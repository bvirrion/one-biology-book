\section*{Chapitre \ref{ch:g3:bones-and-muscles} --- Les os et les muscles}

\begin{solution}{exo:g3:bones-and-muscles:1}
Le \omterm{def:g3:bones-and-muscles:skeleton}{crâne} protège le cerveau~; la \omterm{def:g3:bones-and-muscles:skeleton}{colonne vertébrale} tient le tronc
droit (une colonne souple)~; les \omterm{def:g3:bones-and-muscles:skeleton}{côtes} forment une cage protectrice
autour du cœur et des poumons~; les \omterm{def:g3:bones-and-muscles:skeleton}{os} des \omterm{def:g1:my-body:parts}{membres} portent les \omterm{def:g1:my-body:parts}{bras}
et les \omterm{def:g1:my-body:parts}{jambes}.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:2}
Environ $200$. Le plus long est le fémur.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:3}
Non --- un \omterm{def:g3:bones-and-muscles:skeleton}{os} plié est un \omterm{def:g3:bones-and-muscles:skeleton}{os} cassé. Tout ce qui plie plie aux
\omterm{prop:g3:bones-and-muscles:joints}{articulations}, là où deux \omterm{def:g3:bones-and-muscles:skeleton}{os} se rejoignent.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:4}
C'est une colonne de plus de $20$ petits \omterm{def:g3:bones-and-muscles:skeleton}{os}, chacun pouvant tourner
de quelques degrés sur ses voisins~; les nombreuses petites flexions
s'additionnent en un grand enroulement souple.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:5}
Il raccourcit et s'épaissit, et tire ainsi l'\omterm{def:g3:bones-and-muscles:skeleton}{os} auquel il est
attaché. Aucun \omterm{def:g3:bones-and-muscles:muscle}{muscle} ne peut pousser.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:6}
Parce qu'un \omterm{def:g3:bones-and-muscles:muscle}{muscle} ne fait que tirer, un seul \omterm{def:g3:bones-and-muscles:muscle}{muscle} ne peut bouger
l'\omterm{def:g1:my-body:joint}{articulation} que dans un sens. Flexion~: le \omterm{def:g3:bones-and-muscles:muscle}{muscle} avant se
contracte et tire l'\omterm{def:g1:my-body:parts}{avant-bras} vers le haut pendant que celui de
derrière se relâche. Extension~: le \omterm{def:g3:bones-and-muscles:muscle}{muscle} arrière se contracte
pendant que celui de devant se relâche.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:7}
L'\omterm{def:g3:bones-and-muscles:skeleton}{os} vivant reconstruit lui-même la cassure, lentement~; le plâtre
ne fait que tenir les morceaux immobiles pendant ce travail. Cela
montre que l'\omterm{def:g3:bones-and-muscles:skeleton}{os} est une matière vivante, nourrie par le sang --- pas
morte comme la craie.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:8}
Flexion~: l'avant du \omterm{def:g1:my-body:parts}{bras} gonfle et durcit (ce muscle-là se
contracte). Extension~: l'avant s'assouplit pendant que l'arrière du
\omterm{def:g1:my-body:parts}{bras} se raidit --- la paire échange les rôles.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:9}
Les genoux sont des \omterm{prop:g3:bones-and-muscles:joints}{articulations} faites pour les grands plis, mues
par les \omterm{def:g3:bones-and-muscles:muscle}{muscles} les plus puissants du corps~; la colonne est faite
pour de légères flexions réparties sur beaucoup de petits \omterm{def:g3:bones-and-muscles:skeleton}{os}.
Soulever genoux pliés et dos droit confie le gros du travail aux
\omterm{def:g1:my-body:parts}{jambes} et épargne la colonne.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:10}
Chaque fil, comme chaque \omterm{def:g3:bones-and-muscles:muscle}{muscle}, ne peut que tirer. La réponse du
marionnettiste est celle du biologiste~: deux fils par mouvement ---
l'un pour tirer le \omterm{def:g1:my-body:parts}{bras} vers le haut, l'autre pour le ramener vers
le bas --- qui travaillent à tour de rôle, exactement comme les
paires de \omterm{def:g3:bones-and-muscles:muscle}{muscles} du corps.
\end{solution}
