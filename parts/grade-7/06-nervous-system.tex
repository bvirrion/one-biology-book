\chapter{The Nervous System}\label{ch:g7:nervous}

Your \omterm{def:g1:my-body:body}{body} \omterm{def:g1:senses:sense}{senses} the world and responds in milliseconds. The \omterm{def:g7:nervous:system}{nervous
system} gathers information, processes it and triggers actions --- from
pulling a \omterm{def:g1:my-body:parts}{hand} off a hot pan to solving a maths problem.

\section{Organisation}

\begin{definition}[Nervous system]\label{def:g7:nervous:system}
The \emph{nervous system}\index{nervous system} is the network of \omterm{def:g4:cells:special}{specialised
cells} that detect stimuli, transmit signals and coordinate \omterm{def:g7:nervous:stim}{responses}. In
humans it includes the \emph{central nervous system}\index{central nervous
system} (CNS: \omterm{def:g5:nerves:system}{brain} and \omterm{def:g5:nerves:system}{spinal cord}) and the \emph{peripheral nervous
system}\index{peripheral nervous system} (\omterm{def:g5:nerves:system}{nerves} outside the CNS).
\end{definition}

\begin{definition}[Stimulus and response]\label{def:g7:nervous:stim}
A \emph{stimulus}\index{stimulus} is a detectable change (\omterm{def:g4:plants-need:needs}{light}, sound,
\omterm{def:g1:senses:five}{touch}, chemical, temperature). A \emph{response}\index{response} is the
action that follows.
\end{definition}

\section{Neurons}

\begin{definition}[Neuron]\label{def:g7:nervous:neuron}
A \emph{neuron}\index{neuron} (\omterm{def:g5:nerves:system}{nerve} \omterm{def:g7:microscope:cell}{cell}) carries electrical signals.
Typical parts: \emph{dendrites}\index{dendrite} (receive),
\emph{cell body}\index{cell body (neuron)}, and a long
\emph{axon}\index{axon} (sends). Signals pass to the next \omterm{def:g7:microscope:cell}{cell} at a
\emph{synapse}\index{synapse} using chemical messengers.
\end{definition}

\begin{definition}[Sensory, motor, relay]\label{def:g7:nervous:types}
\emph{Sensory neurons}\index{neuron!sensory} carry signals from receptors
toward the CNS. \emph{Motor neurons}\index{neuron!motor} carry signals
from the CNS to effectors. \emph{Relay (inter)neurons}\index{neuron!relay}
connect within the CNS.
\end{definition}

\section{Reflexes}

\begin{definition}[Reflex]\label{def:g7:nervous:reflex}
A \emph{reflex}\index{reflex} is a fast, automatic \omterm{def:g7:nervous:stim}{response} to a \omterm{def:g7:nervous:stim}{stimulus},
often protective. Many spinal reflexes involve a short pathway called a
\emph{reflex arc}\index{reflex arc}.
\end{definition}

\begin{method}[Describing a reflex arc]\label{met:g7:nervous:arc}
\begin{enumerate}
  \item \omterm{def:g7:nervous:stim}{Stimulus} detected by a receptor.
  \item Sensory \omterm{def:g7:nervous:neuron}{neuron} to \omterm{def:g5:nerves:system}{spinal cord} (or brainstem).
  \item Relay \omterm{def:g7:nervous:neuron}{neuron} (in many arcs).
  \item Motor \omterm{def:g7:nervous:neuron}{neuron} to \omterm{def:g3:bones:muscle}{muscle}/gland.
  \item \omterm{def:g7:nervous:stim}{Response} (for example withdraw \omterm{def:g1:my-body:parts}{hand}).
\end{enumerate}
The \omterm{def:g5:nerves:system}{brain} may be informed, but the quick path can act first.
\end{method}

\begin{example}[Hot pan]\label{ex:g7:nervous:hot}
Heat receptors in \omterm{def:g5:nerves:senses}{skin} $\to$ sensory \omterm{def:g7:nervous:neuron}{neuron} $\to$ \omterm{def:g5:nerves:system}{spinal cord} $\to$ motor
\omterm{def:g7:nervous:neuron}{neuron} $\to$ \omterm{def:g1:my-body:parts}{arm} \omterm{def:g3:bones:muscle}{muscles} pull away --- before you fully ``think'' about
it.
\end{example}
