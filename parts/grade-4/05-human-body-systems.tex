\chapter{Human Body Systems}\label{ch:g4:body-systems}

Your \omterm{def:g1:my-body:body}{body} is not one solid lump. Groups of \omterm{def:g4:body-systems:organs}{organs} work as systems ---
teams with shared jobs such as digesting \omterm{def:g2:food-energy:food}{food}, moving blood, or sensing
the world. This chapter introduces several major systems and how they
cooperate.

\section{Cells, organs and systems}

\begin{definition}[Organ and system]\label{def:g4:body-systems:organs}
An \emph{organ}\index{organ} is a \omterm{def:g1:my-body:body}{body} part made of tissues that work
together on a main job (\omterm{def:g4:body-systems:circ}{heart}, lungs, \omterm{def:g3:teeth:path}{stomach}, brain). An
\emph{organ system}\index{organ system} is a group of organs that
cooperate for a larger task (for example \omterm{def:g3:teeth:digestion}{digestion} or circulation).
\end{definition}

\begin{proposition}[Teamwork]\label{prop:g4:body-systems:team}
\omterm{def:g4:body-systems:organs}{Organ systems} depend on one another. \omterm{def:g3:teeth:digestion}{Digestion} supplies fuel; blood
carries it; lungs supply oxygen; \omterm{def:g3:bones:muscle}{muscles} use both to move.
\end{proposition}
\admitted

\section{Digestive system}

\begin{definition}[Digestion]\label{def:g4:body-systems:digestion}
The \emph{digestive system}\index{digestive system} breaks \omterm{def:g2:food-energy:food}{food} into
smaller pieces and simpler substances the \omterm{def:g1:my-body:body}{body} can absorb and use. Main
\omterm{def:g4:body-systems:organs}{organs} include the \omterm{def:g3:teeth:path}{mouth}, oesophagus, \omterm{def:g3:teeth:path}{stomach}, \omterm{def:g3:teeth:path}{intestines} and helpers
such as the liver.
\end{definition}

\omimg{digestive-system.jpg}{0.52\linewidth}{\omterm{def:g2:food-energy:food}{Food} travels from \omterm{def:g3:teeth:path}{mouth} to
\omterm{def:g3:teeth:path}{stomach} and \omterm{def:g3:teeth:path}{intestines} while it is broken down and useful parts are
absorbed.}

\begin{example}[A sandwich's journey]\label{ex:g4:body-systems:sandwich}
Teeth chew; saliva moistens. The \omterm{def:g3:teeth:path}{stomach} churns \omterm{def:g2:food-energy:food}{food} with acid and
enzymes. The small \omterm{def:g3:teeth:path}{intestine} absorbs \omterm{def:g4:plants-need:needs}{nutrients} into the blood. Leftover
material \omterm{def:g2:plant-parts:main}{leaves} via the large \omterm{def:g3:teeth:path}{intestine}.
\end{example}

\section{Breathing and circulation}

\begin{definition}[Respiratory system]\label{def:g4:body-systems:resp}
The \emph{respiratory system}\index{respiratory system} brings oxygen
into the \omterm{def:g1:my-body:body}{body} and removes carbon dioxide. \omterm{def:g4:plants-need:needs}{Air} flows through the \omterm{def:g1:senses:five}{nose} or
\omterm{def:g3:teeth:path}{mouth} to the lungs, where gases exchange with the blood.
\end{definition}

\begin{definition}[Circulatory system]\label{def:g4:body-systems:circ}
The \emph{circulatory system}\index{circulatory system} moves blood
around the \omterm{def:g1:my-body:body}{body}. The \emph{heart}\index{heart} pumps; blood vessels
carry blood that transports oxygen, \omterm{def:g4:plants-need:needs}{nutrients} and wastes.
\end{definition}

\section{Support, movement and control}

\begin{definition}[Skeleton and muscles]\label{def:g4:body-systems:move}
The \emph{skeletal system}\index{skeletal system} (\omterm{def:g3:bones:bone}{bones}) supports the
\omterm{def:g1:my-body:body}{body} and protects soft \omterm{def:g4:body-systems:organs}{organs}. The \emph{muscular
system}\index{muscular system} pulls on \omterm{def:g3:bones:bone}{bones} (and other structures) to
create movement.
\end{definition}

\begin{definition}[Nervous system]\label{def:g4:body-systems:nervous}
The \emph{nervous system}\index{nervous system} --- brain, spinal cord
and nerves --- \omterm{def:g1:senses:sense}{senses} the surroundings, processes information and
sends signals that control \omterm{def:g3:bones:muscle}{muscles} and many \omterm{def:g4:body-systems:organs}{organs}.
\end{definition}

\begin{example}[Catching a ball]\label{ex:g4:body-systems:ball}
\omterm{def:g1:senses:five}{Eyes} \omterm{def:g1:senses:sense}{sense} the ball (\omterm{def:g4:body-systems:nervous}{nervous system}); the brain plans; \omterm{def:g3:bones:muscle}{muscles} contract
against \omterm{def:g3:bones:bone}{bones} (muscular and \omterm{def:g4:body-systems:move}{skeletal systems}); breathing and blood flow
rise to supply active tissues.
\end{example}

\section{Keeping systems healthy}

\begin{method}[Daily habits that help]\label{met:g4:body-systems:habits}
\begin{enumerate}
  \item Eat a balanced variety of \omterm{def:g2:food-energy:food}{foods} for the \omterm{def:g4:body-systems:digestion}{digestive system} and
        whole \omterm{def:g1:my-body:body}{body}.
  \item Stay active to strengthen \omterm{def:g4:body-systems:circ}{heart}, lungs, \omterm{def:g3:bones:bone}{bones} and \omterm{def:g3:bones:muscle}{muscles}.
  \item Sleep enough for the brain and recovery.
  \item Avoid tobacco smoke and other lung-damaging exposures.
  \item Wash \omterm{def:g1:my-body:parts}{hands} to reduce infection risk.
\end{enumerate}
\end{method}
