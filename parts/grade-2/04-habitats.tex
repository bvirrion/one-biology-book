\chapter{Habitats}\label{ch:g2:habitats}

Every \omterm{def:g1:living-nonliving:living}{living thing} has a home. Not a house with a door --- a \omterm{def:g1:caring:needs}{place}
in nature that gives \omterm{def:g1:caring:needs}{food}, \omterm{def:g1:caring:needs}{water}, \omterm{def:g1:caring:needs}{shelter} and space. That \omterm{def:g1:caring:needs}{place} is
called a \omterm{def:g2:habitats:habitat}{habitat}.

\section{What is a habitat?}

\begin{definition}[Habitat]\label{def:g2:habitats:habitat}
A \emph{habitat}\index{habitat} is the \omterm{def:g1:caring:needs}{place} where an \omterm{def:g2:what-alive:organism}{organism}
lives. A good habitat meets its \omterm{def:g1:caring:needs}{needs}: \omterm{def:g1:caring:needs}{food}, \omterm{def:g1:caring:needs}{water}, \omterm{def:g1:caring:needs}{shelter} and
room to \omterm{def:g2:what-alive:signs}{grow} or move.
\end{definition}

\omimg{habitats.jpg}{0.52\linewidth}{\omterm{def:g2:habitats:kinds}{Forest}, \omterm{def:g2:habitats:kinds}{pond}, \omterm{def:g2:habitats:kinds}{desert} and shore
--- different homes for different lives.}

\begin{example}[Local habitats]\label{ex:g2:habitats:local}
A worm's \omterm{def:g2:habitats:habitat}{habitat} may be the soil in a garden. A frog's \omterm{def:g2:habitats:habitat}{habitat} may
be a \omterm{def:g2:habitats:kinds}{pond} edge. A sparrow's \omterm{def:g2:habitats:habitat}{habitat} may be trees and rooftops in a
town. Your \omterm{def:g2:habitats:habitat}{habitat} includes your home and neighbourhood.
\end{example}

\section{Kinds of habitats}

\begin{definition}[Some habitats]\label{def:g2:habitats:kinds}
\begin{itemize}
  \item \emph{forest}\index{forest}: many trees; shade; homes for
        \omterm{def:g2:animal-groups:five}{birds}, deer, insects;
  \item \emph{pond}\index{pond} or lake: fresh \omterm{def:g1:caring:needs}{water}; \omterm{def:g2:animal-groups:five}{fish}, frogs,
        ducks, \omterm{def:g1:caring:needs}{water} \omterm{def:g1:plants-animals:plant}{plants};
  \item \emph{desert}\index{desert}: very dry; \omterm{def:g1:plants-animals:plant}{plants} and \omterm{def:g1:plants-animals:animal}{animals}
        that save \omterm{def:g1:caring:needs}{water};
  \item \emph{ocean}\index{ocean}: salt \omterm{def:g1:caring:needs}{water}; \omterm{def:g2:animal-groups:five}{fish}, whales, sea
        weeds along coasts;
  \item \emph{grassland}\index{grassland}: open grass; grazing
        \omterm{def:g1:plants-animals:animal}{animals} and their hunters.
\end{itemize}
\end{definition}

\begin{method}[Match animal to habitat]\label{met:g2:habitats:match}
\begin{enumerate}
  \item What does the \omterm{def:g1:plants-animals:animal}{animal} eat?
  \item Does it need lots of \omterm{def:g1:caring:needs}{water} or very little?
  \item Where can it hide or build a nest?
  \item Choose the \omterm{def:g2:habitats:habitat}{habitat} that fits those answers.
\end{enumerate}
\end{method}

\begin{example}[Fit]\label{ex:g2:habitats:fit}
A \omterm{def:g2:animal-groups:five}{fish} \omterm{def:g1:caring:needs}{needs} \omterm{def:g1:caring:needs}{water} all around it $\to$ \omterm{def:g2:habitats:kinds}{pond} or \omterm{def:g2:habitats:kinds}{ocean}, not \omterm{def:g2:habitats:kinds}{desert}. A
camel saves \omterm{def:g1:caring:needs}{water} and handles heat $\to$ \omterm{def:g2:habitats:kinds}{desert} can work. A woodpecker
\omterm{def:g1:caring:needs}{needs} trees $\to$ \omterm{def:g2:habitats:kinds}{forest}.
\end{example}
