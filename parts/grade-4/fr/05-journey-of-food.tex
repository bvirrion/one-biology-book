\chapter{Le voyage des aliments}\label{ch:g4:journey-of-food}

Une bouchée de pain~: mâchée, avalée --- disparue. Disparue où~? Ce
soir, une partie de ce pain aidera tes \omterm{def:g1:my-body:parts}{jambes} à courir et tes \omterm{def:g3:bones-and-muscles:skeleton}{os} à
grandir, ce qui veut dire qu'elle doit d'une façon ou d'une autre
\emph{sortir} du tube où on l'a avalée pour \emph{entrer} dans le
reste du corps. Suis la bouchée~: son voyage est l'une des plus
belles histoires du corps.

\section{Le tube digestif}

\begin{definition}[Le tube digestif]\label{def:g4:journey-of-food:tract}
Les aliments traversent le \emph{tube digestif}\index{tube
digestif}, un long tuyau unique qui traverse le corps de part en
part~:
\begin{enumerate}
  \item la \emph{bouche}\index{bouche}, où les dents broient
        (\cref{ch:g2:teeth-and-eating-well}) et où la
        \emph{salive}\index{salive} humecte~;
  \item l'\emph{œsophage}\index{œsophage}, le toboggan qui va de la
        gorge à l'estomac~;
  \item l'\emph{estomac}\index{estomac}, une poche musclée où les
        aliments sont brassés et mélangés à des \omterm{def:g4:journey-of-food:digestion}{sucs digestifs}
        pendant des heures~;
  \item l'\emph{intestin grêle}\index{intestin grêle}, un tuyau
        replié de plusieurs mètres, où la \omterm{def:g4:journey-of-food:digestion}{digestion} s'achève et où
        ce que les aliments ont de bon passe dans le corps~;
  \item le \emph{gros intestin}\index{gros intestin}, où l'eau est
        récupérée sur ce qui reste, avant que les restes ne
        quittent le corps.
\end{enumerate}
\end{definition}

\begin{example}[Chronométrer le voyage]\label{ex:g4:journey-of-food:timing}
La bouchée de pain passe quelques secondes dans la bouche, deux
secondes à glisser dans l'\omterm{def:g4:journey-of-food:tract}{œsophage}, quelques heures à être brassée
dans l'\omterm{def:g4:journey-of-food:tract}{estomac}, plusieurs heures encore à cheminer dans l'\omterm{def:g4:journey-of-food:tract}{intestin
grêle}, et jusqu'à un jour ou plus à finir dans le \omterm{def:g4:journey-of-food:tract}{gros intestin}. Le
voyage entier prend couramment un à deux jours.
\end{example}

\section{La digestion~: des aliments aux nutriments}

\begin{definition}[Digestion et nutriments]\label{def:g4:journey-of-food:digestion}
La \emph{digestion}\index{digestion} est la transformation des
aliments, tout au long du tube, en éléments assez petits et assez
simples pour entrer dans le corps~: les
\emph{nutriments}\index{nutriment}. Les dents et le brassage de
l'\omterm{def:g4:journey-of-food:tract}{estomac} cassent les aliments~; les \emph{sucs
digestifs}\index{suc digestif} --- dont la \omterm{def:g4:journey-of-food:tract}{salive} --- les démontent
plus loin encore, morceau par morceau, jusqu'à ce que ce qui était
du pain soit un mélange liquide de nutriments.
\end{definition}

\begin{example}[Le pain devient sucré]\label{ex:g4:journey-of-food:sweet}
Mâche très longtemps un morceau de pain nature --- le temps de
compter lentement jusqu'à soixante --- et une légère saveur sucrée
apparaît. On n'a rien ajouté~: la \omterm{def:g4:journey-of-food:tract}{salive} démonte déjà le pain en
morceaux plus simples et plus sucrés. Tu viens de goûter la
\omterm{def:g4:journey-of-food:digestion}{digestion} qui commence.
\end{example}

\begin{proposition}[Passer dans le corps]\label{prop:g4:journey-of-food:absorb}
L'intérieur du tube est encore, en un sens, à l'extérieur du corps
--- comme le trou d'un beignet. Le passage se fait dans
l'\emph{\omterm{def:g4:journey-of-food:tract}{intestin grêle}}~: à travers ses parois minces et profondément
repliées, les \omterm{def:g4:journey-of-food:digestion}{nutriments} passent dans le \emph{sang}\index{sang},
qui les porte à toutes les parties du corps (le service de livraison
qu'explore \cref{ch:g4:heart-and-blood}). Ce qui ne peut pas passer
--- fibres, enveloppes --- poursuit sa route, perd son eau dans le
\omterm{def:g4:journey-of-food:tract}{gros intestin}, et sort sous forme d'excréments.
\end{proposition}
\admitted

\begin{remark}[Pourquoi les replis et les mètres]\label{rem:g4:journey-of-food:surface}
Le passage demande de la surface de paroi, et l'\omterm{def:g4:journey-of-food:tract}{intestin grêle} est
bâti pour cela~: des mètres de longueur, et des parois repliées et
repliées encore à l'intérieur. Étalée à plat, la surface de passage
tapisserait une petite pièce --- rangée dans ton ventre. Dans le
corps, les grands travaux reçoivent de grandes surfaces pliées
petit.
\end{remark}

\section{Aider la digestion à faire son travail}

\begin{method}[Des habitudes amies de la digestion]\label{met:g4:journey-of-food:habits}
\begin{enumerate}
  \item Mâche comme il faut~: l'\omterm{def:g4:journey-of-food:tract}{estomac} n'a pas de dents --- ce que
        la bouche saute, l'\omterm{def:g4:journey-of-food:tract}{estomac} doit le subir~;
  \item mange calmement, à des heures régulières~;
  \item bois de l'eau aux repas~;
  \item mange des aliments riches en fibres --- légumes, \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits},
        pain complet~: les fibres font avancer les restes~;
  \item garde les jeux violents et la baignade pour après que le
        repas s'est tassé~: un \omterm{def:g4:journey-of-food:tract}{estomac} au travail et des \omterm{def:g1:my-body:parts}{jambes} au
        travail se font concurrence.
\end{enumerate}
\end{method}

\begin{example}[L'enquête du mal de ventre]\label{ex:g4:journey-of-food:ache}
Un déjeuner englouti, puis un sprint~: le classique mal de ventre de
l'après-midi. Le repas a été avalé à moitié mâché (règle 1 non
respectée) et l'\omterm{def:g4:journey-of-food:tract}{estomac} en pleine \omterm{def:g4:journey-of-food:digestion}{digestion} a été entraîné dans une
course (règle 5). Le remède n'est pas un médicament mais de la
politesse --- envers son propre \omterm{def:g4:journey-of-food:tract}{tube digestif}.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:1}
Nomme les parties du \omterm{def:g4:journey-of-food:tract}{tube digestif} dans l'ordre, de la bouche à la
sortie.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:2}
Qu'arrive-t-il aux aliments dans l'\omterm{def:g4:journey-of-food:tract}{estomac}, et combien de temps y
restent-ils à peu près~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:3}
Qu'est-ce que la \omterm{def:g4:journey-of-food:digestion}{digestion}~? Que sont les \omterm{def:g4:journey-of-food:digestion}{nutriments}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:4}
Dans quelle partie les \omterm{def:g4:journey-of-food:digestion}{nutriments} passent-ils dans le \omterm{prop:g4:journey-of-food:absorb}{sang}~?
Qu'arrive-t-il à ce qui ne peut pas passer~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:5}
Pourquoi le pain longuement mâché devient-il légèrement sucré~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:6}
Pourquoi l'\omterm{def:g4:journey-of-food:tract}{intestin grêle} est-il si long et si replié à l'intérieur~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:7}
Donne trois habitudes amies de la \omterm{def:g4:journey-of-food:digestion}{digestion} et la raison de chacune.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:8}
À toi d'essayer (à table)~: chronomètre ton prochain repas. Combien
de temps tes aliments passent-ils dans ta bouche --- et comment cela
se compare-t-il aux jours qui les attendent~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:9}
Diagnostique le mal de ventre de
\cref{ex:g4:journey-of-food:ache}~: quelles deux habitudes ont été
enfreintes, et que changerais-tu demain~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:10}
«~Les aliments entrent dans le corps dès qu'on les avale.~» Corrige
avec \cref{prop:g4:journey-of-food:absorb}~: où les aliments
entrent-ils vraiment dans le corps, et que doit-il leur arriver
avant~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:journey-of-food:11}
L'herbe de la vache est un aliment bien plus coriace que ton pain
--- et la vache a quatre compartiments d'\omterm{def:g4:journey-of-food:tract}{estomac}, mâche sa
nourriture deux fois et héberge dans son ventre des \omterm{def:g1:living-or-not:living}{êtres vivants}
qui l'aident à démonter l'herbe. À l'aide de
\cref{def:g4:journey-of-food:digestion} et de
\cref{prop:g2:what-animals-eat:match}, explique pourquoi un mangeur
d'herbe a besoin de cette machinerie plus lourde.
\end{exercise}
