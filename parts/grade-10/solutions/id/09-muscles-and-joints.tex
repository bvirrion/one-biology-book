\section*{Bab \ref{ch:g10:muscles-and-joints} --- Otot dan Sendi}

\begin{solution}{exo:g10:muscles-and-joints:1}
\omterm{def:g10:muscles-and-joints:joint}{Tulang rawan} pada ujung tulangnya (meluncur licin, menyebarkan beban);
kapsul (menutup \omterm{def:g10:muscles-and-joints:joint}{sendinya}) beserta cairan sinovialnya (pelumasan);
\omterm{def:g10:muscles-and-joints:joint}{ligamen} (menahan tulangnya tetap bersatu, membatasi gerakannya); pada
lutut, meniskus (menyebarkan bebannya).
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:2}
\omterm{def:g10:muscles-and-joints:muscle}{Tendon} menyambungkan otot ke tulang dan meneruskan tarikannya; \omterm{def:g10:muscles-and-joints:joint}{ligamen}
menyambungkan tulang ke tulang melintasi \omterm{def:g10:muscles-and-joints:joint}{sendi} dan membatasi
gerakannya. Tidak satu pun dari keduanya dapat berkerut.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:3}
Otot hanya dapat menarik dengan memendek; ia tidak dapat mendorong.
Untuk menggerakkan \omterm{def:g10:muscles-and-joints:joint}{sendi} ke dua arah, diperlukan sebuah fleksor dan
sebuah ekstensor, yang masing-masing membatalkan apa yang dikerjakan
yang lain.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:4}
Keseleo: \omterm{def:g10:muscles-and-joints:joint}{ligamen} yang terentang atau robek. Robekan otot: \omterm{def:g10:muscles-and-joints:muscle}{serat otot}
yang putus. Lepas \omterm{def:g10:muscles-and-joints:joint}{sendi}: ujung tulang yang terpaksa keluar dari
\omterm{def:g10:muscles-and-joints:joint}{sendinya}, dengan kapsul dan \omterm{def:g10:muscles-and-joints:joint}{ligamennya} terentang atau robek.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:5}
Serat lambat: lambat, lemah, tanpa lelah, kaya \omterm{def:g10:cells-common-unit:organelle}{mitokondria} dan kapiler,
merah, berjalan dengan respirasi. Serat cepat: cepat, kuat, lelah dalam
satu menit, sedikit \omterm{def:g10:cells-common-unit:organelle}{mitokondria}, pucat, berjalan dengan simpanan fosfat
dan \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:6}
Bebannya $2 \times 9.8 = \qty{19.6}{N}$; gaya bisepnya $19.6 \times 32/4
\approx \qty{157}{N}$.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:7}
Dorongan dapat datang dari arah mana pun, dan tiap otot hanya dapat
menahan satu arah. Keduanya yang berkerut mengakukan \omterm{def:g10:muscles-and-joints:joint}{sendinya} ke dua
arah sekaligus; ke arah mana pun dorongannya datang, salah satu dari
keduanya sudah menarik melawannya.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:8}
Seratnya memendek \qty{0.6}{cm}; \omterm{def:g10:muscles-and-joints:muscle}{tendonnya} menggerakkan tulangnya
sejauh \qty{0.6}{cm} pada \qty{5}{cm} dari sikunya, sehingga tangannya
bergerak $0.6 \times 35/5 = \qty{4.2}{cm}$.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:9}
Keleluasaan gerak menuntut mangkuk yang dangkal dan \omterm{def:g10:muscles-and-joints:joint}{ligamen} yang
longgar; kelonggaran yang sama membiarkan kepala humerusnya
meninggalkan mangkuknya ketika sebuah gaya mendorongnya melampaui
jangkauan yang dapat ditahan \omterm{def:g10:muscles-and-joints:joint}{ligamennya}.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:10}
Kekurangan cairan dan hilangnya garam lewat keringat, serta kelelahan
seratnya pada akhir lombanya. Penawarnya: regangkan ototnya dengan
lembut dan minumlah air bergaram.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:11}
Paha pelari cepat itu 70\% berserat cepat dan perbandingannya sebagian
besar diwariskan; latihan membesarkan serat dan memperbaiki
perlengkapannya tetapi tidak mengubah jumlahnya. Orang yang tak
terlatih berangkat dari separuh-separuh dan punya lebih banyak serat
lambat untuk dikembangkan.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:12}
$4 \times 70 \times 9.8 \approx \qty{2.7}{kN}$. Tegangannya $2744/100
\approx \qty{27}{N/mm^2}$: sekitar seperempat tegangan putusnya, yaitu
faktor keamanan empat.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:13}
Otot kaya pasokan darah dan memuat \omterm{def:g10:cells-common-unit:cell}{sel} cadangan yang membangun kembali
seratnya: hitungan pekan. \omterm{def:g10:muscles-and-joints:joint}{Ligamen} punya sedikit pembuluh, sehingga zat
gizi dan \omterm{def:g10:cells-common-unit:cell}{sel} pemulihnya tiba dengan lambat: hitungan bulan. \omterm{def:g10:muscles-and-joints:joint}{Tulang
rawan} sama sekali tidak punya pembuluh darah dan hampir tidak punya \omterm{def:g10:cells-common-unit:cell}{sel}
yang membelah; ia hanya diberi makan cairan sinovial dan nyaris tidak
memperbaiki diri.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:14}
Binaraga menebalkan seratnya (lebih banyak miofibril dalam
masing-masing) dan peregangan memanjangkan satuan otot--tendon serta
ketahanan kapsulnya. Jumlah seratnya sudah tetap sejak awal kehidupan;
otot orang dewasa tumbuh atau menyusut dengan mengubah ukuran serat
yang sudah ada.
\end{solution}

\begin{solution}{exo:g10:muscles-and-joints:15}
Di tangga \omterm{def:g10:muscles-and-joints:joint}{sendinya} membawa beberapa kali berat tubuhnya; \omterm{def:g10:muscles-and-joints:joint}{tulang rawan}
yang lebih tipis meneruskan tekanan itu ke tulang di bawahnya, yang
punya saraf dan terasa nyeri. Dengan \omterm{def:g10:muscles-and-joints:joint}{tulang rawan} yang lebih sedikit,
meniskusnya menjadi hal terakhir yang menyebarkan bebannya. Otot yang
kuat di sekeliling lututnya menyerap sebagian kejutnya dan memantapkan
\omterm{def:g10:muscles-and-joints:joint}{sendinya}, sehingga puncak tekanan pada \omterm{def:g10:muscles-and-joints:joint}{tulang rawan} yang tersisa berkurang.
\end{solution}

\begin{solution}{pb:g10:muscles-and-joints:1}
\textbf{1.} $70 \times 9.8 = \qty{686}{N}$.

\textbf{2.} $686 \times 10/5 \approx \qty{1.4}{kN}$.

\textbf{3.} $686 \times 20/5 \approx \qty{2.7}{kN}$.

\textbf{4.} Dua dan empat kali berat tubuhnya: pada tiap anak tangga
kuadrisepnya menarik dengan berat empat orang, dan itulah sebabnya paha
terbakar pada tanjakan yang panjang.

\textbf{5.} Otot paha belakangnya, di belakang pahanya; otot itu
mengendur dan terentang ketika kuadrisepnya meluruskan lututnya, sambil
berkerut ringan untuk memantapkan \omterm{def:g10:muscles-and-joints:joint}{sendinya}.

\textbf{6.} Gaya tanahnya $3 \times 686 \approx \qty{2.06}{kN}$; gaya
\omterm{def:g10:muscles-and-joints:muscle}{tendonnya} $2058 \times 20/5 \approx \qty{8.2}{kN}$, dua belas kali
berat tubuhnya.

\textbf{7.} $8232/120 \approx \qty{69}{N/mm^2}$: sekitar dua pertiga
tegangan putusnya. Tiap pendaratan memakai sebagian besar selisih aman
\omterm{def:g10:muscles-and-joints:muscle}{tendonnya}.

\textbf{8.} Bebannya $8.2 + 2.1 \approx \qty{10.3}{kN}$. Dengan
meniskusnya: $10300/600 \approx \qty{17}{N/mm^2}$; tanpanya: $10300/100 \approx
\qty{103}{N/mm^2}$, enam kali lebih besar.

\textbf{9.} Tekanan enam kali lipat pada \omterm{def:g10:muscles-and-joints:joint}{tulang rawan} yang sama pada
tiap pendaratan: permukaannya tergerus lebih cepat daripada yang dapat
dirawatnya, dan ia pun aus.

\textbf{10.} Selaput cairan di antara \omterm{def:g10:muscles-and-joints:joint}{tulang rawannya}, sehingga
keduanya meluncur selama pembengkokannya tanpa gesekan kering, serta
sedikit peredaman tumbukannya.

\textbf{11.} Puntiran tibia di bawah femurnya, disertai pembengkokan
menyamping: perputaran yang bukan rancangan lututnya, dan ligamennyalah
yang ada untuk menahannya.

\textbf{12.} Kuadrisepnya hanya menarik sepanjang kakinya, yang
meluruskan lututnya; ia tidak punya tindakan melawan puntiran. Lagi
pula cederanya terjadi dalam beberapa perseratus sekon, lebih cepat
daripada tanggapan otot mana pun.

\textbf{13.} Darah dari pembuluh \omterm{def:g10:muscles-and-joints:joint}{ligamen} dan kapsulnya yang robek,
ditambah cairan sinovial yang berlebih: kapsulnya jebol atau meradang
dan \omterm{def:g10:muscles-and-joints:joint}{sendinya} pun terisi.

\textbf{14.} \omterm{def:g10:muscles-and-joints:joint}{Ligamennya}, yang buruk pasokan darahnya, menuntut
berbulan-bulan dan sering pembedahan; meniskusnya yang robek, yaitu
\omterm{def:g10:muscles-and-joints:joint}{tulang rawan} yang nyaris tanpa pembuluh, lazimnya tidak sembuh dan
bagian yang robek dipangkas atau dijahit.

\textbf{15.} Otot paha belakangnya adalah lawan kuadrisepnya dan
menarik tibianya ke belakang; sepasang otot yang kuat dan berkerut
bersama mengakukan lututnya melawan luncuran ke depan yang tidak lagi
dicegah \omterm{def:g10:muscles-and-joints:joint}{ligamennya} yang robek, sehingga \omterm{def:g10:muscles-and-joints:joint}{sendinya} terlindungi sebelum
dan sesudah pembedahan.

\textbf{16.} Sekitar 4000 kali sehari pada \qty{1.4}{kN} atau lebih.

\textbf{17.} Sekitar $\num{1.5e6}$ pembebanan tiap tahun. Tendinitis
adalah kerusakan yang menumpuk lebih cepat daripada perbaikan jaringan
\omterm{def:g10:muscles-and-joints:muscle}{tendon} yang lambat; hanya istirahat yang membiarkan perbaikannya
menyusul, karena kekuatan tidak mengurangi beban tiap langkahnya.

\textbf{18.} Pendaratannya $40 \times 8.2 \approx \qty{330}{kN}$
seluruhnya; langkahnya $4000 \times 1.4 \approx \qty{5600}{kN}$ ---
langkahnya membawa beban seluruhnya tujuh belas kali lebih besar,
tetapi pendaratannyalah yang membawa puncak yang mengancam putus.

\textbf{19.} Melipatduakan gaya tanahnya melipatduakan gaya \omterm{def:g10:muscles-and-joints:muscle}{tendon} dan
\omterm{def:g10:muscles-and-joints:joint}{sendinya} pada tiap langkahnya; bersepeda dan berenang melatih ototnya
sambil membebani lututnya jauh di bawah berat tubuhnya.

\textbf{20.} \omterm{def:g10:muscles-and-joints:muscle}{Tendon} kuadrisepnya membawa empat kali berat tubuh di
tangga dan sekitar dua belas kali saat mendarat; \omterm{def:g10:muscles-and-joints:joint}{ligamennya}, yang tidak
pernah diuji beban itu, adalah jaringan yang ditemukan puntirannya.
\end{solution}
