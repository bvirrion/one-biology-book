\begin{solution}{exo:g11:gene-expression:1}
\omterm{def:g11:gene-expression:transcription}{RNA} is single-stranded, its sugar is ribose, and it uses uracil in
place of thymine (it is also short-lived and leaves the \omterm{def:g10:cells-common-unit:organelle}{nucleus}).
\end{solution}

\end{solution}

\begin{solution}{exo:g11:gene-expression:3}

\end{solution}

\begin{solution}{exo:g11:gene-expression:4}
The \omterm{def:g10:cells-common-unit:organelle}{ribosome} reads the mRNA \omterm{def:g11:gene-expression:code}{codon} by \omterm{def:g11:gene-expression:code}{codon} and joins the \omterm{def:g10:chemistry-of-life:families}{amino acids}
into a chain; each transfer \omterm{def:g11:gene-expression:transcription}{RNA} matches one \omterm{def:g11:gene-expression:code}{codon} with its anticodon
and brings the corresponding \omterm{def:g10:chemistry-of-life:families}{amino acid}.
\end{solution}

\begin{solution}{exo:g11:gene-expression:5}
With 4 letters, words of two give $4^2 = 16$ combinations, fewer than
20 \omterm{def:g10:chemistry-of-life:families}{amino acids}; words of three give 64, enough.
\end{solution}

\begin{solution}{exo:g11:gene-expression:6}
$412 \times 3 + 3 = 1239$ \omterm{def:g10:universal-dna:nucleotide}{nucleotides}; a \omterm{def:g10:universal-dna:gene}{gene} of at least 1239 base
pairs (more in a eukaryote, with \omterm{prop:g11:gene-expression:splicing}{introns}).
\end{solution}

\begin{solution}{exo:g11:gene-expression:7}
Original: Met--Lys--Gly--Trp. Mutant: UGA is a stop, so Met--Lys--Gly:
a nonsense \omterm{def:g11:mutations:mutation}{mutation}, truncating the \omterm{def:g11:gene-expression:protein}{protein}.
\end{solution}

\begin{solution}{exo:g11:gene-expression:8}
\texttt{AAG} is still Lys: silent, Met--Lys--Gly--Trp unchanged. The
second is an insertion of G shifting the frame: \texttt{AUG AAA GGG
CUG GUA A}\dots{} Met--Lys--Gly--Leu--Val\dots, a frameshift with no
stop in the fragment.
\end{solution}

\begin{solution}{exo:g11:gene-expression:9}
In most blocks of the table the four \omterm{def:g11:gene-expression:code}{codons} sharing the first two
letters specify the same \omterm{def:g10:chemistry-of-life:families}{amino acid} (Pro, Thr, Ala, Gly, Val, Ser,
Leu\dots): changing the third letter then changes nothing. The
degeneracy of the code sits mainly in the third position.
\end{solution}

\begin{solution}{exo:g11:gene-expression:10}
Read in threes, \texttt{UCU CUC UCU CUC}\dots{} alternates two \omterm{def:g11:gene-expression:code}{codons},
UCU and CUC, so the \omterm{def:g11:gene-expression:protein}{protein} alternates two \omterm{def:g10:chemistry-of-life:families}{amino acids} --- as
observed. With a two-letter code the same repeat would give a single
\omterm{def:g11:gene-expression:code}{codon} and a single \omterm{def:g10:chemistry-of-life:families}{amino acid}. Since poly-U gives Phe and the table's
UCU is Ser and CUC Leu, the experiment assigns UCU to Ser and CUC to
Leu (or the reverse, settled by other experiments).
\end{solution}

\begin{solution}{exo:g11:gene-expression:11}
Only 1503 base pairs are needed for the coding sequence; the rest are
\omterm{prop:g11:gene-expression:splicing}{introns}, transcribed and then cut out of the pre-mRNA, plus
regulatory sequences at the \omterm{def:g10:universal-dna:gene}{gene}'s ends.
\end{solution}

\begin{solution}{exo:g11:gene-expression:12}
Blocking translation kills the bacterium while leaving the patient's
\omterm{def:g10:cells-common-unit:cell}{cells} working. The drug's selectivity implies that bacterial and human
\omterm{def:g10:cells-common-unit:organelle}{ribosomes}, though they do the same job with the same code, differ in
structure enough for a molecule to bind one and not the other --- a
difference accumulated since their separation.
\end{solution}

\begin{solution}{exo:g11:gene-expression:13}
The mouse's \omterm{def:g11:gene-expression:transcription}{RNA} polymerase transcribed the jellyfish \omterm{def:g10:universal-dna:gene}{gene}, and its
\omterm{def:g10:cells-common-unit:organelle}{ribosomes} and tRNAs translated the mRNA with the same \omterm{def:g11:gene-expression:code}{codon} table, so
the same \omterm{def:g10:chemistry-of-life:families}{amino acid} sequence, hence the same fluorescent \omterm{def:g11:gene-expression:protein}{protein}, was
made. With a different code the \omterm{def:g11:gene-expression:code}{codons} would be read as other \omterm{def:g10:chemistry-of-life:families}{amino
acids} and the \omterm{def:g11:gene-expression:protein}{protein} would be a meaningless, non-fluorescent chain.
\end{solution}

\begin{solution}{exo:g11:gene-expression:14}
UAG would no longer stop translation: every \omterm{def:g11:gene-expression:protein}{protein} whose \omterm{def:g10:universal-dna:gene}{gene} ends in
UAG would be extended past its normal end, often losing function ---
a widespread harm. But a \omterm{def:g10:universal-dna:gene}{gene} truncated by a premature UAG would now
be read through, restoring a nearly full-length \omterm{def:g11:gene-expression:protein}{protein}: the second
\omterm{def:g11:mutations:mutation}{mutation} suppresses the first.
\end{solution}

\begin{solution}{exo:g11:gene-expression:15}
A frameshift scrambles everything downstream and usually meets a stop
within a few \omterm{def:g11:gene-expression:code}{codons}: near the start, essentially no correct \omterm{def:g11:gene-expression:protein}{protein} is
made; near the end, most of the \omterm{def:g11:gene-expression:protein}{protein} is intact and may still fold
and work. A missense changes one \omterm{def:g10:chemistry-of-life:families}{amino acid}: harmless if that position
tolerates the change, fatal if it is at the active site or breaks the
fold --- the effect depends on which \omterm{def:g10:chemistry-of-life:families}{amino acid}, where.
\end{solution}

\begin{solution}{pb:g11:gene-expression:1}
\textbf{1.} \texttt{3'-TAC CGA ACC TTT GGG CTT ATG CCA GTA AAG ACT-5'}.

\textbf{2.} \texttt{AUG GCU UGG AAA CCC GAA UAC GGU CAU UUC UGA}.

\textbf{3.} Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--His--Phe, stop.

\textbf{4.} 10 \omterm{def:g10:chemistry-of-life:families}{amino acids}; 33 \omterm{def:g10:universal-dna:nucleotide}{nucleotides} (11 \omterm{def:g11:gene-expression:code}{codons} including the
stop).

\textbf{5.} AUG; the position of that first AUG fixes the frame, and
every following \omterm{def:g11:gene-expression:code}{codon} is read in step from it.

\textbf{6.} CCG is still Pro: silent, \omterm{def:g11:gene-expression:protein}{protein} unchanged.

\textbf{7.} CGU is Arg in place of His: missense,
Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--Arg--Phe.

\textbf{8.} UGA is stop: Met--Ala, nonsense. A single substitution G
to A in the third \omterm{def:g11:gene-expression:code}{codon} (TGG to TGA).

\textbf{9.} mRNA \texttt{AUG GCU UUG GAA ACC CGA AUA CGG UCA UUU CUG
A}: Met--Ala--Leu--Glu--Thr--Arg--Ile--Arg--Ser--Phe--Leu\dots, no stop
in the fragment: frameshift, every \omterm{def:g10:chemistry-of-life:families}{amino acid} after the second is
changed.

\textbf{10.} Mutant 1 (silent, no change) $<$ mutant 2 (one \omterm{def:g10:chemistry-of-life:families}{amino acid}
changed, function possibly kept) $<$ mutant 4 (frameshift, all but two
\omterm{def:g10:chemistry-of-life:families}{amino acids} wrong) $\approx$ mutant 3 (nonsense, a two-amino-acid
fragment: the \omterm{def:g11:gene-expression:protein}{protein} is gone).

\textbf{11.} Leu 6, Ser 6, Trp 1. Trp, with a single \omterm{def:g11:gene-expression:code}{codon}, is changed
by any substitution; Leu or Ser, with six, is most often unchanged.

\textbf{12.} CUU, CUC, CUA, CUG are all Leu: every one of the 3
possible substitutions at the third position is silent --- 100\%.

\textbf{13.} Two letters give 16 \omterm{def:g11:gene-expression:code}{codons}, too few for 20 \omterm{def:g10:chemistry-of-life:families}{amino acids}
plus a stop; three give 64, already more than enough, so four (256)
would be waste.

\textbf{14.} UUU $=$ Phe and AAA $=$ Lys.

\textbf{15.} A \omterm{def:g10:universal-dna:gene}{gene} transferred between \omterm{def:g10:biodiversity-scales:species}{species} is read identically,
so \omterm{prop:g10:universal-dna:universal}{transgenesis} works; and a shared code, inherited unchanged, is
evidence that all organisms descend from a common ancestor that
already used it.

\textbf{16.} 10 \omterm{def:g10:chemistry-of-life:families}{amino acids}: \qty{1}{s}. 600 \omterm{def:g10:chemistry-of-life:families}{amino acids}: \qty{60}{s}.

\textbf{17.} $1800/80 \approx 22$ \omterm{def:g10:cells-common-unit:organelle}{ribosomes} at once.

\textbf{18.} $7200/8 = 900$ copies.

\textbf{19.} One mRNA gives $600/8 = 75$ copies in 10 minutes; about
27 mRNAs are needed.

\textbf{20.} The \omterm{def:g11:gene-expression:protein}{protein}
Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--His--Phe; mutant 3, the single
G-to-A substitution creating a stop at the third \omterm{def:g11:gene-expression:code}{codon}, abolished it;
one \omterm{def:g10:cells-common-unit:organelle}{ribosome} builds the ten \omterm{def:g10:chemistry-of-life:families}{amino acids} in about one second.
\end{solution}
