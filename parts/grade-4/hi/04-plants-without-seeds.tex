\chapter{बिना बीज वाले पौधे}\label{ch:g4:plants-without-seeds}

पुरानी दीवार की छाया वाली तरफ़ \omterm{prop:g4:plants-without-seeds:damp}{मॉस} की गद्दी बिछी है; बीच के \omterm{def:g1:plants-around-us:tree}{पेड़ों} के
नीचे \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} अपने बड़े हरे पत्ते खोलते हैं। पूरी गर्मी इन्हें छानते रहो:
तुम्हें कोई \omterm{def:g3:flowers-fruits-seeds:parts}{फूल}, कोई \omterm{prop:g3:flowers-fruits-seeds:transform}{फल}, कोई \omterm{def:g2:from-seed-to-plant:seed}{बीज} नहीं मिलेगा --- कभी नहीं। फिर भी \omterm{prop:g4:plants-without-seeds:damp}{मॉस}
और \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} बहुत अच्छी तरह फैलते हैं। पता चलता है कि \omterm{def:g1:plants-around-us:plant}{पौधों} के पास अगली
पीढ़ी बनाने के एक से ज़्यादा तरीक़े हैं।

\section{फ़र्न और मॉस: बीजाणुओं से जनन}

\begin{definition}[बीजाणु]\label{def:g4:plants-without-seeds:spore}
\emph{बीजाणु}\index{बीजाणु} एक सूक्ष्म कण है, जो बहुत बड़ी संख्या में
बनता है और जिससे नया \omterm{def:g1:plants-around-us:plant}{पौधा} उग सकता है। \omterm{def:g2:from-seed-to-plant:seed}{बीज}
(\cref{def:g2:from-seed-to-plant:seed}) के उलट, बीजाणु एक अकेला कण है
जिसके \emph{भीतर न कोई \omterm{def:g3:animals-through-seasons:active}{भोजन-भंडार} होता है, न पहले से बना नन्हा \omterm{def:g1:plants-around-us:plant}{पौधा}}
--- वह शुरुआत का सबसे नंगा रूप है।
\end{definition}

\begin{example}[फ़र्न के पत्तों के नीचे]\label{ex:g4:plants-without-seeds:fern}
गर्मियों के आख़िर में \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} का पत्ता उलटकर देखो: उसकी निचली सतह पर भूरे
बिंदुओं की सुथरी क़तारें बनी होती हैं। हर बिंदु हज़ारों \omterm{def:g4:plants-without-seeds:spore}{बीजाणुओं} की एक
पुड़िया है; पकने पर पुड़ियाँ फटती हैं, और \omterm{def:g4:plants-without-seeds:spore}{बीजाणु} सबसे महीन धूल की तरह
हवा में बह जाते हैं। नम, छायादार ज़मीन पर गिरा \omterm{def:g4:plants-without-seeds:spore}{बीजाणु} --- कई चरणों से
गुज़रकर --- नया \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} बन सकता है।
\end{example}

\begin{omfigure}
\includegraphics[width=0.56\linewidth]{images/book1/ai/g4-ferns-sori.png}

{\small पत्ते के नीचे: भूरे बिंदुओं की सुथरी क़तारें, हर एक हज़ारों
\omterm{def:g4:plants-without-seeds:spore}{बीजाणुओं} की पुड़िया।}
\end{omfigure}

\begin{proposition}[बीजाणु वाले पौधों को नमी चाहिए]\label{prop:g4:plants-without-seeds:damp}
\omterm{def:g4:plants-without-seeds:spore}{बीजाणु} वाले \omterm{def:g1:plants-around-us:plant}{पौधे} --- \emph{मॉस}\index{मॉस} और
\emph{फ़र्न}\index{फ़र्न} --- सिर्फ़ वहीं जनन कर सकते हैं जहाँ
\emph{नमी} हो: उनके \omterm{def:g4:plants-without-seeds:spore}{बीजाणु}, और उनसे उगने वाली नाज़ुक अवस्थाएँ, सूखी हवा
में सूखकर मर जाती हैं। इसीलिए \omterm{prop:g4:plants-without-seeds:damp}{मॉस} छायादार उत्तरी दीवार पर तो गद्दी बिछा
देती है पर धूप वाली दक्षिणी दीवार पर कभी नहीं, और इसीलिए \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} नम जंगलों,
नदी-किनारों और कुओं पर भीड़ लगाते हैं।
\end{proposition}
\admitted

\begin{remark}[बीजाणु बीज नहीं है]\label{rem:g4:plants-without-seeds:sporeseed}
दोनों सफ़र करने वाली शुरुआतें हैं, पर \omterm{def:g2:from-seed-to-plant:seed}{बीज} एक पूरा उत्तरजीविता-सामान है
--- छिलका, \omterm{def:g3:animals-through-seasons:active}{भोजन-भंडार}, पहले से बना नन्हा \omterm{def:g1:plants-around-us:plant}{पौधा} --- जबकि \omterm{def:g4:plants-without-seeds:spore}{बीजाणु} नंगा है।
संख्याओं का कारण यही है: \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} के लाखों \omterm{def:g4:plants-without-seeds:spore}{बीजाणु}, और सेम की फली के दर्जन
भर \omterm{def:g2:from-seed-to-plant:seed}{बीज}। नंगी शुरुआतें बनाने में सस्ती और खोने में आसान होती हैं; ये
दोनों योजनाएँ \cref{prop:g4:animal-reproduction:strategies} की कॉड
मछली और हथिनी की ही गूँज हैं।
\end{remark}

\section{बिना बोए नए पौधे}

\begin{definition}[कायिक जनन]\label{def:g4:plants-without-seeds:vegetative}
बहुत-से \omterm{def:g1:plants-around-us:plant}{पौधे} नए \omterm{def:g1:plants-around-us:plant}{पौधे} \emph{अपने ही किसी टुकड़े से सीधे} भी बना सकते हैं
--- न \omterm{def:g4:plants-without-seeds:spore}{बीजाणु}, न \omterm{def:g2:from-seed-to-plant:seed}{बीज}, और सिर्फ़ एक जनक। इसे \emph{कायिक
जनन}\index{कायिक जनन} कहते हैं। नया \omterm{def:g1:plants-around-us:plant}{पौधा} पुराने का जीवित आगे बढ़ा हुआ
रूप है, और उसी जैसा हूबहू दिखता है।
\end{definition}

\begin{example}[स्ट्रॉबेरी के भूस्तारी]\label{ex:g4:plants-without-seeds:runner}
स्ट्रॉबेरी का \omterm{def:g1:plants-around-us:plant}{पौधा} मिट्टी पर लंबे, पतले, आड़े \omterm{def:g1:plants-around-us:parts}{डंठल} फैलाता है ---
\emph{भूस्तारी}\index{भूस्तारी}। जहाँ \omterm{ex:g4:plants-without-seeds:runner}{भूस्तारी} ज़मीन छूता है, वहाँ वह
\omterm{def:g1:plants-around-us:parts}{जड़ें} जमाता है, \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} निकालता है, और स्ट्रॉबेरी का पूरा नया \omterm{def:g1:plants-around-us:plant}{पौधा} बन
जाता है; फिर \omterm{ex:g4:plants-without-seeds:runner}{भूस्तारी} सूख जाता है और बच्चा अपने पैरों पर खड़ा रह जाता
है। एक \omterm{def:g1:plants-around-us:plant}{पौधा} बिना एक भी \omterm{def:g2:from-seed-to-plant:seed}{बीज} बोए पूरी क्यारी बन जाता है।
\end{example}

\begin{example}[शल्ककंद और कंद]\label{ex:g4:plants-without-seeds:bulbtuber}
ट्यूलिप का \emph{शल्ककंद}\index{शल्ककंद} और आलू का
\emph{कंद}\index{कंद} ज़मीन के नीचे के भंडार-घर हैं --- और शुरू करने
के सामान भी। ज़मीन में पड़ा रहने दिया जाए तो \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद} बँटकर बेटी-शल्ककंद
बना देता है, और हर एक आगे चलकर ट्यूलिप बनेगा; अंकुरित आलू की हर
``\omterm{def:g1:the-five-senses:sense}{आँख}'' से एक पूरा नया \omterm{def:g1:plants-around-us:plant}{पौधा} उग सकता है। माली आलू बोते हैं, आलू के \omterm{def:g2:from-seed-to-plant:seed}{बीज}
नहीं।
\end{example}

\begin{method}[कलम लेना]\label{met:g4:plants-without-seeds:cutting}
माली \emph{कलमों}\index{कलम} के ज़रिए \omterm{def:g4:plants-without-seeds:vegetative}{कायिक जनन} की नक़ल करते हैं:
\begin{enumerate}
  \item जनक \omterm{def:g1:plants-around-us:plant}{पौधे} से एक बालिश्त भर लंबा, स्वस्थ और नया शाख़ काटो
        (पुदीना, बेंत और जेरेनियम अच्छे चलते हैं);
  \item नीचे की \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} हटा दो और शाख़ को पानी या नम मिट्टी में
        खड़ा करो;
  \item रुको: दिनों या हफ़्तों में कटे \omterm{def:g1:plants-around-us:parts}{डंठल} से \omterm{def:g1:plants-around-us:parts}{जड़ें} निकल आती हैं;
  \item उसे बाहर रोप दो --- यानी एक पूरा नया \omterm{def:g1:plants-around-us:plant}{पौधा}, जो अपने अकेले जनक
        जैसा हूबहू है।
\end{enumerate}
\end{method}

\begin{remark}[एक जनक, हूबहू नक़लें]\label{rem:g4:plants-without-seeds:copy}
\omterm{def:g2:from-seed-to-plant:seed}{बीज} दो जनकों को मिलाते हैं
(\cref{rem:g4:animal-reproduction:mix} \omterm{def:g3:flowers-fruits-seeds:parts}{पराग} के ज़रिए \omterm{def:g1:plants-around-us:plant}{पौधों} पर भी लागू
होता है); \omterm{def:g4:plants-without-seeds:vegetative}{कायिक जनन} एक की नक़ल उतारता है। इसीलिए सेब की किसी नामी क़िस्म
का हर \omterm{def:g1:plants-around-us:plant}{पौधा} \omterm{met:g4:plants-without-seeds:cutting}{कलमों} और क़लम-बंदी से उगाया जाता है, \omterm{def:g2:from-seed-to-plant:seed}{बीजों} से कभी नहीं: बाग़
को एक बेहतरीन \omterm{def:g1:plants-around-us:tree}{पेड़} की हूबहू नक़लें चाहिए। नक़ल का दाम है एकरूपता ---
एक जैसे \omterm{def:g1:plants-around-us:tree}{पेड़ों} के बाग़ की कमज़ोरियाँ भी एक जैसी होती हैं, जैसा आगे का
कोई साल दिखाएगा।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:1}
दो ऐसे \omterm{def:g1:plants-around-us:plant}{पौधों} के नाम बताओ जो कभी \omterm{def:g2:from-seed-to-plant:seed}{बीज} नहीं बनाते। वे जनन कैसे करते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:2}
\omterm{def:g4:plants-without-seeds:spore}{बीजाणु} और \omterm{def:g2:from-seed-to-plant:seed}{बीज} के बीच दो फ़र्क़ बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:3}
\omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} के \omterm{def:g4:plants-without-seeds:spore}{बीजाणु} कहाँ बनते हैं? बताओ कि गर्मियों के आख़िर में पत्ते के
नीचे तुम्हें क्या मिलता है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:4}
\omterm{prop:g4:plants-without-seeds:damp}{मॉस} छायादार उत्तरी दीवार पर क्यों उगती है और धूप वाली दक्षिणी पर
क्यों नहीं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:5}
\omterm{def:g4:plants-without-seeds:vegetative}{कायिक जनन} क्या है? इसमें कितने जनक लगते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:6}
स्ट्रॉबेरी के \omterm{ex:g4:plants-without-seeds:runner}{भूस्तारी} की कहानी पहले \omterm{def:g1:plants-around-us:parts}{डंठल} से लेकर अपने पैरों पर खड़े
बच्चा-पौधे तक बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:7}
माली आलू के \omterm{def:g2:from-seed-to-plant:seed}{बीज} बोने के बजाय आलू ही क्यों बोते हैं? हर ``\omterm{def:g1:the-five-senses:sense}{आँख}'' क्या
कर सकती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:8}
करके देखो: \cref{met:g4:plants-without-seeds:cutting} से पुदीने की एक
\omterm{met:g4:plants-without-seeds:cutting}{कलम} लो और उसे खिड़की पर पानी के गिलास में खड़ा करो। क्या निकलता है,
और लगभग कितने समय बाद?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:9}
\omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} लाखों \omterm{def:g4:plants-without-seeds:spore}{बीजाणु} बनाता है; सेम की फली दर्जन भर \omterm{def:g2:from-seed-to-plant:seed}{बीज}। संख्याओं का यह
फ़र्क़ \cref{rem:g4:plants-without-seeds:sporeseed} से समझाओ --- और उन
दोनों \omterm{def:g1:animals-around-us:animal}{जंतुओं} के नाम बताओ जिनकी रणनीतियों की यह गूँज है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:10}
एक माली पाती है कि स्ट्रॉबेरी के नए \omterm{def:g1:plants-around-us:plant}{पौधे} जनक जैसे हूबहू हैं, जबकि उसके
सेम के \omterm{def:g1:plants-around-us:plant}{पौधे} अपने जनकों से थोड़ा अलग हैं। दोनों प्रेक्षण
\cref{rem:g4:plants-without-seeds:copy} से समझाओ।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:plants-without-seeds:11}
सर्दी की बाढ़ से टूटी बेंत की शाखा बहकर नीचे जाती है और कीचड़ के किसी
किनारे पर अटक जाती है, और अकसर बढ़कर बेंत का नया \omterm{def:g1:plants-around-us:tree}{पेड़} बन जाती है। यह
किस तरह का जनन है --- और नदियों के किनारे इतनी बार बेंतों की क़तार
क्यों होती है? बाढ़ ने वह काम किया जो
\cref{met:g4:plants-without-seeds:cutting} का माली कैंची से करता है ---
वह काम क्या था?
\end{exercise}
