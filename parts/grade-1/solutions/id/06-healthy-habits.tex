\section*{Bab \ref{ch:g1:healthy-habits} --- Kebiasaan Sehat}

\begin{solution}{exo:g1:healthy-habits:1}
\omterm{def:g1:healthy-habits:germs}{Kuman} adalah \omterm{def:g1:living-or-not:living}{makhluk hidup} yang jauh terlalu kecil untuk dilihat, dan
ditemukan di mana-mana; sebagian dapat membuat kita sakit ketika masuk
ke dalam tubuh. Karena tidak terlihat, tangan dapat tampak bersih
sempurna dan tetap membawanya.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:2}
Sebelum makan, sesudah dari kamar kecil, dan sesudah bermain di luar
atau menyentuh \omterm{def:g1:animals-around-us:animal}{hewan}.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:3}
Sambil menghitung pelan sampai $20$ (kira-kira selama satu lagu
pendek). Yang mudah terlupa: sela-sela jari, sekeliling ibu jari, dan
punggung tangan.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:4}
Sisa makanan tertinggal di gigi sesudah makan. \omterm{def:g1:healthy-habits:germs}{Kuman} di dalam mulut
memakan sisa itu, terutama yang manis, dan sedikit demi sedikit mereka
membuat lubang di gigi, yang terasa sakit. Menyikat menyapu sisanya
sebelum pesta itu dimulai.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:5}
Sesudah sarapan dan sebelum \omterm{prop:g1:healthy-habits:needs}{tidur}, setiap hari --- dan waktunya
kira-kira selama satu lagu, pada setiap sisi setiap gigi.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:6}
\omterm{prop:g1:healthy-habits:needs}{Makanan beragam} (buah saat sarapan, air untuk diminum), gerak (bermain
di luar, bersepeda), dan \omterm{prop:g1:healthy-habits:needs}{tidur} (malam yang panjang dengan waktu \omterm{prop:g1:healthy-habits:needs}{tidur}
yang awal). Contohnya terbuka.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:7}
Ia beristirahat, memperbaiki dirinya dan tumbuh. Seorang anak
membutuhkan malam yang lebih panjang --- jauh lebih panjang daripada
orang dewasa.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:8}
Misalnya: menguap, \omterm{def:g1:the-five-senses:sense}{mata} yang pedih atau terasa berat, uring-uringan
tanpa sebab.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:9}
Gigi susu masih harus mengunyah bertahun-tahun sebelum tanggal; lubang
padanya sama sakitnya dengan lubang pada gigi tetap; dan gigi tetap
yang menunggu di bawahnya berhak tiba di dalam mulut yang bersih dan
sehat. Dua di antara alasan ini sudah cukup.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:10}
Bubuk kilapnya melambangkan \omterm{def:g1:healthy-habits:germs}{kuman}. Bersalaman, membuka pintu dan
berbagi pensil memindahkannya dari tangan ke tangan tanpa terlihat ---
dan mencuci dengan sabun selama hitungan pelan sampai $20$
menghilangkannya, baik bubuk kilap maupun \omterm{def:g1:healthy-habits:germs}{kuman}.
\end{solution}
