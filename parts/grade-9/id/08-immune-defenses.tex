\chapter{Pertahanan Kekebalan}\label{ch:g9:immune-defenses}

Pintunya sudah ditahan
(\cref{prop:g9:microbes-and-infection:doors}) --- namun tetap
saja, kadang, seorang penyusup mendarat: \omterm{def:g9:microbes-and-infection:sorted}{bakteri} serpihan
kayunya memperbanyak diri di dalam jaringan yang hangat,
virusnya sudah mencetak di dalam \omterm{def:g5:first-look-at-cells:cell}{sel} saluran napasnya. Apa
yang terjadi berikutnya adalah sistem besar ketiga tubuhnya
sesudah \omterm{def:g8:nervous-communication:neuron}{saraf} dan \omterm{def:g8:hormonal-communication:glands}{hormon}: pertahanan \omterm{def:g9:immune-defenses:memory}{kekebalannya} --- sebuah
tentara tetap, sebuah korps ahli yang beringatan, dan kiat
\omterm{prop:g9:immune-defenses:vaccination}{vaksinasi} yang melatihnya lebih dahulu.

\section{Tanggapan cepatnya: peradangan}

\begin{definition}[Tentara tetapnya]\label{def:g9:immune-defenses:standing}
Yang berpatroli di dalam \omterm{prop:g4:journey-of-food:absorb}{darah} dan jaringannya adalah \omterm{def:g5:first-look-at-cells:cell}{sel}
\omterm{prop:g4:journey-of-food:absorb}{darah} putih\index{sel darah putih} --- pasukan sistem
\omterm{def:g9:immune-defenses:memory}{kekebalannya}, dibuat di dalam sumsum \omterm{def:g3:bones-and-muscles:skeleton}{tulangnya}, menumpang
jaringan \cref{prop:g4:heart-and-blood:carries}. Penanggap
pertamanya adalah para \emph{penelan}: \omterm{def:g5:first-look-at-cells:cell}{sel} yang mengalir
mengelilingi seorang penyusup lalu mencernanya (gumpalan
mengalir pada \cref{ex:g6:cell-unit-of-life:drop}
menjalankan manuver yang sama dengan seorang pemburu yang
hidup bebas --- tubuhnya mempekerjakan kiat purba itu
sebagai pertahanan).
\end{definition}

\begin{proposition}[Peradangan, dibaca dengan benar]\label{prop:g9:immune-defenses:inflammation}
Akibat sebuah serpihan kayu --- kemerahan, kehangatan,
pembengkakan, denyut nyeri --- bukanlah \omterm{def:g9:microbes-and-infection:stages}{infeksinya} yang
menang melainkan pertahanannya yang tiba. Isyarat siaga dari
jaringan yang rusak melebarkan \omterm{def:g4:heart-and-blood:vessels}{pembuluh} setempatnya
(pengalihan rute pada \cref{prop:g7:organs-at-work:supply},
yang direkuisisi): lebih banyak \omterm{prop:g4:journey-of-food:absorb}{darah} hangat, dindingnya
dibuat bocor, para penelannya memeras diri keluar dari
\omterm{def:g7:heart-and-circulation:vessels}{kapilernya} lalu merangkak ke lokasinya, sambil memakan
\omterm{def:g9:microbes-and-infection:sorted}{bakteri} di sepanjang jalannya. Nanah adalah daftar korban
medan perangnya --- penelan yang gugur, penyusup yang mati,
puing-puingnya. Sebagian besar pendaratannya berakhir di
sini, dalam hitungan hari, tanpa dibicarakan.
\end{proposition}
\admitted

\begin{example}[Membaca sebuah luka sayat yang menyembuh]\label{ex:g9:immune-defenses:cut}
Hari pertama: lingkaran merah, hangat, nyeri --- \omterm{def:g4:heart-and-blood:vessels}{pembuluhnya}
lebar, pasukannya tiba. Hari kedua: sedikit nanah --- daftar
korbannya. Hari keempat: lingkarannya memudar, keropengnya
kering --- perbanyakannya kalah berlomba dengan
penelanannya, dan pembangunan ulang \omterm{def:g1:the-five-senses:sense}{kulitnya}
(\cref{prop:g5:first-look-at-cells:growth}) menutup pintunya
di belakang perangnya. Seluruh kampanyenya, diperangi lalu
dimenangkan di bawah skala kecemasannya.
\end{example}

\section{Para ahlinya: pengenalan dan ingatan}

\begin{proposition}[Tanggapan khususnya]\label{prop:g9:immune-defenses:specific}
Ketika tanggapan cepatnya terkejar, korps ahlinya turun ke
medan --- lebih lambat, dan tepat:
\begin{itemize}
  \item setiap penyusup mengangkut tanda permukaan --- yaitu
        \emph{antigen}\index{antigen} miliknya --- bentuk
        molekul yang dibaca para ahlinya sebagai asing
        (penanda-diri pada
        \cref{ex:g9:unique-individuals:medicine} adalah
        pembacaan yang sama, dibidikkan ke dalam);
  \item di antara \omterm{def:g5:first-look-at-cells:cell}{sel} putih ahlinya, sedikit yang mengangkut
        reseptor yang cocok dengan \omterm{prop:g9:immune-defenses:specific}{antigen} \emph{ini};
        persentuhannya menyeleksi mereka, dan mereka
        memperbanyak diri menjadi sebuah tentara klona
        (logika seleksinya, dijalankan di dalam tubuh dalam
        hitungan hari);
  \item senjatanya: \emph{antibodi}\index{antibodi} ---
        molekul yang dipaskan lalu dicurahkan ke dalam
        \omterm{prop:g4:journey-of-food:absorb}{darah}, yang memakukan dan menggumpalkan penyusupnya
        bagi para penelannya --- dan ahli pembunuh yang
        memusnahkan \omterm{def:g5:first-look-at-cells:cell}{sel} tubuhnya sendiri yang sudah dibajak,
        satu-satunya cara menghentikan mesin cetak sebuah
        \omterm{def:g9:microbes-and-infection:sorted}{virus} (\cref{ex:g9:microbes-and-infection:contrast});
  \item ongkos pertempurannya adalah bagian dari sakitnya:
        demamnya (tubuh yang dijalankan panas dengan sengaja
        --- banyak \omterm{prop:g9:microbes-and-infection:mostly}{patogen} memperbanyak diri lebih buruk
        dalam keadaan hangat), nyeri-nyerinya, \emph{\omterm{def:g8:hormonal-communication:glands}{kelenjar}
        getah bening} yang membengkak --- barak para ahlinya,
        yang sesak pada masa perang.
\end{itemize}
\end{proposition}
\admitted

\begin{definition}[Ingatan kekebalannya]\label{def:g9:immune-defenses:memory}
Sesudah kemenangannya, sebagian tentara klonanya mundur
menjadi \emph{sel ingatan}\index{sel ingatan}: ahli yang
berumur panjang, yang memegang perian \omterm{prop:g9:immune-defenses:specific}{antigennya}. Sebuah
pendaratan kedua oleh penyusup yang \emph{sama} bertemu
sebuah korps yang siap --- tanggapan dalam hitungan jam,
penyusupnya dibersihkan sebelum gejalanya muncul: kamu
\emph{kebal}\index{kekebalan}. Sekali campak, tidak pernah
dua kali; galur musim dingin lalu tak sanggup menyalakan
lagi wabah tahun ini
(\cref{pb:g9:microbes-and-infection:1}). Ingatan itulah
sebabnya pilek yang sama seolah tak pernah datang dua kali
--- dan sebabnya pilek terus berdatangan: virusnya bermutasi
cepat (\cref{prop:g9:genes-and-dna:explains}) menjadi perian
yang tidak dipegang ingatannya.
\end{definition}

\section{Melatih pertahanannya: vaksinasi}

\begin{proposition}[Vaksinasi]\label{prop:g9:immune-defenses:vaccination}
Yang disebut \emph{vaksinasi}\index{vaksinasi} adalah
ingatan \omterm{def:g9:immune-defenses:memory}{kekebalan} yang diciptakan tanpa penyakitnya:
sodorkan kepada para ahlinya \emph{perian} penyusupnya ---
\omterm{def:g6:producing-our-food:fermentation}{mikroba} yang dimatikan atau dilemahkan, serpihan tak
merugikan yang membawa \omterm{prop:g9:immune-defenses:specific}{antigennya} --- dan seluruh program
seleksi-dan-ingatannya berjalan, dikurangi bahayanya. Korps
terlatihnya lalu menemui \omterm{prop:g9:microbes-and-infection:mostly}{patogen} yang sungguhan, kalaupun
pernah, sebagai sebuah pendaratan kedua. Dua akibatnya
memikul bobot kesehatan masyarakatnya:
\begin{enumerate}
  \item yang divaksinasi sudah terlindung sebelum
        pertempuran pertamanya --- menentukan bagi penyakit
        yang pertempuran pertamanya dapat membunuh atau
        melumpuhkan;
  \item tetangga yang cukup banyak divaksinasi melaparkan
        sebuah wabah dari inang yang dapat tertular
        (padamnya wabah pada
        \cref{pb:g9:microbes-and-infection:1}):
        perlindungannya berkumpul --- perisainya menaungi
        mereka yang terlalu muda atau terlalu sakit untuk
        dilatih sendiri.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Dari taruhan cacar sapinya sampai pembasmiannya]\label{ex:g9:immune-defenses:history}
Sejarahnya berjalan dari taruhan Jenner tahun 1796 --- para
pemerah susu tertular cacar sapi yang ringan lalu tidak
pernah terkena cacar; sebuah inokulasi cacar sapi melatih
mereka terhadap sang pembunuh --- menembus vaksin
mikroba-yang-dilemahkan Pasteur yang disengaja (penulis pada
\cref{ex:g9:microbes-and-infection:history}, yang melengkapi
teorinya), sampai daftar hadir abad kedua puluh: cacar,
momok lama umat manusia, \emph{dibasmi} --- virusnya \omterm{prop:g5:biodiversity:loss}{punah}
di alam liar sejak akhir dasawarsa 1970 --- dan polio
didesak ke sudut terakhir petanya. Ingatan yang terlatih,
bila dikumpulkan, adalah satu-satunya senjata yang pernah
memusnahkan sebuah penyakit manusia.
\end{example}

\begin{method}[Membaca peristiwa kekebalan apa pun]\label{met:g9:immune-defenses:read}
Untuk \omterm{def:g9:microbes-and-infection:stages}{infeksi}, \omterm{def:g7:organs-at-work:recovery}{pemulihan}, vaksin atau hasil uji apa pun:
\begin{enumerate}
  \item tetapkan tahap perjumpaannya: pendaratan pertama
        atau kedua? (ingatan mengubah segalanya);
  \item sebutkan penanggapnya: tanggapan cepat para
        penelannya, tentara klona ahlinya, atau ingatan tetapnya;
  \item bacalah tandanya dengan benar: peradangan dan demam
        sebagai pertahanan yang bekerja, \omterm{def:g8:hormonal-communication:glands}{kelenjar} bengkak
        sebagai baraknya;
  \item untuk sebuah vaksin: kenali perian yang disodorkan,
        dan perlindungan siapa yang dikumpulkannya.
\end{enumerate}
\end{method}

\section{Soal: Buku Besar Vaksinasinya}

\begin{problem}[{Soal akhir pekan --- sepekan sebuah pusat
kesehatan, dibaca dengan cara kekebalannya}]\label{pb:g9:immune-defenses:1}
Sepekan (rekaan) sebuah pusat kesehatan: Senin, \omterm{prop:g9:immune-defenses:vaccination}{vaksinasi}
bayi; Selasa, luka robek kawat berkarat seorang petani
beserta suntikan ulangnya; Rabu, sebuah kasus campak pada
anak yang tidak divaksinasi, dan kontaknya diperiksa; Kamis,
uji \omterm{prop:g4:journey-of-food:absorb}{darah} yang membaca kadar \omterm{prop:g9:immune-defenses:specific}{antibodinya}; Jumat, vaksin
influenza tahunan bagi stafnya.

\medskip\noindent\textbf{Bagian I --- Kasus pekannya.}
\begin{enumerate}
  \item Bayi hari Senin menerima tetes \omterm{def:g6:producing-our-food:fermentation}{mikroba} yang
        dilemahkan dan suntikan serpihan. Sebutkan apa yang
        sedang disodorkan, dan program mana yang berjalan
        pada pekan-pekan berikutnya.
  \item Petani hari Selasa: luka robeknya dibersihkan (garis
        pertahanan yang mana?), lalu suntikan ulangnya
        diberikan. Apa yang dilakukan sebuah \emph{suntikan
        ulang} terhadap ingatan yang terlatih --- dan mengapa
        tetanus tidak mengizinkan taruhan pertempuran pertama?
  \item Anak campak hari Rabu: pendaratan pertamanya,
        program penuhnya. Buatlah garis waktu sakitnya:
        tanggapan cepatnya terkejar, tentara klonanya
        dibangkitkan, \omterm{def:g7:organs-at-work:recovery}{pemulihannya} --- dan apa yang mundur
        sesudahnya.
  \item Kontak yang diperiksa itu terbagi: yang divaksinasi
        (tak terganggu), yang lama sembuh (tak terganggu),
        dan satu bayi yang tidak divaksinasi, yang diawasi
        dengan cemas. Bacalah masing-masing dengan pertanyaan
        pertama \cref{met:g9:immune-defenses:read}.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pengukurannya.}
\begin{enumerate}[resume]
  \item Uji hari Kamis membaca kadar \omterm{prop:g9:immune-defenses:specific}{antibodi} terhadap
        penyakit dan vaksin masa lalunya. Apa yang disahkan
        sebuah kadar yang tinggi, dan persediaan \omterm{def:g5:first-look-at-cells:cell}{sel} apa yang
        disiratkannya?
  \item Kadar seorang pasien sudah memudar bertahun-tahun
        sesudah \omterm{prop:g9:immune-defenses:vaccination}{vaksinasinya}, dan sebuah suntikan ulang
        diresepkan. Jelaskan pemudaran dan penyuntikan
        ulangnya dalam istilah \omterm{def:g9:immune-defenses:memory}{sel ingatannya}.
  \item Vaksin influenza hari Jumat bersifat \emph{tahunan},
        tidak seperti program sekali-atau-dua-kali hari
        Senin. \omterm{def:g9:heredity-and-traits:traits}{Sifat} \omterm{def:g9:microbes-and-infection:sorted}{virus} influenza yang mana
        (\cref{def:g9:immune-defenses:memory}) yang memaksa
        pelatihan ulang tiap tahunnya?
  \item Mengapa pusatnya memvaksinasi \emph{stafnya} dengan
        desakan khusus? (Perlindungan siapa berkumpul di
        mana, di dalam sebuah gedung yang penuh orang sakit
        dan orang yang rapuh?)
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Makna buku
besarnya.}
\begin{enumerate}[resume]
  \item Orang tua anak campak yang tidak divaksinasi itu
        takut kepada vaksin lalu memercayai pertempuran
        pertama yang ``alami''. Susunlah hitungan lembut
        dokternya: bandingkan ongkos pelatihannya dengan
        ongkos pertempuran pertamanya, bagi campak.
  \item Bayi yang diawasi itu terlalu muda untuk vaksinnya.
        Nyatakan siapa yang semestinya melindunginya, dan
        kewajiban itu disebut apa pada
        \cref{prop:g9:immune-defenses:vaccination}.
  \item Hubungkan hari Selasa dan Rabu dengan kedua nama
        bersejarah pada \cref{ex:g9:immune-defenses:history}:
        gagasan siapa yang berjalan pada tiap konsultasinya?
  \item Tutuplah buku besarnya dengan ringkasan satu kalimat
        pekannya: apa yang sebenarnya diberikan pusatnya,
        dalam kosakata bab ini.
\end{enumerate}
\end{problem}
