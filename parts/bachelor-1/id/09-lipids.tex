\section{Asam lemak}

\begin{definition}[Lipid, asam lemak]\label{def:b1:lipids:fattyacid}
Yang disebut \emph{lipid}\index{lipid} adalah molekul biologis yang
ditetapkan bukan oleh struktur bersama melainkan oleh sifat bersama:
ia tidak larut dalam air dan larut dalam pelarut nonpolar. Sebagian
besarnya dibangun atas \emph{asam lemak}\index{asam lemak}: asam
karboksilat dengan rantai hidrokarbon tak bercabang sepanjang
\qtyrange{12}{24}{} karbon, yang hampir selalu berjumlah genap (mereka
dirakit dua karbon sekali jalan,
\cref{ch:b1:biosyntheses-integration}). Sebuah asam \omterm{def:b1:lipids:triglyceride}{lemak}
\emph{jenuh}\index{asam lemak jenuh} hanya memiliki ikatan tunggal dan
rantai lurus yang lentur; sebuah asam \omterm{def:b1:lipids:triglyceride}{lemak} \emph{tak jenuh}\index{asam lemak tak jenuh}
memiliki satu ikatan rangkap atau lebih, dalam susunan
\emph{cis}, yang masing-masing menaruh tekukan kaku sekitar $30^\circ$
pada rantainya. Notasinya: C18:1 $\Delta^9$ adalah asam berkarbon 18
dengan satu ikatan rangkap sesudah karbon 9 (asam oleat); sebuah asam
\emph{omega-3} memiliki ikatan rangkap terakhirnya tiga karbon dari
ujung metilnya.
\end{definition}

\begin{proposition}[Titik leleh]\label{prop:b1:lipids:melting}
Titik leleh sebuah \omterm{def:b1:lipids:fattyacid}{asam lemak}, dan \omterm{def:b1:lipids:fluidity}{kecairan} setiap rakitan \omterm{def:b1:lipids:fattyacid}{lipid} yang
memuatnya, ditetapkan oleh serapat apa rantainya dapat berkemas: ia
naik bersama panjang rantainya (\qty{44}{\celsius} bagi C12:0,
\qty{63}{\celsius} bagi C16:0, \qty{70}{\celsius} bagi C18:0) dan turun
tajam pada tiap ikatan rangkap \emph{cis} (C18:0 \qty{70}{\celsius},
C18:1 \qty{13}{\celsius}, C18:2 \qty{-5}{\celsius}, C18:3
\qty{-11}{\celsius}). \omterm{def:b1:lipids:triglyceride}{Lemak} hewan, yang kaya asam \omterm{def:b1:lipids:fattyacid}{jenuh}, padat pada
suhu ruang; \omterm{def:b1:lipids:triglyceride}{minyak} tumbuhan dan \omterm{def:b1:lipids:triglyceride}{minyak} ikan, yang kaya asam \omterm{def:b1:lipids:fattyacid}{tak jenuh},
bersifat cair.
\end{proposition}
\begin{proof}
Rantai yang lurus berbaring berdampingan dan tiap $\mathrm{CH_2}$
bersentuhan \omterm{def:b1:water-small-molecules:bonds}{van der Waals} dengan tetangganya, masing-masing bernilai
beberapa kilojoule tiap mol, terjumlahkan sepanjang rantainya: makin
banyak karbon, makin banyak sentuhan, makin banyak kalor untuk
memisahkannya. Sebuah tekukan mencegah sentuhan rapat sepanjang
beberapa karbon pada tiap sisinya dan menghapus banyak sentuhan
sekaligus.
\end{proof}

\section{Lemak: simpanan energi}

\begin{definition}[Trigliserida]\label{def:b1:lipids:triglyceride}
Yang disebut \emph{trigliserida}\index{trigliserida} (triasilgliserol,
sebuah \emph{lemak}\index{lemak} bila padat, sebuah
\emph{minyak}\index{minyak} bila cair) adalah gliserol --- alkohol
berkarbon tiga --- yang diesterkan pada ketiga hidroksilnya oleh tiga
\omterm{def:b1:lipids:fattyacid}{asam lemak}. Ia tidak bermuatan dan tak menyisakan gugus polar: ia
seluruhnya hidrofob, dan ia ditimbun sebagai tetesan tanpa air di dalam
\emph{adiposit}\index{adiposit}, yaitu \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:body-plans-tissues:tissue}{jaringan} lemak yang tak lebih
daripada setetes lemak dengan sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} yang terdorong ke satu sisi.
\end{definition}

\begin{proposition}[Lemak adalah simpanan energi terpadat]\label{prop:b1:lipids:energystore}
Mengoksidasi \qty{1}{g} \omterm{def:b1:lipids:triglyceride}{lemak} melepas \qty{38}{kJ}, berhadapan dengan
\qty{17}{kJ} bagi \qty{1}{g} karbohidrat atau protein: karbon sebuah
\omterm{def:b1:lipids:fattyacid}{asam lemak} lebih tereduksi (lebih banyak ikatan C--H, lebih sedikit
C--O) sehingga menyerahkan lebih banyak elektron kepada oksigen.
Lagi pula \omterm{def:b1:lipids:triglyceride}{lemak} ditimbun kering, sedangkan glikogen mengikat sekitar
\qty{3}{g} air tiap gram: \qty{1}{g} glikogen simpanan menghasilkan
\qty{4}{kJ} tiap gram massa yang ditimbun, \omterm{def:b1:lipids:triglyceride}{lemak} sembilan kali lebih
banyak. Seorang manusia yang kurus membawa \qty{0.5}{kg} glikogen
(energi sehari) dan \qty{12}{kg} \omterm{def:b1:lipids:triglyceride}{lemak} (lebih dari sebulan).
\end{proposition}

\begin{example}[Mengapa tidak menimbun semuanya sebagai lemak]\label{ex:b1:lipids:whyglycogen}
\omterm{def:b1:lipids:triglyceride}{Lemak} hanya dapat dibakar dengan oksigen, secara lambat, dan pada hewan
tidak dapat diubah kembali menjadi glukosa
(\cref{ch:b1:biosyntheses-integration}); otak dan \omterm{def:b1:cell-unit-of-life:cell}{sel} darah merah
memerlukan glukosa, dan otot yang berlari cepat memerlukannya lebih
cepat daripada yang sanggup dipasok \omterm{def:b1:lipids:triglyceride}{lemak}. Glikogen adalah simpanan
glukosa yang cepat dan tanpa oksigen bagi hitungan jam; \omterm{def:b1:lipids:triglyceride}{lemak}, simpanan
padat bagi hitungan minggu. Seekor burung yang bermigrasi, seekor tikus
gunung yang berhibernasi dan seekor unta terbuat dari \omterm{def:b1:lipids:triglyceride}{lemak}; kaki
seorang pelari cepat terbuat dari glikogen.
\end{example}

\begin{definition}[Lilin]\label{def:b1:lipids:wax}
Yang disebut \emph{lilin}\index{lilin} adalah ester sebuah \omterm{def:b1:lipids:fattyacid}{asam lemak}
dengan sebuah alkohol berantai panjang: padat, seluruhnya hidrofob, dan
dipakai sebagai pelapis --- \omterm{def:b1:flowering-plant-organization:tissues}{kutikula} \omterm{def:b1:flowering-plant-organization:organs}{daun}
(\cref{ch:b1:flowering-plant-organization}), permukaan serangga,
pengedap air bulu unggas dan rambut, serta sarang lebah.
\end{definition}

\section{Lipid membran dan perakitan diri}

\begin{definition}[Lipid amfifilik]\label{def:b1:lipids:amphiphilic}
\omterm{def:b1:lipids:fattyacid}{Lipid} membran bersifat \emph{amfifilik}\index{molekul amfifilik}:
sebuah \emph{kepala} polar yang disolvasi air dan satu atau dua
\emph{ekor} nonpolar yang dikucilkannya.
\emph{Gliserofosfolipid}\index{fosfolipid} adalah gliserol yang membawa
dua \omterm{def:b1:lipids:fattyacid}{asam lemak} dan, pada karbon ketiganya, sebuah fosfat yang bertaut
pada alkohol polar yang kecil (kolina, etanolamina, serina, inositol):
fosfatidilkolina adalah \omterm{def:b1:lipids:fattyacid}{lipid} tersering pada membran hewan.
\emph{Sfingolipid}\index{sfingolipid} dibangun atas amino alkohol
sfingosina alih-alih gliserol, dengan satu \omterm{def:b1:lipids:fattyacid}{asam lemak} dan kepala berupa
fosfokolina (sfingomielin) atau gula (glikolipid), dan diperkaya pada
lembar luarnya serta pada selubung saraf. \emph{Sterol}\index{sterol}
--- \emph{kolesterol}\index{kolesterol} pada hewan, sterol kerabatnya
pada tumbuhan dan jamur --- adalah molekul kaku bercincin empat dengan
satu hidroksil sebagai kepalanya, yang berbaring di antara ekor \omterm{def:b1:lipids:fattyacid}{lipid}
yang lain.
\end{definition}

\begin{proposition}[Perakitan diri]\label{prop:b1:lipids:selfassembly}
\index{perakitan diri}\index{misel}\index{liposom}
Dalam air, \omterm{def:b1:lipids:amphiphilic}{molekul amfifilik} merakit diri secara spontan sedemikian
sehingga ekornya tersembunyi dan kepalanya terpapar. Bentuk molekulnya
menentukan strukturnya: \omterm{def:b1:lipids:fattyacid}{lipid} berekor tunggal (\omterm{def:b1:lipids:fattyacid}{asam lemak}, deterjen),
yang berbentuk kerucut, membentuk \emph{misel} --- bola selebar
beberapa nanometer dengan ekornya di dalam; \omterm{def:b1:lipids:amphiphilic}{fosfolipid} berekor dua, yang
berbentuk tabung, membentuk \emph{dwilapis}, yang menutup pada dirinya
sendiri menjadi \emph{liposom} (vesikel) agar tak ada tepi yang
terpapar. Perakitannya digerakkan \omterm{prop:b1:water-small-molecules:properties}{efek hidrofob}
(\cref{ch:b1:water-small-molecules}): ia tidak memakan energi dan tidak
memerlukan enzim, dan sebuah dwilapis yang robek menutup sendiri.
Membran adalah lembar yang memulihkan diri, yang tumbuh lewat
penyisipan \omterm{def:b1:lipids:fattyacid}{lipid} baru dan tak pernah terbentuk dari ketiadaan: setiap
membran berasal dari sebuah membran.
\end{proposition}
\begin{proof}[Bukti eksperimental]
\omterm{def:b1:lipids:amphiphilic}{Fosfolipid} kering yang didispersikan dalam air membentuk vesikel
tertutup berdinding dwilapis yang tampak dalam \omterm{prop:b1:cell-unit-of-life:microscopes}{mikroskop elektron}
(Bangham, 1965); permeabilitasnya terhadap ion dan air sepadan dengan
permeabilitas membran \omterm{def:b1:cell-unit-of-life:cell}{sel}, dan vesikel itu dapat dimuati obat lalu
dileburkan dengan \omterm{def:b1:cell-unit-of-life:cell}{sel}. Menambahkan deterjen (berekor tunggal, berbentuk
kerucut) melarutkan dwilapisnya menjadi misel campuran; menyingkirkannya
lewat dialisis membiarkan dwilapisnya terbentuk kembali.
\end{proof}

\section{Kecairan membran}

\begin{definition}[Kecairan, transisi fase]\label{def:b1:lipids:fluidity}
Sebuah \omterm{def:b1:membranes-transport:membrane}{dwilapis lipid} memiliki \emph{suhu transisi fase}\index{suhu transisi fase}
$T_m$: di bawahnya rantainya berkemas dalam keadaan
teratur serupa gel yang \omterm{def:b1:lipids:fattyacid}{lipidnya} hampir tak bergerak; di atasnya
rantainya tak teratur dan dwilapisnya menjadi fluida dua dimensi tempat
sebuah \omterm{def:b1:lipids:fattyacid}{lipid} bertukar tempat dengan tetangganya sejuta kali tiap sekon
dan melintasi bidang membrannya pada \qty{1}{\micro m/s}. \omterm{def:b1:cell-unit-of-life:cell}{Sel} menjaga
membrannya di atas $T_m$: \emph{kecairan}\index{kecairan membran} itu
membuat protein dapat berdifusi dan berputar, vesikel dapat bertunas
dan melebur, serta dwilapisnya dapat menutup kembali.
\end{definition}

\begin{proposition}[Apa yang menetapkan kecairannya]\label{prop:b1:lipids:fluidityfactors}
$T_m$ naik bersama panjang rantainya dan turun bersama ketakjenuhannya,
sebagaimana pada \omterm{def:b1:lipids:fattyacid}{asam lemak} bebas; dwilapis berantai C18:0 meleleh pada
\qty{55}{\celsius}, dwilapis berantai C18:1 pada \qty{-20}{\celsius}.
\omterm{def:b1:lipids:amphiphilic}{Kolesterol}, yang disisipkan di antara ekornya, berpengaruh ganda: di
atas $T_m$ cincin kakunya membatasi gerak rantainya dan mengakukan
membrannya; di bawah $T_m$ cincin itu mencegah rantainya berkemas dan
menahannya agar tidak membeku. Ia melebarkan transisinya menjadi
perubahan yang berangsur dan menjaga \omterm{def:b1:lipids:fluidity}{kecairannya} hampir tetap pada
rentang suhu yang lebar --- sebuah \omterm{def:b1:water-small-molecules:buffer}{penyangga} \omterm{def:b1:lipids:fluidity}{kecairan}. \omterm{def:b1:membranes-transport:membrane}{Membran plasma}
hewan mengandung sampai satu \omterm{def:b1:lipids:amphiphilic}{kolesterol} tiap \omterm{def:b1:lipids:amphiphilic}{fosfolipid}.
\end{proposition}

\begin{proposition}[Adaptasi homeoviskos]\label{prop:b1:lipids:homeoviscous}
\index{adaptasi homeoviskos}
\omterm{def:b1:organism-environment:organism}{Organisme} yang tidak sanggup mengatur suhunya menyetel susunan
membrannya agar \omterm{def:b1:lipids:fluidity}{kecairannya} tetap: seiring turunnya suhu, mereka
mengganti \omterm{def:b1:lipids:fattyacid}{asam lemak jenuh} dengan yang \omterm{def:b1:lipids:fattyacid}{tak jenuh} dan rantai panjang
dengan yang lebih pendek, dan sebaliknya ketika suhunya naik.
\end{proposition}
\begin{proof}[Bukti eksperimental]
\emph{E. coli} yang ditumbuhkan pada \qty{10}{\celsius} memiliki dua
kali lipat \omterm{def:b1:lipids:fattyacid}{asam lemak tak jenuh} dibandingkan galur yang sama yang
ditumbuhkan pada \qty{40}{\celsius}, dan membran kedua biakan itu,
diukur lewat kelincahan sebuah pelacak, sama cairnya pada suhu
tumbuhnya masing-masing. Ikan trout yang diaklimatisasi pada
\qty{5}{\celsius} membawa lebih banyak asam \omterm{def:b1:lipids:fattyacid}{tak jenuh} ganda dalam \omterm{def:b1:lipids:fattyacid}{lipid}
membrannya daripada trout pada \qty{20}{\celsius}; \omterm{def:b1:lipids:triglyceride}{lemak} kaki rusa
kutub, yang dekat dengan salju, lebih \omterm{def:b1:lipids:fattyacid}{tak jenuh} daripada \omterm{def:b1:lipids:triglyceride}{lemak}
\omterm{def:b1:flowering-plant-organization:organs}{batang} tubuhnya. Tumbuhan yang selamat dari beku memperkaya membrannya
dengan \omterm{def:b1:lipids:fattyacid}{lipid} \omterm{def:b1:lipids:fattyacid}{tak jenuh} pada musim gugur, dan mutan yang tak sanggup
melakukannya mati ketika didinginkan.
\end{proof}

\begin{example}[Mentega, minyak zaitun, minyak ikan]\label{ex:b1:lipids:kitchen}
Mentega (\qty{65}{\%} \omterm{def:b1:lipids:fattyacid}{jenuh}) padat di dalam lemari pendingin dan lembut
pada suhu ruang; \omterm{def:b1:lipids:triglyceride}{minyak} zaitun (\qty{75}{\%} asam oleat) cair pada suhu
ruang dan mengeruh di dalam lemari pendingin; \omterm{def:b1:lipids:triglyceride}{minyak} ikan, yang kaya
asam omega-3 dengan lima dan enam ikatan rangkap, tetap cair pada
\qty{-20}{\celsius}, dan itulah yang dibutuhkan membran seekor ikan kod
di Atlantik Utara.
\end{example}

\section{Sterol, pigmen dan isyarat}

\begin{definition}[Steroid, isoprenoid]\label{def:b1:lipids:steroids}
\emph{Steroid}\index{steroid} berbagi keempat cincin bergabung milik
\omterm{def:b1:lipids:amphiphilic}{kolesterol}: \omterm{def:b1:lipids:amphiphilic}{kolesterol} itu sendiri (membran, dan prazat selebihnya),
garam empedu, vitamin D, serta hormon steroid --- kortisol,
aldosteron, hormon kelamin --- yaitu molekul kecil dan hidrofob yang
melintasi membran dengan bebas dan bekerja pada reseptor di dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}.
\emph{Isoprenoid}\index{isoprenoid} (terpena) adalah rantai dan cincin
satuan isoprena berkarbon lima: \emph{karotenoid} yang mewarnai wortel
dan melindungi kloroplas (\cref{ch:b1:photosynthesis}), rantai samping
klorofil, kuinon rantai pengangkut elektron, karet, dan vitamin A, E
serta K yang larut dalam \omterm{def:b1:lipids:triglyceride}{lemak}.
\end{definition}

\begin{proposition}[Apa yang dikerjakan lipid]\label{prop:b1:lipids:functions}
Penimbunan energi (\omterm{def:b1:lipids:triglyceride}{trigliserida}); membran (\omterm{def:b1:lipids:amphiphilic}{fosfolipid}, \omterm{def:b1:lipids:amphiphilic}{sfingolipid},
\omterm{def:b1:lipids:amphiphilic}{sterol}); penyekatan dan pengedapan air (\omterm{def:b1:lipids:triglyceride}{lemak}, \omterm{def:b1:lipids:wax}{lilin}); penyerapan dan
perlindungan terhadap cahaya (karotenoid); pengangkutan elektron
(kuinon); pengisyaratan antarsel (hormon \omterm{def:b1:lipids:steroids}{steroid}, prostaglandin) dan di
dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} (\omterm{def:b1:lipids:amphiphilic}{fosfolipid} inositol, diasilgliserol); vitamin. Satu sifat ---
ketaklarutan dalam air --- mendasari semuanya: sebuah \omterm{def:b1:lipids:fattyacid}{lipid} tinggal di
tempat ia ditaruh, di dalam sebuah tetesan, sebuah selaput atau sebuah
membran, atau melintasi sebuah membran tanpa pembawa.
\end{proposition}

\begin{example}[Hormon yang tak memerlukan reseptor di permukaan]\label{ex:b1:lipids:steroidhormone}
Kortisol, yang disekresikan kelenjar adrenal, mengembara dalam darah
terikat pada protein pembawa, meninggalkannya pada \omterm{def:b1:cell-unit-of-life:cell}{sel} sasaran, larut
menembus membran plasma dalam beberapa sekon, dan mengikat sebuah
reseptor dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosol} yang lalu masuk ke \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} dan menyalakan gen
(\cref{ch:b1:expression-control}). Sebuah hormon protein seperti
insulin, yang larut dalam air dan terlalu besar untuk melintas, harus
mengikat sebuah reseptor di sisi luar lalu pesannya diteruskan ke dalam.
Kimia sang pembawa pesanlah yang menentukan rute pesannya.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:lipids:1}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:lipids:2}
Tuliskan notasi asam linoleat (18 karbon, ikatan rangkap sesudah karbon
9 dan 12) dan sebutkan apakah ia asam omega-3 atau omega-6.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:lipids:3}
Mengapa sebuah \omterm{def:b1:lipids:triglyceride}{trigliserida} membentuk tetesan sedangkan sebuah
\omterm{def:b1:lipids:amphiphilic}{fosfolipid} membentuk dwilapis? Perbedaan struktur apa yang menjadi
sebabnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:lipids:4}
Dari gambar \omterm{def:b1:lipids:fluidity}{kecairannya}, pada suhu berapa masing-masing dari ketiga

ada pada sebuah bakteri yang hidup pada \qty{10}{\celsius}?
\end{exercise}

istirahat \qty{8}{MJ/d}, dan massa glikogen terhidrasi yang akan

\begin{exercise}[$\star\star$]\label{exo:b1:lipids:6}

\begin{exercise}[$\star\star\star$]\label{exo:b1:lipids:10}

\section{Soal: Air Dingin dan Lemak Gurun}

\begin{problem}[{Soal akhir pekan --- membran seekor trout di aliran
gunung dan punuk seekor unta di gurun: ketakjenuhan, suhu transisi,
kerapatan energi dan air metabolik, berakhir pada massa yang dihemat
dengan menimbun lemak alih-alih glikogen}]
\label{pb:b1:lipids:1}
Bagian I membahas seekor trout yang \omterm{def:b1:lipids:amphiphilic}{fosfolipid} membrannya membawa rantai
C16:0, C18:1 dan C22:6 (asam omega-3 dengan enam ikatan rangkap).
Bagian II membahas seekor unta berpunuk satu bermassa \qty{500}{kg}
dengan punuk \omterm{def:b1:lipids:triglyceride}{trigliserida} \qty{30}{kg}, yang melintasi gurun pada \omterm{def:b1:organism-environment:allometry}{laju
metabolisme} \qty{60}{MJ/d}. Energi: \omterm{def:b1:lipids:triglyceride}{lemak} \qty{38}{kJ/g}, glikogen
\qty{17}{kJ/g} kering, yang ditimbun dengan \qty{3}{g} air tiap gram.
Air metabolik: \qty{1.07}{g/g} \omterm{def:b1:lipids:triglyceride}{lemak}. Oksigen: \qty{2.0}{L} tiap gram
\omterm{def:b1:lipids:triglyceride}{lemak}; udara mengandung \qty{21}{\%} oksigen; unta itu menyarikan
\qty{5}{\%} oksigen yang dihirupnya; udara yang diembuskan membawa
\qty{30}{g/m^3} air lebih banyak daripada udara gurun yang dihirupnya.

\medskip
\noindent\textbf{Bagian I --- Sang trout.}
Trout yang diaklimatisasi pada \qty{20}{\celsius} bermembran dengan
rantai \qty{40}{\%} C16:0, \qty{45}{\%} C18:1 dan \qty{15}{\%} C22:6;
trout pada \qty{5}{\celsius}, \qty{25}{\%}, \qty{40}{\%} dan \qty{35}{\%}.
\begin{enumerate}
  \item Tuliskan notasi ketiga asam itu dan hitunglah ikatan rangkap
        tiap rantai pada masing-masing.
  \item Hitunglah rata-rata banyaknya ikatan rangkap tiap rantai pada
        tiap suhu.
  \item Jelaskan, dari pengemasan rantainya, mengapa perubahan itu
        menurunkan suhu transisi membrannya.
  \item $T_m$ sebuah dwilapis turun sekitar \qty{20}{\celsius} bagi
        tiap tambahan \qty{0.5}{} ikatan rangkap tiap rantai
        rata-ratanya. Taksirlah geseran $T_m$ antara kedua trout itu.
  \item Membran trout yang hangat cair pada \qty{20}{\celsius} tetapi
        akan menggel pada \qty{5}{\celsius}. Apa yang akan terjadi pada
        protein pengangkutnya, pompanya dan penghantaran sarafnya pada
        sebuah malam yang tiba-tiba dingin?
  \item Adaptasinya memakan waktu berhari-hari. Sebutkan enzim yang
        harus dibuat lebih banyak oleh trout itu, dan di sebelah mana
        dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya} enzim itu bekerja (\cref{ch:b1:eukaryotic-cell}).
  \item \omterm{def:b1:lipids:amphiphilic}{Kolesterolnya} satu molekul tiap dua \omterm{def:b1:lipids:amphiphilic}{fosfolipid} pada kedua
        trout. Jelaskan apa sumbangannya pada masing-masing.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Energi sang unta.}
\begin{enumerate}[resume]
  \item Hitunglah energi yang ditimbun dalam punuknya.
  \item Untuk berapa hari lintasan itu energi tersebut mencukupi?
  \item Hitunglah massa glikogen kering yang memegang energi yang sama,
        dan massa glikogen terhidrasinya.
  \item Hitunglah massa yang dihemat dengan menimbun \omterm{def:b1:lipids:triglyceride}{lemak} alih-alih
        glikogen, lalu nyatakan sebagai fraksi massa tubuh untanya.
  \item \omterm{def:b1:lipids:triglyceride}{Lemaknya} ditimbun dalam sebuah punuk alih-alih disebar di bawah
        kulitnya. Ajukan sebuah alasan, dari \omterm{prop:b1:mammal-organization:heatbudget}{neraca kalor}
        \cref{ch:b1:mammal-organization}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Apakah punuknya sebuah simpanan air?}
\begin{enumerate}[resume]
  \item Hitunglah air metabolik yang dihasilkan dengan mengoksidasi
        seluruh punuknya.
  \item Hitunglah oksigen yang dibutuhkan, dalam liter.
  \item Hitunglah volume udara yang harus diventilasikan unta itu untuk
        memperolehnya.
  \item Hitunglah air yang hilang saat mengembuskan udara itu.
  \item Bandingkan air yang diperoleh dengan air yang hilang lalu
        simpulkan.
  \item Alih-alih berkeringat, unta itu membiarkan suhu tubuhnya naik
        \qty{6}{\celsius} pada siang hari lalu mendingin pada malam
        hari. Hitunglah kalor yang ditimbunnya dengan cara ini
        (kapasitas kalor \qty{3.5}{kJ.kg^{-1}.K^{-1}}) dan air yang akan
        dituntut penguapan kalor yang sama (\qty{2.4}{kJ/g}).
  \item Seekor unta dapat kehilangan \qty{25}{\%} air tubuhnya dan tetap
        hidup, dan \omterm{def:b1:cell-unit-of-life:cell}{sel} darah merahnya membengkak sampai dua kali
        volumenya tanpa pecah ketika ia minum \qty{100}{L} dalam sepuluh
        menit. Kaitkan sifat kedua itu dengan membran
        \cref{ch:b1:membranes-transport}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Aritmetika dwilapisnya.}
Sebuah kepala \omterm{def:b1:lipids:amphiphilic}{fosfolipid} menempati \qty{0.65}{nm^2}; sebuah dwilapis
setebal \qty{5}{nm}; \omterm{def:b1:cell-unit-of-life:cell}{sel} darah merah unta itu berpermukaan
\qty{80}{\micro m^2} dan berjumlah $\num{8e12}$ tiap liter darah.
\begin{enumerate}[resume]
  \item Hitunglah banyaknya \omterm{def:b1:lipids:amphiphilic}{fosfolipid} dalam satu membran \omterm{def:b1:cell-unit-of-life:cell}{sel} merah.
  \item Hitunglah banyaknya dalam semua \omterm{def:b1:cell-unit-of-life:cell}{sel} merah pada \qty{40}{L} darah.
  \item Pada \qty{750}{g/mol}, hitunglah massa \omterm{def:b1:lipids:fattyacid}{lipid} itu dalam gram.
  \item Sebuah \omterm{def:b1:lipids:amphiphilic}{fosfolipid} berbalik dari satu lembar ke lembar yang lain
        secara spontan kira-kira sekali seminggu, tetapi berdifusi
        menyamping \qty{1}{\micro m} dalam sesekon. Jelaskan kedua angka
        itu dari \omterm{prop:b1:water-small-molecules:properties}{efek hidrofob}.
  \item Jelaskan mengapa, dengan laju berbalik seperti itu, kedua lembar
        sebuah membran dapat tetap berbeda susunannya seumur hidup
        sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}.
  \item Nyatakan hasilnya: massa yang dihemat unta itu dengan membawa
        \qty{30}{kg} \omterm{def:b1:lipids:triglyceride}{lemak} alih-alih glikogen yang setara energinya,
        dan putusan tentang punuknya sebagai simpanan air.
\end{enumerate}
\end{problem}
