\chapter{Voedselwebben}\label{ch:g5:food-webs}

\Cref{ch:g3:food-chains} waarschuwde ons dat zijn nette ketens losse
draden waren die uit een net getrokken werden: de buizerd pakt ook
muizen, de hagedis eet ook vliegen, het gras voedt ook konijnen. Dit
jaar houden we op met draden trekken en kijken we naar het hele net ---
het \omterm{def:g5:food-webs:web}{voedselweb} --- en naar wat het in evenwicht houdt.

\section{Van ketens naar webben}

\begin{definition}[Voedselweb]\label{def:g5:food-webs:web}
Het \emph{voedselweb}\index{voedselweb} van een \omterm{def:g2:where-animals-live:habitat}{leefgebied} is het
netwerk van \emph{alle} \omterm{def:g3:food-chains:chain}{voedselketens} tegelijk: elke \omterm{def:g5:biodiversity:species}{soort} één keer
getekend, met een pijl (die nog steeds \emph{\omterm{def:g3:food-chains:chain}{wordt gegeten door}}
betekent, \cref{rem:g3:food-chains:arrow}) voor elke maaltijdrelatie. De
meeste \omterm{def:g1:animals-around-us:animal}{dieren} eten meerdere \omterm{def:g5:biodiversity:species}{soorten} en worden door meerdere \omterm{def:g5:biodiversity:species}{soorten}
gegeten --- dus kruisen de ketens elkaar, en wordt de tekening een web.
\end{definition}

\begin{example}[Het web lezen]\label{ex:g5:food-webs:reading}
Twee manieren om de figuur te lezen. Volg de pijlen van het konijn: het
eet gras \emph{en} klaver, en \omterm{def:g3:food-chains:chain}{wordt gegeten door} de vos \emph{en} de
buizerd --- vier relaties, waar een keten er twee liet zien. En tel de
voedselpijlen van de buizerd: hagedis en konijn --- valt er een weg, dan
blijft de ander. Het web legt precies de reservekeuzes vast die
\cref{prop:g5:biodiversity:web} prees.
\end{example}

\section{Rollen in het web}

\begin{definition}[Producenten en consumenten]\label{def:g5:food-webs:roles}
De \omterm{def:g5:biodiversity:species}{soorten} van het web vallen uiteen in rollen:
\begin{itemize}
  \item de \emph{producenten}\index{producent}: de \omterm{def:g1:plants-around-us:plant}{planten}, die hun
        eigen materie maken
        (\cref{prop:g4:what-plants-need:make}) --- elk pijlenpad begint
        bij een van hen
        (\cref{prop:g3:food-chains:start});
  \item de \emph{consumenten}\index{consument}: de \omterm{def:g1:animals-around-us:animal}{dieren}, die van
        anderen leven --- \omterm{def:g2:what-animals-eat:kinds}{planteneters} die rechtstreeks producenten
        eten, \omterm{def:g2:what-animals-eat:kinds}{vleeseters} die andere consumenten eten, \omterm{def:g2:what-animals-eat:kinds}{alleseters} beide
        (\cref{def:g2:what-animals-eat:kinds});
  \item en, werkend in de schaduw, de
        \emph{afbrekers}\index{afbreker} --- de verteerders uit
        \cref{ex:g2:caring-for-nature:compost}, die zich met dode
        resten voeden en hun stof aan de bodem teruggeven. Een later
        jaar geeft hun een eigen hoofdstuk.
\end{itemize}
\end{definition}

\begin{example}[De figuur indelen]\label{ex:g5:food-webs:sorting}
In het web in zakformaat: \omterm{def:g5:food-webs:roles}{producenten} --- gras, klaver, eik;
plantenetende \omterm{def:g5:food-webs:roles}{consumenten} --- sprinkhaan, konijn, \omterm{ex:g2:animals-grow-up:butterfly}{rups}; vleesetende
\omterm{def:g5:food-webs:roles}{consumenten} --- hagedis, pimpelmees (grotendeels), vos (in werkelijkheid
een \omterm{def:g2:what-animals-eat:kinds}{alleseter}), buizerd, sperwer. Drie verdiepingen: groen,
plantenetend, vleesetend --- en elke bovenste verdieping staat op de
verdiepingen eronder.
\end{example}

\begin{proposition}[De vorm van de aantallen]\label{prop:g5:food-webs:pyramid}
De bewoners van het web verdieping voor verdieping tellen geeft steeds
dezelfde vorm: \emph{veel} \omterm{def:g5:food-webs:roles}{producenten}, \emph{minder} \omterm{def:g2:what-animals-eat:kinds}{planteneters},
\emph{weinig} \omterm{def:g2:what-animals-eat:kinds}{vleeseters} --- een wei met miljoenen grasplanten voedt
duizenden sprinkhanen, die een handvol hagedissen voeden, die één paar
buizerds voeden. Elke verdieping leeft van de verdieping eronder en kan
haar voedsel niet in aantal overtreffen. Daarom zijn grote jagers altijd
zeldzaam.
\end{proposition}
\admitted

\section{Evenwicht, en wat het verstoort}

\begin{proposition}[Het evenwicht van het web]\label{prop:g5:food-webs:balance}
Aan zichzelf overgelaten houdt een web zijn aantallen in ruw
\emph{evenwicht}\index{evenwicht (van een voedselweb)}: als konijnen
zich vermeerderen, eten hun jagers beter en vermeerderen ze zich achter
hen aan, en smelt het overschot weg; als konijnen teruglopen, stappen
hongerige jagers op andere prooi over (de reservepijlen) en herstelt het
aantal konijnen zich. De correcties uit
\cref{prop:g3:food-chains:depend}, die tegelijk langs veel pijlen lopen,
houden het geheel stabiel.
\end{proposition}
\admitted

\begin{example}[De klassieke vergissing]\label{ex:g5:food-webs:mistake}
Boeren voerden ooit campagne om roofvogels uit te roeien ---
``kippendieven''. Waar dat lukte, vermeerderden muizen en woelmuizen,
bevrijd van hun voornaamste jagers, zich tot plagen die veel meer graan
opaten dan de incidentele kip van de haviken waard was. Het slakkenverhaal
van de tuinder (\cref{exo:g3:food-chains:9}), op de schaal van een heel
platteland: haal de bovenste verdieping van een web weg en de verdieping
eronder barst uit haar voegen.
\end{example}

\begin{method}[Een verstoring voorspellen]\label{met:g5:food-webs:predict}
Om te voorspellen wat het weghalen (of toevoegen) van een \omterm{def:g5:biodiversity:species}{soort} doet:
\begin{enumerate}
  \item zoek de \omterm{def:g5:biodiversity:species}{soort} in het web en noem \emph{al} haar pijlen, in en
        uit;
  \item haar voedsel, dat minder opgegeten wordt, neemt meestal toe;
        haar eters, die het armer krijgen, nemen af of stappen op
        andere prooi over;
  \item volg elke verandering nog één stap verder door het web --- de
        gevolgen van de tweede stap verrassen vaak;
  \item kijk welke buren reservepijlen hebben (die vangen de klap op)
        en welke alleen van deze \omterm{def:g5:biodiversity:species}{soort} afhingen (die lopen gevaar).
\end{enumerate}
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:1}
Wat is een \omterm{def:g5:food-webs:web}{voedselweb}, en waarom kruisen de ketens elkaar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:2}
Noem de drie rollen uit \cref{def:g5:food-webs:roles}, met van elk een
voorbeeld uit de figuur (de \omterm{def:g5:food-webs:roles}{afbrekers} uitgezonderd).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:3}
Noem uit de figuur alles wat de pimpelmees eet, en alles wat haar eet.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:4}
Haal uit de figuur twee volledige ketens die de sprinkhaan delen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:5}
Beschrijf de vorm van de aantallen, verdieping voor verdieping. Waarom
kan geen verdieping de verdieping eronder in aantal overtreffen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:6}
Waarom zijn buizerds zeldzaam en sprinkhanen ontelbaar, in één zin?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:7}
Konijnen hebben een goed jaar en vermeerderen zich. Wat gebeurt er
volgens \cref{prop:g5:food-webs:balance} daarna --- en wat laat het
overschot wegsmelten?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:8}
Probeer het: teken het \omterm{def:g5:food-webs:web}{voedselweb} van een \omterm{def:g2:where-animals-live:habitat}{leefgebied} dat je kent ---
tuin, park, vijver --- met minstens zes \omterm{def:g5:biodiversity:species}{soorten} en elke pijl die je kunt
verdedigen. Kleur de \omterm{def:g5:food-webs:roles}{producenten} groen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:food-webs:9}
Laat \cref{met:g5:food-webs:predict} lopen over het weghalen van de
hagedis uit de figuur: eerst de gevolgen van de eerste stap, dan nog één
stap verder.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:food-webs:10}
Vertel \cref{ex:g5:food-webs:mistake} na als een reeks voorspellingen:
haviken weg, en dan, en dan? Welke stap hebben de campagnevoerders niet
voorspeld?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:food-webs:11}
Vergelijk in het web van de figuur het weghalen van de eik (een
\omterm{def:g5:food-webs:roles}{producent}) met het weghalen van de sperwer (een toproofdier). Welke
buren voelen elk verlies het eerst, in welke richting reist elke
verstoring, en welk verlies vangt het web hier gemakkelijker op? Gebruik
\cref{ex:g3:food-chains:dry,ex:g3:food-chains:hunter}.
\end{exercise}
