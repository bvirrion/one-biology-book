\chapter{Les réseaux alimentaires}\label{ch:g5:food-webs}

\Cref{ch:g3:food-chains} nous avait prévenus~: ses jolies chaînes
étaient des fils isolés tirés d'un filet --- la buse prend aussi des
souris, le lézard mange aussi des mouches, l'herbe nourrit également
les lapins. Cette année, nous cessons de tirer des fils et nous
regardons le filet entier --- le \omterm{def:g5:food-webs:web}{réseau alimentaire} --- et ce qui le
tient en équilibre.

\section{Des chaînes aux réseaux}

\begin{definition}[Réseau alimentaire]\label{def:g5:food-webs:web}
Le \emph{réseau alimentaire}\index{réseau alimentaire} d'un milieu
est l'ensemble de \emph{toutes} ses \omterm{def:g3:food-chains:chain}{chaînes alimentaires} à la fois~:
chaque \omterm{def:g5:biodiversity:species}{espèce} dessinée une fois, avec une flèche (qui veut toujours
dire \emph{\omterm{def:g3:food-chains:chain}{est mangé par}},
\cref{rem:g3:food-chains:arrow}) pour chaque relation de repas. La
plupart des \omterm{def:g1:animals-around-us:animal}{animaux} mangent plusieurs \omterm{def:g5:biodiversity:species}{espèces} et sont mangés par
plusieurs --- les chaînes se croisent donc, et le dessin devient un
réseau.
\end{definition}

\begin{example}[Lire le réseau]\label{ex:g5:food-webs:reading}
Deux lectures de la figure. Suis les flèches du lapin~: il mange de
l'herbe \emph{et} du trèfle, et il \omterm{def:g3:food-chains:chain}{est mangé par} le renard \emph{et}
par la buse --- quatre relations, là où une chaîne en montrait deux.
Et compte les flèches de nourriture de la buse~: le lézard et le
lapin --- si l'un manque, l'autre reste. Le réseau enregistre
exactement les solutions de rechange que
\cref{prop:g5:biodiversity:web} vantait.
\end{example}

\section{Les rôles dans le réseau}

\begin{definition}[Producteurs et consommateurs]\label{def:g5:food-webs:roles}
Les \omterm{def:g5:biodiversity:species}{espèces} du réseau se rangent en rôles~:
\begin{itemize}
  \item les \emph{producteurs}\index{producteur}~: les \omterm{def:g1:plants-around-us:plant}{plantes}, qui
        fabriquent leur propre matière
        (\cref{prop:g4:what-plants-need:make}) --- tout chemin de
        flèches part de l'un d'eux
        (\cref{prop:g3:food-chains:start})~;
  \item les \emph{consommateurs}\index{consommateur}~: les \omterm{def:g1:animals-around-us:animal}{animaux},
        qui vivent des autres --- les \omterm{def:g2:what-animals-eat:kinds}{herbivores} mangeant
        directement des producteurs, les \omterm{def:g2:what-animals-eat:kinds}{carnivores} mangeant
        d'autres consommateurs, les \omterm{def:g2:what-animals-eat:kinds}{omnivores} les deux
        (\cref{def:g2:what-animals-eat:kinds})~;
  \item et, travaillant dans l'ombre, les
        \emph{décomposeurs}\index{décomposeur} --- les artisans de
        la décomposition de
        \cref{ex:g2:caring-for-nature:compost}, qui se nourrissent
        des restes morts et rendent leur substance au sol. Une
        année ultérieure leur consacre un chapitre entier.
\end{itemize}
\end{definition}

\begin{example}[Trier la figure]\label{ex:g5:food-webs:sorting}
Dans le réseau de poche~: \omterm{def:g5:food-webs:roles}{producteurs} --- herbe, trèfle, chêne~;
\omterm{def:g5:food-webs:roles}{consommateurs} \omterm{def:g2:what-animals-eat:kinds}{herbivores} --- sauterelle, lapin, \omterm{ex:g2:animals-grow-up:butterfly}{chenille}~;
\omterm{def:g5:food-webs:roles}{consommateurs} \omterm{def:g2:what-animals-eat:kinds}{carnivores} --- lézard, mésange bleue (surtout), renard
(en vérité \omterm{def:g2:what-animals-eat:kinds}{omnivore}), buse, épervier. Trois étages~: le vert, les
mangeurs de \omterm{def:g1:plants-around-us:plant}{plantes}, les mangeurs de viande --- chaque étage
supérieur reposant sur ceux du dessous.
\end{example}

\begin{proposition}[La forme des nombres]\label{prop:g5:food-webs:pyramid}
Compter les habitants du réseau étage par étage donne toujours la
même forme~: \emph{beaucoup} de \omterm{def:g5:food-webs:roles}{producteurs}, \emph{moins}
d'\omterm{def:g2:what-animals-eat:kinds}{herbivores}, \emph{peu} de \omterm{def:g2:what-animals-eat:kinds}{carnivores} --- un pré de millions de
brins d'herbe nourrit des milliers de sauterelles, qui nourrissent
une poignée de lézards, qui nourrissent un couple de buses. Chaque
étage vit de celui du dessous et ne peut pas être plus nombreux que
sa nourriture. Voilà pourquoi les grands chasseurs sont toujours
rares.
\end{proposition}
\admitted

\section{L'équilibre, et ce qui le dérange}

\begin{proposition}[L'équilibre du réseau]\label{prop:g5:food-webs:balance}
Laissé à lui-même, un réseau maintient ses effectifs en
\emph{équilibre}\index{équilibre (d'un réseau alimentaire)}
approximatif~: si les lapins se multiplient, leurs chasseurs mangent
mieux et se multiplient à leur suite, et le surplus fond~; si les
lapins s'amenuisent, les chasseurs affamés se tournent vers d'autres
proies (les flèches de rechange) et les effectifs de lapins se
rétablissent. Les corrections de
\cref{prop:g3:food-chains:depend}, courant sur de nombreuses flèches
à la fois, stabilisent l'ensemble.
\end{proposition}
\admitted

\begin{example}[L'erreur classique]\label{ex:g5:food-webs:mistake}
Les agriculteurs ont autrefois fait campagne pour détruire les
rapaces --- ces «~voleurs de poules~». Là où ils ont réussi, souris
et campagnols, débarrassés de leurs principaux chasseurs, se sont
multipliés en fléaux qui ont mangé bien plus de grain que ne valait
la poule occasionnelle des buses. C'est l'histoire des escargots du
jardinier (\cref{exo:g3:food-chains:9}) à l'échelle de toute une
campagne~: enlève l'étage supérieur d'un réseau et celui du dessous
explose.
\end{example}

\begin{method}[Prévoir un dérangement]\label{met:g5:food-webs:predict}
Pour prévoir l'effet du retrait (ou de l'ajout) d'une \omterm{def:g5:biodiversity:species}{espèce}~:
\begin{enumerate}
  \item trouve l'\omterm{def:g5:biodiversity:species}{espèce} dans le réseau et énumère \emph{toutes} ses
        flèches, entrantes et sortantes~;
  \item ses nourritures, moins mangées, tendent à augmenter~; ses
        mangeurs, appauvris, tendent à diminuer ou à changer de
        proie~;
  \item suis chaque changement d'un pas de plus le long du réseau
        --- les effets de second pas surprennent souvent~;
  \item repère quels voisins ont des flèches de rechange (ils
        encaissent le choc) et lesquels dépendaient de cette seule
        \omterm{def:g5:biodiversity:species}{espèce} (ils sont en danger).
\end{enumerate}
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:1}
Qu'est-ce qu'un \omterm{def:g5:food-webs:web}{réseau alimentaire}, et pourquoi les chaînes se
croisent-elles~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:2}
Nomme les trois rôles de \cref{def:g5:food-webs:roles}, avec un
exemple de chacun tiré de la figure (les \omterm{def:g5:food-webs:roles}{décomposeurs} exceptés).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:3}
D'après la figure~: énumère tout ce que mange la mésange bleue, et
tout ce qui la mange.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:4}
Extrais de la figure deux chaînes complètes qui ont la sauterelle en
commun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:5}
Énonce la forme des nombres, étage par étage. Pourquoi aucun étage
ne peut-il être plus nombreux que celui du dessous~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:6}
Pourquoi les buses sont-elles rares et les sauterelles innombrables,
en une phrase~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:7}
Les lapins connaissent une bonne année et se multiplient. D'après
\cref{prop:g5:food-webs:balance}, que se passe-t-il ensuite --- et
qu'est-ce qui fait fondre le surplus~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:8}
À toi d'essayer~: dessine le \omterm{def:g5:food-webs:web}{réseau alimentaire} d'un milieu que tu
connais --- jardin, parc, mare --- avec au moins six \omterm{def:g5:biodiversity:species}{espèces} et
toutes les flèches que tu peux défendre. Marque les \omterm{def:g5:food-webs:roles}{producteurs} en
vert.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:food-webs:9}
Applique \cref{met:g5:food-webs:predict} au retrait du lézard dans
la figure~: effets du premier pas, puis un pas de plus.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:food-webs:10}
Raconte \cref{ex:g5:food-webs:mistake} comme une suite de
prévisions~: les rapaces retirés, puis quoi, puis quoi~? Quelle
étape les militants n'ont-ils pas su prévoir~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:food-webs:11}
Dans le réseau de la figure, compare le retrait du chêne (un
\omterm{def:g5:food-webs:roles}{producteur}) à celui de l'épervier (un chasseur du sommet). Quels
voisins ressentent chaque perte en premier, dans quel sens voyage
chaque dérangement, et quelle perte le réseau encaisse-t-il ici le
plus facilement~? Sers-toi de
\cref{ex:g3:food-chains:dry,ex:g3:food-chains:hunter}.
\end{exercise}
