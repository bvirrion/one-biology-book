\chapter{Planten zonder zaden}\label{ch:g4:plants-without-seeds}

De schaduwkant van de oude muur is bekleed met mos; onder de beuken
rollen \omterm{prop:g4:plants-without-seeds:damp}{varens} hun grote groene \omterm{def:g1:plants-around-us:parts}{bladeren} uit. Zoek ze de hele zomer af:
je vindt er geen \omterm{def:g3:flowers-fruits-seeds:parts}{bloem}, geen \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht}, geen \omterm{def:g2:from-seed-to-plant:seed}{zaad} --- nooit. En toch
verspreiden mos en \omterm{prop:g4:plants-without-seeds:damp}{varens} zich uitstekend. \omterm{def:g1:plants-around-us:plant}{Planten} blijken meer dan één
manier te hebben om de volgende generatie te maken.

\section{Varens en mossen: voortplanting met sporen}

\begin{definition}[Spore]\label{def:g4:plants-without-seeds:spore}
Een \emph{spore}\index{spore} is een microscopisch korreltje, in
enorme aantallen gemaakt, waaruit een nieuwe \omterm{def:g1:plants-around-us:plant}{plant} kan groeien. Anders
dan een \omterm{def:g2:from-seed-to-plant:seed}{zaad} (\cref{def:g2:from-seed-to-plant:seed}) is een spore één
enkel stipje \emph{zonder \omterm{def:g3:animals-through-seasons:active}{voedselvoorraad} en zonder kant-en-klaar
plantje erin} --- een begin in zijn kaalste vorm.
\end{definition}

\begin{example}[Onder de bladeren van de varen]\label{ex:g4:plants-without-seeds:fern}
Draai in de late zomer een varenblad om: nette rijen bruine stippen
staan aan de onderkant. Elke stip is een pakketje met duizenden \omterm{def:g4:plants-without-seeds:spore}{sporen};
rijp splijten de pakketjes open, en de \omterm{def:g4:plants-without-seeds:spore}{sporen} zweven weg op de lucht
als het fijnste stof. Een \omterm{def:g4:plants-without-seeds:spore}{spore} die op vochtige, schaduwrijke grond
landt, kan --- in stappen --- tot een nieuwe varen uitgroeien.
\end{example}

\begin{proposition}[Sporenplanten hebben vocht nodig]\label{prop:g4:plants-without-seeds:damp}
Sporenplanten --- \emph{mossen}\index{mos} en \emph{varens}\index{varen}
--- kunnen zich alleen voortplanten waar het \emph{vochtig} is: hun
\omterm{def:g4:plants-without-seeds:spore}{sporen}, en de tere stadia die eruit groeien, drogen uit en gaan dood in
droge lucht. Daarom bedekt mos de schaduwrijke noordmuur maar nooit de
zonnige zuidmuur, en daarom verdringen \omterm{prop:g4:plants-without-seeds:damp}{varens} zich in vochtige bossen,
op beekoevers en bij putten.
\end{proposition}
\admitted

\begin{remark}[Een spore is geen zaad]\label{rem:g4:plants-without-seeds:sporeseed}
Allebei zijn het beginnetjes die reizen, maar het \omterm{def:g2:from-seed-to-plant:seed}{zaad} is een
overlevingspakket --- schil, \omterm{def:g3:animals-through-seasons:active}{voedselvoorraad}, kant-en-klaar plantje ---
terwijl de \omterm{def:g4:plants-without-seeds:spore}{spore} kaal is. Vandaar de aantallen: de miljoenen \omterm{def:g4:plants-without-seeds:spore}{sporen} van
de varen tegenover de twaalf \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} van de bonenpeul. Kale beginnetjes
zijn goedkoop te maken en makkelijk te verliezen; de twee plannen echoën
de kabeljauw en de olifant uit
\cref{prop:g4:animal-reproduction:strategies}.
\end{remark}

\section{Nieuwe planten zonder ook maar te zaaien}

\begin{definition}[Vegetatieve voortplanting]\label{def:g4:plants-without-seeds:vegetative}
Veel \omterm{def:g1:plants-around-us:plant}{planten} kunnen ook nieuwe \omterm{def:g1:plants-around-us:plant}{planten} maken \emph{rechtstreeks uit een
stuk van zichzelf} --- geen \omterm{def:g4:plants-without-seeds:spore}{sporen}, geen \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}, maar één ouder. Dat is
\emph{vegetatieve voortplanting}\index{vegetatieve voortplanting}. De
nieuwe \omterm{def:g1:plants-around-us:plant}{plant} is een levende voortzetting van de oude, en lijkt er
precies op.
\end{definition}

\begin{example}[De uitlopers van de aardbei]\label{ex:g4:plants-without-seeds:runner}
Een aardbeienplant stuurt lange dunne horizontale \omterm{def:g1:plants-around-us:parts}{stengels} ---
\emph{uitlopers}\index{uitloper} --- over de grond. Waar een uitloper
de grond raakt, wortelt hij, krijgt hij blaadjes en wordt hij een
volledige nieuwe aardbeienplant; de uitloper verdort daarna en het kind
staat vrij. Eén \omterm{def:g1:plants-around-us:plant}{plant} wordt een heel bed, zonder dat er één zaadje
gezaaid is.
\end{example}

\begin{example}[Bollen en knollen]\label{ex:g4:plants-without-seeds:bulbtuber}
De \emph{bol}\index{bol} van de tulp en de \emph{knol}\index{knol} van
de aardappel zijn ondergrondse voorraadkamers --- én startpakketten. In
de grond gelaten splitst een \omterm{ex:g4:plants-without-seeds:bulbtuber}{bol} zich in dochterbollen, elk een
toekomstige tulp; elk ``\omterm{def:g1:the-five-senses:sense}{oog}'' van een uitlopende aardappel kan tot een
hele nieuwe \omterm{def:g1:plants-around-us:plant}{plant} uitgroeien. Tuinders poten aardappelen, geen
aardappelzaden.
\end{example}

\begin{method}[Een stek nemen]\label{met:g4:plants-without-seeds:cutting}
Tuinders bootsen de \omterm{def:g4:plants-without-seeds:vegetative}{vegetatieve voortplanting} na met
\emph{stekken}\index{stek}:
\begin{enumerate}
  \item snijd een gezonde jonge scheut van een handlengte van de
        moederplant (munt, wilg en geraniums lukken goed);
  \item haal de onderste blaadjes eraf en zet de scheut in water of
        vochtige aarde;
  \item wacht: in dagen of weken groeien er \omterm{def:g1:plants-around-us:parts}{wortels} uit de snee;
  \item \omterm{def:g1:plants-around-us:plant}{plant} hem uit --- een volledige nieuwe \omterm{def:g1:plants-around-us:plant}{plant}, identiek aan zijn
        ene ouder.
\end{enumerate}
\end{method}

\begin{remark}[Eén ouder, exacte kopieën]\label{rem:g4:plants-without-seeds:copy}
\omterm{prop:g3:flowers-fruits-seeds:transform}{Zaden} mengen twee ouders
(\cref{rem:g4:animal-reproduction:mix} geldt ook voor \omterm{def:g1:plants-around-us:plant}{planten}, via het
\omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel}); \omterm{def:g4:plants-without-seeds:vegetative}{vegetatieve voortplanting} kopieert er één. Daarom wordt elke
\omterm{def:g1:plants-around-us:plant}{plant} van een benoemd appelras uit \omterm{met:g4:plants-without-seeds:cutting}{stekken} en entingen gekweekt, nooit
uit pitjes: de boomgaard wil exacte kopieën van één uitstekende boom. De
prijs van kopiëren is eenvormigheid --- een boomgaard van identieke
bomen deelt identieke zwakke plekken, zoals een later jaar zal laten zien.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:1}
Noem twee \omterm{def:g1:plants-around-us:plant}{planten} die nooit \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} maken. Hoe \omterm{def:g1:plants-around-us:plant}{planten} ze zich voort?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:2}
Geef twee verschillen tussen een \omterm{def:g4:plants-without-seeds:spore}{spore} en een \omterm{def:g2:from-seed-to-plant:seed}{zaad}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:3}
Waar worden de \omterm{def:g4:plants-without-seeds:spore}{sporen} van de varen gemaakt? Beschrijf wat je in de late
zomer onder een varenblad vindt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:4}
Waarom groeit mos op de schaduwrijke noordmuur en niet op de zonnige
zuidmuur?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:5}
Wat is \omterm{def:g4:plants-without-seeds:vegetative}{vegetatieve voortplanting}? Hoeveel ouders zijn ervoor nodig?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:6}
Vertel het verhaal van een aardbeienuitloper, van de eerste \omterm{def:g1:plants-around-us:parts}{stengel} tot
het vrijstaande kindplantje.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:7}
Waarom poten tuinders aardappelen in plaats van aardappelzaden te
zaaien? Wat kan elk ``\omterm{def:g1:the-five-senses:sense}{oog}'' doen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:8}
Probeer het: neem een muntstek volgens
\cref{met:g4:plants-without-seeds:cutting} en zet hem in een glas water
op de vensterbank. Wat verschijnt er, en ongeveer na hoe lang?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:9}
De varen maakt miljoenen \omterm{def:g4:plants-without-seeds:spore}{sporen}; de bonenpeul een stuk of twaalf \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}.
Verklaar het verschil in aantallen met
\cref{rem:g4:plants-without-seeds:sporeseed} --- en noem het dierenpaar
waarvan hun strategieën de echo zijn.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:10}
Een tuinierster ziet dat haar nieuwe aardbeienplanten identiek zijn aan
de moederplant, terwijl haar bonenplanten een beetje van hun ouders
verschillen. Verklaar allebei de waarnemingen met
\cref{rem:g4:plants-without-seeds:copy}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:plants-without-seeds:11}
Een wilgentak die door een winterse overstroming is afgebroken,
stroomafwaarts meegevoerd en op een modderbank gestrand, groeit vaak
uit tot een nieuwe wilg. Wat voor voortplanting is dat --- en waarom
staan er langs rivieren zo vaak wilgen? Wat deed de overstroming dat de
tuinder uit \cref{met:g4:plants-without-seeds:cutting} met een schaar doet?
\end{exercise}
