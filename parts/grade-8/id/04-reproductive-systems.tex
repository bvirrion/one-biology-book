\chapter{Sistem Reproduksi}\label{ch:g8:reproductive-systems}

\Cref{ch:g8:sexual-reproduction} membutuhkan \omterm{def:g8:sexual-reproduction:gametes}{gamet};
\cref{ch:g8:puberty} menyalakan pabriknya. Bab ini memetakan
pabriknya sendiri: sistem reproduksi laki-laki dan
perempuan --- organnya, \omterm{def:g8:sexual-reproduction:gametes}{gamet} yang dihasilkan
masing-masing, dan irama bulanan sistem perempuan yang
menakjubkan. Anatomi biasa, dinamai secara biasa: ini organ
seperti \omterm{def:g4:heart-and-blood:heart}{jantung} dan \omterm{def:g7:how-animals-breathe:four}{paru-paru}, dan inilah babnya.

\section{Sistem laki-laki}

\begin{definition}[Sistem reproduksi laki-laki]\label{def:g8:reproductive-systems:male}
Organ pada sistem laki-laki:
\begin{itemize}
  \item kedua \emph{buah zakar}\index{buah zakar}, yang
        ditahan di luar kehangatan tubuh di dalam
        \emph{skrotum} --- pabrik \omterm{def:g8:sexual-reproduction:gametes}{gametnya}: sejak \omterm{def:g8:puberty:puberty}{pubertas}
        ia menghasilkan \emph{sel sperma}\index{sel sperma}
        tanpa henti, ratusan juta banyaknya;
  \item saluran yang menyimpan lalu mengangkut spermanya,
        disertai di jalan oleh kelenjar yang cairannya
        memberinya makan dan membawanya --- bersama-sama
        membentuk \emph{air mani};
  \item \emph{penis}\index{penis}, tempat air mani
        meninggalkan tubuhnya --- organ yang sama, pada
        tugasnya yang lain, mengangkut \omterm{def:g7:removing-wastes:kidneys}{urine}
        (\cref{def:g7:removing-wastes:kidneys}); sebuah
        susunan \omterm{def:g7:heart-and-circulation:rooms}{katup} menjaga kedua tugas itu tetap
        terpisah tegas.
\end{itemize}
\end{definition}

\begin{example}[Sel sperma, dari dekat]\label{ex:g8:reproductive-systems:sperm}
Di bawah \omterm{def:g5:first-look-at-cells:microscope}{mikroskop}
(\cref{def:g5:first-look-at-cells:microscope}), \omterm{def:g8:reproductive-systems:male}{sel sperma}
adalah \omterm{def:g5:first-look-at-cells:cell}{sel} yang ditelanjangi untuk perjalanan: kepala yang
hampir seluruhnya \omterm{def:g6:cell-unit-of-life:parts}{inti sel} --- sumbangan ayahnya, dikemas
(\cref{rem:g5:first-look-at-cells:nucleus}) --- dan ekor
yang mencambuk. \omterm{def:g5:first-look-at-cells:cell}{Sel} terkecil di dalam tubuh, dibangun untuk
berenang, dibekali untuk beberapa hari. Produksinya berjalan
dari \omterm{def:g8:puberty:puberty}{pubertas} sampai \omterm{def:g3:stages-of-life:stages}{usia tua}: pabrik laki-lakinya tanpa henti.
\end{example}

\section{Sistem perempuan}

\begin{definition}[Sistem reproduksi perempuan]\label{def:g8:reproductive-systems:female}
Organ pada sistem perempuan, semuanya di dalam perut bawah:
\begin{itemize}
  \item kedua \emph{indung telur}\index{indung telur} ---
        gudang sekaligus pabrik \omterm{def:g8:sexual-reproduction:gametes}{gametnya}: tidak seperti buah
        zakar, ia menyimpan persediaan calon \emph{\omterm{def:g4:animal-reproduction:sexual}{sel telur}}
        sejak lahir, lalu sejak \omterm{def:g8:puberty:puberty}{pubertas} melepaskannya satu
        per satu;
  \item kedua \emph{saluran telur}\index{saluran telur},
        pipa berumbai yang menangkap \omterm{def:g4:animal-reproduction:sexual}{sel telur} yang
        dilepaskan --- tempat pertemuan yang biasa pada
        langkah 2 \cref{prop:g8:sexual-reproduction:core};
  \item \emph{\omterm{prop:g5:human-reproduction:plan}{rahim}} (kandungan pada
        \cref{prop:g5:human-reproduction:plan}), yang
        dinding berototnya yang tebal serta lapisan dalamnya
        yang dapat diperbarui menjadi penginapan \omterm{prop:g8:sexual-reproduction:core}{zigotnya};
  \item \emph{vagina}, yang menghubungkan \omterm{prop:g5:human-reproduction:plan}{rahim} dengan dunia
        luar --- tempat air mani diterima, dan jalan lahir
        pada \cref{ex:g5:human-reproduction:birth}.
\end{itemize}
\end{definition}

\begin{example}[Sel telur, dari dekat]\label{ex:g8:reproductive-systems:egg}
\omterm{def:g4:animal-reproduction:sexual}{Sel telur} adalah kebalikan \omterm{def:g8:reproductive-systems:male}{sel sperma} dalam segala hal
kecuali perannya: \omterm{def:g5:first-look-at-cells:cell}{sel} \emph{terbesar} di dalam tubuh ---
sepersepuluh milimeter, di ambang batas \omterm{def:g1:the-five-senses:sense}{mata} --- bulat, tak
bergerak, dipenuhi bekal untuk hari pertama perjalanannya
(asas kuning telur pada \cref{rem:g2:animals-grow-up:egg},
pada skala \omterm{def:g5:first-look-at-cells:cell}{sel}). Satu pabrik menjalankan jutaan tanpa henti;
yang lain melepaskan satu \omterm{def:g5:first-look-at-cells:cell}{sel} berbekal sebulan sekali:
hitungan banyak-melawan-rawatan pada
\cref{prop:g4:animal-reproduction:strategies}, tertulis di
dalam \omterm{def:g8:sexual-reproduction:gametes}{gametnya} sendiri.
\end{example}

\section{Daurnya}

\begin{proposition}[Daur haid]\label{prop:g8:reproductive-systems:cycle}
Dari \omterm{def:g8:puberty:puberty}{pubertas} sampai kira-kira lima puluh tahun, sistem
perempuan menjalankan sebuah program berulang selama
kira-kira \emph{28 hari} --- yaitu \emph{daur
haid}\index{daur haid}, yang diperintah oleh rantai \omterm{def:g8:puberty:hormones}{hormon}
pada \cref{def:g8:puberty:hormones}:
\begin{enumerate}
  \item lapisan dalam \omterm{prop:g5:human-reproduction:plan}{rahimnya} membangun ulang dirinya,
        tebal dan kaya \omterm{prop:g4:journey-of-food:absorb}{darah} --- penginapannya disiapkan;
  \item di pertengahan daurnya, sekitar hari ke-14, sebuah
        \omterm{def:g8:reproductive-systems:female}{indung telur} melepaskan satu \omterm{def:g4:animal-reproduction:sexual}{sel telur} ---
        \emph{ovulasi}\index{ovulasi}; \omterm{def:g8:reproductive-systems:female}{saluran telurnya}
        menangkapnya, dan sekitar sehari ia dapat dibuahi;
  \item tanpa \omterm{def:g5:flower-to-fruit:steps}{pembuahan}: lapisan dalamnya tidak dibutuhkan;
        ia luruh lalu keluar lewat vagina selama beberapa
        hari --- itulah \emph{haid}\index{haid} --- lalu
        programnya mulai lagi;
  \item dengan \omterm{def:g5:flower-to-fruit:steps}{pembuahan}: \omterm{prop:g8:sexual-reproduction:core}{zigotnya} menetap di dalam lapisan
        yang sudah disiapkan itu
        (\cref{def:g5:human-reproduction:pregnancy}),
        daurnya tertunda, dan \omterm{def:g5:human-reproduction:pregnancy}{kehamilan} dimulai.
\end{enumerate}
\end{proposition}
\admitted

\begin{remark}[Hidup bersama daurnya]\label{rem:g8:reproductive-systems:living}
Hal praktisnya, dikatakan apa adanya: daurnya tidak teratur
pada tahun-tahun pertamanya --- permesinannya sedang
menyetel diri; panjangnya berbeda dari orang ke orang dan
dari bulan ke bulan; \omterm{prop:g8:reproductive-systems:cycle}{haid} dapat membawa kram (\omterm{prop:g5:human-reproduction:plan}{rahimnya}
adalah \omterm{def:g3:bones-and-muscles:muscle}{otot}, \cref{def:g3:bones-and-muscles:muscle}, dan ia
bekerja selama peluruhannya) --- kehangatan, gerak dan, bila
perlu, nasihat perawat sekolahnya sama-sama menolong; dan
pembalut, tampon serta cawan hanyalah perlengkapan praktis
\omterm{prop:g8:reproductive-systems:cycle}{haidnya}, satu lorong biasa di setiap apotek. Tidak satu pun
dari ini penyakit: daurnya adalah sistem yang berjalan
persis seperti yang dibangun.
\end{remark}

\begin{method}[Membaca kedua sistem berdampingan]\label{met:g8:reproductive-systems:sides}
Agar petanya tetap lurus, bandingkanlah menurut fungsinya:
\begin{enumerate}
  \item pabrik \omterm{def:g8:sexual-reproduction:gametes}{gametnya}: buah zakar --- \omterm{def:g8:reproductive-systems:female}{indung telur};
  \item \omterm{def:g8:sexual-reproduction:gametes}{gametnya}: jutaan perenang tanpa henti --- satu \omterm{def:g4:animal-reproduction:sexual}{sel
        telur} berbekal sebulan sekali;
  \item tempat pertemuannya: \omterm{def:g8:reproductive-systems:female}{saluran telurnya}, dicapai lewat
        renang spermanya dari vagina menembus \omterm{prop:g5:human-reproduction:plan}{rahim};
  \item penginapannya: tidak ada pada peta laki-lakinya ---
        \omterm{prop:g5:human-reproduction:plan}{rahimnya}, disiapkan tiap bulan, pada peta perempuan;
  \item jadwalnya: tanpa henti sejak \omterm{def:g8:puberty:puberty}{pubertas} --- berdaur,
        dari \omterm{def:g8:puberty:puberty}{pubertas} sampai kira-kira lima puluh tahun.
\end{enumerate}
\end{method}

\section{Soal: Model Peraga Kliniknya}

\begin{problem}[{Soal akhir pekan --- menjelaskan kedua
sistem dengan dua model dan sebuah penanggalan}]\label{pb:g8:reproductive-systems:1}
Hari terbuka sebuah pusat kesehatan memakai tiga alat
peraga: model sistem laki-laki, model sistem perempuan, dan
sebuah penanggalan daur 28 hari yang besar. Kamulah
penjelas sukarelawannya.

\medskip\noindent\textbf{Bagian I --- Kedua modelnya.}
\begin{enumerate}
  \item Seorang pengunjung menelusuri model laki-lakinya
        dari buah zakar sampai pintu keluarnya. Ceritakan
        rutenya, sebutkan apa yang ditambahkan di jalan dan
        susunan pada organ yang bertugas ganda itu.
  \item Pada model perempuannya, tunjukkan tempat persediaan
        \omterm{def:g8:sexual-reproduction:gametes}{gametnya} menunggu, tempat \omterm{def:g5:flower-to-fruit:steps}{pembuahan} biasanya
        terjadi, dan tempat sebuah \omterm{def:g5:human-reproduction:pregnancy}{kehamilan} menginap.
  \item Seorang pengunjung bertanya mengapa pabrik \omterm{def:g8:sexual-reproduction:gametes}{gamet}
        kedua modelnya berbeda begitu tajam --- jutaan
        sehari melawan satu sebulan. Jawablah dengan kalimat
        penutup \cref{ex:g8:reproductive-systems:egg}.
  \item Yang lain bertanya apa yang dikerjakan kedua
        sistemnya sebelum \omterm{def:g8:puberty:puberty}{pubertas}. Jawablah dengan
        permesinan pada \cref{ch:g8:puberty} dalam satu baris.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Penanggalannya.}
\begin{enumerate}[resume]
  \item Telusurilah penanggalannya: tempatkan \omterm{prop:g8:reproductive-systems:cycle}{haidnya},
        pembangunan ulangnya, \omterm{prop:g8:reproductive-systems:cycle}{ovulasi} dan hari suburnya,
        beserta tanggal kasarnya.
  \item Seorang pengunjung menganggap hari ke-28 sama bagi
        semua orang. Betulkan anggapan itu dengan
        \cref{rem:g8:reproductive-systems:living} --- dua
        keberubahan.
  \item Perlihatkan pada penanggalannya apa yang berubah
        bila \omterm{def:g5:flower-to-fruit:steps}{pembuahan} terjadi di dekat hari ke-14 ---
        peristiwa mana yang dibatalkan, mana yang dimulai
        (\cref{def:g5:human-reproduction:pregnancy}
        melanjutkan ceritanya).
  \item Seorang \omterm{def:g3:stages-of-life:stages}{remaja} bertanya, dengan pelan, apakah kram
        berarti ada yang salah. Berikan jawaban \omterm{def:g3:bones-and-muscles:muscle}{ototnya} dan
        kedua penolongnya --- dan alamat bagi apa pun yang
        melampauinya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Keterampilan
penjelasnya.}
\begin{enumerate}[resume]
  \item Pengunjungnya cekikikan melihat modelnya. Kalimat
        pembukamu memperlakukan kedua sistemnya seperti
        alinea pertama \cref{ch:g8:reproductive-systems}
        memperlakukannya. Tuliskanlah.
  \item Ringkaslah kedua sistemnya dengan perbandingan lima
        baris pada \cref{met:g8:reproductive-systems:sides},
        dari ingatan.
  \item Seorang pengunjung bertanya di mana permulaan
        bayinya cocok pada alat peraganya. Rangkaikan ketiga
        alat peraganya menjadi ketiga langkah pada
        \cref{prop:g8:sexual-reproduction:core}.
  \item Tutuplah hari terbukanya dengan satu kalimat tentang
        mengapa pusat kesehatan mengajarkan anatomi ini sama
        sekali --- pengetahuan melawan dongeng, seperti yang
        dibantahkan \cref{ex:g8:puberty:questions}.
\end{enumerate}
\end{problem}
