\chapter{الجمهرات والديموغرافيا}\label{ch:b1:populations}

دورق خميرة يبدأ بمليون \omterm{def:b1:cell-unit-of-life:cell}{خلية} فيحمل، بعد ست عشرة ساعة، ستين مليونًا
ويكفّ عن النمو؛ وقطيع أيائل في جزيرة ينمو بالعشر كل سنة حتى الشتاء
الذي ينفد فيه غذاؤه؛ ووشوق الغابة الشمالية ترتفع وتنهار كل عشر
سنوات، متأخرة سنة عن الأرانب التي تأكلها. وكل واحدة \emph{\omterm{def:b1:populations:population}{جمهرة}}
--- أي أفراد نوع واحد في مكان واحد --- وكل واحدة تخضع لحساب يمكن
كتابته. ويصف هذا الفصل كيف تُعدّ \omterm{def:b1:populations:population}{الجمهرات}، وكيف تُجدول ولاداتها
ووفياتها، ومعادلتي النمو، والقوى التي تكبح الأعداد، وما يقوله
الحساب نفسه عن نوعنا نحن.

\section{عدّ جمهرة}

\begin{definition}[الجمهرة والكثافة والتوزع]\label{def:b1:populations:population}
\index{جمهرة}\index{كثافة جمهرية}
\emph{الجمهرة} مجموعة أفراد نوع واحد تعيش في مساحة معطاة في زمن
معطى وتستطيع التزاوج فيما بينها. و\emph{حجمها} $N$ هو عدد الأفراد؛
و\emph{كثافتها} عددهم في وحدة المساحة أو الحجم؛ و\emph{توزعها} نمط
تباعدهم --- \emph{متكتل} حيث تجمعهم الموارد أو الحياة الاجتماعية
(القطعان والأسراب وبقع النبات)، و\emph{منتظم} حيث يتنافرون (الطيور
ذات الحمى، وجنبات الصحراء المتنافسة على الماء)، و\emph{عشوائي} حيث
لا يعمل أي منهما. وللجمهرات كذلك \emph{بنية عمرية}، و\emph{نسبة
جنسية}، ومعدلات ولادة ووفاة وهجرة داخلة وخارجة، ميزانُها هو النمو.
\end{definition}

\begin{method}[تقدير جمهرة]\label{met:b1:populations:counting}
\index{الوسم وإعادة الصيد}
\begin{enumerate}
  \item \omterm{def:b1:organism-environment:organism}{الكائنات} اللاصقة أو البطيئة: عُدّها في \emph{مربعات عيّنة}
        معلومة المساحة موضوعة عشوائيًا، واضرب الكثافة الوسطية في
        المساحة الكلية؛ وتبعثر المربعات يعطي الدقة ويكشف التوزع.
  \item الحيوانات المتحركة: \emph{\omterm{met:b1:populations:counting}{الوسم وإعادة الصيد}}. اصطد $M$
        فردًا، وسِمها وأطلقها؛ ثم اصطد لاحقًا $C$ منها $R$ موسوم.
        فإذا اختلط الموسوم بحرية ولم يتغير شيء، كان كسر الموسوم في
        الصيدة الثانية مساويًا لكسر الموسوم في \omterm{def:b1:populations:population}{الجمهرة}، أي
        $R/C = M/N$، فيكون
        $N \approx MC/R$ (تقدير لنكولن-بيترسن).
  \item تحقق من الفرضيات: لا ولادات ولا وفيات ولا هجرة بين
        الصيدتين، ووسوم لا تسقط ولا تغيّر سلوك الحيوان، وقابلية صيد
        متساوية. وكل إخفاق يحيّز $N$ في اتجاه يمكن التنبؤ به.
  \item العلامات غير المباشرة حيث يستحيل العدّ: البراز، والآثار،
        والنداءات، والأعشاش، \omterm{def:b1:nucleic-acids:chain}{وDNA} في لتر من ماء بركة.
\end{enumerate}
\end{method}

\begin{example}[فئران حقل في مرج]\label{ex:b1:populations:voles}
ستون فأر حقل تُصاد وتُوسم وتُطلق في هكتار من المرج؛ وبعد ثلاث ليال
يُصاد ثمانون منها اثنا عشر تحمل وسومًا:
$N \approx 60\times 80/12 = 400$ فأر، أي أربعة في كل مئة متر مربع.
فإن كانت الفئران الموسومة، وقد تعلمت أن المصائد تحوي طعامًا، تُصاد
أسهل، انتفخ $R$ وقُدّر $N$ أقل من حقيقته؛ وإن كان الوسم قد نفّرها
من المصائد فالعكس.
\end{example}

\section{الديموغرافيا: الولادات والوفيات بحسب العمر}

\begin{definition}[جدول الحياة]\label{def:b1:populations:lifetable}
\index{جدول الحياة}\index{بقيا}
\emph{جدول الحياة} يتابع فوجًا من الأفراد وُلدوا معًا عبر حياتهم.
فهو يسجل لكل عمر $x$ مقدارَ $l_x$، وهو \emph{البقيا} --- أي كسر
الفوج الذي ما يزال حيًا عند العمر $x$ --- ومقدارَ $m_x$، وهو
\emph{الخصوبة} --- أي متوسط عدد البنات المولودات لأنثى عمرها $x$.
ومنهما: الوفيات بين عمرين، $q_x = 1 - l_{x+1}/l_x$؛ و\emph{معدل
التكاثر الصافي} $R_0 = \sum_x l_x m_x$، وهو عدد البنات اللواتي
ستنجبهن أنثى مولودة وسطيًا في حياتها؛ و\emph{زمن الجيل}
$T = \sum_x x\,l_x m_x / R_0$، وهو متوسط عمر أمهات تلك البنات.
\omterm{def:b1:populations:population}{والجمهرة} التي $R_0 > 1$ فيها تنمو، والتي $R_0 < 1$ فيها تتراجع؛
وتُضرب في جيل واحد بالمقدار $R_0$.
\end{definition}

\begin{proposition}[ثلاثة أشكال للبقيا]\label{prop:b1:populations:survivorship}
رسم $\log l_x$ في مقابل العمر يعطي ثلاثة أنماط من المنحنيات.
\emph{النمط I}: أكثر الأفراد يعيشون إلى الشيخوخة ويموتون في مدى
ضيق (الثدييات الكبيرة، والبشر في البلاد الغنية). و\emph{النمط II}:
احتمال موت ثابت في كل عمر، أي خط مستقيم على السلّم اللوغاريتمي
(كثير من الطيور والسحالي والقوارض). و\emph{النمط III}: وفيات هائلة
بين الصغار وعمر طويل للقلة الناجية (المحار، والأسماك، وأكثر
النباتات والحشرات). والشكل يتبع نمط عيش الكائن: فنوع يستثمر قليلًا
في كل واحد من نسل كثير له منحني نمط III؛ ونوع يستثمر كثيرًا في كل
واحد من نسل قليل له النمط I.
\end{proposition}

\begin{example}[جدول حياة لقطيع أيائل]\label{ex:b1:populations:deer}
من ألف خشف يبلغ ست مئة عيد ميلادها الأول، ثم يفقد الناجون عشر
عددهم كل سنة حتى لا يتجاوز التاسعة إلا قليل؛ وتلد الإناث من عمر
سنتين نحو ثمانية أعشار البنت في السنة في عزّها. وجمع $l_x m_x$
يعطي $R_0 \approx 1.5$: فكل أنثى تترك بنتًا ونصفًا، وينمو القطيع
بالنصف في كل جيل مدته نحو أربع سنوات --- أي زيادة أحد عشر في المئة
في السنة حتى يوقفها شيء.
\end{example}

\section{النمو}

\begin{theorem}[النمو الأسّي]\label{thm:b1:populations:exponential}
\index{نمو أسّي}\index{معدل الزيادة الذاتي}
\omterm{def:b1:populations:population}{الجمهرة} التي يكون فيها معدلا الولادة والوفاة للفرد $b$ و$d$ ثابتين
تنمو بمعدل يتناسب مع حجمها،
\[
\frac{\dd N}{\dd t} = rN, \qquad r = b - d, \qquad N(t) = N_0\,e^{rt},
\]
حيث $r$ هو \emph{\omterm{thm:b1:populations:exponential}{معدل الزيادة الذاتي}}. وهي تتضاعف كل
$\ln 2/r$؛ \omterm{def:b1:populations:population}{فجمهرة} عند $r = \qty{0.11}{yr^{-1}}$ تتضاعف في
\qty{6.3}{yr}، وأخرى عند $r = \qty{0.5}{h^{-1}}$ في \qty{83}{min}.
ومن جدول حياة، $r \approx \ln R_0 / T$.
\end{theorem}
\begin{proof}
في فترة قصيرة $\dd t$ تكون الولادات $bN\,\dd t$ والوفيات
$dN\,\dd t$، فيكون $\dd N = (b - d)N\,\dd t$؛ وللمعادلة
$\dd N/\dd t = rN$ الحل $N_0 e^{rt}$ (فمشتق
$e^{rt}$ هو $re^{rt}$). والتضاعف: $e^{rt} = 2$ يعطي $t = \ln 2/r$.
وعلى مدى جيل واحد $T$ تُضرب \omterm{def:b1:populations:population}{الجمهرة} في $R_0 = e^{rT}$،
ومنه $r = \ln R_0/T$.
\end{proof}

\begin{theorem}[النمو اللوجستي]\label{thm:b1:populations:logistic}
\index{نمو لوجستي}\index{سعة حمولية}
إذا هبط معدل النمو للفرد هبوطًا خطيًا مع اقتراب \omterm{def:b1:populations:population}{الجمهرة} من
\emph{\omterm{thm:b1:populations:logistic}{سعة حمولية}} $K$ --- وهي الحجم الذي تستطيع البيئة إعالته ---
كان
\[
\frac{\dd N}{\dd t} = rN\left(1 - \frac{N}{K}\right), \qquad
N(t) = \frac{K}{1 + \dfrac{K - N_0}{N_0}\,e^{-rt}} .
\]
\omterm{thm:b1:populations:exponential}{والنمو أسّي} تقريبًا ما دام $N \ll K$، وأسرع ما يكون عند $N = K/2$
(حيث $\dd N/\dd t = rK/4$)، ويتلاشى عند $N = K$، التي تقاربها
\omterm{def:b1:populations:population}{الجمهرة} هضبةً: أي منحنٍ على شكل S.
\end{theorem}
\begin{proof}
اكتب المعادلة $\dd N/[N(1 - N/K)] = r\,\dd t$ واقسم الطرف
الأيسر إلى $\dd N/N + \dd N/(K - N)$؛ فيعطي التكامل
$\ln[N/(K - N)] = rt + \text{ثابت}$، وحلّ $N$ مع $N(0) =
N_0$ يعطي الصيغة. واشتقاق $rN(1 - N/K)$ بالنسبة إلى
$N$ ومساواته بالصفر يعطي الأعظمية عند $N = K/2$.
\end{proof}

\begin{proposition}[التعلق بالكثافة]\label{prop:b1:populations:densitydependence}
\index{تعلق بالكثافة}
يبطؤ النمو مع ارتفاع الكثافة لأن معدل الولادة للفرد يهبط أو معدل
الوفاة يرتفع مع الازدحام: فالغذاء لكل رأس يتناقص، والحمى تصير غير
متاحة، وتتراكم الفضلات والأمراض، وتتركز المفترسات. وهذه العوامل
\emph{المتعلقة بالكثافة} تنظّم \omterm{def:b1:populations:population}{الجمهرة} نحو $K$. أما العوامل
\emph{غير المتعلقة بالكثافة} --- الصقيع والفيضان والحريق والجفاف
--- فتقتل كسرًا بصرف النظر عن الأعداد ولا تستطيع أن تنظّم، بل أن
تزعج فقط؛ \omterm{def:b1:populations:population}{وجمهرات} كثير من الحشرات يحكمها الطقس في أكثرها ولا تقارب
\omterm{thm:b1:populations:logistic}{سعة حمولية} أبدًا.
\end{proposition}
\begin{proof}[شواهد]
نمّى غوز (1934) \emph{Paramecium} في أنابيب وسط يُجدَّد يوميًا:
فتبعت الأعداد منحنيًا لوجستيًا إلى هضبة يحددها الغذاء، ومضاعفة
الغذاء ضاعفت الهضبة. ووجد بيرل (1927) الشيء نفسه في ذباب الفاكهة في
قوارير، ووجده كارلسون (1913) في الخميرة. وفي الحقل، تضع عصافير
الأغاني في جزيرة أعشاشًا أصغر وتفقد صغارًا أكثر حين تكبر \omterm{def:b1:populations:population}{جمهرة}
التوالد، ويحدّ عدد الحمى عدد المتوالدات؛ وإزالة طيور يتبعها إحلالها
من فائض كان مستبعدًا.
\end{proof}

\begin{definition}[استراتيجيات تاريخ الحياة]\label{def:b1:populations:rK}
\index{استراتيجية r}\index{استراتيجية K}
تختلف الأنواع في موقعها بين طرفين. فالنوع \emph{المنتخَب على $r$}
--- الأعشاب الضارة، والمنّ، والفئران، وأكثر الأسماك --- مبني
لمعدل $r$ عالٍ: ينضج باكرًا، وينتج نسلًا كثيرًا صغيرًا، ويستثمر
قليلًا في كل واحد، ويعيش قصيرًا، ويستغل الموائل الجديدة أو المضطربة
قبل أن تصل المنافسة، وبقياه من النمط III. والنوع \emph{المنتخَب على
$K$} --- البلوط والفيلة والقطارس --- مبني ليدوم قرب $K$: ينضج
متأخرًا، وينتج نسلًا قليلًا كبيرًا، ويعتني به، ويعيش طويلًا،
وينافس جيدًا في موائل مزدحمة مستقرة، ومنحنيه من النمط I. وأكثر
الأنواع يخلط بينهما؛ واللقبان يسميان طرفي السلّم.
\end{definition}

\section{التذبذب والتنظيم}

\begin{proposition}[دورات المفترس والفريسة]\label{prop:b1:populations:cycles}
\index{دورة جمهرية}
بعض \omterm{def:b1:populations:population}{الجمهرات} لا تستقر بل تتذبذب. فالأرنب الثلجي في الغابة الشمالية
والوشق الذي يأكله يرتفعان ويهبطان معًا بدور نحو عشر سنوات، والوشق
متأخر عن الأرنب سنة أو سنتين: فالأرانب الكثيرة تطعم وشوقًا كثيرة،
والوشوق الكثيرة تأكل الأرانب حتى تقل، والأرانب القليلة تجيع
الوشوق، والوشوق القليلة تدع الأرانب تتعافى. والدورة يسوقها جزئيًا
المفترس وجزئيًا غذاء الأرانب نفسه، الذي يستغرق تعافيه بعد الرعي
المفرط سنوات؛ ونماذج المفترس والفريسة المقترنة
(\cref{ch:b1:species-interactions}) تعيد إنتاج مثل هذه التذبذبات من
معادلتين.
\end{proposition}
\begin{proof}[شواهد]
سجلات فراء شركة تجارية، محفوظة تسعين سنة من أربعينيات القرن التاسع
عشر، تدوّن عدد جلود الوشق والأرنب المجلوبة كل سنة عبر شبه قارة:
فتُظهر السلسلتان قممًا كل تسع إلى عشر سنوات، وقمة الوشق متأخرة عن
قمة الأرنب، على مدى تسع دورات كاملة. والدراسات الحقلية منذئذ، مع
أرانب مسيَّجة عن المفترسات أو مُعطاة غذاءً إضافيًا، تبيّن أن إزالة
أي من العاملين تخمد الدورة وإزالتهما معًا تلغيها.
\end{proof}

إلى عمود. وحساب $R_0$ ينطبق علينا؛ والذي تغيّر هو $m_x$.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:populations:1}
عرّف \omterm{def:b1:populations:population}{الجمهرة} والكثافة والتوزع، وأعطِ مثالًا على كل نمط من أنماط
التوزع.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:populations:2}
عرّف $l_x$ و$m_x$ و$R_0$ و$T$، وقل ماذا يعني $R_0 = 1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:populations:3}
من شكل \omterm{def:b1:populations:lifetable}{البقيا}، أي كسر من فوج من النمط III ينجو من عُشر عمره الأول؟
وأي كسر من فوج من النمط I يبلغ ثلاثة أرباعه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:populations:4}
اذكر المعادلتين الأسّية واللوجستية وقل ماذا يعني $r$ و$K$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:5}
مزرعة بكتيرية تنمو من $10^4$ إلى $10^7$ \omterm{def:b1:cell-unit-of-life:cell}{خلية} في الميليلتر
في \qty{5}{h}. احسب $r$ وزمن التضاعف.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:6}
جدول حياة: $l_x = 1, 0.5, 0.3, 0.1, 0$ عند الأعمار 0 إلى 4؛ و$m_x = 0,
1, 2, 2, 0$. احسب $R_0$ و$T$ و$r$، وقل هل تنمو
\omterm{def:b1:populations:population}{الجمهرة}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:7}
خمسون سمكة تُوسم وتُطلق في بركة؛ وبعد أسبوع تُصاد 70 منها 7
موسومة. قدّر $N$. وإذا ماتت عشر من السمكات الموسومة من المناولة،
فهل التقدير أعلى أم أدنى من الحقيقة، وبكم؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:8}
\omterm{def:b1:populations:population}{جمهرة} لوجستية لها $K = 1000$ و$r = \qty{0.2}{yr^{-1}}$.
احسب $\dd N/\dd t$ عند $N = 100$ و$500$ و$900$، وأكبر عدد
يمكن حصده كل سنة بلا تراجع.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:9}
صنّف كلًّا مما يلي منتخَبًا على $r$ أو على $K$، بسمتين لكل واحد:
هندباء برية، وفيل، وسمكة قدّ، وغوريلا، وذبابة منزل.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:populations:10}
من شكل الخميرة، قدّر $r$ من أول نقطتين و$K$ من الأخيرة، ثم تنبأ
بقيمة $N$ عند \qty{8}{h} من الصيغة اللوجستية وقارنه بالعدّ. وفي أي
زمن يبلغ معدل النمو ذروته، وكم يكون؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:populations:11}
فسّر لماذا لا يستطيع عامل غير متعلق بالكثافة أن ينظّم \omterm{def:b1:populations:population}{جمهرة} مع
أنه يستطيع تقليلها، مستعملًا المعادلات، ولماذا لا تنمو مع ذلك
\omterm{def:b1:populations:population}{جمهرات} الحشرات التي يحكمها الطقس بلا حد.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:populations:12}
«\omterm{def:b1:populations:population}{الجمهرة} ليست عددًا بل توزيعًا.» ناقش في فقرة: البنية العمرية،
والتأخر بين هبوط $m_x$ وهبوط $N$، وما الذي يتنبأ به الهرم مما لا
يتنبأ به $N$ وحده.
\end{exercise}

\section{مسألة: خميرة في دورق وأيائل في جزيرة}

\begin{problem}[{مسألة نهاية الأسبوع --- مزرعة تُعدّ، وقطيع
يُجدول، وبحيرة تُعيَّن، وحصاد يُضبط، وختامًا السعة الحمولية ومعدل
النمو الذاتي للخميرة}]
\label{pb:b1:populations:1}
مزرعة خميرة تُعدّ كل ساعتين (\omterm{def:b1:cell-unit-of-life:cell}{خلايا} في الميليلتر،
$\times 10^5$): $10, 29, 71, 175, 351, 513, 595, 641, 656$ عند $t = 0,
2, \ldots, 16$ ساعة. \omterm{def:b1:populations:population}{وجمهرة} أيائل في جزيرة لها \omterm{def:b1:populations:lifetable}{جدول الحياة}:
$l_x = 1.00, 0.60, 0.52, 0.46, 0.40, 0.34, 0.26, 0.16, 0.06, 0$ و
$m_x = 0, 0, 0.6, 0.8, 0.8, 0.8, 0.7, 0.5, 0.3, 0$ عند $x = 0$ إلى $9$
سنوات. وفي بحيرة، 120 سمكة فرخ تُوسم وتُطلق؛ وبعد أسبوع تُصاد 150
منها 18 موسومة.

\medskip
\noindent\textbf{الجزء الأول --- الخميرة.}
\begin{enumerate}
  \item قدّر $r$ من أول عدّتين، بفرض \omterm{thm:b1:populations:exponential}{نمو أسّي} على مدى الساعتين
        الأوليين.
  \item احسب زمن التضاعف.
  \item قدّر $K$ من العدّات الأخيرة.
  \item احسب التنبؤ اللوجستي عند $t = 4$ و$8$ و\qty{12}{h}
        وقارنه بالعدّات.
  \item عند أي \omterm{def:b1:populations:population}{جمهرة} يكون النمو أسرع، ومتى تُبلَغ؟ واحسب معدل
        النمو الأعظمي \omterm{def:b1:cell-unit-of-life:cell}{بالخلايا} في الميليلتر في الساعة.
  \item بماذا كان النموذج الأسّي يتنبأ عند \qty{12}{h}؟ وبأي معامل
        يكون خاطئًا؟
  \item سمّ شيئين قد يحددان $K$ في دورق خميرة محكم.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الأيائل.}
\begin{enumerate}[resume]
  \item احسب الوفيات $q_x$ في السنة الأولى وفي الخامسة. فأي نمط
        \omterm{def:b1:populations:lifetable}{بقيا} هذا؟
  \item احسب $l_x m_x$ لكل عمر واجمعها: أي $R_0$.
  \item احسب $T$.
  \item احسب $r$ وزمن تضاعف القطيع.
  \item في الجزيرة اليوم 200 أيّل. تنبأ بالعدد بعد 10
        سنوات \omterm{thm:b1:populations:exponential}{بالنمو الأسّي}.
  \item تستطيع الجزيرة إطعام 600 أيّل. تنبأ بالعدد بعد 10 سنوات
        \omterm{thm:b1:populations:logistic}{بالنمو اللوجستي}، وبالسنة التي يتجاوز فيها القطيع 500.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الفرخ والحصاد.}
\begin{enumerate}[resume]
  \item قدّر \omterm{def:b1:populations:population}{جمهرة} الفرخ في البحيرة.
  \item إذا فقدت 20 من السمكات الموسومة وسومها قبل الصيدة الثانية،
        أعد الحساب بالعدد الحقيقي للسمك الموسوم، واذكر اتجاه الخطأ
        في التقدير الأول.
  \item \omterm{def:b1:populations:population}{لجمهرة} الفرخ $K = 1000$ و$r = \qty{0.4}{yr^{-1}}$.
        احسب أعظم غلة مستدامة \omterm{def:b1:populations:population}{والجمهرة} التي تُبلَغ عندها.
  \item احسب زمن تضاعف \omterm{def:b1:populations:population}{جمهرة} الفرخ حين تكون أدنى بكثير من $K$.
  \item يقرر النادي بدل ذلك أخذ \qty{20}{\%} ثابتة من الرصيد كل
        سنة. احسب \omterm{def:b1:populations:population}{جمهرة} التوازن والصيد السنوي عند ذلك التوازن.
  \item نادي صيد يأخذ 120 سمكة فرخ في السنة. بيّن هل تستطيع
        \omterm{def:b1:populations:population}{الجمهرة} إعالة ذلك، بحساب $\dd N/\dd t$ عند $N =
        500$ وعند $N = 300$.
  \item لماذا يكون الحصاد عند $K/2$ خطرًا عمليًا، مع مراعاة الطقس
        وخطأ العدّ في السؤال 14؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- التنظيم.}
\begin{enumerate}[resume]
  \item يهبط $m_x$ عند الأيائل إلى النصف حين يتجاوز القطيع 400.
        بيّن أن $R_0$ يهبط عندئذ دون 1 --- فماذا يفعل هذا بالقطيع،
        وأي نوع من العوامل يعمل؟
  \item شتاء قاسٍ يقتل \qty{40}{\%} من الأيائل بصرف النظر عن
        عددها. فهل هذا العامل منظِّم؟ وماذا يحدث للقطيع في السنوات
        التالية؟
  \item تصل الذئاب فتأخذ 30 أيلًا ثابتة في السنة. احسب أحجام
        القطيع عند التوازن في النموذج اللوجستي عند $K = 600$
        وعند $r$ في السؤال 11 (حُلّ $rN(1 - N/K) = 30$)، وقل
        أي التوازنين مستقر.
  \item فسّر في جملتين لماذا تتأخر دورة الوشق عن دورة الأرنب.
  \item اذكر النتيجة: \omterm{thm:b1:populations:logistic}{السعة الحمولية} ومعدل النمو الذاتي للخميرة،
        و$R_0$ و$T$ و$r$ عند الأيائل.
\end{enumerate}
\end{problem}
