\chapter{جسمي}\label{ch:g1:my-body}

قف أمام المرآة ولوّح: يلوّح لك جسم حي كامل. إنه ينمو منذ ما قبل
ولادتك، وينثني في مواضع دون أخرى، وهو لك مدى الحياة. حان وقت
النظر إليه كما نظرنا إلى القطة والأقحوانة --- جزءًا جزءًا.

\section{أجزاء جسمي}

\begin{definition}[أجزاء جسم الإنسان]\label{def:g1:my-body:parts}
لجسم الإنسان ثلاثة أجزاء رئيسة:
\begin{enumerate}
  \item \emph{الرأس}\index{رأس}، وهو يحمل الوجه والعينين والأذنين
        والأنف والفم؛
  \item \emph{البدن}\index{بدن}، وهو وسط الجسم --- الصدر والبطن
        والظهر؛
  \item \emph{الأطراف}\index{طرف}: \emph{ذراعان}\index{ذراع}
        تنتهيان بيدين، و\emph{رجلان}\index{رجل} تنتهيان بقدمين.
\end{enumerate}
ويستقر الرأس على \emph{العنق}\index{عنق} الذي يصله بالبدن.
\end{definition}

\begin{example}[الخطة نفسها كالقطة؟]\label{ex:g1:my-body:cat}
قارن نفسك بالقطة: رأس، وجسم، وأربعة \omterm{def:g1:animals-around-us:parts}{أطراف} --- تبدو الخطة متشابهة.
لكن القطة تمشي على أربع؛ أما أنت فتقف على \omterm{def:g1:my-body:parts}{رجلين} وتُبقي \omterm{def:g1:my-body:parts}{ذراعين}
حرتين للحمل والرسم والتلويح. وحيث للقطة ذنب، لا ذنب لك.
\end{example}

\section{الانثناء: المفاصل}

\begin{definition}[المفصل]\label{def:g1:my-body:joint}
\emph{المفصل}\index{مفصل} موضع يستطيع الجسم عنده أن ينثني أو يدور.
يثني \emph{المرفق}\index{مرفق} \omterm{def:g1:my-body:parts}{الذراع}، وتثني
\emph{الركبة}\index{ركبة} \omterm{def:g1:my-body:parts}{الرجل}، ويدير \emph{الكتف}\index{كتف}
\omterm{def:g1:my-body:parts}{الذراع} كلها، ويثني \emph{الرسغ}\index{رسغ}
و\emph{الكاحل}\index{كاحل} اليد والقدم. وبين مفصلين يبقى
الجسم مستقيمًا.
\end{definition}

\begin{example}[جرّب: المشي بركبتين جامدتين]\label{ex:g1:my-body:stiff}
جرّب أن تمشي من غير أن تثني ركبتيك. يمكن ذلك --- بالكاد --- بخطوات
بطة صغيرة جامدة. والآن جرّب أن تلتقط قلمًا عن الأرض من غير أن تثني
ركبتيك أو وركيك أو ظهرك. \omterm{def:g1:my-body:joint}{المفاصل} هي التي تجعل حركات الجسم سلسة
سريعة سهلة.
\end{example}

\begin{remark}[اليمين واليسار]\label{rem:g1:my-body:sides}
للجسم جانبان، يمين ويسار، يكاد أحدهما يكون مرآة الآخر: \omterm{def:g1:my-body:parts}{ذراع} \omterm{def:g1:my-body:parts}{ورجل}
في كل جانب، وعين وأذن في كل جانب من الرأس. ومعرفة يدك اليمنى من
اليسرى معرفةٌ بالجسم أيضًا --- فهي تدل الناس على الجهة التي تعنيها.
\end{remark}

\section{جسمي ينمو}

\begin{proposition}[الكبر]\label{prop:g1:my-body:growth}
جسم الطفل \emph{ينمو}\index{نمو}: فيزداد طولًا ووزنًا، سنة بعد
سنة، حتى يبلغ حجم الكبار. والنمو بطيء ثابت --- أبطأ من أن يُحسّ
به، وأسهل من أن يُقاس.
\end{proposition}
\admitted

\begin{example}[علامات على إطار الباب]\label{ex:g1:my-body:doorframe}
قف مستندًا إلى إطار الباب؛ فيعلّم كبير طولك بقلم رصاص. وأعد ذلك
بعد بضعة أشهر: العلامة الجديدة أعلى! لم تشعر قط بنفسك تنمو، لكن
العلامتين، إحداهما فوق الأخرى، تبينان ذلك بوضوح.
\end{example}

\begin{method}[قياس طولي]\label{met:g1:my-body:measure}
\begin{enumerate}
  \item اخلع حذاءك؛
  \item قف مستقيمًا، وظهرك وعقباك إلى الجدار، ونظرك إلى الأمام؛
  \item يضع مساعد كتابًا مسطحًا على رأسك، أفقيًّا، ملامسًا للجدار؛
  \item علّم الموضع تحت الكتاب، واكتب التاريخ بجانبه.
\end{enumerate}
والعلامات المأخوذة بالطريقة نفسها في كل مرة يُوثق بها --- وتُقارن.
\end{method}

\begin{remark}[كل جسم مختلف]\label{rem:g1:my-body:different}
في الصف الواحد أطفال أطول وأطفال أقصر؛ وبعضهم ينمو باكرًا وبعضهم
متأخرًا. وكل ذلك طبيعي. الأجسام كالوجوه: مبنية على الخطة نفسها،
وكل واحد منها مختلف. والعلامات على إطار بابك لك أنت --- لا لسباق
مع الجار.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:my-body:1}
سمِّ أجزاء جسم الإنسان الرئيسة الثلاثة. وما الذي يصل الرأس \omterm{def:g1:my-body:parts}{بالبدن}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:2}
بم ينتهي كل طرف --- ما الذي في نهاية \omterm{def:g1:my-body:parts}{الذراع}، وما الذي في نهاية
\omterm{def:g1:my-body:parts}{الرجل}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:3}
أي \omterm{def:g1:my-body:joint}{مفصل} يثني: (أ) \omterm{def:g1:my-body:parts}{الذراع} في وسطها؟ (ب) \omterm{def:g1:my-body:parts}{الرجل} في وسطها؟ (ج) اليد
في طرف \omterm{def:g1:my-body:parts}{الذراع}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:4}
المس أذنك اليمنى بيدك اليسرى. أي \omterm{def:g1:my-body:joint}{مفاصل} استعملتها ذراعك للتو؟ سمِّ
مفصلين على الأقل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:5}
في \cref{ex:g1:my-body:cat}، جد وجهًا واحدًا يشبه فيه جسمك جسم
القطة، ووجهين يختلف فيهما عنه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:6}
لماذا يضع المساعد في \cref{met:g1:my-body:measure} كتابًا مسطحًا
على رأسك بدل أن ينظر فحسب؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:7}
علامتان على إطار الباب: علامة العام الماضي، وعلامة اليوم أعلى
منها قليلًا. علام تدل العلامتان؟ وهل شعرت بذلك يحدث؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:8}
جرّب: امشِ بركبتين جامدتين كما في \cref{ex:g1:my-body:stiff}، ثم
حاول أن تصفق من غير أن تثني مرفقيك. أي الحركات اليومية يتبين أنها
تحتاج إلى \omterm{def:g1:my-body:joint}{مفاصل}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:9}
ميّا أصغر أطفال صفها حجمًا وهي قلقة من ذلك. بم تجيبها
\cref{rem:g1:my-body:different}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:10}
دمية مصنوعة من عصا خشبية واحدة لا تستطيع الجلوس على كرسي. أين
تقطع العصا وتضيف نقاط دوران لتستطيع الجلوس؟ سمِّ \omterm{def:g1:my-body:joint}{مفاصل} الجسم التي
تحاكيها قطوعك.
\end{exercise}
