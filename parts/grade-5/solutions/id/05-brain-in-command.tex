\section*{Bab \ref{ch:g5:brain-in-command} --- Otak yang Memerintah}

\begin{solution}{exo:g5:brain-in-command:1}
\omterm{def:g5:brain-in-command:system}{Otak}: menerima semua laporan, membuat semua keputusan, mengeluarkan
semua perintah. \omterm{def:g5:brain-in-command:system}{Sumsum tulang belakang}: kabel tebal yang berjalan dari
\omterm{def:g5:brain-in-command:system}{otak} turun ke dalam \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}. \omterm{def:g5:brain-in-command:system}{Saraf}: benang bercabang yang
membawa pesan antara \omterm{def:g5:brain-in-command:system}{otak} atau sumsumnya dan setiap organ.
\end{solution}

\begin{solution}{exo:g5:brain-in-command:2}
\omterm{def:g1:the-five-senses:sense}{Alat indra} melapor kepada \omterm{def:g5:brain-in-command:system}{otak}; \omterm{def:g5:brain-in-command:system}{otak} memutuskan lalu menyusun
perintah; \omterm{def:g5:brain-in-command:system}{saraf} membawa perintahnya ke \omterm{def:g3:bones-and-muscles:muscle}{otot}, yang lalu \omterm{def:g3:bones-and-muscles:muscle}{berkontraksi}.
\omterm{def:g1:the-five-senses:sense}{Alat indra} melapor, \omterm{def:g5:brain-in-command:system}{otak} memutuskan, \omterm{def:g3:bones-and-muscles:muscle}{otot} menurut.
\end{solution}

\begin{solution}{exo:g5:brain-in-command:3}
\omterm{def:g5:brain-in-command:system}{Otaknya} di dalam \omterm{def:g3:bones-and-muscles:skeleton}{tengkorak}; \omterm{def:g5:brain-in-command:system}{sumsum tulang belakangnya} di dalam tiang
\omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} yang tersusun atas \omterm{def:g3:bones-and-muscles:skeleton}{tulang} kecil bertumpuk.
\end{solution}

\begin{solution}{exo:g5:brain-in-command:4}
Jeda antara sebuah isyarat dan tanggapan tubuh --- kira-kira seperlima
detik untuk tugas sederhana pada orang yang terlatih, sedikit lebih
lama pada anak-anak.
\end{solution}

\begin{solution}{exo:g5:brain-in-command:5}
\omterm{def:g1:the-five-senses:sense}{Kulit} kakinya melapor (rabaan dan nyeri), \omterm{def:g5:brain-in-command:system}{sistem sarafnya} memutuskan
untuk menarik diri, dan \omterm{def:g3:bones-and-muscles:muscle}{otot} \omterm{def:g1:my-body:parts}{tungkainya} menurut dengan menyentakkan
kakinya menjauh. (Penarikan yang secepat itu sebagian ditangani \omterm{def:g5:brain-in-command:system}{sumsum
tulang belakang} sendiri --- meja darurat milik pusat komandonya.)
\end{solution}

\begin{solution}{exo:g5:brain-in-command:6}
Ia menjaga \omterm{def:g4:breathing:movements}{pernapasan} tetap berjalan siang dan malam, dan menyetel
irama \omterm{def:g4:heart-and-blood:heart}{jantung} terhadap kebutuhan tubuh (juga: menyimpan ingatan,
bermimpi).
\end{solution}

\begin{solution}{exo:g5:brain-in-command:7}
\omterm{prop:g1:healthy-habits:needs}{Tidur} --- waktu pengarsipan dan perbaikannya --- dan bagian yang besar
serta tetap dari \omterm{prop:g3:balanced-diet:jobs}{energi} dan oksigen tubuh. Kalau kekurangan salah
satunya, ia bereaksi lebih lambat, perhatiannya berserak dan
keputusannya buruk.
\end{solution}

\begin{solution}{exo:g5:brain-in-command:8}
Jawaban terbuka. Tangkapan yang lazim berkisar $10$--$20$ cm;
beberapa kali percobaan biasanya memperbaiki dan memantapkan nilainya
sedikit.
\end{solution}

\begin{solution}{exo:g5:brain-in-command:9}
Karena setiap tangkapan harus menjalankan ketiga langkahnya ---
laporan \omterm{def:g1:the-five-senses:sense}{matanya} harus mengembara, \omterm{def:g5:brain-in-command:system}{otaknya} harus menghitung lalu
memutuskan, perintahnya harus mengembara balik dan \omterm{def:g3:bones-and-muscles:muscle}{ototnya} harus
\omterm{def:g3:bones-and-muscles:muscle}{berkontraksi} --- dan tiap langkah memakan waktu yang sungguhan.
Kemahiran memperpendek gelangnya; tidak ada yang menghapusnya.
\end{solution}

\begin{solution}{exo:g5:brain-in-command:10}
Karena rem yang terbaik pun baru bekerja sesudah gelang pengemudinya
berjalan --- dan pada laju jalan bebas hambatan mobilnya menempuh
kira-kira \qty{7}{m} sebelum perintah ``rem!'' bahkan sampai di
kakinya. Jarak antarmobil adalah ruang bagi gelangnya. Perhatian
memperpendek langkah melapor-dan-memutuskan; \omterm{def:g5:brain-in-command:system}{otak} yang teralihkan
menambahkan meter.
\end{solution}

\begin{solution}{exo:g5:brain-in-command:11}
Bagian \omterm{def:g5:brain-in-command:system}{otaknya} --- yaitu memutuskan dan memerintah: latihan
menghamparkan jalur yang lebih baik, sehingga laporan yang sama
sekarang dijalurkan ke perintah yang tepat dengan cepat dan tanpa
usaha. Melatih \omterm{def:g3:bones-and-muscles:muscle}{otot} menumbuhkan kekuatan \omterm{def:g3:bones-and-muscles:muscle}{ototnya}; melatih keterampilan
menumbuhkan penjaluran \omterm{def:g5:brain-in-command:system}{otaknya}. Yang satu membangun mesinnya, yang
lain membangun panglimanya.
\end{solution}
