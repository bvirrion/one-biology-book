\chapter{The Cell Cycle and Mitosis}\label{ch:g11:cell-cycle-mitosis}

Squash the tip of an onion root under a cover slip, stain it, and look:
among hundreds of quiet \omterm{def:g10:cells-common-unit:cell}{cells} with a round \omterm{def:g10:cells-common-unit:organelle}{nucleus}, a few show something
else --- dark X-shaped rods gathered in the middle of the \omterm{def:g10:cells-common-unit:cell}{cell}, or two
sets of rods pulled apart towards the ends, or two small nuclei with a
wall forming between them. You are watching \omterm{def:g10:cells-common-unit:cell}{cell} division caught at
different moments, in a tissue that grows a millimetre a day. Each of
those divisions hands two complete, identical sets of \omterm{def:g10:universal-dna:information}{chromosomes} to two
daughter \omterm{def:g10:cells-common-unit:cell}{cells}. This chapter follows a \omterm{def:g10:cells-common-unit:cell}{cell} through one full cycle, and
looks closely at the hour in which the \omterm{def:g10:universal-dna:information}{chromosomes} are shared out.

\section{The cell cycle}

\begin{definition}[Cell cycle]\label{def:g11:cell-cycle-mitosis:cycle}
The \emph{cell cycle}\index{cell cycle} is the sequence of events from
the birth of a \omterm{def:g10:cells-common-unit:cell}{cell}, by division, to its own division into two. It
comprises the \emph{interphase}, during which the \omterm{def:g10:cells-common-unit:cell}{cell} grows and
copies its \omterm{def:g10:universal-dna:information}{DNA}, and the \emph{M phase}, during which the \omterm{def:g10:cells-common-unit:organelle}{nucleus}
divides (\emph{mitosis}\index{mitosis}) and then the \omterm{def:g10:cells-common-unit:cell}{cytoplasm}
(\emph{cytokinesis}). Interphase has three parts: $\mathrm{G_1}$
(growth), S (\omterm{def:g11:dna-replication:replication}{DNA replication}, \cref{ch:g11:dna-replication}) and
$\mathrm{G_2}$ (preparation for division).
\end{definition}

\begin{proposition}[DNA through the cycle]\label{prop:g11:cell-cycle-mitosis:dna}
Call $q$ the amount of \omterm{def:g10:universal-dna:information}{DNA} in a \omterm{def:g10:cells-common-unit:organelle}{nucleus} just after division. It stays
at $q$ through $\mathrm{G_1}$, rises steadily to $2q$ during S as the
\omterm{def:g10:universal-dna:information}{chromosomes} are copied, stays at $2q$ through $\mathrm{G_2}$ and the
first part of \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}, and falls back to $q$ in each daughter \omterm{def:g10:cells-common-unit:organelle}{nucleus}
at the end of \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}. The number of \omterm{def:g10:universal-dna:information}{chromosomes} does not change during
S: each \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} is copied into two identical \emph{\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}} held
together at their \emph{\omterm{def:g11:cell-cycle-mitosis:chromosome}{centromere}}, and only separates into two
\omterm{def:g10:universal-dna:information}{chromosomes} during \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}.
\end{proposition}
\admitted

\begin{example}[Reading the curve]\label{ex:g11:cell-cycle-mitosis:reading}
A \omterm{def:g10:cells-common-unit:cell}{cell} measured at $2q$ may be in $\mathrm{G_2}$ or in early \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis};
at $1.5q$ it is certainly in S, half-way through replication; at $q$ it
is in $\mathrm{G_1}$ --- or has left the cycle for good, like a neuron
or a muscle fibre, which stay at $q$ for life. A tissue in which every
\omterm{def:g10:cells-common-unit:cell}{cell} is at $q$ is a tissue that has stopped dividing.
\end{example}

\section{Chromosomes, seen}

\begin{definition}[Chromosome, chromatid, karyotype]\label{def:g11:cell-cycle-mitosis:chromosome}
A \emph{chromosome} is one \omterm{def:g10:universal-dna:information}{DNA} molecule with its packing \omterm{def:g10:chemistry-of-life:families}{proteins}.
Between divisions it is a loose, invisible thread; at the start of
\omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} it coils into a compact rod visible under the light microscope.
After replication each chromosome consists of two identical
\emph{chromatids}\index{chromatid}, joined at the
\emph{centromere}\index{centromere}: the familiar X shape. The
\emph{karyotype}\index{karyotype} of a species is its complete set of
\omterm{def:g10:universal-dna:information}{chromosomes}, photographed at metaphase and arranged in pairs by size:
a human \omterm{def:g10:cells-common-unit:cell}{cell} has 46, in 23 pairs, the members of a pair being
\emph{\omterm{def:g10:common-ancestry:homology}{homologous}} --- same \omterm{def:g10:universal-dna:gene}{genes} at the same positions, one from each
parent.
\end{definition}

\section{Mitosis, step by step}

\begin{proposition}[The four stages of mitosis]\label{prop:g11:cell-cycle-mitosis:stages}
\omterm{def:g11:cell-cycle-mitosis:cycle}{Mitosis} is a continuous movement, described in four stages.
\begin{enumerate}
  \item \emph{Prophase}: the \omterm{def:g10:universal-dna:information}{chromosomes} condense into visible
        \omterm{def:g11:cell-cycle-mitosis:chromosome}{two-chromatid} rods; the nuclear envelope breaks down; a
        \emph{spindle} of \omterm{def:g10:chemistry-of-life:families}{protein} fibres forms between the two poles
        of the \omterm{def:g10:cells-common-unit:cell}{cell}.
  \item \emph{Metaphase}: the \omterm{def:g10:universal-dna:information}{chromosomes}, attached to spindle fibres
        by their \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromeres}, line up on the equatorial plane of the
        \omterm{def:g10:cells-common-unit:cell}{cell}.
  \item \emph{Anaphase}: the \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromeres} split; the two \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} of
        each \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}, now two \omterm{def:g10:universal-dna:information}{chromosomes}, are pulled to opposite
        poles by the shortening fibres.
  \item \emph{Telophase}: a nuclear envelope re-forms around each set;
        the \omterm{def:g10:universal-dna:information}{chromosomes} uncoil. Cytokinesis then divides the
        \omterm{def:g10:cells-common-unit:cell}{cytoplasm} --- by pinching in animal \omterm{def:g10:cells-common-unit:cell}{cells}, by building a new
        wall in plant \omterm{def:g10:cells-common-unit:cell}{cells}.
\end{enumerate}
Each daughter \omterm{def:g10:cells-common-unit:cell}{cell} receives one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid} of every \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}: the same
number of \omterm{def:g10:universal-dna:information}{chromosomes} as the mother, carrying the same \omterm{def:g10:universal-dna:information}{DNA}.
\end{proposition}
\admitted

\begin{example}[Counting in the root tip]\label{ex:g11:cell-cycle-mitosis:count}
Of 600 \omterm{def:g10:cells-common-unit:cell}{cells} counted in a root tip, 540 are in interphase and 60 in
\omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}: a \emph{mitotic index} of 10\%. If the cycle lasts 20 hours,
\omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} lasts about $0.10 \times 20 = \qty{2}{h}$; and if 36 of the 60
dividing \omterm{def:g10:cells-common-unit:cell}{cells} are in prophase, prophase takes $36/60$ of those two
hours, some 72 minutes, while anaphase, seen in only 4 \omterm{def:g10:cells-common-unit:cell}{cells}, takes
about 8 minutes. The rarest stage is the fastest.
\end{example}

\begin{method}[Counting chromosomes and chromatids]\label{met:g11:cell-cycle-mitosis:counting}
For a species with $2n$ \omterm{def:g10:universal-dna:information}{chromosomes}:
\begin{enumerate}
  \item in $\mathrm{G_1}$: $2n$ \omterm{def:g10:universal-dna:information}{chromosomes}, one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid} each, \omterm{def:g10:universal-dna:information}{DNA}
        $q$;
  \item after S, through $\mathrm{G_2}$, prophase and metaphase: $2n$
        \omterm{def:g10:universal-dna:information}{chromosomes}, two \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} each ($4n$ \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}), \omterm{def:g10:universal-dna:information}{DNA} $2q$;
  \item from anaphase: $4n$ \omterm{def:g11:cell-cycle-mitosis:chromosome}{single-chromatid} \omterm{def:g10:universal-dna:information}{chromosomes} in the \omterm{def:g10:cells-common-unit:cell}{cell},
        $2n$ heading to each pole;
  \item in each daughter \omterm{def:g10:cells-common-unit:cell}{cell}: $2n$ \omterm{def:g10:universal-dna:information}{chromosomes}, one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid} each,
        \omterm{def:g10:universal-dna:information}{DNA} $q$.
\end{enumerate}
A \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid} becomes a \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} the moment its \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromere} splits.
\end{method}

\begin{example}[Human numbers]\label{ex:g11:cell-cycle-mitosis:human}
A human \omterm{def:g10:cells-common-unit:cell}{cell} in metaphase holds 46 \omterm{def:g10:universal-dna:information}{chromosomes} and 92 \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}; in
anaphase, 92 \omterm{def:g10:universal-dna:information}{chromosomes}, 46 moving to each pole; each daughter, 46
\omterm{def:g10:universal-dna:information}{chromosomes} of one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid} each, containing exactly the \omterm{def:g10:universal-dna:information}{DNA} the
mother had at the start of its cycle.
\end{example}

\section{What mitosis is for}

\begin{proposition}[Conformity and its uses]\label{prop:g11:cell-cycle-mitosis:conformity}
Because replication is faithful and \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} distributes the \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}
exactly, every \omterm{def:g10:cells-common-unit:cell}{cell} of a multicellular body carries the same \omterm{def:g10:universal-dna:information}{genetic
information} as the fertilised egg it descends from. \omterm{def:g11:cell-cycle-mitosis:cycle}{Mitosis} is the
mechanism of growth (the embryo, the root tip), of renewal (skin,
gut lining, blood: the bone marrow alone makes some $\num{2e11}$ \omterm{def:g10:cells-common-unit:cell}{cells}
a day) and of repair (wound healing); in single-celled eukaryotes it
is reproduction itself. Its control --- the decision to enter the
cycle, the checks before S and before anaphase --- is the subject of
\cref{ch:g11:cancer}.
\end{proposition}
\admitted

\begin{remark}[Not everything divides]\label{rem:g11:cell-cycle-mitosis:notall}
Most neurons and heart muscle \omterm{def:g10:cells-common-unit:cell}{cells} leave the cycle in early
childhood and never re-enter it: damage to them is not repaired by
division, which is why a spinal injury or a heart attack leaves lasting
loss. Liver \omterm{def:g10:cells-common-unit:cell}{cells} rest in $\mathrm{G_1}$ for months but re-enter the
cycle when part of the liver is removed, regrowing the organ within
weeks. Which \omterm{def:g10:cells-common-unit:cell}{cells} may divide, and when, is one of the tightest
regulations of the body.
\end{remark}

\section{Problem: The Root Tip's Timetable}

\begin{problem}[{Weekend problem --- a root tip counted cell by cell:
the length of every stage of the cycle, the DNA at each moment, the
cells a root makes in a day, and the eight minutes of anaphase}]
\label{pb:g11:cell-cycle-mitosis:1}
A class squashes and stains a garlic root tip ($2n = 16$) and counts
1000 \omterm{def:g10:cells-common-unit:cell}{cells} in the growing zone: 900 in interphase, 60 in prophase, 18
in metaphase, 8 in anaphase, 14 in telophase. Independent measurements
give the cycle of these \omterm{def:g10:cells-common-unit:cell}{cells} as 20 hours. Take the growing zone to
contain 20\,000 \omterm{def:g10:cells-common-unit:cell}{cells}.

\medskip
\noindent\textbf{Part I --- The mitotic index.}
\begin{enumerate}
  \item Compute the mitotic index of the tissue.
  \item Assuming that the fraction of \omterm{def:g10:cells-common-unit:cell}{cells} in a stage equals the
        fraction of the cycle spent in it, compute the duration of
        \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}.
  \item Compute the duration of each of the four stages.
  \item Which stage is the shortest? Suggest why it can be so quick.
  \item Why does the method require counting many \omterm{def:g10:cells-common-unit:cell}{cells}, and why must
        they come from the growing zone only?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- \omterm{def:g10:universal-dna:information}{Chromosomes} and \omterm{def:g10:universal-dna:information}{DNA}.}
\begin{enumerate}[resume]
  \item How many \omterm{def:g10:universal-dna:information}{chromosomes}, and how many \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}, does a garlic
        \omterm{def:g10:cells-common-unit:cell}{cell} hold at metaphase?
  \item How many \omterm{def:g10:universal-dna:information}{chromosomes} are moving in an anaphase \omterm{def:g10:cells-common-unit:cell}{cell}, and how
        many reach each pole?
  \item Calling $q$ the \omterm{def:g10:universal-dna:information}{DNA} of a daughter \omterm{def:g10:cells-common-unit:cell}{cell}, give the \omterm{def:g10:universal-dna:information}{DNA} content of
        a \omterm{def:g10:cells-common-unit:cell}{cell} in prophase, in anaphase (whole \omterm{def:g10:cells-common-unit:cell}{cell}) and in telophase
        (each \omterm{def:g10:cells-common-unit:organelle}{nucleus}).
  \item Of the 900 interphase \omterm{def:g10:cells-common-unit:cell}{cells}, roughly how many are in S, if
        $\mathrm{G_1}$, S and $\mathrm{G_2}$ take 8, 7 and 4 hours?
  \item A student finds a \omterm{def:g10:cells-common-unit:cell}{cell} with 32 \omterm{def:g11:cell-cycle-mitosis:chromosome}{single-chromatid} \omterm{def:g10:universal-dna:information}{chromosomes}
        spread through the \omterm{def:g10:cells-common-unit:cell}{cytoplasm}. What stage is it, and what has
        just happened?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The root's output.}
\begin{enumerate}[resume]
  \item If every \omterm{def:g10:cells-common-unit:cell}{cell} of the growing zone cycles in 20 hours, how many
        new \omterm{def:g10:cells-common-unit:cell}{cells} does the zone produce per hour?
  \item Each new \omterm{def:g10:cells-common-unit:cell}{cell}, once out of the growing zone, elongates to about
        \qty{100}{\micro m}. In a file of \omterm{def:g10:cells-common-unit:cell}{cells} one \omterm{def:g10:cells-common-unit:cell}{cell} wide, how much
        does the root lengthen per day? Compare with the observed
        \qty{1}{mm} a day and explain the difference.
  \item Each daughter \omterm{def:g10:cells-common-unit:cell}{cell} must copy $2n = 16$ \omterm{def:g10:universal-dna:information}{chromosomes}, some
        $\num{1.6e10}$ base pairs in all. With forks of 50 \omterm{def:g10:universal-dna:nucleotide}{nucleotides}
        per second, how many origins are needed to finish in the 7
        hours of S?
  \item The root is put in the cold (\qty{4}{\celsius}) for a day.
        Predict the change in the counts, and in the mitotic index if
        cold slows all phases equally.
  \item A herbicide blocks \omterm{prop:g11:dna-replication:forks}{DNA polymerase}. After a day, which stages
        would still be seen in the squash, and which would have
        disappeared? Explain.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The verdict of the counts.}
\begin{enumerate}[resume]
  \item A second class counts the same tissue and finds 4 anaphases in
        1000 \omterm{def:g10:cells-common-unit:cell}{cells}. What duration do they compute? Is the disagreement
        surprising for a stage seen in a handful of \omterm{def:g10:cells-common-unit:cell}{cells}?
  \item Combining both classes (2000 \omterm{def:g10:cells-common-unit:cell}{cells}, 12 anaphases), recompute
        the duration of anaphase.
  \item Human \omterm{def:g10:cells-common-unit:cell}{cells} in culture have a mitotic index of 4\% and a cycle
        of 24 hours. Compare the duration of their \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} with the
        garlic's.
  \item Explain why the mitotic index of a tumour is often several
        times that of the tissue it arose from, and what that means
        for a drug that acts on the spindle.
  \item State the result: the timetable of a garlic root \omterm{def:g10:cells-common-unit:cell}{cell}'s cycle,
        stage by stage, and the stage whose eight minutes decide that
        each daughter gets sixteen \omterm{def:g10:universal-dna:information}{chromosomes}.
\end{enumerate}
\end{problem}
