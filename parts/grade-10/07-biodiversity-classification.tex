\chapter{Biodiversity and Classification}\label{ch:g10:biodiversity-classification}

Life is not a random pile of species: it is a \omterm{def:g10:biodiversity-classification:hierarchy}{family} tree shaped by
descent with modification. This chapter defines \omterm{def:g10:biodiversity-classification:biodiversity}{biodiversity} and the
species concept in practical terms, builds the Linnaean hierarchy, reads
simple \omterm{def:g10:biodiversity-classification:clade}{cladograms}, and introduces the \omterm{def:g10:biodiversity-classification:hierarchy}{three-domain} view of life.

\section{Biodiversity}

\begin{definition}[Biodiversity]\label{def:g10:biodiversity-classification:biodiversity}
\emph{Biodiversity}\index{biodiversity} is the variety of life. It is
often discussed at three levels: \emph{genetic diversity}\index{genetic
diversity} within species; \emph{species diversity}\index{species
diversity} (number and evenness of species); and \emph{ecosystem
diversity}\index{ecosystem diversity} (variety of habitats and
communities).
\end{definition}

\begin{remark}[Why it matters]\label{rem:g10:biodiversity-classification:why}
Diverse communities are often more stable under stress; humans depend on
wild and cultivated species for food, medicines, materials and ecosystem
services (pollination, water purification, climate regulation).
Classification is the map that makes diversity discussable.
\end{remark}

\section{What is a species?}

\begin{definition}[Biological species concept]\label{def:g10:biodiversity-classification:species}
Under the \emph{biological species concept}\index{species!biological
concept}, a \emph{species}\index{species} is a group of populations
whose members can interbreed in nature and produce viable, fertile
offspring --- and are reproductively isolated from other such groups.
\end{definition}

\begin{remark}[Limits of the concept]\label{rem:g10:biodiversity-classification:species-limits}
The concept fits many animals poorly for fossils, asexual \omterm{def:g10:scientific-biology:organism}{organisms}
(including most bacteria) and some plants that hybridise. Taxonomists
then use morphology, ecology or \omterm{def:g10:molecules-of-life:nucleic}{DNA} similarity. ``Species'' is a powerful
tool, not a single magic line.
\end{remark}

\begin{example}[Horses and donkeys]\label{ex:g10:biodiversity-classification:mule}
Horses and donkeys can mate to produce mules, but mules are sterile:
horses and donkeys remain separate species under the biological concept.
\end{example}

\section{Taxonomic hierarchy}

\begin{definition}[Taxonomy and binomial names]\label{def:g10:biodiversity-classification:taxonomy}
\emph{Taxonomy}\index{taxonomy} is the \omterm{def:g10:scientific-biology:science}{science} of naming and classifying
\omterm{def:g10:scientific-biology:organism}{organisms}. In \emph{binomial nomenclature}\index{binomial nomenclature}
each species has a two-part Latin name: \emph{Genus species}\index{genus}
(e.g.\ \emph{Homo sapiens}), italicised, genus capitalised.
\end{definition}

\begin{definition}[Hierarchy]\label{def:g10:biodiversity-classification:hierarchy}
Nested ranks (broad to narrow):
\emph{domain}\index{domain (taxonomy)} $\supset$
\emph{kingdom}\index{kingdom} $\supset$
\emph{phylum}\index{phylum} $\supset$
\emph{class}\index{class (taxonomy)} $\supset$
\emph{order}\index{order (taxonomy)} $\supset$
\emph{family}\index{family (taxonomy)} $\supset$
\emph{genus}\index{genus} $\supset$
\emph{species}\index{species}.
Each rank is a \emph{taxon}\index{taxon}. \omterm{def:g10:scientific-biology:organism}{Organisms} in a smaller taxon
share a more recent common ancestor (in evolutionary classification).
\end{definition}

\begin{example}[Human classification (abbreviated)]\label{ex:g10:biodiversity-classification:human}
\omterm{def:g10:biodiversity-classification:hierarchy}{Domain} \omterm{def:g10:biodiversity-classification:domains}{Eukarya}; \omterm{def:g10:biodiversity-classification:hierarchy}{kingdom} Animalia; \omterm{def:g10:biodiversity-classification:hierarchy}{phylum} Chordata; \omterm{def:g10:biodiversity-classification:hierarchy}{class} Mammalia; \omterm{def:g10:biodiversity-classification:hierarchy}{order}
Primates; \omterm{def:g10:biodiversity-classification:hierarchy}{family} Hominidae; genus \emph{Homo}; species \emph{Homo
sapiens}.
\end{example}

\begin{method}[Reading a scientific name]\label{met:g10:biodiversity-classification:name}
\begin{enumerate}
  \item Genus first, capitalised; specific epithet second, lower case.
  \item Both italicised (or underlined if handwriting).
  \item After first mention, genus may be abbreviated (\emph{H.\ sapiens}).
  \item The name refers to one species worldwide --- a universal label.
\end{enumerate}
\end{method}

\begin{method}[Using binomial nomenclature]
\label{met:g10:biodiversity-classification:binomial}
\begin{enumerate}
  \item Identify the \omterm{def:g10:scientific-biology:organism}{organism} to species level with keys, experts or \omterm{def:g10:molecules-of-life:nucleic}{DNA} as
        available --- never invent a species name for a real specimen without
        taxonomic process.
  \item Write \emph{\omterm{def:g10:biodiversity-classification:taxonomy}{Genus species}} in italics; include authority only when
        required by formal context.
  \item Use the same binomial in every language community so that
        ``bluebell'' ambiguities disappear.
  \item Remember: the specific epithet is not a stand-alone name; \emph{sapiens}
        alone does not identify a species without its genus.
  \item Closely related species share a genus (e.g.\ \emph{Panthera leo},
        \emph{Panthera tigris}) --- the hierarchy is already in the name.
\end{enumerate}
\end{method}

\begin{example}[Why common names fail]
\label{ex:g10:biodiversity-classification:common-names}
``Robin'' means different birds in Europe and North America; ``panther'' is
used for several big cats. Binomials such as \emph{Erithacus rubecula} and
\emph{Turdus migratorius} remove that ambiguity for \omterm{def:g10:scientific-biology:science}{science} and conservation
lists.
\end{example}

\section{Cladograms and common ancestry}

\begin{definition}[Clade and cladogram]\label{def:g10:biodiversity-classification:clade}
A \emph{clade}\index{clade} is an ancestor and all its descendants. A
\emph{cladogram}\index{cladogram} (or phylogenetic tree, in simple form)
is a branching diagram of hypothesized evolutionary relationships.
Branch points are common ancestors; tips are taxa.
\end{definition}

\begin{definition}[Shared characters]\label{def:g10:biodiversity-classification:characters}
A \emph{shared derived character}\index{shared derived character}
appears in a group and its common ancestor but not in more distant
relatives --- useful for defining \omterm{def:g10:biodiversity-classification:clade}{clades} (e.g.\ feathers in birds).
\emph{Shared ancestral characters}\index{shared ancestral character}
are older and less useful for fine splits (e.g.\ vertebrae for mammals
versus birds).
\end{definition}

\begin{method}[Reading a cladogram]\label{met:g10:biodiversity-classification:read-tree}
\begin{enumerate}
  \item Tips are the \omterm{def:g10:scientific-biology:organism}{organisms} (or groups) being compared.
  \item Find the node where two lineages last meet: that is their most
        recent common ancestor.
  \item Rotate branches around a node without changing relationships.
  \item Do not read left-to-right as ``more advanced''; only branching
        \omterm{def:g10:biodiversity-classification:hierarchy}{order} matters.
\end{enumerate}
\end{method}

\section{Domains of life}

\begin{definition}[Three domains]\label{def:g10:biodiversity-classification:domains}
Modern classification recognises three \emph{domains}\index{domain
(taxonomy)}:
\begin{enumerate}
  \item \emph{Bacteria}\index{Bacteria} --- \omterm{def:g10:the-cell:prok-euk}{prokaryotes} with
        peptidoglycan \omterm{def:g10:the-cell:support}{cell walls} (typical);
  \item \emph{Archaea}\index{Archaea} --- \omterm{def:g10:the-cell:prok-euk}{prokaryotes} with distinct
        membranes and genetics; many extremophiles, but not only;
  \item \emph{Eukarya}\index{Eukarya} --- \omterm{def:g10:scientific-biology:organism}{organisms} with \omterm{def:g10:the-cell:prok-euk}{eukaryotic
        cells} (protists, fungi, plants, animals).
\end{enumerate}
Archaea and Eukarya share a closer evolutionary relationship with each
other than either does with Bacteria, despite both Bacteria and Archaea
being prokaryotic in \omterm{def:g10:the-cell:cell}{cell} plan.
\end{definition}

\begin{proposition}[Classification aims to reflect phylogeny]\label{prop:g10:biodiversity-classification:phylo}
Modern systematics groups \omterm{def:g10:scientific-biology:organism}{organisms} by evolutionary relatedness, not
only by overall similarity. \omterm{def:g10:molecules-of-life:nucleic}{DNA} \omterm{def:g10:molecules-of-life:proteins}{sequences} now supply many characters,
but morphology and \omterm{def:g10:scientific-biology:properties}{development} still matter --- especially for fossils.
\end{proposition}
\admitted

\section{Problem: Naming and placing a new find}

\begin{problem}[{Weekend problem --- from field notes to a tree: apply
species thinking, hierarchy and cladogram logic to a hypothetical
organism}]
\label{pb:g10:biodiversity-classification:1}
Field biologists find a small vertebrate on an island. It has fur, lays
eggs, and \omterm{def:g10:molecules-of-life:nucleic}{DNA} places it closer to echidnas than to any other living mammal.

\medskip
\noindent\textbf{Part I --- Species status.}
\begin{enumerate}
  \item What evidence would support calling it a new species under the
        \omterm{def:g10:biodiversity-classification:species}{biological species concept}?
  \item What evidence would you use if it never breeds in captivity and
        the wild population is tiny?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Classification.}
\begin{enumerate}
  \item Place it as far as you can in the hierarchy (\omterm{def:g10:biodiversity-classification:hierarchy}{domain} through
        \omterm{def:g10:biodiversity-classification:hierarchy}{class} at least), with reasoning.
  \item Propose a binomial (invented) and write it in correct form.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Tree thinking.}
\begin{enumerate}
  \item Sketch a \omterm{def:g10:biodiversity-classification:clade}{cladogram} including: bony fish, lizard, echidna, the
        new \omterm{def:g10:scientific-biology:organism}{organism}, human. Use only nesting of shared ancestry as
        taught here (no need for real branch lengths).
  \item Mark a \omterm{def:g10:biodiversity-classification:clade}{clade} that includes all mammals on your tree and name one
        \omterm{def:g10:biodiversity-classification:characters}{shared derived character} of that \omterm{def:g10:biodiversity-classification:clade}{clade}.
\end{enumerate}
\end{problem}
