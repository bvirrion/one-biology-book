\chapter{Bewijs voor evolutie}\label{ch:g9:evidence-of-evolution}

Twee keer is dit boek aan de rand van een grote bewering blijven
staan: de takken van de verwantschapsboom komen allemaal samen
(\cref{rem:g6:classification-kinship:implies}), en elk \omterm{def:g6:exploring-our-environment:components}{levend
wezen} leest één code
(\cref{rem:g9:genes-and-dna:universal}). De bewering waar ze naar
wijzen heeft een naam --- \emph{evolutie}: \omterm{def:g5:biodiversity:species}{soorten} veranderen van
generatie op generatie, en ze stammen allemaal van
gemeenschappelijke voorouders af. Voordat het volgende hoofdstuk
naar het \emph{hoe} vraagt, doet dit hoofdstuk wat
\cref{exo:g6:classification-kinship:12} eiste: het legt het bewijs
uit.

\section{De getuigenis van de gesteenten}

\begin{definition}[Fossielen]\label{def:g9:evidence-of-evolution:fossils}
Een \emph{fossiel}\index{fossiel} is een bewaard spoor van leven
uit het verleden --- een \omterm{def:g1:animals-around-us:covering}{schelp}, een \omterm{def:g3:bones-and-muscles:skeleton}{bot}, een bladafdruk, een
voetafdruk --- ingesloten in sedimentgesteente. Gesteentelagen
stapelen zich in de volgorde van de tijd, de oudste onderop; naar
boven lezen is geschiedenis lezen. Fossielen zijn de resten uit de
inventaris van
\cref{def:g6:exploring-our-environment:components}, miljoenen
jaren bewaard.
\end{definition}

\begin{proposition}[Wat het gelaagde archief laat zien]\label{prop:g9:evidence-of-evolution:record}
Op volgorde gelezen getuigt het archief drie keer:
\begin{enumerate}
  \item \emph{de fauna's veranderen}: elke laagouderdom heeft
        haar eigen bezetting --- geen konijnen tussen de
        trilobieten, geen trilobieten in de konijnenlagen; de
        meeste \omterm{def:g5:biodiversity:species}{soorten} uit elke diepe laag zijn
        \emph{\omterm{prop:g5:biodiversity:loss}{uitgestorven}} (het woord van
        \cref{prop:g5:biodiversity:loss}, nu met diepe tijd
        erachter);
  \item \emph{de volgorde is genest als de boom}: \omterm{prop:g3:sorting-living-things:groups}{vissen}
        verschijnen in diepe lagen, viervoetige amfibieverwanten
        daarboven, reptielverwanten daarboven, \omterm{prop:g3:sorting-living-things:groups}{zoogdieren} en
        \omterm{prop:g3:sorting-living-things:groups}{vogels} nog hoger --- de dozen van de indeling
        (\cref{pb:g6:classification-kinship:1}) stapelen zich in
        de tijd precies zoals de \omterm{prop:g6:classification-kinship:kinship}{verwantschap} voorspelt;
  \item \emph{er bestaan tussenvormen}: de beroemde steen met
        \emph{Archaeopteryx} --- \omterm{def:g1:animals-around-us:covering}{veren} en vleugels op een
        reptielengestel met tanden en een lange benige \omterm{def:g1:animals-around-us:parts}{staart} ---
        staat waar de takken van vogel en reptiel uiteengaan;
        fossiele walvissen met werkende achterpoten staan waar de
        landzoogdierafstamming van
        \cref{ex:g6:classification-kinship:whale} ze eiste.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[De walvisreeks]\label{ex:g9:evidence-of-evolution:whales}
Het dossier van de walvis, door gesteenten gevuld:
opeenvolgende \omterm{def:g9:evidence-of-evolution:fossils}{fossielen} lopen, op volgorde van tijd, van een
viervoetig kustzoogdier zo groot als een wolf, via vormen met
krimpende poten en sterker wordende \omterm{def:g1:animals-around-us:parts}{staarten}, naar pootloze
volledige zwemmers --- terwijl de walvis van vandaag nog steeds
kleine verborgen heup- en pootbeenderen bouwt
(\cref{ex:g6:classification-kinship:placing} wees op de
overblijfselen), nog steeds zoogt, lucht ademt en als \omterm{def:g5:human-reproduction:pregnancy}{foetus} haar
laat groeien. Voorspelling, \omterm{def:g9:evidence-of-evolution:fossils}{fossielen}, overblijfselen: drie
sleutels, één slot.
\end{example}

\section{De getuigenis van levende lichamen}

\begin{proposition}[De anatomie onthoudt]\label{prop:g9:evidence-of-evolution:anatomy}
Levende lichamen dragen hun geschiedenis met zich mee:
\begin{itemize}
  \item \emph{het gedeelde bouwplan}: de vleugel van een
        vleermuis, de flipper van een walvis, het \omterm{def:g1:my-body:parts}{been} van een
        paard en jouw \omterm{def:g1:my-body:parts}{arm} zijn één beenderopstelling (de
        viervoetsplitsing van
        \cref{pb:g6:classification-kinship:1}), uitgerekt,
        ingekort en van baan veranderd --- afstamming verklaart
        wat verse techniek nooit zou herhalen;
  \item \emph{overblijfselen}: de heupen van de walvis, de
        dichtgegroeide vleugels van de vleugelloze kever, je
        eigen staartbeen en de geslonken staat van de blindedarm
        --- delen die behouden zijn omdat afstamming voortbouwt
        en niet opnieuw begint;
  \item \emph{de \omterm{def:g5:human-reproduction:pregnancy}{embryo}'s stemmen overeen}: de \omterm{def:g5:human-reproduction:pregnancy}{embryo}'s van vis,
        kuiken en mens verlopen door hun eerste weken heen
        opvallend gelijk (het aanleggen van het bouwplan uit
        \cref{prop:g8:from-egg-to-birth:timetable}) --- gedeelde
        bouwinstructies, die pas laat uiteenlopen.
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[De moleculen stemmen]\label{prop:g9:evidence-of-evolution:molecules}
De nieuwste getuige is de code van
\cref{rem:g9:genes-and-dna:universal}. Vergelijk de spelling van
één goed bewaard \omterm{def:g9:genes-and-dna:gene}{gen} tussen \omterm{def:g5:biodiversity:species}{soorten}: mens en chimpansee
verschillen in weinig letters, mens en muis in meer, mens en vis
in nog meer, mens en \omterm{ex:g6:producing-our-food:bread}{gist} het meest --- en de afstanden,
\omterm{def:g9:genes-and-dna:gene}{gen} na \omterm{def:g9:genes-and-dna:gene}{gen} opgeteld, tekenen \emph{dezelfde boom} die de
anatomen uit \omterm{def:g3:bones-and-muscles:skeleton}{botten} en de paleontologen uit lagen tekenden. Drie
onafhankelijke getuigen, één \omterm{def:g6:classification-kinship:tree}{stamboom}: door die overeenstemming
hielden biologen op te twisten over het \emph{of} en wendden ze
zich tot het \emph{hoe}.
\end{proposition}
\admitted

\begin{example}[Het in beweging zien]\label{ex:g9:evidence-of-evolution:live}
Verandering is zelfs te bekijken waar generaties snel wisselen. De
donkere vorm van de berkenspanner verspreidde zich door met roet
bevlekte industriebossen en liep terug toen de lucht schoner werd;
\omterm{prop:g3:sorting-living-things:groups}{insecten} die niet meer voor insecticiden wijken en \omterm{def:g6:producing-our-food:fermentation}{microben} die
antibiotica trotseren
(\cref{ch:g9:microbes-and-infection} zal het medische geval
ontmoeten) ontstaan binnen enkele jaren; de verdeling van
snavelgrootten bij vinken op eilanden verschuift met droogte en
overvloed, live gemeten met ring en schuifmaat. Kleine bewegingen,
korte tijden --- het mechanisme betrapt terwijl het loopt, wat
zijn bereik op de lange duur ook is.
\end{example}

\begin{method}[Historisch bewijs wegen]\label{met:g9:evidence-of-evolution:weigh}
Evolutie is geschiedenis, en historische beweringen worden getoetst
zoals de koude zaak van
\cref{pb:g9:unique-individuals:1} --- door het samenvallen van
onafhankelijke getuigen:
\begin{enumerate}
  \item vraag wat de bewering \emph{voorspelt}: welke
        tussenvormen, welke overblijfselen, welke moleculaire
        afstanden;
  \item toets elke getuige afzonderlijk: gesteenten, lichamen,
        moleculen, waarnemingen in het veld;
  \item weeg hun onafhankelijkheid: getuigen die niet hebben
        kunnen samenspannen en het toch eens zijn,
        vermenigvuldigen de geloofwaardigheid;
  \item houd het oordeel open voor de volgende vondst --- en let
        erop wanneer vondsten blijven landen waar de bewering zei
        te graven (de walvispoten werden gezocht, en gevonden).
\end{enumerate}
\end{method}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:9}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:10}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:evidence-of-evolution:12}

\end{exercise}

\section{Probleem: De nieuwe vleugel van het museum}

\begin{problem}[{Weekendprobleem --- de bewijszaal
inrichten}]\label{pb:g9:evidence-of-evolution:1}
Het natuurhistorisch museum bouwt een evolutievleugel, en de klas
richt vier zalen in: Gesteenten, Lichamen, Moleculen,
Waarnemingen in het veld.

\medskip\noindent\textbf{Deel I --- Zaal voor zaal.}
\begin{enumerate}
  \item Gesteenten: orden de wand --- trilobietensteen, eerste
        vis, sporen van amfibieverwanten, reptielenskeletten,
        afgietsel van \emph{Archaeopteryx}, vroege
        zoogdierenkaak --- en schrijf het bijschrift van de zaal
        in één zin, vanuit
        \cref{prop:g9:evidence-of-evolution:record}.
  \item Lichamen: het middelstuk spreidt vleermuisvleugel,
        walvisflipper, paardenbeen en mensenarm vanuit één
        skeletstandaard uit. Schrijf een bijschrift bij het
        gedeelde bouwplan --- en voeg de drie stukken van de
        overblijfselenkast toe.
  \item Moleculen: de wandkaart telt de letterverschillen van
        één \omterm{def:g9:genes-and-dna:gene}{gen} op: chimpansee 1, muis 8, vis 24, \omterm{ex:g6:producing-our-food:bread}{gist} 41
        (illustratieve aantallen). Waarmee moet de boom van de
        kaart overeenkomen, wil de zaal werken?
  \item Waarnemingen in het veld: kies twee stukken uit
        \cref{ex:g9:evidence-of-evolution:live} en zeg wat elk
        in beweging laat zien.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De vragen van de
bezoekers.}
\begin{enumerate}[resume]
  \item ``Waarom zijn er nog apen, als wij van met hen gedeelde
        voorouders afstammen?'' Antwoord met het vertakken van de
        boom --- neven, geen ladders
        (\cref{def:g6:classification-kinship:tree}).
  \item ``Waar zijn de ontbrekende schakels?'' Wijs de gevonden
        schakels aan --- twee uit dit hoofdstuk --- en zeg
        waartussen elk stond.
  \item ``Is evolutie niet maar een theorie?'' Antwoord met de
        methode van het historische bewijs: wat ``theorie'' hier
        betekent, en wat de overeenstemming van de vier zalen
        vaststelt.
  \item ``Zou de opgraving van morgen het kunnen omverwerpen?''
        Zeg wat voor vondst dat zou doen --- konijnen tussen de
        trilobieten --- en wat de afwezigheid ervan, opgraving na
        opgraving, betekend heeft.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Het slotpaneel van de
vleugel.}
\begin{enumerate}[resume]
  \item Ontwerp het paneel dat de vleugel met de indelingszaal
        hiernaast verbindt: waarom de oude geneste dozen altijd
        al de boom waren.
  \item Voeg de alinea over de moleculen toe: wat de gedeelde
        code over de stam zegt.
  \item Voeg de eerlijke alinea toe: wat de vleugel laat zien
        (verandering, \omterm{prop:g6:classification-kinship:kinship}{verwantschap}, tussenvormen) en wat ze nog
        niet verklaart --- met de naam van de mechanismevraag die
        de volgende zaal zal oppakken.
  \item Sluit af met één zin die een twaalfjarige zal onthouden,
        met daarin \emph{familie} en
        \emph{tijd}.
\end{enumerate}
\end{problem}
