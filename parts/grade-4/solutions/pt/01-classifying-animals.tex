\section*{Capítulo \ref{ch:g4:classifying-animals} --- Classificar os animais}

\begin{solution}{exo:g4:classifying-animals:1}
Um \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} tem um \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto} interno construído em volta de uma
\omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral} (a truta, o gato); um \omterm{def:g4:classifying-animals:vertebrate}{invertebrado} não tem \omterm{def:g4:classifying-animals:vertebrate}{coluna
vertebral} (o caracol, a abelha, a minhoca, a água-viva).
\end{solution}

\begin{solution}{exo:g4:classifying-animals:2}
\omterm{prop:g3:sorting-living-things:groups}{Peixes}: escamas, nadadeiras, respiram na água. \omterm{prop:g3:sorting-living-things:groups}{Anfíbios}: \omterm{def:g1:the-five-senses:sense}{pele} nua e
úmida, \omterm{def:g2:animals-grow-up:born}{ovos} na água, infância de girino. \omterm{prop:g4:classifying-animals:classes}{Répteis}: \omterm{def:g1:the-five-senses:sense}{pele} seca e
escamosa, quase todos pondo \omterm{def:g2:animals-grow-up:born}{ovos} com casca em terra. \omterm{prop:g3:sorting-living-things:groups}{Aves}: \omterm{def:g1:animals-around-us:covering}{penas},
bico, \omterm{def:g2:animals-grow-up:born}{ovos} com casca. \omterm{prop:g3:sorting-living-things:groups}{Mamíferos}: pelo, filhotes alimentados com
\omterm{def:g2:animals-grow-up:born}{leite}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:3}
Sardinha: peixe. Crocodilo: réptil. Salamandra: anfíbio. Ema: ave.
Baleia: mamífero. Minhoca: \omterm{def:g4:classifying-animals:vertebrate}{invertebrado} --- sem \omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral},
logo, sem classe de \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} nenhuma.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:4}
O gato, dentro dos \omterm{prop:g3:sorting-living-things:groups}{mamíferos}, dentro dos \omterm{def:g4:classifying-animals:vertebrate}{vertebrados}, dentro dos
\omterm{def:g1:animals-around-us:animal}{animais}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:5}
A \omterm{def:g1:the-five-senses:sense}{pele} --- seca e escamosa no lagarto, úmida e nua na salamandra ---
e os \omterm{def:g2:animals-grow-up:born}{ovos}: com casca e postos em terra no lagarto, nus e postos na
água na salamandra.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:6}
Os \omterm{def:g4:classifying-animals:vertebrate}{invertebrados}. Por exemplo: \omterm{prop:g3:sorting-living-things:groups}{insetos}, aranhas, caracóis, minhocas
(e águas-vivas).
\end{solution}

\begin{solution}{exo:g4:classifying-animals:7}
Pinguim: tem \omterm{def:g1:animals-around-us:covering}{penas}? Tem --- é uma ave; para na primeira pergunta.
Golfinho: tem \omterm{def:g1:animals-around-us:covering}{penas}? Não. Tem pelo ou \omterm{def:g2:animals-grow-up:born}{leite}? Ele alimenta o filhote
com \omterm{def:g2:animals-grow-up:born}{leite} --- é um mamífero; para aí.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:8}
Por exemplo: \omterm{def:g1:animals-around-us:covering}{penas}, um bico, asas, \omterm{def:g2:animals-grow-up:born}{ovos} com casca --- e uma \omterm{def:g4:classifying-animals:vertebrate}{coluna
vertebral}, já que as \omterm{prop:g3:sorting-living-things:groups}{aves} são \omterm{def:g4:classifying-animals:vertebrate}{vertebrados}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:9}
As perguntas param em ``\omterm{def:g2:animals-grow-up:born}{leite} para os filhotes'': a baleia amamenta
o filhote, então é um mamífero, com pelo ou quase sem ele. ``Vive no
mar'' é um \omterm{def:g2:where-animals-live:habitat}{hábitat}, e não uma característica do corpo --- e a
\omterm{def:g4:classifying-animals:classification}{classificação} foi construída sobre o que um \omterm{def:g1:animals-around-us:animal}{animal} \emph{tem}
justamente porque hábitos e casas podem ser divididos por
desconhecidos: o tubarão e o golfinho dividem um endereço, não uma
classe.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:10}
A \omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral}. A cobra tem uma \omterm{def:g3:bones-and-muscles:skeleton}{coluna} de centenas de ossinhos
--- é um \omterm{def:g4:classifying-animals:vertebrate}{vertebrado}, com a \omterm{def:g1:the-five-senses:sense}{pele} seca e escamosa de um réptil; a
minhoca não tem \omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral} nenhuma --- é um \omterm{def:g4:classifying-animals:vertebrate}{invertebrado}. A
falta de patas é só uma característica que falta; a pergunta da
\omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral} vem primeiro.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:11}
Ganharam os \omterm{prop:g3:sorting-living-things:groups}{mamíferos}: pelo e \omterm{def:g2:animals-grow-up:born}{leite} para os filhotes são o pacote
dos \omterm{prop:g3:sorting-living-things:groups}{mamíferos}, e o ornitorrinco tem os dois --- o \omterm{def:g2:animals-grow-up:born}{ovo} é uma
característica fora do compasso. A \omterm{def:g4:classifying-animals:classification}{classificação} pesa o pacote
inteiro em vez de deixar uma característica sozinha decidir; um
\omterm{def:g1:animals-around-us:animal}{animal} estranho num aspecto ainda fica onde a maioria das
características dele aponta. (O ornitorrinco é estranho o bastante
para que os \omterm{prop:g3:sorting-living-things:groups}{mamíferos} com \omterm{def:g2:animals-grow-up:born}{ovos} formem um subgrupo próprio.)
\end{solution}
