\chapter{Les os et les muscles}\label{ch:g3:bones-and-muscles}

Saute, puis retombe~: quelque chose en toi vient de fonctionner
comme une machine parfaite. Des \omterm{def:g1:plants-around-us:parts}{tiges} rigides ont tenu ta forme, des
charnières ont laissé tes \omterm{def:g1:my-body:parts}{jambes} se plier et se détendre, et des
centaines de petits moteurs ont tiré exactement au bon moment. Les
\omterm{def:g1:plants-around-us:parts}{tiges} sont tes \omterm{def:g3:bones-and-muscles:skeleton}{os}, les charnières tes \omterm{prop:g3:bones-and-muscles:joints}{articulations}, les moteurs tes
\omterm{def:g3:bones-and-muscles:muscle}{muscles} --- l'équipement du corps pour tous les mouvements qu'il
fera jamais.

\section{Le squelette}

\begin{definition}[Squelette et os]\label{def:g3:bones-and-muscles:skeleton}
Le \emph{squelette}\index{squelette} est la charpente d'environ
$200$ \emph{os}\index{os} qui se trouve dans le corps. Ses grandes
régions~:
\begin{itemize}
  \item le \emph{crâne}\index{crâne}, un casque d'os solide autour
        du cerveau~;
  \item la \emph{colonne vertébrale}\index{colonne vertébrale}, une
        colonne souple de petits os empilés du cou aux hanches, qui
        tient le tronc droit~;
  \item les \emph{côtes}\index{côte}, des os courbes qui forment
        une cage autour du cœur et des poumons~;
  \item les \emph{os des \omterm{def:g1:my-body:parts}{membres}}~: de longs os dans les \omterm{def:g1:my-body:parts}{bras} et
        les \omterm{def:g1:my-body:parts}{jambes} --- le fémur est le plus long du corps.
\end{itemize}
\end{definition}

\begin{example}[Sens ta charpente]\label{ex:g3:bones-and-muscles:feel}
Pas besoin de radiographie~: appuie doucement et le \omterm{def:g3:bones-and-muscles:skeleton}{squelette}
répond. Les jointures des doigts~; l'arête du tibia~; la file de
petites bosses au milieu du dos --- la colonne, un petit \omterm{def:g3:bones-and-muscles:skeleton}{os} après
l'autre~; les \omterm{def:g3:bones-and-muscles:skeleton}{côtes} que tu peux compter sous tes doigts. La
charpente est juste sous la \omterm{def:g1:the-five-senses:sense}{peau}.
\end{example}

\begin{remark}[Dur dehors, vivant dedans]\label{rem:g3:bones-and-muscles:alive}
Un \omterm{def:g3:bones-and-muscles:skeleton}{os} n'est pas une matière morte comme un bâton de craie~: il est
vivant, nourri par le sang, il grandit pendant que tu grandis --- et
il sait se réparer. Un \omterm{def:g1:my-body:parts}{bras} cassé dans un plâtre se ressoude parce
que l'\omterm{def:g3:bones-and-muscles:skeleton}{os} lui-même reconstruit lentement la cassure. Les bâtons de
craie ne font jamais cela.
\end{remark}

\section{Les articulations, là où la charpente plie}

\begin{proposition}[Des tiges rigides, des endroits qui plient]\label{prop:g3:bones-and-muscles:joints}
Les \omterm{def:g3:bones-and-muscles:skeleton}{os} eux-mêmes ne plient jamais --- un \omterm{def:g3:bones-and-muscles:skeleton}{os} plié est un \omterm{def:g3:bones-and-muscles:skeleton}{os} cassé.
Tout ce qui plie dans le corps plie aux
\emph{articulations}\index{articulation}, là où deux \omterm{def:g3:bones-and-muscles:skeleton}{os} se
rejoignent~: \omterm{def:g1:my-body:joint}{épaule}, \omterm{def:g1:my-body:joint}{coude}, \omterm{def:g1:my-body:joint}{poignet}, hanche, \omterm{def:g1:my-body:joint}{genou}, \omterm{def:g1:my-body:joint}{cheville}, et les
nombreuses petites \omterm{prop:g3:bones-and-muscles:joints}{articulations} de la colonne, dont les petites
flexions s'additionnent en un dos souple.
\end{proposition}
\admitted

\begin{example}[Le tour de la colonne]\label{ex:g3:bones-and-muscles:spinetrick}
Aucun \omterm{def:g3:bones-and-muscles:skeleton}{os} de ton dos ne peut se recourber --- et pourtant tu peux
t'enrouler presque en boule. La colonne est une pile de plus de $20$
petits \omterm{def:g3:bones-and-muscles:skeleton}{os}, chacun tournant de quelques degrés sur ses voisins~:
beaucoup de petites flexions qui en font une grande. C'est aussi
pourquoi il faut plier les genoux, et non le dos, pour soulever une
charge lourde --- les genoux sont faits pour les grands plis, la
colonne pour les petits.
\end{example}

\section{Les muscles, les moteurs du mouvement}

\begin{definition}[Muscle]\label{def:g3:bones-and-muscles:muscle}
Un \emph{muscle}\index{muscle} est un moteur de chair attaché aux
\omterm{def:g3:bones-and-muscles:skeleton}{os}. Un muscle peut se \emph{contracter}\index{contracter} ---
raccourcir et s'épaissir --- et en se contractant il \emph{tire}
l'\omterm{def:g3:bones-and-muscles:skeleton}{os} auquel il est attaché. Les muscles ne font que tirer~; aucun
muscle ne peut pousser.
\end{definition}

\begin{proposition}[Travailler par paires]\label{prop:g3:bones-and-muscles:pairs}
Comme les \omterm{def:g3:bones-and-muscles:muscle}{muscles} ne font que tirer, bouger une \omterm{def:g1:my-body:joint}{articulation} dans
les deux sens demande une \emph{paire}~: un \omterm{def:g3:bones-and-muscles:muscle}{muscle} à l'avant du \omterm{def:g1:my-body:parts}{bras}
plie le \omterm{def:g1:my-body:joint}{coude}, son partenaire à l'arrière le tend~; chacun se
relâche pendant que l'autre tire. Presque toutes les \omterm{prop:g3:bones-and-muscles:joints}{articulations}
du corps sont mues par de telles paires.
\end{proposition}
\admitted

\begin{example}[Sentir la paire au travail]\label{ex:g3:bones-and-muscles:pair}
Saisis ton \omterm{def:g1:my-body:parts}{bras} avec l'autre main et plie lentement le \omterm{def:g1:my-body:joint}{coude}~: le
\omterm{def:g3:bones-and-muscles:muscle}{muscle} de devant gonfle et durcit --- il se contracte. Tends
maintenant le \omterm{def:g1:my-body:parts}{bras}~: l'avant s'assouplit, et derrière le \omterm{def:g1:my-body:parts}{bras}, c'est
son partenaire qui se raidit. Tu viens de sentir
\cref{prop:g3:bones-and-muscles:pairs} se produire.
\end{example}

\begin{method}[Prendre soin de ses os et de ses muscles]\label{met:g3:bones-and-muscles:care}
\begin{enumerate}
  \item Bouge chaque jour --- les \omterm{def:g3:bones-and-muscles:skeleton}{os} deviennent plus solides, les
        \omterm{def:g3:bones-and-muscles:muscle}{muscles} aussi, avec l'usage~;
  \item mange les aliments des bâtisseurs~: des produits laitiers
        pour les \omterm{def:g3:bones-and-muscles:skeleton}{os}, et des repas variés
        (\cref{ch:g2:teeth-and-eating-well})~;
  \item échauffe-toi avant un effort intense, pour que les \omterm{def:g3:bones-and-muscles:muscle}{muscles}
        s'étirent au lieu de se déchirer~;
  \item soulève les charges lourdes genoux pliés et dos droit~;
  \item casque et protections là où les chutes sont probables~: les
        \omterm{def:g3:bones-and-muscles:skeleton}{os} se réparent, mais lentement.
\end{enumerate}
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:1}
Nomme les quatre régions du \omterm{def:g3:bones-and-muscles:skeleton}{squelette} de
\cref{def:g3:bones-and-muscles:skeleton}, et ce que chacune protège
ou soutient.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:2}
Combien le \omterm{def:g3:bones-and-muscles:skeleton}{squelette} compte-t-il d'\omterm{def:g3:bones-and-muscles:skeleton}{os} environ~? Lequel est le plus
long~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:3}
Un \omterm{def:g3:bones-and-muscles:skeleton}{os} peut-il plier~? Où se produit tout ce qui plie dans le corps~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:4}
La colonne peut s'enrouler alors qu'aucun de ses \omterm{def:g3:bones-and-muscles:skeleton}{os} ne le peut.
Explique son tour de force.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:5}
Que fait un \omterm{def:g3:bones-and-muscles:muscle}{muscle} quand il se contracte~? Que ne pourra jamais
faire aucun \omterm{def:g3:bones-and-muscles:muscle}{muscle}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:6}
Pourquoi plier et tendre le \omterm{def:g1:my-body:joint}{coude} demande-t-il deux \omterm{def:g3:bones-and-muscles:muscle}{muscles}~? Dis ce
que fait chacun à chaque instant.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:7}
Comment un \omterm{def:g1:my-body:parts}{bras} cassé se ressoude-t-il dans un plâtre~? Que montre
cela sur l'\omterm{def:g3:bones-and-muscles:skeleton}{os}, d'après \cref{rem:g3:bones-and-muscles:alive}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:8}
À toi d'essayer~: fais le test de \cref{ex:g3:bones-and-muscles:pair}
sur ton propre \omterm{def:g1:my-body:parts}{bras}. Décris ce que fait l'avant du \omterm{def:g1:my-body:parts}{bras} quand tu
plies, et quand tu tends.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:9}
Pourquoi les adultes disent-ils «~soulève avec les genoux, pas avec
le dos~»~? Réponds avec \cref{ex:g3:bones-and-muscles:spinetrick}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:10}
Une marionnette à fils peut être animée pour saluer~: chaque fil ne
peut que \emph{tirer} son \omterm{def:g1:my-body:parts}{bras} dans un seul sens. En quoi est-ce
exactement la situation des \omterm{def:g3:bones-and-muscles:muscle}{muscles}, et quelle réponse le
marionnettiste y apporte-t-il~? Compare avec
\cref{prop:g3:bones-and-muscles:pairs}.
\end{exercise}
