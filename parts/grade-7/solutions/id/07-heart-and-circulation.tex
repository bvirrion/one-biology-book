\begin{solution}{exo:g7:heart-and-circulation:1}
\omterm{def:g7:heart-and-circulation:rooms}{Serambi} kanan: menerima \omterm{prop:g4:journey-of-food:absorb}{darah} tubuhnya yang kembali. \omterm{def:g7:heart-and-circulation:rooms}{Bilik} kanan:
memompanya ke \omterm{def:g7:how-animals-breathe:four}{paru-parunya}. \omterm{def:g7:heart-and-circulation:rooms}{Serambi} kiri: menerima \omterm{prop:g4:journey-of-food:absorb}{darah} \omterm{def:g7:how-animals-breathe:four}{paru-parunya}
yang kaya \omterm{def:g7:what-is-respiration:respiration}{oksigen}. \omterm{def:g7:heart-and-circulation:rooms}{Bilik} kiri: memompanya berkeliling seluruh tubuhnya
--- yang paling tebal dindingnya di antara keempatnya.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:2}
Keduanya adalah pintu satu arah yang menghentikan \omterm{prop:g4:journey-of-food:absorb}{darah} menyembur
mundur. Lub-dub adalah sepasang penutupannya: \omterm{def:g7:heart-and-circulation:rooms}{katup} masuknya membanting
ketika \omterm{def:g7:heart-and-circulation:rooms}{biliknya} mendorong (lub), \omterm{def:g7:heart-and-circulation:rooms}{katup} keluarnya membanting ketika
\omterm{def:g7:heart-and-circulation:rooms}{biliknya} mengendur (dub).
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:3}
Peredaran paru: dari \omterm{def:g7:heart-and-circulation:rooms}{bilik} kanan ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler} \omterm{def:g7:how-animals-breathe:four}{paru-parunya} (memuat
\omterm{def:g7:what-is-respiration:respiration}{oksigen}, membongkar \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}), lalu kembali ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} kiri.
Peredaran tubuh: dari \omterm{def:g7:heart-and-circulation:rooms}{bilik} kiri ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler} seluruh tubuhnya
(mengantarkan, memungut), lalu kembali ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} kanan.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:4}
\omterm{def:g7:heart-and-circulation:vessels}{Arteri}: dinding yang tebal dan lentur --- membawa \omterm{prop:g4:journey-of-food:absorb}{darah} menjauh dari
\omterm{def:g4:heart-and-blood:heart}{jantung} di bawah tekanan. \omterm{def:g7:heart-and-circulation:vessels}{Kapiler}: dinding setebal satu \omterm{def:g5:first-look-at-cells:cell}{sel} --- lokasi
setiap pertukarannya. \omterm{def:g7:heart-and-circulation:vessels}{Vena}: dinding yang lebar dan lebih tipis dengan
\omterm{def:g7:heart-and-circulation:rooms}{katup} satu arah --- mengembalikan \omterm{prop:g4:journey-of-food:absorb}{darah} ke \omterm{def:g4:heart-and-blood:heart}{jantung} pada tekanan yang
rendah.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:5}
Ia menggerakkan peredaran tubuhnya --- gelang yang jauh lebih besar,
keluar sampai ke jari kaki dan \omterm{def:g5:brain-in-command:system}{otak} lalu kembali --- sehingga ia harus
mendorong paling keras, dan dindingnya dibangun sepadan dengan itu.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:6}
Di dalam \omterm{def:g7:heart-and-circulation:vessels}{kapilernya}. \omterm{def:g7:heart-and-circulation:vessels}{Kapiler} \omterm{def:g7:how-animals-breathe:four}{paru-parunya}: \omterm{def:g7:what-is-respiration:respiration}{oksigen} masuk, \omterm{def:g7:what-is-respiration:respiration}{karbon
dioksida} keluar; \omterm{def:g7:heart-and-circulation:vessels}{kapiler} \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot ususnya}: \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} masuk; \omterm{def:g7:heart-and-circulation:vessels}{kapiler}
\omterm{def:g3:bones-and-muscles:muscle}{ototnya}: bahan bakar dan \omterm{def:g7:what-is-respiration:respiration}{oksigen} turun, \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} naik (\omterm{def:g7:heart-and-circulation:vessels}{kapiler}
\omterm{def:g1:the-five-senses:sense}{kulitnya}: panas keluar).
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:7}
Dari \omterm{def:g7:heart-and-circulation:rooms}{bilik} kanan ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler} paru (\omterm{def:g7:what-is-respiration:respiration}{oksigen} naik), lalu ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} kiri
dan ke \omterm{def:g7:heart-and-circulation:rooms}{bilik} kiri, lalu ke \omterm{def:g7:heart-and-circulation:vessels}{arteri} leher dan ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler} \omterm{def:g5:brain-in-command:system}{otaknya}, lalu
ke \omterm{def:g7:heart-and-circulation:vessels}{vena} leher dan akhirnya ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} kanan. Pertukaran \omterm{def:g5:brain-in-command:system}{otaknya} mengambil bagian \omterm{def:g7:what-is-respiration:respiration}{oksigen} dan \omterm{prop:g7:organs-at-work:bill}{glukosa} yang
besar serta tetap lalu meninggalkan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} --- perintah tetap
milik sang panglima yang menuntut.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:8}
\omterm{prop:g4:heart-and-blood:pulse}{Denyut nadi} adalah regangan lentur dinding \omterm{def:g7:heart-and-circulation:vessels}{arterinya} pada tiap
semburan \omterm{def:g7:heart-and-circulation:rooms}{biliknya} --- tekanan yang tinggi berjumpa dinding tebal yang
lentur. Menjelang \omterm{def:g7:heart-and-circulation:vessels}{venanya}, semburannya sudah terpakai habis untuk
menyeberangi \omterm{def:g7:heart-and-circulation:vessels}{kapilernya}: tekanan yang rendah dan mulus di dalam pipa
lebar yang lunak --- tidak ada lagi yang tersisa untuk berdenyut.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:9}
Kembalian yang mendaki: dari \omterm{def:g1:my-body:parts}{tungkainya}, \omterm{prop:g4:journey-of-food:absorb}{darah} \omterm{def:g7:heart-and-circulation:vessels}{vena} harus memanjat
melawan gravitasi pada tekanan yang rendah, dan \omterm{def:g7:heart-and-circulation:rooms}{katupnya} menahan tiap
kemajuannya. \omterm{def:g3:bones-and-muscles:muscle}{Otot} betis prajurit yang berdiri kaku tidak pernah
memeras \omterm{def:g7:heart-and-circulation:vessels}{venanya}, pendakiannya mandek, \omterm{prop:g4:journey-of-food:absorb}{darahnya} menggenang di
\omterm{def:g1:my-body:parts}{tungkainya} dan pasokan \omterm{def:g5:brain-in-command:system}{otaknya} melorot --- pingsannya. Kaki yang
berbaris adalah pompa \omterm{def:g7:heart-and-circulation:vessels}{vena}.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:10}
\omterm{def:g7:heart-and-circulation:vessels}{Arteri} masuk dari peredaran tubuhnya; di dalamnya, \omterm{def:g7:heart-and-circulation:vessels}{kapiler} \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot
ususnya} menyerap \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} santapannya
(\cref{prop:g7:digestion-and-nutrients:absorption}); sambungan
khususnya: \omterm{def:g7:heart-and-circulation:vessels}{venanya} berjalan lebih dahulu ke \omterm{prop:g7:digestion-and-nutrients:absorption}{hatinya} untuk disortir dan
ditabung, dan baru sesudah itu \omterm{prop:g4:journey-of-food:absorb}{darahnya} bergabung kembali dengan
kembalian umum menuju \omterm{def:g7:heart-and-circulation:rooms}{serambi} kanan. Bagannya menceritakan ulang bab
\omterm{def:g4:journey-of-food:digestion}{pencernaannya}.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:11}
\omterm{prop:g4:journey-of-food:absorb}{Darah} di dalam ruangnya melesat lewat; dinding tebal \omterm{def:g4:heart-and-blood:heart}{jantungnya}
sendiri, yang bekerja tanpa istirahat, membutuhkan pengantaran
\omterm{def:g7:heart-and-circulation:vessels}{kapilernya} sendiri seperti \omterm{def:g3:bones-and-muscles:muscle}{otot} mana pun --- sehingga \omterm{def:g7:heart-and-circulation:vessels}{arteri} kecil
pemberi makan bercabang di atas \omterm{def:g4:heart-and-blood:heart}{jantungnya} sendiri. Kalau mengerak,
keduanya mencekik jalur bahan bakar pompanya sendiri: satu-satunya
organ yang tidak pernah dapat berhenti adalah organ yang dibuat
kelaparan oleh penyumbatannya.
\end{solution}

\begin{solution}{exo:g7:heart-and-circulation:12}
Setiap pertukaran yang menjadi sandaran hidup tubuhnya --- \omterm{def:g7:what-is-respiration:respiration}{oksigen} di
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya}, \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} di \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot ususnya}, bahan bakar dan buangan di
setiap \omterm{def:g3:bones-and-muscles:muscle}{otot} --- hanya terjadi menyeberangi dinding \omterm{def:g7:heart-and-circulation:vessels}{kapiler}; \omterm{def:g7:heart-and-circulation:vessels}{arteri} dan
\omterm{def:g7:heart-and-circulation:vessels}{vena} sekadar membawa \omterm{prop:g4:journey-of-food:absorb}{darah} ke dinding itu lalu mengangkutnya pergi,
dan \omterm{def:g4:heart-and-blood:heart}{jantung} sekadar menjaganya tetap bergerak. Pompa dan jalan rayanya
ada demi jalur sehalus rambutnya: hentikanlah lalu lintas \omterm{def:g7:heart-and-circulation:vessels}{kapilernya} di
mana pun, dan organ itu mati, sebaik apa pun perpipaannya.
\end{solution}

\begin{solution}{pb:g7:heart-and-circulation:1}
\textbf{1.} \omterm{prop:g4:heart-and-blood:pulse}{Denyut nadi}: hitungan semburan \omterm{def:g7:heart-and-circulation:vessels}{arterinya} --- laju pompanya
yang dibaca pada sebuah \omterm{def:g1:my-body:joint}{pergelangan tangan}. Tekanan \omterm{prop:g4:journey-of-food:absorb}{darah}: tekanan
\omterm{def:g7:heart-and-circulation:vessels}{arterinya} ketika menyembur dan di antaranya --- beban jalan rayanya.
Lub-dub pada stetoskopnya: \omterm{def:g7:heart-and-circulation:rooms}{katupnya} yang terdengar menutup ---
kekukuhan pintunya.

\textbf{2.} \omterm{prop:g4:heart-and-blood:pulse}{Denyut nadi} istirahat yang rendah pada pasien yang bugar
berarti sebuah \omterm{def:g7:heart-and-circulation:rooms}{bilik} yang kuat yang memindahkan lebih banyak \omterm{prop:g4:journey-of-food:absorb}{darah} tiap
denyut: \omterm{def:g4:heart-and-blood:heart}{jantung} yang terlatih menganggur dengan tenang, dengan cadangan
di tangan.

\textbf{3.} Pintu \omterm{def:g7:heart-and-circulation:rooms}{katupnya}: \omterm{def:g7:heart-and-circulation:rooms}{katup} masuk ketika \omterm{def:g7:heart-and-circulation:rooms}{biliknya} mendorong
(lub), \omterm{def:g7:heart-and-circulation:rooms}{katup} keluar ketika \omterm{def:g7:heart-and-circulation:rooms}{biliknya} mengendur (dub) --- empat pintu
yang menutup dalam dua pasangan, dengan bersih.

\textbf{4.} Dari \omterm{def:g7:heart-and-circulation:rooms}{serambi} kanan ke \omterm{def:g7:heart-and-circulation:rooms}{bilik} kanan, lalu ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler} paru,
lalu ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} kiri dan ke \omterm{def:g7:heart-and-circulation:rooms}{bilik} kiri, lalu ke \omterm{def:g7:heart-and-circulation:vessels}{arteri} tubuh dan ke
\omterm{def:g7:heart-and-circulation:vessels}{kapiler} tubuh (misalnya pada sebuah \omterm{def:g3:bones-and-muscles:muscle}{otot}), lalu ke \omterm{def:g7:heart-and-circulation:vessels}{vena} --- dan pulang
ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} kanan.

\textbf{5.} Tiap denyutnya kehilangan sebagian semburannya ke belakang
lewat kebocorannya, sehingga pengantaran peredaran tubuhnya tiap
denyut menyusut; \omterm{def:g4:heart-and-blood:heart}{jantungnya} mengimbanginya dengan lebih banyak denyut
dan kerja yang lebih keras, dan daya tahannya turun --- bising
\omterm{def:g4:heart-and-blood:heart}{jantungnya} adalah pengantaran yang hilang sebagai arus balik.

\textbf{6.} Tekanan \omterm{prop:g4:journey-of-food:absorb}{darahnya} naik (jalan raya yang kaku dan menyempit
melawan semburannya) dan regangan \omterm{def:g7:heart-and-circulation:vessels}{arteri} \omterm{prop:g4:heart-and-blood:pulse}{denyut nadinya} berubah; \omterm{def:g7:heart-and-circulation:rooms}{bilik}
kirinya, yang mendorong tiap ronde melawan peredaran yang mengaku,
bekerja lembur lalu menebal.

\textbf{7.} Kerak pada \omterm{def:g7:heart-and-circulation:vessels}{arteri} pemberi makan \omterm{def:g4:heart-and-blood:heart}{jantungnya} sendiri
mencekik jalur bahan bakar \omterm{def:g3:bones-and-muscles:muscle}{otot} yang tidak pernah dapat beristirahat
--- jalan klasik menuju serangan \omterm{def:g4:heart-and-blood:heart}{jantung}.

\textbf{8.} \omterm{def:g7:heart-and-circulation:rooms}{Katup} satu arah \omterm{def:g7:heart-and-circulation:vessels}{venanya} dan pompa \omterm{def:g3:bones-and-muscles:muscle}{otot} dari gerakan yang
hilang: karena berdiri diam, betisnya tidak pernah memeras,
kemajuan \omterm{def:g7:heart-and-circulation:rooms}{katupnya} mandek, dan \omterm{def:g1:my-body:joint}{pergelangan kakinya} menggenang.
Berjalan memeras \omterm{def:g7:heart-and-circulation:vessels}{venanya} secara berirama --- keringanannya bersifat
mekanis.

\textbf{9.} Karena keliling tubuhnya menghabiskan \omterm{def:g7:what-is-respiration:respiration}{oksigen} \omterm{prop:g4:journey-of-food:absorb}{darahnya}:
hanya peredaran parunya yang dapat memuatnya ulang. Melewatkan
\omterm{def:g7:how-animals-breathe:four}{paru-parunya} akan mengirim \omterm{prop:g4:journey-of-food:absorb}{darah} yang terkuras untuk mengantar dengan
van yang kosong --- gelang gandanya adalah isi-ulang-lalu-antar, yang
dipaksakan oleh kedua sisi \omterm{def:g4:heart-and-blood:heart}{jantungnya}.

\textbf{10.} \omterm{def:g7:heart-and-circulation:vessels}{Arteri}: tebal, lentur, bertekanan tinggi --- jalan raya
keluarnya. \omterm{def:g7:heart-and-circulation:vessels}{Kapiler}: setebal satu \omterm{def:g5:first-look-at-cells:cell}{sel}, menyelusupi segalanya --- jalur
pengantaran tempat semua pertukarannya terjadi. \omterm{def:g7:heart-and-circulation:vessels}{Vena}: lebar, berkatup,
bertekanan rendah --- kembaliannya.

\textbf{11.} Masalah pada setiap konsultasinya berarti persis di
tempat ia menyentuh layanan \omterm{def:g7:heart-and-circulation:vessels}{kapilernya}: kebocorannya membuat
pengantarannya kelaparan, keraknya mencekik jalur pasokannya,
genangannya memandekkan kembaliannya. Bacaan, \omterm{def:g7:heart-and-circulation:rooms}{katup} dan jalan rayanya
adalah sarana; lalu lintas \omterm{def:g7:heart-and-circulation:vessels}{kapilernya} --- yaitu pertukarannya ---
adalah tujuannya.

\textbf{12.} Bergeraklah setiap hari --- pompanya menguat dan jalan
rayanya tetap lentur; makanlah dengan seimbang --- dinding \omterm{def:g7:heart-and-circulation:vessels}{arterinya}
tetap bebas dari kerak; jangan ada asap --- \omterm{def:g4:heart-and-blood:vessels}{pembuluhnya} tidak
menyempit dan tidak mengaku, dan kursi \omterm{prop:g4:journey-of-food:absorb}{darahnya} tetap bebas bagi
\omterm{def:g7:what-is-respiration:respiration}{oksigen}.
\end{solution}
