\section*{Bab \ref{ch:b1:membranes-transport} --- Membran dan Pengangkutan Lintas Membran}

\begin{solution}{exo:b1:membranes-transport:1}
Sebuah dwilapis fosfolipid, ekornya ke dalam, dengan kolesterol,
membentuk selembar \qty{5}{nm}. Proteinnya terbenam menembusnya atau
melekat pada satu sisi, sebagai satuan yang terpisah. Lipid dan
sebagian besar proteinnya berdifusi menyamping di dalam bidangnya, yang
merupakan fluida dua dimensi. Kedua lembarnya berbeda susunan lipidnya,
dan rantai gulanya hanya ada pada sisi luar.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:2}
$\mathrm{O_2}$ (kecil, nonpolar) $>$ etanol (kecil, sedikit polar)
$>$ air (kecil, polar) $\gg$ glukosa (besar, polar) $\gg$
$\mathrm{Na^+}$ (bermuatan, terhidrasi). Permeabilitasnya turun bersama
kepolaran, ukuran dan terutama muatan, sebab zat terlarutnya harus larut
dalam teras hidrofobnya.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:3}
Sukrosa: $-2.48\times 0.1 = \qty{-0.25}{MPa}$. NaCl terurai menjadi dua
ion, \qty{0.2}{osmol/L}: \qty{-0.50}{MPa}.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:4}
$E_{Cl} = (61.5/(-1))\log(120/4) = -61.5\times 1.48 = \qty{-91}{mV}$.
Pada \qty{-89}{mV} klorida berada dalam \qty{2}{mV} dari
kesetimbangannya: ia tersebar secara pasif.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:5}
Permukaan \omterm{def:b1:cell-unit-of-life:cell}{selnya} $\num{4.7e9}\times 10^{-10}\,\mathrm{m^2} =
\qty{0.47}{m^2}$; rasionya $0.92/0.47 = 1.96 \approx 2$: lipidnya
menyelimuti \omterm{def:b1:cell-unit-of-life:cell}{selnya} dua kali, jadi membrannya sebuah dwilapis.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:6}
\omterm{def:b1:cell-unit-of-life:cell}{Sel} $\Psi = -0.9 + 0.5 = \qty{-0.4}{MPa}$; larutan \qty{-0.6}{MPa}: air
keluar dari \omterm{def:b1:cell-unit-of-life:cell}{selnya}. Kesetimbangannya ketika $\Psi_{\text{sel}} = -0.6$:
$\Psi_p = -0.6 + 0.9 = \qty{0.3}{MPa}$; \omterm{def:b1:cell-unit-of-life:cell}{sel} itu kehilangan sebagian
\omterm{def:b1:eukaryotic-cell:plantcell}{turgornya} tetapi tetap menegang.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:7}
Melipatduakan kepekatannya dari 10 ke 20 hanya menaikkan lajunya
\qty{15}{\%}: penjenuhan. Memplot $1/v$ terhadap $1/c$ (atau mencatat
bahwa $v = 1.0$ pada \qty{5}{mmol/L} dan sekitar $1.67$ pada $c$ tak
hingga) memberikan $V_{\max} \approx \qty{1.7}{\micro mol/min}$ dan
setengah maksimumnya pada sekitar \qty{3}{mmol/L}: \omterm{def:b1:membranes-transport:transporters}{difusi terbantu} oleh
sebuah pembawa.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:8}
Saat beristirahat permeabilitas membrannya didominasi saluran kalium
yang terbuka; kalium bocor keluar sampai bagian dalamnya cukup negatif
untuk menahannya, yakni dekat $E_K$. Bila saluran natriumnya terbuka,
permeabilitasnya akan didominasi natrium dan potensialnya akan berayun
ke arah $E_{Na}$, sekitar \qty{+60}{mV} --- yakni potensial aksi.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:9}
Tiap mol $\mathrm{Na^+}$ masuk: $RT\ln(12/145) + F(-0.060) = -6.4 - 5.8
= \qty{-12.2}{kJ}$; dua: \qty{-24.4}{kJ}. Glukosa mendaki menuntut biaya
$RT\ln(c_{\text{dalam}}/c_{\text{luar}})$; kesamaannya memberikan
$\ln(c_{\text{dalam}}/c_{\text{luar}}) = 24\,400/2577 = 9.5$, yakni
rasio sekitar \num{13000}.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:10}
Kecairannya turun (ekornya memadat, dwilapisnya mendekati gel); difusi
oksigennya melambat tetapi berlanjut; pompanya, yang sebuah enzim,
nyaris berhenti; kebocorannya berlanjut, sehingga sepanjang beberapa jam
natrium masuk dan kalium keluar dan gradiennya meluruh; potensialnya
turun ke arah nol seiring pudarnya gradien itu; dengan natrium yang
masuk dan tarikan osmotik proteinnya tidak lagi terimbangi, air masuk
dan \omterm{def:b1:cell-unit-of-life:cell}{sel} itu membengkak --- \omterm{def:b1:cell-unit-of-life:cell}{sel} dan \omterm{def:b1:mammal-organization:organ}{organ} simpanan dingin membengkak
justru karena hal ini.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:11}
Pencampuran pada \qty{37}{\celsius} tetapi tidak pada
\qty{4}{\celsius} menyiratkan sebuah proses yang bergantung suhu ---
difusi dalam sebuah fluida yang memadat dalam dingin. Tidak adanya
pencampuran tanpa ATP akan menyiratkan penyebaran ulang yang aktif dan
digerakkan energi (motor, perputaran vesikel) alih-alih difusi. Hasil
yang terkoreksi, bahwa pencampurannya tidak memerlukan ATP, adalah apa
yang dituntut \omterm{def:b1:membranes-transport:membrane}{model mosaik cair}: proteinnya bergerak lewat difusi termal
dalam sebuah fluida dua dimensi, tanpa \omterm{def:b1:cell-unit-of-life:cell}{selnya} melakukan kerja.
\end{solution}

\begin{solution}{exo:b1:membranes-transport:12}
Potensialnya adalah nilai ketika gaya listriknya mengimbangi difusi ion
yang paling tertembuskan, yang diberikan Nernst; \omterm{def:b1:cell-unit-of-life:cell}{sel} itu menetapkan
gradiennya (lewat pompanya) dan permeabilitasnya (lewat salurannya), dan
potensialnya mengikuti. Sebuah membran \qty{1}{\micro F/cm^2} pada
\qty{-70}{mV} membawa $\qty{7e-8}{C/cm^2}$, sekitar $\num{4e11}$ ion
tiap sentimeter persegi --- bagi sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} \qty{20}{\micro m} sekitar
$10^5$ ion, berhadapan dengan $10^{10}$ ion kalium di dalamnya: sebuah
fraksi ion yang dapat diabaikan, yang bergerak sejarak yang dapat
diabaikan, memuati kapasitornya. Gradiennya adalah simpanannya;
potensialnya adalah bacaan atasnya, dan sumbangan langsung pompanya
(tiga muatan keluar bagi dua yang masuk) hanya beberapa milivolt.
\end{solution}

\begin{solution}{pb:b1:membranes-transport:1}
\textbf{1.} $E_{Na} = 26.7\ln(145/12) = \qty{+66}{mV}$; $E_K =
26.7\ln(5/140) = \qty{-89}{mV}$.
\textbf{2.} $\mathrm{K^+}$ (berjarak \qty{29}{mV}) lebih dekat daripada
$\mathrm{Na^+}$ (berjarak \qty{126}{mV}): membrannya jauh lebih
tertembus $\mathrm{K^+}$.
\textbf{3.} $\mathrm{Na^+}$ masuk: $RT\ln(12/145) + F(-0.060) = -6.4 -
5.8 = \qty{-12.2}{kJ/mol}$. $\mathrm{K^+}$ keluar: $RT\ln(5/140) -
F(-0.060) = -8.6 + 5.8 = \qty{-2.8}{kJ/mol}$.
\textbf{4.} Biayanya $= 3\times 12.2 + 2\times 2.8 = \qty{42.2}{kJ}$
tiap daur berhadapan dengan \qty{50}{kJ} tiap ATP: efisiensinya
\qty{84}{\%}.
\textbf{5.} \omterm{def:b1:membranes-transport:transporters}{Pompanya} menjaga natrium \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya} tetap rendah;
\omterm{prop:b1:membranes-transport:secondary}{simporter} pada sisi apikalnya memakai gradien natrium ke dalam untuk
menarik glukosa masuk dari lumen; pompanya harus berada pada sisi yang
lain agar natrium yang diekspornya pergi ke darah, bukan kembali ke
lumen, sebab itu akan membuat natrium lumen berputar kembali dan
perpindahan bersihnya nol.
\textbf{6.} $2\times 12.2 = \qty{24.4}{kJ/mol}$ glukosa.
\textbf{7.} $\Delta G = RT\ln(c_C/c_L)$ (glukosa tidak bermuatan).
\textbf{8.} $\ln(c_C/c_L) = 24\,400/2577 = 9.47$; $c_C/c_L =
\num{1.3e4}$.
\textbf{9.} Sampai \qty{1300}{mmol/L} pada asasnya: rasio itu tak pernah
tercapai karena glukosanya pergi lewat pembawa basolateralnya; dalam
praktik \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya} tinggal dekat \qtyrange{5}{20}{mmol/L}.
\textbf{10.} \omterm{def:b1:eukaryotic-cell:organelle}{Sitosolnya} (di atas \qty{5}{mmol/L}) lebih pekat daripada
darahnya; pembawanya membiarkan glukosa berdifusi menuruni gradien itu
ke dalam darah, secara pasif.
\textbf{11.} Bila dibalik, \omterm{prop:b1:membranes-transport:secondary}{simporternya} akan memompa glukosa dari darah
ke dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya} dan pembawanya akan membiarkannya bocor ke dalam lumen:
ususnya akan menyekresikan glukosa. Kekutuban pengangkutnya adalah arah
penyerapannya.
\textbf{12.} \omterm{prop:b1:membranes-transport:secondary}{Simporternya} hanya menyerap natrium bersama glukosa; karena
itu glukosa di dalam lumen menggerakkan penyerapan natrium lewatnya
bahkan ketika racun kolera menghambat rute yang lain; klorida mengikuti
natrium secara listrik dan air mengikuti garamnya secara osmotik.
\textbf{13.} $\qty{4e-15}{mol/s}$ $\mathrm{Na^+}$, yakni $\num{2.4e9}$
ion tiap sekon; pada tiga tiap daur, $\num{8e8}$ daur tiap sekon.
\textbf{14.} $\num{8e8}$ ATP tiap sekon $= \qty{1.3e-15}{mol/s}$; pada
30 ATP tiap glukosa, \qty{4.4e-17}{mol/s} glukosa, yakni \qty{2.2}{\%}
glukosa yang terserap.
\textbf{15.} $\num{1.2e9}$ glukosa tiap sekon pada 50 tiap \omterm{prop:b1:membranes-transport:secondary}{simporter}:
$\num{2.4e7}$ \omterm{prop:b1:membranes-transport:secondary}{simporter}; membrannya \qty{500}{\micro m^2}: \num{48000}
tiap mikrometer persegi, yakni satu tiap persegi \qty{20}{nm} --- sebuah
membran yang padat berisi pengangkut.
\textbf{16.} Osmol yang masuk: $4\times\qty{2e-15}{mol/s} =
\qty{8e-15}{osmol/s}$; pada \qty{0.3}{osmol/L}, airnya
$= \qty{2.7e-14}{L/s} = \qty{27}{\micro m^3/s}$. Volume \omterm{def:b1:cell-unit-of-life:cell}{selnya}
$25\times 5\times 5 = \qty{625}{\micro m^3}$: berlipat dua dalam
\qty{23}{s}. Airnya pergi secepat masuknya, melintasi membran
basolateralnya, dan beginilah usus menyerap air.
\textbf{17.} $\qty{2.7e-14}{L/s}$ air adalah \qty{1.5e-12}{mol/s}; tiap glukosa ($\qty{2e-15}{mol/s}$): sekitar 750 molekul air.
\textbf{18.} $S = 2(25\times 5 + 25\times 5 + 5\times 5) =
\qty{550}{\micro m^2} = \qty{5.5e-6}{cm^2}$; $C = \qty{5.5e-12}{F}$.
\textbf{19.} $Q = CV = 5.5\times 10^{-12}\times 0.060 =
\qty{3.3e-13}{C}$; $/\num{1.6e-19} = \num{2.1e6}$ ion.
\textbf{20.} $\num{2.1e6}/550 = 3800$ ion tiap mikrometer persegi, satu kelebihan muatan tiap persegi \qty{16}{nm} membran.
\textbf{21.} $\mathrm{K^+}$ di dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya}: $0.140\,\mathrm{mol/L}\times
\qty{6.25e-13}{L}\times\num{6e23} = \num{5.3e10}$ ion; muatannya
memerlukan \qty{0.004}{\%} di antaranya: kepekatannya tidak berubah oleh
pemuatan membrannya.
\textbf{22.} Muatan bersih tiap daur $+e$ keluar: arusnya $\num{8e8}\times
\num{1.6e-19} = \qty{1.3e-10}{A}$; dalam satu sekon membrannya akan
memperoleh \qty{1.3e-10}{C}, yakni $\Delta V = Q/C = \qty{24}{V}$. Hal
itu tidak terjadi, karena tiap muatan yang dipompa keluar segera
diimbangi ion yang mengalir kembali lewat salurannya (terutama
$\mathrm{K^+}$ yang lebih sedikit bocor masuk, atau $\mathrm{Na^+}$ yang
masuk kembali): konduktansi membrannya memintas arus pompanya, dan
pompanya hanya menambahkan beberapa milivolt.
\textbf{23.} Natrium \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya} naik sepanjang beberapa menit;
\omterm{prop:b1:membranes-transport:secondary}{simporternya} kehilangan gaya penggeraknya dan penyerapan glukosanya
berhenti; potensialnya meluruh ke arah nol seiring habisnya gradiennya;
natrium dan air masuk dan \omterm{def:b1:cell-unit-of-life:cell}{sel} itu membengkak.
\textbf{24.} Dengan kebocoran kaliumnya terhambat, kebocoran natriumnya
mendominasi: potensialnya bergerak ke arah $E_{Na}$, sehingga jauh
kurang negatif (depolarisasi).
\textbf{25.} Sebuah rasio sekitar \num{13000} (\omterm{def:b1:eukaryotic-cell:organelle}{sitosol} atas lumen), yang
ditetapkan dua ion natrium tiap glukosa pada \qty{-60}{mV}; dimungkinkan
oleh \omterm{def:b1:membranes-transport:transporters}{pompa natrium--kalium} (basolateral) dan \omterm{prop:b1:membranes-transport:secondary}{simporter} natrium--glukosa
(apikal).
\end{solution}
