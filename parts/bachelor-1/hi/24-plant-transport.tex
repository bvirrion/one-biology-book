\chapter{पादपों में गैस-विनिमय और रस-परिवहन}\label{ch:b1:plant-transport}

ज़मीन से सौ मीटर ऊपर किसी सिकोया की चोटी पर जल को \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} से होते
वाष्पन द्वारा मिट्टी से बाहर खींचा जा रहा है, बाल से भी पतली मृत
\omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} के एक स्तंभ में ऊपर, ऐसे तनाव के नीचे जो उतने ही अनुप्रस्थ काट
के किसी इस्पात के तार को तोड़ डालता। कोई पंप नहीं है; जल को वही
\omterm{def:b1:water-small-molecules:water}{हाइड्रोजन बंध} ढोते हैं जो उसे स्वयं से चिपकाए रखते हैं। इसी बीच पहली के
बगल में नलियों के एक दूसरे समूह में, उन्हीं \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} की बनाई शर्करा
दूसरी दिशा में बह रही है, उसी दाब से ठेली जो \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} स्वयं पैदा करती
हैं। किसी पौधे की दोनों नलसाज़ी प्रणालियाँ, \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} और \omterm{def:b1:plant-transport:phloem}{फ्लोएम}, उस भौतिकी
से चलती हैं जो पौधा खड़ी करके चलने के लिए छोड़ देता है। यह अध्याय उन
\omterm{def:b1:flowering-plant-organization:tissues}{रंध्रों} का, जिनसे कोई \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} जल देकर कार्बन डाइऑक्साइड लेती है, रस को
उठाने वाली संसंजन--तनाव क्रियाविधि का, उसे ढोने वाली वाहिकाओं का, शर्करा
को चलाने वाली \omterm{thm:b1:plant-transport:pressureflow}{दाब-प्रवाह क्रियाविधि} का, और पीने तथा खाने के बीच के उस
समझौते का वर्णन करता है जो सब पर शासन करता है।

\section{रंध्र: द्वार}

\begin{definition}[रंध्र, द्वार कोशिकाएँ]\label{def:b1:plant-transport:stoma}
\index{रंध्र}\index{द्वार कोशिका}
\emph{रंध्र} किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} (या किसी हरे \omterm{def:b1:flowering-plant-organization:organs}{तने}) के \omterm{def:b1:flowering-plant-organization:tissues}{अधिचर्म} में एक छिद्र है जो
दो \emph{द्वार \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं}} से घिरा होता है, वृक्काकार \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} जिनकी
दीवारें छिद्र की ओर वाली तरफ़ मोटी होती हैं और जिनके \omterm{def:b1:carbohydrates:polysaccharide}{सेलुलोस} के
सूक्ष्मतंतु उनके चारों ओर छल्लों की तरह दौड़ते हैं। जब द्वार \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} जल
लेकर फूलती हैं, तो छल्ले उन्हें चौड़ा नहीं होने देते और उन्हें अलग-अलग
मुड़ने पर मजबूर करते हैं: छिद्र खुल जाता है। जब वे जल खोती हैं, तो वह
बंद हो जाता है। कोई \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} प्रति वर्ग मिलीमीटर कुछ सौ रंध्र ढोती है,
अधिकतर अपने निचले पृष्ठ पर; खुले हुए, उनके छिद्र \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} के क्षेत्रफल के
एक-दो प्रतिशत होते हैं और उसका लगभग सारा गैस-विनिमय उन्हीं से गुज़रता है।
\end{definition}

\begin{proposition}[रंध्र कैसे खुलता है, और कब]\label{prop:b1:plant-transport:opening}
\index{ऐब्सिसिक अम्ल}
\omterm{def:b1:plant-transport:stoma}{द्वार कोशिकाएँ} प्रोटॉन बाहर पंप करके छिद्र खोलती हैं (\cref{ch:b1:plant-water-minerals}),
जिससे पोटैशियम और क्लोराइड भीतर आते हैं और भीतर मैलेट बनता है; ये विलेय
उनका जल-विभव नीचा करते हैं, पड़ोसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} से जल पीछे आता है, और उनकी
\omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} एक मेगापास्कल चढ़ जाती है। आयन-अभिवाह उलट देने पर छिद्र मिनटों में
बंद हो जाता है। संकेत ये हैं: \emph{प्रकाश} (सीधे \omterm{def:b1:plant-transport:stoma}{द्वार कोशिकाओं} पर नीला
प्रकाश, और प्रकाश-संश्लेषण शुरू होते ही भीतरी $\mathrm{CO_2}$ का गिरना)
खोलता है; \emph{ऊँची भीतरी $\mathrm{CO_2}$} बंद करती है;
\emph{जल-प्रतिबल} बंद करता है, \emph{\omterm{prop:b1:plant-transport:opening}{ऐब्सिसिक अम्ल}} हार्मोन के रास्ते,
जो मुरझाती \omterm{def:b1:flowering-plant-organization:organs}{जड़ों} और \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} में बनता है और मिनटों के भीतर \omterm{def:b1:plant-transport:stoma}{द्वार
कोशिकाओं} से आयन निकलवा देता है; सूखी हवा बंद करती है। \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} अपना द्वार तब
खोलती है जब कार्बन जल के मोल का हो और तब बंद कर देती है जब जल दुर्लभ हो
--- और अधिकांश पौधे ऐसा एक दैनिक लय में करते हैं, भोर में खोलते और साँझ
में बंद करते, जो अचर प्रकाश में भी दिनों तक बनी रहती है।
\end{proposition}

\section{जल उठाना: संसंजन और तनाव}

\begin{theorem}[संसंजन--तनाव क्रियाविधि]\label{thm:b1:plant-transport:cohesiontension}
\index{संसंजन--तनाव सिद्धांत}\index{वाष्पोत्सर्जन}
जल \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} में इसलिए चढ़ता है कि उसे ऊपर से \emph{खींचा} जाता है, नीचे से
ठेला नहीं जाता। \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} की मध्योतक \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} की गीली दीवारों से वायु-अवकाशों
में वाष्पन (\emph{वाष्पोत्सर्जन}) दीवार के महीन छिद्रों के जल को वक्र
अर्धचंद्रकों में खींचता है, जिनका पृष्ठ-तनाव उनके पीछे के जल को तनाव में
डाल देता है; यह तनाव \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} में संसंजन से (\cref{ch:b1:water-small-molecules}
के \omterm{def:b1:water-small-molecules:water}{हाइड्रोजन बंधों} से) बँधे जल के सतत स्तंभों से होकर \omterm{def:b1:flowering-plant-organization:organs}{तने} के नीचे और
\omterm{def:b1:flowering-plant-organization:organs}{जड़ों} तक संचरित होता है, और मिट्टी से जल उठाता है। $h$ ऊँचाई के किसी
स्तंभ को गुरुत्व के विरुद्ध थामे रखने के लिए $\rho g h$ तनाव चाहिए ---
प्रति सौ मीटर \qty{1}{MPa} --- जोड़ उतना जितना बहाव के घर्षण पर पार पाने
में लगे, लगभग उतना ही फिर से; किसी ऊँचे वृक्ष के स्तंभ
\qtyrange{-2}{-3}{MPa} पर होते हैं, जो किसी महीन नली में जल की तनन-सामर्थ्य
के भीतर ही है।
\end{theorem}
\begin{proof}[साक्ष्य]
डिक्सन और जॉली (1894) ने दिखाया कि किसी बंद काँच की नली में जल कई
मेगापास्कल का तनाव बिना टूटे सह लेता है, और यह भी कि पारे में डूबी जल की
किसी नली से जुड़ी वाष्पोत्सर्जन करती कोई पत्तीदार शाखा पारे को वायुमंडलीय
दाब से जितना उठा सकता उससे ऊँचा उठा देती है। शोलैंडर का \emph{दाब कक्ष}
(1965) तनाव सीधे नापता है: कटी हुई किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} को उसका वृंत बाहर निकाले
हुए एक कक्ष में बंद किया जाता है, और जो गैस-दाब रस को ठीक कटे पृष्ठ तक
वापस ठेल दे वही उस तनाव के बराबर है जिसमें रस था --- दोपहर में
भरपूर सींची किसी फ़सल में \qtyrange{0.5}{1}{MPa}, किसी मरुस्थली झाड़ी या
किसी ऊँचे वृक्ष के शिखर में \qtyrange{2}{4}{MPa}। कोई वृक्षमापी दिखाता है
कि किसी वृक्ष का \omterm{def:b1:flowering-plant-organization:organs}{तना} दिन में सिकुड़ता है, जब तनाव वाहिकाओं को दबाता है,
और रात में फूलता है; लकड़ी में लगा कोई तापमापी दिखाता है कि रस तब सबसे
तेज़ बहता है जब वाष्पोत्सर्जन सबसे ऊँचा हो, और रात में रुक जाता है। किसी
जीवित \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की ज़रूरत नहीं: विष या ऊष्मा से मारा गया \omterm{def:b1:flowering-plant-organization:organs}{तना} भी चालन करता
है।
\end{proof}

\begin{definition}[जाइलम की नलिकाएँ, गुहिकायन]\label{def:b1:plant-transport:xylem}
\index{जाइलम वाहिका}\index{गुहिकायन}
जल \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} की \emph{\omterm{def:b1:flowering-plant-organization:tissues}{वाहिनिकाओं}} और \emph{वाहिकाओं} में चलता है
(\cref{ch:b1:flowering-plant-organization}): मृत \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} जिनसे उनकी
सामग्री खाली कर दी गई है, जिनकी लिग्निनी दीवारें छल्लों और सर्पिलों में
मोटी की गई हैं ताकि वे भीतर के तनाव से चिपक न जाएँ, और जो सिरे से सिरा
वेधों (वाहिकाएँ) या अपनी बगल की दीवारों के गर्तों (वाहिनिकाएँ) से जुड़ी
हैं। कोई वाहिका \qtyrange{20}{300}{\micro m} चौड़ी और सेंटीमीटरों से
मीटरों तक लंबी हो सकती है: चौड़ी नलियाँ कहीं अधिक ढोती हैं (किसी नली से
बहाव उसकी त्रिज्या की चौथी घात के अनुपात में चढ़ता है) पर
\emph{गुहिकायन} के प्रति अधिक भंगुर होती हैं --- तनाव के नीचे जल में
किसी गैस-बुलबुले का अचानक बन जाना, जो उस नलिका को खाली कर देता है और उसे
सेवा से बाहर कर देता है। नलिकाओं के बीच के गर्त बुलबुलों को फैलने से
रोकते हैं; तुषारण और सूखा दोनों गुहिकायन कराते हैं, और किसी वृक्ष की
वाहिकाएँ वसंत में \omterm{prop:b1:plant-water-minerals:rootpressure}{मूल दाब} से आंशिक रूप से फिर भर जाती हैं या हर साल नई
लकड़ी से बदल दी जाती हैं।
\end{definition}

\begin{example}[वृक्ष की चोटी]\label{ex:b1:plant-transport:top}

वाहिकाओं का घर्षण दिन में उतना ही और जोड़ देता है, इसलिए शिखर पर \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम}

लिए, उससे भी नीची: उनके \omterm{def:b1:flowering-plant-organization:tissues}{रंध्रों} को दिन में पहले बंद होना पड़ता है और
उनका प्रकाश-संश्लेषण किसी नीची शाखा से धीमा होता है, और चोटी की \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ}
छोटी तथा मोटी होती हैं। सबसे ऊँचे वृक्षों की ऊँचाई, लगभग \qty{120}{m},
शायद इसी से तय होती है: उससे ऊपर \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} किसी सूखी दोपहर में बिना
\omterm{def:b1:plant-transport:xylem}{गुहिकायन} के तनाव नहीं थाम सकतीं।
\end{example}

\section{शर्करा चलाना: दाब-प्रवाह}

\begin{definition}[फ्लोएम, चालनी नलिकाएँ, स्रोत और ग्राही]\label{def:b1:plant-transport:phloem}
\index{फ्लोएम}\index{चालनी नलिका}\index{स्रोत}\index{ग्राही}
शर्करा \emph{फ्लोएम} में, \emph{\omterm{def:b1:flowering-plant-organization:tissues}{चालनी नलिकाओं}} में चलती है: जीवित
\omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} की पंक्तियाँ जिन्होंने अपने केंद्रक और अधिकांश \omterm{def:b1:eukaryotic-cell:organelle}{अंगक} खो दिए हैं
पर अपनी झिल्लियाँ रखी हैं, और जो सिरे से सिरा छिद्रित \emph{चालनी
पट्टिकाओं} से जुड़ी हैं, हर \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को एक \emph{सहचर \omterm{def:b1:cell-unit-of-life:cell}{कोशिका}} पालती है जो
उसके \omterm{def:b1:proteins:peptide}{प्रोटीन} और \omterm{def:b1:water-small-molecules:atp}{ATP} देती है। रस \qtyrange{10}{30}{\%} सुक्रोस होता है,
साथ में \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल}, आयन और संकेतक अणु, और वह \emph{स्रोतों} ---
प्रकाश-संश्लेषण करती \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ}, या खाली होते संचय-अंग --- से
\emph{ग्राहियों} --- बढ़ते शीर्ष, \omterm{def:b1:flowering-plant-organization:organs}{जड़ें}, फल, बीज, भरे जाते संचय-अंग ---
तक \qtyrange{0.5}{1}{m/h} पर बहता है, किसी भी दिशा में, और अलग-अलग
नलिकाओं में अलग-अलग दिशाओं में।
\end{definition}

\begin{theorem}[दाब-प्रवाह]\label{thm:b1:plant-transport:pressureflow}
\index{दाब-प्रवाह क्रियाविधि}
\omterm{def:b1:plant-transport:phloem}{फ्लोएम} का रस स्रोत और ग्राही के बीच द्रवस्थैतिक दाब के अंतर से चलते थोक
बहाव द्वारा चलता है (म्युंश, 1930)। स्रोत पर सहचर \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} सुक्रोस को
उसकी प्रवणता के विरुद्ध \omterm{def:b1:flowering-plant-organization:tissues}{चालनी नलिकाओं} में \emph{लादती} हैं (प्रोटॉन
सहवहन से, \omterm{def:b1:plant-water-minerals:pathways}{अपप्रवेश्य पथ} से, या जीवद्रव्यतंतुओं से होकर); नलिका का
जल-विभव गिरता है, पड़ोसी \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} से जल घुसता है, और \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} चढ़कर
\qty{1}{MPa} या उससे अधिक हो जाती है। ग्राही पर सुक्रोस \emph{उतारा}
जाता है और खर्च या संचित होता है, जल-विभव चढ़ता है, जल निकलता है, और
\omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} नीची रहती है। उनके बीच रस चालनी पट्टिकाओं से होकर दाब-प्रवणता के
नीचे बहता है, और जो कुछ उसमें घुला है सब ढो ले जाता है। पौधा ऊर्जा केवल
दोनों सिरों पर खर्च करता है; बहाव स्वयं निष्क्रिय है।
\end{theorem}
\begin{proof}[साक्ष्य]
किसी \omterm{def:b1:flowering-plant-organization:organs}{तने} पर भोजन करता कोई एफ़िड अपनी सूची एक ही \omterm{def:b1:plant-transport:phloem}{चालनी नलिका} में घुसा
देता है; एफ़िड को काटकर हटा दीजिए और वहीं छोड़ी गई सूची नलिका के अपने ही
दाब से घंटों रस बहाती रहती है --- शुद्ध \omterm{def:b1:plant-transport:phloem}{फ्लोएम} रस, जिसकी सुक्रोस-सांद्रता
और बहाव-दर नापी जा सकती है, और जिसका दाब, सूची पर लगे किसी दाबमापी से
नापा हुआ, किसी स्रोत के पास लगभग \qty{1}{MPa} और ग्राही की ओर नीचा होता
है। किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} को खिलाई गई अंकित $\mathrm{^{14}CO_2}$ मिनटों के भीतर \omterm{def:b1:flowering-plant-organization:organs}{तने}
के \omterm{def:b1:plant-transport:phloem}{फ्लोएम} में प्रकट होती है और उसी चाल से चलती है जिसकी भविष्यवाणी सूची
का बहाव करता है; \omterm{def:b1:flowering-plant-organization:organs}{तने} का कोई हिस्सा ठंडा करना, जो जीवित \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} को धीमा
करता है पर किसी दाब-चालित बहाव पर मुश्किल से असर डालता है, परिवहन को
ज़रा ही धीमा करता है; स्रोत-पत्ती का लदान विषाक्त कर देने पर वह रुक जाता
है।
\end{proof}

\begin{method}[किसी स्थानांतरण प्रयोग को पढ़ना]\label{met:b1:plant-transport:translocation}
\begin{enumerate}
  \item किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} को अंकित $\mathrm{CO_2}$ खिलाइए; \omterm{def:b1:flowering-plant-organization:organs}{तने} के ऊपर और नीचे
        अंतराल पर नमूने लीजिए: समय के सापेक्ष अंक की स्थिति बहाव की
        दिशा और चाल देती है।
  \item \omterm{def:b1:flowering-plant-organization:organs}{तने} पर वलय काटिए (छाल का एक छल्ला हटाइए, जो \omterm{def:b1:plant-transport:phloem}{फ्लोएम} ढोती है,
        \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} छोड़ते हुए): छल्ले के ऊपर शर्करा जमा होती है और नीचे
        \omterm{def:b1:flowering-plant-organization:organs}{जड़ें} भूखी रहती हैं --- उसे \omterm{def:b1:plant-transport:phloem}{फ्लोएम} ढोता है, \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} नहीं; वृक्ष एक
        ही मौसम में मर जाता है।
  \item दो ऊँचाइयों पर कटी एफ़िड-सूचियों से रस इकट्ठा कीजिए: उनके बीच
        सुक्रोस और दाब की गिरावट ही वह प्रवणता है जो बहाव चलाती है।
  \item ग्राही बदलिए: फल हटा दीजिए, और अंक \omterm{def:b1:flowering-plant-organization:organs}{जड़ों} को चला जाता है; \omterm{def:b1:flowering-plant-organization:organs}{पत्ती}
        को छाया में कर दीजिए, और वह स्वयं ग्राही बन जाती है। बहाव का
        ढर्रा माँग का ढर्रा है।
\end{enumerate}
\end{method}

\begin{example}[दो ऋतुओं में एक आलू]\label{ex:b1:plant-transport:potato}
गरमी में \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} स्रोत होती हैं और ज़मीन के नीचे के कंद ग्राही: सुक्रोस
नीचे बहता है, उतारा जाता है, \omterm{def:b1:carbohydrates:polysaccharide}{मंड} में बदला जाता है, और कंद फूल जाते हैं।
वसंत में कंद अंकुरित होता है: उसका \omterm{def:b1:carbohydrates:polysaccharide}{मंड} फिर से सुक्रोस में बदला जाता है,
\omterm{def:b1:plant-transport:phloem}{फ्लोएम} में लादा जाता है, और बढ़ते प्ररोह तक ऊपर बहता है --- वही नलिका,
उलटी दिशा, क्योंकि स्रोत और ग्राही ने जगहें बदल ली हैं।
\end{example}

\section{समझौता}

\begin{proposition}[जल-उपयोग दक्षता]\label{prop:b1:plant-transport:wue}
\index{जल-उपयोग दक्षता}
जो भी \omterm{def:b1:plant-transport:stoma}{रंध्र} कार्बन डाइऑक्साइड भीतर आने देता है वह जल खोता है, और यह
विनिमय असमान है: \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} का भीतरी भाग वाष्प से संतृप्त है और बाहर की हवा
सूखी, जबकि $\mathrm{CO_2}$ बाहर \qty{0.04}{\%} पर और भीतर उससे नीची है,
इसलिए जल की बाहरी प्रवणता कार्बन डाइऑक्साइड की भीतरी प्रवणता से सैकड़ों
गुनी है। कोई C3 \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} प्रति बंधी $\mathrm{CO_2}$ जल के
\qtyrange{200}{500}{} अणु खोती है; कोई C4 \omterm{def:b1:flowering-plant-organization:organs}{पत्ती}, जो $\mathrm{CO_2}$
सांद्रित करती और अपने \omterm{def:b1:plant-transport:stoma}{रंध्र} सँकरे रखती है, \qtyrange{100}{200}{}; कोई \omterm{def:b1:photosynthesis:c4cam}{CAM
पौधा}, जो केवल रात में खोलता है, \qtyrange{20}{50}{}
(\cref{ch:b1:photosynthesis})। \emph{\omterm{prop:b1:plant-transport:wue}{जल-उपयोग दक्षता}} प्रति खोए जल कमाया
कार्बन है, और उसका सुधार --- दोपहर में बंद करके, धँसे \omterm{def:b1:flowering-plant-organization:tissues}{रंध्रों} और मोटी
उपत्वचाओं से, C4 तथा CAM पथों से --- किसी सूखे वायुमंडल से हुए सौदे में
पौधे का पक्ष है।
\end{proposition}

\begin{example}[अंकों में यह सौदा]\label{ex:b1:plant-transport:bargain}
दिन में डेढ़ लीटर वाष्पोत्सर्जन करता मक्का का कोई पौधा लगभग \qty{10}{g}
कार्बन डाइऑक्साइड बाँधता है: \qty{0.23}{mol} $\mathrm{CO_2}$ के लिए
\qty{80}{mol} जल, तीन सौ पचास के मुकाबले एक। किसी मौसम में गेहूँ का कोई
खेत आठ टन अनाज बनाने के लिए प्रति हेक्टेयर पाँच सौ टन जल वाष्पोत्सर्जित
करता है; जल कार्बन का दाम है, और दुनिया के अधिकांश खेतों में वही दाम है
जो फ़सल की सीमा बाँधता है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:plant-transport:1}
समझाइए कि \omterm{def:b1:plant-transport:stoma}{द्वार कोशिकाओं} का आकार और उनकी दीवारें \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} की चढ़ाई को किसी
खुले छिद्र में कैसे बदल देती हैं।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:plant-transport:2}
संसंजन--तनाव क्रियाविधि तीन वाक्यों में बताइए: बल कहाँ से आता है, उसे
क्या संचरित करता है, वह \omterm{def:b1:flowering-plant-organization:organs}{जड़} पर क्या करता है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:plant-transport:3}
दैनिक आलेख से बताइए कि किस घंटे वाष्पोत्सर्जन सबसे ऊँचा है, \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} का
जल-विभव अपने न्यूनतम पर कब आता है, और \omterm{def:b1:plant-transport:stoma}{रंध्र} दोपहर में आंशिक रूप से क्यों
बंद हो जाते हैं।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:plant-transport:4}
स्रोत और ग्राही की परिभाषा दीजिए, और बताइए कि \omterm{thm:b1:plant-transport:pressureflow}{दाब-प्रवाह क्रियाविधि} में
हर एक पर क्या होता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:plant-transport:5}
\qty{30}{m} ऊँचे जल के किसी स्तंभ को थामने के लिए चाहिए तनाव निकालिए
($\rho g h$), और घर्षण उसे दुगना कर दे तो कुल भी। उस ऊँचाई पर \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों}
का जल-विभव कम से कम कितना होना ही चाहिए?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:plant-transport:6}
किसी नली से बहाव $r^4$ के अनुपात में चलता है। \qty{100}{\micro m}
त्रिज्या वाली एक वाहिका से बहाव की तुलना उतने ही कुल अनुप्रस्थ काट की
\qty{10}{\micro m} \omterm{def:b1:flowering-plant-organization:tissues}{वाहिनिकाओं} की संख्या से होते बहाव से कीजिए। पौधे
सँकरी वाली क्यों रखते हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:plant-transport:7}
\qty{20}{\%} सुक्रोस (\qty{0.6}{mol/L}) वाले \omterm{def:b1:plant-transport:phloem}{फ्लोएम} रस का विलेय-विभव
लगभग \qty{-1.5}{MPa} होता है। \qty{-0.5}{MPa} पर किसी \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} के बगल में
\omterm{def:b1:plant-transport:phloem}{चालनी नलिका} साम्य पर कौन-सी \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} पा लेती है? किसी ग्राही पर जहाँ रस
\qty{5}{\%} सुक्रोस ($\Psi_s = \qty{-0.4}{MPa}$) हो और वही \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} बगल में
हो, कौन-सी \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति}? कौन-सा दाब-अंतर बहाव चलाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:plant-transport:8}
कोई एफ़िड-सूची \qty{20}{\%} सुक्रोस वाला \qty{1}{\micro L} रस प्रति घंटा
बहाती है। यदि \omterm{def:b1:plant-transport:phloem}{चालनी नलिका} की त्रिज्या \qty{10}{\micro m} हो, तो रस की
चाल और प्रति घंटा पहुँचाई शर्करा निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:plant-transport:9}
समझाइए कि \omterm{def:b1:flowering-plant-organization:organs}{तने} पर वलय काटने से कोई वृक्ष एक ही मौसम में क्यों मर जाता है,
फिर भी छल्ला काटे जाने के बाद हफ़्तों उसकी \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} हरी क्यों रहती हैं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:plant-transport:10}
किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} के वायु-अवकाश \qty{25}{\celsius} पर संतृप्त हैं
(\qty{23}{g/m^3} वाष्प); बाहर की हवा \qty{9}{g/m^3} रखती है।
$\mathrm{CO_2}$ बाहर \qty{0.72}{g/m^3} और भीतर \qty{0.45}{g/m^3} है।
दोनों गैसें उन्हीं छिद्रों से विसरित होती हैं, और जल $\mathrm{CO_2}$ से
1.6 गुना तेज़ विसरित होता है। खोए जल और कमाई $\mathrm{CO_2}$ का अनुपात
द्रव्यमान से और अणु से निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:plant-transport:11}
किसी गरम सूखी दोपहर में एक वाहिका में \omterm{def:b1:plant-transport:xylem}{गुहिकायन} हो जाता है। बताइए कि
जल-स्तंभ का, पड़ोसी वाहिकाओं के तनाव का, और जिस \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} को वह पालती थी
उसका क्या होता है; समझाइए कि गर्त हानि को कैसे सीमित करते हैं और पौधा
रात में या वसंत में कैसे उबरता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:plant-transport:12}
``वृक्ष मिट्टी और आकाश के बीच एक बत्ती है, जिसके साथ एक शर्करा-नली उलटी
दिशा में दौड़ती है।'' एक अनुच्छेद में विवेचना कीजिए: हर प्रणाली क्या
ढोती है, उसे क्या चलाता है, पौधा ऊर्जा कहाँ खर्च करता है, और कहाँ
बिलकुल नहीं करता।
\end{exercise}

\section{समस्या: सिकोया}

\begin{problem}[{सप्ताहांत समस्या --- गरमी के किसी दिन सौ मीटर का एक
वृक्ष: उसके शिखर पर तनाव, रस की चाल, वह जल जो वह चलाता है, वह शर्करा जो
वह नीचे भेजता है और वह कार्बन जो वह खरीदता है, और अंत में शिखर पर
जाइलम-तनाव}]
\label{pb:b1:plant-transport:1}
कोई सिकोया \qty{100}{m} ऊँचा है और उसके शिखर पर \qty{2000}{m^2} \omterm{def:b1:flowering-plant-organization:organs}{पत्ती}
है। गरमी के किसी दिन वह \qty{0.5}{m^2} अनुप्रस्थ काट की रसदारु से होकर
\qty{12}{h} में \qty{600}{L} जल वाष्पोत्सर्जित करता है, जिसका
\qty{20}{\%} चालक अवकाशिका है। जल: $\rho = \qty{1000}{kg/m^3}$, $g =
\qty{9.8}{m/s^2}$, \qty{18}{g/mol}। \omterm{def:b1:flowering-plant-organization:organs}{तने} भर घर्षण गुरुत्व जितना ही तनाव
माँगता है। मिट्टी का जल-विभव \qty{-0.1}{MPa}। शिखर \qty{12}{h} तक प्रति
वर्ग मीटर \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} प्रति सेकंड \qty{4}{\micro mol} $\mathrm{CO_2}$ बाँधता
है; सुक्रोस \qty{342}{g/mol} है और बारह कार्बन ढोता है; \omterm{def:b1:plant-transport:phloem}{फ्लोएम} का रस
द्रव्यमान से \qty{15}{\%} सुक्रोस है (घनत्व \qty{1.06}{g/mL}) और
\qty{0.6}{m/h} पर बहता है।

\medskip
\noindent\textbf{भाग I --- तनाव।}
\begin{enumerate}
  \item स्तंभ को गुरुत्व के विरुद्ध थामने के लिए चाहिए तनाव
        निकालिए।
  \item घर्षण जोड़िए: शिखर पर जाइलम-तनाव कितना है?
  \item रस के लिए $\Psi_s = 0$ लेते हुए शिखर पर \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} का जल-विभव
        निकालिए।
  \item शिखर पर \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में जल घुसने के लिए उनका जल-विभव
        कितना होना ही चाहिए? यदि उनका $\Psi_s = \qty{-2.5}{MPa}$ हो, तो
        उनकी \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} कितनी है?
  \item उसी वृक्ष पर \qty{10}{m} की किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} से तुलना कीजिए ($\Psi_s =
        \qty{-1.5}{MPa}$)।
  \item महीन नलियों में जल लगभग \qty{-30}{MPa} सह लेता है; तो वृक्ष
        सीधे \qty{1000}{m} तक क्यों नहीं बढ़ जाता?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- बहाव।}
\begin{enumerate}[resume]
  \item वाष्पोत्सर्जन की दर लीटर प्रति घंटा में और घन मीटर प्रति सेकंड
        में निकालिए।
  \item रसदारु का चालक अनुप्रस्थ काट निकालिए।
  \item रस की माध्य चाल निकालिए।
  \item \omterm{def:b1:flowering-plant-organization:organs}{तने} की चालक अवकाशिका में रखे जल का द्रव्यमान निकालिए (उसके
        \qty{100}{m} का)।
  \item उस चाल पर किसी जल-अणु को \omterm{def:b1:flowering-plant-organization:organs}{जड़} से शिखर तक जाने में कितना समय
        लगता है?
  \item प्रति वर्ग मीटर \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} प्रति घंटा वाष्पोत्सर्जन ग्राम में, और
        प्रति सेकंड मिलीमोल में निकालिए।
  \item रस मिट्टी के \qty{-0.1}{} से शिखर के प्रश्न 3 वाले मान तक
        \qty{100}{m} पर विभव-प्रवणता के नीचे बहता है। प्रवणता
        मेगापास्कल प्रति मीटर में कितनी है? अकेले गुरुत्वीय भाग से उसकी
        तुलना कैसी है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- शर्करा।}
\begin{enumerate}[resume]
  \item शिखर द्वारा प्रति दिन बाँधी $\mathrm{CO_2}$ मोल में
        निकालिए।
  \item उसके अनुरूप सुक्रोस मोल में और किलोग्राम में निकालिए।
  \item आधी शिखर में ही श्वसित हो जाती है; बाकी \omterm{def:b1:plant-transport:phloem}{फ्लोएम} से नीचे भेजी
        जाती है। प्रति दिन स्थानांतरित सुक्रोस का द्रव्यमान और उसे ढोते
        रस का द्रव्यमान निकालिए।
  \item रस का आयतन निकालिए और, \qty{0.6}{m/h} पर, उसे \qty{24}{h} में
        ढोने के लिए चाहिए \omterm{def:b1:flowering-plant-organization:tissues}{चालनी नलिकाओं} का अनुप्रस्थ काट भी।
  \item सुक्रोस को शिखर से \omterm{def:b1:flowering-plant-organization:organs}{जड़} तक जाने में लगने वाला समय निकालिए।
  \item जल-उपयोग अनुपात निकालिए: प्रति मोल बंधी $\mathrm{CO_2}$
        वाष्पोत्सर्जित जल के मोल।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- समझौते।}
\begin{enumerate}[resume]
  \item $\Psi_s = \qty{-1.2}{MPa}$ और \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} \qty{1.0}{MPa} के साथ
        शिखर पर \omterm{def:b1:plant-transport:phloem}{फ्लोएम} रस का जल-विभव: उसे निकालिए और वहाँ के \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} से
        तुलना कीजिए। दोनों के बीच जल किस ओर चलता है, और क्या वह लदान
        के अनुकूल है?
  \item \omterm{def:b1:flowering-plant-organization:organs}{जड़} पर \omterm{def:b1:plant-transport:phloem}{फ्लोएम} का रस \qty{5}{\%} सुक्रोस ($\Psi_s =
        \qty{-0.4}{MPa}$) है और \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} \qty{-0.3}{MPa} पर। कौन-सी \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति}
        \omterm{def:b1:plant-transport:phloem}{फ्लोएम} के $\Psi$ को \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} के बराबर कर देती है? शिखर और \omterm{def:b1:flowering-plant-organization:organs}{जड़} के
        बीच कौन-सा दाब-अंतर बहाव चलाता है?
  \item दोपहर में \omterm{def:b1:plant-transport:stoma}{रंध्र} बंद हो जाते हैं और वाष्पोत्सर्जन आधा रह जाता
        है। घर्षण-तनाव और शिखर का जाइलम-विभव फिर से निकालिए। वृक्ष क्या
        पाता और क्या खोता है?
  \item किसी सूखे साल में मिट्टी गिरकर \qty{-1.0}{MPa} पर आ जाती है।
        पूरे वाष्पोत्सर्जन के साथ शिखर का विभव फिर से निकालिए। क्या
        शिखर की \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} ($\Psi_s = \qty{-2.5}{MPa}$) कोई \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} रख
        सकती हैं? वृक्ष को क्या करना ही पड़ेगा?
  \item समझाइए कि सबसे ऊँची \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} सबसे छोटी और सबसे मोटी क्यों
        होती हैं।
  \item परिणाम बताइए: गरमी के किसी दिन \qty{100}{m} के किसी वृक्ष के
        शिखर पर जाइलम-तनाव, और उसके साथ चलने वाली रस की चाल।
\end{enumerate}
\end{problem}
