\chapter{La reproduction et l'environnement}\label{ch:g8:reproduction-and-environment}

Pourquoi le chœur des grenouilles commence-t-il en mai et non en
novembre~? Pourquoi une bonne année de hêtres emplit-elle les bois
de plantules, et un juin froid et pluvieux vide-t-il la mare des
grenouillets du printemps suivant~? Le chapitre précédent a bâti le
mécanisme de la génération suivante~; celui-ci le branche sur le
monde~: la reproduction est calée dans le temps, pourvue --- et
limitée --- par l'\omterm{def:g6:exploring-our-environment:components}{environnement}.

\section{Calée sur les saisons}

\begin{proposition}[La reproduction suit le calendrier de l'abondance]\label{prop:g8:reproduction-and-environment:timing}
Les \omterm{def:g5:biodiversity:species}{espèces} calent leur reproduction pour que les \emph{jeunes}
arrivent quand les \omterm{def:g6:exploring-our-environment:conditions}{conditions} et la nourriture les serviront
(\cref{prop:g6:life-through-seasons:conditions})~:
\begin{itemize}
  \item la mésange bleue pond de façon que ses poussins éclosent au
        pic printanier des \omterm{ex:g2:animals-grow-up:butterfly}{chenilles} (la chaîne du chêne de
        \cref{ex:g3:food-chains:three}, calée au jour près)~;
  \item la grenouille pond dans la mare qui se réchauffe, ce qui
        donne aux \omterm{ex:g2:animals-grow-up:frog}{têtards} un été de croissance avant l'hiver~;
  \item les \omterm{def:g1:plants-around-us:plant}{plantes} fleurissent à la saison de leurs pollinisateurs
        et mûrissent leurs \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} pour le calendrier de la
        \omterm{def:g6:how-plants-spread:dispersal}{dissémination} (\cref{pb:g6:life-through-seasons:1})~;
  \item le brame d'automne du cerf place les \omterm{def:g2:born-grow-die:cycle}{naissances} au
        \emph{printemps} suivant --- la gestation comptée d'avance.
\end{itemize}
Les signaux qui déclenchent la machinerie sont ceux de
l'\omterm{def:g6:exploring-our-environment:components}{environnement} lui-même~: l'allongement des jours avant tout, la
chaleur, la pluie.
\end{proposition}
\admitted

\begin{example}[Les agneaux d'intérieur]\label{ex:g8:reproduction-and-environment:lambs}
Les éleveurs de moutons exploitent le signal directement~: les
brebis entrent en condition de reproduction quand les jours
\emph{raccourcissent}~; sous une \omterm{def:g6:exploring-our-environment:conditions}{lumière} artificielle d'abord longue
puis raccourcie, un troupeau peut être amené à agneler pour le
marché précoce, plus rentable. La machinerie lit la \omterm{def:g6:exploring-our-environment:conditions}{lumière}, pas les
calendriers --- change la \omterm{def:g6:exploring-our-environment:conditions}{lumière} et le calendrier suit. Un signal
connu est un levier.
\end{example}

\section{Pourvue par le milieu}

\begin{proposition}[La reproduction se paie en matière]\label{prop:g8:reproduction-and-environment:cost}
Les \omterm{def:g2:animals-grow-up:born}{œufs}, le \omterm{def:g2:animals-grow-up:born}{lait}, le nectar, les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} et les bois du cerf sont
tous de la \emph{production} au sens de
\cref{def:g6:matter-from-food:production} --- de la \omterm{def:g6:plants-are-producers:matter}{matière
organique} que les parents doivent d'abord acquérir. La richesse du
milieu fixe donc le budget~:
\begin{itemize}
  \item une poule bien nourrie pond chaque jour~; une poule affamée
        s'arrête --- le registre de
        \cref{pb:g6:matter-from-food:1} qui s'équilibre sur la
        ligne des \omterm{def:g2:animals-grow-up:born}{œufs}~;
  \item le hêtre ne fructifie massivement que les «~années de
        glandée~» qui suivent de bonnes saisons de croissance ---
        une récolte de \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} est un été mis en banque
        (\cref{ex:g6:plants-are-producers:pantry})~;
  \item les années dures amincissent la ponte~: beaucoup d'\omterm{prop:g3:sorting-living-things:groups}{oiseaux}
        pondent moins d'\omterm{def:g2:animals-grow-up:born}{œufs}, ou pas du tout, quand les proies
        manquent.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Compter la ponte des mésanges]\label{ex:g8:reproduction-and-environment:clutch}
Les études en nichoirs montrent des pontes de mésange bleue qui
suivent la ressource en \omterm{ex:g2:animals-grow-up:butterfly}{chenilles}~: dans les riches chênaies, dix
\omterm{def:g2:animals-grow-up:born}{œufs} ou plus~; dans les jardins pauvres, six ou sept. L'oiselle ne
décide pas cela avec un registre --- c'est son état corporel, bâti
sur la nourriture du milieu, qui fixe ce qu'elle peut pourvoir. La
taille de la ponte, c'est la qualité du milieu, écrite en \omterm{def:g2:animals-grow-up:born}{œufs}.
\end{example}

\section{Limitée par le monde~: les populations}

\begin{definition}[Population]\label{def:g8:reproduction-and-environment:population}
Une \emph{population}\index{population} est l'ensemble des individus
d'une même \omterm{def:g5:biodiversity:species}{espèce} vivant en un même lieu --- les grenouilles de la
mare, les hêtres du bois. La reproduction y ajoute~; les morts et
les départs en retranchent~; l'\omterm{def:g6:exploring-our-environment:components}{environnement} fixe le plafond.
\end{definition}

\begin{proposition}[Pourquoi les populations n'explosent pas]\label{prop:g8:reproduction-and-environment:limits}
Toute \omterm{def:g5:biodiversity:species}{espèce} surproduit --- des milliers d'\omterm{def:g2:animals-grow-up:born}{œufs} par couple de
grenouilles, des milliers de \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} par hêtre
(\cref{prop:g4:animal-reproduction:strategies}) --- et pourtant les
mares ne sont pas compactes de grenouilles. Le surplus est dépensé
face aux freins de l'\omterm{def:g6:exploring-our-environment:components}{environnement}~:
\begin{itemize}
  \item la \emph{prédation} --- les hérons et les larves de
        libellule de \cref{ex:g4:animal-reproduction:pondcount}~;
  \item la \emph{nourriture et la place} --- il n'y a qu'un certain
        nombre de \omterm{ex:g2:animals-grow-up:butterfly}{chenilles}, de trous à nid, de taches de \omterm{def:g6:exploring-our-environment:conditions}{lumière}
        au \omterm{def:g6:decomposers-and-soil:soil}{sol} de la forêt~;
  \item les \emph{\omterm{def:g6:exploring-our-environment:conditions}{conditions}} --- une gelée tardive sur les \omterm{def:g3:flowers-fruits-seeds:parts}{fleurs}
        (\cref{exo:g3:flowers-fruits-seeds:9}), une sécheresse sur
        la mare qui rétrécit des \omterm{ex:g2:animals-grow-up:frog}{têtards}.
\end{itemize}
En moyenne, dans un milieu stable, chaque couple se remplace
lui-même --- l'équilibre de
\cref{prop:g5:food-webs:balance}, vu du côté des parents.
\end{proposition}
\admitted

\begin{example}[Deux printemps à la mare]\label{ex:g8:reproduction-and-environment:twosprings}
Printemps 1, doux et humide~: eau abondante, \omterm{prop:g3:sorting-living-things:groups}{insectes} précoces ---
des milliers d'\omterm{def:g2:animals-grow-up:born}{œufs}, des centaines de grenouillets, la \omterm{def:g8:reproduction-and-environment:population}{population}
grimpe. Printemps 2, une sécheresse~: la mare se réduit à une
flaque, les \omterm{ex:g2:animals-grow-up:frog}{têtards} s'entassent, le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} baisse
(\cref{prop:g7:oxygen-in-the-water:drains}), les hérons festoient
sur le banc concentré --- une poignée de grenouillets. Mêmes
grenouilles, même mécanisme, deux verdicts~: l'\omterm{def:g6:exploring-our-environment:components}{environnement}
contresigne chaque génération.
\end{example}

\begin{remark}[Les humains déplacent les freins]\label{rem:g8:reproduction-and-environment:humans}
La plupart des effets humains sur la faune sauvage passent par ces
freins, et non par le mécanisme central~: les mares asséchées et les
haies arrachées suppriment des lieux de reproduction
(\cref{prop:g2:where-animals-live:loss})~; les pesticides vident le
pic de \omterm{ex:g2:animals-grow-up:butterfly}{chenilles} sur lequel les mésanges s'étaient calées~; les
nichoirs de jardin et les prairies humides restaurées de
\cref{ex:g5:biodiversity:storks} redonnent du plafond. Protéger une
\omterm{def:g5:biodiversity:species}{espèce}, en pratique, c'est protéger l'\omterm{def:g6:exploring-our-environment:components}{environnement} sur lequel sa
reproduction est branchée --- signal, budget et freins ensemble.
\end{remark}

\begin{method}[Faire le bilan d'un changement de population]\label{met:g8:reproduction-and-environment:audit}
Quand une \omterm{def:g8:reproduction-and-environment:population}{population} monte ou baisse, fais le bilan dans l'ordre~:
\begin{enumerate}
  \item le \emph{calage}~: le signal et le pic de ressource
        coïncidaient-ils encore~? (un printemps trop précoce peut
        laisser les poussins après les \omterm{ex:g2:animals-grow-up:butterfly}{chenilles})~;
  \item le \emph{budget}~: qu'est-ce que le milieu de la saison a
        permis aux parents de pourvoir~?
  \item les \emph{freins}~: lequel a changé --- prédateurs,
        nourriture, place, \omterm{def:g6:exploring-our-environment:conditions}{conditions}~?
  \item le \emph{milieu}~: le terrain de reproduction lui-même
        a-t-il été gagné ou perdu~?
\end{enumerate}
\end{method}

\section{Problème~: le registre des nichoirs}

\begin{problem}[{Problème du week-end --- dix ans de mésanges
bleues, mises en bilan nichoir par nichoir}]\label{pb:g8:reproduction-and-environment:1}
Une école suit vingt nichoirs pendant dix ans~: dates de ponte,
tailles de ponte, jeunes à l'envol par nichoir --- à côté du relevé
météorologique et du carnet du jardinier.

\medskip\noindent\textbf{Partie I --- Lire les années
normales.}
\begin{enumerate}
  \item Années 1 à 4~: ponte fin avril, huit à dix \omterm{def:g2:animals-grow-up:born}{œufs}, la plupart
        des \omterm{def:g2:animals-grow-up:born}{œufs} menés à l'envol. Nomme le signal qui fixe la fin
        avril, et le pic que ce calage vise.
  \item Les pontes de l'année 3 comptaient un ou deux \omterm{def:g2:animals-grow-up:born}{œufs} de plus,
        après un mai chaud et riche en \omterm{ex:g2:animals-grow-up:butterfly}{chenilles} en année 2 --- les
        parents avaient bien passé l'hiver. Quelle proposition cette
        corrélation illustre-t-elle~?
  \item Même les bonnes années, environ un cinquième des \omterm{def:g2:animals-grow-up:born}{œufs}
        n'arrive pas à l'envol. Énumère les freins probablement
        responsables, d'après
        \cref{prop:g8:reproduction-and-environment:limits}.
  \item Pourquoi l'école consigne-t-elle la météo et les travaux du
        parc à côté des \omterm{def:g2:animals-grow-up:born}{œufs}~? (Réponds avec les quatre questions
        du bilan.)
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les anomalies.}
\begin{enumerate}[resume]
  \item Année 5~: un mars anormalement chaud a fait sortir les
        \omterm{ex:g2:animals-grow-up:butterfly}{chenilles} deux semaines plus tôt~; les dates de ponte n'ont
        presque pas bougé. Prédis les envols de l'année 5, et nomme
        la ligne du bilan qui a échoué.
  \item Année 6~: le carnet du jardinier note que les \omterm{def:g1:plants-around-us:tree}{branches}
        mortes des vieux chênes ont été enlevées et la bande de
        prairie tondue tous les mois. Les envols chutent alors que
        la météo a été clémente. Quelles lignes du bilan le carnet
        met-il en cause~?
  \item Année 7~: un couple d'éperviers s'installe. Les pontes sont
        inchangées, mais le nombre de jeunes à l'envol baisse. Quel
        frein a bougé, et pourquoi la ligne de la \emph{ponte} n'a-t-elle
        pas bougé avec lui~?
  \item Année 8~: les nichoirs passent à quarante~; le total des
        envols monte, mais la réussite \emph{par nichoir} reste
        stable. Quel plafond les nouveaux nichoirs ont-ils relevé,
        et lequel ont-ils laissé intact~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le rapport.}
\begin{enumerate}[resume]
  \item Rédige la phrase de synthèse des dix ans reliant la taille
        de la ponte à la qualité du milieu, en citant
        \cref{ex:g8:reproduction-and-environment:clutch}.
  \item La commune propose d'abattre la rangée de chênes pour un
        parking, en «~compensant~» par cinquante nichoirs de plus.
        Fais le bilan de l'offre~: parmi le signal, le budget, les
        freins et le milieu, que remplacent les nichoirs, et
        qu'est-ce que seuls les chênes fournissent~?
  \item Recommande deux mesures d'aménagement pour l'année 11,
        chacune reliée à sa ligne du bilan.
  \item Termine par la morale du chapitre en une phrase~: de quoi
        chaque jeune à l'envol a-t-il besoin, en plus de deux
        parents~?
\end{enumerate}
\end{problem}
