\chapter{Organização funcional da célula eucariótica}\label{ch:b1:eukaryotic-cell}

Uma \omterm{def:b1:cell-unit-of-life:cell}{célula} do pâncreas fabrica, todo dia, cerca da própria massa em
enzimas digestivas e as despeja num ducto. As enzimas são proteínas que
digeririam a \omterm{def:b1:cell-unit-of-life:cell}{célula} que as fez; por isso a \omterm{def:b1:cell-unit-of-life:cell}{célula} as constrói dentro de
uma lâmina fechada de membrana, as envia a um segundo compartimento onde
são triadas e concentradas, as armazena em grânulos, e só as libera ao
sinal de uma refeição. Cada etapa acontece num espaço diferente
delimitado por membrana, e o tráfego entre os espaços é feito por
vesículas. Este capítulo descreve os compartimentos da \omterm{def:b1:cell-unit-of-life:prokeuk}{célula
eucariótica}, o \omterm{def:b1:eukaryotic-cell:endomembrane}{sistema de endomembranas} pelo qual as proteínas fluem, as
\omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} que transformam energia, o esqueleto que mantém a forma da
\omterm{def:b1:cell-unit-of-life:cell}{célula} e move as partes dela, e as características que distinguem uma
\omterm{def:b1:cell-unit-of-life:cell}{célula} vegetal de uma animal.

\section{Compartimentos}

\begin{definition}[Organela, citosol]\label{def:b1:eukaryotic-cell:organelle}
Uma \emph{organela}\index{organela} é uma estrutura dentro de uma \omterm{def:b1:cell-unit-of-life:cell}{célula}
com composição e função definidas; as \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} delimitadas por membrana
são compartimentos cujo interior difere em composição do
\emph{citosol}\index{citosol}, a solução aquosa do \omterm{def:b1:cell-unit-of-life:cell}{citoplasma} fora
delas. Compartimentos delimitados por membrana: o \omterm{def:b1:eukaryotic-cell:nucleus}{núcleo}, o \omterm{def:b1:eukaryotic-cell:endomembrane}{retículo
endoplasmático}, o \omterm{def:b1:eukaryotic-cell:endomembrane}{complexo de Golgi}, os \omterm{def:b1:eukaryotic-cell:endomembrane}{lisossomos}, os \omterm{def:b1:eukaryotic-cell:peroxisome}{peroxissomos}, as
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitocôndrias} e, nas plantas, os \omterm{def:b1:eukaryotic-cell:plastid}{plastídios} e o \omterm{def:b1:eukaryotic-cell:plantcell}{vacúolo}. \omterm{def:b1:cell-unit-of-life:prokeuk}{Organelas} sem
membrana: os ribossomos, o \omterm{def:b1:eukaryotic-cell:cytoskeleton}{centrossomo} e o \omterm{def:b1:eukaryotic-cell:cytoskeleton}{citoesqueleto}.
\end{definition}

\begin{definition}[O núcleo]\label{def:b1:eukaryotic-cell:nucleus}
O \emph{núcleo}\index{núcleo} guarda os cromossomos na forma de
\emph{cromatina}\index{cromatina}, DNA enrolado em proteínas histonas
(\cref{ch:b1:genomes}). Ele é delimitado pelo
\emph{envoltório nuclear}\index{envoltório nuclear}, duas membranas, a
externa contínua com o \omterm{def:b1:eukaryotic-cell:endomembrane}{retículo endoplasmático}, perfurado por
\emph{poros nucleares}\index{poro nuclear} pelos quais RNA e proteínas
passam nos dois sentidos sob controle. O \emph{nucléolo}\index{nucléolo}
é a região em que o RNA ribossômico é fabricado e os ribossomos são
montados. O núcleo separa a transcrição (dentro) da tradução (fora),
uma separação que os procariotos não têm.
\end{definition}

\begin{proposition}[Os compartimentos de uma célula do fígado]\label{prop:b1:eukaryotic-cell:budget}
Num hepatócito de \qty{5000}{\micro m^3}, o \omterm{def:b1:eukaryotic-cell:organelle}{citosol} ocupa cerca de
metade do volume, as \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitocôndrias} um quinto, o \omterm{def:b1:eukaryotic-cell:endomembrane}{retículo endoplasmático}
um sexto, o \omterm{def:b1:eukaryotic-cell:nucleus}{núcleo} um dezesseis avos, e o \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}, os \omterm{def:b1:eukaryotic-cell:endomembrane}{lisossomos} e os
\omterm{def:b1:eukaryotic-cell:peroxisome}{peroxissomos} alguns por cento juntos. As membranas contam outra
história: dos \qty{110000}{\micro m^2} de membrana da \omterm{def:b1:cell-unit-of-life:cell}{célula}, metade é
\omterm{def:b1:eukaryotic-cell:endomembrane}{retículo endoplasmático}, um terço é a membrana interna das \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitocôndrias},
e apenas \qty{2}{\%} é a membrana plasmática. Quase toda a membrana de
uma \omterm{def:b1:cell-unit-of-life:cell}{célula} está dentro dela.
\end{proposition}

\section{O sistema de endomembranas}

\begin{definition}[Retículo endoplasmático, complexo de Golgi, lisossomo]\label{def:b1:eukaryotic-cell:endomembrane}
O \emph{retículo endoplasmático}\index{retículo endoplasmático} (RE) é
uma rede de lâminas e túbulos de membrana que encerra um único espaço
contínuo, o lúmen, e é contínua com o \omterm{def:b1:eukaryotic-cell:nucleus}{envoltório nuclear}. A parte
\emph{rugosa}\index{retículo endoplasmático rugoso} dele é cravejada de
ribossomos que injetam no lúmen ou na membrana as proteínas que
fabricam; a parte \emph{lisa}\index{retículo endoplasmático liso}
fabrica lipídios, armazena cálcio e faz desintoxicação. O
\emph{complexo de Golgi}\index{complexo de Golgi} é uma pilha de sacos
achatados (as cisternas), com uma face \emph{cis} de recepção junto ao
RE e uma face \emph{trans} de expedição, em que as proteínas vindas do
RE são modificadas (açúcares aparados e acrescentados), triadas e
empacotadas em vesículas rumo aos destinos delas. Os
\emph{lisossomos}\index{lisossomo} são vesículas ácidas (pH
$\approx 5$) cheias de enzimas hidrolíticas que digerem material trazido
de fora ou \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} gastas. Junto com as vesículas que se deslocam
entre eles e a membrana plasmática, essas estruturas formam o
\emph{sistema de endomembranas}\index{sistema de endomembranas}: um
conjunto conectado de compartimentos cujos lumens são, do ponto de vista
topológico, o exterior da \omterm{def:b1:cell-unit-of-life:cell}{célula}.
\end{definition}

\begin{proposition}[A via secretora]\label{prop:b1:eukaryotic-cell:secretory}
\index{via secretora}
Uma proteína destinada à secreção, à membrana plasmática ou a um
\omterm{def:b1:eukaryotic-cell:endomembrane}{lisossomo} é sintetizada por ribossomos ligados ao RE rugoso, entra no
lúmen do RE à medida que é feita, dobra-se ali, viaja em vesículas até a
face cis do \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}, atravessa a pilha enquanto é modificada, e deixa a
face trans numa vesícula endereçada ao destino dela: um grânulo de
secreção que se funde com a membrana plasmática mediante um sinal
(\emph{exocitose regulada}\index{exocitose}), uma vesícula que se funde
continuamente (exocitose \emph{constitutiva}, que também entrega
membrana nova) ou um \omterm{def:b1:eukaryotic-cell:endomembrane}{lisossomo}. As proteínas citosólicas e nucleares são
feitas em ribossomos livres e nunca entram no sistema. O que decide é
uma \emph{sequência-sinal}\index{sequência-sinal} no início da proteína
(\cref{ch:b1:gene-expression}).
\end{proposition}
\begin{proof}[Evidências]
Palade (anos 1960) forneceu a fatias de pâncreas de cobaia aminoácidos
radioativos por três minutos (o \emph{pulso}), depois aminoácidos não
marcados (a \emph{caça}), e localizou a radioatividade por
autorradiografia de micrografias eletrônicas em intervalos sucessivos.
Aos \qty{3}{min} os grãos de prata estavam sobre o RE rugoso; aos
\qty{20}{min} sobre o \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}; aos \qty{40}{min} sobre \omterm{def:b1:eukaryotic-cell:plantcell}{vacúolos} de
condensação e grânulos novos; às \qty{2}{h} no lúmen do ducto. O
\omterm{def:b1:cell-unit-of-life:fractionation}{fracionamento celular} deu quimicamente a mesma sequência. Blobel (anos
1970) mostrou que as proteínas de secreção feitas em tubo de ensaio sem
membranas são vinte resíduos mais longas que a forma secretada e não são
protegidas de uma protease acrescentada, ao passo que, feitas na
presença de membranas de RE, elas são clivadas e protegidas: os
resíduos extras são o sinal que dirige a cadeia em crescimento para
dentro do RE.
\end{proof}

\begin{method}[Acompanhar uma molécula pela célula]\label{met:b1:eukaryotic-cell:pulsechase}
\begin{enumerate}
  \item Marque uma coorte curta: exponha as \omterm{def:b1:cell-unit-of-life:cell}{células} a um precursor
        radioativo ou fluorescente por alguns minutos (o pulso), e
        substitua-o depois por um excesso de precursor não marcado (a
        caça), de modo que só as moléculas feitas durante o pulso levem
        a marcação.
  \item Colha amostras em intervalos, fixe e localize a marcação: por
        autorradiografia de cortes, por microscopia de fluorescência ou
        por fracionamento e contagem.
  \item O compartimento cuja marcação atinge o máximo primeiro está a
        montante; a sequência dos máximos é a rota; os intervalos são os
        tempos de trânsito.
  \item Bloqueie uma etapa (frio, um inibidor de energia, um mutante) e
        veja onde a marcação se acumula: no compartimento anterior ao
        bloqueio.
\end{enumerate}
\end{method}

\begin{definition}[Endocitose]\label{def:b1:eukaryotic-cell:endocytosis}
A \emph{endocitose}\index{endocitose} traz material para dentro: a
membrana plasmática se invagina em torno dele e se estrangula numa
vesícula, que em geral se funde com um \omterm{def:b1:eukaryotic-cell:endomembrane}{lisossomo}. A \emph{fagocitose}
engloba partículas (uma bactéria, uma \omterm{def:b1:cell-unit-of-life:cell}{célula} morta); a
\emph{pinocitose} capta gotículas de líquido; a
\emph{endocitose mediada por receptor} concentra uma molécula
específica (partículas que carregam colesterol, a transferrina que
carrega ferro) em receptores de fossetas revestidas antes de captá-la. A
exocitose acrescenta membrana à superfície da \omterm{def:b1:cell-unit-of-life:cell}{célula} e a endocitose a
retira; numa \omterm{def:b1:cell-unit-of-life:cell}{célula} em secreção as duas se equilibram, de modo que a
superfície permanece constante enquanto áreas inteiras de membrana
passam por ela a cada hora.
\end{definition}

\section{Mitocôndrias e plastídios}

\begin{definition}[Mitocôndria]\label{def:b1:eukaryotic-cell:mitochondrion}
Uma \emph{mitocôndria}\index{mitocôndria} é uma \omterm{def:b1:eukaryotic-cell:organelle}{organela} de
\qtyrange{0.5}{1}{\micro m} de largura e alguns micrômetros de
comprimento, delimitada por duas membranas: uma membrana externa \omterm{def:b1:eukaryotic-cell:endomembrane}{lisa},
permeável a moléculas pequenas, e uma membrana interna dobrada em
\emph{cristas}\index{crista}, impermeável a íons, que carrega a cadeia
respiratória e a ATP-sintase
(\cref{ch:b1:respiration-fermentation}). Dentro está a
\emph{matriz}\index{matriz mitocondrial}, que abriga as enzimas do ciclo
de Krebs, ribossomos do tipo bacteriano e várias cópias de um pequeno
DNA circular. As mitocôndrias se dividem por fissão, deslocam-se ao
longo do \omterm{def:b1:eukaryotic-cell:cytoskeleton}{citoesqueleto}, e são de algumas centenas a alguns milhares por
\omterm{def:b1:cell-unit-of-life:cell}{célula}.
\end{definition}

\begin{definition}[Plastídios]\label{def:b1:eukaryotic-cell:plastid}
Os \emph{plastídios}\index{plastídio} são as \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} de membrana dupla
das \omterm{def:b1:cell-unit-of-life:cell}{células} vegetais: os \emph{cloroplastos}\index{cloroplasto}, em que
um terceiro sistema de membranas, os \emph{tilacoides}\index{tilacoide},
empilhados em grana e banhados no \emph{estroma}\index{estroma},
realizam a fotossíntese (\cref{ch:b1:photosynthesis}); os amiloplastos,
que armazenam amido; os cromoplastos, que guardam os pigmentos dos
frutos e das pétalas. Todos derivam de proplastídios e, como as
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitocôndrias}, contêm DNA circular próprio e ribossomos do tipo
bacteriano, e se dividem por fissão.
\end{definition}

\begin{proposition}[Origem endossimbiótica]\label{prop:b1:eukaryotic-cell:endosymbiosis}
\index{endossimbiose}
As \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitocôndrias} e os \omterm{def:b1:eukaryotic-cell:plastid}{plastídios} descendem de bactérias de vida livre
englobadas por uma \omterm{def:b1:cell-unit-of-life:prokeuk}{célula eucariótica} ancestral e mantidas como
simbiontes: as \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitocôndrias}, de uma bactéria aeróbia; os \omterm{def:b1:eukaryotic-cell:plastid}{plastídios}, de
uma cianobactéria.
\end{proposition}
\begin{proof}[Evidências]
As duas \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} têm duas membranas, a interna de composição bacteriana
e a externa parecida com a do hospedeiro; um cromossomo circular sem
histonas; ribossomos do tamanho e da sensibilidade a antibióticos dos
bacterianos; um código genético com variantes bacterianas; divisão por
fissão binária independente da \omterm{def:b1:cell-unit-of-life:cell}{célula}; e os genes delas, quando
sequenciados, agrupam-se com os de linhagens bacterianas específicas
(alfaproteobactérias, cianobactérias), e não com os genes nucleares da
\omterm{def:b1:cell-unit-of-life:cell}{célula} que as abriga. A maior parte dos genes originais delas migrou
desde então para o \omterm{def:b1:eukaryotic-cell:nucleus}{núcleo}, e as \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} já não conseguem viver
sozinhas.
\end{proof}

\begin{definition}[Peroxissomo]\label{def:b1:eukaryotic-cell:peroxisome}
Um \emph{peroxissomo}\index{peroxissomo} é uma \omterm{def:b1:eukaryotic-cell:organelle}{organela} pequena, de
membrana única, que contém enzimas oxidativas que transferem hidrogênio
de substratos (ácidos graxos, álcool) para o oxigênio, produzindo
peróxido de hidrogênio, e catalase, que destrói o peróxido no mesmo
lugar. Ele confina uma química perigosa.
\end{definition}

\section{O citoesqueleto}

\begin{definition}[Citoesqueleto]\label{def:b1:eukaryotic-cell:cytoskeleton}
O \emph{citoesqueleto}\index{citoesqueleto} é uma rede de filamentos
proteicos que dá à \omterm{def:b1:cell-unit-of-life:cell}{célula} a forma dela, ancora as \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} dela, e move
tanto elas quanto a própria \omterm{def:b1:cell-unit-of-life:cell}{célula}. São três tipos: os
\emph{microfilamentos}\index{microfilamento} de actina (\qty{7}{nm},
duas fitas torcidas de subunidades globulares), concentrados sob a
membrana plasmática, responsáveis pela forma da \omterm{def:b1:cell-unit-of-life:cell}{célula}, pelo
rastejamento, pelo anel contrátil da divisão e pela contração muscular
junto com a miosina; os \emph{microtúbulos}\index{microtúbulo} (tubos
ocos de \qty{25}{nm} de tubulina), que irradiam do
\emph{centrossomo}\index{centrossomo} e são os trilhos ao longo dos
quais as proteínas motoras (a cinesina para fora, a dineína para dentro)
carregam vesículas e \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas}, além de formarem o fuso da divisão e o
eixo dos cílios e dos flagelos; e os
\emph{filamentos intermediários}\index{filamento intermediário}
(\qty{10}{nm}, semelhantes a cordas, de queratinas e proteínas
aparentadas), as fibras mecanicamente resistentes que enrijecem as
\omterm{def:b1:cell-unit-of-life:cell}{células} e se ancoram nos desmossomos. Microfilamentos e microtúbulos se
montam e se desmontam continuamente a partir dos estoques de subunidades
deles.
\end{definition}

\begin{example}[Mover uma vesícula]\label{ex:b1:eukaryotic-cell:transport}
Uma vesícula de secreção que deixa o \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} é recolhida pela cinesina, um
motor que caminha ao longo de um \omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtúbulo} em direção à periferia da
\omterm{def:b1:cell-unit-of-life:cell}{célula} a cerca de \qty{1}{\micro m/s}, gastando um ATP por passo de
\qty{8}{nm}; num \omterm{def:b1:body-plans-tissues:nervous}{neurônio} o mesmo motor carrega vesículas ao longo de um
axônio de um metro --- uma viagem de dez dias. Perto da superfície, a
vesícula passa ao córtex de actina e à miosina para o último micrômetro.
A difusão levaria uma vesícula de \qty{100}{nm} cerca de um minuto para
atravessar uma \omterm{def:b1:cell-unit-of-life:cell}{célula} de \qty{20}{\micro m}, e anos para percorrer um
metro: as \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} são carregadas, e não deixadas à deriva.
\end{example}

\section{Células vegetais e animais}

\begin{definition}[Parede celular, vacúolo, plasmodesmos]\label{def:b1:eukaryotic-cell:plantcell}
Uma \omterm{def:b1:cell-unit-of-life:cell}{célula} vegetal acrescenta três estruturas ao plano comum. A
\emph{parede celular}\index{parede celular}, por fora da membrana
plasmática, é um compósito de microfibrilas de celulose numa matriz de
outros polissacarídeos (\cref{ch:b1:carbohydrates}); ela fixa a forma da
\omterm{def:b1:cell-unit-of-life:cell}{célula}, resiste à pressão interna e cimenta as \omterm{def:b1:cell-unit-of-life:cell}{células} vizinhas por meio
de uma lamela média compartilhada. O \emph{vacúolo}\index{vacúolo}, um
compartimento de membrana única que ocupa até \qty{90}{\%} do volume da
\omterm{def:b1:cell-unit-of-life:cell}{célula}, armazena água, íons, açúcares, pigmentos e resíduos; a pressão
osmótica dele empurra a membrana contra a parede, e é esse
\emph{turgor}\index{turgor} que mantém uma \omterm{def:b1:flowering-plant-organization:organs}{folha} erguida. Os
\emph{plasmodesmos}\index{plasmodesmo} são canais que atravessam as
paredes, revestidos por membrana plasmática, pelos quais os citosóis de
\omterm{def:b1:cell-unit-of-life:cell}{células} vizinhas ficam contínuos, de modo que um \omterm{def:b1:body-plans-tissues:tissue}{tecido} vegetal é em
parte um único \omterm{def:b1:cell-unit-of-life:cell}{citoplasma} conectado (o simplasto).
\end{definition}

\begin{proposition}[Células vegetal e animal comparadas]\label{prop:b1:eukaryotic-cell:compare}
As duas têm \omterm{def:b1:eukaryotic-cell:nucleus}{núcleo}, RE, \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}, \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitocôndrias}, \omterm{def:b1:eukaryotic-cell:peroxisome}{peroxissomos}, ribossomos e
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{citoesqueleto}. A \omterm{def:b1:cell-unit-of-life:cell}{célula} vegetal tem parede de celulose, um \omterm{def:b1:eukaryotic-cell:plantcell}{vacúolo}
grande, \omterm{def:b1:eukaryotic-cell:plastid}{plastídios} e \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesmos}, e não tem centríolos nem \omterm{def:b1:eukaryotic-cell:endomembrane}{lisossomos}
(o \omterm{def:b1:eukaryotic-cell:plantcell}{vacúolo} digere); ela não rasteja nem engloba, e se divide construindo
uma parede nova no meio. A \omterm{def:b1:cell-unit-of-life:cell}{célula} animal não tem parede (e por isso pode
mudar de forma, rastejar e fagocitar), tem \omterm{def:b1:eukaryotic-cell:endomembrane}{lisossomos} e um \omterm{def:b1:eukaryotic-cell:cytoskeleton}{centrossomo}
com centríolos, e se comunica por junções em vez de \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesmos}; ela se
divide estrangulando-se em duas.
\end{proposition}

\begin{example}[Turgor]\label{ex:b1:eukaryotic-cell:turgor}
Uma \omterm{def:b1:cell-unit-of-life:cell}{célula} de \omterm{def:b1:flowering-plant-organization:organs}{folha} cujo \omterm{def:b1:eukaryotic-cell:plantcell}{vacúolo} contém \qty{0.3}{mol/L} de solutos puxa
água para dentro até que a parede reaja com uma pressão de cerca de
\qty{0.7}{MPa}, sete atmosferas; a \omterm{def:b1:cell-unit-of-life:cell}{célula} fica então túrgida e a \omterm{def:b1:flowering-plant-organization:organs}{folha},
firme. Privada de água, o \omterm{def:b1:eukaryotic-cell:plantcell}{vacúolo} perde volume, a pressão cai a zero e a
\omterm{def:b1:flowering-plant-organization:organs}{folha} murcha --- a parede inalterada, a pressão desaparecida
(\cref{ch:b1:membranes-transport}). Uma \omterm{def:b1:cell-unit-of-life:cell}{célula} animal na mesma solução,
por não ter parede, simplesmente estouraria.
\end{example}

\section{Problema: A célula acinar do pâncreas}

\begin{problem}[{Problema de fim de semana --- uma célula que secreta a
própria massa em proteína todo dia: o relógio de Palade, a membrana que
ela precisa reciclar, o sinal que endereça a proteína, terminando no
tempo de trânsito de uma enzima digestiva}]
\label{pb:b1:eukaryotic-cell:1}
Uma \omterm{def:b1:cell-unit-of-life:cell}{célula} acinar pancreática é um cubo de \qty{12}{\micro m} de aresta,
com a face apical (\qty{144}{\micro m^2}, voltada para o ducto) e a face
basal voltada para o sangue. Um pâncreas humano de \qty{80}{g} contém
cerca de $\num{8e10}$ \omterm{def:b1:cell-unit-of-life:cell}{células} desse tipo e secreta \qty{10}{g} de
proteína enzimática por dia. Um grânulo de secreção é uma esfera de
\qty{1}{\micro m} de diâmetro que contém proteína a \qty{200}{mg/mL}. Os
ribossomos acrescentam \qty{5}{} aminoácidos por segundo; uma enzima
típica tem \qty{250}{} resíduos de massa média \qty{110}{g/mol}.

\medskip
\noindent\textbf{Parte I --- O relógio de Palade.}
A figura do pulso e caça do capítulo dá a localização da marcação ao
longo do tempo.
\begin{enumerate}
  \item Por que o pulso precisa ser curto e a caça precisa conter um
        grande excesso de aminoácido não marcado?
  \item Leia na figura o instante em que a marcação atinge o máximo no
        RE, no \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} e nos grânulos.
  \item Estime o tempo de trânsito do RE ao \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} e do \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} ao grânulo.
  \item Onde está a marcação aos \qty{60}{min}, e por que parte dela já
        deixou a \omterm{def:b1:cell-unit-of-life:cell}{célula} enquanto a maior parte ainda está nos grânulos?
  \item Um segundo experimento é feito a \qty{15}{\celsius}: a marcação
        fica no RE por mais de uma hora. Interprete.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A massa de proteína.}
\begin{enumerate}[resume]
  \item Calcule a proteína secretada por \omterm{def:b1:cell-unit-of-life:cell}{célula} e por dia, em
        picogramas.
  \item Calcule o volume de um grânulo e a massa de proteína que ele
        contém.
  \item Calcule o número de grânulos liberados por \omterm{def:b1:cell-unit-of-life:cell}{célula} e por dia, e
        por minuto.
  \item Calcule o volume da \omterm{def:b1:cell-unit-of-life:cell}{célula} e, com uma densidade de
        \qty{1.05}{g/cm^3}, a massa dela. Que fração da própria massa
        ela secreta a cada dia?
  \item Calcule o número de moléculas de enzima em um grânulo (número de
        Avogadro $\num{6.0e23}$).
  \item Calcule o tempo de que um ribossomo precisa para fabricar uma
        molécula de enzima.
  \item Quantos ribossomos precisam trabalhar continuamente para
        produzir o total diário? Compare com os $\num{4e6}$ ribossomos
        que uma \omterm{def:b1:cell-unit-of-life:cell}{célula} dessas contém.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Membrana.}
\begin{enumerate}[resume]
  \item Calcule a área de membrana de um grânulo.
  \item Calcule a membrana entregue por dia à face apical pela exocitose
        dos grânulos da questão 8.
  \item Quantas vezes por dia a área da face apical é acrescentada? O
        que a \omterm{def:b1:cell-unit-of-life:cell}{célula} precisa fazer com ela, e por qual processo?
  \item A membrana dos grânulos vem do \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}, que a recebe do RE. Se o
        RE da \omterm{def:b1:cell-unit-of-life:cell}{célula} tem \qty{20000}{\micro m^2} de membrana, a quantos
        dias de membrana de grânulo isso corresponde? O que isso diz
        sobre a reciclagem dentro do \omterm{def:b1:eukaryotic-cell:endomembrane}{sistema de endomembranas}?
  \item A face apical carrega \qty{2000}{} microvilosidades de
        \qty{0.1}{\micro m} de diâmetro e \qty{1}{\micro m} de
        comprimento. Por qual fator elas ampliam a face de
        \qty{144}{\micro m^2}?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O endereço na proteína.}
\begin{enumerate}[resume]
  \item O mRNA de uma enzima traduzido em tubo de ensaio sem membranas
        dá uma cadeia \qty{20}{} resíduos mais longa que a enzima
        encontrada nos grânulos. Onde estão os resíduos extras e o que
        eles são?
  \item Na presença de vesículas de RE, o produto tem o comprimento
        certo, está dentro das vesículas e é protegido de uma protease.
        Interprete.
  \item O mesmo experimento com o mRNA de uma enzima citosólica dá um
        produto fora das vesículas e desprotegido. Conclua.
  \item Uma mutação apaga os vinte resíduos. Preveja onde a enzima vai
        parar e o que acontece com a \omterm{def:b1:cell-unit-of-life:cell}{célula}.
  \item Outra enzima é encontrada nos \omterm{def:b1:eukaryotic-cell:endomembrane}{lisossomos}, e não nos grânulos.
        Ela carrega o mesmo tipo de \omterm{prop:b1:eukaryotic-cell:secretory}{sequência-sinal}. Em que ponto da via
        a segunda decisão precisa ser tomada, e por qual mecanismo, em
        linhas gerais?
  \item Uma \omterm{def:b1:cell-unit-of-life:cell}{célula} é tratada com um fármaco que bloqueia a síntese de
        ATP. Preveja onde a marcação de um pulso e caça se acumula e por
        quê (duas razões).
  \item Combine as questões 3 e 8: em regime estacionário, quantos
        grânulos estão ``em trânsito'' entre o RE e o grânulo a cada
        instante?
  \item Resuma o resultado: o tempo de trânsito da síntese ao grânulo, o
        tempo até a secreção, os grânulos liberados por minuto, e a
        fração da própria massa que a \omterm{def:b1:cell-unit-of-life:cell}{célula} secreta por dia.
\end{enumerate}
\end{problem}
