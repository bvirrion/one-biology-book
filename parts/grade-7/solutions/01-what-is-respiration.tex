\section*{Chapter \ref{ch:g7:what-is-respiration} --- What Is Respiration?}

\begin{solution}{exo:g7:what-is-respiration:1}
\omterm{def:g7:what-is-respiration:respiration}{Respiration} is the gas exchange by which a \omterm{def:g6:exploring-our-environment:components}{living thing} takes
\omterm{def:g7:what-is-respiration:respiration}{oxygen} from its surroundings and releases \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide}.
\omterm{def:g4:breathing:movements}{Breathing} is only the chest's visible pumping movements ---
one way of serving the exchange; the exchange itself runs in
every living \omterm{def:g5:first-look-at-cells:cell}{cell}.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:2}
An \omterm{def:g7:what-is-respiration:respiration}{oxygen} probe in a closed container --- shows the \omterm{def:g7:what-is-respiration:respiration}{oxygen}
level falling; and \omterm{prop:g7:what-is-respiration:shown}{limewater} --- a clear liquid that turns
milky when \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide} passes through it.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:3}
To prove the change comes from the living subject and not
from the container, the day's warmth or the \omterm{prop:g7:what-is-respiration:shown}{limewater}
itself: the fair test's twin, with life the one difference.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:4}
\omterm{prop:g3:balanced-diet:jobs}{Energy}: in every \omterm{def:g5:first-look-at-cells:cell}{cell}, \omterm{def:g4:journey-of-food:digestion}{nutrients} are slowly burned with the
\omterm{def:g7:what-is-respiration:respiration}{oxygen}, releasing their \omterm{prop:g3:balanced-diet:jobs}{energy} for the \omterm{def:g5:first-look-at-cells:cell}{cell}'s work. \omterm{def:g7:what-is-respiration:respiration}{Carbon
dioxide} is the burnt-out remainder, carted away.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:5}
A mouse, a woodlouse, germinating peas, a mushroom, pond
snails --- and a spoonful of living \omterm{def:g6:decomposers-and-soil:soil}{soil}.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:6}
Test it in the dark: with the light-driven manufacture
stopped, the jar's \omterm{def:g7:what-is-respiration:respiration}{oxygen} falls and \omterm{prop:g7:what-is-respiration:shown}{limewater} clouds ---
\omterm{def:g7:what-is-respiration:respiration}{respiration} unmasked.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:7}
Likenesses: both consume \omterm{def:g7:what-is-respiration:respiration}{oxygen} and release \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide};
both release the fuel's \omterm{prop:g3:balanced-diet:jobs}{energy}. Differences: the \omterm{def:g5:first-look-at-cells:cell}{cell}'s
burning is slow and flameless, in thousands of small steps;
and its \omterm{prop:g3:balanced-diet:jobs}{energy} is captured for work instead of lost as
\omterm{def:g6:exploring-our-environment:conditions}{light} and heat all at once.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:8}
Sprinting child (hard muscular work) $>$ reading child (the
body's quiet standing charge) $>$ germinating pea (building
fast, but small) $>$ dry pea (dormant --- \omterm{def:g7:what-is-respiration:respiration}{respiration}
idling near zero). Rate follows work.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:9}
All night the sprouting peas' \omterm{def:g7:what-is-respiration:respiration}{respiration} consumed the
jar's \omterm{def:g7:what-is-respiration:respiration}{oxygen} and released \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide}; by morning too
little \omterm{def:g7:what-is-respiration:respiration}{oxygen} remains to keep a flame alight. The \omterm{prop:g7:what-is-respiration:shown}{limewater}
would have clouded steadily through the night.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:10}
Two containers with probes and \omterm{prop:g7:what-is-respiration:shown}{limewater} traps: one with
fresh living \omterm{def:g6:decomposers-and-soil:soil}{soil}, one with the same \omterm{def:g6:decomposers-and-soil:soil}{soil} baked lifeless
--- the one difference. If the living jar's \omterm{def:g7:what-is-respiration:respiration}{oxygen} falls
and its \omterm{prop:g7:what-is-respiration:shown}{limewater} clouds while the baked twin stays
unchanged, the \omterm{def:g6:decomposers-and-soil:soil}{soil}'s life did it.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:11}
Damp grain wakes: \omterm{def:g2:from-seed-to-plant:germination}{germination} begins, and \omterm{def:g7:what-is-respiration:respiration}{respiration} rises
from a whisper toward the sprouting pea's pace --- billions
of grains burning their stores at once, heating the heap
like the compost's working crews. Warmth speeds the work
further: a damp heap can heat itself toward spoilage. Dry
grain \omterm{prop:g1:healthy-habits:needs}{sleeps} cold; damp grain is a slow furnace.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:12}
Behavior: \omterm{def:g7:what-is-respiration:respiration}{breathing} can be watched --- the chest's
movements, fifteen times a minute
(\cref{ex:g4:breathing:watch}). Chemistry: \omterm{def:g7:what-is-respiration:respiration}{respiration} runs
in things with no chest at all --- peas, mushrooms, \omterm{def:g6:decomposers-and-soil:soil}{soil}
--- detected only by gas tests. Service: \omterm{def:g7:what-is-respiration:respiration}{breathing} exists
to renew the \omterm{def:g4:breathing:organs}{lungs}' air so the gas swap can continue ---
racing exactly when the \omterm{def:g5:first-look-at-cells:cell}{cells}' burning races
(\cref{ex:g4:breathing:running}).
\end{solution}

\begin{solution}{pb:g7:what-is-respiration:1}
\textbf{1.} No change: \omterm{def:g7:what-is-respiration:respiration}{oxygen} steady, \omterm{prop:g7:what-is-respiration:shown}{limewater} clear.
A is the lifeless control --- the stillness every other
jar is read against.

\textbf{2.} B: barely measurable change --- dry peas are
dormant, \omterm{def:g7:what-is-respiration:respiration}{respiration} idling. C: clear \omterm{def:g7:what-is-respiration:respiration}{oxygen} fall and
clouding --- \omterm{def:g2:from-seed-to-plant:germination}{germination} is full-speed building, and the
work is paid in \omterm{def:g7:what-is-respiration:respiration}{respiration}.

\textbf{3.} D: the fastest fall and clouding of all --- a
warm, moving mammal burns hard. The hours are limited
because the mouse is consuming its jar's \omterm{def:g7:what-is-respiration:respiration}{oxygen}; the
experiment must end long before the air runs short.

\textbf{4.} E: \omterm{def:g7:what-is-respiration:respiration}{oxygen} steady or rising --- the lit \omterm{def:g1:plants-around-us:plant}{plant}'s
manufacture takes in \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide} and releases more
\omterm{def:g7:what-is-respiration:respiration}{oxygen} than its \omterm{def:g7:what-is-respiration:respiration}{respiration} uses. F: \omterm{def:g7:what-is-respiration:respiration}{oxygen} falling,
\omterm{prop:g7:what-is-respiration:shown}{limewater} clouding --- in the dark the manufacture stops
and \omterm{def:g7:what-is-respiration:respiration}{respiration}, never stopped, shows plain.

\textbf{5.} F shows that the \omterm{def:g1:plants-around-us:plant}{plant} respires always ---
\omterm{def:g7:what-is-respiration:respiration}{oxygen} in, \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide} out, like the mouse. E hides the
same \omterm{def:g7:what-is-respiration:respiration}{respiration} under the bigger daytime trade: the
manufacture's intake and output swamp it. Together the twins
separate the two exchanges.

\textbf{6.} The germinating pea is working three times as
hard per gram --- building \omterm{def:g1:plants-around-us:parts}{roots} and shoot at full pace ---
and \omterm{def:g7:what-is-respiration:respiration}{respiration} is the engine note of that work.

\textbf{7.} The mouse's \omterm{def:g3:bones-and-muscles:muscle}{muscles} and warm body burn
\omterm{def:g4:journey-of-food:digestion}{nutrients} with \omterm{def:g7:what-is-respiration:respiration}{oxygen}; the burning releases \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide}
into the jar's air; the trapped air passes through the
\omterm{prop:g7:what-is-respiration:shown}{limewater}; the \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide} turns it milky.

\textbf{8.} ``Every \omterm{def:g6:exploring-our-environment:components}{living thing} respires --- \omterm{def:g1:animals-around-us:animal}{animal},
\omterm{def:g1:plants-around-us:plant}{plant}, \omterm{def:g2:from-seed-to-plant:seed}{seed} awake or barely asleep --- taking \omterm{def:g7:what-is-respiration:respiration}{oxygen} and
releasing \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide}.''

\textbf{9.} \omterm{def:g7:what-is-respiration:respiration}{Oxygen} falling, \omterm{prop:g7:what-is-respiration:shown}{limewater} clouding --- the
mushroom respires. It adds that the rule covers \omterm{def:g6:decomposers-and-soil:decomposers}{fungi} too:
neither \omterm{def:g1:plants-around-us:plant}{plant} nor \omterm{def:g1:animals-around-us:animal}{animal}, and no exception.

\textbf{10.} Because a granary is jar C at warehouse scale:
grain kept damp begins germinating and respiring hard,
heating and spoiling the store --- the manager's dryness
rules are \omterm{def:g7:what-is-respiration:respiration}{respiration} control.

\textbf{11.} Everything in it: the pond snails and small
\omterm{def:g1:animals-around-us:animal}{animals}, the mud's \omterm{def:g6:decomposers-and-soil:decomposers}{decomposers}, the \omterm{def:g1:plants-around-us:plant}{plants} and green
single \omterm{def:g5:first-look-at-cells:cell}{cells} in their dark hours. It matters because the
pond's inhabitants all draw on one small dissolved \omterm{def:g7:what-is-respiration:respiration}{oxygen}
stock --- the economy the next chapters open.

\textbf{12.} Only D --- the mouse --- showed \omterm{def:g7:what-is-respiration:respiration}{breathing}:
visible chest movements serving its gas swap. Every jar
with life in it showed \omterm{def:g7:what-is-respiration:respiration}{respiration}: B faintly, C, D, F
plainly, E masked by daylight manufacture but respiring
all the same.
\end{solution}
