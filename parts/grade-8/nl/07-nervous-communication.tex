\chapter{Communicatie via zenuwen}\label{ch:g8:nervous-communication}

\Cref{ch:g5:brain-in-command} tekende de lus --- melden, beslissen,
bevelen --- en mat haar vijfde seconde. Dit jaar openen we de kabels. Wat
reist er eigenlijk door een \omterm{def:g8:nervous-communication:neuron}{zenuw}? Waaruit zijn \omterm{def:g5:brain-in-command:system}{zenuwen} opgebouwd? En wat
gebeurt er in de spleten waar de ene draad ophoudt en de volgende begint
--- de spleten waar, zo zal blijken, zowel het geheugen als veel giffen
hun werk doen?

\section{De bedrading en haar cellen}

\begin{definition}[Zenuwcellen en zenuwimpulsen]\label{def:g8:nervous-communication:neuron}
De werkcellen van het \omterm{def:g5:brain-in-command:system}{zenuwstelsel} zijn de
\emph{zenuwcellen}\index{zenuwcel}
(\cref{rem:g6:cell-unit-of-life:twoways} noemde ze al): elk heeft een
cellichaam en een lange dunne vezel --- sommige vezels reiken van de
\omterm{def:g3:bones-and-muscles:skeleton}{wervelkolom} tot de teen, en zijn de langste cellen van het lichaam. Een
\emph{zenuw}\index{zenuw} is een kabel die duizenden van zulke vezels
bundelt. Door elke vezel reizen \emph{zenuwimpulsen}\index{zenuwimpuls}:
korte elektrische signalen, die tot \qty{100}{m/s} razen (het getal uit
\cref{def:g5:brain-in-command:system}, nu met zijn drager erbij).
\end{definition}

\begin{proposition}[Coderen met frequentie]\label{prop:g8:nervous-communication:code}
De signalen zelf zijn allemaal gelijk --- wat varieert is hun
\emph{tempo}. Een zachte aanraking stuurt een traag getikkel de
gevoelsvezel op; een stevige aanraking een snelle roffel. De taal van het
\omterm{def:g5:brain-in-command:system}{zenuwstelsel} heeft één woord dat op verschillende snelheden herhaald
wordt: de sterkte wordt als frequentie gecodeerd, op elke lijn, heen en
terug.
\end{proposition}
\admitted

\begin{example}[Drie lijnen, één taal]\label{ex:g8:nervous-communication:lines}
De bezetting van de lus (\cref{prop:g5:brain-in-command:loop}), opnieuw
bedraad: \emph{gevoelszenuwcellen} dragen de meldingen --- \omterm{def:g1:the-five-senses:sense}{huid} naar
\omterm{def:g5:brain-in-command:system}{ruggenmerg}, \omterm{def:g1:the-five-senses:sense}{oog} naar \omterm{def:g5:brain-in-command:system}{hersenen}; \emph{bewegingszenuwcellen} dragen de
bevelen --- \omterm{def:g5:brain-in-command:system}{ruggenmerg} naar \omterm{def:g3:bones-and-muscles:muscle}{spier}, en eindigen daar waar de vezel de \omterm{def:g3:bones-and-muscles:muscle}{spier}
raakt en hun roffel de kracht van de samentrekking bepaalt
(\cref{def:g3:bones-and-muscles:muscle}); en daartussen doen ontelbare
\emph{schakelzenuwcellen} in \omterm{def:g5:brain-in-command:system}{hersenen} en \omterm{def:g5:brain-in-command:system}{ruggenmerg} het beslissen. Reflex
of symfonie, het verkeer bestaat uit dezelfde signalen op verschillende
tempo's op verschillende lijnen.
\end{example}

\section{De spleten: synapsen}

\begin{definition}[De synaps]\label{def:g8:nervous-communication:synapse}
\omterm{def:g8:nervous-communication:neuron}{Zenuwcellen} raken elkaar niet. Waar het uiteinde van de ene vezel de
volgende \omterm{def:g5:first-look-at-cells:cell}{cel} ontmoet, ligt een microscopische spleet --- de
\emph{synaps}\index{synaps}. Het elektrische signaal kan er niet
overheen springen; in plaats daarvan geeft de aankomende vezel een
\emph{chemische boodschapper} in de spleet af, die oversteekt, aan de
overkant aanmeert en --- als er genoeg aankomt --- het signaal in de
volgende \omterm{def:g8:nervous-communication:neuron}{zenuwcel} opnieuw start. Elektrisch langs de draad, chemisch bij
elke koppeling: het hele stelsel wisselt de twee af.
\end{definition}

\begin{proposition}[Waarom spleten en geen soldeer]\label{prop:g8:nervous-communication:whygaps}
De spleten lijken een gebrek en zijn juist de vondst van het stelsel:
\begin{itemize}
  \item ze maken koppelingen \emph{instelbaar} --- een \omterm{def:g8:nervous-communication:synapse}{synaps} die vaak
        overgestoken wordt, wordt sterker, en een ongebruikte vervaagt:
        leren (de ``betere paden'' uit
        \cref{rem:g5:brain-in-command:training}, gevonden --- het zijn
        versterkte synapsen);
  \item ze maken signalen \emph{eenrichtings} --- boodschappers worden
        maar aan één kant afgegeven, dus kan het verkeer niet
        achteruitlopen;
  \item ze laten één \omterm{def:g8:nervous-communication:neuron}{zenuwcel} duizenden ingangen \emph{afwegen} ---
        aanjagende en dempende stemmen die bij elke \omterm{def:g5:first-look-at-cells:cell}{cel} opgeteld worden
        voordat ze vuurt: het beslissen van de lus, uitgevoerd bij de
        koppelingen.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[De synapsen van de jongleur]\label{ex:g8:nervous-communication:juggler}
De jongleur uit \cref{rem:g5:brain-in-command:training}, eindelijk
verklaard: een maand oefenen leidde dezelfde signalen over dezelfde
koppelingen tot die synapsen sterker werden --- de routering die moeizaam
afgewogen werd, loopt nu voorafgewogen. Vaardigheid is scheikunde die zich
bij de spleten vastgezet heeft; ``oefening maakt blijvend'' is een
uitspraak over synapsen.
\end{example}

\begin{remark}[De spleten zijn de kwetsbare punten]\label{rem:g8:nervous-communication:vulnerable}
Alles wat van buitenaf op het \omterm{def:g5:brain-in-command:system}{zenuwstelsel} inwerkt, werkt in bij de
synapsen --- de enige plek waar het signaal scheikunde wordt, aan te meren
en te misleiden. Verdovingsmiddelen leggen ze stil; gif blokkeert ze
(sommige verlammen door de boodschapper bij de spierkoppeling te blokkeren,
andere door haar te overspoelen); en de alcohol uit
\cref{prop:g5:healthy-choices:substances}, en elke drug die stemming of
waarneming verandert, werkt door de boodschappers van de spleten na te
bootsen, te blokkeren of uit te putten.
\Cref{ch:g8:protecting-the-nervous-system} bouwt precies op dat punt voort.
\end{remark}

\section{Snelle draden, trage post: twee stelsels vergeleken}

\begin{proposition}[Zenuwen tegenover hormonen]\label{prop:g8:nervous-communication:compare}
Het lichaam laat nu twee communicatiestelsels zien, gebouwd voor
tegenovergestelde taken:
\begin{itemize}
  \item het \emph{\omterm{def:g5:brain-in-command:system}{zenuwstelsel}}: signalen in milliseconden, langs eigen
        lijnen, naar precieze adressen --- één \omterm{def:g3:bones-and-muscles:muscle}{spier}, één klier --- en weg
        zodra het signaal ophoudt: het stelsel van vangbewegingen, spraak
        en reflexen;
  \item het \emph{hormoonstelsel}
        (\cref{def:g8:puberty:hormones}): boodschappers in minuten tot
        maanden, door het \omterm{prop:g4:journey-of-food:absorb}{bloed} omgeroepen naar elk orgaan dat gebouwd is
        om te horen, en die trage programma's door het hele lichaam
        bevelen --- groei, cycli, de verbouwing van de \omterm{def:g8:puberty:puberty}{puberteit}.
\end{itemize}
Telegraaf en postdienst: het lichaam draait allebei, en
\cref{ch:g8:hormonal-communication} geeft de postdienst haar eigen
hoofdstuk.
\end{proposition}
\admitted

\begin{method}[Een communicatie stelsel voor stelsel benoemen]\label{met:g8:nervous-communication:diagnose}
Wijs bij elke coördinatie in het lichaam haar stelsel aan:
\begin{enumerate}
  \item klok haar: milliseconden --- \omterm{def:g5:brain-in-command:system}{zenuwen}; minuten tot jaren ---
        \omterm{def:g8:puberty:hormones}{hormonen} (of allebei, in estafette);
  \item breng haar bereik in kaart: één precies adres --- \omterm{def:g5:brain-in-command:system}{zenuwen}; door
        het hele lichaam --- \omterm{def:g8:puberty:hormones}{hormonen};
  \item zoek de route: een te benoemen draadpad --- \omterm{def:g5:brain-in-command:system}{zenuwen}; een omroep
        door de bloedbaan --- \omterm{def:g8:puberty:hormones}{hormonen};
  \item toets hoe lang ze aanhoudt: weg met het signaal --- \omterm{def:g5:brain-in-command:system}{zenuwen};
        lopend zolang de boodschapper circuleert --- \omterm{def:g8:puberty:hormones}{hormonen}.
\end{enumerate}
\end{method}

\begin{example}[De methode op drie gevallen]\label{ex:g8:nervous-communication:cases}
De knipoog bij een luchtstootje: milliseconden, één spierring, een bedraad
pad, meteen voorbij --- \omterm{def:g5:brain-in-command:system}{zenuwen}. De groeistoot: jaren, het hele lichaam,
de bloedbaan, seizoenen lang lopend --- \omterm{def:g8:puberty:hormones}{hormonen}. Het razende \omterm{def:g4:heart-and-blood:heart}{hart} bij de
sprint (\cref{prop:g7:organs-at-work:supply}): allebei in estafette ---
zenuwsignalen zetten het tempo binnen één slag omhoog, en een boodschapper
in de bloedbaan (de befaamde stoot die je als een hete golf van schrik
voelt) houdt het alarm van het hele lichaam minutenlang vast. Eerst de
telegraaf, dan de post.
\end{example}

\section{Probleem: De reflexopstelling}

\begin{problem}[{Weekendprobleem --- de kniepeesreflex, van instrumenten
voorzien}]\label{pb:g8:nervous-communication:1}
De klassieke demonstratie op de werkbank: de zittende proefpersoon, het
hamertje onder de knieschijf, de onwillekeurige schop van de voet --- nu
van timers en van dit hoofdstuk voorzien.

\medskip\noindent\textbf{Deel I --- De schakeling.}
\begin{enumerate}
  \item Het hamertje rekt een pees op; ongeveer een vijftigste seconde
        later schopt de voet. Zet de schakeling uiteen: welke \omterm{def:g5:biodiversity:species}{soorten}
        \omterm{def:g8:nervous-communication:neuron}{zenuwcellen}, waar doorheen --- en waarom de \omterm{def:g5:brain-in-command:system}{hersenen} niet in deze
        lus zitten.
  \item Een vijftigste seconde is sneller dan de vijfde seconde van de
        balvangst. Verklaar het verschil met de lengte van de schakeling
        en het aantal koppelingen.
  \item De proefpersoon kan de schop niet onderdrukken, ook al is hij
        gewaarschuwd. Wat zegt dat over waar de reflex beslist wordt ---
        en over het eenrichtingsverkeer bij de synapsen?
  \item Meteen na de schop \emph{voelt} de proefpersoon de tik. Verklaar
        de tweede, tragere reis van die vertraging.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- Variaties op de werkbank.}
\begin{enumerate}[resume]
  \item Tik harder: de schop wordt sterker. Welke coderingsregel uit
        \cref{prop:g8:nervous-communication:code} liet zich zien?
  \item De schoolarts legt uit dat deze reflex bij elke controle getest
        wordt. Wat bevestigt een aanwezige, symmetrische kniepeesreflex
        over \omterm{def:g5:brain-in-command:system}{ruggenmerg}, \omterm{def:g5:brain-in-command:system}{zenuwen} en koppelingen?
  \item Een proefpersoon die zijn kaken op elkaar klemt, schopt harder ---
        een bekende versterking. Wat onthult dat over de stemmen van
        schakelzenuwcellen die zelfs ``hersenloze'' lussen bereiken?
  \item Plaatselijke verdoving in het \omterm{def:g1:my-body:parts}{been} heft zowel de schop als het
        gevoel op. Noem het aangrijpingspunt, volgens
        \cref{rem:g8:nervous-communication:vulnerable}.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Het verslag.}
\begin{enumerate}[resume]
  \item Teken de reflex als de lus van drie stappen uit
        \cref{prop:g5:brain-in-command:loop}, en markeer waar deze lus
        afsnijdt.
  \item Vergelijk reflex en vangbeweging in een tabel met twee kolommen:
        pad, koppelingen, tijd, onderdrukbaarheid, rol van oefening.
  \item De laatste paragraaf van het verslag vraagt: ``waarom omzeilen
        snelle beschermende lussen de \omterm{def:g5:brain-in-command:system}{hersenen}?'' Antwoord met de logica
        van de hete pan
        (\cref{exo:g5:brain-in-command:5}).
  \item Eén zin ter afsluiting: wat de werkbank tegelijk liet zien over de
        taal, de bedrading en de koppelingen van het \omterm{def:g5:brain-in-command:system}{zenuwstelsel}.
\end{enumerate}
\end{problem}
