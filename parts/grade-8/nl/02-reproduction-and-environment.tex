\chapter{Voortplanting en de omgeving}\label{ch:g8:reproduction-and-environment}

Waarom begint het kikkerkoor in mei en niet in november? Waarom vult een
goed beukenjaar het bos met zaailingen, en maakt een koude natte juni de
vijver van het volgende voorjaar leeg van kikkertjes? Het vorige hoofdstuk
bouwde het mechanisme van de volgende generatie; dit hoofdstuk steekt het
in de wereld: de voortplanting wordt door de \omterm{def:g6:exploring-our-environment:components}{omgeving} getimed, van
proviand voorzien --- en begrensd.

\section{Getimed door de seizoenen}

\begin{proposition}[Het voortplanten volgt de kalender van de overvloed]\label{prop:g8:reproduction-and-environment:timing}
\omterm{def:g5:biodiversity:species}{Soorten} timen hun voortplanting zo dat de \emph{jongen} aankomen wanneer
de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} en het voedsel hen van dienst zijn
(\cref{prop:g6:life-through-seasons:conditions}):
\begin{itemize}
  \item de pimpelmees legt zo dat haar jongen uitkomen bij de
        voorjaarspiek van de rupsen (de eikenketen uit
        \cref{ex:g3:food-chains:three}, op de dag getimed);
  \item de kikker zet af in de opwarmende vijver, en geeft de
        \omterm{ex:g2:animals-grow-up:frog}{donderpadden} zo een zomer om te groeien vóór de winter;
  \item \omterm{def:g1:plants-around-us:plant}{planten} bloeien op het seizoen van hun bestuivers en laten hun
        \omterm{def:g2:from-seed-to-plant:seed}{zaad} rijpen op de verspreidingskalender
        (\cref{pb:g6:life-through-seasons:1});
  \item het burlen van het hert in de herfst zet de \omterm{def:g2:born-grow-die:cycle}{geboortes} op het
        \emph{volgende voorjaar} --- met de draagtijd meegerekend.
\end{itemize}
De seinen die de machinerie starten zijn de eigen signalen van de
\omterm{def:g6:exploring-our-environment:components}{omgeving}: vooral de lengende dagen, en verder warmte en regen.
\end{proposition}
\admitted

\begin{example}[De lammeren binnen]\label{ex:g8:reproduction-and-environment:lambs}
Schapenhouders buiten het sein rechtstreeks uit: ooien komen in
dekconditie naarmate de dagen \emph{korter} worden; onder kunstlicht dat
eerst lang en dan korter gemaakt wordt, kun je een kudde zo ver krijgen
dat ze voor de lucratieve vroege markt aflammert. De machinerie leest
licht, geen kalenders --- verander het licht en de kalender volgt. Een
bekend sein is een hefboom.
\end{example}

\section{Van proviand voorzien door het leefgebied}

\begin{proposition}[Voortplanting wordt in materie betaald]\label{prop:g8:reproduction-and-environment:cost}
Eieren, \omterm{def:g2:animals-grow-up:born}{melk}, nectar, \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} en geweien zijn allemaal \emph{productie} in
de zin van \cref{def:g6:matter-from-food:production} --- \omterm{def:g6:plants-are-producers:matter}{organische
materie} die de ouders eerst moeten verwerven. Dus stelt de rijkdom van het
\omterm{def:g2:where-animals-live:habitat}{leefgebied} het budget vast:
\begin{itemize}
  \item een goed gevoede kip legt dagelijks; een uitgehongerde stopt ---
        de boekhouding uit \cref{pb:g6:matter-from-food:1} die op de
        eierregel uitkomt;
  \item de beuk draagt alleen massaal \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht} in ``mastjaren'' na goede
        groeiseizoenen --- zaadoogsten zijn gebankierde zomers
        (\cref{ex:g6:plants-are-producers:pantry});
  \item magere jaren dunnen het legsel uit: veel \omterm{prop:g3:sorting-living-things:groups}{vogels} leggen minder
        eieren, of geen, als er weinig prooi is.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Het legsel van de mezen tellen]\label{ex:g8:reproduction-and-environment:clutch}
Onderzoek met nestkasten laat zien dat de legsels van pimpelmezen het
rupsenaanbod volgen: in rijke eikenbossen tien of meer eieren; in arme
tuinen zes of zeven. De vogel beslist dat niet met een boekhouding ---
haar conditie, opgebouwd uit het voedsel van het \omterm{def:g2:where-animals-live:habitat}{leefgebied}, bepaalt
waarvoor ze proviand kan leveren. De legselgrootte is de kwaliteit van het
\omterm{def:g2:where-animals-live:habitat}{leefgebied}, in eieren geschreven.
\end{example}

\section{Begrensd door de wereld: populaties}

\begin{definition}[Populatie]\label{def:g8:reproduction-and-environment:population}
Een \emph{populatie}\index{populatie} is het geheel van alle individuen
van één \omterm{def:g5:biodiversity:species}{soort} die op één plek leven --- de kikkers van de vijver, de
beuken van het bos. De voortplanting voegt eraan toe; sterfte en vertrek
trekken ervan af; de \omterm{def:g6:exploring-our-environment:components}{omgeving} stelt het plafond vast.
\end{definition}

\begin{proposition}[Waarom populaties niet ontploffen]\label{prop:g8:reproduction-and-environment:limits}
Elke \omterm{def:g5:biodiversity:species}{soort} produceert te veel --- duizenden eitjes per kikkerpaar,
duizenden \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} per beuk
(\cref{prop:g4:animal-reproduction:strategies}) --- en toch staan vijvers
niet vol kikkers. Het overschot wordt tegen de rem van de \omterm{def:g6:exploring-our-environment:components}{omgeving}
uitgegeven:
\begin{itemize}
  \item \emph{predatie} --- de reigers en libellenlarven uit
        \cref{ex:g4:animal-reproduction:pondcount};
  \item \emph{voedsel en ruimte} --- er zijn maar zo veel rupsen,
        nestholtes en lichtplekken op de bosbodem;
  \item \emph{\omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}} --- late vorst op de bloesem
        (\cref{exo:g3:flowers-fruits-seeds:9}), droogte op de krimpende
        vijver van de \omterm{ex:g2:animals-grow-up:frog}{donderpadden}.
\end{itemize}
Gemiddeld vervangt elk paar in een stabiel \omterm{def:g2:where-animals-live:habitat}{leefgebied} zichzelf --- het
evenwicht uit \cref{prop:g5:food-webs:balance}, van de kant van de ouders
bekeken.
\end{proposition}
\admitted

\begin{example}[Twee voorjaren bij de vijver]\label{ex:g8:reproduction-and-environment:twosprings}
Voorjaar 1, zacht en nat: hoog water, vroege \omterm{prop:g3:sorting-living-things:groups}{insecten} --- duizenden
eitjes, honderden kikkertjes, de \omterm{def:g8:reproduction-and-environment:population}{populatie} klimt. Voorjaar 2, een droogte:
de vijver krimpt tot een plas, de \omterm{ex:g2:animals-grow-up:frog}{donderpadden} verdringen elkaar, de
\omterm{def:g7:what-is-respiration:respiration}{zuurstof} zakt (\cref{prop:g7:oxygen-in-the-water:drains}), en de reigers
smullen van de samengedrukte school --- een handvol kikkertjes. Dezelfde
kikkers, hetzelfde mechanisme, twee vonnissen: de \omterm{def:g6:exploring-our-environment:components}{omgeving}
medeondertekent elke generatie.
\end{example}

\begin{remark}[Mensen verzetten de remmen]\label{rem:g8:reproduction-and-environment:humans}
De meeste menselijke effecten op wilde \omterm{def:g1:animals-around-us:animal}{dieren} lopen via die remmen en niet
via het kernmechanisme: gedempte vijvers en gerooide heggen halen
voortplantingsgebied weg
(\cref{prop:g2:where-animals-live:loss}); bestrijdingsmiddelen legen de
rupsenpiek waarop de mezen zich getimed hadden; nestkasten in de tuin en
de herstelde natte weiden uit \cref{ex:g5:biodiversity:storks} geven
plafond terug. Een \omterm{def:g5:biodiversity:species}{soort} beschermen betekent in de praktijk de \omterm{def:g6:exploring-our-environment:components}{omgeving}
beschermen waarin haar voortplanting is ingestoken --- sein, budget en
remmen samen.
\end{remark}

\begin{method}[Een verandering in een populatie doorlichten]\label{met:g8:reproduction-and-environment:audit}
Als een \omterm{def:g8:reproduction-and-environment:population}{populatie} stijgt of daalt, licht dan op volgorde door:
\begin{enumerate}
  \item \emph{timing}: vielen het sein en de piek in de voorraad nog
        samen? (een te vroeg voorjaar kan de jongen na de rupsen laten
        stranden);
  \item \emph{budget}: waarvoor liet het \omterm{def:g2:where-animals-live:habitat}{leefgebied} van dat seizoen de
        ouders proviand leveren?
  \item \emph{remmen}: welke veranderden --- roofdieren, voedsel, ruimte,
        \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}?
  \item \emph{\omterm{def:g2:where-animals-live:habitat}{leefgebied}}: is er voortplantingsgebied zelf bij gekomen of
        verdwenen?
\end{enumerate}
\end{method}

\section{Probleem: De nestkastenboekhouding}

\begin{problem}[{Weekendprobleem --- tien jaar pimpelmezen, kast voor kast
doorgelicht}]\label{pb:g8:reproduction-and-environment:1}
Een school volgt tien jaar lang twintig nestkasten: legdatums,
legselgroottes, uitgevlogen jongen per kast --- naast het weerlogboek en
het dagboek van de terreinbeheerder.

\medskip\noindent\textbf{Deel I --- De normale jaren
lezen.}
\begin{enumerate}
  \item Jaar 1--4: leggen eind april, legsels van acht tot tien, en de
        meeste eieren vliegen uit. Noem het sein dat eind april vastzet,
        en de piek waarop de timing mikt.
  \item De legsels van jaar 3 waren een \omterm{def:g2:animals-grow-up:born}{ei} of twee groter na een warme,
        rupsenrijke mei in jaar 2 --- de ouders overwinterden goed. Welke
        propositie illustreert dat verband?
  \item Zelfs in goede jaren vliegt ongeveer een vijfde van de eieren
        nooit uit. Noem de remmen die daar waarschijnlijk verantwoordelijk
        voor zijn, uit
        \cref{prop:g8:reproduction-and-environment:limits}.
  \item Waarom noteert de school naast de eieren ook het weer en het werk
        op het terrein? (Antwoord met de vier vragen van de doorlichting.)
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De afwijkingen.}
\begin{enumerate}[resume]
  \item Jaar 5: een uitzonderlijk warme maart bracht de rupsen twee weken
        te vroeg; de legdatums schoven nauwelijks op. Voorspel het
        uitvliegen van jaar 5, en noem de regel in de doorlichting die
        faalt.
  \item Jaar 6: het dagboek van de terreinbeheerder vermeldt dat de dode
        takken van de oude eiken verwijderd zijn en dat de weidestrook
        maandelijks gemaaid is. Het uitvliegen daalt hoewel het weer
        vriendelijk was. Welke regels van de doorlichting wijst het
        dagboek aan?
  \item Jaar 7: er vestigt zich een sperwerpaar. De legsels blijven
        gelijk, maar het aantal uitgevlogen jongen zakt. Welke rem
        verschoof er, en waarom verschoof de \emph{legsel}regel niet mee?
  \item Jaar 8: het aantal kasten wordt naar veertig verdubbeld; het
        totale aantal uitgevlogen jongen stijgt, maar het succes
        \emph{per kast} blijft gelijk. Welk plafond hebben de nieuwe
        kasten opgetild, en welk lieten ze onaangeroerd?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Het verslag.}
\begin{enumerate}[resume]
  \item Schrijf de samenvattende zin van tien jaar die de legselgrootte
        aan de kwaliteit van het \omterm{def:g2:where-animals-live:habitat}{leefgebied} koppelt, met
        \cref{ex:g8:reproduction-and-environment:clutch}.
  \item De gemeente stelt voor de eikenrij te kappen voor een
        parkeerplaats, en dat te ``compenseren'' met vijftig extra
        nestkasten. Licht dat aanbod door: welke van sein, budget, remmen
        en \omterm{def:g2:where-animals-live:habitat}{leefgebied} vervangen de kasten, en welke leveren alleen de
        eiken?
  \item Beveel twee terreinmaatregelen voor jaar 11 aan, elk gekoppeld aan
        haar regel in de doorlichting.
  \item Sluit af met de moraal van het hoofdstuk in één zin: wat heeft elk
        uitvliegend jong, naast twee ouders, nodig?
\end{enumerate}
\end{problem}
