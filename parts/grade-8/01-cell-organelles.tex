\chapter{Cell Organelles}\label{ch:g8:organelles}

Eukaryotic \omterm{def:g7:microscope:cell}{cells} are busy factories. Membrane-bound compartments ---
\omterm{def:g8:organelles:organelle}{organelles} --- divide the work of \omterm{def:g6:nutrition:def}{energy}, \omterm{def:g6:nutrition:groups}{protein} production, storage and
\omterm{def:g6:method:control}{control}. This chapter maps the main \omterm{def:g8:organelles:organelle}{organelles} and links structure to
function.

\section{Why compartments help}

\begin{definition}[Organelle]\label{def:g8:organelles:organelle}
An \emph{organelle}\index{organelle} is a specialised structure inside a
\omterm{def:g7:microscope:cell}{cell} that performs a particular job. Many eukaryotic organelles are
bounded by membranes.
\end{definition}

\begin{proposition}[Division of labour]\label{prop:g8:organelles:labour}
Separating jobs into \omterm{def:g8:organelles:organelle}{organelles} raises efficiency: enzymes and substrates
for one process stay together, harmful \omterm{def:g7:digestion:def}{digestion} can be contained, and
surfaces (inner membranes) expand where reactions need area.
\end{proposition}
\admitted

\section{Control and instructions}

\begin{definition}[Nucleus]\label{def:g8:organelles:nucleus}
The \emph{nucleus}\index{nucleus (cell)} holds most of the \omterm{def:g7:microscope:cell}{cell}'s DNA as
chromatin/\omterm{def:g7:heredity:gene}{chromosomes}. The nuclear envelope is a double membrane with
pores. The \emph{nucleolus}\index{nucleolus} helps build \omterm{def:g8:organelles:ribosome}{ribosome} parts.
\end{definition}

\begin{definition}[Ribosomes]\label{def:g8:organelles:ribosome}
\emph{Ribosomes}\index{ribosome} are not membrane-bound; they assemble
\omterm{def:g6:nutrition:groups}{proteins}. Free ribosomes work in the cytosol; others stud the rough ER.
\end{definition}

\section{Endomembrane system}

\begin{definition}[ER and Golgi]\label{def:g8:organelles:endo}
The \emph{endoplasmic reticulum}\index{endoplasmic reticulum} (ER) is a
membrane network. Rough ER (with \omterm{def:g8:organelles:ribosome}{ribosomes}) helps make and fold \omterm{def:g6:nutrition:groups}{proteins}
for membranes or export. Smooth ER is involved in \omterm{def:g6:nutrition:groups}{lipids} and some
detoxification. The \emph{Golgi apparatus}\index{Golgi apparatus} modifies,
sorts and packages materials into vesicles.
\end{definition}

\begin{definition}[Lysosomes and vacuoles]\label{def:g8:organelles:lyso}
\emph{Lysosomes}\index{lysosome} (common in \omterm{def:g1:plants-animals:animal}{animal} \omterm{def:g7:microscope:cell}{cells}) contain
digestive enzymes that break down macromolecules and worn parts.
\emph{Vacuoles}\index{vacuole} store materials; \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g7:microscope:cell}{cells} often have a
large central vacuole that helps maintain firmness (turgor).
\end{definition}

\section{Energy organelles}

\begin{definition}[Mitochondria and chloroplasts]\label{def:g8:organelles:energy}
\emph{Mitochondria}\index{mitochondrion} carry out most of aerobic
respiration, producing ATP (usable \omterm{def:g6:nutrition:def}{energy} currency).
\emph{Chloroplasts}\index{chloroplast} in \omterm{def:g1:plants-animals:plant}{plants} and algae perform
\omterm{def:g7:photo-resp:photo}{photosynthesis}. Both have double membranes and their own DNA --- clues to
ancient bacterial origins (endosymbiosis idea).
\end{definition}

\begin{definition}[Cell wall]\label{def:g8:organelles:wall}
\omterm{def:g1:plants-animals:plant}{Plant} \omterm{def:g7:microscope:cell}{cells} have a cellulose \emph{cell wall}\index{cell wall} outside
the plasma membrane for support and protection. \omterm{def:g1:plants-animals:animal}{Animal} \omterm{def:g7:microscope:cell}{cells} lack this
wall.
\end{definition}

\begin{example}[Specialisation]\label{ex:g8:organelles:spec}
\omterm{def:g3:bones:muscle}{Muscle} \omterm{def:g7:microscope:cell}{cells} are rich in mitochondria; \omterm{def:g7:plant-structure:organs}{leaf} palisade \omterm{def:g7:microscope:cell}{cells} are rich in
\omterm{def:g7:microscope:parts}{chloroplasts}; pancreatic secretory \omterm{def:g7:microscope:cell}{cells} are rich in rough ER and Golgi.
\end{example}

\begin{method}[Organelle $\to$ job card]\label{met:g8:organelles:card}
For each \omterm{def:g8:organelles:organelle}{organelle}: (1) name it; (2) say if \omterm{def:g1:plants-animals:plant}{plants}, \omterm{def:g1:plants-animals:animal}{animals} or both;
(3) state one main function; (4) note one structural clue (e.g.\ double
membrane, \omterm{def:g8:organelles:ribosome}{ribosomes} attached).
\end{method}
