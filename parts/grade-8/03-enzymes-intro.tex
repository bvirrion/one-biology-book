\chapter{Enzymes at Work}\label{ch:g8:enzymes}

Chemical reactions of life must run fast enough at \omterm{def:g1:my-body:body}{body} temperature.
\omterm{def:g8:enzymes:enzyme}{Enzymes} make that possible. This chapter treats \omterm{def:g8:enzymes:enzyme}{enzymes} as biological
\omterm{def:g8:enzymes:catalyst}{catalysts} with a lock-and-key fit that depends on shape.

\section{What enzymes are}

\begin{definition}[Enzyme]\label{def:g8:enzymes:enzyme}
An \emph{enzyme}\index{enzyme} is a biological \omterm{def:g8:enzymes:catalyst}{catalyst} --- usually a
\omterm{def:g6:nutrition:groups}{protein} --- that speeds up a specific reaction without being used up.
Each enzyme has an \emph{active site}\index{active site} that binds its
\emph{substrate}\index{substrate (enzyme)}.
\end{definition}

\begin{definition}[Catalyst]\label{def:g8:enzymes:catalyst}
A \emph{catalyst}\index{catalyst} increases reaction rate without being
consumed. \omterm{def:g8:enzymes:enzyme}{Enzymes} lower the activation \omterm{def:g6:nutrition:def}{energy} needed for a reaction to
proceed.
\end{definition}

\begin{proposition}[Lock and key]\label{prop:g8:enzymes:lock}
In the simple \emph{lock-and-key model}\index{lock-and-key model}, the
\omterm{def:g8:enzymes:enzyme}{substrate} fits the \omterm{def:g8:enzymes:enzyme}{active site} like a key in a lock. Only the correct
shape reacts efficiently --- explaining \omterm{def:g8:enzymes:enzyme}{enzyme} specificity.
\end{proposition}
\admitted

\begin{example}[Digestive enzymes]\label{ex:g8:enzymes:digest}
Amylase breaks starch toward sugars; proteases break \omterm{def:g6:nutrition:groups}{proteins} toward amino
acids; lipases break \omterm{def:g6:nutrition:groups}{fats}. Each has a different \omterm{def:g8:enzymes:enzyme}{active site} shape.
\end{example}

\section{Conditions that matter}

\begin{definition}[Optimum]\label{def:g8:enzymes:optimum}
An \omterm{def:g8:enzymes:enzyme}{enzyme} has an \emph{optimum temperature}\index{optimum temperature} and
often an \emph{optimum pH}\index{optimum pH} where activity is highest
under the conditions tested.
\end{definition}

\begin{proposition}[Temperature effects]\label{prop:g8:enzymes:temp}
Raising temperature (within a range) increases collision rate and speeds
reactions. Too high a temperature can \emph{denature}\index{denaturation}
the \omterm{def:g8:enzymes:enzyme}{enzyme}: the \omterm{def:g8:enzymes:enzyme}{active site} shape is destroyed and activity falls sharply.
Cooling slows reactions without always permanently denaturing.
\end{proposition}
\admitted

\begin{proposition}[pH effects]\label{prop:g8:enzymes:pH}
Extreme pH can also \omterm{prop:g8:enzymes:temp}{denature} \omterm{def:g8:enzymes:enzyme}{enzymes} or change \omterm{def:g8:enzymes:enzyme}{active site} charges.
\omterm{def:g3:teeth:path}{Stomach} protease works in acid; intestinal \omterm{def:g8:enzymes:enzyme}{enzymes} need milder conditions.
\end{proposition}
\admitted

\begin{definition}[Substrate concentration]\label{def:g8:enzymes:sub}
At low \omterm{def:g8:enzymes:enzyme}{substrate} levels, rate rises as more \omterm{def:g8:enzymes:enzyme}{active sites} fill. At high
\omterm{def:g8:enzymes:enzyme}{substrate}, the \omterm{def:g8:enzymes:enzyme}{enzyme} becomes saturated: rate levels off until more \omterm{def:g8:enzymes:enzyme}{enzyme}
is added.
\end{definition}

\begin{method}[Investigating an enzyme]\label{met:g8:enzymes:invest}
\begin{enumerate}
  \item Choose a clear dependent measure (gas volume, colour change time,
        mass loss).
  \item Change one \omterm{def:g6:method:variables}{independent variable} (temperature, pH, concentration).
  \item Keep other conditions constant; use repeats.
  \item Include a \omterm{def:g6:method:control}{control} without \omterm{def:g8:enzymes:enzyme}{enzyme} or with boiled \omterm{def:g8:enzymes:enzyme}{enzyme}.
\end{enumerate}
\end{method}
