\chapter{الدهون}\label{ch:b1:lipids}

جمل يقطع صحراء يحمل ثلاثين كيلوغرامًا من \omterm{def:b1:lipids:triglyceride}{الشحم} في سنامه ولا يحمل فيه
ماءً البتة؛ وسلمون مرقط في جدول جبلي عند \qty{4}{\celsius} أغشيته
مائعة ميوعة \omterm{def:b1:membranes-transport:membrane}{أغشية} شبوط في بركة دافئة؛ وقطرة من \omterm{def:b1:lipids:triglyceride}{زيت} الزيتون ترجّ في
ماء فتتفتت إلى سحابة من القطيرات، إن تُركت اجتمعت من جديد في واحدة.
وهذه الثلاثة كلها دهون في العمل: أكثف مخزن للطاقة الكيميائية يستطيع
\omterm{def:b1:organism-environment:organism}{كائن حي} أن يحمله، والغلالة التي تحد كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} وكل عضية، وصنف الجزيئات
الذي يأبى الماء أن يذيبه. يصف هذا الفصل \omterm{def:b1:lipids:fattyacid}{الحموض الدسمة} والشحوم
المبنية منها، \omterm{def:b1:lipids:fattyacid}{والدهون} \omterm{def:b1:lipids:amphiphilic}{المزدوجة الألفة} التي تتجمع في \omterm{def:b1:membranes-transport:membrane}{أغشية}، وما يجعل
الغشاء أكثر ميوعة أو أقل، \omterm{def:b1:lipids:fattyacid}{والدهون} الأخرى --- \omterm{def:b1:lipids:amphiphilic}{الستيرولات}، والأصبغة،
والهرمونات --- التي يمكن أن تكونها بضع ذرات كربون منتظمة في حلقات
وسلاسل.

\section{الحموض الدسمة}

\begin{definition}[الدهن والحمض الدسم]\label{def:b1:lipids:fattyacid}
\emph{الدهون}\index{دهن} هي الجزيئات الحيوية التي لا تُعرَّف ببنية
مشتركة بل بخاصية مشتركة: أنها غير منحلة في الماء ومنحلة في المذيبات
غير القطبية. ومعظمها مبني على \emph{حموض دسمة}\index{حمض دسم}: وهي
حموض كربوكسيلية بسلسلة هيدروكربونية غير متفرعة من
\qtyrange{12}{24}{} ذرة كربون، وعددها زوجي دائمًا تقريبًا (لأنها
تُركَّب ذرتي كربون في كل مرة، \cref{ch:b1:biosyntheses-integration}).
\omterm{def:b1:water-small-molecules:ph}{والحمض} الدسم \emph{المشبع}\index{حمض دسم مشبع} ليس فيه إلا روابط
مفردة وسلسلته مستقيمة مرنة؛ و\emph{غير المشبع}\index{حمض دسم غير مشبع}
فيه رابطة مزدوجة أو أكثر، في تشكيل \emph{سيس}، وكل منها تضع في
السلسلة انثناءً صلبًا قدره نحو $30^\circ$. والترميز: C18:1
$\Delta^9$ \omterm{def:b1:water-small-molecules:ph}{حمض} من 18 كربونًا برابطة مزدوجة واحدة بعد الكربون 9 (\omterm{def:b1:water-small-molecules:ph}{حمض}
الأولييك)؛ \omterm{def:b1:water-small-molecules:ph}{والحمض} \emph{أوميغا-3} آخر رابطة مزدوجة فيه على بعد ثلاث
ذرات كربون من نهاية الميثيل.
\end{definition}

\begin{proposition}[درجات الانصهار]\label{prop:b1:lipids:melting}
درجة انصهار \omterm{def:b1:lipids:fattyacid}{الحمض الدسم}، وميوعة أي تجمع دهني ينتمي إليه، تحددها شدة
ارتصاص السلاسل: فهي ترتفع مع طول السلسلة (\qty{44}{\celsius} عند
C12:0، و\qty{63}{\celsius} عند C16:0، و\qty{70}{\celsius} عند C18:0)
وتهبط هبوطًا حادًا مع كل رابطة مزدوجة \emph{سيس} (C18:0
\qty{70}{\celsius}، وC18:1 \qty{13}{\celsius}، وC18:2
\qty{-5}{\celsius}، وC18:3 \qty{-11}{\celsius}). فالشحوم الحيوانية،
الغنية بالحموض المشبعة، صلبة في حرارة الغرفة؛ والزيوت النباتية
والسمكية، الغنية بغير المشبعة، سائلة.
\end{proposition}
\begin{proof}
السلاسل المستقيمة تصطف جنبًا إلى جنب وتعقد كل زمرة $\mathrm{CH_2}$
تماسات \omterm{def:b1:water-small-molecules:bonds}{فان دير فالس} مع جاراتها، كل منها بضعة كيلوجولات في المول،
مجموعة على امتداد السلسلة: فكلما زاد الكربون زادت التماسات ولزم
لفصلها حرارة أكثر. والانثناء يمنع التماس الوثيق على عدة ذرات كربون
من كل جانب فيزيل تماسات كثيرة دفعة واحدة.
\end{proof}

\section{الشحوم: مخزن الطاقة}

\begin{definition}[ثلاثي الغليسيريد]\label{def:b1:lipids:triglyceride}
\emph{ثلاثي الغليسيريد}\index{ثلاثي الغليسيريد} (ثلاثي أسيل
الغليسرول، وهو \emph{شحم}\index{شحم} حين يكون صلبًا
و\emph{زيت}\index{زيت} حين يكون سائلًا) هو غليسرول --- وهو كحول من
ثلاث ذرات كربون --- مؤستَر على هيدروكسيلاته الثلاثة بثلاثة \omterm{def:b1:lipids:fattyacid}{حموض
دسمة}. ولا شحنة له ولا زمرة قطبية باقية فيه: فهو كاره للماء كله،
ويُخزَّن قطيرات لا ماء فيها في \emph{الخلايا الشحمية}\index{خلية شحمية}،
وهي \omterm{def:b1:cell-unit-of-life:cell}{خلايا} \omterm{def:b1:body-plans-tissues:tissue}{النسيج} الشحمي التي لا تكاد تكون أكثر من قطيرة شحم ونواة
مدفوعة إلى جانب.
\end{definition}

\begin{proposition}[الشحم أكثف مخازن الطاقة]\label{prop:b1:lipids:energystore}
أكسدة \qty{1}{g} من \omterm{def:b1:lipids:triglyceride}{الشحم} تحرر \qty{38}{kJ}، مقابل \qty{17}{kJ}
لكل \qty{1}{g} من الكربوهيدرات أو البروتين: فكربونات \omterm{def:b1:lipids:fattyacid}{الحمض الدسم} أشد
اختزالًا (روابط C--H أكثر وروابط C--O أقل) فتعطي الأكسجين
إلكترونات أكثر. ثم إن \omterm{def:b1:lipids:triglyceride}{الشحم} يُخزَّن جافًا، بينما يربط الغليكوجين نحو
\qty{3}{g} من الماء لكل غرام: أي إن \qty{1}{g} من الغليكوجين المخزون
يعطي \qty{4}{kJ} لكل غرام من الكتلة المخزونة، \omterm{def:b1:lipids:triglyceride}{والشحم} أكثر بتسع مرات.
والإنسان النحيل يحمل \qty{0.5}{kg} من الغليكوجين (طاقة يوم)
و\qty{12}{kg} من \omterm{def:b1:lipids:triglyceride}{الشحم} (أكثر من طاقة شهر).
\end{proposition}

\begin{example}[لماذا لا يُخزَّن كل شيء شحمًا]\label{ex:b1:lipids:whyglycogen}
لا يُحرق \omterm{def:b1:lipids:triglyceride}{الشحم} إلا بالأكسجين، وببطء، ولا يمكن تحويله راجعًا إلى
غلوكوز عند الحيوان (\cref{ch:b1:biosyntheses-integration})؛ والدماغ
والكريات الحمراء تحتاج إلى الغلوكوز، والعضلة العادية تحتاج إليه أسرع
مما يستطيع \omterm{def:b1:lipids:triglyceride}{الشحم} أن يمده. فالغليكوجين هو مخزن الغلوكوز السريع الخالي
من الأكسجين لساعات؛ \omterm{def:b1:lipids:triglyceride}{والشحم} هو المخزن الكثيف لأسابيع. فالطائر المهاجر
والزغبة المسبتة والجمل \omterm{def:b1:lipids:triglyceride}{شحم}؛ وساقا العدّاء غليكوجين.
\end{example}

\begin{definition}[الشموع]\label{def:b1:lipids:wax}
\emph{الشموع}\index{شمع} إسترات \omterm{def:b1:lipids:fattyacid}{حمض دسم} مع كحول طويل السلسلة: صلبة،
كارهة للماء كلها، تُستعمل طلاءات --- \omterm{def:b1:flowering-plant-organization:tissues}{أديمة} \omterm{def:b1:flowering-plant-organization:organs}{الأوراق}
(\cref{ch:b1:flowering-plant-organization})، وسطح الحشرات، وكتم ماء
الريش والفرو، وقرص النحل.
\end{definition}

\section{دهون الأغشية والتجمع الذاتي}

\begin{definition}[الدهون المزدوجة الألفة]\label{def:b1:lipids:amphiphilic}
دهون \omterm{def:b1:membranes-transport:membrane}{الأغشية} \emph{مزدوجة الألفة}\index{جزيء مزدوج الألفة}:
\emph{رأس} \omterm{def:b1:water-small-molecules:water}{قطبي} يذيبه الماء و\emph{ذيل} أو ذيلان غير قطبيين
يستبعدهما. و\emph{الفوسفوليبيدات الغليسرولية}\index{فوسفوليبيد}
غليسرول يحمل حمضين دسمين، وعلى الكربون الثالث فوسفاتًا موصولًا بكحول
\omterm{def:b1:water-small-molecules:water}{قطبي} صغير (كولين، وإيثانولامين، وسيرين، وإينوزيتول): وفوسفاتيديل
الكولين أشيع دهون \omterm{def:b1:membranes-transport:membrane}{الأغشية} الحيوانية.
و\emph{السفينغوليبيدات}\index{سفينغوليبيد} مبنية على الكحول الأميني
سفينغوزين بدل الغليسرول، \omterm{def:b1:lipids:fattyacid}{بحمض دسم} واحد ورأس من فوسفوكولين
(سفينغوميالين) أو من سكريات (غليكوليبيدات)، وهي غنية في الصفيحة
الخارجية وفي أغماد الأعصاب. و\emph{الستيرولات}\index{ستيرول} ---
و\emph{الكولستيرول}\index{كولستيرول} عند الحيوان، وستيرولات قريبة
منه عند النبات والفطر --- جزيئات صلبة رباعية الحلقات لها هيدروكسيل
واحد رأسًا، وهي تقع بين ذيول \omterm{def:b1:lipids:fattyacid}{الدهون} الأخرى.
\end{definition}

\begin{proposition}[التجمع الذاتي]\label{prop:b1:lipids:selfassembly}
\index{التجمع الذاتي}\index{مذيلة}\index{ليبوزوم}
في الماء تتجمع الجزيئات \omterm{def:b1:lipids:amphiphilic}{المزدوجة الألفة} تلقائيًا بحيث تخفي ذيولها
وتعرض رؤوسها. وشكل الجزيء يحسم البنية: \omterm{def:b1:lipids:fattyacid}{فالدهون} ذوات الذيل الواحد
(\omterm{def:b1:lipids:fattyacid}{الحموض الدسمة}، والمنظفات)، وشكلها مخروطي، تكوّن \emph{مذيلات} ---
وهي كرات قطرها بضعة نانومترات وذيولها في الداخل؛ والفوسفوليبيدات ذوات
الذيلين، وشكلها أسطواني، تكوّن \emph{طبقات مضاعفة} تنغلق على نفسها
في \emph{ليبوزومات} (حويصلات) حتى لا تترك حافة مكشوفة. ويقود
التجمعَ الأثرُ الكاره للماء (\cref{ch:b1:water-small-molecules}): فهو
لا يكلف طاقة ولا يحتاج إلى إنزيم، والطبقة المضاعفة الممزقة تلتئم من
تلقاء نفسها. \omterm{def:b1:membranes-transport:membrane}{فالأغشية} صفائح ذاتية الالتئام تنمو بإقحام دهون جديدة
ولا تتكون من عدم أبدًا: فكل غشاء يأتي من غشاء.
\end{proposition}
\begin{proof}[شواهد]
\omterm{def:b1:lipids:amphiphilic}{الفوسفوليبيد} المجفف المشتت في الماء يكوّن حويصلات مغلقة جدرها طبقات
مضاعفة تُرى في \omterm{prop:b1:cell-unit-of-life:microscopes}{المجهر الإلكتروني} (بانغهام، 1965)؛ ونفوذيتها للشوارد
والماء توافق نفوذية \omterm{def:b1:membranes-transport:membrane}{أغشية} \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}، ويمكن تحميلها بالأدوية وإدماجها
\omterm{def:b1:cell-unit-of-life:cell}{بالخلايا}. وإضافة منظف (ذي ذيل واحد مخروطي الشكل) تذيب الطبقة
المضاعفة إلى مذيلات مختلطة؛ وإزالته بالديلزة تدع الطبقة تتكون من
جديد.
\end{proof}

\begin{example}[الأملاح الصفراوية]\label{ex:b1:lipids:bile}
يصل \omterm{def:b1:lipids:triglyceride}{الشحم} الغذائي إلى المعي زيتًا؛ واللِّيباز الذي يهضمه لا يعمل إلا
عند سطح الفصل بين \omterm{def:b1:lipids:triglyceride}{الزيت} والماء. والأملاح الصفراوية، وهي مشتقات
\omterm{def:b1:lipids:amphiphilic}{مزدوجة الألفة} من \omterm{def:b1:lipids:amphiphilic}{الكولستيرول}، تغلف \omterm{def:b1:lipids:triglyceride}{الشحم} في قطيرات قطرها ميكرومتر،
فتضاعف سطح الفصل ألف مرة، ثم تحمل ما تحرر من \omterm{def:b1:lipids:fattyacid}{حموض دسمة}، في مذيلات،
إلى \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الماصة (\cref{ch:b1:digestion-absorption}). والصابون
يستعمل الحيلة نفسها.
\end{example}

\section{ميوعة الغشاء}

\begin{definition}[الميوعة والانتقال الطوري]\label{def:b1:lipids:fluidity}
\omterm{def:b1:membranes-transport:membrane}{للطبقة الدهنية المضاعفة} \emph{درجة حرارة انتقال طوري}\index{درجة حرارة الانتقال الطوري}
$T_m$: فدونها ترتص السلاسل في حالة منتظمة شبيهة بالهلام لا تكاد
\omterm{def:b1:lipids:fattyacid}{الدهون} تتحرك فيها؛ وفوقها تكون مضطربة وتكون الطبقة المضاعفة مائعًا
ثنائي البعد يبدل فيه \omterm{def:b1:lipids:fattyacid}{الدهن} مكانه مع جاره مليون مرة في الثانية ويقطع
مستوي الغشاء بسرعة \qty{1}{\micro m/s}. \omterm{def:b1:cell-unit-of-life:cell}{والخلايا} تبقي أغشيتها فوق
$T_m$: إذ إن \emph{الميوعة}\index{ميوعة الغشاء} تتيح للبروتينات أن تنتشر
وتدور، وللحويصلات أن تتبرعم وتندمج، وللطبقة المضاعفة أن تلتئم.
\end{definition}

\begin{proposition}[ما الذي يضبط الميوعة]\label{prop:b1:lipids:fluidityfactors}
ترتفع $T_m$ مع طول السلاسل وتهبط مع عدم إشباعها، كما عند \omterm{def:b1:lipids:fattyacid}{الحموض
الدسمة} الحرة؛ فطبقة مضاعفة من سلاسل C18:0 تنصهر عند
\qty{55}{\celsius}، وأخرى من سلاسل C18:1 عند \qty{-20}{\celsius}.
\omterm{def:b1:lipids:amphiphilic}{والكولستيرول}، مقحمًا بين الذيول، له أثر مزدوج: ففوق $T_m$ تقيّد
حلقاته الصلبة حركة السلاسل فتصلّب الغشاء؛ ودون $T_m$ تمنع السلاسل من
الارتصاص فتحول دون تجمده. فهو يوسّع الانتقال إلى تغير تدريجي ويبقي
\omterm{def:b1:lipids:fluidity}{الميوعة} شبه ثابتة على مدى واسع من الحرارة --- أي \omterm{def:b1:water-small-molecules:buffer}{هادئة} ميوعة.
\omterm{def:b1:membranes-transport:membrane}{والأغشية} البلازمية الحيوانية تبلغ كولستيرولًا واحدًا لكل
\omterm{def:b1:lipids:amphiphilic}{فوسفوليبيد}.
\end{proposition}

\begin{proposition}[التكيف بثبات اللزوجة]\label{prop:b1:lipids:homeoviscous}
\index{التكيف بثبات اللزوجة}
\omterm{def:b1:organism-environment:organism}{الكائنات} التي لا تستطيع تنظيم حرارتها تعدّل تركيب أغشيتها لتبقي
\omterm{def:b1:lipids:fluidity}{الميوعة} ثابتة: فكلما هبطت الحرارة استبدلت \omterm{def:b1:lipids:fattyacid}{بالحموض الدسمة} المشبعة
حموضًا غير مشبعة وبالسلاسل الطويلة سلاسل أقصر، والعكس حين ترتفع.
\end{proposition}
\begin{proof}[شواهد]
\emph{E. coli} المزروعة عند \qty{10}{\celsius} فيها ضعف نسبة \omterm{def:b1:lipids:fattyacid}{الحموض
الدسمة} غير المشبعة التي في السلالة نفسها مزروعةً عند
\qty{40}{\celsius}، \omterm{def:b1:membranes-transport:membrane}{وأغشية} المزرعتين، مقيسةً بحركية مسبار، متساوية
\omterm{def:b1:lipids:fluidity}{الميوعة} عند حرارتي نموهما. والسلمون المرقط المؤقلم عند
\qty{5}{\celsius} يحمل في دهون أغشيته حموضًا متعددة عدم الإشباع أكثر
مما يحمله السلمون عند \qty{20}{\celsius}؛ \omterm{def:b1:lipids:triglyceride}{وشحم} سيقان الرنة، القريبة
من الثلج، أقل إشباعًا من \omterm{def:b1:lipids:triglyceride}{شحم} الجذع. والنباتات التي تنجو من الصقيع
تُغني أغشيتها \omterm{def:b1:lipids:fattyacid}{بالدهون} غير المشبعة في الخريف، والطافرات العاجزة عن
ذلك تموت حين تُبرَّد.
\end{proof}

\begin{example}[الزبدة وزيت الزيتون وزيت السمك]\label{ex:b1:lipids:kitchen}
الزبدة (\qty{65}{\%} مشبعة) صلبة في البراد وطرية في حرارة الغرفة؛
\omterm{def:b1:lipids:triglyceride}{وزيت} الزيتون (\qty{75}{\%} \omterm{def:b1:water-small-molecules:ph}{حمض} أولييك) سائل في حرارة الغرفة ويتعكر
في البراد؛ \omterm{def:b1:lipids:triglyceride}{وزيت} السمك، الغني بحموض أوميغا-3 ذوات خمس وست روابط
مزدوجة، يبقى سائلًا عند \qty{-20}{\celsius}، وهو ما تحتاج إليه \omterm{def:b1:membranes-transport:membrane}{أغشية}
سمك القد في شمال الأطلسي.
\end{example}

\section{الستيرولات والأصبغة والإشارات}

\begin{definition}[الستيروئيدات والإيزوبرينويدات]\label{def:b1:lipids:steroids}
\emph{الستيروئيدات}\index{ستيروئيد} تشترك مع \omterm{def:b1:lipids:amphiphilic}{الكولستيرول} في حلقاته
الأربع الملتحمة: \omterm{def:b1:lipids:amphiphilic}{الكولستيرول} نفسه (في \omterm{def:b1:membranes-transport:membrane}{الأغشية}، وهو طليعة الباقي)،
والأملاح الصفراوية، وفيتامين D، والهرمونات الستيروئيدية ---
الكورتيزول، والألدوستيرون، والهرمونات الجنسية --- وهي جزيئات صغيرة
كارهة للماء تعبر \omterm{def:b1:membranes-transport:membrane}{الأغشية} بحرية وتعمل على مستقبِلات داخل \omterm{def:b1:cell-unit-of-life:cell}{الخلية}. أما
\emph{الإيزوبرينويدات}\index{إيزوبرينويد} (التربينات) فسلاسل وحلقات
من وحدات الإيزوبرين ذوات الخمس ذرات كربون: \emph{الكاروتينويدات}
التي تلوّن الجزر وتحمي \omterm{def:b1:eukaryotic-cell:plastid}{البلاستيدات الخضراء}
(\cref{ch:b1:photosynthesis})، والسلسلة الجانبية لليخضور، وكينونات
سلاسل نقل الإلكترون، والمطاط، والفيتامينات المنحلة في الدسم A وE
وK.
\end{definition}

\begin{proposition}[ماذا تفعل الدهون]\label{prop:b1:lipids:functions}
ادخار الطاقة (ثلاثيات الغليسيريد)؛ \omterm{def:b1:membranes-transport:membrane}{والأغشية} (الفوسفوليبيدات،
\omterm{def:b1:lipids:amphiphilic}{والسفينغوليبيدات}، \omterm{def:b1:lipids:amphiphilic}{والستيرولات})؛ والعزل وكتم الماء (\omterm{def:b1:lipids:triglyceride}{الشحم}، \omterm{def:b1:lipids:wax}{والشموع})؛
وامتصاص الضوء والحماية (الكاروتينويدات)؛ ونقل الإلكترون (الكينونات)؛
والإشارة بين \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} (الهرمونات الستيروئيدية، والبروستاغلاندينات)
وداخلها (الفوسفوليبيدات الإينوزيتولية، وثنائي أسيل الغليسرول)؛
والفيتامينات. وخاصية واحدة --- عدم الانحلال في الماء --- تقوم تحتها
كلها: \omterm{def:b1:lipids:fattyacid}{فالدهن} يبقى حيث يوضع، في قطيرة أو غلالة أو غشاء، أو يعبر غشاءً
بلا \omterm{def:b1:membranes-transport:transporters}{ناقل}.
\end{proposition}

\begin{example}[هرمون لا يحتاج إلى مستقبِل على السطح]\label{ex:b1:lipids:steroidhormone}
الكورتيزول، الذي تفرزه الغدة الكظرية، يسافر في الدم مرتبطًا ببروتين
حامل، ويفارقه عند \omterm{def:b1:cell-unit-of-life:cell}{الخلية} الهدف، وينحل عبر \omterm{def:b1:membranes-transport:membrane}{الغشاء البلازمي} في ثوان،
ويرتبط بمستقبِل في \omterm{def:b1:eukaryotic-cell:organelle}{العصارة الخلوية} يدخل بعد ذلك \omterm{def:b1:eukaryotic-cell:nucleus}{النواة} ويشغّل
مورّثات (\cref{ch:b1:expression-control}). أما هرمون بروتيني مثل
الأنسولين، وهو منحل في الماء وأكبر من أن يعبر، فلا بد أن يرتبط
بمستقبِل في الخارج وأن تُنقل رسالته إلى الداخل. فكيمياء الرسول تحسم
طريق الرسالة.
\end{example}

\section{مسألة: ماء بارد وشحم صحراوي}

\begin{problem}[{مسألة نهاية الأسبوع --- أغشية سلمون مرقط في جدول
جبلي وسنام جمل في الصحراء: عدم الإشباع، ودرجات الانتقال، وكثافة
الطاقة، والماء الاستقلابي، وصولًا إلى الكتلة التي يوفرها ادخار الشحم
بدل الغليكوجين}]
\label{pb:b1:lipids:1}
الجزء الأول عن سلمون مرقط تحمل فوسفوليبيدات أغشيته سلاسل C16:0
وC18:1 وC22:6 (وهو \omterm{def:b1:water-small-molecules:ph}{حمض} أوميغا-3 بست روابط مزدوجة). والجزء الثاني عن
جمل وحيد السنام كتلته \qty{500}{kg} بسنام من \omterm{def:b1:lipids:triglyceride}{ثلاثي الغليسيريد} كتلته
\qty{30}{kg}، يقطع صحراء بمعدل استقلاب قدره \qty{60}{MJ/d}. والطاقة:
\omterm{def:b1:lipids:triglyceride}{الشحم} \qty{38}{kJ/g}، والغليكوجين \qty{17}{kJ/g} جافًا، مخزونًا مع
\qty{3}{g} من الماء لكل غرام. والماء الاستقلابي: \qty{1.07}{g/g} من
\omterm{def:b1:lipids:triglyceride}{الشحم}. والأكسجين: \qty{2.0}{L} لكل غرام \omterm{def:b1:lipids:triglyceride}{شحم}؛ والهواء \qty{21}{\%}
أكسجينًا؛ والجمل يستخلص \qty{5}{\%} من الأكسجين الذي يتنفسه؛ والهواء
المزفور يحمل \qty{30}{g/m^3} من الماء أكثر مما يحمله هواء الصحراء
المستنشق.

\medskip
\noindent\textbf{الجزء الأول --- السلمون المرقط.}
السلمون المؤقلم عند \qty{20}{\celsius} سلاسل أغشيته \qty{40}{\%}
C16:0 و\qty{45}{\%} C18:1 و\qty{15}{\%} C22:6؛ والسلمون عند
\qty{5}{\celsius}: \qty{25}{\%} و\qty{40}{\%} و\qty{35}{\%}.
\begin{enumerate}
  \item اكتب ترميز الحموض الثلاثة وعُدّ الروابط المزدوجة في كل
        سلسلة منها.
  \item احسب متوسط عدد الروابط المزدوجة لكل سلسلة عند كل درجة حرارة.
  \item فسّر، انطلاقًا من ارتصاص السلاسل، لماذا يخفض التغيرُ درجةَ
        انتقال الغشاء.
  \item تهبط $T_m$ لطبقة مضاعفة نحو \qty{20}{\celsius} مقابل كل
        \qty{0.5}{} رابطة مزدوجة إضافية لكل سلسلة بالمتوسط. قدّر
        إزاحة $T_m$ بين السلمونين.
  \item غشاء السلمون الدافئ مائع عند \qty{20}{\celsius} لكنه كان
        ليتهلمن عند \qty{5}{\celsius}. فماذا يحل ببروتينات نقله
        ومضخاته وبنقله العصبي في ليلة باردة مفاجئة؟
  \item يستغرق التكيف أيامًا. سمِّ الإنزيمات التي على السلمون أن
        يصنع منها أكثر، وأين تعمل في \omterm{def:b1:cell-unit-of-life:cell}{الخلية} (\cref{ch:b1:eukaryotic-cell}).
  \item \omterm{def:b1:lipids:amphiphilic}{الكولستيرول} عند جزيء واحد لكل فوسفوليبيدين عند السلمونين
        كليهما. فسّر ما الذي يسهم به في كل حالة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- طاقة الجمل.}
\begin{enumerate}[resume]
  \item احسب الطاقة المخزونة في السنام.
  \item كم يومًا من العبور تكفي؟
  \item احسب كتلة الغليكوجين الجاف الذي يحمل الطاقة نفسها، وكتلة
        الغليكوجين المماه.
  \item احسب الكتلة الموفَّرة بادخار \omterm{def:b1:lipids:triglyceride}{الشحم} بدل الغليكوجين، وعبّر عنها
        بكسر من كتلة جسم الجمل.
  \item يُدخَّر \omterm{def:b1:lipids:triglyceride}{الشحم} في سنام لا منشورًا تحت الجلد. اقترح سببًا،
        انطلاقًا من \omterm{prop:b1:mammal-organization:heatbudget}{الميزانية الحرارية} في
        \cref{ch:b1:mammal-organization}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- هل السنام مخزن ماء؟}
\begin{enumerate}[resume]
  \item احسب الماء الاستقلابي الناتج من أكسدة السنام كله.
  \item احسب الأكسجين اللازم، باللتر.
  \item احسب حجم الهواء الذي على الجمل أن يهوّيه للحصول عليه.
  \item احسب الماء المفقود بزفير ذلك الهواء.
  \item قارن الماء المكتسب بالماء المفقود، واستنتج.
  \item بدل التعرق، يترك الجمل حرارة جسمه ترتفع \qty{6}{\celsius}
        نهارًا وتبرد ليلًا. احسب الحرارة التي يدخرها بهذه الطريقة
        (بسعة حرارية \qty{3.5}{kJ.kg^{-1}.K^{-1}}) والماء الذي كان
        تبخير الحرارة نفسها ليكلفه (\qty{2.4}{kJ/g}).
  \item يستطيع الجمل أن يفقد \qty{25}{\%} من ماء جسمه وينجو، وتنتفخ
        كرياته الحمراء إلى ضعف حجمها دون أن تنفجر حين يشرب
        \qty{100}{L} في عشر دقائق. اربط الخاصية الثانية \omterm{def:b1:membranes-transport:membrane}{بأغشية}
        \cref{ch:b1:membranes-transport}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- حساب الطبقة المضاعفة.}
يشغل رأس \omterm{def:b1:lipids:amphiphilic}{الفوسفوليبيد} \qty{0.65}{nm^2}؛ وسماكة الطبقة المضاعفة
\qty{5}{nm}؛ وللكريات الحمراء عند الجمل سطح قدره
\qty{80}{\micro m^2} وعددها $\num{8e12}$ في لتر الدم.
\begin{enumerate}[resume]
  \item احسب عدد الفوسفوليبيدات في غشاء كرية حمراء واحدة.
  \item احسب عددها في كل الكريات الحمراء في \qty{40}{L} من الدم.
  \item عند \qty{750}{g/mol}، احسب كتلة ذلك \omterm{def:b1:lipids:fattyacid}{الدهن} بالغرام.
  \item ينقلب \omterm{def:b1:lipids:amphiphilic}{الفوسفوليبيد} من صفيحة إلى أخرى تلقائيًا مرة في الأسبوع
        تقريبًا، لكنه ينتشر جانبيًا \qty{1}{\micro m} في ثانية.
        فسّر العددين انطلاقًا من \omterm{prop:b1:water-small-molecules:properties}{الأثر الكاره للماء}.
  \item فسّر لماذا تستطيع صفيحتا غشاء، بمعدل الانقلاب هذا، أن تبقيا
        مختلفتي التركيب طول حياة \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.
  \item اذكر النتيجة: الكتلة التي يوفرها الجمل بحمل \qty{30}{kg} من
        \omterm{def:b1:lipids:triglyceride}{الشحم} بدل الغليكوجين المكافئ لها طاقةً، والحكم على السنام
        مخزنًا للماء.
\end{enumerate}
\end{problem}
