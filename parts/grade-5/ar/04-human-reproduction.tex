\chapter{تكاثر الإنسان}\label{ch:g5:human-reproduction}

كل شخص تعرفه --- الأصدقاء، والمعلمون، وأكبر فرد في أسرتك --- وصل
بالطريقة نفسها: رضيعًا، وُلد بعد شهور من النمو داخل أمه. والبشر
ثدييات (\cref{prop:g4:classifying-animals:classes})، وتكاثر
الإنسان يجري على خطة \omterm{prop:g3:sorting-living-things:groups}{الثدييات} في
\cref{ch:g4:animal-reproduction} --- بأعداد نوعنا وتواريخه
وتفاصيله، وهي التي يعرضها هذا الفصل بوضوح.

\section{النسخة البشرية من الخطة}

\begin{proposition}[تكاثر الإنسان تكاثر ثدييات]\label{prop:g5:human-reproduction:plan}
\omterm{prop:g3:sorting-living-things:groups}{كالثدييات} كلها، يتكاثر البشر تكاثرًا جنسيًّا
(\cref{def:g4:animal-reproduction:sexual}):
\begin{itemize}
  \item جسم الأم ينتج \emph{\omterm{def:g4:animal-reproduction:sexual}{بويضات}}؛ وجسم الأب ينتج
        \emph{نطفًا}؛
  \item \omterm{def:g5:flower-to-fruit:steps}{والإخصاب} --- التحام \omterm{def:g4:animal-reproduction:sexual}{نطفة} واحدة \omterm{def:g4:animal-reproduction:sexual}{ببويضة} واحدة --- يحدث
        \emph{داخل جسم الأم}
        (\cref{prop:g4:animal-reproduction:where})؛
  \item \omterm{def:g4:animal-reproduction:sexual}{والبويضة} المخصَّبة --- وهي \omterm{def:g5:first-look-at-cells:cell}{خلية} واحدة مفردة
        (\cref{ex:g5:first-look-at-cells:one}) --- تستقر في
        \emph{رحم}\index{رحم} الأم وتنمو هناك؛
  \item والبشر \omterm{def:g4:animal-reproduction:ovivivi}{ولود}
        (\cref{def:g4:animal-reproduction:ovivivi}): فالرضيع
        \omterm{def:g2:animals-grow-up:born}{يولد حيًّا}، ويتغذى في البداية على \omterm{def:g2:animals-grow-up:born}{الحليب}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[التكاثر يحتاج إلى أجسام بالغة]\label{rem:g5:human-reproduction:adults}
جسم الطفل لا يستطيع أن يصنع أطفالًا: فأعضاء التكاثر لا تبدأ العمل
إلا عند \emph{\omterm{def:g3:stages-of-life:stages}{البلوغ}}، وهو تغيّر مرحلة \omterm{def:g3:stages-of-life:stages}{المراهقة} في
\cref{def:g3:stages-of-life:stages}، حين يعيد الجسم بناء نفسه من
جسم طفل إلى جسم بالغ. أما ما يغيّره \omterm{def:g3:stages-of-life:stages}{البلوغ} وكيف، فموضوع فصل كامل
في سنوات المرحلة الإعدادية من هذا الكتاب.
\end{remark}

\section{تسعة أشهر في الرحم}

\begin{definition}[الحمل]\label{def:g5:human-reproduction:pregnancy}
\emph{الحمل}\index{حمل} هو المدة التي يقضيها الرضيع المستقبلي
نموًّا في \omterm{prop:g5:human-reproduction:plan}{رحم} أمه: نحو \emph{تسعة أشهر} عند البشر. ويسمى الوليد
النامي أولًا \emph{مضغة}\index{مضغة}، ثم --- بعد أن تُرسم خطة
جسمه، بعد نحو شهرين --- \emph{جنينًا}\index{جنين}.
\end{definition}

\begin{proposition}[التغذي عبر الحبل]\label{prop:g5:human-reproduction:cord}
داخل \omterm{prop:g5:human-reproduction:plan}{الرحم} يطفو الوليد في جيب من السائل ولا يستطيع أن يأكل ولا أن
يتنفس. وكل شيء يأتيه جاهزًا من دم أمه: \omterm{def:g4:journey-of-food:digestion}{فالمغذيات} والأكسجين
(\cref{prop:g4:heart-and-blood:carries}) تعبر إلى دمه عبر
\emph{الحبل السرّي}\index{حبل سري} عند بطنه --- وفضلاته تعبر
راجعةً بالطريق نفسه، ليتولاها جسم الأم. وسُرّتك هي موضع اتصال
حبلك أنت.
\end{proposition}
\admitted

\begin{example}[جدول الأشهر التسعة]\label{ex:g5:human-reproduction:timetable}
الشهر الأول: عنقود من \omterm{def:g5:first-look-at-cells:cell}{الخلايا} المنقسمة، أصغر من حبة أرز، لكنه ينمو
بسرعة منذ الآن. والشهر الثاني: \omterm{def:g5:human-reproduction:pregnancy}{مضغة} فيها رسم رأس، \omterm{def:g4:heart-and-blood:heart}{وقلب} ينبض،
وبراعم \omterm{def:g1:my-body:parts}{للذراعين} \omterm{def:g1:my-body:parts}{والرجلين} --- وهي ما زالت أصغر من إبهام. والشهران
الثالث والرابع: \omterm{def:g5:human-reproduction:pregnancy}{جنين} مكتمل الخطة يبدأ التحرك؛ وتحسّ الأم بركلاته
قريبًا. والشهور من الخامس إلى الثامن: نموّ ونضج --- فيأتي السمع،
وأول كساء من الشعر. والشهر التاسع: الوليد، ورأسه إلى أسفل عادةً
ووزنه نحو \qty{3}{kg}، مستعدّ.
\end{example}

\begin{example}[الولادة]\label{ex:g5:human-reproduction:birth}
عند \omterm{def:g2:born-grow-die:cycle}{الولادة} تدفع \omterm{def:g3:bones-and-muscles:muscle}{عضلات} \omterm{prop:g5:human-reproduction:plan}{الرحم} القوية الوليد إلى العالم، ورأسه أولًا
في العادة. والصرخة الأولى تملأ الرئتين هواءً --- ولم تكونا قد
تنفستا قط (\cref{prop:g5:human-reproduction:cord}) --- ويُقطع
الحبل بلا ألم إذ لم تعد إليه حاجة، وتستمر خطة \omterm{prop:g3:sorting-living-things:groups}{الثدييات} في الخارج:
\omterm{def:g2:animals-grow-up:born}{حليب}، ودفء، وعناية (وسيرضى بذلك
\cref{ex:g4:animal-reproduction:whalecalf}).
\end{example}

\begin{remark}[التوائم]\label{rem:g5:human-reproduction:twins}
أحيانًا ينمو وليدان في \omterm{prop:g5:human-reproduction:plan}{الرحم} معًا. فإن انشطرت \omterm{def:g4:animal-reproduction:sexual}{بويضة} مخصَّبة واحدة
إلى مضغتين، كان التوأمان \emph{متطابقين} --- مبنيين من \omterm{def:g5:first-look-at-cells:cell}{الخلية}
الأولى نفسها، متشابهين كنسختين. وإن خُصِّبت بويضتان بنطفتين، لم
يكن التوأمان أشبه أحدهما بالآخر من إخوة عاديين --- فهما لم يتشاركا
إلا \omterm{prop:g5:human-reproduction:plan}{الرحم}. صنفان من التوائم، يفرّق بينهما مبدؤهما.
\end{remark}

\section{العناية قبل وبعد}

\begin{proposition}[الوليد يشارك أمه تموينها]\label{prop:g5:human-reproduction:care}
كل ما يبلغ الوليد يسافر في دم الأم --- ولهذا تُحسن الحامل غذاءها
(\cref{ch:g3:balanced-diet})، وتستريح، ويتابعها الأطباء أو
القابلات، وتتجنب ما قد يؤذي الوليد: فدخان التبغ والكحول يعبران في
الحبل أيضًا، \omterm{def:g5:human-reproduction:pregnancy}{والجنين} أهشّ بكثير من البالغ. فالعناية بالأم
\emph{هي} العناية بالوليد.
\end{proposition}
\admitted

\begin{example}[نوع شديد الاعتناء بصغاره]\label{ex:g5:human-reproduction:longcare}
حتى بين \omterm{prop:g3:sorting-living-things:groups}{الثدييات}، يدفع البشر خطة القليل المعتنى به في
\cref{prop:g4:animal-reproduction:strategies} إلى أقصاها: فوليد
واحد في المرة عادةً، عاجز مدة أطول بكثير من مهر أو هرّ صغير،
ومعتنًى به --- إطعامًا وتعليمًا ومحبةً --- \emph{سنوات} كثيرة.
\omterm{def:g3:stages-of-life:stages}{والطفولة} البشرية من أطول الطفولات في عالم \omterm{def:g1:animals-around-us:animal}{الحيوان}؛ وهذا الكتاب،
وهو مقرر يمتد عقدًا، قطعة من تلك العناية الطويلة.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:1}
أي \omterm{def:g5:first-look-at-cells:cell}{خلية} تأتي من الأم، وأيها من الأب، وما اسم التحامهما؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:2}
أين ينمو الوليد البشري، وكم يستغرق ذلك تقريبًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:3}
ما اسما الوليد النامي بالترتيب، ومتى يصير الأول الثاني تقريبًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:4}
الوليد في \omterm{prop:g5:human-reproduction:plan}{الرحم} لا يأكل ولا يتنفس. فكيف ينال \omterm{def:g4:journey-of-food:digestion}{المغذيات} والأكسجين،
وإلى أين تذهب فضلاته؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:5}
علام تدل سُرّتك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:6}
ماذا يحدث عند الصرخة الأولى، ولماذا كانت تغيّرًا كبيرًا لجسم
الوليد؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:7}
أي الصفات تجعل تكاثر الإنسان تكاثر ثدييات نموذجيًّا؟ سمِّ ثلاثًا،
مستشهدًا بما في \cref{prop:g4:classifying-animals:classes} أو
في \cref{ch:g4:animal-reproduction}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:8}
لماذا يطلب الأطباء من الحوامل تجنب دخان التبغ والكحول؟ استعمل
\cref{prop:g5:human-reproduction:care}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:9}
اشرح صنفي التوائم بمبدأ كل منهما. وأي صنف يتشابه ``كنسختين''،
ولماذا يلزم ذلك من \cref{ex:g5:first-look-at-cells:one}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:10}
ضع الإنسان على خط الخطط في
\cref{prop:g4:animal-reproduction:strategies}، مقارنًا بسمكة القدّ
والفيلة: عدد الصغار، وطول العناية، وما تحتويه تلك العناية الطويلة
عند البشر.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:human-reproduction:11}
عند \omterm{def:g2:born-grow-die:cycle}{الولادة} ينهي الحبل المقطوع خط تموين الوليد في اللحظة نفسها
التي تأخذ فيها الرئتان أول نفَس. استعمل
\cref{prop:g5:human-reproduction:cord}
و\cref{prop:g4:breathing:exchange} لتشرح أي عمل ينتقل في تلك
الدقيقة من جسم الأم إلى رئتي الوليد نفسه --- ولماذا كانت الصرخة
الأولى حسنة التوقيت.
\end{exercise}
