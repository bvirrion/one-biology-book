\chapter{Kendali Hormon atas Perkembangbiakan}\label{ch:g11:hormones-and-reproduction}

Seorang perempuan menyimpan sebuah bagan: suhunya tiap pagi, dengan
kenaikan kecil sekitar pertengahan tiap bulan; sebuah perdarahan tiap
dua puluh delapan hari atau lebih kurang. Sebuah laboratorium yang
mengukur empat \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} dalam darahnya tiap hari akan menggambar empat
lengkung yang naik dan turun dalam urutan yang tetap, bulan demi bulan,
dengan tiap puncaknya menyebabkan puncak berikutnya. Bagan laboratorium
seorang lelaki akan memperlihatkan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} yang sama pada aras yang
nyaris tetap, hari demi hari. Organnya berbeda; sistem kendalinya ---
sebuah pusat di otak, sebuah kelenjar di bawahnya, gonadnya, dan pesan
yang dibawa darah ke dua arah --- sama. Bab ini membaca bagannya, dan
menunjukkan bagaimana memahaminya memungkinkan orang mencegah sebuah
kehamilan, atau memperolehnya.

\section{Hormon dan gelungnya}

\begin{definition}[Hormon, umpan balik]\label{def:g11:hormones-and-reproduction:hormone}
Sebuah \emph{hormon}\index{hormon} adalah molekul yang dikeluarkan ke
dalam darah oleh sebuah kelenjar dan bekerja, dari jauh, pada \omterm{def:g10:cells-common-unit:cell}{sel} yang
membawa \emph{reseptor} baginya. Sistem hormon diatur oleh \emph{umpan
balik}\index{umpan balik}: hormon yang dihasilkan pada ujung sebuah
rantai bekerja balik pada kelenjar di pangkalnya. \emph{Umpan balik
negatif} --- hasilnya mengurangi perintahnya --- menjaga sebuah aras
tetap; \emph{umpan balik positif} --- hasilnya menambah perintahnya ---
menghasilkan sebuah lonjakan.
\end{definition}

\begin{proposition}[Rantai bersamanya]\label{prop:g11:hormones-and-reproduction:chain}
Pada kedua kelaminnya, rantai bertingkat tiga yang sama mengendalikan
gonadnya. Sebuah pusat otaknya, yaitu \emph{hipotalamus}, mengeluarkan
dalam denyut sebuah \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} pelepas, yaitu GnRH, ke dalam pembuluh kecil
yang menuju \emph{kelenjar hipofisis} tepat di bawahnya; hipofisisnya
menanggapi dengan mengeluarkan dua \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} ke dalam darah umum, yaitu
\emph{LH} dan \emph{FSH}; keduanya bekerja pada gonadnya, yang
menghasilkan gamet sekaligus \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} kelamin --- testosteron dalam
testis, estrogen dan progesteron dalam ovarium. \omterm{def:g11:hormones-and-reproduction:hormone}{Hormon} kelaminnya
bekerja pada tubuhnya, dan bekerja balik pada hipotalamus serta
hipofisisnya.
\end{proposition}
\admitted

\section{Lelaki: sebuah aras yang tetap}

\begin{proposition}[Kendali testisnya]\label{prop:g11:hormones-and-reproduction:male}
Testisnya punya dua ruang. \emph{Tubulus seminiferus}, yang dirangsang
FSH dan testosteron, menghasilkan sperma terus-menerus sejak
\omterm{prop:g11:becoming-male-female:puberty}{pubertasnya}, sekitar seratus juta tiap hari. \omterm{def:g10:cells-common-unit:cell}{Sel} di antara tubulusnya,
yang dirangsang LH, mengeluarkan \emph{testosteron}, yang menjaga
saluran perkembangbiakannya, ciri sekundernya dan penghasilan
spermanya. \omterm{prop:g11:becoming-male-female:hormones}{Testosteron} menghambat hipotalamus dan hipofisisnya: ketika
ia naik, GnRH, LH dan FSH turun dan testisnya mengeluarkan lebih
sedikit; ketika ia turun, ketiganya naik. \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan balik} negatif ini
menjaga arasnya nyaris tetap, dalam ayunan harian yang kecil, sejak
\omterm{prop:g11:becoming-male-female:puberty}{pubertas} sampai usia lanjut.
\end{proposition}
\begin{proof}[Bukti percobaan]
Pengebirian seekor hewan menaikkan LH-nya beberapa kali lipat dalam
beberapa hari; suntikan testosteron menurunkan LH-nya kembali. Luka
pada hipotalamusnya menghapus LH, FSH dan testosteronnya sekaligus;
menyuntikkan GnRH dalam denyut memulihkan ketiganya, sedangkan aliran
GnRH yang menerus tidak --- denyutnyalah yang menentukan. Lelaki yang
diobati dengan testosteron karena alasan lain melihat penghasilan
spermanya turun: \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} dari luar itu memadamkan hipofisisnya sendiri.
\end{proof}

\section{Perempuan: daurnya}

\begin{proposition}[Daur ovarium]\label{prop:g11:hormones-and-reproduction:ovary}
Sejak \omterm{prop:g11:becoming-male-female:puberty}{pubertas} sampai mati haid, ovariumnya menjalankan daur sekitar 28 hari.
\begin{itemize}
  \item \emph{Fase folikel} (hari 1--13): di bawah FSH, sebuah folikel
        --- \omterm{def:g10:cells-common-unit:cell}{sel} telur yang dikelilingi \omterm{def:g10:cells-common-unit:cell}{sel} perawatnya --- tumbuh dan
        mengeluarkan \emph{estrogen} dalam jumlah yang meningkat.
  \item \emph{Ovulasi} (hari ke-14): sebuah lonjakan LH memecahkan
        folikel yang matang, yang lalu melepaskan \omterm{def:g10:cells-common-unit:cell}{sel} telurnya ke
        dalam saluran telurnya.
  \item \emph{Fase luteal} (hari 15--28): folikel yang telah kosong
        menjadi \emph{korpus luteum}, yang mengeluarkan
        \emph{progesteron} dan estrogen; jika tidak ada kehamilan yang
        mulai, ia merosot setelah sekitar dua belas hari, kedua
        \omterm{def:g11:hormones-and-reproduction:hormone}{hormonnya} turun, dan daur baru pun mulai.
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[Daur rahim dan umpan baliknya]\label{prop:g11:hormones-and-reproduction:uterus}
Lapisan rahimnya mengikuti \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} ovariumnya: estrogen membuatnya
menebal selama fase folikelnya; progesteron membuatnya menghasilkan
getah dan siap menerima sebuah embrio selama fase lutealnya; turunnya
keduanya pada akhir daurnya membuatnya terurai dan berdarah ---
\emph{haid}, hari ke-1 sampai ke-5 daur berikutnya. \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan baliknya}
berganti tanda selama daurnya: pada aras yang rendah dan sedang,
estrogen dan progesteron menghambat hipotalamus dan hipofisisnya (\omterm{def:g11:hormones-and-reproduction:hormone}{umpan
balik} negatif); tetapi estrogen yang tinggi dari folikel yang matang,
yang bertahan dua hari, \emph{merangsang} keduanya (\omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik}
positif), yang lalu menghasilkan lonjakan LH yang memicu ovulasinya.
Setelah ovulasinya, progesteron korpus luteumnya memulihkan \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik}
negatifnya, dan tidak ada lonjakan kedua yang terjadi.
\end{proposition}
\begin{proof}[Bukti percobaan]
Pengukuran darah harian memberi lengkung pada gambar berikutnya:
lonjakan LH-nya menyusul puncak estrogennya satu hari kemudian dan
mendahului ovulasinya satu hari; progesteronnya muncul hanya setelah
ovulasinya. Menyingkirkan ovariumnya menaikkan LH dan FSH; dosis kecil
estrogen menurunkan keduanya; dosis besar yang bertahan yang diberikan
pada pertengahan daurnya memancing lonjakan LH --- yaitu tanda bahwa
\omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya} berbalik seiring dosisnya. Perempuan yang hipofisisnya
tidak dapat menanggapi kenaikan estrogennya tidak punya lonjakan itu
dan tidak berovulasi, walaupun folikelnya tumbuh.
\end{proof}

\begin{method}[Membaca bagan sebuah daur]\label{met:g11:hormones-and-reproduction:read}
\begin{enumerate}
  \item Carilah puncak LH-nya: ovulasinya terjadi keesokan harinya;
        fase folikelnya adalah segalanya sebelumnya, fase lutealnya
        segalanya sesudahnya.
  \item Periksa urutannya: puncak estrogen, lalu lonjakan LH, lalu
        ovulasi, lalu progesteron. Lengkung yang tidak berurutan adalah
        tanda sebuah gangguan.
  \item Hitunglah: fase lutealnya mendekati 14 hari pada tiap
        perempuan; panjang daurnya berubah lewat fase folikelnya.
        Karena itu ovulasinya jatuh 14 hari sebelum haid berikutnya,
        bukan 14 hari setelah haid yang terakhir.
  \item Tafsirkan sebuah obat atau luka dengan bertanya \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} mana
        yang ditambahkan atau disingkirkannya, dan apa yang lalu
        diperbuat \omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya} pada \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} yang lain.
\end{enumerate}
\end{method}

\section{Memilih: pencegahan kehamilan}

\begin{proposition}[Pencegahan kehamilan lewat hormon]\label{prop:g11:hormones-and-reproduction:contraception}
Pil gabungan memasok, tiap hari, dosis kecil sebuah estrogen dan sebuah
molekul yang menyerupai progesteron. Pada aras itu \omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya}
negatif sepanjang waktu: FSH-nya tetap rendah, tidak ada folikel yang
matang, tidak ada puncak estrogen, tidak ada lonjakan LH, tidak ada
ovulasi. Lapisan rahimnya, yang dijaga tetap tipis, dan lendir leher
rahimnya, yang dijaga tetap kental, menambahkan dua penghalang.
Berhenti minum pilnya selama sepekan membiarkan \omterm{def:g11:hormones-and-reproduction:hormone}{hormonnya} turun dan
lapisannya berdarah. Cara lain bekerja di tempat lain: alat tembaga di
dalam rahim mencegah pembuahan dan penanaman; pil darurat, yang diminum
dalam beberapa hari setelah bersanggama, menunda ovulasinya; kondom
menghentikan gametnya bertemu --- dan menjadi satu-satunya cara yang
juga menghentikan infeksi menular seksual.
\end{proposition}
\admitted

\begin{example}[Membaca pengaruh pilnya]\label{ex:g11:hormones-and-reproduction:pill}
Pada bagan seorang perempuan yang minum pilnya, keempat lengkungnya
mendatar: estrogen dan progesteronnya pada aras rendah yang mantap yang
dipasok pilnya, LH dan FSH-nya di bawah minimum daurnya, tanpa puncak
di mana pun. Lewatkan pilnya selama beberapa hari pada awal
kemasannya dan FSH-nya lolos: sebuah folikel dapat mulai, dan
selebihnya dapat menyusul. Pilnya berhasil karena \omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya}
bekerja, dan satu pil yang terlupakan adalah lubang dalam \omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya}.
\end{example}

\section{Memperoleh: perkembangbiakan berbantu}

\begin{proposition}[Ketika kehamilannya tidak datang]\label{prop:g11:hormones-and-reproduction:art}
Sekitar satu pasangan dari tujuh berkonsultasi setelah setahun mencoba.
Sebabnya terbagi: ovulasi yang tidak ada atau tidak teratur, saluran
telur yang tersumbat, sperma yang terlalu sedikit atau tidak bergerak,
dan pada sepertiga kasusnya tanpa sebab yang dikenali. Pengobatannya
mengikuti biologinya:
\begin{itemize}
  \item \emph{pengimbasan ovulasi}: suntikan FSH, atau obat yang
        menghalangi \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} negatif estrogennya, membuat folikelnya
        matang; sebuah suntikan yang menyerupai LH lalu memicu
        ovulasinya pada jam yang terpilih;
  \item \emph{inseminasi}: sperma yang telah disiapkan ditaruh dalam
        rahimnya pada saat ovulasinya;
  \item \emph{pembuahan di luar tubuh}: setelah perangsangannya, \omterm{def:g10:cells-common-unit:cell}{sel}
        telurnya dikumpulkan dari ovariumnya dengan sebuah jarum,
        dibuahi dalam sebuah cawan --- bila perlu dengan menyuntikkan
        satu sperma ke dalam tiap \omterm{def:g10:cells-common-unit:cell}{sel} telurnya --- dan satu atau dua
        embrio ditaruh dalam rahimnya tiga sampai lima hari kemudian.
        Sekitar satu upaya dari tiga atau empat berujung pada kelahiran;
        selebihnya dapat diulang dengan embrio yang dibekukan.
\end{itemize}
Tiap langkah rantai alaminya yang gagal digantikan campur tangan yang
sepadan.
\end{proposition}
\admitted

\begin{example}[Angka satu upaya]\label{ex:g11:hormones-and-reproduction:ivf}
Perangsangannya menghasilkan 10 \omterm{def:g10:cells-common-unit:cell}{sel} telur; 8 matang; 5 terbuahi; 3
mencapai tahap lima hari; 1 dipindahkan dan 2 dibekukan.
Pemindahannya berhasil pada sekitar sepertiga kasusnya; dengan kedua
embrio yang dibekukan, peluang sebuah kelahiran dari satu pengumpulannya
naik menjadi sekitar 60\%. Tekniknya telah menghasilkan beberapa juta
anak sejak 1978; ia juga menimbulkan pertanyaan --- tentang embrio yang
tidak terpakai, tentang umur orang tuanya, tentang apa yang boleh
dipilih --- yang tidak dijawab biologinya.
\end{example}

\begin{remark}[Dua sistem yang bertemu]\label{rem:g11:hormones-and-reproduction:brain}
\omterm{def:g11:hormones-and-reproduction:hormone}{Hormon} bab ini mengendalikan gonad dan tubuhnya; pada manusia \omterm{def:g11:hormones-and-reproduction:hormone}{hormon}
itu tidak mengendalikan perilaku seksualnya sebagaimana pada banyak
hewan, yang perkawinannya terikat pada daur ovariumnya. Kegiatan
seksual manusia sebagian besar tidak bergantung pada daurnya dan
bergantung pada sistem hasrat, kenikmatan dan kelekatan pada otaknya
--- yaitu untai ganjaran yang dibagi bersama dorongan lainnya --- serta
pada riwayat, budaya dan pilihan orangnya. Perkembangbiakan dan
seksualitasnya bertumpang tindih; keduanya bukan hal yang sama, dan
\omterm{def:g11:hormones-and-reproduction:hormone}{hormonnya} hanya mengatur yang pertama.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g11:hormones-and-reproduction:1}
Rumuskan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} dan \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik}, dan bedakan \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} negatif dari
\omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} positif.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:hormones-and-reproduction:2}
Sebutkan ketiga tingkat rantai yang mengendalikan gonadnya dan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon}
yang dikeluarkan masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:hormones-and-reproduction:3}
Apa kedua ruang testisnya dan apa yang dihasilkan masing-masing?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:hormones-and-reproduction:4}
Sebutkan fase daur ovariumnya beserta \omterm{def:g11:hormones-and-reproduction:hormone}{hormonnya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:hormones-and-reproduction:5}
Apa yang menyebabkan haid?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:hormones-and-reproduction:6}
Dari gambar daurnya, sebutkan hari puncak estrogennya, hari puncak
LH-nya, hari ovulasinya dan hari puncak progesteronnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:hormones-and-reproduction:7}
Jelaskan mengapa pengebirian menaikkan LH, dan mengapa suntikan
testosteron menurunkannya lagi.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:hormones-and-reproduction:8}
Jelaskan mengapa puncak estrogen pada hari ke-12 menghasilkan lonjakan
LH sedangkan estrogen pada hari ke-20 tidak.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:hormones-and-reproduction:9}
Seorang perempuan punya daur 35 hari. Pada hari keberapa ia
berovulasi? Berikan alasannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:hormones-and-reproduction:10}
Jelaskan bagaimana pil gabungan mencegah ovulasi, dengan memakai tanda
\omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:hormones-and-reproduction:11}
Seorang lelaki yang minum testosteron untuk binaraga mendapati
testisnya mengerut dan cacah spermanya turun. Jelaskan.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:hormones-and-reproduction:12}
Sebuah obat menghalangi reseptor estrogen pada hipotalamusnya.
Ramalkan pengaruhnya pada FSH, pada pertumbuhan folikelnya dan pada
ovulasinya, lalu sebutkan mengapa obat itu dipakai untuk mengobati
sebagian kemandulan.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:hormones-and-reproduction:13}
Bagan seorang perempuan memperlihatkan puncak estrogen tetapi tanpa
lonjakan LH dan tanpa progesteron. Tentukan letak cacatnya dalam

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:hormones-and-reproduction:14}
Selama kehamilan, jaringan embrionya mengeluarkan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} yang
menyerupai LH yang menjaga korpus luteumnya tetap hidup. Jelaskan
mengapa hal ini mencegah haid dan ovulasi berikutnya, dan mengapa uji
kehamilan mengenali \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} ini.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:hormones-and-reproduction:15}
Sebuah daur pembuahan di luar tubuh mengumpulkan 12 \omterm{def:g10:cells-common-unit:cell}{sel} telur, yang

\section{Soal: Bagannya}

\begin{problem}[{Soal akhir pekan --- daur seorang perempuan yang
dibagankan hormon demi hormon, dibaca umpan baliknya, diganggu sebuah
pil, dan dibantu ketika gagal}]
\label{pb:g11:hormones-and-reproduction:1}
Cuplikan darah diambil tiap hari sepanjang satu daur. Nilai yang
dibulatkan, sebagai persentase maksimum tiap \omterm{def:g11:hormones-and-reproduction:hormone}{hormonnya}:

\begin{center}
\begin{tabular}{lccccccccc}
hari & 2 & 6 & 10 & 12 & 13 & 14 & 18 & 22 & 27 \\ \hline
estrogen & 10 & 20 & 55 & 100 & 80 & 45 & 45 & 60 & 15 \\
LH & 15 & 15 & 20 & 35 & 100 & 60 & 15 & 15 & 12 \\
FSH & 45 & 35 & 30 & 35 & 70 & 45 & 22 & 25 & 25 \\
progesteron & 3 & 3 & 3 & 3 & 4 & 5 & 55 & 100 & 15 \\
\end{tabular}
\end{center}

\medskip
\noindent\textbf{Bagian I --- Membaca bagannya.}
\begin{enumerate}
  \item Pada hari keberapa ovulasinya berlangsung? Berikan alasannya
        dari dua \omterm{def:g11:hormones-and-reproduction:hormone}{hormon}.
  \item Sebutkan fase folikel dan fase lutealnya, beserta panjangnya.
  \item Struktur ovarium mana yang mengeluarkan estrogen hari ke-10?
        Hari ke-22? Mana yang mengeluarkan progesteron hari ke-22?
  \item Lukiskan keadaan lapisan rahimnya pada hari ke-6, ke-22 dan ke-28.
  \item Mengapa FSH-nya tertinggi pada awal daurnya dan terendah pada
        fase lutealnya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan baliknya}.}
\begin{enumerate}[resume]
  \item Antara hari ke-2 dan ke-10, estrogennya naik sedangkan FSH-nya
        turun. \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan balik} mana yang bekerja? Jelaskan.
  \item Antara hari ke-12 dan ke-13, estrogennya berada pada puncaknya
        dan LH-nya naik tujuh kali lipat. \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan balik} mana? Syarat apa
        pada aras estrogennya yang memungkinkannya?
  \item Antara hari ke-18 dan ke-27, estrogennya sedang, progesteronnya
        tinggi, dan LH-nya rendah. \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan balik} mana, dan mengapa tidak
        ada lonjakan kedua?
  \item Korpus luteumnya merosot pada hari ke-26. Ramalkan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} hari
        ke-28 dan apa yang menyusulnya.
  \item Jelaskan mengapa daurnya memulai dirinya sendiri: kenaikan apa
        pada hari ke-1 yang meluncurkan folikel berikutnya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Pilnya.}
Perempuan yang sama minum pil gabungan selama sebulan, yang memasok
estrogen pada 25\% dan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} yang menyerupai progesteron pada 40\%
maksimumnya, tiap hari.
\begin{enumerate}[resume]
  \item Ramalkan aras FSH dan LH-nya sepanjang bulan itu, dan nasib
        folikelnya.
  \item Akankah ada puncak estrogen? Lonjakan LH? Ovulasi?
  \item Selama tujuh hari tanpa pilnya, apa yang terjadi pada lapisan
        rahimnya, dan mengapa ini bukan haid yang sebenarnya?
  \item Ia melupakan pilnya pada hari ke-5 sampai ke-8. Jelaskan
        risikonya, langkah demi langkah.
  \item Jelaskan mengapa pil berisi progesteron saja, pada aras yang
        menjaga \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} negatifnya, juga dapat mencegah ovulasi.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Ketika rantainya gagal.}
\begin{enumerate}[resume]
  \item Bagan perempuan lain memperlihatkan FSH dan LH yang selalu di
        bawah 10\% dan estrogen di bawah 10\%. Di mana cacatnya?
        Pengobatan apa yang akan memulihkan ovulasinya, dan lewat jalan apa?
  \item Bagan perempuan ketiga normal tetapi saluran telurnya
        tersumbat. Langkah mana pada
        \cref{prop:g11:hormones-and-reproduction:art} yang menggantikan
        langkah yang tersumbat itu?
  \item Dalam pembuahan di luar tubuh, satu suntikan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} yang
        menyerupai LH diberikan 36 jam sebelum \omterm{def:g10:cells-common-unit:cell}{sel} telurnya
        dikumpulkan. Jelaskan apa yang ditirunya dan mengapa saatnya tepat.
  \item Setelah perangsangannya, 10 folikel matang alih-alih 1.
        Jelaskan, dengan FSH, mengapa daur alaminya lazimnya hanya
        mematangkan satu.
  \item Nyatakan hasilnya: hari ovulasi pada bagannya, kedua tanda
        \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} yang menghasilkannya, dan mekanisme yang dengannya
        pilnya menekannya.
\end{enumerate}
\end{problem}
