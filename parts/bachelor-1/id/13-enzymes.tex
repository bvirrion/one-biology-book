\chapter{Enzim dan Katalisis Biokimia}\label{ch:b1:enzymes}

Tuangkan hidrogen peroksida pada sebutir kentang yang terpotong dan ia
berbuih: satu molekul katalase dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} kentang itu membelah empat
puluh juta molekul peroksida tiap sekon, sebuah reaksi yang, bila
dibiarkan sendiri, akan memakan waktu bertahun-tahun. Peroksida itu
memang akan terurai --- reaksinya menurun --- tetapi tidak dalam waktu
yang berguna. Tiap reaksi \omterm{def:b1:cell-unit-of-life:cell}{sel} seperti ini: termodinamika menyatakan ke
arah mana ia dapat berjalan, dan sebuah \emph{\omterm{def:b1:enzymes:enzyme}{enzim}} menyatakan apakah
ia berjalan sekarang. Bab ini memerikan apa yang dilakukan sebuah \omterm{def:b1:enzymes:enzyme}{enzim}
pada sebuah reaksi, kinetika yang dipakai untuk mengukur dan
membandingkan \omterm{def:b1:enzymes:enzyme}{enzim}, cara \omterm{def:b1:enzymes:enzyme}{enzim} dihambat, dan cara \omterm{def:b1:cell-unit-of-life:cell}{sel} menyalakan serta
memadamkannya.

\section{Katalisis}

\begin{definition}[Enzim, substrat, sisi aktif]\label{def:b1:enzymes:enzyme}
Yang disebut \emph{enzim}\index{enzim} adalah sebuah \omterm{def:b1:proteins:peptide}{protein} (jarang
sebuah \omterm{def:b1:nucleic-acids:chain}{RNA}) yang \emph{mengatalisis} sebuah reaksi: ia menaikkan
lajunya tanpa habis terpakai dan tanpa mengubah kesetimbangannya.
Molekul yang dikenainya adalah \emph{substrat}\index{substrat}-nya;
molekul itu mengikat pada \emph{sisi aktif}\index{sisi aktif}, yaitu
sebuah celah yang dilapisi beberapa residu, tempat kimianya
berlangsung. Enzim bersifat \emph{khas} --- bagi satu substrat atau satu
keluarga, bagi satu ikatan, bagi satu stereoisomer --- dan dinamai
menurut substrat serta reaksinya dengan akhiran \emph{-ase} (laktase,
\omterm{def:b1:nucleic-acids:chain}{DNA} polimerase, suksinat dehidrogenase), dalam enam golongan:
oksidoreduktase, transferase, hidrolase, liase, isomerase, ligase.
Banyak di antaranya memerlukan sebuah \emph{kofaktor}\index{kofaktor}
bukan \omterm{def:b1:proteins:peptide}{protein}: sebuah ion logam (Zn, Mg, Fe), atau sebuah
\emph{koenzim}\index{koenzim} organik --- $\mathrm{NAD^+}$, FAD,
koenzim A, yang sebagian besarnya dibuat dari vitamin, dan itulah guna
vitamin.
\end{definition}

\begin{proposition}[Apa yang diubah sebuah enzim dan apa yang tidak]\label{prop:b1:enzymes:whatitdoes}
\index{energi pengaktifan}\index{keadaan transisi}
Sebuah reaksi melewati sebuah \emph{\omterm{prop:b1:enzymes:whatitdoes}{keadaan transisi}}, yaitu susunan
atom yang tegang dan berenergi bebas lebih tinggi daripada pereaksi
maupun hasilnya sebesar \emph{\omterm{prop:b1:enzymes:whatitdoes}{energi pengaktifan}} $E_a$; hanya molekul
yang dibawa guncangan termal melewati penghalang ini yang bereaksi,
dalam fraksi yang sebanding dengan $e^{-E_a/RT}$. Sebuah \omterm{def:b1:enzymes:enzyme}{enzim} mengikat
\omterm{prop:b1:enzymes:whatitdoes}{keadaan transisi} itu lebih erat daripada \omterm{def:b1:enzymes:enzyme}{substratnya} --- \omterm{def:b1:enzymes:enzyme}{sisi aktifnya}
melengkapi bentuk yang tegang itu --- sehingga menurunkan $E_a$:
menurunkannya sebesar \qty{34}{kJ/mol} (senilai beberapa \omterm{def:b1:water-small-molecules:water}{ikatan
hidrogen}) pada \qty{37}{\celsius} melipatgandakan lajunya
$e^{34\,000/2580} \approx
10^6$ kali. Ia tidak mengubah $\Delta G$ maupun tetapan
kesetimbangannya, yang hanya bergantung pada pereaksi dan hasilnya:
sebuah \omterm{def:b1:enzymes:enzyme}{enzim} mempercepat reaksi maju dan mundurnya sama besar lalu
membawa sistemnya ke kesetimbangan yang sama lebih cepat.
\end{proposition}

\begin{proposition}[Cara kerja sisi aktifnya]\label{prop:b1:enzymes:mechanisms}
Sebuah \omterm{def:b1:enzymes:enzyme}{sisi aktif} mempercepat reaksinya lewat beberapa cara sekaligus:
ia \emph{mengikat dan mengarahkan} \omterm{def:b1:enzymes:enzyme}{substratnya} sehingga gugus yang
bereaksi bertemu dalam geometri yang tepat, sehingga tumbukan yang
langka digantikan tumbukan yang pasti; ia \emph{menegangkan} \omterm{def:b1:enzymes:enzyme}{substratnya}
ke arah \omterm{prop:b1:enzymes:whatitdoes}{keadaan transisi} (\emph{kecocokan terinduksi}\index{kecocokan terinduksi}
: \omterm{def:b1:enzymes:enzyme}{enzimnya} mengatup mengelilingi \omterm{def:b1:enzymes:enzyme}{substratnya}, dan
kecocokannya terbaik bagi \omterm{prop:b1:enzymes:whatitdoes}{keadaan transisinya}); \omterm{def:b1:proteins:aminoacid}{rantai sampingnya}
berlaku sebagai \emph{asam dan basa}, yang menyerahkan proton ke dan
dari \omterm{def:b1:enzymes:enzyme}{substratnya} pada saat yang tepat (histidina, dengan p$K_a$-nya
dekat 7, melakukan hal ini pada banyak \omterm{def:b1:enzymes:enzyme}{enzim}); sebagian membentuk
\emph{\omterm{def:b1:water-small-molecules:bonds}{ikatan kovalen}} sementara dengan \omterm{def:b1:enzymes:enzyme}{substratnya} (protease serina);
dan ion logamnya mengutubkan ikatan serta memantapkan muatan. Air
sebagian besar dikucilkan dari sisinya, sehingga muatan dan \omterm{def:b1:water-small-molecules:water}{ikatan
hidrogennya} bekerja pada kekuatan penuh.
\end{proposition}

\begin{example}[Tiga enzim]\label{ex:b1:enzymes:three}
Lisozim, dalam air mata dan putih telur, menegangkan sebuah cincin gula
dinding bakteri menjadi bentuk \omterm{prop:b1:enzymes:whatitdoes}{keadaan transisinya} lalu memotong
rantainya: percepatan $10^8$ kali. Karbonat anhidrase, dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} merah,
memegang sebuah ion seng yang mengubah air menjadi hidroksida yang siap
menyerang $\mathrm{CO_2}$: sejuta molekul tiap sekon, \omterm{def:b1:enzymes:enzyme}{enzim} tercepat
sesudah katalase. Kimotripsin, dalam getah pankreas, memakai sebuah
serina yang dibuat reaktif oleh sebuah histidina untuk memotong \omterm{def:b1:proteins:peptide}{protein}
sesudah residu aromatik: seratus tiap sekon, dengan sebuah zat antara
kovalen.
\end{example}

\section{Kinetika enzim}

\begin{definition}[Laju awal, penjenuhan]\label{def:b1:enzymes:rate}
Yang disebut \emph{laju}\index{laju reaksi} $v$ sebuah reaksi \omterm{def:b1:enzymes:enzyme}{enzim}
adalah banyaknya hasil yang terbentuk tiap satuan waktu (\unit{mol/s},
atau satuan \omterm{def:b1:enzymes:enzyme}{enzim}: \qty{1}{\micro mol/min}). Ia diukur sebagai
\emph{laju awal} $v_0$, sebelum \omterm{def:b1:enzymes:enzyme}{substratnya} terkuras atau hasilnya
menumpuk. Pada kepekatan \omterm{def:b1:enzymes:enzyme}{enzim} yang tetap $v_0$ naik bersama kepekatan
\omterm{def:b1:enzymes:enzyme}{substrat} $[S]$ lalu \emph{menjenuh} pada sebuah nilai maksimum
$V_{\max}$: \omterm{def:b1:enzymes:enzyme}{enzimnya} terisi penuh dan tak sanggup lebih cepat lagi.
\end{definition}

\begin{theorem}[Michaelis--Menten]\label{thm:b1:enzymes:mm}
\index{persamaan Michaelis--Menten}\index{tetapan Michaelis}
Bagi sebuah \omterm{def:b1:enzymes:enzyme}{enzim} $E$ yang mengikat \omterm{def:b1:enzymes:enzyme}{substratnya} secara berbalikan lalu
mengubahnya,
\[
E + S \underset{k_{-1}}{\overset{k_1}{\rightleftharpoons}} ES
\overset{k_{\text{cat}}}{\longrightarrow} E + P ,
\]
laju awalnya pada kepekatan \omterm{def:b1:enzymes:enzyme}{substrat} $[S]$ adalah
\[
v_0 = \frac{V_{\max}\,[S]}{K_m + [S]}, \qquad V_{\max} = k_{\text{cat}}\,[E]_{\text{tot}},
\qquad K_m = \frac{k_{-1} + k_{\text{cat}}}{k_1} .
\]
$K_m$, \emph{\omterm{thm:b1:enzymes:mm}{tetapan Michaelis}}, adalah kepekatan \omterm{def:b1:enzymes:enzyme}{substrat} ketika
lajunya setengah nilai maksimumnya, dan mengukur berapa banyak \omterm{def:b1:enzymes:enzyme}{substrat}
yang dibutuhkan \omterm{def:b1:enzymes:enzyme}{enzimnya}; $k_{\text{cat}}$, \emph{bilangan
putaran}\index{bilangan putaran}, adalah banyaknya molekul \omterm{def:b1:enzymes:enzyme}{substrat} yang
diubah satu \omterm{def:b1:enzymes:enzyme}{enzim} tiap sekon ketika \omterm{def:b1:lipids:fattyacid}{jenuh}; rasio $k_{\text{cat}}/K_m$
adalah \emph{efisiensi katalitiknya}, yaitu tetapan laju pada \omterm{def:b1:enzymes:enzyme}{substrat}
yang rendah, yang dibatasi di atas oleh laju \omterm{def:b1:enzymes:enzyme}{substrat} dapat mencapai
\omterm{def:b1:enzymes:enzyme}{enzimnya} lewat difusi, sekitar
$10^8$--$10^9$ \unit{L.mol^{-1}.s^{-1}}.
\end{theorem}
\begin{proof}
Andaikan sebuah \emph{keadaan tunak} yang $ES$-nya terbentuk secepat ia
lenyap (berlaku sesudah beberapa milisekon pertama, selama
$[S] \gg [E]$): $k_1[E][S] = (k_{-1} + k_{\text{cat}})[ES]$, sehingga
$[E][S] = K_m[ES]$ dengan $K_m$ seperti yang ditetapkan tadi.
Kekekalannya memberikan $[E] = [E]_{\text{tot}} -
[ES]$; dengan mensubstitusikannya, $([E]_{\text{tot}} - [ES])[S] = K_m[ES]$,
sehingga $[ES] = [E]_{\text{tot}}[S]/(K_m + [S])$. Lajunya adalah
$v_0 = k_{\text{cat}}[ES]$, yang memberikan rumus tadi; pada
$[S] \gg K_m$, $v_0 \to k_{\text{cat}}[E]_{\text{tot}} = V_{\max}$, dan
pada $[S] = K_m$,
$v_0 = V_{\max}/2$.
\end{proof}

\begin{method}[Mengukur $K_m$ dan $V_{\max}$]\label{met:b1:enzymes:lb}
\begin{enumerate}
  \item Siapkan sederet kepekatan \omterm{def:b1:enzymes:enzyme}{substrat} yang membentang dari
        $K_m/4$ sampai $10K_m$, dengan kepekatan \omterm{def:b1:enzymes:enzyme}{enzim} yang sama;
        ikutilah hasilnya (warna, serapan, gas) selama menit pertamanya
        lalu ambil kemiringannya sebagai $v_0$.
  \item Plot $v_0$ terhadap $[S]$: hiperbolanya memberikan $V_{\max}$
        secara kasar (dataran puncaknya) dan $K_m$ (kepekatan pada
        setengahnya).
  \item Untuk ketelitiannya, baliklah: $1/v_0 = (K_m/V_{\max})(1/[S]) +
        1/V_{\max}$ (\emph{Lineweaver--Burk}). Perpotongan garisnya
        pada sumbu $y$ adalah $1/V_{\max}$, pada sumbu $x$ adalah
        $-1/K_m$, dan kemiringannya $K_m/V_{\max}$.
  \item Bagilah $V_{\max}$ dengan kepekatan \omterm{def:b1:enzymes:enzyme}{enzimnya} untuk memperoleh
        $k_{\text{cat}}$; bagilah dengan $K_m$ untuk efisiensinya.
        Ulangi dengan sebuah \omterm{def:b1:enzymes:inhibitors}{penghambat} untuk melihat parameter mana
        yang diubahnya.
\end{enumerate}
\end{method}

\section{Penghambatan}

\begin{definition}[Penghambat]\label{def:b1:enzymes:inhibitors}
Sebuah \emph{penghambat}\index{penghambat enzim} menurunkan laju sebuah
\omterm{def:b1:enzymes:enzyme}{enzim}. Penghambat \emph{tak berbalik} mengikat secara kovalen dan
memusnahkan keaktifan \omterm{def:b1:enzymes:enzyme}{enzimnya} untuk selamanya (zat saraf pada
asetilkolinesterase, penisilin pada \omterm{def:b1:enzymes:enzyme}{enzim} yang menyilang ikat dinding
bakteri, aspirin pada \omterm{def:b1:enzymes:enzyme}{enzim} yang membuat prostaglandin). Penghambat
\emph{berbalik} mengikat lewat ikatan lemah dan dapat dibilas pergi;
menurut tempatnya mengikat:
\begin{itemize}
  \item \emph{bersaing}\index{penghambatan bersaing}: penghambatnya
        mirip \omterm{def:b1:enzymes:enzyme}{substratnya} dan menempati \omterm{def:b1:enzymes:enzyme}{sisi aktifnya}; ia menaikkan
        $K_m$ tampaknya (sebesar faktor $1 + [I]/K_i$) dan membiarkan
        $V_{\max}$ tak berubah --- \omterm{def:b1:enzymes:enzyme}{substrat} yang cukup banyak
        menyingkirkannya;
  \item \emph{tak bersaing}: penghambatnya mengikat di tempat lain dan
        merusak katalisisnya baik \omterm{def:b1:enzymes:enzyme}{substratnya} terikat maupun tidak; ia
        menurunkan $V_{\max}$ dan membiarkan $K_m$ tak berubah ---
        berapa pun banyak \omterm{def:b1:enzymes:enzyme}{substratnya} tidak mengatasinya;
  \item \emph{antibersaing}: penghambatnya hanya mengikat kompleks
        $ES$; $V_{\max}$ maupun $K_m$ sama-sama turun.
\end{itemize}
$K_i$, tetapan penguraian penghambatnya, mengukur kekuatannya: makin
rendah, makin kuat.
\end{definition}

\begin{example}[Persaingan sebagai obat]\label{ex:b1:enzymes:medicine}
Metanol tak berbahaya sampai alkohol dehidrogenase hati mengubahnya
menjadi formaldehida dan asam format, yang membutakan dan membunuh.
Pengobatannya adalah etanol: sebuah \omterm{def:b1:enzymes:enzyme}{substrat} penyaing dengan $K_m$
lebih rendah, yang menyibukkan \omterm{def:b1:enzymes:enzyme}{enzimnya} sementara metanolnya
diekskresikan tanpa berubah. Statin adalah \omterm{def:b1:enzymes:inhibitors}{penghambat} bersaing, yang
mirip \omterm{prop:b1:enzymes:whatitdoes}{keadaan transisi}, bagi \omterm{def:b1:enzymes:enzyme}{enzim} yang menyerahkan karbon ke
\omterm{def:b1:lipids:amphiphilic}{kolesterol}; sulfonamida mirip \omterm{def:b1:enzymes:enzyme}{substrat} sebuah \omterm{def:b1:enzymes:enzyme}{enzim} bakteri yang
membuat folat, yang tidak dibuat manusia sehingga tidak dirindukannya.
\end{example}

\section{Pengaturan}

\begin{definition}[Enzim alosterik]\label{def:b1:enzymes:allosteric}
Sebuah \emph{enzim alosterik}\index{enzim alosterik} memiliki beberapa
subunit dan, di samping \omterm{def:b1:enzymes:enzyme}{sisi aktifnya}, \emph{sisi pengatur} tempat
efektor mengikat: \emph{pengaktif} menggesernya ke arah konformasi
aktifnya, \emph{\omterm{def:b1:enzymes:inhibitors}{penghambat}} ke arah yang tidak aktif. Lajunya terhadap
$[S]$ berbentuk sigmoid, bukan hiperbolik (\omterm{def:b1:proteins:allostery}{kerja sama} di antara \omterm{def:b1:enzymes:enzyme}{sisi
aktifnya}, seperti pada hemoglobin, \cref{ch:b1:proteins}), sehingga
perubahan kecil \omterm{def:b1:enzymes:enzyme}{substrat} dekat bagian curamnya mengubah lajunya besar
sekali; efektornya menggeser kurvanya ke samping (mengubah kepekatan
\omterm{def:b1:enzymes:enzyme}{substrat} yang dibutuhkan) atau naik turun (mengubah laju maksimumnya).
\omterm{def:b1:enzymes:enzyme}{Enzim} \omterm{def:b1:proteins:allostery}{alosterik} berdiri pada titik cabang metabolisme, dan efektornya
adalah hasil lintasan itu sendiri serta isyarat energi \omterm{def:b1:cell-unit-of-life:cell}{sel} (\omterm{def:b1:water-small-molecules:atp}{ATP}, ADP,
AMP).
\end{definition}

\begin{proposition}[Empat cara sel mengendalikan sebuah enzim]\label{prop:b1:enzymes:control}
\begin{enumerate}
  \item \emph{Penghambatan umpan balik}\index{penghambatan umpan balik}:
        hasil akhir sebuah lintasan menjadi \omterm{def:b1:enzymes:inhibitors}{penghambat} \omterm{def:b1:proteins:allostery}{alosterik} bagi
        \omterm{def:b1:enzymes:enzyme}{enzim} pertamanya yang menentukan, sehingga lintasannya berjalan
        hanya secepat hasilnya dipakai (isoleusina pada treonina
        deaminase; \omterm{def:b1:water-small-molecules:atp}{ATP} pada fosfofruktokinase,
        \cref{ch:b1:respiration-fermentation}). Waktu tanggapnya:
        milisekon.
  \item \emph{Perubahan kovalen}\index{perubahan kovalen}: sebuah
        kinase memautkan sebuah fosfat pada sebuah serina, treonina
        atau tirosina \omterm{def:b1:enzymes:enzyme}{enzimnya}, sebuah fosfatase menyingkirkannya, dan
        kedua bentuknya berbeda keaktifannya (\omterm{def:b1:carbohydrates:polysaccharide}{glikogen} fosforilase
        dinyalakan lewat fosforilasi, \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} sintase dipadamkan, oleh
        isyarat hormon yang sama). Sekon sampai menit; berbalikan; dapat
        diperkuat dalam untaian.
  \item \emph{Pengaktifan proteolitik}: sebagian \omterm{def:b1:enzymes:enzyme}{enzim} dibuat sebagai
        prazat tak aktif (\emph{zimogen}: tripsinogen, pepsinogen,
        faktor pembekuan) lalu dinyalakan dengan memotong sebuah
        peptida, sekali untuk selamanya, di tempat dan saat ia
        diinginkan.
  \item \emph{Jumlah}: \omterm{def:b1:cell-unit-of-life:cell}{sel} membuat \omterm{def:b1:enzymes:enzyme}{enzimnya} lebih banyak atau lebih
        sedikit dengan mengendalikan gennya
        (\cref{ch:b1:expression-control}) dan menguraikannya lebih cepat
        atau lebih lambat. Menit sampai jam.
\end{enumerate}
\end{proposition}

\begin{example}[Isoenzim]\label{ex:b1:enzymes:isoenzymes}
Laktat dehidrogenase hadir dalam lima bentuk, tetramer dua jenis subunit
dalam segala gabungannya; bentuk jantungnya ber-$K_m$ rendah bagi
laktat dan dihambat piruvat (ia mengoksidasi laktat untuk memberi makan
daur Krebs), bentuk ototnya ber-$V_{\max}$ tinggi dan tahan piruvat (ia
membuat laktat dalam sebuah lari cepat). Pola bentuknya dalam darah
menyingkapkan \omterm{def:b1:mammal-organization:organ}{organ} mana yang telah rusak --- sebuah serangan jantung
melepas bentuk jantungnya.
\end{example}

\section{Suhu dan pH}

\begin{proposition}[Enzim dan lingkungannya]\label{prop:b1:enzymes:environment}
Laju sebuah reaksi \omterm{def:b1:enzymes:enzyme}{enzim} kira-kira berlipat dua tiap
\qty{10}{\celsius} ($Q_{10} \approx 2$) seiring energi termal membawa
lebih banyak molekul melewati penghalangnya, sampai \omterm{def:b1:enzymes:enzyme}{enzimnya} mulai
terbuka lipatannya (\qtyrange{40}{60}{\celsius} bagi sebagian besar;
\qty{100}{\celsius} bagi \omterm{def:b1:enzymes:enzyme}{enzim} bakteri mata air panas) dan lajunya
runtuh. Tiap \omterm{def:b1:enzymes:enzyme}{enzim} memiliki sebuah \emph{pH optimum}, ketika pengionan
residu katalitiknya dan \omterm{def:b1:enzymes:enzyme}{substratnya} tepat: pepsin, di lambung, dekat pH
2; tripsin, di usus, dekat 8; sebagian besar \omterm{def:b1:enzymes:enzyme}{enzim} intrasel dekat 7.
Jauh dari optimumnya lajunya turun, dan jauh sekali darinya \omterm{def:b1:proteins:peptide}{proteinnya}
terdenaturasi. Suhu dan pH bekerja pada \omterm{def:b1:proteins:peptide}{proteinnya}; \omterm{def:b1:cell-unit-of-life:cell}{sel} menjaga keduanya
dalam rentang yang membuat seribu \omterm{def:b1:enzymes:enzyme}{enzimnya} semua bekerja.
\end{proposition}

\begin{example}[Sebuah demam dan sebuah mata air panas]\label{ex:b1:enzymes:fever}
Demam \qty{40}{\celsius} mempercepat tiap reaksi tubuh seperempat kali
dan mulai membuka lipatan \omterm{def:b1:proteins:peptide}{protein} yang paling rapuh; pada
\qty{42}{\celsius} \omterm{def:b1:proteins:peptide}{protein} otak gagal. \omterm{def:b1:nucleic-acids:chain}{DNA} polimerase sebuah bakteri
dari mata air \qty{75}{\celsius} selamat pada \qty{95}{\celsius}, dan
itulah sebabnya ia dapat didaurkan lewat peleburan \omterm{def:b1:nucleic-acids:chain}{DNA} ribuan kali dan
menjadi \omterm{def:b1:enzymes:enzyme}{enzim} reaksi berantai polimerase (jilid Tahun ke-3):
ketahanan panas adalah sifat urutannya, bukan sifat kimia yang
dikatalisisnya.
\end{example}

\begin{exercise}[$\star$]\label{exo:b1:enzymes:4}
Sebutkan keempat cara \omterm{def:b1:cell-unit-of-life:cell}{sel} mengatur keaktifan sebuah \omterm{def:b1:enzymes:enzyme}{enzim} dan berikan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:enzymes:8}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:enzymes:10}
Tripsinogen diaktifkan di usus, bukan di pankreas yang membuatnya;

\section{Soal: Sebuah Enzim yang Diukur}

\begin{problem}[{Soal akhir pekan --- sebuah esterase yang diuji pada
enam kepekatan \omterm{def:b1:enzymes:enzyme}{substrat}, sendirian dan dengan dua \omterm{def:b1:enzymes:inhibitors}{penghambat},
tetapannya yang disarikan dan perilakunya dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} yang diramalkan,
berakhir pada $K_m$, $V_{\max}$ dan $K_i$-nya}]
\label{pb:b1:enzymes:1}
Sebuah esterase diuji dalam \qty{1.0}{mL} pada \qty{25}{\celsius}, pH
7.5, dengan \qty{10}{nmol/L} \omterm{def:b1:enzymes:enzyme}{enzim}. Laju awalnya (\unit{\micro mol/min})
pada enam kepekatan \omterm{def:b1:enzymes:enzyme}{substrat}, sendirian dan dengan hadirnya
\qty{1.0}{mmol/L} \omterm{def:b1:enzymes:inhibitors}{penghambat} I atau \qty{1.0}{mmol/L} \omterm{def:b1:enzymes:inhibitors}{penghambat} J:

\begin{center}
\small
\begin{tabular}{lcccccc}
\hline
$[S]$ (\unit{mmol/L}) & 0.5 & 1 & 2 & 5 & 10 & 20 \\
\hline
$v_0$, tanpa \omterm{def:b1:enzymes:inhibitors}{penghambat} & 0.200 & 0.333 & 0.500 & 0.714 & 0.833 & 0.909 \\
$v_0$, dengan I & 0.111 & 0.200 & 0.333 & 0.556 & 0.714 & 0.833 \\
$v_0$, dengan J & 0.100 & 0.167 & 0.250 & 0.357 & 0.417 & 0.455 \\
\hline
\end{tabular}
\end{center}

\medskip
\noindent\textbf{Bagian I --- \omterm{def:b1:enzymes:enzyme}{Enzimnya} sendirian.}
\begin{enumerate}
  \item Hitunglah $1/[S]$ dan $1/v_0$ bagi keenam titik tanpa
        \omterm{def:b1:enzymes:inhibitors}{penghambatnya}.
  \item Tunjukkan bahwa titiknya terletak pada sebuah garis lurus lalu
        carilah kemiringan dan perpotongannya.
  \item Simpulkan $V_{\max}$ dan $K_m$.
  \item Periksalah dengan \omterm{thm:b1:enzymes:mm}{persamaan Michaelis--Menten} pada $[S] =
        \qty{5}{mmol/L}$.
  \item Hitunglah banyaknya \omterm{def:b1:enzymes:enzyme}{enzim} dalam pengujiannya (mol) dan
        $k_{\text{cat}}$ dalam \unit{s^{-1}}.
  \item Hitunglah $k_{\text{cat}}/K_m$ lalu bandingkan dengan batas
        difusinya.
  \item Pada kepekatan \omterm{def:b1:enzymes:enzyme}{substrat} berapa \omterm{def:b1:enzymes:enzyme}{enzimnya} bekerja pada
        \qty{95}{\%} $V_{\max}$?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- \omterm{def:b1:enzymes:inhibitors}{Penghambat} I.}
\begin{enumerate}[resume]
  \item Hitunglah $1/v_0$ bagi keenam titik dengan I lalu carilah
        kemiringan dan perpotongan garisnya.
  \item Simpulkan $V_{\max}$ dan $K_m$ tampaknya dengan I.
  \item Parameter mana yang berubah? Golongkan \omterm{def:b1:enzymes:inhibitors}{penghambatnya}.
  \item Hitunglah $K_i$ dari $K_m^{\text{app}} = K_m(1 + [I]/K_i)$.
  \item Pada $[S] = \qty{2}{mmol/L}$, kepekatan I berapa yang membagi
        dua lajunya? Pada $[S] = \qty{20}{mmol/L}$?
  \item Ajukan I itu mungkin apa, secara struktur, dan di mana ia
        mengikat.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- \omterm{def:b1:enzymes:inhibitors}{Penghambat} J.}
\begin{enumerate}[resume]
  \item Hitunglah kemiringan dan perpotongan garisnya dengan J lalu
        simpulkan $V_{\max}$ dan $K_m$ tampaknya.
  \item Golongkan J lalu hitunglah $K_i$-nya dari $V_{\max}^{\text{app}} =
        V_{\max}/(1 + [J]/K_i)$.
  \item Jelaskan mengapa menaikkan \omterm{def:b1:enzymes:enzyme}{substratnya} tidak sanggup mengatasi J.
  \item Melipatduakan kepekatan \omterm{def:b1:enzymes:enzyme}{enzimnya} dengan hadirnya J berbuat apa
        pada lajunya? Dan dengan hadirnya I?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Di dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}.}
\omterm{def:b1:cell-unit-of-life:cell}{Sel} menjaga \omterm{def:b1:enzymes:enzyme}{substratnya} pada \qty{0.5}{mmol/L} dan \omterm{def:b1:enzymes:enzyme}{enzimnya} pada
\qty{10}{nmol/L} dalam volume \qty{1000}{\micro m^3}, pada
\qty{37}{\celsius}, dengan $Q_{10} = 2$.
\begin{enumerate}[resume]
  \item Hitunglah lajunya dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} (molekul hasil tiap sekon dalam
        seluruh \omterm{def:b1:cell-unit-of-life:cell}{selnya}).
  \item Hasilnya dibutuhkan sebanyak \qty{3e6}{} molekul tiap sekon.
        Berapa kali lipat \omterm{def:b1:cell-unit-of-life:cell}{sel} harus menaikkan lajunya, dan sebutkan dua
        cara yang mungkin dipakainya.
  \item Hasilnya adalah \omterm{def:b1:enzymes:inhibitors}{penghambat} bersaing bagi \omterm{def:b1:enzymes:enzyme}{enzimnya} dengan
        $K_i = \qty{0.2}{mmol/L}$, dan menumpuk sampai
        \qty{0.4}{mmol/L}. Hitunglah lajunya pada saat itu, lalu beri
        tanggapan atas apa yang dicapainya.
  \item \omterm{def:b1:enzymes:enzyme}{Enzimnya} difosforilasi sebuah kinase, yang menurunkan $K_m$-nya
        ke \qty{0.4}{mmol/L}. Hitunglah lajunya pada \omterm{def:b1:enzymes:enzyme}{substrat}
        \qty{0.5}{mmol/L} sebelum dan sesudahnya.
  \item pH optimum \omterm{def:b1:enzymes:enzyme}{enzimnya} 7.5 dan keaktifannya terbagi dua pada pH
        6.5. Sebuah \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosom} berada pada pH 5. Ramalkan, secara
        kualitatif, keaktifannya di sana lalu jelaskan dari residu yang
        dapat terion.
  \item Sebuah mutasi mengganti histidina \omterm{def:b1:enzymes:enzyme}{sisi aktifnya} dengan alanina.
        Ramalkan pengaruhnya pada $K_m$ dan pada $k_{\text{cat}}$,
        beserta alasan masing-masing.
  \item Jelaskan mengapa \omterm{def:b1:cell-unit-of-life:cell}{sel} menjaga \omterm{def:b1:enzymes:enzyme}{substratnya} dekat $K_m$ alih-alih
        jauh di atasnya.
  \item Nyatakan hasilnya: $K_m$, $V_{\max}$, $k_{\text{cat}}$
        esterasenya, serta jenis dan $K_i$ tiap \omterm{def:b1:enzymes:inhibitors}{penghambatnya}.
\end{enumerate}
\end{problem}
