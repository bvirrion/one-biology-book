\section*{Bab \ref{ch:g6:how-plants-spread} --- Bagaimana Tumbuhan Menyebar}

\begin{solution}{exo:g6:how-plants-spread:1}
\omterm{prop:g6:how-plants-spread:travel}{Kolonisasi} adalah pendudukan tanah baru oleh sebuah \omterm{def:g5:biodiversity:species}{spesies}.
\omterm{def:g6:how-plants-spread:dispersal}{Penyebaran} adalah pengangkutan \omterm{def:g2:from-seed-to-plant:seed}{biji}, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} atau \omterm{def:g4:plants-without-seeds:spore}{spora} menjauh dari
induknya; pengangkutnya adalah angin, \omterm{def:g1:animals-around-us:animal}{hewan} (di luar dan di dalam),
air --- dan sebagian \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} yang melontar sendiri.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:2}
Di kaki induknya ia akan \omterm{def:g2:from-seed-to-plant:germination}{berkecambah} di dalam naungan induknya dan di
antara \omterm{def:g1:plants-around-us:parts}{akarnya}, bersaing memperebutkan \omterm{def:g6:exploring-our-environment:conditions}{cahaya} dan air melawan seorang
lawan yang seratus kali lebih besar. Jarak melarikan diri dari
pertandingan yang kalah itu.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:3}
Jambul jombang: anginnya. \omterm{prop:g3:flowers-fruits-seeds:transform}{Buah} berkait burdok: \omterm{def:g1:animals-around-us:covering}{bulu} \omterm{def:g1:animals-around-us:animal}{hewan}, di luar.
\omterm{prop:g3:flowers-fruits-seeds:transform}{Buah} ceri: usus \omterm{def:g1:animals-around-us:animal}{hewan}, di dalam. \omterm{prop:g3:flowers-fruits-seeds:transform}{Buah} bersayap maple: anginnya (sayap
yang berputar). \omterm{prop:g3:flowers-fruits-seeds:transform}{Buah} kelapa: airnya.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:4}
\omterm{def:g1:the-five-senses:sense}{Kulitnya} yang liat membuatnya dapat melintasi \omterm{def:g4:journey-of-food:digestion}{pencernaan} \omterm{prop:g3:sorting-living-things:groups}{burungnya}
tanpa terluka; mereka tiba dalam keadaan diendapkan jauh dari
rumahnya, berikut sedosis pupuk.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:5}
Melontar sendiri: polongnya mengering, menegang seperti pegas, lalu
meletik terbuka, melontarkan \omterm{def:g2:from-seed-to-plant:seed}{bijinya} bermeter-meter dari induknya.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:6}
Rumput grinting, yang maju di bawah tanah lewat \omterm{def:g1:plants-around-us:parts}{batang} yang menjalar;
semak beri hitam, yang melengkung ke atas lalu berakar pada ujungnya
(atau stroberi, yang melompat lewat \omterm{ex:g4:plants-without-seeds:runner}{geragihnya}).
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:7}
Jombang: terbang masuk bersama angin oleh jambulnya. \omterm{def:g3:flowers-fruits-seeds:parts}{Bunga} popi:
\omterm{def:g2:from-seed-to-plant:seed}{bijinya} sudah terbaring di dalam tanah, lalu terbangun ketika tanahnya
dipindahkan. Beri hitam: diendapkan \omterm{prop:g3:sorting-living-things:groups}{burung} yang bertengger, \omterm{def:g2:from-seed-to-plant:seed}{bijinya} di
dalam tinjanya. Burdok: terbawa masuk dengan berkait pada \omterm{def:g1:animals-around-us:covering}{bulu} seekor
anjing.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:8}
(a) Perlengkapan terbang --- jambul: anginnya. (b) Berdaging, berwarna,
berbiji \omterm{def:g1:the-five-senses:sense}{kulit} liat di dalamnya: dimakan \omterm{def:g1:animals-around-us:animal}{hewan}, \omterm{def:g2:from-seed-to-plant:seed}{bijinya} menumpang di
ususnya. (c) Kait yang mencengkeram wol: \omterm{def:g1:animals-around-us:covering}{bulu} \omterm{def:g1:animals-around-us:animal}{hewan} --- dan uji baju
hangatnya baru saja menegaskannya.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:9}
\omterm{def:g1:plants-around-us:parts}{Batang} yang melengkung dan berakar di ujungnya mengerjakan semeternya
--- pendudukan yang rapat dan pasti atas tanah di sebelahnya. Beri
yang dimakan \omterm{prop:g3:sorting-living-things:groups}{burung} mengerjakan sekilometernya --- pendaratan jarak
jauh yang jarang, yang mendirikan petak baru di tempat yang jauh.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:10}
\omterm{def:g4:plants-without-seeds:spore}{Spora} mencapai kedua tebingnya --- keduanya mengapung ke mana-mana ---
tetapi perkembangbiakan \omterm{def:g1:plants-around-us:plant}{tumbuhan} berspora membutuhkan \omterm{def:g6:exploring-our-environment:conditions}{kelembapan},
sehingga hanya \omterm{def:g6:exploring-our-environment:conditions}{kondisi} tebing utaranya yang menerima kedatangannya.
\omterm{def:g6:how-plants-spread:dispersal}{Penyebaran} mengantarkan; kondisilah yang memutuskan siapa yang boleh
tinggal: penghuninya cocok dengan \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}, sebanyak apa pun yang
datang.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:11}
Hanya layanan jarak jauh: angin (\omterm{def:g2:from-seed-to-plant:seed}{biji} berjambul dan sehalus debu,
\omterm{def:g4:plants-without-seeds:spore}{spora}), air laut (\omterm{prop:g3:flowers-fruits-seeds:transform}{buah} yang mengapung seperti kelapa), dan \omterm{prop:g3:sorting-living-things:groups}{burung}
(\omterm{def:g2:from-seed-to-plant:seed}{biji} \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} berdaging yang terbawa usus). Ramalkanlah perlengkapan
berjambul, sehalus debu, mengapung atau terbawa beri di antara para
perintisnya --- tidak ada \omterm{def:g2:from-seed-to-plant:seed}{biji} yang berat dan tanpa pengangkut.
\end{solution}

\begin{solution}{exo:g6:how-plants-spread:12}
Gulanya membeli pengangkutan: seekor \omterm{prop:g3:sorting-living-things:groups}{burung} yang dibayar dengan daging
membawa \omterm{def:g2:from-seed-to-plant:seed}{bijinya} jauh dari naungan induknya lalu mengendapkannya
berikut pupuk --- jarak, pemberian yang pertama. \omterm{def:g1:the-five-senses:sense}{Kulit} yang liat
adalah paruh yang lain karena kurirnya berupa \omterm{def:g4:journey-of-food:tract}{saluran pencernaan}:
pembayarannya harus dapat dicerna dan penumpangnya tidak boleh dapat
dicerna. Rasa manis merekrut pengangkutnya; \omterm{def:g1:the-five-senses:sense}{kulitnya} bertahan
melewatinya.
\end{solution}

\begin{solution}{pb:g6:how-plants-spread:1}
\textbf{1.} \omterm{def:g3:flowers-fruits-seeds:parts}{Bunga} popi: terbangun dari \omterm{def:g2:from-seed-to-plant:seed}{biji} yang sudah ada di dalam
tanah yang dibajak --- persediaan tunggu milik \omterm{prop:g6:life-through-seasons:herbs}{tumbuhan semusim}.
Jombang dan rumput duri: terbang masuk bersama angin, oleh \omterm{def:g2:from-seed-to-plant:seed}{bijinya}
yang berjambul.

\textbf{2.} \omterm{prop:g3:sorting-living-things:groups}{Burung} yang memakan beri hitam di tempat lain bertengger
di tiang pagarnya lalu mengendapkan \omterm{def:g2:from-seed-to-plant:seed}{bijinya} di bawahnya, berikut
pupuknya --- sehingga \omterm{prop:g2:from-seed-to-plant:cycle}{kecambahnya} memetakan tenggerannya, bukan
anginnya.

\textbf{3.} Kekeringannya: sebuah ladang yang terbuka dan tersinari
sama sekali tidak menawarkan \omterm{def:g6:exploring-our-environment:conditions}{kelembapan} yang dibutuhkan tahap halus
\omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} yang tumbuh dari \omterm{def:g4:plants-without-seeds:spore}{spora}. Kedatangannya terjadi;
penegakannya ditolak.

\textbf{4.} Kemajuan vegetatif --- \omterm{def:g1:plants-around-us:parts}{batang} yang menjalar di bawah
tanah. Ia hanya berpindah dari tanah ke tanah yang bersebelahan mulai
dari \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang sudah tegak, sehingga ia harus berjalan masuk dari
tepinya; ia tidak punya tahap lewat udara untuk mendarat di tengah.

\textbf{5.} Di dalam tanah, sebagai \omterm{def:g2:from-seed-to-plant:seed}{biji}: \omterm{prop:g6:life-through-seasons:herbs}{tumbuhan semusim} yang kalah
dalam persaingan tegak bertahan sebagai bank \omterm{def:g2:from-seed-to-plant:seed}{biji}, yang menunggu tanah
telanjang berikutnya.

\textbf{6.} Layanan di-dalam-hewannya (beri yang dimakan, \omterm{def:g2:from-seed-to-plant:seed}{bijinya}
diendapkan dari tenggerannya --- paling lebat di bawah garis pagar dan
pagar tanaman) dan, bagi hawthorn juga, kurir \omterm{prop:g3:sorting-living-things:groups}{burung} yang sama; pola
yang berbentuk tenggeran itulah buktinya.

\textbf{7.} \omterm{def:g2:from-seed-to-plant:seed}{Biji} pohon ek dan berangan itu berat, dan tanpa
pertolongan tidak punya pengangkut --- keduanya jatuh di kaki
induknya. Kurirnya adalah \omterm{prop:g3:sorting-living-things:groups}{burung} jay (dan tupai): dengan mengubur \omterm{def:g2:from-seed-to-plant:seed}{biji}
ek sebagai cadangan musim dingin lalu melupakan sebagian, mereka
menanam pohon ek jauh dari induknya.

\textbf{8.} \omterm{def:g1:plants-around-us:plant}{Tumbuhan} tiap tahunnya menaungi tanah di bawahnya,
sehingga permukaannya menjadi lebih sejuk dan lebih gelap: keberhasilan
para perintis pencinta matahari itu sendiri menyingkirkan \omterm{def:g6:exploring-our-environment:conditions}{kondisinya},
dan \omterm{def:g6:exploring-our-environment:conditions}{kondisi} yang berubah itu menguntungkan para penahan naungan
berikutnya --- penjajahnya mengubah lokasinya, dan lokasi yang berubah
itu memilih penjajah yang baru.

\textbf{9.} Gelombang 1, \omterm{def:g2:from-seed-to-plant:seed}{biji} yang menunggu di dalam tanah (seketika
--- mereka sudah ada di situ). Gelombang 2, para penumpang angin
(beberapa minggu: udaranya mengantar terus-menerus). Gelombang 3,
semak yang dibawa \omterm{def:g1:animals-around-us:animal}{hewan} (bertahun-tahun: tenggeran demi tenggeran).
Gelombang 4, pohon berbiji berat (terakhir: mereka menunggu kelupaan
\omterm{prop:g3:sorting-living-things:groups}{burung} jay dan tupai atas kuburannya).

\textbf{10.} Celahnya tidak menawarkan ruang: \omterm{def:g1:plants-around-us:parts}{akar} sebuah semak
membutuhkan volume tanah yang tidak dimiliki sebuah celah trotoar; dan
gelombang yang lebih akhir pada suksesinya membutuhkan kurir dan
\omterm{def:g6:exploring-our-environment:conditions}{kondisinya} --- endapan \omterm{prop:g3:sorting-living-things:groups}{burung} yang bertengger kebanyakan meleset dari
celahnya, dan celah kering yang terpanggang itu menolak apa yang
memang tiba. Penumpang angin tahun pertama adalah satu-satunya layanan
yang dapat diterima lokasinya.

\textbf{11.} Kecenderungannya berjalan dari tanah telanjang, \omterm{def:g1:plants-around-us:plant}{tumbuhan}
kecil, semak, sampai pohon perintis --- tiap gelombang lebih tinggi
dan lebih menaungi: kalau diperpanjang, ia berakhir menjadi hutan,
sebagaimana sudah diperlihatkan pohon dedalu dan birchnya.
Perkecualiannya: sebuah padang rumput yang dipangkas atau digembalai,
tempat sentuhan manusia menyingkirkan gelombang semaknya setiap tahun
--- bukti bahwa hanya \emph{kalau dibiarkan sendiri} ladangnya berjalan
menuju hutan.

\textbf{12.} Misalnya: ``\omterm{def:g6:how-plants-spread:dispersal}{Penyebaran} mengantarkan para penjajah ke
ladang telanjang itu --- dari tanahnya, dari anginnya, dari \omterm{prop:g3:sorting-living-things:groups}{burungnya}
--- dan pada tiap tahapnya \omterm{def:g6:exploring-our-environment:conditions}{kondisi} ladangnya memutuskan kedatangan
mana yang boleh tinggal. Dengan begitu \omterm{prop:g6:how-plants-spread:travel}{kolonisasinya} berjalan dalam
gelombang, dan tiap gelombang mengubah \omterm{def:g6:exploring-our-environment:conditions}{kondisinya} lalu memilih
gelombang berikutnya.''
\end{solution}
