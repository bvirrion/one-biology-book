\chapter{Evidence for Evolution}\label{ch:g8:evolution}

Life on Earth has changed over vast time. \omterm{def:g8:evolution:def}{Evolution} is the scientific
explanation for that pattern. This chapter gathers lines of evidence ---
\omterm{def:g8:evolution:fossil}{fossils}, anatomy, embryos, DNA and \omterm{def:g8:evolution:biogeo}{biogeography} --- without full \omterm{def:g5:evolution:ns}{natural
selection} maths (that deepens in Grade 9).

\section{What evolution means}

\begin{definition}[Evolution]\label{def:g8:evolution:def}
\emph{Evolution}\index{evolution} is change in the \omterm{def:g4:traits:inherited}{inherited traits} of
\omterm{def:g6:ecology:levels}{populations} over generations. On long timescales it produces new forms
and the diversity of life from common ancestors.
\end{definition}

\section{Fossils}

\begin{definition}[Fossil]\label{def:g8:evolution:fossil}
A \emph{fossil}\index{fossil} is a preserved remain or trace of an
\omterm{def:g4:ecosystems:levels}{organism} from the past. The \emph{fossil record}\index{fossil record}
shows sequences of life through rock layers (older generally deeper in
undisturbed strata).
\end{definition}

\begin{proposition}[What fossils show]\label{prop:g8:evolution:fossils}
\omterm{def:g8:evolution:fossil}{Fossils} document extinct \omterm{def:g6:classification:species}{species}, transitional forms, and the order in
which major groups appear. Dating methods estimate ages; patterns match
branching descent better than a single unchanging catalogue of life.
\end{proposition}
\admitted

\begin{example}[Horses and whales]\label{ex:g8:evolution:trans}
Sequences of \omterm{def:g8:evolution:fossil}{fossil} horses show changes in teeth and limbs over time.
\omterm{def:g8:evolution:fossil}{Fossil} whales with hind-limb remnants fit a \omterm{def:g5:vertebrates:groups}{land-mammal} ancestry story
also supported by other evidence.
\end{example}

\section{Anatomy and embryos}

\begin{definition}[Homologous structures]\label{def:g8:evolution:homol}
\emph{Homologous structures}\index{homologous structures} are \omterm{def:g1:my-body:body}{body} parts
in different \omterm{def:g6:classification:species}{species} that share an underlying design from common ancestry,
even if functions differ (for example \omterm{def:g5:vertebrates:split}{vertebrate} forelimbs).
\end{definition}

\begin{definition}[Analogous structures]\label{def:g8:evolution:anal}
\emph{Analogous structures}\index{analogous structures} serve similar
functions but do not share the same deep ancestral design (for example
\omterm{def:g3:insects:insect}{insect} wing vs \omterm{def:g5:vertebrates:groups}{bird} wing). They arise from convergent \omterm{def:g8:evolution:def}{evolution}.
\end{definition}

\section{DNA and distribution}

\begin{proposition}[Molecular evidence]\label{prop:g8:evolution:dna}
\omterm{def:g6:classification:species}{Species} that look different can share similar \omterm{def:g8:genetics:gene}{genes} and \omterm{def:g6:nutrition:groups}{proteins}. Greater
DNA similarity usually means more recent common ancestry --- a family
tree readable in molecules.
\end{proposition}
\admitted

\begin{definition}[Biogeography]\label{def:g8:evolution:biogeo}
\emph{Biogeography}\index{biogeography} is the study of where \omterm{def:g6:classification:species}{species} live.
Island \omterm{def:g6:classification:species}{species} often resemble mainland relatives more than distant island
look-alikes --- patterns expected from colonisation and local change.
\end{definition}

\begin{method}[Weighing evidence]\label{met:g8:evolution:weigh}
\begin{enumerate}
  \item Ask what \omterm{def:g6:method:obs}{observation} \omterm{def:g1:caring:needs}{needs} explaining.
  \item Check whether \omterm{def:g8:evolution:fossil}{fossils}, anatomy, molecules and geography agree.
  \item Prefer explanations that unite many independent data sets.
\end{enumerate}
\end{method}
