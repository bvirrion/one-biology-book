\chapter{Onde os animais vivem}\label{ch:g2:where-animals-live}

Você não vai encontrar uma truta no pátio da escola, nem um esquilo
no fundo da lagoa. Cada \omterm{def:g1:animals-around-us:animal}{animal} vive no tipo de lugar que combina com
ele --- onde acha o alimento, o abrigo e os vizinhos. Os biólogos
têm um nome para o lugar onde um \omterm{def:g1:animals-around-us:animal}{animal} vive.

\section{Hábitats}

\begin{definition}[Hábitat]\label{def:g2:where-animals-live:habitat}
O \emph{hábitat}\index{hábitat} de um \omterm{def:g1:animals-around-us:animal}{animal} é o tipo de lugar onde
ele vive naturalmente e acha tudo de que precisa: alimento, água,
abrigo e um canto seguro para os filhotes. A lagoa, a floresta, o
campo, a cerca viva e a beira do mar são hábitats.
\end{definition}

\begin{example}[Quatro hábitats, quatro vizinhanças]\label{ex:g2:where-animals-live:four}
Na \emph{lagoa}: sapos, libélulas, peixinhos, caramujos de água. Na
\emph{floresta}: esquilos, veados, pica-paus, formigas. No
\emph{campo}: coelhos, gafanhotos, borboletas, ratinhos-do-campo.
Debaixo das pedras de um \emph{muro} velho: tatuzinhos, aranhas e
às vezes um lagarto tomando sol lá em cima.
\end{example}

\begin{remark}[Um hábitat, muitas casas]\label{rem:g2:where-animals-live:layers}
Um \omterm{def:g2:where-animals-live:habitat}{hábitat} tem muitos cantos. Numa mesma floresta, o pica-pau vive
lá no alto dos troncos, o esquilo nos \omterm{def:g1:plants-around-us:tree}{galhos}, o veado no meio dos
arbustos e as formigas no chão, no formigueiro amontoado. A mesma
floresta, andares diferentes --- e é por isso que tantos \omterm{def:g1:animals-around-us:animal}{animais}
conseguem dividir um \omterm{def:g2:where-animals-live:habitat}{hábitat} sem apertar ninguém.
\end{remark}

\section{Feito sob medida para o hábitat}

\begin{proposition}[Os animais combinam com o hábitat]\label{prop:g2:where-animals-live:fit}
O corpo e os hábitos de um \omterm{def:g1:animals-around-us:animal}{animal} combinam com o lugar onde ele vive:
\begin{itemize}
  \item os pés com membrana do sapo empurram a água --- feitos para
        a lagoa;
  \item as garras de agarrar e o rabo grandão de equilíbrio do
        esquilo são feitos para as árvores;
  \item as patas de cavar e as orelhas compridas de alerta do
        coelho são feitas para o campo aberto com tocas embaixo;
  \item as \omterm{def:g1:animals-around-us:covering}{penas} do pato escorrem a água, e os pés com membrana dele
        remam.
\end{itemize}
Levado para o \omterm{def:g2:where-animals-live:habitat}{hábitat} errado, um \omterm{def:g1:animals-around-us:animal}{animal} perde essas vantagens.
\end{proposition}
\admitted

\begin{example}[O equipamento da toupeira]\label{ex:g2:where-animals-live:mole}
A toupeira vive debaixo da terra, em túneis que ela cava embaixo do
campo. As mãos dela são pás largas de cavar, o pelo dela se deita
nos dois sentidos para que a terra nunca o embarace, e os \omterm{def:g1:the-five-senses:sense}{olhos}
fracos quase não fazem falta no escuro. Na superfície ela é
desajeitada; no \omterm{def:g2:where-animals-live:habitat}{hábitat} dela é uma campeã.
\end{example}

\begin{method}[Ligar um animal ao hábitat dele]\label{met:g2:where-animals-live:matching}
Diante de um \omterm{def:g1:animals-around-us:animal}{animal} desconhecido:
\begin{enumerate}
  \item olhe o equipamento dele: pés com membrana? garras? asas?
        patas de cavar?
  \item pergunte o que ele come e onde esse alimento é encontrado;
  \item pergunte onde ele pode se abrigar e criar os filhotes;
  \item o \omterm{def:g2:where-animals-live:habitat}{hábitat} que responde às três perguntas é a casa provável
        dele.
\end{enumerate}
\end{method}

\section{Os hábitats podem se perder}

\begin{proposition}[Sem hábitat, sem moradores]\label{prop:g2:where-animals-live:loss}
Um \omterm{def:g1:animals-around-us:animal}{animal} não consegue viver sem o \omterm{def:g2:where-animals-live:habitat}{hábitat} dele. Se uma lagoa é
aterrada, os sapos e as libélulas dela somem junto; se uma cerca
viva é arrancada, as aves que faziam ninho ali perdem o abrigo.
Proteger os \omterm{def:g1:animals-around-us:animal}{animais} começa por proteger os lugares onde eles vivem.
\end{proposition}
\admitted

\begin{example}[A lagoa da escola]\label{ex:g2:where-animals-live:schoolpond}
Uma escola cava uma lagoinha num canto do pátio. Em dois anos, as
libélulas fazem ronda por ali, os insetos patinadores andam na água
e os sapos se mudaram para lá --- ninguém trouxe esses bichos.
Faça o \omterm{def:g2:where-animals-live:habitat}{hábitat}, e os moradores acham o caminho.
\end{example}

\begin{remark}[Olhar sem incomodar]\label{rem:g2:where-animals-live:respect}
Ao visitar um \omterm{def:g2:where-animals-live:habitat}{hábitat}, somos convidados. Observe em silêncio,
recoloque as pedras exatamente como estavam, não mexa em \omterm{def:g2:animals-grow-up:born}{ovos} nem em
ninhos e leve o seu lixo para casa --- o \omterm{def:g2:where-animals-live:habitat}{hábitat} deve esquecer que
você esteve lá. Os próximos capítulos trazem mais jeitos de cuidar
da natureza.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:1}
O que é um \omterm{def:g2:where-animals-live:habitat}{hábitat}? Diga as quatro coisas que um \omterm{def:g1:animals-around-us:animal}{animal} acha nele.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:2}
Dê dois \omterm{def:g1:animals-around-us:animal}{animais} da lagoa, dois da floresta e dois do campo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:3}
Qual \omterm{def:g2:where-animals-live:habitat}{hábitat} combina com cada \omterm{def:g1:animals-around-us:animal}{animal}: o peixinho, o pica-pau, o
gafanhoto, o tatuzinho?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:4}
Qual é o equipamento de nadar do sapo? E o equipamento de subir em
árvore do esquilo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:5}
Liste os três equipamentos subterrâneos da toupeira do
\cref{ex:g2:where-animals-live:mole}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:6}
Na \cref{rem:g2:where-animals-live:layers}, que ``andar'' da
floresta cada \omterm{def:g1:animals-around-us:animal}{animal} usa: pica-pau, esquilo, veado, formiga?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:7}
O que acontece com os sapos e as libélulas quando uma lagoa é
aterrada? O que isso ensina sobre proteger os \omterm{def:g1:animals-around-us:animal}{animais}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:8}
Faça a experiência: escolha um \omterm{def:g2:where-animals-live:habitat}{hábitat} pequeno perto de você ---
debaixo de uma pedra, um arbusto, um canteiro. Observe em silêncio e
liste todos os \omterm{def:g1:animals-around-us:animal}{animais} que você vir num tempo combinado. Recoloque
cada pedra. Quantos moradores o seu \omterm{def:g2:where-animals-live:habitat}{hábitat} pequenininho tinha?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:where-animals-live:9}
Um \omterm{def:g1:animals-around-us:animal}{animal} misterioso tem pés com membrana, \omterm{def:g1:animals-around-us:covering}{penas} densas que escorrem
a água e um bico chato de peneirar. Use o
\cref{met:g2:where-animals-live:matching} para dizer o \omterm{def:g2:where-animals-live:habitat}{hábitat} dele
--- e talvez o \omterm{def:g1:animals-around-us:animal}{animal}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:where-animals-live:10}
O Tomás quer ``ajudar'' um sapo levando ele para casa, para um
terrário seco e cheio de comida. Usando as
\cref{prop:g2:where-animals-live:fit} e
\cref{prop:g2:where-animals-live:loss}, explique por que isso não
ajudaria o sapo em nada.
\end{exercise}
