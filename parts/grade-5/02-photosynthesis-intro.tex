\chapter{How Plants Make Food}\label{ch:g5:photosynthesis}

Green \omterm{def:g1:plants-animals:plant}{plants} do not shop for meals. In \omterm{def:g4:plants-need:needs}{light}, they build sugars from
simple materials. That process --- \omterm{def:g5:photosynthesis:def}{photosynthesis} --- feeds \omterm{def:g1:plants-animals:plant}{plants} and,
indirectly, almost every \omterm{def:g3:food-chains:chain}{food chain} on land.

\section{The big idea}

\begin{definition}[Photosynthesis]\label{def:g5:photosynthesis:def}
\emph{Photosynthesis}\index{photosynthesis} is the process by which
green \omterm{def:g1:plants-animals:plant}{plants} (and algae) use \omterm{def:g4:plants-need:needs}{light} \omterm{def:g2:food-energy:energy}{energy} to make \omterm{def:g2:food-energy:food}{food} (sugars) from
carbon dioxide and \omterm{def:g4:plants-need:needs}{water}, releasing oxygen as a by-product.
\end{definition}

\begin{proposition}[Word equation]\label{prop:g5:photosynthesis:eq}
In words:
\[
\text{carbon dioxide + water}
  \xrightarrow{\text{light, chlorophyll}}
  \text{sugar + oxygen}.
\]
\end{proposition}
\admitted

\section{Where it happens}

\begin{definition}[Chlorophyll and chloroplasts]\label{def:g5:photosynthesis:chloro}
\emph{Chlorophyll}\index{chlorophyll} is the green pigment that captures
\omterm{def:g4:plants-need:needs}{light} \omterm{def:g2:food-energy:energy}{energy}. It is found in \emph{chloroplasts}\index{chloroplast}
inside \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g5:cell-structure:cell}{cells}, especially in \omterm{def:g2:plant-parts:main}{leaves}.
\end{definition}

\omimg{plant-parts.jpg}{0.52\linewidth}{\omterm{def:g4:plants-need:parts}{Leaves} are the main
photosynthetic \omterm{def:g4:body-systems:organs}{organs}; \omterm{def:g2:plant-parts:main}{roots} supply \omterm{def:g4:plants-need:needs}{water}; \omterm{def:g4:plants-need:parts}{stems} support and transport.}

\begin{example}[Why leaves are flat and green]\label{ex:g5:photosynthesis:leaf}
A broad flat shape catches \omterm{def:g4:plants-need:needs}{light}. Green colour comes from \omterm{def:g5:photosynthesis:chloro}{chlorophyll}.
Stomata (tiny pores) let gases in and out --- details expand later.
\end{example}

\section{What plants do with sugar}

\begin{definition}[Uses of sugar]\label{def:g5:photosynthesis:uses}
\omterm{def:g1:plants-animals:plant}{Plants} use sugar for \omterm{def:g2:food-energy:energy}{energy} (in respiration), \omterm{def:g2:growing:growth}{growth}, and storage as
starch in \omterm{def:g2:plant-parts:main}{seeds}, \omterm{def:g2:plant-parts:main}{roots} or tubers. \omterm{def:g1:plants-animals:animal}{Animals} that eat \omterm{def:g1:plants-animals:plant}{plants} take in that
stored \omterm{def:g2:food-energy:energy}{energy}.
\end{definition}

\begin{proposition}[Oxygen bonus]\label{prop:g5:photosynthesis:oxygen}
\omterm{def:g5:photosynthesis:def}{Photosynthesis} adds oxygen to the \omterm{def:g4:plants-need:needs}{air}. \omterm{def:g1:plants-animals:animal}{Animals} and \omterm{def:g1:plants-animals:plant}{plants} both need
oxygen for respiration, so photosynthetic \omterm{def:g4:ecosystems:levels}{organisms} support the whole
living world.
\end{proposition}
\admitted

\section{What limits photosynthesis}

\begin{method}[Thinking about limiting factors]\label{met:g5:photosynthesis:limits}
\begin{enumerate}
  \item Ask which input is in short supply: \omterm{def:g4:plants-need:needs}{light}, \omterm{def:g4:plants-need:needs}{water}, carbon
        dioxide, or healthy \omterm{def:g5:photosynthesis:chloro}{chlorophyll}.
  \item Predict: increasing a scarce factor may raise the rate until
        something else limits it.
  \item Remember temperature also affects living processes.
\end{enumerate}
\end{method}

\begin{example}[Cupboard plants]\label{ex:g5:photosynthesis:dark}
\omterm{def:g1:plants-animals:plant}{Plants} kept in darkness cannot photosynthesise. They may use stored
\omterm{def:g2:food-energy:food}{food} for a while, then weaken. Pale colour can appear when \omterm{def:g4:plants-need:needs}{light} is too
low for normal \omterm{def:g5:photosynthesis:chloro}{chlorophyll} development.
\end{example}
