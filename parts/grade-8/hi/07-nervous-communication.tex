\chapter{तंत्रिका वाला संवाद}\label{ch:g8:nervous-communication}

\Cref{ch:g5:brain-in-command} ने वह फंदा खींचा था --- ख़बर, फ़ैसला,
हुक्म --- और उसका सेकंड का पाँचवाँ हिस्सा भी नापा था। इस साल हम वे तार
खोलते हैं। किसी \omterm{def:g8:nervous-communication:neuron}{तंत्रिका} के साथ-साथ असल में सफ़र क्या करता है?
\omterm{def:g5:brain-in-command:system}{तंत्रिकाएँ} बनी किस चीज़ से हैं? और उन ख़ाली जगहों पर क्या होता है जहाँ एक
तार ख़त्म होता है और अगला शुरू --- यानी वे ख़ाली जगहें, जहाँ, जैसा
निकलेगा, याददाश्त भी अपना काम करती है और बहुत-से ज़हर भी?

\section{तार और उनकी कोशिकाएँ}

\begin{definition}[न्यूरॉन और तंत्रिका-संदेश]\label{def:g8:nervous-communication:neuron}
\omterm{def:g5:brain-in-command:system}{तंत्रिका तंत्र} की काम करती \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} \emph{न्यूरॉन}\index{न्यूरॉन} हैं
(\cref{rem:g6:cell-unit-of-life:twoways} ने उन्हें गिनाया था): हर एक
में एक कोशिका-काया होती है और एक लंबा पतला तंतु --- और कुछ तंतु \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} से
पैर की उँगली तक पहुँचते हैं, यानी शरीर की सबसे लंबी \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ}। और
\emph{तंत्रिका}\index{तंत्रिका} ऐसे हज़ारों तंतुओं को बाँधने वाली एक
केबल है। हर तंतु के साथ-साथ \emph{तंत्रिका-संदेश}\index{तंत्रिका-संदेश}
चलते हैं: यानी छोटे-छोटे बिजली के इशारे, जो \qty{100}{m/s} तक की
रफ़्तार से दौड़ते हैं (\cref{def:g5:brain-in-command:system} वाला आँकड़ा,
अब अपने वाहक के साथ)।
\end{definition}

\begin{proposition}[आवृत्ति से कूट-भाषा]\label{prop:g8:nervous-communication:code}
ख़ुद वे इशारे सब एक जैसे हैं --- जो बदलता है वह उनकी \emph{दर} है।
हलका-सा छूना ख़बर वाले तंतु पर धीमी थपकियाँ भेजता है; ज़ोरदार छूना, तेज़
ढोल। \omterm{def:g5:brain-in-command:system}{तंत्रिका तंत्र} की भाषा में एक ही शब्द है, जो अलग-अलग रफ़्तारों पर
दोहराया जाता है: तीव्रता की कूट-भाषा आवृत्ति है, हर लाइन पर, भीतर की ओर
भी और बाहर की ओर भी।
\end{proposition}
\admitted

\begin{example}[तीन लाइनें, एक ही भाषा]\label{ex:g8:nervous-communication:lines}
फंदे का वही कलाकार-दल (\cref{prop:g5:brain-in-command:loop}), नए सिरे
से जुड़ा हुआ: \emph{ख़बर वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन}} ख़बरें ढोते हैं --- \omterm{def:g1:the-five-senses:sense}{त्वचा} से
\omterm{def:g5:brain-in-command:system}{मेरुरज्जु} तक, \omterm{def:g1:the-five-senses:sense}{आँख} से \omterm{def:g5:brain-in-command:system}{मस्तिष्क} तक; \emph{गति वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन}} हुक्म ढोते
हैं --- \omterm{def:g5:brain-in-command:system}{मेरुरज्जु} से \omterm{def:g3:bones-and-muscles:muscle}{पेशी} तक, और वहीं ख़त्म होते हैं जहाँ तंतु \omterm{def:g3:bones-and-muscles:muscle}{पेशी} से
मिलता है तथा उसका ढोल सिकुड़न की ताक़त तय करता है
(\cref{def:g3:bones-and-muscles:muscle}); और उनके बीच, \omterm{def:g5:brain-in-command:system}{मस्तिष्क} तथा
\omterm{def:g5:brain-in-command:system}{मेरुरज्जु} में, अनगिनत \emph{जोड़ने वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन}} फ़ैसला करते हैं।
प्रतिवर्त हो या पूरा संगीत, आवाजाही वही इशारे हैं, अलग-अलग दरों पर,
अलग-अलग लाइनों पर।
\end{example}

\section{ख़ाली जगहें: तंत्रिका-संधियाँ}

\begin{definition}[तंत्रिका-संधि]\label{def:g8:nervous-communication:synapse}
\omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} आपस में छूते नहीं। जहाँ एक तंतु का सिरा अगली \omterm{def:g5:first-look-at-cells:cell}{कोशिका} से मिलता है
वहाँ एक बेहद सूक्ष्म ख़ाली जगह पड़ी होती है --- यानी
\emph{तंत्रिका-संधि}\index{तंत्रिका-संधि}। बिजली का इशारा उसे कूद नहीं
सकता; उसके बजाय आता हुआ तंतु उस ख़ाली जगह में एक \emph{रासायनिक
संदेशवाहक} छोड़ता है, जो पार करता है, दूसरी तरफ़ जाकर जुड़ता है, और ---
अगर काफ़ी मात्रा में पहुँचे --- अगले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} में इशारा नए सिरे से शुरू कर
देता है। तार पर बिजली, और हर जोड़ पर रसायन: पूरा तंत्र बारी-बारी से
दोनों चलाता है।
\end{definition}

\begin{proposition}[टाँका क्यों नहीं, ख़ाली जगह क्यों]\label{prop:g8:nervous-communication:whygaps}
ये ख़ाली जगहें ख़ामी जैसी दिखती हैं और असल में इस तंत्र की प्रतिभा हैं:
\begin{itemize}
  \item वे जोड़ों को \emph{सधने लायक़} बनाती हैं --- बार-बार पार की
        गई \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधि} मज़बूत होती जाती है, और बेकार पड़ी मुरझा जाती
        है: यही सीखना है
        (\cref{rem:g5:brain-in-command:training} वाले ``बेहतर
        रास्ते'', अब मिल गए --- वे मज़बूत हुई \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियाँ} ही
        हैं);
  \item वे इशारों को \emph{एकतरफ़ा} बनाती हैं --- संदेशवाहक सिर्फ़ एक
        ही तरफ़ छूटते हैं, इसलिए आवाजाही उलटी नहीं चल सकती;
  \item और वे एक \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} को हज़ारों आवक इशारों को \emph{तौलने} देती
        हैं --- उकसाने वाली और दबाने वाली आवाज़ें हर \omterm{def:g5:first-look-at-cells:cell}{कोशिका} पर उसके
        चलने से पहले जोड़ी जाती हैं: यानी फंदे का फ़ैसला, जोड़ों पर
        किया हुआ।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[बाज़ीगर की तंत्रिका-संधियाँ]\label{ex:g8:nervous-communication:juggler}
\Cref{rem:g5:brain-in-command:training} वाला बाज़ीगर, आख़िरकार समझाया
हुआ: महीने भर की प्रैक्टिस ने वही इशारे उन्हीं जोड़ों के पार तब तक भेजे,
जब तक वे \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियाँ} मज़बूत नहीं हो गईं --- और जो रास्ता पहले
मेहनत से तौला जाता था वह अब पहले से तुला हुआ चलता है। हुनर जोड़ों पर जमा
हुआ रसायन है; और ``प्रैक्टिस पक्का कर देती है'' दरअसल \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियों}
के बारे में कहा गया वाक्य है।
\end{example}

\begin{remark}[ख़ाली जगहें ही कमज़ोर बिंदु हैं]\label{rem:g8:nervous-communication:vulnerable}
बाहर से \omterm{def:g5:brain-in-command:system}{तंत्रिका तंत्र} पर जो कुछ भी असर करता है वह \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियों} पर
ही करता है --- क्योंकि वही एक जगह है जहाँ इशारा रसायन बन जाता है, यानी
जुड़ने लायक़ और ठगे जाने लायक़। निश्चेतक उन्हें चुप करा देते हैं; विष
उन्हें जाम कर देते हैं (कुछ \omterm{def:g3:bones-and-muscles:muscle}{पेशी} वाले जोड़ का संदेशवाहक रोककर लक़वा मार
देते हैं, कुछ उसे बाढ़ की तरह भरकर); और
\cref{prop:g5:healthy-choices:substances} वाली शराब तथा हर वह नशा, जो
मिज़ाज या बोध बदलता है, इन्हीं ख़ाली जगहों के संदेशवाहकों की नक़ल करके,
उन्हें रोककर या उन्हें चुकाकर काम करता है।
\Cref{ch:g8:protecting-the-nervous-system} ठीक इसी बात पर आगे बढ़ता
है।
\end{remark}

\section{तेज़ तार, धीमी डाक: दो तंत्रों की तुलना}

\begin{proposition}[तंत्रिकाएँ बनाम हॉर्मोन]\label{prop:g8:nervous-communication:compare}
अब शरीर संवाद के दो तंत्र दिखा चुका है, और दोनों उलटे कामों के लिए बने
हैं:
\begin{itemize}
  \item \emph{\omterm{def:g8:nervous-communication:neuron}{तंत्रिका} वाला}: मिलीसेकंडों में इशारे, अपनी ही ख़ास
        लाइनों पर, और ठीक-ठीक पतों तक --- एक \omterm{def:g3:bones-and-muscles:muscle}{पेशी}, एक ग्रंथि --- और
        इशारा रुकते ही ख़त्म: यानी कैचों, बोली और प्रतिवर्तों का
        तंत्र;
  \item \emph{\omterm{def:g8:puberty:hormones}{हॉर्मोन} वाला} (\cref{def:g8:puberty:hormones}):
        मिनटों से महीनों में संदेशवाहक, जिन्हें \omterm{prop:g4:journey-of-food:absorb}{रक्त} हर उस अंग तक
        प्रसारित करता है जो सुनने के लिए बना है, और जो धीमे,
        पूरे-शरीर वाले कार्यक्रमों का हुक्म देते हैं --- वृद्धि, चक्र,
        और \omterm{def:g8:puberty:puberty}{यौवनारंभ} की दोबारा बनावट।
\end{itemize}
तार का संदेश और डाक की सेवा: शरीर दोनों चलाता है, और
\cref{ch:g8:hormonal-communication} डाक वाली सेवा को उसका अपना अध्याय दे
देता है।
\end{proposition}
\admitted

\begin{method}[किसी संवाद की तंत्र-दर-तंत्र पहचान]\label{met:g8:nervous-communication:diagnose}
शरीर के किसी भी तालमेल के लिए, उसका तंत्र तय करो:
\begin{enumerate}
  \item उसका वक़्त नापो: मिलीसेकंड --- \omterm{def:g8:nervous-communication:neuron}{तंत्रिका} वाला; मिनटों से साल
        --- \omterm{def:g8:puberty:hormones}{हॉर्मोन} वाला (या दोनों, रिले में);
  \item उसकी पहुँच का नक़्शा बनाओ: एक ठीक-ठीक पता --- \omterm{def:g8:nervous-communication:neuron}{तंत्रिका} वाला;
        पूरा शरीर --- \omterm{def:g8:puberty:hormones}{हॉर्मोन} वाला;
  \item रास्ता ढूँढ़ो: नाम लेने लायक़ कोई तार वाला रास्ता --- \omterm{def:g8:nervous-communication:neuron}{तंत्रिका}
        वाला; रक्त-धारा का प्रसारण --- \omterm{def:g8:puberty:hormones}{हॉर्मोन} वाला;
  \item और टिकाव जाँचो: इशारे के साथ ही ख़त्म --- \omterm{def:g8:nervous-communication:neuron}{तंत्रिका} वाला;
        संदेशवाहक जब तक घूमता रहे तब तक चलता हुआ --- \omterm{def:g8:puberty:hormones}{हॉर्मोन} वाला।
\end{enumerate}
\end{method}

\begin{example}[तीन मामलों पर यह तरीक़ा]\label{ex:g8:nervous-communication:cases}
हवा की फुहार पर पलक झपकना: मिलीसेकंड, एक \omterm{def:g3:bones-and-muscles:muscle}{पेशी} का छल्ला, तार वाला रास्ता,
और फ़ौरन ख़त्म --- \omterm{def:g8:nervous-communication:neuron}{तंत्रिका} वाला। वृद्धि की उछाल: साल, पूरा शरीर,
रक्त-धारा, और मौसमों तक चलती हुई --- \omterm{def:g8:puberty:hormones}{हॉर्मोन} वाला। और दौड़ में दौड़ता
\omterm{def:g4:heart-and-blood:heart}{दिल} (\cref{prop:g7:organs-at-work:supply}): दोनों, रिले में ---
\omterm{def:g8:nervous-communication:neuron}{तंत्रिका} के इशारे रफ़्तार एक ही धड़कन के भीतर चढ़ा देते हैं, और
रक्त-धारा का एक संदेशवाहक (वही मशहूर लहर, जो डर की गरम लहर की तरह महसूस
होती है) पूरे शरीर की चौकसी मिनटों तक थामे रखता है। पहले तार, फिर डाक।
\end{example}

\begin{exercise}[$\star\star$]\label{exo:g8:nervous-communication:8}
``प्रैक्टिस पक्का कर देती है।'' इस नारे को बाज़ीगर के साथ, ख़ाली जगहों पर
समझाओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:nervous-communication:9}
बाहर से \omterm{def:g5:brain-in-command:system}{तंत्रिका तंत्र} पर असर करने वाली हर चीज़ \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियों} पर ही
क्यों असर करती है? उनमें ख़ास क्या है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:nervous-communication:10}
\cref{met:g8:nervous-communication:diagnose} इन पर चलाओ: (क) गरम कड़ाही
से हाथ खींचना; (ख) \omterm{def:g8:puberty:puberty}{यौवनारंभ} में आवाज़ का बदलना; (ग) डर की वह गरम लहर, जो

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:nervous-communication:11}

भविष्यवाणी करो, और फंदे का वह चरण भी बताओ जो कट गया।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:nervous-communication:12}
``तार पर बिजली, जोड़ पर रसायन --- और दिलचस्प सब कुछ जोड़ों पर ही होता
है।'' आख़िरी उपवाक्य का बचाव तीन तरह से करो: सीखने से, फ़ैसले से, और
कमज़ोरी से।
\end{exercise}

\section{समस्या: प्रतिवर्त की मेज़}

\begin{problem}[{सप्ताहांत समस्या --- घुटने की उछाल, साज़ों के
साथ}]\label{pb:g8:nervous-communication:1}
वही चिर-परिचित मेज़ वाला प्रदर्शन: बैठा हुआ स्वयंसेवक, घुटने की चपनी के
नीचे नन्हा हथौड़ा, और पैर की बिना चाहे उठती ठोकर --- अब घड़ियों और इस
अध्याय के साथ।

\medskip\noindent\textbf{भाग I --- परिपथ।}
\begin{enumerate}
  \item हथौड़ा एक कंडरा को खींचता है; और सेकंड के लगभग पचासवें हिस्से
        बाद पैर ठोकर मारता है। परिपथ बिछाओ: कौन-से \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन}, किन-किन से
        होकर --- और इस फंदे पर \omterm{def:g5:brain-in-command:system}{मस्तिष्क} क्यों नहीं है।
  \item सेकंड का पचासवाँ हिस्सा गेंद के कैच वाले पाँचवें हिस्से से
        तेज़ है। इस फ़र्क़ का हिसाब परिपथ की लंबाई और जोड़ों की गिनती
        से दो।
  \item चेतावनी देने पर भी स्वयंसेवक वह ठोकर दबा नहीं पाता। यह इस बारे
        में क्या कहता है कि यह प्रतिवर्त कहाँ तय होता है --- और
        \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियों} पर एकतरफ़ा आवाजाही के बारे में क्या?
  \item ठोकर के फ़ौरन बाद स्वयंसेवक को वह चोट \emph{महसूस} होती है।
        इस देरी वाले दूसरे, धीमे सफ़र को समझाओ।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- मेज़ पर फेरबदल।}
\begin{enumerate}[resume]
  \item ज़्यादा ज़ोर से मारो: ठोकर तेज़ हो जाती है।
        \cref{prop:g8:nervous-communication:code} का कौन-सा कूट-नियम
        ख़ुद को दिखा गया?
  \item कक्षा के डॉक्टर बताते हैं कि हर जाँच में यह प्रतिवर्त परखा
        जाता है। घुटने की मौजूद और दोनों तरफ़ बराबर उछाल \omterm{def:g5:brain-in-command:system}{मेरुरज्जु},
        \omterm{def:g5:brain-in-command:system}{तंत्रिकाओं} और जोड़ों के बारे में किस बात की गवाही देती है?
  \item जबड़ा भींचे हुए स्वयंसेवक की ठोकर ज़्यादा ज़ोरदार होती है ---
        यह एक जाना-पहचाना बढ़ाव है। इससे इस बारे में क्या पता चलता है
        कि जोड़ने वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉनों} की आवाज़ें ``बिना \omterm{def:g5:brain-in-command:system}{मस्तिष्क} वाले''
        फंदों तक भी पहुँचती हैं?
  \item \omterm{def:g1:my-body:parts}{टाँग} में लगा स्थानीय निश्चेतक ठोकर और महसूस होना, दोनों मिटा
        देता है। \cref{rem:g8:nervous-communication:vulnerable} के
        अनुसार हमले की जगह बताओ।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- लिखित रिपोर्ट।}
\begin{enumerate}[resume]
  \item इस प्रतिवर्त को \cref{prop:g5:brain-in-command:loop} वाले तीन
        क़दमों के फंदे की तरह खींचो, और यह भी निशान लगाओ कि यह कहाँ से
        छोटा रास्ता लेता है।
  \item प्रतिवर्त और कैच की तुलना दो स्तंभों की तालिका में करो:
        रास्ता, जोड़, समय, दबा पाने की गुंजाइश, और प्रैक्टिस की
        भूमिका।
  \item रिपोर्ट का आख़िरी हिस्सा पूछता है: ``तेज़, हिफ़ाज़त वाले फंदे
        \omterm{def:g5:brain-in-command:system}{मस्तिष्क} को छोड़कर क्यों निकल जाते हैं?'' जवाब गरम कड़ाही वाले
        तर्क से दो (\cref{exo:g5:brain-in-command:5})।
  \item समापन के लिए एक वाक्य: इस मेज़ ने \omterm{def:g5:brain-in-command:system}{तंत्रिका तंत्र} की भाषा, उसके
        तारों और उसके जोड़ों, तीनों के बारे में एक साथ क्या दिखाया।
\end{enumerate}
\end{problem}
