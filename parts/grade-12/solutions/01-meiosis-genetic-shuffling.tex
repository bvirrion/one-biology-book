\section*{Chapter \ref{ch:g12:meiosis-genetic-shuffling} --- Meiosis and Genetic Shuffling}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:1}
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{Diploid}: two copies of each \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} ($2n$); \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{haploid}: one ($n$).
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{Meiosis} halves the \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} number, from $2n$ to $n$, and reduces
the \omterm{def:g10:universal-dna:information}{DNA} from $2q$ (after replication) to $q/2$ per gamete.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:2}
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{Meiosis} I separates the homologous chromosomes of each pair; \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis}
II separates the two \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} of each \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}. The first is
reductional.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:3}
After \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis} I: 4 \omterm{def:g10:universal-dna:information}{chromosomes} (two \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} each), \omterm{def:g10:universal-dna:information}{DNA} $q$. After
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis} II: 4 \omterm{def:g10:universal-dna:information}{chromosomes} (one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid} each), \omterm{def:g10:universal-dna:information}{DNA} $q/2$.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:4}
$2^4 = 16$.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:5}
A cross with a partner carrying only \omterm{def:g11:genes-and-disease:genetic}{recessive} \omterm{def:g10:universal-dna:gene}{alleles}, so that each
offspring shows the tested parent's gamete. The proportion of
recombinants reveals whether two \omterm{def:g10:universal-dna:gene}{genes} are on the same \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} and,
if so, how far apart.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:6}
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{Meiosis} I: $2q$; between the divisions: $q$; gamete: $q/2$. Without an
intervening S phase, the second division halves the \omterm{def:g10:universal-dna:information}{DNA} again, which
is what brings the gamete to half a \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{diploid} \omterm{def:g10:cells-common-unit:cell}{cell}'s content.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:7}
Four equal classes: not linked, on different \omterm{def:g10:universal-dna:information}{chromosomes}. Gametes
$AB$, $Ab$, $aB$, $ab$, a quarter each.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:8}
Linked: two large classes ($AB$, $ab$, parental) and two small ($Ab$,
$aB$, recombinant). Recombinants $85/1000 = 8.5\%$.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:9}
Only $AB$ and $ab$: an exchange above both \omterm{def:g10:universal-dna:gene}{genes} swaps segments that
carry neither, so the combinations of $A/a$ and $B/b$ are unchanged.
Recombination between two \omterm{def:g10:universal-dna:gene}{genes} requires an exchange between them.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:10}
About 0.8, 2.9 and 16 per 1000: a factor of 20 between 25 and 42.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:11}
At \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis} I, both homologues go to one \omterm{def:g10:cells-common-unit:cell}{cell}: two gametes with two
copies and two with none. At \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis} II, both \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} of one
\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} go to one \omterm{def:g10:cells-common-unit:cell}{cell}: one gamete with two copies, one with none,
and two normal.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:12}
Recombination is proportional to distance: 2\% means the \omterm{def:g10:universal-dna:gene}{genes} are
very close, so an exchange rarely falls between them; 48\% means far
apart, with an exchange between them in nearly every \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis}. Since
one exchange gives 50\% recombinant gametes, distant linked \omterm{def:g10:universal-dna:gene}{genes}
recombine as freely as independent ones.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:13}
$AB$, $Ab$, $aB$, $ab$, a quarter each, by independent assortment of
the two pairs. \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{Crossing-over} changes nothing for these \omterm{def:g10:universal-dna:gene}{genes}: it
recombines \omterm{def:g10:universal-dna:gene}{alleles} along one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}, and these are on different
ones.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:14}
Each parent passes each child one of two \omterm{def:g10:universal-dna:gene}{alleles} at every \omterm{def:g10:universal-dna:gene}{gene}, chosen
at random; at a given \omterm{def:g10:universal-dna:gene}{gene}, two siblings received the same \omterm{def:g10:universal-dna:gene}{allele} from
a parent with probability $1/2$. Over many \omterm{def:g10:universal-dna:gene}{genes} the fraction shared
averages $1/2$, but for any one pair of siblings it varies around it.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:15}
Clones of the mother plant, genetically identical to it and to each
other. The plant gains a fast, sure multiplication of a \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} that
works here and now; it loses the variety that would let some
descendants survive a change of environment or a new parasite.
\end{solution}

\begin{solution}{pb:g12:meiosis-genetic-shuffling:1}
\textbf{1.} Female $b^+b$, $vg^+vg$, with $b^+vg^+$ on one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}
and $b\,vg$ on the other (from her two pure parents). Male $bb$,
$vg\,vg$.

\textbf{2.} Only $b\,vg$ gametes: the male contributes \omterm{def:g11:genes-and-disease:genetic}{recessive}
\omterm{def:g10:universal-dna:gene}{alleles} only, so each offspring's \omterm{def:g11:enzymes-and-phenotype:phenotype}{phenotype} reveals the female's
gamete.

\textbf{3.} $b^+vg^+$ (grey-normal), $b\,vg$ (black-vestigial),
$b^+vg$ (grey-vestigial), $b\,vg^+$ (black-normal).

\textbf{4.} Grey-normal 42\%, black-vestigial 41\%, grey-vestigial 9\%,
black-normal 8\%.

\textbf{5.} On the same \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}: two large and two small classes
instead of $1:1:1:1$.

\textbf{6.} Grey-normal and black-vestigial: the combinations carried
by her two homologues, inherited intact from her pure parents.

\textbf{7.} Grey-vestigial and black-normal, from a \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{crossing-over}
between the two \omterm{def:g10:universal-dna:gene}{genes} in prophase I.

\textbf{8.} $(206 + 185)/2300 \approx 17\%$. A \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{crossing-over} between
the \omterm{def:g10:universal-dna:gene}{genes} makes two of the four \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} recombinant, so 17\%
recombinant gametes means an exchange in about 34\% of meioses.

\textbf{9.} A single exchange produces the two recombinant \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}
together, one of each kind; and the two homologues go to the gametes
equally, so the two parental classes match too.

\textbf{10.} Parental classes now grey-vestigial and black-normal,
about 41.5\% each; recombinants grey-normal and black-vestigial, about
8.5\% each.

\textbf{11.} Yes: recombination well below 50\% with both.

\textbf{12.} $b$ --- $cn$ --- $vg$: 9 and 8 add up to 17.

\textbf{13.} The chance of an exchange in an interval is proportional
to its length, so the frequencies of adjacent intervals add:
recombination frequency measures distance along the \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}.

\textbf{14.} One exchange gives two recombinant \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} out of four:
50\% at most; additional exchanges reshuffle but never raise the
recombinant fraction above half.

\textbf{15.} It is on another \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} or very far along the same
one; the crosses cannot tell which.

\textbf{16.} $2^4 = 16$ gametes; $16 \times 16 = 256$ combinations.

\textbf{17.} Every exchange produces \omterm{def:g10:universal-dna:information}{chromosomes} that never existed
before, at a position that varies from one \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis} to the next; the
number of distinct \omterm{def:g10:universal-dna:information}{chromosomes}, hence of gametes, has no practical
limit.

\textbf{18.} $2^{23} \times 2^{23} \approx \num{7e13}$: seven hundred
times more combinations than the number of humans who have ever lived,
and that before \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{crossing-over}.

\textbf{19.} Siblings draw their \omterm{def:g10:universal-dna:gene}{alleles} from the same four sets
(two per parent) and share, on average, half of them; strangers draw
from different sets.

\textbf{20.} 17\%, the fraction of the female's gametes recombined
between $b$ and $vg$, measures the distance between the \omterm{def:g10:universal-dna:gene}{genes};
independent assortment of the pairs and \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{crossing-over} within each
pair make every gamete different.
\end{solution}
