\section*{الفصل \ref{ch:g1:the-five-senses} --- الحواس الخمس}

\begin{solution}{exo:g1:the-five-senses:1}
البصر --- \omterm{def:g1:the-five-senses:sense}{العينان}؛ والسمع --- \omterm{def:g1:the-five-senses:sense}{الأذنان}؛ والشمّ --- الأنف؛ والذوق
--- اللسان؛ واللمس --- \omterm{def:g1:the-five-senses:sense}{الجلد}.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:2}
(أ) \omterm{def:g1:the-five-senses:sense}{العينان}. (ب) \omterm{def:g1:the-five-senses:sense}{الجلد}. (ج) الأنف. (د) \omterm{def:g1:the-five-senses:sense}{الأذنان}.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:3}
\omterm{def:g1:the-five-senses:sense}{الجلد}، وهو عضو اللمس. وهو يغطي الجسم كله، من الرأس إلى القدم.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:4}
\omterm{def:g1:the-five-senses:sense}{جلد} اليدين أرسل الرسالة (بارد، صلب، مسنّن)؛ والدماغ فهمها وسمّى
الشيء.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:5}
الذوق والشمّ. فالأنف تؤدي نصف عمل النكهة، ولذلك يبدو الطعام بلا
طعم حين تنسدّ.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:6}
مثلًا: لا تنظر إلى الشمس مباشرة --- يحمي \omterm{def:g1:the-five-senses:sense}{العينين}؛ وأبقِ الأصوات
هادئة --- يحمي \omterm{def:g1:the-five-senses:sense}{الأذنين}؛ وانفخ في الطعام الساخن --- يحمي اللسان؛
ولا تدخل شيئًا في \omterm{def:g1:the-five-senses:sense}{الأذن} أو الأنف --- يحمي \omterm{def:g1:the-five-senses:sense}{الأذنين} والأنف؛ والضوء
الجيد للقراءة --- يحمي \omterm{def:g1:the-five-senses:sense}{العينين}.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:7}
لأن الصوت الشديد العلو يؤذي \omterm{def:g1:the-five-senses:sense}{الأذنين}، \omterm{def:g1:the-five-senses:sense}{والأذن} المصابة قد تبقى
مصابة. والواقيات تُبقي ضجيج الآلات هادئًا بما يكفي لعمر كامل من
السمع.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:8}
الجواب مفتوح --- وأكثر الناس يسمّون اثنين أو ثلاثة من الثلاثة.
الأنف تتعرف الأطعمة المألوفة جيدًا، ولو بلا عون من \omterm{def:g1:the-five-senses:sense}{العينين}.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:9}
\omterm{def:g1:the-five-senses:sense}{العين} \emph{ترى} فحسب --- فهي تجمع الضوء وترسل رسالة. أما الفهم
فللدماغ: يتلقى الرسالة فيقول ``هذه الجدة''، ``هذه مدرستي''.
الحواس تجمع، والدماغ يفهم.
\end{solution}

\begin{solution}{exo:g1:the-five-senses:10}
القراءة: \omterm{def:g1:animals-around-us:parts}{بأطراف} الأصابع --- فاللمس يتولى عن البصر، ويحسّ بالنقاط
البارزة على الصفحة. وعبور الشارع: \omterm{def:g1:the-five-senses:sense}{بالأذنين} --- بالإصغاء إلى
السيارات مقبلة ومدبرة --- وبعصا طرفها يدع اللمس يفحص الأرض أمامه.
\end{solution}
