\chapter{إزالة الفضلات}\label{ch:g7:removing-wastes}

سيطرت التوصيلات على هذه السنة: \omterm{def:g7:what-is-respiration:respiration}{أكسجين} إلى الداخل، \omterm{def:g7:digestion-and-nutrients:families}{ومغذيات} إلى
الداخل، ووقود إلى كل \omterm{def:g5:first-look-at-cells:cell}{خلية} عاملة. لكن كل مشغل يصنع فضلات، وخدمة
توصيل لا تجمع النفايات أبدًا تسمّم مدينتها سريعًا. أما حساب \omterm{def:g7:what-is-respiration:respiration}{ثاني
أكسيد الكربون} فقد سُوّي --- إذ يُزفر عند \omterm{def:g7:lungs-and-gas-exchange:alveoli}{الأسناخ}. وهذا الفصل
الختامي يتتبع الفضلات \emph{الأخرى}، إلى أشدّ أعضاء الجسم تكتمًا:
\omterm{def:g7:removing-wastes:kidneys}{الكليتين}.

\section{الفضلات غير ثاني أكسيد الكربون}

\begin{proposition}[نفايات الخلايا الأخرى]\label{prop:g7:removing-wastes:rubbish}
احتراق \omterm{def:g5:first-look-at-cells:cell}{الخلايا} وبناؤها
(\cref{prop:g7:what-is-respiration:power}،
\cref{prop:g6:matter-from-food:transform}) يعطيان أكثر من \omterm{def:g7:what-is-respiration:respiration}{ثاني
أكسيد الكربون}. فاستعمال \omterm{def:g7:digestion-and-nutrients:families}{مغذيات} مادة البناء خاصة يترك فضلة تحمل
الآزوت --- هي \emph{البولينا}\index{بولينا} --- يجمعها \omterm{prop:g4:journey-of-food:absorb}{الدم} من كل
نسيج كما يجمع \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون} تمامًا. \omterm{prop:g7:removing-wastes:rubbish}{والبولينا} غير مؤذية في
حركة المرور العابرة وسامّة إذا تكوّمت: فعليها أن تغادر باستمرار.
\end{proposition}
\admitted

\begin{example}[جولتا جمع ودم واحد]\label{ex:g7:removing-wastes:rounds}
القطرة نفسها في \cref{ex:g7:heart-and-circulation:tour} تجري
الجمعين معًا: فعند شعيرات الفخذ تأخذ \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون}
\emph{\omterm{prop:g7:removing-wastes:rubbish}{والبولينا}}. وتُنزل الأول عند \omterm{def:g7:how-animals-breathe:four}{الرئتين} في الدارة التالية. أما
الثانية فتركب ماضيةً --- متجاوزةً \omterm{def:g7:how-animals-breathe:four}{الرئتين}، وهما لا تستطيعان مسّها
--- حتى تحملها الدارة الجسمية عبر الأعضاء المبنية لهذه الحمولة
بعينها.
\end{example}

\section{الكليتان، مصفاتا الدم}

\begin{definition}[الكليتان والجهاز البولي]\label{def:g7:removing-wastes:kidneys}
\emph{الكليتان}\index{كلية} --- في حجم القبضة، على شكل الفاصولياء،
في أسفل الظهر --- مصفاتان لتنقية \omterm{prop:g4:journey-of-food:absorb}{الدم}. وتتلقى كل واحدة شريانًا
سخيًّا مباشرة من الخط الرئيس في الدارة الجسمية. وفي الداخل يُصفَّى
\omterm{prop:g4:journey-of-food:absorb}{الدم} عبر نحو مليون وحدة ترشيح مجهرية؛ فيعود \omterm{prop:g4:journey-of-food:absorb}{الدم} النقي \omterm{def:g7:heart-and-circulation:vessels}{بوريد}
الكلية، ويغادر \omterm{def:g5:human-reproduction:pregnancy}{الحمل} المزال --- بولينا، وماء فائض، وأملاح فائضة
--- \emph{بولًا}\index{بول}، ينزّ في أنبوبين إلى
\emph{المثانة}\index{مثانة}، وهي خزان الحفظ يُفرَّغ عند الإرادة.
\end{definition}

\begin{proposition}[صفِّ ثم استردّ]\label{prop:g7:removing-wastes:filter}
حيلة \omterm{def:g7:removing-wastes:kidneys}{الكلية} سخاء يتبعه اقتصاد. فوحدات الترشيح تدفع أولًا حجمًا
هائلًا من الجزء المائي في \omterm{prop:g4:journey-of-food:absorb}{الدم} --- عشرات اللترات في اليوم --- عبر
مناخلها الدقيقة، \omterm{prop:g7:removing-wastes:rubbish}{بالبولينا} والسكريات والأملاح وكل شيء. ثم، على
امتداد الأنبوب المتعرج في كل وحدة، \emph{يستردّ} الجسم ما يريد
استعادته: كل الماء تقريبًا، وكل \omterm{prop:g7:organs-at-work:bill}{الغلوكوز}، والأملاح بقدر --- تاركًا
\omterm{prop:g7:removing-wastes:rubbish}{البولينا} والفوائض وراءه. ومن عشرات اللترات المنخولة يغادر نحو لتر
ونصف بولًا: صفِّ كل شيء، واسترد الجيد، فلا يبقى للفضلة مخبأ.
\end{proposition}
\admitted

\begin{example}[قراءة الاقتصاد]\label{ex:g7:removing-wastes:thrift}
طور الاسترداد يشرح الملاحظات اليومية. فاشرب كثيرًا تستردّ \omterm{def:g7:removing-wastes:kidneys}{الكليتان}
ماءً أقل: \omterm{def:g7:removing-wastes:kidneys}{فالبول} وفير شاحب. وتعرّق في مباراة حارة حتى تجفّ
تستردّان كل الماء تقريبًا: \omterm{def:g7:removing-wastes:kidneys}{فالبول} شحيح داكن --- والجسم يحرس ماءه
(وقد علّمت \cref{prop:g4:what-plants-need:needs} قيمة الماء
\omterm{def:g1:plants-around-us:plant}{للنباتات}؛ \omterm{def:g7:removing-wastes:kidneys}{والكلية} هي الوجه الحيواني من الاقتصاد نفسه). أما
\omterm{prop:g7:organs-at-work:bill}{الغلوكوز} في \omterm{def:g7:removing-wastes:kidneys}{البول}، وهو ما يمرّره المنخل وينبغي للاسترداد أن
يستعيده كله، فإشارة طبية كلاسيكية على أن شيئًا ما --- الاسترداد أو
سكر \omterm{prop:g4:journey-of-food:absorb}{الدم} --- ليس على ما يرام.
\end{example}

\begin{remark}[الجلد يمدّ يد العون]\label{rem:g7:removing-wastes:sweat}
\emph{العرق}\index{عرق} فرع من العمل نفسه: ماء وأملاح (مع أثر من
\omterm{prop:g7:removing-wastes:rubbish}{البولينا}) تُطلق عند \omterm{def:g1:the-five-senses:sense}{الجلد} --- وإن كان عمله الأول هو التبريد في
\cref{ex:g7:organs-at-work:felt}. \omterm{def:g7:how-animals-breathe:four}{والرئتان} \omterm{def:g7:removing-wastes:kidneys}{والكليتان} \omterm{def:g1:the-five-senses:sense}{والجلد}
تقتسم إزالة فضلات الجسم كلها: \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون} للأولى، \omterm{prop:g7:removing-wastes:rubbish}{والبولينا}
والفوائض للثانية، والحرارة ونزّ مالح للثالث. فلا يبقى شيء سامّ
يتكوّم --- وفي الجسم السليم لا تنتصر النفايات أبدًا.
\end{remark}

\section{آلة السنة كاملةً}

\begin{proposition}[حلقة التموين والإزالة الكاملة]\label{prop:g7:removing-wastes:complete}
بوجود \omterm{def:g7:removing-wastes:kidneys}{الكليتين} تنغلق جولة هذه السنة في آلة واحدة. \omterm{prop:g4:journey-of-food:absorb}{فالدم}، مضخوخًا في
الدارة المزدوجة
(\cref{def:g7:heart-and-circulation:double})، يحمّل \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} عند
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{الأسناخ} \omterm{def:g7:digestion-and-nutrients:families}{والمغذيات} عند \omterm{prop:g7:digestion-and-nutrients:absorption}{الزغابات}؛ ويسلّم الاثنين إلى كل \omterm{def:g5:first-look-at-cells:cell}{خلية} عاملة
للاحتراق والبناء؛ ويجمع الفضلات --- \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون} إلى
\omterm{def:g7:how-animals-breathe:four}{الرئتين}، \omterm{prop:g7:removing-wastes:rubbish}{والبولينا} والفوائض إلى \omterm{def:g7:removing-wastes:kidneys}{الكليتين}. وكل عضو من أعضاء الصفين
الرابع والسابع محطة من هذه الحلقة الواحدة؛ وكل تبادل يعبر جدار
شعيرة؛ والكل يجري على النار البطيئة في
\cref{prop:g7:what-is-respiration:power}، خليةً \omterm{def:g5:first-look-at-cells:cell}{خلية}.
\end{proposition}
\admitted

\begin{method}[تدقيق الجسم كله]\label{met:g7:removing-wastes:audit}
لأي مادة، دقّق في مرورها:
\begin{enumerate}
  \item الدخول: الفم \omterm{prop:g7:digestion-and-nutrients:absorption}{والزغابات}، أو الأنف \omterm{def:g7:lungs-and-gas-exchange:alveoli}{والأسناخ}؛
  \item والنقل: أي أشواط الدارتين، وأي محطات خاصة (\omterm{prop:g7:digestion-and-nutrients:absorption}{الكبد})؛
  \item والاستعمال: وقود يُحرَق، أو مادة تُبنى، أو لا هذا ولا
        ذاك؛
  \item والخروج: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{الأسناخ} (غازًا)، \omterm{def:g7:removing-wastes:kidneys}{والكليتان} (بولينا وفوائض)،
        \omterm{def:g1:the-five-senses:sense}{والجلد} (حصة \omterm{rem:g7:removing-wastes:sweat}{العرق}) --- أو، إن لم تُمتصّ البتة، طريق
        البقايا (\cref{prop:g4:journey-of-food:absorb}).
\end{enumerate}
أجرِ ذلك على الماء، وعلى \omterm{prop:g7:organs-at-work:bill}{الغلوكوز}، وعلى رقاقة مالحة: فللآلة محطة
لكل خطوة، وهذا ما وجدته السنة في تمرين واحد.
\end{method}

\section{مسألة: يوم في حياة جزيء ماء}

\begin{problem}[{مسألة نهاية الأسبوع --- كوب ماء واحد، مدقَّقًا
عبر الآلة كلها}]\label{pb:g7:removing-wastes:1}
تتبّع كوب ماء يُشرب على الفطور في يوم رياضي، بوصفه تدقيق السنة
الكبير.

\medskip\noindent\textbf{القسم الأول --- الدخول والنقل.}
\begin{enumerate}
  \item الماء لا يحتاج إلى تفكيك
        (\cref{def:g7:digestion-and-nutrients:families}).
        سمِّ عبوره عند الدخول وشبكة \omterm{def:g4:heart-and-blood:vessels}{الأوعية} التي تستقبله.
  \item ويقوم دمه بالمحطة الأولى المعتادة. فأي عضو، ولماذا كان
        المسار معقولًا \omterm{def:g7:digestion-and-nutrients:families}{للمغذيات} وإن كان الماء نفسه لا يحتاج إلى
        فرز؟
  \item وفي وسط الصباح يكون الماء في كل مكان يبلغه \omterm{prop:g4:journey-of-food:absorb}{الدم}. سمِّ
        الدارتين اللتين يركبهما الآن بالتناوب.
  \item وبعضه يصير سريعًا داخل \omterm{def:g5:first-look-at-cells:cell}{خلايا} عاملة. عدّد عملين يؤديهما
        الماء هناك (وتذكّر أن كيمياء \omterm{def:g5:first-look-at-cells:cell}{الخلايا} --- العصارات
        والاحتراق --- تجري رطبةً).
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- مباراة بعد
الظهر.}
\begin{enumerate}[resume]
  \item في أثناء المباراة يغادر جزء من الماء عبر حدّ ليس عضو
        إخراج في المقام الأول. سمِّه، وحمولته غير الماء، وعمله
        الحقيقي في ذلك اليوم.
  \item وبعد انتهاء المباراة تعطش اللاعبة ويكون بولها في المساء
        شحيحًا داكنًا. اشرح الأمرين بطور الاسترداد.
  \item وفي تلك الأثناء صنعت \omterm{def:g3:bones-and-muscles:muscle}{عضلات} المباراة الكادحة ثاني أكسيد
        كربون وبولينا زائدين. أعطِ كل واحد حدّ خروجه، وأشواط
        الدارة التي حملته إليه.
  \item ولماذا واصلت \omterm{def:g7:removing-wastes:kidneys}{الكليتان} العمل طول المباراة، مع أن إنتاج
        \omterm{def:g7:removing-wastes:kidneys}{البول} انخفض؟ وما الذي كانتا ما زالتا تزيلانه؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- إغلاق التدقيق.}
\begin{enumerate}[resume]
  \item في صباح اليوم التالي يكون بعض الكوب قد غادر في النفَس.
        اشرح كيف يخرج الماء عند \omterm{def:g7:lungs-and-gas-exchange:alveoli}{الأسناخ}
        (وقد أظهرت \cref{ex:g4:breathing:mirror} ذلك منذ
        سنوات).
  \item حرّر دفتر الكوب الكامل: الدخول، والمخازن والاستعمالات،
        ومخارجه الأربعة --- \omterm{def:g7:removing-wastes:kidneys}{البول}، \omterm{rem:g7:removing-wastes:sweat}{والعرق}، والنفَس، والحصة
        الصغيرة في البقايا.
  \item يبين التدقيق أن دمًا واحدًا يخدم \omterm{def:g7:how-animals-breathe:four}{الرئتين} \omterm{prop:g7:digestion-and-nutrients:absorption}{والزغابات}
        \omterm{def:g3:bones-and-muscles:muscle}{والعضلات} \omterm{def:g7:removing-wastes:kidneys}{والكليتين} \omterm{def:g1:the-five-senses:sense}{والجلد} بالتناوب. اذكر، في جملتين،
        لماذا يجعل تصميم الشبكة الواحدة هذا صحةَ كل عضو شأنًا
        لكل عضو آخر.
  \item اختم السنة: أي فكرة واحدة --- سمِّها --- تبين أن كل محطة
        في هذا التدقيق تخدمها؟ (والجواب في
        \cref{prop:g7:what-is-respiration:power}.)
\end{enumerate}
