\chapter{Classification et parenté}\label{ch:g6:classification-kinship}

Pourquoi la \omterm{def:g4:classifying-animals:classification}{classification} de
\cref{ch:g4:classifying-animals} fonctionne-t-elle si bien~? Trie
les \omterm{def:g1:animals-around-us:animal}{animaux} d'après les \omterm{def:g1:animals-around-us:covering}{plumes} et les poils, et mille autres
caractères se rangent docilement~: les nourrisseurs au \omterm{def:g2:animals-grow-up:born}{lait} ont une
\omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale}, les porteurs de \omterm{def:g1:animals-around-us:covering}{plumes} pondent des \omterm{def:g2:animals-grow-up:born}{œufs} à
\omterm{def:g1:animals-around-us:covering}{coquille}, et personne n'a besoin de vérifier.
\Cref{rem:g4:classifying-animals:tree} laissait la question en
suspens. Ce chapitre y répond en un mot --- la \omterm{prop:g6:classification-kinship:kinship}{parenté} --- et
rebâtit les boîtes en \omterm{def:g6:classification-kinship:tree}{arbre} généalogique.

\section{Des attributs, partagés et emboîtés}

\begin{definition}[Attribut]\label{def:g6:classification-kinship:attribute}
Un \emph{attribut}\index{attribut} est un caractère qu'un organisme
\emph{possède}, énoncé de façon vérifiable~: possède un \omterm{def:g3:bones-and-muscles:skeleton}{squelette}
avec une \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale}~; possède quatre \omterm{def:g1:my-body:parts}{membres}~; possède des
poils et allaite~; possède des \omterm{def:g1:animals-around-us:covering}{plumes}. La \omterm{def:g4:classifying-animals:classification}{classification} est bâtie
sur des attributs --- les bons \omterm{def:g3:sorting-living-things:criterion}{critères} de
\cref{def:g3:sorting-living-things:criterion}, mis en forme.
\end{definition}

\begin{proposition}[Les attributs s'emboîtent]\label{prop:g6:classification-kinship:nest}
Vérifiés sur de nombreuses \omterm{def:g5:biodiversity:species}{espèces}, les \omterm{def:g6:classification-kinship:attribute}{attributs} se répartissent en
ensembles \emph{emboîtés} --- tout \omterm{def:g1:animals-around-us:animal}{animal} à poils a aussi quatre
\omterm{def:g1:my-body:parts}{membres}~; tout \omterm{def:g1:animals-around-us:animal}{animal} à quatre \omterm{def:g1:my-body:parts}{membres} a aussi une \omterm{def:g4:classifying-animals:vertebrate}{colonne
vertébrale}~; jamais le motif inverse (des poils sans colonne) et
jamais des ensembles qui se croisent (poils et \omterm{def:g1:animals-around-us:covering}{plumes} ensemble). Cet
emboîtement est la raison pour laquelle les groupes dans des groupes
(\cref{def:g4:classifying-animals:classification}) conviennent au
monde vivant.
\end{proposition}
\admitted

\begin{example}[Placer trois animaux]\label{ex:g6:classification-kinship:placing}
Vérifie les boîtes pour la baleine, la truite et le merle. Baleine~:
colonne oui, quatre \omterm{def:g1:my-body:parts}{membres} oui (des nageoires avant, et des
vestiges cachés à l'arrière), poils et \omterm{def:g2:animals-grow-up:born}{lait} oui --- la boîte la plus
intérieure, avec le chat. Truite~: colonne oui, stop --- la boîte
extérieure seulement. Merle~: colonne, quatre \omterm{def:g1:my-body:parts}{membres} (deux pattes,
deux ailes), \omterm{def:g1:animals-around-us:covering}{plumes} --- la boîte des quatre \omterm{def:g1:my-body:parts}{membres}, dans le
compartiment propre aux \omterm{def:g1:animals-around-us:covering}{plumes}. Aucun \omterm{def:g1:animals-around-us:animal}{animal} n'exige jamais deux
boîtes qui se croisent.
\end{example}

\section{La parenté~: la raison de l'emboîtement}

\begin{proposition}[Des attributs partagés signifient une ascendance commune]\label{prop:g6:classification-kinship:kinship}
L'explication de l'emboîtement est l'hérédité
(\cref{rem:g4:animal-reproduction:mix}~: les jeunes reçoivent de
leurs parents). Les \omterm{def:g5:biodiversity:species}{espèces} qui partagent un \omterm{def:g6:classification-kinship:attribute}{attribut} le partagent
parce qu'elles l'ont \emph{hérité d'un ancêtre commun} --- une
\omterm{def:g5:biodiversity:species}{espèce} grand-parente lointaine. Plus deux \omterm{def:g5:biodiversity:species}{espèces} partagent
d'\omterm{def:g6:classification-kinship:attribute}{attributs}, plus leur \emph{parenté}\index{parenté} est proche~: le
chat et la baleine, tous deux velus et nourrisseurs au \omterm{def:g2:animals-grow-up:born}{lait}, sont
des cousins plus proches l'un de l'autre que ne l'est chacun de la
truite. Un groupe de la \omterm{def:g4:classifying-animals:classification}{classification} est, au fond, une
\emph{famille}~: un ancêtre et tous ses descendants.
\end{proposition}
\admitted

\begin{example}[Lire enfin correctement la baleine]\label{ex:g6:classification-kinship:whale}
Voilà pourquoi l'énigme de la baleine
(\cref{exo:g4:classifying-animals:9}) devait se résoudre comme elle
l'a fait~: les poils et le \omterm{def:g2:animals-grow-up:born}{lait} marquent la lignée familiale de la
baleine --- ses ancêtres étaient des \omterm{prop:g3:sorting-living-things:groups}{mammifères} terrestres qui
allaitaient --- tandis que sa forme de poisson est l'ouvrage de
l'eau (\cref{exo:g4:adapted-to-their-world:11}), imprimée sur une
famille étrangère par le même mode de vie. Les \omterm{def:g6:classification-kinship:attribute}{attributs}
d'ascendance trient les familles~; les ressemblances de milieu
peuvent être empruntées par des étrangers. La \omterm{def:g4:classifying-animals:classification}{classification} se fie
aux premiers et se méfie des secondes.
\end{example}

\begin{definition}[L'arbre de parenté]\label{def:g6:classification-kinship:tree}
La \omterm{prop:g6:classification-kinship:kinship}{parenté} se dessine en \emph{arbre}\index{arbre de parenté}~: les
points de branchement représentent des ancêtres communs, et chaque
\omterm{def:g6:classification-kinship:attribute}{attribut} n'apparaît qu'une fois, à l'embranchement où il est
apparu --- toute \omterm{def:g5:biodiversity:species}{espèce} au-delà de cette fourche en hérite. Les
boîtes emboîtées et l'arbre portent la même information~: une boîte
est une \omterm{def:g1:plants-around-us:tree}{branche} avec tous ses rameaux.
\end{definition}

\begin{method}[Construire un petit arbre de parenté]\label{met:g6:classification-kinship:build}
Devant une poignée d'\omterm{def:g5:biodiversity:species}{espèces}~:
\begin{enumerate}
  \item dresse le tableau de leurs \omterm{def:g6:classification-kinship:attribute}{attributs} --- les \omterm{def:g5:biodiversity:species}{espèces} en
        lignes, les \omterm{def:g6:classification-kinship:attribute}{attributs} vérifiables en colonnes, une croix là
        où l'\omterm{def:g6:classification-kinship:attribute}{attribut} est possédé~;
  \item regroupe par croix partagées, l'\omterm{def:g6:classification-kinship:attribute}{attribut} le plus large
        d'abord~: l'emboîtement apparaît dans le tableau~;
  \item dessine l'\omterm{def:g6:classification-kinship:tree}{arbre}~: une fourche par \omterm{def:g6:classification-kinship:attribute}{attribut}, la plus large
        au plus près du tronc~; place chaque \omterm{def:g5:biodiversity:species}{espèce} au-delà
        exactement des fourches dont elle possède les \omterm{def:g6:classification-kinship:attribute}{attributs}~;
  \item lis la \omterm{prop:g6:classification-kinship:kinship}{parenté}~: la proximité de deux \omterm{def:g5:biodiversity:species}{espèces} est la
        dernière fourche qu'elles partagent.
\end{enumerate}
\end{method}

\begin{remark}[Ce que l'arbre laisse entendre]\label{rem:g6:classification-kinship:implies}
Prends cet \omterm{def:g6:classification-kinship:tree}{arbre} au sérieux et il dit quelque chose d'énorme~: suis
deux \omterm{def:g1:plants-around-us:tree}{branches} quelconques vers l'arrière et elles \emph{se
rejoignent}. Le chat et la truite ont une \omterm{def:g5:biodiversity:species}{espèce} grand-parente
commune~; il doit donc en aller de même, plus loin en arrière, du
chat et de l'escargot, de la pâquerette et de l'enfant. Tous les
\omterm{def:g6:exploring-our-environment:components}{êtres vivants} formeraient alors une seule famille, dont l'unité
(\cref{prop:g5:first-look-at-cells:principle}) serait un air de
famille --- et les \omterm{def:g5:biodiversity:species}{espèces} doivent pouvoir \emph{changer} au fil des
générations, sans quoi le branchement n'aurait jamais pu avoir lieu.
Savoir si c'est vrai, et comment cela pourrait fonctionner, est la
grande question de l'année de \cref{ch:g9:evidence-of-evolution} ---
cet \omterm{def:g6:classification-kinship:tree}{arbre} est le schéma qui l'impose.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g6:classification-kinship:1}
Qu'est-ce qu'un \omterm{def:g6:classification-kinship:attribute}{attribut}~? Donnes-en trois, énoncés de façon
vérifiable.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:classification-kinship:2}
Que veut dire que les \omterm{def:g6:classification-kinship:attribute}{attributs} \emph{s'emboîtent}~? Donne l'exemple
des trois boîtes du chapitre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:classification-kinship:3}
Pourquoi des \omterm{def:g5:biodiversity:species}{espèces} partagent-elles des \omterm{def:g6:classification-kinship:attribute}{attributs}, d'après
\cref{prop:g6:classification-kinship:kinship}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:classification-kinship:4}
Sur l'\omterm{def:g6:classification-kinship:tree}{arbre} de poche, au-delà de quel point d'\omterm{def:g6:classification-kinship:attribute}{attribut} se tient la
truite~? Et au-delà de quels points se tient la baleine~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:classification-kinship:5}
Qui sont les cousins les plus proches~: la baleine et le chat, ou la
baleine et la truite~? Lis la réponse sur les fourches de l'\omterm{def:g6:classification-kinship:tree}{arbre}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:classification-kinship:6}
Énonce la relation entre une boîte emboîtée et une \omterm{def:g1:plants-around-us:tree}{branche} de
l'\omterm{def:g6:classification-kinship:tree}{arbre}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:classification-kinship:7}
Pourquoi la \omterm{def:g4:classifying-animals:classification}{classification} se fie-t-elle aux poils et au \omterm{def:g2:animals-grow-up:born}{lait} mais
se méfie-t-elle de «~a la forme d'un poisson~»~? Sers-toi de
\cref{ex:g6:classification-kinship:whale}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:classification-kinship:8}
Construis le tableau d'\omterm{def:g6:classification-kinship:attribute}{attributs} de
\cref{met:g6:classification-kinship:build} pour~: la truite, la
grenouille, le chat, le merle --- \omterm{def:g6:classification-kinship:attribute}{attributs}~: \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale},
quatre \omterm{def:g1:my-body:parts}{membres}, poils et \omterm{def:g2:animals-grow-up:born}{lait}, \omterm{def:g1:animals-around-us:covering}{plumes}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:classification-kinship:9}
Ajoute le lézard à l'\omterm{def:g6:classification-kinship:tree}{arbre} de poche~: quelles fourches
franchit-il, et où sa \omterm{def:g1:plants-around-us:tree}{branche} part-elle~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:classification-kinship:10}
La chauve-souris vole comme un merle mais allaite comme un chat.
Quelle ressemblance marque la \omterm{prop:g6:classification-kinship:kinship}{parenté} et laquelle est empruntée au
mode de vie~? Décide avec la logique de l'\omterm{def:g6:classification-kinship:tree}{arbre}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:classification-kinship:11}
«~Un groupe est un ancêtre et tous ses descendants.~» Éprouve la
phrase sur les \omterm{prop:g3:sorting-living-things:groups}{mammifères} et sur l'ornithorynque de
\cref{exo:g4:classifying-animals:11}~: cette bizarrerie pondeuse
d'\omterm{def:g2:animals-grow-up:born}{œufs} brise-t-elle la famille, ou lui appartient-elle~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:classification-kinship:12}
Pousse \cref{rem:g6:classification-kinship:implies} jusqu'au bout~:
que doit-il être vrai des \omterm{def:g5:biodiversity:species}{espèces} sur de longues durées, si chaque
paire de \omterm{def:g1:plants-around-us:tree}{branches} se rejoint vraiment~? Énonce l'implication en une
phrase --- et nomme le genre de preuves que tu exigerais.
\end{exercise}

\section{Problème~: le tiroir du muséum}

\begin{problem}[{Problème du week-end --- six spécimens, un tableau
d'attributs, un arbre}]\label{pb:g6:classification-kinship:1}
Un muséum prête à la classe un tiroir de six spécimens~: une truite,
une grenouille, un lézard, un pigeon, une souris, et un \omterm{def:g3:bones-and-muscles:skeleton}{crâne} de
dauphin avec des moulages de nageoires. La tâche~: les classer par
\omterm{def:g6:classification-kinship:attribute}{attributs} et dessiner leur \omterm{def:g6:classification-kinship:tree}{arbre de parenté}.

\medskip\noindent\textbf{Partie I --- Le tableau.}
\begin{enumerate}
  \item Construis le tableau des six, avec les \omterm{def:g6:classification-kinship:attribute}{attributs}~: \omterm{def:g4:classifying-animals:vertebrate}{colonne
        vertébrale}~; quatre \omterm{def:g1:my-body:parts}{membres}~; \omterm{def:g1:the-five-senses:sense}{peau} sèche et écailleuse~;
        \omterm{def:g1:animals-around-us:covering}{plumes}~; poils et \omterm{def:g2:animals-grow-up:born}{lait}. (Pour le dauphin, presque sans
        poils, l'étiquette du muséum l'atteste~: petits nourris au
        \omterm{def:g2:animals-grow-up:born}{lait} --- \cref{ex:g3:sorting-living-things:surprises}
        explique pourquoi ce témoignage tranche.)
  \item Quel \omterm{def:g6:classification-kinship:attribute}{attribut} les six partagent-ils tous~? Que signifie le
        partage le plus large, en termes de \omterm{prop:g6:classification-kinship:kinship}{parenté}~?
  \item Quels spécimens cochent «~quatre \omterm{def:g1:my-body:parts}{membres}~»~? Explique les
        croix du pigeon et du dauphin --- que sont des ailes et des
        nageoires, selon la logique de l'\omterm{def:g6:classification-kinship:tree}{arbre}~?
  \item Pourquoi «~vit dans l'eau~» n'apparaît-il nulle part dans
        le tableau, alors que la truite et le dauphin y vivent tous
        deux~? Cite la règle du chapitre.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- L'\omterm{def:g6:classification-kinship:tree}{arbre}.}
\begin{enumerate}[resume]
  \item Dessine l'\omterm{def:g6:classification-kinship:tree}{arbre}~: ordonne les fourches --- \omterm{def:g4:classifying-animals:vertebrate}{colonne
        vertébrale}, quatre \omterm{def:g1:my-body:parts}{membres}, puis les revêtements (\omterm{def:g1:animals-around-us:covering}{écailles},
        \omterm{def:g1:animals-around-us:covering}{plumes}, poils) sur leurs propres \omterm{def:g1:plants-around-us:tree}{branches} --- et place les
        six.
  \item Lis sur ton \omterm{def:g6:classification-kinship:tree}{arbre}~: le cousin le plus proche de la souris
        dans le tiroir, et celui de la truite.
  \item La grenouille se tient au-delà de «~quatre \omterm{def:g1:my-body:parts}{membres}~» mais
        avant toute fourche de revêtement. Comme quoi sa \omterm{def:g1:the-five-senses:sense}{peau} nue
        et humide (\cref{rem:g4:classifying-animals:confusion})
        la désigne-t-elle, et où sa \omterm{def:g1:plants-around-us:tree}{branche} part-elle~?
  \item Une étiquette tombe d'un septième spécimen~: \omterm{def:g1:the-five-senses:sense}{peau} sèche et
        écailleuse, quatre \omterm{def:g1:my-body:parts}{membres}, \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale}. Place-le
        sur l'\omterm{def:g6:classification-kinship:tree}{arbre} et nomme sa classe --- sans voir l'étiquette.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le sens.}
\begin{enumerate}[resume]
  \item Le dauphin et la truite partagent un corps fuselé~; le
        dauphin et la souris partagent une ascendance à poils et à
        \omterm{def:g2:animals-grow-up:born}{lait}. Quelle paire se tient la plus proche sur l'\omterm{def:g6:classification-kinship:tree}{arbre}, et
        pourquoi l'\omterm{def:g6:classification-kinship:tree}{arbre} doit-il ignorer le corps fuselé~?
  \item Énonce ce que la fourche «~quatre \omterm{def:g1:my-body:parts}{membres}~» affirme
        historiquement~: de quelle chose unique, il y a très
        longtemps, la grenouille, le lézard, le pigeon, la souris
        et le dauphin descendent-ils tous~?
  \item L'\omterm{def:g6:classification-kinship:tree}{arbre} de la classe fait finalement se rejoindre toutes
        les \omterm{def:g1:plants-around-us:tree}{branches} au tronc. Dis en une phrase ce que ce dessin
        affirme discrètement des six spécimens --- et des élèves
        qui le dessinent.
  \item Une sceptique dit~: «~votre \omterm{def:g6:classification-kinship:tree}{arbre} n'est qu'un système de
        classement~». Donne la réponse en deux parties~: sur quoi
        l'emboîtement des \omterm{def:g6:classification-kinship:attribute}{attributs} est un \emph{fait}, et ce que
        l'\omterm{def:g6:classification-kinship:tree}{arbre} \emph{explique} qu'un simple classement laisserait
        inexpliqué.
\end{enumerate}
\end{problem}
