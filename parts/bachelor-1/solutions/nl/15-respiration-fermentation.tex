\section*{Hoofdstuk \ref{ch:b1:respiration-fermentation} --- Celademhaling en gisting}

\begin{solution}{exo:b1:respiration-fermentation:1}
\omterm{def:b1:respiration-fermentation:glycolysis}{Glycolyse}, \omterm{def:b1:eukaryotic-cell:organelle}{cytosol}: 2 pyruvaat, 2 \omterm{def:b1:water-small-molecules:atp}{ATP}, 2 NADH. Pyruvaatoxidatie en \omterm{def:b1:respiration-fermentation:krebs}{krebscyclus},
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondriale matrix}: 6 $\mathrm{CO_2}$, 8 NADH, 2 $\mathrm{FADH_2}$, 2 GTP.
Oxidatieve fosforylering, binnenmembraan: water uit zuurstof, ongeveer 26 \omterm{def:b1:water-small-molecules:atp}{ATP}.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:2}
Glucose $+ 2\,\mathrm{NAD^+} + 2\,\mathrm{ADP} + 2\,\mathrm{P_i} \to
2$ pyruvaat $+ 2\,\mathrm{NADH} + 2\,\mathrm{H^+} + 2\,\mathrm{ATP} +
2\,\mathrm{H_2O}$. \omterm{def:b1:respiration-fermentation:glycolysis}{Substraatfosforylering}: een fosfaat wordt door een \omterm{def:b1:enzymes:enzyme}{enzym}
rechtstreeks van een energierijk tussenproduct (1,3-bisfosfoglyceraat,
fosfo-enolpyruvaat) aan ADP overgedragen, zonder membraangradiënt of zuurstof.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:3}
NADH en $\mathrm{CO_2}$ (isocitraat naar $\alpha$-ketoglutaraat), NADH en
$\mathrm{CO_2}$ ($\alpha$-ketoglutaraat naar succinyl-CoA), GTP (succinyl-CoA naar
succinaat), $\mathrm{FADH_2}$ (succinaat naar fumaraat), NADH (malaat naar
oxaalacetaat). De twee $\mathrm{CO_2}$ komen van de twee decarboxyleringen van de
zuren met zes en met vijf koolstoffen.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:4}
Zij herstelt $\mathrm{NAD^+}$ zodat de \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} door kan gaan en haar 2 \omterm{def:b1:water-small-molecules:atp}{ATP} per
glucose kan leveren. Zij oxideert de koolstof niet, haalt de resterende energie er
niet uit (het product houdt haar vrijwel volledig) en maakt niet meer dan een
vijftiende van het \omterm{def:b1:water-small-molecules:atp}{ATP} van de ademhaling.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:5}
Naar zuurstof: $-2\times 96\,500\times 1.14 = \qty{-220}{kJ/mol}$, vier \omterm{def:b1:water-small-molecules:atp}{ATP} van
\qty{50}{kJ}. Naar ubichinon: $-2\times 96\,500\times 0.37 =
\qty{-71}{kJ/mol}$, één \omterm{def:b1:water-small-molecules:atp}{ATP}.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:6}
Spier: $2 + 2$ (GTP) $+ 8\times 2.5$ (mitochondriaal NADH) $+ 2\times
1.5$ ($\mathrm{FADH_2}$) $+ 2\times 1.5$ (cytosolisch NADH via \omterm{def:b1:respiration-fermentation:carriers}{FAD}) $=
30$. Lever: het cytosolische NADH geeft $2\times 2.5$: 32.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:7}
RQ $= 2.2$. Verademde glucose $x$: $6x = 1.0$, $x = \qty{0.167}{mmol}$;
$\mathrm{CO_2}$: $6x + 2y = 2.2$, $2y = 1.2$, $y = \qty{0.6}{mmol}$. Vergist wordt
$0.6/0.767 = \qty{78}{\%}$ van de glucose, verademd \qty{22}{\%}.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:8}
Het zuurstofverbruik stijgt (de keten loopt, bevrijd van de tegendruk van de
gradiënt, op volle snelheid), de ATP-synthese stopt (er is geen gradiënt om het
synthase aan te drijven), en alle energie van de elektronen verschijnt als warmte.
Patiënten verbrandden hun \omterm{def:b1:lipids:triglyceride}{vet} snel en stierven aan een lichaamstemperatuur die
niets omlaag kon brengen.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:9}
De keten loopt vast: elke drager raakt gereduceerd, er worden geen protonen gepompt
en er wordt geen \omterm{def:b1:water-small-molecules:atp}{ATP} gemaakt. NADH wordt niet weer geoxideerd, zodat de \omterm{def:b1:respiration-fermentation:krebs}{krebscyclus}
en het pyruvaatdehydrogenase stilvallen bij gebrek aan $\mathrm{NAD^+}$. De
\omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse}, verlost van de remming door \omterm{def:b1:water-small-molecules:atp}{ATP}, loopt snel en moet haar
$\mathrm{NAD^+}$ herstellen door pyruvaat tot lactaat te reduceren, dat het bloed
overspoelt.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:10}
$8\times 10 + 7\times 2.5 + 7\times 1.5 - 2 = 80 + 17.5 + 10.5 - 2 =
106$ \omterm{def:b1:water-small-molecules:atp}{ATP}; $106/16 = 6.6$ per koolstof. Zuurstof: palmitaat 23 $\mathrm{O_2}$ voor
106 \omterm{def:b1:water-small-molecules:atp}{ATP}, \qty{0.22}{O_2} per \omterm{def:b1:water-small-molecules:atp}{ATP}; glucose 6 voor 30, \qty{0.20}{O_2} per \omterm{def:b1:water-small-molecules:atp}{ATP}. Per
eenheid zuurstof is glucose \qty{10}{\%} beter: de zeehond, onder water door
zuurstof begrensd, verkiest \omterm{def:b1:carbohydrates:monosaccharide}{koolhydraat}; per eenheid massa is \omterm{def:b1:lipids:triglyceride}{vet} twee keer zo
goed: de kolibrie, in de vlucht door massa begrensd, trekt op \omterm{def:b1:lipids:triglyceride}{vet}.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:11}
\omterm{def:b1:water-small-molecules:atp}{ATP} alleen: $5/3 < \qty{2}{s}$. Fosfocreatine: $25/3 = \qty{8}{s}$. \omterm{def:b1:respiration-fermentation:fermentation}{Gisting}:
$2\times 3 = \qty{6}{mmol/L}$ \omterm{def:b1:water-small-molecules:atp}{ATP} per seconde, genoeg voor de vraag, zolang het
\omterm{def:b1:carbohydrates:polysaccharide}{glycogeen} en de verdraagzaamheid voor lactaat het volhouden (tientallen seconden).
De zuurstoftoevoer heeft een minuut of meer nodig om te stijgen en levert in een
vezel hoogstens ongeveer \qty{1}{mmol/L} \omterm{def:b1:water-small-molecules:atp}{ATP} per seconde: een race van tien
seconden is voorbij voordat de ademhaling heeft kunnen helpen, en wordt door
fosfocreatine en \omterm{def:b1:respiration-fermentation:fermentation}{gisting} betaald.
\end{solution}

\begin{solution}{exo:b1:respiration-fermentation:12}
De gradiënt slaat energie op als een verschil in protonenconcentratie en in
elektrische potentiaal over een isolerende membraan --- precies een geladen
condensator plus een concentratiecel; de keten is de lader, aangedreven door de val
van de elektronen; het synthase is een motor die door de protonenstroom wordt
rondgedraaid, precies een turbine, die de stroming aan een mechanische draaiing
koppelt die \omterm{def:b1:water-small-molecules:atp}{ATP} maakt. De vergelijking zit los doordat de ``batterij'' maar
milliseconden van het verbruik van de \omterm{def:b1:cell-unit-of-life:cell}{cel} bevat en onafgebroken moet worden
opgeladen, doordat de overige transporteiwitten van diezelfde membraan er ook uit
putten, en doordat de turbine achteruit kan draaien en \omterm{def:b1:water-small-molecules:atp}{ATP} hydrolyseert om protonen
te pompen wanneer de keten faalt.
\end{solution}

\begin{solution}{pb:b1:respiration-fermentation:1}
\textbf{1.} \omterm{def:b1:respiration-fermentation:glycolysis}{Glycolyse}: 2 \omterm{def:b1:water-small-molecules:atp}{ATP}, 2 NADH (cytosolisch). Pyruvaatdehydrogenase: 2 NADH.
Krebs: 6 NADH, 2 $\mathrm{FADH_2}$, 2 GTP.
\textbf{2.} Mitochondriaal NADH $8\times 10 = 80$; $\mathrm{FADH_2}$
$2\times 6 = 12$; cytosolisch NADH als $\mathrm{FADH_2}$ $2\times 6 = 12$: 104
protonen.
\textbf{3.} $104/4 = 26$ \omterm{def:b1:water-small-molecules:atp}{ATP}; samen $26 + 2 + 2 = 30$.
\textbf{4.} 2 \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{5.} 15.
\textbf{6.} Met lucht $30\times 50 = \qty{1500}{kJ}$: \qty{52}{\%} van 2870.
Zonder: \qty{100}{kJ}, \qty{3.5}{\%} van 2870 maar \qty{43}{\%} van de 235 die
vrijkomen --- de \omterm{def:b1:respiration-fermentation:fermentation}{gisting} legt doeltreffend vast wat zij vrijmaakt; zij maakt alleen
weinig vrij.
\textbf{7.} Belucht $30/30 = \qty{1}{mmol}$ glucose per uur; afgesloten
$30/2 = \qty{15}{mmol}$.
\textbf{8.} Afgesloten: \qty{30}{mmol} ethanol en \qty{30}{mmol} $\mathrm{CO_2}$
per uur. Belucht: \qty{6}{mmol} $\mathrm{CO_2}$ afgegeven en \qty{6}{mmol}
$\mathrm{O_2}$ verbruikt.
\textbf{9.} $2\times 10.5 = \qty{21}{g}$ per mol, \qty{0.12}{g} per gram glucose.
\textbf{10.} $0.5\times 180 = \qty{90}{g}$ per mol, oftewel $90/10.5 =
8.6$ \omterm{def:b1:water-small-molecules:atp}{ATP} waard tegen 30 beschikbaar: \omterm{def:b1:water-small-molecules:atp}{ATP} is in overmaat; de opbrengst wordt
begrensd door koolstof (de helft van de glucose wordt tot $\mathrm{CO_2}$
geoxideerd om het \omterm{def:b1:water-small-molecules:atp}{ATP} te maken, en biomassa vraagt koolstofskeletten en
reducerende kracht), en het overtollige \omterm{def:b1:water-small-molecules:atp}{ATP} gaat naar het onderhoud.
\textbf{11.} Om hetzelfde \omterm{def:b1:water-small-molecules:atp}{ATP} te krijgen verbruikt de afgesloten kweek vijftien
keer meer glucose (vraag 7) en groeit zij per gram een vijfde zoveel (vraag 9):
snel suikerverbruik, weinig groei, alcohol --- de waarneming van Pasteur.
Fosfofructokinase, dat bij belucht kweken door \omterm{def:b1:water-small-molecules:atp}{ATP} en citraat wordt geremd, komt
vrij zodra de ademhaling stopt.
\textbf{12.} $\mathrm{CO_2}$ $44/44 = \qty{1.0}{mmol}$; $\mathrm{O_2}$
$20/32 = \qty{0.625}{mmol}$: RQ $= 1.6$.
\textbf{13.} $\mathrm{O_2}$: $6x = 0.625$, $x = \qty{0.104}{mmol}$.
$\mathrm{CO_2}$: $6x + 2y = 1.0$, $2y = 0.375$, $y = \qty{0.1875}{mmol}$.
\textbf{14.} Vergist wordt $0.1875/0.292 = \qty{64}{\%}$ van de glucose. Het \omterm{def:b1:water-small-molecules:atp}{ATP}
uit de \omterm{def:b1:respiration-fermentation:fermentation}{gisting} is $2y = 0.375$ tegen $30x = 3.12$ uit de ademhaling: \qty{11}{\%}.
\textbf{15.} $16/23 = 0.70$.
\textbf{16.} Nuchter verbrandt het dier \omterm{def:b1:lipids:triglyceride}{vet} (RQ 0,7); gevoed verbrandt het het
\omterm{def:b1:carbohydrates:monosaccharide}{koolhydraat} van de maaltijd (1,0).
\textbf{17.} $96\,500\times 0.22 = \qty{21.2}{kJ/mol}$.
\textbf{18.} $4\times 21.2 = \qty{85}{kJ}$ per \omterm{def:b1:water-small-molecules:atp}{ATP} van \qty{50}{kJ}: \qty{59}{\%}.
\textbf{19.} $2\times 96\,500\times 1.14 = \qty{220}{kJ}$; tien protonen slaan
$10\times 21.2 = \qty{212}{kJ}$ op: \qty{96}{\%} --- de keten verspilt weinig; de
verliezen zitten bij het synthase en de transporteiwitten.
\textbf{20.} $0.96\times 0.59 = 0.57$; $2.5\times 50/220 = 0.57$: langs beide wegen
hetzelfde cijfer.
\textbf{21.} Het zuurstofverbruik stijgt, de ATP-opbrengst daalt naar die van de
\omterm{def:b1:respiration-fermentation:fermentation}{gisting} (2 per glucose uit de \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse}, en PFK komt vrij, zodat het
glucoseverbruik stijgt), de warmteafgifte stijgt, de groei stort in.
\textbf{22.} De ATP-synthese stopt; de gradiënt kan zich niet ontladen, de keten
loopt ertegen vast, het zuurstofverbruik stopt, en de \omterm{def:b1:cell-unit-of-life:cell}{cel} vergist wat zij kan. Wordt
er dan een ontkoppelaar toegevoegd, dan lekken de protonen terug: de keten en het
zuurstofverbruik hervatten, nog altijd zonder \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{23.} NADH pompt alleen via de complexen III en IV: $6/4 = 1.5$; per
glucose $8\times 1.5 + 2\times 1.5 + 2\times 1.5 + 4 = 22$ \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{24.} Zonder zuurstof kan de keten geen elektronen opnemen, zodat NADH niet
weer wordt geoxideerd en de voorraad gereduceerd blijft; de drie NAD-afhankelijke
dehydrogenasen van de cyclus hebben geen $\mathrm{NAD^+}$ en de cyclus stopt ---
en juist daarom moet de \omterm{def:b1:respiration-fermentation:fermentation}{gisting} in het \omterm{def:b1:eukaryotic-cell:organelle}{cytosol} $\mathrm{NAD^+}$ herstellen opdat de
\omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} door kan gaan.
\textbf{25.} 30 \omterm{def:b1:water-small-molecules:atp}{ATP} met lucht, 2 zonder: een verhouding van 15; P/O 2,5 voor NADH
(10 protonen gedeeld door 4 per \omterm{def:b1:water-small-molecules:atp}{ATP}) en 1,5 voor $\mathrm{FADH_2}$ (6 gedeeld door
4).
\end{solution}
