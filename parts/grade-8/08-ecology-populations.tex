\chapter{Populations and Communities}\label{ch:g8:populations}

\omterm{def:g6:ecology:def}{Ecology} scales from single \omterm{def:g4:ecosystems:levels}{organisms} to the \omterm{def:g6:ecology:levels}{biosphere}. Here we focus on
\omterm{def:g8:populations:pop}{populations} and communities: how numbers change, and how \omterm{def:g6:classification:species}{species} interact.

\section{Population basics}

\begin{definition}[Population]\label{def:g8:populations:pop}
A \emph{population}\index{population} is a group of individuals of the
same \omterm{def:g6:classification:species}{species} living in the same area at the same time.
\end{definition}
\omimg{food-web.jpg}{0.50\linewidth}{Populations interact in food webs.}


\begin{definition}[Population size and density]\label{def:g8:populations:size}
\emph{Population size}\index{population size} is the number of
individuals. \emph{Population density}\index{population density} is size
per unit area (or volume).
\end{definition}

\begin{proposition}[What changes size]\label{prop:g8:populations:change}
Births and immigration increase a \omterm{def:g8:populations:pop}{population}; deaths and emigration
decrease it. Net change depends on all four.
\end{proposition}
\admitted

\begin{definition}[Limiting factors]\label{def:g8:populations:limit}
A \emph{limiting factor}\index{limiting factor} restricts \omterm{def:g8:populations:pop}{population}
\omterm{def:g2:growing:growth}{growth} (\omterm{def:g2:food-energy:food}{food}, \omterm{def:g4:plants-need:needs}{water}, space, disease, \omterm{def:g5:food-webs:pred}{predators}, weather extremes).
\emph{Carrying capacity}\index{carrying capacity} is the approximate
maximum \omterm{def:g8:populations:size}{population size} an \omterm{def:g4:traits:env}{environment} can sustain over time.
\end{definition}

\section{Communities and interactions}

\begin{definition}[Community]\label{def:g8:populations:community}
A \emph{community}\index{community (ecology)} is all the \omterm{def:g8:populations:pop}{populations} of
different \omterm{def:g6:classification:species}{species} living and interacting in an area.
\end{definition}

\begin{definition}[Key interactions]\label{def:g8:populations:interact}
\begin{itemize}
  \item \emph{Predation}\index{predation}: \omterm{def:g5:food-webs:pred}{predator} eats \omterm{def:g5:food-webs:pred}{prey}.
  \item \emph{Competition}\index{competition (ecology)}: \omterm{def:g6:classification:species}{species} (or
        individuals) seek the same limited resource.
  \item \emph{Mutualism}\index{mutualism}: both \omterm{def:g6:classification:species}{species} benefit (bees and
        \omterm{def:g2:plant-parts:main}{flowers}).
  \item \emph{Parasitism}\index{parasitism}: parasite benefits; host is
        harmed.
\end{itemize}
\end{definition}

\begin{example}[Predator--prey idea]\label{ex:g8:populations:pred}
If \omterm{def:g5:food-webs:pred}{prey} become scarce, \omterm{def:g5:food-webs:pred}{predator} numbers may later fall; if \omterm{def:g5:food-webs:pred}{predators} fall,
\omterm{def:g5:food-webs:pred}{prey} may rebound --- linked dynamics, not a simple seesaw every year.
\end{example}

\begin{method}[Describing a community]\label{met:g8:populations:describe}
\begin{enumerate}
  \item List major \omterm{def:g6:classification:species}{species} or groups.
  \item Note who eats whom (links).
  \item Identify one \omterm{def:g8:populations:interact}{mutualism} and one \omterm{def:g6:ecology:rel}{competition} if possible.
  \item Name a \omterm{def:g8:populations:limit}{limiting factor} for a key \omterm{def:g8:populations:pop}{population}.
\end{enumerate}
\end{method}
