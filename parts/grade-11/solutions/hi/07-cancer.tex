\section*{अध्याय \ref{ch:g11:cancer} --- जीनोम की क्षति और कैंसर}

\begin{solution}{exo:g11:cancer:1}
\omterm{def:g11:cancer:cancer}{कैंसर}: सामान्य नियंत्रणों के बिना विभाजित होतीं और पड़ोसी ऊतक में घुसतीं
\omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}। \omterm{def:g11:cancer:cancer}{अर्बुद}: ऐसे किसी क्लोन का बनाया हुआ पिंड। विक्षेप: रक्त या लसीका
से चलकर गई \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} का कहीं और बसाया हुआ नया \omterm{def:g11:cancer:cancer}{अर्बुद}।
\end{solution}

\begin{solution}{exo:g11:cancer:2}
वह \omterm{def:g10:universal-dna:gene}{जीन} जिसका \omterm{def:g11:gene-expression:protein}{प्रोटीन} वृद्धि के संकेत आगे ढोता है और \omterm{def:g10:cells-common-unit:cell}{कोशिका} को चक्र में
धकेलता है; और ऐसे उत्परिवर्तित होने पर कि वह \omterm{def:g11:gene-expression:protein}{प्रोटीन} हमेशा सक्रिय रहे, वह
ऑन्कोजीन बन जाता है। एक जाम हुआ गति-यंत्र \omterm{def:g10:cells-common-unit:cell}{कोशिका} को चलाता ही रहता है, चाहे
दूसरा \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} कुछ भी करे।
\end{solution}

\begin{solution}{exo:g11:cancer:3}
वह \omterm{def:g10:universal-dna:information}{DNA} क्षतिग्रस्त होने पर चक्र रोक देता है और मरम्मत या आत्म-विनाश शुरू
करा देता है। एक काम करता \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} भी ब्रेक लगाने भर \omterm{def:g11:gene-expression:protein}{प्रोटीन} बना देता है;
इसलिए ब्रेक केवल दोनों के खोने पर ही हटता है।
\end{solution}

\begin{solution}{exo:g11:cancer:4}
पराबैंगनी प्रकाश (त्वचा के \omterm{def:g11:cancer:cancer}{कैंसर}), तंबाकू का तारकोल (फेफड़े का \omterm{def:g11:cancer:cancer}{कैंसर}),
आयनकारी विकिरण, एस्बेस्टस, शराब, पैपिलोमा विषाणु (ग्रीवा का \omterm{def:g11:cancer:cancer}{कैंसर})।
\end{solution}

\begin{solution}{exo:g11:cancer:5}
उसकी सारी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} उसी एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} से उतरी हैं जिसमें पहला \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} हुआ
था; वे उस \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} को और उसके बाद वालों को साझा करती हैं, तथा वही
निष्क्रिय X \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} जैसे निशान भी।
\end{solution}

\begin{solution}{exo:g11:cancer:7}

\begin{solution}{exo:g11:cancer:8}
p53 के बिना कोई \omterm{def:g10:cells-common-unit:cell}{कोशिका} प्रतिकृतिकरण से पहले क्षतिग्रस्त \omterm{def:g10:universal-dna:information}{DNA} की मरम्मत के
लिए रुकती ही नहीं, और न क्षति गंभीर होने पर ख़ुद को नष्ट करती है: इसलिए हर
विभाजन पर क्षतियाँ \omterm{def:g11:mutations:mutation}{उत्परिवर्तनों} में नक़ल हो जाती हैं, और गड्डमड्ड जीनोम
वाली \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} बची रह जाती हैं।
\end{solution}

\begin{solution}{exo:g11:cancer:9}
उसके शरीर की हर \omterm{def:g10:cells-common-unit:cell}{कोशिका} में एक \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} पहले ही नहीं है; इसलिए दूसरे का
अकेला एक कायिक \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} वह हानि पूरी कर देता है, दो स्वतंत्र
\omterm{def:g11:mutations:mutation}{उत्परिवर्तनों} के बजाय। जन्म से ही एक क़दम उठा होने पर बचे हुए क़दम जल्दी
और अधिक \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में पूरे हो जाते हैं। और उसकी बहनों को वह \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} उसी जनक
से $1/2$ प्रायिकता से मिलता है।
\end{solution}

\begin{solution}{exo:g11:cancer:10}
वे दवाएँ विभाजित होती \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} को बिना भेद किए मारती हैं: बालों की जड़ों
की \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}, आँत की परत और अस्थि मज्जा तेज़ी से विभाजित होते हैं और \omterm{def:g11:cancer:cancer}{अर्बुद}
के साथ-साथ वे भी चपेट में आते हैं --- इसीलिए बाल झड़ना, मितली और श्वेत
\omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की कमी।
\end{solution}

\begin{solution}{exo:g11:cancer:11}
छोड़ देने से नए \omterm{def:g11:mutations:mutation}{उत्परिवर्तनों} की आपूर्ति बंद हो जाती है, और उस परत की
\omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} धीरे-धीरे बदल दी जाती हैं, जिससे शुरुआती क़दम ढोते बहुत-से क्लोन
हट जाते हैं। पर कई क़दम उठा चुके कुछ क्लोन टिके रह जाते हैं, और पहले से
मौजूद \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} मिटते नहीं; इसलिए जोखिम धूम्रपान न करने वाले से ऊपर बना
रहता है।
\end{solution}

\begin{solution}{exo:g11:cancer:12}
$10^{12} \times 10^{-7} = 10^5$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} पहला \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} पा लेती हैं।
उनमें से हर एक ऐसा क्लोन बसाती है जो हज़ारों बार विभाजित हो सकता है, इसलिए
दूसरा \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} किसी एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} में नहीं, बल्कि पूरे क्लोन के विभाजनों में
ढूँढ़ा जाता है, और वैसे ही तीसरा: यानी वे क़दम निशाना गुणा कर देते हैं, और
सच्ची प्रायिकता $10^{-21}$ से कई कोटियाँ ऊपर है।
\end{solution}

\begin{solution}{exo:g11:cancer:13}
वह विषाणु \omterm{def:g11:gene-expression:protein}{प्रोटीन} \omterm{def:g10:universal-dna:gene}{जीन} को छुए बिना ही ब्रेक हटा देता है: संक्रमित \omterm{def:g10:cells-common-unit:cell}{कोशिका}
ऐसे बरतती है जैसे p53 खो गया हो, और वे \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} जमा करती है जो यह
प्रक्रिया पूरी कर देते हैं। टीका प्रतिरक्षा तंत्र से उस विषाणु को गर्भाशय
ग्रीवा में संक्रमण फैलाने से पहले ही नष्ट करा देता है; और एक बार \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}
संक्रमित तथा रूपांतरित हो जाएँ तो टीका काम का नहीं रहता --- इसीलिए
किशोरावस्था के आरंभ में टीकाकरण।
\end{solution}

\begin{solution}{exo:g11:cancer:14}
पॉलिप वह क्लोन है जो एक या दो क़दम उठा चुका है; उसे हटा देने से वे
\omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} हट जाती हैं जिनमें बचे हुए क़दम उठ सकते थे, इसलिए वह \omterm{def:g11:cancer:cancer}{कैंसर} बनता
ही नहीं। और चूँकि पॉलिप से \omterm{def:g11:cancer:cancer}{कैंसर} तक का क्रम दस से बीस साल लेता है, इसलिए
दस साल का अंतराल अधिकांश पॉलिप उन्हें पूरा करने से पहले ही पकड़ लेता है।
\end{solution}

\begin{solution}{exo:g11:cancer:15}
घटना-दर उम्र के साथ इसलिए चढ़ती है कि वे क़दम दशकों लेते हैं, पर वे क़दम
जीवन के दौरान ही उठते हैं, और बड़े हिस्से में संपर्कों के नीचे ---
अकेला तंबाकू \omterm{def:g11:cancer:cancer}{कैंसर} से होने वाली एक-तिहाई मौतों का हिसाब देता है, और बाक़ी
के अधिकांश का संक्रमण, शराब, धूप तथा आहार। किसी भी उम्र पर कोई संपर्क हटा
देना आगे आने वाले \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} घटा देता है; और छानबीन आख़िरी क़दम से पहले
क्लोन हटा देती है। बुढ़ापा वह समय है जब यह बही पेश की जाती है, वह नहीं जब
वह लिखी जाती है।
\end{solution}

\begin{solution}{pb:g11:cancer:1}
\textbf{1.} वे \omterm{def:g11:cancer:cancer}{अर्बुद} में मौजूद हैं और रोगी की बाक़ी \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में
अनुपस्थित, इसलिए वे किसी शरीर \omterm{def:g10:cells-common-unit:cell}{कोशिका} में उठे। जनन-वंश वाले \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} हर
\omterm{def:g10:cells-common-unit:cell}{कोशिका} में होते, स्वस्थ ऊतक समेत।

\textbf{2.} यह कि \omterm{def:g11:cancer:cancer}{अर्बुद} के अधिकांश \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} बेंज़ोपाइरीन ने किए ---
यानी तंबाकू के धुएँ के \omterm{def:g11:mutations:mutagen}{उत्परिवर्तजन} ने अपना निशान छोड़ दिया।

\textbf{3.} एक निरर्थक \omterm{def:g11:mutations:mutation}{उत्परिवर्तन}: समय से पहले आया विराम \omterm{def:g11:gene-expression:code}{कोडॉन} p53 को
काट देता है, और वह काम नहीं कर पाता।

\textbf{4.} p53 एक ब्रेक है: एक काम करता \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} भी ब्रेक लगा देता है,
इसलिए दोनों को जाना पड़ा (एक उत्परिवर्तित, एक विलोपित)। और ग्राही एक
गति-यंत्र है: हमेशा सक्रिय एक \omterm{def:g11:gene-expression:protein}{प्रोटीन} \omterm{def:g10:cells-common-unit:cell}{कोशिका} को चलाता ही रहता है, चाहे
दूसरा \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} कुछ भी करे।

\textbf{5.} ग्राही वाला \omterm{def:g10:universal-dna:gene}{जीन} गति-यंत्र है (\omterm{prop:g11:cancer:genes}{प्रोटो-ऑन्कोजीन} जो ऑन्कोजीन बन
गया); और p53 ब्रेक है (अर्बुद-दमनकारी)।

\textbf{6.} $20 \times 365 \times 35 \approx \num{255000}$ सिगरेट।

\textbf{7.} प्रति \omterm{def:g10:cells-common-unit:cell}{कोशिका} लगभग $\num{255000}$ प्रतिस्थापन --- यानी मिले
हुए 5000 से पचास गुना। अधिकांश क्षतियों की मरम्मत हो जाती है, और बहुत अधिक
\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} वाली \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} मर जाती हैं; 5000 वही अंश है जो बचा रह गया।

\textbf{8.} वे जल्दी हुए, यानी उस बसाने वाली \omterm{def:g10:cells-common-unit:cell}{कोशिका} में या उसके पहले
वंशजों में, इसलिए हर \omterm{def:g11:cancer:cancer}{अर्बुद} \omterm{def:g10:cells-common-unit:cell}{कोशिका} को वे विरासत में मिले; और बाक़ी बाद में
उप-क्लोनों में उठे।

\textbf{9.} $10^{10} \approx 2^{33}$: यानी 33 दोगुनन, और हर एक 100 दिन
का, यानी लगभग 9 साल।

\textbf{10.} यदि उस \omterm{def:g11:cancer:cancer}{अर्बुद} को अपनी बसाने वाली \omterm{def:g10:cells-common-unit:cell}{कोशिका} से बढ़ने में 9 साल या
उससे अधिक लगे, तो आख़िरी मुख्य \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} धूम्रपान के उन 35 सालों में से
क़रीब 25वें साल के आसपास, या उससे पहले हुआ।

\textbf{11.} $17 \times 1\% = 17\%$।

\textbf{12.} 17 \omterm{def:g11:cancer:cancer}{कैंसर}, जिनमें से 1 वैसे भी होता: यानी 100 में 16 धूम्रपान
के खाते में।

\textbf{13.} कोई नया \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} नहीं जुड़ता; वह परत ख़ुद को नया करती रहती
है और शुरुआती क़दम ढोते बहुत-से क्लोन सालों में झड़कर बदल जाते हैं; और बचे
हुए क्लोनों में उतने क़दम नहीं होते जितने जारी रखने पर होते।

\textbf{14.} कई विरले \omterm{def:g11:mutations:mutation}{उत्परिवर्तनों} को एक ही वंशक्रम में जमा होना पड़ता
है, और फिर उस वंशक्रम को पकड़ में आने लायक़ आकार तक बढ़ना पड़ता है: हर चरण
साल लेता है, और उनका जोड़ दशक।

\textbf{15.} प्रति सिगरेट प्रति \omterm{def:g10:cells-common-unit:cell}{कोशिका} एक \omterm{def:g11:mutations:mutation}{उत्परिवर्तन}, परत की क़रीब
$10^9$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में, $\num{255000}$ सिगरेटों पर, यानी पूरे फेफड़े में
बिखरे $10^{14}$ \omterm{def:g11:mutations:mutation}{उत्परिवर्तन}; और उनमें से कुछ ऐसी \omterm{def:g10:cells-common-unit:cell}{कोशिका} के p53 या किसी
\omterm{prop:g11:cancer:genes}{प्रोटो-ऑन्कोजीन} पर पड़ेंगे ही जो पहले से कोई और क़दम ढो रही हो। कोशिशों की
संख्या ही असंभव-से को निश्चित बना देती है।

\textbf{16.} धूम्रपान करने वालों में $\num{27000}$ मौतें, कुल
$\num{11.25e6}$ में से, यानी प्रति 100\,000 लगभग 240; और न करने वालों में
3000, कुल $\num{33.75e6}$ में से, यानी प्रति 100\,000 लगभग 9.

\textbf{17.} लगभग 27 गुना --- यानी भारी धूम्रपान करने वालों के लिए चित्र
के 17 से 25 की ही कोटि का; और आबादी वाला आँकड़ा हल्के तथा भारी धूम्रपान
करने वालों और छोड़ चुके लोगों को आपस में मिला देता है।

\textbf{18.} साल भर में लगभग 3000, यानी धूम्रपान न करने वालों का हिस्सा।
तुरंत नहीं: आज के और पहले के धूम्रपान करने वालों में पहले से मौजूद
\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} अगले बीस-तीस साल तक \omterm{def:g11:cancer:cancer}{कैंसर} देते रहेंगे।

\textbf{19.} हर धूम्रपान करने वाले की छानबीन $\num{27000}$ का लगभग 20\%
बचाती है, यानी साल भर में क़रीब 5000 मौतें; और धूम्रपान छोड़ना आख़िरकार
लगभग $\num{24000}$ रोक देता --- यानी पाँच गुना अधिक।

\textbf{20.} G-से-T निशान ने सिद्ध किया कि \omterm{def:g11:cancer:cancer}{अर्बुद} के \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} तंबाकू के
धुएँ से आए; वह \omterm{def:g11:cancer:cancer}{कैंसर} एक ऑन्कोजीन के सक्रिय होने और p53 के दोनों \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी}
खोने से बना; और धूम्रपान न करना उसे रोक देता।
\end{solution}
