\chapter{Cadeias alimentares}\label{ch:g3:food-chains}

O capim cresce no campo. Um gafanhoto come o capim. Um lagarto pega
o gafanhoto. Um gavião, planando lá em cima, leva o lagarto. Quatro
refeições --- e de repente a tarde do gavião depende de quão verde o
capim cresceu. As refeições da natureza se ligam em cadeias, e
aprender a ler essas cadeias é aprender como um campo inteiro se
sustenta.

\section{Ler e escrever uma cadeia alimentar}

\begin{definition}[Cadeia alimentar]\label{def:g3:food-chains:chain}
Uma \emph{cadeia alimentar}\index{cadeia alimentar} é uma fila de
\omterm{def:g1:living-or-not:living}{seres vivos} em que cada um é comido pelo seguinte. Ela é escrita com
setas, e cada seta quer dizer \emph{``é comido
por''}\index{é comido por}:
\[
\text{capim} \longrightarrow \text{gafanhoto}
\longrightarrow \text{lagarto} \longrightarrow \text{gavião}.
\]
Cada \omterm{def:g1:living-or-not:living}{ser vivo} é um \emph{elo}\index{elo} da cadeia.
\end{definition}

\begin{remark}[Cuidado com o sentido da seta]\label{rem:g3:food-chains:arrow}
A seta aponta do comido para o comedor --- ela segue o alimento.
``Capim $\rightarrow$ gafanhoto'' se lê ``o capim é comido pelo
gafanhoto''. Escrever as setas ao contrário --- apontando para a
refeição --- é o erro mais comum das \omterm{def:g3:food-chains:chain}{cadeias alimentares}; lembre que
a seta mostra para onde o alimento \emph{vai}.
\end{remark}

\begin{proposition}[Toda cadeia começa por uma planta]\label{prop:g3:food-chains:start}
O primeiro elo de uma \omterm{def:g3:food-chains:chain}{cadeia alimentar} é sempre uma \omterm{def:g1:plants-around-us:plant}{planta} (ou outro
\omterm{def:g1:living-or-not:living}{ser vivo} que fabrica o próprio alimento). Só as \omterm{def:g1:plants-around-us:plant}{plantas} se alimentam
sem comer ninguém --- a partir da luz, da água e do ar, como disse a
\cref{def:g1:plants-around-us:plant}. Todo elo \omterm{def:g1:animals-around-us:animal}{animal} vive,
diretamente ou por meio de outros, desse primeiro elo verde.
\end{proposition}
\admitted

\begin{example}[Cadeias de três hábitats]\label{ex:g3:food-chains:three}
Na lagoa: \omterm{def:g1:plants-around-us:plant}{plantas} aquáticas $\rightarrow$ girino $\rightarrow$ larva
de libélula $\rightarrow$ peixinho $\rightarrow$ garça. Na floresta:
\omterm{def:g1:plants-around-us:parts}{folhas} de carvalho $\rightarrow$ \omterm{ex:g2:animals-grow-up:butterfly}{lagarta} $\rightarrow$ chapim
$\rightarrow$ gavião-miúdo. No quintal: alface $\rightarrow$ caracol
$\rightarrow$ ouriço. Primeiro elo: sempre verde.
\end{example}

\begin{method}[Montar uma cadeia alimentar]\label{met:g3:food-chains:build}
\begin{enumerate}
  \item Escolha um \omterm{def:g1:animals-around-us:animal}{animal} e pergunte: o que ele come? Escreva o
        alimento à esquerda dele, com a seta apontando para o
        comedor;
  \item faça a mesma pergunta ao alimento e continue para a
        esquerda, até chegar a uma \omterm{def:g1:plants-around-us:plant}{planta};
  \item pergunte: quem come o \omterm{def:g1:animals-around-us:animal}{animal} do começo? Estenda para a
        direita enquanto houver um comedor;
  \item confira se toda seta se lê \omterm{def:g3:food-chains:chain}{``é comido por''} --- com o
        alimento fluindo para a direita.
\end{enumerate}
\end{method}

\section{Quando um elo muda}

\begin{proposition}[Os elos dependem uns dos outros]\label{prop:g3:food-chains:depend}
Numa \omterm{def:g3:food-chains:chain}{cadeia alimentar}, todo elo depende dos elos anteriores e
alimenta os seguintes. Se um elo fica escasso, os elos seguintes
passam fome --- e o elo \emph{anterior}, já não tão comido, pode se
multiplicar. Uma mudança num elo viaja pela cadeia inteira.
\end{proposition}
\admitted

\begin{example}[O verão seco]\label{ex:g3:food-chains:dry}
Uma seca deixa o capim do campo marrom. Os gafanhotos, sem alimento,
ficam poucos; os lagartos que os caçavam passam fome e, por sua vez,
alimentam mal o gavião. Um primeiro elo marrom, e a cadeia inteira
acima dele fica magra --- o gavião nunca comeu capim, mas sente a
seca.
\end{example}

\begin{example}[Tirando o caçador]\label{ex:g3:food-chains:hunter}
Suponha que os gaviões somem do vale. Os lagartos, menos caçados, se
multiplicam; os gafanhotos passam a ser comidos com mais força e
diminuem; o capim, menos mordiscado, pode até ficar mais cerrado.
Tirar o \emph{último} elo também viaja pela cadeia --- só que no
outro sentido.
\end{example}

\begin{remark}[Os campos de verdade são teias]\label{rem:g3:food-chains:web}
A nossa cadeia é um fio só, mas o gavião também leva ratinhos, o
lagarto também come moscas e o capim também alimenta coelhos. Muitas
cadeias se cruzando fazem uma rede --- a \emph{teia alimentar} que
um ano mais adiante vai estudar (o
\cref{ch:g2:what-animals-eat} já deu uma pista disso com a raposa, o
coelho e o capim). Primeiro as cadeias: elas são os fios com que a
rede é tecida.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:1}
O que a seta de uma \omterm{def:g3:food-chains:chain}{cadeia alimentar} quer dizer? Para que lado ela
aponta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:2}
Que tipo de \omterm{def:g1:living-or-not:living}{ser vivo} é sempre o primeiro elo? Por quê?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:3}
Escreva a cadeia do campo deste capítulo, com os quatro elos e as
setas dela.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:4}
Corrija esta cadeia: raposa $\rightarrow$ coelho $\rightarrow$
capim.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:5}
Monte uma cadeia com estes elos: chapim, \omterm{def:g1:plants-around-us:parts}{folhas} de carvalho,
gavião-miúdo, \omterm{ex:g2:animals-grow-up:butterfly}{lagarta}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:6}
Na cadeia do quintal alface $\rightarrow$ caracol $\rightarrow$
ouriço, quem é comido pelo caracol? Quem come o caracol?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:7}
Use o \cref{met:g3:food-chains:build} para montar uma cadeia de pelo
menos três elos terminando na garça da lagoa.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:8}
No verão seco do \cref{ex:g3:food-chains:dry}, por que o gavião
sofre, se ele nunca come capim?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:food-chains:9}
Quem cuida de horta e envenena todos os caracóis às vezes perde o
ouriço --- e depois encontra \emph{mais} lesmas e caracóis do que
antes. Explique as duas metades com a
\cref{prop:g3:food-chains:depend}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:food-chains:10}
``O lagarto depende do capim.'' Verdadeiro ou falso? Defenda a sua
resposta com a cadeia do campo --- e diga em que sentido uma falha
do capim viaja.
\end{exercise}
