\chapter{العناية بالطبيعة}\label{ch:g2:caring-for-nature}

بعد النزهة تترك أسرة المرج كما وجدته تمامًا؛ وتترك أخرى أكياسًا
وأغطية قناني وقرية نمل مدوسة. أرانب المرج ونمله وأقحوانه لا تستطيع
أن تسألنا أن نحترس --- لكنها كائنات حية تشاركنا أماكننا، وسلوكنا
في \omterm{def:g2:caring-for-nature:environment}{الطبيعة} يهمّها. والعناية \omterm{def:g2:caring-for-nature:environment}{بالطبيعة} شيء يستطيع كل امرئ أن يتعلمه،
في أي سن.

\section{الطبيعة حيث تعيش الكائنات الحية}

\begin{definition}[الطبيعة والبيئة]\label{def:g2:caring-for-nature:environment}
\emph{الطبيعة}\index{طبيعة} عالم الكائنات الحية وما يحيط بها:
المروج، والغابات، والبرك، والبحار --- والهواء والماء والتربة التي
تعتمد عليها. ويسمى ما يحيط \omterm{def:g1:living-or-not:living}{بالكائن الحي} ويعتمد عليه
\emph{بيئته}\index{بيئة}.
\end{definition}

\begin{example}[كل شيء بيت لأحد]\label{ex:g2:caring-for-nature:home}
جذع متعفن في الغابة يبدو نفاية --- وهو في الحقيقة قصر: خنافس تحت
اللحاء، وحشرات قمل خشبي تحته، وطحلب فوقه، وقنفذ يشتّي في جوفه.
اركله لهوًا فتفقد أربع أُسر بيتها. وفي \omterm{def:g2:caring-for-nature:environment}{الطبيعة} يكاد كل شيء يكون
بيتًا لأحد، أو غذاءه، أو مأواه.
\end{example}

\begin{omfigure}
\includegraphics[width=0.62\linewidth]{images/book1/ai/g2-nature-log.png}

{\small الجذع المتعفن الذي هو قصر في الحقيقة: طحلب على السقف،
وفطر على الجدران، وخنفساء في الممر.}
\end{omfigure}

\begin{remark}[نحن جزء منها]\label{rem:g2:caring-for-nature:part}
والناس كائنات حية أيضًا: فالهواء الذي نتنفسه، والماء الذي نشربه،
والغذاء الذي نأكله كلها تأتي من \omterm{def:g2:caring-for-nature:environment}{الطبيعة}. والعناية بها ليست إحسانًا
إلى الأرانب فحسب --- بل هي رعاية للعالم الذي نعيش منه نحن أنفسنا،
كما قالت \cref{ch:g2:where-animals-live} عن كل \omterm{def:g2:where-animals-live:habitat}{موئل}.
\end{remark}

\section{ما يضرّ وما ينفع}

\begin{proposition}[أفعال صغيرة وضرر حقيقي]\label{prop:g2:caring-for-nature:harm}
الضرر في \omterm{def:g2:caring-for-nature:environment}{الطبيعة} قلما يأتي من إرادة الإساءة --- بل يأتي من قلة
التفكير:
\begin{itemize}
  \item النفايات تبقى سنوات وقد توقع \omterm{def:g1:animals-around-us:animal}{الحيوانات} في شرك أو تسمّمها؛
  \item وقطف كل \omterm{def:g1:plants-around-us:parts}{زهرة} لا يترك ما يصنع \omterm{def:g2:from-seed-to-plant:seed}{البذور} --- ولا يترك للنحل
        شيئًا؛
  \item والصياح العالي والمطاردة يفزعان \omterm{def:g1:animals-around-us:animal}{الحيوانات} عن غذائها
        وصغارها؛
  \item والضفاف المدوسة والحجارة المقلوبة \omterm{def:g1:plants-around-us:tree}{والأغصان} المكسورة كل
        واحد منها ينقض بيتًا صغيرًا.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[غطاء القنينة]\label{ex:g2:caring-for-nature:cap}
غطاء بلاستيكي يسقط منا لا شيء عندنا، ويبقى في المرج سنوات كثيرة
كثيرة --- تكفي لأن يصير للكبار أولاد يجدون الغطاء نفسه. \omterm{def:g2:caring-for-nature:environment}{والطبيعة}
لا شاحنة نفايات لها: فما نتركه يبقى.
\end{example}

\begin{method}[قواعد الزائر]\label{met:g2:caring-for-nature:rules}
في \omterm{def:g2:caring-for-nature:environment}{الطبيعة}:
\begin{enumerate}
  \item احمل نفاياتك كلها إلى البيت --- كلها؛
  \item وانظر إلى الأزهار والحشرات والأعشاش بدل أن تجمعها؛ واقطف
        على الأكثر أزهارًا قليلة شائعة، ولا تقطف حزمة كاملة أبدًا؛
  \item وأعد كل حجر وكل جذع تقلبه --- فتحته من يسكن؛
  \item والزم الممرات قرب المواضع الهشّة كضفاف الجداول؛
  \item وراقب \omterm{def:g1:animals-around-us:animal}{الحيوانات} من بعيد، في هدوء، ودع صغارها وشأنها.
\end{enumerate}
واترك المكان بحيث لا يستطيع أحد أن يعرف أنك كنت هناك.
\end{method}

\begin{example}[المساعدة عن قصد]\label{ex:g2:caring-for-nature:help}
والعناية قد تتجاوز ترك الأذى. فصحن ماء في موجة حرّ، وركن بري صغير
في الحديقة يُترك بلا جزّ، \omterm{def:g2:from-seed-to-plant:seed}{وبذور} للطيور في البرد، وبركة مدرسة
كالبركة في \cref{ex:g2:where-animals-live:schoolpond}: كل واحد
منها يعطي كائنات حية غذاءً أو ماءً أو بيتًا كانت قد فقدته.
\end{example}

\section{الفرز والاقتصاد}

\begin{proposition}[النفايات يمكن أن تُردّ]\label{prop:g2:caring-for-nature:recycle}
كثير مما نرميه يمكن أن يُستعمل من جديد.
و\emph{فرز}\index{فرز النفايات} النفايات --- الزجاج مع الزجاج،
والورق مع الورق، والقشور إلى السماد --- يتيح للقناني أن تصير
قناني جديدة وللقشور أن تصير تربة، بدل أن يتكوّم كل شيء. وما
\emph{يُقتصد} --- من ماء وورق وكهرباء --- لا يصير نفاية البتة.
\end{proposition}
\admitted

\begin{example}[صندوق السماد]\label{ex:g2:caring-for-nature:compost}
قلوب التفاح وقشور الجزر، في صندوق سماد، تتحول ببطء إلى تربة داكنة
هشّة --- بعون الديدان وحشرات القمل الخشبي وكائنات حية أصغر من أن
تُرى. وفي الربيع التالي تغذّي تلك التربة الطماطم. لم يُهدر شيء؛
فالقشور عادت إلى \omterm{def:g2:born-grow-die:cycle}{دورة الحياة}.
\end{example}

\begin{remark}[الأيدي الصغيرة تُحسب]\label{rem:g2:caring-for-nature:count}
إغلاق الصنبور في أثناء تنظيف الأسنان، واستعمال وجهي \omterm{def:g1:plants-around-us:parts}{الورقة}، وصندوق
السماد، والنفايات المحمولة إلى البيت: كل واحد منها صغير، وملايين
الناس يفعلونها --- أو لا يفعلونها. \omterm{def:g2:caring-for-nature:environment}{والطبيعة} يُعتنى بها، أو تُؤذى،
في الخيارات اليومية الصغيرة كخياراتك في الغالب.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:1}
ما \omterm{def:g2:caring-for-nature:environment}{بيئة} \omterm{def:g1:living-or-not:living}{الكائن الحي}؟ وسمِّ ثلاثة أشياء في بيئتك تأتي من \omterm{def:g2:caring-for-nature:environment}{الطبيعة}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:2}
من يمكن أن يفقد بيته حين يُركل جذع متعفن؟ سمِّ ثلاثة من سكانه في
\cref{ex:g2:caring-for-nature:home}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:3}
أعطِ أربعًا من قواعد الزائر في \cref{met:g2:caring-for-nature:rules}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:4}
لماذا يجب أن تعود النفايات معنا إلى البيت؟ أجب بغطاء القنينة في
\cref{ex:g2:caring-for-nature:cap}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:5}
لماذا لا ينبغي أن نقطف كل أزهار حزمة؟ أعطِ السببين في
\cref{prop:g2:caring-for-nature:harm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:6}
ماذا يحدث لقلوب التفاح في صندوق السماد؟ ومن يقوم بالعمل، وماذا
ننال في النهاية؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:7}
اذكر طريقتين لمساعدة \omterm{def:g2:caring-for-nature:environment}{الطبيعة} عن قصد، من
\cref{ex:g2:caring-for-nature:help}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:8}
جرّب: في نزهتك القادمة، عدّ قطع النفايات التي تراها (من غير أن
تلمس الخطر منها). أي صنف أكثرها شيوعًا؟ وإلى أين كان ينبغي أن يذهب
كل صنف؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:caring-for-nature:9}
عند البركة تريد إيما أن تأخذ ثلاثة \omterm{ex:g2:animals-grow-up:frog}{شراغيف} إلى بيتها في مرطبان
``لتساعدها على النمو بأمان''. مستعملًا هذا الفصل
و\cref{exo:g2:where-animals-live:10}، بم تنصحها بدل ذلك؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:caring-for-nature:10}
``غطاء قنينة واحد لا شيء.'' استعمل \cref{rem:g2:caring-for-nature:count}
للجواب: ماذا يحدث حين يفكر مليون زائر بالجملة نفسها --- في
الاتجاه السيئ، وفي الاتجاه الحسن؟
\end{exercise}
