\chapter{De vijf zintuigen}\label{ch:g1:the-five-senses}

Doe je \omterm{def:g1:the-five-senses:sense}{ogen} dicht en de wereld verdwijnt niet: je hoort nog steeds de
klas, je ruikt het middageten dat klaarstaat, je voelt de stoel onder
je. Het lichaam heeft vijf deuren naar de wereld --- de \omterm{def:g1:the-five-senses:sense}{zintuigen} ---
en elke deur is een orgaan van het lichaam dat de hele dag voor je werkt.

\section{Vijf zintuigen, vijf organen}

\begin{definition}[Zintuig en zintuigorgaan]\label{def:g1:the-five-senses:sense}
Een \emph{zintuig}\index{zintuig} is een manier waarop het lichaam iets
over de wereld te weten komt. Elk zintuig werkt via een
\emph{zintuigorgaan}\index{zintuigorgaan}, een lichaamsdeel dat voor
die taak gebouwd is:
\begin{enumerate}
  \item \emph{zien}\index{zien} --- de \emph{ogen}\index{oog} zien
        licht, kleuren en vormen;
  \item \emph{horen}\index{horen} --- de \emph{oren}\index{oor} horen
        geluiden;
  \item \emph{ruiken}\index{ruiken} --- de \emph{neus}\index{neus}
        ruikt geuren;
  \item \emph{proeven}\index{proeven} --- de \emph{tong}\index{tong}
        proeft het eten;
  \item \emph{voelen}\index{voelen} --- de \emph{huid}\index{huid},
        over het hele lichaam, voelt ruw en glad, warm en koud.
\end{enumerate}
\end{definition}

\begin{example}[Ontbijt, zintuig voor zintuig]\label{ex:g1:the-five-senses:breakfast}
Eén kom warme chocolademelk gebruikt alle vijf: de \omterm{def:g1:the-five-senses:sense}{ogen} zien de damp,
de \omterm{def:g1:the-five-senses:sense}{neus} ruikt de cacao, de \omterm{def:g1:the-five-senses:sense}{huid} van de handen voelt de warme kom, de
\omterm{def:g1:the-five-senses:sense}{oren} horen de lepel tikken, en de \omterm{def:g1:the-five-senses:sense}{tong} --- het beste deel --- proeft.
\end{example}

\begin{remark}[Voelen zit overal]\label{rem:g1:the-five-senses:skin}
Vier zintuigorganen zitten op het \omterm{def:g1:my-body:parts}{hoofd}, maar het orgaan van het voelen
is de \omterm{def:g1:the-five-senses:sense}{huid}, en de \omterm{def:g1:the-five-senses:sense}{huid} bedekt je van top tot teen. Daarom voel je een
zuchtje wind in je nek, een steentje in je schoen en een
lieveheersbeestje dat op je \omterm{def:g1:my-body:parts}{arm} landt --- allemaal zonder te kijken.
\end{remark}

\section{De zintuigen melden aan de hersenen}

\begin{proposition}[Boodschappen naar de hersenen]\label{prop:g1:the-five-senses:brain}
Een \omterm{def:g1:the-five-senses:sense}{zintuigorgaan} begrijpt niet wat het waarneemt --- het \emph{stuurt
een boodschap}. Alle boodschappen reizen binnen door het lichaam naar
de \emph{hersenen}\index{hersenen}, in het \omterm{def:g1:my-body:parts}{hoofd}, die ze samenvoegen:
``dat is de stem van oma'', ``die geur is geroosterd brood''. Zonder
hersenen zouden de \omterm{def:g1:the-five-senses:sense}{ogen} en de \omterm{def:g1:the-five-senses:sense}{oren} aan niemand melden.
\end{proposition}
\admitted

\begin{example}[Dingen benoemen in het donker]\label{ex:g1:the-five-senses:dark}
Een vriend legt een bekend voorwerp in je handen terwijl je \omterm{def:g1:the-five-senses:sense}{ogen} dicht
zijn: een sleutel, een spons, een lepel. Je noemt het bijna meteen. De
\omterm{def:g1:the-five-senses:sense}{huid} stuurde haar boodschap --- koud, hard, gekarteld --- en de
hersenen antwoordden: ``een sleutel''. \omterm{def:g1:the-five-senses:sense}{Zintuigen} verzamelen; de
hersenen begrijpen.
\end{example}

\begin{remark}[Elkaar helpen]\label{rem:g1:the-five-senses:together}
De \omterm{def:g1:the-five-senses:sense}{zintuigen} werken als een team. Eten ``smaakt'' flauw als een
verkoudheid je \omterm{def:g1:the-five-senses:sense}{neus} verstopt, want de reuk deed de helft van het werk
van de smaak. En als één \omterm{def:g1:the-five-senses:sense}{zintuig} zijn werk helemaal niet kan doen,
leren de andere meer te doen: mensen die niet kunnen zien lezen met
hun vingertoppen, met het gevoel, en steken de straat over met hun \omterm{def:g1:the-five-senses:sense}{oren}.
\end{remark}

\section{Zorgen voor de zintuigorganen}

\begin{proposition}[Kwetsbare deuren]\label{prop:g1:the-five-senses:fragile}
Zintuigorganen zijn kostbaar en gemakkelijk te beschadigen. Te hard
geluid beschadigt de \omterm{def:g1:the-five-senses:sense}{oren}; naar de Zon kijken beschadigt de \omterm{def:g1:the-five-senses:sense}{ogen}; een
te hete lepel verbrandt de \omterm{def:g1:the-five-senses:sense}{tong}. Een beschadigd \omterm{def:g1:the-five-senses:sense}{zintuigorgaan} blijft
soms beschadigd --- dus beschermen we ze.
\end{proposition}
\admitted

\begin{method}[Mijn zintuigen beschermen]\label{met:g1:the-five-senses:protect}
\begin{enumerate}
  \item Kijk nooit recht in de Zon, zelfs niet even;
  \item houd geluiden zacht: niet in \omterm{def:g1:the-five-senses:sense}{oren} schreeuwen, koptelefoon zacht
        gezet;
  \item duw nooit iets in je \omterm{def:g1:the-five-senses:sense}{oor} of je \omterm{def:g1:the-five-senses:sense}{neus};
  \item blaas op heet eten voor je proeft;
  \item lees en teken bij goed licht.
\end{enumerate}
\end{method}

\begin{example}[Het concert hiernaast]\label{ex:g1:the-five-senses:loud}
Na heel harde muziek fluiten de \omterm{def:g1:the-five-senses:sense}{oren} soms een tijdje na. Dat fluiten is
het \omterm{def:g1:the-five-senses:sense}{oor} dat ``te veel!'' zegt. Mensen die met lawaaiige machines werken
dragen precies daarom gehoorbeschermers --- hun \omterm{def:g1:the-five-senses:sense}{oren} moeten hun een
leven lang meegaan.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:1}
Noem de vijf \omterm{def:g1:the-five-senses:sense}{zintuigen} en, bij elk, het bijbehorende \omterm{def:g1:the-five-senses:sense}{zintuigorgaan}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:2}
Welk \omterm{def:g1:the-five-senses:sense}{zintuigorgaan} meldt: (a) de kleur van een ballon? (b) de zachtheid
van een deken? (c) de geur van regen? (d) het geluid van een bel?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:3}
Welk \omterm{def:g1:the-five-senses:sense}{zintuigorgaan} zit niet op het \omterm{def:g1:my-body:parts}{hoofd}? Hoe ver reikt het?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:4}
In \cref{ex:g1:the-five-senses:dark} waren de \omterm{def:g1:the-five-senses:sense}{ogen} dicht. Welk orgaan
stuurde de boodschap, en welk lichaamsdeel begreep hem?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:5}
Met een verkoudheid en een verstopte \omterm{def:g1:the-five-senses:sense}{neus} lijkt eten zijn smaak te
verliezen. Welke twee \omterm{def:g1:the-five-senses:sense}{zintuigen} werken bij het eten meestal samen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:6}
Geef drie regels uit \cref{met:g1:the-five-senses:protect}, en zeg welk
\omterm{def:g1:the-five-senses:sense}{zintuigorgaan} elke regel beschermt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:7}
Waarom dragen mensen die met lawaaiige machines werken
gehoorbeschermers? Gebruik \cref{prop:g1:the-five-senses:fragile}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:8}
Probeer het: laat met je \omterm{def:g1:the-five-senses:sense}{ogen} dicht een helper drie soorten eten onder
je \omterm{def:g1:the-five-senses:sense}{neus} houden --- bijvoorbeeld banaan, kaas, sinaasappel. Hoeveel
noemde je \omterm{def:g1:the-five-senses:sense}{neus} er goed, zonder hulp van de \omterm{def:g1:the-five-senses:sense}{ogen}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:the-five-senses:9}
Een vriend beweert: ``Het \omterm{def:g1:the-five-senses:sense}{oog} begrijpt wat het ziet.'' Verbeter de zin
met \cref{prop:g1:the-five-senses:brain}: wat doet het \omterm{def:g1:the-five-senses:sense}{oog} echt, en wie
doet het begrijpen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:the-five-senses:10}
Hoe kan iemand die niet kan zien een boek ``lezen'' en veilig een
straat oversteken? Zeg welke \omterm{def:g1:the-five-senses:sense}{zintuigen} welke taken overnemen.
\end{exercise}
