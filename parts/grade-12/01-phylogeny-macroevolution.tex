\chapter{Phylogeny and Macroevolution}\label{ch:g12:phylogeny-macroevolution}

A fin, a wing and a human arm look different and do different jobs --- yet
they share the same bone layout. That shared skeleton is a message from
deep time: the three structures are modifications of one ancestral
limb. This chapter turns such messages into \emph{trees}, distinguishes
true kinship from superficial resemblance, and asks how new \omterm{def:g12:phylogeny-macroevolution:species}{species}
arise and how life's major patterns are read from fossils and \omterm{prop:g11:dna-replication:genome}{genomes}.

\section{Reading evolutionary trees}

\begin{definition}[Phylogeny]\label{def:g12:phylogeny-macroevolution:phylogeny}
A \emph{phylogeny}\index{phylogeny} (or \emph{phylogenetic
tree}\index{phylogenetic tree}) is a branching diagram that represents
the evolutionary relationships among a set of taxa (\omterm{def:g12:phylogeny-macroevolution:species}{species}, \omterm{def:g11:evolution-mechanisms:pool}{populations}
or \omterm{def:g11:gene-expression:gene}{genes}). Time runs from the root (the most recent common ancestor of
all tips) toward the tips (the living or sampled taxa).
\end{definition}

\begin{definition}[Node, branch, clade]\label{def:g12:phylogeny-macroevolution:node}
On a tree, an internal \emph{node}\index{node (phylogeny)} marks a
hypothesised common ancestor; a \emph{branch}\index{branch (phylogeny)}
(edge) is a lineage through time. A \emph{clade}\index{clade} is a
group consisting of an ancestor and \emph{all} of its descendants ---
equivalently, a \omterm{def:g12:phylogeny-macroevolution:monophyly}{monophyletic} group (\cref{def:g12:phylogeny-macroevolution:monophyly}).
\end{definition}

\begin{remark}[Rotation of branches]\label{rem:g12:phylogeny-macroevolution:rotation}
\omterm{def:g12:phylogeny-macroevolution:node}{Branches} may be rotated around any \omterm{def:g12:phylogeny-macroevolution:node}{node} without changing the
relationships: only the \emph{topology} (who is closer to whom) matters,
not left--right \omterm{def:g10:biodiversity-classification:hierarchy}{order} on the page.
\end{remark}

\begin{definition}[Sister groups]\label{def:g12:phylogeny-macroevolution:sister}
Two \omterm{def:g12:phylogeny-macroevolution:node}{clades} that share an exclusive common ancestor not shared with any
other \omterm{def:g12:phylogeny-macroevolution:node}{clade} in the tree are \emph{sister groups}\index{sister group}:
they are each other's closest relatives among the taxa shown.
\end{definition}

\begin{method}[Reading a tree]\label{met:g12:phylogeny-macroevolution:read}
\begin{enumerate}
  \item Find the most recent common ancestor of the taxa of interest
        (their shared \omterm{def:g12:phylogeny-macroevolution:node}{node}).
  \item Taxa that share a more recent common ancestor are more closely
        related than taxa whose common ancestor sits deeper.
  \item Do \emph{not} read relatedness from tip-to-tip distance along
        the paper, nor from who sits ``higher'' or ``more evolved''.
\end{enumerate}
\end{method}

\section{Homology, analogy and character evidence}

\begin{definition}[Homology]\label{def:g12:phylogeny-macroevolution:homology}
A \emph{homology}\index{homology} is a similarity due to inheritance
from a common ancestor. \omterm{prop:g11:evolution-mechanisms:evidence}{Homologous structures} may look very different
if descendants specialised for different functions (the vertebrate
forelimb in bats, whales and horses).
\end{definition}

\begin{definition}[Analogy (homoplasy)]\label{def:g12:phylogeny-macroevolution:analogy}
An \emph{analogy}\index{analogy} (or \emph{homoplasy}\index{homoplasy})
is a similarity that evolved independently --- often by convergent
\omterm{def:g11:evolution-mechanisms:fitness}{adaptation} to similar environments --- and does \emph{not} mark a unique
shared ancestry. The wings of bats and insects are analogous as wings
(flight surfaces) though both groups have homologous bones/structures
at deeper levels of organisation.
\end{definition}

\begin{example}[Forelimbs and wings]\label{ex:g12:phylogeny-macroevolution:wings}
The wing of a bat and the flipper of a dolphin are homologous as
tetrapod forelimbs (same skeletal plan from a common tetrapod ancestor).
The wing of a bat and the wing of a butterfly are analogous as flying
appendages: flight evolved separately in insects and vertebrates.
\end{example}

\begin{proposition}[Shared derived characters]\label{prop:g12:phylogeny-macroevolution:synapomorphy}
\omterm{def:g12:phylogeny-macroevolution:node}{Clades} are supported by \emph{shared derived
characters}\index{shared derived character} (\emph{synapomorphies}):
traits that originated in the common ancestor of the \omterm{def:g12:phylogeny-macroevolution:node}{clade} and are
inherited by its members. Shared ancestral traits (present in a broader
group) do not by themselves define a smaller \omterm{def:g12:phylogeny-macroevolution:node}{clade}.
\end{proposition}
\admitted

\begin{remark}[Molecular characters]\label{rem:g12:phylogeny-macroevolution:molecular}
Modern trees often use \omterm{def:g11:dna-replication:helix}{DNA} or \omterm{def:g10:molecules-of-life:proteins}{protein} \omterm{def:g10:molecules-of-life:proteins}{sequences}: each aligned position
is a character. Convergence and reverse \omterm{def:g11:evolution-mechanisms:mutation}{mutations} still produce
\omterm{def:g12:phylogeny-macroevolution:analogy}{homoplasy}, so many independent sites and careful methods are needed.
\end{remark}

\section{Monophyly, paraphyly and polyphyly}

\begin{definition}[Monophyletic, paraphyletic, polyphyletic]
\label{def:g12:phylogeny-macroevolution:monophyly}
A group is
\begin{itemize}
  \item \emph{monophyletic}\index{monophyletic} if it contains an
        ancestor and \emph{all} of its descendants (a \omterm{def:g12:phylogeny-macroevolution:node}{clade});
  \item \emph{paraphyletic}\index{paraphyletic} if it contains an
        ancestor and \emph{some but not all} of its descendants;
  \item \emph{polyphyletic}\index{polyphyletic} if its members do not
        share an exclusive common ancestor within the group (the most
        recent common ancestor is excluded, often because of convergent
        traits).
\end{itemize}
\end{definition}

\begin{example}[``Reptiles'' without birds]\label{ex:g12:phylogeny-macroevolution:reptiles}
Traditional ``reptiles'' (lizards, snakes, turtles, crocodiles) exclude
birds, yet birds nest inside the archosaur \omterm{def:g12:phylogeny-macroevolution:node}{clade} with crocodiles. The
traditional group is therefore \omterm{def:g12:phylogeny-macroevolution:monophyly}{paraphyletic}. A \omterm{def:g12:phylogeny-macroevolution:monophyly}{monophyletic}
\emph{Sauropsida} includes birds.
\end{example}

\begin{proposition}[Natural classification]\label{prop:g12:phylogeny-macroevolution:natural}
In cladistic classification, only \omterm{def:g12:phylogeny-macroevolution:monophyly}{monophyletic} groups are recognised as
formal taxa: classification then mirrors the nested structure of the tree
of life.
\end{proposition}
\admitted

\section{Speciation}

\begin{definition}[Species (biological concept)]
\label{def:g12:phylogeny-macroevolution:species}
Under the \emph{biological species concept}\index{biological species
concept}, a \emph{species}\index{species} is a group of actually or
potentially interbreeding \omterm{def:g11:evolution-mechanisms:pool}{populations} that are reproductively isolated
from other such groups. (\omterm{prop:g11:evolution-mechanisms:evidence}{Fossils}, asexual \omterm{def:g10:scientific-biology:organism}{organisms} and hybrids require
other operational definitions.)
\end{definition}

\begin{definition}[Speciation]\label{def:g12:phylogeny-macroevolution:speciation}
\emph{Speciation}\index{speciation} is the process by which one lineage
splits into two or more lineages that no longer exchange \omterm{def:g11:gene-expression:gene}{genes} freely.
\end{definition}

\begin{definition}[Allopatric speciation]
\label{def:g12:phylogeny-macroevolution:allopatric}
\emph{Allopatric speciation}\index{allopatric speciation} occurs when
\omterm{def:g11:evolution-mechanisms:pool}{populations} are separated by a geographic barrier (mountain, ocean,
glacier). Isolated \omterm{def:g11:evolution-mechanisms:pool}{gene pools} diverge by selection, drift and \omterm{def:g11:evolution-mechanisms:mutation}{mutation}
until \omterm{def:g12:phylogeny-macroevolution:isolation}{reproductive isolation} evolves.
\end{definition}

\begin{definition}[Sympatric speciation]
\label{def:g12:phylogeny-macroevolution:sympatric}
\emph{Sympatric speciation}\index{sympatric speciation} occurs without
complete geographic isolation: \omterm{def:g11:evolution-mechanisms:geneflow}{gene flow} is reduced by ecological
specialisation, polyploidy (especially in plants), or assortative
mating within the same region.
\end{definition}

\begin{definition}[Reproductive isolation]
\label{def:g12:phylogeny-macroevolution:isolation}
\emph{Reproductive isolating
mechanisms}\index{reproductive isolation} prevent successful \omterm{def:g11:gene-expression:gene}{gene}
exchange. They may act before fertilisation (behavioural, temporal,
mechanical, gametic) or after (hybrid inviability, sterility, breakdown
in later generations).
\end{definition}

\begin{example}[Polyploidy in plants]\label{ex:g12:phylogeny-macroevolution:polyploid}
A chromosome-doubling event can instantly create a polyploid plant that
cannot produce fertile offspring with its \omterm{def:g11:meiosis-life-cycles:ploidy}{diploid} parent \omterm{def:g11:evolution-mechanisms:pool}{population}.
\omterm{def:g12:phylogeny-macroevolution:sympatric}{Sympatric speciation} by polyploidy is common in flowering plants.
\end{example}

\section{Macroevolution and the fossil record}

\begin{definition}[Macroevolution]\label{def:g12:phylogeny-macroevolution:macroevolution}
\emph{Macroevolution}\index{macroevolution} refers to evolutionary
change above the \omterm{def:g12:phylogeny-macroevolution:species}{species} level: the origin of major body plans,
\omterm{def:g12:phylogeny-macroevolution:radiation}{adaptive radiations}, mass extinctions and long-term trends visible in
the fossil record and large phylogenies. It is continuous with
microevolution (allele-frequency change) but observed on deeper time
scales.
\end{definition}

\begin{definition}[Adaptive radiation]
\label{def:g12:phylogeny-macroevolution:radiation}
An \emph{adaptive radiation}\index{adaptive radiation} is the rapid
diversification of a lineage into many \omterm{def:g12:phylogeny-macroevolution:species}{species} that exploit different
ecological \omterm{def:g10:ecosystems:habitat-niche}{niches}, often after colonising a new area or surviving a mass
extinction that emptied \omterm{def:g10:ecosystems:habitat-niche}{niches}.
\end{definition}

\begin{proposition}[Mass extinctions reset opportunity]
\label{prop:g12:phylogeny-macroevolution:extinction}
Mass extinctions remove many lineages and open ecological space;
surviving \omterm{def:g12:phylogeny-macroevolution:node}{clades} may radiate into vacated roles (for example mammals
after the end-Cretaceous extinction of non-avian dinosaurs).
\end{proposition}
\admitted

\begin{remark}[Tempo]\label{rem:g12:phylogeny-macroevolution:tempo}
Rates of morphological change and \omterm{def:g12:phylogeny-macroevolution:speciation}{speciation} vary: some lineages are
conservative for long periods; others diversify quickly. \omterm{prop:g11:evolution-mechanisms:evidence}{Fossils} and
dated molecular trees constrain these tempos but never show every
intermediate.
\end{remark}

\begin{method}[Building a simple character table]
\label{met:g12:phylogeny-macroevolution:characters}
\begin{enumerate}
  \item Choose an outgroup (a relative known to lie outside the group
        of interest) to polarise characters: ancestral vs.\ derived.
  \item Score taxa for independent characters (morphology or \omterm{def:g11:dna-replication:helix}{DNA}).
  \item Group taxa that share derived states; nest those groups into a
        tree that minimises unnecessary independent origins when
        possible.
\end{enumerate}
\end{method}

\section{Problem: Reconstructing a clade from characters}

\begin{problem}[{Weekend problem --- from a character matrix to a
hypothesised phylogeny, with tests of monophyly and of speciation
scenarios}]
\label{pb:g12:phylogeny-macroevolution:1}
Four living taxa A--D and an outgroup O are scored for six binary
characters ($0$ = ancestral, $1$ = derived relative to O):

\begin{center}
\begin{tabular}{c|cccccc}
\omterm{def:g10:biodiversity-classification:hierarchy}{Taxon} & 1 & 2 & 3 & 4 & 5 & 6 \\
\hline
O & 0 & 0 & 0 & 0 & 0 & 0 \\
A & 1 & 1 & 0 & 0 & 1 & 0 \\
B & 1 & 1 & 0 & 1 & 0 & 0 \\
C & 1 & 0 & 1 & 0 & 0 & 1 \\
D & 1 & 0 & 1 & 0 & 0 & 1 \\
\end{tabular}
\end{center}

\medskip
\noindent\textbf{Part I --- Characters and \omterm{def:g12:phylogeny-macroevolution:homology}{homology}.}
\begin{enumerate}
  \item Which character is a synapomorphy of the whole ingroup $\{A,B,C,D\}$?
  \item Which characters support grouping $A$ with $B$? Which support
        $C$ with $D$?
  \item Characters 5 and 4 each appear in only one \omterm{def:g10:biodiversity-classification:hierarchy}{taxon}. What do they
        tell you about relationships among A--D?
  \item Suppose character 2 also appears (convergently) in D. How does
        that change the evidence for $\{A,B\}$?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Tree topology.}
\begin{enumerate}[resume]
  \item Draw a rooted tree consistent with the \omterm{def:g10:cellular-respiration:mito}{matrix} (O as outgroup)
        that groups $A$ with $B$ and $C$ with $D$.
  \item On your tree, mark the \omterm{def:g12:phylogeny-macroevolution:node}{clade} $\{C,D\}$ and the \omterm{def:g12:phylogeny-macroevolution:sister}{sister group} of
        $\{A,B\}$.
  \item Is $\{A,B,C\}$ \omterm{def:g12:phylogeny-macroevolution:monophyly}{monophyletic} on your tree? Justify.
  \item Give one \omterm{def:g12:phylogeny-macroevolution:monophyly}{monophyletic}, one \omterm{def:g12:phylogeny-macroevolution:monophyly}{paraphyletic} and one \omterm{def:g12:phylogeny-macroevolution:monophyly}{polyphyletic}
        subset of $\{A,B,C,D\}$ on your tree.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- \omterm{def:g12:phylogeny-macroevolution:speciation}{Speciation} and \omterm{def:g12:phylogeny-macroevolution:macroevolution}{macroevolution}.}
\begin{enumerate}[resume]
  \item Propose an allopatric scenario that could produce the split
        between the $\{A,B\}$ and $\{C,D\}$ lineages.
  \item Propose a sympatric mechanism that could later split $C$ from
        $D$ without a geographic barrier.
  \item After a mass extinction removes many competitors, how might an
        \omterm{def:g12:phylogeny-macroevolution:radiation}{adaptive radiation} reshape the tree of survivors over the next
        few million years?
  \item In one paragraph, explain why classification that insists on
        \omterm{def:g12:phylogeny-macroevolution:monophyly}{monophyletic} taxa is more predictive than one that groups by
        overall similarity alone (including analogies).
\end{enumerate}
\end{problem}
