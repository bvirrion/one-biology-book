\chapter{Zuurstof in het water}\label{ch:g7:oxygen-in-the-water}

De bergbeek en de zomerse vijver zijn allebei ``water'' --- en toch zou de
forel die in de een rondschiet in de ander naar adem happen, en zou de
karper die in de vijver gedijt de beek kaal vinden van zijn voedsel. Het
verschil tussen die twee wateren is grotendeels één onzichtbaar getal:
hoeveel \omterm{def:g7:what-is-respiration:respiration}{zuurstof} er in opgelost is. Dit hoofdstuk leert dat getal lezen,
meten en voorspellen --- en daarmee de kaart van het waterleven.

\section{Opgeloste zuurstof, een schaarse voorraad}

\begin{definition}[Opgeloste zuurstof]\label{def:g7:oxygen-in-the-water:dissolved}
Water bevat \omterm{def:g7:what-is-respiration:respiration}{zuurstof} zoals zoete thee suiker bevat --- onzichtbaar,
\emph{opgelost}\index{opgeloste zuurstof}. De hoeveelheid is piepklein:
een liter goed belucht water draagt ongeveer \qty{10}{mg} \omterm{def:g7:what-is-respiration:respiration}{zuurstof}, zo'n
dertig keer minder dan een liter lucht. Elke \omterm{def:g7:how-animals-breathe:four}{kieuw} en elke onderwaterademer
uit \cref{ch:g7:how-animals-breathe} put uit die ene dunne voorraad
(\cref{prop:g7:what-is-respiration:habitats}).
\end{definition}

\begin{proposition}[Wat de voorraad aanvult]\label{prop:g7:oxygen-in-the-water:fills}
Twee leveranciers vullen de \omterm{def:g7:what-is-respiration:respiration}{zuurstof} van een water aan:
\begin{enumerate}
  \item de \emph{lucht erboven}: aan het oppervlak lost \omterm{def:g7:what-is-respiration:respiration}{zuurstof} op ---
        sneller waar het water geroerd, opgespat en geschuimd wordt
        (stroomversnellingen, watervallen, golven, regen);
  \item de \emph{\omterm{def:g5:food-webs:roles}{producenten} eronder}: overdag geven waterplanten en
        groene losse cellen \omterm{def:g7:what-is-respiration:respiration}{zuurstof} aan het water af terwijl ze
        fabriceren (\cref{rem:g6:plants-are-producers:gas}).
\end{enumerate}
\end{proposition}
\admitted

\begin{proposition}[Wat haar leegtrekt, en wat het water kan bevatten]\label{prop:g7:oxygen-in-the-water:drains}
De voorraad wordt leeggetrokken door \emph{elke ademhaler} --- \omterm{def:g1:animals-around-us:animal}{dieren},
\omterm{def:g1:plants-around-us:plant}{planten} 's nachts, en vooral de \omterm{def:g6:decomposers-and-soil:decomposers}{afbrekers} wanneer er veel dode materie te
eten is (de achtste pot uit \cref{pb:g7:what-is-respiration:1}). En het
plafond verschuift: \emph{koud} water kan aanzienlijk meer \omterm{def:g7:oxygen-in-the-water:dissolved}{opgeloste
zuurstof} bevatten dan warm --- een beek van \qty{5}{\degreeCelsius}
draagt anderhalf keer zo veel als een vijver van
\qty{25}{\degreeCelsius}. Warmte knijpt dubbel: minder \omterm{def:g7:what-is-respiration:respiration}{zuurstof} bevat,
terwijl elk motorgeluid
(\cref{ex:g7:what-is-respiration:demand}) sneller loopt.
\end{proposition}
\admitted

\begin{example}[Beek tegenover vijver]\label{ex:g7:oxygen-in-the-water:streampond}
Nu lezen de twee wateren als boekhoudingen. De bergbeek: koud (hoog
plafond), bij elk rotsblok wit geklopt (snelle aanvulling) --- het hele
jaar \omterm{def:g7:what-is-respiration:respiration}{zuurstof} dicht bij het plafond. De zomerse vijver: warm (laag
plafond), stil (trage aanvulling), rijk aan modder en dode materie
(hongerige \omterm{def:g6:decomposers-and-soil:decomposers}{afbrekers}) --- \omterm{def:g7:what-is-respiration:respiration}{zuurstof} die over de vloer schraapt, en het
ergst in de kleine uurtjes voor zonsopgang, als de \omterm{def:g1:plants-around-us:plant}{planten} de hele nacht
geademd hebben zonder daglichtfabricage om het te compenseren.
\end{example}

\begin{example}[De curve van de vijver lezen]\label{ex:g7:oxygen-in-the-water:curve}
De dagelijkse golf van de vijver is
\cref{rem:g6:plants-are-producers:gas} als gegevens getekend: bij
zonsopgang start de fabricage van de \omterm{def:g5:food-webs:roles}{producenten} en klimt de curve; laat
in de middag piekt ze; 's nachts putten alle ademhalers uit de voorraad
zonder andere leverancier dan het stille oppervlak --- en glijdt de curve
naar haar minimum bij zonsopgang. Vissterfte in dichtgegroeide vijvers
gebeurt bij het ochtendgloren, en nu weet je waarom.
\end{example}

\section{De kaart van de bewoners}

\begin{proposition}[Zuurstof tekent de kaart]\label{prop:g7:oxygen-in-the-water:map}
Omdat \omterm{def:g5:biodiversity:species}{soorten} verschillen in hun zuurstofbehoefte, brengt het opgeloste
getal de bewoners in kaart
(\cref{prop:g6:exploring-our-environment:decide}, onder water):
\begin{itemize}
  \item \emph{veeleisende} \omterm{def:g5:biodiversity:species}{soorten} --- forel, haftlarven met hun wuivende
        kieuwpluimen (\cref{pb:g7:how-animals-breathe:1}) --- leven
        alleen in koud, geklopt, zuurstofrijk water;
  \item \emph{verdraagzame} \omterm{def:g5:biodiversity:species}{soorten} --- karper, veel poelslakken,
        rattenstaartlarven met snorkels naar het oppervlak --- redden
        het in warm, stilstaand, armer water;
  \item dus laat de telling zich achterstevoren lezen: vind je haftlarven
        onder de stenen, dan is het water goed; vind je alleen snorkelaars
        en luchthappers, dan is het slecht.
\end{itemize}
Rivierbeheerders tellen juist daarom het kleine leven: de bewoners zijn
een verslag van de slechtste dagen die het water onlangs had, en die kan
één enkele metersaflezing missen.
\end{proposition}
\admitted

\begin{example}[Het verhaal van het organische afval]\label{ex:g7:oxygen-in-the-water:waste}
Het ongezuiverde afval van een dorp --- rijk aan \omterm{def:g6:plants-are-producers:matter}{organische materie} ---
bereikt een heldere beek. Stroomafwaarts van de pijp smullen de \omterm{def:g6:decomposers-and-soil:decomposers}{afbrekers}
en vermeerderen ze zich
(\cref{def:g6:decomposers-and-soil:decomposers}), en hun massale
\omterm{def:g7:what-is-respiration:respiration}{ademhaling} trekt de \omterm{def:g7:what-is-respiration:respiration}{zuurstof} leeg: de forel en de haftlarven verdwijnen,
de verdraagzame snorkelaars nemen het over, en nog verder stroomafwaarts
--- het afval opgegeten, het water over de ondiepten opnieuw belucht ---
keren de veeleisende \omterm{def:g5:biodiversity:species}{soorten} terug. De vervuiling was \omterm{def:g6:plants-are-producers:matter}{organische materie};
de moordenaar was zuurstofhonger; de kaart legde het allemaal vast.
\end{example}

\begin{method}[Een water beoordelen]\label{met:g7:oxygen-in-the-water:judge}
Om een beek of vijver te beoordelen:
\begin{enumerate}
  \item meet of schat de leveranciers: \omterm{def:g6:exploring-our-environment:conditions}{temperatuur}, beroering, groene
        begroeiing;
  \item weeg de afvoeren: modder en dode materie, drukte, elke instroom
        van organisch afval;
  \item tel het kleine leven onder stenen en tussen de waterplanten ---
        veeleisende \omterm{def:g5:biodiversity:species}{soorten} aanwezig, of alleen verdraagzame?
  \item bemonster je op één uur, denk dan aan de vijvercurve: een mooie
        middagaflezing kan een dodelijke dageraad verbergen.
\end{enumerate}
\end{method}

\section{Probleem: Het jaarboek van de beheerder}

\begin{problem}[{Weekendprobleem --- één rivier, vier meetpunten, een jaar
metingen}]\label{pb:g7:oxygen-in-the-water:1}
Een rivierbeheerder houdt een jaarboek bij voor vier meetpunten: K1, een
koude stroomversnelling onder een stuw; K2, een traag warm traject door
weilanden; K3, vlak onder de afvalpijp van een zuivelfabriek; K4, twee
kilometer verderop, voorbij een lange stenige ondiepte. Elk meetpunt
krijgt metingen met de meter en een telling van het kleine leven.

\medskip\noindent\textbf{Deel I --- De meetpunten lezen.}
\begin{enumerate}
  \item K1 leest \qty{11}{mg/L} bij \qty{8}{\degreeCelsius}; K2 leest
        \qty{7}{mg/L} bij \qty{22}{\degreeCelsius}. Wijs het verschil toe
        aan de juiste leveranciers en plafonds.
  \item De telling van K1: jonge forellen, haftlarven. Die van K2:
        karpers, poelslakken, rattenstaartlarven. Duid allebei met
        \cref{prop:g7:oxygen-in-the-water:map}.
  \item K3 leest \qty{3}{mg/L} hoewel het water niet warmer is dan bij
        K2. Noem de afvoer, en de werkers erachter.
  \item K4 leest \qty{9}{mg/L} en de telling laat weer haftlarven zien.
        Verklaar het herstel met allebei de leveranciers en met het lot
        van het afval.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De ritmes van het jaar.}
\begin{enumerate}[resume]
  \item De metingen bij zonsopgang bij K2 zakken in augustus tot
        \qty{4}{mg/L} terwijl de middagmetingen \qty{9}{mg/L} halen.
        Teken of beschrijf de dagcurve, en noem de twee processen die van
        plaats wisselen.
  \item De januarimetingen bij K2 zijn stabieler en hoger dan die van
        augustus, hoewel het weilandtraject net zo stil is als altijd.
        Geef de twee winterredenen.
  \item Er wordt een hete, windstille twee weken voorspeld. Zet de vier
        meetpunten op volgorde van risico op vissterfte, met bij elk één
        reden.
  \item De beheerder \omterm{def:g1:plants-around-us:plant}{plant} haar augustusmetingen vlak voor zonsopgang.
        Verdedig dat schema.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Oordelen en
remedies.}
\begin{enumerate}[resume]
  \item De zuivelfabriek plaatst een zuivering die het meeste organische
        materiaal uit haar uitstroom haalt. Voorspel de meter en de
        telling van K3 een jaar later, op volgorde van herstel.
  \item Een landeigenaar stelt voor de oeverbomen bij K2 te kappen ``om
        meer licht meer \omterm{def:g7:what-is-respiration:respiration}{zuurstof} te laten maken''. Weeg dat voorstel: wat
        doet de schaduw ook, volgens
        \cref{prop:g7:oxygen-in-the-water:drains}, en wat zou jij
        antwoorden?
  \item De telling vindt haftlarven bij een meetpunt waarvan de enige
        middagaflezing er middelmatig uitziet. Welke getuige geloof je
        over het seizoen als geheel, en waarom?
  \item Sluit het jaarboek af met de regel van de beheerder voor de hele
        rivier in één zin, met de woorden \emph{koud}, \emph{geroerd} en
        \emph{onbelast} (met organisch afval).
\end{enumerate}
\end{problem}
