\section*{Hoofdstuk \ref{ch:g6:plants-are-producers} --- Planten zijn producenten}

\begin{solution}{exo:g6:plants-are-producers:1}
\omterm{def:g6:plants-are-producers:matter}{Minerale materie}: materie van de niet-levende wereld --- water, gesteente
(en ook de gassen van de lucht, \omterm{prop:g4:what-plants-need:needs}{minerale zouten}). \omterm{def:g6:plants-are-producers:matter}{Organische materie}:
materie die door \omterm{def:g6:exploring-our-environment:components}{levende wezens} gemaakt is en waaruit hun lichaam bestaat
--- hout, \omterm{def:g3:bones-and-muscles:muscle}{spier} (en ook suiker, wol, vet).
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:2}
Grondstoffen: water en \omterm{prop:g4:what-plants-need:needs}{minerale zouten} uit de bodem, een gas
(koolstofdioxide) uit de lucht. \omterm{prop:g3:balanced-diet:jobs}{Energie}: licht, opgevangen door de groene
stof. Product: \omterm{def:g6:plants-are-producers:matter}{organische materie} --- suiker, en daarna hout, \omterm{def:g1:plants-around-us:parts}{blad},
\omterm{def:g1:plants-around-us:parts}{wortel}, \omterm{def:g2:from-seed-to-plant:seed}{zaad}. Werkplaats: het groene \omterm{def:g1:plants-around-us:parts}{blad}, overdag.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:3}
Mineraal: dauw, graniet, zeezout. Organisch: bijenwas, bakolie --- en het
slakkenhuis is het maaksel van de slak, gebouwd uit minerale stof die het
\omterm{def:g1:animals-around-us:animal}{dier} heeft afgezet: gemaakt door een \omterm{def:g6:exploring-our-environment:components}{levend wezen}, uit minerale
grondstoffen. (Beide antwoorden gelden, met die redenering.)
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:4}
Vooral uit lucht en water --- het gas dat uit de lucht gehaald wordt en
het water dat de \omterm{def:g1:plants-around-us:parts}{wortels} drinken --- in de \omterm{def:g1:plants-around-us:parts}{bladeren} samengevoegd met de
\omterm{prop:g3:balanced-diet:jobs}{energie} van het licht; de bodem gaf alleen kleine hoeveelheden \omterm{prop:g4:what-plants-need:needs}{minerale
zouten}.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:5}
Het neemt koolstofdioxide op en geeft zuurstof af. Elk ademhalend \omterm{def:g6:exploring-our-environment:components}{levend
wezen} ademt het eerste uit en heeft het tweede nodig --- wijzelf
inbegrepen.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:6}
De aardappelknol en de tulpenbol (ondergrondse voorraden), de dikke
helften van de boon (de \omterm{def:g3:animals-through-seasons:active}{voedselvoorraad} van het \omterm{def:g2:from-seed-to-plant:seed}{zaad}), de dikke \omterm{def:g1:plants-around-us:parts}{wortel}
van de peen, de suikerrijke biet --- allemaal organische reserves, van de
zomerfabricage op de bank gezet.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:7}
Omdat de fabricage op licht draait: zonder licht is er geen \omterm{prop:g3:balanced-diet:jobs}{energie} om
\omterm{def:g6:plants-are-producers:matter}{organische materie} te bouwen, hoeveel mineralen er ook in de bodem
wachten. Licht is de enige grondstof zonder vervanger --- de zouten zijn
kruiderij, geen brandstof.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:8}
Omdat de eerste schakel van een keten nieuwe \omterm{def:g6:plants-are-producers:matter}{organische materie} de
levende wereld in moet brengen, en \omterm{def:g1:animals-around-us:animal}{dieren} \omterm{def:g5:food-webs:roles}{consumenten} zijn: die kunnen
alleen \omterm{def:g6:plants-are-producers:matter}{organische materie} omzetten die iemand anders al gemaakt heeft.
Alleen \omterm{def:g5:food-webs:roles}{producenten} fabriceren haar uit minerale grondstoffen --- dus is
de eerste schakel altijd groen.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:9}
Brood: tarwe --- de \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} van de tarweplant; het spoor eindigt in het
tarweblad. Boter: room --- \omterm{def:g2:animals-grow-up:born}{melk} --- de koe --- gras; het spoor eindigt in
het grasblad. Kaas: \omterm{def:g2:animals-grow-up:born}{melk} --- de koe --- opnieuw gras. Drie
voedingsmiddelen, drie (of twee) \omterm{def:g1:plants-around-us:parts}{bladeren}, allemaal in het licht.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:10}
Hooi is het voedsel van de koe --- \omterm{def:g6:plants-are-producers:matter}{organische materie} die ze in haar
lichaam omzet. Compost is niet het voedsel van de \omterm{def:g1:plants-around-us:plant}{plant}: het vult onder
het verteren de \emph{minerale} zouten van de bodem aan. De \omterm{def:g1:plants-around-us:plant}{plant} maakt
daarna haar eigen \omterm{def:g6:plants-are-producers:matter}{organische materie} uit die mineralen, uit lucht, water
en licht. De koe eet maaltijden; de \omterm{def:g1:plants-around-us:plant}{plant} fabriceert ze.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:11}
In het licht geeft de dagfabricage van de \omterm{def:g1:plants-around-us:plant}{planten} zuurstof af en neemt ze
koolstofdioxide op, wat de ademhaling van de \omterm{prop:g3:sorting-living-things:groups}{vissen} in evenwicht houdt
--- de gaswisseling loopt beide kanten op en de pot draait rond. In het
donker stopt de fabricage terwijl elke bewoner (de \omterm{def:g1:plants-around-us:plant}{planten} inbegrepen)
blijft ademen: de zuurstof raakt op, de koolstofdioxide stapelt zich op,
en de afgesloten wereld stikt.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:12}
Goed: de materie van de boom kwam niet uit de aarde --- zijn weging
bewees dat voorgoed. Onvolledig: water was maar één van de
massa-ingrediënten; hij kon niets weten van het gas uit de lucht, dat
veel van de stof van het hout levert. De moderne aanvulling:
koolstofdioxide uit de lucht (plus de kleine \omterm{prop:g4:what-plants-need:needs}{minerale zouten} van de
bodem) --- water en lucht samen, samengevoegd door het licht.
\end{solution}

\begin{solution}{pb:g6:plants-are-producers:1}
\textbf{1.} De tonnen zijn \omterm{def:g6:plants-are-producers:matter}{organische materie} die de \omterm{def:g1:plants-around-us:plant}{planten} zelf
gefabriceerd hebben uit minerale grondstoffen --- water, de
koolstofdioxide uit de lucht, de zouten van de bodem --- met licht als
\omterm{prop:g3:balanced-diet:jobs}{energie}. De zakken vullen de mineralen aan; de lucht en de regen leveren
de massa; de \omterm{def:g1:plants-around-us:parts}{bladeren} doen het maken.

\textbf{2.} Aardappel: ondergronds opslagorgaan (\omterm{ex:g4:plants-without-seeds:bulbtuber}{knol}). \omterm{def:g1:plants-around-us:parts}{Wortel}:
opslagwortel. Bonenzaad: de \omterm{def:g3:animals-through-seasons:active}{voedselvoorraad} van het \omterm{def:g2:from-seed-to-plant:seed}{zaad}, ingepakt voor
de volgende \omterm{def:g1:plants-around-us:plant}{plant}. Slablad: het werkende orgaan zelf --- de
lichtvangwerkplaats, vers gegeten.

\textbf{3.} De echte leverancier is het licht: elke gram opbrengst is met
de \omterm{prop:g3:balanced-diet:jobs}{energie} van het licht gebouwd, overdag, in de \omterm{def:g1:plants-around-us:parts}{bladeren}.

\textbf{4.} Voor de \omterm{prop:g4:what-plants-need:needs}{minerale zouten}: oogsten voeren die af, en zonder
aanvulling zouden de \omterm{def:g1:plants-around-us:plant}{planten} mineraaltekort krijgen zoals de munt in het
zand --- klein in massa, onmisbaar in functie.

\textbf{5.} De proef met de donkere kast (de bleke gevangene): het doek
onttrok het licht, en een week ervan is al te zien --- bleek, uitgerekt,
zwakke fabricage.

\textbf{6.} De knollen worden door het loof gevuld: \omterm{def:g1:plants-around-us:parts}{bladeren} fabriceren
overdag suiker, en het overschot wordt omlaag geleid en in de zwellende
knollen op de bank gezet. Weelderig werkend \omterm{def:g1:plants-around-us:parts}{blad} betekent een groot
overschot --- goed loof, goede knollen.

\textbf{7.} Water. De fabricage heeft al haar grondstoffen tegelijk
nodig: licht levert \omterm{prop:g3:balanced-diet:jobs}{energie}, geen stof --- met droge \omterm{def:g1:plants-around-us:parts}{wortels} is er niets
om mee te bouwen, hoe helder de hemel ook is.

\textbf{8.} De ondergespitte \omterm{def:g1:plants-around-us:parts}{wortels} zijn \omterm{def:g6:plants-are-producers:matter}{organische materie} die aan de
bodem teruggegeven wordt: al verterend maken ze hun mineralen vrij voor
de volgende gewassen --- de regel van teruggeven uit
\cref{prop:g5:people-and-nature:lasting}, betaald in plantenmateriaal.
De verspilling zou zijn ze te verbranden.

\textbf{9.} Aardappelplant $\rightarrow$ varken $\rightarrow$ het gezin.
Al haar \omterm{def:g6:plants-are-producers:matter}{organische materie} is eerst in de \omterm{def:g1:plants-around-us:parts}{bladeren} van de aardappelplant
gefabriceerd --- het varken en het gezin hebben haar alleen omgezet.

\textbf{10.} Ingrediënten: water en \omterm{prop:g4:what-plants-need:needs}{minerale zouten} uit de bodem,
koolstofdioxide uit de lucht --- samengevoegd tot bladmaterie. De
precieze rol van de zon: \omterm{prop:g3:balanced-diet:jobs}{energie}, geen stof. Haar salade is water, lucht
en aarde, \emph{gebouwd door} zonlicht --- de poëtische eer verdiend, de
ingrediëntenlijst fout.

\textbf{11.} De \omterm{def:g5:food-webs:roles}{afbrekers} --- bodemleven van wormen en pissebedden tot
\omterm{def:g6:exploring-our-environment:components}{levende wezens} die te klein zijn om te zien --- voeden zich met de dode
\omterm{def:g1:plants-around-us:parts}{bladeren} en breken hun \omterm{def:g6:plants-are-producers:matter}{organische materie} weer af richting \omterm{def:g6:plants-are-producers:matter}{minerale
materie}, en vullen zo de bodem aan waar de volgende gewassen uit putten.

\textbf{12.} Bijvoorbeeld: ``De \omterm{def:g5:food-webs:roles}{producenten} van de tuin bouwen
\omterm{def:g6:plants-are-producers:matter}{organische materie} --- de oogst --- uit \omterm{def:g6:plants-are-producers:matter}{minerale materie} uit bodem, water
en lucht, met licht als hun \omterm{prop:g3:balanced-diet:jobs}{energie}.''
\end{solution}
