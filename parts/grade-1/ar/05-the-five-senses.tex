\chapter{الحواس الخمس}\label{ch:g1:the-five-senses}

أغمض عينيك فلا يختفي العالم: ما زلت تسمع الصف، وتشمّ الغداء وهو
يُطهى، وتحسّ بالكرسي تحتك. للجسم خمسة أبواب إلى العالم --- هي
الحواس --- وكل باب عضو من الجسم يعمل لك طول النهار.

\section{خمس حواس وخمسة أعضاء}

\begin{definition}[الحاسة وعضو الحس]\label{def:g1:the-five-senses:sense}
\emph{الحاسة}\index{حاسة} طريقة يعرف بها الجسم ما في العالم. وكل
حاسة تعمل من خلال \emph{عضو حس}\index{عضو الحس}، وهو جزء من الجسم
مبني لهذا العمل:
\begin{enumerate}
  \item \emph{البصر}\index{بصر} --- \emph{العينان}\index{عين}
        تريان الضوء والألوان والأشكال؛
  \item \emph{السمع}\index{سمع} --- \emph{الأذنان}\index{أذن}
        تسمعان الأصوات؛
  \item \emph{الشم}\index{شم} --- \emph{الأنف}\index{أنف} تشمّ
        الروائح؛
  \item \emph{الذوق}\index{ذوق} --- \emph{اللسان}\index{لسان}
        يذوق الطعام؛
  \item \emph{اللمس}\index{لمس} --- \emph{الجلد}\index{جلد}، وهو
        يغطي الجسم كله، يحسّ بالخشن والناعم والدافئ والبارد.
\end{enumerate}
\end{definition}

\begin{example}[الفطور حاسةً حاسة]\label{ex:g1:the-five-senses:breakfast}
قدح واحد من الشوكولاتة الساخنة يستعمل الحواس الخمس جميعًا:
\omterm{def:g1:the-five-senses:sense}{فالعينان} تريان البخار، والأنف تشمّ الكاكاو، \omterm{def:g1:the-five-senses:sense}{وجلد} اليدين يحسّ
بالقدح الدافئ، \omterm{def:g1:the-five-senses:sense}{والأذنان} تسمعان رنّة الملعقة، واللسان --- وهو
أفضلها --- يذوقها.
\end{example}

\begin{remark}[اللمس في كل مكان]\label{rem:g1:the-five-senses:skin}
أربعة من \omterm{def:g1:the-five-senses:sense}{أعضاء الحس} تسكن في الرأس، أما عضو اللمس فهو \omterm{def:g1:the-five-senses:sense}{الجلد}،
\omterm{def:g1:the-five-senses:sense}{والجلد} يغطيك من رأسك إلى قدمك. ولهذا تحسّ بنسمة على عنقك، وبحصاة
في حذائك، وبدعسوقة تحطّ على ذراعك --- من غير أن تنظر.
\end{remark}

\section{الحواس ترفع تقاريرها إلى الدماغ}

\begin{proposition}[رسائل إلى الدماغ]\label{prop:g1:the-five-senses:brain}
\omterm{def:g1:the-five-senses:sense}{عضو الحس} لا يفهم ما يحسّ به --- بل \emph{يرسل رسالة}. وتسافر
الرسائل كلها داخل الجسم إلى \emph{الدماغ}\index{دماغ} في الرأس،
فيجمعها معًا: ``هذا صوت الجدة''، ``هذه رائحة الخبز المحمّص''.
ولولا الدماغ لما رفعت \omterm{def:g1:the-five-senses:sense}{العينان} \omterm{def:g1:the-five-senses:sense}{والأذنان} تقاريرها إلى أحد.
\end{proposition}
\admitted

\begin{example}[تسمية الأشياء في الظلام]\label{ex:g1:the-five-senses:dark}
يضع صديق شيئًا مألوفًا في يديك وعيناك مغمضتان: مفتاح، أو إسفنجة،
أو ملعقة. فتسميه في الحال تقريبًا. لقد أرسل \omterm{def:g1:the-five-senses:sense}{الجلد} رسالته --- بارد،
صلب، مسنّن --- فأجاب الدماغ: ``مفتاح''. الحواس تجمع، والدماغ يفهم.
\end{example}

\begin{remark}[يعين بعضها بعضًا]\label{rem:g1:the-five-senses:together}
تعمل الحواس فريقًا واحدًا. فالطعام ``يفقد طعمه'' حين يسدّ الزكام
أنفك، لأن الشمّ كان يؤدي نصف عمل النكهة. وحين لا تستطيع حاسة أن
تؤدي عملها البتة، تتعلم الأخريات أن تؤدي أكثر: فمن لا يبصرون
يقرؤون \omterm{def:g1:animals-around-us:parts}{بأطراف} أصابعهم باللمس، ويعبرون الشارع بآذانهم.
\end{remark}

\section{العناية بأعضاء الحس}

\begin{proposition}[أبواب هشّة]\label{prop:g1:the-five-senses:fragile}
\omterm{def:g1:the-five-senses:sense}{أعضاء الحس} ثمينة وسهلة الأذى. فالصوت الشديد العلو يؤذي \omterm{def:g1:the-five-senses:sense}{الأذنين}؛
والنظر إلى الشمس يؤذي \omterm{def:g1:the-five-senses:sense}{العينين}؛ والملعقة الشديدة السخونة تحرق
اللسان. \omterm{def:g1:the-five-senses:sense}{وعضو الحس} المصاب قد يبقى مصابًا --- ولذلك نحميها.
\end{proposition}
\admitted

\begin{method}[حماية حواسي]\label{met:g1:the-five-senses:protect}
\begin{enumerate}
  \item لا تنظر إلى الشمس مباشرة أبدًا، ولو للحظة؛
  \item وأبقِ الأصوات هادئة: لا صياح في \omterm{def:g1:the-five-senses:sense}{الآذان}، والسماعات على صوت
        منخفض؛
  \item ولا تدخل شيئًا في أذنك أو أنفك أبدًا؛
  \item وانفخ في الطعام الساخن قبل ذوقه؛
  \item واقرأ وارسم في ضوء جيد.
\end{enumerate}
\end{method}

\begin{example}[الحفل في الجوار]\label{ex:g1:the-five-senses:loud}
بعد موسيقى شديدة العلو، تصفر \omterm{def:g1:the-five-senses:sense}{الأذنان} أحيانًا بعض الوقت. وذلك
الصفير هو \omterm{def:g1:the-five-senses:sense}{الأذن} تقول: ``هذا أكثر من اللازم!''. ومن يعملون بآلات
صاخبة يلبسون واقيات \omterm{def:g1:the-five-senses:sense}{للأذنين} لهذا السبب بعينه --- فآذانهم يجب أن
تكفيهم العمر كله.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:1}
سمِّ الحواس الخمس، واذكر لكل واحدة عضو حسها.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:2}
أي \omterm{def:g1:the-five-senses:sense}{عضو حس} يخبر: (أ) بلون بالون؟ (ب) بنعومة بطانية؟ (ج) برائحة
المطر؟ (د) بصوت جرس؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:3}
أي \omterm{def:g1:the-five-senses:sense}{عضو حس} ليس في الرأس؟ وإلى أين يمتد؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:4}
في \cref{ex:g1:the-five-senses:dark} كانت \omterm{def:g1:the-five-senses:sense}{العينان} مغمضتين. فأي
عضو أرسل الرسالة، وأي جزء من الجسم فهمها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:5}
مع الزكام وانسداد الأنف، يبدو أن الطعام يفقد نكهته. فأي حاستين
تتعاونان عادة حين تأكل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:6}
اذكر ثلاث قواعد من \cref{met:g1:the-five-senses:protect}، وقل أي
\omterm{def:g1:the-five-senses:sense}{عضو حس} تحميه كل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:7}
لماذا يلبس من يعملون بآلات صاخبة واقيات \omterm{def:g1:the-five-senses:sense}{للأذنين}؟ استعمل
\cref{prop:g1:the-five-senses:fragile}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:8}
جرّب: أغمض عينيك، ودع مساعدًا يمسك ثلاثة أطعمة تحت أنفك --- موزًا،
وجبنًا، وبرتقالة مثلًا. كم واحدًا سمّته أنفك على صواب، بلا عون من
\omterm{def:g1:the-five-senses:sense}{العينين}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:the-five-senses:9}
يزعم صديق: ``\omterm{def:g1:the-five-senses:sense}{العين} تفهم ما تراه.'' صحّح الجملة مستعملًا
\cref{prop:g1:the-five-senses:brain}: ماذا تفعل \omterm{def:g1:the-five-senses:sense}{العين} حقًّا، ومن
الذي يفهم؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:the-five-senses:10}
كيف يستطيع من لا يبصر أن ``يقرأ'' كتابًا، وأن يعبر الشارع بأمان؟
اذكر أي الحواس تتولى أي الأعمال.
\end{exercise}
