\chapter{النظم البيئية والبنية الغذائية}\label{ch:b1:ecosystem-organization}

نبع في سهل كلسي يصبّ ماءً من صفائه يمكن عدّ كل نبتة وسمكة وسلحفاة
فيه من قارب. ويسقط ضوء الشمس على عشب البحر بمقدار \qty{1.7}{} مليون
كيلوسعرة في المتر المربع في السنة؛ فيخزن العشب واحدًا في المئة منها؛
وتنال الحلزونات والسلاحف التي تأكل العشب سدس ذلك؛ وتنال الأسماك
التي تأكل الحلزونات عُشره ثانيةً؛ ويعيش البلشون الوحيد في القمة على
جزء من مئة ألف مما سقط على الماء. وكل \omterm{def:b1:ecosystem-organization:ecosystem}{نظام بيئي} مبني هكذا: تيار
طاقة يدخل مرة واحدة، وينتقل من الآكلين إلى المأكولين، ويغادر
حرارة، بينما تُعاد دورة المادة التي يحركها. ويعرّف هذا الفصل \omterm{def:b1:ecosystem-organization:ecosystem}{النظام
البيئي} وأجزاءه، \omterm{def:b1:ecosystem-organization:niche}{والمكانة} التي يشغلها كل نوع، \omterm{def:b1:ecosystem-organization:trophic}{والمستويات الغذائية}
والشبكات الغذائية التي تسري الطاقة فيها، والإنتاجية التي تحدد حجم
الكل، وطرائق عدّ كم صنفًا من الأشياء يعيش فيه.

\section{النظام البيئي والمكانة}

\begin{definition}[النظام البيئي والوسط الحيوي والمجتمع]\label{def:b1:ecosystem-organization:ecosystem}
\index{نظام بيئي}\index{وسط حيوي}\index{مجتمع حيوي}
\emph{النظام البيئي} مجموع \omterm{def:b1:organism-environment:organism}{الكائنات} التي تعيش في مكان محدد مع
البيئة الفيزيائية التي تشغلها وتتبادل معها: أي \emph{المجتمع}
(أو البيوسينوز) --- وهو كل \omterm{def:b1:populations:population}{جمهرات} كل الأنواع --- و\emph{الوسط
الحيوي} --- وهو التربة والماء والهواء والضوء والمناخ. وحدوده
تُختار بحسب السؤال: جذع متعفن، أو بركة، أو غابة، أو حوض محيطي، أو
المحيط الحيوي كله. والنظام البيئي جملة مفتوحة
(\cref{ch:b1:organism-environment}): تسري الطاقة خلاله، وتدور
المادة داخله وعبر حدوده، وبنيته هي نمط من يأكل من.
\end{definition}

\begin{definition}[الموئل والمكانة]\label{def:b1:ecosystem-organization:niche}
\index{مكانة بيئية}\index{موئل}
\emph{موئل} النوع هو حيث يعيش؛ و\emph{مكانته} هي كيف يعيش: أي
مجموع الشروط التي يحتملها والموارد التي يستعملها ---
الحرارة والرطوبة والضوء والغذاء وزمن النشاط ومواضع التعشيش ---
وكل واحد منها بُعد يشغل النوع مدى عليه. و\emph{المكانة الأساسية}
هي المدى الذي يستطيع أن يشغله وحده؛ و\emph{المكانة المحققة} هي
الجزء الذي يشغله فعلًا بحضور المنافسين والمفترسات
(\cref{ch:b1:species-interactions}).
ونوعان لهما المكانة نفسها لا يستطيعان التعايش في مكان واحد طويلًا؛
ومكانات الأنواع المتعايشة مختلفة، والمجتمع مجموعة من المكانات
مركّبة معًا.
\end{definition}

\begin{example}[خمس هوازج في تنوبة واحدة]\label{ex:b1:ecosystem-organization:warblers}
خمسة أنواع من الهازجة الصغيرة الآكلة للحشرات تعشش في غابات التنوب
نفسها وتأكل الحشرات نفسها؛ وبمراقبة دقيقة يتبين أن كل نوع يعتلف في
جزء مختلف من الشجرة --- القمة، والإبر الجديدة الخارجية، والأغصان
الوسطى، والجذع والأغصان السفلى، والأرض تحتها ---
وعلى ارتفاعات وأوقات مختلفة. فموئلها واحد؛ ومكاناتها خمس، وتحمل
الغابة خمسة أنواع حيث يوحي الغذاء وحده بواحد.
\end{example}

\section{البنية الغذائية}

\begin{definition}[المنتِجات والمستهلِكات والمحلِّلات]\label{def:b1:ecosystem-organization:trophic}
\index{مستوى غذائي}\index{منتِج}\index{مستهلِك}\index{محلِّل}
تُصنَّف \omterm{def:b1:organism-environment:organism}{كائنات} \omterm{def:b1:ecosystem-organization:ecosystem}{النظام البيئي} بحسب ما تأكله في \emph{مستويات
غذائية}. وتصنع \emph{المنتِجات} (ذاتيات التغذي: النباتات والطحالب
والبكتيريا الزرقاء وبكتيريا التخليق الكيميائي) تصنع المادة العضوية
من غير العضوية؛ وهي المستوى الأول. و\emph{المستهلِكات} تأكلها:
فتأكل \emph{المستهلِكات الأولية} (آكلات العشب) المنتِجات،
و\emph{المستهلِكات الثانوية} (آكلات اللحوم) تأكل آكلات العشب،
و\emph{الثالثية} تأكل آكلات اللحوم؛ وقارتات اللحم والعشب تأكل عند
مستويات عدة. و\emph{المحلِّلات} (البكتيريا والفطريات)
و\emph{آكلات الفتات} (ديدان الأرض وقمل \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} وخنافس الروث) تعيش على
المادة الميتة والفضلات من كل مستوى، وتعيد معادنها إلى التربة والماء
لتلتقطها المنتِجات ثانية --- وهي الوصلة التي تغلق دورة المادة.
\end{definition}

\begin{definition}[السلسلة الغذائية والشبكة الغذائية]\label{def:b1:ecosystem-organization:foodweb}
\index{سلسلة غذائية}\index{شبكة غذائية}
\emph{السلسلة الغذائية} متتالية من \omterm{def:b1:organism-environment:organism}{الكائنات} يأكل كل واحد منها الذي
قبله: عشب، فجندب، فضفدع، فأفعى، فصقر. وفي مجتمع حقيقي يأكل كل نوع
عدة أنواع ويأكله عدة أنواع، فتتشابك السلاسل في \emph{شبكة غذائية}
تشير أسهمها من المأكول إلى الآكل --- أي في اتجاه الطاقة. والسلاسل
قصيرة: نادرًا ما تزيد على أربع أو خمس وصلات، لسبب يعطيه القسم
التالي.
\end{definition}

\section{سريان الطاقة}

\begin{definition}[الإنتاجية]\label{def:b1:ecosystem-organization:productivity}
\index{إنتاجية أولية}\index{إنتاجية أولية إجمالية}\index{إنتاجية أولية صافية}
\emph{الإنتاجية الأولية الإجمالية} في \omterm{def:b1:ecosystem-organization:ecosystem}{نظام بيئي} هي السرعة التي
تثبّت بها منتِجاته الطاقةَ بالتركيب الضوئي، في وحدة المساحة والزمن
(\unit{kJ.m^{-2}.yr^{-1}}، أو غرامات الكربون أو المادة الجافة)؛
و\emph{الإنتاجية الأولية الصافية} هي ما يبقى بعد تنفس \omterm{def:b1:ecosystem-organization:trophic}{المنتِجات}
نفسها --- أي المادة العضوية المتاحة فعلًا لبقية الجملة.
و\emph{الإنتاجية الثانوية} هي السرعة التي تبني بها \omterm{def:b1:ecosystem-organization:trophic}{المستهلِكات}
كتلتها الحيوية مما تأكله. و\emph{الكتلة الحيوية} هي المخزون
القائم، أي الكتلة الحاضرة في لحظة؛ فالإنتاجية دفق، والكتلة الحيوية
مخزون، ونسبتهما هي زمن الدوران.
\end{definition}

\begin{theorem}[قاعدة العشرة في المئة]\label{thm:b1:ecosystem-organization:lindeman}
\index{كفاءة غذائية}
من الطاقة الداخلة إلى \omterm{def:b1:ecosystem-organization:trophic}{مستوى غذائي} واحد غذاءً، لا يبلغ المستوى
التالي غذاءً له إلا نحو العُشر --- بين \qty{5}{} و\qty{20}{\%}.
والباقي لا يُؤكل، أو لا يُمثَّل (فيغادر برازًا)، أو يتنفسه المستوى
نفسه. فالطاقة المتاحة إذن تهبط عشرة أمثال في كل مستوى، ولهذا نادرًا
ما تتجاوز السلاسل الغذائية أربع أو خمس وصلات، ولهذا تكون كتلة
المفترسات العليا وأعدادها صغيرة.
\end{theorem}
\begin{proof}[برهان جزئي]
الطاقة التي يمثّلها مستوى هي ما يأكله ناقص ما يطرحه؛ ومن ذلك يأخذ
التنفس ما ينفقه المستوى على العيش
(\cref{ch:b1:organism-environment})، ولا يكون متاحًا للأكل إلا
الباقي --- أي نموه وتكاثره. وكل واحد من الكسور الثلاثة (المأكول،
والممثَّل، والمحوَّل إلى كتلة حيوية) أدنى من واحد بكثير: فآكل العشب
في مرج يأكل نحو \qty{20}{\%} من نمو النبات، ويمثّل \qty{50}{\%} من
ذلك، ويحوّل \qty{10}{\%} من ذلك إلى لحم، فيعطي \qty{1}{\%}؛ وآكل
اللحم، وهو يأكل ويمثّل من فريسته أكثر، نحو
\qty{10}{\%}. \omterm{def:b1:nucleic-acids:nucleotide}{والقاعدة} انتظام ملاحَظ لا قانون؛ ومقدارها هو ما
تعطيه القياسات.
\end{proof}
\begin{proof}[شواهد]
قاس لينديمان (1942)، في بحيرة صغيرة، الطاقة التي تثبّتها العوالق
والطاقة المحفوظة والمتنفَّسة عند كل مستوى فوقها، فوجد كل مستوى
يمرر نحو العُشر. ورسم أودوم (1957) الميزانية الكاملة لنبع في
فلوريدا: فمن \qty{1.7e6}{kcal} في المتر المربع في
السنة من ضوء الشمس، ثبّتت \omterm{def:b1:ecosystem-organization:trophic}{المنتِجات} \num{20800} (إجماليًا)، وتنفست
\num{12000} وتركت \num{8800} صافيًا؛ وأخذت آكلات العشب \num{3400}
ومررت \num{380} إلى آكلات اللحوم؛ ومررت آكلات اللحوم
\num{20} إلى آكلات اللحوم العليا. وكانت الكفاءات \qty{1.2}{\%}
و\qty{16}{\%} و\qty{11}{\%} و\qty{5}{\%} --- أي قاعدة العشرة في المئة
ضمن معامل اثنين عند كل خطوة.
\end{proof}

\begin{proposition}[الأهرام]\label{prop:b1:ecosystem-organization:pyramids}
\index{هرم بيئي}
\omterm{prop:b1:nucleic-acids:stability}{تكديس} \omterm{def:b1:ecosystem-organization:trophic}{المستويات الغذائية} يعطي \emph{هرمًا}. فهرم \emph{الطاقة}
(الدفق في كل مستوى) قائم دائمًا: إذ لا يستطيع أي مستوى أن يمرر إلا
أقل مما تلقاه. وهرم \emph{الكتلة الحيوية} (المخزون القائم) قائم
عادة لكنه قد ينقلب: ففي عرض البحر تكون العوالق النباتية، وهي
تُؤكل بسرعة نموها، أخفّ في أي لحظة من العوالق الحيوانية التي تعيش
عليها، لأنها تدور في أيام بينما تدوم الحيوانات شهورًا. وهرم
\emph{الأعداد} قد يأخذ أي شكل: فبلوطة واحدة تطعم آلاف اليرقات.
فالطاقة هي المقياس الصادق.
\end{proposition}

\begin{method}[وضع ميزانية طاقة]\label{met:b1:ecosystem-organization:budget}
\begin{enumerate}
  \item قِس الإنتاجية الإجمالية: بالتبادل الغازي (الأكسجين المتحرر
        أو $\mathrm{CO_2}$ الملتقط في قطعة أو في قنينة في الضوء وفي
        الظلام: فالضوء ناقص الظلام يعطي الإجمالي، والضوء وحده يعطي
        الصافي)، أو بالحصاد (الكتلة الحيوية المكتسبة في السنة زائد
        ما أُكل وما تساقط).
  \item قِس، لكل مستوى \omterm{def:b1:ecosystem-organization:trophic}{مستهلِك}، المدخول (ما يأكله)، والتمثيل
        (المدخول ناقص البراز)، والتنفس (استهلاكه للأكسجين)، والإنتاج
        (النمو زائد النسل $=$ التمثيل ناقص التنفس).
  \item \omterm{thm:b1:ecosystem-organization:lindeman}{الكفاءة الغذائية} بين مستويين $=$ إنتاج المستوى الأعلى
        $/$ إنتاج المستوى الأدنى. وتحقق من أن التنفس زائد
        الإنتاج زائد التحلل في كل مستوى يجمع إلى تمثيله.
  \item حوّل إلى وحدات مشتركة (\unit{kJ}، أو غرامات الكربون عند
        نحو \qty{40}{kJ/g}) وارسم الهرم؛ ومجموع كل التنفس ينبغي أن
        يقارب الإنتاجية الإجمالية على مدى سنة في \omterm{def:b1:ecosystem-organization:ecosystem}{نظام بيئي} مستقر.
\end{enumerate}
\end{method}

\begin{example}[لماذا القمة نحيلة]\label{ex:b1:ecosystem-organization:top}
كيلومتر مربع من السافانا يصنع نحو \qty{5000}{t} من العشب في
السنة؛ وتحوّله الراعيات إلى \qty{50}{t} من الظباء والجاموس؛

ما يزول حين ينكمش موئلها.
\end{example}

\begin{proposition}[إنتاجية المناطق الحيوية]\label{prop:b1:ecosystem-organization:biomes}

\section{عدّ التنوع}

\begin{definition}[الغنى النوعي ودليل التنوع]\label{def:b1:ecosystem-organization:diversity}
\index{غنى نوعي}\index{دليل شانون}
\emph{الغنى النوعي} $S$ في مجتمع هو عدد الأنواع فيه.
والغنى وحده يتجاهل الوفرة: فغابة فيها مئة بلوطة وواحدة من كل
شجرة من تسع شجرات أخرى أقل تنوعًا من غابة فيها عشر من كل
واحدة. و\emph{دليل شانون}
\[
H = -\sum_{i=1}^{S} p_i \ln p_i ,
\]
حيث $p_i$ كسر الأفراد المنتمين إلى النوع $i$،
يرتفع مع الغنى ومع \emph{التساوي} معًا؛ وهو $\ln S$ حين تتساوى
وفرة كل الأنواع ويقارب الصفر حين يسود نوع واحد. و\emph{التساوي}
هو $H/\ln S$. والتنوع يرتفع عمومًا من القطبين
إلى المدارين، ومع المساحة، ومع الزمن منذ آخر اضطراب،
ومع إنتاجية الموقع إلى حد.
\end{definition}

\begin{method}[أخذ عينات من مجتمع]\label{met:b1:ecosystem-organization:sampling}
\begin{enumerate}
  \item خذ العينات بطريقة تناسب \omterm{def:b1:organism-environment:organism}{الكائنات} (مربعات عيّنة، أو شباك،
        أو مصائد، أو مقاطع خطية) وسجّل عدد أفراد كل
        نوع.
  \item ارسم عدد الأنواع الموجودة في مقابل عدد العينات
        (أو الأفراد): فيرتفع المنحني بحدة ثم
        ينبسط؛ وحيث ينبسط يكون الغنى قد اكتمل تقريبًا.
        ولا تقارن بين مجتمعات إلا بجهد أخذ عينات متساوٍ.
  \item احسب $H$ والتساوي؛ وقارنهما بمخطط الرتبة والوفرة
        (الأنواع مرتبة بالوفرة، على سلّم لوغاريتمي)، الذي يبيّن
        ميلُه السيادةَ.
  \item فسّر: فالتساوي المنخفض يشير إلى نوع سائد أو إلى
        إجهاد؛ وهبوط $H$ مع الزمن يشير إلى تغير في
        \omterm{def:b1:ecosystem-organization:ecosystem}{الوسط الحيوي}.
\end{enumerate}
\end{method}

\begin{example}[غابتان]\label{ex:b1:ecosystem-organization:twowoods}
الغابة A: مئة بلوطة وواحدة من كل شجرة من تسع أخرى؛ فيكون $H = -(0.917\ln
0.917 + 9\times 0.0092\ln 0.0092) = 0.08 + 0.39 = 0.47$، والتساوي
$0.47/2.30 = 0.20$. والغابة B: عشر من كل نوع من الأنواع العشرة نفسها؛ فيكون $H =
\ln 10 = 2.30$، والتساوي 1. أي الغنى نفسه، وخمسة أمثال التنوع.
\end{example}

\section{مسألة: ميزانية نبع}

\begin{problem}[{مسألة نهاية الأسبوع --- نبع أودوم يُوضع مستوًى
مستوًى: من الشمس إلى العشب، ومن العشب إلى الحلزون، ومن الحلزون إلى
السمكة، ومن السمكة إلى البلشون، مع التنفس والمحلِّلات، وختامًا
الكفاءات الغذائية عند كل مستوى}]
\label{pb:b1:ecosystem-organization:1}
نبع يتلقى \qty{1.7e6}{kcal} من ضوء الشمس في المتر المربع في
السنة. وتثبّت منتِجاته \qty{20800}{kcal} (إجماليًا) وتتنفس
\qty{12000}{}. وتأكل آكلات العشب \qty{3400}{kcal} من المادة
النباتية، تمثّل منها \qty{1500}{}، وتتنفس \qty{1100}{}، وتحوّل
الباقي إلى نمو ونسل. وتأكل آكلات اللحوم \qty{380}{kcal}، وتتنفس
\qty{300}{} وتنتج الباقي؛ وتأكل آكلات اللحوم العليا \qty{20}{kcal}،
وتتنفس \qty{14}{} وتنتج \qty{6}{}. وكل ما لا يُؤكل ولا يُتنفس يذهب
إلى \omterm{def:b1:ecosystem-organization:trophic}{المحلِّلات} التي تتنفسه. وخذ
\qty{1}{kcal} $= \qty{4.18}{kJ}$ و\qty{10}{kcal} لكل غرام من الكتلة
الحيوية الجافة.

\medskip
\noindent\textbf{الجزء الأول --- \omterm{def:b1:ecosystem-organization:trophic}{المنتِجات}.}
\begin{enumerate}
  \item احسب \omterm{def:b1:ecosystem-organization:productivity}{الإنتاجية الأولية الصافية}.
  \item احسب كفاءة التركيب الضوئي: الإنتاج الإجمالي
        على ضوء الشمس. وقارنها بالقيمة \qty{34}{\%} في
        \cref{ch:b1:photosynthesis} واسرد ثلاثة فواقد
        تفصل بينهما.
  \item أي كسر من الإنتاج الإجمالي تتنفسه
        \omterm{def:b1:ecosystem-organization:trophic}{المنتِجات}؟
  \item كم من الإنتاج الصافي تأكله آكلات العشب، وكم
        يذهب مباشرة إلى \omterm{def:b1:ecosystem-organization:trophic}{المحلِّلات}؟
  \item عبّر عن الإنتاجية الصافية بغرامات المادة الجافة في المتر
        المربع في السنة وضع النبع بين المناطق الحيوية في الفصل.
  \item كم من ضوء الشمس، في المتر المربع وفي السنة، يغادر
        النبع حرارة دون أن يمر قط \omterm{def:b1:organism-environment:organism}{بكائن حي}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- \omterm{def:b1:ecosystem-organization:trophic}{المستهلِكات}.}
\begin{enumerate}[resume]
  \item احسب إنتاج آكلات العشب وكفاءة تمثيلها
        (الممثَّل على المأكول).
  \item احسب كفاءة إنتاج آكلات العشب
        (الإنتاج على الممثَّل).
  \item احسب \omterm{thm:b1:ecosystem-organization:lindeman}{الكفاءة الغذائية} من \omterm{def:b1:ecosystem-organization:trophic}{المنتِجات} إلى آكلات العشب:
        إنتاج آكلات العشب على الإنتاج الأولي الصافي.
  \item احسب إنتاج آكلات اللحوم \omterm{thm:b1:ecosystem-organization:lindeman}{والكفاءة الغذائية}
        من آكلات العشب إلى آكلات اللحوم.
  \item احسب \omterm{thm:b1:ecosystem-organization:lindeman}{الكفاءة الغذائية} من آكلات اللحوم إلى العليا منها.
  \item كم طاقة في المتر المربع في السنة تبلغ \omterm{def:b1:body-plans-tissues:tissue}{أنسجة} آكل اللحم
        الأعلى نفسها؟ وأي كسر من ضوء الشمس هي؟
  \item لماذا ترتفع كفاءة التمثيل من آكلات العشب إلى
        آكلات اللحوم؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{def:b1:ecosystem-organization:trophic}{المحلِّلات} والميزان.}
\begin{enumerate}[resume]
  \item اسرد كل دفق إلى \omterm{def:b1:ecosystem-organization:trophic}{المحلِّلات} (الإنتاج الصافي غير المأكول،
        وبراز آكلات العشب وإنتاجها غير المأكول، وهكذا)
        واجمعها.
  \item احسب التنفس الكلي \omterm{def:b1:ecosystem-organization:ecosystem}{للنظام البيئي}: \omterm{def:b1:ecosystem-organization:trophic}{المنتِجات}،
        وآكلات العشب، وآكلات اللحوم، والعليا منها، \omterm{def:b1:ecosystem-organization:trophic}{والمحلِّلات}.
  \item قارنه بالإنتاج الإجمالي. فماذا تعني المقارنة
        لكتلة النبع الحيوية على مدى سنة؟
  \item أي كسر من الإنتاج الإجمالي تتنفسه
        \omterm{def:b1:ecosystem-organization:trophic}{المحلِّلات}؟
  \item ارسم هرم الطاقة بإنتاج المستويات الأربعة
        وعنون الكفاءات.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- المخزونات والدوران والتغير.}
الكتلة الحيوية القائمة \qty{800}{g/m^2} من \omterm{def:b1:ecosystem-organization:trophic}{المنتِجات}، و\qty{40}{g/m^2}
من آكلات العشب، و\qty{10}{g/m^2} من آكلات اللحوم، و\qty{1.5}{g/m^2} من
العليا منها.
\begin{enumerate}[resume]
  \item حوّل كل كتلة حيوية إلى كيلوسعرات وارسم هرم
        الكتلة الحيوية. أهو قائم؟
  \item احسب زمن الدوران (الكتلة الحيوية على الإنتاج) لكل مستوى.
        فأيها أسرع دورانًا، ولماذا؟
  \item بلشون كتلته \qty{2}{kg} يعيش على إنتاج آكلات اللحوم العليا
        في النبع. فكم مترًا مربعًا يحتاج؟
  \item يُخصَّب النبع بالسماد فيتضاعف إنتاجه الإجمالي، لكن آكلات
        العشب لا تستطيع أن تأكل أسرع.
        فماذا يحدث للإنتاج الزائد، ولتنفس \omterm{def:b1:ecosystem-organization:trophic}{المحلِّلات}،
        ولأكسجين الماء ليلًا؟
  \item تُدخَل سمكة جديدة تأكل بيوض آكلات العشب
        فتنصّف إنتاجها. تنبأ بإنتاج آكلات اللحوم
        وبأعداد البلاشين، وبمصير عشب البحر.
  \item فسّر لماذا تغيّر إزالة آكلات اللحوم العليا عشبَ البحر
        أكثر بكثير مما تغيّره إزالة ما يكافئها من طاقة آكلات
        العشب.
  \item اذكر النتيجة: الكفاءات الغذائية عند كل من الانتقالات
        الثلاثة، وكفاءة ضوء الشمس إلى الإنتاج الإجمالي، وكسر طاقة
        الشمس الذي يبلغ آكلات اللحوم العليا.
\end{enumerate}
\end{problem}
