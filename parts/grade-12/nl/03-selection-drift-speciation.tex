\chapter{Selectie, drift en soortvorming}\label{ch:g12:selection-drift-speciation}

In 1848 werd bij een industriestad, waar elke boomstam met roet was
bedekt, een zwarte vorm van de berkenspanner gevangen; in 1895 was
vrijwel elke vlinder in de streek zwart. Een eeuw later, met het roet
verdwenen en de stammen weer bleek, was de zwarte vorm bijna weg. Niemand
kweekte de vlinders; de vogels die hen eten deden het sorteerwerk. De
vorige twee hoofdstukken beschreven hoe variatie ontstaat; dit hoofdstuk
gaat over wat er in een \omterm{def:g12:selection-drift-speciation:population}{populatie} mee gebeurt --- hoe sommige \omterm{def:g10:universal-dna:gene}{allelen}
algemeen worden en andere zeldzaam, door selectie en door toeval --- en
hoe uiteindelijk één \omterm{def:g12:selection-drift-speciation:population}{populatie} twee \omterm{def:g10:biodiversity-scales:species}{soorten} wordt.

\section{Populaties en hun allelen}

\begin{definition}[Populatie, allelfrequentie]\label{def:g12:selection-drift-speciation:population}
Een \emph{populatie}\index{populatie} is het geheel van individuen van
één \omterm{def:g10:biodiversity-scales:species}{soort} die op dezelfde plaats leven en zich onderling voortplanten.
Haar genetische samenstelling wordt beschreven door de
\emph{frequenties}\index{allelfrequentie} van de \omterm{def:g10:universal-dna:gene}{allelen} van elk \omterm{def:g10:universal-dna:gene}{gen}:
voor een \omterm{def:g10:universal-dna:gene}{gen} met twee \omterm{def:g10:universal-dna:gene}{allelen} $A$ en $a$, het deel $p$ van alle kopieën
dat $A$ is en het deel $q = 1 - p$ dat $a$ is. Evolutie is op deze schaal
een verandering van allelfrequenties van de ene generatie op de volgende.
\end{definition}

\begin{proposition}[Frequenties zonder enige kracht]\label{prop:g12:selection-drift-speciation:hw}
In een grote \omterm{def:g12:selection-drift-speciation:population}{populatie} waarin willekeurig wordt gepaard, waarin geen
\omterm{def:g10:universal-dna:gene}{allel} een voordeel geeft en waarin geen \omterm{def:g11:mutations:mutation}{mutatie} of migratie optreedt,
blijven de \omterm{def:g12:selection-drift-speciation:population}{allelfrequenties} van generatie op generatie gelijk en
verschijnen de \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypen} in de verhoudingen
\[
AA : Aa : aa = p^2 : 2pq : q^2 .
\]
Dat is de referentietoestand: elke afwijking ervan, over de generaties
waargenomen, is het teken dat een van de voorwaarden niet is vervuld ---
dat selectie, toeval, migratie of \omterm{def:g11:mutations:mutation}{mutatie} aan het werk is.
\end{proposition}
\begin{proof}
Elke gameet draagt $A$ met kans $p$ en $a$ met kans $q$; twee
willekeurig getrokken gameten geven $AA$ met kans $p^2$, $aa$ met $q^2$,
en $Aa$ met $pq + qp = 2pq$. Als je de \omterm{def:g10:universal-dna:gene}{allelen} van de nieuwe generatie
telt, vormen de kopieën van $A$ $p^2 + \tfrac12 (2pq) =
p(p + q) = p$ van het totaal. De frequentie is onveranderd.
\end{proof}

\begin{example}[Allelen tellen]\label{ex:g12:selection-drift-speciation:count}
Van 1000 mensen zijn er 490 $AA$, 420 $Aa$ en 90 $aa$. Kopieën van $a$:
$420 + 2 \times 90 = 600$ op 2000, dus $q = 0.3$ en $p = 0.7$ --- en de
genotype-aantallen zijn precies $p^2$, $2pq$ en $q^2$ van 1000: de
\omterm{def:g12:selection-drift-speciation:population}{populatie} verkeert voor dit \omterm{def:g10:universal-dna:gene}{gen} in de referentietoestand. Een recessieve
ziekte die één pasgeborene op 2500 treft ($q^2 = 1/2500$) betekent
$q = 1/50$ en dragers $2pq \approx 1/25$: de cijfers uit
\cref{ch:g11:genes-and-disease}, afgeleid.
\end{example}

\section{Selectie}

\begin{definition}[Natuurlijke selectie]\label{def:g12:selection-drift-speciation:selection}
\emph{Natuurlijke selectie}\index{natuurlijke selectie} is het verschil
in voortplanting tussen individuen die verschillende \omterm{def:g10:universal-dna:gene}{allelen} dragen, in
een gegeven omgeving: de dragers van het ene \omterm{def:g10:universal-dna:gene}{allel} laten gemiddeld meer
nakomelingen na dan de dragers van het andere, en de frequentie van dat
\omterm{def:g10:universal-dna:gene}{allel} stijgt van generatie op generatie. Wat wordt geselecteerd is het
hele individu --- zijn overleving, zijn paring, zijn vruchtbaarheid ---
en de \omterm{def:g10:universal-dna:gene}{allelen} liften mee.
\end{definition}

\begin{proposition}[Selectie verandert frequenties in een richting]\label{prop:g12:selection-drift-speciation:direction}
Een \omterm{def:g10:universal-dna:gene}{allel} waarvan de dragers ook maar iets meer voortplantingssucces
hebben, verbreidt zich; een \omterm{def:g10:universal-dna:gene}{allel} waarvan de dragers minder hebben, wordt
zeldzaam. De richting wordt door de omgeving bepaald en keert om als die
omkeert; de snelheid hangt af van de grootte van het voordeel en van de
vraag of het \omterm{def:g10:universal-dna:gene}{allel} bij één kopie tot uiting komt (\omterm{def:g11:genes-and-disease:genetic}{dominant}, snel) of pas
bij twee (\omterm{def:g11:genes-and-disease:genetic}{recessief}, aanvankelijk traag, en nooit helemaal weg te krijgen,
omdat zijn zeldzame kopieën zich in heterozygoten verschuilen).
\end{proposition}
\begin{proof}[Bewijsmateriaal]
De berkenspanner: vogels pakken de vlinders die zij zien; op beroete
stammen valt de bleke vorm op, op schone stammen de donkere; de frequentie
van het donkere \omterm{def:g10:universal-dna:gene}{allel} steeg in vijftig jaar vervuiling van vrijwel nul
naar 98\% en zakte in veertig jaar schone lucht terug tot onder 10\%,
precies mee met de stammen. \omterm{prop:g11:antibiotic-resistance:mechanisms}{Antibioticaresistentie}
(\cref{ch:g11:antibiotic-resistance}): in aanwezigheid van het middel
planten alleen de dragers van het bestendige \omterm{def:g10:universal-dna:gene}{allel} zich voort.
Lactasepersistentie (\cref{ch:g11:enzymes-and-phenotype}): het \omterm{def:g10:universal-dna:gene}{allel} is
juist algemeen waar melk al duizenden jaren voedsel voor volwassenen is.
In elk geval volgt het lot van het \omterm{def:g10:universal-dna:gene}{allel} de omgeving.
\end{proof}

\begin{example}[Selectie door de andere sekse]\label{ex:g12:selection-drift-speciation:sexual}
De sleep van een pauw maakt hem trager en beter zichtbaar voor roofdieren,
en toch blijft die behouden omdat pauwinnen de mannetjes met de grootste
en regelmatigste sleep uitkiezen: een \omterm{def:g10:universal-dna:gene}{allel} dat de sleep verbetert, wordt
vaker doorgegeven, wat het aan overleving ook kost. De \emph{seksuele
selectie} --- de keuze van een partner --- brengt de versierselen, de
zangen en de gevechten van dieren voort, en kan een kenmerk in een
richting duwen die de overleving alleen nooit zou bevoordelen.
\end{example}

\section{Toeval: genetische drift}

\begin{proposition}[Genetische drift]\label{prop:g12:selection-drift-speciation:drift}
In elke \omterm{def:g12:selection-drift-speciation:population}{populatie} van eindige omvang zijn de \omterm{def:g10:universal-dna:gene}{allelen} van de ene generatie
een willekeurige greep uit de vorige: welke individuen zich toevallig
voortplanten, en welke \omterm{def:g10:universal-dna:gene}{allelen} hun gameten toevallig dragen, schommelt
door het toeval. De \omterm{def:g12:selection-drift-speciation:population}{allelfrequenties} dwalen dus van generatie op generatie
zonder dat er enig voordeel bij komt kijken --- de \emph{genetische
drift}\index{genetische drift}. Hoe kleiner de \omterm{def:g12:selection-drift-speciation:population}{populatie}, des te groter
het dwalen; in een kleine \omterm{def:g12:selection-drift-speciation:population}{populatie} kan een \omterm{def:g10:universal-dna:gene}{allel} binnen enkele
generaties door louter toeval verdwijnen, of 100\% halen, wat zijn waarde
ook is.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
\omterm{def:g12:selection-drift-speciation:population}{Populaties} die door een handvol individuen zijn gesticht --- een eiland
dat door een paar vogels wordt gekoloniseerd, een menselijke gemeenschap
die van enkele tientallen kolonisten afstamt --- dragen een vertekende
greep uit de \omterm{def:g10:universal-dna:gene}{allelen} van de bronpopulatie: een zeldzaam ziekteallel kan er
algemeen zijn, en een groot deel van de verscheidenheid van de bron
ontbreken. \omterm{def:g10:biodiversity-scales:species}{Soorten} die door een \emph{flessenhals} van heel weinig
overlevenden zijn gegaan (het jachtluipaard uit
\cref{ch:g10:biodiversity-scales}, de noordelijke zeeolifant die in 1890
tot twintig dieren was teruggebracht) vertonen vandaag vrijwel geen
\omterm{def:g10:biodiversity-scales:biodiversity}{genetische diversiteit}, ook al zijn zij tot duizenden hersteld. En in het
laboratorium waaieren vele kleine parallelle \omterm{def:g12:selection-drift-speciation:population}{populaties} die bij dezelfde
frequentie beginnen, binnen enkele generaties alle kanten op, terwijl
grote blijven staan.
\end{proof}

\begin{example}[Een stichterseffect]\label{ex:g12:selection-drift-speciation:founder}
Een eilandgemeenschap werd gesticht door enkele tientallen kolonisten, van
wie er één een \omterm{def:g10:universal-dna:gene}{allel} voor een zeldzame oogaandoening droeg. Tien
generaties later is de frequentie van dat \omterm{def:g10:universal-dna:gene}{allel} in de gemeenschap één op
tien --- honderd keer de waarde op het vasteland --- omdat één drager
onder dertig stichters al 1.7\% van de kopieën vormt, en de drift in een
kleine, afgezonderde \omterm{def:g12:selection-drift-speciation:population}{populatie} haar verder heeft verschoven. Het \omterm{def:g10:universal-dna:gene}{allel}
geeft geen enkel voordeel; zijn talrijkheid is een toevalligheid van wie
er aan boord ging.
\end{example}

\begin{method}[Selectie of drift?]\label{met:g12:selection-drift-speciation:which}
Sta je voor een verandering van een \omterm{def:g12:selection-drift-speciation:population}{allelfrequentie}:
\begin{enumerate}
  \item Schat de omvang van de \omterm{def:g12:selection-drift-speciation:population}{populatie}: drift is sterk onder een paar
        honderd en verwaarloosbaar bij miljoenen.
  \item Zoek naar een richting die de omgeving volgt (selectie), of naar
        een willekeurige wandeling (drift). Parallelle \omterm{def:g12:selection-drift-speciation:population}{populaties} die
        dezelfde kant op bewegen wijzen op selectie; bewegen zij
        verschillende kanten op, dan op drift.
  \item Zoek een mechanisme dat het \omterm{def:g10:universal-dna:gene}{allel} met overleving of voortplanting
        verbindt; is er geen, denk dan aan drift.
  \item Bedenk dat beide tegelijk werken: de selectie zet een tendens,
        de drift voegt ruis toe, en in kleine \omterm{def:g12:selection-drift-speciation:population}{populaties} kan de ruis de
        tendens overstemmen.
\end{enumerate}
\end{method}

\section{Van populatie tot soort}

\begin{proposition}[Soortvorming]\label{prop:g12:selection-drift-speciation:speciation}
Twee \omterm{def:g12:selection-drift-speciation:population}{populaties} van één \omterm{def:g10:biodiversity-scales:species}{soort} die geen \omterm{def:g10:universal-dna:gene}{genen} meer uitwisselen ---
gescheiden door een barrière van aardrijkskunde, van gedrag of van timing
--- evolueren uit elkaar: elk hoopt haar eigen \omterm{def:g11:mutations:mutation}{mutaties} op, wordt door
haar eigen omgeving geselecteerd en drift haar eigen kant op. Is het
uiteengroeien zo ver gevorderd dat individuen van de twee \omterm{def:g12:selection-drift-speciation:population}{populaties} zich
niet meer onderling voortplanten, of onvruchtbare nakomelingen geven, ook
wanneer zij elkaar weer ontmoeten, dan bestaan er twee \omterm{def:g10:biodiversity-scales:species}{soorten} waar er één
was. De \emph{soortvorming}\index{soortvorming} is dat proces; het kost
meestal duizenden tot miljoenen generaties, al bereikt de polyploïdie
(\cref{ch:g12:diversification-of-life}) het in één keer.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
Eilanden: elk van de Galapagoseilanden draagt zijn eigen vinken, die van
een vastelandsoort afstammen en het dichtst bij die van de naburige
eilanden staan. Meren: één voorouderlijke cichlide heeft in één Afrikaans
meer binnen enkele honderdduizenden jaren honderden \omterm{def:g10:biodiversity-scales:species}{soorten} opgeleverd,
gescheiden door leefgebied en door de kleurkeuze van de vrouwtjes.
Ringsoorten: \omterm{def:g12:selection-drift-speciation:population}{populaties} van een meeuw hebben zich rond het poolgebied
verbreid en planten zich de hele ring rond met hun buren voort, tot de
twee uiteinden elkaar in Europa ontmoeten als vormen die zich niet meer
onderling voortplanten --- een soortgrens die betrapt wordt terwijl zij
ontstaat. En \omterm{def:g10:biodiversity-scales:species}{soorten} die zich pas kortgeleden hebben gescheiden, zoals de
ijsbeer en de bruine beer, geven nog af en toe vruchtbare kruisingen: de
grens is een kwestie van graad.
\end{proof}

\begin{proposition}[De soort, opnieuw bekeken]\label{prop:g12:selection-drift-speciation:concept}
Een \omterm{def:g10:biodiversity-scales:species}{soort} is een \omterm{def:g12:selection-drift-speciation:population}{populatie}, of een geheel van \omterm{def:g12:selection-drift-speciation:population}{populaties}, waarvan de leden
zich onderling voortplanten en genetisch van andere zulke gehelen zijn
afgezonderd --- een definitie die voor de meeste dieren en planten op een
gegeven moment werkt, maar randen heeft: \omterm{def:g12:selection-drift-speciation:population}{populaties} die bezig zijn zich te
scheiden, \omterm{def:g10:biodiversity-scales:species}{soorten} die nog kruisen, organismen die zich zonder geslacht
voortplanten. Een \omterm{def:g10:biodiversity-scales:species}{soort} is geen vast type maar een lijn in de tijd: zij
begint wanneer een \omterm{def:g12:selection-drift-speciation:population}{populatie} afgezonderd raakt, bestaat zolang haar leden
zich onderling blijven voortplanten, en eindigt door uitsterven of door in
nieuwe \omterm{def:g10:biodiversity-scales:species}{soorten} uiteen te vallen. De \omterm{def:g10:biodiversity-scales:species}{soorten} die vandaag leven, zijn de
toppen van een boom waarvan de takken het onderwerp zijn van
\cref{ch:g12:phylogenetic-trees}.
\end{proposition}
\admitted

\begin{remark}[De evolutie, samengesteld]\label{rem:g12:selection-drift-speciation:assembled}
\omterm{def:g11:mutations:mutation}{Mutatie} en het schudden leveren variatie; verdubbeling, overdracht,
polyploïdie, regeling en \omterm{def:g12:diversification-of-life:symbiosis}{symbiose} leveren er meer; de selectie sorteert
haar naar de omgeving, de drift naar het toeval; afzondering laat
\omterm{def:g12:selection-drift-speciation:population}{populaties} uiteengroeien tot zij \omterm{def:g10:biodiversity-scales:species}{soorten} zijn; uitsterven neemt hen weg.
Geen van die stappen heeft een doel, en geen ervan kijkt vooruit; samen
hebben zij, over de vier miljard jaar van het fossielenarchief, de
verscheidenheid uit \cref{ch:g10:biodiversity-scales} voortgebracht. De
theorie die die stappen en hun samenspel benoemt, is de evolutietheorie,
en elk hoofdstuk van dit jaar is een stukje van haar bewijs geweest.
\end{remark}

\section{Probleem: De muizen van het eiland}

\begin{problem}[{Weekendprobleem --- een populatie muizen op een eiland:
haar allelfrequenties berekend, de selectie door een nieuw roofdier
gevolgd, de drift van een piepkleine kolonie nagebootst, en de geboorte
van een soort voorzien}]
\label{pb:g12:selection-drift-speciation:1}
Op een rotsachtig eiland dragen muizen een \omterm{def:g10:universal-dna:gene}{gen} voor de vachtkleur met twee
\omterm{def:g10:universal-dna:gene}{allelen}: $D$ (donker, \omterm{def:g11:genes-and-disease:genetic}{dominant}) en $d$ (bleek). Een telling van 2000
muizen vindt 720 donkere muizen met \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} $DD$, 960 donkere $Dd$ en
320 bleke $dd$.

\medskip
\noindent\textbf{Deel I --- De \omterm{def:g12:selection-drift-speciation:population}{populatie} nu.}
\begin{enumerate}
  \item Bereken de \omterm{def:g12:selection-drift-speciation:population}{frequenties} $p$ van $D$ en $q$ van $d$ door de kopieën
        te tellen.
  \item Bereken de aantallen \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypen} die in de referentietoestand
        worden verwacht en vergelijk die met de telling. Verkeert de
        \omterm{def:g12:selection-drift-speciation:population}{populatie} in de referentietoestand?
  \item Welk deel van de donkere muizen is drager van $d$?
  \item Leg met het vorige antwoord uit waarom het één generatie lang
        weghalen van alle bleke muizen van het eiland $q$ maar weinig zou
        verlagen. Bereken de nieuwe $q$ na zo'n verwijdering, als de
        overgebleven muizen willekeurig paren.
  \item Welke van de vier voorwaarden van de referentietoestand is op een
        klein eiland het waarschijnlijkst niet vervuld, en met welk
        gevolg?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Er komt een roofdier.}
Uilen koloniseren het eiland. Op het bleke gesteente worden donkere muizen
gezien en twee keer zo vaak gegeten: elke generatie sterft de helft van de
donkere muizen vóór de voortplanting, terwijl alle bleke overleven.
\begin{enumerate}[resume]
  \item Bereken, uitgaande van de getelde aantallen, hoeveel muizen van
        elk \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} de eerste generatie overleven om zich voort te
        planten.
  \item Bereken $p$ en $q$ onder de overlevenden.
  \item Bereken, als de overlevenden willekeurig paren, de verhoudingen
        van de \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypen} van de volgende generatie, laat daarna de uilen
        opnieuw hun werk doen en bereken $q$ onder de overlevenden ervan.
  \item Herhaal dat nog één keer (twee significante cijfers). Beschrijf
        het verloop van $q$ over de drie generaties.
  \item Leg uit waarom $q$ gestaag stijgt, en waarom $D$, anders dan een
        \omterm{def:g11:genes-and-disease:genetic}{recessief} \omterm{def:g10:universal-dna:gene}{allel} onder dezelfde selectie, volledig kan worden
        weggewerkt.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Een kolonie van zes.}
Een storm voert zes muizen naar een naburig eilandje: 2 $DD$, 3 $Dd$, 1 $dd$.
\begin{enumerate}[resume]
  \item Bereken $q$ in de kolonie en vergelijk die met het eiland.
  \item In het eerste nest planten zich, bij toeval, alleen de twee $DD$
        en één $Dd$ voort. Welke waarden kan $q$ in de volgende generatie
        aannemen? Wat is er met de variatie van de \omterm{def:g12:selection-drift-speciation:population}{populatie} gebeurd?
  \item Leg uit waarom op het eilandje een \omterm{def:g10:universal-dna:gene}{allel} binnen enkele generaties
        kan verdwijnen zonder uilen en zonder enig voordeel.
  \item Op het eilandje zijn geen uilen. Voorspel het lot van $d$ daar, en
        zeg waarom twee naburige eilanden zonder enige reden van
        aanpassing met verschillende vachtkleuren kunnen eindigen.
  \item Geef de naam van het proces, en de eigenschap van de kolonie die
        het daar overheersend maakt.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Tienduizend jaar later.}
De muizen van het eilandje zijn, afgezonderd, uiteengegroeid: kleiner,
bleker, en zij planten zich in een ander seizoen voort. Met eilandmuizen
in het laboratorium samengebracht paren zij zelden, en de weinige
kruisingen zijn onvruchtbaar.
\begin{enumerate}[resume]
  \item Zijn de muizen van het eilandje een nieuwe \omterm{def:g10:biodiversity-scales:species}{soort}? Verantwoord dat
        met de definitie.
  \item Noem de barrière die het proces in gang zette en de twee
        mechanismen die het uiteengroeien aandreven.
  \item Het voortplantingsseizoen verschilt tussen de twee. Leg uit hoe
        zo'n verschil, eenmaal aanwezig, de afzondering versterkt.
  \item Zou het eilandje zich morgen weer bij het eiland voegen, zouden
        de twee dan weer tot één \omterm{def:g12:selection-drift-speciation:population}{populatie} samensmelten? Onderscheid het
        geval van vraag 17 van een eerder stadium, zeg na 100 jaar.
  \item Geef het resultaat: de frequentie van $d$ op het eiland vóór en
        na drie generaties uilen, het lot van $d$ op het eilandje en het
        proces dat daarvoor verantwoordelijk is, en het aantal \omterm{def:g10:biodiversity-scales:species}{soorten} aan
        het eind.
\end{enumerate}
\end{problem}
