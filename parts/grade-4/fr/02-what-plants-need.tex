\chapter{Ce dont les plantes ont besoin}\label{ch:g4:what-plants-need}

Deux pots de cresson sont posés sur le même rebord de fenêtre, semés
le même jour. L'un est arrosé, l'autre non --- et en une semaine la
différence s'écrit en vert et en jaune. Avec des pots, des \omterm{prop:g3:flowers-fruits-seeds:transform}{graines}
et de la patience, tu peux faire répondre les \omterm{def:g1:plants-around-us:plant}{plantes} elles-mêmes à
la question de ce chapitre~: de quoi une \omterm{def:g1:plants-around-us:plant}{plante} a-t-elle exactement
besoin pour vivre et grandir~?

\section{Poser des questions aux plantes~: l'expérience honnête}

\begin{method}[L'expérience honnête]\label{met:g4:what-plants-need:fairtest}
Pour savoir si une \omterm{def:g1:plants-around-us:plant}{plante} a besoin de quelque chose --- eau,
lumière, sol, chaleur~:
\begin{enumerate}
  \item prends \emph{deux} pots des mêmes \omterm{def:g1:plants-around-us:plant}{plantes}, semées de la
        même façon~;
  \item traite-les exactement pareil en tout, \emph{sauf pour la
        seule chose testée} --- l'un l'a, l'autre non~;
  \item attends des jours ou des semaines, et compare~;
  \item toute différence vient forcément de la seule chose que tu
        as changée~: l'expérience était honnête.
\end{enumerate}
Le pot auquel on ne touche pas s'appelle le
\emph{témoin}\index{témoin (expérience)}~: il montre ce qui se passe
quand rien ne manque.
\end{method}

\begin{example}[Pourquoi deux pots et non un seul]\label{ex:g4:what-plants-need:control}
Supposons qu'un pot non arrosé se fane. Était-ce le manque d'eau ---
ou un coup de froid, ou de mauvaises \omterm{prop:g3:flowers-fruits-seeds:transform}{graines}~? Avec un jumeau arrosé
qui reste vert à côté, poussé du même sachet dans la même pièce, le
doute disparaît~: les jumeaux ne différaient que par l'eau. Les
expériences à un seul pot ne convainquent personne, et surtout pas
un biologiste.
\end{example}

\section{Ce que révèlent les expériences}

\begin{proposition}[Les besoins d'une plante]\label{prop:g4:what-plants-need:needs}
Des expériences honnêtes, menées pour chaque besoin tour à tour,
montrent qu'une \omterm{def:g1:plants-around-us:plant}{plante} a besoin~:
\begin{itemize}
  \item d'\emph{eau} --- le pot non arrosé se fane et meurt~;
  \item de \emph{lumière} --- le pot du placard devient pâle, long
        et faible, puis renonce~: pas de lumière, pas de nourriture
        fabriquée~;
  \item d'\emph{air} --- coupée de l'air frais, la croissance
        s'arrête~;
  \item de \emph{chaleur} --- dans le froid, les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} attendent
        (\cref{prop:g2:from-seed-to-plant:needs}) et la croissance
        se fige~;
  \item de \emph{sels minéraux}\index{sels minéraux}~: de toutes
        petites quantités de substances nourrissantes que les
        \omterm{def:g1:plants-around-us:parts}{racines} boivent dissoutes dans l'eau du sol --- dans de
        l'eau pure seule, une \omterm{def:g1:plants-around-us:plant}{plante} manque vite.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Le prisonnier pâle]\label{ex:g4:what-plants-need:pale}
L'expérience du placard de \cref{exo:g1:plants-around-us:8}, menée
correctement avec un jumeau éclairé, donne un résultat frappant~: le
prisonnier n'est pas seulement triste mais \emph{jaune pâle} --- il
a cessé de fabriquer sa substance verte --- et étrangement
\emph{grand}, étiré comme s'il cherchait la lumière absente.
Grandir sans lumière, c'est dépenser sans gagner~; les réserves de
la \omterm{def:g2:from-seed-to-plant:seed}{graine} une fois brûlées, il s'arrête.
\end{example}

\begin{remark}[Pourquoi les jardiniers nourrissent le sol]\label{rem:g4:what-plants-need:soil}
Le compost et le fumier ne sont pas des «~repas~» pour les \omterm{def:g1:plants-around-us:plant}{plantes}
--- une \omterm{def:g1:plants-around-us:plant}{plante} ne mange personne. Ils réapprovisionnent le sol en
\omterm{prop:g4:what-plants-need:needs}{sels minéraux}, que les récoltes emportent année après année. Le
composteur de \cref{ex:g2:caring-for-nature:compost} referme la
boucle~: ce que le sol a donné aux légumes, les épluchures le
rendent au sol.
\end{remark}

\section{Ce que les plantes font de ce qu'elles prennent}

\begin{proposition}[Les plantes fabriquent leur propre matière]\label{prop:g4:what-plants-need:make}
Les \omterm{def:g1:animals-around-us:animal}{animaux} grandissent en mangeant d'autres \omterm{def:g1:living-or-not:living}{êtres vivants}. Une
\omterm{def:g1:plants-around-us:plant}{plante} ne mange personne~: avec l'eau et les \omterm{prop:g4:what-plants-need:needs}{sels minéraux} du sol,
l'air, et l'\omterm{prop:g3:balanced-diet:jobs}{énergie} de la \emph{lumière} captée par ses \omterm{def:g1:plants-around-us:parts}{feuilles}
vertes, elle \emph{fabrique sa propre matière} --- de nouvelles
\omterm{def:g1:plants-around-us:parts}{tiges}, \omterm{def:g1:plants-around-us:parts}{feuilles}, \omterm{def:g1:plants-around-us:parts}{racines} et \omterm{prop:g3:flowers-fruits-seeds:transform}{graines}. Voilà pourquoi la lumière est
pour une \omterm{def:g1:plants-around-us:plant}{plante} une affaire de boire et de manger, et pourquoi la
\omterm{def:g1:plants-around-us:parts}{feuille} verte se tourne vers le ciel.
\end{proposition}
\admitted

\begin{example}[Un arbre à partir de presque rien]\label{ex:g4:what-plants-need:tree}
\omterm{def:g1:plants-around-us:plant}{Plante} un pépin dans un grand pot de terre pesée, attends des
années, et un petit arbre se dresse là --- et pourtant la terre n'a
presque rien perdu de son poids. Le bois de l'arbre n'a pas été
puisé dans le pot~: il a été \emph{fabriqué}, à partir d'eau, d'air
et de lumière, dans les ateliers verts des \omterm{def:g1:plants-around-us:parts}{feuilles}. D'où vient la
matière des \omterm{def:g1:living-or-not:living}{êtres vivants} nous occupera encore dans les années à
venir.
\end{example}

\begin{remark}[Le vert fait vivre tout le monde]\label{rem:g4:what-plants-need:everyone}
Rappelle-toi \cref{prop:g3:food-chains:start}~: toute \omterm{def:g3:food-chains:chain}{chaîne
alimentaire} commence par une \omterm{def:g1:plants-around-us:plant}{plante}. Nous voyons maintenant pourquoi
il en est forcément ainsi --- les \omterm{def:g1:plants-around-us:plant}{plantes} sont le seul maillon qui
fabrique de la matière neuve à partir d'ingrédients non vivants.
Chaque bouchée que quiconque avale, où que ce soit, a d'abord été
fabriquée dans une \omterm{def:g1:plants-around-us:parts}{feuille}.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:1}
Énumère les cinq besoins d'une \omterm{def:g1:plants-around-us:plant}{plante}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:2}
Dans une expérience honnête, combien de choses peuvent différer
entre les deux pots~? Comment appelle-t-on le pot auquel on ne
touche pas~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:3}
Décris l'aspect d'une \omterm{def:g1:plants-around-us:plant}{plante} poussée quinze jours dans le noir. Quel
besoin manquait, et que ne pouvait-elle pas fabriquer~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:4}
Que sont les \omterm{prop:g4:what-plants-need:needs}{sels minéraux}, et comment la \omterm{def:g1:plants-around-us:plant}{plante} les prend-elle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:5}
Pourquoi les jardiniers épandent-ils du compost, si les \omterm{def:g1:plants-around-us:plant}{plantes} ne
mangent pas~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:6}
En quoi la façon de grandir d'une \omterm{def:g1:plants-around-us:plant}{plante} diffère-t-elle de celle
d'un \omterm{def:g1:animals-around-us:animal}{animal}, d'après \cref{prop:g4:what-plants-need:make}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:7}
Dans \cref{ex:g4:what-plants-need:tree}, la terre du pot n'a presque
pas perdu de poids pendant que l'arbre poussait. D'où venait la
matière de l'arbre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:8}
À toi d'essayer~: conçois et réalise l'expérience honnête sur l'eau
de la figure avec du cresson --- deux pots, une seule différence.
Note ce que tu vois au bout d'une semaine.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:what-plants-need:9}
Lucas «~prouve~» que les \omterm{def:g1:plants-around-us:plant}{plantes} ont besoin de musique~: il a chanté
à un pot pendant un mois, et il s'est bien porté. Que manque-t-il à
son expérience, d'après \cref{met:g4:what-plants-need:fairtest}~?
Conçois la version qui la testerait vraiment.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:what-plants-need:10}
Une \omterm{def:g1:plants-around-us:plant}{plante} aquatique pousse complètement immergée, sans \omterm{def:g1:plants-around-us:parts}{racines}, et
pourtant en pleine santé. Lequel des cinq besoins satisfait-elle
manifestement encore, et comment --- et quel besoin son eau
fournit-elle à la place du sol~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:what-plants-need:11}
Deux pots de menthe identiques~: l'un dans du sable arrosé d'eau de
pluie, l'autre dans de la terre de jardin, tous deux arrosés et
éclairés pareil. Au bout de deux mois, le pot de sable est petit et
jaunissant. Quel besoin cette expérience honnête isole-t-elle --- et
pourquoi la \omterm{def:g1:plants-around-us:plant}{plante} du sable a-t-elle quand même poussé au début~?
\end{exercise}
