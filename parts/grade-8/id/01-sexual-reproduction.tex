\chapter{Reproduksi Seksual}\label{ch:g8:sexual-reproduction}

Nyanyian katak pada bulan Mei, \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} ceri pada bulan April, lenguhan
musim gugur seekor rusa \omterm{def:g4:animal-reproduction:sexual}{jantan}, \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} matahari yang memalingkan
wajahnya yang padat kepada lebah: di seluruh dunia yang hidup, \omterm{prop:g3:balanced-diet:jobs}{energi}
yang amat besar dibelanjakan untuk satu proyek --- generasi
berikutnya. Tahun-tahun sebelumnya mengumpulkan kepingannya
(\cref{ch:g4:animal-reproduction},
\cref{ch:g5:flower-to-fruit}); tahun ini merangkainya menjadi satu
teori, lalu mengajukan pertanyaan yang lebih tajam: mengapa harus
\emph{dua} induk?

\section{Satu mekanisme di seluruh dunia yang hidup}

\begin{definition}[Gamet]\label{def:g8:sexual-reproduction:gametes}
\omterm{def:g5:first-look-at-cells:cell}{Sel} kelamin pada \cref{def:g4:animal-reproduction:sexual} menyandang
nama umum \emph{gamet}\index{gamet}: gamet \omterm{def:g4:animal-reproduction:sexual}{jantan} (\omterm{def:g4:animal-reproduction:sexual}{sel sperma} pada
\omterm{def:g1:animals-around-us:animal}{hewan}, \omterm{def:g5:first-look-at-cells:cell}{sel} butir \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} pada \omterm{def:g1:plants-around-us:plant}{tumbuhan} berbunga) dan gamet \omterm{def:g4:animal-reproduction:sexual}{betina}
(\omterm{def:g4:animal-reproduction:sexual}{sel telur}, di dalam ovarium atau \omterm{prop:g5:flower-to-fruit:roles}{bakal biji}). Gamet adalah \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal
(\cref{ch:g5:first-look-at-cells}), dan masing-masingnya separuh
permulaan: kalau sendirian, sebuah gamet mati; kalau bergabung, dua
gamet membuat satu kehidupan.
\end{definition}

\begin{proposition}[Intinya yang tak berubah]\label{prop:g8:sexual-reproduction:core}
Di mana pun \omterm{def:g4:animal-reproduction:sexual}{reproduksi seksual} berjalan --- \omterm{prop:g3:sorting-living-things:groups}{ikan} atau pohon cemara,
siput atau manusia --- intinya tidak berubah:
\begin{enumerate}
  \item dua induk menghasilkan \omterm{def:g8:sexual-reproduction:gametes}{gamet}, \omterm{def:g4:animal-reproduction:sexual}{jantan} dan \omterm{def:g4:animal-reproduction:sexual}{betina};
  \item \emph{\omterm{def:g5:flower-to-fruit:steps}{pembuahan}}: satu \omterm{def:g8:sexual-reproduction:gametes}{gamet} \omterm{def:g4:animal-reproduction:sexual}{jantan} bergabung dengan satu
        \omterm{def:g8:sexual-reproduction:gametes}{gamet} \omterm{def:g4:animal-reproduction:sexual}{betina}, membentuk satu \omterm{def:g5:first-look-at-cells:cell}{sel} baru yang tunggal --- yaitu
        \emph{\omterm{def:g4:animal-reproduction:sexual}{sel telur} yang terbuahi}, atau
        \emph{zigot}\index{zigot};
  \item \omterm{prop:g8:sexual-reproduction:core}{zigotnya} membelah lalu membelah lagi
        (\cref{prop:g5:first-look-at-cells:growth}), membangun sebuah
        individu baru yang membawa sesuatu dari \emph{kedua} induknya
        (\cref{rem:g4:animal-reproduction:mix}).
\end{enumerate}
Di sekeliling inti ini, segala hal lainnya adalah ragam: di mana
\omterm{def:g8:sexual-reproduction:gametes}{gametnya} bertemu, berapa banyak, siapa yang merawat anaknya.
\end{proposition}
\admitted

\begin{example}[Ragamnya, yang dikelompokkan]\label{ex:g8:sexual-reproduction:variations}
Kumpulan lama itu terarsipkan dengan rapi di bawah intinya. \omterm{def:g5:flower-to-fruit:steps}{Pembuahan}
luar melawan \omterm{def:g5:flower-to-fruit:steps}{pembuahan} dalam
(\cref{prop:g4:animal-reproduction:where}): tempat pertemuannya. \omterm{def:g4:animal-reproduction:ovivivi}{Ovipar}
melawan \omterm{def:g4:animal-reproduction:ovivivi}{vivipar}
(\cref{def:g4:animal-reproduction:ovivivi}): pemondokan \omterm{prop:g8:sexual-reproduction:core}{zigotnya}.
Jutaan \omterm{def:g2:animals-grow-up:born}{telur} \omterm{prop:g3:sorting-living-things:groups}{ikan} kod melawan anak tunggal gajah
(\cref{prop:g4:animal-reproduction:strategies}): pembukuan jumlahnya.
\omterm{def:g5:flower-to-fruit:steps}{Penyerbukan} oleh lebah atau angin
(\cref{def:g5:flower-to-fruit:steps}): layanan perjalanan \omterm{def:g8:sexual-reproduction:gametes}{gamet}
\omterm{def:g4:animal-reproduction:sexual}{jantannya}. Satu mekanisme, sebuah katalog setelan.
\end{example}

\begin{remark}[Tumbuhan menjalankan inti yang sama]\label{rem:g8:sexual-reproduction:plants}
Letakkan \cref{ch:g5:flower-to-fruit} di samping
\cref{ch:g4:animal-reproduction} dan kecocokannya persis: \omterm{def:g3:flowers-fruits-seeds:parts}{benang sari}
dan \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya} menghasilkan \omterm{def:g8:sexual-reproduction:gametes}{gametnya}, buluh \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sarinya} adalah lorong
\omterm{def:g5:flower-to-fruit:steps}{pembuahan} dalamnya, \omterm{def:g2:from-seed-to-plant:seed}{bijinya} memondokkan \omterm{def:g5:human-reproduction:pregnancy}{embrio} \omterm{def:g1:plants-around-us:plant}{tumbuhannya} berikut
bekalnya (\cref{def:g2:from-seed-to-plant:seed}). Hanya
\omterm{def:g4:plants-without-seeds:vegetative}{perkembangbiakan vegetatif}
(\cref{def:g4:plants-without-seeds:vegetative}) yang berdiri terpisah
--- satu induk, tanpa \omterm{def:g8:sexual-reproduction:gametes}{gamet}, salinan yang persis: pertentangan yang
membuat pertanyaan pada bagian berikutnya menjadi tajam.
\end{remark}

\section{Mengapa dua induk?}

\begin{proposition}[Reproduksi seksual mengocok]\label{prop:g8:sexual-reproduction:shuffle}
Salinan stroberi lewat \omterm{ex:g4:plants-without-seeds:runner}{geragihnya}
(\cref{rem:g4:plants-without-seeds:copy}) memperlihatkan apa yang
dihasilkan satu induk: keseragaman. Dua induk menghasilkan yang
sebaliknya. Karena setiap \omterm{prop:g8:sexual-reproduction:core}{zigot} memadukan satu \omterm{def:g8:sexual-reproduction:gametes}{gamet} dari tiap
induknya --- dan, seperti akan ditunjukkan sebuah bab berikutnya,
tidak ada dua \omterm{def:g8:sexual-reproduction:gametes}{gamet} bahkan dari satu induk yang membawa warisan yang
benar-benar sama --- \omterm{def:g4:animal-reproduction:sexual}{reproduksi seksual} membagikan kepada tiap anaknya
sebuah \emph{paduan yang baru}. Saudara kandung berbeda; tidak ada dua
\omterm{prop:g2:from-seed-to-plant:cycle}{kecambah} dari satu polong yang cocok; keragaman di dalam setiap \omterm{def:g5:biodiversity:species}{spesies}
(\cref{rem:g5:biodiversity:scales}) sebagian besar adalah hasil
kocokan ini.
\end{proposition}
\admitted

\begin{example}[Dua taruhan kebun buahnya]\label{ex:g8:sexual-reproduction:bets}
Kebun \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} pada \cref{rem:g4:plants-without-seeds:copy} memainkan
kedua permainannya dengan sadar. Sambungan dan \omterm{met:g4:plants-without-seeds:cutting}{stek}: satu induk,
salinan yang persis dari pohon yang istimewa --- dan satu kelemahan
yang dibagi bersama, sehingga sebuah penyakit hawar yang menumbangkan
satu dapat menumbangkan semuanya. \omterm{def:g2:from-seed-to-plant:seed}{Biji}: dua induk, tiap \omterm{prop:g2:from-seed-to-plant:cycle}{kecambahnya}
sebuah paduan yang baru --- kebanyakan biasa saja, sebagian lebih
buruk, satu yang langka lebih baik daripada kedua induknya, dan tidak
ada dua yang serupa untuk disapu sebuah penyakit hawar. Menyalin
menabung masa kini; mengocok menjelajahi masa depan.
\end{example}

\begin{remark}[Keragaman adalah cadangan spesiesnya]\label{rem:g8:sexual-reproduction:reserve}
Mengapa kocokannya berarti melampaui kebun \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya}:
\cref{prop:g5:biodiversity:web} memperlihatkan jaring yang menyerap
guncangan lewat keragaman spesiesnya; di dalam satu \omterm{def:g5:biodiversity:species}{spesies}, keragaman
individunya memainkan peran yang sama. Ketika \omterm{def:g6:exploring-our-environment:conditions}{kondisinya} berbalik ---
sebuah penyakit baru, sebuah musim dingin yang lebih keras --- sebuah
\omterm{def:g5:biodiversity:species}{spesies} yang beragam lebih mungkin memegang seseorang yang cocok
dengan syarat yang baru. Apa yang disiratkannya bagi sejarah kehidupan
akan dinyatakan dengan terang oleh
\cref{ch:g9:how-species-change}; di sini catatlah saja bahwa
\omterm{def:g4:animal-reproduction:sexual}{reproduksi seksual} adalah yang menjaga cadangannya tetap terisi.
\end{remark}

\section{Membaca reproduksi spesies apa pun}

\begin{method}[Berkas reproduksinya]\label{met:g8:sexual-reproduction:dossier}
Untuk \omterm{def:g5:biodiversity:species}{spesies} apa pun, susunlah:
\begin{enumerate}
  \item \omterm{def:g8:sexual-reproduction:gametes}{gametnya}: siapa menghasilkan yang mana, di mana;
  \item pertemuannya: \omterm{def:g5:flower-to-fruit:steps}{pembuahan} luar atau dalam, dan layanan
        perjalanannya (air, perkawinan, penyerbuk, angin);
  \item pemondokannya: \omterm{def:g2:animals-grow-up:born}{telur} dikeluarkan atau anak dikandung; dengan
        bekal apa;
  \item jumlah dan perawatannya: kedudukannya pada garis siasat
        (\cref{prop:g4:animal-reproduction:strategies});
  \item musimnya: kapan, dan diatur waktunya oleh apa --- pertanyaan
        bab berikutnya.
\end{enumerate}
\end{method}

\begin{example}[Dua berkas berdampingan]\label{ex:g8:sexual-reproduction:dossiers}
\omterm{prop:g3:sorting-living-things:groups}{Ikan} duri (\cref{exo:g4:animal-reproduction:10}): \omterm{def:g8:sexual-reproduction:gametes}{gamet} yang
dikeluarkan ke dalam air, \omterm{def:g5:flower-to-fruit:steps}{pembuahan} luar, \omterm{def:g2:animals-grow-up:born}{telur} di dalam sebuah sarang
yang dibangun, berlusin-lusin, dijaga oleh ayahnya, pada musim semi.
Pohon ceri: \omterm{def:g8:sexual-reproduction:gametes}{gamet} di dalam \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya}, \omterm{def:g5:flower-to-fruit:steps}{pembuahan} dalam menuruni buluh
\omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sarinya}, \omterm{def:g5:human-reproduction:pregnancy}{embrio} yang dipondokkan di dalam \omterm{def:g2:from-seed-to-plant:seed}{biji} di dalam \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya},
ribuan, yang dibekali lalu ditinggalkan kepada layanan pengantaran
\omterm{prop:g3:sorting-living-things:groups}{burungnya} (\cref{exo:g6:how-plants-spread:12}), yang diatur waktunya
pada bulan April. Bentuk yang sama, \omterm{def:g5:biodiversity:species}{spesies} apa pun: berkasnya adalah
teorinya yang sedang dipakai.
\end{example}

\section{Soal: Program Pemuliaan}

\begin{problem}[{Soal akhir pekan --- setahun perjalanan seorang
pemulia tanaman, yang dijalankan atas teorinya}]\label{pb:g8:sexual-reproduction:1}
Seorang pemulia menginginkan sebuah apel baru: renyah seperti jenis A,
tahan penyakit hawar seperti jenis B. Perkakasnya persis perkakas bab
ini.

\medskip\noindent\textbf{Bagian I --- Persilangannya.}
\begin{enumerate}
  \item Ia mengambil \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} dari \omterm{def:g3:flowers-fruits-seeds:parts}{benang sari} A dengan sebuah kuas
        lalu menyentuhkannya pada \omterm{def:g3:flowers-fruits-seeds:parts}{putik} B, kemudian membungkus \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya}
        agar terhindar dari lebah. Sebutkan peran tiap langkahnya di
        dalam intinya --- dan mengapa bungkusnya berarti
        (\cref{met:g5:flower-to-fruit:diagnose} memakai kuas yang
        sama).
  \item Apa yang tumbuh di dalam \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} B sesudah kunjungannya, dan dua
        warisan apa yang dipadukan tiap \omterm{def:g2:from-seed-to-plant:seed}{bijinya}?
  \item Ia menyemai tiga ratus \omterm{def:g2:from-seed-to-plant:seed}{biji}. Mengapa ia mengharapkan tiga
        ratus pohon yang \emph{berbeda}, menurut
        \cref{prop:g8:sexual-reproduction:shuffle}?
  \item Kebanyakan \omterm{prop:g2:from-seed-to-plant:cycle}{kecambahnya} akan biasa saja, beberapa lebih buruk
        daripada kedua induknya, barangkali satu tepat sasaran. Contoh
        yang mana pada bab ini yang persis meramalkan sebaran itu?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Memaku pemenangnya.}
\begin{enumerate}[resume]
  \item Tahun kedelapan: satu pohon berbuah apel yang renyah dan tahan
        penyakit hawar. Kalau ia menyemai \omterm{def:g2:from-seed-to-plant:seed}{bijinya}, apa yang terjadi
        pada paduan yang berharga itu --- dan mengapa?
  \item Sebagai gantinya ia melipatgandakan pemenangnya lewat
        penyambungan --- reproduksi yang mana itu, dan apa yang
        dijaminnya?
  \item Seluruh kebun \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya} di masa depan akan berupa salinan satu
        pohon itu. Sebutkan risiko yang sudah diterimanya, dan frasa
        bab ini untuknya.
  \item Ajukanlah asuransi yang baku: apa lagi yang harus dipertahankan
        kebun \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya}, dan mengapa siasatnya menggemakan
        \cref{rem:g8:sexual-reproduction:reserve}?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Teorinya, yang
diaudit.}
\begin{enumerate}[resume]
  \item Programnya memakai kedua taruhan kebun \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya} secara
        berurutan. Sebutkan tahap mana memakai yang mana, dan apa yang
        dibeli masing-masingnya baginya.
  \item Seorang kolega memuliakan \omterm{prop:g3:sorting-living-things:groups}{ikan} duri dengan cara yang sama ---
        memilih induknya, membesarkan anaknya yang terkocok,
        mempertahankan yang terbaik. Daftarkan catatan berkas
        (\cref{met:g8:sexual-reproduction:dossier}) tempat praktik
        \omterm{prop:g3:sorting-living-things:groups}{ikannya} berbeda dari praktik apelnya, dan langkah inti tempat
        keduanya tidak mungkin berbeda.
  \item Pemulianya berkata: ``seks mengusulkan, seleksi memutuskan ---
        aku hanya memilih di antara yang dibagikan kocokannya.''
        Hubungkanlah semboyannya dengan cadangan keragaman pada
        \cref{rem:g8:sexual-reproduction:reserve}.
  \item Satu kalimat untuk menutup: programnya membutuhkan \omterm{def:g4:animal-reproduction:sexual}{reproduksi
        seksual} \emph{untuk apa}, dan membutuhkan \omterm{def:g4:plants-without-seeds:vegetative}{perkembangbiakan
        vegetatif} untuk apa?
\end{enumerate}
\end{problem}
