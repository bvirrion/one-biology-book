\chapter{Kegiatan Jasmani dan Kesehatan}\label{ch:g10:sport-and-health}

Dua murid berumur enam belas tahun. Yang satu berjalan kaki ke sekolah,
bermain bola basket dua kali sepekan dan tidur delapan jam; yang lain
diantar berkendara, menghabiskan malamnya duduk di depan layar, dan
tidur enam jam. Pada umur enam belas keduanya tampak serupa. Pada umur
lima puluh, dokternya akan melihat dua orang yang berbeda: yang pertama
dengan jantung, tulang, otot dan gula darah seorang yang dua puluh
tahun lebih muda, yang kedua dengan sederet keadaan yang diawali kata
``menahun''. Ketiga bab sebelumnya telah menunjukkan apa yang dikerjakan
tubuh yang bekerja; bab ini membahas apa yang dilakukan kerja yang
teratur padanya --- dan apa yang dirusak penyalahgunaan, dalam bentuk
latihan berlebih atau obat.

\section{Apa yang diubah latihan}

\begin{definition}[Kegiatan jasmani dan latihan]\label{def:g10:sport-and-health:training}
Yang disebut \emph{kegiatan jasmani}\index{kegiatan jasmani} adalah
gerakan tubuh apa pun yang dihasilkan otot yang menaikkan \omterm{def:g10:exercise-and-energy:expenditure}{pengeluaran
energi} di atas keadaan istirahat: berjalan, menaiki tangga, pekerjaan
rumah, olahraga. \emph{Latihan} adalah kegiatan jasmani yang diulang
dengan tujuan meningkatkan sebuah kemampuan. Tubuh menanggapi tuntutan
yang berulang dengan \emph{menyesuaikan diri}: memperbesar, menguatkan
atau melengkapi ulang struktur yang dibebani tuntutan itu --- asalkan
tuntutannya disusul istirahat tempat penyesuaian itu dibangun.
\end{definition}

\begin{proposition}[Penyesuaian terhadap latihan ketahanan]\label{prop:g10:sport-and-health:endurance}
Olahraga berulang yang lama dan berkegiatan sedang, sepanjang beberapa
bulan, menghasilkan:
\begin{itemize}
  \item jantung yang lebih besar dan lebih kuat: volume sekuncupnya
        naik, denyut istirahatnya turun (dari 70 menjadi 50 atau di
        bawahnya), curah maksimal dan $\dot V\!\mathrm{O_2}$maks-nya
        naik 10\% sampai 30\%;
  \item lebih banyak kapiler dan lebih banyak \omterm{def:g10:cells-common-unit:organelle}{mitokondria} dalam \omterm{def:g10:muscles-and-joints:muscle}{serat
        ototnya}, yang membakar bagian lemak yang lebih besar dan
        menghasilkan asam laktat yang lebih sedikit pada daya tertentu;
  \item lebih banyak \omterm{def:g10:cells-common-unit:cell}{sel} darah merah, sehingga lebih banyak oksigen
        yang terbawa tiap liter;
  \item simpanan \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} yang lebih besar, dan tubuh yang lebih mudah
        mengatur gula darah serta tekanan darahnya.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Angka penyesuaiannya]\label{ex:g10:sport-and-health:numbers}
Seorang berumur 17 tahun yang kurang gerak dengan
$\dot V\!\mathrm{O_2}$maks \qty{40}{mL/min} tiap kilogram berlari tiga
kali sepekan selama enam bulan: \num{48}, dan denyut jantung
istirahatnya turun dari 72 menjadi 58. Biopsi ototnya memperlihatkan
\omterm{def:g10:cells-common-unit:organelle}{mitokondria} 40\% lebih banyak dan kapiler 30\% lebih banyak tiap
seratnya. Seratnya tetap sama jumlahnya dan sama jenisnya; seratnya
telah dilengkapi ulang.
\end{example}

\begin{proposition}[Penyesuaian terhadap latihan kekuatan]\label{prop:g10:sport-and-health:strength}
Kerutan berulang melawan beban yang berat menebalkan \omterm{def:g10:muscles-and-joints:muscle}{serat ototnya}
(lebih banyak miofibril dalam masing-masing) dan memperbaiki perintah
saraf yang mengerahkannya; kekuatannya dapat berlipat dua dalam setahun
sementara jumlah seratnya tidak berubah. \omterm{def:g10:muscles-and-joints:muscle}{Tendon} dan tulang yang
dibebani latihannya ikut menebal, dengan lebih lambat. Tulang,
khususnya, dibangun dan dipelihara sebagai tanggapan terhadap beban
yang dibawanya: masa remaja yang aktif menetapkan massa tulang yang
akan ditarik pada usia lanjut, dan anggota badan yang tak bergerak
dalam gips kehilangan tulang dalam hitungan pekan.
\end{proposition}
\admitted

\section{Kegiatan dan kesehatan}

\begin{proposition}[Pengaruh kegiatan teratur pada kesehatan]\label{prop:g10:sport-and-health:health}
\omterm{def:g10:sport-and-health:training}{Kegiatan jasmani} yang teratur --- setara dengan sejam sehari kegiatan
sedang bagi remaja, setengah jam bagi orang dewasa --- menurunkan
risiko penyakit menahun yang terlazim: penyakit jantung dan arteri,
diabetes tipe 2, beberapa kanker, osteoporosis, depresi. Ia berbuat
demikian lewat penyesuaian di atas: jantung dan pembuluh yang bekerja
pada tekanan yang lebih rendah, otot yang mudah menarik glukosa dari
darah, tulang yang menjaga massanya, massa tubuh yang terjaga dalam
rentang yang sehat. Pengaruhnya besar --- setara dengan pengaruh tidak
merokok --- dan itu tidak menuntut olahraga, hanya gerakan.
\end{proposition}
\begin{proof}[Bukti percobaan]
Penelitian yang mengikuti puluhan ribu orang selama puluhan tahun
menemukan bahwa mereka yang berada dalam seperlima penduduk yang paling
aktif meninggal, pada umur tertentu, dengan laju kira-kira separuh laju
mereka dalam seperlima yang paling tidak aktif, dengan penyakit jantung
menjelaskan sebagian besar bedanya; hubungan itu tetap berlaku setelah
memperhitungkan pola makan, merokok dan keadaan sosialnya, dan ia
bertingkat --- tiap langkah naik dalam kegiatannya membawa satu langkah
turun dalam risikonya. Uji coba yang menugasi orang kurang gerak
mengikuti program jalan kaki memperlihatkan turunnya tekanan darah dan
gula darah dalam hitungan bulan.
\end{proof}

\begin{example}[Gula darah dan otot]\label{ex:g10:sport-and-health:sugar}
Otot yang bekerja mengambil glukosa dari darah tanpa memerlukan hormon
insulin, dan otot yang terlatih tetap lebih peka terhadap insulin di
antara sesinya. Jalan cepat setelah makan menurunkan puncak glukosa
darah yang menyusulnya; tubuh yang kurang gerak, yang diminta menyimpan
gula yang sama tanpa bergerak, memerlukan insulin yang lebih banyak
tiap tahun --- itulah jalan menuju diabetes tipe 2, yang mekanismenya
menjadi pokok tahun terakhir.
\end{example}

\section{Terlalu banyak, terlalu cepat, terlalu keliru}

\begin{proposition}[Latihan berlebih dan cedera]\label{prop:g10:sport-and-health:overtraining}
Penyesuaian dibangun selama istirahat. Tuntutan yang diulang tanpa
pemulihan yang cukup --- lebih banyak sesi, atau sesi yang lebih berat,
daripada yang dapat diperbaiki jaringannya di antaranya ---
menghasilkan kebalikan penyesuaian: kelelahan yang menetap, denyut
jantung istirahat yang naik, penampilan yang menurun, tidur yang
terganggu, infeksi yang sering, dan, pada jaringan yang paling lambat
memperbaiki diri (\cref{ch:g10:muscles-and-joints}), tendinitis serta
patah tulang karena tekanan. Penyebab cedera yang terlazim pada remaja
bukanlah olahraganya melainkan laju kenaikan bebannya.
\end{proposition}
\admitted

\begin{example}[Aturan sepuluh persen]\label{ex:g10:sport-and-health:tenpercent}
Pelari yang menaikkan jarak mingguannya tidak lebih dari 10\% tiap
pekan jarang mencederai dirinya; pelari yang melipatduakannya dalam dua
pekan sering mencederai dirinya. Tulang dan \omterm{def:g10:muscles-and-joints:muscle}{tendon} menyesuaikan diri
sepanjang beberapa bulan, otot dan jantung sepanjang beberapa pekan:
jaringan yang lambatlah yang menetapkan laju kenaikan bebannya.
\end{example}

\begin{proposition}[Doping]\label{prop:g10:sport-and-health:doping}
Yang disebut \emph{doping}\index{doping} adalah pemakaian zat atau cara
yang menaikkan penampilan dengan bekerja pada pengaturan tubuh, sebagai
ganti penyesuaian yang akan dibangun latihan. Masing-masing menirukan
sebuah penyesuaian dan membawa harga karena melewatinya:
\begin{itemize}
  \item hormon eritropoietin (EPO), atau pemindahan darah simpanan
        sendiri, menaikkan jumlah \omterm{def:g10:cells-common-unit:cell}{sel} darah merah --- dan mengentalkan
        darahnya sampai ia dapat membeku dalam pembuluh jantung atau
        otaknya;
  \item steroid anabolik, yang berasal dari testosteron, menebalkan
        \omterm{def:g10:muscles-and-joints:muscle}{serat ototnya} --- dan mengganggu hormon tubuhnya sendiri,
        hatinya, jantungnya dan suasana hatinya;
  \item perangsang menutupi kelelahan --- dan menyingkirkan peringatan
        yang melindungi jantung dari kepanasan dan kehabisan tenaga;
  \item hormon pertumbuhan, dan obat yang menyembunyikan yang lainnya,
        melengkapi daftarnya.
\end{itemize}
\omterm{prop:g10:sport-and-health:doping}{Doping} dilarang dalam pertandingan karena tidak adil; ia berbahaya
karena setiap zat itu mengalahkan sebuah pengaturan yang ada karena
sebuah alasan.
\end{proposition}
\admitted

\begin{example}[EPO dalam angka]\label{ex:g10:sport-and-health:epo}
\omterm{def:g10:cells-common-unit:cell}{Sel} darah merah menyusun sekitar 45\% volume darah yang normal.
Penyalahgunaan EPO telah mendorong atlet sampai 55\% dan lebih, yaitu
darah yang begitu kental sehingga jantungnya, yang melambat oleh tidur,
tidak dapat memindahkannya: beberapa pesepeda muda meninggal dalam
tidurnya pada tahun-tahun sebelum ujinya ada. Berlatih di tempat tinggi
menaikkan angka yang sama dua atau tiga poin, sepanjang beberapa pekan,
di bawah kendali tubuhnya sendiri.
\end{example}

\section{Aturan sehari-hari}

\begin{method}[Berlatih dengan aman]\label{met:g10:sport-and-health:rules}
\begin{enumerate}
  \item \emph{Majulah perlahan}: naikkan bebannya sekitar sepersepuluh
        tiap pekan, dan biarkan tulang serta \omterm{def:g10:muscles-and-joints:muscle}{tendon} menetapkan lajunya.
  \item \emph{Pemanasan dan pendinginan}: sepuluh menit masing-masing,
        karena alasan \cref{ch:g10:muscles-and-joints}.
  \item \emph{Minumlah}: satu jam olahraga berat kehilangan satu sampai
        dua liter keringat; gantilah, dengan sedikit garam, sebelum
        rasa haus mengatakannya --- kehilangan 2\% massa tubuh sudah
        menurunkan penampilan.
  \item \emph{Makanlah untuk usahanya}: \omterm{def:g10:chemistry-of-life:families}{karbohidrat} sebelum dan selama
        usaha yang panjang, \omterm{def:g10:chemistry-of-life:families}{protein} untuk memperbaiki diri sesudahnya
        (\cref{ch:g10:exercise-and-energy}); tidak ada suplemen yang
        menggantikan pola makan yang beragam.
  \item \emph{Beristirahatlah}: tidur adalah saat penyesuaian dibangun;
        denyut jantung istirahat yang naik adalah tanda paling awal
        bahwa istirahatnya terlalu sedikit.
  \item \emph{Dengarkan nyerinya}: nyeri tajam pada \omterm{def:g10:muscles-and-joints:joint}{sendi} atau \omterm{def:g10:muscles-and-joints:muscle}{tendon}
        adalah sebuah isyarat, bukan sebuah rintangan.
\end{enumerate}
\end{method}

\begin{example}[Pemenuhan cairan dalam angka]\label{ex:g10:sport-and-health:water}
Seorang pelari \qty{60}{kg} berkeringat \qty{1.2}{L} tiap jam pada
cuaca hangat. Dalam lari dua jam ia kehilangan \qty{2.4}{kg}, yaitu 4\%
massanya: volume darahnya turun, denyut jantungnya naik sepuluh denyut
untuk menjaga curahnya, suhu tubuhnya memanjat, dan kecepatannya turun
sepersepuluh. Minum setengah liter tiap jam akan menyusutkan
kehilangannya menjadi separuh.
\end{example}

\begin{remark}[Tubuh menyimpan catatannya]\label{rem:g10:sport-and-health:account}
Tiap penyesuaian dalam bab ini adalah struktur yang dibangun tubuh
sebagai tanggapan atas pemakaiannya, dan setiap satunya dibongkar lagi
ketika pemakaiannya berhenti: sebulan di tempat tidur kehilangan
seperlima ototnya, setahun tanpa gerak kehilangan sebagian besar
perolehan ketahanannya. Catatannya disimpan terus-menerus, dan
neracanya pada umur lima puluh adalah jumlah simpanan yang disetorkan
pada umur enam belas, dua puluh enam dan tiga puluh enam.
\end{remark}

\section{Soal: Dua Murid, Tiga Puluh Tahun}

\begin{problem}[{Soal akhir pekan --- seorang remaja enam belas tahun
yang aktif dan seorang yang kurang gerak, yang diikuti sepanjang
latihan, satu musim latihan berlebih, dan catatan yang disimpan
tubuhnya sampai usia pertengahan}]
\label{pb:g10:sport-and-health:1}
Alex dan Sam berumur enam belas tahun, keduanya \qty{62}{kg}, keduanya
berdenyut jantung istirahat 72 dan ber-$\dot V\!\mathrm{O_2}$maks
\qty{42}{mL/min} tiap kilogram. Alex mulai berlari tiga kali sepekan;
Sam tidak.

\medskip
\noindent\textbf{Bagian I --- Alex berlatih.}
Setelah enam bulan denyut jantung istirahat Alex 56 dan
$\dot V\!\mathrm{O_2}$maks-nya \qty{50}{mL/min} tiap kilogram; denyut
jantung maksimalnya (200) tidak berubah.
\begin{enumerate}
  \item \omterm{def:g10:heart-lungs-effort:output}{Curah jantung} istirahatnya masih \qty{4.8}{L/min}. Hitung
        volume sekuncupnya sebelum dan sesudahnya.
  \item Hitung $\dot V\!\mathrm{O_2}$maks-nya dalam liter tiap menit
        sebelum dan sesudahnya, serta perolehannya dalam persen.
  \item Dengan menganggap pengambilan oksigen maksimalnya
        (\qty{150}{mL} tiap liter) tidak berubah, hitung \omterm{def:g10:heart-lungs-effort:output}{curah jantung}
        maksimalnya sebelum dan sesudahnya, serta volume sekuncup
        maksimalnya sebelum dan sesudahnya.
  \item Penyesuaian mana pada
        \cref{prop:g10:sport-and-health:endurance} yang menjelaskan
        perolehan volume sekuncupnya, dan mana yang akan menjelaskan
        perolehan pengambilannya?
  \item Jarak mingguan Alex berjalan dari \qty{6}{km} menjadi
        \qty{24}{km} dalam enam bulan. Apakah aturan sepuluh persennya
        dihormati? Perlihatkan hitungannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Alex berlatih berlebih.}
Pada musim semi Alex melipatduakan jaraknya dalam dua pekan untuk
menyiapkan sebuah lomba. Tiga pekan kemudian: denyut jantung istirahat
64, tidur yang buruk, pilek, nyeri pada tulang keringnya, waktu
lombanya lebih lambat.
\begin{enumerate}[resume]
  \item Sebutkan nama keadaannya dan daftarkan tanda dari bab ini yang
        cocok dengannya.
  \item Apa sebab yang paling mungkin bagi nyeri tulang keringnya,
        dalam istilah \cref{ch:g10:muscles-and-joints}, dan mengapa
        jaringan itu?
  \item Mengapa denyut jantung istirahatnya naik, padahal latihan telah
        menurunkannya?
  \item Usulkan sebuah rencana tiga pekan dan berikan alasan tiap
        unsurnya dengan \cref{met:g10:sport-and-health:rules}.
  \item Seorang rekan setimnya menawarkan sebuah ``suplemen'' yang
        menyingkirkan rasa lelah. Golongan mana pada
        \cref{prop:g10:sport-and-health:doping} yang memuatnya, dan
        pengaturan apa yang dikalahkannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Sam duduk saja.}
Pada umur tiga puluh Sam bermassa \qty{80}{kg}, dengan denyut jantung
istirahat 78 dan $\dot V\!\mathrm{O_2}$maks \qty{32}{mL/min} tiap
kilogram; Alex, yang masih aktif, \qty{64}{kg}, 54 dan \num{48}.
\begin{enumerate}[resume]
  \item Hitung $\dot V\!\mathrm{O_2}$maks kedua lelaki itu dalam liter
        tiap menit. Nilai mutlak mana yang lebih besar, dan mengapa
        angka tiap kilogramlah yang menentukan bagi kehidupan
        sehari-hari?
  \item Menaiki empat lantai tangga menuntut masing-masing sekitar
        \qty{12}{mL} oksigen tiap kilogram tiap menit di atas keadaan
        istirahat. Berapa bagian nilai maksimalnya yang dipakai
        masing-masing? Siapa yang tiba dengan terengah?
  \item Dokter Sam menemukan gula darah yang sedikit di atas normal.
        Jelaskan, dengan \cref{ex:g10:sport-and-health:sugar},
        kaitannya dengan kekurangan geraknya.
  \item Dokternya meresepkan setengah jam jalan cepat sehari. Dengan
        gambar risiko nisbinya, perkirakan penurunan risiko jangka
        panjangnya jika Sam berpindah dari seperlima yang paling tidak
        aktif ke seperlima yang sedang.
  \item Sam membantah bahwa ia tidak punya waktu untuk olahraga.
        Jawablah dia dengan \cref{def:g10:sport-and-health:training}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Catatannya pada umur lima puluh.}
\begin{enumerate}[resume]
  \item Massa tulang memuncak sekitar umur dua puluh lima lalu menurun
        sekitar 1\% tiap tahun. Jika puncak Alex 10\% lebih tinggi
        daripada puncak Sam karena masa remajanya yang aktif, berapa
        tahun penurunan yang diwakili selisih itu?
  \item Setelah terjatuh pada umur lima puluh, pergelangan tangan Sam
        patah dan pergelangan Alex tidak. Hubungkan hal ini dengan
        pertanyaan 16 dan dengan
        \cref{prop:g10:sport-and-health:strength}.
  \item Alex memutus latihannya selama setahun pada umur empat puluh
        setelah cedera dan kehilangan sebagian besar perolehan
        ketahanannya, lalu membangunnya kembali. Apa yang ditunjukkannya
        tentang sifat sebuah penyesuaian?
  \item Sebutkan, bagi Sam pada umur lima puluh, tiga keadaan pada
        \cref{prop:g10:sport-and-health:health} yang menjadi lebih
        mungkin oleh kekurangan geraknya, dan mekanisme salah satunya.
  \item Nyatakan hasilnya: dua bilangan (denyut jantung istirahat,
        $\dot V\!\mathrm{O_2}$maks tiap kilogram) yang memisahkan kedua
        lelaki itu pada umur tiga puluh, dan satu kebiasaan, yang
        dinyatakan dalam jam tiap pekan, yang menghasilkan bedanya.
\end{enumerate}
\end{problem}
