\section{Les dents~: une boîte à outils dans la bouche}

\begin{definition}[Les trois sortes de dents]\label{def:g2:teeth-and-eating-well:kinds}
Les dents humaines sont de trois sortes, chacune avec son travail~:
\begin{enumerate}
  \item les \emph{incisives}\index{incisive}, plates et
        tranchantes à l'avant, \emph{coupent} --- elles tranchent
        dans la pomme~;
  \item les \emph{canines}\index{canine}, pointues, aux coins,
        \emph{déchirent} --- elles saisissent et tirent les
        aliments plus coriaces~;
  \item les \emph{molaires}\index{molaire}, larges et bosselées au
        fond, \emph{broient} --- elles écrasent les aliments en une
        purée prête à être avalée.
\end{enumerate}
\end{definition}

\begin{example}[Une bouchée de pomme, trois outils]\label{ex:g2:teeth-and-eating-well:apple}
Manger un morceau de pomme emploie toute la boîte à outils dans
l'ordre~: les \omterm{def:g2:teeth-and-eating-well:kinds}{incisives} tranchent la bouchée, les \omterm{def:g2:teeth-and-eating-well:kinds}{canines} aident à
la détacher, et la langue la fait passer vers le fond, aux
\omterm{def:g2:teeth-and-eating-well:kinds}{molaires}, qui la broient. C'est seulement alors que tu avales.
\end{example}

\begin{remark}[Souviens-toi des animaux]\label{rem:g2:teeth-and-eating-well:animals}
Ta boîte à outils mélangée raconte une histoire~: des coupoirs
plats et des broyeuses comme un \omterm{def:g2:what-animals-eat:kinds}{herbivore}, des \omterm{def:g2:teeth-and-eating-well:kinds}{canines} pointues
comme un \omterm{def:g2:what-animals-eat:kinds}{carnivore}. C'est une bouche d'\omterm{def:g2:what-animals-eat:kinds}{omnivore} --- les humains sont
équipés pour les \omterm{def:g1:plants-around-us:plant}{plantes} \emph{et} pour la viande, exactement comme
\cref{ch:g2:what-animals-eat} nous a rangés.
\end{remark}

\section{Dents de lait, dents d'adulte}

\begin{proposition}[Deux jeux dans une vie]\label{prop:g2:teeth-and-eating-well:sets}
Les humains font pousser deux jeux de dents. Les $20$ \emph{dents de
lait}\index{dents de lait} arrivent dans les premières années de la
vie. Vers six ans, elles se déchaussent et tombent une à une,
poussées par-dessous par les \emph{dents d'adulte}\index{dents
d'adulte} --- le jeu définitif, jusqu'à $32$ dents, qui doit durer
tout le reste de ta vie.
\end{proposition}
\admitted

\begin{example}[L'incisive qui bouge]\label{ex:g2:teeth-and-eating-well:wobbly}
Une dent de devant se met à bouger juste à l'âge où les enfants
entrent à la grande école. Dessous, une \omterm{def:g2:teeth-and-eating-well:kinds}{incisive} d'adulte a poussé
en silence et dissous la \omterm{def:g1:plants-around-us:parts}{racine} de la dent de \omterm{def:g2:animals-grow-up:born}{lait}, jusqu'à ce que
la petite dent lâche prise. Le trou n'est pas un accident --- c'est
une relève.
\end{example}

\begin{remark}[Pas de troisième jeu]\label{rem:g2:teeth-and-eating-well:nothird}
Après le jeu d'adulte, aucune dent nouvelle ne viendra plus. Une
dent de \omterm{def:g2:animals-grow-up:born}{lait} perdue est remplacée~; une dent d'adulte perdue est
perdue. C'est la vraie raison pour laquelle les grands insistent sur
le brossage~: la seconde boîte à outils doit durer soixante-dix ans
ou plus.
\end{remark}

\section{Bien manger}

\begin{proposition}[Une assiette équilibrée]\label{prop:g2:teeth-and-eating-well:balanced}
Pour grandir et rester en bonne santé, le corps a besoin, chaque
jour, d'aliments de chaque famille~:
\begin{itemize}
  \item fruits et légumes --- pour les vitamines, à chaque repas~;
  \item pain, céréales, pommes de terre --- pour l'énergie qui
        dure~;
  \item \omterm{def:g2:animals-grow-up:born}{lait}, yaourt, fromage --- pour construire les os et les
        dents~;
  \item viande, poisson ou \omterm{def:g2:animals-grow-up:born}{œufs} --- pour construire le corps, une
        ou deux fois par jour suffit~;
  \item eau --- la seule boisson dont le corps ait vraiment
        besoin~;
  \item douceurs sucrées et grasses --- pour le plaisir, en petites
        quantités, pas à chaque repas.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Deux goûters]\label{ex:g2:teeth-and-eating-well:snack}
Goûter A~: une pomme, un morceau de pain, de l'eau. Goûter B~: des
bonbons, un soda. Le goûter B a le goût du sucre et de pas grand
chose d'autre --- et le sucre est exactement ce dont les \omterm{def:g1:healthy-habits:germs}{microbes} de
\cref{ch:g1:healthy-habits} font leur festin. Le goûter A nourrit le
corps et donne aux dents un travail honnête. Devine lequel la
dentiste prépare pour ses propres enfants.
\end{example}

\begin{method}[Vérifier le menu d'une journée]\label{met:g2:teeth-and-eating-well:check}
Prends les repas d'une journée et vérifie~:
\begin{enumerate}
  \item des fruits ou des légumes à chaque repas~?
  \item quelque chose des familles qui construisent --- produits
        laitiers, et viande, poisson ou \omterm{def:g2:animals-grow-up:born}{œufs}~?
  \item de l'eau comme boisson de tous les jours~?
  \item des douceurs gardées petites et occasionnelles~?
\end{enumerate}
S'il manque une famille, le prochain repas est l'occasion de la
réinviter.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:1}
Nomme les trois sortes de dents et le travail de chacune.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:2}
Quelles dents sont à l'avant de la bouche, et lesquelles au fond~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:3}
Suis une bouchée de carotte à travers la bouche~: quelle dent fait
quoi, et dans quel ordre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:4}
Combien y a-t-il de \omterm{prop:g2:teeth-and-eating-well:sets}{dents de lait}~? Que leur arrive-t-il à partir de
six ans environ~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:5}
Pourquoi faut-il prendre encore plus soin des \omterm{prop:g2:teeth-and-eating-well:sets}{dents d'adulte} que des
\omterm{prop:g2:teeth-and-eating-well:sets}{dents de lait}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:6}
Nomme les familles d'aliments qu'une journée saine devrait
comporter. De quelle boisson le corps a-t-il vraiment besoin~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:7}
Compare les deux goûters de
\cref{ex:g2:teeth-and-eating-well:snack}~: que donne le goûter A au
corps que le goûter B ne donne pas~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:8}
À toi d'essayer~: regarde tes propres dents dans un miroir. Trouve
une \omterm{def:g2:teeth-and-eating-well:kinds}{incisive}, une \omterm{def:g2:teeth-and-eating-well:kinds}{canine} et une \omterm{def:g2:teeth-and-eating-well:kinds}{molaire}, et compare leurs formes à
celles de la figure.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:teeth-and-eating-well:9}
À l'aide de \cref{rem:g2:teeth-and-eating-well:animals}, explique ce
que la forme des dents humaines dit du menu humain --- et à quels
\omterm{def:g1:animals-around-us:animal}{animaux} de \cref{ch:g2:what-animals-eat} nous ressemblons.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:teeth-and-eating-well:10}
La journée de Lila~: petit déjeuner --- céréales et \omterm{def:g2:animals-grow-up:born}{lait}~; déjeuner
--- pâtes et bonbons~; goûter --- soda et biscuits~; dîner ---
pizza. Applique-lui
\cref{met:g2:teeth-and-eating-well:check}~: qu'est-ce qui manque, et
qu'est-ce qui revient trop souvent~?
\end{exercise}
