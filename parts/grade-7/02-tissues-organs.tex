\chapter{Tissues and Organs}\label{ch:g7:tissues}

A single \omterm{def:g7:microscope:cell}{cell} can be a whole \omterm{def:g4:ecosystems:levels}{organism} (for example a bacterium), but your
\omterm{def:g1:my-body:body}{body} has trillions of \omterm{def:g7:microscope:cell}{cells}. They are not a random pile: \omterm{def:g7:microscope:cell}{cells} form
\omterm{def:g7:tissues:tissue}{tissues}, \omterm{def:g7:tissues:tissue}{tissues} form \omterm{def:g7:tissues:organ}{organs}, and \omterm{def:g7:tissues:organ}{organs} work in systems. This chapter
maps those levels of organisation.

\section{Levels of organisation}

\begin{definition}[Tissue]\label{def:g7:tissues:tissue}
A \emph{tissue}\index{tissue} is a group of similar \omterm{def:g7:microscope:cell}{cells} that work
together to perform a shared function.
\end{definition}

\begin{definition}[Organ]\label{def:g7:tissues:organ}
An \emph{organ}\index{organ} is a structure made of several \omterm{def:g7:tissues:tissue}{tissue} types
working together (for example the \omterm{def:g5:circulation:heart}{heart}, \omterm{def:g7:tissues:plant-organs}{leaf} or \omterm{def:g3:teeth:path}{stomach}).
\end{definition}

\begin{definition}[Organ system]\label{def:g7:tissues:system}
An \emph{organ system}\index{organ system} is a group of \omterm{def:g7:tissues:organ}{organs} that
cooperate in a major \omterm{def:g1:my-body:body}{body} function (for example the \omterm{def:g4:body-systems:digestion}{digestive system} or
\omterm{def:g4:body-systems:circ}{circulatory system}).
\end{definition}

\begin{proposition}[Hierarchy]\label{prop:g7:tissues:hierarchy}
In \omterm{def:g6:cell-theory:uni-multi}{multicellular} \omterm{def:g1:plants-animals:animal}{animals} and \omterm{def:g1:plants-animals:plant}{plants}:
\[
\text{cell} \to \text{tissue} \to \text{organ} \to \text{organ system}
\to \text{organism}.
\]
Each level has properties that single \omterm{def:g7:microscope:cell}{cells} alone do not show.
\end{proposition}
\admitted

\section{Animal tissues (overview)}

\begin{definition}[Main animal tissue types]\label{def:g7:tissues:animal}
Four broad types appear again and again:
\begin{itemize}
  \item \emph{epithelial tissue}\index{epithelial tissue} --- covers
        surfaces and lines cavities (\omterm{def:g5:nerves:senses}{skin} surface, gut lining);
  \item \emph{connective tissue}\index{connective tissue} --- supports and
        binds (\omterm{def:g3:bones:bone}{bone}, \omterm{def:g5:circulation:blood}{blood}, cartilage, \omterm{def:g6:nutrition:groups}{fat});
  \item \emph{muscle tissue}\index{muscle tissue} --- contracts to produce
        movement;
  \item \emph{nervous tissue}\index{nervous tissue} --- carries electrical
        signals (\omterm{def:g5:nerves:neuron}{neurons} and support \omterm{def:g7:microscope:cell}{cells}).
\end{itemize}
\end{definition}

\begin{example}[Stomach as an organ]\label{ex:g7:tissues:stomach}
The \omterm{def:g3:teeth:path}{stomach} has a muscular wall (churning), an epithelial lining (mucus
and acid), \omterm{def:g7:tissues:animal}{connective tissue} layers, and \omterm{def:g5:nerves:system}{nerves} coordinating activity.
Several \omterm{def:g7:tissues:tissue}{tissues}, one \omterm{def:g7:tissues:organ}{organ}.
\end{example}

\section{Plant tissues and organs}

\begin{definition}[Plant organs]\label{def:g7:tissues:plant-organs}
Typical \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g7:tissues:organ}{organs} include \emph{roots}\index{root} (anchor, absorb
\omterm{def:g4:plants-need:needs}{water} and \omterm{def:g6:nutrition:groups}{minerals}), \emph{stems}\index{stem} (support, transport) and
\emph{leaves}\index{leaf} (\omterm{def:g5:photosynthesis:def}{photosynthesis}). \omterm{def:g2:plant-parts:main}{Flowers} and \omterm{def:g6:reproduction:plant}{fruits} are
reproductive \omterm{def:g7:tissues:organ}{organs} in flowering \omterm{def:g1:plants-animals:plant}{plants}.
\end{definition}

\begin{definition}[Plant tissue ideas]\label{def:g7:tissues:plant-tissue}
\omterm{def:g1:plants-animals:plant}{Plants} have dermal \omterm{def:g7:tissues:tissue}{tissue} (outer protection), vascular \omterm{def:g7:tissues:tissue}{tissue}
(\emph{xylem}\index{xylem} and \emph{phloem}\index{phloem} for transport)
and ground \omterm{def:g7:tissues:tissue}{tissue} (\omterm{def:g5:photosynthesis:def}{photosynthesis}, storage, support). Details return in
\cref{ch:g7:plant-structure}.
\end{definition}

\begin{example}[Leaf]\label{ex:g7:tissues:leaf}
A \omterm{def:g7:tissues:plant-organs}{leaf} is an \omterm{def:g7:tissues:organ}{organ}: upper and lower epidermis, photosynthetic mesophyll
\omterm{def:g7:tissues:tissue}{tissue}, and \omterm{def:g5:circulation:heart}{veins} with \omterm{def:g7:tissues:plant-tissue}{xylem} and \omterm{def:g7:tissues:plant-tissue}{phloem}.
\end{example}

\section{Specialisation}

\begin{proposition}[Division of labour]\label{prop:g7:tissues:labour}
In \omterm{def:g6:uni-multi:multi}{multicellular organisms}, \omterm{def:g7:microscope:cell}{cells} become specialised for particular jobs
(for example red \omterm{def:g5:circulation:blood}{blood} \omterm{def:g7:microscope:cell}{cells} for oxygen transport, \omterm{def:g7:tissues:plant-organs}{root} hair \omterm{def:g7:microscope:cell}{cells} for
absorption). Specialisation raises efficiency but means \omterm{def:g7:microscope:cell}{cells} depend on
one another --- hence \omterm{def:g7:tissues:tissue}{tissues}, \omterm{def:g7:tissues:organ}{organs} and systems.
\end{proposition}
\admitted
