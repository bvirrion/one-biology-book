\chapter{Chromosomen}\label{ch:g9:chromosomes}

\Cref{rem:g9:heredity-and-traits:question} dreef de bepalers
de \omterm{def:g6:cell-unit-of-life:parts}{celkern} in. Betrap een \omterm{def:g5:first-look-at-cells:cell}{cel} op het juiste moment en de kern
laat haar kaarten zien: haar inhoud windt zich op tot telbare,
fotografeerbare lichaampjes --- de chromosomen. Ze tellen, in
lichaamscellen en in \omterm{def:g8:sexual-reproduction:gametes}{geslachtscellen}, blijkt bijna alles te
verklaren wat het familiealbum eiste: de helften, de paren, het
schudden, en de worp die beslist of een \omterm{def:g3:stages-of-life:stages}{baby} een jongen of een
meisje wordt.

\section{De celkern, op heterdaad betrapt}

\begin{definition}[Chromosomen]\label{def:g9:chromosomes:chromosomes}
Een \emph{chromosoom}\index{chromosoom} is een draadvormig
lichaampje in de \omterm{def:g6:cell-unit-of-life:parts}{celkern}, zichtbaar (goed gekleurd, op het
moment van de deling) doordat de inhoud van de kern zich voor
het vervoer oprolt. Tussen twee delingen door lopen dezelfde
draden los en onzichtbaar --- werkstand; bij de deling
verdichten ze zich --- verplaatsstand. Gefotografeerd en
gesorteerd vormen de chromosomen van een \omterm{def:g5:first-look-at-cells:cell}{cel} haar
\emph{karyotype}\index{karyotype}: het hele spel, uitgelegd.
\end{definition}

\begin{proposition}[De menselijke aantallen]\label{prop:g9:chromosomes:counts}
De \omterm{def:g9:chromosomes:chromosomes}{karyotypes} leveren drie getallen die het hele hoofdstuk
dragen:
\begin{enumerate}
  \item elke menselijke lichaamscel bevat \emph{46}
        chromosomen --- \omterm{def:g1:the-five-senses:sense}{huid}, \omterm{prop:g7:digestion-and-nutrients:absorption}{lever} en \omterm{def:g8:nervous-communication:neuron}{zenuwcel} evengoed
        (de delingen van
        \cref{prop:g5:first-look-at-cells:growth} kopiëren het
        spel getrouw naar elke \omterm{def:g5:first-look-at-cells:cell}{cel});
  \item uitgelegd sorteren de 46 zich tot \emph{23 paren},
        waarbij de twee partners van een paar in grootte en
        bandering overeenkomen;
  \item de \emph{\omterm{def:g8:sexual-reproduction:gametes}{geslachtscellen}} doorbreken de regel: \omterm{def:g4:animal-reproduction:sexual}{eicel}
        en \omterm{def:g8:reproductive-systems:male}{zaadcel} dragen er \emph{23} --- één \omterm{def:g9:chromosomes:chromosomes}{chromosoom} uit
        elk paar, geen enkel paar compleet.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[De rekensom van de bevruchting]\label{ex:g9:chromosomes:arithmetic}
De aantallen klikken als een bewijs in elkaar. Twee
\omterm{def:g8:sexual-reproduction:gametes}{geslachtscellen} met elk 23; de \omterm{def:g5:flower-to-fruit:steps}{bevruchting}
(\cref{prop:g8:sexual-reproduction:core}) telt ze op: de \omterm{prop:g8:sexual-reproduction:core}{zygote}
bevat $23 + 23 = 46$ --- de paren hersteld, van elk paar één
partner van de moeder en één van de vader. Elke lichaamscel van
het kind draagt dan de bijdrage van beide ouders --- precies wat
\cref{rem:g4:animal-reproduction:mix} van buitenaf waarnam, met
het veulen en de bles, een hoofdstukgeneratie geleden.
\end{example}

\begin{remark}[Waarom geslachtscellen moeten halveren]\label{rem:g9:chromosomes:whyhalve}
Stel dat ze het niet deden: \omterm{def:g8:sexual-reproduction:gametes}{geslachtscellen} met 46 chromosomen
zouden kinderen met 92 maken, kleinkinderen met 184 --- het
spel zou elke generatie verdubbelen. De bijzondere halverende
deling van de \omterm{def:g8:sexual-reproduction:gametes}{geslachtscellen} --- elk krijgt \emph{één partner
uit elk paar} --- houdt het aantal van de \omterm{def:g5:biodiversity:species}{soort} generatie na
generatie op 46. En let op de bonus: \emph{welke} partner van
elk paar een \omterm{def:g8:sexual-reproduction:gametes}{geslachtscel} krijgt, wordt paar voor paar
onafhankelijk beslist --- $2^{23}$ mogelijke spellen van één
ouder, het schudden van
\cref{prop:g8:sexual-reproduction:shuffle} eindelijk geteld.
\end{remark}

\section{De chromosomen dragen de bepalers}

\begin{proposition}[Bewijs dat het spel ertoe doet]\label{prop:g9:chromosomes:evidence}
Dat de chromosomen de voertuigen van de bepalers \emph{zijn},
is geen aanname maar bewijs:
\begin{itemize}
  \item ze gedragen zich precies zoals
        \cref{prop:g9:heredity-and-traits:conclusions} het
        eist: in elke \omterm{def:g5:first-look-at-cells:cell}{cel} aanwezig, tot \omterm{def:g8:sexual-reproduction:gametes}{geslachtscellen}
        gehalveerd, bij de \omterm{def:g5:flower-to-fruit:steps}{bevruchting} hersteld, partner voor
        partner selecteerbaar;
  \item ongelukken in het \omterm{def:g9:chromosomes:chromosomes}{karyotype} bewijzen de lading: een
        spel met één \omterm{def:g9:chromosomes:chromosomes}{chromosoom} te veel of te weinig verandert
        de ontwikkeling van het hele lichaam --- een extra
        exemplaar van het kleine paar 21 bijvoorbeeld
        veroorzaakt de herkenbare reeks \omterm{def:g9:heredity-and-traits:traits}{kenmerken} die
        downsyndroom heet. Eén draad verkeerd geteld, veel
        \omterm{def:g9:heredity-and-traits:traits}{kenmerken} verschoven: de draden dragen instructies
        voor \omterm{def:g9:heredity-and-traits:traits}{kenmerken};
  \item en één paar kondigt zijn lading in elk \omterm{def:g9:chromosomes:chromosomes}{karyotype} aan:
        de \emph{geslachtschromosomen}.
\end{itemize}
\end{proposition}
\admitted

\begin{definition}[X, Y en de kruis-of-munt]\label{def:g9:chromosomes:xy}
Paar 23 komt in twee uitvoeringen: twee overeenkomende
\emph{X}-chromosomen --- het \omterm{def:g9:chromosomes:chromosomes}{karyotype} van een meisje; of één X
en een kleine \emph{Y} --- een jongen. De halvering voert
vervolgens de worp uit: elke \omterm{def:g4:animal-reproduction:sexual}{eicel} draagt een X; een \omterm{def:g8:reproductive-systems:male}{zaadcel}
draagt X of Y, half om half. \omterm{def:g5:flower-to-fruit:steps}{Bevruchting} door een \omterm{def:g8:reproductive-systems:male}{X-zaadcel}
maakt XX, door een \omterm{def:g8:reproductive-systems:male}{Y-zaadcel} XY --- één kans op twee (de taal
van kans uit \cref{ch:g9:heredity-and-traits}, eindelijk met
haar mechanisme), beslist door welke zwemmer aankomt, op het
ogenblik van de \omterm{def:g5:flower-to-fruit:steps}{bevruchting}.
\end{definition}

\begin{example}[Oude vragen, korte antwoorden]\label{ex:g9:chromosomes:answers}
Het spel beslecht historische twisten. Of een kind een jongen of
een meisje is, dankt niets aan de wensen, diëten of jaargetijden
van welke ouder ook --- het is de worp, uitgevoerd in het
mechanisme van \cref{def:g9:chromosomes:xy} (en, voor het
register van oud onrecht: het beslissende verschil rijdt mee in
de \emph{\omterm{def:g8:reproductive-systems:male}{zaadcel}}). Gelijke helften beantwoorden ook het
familiealbum: moeder en vader dragen elk één vol half spel bij
--- 23 en 23 --- en de rekenkunde van de \omterm{def:g9:heredity-and-traits:heredity}{erfelijkheid} eert beide
lijnen gelijk.
\end{example}

\begin{method}[Een karyotype lezen]\label{met:g9:chromosomes:read}
Met een foto van een \omterm{def:g9:chromosomes:chromosomes}{karyotype} in de hand:
\begin{enumerate}
  \item tel: 46 verwacht in een lichaamscel, 23 in een
        \omterm{def:g8:sexual-reproduction:gametes}{geslachtscel};
  \item paar: partners gekoppeld op grootte en bandering, 22
        gewone paren plus het 23e;
  \item lees het 23e: XX of XY;
  \item controleer de aantallen: elk paar met één of drie leden
        verklaart, via
        \cref{prop:g9:chromosomes:evidence}, een verschoven
        ontwikkeling.
\end{enumerate}
\end{method}

\section{Probleem: De cytogenetische werkbank}

\begin{problem}[{Weekendprobleem --- vier karyotypes op
één lichtbak}]\label{pb:g9:chromosomes:1}
De lesbak van een cytogenetisch ziekenhuislaboratorium:
\omterm{def:g9:chromosomes:chromosomes}{karyotype} A --- 46 chromosomen, 23e paar XY; B --- 46, 23e paar
XX; C --- 23 chromosomen, met daaronder één X; D --- 47, met
drie exemplaren bij paar 21, 23e paar XX.

\medskip\noindent\textbf{Deel I --- De aflezingen.}
\begin{enumerate}
  \item Lees A, B en C met
        \cref{met:g9:chromosomes:read}: van wie zouden het de
        cellen kunnen zijn?
  \item Het aantal van C is geen fout. Wat voor \omterm{def:g5:first-look-at-cells:cell}{cel} is het, en
        welke delingsgeschiedenis legt haar 23 vast?
  \item Lees D: aantal, paren, en de ontwikkeling die het
        aankondigt.
  \item De \omterm{def:g8:sexual-reproduction:gametes}{geslachtscel} van welke ouder besliste het 23e paar
        van A, en hoe stonden de kansen bij die \omterm{def:g5:flower-to-fruit:steps}{bevruchting}?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De rekenkundige
vragen.}
\begin{enumerate}[resume]
  \item Een stagiair vraagt waarom de lichaamscellen van één
        persoon allemaal met \omterm{def:g9:chromosomes:chromosomes}{karyotype} A overeenkomen.
        Antwoord met de kopiërende delingen van
        \cref{prop:g5:first-look-at-cells:growth}.
  \item Een ander vraagt hoe twee ouders met 46-cellen de 23
        van C maakten. Noem de bijzondere deling en haar regel.
  \item Combineer C met een \omterm{def:g4:animal-reproduction:sexual}{eicel} die een X draagt, en met de
        Y-dragende buurcel uit haar eigen laboratorium --- wacht:
        C \emph{is} Y-vrij. Zeg wat de X van C garandeert over
        elk kind dat ze mee helpt maken.
  \item Bereken de mogelijke spellen die één ouder kan
        uitdelen: geef de onafhankelijkheid paar voor paar en de
        conclusie $2^{23}$ --- voeg daar dan het koppelen van
        twee zulke uitdelingen door de \omterm{def:g5:flower-to-fruit:steps}{bevruchting} aan toe.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De
counselingkamer.}
\begin{enumerate}[resume]
  \item De ouders van een kind met \omterm{def:g9:chromosomes:chromosomes}{karyotype} D vragen: ``wat
        hebben wij verkeerd gedaan?'' Geef het eerlijke antwoord
        van het mechanisme over ongelukken bij het halveren ---
        en wat de 47 draden van het kind wel en niet veranderen
        aan de liefde in die spreekkamer. (Antwoord als de counselor,
        met de \omterm{def:g1:living-or-not:living}{biologie} recht voor zijn raap.)
  \item Een echtpaar met drie dochters vraagt naar de kans op
        een zoon. Antwoord met de worp, en corrigeer het gevoel
        dat het geluk aan de beurt is.
  \item Een bezoeker beweert dat een dieet het geslacht van een
        \omterm{def:g3:stages-of-life:stages}{baby} kan kiezen. Toets die bewering aan
        \cref{def:g9:chromosomes:xy}: waar wordt de beslissing
        genomen, en waar zou een dieet bij kunnen?
  \item Sluit de werkbank af met één zin: wat de vier foto's
        samen bewijzen over waar de \omterm{def:g9:heredity-and-traits:heredity}{erfelijkheid} meerijdt.
\end{enumerate}
\end{problem}
