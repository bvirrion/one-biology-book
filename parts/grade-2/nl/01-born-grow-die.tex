\chapter{Geboren worden, groeien, doodgaan}\label{ch:g2:born-grow-die}

In de lente staat de kersenboom bij het hek vol bloesem; eronder zoogt
een poes haar pasgeboren kittens; en opa laat een foto zien van
zichzelf toen hij zo oud was als jij --- klein, met gaten in zijn
gebit, en toen al dol op kersen. Elk \omterm{def:g1:living-or-not:living}{levend wezen} legt dezelfde weg af:
het begint, het groeit, en op een dag houdt het op. Die weg heet de
\omterm{def:g2:born-grow-die:cycle}{levenscyclus}.

\section{De stadia van een leven}

\begin{definition}[Levenscyclus]\label{def:g2:born-grow-die:cycle}
De \emph{levenscyclus}\index{levenscyclus} van een \omterm{def:g1:living-or-not:living}{levend wezen} is de
rij \emph{stadia}\index{stadium} die zijn leven doorloopt:
\begin{enumerate}
  \item het wordt \emph{geboren}\index{geboorte} --- het begin;
  \item het is \emph{jong}\index{jong} --- het groeit en verandert snel;
  \item het is \emph{volwassen}\index{volwassen} --- helemaal
        uitgegroeid; volwassenen kunnen zelf jongen krijgen;
  \item het wordt \emph{oud}\index{ouderdom}, en aan het eind
        \emph{gaat het dood}\index{dood}.
\end{enumerate}
\end{definition}

\begin{example}[Drie levens, dezelfde weg]\label{ex:g2:born-grow-die:three}
Een kitten wordt \omterm{def:g2:born-grow-die:cycle}{geboren}, speelt en groeit uit tot een kat, die zelf
kittens kan krijgen, en wordt oud. Een kersenpit ontkiemt, de jonge
boom wordt jarenlang hoger, de \omterm{def:g2:born-grow-die:cycle}{volwassen} boom maakt bloesem en kersen,
en na vele tientallen jaren gaat hij dood. En jij: baby, kind, en op
een dag \omterm{def:g2:born-grow-die:cycle}{volwassen}. Kitten, kersenboom, kind --- dezelfde vier \omterm{def:g2:born-grow-die:cycle}{stadia}.
\end{example}

\begin{remark}[Waarom het een cyclus heet]\label{rem:g2:born-grow-die:wheel}
Elk afzonderlijk leven loopt van \omterm{def:g2:born-grow-die:cycle}{geboorte} tot dood, als een weg met
twee einden. Maar vóór het einde krijgen \omterm{def:g2:born-grow-die:cycle}{volwassenen} jongen, en die
jongen beginnen de weg opnieuw. Over vele generaties gezien draait het
leven als een wiel --- daarom spreken we van een levens\emph{cyclus}.
\end{remark}

\section{Groeien en veranderen}

\begin{proposition}[Opgroeien is meer dan groter worden]\label{prop:g2:born-grow-die:change}
Terwijl het groeit, \emph{verandert} een jong \omterm{def:g1:living-or-not:living}{levend wezen} ook: een
baby krijgt tanden, verliest ze en krijgt sterkere terug; het gele dons
van een kuiken wordt echte \omterm{def:g1:animals-around-us:covering}{veren}; de zachte groene \omterm{def:g1:plants-around-us:parts}{stengel} van een
jonge boom wordt hard hout. Groeien is groter worden \emph{en} anders worden.
\end{proposition}
\admitted

\begin{example}[De fototest]\label{ex:g2:born-grow-die:photo}
Op de oude foto's van opa kun je één mens door alle \omterm{def:g2:born-grow-die:cycle}{stadia} volgen:
baby, schoolkind, jonge man, vader, grootvader. Dezelfde mens, een
veranderend lichaam. Foto's zijn kleine tijdmachines om een
\omterm{def:g2:born-grow-die:cycle}{levenscyclus} te bekijken.
\end{example}

\begin{method}[Levensstadia op volgorde zetten]\label{met:g2:born-grow-die:order}
Als je door elkaar gehusselde plaatjes van het leven van een \omterm{def:g1:living-or-not:living}{levend
wezen} krijgt:
\begin{enumerate}
  \item zoek het plaatje van de \omterm{def:g2:born-grow-die:cycle}{geboorte} --- ei, zaadje of pasgeboren baby;
  \item zoek de volwassene --- helemaal uitgegroeid, in staat jongen te krijgen;
  \item leg de jonge \omterm{def:g2:born-grow-die:cycle}{stadia} ertussen, van kleiner naar groter;
  \item controleer je rij: leidt elk plaatje naar het volgende?
\end{enumerate}
\end{method}

\section{Elk leven eindigt --- en het leven gaat door}

\begin{proposition}[Levensduur]\label{prop:g2:born-grow-die:lifespan}
Verschillende levende wezens leven verschillend lang --- een
\emph{levensduur}\index{levensduur}. Een vlinder leeft misschien een
paar weken; een kat en een hond tien tot twintig jaar; een mens vaak
meer dan tachtig jaar; sommige bomen honderden jaren. Lang of kort, elk
leven heeft een einde.
\end{proposition}
\admitted

\begin{example}[De oude appelboom]\label{ex:g2:born-grow-die:oldtree}
De kromme oude appelboom achter in de tuin geeft elk jaar minder
appels --- hij is in zijn \omterm{def:g2:born-grow-die:cycle}{ouderdom}. Maar in zijn appels zitten pitten,
en één geplante pit is al een jonge appelboom die aan de weg begint.
Als de oude boom doodgaat, zijn de appelbomen niet verdwenen: zijn
jongen gaan verder.
\end{example}

\begin{remark}[Praten over de dood]\label{rem:g2:born-grow-die:death}
Doodgaan hoort bij elke \omterm{def:g2:born-grow-die:cycle}{levenscyclus}, en het is normaal om daar
verdrietig over te zijn --- dat zijn mensen ook, vooral om huisdieren
en om mensen van wie ze houden. De \omterm{def:g1:living-or-not:living}{biologie} voegt één troost toe: omdat
\omterm{def:g2:born-grow-die:cycle}{volwassenen} jongen krijgen voordat ze doodgaan, gaat het leven zelf
door, cyclus na cyclus.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:1}
Noem de vier \omterm{def:g2:born-grow-die:cycle}{stadia} van een \omterm{def:g2:born-grow-die:cycle}{levenscyclus}, op volgorde.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:2}
In welk \omterm{def:g2:born-grow-die:cycle}{stadium} kan een \omterm{def:g1:living-or-not:living}{levend wezen} zelf jongen krijgen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:3}
Zet op volgorde, van oudste naar jongste: een schoolkind, een baby, een
grootmoeder, een moeder.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:4}
Geef de \omterm{def:g2:born-grow-die:cycle}{levenscyclus} van de kersenboom uit
\cref{ex:g2:born-grow-die:three}, \omterm{def:g2:born-grow-die:cycle}{stadium} voor \omterm{def:g2:born-grow-die:cycle}{stadium}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:5}
Geef twee voorbeelden uit \cref{prop:g2:born-grow-die:change} van een
jong \omterm{def:g1:living-or-not:living}{levend wezen} dat \emph{verandert} terwijl het groeit, en niet
alleen maar groter wordt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:6}
Wie leeft meestal het langst: een vlinder, een kat of een eik? Zet de
drie op volgorde van \omterm{prop:g2:born-grow-die:lifespan}{levensduur}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:7}
Waarom gebruiken we het woord \emph{cyclus}, terwijl elk afzonderlijk
leven als een weg van \omterm{def:g2:born-grow-die:cycle}{geboorte} tot dood loopt? Gebruik de paarse pijl
uit de figuur.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:8}
Probeer het: vraag je familie om foto's van één volwassene op
verschillende leeftijden, en leg ze in de volgorde van de \omterm{def:g2:born-grow-die:cycle}{levenscyclus}
met \cref{met:g2:born-grow-die:order}. Welke \omterm{def:g2:born-grow-die:cycle}{stadia} laten de foto's
zien? Welk \omterm{def:g2:born-grow-die:cycle}{stadium} moet nog komen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:born-grow-die:9}
De oude appelboom uit \cref{ex:g2:born-grow-die:oldtree} gaat op een
dag dood, en toch hoeft de tuin zijn appelbomen niet kwijt te raken.
Leg uit waarom, met het woord \emph{generatie} of met de paarse pijl
uit de figuur.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:born-grow-die:10}
Een kiezel gaat nooit dood. Komt dat doordat hij zo sterk is? Gebruik
het idee van levend en niet-levend uit \cref{ch:g1:living-or-not} om
het uit te leggen.
\end{exercise}
