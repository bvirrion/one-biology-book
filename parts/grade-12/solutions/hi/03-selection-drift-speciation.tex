\section*{अध्याय \ref{ch:g12:selection-drift-speciation} --- वरण, अपवाह और जातिउद्भव}

\begin{solution}{exo:g12:selection-drift-speciation:1}
किसी \omterm{def:g12:selection-drift-speciation:population}{आबादी} में किसी \omterm{def:g10:universal-dna:gene}{जीन} की सारी प्रतियों का वह अंश जो कोई दिया गया
\omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} हो। $p = 0.8$, $q = 0.2$ के साथ: $AA$ 64\%, $Aa$ 32\%, $aa$ 4\%।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:2}
वरण: अलग-अलग \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोते वाहकों के बीच प्रजनन का अंतर, जो पर्यावरण
बिठाता है और जो बारंबारताएँ एक दिशा में बदलता है। अपवाह: किसी सीमित \omterm{def:g12:selection-drift-speciation:population}{आबादी}
के प्रतिचयन से \omterm{def:g12:selection-drift-speciation:population}{बारंबारताओं} का यादृच्छिक उतार-चढ़ाव, जिसकी कोई दिशा नहीं।
वरण इस पर निर्भर है कि वह \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} करता क्या है; अपवाह नहीं।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:3}
1880 में लगभग 75\%, जब कालिख तनों को काला कर चुकी थी; और 1980 में लगभग
30\%, जब स्वच्छ-वायु क़ानून तनों को फिर फीका पड़ने दे चुके थे।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:4}
अगली पीढ़ी के \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} पिछली पीढ़ी का एक प्रतिदर्श हैं; और कोई छोटा
प्रतिदर्श स्रोत से किसी बड़े के मुक़ाबले अधिक हटता है, इसलिए बारंबारताएँ
अधिक डोलती हैं, और कोई \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} संयोग से खो सकता है।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:5}
दो \omterm{def:g12:selection-drift-speciation:population}{आबादियों} के बीच \omterm{def:g10:universal-dna:gene}{जीन} के लेन-देन में इतनी लंबी रुकावट कि वे तब तक अपसरित हो
जाएँ जब तक आपस में प्रजनन न कर सकें।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:6}
$a$ की प्रतियाँ: 1000 में से $160 + 40 = 200$, इसलिए $q = 0.2$, $p = 0.8$।
अपेक्षित: $0.64 \times 500 = 320$, $0.32 \times 500 = 160$, $0.04
\times 500 = 20$ --- यानी ठीक-ठीक वही गणना: वह \omterm{def:g12:selection-drift-speciation:population}{आबादी} संदर्भ अवस्था में
है।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:7}
$q^2 = 0.0001$: दस हज़ार में एक वह लक्षण दिखाता है; $2pq \approx
0.02$: पचास में एक उसे ढोता है। प्रतियों के निन्यानबे प्रतिशत
विषमयुग्मजियों में बैठे हैं, जिन पर उस लक्षण के ख़िलाफ़ वरण की कोई पकड़ नहीं।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:8}
दोनों बारहवीं पीढ़ी पर किसी सिरे तक पहुँचे, एक 100\% पर स्थिर हुआ और दूसरा
खो गया। तीसरा 20 पीढ़ियों में किसी भी सीमा को छुए बिना भटकता रहा --- फिर से
संयोग; और समय मिलता तो वह भी पहुँचता।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:9}
फीके पतंगे बिना प्रदूषण वाले देहात में प्रजनन करते रहे, जहाँ वे पसंद किए
जाते थे, और पतंगे उड़ते हैं: प्रवासी हर पीढ़ी में फीका \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} उस
कालिख भरे इलाक़े में वापस ढोते रहे, इसलिए वह कभी ग़ायब नहीं हुआ।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:10}
बीस व्यक्ति हर \omterm{def:g10:universal-dna:gene}{जीन} की अधिक से अधिक चालीस प्रतियाँ ढोते हैं, यानी मूल
\omterm{def:g10:universal-dna:gene}{युग्मविकल्पियों} का एक नन्हा प्रतिदर्श; अधिकांश उस सँकरी गली में खो गए, और
उसके बाद की छोटी \omterm{def:g12:selection-drift-speciation:population}{आबादी} में अपवाह ने और खो दिए। संख्याएँ लौट आईं;
\omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} नहीं लौट सकते, सिवाय किसी नए \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} से।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:11}
एक ही पर्यावरणीय बदलाव के बाद एक ही ढंग से हिलती दस \omterm{def:g12:selection-drift-speciation:population}{आबादियाँ}, और वह भी किसी
ठीक-ठाक क्रियाविधि के साथ: वरण। और तीस की कोई अकेली नन्ही \omterm{def:g12:selection-drift-speciation:population}{आबादी}, बिना किसी
बताए गए कारण के: जब तक कोई क्रियाविधि न दिखे तब तक अपवाह ही तयशुदा
व्याख्या है।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:12}
$AA$ $0.85^2 \approx 72\%$, $Aa$ $2 \times 0.85 \times 0.15 \approx
26\%$, $aa$ $0.15^2 \approx 2\%$। $aa$ मर जाते हैं, जिससे $a$ की
प्रतियाँ हटती हैं; पर $Aa$, यानी \omterm{def:g12:selection-drift-speciation:population}{आबादी} का एक चौथाई, $AA$ से बेहतर मलेरिया
झेलते हैं और $a$ की अपनी प्रतियाँ आगे बढ़ाते हैं। $aa$ से हुआ नुक़सान
$Aa$ के फ़ायदे से संतुलित हो जाता है, और $a$ 15\% पर टिका रहता है।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:13}
रंग से साथी का चुनाव: यानी बरताव की एक रुकावट। गँदले पानी में वह रुकावट चूक
जाती है और वे \omterm{def:g10:biodiversity-scales:species}{जातियाँ} संकरित हो जाती हैं, इसलिए अब तक कोई आनुवंशिक
असंगति जमा नहीं हुई है: वह अलगाव हाल का है और अब भी पलटा जा सकता है।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:14}
उस वलय के चारों ओर हर \omterm{def:g12:selection-drift-speciation:population}{आबादी} अपने पड़ोसियों से प्रजनन करती है, और सटी हुई
\omterm{def:g12:selection-drift-speciation:population}{आबादियों} के बीच केवल छोटे अंतर हैं; वे अंतर उस घेरे भर जुड़ते जाते हैं जब तक
मिलते हुए सिरे आपस में प्रजनन न करने लगें। पूरा प्रवणता-क्रम एक साथ दिखता
है: यानी \omterm{prop:g12:selection-drift-speciation:speciation}{जातिउद्भव} जमा हुए छोटे-छोटे क़दमों का मामला है। दोनों सिरे इसलिए
\omterm{def:g10:biodiversity-scales:species}{जातियाँ} कहलाते हैं क्योंकि जहाँ वे मिलते हैं वहाँ वे \omterm{def:g10:biodiversity-scales:species}{जातियों} जैसा बरताव
करते हैं: कोई \omterm{def:g10:universal-dna:gene}{जीन} लेन-देन नहीं।
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:15}
वरण कुछ \omterm{def:g10:universal-dna:gene}{युग्मविकल्पियों} के वाहकों को \emph{यहाँ और अभी} अधिक प्रजनन कराता
है: यानी इस पर्यावरण में वंशज छोड़ने में बेहतर, और वह भी उसी \omterm{def:g12:selection-drift-speciation:population}{आबादी} के बाक़ी
सदस्यों के मुक़ाबले --- न कि किसी परम अर्थ में बेहतर। पर्यावरण बदले तो वह
दिशा पलट जाती है (वे पतंगे); अपवाह \omterm{def:g12:selection-drift-speciation:population}{आबादियों} को बिना अनुकूलन के किसी कारण के
अलग कर देता है; और लैंगिक वरण ऐसे आभूषण पसंद करता है जो बचने में रोड़ा हैं।
विकास में सुधार की कोई दिशा नहीं है, केवल स्थानीय छँटाई है।
\end{solution}

\begin{solution}{pb:g12:selection-drift-speciation:1}
\textbf{1.} $D$ की प्रतियाँ: $2 \times 720 + 960 = 2400$; $d$ की: $960 +
2 \times 320 = 1600$; 4000 में से: $p = 0.6$, $q = 0.4$।

\textbf{2.} अपेक्षित $0.36 \times 2000 = 720$, $0.48 \times 2000 =
960$, $0.16 \times 2000 = 320$: यानी ठीक-ठीक वही गणना। हाँ।

\textbf{3.} $960/1680 \approx 57\%$।

\textbf{4.} $d$ की अधिकांश प्रतियाँ गहरे वाहकों में हैं। हटाव के बाद:
1680 चूहे, $D$ प्रतियाँ 2400, $d$ प्रतियाँ 960: $q = 960/3360 \approx
0.29$ --- यानी 0.40 से गिरावट, पर मिटने से कोसों दूर।

\textbf{5.} बड़ा आकार: किसी छोटी द्वीप \omterm{def:g12:selection-drift-speciation:population}{आबादी} में अपवाह होता है, इसलिए
बारंबारताएँ संयोग से भटकती हैं (और मुख्य भूमि से प्रवासन ग़ैरहाज़िर है,
\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} नगण्य)।

\textbf{6.} $DD$ 360, $Dd$ 480, $dd$ 320: यानी 1160 प्रजनक।

\textbf{7.} $D$: $720 + 480 = 1200$; $d$: $480 + 640 = 1120$; 2320 में से:
$p \approx 0.52$, $q \approx 0.48$।

\textbf{8.} अगली पीढ़ी (प्रति 2000): $DD$ $0.52^2 \approx 27\%$
(535), $Dd$ $\approx 50\%$ (999), $dd$ $\approx 23\%$ (466)। उल्लुओं
के बाद: 267, 500, 466। $D$: $534 + 500 = 1034$; $d$: $500 + 932 =
1432$; $q \approx 0.58$।

\textbf{9.} $q = 0.58$ के साथ: $DD$ 18\% (353), $Dd$ 49\% (974), $dd$
34\% (673); बचे 176, 487, 673; $d$: 2672 में से $487 + 1346 = 1833$:
$q \approx 0.69$। झुकाव: 0.40, 0.48, 0.58, 0.69 --- यानी प्रति पीढ़ी
लगभग 0.1 की चढ़ाई।

\textbf{10.} $D$ का हर वाहक गहरा है और उल्लुओं के सामने खुला है, इसलिए वरण
हर पीढ़ी में $D$ की हर प्रति पर काम करता है; और $q$ लगातार चढ़ता है। कोई
\omterm{def:g11:genes-and-disease:genetic}{अप्रभावी} \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} दुर्लभ होते ही विषमयुग्मजियों में छिप जाता, पर कोई
\omterm{def:g11:genes-and-disease:genetic}{प्रभावी} कभी नहीं छिपता: $D$ शून्य तक हाँका जा सकता है।

\textbf{11.} $D$: $4 + 3 = 7$; $d$: $3 + 2 = 5$; $q = 5/12 \approx
0.42$, जो निकाले की क़िस्मत से द्वीप वाले 0.40 के पास है।

\textbf{12.} प्रजनन करने वालों में $D$ की 5 प्रतियाँ हैं और $d$ की 1:
अगली पीढ़ी में $q$ औसतन $1/6$ है, पर $d$ की वह अकेली प्रति किसी भी बच्चे
तक न पहुँचे ($q = 0$) या कई तक पहुँचे; और $dd$ \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} पहले ही ग़ायब हो
चुका है, तथा उसके साथ बहुत-सी विविधता।

\textbf{13.} इतने कम प्रजनकों के साथ, कौन-से \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} आगे बढ़ते हैं यह
हर पीढ़ी में संयोग का मामला है; और कोई दुर्लभ \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} एक ही
बदक़िस्मत बार में खो सकता है।

\textbf{14.} $d$ अपवाहित होगा, और कुछ पीढ़ियों के भीतर स्थिर हो जाएगा या खो
जाएगा --- दोनों में से कोई भी नतीजा संयोग से। कोई पड़ोसी टापू, वैसी ही
शुरुआत से, उल्टे \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} पर ख़त्म हो सकता है: यानी वे दोनों द्वीप बिना
किसी अनुकूली कारण के कोट के रंग में अलग हैं।

\textbf{15.} \omterm{prop:g12:selection-drift-speciation:drift}{आनुवंशिक अपवाह}; और उस बस्ती का नन्हा आकार।

\textbf{16.} हाँ: वे द्वीप के चूहों से शायद ही प्रजनन करते हैं और उनके संकर
बंध्य हैं --- यानी प्रजनन अलगाव।

\textbf{17.} वह समुद्र, यानी एक भौगोलिक रुकावट; किसी अलग पर्यावरण में वरण
(कोई उल्लू नहीं, और दूसरी परिस्थितियाँ) तथा किसी छोटी \omterm{def:g12:selection-drift-speciation:population}{आबादी} में \omterm{prop:g12:selection-drift-speciation:drift}{आनुवंशिक
अपवाह}।

\textbf{18.} अलग-अलग मौसमों में प्रजनन करते चूहे साथी की तरह कभी मिलते ही
नहीं: यानी भूगोल की किसी रुकावट के बिना भी कोई \omterm{def:g10:universal-dna:gene}{जीन} नहीं बहते, और वे \omterm{def:g12:selection-drift-speciation:population}{आबादियाँ}
अपसरित होती रहती हैं।

\textbf{19.} अभी नहीं: वे बरताव और बंध्यता से अलग-थलग हैं और अग़ल-बग़ल दो
\omterm{def:g10:biodiversity-scales:species}{जातियाँ} ही बने रहेंगे। जबकि केवल 100 साल बाद वे आज़ादी से प्रजनन करते और
वापस एक ही \omterm{def:g12:selection-drift-speciation:population}{आबादी} में मिल जाते।

\textbf{20.} उल्लुओं की तीन पीढ़ियों के बाद $q$ 0.40 से लगभग 0.69 तक; उस
टापू पर $d$ \omterm{prop:g12:selection-drift-speciation:drift}{आनुवंशिक अपवाह} से स्थिर हुआ या खो गया; और आख़िर में दो
\omterm{def:g10:biodiversity-scales:species}{जातियाँ}।
\end{solution}
