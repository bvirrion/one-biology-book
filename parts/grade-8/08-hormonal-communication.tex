\section{Glands, hormones, targets}

\begin{definition}[The endocrine glands]\label{def:g8:hormonal-communication:glands}
A \emph{gland}\index{gland} whose product is released
\emph{into the \omterm{prop:g4:journey-of-food:absorb}{blood}} (rather than down a duct, like
\omterm{def:g4:journey-of-food:tract}{saliva}'s) is an \emph{endocrine} gland, and its products
are the \emph{hormones}\index{hormone}. The main writers:
the brain-base \emph{control gland} --- \omterm{def:g8:puberty:puberty}{puberty}'s
starter, and the conductor of most of the others; the
\emph{thyroid} in the neck --- its hormone sets the whole
body's running pace; the \emph{adrenal} caps on the
\omterm{def:g7:removing-wastes:kidneys}{kidneys} --- the alarm hormone of frights and sprints; the
\emph{pancreas} --- its hormone manages the \omterm{prop:g4:journey-of-food:absorb}{blood}'s
\omterm{prop:g7:organs-at-work:bill}{glucose}; and the \emph{\omterm{def:g8:reproductive-systems:female}{ovaries} and \omterm{def:g8:reproductive-systems:male}{testes}} --- the cycle
and rebuild hormones of
\cref{ch:g8:reproductive-systems}.
\end{definition}

\begin{proposition}[Addressing by receptor]\label{prop:g8:hormonal-communication:address}
The \omterm{prop:g4:journey-of-food:absorb}{blood} delivers every \omterm{def:g8:hormonal-communication:glands}{hormone} everywhere
(\cref{ex:g8:puberty:sites}) --- yet each acts only on
its \emph{target organs}\index{target organ}: the \omterm{def:g5:first-look-at-cells:cell}{cells}
built with docking sites, \emph{receptors}, for that
messenger. Broadcast carriage, receiver-side addressing:
a letter every house receives, readable only where the
key fits. (The \omterm{def:g8:nervous-communication:synapse}{synapse}'s docking of
\cref{def:g8:nervous-communication:synapse} is the same
trick at nanometre range.)
\end{proposition}
\admitted

\begin{example}[One messenger's tour: the alarm hormone]\label{ex:g8:hormonal-communication:alarm}
A near-miss on the bicycle: the adrenal \omterm{def:g8:hormonal-communication:glands}{glands} dump
their alarm \omterm{def:g8:hormonal-communication:glands}{hormone} into the \omterm{prop:g4:journey-of-food:absorb}{blood}, and within seconds
--- one circulation lap
(\cref{ex:g4:heart-and-blood:round}) --- every fitted
receiver responds at once: the \omterm{def:g4:heart-and-blood:heart}{heart}'s pace jumps, the
airways widen, the \omterm{prop:g7:digestion-and-nutrients:absorption}{liver} releases banked \omterm{prop:g7:organs-at-work:bill}{glucose}
(\cref{ex:g7:digestion-and-nutrients:breadtrace}'s
bank, called), \omterm{def:g3:bones-and-muscles:muscle}{muscle} \omterm{def:g4:heart-and-blood:vessels}{vessels} widen and the \omterm{def:g4:journey-of-food:tract}{digestive
tract}'s narrow
(\cref{prop:g7:organs-at-work:supply}'s rerouting,
hormone-held), and the \omterm{def:g1:the-five-senses:sense}{skin} goes pale and prickling.
One letter, six answers, one coordinated state: ready.
\end{example}

\section{Balance by feedback}

\begin{proposition}[The feedback principle]\label{prop:g8:hormonal-communication:feedback}
\omterm{def:g8:hormonal-communication:glands}{Hormone} systems regulate themselves by
\emph{feedback}\index{feedback}: the writing \omterm{def:g8:hormonal-communication:glands}{gland}
\emph{monitors the result} of its own letters, and slows
when the result is reached --- a thermostat, not a tap.
The showcase is \omterm{prop:g4:journey-of-food:absorb}{blood} \omterm{prop:g7:organs-at-work:bill}{glucose}: after a meal, the rising
\omterm{prop:g7:organs-at-work:bill}{glucose} itself triggers the pancreas's storage \omterm{def:g8:hormonal-communication:glands}{hormone};
\omterm{prop:g7:organs-at-work:bill}{glucose} banked, the level falls, and the secretion
subsides with it; between meals a partner \omterm{def:g8:hormonal-communication:glands}{hormone}
releases the bank the other way. The level rides a
narrow lane, day and night, with no conscious driver.
\end{proposition}
\admitted

\begin{example}[When feedback fails: diabetes]\label{ex:g8:hormonal-communication:diabetes}
In \emph{diabetes}\index{diabetes}, the \omterm{def:g8:hormonal-communication:glands}{storage-hormone}
signal fails --- the pancreas stops writing it, or the
receivers stop answering. \omterm{prop:g7:organs-at-work:bill}{Glucose} piles up in the \omterm{prop:g4:journey-of-food:absorb}{blood}
after every meal; the \omterm{def:g7:removing-wastes:kidneys}{kidney}'s reclaiming is overwhelmed
and sugar appears in the \omterm{def:g7:removing-wastes:kidneys}{urine} ---
\cref{ex:g7:removing-wastes:thrift}'s classic signal,
explained at last. Treatment is applied chapter: measured
doses of the missing \omterm{def:g8:hormonal-communication:glands}{hormone}, timed to meals --- the
letter written by syringe. Millions live well on exactly
this arrangement.
\end{example}

\begin{remark}[The two systems in partnership]\label{rem:g8:hormonal-communication:partnership}
\Cref{prop:g8:nervous-communication:compare}'s telegraph
and post are one administration. The brain-base control
\omterm{def:g8:hormonal-communication:glands}{gland} is the interface: \omterm{def:g5:brain-in-command:system}{nervous system} above, \omterm{def:g8:hormonal-communication:glands}{hormones}
below --- the cyclist's fright ran nerve-first,
hormone-sustained; \omterm{def:g8:puberty:puberty}{puberty} runs hormone-slow under a
brain-based conductor; the sheep of
\cref{ex:g8:reproduction-and-environment:lambs} turn
day-length (a nervous perception) into breeding \omterm{def:g8:hormonal-communication:glands}{hormones}.
Fast wires where speed matters, slow post where the
program must run for months, and translation offices
between.
\end{remark}

\section{Reading a hormone system}

\begin{method}[The four questions of any hormone]\label{met:g8:hormonal-communication:read}
For any \omterm{def:g8:hormonal-communication:glands}{hormone} you meet, establish:
\begin{enumerate}
  \item \emph{writer}: which \omterm{def:g8:hormonal-communication:glands}{gland}, into the \omterm{prop:g4:journey-of-food:absorb}{blood}, on
        what trigger?
  \item \emph{receivers}: which \omterm{prop:g8:hormonal-communication:address}{target organs} carry its
        receptors, and what does each do on delivery?
  \item \emph{\omterm{prop:g8:hormonal-communication:feedback}{feedback}}: what result reports back, and
        how is over-writing prevented?
  \item \emph{failure and use}: what happens when the
        letter fails --- and does medicine imitate it
        (the pill, the diabetic's doses)?
\end{enumerate}
\end{method}

\begin{example}[The method on the thyroid's pace-setter]\label{ex:g8:hormonal-communication:thyroid}
Writer: the thyroid, conducted by the control \omterm{def:g8:hormonal-communication:glands}{gland}.
Receivers: nearly every \omterm{def:g5:first-look-at-cells:cell}{cell} --- the \omterm{def:g8:hormonal-communication:glands}{hormone} sets the
rate of \cref{prop:g7:what-is-respiration:power}'s slow
fire. \omterm{prop:g8:hormonal-communication:feedback}{Feedback}: the control \omterm{def:g8:hormonal-communication:glands}{gland} reads the level and
trims its conducting. Failure: too little, and the whole
engine idles --- tiredness, cold, slowed everything; too
much, and it races. Use: the missing \omterm{def:g8:hormonal-communication:glands}{hormone} is a daily
tablet --- among medicine's oldest letters-by-post.
\end{example}

\section{Problem: The Glucose Logbook}

\begin{problem}[{Weekend problem --- a day of blood
sugar, read as correspondence}]\label{pb:g8:hormonal-communication:1}
A continuous \omterm{prop:g7:organs-at-work:bill}{glucose} monitor's (invented) day-log:
steady overnight; breakfast at 8 --- a rise, then a calm
descent by 10; sport at 12 --- a brief dip, quickly
caught; a missed lunch --- steady all afternoon
regardless; a fright at 17 (a near-miss on the bicycle)
--- a sharp brief rise; supper at 19 --- rise and
descent again.

\medskip\noindent\textbf{Part I --- The morning.}
\begin{enumerate}
  \item Overnight steadiness: which two-letter
        correspondence held the lane while its owner
        slept?
  \item Breakfast's rise and descent: name the trigger,
        the writer, the letter, and the receivers that
        banked the surplus.
  \item Why is the descent's \emph{stopping} --- no
        plunge below the lane --- itself \omterm{prop:g8:hormonal-communication:feedback}{feedback}'s
        signature?
  \item The 10 o'clock level matches the 7 o'clock one.
        What does the thermostat image predict about the
        secretion's course between?
\end{enumerate}

\medskip\noindent\textbf{Part II --- The afternoon.}
\begin{enumerate}[resume]
  \item Sport's dip and catch: which letter called the
        \omterm{prop:g7:digestion-and-nutrients:absorption}{liver}'s bank, and which chapter's account was
        being spent?
  \item The missed lunch changed nothing visible. Which
        \omterm{def:g8:hormonal-communication:glands}{partner-hormone}'s quiet work bridged the
        afternoon?
  \item The fright's sharp rise: name the writer and
        the point of the sudden sugar --- ready for
        what, by
        \cref{ex:g8:hormonal-communication:alarm}?
  \item Why did the fright's rise subside within the
        hour, unlike \omterm{ex:g8:hormonal-communication:diabetes}{diabetes}'s lasting pile-up?
\end{enumerate}

\medskip\noindent\textbf{Part III --- The consultation.}
\begin{enumerate}[resume]
  \item A diabetic's log of the same day would differ
        at every meal. Describe breakfast's curve
        untreated, and where the \omterm{def:g7:removing-wastes:kidneys}{kidney}'s signal would
        appear.
  \item The treated version: what does the measured
        dose before each meal imitate, and why must
        its size match the plate?
  \item The doctor calls the healthy log ``a
        conversation you never hear''. List the
        speakers of the day --- \omterm{def:g8:hormonal-communication:glands}{glands}, letters,
        receivers --- in one summary table or
        paragraph.
  \item Close with the chapter's moral in a sentence:
        what kind of system keeps a value in a lane
        for a lifetime without a driver?
\end{enumerate}
\end{problem}
