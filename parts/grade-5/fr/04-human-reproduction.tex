\chapter{La reproduction humaine}\label{ch:g5:human-reproduction}

Toutes les personnes que tu connais --- tes amis, tes enseignants,
la plus âgée de ta famille --- sont arrivées de la même façon~: sous
la forme d'un bébé, né après des mois passés à grandir dans le corps
de sa mère. Les humains sont des \omterm{prop:g3:sorting-living-things:groups}{mammifères}
(\cref{prop:g4:classifying-animals:classes}), et la reproduction
humaine suit le plan des \omterm{prop:g3:sorting-living-things:groups}{mammifères} de
\cref{ch:g4:animal-reproduction} --- avec les nombres, les dates et
les détails de notre propre \omterm{def:g5:biodiversity:species}{espèce}, que ce chapitre expose
simplement.

\section{La version humaine du plan}

\begin{proposition}[La reproduction humaine est une reproduction de mammifère]\label{prop:g5:human-reproduction:plan}
Comme tous les \omterm{prop:g3:sorting-living-things:groups}{mammifères}, les humains se reproduisent de façon
sexuée (\cref{def:g4:animal-reproduction:sexual})~:
\begin{itemize}
  \item le corps de la mère produit des \emph{\omterm{def:g4:animal-reproduction:sexual}{cellules œufs}}~;
        celui du père produit des \emph{\omterm{def:g4:animal-reproduction:sexual}{cellules mâles}}~;
  \item la \omterm{def:g5:flower-to-fruit:steps}{fécondation} --- une \omterm{def:g4:animal-reproduction:sexual}{cellule mâle} qui rejoint une
        \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} --- se produit \emph{dans le corps de la mère}
        (\cref{prop:g4:animal-reproduction:where})~;
  \item la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée --- une seule \omterm{def:g5:first-look-at-cells:cell}{cellule}
        (\cref{ex:g5:first-look-at-cells:one}) --- s'installe dans
        l'\emph{utérus}\index{utérus} de la mère et y grandit~;
  \item les humains sont \omterm{def:g4:animal-reproduction:ovivivi}{vivipares}
        (\cref{def:g4:animal-reproduction:ovivivi})~: le bébé naît
        vivant, et est d'abord nourri de \omterm{def:g2:animals-grow-up:born}{lait}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[La reproduction demande des corps adultes]\label{rem:g5:human-reproduction:adults}
Le corps d'un enfant ne peut pas faire de bébés~: les organes
reproducteurs ne se mettent à fonctionner qu'à la \emph{puberté}, le
changement de l'étape adolescente de
\cref{def:g3:stages-of-life:stages}, quand le corps se reconstruit
d'enfant en adulte. Ce que la puberté change, et comment, fait
l'objet d'un chapitre entier dans les années de collège de ce livre.
\end{remark}

\section{Neuf mois dans l'utérus}

\begin{definition}[Grossesse]\label{def:g5:human-reproduction:pregnancy}
La \emph{grossesse}\index{grossesse} est le temps que le futur bébé
passe à grandir dans l'\omterm{prop:g5:human-reproduction:plan}{utérus} de sa mère~: environ \emph{neuf mois}
chez les humains. Le bébé en croissance s'appelle d'abord un
\emph{embryon}\index{embryon}, puis --- une fois son plan
d'organisation en place, après deux mois environ --- un
\emph{fœtus}\index{fœtus}.
\end{definition}

\begin{proposition}[Nourri par le cordon]\label{prop:g5:human-reproduction:cord}
Dans l'\omterm{prop:g5:human-reproduction:plan}{utérus}, le bébé flotte dans une poche de liquide et ne peut
ni manger ni respirer. Tout lui arrive tout prêt du \omterm{prop:g4:journey-of-food:absorb}{sang} de sa
mère~: les \omterm{def:g4:journey-of-food:digestion}{nutriments} et le dioxygène
(\cref{prop:g4:heart-and-blood:carries}) passent dans le \omterm{prop:g4:journey-of-food:absorb}{sang} du
bébé par le \emph{cordon ombilical}\index{cordon ombilical} attaché
à son ventre --- et ses déchets repassent par le même chemin, pour
que le corps de la mère s'en occupe. Ton nombril est l'endroit où
ton propre cordon était attaché.
\end{proposition}
\admitted

\begin{example}[Le calendrier des neuf mois]\label{ex:g5:human-reproduction:timetable}
Mois 1~: un amas de \omterm{def:g5:first-look-at-cells:cell}{cellules} qui se divisent, plus petit qu'un grain
de riz, mais qui grandit déjà vite. Mois 2~: un \omterm{def:g5:human-reproduction:pregnancy}{embryon} avec
l'ébauche d'une tête, un \omterm{def:g4:heart-and-blood:heart}{cœur} qui bat, des bourgeons de \omterm{def:g1:my-body:parts}{bras} et de
\omterm{def:g1:my-body:parts}{jambes} --- toujours plus petit qu'un pouce. Mois 3--4~: un \omterm{def:g5:human-reproduction:pregnancy}{fœtus},
complet dans son plan, qui commence à bouger~; la mère sent bientôt
les coups de pied. Mois 5--8~: il grandit et mûrit --- l'ouïe
arrive, ainsi qu'un premier duvet. Mois 9~: le bébé, d'ordinaire la
tête en bas et pesant environ \qty{3}{kg}, est prêt.
\end{example}

\begin{example}[La naissance]\label{ex:g5:human-reproduction:birth}
À la \omterm{def:g2:born-grow-die:cycle}{naissance}, les puissants \omterm{def:g3:bones-and-muscles:muscle}{muscles} de l'\omterm{prop:g5:human-reproduction:plan}{utérus} poussent le bébé
dehors, la tête la première le plus souvent. Le premier cri remplit
les \omterm{def:g4:breathing:organs}{poumons} d'air --- ils n'avaient jamais respiré
(\cref{prop:g5:human-reproduction:cord}) --- le cordon, devenu
inutile, est coupé sans douleur, et le plan des \omterm{prop:g3:sorting-living-things:groups}{mammifères} se
poursuit au-dehors~: \omterm{def:g2:animals-grow-up:born}{lait}, chaleur et soins
(\cref{ex:g4:animal-reproduction:whalecalf} approuverait).
\end{example}

\begin{remark}[Les jumeaux]\label{rem:g5:human-reproduction:twins}
Parfois deux bébés grandissent ensemble dans l'\omterm{prop:g5:human-reproduction:plan}{utérus}. Si une seule
\omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée s'est partagée en deux \omterm{def:g5:human-reproduction:pregnancy}{embryons}, les jumeaux
sont \emph{vrais jumeaux} --- bâtis à partir de la même \omterm{def:g5:first-look-at-cells:cell}{cellule} de
départ, semblables comme deux copies. Si deux \omterm{def:g4:animal-reproduction:sexual}{cellules œufs} ont été
fécondées par deux \omterm{def:g4:animal-reproduction:sexual}{cellules mâles}, les jumeaux ne se ressemblent pas
plus que des frères et sœurs ordinaires --- ils n'ont partagé que
l'\omterm{prop:g5:human-reproduction:plan}{utérus}. Deux sortes de jumeaux, que leur commencement distingue.
\end{remark}

\section{Les soins avant et après}

\begin{proposition}[Le bébé partage les approvisionnements de sa mère]\label{prop:g5:human-reproduction:care}
Tout ce qui parvient au bébé passe par le \omterm{prop:g4:journey-of-food:absorb}{sang} de sa mère --- c'est
pourquoi une future mère mange bien
(\cref{ch:g3:balanced-diet}), se repose, est suivie par des médecins
ou des sages-femmes, et évite ce qui pourrait nuire au bébé~: la
fumée de tabac et l'alcool passent aussi par le cordon, et un \omterm{def:g5:human-reproduction:pregnancy}{fœtus}
est bien plus fragile qu'un adulte. Prendre soin de la mère,
\emph{c'est} prendre soin du bébé.
\end{proposition}
\admitted

\begin{example}[Une espèce très soignée]\label{ex:g5:human-reproduction:longcare}
Même parmi les \omterm{prop:g3:sorting-living-things:groups}{mammifères}, les humains poussent à l'extrême la
stratégie du «~peu et bien soignés~» de
\cref{prop:g4:animal-reproduction:strategies}~: d'ordinaire un seul
bébé à la fois, sans défense bien plus longtemps qu'un poulain ou un
chaton, et soigné --- nourri, instruit, aimé --- pendant de longues
\emph{années}. L'enfance humaine est l'une des plus longues du monde
\omterm{def:g1:animals-around-us:animal}{animal}~; ce livre, un cours long d'une décennie, est lui-même un
morceau de ce long soin.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:1}
Quelle \omterm{def:g5:first-look-at-cells:cell}{cellule} vient de la mère, laquelle du père, et comment
appelle-t-on leur rencontre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:2}
Où grandit le bébé humain, et pendant combien de temps environ~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:3}
Quels sont les deux noms du bébé en croissance, dans l'ordre, et à
peu près quand le premier devient-il le second~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:4}
Le bébé dans l'\omterm{prop:g5:human-reproduction:plan}{utérus} ne mange ni ne respire. Comment reçoit-il
\omterm{def:g4:journey-of-food:digestion}{nutriments} et dioxygène, et où vont ses déchets~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:5}
De quoi ton nombril est-il la marque~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:6}
Que se passe-t-il au premier cri, et pourquoi est-ce un tel
changement pour le corps du bébé~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:7}
Quels caractères font de la reproduction humaine une reproduction
typiquement mammifère~? Nomme-en trois, en citant
\cref{prop:g4:classifying-animals:classes} ou
\cref{ch:g4:animal-reproduction}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:8}
Pourquoi les médecins demandent-ils aux futures mères d'éviter la
fumée de tabac et l'alcool~? Utilise
\cref{prop:g5:human-reproduction:care}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:9}
Explique les deux sortes de jumeaux par leurs commencements.
Laquelle est semblable «~comme deux copies~», et pourquoi cela
découle-t-il de \cref{ex:g5:first-look-at-cells:one}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:10}
Place l'humain sur la ligne des stratégies de
\cref{prop:g4:animal-reproduction:strategies}, en le comparant à la
morue et à l'éléphante~: nombre de jeunes, durée des soins, et ce
que ce long soin contient chez les humains.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:human-reproduction:11}
À la \omterm{def:g2:born-grow-die:cycle}{naissance}, le cordon coupé met fin à la ligne
d'approvisionnement du bébé à l'instant même où les \omterm{def:g4:breathing:organs}{poumons} prennent
leur première \omterm{def:g4:breathing:movements}{inspiration}. À l'aide de
\cref{prop:g5:human-reproduction:cord} et de
\cref{prop:g4:breathing:exchange}, explique quel travail bascule, en
cette minute, du corps de la mère aux \omterm{def:g4:breathing:organs}{poumons} du bébé --- et
pourquoi le premier cri tombe à point nommé.
\end{exercise}
