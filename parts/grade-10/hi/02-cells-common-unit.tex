\chapter{कोशिकाएँ: जीवन की साझा इकाई}\label{ch:g10:cells-common-unit}

रुई की एक फाहा से अपने गाल के भीतरी हिस्से को खुरचिए, उसे एक स्लाइड पर
लगाइए, नीले रंजक की एक बूँद डालिए, और चार सौ गुना पर सूक्ष्मदर्शी से
देखिए: चपटी, बेढंगी पट्टियों का एक बिखराव, और हर एक के बीचोबीच के पास एक
गहरा धब्बा। प्याज़ के शल्क के भीतर से पतली झिल्ली उतारकर फिर देखिए:
सुथरी आयताकार ईंटें, हर एक के अपने धब्बे के साथ। तालाब के पानी में
अंडाकार जीव रोओं की झालर के साथ खेते हुए निकलते दिखते हैं। नंगी आँख से
इन तीनों नमूनों में कुछ भी साझा नहीं। लेंस के नीचे ये तीनों एक ही वस्तु
से बने निकलते हैं --- \omterm{def:g10:cells-common-unit:cell}{कोशिका} से --- और यह अध्याय इसी के बारे में है कि
वह वस्तु है क्या, उसे देखा कैसे जाता है, और हर सजीव उसी से क्यों बना है।

\section{कोशिका सिद्धांत}

\begin{definition}[कोशिका]\label{def:g10:cells-common-unit:cell}
\emph{कोशिका}\index{कोशिका} किसी सजीव की वह सबसे छोटी इकाई है जो ख़ुद
जीवित हो: जलीय अंतर्वस्तु का एक आयतन, यानी
\emph{कोशिकाद्रव्य}\index{कोशिकाद्रव्य}, जिसे एक पतली
\emph{प्लाज़्मा झिल्ली}\index{प्लाज़्मा झिल्ली} घेरे रहती है और जो उसे
परिवेश से अलग रखते हुए भी चुने हुए पदार्थों को पार जाने देती है। कोई
कोशिका पदार्थ और ऊर्जा भीतर लेती है, उन्हें बदलती है, बढ़ती है, और दो
कोशिकाओं में बँट जाती है।
\end{definition}

\begin{proposition}[कोशिका सिद्धांत]\label{prop:g10:cells-common-unit:theory}
उन्नीसवीं सदी में सूक्ष्मदर्शन से स्थापित तीन कथन मिलकर \omterm{def:g10:cells-common-unit:cell}{कोशिका} सिद्धांत
बनाते हैं:
\begin{enumerate}
  \item हर सजीव एक या अधिक \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से बना है;
  \item \omterm{def:g10:cells-common-unit:cell}{कोशिका} सजीवों की संरचना और क्रिया की मूल इकाई है --- उससे छोटा
        कुछ भी अपने बल पर जीवित नहीं है;
  \item हर \omterm{def:g10:cells-common-unit:cell}{कोशिका} विभाजन द्वारा पहले से मौजूद किसी \omterm{def:g10:cells-common-unit:cell}{कोशिका} से ही आती है।
\end{enumerate}
\end{proposition}
\begin{proof}[साक्ष्य]
पहला कथन बेहतर सूक्ष्मदर्शियों से पौधों (श्लाइडेन, 1838) और जंतुओं
(श्वान, 1839) की व्यवस्थित जाँच पर टिका है: जिस भी ऊतक को देखा गया, वह
दिखने में चाहे कितना ही अलग हो, \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में खुल गया। तीसरा कथन विरचो
(1855) ने और बढ़ते ऊतकों में विभाजित होती \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के प्रेक्षण ने
स्थापित किया; दूसरा विकल्प --- कि \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} तरल से अपने आप बन जाती हैं ---
कभी देखा ही नहीं गया, और सूक्ष्मजीवों पर पास्तर के प्रयोगों ने
(\cref{ch:g11:antibiotic-resistance} उन पर लौटता है) यह सवाल बंद कर दिया।
दूसरा कथन पहले से निकलता है, और इस तथ्य से भी कि उपयुक्त तरल में रखी
अलग-अलग \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} जीती रहती हैं, जबकि किसी \omterm{def:g10:cells-common-unit:cell}{कोशिका} का कोई टुकड़ा नहीं।
\end{proof}

\begin{example}[एक कोशिका या अनेक]\label{ex:g10:cells-common-unit:onemany}
खमीर, पैरामीशियम या कोई जीवाणु अकेली एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} है जो सब कुछ करती है ---
खाती है, चलती है, बँटती है। बलूत या मनुष्य कई तरह की \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की एक
सहकारी है, जिसमें हर \omterm{def:g10:cells-common-unit:cell}{कोशिका} काम का एक हिस्सा सँभालती है: मानव शरीर में
लगभग $\num{3e13}$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} हैं, और मोटे तौर पर दो सौ तरह की। इन दोनों के
बीच में कॉलोनियाँ और सरल बहुकोशिकीय जीव आते हैं, पर ऐसा कोई सजीव नहीं जो
\omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के अलावा किसी और चीज़ से बना हो।
\end{example}

\section{कोशिकाएँ देखना: पैमाने और सूक्ष्मदर्शी}

\begin{definition}[आवर्धन और विभेदन]\label{def:g10:cells-common-unit:magnification}
किसी उपकरण का \emph{आवर्धन}\index{आवर्धन} प्रतिबिंब के आकार और वस्तु के
आकार का अनुपात है। उसका \emph{विभेदन}\index{विभेदन} वह सबसे छोटी दूरी है
जिस पर दो बिंदुओं को वह अब भी अलग-अलग दिखाता है। विभेदन जितना उचित
ठहराता है उससे आगे प्रतिबिंब बढ़ाने पर धुँधलापन बढ़ता है, ब्योरा नहीं।
\end{definition}

\begin{proposition}[दो उपकरण, दो पैमाने]\label{prop:g10:cells-common-unit:microscopes}
प्रकाश सूक्ष्मदर्शी लगभग $\times 1500$ तक \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} देता है और उसका \omterm{def:g10:cells-common-unit:magnification}{विभेदन}
लगभग \qty{0.2}{\micro m} है: वह \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}, \omterm{def:g10:cells-common-unit:organelle}{केंद्रक}, \omterm{def:g10:cells-common-unit:organelle}{हरितलवक} और जीवाणु
रंगों में और जीवित दिखाता है। इलेक्ट्रॉन सूक्ष्मदर्शी प्रकाश की जगह
इलेक्ट्रॉनों का पुंज इस्तेमाल करता है: \omterm{def:g10:cells-common-unit:magnification}{विभेदन} \qty{1}{nm} से नीचे,
\omterm{def:g10:cells-common-unit:magnification}{आवर्धन} $\times 10^6$ तक; वह \omterm{def:g10:cells-common-unit:cell}{कोशिका} की भीतरी संरचना खोल देता है ---
झिल्लियाँ, \omterm{def:g10:cells-common-unit:organelle}{सूत्रकणिकाएँ}, \omterm{def:g10:cells-common-unit:organelle}{राइबोसोम}, यहाँ तक कि बड़े अणु भी --- पर उस
नमूने पर जो मरा हुआ, सूखा हुआ और माइक्रोमीटर से भी पतली फाँकों में कटा
होता है।
\end{proposition}
\admitted

\begin{method}[सूक्ष्मचित्र पर किसी वस्तु को मापना]\label{met:g10:cells-common-unit:measure}
हर सूक्ष्मचित्र या तो \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} बताता है या मापनी पट्टी दिखाता है।
\begin{enumerate}
  \item मापनी पट्टी के साथ: पन्ने पर पट्टी को मापिए (मान लीजिए
        \qty{10}{\micro m} लिखी पट्टी \qty{12}{mm} की निकली) और वस्तु
        को भी (\qty{30}{mm}); तब वास्तविक आकार $30 \times 10/12 =
        \qty{25}{\micro m}$ है।
  \item \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} के साथ: वास्तविक आकार $=$ मापा गया आकार, \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} से भाग
        देकर; $\times 400$ पर \qty{20}{mm} लंबी खींची गई \omterm{def:g10:cells-common-unit:cell}{कोशिका}
        $20/400 = \qty{0.05}{mm} = \qty{50}{\micro m}$ की है।
  \item मात्रक हमेशा लिखिए, और बदलिए भी: $\qty{1}{mm} =
        \qty{1000}{\micro m}$, $\qty{1}{\micro m} = \qty{1000}{nm}$।
\end{enumerate}
\end{method}

\begin{example}[गाल की कोशिकाएँ]\label{ex:g10:cells-common-unit:cheek}
$\times 400$ पर \qty{24}{mm} चौड़ी खींची गई गाल की \omterm{def:g10:cells-common-unit:cell}{कोशिका}
$24/400 = \qty{0.06}{mm} = \qty{60}{\micro m}$ चौड़ी है, और उसका
\omterm{def:g10:cells-common-unit:organelle}{केंद्रक}, जो \qty{3}{mm} खींचा गया, लगभग \qty{8}{\micro m} का। लाल रक्त
\omterm{def:g10:cells-common-unit:cell}{कोशिका} \qty{7}{\micro m} की होती है; और मुँह के जीवाणु, जो
\qty{1}{\micro m} लंबे हैं, उसी \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} पर मुश्किल से ही अलग-अलग
बिंदुओं जैसे दिखते हैं।
\end{example}

\section{कोशिका के भीतर}

\begin{definition}[कोशिकांग]\label{def:g10:cells-common-unit:organelle}
\emph{कोशिकांग}\index{कोशिकांग} \omterm{def:g10:cells-common-unit:cell}{कोशिका} के भीतर की वह रचना है जो अपनी ही
झिल्ली से घिरी होती है और कोई एक निश्चित काम करती है। मुख्य कोशिकांग ये
हैं:
\begin{itemize}
  \item \emph{केंद्रक}\index{केंद्रक}, जिसमें गुणसूत्र रहते हैं --- यानी
        \cref{ch:g10:universal-dna} वाला DNA --- और जो \omterm{def:g10:cells-common-unit:cell}{कोशिका} की
        गतिविधि पर नियंत्रण रखता है;
  \item \emph{सूत्रकणिकाएँ}\index{सूत्रकणिका}, जो लगभग
        \qty{1}{\micro m} लंबी होती हैं और जहाँ कार्बनिक अणुओं को
        ऑक्सीजन के साथ तोड़कर काम में आने वाली ऊर्जा छोड़ी जाती है
        (कोशिकीय श्वसन, \cref{ch:g10:cell-metabolism});
  \item पादप \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में \emph{हरितलवक}\index{हरितलवक}, जो पर्णहरित से
        हरे होते हैं और जहाँ प्रकाशसंश्लेषण होता है, और साथ में एक बड़ी
        जल से भरी \emph{रसधानी}\index{रसधानी}, जो \omterm{def:g10:cells-common-unit:cell}{कोशिका} को तना हुआ रखती
        है।
\end{itemize}
झिल्ली के बाहर पादप \omterm{def:g10:cells-common-unit:cell}{कोशिका} सेलुलोज़ की एक दृढ़
\emph{कोशिका भित्ति}\index{कोशिका भित्ति} में बंद रहती है; जंतु \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं}
में ऐसी कोई भित्ति नहीं होती। हर \omterm{def:g10:cells-common-unit:cell}{कोशिका} के \omterm{def:g10:cells-common-unit:cell}{कोशिकाद्रव्य} में लाखों
\emph{राइबोसोम}\index{राइबोसोम} भी होते हैं, यानी लगभग \qty{25}{nm} के
कण, जो \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} जोड़ते हैं और केवल इलेक्ट्रॉन सूक्ष्मदर्शी से दिखते हैं।
\end{definition}

\begin{definition}[प्राक्केंद्रकी और सुकेंद्रकी]\label{def:g10:cells-common-unit:prokaryote}
जिस \omterm{def:g10:cells-common-unit:cell}{कोशिका} का DNA किसी \omterm{def:g10:cells-common-unit:organelle}{केंद्रक} में बंद रहता है वह
\emph{सुकेंद्रकी कोशिका}\index{सुकेंद्रकी कोशिका} है: जंतु, पौधे, कवक
और बहुत-से एककोशिकीय जीव सुकेंद्रकी हैं।
\emph{प्राक्केंद्रकी कोशिका}\index{प्राक्केंद्रकी कोशिका} में न कोई
\omterm{def:g10:cells-common-unit:organelle}{केंद्रक} होता है, न झिल्ली से घिरे \omterm{def:g10:cells-common-unit:organelle}{कोशिकांग}: उसका DNA \omterm{def:g10:cells-common-unit:cell}{कोशिकाद्रव्य} में
खुला पड़ा रहता है, उसका आकार \qty{1}{\micro m} के आसपास होता है, और वह
अपनी ही तरह की एक भित्ति से घिरी रहती है। जीवाणु प्राक्केंद्रकी हैं।
\end{definition}

\omterm{def:g10:cells-common-unit:cell}{कोशिकाद्रव्य}, और बीच में एक फीका क्षेत्र जहाँ अकेला वृत्ताकार DNA अणु
पड़ा है। न कोई \omterm{def:g10:cells-common-unit:organelle}{केंद्रक}, न \omterm{def:g10:cells-common-unit:organelle}{सूत्रकणिकाएँ}। फिर भी यह \omterm{def:g10:cells-common-unit:cell}{कोशिका} पोषक शोरबे में
खाती है, बढ़ती है और हर बीस मिनट में बँट जाती है: \omterm{def:g10:cells-common-unit:cell}{कोशिका} सिद्धांत जो कुछ
माँगता है, वह सब गाल की \omterm{def:g10:cells-common-unit:cell}{कोशिका} के हज़ारवें हिस्से भर आयतन में।

\section{बहुत-सी कोशिकाएँ, बहुत-से आकार: विशेषीकरण}

\begin{proposition}[विशेषीकृत कोशिकाएँ]\label{prop:g10:cells-common-unit:specialised}
किसी बहुकोशिकीय जीव में सब \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की बुनियादी योजना और DNA एक ही होते
हैं, पर हर तरह की \omterm{def:g10:cells-common-unit:cell}{कोशिका} का आकार और साज़-सामान उसके काम से मेल खाता है:
कोई न्यूरॉन संकेत ढोने के लिए एक मीटर तक लंबा तंतु फैला देता है; लाल रक्त
\omterm{def:g10:cells-common-unit:cell}{कोशिका} अपना \omterm{def:g10:cells-common-unit:organelle}{केंद्रक} खो चुकी है और हीमोग्लोबिन से ठसाठस भरी एक चपटी तश्तरी
है; पेशी तंतु संकुचनशील प्रोटीनों से भरी एक विशाल \omterm{def:g10:cells-common-unit:cell}{कोशिका} है; मूलरोम एक
पतली नली है जो जल सोखने के लिए सतह को अधिकतम कर देती है; और पत्ती की
द्वार \omterm{def:g10:cells-common-unit:cell}{कोशिका} किसी छिद्र को खोलने या बंद करने के लिए अपना आकार बदल लेती
है। विशेषीकरण इसी बात का मामला है कि साझा योजना के कौन-से हिस्से विकसित
किए गए हैं।
\end{proposition}
\admitted

\begin{example}[सतह और आयतन]\label{ex:g10:cells-common-unit:surface}
\omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} इतनी छोटी क्यों होती हैं? कोई \omterm{def:g10:cells-common-unit:cell}{कोशिका} जो कुछ भीतर लेती या बाहर
देती है, वह सब उसकी झिल्ली से पार होता है, और झिल्ली का क्षेत्रफल आकार
के वर्ग के साथ बढ़ता है, जबकि ज़रूरतें आयतन के साथ, यानी घन के साथ।
\qty{10}{\micro m} भुजा की घनाकार \omterm{def:g10:cells-common-unit:cell}{कोशिका} की सतह \qty{600}{\micro m^2} है
और आयतन \qty{1000}{\micro m^3}: यानी प्रति घन माइक्रोमीटर
\qty{0.6}{\micro m^2} झिल्ली। \qty{100}{\micro m} भुजा पर यह अनुपात
गिरकर \qty{0.06}{\micro m^2} रह जाता है: प्रति इकाई अंतर्वस्तु दस गुना
कम झिल्ली। कोई बड़ी \omterm{def:g10:cells-common-unit:cell}{कोशिका} अपने बीच में भूखी मर जाती; जो \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} बड़ी
हैं (मेंढक का अंडा, पेशी तंतु) वे या तो ज़्यादातर निष्क्रिय भंडार हैं या
इतनी लंबी हैं कि कोई भी बिंदु सतह से दूर न पड़े।
\end{example}

\begin{remark}[खुले तंत्र के रूप में कोशिका]\label{rem:g10:cells-common-unit:open}
\omterm{def:g10:cells-common-unit:cell}{कोशिका} कोई बंद थैली नहीं है। जल, ऑक्सीजन, ग्लूकोज़ और आयन लगातार उसकी
झिल्ली पार करते हैं, और उसका कचरा भी उसी रास्ते बाहर जाता है; शुद्ध जल
में रखी गाल की \omterm{def:g10:cells-common-unit:cell}{कोशिका} कुछ ही मिनटों में फूलकर फट जाती है, और तेज़ नमकीन
पानी में रखी \omterm{def:g10:cells-common-unit:cell}{कोशिका} सिकुड़ जाती है। भीतर क्या आए और बाहर क्या जाए, इस पर
झिल्ली का नियंत्रण स्नातक वर्ष 1 वाले खंड का विषय है; अभी इतना याद रखिए
कि \omterm{def:g10:cells-common-unit:cell}{कोशिका} अपने परिवेश से आदान-प्रदान करके जीती है, और आदान-प्रदान रुकते
ही मर जाती है।
\end{remark}

निकालिए। इनमें से कौन अकेले अपनी झिल्ली से होने वाले सीधे विसरण के भरोसे
रह सकती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cells-common-unit:11}
पोषक शोरबे में कोई जीवाणु हर \qty{20}{min} में बँटता है। एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} से

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:cells-common-unit:12}
लाल रक्त \omterm{def:g10:cells-common-unit:cell}{कोशिका} में न \omterm{def:g10:cells-common-unit:organelle}{केंद्रक} होता है, न \omterm{def:g10:cells-common-unit:organelle}{सूत्रकणिकाएँ}। क्या वह

कीजिए कि वह लगभग 120 दिन जीती है, बँट नहीं सकती, और अस्थि मज्जा की

\begin{exercise}[$\star\star\star$]\label{exo:g10:cells-common-unit:13}
1665 में हुक ने कॉर्क की एक फाँक में ``कोठरियाँ'' देखी थीं --- मोटी
दीवारों वाले ख़ाली डिब्बे। इस अध्याय की शब्दावली में समझाइए कि उन्होंने
असल में क्या देखा था, और उसमें कोई \omterm{def:g10:cells-common-unit:organelle}{केंद्रक} क्यों नहीं दिख सकता था।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:cells-common-unit:14}
विषाणु \qty{20}{nm} से \qty{300}{nm} तक के कण हैं, जिनमें \omterm{def:g10:chemistry-of-life:families}{न्यूक्लिक
अम्ल} और \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} होते हैं, जो अपने बल पर न बढ़ सकते हैं न बँट सकते हैं,
और जो केवल किसी \omterm{def:g10:cells-common-unit:cell}{कोशिका} के भीतर ही गुणित होते हैं। क्या वे \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} हैं?
क्या वे जीवित हैं? \omterm{def:g10:cells-common-unit:cell}{कोशिका} सिद्धांत के आधार पर तर्क दीजिए।
\end{exercise}

चौड़ी, जिसमें सैकड़ों \omterm{def:g10:cells-common-unit:organelle}{केंद्रक} होते हैं। उसका आयतन निकालिए और

हैं।
\end{exercise}

\section{समस्या: किसी शरीर की कोशिकाएँ गिनना}

\begin{problem}[{सप्ताहांत समस्या --- सूक्ष्मदर्शन का एक सत्र, स्लाइड पर
रखी प्याज़ की झिल्ली से लेकर मानव शरीर में कोशिकाओं की संख्या के अनुमान
तक}]
\label{pb:g10:cells-common-unit:1}
एक कक्षा सूक्ष्मदर्शी पर एक दोपहर बिताती है। उपलब्ध अभिदृश्यक $\times 40$,
$\times 100$ और $\times 400$ का कुल \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} देते हैं; $\times 100$ पर
दृष्टि-क्षेत्र की चौड़ाई \qty{1.8}{mm} है।

\medskip
\noindent\textbf{भाग I --- प्याज़।}
\begin{enumerate}
  \item $\times 100$ पर प्याज़ की लगभग 9 \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} दृष्टि-क्षेत्र में
        अग़ल-बग़ल आ जाती हैं। एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} की चौड़ाई माइक्रोमीटर में
        आँकिए।
  \item $\times 400$ पर दृष्टि-क्षेत्र चार गुना सँकरा है। वह कितना चौड़ा
        है, और उसकी चौड़ाई में इनमें से कितनी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} आएँगी?
  \item एक विद्यार्थी $\times 400$ के प्रतिबिंब से एक \omterm{def:g10:cells-common-unit:cell}{कोशिका}
        \qty{28}{mm} चौड़ी खींचता है। असली \omterm{def:g10:cells-common-unit:cell}{कोशिका} के मुक़ाबले उस चित्र
        का \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} क्या है?
  \item आयोडीन \omterm{def:g10:cells-common-unit:organelle}{केंद्रक} को पीला-भूरा रँग देता है; विद्यार्थी उसे चित्र पर
        \qty{4}{mm} नापता है। वास्तविक व्यास?
  \item उन रचनाओं के नाम लिखिए जिन पर विद्यार्थी को चित्र में लेबल लगाना
        चाहिए, और एक ऐसी रचना का नाम भी जो मौजूद तो है पर इस \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} पर
        दिख नहीं सकती।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- गाल और तालाब।}
\begin{enumerate}[resume]
  \item मेथिलीन ब्लू से रँगी गाल की \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} लगभग \qty{60}{\micro m}
        चौड़े पर केवल \qty{5}{\micro m} मोटे चपटे बहुभुजों जैसी दिखती
        हैं। वे कहाँ से आती हैं, यह देखते हुए वे चपटी क्यों हैं?
  \item तालाब के पानी की एक बूँद में \qty{240}{\micro m} लंबा एक
        पैरामीशियम दिखता है। क्या उसे नंगी आँख से देखा जा सकता है? किस
        \omterm{def:g10:cells-common-unit:magnification}{आवर्धन} पर वह दृष्टि-क्षेत्र का चौथाई हिस्सा भर देगा?
  \item $\times 400$ पर पैरामीशियम के पास लगभग \qty{1}{\micro m} लंबे
        नन्हे हिलते-डुलते बिंदु दिखते हैं। सबसे संभावित रूप से वे क्या
        हैं, और इलेक्ट्रॉन सूक्ष्मदर्शी उनमें क्या जोड़ता?
  \item पैरामीशियम अकेली एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} है; गाल की \omterm{def:g10:cells-common-unit:cell}{कोशिका} लाखों करोड़ों में
        से एक। एक-एक वाक्य में बताइए कि गाल की \omterm{def:g10:cells-common-unit:cell}{कोशिका} के लिए
        ``विशेषीकृत'' का क्या अर्थ है, और वह पैरामीशियम पर क्यों लागू
        नहीं हो सकता।
  \item एक विद्यार्थी गाल की \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} पर नमकीन पानी की एक बूँद डालता है
        और उन्हें सिकुड़ते देखता है।
        \cref{rem:g10:cells-common-unit:open} की मदद से समझाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:g10:cells-common-unit:cell}{कोशिका} का आकार।}
\begin{enumerate}[resume]
  \item किसी सामान्य मानव \omterm{def:g10:cells-common-unit:cell}{कोशिका} को \qty{20}{\micro m} भुजा का घन
        मानिए। उसका आयतन घन माइक्रोमीटर में निकालिए, फिर घन मिलीमीटर में
        ($\qty{1}{mm^3} = 10^9\,\unit{\micro m^3}$)।
  \item उसका पृष्ठीय क्षेत्रफल और सतह/आयतन अनुपात निकालिए।
  \item \qty{1}{\micro m} भुजा का घन मानकर किसी जीवाणु के लिए वही
        दोहराइए। उसका अनुपात कितने गुना बड़ा है?
  \item \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} मोटे तौर पर उतने ही आयतन के जल के बराबर द्रव्यमान लेती
        हैं (प्रति \unit{cm^3} $\qty{1}{g}$)। \qty{20}{\micro m} वाली
        \omterm{def:g10:cells-common-unit:cell}{कोशिका} के द्रव्यमान का अनुमान नैनोग्राम में लगाइए
        ($\qty{1}{g} = 10^9\,\unit{ng}$)।
  \item लाल रक्त \omterm{def:g10:cells-common-unit:cell}{कोशिका} \qty{7}{\micro m} व्यास और \qty{2}{\micro m}
        मोटी एक तश्तरी है। उसके आयतन का अनुमान लगाइए (उसे बेलन मानिए,
        $V = \pi r^2 h$) और प्रश्न 11 के घन से तुलना कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- किसी शरीर में \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की संख्या।}
\begin{enumerate}[resume]
  \item \qty{60}{kg} द्रव्यमान के किसी विद्यार्थी का आयतन लगभग
        \qty{60}{L} है। यदि शरीर केवल \qty{20}{\micro m} की घनाकार
        \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से बना होता, तो उसमें कितनी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} होतीं?
  \item असल में शरीर के आयतन का लगभग एक-तिहाई \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के बीच का तरल और
        दूसरा अकोशिकीय पदार्थ है। अनुमान को सुधारिए।
  \item अकेली लाल रक्त \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} ही लगभग $\num{2.5e13}$ हैं। प्रश्न 15 के
        आयतन की मदद से बताइए कि वे कितना आयतन घेरती हैं। क्या यह लगभग
        \qty{4.5}{L} के रक्त-आयतन से, जिसका 45\% \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} हैं, संगत है?
  \item प्रश्न 17 और 18 को जोड़िए: शरीर में \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की कुल संख्या की
        परिमाण की कोटि बताइए, और यह भी कि ठीक-ठीक गिनती क्यों असंभव है।
  \item परिणाम लिखिए: मोटे तौर पर कितनी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} मिलकर एक मनुष्य बनाती
        हैं, और किस प्रक्रिया से --- किस शुरुआती संख्या से --- वे सब
        बनीं?
\end{enumerate}
\end{problem}
