\chapter{ऋतुओं में जंतु}\label{ch:g3:animals-through-seasons}

अक्तूबर में अबाबीलें तारों पर क़तार बाँधती हैं और एक सुबह ग़ायब हो जाती
हैं। बाड़ के नीचे का साही अपने पत्तों के घोंसले में सिमट जाता है और
वसंत से पहले दिखाई नहीं देगा। इस बीच लोमड़ी हर बर्फ़ में बाहर रहती है।
सर्दी कठिन ऋतु है --- ठंडी, और सबसे बढ़कर भोजन की कमी वाली --- और
\omterm{def:g1:animals-around-us:animal}{जंतुओं} के पास इससे पार पाने के तीन बड़े रास्ते हैं।

\section{सर्दी, कमी की ऋतु}

\begin{proposition}[सर्दी की दोहरी समस्या]\label{prop:g3:animals-through-seasons:problem}
सर्दी \omterm{def:g1:animals-around-us:animal}{जंतुओं} के सामने एक साथ दो समस्याएँ लाती है:
\begin{enumerate}
  \item \emph{ठंड}, जो शरीर की गरमाहट तेज़ी से खींच लेती है ---
        और गरम रहने में \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} लगती है
        (\cref{prop:g3:balanced-diet:jobs});
  \item \emph{भोजन की कमी}: \omterm{def:g1:plants-around-us:plant}{पौधे} बढ़ना बंद कर देते हैं, \omterm{prop:g3:sorting-living-things:groups}{कीट} ग़ायब हो
        जाते हैं, \cref{ch:g3:food-chains} की \omterm{def:g3:food-chains:chain}{आहार शृंखलाओं} की पहली
        कड़ियाँ पतली पड़ जाती हैं --- और यह कमी ऊपर सबके पास पहुँच
        जाती है।
\end{enumerate}
गरमाहट ज़्यादा चाहिए, और उसका दाम चुकाने को भोजन कम: यही सर्दी की वह
समस्या है जिसे हर \omterm{def:g1:animals-around-us:animal}{जंतु} को सुलझाना है।
\end{proposition}
\admitted

\begin{example}[कीट खाने वालों का भोजन कहाँ गया?]\label{ex:g3:animals-through-seasons:insects}
अबाबील उड़ते \omterm{prop:g3:sorting-living-things:groups}{कीटों} पर पलती है, जिन्हें वह उड़ान में ही पकड़ लेती है।
नवंबर में वे होते ही नहीं: बहुत ठंड है। हवाई शिकार के लिए बने \omterm{prop:g3:sorting-living-things:groups}{पक्षी} के
लिए मेनू बदलकर इसे ठीक नहीं किया जा सकता --- अबाबील का भोजन पूरी तरह
ग़ायब हो जाता है।
\end{example}

\section{सर्दी के तीन उत्तर}

\begin{definition}[प्रवास]\label{def:g3:animals-through-seasons:migration}
\emph{प्रवास करना}\index{प्रवास करना} यानी हर साल, ऋतुओं के बदलने के
साथ एक इलाक़े से दूसरे इलाक़े की यात्रा करना --- सर्दी से पहले गरम
देशों की ओर निकल जाना, जहाँ भोजन बचा रहता है, और वसंत में लौट आना।
अबाबील, सारस और क्रौंच \emph{प्रवासी}\index{प्रवास} हैं; कुछ हज़ारों
किलोमीटर चलते हैं, और ठीक उसी घोंसले तक लौटने का रास्ता खोज लेते हैं।
\end{definition}

\begin{definition}[शीतनिद्रा]\label{def:g3:animals-through-seasons:hibernation}
\emph{शीतनिद्रा लेना}\index{शीतनिद्रा} यानी सर्दी किसी ओट में छिपकर
गहरी, ठंडी \omterm{prop:g1:healthy-habits:needs}{नींद} में बिताना: शरीर ठंडा पड़ जाता है, हृदय और साँस लगभग
शून्य तक धीमे हो जाते हैं, और \omterm{def:g1:animals-around-us:animal}{जंतु} पतझड़ में जमा की गई चरबी पर
धीरे-धीरे जीता है। साही, गिलहरी-मूष, मरमोट और चमगादड़ शीतनिद्रा लेते
हैं।
\end{definition}

\begin{definition}[सक्रिय रहना]\label{def:g3:animals-through-seasons:active}
बाक़ी \emph{सक्रिय रहते}\index{सक्रिय रहना (सर्दियों में)} हैं और अपना
साज़-सामान या अपनी आदतें बदल लेते हैं: मोटा \emph{शीतकालीन
आवरण}\index{शीतकालीन आवरण}, पतझड़ में जुटाए \emph{भोजन-भंडार}\index{भोजन-भंडार}
(गिलहरी के गाड़े हुए मेवे), मेनू का बदलना
(\cref{ex:g2:what-animals-eat:winter} की कलचिड़ी), या बस ज़्यादा मेहनत
से खोजना --- लोमड़ी, हिरन, रॉबिन।
\end{definition}

\begin{example}[तीन पड़ोसी, तीन रणनीतियाँ]\label{ex:g3:animals-through-seasons:three}
जनवरी में एक ही बाड़: उसके ऊपर घोंसला बनाने वाली अबाबीलें अफ़्रीका में
गरम धूप के नीचे \omterm{prop:g3:sorting-living-things:groups}{कीट} पकड़ रही हैं; उसके नीचे का साही अपने पत्तों के
घोंसले में ठंडा और धीमा सोता है और अगस्त की चरबी जलाता है; उसके ऊपर
बैठा रॉबिन, अपने फूले हुए शीतकालीन \omterm{def:g1:animals-around-us:covering}{पंखों} में, आख़िरी जामुनों की गश्त
लगाता है। अप्रैल आते ही तीनों फिर उसी बाड़ में व्यस्त होंगे।
\end{example}

\begin{remark}[नींद आलस नहीं, यात्रा सैर नहीं]\label{rem:g3:animals-through-seasons:cost}
हर रणनीति का अपना दाम है। \omterm{def:g3:animals-through-seasons:migration}{प्रवासी} थका देने वाली यात्रा का ख़तरा उठाता
है; \omterm{def:g3:animals-through-seasons:hibernation}{शीतनिद्रा लेने} वाले को पतझड़ में अपनी सर्दी भर की चरबी का
एक-एक ग्राम जुटाना पड़ता है --- नवंबर में बहुत दुबला साही वसंत में
जागेगा ही नहीं; और सक्रिय \omterm{def:g1:animals-around-us:animal}{जंतु} इस दाँव पर रहता है कि सबसे भूखे महीनों
में उसे भोजन मिल जाएगा। सर्दी का दाम पहले चुकाया जाता है, या रास्ते
में, या दिन दर दिन।
\end{remark}

\begin{method}[जंतु की सर्दी वाली रणनीति बूझना]\label{met:g3:animals-through-seasons:guess}
\begin{enumerate}
  \item पूछो कि वह क्या खाता है, और क्या वह भोजन यहाँ सर्दी में
        मिलता है;
  \item भोजन पूरी तरह ग़ायब (उड़ते \omterm{prop:g3:sorting-living-things:groups}{कीट}, फूलों का रस): \omterm{def:g3:animals-through-seasons:migration}{प्रवास} या
        \omterm{def:g3:animals-through-seasons:hibernation}{शीतनिद्रा} की उम्मीद करो;
  \item भोजन मिल सकता हो (\omterm{def:g2:from-seed-to-plant:seed}{बीज}, जामुन, शिकार): सक्रिय सर्दी की उम्मीद
        करो, शायद मोटे आवरण या भंडारों के साथ;
  \item पतझड़ में जाँचो: ख़ूब खाना और घोंसले के लिए गड्ढा खोदना
        \omterm{def:g3:animals-through-seasons:hibernation}{शीतनिद्रा} की ओर इशारा करता है; झुंड बाँधना निकल जाने की ओर।
\end{enumerate}
\end{method}

\begin{example}[वसंत सब कुछ उलट देता है]\label{ex:g3:animals-through-seasons:spring}
मार्च और अप्रैल में क्रम उलट जाता है: \omterm{prop:g3:sorting-living-things:groups}{कीट} लौट आते हैं, \omterm{def:g3:animals-through-seasons:migration}{प्रवासी} अपने
पुराने घोंसलों पर फिर दिखने लगते हैं, \omterm{def:g3:animals-through-seasons:hibernation}{शीतनिद्रा} वाले दुबले और भूखे
जागते हैं, \omterm{def:g3:animals-through-seasons:active}{शीतकालीन आवरण} उतर जाते हैं। ऋतुएँ घूमती हैं, और \omterm{def:g1:animals-around-us:animal}{जंतुओं} का
पंचांग उनके साथ घूमता है --- अगले पतझड़ अबाबीलें फिर तारों पर क़तार
बाँधेंगी।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:1}
सर्दी हर \omterm{def:g1:animals-around-us:animal}{जंतु} के सामने कौन-सी दोहरी समस्या रखती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:2}
सर्दी की तीनों रणनीतियों के नाम बताओ, और हर एक के दो \omterm{def:g1:animals-around-us:animal}{जंतु} भी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:3}
\omterm{def:g3:animals-through-seasons:hibernation}{शीतनिद्रा} में पड़े साही के शरीर के साथ उसकी लंबी \omterm{prop:g1:healthy-habits:needs}{नींद} के दौरान क्या
होता है? वह किस पर जीता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:4}
अबाबील को ठीक-ठीक क्यों जाना पड़ता है --- उसका भोजन क्या है, और यहाँ
सर्दी में उसका क्या होता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:5}
सक्रिय \omterm{def:g1:animals-around-us:animal}{जंतु} के सर्दी पार करने के तीन तरीक़े बताओ, और हर एक के साथ उसका
\omterm{def:g1:animals-around-us:animal}{जंतु} भी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:6}
\cref{rem:g3:animals-through-seasons:cost} के अनुसार हर रणनीति का दाम
क्या है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:7}
पतझड़ में एक \omterm{def:g1:animals-around-us:animal}{जंतु} ख़ूब खाता हुआ और किसी बिल में सूखी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} बिछाता
हुआ दिखता है। वह कौन-सी रणनीति की तैयारी कर रहा है?
\cref{met:g3:animals-through-seasons:guess} के किस चरण से तुम्हें पता
चला?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:8}
करके देखो: किसी एक ठंडे हफ़्ते में बाहर सचमुच दिखने वाले \omterm{prop:g3:sorting-living-things:groups}{पक्षियों} की
सूची बनाओ। क्या अबाबीलें उसमें हैं? तुम्हारे \omterm{prop:g3:sorting-living-things:groups}{पक्षियों} में से किन्होंने
सक्रिय रणनीति चुनी?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:animals-through-seasons:9}

\begin{exercise}[$\star\star$]\label{exo:g3:animals-through-seasons:10}
हलके पतझड़ की वजह से एक नन्हा साही देर तक जागता और शिकार करता रहता है,

\end{exercise}
