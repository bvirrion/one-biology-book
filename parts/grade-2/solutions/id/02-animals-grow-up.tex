\section*{Bab \ref{ch:g2:animals-grow-up} --- Bagaimana Hewan Menjadi Dewasa}

\begin{solution}{exo:g2:animals-grow-up:1}
Dari \omterm{def:g2:animals-grow-up:born}{telur} yang dikeluarkan induknya (anak ayam, katak, siput), atau
\omterm{def:g2:animals-grow-up:born}{lahir hidup} dari tubuh induknya lalu meminum \omterm{def:g2:animals-grow-up:born}{susunya} (anak kucing,
anak sapi, bayi manusia).
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:2}
(a) Dari \omterm{def:g2:animals-grow-up:born}{telur}. (b) \omterm{def:g2:animals-grow-up:born}{Lahir hidup}. (c) Dari \omterm{def:g2:animals-grow-up:born}{telur}. (d) \omterm{def:g2:animals-grow-up:born}{Lahir hidup}.
(e) Dari \omterm{def:g2:animals-grow-up:born}{telur}.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:3}
\omterm{def:g2:animals-grow-up:born}{Susu} induknya. Semua \omterm{def:g1:animals-around-us:animal}{hewan} yang \omterm{def:g2:animals-grow-up:born}{lahir hidup} dalam bab ini --- anak
anjing, anak sapi, anak kelinci, bayi manusia --- juga memulai dengan
\omterm{def:g2:animals-grow-up:born}{susu}.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:4}
\omterm{def:g2:animals-grow-up:metamorphosis}{Metamorfosis} adalah perubahan besar bentuk tubuh yang dilalui sebagian
\omterm{def:g1:animals-around-us:animal}{hewan} muda untuk menjadi dewasa. \omterm{ex:g2:animals-grow-up:frog}{Berudu} yang menjadi katak, dan \omterm{ex:g2:animals-grow-up:butterfly}{ulat}
yang menjadi kupu-kupu.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:5}
\omterm{def:g2:animals-grow-up:born}{Telur}, \omterm{ex:g2:animals-grow-up:frog}{berudu}, \omterm{ex:g2:animals-grow-up:frog}{berudu} berkaki, katak.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:6}
\omterm{def:g2:animals-grow-up:born}{Telur}, \omterm{ex:g2:animals-grow-up:butterfly}{ulat}, \omterm{ex:g2:animals-grow-up:butterfly}{kepompong}, kupu-kupu. Ulatlah yang memakan \omterm{def:g1:plants-around-us:parts}{daun} dan
mengerjakan hampir seluruh pertumbuhannya.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:7}
Anak kuda --- seekor kuda kecil sejak hari pertamanya. Baginya,
menjadi dewasa berarti bertambah besar dan bertambah kuat sambil
mempertahankan bentuk yang sama.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:8}
Agar tetap hangat. Di dalam tiap \omterm{def:g2:animals-grow-up:born}{telur} yang hangat, seekor anak ayam
sedang dibangun, diberi makan oleh kuning telurnya, sampai ia siap
memecahkan \omterm{def:g1:animals-around-us:covering}{cangkang} dan menetas.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:9}
Ia \omterm{ex:g2:animals-grow-up:frog}{berudu} yang berada di tengah \omterm{def:g2:animals-grow-up:metamorphosis}{metamorfosis}: kaki belakangnya sudah
tumbuh, kaki depannya belum muncul. Sebelumnya ia \omterm{ex:g2:animals-grow-up:frog}{berudu} tanpa kaki
(dan sebelum itu, sebutir \omterm{def:g2:animals-grow-up:born}{telur}); berikutnya kaki depannya akan
muncul, ekornya akan menyusut, dan ia akan menjadi katak kecil.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:10}
Tidak ada yang pindah --- \omterm{ex:g2:animals-grow-up:butterfly}{ulat} dan kupu-kupu adalah \omterm{def:g1:animals-around-us:animal}{hewan} yang sama.
Di dalam \omterm{ex:g2:animals-grow-up:butterfly}{kepompongnya}, tubuh \omterm{ex:g2:animals-grow-up:butterfly}{ulatnya} dibangun ulang menjadi tubuh
kupu-kupu: pembangunan ulang itulah arti \omterm{def:g2:animals-grow-up:metamorphosis}{metamorfosis}.
\end{solution}
