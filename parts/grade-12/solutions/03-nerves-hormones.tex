\section*{Chapter \ref{ch:g12:nerves-hormones} --- Nerves and Hormones}

\begin{solution}{exo:g12:nerves-hormones:1}
\omterm{def:g12:nerves-hormones:neuron}{Dendrites} receive synaptic input and convey graded potentials toward the \omterm{def:g12:nerves-hormones:neuron}{cell
body}. The \omterm{def:g12:nerves-hormones:neuron}{axon} conducts \omterm{def:g12:nerves-hormones:ap}{action potentials} over distance. \omterm{def:g12:nerves-hormones:neuron}{Axon} terminals release
\omterm{met:g12:nerves-hormones:synapse}{neurotransmitter} onto target \omterm{def:g10:the-cell:cell}{cells} at \omterm{def:g12:nerves-hormones:synapse}{synapses}.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:2}
Typical neuronal \omterm{def:g12:nerves-hormones:resting}{resting potential} is about \qty{-70}{mV}: the inside of the
membrane is negative relative to the outside. That polarity is maintained by
ion gradients and selective membrane permeability, especially to $\mathrm{K}^{+}$.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:3}
A \omterm{def:g12:nerves-hormones:hormone}{hormone} is a \omterm{def:g12:nerves-hormones:synapse}{chemical} messenger released into blood or extracellular fluid
that acts on \omterm{def:g10:the-cell:cell}{cells} bearing specific receptors. Examples: pancreas--insulin;
thyroid--thyroxine; adrenal--cortisol or adrenaline; posterior
pituitary--ADH. \omterm{def:g10:membranes-enzymes:ES}{Specificity} comes from receptors, not from exclusive delivery
routes alone.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:4}
\omterm{def:g12:nerves-hormones:negative}{Negative feedback}: the response opposes the deviation (\omterm{ex:g12:nerves-hormones:insulin}{insulin} lowering high
glucose toward set point). \omterm{def:g12:nerves-hormones:positive}{Positive feedback}: the response amplifies the
change (oxytocin in labour; $\mathrm{Na}^{+}$-driven upstroke of the \omterm{def:g12:nerves-hormones:ap}{action
potential}). \omterm{def:g12:nerves-hormones:homeostasis}{Homeostasis} relies mainly on negative loops.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:5}
An \omterm{def:g12:nerves-hormones:ap}{action potential} arrives at the terminal $\to$ voltage-gated
$\mathrm{Ca}^{2+}$ channels open $\to$ vesicles fuse and release transmitter
$\to$ transmitter binds postsynaptic receptors $\to$ postsynaptic potential
$\to$ transmitter is cleared by \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}, reuptake or \omterm{def:g10:membranes-enzymes:enzyme}{enzymes}.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:6}
Above threshold a full regenerative spike occurs; below threshold the spike
fails (all-or-none for the regenerative event). Stimulus strength is encoded
mainly by firing frequency and by how many \omterm{def:g12:nerves-hormones:neuron}{axons} are recruited, not by making
individual spikes taller.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:7}
\omterm{def:g12:nerves-hormones:glia}{Myelin} increases conduction speed and energetic efficiency by insulating
internodes. \omterm{def:g12:nerves-hormones:prop}{Nodes of Ranvier} are unmyelinated gaps where the \omterm{def:g12:nerves-hormones:ap}{action potential}
is regenerated; conduction jumps from \omterm{def:g12:phylogeny-macroevolution:node}{node} to \omterm{def:g12:phylogeny-macroevolution:node}{node} (\omterm{def:g12:nerves-hormones:prop}{saltatory conduction}).
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:8}
High blood glucose is sensed by pancreatic $\beta$ \omterm{def:g10:the-cell:cell}{cells}, which release
\omterm{ex:g12:nerves-hormones:insulin}{insulin}. \omterm{ex:g12:nerves-hormones:insulin}{Insulin} increases glucose uptake and storage in liver, muscle and \omterm{def:g10:molecules-of-life:lipids}{fat},
so glucose falls toward set point and \omterm{ex:g12:nerves-hormones:insulin}{insulin} secretion declines --- classic
\omterm{def:g12:nerves-hormones:negative}{negative feedback}.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:9}
\omterm{def:g10:molecules-of-life:lipids}{Steroid} \omterm{def:g12:nerves-hormones:hormone}{hormones} often act via intracellular receptors that change \omterm{def:g11:gene-expression:gene}{gene}
\omterm{def:g11:gene-expression:transcription}{transcription}, so effects appear over minutes to hours as new \omterm{def:g10:molecules-of-life:proteins}{proteins} are
made. Peptide \omterm{def:g12:nerves-hormones:hormone}{hormones} typically bind membrane receptors and second-messenger
cascades, acting in seconds. Speed and mechanism track \omterm{def:g12:nerves-hormones:synapse}{chemical} \omterm{def:g10:biodiversity-classification:hierarchy}{class}.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:10}
Sympathetic activation raises heart rate and force, dilates bronchi, and
redirects blood flow toward muscle. \omterm{def:g12:nerves-hormones:glands}{Adrenal} adrenaline and noradrenaline
reinforce fuel mobilisation and cardiovascular readiness for fight-or-flight
work.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:11}
If voltage-gated $\mathrm{Na}^{+}$ channels are blocked, \omterm{def:g12:nerves-hormones:ap}{action potentials}
fail because regenerative $\mathrm{Na}^{+}$ influx cannot occur. Terminals do
not depolarise properly, $\mathrm{Ca}^{2+}$-dependent release drops, and
synaptic transmission collapses along affected \omterm{def:g12:nerves-hormones:neuron}{axons}.
\end{solution}

\begin{solution}{exo:g12:nerves-hormones:12}
Neuromuscular \omterm{def:g12:nerves-hormones:ap}{action potentials} act on a millisecond scale at a single
junction and stop when firing stops. Circulating adrenaline acts over seconds
to minutes on many tissues and is cleared more slowly --- broader, longer
modulation than a single synaptic burst.
\end{solution}

\begin{solution}{pb:g12:nerves-hormones:1}
\textbf{Part I.} (1)~Visual/tactile afferents $\to$ spinal/brain circuits $\to$
motor \omterm{def:g12:nerves-hormones:neuron}{neurons} to leg muscles. (2)~\omterm{def:g12:nerves-hormones:synapse}{Synapse} \omterm{def:g10:molecules-of-life:proteins}{sequence} as in the chapter method
with ACh at the neuromuscular junction. (3)~Standard spike sketch (rest,
threshold, rising and falling phases). (4)~Electrical conduction is fast;
endocrine signals require synthesis, circulation and clearance.

\textbf{Part II.} (5)~$\mathrm{Na}^{+}$ in during the upstroke; $\mathrm{K}^{+}$
out during repolarisation. (6)~Prolonged depolarisation or failed
repolarisation if $\mathrm{K}^{+}$ efflux is blocked. (7)~Saltatory jump
between \omterm{def:g12:phylogeny-macroevolution:node}{nodes} under \omterm{def:g12:nerves-hormones:glia}{myelin}.

\textbf{Part III.} (8)~Adrenaline for rapid cardiorespiratory readiness;
cortisol for slower metabolic stress responses. (9)~Parasympathetic tone
returns; \omterm{def:g12:nerves-hormones:hormone}{hormones} clear; set points reassert. (10)~Full glucose--insulin loop
as in the chapter. (11)~Childbirth, clotting, or spike upstroke --- \omterm{def:g12:nerves-hormones:positive}{positive
feedback} ends when the stimulus stops or the system saturates.
(12)~Nervous and \omterm{def:g12:nerves-hormones:hormone}{endocrine systems} are \omterm{prop:g11:dna-replication:base-pairs}{complementary} in speed versus breadth;
the \omterm{def:g12:nerves-hormones:glands}{hypothalamus} links both.
\end{solution}
