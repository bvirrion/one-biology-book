\section*{Chapter \ref{ch:g4:cells} --- A First Look at Cells}

\begin{solution}{exo:g4:cells:1}
A \omterm{def:g4:cells:cell}{cell} is the smallest unit of a \omterm{def:g1:living-nonliving:living}{living thing} that can carry out life's
jobs (taking in materials, using \omterm{def:g2:food-energy:energy}{energy}, and often making more \omterm{def:g4:cells:cell}{cells}).
\end{solution}

\begin{solution}{exo:g4:cells:2}
Most \omterm{def:g4:cells:cell}{cells} are too small to see with the naked \omterm{def:g1:senses:five}{eye}, so a \omterm{prop:g4:cells:building}{microscope} is
needed to magnify them.
\end{solution}

\begin{solution}{exo:g4:cells:3}
\omterm{def:g4:cells:parts}{Cell membrane}, \omterm{def:g4:cells:parts}{cytoplasm} and \omterm{def:g4:cells:parts}{nucleus} (accept clear equivalents).
\end{solution}

\begin{solution}{exo:g4:cells:4}
\omterm{def:g4:cells:plant}{Cell wall} --- supports and protects the \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g4:cells:cell}{cell}. \omterm{def:g4:cells:plant}{Chloroplasts} ---
capture light to help make \omterm{def:g2:food-energy:food}{food}. (A large central vacuole is also
acceptable if mentioned in class.)
\end{solution}

\begin{solution}{exo:g4:cells:5}
Rocks are non-living. \omterm{def:g4:cells:cell}{Cells} are living units (or parts of \omterm{def:g1:living-nonliving:living}{living
things}). Being solid does not mean something is made of \omterm{def:g4:cells:cell}{cells}.
\end{solution}
