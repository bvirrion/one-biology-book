\section*{अध्याय \ref{ch:g4:classifying-animals} --- जंतुओं का वर्गीकरण}

\begin{solution}{exo:g4:classifying-animals:1}
\omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी} का भीतरी \omterm{def:g3:bones-and-muscles:skeleton}{कंकाल} \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} के चारों ओर बना होता है (ट्राउट, बिल्ली);
\omterm{def:g4:classifying-animals:vertebrate}{अकशेरुकी} के कोई \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} नहीं होती (घोंघा, मधुमक्खी, केंचुआ, जेलीफ़िश)।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:2}
\omterm{prop:g3:sorting-living-things:groups}{मछलियाँ}: \omterm{def:g1:animals-around-us:covering}{शल्क}, मीन-पंख, पानी में साँस। \omterm{prop:g3:sorting-living-things:groups}{उभयचर}: नम खुली \omterm{def:g1:the-five-senses:sense}{त्वचा}, पानी में
\omterm{def:g2:animals-grow-up:born}{अंडे}, \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} वाला बचपन। \omterm{prop:g4:classifying-animals:classes}{सरीसृप}: सूखी शल्कदार \omterm{def:g1:the-five-senses:sense}{त्वचा}, ज़्यादातर ज़मीन पर
खोलदार \omterm{def:g2:animals-grow-up:born}{अंडे}। \omterm{prop:g3:sorting-living-things:groups}{पक्षी}: \omterm{def:g1:animals-around-us:covering}{पंख}, चोंच, खोलदार \omterm{def:g2:animals-grow-up:born}{अंडे}। \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी}: \omterm{def:g1:animals-around-us:covering}{रोम} या बाल,
बच्चे \omterm{def:g2:animals-grow-up:born}{दूध} पर पलते हैं।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:3}
सारडीन: मछली। मगरमच्छ: \omterm{prop:g4:classifying-animals:classes}{सरीसृप}। न्यूट: \omterm{prop:g3:sorting-living-things:groups}{उभयचर}। शुतुरमुर्ग़: \omterm{prop:g3:sorting-living-things:groups}{पक्षी}।
ह्वेल: \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी}। केंचुआ: \omterm{def:g4:classifying-animals:vertebrate}{अकशेरुकी} --- \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} नहीं, इसलिए कोई \omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी}
वर्ग भी नहीं।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:4}
बिल्ली, \omterm{prop:g3:sorting-living-things:groups}{स्तनधारियों} के भीतर, \omterm{def:g4:classifying-animals:vertebrate}{कशेरुकियों} के भीतर, \omterm{def:g1:animals-around-us:animal}{जंतुओं} के भीतर।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:5}
\omterm{def:g1:the-five-senses:sense}{त्वचा} --- छिपकली की सूखी और शल्कदार, न्यूट की नम और खुली --- और \omterm{def:g2:animals-grow-up:born}{अंडे}:
छिपकली के खोलदार और ज़मीन पर दिए हुए, न्यूट के खुले और पानी में दिए
हुए।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:6}
\omterm{def:g4:classifying-animals:vertebrate}{अकशेरुकी}। जैसे: \omterm{prop:g3:sorting-living-things:groups}{कीट}, मकड़ियाँ, घोंघे, केंचुए (और जेलीफ़िश)।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:7}
पेंग्विन: \omterm{def:g1:animals-around-us:covering}{पंख}? हाँ --- \omterm{prop:g3:sorting-living-things:groups}{पक्षी}; पहले ही सवाल पर रुक जाओ। डॉल्फ़िन: \omterm{def:g1:animals-around-us:covering}{पंख}?
नहीं। \omterm{def:g1:animals-around-us:covering}{रोम} या \omterm{def:g2:animals-grow-up:born}{दूध}? वह अपने बच्चे को \omterm{def:g2:animals-grow-up:born}{दूध} पिलाती है --- \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी}; वहीं रुक
जाओ।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:8}
जैसे: \omterm{def:g1:animals-around-us:covering}{पंख}, एक चोंच, डैने, खोलदार \omterm{def:g2:animals-grow-up:born}{अंडे} --- और एक \omterm{def:g4:classifying-animals:vertebrate}{रीढ़}, क्योंकि \omterm{prop:g3:sorting-living-things:groups}{पक्षी}
\omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी} हैं।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:9}
सवाल ``बच्चों के लिए \omterm{def:g2:animals-grow-up:born}{दूध}'' पर रुक जाते हैं: ह्वेल अपने बच्चे को \omterm{def:g2:animals-grow-up:born}{दूध}
पिलाती है, इसलिए वह \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} है, बाल हों या न हों। ``समुद्र में रहती
है'' \omterm{def:g2:where-animals-live:habitat}{आवास} है, शरीर की ख़ासियत नहीं --- और \omterm{def:g4:classifying-animals:classification}{वर्गीकरण} इसी पर बनाया गया था
कि \omterm{def:g1:animals-around-us:animal}{जंतु} के पास \emph{क्या है}, ठीक इसलिए कि आदतें और घर अजनबी भी बाँट
सकते हैं: शार्क और डॉल्फ़िन का पता एक है, वर्ग नहीं।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:10}
\omterm{def:g4:classifying-animals:vertebrate}{रीढ़}। साँप की \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} सैकड़ों छोटी \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियों} की होती है --- यानी \omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी},
और \omterm{prop:g4:classifying-animals:classes}{सरीसृप} की सूखी शल्कदार \omterm{def:g1:the-five-senses:sense}{त्वचा} के साथ; केंचुए की कोई \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} ही नहीं ---
यानी \omterm{def:g4:classifying-animals:vertebrate}{अकशेरुकी}। \omterm{def:g1:my-body:parts}{टाँगों} का न होना बस एक ख़ासियत का न होना है; \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} वाला
सवाल पहले पूछा जाता है।
\end{solution}

\begin{solution}{exo:g4:classifying-animals:11}
\omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} जीते: \omterm{def:g1:animals-around-us:covering}{रोम} और बच्चों के लिए \omterm{def:g2:animals-grow-up:born}{दूध} \omterm{prop:g3:sorting-living-things:groups}{स्तनधारियों} का गट्ठर हैं, और
प्लैटिपस के पास दोनों हैं --- \omterm{def:g2:animals-grow-up:born}{अंडा} उस गट्ठर से बाहर की अकेली ख़ासियत
है। \omterm{def:g4:classifying-animals:classification}{वर्गीकरण} किसी एक ख़ासियत को फ़ैसला करने देने के बजाय पूरे गट्ठर को
तौलता है; एक बात में अनोखा \omterm{def:g1:animals-around-us:animal}{जंतु} भी वहीं बैठता है जहाँ उसकी ज़्यादातर
ख़ासियतें इशारा करती हैं। (प्लैटिपस इतना अनोखा है कि \omterm{def:g2:animals-grow-up:born}{अंडे} देने वाले
\omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} अपना अलग छोटा उपसमूह बनाते हैं।)
\end{solution}
