\chapter{Matter Cycles in Ecosystems}\label{ch:g9:cycles}

\omterm{def:g2:food-energy:energy}{Energy} flows through \omterm{def:g7:eco-energy:ecosystem}{ecosystems} and is lost as heat; matter cycles.
\omterm{def:g6:nutrition:groups}{Water}, carbon and nitrogen move between \omterm{def:g1:living-nonliving:living}{living things} and the non-living
world in continuous loops.

\section{Why cycles matter}

\begin{proposition}[Energy vs matter]\label{prop:g9:cycles:vs}
\omterm{def:g2:food-energy:energy}{Energy} enters mainly as sunlight and \omterm{def:g7:tissues:plant-organs}{leaves} as heat --- it does not cycle
fully. Chemical elements are reused: \omterm{def:g7:eco-energy:roles}{decomposers} return materials to soil
and \omterm{def:g4:plants-need:needs}{air} for \omterm{def:g7:eco-energy:roles}{producers}.
\end{proposition}
\admitted

\begin{definition}[Biogeochemical cycle]\label{def:g9:cycles:bio}
A \emph{biogeochemical cycle}\index{biogeochemical cycle} describes how an
element or compound moves through living \omterm{def:g4:ecosystems:levels}{organisms} and physical
reservoirs (\omterm{def:g4:plants-need:needs}{air}, \omterm{def:g4:plants-need:needs}{water}, rock, soil).
\end{definition}

\section{Water cycle}

\begin{definition}[Water cycle]\label{def:g9:cycles:water}
The \emph{water cycle}\index{water cycle} includes evaporation,
\omterm{def:g9:plant-phys:trans}{transpiration} from \omterm{def:g1:plants-animals:plant}{plants}, condensation, precipitation and runoff or
infiltration to groundwater. \omterm{def:g4:ecosystems:levels}{Organisms} are temporary stores of \omterm{def:g4:plants-need:needs}{water}.
\end{definition}

\section{Carbon cycle}

\begin{definition}[Carbon cycle]\label{def:g9:cycles:carbon}
Carbon moves as $\mathrm{CO_2}$ in \omterm{def:g4:plants-need:needs}{air}, as organic molecules in \omterm{def:g4:ecosystems:levels}{organisms},
and in \omterm{def:g8:evolution:fossil}{fossil} stores and rocks. \omterm{def:g7:photo-resp:photo}{Photosynthesis} pulls $\mathrm{CO_2}$ into
\omterm{def:g7:eco-energy:chain}{food webs}; respiration and combustion release it.
\end{definition}

\begin{proposition}[Human acceleration]\label{prop:g9:cycles:fossil}
Burning \omterm{def:g8:evolution:fossil}{fossil} fuels and deforestation raise atmospheric $\mathrm{CO_2}$
faster than natural sinks remove it, enhancing the \omterm{def:g7:impact:ghg}{greenhouse effect}.
\end{proposition}
\admitted

\section{Nitrogen cycle (outline)}

\begin{definition}[Nitrogen cycle ideas]\label{def:g9:cycles:nitrogen}
Most \omterm{def:g4:ecosystems:levels}{organisms} cannot use $\mathrm{N_2}$ gas directly.
\emph{Nitrogen-fixing}\index{nitrogen fixation} \omterm{def:g4:microbes:def}{microbes} convert it to
forms \omterm{def:g1:plants-animals:plant}{plants} can absorb. \omterm{def:g7:eco-energy:roles}{Consumers} get nitrogen in \omterm{def:g6:nutrition:groups}{proteins}; \omterm{def:g7:eco-energy:roles}{decomposers}
and other \omterm{def:g4:microbes:def}{bacteria} return nitrogen compounds to soil and, via denitrification,
to the \omterm{def:g4:plants-need:needs}{air}.
\end{definition}

\begin{example}[Legumes]\label{ex:g9:cycles:legume}
Beans and clover host nitrogen-fixing \omterm{def:g4:microbes:def}{bacteria} in \omterm{def:g7:plant-structure:organs}{root} nodules --- a
\omterm{def:g8:populations:interact}{mutualism} that enriches soil nitrogen.
\end{example}

\begin{method}[Tracking an atom]\label{met:g9:cycles:track}
\begin{enumerate}
  \item Pick C, N or $\mathrm{H_2O}$.
  \item Name at least three reservoirs or \omterm{def:g4:ecosystems:levels}{organisms} it visits.
  \item Name the processes that move it (\omterm{def:g7:photo-resp:photo}{photosynthesis}, eating,
        combustion,\ldots).
\end{enumerate}
\end{method}
