\chapter{Los dientes y comer bien}\label{ch:g2:teeth-and-eating-well}

Muerde una manzana y mira la marca: una hilera curva y ordenada de
huellecitas. Eso lo han hecho tus dientes --- una caja de herramientas
completa que corta, desgarra y tritura todas las comidas antes de que
las tragues. Y, como toda buena caja de herramientas, merece cuidados
y buen material con el que trabajar.

\section{Los dientes: una caja de herramientas en la boca}

\begin{definition}[Las tres clases de dientes]\label{def:g2:teeth-and-eating-well:kinds}
Los dientes humanos son de tres clases, cada una con su tarea:
\begin{enumerate}
  \item los \emph{incisivos}\index{incisivo}, planos y afilados,
        delante, \emph{cortan} --- son los que entran en la manzana;
  \item los \emph{caninos}\index{canino}, puntiagudos, en las
        esquinas, \emph{desgarran} --- agarran y tiran de la comida
        más dura;
  \item los \emph{molares}\index{molar}, anchos y con relieves, al
        fondo, \emph{trituran} --- deshacen la comida en una pasta
        lista para tragar.
\end{enumerate}
\end{definition}

\begin{example}[Un mordisco de manzana, tres herramientas]\label{ex:g2:teeth-and-eating-well:apple}
Comerse un trozo de manzana usa la caja entera y por orden: los
\omterm{def:g2:teeth-and-eating-well:kinds}{incisivos} cortan el bocado, los \omterm{def:g2:teeth-and-eating-well:kinds}{caninos} ayudan a arrancarlo y la
lengua lo pasa hacia atrás a los \omterm{def:g2:teeth-and-eating-well:kinds}{molares}, que lo trituran. Solo
entonces tragas.
\end{example}

\begin{remark}[Acuérdate de los animales]\label{rem:g2:teeth-and-eating-well:animals}
Tu caja de herramientas mixta cuenta una historia: cortadores planos y
trituradores como los de un \omterm{def:g2:what-animals-eat:kinds}{herbívoro}, \omterm{def:g2:teeth-and-eating-well:kinds}{caninos} puntiagudos como los de
un \omterm{def:g2:what-animals-eat:kinds}{carnívoro}. Es una boca de \omterm{def:g2:what-animals-eat:kinds}{omnívoro} --- las personas están equipadas
para las \omterm{def:g1:plants-around-us:plant}{plantas} \emph{y} para la carne, justo como nos clasificó el
\cref{ch:g2:what-animals-eat}.
\end{remark}

\section{Dientes de leche, dientes definitivos}

\begin{proposition}[Dos juegos en una vida]\label{prop:g2:teeth-and-eating-well:sets}
Las personas tenemos dos juegos de dientes. Los $20$ \emph{dientes de
leche}\index{dientes de leche} salen en los primeros años de vida.
Hacia los seis años se aflojan y se caen uno a uno, empujados desde
abajo por los \emph{dientes definitivos}\index{dientes definitivos}
--- el juego final, hasta $32$, que tiene que durar el resto de la vida.
\end{proposition}
\admitted

\begin{example}[El incisivo que se mueve]\label{ex:g2:teeth-and-eating-well:wobbly}
Un diente de delante empieza a moverse justo a la edad en que los
niños empiezan el colegio. Debajo, un \omterm{def:g2:teeth-and-eating-well:kinds}{incisivo} definitivo lleva un
tiempo creciendo en silencio y deshaciendo la \omterm{def:g1:plants-around-us:parts}{raíz} del diente de
\omterm{def:g2:animals-grow-up:born}{leche}, hasta que el dientecillo se suelta. El hueco no es un accidente
--- es un relevo.
\end{example}

\begin{remark}[No hay un tercer juego]\label{rem:g2:teeth-and-eating-well:nothird}
Después del juego definitivo no saldrá ningún diente nuevo. Un diente
de \omterm{def:g2:animals-grow-up:born}{leche} perdido se sustituye; un diente definitivo perdido está
perdido. Esa es la verdadera razón de que las personas mayores insistan
en el cepillo: la segunda caja de herramientas tiene que durar setenta
años o más.
\end{remark}

\section{Comer bien}

\begin{proposition}[Un plato equilibrado]\label{prop:g2:teeth-and-eating-well:balanced}
Para crecer y estar sano, el cuerpo necesita comida de cada familia,
todos los días:
\begin{itemize}
  \item fruta y verdura --- para las vitaminas, en cada comida;
  \item pan, cereales, patatas --- para la energía que dura;
  \item \omterm{def:g2:animals-grow-up:born}{leche}, yogur, queso --- para construir huesos y dientes;
  \item carne, pescado o \omterm{def:g2:animals-grow-up:born}{huevos} --- para construir el cuerpo; con una
        o dos veces al día basta;
  \item agua --- la única bebida que el cuerpo necesita de verdad;
  \item caprichos dulces y grasos --- por gusto, en poca cantidad y no
        en cada comida.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Dos meriendas]\label{ex:g2:teeth-and-eating-well:snack}
Merienda A: una manzana, un trozo de pan, agua. Merienda B: caramelos
y un refresco. La merienda B sabe a azúcar y a poco más --- y el
azúcar es justo lo que les encanta a los \omterm{def:g1:healthy-habits:germs}{microbios} del
\cref{ch:g1:healthy-habits}. La merienda A alimenta el cuerpo y les da
trabajo honrado a los dientes. Adivina cuál prepara la dentista para
sus propios hijos.
\end{example}

\begin{method}[Repasar el menú de un día]\label{met:g2:teeth-and-eating-well:check}
Coge las comidas de un día y comprueba:
\begin{enumerate}
  \item ¿fruta o verdura en cada comida?
  \item ¿algo de las familias que construyen --- lácteos y carne,
        pescado o \omterm{def:g2:animals-grow-up:born}{huevos}?
  \item ¿agua como bebida de todos los días?
  \item ¿los caprichos pequeños y de vez en cuando?
\end{enumerate}
Si falta una familia, la comida siguiente es la ocasión de volver a
invitarla.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:1}
Nombra las tres clases de dientes y la tarea de cada una.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:2}
¿Qué dientes están delante de la boca y cuáles al fondo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:3}
Sigue un mordisco de zanahoria por la boca: ¿qué hace cada diente y en
qué orden?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:4}
¿Cuántos \omterm{prop:g2:teeth-and-eating-well:sets}{dientes de leche} hay? ¿Qué les pasa a partir de los seis años
más o menos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:5}
¿Por qué hay que cuidar los \omterm{prop:g2:teeth-and-eating-well:sets}{dientes definitivos} aún más que los de \omterm{def:g2:animals-grow-up:born}{leche}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:6}
Nombra las familias de alimentos que debería incluir un día sano. ¿Qué
bebida necesita de verdad el cuerpo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:7}
Compara las dos meriendas del
\cref{ex:g2:teeth-and-eating-well:snack}: ¿qué le da al cuerpo la
merienda A que no le da la B?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:8}
Pruébalo: mírate los dientes en un espejo. Encuentra un \omterm{def:g2:teeth-and-eating-well:kinds}{incisivo}, un
\omterm{def:g2:teeth-and-eating-well:kinds}{canino} y un \omterm{def:g2:teeth-and-eating-well:kinds}{molar}, y compara sus formas con las de la figura.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:teeth-and-eating-well:9}
Con la \cref{rem:g2:teeth-and-eating-well:animals}, explica qué dice
la forma de los dientes humanos sobre el menú humano --- y a qué
\omterm{def:g1:animals-around-us:animal}{animales} del \cref{ch:g2:what-animals-eat} nos parecemos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:teeth-and-eating-well:10}
El día de Lila: desayuno --- cereales con \omterm{def:g2:animals-grow-up:born}{leche}; comida --- pasta y
caramelos; merienda --- refresco y galletas; cena --- pizza. Pásale el
\cref{met:g2:teeth-and-eating-well:check}: ¿qué falta y qué aparece
demasiado a menudo?
\end{exercise}
