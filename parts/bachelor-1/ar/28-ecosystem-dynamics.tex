\chapter{ديناميكا النظم البيئية}\label{ch:b1:ecosystem-dynamics}

حيث ما يزال نهر جليدي في شمال المحيط الهادئ يتراجع منذ قرنين، تكون
المسيرة من مقدمته نزولًا في المضيق مسيرةً عبر الزمن. فالصخر الذي
انكشف العام الماضي عارٍ؛ وعلى بعد كيلومتر يلبس الطحالب وفرشًا
مثبِّتة للنتروجين؛ وأبعد، أدغال الأرلة؛ ثم التنوب، ثم غابة من
التنوب والتسوغا عمرها مئتا سنة وتحتها متر من التربة. ولا شيء هنا
ساكن: \omterm{def:b1:ecosystem-organization:ecosystem}{فالنظم البيئية} تتغير بعد كل \omterm{def:b1:ecosystem-dynamics:disturbance}{اضطراب}، على مسار
قابل للتنبؤ جزئيًا، وتتغير معها قدرتها على حفظ مغذّياتها وإعادة
تدويرها. ويتتبع هذا الفصل تلك التغيرات ---
\omterm{def:b1:ecosystem-dynamics:succession}{التعاقب}، \omterm{def:b1:ecosystem-dynamics:disturbance}{والاضطراب} والتعافي، وحفظ المغذّيات،
وتجمّع المجتمعات على الجزر --- والتجارب التي
حوّلتها من سرديات إلى أرقام.

\section{التعاقب}

\begin{definition}[التعاقب البيئي]\label{def:b1:ecosystem-dynamics:succession}
\index{تعاقب}\index{تعاقب أولي}\index{تعاقب ثانوي}\index{ذروة}
\emph{التعاقب} هو التغير الموجَّه لمجتمع مع الزمن
بعد \omterm{def:b1:ecosystem-dynamics:disturbance}{اضطراب}، إذ تحلّ مجموعة أنواع محل أخرى حتى يدوم مجتمع
مستقر نسبيًا هو \emph{الذروة}. و\emph{التعاقب
الأولي} يبدأ على ركيزة جديدة بلا تربة ولا \omterm{def:b1:organism-environment:organism}{كائنات}
حية: صخر تركه نهر جليدي، أو تدفق حمم، أو كثيب، أو جزيرة جديدة.
و\emph{التعاقب الثانوي} يبدأ حيث دُمّر مجتمع
لكن بقيت التربة وبنك بذورها: حقل مهجور،
أو غابة محروقة أو مقطوعة، أو مرج مغمور. والوافدون الأوائل،
وهم \emph{الأنواع الرائدة}، ناشرون يحتملون الأرض العارية؛
والوافدون المتأخرون منافسون يحتملون الظل ويحتاجون إلى تربة.
\end{definition}

\begin{proposition}[آليات الإحلال]\label{prop:b1:ecosystem-dynamics:mechanisms}
\index{تيسير}
كل مرحلة من \omterm{def:b1:ecosystem-dynamics:succession}{التعاقب} قد تجلب التالية بثلاث طرائق.
\emph{التيسير}: فالأنواع الأبكر تجعل الموقع صالحًا
للأنواع اللاحقة --- ببناء التربة، \omterm{def:b1:plant-water-minerals:nodules}{وتثبيت النتروجين}، وإلقاء الظل،
وحفظ الرطوبة --- ثم تزيحها اللاحقة. و\emph{التثبيط}:
فالشاغلون الأوائل يمسكون الموقع ضد كل قادم حتى يموتوا
أو يتضرروا، ويكون ما يحلّ محلهم ما يصل عندئذ.
و\emph{التحمل}: فالأنواع اللاحقة تصل في البداية أيضًا، وتنمو
ببطء وتبقى ببساطة بعد الرواد، الذين لا ينفعها حضورهم
ولا يعوقها. وأكثر التعاقبات يظهر الثلاثة في مراحل
مختلفة؛ ويسود التيسير التعاقبَ الأولي، إذ تكون المهمة
الأولى صنع تربة.
\end{proposition}
\begin{proof}[شواهد]
في خليج غليشير في ألاسكا، حيث تراجع الجليد نحو
\qty{100}{km} منذ 1750، يُعرف عمر كل موقع من الخرائط
وحلقات الأشجار، ويمكن زيارة المراحل بالترتيب: ركام جليدي عارٍ،
وقشرة من البكتيريا الزرقاء والطحالب، وفرش من
\emph{Dryas} المثبِّتة للنتروجين، وأدغال الأرلة (مثبِّتة للنتروجين، تضيف
\qty{50}{kg} من النتروجين في الهكتار في السنة)، ثم تنوب سيتكا،
ثم غابة تنوب وتسوغا بعد قرنين. ويرتفع نتروجين التربة
من لا شيء تقريبًا إلى \qty{300}{kg/ha} تحت الأرلة وتستقر
أشجار التنوب بعدها فقط؛ ويهبط pH من 8 إلى دون 5 مع
تراكم \omterm{def:b1:ecosystem-dynamics:decomposition}{النثار}. وبادرات التنوب المزروعة على الركام الفتي
لا تنمو إلا إذا سُمّدت بالنتروجين: فالأرلة مُيسِّرات.
\end{proof}

\begin{example}[حقل قديم]\label{ex:b1:ecosystem-dynamics:oldfield}
حقل مهجور في شرق الولايات المتحدة تستعمره في السنة الأولى
أعشاب حولية (النجيل الزاحف والأمبروسيا) من بنك البذور
والريح؛ وفي السنة الثالثة تسود أعشاب معمّرة وعصا الذهب؛ وفي
العاشرة، جنبات وبادرات صنوبر؛ وفي الخامسة والعشرين غابة
صنوبر، لا تستطيع بادرات الصنوبر نفسها أن تنمو في ظلها لكن
البلوط والهيكوري يستطيعان؛ وفي المئة غابة بلوط وهيكوري،
بادراتها تحتمل ظلها فتحلّ محل نفسها. وكل مرحلة
تزيحها أنواع كانت أسوأ منها نشرًا وأفضل
منافسةً؛ والتربة كانت موجودة من البداية، والتتابع كله
يستغرق عمر إنسان لا آلاف السنين التي يستغرقها \omterm{def:b1:ecosystem-dynamics:succession}{تعاقب أولي}.
\end{example}

\section{الاضطراب والاستقرار والتنوع}

\begin{definition}[الاضطراب والمقاومة والمرونة]\label{def:b1:ecosystem-dynamics:disturbance}
\index{اضطراب}\index{مرونة}\index{مقاومة}
\emph{الاضطراب} حدث --- حريق، أو عاصفة، أو فيضان، أو جفاف،
أو رعي، أو قطع، أو وباء --- يزيل الكتلة الحيوية ويفتح
مكانًا. و\emph{مقاومة} \omterm{def:b1:ecosystem-organization:ecosystem}{النظام البيئي} هي قدرته على البقاء
بلا تغير خلال اضطراب؛ و\emph{مرونته} هي السرعة
التي يعود بها إلى حاله السابقة بعده. فالغابة
مقاومة لصيف جاف وبطيئة التعافي من حريق؛
والمرج على العكس. والمجتمعات تشكّلها أنظمة
اضطرابها: تواترها وشدتها وامتدادها. وكثير من
المجتمعات فسيفساء من رقع في مراحل مختلفة من
\omterm{def:b1:ecosystem-dynamics:succession}{التعاقب} منذ آخر اضطراب لها، وتنوعها هو
المجموع على الفسيفساء.
\end{definition}

\begin{proposition}[فرضية الاضطراب المتوسط]\label{prop:b1:ecosystem-dynamics:idh}
\index{فرضية الاضطراب المتوسط}
يكون التنوع أعلى ما يكون عند تواترات وشدات متوسطة من
\omterm{def:b1:ecosystem-dynamics:disturbance}{الاضطراب}. فحيث تندر الاضطرابات يقصي أفضل المنافسين
الباقين؛ وحيث تكثر لا ينجو إلا أسرع
المستعمِرين؛ وبين الاثنين يتعايش الرواد والمنافسون،
كل واحد في الرقع التي تناسبه.
\end{proposition}
\begin{proof}[شواهد]
على الشواطئ الصخرية والشعاب المرجانية، يكون عدد أنواع الطحالب أو
المرجان على الجلاميد وأقسام الشعاب أكبر ما يكون على التي
تقلبها العواصف أو تكسرها بمعدلات متوسطة: فالجلاميد الصغيرة
(التي تُقلب كثيرًا) تحمل بضعة طحالب رائدة، والكبيرة (التي نادرًا ما تُقلب)
أحادية زراعية من أفضل المنافسين، والمتوسطة أكثر الأنواع.
ويظهر السنام نفسه في مجاري الجداول في مقابل تواتر الفيضان وفي
المروج في مقابل شدة الرعي.
\end{proof}

\begin{remark}[هل يجعل التنوع النظام البيئي مستقرًا؟]\label{rem:b1:ecosystem-dynamics:diversity-stability}
قطع المروج المزروعة بأنواع أكثر تحفظ كتلتها الحيوية أثبت
عبر الجفاف من القطع القليلة الأنواع، لأنه مع أنواع أكثر
يكون بعضها دائمًا مناسبًا لطقس السنة --- وهو أثر
تأميني. لكن المجتمعات المتنوعة ليست دائمًا أشد مقاومة،
\omterm{def:b1:populations:population}{والجمهرات} المفردة فيها تتذبذب كما تتذبذب في أي مكان:
فالتنوع يثبّت وظائف \omterm{def:b1:ecosystem-organization:ecosystem}{النظام البيئي} (الإنتاج وحفظ
المغذّيات) أكثر مما يثبّت تركيبه.
\end{remark}

\section{التحلل وحفظ المغذّيات}

\begin{definition}[التحلل وميزانية المغذّيات]\label{def:b1:ecosystem-dynamics:decomposition}
\index{تحلل}\index{ميزانية مغذّيات}\index{نثار}
\emph{التحلل} هو تفكيك المادة العضوية الميتة
(\emph{النثار}، والخشب الميت، والجثث، والروث) \omterm{def:b1:ecosystem-organization:trophic}{بآكلات الفتات}
\omterm{def:b1:ecosystem-organization:trophic}{والمحلِّلات} إلى ثاني أكسيد الكربون وماء وأيونات معدنية ---
أي \emph{التمعدن}. وسرعته تتوقف على الحرارة والرطوبة
والأكسجين ونوعية النثار (محتوى النتروجين واللغنين): \omterm{def:b1:flowering-plant-organization:organs}{فالورقة}
تختفي في سنة في غابة دافئة رطبة، وفي عقد تحت الصنوبريات
على منحدر بارد، وفي آلاف السنين في مستنقع. والأيونات المعدنية
المحررة تلتقطها \omterm{def:b1:flowering-plant-organization:organs}{الجذور} والجراثيم ثانية، وفي \omterm{def:b1:ecosystem-organization:ecosystem}{نظام بيئي} ناضج
تحمل هذه الدورة الداخلية أكثر دفق المغذّيات: فالمدخل من
التجوية والمطر والمخرج في ماء الجدول صغيران إلى جانبها.
و\emph{ميزانية المغذّيات} تقارن المداخل والمخارج والمكامن
الداخلية لعنصر واحد في \omterm{def:b1:ecosystem-organization:ecosystem}{نظام بيئي} محدد؛ أما الدورات
الجيوكيميائية الحيوية على مقياس الكوكب فتُتناول في كتاب السنة الثانية.
\end{definition}

\begin{proposition}[المجتمع الحي يحفظ المغذّيات]\label{prop:b1:ecosystem-dynamics:retention}
تُحفظ المغذّيات في \omterm{def:b1:ecosystem-organization:ecosystem}{نظام بيئي} بالتقاط النباتات الحية
والجراثيم؛ فإذا أُزيل الغطاء النباتي استمر التمعدن
بينما توقف الالتقاط، وغُسلت الأيونات المحررة إلى
الجداول --- فيسرّب \omterm{def:b1:ecosystem-organization:ecosystem}{النظام البيئي} حتى ينمو غطاء جديد.
\end{proposition}
\begin{proof}[شواهد]
في هَبَرد بروك في نيوهامبشير، ستة أودية صغيرة مشجّرة
يصرف كل واحد منها جدول مزوَّد بهدّار، حتى يمكن قياس كل ما
يغادر في الماء في مقابل ما يصل في
المطر. وفي 1965--66 قُطع أحد الأودية قطعًا كاملًا وأُبقي عاريًا
بمبيد أعشاب ثلاث سنوات، وتُرك جاره شاهدًا. فارتفع جريان
الجدول من الوادي المقطوع بمقدار \qty{40}{\%} (فلا نتح).
وارتفعت النترات في جدوله أربعين إلى ستين مثلًا، والكالسيوم عشرة أمثال،
والبوتاسيوم عشرين مثلًا؛ وكان مجموع فقد المواد المنحلة ستة
إلى ثمانية أمثال فقد الشاهد --- فالنتروجين الذي كانت الغابة السليمة
تحفظه في دورتها قرونًا غادر في صيفين. ولما سُمح
بالنمو ثانية، هبطت الفواقد إلى سابق عهدها في بضع سنوات،
بسرعة التقاط الغطاء الجديد لما تحرره التربة.
\end{proof}

\section{الجزر وتجمّع المجتمعات}

\begin{theorem}[نظرية التوازن في الجغرافيا الحيوية للجزر]\label{thm:b1:ecosystem-dynamics:island}
\index{جغرافيا حيوية للجزر}\index{علاقة الأنواع بالمساحة}
عدد الأنواع في جزيرة توازن ديناميكي بين
الهجرة الداخلة من مجمع مصدر فيه $P$ نوعًا والانقراض على
الجزيرة. فإذا هبط معدل هجرة الأنواع الجديدة هبوطًا خطيًا مع
العدد $S$ الحاضر أصلًا، $I = I_0(1 - S/P)$، وارتفع معدل الانقراض
ارتفاعًا خطيًا معه، $E = eS$، كان التوازن
\[
S^* = \frac{I_0 P}{I_0 + eP} ,
\]
يُبلَغ حين $I = E$، مع دوران متصل مقداره $eS^*$ نوعًا في
وحدة الزمن. والجزر القريبة من \omterm{def:b1:plant-transport:phloem}{المصدر} لها $I_0$ أعلى وأنواع
أكثر؛ والجزر الكبيرة لها $e$ أخفض (لكبر جمهراتها) وأنواع
أكثر. وعبر مجموعة من الجزر يرتفع عدد الأنواع مع المساحة
على صورة $S = cA^z$، وهو خط مستقيم ميله $z$ على رسم لوغاريتمي مزدوج، ويكون
$z$ عادة بين \num{0.2} و\num{0.35}.
\end{theorem}
\begin{proof}[برهان جزئي]
يرتفع $S$ ما دامت الهجرة تفوق الانقراض ويهبط حين
يفوق الانقراض الهجرة، فتكون $\dd S/\dd t = I_0(1 - S/P) - eS$
معادلة خطية من الرتبة الأولى توازنها الوحيد، $I_0 -
(I_0/P + e)S = 0$، يعطي $S^*$ وهو مستقر، إذ يكون معدل التغير
موجبًا تحته وسالبًا فوقه. وعند التوازن تظل الأنواع
تصل وتزول بالمعدل $eS^*$: فالعدد ثابت،
والقائمة ليست ثابتة. وقانون القوة تجريبي؛ وأسّه القريب من
\num{0.25} يعني أن جزيرة أكبر بعشرة أمثال تحمل نحو \num{1.8}
مثل من الأنواع، وأن فقد \qty{90}{\%} من \omterm{def:b1:ecosystem-organization:niche}{موئل} سيفقد
في النهاية نحو نصف أنواعه.
\end{proof}
\begin{proof}[شواهد]
عدّ سيمبرلوف وويلسون (1969) أنواع مفصليات الأرجل على جزيرات
مانغروف صغيرة قبالة فلوريدا، ثم بخّراها لقتل كل
حيوان، ثم عدّاها ثانية على فترات. ففي غضون سنة كانت كل جزيرة قد
استعادت عددًا قريبًا من عدد أنواعها الأصلي --- أكثر على
القريبة والكبيرة، وأقل على البعيدة والصغيرة --- لكن بقائمة
مختلفة، والقائمة ظلت تتغير: فالعدد كان في
توازن، والتركيب في دوران، كما اقتضت النظرية.
وكراكاتوا، وقد عُقّمت بثورانها سنة 1883، كان فيها \num{30} نوعًا من
النبات بعد ثلاث سنوات وأكثر من \num{270} بعد خمسين، مقاربةً
غنى جزر مشابهة قريبة.
\end{proof}

\begin{proposition}[التجزؤ]\label{prop:b1:ecosystem-dynamics:fragmentation}
\index{تجزؤ الموائل}
\omterm{def:b1:ecosystem-organization:niche}{الموئل} المقطَّع إلى شظايا يسلك سلوك أرخبيل: فكل
شظية تفقد أنواعًا نحو التوازن الذي تسمح به مساحتها وعزلتها،
ببطء أولًا (وهو \emph{دين الانقراض})، والأنواع الكبيرة الجسم
والنادرة والمتخصصة أولًا. والشظايا تكسب كذلك حافة --- أجف
وأشد رياحًا وأكثر اضطرابًا --- على حساب الداخل؛ فشظية مربعة
مساحتها \qty{1}{km^2} بمنطقة حافة عرضها \qty{100}{m} تحتفظ
بنسبة \qty{64}{\%} من مساحتها داخلًا، وشظية مساحتها \qty{0.1}{km^2} لا تحتفظ
إلا بنسبة \qty{15}{\%}. والممرات بين الشظايا ترفع معدل الهجرة،
ولهذا تعيد الأنواع أسرع مما تضيف مساحة.
\end{proposition}

\begin{method}[استعمال علاقة الأنواع بالمساحة]\label{met:b1:ecosystem-dynamics:sar}
\begin{enumerate}
  \item ارسم $\log S$ في مقابل $\log A$ لمجموعة من الجزر أو
        الشظايا مأخوذة عيناتها بجهد متساوٍ؛ وطابق خطًا بالعين أو
        بالمربعات الصغرى.
  \item اقرأ $z$ ميلًا و$c$ قيمةً للعدد $S$ عند $A = 1$.
  \item تنبأ بأنواع مساحة جديدة بالمقدار $cA^z$؛ وتنبأ
        بكسر الأنواع المحفوظة حين تُخفَّض مساحة من
        $A_0$ إلى $A$ بالمقدار $(A/A_0)^z$ --- فعند $z = 0.25$، يحفظ إبقاء عُشر
        \omterm{def:b1:ecosystem-organization:niche}{الموئل} \qty{56}{\%} من الأنواع في النهاية.
  \item احذر: فالفقد لا يُبلَغ إلا بعد سداد دين
        الانقراض، وقد يكون ذلك بعد عقود؛ والمطابقة لا تقول شيئًا عن
        أي الأنواع يذهب.
\end{enumerate}
\end{method}

\section{مسألة: الوادي المقطوع والأرخبيل}

\begin{problem}[{مسألة نهاية الأسبوع --- ميزانية المغذّيات في وادٍ
مقطوع كليًا، وقانون الأنواع والمساحة في أرخبيل،
وختامًا الأسّ $z$ وعدد الأنواع عند التوازن في
جزيرة}]
\label{pb:b1:ecosystem-dynamics:1}

\medskip
\noindent\textbf{الجزء الأول --- قراءة \omterm{def:b1:ecosystem-dynamics:succession}{تعاقب}.}
سلسلة زمنية من مواقع معلومة العمر منذ تراجع نهر جليدي
تعطي: عند 10 سنوات، طحالب وفرشة من مثبِّتات النتروجين، ونتروجين
التربة \qty{10}{kg/ha}؛ وعند 40 سنة، أدغال أرلة، \qty{150}{kg/ha}؛
وعند 80 سنة، غابة تنوب، \qty{300}{kg/ha}؛ وعند 200 سنة،
تنوب وتسوغا، \qty{320}{kg/ha}.
\begin{enumerate}
  \item احسب المعدل الوسطي لتراكم النتروجين في كل
        فترة وقل متى يكون أسرع.
  \item إلى أي آلية إحلال يشير منحني النتروجين،
        ولأي انتقال؟
  \item لماذا يكاد التراكم يتوقف بعد 80 سنة مع أن
        الغابة ما تزال حية؟
  \item تثبّت الأرلة \qty{50}{kg/ha} من النتروجين في السنة. فأي
        كسر منه يُحفظ في التربة بين السنة 40 والسنة
        80، وأين يذهب الباقي؟
  \item اقترح تجربة تميّز التيسير من
        التحمل في الانتقال من الأرلة إلى التنوب.
  \item لماذا تكون السلسلة الزمنية بديلًا عن مراقبة موقع واحد
        قرنين، لا الشيء نفسه؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الوادي المقطوع.}
واديان متجاوران مساحة كل واحد \qty{15}{ha} يتلقيان \qty{1300}{mm} من
المطر في السنة فيه \qty{6.5}{kg/ha} من النتروجين منحلًا. ويصدّر
الوادي الشاهد المشجَّر \qty{820}{mm} من ماء الجدول في السنة
تحمل \qty{2}{kg/ha} من نتروجين النترات، و\qty{9}{kg/ha} من الكالسيوم
و\qty{2}{kg/ha} من البوتاسيوم. ويصدّر الوادي المقطوع، في السنة الثانية
بعد القطع، \qty{1150}{mm} تحمل \qty{120}{kg/ha} من
نتروجين النترات، و\qty{90}{kg/ha} من الكالسيوم و\qty{36}{kg/ha} من
البوتاسيوم. وتحمل تربته ونثاره \qty{3500}{kg/ha} من النتروجين،
وتحمل كتلته الحيوية المقطوعة \qty{350}{kg/ha} أخرى.
\begin{enumerate}[resume]
  \item بكم ارتفع مردود الماء، ولماذا؟
  \item احسب تركيز نتروجين النترات الوسطي في كل جدول
        بالميليغرامات في اللتر (\qty{1}{mm} من الماء على \qty{1}{ha} هي
        \qty{10}{m^3}).
  \item احسب ميزان النتروجين (المدخل ناقص المخرج) في كل
        وادٍ. فأيهما يراكم النتروجين، وبأي معدل؟
  \item أي كسر من نتروجين تربة الوادي المقطوع غادر في
        تلك السنة الواحدة؟ وبذلك المعدل، كم يدوم المكمن؟
  \item بأي معاملات ارتفع تصدير الكالسيوم والبوتاسيوم؟
        ولماذا البوتاسيوم أكثر من الكالسيوم (فكّر أين يُحفظ كل واحد
        في تربة)؟
  \item من أين جاءت النترات المصدَّرة، مع أنه لم يكن ينمو
        غطاء نباتي يحررها؟ سمّ الخطوتين الجرثوميتين.
  \item بعد السماح بالنمو ثانية، يهبط التصدير إلى \qty{20}{kg/ha}
        في السنة 4 وإلى \qty{5}{kg/ha} في السنة 6. فكم يلزم حتى
        يفوق مدخل المطر المخرجَ ثانية، وما الذي أعاد
        الحفظ؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الأرخبيل.}
خمس جزر مساحاتها \qtylist{1;10;100;1000;10000}{km^2} تحمل \num{12} و\num{21} و\num{38} و\num{68} و
\num{120} نوعًا من طيور اليابسة. ومجمع البر الرئيس \num{300}
نوع.
\begin{enumerate}[resume]
  \item احسب $\log_{10}$ لكل مساحة ولكل عدد أنواع
        وارسمهما.
  \item استخرج $z$ من النقطتين الطرفيتين و$c$ من أصغر
        جزيرة؛ وتحقق من الجزر الوسطى.
  \item تنبأ بعدد أنواع جزيرة مساحتها \qty{250}{km^2}.
  \item الجزيرة التي مساحتها \qty{100}{km^2} تتلقى $I_0 = 6$ أنواع جديدة في السنة
        وهي خالية وتفقد كسرًا $e$ من أنواعها في السنة.
        فباستعمال $S^* = 38$، احسب $e$ والدوران عند التوازن.
  \item جزيرة بالمساحة نفسها وأبعد مرتين لها $I_0 = 3$.
        احسب عدد أنواعها عند التوازن بالقيمة $e$ نفسها.
  \item جسر يصل الجزيرة التي مساحتها \qty{100}{km^2} بالبر الرئيس
        فيرتفع $I_0$ إلى \num{30}. احسب التوازن الجديد
        وفسّر لماذا يبقى أدنى بكثير من \num{300}.
  \item لو كان $z$ يساوي \num{0.35} بدل \num{0.25}، أكانت الجزر
        الصغيرة تحمل أنواعًا أكثر أم أقل نسبةً إلى الكبيرة؟ وقل
        ماذا يعني $z$ العالي بيولوجيًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- التجزؤ والتعافي.}
\begin{enumerate}[resume]
  \item تُخطَّط محمية مساحتها \qty{1000}{km^2} في غابة مساحتها
        \qty{100000}{km^2} سيُزال سائرها. فعند $z =
        0.25$، أي كسر من أنواع الغابة تستطيع المحمية أن تحمله
        في النهاية؟
  \item أتحمل عشر محميات مساحة كل واحدة \qty{100}{km^2} أكثر أم أقل، إذا
        كانت معزولة؟ وإذا وُصلت بممرات؟
  \item منطقة حافة المحمية عرضها \qty{300}{m}. احسب كسر
        الداخل في محمية مربعة مساحتها \qty{1000}{km^2} وفي
        أخرى مساحتها \qty{100}{km^2}.
  \item ارسم تخطيطيًا عدد أنواع شظية حديثة العزل
        في مقابل الزمن، وفسّر مصطلح دين الانقراض.
  \item اذكر النتيجة: الأسّ $z$ في الأرخبيل وعدد
        الأنواع عند التوازن في الجزيرة التي مساحتها \qty{100}{km^2}
        قبل الجسر وبعده.
\end{enumerate}
\end{problem}
