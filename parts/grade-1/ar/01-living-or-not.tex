\chapter{حيٌّ أم غير حي؟}\label{ch:g1:living-or-not}

ينزلق حلزون ببطء على الممر المبلل. وبجانبه حصاة رمادية، مثله في
الصغر، ومثله في الاستدارة. الحلزون والحصاة في المطر نفسه --- ومع
ذلك أحدهما حي والآخر ليس بحي. فكيف نعرف؟ هذا هو أول سؤال في \omterm{def:g1:living-or-not:living}{علم
الأحياء}، وهو علم الكائنات الحية.

\section{ماذا تفعل الكائنات الحية}

\begin{definition}[الكائن الحي]\label{def:g1:living-or-not:living}
\emph{الكائن الحي}\index{كائن حي} شيء \emph{حي}\index{حي}: يولد،
ويتغذى، وينمو، ويمكن أن يكون له صغار، ويموت يومًا ما. الحيوانات
والنباتات كائنات حية. وعلم الأحياء\index{علم الأحياء} هو العلم الذي
يدرسها.
\end{definition}

\begin{example}[الحلزون والحصاة]\label{ex:g1:living-or-not:snail}
وُلد الحلزون من بيضة صغيرة. وهو يأكل أوراق الخس. وينمو --- وتنمو
صدفته معه. وقد تكون له حلزونات صغيرة. إذن الحلزون \omterm{def:g1:living-or-not:living}{كائن حي}. أما
الحصاة فلا تأكل شيئًا، ولا تنمو البتة، ولا تكون لها حصيات صغيرة:
الحصاة ليست حية.
\end{example}

\begin{proposition}[علامات الحياة]\label{prop:g1:living-or-not:signs}
يُظهر \omterm{def:g1:living-or-not:living}{الكائن الحي} هذه العلامات:
\begin{enumerate}
  \item \emph{يولد}\index{ولادة}؛
  \item \emph{يتغذى}\index{تغذي} --- فيأخذ غذاءه أو يصنعه بنفسه،
        كما تفعل النباتات بالضوء؛
  \item \emph{ينمو}\index{نمو}؛
  \item يمكن أن يكون له صغار --- كائنات حية جديدة مثله؛
  \item \emph{يموت}\index{موت} يومًا ما.
\end{enumerate}
وما لا يُظهر أيًّا من هذه العلامات فليس بحي.
\end{proposition}
\admitted

\begin{remark}[كل علامة تُحسب]\label{rem:g1:living-or-not:allsigns}
علامة واحدة لا تكفي. فالسيارة ``تشرب'' الوقود، وقنديل الجليد يطول
في البرد --- لكن السيارة صُنعت ولم تولد، وقنديل الجليد لا تكون له
قناديل صغيرة. ولكي يكون الشيء حيًّا، عليه أن يُظهر علامات الحياة
\emph{مجتمعة}: يولد، ويتغذى، وينمو، ويصنع كائنات حية جديدة، ويموت.
\end{remark}

\section{حي، وغير حي، ولم يكن حيًّا قط}

\begin{definition}[الجسم غير الحي]\label{def:g1:living-or-not:nonliving}
\emph{الجسم غير الحي}\index{جسم غير حي} شيء ليس بحي. وهو نوعان:
\begin{itemize}
  \item أشياء \emph{لم تكن حية قط}\index{لم يكن حيًّا قط}: حصاة،
        وماء، وريح، ودراجة؛
  \item أشياء \emph{لم تعد حية}\index{لم يعد حيًّا} أو \emph{تأتي
        من} كائنات حية: ورقة يابسة على الأرض، وطاولة خشبية، وكنزة
        صوفية.
\end{itemize}
\end{definition}

\begin{example}[الطاولة الخشبية]\label{ex:g1:living-or-not:table}
الطاولة الخشبية لا تتغذى ولا تنمو: فهي ليست حية. لكن خشبها كان
حيًّا: إنه يأتي من شجرة وُلدت من بذرة، ونمت سنوات طويلة، ثم قُطعت.
الطاولة \omterm{def:g1:living-or-not:nonliving}{جسم غير حي} --- وهي تأتي من \omterm{def:g1:living-or-not:living}{كائن حي}.
\end{example}

\begin{example}[وماذا عن الإنسان الآلي؟]\label{ex:g1:living-or-not:robot}
إنسان آلي لعبة يتحرك، ويصدر أصواتًا، ويومض ضوءه كعين صغيرة. أهو
حي؟ راجع العلامات: صُنع في مصنع ولم يولد؛ وبطاريته ليست غذاءً
ينميه؛ ولن تكون له أبدًا صغار من الآليين. الإنسان الآلي يتحرك،
لكنه ليس حيًّا.
\end{example}

\begin{method}[كيف تقرر: حي أم غير حي؟]\label{met:g1:living-or-not:decide}
اسأل أسئلة الحياة، واحدًا بعد الآخر:
\begin{enumerate}
  \item أوُلد؟ أيتغذى؟ أينمو؟
  \item أيمكن أن تكون له صغار؟ أيموت يومًا ما؟
  \item فإن كانت الأجوبة نعم: فهو حي.
  \item وإن كانت الأجوبة لا، فاسأل سؤالًا آخر: أيأتي \emph{من}
        شيء حي، كالخشب من الشجرة؟
\end{enumerate}
\end{method}

\begin{remark}[البذور تبدو نائمة]\label{rem:g1:living-or-not:seed}
بذرة فول يابسة في كيس تبدو ساكنة كالحصاة. لكن ضعها على قطن مبلل،
فيظهر بعد أيام قليلة جذير صغير: كانت البذرة حية طوال الوقت،
تنتظر. الكائنات الحية البطيئة الهادئة يسهل أن نخلط بينها وبين
الأجسام غير الحية --- وأسئلة الحياة تعيننا على النظر عن قرب.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:1}
اذكر علامات الحياة الخمس. وأيها ترى هريرة صغيرة تفعله اليوم؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:2}
حي أم غير حي؟ (أ) فراشة؛ (ب) دراجة؛ (ج) زهرة أقحوان؛ (د) سحابة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:3}
ورقة يابسة ملقاة على الممر. أهي حية الآن؟ أكانت حية يومًا؟ ومن
أين أتت؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:4}
يقول توم: ``النهر حي --- انظر، إنه يتحرك ويصدر ضجيجًا!'' استعمل
أسئلة \cref{met:g1:living-or-not:decide} لتجيبه برفق.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:5}
افرز في صناديق الشكل الثلاثة --- حي، وكان حيًّا، \omterm{def:g1:living-or-not:nonliving}{ولم يكن حيًّا قط}:
عنكبوت، وجورب صوفي، وقطرة مطر، وتفاحة على الشجرة، وكوز صنوبر ساقط.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:6}
لهب الشمعة ``يأكل'' الشمع، ويطول، ويموت حين تنفخ فيه. أهو \omterm{def:g1:living-or-not:living}{كائن
حي}؟ وأي سؤال من أسئلة الحياة يقول لا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:7}
سمِّ شيئين في غرفتك يأتيان من كائنات حية، كالطاولة الخشبية في
\cref{ex:g1:living-or-not:table}. واذكر \omterm{def:g1:living-or-not:living}{الكائن الحي} الذي يأتي منه
كل واحد.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:8}
جرّب: ضع فولة يابسة على قطن مبلل قرب النافذة، وحصاة على قطعة قطن
مبلل أخرى بجانبها. وانظر كل يوم مدة أسبوع. أيهما يتغير؟ وعلام يدل
ذلك؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:9}
إنسان آلي لعبة وحلزون يتحركان معًا على الأرض. أعطِ سؤالين من أسئلة
الحياة يختلف فيهما جوابهما.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:10}
الفطر لا يجري كالحيوانات، وليس أخضر كأكثر النباتات. وبعلامات
\cref{prop:g1:living-or-not:signs}، اشرح لماذا يبقى الفطر كائنًا
حيًّا.
\end{exercise}
