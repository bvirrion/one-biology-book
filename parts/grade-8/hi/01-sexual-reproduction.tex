\chapter{लैंगिक जनन}\label{ch:g8:sexual-reproduction}

मई में मेंढकों का गाना, अप्रैल में चेरी का फूलना, पतझड़ में बारहसिंगे की
गरज, और मधुमक्खियों की ओर अपना भरा-भरा चेहरा घुमाता सूरजमुखी: \omterm{def:g6:exploring-our-environment:components}{सजीव}
दुनिया भर में बेतहाशा \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} एक ही परियोजना पर ख़र्च होती है --- अगली
पीढ़ी। पिछले सालों ने टुकड़े बटोरे थे
(\cref{ch:g4:animal-reproduction}, \cref{ch:g5:flower-to-fruit}); यह
साल उन्हें एक ही सिद्धांत में जोड़ता है, और वह तीखा सवाल पूछता है:
आख़िर \emph{दो} जनक क्यों?

\begin{omfigure}
\includegraphics[width=0.6\linewidth]{images/book1/ai/g8-frog-chorus.jpg}

{\small एक ही परियोजना के लिए बेतहाशा \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा}: \omterm{def:g4:animal-reproduction:sexual}{नर} की फूली हुई स्वर-थैली,
जो \omterm{def:g8:sexual-reproduction:gametes}{युग्मकों} के मिलन के लिए पूरे तालाब को बुला रही है।}

\section{सजीव दुनिया भर में एक ही कार्य-प्रणाली}

\begin{definition}[युग्मक]\label{def:g8:sexual-reproduction:gametes}
\cref{def:g4:animal-reproduction:sexual} वाली \omterm{def:g5:first-look-at-cells:cell}{जनन-कोशिकाओं} का आम नाम
\emph{युग्मक}\index{युग्मक} है: \omterm{def:g4:animal-reproduction:sexual}{नर} युग्मक (\omterm{def:g1:animals-around-us:animal}{जंतुओं} में \omterm{def:g4:animal-reproduction:sexual}{शुक्राणु},
फूलदार \omterm{def:g1:plants-around-us:plant}{पौधों} में पराग-कण की \omterm{def:g5:first-look-at-cells:cell}{कोशिका}) और \omterm{def:g4:animal-reproduction:sexual}{मादा} युग्मक (\omterm{def:g4:animal-reproduction:sexual}{अंडाणु}, चाहे
अंडाशय में हो या \omterm{prop:g5:flower-to-fruit:roles}{बीजांड} में)। युग्मक अकेली \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} हैं
(\cref{ch:g5:first-look-at-cells}), और हर एक आधी शुरुआत है: अकेला
युग्मक मर जाता है; जुड़ जाएँ तो दो मिलकर एक जीवन बनाते हैं।
\end{definition}

\begin{proposition}[न बदलने वाला केंद्र]\label{prop:g8:sexual-reproduction:core}
\omterm{def:g4:animal-reproduction:sexual}{लैंगिक जनन} जहाँ भी चले --- मछली हो या देवदार, घोंघा हो या इंसान ---
केंद्र वही रहता है:
\begin{enumerate}
  \item दो जनक \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} बनाते हैं, \omterm{def:g4:animal-reproduction:sexual}{नर} और \omterm{def:g4:animal-reproduction:sexual}{मादा};
  \item \emph{\omterm{def:g5:flower-to-fruit:steps}{निषेचन}}: एक \omterm{def:g4:animal-reproduction:sexual}{नर} \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} एक \omterm{def:g4:animal-reproduction:sexual}{मादा} \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} से जुड़ता है, और
        एक ही नई \omterm{def:g5:first-look-at-cells:cell}{कोशिका} बनती है --- यानी \emph{निषेचित \omterm{def:g4:animal-reproduction:sexual}{अंडाणु}}, या
        \emph{युग्मनज}\index{युग्मनज};
  \item \omterm{prop:g8:sexual-reproduction:core}{युग्मनज} बँटता जाता है
        (\cref{prop:g5:first-look-at-cells:growth}), और एक नया जीव
        बनाता है, जो \emph{दोनों} जनकों से कुछ न कुछ लिए होता है
        (\cref{rem:g4:animal-reproduction:mix})।
\end{enumerate}
इस केंद्र के चारों ओर बाक़ी सब कुछ फेरबदल है: \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} कहाँ मिलते हैं,
कितने होते हैं, और बच्चों की देखभाल कौन करता है।
\end{proposition}
\admitted

\begin{example}[फेरबदल, छाँटकर]\label{ex:g8:sexual-reproduction:variations}
पुराना जमा किया हुआ सब कुछ इस केंद्र के नीचे साफ़-साफ़ दर्ज हो जाता है।
बाहरी बनाम भीतरी \omterm{def:g5:flower-to-fruit:steps}{निषेचन}
(\cref{prop:g4:animal-reproduction:where}): मिलने की जगह। \omterm{def:g4:animal-reproduction:ovivivi}{अंडज} बनाम
\omterm{def:g4:animal-reproduction:ovivivi}{जरायुज} (\cref{def:g4:animal-reproduction:ovivivi}): \omterm{prop:g8:sexual-reproduction:core}{युग्मनज} का ठिकाना।
कॉड मछली के लाखों बनाम हाथी का इकलौता
(\cref{prop:g4:animal-reproduction:strategies}): गिनती का बहीखाता।
मधुमक्खी से या हवा से \omterm{def:g5:flower-to-fruit:steps}{परागण}
(\cref{def:g5:flower-to-fruit:steps}): \omterm{def:g4:animal-reproduction:sexual}{नर} \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} की सफ़र-सेवा। एक ही
कार्य-प्रणाली, और उसकी सेटिंगों की एक पूरी सूची।
\end{example}

\begin{remark}[पौधे भी वही केंद्र चलाते हैं]\label{rem:g8:sexual-reproduction:plants}
\cref{ch:g5:flower-to-fruit} को \cref{ch:g4:animal-reproduction} के
बग़ल में रखो और मेल बिलकुल ठीक बैठता है: \omterm{def:g3:flowers-fruits-seeds:parts}{पुंकेसर} और \omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसर} \omterm{def:g8:sexual-reproduction:gametes}{युग्मक}
बनाते हैं, पराग-नली भीतरी \omterm{def:g5:flower-to-fruit:steps}{निषेचन} का गलियारा है, और \omterm{def:g2:from-seed-to-plant:seed}{बीज} \omterm{def:g1:plants-around-us:plant}{पौधे} के \omterm{def:g5:human-reproduction:pregnancy}{भ्रूण} को
उसके बँधे हुए खाने के साथ ठहराता है
(\cref{def:g2:from-seed-to-plant:seed})। अलग खड़ा है तो सिर्फ़ \omterm{def:g4:plants-without-seeds:vegetative}{कायिक
जनन} (\cref{def:g4:plants-without-seeds:vegetative}) --- एक जनक, कोई
\omterm{def:g8:sexual-reproduction:gametes}{युग्मक} नहीं, और हूबहू नक़लें: और यही वह उलटाव है जो अगले खंड का सवाल
तीखा बना देता है।
\end{remark}

\section{दो जनक क्यों?}

\begin{proposition}[लैंगिक जनन पत्ते फेंटता है]\label{prop:g8:sexual-reproduction:shuffle}
\omterm{ex:g4:plants-without-seeds:runner}{भूस्तारी} से बनी स्ट्रॉबेरी की नक़लें
(\cref{rem:g4:plants-without-seeds:copy}) दिखाती हैं कि एक जनक क्या
देता है: एकरूपता। दो जनक ठीक उलटा देते हैं। चूँकि हर \omterm{prop:g8:sexual-reproduction:core}{युग्मनज} हर जनक से
एक-एक \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} जोड़ता है --- और, जैसा आगे का एक अध्याय दिखाएगा, एक ही जनक
के भी कोई दो \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} बिलकुल एक-सी विरासत नहीं ढोते --- \omterm{def:g4:animal-reproduction:sexual}{लैंगिक जनन} हर
संतान को एक \emph{नया मेल} बाँटता है। भाई-बहन अलग होते हैं; एक ही फली के
कोई दो \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} एक-से नहीं होते; और हर \omterm{def:g5:biodiversity:species}{प्रजाति} के भीतर की विविधता
(\cref{rem:g5:biodiversity:scales}) बड़ी हद तक इसी फेंटाई का काम है।
\end{proposition}
\admitted

\begin{example}[बाग़ के दो दाँव]\label{ex:g8:sexual-reproduction:bets}
\cref{rem:g4:plants-without-seeds:copy} वाला बाग़ दोनों खेल जान-बूझकर
खेलता है। \omterm{met:g4:plants-without-seeds:cutting}{कलम} और पैबंद: एक जनक, और उस बेहतरीन \omterm{def:g1:plants-around-us:tree}{पेड़} की हूबहू नक़लें ---
पर साथ में एक साझी कमज़ोरी भी, इसलिए जो झुलसा एक को गिराए वह सबको गिरा
सकता है। \omterm{def:g2:from-seed-to-plant:seed}{बीज}: दो जनक, और हर \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} एक ताज़ा मेल --- ज़्यादातर मामूली, कुछ
और भी ख़राब, और कोई-कोई दोनों जनकों से बेहतर; और कोई दो एक-से नहीं, कि
कोई झुलसा उन्हें एक साथ बहा ले जाए। नक़ल वर्तमान को बैंक में रखती है;
फेंटाई भविष्य की टोह लेती है।
\end{example}

\begin{remark}[विविधता ही प्रजाति की बचत है]\label{rem:g8:sexual-reproduction:reserve}
यह फेंटाई बाग़ से आगे क्यों मायने रखती है:
\cref{prop:g5:biodiversity:web} ने दिखाया था कि जाल \omterm{def:g5:biodiversity:species}{प्रजातियों} की
विविधता से झटके सोख लेते हैं; और किसी एक \omterm{def:g5:biodiversity:species}{प्रजाति} के भीतर व्यक्तियों की
विविधता वही भूमिका निभाती है। जब \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} पलटती हैं --- कोई नई
बीमारी, कोई ज़्यादा कड़ी सर्दी --- तो विविधता वाली \omterm{def:g5:biodiversity:species}{प्रजाति} में किसी न
किसी के नई शर्तों पर खरा उतरने की उम्मीद ज़्यादा होती है। इससे जीवन के
इतिहास पर जो इशारा मिलता है उसे \cref{ch:g9:how-species-change} खुलकर
कहेगा; यहाँ बस इतना ध्यान रखो कि यह बचत भरी हुई \omterm{def:g4:animal-reproduction:sexual}{लैंगिक जनन} ही रखता है।
\end{remark}

\section{किसी भी प्रजाति का जनन पढ़ना}

\begin{method}[जनन की मिसल]\label{met:g8:sexual-reproduction:dossier}
किसी भी \omterm{def:g5:biodiversity:species}{प्रजाति} के लिए यह जमा करो:
\begin{enumerate}
  \item \omterm{def:g8:sexual-reproduction:gametes}{युग्मक}: कौन कौन-सा बनाता है, और कहाँ;
  \item मिलन: बाहरी \omterm{def:g5:flower-to-fruit:steps}{निषेचन} या भीतरी, और साथ में सफ़र-सेवा (पानी,
        \omterm{prop:g4:animal-reproduction:where}{समागम}, परागकर्ता, हवा);
  \item ठिकाना: \omterm{def:g2:animals-grow-up:born}{अंडे} दिए गए या बच्चे ढोए गए; और किस साज़-सामान के
        साथ;
  \item गिनती और देखभाल: रणनीति की लकीर पर उसकी जगह
        (\cref{prop:g4:animal-reproduction:strategies});
  \item मौसम: कब, और किस चीज़ से सधा हुआ --- यानी अगले अध्याय का
        सवाल।
\end{enumerate}
\end{method}

\begin{example}[दो मिसलें, आमने-सामने]\label{ex:g8:sexual-reproduction:dossiers}
काँटा-मछली (\cref{exo:g4:animal-reproduction:10}): \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} पानी में
छोड़े गए, बाहरी \omterm{def:g5:flower-to-fruit:steps}{निषेचन}, \omterm{def:g2:animals-grow-up:born}{अंडे} एक बनाए हुए घोंसले में, दर्जनों भर, और
रखवाली पिता की, वसंत में। चेरी: \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} में, भीतरी \omterm{def:g5:flower-to-fruit:steps}{निषेचन} पराग-नली
से होकर, \omterm{def:g5:human-reproduction:pregnancy}{भ्रूण} \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} के भीतर \omterm{def:g2:from-seed-to-plant:seed}{बीजों} में ठहरे, हज़ारों, साज़-सामान देकर
\omterm{prop:g3:sorting-living-things:groups}{पक्षियों} की पहुँचाने वाली सेवा के हवाले
(\cref{exo:g6:how-plants-spread:12}), और अप्रैल पर सधे हुए। एक ही
ख़ाका, कोई भी \omterm{def:g5:biodiversity:species}{प्रजाति}: मिसल यानी सिद्धांत, इस्तेमाल में।
\end{example}

\section{समस्या: नस्ल सुधारने का कार्यक्रम}

\begin{problem}[{सप्ताहांत समस्या --- किसी पौध-प्रजनक का एक साल,
सिद्धांत पर चलाया हुआ}]\label{pb:g8:sexual-reproduction:1}
एक प्रजनक नया सेब चाहती है: क़िस्म क जैसा कुरकुरा, और क़िस्म ख जैसा
झुलसा-रोधी। उसके औज़ार ठीक इसी अध्याय वाले हैं।

\medskip\noindent\textbf{भाग I --- संकरण।}
\begin{enumerate}
  \item वह ब्रश से क़िस्म क के \omterm{def:g3:flowers-fruits-seeds:parts}{पुंकेसरों} से \omterm{def:g3:flowers-fruits-seeds:parts}{पराग} लेती है और उसे
        क़िस्म ख के \omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसरों} पर छुआ देती है, फिर मधुमक्खियों से
        बचाने के लिए फूलों पर थैलियाँ चढ़ा देती है। हर क़दम की
        केंद्र में भूमिका बताओ --- और यह भी कि थैलियाँ क्यों मायने
        रखती हैं (\cref{met:g5:flower-to-fruit:diagnose} ने वही ब्रश
        इस्तेमाल किया था)।
  \item उसकी इस मुलाक़ात के बाद क़िस्म ख के \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} में क्या उगता है, और
        हर \omterm{def:g2:from-seed-to-plant:seed}{बीज} कौन-सी दो विरासतें जोड़ता है?
  \item वह तीन सौ \omterm{def:g2:from-seed-to-plant:seed}{बीज} बोती है।
        \cref{prop:g8:sexual-reproduction:shuffle} के अनुसार वह तीन
        सौ \emph{अलग-अलग} \omterm{def:g1:plants-around-us:tree}{पेड़ों} की उम्मीद क्यों करती है?
  \item ज़्यादातर \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} मामूली होंगे, कुछ दोनों जनकों से भी ख़राब, और
        शायद कोई एक बिलकुल ठीक। अध्याय के किस उदाहरण ने ठीक इसी
        फैलाव की भविष्यवाणी की थी?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- जीते हुए को पक्का करना।}
\begin{enumerate}[resume]
  \item आठवाँ साल: एक \omterm{def:g1:plants-around-us:tree}{पेड़} कुरकुरे, झुलसा-रोधी सेब देता है। अगर वह
        उसके \omterm{def:g2:from-seed-to-plant:seed}{बीज} बो दे, तो उस बेशक़ीमती मेल का क्या होगा --- और
        क्यों?
  \item इसके बजाय वह जीते हुए \omterm{def:g1:plants-around-us:tree}{पेड़} को पैबंद लगाकर बढ़ाती है --- यह
        कौन-सा जनन है, और यह क्या पक्का करता है?
  \item उसका आगे का पूरा बाग़ उसी एक \omterm{def:g1:plants-around-us:tree}{पेड़} की नक़लें होगा। उसने जो
        ख़तरा क़बूल किया है उसका नाम बताओ, और उसके लिए अध्याय वाला
        मुहावरा भी।
  \item मानक बीमा सुझाओ: उसके बाग़ को और क्या रखना चाहिए, और यह
        रणनीति \cref{rem:g8:sexual-reproduction:reserve} की गूँज
        क्यों है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- सिद्धांत की जाँच-पड़ताल।}
\begin{enumerate}[resume]
  \item उसके कार्यक्रम ने बाग़ के दोनों दाँव एक के बाद एक इस्तेमाल
        किए। बताओ कि किस चरण ने कौन-सा, और हर एक ने उसे क्या दिया।
  \item एक सहकर्मी काँटा-मछलियों की नस्ल उसी तरह सुधारता है ---
        जनक चुनकर, फेंटी हुई संतान पालकर, और सबसे अच्छे रखकर।
        \cref{met:g8:sexual-reproduction:dossier} की वे प्रविष्टियाँ
        गिनाओ जहाँ मछली वाला तरीक़ा सेब वाले से अलग है, और वे
        केंद्रीय क़दम भी जहाँ वह अलग हो ही नहीं सकता।
  \item प्रजनक कहती है: ``लिंग प्रस्ताव रखता है, चुनाव फ़ैसला करता है
        --- मैं तो बस उसी में से चुनती हूँ जो फेंटाई बाँटती है।''
        उसके इस नारे को \cref{rem:g8:sexual-reproduction:reserve}
        वाली विविधता की बचत से जोड़ो।
  \item समापन के लिए एक वाक्य: कार्यक्रम को \omterm{def:g4:animal-reproduction:sexual}{लैंगिक जनन} \emph{किसलिए}
        चाहिए था, और \omterm{def:g4:plants-without-seeds:vegetative}{कायिक जनन} किसलिए?
\end{enumerate}
\end{problem}
