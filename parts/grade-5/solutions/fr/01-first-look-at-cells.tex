\section*{Chapitre \ref{ch:g5:first-look-at-cells} --- Un premier regard sur les cellules}

\begin{solution}{exo:g5:first-look-at-cells:1}
La plus petite brique vivante d'un \omterm{def:g1:living-or-not:living}{être vivant}. Ton corps en est
fait de dizaines de milliers de milliards.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:2}
Parce que les \omterm{def:g5:first-look-at-cells:cell}{cellules} sont plus petites que la plus petite tache
visible à l'\omterm{def:g1:the-five-senses:sense}{œil}. Le \omterm{def:g5:first-look-at-cells:microscope}{microscope} --- qui grossit des centaines de fois
de fines tranches --- les a révélées.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:3}
Pellicule d'oignon~: des \omterm{def:g5:first-look-at-cells:cell}{cellules} en briques bien rangées, paroi
contre paroi. Grattage de joue~: des \omterm{def:g5:first-look-at-cells:cell}{cellules} plus rondes,
dispersées. En commun~: les deux matières sont entièrement faites de
\omterm{def:g5:first-look-at-cells:cell}{cellules}, la plupart montrant à l'intérieur une petite tache sombre.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:4}
Le noyau --- le centre de commande de la \omterm{def:g5:first-look-at-cells:cell}{cellule}.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:5}
Environ $50$ \omterm{def:g5:first-look-at-cells:cell}{cellules} couvrent un millimètre~; une \omterm{def:g5:first-look-at-cells:cell}{cellule} mesure
environ un cinquantième de millimètre --- $\qty{0.02}{mm}$.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:6}
Grandir en taille~: les \omterm{def:g5:first-look-at-cells:cell}{cellules} du corps se divisent et se
multiplient. Guérir~: des \omterm{def:g5:first-look-at-cells:cell}{cellules} se divisent pour reconstruire la
zone abîmée.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:7}
Une seule. C'était la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée --- la rencontre d'une
\omterm{def:g4:animal-reproduction:sexual}{cellule mâle} et d'une \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:8}
Le doublement à partir de $1$ donne $2$, $4$, $8$, $16$, $32$~: il y
a $32$ \omterm{def:g5:first-look-at-cells:cell}{cellules} après $5$ divisions. Cinq doublements de plus
donnent $64$, $128$, $256$, $512$, $1024$~: donc $1024$ \omterm{def:g5:first-look-at-cells:cell}{cellules}
après $10$.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:9}
Si différentes soient-elles par la taille, la forme et la vie,
toutes deux sont bâties du même genre de petites unités vivantes ---
les \omterm{def:g5:first-look-at-cells:cell}{cellules}. Ce plan de construction partagé, commun à tous les
\omterm{def:g1:living-or-not:living}{êtres vivants}, est ce que veut dire «~\omterm{prop:g5:first-look-at-cells:principle}{unité du vivant}~».
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:10}
Tous~: une \omterm{def:g5:first-look-at-cells:cell}{cellule} unique naît (d'une \omterm{def:g5:first-look-at-cells:cell}{cellule} mère qui se divise),
se nourrit, grandit, se divise --- faisant de nouveaux \omterm{def:g1:living-or-not:living}{êtres vivants}
comme elle --- et meurt. Une seule \omterm{def:g5:first-look-at-cells:cell}{cellule} suffit à une vie
complète.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:11}
Davantage de \omterm{def:g5:first-look-at-cells:cell}{cellules} --- la taille des \omterm{def:g5:first-look-at-cells:cell}{cellules} change à peine de
la souris à l'éléphant. Grandir n'est donc pas gonfler ses \omterm{def:g5:first-look-at-cells:cell}{cellules}
mais les multiplier~: les corps deviennent grands par division après
division, exactement comme le dit la proposition.
\end{solution}
