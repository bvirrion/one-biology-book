\chapter{One Cell or Many}\label{ch:g6:uni-multi}

Some \omterm{def:g4:ecosystems:levels}{organisms} are complete living systems inside a single \omterm{def:g6:cell-theory:cell}{cell}. Others
are teams of trillions. This chapter compares \omterm{def:g6:cell-theory:uni-multi}{unicellular} and
\omterm{def:g6:cell-theory:uni-multi}{multicellular} life and introduces \omterm{def:g6:uni-multi:tissue}{tissues} and \omterm{def:g6:uni-multi:tissue}{organs} as levels of
organisation.

\section{Life in one cell}

\begin{definition}[Unicellular organism]\label{def:g6:uni-multi:uni}
A \emph{unicellular organism}\index{unicellular organism} consists of a
single \omterm{def:g6:cell-theory:cell}{cell} that carries out all life functions: feeding, exchange with
the \omterm{def:g4:traits:env}{environment}, sensing, and \omterm{def:g5:reproduction:def}{reproduction}.
\end{definition}

\begin{example}[Familiar microbes]\label{ex:g6:uni-multi:examples}
Many \omterm{def:g4:microbes:def}{bacteria}, yeasts and some algae (such as \emph{Chlamydomonas}) are
\omterm{def:g6:cell-theory:uni-multi}{unicellular}. Amoebas move and feed as single \omterm{def:g6:cell-theory:cell}{cells} in \omterm{def:g2:habitats:kinds}{pond} \omterm{def:g4:plants-need:needs}{water}.
\end{example}

\begin{proposition}[Small and complete]\label{prop:g6:uni-multi:complete}
Being \omterm{def:g6:cell-theory:uni-multi}{unicellular} does not mean ``simple'' in every way: one \omterm{def:g6:cell-theory:cell}{cell} must
still do every job. Size is limited by the need to exchange materials
efficiently.
\end{proposition}
\admitted

\section{Life with many cells}

\begin{definition}[Multicellular organism]\label{def:g6:uni-multi:multi}
A \emph{multicellular organism}\index{multicellular organism} is made of
many \omterm{def:g6:cell-theory:cell}{cells} that usually specialise and cooperate. \omterm{def:g6:cell-theory:cell}{Cells} depend on one
another; a \omterm{def:g3:bones:muscle}{muscle} \omterm{def:g6:cell-theory:cell}{cell} alone cannot live as a free \omterm{def:g1:plants-animals:animal}{animal}.
\end{definition}

\begin{definition}[Specialisation]\label{def:g6:uni-multi:special}
\emph{Cell specialisation}\index{cell specialisation} (differentiation)
means \omterm{def:g6:cell-theory:cell}{cells} develop different structures for different jobs --- \omterm{def:g5:nerves:system}{nerve},
\omterm{def:g5:circulation:blood}{blood}, \omterm{def:g4:plants-need:parts}{leaf}, \omterm{def:g4:plants-need:parts}{root} hair, and so on.
\end{definition}

\section{Tissues and organs}

\begin{definition}[Tissue and organ]\label{def:g6:uni-multi:tissue}
A \emph{tissue}\index{tissue} is a group of similar \omterm{def:g6:cell-theory:cell}{cells} working
together. An \emph{organ}\index{organ} is a structure of different
tissues with a major job (\omterm{def:g4:plants-need:parts}{leaf}, \omterm{def:g5:circulation:heart}{heart}, \omterm{def:g5:nerves:system}{brain}). Organs form
\emph{organ systems}\index{organ system}.
\end{definition}

\begin{example}[Leaf as organ]\label{ex:g6:uni-multi:leaf}
A \omterm{def:g4:plants-need:parts}{leaf} contains photosynthetic \omterm{def:g6:uni-multi:tissue}{tissue}, transport \omterm{def:g6:uni-multi:tissue}{tissue} and protective
epidermis --- \omterm{def:g6:uni-multi:tissue}{tissues} cooperating as an \omterm{def:g6:uni-multi:tissue}{organ} of \omterm{def:g5:photosynthesis:def}{photosynthesis}.
\end{example}

\begin{example}[Human stomach]\label{ex:g6:uni-multi:stomach}
\omterm{def:g3:bones:muscle}{Muscle} \omterm{def:g6:uni-multi:tissue}{tissue} churns \omterm{def:g2:food-energy:food}{food}; gland \omterm{def:g6:uni-multi:tissue}{tissue} secretes juices; connective and
nervous \omterm{def:g6:uni-multi:tissue}{tissues} support and \omterm{def:g6:method:control}{control} --- one \omterm{def:g6:uni-multi:tissue}{organ}, many \omterm{def:g6:uni-multi:tissue}{tissues}.
\end{example}

\section{Colonial in-between}

\begin{method}[Classifying an organism's organisation]\label{met:g6:uni-multi:class}
\begin{enumerate}
  \item Can one \omterm{def:g6:cell-theory:cell}{cell} live independently as the whole \omterm{def:g4:ecosystems:levels}{organism}?
  \item If many \omterm{def:g6:cell-theory:cell}{cells}: are they specialised into \omterm{def:g6:uni-multi:tissue}{tissues}?
  \item Name \omterm{def:g6:uni-multi:tissue}{organs} if present.
  \item Note \omterm{def:g4:ecosystems:habitat}{habitat} and size as clues (not proof).
\end{enumerate}
\end{method}
