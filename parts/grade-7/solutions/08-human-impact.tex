\section*{Chapter \ref{ch:g7:impact} --- Humans and the Environment}

\begin{solution}{exo:g7:impact:1}
Reduction or destruction of \omterm{def:g1:caring:needs}{places} \omterm{def:g4:ecosystems:levels}{organisms} live. Examples: deforestation
for farms/cities, wetland draining, mining.
\end{solution}

\begin{solution}{exo:g7:impact:2}
Harmful substances/\omterm{def:g6:nutrition:def}{energy} added to the \omterm{def:g4:traits:env}{environment}. Examples: plastic
waste, oil spills, pesticide runoff, smog, excess fertiliser, noise.
\end{solution}

\begin{solution}{exo:g7:impact:3}
It spreads and causes ecological, economic or \omterm{def:g2:healthy:health}{health} harm --- not just
presence outside the native range.
\end{solution}

\begin{solution}{exo:g7:impact:4}
$\mathrm{CO_2}$, methane (also $\mathrm{N_2O}$, etc.). $\mathrm{CO_2}$:
burning \omterm{def:g5:evolution:fossil}{fossil} fuels, deforestation reducing uptake.
\end{solution}

\begin{solution}{exo:g7:impact:5}
Protected parks, fishing quotas, reforestation, \omterm{def:g7:impact:pollution}{pollution} laws, recycling,
renewable \omterm{def:g6:nutrition:def}{energy}, wildlife corridors (any three valid).
\end{solution}
