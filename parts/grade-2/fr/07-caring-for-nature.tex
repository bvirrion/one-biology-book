\chapter{Prendre soin de la nature}\label{ch:g2:caring-for-nature}

Après le pique-nique, une famille laisse le pré exactement comme
elle l'a trouvé~; une autre laisse des sacs, des bouchons de
bouteille et une fourmilière piétinée. Les lapins, les fourmis et
les pâquerettes du pré ne peuvent pas nous demander d'être prudents
--- mais ce sont des \omterm{def:g1:living-or-not:living}{êtres vivants} qui partagent nos lieux, et notre
façon de nous conduire dans la \omterm{def:g2:caring-for-nature:environment}{nature} compte pour eux. Prendre soin
de la \omterm{def:g2:caring-for-nature:environment}{nature} s'apprend, à tout âge.

\section{La nature, c'est là où vivent les êtres vivants}

\begin{definition}[La nature et l'environnement]\label{def:g2:caring-for-nature:environment}
La \emph{nature}\index{nature} est le monde des \omterm{def:g1:living-or-not:living}{êtres vivants} et de
ce qui les entoure~: prés, forêts, mares, mers --- et l'air, l'eau
et le sol dont ils dépendent. Ce qui entoure un \omterm{def:g1:living-or-not:living}{être vivant} et dont
il dépend s'appelle son \emph{environnement}\index{environnement}.
\end{definition}

\begin{example}[Tout est la maison de quelqu'un]\label{ex:g2:caring-for-nature:home}
Un tronc pourri dans la forêt a l'air d'un déchet --- c'est un
palais~: des scarabées sous l'écorce, des cloportes en dessous, de
la mousse sur le dessus, un hérisson qui hiverne dans le creux.
Éparpille-le pour t'amuser, et quatre foyers perdent leur maison.
Dans la \omterm{def:g2:caring-for-nature:environment}{nature}, presque tout est la maison, la nourriture ou l'abri
de quelqu'un.
\end{example}

\begin{remark}[Nous en faisons partie]\label{rem:g2:caring-for-nature:part}
Les humains aussi sont des \omterm{def:g1:living-or-not:living}{êtres vivants}~: l'air que nous
respirons, l'eau que nous buvons et les aliments que nous mangeons
viennent tous de la \omterm{def:g2:caring-for-nature:environment}{nature}. En prendre soin n'est pas seulement de
la gentillesse envers les lapins --- c'est veiller sur le monde dont
nous vivons nous-mêmes, comme
\cref{ch:g2:where-animals-live} le disait de tout \omterm{def:g2:where-animals-live:habitat}{milieu de vie}.
\end{remark}

\section{Ce qui nuit, ce qui aide}

\begin{proposition}[De petits gestes, de vrais dégâts]\label{prop:g2:caring-for-nature:harm}
Dans la \omterm{def:g2:caring-for-nature:environment}{nature}, le mal vient rarement de l'envie de mal faire ---
il vient du manque de réflexion~:
\begin{itemize}
  \item un déchet reste des années et peut piéger ou empoisonner
        des \omterm{def:g1:animals-around-us:animal}{animaux}~;
  \item cueillir toutes les \omterm{def:g1:plants-around-us:parts}{fleurs} n'en laisse aucune pour faire
        des \omterm{def:g2:from-seed-to-plant:seed}{graines} --- ni pour les abeilles~;
  \item crier fort et courir après les bêtes les éloigne de leur
        nourriture et de leurs petits~;
  \item des berges piétinées, des pierres retournées et des
        \omterm{def:g1:plants-around-us:tree}{branches} cassées défont chacune une petite maison.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Le bouchon de bouteille]\label{ex:g2:caring-for-nature:cap}
Un bouchon en plastique tombé n'est rien pour nous et reste dans le
pré pendant de très, très longues années --- assez pour que des
adultes aient des enfants qui retrouvent le même bouchon. La \omterm{def:g2:caring-for-nature:environment}{nature}
n'a pas de camion-poubelle~: tout ce que nous laissons, reste.
\end{example}

\begin{method}[Les règles du visiteur]\label{met:g2:caring-for-nature:rules}
Dehors, dans la \omterm{def:g2:caring-for-nature:environment}{nature}~:
\begin{enumerate}
  \item remporte tous tes déchets --- tous~;
  \item regarde les \omterm{def:g1:plants-around-us:parts}{fleurs}, les insectes et les nids au lieu de les
        collectionner~; cueille au plus quelques \omterm{def:g1:plants-around-us:parts}{fleurs} communes,
        jamais toute une touffe~;
  \item remets en place chaque pierre et chaque bûche que tu
        retournes --- quelqu'un habite dessous~;
  \item reste sur les sentiers près des endroits fragiles comme les
        berges~;
  \item observe les \omterm{def:g1:animals-around-us:animal}{animaux} de loin, en silence, et laisse leurs
        petits tranquilles.
\end{enumerate}
Laisse l'endroit tel que personne ne puisse deviner que tu es passé.
\end{method}

\begin{example}[Aider exprès]\label{ex:g2:caring-for-nature:help}
Prendre soin peut aller plus loin que ne pas nuire. Une soucoupe
d'eau pendant une canicule, un petit coin sauvage laissé sans
tondre au jardin, des \omterm{def:g2:from-seed-to-plant:seed}{graines} pour les oiseaux dans le froid, une
mare d'école comme celle de
\cref{ex:g2:where-animals-live:schoolpond}~: chacun rend à des \omterm{def:g1:living-or-not:living}{êtres
vivants} la nourriture, l'eau ou la maison qu'ils avaient perdues.
\end{example}

\section{Trier et économiser}

\begin{proposition}[Les déchets peuvent être rendus]\label{prop:g2:caring-for-nature:recycle}
Une grande partie de ce que nous jetons peut resservir.
\emph{Trier}\index{tri des déchets} les déchets --- le verre avec le
verre, le papier avec le papier, les épluchures au compost --- permet
aux bouteilles de redevenir des bouteilles et aux épluchures de
redevenir de la terre, au lieu que tout s'entasse. Et ce qui est
\emph{économisé} --- eau, papier, électricité --- ne devient jamais
un déchet du tout.
\end{proposition}
\admitted

\begin{example}[Le composteur]\label{ex:g2:caring-for-nature:compost}
Trognons de pomme et épluchures de carotte, dans un composteur, se
transforment lentement en une terre sombre et friable --- avec
l'aide des vers, des cloportes et d'\omterm{def:g1:living-or-not:living}{êtres vivants} trop petits pour
être vus. Au printemps suivant, cette terre nourrit les tomates.
Rien n'a été perdu~; les épluchures sont retournées dans le cycle de
la vie.
\end{example}

\begin{remark}[Les petites mains comptent]\label{rem:g2:caring-for-nature:count}
Fermer le robinet pendant le brossage des dents, écrire des deux
côtés de la \omterm{def:g1:plants-around-us:parts}{feuille}, le composteur, le déchet remporté~: chacun est
petit, et des millions de gens le font --- ou ne le font pas. La
\omterm{def:g2:caring-for-nature:environment}{nature} est protégée, ou abîmée, surtout par de petits choix
quotidiens comme les tiens.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:1}
Qu'est-ce que l'\omterm{def:g2:caring-for-nature:environment}{environnement} d'un \omterm{def:g1:living-or-not:living}{être vivant}~? Nomme trois choses
du tien qui viennent de la \omterm{def:g2:caring-for-nature:environment}{nature}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:2}
Qui peut perdre sa maison quand on éparpille un tronc pourri~?
Nomme trois habitants de \cref{ex:g2:caring-for-nature:home}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:3}
Donne quatre règles du visiteur de
\cref{met:g2:caring-for-nature:rules}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:4}
Pourquoi les déchets doivent-ils rentrer avec nous~? Réponds avec le
bouchon de \cref{ex:g2:caring-for-nature:cap}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:5}
Pourquoi ne faut-il pas cueillir toutes les \omterm{def:g1:plants-around-us:parts}{fleurs} d'une touffe~?
Donne les deux raisons de \cref{prop:g2:caring-for-nature:harm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:6}
Qu'arrive-t-il aux trognons de pomme dans un composteur~? Qui fait
le travail, et qu'obtient-on à la fin~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:7}
Donne deux façons d'aider la \omterm{def:g2:caring-for-nature:environment}{nature} exprès, d'après
\cref{ex:g2:caring-for-nature:help}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:8}
À toi d'essayer~: lors de ta prochaine promenade, compte les déchets
que tu vois (sans toucher aux dangereux). Quelle sorte est la plus
fréquente~? Où chaque sorte aurait-elle dû aller~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:caring-for-nature:9}
Au bord de la mare, Emma veut ramener trois \omterm{ex:g2:animals-grow-up:frog}{têtards} chez elle dans
un bocal «~pour les aider à grandir en sécurité~». À l'aide de ce
chapitre et de \cref{exo:g2:where-animals-live:10}, que lui
conseillerais-tu à la place~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:caring-for-nature:10}
«~Un bouchon, ce n'est rien.~» Sers-toi de
\cref{rem:g2:caring-for-nature:count} pour répondre~: que se
passe-t-il quand un million de visiteurs pensent tous la même
phrase --- dans le mauvais sens, et dans le bon~?
\end{exercise}
