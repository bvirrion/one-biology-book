\chapter{As provas da evolução}\label{ch:g9:evidence-of-evolution}

Duas vezes este livro parou à beira de uma grande afirmação:
os ramos da árvore de \omterm{prop:g6:classification-kinship:kinship}{parentesco} se encontram todos
(\cref{rem:g6:classification-kinship:implies}) e todo \omterm{def:g6:exploring-our-environment:components}{ser
vivo} lê um código só
(\cref{rem:g9:genes-and-dna:universal}). A afirmação para a
qual eles apontam tem um nome --- a \emph{evolução}: as
\omterm{def:g5:biodiversity:species}{espécies} mudam através das gerações, e todas descendem de
antepassados comuns. Antes que o próximo capítulo pergunte
\emph{como}, este faz o que o
\cref{exo:g6:classification-kinship:12} exigia: expõe as
provas.

\section{O testemunho das rochas}

\begin{definition}[Os fósseis]\label{def:g9:evidence-of-evolution:fossils}
Um \emph{fóssil}\index{fóssil} é um vestígio conservado de
vida do passado --- \omterm{def:g1:animals-around-us:covering}{concha}, \omterm{def:g3:bones-and-muscles:skeleton}{osso}, marca de \omterm{def:g1:plants-around-us:parts}{folha}, pegada ---
selado em rocha sedimentar. As camadas de rocha se empilham
em ordem de tempo, com as mais antigas embaixo; ler para cima
é ler a história. Os fósseis são a metade dos restos do
inventário da
\cref{def:g6:exploring-our-environment:components}, guardada
por milhões de anos.
\end{definition}

\begin{proposition}[O que o registro em camadas mostra]\label{prop:g9:evidence-of-evolution:record}
Lido em ordem, o registro testemunha três vezes:
\begin{enumerate}
  \item \emph{as faunas mudam}: cada idade de camada tem o
        elenco dela --- nenhum coelho entre os trilobitas,
        nenhum trilobita nas camadas dos coelhos; a maioria
        das \omterm{def:g5:biodiversity:species}{espécies} de toda camada profunda está
        \emph{\omterm{prop:g5:biodiversity:loss}{extinta}} (a palavra da
        \cref{prop:g5:biodiversity:loss}, agora com o tempo
        profundo por trás dela);
  \item \emph{a ordem é encaixada como a árvore}: os \omterm{prop:g3:sorting-living-things:groups}{peixes}
        aparecem nas camadas profundas, os parentes \omterm{prop:g3:sorting-living-things:groups}{anfíbios}
        de quatro \omterm{def:g1:my-body:parts}{membros} acima deles, os parentes dos
        \omterm{prop:g4:classifying-animals:classes}{répteis} acima destes, e os \omterm{prop:g3:sorting-living-things:groups}{mamíferos} e as \omterm{prop:g3:sorting-living-things:groups}{aves} mais
        acima ainda --- as caixas da \omterm{def:g4:classifying-animals:classification}{classificação}
        (\cref{pb:g6:classification-kinship:1}) se empilham
        no tempo exatamente como o \omterm{prop:g6:classification-kinship:kinship}{parentesco} prevê;
  \item \emph{existem intermediários}: a famosa laje do
        \emph{Archaeopteryx} --- \omterm{def:g1:animals-around-us:covering}{penas} e asas sobre uma
        armação de réptil com dentes e cauda de \omterm{def:g3:bones-and-muscles:skeleton}{ossos}
        compridos --- fica onde os ramos das \omterm{prop:g3:sorting-living-things:groups}{aves} e dos
        \omterm{prop:g4:classifying-animals:classes}{répteis} se separam; e baleias \omterm{def:g9:evidence-of-evolution:fossils}{fósseis} com patas
        traseiras funcionais ficam onde a ascendência de
        mamífero terrestre do
        \cref{ex:g6:classification-kinship:whale} as exigia.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[A série das baleias]\label{ex:g9:evidence-of-evolution:whales}
O dossiê da baleia, preenchido pelas rochas: \omterm{def:g9:evidence-of-evolution:fossils}{fósseis}
sucessivos, em ordem de tempo, vão de um mamífero de costa de
quatro patas do tamanho de um lobo, passando por formas de
\omterm{def:g1:my-body:parts}{pernas} encolhendo e caudas se fortalecendo, até nadadores
completos sem \omterm{def:g1:my-body:parts}{pernas} --- enquanto a baleia de hoje ainda
constrói ossinhos escondidos de quadril e de \omterm{def:g1:my-body:parts}{perna} (o
\cref{ex:g6:classification-kinship:placing} reparou nos
vestígios) e ainda amamenta, respira ar e cria pelo fetal.
Previsão, \omterm{def:g9:evidence-of-evolution:fossils}{fósseis}, vestígios: três chaves, uma fechadura.
\end{example}

\section{O testemunho dos corpos vivos}

\begin{proposition}[A anatomia se lembra]\label{prop:g9:evidence-of-evolution:anatomy}
Os corpos vivos carregam a história deles:
\begin{itemize}
  \item \emph{o plano compartilhado}: a asa do morcego, a
        nadadeira da baleia, a \omterm{def:g1:my-body:parts}{perna} do cavalo e o seu \omterm{def:g1:my-body:parts}{braço}
        são um mesmo arranjo de \omterm{def:g3:bones-and-muscles:skeleton}{ossos} (a bifurcação dos
        quatro \omterm{def:g1:my-body:parts}{membros} do
        \cref{pb:g6:classification-kinship:1}) esticado,
        encurtado e reempregado --- a herança explica o que
        uma engenharia nova jamais repetiria;
  \item \emph{os vestígios}: os quadris da baleia, as asas
        seladas do besouro que não voa, o seu próprio cóccix
        e o estado encolhido do apêndice --- partes mantidas
        porque a ascendência constrói por cima, e não do
        zero;
  \item \emph{os \omterm{def:g5:human-reproduction:pregnancy}{embriões} concordam}: os \omterm{def:g5:human-reproduction:pregnancy}{embriões}
        \omterm{def:g4:classifying-animals:vertebrate}{vertebrados} de peixe, de galinha e de humano correm
        impressionantemente parecidos pelas primeiras semanas
        deles (o assentamento do plano da
        \cref{prop:g8:from-egg-to-birth:timetable}) ---
        instruções de construção compartilhadas, divergindo
        tarde.
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[As moléculas votam]\label{prop:g9:evidence-of-evolution:molecules}
A testemunha mais nova é o código da
\cref{rem:g9:genes-and-dna:universal}. Compare a grafia de um
\omterm{def:g9:genes-and-dna:gene}{gene} bem conservado em várias \omterm{def:g5:biodiversity:species}{espécies}: o humano e o
chimpanzé diferem em poucas letras, o humano e o camundongo
em mais, o humano e o peixe em mais ainda, e o humano e a
levedura no máximo --- e as distâncias, somadas \omterm{def:g9:genes-and-dna:gene}{gene} após
\omterm{def:g9:genes-and-dna:gene}{gene}, redesenham a \emph{mesma árvore} que os anatomistas
desenharam a partir dos \omterm{def:g3:bones-and-muscles:skeleton}{ossos} e os paleontólogos a partir das
camadas. Três testemunhas independentes, uma árvore
genealógica só: essa concordância é a razão de os biólogos
terem parado de discutir o \emph{se} e passado ao
\emph{como}.
\end{proposition}
\admitted

\begin{example}[Vendo a mudança acontecer]\label{ex:g9:evidence-of-evolution:live}
A mudança até dá para ver onde as gerações se sucedem
depressa. A forma escura da mariposa-salpicada se espalhou
pelos bosques industriais manchados de fuligem e recuou
quando o ar limpou; \omterm{prop:g3:sorting-living-things:groups}{insetos} à prova de inseticida e \omterm{def:g6:producing-our-food:fermentation}{micróbios}
que desafiam os antibióticos (o
\cref{ch:g9:microbes-and-infection} vai encontrar o caso
médico) surgem em poucos anos; a distribuição do tamanho do
bico dos tentilhões das ilhas se desloca com a seca e com a
fartura, medida ao vivo por estudos de anilha e paquímetro.
Movimentos pequenos, tempos curtos --- o mecanismo pego em
funcionamento, seja qual for o alcance dele no longo prazo.
\end{example}

\begin{method}[Pesando provas históricas]\label{met:g9:evidence-of-evolution:weigh}
A evolução é história, e as afirmações históricas se testam
como o caso arquivado do \cref{pb:g9:unique-individuals:1}
--- pela convergência de testemunhas independentes:
\begin{enumerate}
  \item pergunte o que a afirmação \emph{prevê}: que
        intermediários, que vestígios, que distâncias
        moleculares;
  \item confira cada testemunha em separado: as rochas, os
        corpos, as moléculas, as observações ao vivo;
  \item pese a independência delas: testemunhas que não
        poderiam ter combinado a versão, ao concordar,
        multiplicam a credibilidade;
  \item mantenha o veredito aberto ao próximo achado --- e
        repare em quando os achados continuam caindo onde a
        afirmação mandou cavar (as \omterm{def:g1:my-body:parts}{pernas} das baleias foram
        procuradas, e encontradas).
\end{enumerate}
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:1}
O que é um \omterm{def:g9:evidence-of-evolution:fossils}{fóssil}, e por que as camadas de rocha se leem como
uma linha do tempo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:2}
Enuncie os três testemunhos do registro em camadas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:3}
Que mistura de \omterm{def:g9:heredity-and-traits:traits}{características} faz do \emph{Archaeopteryx} um
intermediário?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:4}
Dê três vestígios e diga o que cada um testemunha.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:5}
Como as comparações moleculares redesenham a árvore?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:6}
Nomeie duas observações ao vivo da mudança do
\cref{ex:g9:evidence-of-evolution:live}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:7}
Por que o plano compartilhado dos \omterm{def:g1:my-body:parts}{membros} defende a herança,
e não um bom projeto reaproveitado? (O que uma engenharia
nova teria liberdade de fazer?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:8}
Monte o dossiê completo da baleia: \omterm{def:g9:heredity-and-traits:traits}{características} de classe,
vestígios, série de \omterm{def:g9:evidence-of-evolution:fossils}{fósseis}, \omterm{prop:g6:classification-kinship:kinship}{parentesco} molecular --- uma
linha para cada um.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:9}
Por que a \emph{concordância} das rochas, dos corpos e das
moléculas vale mais do que qualquer testemunha isolada? Use o
passo da independência da
\cref{met:g9:evidence-of-evolution:weigh}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:10}
``Nenhum coelho entre os trilobitas.'' Por que essa ausência
--- sustentada em toda escavação já feita --- é ela mesma uma
prova forte?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:11}
A testemunha dos \omterm{def:g5:human-reproduction:pregnancy}{embriões}: o que concorda, quando diverge e a
maquinaria de qual capítulo diz por que as instruções
iniciais compartilhadas importam mais?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:evidence-of-evolution:12}
Escreva o argumento do capítulo como a
\cref{met:g9:evidence-of-evolution:weigh} aplicada por
inteiro: afirmação, previsões, quatro testemunhas, veredito
--- e a pergunta que o veredito entrega ao
\cref{ch:g9:how-species-change}.
\end{exercise}

\section{Problema: A ala nova do museu}

\begin{problem}[{Problema de fim de semana --- curando a
sala das provas}]\label{pb:g9:evidence-of-evolution:1}
O museu de história natural constrói uma ala da evolução, e a
turma cura quatro salas: Rochas, Corpos, Moléculas e
Observações ao Vivo.

\medskip\noindent\textbf{Parte I --- Sala por sala.}
\begin{enumerate}
  \item Rochas: ordene a parede --- laje de trilobita,
        primeiro peixe, pegadas de parentes \omterm{prop:g3:sorting-living-things:groups}{anfíbios},
        \omterm{def:g3:bones-and-muscles:skeleton}{esqueletos} de réptil, molde de \emph{Archaeopteryx},
        mandíbula de mamífero antigo --- e escreva a legenda
        de uma frase da sala a partir da
        \cref{prop:g9:evidence-of-evolution:record}.
  \item Corpos: a peça central abre asa de morcego, nadadeira
        de baleia, \omterm{def:g1:my-body:parts}{perna} de cavalo e \omterm{def:g1:my-body:parts}{braço} humano a partir de
        um suporte de \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto} só. Legende o plano
        compartilhado --- e acrescente as três peças do
        armário dos vestígios.
  \item Moléculas: o quadro de parede soma as diferenças de
        letras de um \omterm{def:g9:genes-and-dna:gene}{gene}: chimpanzé 1, camundongo 8, peixe
        24, levedura 41 (contagens ilustrativas). Com o que a
        árvore do quadro tem de combinar, para a sala
        funcionar?
  \item Observações ao Vivo: escolha duas peças do
        \cref{ex:g9:evidence-of-evolution:live} e diga o que
        cada uma mostra em movimento.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- As perguntas dos
visitantes.}
\begin{enumerate}[resume]
  \item ``Por que ainda existem macacos, se descendemos de
        antepassados compartilhados com eles?'' Responda com
        a ramificação da árvore --- primos, e não degraus
        (\cref{def:g6:classification-kinship:tree}).
  \item ``Onde estão os elos perdidos?'' Aponte os
        encontrados --- dois deles neste capítulo --- e diga
        entre o que cada um ficava.
  \item ``A evolução não é só uma teoria?'' Responda com o
        método das provas históricas: o que ``teoria''
        significa aqui, e o que a concordância das quatro
        salas estabelece.
  \item ``A escavação de amanhã não poderia derrubar tudo?''
        Diga que tipo de achado derrubaria --- coelhos entre
        trilobitas --- e o que a ausência dele, escavação
        após escavação, significou.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O painel de
encerramento da ala.}
\begin{enumerate}[resume]
  \item Redija o painel que liga a ala à sala da
        \omterm{def:g4:classifying-animals:classification}{classificação} ao lado: por que as antigas caixas
        encaixadas eram a árvore desde sempre.
  \item Acrescente o parágrafo das moléculas: o que o código
        compartilhado diz sobre o tronco.
  \item Acrescente o parágrafo honesto: o que a ala mostra
        (a mudança, o \omterm{prop:g6:classification-kinship:kinship}{parentesco}, os intermediários) e o que
        ela ainda não explica --- nomeando a pergunta do
        mecanismo que a sala seguinte vai assumir.
  \item Encerre com uma frase que um garoto de doze anos vá
        lembrar, contendo \emph{família} e \emph{tempo}.
\end{enumerate}
\end{problem}
