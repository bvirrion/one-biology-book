\chapter{Les bonnes habitudes}\label{ch:g1:healthy-habits}

Pourquoi tout le monde répète-t-il «~lave-toi les mains~», «~brosse-toi
les dents~», «~au lit~»~? Pas pour gâcher le plaisir. Ton corps est
l'\omterm{def:g1:living-or-not:living}{être vivant} que tu garderas le plus longtemps --- toute ta vie ---
et quelques petites habitudes, prises chaque jour, le gardent solide.
Voici lesquelles, et pourquoi elles marchent.

\section{Rester propre}

\begin{definition}[Les microbes]\label{def:g1:healthy-habits:germs}
Les \emph{microbes}\index{microbe} sont des \omterm{def:g1:living-or-not:living}{êtres vivants} bien trop
petits pour être vus, et ils sont partout --- sur les mains, sur les
poignées de porte, dans la terre. La plupart sont inoffensifs, mais
certains peuvent nous rendre malades quand ils entrent dans le
corps. Comme ils sont invisibles, des mains qui ont l'air propres
peuvent quand même en porter.
\end{definition}

\begin{definition}[Hygiène]\label{def:g1:healthy-habits:hygiene}
L'\emph{hygiène}\index{hygiène}, ce sont les habitudes qui gardent
le corps propre et qui tiennent les \omterm{def:g1:healthy-habits:germs}{microbes} dehors~: se laver les
mains, se laver le corps, se brosser les dents, garder les aliments
propres.
\end{definition}

\begin{method}[Bien se laver les mains]\label{met:g1:healthy-habits:hands}
\begin{enumerate}
  \item Mouille tes mains et prends du savon~;
  \item frotte partout --- les paumes, le dessus, entre les doigts,
        autour des pouces --- pendant que tu comptes lentement
        jusqu'à $20$ ou que tu chantes une petite chanson~;
  \item rince à l'eau courante~;
  \item sèche avec une serviette propre.
\end{enumerate}
Lave-toi les mains avant de manger, après les toilettes, et après
avoir joué dehors ou touché des \omterm{def:g1:animals-around-us:animal}{animaux}.
\end{method}

\begin{example}[L'expérience des paillettes]\label{ex:g1:healthy-habits:glitter}
Frotte une pincée de paillettes sur tes paumes, puis serre la main
d'un camarade, ouvre une porte, prends un crayon~: des paillettes
partout. Les \omterm{def:g1:healthy-habits:germs}{microbes} voyagent exactement de cette façon, mais sans
se voir. Lave-toi maintenant au savon en comptant jusqu'à $20$ ---
les paillettes s'en vont, et les \omterm{def:g1:healthy-habits:germs}{microbes} aussi.
\end{example}

\section{Des dents à garder}

\begin{proposition}[Pourquoi il faut se brosser les dents]\label{prop:g1:healthy-habits:teeth}
Après le repas, des morceaux d'aliments restent sur les dents. Les
\omterm{def:g1:healthy-habits:germs}{microbes} de la bouche se nourrissent de ces restes --- surtout des
plus sucrés --- et creusent lentement des trous dans les dents, qui
font mal. Le brossage balaie les restes avant que les \omterm{def:g1:healthy-habits:germs}{microbes} ne
festoient.
\end{proposition}
\admitted

\begin{method}[Me brosser les dents]\label{met:g1:healthy-habits:brush}
\begin{enumerate}
  \item Brosse après le petit déjeuner et avant le coucher --- tous
        les jours~;
  \item brosse chaque face de chaque dent~: le devant, le derrière
        et le dessus, là où tu mâches~;
  \item prends ton temps --- à peu près la durée d'une chanson~;
  \item change de brosse à dents quand les poils s'écartent.
\end{enumerate}
\end{method}

\begin{remark}[Les dents de lait comptent aussi]\label{rem:g1:healthy-habits:babyteeth}
«~Ces dents vont tomber de toute façon --- pourquoi les brosser~?~»
Parce qu'elles doivent encore mâcher pour toi pendant des années,
parce que les trous y font tout aussi mal, et parce que les dents
d'adulte attendent déjà dessous et méritent d'arriver dans une
bouche propre et saine.
\end{remark}

\section{Manger, bouger, dormir}

\begin{proposition}[Les trois besoins de chaque jour]\label{prop:g1:healthy-habits:needs}
Un corps qui grandit a besoin, chaque jour~:
\begin{enumerate}
  \item d'une \emph{alimentation variée}\index{alimentation variée}
        --- un peu de tout~: légumes et fruits, pain et céréales,
        lait et fromage, viande ou poisson ou œufs, et de l'eau à
        boire~; des sucreries de temps en temps seulement~;
  \item de \emph{mouvement}\index{mouvement} --- courir, grimper,
        faire du vélo, jouer dehors~: bouger rend les muscles, les
        os et le cœur plus forts~;
  \item de \emph{sommeil}\index{sommeil} --- la nuit, le corps se
        repose, se répare et grandit~; un enfant a besoin d'une
        longue nuit, bien plus longue que celle d'un adulte.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Une bonne journée pour le corps]\label{ex:g1:healthy-habits:day}
Un petit déjeuner avec des fruits~; les mains lavées avant le
déjeuner~; un après-midi à courir au parc~; les dents brossées après
le dîner~; la lumière éteinte tôt, avec une histoire. Rien
d'extraordinaire --- et c'est justement le point~: la santé est
faite de journées ordinaires comme celle-là.
\end{example}

\begin{remark}[La fatigue est un message]\label{rem:g1:healthy-habits:tired}
Bâiller, avoir les \omterm{def:g1:the-five-senses:sense}{yeux} qui piquent, être grognon sans raison~: c'est
le corps qui dit «~j'ai besoin de dormir~». L'écouter est aussi une
bonne habitude. Un cerveau qui a bien dormi apprend mieux, joue
mieux et rit plus facilement.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:1}
Qu'est-ce que les \omterm{def:g1:healthy-habits:germs}{microbes}~? Pourquoi des mains qui ont l'air
propres ont-elles quand même besoin d'être lavées~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:2}
Donne trois moments de la journée où il faut se laver les mains.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:3}
Dans \cref{met:g1:healthy-habits:hands}, combien de temps le
frottage au savon doit-il durer~? Nomme deux endroits faciles à
oublier.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:4}
Pourquoi les dents se creusent-elles quand on ne les brosse pas~?
Raconte l'histoire avec les \omterm{def:g1:healthy-habits:germs}{microbes} et les restes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:5}
Quand faut-il se brosser les dents, et combien de temps cela
doit-il durer~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:6}
Nomme les trois besoins quotidiens de
\cref{prop:g1:healthy-habits:needs}, et un exemple de chacun dans ta
propre journée.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:7}
Que fait le corps pendant le \omterm{prop:g1:healthy-habits:needs}{sommeil}~? Qui a besoin de la nuit la
plus longue, un enfant ou un adulte~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:8}
Donne deux signes qu'un corps réclame du \omterm{prop:g1:healthy-habits:needs}{sommeil}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:healthy-habits:9}
Sam dit~: «~Mes dents de lait vont tomber de toute façon, donc les
brosser ne sert à rien.~» Réponds-lui avec
\cref{rem:g1:healthy-habits:babyteeth} --- au moins deux raisons.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:healthy-habits:10}
Dans l'expérience des paillettes de
\cref{ex:g1:healthy-habits:glitter}, que représentent les
paillettes~? Que montre l'expérience sur le fait de se serrer la
main --- et sur le savon~?
\end{exercise}
