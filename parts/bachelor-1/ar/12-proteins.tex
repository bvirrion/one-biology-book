\chapter{الحموض الأمينية والبروتينات}\label{ch:b1:proteins}

اسلق بيضة فيصير بياضها الصافي معتمًا متماسكًا؛ ولم يُضف شيء ولم
يُنزع، لكن كل جزيء \omterm{def:b1:proteins:peptide}{بروتين} فيها انفك وتشابك مع جيرانه. وبرّدها فتبقى
مسلوقة. ومع ذلك فإن إنزيمًا صغيرًا يُفكّ برفق في أنبوب اختبار، ثم
يُترك في \omterm{def:b1:water-small-molecules:buffer}{هادئة} عادية، يعود فينطوي في دقائق إلى شكله العامل بالضبط،
بلا عون من شيء: فالشكل كان مكتوبًا في السلسلة. \omterm{def:b1:proteins:peptide}{والبروتينات} هي جزيئات
العمل في \omterm{def:b1:cell-unit-of-life:cell}{الخلية} --- إنزيماتها ومضخاتها ومحركاتها وسقالاتها وأضدادها
المناعية وهرموناتها --- وكل واحد منها سلسلة من الحموض الأمينية مطوية
في شكل يمليه تسلسلها. يصف هذا الفصل الحموض الأمينية العشرين، والرابطة
التي تصلها، ومستويات بنية \omterm{def:b1:proteins:peptide}{البروتين} الأربعة، والطي الذي يحول السلسلة
إلى آلة، والسلوك التعاوني \omterm{def:b1:proteins:peptide}{للبروتينات} ذوات الوحدات المتعددة، والطرق
التي تُفصل بها \omterm{def:b1:proteins:peptide}{البروتينات} وتُرى.

\section{الحموض الأمينية}

\begin{definition}[الحمض الأميني]\label{def:b1:proteins:aminoacid}
\emph{للحمض الأميني}\index{حمض أميني} \emph{كربون $\alpha$} مركزي
يحمل زمرة أمينية ($-\mathrm{NH_2}$)، وزمرة كربوكسيلية
($-\mathrm{COOH}$)، وهيدروجينًا، و\emph{سلسلة جانبية}\index{سلسلة جانبية}
$R$ تميز الحموض الأمينية العشرين المعيارية في \omterm{def:b1:proteins:peptide}{البروتينات}. وعند pH
\omterm{def:b1:cell-unit-of-life:cell}{الخلية} تكون الزمرة الأمينية مبرتنة والكربوكسيلية مشرَّدة: فالجزيء
\emph{شاردة ثنائية القطب}\index{شاردة ثنائية القطب}،
$\mathrm{^+H_3N{-}CHR{-}COO^-}$، بلا شحنة صافية. والكربون $\alpha$
لامتناظر عند الجميع إلا الغليسين، والأحياء لا تستعمل إلا المصاوغ L.
وتتجمع السلاسل الجانبية بحسب كيميائها:
\begin{center}
\footnotesize
\begin{tabular}{lll}
\hline
الصنف & الحموض الأمينية (بالرمزين الثلاثي والأحادي) & السلسلة الجانبية \\
\hline
غير قطبية أليفاتية & Gly G، Ala A، Val V، Leu L، Ile I، Met M، Pro P & هيدروكربونية؛ وPro حلقة \\
عطرية & Phe F، Tyr Y، Trp W & حلقات؛ وTyr وTrp تمتصان فوق البنفسجي \\
قطبية غير مشحونة & Ser S، Thr T، Cys C، Asn N، Gln Q & $-$OH، $-$SH، أميد \\
موجبة الشحنة & Lys K، Arg R، His H & أسس؛ وp$K_a$ عند His تساوي 6.0 \\
سالبة الشحنة & Asp D، Glu E & كربوكسيلات، وp$K_a$ لها 4 \\
\hline
\end{tabular}
\end{center}
وتسعة منها (His، وIle، وLeu، وLys، وMet، وPhe، وThr، وTrp، وVal)
\emph{أساسية} للإنسان: فنحن لا نستطيع صنعها ولا بد أن نأكلها.
\end{definition}

\begin{proposition}[التشرد]\label{prop:b1:proteins:ionisation}
لكل زمرة قابلة للتشرد في \omterm{def:b1:proteins:aminoacid}{الحمض الأميني} p$K_a$ خاصة بها: نحو 2
للكربوكسيل $\alpha$، و9.5 للأمين $\alpha$، وللسلاسل الجانبية 3.9
(Asp)، و4.1 (Glu)، و6.0 (His)، و8.3 (Cys)، و10.1 (Tyr)، و10.5
(Lys)، و12.5 (Arg). و\emph{نقطة التعادل الكهربائي}\index{نقطة التعادل الكهربائي}
p$I$ هي pH التي تنعدم عندها الشحنة الصافية --- وهي متوسط قيمتي
p$K_a$ اللتين تحيطان بالصورة المتعادلة؛ وp$I$ \omterm{def:b1:proteins:peptide}{لبروتين} يحددها محتواه
من السلاسل الجانبية المشحونة، وعند تلك pH يكون أقل ما يكون انحلالًا
ولا يتحرك في حقل كهربائي. والهيستيدين، وp$K_a$ له قريبة من 7، هو
البقية التي تغير شحنتها ضمن المدى الفيزيولوجي، وهو يقع في المواقع
الفعالة لكثير من الإنزيمات.
\end{proposition}

\section{الرابطة الببتيدية}

\begin{definition}[الرابطة الببتيدية وعديد الببتيد]\label{def:b1:proteins:peptide}
\emph{الرابطة الببتيدية}\index{رابطة ببتيدية} تصل كربوكسيل \omterm{def:b1:proteins:aminoacid}{حمض أميني}
بالزمرة الأمينية للتالي \omterm{prop:b1:water-small-molecules:families}{بالتكاثف} (وهي رابطة أميدية،
$\mathrm{-CO{-}NH-}$). وسلسلة من الحموض الأمينية موصولة هكذا هي
\emph{عديد ببتيد}\index{عديد ببتيد}: أي \emph{هيكل} من وحدات
$\mathrm{N{-}C_\alpha{-}C}$ متكررة تبرز منها السلاسل الجانبية،
ذات \emph{نهاية N} (زمرة أمينية حرة) و\emph{نهاية C} (كربوكسيل حر).
وتُكتب التسلسلات من N إلى C، وهو ترتيب التصنيع
(\cref{ch:b1:gene-expression}). و\emph{البروتين}\index{بروتين} عديد
ببتيد واحد أو أكثر مطوي في شكل محدد؛ والسلاسل النموذجية فيها
\qtyrange{100}{1000}{} بقية كتلتها الوسطية \qty{110}{Da}، فبروتين من
\qty{300}{} بقية كتلته \qty{33}{kDa}.
\end{definition}

\begin{proposition}[هندسة الرابطة الببتيدية]\label{prop:b1:proteins:geometry}
لرابطة C--N الببتيدية طابع رابطة مزدوجة جزئي (وهو رنين الأميد): فهي
قصيرة، ولا تستطيع الدوران، وتمسك الذرات الست
$\mathrm{C_\alpha{-}C(=O){-}N(H){-}C_\alpha}$ في مستو واحد، وكربوناها
$\alpha$ في وضع \emph{ترانس} دائمًا تقريبًا. فالهيكل إذن سلسلة من
مستويات صلبة مفصولة بمفاصل عند الكربونات $\alpha$، لكل مفصل دورانان،
$\phi$ (حول N--$\mathrm{C_\alpha}$) و$\psi$ (حول
$\mathrm{C_\alpha}$--C)، والتصادمات الفراغية بين السلاسل الجانبية
والهيكل تمنع معظم تراكيب $\phi$ و$\psi$. وتشكيل \omterm{def:b1:proteins:peptide}{البروتين} هو في جوهره
قائمة زواياه $\phi$ و$\psi$.
\end{proposition}

\section{البنية الثانوية}

\begin{definition}[البنية الثانوية]\label{def:b1:proteins:secondary}
\emph{البنية الثانوية}\index{بنية ثانوية} \omterm{def:b1:proteins:peptide}{لبروتين} هي الطي الموضعي
المنتظم لهيكله، وتمسكه \omterm{def:b1:water-small-molecules:water}{روابط هيدروجينية} بين زمر C=O وN--H في الهيكل.
ونمطان يسيطران. \emph{اللولب $\alpha$}\index{alpha@اللولب $\alpha$}: وهو
ملف يميني من 3.6 بقية في الدورة، يرتفع \qty{0.15}{nm} لكل بقية
(\qty{0.54}{nm} لكل دورة)، وكل زمرة C=O مرتبطة بزمرة N--H على بعد
أربع بقايا أمامها، والسلاسل الجانبية تشير إلى الخارج.
و\emph{الصفيحة $\beta$}\index{beta@الصفيحة $\beta$}: وهي سلاسل ممدودة
شبه تام (\qty{0.35}{nm} لكل بقية) موضوعة جنبًا إلى جنب، متوازية أو
متعاكسة، مرتبطة بين الخصلات المتجاورة، وسلاسلها الجانبية تتناوب فوق
الصفيحة وتحتها. وبينهما \emph{انعطافات} وعرى تعكس اتجاه السلسلة.
والبرولين، الذي تحبس حلقته $\phi$، يكسر اللوالب؛ والغليسين، ولا \omterm{def:b1:proteins:aminoacid}{سلسلة
جانبية} له، يتيح انعطافات لا تستطيعها بقية أخرى.
\end{definition}

\begin{example}[اللولب والصفيحة بالأرقام]\label{ex:b1:proteins:numbers}
اللولب $\alpha$ العابر للغشاء عليه أن يقطع \qty{3}{nm} من اللب
الكاره للماء: أي $3/0.15 = 20$ بقية، وبالفعل فإن المقاطع العابرة
للغشاء في \cref{ch:b1:membranes-transport} مسالك من نحو عشرين بقية
كارهة للماء. أما خصلة $\beta$ بالطول نفسه فتمتد \qty{7}{nm}: وفيبروين
الحرير صفائح متعاكسة مكدسة من الغليسين والألانين، وخيط الحرير أقوى من
فولاذ بوزنه لأن الهياكل التساهمية تمتد على امتداد الليف.
\end{example}

\section{البنية الثالثية والرابعية}

\begin{definition}[البنية الثالثية والرابعية]\label{def:b1:proteins:tertiary}
\emph{البنية الثالثية}\index{بنية ثالثية} هي الترتيب الثلاثي الأبعاد
الكامل \omterm{def:b1:proteins:peptide}{لعديد ببتيد} واحد: لوالبه وصفائحه وعراه مرصوصة معًا، وكثيرًا في
عدة \emph{نطاقات} مضغوطة تنطوي كل منها على حدة. وتمسكها قوى غير
تساهمية --- \emph{\omterm{prop:b1:water-small-molecules:properties}{الأثر الكاره للماء}} الذي يدفن السلاسل الجانبية غير
القطبية في لب بعيد عن الماء، \omterm{def:b1:water-small-molecules:water}{والروابط الهيدروجينية}، والجسور الملحية
بين السلاسل الجانبية المشحونة، وارتصاص \omterm{def:b1:water-small-molecules:bonds}{فان دير فالس} --- وأحيانًا
\emph{جسور ثنائية الكبريت}\index{جسر ثنائي الكبريت}، وهي روابط S--S
تساهمية بين سيستئينين، في \omterm{def:b1:proteins:peptide}{البروتينات} المفرزة إلى الخارج المؤكسد.
و\emph{البنية الرابعية}\index{بنية رابعية} هي تركيب عدة عديدات ببتيد
(وحدات) في \omterm{def:b1:proteins:peptide}{بروتين} واحد: فالهيموغلوبين سلسلتا $\alpha$ وسلسلتا
$\beta$. \omterm{def:b1:proteins:peptide}{والبروتينات} \emph{الكروية} مضغوطة منحلة في الماء (الإنزيمات،
\omterm{def:b1:membranes-transport:transporters}{والنواقل})؛ \omterm{def:b1:proteins:peptide}{والبروتينات} \emph{الليفية} ممدودة غير منحلة (اللولب
الثلاثي \omterm{def:b1:body-plans-tissues:connective}{للكولاجين}، والملفات الملتفة في الكيراتين والميوزين، والحرير).
\end{definition}

\begin{theorem}[التسلسل يحدد الطية]\label{thm:b1:proteins:anfinsen}
\index{طي البروتين}
تحدد بنيةَ \omterm{def:b1:proteins:peptide}{البروتين} الثلاثية الأبعاد تسلسلُ حموضه الأمينية: فالطية
الأصلية هي التشكيل الأدنى طاقةً حرة الذي تستطيع السلسلة بلوغه في
محيطها الخلوي، والمعلومة اللازمة لبلوغه محتواة في السلسلة نفسها.
\end{theorem}
\begin{proof}[شواهد]
فكّ أنفنسن (1961) الريبونوكلياز A، وهو إنزيم من \qty{124}{} بقية له
أربعة \omterm{def:b1:proteins:tertiary}{جسور ثنائية الكبريت}، في يوريا مركزة مع عامل مرجع كسر الجسور:
ففقدت السلسلة كل نشاطها. ثم أزال اليوريا وترك السيستئينات تتأكسد
ببطء في الهواء، فاسترد الإنزيم كامل النشاط، وقد أُعيد تكوين جسوره
الأربعة بين السيستئينات الثمانية نفسها من بين \qty{105}{} اقترانات
ممكنة --- فالسلسلة وجدت طيتها بنفسها. وحين أُعيدت أكسدتها وهي ما
تزال في اليوريا كوّنت جسورًا عشوائية وبقيت خاملة؛ ثم أتاح لها أثر من
العامل المرجع أن تخلطها حتى بلغت المجموعة الأصلية، وهي الأكثر
استقرارًا. ومنذ ذلك الحين طُويت آلاف التسلسلات من الصفر في الأنبوب،
وصار يمكن حساب طية تسلسل جديد منه.
\end{proof}

\begin{proposition}[المسخ]\label{prop:b1:proteins:denaturation}
\index{مسخ البروتين}
الحرارة، وpH المتطرفة، واليوريا، والمنظفات، والمذيبات العضوية تفك
\omterm{def:b1:proteins:peptide}{البروتينات} بالتغلب على الروابط الضعيفة التي تمسك الطية: فتفقد
السلسلة شكلها ووظيفتها، وتكشف لبها الكاره للماء، وتتكتل مع جيرانها.
\omterm{def:b1:proteins:peptide}{والبروتينات} الصغيرة تعيد الانطواء حين يُزال الماسخ؛ أما الكبيرة،
\omterm{def:b1:proteins:peptide}{والبروتينات} في \omterm{def:b1:cell-unit-of-life:cell}{خلية} مزدحمة، فكثيرًا ما لا تستطيع، \omterm{def:b1:cell-unit-of-life:cell}{والخلية} تحتفظ
\emph{ببروتينات مرافقة}\index{بروتين مرافق} --- تربط السلاسل المفكوكة،
وتحجب سطوحها الكارهة للماء، وتمنحها فرصًا متكررة للانطواء انطواءً
صحيحًا --- وتهدم ما بقي مسيء الطي. والحمى ببضع درجات، و\qty{42}{\celsius}
التي تبدأ عندها \omterm{def:b1:proteins:peptide}{بروتينات} الدماغ بالانفكاك، تحددان هامش الاستقرار
الضيق: فالطية النموذجية أكثر استقرارًا من السلسلة المفكوكة
بمقدار \qtyrange{20}{60}{kJ/mol} فقط، أي بقيمة بضع \omterm{def:b1:water-small-molecules:water}{روابط هيدروجينية}.
\end{proposition}

\section{الألوستيرية: الهيموغلوبين}

\begin{definition}[الألوستيرية والتعاونية]\label{def:b1:proteins:allostery}
يكون \omterm{def:b1:proteins:peptide}{البروتين} \emph{ألوستيريًا}\index{ألوستيرية} حين يغير ارتباط
ربيطة في موقع تشكيلَ \omterm{def:b1:proteins:peptide}{البروتين} ومن ثم ألفته في موقع آخر. وحين تتشابه
المواقع ويرفع التغيرُ ألفةَ الباقي منها يكون الارتباط
\emph{تعاونيًا}\index{تعاونية}: فيكون منحني الإشباع سينيًا لا زائديًا،
وينتقل \omterm{def:b1:proteins:peptide}{البروتين} من شبه فارغ إلى شبه ممتلئ على مدى ضيق من تركيز
الربيطة. و\emph{معادلة هيل}\index{معادلة هيل} تصف ذلك:
\[
Y = \frac{P^n}{P_{50}^n + P^n},
\]
حيث $Y$ كسر المواقع المشغولة، و$P$ تركيز الربيطة (أو ضغطها الجزئي)،
و$P_{50}$ القيمة عند نصف الإشباع، و$n$ معامل هيل --- وهو 1 عند
المواقع المستقلة، ويبلغ عدد المواقع عند التعاون التام.
\end{definition}

\begin{proposition}[الهيموغلوبين والميوغلوبين]\label{prop:b1:proteins:hemoglobin}
الميوغلوبين، وهو سلسلة واحدة بهيم واحد، يرتبط بجزيء $\mathrm{O_2}$ وفق
زائدي ($n = 1$، و$P_{50} = \qty{0.37}{kPa}$): فهو مشبع عند أي ضغط
تقدمه \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة} ويعمل مخزنًا في العضلة. أما الهيموغلوبين، وهو أربع
وحدات، فيرتبط ارتباطًا \omterm{def:b1:proteins:allostery}{تعاونيًا} ($n \approx 2.8$،
و$P_{50} = \qty{3.5}{kPa}$): فأول $\mathrm{O_2}$ يرتبط يزيح الرباعي
من \emph{حالة T} ضعيفة الألفة نحو \emph{حالة R} عالية الألفة، فيتحمل
شحنته كاملة تقريبًا عند \qty{13}{kPa} في الرئة ويفرغ كسرًا كبيرًا عند
\qty{5}{kPa} في \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}. وتخفض ألفته أكثر، ويُيسَّر التفريغ، \omterm{def:b1:water-small-molecules:ph}{بالحمض}
وبغاز $\mathrm{CO_2}$ (وهو \emph{أثر بور}) وببسفوغليسيرات 2،3 (BPG)،
التي ترتبط بالحالة T: ففي عضلة عاملة، أو على ارتفاع، يُطلق أكسجين
أكثر عند الضغط نفسه. والجنين يصنع هيموغلوبينًا يرتبط بمركب BPG ارتباطًا
ضعيفًا، فيسحب دمه الأكسجين من دم أمه عبر المشيمة.
\end{proposition}

\begin{method}[استعمال معادلة هيل]\label{met:b1:proteins:hill}
\begin{enumerate}
  \item اقرأ $P_{50}$ من المنحني عند $Y = 0.5$؛ وقدّر $n$ من ميل
        $\log[Y/(1-Y)]$ بدلالة $\log P$ (وهو مخطط هيل)، وهو مستقيم
        ميله $n$ قرب $P_{50}$.
  \item احسب $Y$ عند ضغط التحميل (الرئتان) وعند ضغط التفريغ
        (\omterm{def:b1:body-plans-tissues:tissue}{الأنسجة})؛ والفرق هو الكسر المسلَّم من السعة الحاملة.
  \item اضرب في السعة: فالدم يحمل \qty{150}{g/L} من الهيموغلوبين،
        ويرتبط كل غرام بمقدار \qty{1.34}{mL} من $\mathrm{O_2}$، أي
        \qty{200}{mL} من $\mathrm{O_2}$ في اللتر عند الإشباع.
  \item ولإدخال أثر بور أو BPG، استعمل $P_{50}$ المزاحة عند \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}
        وحدها، حيث \omterm{def:b1:water-small-molecules:ph}{الحمض}.
\end{enumerate}
\end{method}

\section{رؤية البروتينات وفصلها}

\begin{method}[فصل البروتينات]\label{met:b1:proteins:methods}
\begin{enumerate}
  \item \emph{الرحلان الكهربائي في SDS}: يغلف المنظف كل سلسلة بشحنة
        سالبة تتناسب مع طولها، فتهاجر السلاسل عبر هلام البولي
        أكريلاميد بالقياس وحده؛ ويعاير الهلامَ سلمٌ من كتل معلومة،
        وموضع الحزمة المصبوغة يعطي الكتلة في حدود بضعة في المئة.
  \item \emph{الاستشراب}: عبر عمود من حبيبات تؤخر \omterm{def:b1:proteins:peptide}{البروتينات} بالقياس
        (الترشيح الهلامي: فتخرج الكبيرة أولًا)، أو بالشحنة (التبادل
        الشاردي)، أو بالارتباط النوعي (الألفة: فترتبط ربيطة مثبتة
        \omterm{def:b1:proteins:peptide}{ببروتين} واحد وتدع الباقي يمر). ويُتابع التنقية ارتفاعُ النشاط
        النوعي (\cref{ch:b1:cell-unit-of-life}).
  \item \emph{تحديد التسلسل}: تسلسل المورّثة، مترجمًا؛ أو \omterm{def:b1:proteins:peptide}{البروتين}
        نفسه، مقطّعًا إلى ببتيدات ومقروءًا بمطيافية الكتلة.
  \item \emph{البنية}: انعراج الأشعة السينية على بلورة، أو المجهرية
        الإلكترونية لجسيمات مفردة مجمدة، يعطي موضع كل ذرة؛ ورسوم
        الشرائط في هذا الفصل ملخصات لها.
\end{enumerate}
\end{method}

\section{مسألة: الهيموغلوبين في العمل}

\begin{problem}[{مسألة نهاية الأسبوع --- الأكسجين محمولًا من الرئة إلى
العضلة، في الراحة، وفي عدو سريع، وعلى جبل، وقبل الولادة، وصولًا إلى
الأكسجين المسلَّم لكل لتر من الدم}]
\label{pb:b1:proteins:1}
يحمل الدم \qty{150}{g/L} من الهيموغلوبين؛ ويرتبط \qty{1}{g} منه
بمقدار \qty{1.34}{mL} من $\mathrm{O_2}$. والهيموغلوبين يطيع \omterm{def:b1:proteins:allostery}{معادلة هيل}
عند $n = 2.8$ و$P_{50} = \qty{3.5}{kPa}$ وpH 7.4؛ وللميوغلوبين
$n = 1$ و$P_{50} = \qty{0.37}{kPa}$. والضغوط الجزئية
لغاز $\mathrm{O_2}$: \qty{13.3}{kPa} في الرئتين عند مستوى سطح البحر،
و\qty{5.3}{kPa} في \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة} المستريحة. خذ $2.8\ln 3.5 = 3.51$،
و$2.8\ln 5.3 = 4.67$، و$2.8\ln 13.3 = 7.25$، و$2.8\ln 2.7 = 2.78$،
و$2.8\ln 4.5 = 4.21$، و$2.8\ln 7 = 5.45$، و$2.8\ln 4 = 3.88$،
و$2.8\ln 2.6 = 2.68$.

\medskip
\noindent\textbf{الجزء الأول --- في الراحة.}
\begin{enumerate}
  \item احسب السعة الأكسجينية للتر واحد من الدم.
  \item احسب $P^{2.8}$ من أجل $P = 3.5$ و$5.3$ و$13.3$ كيلوباسكال
        (بوصفها $e^{2.8\ln P}$).
  \item احسب إشباع الهيموغلوبين $Y$ في الرئتين وفي \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}.
  \item احسب الكسر المسلَّم من الحمولة والأكسجين المسلَّم لكل لتر من
        الدم، بالمليلتر.
  \item إنسان مستريح يستهلك \qty{250}{mL} من $\mathrm{O_2}$ في
        الدقيقة. فأي صبيب قلبي يقتضيه هذا التسليم؟
  \item احسب إشباع الميوغلوبين عند الضغطين نفسيهما والأكسجين الذي كان
        ليسلّمه لكل لتر لو حل محل الهيموغلوبين. واستنتج.
  \item فسّر في جملة واحدة ماذا تشتري \omterm{def:b1:proteins:allostery}{التعاونية}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- عدو سريع.}
في عضلة تعدو تهبط pH إلى 7.2 ويهبط الضغط الموضعي
لغاز $\mathrm{O_2}$ إلى \qty{2.7}{kPa}؛ وأثر بور يرفع $P_{50}$ هناك إلى
\qty{4.5}{kPa}.
\begin{enumerate}[resume]
  \item احسب $Y$ عند \qty{2.7}{kPa} مع $P_{50} = \qty{3.5}{kPa}$
        (بلا أثر بور).
  \item احسب $Y$ عند \qty{2.7}{kPa} مع $P_{50} = \qty{4.5}{kPa}$.
  \item احسب الأكسجين المسلَّم لكل لتر في الحالتين (والتحميل في
        الرئتين على حاله) والكسب من أثر بور.
  \item ميوغلوبين العضلة عند $P = \qty{2.7}{kPa}$. فما إشباعه، وماذا
        يحل به إذا هبط الضغط إلى \qty{0.5}{kPa} في أثناء تقلص؟
  \item فسّر لماذا كان أثر بور صورة من \omterm{def:b1:proteins:allostery}{الألوستيرية} وأين تعمل
        البروتونات.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- جبل.}
عند \qty{4000}{m} يكون ضغط $\mathrm{O_2}$ في الرئتين \qty{7}{kPa}.
وفي غضون أيام ترفع الكريات الحمراء BPG فيها، فتزيح $P_{50}$ إلى
\qty{4.0}{kPa}؛ وفي غضون أسابيع يرفع النقي الهيموغلوبين إلى
\qty{180}{g/L}.
\begin{enumerate}[resume]
  \item احسب الإشباع في الرئتين عند \qty{7}{kPa} مع
        $P_{50} = \qty{3.5}{kPa}$، والتسليم لكل لتر (\omterm{def:b1:body-plans-tissues:tissue}{والأنسجة} عند
        \qty{5.3}{kPa}). وقارن مع مستوى سطح البحر.
  \item احسب إشباعي الرئة \omterm{def:b1:body-plans-tissues:tissue}{والنسيج} مع $P_{50} = \qty{4.0}{kPa}$،
        والتسليم.
  \item فسّر لماذا يساعد خفض الألفة مع أنه يخفض التحميل في الرئتين.
  \item احسب التسليم لكل لتر مع $P_{50} = \qty{4.0}{kPa}$
        و\qty{180}{g/L} من الهيموغلوبين.
  \item هل تساعد ألفة أدنى بعد ($P_{50} = \qty{6}{kPa}$) عند
        \qty{7}{kPa}؟ احسب الإشباعين واحسم.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- قبل الولادة.}
لهيموغلوبين الجنين $P_{50} = \qty{2.6}{kPa}$؛ وفي المشيمة يكون ضغط
$\mathrm{O_2}$ \qty{4}{kPa} على الجهتين.
\begin{enumerate}[resume]
  \item احسب إشباع هيموغلوبين الأم عند \qty{4}{kPa}.
  \item احسب إشباع هيموغلوبين الجنين عند \qty{4}{kPa}.
  \item فسّر كيف ينتقل الأكسجين من الأم إلى الجنين مع أن الضغط واحد
        على الجهتين.
  \item لهيموغلوبين الجنين سلاسل $\gamma$ بدل $\beta$، وهي ترتبط
        بمركب BPG ارتباطًا ضعيفًا. فسّر كيف ينتج هذا $P_{50}$ الأدنى.
  \item عند الولادة يجب أن يفرغ دم الجنين في \omterm{def:b1:body-plans-tissues:tissue}{أنسجة} عند
        \qty{5.3}{kPa}. احسب التسليم الجنيني لكل لتر من رئتين عند
        \qty{13.3}{kPa}، وقل لماذا كان الانتقال إلى هيموغلوبين البالغ
        في الأشهر الأولى نافعًا.
  \item طفرة ترفع $P_{50}$ عند البالغ إلى \qty{5}{kPa}. تنبأ بإشباع
        الشخص الشرياني عند مستوى سطح البحر، وبالنتيجة في التسليم.
  \item فسّر لماذا كان حامل عند $P_{50} = \qty{3.5}{kPa}$ وبلا \omterm{def:b1:proteins:allostery}{تعاونية}
        ($n = 1$) حاملًا رديئًا للأكسجين: احسب تسليمه.
  \item اذكر النتيجة: الأكسجين المسلَّم لكل لتر من الدم في الراحة وفي
        عضلة تعدو، وخاصتا الهيموغلوبين اللتان تصنعان الفرق.
\end{enumerate}
\end{problem}
