\section*{Bab \ref{ch:g8:reproductive-systems} --- Sistem Reproduksi}

\begin{solution}{exo:g8:reproductive-systems:1}
Buah zakarnya, di dalam skrotum: produksi sperma tanpa henti
sejak \omterm{def:g8:puberty:puberty}{pubertas}. Saluran dan kelenjarnya: penyimpanan,
pengangkutan, dan cairan yang membentuk air mani. Penisnya:
pintu keluar air mani --- dan \omterm{def:g7:removing-wastes:kidneys}{urine}, pada tugasnya yang
terpisah dan dijaga \omterm{def:g7:heart-and-circulation:rooms}{katup}.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:2}
\omterm{def:g8:reproductive-systems:female}{Indung telurnya}: persediaan \omterm{def:g4:animal-reproduction:sexual}{sel telur} sejak lahir, melepas
satu tiap daur sejak \omterm{def:g8:puberty:puberty}{pubertas}. \omterm{def:g8:reproductive-systems:female}{Saluran telurnya}: menangkap
\omterm{def:g2:animals-grow-up:born}{telur} yang dilepaskan --- tempat \omterm{def:g5:flower-to-fruit:steps}{pembuahan} yang biasa.
\omterm{prop:g5:human-reproduction:plan}{Rahimnya}: penginapan berotot yang lapisan dalamnya
diperbarui. Vaginanya: penerimaan air mani, dan jalan lahir.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:3}
\omterm{def:g8:reproductive-systems:male}{Sel sperma}: yang terkecil di dalam tubuh --- kepala yang
padat \omterm{def:g6:cell-unit-of-life:parts}{inti sel} dan ekor perenang, berbekal ringan, dibuat
tanpa henti dalam ratusan juta. \omterm{def:g4:animal-reproduction:sexual}{Sel telur}: yang terbesar di
dalam tubuh, bulat dan tak bergerak, berbekal kaya,
dilepaskan satu sebulan.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:4}
Sebuah \omterm{def:g8:reproductive-systems:female}{indung telur} melepaskan satu \omterm{def:g4:animal-reproduction:sexual}{sel telur}, ditangkap
oleh \omterm{def:g8:reproductive-systems:female}{saluran telurnya} --- sekitar hari ke-14. Ia dapat
dibuahi selama kira-kira sehari.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:5}
Luruh dan lepasnya lapisan dalam \omterm{prop:g5:human-reproduction:plan}{rahim} yang sudah
disiapkan, keluar selama beberapa hari. Kedatangannya
berkata bahwa daur itu berakhir tanpa \omterm{def:g5:flower-to-fruit:steps}{pembuahan}.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:6}
\omterm{def:g5:flower-to-fruit:steps}{Pembuahan}. \omterm{prop:g8:sexual-reproduction:core}{Zigotnya} menetap di dalam lapisan yang sudah
disiapkan, daurnya tertunda, dan \omterm{def:g5:human-reproduction:pregnancy}{kehamilan} dimulai.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:7}
Langkah 1, \omterm{def:g8:sexual-reproduction:gametes}{gametnya}: buah zakar dan \omterm{def:g8:reproductive-systems:female}{indung telur}. Langkah 2,
\omterm{def:g5:flower-to-fruit:steps}{pembuahannya}: di dalam \omterm{def:g8:reproductive-systems:female}{saluran telur}, dicapai lewat renang
spermanya dari vagina menembus \omterm{prop:g5:human-reproduction:plan}{rahim}. Langkah 3,
perkembangan \omterm{prop:g8:sexual-reproduction:core}{zigot} yang membelah: menginap di dalam lapisan
\omterm{prop:g5:human-reproduction:plan}{rahim} yang sudah disiapkan.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:8}
Sisi laki-lakinya menjalankan logika banyak-dan-murah:
jutaan perenang tanpa henti, masing-masing dapat
dikorbankan. Sisi perempuannya menjalankan
sedikit-dan-berbekal: satu \omterm{def:g4:animal-reproduction:sexual}{sel telur} berbekal sebulan, dan
di belakangnya seluruh penanaman modal penginapan \omterm{prop:g5:human-reproduction:plan}{rahimnya}.
Garis siasatnya, tertulis di dalam \omterm{def:g5:first-look-at-cells:cell}{sel}.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:9}
Permesinannya sedang menyetel diri pada tahun-tahun
pertamanya --- rantai \omterm{def:g8:puberty:hormones}{hormonnya} sedang mendapatkan iramanya
--- sehingga ketidakteraturannya memang diharapkan; dan
panjang daurnya sungguh berubah-ubah, dari orang ke orang
dan dari bulan ke bulan. Keduanya adalah sistem yang berjalan
dengan wajar.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:10}
\omterm{prop:g5:human-reproduction:plan}{Rahimnya} adalah \omterm{def:g3:bones-and-muscles:muscle}{otot}, dan peluruhannya adalah kerja \omterm{def:g3:bones-and-muscles:muscle}{otot} ---
kontraksinya terasa sebagai kram. Kehangatan dan gerak
menolong (dan perawat sekolahnya punya nasihat bila keduanya
belum cukup).
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:11}
Tugas spermanya adalah perjalanannya: murah, berenang, dan
dibutuhkan dalam jutaan karena hampir tidak ada yang tiba
--- sehingga pabriknya berjalan tanpa henti. Tugas \omterm{def:g2:animals-grow-up:born}{telurnya}
adalah menjadi bekal pertama \omterm{prop:g8:sexual-reproduction:core}{zigotnya}: mahal, disimpan sejak
lahir, dan dilepaskan dengan persiapan sebulan di
belakangnya --- sehingga pabriknya adalah sebuah
perbendaharaan, yang membelanjakan satu demi satu.
\end{solution}

\begin{solution}{exo:g8:reproductive-systems:12}
Peristiwa 1 adalah pertanyaannya sedang disiapkan: lapisan
dalamnya dibangun ulang, penginapannya disiapkan. Peristiwa 2
mengajukannya: satu \omterm{def:g4:animal-reproduction:sexual}{sel telur} dilepaskan --- akankah
\omterm{def:g5:flower-to-fruit:steps}{pembuahan} datang? Peristiwa 3 adalah jawaban ``tidak'':
lapisan yang tak dibutuhkan itu luruh, dan rodanya bertanya
lagi bulan depan. Peristiwa 4 adalah jawaban ``ya'':
\omterm{prop:g8:sexual-reproduction:core}{zigotnya} menetap, pertanyaannya berhenti, dan penginapannya
melayani maksudnya selama sembilan bulan.
\end{solution}

\begin{solution}{pb:g8:reproductive-systems:1}
\textbf{1.} Dari buah zakarnya --- pabriknya, yang disimpan
di dalam skrotum --- spermanya lewat ke dalam salurannya
untuk disimpan lalu diangkut, disertai cairan kelenjarnya
sehingga menjadi air mani, lalu keluar lewat penisnya; organ
yang sama mengangkut \omterm{def:g7:removing-wastes:kidneys}{urine} pada waktu yang lain, dan sebuah
susunan \omterm{def:g7:heart-and-circulation:rooms}{katup} menjaga kedua tugas itu tetap terpisah tegas.

\textbf{2.} Persediaannya menunggu di dalam \omterm{def:g8:reproductive-systems:female}{indung telurnya},
sejak lahir; \omterm{def:g5:flower-to-fruit:steps}{pembuahan} biasanya terjadi di dalam \omterm{def:g8:reproductive-systems:female}{saluran
telurnya}; sebuah \omterm{def:g5:human-reproduction:pregnancy}{kehamilan} menginap di dalam lapisan \omterm{prop:g5:human-reproduction:plan}{rahim}
yang sudah disiapkan.

\textbf{3.} \omterm{def:g8:sexual-reproduction:gametes}{Gametnya} dibangun menurut siasat yang
berlawanan: jutaan perenang murah melawan satu \omterm{def:g2:animals-grow-up:born}{telur} berbekal
sebulan sekali --- banyak-melawan-rawatan, tertulis di dalam
\omterm{def:g5:first-look-at-cells:cell}{selnya} sendiri.

\textbf{4.} Menunggu perintahnya: kedua sistemnya berdiri
terbangun tetapi menganggur sampai \omterm{def:g8:puberty:hormones}{hormon} pusat kendali di
dasar \omterm{def:g5:brain-in-command:system}{otaknya} menyalakannya pada masa \omterm{def:g8:puberty:puberty}{pubertas}.

\textbf{5.} Hari 1--5 kira-kira: \omterm{prop:g8:reproductive-systems:cycle}{haidnya}. Sampai pertengahan
daurnya: lapisan dalamnya dibangun ulang. Sekitar hari ke-14:
\omterm{prop:g8:reproductive-systems:cycle}{ovulasi}, dengan hari suburnya di sekitarnya. Lalu penantian
yang siap --- sampai hari ke-28 dan putaran rodanya berikutnya.

\textbf{6.} Panjang daurnya berubah-ubah dari orang ke
orang, dan dari bulan ke bulan pada satu orang --- dan
tahun-tahun pertamanya berjalan tidak teratur selagi
permesinannya menyetel diri. Penanggalannya mengajarkan
polanya, bukan tanggal persis siapa pun.

\textbf{7.} Dibatalkan: luruhnya lapisan dalamnya dan \omterm{prop:g8:reproductive-systems:cycle}{haid}
berikutnya --- rodanya berhenti. Dimulai: menetapnya \omterm{prop:g8:sexual-reproduction:core}{zigotnya}
di dalam lapisan itu, dan sembilan bulan \omterm{def:g5:human-reproduction:pregnancy}{kehamilannya}.

\textbf{8.} Kram adalah \omterm{prop:g5:human-reproduction:plan}{rahim} berotot yang bekerja selagi ia
meluruh --- wajar, bukan kerusakan; kehangatan dan gerak
menolong, dan apa pun yang lebih kuat punya alamatnya di
pintu perawat atau dokternya.

\textbf{9.} Misalnya: ``Ini organ seperti \omterm{def:g4:heart-and-blood:heart}{jantung} dan
\omterm{def:g7:how-animals-breathe:four}{paru-paru} --- setiap orang di ruangan ini memiliki satu
perangkat atau yang lainnya, dan inilah babnya.''

\textbf{10.} Pabriknya: buah zakar --- \omterm{def:g8:reproductive-systems:female}{indung telur}.
\omterm{def:g8:sexual-reproduction:gametes}{Gametnya}: jutaan perenang tanpa henti --- satu \omterm{def:g2:animals-grow-up:born}{telur} berbekal
sebulan sekali. Pertemuannya: \omterm{def:g8:reproductive-systems:female}{saluran telurnya}. Penginapannya:
tidak ada --- \omterm{prop:g5:human-reproduction:plan}{rahim} yang disiapkan tiap bulan. Jadwalnya:
tanpa henti sejak \omterm{def:g8:puberty:puberty}{pubertas} --- berdaur, dari \omterm{def:g8:puberty:puberty}{pubertas} sampai
kira-kira lima puluh tahun.

\textbf{11.} Langkah 1 pada kedua modelnya: pabriknya membuat
\omterm{def:g8:sexual-reproduction:gametes}{gametnya}. Langkah 2 pada model perempuannya: bergabungnya
keduanya di dalam \omterm{def:g8:reproductive-systems:female}{saluran telur}. Langkah 3 melintasi model
dan penanggalannya: \omterm{prop:g8:sexual-reproduction:core}{zigotnya} menginap di dalam lapisan yang
disiapkan pada penanggalannya, rodanya berhenti,
perkembangannya dimulai.

\textbf{12.} ``Karena tubuh berjalan lebih baik dengan
pengetahuan daripada dengan dongeng --- dan segala yang
diajarkan hari ini punya alamat yang terlatih dan menjaga
rahasia bagi pertanyaan lanjutannya.''
\end{solution}
