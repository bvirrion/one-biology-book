\section*{Chapter \ref{ch:g10:cells-common-unit} --- Cells: The Common Unit of Life}

\begin{solution}{exo:g10:cells-common-unit:1}
Every living thing is made of one or more \omterm{def:g10:cells-common-unit:cell}{cells}; the \omterm{def:g10:cells-common-unit:cell}{cell} is the basic
unit of structure and function; every \omterm{def:g10:cells-common-unit:cell}{cell} comes from a pre-existing
\omterm{def:g10:cells-common-unit:cell}{cell} by division.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:2}
\qty{50}{\micro m}; \qty{2.5}{\micro m}; \qty{0.007}{mm}.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:3}
$36/600 = \qty{0.06}{mm} = \qty{60}{\micro m}$.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:4}
Plant only: \omterm{def:g10:cells-common-unit:organelle}{cell wall}, \omterm{def:g10:cells-common-unit:organelle}{chloroplasts}, large central \omterm{def:g10:cells-common-unit:organelle}{vacuole}. Every \omterm{def:g10:cells-common-unit:cell}{cell}:
a \omterm{def:g10:cells-common-unit:cell}{plasma membrane} (also \omterm{def:g10:cells-common-unit:cell}{cytoplasm} and \omterm{def:g10:cells-common-unit:organelle}{ribosomes}).
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:5}
A \omterm{def:g10:cells-common-unit:prokaryote}{eukaryotic cell} keeps its DNA inside a \omterm{def:g10:cells-common-unit:organelle}{nucleus} and has
membrane-bound \omterm{def:g10:cells-common-unit:organelle}{organelles} (a cheek \omterm{def:g10:cells-common-unit:cell}{cell}, a yeast); a \omterm{def:g10:cells-common-unit:prokaryote}{prokaryotic cell}
has neither, its DNA lies free in the \omterm{def:g10:cells-common-unit:cell}{cytoplasm}, and it is about
\qty{1}{\micro m} (a bacterium such as \emph{E. coli}).
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:6}
Real length $6 \times 5/15 = \qty{2}{\micro m}$. \omterm{def:g10:cells-common-unit:magnification}{Magnification}: the bar
is \qty{15}{mm} for \qty{5}{\micro m} $= \qty{0.005}{mm}$, so
$15/0.005 = 3000$: $\times 3000$.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:7}
A \omterm{def:g10:cells-common-unit:organelle}{ribosome} (\qty{25}{nm}) is ten times smaller than the light
microscope's \omterm{def:g10:cells-common-unit:magnification}{resolution} (\qty{0.2}{\micro m} $= \qty{200}{nm}$): two
neighbouring \omterm{def:g10:cells-common-unit:organelle}{ribosomes}, or a \omterm{def:g10:cells-common-unit:organelle}{ribosome} and its surroundings, cannot be
told apart. More \omterm{def:g10:cells-common-unit:magnification}{magnification} only enlarges the blur.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:8}
The grains are starch, stored inside the \omterm{def:g10:cells-common-unit:cell}{cells} as solid grains (starch
is insoluble). A plant's reserves are kept inside its \omterm{def:g10:cells-common-unit:cell}{cells}, not in
some space between them.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:9}
The onion scale grows underground, in the dark, and stores reserves:
\omterm{def:g10:cells-common-unit:organelle}{chloroplasts} would be useless there. The moss leaf lives in the light
and makes its food by photosynthesis, so its \omterm{def:g10:cells-common-unit:cell}{cells} are packed with
\omterm{def:g10:cells-common-unit:organelle}{chloroplasts}. Same plan, different equipment.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:10}
Side \qty{1}{\micro m}: surface \qty{6}{\micro m^2}, volume
\qty{1}{\micro m^3}, ratio 6 per micrometre. Side \qty{20}{\micro m}:
surface \qty{2400}{\micro m^2}, volume \qty{8000}{\micro m^3}, ratio
0.3. The bacterium has twenty times more membrane per unit of content
and lives by diffusion alone; the larger \omterm{def:g10:cells-common-unit:cell}{cell} needs internal transport
and \omterm{def:g10:cells-common-unit:organelle}{organelles}.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:11}
\qty{2}{h} is 6 divisions: $2^6 = 64$ \omterm{def:g10:cells-common-unit:cell}{cells}. \qty{4}{h} is 12:
$2^{12} = 4096$.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:12}
It has a membrane, a \omterm{def:g10:cells-common-unit:cell}{cytoplasm} and takes in and gives out substances,
so it fits most of the definition, but it cannot divide and cannot
renew its \omterm{def:g10:chemistry-of-life:families}{proteins}: it is a \omterm{def:g10:cells-common-unit:cell}{cell} that has given up part of the
programme in exchange for room for haemoglobin, and it is made and
replaced by complete \omterm{def:g10:cells-common-unit:cell}{cells}. The definition describes the general case;
the red blood \omterm{def:g10:cells-common-unit:cell}{cell} is a specialised, terminal form of it.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:13}
Cork is dead tissue: Hooke saw the cellulose walls of \omterm{def:g10:cells-common-unit:cell}{cells} whose
living content had disappeared --- empty boxes. A \omterm{def:g10:cells-common-unit:organelle}{nucleus} exists only
in a living \omterm{def:g10:cells-common-unit:cell}{cell}'s \omterm{def:g10:cells-common-unit:cell}{cytoplasm}, and in any case his microscope's
\omterm{def:g10:cells-common-unit:magnification}{resolution} and the absence of stains would have hidden it.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:14}
A virus has no \omterm{def:g10:cells-common-unit:cell}{cytoplasm}, no membrane of its own making, no metabolism,
and cannot divide: it fails the definition of a \omterm{def:g10:cells-common-unit:cell}{cell}. By the \omterm{def:g10:cells-common-unit:cell}{cell}
theory it is not alive on its own; it is a particle that borrows a
\omterm{def:g10:cells-common-unit:cell}{cell}'s machinery to make copies of itself. Whether to call it "alive"
is a matter of definition; that it is not a \omterm{def:g10:cells-common-unit:cell}{cell} is not.
\end{solution}

\begin{solution}{exo:g10:cells-common-unit:15}
Volume $\pi \times (25\,\unit{\micro m})^2 \times 30000\,\unit{\micro m}
\approx \num{5.9e7}\,\unit{\micro m^3}$, about 7000 times the
\qty{8000}{\micro m^3} of the cube. Being a thin cylinder, no point of
the fibre is more than \qty{25}{\micro m} from the membrane, so
exchanges remain fast; and each \omterm{def:g10:cells-common-unit:organelle}{nucleus} governs only its own stretch
of the fibre, so no \omterm{def:g10:cells-common-unit:organelle}{nucleus} has to serve a volume larger than an
ordinary \omterm{def:g10:cells-common-unit:cell}{cell}'s.
\end{solution}

\begin{solution}{pb:g10:cells-common-unit:1}
\textbf{1.} $1.8/9 = \qty{0.2}{mm} = \qty{200}{\micro m}$.

\textbf{2.} $1.8/4 = \qty{0.45}{mm}$; about two \omterm{def:g10:cells-common-unit:cell}{cells} fit across it.

\textbf{3.} $28/0.2 = 140$: the drawing is $\times 140$.

\textbf{4.} $4/140 \approx \qty{0.029}{mm}$, about \qty{30}{\micro m}.

\textbf{5.} \omterm{def:g10:cells-common-unit:organelle}{Cell wall}, \omterm{def:g10:cells-common-unit:cell}{plasma membrane} (pressed against the wall),
\omterm{def:g10:cells-common-unit:cell}{cytoplasm}, \omterm{def:g10:cells-common-unit:organelle}{nucleus}, \omterm{def:g10:cells-common-unit:organelle}{vacuole}. Present but invisible at $\times 400$:
the \omterm{def:g10:cells-common-unit:organelle}{ribosomes} (and the \omterm{def:g10:cells-common-unit:organelle}{mitochondria} are at the limit).

\textbf{6.} They come from the surface of a lining that is several
\omterm{def:g10:cells-common-unit:cell}{cells} thick and constantly rubbed: the outermost \omterm{def:g10:cells-common-unit:cell}{cells} are flattened
and shed, which is why a cotton bud collects them.

\textbf{7.} \qty{0.24}{mm} is just above the naked eye's
\qty{0.1}{mm} limit: a barely visible speck. At $\times 100$ it spans
$0.24/1.8 \approx 13\%$ of the field, at $\times 400$ more than half;
no available objective gives exactly a quarter.

\textbf{8.} Bacteria. The electron microscope would show their wall,
membrane, \omterm{def:g10:cells-common-unit:organelle}{ribosomes} and DNA region, and the absence of a \omterm{def:g10:cells-common-unit:organelle}{nucleus}.

\textbf{9.} The cheek \omterm{def:g10:cells-common-unit:cell}{cell} does one job (protecting a surface) in a
body where other \omterm{def:g10:cells-common-unit:cell}{cells} feed it, oxygenate it and coordinate it; the
paramecium has nobody else and must feed, move, sense and divide as a
single \omterm{def:g10:cells-common-unit:cell}{cell}.

\textbf{10.} The outside is now saltier than the \omterm{def:g10:cells-common-unit:cell}{cytoplasm}, so water
leaves the \omterm{def:g10:cells-common-unit:cell}{cell} across the membrane; the \omterm{def:g10:cells-common-unit:cell}{cell} loses volume and
shrivels.

\textbf{11.} $20^3 = \qty{8000}{\micro m^3} = \num{8e-6}\,\unit{mm^3}$.

\textbf{12.} $6 \times 20^2 = \qty{2400}{\micro m^2}$; ratio
$2400/8000 = 0.3$ per micrometre.

\textbf{13.} Surface \qty{6}{\micro m^2}, volume \qty{1}{\micro m^3},
ratio 6: twenty times larger.

\textbf{14.} $\qty{8000}{\micro m^3} = \num{8e-9}\,\unit{cm^3}$, hence
about $\num{8e-9}\,\unit{g} = \qty{8}{ng}$.

\textbf{15.} $\pi \times 3.5^2 \times 2 \approx \qty{77}{\micro m^3}$,
about a hundredth of the cube.

\textbf{16.} $\qty{60}{L} = \num{6e16}\,\unit{\micro m^3}$; divided by
8000: about $\num{7.5e12}$ \omterm{def:g10:cells-common-unit:cell}{cells}.

\textbf{17.} Two thirds of that: about $\num{5e12}$.

\textbf{18.} $\num{2.5e13} \times 77 \approx \num{1.9e15}\,\unit{\micro
m^3} \approx \qty{1.9}{L}$. Blood \omterm{def:g10:cells-common-unit:cell}{cells}: $0.45 \times 4.5 \approx
\qty{2.0}{L}$ --- consistent.

\textbf{19.} About $\num{5e12}$ large \omterm{def:g10:cells-common-unit:cell}{cells} plus $\num{2.5e13}$ red
\omterm{def:g10:cells-common-unit:cell}{cells}: some $\num{3e13}$, a few tens of trillions. \omterm{def:g10:cells-common-unit:cell}{Cell} sizes span a
factor of a hundred and the mix of kinds varies, so any count is an
estimate to within a factor of two or three.

\textbf{20.} Roughly thirty trillion ($\num{3e13}$) \omterm{def:g10:cells-common-unit:cell}{cells}, all
produced by successive \omterm{def:g10:cells-common-unit:cell}{cell} divisions from a single starting \omterm{def:g10:cells-common-unit:cell}{cell}, the
fertilised egg.
\end{solution}
