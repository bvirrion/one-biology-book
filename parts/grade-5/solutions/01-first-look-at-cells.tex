\section*{Chapter \ref{ch:g5:first-look-at-cells} --- A First Look at Cells}

\begin{solution}{exo:g5:first-look-at-cells:1}
The smallest living building block of a \omterm{def:g1:living-or-not:living}{living thing}. Your body
is made of tens of thousands of billions of them.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:2}
Because \omterm{def:g5:first-look-at-cells:cell}{cells} are smaller than the smallest speck the \omterm{def:g1:the-five-senses:sense}{eye} can
see. The \omterm{def:g5:first-look-at-cells:microscope}{microscope} --- magnifying thin slices hundreds of
times --- revealed them.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:3}
Onion \omterm{def:g1:the-five-senses:sense}{skin}: brick-like \omterm{def:g5:first-look-at-cells:cell}{cells} in neat rows, wall against wall.
Cheek scrape: rounder \omterm{def:g5:first-look-at-cells:cell}{cells}, scattered. In common: both
materials are made entirely of \omterm{def:g5:first-look-at-cells:cell}{cells}, most showing a small
dark spot inside.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:4}
The nucleus --- the \omterm{def:g5:first-look-at-cells:cell}{cell}'s command center.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:5}
About $50$ \omterm{def:g5:first-look-at-cells:cell}{cells} span one millimetre; one \omterm{def:g5:first-look-at-cells:cell}{cell} is about a
fiftieth of a millimetre --- $\qty{0.02}{mm}$.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:6}
Growing taller: the body's \omterm{def:g5:first-look-at-cells:cell}{cells} are dividing and multiplying.
Healing: \omterm{def:g5:first-look-at-cells:cell}{cells} are dividing to rebuild the damaged patch.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:7}
One. It was the fertilized \omterm{def:g4:animal-reproduction:sexual}{egg cell} --- the joining of one
\omterm{def:g4:animal-reproduction:sexual}{sperm cell} and one \omterm{def:g4:animal-reproduction:sexual}{egg cell}.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:8}
Doubling from $1$ gives $2$, $4$, $8$, $16$, $32$: there are
$32$ \omterm{def:g5:first-look-at-cells:cell}{cells} after $5$ divisions. Five more doublings give
$64$, $128$, $256$, $512$, $1024$: so $1024$ \omterm{def:g5:first-look-at-cells:cell}{cells} after
$10$.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:9}
However different in size, shape and life, both are built of
the same kind of small living units --- \omterm{def:g5:first-look-at-cells:cell}{cells}. The shared
construction plan, common to all \omterm{def:g1:living-or-not:living}{living things}, is what
``\omterm{prop:g5:first-look-at-cells:principle}{unity of life}'' means.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:10}
All of them: a single \omterm{def:g5:first-look-at-cells:cell}{cell} is born (of a dividing parent
\omterm{def:g5:first-look-at-cells:cell}{cell}), feeds, grows, divides --- making new \omterm{def:g1:living-or-not:living}{living things}
like itself --- and dies. One \omterm{def:g5:first-look-at-cells:cell}{cell} is enough for a complete
life.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:11}
More \omterm{def:g5:first-look-at-cells:cell}{cells} --- \omterm{def:g5:first-look-at-cells:cell}{cell} size barely changes between mouse and
elephant. Growing, then, is not inflating one's \omterm{def:g5:first-look-at-cells:cell}{cells} but
multiplying them: bodies get big by division after division,
exactly as the proposition says.
\end{solution}
