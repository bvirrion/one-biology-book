\section*{Chapter \ref{ch:g4:classifying-animals} --- Classifying Animals}

\begin{solution}{exo:g4:classifying-animals:1}
A \omterm{def:g4:classifying-animals:vertebrate}{vertebrate} has an inner \omterm{def:g3:bones-and-muscles:skeleton}{skeleton} built around a \omterm{def:g4:classifying-animals:vertebrate}{backbone} (the
trout, the cat); an \omterm{def:g4:classifying-animals:vertebrate}{invertebrate} has no \omterm{def:g4:classifying-animals:vertebrate}{backbone} (the snail, the
bee, the earthworm, the jellyfish).
\end{solution}

\begin{solution}{exo:g4:classifying-animals:2}
\omterm{prop:g3:sorting-living-things:groups}{Fish}: scales, fins, breathe in water. \omterm{prop:g3:sorting-living-things:groups}{Amphibians}: moist bare
\omterm{def:g1:the-five-senses:sense}{skin}, \omterm{def:g2:animals-grow-up:born}{eggs} in water, tadpole childhood. \omterm{prop:g4:classifying-animals:classes}{Reptiles}: dry scaly
\omterm{def:g1:the-five-senses:sense}{skin}, most laying shelled \omterm{def:g2:animals-grow-up:born}{eggs} on land. \omterm{prop:g3:sorting-living-things:groups}{Birds}: \omterm{def:g1:animals-around-us:covering}{feathers}, beak,
shelled \omterm{def:g2:animals-grow-up:born}{eggs}. \omterm{prop:g3:sorting-living-things:groups}{Mammals}: \omterm{def:g1:animals-around-us:covering}{fur} or hair, young fed on \omterm{def:g2:animals-grow-up:born}{milk}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:3}
Sardine: \omterm{prop:g3:sorting-living-things:groups}{fish}. Crocodile: reptile. Newt: amphibian. Ostrich:
bird. Whale: mammal. Earthworm: \omterm{def:g4:classifying-animals:vertebrate}{invertebrate} --- no \omterm{def:g4:classifying-animals:vertebrate}{backbone}, so
no \omterm{def:g4:classifying-animals:vertebrate}{vertebrate} class at all.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:4}
The cat, inside the \omterm{prop:g3:sorting-living-things:groups}{mammals}, inside the \omterm{def:g4:classifying-animals:vertebrate}{vertebrates}, inside the
\omterm{def:g1:animals-around-us:animal}{animals}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:5}
The \omterm{def:g1:the-five-senses:sense}{skin} --- dry and scaly for the lizard, moist and bare for
the newt --- and the \omterm{def:g2:animals-grow-up:born}{eggs}: shelled and laid on land for the
lizard, bare and laid in water for the newt.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:6}
The \omterm{def:g4:classifying-animals:vertebrate}{invertebrates}. For example: \omterm{prop:g3:sorting-living-things:groups}{insects}, spiders, snails,
worms (and jellyfish).
\end{solution}

\begin{solution}{exo:g4:classifying-animals:7}
Penguin: \omterm{def:g1:animals-around-us:covering}{feathers}? Yes --- a bird; stop at the first question.
Dolphin: \omterm{def:g1:animals-around-us:covering}{feathers}? No. \omterm{def:g1:animals-around-us:covering}{Fur} or \omterm{def:g2:animals-grow-up:born}{milk}? It feeds its calf \omterm{def:g2:animals-grow-up:born}{milk} ---
a mammal; stop there.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:8}
For example: \omterm{def:g1:animals-around-us:covering}{feathers}, a beak, wings, shelled \omterm{def:g2:animals-grow-up:born}{eggs} --- and a
\omterm{def:g4:classifying-animals:vertebrate}{backbone}, since \omterm{prop:g3:sorting-living-things:groups}{birds} are \omterm{def:g4:classifying-animals:vertebrate}{vertebrates}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:9}
The questions stop at ``\omterm{def:g2:animals-grow-up:born}{milk} for the young'': the whale nurses
its calf, so it is a mammal, near-hairless or not. ``Lives in
the sea'' is a \omterm{def:g2:where-animals-live:habitat}{habitat}, not a body feature --- and
\omterm{def:g4:classifying-animals:classification}{classification} was built on what an \omterm{def:g1:animals-around-us:animal}{animal} \emph{has} precisely
because habits and homes can be shared by strangers: the shark
and the dolphin share an address, not a class.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:10}
The \omterm{def:g4:classifying-animals:vertebrate}{backbone}. The snake has a \omterm{def:g3:bones-and-muscles:skeleton}{spine} of hundreds of small \omterm{def:g3:bones-and-muscles:skeleton}{bones}
--- a \omterm{def:g4:classifying-animals:vertebrate}{vertebrate}, with the dry scaly \omterm{def:g1:the-five-senses:sense}{skin} of a reptile; the
earthworm has no \omterm{def:g4:classifying-animals:vertebrate}{backbone} at all --- an \omterm{def:g4:classifying-animals:vertebrate}{invertebrate}. Missing
\omterm{def:g1:my-body:parts}{legs} are just a missing feature; the \omterm{def:g4:classifying-animals:vertebrate}{backbone} question is asked
first.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:11}
The \omterm{prop:g3:sorting-living-things:groups}{mammals} won: \omterm{def:g1:animals-around-us:covering}{fur} and \omterm{def:g2:animals-grow-up:born}{milk} for the young are the mammal
bundle, and the platypus has both --- the \omterm{def:g2:animals-grow-up:born}{egg} is one feature
out of step. \omterm{def:g4:classifying-animals:classification}{Classification} weighs the whole bundle rather than
letting a single feature decide; an \omterm{def:g1:animals-around-us:animal}{animal} odd in one respect
still sits where most of its features point. (The platypus is
odd enough that \omterm{prop:g3:sorting-living-things:groups}{mammals} with \omterm{def:g2:animals-grow-up:born}{eggs} form their own small
sub-group.)
\end{solution}
