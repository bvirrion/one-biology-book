\chapter{كيف تتغير الأنواع}\label{ch:g9:how-species-change}

أثبتت قاعة الأدلة الحقيقة؛ وهذا الفصل يبني المحرك. وقطعه
كلها في يديك سلفًا: \omterm{prop:g9:genes-and-dna:explains}{الطفرة} تكتب تهجئات جديدة
(\cref{prop:g9:genes-and-dna:explains})، والخلط يوزع
مجموعات فريدة
(\cref{prop:g9:unique-individuals:theorem})، وكوابح
\omterm{def:g6:exploring-our-environment:components}{البيئة} تجمع أكثر فائض كل جيل
(\cref{prop:g8:reproduction-and-environment:limits}).
وإذا ركّبت جرت من تلقاء نفسها --- وذلك الجريان الذاتي هو
\omterm{prop:g9:how-species-change:selection}{الانتقاء الطبيعي}، وهو أعظم فكرة بسيطة أثرًا في
البيولوجيا.

\section{المحرك، مركّبًا}

\begin{proposition}[الانتقاء الطبيعي]\label{prop:g9:how-species-change:selection}
ثلاث حقائق لا تنكر، ونتيجتها:
\begin{enumerate}
  \item \emph{التنوع}: أفراد \omterm{def:g8:reproduction-and-environment:population}{الجمهرة} يختلفون في \omterm{def:g9:heredity-and-traits:traits}{صفات}
        موروثة
        (\cref{prop:g9:unique-individuals:reserve}
        أودعه)؛
  \item \emph{فرط الإنتاج والكوابح}: ينتج من الصغار أكثر
        بكثير مما تدع \omterm{def:g6:exploring-our-environment:components}{البيئة} يبقى --- فالكوابح تجمع
        الفائض
        (\cref{ex:g8:reproduction-and-environment:twosprings})؛
  \item \emph{الجمع غير المتساوي}: الكوابح لا تجمع
        عشوائيًا --- فالأفراد الذين تلائم صفاتهم الموروثة
        الظروف القائمة يفلتون منها أكثر قليلًا، فيتركون
        نسلًا أكثر، \emph{وهو يرث تلك \omterm{def:g9:heredity-and-traits:traits}{الصفات}}.
\end{enumerate}
والنتيجة: جيلًا بعد جيل، تنمو حصة \omterm{def:g9:genes-and-dna:gene}{الأليلات} الملائمة في
\omterm{def:g8:reproduction-and-environment:population}{الجمهرة}. وذلك الانزياح في التركيب الموروث \omterm{def:g8:reproduction-and-environment:population}{للجمهرة}، ولا
يقوده إلا التنوع وقد لقي الكوابح، هو \emph{الانتقاء
الطبيعي}\index{انتقاء طبيعي} --- ومهلة كافية من الزمن
تجعله \emph{تطورًا}\index{تطور}.
\end{proposition}
\admitted

\begin{example}[العثات، بآليتها]\label{ex:g9:how-species-change:moths}
عثات الفلفل في \Cref{ex:g9:evidence-of-evolution:live}،
قطعة قطعة. التنوع: صورتان موروثتان، شاحبة وداكنة.
والكابح: \omterm{prop:g3:sorting-living-things:groups}{الطيور} تصطاد العثات بالبصر على اللحاء. والجمع
غير المتساوي: على الجذوع المسودة بالسخام تكون الصورة
الشاحبة هي المرئية --- فتؤكل أكثر، فترتفع حصة \omterm{def:g9:genes-and-dna:gene}{الأليل}
الداكن نحو \omterm{prop:g5:people-and-nature:managed}{الغابات} الداكنة المرصودة؛ ثم يصفو الهواء،
ويشحب اللحاء، فيجري المحرك نفسه معكوسًا. ولم تتغير عثة
واحدة؛ بل تغير تركيب \emph{\omterm{def:g8:reproduction-and-environment:population}{الجمهرة}} --- فالانتقاء يحرر
تواترات المجموعة، لا الأفراد.
\end{example}

\begin{remark}[ما ليس الانتقاء]\label{rem:g9:how-species-change:not}
ثلاثة التباسات قائمة، تحال إلى التقاعد:
\begin{itemize}
  \item \emph{لا نية}: فالظروف تنتقي كما ``انتقت''
        \omterm{def:g6:exploring-our-environment:conditions}{الرطوبة} قمل الخشب
        (\cref{ex:g6:exploring-our-environment:chooser})
        --- بالعواقب، لا بالاختيارات؛ ولا شيء يخطط؛
  \item \emph{ولا كمال}: فالانتقاء لا يقارن إلا الصور
        \emph{الحاضرة} --- فيجد الأفضل المتاح، لا الأفضل
        المتخيَّل (وإعادة توظيف مخطط \omterm{def:g1:my-body:parts}{الأطراف} المشترك على
        عناء، \cref{exo:g9:evidence-of-evolution:7}، هي
        توقيعه)؛
  \item \emph{ولا توريث للجهد}: \omterm{def:g1:my-body:parts}{فذراع} الحداد تموت مع
        الحداد
        (\cref{prop:g9:heredity-and-traits:sorting})؛
        والانتقاء لا يعمل إلا فيما تحمله \omterm{def:g8:sexual-reproduction:gametes}{الأمشاج}.
        فالزرافات لم تمط أعناقها فتطول --- بل تفوقت
        الصور الأطول عنقًا في التغذي والتناسل على الأقصر.
\end{itemize}
\end{remark}

\section{المحرك مرصودًا ومستغلًا}

\begin{example}[الانتقاء في المستشفى]\label{ex:g9:how-species-change:resistance}
الحالة الطبية، بآليتها، ليستعملها
\cref{ch:g9:microbes-and-infection}: \omterm{def:g8:reproduction-and-environment:population}{جمهرة} \omterm{def:g6:producing-our-food:fermentation}{ميكروبات}
تحمل طافرات مقاومة نادرة (رشح الطفرات)؛ والمضاد الحيوي
هو الكابح؛ فيجمع القابلة للتأثر ويعفي المقاومة، فترث
الجناح. أيام لا آلاف السنين --- إذ تدور الأجيال في
ساعات، وسرعة الانتقاء تتدرج معها. وكل قاعدة عن إتمام
جرعات المضادات الحيوية وعن عدم نثرها في الزراعة هي إدارة
للمحرك: أضعف الكابح على نحو مهلهل، تربّ ما أردت قتله.
\end{example}

\begin{example}[الانتقاء ويد إنسان على الكابح]\label{ex:g9:how-species-change:artificial}
مربّية \Cref{pb:g8:sexual-reproduction:1} أجرت المحرك
نفسه وهي نفسها الكابح --- تبقي المقرمش والمقاوم، وتستبعد
الباقي. وكل سلالة كلاب، وكل صنف \omterm{def:g6:producing-our-food:farming}{محصول}، وكل خط حلوب هو
\emph{انتقاء اصطناعي} متراكم: برهان على قوة المحرك على
مقياس زمن الإنسان، من الذئب إلى كلب الحضن في بضعة آلاف
من الأجيال. وداروين --- عالم الطبيعة الذي ركّب حجة هذا
الفصل --- افتتح كتابه العظيم بتلك المقارنة بعينها.
\end{example}

\begin{proposition}[من التغير إلى أنواع جديدة]\label{prop:g9:how-species-change:speciation}
الانتقاء يفسر تغير \omterm{def:g8:reproduction-and-environment:population}{جمهرة}؛ ومكوّن واحد آخر يفسر
\emph{انشطار} \omterm{def:g8:reproduction-and-environment:population}{الجمهرات}. افصل جمهرتين من \omterm{def:g5:biodiversity:species}{نوع} واحد ---
يرتفع بحر، أو ينقسم جبل، أو تستقبل جزيرة غرقى
(\cref{exo:g6:how-plants-spread:11}) --- فتجري كل واحدة
المحرك تحت ظروفها هي، فتراكم أليلاتها هي. وإذا بلغ
التباعد مبلغًا كافيًا، لم تعد الجمهرتان تتناسلان حيث
تلتقيان: فبتعريف \cref{def:g5:biodiversity:species}
نفسه، يقوم نوعان حيث قام \omterm{def:g5:biodiversity:species}{نوع} واحد. وفروق الشجرة
(\cref{def:g6:classification-kinship:tree}) هي بالضبط
انشطارات كهذه، بمقدار الزمن السحيق منها.
\end{proposition}
\admitted

\begin{method}[إجراء المحرك على أي حالة]\label{met:g9:how-species-change:run}
في أي مثال يدعى للتطور:
\begin{enumerate}
  \item سمّ \emph{التنوع} وتحقق من كونه موروثًا
        (\cref{met:g9:heredity-and-traits:audit})؛
  \item وسمّ \emph{الكابح}، وكيف يجمع على غير سواء عبر
        ذلك التنوع؛
  \item وتابع \emph{المحاسبة}: \omterm{def:g9:genes-and-dna:gene}{أليلات} من تكسب حصة في
        الجيل التالي؟
  \item وإن كان انشطارًا، فأضف \emph{الفصل} والظروف
        المتباعدة؛
  \item وتحقق من مقياس الزمن: فالأجيال، لا السنون، هي
        نقرات المحرك.
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:1}
اذكر حقائق \omterm{prop:g9:how-species-change:selection}{الانتقاء الطبيعي} الثلاث ونتيجتها.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:2}
في حالة العثات، سمّ التنوع والكابح والجمع غير المتساوي

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:3}
ما الذي يتغير تحت الانتقاء --- الأفراد أم \omterm{def:g8:reproduction-and-environment:population}{الجمهرة}؟

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:4}
صحّح، والآلية في يدك: ``مطّت الزرافات أعناقها ونقلت

لماذا تتطور مقاومة المضادات الحيوية في أيام حيث أخذت

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:6}
ما الذي اشتركت فيه مربّية التفاح وطيور العثات، وما الذي

\begin{exercise}[$\star\star$]\label{exo:g9:how-species-change:9}
ركّب الانتواع لغرقى الجزيرة في

\begin{exercise}[$\star\star$]\label{exo:g9:how-species-change:10}

\section{مسألة: دفاتر الجزر}

\begin{problem}[{مسألة نهاية الأسبوع --- قرنان على أرخبيل
بركاني}]\label{pb:g9:how-species-change:1}
أرخبيل (مركّب، مستوحى من التاريخ): جزر بركانية لم تتصل
باليابسة قط، تأوي طيورًا شبيهة بالعصافير تختلف صور
مناقيرها من جزيرة إلى جزيرة --- وتختلف عن قريبها الوحيد
الظاهر في البر. ودفاتر علماء الطبيعة تمتد قرنين.

\medskip\noindent\textbf{الجزء الأول --- النمط.}
\begin{enumerate}
  \item الجزر لا تحمل إلا أنواعًا استطاعت عبور البحر ---
        طيورًا، وبذورًا طافية، وخفاشًا واحدًا، ولا
        ثدييات يابسة أصلية ولا ضفادع. أي فصل تنبأ بقائمة
        الركاب (\cref{exo:g6:how-plants-spread:11})؟
  \item كل عصافير الأرخبيل تشبه \omterm{def:g5:biodiversity:species}{النوع} البري الواحد أكثر
        مما تشبه أي طائر آخر على الأرض. بم يخلص استدلال
        \omterm{prop:g6:classification-kinship:kinship}{القرابة}
        (\cref{prop:g6:classification-kinship:kinship})
        عن أصلها؟
  \item المناقير تختلف بحسب غذاء الجزيرة: كاسرات \omterm{def:g2:from-seed-to-plant:seed}{بذور} في
        جزر \omterm{def:g2:from-seed-to-plant:seed}{البذور}، وناخسات في جزر الصبار. اقرأ التوافق
        بمساعدة
        \cref{prop:g4:adapted-to-their-world:answers} ---
        ثم قل ما الذي يضيفه هذا الفصل ولم يستطعه فصل
        \omterm{def:g4:adapted-to-their-world:adaptation}{التكيف}.
  \item ولماذا كانت الجزر البركانية التي لم تتصل قط معرض
        المحرك؟ سمّ مكوّن الانتواع الذي توفره بالجملة.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- سنوات دفتر
المحرك.}
\begin{enumerate}[resume]
  \item سنة جفاف: تخفق \omterm{def:g2:from-seed-to-plant:seed}{البذور} اللينة، وتبقى القاسية؛
        وتبين فراجير الباحثين أن فراخ السنة التالية أعمق
        مناقير في المتوسط. أجر محاسبة المحرك على تلك
        الجملة.
  \item ثم تتلوها سنوات ممطرة، فتعود \omterm{def:g2:from-seed-to-plant:seed}{البذور} اللينة،
        وينزاح المتوسط راجعًا. بم يثبت هذا الانعكاس عن
        ``اتجاه'' الانتقاء؟
  \item عصافير جزيرة واحدة، نقلتها عاصفة إلى أخرى،
        تتناسل بضعف مع المقيمين. أي قضية ترقب في منتصف
        جريانها؟
  \item يعترض دفتر من عصر الخلقيين: ``ما رأى أحد نوعًا
        يصير نوعًا آخر.'' أجب بمقياس الزمن ومنهج الشهود
        المتقاربين
        (\cref{met:g9:evidence-of-evolution:weigh}).
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- محاضرة
التركيب.}
\begin{enumerate}[resume]
  \item ألق حكاية الأرخبيل في خطوات المحرك الخمس
        (\cref{met:g9:how-species-change:run})، من وصول
        الغرقى إلى \omterm{def:g5:biodiversity:species}{أنواع} الجزر.
  \item فسّر لماذا كان عصفور كل جزيرة ``الأفضل المتاح،
        لا الأفضل المتخيَّل'' --- ما الذي كانت مجموعة كل
        جزيرة تستطيع أن تعرضه على الانتقاء؟
  \item اربط الأرخبيل بجناح المستشفى في
        \cref{ex:g9:how-species-change:resistance}:
        المحرك نفسه، ومعدل نقر مختلف --- واذكر ما الذي
        يحدد المعدل.
  \item اختم المحاضرة بمقارنة داروين وقد صرّح بها: ما
        الذي تشترك فيه يد مربّي الحمام \omterm{def:g2:from-seed-to-plant:seed}{وبذور} الجفاف
        القاسية.
\end{enumerate}
\end{problem}
