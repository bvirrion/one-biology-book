\chapter{La célula: unidad de vida}\label{ch:b1:cell-unit-of-life}

En 1665 Robert Hooke puso una lámina de corcho bajo una lente y la vio
dividida en cajitas que llamó celdas, por los cuartos de un monasterio.
Diez años después un pañero holandés, Antoni van Leeuwenhoek, que tallaba
sus propias lentes hasta un aumento que nadie más alcanzaba, vio seres vivos
nadando en una gota de agua de charca, en el raspado de sus dientes y en su
propio semen. Hizo falta otro siglo y medio para enunciar lo que ambos
habían visto: que todo \omterm{def:b1:organism-environment:organism}{organismo} está hecho de \omterm{def:b1:cell-unit-of-life:cell}{células} y que toda \omterm{def:b1:cell-unit-of-life:cell}{célula}
procede de una \omterm{def:b1:cell-unit-of-life:cell}{célula}. Este capítulo enuncia la \omterm{prop:b1:cell-unit-of-life:theory}{teoría celular} y sus
pruebas, contrasta las dos clases de \omterm{def:b1:cell-unit-of-life:cell}{célula} y describe los instrumentos ---
los microscopios, la centrífuga, el frasco de cultivo --- que nos permiten
ver las \omterm{def:b1:cell-unit-of-life:cell}{células}, desmontarlas y mantenerlas vivas fuera de un \omterm{def:b1:organism-environment:organism}{organismo}.

\section{La teoría celular}

\begin{definition}[Célula]\label{def:b1:cell-unit-of-life:cell}
Una \emph{célula}\index{célula} es la unidad más pequeña de materia viva: un
volumen de \emph{citoplasma}\index{citoplasma} acuoso delimitado por una
\emph{membrana plasmática}\index{membrana plasmática}, que contiene el
material genético (ADN) y la maquinaria que lo expresa, capaz de tomar
nutrientes, transformar energía, mantener su propia organización y, cuando
las condiciones lo permiten, dividirse en dos células.
\end{definition}

\begin{proposition}[La teoría celular]\label{prop:b1:cell-unit-of-life:theory}
\index{teoría celular}
\begin{enumerate}
  \item Todo \omterm{def:b1:organism-environment:organism}{organismo} está hecho de una o más \omterm{def:b1:cell-unit-of-life:cell}{células}, y la \omterm{def:b1:cell-unit-of-life:cell}{célula} es la
        unidad de estructura y de función de los seres vivos
        (Schleiden y Schwann, 1838--1839).
  \item Toda \omterm{def:b1:cell-unit-of-life:cell}{célula} procede de una \omterm{def:b1:cell-unit-of-life:cell}{célula} preexistente por división
        (Virchow, 1855: \emph{omnis cellula e cellula}); no hay
        generación espontánea.
  \item El material hereditario pasa de \omterm{def:b1:cell-unit-of-life:cell}{célula} a \omterm{def:b1:cell-unit-of-life:cell}{célula} en la división, de
        modo que la continuidad de la vida es la continuidad de las
        \omterm{def:b1:cell-unit-of-life:cell}{células}.
\end{enumerate}
\end{proposition}
\begin{proof}[Evidencia]
El corcho de Hooke (1665) y las secciones vegetales de Grew y Malpighi
mostraron los \omterm{def:b1:body-plans-tissues:tissue}{tejidos} vegetales como conjuntos de cámaras; las lentes de
Leeuwenhoek (década de 1670) mostraron \omterm{def:b1:organism-environment:organism}{organismos} unicelulares, \omterm{def:b1:cell-unit-of-life:cell}{células}
sanguíneas y espermatozoides. Dos siglos de lentes cada vez mejores
encontraron \omterm{def:b1:cell-unit-of-life:cell}{células} en todos los \omterm{def:b1:body-plans-tissues:tissue}{tejidos} examinados, tanto animales como
vegetales, una vez que Schwann reconoció \omterm{def:b1:cell-unit-of-life:cell}{células} nucleadas en el cartílago y
en los embriones. La división se observó directamente en algas y en
embriones, y Virchow no encontró ningún \omterm{def:b1:body-plans-tissues:tissue}{tejido}, sano o enfermo, cuyas
\omterm{def:b1:cell-unit-of-life:cell}{células} no procedieran de otras \omterm{def:b1:cell-unit-of-life:cell}{células}. Los matraces de cuello de cisne de
Pasteur (1861) --- caldos que se mantenían estériles durante meses cuando el
polvo no podía llegar hasta ellos, y que hervían de vida en pocos días
cuando sí podía --- cerraron la cuestión de la generación espontánea. El
tercer enunciado es obra de finales del siglo XIX, que siguió los
cromosomas a lo largo de la división (\cref{ch:b1:replication-mitosis}).
\end{proof}

\begin{example}[Una célula, muchas células]\label{ex:b1:cell-unit-of-life:onemany}
Una bacteria, una levadura o una ameba son \omterm{def:b1:organism-environment:organism}{organismos} enteros de una sola
\omterm{def:b1:cell-unit-of-life:cell}{célula}, que lo hacen todo --- alimentarse, moverse, dividirse --- en unos
pocos micrómetros. Un ser humano son unos $\num{3e13}$ \omterm{def:b1:cell-unit-of-life:cell}{células} de doscientas
clases, ninguna de las cuales podría vivir sola en una charca y cada una de
las cuales desciende por divisiones ininterrumpidas de un único óvulo
fecundado. El conejo del \cref{ch:b1:mammal-organization} y el roble del
\cref{ch:b1:flowering-plant-organization} son la misma teoría aplicada dos
veces.
\end{example}

\section{Dos clases de célula}

\begin{definition}[Células procariotas y eucariotas]\label{def:b1:cell-unit-of-life:prokeuk}
Una \emph{célula procariota}\index{célula procariota} (bacterias, arqueas)
no tiene núcleo: su ADN, por lo general un único cromosoma circular, se
aloja en una región del \omterm{def:b1:cell-unit-of-life:cell}{citoplasma} llamada
\emph{nucleoide}\index{nucleoide}, y no tiene compartimentos delimitados por
membranas. Es pequeña (\qtyrange{0.5}{5}{\micro m}) y está delimitada por
una \omterm{def:b1:cell-unit-of-life:cell}{membrana plasmática} y, por fuera de ella, una \emph{pared
celular}\index{pared celular}. Una \emph{célula
eucariota}\index{célula eucariota} (protistas, hongos, plantas, animales)
guarda su ADN, en forma de varios cromosomas lineales, dentro de un
\emph{núcleo}\index{núcleo} delimitado por una doble membrana, y divide su
\omterm{def:b1:cell-unit-of-life:cell}{citoplasma} en \emph{orgánulos}\index{orgánulo} delimitados por membranas
(\cref{ch:b1:eukaryotic-cell}). Es mayor
(\qtyrange{10}{100}{\micro m}) y tiene un esqueleto interno de filamentos
proteicos.
\end{definition}

\begin{proposition}[La célula bacteriana]\label{prop:b1:cell-unit-of-life:bacterium}
\emph{E. coli}, la bacteria modelo, es un bastón de \qty{2}{\micro m} de
largo y \qty{0.8}{\micro m} de ancho: un volumen de aproximadamente
\qty{1}{\micro m^3} que alberga un cromosoma de $\num{4.6e6}$ pares de bases
(\qty{1.5}{mm} de ADN plegado diez mil veces), unos \num{20000} ribosomas,
unos pocos millones de moléculas de proteína de \num{2000} clases distintas
y, a menudo, uno o varios \emph{plásmidos}\index{plásmido}: pequeñas
moléculas circulares de ADN que llevan genes adicionales. Su envoltura es,
de dentro afuera, la \omterm{def:b1:cell-unit-of-life:cell}{membrana plasmática}, una pared delgada de
\emph{peptidoglucano}\index{peptidoglucano} (una malla de cadenas de azúcar
entrecruzadas por péptidos cortos, que resiste la presión interna de la
\omterm{def:b1:cell-unit-of-life:cell}{célula}) y una membrana externa. Las bacterias con una pared gruesa de
peptidoglucano y sin membrana externa retienen el cristal violeta de la
\emph{tinción de Gram}\index{tinción de Gram} (grampositivas); las que
tienen una pared delgada bajo una membrana externa lo pierden
(gramnegativas, como \emph{E. coli}). Los flagelos giran como hélices; los
pili fijan la \omterm{def:b1:cell-unit-of-life:cell}{célula} a las superficies y a otras \omterm{def:b1:cell-unit-of-life:cell}{células}.
\end{proposition}

\begin{method}[La tinción de Gram]\label{met:b1:cell-unit-of-life:gram}
\begin{enumerate}
  \item Fíjese por calor una extensión fina de bacterias a un porta.
  \item Tíñase con cristal violeta (todas las \omterm{def:b1:cell-unit-of-life:cell}{células} quedan moradas) y
        después con yodo, que forma un complejo voluminoso con el
        colorante dentro de las \omterm{def:b1:cell-unit-of-life:cell}{células}.
  \item Enjuáguese con alcohol: el peptidoglucano grueso de las \omterm{def:b1:cell-unit-of-life:cell}{células}
        grampositivas se contrae y atrapa el complejo; la pared delgada de
        las gramnegativas, una vez que el alcohol ha disuelto la membrana
        externa, lo deja escapar.
  \item Contrastíñase con un colorante rosa (safranina): las \omterm{def:b1:cell-unit-of-life:cell}{células}
        grampositivas aparecen moradas y las gramnegativas, rosas. El
        resultado predice la estructura de la pared y, con ella, la
        sensibilidad a muchos antibióticos.
\end{enumerate}
\end{method}

\begin{example}[Por qué la célula eucariota está compartimentada]\label{ex:b1:cell-unit-of-life:whycompart}
Una \omterm{def:b1:cell-unit-of-life:cell}{célula} de \qty{20}{\micro m} tiene mil veces el volumen de una bacteria,
pero solo cien veces su superficie de membrana
(\cref{prop:b1:organism-environment:sv}). Las membranas son el lugar donde
una \omterm{def:b1:cell-unit-of-life:cell}{célula} aloja sus cadenas transductoras de energía y sus sistemas de
transporte; el eucariota recupera superficie plegando membranas en su
interior --- la membrana interna de una mitocondria, las láminas del
retículo endoplasmático --- y gana, además, cámaras separadas en las que
químicas incompatibles (la digestión en el lisosoma, la síntesis en el
citosol) pueden funcionar una junto a otra.
\end{example}

\section{Ver las células: la microscopía}

\begin{definition}[Aumento y resolución]\label{def:b1:cell-unit-of-life:resolution}
Un microscopio \emph{aumenta}\index{aumento} (agranda la imagen) y
\emph{resuelve}\index{resolución} (separa puntos vecinos). Solo lo segundo
limita lo que puede verse: el \emph{poder de
resolución}\index{poder de resolución} es la menor distancia $d$ entre dos
puntos que siguen apareciendo distintos,
\[
d = \frac{0.61\,\lambda}{\mathrm{NA}},
\]
donde $\lambda$ es la longitud de onda de la iluminación y $\mathrm{NA}$ la
apertura numérica del objetivo (el seno de su semiángulo de captación de luz
por el índice de refracción del medio, como mucho $1.4$ con aceite). Aumentar
más allá del poder de resolución agranda el borrón, no el detalle.
\end{definition}

\begin{proposition}[Microscopios ópticos y electrónicos]\label{prop:b1:cell-unit-of-life:microscopes}
Con luz visible ($\lambda \approx \qty{550}{nm}$) y $\mathrm{NA} = 1.4$, el
\emph{microscopio óptico}\index{microscopio óptico} resuelve
\qty{0.24}{\micro m}: \omterm{def:b1:cell-unit-of-life:cell}{células}, \omterm{def:b1:cell-unit-of-life:prokeuk}{núcleos}, cloroplastos, mitocondrias como
puntos, bacterias como bastones, y muestras vivas en movimiento. El
contraste en una \omterm{def:b1:cell-unit-of-life:cell}{célula} transparente se obtiene por tinción (que suele
matarla), por \emph{contraste de fases} (que convierte diferencias de índice
de refracción en diferencias de brillo) o por \emph{fluorescencia} (marcando
una molécula con un colorante o una proteína fluorescente y captando solo su
luz). El \emph{microscopio electrónico}\index{microscopio electrónico} usa
electrones de unos pocos picómetros de longitud de onda; su resolución
práctica, limitada por sus lentes, es de unos \qty{0.2}{nm} en transmisión
(TEM: los electrones atraviesan una sección de \qty{50}{nm} de espesor
teñida con metales pesados, lo que da la estructura interna de los
\omterm{def:b1:cell-unit-of-life:prokeuk}{orgánulos}) y de unos nanómetros en barrido (SEM: un haz que recorre una
superficie recubierta de metal, lo que da su relieve en profundidad). La
microscopía electrónica exige vacío y muestras muertas, fijadas y
deshidratadas.
\end{proposition}

\begin{method}[Elección del instrumento]\label{met:b1:cell-unit-of-life:choose}
\begin{enumerate}
  \item Para seguir un proceso vivo (una división celular, un movimiento,
        una proteína fluorescente que pasa de un compartimento a otro):
        microscopía óptica, de contraste de fases o de fluorescencia, sobre
        \omterm{def:b1:cell-unit-of-life:cell}{células} sin fijar.
  \item Para contar e identificar \omterm{def:b1:cell-unit-of-life:cell}{células} en un \omterm{def:b1:body-plans-tissues:tissue}{tejido}: una preparación
        fijada, seccionada y teñida al \omterm{prop:b1:cell-unit-of-life:microscopes}{microscopio óptico}.
  \item Para ver la estructura interna de un \omterm{def:b1:cell-unit-of-life:prokeuk}{orgánulo}, de una membrana o de
        un virus: microscopía electrónica de transmisión de secciones
        finas, o de partículas con tinción negativa.
  \item Para ver el relieve de la superficie de una \omterm{def:b1:cell-unit-of-life:cell}{célula} o de un \omterm{def:b1:body-plans-tissues:tissue}{tejido}:
        microscopía electrónica de barrido.
  \item Para localizar una proteína: inmunofluorescencia (óptica) o marcaje
        con inmunooro (electrónica), con un anticuerpo contra ella.
\end{enumerate}
\end{method}

\begin{example}[Una bacteria y una mitocondria]\label{ex:b1:cell-unit-of-life:seeing}
Al \omterm{prop:b1:cell-unit-of-life:microscopes}{microscopio óptico} con $1000\times$, una \omterm{def:b1:cell-unit-of-life:cell}{célula} de \emph{E. coli} es un
bastón de \qty{2}{mm} de largo a escala de la retina, sin interior visible;
una mitocondria es un punto en el límite de la resolución. En una sección de
TEM esa misma bacteria muestra sus dos membranas, su pared, su \omterm{def:b1:cell-unit-of-life:prokeuk}{nucleoide}

de las dos está viva.
\end{example}

\section{Desmontar las células y mantenerlas vivas}

\begin{definition}[Fraccionamiento celular]\label{def:b1:cell-unit-of-life:fractionation}
El \emph{fraccionamiento celular}\index{fraccionamiento celular} separa los
componentes de una \omterm{def:b1:cell-unit-of-life:cell}{célula} para estudiarlos. El \omterm{def:b1:body-plans-tissues:tissue}{tejido} se
\emph{homogeneiza}\index{homogeneizado} --- se rompen las \omterm{def:b1:cell-unit-of-life:cell}{células} en un
tampón frío e isotónico por molienda o cizalladura, dejando intactos los
\omterm{def:b1:cell-unit-of-life:prokeuk}{orgánulos} --- y el homogeneizado se somete a una \emph{centrifugación
diferencial}\index{centrifugación diferencial}: una serie de giros de fuerza
y duración crecientes, cada uno de los cuales sedimenta las partículas
mayores y más densas que quedan en suspensión. Cada fracción se identifica
por su aspecto al \omterm{prop:b1:cell-unit-of-life:microscopes}{microscopio electrónico} y por
\emph{enzimas marcadoras}\index{enzima marcadora}, actividades que se sabe
que residen en un solo compartimento.
\end{definition}

\begin{proposition}[Sedimentación]\label{prop:b1:cell-unit-of-life:stokes}
Una partícula esférica de radio $r$ y densidad $\rho_p$ en un fluido de
densidad $\rho_f$ y viscosidad $\eta$, en un campo centrífugo $g_c$ (un
múltiplo de $g$), sedimenta a la velocidad estacionaria
\[
v = \frac{2}{9}\,\frac{r^2\,(\rho_p - \rho_f)\,g_c}{\eta}
\]
(ley de Stokes). La velocidad crece con el cuadrado del radio: un \omterm{def:b1:cell-unit-of-life:prokeuk}{núcleo} de
\qty{5}{\micro m} cae cien veces más deprisa que una mitocondria de
\qty{0.5}{\micro m}, que a su vez cae mil veces más deprisa que un ribosoma
de \qty{15}{nm}. Por eso un mismo tubo los separa a velocidades
crecientes.
\end{proposition}
\begin{proof}
A velocidad estacionaria, el rozamiento de Stokes, $6\pi\eta r v$, equilibra
el peso neto de la partícula en el fluido,
$\frac{4}{3}\pi r^3(\rho_p - \rho_f)g_c$; despejando $v$ se obtiene la
fórmula.
\end{proof}

\begin{definition}[Cultivo celular]\label{def:b1:cell-unit-of-life:culture}
El \emph{cultivo celular}\index{cultivo celular} mantiene las \omterm{def:b1:cell-unit-of-life:cell}{células} vivas
y dividiéndose fuera del \omterm{def:b1:organism-environment:organism}{organismo}, en un medio estéril que aporta
nutrientes, sales, factores de crecimiento y, para las \omterm{def:b1:cell-unit-of-life:cell}{células} animales, una
superficie a la que adherirse, a la temperatura del \omterm{def:b1:organism-environment:organism}{organismo}. Las bacterias
y las levaduras crecen en medios sencillos y se duplican en minutos u horas;
las \omterm{def:b1:cell-unit-of-life:cell}{células} animales tomadas de un \omterm{def:b1:body-plans-tissues:tissue}{tejido} (\emph{cultivos primarios}) se
dividen un número limitado de veces, mientras que las \emph{líneas
celulares} derivadas de tumores o de \omterm{def:b1:cell-unit-of-life:cell}{células} transformadas se dividen
indefinidamente. El cultivo convierte a la \omterm{def:b1:cell-unit-of-life:cell}{célula} en un objeto experimental:
un solo tipo, un solo medio, una variable cada vez.
\end{definition}

\begin{example}[Una curva de crecimiento]\label{ex:b1:cell-unit-of-life:growthcurve}
Un matraz de \emph{E. coli} en un medio rico a \qty{37}{\celsius}: una
latencia de media hora y después crecimiento exponencial --- el recuento se
duplica cada \qty{20}{min}, de modo que $N = N_0\,2^{t/20}$ --- hasta que se
agota un nutriente o se acumulan desechos y el cultivo alcanza una meseta
cercana a $10^9$ \omterm{def:b1:cell-unit-of-life:cell}{células} por mililitro. A partir de una sola \omterm{def:b1:cell-unit-of-life:cell}{célula},
$2^{30} \approx 10^9$ \omterm{def:b1:cell-unit-of-life:cell}{células} en diez horas. La misma curva, con un tiempo
de duplicación de un día, describe un cultivo de \omterm{def:b1:cell-unit-of-life:cell}{células} humanas.
\end{example}

\section{Problema: fraccionar un hígado}

\begin{problem}[{Problema de fin de semana --- diez gramos de hígado de
rata desmontados por centrifugación: sedimentos, marcadores, ley de Stokes y
el balance de una enzima, hasta llegar al factor de purificación de la
fracción mitocondrial}]
\label{pb:b1:cell-unit-of-life:1}
Se homogeneizan \qty{10}{g} de hígado de rata en \qty{50}{mL} de sacarosa
isotónica fría y se centrifugan en tres etapas: $1000\,g$ durante
\qty{10}{min} (sedimento 1), $\num{10000}\,g$ durante \qty{20}{min}
(sedimento 2) y $\num{100000}\,g$ durante \qty{60}{min} (sedimento 3),
dejando el sobrenadante final. Se miden el contenido de proteína y dos
actividades enzimáticas en el \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado} y en cada fracción:

\begin{center}
\small
\begin{tabular}{lccc}
\hline
fracción & proteína (mg) & succinato deshidrogenasa (U) & lactato deshidrogenasa (U) \\
\hline
\omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado} & 2000 & 100 & 400 \\
sedimento 1 & 500 & 15 & 20 \\
sedimento 2 & 200 & 60 & 12 \\
sedimento 3 & 300 & 5 & 8 \\
sobrenadante final & 950 & 4 & 350 \\
\hline
\end{tabular}
\end{center}

Tómense $\eta = \qty{1e-3}{Pa.s}$ para el medio, un recorrido de
sedimentación de \qty{5}{cm} y $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{Parte I --- Qué hay en cada sitio.}
\begin{enumerate}
  \item Nómbrese el \omterm{def:b1:cell-unit-of-life:prokeuk}{orgánulo} principal de cada sedimento.
  \item ¿Por qué debe ser frío e isotónico el medio de homogeneización?
  \item ¿De qué compartimento es marcadora la succinato deshidrogenasa? ¿Y
        de cuál la lactato deshidrogenasa?
  \item Calcúlese la recuperación de proteína: la suma sobre las fracciones
        como porcentaje del \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado}. Coméntese.
  \item Calcúlese del mismo modo la recuperación de cada actividad
        enzimática.
  \item Calcúlense el contenido de proteína por gramo de hígado y la
        fracción de la masa fresca del hígado que es proteína.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La ley de Stokes.}
Mitocondrias: radio \qty{0.5}{\micro m}, exceso de densidad
\qty{100}{kg/m^3}. \omterm{def:b1:cell-unit-of-life:prokeuk}{Núcleos}: radio \qty{4}{\micro m}, mismo exceso.
Ribosomas: radio \qty{12}{nm}, exceso de densidad \qty{600}{kg/m^3}.
\begin{enumerate}[resume]
  \item Calcúlense la velocidad de sedimentación de una mitocondria a
        $\num{10000}\,g$ y el tiempo que tarda en recorrer \qty{5}{cm}.
        ¿Bastan \qty{20}{min}?
  \item Calcúlese la distancia que recorre una mitocondria durante el
        primer giro ($1000\,g$, \qty{10}{min}). ¿Qué fracción de las
        mitocondrias, repartidas uniformemente al principio, quedaría
        sedimentada en la primera etapa?
  \item Compárese con el \qty{15}{\%} de succinato deshidrogenasa hallado
        en el sedimento 1. ¿Qué más puede arrastrar mitocondrias al
        sedimento 1?
  \item Calcúlese el tiempo que tarda un \omterm{def:b1:cell-unit-of-life:prokeuk}{núcleo} en recorrer \qty{5}{cm} a
        $1000\,g$.
  \item Calcúlese el tiempo que tarda un ribosoma en recorrer \qty{5}{cm} a
        $\num{100000}\,g$. ¿Sedimentaría un ribosoma a $\num{10000}\,g$ en
        \qty{20}{min}?
  \item Explíquese por qué la secuencia de giros va en fuerza creciente y
        por qué cada etapa debe durar lo bastante, pero no demasiado.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Actividad específica y purificación.}
La actividad específica de una enzima en una fracción es su actividad por
miligramo de proteína. El factor de purificación de una fracción para una
enzima es su actividad específica dividida por la del \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado}; el
rendimiento es la actividad de la fracción dividida por la del
\omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado}.
\begin{enumerate}[resume]
  \item Calcúlese la actividad específica de la succinato deshidrogenasa en
        el \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado} y en cada fracción.
  \item Calcúlense el factor de purificación y el rendimiento de la
        succinato deshidrogenasa en el sedimento 2.
  \item Calcúlese el factor de purificación de la succinato deshidrogenasa
        en el sedimento 3 e interprétese su valor.
  \item Calcúlense la actividad específica de la lactato deshidrogenasa en
        el \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado} y en el sobrenadante final, y el factor de
        purificación y el rendimiento correspondientes.
  \item ¿Por qué el factor de purificación de la enzima citosólica es menor
        que el de la enzima mitocondrial, aunque su rendimiento sea mayor?
  \item El sedimento 2 contiene \qty{12}{U} de lactato deshidrogenasa. ¿Es
        la enzima en parte mitocondrial? Propóngase una explicación mejor y
        un modo de comprobarla.
  \item ¿Qué fracción de la proteína del hígado es mitocondrial, si toda la
        succinato deshidrogenasa es mitocondrial y el sedimento 2 tiene una
        pureza del \qty{60}{\%}? (Indicación: úsense la actividad
        específica de las mitocondrias puras, deducida del sedimento 2, y
        la actividad total.)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Comprobar las fracciones.}
\begin{enumerate}[resume]
  \item El sedimento 2 se examina por microscopía electrónica: ¿qué
        instrumento y qué preparación, y qué debería verse?
  \item ¿Cómo se comprobaría si el sedimento 1 contiene \omterm{def:b1:cell-unit-of-life:cell}{células} enteras, y
        qué se haría si se encontraran muchas?
  \item El sedimento 2 se resuspende y se deposita sobre un gradiente de
        densidad de sacarosa, y luego se centrifuga hasta que cada
        partícula se detiene donde su densidad iguala a la del medio. Las
        mitocondrias forman banda a \qty{1.18}{g/mL}, los lisosomas a
        \qty{1.22}{g/mL} y los peroxisomas a \qty{1.24}{g/mL}. Explíquese
        por qué funciona esta separación cuando la \omterm{def:b1:cell-unit-of-life:fractionation}{centrifugación
        diferencial} no podía lograrla.
  \item La actividad de la fosfatasa ácida del sedimento 2 sube bruscamente
        al añadir un detergente al ensayo. Explíquese, y dígase qué indica
        esto sobre el estado de los lisosomas del sedimento.
  \item Un estudiante \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneiza} en agua pura en vez de en sacarosa
        isotónica. Predígase el efecto sobre el sedimento 2 y sobre la
        distribución de la succinato deshidrogenasa y la fosfatasa ácida.
  \item Resúmase el resultado: el factor de purificación y el rendimiento
        de la fracción mitocondrial, y qué limita a cada uno.
\end{enumerate}
\end{problem}
