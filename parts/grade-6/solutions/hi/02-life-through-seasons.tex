\section*{अध्याय \ref{ch:g6:life-through-seasons} --- ऋतुओं में जीवन}

\begin{solution}{exo:g6:life-through-seasons:1}
दिन की लंबाई (\omterm{def:g6:exploring-our-environment:conditions}{प्रकाश}), \omterm{def:g6:exploring-our-environment:conditions}{ताप}, और बारिश तथा सूखे के पीछे चलती मिट्टी या
हवा की नमी।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:2}
\omterm{def:g6:life-through-seasons:trees}{पर्णपाती} --- पतझड़ में अपनी सारी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} गिराने वाले: बलूत, बीच, हॉर्स
चेस्टनट। \omterm{def:g6:life-through-seasons:trees}{सदाबहार} --- पूरी सर्दी काम करती \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} बनाए रखने वाले: चीड़,
होली।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:3}
एक नन्हा पूरा \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} --- यानी तह की हुई नन्ही \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}, मौसम-रोधी \omterm{def:g1:animals-around-us:covering}{शल्कों}
में लिपटी हुईं। वह पिछली गर्मियों में बनाया गया था।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:4}
\omterm{prop:g6:life-through-seasons:herbs}{एकवर्षी पौधा} पूरी तरह मर जाता है और सर्दी \omterm{def:g2:from-seed-to-plant:seed}{बीजों} के रूप में काटता है।
डैफ़ोडिल का ऊपरी हिस्सा सूख जाता है जबकि \omterm{def:g1:plants-around-us:plant}{पौधा} ज़मीन के नीचे \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद} के
रूप में इंतज़ार करता है।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:5}
क्योंकि उसकी \omterm{def:g6:life-through-seasons:buds}{कलियों} के भीतर के \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} पिछली गर्मियों में ही तैयार हो चुके
थे; टहनी में सिर्फ़ गरमाहट की कमी थी, जो घर के भीतर का पानी बग़ीचे से
हफ़्तों पहले दे देता है।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:6}
\omterm{def:g3:animals-through-seasons:migration}{प्रवास}: अबाबील। \omterm{def:g3:animals-through-seasons:hibernation}{शीतनिद्रा}: साही (या \omterm{def:g1:animals-around-us:covering}{खोल} में बंद घोंघा)। सक्रिय रहना:
लोमड़ी, मगपाई, रॉबिन। नन्ही अवस्थाओं के रूप में सर्दी --- \omterm{def:g2:animals-grow-up:born}{अंडा}, डिंभ या
\omterm{ex:g2:animals-grow-up:butterfly}{प्यूपा}: ज़्यादातर \omterm{prop:g3:sorting-living-things:groups}{कीट}, जैसे टिड्डा या तितली।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:7}
तितली: मुँडेर के पत्थर के नीचे \omterm{ex:g2:animals-grow-up:butterfly}{प्यूपा}। टिड्डा: मिट्टी में \omterm{def:g2:animals-grow-up:born}{अंडे}।
गुबरैला: पत्तों के कूड़े में गहरे धँसा सोता हुआ वयस्क। घोंघा: किनारे में
अपने \omterm{def:g1:animals-around-us:covering}{खोल} में बंद। अबाबील: अफ़्रीका में, अपने \omterm{def:g3:animals-through-seasons:migration}{प्रवासी} रूप में।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:8}
ताकि चारों तालिकाओं का कोई भी फ़र्क़ ऋतुओं से आए, सर्वेक्षण के ढंग से
नहीं --- यानी \cref{met:g6:exploring-our-environment:survey} का न्यायी
जाँच वाला नियम, स्थलों के बजाय समय पर लगाया हुआ।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:9}
पहला, बहुत-से ``ग़ैरहाज़िर'' छिपे रूपों में मौजूद हैं --- \omterm{def:g2:animals-grow-up:born}{अंडे}, \omterm{ex:g2:animals-grow-up:butterfly}{प्यूपा},
सोते वयस्क, \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद} --- जिन्हें टहलकर किया गया सर्वेक्षण बिना खोदे और
उठाए छोड़ देता है। दूसरा, कुछ खोए नहीं, बस कहीं और हैं: \omterm{def:g3:animals-through-seasons:migration}{प्रवासी} लौट
आएँगे। जनवरी की सूची बसे हुओं को कम गिनती है और यात्रियों को ग़ायब मान
लेती है।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:10}
\omterm{def:g1:plants-around-us:plant}{पौधा} अपनी भूमिगत भंडार-अवस्था में है --- जीवित, \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंदों} के रूप में
इंतज़ार करता हुआ। मार्च में \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद} फिर \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} और \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} भेजेंगे, और
गुच्छे बढ़े हों तो और घने।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:11}
कहाँ गया, और किस रूप में? मेंढक सर्दी सुप्त अवस्था में काटते हैं ---
तालाब की तलहटी की कीचड़ में दबे या ज़मीन पर नम ओटों में --- और ठंड के
महीने सुस्त पड़े बिताते हैं, जब तक वसंत उन्हें प्रजनन के लिए फिर पानी
में न बुला ले।
\end{solution}

\begin{solution}{exo:g6:life-through-seasons:12}
काम करती \omterm{def:g1:plants-around-us:parts}{पत्ती} के लिए सर्दी की समस्या पानी थामे रखना है: जमी हुई ज़मीन
\omterm{def:g1:plants-around-us:parts}{जड़ों} को लगभग कुछ नहीं देती, जबकि हवा और धूप उसे लगातार खींचती रहती हैं
--- यानी कठ-जूँ वाली सूखने की समस्या, पूरी सर्दी भर। \omterm{def:g6:life-through-seasons:trees}{सदाबहार} \omterm{def:g1:plants-around-us:parts}{पत्ती} की
मोम और सुई वाली बनावट इस नुक़सान को उतना कर देती है जितना \omterm{def:g1:plants-around-us:tree}{पेड़} झेल सके।
चौड़ी पतली \omterm{def:g1:plants-around-us:parts}{पत्ती} को इतनी अच्छी तरह बंद नहीं किया जा सकता; इसलिए \omterm{def:g6:life-through-seasons:trees}{पर्णपाती}
\omterm{def:g1:plants-around-us:tree}{पेड़} का सस्ता उत्तर यह है कि हर पतझड़ अपनी पतली \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} बट्टे खाते
डालकर वसंत को \omterm{def:g6:life-through-seasons:buds}{कलियों} में रख ले।
\end{solution}

\begin{solution}{pb:g6:life-through-seasons:1}
\textbf{1.} हर टहनी पर लगी \omterm{def:g6:life-through-seasons:buds}{कलियाँ} --- यानी सर्दी का वह भंडार जिसमें
पिछली गर्मियों में बनाए गए अगले वसंत के \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} रखे हैं।

\textbf{2.} मार्च और अप्रैल की तस्वीरों के बीच (और नोटों में लंबे होते
दिन तथा नरम पड़ती सुबहें)। \omterm{def:g1:plants-around-us:tree}{पेड़} चढ़ते \omterm{def:g6:exploring-our-environment:conditions}{ताप} और लंबे होते \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} का उत्तर
दे रहा है।

\textbf{3.} \omterm{def:g5:flower-to-fruit:steps}{परागण} करने वाले \omterm{prop:g3:sorting-living-things:groups}{कीटों} को --- सबसे बढ़कर मधुमक्खियों को ---
फूलों पर आना ज़रूरी है, ताकि \omterm{def:g3:flowers-fruits-seeds:parts}{पराग} \omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसरों} तक पहुँचे और \omterm{def:g5:flower-to-fruit:steps}{निषेचन} हो।
वह \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} असल में किसी निषेचित \omterm{prop:g5:flower-to-fruit:roles}{बीजांड} से बना \omterm{def:g2:from-seed-to-plant:seed}{बीज} है, अपने काँटेदार फल-कवच
में।

\textbf{4.} \omterm{def:g6:life-through-seasons:trees}{पर्णपाती}। उसका अगला वसंत सर्दी की तस्वीरों में नंगी टहनियों
पर \omterm{def:g6:life-through-seasons:buds}{कलियों} के रूप में सवार रहता है।

\textbf{5.} लॉन के नीचे, \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंदों} के रूप में --- यानी \omterm{prop:g6:life-through-seasons:herbs}{बहुवर्षी पौधे}
का भूमिगत भंडार।

\textbf{6.} मिट्टी में \omterm{def:g2:from-seed-to-plant:seed}{बीजों} के रूप में: यही \omterm{prop:g6:life-through-seasons:herbs}{एकवर्षी} की रणनीति है ---
\omterm{def:g1:plants-around-us:plant}{पौधे} ख़ुद पतझड़ में मर गए थे।

\textbf{7.} अबाबील: अफ़्रीका में, अपने \omterm{def:g3:animals-through-seasons:migration}{प्रवासी} रूप में, उड़ते \omterm{prop:g3:sorting-living-things:groups}{कीटों} के
पीछे। मधुमक्खियाँ: छत्ते में, सर्दी वाले गुच्छे के रूप में, अपनी रानी
के चारों ओर गरम।

\textbf{8.} सक्रिय रहना। मगपाई को सबसे दुबले महीनों में भोजन ढूँढ़ना
पड़ता है --- ठीक वही समस्या जिससे अबाबील चली जाकर बच जाती है।

\textbf{9.} जैसे: \omterm{def:g6:life-through-seasons:buds}{कलियाँ} फूटती हैं (मार्च--अप्रैल) जब दिन
$11$--$13$ घंटे के हो जाते हैं और सुबहें
\qty{8}{\degreeCelsius} से ऊपर गरम पड़ती हैं; \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} सुनहरी होकर
गिरती हैं (अक्तूबर--नवंबर) जब दिन छोटे होते हैं और सुबहें ठंडी। \omterm{def:g1:plants-around-us:tree}{पेड़} की
घटनाएँ तस्वीर दर तस्वीर \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} के बदलावों से जुड़ती जाती हैं।

\textbf{10.} क्यारी मरी नहीं: वह भंडार में चली गई --- उसी जगह के नीचे
\omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंदों} के रूप में जीवित। ``चला गया'' हमेशा इसका हक़दार है: गया
\emph{कहाँ}, और \emph{किस रूप} में?

\textbf{11.} \omterm{def:g6:life-through-seasons:buds}{कलियाँ} लिए नंगी शाखाएँ; कहीं न कहीं शायद एक मगपाई; और
नीचे पत्तों का कूड़ा तथा \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद} छिपाए नंगी डैफ़ोडिल-ज़मीन; और पगडंडी
की मिट्टी में पोस्ते के \omterm{def:g2:from-seed-to-plant:seed}{बीज}। पक्के दाँव \omterm{def:g6:life-through-seasons:buds}{कलियाँ} और नंगापन हैं --- वे \omterm{def:g1:plants-around-us:tree}{पेड़}
की \omterm{def:g6:life-through-seasons:trees}{पर्णपाती} मशीनरी और एक साल के सबूत से निकलते हैं; मगपाई संभव है, पक्की
नहीं।

\textbf{12.} इससे पता चलता है कि गुम परिस्थिति गरमाहट थी --- फ़रवरी में
घर के भीतर \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} बाहर से लंबा नहीं होता, पर पानी और कमरा गरम हैं, और
इतना ही काफ़ी रहा। न्यायी जाँच: एक ही \omterm{def:g1:plants-around-us:tree}{पेड़} की दो टहनियाँ, दोनों एक ही
खिड़की पर पानी में, एक गरम कमरे में और एक ठंडे बरामदे में --- एक ही
फ़र्क़, गरमाहट; और अगर सिर्फ़ गरम वाली टहनी पत्ते निकाले, तो फ़ैसला पक्का
हो गया।
\end{solution}
