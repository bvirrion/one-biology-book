\chapter{हड्डियाँ और पेशियाँ}\label{ch:g3:bones-and-muscles}

उछलो, और उतरो: तुम्हारे भीतर कुछ अभी-अभी बिलकुल किसी मशीन की तरह काम
कर गया। कड़ी छड़ों ने तुम्हारा आकार थामे रखा, कब्ज़ों ने तुम्हारी
\omterm{def:g1:my-body:parts}{टाँगों} को मुड़ने और उछलने दिया, और सैकड़ों छोटी-छोटी मोटरों ने ठीक
ज़रूरत के समय खींचा। छड़ें तुम्हारी \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ} हैं, कब्ज़े तुम्हारे जोड़,
और मोटरें तुम्हारी \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} --- यही शरीर का वह साज़-सामान है जिससे वह
अपनी हर हरकत करेगा।

\section{कंकाल}

\begin{definition}[कंकाल और हड्डियाँ]\label{def:g3:bones-and-muscles:skeleton}
\emph{कंकाल}\index{कंकाल} शरीर के भीतर लगभग $200$
\emph{हड्डियों}\index{हड्डी} का ढाँचा है। उसके मुख्य हिस्से:
\begin{itemize}
  \item \emph{खोपड़ी}\index{खोपड़ी}, मस्तिष्क के चारों ओर हड्डी का
        ठोस टोप;
  \item \emph{रीढ़ की हड्डी}\index{रीढ़ की हड्डी}, गरदन से कूल्हों तक
        जमी छोटी हड्डियों का लचीला खंभा, जो \omterm{def:g1:my-body:parts}{धड़} को सीधा रखता है;
  \item \emph{पसलियाँ}\index{पसली}, मुड़ी हुई हड्डियाँ, जो हृदय और
        फेफड़ों के चारों ओर पिंजरा बनाती हैं;
  \item \emph{उपांगों की हड्डियाँ}: \omterm{def:g1:my-body:parts}{बाँहों} और \omterm{def:g1:my-body:parts}{टाँगों} की लंबी हड्डियाँ
        --- जाँघ की हड्डी शरीर की सबसे लंबी हड्डी है।
\end{itemize}
\end{definition}

\begin{example}[अपना ढाँचा महसूस करो]\label{ex:g3:bones-and-muscles:feel}
एक्स-रे की ज़रूरत नहीं: धीरे से दबाओ और \omterm{def:g3:bones-and-muscles:skeleton}{कंकाल} ख़ुद ख़बर दे देता है।
पोर; पिंडली की धार; पीठ के बीचोंबीच नीचे तक जाती छोटे-छोटे उभारों की
लड़ी --- यानी रीढ़, एक के बाद एक छोटी \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी}; और उँगलियों के नीचे गिनी
जा सकने वाली \omterm{def:g3:bones-and-muscles:skeleton}{पसलियाँ}। ढाँचा \omterm{def:g1:the-five-senses:sense}{त्वचा} के ठीक नीचे है।
\end{example}

\begin{remark}[बाहर से सख़्त, भीतर से जीवित]\label{rem:g3:bones-and-muscles:alive}
\omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} खड़िया की डंडी जैसी मरी हुई चीज़ नहीं है: वह जीवित है, रक्त से
पोषण पाती है, तुम्हारे बढ़ने के साथ बढ़ती है --- और अपनी मरम्मत भी कर
सकती है। प्लास्टर में बँधी टूटी \omterm{def:g1:my-body:parts}{बाँह} इसीलिए जुड़ती है कि \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} ख़ुद
धीरे-धीरे उस टूट को फिर से बना देती है। खड़िया की डंडियाँ ऐसा कभी नहीं
करतीं।
\end{remark}

\section{जोड़, जहाँ ढाँचा मुड़ता है}

\begin{proposition}[कड़ी छड़ें, मुड़ने की जगहें]\label{prop:g3:bones-and-muscles:joints}
\omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ} ख़ुद कभी नहीं मुड़तीं --- मुड़ी हुई \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} यानी टूटी हुई
\omterm{def:g3:bones-and-muscles:skeleton}{हड्डी}। शरीर का सारा मुड़ना \emph{जोड़ों}\index{जोड़} पर होता है, जहाँ
दो \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ} मिलती हैं: \omterm{def:g1:my-body:joint}{कंधा}, \omterm{def:g1:my-body:joint}{कोहनी}, \omterm{def:g1:my-body:joint}{कलाई}, कूल्हा, \omterm{def:g1:my-body:joint}{घुटना}, \omterm{def:g1:my-body:joint}{टखना}, और
रीढ़ के वे कई छोटे जोड़ जिनके नन्हे-नन्हे मोड़ जुड़कर पीठ को लचीला
बनाते हैं।
\end{proposition}
\admitted

\begin{example}[रीढ़ की करामात]\label{ex:g3:bones-and-muscles:spinetrick}
तुम्हारी पीठ की कोई एक \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} मुड़ नहीं सकती --- फिर भी तुम आगे झुककर
लगभग गेंद बन सकते हो। रीढ़ $20$ से ज़्यादा छोटी \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियों} का खंभा है,
और हर \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} अपने पड़ोसियों पर कुछ ही अंश घूमती है: कई छोटे मोड़ मिलकर
एक बड़ा मोड़ बना देते हैं। भारी चीज़ उठाने के लिए घुटने मोड़ने चाहिए,
पीठ नहीं, इसका भी यही कारण है --- घुटने बड़े मोड़ों के लिए बने हैं,
रीढ़ हलके मोड़ों के लिए।
\end{example}

\section{पेशियाँ, हरकत करने वाली}

\begin{definition}[पेशी]\label{def:g3:bones-and-muscles:muscle}
\emph{पेशी}\index{पेशी} \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियों} से जुड़ी एक माँसल मोटर है। पेशी
\emph{संकुचित}\index{संकुचन} हो सकती है --- यानी छोटी और मोटी हो जाती
है --- और संकुचित होते हुए वह उस \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} को \emph{खींचती} है जिससे वह
जुड़ी है। पेशियाँ सिर्फ़ खींचती हैं; कोई पेशी धकेल नहीं सकती।
\end{definition}

\begin{proposition}[जोड़ियों में काम]\label{prop:g3:bones-and-muscles:pairs}
चूँकि \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} सिर्फ़ खींचती हैं, इसलिए किसी जोड़ को दोनों तरफ़ हिलाने
के लिए एक \emph{जोड़ी} चाहिए: \omterm{def:g1:my-body:parts}{बाँह} के आगे वाली \omterm{def:g3:bones-and-muscles:muscle}{पेशी} \omterm{def:g1:my-body:joint}{कोहनी} को मोड़ती है,
और पीछे वाली उसकी साथी उसे सीधा करती है; जब एक खींचती है तब दूसरी ढीली
रहती है। शरीर का लगभग हर जोड़ ऐसी ही जोड़ियों से हिलता है।
\end{proposition}
\admitted

\begin{example}[जोड़ी को काम करते महसूस करो]\label{ex:g3:bones-and-muscles:pair}
दूसरे हाथ से अपनी ऊपरी \omterm{def:g1:my-body:parts}{बाँह} पकड़ो और धीरे-धीरे \omterm{def:g1:my-body:joint}{कोहनी} मोड़ो: आगे वाली
\omterm{def:g3:bones-and-muscles:muscle}{पेशी} फूलती और सख़्त होती है --- वह \omterm{def:g3:bones-and-muscles:muscle}{संकुचित} हो रही है। अब \omterm{def:g1:my-body:parts}{बाँह} सीधी करो:
आगे वाली नरम पड़ जाती है, और \omterm{def:g1:my-body:parts}{बाँह} के पीछे उसकी साथी कस जाती है। तुमने
अभी-अभी \cref{prop:g3:bones-and-muscles:pairs} को होते हुए महसूस किया।
\end{example}

\begin{method}[हड्डियों और पेशियों की देखभाल]\label{met:g3:bones-and-muscles:care}
\begin{enumerate}
  \item हर रोज़ चलो-फिरो --- इस्तेमाल से \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ} मज़बूत होती हैं,
        \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} भी;
  \item बनाने वाला भोजन खाओ: \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियों} के लिए \omterm{def:g2:animals-grow-up:born}{दूध} से बनी चीज़ें, और
        \omterm{prop:g1:healthy-habits:needs}{विविध भोजन} (\cref{ch:g2:teeth-and-eating-well});
  \item कड़ी क़सरत से पहले शरीर गरम करो, ताकि \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} फटने के बजाय
        खिंचें;
  \item भारी चीज़ें घुटने मोड़कर और पीठ सीधी रखकर उठाओ;
  \item जहाँ गिरने का डर हो वहाँ हेलमेट और गद्दियाँ पहनो: \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ}
        मरम्मत तो करती हैं, पर धीरे।
\end{enumerate}
\end{method}
