\chapter{الرئتان والتبادل الغازي}\label{ch:g7:lungs-and-gas-exchange}

في كل دقيقة من حياتك تمرّ نحو ثمانية لترات من الهواء في عضوين لم
ترهما قط. وقد رسم \cref{ch:g4:breathing} المسار والتبادل؛ ووضع
\cref{ch:g7:how-animals-breathe} الآلة البشرية بين آلات المملكة
الأربع. وهذا الفصل يكمل الصورة: آلة التهوية، والجيوب الهوائية
باسمها الصحيح، وأعداد التبادل --- وما يفعله الدخان بالعمل كله.

\section{آلة التهوية}

\begin{proposition}[كيف يُحرَّك الهواء]\label{prop:g7:lungs-and-gas-exchange:ventilation}
لا \omterm{def:g3:bones-and-muscles:muscle}{عضلات} \omterm{def:g7:how-animals-breathe:four}{للرئتين} خاصة بهما؛ بل يضخّهما الصدر كالمنفاخ:
\begin{itemize}
  \item \omterm{def:g4:breathing:movements}{الشهيق} \emph{عمل}: \omterm{def:g3:bones-and-muscles:muscle}{فعضلات} \omterm{def:g3:bones-and-muscles:skeleton}{الأضلاع} ترفع القفص وتوسّعه،
        و\emph{الحجاب الحاجز}\index{حجاب حاجز} --- وهو أرضية
        العضل العريضة تحت \omterm{def:g7:how-animals-breathe:four}{الرئتين} --- يستوي نازلًا؛ فيمدّ الصدر
        المتسع \omterm{def:g7:how-animals-breathe:four}{الرئتين}، ويدخل الهواء ليملأهما؛
  \item \omterm{def:g4:breathing:movements}{والزفير}، في الراحة، \emph{إطلاق} في معظمه: فترتخي
        \omterm{def:g3:bones-and-muscles:muscle}{العضلات}، ويرتدّ الصدر الممدود، فيخرج الهواء. (ونفخ
        الشموع يضيف عضلًا إلى الارتداد.)
\end{itemize}
نحو $15$ مرة في الدقيقة في الراحة
(\cref{def:g4:breathing:movements})، من غير طلب
(\cref{rem:g5:brain-in-command:more}) --- وأعمق وأسرع عند الطلب
(\cref{prop:g7:organs-at-work:supply}).
\end{proposition}
\admitted

\begin{example}[تحسّس المنفاخ]\label{ex:g7:lungs-and-gas-exchange:feel}
ضع كفّيك على أضلاعك \omterm{def:g7:what-is-respiration:respiration}{وتنفس} عميقًا: فيرتفع القفص ويتسع تحت راحتيك
(\cref{ex:g4:breathing:watch}). والآن الهث كالكلب: فالنفخات
السريعة السطحية كلها \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{حجاب حاجز} تقريبًا. والفواق، تتميمًا للفائدة،
بدايات كاذبة متشنجة من \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{الحجاب الحاجز} --- المنفاخ يخطئ الإشعال على
جدوله الخاص.
\end{example}

\section{الأسناخ والتبادل}

\begin{definition}[الأسناخ]\label{def:g7:lungs-and-gas-exchange:alveoli}
الجيوب الهوائية الصغيرة في \cref{def:g4:breathing:organs} هي
\emph{الأسناخ}\index{سنخ رئوي}: مئات الملايين من الفقاعات الرقيقة
الجدران متجمعةً عند أدقّ \omterm{def:g1:animals-around-us:parts}{أطراف} المسالك الهوائية، كل واحدة ملفوفة
\omterm{def:g4:heart-and-blood:vessels}{بأوعية دموية} دقيقة. ومجموع سطحها هو نصف ملعب التنس في
\cref{rem:g4:breathing:sponge} --- وهو النسخة البشرية من مواصفة
\cref{prop:g7:how-animals-breathe:spec}: كبير رقيق رطب.
\end{definition}

\begin{proposition}[التبادل بالأعداد]\label{prop:g7:lungs-and-gas-exchange:numbers}
قارن الهواء الداخل بالهواء الخارج: فهواء \omterm{def:g4:breathing:movements}{الشهيق} نحو $21$ في المئة
أكسجينًا ويكاد يخلو من \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون}؛ وهواء \omterm{def:g4:breathing:movements}{الزفير} ينزل إلى
نحو $16$ في المئة أكسجينًا ويرتفع إلى نحو $4$ في المئة ثاني أكسيد
كربون (مع كونه دافئًا رطبًا ---
\cref{ex:g4:breathing:mirror}). والفروق هي التبادل: فعبر جدران
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{الأسناخ} يعبر \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} إلى \omterm{prop:g4:journey-of-food:absorb}{الدم} ويعبر \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون} راجعًا
(\cref{prop:g4:breathing:exchange})، كل دقيقة من كل يوم.
\end{proposition}
\admitted

\begin{example}[الحساب يستقيم]\label{ex:g7:lungs-and-gas-exchange:accounting}
ثمانية لترات في الدقيقة، ويُحتفظ بجزء من عشرين من \omterm{def:g7:what-is-respiration:respiration}{أكسجين} كل نفَس:
فالحساب يوفّر تقريبًا فاتورة \omterm{def:g7:what-is-respiration:respiration}{أكسجين} الجسم في الراحة --- وفي الجهد،
مع تثليث \omterm{def:g7:what-is-respiration:respiration}{التنفس} وتعميقه وتسريع جولات \omterm{def:g4:heart-and-blood:heart}{القلب}
(\cref{prop:g7:organs-at-work:supply})، يتضاعف التوصيل ليطابق
فاتورة \omterm{def:g3:bones-and-muscles:muscle}{العضلات}. فأعداد الآلة ودفتر الجسم يتوازنان؛ وهما الدفتران
نفساهما ممسوكان عند شبّاكين.
\end{example}

\section{الآلة تُساء: الدخان}

\begin{proposition}[ماذا يفعل الدخان بالعمل]\label{prop:g7:lungs-and-gas-exchange:smoke}
دخان التبغ يهاجم كل طابق من طوابق هذا الفصل
(وقد أعطت \cref{prop:g5:healthy-choices:substances} التحذير؛ وهذه
هي الهندسة):
\begin{itemize}
  \item فقطرانه يطلي المسالك الهوائية ويهيّجها: فتتعثّر سطوح
        التنظيف، ويكون ``سعال المدخّن'' مكنسةَ المسالك المحمَّلة
        فوق طاقتها؛
  \item وجدران \omterm{def:g7:lungs-and-gas-exchange:alveoli}{الأسناخ}، إذا هُيّجت تهييجًا مزمنًا، تنهار: فتندمج
        الجيوب في فراغات أقل وأكبر وأرخى --- \emph{ويُفقد السطح}،
        ويُفقد معه التبادل (والمواصفة مفكَّكة)؛
  \item وأحد غازات الدخان يركب مقاعد \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} في \omterm{prop:g4:journey-of-food:absorb}{الدم}، فيزحم
        \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} عن وسيلة نقله هو؛
  \item وتضيق المسالك الهوائية --- فكل نفَس عبر صدر مدخّن نفَس
        عبر منفاخ مسدود جزئيًّا.
\end{itemize}
والضرر بطيء تراكمي لا يُرَدّ إلا جزئيًّا: أما \omterm{def:g7:how-animals-breathe:four}{الرئة} التي لم تبدأ قط
فتحتفظ بنصف ملعب التنس كاملًا (\cref{rem:g5:healthy-choices:never}).
\end{proposition}
\admitted

\begin{example}[اختباران على الدرج]\label{ex:g7:lungs-and-gas-exchange:stairs}
طبقات \cref{ch:g7:organs-at-work} الثلاث، يصعدها بالغان متساويان في
السن وفي عادات التمرين، أحدهما مدخّن منذ زمن طويل: فيصل المدخّن
لاهثًا عند \omterm{prop:g4:heart-and-blood:pulse}{نبض} لا يبلغه الآخر إلا في العدو. سطح أقل، ومقاعد
مشغولة، ومسالك ضيقة --- فسلسلة التموين تدفع فاتورة
\cref{prop:g7:organs-at-work:bill} عبر مدخل مخنوق. واختبارات
اللياقة تقرأ تاريخ \omterm{def:g7:how-animals-breathe:four}{الرئتين} بيقين أي تصوير.
\end{example}

\begin{method}[صيانة الآلة]\label{met:g7:lungs-and-gas-exchange:keep}
\cref{met:g4:breathing:care}، مرقّاةً بأسباب هذا
الفصل:
\begin{enumerate}
  \item لا دخان --- لا دخانك ولا دخان غيرك: فالسطح المفقود لا
        يكاد يُستعاد؛
  \item وهوّ الغرف (\cref{exo:g4:breathing:10}) وفضّل هواء
        الخارج النظيف للرياضة؛
  \item ومرّن المنفاخ: فالجهد ذو \omterm{def:g7:what-is-respiration:respiration}{التنفس} العميق يبني \omterm{def:g3:bones-and-muscles:muscle}{عضلات} \omterm{def:g7:what-is-respiration:respiration}{التنفس}
        ويُبقي السطح كله مهوًّى؛
  \item \omterm{def:g7:what-is-respiration:respiration}{وتنفس} بالأنف في البرد والغبار
        (\cref{exo:g4:breathing:7}) --- فالمصفاة تحمي المكنسة.
\end{enumerate}
\end{method}

\section{مسألة: ملف آلة التنفس}

\begin{problem}[{مسألة نهاية الأسبوع --- تفتيش مهندس على منفاخ
الإنسان}]\label{pb:g7:lungs-and-gas-exchange:1}
صف هندسة ``يفتّش'' آلة \omterm{def:g7:what-is-respiration:respiration}{التنفس} البشرية على الورق: المدخل، والمضخة،
وسطح التبادل، ووصلة النقل، وسجلّ الصيانة.

\medskip\noindent\textbf{القسم الأول --- المضخة.}
\begin{enumerate}
  \item حرّر دورة المضخة: الطوران، ومحرّكا كل واحد، وأي طور يكلّف
        \omterm{prop:g3:balanced-diet:jobs}{طاقة} في الراحة.
  \item لا اتصال للمضخة \omterm{def:g3:bones-and-muscles:muscle}{بعضلة} العضو المضخوخ --- فلا \omterm{def:g3:bones-and-muscles:muscle}{عضلات}
        \omterm{def:g7:how-animals-breathe:four}{للرئتين}. فكيف يحرّك اتساع الصدر الهواء أصلًا؟
  \item أعطِ وتيرة المضخة في الراحة وتعديليها في الجهد
        (\cref{prop:g7:organs-at-work:supply}).
  \item اللهاث، والتنهد، ونفخ الشموع: أعطِ كل واحد موضعه في
        ميكانيكا الدورة.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- سطح
التبادل.}
\begin{enumerate}[resume]
  \item حدّد مواصفة السطح: الاسم، والعدد، والجدار، واللفّ،
        والحجم المجموع.
  \item وتحقق من المواصفة في مقابل
        \cref{prop:g7:how-animals-breathe:spec}، بندًا بندًا.
  \item هواء المدخل وهواء العادم يختلفان بنسب الفصل المئوية.
        احسب فرق \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} نسبةً إلى $21$ في المئة في المدخل ---
        فأي حصة من \omterm{def:g7:what-is-respiration:respiration}{أكسجين} \omterm{def:g4:breathing:movements}{الشهيق} يُحتفظ بها تقريبًا؟
  \item ولماذا يجب أن يبقى العادم دافئًا رطبًا --- وأي خاصية من
        خواصّ السطح يكون ذلك ثمنًا لها؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- سجلّ
الصيانة.}
\begin{enumerate}[resume]
  \item صنّف هجمات الدخان الأربع تحت: المدخل، والمضخة، والسطح،
        والنقل. سطر واحد لكل هجمة.
  \item يلاحظ الملف: ``خسائر السطح خسائر رأسمالية.'' اشرح ذلك في
        مقابل السعال والزكام اللذين هما إصلاحات جارية.
  \item اكتب جدول الصيانة ---
        \cref{met:g7:lungs-and-gas-exchange:keep} --- في أربعة
        توجيهات مهندس مع أسبابها.
  \item اختم التفتيش: بجملة واحدة عن سبب كون هذه الآلة، وحدها من
        بين مضخات الجسم وحدوده، نصفها تحت سيطرة صاحبها الإرادية
        --- ومن أجل ماذا تلك السيطرة (الكلام، والغناء، \omterm{def:g4:breathing:movements}{وزفير}
        الغطّاس البطيء).
\end{enumerate}
\end{problem}
