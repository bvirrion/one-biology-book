\chapter{संतुलित आहार}\label{ch:g3:balanced-diet}

निर्माण-स्थल पर ईंटें, सीमेंट, तख़्ते चाहिए --- और मशीनों के लिए ईंधन।
तुम्हारा शरीर वह निर्माण-स्थल है जो कभी बंद नहीं होता: वह बढ़ता है,
अपनी मरम्मत करता है और दिन भर दौड़ता-भागता है। उसकी सामग्री दिन में
तीन-चार बार, थाली पर आती है। उस सामग्री में क्या होना चाहिए? यही आहार
का सवाल है।

\section{शरीर को भोजन क्यों चाहिए}

\begin{proposition}[भोजन के दो काम]\label{prop:g3:balanced-diet:jobs}
भोजन दो अलग-अलग काम करता है:
\begin{enumerate}
  \item वह \emph{ऊर्जा}\index{ऊर्जा (भोजन से)} देता है --- चलने की,
        गरम रहने की, और हृदय तथा मस्तिष्क को चलाते रहने की ताक़त,
        \omterm{prop:g1:healthy-habits:needs}{नींद} के दौरान भी;
  \item वह \emph{बनाने वाला सामान} देता है --- वह सब जिससे शरीर लंबा
        होता है, \omterm{def:g3:bones-and-muscles:muscle}{पेशी} और \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} बनाता है, छिलनों की मरम्मत करता है और
        घिसे हुए हिस्से बदलता है।
\end{enumerate}
आहार को हर रोज़ दोनों पहुँचाने चाहिए।
\end{proposition}
\admitted

\begin{example}[आराम में भी ईंधन]\label{ex:g3:balanced-diet:rest}
बिलकुल निश्चल लेटे रहने पर भी तुम \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} ख़र्च करते हो: हृदय धड़कता है,
छाती उठती-गिरती है, और मौसम चाहे जैसा हो, शरीर लगभग
$\qty{37}{\degreeCelsius}$ पर गरम बना रहता है। यह जो लगातार ख़र्च चलता
रहता है, उसी की वजह से खाना छोड़ने पर सिर्फ़ भूख नहीं लगती, कमज़ोरी और
ठंड भी लगती है।
\end{example}

\section{आहार-वर्ग}

\begin{definition}[आहार-वर्ग]\label{def:g3:balanced-diet:families}
भोजन को इस आधार पर \emph{आहार-वर्गों}\index{आहार-वर्ग} में छाँटा जाता
है कि वह क्या लाता है:
\begin{itemize}
  \item \emph{\omterm{prop:g3:flowers-fruits-seeds:transform}{फल} और सब्ज़ियाँ} --- विटामिन, रेशा और पानी; ताज़गी वाला
        वर्ग;
  \item \emph{अनाज और स्टार्च वाले भोजन} (रोटी, पास्ता, चावल, आलू)
        --- टिकाऊ \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} वाला वर्ग;
  \item \emph{\omterm{def:g2:animals-grow-up:born}{दूध} से बनी चीज़ें} (\omterm{def:g2:animals-grow-up:born}{दूध}, दही, पनीर) --- \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} बनाने वाला
        वर्ग;
  \item \emph{माँस, मछली और \omterm{def:g2:animals-grow-up:born}{अंडे}} --- शरीर बनाने वाला वर्ग;
  \item \emph{चिकनाइयाँ} (मक्खन, तेल) --- गाढ़ी \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा}, थोड़ी मात्रा
        में ज़रूरी;
  \item \emph{मीठे भोजन} --- झटपट मज़ा, ज़रूरत बिलकुल नहीं: चटपटा
        वर्ग;
  \item और \emph{पानी}\index{पानी (पेय)} --- शरीर को दिन भर ज़रूरी
        अकेला पेय।
\end{itemize}
\end{definition}

\begin{example}[दोपहर के खाने को संतुलित करना]\label{ex:g3:balanced-diet:lunch}
टमाटर की तरी वाला पास्ता, मछली का एक टुकड़ा, हरी फलियाँ, एक दही, एक
सेब, पानी: हर वर्ग मौजूद, चटपटी चीज़ें ग़ैरहाज़िर --- यानी संतुलित
दोपहर का खाना। तले चिप्स, एक ठंडा पेय और चॉकलेट की एक पट्टी ठीक दो ही
वर्गों (चिकनाई और चीनी) का पोषण देते हैं और उन्हें छोड़ देते हैं
जिनका निर्माण-स्थल इंतज़ार कर रहा था।
\end{example}

\begin{remark}[कोई भोजन मना नहीं, बस भुलाया हुआ]\label{rem:g3:balanced-diet:balance}
संतुलन दिनों के पैमाने पर नापा जाता है, एक-एक कौर के नहीं। जन्मदिन के
केक का एक टुकड़ा कोई नियम नहीं तोड़ता --- मुश्किल कभी एक मिठाई से नहीं
आती, वह तब आती है जब दिन दर दिन मिठाइयाँ वर्गों की \emph{जगह} ले लें।
हर खाने पर काम का सवाल यह नहीं कि ``क्या यह भोजन बुरा है?'' बल्कि यह
कि ``आज मैं कौन-सा वर्ग भूल गया?''
\end{remark}

\section{भोजन को दिन के हिसाब से मिलाना}

\begin{proposition}[ज़रूरतें दिन और व्यक्ति के साथ बदलती हैं]\label{prop:g3:balanced-diet:needs}
शरीर को कितनी \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} चाहिए, यह इस पर निर्भर है कि वह किसका है और क्या
करता है: बढ़ते बच्चे को बड़े के मुक़ाबले अनुपात में ज़्यादा बनाने वाला
सामान चाहिए; खेलकूद का दिन या सर्दी की ठंड पढ़ाई की दोपहर से ज़्यादा
\omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} जलाती है। भूख शरीर के ख़र्च के साथ बढ़ती है।
\end{proposition}
\admitted

\begin{example}[नाश्ता क्यों मायने रखता है]\label{ex:g3:balanced-diet:breakfast}
सुबह तक शरीर पूरी रात चल चुका होता है --- हृदय, साँस, गरमाहट --- और
इस बीच उसे कोई सामान नहीं मिला होता। नाश्ता उसे सुबह के पाठों और
दौड़-भाग के लिए फिर से भर देता है। इसे छोड़ दो, और आधी सुबह तक
निर्माण-स्थल में कमी पड़ जाती है: पेट गुड़गुड़ाता है, ध्यान भटकता है,
खेल में ताक़त नहीं रहती।
\end{example}

\begin{method}[दिन का संतुलन जाँचना]\label{met:g3:balanced-diet:check}
दिन के अंत में इस सूची पर नज़र दौड़ाओ:
\begin{enumerate}
  \item हर खाने में \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} या सब्ज़ियाँ? (लक्ष्य: दिन में लगभग पाँच बार)
  \item हर खाने में टिकाऊ \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} के लिए स्टार्च वाला भोजन?
  \item बनाने वाले मौजूद --- \omterm{def:g2:animals-grow-up:born}{दूध} से बनी चीज़ें कुछ बार, और माँस, मछली
        या \omterm{def:g2:animals-grow-up:born}{अंडे} एक-दो बार?
  \item दिन के पेय के रूप में पानी?
  \item चटपटी चीज़ें: थोड़ी-सी, मुख्य हिस्सा नहीं?
\end{enumerate}
जो भी वर्ग छूट गया हो, उसे कल बुला लो।
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:1}
भोजन के दोनों काम क्या हैं? हर काम को मुख्य रूप से पूरा करने वाला एक
\omterm{def:g3:balanced-diet:families}{आहार-वर्ग} बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:2}
\omterm{def:g3:balanced-diet:families}{आहार-वर्गों} के नाम बताओ, और वह अकेला पेय भी जो शरीर को ज़रूरी है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:3}
इस सौदे को वर्गों में छाँटो: चावल, दही, गाजर, \omterm{def:g2:animals-grow-up:born}{अंडे}, मक्खन, सेब, रोटी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:4}
\omterm{prop:g1:healthy-habits:needs}{नींद} के दौरान भी शरीर \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} क्यों ख़र्च करता है? रात के दो ख़र्च
बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:5}
``पास्ता, मछली, हरी फलियाँ, दही, सेब, पानी'' वाले दोपहर के खाने को
चित्र की संतुलित थाली से जाँचो: हर चीज़ थाली के किस हिस्से में आती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:6}
नाश्ता ख़ास तौर पर क्यों ज़रूरी है? उसके बिना आधी सुबह तक क्या होता
है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:7}
अनुपात में ज़्यादा बनाने वाला सामान किसे चाहिए: बच्चे को या बड़े को?
क्यों?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:8}
करके देखो: अपने ही दिन पर \cref{met:g3:balanced-diet:check} ईमानदारी
से चलाओ। तुम कौन-सा वर्ग भूल गए --- और कौन-सा अपने हिस्से से ज़्यादा
आया?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:balanced-diet:9}
``मिठाइयाँ मना हैं!'' क्या यह अध्याय यही कहता है?
\cref{rem:g3:balanced-diet:balance} का ज़्यादा सटीक नियम बताओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:balanced-diet:10}
बर्फ़ पर छुट्टियाँ बिताते समय सबको घर से ज़्यादा भूख लगती है।
\cref{prop:g3:balanced-diet:needs} से समझाओ --- और वे दोनों अतिरिक्त
ख़र्च बताओ जो ठंडा, खेलकूद वाला दिन जोड़ देता है।
\end{exercise}
