\section*{Chapter \ref{ch:g9:biotech} --- Biology and Technology}

\begin{solution}{exo:g9:biotech:1}
Using living systems to make products/solve problems. Example: yeast bread,
yoghurt, cheese, brewing.
\end{solution}

\begin{solution}{exo:g9:biotech:2}
Human insulin \omterm{def:g9:dna:gene}{gene} inserted into \omterm{def:g4:microbes:def}{microbes} that produce insulin \omterm{def:g6:nutrition:groups}{protein} in
fermenters.
\end{solution}

\begin{solution}{exo:g9:biotech:3}
Matching biological samples (forensics, identification, kinship).
\end{solution}

\begin{solution}{exo:g9:biotech:4}
\omterm{prop:g9:biotech:breed}{Selective breeding}: choose parents over generations. GM: directly alter
\omterm{def:g9:dna:dna}{DNA} (often specific \omterm{def:g9:dna:gene}{genes}) in a targeted way.
\end{solution}

\begin{solution}{exo:g9:biotech:5}
