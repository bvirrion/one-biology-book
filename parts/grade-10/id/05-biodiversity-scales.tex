\chapter{Keanekaragaman Hayati pada Setiap Skala}\label{ch:g10:biodiversity-scales}

Bentangkan seutas tali membentuk persegi satu meter di halaman rumput
dan hitung apa yang tumbuh di dalamnya: rumput, tentu saja, tetapi juga
semanggi, aster, jombang, daun sendok, lumut, dan, kalau kamu menengok
dengan lensa, selusin serangga serta seekor siput. Pindahkan perseginya
ke aspal tempat parkir: tidak ada apa-apa, atau satu gulma di sebuah
retakan. Pindahkan ke padang rumput tua yang belum pernah dibajak: empat
puluh \omterm{def:g10:biodiversity-scales:species}{spesies} tumbuhan, dan serangga yang tak terhitung. Kata untuk apa
yang diukur persegi itu adalah \emph{\omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman hayati}}, dan bab ini
membahas bagaimana mengukurnya, pada skala apa saja ia ada, dan mengapa
ia tidak tetap dan tidak terjamin.

\section{Tiga skala keragaman}

\begin{definition}[Keanekaragaman hayati]\label{def:g10:biodiversity-scales:biodiversity}
Yang disebut \emph{keanekaragaman hayati}\index{keanekaragaman hayati}
adalah keragaman makhluk hidup, yang ditinjau pada tiga skala:
\begin{itemize}
  \item keragaman \emph{ekosistem}\index{ekosistem} --- yaitu keragaman
        masyarakat hidup dan lingkungan yang ditempatinya (sebuah
        kolam, hutan beech, terumbu karang, padang rumput),
        masing-masing dengan himpunan \omterm{def:g10:biodiversity-scales:species}{spesiesnya} sendiri;
  \item keragaman \omterm{def:g10:biodiversity-scales:species}{spesies} di dalam sebuah ekosistem atau di Bumi ---
        yaitu jumlah \omterm{def:g10:biodiversity-scales:species}{spesies} dan kelimpahan nisbinya;
  \item \emph{keanekaragaman genetik}\index{keanekaragaman genetik} di
        dalam satu \omterm{def:g10:biodiversity-scales:species}{spesies} --- yaitu jumlah \omterm{def:g10:universal-dna:gene}{alel} \omterm{def:g10:universal-dna:gene}{gennya} dan cara \omterm{def:g10:universal-dna:gene}{alel}
        itu terbagi di antara individu dan populasinya.
\end{itemize}
\end{definition}

\begin{definition}[Spesies]\label{def:g10:biodiversity-scales:species}
Sebuah \emph{spesies}\index{spesies} adalah kelompok makhluk hidup yang
saling menyerupai, dapat berbiak bersama, dan menghasilkan keturunan
yang subur; dua populasi yang tidak dapat saling berbiak, atau yang
hibridnya mandul, merupakan spesies yang berbeda. Dalam praktiknya ---
untuk fosil, untuk bakteri, untuk organisme yang tak pernah terlihat
berbiak --- spesies dikenali dari ciri bersamanya, dan
\cref{ch:g12:selection-drift-speciation} akan menunjukkan bahwa
batasnya tidak selalu tegas.
\end{definition}

\begin{example}[Kuda, keledai, bagal]\label{ex:g10:biodiversity-scales:mule}
Kuda dan keledai saling menyerupai dan dapat berbiak: keturunannya
adalah bagal. Tetapi bagal mandul, sehingga kuda dan keledai adalah dua
\omterm{def:g10:biodiversity-scales:species}{spesies}. Anjing pudel dan anjing pemburu serigala sama sekali tidak
mirip, mudah berbiak, dan memberi anak anjing yang subur: satu \omterm{def:g10:biodiversity-scales:species}{spesies},
yaitu anjing --- yang juga dapat berbiak dengan serigala, moyangnya.
\end{example}

\section{Keragaman spesies}

\begin{proposition}[Menghitung spesies]\label{prop:g10:biodiversity-scales:count}
Sekitar dua juta \omterm{def:g10:biodiversity-scales:species}{spesies} telah dilukiskan dan diberi nama. Sebagian
besar berupa serangga (lebih dari sejuta), disusul tumbuhan (sekitar
300\,000), jamur, arakhnida, moluska; vertebrata, kelompok yang paling
kita kenal, berjumlah sekitar 70\,000, yang 6\,400 di antaranya
mamalia. Jumlah \omterm{def:g10:biodiversity-scales:species}{spesies} yang belum dilukiskan ditaksir, dari laju
penemuan \omterm{def:g10:biodiversity-scales:species}{spesies} baru, beberapa juta lagi --- barangkali delapan sampai
sepuluh juta \omterm{def:g10:biodiversity-scales:species}{spesies} eukariot seluruhnya, ditambah keragaman bakteri
yang luas dan sebagian besar belum terhitung.
\end{proposition}
\admitted

\begin{method}[Mengukur keragaman sebuah lokasi]\label{met:g10:biodiversity-scales:sampling}
Sebuah lokasi tidak pernah dihitung seluruhnya; ia \emph{dicuplik}.
\begin{enumerate}
  \item Pilih satuan cuplikan yang serasi dengan organismenya: petak
        \qty{1}{m^2} untuk tumbuhan padang rumput, sapuan jaring untuk
        serangga, ayakan tanah untuk cacing tanah, sepanjang transek
        untuk burung.
  \item Tempatkan satuannya secara acak atau pada selang yang teratur,
        cukup banyak sehingga menambah satu lagi hanya sedikit mengubah
        jumlahnya.
  \item Pada tiap satuan, kenali dan hitung. Dua bilangan melukiskan
        cuplikannya: \emph{kekayaan \omterm{def:g10:biodiversity-scales:species}{spesies}}, yaitu jumlah \omterm{def:g10:biodiversity-scales:species}{spesies} yang
        ditemukan, dan \emph{kelimpahan nisbi} masing-masing --- lokasi
        yang satu \omterm{def:g10:biodiversity-scales:species}{spesiesnya} menyusun 95\% individu kurang beragam
        daripada lokasi yang sepuluh \omterm{def:g10:biodiversity-scales:species}{spesiesnya} berbagi individu secara
        merata, walaupun kekayaannya sama.
  \item Bandingkan lokasi hanya dengan cara yang sama dan usaha yang
        sama, pada musim yang sama.
\end{enumerate}
\end{method}

\begin{example}[Dua halaman rumput]\label{ex:g10:biodiversity-scales:lawns}
Sepuluh petak di halaman rumput sekolah yang dipangkas: 6 \omterm{def:g10:biodiversity-scales:species}{spesies}
tumbuhan, dengan rumput menutupi 90\%. Sepuluh petak di tepian jalan
tak terpangkas di sebelahnya: 23 \omterm{def:g10:biodiversity-scales:species}{spesies}, tidak ada yang di atas 30\%.
Tepian itu lebih kaya dan lebih merata; bedanya bukan tanahnya atau
iklimnya, yang sama, melainkan pemangkasannya, yang menyingkirkan tiap
tumbuhan yang tidak dapat berbunga dekat ke tanah.
\end{example}

\section{Keanekaragaman genetik di dalam satu spesies}

\begin{proposition}[Individu satu spesies tidak serupa]\label{prop:g10:biodiversity-scales:genetic}
Di dalam satu \omterm{def:g10:biodiversity-scales:species}{spesies}, sebagian besar \omterm{def:g10:universal-dna:gene}{gen} hadir dalam beberapa \omterm{def:g10:universal-dna:gene}{alel},
dan individunya membawa gabungan \omterm{def:g10:universal-dna:gene}{alel} yang berbeda-beda.
\emph{Frekuensi} sebuah \omterm{def:g10:universal-dna:gene}{alel} dalam sebuah populasi --- yaitu bagian
seluruh salinan \omterm{def:g10:universal-dna:gene}{gennya} yang berupa \omterm{def:g10:universal-dna:gene}{alel} tersebut --- berbeda dari satu
populasi ke populasi lain. \omterm{def:g10:biodiversity-scales:biodiversity}{Keanekaragaman genetik} itulah yang
memungkinkan sebuah \omterm{def:g10:biodiversity-scales:species}{spesies} menanggapi lingkungan yang berubah:
populasi yang setiap individunya serupa tidak punya apa pun untuk dipilih.
\end{proposition}
\admitted

\begin{example}[Alel golongan darah di seluruh dunia]\label{ex:g10:biodiversity-scales:abo}
\omterm{def:g10:universal-dna:gene}{Gen} golongan darah ABO punya tiga \omterm{def:g10:universal-dna:gene}{alel} yang lazim, yaitu A, B dan O.
Di Eropa barat frekuensinya sekitar 0.27, 0.07 dan 0.66; di Asia timur
sekitar 0.20, 0.20 dan 0.60; pada sebagian populasi asli Amerika
Selatan \omterm{def:g10:universal-dna:gene}{alel} O sendirian mencapai 0.98. \omterm{def:g10:biodiversity-scales:species}{Spesies} sama, \omterm{def:g10:universal-dna:gene}{gen} sama, sebaran
\omterm{def:g10:universal-dna:gene}{alelnya} berbeda: \omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman genetik} manusia itu nyata, dan ia
terbentang lintas populasi alih-alih memisahkannya --- sebagian besar
keragaman \omterm{def:g10:biodiversity-scales:species}{spesiesnya} justru ditemukan di dalam satu populasi mana pun.
\end{example}

\begin{example}[Spesies yang hampir tidak punya]\label{ex:g10:biodiversity-scales:cheetah}
Citah begitu serupa secara genetik sehingga cangkok kulit antarhewan
yang tak berkerabat diterima seolah-olah antarkembar: \omterm{def:g10:biodiversity-scales:species}{spesies} itu
pernah melewati leher botol yang hanya berisi sangat sedikit individu
sekitar sepuluh ribu tahun lalu dan tidak pernah memulihkan \omterm{def:g10:universal-dna:gene}{alelnya}.
Penyakit atau iklim baru apa pun yang datang, setiap citah
menghadapinya dengan perlengkapan yang sama; \omterm{def:g10:biodiversity-scales:species}{spesiesnya} bertahan,
tetapi tanpa cadangan apa pun.
\end{example}

\section{Keanekaragaman hayati berubah: dahulu dan sekarang}

\begin{proposition}[Keanekaragaman hayati adalah keadaan, bukan tetapan]\label{prop:g10:biodiversity-scales:changes}
Himpunan \omterm{def:g10:biodiversity-scales:species}{spesies} yang hidup pada suatu saat adalah satu tahap dalam
sebuah sejarah. Rekaman fosil memperlihatkan \omterm{def:g10:biodiversity-scales:species}{spesies} bermunculan,
bertahan rata-rata beberapa juta tahun, lalu lenyap; keragaman
kehidupan seluruhnya meningkat sepanjang 540 juta tahun terakhir,
diselingi lima \emph{kepunahan massal}\index{kepunahan massal} yang
lebih dari tiga perempat \omterm{def:g10:biodiversity-scales:species}{spesiesnya} lenyap dalam waktu yang singkat
secara geologi. Yang paling akhir, 66 juta tahun lalu, mengakhiri
dinosaurus bukan burung dan membuka jalan bagi penganekaragaman mamalia.
\end{proposition}
\begin{proof}[Bukti percobaan]
Penghitungan famili fosil dalam lapisan batuan yang bertarikh, yang
disusun untuk seluruh rekaman laut, memberi lengkung pada gambar
berikutnya: sebuah kenaikan dari beberapa ratus menjadi hampir seribu
famili, dengan penurunan tajam pada 445, 370, 252, 201 dan 66 juta
tahun lalu. Peristiwa 252 juta tahun lalu, yang terbesar,
menyingkirkan sekitar 95\% \omterm{def:g10:biodiversity-scales:species}{spesies} laut; lapisan tepat di atasnya
hampir kosong dari fosil, dan kelompok baru mengisinya hanya setelah
jutaan tahun berikutnya.
\end{proof}

\begin{proposition}[Perubahan masa kini]\label{prop:g10:biodiversity-scales:present}
\omterm{def:g10:biodiversity-scales:species}{Spesies} kini lenyap dengan laju yang ditaksir seratus sampai seribu
kali rata-rata rekaman fosil, akibat perusakan \omterm{def:g10:biodiversity-scales:biodiversity}{ekosistem} (penebangan
hutan, pengeringan lahan basah, pembangunan kota), akibat pemanfaatan
berlebihan, akibat \omterm{def:g10:biodiversity-scales:species}{spesies} yang diperkenalkan, akibat pencemaran dan
akibat perubahan iklim --- semuanya berasal dari manusia. Separuh hutan
tropis tahun 1950 telah lenyap; sepertiga \omterm{def:g10:biodiversity-scales:species}{spesies} amfibi terancam;
\omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman genetik} tanaman budi daya dan ternak menyempit menjadi
beberapa varietas saja. \omterm{def:g10:biodiversity-scales:biodiversity}{Keanekaragaman hayati} adalah sebuah sumber daya
--- pangan, obat, penyerbukan, tanah, kemantapan \omterm{def:g10:biodiversity-scales:biodiversity}{ekosistem} --- dan
hilangnya tidak dapat dibalik pada skala waktu manusia.
\end{proposition}
\admitted

\begin{example}[Membalikkan sebuah kehilangan]\label{ex:g10:biodiversity-scales:reversing}
Di tempat pagar tanaman yang disingkirkan pada 1960-an ditanam kembali,
kekayaan burung dan serangga kembali dalam sepuluh tahun; di tempat
serigala dilepasliarkan kembali ke sebuah taman nasional pada 1995,
rusa yang diburunya berhenti menggunduli dedalu tepi sungai, dedalunya
tumbuh lagi, berang-berang kembali, dan burung serta ikan yang
bergantung pada kolam berang-berang ikut kembali. Melindungi atau
memulihkan sebuah \omterm{def:g10:biodiversity-scales:biodiversity}{ekosistem} memulihkan keragaman pada ketiga skalanya
sekaligus; \omterm{def:g10:biodiversity-scales:species}{spesies} yang dipelihara hidup di kebun binatang hanya
menjaga salah satunya.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:1}
Sebutkan ketiga skala \omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman hayati} dan berikan satu contoh masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:2}
Rumuskan \omterm{def:g10:biodiversity-scales:species}{spesies}. Apakah singa dan harimau --- yang dapat menghasilkan
hibrid mandul dalam penangkaran --- satu \omterm{def:g10:biodiversity-scales:species}{spesies} atau dua?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:3}
Dari bagan \omterm{def:g10:biodiversity-scales:species}{spesiesnya}, berapa kali lipat \omterm{def:g10:biodiversity-scales:species}{spesies} serangga yang telah
dilukiskan dibandingkan \omterm{def:g10:biodiversity-scales:species}{spesies} mamalia?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:4}
Apa itu frekuensi \omterm{def:g10:universal-dna:gene}{alel}? Dalam populasi yang \omterm{def:g10:universal-dna:gene}{alel} O-nya berfrekuensi
0.66 dan \omterm{def:g10:universal-dna:gene}{alel} A-nya 0.27, berapa frekuensi B?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:5}
Berapa \omterm{prop:g10:biodiversity-scales:changes}{kepunahan massal} yang diperlihatkan rekaman fosil, dan yang mana
yang terbesar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:6}
Dua padang rumput masing-masing punya 12 \omterm{def:g10:biodiversity-scales:species}{spesies} tumbuhan. Pada yang
pertama, satu \omterm{def:g10:biodiversity-scales:species}{spesies} menyusun 80\% tumbuhannya; pada yang kedua, tidak
ada \omterm{def:g10:biodiversity-scales:species}{spesies} yang melampaui 15\%. Mana yang lebih beragam, dan bilangan
mana dari kedua bilangan \cref{met:g10:biodiversity-scales:sampling}
yang membedakan keduanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:7}
Seorang murid menghitung serangga dalam 3 sapuan jaring dan menemukan 8

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:8}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:9}
Pada gambar famili fosilnya, bacalah jumlah famili tepat sebelum dan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:10}
Dengan \cref{ex:g10:biodiversity-scales:abo}, jelaskan dalam arti apa

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:11}
Sebuah kolam dikeringkan untuk membangun tempat parkir. Sebutkan satu

\section{Soal: Laba dari Pagar Tanaman}

\begin{problem}[{Soal akhir pekan --- survei lapangan dua ladang, satu
berpagar tanaman dan satu tanpa: spesies yang terhitung, kelimpahan
yang dibandingkan, alel seekor siput yang diikuti, dan angka yang
setara dengan sebuah pagar}]
\label{pb:g10:biodiversity-scales:1}
Dua ladang bertetangga menanam gandum yang sama pada tanah yang sama.
Ladang A menjaga pagar tanamannya; Ladang B menyingkirkannya pada 1970.
Sebuah kelas mencuplik keduanya pada bulan Juni dengan cara yang sama:
20 petak \qty{1}{m^2} di sepanjang tepi ladang untuk tumbuhan, 20
sapuan jaring untuk serangga, dan 10 menit menyimak pada 5 titik untuk burung.

\medskip
\noindent\textbf{Bagian I --- Tumbuhannya.}
Hasilnya, Ladang A: 31 \omterm{def:g10:biodiversity-scales:species}{spesies} tumbuhan dalam 20 petak; Ladang B: 9.
Di Ladang B, satu jenis rumput menyusun 84\% tumbuhan yang terhitung;
di Ladang A \omterm{def:g10:biodiversity-scales:species}{spesies} yang terlazim menyusun 22\%.
\begin{enumerate}
  \item Sebutkan kekayaan \omterm{def:g10:biodiversity-scales:species}{spesies} tepi tiap ladang dan nisbah antara
        keduanya.
  \item Masyarakat tumbuhan ladang mana yang lebih merata? Jelaskan apa
        yang ditambahkan ``merata'' pada ``kaya''.
  \item Di Ladang A, jumlah \omterm{def:g10:biodiversity-scales:species}{spesies} yang ditemukan setelah 5, 10, 15
        dan 20 petak berturut-turut 18, 25, 29 dan 31. Apakah
        pencuplikannya memadai? Berikan alasannya.
  \item Deret yang sama di Ladang B adalah 6, 8, 9, 9. Bandingkan kedua
        deret itu dan sebutkan bentuk apa yang diambil lengkung
        pencuplikan yang ``lengkap''.
  \item Mengapa kedua ladang harus dicuplik pada bulan yang sama?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Serangga dan burungnya.}
Serangga: Ladang A 46 \omterm{def:g10:biodiversity-scales:species}{spesies}, Ladang B 14. Burung: Ladang A 17
\omterm{def:g10:biodiversity-scales:species}{spesies}, Ladang B 5; di Ladang B, 60\% burung yang terdengar berasal
dari satu \omterm{def:g10:biodiversity-scales:species}{spesies} saja.
\begin{enumerate}[resume]
  \item Hitung nisbah kekayaan serangga dan kekayaan burung antara
        kedua ladang itu. Apakah ketiga kelompok (tumbuhan, serangga,
        burung) menceritakan hal yang sama?
  \item Usulkan rantai sebab yang menghubungkan penyingkiran pagar
        tanaman dengan turunnya keragaman serangga, lalu dengan
        keragaman burung.
  \item Hasil gandum tiap hektar sama pada kedua ladang. Apakah ini
        berarti pagarnya tidak berpengaruh pada panennya? Usulkan satu
        jasa yang mungkin diberikan sebuah pagar yang tidak
        diperlihatkan angka panennya.
  \item Sebutkan satu \omterm{def:g10:biodiversity-scales:species}{spesies} yang diharapkan kelas itu hanya
        ditemukan di Ladang A dan jelaskan mengapa.
  \item Skala \omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman hayati} mana yang paling kentara berkurang
        di Ladang B: \omterm{def:g10:biodiversity-scales:biodiversity}{ekosistem}, \omterm{def:g10:biodiversity-scales:species}{spesies}, atau \omterm{def:g10:universal-dna:gene}{gen}? Berikan alasannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- \omterm{def:g10:universal-dna:gene}{Alel} seekor siput.}
Siput kebun membawa sebuah \omterm{def:g10:universal-dna:gene}{gen} dengan dua \omterm{def:g10:universal-dna:gene}{alel} yang lazim untuk warna
cangkang, yaitu kuning (y) dan cokelat (b). Di pagar Ladang A, 400
siput dicuplik: 100 punya dua \omterm{def:g10:universal-dna:gene}{alel} y, 200 punya satu y dan satu b, 100
punya dua b. Di satu-satunya tanggul yang tersisa di Ladang B, 100
siput: 4 dengan dua y, 32 dengan satu dari masing-masing, 64 dengan dua b.
\begin{enumerate}[resume]
  \item Hitung salinan y dan salinan b pada tiap cuplikan (tiap siput
        membawa dua salinan). Turunkan frekuensi y pada tiap ladang.
  \item Populasi mana yang lebih beragam secara genetik pada \omterm{def:g10:universal-dna:gene}{gen} ini?
        Bandingkan bagian siput yang membawa kedua \omterm{def:g10:universal-dna:gene}{alelnya}.
  \item Burung ciung memburu siput dengan penglihatan; pada tanggul
        gundul Ladang B, cangkang cokelat lebih tersembunyi. Usulkan
        bagaimana frekuensi Ladang B itu muncul.
  \item Jika pagar yang lembap dan teduh menguntungkan cangkang kuning,
        apa yang kamu ramalkan bagi siput Ladang B seandainya pagarnya
        ditanam kembali dan tidak ada lagi yang berubah?
  \item Siput Ladang B berjumlah beberapa ratus; siput Ladang A puluhan
        ribu. Populasi mana yang lebih terbuka terhadap kehilangan satu
        \omterm{def:g10:universal-dna:gene}{alel} sama sekali secara kebetulan? Mengapa?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Labanya.}
\begin{enumerate}[resume]
  \item Untuk ketiga kelompok itu, hitung kekayaan \omterm{def:g10:biodiversity-scales:species}{spesies} seluruhnya
        pada tiap ladang, lalu jumlah Ladang A dibagi jumlah Ladang B.
  \item Pagar tanaman Ladang A menempati 3\% luasnya. Nyatakan dalam
        satu kalimat berapa bagian lahannya yang memuat berapa bagian
        \omterm{def:g10:biodiversity-scales:species}{spesiesnya}.
  \item Pagar yang ditanam kembali di Ladang B pada 1995 disurvei ulang
        pada 2010: tumbuhan 24, serangga 35, burung 12 \omterm{def:g10:biodiversity-scales:species}{spesies}. Berapa
        bagian kekayaan Ladang A yang telah dipulihkan tiap kelompok?
  \item Skala mana dari ketiga skala \omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman hayati} yang paling
        lambat pulih setelah penanaman kembali, dan mengapa?
  \item Nyatakan hasilnya: faktor yang dengannya pagar tanaman
        melipatgandakan kekayaan \omterm{def:g10:biodiversity-scales:species}{spesies} sebuah ladang, dan satu skala
        \omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman hayati} yang tidak dapat dikembalikan sendirian
        oleh pagar yang ditanam kembali.
\end{enumerate}
\end{problem}
