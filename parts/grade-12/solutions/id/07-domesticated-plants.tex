\section*{Bab \ref{ch:g12:domesticated-plants} --- Tumbuhan yang Dijinakkan}

\begin{solution}{exo:g12:domesticated-plants:1}
\omterm{def:g12:domesticated-plants:domestication}{Domestikasi}: perubahan sebuah \omterm{def:g10:biodiversity-scales:species}{spesies} liar menjadi \omterm{def:g10:biodiversity-scales:species}{spesies} yang
bergantung pada manusia dan melayaninya. \omterm{def:g12:domesticated-plants:domestication}{Seleksi buatan}: manusia
menyimpan dan menyemai individu yang bersifat seperti yang mereka
inginkan, sehingga \omterm{def:g12:selection-drift-speciation:population}{frekuensi alel} di baliknya naik.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:2}
Tongkol yang tak rontok (tanpa persebaran); perkecambahan seketika
(tanpa menunggu lewatnya tahun yang buruk); hilangnya racun (dimakan
segalanya); satu tongkol besar pada tanaman tak bercabang (semua telur
dalam satu keranjang); pemasakan serentak (satu pekan yang buruk
memusnahkan seluruh panen).
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:3}
Keturunan seragam dan kuat dari dua galur murni yang homozigot.
Benihnya sendiri memilah menjadi campuran tanaman yang kurang kuat,
sehingga petani membeli benih hibrida yang baru setiap tahun.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:4}
\omterm{def:g10:universal-dna:gene}{Gennya} disisipkan ke dalam \omterm{def:g10:universal-dna:information}{DNA} yang secara alami dipindahkan sejenis
bakteri tanah ke dalam \omterm{def:g10:cells-common-unit:cell}{sel} tumbuhan; \omterm{def:g10:cells-common-unit:cell}{sel} terinfeksi yang menerima \omterm{def:g10:universal-dna:gene}{gen}
itu ditumbuhkan kembali menjadi tumbuhan utuh.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:5}
Dari sekitar \num{1.9} menjadi \qty{7.2}{t/ha}: nyaris empat kali.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:6}
Tanaman yang butirnya rontok kehilangan butir itu sebelum atau selama
panen; mutan bertangkai liat mempertahankan butirnya dan menjadi yang
dipungut, jadi juga yang disemai. Pemanenan itu sendiri menyeleksi
\omterm{def:g10:universal-dna:gene}{alel} tangkai liat, generasi demi generasi.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:7}
Satu \omterm{def:g10:biodiversity-scales:species}{spesies}: ketiganya saling kawin dan memberi keturunan yang subur.
Petani menyeleksi pembesaran organ yang berbeda pada tiap garis
keturunan --- daun, tunas ujung, kuncup bunga --- sehingga ketiganya
berbeda dalam pengaturan pertumbuhan, bukan dalam kemampuan saling
menyilang.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:8}
F1 seluruhnya $Aa$. F2: $AA$ $1/4$, $Aa$ $1/2$, $aa$ $1/4$. F1 seragam
karena setiap tanaman menerima $A$ dari satu induk dan $a$ dari induk
yang lain; pada F2 \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis} dan pembuahan membagikan \omterm{def:g10:universal-dna:gene}{alelnya} secara
acak.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:9}
Di antara jutaan penggerek, beberapa membawa \omterm{def:g11:mutations:mutation}{mutasi} yang membuatnya
\omterm{prop:g11:antibiotic-resistance:mechanisms}{tahan} terhadap racun itu; di ladang yang setiap tanamannya membuat
racun, hanya merekalah yang selamat dan berbiak --- seleksi. Sebuah
suaka menjaga banyak penggerek peka tetap hidup; perkawinannya dengan
penggerek \omterm{prop:g11:antibiotic-resistance:mechanisms}{tahan} yang jarang itu memberi keturunan yang membawa satu
\omterm{def:g10:universal-dna:gene}{alel} ketahanan, yang (karena \omterm{def:g11:genes-and-disease:genetic}{resesif}) mati di jagung beracun, sehingga
\omterm{def:g10:universal-dna:gene}{alel} itu terencerkan setiap generasi.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:10}
\omterm{def:g10:universal-dna:gene}{Gen} yang bersangkutan mengendalikan perkembangan --- percabangan
tanaman, pertumbuhan sarung butirnya, jumlah barisnya; perubahan pada
pengaturan satu \omterm{def:g10:universal-dna:gene}{gen} semacam itu mengubah bentuk sebuah organ utuh.
Sedikit perubahan pengatur, akibat tampak yang besar.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:11}
Kerabat liar memuat \omterm{def:g10:universal-dna:gene}{alel} yang disingkirkan seleksi dari tanaman
budidaya: ketahanan terhadap penyakit dan hama, toleransi terhadap
kekeringan, garam atau dingin. Ketika penyakit atau iklim baru datang,
pemulia menyilangkan \omterm{def:g10:universal-dna:gene}{alel} itu kembali ke tanaman budidayanya; sekali
kerabat liarnya punah, \omterm{def:g10:universal-dna:gene}{alelnya} lenyap.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:12}
Serbuk sari kanola membuahi sesawi itu; hibridanya membawa \omterm{def:g10:universal-dna:gene}{gen}
toleransi; setelah disilangbalikkan ke sesawi sepanjang beberapa
generasi dan diseleksi oleh herbisida yang disemprotkan di ladang,
sesawi liar yang toleran pun menyebar. Prosesnya: hibridisasi dan
perpindahan mendatar sebuah \omterm{def:g10:universal-dna:gene}{gen} antarpopulasi, lalu \omterm{def:g12:selection-drift-speciation:selection}{seleksi alam} oleh
herbisidanya.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:13}
Biologis: \omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} tanamannya dan kemampuannya mengisi butir sudah
mendekati batasnya, dan keragaman genetik varietas unggul, yang
darinya kenaikan selanjutnya harus datang, tergolong sempit.
Nonbiologis: pemakaian pupuk dan pestisida tak lagi meningkat, atau
malah dikurangi, dan tekanan iklim bertambah. Seorang pemulia mungkin
mendatangkan \omterm{def:g10:universal-dna:gene}{alel} dari kerabat liar, atau menyunting \omterm{def:g10:universal-dna:gene}{gen} \omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis}
atau \omterm{def:g10:universal-dna:gene}{gen} toleransi kekeringan.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:14}
Penyilangan: \omterm{def:g10:universal-dna:gene}{alelnya} harus ada pada tumbuhan yang dapat disilangkan
dengan tanaman budidayanya; ribuan \omterm{def:g10:universal-dna:gene}{gen} dikocok ulang dan harus dipilah
oleh seleksi sepanjang sekitar sepuluh generasi. \omterm{prop:g10:universal-dna:universal}{Transgenesis}: \omterm{def:g10:universal-dna:gene}{gennya}
boleh berasal dari makhluk mana pun; satu \omterm{def:g10:universal-dna:gene}{gen} ditambahkan dan sisa
genomnya tak tersentuh; hanya perlu beberapa tahun.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:15}
Hitungannya identik: individu yang membawa \omterm{def:g10:universal-dna:gene}{alel} tertentu bereproduksi
lebih banyak, dan \omterm{def:g12:selection-drift-speciation:population}{frekuensi alel} itu naik. Yang berbeda adalah
pelakunya --- pilihan manusia menggantikan lingkungan --- dan akibatnya
bagi tumbuhan itu: \omterm{def:g12:selection-drift-speciation:selection}{seleksi alam} menjaga sebuah \omterm{def:g10:biodiversity-scales:species}{spesies} tetap mampu
hidup sendiri, sedangkan \omterm{def:g12:domesticated-plants:domestication}{domestikasi} menyeleksi sifat yang melayani
sang penyeleksi dan meninggalkan tumbuhannya bergantung padanya.
\omterm{def:g12:domesticated-plants:domestication}{Domestikasi} adalah \omterm{def:g12:selection-drift-speciation:selection}{seleksi alam} dengan penyeleksi yang berbeda, bukan
kebalikannya.
\end{solution}

\begin{solution}{pb:g12:domesticated-plants:1}
\textbf{1.} Teosinte $12 \times 0.08 \approx \qty{1}{g}$; jagung $600
\times 0.3 = \qty{180}{g}$: nyaris dua ratus kali.

\textbf{2.} Banyak butir pada janggel yang kaku (tanpa persebaran);
butir telanjang (tak terlindung dari hewan dan pembusukan); satu
tongkol besar pada satu batang (tanpa penyebaran risiko); butir yang
terkurung kelobot (sama sekali tak dapat jatuh ke tanah).

\textbf{3.} \omterm{def:g10:universal-dna:gene}{Alel} telanjang bersifat \omterm{def:g11:genes-and-disease:genetic}{resesif}: sebuah tanaman
menampakkan butir telanjang hanya bila membawa dua salinannya, dan
bagi \omterm{def:g10:universal-dna:gene}{alel} yang jarang hal itu terjadi pada segelintir tanaman saja;
\omterm{def:g11:genes-and-disease:genetic}{pembawa} satu salinan tampak biasa saja.

\textbf{4.} $q^2 = 0.0001$: satu tanaman dari sepuluh ribu.

\textbf{5.} Semua benihnya berasal dari tanaman $nn$, yang butirnya
menerima $n$ dari induk betinanya; serbuk sarinya sebagian besar
datang dari ladang biasa ($N$ dengan frekuensi 0.99). Benihnya hampir
seluruhnya $Nn$, sehingga frekuensi $n$ melonjak menjadi sekitar 0.5
--- satu generasi seleksi yang keras.

\textbf{6.} $600/12 = 50$ sepanjang 9000 generasi: kelipatan $50^{1/9000}
\approx 1.0004$ tiap generasi, yaitu kenaikan 0.04\%. Langkah mungil,
yang berbunga sepanjang ribuan generasi, menjadi perubahan yang besar
--- itulah hitungan seleksi.

\textbf{7.} Dengan banyak \omterm{def:g10:universal-dna:gene}{gen} yang masing-masing menambah sedikit,
seleksi menaikkan \omterm{def:g12:selection-drift-speciation:population}{frekuensi alel} yang menguntungkan pada semuanya dan
terus menemukan gabungan baru dalam waktu lama; sifat bergen tunggal
beralih begitu \omterm{def:g10:universal-dna:gene}{alel} \omterm{def:g11:genes-and-disease:genetic}{resesifnya} menjadi homozigot.

\textbf{8.} Satu \omterm{def:g10:biodiversity-scales:species}{spesies}: hibridanya subur. \omterm{def:g12:domesticated-plants:domestication}{Domestikasi} mengubah
\omterm{def:g12:selection-drift-speciation:population}{frekuensi alel} pada beberapa \omterm{def:g10:universal-dna:gene}{gen} dan bentuk tanamannya; ia tidak
menciptakan penghalang reproduksi.

\textbf{9.} Penyakit atau hama yang mengalahkan satu tanaman
mengalahkan semuanya; jawabannya adalah keragaman yang tersimpan pada
varietas lama dan kerabat liar, di bank benih dan di ladang.

\textbf{10.} Petani adalah persebaran tanaman itu, lingkungan yang
menyeleksinya dan satu-satunya jalannya menuju generasi berikutnya:
jagung berada di bawah \omterm{def:g12:domesticated-plants:domestication}{seleksi buatan} pada setiap generasi dan tak
dapat meninggalkannya.

\textbf{11.} $10/5.5 - 1 \approx 82\%$ di atas induknya yang lebih baik;
$1 - 8.2/10 = 18\%$ di bawah F1.

\textbf{12.} Setiap galur homozigot bagi beberapa \omterm{def:g10:universal-dna:gene}{alel} \omterm{def:g11:genes-and-disease:genetic}{resesif} yang
lemah; pada F1 setiap \omterm{def:g10:universal-dna:gene}{alel} semacam itu tertutupi oleh \omterm{def:g10:universal-dna:gene}{alel} bekerja
galur yang lain. Pada F2 seperempat tanamannya homozigot bagi \omterm{def:g10:universal-dna:gene}{alel}
lemah pada tiap \omterm{def:g10:universal-dna:gene}{gen} semacam itu, dan reratanya pun turun.

\textbf{13.} Membeli: $10 - 0.4 = \qty{9.6}{t/ha}$; menyimpan: \qty{8.2}{t/ha}.
Membeli unggul \qty{1.4}{t/ha} tiap tahun, dan itulah sebabnya
perusahaan yang memegang galur murninya dapat menjual benih hibrida
setiap musim.

\textbf{14.} Perusahaan menyimpan galurnya; dengan galur itu, seorang
petani dapat membuat hibridanya tanpa henti dan perusahaannya hanya
akan menjual sekali. Galur itu modal perusahaannya; hanya produknya
yang dijual.

\textbf{15.} Setiap ladang varietas hibrida identik secara genetik, dan
seluruh wilayah menanam beberapa hibrida saja: keragamannya
menyempit. Petaninya tak lagi menghasilkan benih dan bergantung pada
pemasoknya setiap tahun.

\textbf{16.} $q^2 = 10^{-6}$: satu penggerek dari sejuta selamat; di
antara penyintasnya, yang semuanya $rr$, frekuensi $r$ adalah 1.

\textbf{17.} Para penyintas berbiak di antara mereka sendiri;
keturunannya seluruhnya $rr$; dalam beberapa generasi penggerek di
ladang itu menjadi \omterm{prop:g11:antibiotic-resistance:mechanisms}{tahan} --- seperti bakteri pada pasien yang kurnya
tak tuntas.

\textbf{18.} Seekor penyintas yang \omterm{prop:g11:antibiotic-resistance:mechanisms}{tahan} ($rr$) yang kawin dengan
penggerek dari suaka ($RR$, yang jumlahnya jauh lebih besar) memberi
keturunan $Rr$, yang mati di jagung beracun. Selama penggerek peka
jauh melebihi jumlah penggerek \omterm{prop:g11:antibiotic-resistance:mechanisms}{tahan}, hampir setiap anak penggerek
\omterm{prop:g11:antibiotic-resistance:mechanisms}{tahan} bersifat heterozigot dan tersingkirkan, dan $r$ tetap jarang.

\textbf{19.} Sama: keduanya bertujuan mencegah berlipatnya individu
\omterm{prop:g11:antibiotic-resistance:mechanisms}{tahan} yang jarang itu. Berbeda: kur penuh menyingkirkannya dengan
membunuh setiap individu sebelum ketahanan sempat terseleksi;
suakanya menerima adanya penyintas lalu mengencerkannya dengan
pasangan yang peka.

\textbf{20.} Nyaris dua ratus kali lebih banyak butir per tongkol;
sekitar lima \omterm{def:g10:universal-dna:gene}{gen} utama; dan sebuah tanaman yang bergantung pada petani
untuk menyebarkan serta menyemainya.
\end{solution}
