\chapter{Menyingkirkan Buangan}\label{ch:g7:removing-wastes}

Pengantaran mendominasi tahun ini: \omterm{def:g7:what-is-respiration:respiration}{oksigen} masuk, \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} masuk,
bahan bakar ke setiap \omterm{def:g5:first-look-at-cells:cell}{sel} yang bekerja. Tetapi setiap bengkel membuat
buangan, dan sebuah layanan pengantaran yang tidak pernah memungut
sampahnya segera meracuni kotanya sendiri. Rekening \omterm{def:g7:what-is-respiration:respiration}{karbon
dioksidanya} sudah dilunasi --- diembuskan keluar di \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya}. Bab
penutup ini mengikuti buangan yang \emph{lain}, sampai ke organ tubuh
yang paling tidak menonjol: \omterm{def:g7:removing-wastes:kidneys}{ginjalnya}.

\section{Buangan di luar karbon dioksida}

\begin{proposition}[Sampah sel yang lain]\label{prop:g7:removing-wastes:rubbish}
Pembakaran dan pembangunan oleh \omterm{def:g5:first-look-at-cells:cell}{selnya}
(\cref{prop:g7:what-is-respiration:power},
\cref{prop:g6:matter-from-food:transform}) menghasilkan lebih daripada
\omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}. Terutama pemakaian \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} materi pembangun
meninggalkan sebuah buangan yang membawa nitrogen ---
\emph{urea}\index{urea} --- yang dipungut \omterm{prop:g4:journey-of-food:absorb}{darahnya} dari setiap
jaringan persis sebagaimana ia memungut \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}. \omterm{prop:g7:removing-wastes:rubbish}{Urea} tidak
berbahaya ketika berada dalam lalu lintas yang lewat dan beracun
ketika menumpuk: ia harus pergi, terus-menerus.
\end{proposition}
\admitted

\begin{example}[Dua ronde pemungutan, satu darah]\label{ex:g7:removing-wastes:rounds}
Tetes yang sama pada \cref{ex:g7:heart-and-circulation:tour}
menjalankan kedua pemungutannya sekaligus: di \omterm{def:g7:heart-and-circulation:vessels}{kapiler} pahanya ia
menerima \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} \emph{dan} \omterm{prop:g7:removing-wastes:rubbish}{urea}. Yang pertama diturunkannya
di \omterm{def:g7:how-animals-breathe:four}{paru-parunya}, pada peredaran berikutnya. Yang kedua menumpang terus
--- melewati \omterm{def:g7:how-animals-breathe:four}{paru-parunya}, yang tidak dapat menyentuhnya --- sampai
peredaran tubuhnya membawanya melewati organ yang dibangun persis bagi
muatan itu.
\end{example}

\section{Ginjal, penyaring darahnya}

\begin{definition}[Ginjal dan sistem kemih]\label{def:g7:removing-wastes:kidneys}
Kedua \emph{ginjal}\index{ginjal} --- sebesar kepalan tangan,
berbentuk kacang, terletak rendah di punggung --- adalah penyaring
pembersih \omterm{prop:g4:journey-of-food:absorb}{darah}. Masing-masingnya menerima sebuah \omterm{def:g7:heart-and-circulation:vessels}{arteri} yang murah
\omterm{prop:g7:digestion-and-nutrients:absorption}{hati} langsung dari jalur utama peredaran tubuhnya. Di dalamnya,
\omterm{prop:g4:journey-of-food:absorb}{darahnya} disaring melalui kira-kira sejuta satuan penyaring yang
renik; \omterm{prop:g4:journey-of-food:absorb}{darah} yang sudah bersih kembali lewat \omterm{def:g7:heart-and-circulation:vessels}{vena} ginjalnya, dan
muatan yang disingkirkan --- \omterm{prop:g7:removing-wastes:rubbish}{urea}, kelebihan air, kelebihan garam ---
pergi sebagai \emph{urine}\index{urine}, yang menetes turun lewat dua
pipa menuju \emph{kandung kemih}\index{kandung kemih}, yaitu tangki
penampung yang dikosongkan sesuka \omterm{prop:g7:digestion-and-nutrients:absorption}{hati}.
\end{definition}

\begin{proposition}[Saring, lalu ambil kembali]\label{prop:g7:removing-wastes:filter}
Kiat \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} adalah kemurahan \omterm{prop:g7:digestion-and-nutrients:absorption}{hati} yang disusul kehematan. Satuan
penyaringnya mula-mula mendorong sejumlah besar bagian berair \omterm{prop:g4:journey-of-food:absorb}{darahnya}
--- berpuluh-puluh liter sehari --- melewati ayakan halusnya, berikut
\omterm{prop:g7:removing-wastes:rubbish}{urea}, gula, garam dan segalanya. Lalu, di sepanjang pipa berkelok tiap
satuannya, tubuhnya \emph{mengambil kembali} apa yang diinginkannya:
nyaris seluruh airnya, seluruh \omterm{prop:g7:organs-at-work:bill}{glukosanya}, garam secukupnya ---
sehingga \omterm{prop:g7:removing-wastes:rubbish}{urea} dan kelebihannya tertinggal. Dari berpuluh-puluh liter
yang diayak, kira-kira satu setengah liter pergi sebagai \omterm{def:g7:removing-wastes:kidneys}{urine}:
saringlah segalanya, ambil kembali yang baik, dan buangannya tidak
punya tempat bersembunyi.
\end{proposition}
\admitted

\begin{example}[Membaca kehematannya]\label{ex:g7:removing-wastes:thrift}
Tahap pengambilan kembalinya menjelaskan pengamatan sehari-harinya.
Minumlah banyak, dan \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} mengambil kembali lebih sedikit air:
\omterm{def:g7:removing-wastes:kidneys}{urinenya} berlimpah dan pucat. Berkeringatlah sampai kering dalam
pertandingan yang panas, dan \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} mengambil kembali nyaris
seluruhnya: \omterm{def:g7:removing-wastes:kidneys}{urinenya} sedikit dan gelap --- tubuhnya sedang menjaga
airnya (\cref{prop:g4:what-plants-need:needs} mengajarkan nilai air
bagi \omterm{def:g1:plants-around-us:plant}{tumbuhan}; \omterm{def:g7:removing-wastes:kidneys}{ginjal} adalah sisi \omterm{def:g1:animals-around-us:animal}{hewan} dari ekonomi yang sama). Dan
\omterm{prop:g7:organs-at-work:bill}{glukosa} di dalam \omterm{def:g7:removing-wastes:kidneys}{urine}, yang dilewatkan ayakannya tetapi seharusnya
dipulihkan sepenuhnya oleh pengambilan kembalinya, adalah isyarat
kedokteran yang klasik bahwa ada yang tidak beres --- entah pengambilan
kembalinya, entah gula \omterm{prop:g4:journey-of-food:absorb}{darahnya}.
\end{example}

\begin{remark}[Kulit ikut membantu]\label{rem:g7:removing-wastes:sweat}
\emph{Keringat}\index{keringat} adalah usaha sampingan dari bisnis
yang sama: air dan garam (dengan jejak \omterm{prop:g7:removing-wastes:rubbish}{urea}) yang dilepaskan di \omterm{def:g1:the-five-senses:sense}{kulit}
--- walaupun tugas pertamanya adalah pendinginan pada
\cref{ex:g7:organs-at-work:felt}. \omterm{def:g7:how-animals-breathe:four}{Paru-paru}, \omterm{def:g7:removing-wastes:kidneys}{ginjal} dan \omterm{def:g1:the-five-senses:sense}{kulit} membagi
seluruh penyingkiran buangan tubuhnya: \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} kepada yang
pertama, \omterm{prop:g7:removing-wastes:rubbish}{urea} dan kelebihannya kepada yang kedua, panas dan sebuah
tetesan asin kepada yang ketiga. Tidak ada yang beracun yang
dibiarkan menumpuk --- di dalam tubuh yang sehat, sampahnya tidak
pernah menang.
\end{remark}

\section{Mesin tahun ini, yang lengkap}

\begin{proposition}[Gelang pasokan-dan-penyingkiran yang lengkap]\label{prop:g7:removing-wastes:complete}
Dengan \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} pada tempatnya, perjalanan keliling tahun ini menutup
menjadi satu mesin. \omterm{prop:g4:journey-of-food:absorb}{Darahnya}, yang dipompa mengelilingi peredaran
gandanya (\cref{def:g7:heart-and-circulation:double}), memuat \omterm{def:g7:what-is-respiration:respiration}{oksigen}
di \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya} dan \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} di \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot ususnya}; mengantarkan keduanya
ke setiap \omterm{def:g5:first-look-at-cells:cell}{sel} yang bekerja untuk dibakar dan dibangun; lalu memungut
buangannya --- \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} ke \omterm{def:g7:how-animals-breathe:four}{paru-parunya}, \omterm{prop:g7:removing-wastes:rubbish}{urea} dan kelebihannya
ke \omterm{def:g7:removing-wastes:kidneys}{ginjalnya}. Setiap organ pada kelas empat dan kelas tujuh adalah
sebuah stasiun pada gelang tunggal ini; setiap pertukarannya
menyeberangi sebuah dinding \omterm{def:g7:heart-and-circulation:vessels}{kapiler}; dan keseluruhannya berjalan
dengan api lambat pada \cref{prop:g7:what-is-respiration:power}, \omterm{def:g5:first-look-at-cells:cell}{sel}
demi \omterm{def:g5:first-look-at-cells:cell}{sel}.
\end{proposition}
\admitted

\begin{method}[Audit seluruh tubuh]\label{met:g7:removing-wastes:audit}
Untuk zat apa pun, auditlah perlintasannya:
\begin{enumerate}
  \item masuknya: mulut dan \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot usus}, atau hidung dan \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolus};
  \item pengangkutannya: kaki peredaran yang mana, perhentian khusus
        yang mana (\omterm{prop:g7:digestion-and-nutrients:absorption}{hatinya});
  \item pemakaiannya: bahan bakar yang dibakar, bahan yang dibangunkan,
        atau bukan keduanya;
  \item keluarnya: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolus} (gas), \omterm{def:g7:removing-wastes:kidneys}{ginjal} (\omterm{prop:g7:removing-wastes:rubbish}{urea}, kelebihannya), \omterm{def:g1:the-five-senses:sense}{kulit}
        (bagian keringatnya) --- atau, kalau sama sekali tidak
        terserap, jalur sisanya
        (\cref{prop:g4:journey-of-food:absorb}).
\end{enumerate}
Jalankan ia atas air, atas \omterm{prop:g7:organs-at-work:bill}{glukosa}, atas sekeping keripik yang asin:
mesinnya punya sebuah stasiun bagi setiap langkahnya, dan itulah
temuan tahun ini dalam satu latihan.
\end{method}

\section{Soal: Sehari Perjalanan Sebuah Molekul Air}

\begin{problem}[{Soal akhir pekan --- segelas air, yang diaudit
melewati seluruh mesinnya}]\label{pb:g7:removing-wastes:1}
Ikutilah segelas air yang diminum ketika sarapan pada sebuah hari
berolahraga, sebagai audit besar tahun ini.

\medskip\noindent\textbf{Bagian I --- Masuk dan pengangkutannya.}
\begin{enumerate}
  \item Air tidak membutuhkan pembongkaran
        (\cref{def:g7:digestion-and-nutrients:families}).
        Sebutkan penyeberangan masuknya dan jaringan \omterm{def:g4:heart-and-blood:vessels}{pembuluh} yang
        menerimanya.
  \item \omterm{prop:g4:journey-of-food:absorb}{Darahnya} membuat perhentian pertama yang baku. Organ yang
        mana, dan mengapa penjalurannya masuk akal bagi \omterm{def:g7:digestion-and-nutrients:families}{zat gizi}
        walaupun airnya sendiri tidak membutuhkan penyortiran?
  \item Menjelang tengah pagi, airnya berada di mana pun \omterm{prop:g4:journey-of-food:absorb}{darahnya}
        pergi. Sebutkan kedua peredaran yang sekarang ditumpanginya
        secara bergantian.
  \item Sebagiannya segera berada di dalam \omterm{def:g5:first-look-at-cells:cell}{sel} yang bekerja.
        Daftarkan dua tugas yang dikerjakan air di situ (ingatlah
        bahwa kimia \omterm{def:g5:first-look-at-cells:cell}{selnya} --- getah, pembakaran --- berjalan dalam
        keadaan basah).
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pertandingan sore
harinya.}
\begin{enumerate}[resume]
  \item Selama pertandingannya, sebagian airnya pergi lewat sebuah
        perbatasan yang sebenarnya bukan organ pembuangan. Sebutkan
        perbatasannya, muatannya selain air, dan tugas sesungguhnya
        pada hari itu.
  \item Pertandingannya usai, pemainnya kehausan dan \omterm{def:g7:removing-wastes:kidneys}{urinenya} malam
        itu sedikit serta gelap. Jelaskan keduanya dengan tahap
        pengambilan kembalinya.
  \item Sementara itu \omterm{def:g3:bones-and-muscles:muscle}{otot} yang bekerja keras dalam pertandingannya
        membuat \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} dan \omterm{prop:g7:removing-wastes:rubbish}{urea} tambahan. Berikan kepada
        masing-masingnya perbatasan keluarnya, dan kaki peredaran yang
        membawanya ke situ.
  \item Mengapa \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} terus bekerja sepanjang pertandingannya,
        walaupun produksi \omterm{def:g7:removing-wastes:kidneys}{urinenya} turun? Apa yang masih
        disingkirkannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Auditnya yang ditutup.}
\begin{enumerate}[resume]
  \item Keesokan paginya, sebagian isi gelasnya sudah pergi bersama
        napas. Jelaskan bagaimana air keluar di \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya}
        (\cref{ex:g4:breathing:mirror} memperlihatkannya bertahun-tahun
        yang lalu).
  \item Susunlah pembukuan lengkap gelasnya: masuknya, simpanan dan
        pemakaiannya, serta keempat jalan keluarnya --- \omterm{def:g7:removing-wastes:kidneys}{urine},
        keringat, napas, dan bagian kecil di dalam sisanya.
  \item Auditnya memperlihatkan satu \omterm{prop:g4:journey-of-food:absorb}{darah} yang melayani \omterm{def:g7:how-animals-breathe:four}{paru-paru},
        \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot usus}, \omterm{def:g3:bones-and-muscles:muscle}{otot}, \omterm{def:g7:removing-wastes:kidneys}{ginjal} dan \omterm{def:g1:the-five-senses:sense}{kulit} secara bergiliran.
        Nyatakan, dalam dua kalimat, mengapa rancangan jaringan
        tunggal ini membuat kesehatan tiap organ menjadi urusan setiap
        organ yang lain.
  \item Tutuplah tahun ini: satu gagasan tunggal yang mana ---
        sebutkan namanya --- yang ternyata dilayani setiap stasiun
        auditnya? (\cref{prop:g7:what-is-respiration:power} memegang
        jawabannya.)
\end{enumerate}
\end{problem}
