\chapter{مورثومات الخلايا والفيروسات}\label{ch:b1:genomes}

مترا \omterm{def:b1:nucleic-acids:chain}{DNA} مطويان في نواة قطرها ستة ميكرومترات، ومطويان بحيث يمكن إيجاد
أي من عشرين ألف \omterm{def:b1:genomes:gene}{مورّثة} وقراءتها في دقائق. والبكتيريا تطوي مليمترًا
ونصفًا في \omterm{def:b1:cell-unit-of-life:cell}{خلية} أقصر منه بألف مرة، وتنسخه كل عشرين دقيقة. \omterm{def:b1:genomes:virus}{والفيروس}
يحمل بضعة آلاف من الحروف في غلاف بروتيني ويقرؤها \omterm{def:b1:cell-unit-of-life:cell}{بخلية} لا يملكها.
يصف هذا الفصل \emph{\omterm{def:b1:genomes:genome}{المورثوم}} --- أي مجموع \omterm{def:b1:nucleic-acids:chain}{DNA} الكائن --- بوصفه شيئًا
ماديًا: كيف يُنظَّم \omterm{def:b1:nucleic-acids:chain}{DNA} البكتيريا، \omterm{def:b1:cell-unit-of-life:cell}{والخلايا} \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقية النواة}، \omterm{def:b1:cell-unit-of-life:prokeuk}{والعضيات}،
\omterm{def:b1:genomes:virus}{والفيروسات}، وما يحويه فوق \omterm{def:b1:genomes:gene}{المورّثات}، ولماذا يقول مقداره القليل جدًا عن
الكائن الذي يحمله.

\section{المورثوم البكتيري}

\begin{definition}[المورثوم والصبغي والنويد]\label{def:b1:genomes:genome}
\emph{مورثوم}\index{مورثوم} \omterm{def:b1:cell-unit-of-life:cell}{خلية} هو مجموع \omterm{def:b1:nucleic-acids:chain}{DNA} فيها: أي المعلومة لكل
\omterm{def:b1:proteins:peptide}{بروتين} وكل \omterm{def:b1:nucleic-acids:chain}{RNA} تستطيع صنعه، والتسلسلات التي تتحكم في متى تُصنع.
و\emph{الصبغي}\index{صبغي} جزيء \omterm{def:b1:nucleic-acids:chain}{DNA} واحد مع \omterm{def:b1:proteins:peptide}{البروتينات} التي تنظمه.
وللبكتيريا عادة صبغي واحد حلقي من مليون إلى عشرة ملايين \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد}
(و\emph{E. coli}: $\num{4.6e6}$، أي \qty{1.5}{mm} طولًا)، مطوي في
منطقة من \omterm{def:b1:cell-unit-of-life:cell}{السيتوبلازم} هي \emph{النويد}\index{نويد}، بلا غشاء حولها.
وفوق الصبغي تحمل كثير من البكتيريا \emph{بلازميدات}\index{بلازميد}:
وهي حلقات صغيرة من بضعة آلاف إلى بضع مئات الآلاف من الأزواج، في نسخة
إلى مئات النسخ، تتضاعف على حدة وتحمل \omterm{def:b1:genomes:gene}{مورّثات} اختيارية --- مقاومة
المضادات الحيوية، والذيفانات، وآلة نقل نفسها إلى \omterm{def:b1:cell-unit-of-life:cell}{خلية} أخرى.
\end{definition}

\begin{proposition}[كيف تطوي بكتيريا صبغيها]\label{prop:b1:genomes:supercoiling}
\index{الالتفاف الفائق}
\omterm{def:b1:nucleic-acids:chain}{DNA} حلقي لا تستطيع نهايتاه أن تدورا يمكن لَيّه لَيّ شريط مطاطي: وهو
\emph{\omterm{prop:b1:genomes:supercoiling}{الالتفاف الفائق}}. \omterm{def:b1:enzymes:enzyme}{وإنزيمات} تسمى \emph{طوبوإيزوميرازات} تقص
الخصلتين وتمررهما وتعيد لحمهما لتضيف دورات أو تنزعها؛
و\emph{الجيراز} يستعمل \omterm{def:b1:water-small-molecules:atp}{ATP} ليُدخل التفافات فائقة سالبة تفكّ لولبة
الحلزون وتجعله أكثر انضغاطًا وأسهل فتحًا في آن. \omterm{def:b1:genomes:genome}{وصبغي}
\emph{E. coli} منظَّم في نحو خمسين عروة، كل منها ملتفة التفافًا فائقًا
على حدة ومرساة على لب بروتيني، حتى إن \qty{1.5}{mm} تشغل نويدًا بنحو
ميكرومتر --- أي انضغاطًا ألف ضعف --- بينما يمكن فك أي عروة للقراءة
أو النسخ دون إزعاج الباقي. \omterm{def:b1:proteins:peptide}{وبروتينات} أساسية صغيرة (شبيهة بالهستونات،
وإن لم تكن قريبة منها) تثني \omterm{def:b1:nucleic-acids:chain}{DNA} وتجسر بينه في كل مكان.
\end{proposition}

\begin{example}[مورثوم كثيف]\label{ex:b1:genomes:dense}
من \qty{4.6}{Mb} في \omterm{def:b1:genomes:genome}{صبغي} \emph{E. coli}، تشفّر \qty{88}{\%}
\omterm{def:b1:proteins:peptide}{بروتينات}: أي نحو \num{4300} \omterm{def:b1:genomes:gene}{مورّثة} من نحو \num{1000} زوج لكل واحدة،
مرصوصة رأسًا بعقب بمئة زوج بينها، وكثير منها مجموع في
\emph{أوبرونات} تُستنسخ معًا (\cref{ch:b1:expression-control}).
\omterm{def:b1:genomes:genome}{والبلازميد} من \qty{5}{kb} قد يحمل ثلاث \omterm{def:b1:genomes:gene}{مورّثات} ويوجد في خمسين نسخة؛
\omterm{def:b1:genomes:gene}{ومورّثة} المقاومة التي يحمله يمكن أن تنتشر في بكتيريا جناح مستشفى في
أسابيع، بالاقتران، أسرع مما تستطيع أي طفرة أن تنشأ.
\end{example}

\section{المورثوم حقيقي النواة: الصبغين}

\begin{definition}[الجسيم النووي والصبغين]\label{def:b1:genomes:nucleosome}
\index{جسيم نووي}\index{هستون}\index{صبغين}
في نواة حقيقية يُلف \omterm{def:b1:nucleic-acids:chain}{DNA} على \omterm{def:b1:proteins:peptide}{بروتينات}: \qty{147}{} \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} تصنع 1.7
دورة حول ثماني \emph{هستونات} (اثنتان من كل من H2A وH2B وH3 وH4، وهي
\omterm{def:b1:proteins:peptide}{بروتينات} صغيرة غنية باللايسين والأرجينين تعادل الفوسفات)، فتكوّن
\emph{جسيمًا نوويًا}\index{جسيم نووي} قطره \qty{11}{nm}؛ وتتتابع
الجسيمات النووية كل \qty{200}{} زوج تقريبًا، كخرزات على خيط، مع
هستون خامس، H1، يختم كل خرزة. وينطوي هذا \emph{الصبغين} أكثر: في ليف
قطره \qty{30}{nm}، ثم في عرى من عشرات الآلاف من الأزواج مرساة على
سقالة بروتينية، ثم --- عند الانقسام --- في قضبان \omterm{def:b1:genomes:genome}{الصبغي} المنضغطة في
الطور الاستوائي، وهي أقصر من \omterm{def:b1:nucleic-acids:chain}{DNA} العاري بعشرة آلاف مرة.
و\emph{الصبغين الحقيقي} هو الصورة الأرخى الغنية \omterm{def:b1:genomes:gene}{بالمورّثات} المستنسخة
في الطور البيني؛ و\emph{الصبغين المتغاير} هو الصورة المكثفة الفقيرة
\omterm{def:b1:genomes:gene}{بالمورّثات} التي تبقى منضغطة (حول المريكزات، وعند الأطراف، وعلى \omterm{def:b1:genomes:genome}{صبغي}
X مطفأ).
\end{definition}

\begin{definition}[النمط النووي والمريكز والقسيم الطرفي]\label{def:b1:genomes:karyotype}
\index{نمط نووي}\index{مريكز}\index{قسيم طرفي}
\emph{النمط النووي} هو مجموعة \omterm{def:b1:genomes:genome}{صبغيات} \omterm{def:b1:cell-unit-of-life:cell}{خلية}، مرتبة بالقياس ونمط التخطط:
فللإنسان 23 زوجًا (22 صبغيًا جسميًا وXY أو XX)، و$\num{6.4e9}$ \omterm{thm:b1:nucleic-acids:helix}{زوج
قواعد} في المجموع، أي \qty{2.2}{m}؛ وأكبر \omterm{def:b1:genomes:genome}{صبغي} يحمل \qty{250}{Mb}
وأصغرها \qty{50}{}. ويحمل كل \omterm{def:b1:genomes:genome}{صبغي} خطي \emph{مريكزًا}، وهي المنطقة
المضيَّقة التي تبقى النسختان موصولتين عندها بعد التضاعف ويرتبط بها
المغزل (\cref{ch:b1:replication-mitosis})، و\emph{قسيمين طرفيين}، وهما
تسلسلان متكرران عند النهايتين يحميانهما من أن يُحسبا \omterm{def:b1:nucleic-acids:chain}{DNA} مكسورًا ومن
التقصر عند كل تضاعف.
\end{definition}

\section{ماذا يحوي المورثوم}

\begin{definition}[تشريح مورّثة]\label{def:b1:genomes:gene}
\index{مورّثة}\index{إكسون}\index{إنترون}\index{مروّج}
\emph{المورّثة} قطعة من \omterm{def:b1:nucleic-acids:chain}{DNA} تُستنسخ إلى \omterm{def:b1:nucleic-acids:chain}{RNA} وظيفي --- وهو عادة رسول
\omterm{def:b1:proteins:peptide}{لبروتين} واحد. والمورّثة \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقية النواة} المشفِّرة \omterm{def:b1:proteins:peptide}{للبروتين} تشمل
\emph{مروّجًا}، وهو التسلسل الواقع أعلى المجرى حيث يبدأ الاستنساخ
ويُتحكم فيه؛ و\emph{إكسونات}، وهي الأجزاء المحفوظة في الرسالة
الناضجة؛ و\emph{إنترونات}، وهي الأجزاء المستنسخة ثم المقصوصة
(\cref{ch:b1:gene-expression})؛ والمنطقتين غير المترجمتين عند طرفي
الرسالة؛ وخاتمة. والمورّثة البشرية بالمتوسط \qty{27}{kb} بثمانية
إكسونات مجموعها \qty{1.5}{kb}: فتسعة عشر جزءًا من عشرين من المورّثة
النموذجية إنترون. أما المورّثات البكتيرية فلا إنترونات فيها.
\end{definition}

\begin{proposition}[جرد المورثوم البشري]\label{prop:b1:genomes:census}
\index{عنصر قافز}\index{DNA متكرر}

\end{proposition}

\section{مورثومات العضيات والفيروسات}

\begin{definition}[مورثومات العضيات]\label{def:b1:genomes:organelle}
تحتفظ \omterm{def:b1:eukaryotic-cell:mitochondrion}{المتقدرات} \omterm{def:b1:eukaryotic-cell:plastid}{والبلاستيدات الخضراء} ببقية من \omterm{def:b1:genomes:genome}{صبغي} أسلافها البكتيرية
(\cref{ch:b1:eukaryotic-cell}): وهي حلقة صغيرة --- \qty{16.6}{kb}
و37 \omterm{def:b1:genomes:gene}{مورّثة} في \omterm{def:b1:eukaryotic-cell:mitochondrion}{متقدرات} الإنسان، و\qtyrange{120}{160}{kb} ونحو 100
\omterm{def:b1:genomes:gene}{مورّثة} في \omterm{def:b1:eukaryotic-cell:plastid}{البلاستيدات} --- في نسخ كثيرة لكل عضية، تشفّر بعض بروتيناتها
الخاصة \omterm{def:b1:nucleic-acids:chain}{وRNA} ريبوزوماتها. وسائر بروتيناتها يأتي من \omterm{def:b1:genomes:gene}{مورّثات} نووية. وفي
معظم الحيوانات يُورَّث \omterm{def:b1:genomes:genome}{المورثوم} المتقدري من الأم وحدها، مع \omterm{def:b1:cell-unit-of-life:cell}{سيتوبلازم}
البيضة.
\end{definition}

\begin{definition}[الفيروس]\label{def:b1:genomes:virus}
\index{فيروس}\index{قفيصة}
\emph{الفيروس} \omterm{def:b1:genomes:genome}{مورثوم} في غلاف بروتيني، هو \emph{القفيصة} --- ملفوفة
أحيانًا في غشاء، هو \emph{الغلاف}، مأخوذ من \omterm{def:b1:cell-unit-of-life:cell}{خلية} مضيفة --- لا يتكاثر
إلا داخل \omterm{def:b1:cell-unit-of-life:cell}{خلية}، مستعملًا ريبوزوماتها وطاقتها وطلائعها. ومورثومه قد
يكون \omterm{def:b1:nucleic-acids:chain}{DNA} أو \omterm{def:b1:nucleic-acids:chain}{RNA}، مزدوج الخصلة أو مفردها، خطيًا أو حلقيًا، جزيئًا
واحدًا أو عدة جزيئات: من \qty{5}{kb} (بضع \omterm{def:b1:genomes:gene}{مورّثات}) إلى \qty{1.2}{Mb}
(ألف \omterm{def:b1:genomes:gene}{مورّثة}، عند الفيروسات العملاقة). وخارج \omterm{def:b1:cell-unit-of-life:cell}{خلية} لا يفعل الفيروس
شيئًا: فهو ليس \omterm{def:b1:cell-unit-of-life:cell}{خلية}، ولا يستقلب، وهو حي بمعنى واحد فقط هو أنه يحمل
معلومة ويتطور. و\emph{العاثيات} هي فيروسات البكتيريا.
\end{definition}

\begin{proposition}[دورتا العاثية $\lambda$]\label{prop:b1:genomes:lambda}
\index{الدورة الحالة}\index{الدورة المدمجة}
تحقن العاثية $\lambda$ ما لديها من \qty{48.5}{kb} من \omterm{def:b1:nucleic-acids:chain}{DNA}
(\cref{ch:b1:nucleic-acids}) في \emph{E. coli}، حيث يتحلق. ثم يتبع
مصيران. ففي \emph{\omterm{prop:b1:genomes:lambda}{الدورة الحالة}} تُستنسخ \omterm{def:b1:genomes:gene}{مورّثات} العاثية ببوليميراز
المضيف، ويتضاعف \omterm{def:b1:nucleic-acids:chain}{DNA} لها مئة ضعف، وتُصنع \omterm{def:b1:proteins:peptide}{بروتينات} \omterm{def:b1:genomes:virus}{القفيصة} وتُركَّب،
ويُحشر \omterm{def:b1:nucleic-acids:chain}{DNA} فيها، ويذيب \omterm{def:b1:enzymes:enzyme}{إنزيم} الجدار: فتنفجر \omterm{def:b1:cell-unit-of-life:cell}{الخلية}، فتطلق مئة عاثية
بعد أربعين دقيقة من العدوى. وفي \emph{\omterm{prop:b1:genomes:lambda}{الدورة المدمجة}} يُدخل \omterm{def:b1:nucleic-acids:chain}{DNA}
العاثية في \omterm{def:b1:genomes:genome}{صبغي} المضيف ويُسكَّت بكابح من صنعه هو: فتُنسخ
\emph{العاثية الكامنة} مع \omterm{def:b1:genomes:genome}{الصبغي} أجيالًا، غير مرئية، حتى يرفع تلفٌ في
\omterm{def:b1:nucleic-acids:chain}{DNA} المضيف الكبحَ فتستأنف \omterm{prop:b1:genomes:lambda}{الدورة الحالة}. والاختيار مفتاح جزيئي، وهو من
أول ما فُهم منها (\cref{ch:b1:expression-control}).
\end{proposition}

\begin{example}[فيروسات مورثوماتها RNA]\label{ex:b1:genomes:rna}
تحمل الأنفلونزا ثماني قطع من \omterm{def:b1:nucleic-acids:chain}{RNA} مفرد الخصلة وبوليميرازها الخاص
لنسخها، إذ لا \omterm{def:b1:enzymes:enzyme}{إنزيم} في \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} ينسخ \omterm{def:b1:nucleic-acids:chain}{RNA}؛ وبوليميرازها يخطئ خطأً واحدًا
لكل \num{10000} \omterm{def:b1:nucleic-acids:nucleotide}{قاعدة} \omterm{def:b1:genomes:virus}{والفيروس} يتغير كل موسم. أما \omterm{def:b1:genomes:virus}{الفيروس} القهقري
(HIV) فيحمل \omterm{def:b1:nucleic-acids:chain}{RNA} و\emph{ناسخة عكسية} تنسخه إلى \omterm{def:b1:nucleic-acids:chain}{DNA} يُدخل في \omterm{def:b1:genomes:genome}{صبغي}
المضيف مثل عاثية كامنة --- أي عدوى دائمة. والآليات، والاستجابة
المناعية لها، من كتاب السنة الثالثة؛ وهي هنا تعلّم مدى ما يمكن أن
يكونه \omterm{def:b1:genomes:genome}{مورثوم}.
\end{example}

\section{مسألة: مترانِ في ستة ميكرومترات}

\begin{problem}[{مسألة نهاية الأسبوع --- نواة بشرية مقيسة من اللولب
إلى الصبغي: الأطوال، والحجوم، والجسيمات النووية، والهستونات، ونسبة
الرص عند كل مستوى، وصولًا إلى الانضغاط الكلي في الطور الاستوائي}]
\label{pb:b1:genomes:1}
نواة بشرية ثنائية الصيغة كرة قطرها \qty{6}{\micro m} تحمل
$\num{6.4e9}$ \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} على 46 صبغيًا؛ وأكبر \omterm{def:b1:genomes:genome}{صبغي} يحمل \qty{250}{Mb}
وأصغرها \qty{50}{}. خذ \qty{0.34}{nm} لكل زوج، و\qty{2}{nm} لقطر
اللولب، و\qty{650}{Da} لكل زوج، و\num{200} زوج لكل \omterm{def:b1:genomes:nucleosome}{جسيم نووي}، وثماني
هستونات كتلتها \qty{108}{kDa}، و$\qty{1.66e-24}{g}$ لكل دالتون.

\medskip
\noindent\textbf{الجزء الأول --- الأطوال والحجوم.}
\begin{enumerate}
  \item احسب مجموع طول \omterm{def:b1:nucleic-acids:chain}{DNA} في \omterm{def:b1:eukaryotic-cell:nucleus}{النواة}.
  \item احسب عدد دورات \omterm{thm:b1:nucleic-acids:helix}{اللولب المزدوج} التي يحويها.
  \item احسب طول \omterm{def:b1:nucleic-acids:chain}{DNA} أكبر \omterm{def:b1:genomes:genome}{صبغي} وأصغره.
  \item احسب حجم \omterm{def:b1:eukaryotic-cell:nucleus}{النواة}.
  \item احسب حجم \omterm{def:b1:nucleic-acids:chain}{DNA} بوصفه أسطوانة، والكسر الذي يشغله من \omterm{def:b1:eukaryotic-cell:nucleus}{النواة}.
  \item احسب كتلة \omterm{def:b1:nucleic-acids:chain}{DNA} في \omterm{def:b1:eukaryotic-cell:nucleus}{النواة} بالبيكوغرام.
  \item تحوي \omterm{def:b1:eukaryotic-cell:nucleus}{النواة} نحو \qty{25}{pg} من \omterm{def:b1:proteins:peptide}{البروتين}. فأي كسر من كتلة
        \omterm{def:b1:eukaryotic-cell:nucleus}{النواة} هو \omterm{def:b1:nucleic-acids:chain}{DNA}، إذا كان الماء \qty{80}{\%} من \omterm{def:b1:eukaryotic-cell:nucleus}{النواة} (بكثافة
        \qty{1.1}{g/mL})؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الجسيمات النووية.}
\begin{enumerate}[resume]
  \item احسب عدد الجسيمات النووية في \omterm{def:b1:eukaryotic-cell:nucleus}{النواة}.
  \item احسب كتلة الهستونات وقارنها بكتلة \omterm{def:b1:nucleic-acids:chain}{DNA}.
  \item الأزواج \qty{200}{} في \omterm{def:b1:genomes:nucleosome}{جسيم نووي} واحد تمتد على \qty{68}{nm}
        من اللولب لكنها تشغل خرزة طولها \qty{11}{nm}. احسب عامل
        انضغاط «الخرزات على الخيط».
  \item الليف \qty{30}{nm} يحمل ست جسيمات نووية لكل \qty{11}{nm} من
        طوله. احسب عامل انضغاطه بالنسبة إلى \omterm{def:b1:nucleic-acids:chain}{DNA} العاري.
  \item احسب طول الليف \qty{30}{nm} \omterm{def:b1:genomes:genome}{للمورثوم} كله، وقارنه بقطر \omterm{def:b1:eukaryotic-cell:nucleus}{النواة}.
  \item لا بد من تفكيك كل \omterm{def:b1:genomes:nucleosome}{جسيم نووي} وإعادة تركيبه حين يتضاعف \omterm{def:b1:nucleic-acids:chain}{DNA} أو
        يُستنسخ. فكم \omterm{def:b1:genomes:nucleosome}{جسيمًا نوويًا} يُعاد تركيبه في الطور S
        (\qty{8}{h})؟ وكم في الثانية؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- العرى \omterm{def:b1:genomes:genome}{والصبغيات}.}
\begin{enumerate}[resume]
  \item عرى من \qty{75}{kb} من الليف \qty{30}{nm} تتدلى من سقالة.
        احسب طول الليف في عروة واحدة وعدد العرى في \omterm{def:b1:genomes:genome}{المورثوم}.
  \item صُبيغية أكبر \omterm{def:b1:genomes:genome}{صبغي} في الطور الاستوائي طولها
        \qty{10}{\micro m}. احسب عامل الانضغاط من \omterm{def:b1:nucleic-acids:chain}{DNA} العاري إلى \omterm{def:b1:genomes:genome}{صبغي}
        الطور الاستوائي.
  \item احسب حجم تلك الصبيغية بوصفها أسطوانة قطرها
        \qty{0.7}{\micro m}، والكسر الذي هو \omterm{def:b1:nucleic-acids:chain}{DNA} منها.
  \item \omterm{def:b1:genomes:genome}{الصبغيات} الستة والأربعون في الطور الاستوائي، وكل منها قضيب
        بالكثافة نفسها: احسب حجمها الكلي وقارنه \omterm{def:b1:eukaryotic-cell:nucleus}{بالنواة} في الطور
        البيني.
  \item فسّر لماذا لا يمكن رص \omterm{def:b1:genomes:nucleosome}{صبغين} الطور البيني رصًا محكمًا كرص \omterm{def:b1:genomes:nucleosome}{صبغين}
        الطور الاستوائي.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- إيجاد \omterm{def:b1:genomes:gene}{مورّثة}.}
\begin{enumerate}[resume]
  \item على \omterm{def:b1:proteins:peptide}{بروتين} أن يجد موقعًا واحدًا من \qty{20}{bp} في \omterm{def:b1:genomes:genome}{المورثوم}.
        فلو اختبر موقعًا عشوائيًا واحدًا في كل جزء من الألف من الثانية،
        فكم يستغرق البحث؟ وماذا يقول هذا عن كيفية إيجاد المواقع؟
  \item \omterm{def:b1:genomes:gene}{مورّثات} \omterm{def:b1:genomes:genome}{المورثوم} \qty{20000}{} بمتوسط \qty{27}{kb}. فأي كسر من
        \omterm{def:b1:genomes:genome}{المورثوم} داخل \omterm{def:b1:genomes:gene}{المورّثات}؟ وأي كسر مشفِّر، عند \qty{1.5}{kb} من
        \omterm{def:b1:genomes:gene}{الإكسونات} لكل \omterm{def:b1:genomes:gene}{مورّثة}؟
  \item يشغل كل \omterm{def:b1:genomes:genome}{صبغي} إقليمه الخاص من \omterm{def:b1:eukaryotic-cell:nucleus}{النواة}. احسب حجم إقليم أكبر
        \omterm{def:b1:genomes:genome}{صبغي} إذا كانت الأقاليم متناسبة مع محتوى \omterm{def:b1:nucleic-acids:chain}{DNA}، وتركيز \omterm{def:b1:nucleic-acids:chain}{DNA} داخله
        بأزواج القواعد في الميكرومتر المكعب.
  \item نواة سمكة رئة بكثافة \omterm{def:b1:nucleic-acids:chain}{DNA} نفسها يكون حجمها كم من أجل
        \qty{260}{Gb} (ثنائية الصيغة)؟ وما قطرها؟
  \item احسب عدد ثمانيات الهستونات التي يجب أن تصنعها \omterm{def:b1:cell-unit-of-life:cell}{خلية} في الطور S
        والحموض الأمينية التي تحويها (\qty{1000}{} بقية لكل ثمانية).
        قارن مع \qty{3e9}{} بقية بروتينية تصنعها \omterm{def:b1:cell-unit-of-life:cell}{خلية} في اليوم.
  \item فسّر في جملتين لماذا تستطيع البكتيريا الاستغناء عن الجسيمات
        النووية ولا تستطيع \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقيات النواة} ذلك.
  \item اذكر النتيجة: عامل الانضغاط الكلي من اللولب إلى \omterm{def:b1:genomes:genome}{صبغي} الطور
        الاستوائي، والعوامل عند كل مستوى.
\end{enumerate}
\end{problem}
