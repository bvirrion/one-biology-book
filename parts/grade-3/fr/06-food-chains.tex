\chapter{Les chaînes alimentaires}\label{ch:g3:food-chains}

L'herbe pousse dans le pré. Une sauterelle mange l'herbe. Un lézard
attrape la sauterelle. Une buse, qui tourne haut dans le ciel, prend
le lézard. Quatre repas --- et voilà l'après-midi de la buse
suspendu à la verdeur de l'herbe. Dans la nature, les repas
s'enchaînent, et apprendre à les lire, c'est apprendre comment tout
un pré tient ensemble.

\section{Lire et écrire une chaîne alimentaire}

\begin{definition}[Chaîne alimentaire]\label{def:g3:food-chains:chain}
Une \emph{chaîne alimentaire}\index{chaîne alimentaire} est une
suite d'\omterm{def:g1:living-or-not:living}{êtres vivants} dont chacun est mangé par le suivant. On
l'écrit avec des flèches, chaque flèche voulant dire \emph{«~est
mangé par~»}\index{est mangé par}~:
\[
\text{herbe} \longrightarrow \text{sauterelle}
\longrightarrow \text{lézard} \longrightarrow \text{buse}.
\]
Chaque \omterm{def:g1:living-or-not:living}{être vivant} est un \emph{maillon}\index{maillon} de la chaîne.
\end{definition}

\begin{remark}[Attention au sens de la flèche]\label{rem:g3:food-chains:arrow}
La flèche va du mangé vers le mangeur --- elle suit la nourriture.
«~Herbe $\rightarrow$ sauterelle~» se lit «~l'herbe est mangée par
la sauterelle~». Écrire les flèches à l'envers --- vers le repas ---
est l'erreur la plus fréquente dans les \omterm{def:g3:food-chains:chain}{chaînes alimentaires}~;
retiens que la flèche montre où \emph{va} la nourriture.
\end{remark}

\begin{proposition}[Toute chaîne commence par une plante]\label{prop:g3:food-chains:start}
Le premier maillon d'une \omterm{def:g3:food-chains:chain}{chaîne alimentaire} est toujours une \omterm{def:g1:plants-around-us:plant}{plante}
(ou un autre \omterm{def:g1:living-or-not:living}{être vivant} qui fabrique sa propre nourriture). Seules
les \omterm{def:g1:plants-around-us:plant}{plantes} se nourrissent sans manger personne --- à partir de la
lumière, de l'eau et de l'air, comme le disait
\cref{def:g1:plants-around-us:plant}. Chaque maillon \omterm{def:g1:animals-around-us:animal}{animal} vit,
directement ou par d'autres, de ce premier maillon vert.
\end{proposition}
\admitted

\begin{example}[Des chaînes dans trois milieux]\label{ex:g3:food-chains:three}
Dans la mare~: \omterm{def:g1:plants-around-us:plant}{plantes} aquatiques $\rightarrow$ têtard
$\rightarrow$ larve de libellule $\rightarrow$ épinoche
$\rightarrow$ héron. Dans la forêt~: \omterm{def:g1:plants-around-us:parts}{feuilles} de chêne
$\rightarrow$ \omterm{ex:g2:animals-grow-up:butterfly}{chenille} $\rightarrow$ mésange bleue $\rightarrow$
épervier. Dans le jardin~: salade $\rightarrow$ escargot
$\rightarrow$ hérisson. Premier maillon~: toujours vert.
\end{example}

\begin{method}[Construire une chaîne alimentaire]\label{met:g3:food-chains:build}
\begin{enumerate}
  \item Choisis un \omterm{def:g1:animals-around-us:animal}{animal} et demande-toi~: que mange-t-il~? Écris sa
        nourriture à sa gauche, la flèche pointant vers le
        mangeur~;
  \item pose la même question à cette nourriture, et continue vers
        la gauche jusqu'à atteindre une \omterm{def:g1:plants-around-us:plant}{plante}~;
  \item demande-toi~: qui mange mon \omterm{def:g1:animals-around-us:animal}{animal} de départ~? Prolonge
        vers la droite tant qu'il y a un mangeur~;
  \item vérifie que chaque flèche se lit \omterm{def:g3:food-chains:chain}{«~est mangé par~»} --- la
        nourriture va vers la droite.
\end{enumerate}
\end{method}

\section{Quand un maillon change}

\begin{proposition}[Les maillons dépendent les uns des autres]\label{prop:g3:food-chains:depend}
Dans une \omterm{def:g3:food-chains:chain}{chaîne alimentaire}, chaque maillon dépend de ceux qui le
précèdent et nourrit ceux qui le suivent. Si un maillon se raréfie,
les maillons suivants ont faim --- et le maillon \emph{précédent},
moins mangé, peut se multiplier. Un changement dans un maillon
voyage le long de toute la chaîne.
\end{proposition}
\admitted

\begin{example}[L'été sec]\label{ex:g3:food-chains:dry}
Une sécheresse grille l'herbe du pré. Les sauterelles, à court de
nourriture, se font rares~; les lézards qui les chassaient ont faim
et nourrissent mal la buse à leur tour. Un premier maillon jauni, et
toute la chaîne au-dessus s'amincit --- la buse n'a jamais mangé
d'herbe, et pourtant elle ressent la sécheresse.
\end{example}

\begin{example}[Enlever le chasseur]\label{ex:g3:food-chains:hunter}
Supposons que les buses disparaissent de la vallée. Les lézards,
moins chassés, se multiplient~; les sauterelles sont alors mangées
plus dur et s'amenuisent~; l'herbe, moins broutée, peut même
s'épaissir. Enlever le \emph{dernier} maillon voyage aussi le long
de la chaîne --- dans l'autre sens.
\end{example}

\begin{remark}[Les vrais prés sont des réseaux]\label{rem:g3:food-chains:web}
Notre chaîne est un fil unique, mais la buse prend aussi des
souris, le lézard mange aussi des mouches, et l'herbe nourrit
également les lapins. Beaucoup de chaînes qui se croisent font un
filet --- le \emph{réseau alimentaire} qu'une année ultérieure
étudiera (\cref{ch:g2:what-animals-eat} l'a déjà laissé entrevoir
avec le renard, le lapin et l'herbe). Les chaînes d'abord~: ce sont
les fils dont le filet est tissé.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:1}
Que veut dire la flèche d'une \omterm{def:g3:food-chains:chain}{chaîne alimentaire}~? Dans quel sens
va-t-elle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:2}
Quel genre d'\omterm{def:g1:living-or-not:living}{être vivant} est toujours le premier maillon~? Pourquoi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:3}
Écris la chaîne du pré de ce chapitre, avec ses quatre maillons et
ses flèches.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:4}
Corrige cette chaîne~: renard $\rightarrow$ lapin $\rightarrow$
herbe.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:5}
Construis une chaîne avec ces maillons~: mésange bleue, \omterm{def:g1:plants-around-us:parts}{feuilles} de
chêne, épervier, \omterm{ex:g2:animals-grow-up:butterfly}{chenille}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:6}
Dans la chaîne du jardin salade $\rightarrow$ escargot
$\rightarrow$ hérisson, qui \omterm{def:g3:food-chains:chain}{est mangé par} l'escargot~? Qui mange
l'escargot~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:7}
Sers-toi de \cref{met:g3:food-chains:build} pour faire une chaîne
d'au moins trois maillons se terminant par le héron de la mare.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:8}
Dans l'été sec de \cref{ex:g3:food-chains:dry}, pourquoi la buse
souffre-t-elle alors qu'elle ne mange jamais d'herbe~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:food-chains:9}
Les jardiniers qui empoisonnent tous les escargots perdent parfois
leur hérisson --- et rencontrent ensuite \emph{plus} de limaces et
d'escargots qu'avant. Explique les deux moitiés avec
\cref{prop:g3:food-chains:depend}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:food-chains:10}
«~Le lézard dépend de l'herbe.~» Vrai ou faux~? Défends ta réponse
avec la chaîne du pré --- et dis dans quel sens voyage un échec de
l'herbe.
\end{exercise}
