\chapter{O que é a respiração?}\label{ch:g7:what-is-respiration}

Ponha um tatuzinho num pote fechado e, horas depois, o ar do pote
mudou: parte do \omterm{def:g7:what-is-respiration:respiration}{oxigênio} dele sumiu, substituído por \omterm{def:g7:what-is-respiration:respiration}{gás carbônico}.
Repita com uma \omterm{def:g2:from-seed-to-plant:seed}{semente} germinando, um cogumelo, um peixinho dourado,
uma colherada de \omterm{def:g6:decomposers-and-soil:soil}{solo} vivo --- o mesmo resultado, todas as vezes. A
\cref{prop:g4:breathing:exchange} mostrou a troca no seu próprio
peito; neste ano a ideia se abre em toda a largura dela: a troca não é
um costume humano, e sim uma propriedade da própria vida.

\section{A respiração, definida e medida}

\begin{definition}[Respiração]\label{def:g7:what-is-respiration:respiration}
A \emph{respiração}\index{respiração} é a troca gasosa pela qual um \omterm{def:g6:exploring-our-environment:components}{ser
vivo} retira \emph{oxigênio}\index{oxigênio} do que está em volta dele
e libera \emph{gás carbônico}\index{gás carbônico}. Ela não deve ser
confundida com os \emph{movimentos respiratórios}\index{movimentos
respiratórios} --- os movimentos visíveis do peito da
\cref{def:g4:breathing:movements}: esses movimentos são um jeito de
\emph{servir} à respiração; a respiração em si é a troca gasosa, e ela
acontece em toda \omterm{def:g5:first-look-at-cells:cell}{célula} viva, com movimentos ou sem nenhum.
\end{definition}

\begin{proposition}[Como a troca é mostrada]\label{prop:g7:what-is-respiration:shown}
Dois testes clássicos de bancada revelam a \omterm{def:g7:what-is-respiration:respiration}{respiração}:
\begin{itemize}
  \item uma \emph{sonda de \omterm{def:g7:what-is-respiration:respiration}{oxigênio}} num recipiente fechado mostra o
        nível de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} caindo enquanto um \omterm{def:g6:exploring-our-environment:components}{ser vivo} fica lá dentro;
  \item a \emph{água de cal}\index{água de cal}, um líquido
        transparente que fica leitoso na presença de \omterm{def:g7:what-is-respiration:respiration}{gás carbônico},
        turva quando o ar do recipiente passa por ela.
\end{itemize}
\omterm{def:g7:what-is-respiration:respiration}{Oxigênio} caindo mais \omterm{prop:g7:what-is-respiration:shown}{água de cal} turvando, com um recipiente de
controle sem vida ficando inalterado
(\cref{met:g4:what-plants-need:fairtest} --- o teste justo nos
acompanha em toda parte): isso é \omterm{def:g7:what-is-respiration:respiration}{respiração}, medida.
\end{proposition}
\admitted

\begin{example}[A galeria dos que respiram]\label{ex:g7:what-is-respiration:gallery}
Rode os dois testes numa coleção: um camundongo --- queda rápida,
turvação rápida; um tatuzinho --- o mesmo, em escala menor; uma
ervilha germinando --- inconfundível (um pote de ervilhas brotando
pode esgotar o \omterm{def:g7:what-is-respiration:respiration}{oxigênio} dele em um dia); um cogumelo --- sim;
caramujos de lagoa --- sim; uma colherada de \omterm{def:g6:decomposers-and-soil:soil}{solo} vivo --- sim, com os
bilhões de \omterm{def:g6:decomposers-and-soil:decomposers}{decompositores} dele respirando juntos
(\cref{ex:g6:decomposers-and-soil:handful}). Só o pote de controle ---
pedrinhas, areia seca --- não muda nada.
\end{example}

\begin{remark}[As plantas também respiram]\label{rem:g7:what-is-respiration:plants}
A \cref{rem:g6:plants-are-producers:gas} disse isso de passagem; o
pote prova: uma \omterm{def:g1:plants-around-us:plant}{planta} em vaso \emph{no escuro} baixa o \omterm{def:g7:what-is-respiration:respiration}{oxigênio} do
pote e turva a \omterm{prop:g7:what-is-respiration:shown}{água de cal} como qualquer \omterm{def:g1:animals-around-us:animal}{animal}. À \omterm{def:g6:exploring-our-environment:conditions}{luz} do dia, a
fabricação das produtoras mascara o efeito --- consumindo muito mais
\omterm{def:g7:what-is-respiration:respiration}{gás carbônico} do que a \omterm{def:g7:what-is-respiration:respiration}{respiração} libera ---, mas a \omterm{def:g7:what-is-respiration:respiration}{respiração} nunca
para. Ser verde não dispensa uma \omterm{def:g5:first-look-at-cells:cell}{célula} de respirar no sentido largo;
nada que esteja vivo é dispensado.
\end{remark}

\section{Por que os seres vivos respiram}

\begin{proposition}[A respiração dá energia às células]\label{prop:g7:what-is-respiration:power}
O objetivo da troca é a \emph{\omterm{prop:g3:balanced-diet:jobs}{energia}}. Em toda \omterm{def:g5:first-look-at-cells:cell}{célula} viva, os
\omterm{def:g4:journey-of-food:digestion}{nutrientes} (\cref{def:g4:journey-of-food:digestion}) são lentamente
\emph{queimados} com o \omterm{def:g7:what-is-respiration:respiration}{oxigênio} que a \omterm{def:g7:what-is-respiration:respiration}{respiração} traz: a \omterm{prop:g3:balanced-diet:jobs}{energia} deles
é liberada para o trabalho da \omterm{def:g5:first-look-at-cells:cell}{célula} --- contrair, construir, comandar
--- e o \omterm{def:g7:what-is-respiration:respiration}{gás carbônico} é o resto queimado, levado embora. A ``fatia
queimada'' da \cref{prop:g6:matter-from-food:losses} era exatamente
isso, vista pelo balanço; aqui ela é vista pela \omterm{def:g5:first-look-at-cells:cell}{célula}.
\end{proposition}
\admitted

\begin{example}[A comparação com o fogo, usada com honestidade]\label{ex:g7:what-is-respiration:fire}
A chama de uma vela também consome \omterm{def:g7:what-is-respiration:respiration}{oxigênio} e libera \omterm{def:g7:what-is-respiration:respiration}{gás carbônico} ---
queimar e respirar são de fato o mesmo tipo de química. As diferenças
importam tanto quanto: a queima da \omterm{def:g5:first-look-at-cells:cell}{célula} é lenta, sem chama,
espalhada por milhares de passinhos, e a \omterm{prop:g3:balanced-diet:jobs}{energia} dela é capturada para
o trabalho em vez de se perder como \omterm{def:g6:exploring-our-environment:conditions}{luz} e queimadura. A \omterm{def:g7:what-is-respiration:respiration}{respiração} é a
combustão domesticada --- fogo na coleira, em cada \omterm{def:g5:first-look-at-cells:cell}{célula} sua.
\end{example}

\begin{example}[Lendo a demanda]\label{ex:g7:what-is-respiration:demand}
Como a queima dá força ao trabalho, a demanda acompanha o esforço: a
criança em disparada do \cref{ex:g4:breathing:running} respira muito
mais rápido que a criança lendo na poltrona; a ervilha germinando ---
construindo uma \omterm{def:g1:plants-around-us:plant}{planta} a toda velocidade
(\cref{def:g2:from-seed-to-plant:germination}) --- respira muito mais
rápido que a \omterm{def:g2:from-seed-to-plant:seed}{semente} seca e à espera ao lado dela, cuja \omterm{def:g7:what-is-respiration:respiration}{respiração}
fica quase em zero por anos. A \omterm{def:g7:what-is-respiration:respiration}{respiração} é o ronco do motor da vida:
alto no esforço e no crescimento, um sussurro na dormência, mudo só na
morte.
\end{example}

\section{A respiração em todo hábitat}

\begin{proposition}[Uma necessidade, muitos arredores]\label{prop:g7:what-is-respiration:habitats}
Todo \omterm{def:g6:exploring-our-environment:components}{ser vivo} precisa obter o \omterm{def:g7:what-is-respiration:respiration}{oxigênio} dele \emph{do que está em volta}
--- e os arredores diferem. O ar é rico em \omterm{def:g7:what-is-respiration:respiration}{oxigênio}; a água guarda bem
menos, dissolvido (o \cref{exo:g4:breathing:11} encontrou a
consequência); o \omterm{def:g6:decomposers-and-soil:soil}{solo} e a lama guardam menos ainda. Onde e como um \omterm{def:g6:exploring-our-environment:components}{ser
vivo} mora é, portanto, moldado pelo jeito como ele resolve o problema
de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} dele --- o assunto dos dois próximos capítulos, para o
equipamento dos \omterm{def:g1:animals-around-us:animal}{animais} (\cref{ch:g7:how-animals-breathe}) e para a
economia da água (\cref{ch:g7:oxygen-in-the-water}).
\end{proposition}
\admitted

\begin{method}[Testar uma pergunta sobre respiração]\label{met:g7:what-is-respiration:test}
Para testar qualquer afirmação sobre \omterm{def:g7:what-is-respiration:respiration}{respiração} (``as \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} respiram
mais rápido no calor'', ``a lama da lagoa respira''):
\begin{enumerate}
  \item feche o sujeito num recipiente, com uma sonda de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} ou
        uma armadilha de \omterm{prop:g7:what-is-respiration:shown}{água de cal};
  \item monte o controle gêmeo --- o mesmo recipiente, com o sujeito
        ausente ou sem vida;
  \item varie uma condição de cada vez, se a afirmação citar alguma
        (calor, \omterm{def:g6:exploring-our-environment:conditions}{umidade}, escuro);
  \item leia o veredito na diferença: \omterm{def:g7:what-is-respiration:respiration}{oxigênio} caído, \omterm{prop:g7:what-is-respiration:shown}{água de cal}
        turvada --- contra a imobilidade do controle.
\end{enumerate}
\end{method}

\begin{example}[O método numa semente fria]\label{ex:g7:what-is-respiration:coldseed}
Afirmação: ervilhas germinando respiram mais devagar no frio. Dois
potes de ervilhas brotando, um à \omterm{def:g6:exploring-our-environment:conditions}{temperatura} \omterm{def:g6:exploring-our-environment:components}{ambiente} e outro na
geladeira; armadilhas de \omterm{prop:g7:what-is-respiration:shown}{água de cal} nos dois; um terceiro sem vida
como controle. No dia seguinte: a \omterm{prop:g7:what-is-respiration:shown}{água de cal} do pote quente está
leitosa, a do pote frio mal ficou embaçada, a do controle está
transparente. Veredito: a \omterm{def:g7:what-is-respiration:respiration}{respiração} acompanha o ritmo de trabalho das
ervilhas --- a regra das \omterm{def:g6:exploring-our-environment:conditions}{condições} da
\cref{prop:g6:decomposers-and-soil:speed}, medida em gás.
\end{example}

\section{Problema: Os potes lacrados}

\begin{problem}[{Problema de fim de semana --- seis potes, dois
mistérios, uma troca gasosa}]\label{pb:g7:what-is-respiration:1}
Para a feira de ciências, a turma prepara seis potes lacrados, cada um
com uma sonda de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} e uma armadilha de \omterm{prop:g7:what-is-respiration:shown}{água de cal}: A ---
pedrinhas; B --- ervilhas secas; C --- ervilhas germinando; D --- um
camundongo (com comida, água e só por algumas horas, sob a vigilância
do professor); E --- uma \omterm{def:g1:plants-around-us:plant}{planta} verde na \omterm{def:g6:exploring-our-environment:conditions}{luz}; F --- uma \omterm{def:g1:plants-around-us:plant}{planta} do mesmo
tipo dentro de uma caixa escura.

\medskip\noindent\textbf{Parte I --- As previsões.}
\begin{enumerate}
  \item Preveja as leituras de A e diga o papel de A na experiência.
  \item Preveja B e C, e explique a diferença, embora os dois potes
        tenham ``as mesmas'' ervilhas.
  \item Preveja D e diga por que o professor limita as horas.
  \item E e F têm \omterm{def:g1:plants-around-us:plant}{plantas} idênticas. Preveja a tendência do \omterm{def:g7:what-is-respiration:respiration}{oxigênio}
        de cada pote e concilie a diferença usando a
        \cref{rem:g7:what-is-respiration:plants}.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Resultados e leituras.}
\begin{enumerate}[resume]
  \item Os dados da manhã batem com as previsões --- só que uma colega
        fica espantada porque a \omterm{def:g1:plants-around-us:plant}{planta} de F ``respira como um
        \omterm{def:g1:animals-around-us:animal}{animal}''. Exponha o que F de fato mostra e o que E esconde.
  \item O \omterm{def:g7:what-is-respiration:respiration}{oxigênio} de C caiu três vezes mais rápido que o de B, por
        grama de ervilha. Traduza esse fator de três nos termos do
        \cref{ex:g7:what-is-respiration:demand}.
  \item A \omterm{prop:g7:what-is-respiration:shown}{água de cal} de D turvou mais rápido que todas. Dê a cadeia
        --- do corpo quente e em movimento do camundongo até o líquido
        leitoso --- em quatro passos.
  \item Diga a única conclusão que B, C, D e F apoiam, numa frase
        começando com ``Todo \omterm{def:g6:exploring-our-environment:components}{ser vivo}\dots''.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Indo mais fundo.}
\begin{enumerate}[resume]
  \item Um sétimo pote é proposto: fatias de cogumelo. Preveja as
        leituras dele e diga que ponto, no nível dos grandes grupos de
        \omterm{def:g6:exploring-our-environment:components}{seres vivos}, ele acrescentaria ao
        \cref{ex:g7:what-is-respiration:gallery}.
  \item Quem visita a feira pergunta por que o pote C interessa a quem
        \omterm{def:g1:plants-around-us:plant}{planta}. Responda com o armazém de grãos do
        \cref{exo:g7:what-is-respiration:11}.
  \item Um oitavo pote: água de lagoa com a lama dela. O \omterm{def:g7:what-is-respiration:respiration}{oxigênio} cai
        durante a noite. Quem no pote está respirando, e por que esse
        achado importa para a pergunta do
        \cref{ch:g7:oxygen-in-the-water}?
  \item Feche o cartaz da feira com a distinção do capítulo: que potes
        mostraram \emph{movimentos respiratórios} e quais mostraram
        \emph{\omterm{def:g7:what-is-respiration:respiration}{respiração}}? (Cuidado --- uma das palavras cobre todos.)
\end{enumerate}
\end{problem}
