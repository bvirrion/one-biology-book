\section*{Bab \ref{ch:g3:bones-and-muscles} --- Tulang dan Otot}

\begin{solution}{exo:g3:bones-and-muscles:1}
\omterm{def:g3:bones-and-muscles:skeleton}{Tengkorak} melindungi otak; \omterm{def:g3:bones-and-muscles:skeleton}{tulang belakang} menahan batang tubuh tetap
tegak (sebuah tiang yang dapat melengkung); \omterm{def:g3:bones-and-muscles:skeleton}{tulang rusuk} membentuk
sangkar pelindung mengelilingi jantung dan paru-paru; \omterm{def:g3:bones-and-muscles:skeleton}{tulang} \omterm{def:g1:my-body:parts}{anggota
gerak} membawa \omterm{def:g1:my-body:parts}{lengan} dan \omterm{def:g1:my-body:parts}{tungkai}.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:2}
Kira-kira $200$. Yang terpanjang adalah \omterm{def:g3:bones-and-muscles:skeleton}{tulang} paha.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:3}
Tidak --- \omterm{def:g3:bones-and-muscles:skeleton}{tulang} yang tertekuk adalah \omterm{def:g3:bones-and-muscles:skeleton}{tulang} yang patah. Semua tekukan
terjadi pada \omterm{def:g1:my-body:joint}{sendi}, tempat dua \omterm{def:g3:bones-and-muscles:skeleton}{tulang} bertemu.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:4}
Ia sebuah tiang yang terdiri atas lebih dari $20$ \omterm{def:g3:bones-and-muscles:skeleton}{tulang} kecil,
masing-masing mampu berputar beberapa derajat terhadap tetangganya;
banyak tekukan kecil itu menjumlah menjadi satu lengkungan besar yang
lentur.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:5}
Ia memendek dan menebal, lalu menarik \omterm{def:g3:bones-and-muscles:skeleton}{tulang} tempatnya melekat. Tidak
ada \omterm{def:g3:bones-and-muscles:muscle}{otot} yang dapat mendorong.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:6}
Karena sebuah \omterm{def:g3:bones-and-muscles:muscle}{otot} hanya menarik, satu \omterm{def:g3:bones-and-muscles:muscle}{otot} hanya dapat menggerakkan
\omterm{def:g1:my-body:joint}{sendinya} ke satu arah. Menekuk: \omterm{def:g3:bones-and-muscles:muscle}{otot} depan \omterm{def:g3:bones-and-muscles:muscle}{berkontraksi} lalu menarik
\omterm{def:g1:my-body:parts}{lengan} bawah ke atas sementara \omterm{def:g3:bones-and-muscles:muscle}{otot} belakang mengendur. Meluruskan:
\omterm{def:g3:bones-and-muscles:muscle}{otot} belakang \omterm{def:g3:bones-and-muscles:muscle}{berkontraksi} sementara \omterm{def:g3:bones-and-muscles:muscle}{otot} depan mengendur.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:7}
\omterm{def:g3:bones-and-muscles:skeleton}{Tulang} hidupnya perlahan membangun ulang patahannya sendiri; gipsnya
hanya menahan potongannya tetap diam selama \omterm{def:g3:bones-and-muscles:skeleton}{tulangnya} bekerja. Ini
menunjukkan bahwa \omterm{def:g3:bones-and-muscles:skeleton}{tulang} adalah bahan yang hidup, diberi makan oleh
darah --- bukan mati seperti kapur.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:8}
Menekuk: bagian depan \omterm{def:g1:my-body:parts}{lengan} membengkak dan mengeras (\omterm{def:g3:bones-and-muscles:muscle}{otot} itu sedang
\omterm{def:g3:bones-and-muscles:muscle}{berkontraksi}). Meluruskan: bagian depannya melunak sementara bagian
belakang \omterm{def:g1:my-body:parts}{lengan} mengencang --- pasangannya bertukar peran.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:9}
\omterm{def:g1:my-body:joint}{Lutut} adalah \omterm{def:g1:my-body:joint}{sendi} yang dibangun untuk lipatan besar, digerakkan oleh
\omterm{def:g3:bones-and-muscles:muscle}{otot} terkuat di tubuh; \omterm{def:g3:bones-and-muscles:skeleton}{tulang belakang} dibangun untuk tekukan lembut
yang tersebar pada banyak \omterm{def:g3:bones-and-muscles:skeleton}{tulang} kecil. Mengangkat dengan \omterm{def:g1:my-body:joint}{lutut}
tertekuk dan punggung lurus menyerahkan pekerjaan beratnya kepada
\omterm{def:g1:my-body:parts}{tungkai} dan menyelamatkan \omterm{def:g3:bones-and-muscles:skeleton}{tulang belakang}.
\end{solution}

\begin{solution}{exo:g3:bones-and-muscles:10}
Tiap tali, seperti tiap \omterm{def:g3:bones-and-muscles:muscle}{otot}, hanya dapat menarik. Jawaban pembuat
bonekanya sama dengan jawaban ahli \omterm{def:g1:living-or-not:living}{biologi}: dua tali untuk tiap
gerakan --- satu untuk menarik \omterm{def:g1:my-body:parts}{lengannya} ke atas, satu lagi untuk
menariknya kembali ke bawah --- yang bekerja bergantian, persis
seperti pasangan \omterm{def:g3:bones-and-muscles:muscle}{otot} tubuh.
\end{solution}
