\chapter{Cómo se extienden las plantas}\label{ch:g6:how-plants-spread}

En marzo se abre a máquina el desmonte de una carretera nueva: tierra
cruda y desnuda, ni una mota verde. Vuelve en septiembre y estará
cosido de \omterm{prop:g2:from-seed-to-plant:cycle}{plántulas} por todas partes --- amapolas, dientes de león,
brotes de sauce, hierbas. Nadie lo sembró. Y sin embargo las \omterm{def:g1:plants-around-us:plant}{plantas},
que no pueden dar ni un paso, colonizaron el terreno desnudo en unos
meses. Este capítulo trata de su departamento de viajes.

\section{Las plantas viajan como semillas y esporas}

\begin{proposition}[Plantas quietas, comienzos viajeros]\label{prop:g6:how-plants-spread:travel}
Una \omterm{def:g1:plants-around-us:plant}{planta} adulta está fija de por vida
(\cref{def:g1:plants-around-us:plant}) --- así que una \omterm{def:g5:biodiversity:species}{especie} se
extiende enviando fuera sus \emph{comienzos}: \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} y \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos}
(\cref{ch:g5:flower-to-fruit}), \omterm{def:g4:plants-without-seeds:spore}{esporas}
(\cref{def:g4:plants-without-seeds:spore}) o \omterm{ex:g4:plants-without-seeds:runner}{estolones} y otros avances
vegetativos (\cref{def:g4:plants-without-seeds:vegetative}). Ocupar así
terreno nuevo se llama \emph{colonización}\index{colonización}.
\end{proposition}
\admitted

\begin{definition}[Dispersión]\label{def:g6:how-plants-spread:dispersal}
La \emph{dispersión}\index{dispersión} es el transporte de \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas},
\omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} o \omterm{def:g4:plants-without-seeds:spore}{esporas} lejos de la \omterm{def:g1:plants-around-us:plant}{planta} madre. Los transportistas son pocos
y antiguos: el \emph{viento}, los \emph{\omterm{def:g1:animals-around-us:animal}{animales}}, el \emph{agua} ---
y algunos \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} que lanzan ellos mismos sus \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas}.
\end{definition}

\begin{example}[¿Y para qué irse de casa?]\label{ex:g6:how-plants-spread:whyleave}
Una \omterm{def:g2:from-seed-to-plant:seed}{semilla} que cae al pie de su madre \omterm{def:g2:from-seed-to-plant:germination}{germina} a la sombra de su madre
y entre las \omterm{def:g1:plants-around-us:parts}{raíces} de su madre --- compitiendo por la misma \omterm{def:g6:exploring-our-environment:conditions}{luz} y la
misma agua de la \cref{prop:g4:what-plants-need:needs} con una rival
cien veces más grande. La \omterm{def:g6:how-plants-spread:dispersal}{dispersión} es la huida de ese partido
perdido: la distancia es el primer regalo que una \omterm{def:g1:plants-around-us:plant}{planta} le hace a sus
crías.
\end{example}

\section{Los transportistas y los frutos hechos para ellos}

\begin{proposition}[Construidos a la medida del transportista]\label{prop:g6:how-plants-spread:builds}
Las formas de los \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} y de las \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} encajan con sus
transportistas --- el ajuste de la
\cref{def:g4:adapted-to-their-world:adaptation}, aplicado al equipaje:
\begin{itemize}
  \item el \emph{viento}: paracaídas (las \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} con plumón del
        diente de león), alas (las sámaras del arce y del fresno, que
        bajan girando de lado) o \omterm{def:g4:plants-without-seeds:spore}{esporas} finas como el polvo que
        sencillamente flotan (los millones del helecho);
  \item los \emph{\omterm{def:g1:animals-around-us:animal}{animales}, por fuera}: los \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} con garfios ---
        bardana, amor de hortelano --- cuyos ganchos se agarran al \omterm{def:g1:animals-around-us:covering}{pelo}
        y a los calcetines y viajan hasta que se los quitan;
  \item los \emph{\omterm{def:g1:animals-around-us:animal}{animales}, por dentro}: \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} carnosos y dulces ---
        cerezas, moras, escaramujos --- que se comen enteros; las
        \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas}, de cubierta dura, pasan sin daño y se depositan en
        otra parte, con abono incluido (la respuesta del
        \cref{exo:g3:flowers-fruits-seeds:10}, completada);
  \item el \emph{agua}: \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} que flotan --- las balsitas del aliso,
        la cáscara transoceánica del coco;
  \item el \emph{lanzamiento propio}: vainas que se secan, se tensan y
        estallan --- el tojo y las violetas disparan sus \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} a
        metros de la madre.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Leer la cuneta]\label{ex:g6:how-plants-spread:roadside}
El desmonte de la carretera en septiembre, descifrado: dientes de león
y sauces --- \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} con plumón, llegadas volando con el viento;
amapolas --- sus \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} diminutas ya estaban en la tierra removida,
esperando como solo saben esperar las \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas}
(\cref{exo:g2:from-seed-to-plant:9}); \omterm{prop:g2:from-seed-to-plant:cycle}{plántulas} de zarzamora bajo los
postes de la valla --- depositadas por los pájaros posados, con abono
incluido; bardana junto al sendero --- llegada en un perro. La ruta de
llegada de cada colonizadora está escrita en su equipo.
\end{example}

\begin{method}[Nombrar al transportista a partir del fruto]\label{met:g6:how-plants-spread:name}
Ante un \omterm{prop:g3:flowers-fruits-seeds:transform}{fruto} o una \omterm{def:g2:from-seed-to-plant:seed}{semilla}:
\begin{enumerate}
  \item busca equipo de vuelo --- plumones, alas, finura de polvo: el
        viento;
  \item busca ganchos: el \omterm{def:g1:animals-around-us:covering}{pelo} de un \omterm{def:g1:animals-around-us:animal}{animal}, por fuera;
  \item ¿carnoso, de color, dulce? el \omterm{def:g4:journey-of-food:tract}{tubo digestivo} de un \omterm{def:g1:animals-around-us:animal}{animal}, por
        dentro --- comprueba si las \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} tienen cubierta dura;
  \item ¿flota, y está cerca del agua? el agua;
  \item ¿una vaina seca, retorcida y en tensión? lanzamiento propio;
  \item pon a prueba tu veredicto: déjalo caer, ponlo a flotar,
        apriétalo contra una manga de lana.
\end{enumerate}
\end{method}

\section{Colonizar sin semillas}

\begin{example}[El avance por el suelo]\label{ex:g6:how-plants-spread:vegetative}
Junto a las llegadas por aire, algunas colonizadoras andan con sus
propios pies vegetativos (\cref{ch:g4:plants-without-seeds}): la grama
avanza bajo tierra con \omterm{def:g1:plants-around-us:parts}{tallos} rastreros, metro a metro; las zarzas se
arquean y echan \omterm{def:g1:plants-around-us:parts}{raíces} donde tocan sus puntas; los \omterm{ex:g4:plants-without-seeds:runner}{estolones} de la
fresa saltan por el terreno abierto. Lentas, de corto alcance --- e
imparables en un jardín descuidado, como sabe cualquier hortelano.
\end{example}

\begin{remark}[Colonización a dos velocidades]\label{rem:g6:how-plants-spread:twospeeds}
Los dos sistemas se reparten el trabajo. Las \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} y las \omterm{def:g4:plants-without-seeds:spore}{esporas} son
el servicio de \emph{largo alcance} --- aterrizajes afortunados y raros
muy lejos, el diente de león cruzando el pueblo. El avance vegetativo
es el servicio de \emph{corto alcance} --- la ocupación densa y segura
del terreno de al lado. La mata de zarzas que se extiende desde el seto
usa los dos a la vez: \omterm{def:g1:plants-around-us:parts}{tallos} arqueados para el metro, \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} llevadas
por pájaros para el kilómetro.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:1}
¿Qué es la \omterm{prop:g6:how-plants-spread:travel}{colonización}? ¿Qué es la \omterm{def:g6:how-plants-spread:dispersal}{dispersión} y cuáles son los
transportistas principales?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:2}
¿Por qué le conviene más a una \omterm{def:g2:from-seed-to-plant:seed}{semilla} \omterm{def:g2:from-seed-to-plant:germination}{germinar} lejos de su madre?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:3}
Nombra al transportista de cada uno: el plumón del diente de león, el
\omterm{prop:g3:flowers-fruits-seeds:transform}{fruto} con ganchos de la bardana, la cereza, la sámara del arce, el coco.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:4}
¿Cómo sobreviven al viaje las \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} de una mora comida por un
pájaro y con qué llegan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:5}
¿Qué servicio de viajes usan el tojo y las violetas? Describe el
mecanismo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:6}
Nombra dos colonizadoras vegetativas y cómo avanza cada una.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:7}
Descifra el desmonte de la carretera: para el diente de león, la
amapola, la zarzamora y la bardana, da la ruta de llegada que nombra el

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:8}
Pásale el \cref{met:g6:how-plants-spread:name} a: (a) una \omterm{def:g2:from-seed-to-plant:seed}{semilla} de
cardo con plumón; (b) un escaramujo; (c) una bolita de amor de
hortelano que se te pega al jersey.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:9}
Explica la idea de las ``dos velocidades'' de la
\cref{rem:g6:how-plants-spread:twospeeds} con la mata de zarzas: ¿qué
sistema hace el metro y cuál el kilómetro?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:10}
El helecho colonizó el talud norte húmedo del desmonte nuevo, pero no
su talud sur seco. Combina la
\cref{prop:g4:plants-without-seeds:damp} y la
\cref{prop:g6:exploring-our-environment:decide} para dar la explicación.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:11}
Una isla emerge desnuda del mar, a kilómetros de la costa. ¿Qué
servicios de \omterm{def:g6:how-plants-spread:dispersal}{dispersión} pueden llegar siquiera hasta ella? Predice el
equipo de sus primeras \omterm{def:g1:plants-around-us:plant}{plantas} colonizadoras.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:how-plants-spread:12}
La pulpa dulce le sale cara de construir a una \omterm{def:g1:plants-around-us:plant}{planta}. Con el
\cref{ex:g6:how-plants-spread:whyleave} y el servicio de por dentro de
los \omterm{def:g1:animals-around-us:animal}{animales}, explica qué compra el cerezo con todo ese azúcar --- y
por qué las cubiertas duras de las \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} son la otra mitad de la
compra.
\end{exercise}

\section{Problema: El campo abandonado}

\begin{problem}[{Problema de fin de semana --- diez años de
colonización, vistos desde una cancela}]\label{pb:g6:how-plants-spread:1}
Un campo arado se abandona en otoño. Un naturalista lo fotografía desde
la cancela cada junio durante diez años y va tomando notas.

\medskip\noindent\textbf{Parte I --- El primer año.}
\begin{enumerate}
  \item Junio del primer año: la tierra desnuda está salpicada de
        amapolas, dientes de león y cardos. Nombra de cada uno el
        origen y la ruta probables --- las amapolas desde la propia
        tierra, los demás por el aire. ¿Qué equipo hizo posibles las
        llegadas por aire?
  \item En el primer año no aparecen zarzamoras en medio del campo,
        pero hay tres \omterm{prop:g2:from-seed-to-plant:cycle}{plántulas} bajo los postes viejos de la valla.
        Explica el patrón con el servicio de por dentro de los \omterm{def:g1:animals-around-us:animal}{animales}
        --- ¿por qué justo debajo de los postes?
  \item El naturalista no encuentra \omterm{prop:g4:plants-without-seeds:damp}{helechos} por ninguna parte, aunque
        hay un bosque lleno de ellos a barlovento. Las \omterm{def:g4:plants-without-seeds:spore}{esporas} de
        helecho llegaron sin duda (el
        \cref{ex:g4:plants-without-seeds:fern} cuenta sus millones).
        ¿Qué condición del campo abierto, según la
        \cref{prop:g4:plants-without-seeds:damp}, se las rechazó?
  \item La grama invade solo desde el borde del campo, un metro o dos
        al año. ¿Qué sistema de \omterm{prop:g6:how-plants-spread:travel}{colonización} es este y por qué no puede
        aparecer en medio del campo como hicieron los dientes de león?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Los años siguientes.}
\begin{enumerate}[resume]
  \item Para el tercer año, las hierbas y las herbáceas altas han
        cerrado el terreno desnudo y las amapolas han desaparecido de
        la vegetación en pie. ¿Dónde están ahora las amapolas y en qué
        forma? (Acuérdate del desmonte de la carretera y de la
        \cref{prop:g6:life-through-seasons:herbs}.)
  \item A partir del cuarto año, arcos de zarza y majuelos jóvenes
        salpican el campo, más espesos cerca de los setos y bajo la
        línea de la valla. Nombra los dos servicios que los colocaron y
        la prueba que hay en el patrón.
  \item Para el año ocho, los brotes de sauce y de abedul están más
        altos que las zarzas --- los dos árboles de \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas} diminutas,
        con plumón o ligeras como el polvo. ¿Por qué siguen casi
        ausentes los robles y los castaños, de \omterm{def:g2:from-seed-to-plant:seed}{semilla} pesada, y qué
        mensajero \omterm{def:g1:animals-around-us:animal}{animal} acaba plantando incluso a esos?
  \item Las fotografías enseñan que el reparto del campo cambia año a
        año aunque no lo haya tocado ningún hortelano. Enuncia la regla
        que lo impulsa: ¿qué cambia la vegetación de cada año en las
        propias \omterm{def:g6:exploring-our-environment:conditions}{condiciones} del campo --- sobre todo la \omterm{def:g6:exploring-our-environment:conditions}{luz} a ras de
        suelo --- y a quién favorece después ese cambio?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La lección general.}
\begin{enumerate}[resume]
  \item Ordena a las colonizadoras por orden de llegada: las \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas}
        que esperaban en la tierra, las viajeras del viento, los
        arbustos llevados por \omterm{def:g1:animals-around-us:animal}{animales}, los árboles de \omterm{def:g2:from-seed-to-plant:seed}{semilla} pesada.
        Empareja cada oleada con el servicio de \omterm{def:g6:how-plants-spread:dispersal}{dispersión} que marcó su
        momento.
  \item Las grietas del pavimento de una plaza sacan el mismo reparto
        del primer año --- amapolas, dientes de león --- pero nunca
        llegan a los arbustos del cuarto año. Da dos razones por las que
        allí se atasca la sucesión.
  \item El naturalista afirma: ``si se lo deja en paz, este campo será
        un bosque''. Defiende la afirmación con la tendencia de los diez
        años --- y nombra la excepción del prado de la
        \cref{prop:g5:people-and-nature:managed} que demuestra que ``si
        se lo deja en paz'' es la clave de la frase.
  \item Resume el problema entero en dos frases usando las palabras
        \emph{\omterm{def:g6:how-plants-spread:dispersal}{dispersión}}, \emph{\omterm{def:g6:exploring-our-environment:conditions}{condiciones}} y \emph{\omterm{prop:g6:how-plants-spread:travel}{colonización}}.
\end{enumerate}
\end{problem}
