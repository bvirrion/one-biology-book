\chapter{Populations et démographie}\label{ch:b1:populations}

Une fiole de levure part d'un million de \omterm{def:b1:cell-unit-of-life:cell}{cellules} et, seize heures plus
tard, en contient soixante millions et a cessé de croître~; un troupeau
de cerfs sur une île croît d'un dixième chaque année jusqu'à l'hiver où
la nourriture manque~; les lynx de la forêt boréale montent et
s'effondrent tous les dix ans, un an derrière les lièvres qu'ils
mangent. Chacun est une \emph{\omterm{def:b1:populations:population}{population}} --- les individus d'une même
espèce en un même lieu --- et chacun obéit à une arithmétique que l'on
peut écrire. Ce chapitre décrit comment on dénombre une \omterm{def:b1:populations:population}{population},
comment on tabule ses naissances et ses morts, les deux équations de la
croissance, les forces qui contiennent les effectifs, et ce que la même
arithmétique dit de notre propre espèce.

\section{Compter une population}

\begin{definition}[Population, densité, dispersion]\label{def:b1:populations:population}
\index{population}\index{densite de population@densité de population}
Une \emph{population} est l'ensemble des individus d'une même espèce
vivant dans une aire donnée à un moment donné et capables de se
reproduire entre eux. Son \emph{effectif} $N$ est le nombre
d'individus~; sa \emph{densité}, le nombre par unité d'aire ou de
volume~; sa \emph{dispersion}, le motif de leur espacement ---
\emph{agrégée} là où les ressources ou la vie sociale les rassemblent
(troupeaux, bancs, taches d'une plante), \emph{uniforme} là où ils se
repoussent (oiseaux territoriaux, arbustes du désert en concurrence
pour l'eau), \emph{aléatoire} là où ni l'un ni l'autre n'agit. Les
populations ont aussi une \emph{structure d'âges}, une \emph{sex-ratio}
et des taux de natalité, de mortalité, d'immigration et d'émigration,
dont le bilan est la croissance.
\end{definition}

\begin{method}[Estimer une population]\label{met:b1:populations:counting}
\index{capture-marquage-recapture}
\begin{enumerate}
  \item \omterm{def:b1:organism-environment:organism}{Organismes} fixés ou lents~: compter dans des \emph{quadrats}
        d'aire connue placés au hasard, et multiplier la densité
        moyenne par l'aire totale~; la dispersion des quadrats donne la
        précision et révèle le mode de dispersion.
  \item Animaux mobiles~: \emph{capture--marquage--recapture}. Capturer
        $M$ individus, les marquer et les relâcher~; capturer plus tard
        $C$ individus, dont $R$ sont marqués. Si les marqués se mêlent
        librement et que rien n'a changé, la fraction marquée de la
        seconde capture égale la fraction marquée de la \omterm{def:b1:populations:population}{population},
        $R/C = M/N$, d'où $N \approx MC/R$ (l'estimation de
        Lincoln--Petersen).
  \item Vérifier les hypothèses~: ni naissance, ni mort, ni migration
        entre les deux captures, des marques qui ne tombent pas et ne
        changent pas le comportement de l'animal, une capturabilité
        égale. Chaque manquement biaise $N$ dans un sens prévisible.
  \item Indices indirects là où le comptage est impossible~: crottes,
        traces, chants, nids, l'\omterm{def:b1:nucleic-acids:chain}{ADN} d'un litre d'eau d'étang.
\end{enumerate}
\end{method}

\begin{example}[Des campagnols dans une prairie]\label{ex:b1:populations:voles}
Soixante campagnols piégés, marqués et relâchés dans un hectare de
prairie~; trois nuits plus tard, quatre-vingts sont piégés, dont douze
portent une marque~: $N \approx 60\times 80/12 = 400$ campagnols, quatre
pour cent mètres carrés. Si les campagnols marqués, ayant appris que
les pièges contiennent de la nourriture, sont capturés plus volontiers,
$R$ est gonflé et $N$ sous-estimé~; si le marquage les a éloignés des
pièges, c'est l'inverse.
\end{example}

\section{Démographie~: naissances et morts par âge}

\begin{definition}[Table de survie]\label{def:b1:populations:lifetable}
\index{table de survie}\index{survie}
Une \emph{table de survie} suit une cohorte d'individus nés ensemble
tout au long de leur vie. Pour chaque âge $x$, elle enregistre $l_x$,
la \emph{survie} --- la fraction de la cohorte encore vivante à l'âge
$x$ --- et $m_x$, la \emph{fécondité} --- le nombre moyen de filles nées
d'une femelle d'âge $x$. On en tire~: la mortalité entre deux âges,
$q_x = 1 - l_{x+1}/l_x$~; le \emph{taux net de reproduction}
$R_0 = \sum_x l_x m_x$, le nombre de filles qu'une femelle nouveau-née
aura en moyenne au cours de sa vie~; et le \emph{temps de génération}
$T = \sum_x x\,l_x m_x / R_0$, l'âge moyen des mères de ces filles. Une
\omterm{def:b1:populations:population}{population} telle que $R_0 > 1$ croît, telle que $R_0 < 1$ décline~; sur
une génération elle est multipliée par $R_0$.
\end{definition}

\begin{proposition}[Trois formes de courbe de survie]\label{prop:b1:populations:survivorship}
Porter $\log l_x$ en fonction de l'âge donne trois types de courbe.
\emph{Type I}~: la plupart des individus atteignent la vieillesse et
meurent dans un intervalle étroit (grands mammifères, humains des pays
riches). \emph{Type II}~: une probabilité de mourir constante à tout
âge, une droite en échelle logarithmique (beaucoup d'oiseaux, de
lézards, de rongeurs). \emph{Type III}~: une mortalité énorme chez les
jeunes et une longue vie pour les rares survivants (huîtres, poissons,
la plupart des plantes et des insectes). La forme découle du mode de
vie~: une espèce qui investit peu dans chacun de très nombreux
descendants a une courbe de type III~; celle qui investit beaucoup dans
chacun de quelques-uns, une courbe de type I.
\end{proposition}

\begin{example}[Une table de survie pour un troupeau de cerfs]\label{ex:b1:populations:deer}
Sur mille faons, six cents atteignent leur premier anniversaire, et les
survivants perdent ensuite un dixième de leur nombre chaque année
jusqu'à ce que peu dépassent neuf ans~; les biches mettent bas dès l'âge
de deux ans, environ huit dixièmes de fille par an dans leur pleine
force. La somme des $l_x m_x$ donne $R_0 \approx 1.5$~: chaque biche
laisse une fille et demie, et le troupeau croît de moitié à chaque
génération d'environ quatre ans --- une hausse de onze pour cent par an
jusqu'à ce que quelque chose l'arrête.
\end{example}

\section{La croissance}

\begin{theorem}[La croissance exponentielle]\label{thm:b1:populations:exponential}
\index{croissance exponentielle}\index{taux intrinseque d'accroissement@taux intrinsèque d'accroissement}
Une \omterm{def:b1:populations:population}{population} dont les taux de natalité et de mortalité par tête $b$
et $d$ sont constants croît à une vitesse proportionnelle à son effectif,
\[
\frac{\dd N}{\dd t} = rN, \qquad r = b - d, \qquad N(t) = N_0\,e^{rt},
\]
où $r$ est le \emph{\omterm{thm:b1:populations:exponential}{taux intrinsèque d'accroissement}}. Elle double
tous les $\ln 2/r$~; une \omterm{def:b1:populations:population}{population} avec $r = \qty{0.11}{yr^{-1}}$
double en \qty{6.3}{yr}, une autre avec $r = \qty{0.5}{h^{-1}}$ en
\qty{83}{min}. À partir d'une \omterm{def:b1:populations:lifetable}{table de survie}, $r \approx \ln R_0 / T$.
\end{theorem}
\begin{proof}
Pendant un court intervalle $\dd t$, les naissances valent
$bN\,\dd t$ et les morts
$dN\,\dd t$, donc $\dd N = (b - d)N\,\dd t$~; l'équation
$\dd N/\dd t = rN$ a pour solution $N_0 e^{rt}$ (la dérivée de
$e^{rt}$ vaut $re^{rt}$). Doublement~: $e^{rt} = 2$ donne $t = \ln 2/r$.
Sur une génération $T$, la \omterm{def:b1:populations:population}{population} est multipliée par $R_0 = e^{rT}$,
d'où $r = \ln R_0/T$.
\end{proof}

\begin{theorem}[La croissance logistique]\label{thm:b1:populations:logistic}
\index{croissance logistique}\index{capacite de charge@capacité de charge}
Si le taux de croissance par tête décroît linéairement quand la
\omterm{def:b1:populations:population}{population} approche une \emph{\omterm{thm:b1:populations:logistic}{capacité de charge}} $K$ --- l'effectif que
le milieu peut soutenir ---
\[
\frac{\dd N}{\dd t} = rN\left(1 - \frac{N}{K}\right), \qquad
N(t) = \frac{K}{1 + \dfrac{K - N_0}{N_0}\,e^{-rt}} .
\]
La croissance est presque exponentielle tant que $N \ll K$, maximale en
$N = K/2$ (où $\dd N/\dd t = rK/4$), et s'annule en $N = K$, dont la
\omterm{def:b1:populations:population}{population} approche comme d'un plateau~: une courbe en S.
\end{theorem}
\begin{proof}
Écrivons l'équation sous la forme $\dd N/[N(1 - N/K)] = r\,\dd t$ et
décomposons le membre de gauche en $\dd N/N + \dd N/(K - N)$~;
l'intégration donne $\ln[N/(K - N)] = rt + \text{cte}$, et la résolution
en $N$ avec $N(0) =
N_0$ donne la formule. Dériver $rN(1 - N/K)$ par rapport à
$N$ et annuler la dérivée donne le maximum en $N = K/2$.
\end{proof}

\begin{proposition}[La densité-dépendance]\label{prop:b1:populations:densitydependence}
\index{densite-dependance@densité-dépendance}
La croissance ralentit quand la densité monte parce que le taux de
natalité par tête baisse ou que le taux de mortalité monte avec
l'entassement~: la nourriture par tête diminue, les territoires
manquent, déchets et maladies s'accumulent, les prédateurs se
concentrent. Ces facteurs \emph{densité-dépendants} régulent une
\omterm{def:b1:populations:population}{population} vers $K$. Les facteurs \emph{densité-indépendants} --- gel,
crue, incendie, sécheresse --- tuent une fraction quel que soit
l'effectif et ne peuvent réguler, seulement perturber~; les \omterm{def:b1:populations:population}{populations}
de bien des insectes sont gouvernées surtout par le temps qu'il fait et
n'approchent jamais une \omterm{thm:b1:populations:logistic}{capacité de charge}.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Gause (1934) cultiva des \emph{Paramecium} dans des tubes de milieu
renouvelé chaque jour~: les comptages suivirent une courbe logistique
jusqu'à un plateau fixé par la nourriture, et doubler la nourriture
doublait le plateau. Pearl (1927) trouva la même chose chez les
drosophiles en bouteille, et Carlson (1913) chez la levure. Sur le
terrain, les bruants chanteurs d'une île pondent des couvées plus
petites et perdent plus de jeunes quand la \omterm{def:b1:populations:population}{population} reproductrice est
grande, et le nombre de territoires plafonne le nombre de reproducteurs~;
si l'on retire des oiseaux, ils sont remplacés à partir d'un surplus
qui était tenu à l'écart.
\end{proof}

\begin{definition}[Stratégies démographiques]\label{def:b1:populations:rK}
\index{strategie r@stratégie r}\index{strategie K@stratégie K}
Les espèces diffèrent par leur place entre deux extrêmes. Une espèce
\emph{à sélection $r$} --- adventices, pucerons, souris, la plupart des
poissons --- est bâtie pour un $r$ élevé~: elle mûrit tôt, produit de
nombreux descendants petits, investit peu dans chacun, vit brièvement,
et exploite les milieux neufs ou perturbés avant que la concurrence
n'arrive, avec une \omterm{def:b1:populations:lifetable}{survie} de type III. Une espèce \emph{à sélection $K$}
--- chênes, éléphants, albatros --- est bâtie pour persister près de
$K$~: elle mûrit tard, produit peu de descendants grands, en prend soin,
vit longtemps, rivalise bien dans les milieux stables et peuplés, avec
une courbe de type I. La plupart des espèces mêlent les deux~; les
étiquettes nomment les bouts de l'échelle.
\end{definition}

\section{Fluctuation et régulation}

\begin{proposition}[Les cycles du prédateur et de la proie]\label{prop:b1:populations:cycles}
\index{cycle de population}
Certaines \omterm{def:b1:populations:population}{populations} ne se stabilisent pas mais oscillent. Le lièvre
d'Amérique de la forêt du Nord et le lynx qui le mange montent et
descendent ensemble avec une période d'environ dix ans, le lynx un an
ou deux derrière le lièvre~: beaucoup de lièvres nourrissent beaucoup de
lynx, beaucoup de lynx font baisser les lièvres, peu de lièvres
affament les lynx, peu de lynx laissent les lièvres se rétablir. Le
cycle est entraîné en partie par le prédateur et en partie par la
nourriture des lièvres eux-mêmes, dont le rétablissement après un
surbroutage prend des années~; les modèles couplant prédateur et proie
(\cref{ch:b1:species-interactions}) reproduisent de telles oscillations
à partir de deux équations.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Les registres de fourrures d'une compagnie de traite, tenus pendant
quatre-vingt-dix ans à partir des années 1840, donnent le nombre de
peaux de lynx et de lièvres rapportées chaque année à l'échelle d'un
sous-continent~: les deux séries montrent des pics tous les neuf à dix
ans, celui du lynx en retard sur celui du lièvre, sur neuf cycles
complets. Les études de terrain menées depuis, avec des lièvres
protégés des prédateurs par des clôtures ou nourris en supplément,
montrent que retirer l'un des deux facteurs amortit le cycle et que
retirer les deux l'abolit.
\end{proof}

\begin{example}[Notre propre espèce]\label{ex:b1:populations:humans}
La \omterm{def:b1:populations:population}{population} humaine a mis toute l'histoire à atteindre un milliard,
vers 1800, et deux siècles à en atteindre huit. Son taux de croissance
a culminé près de \qty{2}{\%} par an dans les années 1960 (doublement
en 35 ans) et est depuis tombé sous \qty{1}{\%}, non par les morts mais
par les naissances~: là où la mortalité infantile baisse et où les
femmes sont instruites, les familles se réduisent en une génération, et
la pyramide des âges passe du triangle à la colonne. L'arithmétique du
$R_0$ vaut pour nous~; ce qui a changé, c'est $m_x$.
\end{example}

\section{Problème~: des levures en fiole et des cerfs sur une île}

\begin{problem}[{Problème du week-end --- une culture comptée, un
troupeau tabulé, un lac échantillonné et un prélèvement fixé, jusqu'à
la capacité de charge et au taux intrinsèque de la levure}]
\label{pb:b1:populations:1}
Une culture de levure est comptée toutes les deux heures (\omterm{def:b1:cell-unit-of-life:cell}{cellules} par
millilitre, $\times 10^5$)~: $10, 29, 71, 175, 351, 513, 595, 641, 656$ à $t = 0,
2, \ldots, 16$ h. Une \omterm{def:b1:populations:population}{population} de cerfs sur une île a pour \omterm{def:b1:populations:lifetable}{table de
survie} $l_x = 1.00, 0.60, 0.52, 0.46, 0.40, 0.34, 0.26, 0.16, 0.06, 0$ et
$m_x = 0, 0, 0.6, 0.8, 0.8, 0.8, 0.7, 0.5, 0.3, 0$ pour $x = 0$ à $9$
ans. Dans un lac, 120 perches sont marquées et relâchées~; une semaine
plus tard, 150 sont capturées, dont 18 marquées.

\medskip
\noindent\textbf{Partie I --- La levure.}
\begin{enumerate}
  \item Estimer $r$ sur les deux premiers comptages, en supposant une
        \omterm{thm:b1:populations:exponential}{croissance exponentielle} pendant les deux premières heures.
  \item Calculer le temps de doublement.
  \item Estimer $K$ sur les derniers comptages.
  \item Calculer la prédiction logistique en $t = 4$, $8$ et \qty{12}{h}
        et la comparer aux comptages.
  \item À quel effectif la croissance est-elle la plus rapide, et quand
        est-il atteint~? Calculer la vitesse de croissance maximale en
        \omterm{def:b1:cell-unit-of-life:cell}{cellules} par millilitre et par heure.
  \item Que prédirait le modèle exponentiel à \qty{12}{h}~? De quel
        facteur se trompe-t-il~?
  \item Nommer deux choses qui pourraient fixer $K$ dans une fiole
        scellée de levure.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Les cerfs.}
\begin{enumerate}[resume]
  \item Calculer la mortalité $q_x$ la première année et la cinquième.
        De quel type de \omterm{def:b1:populations:lifetable}{survie} s'agit-il~?
  \item Calculer $l_x m_x$ pour chaque âge et en faire la somme~: $R_0$.
  \item Calculer $T$.
  \item Calculer $r$ et le temps de doublement du troupeau.
  \item L'île porte 200 cerfs aujourd'hui. Prédire l'effectif dans 10
        ans par la \omterm{thm:b1:populations:exponential}{croissance exponentielle}.
  \item L'île peut nourrir 600 cerfs. Prédire l'effectif dans 10 ans
        par la \omterm{thm:b1:populations:logistic}{croissance logistique}, et l'année où le troupeau
        dépasse 500.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Les perches et le prélèvement.}
\begin{enumerate}[resume]
  \item Estimer la \omterm{def:b1:populations:population}{population} de perches du lac.
  \item Si 20 des perches marquées ont perdu leur marque avant la
        seconde capture, refaire le calcul avec le vrai nombre de
        poissons marqués, et dire dans quel sens la première
        estimation se trompait.
  \item La \omterm{def:b1:populations:population}{population} de perches a $K = 1000$ et $r = \qty{0.4}{yr^{-1}}$.
        Calculer le rendement soutenable maximal et l'effectif auquel
        on l'obtient.
  \item Calculer le temps de doublement de la \omterm{def:b1:populations:population}{population} de perches
        quand elle est très inférieure à $K$.
  \item Le club décide plutôt de prélever chaque année \qty{20}{\%} du
        stock. Calculer la \omterm{def:b1:populations:population}{population} d'équilibre et la prise annuelle
        à cet équilibre.
  \item Un club de pêcheurs prend 120 perches par an. Montrer si la
        \omterm{def:b1:populations:population}{population} peut le soutenir, en calculant $\dd N/\dd t$ en $N =
        500$ et en $N = 300$.
  \item Pourquoi prélever en $K/2$ est-il risqué en pratique, si l'on
        songe au temps qu'il fait et à l'erreur de comptage de la
        question 14~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La régulation.}
\begin{enumerate}[resume]
  \item Le $m_x$ des cerfs tombe de moitié quand le troupeau dépasse
        400. Montrer que $R_0$ passe alors sous 1 --- qu'est-ce que cela
        fait au troupeau, et quel type de facteur est à l'œuvre~?
  \item Un hiver rude tue \qty{40}{\%} des cerfs quel que soit leur
        nombre. Ce facteur est-il régulateur~? Qu'advient-il du troupeau
        les années suivantes~?
  \item Des loups arrivent et prélèvent 30 cerfs par an, un nombre
        fixe. Calculer les effectifs d'équilibre du modèle logistique
        avec $K = 600$ et le $r$ de la question 11 (résoudre
        $rN(1 - N/K) = 30$), et dire quel équilibre est stable.
  \item Expliquer en deux phrases pourquoi le cycle du lynx est en
        retard sur celui du lièvre.
  \item Énoncer le résultat~: la \omterm{thm:b1:populations:logistic}{capacité de charge} et le taux
        intrinsèque de la levure, ainsi que $R_0$, $T$ et $r$ des cerfs.
\end{enumerate}
\end{problem}
