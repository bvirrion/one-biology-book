\chapter{Mensen en natuur}\label{ch:g5:people-and-nature}

Kijk vanaf een heuveltop omlaag het dal in: \omterm{prop:g5:people-and-nature:managed}{akkers} in nette vierkanten,
een beheerd bos, rechtgetrokken rivieroevers, wegen, daken. Bijna niets
van wat je ziet is vanzelf zo gegroeid. Mensen zijn de enige \omterm{def:g5:biodiversity:species}{soort} die
hele landschappen naar haar behoeften omvormt --- en dit hoofdstuk kijkt
eerlijk naar die macht: wat ze ons geeft, wat ze kost, en hoe je haar
goed kunt gebruiken.

\section{Landschappen gemaakt door mensen}

\begin{proposition}[De meeste landschappen worden beheerd]\label{prop:g5:people-and-nature:managed}
Bijna elk landschap om ons heen is door mensen gevormd:
\begin{itemize}
  \item \emph{akkers}\index{akker} zijn beheerde grond waar één gekozen
        \omterm{def:g1:plants-around-us:plant}{plant} --- tarwe, maïs, aardappelen --- groeit in plaats van het
        wilde mengsel;
  \item de meeste \emph{bossen}\index{bos (beheerd)} zijn geplant,
        gedund en geoogst als trage gewassen;
  \item \emph{weiden} bestaan doordat ze gemaaid of begraasd worden ---
        verlaten groeien ze weer dicht tot struikgewas en daarna tot bos;
  \item steden, wegen en dammen tekenen het land zelf opnieuw.
\end{itemize}
``Onaangeroerde'' hoekjes zijn de uitzondering, en de meeste worden met
opzet beschermd.
\end{proposition}
\admitted

\begin{example}[Het dal lezen]\label{ex:g5:people-and-nature:valley}
Dat uitzicht vanaf de heuveltop, punt voor punt: de tarweakker voedt de
bakkerij; de rijen sparren zijn een houtgewas dat ouder is dan hun
boswachter; de bloemrijke wei wordt elke juli gemaaid --- sla twee
zomers over en de bramen nemen haar in; de rivier werd lang geleden
rechtgetrokken om de \omterm{prop:g5:people-and-nature:managed}{akkers} tegen overstromingen te beschermen. Het dal
is een werkplaats --- een die al eeuwen draait.
\end{example}

\begin{remark}[Geen schurk en geen tuinkabouter]\label{rem:g5:people-and-nature:honest}
Het zou onjuist zijn dit verhaal als pure vernietiging te vertellen ---
de landbouw voedt de mensheid, en akkerland met heggen kan rijk zijn aan
\omterm{def:g5:biodiversity:species}{soorten} (\cref{ex:g5:biodiversity:orchard}) --- en even onjuist om te
doen alsof het omvormen niets kost
(\cref{prop:g5:biodiversity:loss} somde de rekeningen op). De eerlijke
vraag is nooit ``de natuur aanraken of niet?'' maar ``hoe beheren we wat
we aanraken zodat het blijft werken --- voor ons en voor de rest van het
web?''
\end{remark}

\section{Goed beheren en slecht beheren}

\begin{example}[Twee manieren om een heuvel te bewerken]\label{ex:g5:people-and-nature:twofarms}
Heuvel A: één reusachtige akker, heggen gerooid, de bodem de hele winter
kaal. Regen spoelt de losgemaakte grond de helling af; de helpers uit
\cref{ex:g5:biodiversity:orchard} hebben geen thuis; één plaag kan het
enige gewas ongehinderd wegvagen. Heuvel B: kleinere \omterm{prop:g5:people-and-nature:managed}{akkers} met de
heggen behouden, de bodem de winter door bedekt, gewassen die jaar na
jaar wisselen. De heggen houden de grond vast en huisvesten de
plaageters; de wisselteelt laat de grond rusten. Beide heuvels voeden
mensen --- één van beide doet dat over honderd jaar nog.
\end{example}

\begin{proposition}[De regels van duurzaam gebruik]\label{prop:g5:people-and-nature:lasting}
Gebruik van de natuur houdt stand als het drie regels respecteert:
\begin{enumerate}
  \item \emph{neem niet sneller dan het zich vernieuwt}: oogst hout in
        het groeitempo van het bos, vis in het voortplantingstempo van
        de school --- de overbevissing uit
        \cref{prop:g5:biodiversity:loss} is deze regel gebroken;
  \item \emph{houd de werkers}: bodemleven, bestuivers, plaageters ---
        de gratis diensten uit
        \cref{exo:g5:biodiversity:11} --- hebben hun thuis nodig;
  \item \emph{geef terug wat je leent}: de compostkringloop uit
        \cref{rem:g4:what-plants-need:soil}, op elke schaal.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[De rekensom van de boswachter]\label{ex:g5:people-and-nature:forester}
Een goed beheerd bos is het oudste bewijs dat regel 1 werkt:
boswachters tellen wat het bos elk jaar bijgroeit, en kappen niet meer
dan dat. Het bos levert dan elk jaar hout, voor altijd --- terwijl het
een bos blijft, met zijn uilen, herten en kevers. Sommige beheerde
\omterm{prop:g5:people-and-nature:managed}{bossen} ter wereld worden al vele honderden jaren op deze manier geoogst
zonder te krimpen.
\end{example}

\section{Steden, en weer ruimte maken}

\begin{example}[Natuur in de stad]\label{ex:g5:people-and-nature:city}
Een stad lijkt het tegendeel van natuur, en toch is ze ook een
\omterm{def:g2:where-animals-live:habitat}{leefgebied} --- een vreemd rotsachtig \omterm{def:g2:where-animals-live:habitat}{leefgebied}. Gierzwaluwen nestelen
op ``kliffen'' van beton, vossen patrouilleren langs de vuilnisbakken,
wilde \omterm{def:g1:plants-around-us:plant}{planten} splijten de voegen van het asfalt. En stadsbewoners kunnen
het welkom verbreden: parken, groene daken, balkonbloemen voor de bijen,
de schoolvijver uit \cref{ex:g2:where-animals-live:schoolpond}. De
methoden uit \cref{met:g5:biodiversity:help} werken ook tussen gebouwen.
\end{example}

\begin{example}[Een rivier haar bochten teruggeven]\label{ex:g5:people-and-nature:river}
De rechtgetrokken rivier van het dal stroomde snel en overstroomde
stroomafwaarts hevig; haar \omterm{prop:g3:sorting-living-things:groups}{vissen} en libellen waren met de bochten
verdwenen. Ingenieurs geven zulke rivieren nu soms hun bochten
\emph{terug}: herstelde krommingen en natte weiden vertragen het water,
bergen de overstromingen, en de ooievaars uit
\cref{ex:g5:biodiversity:storks} vinden hun weiden terug. Beheer kan ook
achteruit draaien --- herstellen is ook beheren.
\end{example}

\begin{method}[Een gebruik van de natuur beoordelen]\label{met:g5:people-and-nature:judge}
Bij elk gebruik van land of water --- een boerderij, een visserij, een
geplande weg, een tuin:
\begin{enumerate}
  \item vraag wat er \emph{genomen} wordt, en of dat sneller gaat dan
        het zich vernieuwt (regel 1);
  \item vraag welke \emph{thuizen en werkers} behouden blijven of
        verdwijnen (regel 2);
  \item vraag wat er \emph{teruggegeven} wordt --- en welk afval blijft
        liggen (regel 3);
  \item oordeel dan: zou dit honderd jaar door kunnen gaan? Welke ene
        verandering zou het antwoord het meest verbeteren?
\end{enumerate}
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:1}
Geef vier \omterm{def:g5:biodiversity:species}{soorten} door mensen gevormd landschap uit
\cref{prop:g5:people-and-nature:managed}, elk met datgene waarvoor het
beheerd wordt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:2}
Wat gebeurt er met een wei die niet meer gemaaid of begraasd wordt?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:3}
Noem de drie regels van duurzaam gebruik.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:4}
Noem drie verschillen tussen de twee heuvels uit
\cref{ex:g5:people-and-nature:twofarms}, en wat elk verschil verandert.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:5}
Leg de rekensom van de boswachter uit. Welke regel is het, en wat levert
het bos daarvoor terug --- en hoe lang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:6}
Noem drie wilde bewoners van steden, en welk kenmerk van de stad elk als
\omterm{def:g2:where-animals-live:habitat}{leefgebied} gebruikt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:7}
Wat herstelde het teruggeven van de bochten aan de rivier --- voor
mensen, en voor het web?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:8}
Probeer het: kies één uitzicht bij je in de buurt --- straat, akker,
park --- en noem alles wat mensen daar beheren, zoals in
\cref{ex:g5:people-and-nature:valley}. Blijft er iets onbeheerd?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:people-and-nature:9}
``De natuur beschermen betekent haar nooit aanraken.'' Bespreek dit met
de gemaaide wei uit \cref{prop:g5:people-and-nature:managed} --- een
\omterm{def:g2:where-animals-live:habitat}{leefgebied} dat zonder menselijke hand juist \emph{verdwijnt} --- en met
\cref{rem:g5:people-and-nature:honest}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:people-and-nature:10}
Laat \cref{met:g5:people-and-nature:judge} lopen over heuvel A uit
\cref{ex:g5:people-and-nature:twofarms}: welke regels breekt hij, en
welke ene verandering zou hem het meest verbeteren?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:people-and-nature:11}
Een vissersvloot verdubbelt vijf schitterende jaren lang haar vangst; in
het zesde jaar zijn de scholen verdwenen en ligt de vloot stil. Verklaar
de zes jaren met regel 1 --- waarom \emph{leek} het breken ervan in het
begin op succes? Vergelijk met de tegenovergestelde rekensom van de
boswachter.
\end{exercise}
