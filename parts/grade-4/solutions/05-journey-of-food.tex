\section*{Chapter \ref{ch:g4:journey-of-food} --- The Journey of Food}

\begin{solution}{exo:g4:journey-of-food:1}
Mouth, \omterm{def:g4:journey-of-food:tract}{esophagus}, \omterm{def:g4:journey-of-food:tract}{stomach}, \omterm{def:g4:journey-of-food:tract}{small intestine}, \omterm{def:g4:journey-of-food:tract}{large intestine} ---
then the leftovers leave the body.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:2}
It is churned by the \omterm{def:g4:journey-of-food:tract}{stomach}'s muscular walls and mixed with
\omterm{def:g4:journey-of-food:digestion}{digestive juices} --- for some hours.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:3}
\omterm{def:g4:journey-of-food:digestion}{Digestion} is the transformation of food, along the tract, into
components simple and small enough to enter the body. Those
components are the \omterm{def:g4:journey-of-food:digestion}{nutrients}.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:4}
In the \omterm{def:g4:journey-of-food:tract}{small intestine}, through its thin folded walls, into the
\omterm{prop:g4:journey-of-food:absorb}{blood}. What cannot cross travels on to the \omterm{def:g4:journey-of-food:tract}{large intestine},
loses its water there, and \omterm{def:g1:plants-around-us:parts}{leaves} the body as droppings.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:5}
\omterm{def:g4:journey-of-food:tract}{Saliva} is already dismantling the bread into simpler, sweeter
components --- \omterm{def:g4:journey-of-food:digestion}{digestion} begins in the mouth, and the sweetness
is its taste.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:6}
Because crossing into the \omterm{prop:g4:journey-of-food:absorb}{blood} takes wall surface: the metres
of length and the deep inner folds pack an enormous crossing
surface --- flattened out, roughly a small room's carpet ---
into the belly.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:7}
For example: chew properly (the \omterm{def:g4:journey-of-food:tract}{stomach} has no teeth); eat
calmly at regular times (the tract works best unhurried);
drink water (\omterm{def:g4:journey-of-food:digestion}{digestion} runs wet); eat fibre (it keeps
leftovers moving); rest after big meals (\omterm{def:g4:journey-of-food:tract}{stomach} and \omterm{def:g1:my-body:parts}{legs}
compete for the body's effort).
\end{solution}

\begin{solution}{exo:g4:journey-of-food:8}
Open answer. Typically well under half an hour in the mouth
--- against one to two days for the rest of the journey. The
mouth's share is tiny in time and decisive in effect: it is
the only stop with teeth.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:9}
Broken: chew properly (lunch was wolfed half-chewed) and rest
after meals (a sprint straight after). Tomorrow: eat slowly,
and let the meal settle before racing.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:10}
Swallowed food is still inside the tube --- like the hole of a
doughnut, not yet truly in the body. It must first be digested
into \omterm{def:g4:journey-of-food:digestion}{nutrients}; only in the \omterm{def:g4:journey-of-food:tract}{small intestine} do those cross the
wall into the \omterm{prop:g4:journey-of-food:absorb}{blood}. Entering the mouth is not entering the
body.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:11}
\omterm{def:g4:journey-of-food:digestion}{Digestion} must dismantle food into \omterm{def:g4:journey-of-food:digestion}{nutrients}, and grass is far
harder to dismantle than bread --- tough fibres, little
goodness per mouthful
(\cref{prop:g2:what-animals-eat:daily}). So the grass-eater
brings heavier machinery: grinding \omterm{def:g2:teeth-and-eating-well:kinds}{molars} used twice over
(chewing the cud), four \omterm{def:g4:journey-of-food:tract}{stomach} chambers for long processing,
and gut helpers that can break down what its own juices
cannot. Tougher input, bigger factory.
\end{solution}
