\chapter{Een eerste blik op cellen}\label{ch:g5:first-look-at-cells}

Kijk naar de rug van je hand: gladde \omterm{def:g1:the-five-senses:sense}{huid}, één naadloos oppervlak. Stel
je nu een vergrootglas voor dat sterk genoeg is om duizend keer in te
zoomen. Dat naadloze oppervlak zou uiteenvallen in een plaveisel van
piepkleine vakjes, tegen elkaar aan gelegd als stoeptegels --- en elk
\omterm{def:g1:living-or-not:living}{levend wezen} dat je in dit boek bent tegengekomen, van mos tot walvis,
is op dezelfde manier geplaveid. Het grootste geheim van de \omterm{def:g1:living-or-not:living}{biologie} is klein.

\section{Levende wezens bestaan uit cellen}

\begin{definition}[Cel]\label{def:g5:first-look-at-cells:cell}
Een \emph{cel}\index{cel} is het kleinste levende bouwsteentje van een
\omterm{def:g1:living-or-not:living}{levend wezen}. Bijna alle cellen zijn veel te klein om met het blote \omterm{def:g1:the-five-senses:sense}{oog}
te zien; een lichaam is eruit opgebouwd zoals een muur uit bakstenen.
Jouw eigen lichaam bestaat uit tienduizenden miljarden cellen.
\end{definition}

\begin{proposition}[Het celprincipe]\label{prop:g5:first-look-at-cells:principle}
Elk \omterm{def:g1:living-or-not:living}{levend wezen} bestaat uit cellen --- elk \omterm{def:g1:animals-around-us:animal}{dier}, elke \omterm{def:g1:plants-around-us:plant}{plant}, elke
paddenstoel, tot aan levende wezens die uit maar \emph{één} \omterm{def:g5:first-look-at-cells:cell}{cel} bestaan.
Die gedeelde bouw is een van de grote tekenen van de \emph{eenheid van
het leven}\index{eenheid van het leven}: hoe verschillend het madeliefje
en de walvis ook zijn, van dichtbij zijn ze uit dezelfde soort kleine
levende eenheden opgebouwd.
\end{proposition}
\admitted

\begin{example}[Hoe klein is klein?]\label{ex:g5:first-look-at-cells:small}
Een gewone \omterm{def:g5:first-look-at-cells:cell}{cel} van jouw lichaam meet ongeveer een vijftigste
millimeter. Zet er $50$ op een rij en ze beslaan één millimeter --- een
potloodstip zou de hele optocht bedekken. Daarom heeft niemand het
grootste deel van de menselijke geschiedenis iets van cellen vermoed: de
bouwstenen van het leven zijn kleiner dan het kleinste stipje dat het
\omterm{def:g1:the-five-senses:sense}{oog} kan zien.
\end{example}

\section{Het instrument dat ze onthulde}

\begin{definition}[Microscoop]\label{def:g5:first-look-at-cells:microscope}
Een \emph{microscoop}\index{microscoop} is een instrument waarvan de
lenzen heel kleine dingen honderden keren vergroten. Op dunne plakjes
levend materiaal gericht, onthulde hij wat geen \omterm{def:g1:the-five-senses:sense}{oog} ooit had gezien: de
cellen. De naam zelf werd bedacht door een vroege waarnemer die vond dat
de kleine vakjes in een plakje kurk op de kamertjes van monniken leken
--- \emph{cellen}.
\end{definition}

\begin{remark}[De donkere stip]\label{rem:g5:first-look-at-cells:nucleus}
In allebei de beelden laten de meeste cellen vanbinnen een kleiner,
donkerder stipje zien: het commandocentrum van de \omterm{def:g5:first-look-at-cells:cell}{cel}, dat de
\emph{celkern}\index{celkern} heet. Waarmee de celkern commandeert ---
de beroemde draad met instructies die erin opgerold ligt --- is een
verhaal voor de latere jaren van dit boek; onthoud voorlopig alleen dat
de kamertjes niet leeg zijn.
\end{remark}

\section{Van één cel tot een lichaam}

\begin{proposition}[Lichamen groeien via hun cellen]\label{prop:g5:first-look-at-cells:growth}
Een lichaam groeit, en herstelt zichzelf, doordat zijn cellen zich
kunnen \emph{delen}: één \omterm{def:g5:first-look-at-cells:cell}{cel} splitst in twee levende cellen, twee in
vier, enzovoort. Langer worden
(\cref{prop:g3:stages-of-life:animals}) is er onderhuids cellen die zich
vermenigvuldigen; een genezende schaafwond
(\cref{rem:g3:bones-and-muscles:alive}) is er cellen die opnieuw
opbouwen. En elk lichaam, ook het grootste, is begonnen als \emph{één
enkele \omterm{def:g5:first-look-at-cells:cell}{cel}} --- de bevruchte \omterm{def:g4:animal-reproduction:sexual}{eicel} uit
\cref{def:g4:animal-reproduction:sexual}.
\end{proposition}
\admitted

\begin{example}[Van één tot biljoenen]\label{ex:g5:first-look-at-cells:one}
De bevruchte \omterm{def:g4:animal-reproduction:sexual}{eicel} deelt zich in $2$ cellen, dan $4$, $8$,
$16$ \dots{} Herhaalde deling heeft in de loop van maanden en jaren de
tienduizenden miljarden cellen gebouwd die deze zin lezen. Elke \omterm{def:g5:first-look-at-cells:cell}{cel} van
jou stamt van die eerste af.
\end{example}

\begin{remark}[Hele levens in één cel]\label{rem:g5:first-look-at-cells:onecelled}
Sommige levende wezens komen nooit verder dan één \omterm{def:g5:first-look-at-cells:cell}{cel} --- en hebben ook
niet meer nodig. Een druppel vijverwater wemelt ervan: losse cellen die
zwemmen, eten en zich delen, hele levens op microscoopschaal. De
\omterm{def:g1:healthy-habits:germs}{ziektekiemen} uit \cref{def:g1:healthy-habits:germs} zijn meestal zulke
eencellige levende wezens --- oude bekenden, nu met een verklaring erbij.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:1}
Wat is een \omterm{def:g5:first-look-at-cells:cell}{cel}? Ruwweg hoeveel ervan vormen jouw lichaam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:2}
Waarom wist niemand het grootste deel van de geschiedenis iets van
cellen? Welk instrument veranderde dat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:3}
Beschrijf wat de \omterm{def:g5:first-look-at-cells:microscope}{microscoop} laat zien in uienvlies en in een
wangschraapsel. Wat hebben de twee beelden gemeen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:4}
Hoe heet het kleine donkere stipje dat in de meeste cellen te zien is?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:5}
Ongeveer hoeveel lichaamscellen op een rij beslaan één millimeter ---
en hoe lang is één \omterm{def:g5:first-look-at-cells:cell}{cel}, in millimeters?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:6}
Wat zijn de cellen onderhuids aan het doen als een kind langer wordt? En
als een geschaafde \omterm{def:g1:my-body:joint}{knie} geneest?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:7}
Als hoeveel cellen is elk mensenlichaam begonnen? Wat was die eerste
\omterm{def:g5:first-look-at-cells:cell}{cel}, volgens \cref{def:g4:animal-reproduction:sexual}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:8}
Een \omterm{def:g5:first-look-at-cells:cell}{cel} deelt zich in $2$; elk van die twee deelt zich weer, enzovoort.
Zet het verdubbelen vanaf $1$ voort: hoeveel cellen na $5$ delingen? En
na $10$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:first-look-at-cells:9}
Leg de zin uit: ``het madeliefje en de walvis laten de \omterm{prop:g5:first-look-at-cells:principle}{eenheid van het
leven} zien.'' Wat delen ze, volgens
\cref{prop:g5:first-look-at-cells:principle}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:first-look-at-cells:10}
Een druppel vijverwater bevat levende wezens van elk één enkele \omterm{def:g5:first-look-at-cells:cell}{cel}.
Toets ze aan de kenmerken van leven uit
\cref{prop:g1:living-or-not:signs}: welke kenmerken kan één \omterm{def:g5:first-look-at-cells:cell}{cel} vertonen?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:first-look-at-cells:11}
De cellen van een olifant zijn ongeveer even groot als die van een muis.
Wat maakt de olifant dan groter --- grotere cellen of meer cellen? Wat
zegt jouw antwoord over hoe groeien werkt, volgens
\cref{prop:g5:first-look-at-cells:growth}?
\end{exercise}
