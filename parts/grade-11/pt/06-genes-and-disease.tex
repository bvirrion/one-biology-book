\chapter{Variação genética e doença}\label{ch:g11:genes-and-disease}

Um casal com saúde perfeita tem um primeiro filho que, desde as
primeiras semanas, tosse, não ganha peso e tem um suor tão salgado que
um beijo na testa tem o gosto dele. O diagnóstico é fibrose cística: os
dois pais, sem saber, carregavam uma cópia não funcional do mesmo \omterm{def:g10:universal-dna:gene}{gene},
e a criança recebeu as duas. Uma em vinte e cinco pessoas de
ascendência europeia é portadora assim; os pais poderiam ter sido
avisados, antes da gravidez, por um teste numa gota de sangue. Este
capítulo trata das doenças escritas nos \omterm{def:g10:universal-dna:gene}{alelos}: como elas são herdadas,
como o risco é calculado a partir de uma árvore de família, como um
laboratório lê o \omterm{def:g10:universal-dna:gene}{gene}, e por que a maioria das doenças comuns é apenas
em parte genética.

\section{Doenças de um único gene}

\begin{definition}[Doença genética, dominante e recessivo]\label{def:g11:genes-and-disease:genetic}
Uma \emph{doença genética}\index{doença genética} é uma doença causada
por um ou mais \omterm{def:g10:universal-dna:gene}{alelos} mutantes. Numa doença de um único \omterm{def:g10:universal-dna:gene}{gene}, um \omterm{def:g10:universal-dna:gene}{alelo}
é \emph{recessivo}\index{recessivo} se a doença só aparece em pessoas
que carregam duas cópias dele --- uma pessoa com uma cópia é um
\emph{portador}\index{portador} saudável --- e
\emph{dominante}\index{dominante} se uma cópia basta. Por convenção, os
\omterm{def:g10:universal-dna:gene}{alelos} de um \omterm{def:g10:universal-dna:gene}{gene} são escritos com uma letra: $A$ para o \omterm{def:g10:universal-dna:gene}{alelo} comum,
$a$ para um mutante recessivo, de modo que um portador é $Aa$ e uma
pessoa afetada é $aa$.
\end{definition}

\begin{example}[A fibrose cística]\label{ex:g11:genes-and-disease:cf}
O \omterm{def:g10:universal-dna:gene}{gene} CFTR codifica uma \omterm{def:g11:gene-expression:protein}{proteína} que deixa os íons cloreto
atravessarem a membrana das \omterm{def:g10:cells-common-unit:cell}{células} que revestem as vias aéreas, o
intestino e as glândulas sudoríparas. O \omterm{def:g10:universal-dna:gene}{alelo} mutante mais comum não
tem três \omterm{def:g10:universal-dna:nucleotide}{nucleotídeos}, logo um \omterm{def:g10:chemistry-of-life:families}{aminoácido}; a \omterm{def:g11:gene-expression:protein}{proteína} se dobra errado e
é destruída. Sem ela o muco das vias aéreas fica espesso e pegajoso,
as infecções se instalam nos pulmões e os ductos do pâncreas entopem. A
doença é recessiva: cerca de um europeu em 25 é \omterm{def:g11:genes-and-disease:genetic}{portador} saudável, e um
recém-nascido em 2500 é afetado. A anemia falciforme
(\cref{ch:g11:enzymes-and-phenotype}) também é recessiva; a doença de
Huntington, uma degeneração do cérebro que começa por volta dos
quarenta anos, é \omterm{def:g11:genes-and-disease:genetic}{dominante} --- um \omterm{def:g10:universal-dna:gene}{alelo} mutante basta, e toda pessoa
afetada teve um genitor afetado.
\end{example}

\begin{proposition}[Transmissão de uma doença recessiva]\label{prop:g11:genes-and-disease:recessive}
Cada genitor passa um dos dois \omterm{def:g10:universal-dna:gene}{alelos} dele, escolhido ao acaso, a cada
filho. Dois \omterm{def:g11:genes-and-disease:genetic}{portadores} $Aa \times Aa$ têm, portanto, para cada filho,
uma chance de $1/4$ de uma criança afetada $aa$, $1/2$ de uma portadora
$Aa$ e $1/4$ de uma não portadora $AA$. Um \omterm{def:g11:genes-and-disease:genetic}{portador} e um não \omterm{def:g11:genes-and-disease:genetic}{portador}
$Aa \times AA$ não têm nenhum filho afetado e têm uma chance em duas de
um \omterm{def:g11:genes-and-disease:genetic}{portador}. Uma pessoa afetada e um não \omterm{def:g11:genes-and-disease:genetic}{portador} $aa \times AA$ só têm
filhos \omterm{def:g11:genes-and-disease:genetic}{portadores}.
\end{proposition}
\admitted

\section{Ler uma árvore de família}

\begin{definition}[Heredograma]\label{def:g11:genes-and-disease:pedigree}
Um \emph{heredograma}\index{heredograma} é um diagrama de uma família
ao longo de várias gerações: quadrados para os homens, círculos para as
mulheres, uma linha horizontal ligando um casal, uma linha vertical
descendo até os filhos, símbolos preenchidos para as pessoas afetadas.
As gerações são numeradas I, II, III de cima para baixo; os indivíduos
são numerados da esquerda para a direita.
\end{definition}

\begin{method}[Analisar um heredograma]\label{met:g11:genes-and-disease:analyse}
\begin{enumerate}
  \item \emph{\omterm{def:g11:genes-and-disease:genetic}{Dominante} ou recessiva?} Uma criança afetada de dois
        pais saudáveis quer dizer recessiva (os pais são \omterm{def:g11:genes-and-disease:genetic}{portadores}).
        Uma pessoa afetada com um genitor afetado em cada geração
        sugere \omterm{def:g11:genes-and-disease:genetic}{dominante}.
  \item \emph{Num \omterm{def:g11:cell-cycle-mitosis:chromosome}{cromossomo} sexual?} Se só os homens são afetados e
        os irmãos ou os pais das mães deles também, suspeite do
        \omterm{def:g11:cell-cycle-mitosis:chromosome}{cromossomo} X (mais adiante). Caso contrário o \omterm{def:g10:universal-dna:gene}{gene} está num
        \omterm{def:g11:cell-cycle-mitosis:chromosome}{cromossomo} comum.
  \item \emph{Atribua os \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipos}}: as pessoas afetadas numa doença
        recessiva são $aa$; os pais e os filhos delas carregam pelo
        menos um $a$; os irmãos saudáveis de uma pessoa afetada são
        $AA$ ou $Aa$.
  \item \emph{Calcule o risco} para um filho planejado combinando os
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipos} dos pais num quadro de Punnett; quando o \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipo} de
        um dos pais é incerto, pondere cada possibilidade pela
        probabilidade dela.
\end{enumerate}
\end{method}

\begin{example}[Os dois terços]\label{ex:g11:genes-and-disease:twothirds}
Por que um irmão saudável de uma criança afetada é \omterm{def:g11:genes-and-disease:genetic}{portador} com
probabilidade $2/3$ e não $1/2$? Os pais são $Aa \times Aa$; um filho é
$AA$, $Aa$, $aA$ ou $aa$ com chances iguais. Saber que ele é saudável
elimina $aa$: restam três casos, dos quais dois são \omterm{def:g11:genes-and-disease:genetic}{portadores}. Se ele
se casar com uma mulher da população geral (portadora com probabilidade
$1/25$), o risco de um primeiro filho afetado é
$2/3 \times 1/25 \times 1/4 = 1/150$.
\end{example}

\section{Genes do cromossomo X}

\begin{proposition}[Herança ligada ao X]\label{prop:g11:genes-and-disease:xlinked}
As mulheres carregam dois \omterm{def:g10:universal-dna:information}{cromossomos} X, os homens um X e um Y. Um
\omterm{def:g10:universal-dna:gene}{alelo} \omterm{def:g11:genes-and-disease:genetic}{recessivo} do \omterm{def:g11:cell-cycle-mitosis:chromosome}{cromossomo} X é mascarado numa mulher pelo segundo X
dela, mas um homem, que não tem um segundo X, o expressa. Uma doença
assim afeta, portanto, sobretudo os homens; as filhas de um homem
afetado são todas portadoras; uma mãe portadora passa o \omterm{def:g10:universal-dna:gene}{alelo} a metade
dos filhos homens, que são afetados, e a metade das filhas, que são
portadoras. A hemofilia, em que o sangue não consegue coagular, é o
exemplo clássico; o daltonismo para o vermelho e o verde
(\cref{ch:g11:the-eye}) é outro.
\end{proposition}
\admitted

\section{Ler o próprio gene}

\begin{proposition}[Testes genéticos]\label{prop:g11:genes-and-disease:testing}
Os \omterm{def:g10:universal-dna:gene}{alelos} que uma pessoa carrega podem ser lidos diretamente no \omterm{def:g10:universal-dna:information}{DNA}
dela, extraído de algumas \omterm{def:g10:cells-common-unit:cell}{células} (sangue, saliva e, para um feto, uma
amostra da placenta ou do líquido que o cerca). A região do \omterm{def:g10:universal-dna:gene}{gene} é
copiada milhões de vezes por uma reação que imita a replicação
(\cref{ch:g11:dna-replication}) num tubo de ensaio, e depois é
sequenciada ou cortada por \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimas} que reconhecem sequências
específicas e separada por tamanho num gel: um \omterm{def:g10:universal-dna:gene}{alelo} mutante que ganhou,
perdeu ou trocou um sítio de corte dá fragmentos de comprimentos
diferentes, visíveis como bandas. Testes assim identificam \omterm{def:g11:genes-and-disease:genetic}{portadores}
antes de qualquer criança ser concebida, diagnosticam um feto e
detectam a \omterm{def:g11:mutations:mutation}{mutação} num recém-nascido antes de os sintomas aparecerem.
\end{proposition}
\admitted

\begin{example}[O que um teste pode e não pode dizer]\label{ex:g11:genes-and-disease:testsays}
Um teste de \omterm{def:g11:genes-and-disease:genetic}{portador} para a fibrose cística procura os cinquenta \omterm{def:g10:universal-dna:gene}{alelos}
mutantes mais comuns e encontra cerca de 90\% dos \omterm{def:g11:genes-and-disease:genetic}{portadores}: um
resultado negativo reduz a chance de uma pessoa ser portadora de $1/25$
para cerca de $1/250$, mas não a zera. Um diagnóstico pré-natal num
feto de dois \omterm{def:g11:genes-and-disease:genetic}{portadores} conhecidos é quase certo, e coloca os pais
diante de uma decisão que a biologia não toma por eles. Um teste para a
doença de Huntington diz a uma pessoa saudável de trinta anos se a
doença virá; um terço das que têm direito a ele preferem não saber.
\end{example}

\section{Doenças de muitos genes e de um modo de vida}

\begin{proposition}[Doenças multifatoriais]\label{prop:g11:genes-and-disease:multifactorial}
A maioria das doenças comuns --- o diabetes tipo 2, as doenças do
coração e das artérias, a maior parte dos cânceres, a asma --- é
\emph{multifatorial}\index{doença multifatorial}: dezenas ou centenas
de \omterm{def:g10:universal-dna:gene}{alelos} deslocam o risco um pouco cada um, e o ambiente
(alimentação, atividade, tabagismo, infecções) o desloca muito. Uma
doença assim corre nas famílias sem seguir as regras de um único \omterm{def:g10:universal-dna:gene}{gene},
e uma predisposição genética é um risco, não uma sentença: os mesmos
\omterm{def:g10:universal-dna:gene}{alelos} produzem a doença num modo de vida e não noutro.
\end{proposition}
\begin{proof}[Evidências]
Gêmeos idênticos compartilham todos os \omterm{def:g10:universal-dna:gene}{alelos}; quando um tem diabetes
tipo 2, o outro tem em cerca de 70\% dos casos --- bem acima dos 8\% da
população, de modo que os \omterm{def:g10:universal-dna:gene}{genes} importam --- mas não em 100\%, de modo
que os \omterm{def:g10:universal-dna:gene}{genes} não decidem. Populações que passaram numa geração de uma
alimentação rural a uma urbana viram a doença subir várias vezes com os
mesmos \omterm{def:g10:universal-dna:gene}{genes}. Entre as pessoas que carregam muitos \omterm{def:g10:universal-dna:gene}{alelos} de risco, as
que se mantêm ativas e magras desenvolvem a doença a uma fração da taxa
das que não se mantêm.
\end{proof}

\begin{remark}[O que quer dizer ``genética'', então]\label{rem:g11:genes-and-disease:meaning}
A fibrose cística é genética no sentido estrito: o \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipo} determina a
doença, seja qual for o ambiente, e o ambiente só muda o modo como ela
é vivida. O diabetes tipo 2 é genético num sentido mais fraco: o
\omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipo} fixa uma suscetibilidade que o ambiente realiza ou não. Entre
os dois estão a fenilcetonúria, em que uma dieta cancela o efeito do
\omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipo}, e o \omterm{def:g11:genes-and-disease:genetic}{portador} de anemia falciforme, cujo \omterm{def:g10:universal-dna:gene}{alelo} é uma doença
num lugar e uma proteção em outro. ``Isso é genético?'' raramente é uma
pergunta de sim ou não; ``quanto, e em que condições?'' é a pergunta a
fazer.
\end{remark}

\section{Problema: uma família, um gene}

\begin{problem}[{Problema de fim de semana --- uma família com fibrose
cística acompanhada por três gerações: genótipos atribuídos, riscos
calculados para cada filho planejado, um teste de laboratório lido, e a
aritmética de uma população de portadores}]
\label{pb:g11:genes-and-disease:1}
Léa e Tom, os dois saudáveis, têm três filhos: Max (saudável), Zoé
(fibrose cística) e Lou (saudável). Ana, irmã de Tom, é saudável e
casada com Karim, cuja família não tem história da doença; eles estão
esperando um filho. Na população, uma pessoa em 25 é portadora.

\medskip
\noindent\textbf{Parte I --- Os \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipos}.}
\begin{enumerate}
  \item A doença é \omterm{def:g11:genes-and-disease:genetic}{dominante} ou recessiva? Justifique a partir da
        família.
  \item Dê os \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipos} de Léa, de Tom e de Zoé.
  \item Quais são os \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipos} possíveis de Max, com que
        probabilidades?
  \item Qual é a probabilidade de Ana ser portadora? (Considere os pais
        dela.)
  \item Qual é a probabilidade de Karim ser \omterm{def:g11:genes-and-disease:genetic}{portador}?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Os riscos.}
\begin{enumerate}[resume]
  \item Calcule a probabilidade de o filho de Ana e Karim ser afetado.
  \item Léa e Tom cogitam um quarto filho. Probabilidade de ele ser
        afetado? E de ser \omterm{def:g11:genes-and-disease:genetic}{portador}?
  \item Max, adulto, se casa com uma mulher da população geral.
        Probabilidade de um primeiro filho afetado?
  \item Zoé, adulta, se casa com um homem da população geral.
        Probabilidade de um filho afetado? E de um filho \omterm{def:g11:genes-and-disease:genetic}{portador}?
  \item Lou se casa com um primo de primeiro grau, filho de Ana e
        Karim. Calcule a probabilidade de um filho afetado. Por que ela
        é maior que para Max?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O teste.}
O \omterm{def:g10:universal-dna:gene}{alelo} mutante carregado nesta família acrescenta um sítio de corte: o
\omterm{def:g10:universal-dna:gene}{alelo} comum dá um fragmento de 900 pares de bases, o mutante dá um de
600 e um de 300.
\begin{enumerate}[resume]
  \item Desenhe, ou descreva, as bandas esperadas para Zoé, para Tom e
        para um não \omterm{def:g11:genes-and-disease:genetic}{portador}.
  \item Max é testado: uma banda em 900. Qual é o \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipo} dele agora,
        e o novo risco para o primeiro filho dele (questão 8)?
  \item Ana é testada: bandas em 900, 600 e 300. Recalcule o risco para
        o filho dela.
  \item O teste de Karim não encontra nenhum dos cinquenta \omterm{def:g10:universal-dna:gene}{alelos}
        mutantes mais comuns, que cobrem 90\% dos \omterm{def:g11:genes-and-disease:genetic}{portadores}. Calcule a
        probabilidade restante de ele ser \omterm{def:g11:genes-and-disease:genetic}{portador} e recalcule o risco.
  \item Um teste pré-natal do feto de Ana e Karim mostra bandas em 900,
        600 e 300. Qual é o \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipo} e qual é o \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenótipo} do feto?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A população.}
\begin{enumerate}[resume]
  \item Se uma pessoa em 25 é portadora, qual é a probabilidade de os
        dois membros de um casal ao acaso serem \omterm{def:g11:genes-and-disease:genetic}{portadores} e, portanto,
        a frequência de recém-nascidos afetados?
  \item Compare com o um em 2500 observado. Comente.
  \item Quase todas as pessoas afetadas morriam na infância. Explique
        por que, apesar disso, o \omterm{def:g10:universal-dna:gene}{alelo} mutante se manteve numa
        frequência de uma em 50 cópias, usando o número de cópias
        carregadas pelos \omterm{def:g11:genes-and-disease:genetic}{portadores} saudáveis contra o das carregadas
        pelas pessoas afetadas.
  \item Os tratamentos agora permitem que a maioria das pessoas
        afetadas chegue à vida adulta e tenha filhos. Preveja o efeito
        sobre a frequência do \omterm{def:g10:universal-dna:gene}{alelo} nas gerações vindouras, e o tamanho
        desse efeito.
  \item Enuncie o resultado: o um em quatro de dois \omterm{def:g11:genes-and-disease:genetic}{portadores}, o dois
        em três de um irmão saudável, e o que o gel acrescentou à
        aritmética desta família.
\end{enumerate}
\end{problem}
