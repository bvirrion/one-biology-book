\chapter{Muscles et articulations}\label{ch:g10:muscles-and-joints}

Un footballeur plante son pied, pivote, et s'écroule en se tenant le
genou. Le diagnostic, une heure plus tard~: un \omterm{def:g10:muscles-and-joints:joint}{ligament} rompu. Aucun os
n'est cassé, aucun muscle n'a lâché~; une sangle de tissu de quelques
centimètres, chargée de tenir deux os alignés, a cédé sous une torsion
pour laquelle elle n'avait jamais été bâtie. Pour comprendre ce qui
s'est rompu, et pourquoi le genou mettra des mois à se remettre, il
faut savoir de quoi une \omterm{def:g10:muscles-and-joints:joint}{articulation} est faite, comment les muscles
tirent dessus, et quelles forces la traversent quand un corps de
soixante-dix kilogrammes retombe sur une seule jambe.

\section{L'articulation}

\begin{definition}[Articulation]\label{def:g10:muscles-and-joints:joint}
Une \emph{articulation}\index{articulation} est le lieu où deux os se
rencontrent. Dans une articulation mobile comme le genou, la hanche,
l'épaule ou le coude, les extrémités des os sont revêtues d'une couche
de \emph{cartilage}\index{cartilage} lisse de quelques millimètres
d'épaisseur, et enfermées dans un sac étanche, la \emph{capsule}, dont
la doublure sécrète le \emph{liquide synovial} qui lubrifie les
surfaces. Des bandes de tissu résistant, les
\emph{ligaments}\index{ligament}, vont d'un os à l'autre en travers de
l'articulation et limitent ses mouvements à ceux qui sont prévus. Dans
le genou, deux croissants de cartilage, les \emph{ménisques}, sont
placés entre les os et répartissent la charge.
\end{definition}

\begin{proposition}[À quoi sert chaque partie]\label{prop:g10:muscles-and-joints:parts}
Le \omterm{def:g10:muscles-and-joints:joint}{cartilage} et le liquide synovial font glisser les surfaces avec
moins de frottement que la glace sur la glace~; les ménisques
répartissent une charge qui se concentrerait sinon sur quelques
millimètres carrés~; les \omterm{def:g10:muscles-and-joints:joint}{ligaments} laissent le genou se plier et se
tendre mais résistent à la torsion et à la flexion latérale~; la
capsule retient le liquide. Chaque partie a sa défaillance propre~: le
\omterm{def:g10:muscles-and-joints:joint}{cartilage} s'use (l'arthrose), un ménisque se déchire sous une torsion
genou plié, un \omterm{def:g10:muscles-and-joints:joint}{ligament} s'entorse ou se rompt quand l'\omterm{def:g10:muscles-and-joints:joint}{articulation} est
forcée au-delà de son amplitude.
\end{proposition}
\admitted

\begin{example}[Entorse, déchirure, luxation]\label{ex:g10:muscles-and-joints:injuries}
Tordez une cheville sur un trottoir~: les \omterm{def:g10:muscles-and-joints:joint}{ligaments} externes sont
étirés au-delà de leur limite --- c'est une \emph{entorse}~;
l'\omterm{def:g10:muscles-and-joints:joint}{articulation} enfle car la capsule saigne et du liquide s'accumule.
Tirez fort sur des ischio-jambiers froids~: une partie de leurs fibres
se rompt --- c'est une \emph{déchirure} musculaire. Tombez sur un bras
tendu~: la tête de l'humérus peut quitter sa cavité --- c'est une
\emph{luxation}, les \omterm{def:g10:muscles-and-joints:joint}{ligaments} et la capsule étant étirés ou déchirés
au passage. Aucune n'est une fracture~; toutes peuvent mettre plus
longtemps à guérir qu'une fracture.
\end{example}

\section{Les muscles tirent, par paires}

\begin{definition}[Muscle squelettique et tendon]\label{def:g10:muscles-and-joints:muscle}
Un \emph{muscle squelettique}\index{muscle squelettique} est un organe
fait de milliers de \omterm{def:g10:cells-common-unit:cell}{cellules} allongées, les \emph{fibres musculaires},
réunies en faisceaux~; chaque fibre se raccourcit quand un nerf la
stimule. À ses deux extrémités, le muscle se resserre en un
\emph{tendon}\index{tendon}, cordon de tissu non contractile fixé à
l'os. Un muscle ne peut que \emph{tirer}, en se raccourcissant, et
seulement selon sa propre longueur~: il meut une \omterm{def:g10:muscles-and-joints:joint}{articulation} en tirant
sur l'os situé au-delà.
\end{definition}

\begin{proposition}[Les paires antagonistes]\label{prop:g10:muscles-and-joints:antagonists}
Puisqu'un muscle ne peut pas pousser, tout mouvement d'une \omterm{def:g10:muscles-and-joints:joint}{articulation}
exige deux muscles ou deux groupes travaillant en opposition~: un
\emph{fléchisseur} qui plie l'\omterm{def:g10:muscles-and-joints:joint}{articulation} et un \emph{extenseur} qui
la tend. Quand l'un se contracte, l'autre se relâche et s'étire~; pour
tenir une position, les deux se contractent ensemble. Le biceps
fléchit le coude et le triceps l'étend~; le quadriceps, à l'avant de la
cuisse, étend le genou, et les ischio-jambiers, derrière, le
fléchissent.
\end{proposition}
\admitted

\begin{proposition}[Leviers et forces]\label{prop:g10:muscles-and-joints:lever}
Un os qui pivote autour d'une \omterm{def:g10:muscles-and-joints:joint}{articulation} est un levier. Le \omterm{def:g10:muscles-and-joints:muscle}{tendon} du
muscle s'attache à une courte distance $d$ du pivot, tandis que la
charge agit à une distance $D$ plus grande~; à l'équilibre, le muscle
doit tirer avec une force
\[
F_{\text{muscle}} = F_{\text{charge}} \times \frac{D}{d},
\]
soit plusieurs fois la charge. Ce montage échange de la force contre de
la vitesse et de l'amplitude~: un faible raccourcissement du muscle
produit un mouvement grand et rapide de la main ou du pied, au prix de
forces élevées dans le \omterm{def:g10:muscles-and-joints:muscle}{tendon} et dans l'\omterm{def:g10:muscles-and-joints:joint}{articulation}.
\end{proposition}
\admitted

\begin{example}[Porter un sac]\label{ex:g10:muscles-and-joints:bag}
Un sac de \qty{5}{kg} (poids \qty{49}{N}) pend à la main, à
\qty{35}{cm} du coude~; le \omterm{def:g10:muscles-and-joints:muscle}{tendon} du biceps s'attache à \qty{5}{cm} de
celui-ci. Le biceps tire avec $49 \times 35/5 = \qty{343}{N}$ --- sept
fois la charge, le poids de \qty{35}{kg} --- et l'\omterm{def:g10:muscles-and-joints:joint}{articulation} du coude
est comprimée avec environ \qty{300}{N}. Les muscles, les \omterm{def:g10:muscles-and-joints:muscle}{tendons} et
le \omterm{def:g10:muscles-and-joints:joint}{cartilage} sont bâtis pour ces forces~; ce qui les abîme, c'est une
force brusque dans la mauvaise direction, ou une charge répétée au-delà
de ce que le tissu sait réparer.
\end{example}

\section{À l'intérieur du muscle}

\begin{proposition}[Structure d'une fibre musculaire]\label{prop:g10:muscles-and-joints:fibre}
Une fibre musculaire est une \omterm{def:g10:cells-common-unit:cell}{cellule} géante unique, large de
\num{10}--\qty{100}{\micro m} et longue de plusieurs centimètres, avec
de nombreux \omterm{def:g10:cells-common-unit:organelle}{noyaux}. Elle est bourrée de \emph{myofibrilles} parallèles,
chacune faite d'une chaîne d'unités répétées où deux sortes de
filaments protéiques, épais et fins, s'interpénètrent. La contraction
est le glissement des filaments fins entre les épais, mû par l'ATP~:
chaque unité se raccourcit d'une fraction, la myofibrille de leur
somme, et la fibre avec elle. Les \omterm{def:g10:cells-common-unit:organelle}{mitochondries} logées entre les
myofibrilles fournissent l'ATP~; une terminaison nerveuse posée sur
chaque fibre en donne l'ordre.
\end{proposition}
\admitted

\begin{definition}[Types de fibres]\label{def:g10:muscles-and-joints:types}
Les \omterm{def:g10:muscles-and-joints:muscle}{fibres musculaires} existent en deux grands types. Les
\emph{fibres lentes} se contractent lentement et faiblement mais sans
se fatiguer~: riches en \omterm{def:g10:cells-common-unit:organelle}{mitochondries} et en capillaires, rouges grâce à
une \omterm{def:g10:chemistry-of-life:families}{protéine} qui stocke l'oxygène, elles tournent sur la respiration et
dominent dans les muscles posturaux et chez les athlètes d'endurance.
Les \emph{fibres rapides} se contractent vite et fort mais se fatiguent
en moins d'une minute~: plus pauvres en \omterm{def:g10:cells-common-unit:organelle}{mitochondries}, plus pâles,
elles reposent sur la réserve phosphatée et la \omterm{def:g10:cell-metabolism:fermentation}{fermentation} et dominent
chez les sprinteurs. La proportion dans un muscle donné est largement
héritée~; l'entraînement change la taille et l'équipement des fibres
plus que leur nombre.
\end{definition}

\begin{example}[Lire une biopsie musculaire]\label{ex:g10:muscles-and-joints:biopsy}
Une coloration des \omterm{def:g10:cell-metabolism:metabolism}{enzymes} mitochondriales sur une coupe fine de muscle
de cuisse montre une mosaïque~: des fibres sombres (lentes) et des
fibres pâles (rapides). L'échantillon d'un marathonien est sombre à
80~\%~; celui d'un sprinteur à 30~\%~; celui d'une personne non
entraînée à moitié. La même coloration chez le même coureur après un an
d'entraînement au sprint montre les fibres pâles grossies, mais pas
plus nombreuses.
\end{example}

\section{Fatigue, dommage, réparation}

\begin{proposition}[Les limites du muscle]\label{prop:g10:muscles-and-joints:limits}
Un muscle défaille de trois façons distinctes. La \emph{fatigue}~: la
force décline quand l'apport d'ATP ne suit plus --- en moins d'une
minute dans les fibres rapides, en quelques heures dans les lentes ---
et se récupère au repos. La \emph{crampe}~: une contraction
involontaire, brusque et durable, souvent due à la déshydratation ou à
une perte de sel. La \emph{lésion}~: des fibres rompues par une force
supérieure à leur résistance, typiquement quand le muscle est froid,
fatigué, ou étiré alors qu'il se contracte. Les fibres se réparent en
quelques semaines à partir de \omterm{def:g10:cells-common-unit:cell}{cellules} de réserve du muscle~; un \omterm{def:g10:muscles-and-joints:muscle}{tendon}
ou un \omterm{def:g10:muscles-and-joints:joint}{ligament}, mal irrigué, se répare en quelques mois.
\end{proposition}
\admitted

\begin{method}[Analyser une lésion de l'appareil locomoteur]\label{met:g10:muscles-and-joints:analyse}
\begin{enumerate}
  \item Repérer le mouvement qui l'a causée et sa direction~:
        l'\omterm{def:g10:muscles-and-joints:joint}{articulation} a-t-elle été mue dans son amplitude (problème
        musculaire) ou au-delà (problème de \omterm{def:g10:muscles-and-joints:joint}{ligament} ou de capsule)~?
  \item Repérer le tissu~: les \omterm{def:g10:muscles-and-joints:muscle}{fibres musculaires} se déchirent
        (douleur dans le corps du muscle, faiblesse)~; les \omterm{def:g10:muscles-and-joints:joint}{ligaments}
        s'entorsent (douleur et gonflement de l'\omterm{def:g10:muscles-and-joints:joint}{articulation},
        instabilité)~; les \omterm{def:g10:muscles-and-joints:muscle}{tendons} s'enflamment sous une charge
        répétée~; le \omterm{def:g10:muscles-and-joints:joint}{cartilage} s'use en silence pendant des années.
  \item Estimer la force~: utiliser la règle du levier~; se rappeler
        qu'une réception ou un freinage multiplie plusieurs fois le
        poids du corps.
  \item Prévoir la guérison d'après l'irrigation du tissu~: le muscle
        en quelques semaines, le \omterm{def:g10:muscles-and-joints:joint}{ligament} et le \omterm{def:g10:muscles-and-joints:muscle}{tendon} en quelques
        mois, le \omterm{def:g10:muscles-and-joints:joint}{cartilage} presque pas.
\end{enumerate}
\end{method}

\begin{remark}[Pourquoi l'échauffement fonctionne]\label{rem:g10:muscles-and-joints:warmup}
Dix minutes de mouvement doux élèvent la température des muscles d'un
ou deux degrés, ce qui rend leurs \omterm{def:g10:chemistry-of-life:families}{protéines} plus extensibles et leurs
\omterm{def:g10:cell-metabolism:metabolism}{enzymes} plus rapides~; cela ouvre les capillaires qui livreront
l'oxygène du \cref{ch:g10:heart-lungs-effort}, et cela épaissit le film
de liquide synovial sur le \omterm{def:g10:muscles-and-joints:joint}{cartilage}. Un muscle froid à qui l'on
demande une contraction maximale est la cause la plus fréquente d'une
déchirure~; une \omterm{def:g10:muscles-and-joints:joint}{articulation} chargée avant que son liquide se soit
réparti s'use plus vite.
\end{remark}

\section{Problème~: les forces dans un genou}

\begin{problem}[{Problème du week-end --- le genou d'un footballeur
démonté~: le levier de la jambe, la traction du muscle de la cuisse, le
ligament qui a rompu, et les nombres qu'un tendon porte à chaque pas}]
\label{pb:g10:muscles-and-joints:1}
Karim, \qty{70}{kg}, s'est blessé au genou en pivotant brusquement sur
un pied planté. On prendra $g = \qty{9.8}{N/kg}$. Le \omterm{def:g10:muscles-and-joints:muscle}{tendon} du
quadriceps s'attache au tibia à \qty{5}{cm} du pivot du genou~; la
poussée du sol sur le pied s'exerce à environ \qty{20}{cm} de ce pivot
quand le genou est bien plié, comme à la réception d'un saut.

\medskip
\noindent\textbf{Partie I --- Debout et en marchant.}
\begin{enumerate}
  \item Calculer le poids de Karim.
  \item Debout sur une jambe, genou légèrement plié, le sol pousse sur
        le pied avec tout son poids, à \qty{10}{cm} du pivot du genou.
        Calculer la force dans le \omterm{def:g10:muscles-and-joints:muscle}{tendon} du quadriceps.
  \item En montant une marche, le genou est plus plié et la force du
        sol s'exerce à \qty{20}{cm} du pivot. Calculer de nouveau la
        force dans le \omterm{def:g10:muscles-and-joints:muscle}{tendon}.
  \item Exprimer les deux résultats en multiples du poids du corps, et
        expliquer pourquoi les escaliers fatiguent les cuisses.
  \item Quel groupe musculaire est l'antagoniste du quadriceps, et que
        fait-il pendant la montée~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La réception.}
\begin{enumerate}[resume]
  \item À la réception d'une tête, le sol pousse avec trois fois le
        poids de Karim, à \qty{20}{cm} du pivot. Calculer la force dans
        le \omterm{def:g10:muscles-and-joints:muscle}{tendon}.
  \item La section du \omterm{def:g10:muscles-and-joints:muscle}{tendon} vaut \qty{1.2}{cm^2}. Calculer la
        contrainte qui s'y exerce, en newtons par millimètre carré, et
        la comparer à la contrainte de rupture d'environ
        \qty{100}{N/mm^2}.
  \item Les surfaces articulaires sont comprimées l'une contre l'autre
        par environ la force du \omterm{def:g10:muscles-and-joints:muscle}{tendon} plus la force du sol. Calculer
        cette charge, puis la pression sur le \omterm{def:g10:muscles-and-joints:joint}{cartilage} si l'aire de
        contact vaut \qty{6}{cm^2} avec les ménisques et
        \qty{1}{cm^2} sans eux.
  \item En déduire pourquoi un genou privé d'un ménisque use son
        \omterm{def:g10:muscles-and-joints:joint}{cartilage} plus vite.
  \item Expliquer ce qu'apporte le liquide synovial pendant la
        réception.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La blessure.}
\begin{enumerate}[resume]
  \item Le pied de Karim était planté et son corps a pivoté~: quelle
        direction de mouvement a été imposée au genou, et quel tissu
        est bâti pour y résister~?
  \item Pourquoi le quadriceps, capable de tirer avec plusieurs
        kilonewtons, n'a-t-il pas protégé le \omterm{def:g10:muscles-and-joints:joint}{ligament}~?
  \item Le genou a enflé en une heure. Qu'est-ce qui l'a rempli, et
        quelle partie de l'\omterm{def:g10:muscles-and-joints:joint}{articulation} a été atteinte~?
  \item Une imagerie montre le \omterm{def:g10:muscles-and-joints:joint}{ligament} rompu et un ménisque déchiré.
        À l'aide de la \cref{met:g10:muscles-and-joints:analyse},
        prévoir le temps de guérison de chacun et le justifier par leur
        irrigation.
  \item Karim demande pourquoi le chirurgien veut renforcer ses
        ischio-jambiers avant l'opération. Répondre avec la
        \cref{prop:g10:muscles-and-joints:antagonists}.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Chaque pas compte.}
\begin{enumerate}[resume]
  \item Karim fait environ 8000 pas par jour~; la moitié sur la jambe
        blessée. À l'aide de la force de la question 2, combien de fois
        par jour le \omterm{def:g10:muscles-and-joints:muscle}{tendon} porte-t-il au moins cette charge~?
  \item Sur une année, quel est le nombre total de mises en charge~?
        Expliquer pourquoi un \omterm{def:g10:muscles-and-joints:muscle}{tendon} enflammé par le surmenage (une
        tendinite) a besoin de repos plutôt que de force.
  \item Au cours d'un match, il retombe de saut quelque 40 fois.
        Comparer la charge totale portée aux réceptions (question 6) à
        celle portée par les pas de la journée.
  \item La course double la force du sol par rapport à la marche.
        Pourquoi les médecins recommandent-ils le vélo et la natation
        plutôt que la course pour un genou en cours de guérison~?
  \item Énoncer le résultat~: le multiple du poids du corps que porte
        le \omterm{def:g10:muscles-and-joints:muscle}{tendon} du quadriceps dans un escalier et à la réception d'un
        saut, et le seul tissu du genou dont ces nombres n'ont jamais
        éprouvé la résistance --- jusqu'au pivot.
\end{enumerate}
\end{problem}
