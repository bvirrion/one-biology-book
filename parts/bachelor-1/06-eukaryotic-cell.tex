\chapter{Functional Organization of the Eukaryotic Cell}\label{ch:b1:eukaryotic-cell}

A \omterm{def:b1:cell-unit-of-life:cell}{cell} of the pancreas makes, every day, about its own mass of digestive
enzymes and pours them into a duct. The enzymes are proteins that would
digest the \omterm{def:b1:cell-unit-of-life:cell}{cell} that made them; so the \omterm{def:b1:cell-unit-of-life:cell}{cell} builds them inside a sealed
sheet of membrane, ships them through a second compartment where they
are sorted and concentrated, stores them in granules, and releases them
only at the signal of a meal. Every step happens in a different
membrane-bounded space, and the traffic between the spaces is carried by
vesicles. This chapter describes the compartments of the \omterm{def:b1:cell-unit-of-life:prokeuk}{eukaryotic
cell}, the \omterm{def:b1:eukaryotic-cell:endomembrane}{endomembrane system} through which proteins flow, the
\omterm{def:b1:cell-unit-of-life:prokeuk}{organelles} that transform energy, the skeleton that holds the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s
shape and moves its parts, and the features that distinguish a plant
\omterm{def:b1:cell-unit-of-life:cell}{cell} from an animal one.

\section{Compartments}

\begin{definition}[Organelle, cytosol]\label{def:b1:eukaryotic-cell:organelle}
An \emph{organelle}\index{organelle} is a structure inside a \omterm{def:b1:cell-unit-of-life:cell}{cell} with
a definite composition and function; the membrane-bounded \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles}
are compartments whose interior differs in composition from the
\emph{cytosol}\index{cytosol}, the aqueous solution of the \omterm{def:b1:cell-unit-of-life:cell}{cytoplasm}
outside them. Membrane-bounded compartments: the \omterm{def:b1:eukaryotic-cell:nucleus}{nucleus}, the
\omterm{def:b1:eukaryotic-cell:endomembrane}{endoplasmic reticulum}, the \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi apparatus}, \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes}, \omterm{def:b1:eukaryotic-cell:peroxisome}{peroxisomes},
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondria}, and in plants the \omterm{def:b1:eukaryotic-cell:plastid}{plastids} and the \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole}. \omterm{def:b1:cell-unit-of-life:prokeuk}{Organelles}
without a membrane: ribosomes, the \omterm{def:b1:eukaryotic-cell:cytoskeleton}{centrosome}, the \omterm{def:b1:eukaryotic-cell:cytoskeleton}{cytoskeleton}.
\end{definition}

\begin{definition}[The nucleus]\label{def:b1:eukaryotic-cell:nucleus}
The \emph{nucleus}\index{nucleus} holds the chromosomes as
\emph{chromatin}\index{chromatin}, DNA wound on histone proteins
(\cref{ch:b1:genomes}). It is bounded by the \emph{nuclear
envelope}\index{nuclear envelope}, two membranes, the outer one
continuous with the \omterm{def:b1:eukaryotic-cell:endomembrane}{endoplasmic reticulum}, pierced by \emph{nuclear
pores}\index{nuclear pore} through which RNA and proteins pass in both
directions under control. The \emph{nucleolus}\index{nucleolus} is the
region where ribosomal RNA is made and ribosomes are assembled. The
nucleus separates transcription (inside) from translation (outside),
a separation prokaryotes do not have.
\end{definition}

\begin{proposition}[The compartments of a liver cell]\label{prop:b1:eukaryotic-cell:budget}
In a hepatocyte of \qty{5000}{\micro m^3}, the \omterm{def:b1:eukaryotic-cell:organelle}{cytosol} takes about
half the volume, the \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondria} a fifth, the \omterm{def:b1:eukaryotic-cell:endomembrane}{endoplasmic reticulum} a
sixth, the \omterm{def:b1:eukaryotic-cell:nucleus}{nucleus} a sixteenth, the Golgi, \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes} and \omterm{def:b1:eukaryotic-cell:peroxisome}{peroxisomes} a
few percent together. The membranes tell a different story: of the
\qty{110000}{\micro m^2} of membrane in the \omterm{def:b1:cell-unit-of-life:cell}{cell}, half is \omterm{def:b1:eukaryotic-cell:endomembrane}{endoplasmic
reticulum}, a third the inner membrane of the \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondria}, and only
\qty{2}{\%} the plasma membrane. Most of a \omterm{def:b1:cell-unit-of-life:cell}{cell}'s membrane is inside
it.
\end{proposition}

\section{The endomembrane system}

\begin{definition}[Endoplasmic reticulum, Golgi apparatus, lysosome]\label{def:b1:eukaryotic-cell:endomembrane}
The \emph{endoplasmic reticulum}\index{endoplasmic reticulum} (ER) is
a network of membrane sheets and tubes enclosing one continuous space,
the lumen, and continuous with the \omterm{def:b1:eukaryotic-cell:nucleus}{nuclear envelope}. Its
\emph{rough}\index{rough endoplasmic reticulum} part is studded with
ribosomes that inject the proteins they make into the lumen or into the
membrane; its \emph{smooth}\index{smooth endoplasmic reticulum} part
makes lipids, stores calcium and detoxifies. The \emph{Golgi
apparatus}\index{Golgi apparatus} is a stack of flattened sacs
(cisternae), with a receiving \emph{cis} face near the ER and a
shipping \emph{trans} face, in which proteins from the ER are modified
(sugars trimmed and added), sorted, and packed into vesicles for their
destinations. \emph{Lysosomes}\index{lysosome} are acidic vesicles (pH
$\approx 5$) full of hydrolytic enzymes that digest material brought in
from outside or worn-out \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles}. Together with the vesicles that
move between them and the plasma membrane, these form the
\emph{endomembrane system}\index{endomembrane system}: one connected set
of compartments whose lumens are topologically the outside of the \omterm{def:b1:cell-unit-of-life:cell}{cell}.
\end{definition}

\begin{proposition}[The secretory pathway]\label{prop:b1:eukaryotic-cell:secretory}
\index{secretory pathway}
A protein destined for secretion, for the plasma membrane, or for a
\omterm{def:b1:eukaryotic-cell:endomembrane}{lysosome} is synthesised by ribosomes bound to the rough ER, enters the
ER lumen as it is made, folds there, travels in vesicles to the cis
face of the Golgi, crosses the stack while being modified, and \omterm{def:b1:flowering-plant-organization:organs}{leaves}
the trans face in a vesicle addressed to its destination: a secretory
granule that fuses with the plasma membrane on a signal
(\emph{regulated exocytosis}\index{exocytosis}), a vesicle that fuses
continuously (\emph{constitutive} exocytosis, which also delivers new
membrane), or a \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosome}. Cytosolic and nuclear proteins are made on
free ribosomes and never enter the system. What decides is a
\emph{signal sequence}\index{signal sequence} at the start of the
protein (\cref{ch:b1:gene-expression}).
\end{proposition}
\begin{proof}[Evidence]
Palade (1960s) fed slices of guinea-pig pancreas radioactive amino
acids for three minutes (the \emph{pulse}), then unlabelled ones (the
\emph{chase}), and located the radioactivity by autoradiography of
electron micrographs at intervals. At \qty{3}{min} the silver grains
lay over the rough ER; at \qty{20}{min} over the Golgi; at \qty{40}{min}
over condensing \omterm{def:b1:eukaryotic-cell:plantcell}{vacuoles} and new granules; at \qty{2}{h} in the duct
lumen. \omterm{def:b1:cell-unit-of-life:fractionation}{Cell fractionation} gave the same sequence chemically. Blobel
(1970s) showed that secretory proteins made in a test tube without
membranes are twenty residues longer than the secreted form and are not
protected from added protease, whereas made with ER membranes present
they are cleaved and protected: the extra residues are the signal that
directs the growing chain into the ER.
\end{proof}

\begin{method}[Following a molecule through the cell]\label{met:b1:eukaryotic-cell:pulsechase}
\begin{enumerate}
  \item Label a short cohort: expose \omterm{def:b1:cell-unit-of-life:cell}{cells} to a radioactive or
        fluorescent precursor for a few minutes (the pulse), then
        replace it by excess unlabelled precursor (the chase), so that
        only molecules made during the pulse carry the label.
  \item Sample at intervals, fix, and locate the label: by
        autoradiography of sections, by fluorescence microscopy, or by
        fractionation and counting.
  \item The compartment whose label peaks first is upstream; the
        sequence of peaks is the route; the intervals are the transit
        times.
  \item Block a step (cold, an inhibitor of energy, a mutant) and see
        where the label accumulates: the compartment before the block.
\end{enumerate}
\end{method}

\begin{definition}[Endocytosis]\label{def:b1:eukaryotic-cell:endocytosis}
\emph{Endocytosis}\index{endocytosis} brings material in: the plasma
membrane invaginates around it and pinches off a vesicle, which
generally fuses with a \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosome}. \emph{Phagocytosis} engulfs particles
(a bacterium, a dead \omterm{def:b1:cell-unit-of-life:cell}{cell}); \emph{pinocytosis} takes in droplets of
fluid; \emph{receptor-mediated endocytosis} concentrates a particular
molecule (cholesterol-carrying particles, iron-carrying transferrin) on
receptors in coated pits before taking it in. Exocytosis adds membrane
to the \omterm{def:b1:cell-unit-of-life:cell}{cell} surface and endocytosis removes it; in a secreting \omterm{def:b1:cell-unit-of-life:cell}{cell} the
two balance so that the surface stays constant while whole areas of
membrane cycle through it every hour.
\end{definition}

\section{Mitochondria and plastids}

\begin{definition}[Mitochondrion]\label{def:b1:eukaryotic-cell:mitochondrion}
A \emph{mitochondrion}\index{mitochondrion} is an \omterm{def:b1:eukaryotic-cell:organelle}{organelle}
\qtyrange{0.5}{1}{\micro m} across and a few micrometres long, bounded
by two membranes: a smooth outer membrane permeable to small molecules,
and an inner membrane folded into \emph{cristae}\index{crista} that is
impermeable to ions and carries the respiratory chain and the ATP
synthase (\cref{ch:b1:respiration-fermentation}). Inside is the
\emph{matrix}\index{mitochondrial matrix}, holding the enzymes of the
Krebs cycle, ribosomes of the bacterial type and several copies of a
small circular DNA. Mitochondria divide by fission, move along the
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{cytoskeleton}, and number from a few hundred to a few thousand per \omterm{def:b1:cell-unit-of-life:cell}{cell}.
\end{definition}

\begin{definition}[Plastids]\label{def:b1:eukaryotic-cell:plastid}
\emph{Plastids}\index{plastid} are the double-membraned \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles} of
plant \omterm{def:b1:cell-unit-of-life:cell}{cells}: \emph{chloroplasts}\index{chloroplast}, in which a third
system of membranes, the \emph{thylakoids}\index{thylakoid}, stacked in
grana and bathed in the \emph{stroma}\index{stroma}, carries out
photosynthesis (\cref{ch:b1:photosynthesis}); amyloplasts, which store
starch; chromoplasts, which hold the pigments of fruits and petals. All
derive from proplastids and, like \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondria}, contain their own
circular DNA and bacterial-type ribosomes and divide by fission.
\end{definition}

\begin{proposition}[Endosymbiotic origin]\label{prop:b1:eukaryotic-cell:endosymbiosis}
\index{endosymbiosis}
\omterm{def:b1:eukaryotic-cell:mitochondrion}{Mitochondria} and \omterm{def:b1:eukaryotic-cell:plastid}{plastids} descend from free-living bacteria engulfed by
an ancestral \omterm{def:b1:cell-unit-of-life:prokeuk}{eukaryotic cell} and retained as symbionts: \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondria}
from an aerobic bacterium, \omterm{def:b1:eukaryotic-cell:plastid}{plastids} from a cyanobacterium.
\end{proposition}
\begin{proof}[Evidence]
Both \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles} have two membranes, the inner of bacterial composition
and the outer resembling the host's; a circular chromosome without
histones; ribosomes of the bacterial size and sensitivity to
antibiotics; a genetic code with bacterial variants; division by
binary fission independent of the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s; and their genes, when
sequenced, group with those of specific bacterial lineages
(alphaproteobacteria, cyanobacteria) rather than with the nuclear
genes of the \omterm{def:b1:cell-unit-of-life:cell}{cell} that houses them. Most of their original genes have
since moved to the \omterm{def:b1:eukaryotic-cell:nucleus}{nucleus}, and the \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles} can no longer live
alone.
\end{proof}

\begin{definition}[Peroxisome]\label{def:b1:eukaryotic-cell:peroxisome}
A \emph{peroxisome}\index{peroxisome} is a small single-membraned
\omterm{def:b1:eukaryotic-cell:organelle}{organelle} containing oxidative enzymes that transfer hydrogen from
substrates (fatty acids, alcohol) to oxygen, producing hydrogen
peroxide, and catalase, which destroys the peroxide on the spot. It
confines a dangerous chemistry.
\end{definition}

\section{The cytoskeleton}

\begin{definition}[Cytoskeleton]\label{def:b1:eukaryotic-cell:cytoskeleton}
The \emph{cytoskeleton}\index{cytoskeleton} is a network of protein
filaments that gives the \omterm{def:b1:cell-unit-of-life:cell}{cell} its shape, anchors its \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles}, and
moves them and the \omterm{def:b1:cell-unit-of-life:cell}{cell} itself. Three kinds:
\emph{microfilaments}\index{microfilament} of actin (\qty{7}{nm},
two twisted strands of globular subunits), concentrated under the
plasma membrane, responsible for \omterm{def:b1:cell-unit-of-life:cell}{cell} shape, crawling, the contractile
ring of division and muscle contraction with myosin;
\emph{microtubules}\index{microtubule} (\qty{25}{nm} hollow tubes of
tubulin), radiating from the \emph{centrosome}\index{centrosome},
tracks along which motor proteins (kinesin outward, dynein inward)
carry vesicles and \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles}, the spindle of division, and the core of
cilia and flagella; \emph{intermediate filaments}\index{intermediate
filament} (\qty{10}{nm}, rope-like, of keratins and related proteins),
the mechanically resistant fibres that stiffen \omterm{def:b1:cell-unit-of-life:cell}{cells} and anchor at
desmosomes. Microfilaments and microtubules assemble and disassemble
continuously from their subunit pools.
\end{definition}

\begin{example}[Moving a vesicle]\label{ex:b1:eukaryotic-cell:transport}
A secretory vesicle leaving the Golgi is picked up by kinesin, a
motor that walks along a \omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtubule} toward the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s periphery at
about \qty{1}{\micro m/s}, spending one ATP per \qty{8}{nm} step; in a
\omterm{def:b1:body-plans-tissues:nervous}{neuron} the same motor carries vesicles the length of an axon a metre
long --- a journey of ten days. Near the surface, the vesicle passes to
the actin cortex and to myosin for the last micrometre. Diffusion would
take a vesicle of \qty{100}{nm} about a minute to cross a \omterm{def:b1:cell-unit-of-life:cell}{cell} of
\qty{20}{\micro m}, and years to travel a metre: \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles} are carried,
not left to wander.
\end{example}

\section{Plant and animal cells}

\begin{definition}[Cell wall, vacuole, plasmodesmata]\label{def:b1:eukaryotic-cell:plantcell}
A plant \omterm{def:b1:cell-unit-of-life:cell}{cell} adds three structures to the common plan. The \emph{cell
wall}\index{cell wall}, outside the plasma membrane, is a composite of
cellulose microfibrils in a matrix of other polysaccharides
(\cref{ch:b1:carbohydrates}); it fixes the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s shape, resists the
pressure inside, and cements neighbouring \omterm{def:b1:cell-unit-of-life:cell}{cells} together through a
shared middle lamella. The \emph{vacuole}\index{vacuole}, a
single-membraned compartment holding up to \qty{90}{\%} of the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s
volume, stores water, ions, sugars, pigments and wastes; its osmotic
pressure pushes the membrane against the wall, and this
\emph{turgor}\index{turgor} is what holds a \omterm{def:b1:flowering-plant-organization:organs}{leaf} up. \emph{Plasmodesmata}\index{plasmodesma}
are channels through the walls, lined by plasma membrane, through which
the \omterm{def:b1:eukaryotic-cell:organelle}{cytosols} of neighbouring \omterm{def:b1:cell-unit-of-life:cell}{cells} are continuous, so that a plant
\omterm{def:b1:body-plans-tissues:tissue}{tissue} is in part one connected \omterm{def:b1:cell-unit-of-life:cell}{cytoplasm} (the symplast).
\end{definition}

\begin{proposition}[Plant and animal cell compared]\label{prop:b1:eukaryotic-cell:compare}
Both have a \omterm{def:b1:eukaryotic-cell:nucleus}{nucleus}, ER, Golgi, \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondria}, \omterm{def:b1:eukaryotic-cell:peroxisome}{peroxisomes}, ribosomes
and a \omterm{def:b1:eukaryotic-cell:cytoskeleton}{cytoskeleton}. The plant \omterm{def:b1:cell-unit-of-life:cell}{cell} has a cellulose wall, a large
\omterm{def:b1:eukaryotic-cell:plantcell}{vacuole}, \omterm{def:b1:eukaryotic-cell:plastid}{plastids} and \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesmata}, and no centrioles or \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes}
(the \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole} digests); it does not crawl or engulf, and divides by
building a new wall across the middle. The animal \omterm{def:b1:cell-unit-of-life:cell}{cell} has no wall
(and so can change shape, crawl and phagocytose), \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes}, a
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{centrosome} with centrioles, and communicates through junctions rather
than \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesmata}; it divides by pinching in two.
\end{proposition}

\begin{example}[Turgor]\label{ex:b1:eukaryotic-cell:turgor}
A \omterm{def:b1:flowering-plant-organization:organs}{leaf} \omterm{def:b1:cell-unit-of-life:cell}{cell} whose \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole} holds \qty{0.3}{mol/L} of solutes draws water
in until the wall pushes back with a pressure of about \qty{0.7}{MPa},
seven atmospheres; the \omterm{def:b1:cell-unit-of-life:cell}{cell} is then turgid and the \omterm{def:b1:flowering-plant-organization:organs}{leaf} firm. Deprived
of water, the \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole} loses volume, the pressure falls to zero, and the
\omterm{def:b1:flowering-plant-organization:organs}{leaf} wilts --- the wall unchanged, the pressure gone
(\cref{ch:b1:membranes-transport}). An animal \omterm{def:b1:cell-unit-of-life:cell}{cell} in the same
solution, having no wall, would simply burst.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:eukaryotic-cell:1}
List the membrane-bounded \omterm{def:b1:cell-unit-of-life:prokeuk}{organelles} of an animal \omterm{def:b1:cell-unit-of-life:cell}{cell} and give the
main function of each in a few words.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:eukaryotic-cell:2}
Trace the path of a digestive enzyme from its synthesis to the duct,
naming each compartment and how the protein passes between them.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:eukaryotic-cell:3}
From the membrane-budget figure, what fraction of the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s membrane
is mitochondrial (outer plus inner)? Why is the inner membrane so much
larger than the outer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:eukaryotic-cell:4}
Give three structures a plant \omterm{def:b1:cell-unit-of-life:cell}{cell} has that an animal \omterm{def:b1:cell-unit-of-life:cell}{cell} lacks, and
two the animal \omterm{def:b1:cell-unit-of-life:cell}{cell} has that the plant \omterm{def:b1:cell-unit-of-life:cell}{cell} lacks.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:5}
From the pulse-chase figure, at what time does the label peak in the
ER, in the Golgi, and in the granules? Estimate the transit time from
ER to granule and the time before half the label has been secreted.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:6}
A \omterm{def:b1:cell-unit-of-life:cell}{cell} is treated with a drug that depolymerises \omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtubules}.
Predict the effects on the position of the Golgi, on vesicle traffic,
on \omterm{def:b1:cell-unit-of-life:cell}{cell} division and on the beating of cilia.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:7}
Give four pieces of evidence for the bacterial origin of \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondria},
and explain why a \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrion} nevertheless cannot live outside a
\omterm{def:b1:cell-unit-of-life:cell}{cell}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:8}
The lumen of the ER is "topologically outside the \omterm{def:b1:cell-unit-of-life:cell}{cell}". Explain the
phrase by following a membrane protein whose sugar chains face the
lumen from the ER to the plasma membrane: which side do the sugars face
at the end?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:9}
A secretory \omterm{def:b1:cell-unit-of-life:cell}{cell} releases \qty{2000}{} granules of \qty{1}{\micro m}
diameter per day through an apical surface of \qty{40}{\micro m^2}.
Compute the membrane added per day and how many times the apical
surface must be recycled. What does this imply?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:eukaryotic-cell:10}
In a test tube containing ribosomes, mRNA for a secretory protein and
amino acids, the product is a chain of \qty{320}{} residues; with ER
vesicles added, a chain of \qty{300}{} residues appears inside the
vesicles and is not digested by a protease added afterwards. Interpret
each observation, and say what a mutant lacking the first twenty
residues would do.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:eukaryotic-cell:11}
A patient's \omterm{def:b1:cell-unit-of-life:cell}{cells} lack a lysosomal enzyme that degrades a lipid; the
lipid accumulates in swollen \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes} and the \omterm{def:b1:cell-unit-of-life:cell}{cells} die. Explain the
\omterm{def:b1:cell-unit-of-life:cell}{cell} biology, say why the disease is worst in \omterm{def:b1:cell-unit-of-life:cell}{cells} that renew their
membranes fastest, and propose how an enzyme injected into the blood
could reach the \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:eukaryotic-cell:12}
"The \omterm{def:b1:cell-unit-of-life:prokeuk}{eukaryotic cell} is a federation of former bacteria held together
by a membrane system." Discuss in a paragraph, weighing what the
endosymbiotic theory explains against what it does not (\omterm{def:b1:eukaryotic-cell:nucleus}{nucleus}, ER,
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{cytoskeleton}).
\end{exercise}

\section{Problem: The Acinar Cell of the Pancreas}

\begin{problem}[{Weekend problem --- a cell that secretes its own mass
of protein every day: Palade's clock, the membrane it must recycle, the
signal that routes the protein, ending on the transit time of a
digestive enzyme}]
\label{pb:b1:eukaryotic-cell:1}
A pancreatic acinar \omterm{def:b1:cell-unit-of-life:cell}{cell} is a cube of side \qty{12}{\micro m}, with its
apical face (\qty{144}{\micro m^2}, on the duct) and its basal face on
the blood. A human pancreas of \qty{80}{g} holds about $\num{8e10}$
such \omterm{def:b1:cell-unit-of-life:cell}{cells} and secretes \qty{10}{g} of enzyme protein per day. A
secretory granule is a sphere \qty{1}{\micro m} in diameter holding
protein at \qty{200}{mg/mL}. Ribosomes add \qty{5}{} amino acids per
second; a typical enzyme has \qty{250}{} residues of mean mass
\qty{110}{g/mol}.

\medskip
\noindent\textbf{Part I --- Palade's clock.}
The pulse-chase figure of the chapter gives the location of the label
with time.
\begin{enumerate}
  \item Why must the pulse be short and the chase contain a large
        excess of unlabelled amino acid?
  \item Read off the time at which the label peaks in the ER, in the
        Golgi and in the granules.
  \item Estimate the transit time from ER to Golgi and from Golgi to
        granule.
  \item Where is the label at \qty{60}{min}, and why has some of it
        already left the \omterm{def:b1:cell-unit-of-life:cell}{cell} while most of it is still in granules?
  \item A second experiment is done at \qty{15}{\celsius}: the label
        stays in the ER for over an hour. Interpret.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The mass of protein.}
\begin{enumerate}[resume]
  \item Compute the protein secreted per \omterm{def:b1:cell-unit-of-life:cell}{cell} per day, in picograms.
  \item Compute the volume of one granule and the mass of protein it
        holds.
  \item Compute the number of granules released per \omterm{def:b1:cell-unit-of-life:cell}{cell} per day, and
        per minute.
  \item Compute the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s volume and, with a density of
        \qty{1.05}{g/cm^3}, its mass. What fraction of its own mass
        does it secrete each day?
  \item Compute the number of enzyme molecules in one granule
        (Avogadro's number $\num{6.0e23}$).
  \item Compute the time a ribosome needs to make one enzyme molecule.
  \item How many ribosomes must work continuously to make the daily
        output? Compare with the $\num{4e6}$ ribosomes such a \omterm{def:b1:cell-unit-of-life:cell}{cell}
        contains.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Membrane.}
\begin{enumerate}[resume]
  \item Compute the membrane area of one granule.
  \item Compute the membrane delivered to the apical face per day by
        exocytosis of the granules of question 8.
  \item How many times per day is the area of the apical face added?
        What must the \omterm{def:b1:cell-unit-of-life:cell}{cell} do with it, and by what process?
  \item The granule membrane arrives from the Golgi, which receives it
        from the ER. If the ER of the \omterm{def:b1:cell-unit-of-life:cell}{cell} has \qty{20000}{\micro m^2}
        of membrane, how many days' worth of granule membrane is that?
        What does this say about recycling within the \omterm{def:b1:eukaryotic-cell:endomembrane}{endomembrane
        system}?
  \item The apical face carries \qty{2000}{} microvilli of
        \qty{0.1}{\micro m} diameter and \qty{1}{\micro m} length. By
        what factor do they enlarge the \qty{144}{\micro m^2} face?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The address on the protein.}
\begin{enumerate}[resume]
  \item An enzyme's mRNA translated in a test tube without membranes
        gives a chain \qty{20}{} residues longer than the enzyme found
        in the granules. Where are the extra residues and what are
        they?
  \item With ER vesicles present, the product is the right length,
        inside the vesicles, and protected from a protease. Interpret.
  \item The same experiment with the mRNA of a cytosolic enzyme gives
        a product outside the vesicles and unprotected. Conclude.
  \item A mutation deletes the twenty residues. Predict where the
        enzyme ends up and what happens to the \omterm{def:b1:cell-unit-of-life:cell}{cell}.
  \item Another enzyme is found in the \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes} rather than the
        granules. It carries the same kind of \omterm{prop:b1:eukaryotic-cell:secretory}{signal sequence}. Where
        along the pathway must the second decision be taken, and by
        what mechanism, in outline?
  \item A \omterm{def:b1:cell-unit-of-life:cell}{cell} is treated with a drug that blocks ATP synthesis.
        Predict where the label of a pulse-chase accumulates and why
        (two reasons).
  \item Combine questions 3 and 8: at a steady state, how many granules
        are "in transit" between ER and granule at any moment?
  \item Summarise the result: the transit time from synthesis to
        granule, the time to secretion, the granules released per
        minute, and the fraction of its own mass the \omterm{def:b1:cell-unit-of-life:cell}{cell} secretes
        daily.
\end{enumerate}
\end{problem}
