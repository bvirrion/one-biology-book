\section*{Chapter \ref{ch:b1:plant-water-minerals} --- Plant Water and Mineral Nutrition}

\begin{solution}{exo:b1:plant-water-minerals:1}
\omterm{def:b1:plant-water-minerals:pathways}{Apoplast}: the continuous walls and spaces, crossed without passing a
membrane. \omterm{def:b1:plant-water-minerals:pathways}{Symplast}: the connected \omterm{def:b1:cell-unit-of-life:cell}{cytoplasm} of the \omterm{def:b1:cell-unit-of-life:cell}{cells}, entered
through one membrane and continued through \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesmata}. The
\omterm{def:b1:plant-water-minerals:pathways}{Casparian strip} blocks the \omterm{def:b1:plant-water-minerals:pathways}{apoplast} at the \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}, forcing
everything through a membrane and preventing back-leakage from the
\omterm{def:b1:flowering-plant-organization:tissues}{xylem}.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:2}
N (\omterm{def:b1:proteins:peptide}{proteins}, nucleic acids), P (\omterm{def:b1:nucleic-acids:nucleotide}{nucleotides}, membranes), K (osmotic
agent, \omterm{def:b1:flowering-plant-organization:tissues}{stomata}), Ca (walls, signalling), Mg (\omterm{def:b1:photosynthesis:pigments}{chlorophyll}), S (\omterm{def:b1:proteins:aminoacid}{amino
acids} cysteine and methionine).
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:3}
Soil \qty{-0.2}{}, \omterm{def:b1:flowering-plant-organization:organs}{root} \omterm{def:b1:flowering-plant-organization:tissues}{xylem} \qty{-0.6}{}, \omterm{def:b1:flowering-plant-organization:organs}{leaf} \omterm{def:b1:cell-unit-of-life:cell}{cell} \qty{-1.3}{MPa}:
from soil into the \omterm{def:b1:flowering-plant-organization:organs}{root}, from \omterm{def:b1:flowering-plant-organization:organs}{root} \omterm{def:b1:flowering-plant-organization:tissues}{xylem} up to the \omterm{def:b1:flowering-plant-organization:organs}{leaf} --- always
toward the more negative value.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:4}
Nitrogen is mobile in the phloem: the plant withdraws it from old
\omterm{def:b1:flowering-plant-organization:organs}{leaves} to feed the young, so old \omterm{def:b1:flowering-plant-organization:organs}{leaves} yellow first. Iron is not
remobilised: the young \omterm{def:b1:flowering-plant-organization:organs}{leaves}, built without it, are the ones that
yellow.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:5}
\omterm{def:b1:cell-unit-of-life:cell}{Cell} $\Psi = -0.9 + 0.5 = \qty{-0.4}{MPa}$: soil ($-0.2$) $>$ \omterm{def:b1:cell-unit-of-life:cell}{cell}
($-0.4$) $>$ \omterm{def:b1:flowering-plant-organization:tissues}{xylem} ($-0.6$), so water flows soil $\to$ \omterm{def:b1:cell-unit-of-life:cell}{cell} $\to$
\omterm{def:b1:flowering-plant-organization:tissues}{xylem}. Uptake stops when the \omterm{def:b1:cell-unit-of-life:cell}{cell} reaches \qty{-0.2}{MPa}, i.e.\
$\Psi_p = -0.2 + 0.9 = \qty{0.7}{MPa}$.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:6}
$\Psi = 137\ln(\text{RH})$: \qty{-14}{MPa}, \qty{-95}{MPa},
\qty{-315}{MPa}. Even humid air is ten times drier, in this sense,
than soil at the wilting point: a \omterm{def:b1:flowering-plant-organization:organs}{leaf} always loses water to air; the
only question is how fast.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:7}
$E_K = 25.7\ln(0.1/100) = \qty{-178}{mV}$: at \qty{-150}{mV} the \omterm{def:b1:cell-unit-of-life:cell}{cell}
is slightly less negative than needed, so a thousandfold accumulation
needs a little more than \omterm{def:b1:membranes-transport:transporters}{channels} --- but a ratio of 100 would be at
equilibrium: \omterm{def:b1:membranes-transport:transporters}{channels} do nearly all of it. Nitrate: $E = -25.7\ln(0.5/5)
= \qty{+59}{mV}$; the \omterm{def:b1:cell-unit-of-life:cell}{cell} is at \qty{-150}{}, \qty{209}{mV} away and
in the wrong direction: an anion must be pumped in, by proton symport.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:8}
At night, with no transpiration, ion loading into the \omterm{def:b1:flowering-plant-organization:tissues}{xylem} draws
water in and a \omterm{prop:b1:plant-water-minerals:rootpressure}{root pressure} of a few tenths of a megapascal pushes
sap up a few metres at most --- enough to reach the tips of grass
blades and force droplets out, not to reach the top of a tree. When
the \omterm{def:b1:flowering-plant-organization:tissues}{stomata} open, transpiration puts the \omterm{def:b1:flowering-plant-organization:tissues}{xylem} under tension and \omterm{prop:b1:plant-water-minerals:rootpressure}{root
pressure} disappears.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:9}
Without molybdenum \omterm{prop:b1:plant-water-minerals:nitrate}{nitrate reductase} does not work: on nitrate the
plant cannot make \omterm{def:b1:proteins:aminoacid}{amino acids} and shows nitrogen deficiency though
nitrate is abundant. On ammonium it grows normally, since ammonium
enters the \omterm{def:b1:proteins:aminoacid}{amino acids} without reduction. The difference locates the
element in one \omterm{def:b1:enzymes:enzyme}{enzyme}.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:10}
\qty{30}{mg} of N $= \qty{2.14}{mmol}$; $\times 8 = \qty{17}{mmol}$ of
electrons, \qty{8.6}{mmol} of NADPH-equivalents. \omterm{def:b1:flowering-plant-organization:organs}{Leaf}: $0.05\times
4\times 10^{-6}\times 43\,200 = \qty{8.6}{mmol}$ of $\mathrm{CO_2}$,
\qty{35}{mmol} of electrons: nitrate reduction takes half as many
electrons as carbon fixation on this reckoning --- a large fraction,
which is why plants do it in the light and why fast-growing \omterm{def:b1:flowering-plant-organization:organs}{leaves}
are heavy consumers of reductant.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:11}
\qty{2.14}{mmol} of N $= \qty{1.07}{mmol}$ of $\mathrm{N_2}$: \omterm{def:b1:water-small-molecules:atp}{ATP}
$16\times 1.07 = \qty{17}{mmol}$, i.e.\ \qty{0.57}{mmol} of glucose,
\qty{0.10}{g}; plus 8 electrons per $\mathrm{N_2}$ (another
\qty{0.05}{g} of glucose). The \qty{6}{g/g} rule gives \qty{0.18}{g} a
day: nitrogenase itself is most of it, and the rest is building and
maintaining the nodules, their leghaemoglobin, and the bacteria's own
respiration to keep oxygen low.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:12}
The \omterm{def:b1:flowering-plant-organization:organs}{root} itself takes water and the mobile ions (nitrate, potassium)
that the soil water brings to its hairs, spending \omterm{def:b1:water-small-molecules:atp}{ATP} only on the
\omterm{prop:b1:plant-water-minerals:uptake}{proton pump}. For phosphate, which will not come to it, it pays a
fungus in sugar to extend its reach; for nitrogen in a soil without
nitrate it pays bacteria in sugar to fix it. The fee is a tenth or
more of its photosynthesis, and the plant stops paying when the
service is not needed: on phosphate-rich soil mycorrhizae are reduced,
on nitrate-rich soil legumes make fewer nodules --- contractors are
kept only while they are cheaper than doing the job alone.
\end{solution}

\begin{solution}{pb:b1:plant-water-minerals:1}
\textbf{1.} $-0.9 + 0.5 = \qty{-0.4}{MPa}$.
\textbf{2.} $-0.2 > -0.4 > -0.6 > -1.3$: each step lower, water moves
soil $\to$ cortex $\to$ \omterm{def:b1:flowering-plant-organization:tissues}{xylem} $\to$ \omterm{def:b1:flowering-plant-organization:organs}{leaf}.
\textbf{3.} $137\ln 0.5 = \qty{-95}{MPa}$.
\textbf{4.} Soil to \omterm{def:b1:flowering-plant-organization:organs}{leaf} \qty{1.1}{MPa}; \omterm{def:b1:flowering-plant-organization:organs}{leaf} to air \qty{94}{MPa}:
\qty{99}{\%} of the total is the last step.
\textbf{5.} Soil at $-0.9$ is below the cortex \omterm{def:b1:cell-unit-of-life:cell}{cells}' $-0.4$: water
\omterm{def:b1:flowering-plant-organization:organs}{leaves} the \omterm{def:b1:cell-unit-of-life:cell}{cells} until they fall to $-0.9$, i.e.\ their \omterm{def:b1:eukaryotic-cell:plantcell}{turgor} drops
to zero; uptake stops until the \omterm{def:b1:cell-unit-of-life:cell}{cells} lower their $\Psi_s$ by
accumulating solutes.
\textbf{6.} The soil is now below the \omterm{def:b1:flowering-plant-organization:organs}{leaves}' $-1.3$: water flows out
of the plant; the \omterm{def:b1:flowering-plant-organization:organs}{leaves} lose \omterm{def:b1:eukaryotic-cell:plantcell}{turgor} and wilt. Watering raises the
soil's $\Psi$ above the \omterm{def:b1:flowering-plant-organization:organs}{root}'s and the flow resumes; the \omterm{def:b1:flowering-plant-organization:organs}{leaves}, whose
walls were intact, refill.
\textbf{7.} $2\times 10^{-7}\times 0.2\times(-0.2 - (-0.6)) =
\qty{1.6e-8}{m^3/s}$.
\textbf{8.} $1.6\times 10^{-8}\times 86\,400 = \qty{1.4e-3}{m^3} =
\qty{1.4}{L}$ per day.
\textbf{9.} \qty{1400}{g} over \qty{0.5}{m^2} and \qty{12}{h} of daylight
(most of it): about \qty{230}{g} per square metre per hour.
\textbf{10.} $230/18 = \qty{12.8}{mol}$ per square metre per hour, \qty{3.5}{mmol} per square metre per second --- a thousand water molecules for every $\mathrm{CO_2}$ fixed at \qty{4}{\micro mol}.
\textbf{11.} Difference $-0.2 - (-0.4) = 0.2$: flux halved,
\qty{0.7}{L} a day. Closing the \omterm{def:b1:flowering-plant-organization:tissues}{stomata} has cut the loss, the tension
has relaxed, and the whole ladder has risen: the plant trades carbon
gain for water when the air is driest.
\textbf{12.} Water moves down its own potential gradient, created by
the \omterm{def:b1:flowering-plant-organization:organs}{leaf}'s evaporation, so the \omterm{def:b1:flowering-plant-organization:organs}{root} is passive; nitrate must be
concentrated against both a concentration and an electrical gradient,
which only coupling to the \omterm{prop:b1:plant-water-minerals:uptake}{proton pump} can pay for.
\textbf{13.} $\qty{1.4}{L}\times\qty{2}{mmol/L} = \qty{2.8}{mmol}$,
\qty{39}{mg} of nitrogen.
\textbf{14.} Yes, with a third to spare; the surplus is stored in the
\omterm{def:b1:eukaryotic-cell:plantcell}{vacuoles} as nitrate or excluded at the \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}.
\textbf{15.} $\Delta G = RT\ln(5/2) + zFV = 2.48\times 0.92 + (-1)(96.5)(-0.15)
= 2.3 + 14.5 = \qty{16.8}{kJ/mol}$.
\textbf{16.} Per proton: $RT\ln(10^{-7.5}/10^{-5.5}) + F(-0.15) =
-11.4 - 14.5 = \qty{-25.9}{kJ}$; two protons: \qty{-52}{kJ}, three
times what is needed.
\textbf{17.} $\qty{2.14}{mmol}\times 8 = \qty{17}{mmol}$ of electrons.
\textbf{18.} $0.5\times 4\times 10^{-6}\times 43\,200 = \qty{86}{mmol}$
of $\mathrm{CO_2}$, \qty{346}{mmol} of electrons: \qty{5}{\%} to
nitrate.
\textbf{19.} It would save the 8 electrons per nitrogen; but taking
up a cation and releasing a proton for each acidifies the soil around
the \omterm{def:b1:flowering-plant-organization:organs}{root} and the \omterm{def:b1:cell-unit-of-life:cell}{cell} must export acid, and ammonium is toxic if it
accumulates, so it must be assimilated at once in the \omterm{def:b1:flowering-plant-organization:organs}{root}.
\textbf{20.} \qty{1.07}{mmol} of $\mathrm{N_2}$ $\times 16 =
\qty{17}{mmol}$ of \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{21.} $6\times 0.03 = \qty{0.18}{g}$ of sugar, \qty{6}{\%} of
\qty{3}{g}. With \qty{2}{mmol/L} of nitrate available for the cost of
its reduction (\qty{5}{\%} of the electrons), fixation is a poor
bargain and a legume makes few nodules; with no nitrate it is the only
way, and \qty{6}{\%} of production is cheap for the whole of the
plant's \omterm{def:b1:proteins:peptide}{protein}.
\textbf{22.} $1.4\times 2\times 10^{-6} = \qty{2.8}{\micro mol}$,
\qty{0.09}{mg} of phosphorus --- a fortieth of the need; the plant
must rely on diffusion to its \omterm{def:b1:flowering-plant-organization:organs}{root hairs}, which soon exhaust their
surroundings, and above all on its mycorrhizal fungus.
\textbf{23.} The strip blocks the \omterm{def:b1:plant-water-minerals:pathways}{apoplast}, so nitrate can enter the
\omterm{def:b1:flowering-plant-organization:tissues}{xylem} only through endodermal membranes that pump it inward; and it
prevents the concentrated \omterm{def:b1:flowering-plant-organization:tissues}{xylem} sap from leaking back out along the
walls, so the gradient is held.
\textbf{24.} Its \omterm{def:b1:flowering-plant-organization:tissues}{xylem} sap would resemble the soil solution --- dilute,
unselected, with sodium and other unwanted ions --- and could not be
kept concentrated; in salty soil sodium would reach the \omterm{def:b1:flowering-plant-organization:organs}{leaves} freely
and the plant would be poisoned, where a normal plant excludes it at
the \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}.
\textbf{25.} About \qty{1.4}{L} of water a day through \qty{0.2}{m^2}
of \omterm{def:b1:flowering-plant-organization:organs}{root}, carrying \qty{39}{mg} of nitrogen against a need of
\qty{30}{}, whose reduction takes about \qty{5}{\%} of the \omterm{def:b1:flowering-plant-organization:organs}{leaf}'s
photosynthetic electrons.
\end{solution}
