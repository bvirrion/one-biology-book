\chapter{المناعة الفطرية}\label{ch:g12:innate-immunity}

شظية تدخل في طرف إصبع وتنزع في غضون دقيقة. وبحلول المساء يصير الموضع
أحمر حارًا متورمًا نابضًا؛ وفي اليوم التالي تتكون تحت الجلد حبة من القيح
الأصفر؛ وفي اليوم الثالث تزول. ولم يقرر شيء، ولم يتعلم شيء: فالعلامات
الأربع نفسها تظهر في كل جرح، وعند كل إنسان، وفي كل ثديي، وقد فعلت ذلك
قبل أن يعلم أحد بزمن طويل أن الشظية تحمل جراثيم. وهي السطح المرئي لخط
الدفاع الأول في الجسم --- طقم من \omterm{def:g10:cells-common-unit:cell}{الخلايا} والجزيئات يتعرف الدخيل في
غضون دقائق ويهاجمه من غير أن يكون قد لقيه قط. ويصف هذا الفصل ذلك
الخط؛ ويصف التالي الدفاع الأبطأ المتعلِّم الذي يستدعيه.

\section{مناعتان}

\begin{definition}[المناعة الفطرية والمناعة التكيّفية]\label{def:g12:innate-immunity:immunity}
\emph{الجهاز المناعي}\index{جهاز مناعي} هو طقم الأعضاء \omterm{def:g10:cells-common-unit:cell}{والخلايا}
والجزيئات التي تدافع عن الجسم في وجه العدوى والأذى. وله جزآن.
\emph{فالمناعة الفطرية}\index{مناعة فطرية} حاضرة منذ الولادة، وتعمل في
غضون دقائق إلى ساعات، وتتعرف أصنافًا عريضة من الدخلاء --- جراثيم،
وفيروسات، وفطورًا، \omterm{def:g10:cells-common-unit:cell}{وخلايا} متضررة --- بسمات تشترك فيها، وتستجيب
الاستجابة نفسها في كل مرة. أما \emph{المناعة التكيّفية}\index{مناعة
تكيّفية} فتأخذ أيامًا لتعمل، وتتعرف كل دخيل على حدة، وتتذكره
(\cref{ch:g12:adaptive-immunity}). والجهاز الفطري يشترك فيه الحيوان
كله؛ أما التكيّفي فلا يوجد إلا عند \omterm{def:g10:common-ancestry:bodyplan}{الفقاريات}، وهو يتوقف على الفطري
ليطلق.
\end{definition}

\begin{proposition}[الحواجز أولًا]\label{prop:g12:innate-immunity:barriers}
معظم الدخلاء لا يدخلون قط. فالجلد، وهو جاف حمضي ينسل سطحه باستمرار،
يوقف كل شيء تقريبًا؛ وبطانات المسالك الهوائية والمعي والسبيل التناسلي
رطبة لكنها مكسوة بمخاط يحبس الجراثيم ويكنس أو يجدَّد؛ والدموع واللعاب
والمخاط تحمل إنزيمًا يشق جدر الجراثيم؛ وحمض المعدة وجراثيم المعي نفسها،
وهي تشغل المكان، لا تدع للوافدين إلا قليلًا. والاستجابة المناعية في هذا
الفصل تبدأ حين يخرق حاجز.
\end{proposition}
\admitted

\section{الاستجابة الالتهابية}

\begin{proposition}[العلامات الأربع وما يصنعها]\label{prop:g12:innate-immunity:inflammation}
الخرق --- جرح، أو عدوى، أو حرق --- يثير في غضون دقائق
\emph{الاستجابة الالتهابية}\index{التهاب}: احمرارًا وحرارة وتورمًا
وألمًا. ولكل واحدة سبب:
\begin{itemize}
  \item \emph{\omterm{def:g10:cells-common-unit:cell}{الخلايا} الحارسة} المقيمة في النسيج --- البلاعم الكبيرة،
        \omterm{def:g10:cells-common-unit:cell}{والخلايا} التغصنية، \omterm{def:g10:cells-common-unit:cell}{والخلايا} البدينة --- تحمل مستقبلات تتعرف
        جزيئات مشتركة بين أصناف كاملة من الجراثيم (قطعًا من جدر
        الجراثيم، وحموضًا نووية فيروسية) وجزيئات تحررها \omterm{def:g10:cells-common-unit:cell}{الخلايا}
        المتضررة. وهي، عند التعرف، تحرر
        \emph{وسائط}\index{وسيط التهابي}: الهيستامين،
        والبروستاغلاندينات، وبروتينات إشارة تسمى السيتوكينات.
  \item \omterm{prop:g12:innate-immunity:inflammation}{والوسائط} توسّع الأوعية الدموية الموضعية وتجعل جدرها راشحة:
        فدم أكثر (احمرار، وحرارة)، وبلازما ترشح إلى النسيج (تورم). وهي
        تحسّس أيضًا النهايات العصبية (ألم)، وتجعل جدر الأوعية لاصقة
        \omterm{def:g10:cells-common-unit:cell}{للخلايا} البيض المارة.
  \item و\emph{\omterm{def:g12:innate-immunity:phagocytosis}{البالعات}} --- العدلات من الدم في غضون ساعات، والوحيدات
        التي تصير بلاعم كبيرة في غضون يوم --- تعصر نفسها عبر جدر
        الأوعية، وتزحف صاعدة في تدرج \omterm{prop:g12:innate-immunity:inflammation}{الوسائط} إلى الموضع، وتبتلع
        الجراثيم والحطام وتهضمها. \omterm{def:g12:innate-immunity:phagocytosis}{والبالعات} الميتة والجراثيم والسائل
        هي القيح.
\end{itemize}
والاستجابة موضعية سريعة، وهي هي للشظية وللجرثومة: فلا تحتاج تماسًا
سابقًا مع الدخيل.
\end{proposition}
\begin{proof}[شواهد]
حقن قطعة من جدار جرثومي وحدها في الجلد، بلا أي جرثومة حية، ينتج
الاستجابة كاملة في غضون ساعة: فالإشارة نمط جزيئي لا الجرثومة نفسها.
وسد الهيستامين (بمضادات الهيستامين المستعملة في حمى الكلأ) يبطل
الاحمرار والتورم الباكرين؛ وسد \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} الذي يصنع البروستاغلاندينات
(بالأسبرين أو الإيبوبروفين) يقلل التورم والألم والحمى. ومشاهدة جرح
تحت المجهر تبدي العدلات تتدحرج على جدار الوعاء، ثم تقف، ثم تعبره
وتحتشد إلى الموضع في غضون ساعتين؛ والفئران التي استنفدت عدلاتها تموت
من عداوى يصفّيها فأر عادي في ليلة.
\end{proof}

\begin{definition}[البلعمة]\label{def:g12:innate-immunity:phagocytosis}
\emph{البلعمة}\index{بلعمة} ابتلاع جسيم --- جرثومة، أو \omterm{def:g10:cells-common-unit:cell}{خلية} ميتة، أو
ذرة غبار --- من \omterm{def:g10:cells-common-unit:cell}{خلية} تلف غشاءها حوله، وتسحبه إلى الداخل في حويصلة،
وتهضمه \omterm{def:g11:enzymes-and-phenotype:enzyme}{بالإنزيمات} وبمواد كيميائية فعالة. و\emph{البالعات}\index{بالعة}
في الاستجابة الفطرية هي العدلات، وهي قصيرة العمر كثيرة العدد،
والبلاعم الكبيرة، وهي طويلة العمر مقيمة. والبلعم الكبير الذي هضم
جرثومة يبقي قطعًا من بروتيناتها ويعرضها على سطحه --- وهي الخطوة التي
ستصل هذه الاستجابة باستجابة الفصل التالي.
\end{definition}

\begin{example}[جدول زمني لشظية]\label{ex:g12:innate-immunity:timetable}
الدقيقة 0: جراثيم من الشظية في النسيج. والدقائق من 1 إلى 10: \omterm{def:g10:cells-common-unit:cell}{الخلايا}
البدينة والبلاعم الكبيرة تتعرف قطع جدرها وتحرر الهيستامين
والسيتوكينات؛ وتتسع الأوعية. والساعة 1: احمرار ودفء، وأول العدلات
تعبر جدر الأوعية. والساعات من 2 إلى 12: العدلات تحتشد، وتبتلع
الجراثيم، وتموت؛ والتورم والألم في ذروتهما. واليوم 1: تصل الوحيدات
وتصير بلاعم كبيرة، فتصفّي الحطام؛ والقيح مرئي. واليومان 2 و3: الجراثيم
زالت، \omterm{prop:g12:innate-immunity:inflammation}{والوسائط} لم تعد تحرر، والأوعية عادت إلى حالها، والإصلاح يبدأ.
فإن تكاثرت الجراثيم أسرع مما تصفّيها \omterm{def:g12:innate-immunity:phagocytosis}{البالعات}، اتسعت الاستجابة، وظهرت
الحمى، وتولّت الاستجابة التكيّفية --- وقد بدأت أصلًا في العقدة اللمفية.
\end{example}

\section{تهدئة الاستجابة}

\begin{proposition}[الأدوية المضادة للالتهاب]\label{prop:g12:innate-immunity:drugs}
\omterm{prop:g12:innate-immunity:inflammation}{الاستجابة الالتهابية} نافعة ومكلفة: فتورمها وألمها ثمن تصفية الدخيل،
وهي إذا أفرطت، أو أسيء توجيهها (ضد غبار طلع غير ضار، أو ضد مفاصل الجسم
نفسه)، أو طالت، أحدثت أذى من عندها. و\emph{الأدوية المضادة
للالتهاب}\index{دواء مضاد للالتهاب} تعمل على وسائطها: فالأسبرين
والإيبوبروفين وأقرباؤهما تسد \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} الذي يصنع البروستاغلاندينات،
فتقلل الألم والتورم والحمى؛ والستيرويدات القشرية، المشتقة من \omterm{def:g11:hormones-and-reproduction:hormone}{هرمون}
الكظر، تطفئ إنتاج معظم \omterm{prop:g12:innate-immunity:inflammation}{الوسائط} وتستعمل في الالتهاب الشديد أو المزمن.
وكلها تقلل العلامات؛ ولا واحد منها يزيل السبب، وهي بتخميدها الاستجابة
قد تبطئ تصفية عدوى حقيقية.
\end{proposition}
\admitted

\begin{example}[الحمى]\label{ex:g12:innate-immunity:fever}
السيتوكينات المتحررة عند موضع التهاب كبير أو مديم تبلغ الدماغ، حيث
ترفع نقطة ضبط حرارة الجسم: وهي الحمى. ودرجة أو درجتان من الحرارة
الزائدة تسرّعان \omterm{def:g12:innate-immunity:phagocytosis}{البالعات} وتبطئان كثيرًا من الجراثيم، بكلفة من الطاقة
والانزعاج؛ وفوق \qty{40}{\celsius} تفوق الكلفة الفائدة، وهناك تستعمل
الأدوية الخافضة للحرارة (التي تسد البروستاغلاندين). فالحمى استجابة
منظَّمة، لا إخفاق في التنظيم.
\end{example}

\section{من الفطرية إلى التكيّفية}

\begin{proposition}[التسليم]\label{prop:g12:innate-immunity:handover}
بين \omterm{def:g10:cells-common-unit:cell}{الخلايا} الحارسة، لل\emph{خلايا التغصنية}\index{خلية تغصنية} عمل
ثان. فهي، بعد أن تبتلع الدخيل، تغادر النسيج عبر الأوعية اللمفية،
حاملةً قطع بروتيناته معروضةً على سطحها، وتسافر إلى أقرب \emph{عقدة
لمفية}. وهناك تعرض القطع على لمفاويات الجهاز التكيّفي، والسيتوكينات
التي تحملها تخبر تلك اللمفاويات أي صنف من الدخلاء كان. فالاستجابة
الفطرية إذن لا تقاتل الأيام الأولى وحسب؛ بل تعيّن العدو وتسلّم وصفه.
ومن غير ذلك التسليم لا تبدأ استجابة تكيّفية --- ولهذا كان اللقاح يحوي،
إلى جانب المستضد، مواد تثير التهابًا صغيرًا.
\end{proposition}
\admitted

{\small استجابتان لعدوى أولى. الاستجابة الفطرية تبلغ كامل قوتها في
غضون ساعات وتصمد؛ والتكيّفية، التي أطلقها تقرير \omterm{def:g10:cells-common-unit:cell}{الخلايا} التغصنية،
تأخذ أسبوعًا لتبلغ ذروتها --- وهي، بخلاف الأولى، ستتذكر.}
\end{omfigure}

\begin{method}[قراءة التهاب]\label{met:g12:innate-immunity:read}
\begin{enumerate}
  \item سمّ الخرق والدخيل المرجَّح (أو الضرر بلا دخيل: فالالتواء يلتهب
        أيضًا).
  \item وانسب كل علامة إلى وسيطها وإلى تغير الأوعية: الاحمرار والحرارة
        إلى أوعية متسعة، والتورم إلى أوعية راشحة، والألم إلى أعصاب
        متحسسة.
  \item وعيّن \omterm{def:g10:cells-common-unit:cell}{الخلايا} الحاضرة في كل طور: الحارسات أولًا، ثم العدلات في
        غضون ساعات، ثم البلاعم الكبيرة في غضون يوم؛ والقيح بقاياها.
  \item واحكم هل الاستجابة متناسبة: موضعية تنحل في أيام (تعمل كما
        ينبغي)، أم منتشرة مع حمى (العدوى تكسب، وتلزم استجابة تكيّفية)،
        أم موجَّهة إلى شيء غير ضار أو مزمنة (وهي حالة تستدعي علاجًا
        مضادًا للالتهاب).
\end{enumerate}
\end{method}

\begin{remark}[قديمة وسريعة وكليلة]\label{rem:g12:innate-immunity:old}
\omterm{def:g12:innate-immunity:immunity}{المناعة الفطرية} مناعة الإسفنج والحشرة والبلوطة كما هي مناعة الإنسان:
فمستقبلات الأنماط \omterm{def:g12:innate-immunity:phagocytosis}{والبالعات} أقدم من \omterm{def:g10:common-ancestry:bodyplan}{الفقاريات} بمئات الملايين من
السنين. وهي سريعة لأنها مهيأة سلفًا لأصناف من الأعداء، وكليلة للسبب
نفسه: فهي لا تستطيع أن تميز جرثومة من جرثومة، ولا تستطيع أن تتحسن.
وجهاز الفصل التالي يستطيع الأمرين --- لكن بعد أن يكون هذا قد دق ناقوس
الخطر.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:innate-immunity:1}
أعط ثلاثة فروق بين \omterm{def:g12:innate-immunity:immunity}{المناعة الفطرية} \omterm{def:g12:innate-immunity:immunity}{والمناعة التكيّفية}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:innate-immunity:2}
اذكر علامات الالتهاب الأربع وتغير الأوعية أو الأعصاب وراء كل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:innate-immunity:3}
ماذا تتعرف \omterm{def:g10:cells-common-unit:cell}{الخلايا} الحارسة، وماذا تحرر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:innate-immunity:4}
صف \omterm{def:g12:innate-immunity:phagocytosis}{البلعمة} في أربع خطوات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:innate-immunity:5}
كيف يقلل الأسبرين والإيبوبروفين الالتهاب؟ وما الذي لا يفعلانه؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:6}
من شكل الاستجابتين، متى تبلغ الاستجابة الفطرية ذروتها، ومتى تبلغها
التكيّفية، وفي أي يوم يتقاطع المنحنيان؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:7}
قطعة من جدار جرثومي محقونة وحدها تسبب \omterm{prop:g12:innate-immunity:inflammation}{الاستجابة الالتهابية} كاملة.
فماذا يبين ذلك، وماذا كان سيحدث بحقنة من محلول ملحي عقيم؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:8}
مم يتألف القيح؟ ولماذا يظهر بعد يوم من الجرح لا في الحال؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:9}
فسّر لماذا يخفف مضاد الهيستامين حمى الكلأ ولا يفعل شيئًا في الأطوار
المتأخرة من عدوى جرثومية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:10}
من شكل التورم، قارن التورم عند 12 ساعة وعند 36 ساعة مع الدواء ومن
غيره. أيقصّر الدواء الاستجابة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:11}
كاحل ملتوٍ، بلا جرح ولا جرثومة، يصير أحمر حارًا متورمًا مؤلمًا. فسّر ما
الذي أثار \omterm{def:g10:cells-common-unit:cell}{الخلايا} الحارسة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:innate-immunity:12}
الفئران التي لا عدلات لها تموت من عداوى تصفّيها الفئران العادية في
ليلة؛ والفئران التي لا \omterm{def:g12:innate-immunity:immunity}{مناعة تكيّفية} لها تنجو منها لكنها تموت من عداوى
بعد بضعة أسابيع. فسّر ماذا تقول كل نتيجة عن دور الجهازين وعن توقيتهما.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:innate-immunity:13}
فسّر لماذا قد يرفض طبيب دواء مضادًا للالتهاب لمريض عنده خراج ويصفه
لمريض عنده التهاب مفاصل، مستعينًا بسبب كل التهاب.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:innate-immunity:14}
فسّر لماذا لا تستطيع الاستجابة الفطرية أن تتحسن مع تكرر العداوى،
ولماذا تستطيع التكيّفية ذلك، من طبيعة ما تتعرفه كل منهما.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:innate-immunity:15}
"الالتهاب مرض." ناقش في فقرة: متى يكون الدواء، ومتى يكون المرض، وما
الذي يقرر ذلك.
\end{exercise}

\section{مسألة: ثلاثة أيام من شظية}

\begin{problem}[{مسألة نهاية الأسبوع --- جرح يتابع ساعةً ساعةً: الخلايا
تعد، والوسائط تتتبع، والأدوية تختبر، والتسليم إلى العقدة اللمفية
يوقَّت}]
\label{pb:g12:innate-immunity:1}
شظية تحمل نحو $\num{10000}$ جرثومة إلى طرف إصبع؛ وهي تنقسم كل 40
دقيقة. والنسيج يحمل 200 بلعم كبير مقيم في حدود ميليمتر من الجرح؛
والعدلات تبدأ بالوصول من الدم بعد ساعة بمعدل $\num{50000}$ في الساعة،
وتستطيع كل واحدة أن تبتلع نحو 20 جرثومة قبل أن تموت.

\medskip
\noindent\textbf{القسم الأول --- السباق.}
\begin{enumerate}
  \item كم جرثومة تكون بعد ساعتين، ثم بعد أربع، لو لم يقتلها شيء؟
  \item البلاعم الكبيرة المئتان تبتلع كل واحدة نحو 5 جراثيم في الساعة.
        فكم تنزع في الساعة الأولى، وهل يوقف ذلك النمو؟
  \item ومن الساعة الأولى تصل العدلات. فكم جرثومة تستطيع عدلات الساعة
        الأولى أن تنزع جملةً؟
  \item قارن ذلك \omterm{def:g12:selection-drift-speciation:population}{بجمهرة} الجراثيم عند الساعة الثانية. من الرابح؟
  \item وعند أي ساعة، تقريبًا، يصفّى الجرح؟ وماذا يبقى في الموضع؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثاني --- العلامات.}
\begin{enumerate}[resume]
  \item انسب كلًا من الاحمرار والحرارة والتورم والألم إلى وسيط وإلى
        تغير في الأوعية أو في الأعصاب.
  \item فسّر لماذا يعين التورم العدلات، ولماذا يؤلم.
  \item طرف الإصبع ينبض مع كل ضربة قلب. فسّر ذلك من الأوعية المتسعة
        ومن الضغط في النسيج المتورم.
  \item وبعد يوم تكونت حبة قيح. اذكر محتوياتها.
  \item وفي اليوم الثالث زالت العلامات. فما الذي أوقف تحرر \omterm{prop:g12:innate-immunity:inflammation}{الوسائط}؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثالث --- الأدوية.}
\begin{enumerate}[resume]
  \item يؤخذ مضاد هيستامين عند الساعة الأولى. فأي العلامات يقلل، وأيها
        لا يقلل؟
  \item ويؤخذ إيبوبروفين عند الساعة السادسة. فأي وسيط يسد، وأي
        العلامات يقلل؟
  \item ويدهن مرهم ستيرويد قشري. فماذا يحل باستقدام العدلات، وأي خطر
        يترتب على ذلك في العدوى؟
  \item فسّر لماذا لا يقصّر أي من الأدوية الثلاثة العدوى، وأيها قد
        يطيلها.
  \item وعند المريض أيضًا حمى قدرها \qty{38.5}{\celsius}. فأي الجزيئات
        سببتها، وما فائدتها؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الرابع --- التقرير.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:cells-common-unit:cell}{الخلايا} التغصنية التي ابتلعت جراثيم عند الساعة الثانية تبلغ
        عقدة المرفق اللمفية عند الساعة العشرين، عارضةً قطعًا جرثومية.
        فماذا تحمل إلى جانب القطع، وإلى من ترفع تقريرها؟
  \item ومن شكل الاستجابتين، كم يومًا يمر قبل أن تكثر الأجسام المضادة ضد هذه
        الجراثيم؟ وهل ستلزم لهذه الشظية؟
  \item والجراثيم نفسها تدخل عبر شظية ثانية بعد شهر. فأي الاستجابتين
        تختلف في المرة الثانية، وأيهما تكون هي هي؟
  \item وشخص ولد بلا عدلات عاملة ينجو \omterm{def:g11:antibiotic-resistance:antibiotic}{بمضادات حيوية} يومية. فسّر أي جزء
        من الدفاع تعوّضه \omterm{def:g11:antibiotic-resistance:antibiotic}{المضادات الحيوية} وأيها لا تستطيع تعويضه.
  \item اذكر النتيجة: الساعة التي تتجاوز عندها \omterm{def:g12:innate-immunity:phagocytosis}{البالعات} الجراثيم،
        والوسيط وراء كل علامة من العلامات الأربع، \omterm{def:g10:cells-common-unit:cell}{والخلية} التي تحمل
        التقرير إلى العقدة اللمفية.
\end{enumerate}
\end{problem}
