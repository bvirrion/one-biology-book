\chapter{Bagaimana Tumbuhan Menyebar}\label{ch:g6:how-plants-spread}

Sebuah galian jalan baru dibuldoser pada bulan Maret: tanah mentah
yang telanjang, tanpa satu bintik hijau pun. Kembalilah pada bulan
September dan ia sudah terjahit di mana-mana oleh \omterm{prop:g2:from-seed-to-plant:cycle}{kecambah} --- \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}
popi, jombang, tunas dedalu, rerumputan. Tidak seorang pun
menyemainya. Namun \omterm{def:g1:plants-around-us:plant}{tumbuhan}, yang tidak dapat berjalan satu langkah
pun, menjajah tanah telanjang itu dalam hitungan bulan. Bab ini
tentang bagian perjalanan mereka.

\section{Tumbuhan mengembara sebagai biji dan spora}

\begin{proposition}[Tumbuhan yang menetap, permulaan yang mengembara]\label{prop:g6:how-plants-spread:travel}
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} yang dewasa terpasang seumur hidup
(\cref{def:g1:plants-around-us:plant}) --- sehingga sebuah \omterm{def:g5:biodiversity:species}{spesies}
menyebar dengan mengirimkan \emph{permulaannya}: \omterm{def:g2:from-seed-to-plant:seed}{biji} dan \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}
(\cref{ch:g5:flower-to-fruit}), \omterm{def:g4:plants-without-seeds:spore}{spora}
(\cref{def:g4:plants-without-seeds:spore}), atau \omterm{ex:g4:plants-without-seeds:runner}{geragih} dan
kemajuan vegetatif yang lain
(\cref{def:g4:plants-without-seeds:vegetative}). Menduduki tanah baru
dengan cara ini disebut
\emph{kolonisasi}\index{kolonisasi}.
\end{proposition}
\admitted

\begin{definition}[Penyebaran]\label{def:g6:how-plants-spread:dispersal}
Yang disebut \emph{penyebaran}\index{penyebaran} adalah pengangkutan
\omterm{def:g2:from-seed-to-plant:seed}{biji}, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} atau \omterm{def:g4:plants-without-seeds:spore}{spora} menjauh dari \omterm{def:g1:plants-around-us:plant}{tumbuhan} induknya. Pengangkutnya
sedikit dan tua: \emph{angin}, \emph{\omterm{def:g1:animals-around-us:animal}{hewan}}, \emph{air} --- dan
sebagian \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} yang melontarkan \omterm{def:g2:from-seed-to-plant:seed}{bijinya} sendiri.
\end{definition}

\begin{example}[Mengapa harus pergi dari rumah?]\label{ex:g6:how-plants-spread:whyleave}
Sebutir \omterm{def:g2:from-seed-to-plant:seed}{biji} yang jatuh di kaki induknya \omterm{def:g2:from-seed-to-plant:germination}{berkecambah} di dalam naungan
induknya, di antara \omterm{def:g1:plants-around-us:parts}{akar} induknya --- bersaing memperebutkan \omterm{def:g6:exploring-our-environment:conditions}{cahaya}
dan air yang persis sama pada
\cref{prop:g4:what-plants-need:needs} dengan seorang lawan yang
seratus kali lebih besar. \omterm{def:g6:how-plants-spread:dispersal}{Penyebaran} adalah pelarian dari pertandingan
yang kalah itu: jarak adalah pemberian pertama sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan} kepada
anaknya.
\end{example}

\section{Para pengangkut, dan buah yang dibangun untuknya}

\begin{proposition}[Dibangun untuk cocok dengan pengangkutnya]\label{prop:g6:how-plants-spread:builds}
Bentuk \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} dan \omterm{def:g2:from-seed-to-plant:seed}{biji} cocok dengan pengangkutnya --- kecocokan pada
\cref{def:g4:adapted-to-their-world:adaptation}, yang diterapkan pada
barang bawaan:
\begin{itemize}
  \item \emph{angin}: payung terjun (\omterm{def:g2:from-seed-to-plant:seed}{biji} jombang yang berjambul),
        sayap (\omterm{prop:g3:flowers-fruits-seeds:transform}{buah} bersayap maple dan ash yang berputar turun
        menyerong), atau \omterm{def:g4:plants-without-seeds:spore}{spora} sehalus debu yang sekadar mengapung
        (jutaan \omterm{def:g4:plants-without-seeds:spore}{spora} \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku});
  \item \emph{\omterm{def:g1:animals-around-us:animal}{hewan}, di luar}: \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} berkait --- burdok, rumput lengket
        --- yang kaitnya mencengkeram \omterm{def:g1:animals-around-us:covering}{bulu} dan kaus kaki lalu
        menumpang sampai dibersihkan;
  \item \emph{\omterm{def:g1:animals-around-us:animal}{hewan}, di dalam}: \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} berdaging yang manis --- ceri,
        beri hitam, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} mawar --- yang dimakan utuh; \omterm{def:g2:from-seed-to-plant:seed}{bijinya} yang
        berkulit liat melintas tanpa terluka lalu diendapkan di tempat
        lain, berikut pupuknya
        (jawaban \cref{exo:g3:flowers-fruits-seeds:10}, yang
        dilengkapi);
  \item \emph{air}: \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} yang mengapung --- rakit mungil pohon alder,
        \omterm{def:g1:animals-around-us:covering}{cangkang} kelapa yang berlayar di samudra;
  \item \emph{melontar sendiri}: polong yang mengering, menegang lalu
        meletik --- semak gorse dan \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} violet melontarkan \omterm{def:g2:from-seed-to-plant:seed}{bijinya}
        bermeter-meter dari induknya.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Membaca tepi jalannya]\label{ex:g6:how-plants-spread:roadside}
Galian jalan bulan September itu, yang sudah dipecahkan sandinya:
jombang dan dedalu --- \omterm{def:g2:from-seed-to-plant:seed}{biji} berjambul, yang terbang masuk bersama
angin; \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} popi --- \omterm{def:g2:from-seed-to-plant:seed}{bijinya} yang mungil sudah terbaring di dalam
tanah yang dipindahkan, menunggu sebagaimana hanya \omterm{def:g2:from-seed-to-plant:seed}{biji} yang bisa
(\cref{exo:g2:from-seed-to-plant:9}); \omterm{prop:g2:from-seed-to-plant:cycle}{kecambah} beri hitam di bawah
tiang pagarnya --- diendapkan \omterm{prop:g3:sorting-living-things:groups}{burung} yang bertengger, berikut pupuknya;
burdok di dekat jalan setapak --- terbawa masuk pada seekor anjing.
Rute kedatangan tiap penjajah tertulis pada perlengkapannya.
\end{example}

\begin{method}[Menyebutkan pengangkut dari buahnya]\label{met:g6:how-plants-spread:name}
Diberikan sebuah \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} atau \omterm{def:g2:from-seed-to-plant:seed}{biji}:
\begin{enumerate}
  \item carilah perlengkapan terbang --- jambul, sayap, kehalusan
        debu: anginnya;
  \item carilah kait: \omterm{def:g1:animals-around-us:covering}{bulu} \omterm{def:g1:animals-around-us:animal}{hewan}, di luar;
  \item berdaging, berwarna, manis? usus \omterm{def:g1:animals-around-us:animal}{hewan}, di dalam --- periksa
        \omterm{def:g2:from-seed-to-plant:seed}{bijinya}, apakah berkulit liat;
  \item mengapung, dekat air? airnya;
  \item polong yang kering, terpilin, dan bertegangan? melontar
        sendiri;
  \item ujilah putusanmu: jatuhkan ia, apungkan ia, tekankan ia pada
        \omterm{def:g1:my-body:parts}{lengan} baju berbulu wol.
\end{enumerate}
\end{method}

\section{Menjajah tanpa biji}

\begin{example}[Kemajuan di permukaan tanah]\label{ex:g6:how-plants-spread:vegetative}
Di samping para pendatang lewat udara, sebagian penjajah berjalan
dengan kaki vegetatifnya sendiri
(\cref{ch:g4:plants-without-seeds}): rumput grinting maju di bawah
tanah lewat \omterm{def:g1:plants-around-us:parts}{batang} yang menjalar, meter demi meter; semak beri hitam
melengkung ke atas lalu berakar di tempat ujungnya menyentuh tanah;
\omterm{ex:g4:plants-without-seeds:runner}{geragih} stroberi melompat melintasi tanah terbuka. Lambat, berjarak
pendek --- dan tak terhentikan melintasi sebuah kebun yang
terlantar, sebagaimana diketahui setiap tukang kebun.
\end{example}

\begin{remark}[Kolonisasi dua kecepatan]\label{rem:g6:how-plants-spread:twospeeds}
Kedua sistemnya membagi pekerjaannya. \omterm{def:g2:from-seed-to-plant:seed}{Biji} dan \omterm{def:g4:plants-without-seeds:spore}{spora} adalah layanan
\emph{jarak jauh} --- pendaratan yang jarang dan beruntung di tempat
yang jauh, jombang yang menyeberangi kotanya. Kemajuan vegetatif
adalah layanan \emph{jarak pendek} --- pendudukan yang rapat dan pasti
atas tanah di sebelahnya. Petak semak beri hitam yang menyebar dari
pagar tanamannya memakai keduanya sekaligus: \omterm{def:g1:plants-around-us:parts}{batang} yang melengkung
untuk semeternya, \omterm{def:g2:from-seed-to-plant:seed}{biji} yang dibawa \omterm{prop:g3:sorting-living-things:groups}{burung} untuk sekilometernya.
\end{remark}

\section{Latihan}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:2}

induknya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:3}
Sebutkan pengangkut bagi masing-masing: jambul jombang, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} berkait

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:4}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:5}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:6}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:8}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:10}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:11}

\begin{exercise}[$\star\star\star$]\label{exo:g6:how-plants-spread:12}

\section{Soal: Ladang yang Terlantar}

\begin{problem}[{Soal akhir pekan --- sepuluh tahun kolonisasi, yang
diamati dari satu gerbang}]\label{pb:g6:how-plants-spread:1}
Sebuah ladang yang sudah dibajak ditinggalkan pada musim gugur. Seorang
naturalis memotretnya dari gerbangnya setiap bulan Juni selama sepuluh
tahun, dan menyimpan catatan.

\medskip\noindent\textbf{Bagian I --- Tahun pertama.}
\begin{enumerate}
  \item Bulan Juni tahun pertama: tanah telanjangnya bertitik-titik
        \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} popi, jombang dan rumput duri. Untuk masing-masing,
        sebutkan sumber dan rute yang mungkin --- \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} popi dari
        tanahnya sendiri, yang lain lewat udara. Perlengkapan apa yang
        memungkinkan kedatangan lewat udara itu?
  \item Tidak ada beri hitam yang muncul di tengah ladangnya pada
        tahun pertama, tetapi tiga \omterm{prop:g2:from-seed-to-plant:cycle}{kecambah} berdiri di bawah tiang
        pagar yang tua. Jelaskan polanya dengan layanan
        di-dalam-hewannya --- mengapa persis di bawah tiangnya?
  \item Naturalisnya tidak menemukan \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} di mana pun,
        walaupun sebuah hutan yang penuh dengannya terletak di arah
        angin datang. \omterm{def:g4:plants-without-seeds:spore}{Spora} \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan pakunya} pasti tiba
        (\cref{ex:g4:plants-without-seeds:fern} menghitung jutaannya).
        \omterm{def:g6:exploring-our-environment:conditions}{Kondisi} ladang terbuka yang mana, menurut
        \cref{prop:g4:plants-without-seeds:damp}, yang menolak mereka?
  \item Rumput grinting menyerbu hanya dari tepi ladangnya, satu atau
        dua meter setahun. Sistem \omterm{prop:g6:how-plants-spread:travel}{kolonisasi} yang mana ini, dan
        mengapa ia tidak dapat muncul di tengah ladangnya seperti yang
        dilakukan jombang?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Tahun-tahun sesudahnya.}
\begin{enumerate}[resume]
  \item Menjelang tahun ketiga, rerumputan dan \omterm{def:g1:plants-around-us:plant}{tumbuhan} tinggi sudah
        menutup tanah telanjangnya, dan \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} popinya lenyap dari
        \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang berdiri. Di mana \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} popinya sekarang, dan
        dalam wujud apa? (Ingatlah galian jalannya dan
        \cref{prop:g6:life-through-seasons:herbs}.)
  \item Mulai tahun keempat, lengkung semak beri hitam dan semak
        hawthorn muda bertitik-titik di ladangnya, paling lebat di
        dekat pagar tanaman dan di bawah garis pagarnya. Sebutkan kedua
        layanan yang menempatkannya, dan buktinya pada polanya.
  \item Menjelang tahun kedelapan, anakan pohon dedalu dan birch
        berdiri lebih tinggi daripada semak beri hitamnya --- keduanya
        pohon yang \omterm{def:g2:from-seed-to-plant:seed}{bijinya} berjambul mungil atau seringan debu.
        Mengapa pohon ek dan berangan yang berbiji berat masih nyaris
        tidak ada, dan kurir \omterm{def:g1:animals-around-us:animal}{hewan} mana yang akhirnya menanam bahkan
        keduanya?
  \item Foto-fotonya memperlihatkan rombongan ladangnya berubah tahun
        demi tahun walaupun tidak ada tukang kebun yang menyentuhnya.
        Nyatakan aturan penggeraknya: apa yang diubah \omterm{def:g1:plants-around-us:plant}{tumbuhan} tiap
        tahunnya terhadap \omterm{def:g6:exploring-our-environment:conditions}{kondisi} ladangnya sendiri --- terutama
        \omterm{def:g6:exploring-our-environment:conditions}{cahaya} pada permukaan tanah --- dan siapa yang diuntungkan
        perubahan itu berikutnya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Pelajaran umumnya.}
\begin{enumerate}[resume]
  \item Urutkan para penjajahnya menurut kedatangan: \omterm{def:g2:from-seed-to-plant:seed}{biji} yang
        menunggu di dalam tanah, penumpang angin, semak yang dibawa
        \omterm{def:g1:animals-around-us:animal}{hewan}, pohon yang berbiji berat. Pasangkan tiap gelombangnya
        dengan layanan \omterm{def:g6:how-plants-spread:dispersal}{penyebaran} yang mengatur waktunya.
  \item Celah trotoar sebuah alun-alun kota bertunas rombongan tahun
        pertama yang sama --- \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} popi, jombang --- tetapi tidak
        pernah mencapai semak tahun keempatnya. Berikan dua alasan
        mengapa suksesinya mandek di situ.
  \item Naturalisnya mengaku: ``kalau dibiarkan sendiri, ladang ini
        akan menjadi hutan.'' Belalah pengakuannya dengan
        kecenderungan sepuluh tahunnya --- lalu sebutkan perkecualian
        padang rumput pada
        \cref{prop:g5:people-and-nature:managed} yang membuktikan
        bahwa ``dibiarkan sendiri'' adalah frasa yang menentukan.
  \item Rangkumlah seluruh soalnya dalam dua kalimat dengan memakai
        kata \emph{\omterm{def:g6:how-plants-spread:dispersal}{penyebaran}}, \emph{\omterm{def:g6:exploring-our-environment:conditions}{kondisi}} dan
        \emph{\omterm{prop:g6:how-plants-spread:travel}{kolonisasi}}.
\end{enumerate}
\end{problem}
