\chapter{Le cœur et le sang}\label{ch:g4:heart-and-blood}

Appuie deux doigts --- pas le pouce --- sur l'intérieur de ton
\omterm{def:g1:my-body:joint}{poignet}, sous la base du pouce, et attends. Voilà~: un petit coup
régulier, encore et encore, à peu près une fois par seconde. Ce coup,
c'est ton \omterm{def:g4:heart-and-blood:heart}{cœur} au travail, senti à distance. Il bat depuis avant ta
\omterm{def:g2:born-grow-die:cycle}{naissance}, et c'est le moteur du service de livraison du corps.

\section{Le sang, service de livraison du corps}

\begin{proposition}[Ce que transporte le sang]\label{prop:g4:heart-and-blood:carries}
Le \emph{sang}\index{sang} est un liquide qui atteint tous les
recoins du corps, et c'est le transporteur du corps~:
\begin{itemize}
  \item à l'\omterm{def:g4:journey-of-food:tract}{intestin grêle}, il recueille les \emph{\omterm{def:g4:journey-of-food:digestion}{nutriments}}
        (\cref{prop:g4:journey-of-food:absorb})~;
  \item aux \omterm{def:g4:breathing:organs}{poumons}, il recueille le \emph{dioxygène}
        (\cref{prop:g4:breathing:exchange})~;
  \item il livre les deux à toutes les parties du corps ---
        \omterm{def:g3:bones-and-muscles:muscle}{muscles}, cerveau, \omterm{def:g3:bones-and-muscles:skeleton}{os}, \omterm{def:g1:the-five-senses:sense}{peau}~;
  \item au retour, il emporte les déchets, en livrant le dioxyde de
        carbone aux \omterm{def:g4:breathing:organs}{poumons} pour qu'il soit expiré.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Une tournée de livraison]\label{ex:g4:heart-and-blood:round}
Suis une goutte de \omterm{prop:g4:journey-of-food:absorb}{sang}~: dans les \omterm{def:g4:breathing:organs}{poumons} elle devient rouge vif,
chargée de dioxygène~; le \omterm{def:g4:heart-and-blood:heart}{cœur} l'expédie~; dans un \omterm{def:g3:bones-and-muscles:muscle}{muscle} de \omterm{def:g1:my-body:parts}{jambe}
qui court, elle remet dioxygène et \omterm{def:g4:journey-of-food:digestion}{nutriments}, se charge de dioxyde
de carbone et fonce~; de retour au \omterm{def:g4:heart-and-blood:heart}{cœur}, elle est envoyée aux \omterm{def:g4:breathing:organs}{poumons}
pour décharger et recharger --- et repartir. Une tournée complète
prend bien moins d'une minute.
\end{example}

\begin{definition}[Vaisseaux sanguins]\label{def:g4:heart-and-blood:vessels}
Le \omterm{prop:g4:journey-of-food:absorb}{sang} circule dans des tubes appelés \emph{vaisseaux
sanguins}\index{vaisseau sanguin}~: de larges vaisseaux quittent le
\omterm{def:g4:heart-and-blood:heart}{cœur} et y reviennent, se ramifiant de plus en plus fin --- jusqu'à
des vaisseaux minces comme un cheveu qui se faufilent entre les
parties du corps, là où les livraisons sont remises. Mis bout à
bout, les vaisseaux d'un seul corps s'étendraient sur des dizaines
de milliers de kilomètres.
\end{definition}

\begin{remark}[Voir les vaisseaux]\label{rem:g4:heart-and-blood:see}
Regarde le dessous de ton \omterm{def:g1:my-body:joint}{poignet}~: des lignes bleutées sous la \omterm{def:g1:the-five-senses:sense}{peau}
--- des \omterm{def:g4:heart-and-blood:vessels}{vaisseaux sanguins} en route vers le \omterm{def:g4:heart-and-blood:heart}{cœur}. Un \omterm{def:g1:my-body:joint}{genou} éraflé
montre l'autre bout du réseau~: même la plus fine coupure trouve du
\omterm{prop:g4:journey-of-food:absorb}{sang}, parce que les vaisseaux les plus fins sont partout.
\end{remark}

\section{Le cœur, la pompe}

\begin{definition}[Le cœur]\label{def:g4:heart-and-blood:heart}
Le \emph{cœur}\index{cœur} est un \omterm{def:g3:bones-and-muscles:muscle}{muscle} creux, à peu près de la
taille du poing de son propriétaire, au milieu de la poitrine
derrière la cage thoracique. En se serrant --- un
\emph{battement}\index{battement de cœur} --- il pousse le \omterm{prop:g4:journey-of-food:absorb}{sang} dans
les vaisseaux. Il a deux côtés~: le côté droit envoie le \omterm{prop:g4:journey-of-food:absorb}{sang} aux
\emph{\omterm{def:g4:breathing:organs}{poumons}} pour charger du dioxygène~; le côté gauche envoie le
\omterm{prop:g4:journey-of-food:absorb}{sang} riche en dioxygène à tout le \emph{corps}.
\end{definition}

\begin{proposition}[Le pouls]\label{prop:g4:heart-and-blood:pulse}
Chaque \omterm{def:g4:heart-and-blood:heart}{battement} pousse une petite vague de \omterm{prop:g4:journey-of-food:absorb}{sang} le long des
vaisseaux~; sentie au \omterm{def:g1:my-body:joint}{poignet} ou au cou, cette vague est le
\emph{pouls}\index{pouls}. Compter le \omterm{prop:g4:heart-and-blood:pulse}{pouls}, c'est compter les
\omterm{def:g4:heart-and-blood:heart}{battements} du \omterm{def:g4:heart-and-blood:heart}{cœur}~: environ $80$ à $100$ par minute pour un enfant
au repos, $60$ à $80$ pour un adulte --- et bien davantage à
l'effort.
\end{proposition}
\admitted

\begin{example}[Le cœur suit les jambes]\label{ex:g4:heart-and-blood:effort}
Quand tu cours, tes \omterm{def:g3:bones-and-muscles:muscle}{muscles} de \omterm{def:g1:my-body:parts}{jambe} brûlent vite \omterm{def:g4:journey-of-food:digestion}{nutriments} et
dioxygène~; les livraisons doivent accélérer. Le \omterm{def:g4:heart-and-blood:heart}{cœur} bat donc plus
vite \emph{et} plus fort --- le \omterm{prop:g4:heart-and-blood:pulse}{pouls} d'un enfant peut dépasser $150$
dans un sprint --- pendant que la \omterm{def:g4:breathing:movements}{respiration} s'accélère pour
recharger le \omterm{prop:g4:journey-of-food:absorb}{sang} (\cref{ex:g4:breathing:running}). Le \omterm{def:g4:heart-and-blood:heart}{cœur} et les
\omterm{def:g4:breathing:organs}{poumons} sont les deux moitiés d'une même chaîne
d'approvisionnement.
\end{example}

\begin{method}[Prendre un pouls]\label{met:g4:heart-and-blood:pulse}
\begin{enumerate}
  \item Retourne une main paume vers le haut~; pose deux doigts de
        l'autre main sur le \omterm{def:g1:my-body:joint}{poignet}, juste sous la base du pouce ---
        jamais le pouce, qui a son propre petit \omterm{prop:g4:heart-and-blood:pulse}{pouls}~;
  \item appuie doucement jusqu'à sentir les coups~;
  \item compte les coups pendant une minute (une montre à secondes
        aide)~;
  \item note le nombre, avec ce que le corps était en train de
        faire~: repos, marche, après un sprint.
\end{enumerate}
\end{method}

\begin{remark}[Un muscle qui ne se repose jamais --- ou presque]\label{rem:g4:heart-and-blood:rest}
Entre deux \omterm{def:g4:heart-and-blood:heart}{battements}, le \omterm{def:g4:heart-and-blood:heart}{cœur} se relâche pendant une fraction de
seconde --- ce court instant est tout le repos qu'il prend jamais.
Jour et nuit, environ $100000$ \omterm{def:g4:heart-and-blood:heart}{battements} par jour, toute une vie.
Comme tout \omterm{def:g3:bones-and-muscles:muscle}{muscle} (\cref{met:g3:bones-and-muscles:care}), il devient
plus fort avec l'exercice~: un \omterm{def:g4:heart-and-blood:heart}{cœur} entraîné déplace plus de \omterm{prop:g4:journey-of-food:absorb}{sang} à
chaque \omterm{def:g4:heart-and-blood:heart}{battement}, et peut donc battre plus tranquillement.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:1}
Nomme deux choses que le \omterm{prop:g4:journey-of-food:absorb}{sang} livre, et où il recueille chacune.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:2}
Quel gaz de rebut le \omterm{prop:g4:journey-of-food:absorb}{sang} emporte-t-il, et où le dépose-t-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:3}
De quoi le \omterm{def:g4:heart-and-blood:heart}{cœur} est-il fait, quelle est sa taille, et où se
trouve-t-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:4}
Que fait chaque côté du \omterm{def:g4:heart-and-blood:heart}{cœur}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:5}
Qu'est-ce que le \omterm{prop:g4:heart-and-blood:pulse}{pouls}, et que t'apprend le fait de le compter~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:6}
Pourquoi faut-il prendre le \omterm{prop:g4:heart-and-blood:pulse}{pouls} avec les doigts et non avec le
pouce~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:7}
Pourquoi le \omterm{prop:g4:heart-and-blood:pulse}{pouls} s'emballe-t-il pendant un sprint~? Raconte
l'histoire du point de vue des \omterm{def:g3:bones-and-muscles:muscle}{muscles} des \omterm{def:g1:my-body:parts}{jambes}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:8}
À toi d'essayer~: prends ton \omterm{prop:g4:heart-and-blood:pulse}{pouls} au repos avec
\cref{met:g4:heart-and-blood:pulse}, fais trente secondes de sauts
étoilés, et reprends-le. Donne les deux nombres.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:heart-and-blood:9}
Dans \cref{ex:g4:heart-and-blood:round}, la goutte de \omterm{prop:g4:journey-of-food:absorb}{sang} est rouge
vif en quittant les \omterm{def:g4:breathing:organs}{poumons} et plus foncée en quittant le \omterm{def:g3:bones-and-muscles:muscle}{muscle} de
la \omterm{def:g1:my-body:parts}{jambe}. Qu'a-t-il changé en elle, à chaque fois~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:heart-and-blood:10}
Le \omterm{prop:g4:journey-of-food:absorb}{sang} relie trois organes de ton année~: l'\omterm{def:g4:journey-of-food:tract}{intestin grêle}, les
\omterm{def:g4:breathing:organs}{poumons}, le \omterm{def:g4:heart-and-blood:heart}{cœur}. Dessine ou décris le réseau~: qu'apporte chacun au
service de livraison~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:heart-and-blood:11}
Le \omterm{prop:g4:heart-and-blood:pulse}{pouls} au repos d'un cycliste entraîné peut être de $50$~; celui
d'un adulte non entraîné, de $75$. À l'aide de
\cref{rem:g4:heart-and-blood:rest}, explique comment \emph{plus lent}
peut vouloir dire \emph{plus fort} --- que doit faire chacun des
\omterm{def:g4:heart-and-blood:heart}{battements} du cycliste~?
\end{exercise}
