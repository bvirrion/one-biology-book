\section*{Bab \ref{ch:g4:journey-of-food} --- Perjalanan Makanan}

\begin{solution}{exo:g4:journey-of-food:1}
Mulut, \omterm{def:g4:journey-of-food:tract}{kerongkongan}, \omterm{def:g4:journey-of-food:tract}{lambung}, \omterm{def:g4:journey-of-food:tract}{usus halus}, \omterm{def:g4:journey-of-food:tract}{usus besar} --- lalu sisanya
meninggalkan tubuh.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:2}
Ia diaduk oleh dinding berotot \omterm{def:g4:journey-of-food:tract}{lambungnya} dan dicampur dengan \omterm{def:g4:journey-of-food:digestion}{getah
pencernaan} --- selama beberapa jam.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:3}
\omterm{def:g4:journey-of-food:digestion}{Pencernaan} adalah perubahan makanan, di sepanjang salurannya, menjadi
bagian-bagian yang cukup sederhana dan cukup kecil untuk masuk ke
dalam tubuh. Bagian-bagian itulah \omterm{def:g4:journey-of-food:digestion}{zat gizinya}.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:4}
Di \omterm{def:g4:journey-of-food:tract}{usus halus}, melalui dindingnya yang tipis dan terlipat, ke dalam
\omterm{prop:g4:journey-of-food:absorb}{darah}. Apa yang tidak dapat menyeberang meneruskan perjalanan ke \omterm{def:g4:journey-of-food:tract}{usus
besar}, kehilangan airnya di situ, lalu meninggalkan tubuh sebagai
tinja.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:5}
\omterm{def:g4:journey-of-food:tract}{Air liur} sudah membongkar rotinya menjadi bagian yang lebih sederhana
dan lebih manis --- \omterm{def:g4:journey-of-food:digestion}{pencernaan} bermula di dalam mulut, dan rasa
manisnya adalah rasa \omterm{def:g4:journey-of-food:digestion}{pencernaan} itu.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:6}
Karena menyeberang ke dalam \omterm{prop:g4:journey-of-food:absorb}{darah} membutuhkan luas dinding:
meter-meter panjangnya dan lipatan dalam di sisi dalamnya mengemas
permukaan penyeberangan yang amat luas --- kalau dibentangkan datar,
kira-kira sehampar karpet kamar kecil --- ke dalam perut.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:7}
Misalnya: mengunyah dengan benar (\omterm{def:g4:journey-of-food:tract}{lambung} tidak punya gigi); makan
dengan tenang pada waktu yang teratur (salurannya bekerja paling baik
tanpa tergesa); minum air (\omterm{def:g4:journey-of-food:digestion}{pencernaan} berjalan dalam keadaan basah);
makan serat (serat menjaga sisanya tetap bergerak); beristirahat
sesudah santapan besar (\omterm{def:g4:journey-of-food:tract}{lambung} dan \omterm{def:g1:my-body:parts}{tungkai} bersaing memperebutkan
tenaga tubuh).
\end{solution}

\begin{solution}{exo:g4:journey-of-food:8}
Jawaban terbuka. Biasanya jauh di bawah setengah jam di dalam mulut
--- melawan satu sampai dua hari bagi sisa perjalanannya. Bagian
mulutnya kecil dalam hal waktu dan menentukan dalam hal pengaruh: ia
satu-satunya perhentian yang bergigi.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:9}
Yang dilanggar: mengunyah dengan benar (makan siangnya dilahap
setengah terkunyah) dan beristirahat sesudah makan (lari cepat tepat
sesudahnya). Besok: makanlah pelan-pelan, dan biarkan santapannya
mengendap sebelum berlomba.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:10}
Makanan yang ditelan masih berada di dalam pipanya --- seperti lubang
sebuah donat, belum sungguh-sungguh berada di dalam tubuh. Ia harus
dicerna lebih dahulu menjadi \omterm{def:g4:journey-of-food:digestion}{zat gizi}; hanya di usus haluslah \omterm{def:g4:journey-of-food:digestion}{zat gizi}
itu menyeberangi dindingnya ke dalam \omterm{prop:g4:journey-of-food:absorb}{darah}. Masuk ke mulut bukan
berarti masuk ke tubuh.
\end{solution}

\begin{solution}{exo:g4:journey-of-food:11}
\omterm{def:g4:journey-of-food:digestion}{Pencernaan} harus membongkar makanan menjadi \omterm{def:g4:journey-of-food:digestion}{zat gizi}, dan rumput jauh
lebih sukar dibongkar daripada roti --- seratnya liat, kebaikannya
sedikit pada tiap suapan
(\cref{prop:g2:what-animals-eat:daily}). Karena itu si pemakan rumput
membawa permesinan yang lebih berat: \omterm{def:g2:teeth-and-eating-well:kinds}{gigi geraham} penggilas yang
dipakai dua kali (memamah biak), empat bilik \omterm{def:g4:journey-of-food:tract}{lambung} untuk pengolahan
yang lama, dan penolong di dalam usus yang dapat memecah apa yang
tidak dapat dipecah getahnya sendiri. Masukan yang lebih liat, pabrik
yang lebih besar.
\end{solution}
