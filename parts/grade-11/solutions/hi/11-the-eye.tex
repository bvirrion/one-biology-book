\section*{अध्याय \ref{ch:g11:the-eye} --- आँख और उसके प्रकाशग्राही}

\begin{solution}{exo:g11:the-eye:1}
कॉर्निया, पुतली (परितारिका के छेद से), लेंस, काचाभ द्रव, और फिर दृष्टिपटल
की पारदर्शी परतें (गुच्छिका तथा द्विध्रुवी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}) पार करके पीछे के
\omterm{def:g11:the-eye:photoreceptors}{प्रकाशग्राही}।
\end{solution}

\begin{solution}{exo:g11:the-eye:2}
\omterm{def:g11:the-eye:photoreceptors}{शलाकाएँ}: क़रीब 12 करोड़, बहुत संवेदनशील, एक वर्णक, और धूसर में रात की
दृष्टि। \omterm{def:g11:the-eye:photoreceptors}{शंकु}: क़रीब 60 लाख, तेज़ प्रकाश माँगते हैं, तीन वर्णक, दिन की तथा
रंग की दृष्टि, और दृष्टि-गर्तिका पर सबसे तीखी।
\end{solution}

\begin{solution}{exo:g11:the-eye:3}
दृष्टि-गर्तिका दृष्टिपटल का वह केंद्रीय गड्ढा है जो केवल ठसाठस भरे \omterm{def:g11:the-eye:photoreceptors}{शंकुओं}
से बना है और जहाँ दृष्टि सबसे तीखी है। और अंध बिंदु वह जगह है जहाँ से दृक्
तंत्रिका निकलती है और जहाँ कोई \omterm{def:g11:the-eye:photoreceptors}{प्रकाशग्राही} नहीं।
\end{solution}

\begin{solution}{exo:g11:the-eye:4}
S \omterm{def:g11:the-eye:photoreceptors}{शंकु} लगभग 45\%, \omterm{def:g11:the-eye:photoreceptors}{शलाकाएँ} लगभग 60\%, M \omterm{def:g11:the-eye:photoreceptors}{शंकु} लगभग 20\%, और L \omterm{def:g11:the-eye:photoreceptors}{शंकु} लगभग
10\%।
\end{solution}

\begin{solution}{exo:g11:the-eye:5}
X \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} पर; और यह स्थिति X-सहलग्न \omterm{def:g11:genes-and-disease:genetic}{अप्रभावी} है: वह \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोते हर पुरुष
में व्यक्त, और स्त्रियों में केवल दो प्रतियों के साथ, इसलिए वह पुरुषों में
कहीं अधिक आम है।
\end{solution}

\begin{solution}{exo:g11:the-eye:6}
रात में केवल \omterm{def:g11:the-eye:photoreceptors}{शलाकाएँ} जवाब देती हैं, और उनका अकेला एक वर्णक है: वे चमक बताती
हैं, तरंगदैर्घ्य नहीं, इसलिए कोई रंग नहीं। दृष्टि-गर्तिका में केवल \omterm{def:g11:the-eye:photoreceptors}{शंकु}
हैं, जो उस तारे के लिए बहुत असंवेदनशील हैं; और बग़ल से देखने पर उस तारे का
प्रतिबिंब \omterm{def:g11:the-eye:photoreceptors}{शलाकाओं} पर पड़ता है, जो उसे पकड़ लेती हैं।
\end{solution}

\begin{solution}{exo:g11:the-eye:7}
वही रंग: दोनों में S:M:L के अनुपात 0:0.6:1 हैं। पर चमक अलग: प्रकाश 2 हर
\omterm{def:g11:the-eye:photoreceptors}{शंकु} को आधा ही उकसाता है। रंग वह अनुपात है; और चमक वह कुल।
\end{solution}

\begin{solution}{exo:g11:the-eye:8}
बेटों को अपना X माता से मिलता है: इसलिए सब सामान्य। बेटियों को पिता का X
मिलता है: इसलिए सब वाहक, और दृष्टि सामान्य। और हर वाहक बेटी के बेटे आधी
प्रायिकता से वर्णांध होते हैं।
\end{solution}

\begin{solution}{exo:g11:the-eye:9}
किसी पुरुष को एक उत्परिवर्ती X चाहिए, और किसी स्त्री को दो। यदि उस \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी}
की बारंबारता 0.08 हो, तो कोई स्त्री दो $0.08^2 = 0.0064$ प्रायिकता से ढोती
है, यानी लगभग 0.6\% --- जो 8\% से सोलह गुना कम है।
\end{solution}

\begin{solution}{exo:g11:the-eye:10}
किसी द्विगुणन के बाद बीते समय के साथ फ़र्क़ जमा होते जाते हैं: L और M के
बीच 4\% का मतलब है कोई हालिया नक़ल; और इनमें से किसी का S से 57\% का मतलब
है कोई कहीं पुरानी। इसलिए L/M द्विगुणन सबसे हालिया घटना है, और S का अलगाव
उससे कहीं पहले का।
\end{solution}

\begin{solution}{exo:g11:the-eye:11}
पुरानी दुनिया के नरवानरों के पूर्वज ने अपने अकेले X-सहलग्न \omterm{def:g11:the-eye:photoreceptors}{ऑप्सिन} \omterm{def:g10:universal-dna:gene}{जीन} का
द्विगुणन किया; और एक नक़ल ने अपनी चोटी लंबे तरंगदैर्घ्यों की ओर खिसका ली,
जिससे तीसरा शंकु-प्रकार और रंग की एक नई विमा मिली। वह शायद इसलिए बचाकर रखा
गया कि उसने पत्तों के बीच पके फल और नई कोंपलें ढूँढ़ने में मदद की।
\end{solution}

\begin{solution}{exo:g11:the-eye:12}
नीले और पीले (और हरे में से नीला-हरा)। S \omterm{def:g10:universal-dna:gene}{जीन} किसी साधारण \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} पर है,
इसलिए दोनों में से किसी भी लिंग में दोनों \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} उत्परिवर्ती होने चाहिए ---
यानी विरला और लिंग से स्वतंत्र।
\end{solution}

\begin{solution}{exo:g11:the-eye:13}
$\num{2.5e-6}/\num{17e-3} \approx \num{1.5e-4}\,\unit{rad}$, यानी चाप का
लगभग आधा मिनट। और \qty{10}{m} पर: $\num{1.5e-4} \times 10 =
\qty{1.5}{mm}$।
\end{solution}

\begin{solution}{exo:g11:the-eye:14}
किसी नर के पास एक X है, इसलिए एक \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी}, इसलिए M/L प्रकार का एक ही \omterm{def:g11:the-eye:photoreceptors}{शंकु}:
यानी S समेत दो प्रकार। और जिस मादा के पास दो अलग \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} हों वह कुछ \omterm{def:g11:the-eye:photoreceptors}{शंकुओं}
में एक और बाक़ी में दूसरा व्यक्त करती है: यानी तीन प्रकार, और इसलिए तीन-रंग
की दृष्टि; जबकि दो एक जैसे \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} वाली मादा के पास किसी नर जैसे दो ही
प्रकार होते हैं।
\end{solution}

\begin{solution}{exo:g11:the-eye:15}
गुच्छिका \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} और दृक् तंत्रिका सही-सलामत हैं, इसलिए उन तक पहुँचाई गई
विद्युत धड़कनें दृष्टि-मस्तिष्क तक संकेतों की तरह पहुँच जाती हैं। गुणवत्ता
को इलेक्ट्रोडों की संख्या सीमित करती है (सैकड़ों, जबकि दस लाख तंतु और लाखों
ग्राही होते हैं) और दृष्टिपटल की अपनी छँटाई का खो जाना भी।
\end{solution}

\begin{solution}{pb:g11:the-eye:1}
\textbf{1.} \qty{470}{nm}: S 40, M 35, L 18. \qty{530}{nm}: S 0, M 100,
L 80. \qty{590}{nm}: S 0, M 40, L 90. \qty{650}{nm}: S 0, M 2, L 12.

\textbf{2.} \qty{470}{nm} तीनों को उकसाता है; और \qty{650}{nm} असल में
केवल L को।

\textbf{3.} केवल S और M के साथ: \qty{590}{nm} (0, 40) देता है और
\qty{650}{nm} (0, 2); जबकि \qty{530}{nm} (0, 100) देता है। इसलिए
\qty{590}{nm} (नारंगी) और \qty{530}{nm} (हरा) प्रकाश केवल M की तीव्रता में
अलग हैं, यानी किसी मंद और किसी चमकीले हरे जैसे, और वही मुश्किल जोड़ा हैं;
और \qty{650}{nm} किसी बहुत मंद प्रकाश जैसा दिखता है।

\textbf{4.} केवल S और L के साथ: \qty{530}{nm} (0, 80) देता है और
\qty{590}{nm} (0, 90) --- यानी लगभग एक जैसे: इसलिए हरा और नारंगी गड्डमड्ड
हो जाते हैं।

\textbf{5.} दोनों लाल प्रकाश अपने बचे हुए शंकु-प्रकार से पकड़ तो लेते हैं;
जो खोता है वह M और L के जवाबों के बीच की तुलना है, और वही लाल को हरे से
अलग करती है।

\textbf{6.} $X^NX^c$: उसके पिता ने उसे अपना अकेला X दिया, जो वह
उत्परिवर्ती \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोता है; और उसकी सामान्य दृष्टि दिखाती है कि दूसरा X
सामान्य है।

\textbf{7.} बेटे: आधे-आधे $X^NY$ सामान्य या $X^cY$ वर्णांध। बेटियाँ:
आधी-आधी $X^NX^N$ या $X^NX^c$, और दोनों की दृष्टि सामान्य (दूसरी वाली
वाहक)।

\textbf{8.} इनेस $1/2$ प्रायिकता से वाहक है। कोई बेटी तभी वर्णांध होगी जब
उसे दोनों जनकों से $X^c$ मिले: पिता से तो पक्के तौर पर, और इनेस से
$1/2 \times 1/2$ प्रायिकता से: यानी $1/4$।

\textbf{9.} किसी वर्णांध बेटी के पास दो उत्परिवर्ती X \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} होते हैं,
हर जनक से एक; और पिता अपना अकेला X हर बेटी को देता है, इसलिए उसका X
उत्परिवर्ती है और वह ख़ुद वर्णांध है।

\textbf{10.} हाँ: उनके बेटे को अपना X उन्हीं से मिला, और वह उत्परिवर्ती
\omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोता है; और सामान्य दृष्टि के साथ वे $X^NX^c$ रही होंगी।

\textbf{11.} पहले रोडॉप्सिन अलग होता है; फिर S, L/M रेखा से अलग होता है;
और L तथा M सबसे आख़िर में बँटते हैं: यानी (रोडॉप्सिन, (S, (L, M)))।

\textbf{12.} 3.5 करोड़ साल के लिए 4\% का फ़र्क़; इसलिए 57\% में लगभग
$35 \times 57/4 \approx 500$ लाख साल, यानी 50 करोड़ साल लगते --- जो
\omterm{def:g10:common-ancestry:bodyplan}{कशेरुकियों} के शुरुआती उद्गम से संगत है (बड़े फ़र्क़ों पर यह संबंध ठीक-ठीक
समानुपाती नहीं है, इसलिए यह परिमाण की कोटि भर है)।

\textbf{13.} अग़ल-बग़ल पड़े वे दो लगभग एक जैसे \omterm{def:g10:universal-dna:gene}{जीन} दोनों X \omterm{def:g10:universal-dna:information}{गुणसूत्रों} के
जुड़ने के दौरान ग़लत सीध में बैठ सकते हैं; और तब कोई अदला-बदली एक X तीन
प्रतियों वाला और एक अकेली प्रति वाला दे देती है। जिस नर को वह एक-प्रति वाला
X मिले उसके पास M/L प्रकार का केवल एक \omterm{def:g11:the-eye:photoreceptors}{शंकु} होता है: यानी लाल--हरी
वर्णांधता।

\textbf{14.} जुड़ना अनुक्रम की समानता पर टिका है: इसलिए अग़ल-बग़ल पड़े 96\%
एक जैसे दो \omterm{def:g10:universal-dna:gene}{जीन} बेसुरे ढंग से, यानी \omterm{def:g10:universal-dna:gene}{जीन} 1 \omterm{def:g10:universal-dna:gene}{जीन} 2 के साथ, जुड़ सकते हैं और
असमान अदला-बदली कर सकते हैं।

\textbf{15.} वह एक ही बार हुआ, यानी पुरानी दुनिया के नरवानरों के साझा
पूर्वज में, दूसरे स्तनधारियों तथा नई दुनिया के बंदरों से उनके अलग होने के
बाद, क़रीब 3.5 करोड़ साल पहले, और वह उनके सारे वंशजों को विरासत में मिला।

\textbf{16.} $\num{2.5e-6}/\num{17e-3} \approx \num{1.5e-4}$ रेडियन,
यानी लगभग $0.5'$।

\textbf{17.} $\num{1.5e-4} \times 6 \approx \qty{0.9}{mm}$: यानी
\qty{1.7}{mm} की उन लकीरों से महीन, और इसीलिए सामान्य दृष्टि वह पंक्ति
आराम से पढ़ लेती है (व्यवहार में प्रकाशीय धुँधलापन उस सीमा को
$1'$ के पास ले आता है)।

\textbf{18.} कम से कम दस गुना मोटा (100 \omterm{def:g11:the-eye:photoreceptors}{शलाकाएँ} एक ही संकेत बाँटती हैं,
यानी क़रीब 10 गुणा 10): यानी चाप के कई मिनट; और बदले में सौ ग्राही अपना
प्रकाश जोड़ लेते हैं तथा कहीं मंद स्रोत भी पकड़ लेते हैं।

\textbf{19.} दोगुना घनत्व यानी $\sqrt{2}$ गुना क़रीब की दूरी, और बड़ी \omterm{def:g11:the-eye:parts}{आँख}
लंबी फ़ोकस दूरी जोड़ देती है: यानी लगभग दोगुना महीन।

\textbf{20.} S, M और L \omterm{def:g11:the-eye:photoreceptors}{शंकु}, जिनमें से हर एक अपनी चोटी से दूर स्पेक्ट्रम
के सिरे के प्रति लगभग अंधा है; X-सहलग्न \omterm{def:g11:the-eye:photoreceptors}{ऑप्सिन} \omterm{def:g10:universal-dna:gene}{जीन} का 3.5 करोड़ साल पहले
हुआ एक अनुक्रमित द्विगुणन, जिसने नरवानरों को तीसरा दिया; और
दृष्टि-गर्तिका चाप का लगभग आधा मिनट विभेदित कर लेती है।
\end{solution}
