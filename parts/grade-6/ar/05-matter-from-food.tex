\chapter{المادة من الغذاء}\label{ch:g6:matter-from-food}

عجل وزنه \qty{40}{kg} يصير في سنتين بقرة وزنها
\qty{600}{kg}. نصف طنّ من البقرة الجديدة --- لحمًا وعظمًا وجلدًا
وقرنًا --- ظهر من مكان ما. أكلت البقرة العشب؛ والمادة الجديدة
\emph{ليست عشبًا}؛ ومع ذلك لولا العشب لما وُجد شيء منها. وما تفعله
\omterm{def:g1:animals-around-us:animal}{الحيوانات} بغذائها هو شأن هذا الفصل --- جانب \omterm{def:g5:food-webs:roles}{المستهلِكين} من دفتر
حسابات \cref{ch:g6:plants-are-producers}.

\section{الحيوانات تبني مادتها من الغذاء}

\begin{proposition}[المستهلِكون يحوّلون المادة العضوية]\label{prop:g6:matter-from-food:transform}
يبني \omterm{def:g1:animals-around-us:animal}{الحيوان} مادته العضوية الخاصة --- عضلًا وعظمًا وشحمًا وريشًا
\omterm{def:g1:animals-around-us:covering}{وصدفة} --- من \emph{\omterm{def:g6:plants-are-producers:matter}{المادة العضوية} التي يأكلها}. \omterm{def:g4:journey-of-food:digestion}{فالهضم}
(\cref{def:g4:journey-of-food:digestion}) يفكّك الغذاء إلى \omterm{def:g4:journey-of-food:digestion}{مغذيات}؛
\omterm{prop:g4:journey-of-food:absorb}{والدم} يوصّلها (\cref{prop:g4:heart-and-blood:carries})؛ \omterm{def:g5:first-look-at-cells:cell}{وخلايا}
الجسم (\cref{prop:g5:first-look-at-cells:growth}) تعيد تركيبها
مادةً خاصة \omterm{def:g1:animals-around-us:animal}{بالحيوان}. \omterm{def:g1:animals-around-us:animal}{فالحيوانات} \emph{محوِّلة} للمادة، لا صانعة لها
من المعدني قط: فتلك تجارة \omterm{def:g5:food-webs:roles}{المنتِجين} وحدهم.
\end{proposition}
\admitted

\begin{example}[من العشب إلى البقرة]\label{ex:g6:matter-from-food:cow}
تتبّع نصف الطنّ: البقرة ترعى ساعة بعد ساعة
(وقد شرحت \cref{prop:g2:what-animals-eat:daily} طول الساعات)؛
وهضمها الثقيل (\cref{exo:g4:journey-of-food:11}) يفكّك العشب إلى
\omterm{def:g4:journey-of-food:digestion}{مغذيات}؛ ودمها يحملها إلى الضرع والعضل والعجل النامي في داخلها؛
وخلاياها تبني مادة بقرة من مادة عشب. فالمرعى، حرفيًّا، يمشي في
صورة جديدة.
\end{example}

\begin{remark}[المادة تغيّر أصحابها لا طبيعتها]\label{rem:g6:matter-from-food:hands}
عند كل سهم من أسهم \omterm{def:g3:food-chains:chain}{السلسلة الغذائية}
(\cref{def:g3:food-chains:chain}) تغيّر \omterm{def:g6:plants-are-producers:matter}{المادة العضوية} أصحابها
ويُعاد بناؤها --- مادة عشب إلى مادة بقرة، ومادة بقرة (\omterm{def:g2:animals-grow-up:born}{الحليب}،
واللحم) إلى مادة طفل. و\emph{العضوي} لا يأتي من العدم قط:
\omterm{def:g5:food-webs:roles}{فالمستهلِكون} يمرّرونه ويعيدون تشكيله. ولم يُصنع جديدًا إلا عند بداية
السلسلة الخضراء
(\cref{prop:g6:plants-are-producers:firstlink})؛ ولا يعود معدنيًّا
إلا عند نهاية المحلِّلين في الفصل التالي.
\end{remark}

\section{النمو إنتاج للمادة}

\begin{definition}[إنتاج المادة عند الحيوان]\label{def:g6:matter-from-food:production}
\emph{إنتاج المادة}\index{إنتاج المادة} عند \omterm{def:g1:animals-around-us:animal}{الحيوان} هو \omterm{def:g6:plants-are-producers:matter}{المادة
العضوية} الجديدة التي يبنيها: نمو الصغير (كيلوغرامات العجل
المضافة)، والتجديدات السنوية (\omterm{def:g1:animals-around-us:covering}{فرو} يُطرح ويعود، وقرون، \omterm{def:g1:animals-around-us:covering}{وريش})،
والإصلاحات (\omterm{def:g3:bones-and-muscles:skeleton}{كالعظم} الملتئم في
\cref{rem:g3:bones-and-muscles:alive})، والمنتجات (\omterm{def:g2:animals-grow-up:born}{حليب}، \omterm{def:g2:animals-grow-up:born}{وبيض}،
وصوف). وكل ذلك غذاء أُعيد بناؤه.
\end{definition}

\begin{example}[وزن الإنتاج]\label{ex:g6:matter-from-food:weighing}
يزن المزارعون الإنتاج كل يوم: فبقرة حلوب تعطي نحو
\qty{25}{kg} من \omterm{def:g2:animals-grow-up:born}{الحليب} في اليوم --- وهي \omterm{def:g6:plants-are-producers:matter}{مادة عضوية} عليها أن تعيد
بناءها من العلف يومًا بعد يوم؛ \omterm{def:g2:animals-grow-up:born}{وبيضة} الدجاجة نحو
\qty{60}{g} من المؤن المحزومة
(\cref{rem:g2:animals-grow-up:egg})، تُركَّب في ليلة من حبّها. فلا
علف، فلا إنتاج: والدفتر يوازن بلا رحمة.
\end{example}

\begin{proposition}[لا يصير كل الغذاء جسمًا]\label{prop:g6:matter-from-food:losses}
لا يحوّل المستهلِك غذاءه كله مادةً جديدة أبدًا:
\begin{itemize}
  \item فجزء لا يُمتصّ البتة، ويغادر فضلات
        (\cref{prop:g4:journey-of-food:absorb})؛
  \item وجزء كبير \emph{يُحرَق} --- إذ يُجمع مع أكسجين النفَس
        (\cref{prop:g4:breathing:exchange}) --- لتشغيل الحركة
        والدفء: مادة تُنفق وقودًا وتغادر غازًا مزفورًا؛
  \item والباقي وحده يصير نموًّا ومنتجات.
\end{itemize}
فمن العشب الذي تأكله البقرة لا ينتهي بقرةً جديدة إلا كسر متواضع
--- ولهذا كان كل طابق من هرم
\cref{prop:g5:food-webs:pyramid} أصغر بكثير من الذي تحته.
\end{proposition}
\admitted

\begin{example}[الهرم مشروحًا أخيرًا]\label{ex:g6:matter-from-food:pyramid}
أطنان عشب مرج تغذّي قطيعًا من الأرانب؛ وكيلوغرامات الأرانب تغذّي
ثعلبًا أو ثعلبين. وعند كل سهم تُحرق أكثر المادة للعيش أو تُفقد،
ولا يُمرَّر إلى أعلى جسمًا إلا كسر. وكبار الصائدين نادرون
(\cref{prop:g5:food-webs:pyramid}) لأنهم يعيشون على
\emph{بقايا البقايا} --- حسابًا لا سوء حظ.
\end{example}

\section{موائد مختلفة وتجارة واحدة}

\begin{proposition}[موائد مختلفة والتحويل نفسه]\label{prop:g6:matter-from-food:menus}
\omterm{def:g2:what-animals-eat:kinds}{العاشب} \omterm{def:g2:what-animals-eat:kinds}{واللاحم} \omterm{def:g2:what-animals-eat:kinds}{والقارت}
(\cref{def:g2:what-animals-eat:kinds}) أسماء \emph{لمصادر} مختلفة
\omterm{def:g6:plants-are-producers:matter}{للمادة العضوية}، لا لتجارات مختلفة. فالبقرة التي تعيد بناء العشب
والثعلب الذي يعيد بناء الأرنب كلاهما يجري تحويل
\cref{prop:g6:matter-from-food:transform} --- بعتاد يناسب المائدة
(\cref{prop:g2:what-animals-eat:match}،
\cref{exo:g4:journey-of-food:11}): فالعشب يقتضي طواحن وهضمًا
هائلًا، واللحم يكافئ أسنان الإمساك ومعًى قصيرًا.
\end{proposition}
\admitted

\begin{method}[تدقيق مادة حيوان]\label{met:g6:matter-from-food:audit}
لأي \omterm{def:g1:animals-around-us:animal}{حيوان}، حرّر دفتر مادته:
\begin{enumerate}
  \item \emph{الداخل}: أي \omterm{def:g6:plants-are-producers:matter}{مادة عضوية} يأكل، وممن؟ (أي أسهم شبكته
        الغذائية الداخلة،
        \cref{def:g5:food-webs:web})؛
  \item و\emph{المبني}: أي نمو وتجديدات ومنتجات يصنع؟
  \item و\emph{الخارج}: الفضلات، والمادة المحروقة للحركة
        والدفء؛
  \item وراجع الميزان: فالإنتاج لا بد أن يكون الباقي المتواضع
        --- فإن جعله زعم أكثر من ذلك (``هذه البركة تغلّ من مادة
        السمك أكثر مما تنمّيه من غذاء'')، فالزعم خاطئ.
\end{enumerate}
\end{method}

\begin{example}[التدقيق في العمل]\label{ex:g6:matter-from-food:audituse}
دقّق في الشحرور (\cref{ex:g2:what-animals-eat:winter}): الداخل ---
ديدان، وحشرات، وتوت؛ والمبني --- نموه هو، \omterm{def:g1:animals-around-us:covering}{وريش} كل سنة الجديد،
وحضنة \omterm{def:g2:animals-grow-up:born}{البيض}؛ والخارج --- فضلات على الجدار، والحصة المحروقة الكبيرة
التي شغّلت سنة من الطيران والغناء. ووزن غذاء الطائر المغرّد اليومي
هائل بالقياس إلى حجمه لأن الطيران يحرق حصة كبيرة من الدفتر بالضبط.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:1}
اذكر تجارة \omterm{def:g5:food-webs:roles}{المستهلِكين}، وقابلها بتجارة \omterm{def:g5:food-webs:roles}{المنتِجين} في جملة لكل
واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:2}
عدّد الخطوات التي تحوّل العشب عضلَ بقرة، مع تسمية الفصل الذي غطى
كل خطوة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:3}
أعطِ أربع صور من صور \omterm{def:g6:matter-from-food:production}{إنتاج المادة} عند \omterm{def:g1:animals-around-us:animal}{الحيوان}، مع مثال على كل
واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:4}
سمِّ وجهات غذاء المستهلِك الثلاث، وفق
\cref{prop:g6:matter-from-food:losses}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:5}
لماذا تحتاج البقرة الحلوب إلى العلف كل يوم لتواصل إعطاء \omterm{def:g2:animals-grow-up:born}{الحليب}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:6}
تحوي \omterm{def:g2:animals-grow-up:born}{البيضة} \qty{60}{g} من \omterm{def:g6:plants-are-producers:matter}{المادة العضوية}. فأين صُنعت --- في
الدجاجة أم قبلها؟ فكّك ذلك بعناية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:7}
استعمل \cref{prop:g6:matter-from-food:losses} لتشرح لماذا يضيق
هرم الغذاء عند كل طابق.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:8}
طبّق \cref{met:g6:matter-from-food:audit} على قطة أليفة: الداخل،
والمبني، والخارج --- ببندين على الأقل في كل سطر.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:9}
``الثعلب والبقرة يفعلان بالغذاء أشياء مختلفة.'' صحّح الجملة وفق
\cref{prop:g6:matter-from-food:menus} --- ما الذي يختلف، وما الذي
يتطابق؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:10}
ينمو طفل \qty{3}{kg} في سنة وهو يأكل مئات الكيلوغرامات من
الغذاء. فأين ذهب الباقي؟ أجب بالدفتر.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:11}
\omterm{def:g1:animals-around-us:covering}{صدفة} الحلزون \omterm{def:g6:plants-are-producers:matter}{مادة معدنية}، ومع ذلك عدّتها
\cref{ex:g6:plants-are-producers:sorting} صنعَ الحلزون. وفّق بين
الأمرين: ماذا فعل جسم الحلزون، ومن أي موادّ، ليبنيها؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:matter-from-food:12}
هكتار واحد من الأرض يستطيع أن يطعم بالبطاطا عددًا من الناس أكبر
مما يطعمه بلحم خنزير رُبّي على تلك البطاطا. اشرح ذلك وفق
\cref{prop:g6:matter-from-food:losses} --- إلى أين يذهب الفرق حين
تسلك البطاطا منعطف الخنزير؟
\end{exercise}

\section{مسألة: دفتر المزرعة}

\begin{problem}[{مسألة نهاية الأسبوع --- مزرعة صغيرة مدقَّقة،
من الحقل إلى المائدة}]\label{pb:g6:matter-from-food:1}
حيازة الجدة الصغيرة: مرعًى فيه بقرة واحدة، وقنّ فيه عشر دجاجات
تُعلف بالحبّ، وحوض خضر، ومطبخ الأسرة. وفي نهاية الأسبوع هذه، أنت
من يمسك دفتر مادتها.

\medskip\noindent\textbf{القسم الأول --- طابق \omterm{def:g5:food-webs:roles}{المنتِجين}.}
\begin{enumerate}
  \item عدّد منتِجي الحيازة. وما مدخلاتهم المعدنية، وما مصدر
        طاقتهم؟
  \item ينمّي المرعى نحو \qty{10000}{kg} من مادة العشب في الموسم.
        ففي \omterm{def:g1:plants-around-us:parts}{أوراق} مَن صُنع كل غرام منها؟
  \item تنثر الجدة روث القنّ على حوض الخضر. فأي قاعدة من قواعد
        \cref{prop:g5:people-and-nature:lasting} تحفظها، وما
        الذي يعيد الروث تموينه بالضبط؟
  \item ولماذا لا يمكن أن تُعلف البقرة أملاحًا معدنية وماءً
        مباشرة، توفيرًا للمرعى؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- طابق \omterm{def:g5:food-webs:roles}{المستهلِكين}.}
\begin{enumerate}[resume]
  \item تأكل البقرة نحو \qty{70}{kg} من العشب في اليوم وتعطي نحو
        \qty{20}{kg} من \omterm{def:g2:animals-grow-up:born}{الحليب}. سمِّ الوجهات الثلاث لكمية
        \qty{50}{kg} الناقصة.
  \item تحوّل الدجاجات الحبّ بيضًا. اكتب دفتر الدجاجة --- الداخل،
        والمبني، والخارج --- واذكر لماذا تبيض الدجاجة التي تطرح
        ريشها (وهي تجدد ريشها كله) بيضًا أقل مدةً من الزمن.
  \item يأخذ صقر كتكوتًا. مُدّ سلسلة الحيازة الغذائية إليه، واذكر
        عند أي حلقات \emph{صُنعت} المادة وعند أيها حُوّلت فحسب.
  \item تأكل الأسرة \omterm{def:g2:animals-grow-up:born}{البيض} \omterm{def:g2:animals-grow-up:born}{والحليب} والخضر. تتبّع \omterm{def:g6:plants-are-producers:matter}{المادة العضوية} في
        فطور واحد --- خبز، \omterm{def:g2:animals-grow-up:born}{وبيضة} مسلوقة، \omterm{def:g2:animals-grow-up:born}{وحليب} --- كلًّا منها إلى
        \omterm{def:g1:plants-around-us:parts}{الورقة} التي صُنعت فيها أول مرة.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- حساب
الطوابق.}
\begin{enumerate}[resume]
  \item تقول الجدة: ``المرعى يطعم بقرة واحدة؛ ولن يطعم أكثر من
        وجار صغير من الثعالب لو أكلت الثعالب أرانب تغذّت على
        عشبه.'' برّر تقديرها بخسائر كل طابق.
  \item حوض الخضر يطعم الأسرة مباشرة. مستعملًا منطق
        \cref{exo:g6:matter-from-food:12}، اشرح لماذا يطعم
        الحوض عددًا من الناس في المتر المربع أكبر مما يطعمه
        طريق المرعى فالبقرة \omterm{def:g2:animals-grow-up:born}{فالحليب}.
  \item في الجفاف يتوقف نمو العشب. تنبأ، بالترتيب، بما يحدث
        \omterm{def:g2:animals-grow-up:born}{لحليب} البقرة، ولوزنها، ولسطر ``المبني'' في الدفتر ---
        ولماذا لا تفيد شمس زائدة ما دام المطر مخفقًا
        (والمفتاح في \cref{pb:g6:plants-are-producers:1}،
        السؤال السابع).
  \item أغلق الدفتر بجملة لكل طابق: \omterm{def:g5:food-webs:roles}{المنتِجون}، \omterm{def:g5:food-webs:roles}{والمستهلِكون} ---
        وسمِّ من ما زال غائبًا عن الحساب (وهو موضوع الفصل
        التالي).
\end{enumerate}
\end{problem}
