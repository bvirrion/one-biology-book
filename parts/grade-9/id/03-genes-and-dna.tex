\chapter{Gen dan DNA}\label{ch:g9:genes-and-dna}

\omterm{def:g9:chromosomes:chromosomes}{Kromosomnya} mengangkut penentunya --- tetapi sebuah benang
belumlah sebuah teks. Apa, di sepanjang sebuah \omterm{def:g9:chromosomes:chromosomes}{kromosom},
yang \emph{merupakan} satu penentu? Dan benangnya terbuat
dari apa, sehingga sebuah \omterm{def:g6:cell-unit-of-life:parts}{inti sel} yang lebarnya seperseribu
milimeter dapat memuat petunjuk bagi seorang manusia utuh
(\cref{exo:g9:heredity-and-traits:12})? Dua nama
menjawabnya, dan keduanya yang paling termasyhur dalam
\omterm{def:g1:living-or-not:living}{biologi} modern: \omterm{def:g9:genes-and-dna:gene}{gen}, dan \omterm{def:g9:genes-and-dna:dna}{DNA}.

\section{Gen: penentunya, ditemukan tempatnya}

\begin{definition}[Gen dan alel]\label{def:g9:genes-and-dna:gene}
Sebuah \emph{gen}\index{gen} adalah satu penentu yang dibuat
tepat: sebuah \emph{ruas \omterm{def:g9:chromosomes:chromosomes}{kromosom}} yang mengangkut petunjuk
bagi satu keping pembangunan sebuah \omterm{def:g9:heredity-and-traits:traits}{sifat} --- satu gen bagi
permesinan golongan \omterm{prop:g4:journey-of-food:absorb}{darahnya}, yang lain bagi zat warna
\omterm{def:g1:the-five-senses:sense}{matanya}, ribuan demi ribuan di sepanjang ke-46 benangnya,
masing-masing pada alamatnya yang tetap. Sebuah gen dapat
hadir dalam beberapa versi --- \emph{alel}\index{alel}:
versi A, B dan O pada gen golongan \omterm{prop:g4:journey-of-food:absorb}{darahnya}, versi cokelat
dan biru pada gen warna \omterm{def:g1:the-five-senses:sense}{matanya}. Alamat sama, ejaan berbeda.
\end{definition}

\begin{proposition}[Dua salinan, dan apa arti memperlihatkan]\label{prop:g9:genes-and-dna:twocopies}
\omterm{def:g9:chromosomes:chromosomes}{Kromosomnya} hadir berpasangan
(\cref{prop:g9:chromosomes:counts}) --- sehingga setiap \omterm{def:g9:genes-and-dna:gene}{gen}
hadir dalam \emph{dua salinan}, satu pada tiap mitranya,
satu dari tiap orang tuanya. Kedua salinannya boleh \omterm{def:g9:genes-and-dna:gene}{alel}
yang sama atau berbeda; bila berbeda, yang satu dapat
\emph{diungkapkan} dan yang lain diangkut diam-diam --- dan
misteri album keluarganya terurai menjadi mekanisme:
\begin{itemize}
  \item pengangkutan-tanpa-memperlihatkan
        (\cref{ex:g9:heredity-and-traits:skip}): \omterm{def:g9:genes-and-dna:gene}{alel} yang
        diam menumpang pada mitra keduanya;
  \item anak bergolongan O dari orang tua A dan B
        (\cref{pb:g9:heredity-and-traits:1}): tiap orang tua
        mengangkut O secara diam, dan pembagian duanya
        membagikan kedua O itu bersamaan;
  \item anak bermata biru dari orang tua bermata cokelat:
        dua biru yang diam, bertemu --- satu peluang dari
        empat, bila kedua orang tuanya masing-masing
        mengangkut satu: pembagian duanya membagikan biru
        tiap orang tua dengan satu peluang dari dua, dan
        kedua pembagiannya saling bebas.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Golongan darahnya, dimesinkan sepenuhnya]\label{ex:g9:genes-and-dna:bloodgroups}
Satu \omterm{def:g9:genes-and-dna:gene}{gen}, tiga \omterm{def:g9:genes-and-dna:gene}{alel}, dua salinan tiap orang: A bersama A
atau O memperlihatkan golongan A; B bersama B atau O
memperlihatkan B; A bersama B memperlihatkan AB (keduanya
diungkapkan --- \omterm{def:g9:genes-and-dna:gene}{alel} tidak selalu dapat didiamkan); O
bersama O memperlihatkan O. Seluruh pola daftar donornya ---
termasuk setiap ``kejutan'' --- adalah hitungan pembagian
dua salinannya.
\end{example}

\section{DNA: teksnya tertulis dalam apa}

\begin{definition}[DNA]\label{def:g9:genes-and-dna:dna}
Secara kimiawi, sebuah \omterm{def:g9:chromosomes:chromosomes}{kromosom} adalah satu molekul
\emph{DNA}\index{DNA} yang teramat panjang dan menggulung
(beserta protein pengemasnya): sebuah untai ganda --- tangga
terpilin yang termasyhur itu --- yang anak tangganya adalah
pasangan empat huruf kimiawi. \emph{Urutan hurufnya di
sepanjang untainya itulah keterangannya}: sebuah \omterm{def:g9:genes-and-dna:gene}{gen} adalah
sepenggal urutan itu, seperti sebuah kalimat adalah
sepenggal halaman. Empat huruf, tiga miliar banyaknya dalam
satu tumpukan manusia --- cukup teks bagi seluruh perangkat
petunjuk sebuah pembangunan, tergulung di setiap \omterm{def:g6:cell-unit-of-life:parts}{inti selnya}.
\end{definition}

\begin{example}[Melihatnya sendiri]\label{ex:g9:genes-and-dna:extraction}
\omterm{def:g9:genes-and-dna:dna}{DNA} dapat disarikan di dapur: lumatkan sebuah pisang atau
sebutir bawang, tambahkan air garam dan sedikit sabun cuci
piring (\omterm{def:g6:cell-unit-of-life:parts}{membrannya} diuraikan --- selubung pada
\cref{def:g6:cell-unit-of-life:parts} adalah sasarannya),
saring, lalu lapiskan alkohol yang dingin di atasnya: pada
batasnya naiklah segumpal benang keputihan, yang dapat
digulung pada sebatang tusuk gigi. Benang itulah \omterm{def:g9:genes-and-dna:dna}{DNA} ---
molekul sungguhan bab ini, dalam sentimeter yang sungguhan.
\end{example}

\begin{proposition}[Apa yang dijelaskan struktur DNA]\label{prop:g9:genes-and-dna:explains}
Untai gandanya bukanlah hiasan --- tiap \omterm{def:g9:heredity-and-traits:traits}{sifatnya} menjawab
sebuah pertanyaan lama:
\begin{itemize}
  \item \emph{penyalinannya}
        (pembelahan setia pada
        \cref{prop:g5:first-look-at-cells:growth}): kedua
        untainya saling melengkapi huruf demi huruf,
        sehingga bila ditarik terpisah, masing-masing
        membangun ulang mitranya --- satu tangga menjadi dua
        tangga yang serupa, satu bagi tiap \omterm{def:g5:first-look-at-cells:cell}{sel} anaknya;
  \item \emph{\omterm{def:g9:genes-and-dna:gene}{alelnya}}: versi sebuah \omterm{def:g9:genes-and-dna:gene}{gen} adalah perbedaan
        ejaan yang kecil pada urutannya ---
        alamat yang sama, huruf yang berubah;
  \item \emph{mutasi}\index{mutasi}: penyalinannya nyaris
        sempurna, bukan sempurna --- sebuah huruf yang
        jarang tersalin keliru menciptakan sebuah \omterm{def:g9:genes-and-dna:gene}{alel}
        \emph{baru}. Sebagian besar netral, kadang
        merugikan, sesekali berguna: \omterm{prop:g9:genes-and-dna:explains}{mutasi} adalah asal
        ejaan yang segar, dan akhirnya asal semua \omterm{def:g9:genes-and-dna:gene}{alel} ---
        kebaruan mentah yang akan dibutuhkan \cref{ch:g9:how-species-change};
  \item \emph{pembacaannya}: sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} memakai sebuah \omterm{def:g9:genes-and-dna:gene}{gen}
        dengan membaca urutannya lalu membangun molekul
        kerja yang bersesuaian --- dan tiap jenis \omterm{def:g5:first-look-at-cells:cell}{sel}
        membaca pilihannya sendiri dari perpustakaannya:
        teka-teki pada \cref{exo:g9:chromosomes:12},
        terpecahkan seperti dijanjikan.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Satu sandi, seluruh kehidupan]\label{rem:g9:genes-and-dna:universal}
Fakta yang terdalam datang terakhir: \emph{setiap} \omterm{def:g6:exploring-our-environment:components}{makhluk
hidup} menuliskan \omterm{def:g9:genes-and-dna:gene}{gennya} dengan huruf \omterm{def:g9:genes-and-dna:dna}{DNA} yang sama lalu
membacanya dengan sandi yang sama --- pohon ek, \omterm{ex:g6:producing-our-food:bread}{ragi}, paus,
\omterm{def:g6:producing-our-food:fermentation}{mikroba} pada \cref{ch:g6:producing-our-food}, kamu. Inilah
\omterm{prop:g5:first-look-at-cells:principle}{kesatuan kehidupan}
(\cref{prop:g5:first-look-at-cells:principle}) pada
dasarnya, dan bukti terkuat sejauh ini bagi \omterm{def:g6:classification-kinship:tree}{pohon
kekerabatannya} (\cref{rem:g6:classification-kinship:implies}):
bahasa yang sama menghujahkan asal yang sama. Inilah juga
sebabnya rekayasa genetika dapat berjalan sama sekali ---
sebuah \omterm{def:g9:genes-and-dna:gene}{gen} yang dipindahkan antarspesies tetap dapat dibaca,
seperti sebuah kalimat yang difotokopi ke dalam buku lain.
\end{remark}

\begin{method}[Pembacaan tiga tingkatnya]\label{met:g9:genes-and-dna:levels}
Setiap pertanyaan \omterm{def:g9:heredity-and-traits:heredity}{pewarisan sifat} kini dapat dibaca pada
tiga tingkat --- berlatihlah berpindah-pindah:
\begin{enumerate}
  \item \emph{tingkat keluarganya}: siapa yang
        memperlihatkan, siapa yang mengangkut, siapa yang
        meneruskan (\cref{ch:g9:heredity-and-traits});
  \item \emph{tingkat \omterm{def:g9:chromosomes:chromosomes}{kromosomnya}}: benang mana, dibagi dua
        lalu dipulihkan, yang mengangkut pembagiannya
        (\cref{ch:g9:chromosomes});
  \item \emph{tingkat molekulnya}: \omterm{def:g9:genes-and-dna:gene}{gen} mana, \omterm{def:g9:genes-and-dna:gene}{alel} mana,
        ejaan apa --- dan, bagi sebuah \omterm{def:g9:heredity-and-traits:traits}{sifat} yang baru,
        \omterm{prop:g9:genes-and-dna:explains}{mutasi} apa.
\end{enumerate}
Satu kisah, tiga perbesaran; sebuah jawaban lengkap bila ia
berjalan pada ketiganya.
\end{method}

\section{Soal: Mesin Fotokopi yang Rusak}

\begin{problem}[{Soal akhir pekan --- satu gen diikuti dari
ejaannya sampai pohon keluarganya}]\label{pb:g9:genes-and-dna:1}
Ikutilah satu kasus (rekaan tetapi khas): sebuah \omterm{def:g9:genes-and-dna:gene}{gen} zat
warna yang \omterm{def:g9:genes-and-dna:gene}{alel} bekerjanya, P, membangun zat warna
kulit-dan-rambut; lama berselang, sebuah kekeliruan
penyalinan menciptakan \omterm{def:g9:genes-and-dna:gene}{alel} rusaknya, p --- salah eja sampai
tak terbaca, tidak membangun apa pun. Pengangkut P bersama p
tampak biasa; seorang anak yang dibagikan p dan p tidak
membangun zat warna sama sekali --- rambut dan \omterm{def:g1:the-five-senses:sense}{kulit} yang
sangat pucat, \omterm{def:g1:the-five-senses:sense}{mata} yang rapuh terhadap matahari (keadaan ini
ada pada banyak \omterm{def:g5:biodiversity:species}{spesies}).

\medskip\noindent\textbf{Bagian I --- Tingkat molekulnya.}
\begin{enumerate}
  \item Sebutkan asal p, menurut
        \cref{prop:g9:genes-and-dna:explains} --- apa yang
        secara ragawi terjadi, sekali, pada garis benih
        seorang nenek moyangnya?
  \item Mengapa P-bersama-p tampak biasa? Kata kosakata yang
        mana yang memerikan tumpangan p?
  \item Mengapa p-bersama-p mengubah pembangunannya? Apa
        yang hilang, dan dari jenis permesinan bab yang mana
        (\cref{def:g9:genes-and-dna:gene})?
  \item Keadaannya muncul pada banyak \omterm{def:g5:biodiversity:species}{spesies}. Apa yang
        diramalkan \cref{rem:g9:genes-and-dna:universal}
        tentang ejaan \omterm{def:g9:genes-and-dna:gene}{gen} zat warnanya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Tingkat \omterm{def:g9:chromosomes:chromosomes}{kromosom}
dan keluarganya.}
\begin{enumerate}[resume]
  \item Dua orang tua pengangkut (masing-masing P bersama
        p): sebutkan keempat pembagian yang sama mungkinnya
        bagi satu anak, dan peluang pembangunan p-bersama-p.
  \item Anak perempuan p-bersama-p mereka lahir dari dua
        orang tua yang tampak biasa. Misteri album yang mana
        (\cref{ex:g9:heredity-and-traits:skip}) yang baru
        saja berjalan dengan mekanisme penuh?
  \item Anak sang putri sendiri bersama seorang pasangan
        P-bersama-P: apa yang diangkut mereka semua, dan apa
        yang tidak diperlihatkan seorang pun dari mereka?
  \item Telusuri kendaraan \omterm{def:g9:genes-and-dna:gene}{alelnya} untuk satu penerusan:
        tumpangan p dari neneknya ke cucunya, benang dan
        \omterm{def:g8:sexual-reproduction:gametes}{gamet} demi benang dan \omterm{def:g8:sexual-reproduction:gametes}{gamet}.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Laporan tiga
tingkatnya.}
\begin{enumerate}[resume]
  \item Tuliskan kasusnya pada ketiga tingkat
        \cref{met:g9:genes-and-dna:levels}, satu kalimat
        masing-masing.
  \item Seorang teman sekelas bertanya mengapa ejaan yang
        rusak itu tidak sekadar ``diperbaiki hilang'' sejak
        lama. Jawablah dengan penampilan biasa pengangkutnya
        --- di mana p bersembunyi dari penghakiman atas
        akibatnya?
  \item Perlindungan matahari lebih penting bagi sang putri
        daripada bagi orang tua pengangkutnya. Hubungkan
        \omterm{def:g9:heredity-and-traits:traits}{sifat}, pasangan \omterm{def:g9:genes-and-dna:gene}{alel} dan \omterm{def:g6:exploring-our-environment:components}{lingkungannya} dalam satu
        kalimat (tenunan pada
        \cref{prop:g9:heredity-and-traits:sorting}).
  \item Tutuplah dengan gambaran mesin fotokopinya yang
        dibuat tepat: apa yang menyalin, apa yang sekali
        tersalin keliru, dan mengapa kekeliruannya kini
        disalin dengan setia.
\end{enumerate}
\end{problem}
