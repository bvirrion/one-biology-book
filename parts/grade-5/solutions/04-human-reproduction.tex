\section*{Chapter \ref{ch:g5:human-reproduction} --- Human Reproduction}

\begin{solution}{exo:g5:human-reproduction:1}
The \omterm{def:g4:animal-reproduction:sexual}{egg cell} from the mother, the \omterm{def:g4:animal-reproduction:sexual}{sperm cell} from the father;
their joining is \omterm{def:g5:flower-to-fruit:steps}{fertilization}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:2}
In the mother's \omterm{prop:g5:human-reproduction:plan}{womb}, for about nine months --- the \omterm{def:g5:human-reproduction:pregnancy}{pregnancy}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:3}
\omterm{def:g5:human-reproduction:pregnancy}{Embryo} first, then \omterm{def:g5:human-reproduction:pregnancy}{fetus} --- from about two months, once the
body plan is laid out.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:4}
Through the \omterm{prop:g5:human-reproduction:cord}{umbilical cord}: \omterm{def:g4:journey-of-food:digestion}{nutrients} and oxygen cross from
the mother's \omterm{prop:g4:journey-of-food:absorb}{blood} into the baby's, and the baby's wastes
cross back for the mother's body to deal with.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:5}
The place where your own \omterm{prop:g5:human-reproduction:cord}{umbilical cord} was attached, before
it was cut at \omterm{def:g2:born-grow-die:cycle}{birth}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:6}
The first cry fills the \omterm{def:g4:breathing:organs}{lungs} with air for the first time:
until that moment they had never breathed --- oxygen had
always arrived through the cord.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:7}
\omterm{def:g5:flower-to-fruit:steps}{Fertilization} inside the mother's body; a \omterm{def:g4:animal-reproduction:ovivivi}{viviparous} \omterm{def:g2:born-grow-die:cycle}{birth}
(the baby grows in the \omterm{prop:g5:human-reproduction:plan}{womb} and is \omterm{def:g2:animals-grow-up:born}{born alive}); \omterm{def:g2:animals-grow-up:born}{milk} feeding
afterward --- the mammal class features exactly.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:8}
Because whatever is in the mother's \omterm{prop:g4:journey-of-food:absorb}{blood} reaches the baby
through the cord --- smoke's and alcohol's harmful substances
included --- and a \omterm{def:g5:human-reproduction:pregnancy}{fetus} is far more fragile than an adult.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:9}
Identical twins: one fertilized \omterm{def:g4:animal-reproduction:sexual}{egg cell} split into two
\omterm{def:g5:human-reproduction:pregnancy}{embryos}. Ordinary (fraternal) twins: two \omterm{def:g4:animal-reproduction:sexual}{egg cells} fertilized
by two \omterm{def:g4:animal-reproduction:sexual}{sperm cells}, sharing only the \omterm{prop:g5:human-reproduction:plan}{womb}. The identical pair
are alike as copies because a body is built from its first
\omterm{def:g5:first-look-at-cells:cell}{cell} --- and they started from the same one.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:10}
At the elephant's end, and beyond it: usually one baby, years
of helplessness, and care that includes not only feeding and
protection but teaching --- language, skills, school. The
codfish's millions-with-no-care is the opposite pole.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:11}
The job of loading oxygen and unloading carbon dioxide: until
\omterm{def:g2:born-grow-die:cycle}{birth} the mother's \omterm{prop:g4:journey-of-food:absorb}{blood} did it through the cord; from the
first breath, the baby's own \omterm{def:g4:breathing:organs}{lungs}' air pockets take it over.
The cry is well timed because it comes at the exact minute
the old supply line is cut --- the \omterm{def:g4:breathing:organs}{lungs} must start precisely
when the cord stops.
\end{solution}
