\section*{Bab \ref{ch:g11:hormones-and-reproduction} --- Kendali Hormon atas Perkembangbiakan}

\begin{solution}{exo:g11:hormones-and-reproduction:1}
\omterm{def:g11:hormones-and-reproduction:hormone}{Hormon} adalah molekul yang dikeluarkan ke dalam darah oleh sebuah
kelenjar dan bekerja pada \omterm{def:g10:cells-common-unit:cell}{sel} yang jauh yang membawa reseptornya. \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan
balik} adalah kerja hasil akhirnya pada kelenjar di pangkal rantainya:
negatif ketika ia mengurangi perintahnya (sehingga arasnya tetap),
positif ketika ia menambahnya (sehingga menghasilkan sebuah lonjakan).
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:2}
Hipotalamus (GnRH), hipofisis (LH dan FSH), gonad (testosteron dalam
testisnya; estrogen dan progesteron dalam ovariumnya).
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:3}
Tubulus seminiferusnya menghasilkan sperma; \omterm{def:g10:cells-common-unit:cell}{sel} di antaranya
mengeluarkan testosteron.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:4}
Fase folikel (hari 1--13): folikel yang tumbuh, estrogen. Ovulasi (hari
ke-14): lonjakan LH. Fase luteal (hari 15--28): korpus luteum,
progesteron dan estrogen.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:5}
Merosotnya korpus luteumnya pada akhir daurnya: turunnya progesteron
dan estrogennya membuat lapisan rahim yang menebal itu terurai dan
berdarah.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:6}
Puncak estrogennya pada hari ke-12, puncak LH-nya pada hari ke-13,
ovulasinya pada hari ke-14, dan puncak progesteronnya pada hari ke-21.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:7}
Pengebirian menyingkirkan testosteron, dan dengan demikian hambatannya
pada hipotalamus dan hipofisisnya: LH pun naik. \omterm{prop:g11:becoming-male-female:hormones}{Testosteron} yang
disuntikkan memulihkan hambatannya dan LH turun --- yaitu \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik}
negatif pada kedua arahnya.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:8}
Pada hari ke-12 aras estrogennya tinggi dan bertahan, yang membalikkan
\omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya} menjadi positif dan memicu lonjakannya. Pada hari ke-20
estrogennya sedang dan progesteronnya tinggi, dan \omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya}
negatif: LH-nya tertahan rendah.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:9}
Hari ke-21: fase lutealnya berlangsung sekitar 14 hari berapa pun
panjang daurnya, sehingga ovulasinya jatuh 14 hari sebelum haid
berikutnya; sepekan tambahannya berada pada fase folikelnya.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:10}
Estrogen dan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} yang menyerupai progesteron dari pilnya, yang
mantap dan sedang, menjaga \omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya} tetap negatif sepanjang
bulan: FSH-nya tetap terlalu rendah bagi folikel untuk matang, sehingga
tidak ada puncak estrogen, tidak ada \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} positif, tidak ada
lonjakan LH, tidak ada ovulasi.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:11}
\omterm{prop:g11:becoming-male-female:hormones}{Testosteron} dari luar itu menghambat hipotalamus dan hipofisisnya; LH
dan FSH-nya turun; testisnya, yang tidak lagi dirangsang, berhenti
mengeluarkan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} dan berhenti membuat sperma, lalu mengerut.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:12}
Tanpa hambatan estrogennya, GnRH dan FSH naik; folikelnya matang;
puncak estrogennya (yang bekerja pada hipofisisnya, yang masih mampu
menanggapi) memicu lonjakan LH dan ovulasinya. Obat itu dipakai ketika
ovulasinya gagal karena folikelnya menerima FSH yang terlalu sedikit.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:13}
Folikelnya tumbuh dan mengeluarkan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} (ovarium dan FSH-nya
bekerja), tetapi puncak estrogennya tidak menghasilkan lonjakan: \omterm{def:g11:hormones-and-reproduction:hormone}{umpan
balik} positifnya gagal pada hipotalamus atau hipofisisnya. Suntikan
\omterm{def:g11:hormones-and-reproduction:hormone}{hormon} yang menyerupai LH pada saat puncaknya dapat memicu ovulasinya.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:14}
Karena dijaga tetap hidup, korpus luteumnya terus mengeluarkan
progesteron: lapisannya terjaga (tanpa haid) dan \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} negatifnya
menahan LH dan FSH tetap rendah (tanpa folikel baru atau ovulasi).
\omterm{def:g11:hormones-and-reproduction:hormone}{Hormonnya} muncul dalam darah dan urinenya dalam beberapa hari setelah
penanamannya, dan itulah yang dikenali ujinya.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:15}
$12 \times 0.6 \times 0.4 \approx 2.9$: sekitar 3 embrio. Peluang tidak
ada kelahiran dalam tiga pemindahan: $0.65^3 \approx 0.27$; sedikitnya
satu kelahiran: sekitar 73\%.
\end{solution}

\begin{solution}{pb:g11:hormones-and-reproduction:1}
\textbf{1.} Hari ke-14: lonjakan LH hari ke-13 mendahului ovulasinya
satu hari, dan progesteronnya, yang muncul hanya setelah ovulasinya,
naik sejak hari ke-14.

\textbf{2.} Fase folikelnya hari 1--13 (13 hari); fase lutealnya hari
14--27 atau 28 (sekitar 14 hari).

\textbf{3.} Folikel yang tumbuh pada hari ke-10; korpus luteumnya pada
hari ke-22, bagi estrogen maupun progesteronnya.

\textbf{4.} Hari ke-6: tipis, mulai menebal di bawah estrogennya. Hari
ke-22: tebal dan bergetah di bawah progesteronnya, siap bagi sebuah
embrio. Hari ke-28: terurai dan berdarah, karena kedua \omterm{def:g11:hormones-and-reproduction:hormone}{hormonnya} telah
turun.

\textbf{5.} Pada awalnya \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} daur sebelumnya telah turun dan
hambatannya ikut turun, sehingga FSH-nya naik; pada fase lutealnya
progesteron dan estrogennya menghambatnya dengan kuat.

\textbf{6.} \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan balik} negatif: estrogen folikelnya yang sedang naik
menghambat hipofisisnya, dan FSH-nya turun.

\textbf{7.} \omterm{def:g11:hormones-and-reproduction:hormone}{Umpan balik} positif: estrogen pada maksimumnya selama
sekitar dua hari merangsang hipotalamus dan hipofisisnya alih-alih
menghambatnya, dan LH-nya melonjak.

\textbf{8.} Negatif: progesteron bersama estrogen yang sedang
menghambat hipofisisnya; tanpa puncak estrogen yang bertahan, tidak ada
\omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} positif dan tidak ada lonjakan kedua.

\textbf{9.} Progesteron dan estrogennya runtuh; lapisannya terurai dan
berdarah pada hari ke-1 daur berikutnya; FSH-nya, yang lepas dari
hambatannya, naik.

\textbf{10.} Turunnya \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} lutealnya menyingkirkan \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik}
negatifnya, FSH-nya naik, dan folikel baru mulai tumbuh: akhir satu
daurnya adalah isyarat bagi daur berikutnya.

\textbf{11.} FSH dan LH-nya rendah sepanjang bulan (\omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} negatif
yang dijaga \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} pilnya); tidak ada folikel yang matang.

\textbf{12.} Tanpa puncak (tanpa folikel yang matang), tanpa lonjakan,
tanpa ovulasi.

\textbf{13.} \omterm{def:g11:hormones-and-reproduction:hormone}{Hormon} pilnya turun dan lapisannya, yang tipis, terurai
dan berdarah; itu adalah perdarahan lepas obat, bukan akhir sebuah fase
luteal --- karena tidak ada korpus luteum.

\textbf{14.} Tanpa \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} pilnya, \omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya} terangkat, FSH-nya
naik, sebuah folikel dapat mulai tumbuh dan mengeluarkan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon}; jika
ia mencapai puncak estrogennya, lonjakan LH dan ovulasi dapat menyusul,
dan lapisan yang tipis serta lendir yang kental tidak lagi terjaga.

\textbf{15.} Progesteron sendirian menghambat hipotalamus dan
hipofisisnya cukup untuk mencegah lonjakan LH-nya, dan mengentalkan
lendir leher rahimnya.

\textbf{16.} Hipotalamus atau hipofisisnya: ovariumnya tidak
dirangsang. Denyut GnRH (jika hipofisisnya bekerja) atau suntikan FSH
dan LH memulihkan pertumbuhan folikel dan ovulasinya, dengan melewati
tingkat yang hilang itu.

\textbf{17.} Pembuahan di luar tubuh: \omterm{def:g10:cells-common-unit:cell}{sel} telurnya dikumpulkan dari
ovariumnya, dibuahi dalam sebuah cawan, embrionya ditaruh dalam
rahimnya --- yaitu pertemuan gametnya dan pengangkutan yang akan
dikerjakan saluran telurnya.

\textbf{18.} Lonjakan LH yang alami; ovulasinya menyusul lonjakannya
sekitar 36 jam kemudian, sehingga \omterm{def:g10:cells-common-unit:cell}{sel} telurnya dikumpulkan tepat
sebelum \omterm{def:g10:cells-common-unit:cell}{sel} itu akan dilepaskan dan hilang.

\textbf{19.} Secara alami, estrogen folikel terdepannya yang naik
menurunkan FSH-nya, dan folikel yang lebih kecil, yang kelaparan FSH,
merosot; hanya yang terbesar yang bertahan. FSH yang disuntikkan
mengalahkan penyaringan itu dan menjaga semuanya tetap tumbuh.

\textbf{20.} Ovulasinya pada hari ke-14; \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} negatif yang
menurunkan FSH-nya seiring tumbuhnya folikelnya, lalu \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik}
positif yang mengubah puncak estrogennya menjadi lonjakan LH; pilnya
menjaga \omterm{def:g11:hormones-and-reproduction:hormone}{umpan baliknya} tetap negatif sepanjang bulan, sehingga tidak
ada folikel yang matang dan tidak ada lonjakan yang terjadi.
\end{solution}
