\chapter{Enzim dan Fenotipe}\label{ch:g11:enzymes-and-phenotype}

Seekor kucing Siam berwarna krem pada sebagian besar tubuhnya dan
cokelat tua pada telinga, moncong, kaki dan ekornya --- bagian yang
dingin. Cukurlah sebidang bulu pada punggungnya, jagalah bidang itu
tetap sejuk di bawah kantong es sementara bulunya tumbuh kembali, dan
bulu barunya tumbuh gelap. \omterm{def:g10:universal-dna:gene}{Gen} kucing itu tidak berubah; \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} pembuat
pigmennya sekadar bekerja di bawah \qty{35}{\celsius} dan tidak di
atasnya. Di antara sebuah \omterm{def:g10:universal-dna:gene}{gen} dan sifat yang kasatmata berdiri, hampir
selalu, sebuah \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}; dan di antara \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimnya} dan sifatnya berdiri
lingkungan tempat \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} itu bekerja. Bab ini membahas kedua perantara itu.

\section{Fenotipe pada tiga skala}

\begin{definition}[Genotipe dan fenotipe]\label{def:g11:enzymes-and-phenotype:phenotype}
Yang disebut \emph{genotipe}\index{genotipe} seorang individu adalah
perangkat \omterm{def:g10:universal-dna:gene}{alel} yang dibawanya; \emph{fenotipe}\index{fenotipe}-nya
adalah perangkat ciri yang dapat diamati. Fenotipenya dapat dilukiskan
pada tiga skala:
\begin{itemize}
  \item skala \emph{molekuler}: \omterm{def:g11:gene-expression:protein}{protein} mana yang hadir, dalam bentuk
        yang mana, dengan kegiatan yang bagaimana;
  \item skala \emph{seluler}: bentuk, isi dan perilaku \omterm{def:g10:cells-common-unit:cell}{selnya};
  \item skala \emph{organisme}: sifat yang kasatmata, dari warna mata
        sampai gejala sebuah penyakit.
\end{itemize}
Tiap skalanya adalah akibat skala di bawahnya.
\end{definition}

\begin{example}[Penyakit sel sabit pada tiga skala]\label{ex:g11:enzymes-and-phenotype:sickle}
Molekuler: satu \omterm{def:g10:chemistry-of-life:families}{asam amino} rantai hemoglobinnya (glutamat, yang
keenam) digantikan valin; molekulnya yang berubah saling melekat
menjadi serat panjang ketika oksigennya rendah. Seluler: seratnya
memuntir \omterm{def:g10:cells-common-unit:cell}{sel} darah merahnya menjadi sabit yang kaku yang menyumbat
pembuluh kecil dan mudah pecah. Organisme: anemia, serangan nyeri
ketika pembuluhnya tersumbat, kerusakan limpa dan ginjalnya. Satu
substitusi pada satu \omterm{def:g11:gene-expression:code}{kodon} satu \omterm{def:g10:universal-dna:gene}{gen}, dan tiga lukisan akibatnya.
\end{example}

\begin{proposition}[Protein membuat fenotipenya]\label{prop:g11:enzymes-and-phenotype:proteins}
Tiap aras \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipenya} dihasilkan \omterm{def:g11:gene-expression:protein}{protein}: sebuah \omterm{def:g10:universal-dna:gene}{alel} menetapkan sebuah
\omterm{def:g11:gene-expression:protein}{protein} (\cref{ch:g11:gene-expression}); kegiatan \omterm{def:g11:gene-expression:protein}{proteinnya} menetapkan
kimia dan struktur \omterm{def:g10:cells-common-unit:cell}{selnya}; \omterm{def:g10:cells-common-unit:cell}{selnya} menetapkan sifat organismenya.
Sebagian besar \omterm{def:g11:gene-expression:protein}{protein} itu berupa \emph{\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}}, dan karena itu sebagian
besar sifatnya adalah hasil rantai reaksi yang dijalankan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}.
\end{proposition}
\admitted

\section{Enzim}

\begin{definition}[Enzim, substrat, tapak aktif]\label{def:g11:enzymes-and-phenotype:enzyme}
Sebuah \emph{enzim}\index{enzim} adalah \omterm{def:g11:gene-expression:protein}{protein} yang mengatalisis satu
reaksi kimia: ia membuat reaksinya berjalan jutaan kali lebih cepat
tanpa terpakai habis. Molekul yang diubahnya adalah
\emph{substrat}\index{substrat}-nya; reaksinya berlangsung dalam sebuah
kantong enzimnya, yaitu \emph{tapak aktif}\index{tapak aktif}, yang
bentuk dan kimianya cocok dengan substrat itu dan bukan yang lain.
\emph{Kekhususan} ini mengikuti dari lipatan tiga dimensi enzimnya,
sehingga dari urutan \omterm{def:g10:chemistry-of-life:families}{asam aminonya}, sehingga dari \omterm{def:g10:universal-dna:gene}{gennya}.
\end{definition}

\begin{example}[Laktase]\label{ex:g11:enzymes-and-phenotype:lactase}
Gula susu, yaitu laktosa, dipecah menjadi dua gula sederhana oleh \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}
laktase pada lapisan usus halus; tanpa pemecahan itu ia lewat tanpa
tercerna menuju usus besar, tempat bakteri memfermentasinya, dengan
gas, kram dan diare sebagai hasilnya. Semua bayi membuat laktase; pada
sebagian besar umat manusia \omterm{def:g10:universal-dna:gene}{gennya} dimatikan setelah penyapihan, dan
orang dewasanya tidak tahan laktosa. Pada populasi dengan riwayat
peternakan susu yang panjang, sebuah \omterm{def:g10:universal-dna:gene}{alel} yang menjaga \omterm{def:g10:universal-dna:gene}{gennya} tetap
aktif seumur hidup menjadi lazim --- 90\% di Eropa utara, 10\% di Asia
timur. Tetes laktase yang dijual kepada mereka yang tidak tahan memuat
\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} yang sama, yang dibuat jamur.
\end{example}

\begin{proposition}[Syarat kegiatannya]\label{prop:g11:enzymes-and-phenotype:conditions}
Kegiatan sebuah \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} bergantung pada sekelilingnya. Ia naik seiring
suhu sampai \omterm{def:g11:gene-expression:protein}{proteinnya} mulai terbuka lipatannya, lalu runtuh --- \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}
manusia bekerja paling baik mendekati \qty{37}{\celsius} dan rusak di
atas sekitar \num{45}; tiap \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} bekerja dalam rentang keasaman yang
serasi dengan tempatnya (pepsin dalam asam lambung, tripsin dalam usus
yang netral); dan lajunya naik seiring kepekatan \omterm{def:g11:enzymes-and-phenotype:enzyme}{substratnya} sampai
sebuah maksimum, ketika tiap \omterm{def:g11:enzymes-and-phenotype:enzyme}{tapak aktifnya} sibuk.
\end{proposition}
\admitted

\section{Rantai reaksi}

\begin{proposition}[Jalur metabolisme]\label{prop:g11:enzymes-and-phenotype:pathway}
Sebuah \omterm{def:g10:cells-common-unit:cell}{sel} jarang membuat hasilnya dalam satu langkah: molekul awalnya
diubah oleh serangkaian \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}, yang masing-masing bekerja pada hasil
\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} sebelumnya, di sepanjang sebuah \emph{jalur
metabolisme}\index{jalur metabolisme}. Tiap \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimnya} disandikan \omterm{def:g10:universal-dna:gene}{gennya}
sendiri. \omterm{def:g10:universal-dna:gene}{Alel} yang tidak bekerja dari salah satunya menghalangi jalurnya
pada langkah itu: hasilnya tidak terbuat, dan \omterm{def:g11:enzymes-and-phenotype:enzyme}{substrat} \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} yang
terhalang itu dapat menumpuk. Karena itu dua \omterm{def:g10:universal-dna:gene}{gen} yang berbeda dapat
menghasilkan cacat kasatmata yang sama, dan cacat satu \omterm{def:g10:universal-dna:gene}{gen} dapat
menghasilkan beberapa cacat.
\end{proposition}
\admitted

\begin{example}[Fenilketonuria, dan bagaimana pola makan mengubah fenotipe]\label{ex:g11:enzymes-and-phenotype:pku}
Sekitar satu bayi baru lahir tiap sepuluh ribu membawa dua \omterm{def:g10:universal-dna:gene}{alel} \omterm{def:g10:universal-dna:gene}{gen}
\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} 1 yang tidak bekerja. Fenilalanin dari makanannya tidak diubah;
kepekatannya dalam darah naik dua puluh kali lipat dan ia meracuni otak
yang sedang berkembang. Tanpa pengobatan, anaknya menderita ketidakmampuan
berpikir yang berat. Jika dikenali sejak lahir lewat setetes darah pada
sebuah kartu --- tiap bayi baru lahir diuji --- dan diberi pola makan
yang hampir bebas fenilalanin, anak yang sama tumbuh normal. \omterm{def:g11:enzymes-and-phenotype:phenotype}{Genotipenya}
tidak berubah; \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipenya}, pada tiap skala di atas skala molekulernya,
adalah \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipe} orang yang sehat. Lingkungannya, di sini pola makannya,
yang memutuskan.
\end{example}

\section{Genotipe, lingkungan, fenotipe}

\begin{proposition}[Lingkungan membentuk fenotipe]\label{prop:g11:enzymes-and-phenotype:environment}
\omterm{def:g11:enzymes-and-phenotype:phenotype}{Fenotipenya} adalah hasil \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotipenya} \emph{dan} lingkungannya.
Lingkungannya bekerja pada skala molekuler (suhu dan keasaman yang
menetapkan kegiatan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimnya}, kehadiran sebuah \omterm{def:g11:enzymes-and-phenotype:enzyme}{substrat} yang menetapkan
apakah sebuah jalur berjalan), pada skala seluler (\omterm{def:g10:cells-common-unit:cell}{sel} yang menanggapi
isyarat, zat gizi, cahaya) dan pada skala organisme (pola makan,
olahraga, penyakit). Individu bergenotipe serupa --- kembar identik,
setek satu tumbuhan --- berbeda \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipenya} menurut riwayatnya; dan
individu bergenotipe berbeda dapat bertemu pada \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipe} yang sama di
bawah keadaan yang sesuai.
\end{proposition}
\begin{proof}[Bukti percobaan]
Tirosinase kucing Siam yang peka suhu: pigmennya terbentuk hanya pada
ujung tubuhnya yang sejuk, dan bidang cukur yang didinginkan tumbuh
kembali gelap. Fenilketonuria yang terkendalikan oleh pola makan.
Kembar identik yang dibesarkan terpisah berbeda tingginya beberapa
sentimeter dan massa tubuhnya lebih jauh lagi. Tumbuhan dari klon yang
sama yang ditumbuhkan di permukaan laut dan di tempat tinggi berbeda
tingginya sepuluh kali lipat, dan kembali seperti semula ketika
dipindahkan. Pada tiap hal, \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotipenya} tetap dan \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipenya} bergerak
mengikuti keadaannya.
\end{proof}

\begin{method}[Menalar dari gen ke sifat]\label{met:g11:enzymes-and-phenotype:reason}
\begin{enumerate}
  \item Kenali \omterm{def:g11:gene-expression:protein}{proteinnya}: \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} yang mana, dalam jalur yang mana, dan
        apa kerjanya.
  \item Kenali pengaruh \omterm{def:g10:universal-dna:gene}{alelnya} pada \omterm{def:g11:gene-expression:protein}{proteinnya}: normal, berkurang,
        tidak ada, atau berubah syarat kegiatannya.
  \item Ikuti akibatnya naik menyusuri skalanya: apa yang kurang atau
        menumpuk pada \omterm{def:g10:cells-common-unit:cell}{selnya}, lalu apa yang diperlihatkan organismenya.
  \item Tanyakan berapa \omterm{def:g10:universal-dna:gene}{alel} yang bekerja yang diperlukan: bagi
        sebagian besar \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} satu sudah cukup, sehingga cacatnya muncul
        hanya dengan dua \omterm{def:g10:universal-dna:gene}{alel} yang tidak bekerja (sifat resesif).
  \item Tanyakan apa yang ditambahkan lingkungannya: \omterm{def:g11:enzymes-and-phenotype:enzyme}{substratnya} hadir
        atau tidak, suhunya, sebuah pengobatan.
\end{enumerate}
\end{method}

\begin{remark}[Mengapa satu alel yang bekerja lazimnya cukup]\label{rem:g11:enzymes-and-phenotype:recessive}
\omterm{def:g11:enzymes-and-phenotype:enzyme}{Enzim} adalah katalis: jumlah yang kecil mengolah \omterm{def:g11:enzymes-and-phenotype:enzyme}{substrat} dalam jumlah
yang besar. Separuh jumlah tirosinase yang normal tetap membuat cukup
melanin, separuh laktase yang normal tetap mencerna segelas susu.
Karena itu pembawa satu \omterm{def:g10:universal-dna:gene}{alel} yang tidak bekerja lazimnya tidak dapat
dibedakan dari yang bukan pembawa --- yaitu ``resesif'' pada jilid
Dasar, yang kini terjelaskan --- dan sifatnya muncul hanya ketika kedua
\omterm{def:g10:universal-dna:gene}{alelnya} gagal. \omterm{def:g11:gene-expression:protein}{Protein} yang bukan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}, yang diperlukan dalam jumlah
banyak atau dalam bentuk yang tepat, sering menjadi kekecualian:
hemoglobin separuh normal menimbulkan perbedaan.
\end{remark}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:enzymes-and-phenotype:4}

\end{exercise}

Seorang anak dengan fenilketonuria diberi makan biasa. Molekul mana

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:enzymes-and-phenotype:9}
Ketidaktahanan laktosa pada orang dewasa adalah keadaan manusia yang
asli dan kelestarian laktase adalah \omterm{def:g11:mutations:mutation}{mutasi}. Jelaskan dalam arti apa
\omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipe} mutannya ``lebih sehat'', dan mengapa hanya dalam sebagian
lingkungan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:enzymes-and-phenotype:10}
Sebuah \omterm{def:g11:mutations:mutation}{mutasi} menggantikan satu \omterm{def:g10:chemistry-of-life:families}{asam amino} pada \omterm{def:g11:enzymes-and-phenotype:enzyme}{tapak aktif} sebuah
\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} dengan \omterm{def:g10:chemistry-of-life:families}{asam amino} lain yang sangat berbeda bentuknya. Ramalkan
akibatnya pada ketiga skalanya bagi jalur pilihanmu.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:enzymes-and-phenotype:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:enzymes-and-phenotype:12}

\begin{exercise}[$\star\star\star$]\label{exo:g11:enzymes-and-phenotype:13}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:enzymes-and-phenotype:14}

sebutkan apa yang diberitahukan masing-masing tentang sifat ``genetik''.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:enzymes-and-phenotype:15}

begitu menentukan.'' Bahaslah dengan fenilketonuria, albinisme dan

yang dapat serta tidak dapat diubah lingkungannya.
\end{exercise}

\section{Soal: Kucing Bertelinga Dingin}

\begin{problem}[{Soal akhir pekan --- bulu seekor kucing Siam yang
dijelaskan enzim demi enzim: tirosinase yang peka suhu, peta suhu
tubuhnya, bidang bulu yang dicukur, dan pola makan yang menulis ulang
sebuah fenotipe}]
\label{pb:g11:enzymes-and-phenotype:1}
\omterm{def:g10:universal-dna:gene}{Alel} Siam \omterm{def:g10:universal-dna:gene}{gen} tirosinase menyandikan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} yang kegiatannya, jika
diukur di laboratorium, adalah: \qty{100}{\percent} pada
\qty{30}{\celsius}, \qty{80}{\percent} pada \qty{33}{\celsius},
\qty{40}{\percent} pada \qty{35}{\celsius}, \qty{5}{\percent} pada
\qty{37}{\celsius}, \qty{0}{\percent} pada \qty{39}{\celsius}. \omterm{def:g11:enzymes-and-phenotype:enzyme}{Enzim}
\omterm{def:g10:universal-dna:gene}{alel} biasanya menjaga \qty{100}{\percent} sampai \qty{40}{\celsius}.
Suhu kulit seekor kucing: badan \qty{38}{\celsius}, kaki
\qty{36}{\celsius}, telapak \qty{33}{\celsius}, telinga dan moncong
\qty{31}{\celsius}. Melanin terlihat pada bulunya ketika \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimnya}
bekerja lebih dari sekitar \qty{30}{\percent}.

\medskip
\noindent\textbf{Bagian I --- Petanya.}
\begin{enumerate}
  \item Untuk tiap bagian tubuhnya, bacalah kegiatan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} Siamnya dan
        sebutkan apakah bulunya berpigmen.
  \item Gambarkan, atau lukiskan, bulu yang dihasilkannya. Apakah ia
        cocok dengan seekor Siam?
  \item Lakukan hal yang sama untuk kucing dengan \omterm{def:g10:universal-dna:gene}{alel} biasanya.
  \item Anak kucing Siam lahir seluruhnya krem. Jelaskan, jika
        diketahui bahwa rahimnya bersuhu \qty{38}{\celsius}.
  \item Pada suhu kulit berapa batas antara bulu gelap dan bulu pucat
        terletak? Bagian mana dari lengkung \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimnya} yang menetapkannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Percobaannya.}
\begin{enumerate}[resume]
  \item Sebidang bulu pada badannya dicukur dan dijaga pada
        \qty{30}{\celsius} di bawah kantong dingin sementara bulunya
        tumbuh kembali. Ramalkan warnanya.
  \item Sebidang bulu pada telapaknya dicukur dan dijaga tetap hangat
        pada \qty{38}{\celsius} dengan perban. Ramalkan.
  \item Setelah kantongnya disingkirkan, bulu yang tumbuh kembali
        menjaga warnanya berbulan-bulan, lalu berangsur digantikan.
        Jelaskan kedua kenyataan itu.
  \item Jelaskan mengapa percobaan ini menunjukkan bahwa lingkungannya
        bekerja pada skala molekuler, bukan pada \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotipenya}.
  \item Seorang murid mengusulkan bahwa ujung tubuhnya yang gelap
        disebabkan \omterm{def:g10:universal-dna:gene}{alel} yang berbeda pada \omterm{def:g10:cells-common-unit:cell}{sel} di sana. Berikan dua
        alasan yang membantahnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Ketiga skalanya.}
\begin{enumerate}[resume]
  \item Lukiskan \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipe} Siamnya pada skala molekuler, seluler dan
        organisme.
  \item \omterm{def:g10:universal-dna:gene}{Alel} Siamnya berbeda dari \omterm{def:g10:universal-dna:gene}{alel} biasanya oleh satu substitusi
        yang mengubah satu \omterm{def:g10:chemistry-of-life:families}{asam amino}. Jelaskan bagaimana satu \omterm{def:g10:chemistry-of-life:families}{asam
        amino} dapat membuat sebuah \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} peka suhu.
  \item Seekor kucing yang membawa satu \omterm{def:g10:universal-dna:gene}{alel} Siam dan satu \omterm{def:g10:universal-dna:gene}{alel} biasa
        berpigmen sepenuhnya. Jelaskan dengan
        \cref{rem:g11:enzymes-and-phenotype:recessive}.
  \item Seekor kucing albino punya dua \omterm{def:g10:universal-dna:gene}{alel} \omterm{def:g10:universal-dna:gene}{gen} yang sama yang tidak
        menyandikan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} yang bekerja pada suhu berapa pun.
        Bandingkan \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipenya} dengan \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipe} Siamnya pada ketiga skalanya.
  \item Kelinci Himalaya dan mencit tertentu memperlihatkan pola yang
        sama. Apa yang disarankannya tentang \omterm{def:g10:universal-dna:gene}{gen} tirosinasenya, dan apa
        yang kamu harapkan dari \omterm{def:g11:mutations:mutation}{mutasinya}?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pola makan yang menulis ulang \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipe}.}
Seorang bayi baru lahir diketahui mengidap fenilketonuria: fenilalanin
darahnya \qty{1200}{\micro mol/L} alih-alih \num{60}. Aras yang aman
bagi perkembangan otaknya di bawah \qty{360}{\micro mol/L}.
\begin{enumerate}[resume]
  \item Dengan faktor berapa fenilalaninnya naik, dan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} mana pada
        gambar jalurnya yang hilang?
  \item Jelaskan mengapa otak anaknya, dan bukan, misalnya, kulitnya,
        yang menjadi organ yang berisiko.
  \item Pola makannya menurunkan asupan fenilalanin menjadi seperlima
        yang normal; aras darahnya turun sebanding. Apakah anaknya
        aman? Apa yang dinyatakannya tentang hubungan antara pasokan
        \omterm{def:g11:enzymes-and-phenotype:enzyme}{substrat} dan \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipenya} ketika \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimnya} tidak ada?
  \item Sebagian fenilalanin tetap tak tergantikan, karena tubuhnya
        tidak dapat membuatnya. Jelaskan mengapa pola makannya harus
        dikendalikan alih-alih fenilalaninnya sekadar disingkirkan.
  \item Nyatakan hasilnya: bagi kucing Siamnya dan bagi anaknya, apa
        yang ditetapkan \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotipenya}, apa yang ditetapkan lingkungannya,
        dan satu kalimat yang menghubungkan \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotipe}, lingkungan dan \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotipe}.
\end{enumerate}
\end{problem}
