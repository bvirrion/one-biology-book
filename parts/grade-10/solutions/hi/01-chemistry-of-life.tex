\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:2}
खनिज: जल, सोडियम क्लोराइड, कैल्शियम फ़ॉस्फ़ेट, कार्बन डाइऑक्साइड (यह
कार्बन का यौगिक तो है, पर बनता जीवन के बाहर है और उसमें कार्बन-हाइड्रोजन
का ढाँचा नहीं)। कार्बनिक: ग्लूकोज़, ट्राइग्लिसराइड, \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्ल}।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:3}

95\% जल है।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:4}
(क) कुचले हुए बीज पर आयोडीन विलयन: नीला-काला। (ख) ब्यूरे अभिकर्मक:
बैंगनी। (ग) बीज को काग़ज़ पर कुचलिए: एक पारभासी धब्बा जो सूखता नहीं;

\begin{solution}{exo:g10:chemistry-of-life:5}
\omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट}: ग्लूकोज़ जैसी सरल शर्कराएँ। \omterm{def:g10:chemistry-of-life:families}{लिपिड}: ग्लिसरॉल और
वसा-अम्ल। \omterm{def:g10:chemistry-of-life:families}{प्रोटीन}: \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्ल} (बीस तरह के)। \omterm{def:g10:chemistry-of-life:families}{न्यूक्लिक अम्ल}:
न्यूक्लियोटाइड (चार तरह के)।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:6}
कार्बन: $18.5/0.02 \approx 900$ गुना अधिक सांद्रित। नाइट्रोजन:
$3.2/0.002 = 1600$ गुना। सजीव पदार्थ अपने परिवेश का नमूना नहीं है: वह
कुछ विरल तत्वों को हज़ार-हज़ार गुना बटोरता है, और यही जीवन का एक
रासायनिक हस्ताक्षर है।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:7}
जल: $0.62 \times 70 = \qty{43.4}{kg}$। शुष्क पदार्थ: $70 - 43.4 =
\qty{26.6}{kg}$। \omterm{def:g10:chemistry-of-life:families}{प्रोटीन}: $0.17 \times 70 = \qty{11.9}{kg}$।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:8}
\omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} $60 \times 17 = \qty{1020}{kJ}$; \omterm{def:g10:chemistry-of-life:families}{लिपिड} $12 \times 38 =
\qty{456}{kJ}$; \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} $8 \times 17 = \qty{136}{kJ}$; कुल
\qty{1612}{kJ}, यानी लगभग \qty{1600}{kJ}। \omterm{def:g10:chemistry-of-life:families}{लिपिड} ऊर्जा का $456/1612 \approx
28\%$ देते हैं, जबकि द्रव्यमान का केवल 12\% हैं।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:9}
ब्रेड: नीला-काला --- आटा ज़्यादातर मंड है। पनीर: आयोडीन की अपनी रंगत के
अलावा कोई रंग-परिवर्तन नहीं --- \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} और \omterm{def:g10:chemistry-of-life:families}{लिपिड}, मंड नहीं। चीनी: कोई
बदलाव नहीं --- सुक्रोज़ एक सरल शर्करा है, मंड नहीं; आयोडीन केवल मंड की
लंबी शृंखला को पकड़ता है।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:10}
शुष्क पदार्थ ज़्यादातर कार्बनिक है: कार्बन-हाइड्रोजन वाले अणु, जो
ऑक्सीजन से जुड़कर ऊर्जा, कार्बन डाइऑक्साइड और जल छोड़ते हैं। राख \omterm{def:g10:chemistry-of-life:mineral-organic}{खनिज
लवण} है, जो पहले ही ऑक्सीजन से पूरी तरह जुड़ चुके हैं; उनमें जलने लायक़
कुछ बचा ही नहीं।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:11}
फ़ेलिंग अभिकर्मक इकहरे ग्लूकोज़ अणुओं के एक खुले रासायनिक समूह से
अभिक्रिया करता है; मंड में ग्लूकोज़ इकाइयाँ एक-दूसरे से जुड़ी हैं और वे
समूह शृंखला के भीतर बंद हैं, इसलिए अभिकर्मक को कुछ मिलता ही नहीं।
इसके उलट, आयोडीन मंड की लंबी कुंडलित शृंखला में जा बैठता है, जो इकहरे
ग्लूकोज़ अणु बनाते ही नहीं। हर परीक्षण किसी संरचना को पकड़ता है, किसी
तत्व को नहीं।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:12}
दाने ने भारी मात्रा में जल सोख लिया है। 13\% पर भंडार ठोस रहते हैं और
एंजाइम काम नहीं कर सकते: दाने का रसायन बंद रहता है और वह सुप्त पड़ा
जीवित रहता है। जल भंडारों को घोलता है, एंजाइमों को काम करने देता है और
उत्पादों को बढ़ते भ्रूण तक पहुँचाता है: रसायन तभी दोबारा चलता है जब जल
प्रचुर हो।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:13}
नमक: $9 \times 0.5 = \qty{4.5}{g}$; ग्लूकोज़: $1 \times 0.5 =
\qty{0.5}{g}$। शुद्ध जल प्लाज़्मा को पतला कर देता; तब जल लाल रक्त
कोशिकाओं में घुस जाता, जो फूलकर फट जातीं।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:14}
ग्लाइकोजन: $500 \times 17 = \qty{8500}{kJ}$। \omterm{def:g10:chemistry-of-life:families}{लिपिड}: $10000 \times 38 =
\qty{3.8e5}{kJ}$। \qty{3.8e5}{kJ} को ग्लाइकोजन के रूप में रखने के लिए चाहिए $380000/17
\approx \qty{22.4}{kg}$ ग्लाइकोजन, और उसके साथ उससे तिगुना द्रव्यमान
जल, यानी कुल मिलाकर लगभग \qty{90}{kg} --- व्यक्ति से भी अधिक।
ढोए गए प्रति किलोग्राम वसा लगभग नौ गुना अधिक ऊर्जा संचित करती है, और
इसीलिए जिन जंतुओं को चलना-फिरना है वे अपना भंडार \omterm{def:g10:chemistry-of-life:families}{लिपिड} के रूप में रखते
हैं।
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:15}
परमाणु सचमुच साधारण हैं, पर (1) अनुपात साधारण नहीं: सजीव पदार्थ कार्बन
को हज़ार गुना और नाइट्रोजन को उससे कहीं अधिक सांद्रित करता है, जबकि
भूपर्पटी ऑक्सीजन-सिलिकॉन वाली है; (2) अणु साधारण नहीं: केवल सजीव ही
कार्बन को विशाल शृंखलाओं में जोड़ते हैं --- \omterm{def:g10:chemistry-of-life:families}{प्रोटीन}, मंड, DNA --- और
उनके आकार का कोई खनिज समकक्ष नहीं; (3) वे अणु अपनी इकाइयों के क्रम में
सूचना ढोते हैं, और कोई क्रिस्टल ऐसा नहीं करता। जीवन रासायनिक रूप से
अपने संगठन में ख़ास है, अपने परमाणुओं में नहीं।
\end{solution}

\begin{solution}{pb:g10:chemistry-of-life:1}
\textbf{1.} जल $0.13 \times 45 = \qty{5.9}{mg}$; शुष्क द्रव्यमान
\qty{39.2}{mg}।

\textbf{2.} मंड $0.70 \times 39.2 \approx \qty{27.4}{mg}$; \omterm{def:g10:chemistry-of-life:families}{प्रोटीन}
$0.12 \times 39.2 \approx \qty{4.7}{mg}$; \omterm{def:g10:chemistry-of-life:families}{लिपिड} $0.02 \times 39.2
\approx \qty{0.8}{mg}$।

\textbf{3.} मंड के लिए आयोडीन (नीला-काला); \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} के लिए ब्यूरे
(बैंगनी); \omterm{def:g10:chemistry-of-life:families}{लिपिड} के लिए काग़ज़ पर पारभासी धब्बा या सूडान III।

\textbf{4.} 13\% पर भंडार ठोस और एंजाइम निष्क्रिय रहते हैं; भ्रूण के
बढ़ने से पहले जल को भंडारों को घोलना और अभिक्रियाओं को चलने देना पड़ता
है। अंकुरण जल सोखने से ही शुरू होता है, और अंकुर का 88\% इसी बात का
निशान है कि उसका रसायन चालू है।

\textbf{5.} जल $0.13 \times 8 = \qty{1.04}{t}$; शुष्क पदार्थ
\qty{6.96}{t}, यानी लगभग \qty{7}{t}।

\textbf{6.} द्रव्यमान $500 + 320 + 10 - 130 = \qty{700}{g}$। रोटी में
जल: आटे से $0.13 \times 500 = \qty{65}{g}$, और \qty{320}{g} जोड़कर,
\qty{130}{g} घटाकर: \qty{255}{g}, यानी $255/700 \approx 36\%$।

\textbf{7.} आटे का शुष्क पदार्थ $0.87 \times 500 = \qty{435}{g}$; मंड
$0.70 \times 435 \approx \qty{305}{g}$; \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} $0.12 \times 435 \approx
\qty{52}{g}$ (\omterm{def:g10:chemistry-of-life:families}{लिपिड} लगभग \qty{9}{g})।

\textbf{8.} $305 \times 17 + 52 \times 17 + 9 \times 38 \approx 5185 +
884 + 342 \approx \qty{6400}{kJ}$।

\textbf{9.} रोटी का $120/700 \approx 0.17$: लगभग \qty{1100}{kJ}, यानी
\qty{10000}{kJ} का 11\%।

\textbf{10.} नमक खनिज है: वह पहले ही पूरी तरह ऑक्सीकृत है और उसमें जलने
लायक़ कार्बन-हाइड्रोजन बंध नहीं हैं। शरीर में उसके आयन रक्त की लवणता तय
करते हैं और तंत्रिका तथा पेशी के संकेतों के लिए ज़रूरी हैं।

\textbf{11.} $0.60 \times 60 = \qty{36}{kg}$ जल।

\textbf{12.} \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} $0.17 \times 60 = \qty{10.2}{kg}$; \omterm{def:g10:chemistry-of-life:families}{लिपिड} $0.16
\times 60 = \qty{9.6}{kg}$; खनिज \qty{3.6}{kg}; \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट}
\qty{0.6}{kg}।

\textbf{13.} उसके जल का $1.2/36 \approx 3.3\%$। रक्त का आयतन और ताप का
नियंत्रण, दोनों उस पर टिके हैं; 10\% की हानि ख़तरनाक है, इसलिए पीकर उसे
कुछ ही घंटों में लौटाना चाहिए।

\textbf{14.} प्लाज़्मा $0.55 \times 4.5 \approx \qty{2.5}{L}$; जल $0.9
\times 2.5 \approx \qty{2.2}{kg}$, यानी शरीर के जल का लगभग 6\%। बाक़ी
कोशिकाओं के भीतर (शरीर के कुल जल का लगभग दो-तिहाई) और कोशिकाओं के बीच के
तरल में है।

\textbf{15.} शिशु अपने जल का कहीं बड़ा अंश हर दिन बदलता है (पीना, मूत्र,
त्वचा) --- वयस्क से कहीं बड़ा; इसलिए बिना कुछ पिए कुछ ही घंटों की हानि
उसके कुल का ख़तरनाक अंश निकाल ले जाती है।

\textbf{16.} \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} $0.40 \times 0.6 = \qty{0.24}{kg}$; \omterm{def:g10:chemistry-of-life:families}{लिपिड}
$0.77 \times 9.6 \approx \qty{7.4}{kg}$; \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} $0.53 \times 10.2
\approx \qty{5.4}{kg}$। कुल लगभग \qty{13}{kg} कार्बन।

\textbf{17.} $13/60 \approx 22\%$, जबकि 18.5\% अपेक्षित था। कुलों के
गोल किए हुए अंशों और व्यक्तियों के बीच लिपिड-मात्रा के फेर को देखते हुए
मेल अच्छा है; दुबले शरीर पर आँकड़ा कम आता है।

\textbf{18.} $0.16 \times 10.2 \approx \qty{1.6}{kg}$ नाइट्रोजन। वह हवा
की नाइट्रोजन से आई, जिसे मिट्टी के जीवाणुओं ने पकड़ा, पौधों ने ग्रहण
किया और आहार शृंखला के साथ खाया गया।

\textbf{19.} $0.89 \times 36 \approx \qty{32}{kg}$ ऑक्सीजन, यानी शरीर के
द्रव्यमान का $32/60 \approx 53\%$। अकेला जल ही उन 65\% का अधिकांश
समझा देता है; बाक़ी कार्बनिक अणुओं के भीतर की ऑक्सीजन है।

\textbf{20.} \qty{60}{kg} का व्यक्ति लगभग \qty{13}{kg} कार्बन ढोता है।
वह कार्बन पौधों में प्रकाशसंश्लेषण से कार्बन डाइऑक्साइड के रूप में हवा
से निकला, और आहार शृंखला के साथ खाने तथा पाचन से शरीर में पहुँचा।
\end{solution}
