\chapter{Plants and Animals}\label{ch:g1:plants-animals}

\omterm{def:g1:living-nonliving:living}{Living things} come in many shapes. Two big groups you already know
are \emph{\omterm{def:g1:plants-animals:plant}{plants}} and \emph{\omterm{def:g1:plants-animals:animal}{animals}}. This chapter helps you tell
them apart.

\section{Plants}

\begin{definition}[Plant]\label{def:g1:plants-animals:plant}
A \emph{plant}\index{plant} is a \omterm{def:g1:living-nonliving:living}{living thing} that usually stays in
one place. Most plants are green. They make their own food using
light from the Sun, air and water.
\end{definition}

\begin{example}[Plants near you]\label{ex:g1:plants-animals:near}
Grass, trees, flowers, bushes and many house \omterm{def:g1:plants-animals:plant}{plants} are \omterm{def:g1:plants-animals:plant}{plants}.
Moss on a wall is a \omterm{def:g1:plants-animals:plant}{plant} too, even though it is small and soft.
\end{example}

\section{Animals}

\begin{definition}[Animal]\label{def:g1:plants-animals:animal}
An \emph{animal}\index{animal} is a \omterm{def:g1:living-nonliving:living}{living thing} that usually can
move from place to place. Animals cannot make food from sunlight
the way \omterm{def:g1:plants-animals:plant}{plants} do. They eat \omterm{def:g1:plants-animals:plant}{plants} or other animals.
\end{definition}

\begin{example}[Animals near you]\label{ex:g1:plants-animals:animals}
Dogs, cats, birds, fish, insects and people are \omterm{def:g1:plants-animals:animal}{animals}. A snail is
slow, but it is still an \omterm{def:g1:plants-animals:animal}{animal} --- it moves and it eats.
\end{example}

\section{Comparing plants and animals}

\begin{proposition}[Plants and animals]\label{prop:g1:plants-animals:compare}
\begin{itemize}
  \item \omterm{def:g1:plants-animals:plant}{Plants} make their own food from light; \omterm{def:g1:plants-animals:animal}{animals} eat food.
  \item Most \omterm{def:g1:plants-animals:plant}{plants} stay rooted in one place; most \omterm{def:g1:plants-animals:animal}{animals} move.
  \item Both are living: both grow and both can make young.
\end{itemize}
\end{proposition}
\admitted

\begin{method}[Plant or animal?]\label{met:g1:plants-animals:sort}
\begin{enumerate}
  \item Does it make food from sunlight, or does it eat?
  \item Does it stay put with roots, or can it walk, swim, fly or
        crawl?
  \item Use both clues together --- a Venus flytrap is still a
        \omterm{def:g1:plants-animals:plant}{plant} even if it snaps shut.
\end{enumerate}
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g1:plants-animals:1}
Name two \omterm{def:g1:plants-animals:plant}{plants} and two \omterm{def:g1:plants-animals:animal}{animals} you know.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-animals:2}
Is a butterfly a \omterm{def:g1:plants-animals:plant}{plant} or an \omterm{def:g1:plants-animals:animal}{animal}? Why?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-animals:3}
Copy and complete: \omterm{def:g1:plants-animals:plant}{Plants} usually \underline{\hspace{1.5cm}}.
\omterm{def:g1:plants-animals:animal}{Animals} usually \underline{\hspace{1.5cm}}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-animals:4}
True or false? ``All \omterm{def:g1:living-nonliving:living}{living things} are either \omterm{def:g1:plants-animals:plant}{plants} or \omterm{def:g1:plants-animals:animal}{animals}.''
(Think of a mushroom if you know one.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-animals:5}
How do \omterm{def:g1:plants-animals:plant}{plants} get food? How do \omterm{def:g1:plants-animals:animal}{animals} get food?
\end{exercise}
