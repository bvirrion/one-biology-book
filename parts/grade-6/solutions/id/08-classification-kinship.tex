\section*{Bab \ref{ch:g6:classification-kinship} --- Klasifikasi dan Kekerabatan}

\begin{solution}{exo:g6:classification-kinship:1}
Ciri yang dimiliki sebuah organisme, yang dinyatakan sehingga dapat
diperiksa: punya \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}; punya empat \omterm{def:g1:my-body:parts}{anggota gerak}; punya
\omterm{def:g1:animals-around-us:covering}{bulu} dan memberi \omterm{def:g2:animals-grow-up:born}{susu} kepada anaknya (atau: punya \omterm{def:g1:animals-around-us:covering}{bulu burung}).
\end{solution}

\begin{solution}{exo:g6:classification-kinship:2}
Himpunan pemilik tiap \omterm{def:g6:classification-kinship:attribute}{atributnya} duduk sepenuhnya di dalam himpunan
\omterm{def:g6:classification-kinship:attribute}{atribut} yang lain: semua pembawa \omterm{def:g1:animals-around-us:covering}{bulu} punya empat \omterm{def:g1:my-body:parts}{anggota gerak}, semua
\omterm{def:g1:animals-around-us:animal}{hewan} berkaki empat punya \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} --- tidak pernah
sebaliknya, tidak pernah bersilangan. Kotak di dalam kotak, tanpa ada
yang mengangkangi keduanya.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:3}
Karena mereka mewarisinya dari seorang nenek moyang bersama: sebuah
\omterm{def:g6:classification-kinship:attribute}{atribut} menuruni sebuah garis keluarga, sehingga pemiliknya adalah
keturunan nenek moyang itu --- sebuah keluarga.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:4}
\omterm{prop:g3:sorting-living-things:groups}{Ikan} troutnya berdiri melewati titik \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} saja. Pausnya
berdiri melewati \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak}, dan
bulu-dan-susu --- ketiganya.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:5}
Paus dan kucing: keduanya berbagi percabangan bulu-dan-susu, yang
termuda dari yang digambar; \omterm{prop:g3:sorting-living-things:groups}{ikan} troutnya berpisah pada percabangan
yang tertua.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:6}
Keduanya adalah keterangan yang sama dua kali: sebuah kotak adalah
sebuah cabang beserta semua \omterm{def:g1:plants-around-us:tree}{rantingnya} --- segala yang melewati sebuah
percabangan mewarisi \omterm{def:g6:classification-kinship:attribute}{atribut} percabangan itu lalu duduk di dalam
kotaknya.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:7}
Bulu-dan-susu diwariskan menuruni satu garis keluarga --- ia menandai
keturunan. ``Berbentuk seperti \omterm{prop:g3:sorting-living-things:groups}{ikan}'' dapat dicapkan pada orang asing
oleh cara hidup berair yang sama, seperti diperlihatkan pausnya:
bentuknya dipinjam, keluarganya di tempat lain. \omterm{def:g4:classifying-animals:classification}{Klasifikasi}
mengelompokkan keluarga, sehingga ia memercayai barang pusakanya dan
mencurigai mantel pinjamannya.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:8}
\omterm{prop:g3:sorting-living-things:groups}{Ikan} trout: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} saja. Katak: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat
\omterm{def:g1:my-body:parts}{anggota gerak}. Kucing: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak},
bulu-dan-susu. \omterm{prop:g3:sorting-living-things:groups}{Burung} hitam: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak},
\omterm{def:g1:animals-around-us:covering}{bulu burung}. Persarangannya terlihat pada centangnya: centang tiap
barisnya memuat \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}; tiap baris yang mencentang penutupnya
juga mencentang empat \omterm{def:g1:my-body:parts}{anggota gerak}.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:9}
Ia melewati \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} dan empat \omterm{def:g1:my-body:parts}{anggota gerak}, lalu berpisah
pada cabangnya sendiri dengan \omterm{def:g1:the-five-senses:sense}{kulit} bersisiknya yang kering --- di
samping, bukan di seberang, percabangan \omterm{def:g1:animals-around-us:covering}{bulu} dan \omterm{def:g1:animals-around-us:covering}{bulu burungnya}.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:10}
Menyusuinyalah yang menandai \omterm{prop:g6:classification-kinship:kinship}{kekerabatan}: bulu-dan-susu adalah barang
pusaka keluarga \omterm{prop:g3:sorting-living-things:groups}{mamalia}, sehingga kelelawarnya bercabang bersama
kucingnya. Terbang dipinjam lewat cara hidup --- sayap timbul secara
terpisah pada cabang \omterm{prop:g3:sorting-living-things:groups}{burungnya} dan pada cabang kelelawarnya,
sebagaimana kerampingan timbul pada paus dan \omterm{prop:g3:sorting-living-things:groups}{ikan} trout.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:11}
Ia termasuk di dalamnya. Platipus berketurunan dari nenek moyang
bersama para \omterm{prop:g3:sorting-living-things:groups}{mamalia} dan membawa barang pusaka keluarganya (\omterm{def:g1:animals-around-us:covering}{bulu},
\omterm{def:g2:animals-grow-up:born}{susu}); \omterm{def:g2:animals-grow-up:born}{telurnya} adalah sebuah ciri tua yang dipertahankan cabang
awalnya sementara yang lain kehilangannya. Keluarganya adalah nenek
moyangnya beserta \emph{semua} keturunannya --- termasuk yang ganjil;
berkas cirinya (\cref{exo:g4:classifying-animals:11}) sudah memilihnya
masuk.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:12}
Kalau setiap pasang cabang bertemu, maka \omterm{def:g5:biodiversity:species}{spesies} pastilah berubah
lintas generasi sepanjang waktu yang panjang --- \omterm{def:g6:classification-kinship:attribute}{atribut} baru timbul
lalu diwariskan --- sehingga satu garis nenek moyang dapat bercabang
menjadi \omterm{prop:g3:sorting-living-things:groups}{ikan} trout, kucing dan \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} aster. Siratnya dalam satu
kalimat: semua \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} berketurunan, dengan perubahan, dari
nenek moyang bersama. Bukti yang harus dituntut: jejak perubahannya
--- bentuk peralihan, sisa yang terawetkan di dalam batuan tua, dan
kemiripan keluarga yang terletak persis di tempat yang diramalkan
\omterm{def:g6:classification-kinship:tree}{pohonnya} (urusan sebuah tahun berikutnya).
\end{solution}

\begin{solution}{pb:g6:classification-kinship:1}
\textbf{1.} \omterm{prop:g3:sorting-living-things:groups}{Ikan} trout: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}. Katak: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat
\omterm{def:g1:my-body:parts}{anggota gerak}. Kadal: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak}, \omterm{def:g1:the-five-senses:sense}{kulit}
bersisik yang kering. Merpati: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak},
\omterm{def:g1:animals-around-us:covering}{bulu burung}. Tikus: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak},
bulu-dan-susu. Lumba-lumba: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak},
bulu-dan-susu (kesaksian labelnya tentang \omterm{def:g2:animals-grow-up:born}{susu}; cirinyalah yang
memutuskan, dan \omterm{def:g2:animals-grow-up:born}{susu} adalah ciri yang menentukan).

\textbf{2.} \omterm{def:g4:classifying-animals:vertebrate}{Tulang belakangnya} --- keenamnya adalah \omterm{def:g4:classifying-animals:vertebrate}{vertebrata}.
Perkongsian yang terluas menandai nenek moyang bersama yang tertua:
keenamnya adalah satu keluarga pada kedalaman itu.

\textbf{3.} Katak, kadal, merpati, tikus, lumba-lumba. Sayap merpatinya
dan sirip dayung lumba-lumbanya adalah empat \omterm{def:g1:my-body:parts}{anggota gerak} dalam
pakaian kerja: \omterm{def:g1:my-body:parts}{anggota gerak} warisan yang sama, yang dibentuk oleh
kehidupan yang berbeda --- \omterm{def:g6:classification-kinship:tree}{pohonnya} menghitung warisannya, bukan
pakaiannya.

\textbf{4.} Karena alamat yang sama bukanlah keluarga yang sama:
``hidup di air'' adalah \omterm{def:g2:where-animals-live:habitat}{habitat}, yang dapat dipinjam orang asing,
bukan \omterm{def:g6:classification-kinship:attribute}{atribut} keturunan tubuhnya ---
aturan lama \cref{def:g3:sorting-living-things:criterion}, yang
sekarang punya alasannya.

\textbf{5.} \omterm{def:g1:plants-around-us:parts}{Batang} $\rightarrow$ percabangan \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} (\omterm{prop:g3:sorting-living-things:groups}{ikan}
troutnya berpisah sesudahnya) $\rightarrow$ percabangan empat \omterm{def:g1:my-body:parts}{anggota
gerak} $\rightarrow$ tiga cabang penutup: \omterm{def:g1:animals-around-us:covering}{sisik} kering (kadal), \omterm{def:g1:animals-around-us:covering}{bulu
burung} (merpati), bulu-dan-susu (tikus dan lumba-lumba bersama pada
percabangan yang termuda); kataknya berpisah sesudah empat \omterm{def:g1:my-body:parts}{anggota
gerak}, sebelum setiap penutupnya.

\textbf{6.} Sepupu terdekat tikusnya di dalam lacinya adalah
lumba-lumbanya (keduanya berbagi percabangan bulu-dan-susu). Sepupu
terdekat \omterm{prop:g3:sorting-living-things:groups}{ikan} troutnya adalah semuanya sama saja --- ia berpisah pada
percabangan yang pertama, sehingga kelima yang lain sama jauh
kerabatnya.

\textbf{7.} Seekor \omterm{prop:g3:sorting-living-things:groups}{amfibi} --- berkaki empat, berkulit telanjang yang
lembap, tanpa percabangan penutupnya sendiri; cabangnya berpisah di
antara percabangan empat \omterm{def:g1:my-body:parts}{anggota gerak} dan penutupnya.

\textbf{8.} Melewati \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} dan empat \omterm{def:g1:my-body:parts}{anggota gerak}, lalu
masuk ke cabang bersisik kering: seekor \omterm{prop:g4:classifying-animals:classes}{reptil} --- yang
diklasifikasikan tanpa melihat, karena \omterm{def:g6:classification-kinship:attribute}{atributnya}, bukan labelnya,
yang memutuskan.

\textbf{9.} Lumba-lumba dan tikus duduk lebih dekat: bulu-dan-susu
adalah percabangan warisan. Kerampingannya adalah seragam airnya, yang
dibagikan sama kepada perenang yang tidak berkerabat
(\cref{exo:g4:adapted-to-their-world:11}); kalau dihitung, ia akan
mengarsipkan pakaian pinjaman sebagai keluarga lalu merusak
persarangannya.

\textbf{10.} Satu \omterm{def:g5:biodiversity:species}{spesies} nenek moyang bersama --- seekor \omterm{def:g4:classifying-animals:vertebrate}{vertebrata}
berkaki empat yang awal --- yang darinya katak, kadal, merpati, tikus
dan lumba-lumba semuanya berketurunan, masing-masing membawa keempat
\omterm{def:g1:my-body:parts}{anggota geraknya} dalam pakaian yang berbeda.

\textbf{11.} Bahwa keenamnya --- dan kelas yang menggambarnya ---
termasuk dalam satu keluarga yang bercabang, yang berketurunan dengan
perubahan dari nenek moyang bersama, dengan manusia berada pada cabang
bulu-dan-susu di antara yang lain.

\textbf{12.} Persarangannya adalah sebuah kenyataan tentang dunia yang
hidup --- \omterm{def:g6:classification-kinship:attribute}{atribut} memang sungguh-sungguh jatuh ke dalam
kotak-di-dalam-kotak, yang tidak ditetapkan juru arsip mana pun.
\omterm{def:g6:classification-kinship:tree}{Pohonnya} menjelaskan kenyataan itu: pewarisan dari nenek moyang
bersama \emph{pasti} menghasilkan \omterm{def:g6:classification-kinship:attribute}{atribut} yang bersarang, sedangkan
pengarsipan belaka akan membiarkan kerapian dunia yang penurut itu
tetap menjadi sebuah keajaiban.
\end{solution}
