\chapter{कचरा बाहर करना}\label{ch:g7:removing-wastes}

इस साल पहुँचाने की ही धूम रही: \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} भीतर, \omterm{def:g7:digestion-and-nutrients:families}{पोषक तत्व} भीतर, और हर काम
करती \omterm{def:g5:first-look-at-cells:cell}{कोशिका} तक ईंधन। पर हर कारख़ाना कचरा भी बनाता है, और जो पहुँचाने की
सेवा कूड़ा कभी उठाती ही न हो वह जल्द अपने ही शहर को ज़हरीला कर देती है।
\omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} का हिसाब तो चुक चुका --- वह \omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाओं} पर साँस के साथ
बाहर हो जाती है। यह आख़िरी अध्याय \emph{बाक़ी} कचरों के पीछे-पीछे शरीर
के सबसे कम चर्चा वाले अंगों तक जाता है: \omterm{def:g7:removing-wastes:kidneys}{गुर्दों} तक।

\section{कार्बन डाइऑक्साइड से आगे के कचरे}

\begin{proposition}[कोशिकाओं का बाक़ी कूड़ा]\label{prop:g7:removing-wastes:rubbish}
\omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} का जलना और बनाना
(\cref{prop:g7:what-is-respiration:power},
\cref{prop:g6:matter-from-food:transform}) \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} से
ज़्यादा कुछ देता है। ख़ास तौर पर शरीर बनाने वाले पोषक तत्वों के
इस्तेमाल से नाइट्रोजन वाला एक कचरा पीछे छूटता है ---
\emph{यूरिया}\index{यूरिया} --- जिसे \omterm{prop:g4:journey-of-food:absorb}{रक्त} हर ऊतक से ठीक वैसे ही बटोरता
है जैसे \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} को। \omterm{prop:g7:removing-wastes:rubbish}{यूरिया} गुज़रती आवाजाही में बेज़रर है और
ढेर बन जाए तो ज़हरीली: उसे लगातार बाहर जाना ही चाहिए।
\end{proposition}
\admitted

\begin{example}[दो कूड़ा-चक्कर, रक्त एक]\label{ex:g7:removing-wastes:rounds}
\cref{ex:g7:heart-and-circulation:tour} वाली वही बूँद दोनों कूड़ा-चक्कर
एक ही साथ चलाती है: जाँघ की \omterm{def:g7:heart-and-circulation:vessels}{केशिकाओं} पर वह \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड}
\emph{और} \omterm{prop:g7:removing-wastes:rubbish}{यूरिया}, दोनों उठाती है। पहली को वह अगले चक्कर में फेफड़ों पर
उतार देती है। दूसरी सवार ही रहती है --- फेफड़ों के आगे निकलकर, जो उसे
छू तक नहीं सकते --- जब तक शरीर वाला चक्कर उसे उन अंगों तक न ले जाए जो
ठीक इसी माल के लिए बने हैं।
\end{example}

\section{गुर्दे, रक्त के छन्ने}

\begin{definition}[गुर्दे और मूत्र-तंत्र]\label{def:g7:removing-wastes:kidneys}
दोनों \emph{गुर्दे}\index{गुर्दा} --- मुट्ठी जितने, सेम के दाने की
शक्ल के, पीठ में नीचे की ओर --- \omterm{prop:g4:journey-of-food:absorb}{रक्त} साफ़ करने वाले छन्ने हैं। हर एक को
शरीर वाले चक्कर की मुख्य लाइन से सीधे एक भरी-पूरी \omterm{def:g7:heart-and-circulation:vessels}{धमनी} मिलती है। भीतर
\omterm{prop:g4:journey-of-food:absorb}{रक्त} लगभग दस लाख सूक्ष्म छानने वाली इकाइयों से छाना जाता है; साफ़
हुआ \omterm{prop:g4:journey-of-food:absorb}{रक्त} गुर्दे की \omterm{def:g7:heart-and-circulation:vessels}{शिरा} से लौट जाता है, और हटाया हुआ बोझ --- \omterm{prop:g7:removing-wastes:rubbish}{यूरिया},
फ़ालतू पानी, फ़ालतू लवण --- \emph{मूत्र}\index{मूत्र} बनकर निकलता है,
जो दो नलियों से टपकता हुआ \emph{मूत्राशय}\index{मूत्राशय} तक जाता है,
यानी उस भंडार-टंकी तक जो मरज़ी से ख़ाली की जाती है।
\end{definition}

\begin{proposition}[छानो, फिर वापस लो]\label{prop:g7:removing-wastes:filter}
\omterm{def:g7:removing-wastes:kidneys}{गुर्दे} की तरकीब है दरियादिली, और उसके पीछे-पीछे किफ़ायत। छानने वाली
इकाइयाँ पहले \omterm{prop:g4:journey-of-food:absorb}{रक्त} के पानी वाले हिस्से का एक बहुत बड़ा आयतन --- दिन के
कई दसियों लीटर --- अपनी महीन छननियों से धकेल देती हैं, \omterm{prop:g7:removing-wastes:rubbish}{यूरिया}, शर्कराएँ,
लवण, सब कुछ समेत। फिर हर इकाई की घुमावदार नली के साथ-साथ शरीर वह सब
\emph{वापस ले लेता} है जो उसे चाहिए: क़रीब-क़रीब सारा पानी, सारी
\omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़}, और नपे-तुले लवण --- और पीछे छूट जाते हैं \omterm{prop:g7:removing-wastes:rubbish}{यूरिया} तथा फ़ालतू
चीज़ें। छाने गए दसियों लीटर में से लगभग डेढ़ लीटर \omterm{def:g7:removing-wastes:kidneys}{मूत्र} बनकर निकलता है:
सब कुछ छानो, अच्छा वापस ले लो, और कचरे के छिपने की कोई जगह नहीं बचती।
\end{proposition}
\admitted

\begin{example}[किफ़ायत को पढ़ना]\label{ex:g7:removing-wastes:thrift}
वापस लेने वाला चरण रोज़मर्रा की देखी हुई बातें समझा देता है। ख़ूब पियो,
और \omterm{def:g7:removing-wastes:kidneys}{गुर्दे} कम पानी वापस लेते हैं: \omterm{def:g7:removing-wastes:kidneys}{मूत्र} भरपूर और हलके रंग का। किसी गरम
मुक़ाबले में \omterm{rem:g7:removing-wastes:sweat}{पसीना} बहा दो, और वे क़रीब-क़रीब सारा वापस ले लेते हैं:
\omterm{def:g7:removing-wastes:kidneys}{मूत्र} थोड़ा और गहरे रंग का --- यानी शरीर अपने पानी की रखवाली करता हुआ
(\cref{prop:g4:what-plants-need:needs} ने \omterm{def:g1:plants-around-us:plant}{पौधों} के लिए पानी की क़ीमत
सिखाई थी; \omterm{def:g7:removing-wastes:kidneys}{गुर्दा} उसी अर्थशास्त्र का \omterm{def:g1:animals-around-us:animal}{जंतु} वाला पहलू है)। और \omterm{def:g7:removing-wastes:kidneys}{मूत्र} में
\omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़}, जिसे छननी तो जाने देती है पर वापस लेना पूरी की पूरी लौटा लेना
चाहिए, एक चिर-परिचित चिकित्सा-संकेत है कि कुछ न कुछ --- या तो वापस लेना
या \omterm{prop:g4:journey-of-food:absorb}{रक्त} की शर्करा --- गड़बड़ है।
\end{example}

\begin{remark}[त्वचा भी हाथ बँटाती है]\label{rem:g7:removing-wastes:sweat}
\emph{पसीना}\index{पसीना} उसी कारोबार की एक बग़ली शाख़ है: \omterm{def:g1:the-five-senses:sense}{त्वचा} पर
छोड़े गए पानी और लवण (और साथ में \omterm{prop:g7:removing-wastes:rubbish}{यूरिया} की हलकी मात्रा) --- हालाँकि
उसका पहला काम \cref{ex:g7:organs-at-work:felt} वाली ठंडक है। \omterm{def:g7:how-animals-breathe:four}{फेफड़े},
\omterm{def:g7:removing-wastes:kidneys}{गुर्दे} और \omterm{def:g1:the-five-senses:sense}{त्वचा} मिलकर शरीर का पूरा कचरा-हटाव बाँट लेते हैं: \omterm{def:g7:what-is-respiration:respiration}{कार्बन
डाइऑक्साइड} पहले को, \omterm{prop:g7:removing-wastes:rubbish}{यूरिया} और फ़ालतू चीज़ें दूसरे को, और ऊष्मा तथा एक
नमकीन रिसाव तीसरे को। ज़हरीला कुछ भी ढेर बनने को नहीं छूटता --- स्वस्थ
शरीर में कूड़ा कभी नहीं जीतता।
\end{remark}

\section{साल की मशीन, पूरी}

\begin{proposition}[आपूर्ति और हटाव का पूरा फंदा]\label{prop:g7:removing-wastes:complete}
\omterm{def:g7:removing-wastes:kidneys}{गुर्दों} के अपनी जगह पर आते ही इस साल का दौरा एक मशीन बनकर बंद हो जाता
है। \omterm{prop:g4:journey-of-food:absorb}{रक्त}, जो दोहरे चक्कर भर पंप किया जाता है
(\cref{def:g7:heart-and-circulation:double}), \omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाओं} पर \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} और
\omterm{prop:g7:digestion-and-nutrients:absorption}{रसांकुरों} पर \omterm{def:g7:digestion-and-nutrients:families}{पोषक तत्व} लादता है; दोनों को जलने तथा बनाने के लिए हर काम
करती \omterm{def:g5:first-look-at-cells:cell}{कोशिका} तक पहुँचाता है; और कचरे बटोरता है --- \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड}
फेफड़ों को, \omterm{prop:g7:removing-wastes:rubbish}{यूरिया} तथा फ़ालतू चीज़ें \omterm{def:g7:removing-wastes:kidneys}{गुर्दों} को। चौथी और सातवीं कक्षा
का हर अंग इसी एक फंदे का एक ठिकाना है; हर अदला-बदली किसी \omterm{def:g7:heart-and-circulation:vessels}{केशिका} की
दीवार पार करती है; और यह सारा कुछ
\cref{prop:g7:what-is-respiration:power} वाली धीमी आग पर चलता है,
कोशिका-दर-कोशिका।
\end{proposition}
\admitted

\begin{method}[पूरे शरीर की जाँच-पड़ताल]\label{met:g7:removing-wastes:audit}
किसी भी पदार्थ के लिए उसके गुज़रने की जाँच-पड़ताल करो:
\begin{enumerate}
  \item भीतर आना: मुँह और \omterm{prop:g7:digestion-and-nutrients:absorption}{रसांकुर}, या नाक और \omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाएँ};
  \item ढुलाई: कौन-से चक्कर, और कौन-से ख़ास पड़ाव
        (\omterm{prop:g7:digestion-and-nutrients:absorption}{यकृत});
  \item इस्तेमाल: ईंधन जला, सामान बनकर लगा, या इनमें से कुछ नहीं;
  \item बाहर जाना: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाएँ} (गैस), \omterm{def:g7:removing-wastes:kidneys}{गुर्दे} (\omterm{prop:g7:removing-wastes:rubbish}{यूरिया}, फ़ालतू चीज़ें),
        \omterm{def:g1:the-five-senses:sense}{त्वचा} (\omterm{rem:g7:removing-wastes:sweat}{पसीने} का हिस्सा) --- या फिर, कभी अवशोषित ही न होकर,
        बचे-खुचे का रास्ता
        (\cref{prop:g4:journey-of-food:absorb})।
\end{enumerate}
इसे पानी पर, \omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़} पर, और नमकीन चिप्स पर चलाओ: मशीन के पास हर क़दम
के लिए एक ठिकाना है, और एक ही अभ्यास में यही इस साल की खोज है।
\end{method}

\section{समस्या: पानी के एक अणु का दिन}

\begin{problem}[{सप्ताहांत समस्या --- पानी का एक गिलास, पूरी मशीन भर
जाँचा हुआ}]\label{pb:g7:removing-wastes:1}
खेल वाले किसी दिन नाश्ते में पिए गए पानी के एक गिलास का पीछा करो, और
वह भी साल की उस बड़ी जाँच-पड़ताल के रूप में।

\medskip\noindent\textbf{भाग I --- भीतर आना और ढुलाई।}
\begin{enumerate}
  \item पानी को कोई तोड़-फोड़ नहीं चाहिए
        (\cref{def:g7:digestion-and-nutrients:families})। उसके भीतर
        आने की सरहद-पार बताओ, और वह वाहिका-जाल भी जो उसे लेता है।
  \item उसका \omterm{prop:g4:journey-of-food:absorb}{रक्त} वही जाना-पहचाना पहला पड़ाव करता है। कौन-सा अंग, और
        पानी को ख़ुद कोई छँटाई न चाहिए होने पर भी यह रास्ता पोषक
        तत्वों के लिहाज़ से समझदारी भरा क्यों है?
  \item दोपहर से पहले-पहले वह पानी वहाँ-वहाँ पहुँच चुका है जहाँ-जहाँ
        \omterm{prop:g4:journey-of-food:absorb}{रक्त} जाता है। अब वह जिन दो चक्करों पर बारी-बारी से सवार है,
        उनके नाम बताओ।
  \item उसमें से कुछ जल्द ही काम करती \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} के भीतर होता है। वहाँ
        पानी के दो काम गिनाओ (याद रखो कि \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} का रसायन ---
        रस, जलना --- गीले में ही चलता है)।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दोपहर का मुक़ाबला।}
\begin{enumerate}[resume]
  \item मुक़ाबले के दौरान उस पानी का एक हिस्सा ऐसी सरहद से बाहर जाता
        है जो पहले-पहल कोई उत्सर्जन का अंग है ही नहीं। उसका नाम
        बताओ, पानी के अलावा उसका माल, और उस दिन उसका असली काम भी।
  \item मुक़ाबला ख़त्म होने पर खिलाड़ी को प्यास लगी है और उस शाम उसका
        \omterm{def:g7:removing-wastes:kidneys}{मूत्र} थोड़ा तथा गहरे रंग का है। दोनों की व्याख्या वापस लेने
        वाले चरण से करो।
  \item इस बीच मुक़ाबले में जमकर काम करती \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} ने ग़ैर-मामूली
        \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} और \omterm{prop:g7:removing-wastes:rubbish}{यूरिया} बनाई। हर एक को उसकी बाहर जाने की
        सरहद दो, और वह चक्कर भी जो उसे वहाँ ले गया।
  \item \omterm{def:g7:removing-wastes:kidneys}{मूत्र} का बनना घट जाने पर भी \omterm{def:g7:removing-wastes:kidneys}{गुर्दे} मुक़ाबले भर लगातार काम
        क्यों करते रहे? वे तब भी क्या हटा रहे थे?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- जाँच-पड़ताल बंद।}
\begin{enumerate}[resume]
  \item अगली सुबह उस गिलास का कुछ हिस्सा साँस के साथ जा चुका है।
        समझाओ कि पानी \omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाओं} पर कैसे बाहर निकलता है
        (\cref{ex:g4:breathing:mirror} ने बरसों पहले यह दिखाया था)।
  \item उस गिलास का पूरा बहीखाता तैयार करो: भीतर आना, भंडार तथा
        इस्तेमाल, और उसके चारों निकास --- \omterm{def:g7:removing-wastes:kidneys}{मूत्र}, \omterm{rem:g7:removing-wastes:sweat}{पसीना}, साँस, और
        बचे-खुचे में गया छोटा हिस्सा।
  \item जाँच-पड़ताल दिखाती है कि एक ही \omterm{prop:g4:journey-of-food:absorb}{रक्त} बारी-बारी से फेफड़ों,
        \omterm{prop:g7:digestion-and-nutrients:absorption}{रसांकुरों}, \omterm{def:g3:bones-and-muscles:muscle}{पेशियों}, \omterm{def:g7:removing-wastes:kidneys}{गुर्दों} और \omterm{def:g1:the-five-senses:sense}{त्वचा} की सेवा करता है। दो
        वाक्यों में बताओ कि इस एक-ही-जाल वाली बनावट के चलते हर अंग
        की सेहत बाक़ी हर अंग का मामला क्यों बन जाती है।
  \item साल का समापन करो: इस पूरी जाँच-पड़ताल का हर ठिकाना आख़िरकार
        किस एक विचार की सेवा में निकला --- उसका नाम बताओ।
        (\cref{prop:g7:what-is-respiration:power} में जवाब है।)
\end{enumerate}
\end{problem}
