\chapter{Classer les animaux}\label{ch:g4:classifying-animals}

Un muséum d'histoire naturelle conserve des millions d'\omterm{def:g1:animals-around-us:animal}{animaux} ---
des tiroirs de scarabées, des salles de \omterm{def:g3:bones-and-muscles:skeleton}{squelettes}, des bocaux de
\omterm{prop:g3:sorting-living-things:groups}{poissons}. Et pourtant un conservateur peut marcher jusqu'à n'importe
lequel d'entre eux en quelques minutes. Le secret n'est pas la
mémoire, c'est l'\emph{ordre}~: chaque \omterm{def:g1:animals-around-us:animal}{animal} a sa place dans un
grand système de \omterm{def:g4:classifying-animals:classification}{groupes emboîtés}, bâti sur les règles de tri de
\cref{ch:g3:sorting-living-things} --- et cette année, nous
apprenons comment ce système fonctionne.

\section{Des groupes dans des groupes}

\begin{definition}[Classification]\label{def:g4:classifying-animals:classification}
\emph{Classer}\index{classer} les \omterm{def:g1:living-or-not:living}{êtres vivants}, c'est les trier
d'après des caractères du corps qu'ils partagent, puis trier ces
groupes en groupes plus grands --- des \emph{groupes emboîtés les
uns dans les autres}\index{groupes emboîtés}, comme des boîtes dans
des boîtes. L'ensemble ainsi obtenu est une
\emph{classification}\index{classification}.
\end{definition}

\begin{example}[Les adresses du chat]\label{ex:g4:classifying-animals:cat}
Un seul chat, plusieurs adresses emboîtées~: le chat est un
\emph{mammifère} (des poils, du \omterm{def:g2:animals-grow-up:born}{lait} pour les petits)~; tout
mammifère est un \emph{\omterm{def:g4:classifying-animals:vertebrate}{vertébré}} (un \omterm{def:g3:bones-and-muscles:skeleton}{squelette} intérieur avec une
\omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale})~; tout \omterm{def:g4:classifying-animals:vertebrate}{vertébré} est un \emph{\omterm{def:g1:animals-around-us:animal}{animal}}. Le chat
dans les \omterm{prop:g3:sorting-living-things:groups}{mammifères}, les \omterm{prop:g3:sorting-living-things:groups}{mammifères} dans les \omterm{def:g4:classifying-animals:vertebrate}{vertébrés}, les
\omterm{def:g4:classifying-animals:vertebrate}{vertébrés} dans les \omterm{def:g1:animals-around-us:animal}{animaux}~: chaque boîte se range dans une plus
grande.
\end{example}

\begin{definition}[Vertébrés et invertébrés]\label{def:g4:classifying-animals:vertebrate}
Un \emph{vertébré}\index{vertébré} est un \omterm{def:g1:animals-around-us:animal}{animal} doté d'un \omterm{def:g3:bones-and-muscles:skeleton}{squelette}
intérieur bâti autour d'une \emph{colonne
vertébrale}\index{colonne vertébrale} --- celle que tu as sentie
dans \cref{ex:g3:bones-and-muscles:feel}. Un \omterm{def:g1:animals-around-us:animal}{animal} sans colonne
vertébrale est un \emph{invertébré}\index{invertébré}~: \omterm{prop:g3:sorting-living-things:groups}{insectes},
araignées, escargots, vers, méduses --- la grande majorité de tous
les \omterm{def:g1:animals-around-us:animal}{animaux}.
\end{definition}

\section{Les cinq classes de vertébrés}

\begin{proposition}[Les cinq classes de vertébrés]\label{prop:g4:classifying-animals:classes}
Les \omterm{def:g4:classifying-animals:vertebrate}{vertébrés} se classent en cinq grands groupes, appelés
\emph{classes}\index{classe (de vertébrés)}~:
\begin{enumerate}
  \item les \emph{\omterm{prop:g3:sorting-living-things:groups}{poissons}}~: des \omterm{def:g1:animals-around-us:covering}{écailles}, des nageoires, une
        respiration dans l'eau --- truite, requin, sardine~;
  \item les \emph{amphibiens}\index{amphibien}~: une \omterm{def:g1:the-five-senses:sense}{peau} nue et
        humide, des \omterm{def:g2:animals-grow-up:born}{œufs} dans l'eau, une enfance de têtard ---
        grenouille, crapaud, triton~;
  \item les \emph{reptiles}\index{reptile}~: une \omterm{def:g1:the-five-senses:sense}{peau} sèche et
        écailleuse, la plupart pondent des \omterm{def:g2:animals-grow-up:born}{œufs} à \omterm{def:g1:animals-around-us:covering}{coquille} sur la
        terre ferme --- lézard, serpent, tortue, crocodile~;
  \item les \emph{\omterm{prop:g3:sorting-living-things:groups}{oiseaux}}~: des \omterm{def:g1:animals-around-us:covering}{plumes}, un bec, des \omterm{def:g2:animals-grow-up:born}{œufs} à
        \omterm{def:g1:animals-around-us:covering}{coquille} --- merle, autruche, manchot~;
  \item les \emph{\omterm{prop:g3:sorting-living-things:groups}{mammifères}}~: des poils, des petits nourris au
        \omterm{def:g2:animals-grow-up:born}{lait} --- chat, baleine, chauve-souris, humain.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Trier une livraison de zoo]\label{ex:g4:classifying-animals:zoo}
Six caisses arrivent~: un python (\omterm{def:g1:the-five-senses:sense}{peau} sèche et écailleuse, \omterm{def:g2:animals-grow-up:born}{œufs} à
\omterm{def:g1:animals-around-us:covering}{coquille} --- reptile), un crapaud (\omterm{def:g1:the-five-senses:sense}{peau} nue et humide ---
amphibien), un manchot (des \omterm{def:g1:animals-around-us:covering}{plumes} --- oiseau), un requin (nageoires
et branchies --- poisson), un dauphin (du \omterm{def:g2:animals-grow-up:born}{lait} pour son petit ---
mammifère), et une caisse d'escargots géants (aucune \omterm{def:g4:classifying-animals:vertebrate}{colonne
vertébrale} --- des \omterm{def:g4:classifying-animals:vertebrate}{invertébrés}, dans aucune classe de \omterm{def:g4:classifying-animals:vertebrate}{vertébrés}).
\end{example}

\begin{remark}[Reptile ou amphibien~?]\label{rem:g4:classifying-animals:confusion}
Le lézard et le triton se ressemblent au premier coup d'\omterm{def:g1:the-five-senses:sense}{œil} ---
quatre pattes, une longue queue. C'est la \omterm{def:g1:the-five-senses:sense}{peau} qui décide~: celle du
lézard est sèche et écailleuse, celle du triton nue et humide~; les
\omterm{def:g2:animals-grow-up:born}{œufs} du lézard ont une \omterm{def:g1:animals-around-us:covering}{coquille} et sont pondus sur la terre ferme,
ceux du triton sont pondus nus dans l'eau. Même silhouette, classes
différentes.
\end{remark}

\section{Se servir d'une classification}

\begin{method}[Placer un vertébré dans sa classe]\label{met:g4:classifying-animals:place}
Pose les questions dans l'ordre, en t'arrêtant au premier oui~:
\begin{enumerate}
  \item Des \omterm{def:g1:animals-around-us:covering}{plumes}~? --- un oiseau.
  \item Des poils (ou du \omterm{def:g2:animals-grow-up:born}{lait} pour les petits)~? --- un mammifère.
  \item Une \omterm{def:g1:the-five-senses:sense}{peau} sèche et écailleuse~? --- un reptile.
  \item Une \omterm{def:g1:the-five-senses:sense}{peau} nue et humide, et une enfance de têtard~? --- un
        amphibien.
  \item Des nageoires et des branchies, une vie dans l'eau~? --- un
        poisson.
\end{enumerate}
Aucune \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale}~? Alors c'était un \omterm{def:g4:classifying-animals:vertebrate}{invertébré}, et les
questions ne s'appliquent pas.
\end{method}

\begin{proposition}[Ce qu'une classe t'apprend gratuitement]\label{prop:g4:classifying-animals:free}
Connaître la classe d'un \omterm{def:g1:animals-around-us:animal}{animal}, c'est connaître tout un paquet de
faits avant de l'avoir vu~: si l'on te dit seulement «~c'est un
mammifère~», tu t'attends déjà à des poils, à du \omterm{def:g2:animals-grow-up:born}{lait} pour les
petits et --- comme presque tous les \omterm{prop:g3:sorting-living-things:groups}{mammifères} --- à des petits nés
vivants. Cette économie est tout l'intérêt de la \omterm{def:g4:classifying-animals:classification}{classification}~:
apprends le groupe une fois, et chaque \omterm{def:g1:my-body:parts}{membre} arrive à moitié connu.
\end{proposition}
\admitted

\begin{example}[Une nouvelle espèce arrive]\label{ex:g4:classifying-animals:new}
Les naturalistes découvrent encore des \omterm{def:g1:animals-around-us:animal}{animaux} jamais vus. Quand on
trouve un nouvel \omterm{def:g1:animals-around-us:animal}{animal} proche de la chauve-souris, avec des poils
et du \omterm{def:g2:animals-grow-up:born}{lait}, il va tout droit chez les \omterm{prop:g3:sorting-living-things:groups}{mammifères} --- et aussitôt les
biologistes s'attendent à une \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale}, à un cœur à quatre
cavités et à du sang chaud, avant même d'avoir vérifié quoi que ce
soit. La \omterm{def:g4:classifying-animals:classification}{classification} rapporte des prédictions.
\end{example}

\begin{remark}[Vers l'arbre de parenté]\label{rem:g4:classifying-animals:tree}
Pourquoi les poils, le \omterm{def:g2:animals-grow-up:born}{lait} et la \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale} voyagent-ils
si fidèlement ensemble~? La réponse complète --- que les \omterm{def:g1:my-body:parts}{membres}
d'un groupe héritent leurs caractères communs d'ancêtres communs ---
appartient à l'histoire de la parenté et de l'évolution, reprise
dans des années ultérieures. Pour l'instant, les boîtes emboîtées
fonctionnent, et fonctionnent à merveille.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:1}
Qu'est-ce qu'un \omterm{def:g4:classifying-animals:vertebrate}{vertébré}~? Qu'est-ce qu'un \omterm{def:g4:classifying-animals:vertebrate}{invertébré}~? Donne deux
exemples de chacun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:2}
Nomme les cinq classes de \omterm{def:g4:classifying-animals:vertebrate}{vertébrés} avec leurs caractères
définissants.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:3}
Place chaque \omterm{def:g1:animals-around-us:animal}{animal}~: la sardine, le crocodile, le triton,
l'autruche, la baleine, le ver de terre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:4}
Écris les adresses emboîtées du chat, de la plus petite boîte à la
plus grande, comme dans \cref{ex:g4:classifying-animals:cat}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:5}
Un lézard et un triton se ressemblent. Quels deux caractères
séparent leurs classes~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:6}
Quel groupe rassemble la grande majorité de tous les \omterm{def:g1:animals-around-us:animal}{animaux}~? Nomme
quatre de ses \omterm{def:g1:my-body:parts}{membres}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:7}
Applique \cref{met:g4:classifying-animals:place} à voix haute au
manchot puis au dauphin, question après question.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:8}
On te dit seulement~: «~cet \omterm{def:g1:animals-around-us:animal}{animal} est un oiseau~». Énumère trois
choses auxquelles tu t'attends désormais sans regarder, d'après
\cref{prop:g4:classifying-animals:free}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:9}
La baleine n'a presque pas de poils et vit comme un poisson. À
l'aide de \cref{met:g4:classifying-animals:place} et de
\cref{ex:g3:sorting-living-things:surprises}, explique pourquoi
c'est malgré tout un mammifère --- et pourquoi «~vit dans la mer~»
n'a jamais été un caractère de \omterm{def:g4:classifying-animals:classification}{classification}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:10}
Le serpent n'a pas de pattes, le ver de terre non plus. Pourquoi
l'un est-il un reptile \omterm{def:g4:classifying-animals:vertebrate}{vertébré} et l'autre un \omterm{def:g4:classifying-animals:vertebrate}{invertébré}~? Quel
caractère unique l'emporte sur les pattes manquantes~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:classifying-animals:11}
L'ornithorynque a intrigué les naturalistes~: des poils et du \omterm{def:g2:animals-grow-up:born}{lait},
et pourtant il pond des \omterm{def:g2:animals-grow-up:born}{œufs} à \omterm{def:g1:animals-around-us:covering}{coquille}. Quelle classe l'a emporté,
et pourquoi~? Que montre cette énigme sur l'intérêt de \omterm{def:g4:classifying-animals:classification}{classer} par
\emph{paquets} de caractères plutôt que par un seul~?
\end{exercise}
