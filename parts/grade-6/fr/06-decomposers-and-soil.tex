\chapter{Les décomposeurs et le sol}\label{ch:g6:decomposers-and-soil}

Chaque automne, le bois laisse tomber un épais tapis de \omterm{def:g1:plants-around-us:parts}{feuilles} ---
des tonnes à l'hectare. Et pourtant, parcours le même bois chaque
printemps~: tu ne patauges pas dans une mer de litière qui monte~;
le tapis est de nouveau mince, année après année. Quelqu'un mange
les morts de la forêt. Ce chapitre descend dans le \omterm{def:g6:decomposers-and-soil:soil}{sol} à la
rencontre des mangeurs --- et referme enfin le cercle de la matière.

\section{Le sol est vivant}

\begin{definition}[Le sol]\label{def:g6:decomposers-and-soil:soil}
Le \emph{sol}\index{sol} est la \omterm{def:g1:the-five-senses:sense}{peau} vivante de la terre~: un
mélange de grains \emph{minéraux} (roche broyée, du sable à
l'argile), de restes \emph{organiques} à tous les stades de
\omterm{prop:g6:decomposers-and-soil:decomposition}{décomposition}, d'eau, d'air --- et d'une énorme population d'\omterm{def:g6:exploring-our-environment:components}{êtres
vivants}. Sa couche supérieure sombre, la plus riche en matière
décomposée, est l'\emph{humus}\index{humus} que les jardiniers
prisent.
\end{definition}

\begin{example}[Une poignée, recensée]\label{ex:g6:decomposers-and-soil:handful}
Une poignée de terre de bois, triée sur un plateau blanc~: des
grains minéraux~; des filaments de \omterm{def:g1:plants-around-us:parts}{racines}~; des \omterm{def:g1:plants-around-us:parts}{demi-feuilles}, des
\omterm{def:g3:bones-and-muscles:skeleton}{squelettes} de \omterm{def:g1:plants-around-us:parts}{feuilles} et des miettes noires --- de la \omterm{def:g6:plants-are-producers:matter}{matière
organique} en cours de descente~; et le recensement --- un ver de
terre, deux cloportes, un mille-pattes, de minuscules collemboles
blancs, des feutrages de \omterm{def:g6:decomposers-and-soil:decomposers}{champignons} fins comme des fils. Et le
plateau ne montre que le visible~: un seul gramme de terre abrite
plus d'\omterm{def:g6:exploring-our-environment:components}{êtres vivants} trop petits pour être vus qu'il n'y a
d'habitants dans une grande ville.
\end{example}

\begin{definition}[Les décomposeurs]\label{def:g6:decomposers-and-soil:decomposers}
Les \emph{décomposeurs}\index{décomposeur} sont les \omterm{def:g6:exploring-our-environment:components}{êtres vivants}
qui se nourrissent de \emph{\omterm{def:g6:plants-are-producers:matter}{matière organique} morte} --- \omterm{def:g1:plants-around-us:parts}{feuilles}
tombées, bois mort, excréments, cadavres~:
\begin{itemize}
  \item l'équipe visible~: vers de terre, cloportes, mille-pattes,
        collemboles, larves de mouches, bousiers~;
  \item les \emph{champignons}\index{champignon}~: moisissures et
        champignons à chapeau, dont le corps nourricier est un
        feutrage de fils blancs étalé dans le \omterm{def:g6:decomposers-and-soil:soil}{sol} et le bois mort
        --- le champignon de \cref{exo:g1:living-or-not:10} n'en
        est que le \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}~;
  \item et, pour finir chaque travail, une vie du \omterm{def:g6:decomposers-and-soil:soil}{sol} trop petite
        pour être vue --- les démonteurs ultimes.
\end{itemize}
Ce sont les travailleurs de l'ombre de
\cref{def:g5:food-webs:roles}, à qui un chapitre était promis~: le
voici.
\end{definition}

\section{La décomposition~: de l'organique au minéral}

\begin{proposition}[Ce que fait la décomposition]\label{prop:g6:decomposers-and-soil:decomposition}
La \emph{décomposition}\index{décomposition} est l'alimentation des
\omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs}, et son résultat est l'inverse du métier des
\omterm{def:g5:food-webs:roles}{producteurs}~: la matière \emph{organique} morte est démontée, étape
par étape et mangeur après mangeur, jusqu'à ce que sa substance
retourne au monde \emph{minéral} --- \omterm{def:g6:decomposers-and-soil:soil}{humus} friable, puis \omterm{prop:g4:what-plants-need:needs}{sels
minéraux} dissous dans l'eau du \omterm{def:g6:decomposers-and-soil:soil}{sol}, et gaz rendu à l'air. Le tapis
d'automne devient, au printemps, exactement ce que boivent les
\omterm{def:g1:plants-around-us:parts}{racines} (\cref{prop:g4:what-plants-need:needs}).
\end{proposition}
\admitted

\begin{example}[La dernière année d'une feuille]\label{ex:g6:decomposers-and-soil:leaf}
Suis une \omterm{def:g1:plants-around-us:parts}{feuille} de chêne vers le bas. Novembre~: tombée, entière.
Hiver~: ramollie, minée par les fils des \omterm{def:g6:decomposers-and-soil:decomposers}{champignons}, grignotée
jusqu'à n'être qu'un \omterm{def:g3:bones-and-muscles:skeleton}{squelette} de dentelle par les collemboles et
les cloportes. Printemps~: entraînée sous terre par un ver, mangée,
et rejetée en miettes travaillées par le ver. Été~: ses derniers
fragments sont de l'\omterm{def:g6:decomposers-and-soil:soil}{humus} noir, et ses minéraux voyagent dans l'eau
du \omterm{def:g6:decomposers-and-soil:soil}{sol} --- jusque dans une \omterm{def:g1:plants-around-us:parts}{racine}, peut-être celle du chêne
lui-même. Mangée quatre fois, la \omterm{def:g1:plants-around-us:parts}{feuille} est devenue nourriture pour
son propre arbre.
\end{example}

\begin{remark}[Le cercle se referme]\label{rem:g6:decomposers-and-soil:circle}
Avec les \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs} à leur place, l'histoire de la matière de
cette année se referme en roue~: les \omterm{def:g5:food-webs:roles}{producteurs} bâtissent
l'organique à partir du minéral
(\cref{ch:g6:plants-are-producers})~; les \omterm{def:g5:food-webs:roles}{consommateurs} se le
passent (\cref{ch:g6:matter-from-food})~; les \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs} le
rendent au minéral, prêt pour les \omterm{def:g5:food-webs:roles}{producteurs} de nouveau. La matière
ne s'épuise pas --- elle tourne. Les atomes du papier de cette page
ont été \omterm{def:g1:plants-around-us:parts}{feuille}, \omterm{def:g6:decomposers-and-soil:soil}{humus}, \omterm{def:g1:plants-around-us:parts}{racine} puis \omterm{def:g1:plants-around-us:parts}{feuille} encore, plus de fois
qu'on ne peut les compter.
\end{remark}

\section{Travailler avec les décomposeurs}

\begin{example}[Le compost, enfin expliqué]\label{ex:g6:decomposers-and-soil:compost}
Le composteur de \cref{ex:g2:caring-for-nature:compost} se lit
maintenant comme un appareil~: un atelier de \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs}
approvisionné en épluchures. Vers, cloportes et équipes invisibles
démontent les déchets organiques de la cuisine en \omterm{def:g6:decomposers-and-soil:soil}{humus} sombre~; le
jardinier l'épand, et ses minéraux nourrissent les planches
(\cref{rem:g4:what-plants-need:soil}). Le tas chauffe même tout
seul --- la combustion propre des \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs}
(\cref{prop:g6:matter-from-food:losses}) chauffe leur atelier.
\end{example}

\begin{proposition}[Ce qui accélère et ce qui ralentit le travail]\label{prop:g6:decomposers-and-soil:speed}
Les \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs} sont des \omterm{def:g6:exploring-our-environment:components}{êtres vivants} avec leurs \omterm{def:g6:exploring-our-environment:conditions}{conditions}
(\cref{prop:g6:exploring-our-environment:decide})~: ils travaillent
le plus vite au \emph{chaud}, à l'\emph{humide} et à l'\emph{aéré},
et s'arrêtent presque quand il fait froid, sec, ou noyé sans air.
D'où le tapis d'hiver qui traîne et le nettoyage rapide du
printemps~; d'où aussi la recette du compost du jardinier ---
humide, retourné pour l'air, jamais une masse détrempée et
étanche.
\end{proposition}
\admitted

\begin{method}[Lire la santé d'un sol]\label{met:g6:decomposers-and-soil:read}
Pour juger un \omterm{def:g6:decomposers-and-soil:soil}{sol} à la manière du naturaliste~:
\begin{enumerate}
  \item creuse sur une main de profondeur et regarde~: un \omterm{def:g6:decomposers-and-soil:soil}{humus}
        sombre et friable, ou une poussière pâle et sans vie~?
  \item recense une poignée sur un plateau blanc --- vers,
        cloportes, collemboles~: une équipe riche veut dire que le
        cycle tourne~;
  \item vérifie la surface~: la matière morte disparaît-elle au fil
        des saisons, ou s'entasse-t-elle sans pourrir~?
  \item traite l'équipe comme un cheptel~: nourris-la (restes
        enfouis, compost, paillis), garde-la humide, épargne-lui
        les poisons.
\end{enumerate}
\end{method}

\begin{example}[Le ver de terre, jardinier des jardiniers]\label{ex:g6:decomposers-and-soil:worm}
Le ver de terre fait double service~: \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeur} --- il entraîne
les \omterm{def:g1:plants-around-us:parts}{feuilles} vers le bas et mange la matière décomposée --- et
\emph{laboureur}, ses galeries aérant le \omterm{def:g6:decomposers-and-soil:soil}{sol} et ses turricules
pétrissant ensemble le minéral et l'organique. Un seul pré peut
porter plus de poids de vers sous l'herbe que de poids de vaches
au-dessus~; les vieux naturalistes appelaient le ver le véritable
fabricant des terres cultivées, et ils avaient raison.
\end{example}

\section{Problème~: les sacs de feuilles}

\begin{problem}[{Problème du week-end --- des sacs à mailles, des
feuilles enterrées et un hiver de science du sol}]\label{pb:g6:decomposers-and-soil:1}
La classe enterre des sacs de \omterm{def:g1:plants-around-us:parts}{feuilles} dans le talus de la haie de
l'école en novembre~: des poids égaux de \omterm{def:g1:plants-around-us:parts}{feuilles} de chêne fraîches
dans des sacs à mailles, à déterrer et à repeser tout au long de
l'année. Trois types de sacs~: mailles larges (qui laissent entrer
les vers, les cloportes, tout le monde), mailles fines (qui excluent
l'équipe visible et n'admettent que les invisibles) et plastique
scellé (qui exclut tout le monde, et l'air en plus).

\medskip\noindent\textbf{Partie I --- Le montage.}
\begin{enumerate}
  \item Pourquoi tous les sacs doivent-ils contenir le \emph{même}
        poids des \emph{mêmes} \omterm{def:g1:plants-around-us:parts}{feuilles}, enterrés le \emph{même}
        jour dans le \emph{même} talus~? Nomme la méthode.
  \item Quelle question pose le sac à mailles fines, que le sac à
        mailles larges ne peut pas poser à lui seul~?
  \item Une équipe veut un quatrième sac~: mailles larges, mais
        enterré dans le gravier sec de la cour. Quelle question
        poserait-il, d'après
        \cref{prop:g6:decomposers-and-soil:speed}~?
  \item Prédis, avant de déterrer~: classe les trois sacs du talus
        selon la quantité de poids de \omterm{def:g1:plants-around-us:parts}{feuilles} qu'ils auront perdue
        en juin.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les résultats.}
\begin{enumerate}[resume]
  \item Pesées de juin~: mailles larges --- le plus de poids parti,
        \omterm{def:g1:plants-around-us:parts}{feuilles} réduites à des \omterm{def:g3:bones-and-muscles:skeleton}{squelettes} et à des miettes~;
        mailles fines --- nettement allégé, \omterm{def:g1:plants-around-us:parts}{feuilles} ramollies et
        noircies mais plus lentement~; plastique scellé --- poids
        presque inchangé, contenu en boue aigre. Interprète chaque
        sac.
  \item Dans le sac à mailles larges, la classe trouve des
        cloportes, un ver, des collemboles et des fils blancs de
        \omterm{def:g6:decomposers-and-soil:decomposers}{champignon}. Attribue à chacun son rôle dans le démontage de
        la \omterm{def:g1:plants-around-us:parts}{feuille} (\cref{ex:g6:decomposers-and-soil:leaf} sert de
        modèle).
  \item Le sac à mailles fines a perdu du poids réel alors que tout
        \omterm{def:g1:animals-around-us:animal}{animal} visible en était exclu. Que cela prouve-t-il sur la
        vie du \omterm{def:g6:decomposers-and-soil:soil}{sol} invisible~?
  \item Le sac de la cour en gravier (celui de la question 3) a
        beaucoup moins perdu que son jumeau du talus. Quelles
        \omterm{def:g6:exploring-our-environment:conditions}{conditions} de travail le gravier refuse-t-il~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La leçon plus large.}
\begin{enumerate}[resume]
  \item Où est \emph{passé} le poids perdu par le sac à mailles
        larges~? Réponds avec
        \cref{prop:g6:decomposers-and-soil:decomposition} --- deux
        destinations au moins.
  \item Les orties du printemps poussent les plus vertes exactement
        le long du talus où la classe a enterré ses sacs. Une
        coïncidence~? Argumente à partir du cycle de la matière.
  \item Sers-toi du sac scellé pour expliquer pourquoi des déchets
        organiques enfouis en décharge, emballés hermétiquement
        dans du plastique, persistent des dizaines d'années alors
        que les mêmes déchets se compostent en quelques mois.
  \item Conclus en deux phrases~: qu'a prouvé l'hiver sur ceux qui
        nettoient le tapis de la forêt, et que nourrit ce
        nettoyage~?
\end{enumerate}
\end{problem}
