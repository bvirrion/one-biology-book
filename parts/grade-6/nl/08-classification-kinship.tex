\chapter{Classificatie en verwantschap}\label{ch:g6:classification-kinship}

Waarom werkt de \omterm{def:g4:classifying-animals:classification}{classificatie} uit \cref{ch:g4:classifying-animals} zo
goed? Deel \omterm{def:g1:animals-around-us:animal}{dieren} in naar \omterm{def:g1:animals-around-us:covering}{veren} en \omterm{def:g1:animals-around-us:covering}{vacht}, en duizend andere \omterm{def:g6:classification-kinship:attribute}{kenmerken}
vallen gehoorzaam op hun plaats --- melkgevers hebben een \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat},
verendragers leggen eieren met een schaal, en niemand hoeft het na te
kijken. \Cref{rem:g4:classifying-animals:tree} liet die vraag in de lucht
hangen. Dit hoofdstuk beantwoordt haar met één woord --- \omterm{prop:g6:classification-kinship:kinship}{verwantschap} ---
en bouwt de dozen om tot een \omterm{def:g6:classification-kinship:tree}{stamboom}.

\section{Kenmerken, gedeeld en genest}

\begin{definition}[Kenmerk]\label{def:g6:classification-kinship:attribute}
Een \emph{kenmerk}\index{kenmerk} is een eigenschap die een organisme
\emph{bezit}, zo geformuleerd dat je haar kunt nakijken: heeft een \omterm{def:g3:bones-and-muscles:skeleton}{skelet}
met een \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}; heeft vier \omterm{def:g1:my-body:parts}{ledematen}; heeft \omterm{def:g1:animals-around-us:covering}{vacht} en zoogt zijn
jongen; heeft \omterm{def:g1:animals-around-us:covering}{veren}. De \omterm{def:g4:classifying-animals:classification}{classificatie} is op kenmerken gebouwd --- de
goede criteria uit \cref{def:g3:sorting-living-things:criterion},
formeel gemaakt.
\end{definition}

\begin{proposition}[Kenmerken nesten]\label{prop:g6:classification-kinship:nest}
Over veel \omterm{def:g5:biodiversity:species}{soorten} nagekeken vallen \omterm{def:g6:classification-kinship:attribute}{kenmerken} in \emph{geneste}
verzamelingen uiteen --- elk \omterm{def:g1:animals-around-us:animal}{dier} met \omterm{def:g1:animals-around-us:covering}{vacht} heeft ook vier \omterm{def:g1:my-body:parts}{ledematen}; elk
viervoetig \omterm{def:g1:animals-around-us:animal}{dier} heeft ook een \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}; nooit het omgekeerde patroon
(\omterm{def:g1:animals-around-us:covering}{vacht} zonder \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}) en nooit kruisende verzamelingen (\omterm{def:g1:animals-around-us:covering}{vacht} en
\omterm{def:g1:animals-around-us:covering}{veren} tegelijk). Dat nesten is de reden dat groepen-binnen-groepen
(\cref{def:g4:classifying-animals:classification}) überhaupt bij de
levende wereld passen.
\end{proposition}
\admitted

\begin{example}[Drie dieren plaatsen]\label{ex:g6:classification-kinship:placing}
Loop de dozen na voor de walvis, de forel en de merel. Walvis:
\omterm{def:g4:classifying-animals:vertebrate}{ruggengraat} ja, vier \omterm{def:g1:my-body:parts}{ledematen} ja (voorflippers, en verborgen resten
achteraan), vacht-en-melk ja --- de binnenste doos, samen met de kat.
Forel: \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat} ja, stop --- alleen de buitenste doos. Merel:
\omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}, vier \omterm{def:g1:my-body:parts}{ledematen} (twee poten, twee vleugels), \omterm{def:g1:animals-around-us:covering}{veren} --- de
doos van de viervoetigen, in het eigen vakje van de \omterm{def:g1:animals-around-us:covering}{veren}. Geen enkel
\omterm{def:g1:animals-around-us:animal}{dier} vraagt ooit om twee kruisende dozen tegelijk.
\end{example}

\section{Verwantschap: de reden achter het nesten}

\begin{proposition}[Gedeelde kenmerken betekenen gedeelde voorouders]\label{prop:g6:classification-kinship:kinship}
De verklaring van het nesten is de overerving
(\cref{rem:g4:animal-reproduction:mix}: jongen krijgen iets van hun
ouders mee). \omterm{def:g5:biodiversity:species}{Soorten} die een \omterm{def:g6:classification-kinship:attribute}{kenmerk} delen, delen het omdat ze het
\emph{van een gemeenschappelijke voorouder geërfd hebben} --- een verre
grootoudersoort. Hoe meer \omterm{def:g6:classification-kinship:attribute}{kenmerken} twee \omterm{def:g5:biodiversity:species}{soorten} delen, hoe nauwer hun
\emph{verwantschap}\index{verwantschap}: de kat en de walvis, allebei
melkgevers met \omterm{def:g1:animals-around-us:covering}{vacht}, zijn nauwere neven van elkaar dan elk van beide van
de forel is. Een classificatiegroep is in de grond een \emph{familie}:
een voorouder en al zijn nakomelingen.
\end{proposition}
\admitted

\begin{example}[De walvis eindelijk goed gelezen]\label{ex:g6:classification-kinship:whale}
Dit is de reden dat het walvisraadsel
(\cref{exo:g4:classifying-animals:9}) wel zo moest aflopen als het
afliep: vacht-en-melk markeert de familielijn van de walvis --- haar
voorouders waren melkgevende landzoogdieren --- terwijl haar visachtige
vorm het werk van het water is
(\cref{exo:g4:adapted-to-their-world:11}), door dezelfde manier van leven
op een niet-verwante familie gestempeld. \omterm{def:g6:classification-kinship:attribute}{Kenmerken} van afstamming delen
families in; gelijkenissen van \omterm{def:g2:where-animals-live:habitat}{leefgebied} kunnen door vreemden geleend
worden. De \omterm{def:g4:classifying-animals:classification}{classificatie} vertrouwt de eerste en wantrouwt de tweede.
\end{example}

\begin{definition}[De stamboom van de verwantschap]\label{def:g6:classification-kinship:tree}
\omterm{prop:g6:classification-kinship:kinship}{Verwantschap} wordt getekend als een \emph{stamboom}\index{stamboom}:
vertakkingspunten staan voor gemeenschappelijke voorouders, en elk
\omterm{def:g6:classification-kinship:attribute}{kenmerk} verschijnt één keer, bij de vertakking waar het is ontstaan ---
elke \omterm{def:g5:biodiversity:species}{soort} voorbij die splitsing erft het. De geneste dozen en de
stamboom dragen dezelfde informatie: een doos is een tak met al zijn
twijgen.
\end{definition}

\begin{method}[Een kleine stamboom bouwen]\label{met:g6:classification-kinship:build}
Bij een handvol \omterm{def:g5:biodiversity:species}{soorten}:
\begin{enumerate}
  \item zet hun \omterm{def:g6:classification-kinship:attribute}{kenmerken} in een tabel --- \omterm{def:g5:biodiversity:species}{soorten} in de rijen, na te
        kijken \omterm{def:g6:classification-kinship:attribute}{kenmerken} in de kolommen, vinkjes waar ze het bezitten;
  \item groepeer op gedeelde vinkjes, het wijdste \omterm{def:g6:classification-kinship:attribute}{kenmerk} eerst: het
        nesten verschijnt in de tabel;
  \item teken de \omterm{def:g6:classification-kinship:tree}{stamboom}: één splitsing per \omterm{def:g6:classification-kinship:attribute}{kenmerk}, de wijdste het
        dichtst bij de stam; zet elke \omterm{def:g5:biodiversity:species}{soort} voorbij precies die
        splitsingen waarvan ze de \omterm{def:g6:classification-kinship:attribute}{kenmerken} bezit;
  \item lees de \omterm{prop:g6:classification-kinship:kinship}{verwantschap} af: hoe nauw twee \omterm{def:g5:biodiversity:species}{soorten} verwant zijn, is
        de laatste splitsing die ze delen.
\end{enumerate}
\end{method}

\begin{remark}[Wat de stamboom stilletjes impliceert]\label{rem:g6:classification-kinship:implies}
Neem de \omterm{def:g6:classification-kinship:tree}{stamboom} serieus en hij zegt iets enorms: volg twee willekeurige
takken terug en ze \emph{komen bij elkaar}. Kat en forel delen een
grootoudersoort; en dus, nog verder terug, ook kat en slak, madeliefje en
kind. Alle \omterm{def:g6:exploring-our-environment:components}{levende wezens} zouden dan één familie zijn, en hun eenheid
(\cref{prop:g5:first-look-at-cells:principle}) een familiegelijkenis ---
en \omterm{def:g5:biodiversity:species}{soorten} moeten dan over generaties heen kunnen \emph{veranderen}, want
anders had het vertakken nooit kunnen gebeuren. Of dat waar is, en hoe
het zou kunnen werken, is de grote vraag van het jaar van
\cref{ch:g9:evidence-of-evolution} --- deze \omterm{def:g6:classification-kinship:tree}{stamboom} is de tekening die
haar afdwingt.
\end{remark}

\section{Probleem: De museumla}

\begin{problem}[{Weekendprobleem --- zes exemplaren, één kenmerkentabel,
één stamboom}]\label{pb:g6:classification-kinship:1}
Een museum leent de klas een la met zes exemplaren: een forel, een
kikker, een hagedis, een duif, een muis, en een dolfijnenschedel met
afgietsels van de flippers. De opdracht: deel ze in naar \omterm{def:g6:classification-kinship:attribute}{kenmerken} en
teken hun \omterm{def:g6:classification-kinship:tree}{stamboom}.

\medskip\noindent\textbf{Deel I --- De tabel.}
\begin{enumerate}
  \item Bouw de tabel voor de zes, met de \omterm{def:g6:classification-kinship:attribute}{kenmerken}: \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}; vier
        \omterm{def:g1:my-body:parts}{ledematen}; droge schubbige \omterm{def:g1:the-five-senses:sense}{huid}; \omterm{def:g1:animals-around-us:covering}{veren}; \omterm{def:g1:animals-around-us:covering}{vacht} en \omterm{def:g2:animals-grow-up:born}{melk}. (Voor de
        vrijwel kale dolfijn staat het museumlabel ervoor in: jongen die
        \omterm{def:g2:animals-grow-up:born}{melk} krijgen ---
        \cref{ex:g3:sorting-living-things:surprises} legt uit waarom die
        getuigenis de zaak beslist.)
  \item Welk \omterm{def:g6:classification-kinship:attribute}{kenmerk} delen alle zes? Wat betekent de wijdste deling, in
        termen van \omterm{prop:g6:classification-kinship:kinship}{verwantschap}?
  \item Welke exemplaren krijgen een vinkje bij ``vier \omterm{def:g1:my-body:parts}{ledematen}''?
        Verklaar de vinkjes van de duif en de dolfijn --- wat zijn
        vleugels en flippers, volgens de logica van de \omterm{def:g6:classification-kinship:tree}{stamboom}?
  \item Waarom komt ``leeft in het water'' nergens in de tabel voor,
        hoewel forel en dolfijn dat allebei doen? Noem de regel van het
        hoofdstuk.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De \omterm{def:g6:classification-kinship:tree}{stamboom}.}
\begin{enumerate}[resume]
  \item Teken de \omterm{def:g6:classification-kinship:tree}{stamboom}: zet de splitsingen op volgorde ---
        \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}, vier \omterm{def:g1:my-body:parts}{ledematen}, en dan de bekledingen (\omterm{def:g1:animals-around-us:covering}{schubben},
        \omterm{def:g1:animals-around-us:covering}{veren}, \omterm{def:g1:animals-around-us:covering}{vacht}) op hun eigen takken --- en plaats alle zes.
  \item Lees van je \omterm{def:g6:classification-kinship:tree}{stamboom} af: de nauwste neef van de muis in de la,
        en die van de forel.
  \item De kikker staat voorbij ``vier \omterm{def:g1:my-body:parts}{ledematen}'' maar vóór elke
        bekledingssplitsing. Waar markeert zijn kale, vochtige \omterm{def:g1:the-five-senses:sense}{huid}
        (\cref{rem:g4:classifying-animals:confusion}) hem als, en waar
        gaat zijn tak eruit?
  \item Van een zevende exemplaar valt het label eraf: droge schubbige
        \omterm{def:g1:the-five-senses:sense}{huid}, vier \omterm{def:g1:my-body:parts}{ledematen}, \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}. Plaats het op de \omterm{def:g6:classification-kinship:tree}{stamboom} en
        noem zijn klasse --- zonder het label te zien.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De betekenis.}
\begin{enumerate}[resume]
  \item De dolfijn en de forel delen hun stroomlijn; de dolfijn en de
        muis delen de vacht-en-melkafstamming. Welk paar zit dichter bij
        elkaar op de \omterm{def:g6:classification-kinship:tree}{stamboom}, en waarom moet de \omterm{def:g6:classification-kinship:tree}{stamboom} de stroomlijn
        negeren?
  \item Zeg wat de splitsing ``vier \omterm{def:g1:my-body:parts}{ledematen}'' historisch beweert: van
        welk ene ding, lang geleden, stammen kikker, hagedis, duif, muis
        en dolfijn allemaal af?
  \item Op de \omterm{def:g6:classification-kinship:tree}{stamboom} van de klas komt elke tak uiteindelijk bij de
        stam samen. Zeg in één zin wat die tekening stilletjes beweert
        over alle zes de exemplaren --- en over de klasgenoten die haar
        tekenen.
  \item Een scepticus zegt: ``jullie \omterm{def:g6:classification-kinship:tree}{stamboom} is gewoon een
        archiefsysteem''. Geef het tweeledige antwoord: waarover het
        nesten van \omterm{def:g6:classification-kinship:attribute}{kenmerken} een \emph{feit} is, en wat de \omterm{def:g6:classification-kinship:tree}{stamboom}
        \emph{verklaart} dat louter archiveren onverklaard zou laten.
\end{enumerate}
\end{problem}
