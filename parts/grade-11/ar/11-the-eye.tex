\chapter{العين ومستقبلاتها الضوئية}\label{ch:g11:the-eye}

في ليلة بلا قمر، بعيدًا عن أي مصباح، ترى الدرب ولا ترى ألوانه؛ وسترة
صديقك الحمراء شكل رمادي. وانظر مباشرةً إلى نجم خافت فيختفي؛ وانظر إلى
جانبه قليلًا فيعود. وكلتا الغرابتين تأتيان من الشبكية، وهي الصفيحة
الرقيقة في مؤخرة \omterm{def:g11:the-eye:parts}{العين} حيث يصير الضوء إشارة عصبية: صنفان من \omterm{def:g10:cells-common-unit:cell}{الخلايا}
المستقبلة، غير متساويي التوزع، أحدهما أعمى عن اللون والآخر أعمى في
الظلام. ويفكك هذا الفصل \omterm{def:g11:the-eye:parts}{العين} من القرنية إلى العصب البصري، ويقرأ، في
\omterm{def:g10:universal-dna:gene}{مورّثات} أصبغة الشبكية، فصلًا من تاريخنا التطوري.

\section{العين أداةً بصرية}

\begin{definition}[أجزاء العين]\label{def:g11:the-eye:parts}
\emph{العين}\index{عين} كرة قطرها نحو \qty{24}{mm}. يدخل الضوء عبر
\emph{القرنية} الشفافة، ويمر عبر \emph{البؤبؤ} --- وهو الفتحة في
\emph{القزحية} الملونة، وهو يتسع في الظلام ويضيق في الضوء الساطع ---
وعبر \emph{العدسة}، ويعبر هلام \emph{الجسم الزجاجي} الصافي، ويبلغ
\emph{الشبكية}، وهي الطبقة الحساسة للضوء المبطنة لمؤخرة الكرة.
وتكوّن القرنية والعدسة معًا، على الشبكية، صورة صغيرة مقلوبة للمشهد؛
والعدسة، وتستطيع عضلات تغيير تحدبها، تضبط البؤرة من الأشياء البعيدة
إلى القريبة (\emph{المطابقة}). ويغادر \emph{العصب البصري} مؤخرة العين
حاملًا إشارات الشبكية إلى الدماغ.
\end{definition}

\begin{example}[البؤرة وعيوبها]\label{ex:g11:the-eye:focus}
\omterm{def:g11:the-eye:parts}{العين} المرتاحة تبئر الأشياء البعيدة على الشبكية؛ وللقراءة على بعد
\qty{25}{cm} يجب أن تنتفخ العدسة. \omterm{def:g11:the-eye:parts}{والعين} الأطول قليلًا مما ينبغي تبئر
الأشياء البعيدة أمام الشبكية --- وهو قصر البصر، يصحح بعدسة نظارة
مفرقة؛ \omterm{def:g11:the-eye:parts}{والعين} الأقصر مما ينبغي، أو العدسة التي تصلبت مع العمر، لا
تستطيع تبئير الأشياء القريبة --- وهو طول البصر، يصحح بعدسة جامعة. وفي
كل حالة تكون الشبكية سليمة: فالعيب بصري، والنظارة تعيد الصورة إلى حيث
المستقبلات.
\end{example}

\section{الشبكية}

\begin{definition}[المستقبلات الضوئية]\label{def:g11:the-eye:photoreceptors}
تحوي الشبكية صنفين من \omterm{def:g10:cells-common-unit:cell}{الخلايا} \emph{المستقبلة للضوء}\index{مستقبل ضوئي}،
سميا بشكليهما. \emph{فالعصيات}\index{عصية (شبكية)}، ونحوها 120
مليونًا، تستجيب لضوء بالغ الخفوت لكن كلها بالصباغ نفسه: فتعطي رؤية
بدرجات الرمادي في الليل. أما \emph{المخاريط}\index{مخروط (شبكية)}،
ونحوها 6 ملايين، فتحتاج إلى ضوء أسطع وتأتي على ثلاثة أصناف، لكل منها
صباغ أشد حساسية لجزء مختلف من الطيف: فتعطي رؤية النهار واللون. ويحوي
كل مستقبل ضوئي \emph{صباغًا ضوئيًا} --- وهو \omterm{def:g11:gene-expression:protein}{بروتين}، أي
\emph{أوبسين}\index{أوبسين}، يحمل جزيئًا صغيرًا ماصًّا للضوء مشتقًا من
فيتامين A --- ويبدأ تغير شكله عند امتصاص فوتون استجابةَ \omterm{def:g10:cells-common-unit:cell}{الخلية}
الكهربائية.
\end{definition}

\begin{proposition}[تنظيم الشبكية]\label{prop:g11:the-eye:organisation}
تقع المستقبلات الضوئية في مؤخرة الشبكية، ملاصقة لطبقة داكنة؛ وعلى
الضوء أن يعبر الطبقات الأخرى ليبلغها. وتمر إشاراتها إلى \emph{\omterm{def:g10:cells-common-unit:cell}{خلايا}
ثنائية القطب}، ثم إلى \emph{\omterm{def:g10:cells-common-unit:cell}{خلايا} عقدية}، تتجمع أليافها الطويلة ---
ونحوها مليون --- في العصب البصري. والتوزع غير متساو: ففي \emph{النقرة}،
وهي الحفرة المركزية الصغيرة التي توجه \omterm{def:g11:the-eye:parts}{العين} نحوها، لا توجد إلا
\omterm{def:g11:the-eye:photoreceptors}{المخاريط}، مرصوصة رصًا وكل منها موصول بخليته العقدية الخاصة، فتعطي أحدّ
رؤية؛ وبعيدًا عن المركز تسود العصيات ويتقاسم كثير منها \omterm{def:g10:cells-common-unit:cell}{خلية} عقدية
واحدة، فتعطي حساسية بثمن التفصيل. وحيث يغادر العصب البصري لا مستقبلات
البتة: وهي \emph{البقعة العمياء}.
\end{proposition}
\admitted

\begin{example}[تفسير الغرابتين]\label{ex:g11:the-eye:oddities}
النجم الخافت يختفي حين ينظر إليه مباشرةً لأن في النقرة مخاريط فقط، وهي
تحتاج إلى ضوء أكثر مما يقدمه النجم؛ فإذا نظر إليه من الجانب وقع ضوءه
على العصيات، وهي تكشفه. والسترة رمادية في الليل لأن العصيات التي تراها
لها صباغ واحد: فتستطيع الإخبار بمقدار الضوء، لا بطول موجته. فاللون
مقارنة بين أصناف \omterm{def:g11:the-eye:photoreceptors}{المخاريط} الثلاثة، \omterm{def:g11:the-eye:photoreceptors}{والمخاريط} صامتة في الظلام.
\end{example}

\section{اللون: ثلاثة أصبغة}

\begin{proposition}[رؤية اللون]\label{prop:g11:the-eye:colour}
يحوي كل صنف من \omterm{def:g11:the-eye:photoreceptors}{المخاريط} واحدًا من ثلاثة أوبسينات، تقع ذرى امتصاصها قرب
\qty{420}{nm} (\omterm{def:g11:the-eye:photoreceptors}{المخاريط} S، الزرقاء)، و\qty{530}{nm} (\omterm{def:g11:the-eye:photoreceptors}{المخاريط} M،
الخضراء)، و\qty{560}{nm} (\omterm{def:g11:the-eye:photoreceptors}{المخاريط} L، الحمراء). والضوء المعطى يثير
الأصناف الثلاثة بنسب تتوقف على طول موجته؛ ويقرأ الدماغ اللون من تلك
النسب. والشخص الذي ينقصه صنف من \omterm{def:g11:the-eye:photoreceptors}{المخاريط} يخلط بين الألوان التي كان ذلك
الصنف يميزها: فمن دون مخاريط L أو M عاملة، يصعب التمييز بين الأحمر
والأخضر --- وهو \emph{عمى الألوان الأحمر والأخضر}، ويصيب نحو 8\% من
الرجال و0.5\% من النساء.
\end{proposition}
\admitted

\begin{example}[قراءة المنحنيات]\label{ex:g11:the-eye:curves}
ضوء طوله \qty{500}{nm}: \omterm{def:g11:the-eye:photoreceptors}{المخاريط} S عند 8\%، وM عند 75\%، وL عند 40\%

لون واحد.
\end{example}

\section{مورّثات الأوبسين: عائلة لها تاريخ}

\begin{proposition}[مورّثات الأوبسينات]\label{prop:g11:the-eye:genes}
لكل \omterm{def:g11:the-eye:photoreceptors}{أوبسين} مورّثته الخاصة. \omterm{def:g10:universal-dna:gene}{فمورّثة} \omterm{def:g11:the-eye:photoreceptors}{الأوبسين} S تقع على \omterm{def:g10:universal-dna:information}{صبغي} عادي؛ أما
مورّثتا الأوبسينين M وL فتقعان جنبًا إلى جنب على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} X. والبروتينان
L وM يختلفان في 15 حمضًا أمينيًا فقط من حموضهما الأربعة والستين بعد الثلاثمئة (فهما متطابقان
بنسبة 96\%)، أما \omterm{def:g11:gene-expression:protein}{البروتين} S فيختلف عن أي منهما في أكثر من نصف مواضعه.
ولما كان \omterm{def:g10:universal-dna:gene}{الأليل} غير العامل \omterm{def:g10:universal-dna:gene}{للمورّثة} M أو L على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} X، فهو يعبَّر عنه عند كل
ذكر يحمله ولا يعبَّر عنه عند الإناث إلا إذا حملن اثنين --- وهو نمط
\cref{ch:g11:genes-and-disease}، وسبب كون عمى الألوان الأحمر والأخضر
أشيع ستة عشر ضعفًا عند الرجال.
\end{proposition}
\admitted

\begin{proposition}[عائلة من المورّثات نشأت بالمضاعفات]\label{prop:g11:the-eye:family}
\omterm{def:g10:universal-dna:gene}{مورّثات} الأوبسينات الثلاث، \omterm{def:g10:universal-dna:gene}{ومورّثة} صباغ العصيات، نسخ من \omterm{def:g10:universal-dna:gene}{مورّثة} سلفية
واحدة نسخت \emph{بالمضاعفة} ثم تباعدت \omterm{def:g11:mutations:mutation}{بالطفرة}. ودرجات تشابهها تؤرخ
النسخ: فخطّا S وL/M افترقا باكرًا جدًا في تاريخ \omterm{def:g10:common-ancestry:bodyplan}{الفقاريات}، قبل نحو 500
مليون سنة؛ ونشأت مورّثتا L وM من مضاعفة واحدة قبل نحو 30 إلى 40 مليون
سنة عند سلف رئيسيات العالم القديم، ولدى نسلها --- القردة والقردة
العليا والإنسان --- ثلاثة أصناف من \omterm{def:g11:the-eye:photoreceptors}{المخاريط} بينما لمعظم الثدييات
الأخرى صنفان.
\end{proposition}
\begin{proof}[شواهد]
مقارنة المتتاليات: فمورّثتا L وM متطابقتان بنسبة 96\% ومتجاورتان على
\omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} X --- وهي بصمة مضاعفة ترادفية حديثة؛ وكل منهما مطابقة نحو 43\%
\omterm{def:g10:universal-dna:gene}{لمورّثة} S، والثلاث جميعًا مطابقة نحو 40\% \omterm{def:g10:universal-dna:gene}{لمورّثة} \omterm{def:g11:the-eye:photoreceptors}{أوبسين} العصيات، وهي
النسخ الأبعد. والكلاب والأبقار ومعظم الثدييات لها \omterm{def:g10:universal-dna:gene}{مورّثة} \omterm{def:g11:the-eye:photoreceptors}{أوبسين} واحدة
مرتبطة \omterm{def:g11:cell-cycle-mitosis:chromosome}{بالصبغي} X، في الموضع الذي للرئيسيات فيه مورّثتان؛ ولقردة العالم
الجديد واحدة، لها عدة \omterm{def:g10:universal-dna:gene}{أليلات}، ولا ترى ثلاثة ألوان إلا الإناث الحاملات
\omterm{def:g10:universal-dna:gene}{لأليلين} مختلفين. فعدد \omterm{def:g10:universal-dna:gene}{المورّثات} يوافق عدد \omterm{def:g11:the-eye:photoreceptors}{المخاريط} في كل سلالة، وشجرة
\omterm{def:g10:universal-dna:gene}{المورّثات} توافق شجرة \omterm{def:g10:biodiversity-scales:species}{الأنواع}.
\end{proof}

\begin{example}[ما الذي غيرته مضاعفة واحدة]\label{ex:g11:the-eye:duplication}
القرد ذو صنفي \omterm{def:g11:the-eye:photoreceptors}{المخاريط} يرى الغابة ببعدين من اللون، ويصعب عليه تمييز
الثمرة الناضجة من الأوراق. وبعد مضاعفة L/M، أمكن أن تطفر إحدى النسختين
نحو أطوال موجة أكبر بينما تحفظ الأخرى الأصل --- فمخروط ثالث، وأحمر ناضج
في مقابل الأخضر. أما \omterm{def:g11:mutations:mutation}{الطفرة} التي تكسر لاحقًا إحدى النسختين عند 8\% من
الرجال فلا تفعل إلا أن ترد \omterm{def:g11:the-eye:parts}{العين} إلى حال المخروطين السلفية.
\end{example}

\begin{method}[الاستدلال في حالة رؤية لونية]\label{met:g11:the-eye:case}
\begin{enumerate}
  \item أي صنف من \omterm{def:g11:the-eye:photoreceptors}{المخاريط} مفقود أو متغير؟ اقرأه من الألوان الملتبسة
        (الأحمر والأخضر: L أو M؛ والأزرق والأصفر، وهو نادر: S).
  \item أي \omterm{def:g10:universal-dna:gene}{مورّثة}، وعلى أي \omterm{def:g10:universal-dna:information}{صبغي}؟ L وM على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} X، وS في غيره.
  \item طبّق الوراثة: متنحية مرتبطة \omterm{def:g11:cell-cycle-mitosis:chromosome}{بالصبغي} X للأحمر والأخضر ---
        فأبناء الأمهات الحاملات يصابون مرة من اثنتين، وبنات الآباء
        المصابين حاملات كلهن.
  \item اربط بعائلة \omterm{def:g10:universal-dna:gene}{المورّثات}: نسخة مفقودة، أو نسختا L وM جعلهما تبادل
        بين المورّثتين المتجاورتين متشابهتين أكثر مما ينبغي.
\end{enumerate}
\end{method}

\begin{remark}[من العين إلى الدماغ]\label{rem:g11:the-eye:tobrain}
الشبكية لا ترى؛ إنما تحوّل وتبدأ الفرز. وألياف العصب البصري المليون
تحمل وصفًا مرمَّزًا --- تباينات، وحوافّ، ونسبًا بين استجابات \omterm{def:g11:the-eye:photoreceptors}{المخاريط}
--- إلى الدماغ، حيث يلتقطه الفصل التالي. وما نسميه رؤية يحدث هناك،
مبنيًا على إشارات \omterm{def:g10:cells-common-unit:cell}{الخلايا} الموصوفة هنا ولا شيء غيرها: فمن تعمل شبكيتاه
لكن دماغه البصري متضرر أعمى، ومن يعمل دماغه البصري لكن ماتت مستقبلاته
الضوئية أعمى كذلك.
\end{remark}

\section{مسألة: ثلاثة أصبغة ومضاعفة واحدة}

\begin{problem}[{مسألة نهاية الأسبوع --- تفكيك الرؤية اللونية:
استجابات المخاريط لطيف، وعائلة فيها عمى ألوان، ومورّثات الأوبسين تقارن،
والحدة التي تبلغها نقرة}]
\label{pb:g11:the-eye:1}
استعمل شكل الامتصاص. يعرض اختبار أضواء بأطوال \qty{470}{nm}
و\qty{530}{nm} و\qty{590}{nm} و\qty{650}{nm}.

\medskip
\noindent\textbf{الجزء الأول --- \omterm{def:g11:the-eye:photoreceptors}{المخاريط} تستجيب.}
\begin{enumerate}
  \item لكل ضوء، اقرأ الاستجابة التقريبية للمخاريط S وM وL.
  \item أي ضوء يثير الأصناف الثلاثة؟ وأيها لا يثير إلا واحدًا؟
  \item الشخص الذي تنقصه \omterm{def:g11:the-eye:photoreceptors}{المخاريط} L لا يتلقى إلا قيمتي S وM. فأي
        ضوءين من الأربعة يصعب عليه التمييز بينهما؟ علّل بالأرقام.
  \item والشخص الذي تنقصه \omterm{def:g11:the-eye:photoreceptors}{المخاريط} M: أي زوج يخلط بينه؟
  \item فسّر لماذا لا يكون أي منهما «أعمى عن الأحمر» لكن كليهما ضعيف
        في تمييز الأحمر من الأخضر.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- عائلة.}
نورا ترى الألوان رؤية طبيعية؛ وأبوها مصاب بعمى الألوان الأحمر والأخضر.
وهي تتزوج سامًا، وهو يرى رؤية طبيعية.
\begin{enumerate}[resume]
  \item أعطِ \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي} لنورا في \omterm{def:g10:universal-dna:gene}{مورّثة} \omterm{def:g11:the-eye:photoreceptors}{الأوبسين} المرتبطة \omterm{def:g11:cell-cycle-mitosis:chromosome}{بالصبغي}
        X، وعلّل.
  \item أعطِ الأنماط الوراثية والرؤية اللونية لأبنائهما وبناتهما
        الممكنين، مع الاحتمالات.
  \item ابنتهما إيناس، ورؤيتها اللونية طبيعية، تتزوج رجلًا مصابًا بعمى
        الألوان. احسب احتمال أن تكون إيناس حاملة، ثم احتمال أن تكون
        ابنة لهما مصابة بعمى الألوان.
  \item فسّر لماذا يكون لكل بنت مصابة بعمى الألوان أب مصاب دائمًا.
  \item أم أبي نورا كانت ترى الألوان رؤية طبيعية. فهل كانت حاملة
        بالضرورة؟ فسّر.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{def:g10:universal-dna:gene}{المورّثات}.}
نسب التطابق بين متتاليات بروتينات \omterm{def:g11:the-eye:photoreceptors}{الأوبسين}: L--M 96\%، وL--S 43\%،
وM--S 43\%، وS--رودوبسين 41\%.
\begin{enumerate}[resume]
  \item ارسم، أو صف، شجرة \omterm{def:g10:universal-dna:gene}{المورّثات} الأربع التي تقتضيها هذه الأرقام.
  \item إذا تراكمت الفروق بنسبة ثابتة وكانت مضاعفة L/M عمرها 35 مليون
        سنة، فقدّر عمر افتراق S عن L/M.
  \item تقع مورّثتا L وM جنبًا إلى جنب على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} X. فسّر كيف يستطيع
        تبادل غير متساوٍ بينهما أثناء الانقسام المنصّف أن ينتج \omterm{def:g10:universal-dna:information}{صبغي} X
        لا يحمل إلا واحدة منهما، وأي رؤية لونية تنتج.
  \item ومثل هذه التبادلات أشيع سبب لعمى الألوان الأحمر والأخضر.
        فلماذا تكون مورّثتان متجاورتان تكادان تتطابقان معرضتين لها
        بخاصة؟
  \item توجد مضاعفة L/M عند كل رئيسيات العالم القديم ولا توجد عند أي
        ثديي آخر. فما الذي يقوله ذلك عن زمن وقوعها وعند من وقعت؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- حدة النقرة.}
مخاريط النقرة تتباعد \qty{2.5}{\micro m}؛ والبعد البؤري \omterm{def:g11:the-eye:parts}{للعين}
\qty{17}{mm}. استعمل: الزاوية (بالتقدير الدائري) $=$ الحجم $/$ المسافة.
\begin{enumerate}[resume]
  \item احسب الزاوية التي يحصرها مخروطان متجاوران، بالتقدير الدائري
        وبدقائق القوس ($1' = \num{2.9e-4}$ راديان).
  \item ما أصغر تفصيل في حرف يمكن تمييزه على بعد \qty{6}{m}؟ قارنه
        بخطوط الحروف على لوحة فحص البصر، وهي نحو \qty{1.7}{mm} على تلك
        المسافة للرؤية «الطبيعية».
  \item بعيدًا عن النقرة، تتقاسم 100 عصية \omterm{def:g10:cells-common-unit:cell}{خلية} عقدية واحدة. قدّر
        الزاوية القابلة للتمييز هناك، وفسّر ما الذي يكسب في المقابل.
  \item مخاريط نقرة العقاب أكثف رصًا مرتين وعينه أكبر قليلًا. فبأي
        عامل تكون قدرته على التمييز أدق؟
  \item اذكر النتيجة: أصناف \omterm{def:g11:the-eye:photoreceptors}{المخاريط} الثلاثة والضوء الذي يعمى عنه كل
        منها، والحدث الوراثي الواحد الذي أعطى الرئيسيات الثالث،
        والزاوية التي تميزها النقرة.
\end{enumerate}
\end{problem}
