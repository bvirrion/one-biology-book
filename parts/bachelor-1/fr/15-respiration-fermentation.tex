\chapter{Respiration cellulaire et fermentation}\label{ch:b1:respiration-fermentation}

Une fiole de levure dans une solution sucrée, à l'abri de l'air,
bouillonne de dioxyde de carbone et transforme le sucre en alcool~;
laissez entrer l'air et le bouillonnement ralentit, l'alcool cesse, et
la levure croît dix fois plus vite sur le même sucre. Pasteur observa
cela en 1861 sans pouvoir l'expliquer~; l'explication a demandé un
siècle et traverse tout ce chapitre. L'oxygène permet à une \omterm{def:b1:cell-unit-of-life:cell}{cellule} de
tirer du glucose quinze fois l'énergie qu'elle en tire sans lui, et
elle y parvient en faisant descendre les électrons du sucre le long
d'une chaîne de transporteurs jusqu'à l'oxygène, en pompant des protons
à travers une membrane à mesure qu'ils tombent, et en laissant les
protons revenir par une turbine qui fabrique de l'\omterm{def:b1:water-small-molecules:atp}{ATP}. Ce chapitre
décrit la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse}, le \omterm{def:b1:respiration-fermentation:krebs}{cycle de Krebs}, la \omterm{def:b1:respiration-fermentation:chain}{chaîne respiratoire} et son
couplage chimiosmotique, les \omterm{def:b1:respiration-fermentation:fermentation}{fermentations} qui se passent d'oxygène, et
la combustion des \omterm{def:b1:lipids:triglyceride}{graisses} et des \omterm{def:b1:proteins:peptide}{protéines} par la même machinerie.

\section{L'oxydation du glucose}

\begin{proposition}[La réaction globale et ses étapes]\label{prop:b1:respiration-fermentation:overview}
\index{respiration cellulaire}
L'oxydation complète du glucose,
\[
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \to 6\,\mathrm{CO_2} + 6\,\mathrm{H_2O},
\qquad \Delta G^{\circ\prime} = \qty{-2870}{kJ/mol},
\]
se déroule en trois étapes qui ne laissent jamais les électrons
rencontrer directement l'oxygène. La \emph{\omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse}}, dans le \omterm{def:b1:eukaryotic-cell:organelle}{cytosol},
scinde le glucose en deux pyruvates et rapporte 2 \omterm{def:b1:water-small-molecules:atp}{ATP} et 2 NADH.
L'\emph{oxydation du pyruvate et le \omterm{def:b1:respiration-fermentation:krebs}{cycle de Krebs}}, dans la \omterm{def:b1:eukaryotic-cell:mitochondrion}{matrice
mitochondriale}, oxydent le pyruvate en $\mathrm{CO_2}$, chargeant les
électrons sur 8 NADH et 2 $\mathrm{FADH_2}$ et fabriquant 2 GTP. La
\emph{phosphorylation oxydative}, dans la membrane interne de la
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrie}, transmet ces électrons à l'oxygène par une chaîne de
transporteurs qui \omterm{def:b1:membranes-transport:transporters}{pompe} des protons, et le gradient de protons
entretient la synthèse d'environ 26 \omterm{def:b1:water-small-molecules:atp}{ATP}. En tout, quelque 30 \omterm{def:b1:water-small-molecules:atp}{ATP} par
glucose~: environ la moitié de l'\omterm{prop:b1:water-small-molecules:gibbs}{énergie libre} de combustion captée, le
reste libéré en chaleur.
\end{proposition}

\begin{definition}[Transporteurs d'électrons]\label{def:b1:respiration-fermentation:carriers}
\index{NAD}\index{FAD}
Le \emph{$\mathrm{NAD^+}$} (nicotinamide adénine dinucléotide) accepte
deux électrons et un proton d'un substrat pour devenir NADH~; le
\emph{FAD} accepte deux électrons et deux protons pour devenir
$\mathrm{FADH_2}$. Tous deux sont des \omterm{def:b1:enzymes:enzyme}{coenzymes} de déshydrogénases, et
tous deux sont recyclés~: réduits par le catabolisme, réoxydés par la
\omterm{def:b1:respiration-fermentation:chain}{chaîne respiratoire}. Une \omterm{def:b1:cell-unit-of-life:cell}{cellule} n'en contient que des micromoles, qui
se renouvellent des milliers de fois par heure. Leurs potentiels rédox,
$E'_0 = \qty{-0.32}{V}$ pour $\mathrm{NAD^+}/\mathrm{NADH}$, les placent
très au-dessous de l'oxygène ($\qty{+0.82}{V}$)~: chaque paire
d'électrons que le NADH cède à l'oxygène libère
$\Delta G^{\circ\prime} = -2F\Delta E =
\qty{-220}{kJ/mol}$, de quoi payer plus de quatre \omterm{def:b1:water-small-molecules:atp}{ATP}.
\end{definition}

\section{La glycolyse}

\begin{definition}[Glycolyse]\label{def:b1:respiration-fermentation:glycolysis}
\index{glycolyse}
La \emph{glycolyse} est une suite de dix réactions enzymatiques du
\omterm{def:b1:eukaryotic-cell:organelle}{cytosol} qui convertit un glucose (C6) en deux \emph{pyruvates} (C3).
Dans la \emph{phase d'investissement}, deux \omterm{def:b1:water-small-molecules:atp}{ATP} sont dépensés pour
phosphoryler le sucre (l'hexokinase, puis la
\emph{phosphofructokinase}, PFK) et le diphosphate à six carbones est
scindé en deux phosphates à trois carbones. Dans la \emph{phase de
remboursement}, chaque triose est oxydé par le $\mathrm{NAD^+}$ (la
seule oxydation de la glycolyse) et ses phosphates sont transférés à
l'ADP par \emph{phosphorylation au niveau du substrat}\index{phosphorylation au niveau du substrat} --- un phosphate cédé directement par un
intermédiaire riche en énergie à l'ADP, sans membrane ni oxygène. Le
bilan~:
\[
\text{glucose} + 2\,\mathrm{NAD^+} + 2\,\mathrm{ADP} + 2\,\mathrm{P_i}
\to 2\,\text{pyruvate} + 2\,\mathrm{NADH} + 2\,\mathrm{H^+} + 2\,\mathrm{ATP} + 2\,\mathrm{H_2O} .
\]
Elle n'a pas besoin d'oxygène~; c'est la voie la plus ancienne et la
plus universelle du métabolisme.
\end{definition}

\begin{proposition}[La commande à la phosphofructokinase]\label{prop:b1:respiration-fermentation:pfk}
La PFK, première étape irréversible engagée dans la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse}, est une
\omterm{def:b1:enzymes:allosteric}{enzyme allostérique} (\cref{ch:b1:enzymes}) inhibée par l'\omterm{def:b1:water-small-molecules:atp}{ATP} et par le
citrate --- les signaux que l'énergie et le carburant du \omterm{def:b1:respiration-fermentation:krebs}{cycle de Krebs}
abondent --- et activée par l'AMP et l'ADP, les signaux qu'ils manquent.
Quand l'\omterm{def:b1:water-small-molecules:atp}{ATP} est élevé, le flux de la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} tombe en quelques
secondes~; quand la \omterm{def:b1:cell-unit-of-life:cell}{cellule} dépense de l'\omterm{def:b1:water-small-molecules:atp}{ATP}, l'AMP monte brusquement
(par l'adénylate kinase,
$2\,\mathrm{ADP} \rightleftharpoons \mathrm{ATP} + \mathrm{AMP}$), et
la PFK s'ouvre. La \emph{charge énergétique} de la \omterm{def:b1:cell-unit-of-life:cell}{cellule} règle la voie
qui la remplit.
\end{proposition}

\section{Oxydation du pyruvate et cycle de Krebs}

\begin{definition}[Pyruvate déshydrogénase, acétyl-CoA]\label{def:b1:respiration-fermentation:pdh}
Le pyruvate entre dans la \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrie} et est oxydé par le complexe
\emph{pyruvate déshydrogénase} en \emph{acétyl-CoA}\index{acetyl-CoA@acétyl-CoA}
--- un groupe acétyle à deux carbones porté par le \omterm{def:b1:enzymes:enzyme}{coenzyme} A
(\cref{ch:b1:nucleic-acids}) sur une liaison thioester riche en énergie
--- avec libération d'un $\mathrm{CO_2}$ et d'un NADH. L'acétyl-CoA est
l'entrée commune de tous les carburants dans le cycle~: les \omterm{def:b1:carbohydrates:monosaccharide}{glucides} par
le pyruvate, les \omterm{def:b1:lipids:triglyceride}{graisses} par la $\beta$-oxydation, beaucoup d'\omterm{def:b1:proteins:aminoacid}{acides
aminés} directement.
\end{definition}

\begin{definition}[Le cycle de Krebs]\label{def:b1:respiration-fermentation:krebs}
\index{cycle de Krebs}\index{cycle de l'acide citrique}
Le \emph{cycle de Krebs} (cycle de l'acide citrique), dans la \omterm{def:b1:eukaryotic-cell:mitochondrion}{matrice
mitochondriale}, condense l'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} (C2) avec l'oxaloacétate (C4) en
citrate (C6) et, en huit étapes, oxyde les deux carbones en
$\mathrm{CO_2}$ tout en régénérant l'oxaloacétate. Par groupe acétyle~:
3 NADH, 1 $\mathrm{FADH_2}$, 1 GTP (par \omterm{def:b1:respiration-fermentation:glycolysis}{phosphorylation au niveau du
substrat}) et 2 $\mathrm{CO_2}$. Par glucose (deux acétyles)~: 6 NADH,
2 $\mathrm{FADH_2}$, 2 GTP, 4 $\mathrm{CO_2}$ --- ce qui, avec les
2 $\mathrm{CO_2}$ de la pyruvate déshydrogénase, fait les six de
l'équation globale. Aucun oxygène n'est consommé dans le cycle
lui-même~; il ne tourne qu'aussi longtemps que la \omterm{def:b1:respiration-fermentation:chain}{chaîne respiratoire}
réoxyde son NADH. Il est en outre \emph{amphibolique}~: ses
intermédiaires sont prélevés pour les biosynthèses
(\cref{ch:b1:biosyntheses-integration}) et réapprovisionnés à partir
d'\omterm{def:b1:proteins:aminoacid}{acides aminés} et de pyruvate.
\end{definition}

\begin{example}[Suivre les carbones]\label{ex:b1:respiration-fermentation:carbons}
Donnez à une \omterm{def:b1:cell-unit-of-life:cell}{cellule} du glucose marqué au $\mathrm{^{14}C}$ sur le
carbone 1~: le marquage apparaît sur le carbone méthyle du pyruvate,
puis de l'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA}, entre dans le citrate, et ne sort en
$\mathrm{CO_2}$ qu'au deuxième ou au troisième tour du cycle --- les
deux carbones libérés à un tour donné appartiennent à l'oxaloacétate du
tour précédent. Krebs (1937) a déduit le cycle de l'observation que des
quantités catalytiques de n'importe lequel de ses acides stimulaient
l'oxydation de bien plus de pyruvate qu'elles ne pouvaient en rendre
compte~: elles étaient régénérées.
\end{example}

\section{La phosphorylation oxydative}

\begin{definition}[La chaîne respiratoire]\label{def:b1:respiration-fermentation:chain}
\index{chaine respiratoire@chaîne respiratoire}\index{chaine de transport d'electrons@chaîne de transport d'électrons}
La \emph{chaîne respiratoire} est faite de quatre complexes protéiques
de la membrane interne de la \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrie} et de deux transporteurs
mobiles. Le NADH donne ses électrons au \emph{complexe I}, le
$\mathrm{FADH_2}$ (par la succinate déshydrogénase, \emph{complexe II})
donne les siens~; tous deux réduisent l'\emph{ubiquinone} (Q), quinone
liposoluble qui diffuse dans la membrane~; Q les passe au
\emph{complexe III}, qui les remet au \emph{cytochrome $c$}, petite
\omterm{def:b1:proteins:peptide}{protéine} de la face externe~; et le \emph{complexe IV} (la cytochrome
oxydase) les livre, quatre à la fois, à l'$\mathrm{O_2}$, ce qui fait de
l'eau. À chacun des complexes I, III et IV, les électrons chutent en
potentiel et l'énergie \omterm{def:b1:membranes-transport:transporters}{pompe} des protons de la matrice vers l'espace
intermembranaire~: environ 4, 4 et 2 par paire d'électrons, dix par
NADH, six par $\mathrm{FADH_2}$ (qui entre en aval du complexe I).
\end{definition}

\begin{theorem}[Le couplage chimiosmotique]\label{thm:b1:respiration-fermentation:chemiosmosis}
\index{theorie chimiosmotique@théorie chimiosmotique}\index{force protomotrice}
Le gradient de protons bâti par la chaîne --- environ \qty{0.16}{V} de
\omterm{def:b1:membranes-transport:potential}{potentiel de membrane} (matrice négative) plus une unité de pH, soit une
\emph{\omterm{thm:b1:respiration-fermentation:chemiosmosis}{force protomotrice}} d'environ \qty{0.22}{V}, ou \qty{21}{kJ} par
mole de protons --- est le seul lien entre le transport d'électrons et
la synthèse d'\omterm{def:b1:water-small-molecules:atp}{ATP}. L'\omterm{thm:b1:photosynthesis:chemiosmosis}{ATP synthase}, moteur rotatif de la membrane
interne, laisse passer environ \qty{3}{H^+} par \omterm{def:b1:water-small-molecules:atp}{ATP} fabriqué, et un de
plus est dépensé à importer le phosphate et à exporter l'\omterm{def:b1:water-small-molecules:atp}{ATP}~: 4 par
\omterm{def:b1:water-small-molecules:atp}{ATP}. D'où le \emph{rapport P/O}\index{rapport P/O}~: $10/4 = 2.5$ \omterm{def:b1:water-small-molecules:atp}{ATP}
par NADH et $6/4 = 1.5$ par $\mathrm{FADH_2}$.
\end{theorem}
\begin{proof}[Faits expérimentaux]
Les \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondries} ne fabriquent de l'\omterm{def:b1:water-small-molecules:atp}{ATP} que si leur membrane interne
est intacte et close~; des fragments qui transportent les électrons mais
ne peuvent pas retenir de gradient n'en font pas. Les découplants comme
le dinitrophénol, qui font passer des protons à travers la membrane,
abolissent la synthèse d'\omterm{def:b1:water-small-molecules:atp}{ATP} alors que la consommation d'oxygène se
poursuit et même s'accélère, l'énergie partant en chaleur --- la \omterm{def:b1:lipids:triglyceride}{graisse}
brune le fait délibérément avec sa propre \omterm{def:b1:proteins:peptide}{protéine} découplante
(\cref{ch:b1:mammal-organization}), et le dinitrophénol a été vendu comme
amaigrissant, qui tuait par hyperthermie. Les \omterm{def:b1:enzymes:inhibitors}{inhibiteurs} de la chaîne
(le cyanure sur le complexe IV) arrêtent les deux~; les \omterm{def:b1:enzymes:inhibitors}{inhibiteurs} de
la synthase (l'oligomycine) arrêtent la synthèse d'\omterm{def:b1:water-small-molecules:atp}{ATP} et, le gradient
ne pouvant plus se décharger, arrêtent aussi la respiration, qu'un
découplant rétablit alors. Un gradient de pH artificiel imposé à des
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondries} dans l'obscurité entretient la synthèse d'\omterm{def:b1:water-small-molecules:atp}{ATP}, comme dans
les \omterm{def:b1:eukaryotic-cell:plastid}{chloroplastes} (\cref{ch:b1:photosynthesis}).
\end{proof}

\begin{method}[Le bilan en ATP d'un glucose]\label{met:b1:respiration-fermentation:balance}
\begin{enumerate}
  \item \omterm{def:b1:respiration-fermentation:glycolysis}{Glycolyse}~: $+2$ \omterm{def:b1:water-small-molecules:atp}{ATP} (net), $+2$ NADH dans le \omterm{def:b1:eukaryotic-cell:organelle}{cytosol}.
  \item Pyruvate déshydrogénase~: $+2$ NADH. \omterm{def:b1:respiration-fermentation:krebs}{Cycle de Krebs}~: $+6$ NADH,
        $+2$ $\mathrm{FADH_2}$, $+2$ GTP ($=$ \omterm{def:b1:water-small-molecules:atp}{ATP}).
  \item Phosphorylation oxydative~: $2.5$ \omterm{def:b1:water-small-molecules:atp}{ATP} par NADH mitochondrial
        (8~: 20 \omterm{def:b1:water-small-molecules:atp}{ATP}), $1.5$ par $\mathrm{FADH_2}$ (2~: 3 \omterm{def:b1:water-small-molecules:atp}{ATP}).
  \item Le NADH cytosolique ne peut pas traverser la membrane interne~;
        ses électrons entrent par une navette qui les livre au
        $\mathrm{FAD}$ (1.5 chacun, dans le muscle et l'encéphale) ou au
        $\mathrm{NAD^+}$ (2.5 chacun, dans le foie et le cœur)~: 3 ou 5
        \omterm{def:b1:water-small-molecules:atp}{ATP}.
  \item Total~: $2 + 2 + 20 + 3 + 3 = 30$, ou $32$ avec la meilleure
        navette. Rendement~: $30\times 50/2870 \approx \qty{52}{\%}$
        dans les conditions cellulaires.
\end{enumerate}
\end{method}

\section{La fermentation}

\begin{definition}[Fermentation]\label{def:b1:respiration-fermentation:fermentation}
\index{fermentation}
Sans oxygène, la \omterm{def:b1:respiration-fermentation:chain}{chaîne respiratoire} s'arrête, le NADH ne peut plus être
réoxydé, et la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} s'arrêterait en quelques secondes faute de
$\mathrm{NAD^+}$. La \emph{fermentation} le régénère en employant le
pyruvate lui-même comme accepteur d'électrons~: dans la
\emph{fermentation lactique} (muscle, globules rouges, bactéries
lactiques), le pyruvate est réduit en lactate~; dans la
\emph{fermentation alcoolique} (levures, quelques végétaux), il est
décarboxylé en acétaldéhyde, réduit ensuite en éthanol, avec libération
de $\mathrm{CO_2}$. Le rendement est celui de la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} seule~: 2 \omterm{def:b1:water-small-molecules:atp}{ATP}
par glucose, un quinzième de celui de la respiration~; le carbone est à
peine oxydé et les produits contiennent encore presque toute l'énergie
du carburant. La \emph{respiration anaérobie} --- une \omterm{def:b1:respiration-fermentation:chain}{chaîne de
transport d'électrons} qui finit sur du nitrate, du sulfate ou du
carbonate au lieu de l'oxygène, chez beaucoup de bactéries --- est autre
chose, traitée dans le volume de deuxième année.
\end{definition}

\begin{proposition}[L'effet Pasteur]\label{prop:b1:respiration-fermentation:pasteur}
\index{effet Pasteur}
Une \omterm{def:b1:cell-unit-of-life:cell}{cellule} capable de respirer consomme le glucose bien plus vite
lorsqu'elle est privée d'oxygène que lorsqu'elle en dispose, et elle y
croît bien moins.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Pasteur (1861) trouva que la levure d'une cuve aérée consommait le sucre
lentement et se multipliait, tandis que dans une cuve close elle le
consommait dix fois plus vite, se multipliait peu et faisait de
l'alcool. L'explication est le rendement en \omterm{def:b1:water-small-molecules:atp}{ATP}~: pour obtenir autant
d'\omterm{def:b1:water-small-molecules:atp}{ATP} à 2 par glucose qu'à 30, la \omterm{def:b1:cell-unit-of-life:cell}{cellule} doit faire tourner la
\omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} quinze fois plus vite, et la PFK, libérée de l'inhibition par
l'\omterm{def:b1:water-small-molecules:atp}{ATP} et le citrate que la respiration entretient, le lui permet. Un
muscle qui sprinte montre le même effet en quelques secondes~: son
\omterm{def:b1:carbohydrates:polysaccharide}{glycogène} baisse vingt fois plus vite qu'au repos, et le lactate monte.
\end{proof}

\begin{definition}[Quotient respiratoire]\label{def:b1:respiration-fermentation:rq}
\index{quotient respiratoire}
Le \emph{quotient respiratoire} QR est le rapport du $\mathrm{CO_2}$
produit à l'$\mathrm{O_2}$ consommé, en moles. Les \omterm{def:b1:carbohydrates:monosaccharide}{glucides} donnent 1.0
(six de chaque par glucose)~; les \omterm{def:b1:lipids:triglyceride}{graisses} environ 0.7 (palmitate~:
$\mathrm{C_{16}H_{32}O_2} + 23\,\mathrm{O_2} \to 16\,\mathrm{CO_2} +
16\,\mathrm{H_2O}$, $16/23 = 0.70$), parce que leurs carbones sont plus
réduits et demandent plus d'oxygène par $\mathrm{CO_2}$~; les \omterm{def:b1:proteins:peptide}{protéines}
environ 0.8. Une culture qui fermente libère du $\mathrm{CO_2}$ sans
prendre d'oxygène~: son QR est infini, et le QR d'une culture mixte,
supérieur à 1, mesure la part de la \omterm{def:b1:respiration-fermentation:fermentation}{fermentation}. Mesuré sur un animal
entier par analyse des gaz, le QR dit quel carburant il brûle.
\end{definition}

\begin{example}[Lire un QR]\label{ex:b1:respiration-fermentation:readrq}
Un humain au repos après un repas glucidique~: QR 0.95, il brûle du
glucose. La même personne après une nuit de jeûne~: 0.75, de la \omterm{def:b1:lipids:triglyceride}{graisse}.
Un marathonien au trentième kilomètre, \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} épuisé~: 0.72. Une
fiole de levure sur glucose avec un peu d'air~: QR 4 --- les trois
quarts de son glucose sont fermentés.
\end{example}

\section{Autres carburants et régulation de l'ensemble}

\begin{proposition}[Graisses et protéines entrent dans la même machinerie]\label{prop:b1:respiration-fermentation:otherfuels}
\index{beta-oxydation@$\beta$-oxydation}
Les \omterm{def:b1:lipids:fattyacid}{acides gras} sont activés en acyl-CoA et, dans la \omterm{def:b1:eukaryotic-cell:mitochondrion}{matrice
mitochondriale}, raccourcis de deux carbones à la fois par la
\emph{$\beta$-oxydation}, chaque tour donnant un \omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA}, un NADH et
un $\mathrm{FADH_2}$~: le palmitate (C16) donne 8 \omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA}, 7 NADH et
7 $\mathrm{FADH_2}$ --- environ 106 \omterm{def:b1:water-small-molecules:atp}{ATP}, soit \qty{6.6}{} \omterm{def:b1:water-small-molecules:atp}{ATP} par
carbone contre 5 pour le glucose. Les \omterm{def:b1:proteins:aminoacid}{acides aminés} perdent leur azote
sous forme d'ammoniac (converti en urée dans le foie) et leurs
squelettes carbonés entrent sous forme de pyruvate, d'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} ou
d'intermédiaires du \omterm{def:b1:respiration-fermentation:krebs}{cycle de Krebs}. Tout carburant converge vers
l'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} et la chaîne~; la \omterm{def:b1:cell-unit-of-life:cell}{cellule} choisit entre eux par les
hormones et par l'état de son \omterm{def:b1:water-small-molecules:atp}{ATP}, comme le décrit le
\cref{ch:b1:biosyntheses-integration}.
\end{proposition}

\begin{example}[Un gramme de chaque]\label{ex:b1:respiration-fermentation:gramofeach}
Un gramme de glucose (\qty{5.6}{mmol}) rapporte environ \qty{170}{mmol}
d'\omterm{def:b1:water-small-molecules:atp}{ATP}~; un gramme de palmitate (\qty{3.9}{mmol}) environ
\qty{410}{mmol} --- les \qty{38}{kJ/g} contre \qty{17}{kJ/g} du
\cref{ch:b1:lipids}, convertis en monnaie. La \omterm{def:b1:lipids:triglyceride}{graisse} coûte plus
d'oxygène par \omterm{def:b1:water-small-molecules:atp}{ATP}, ce qui explique qu'un muscle de sprinteur, limité par
l'oxygène, brûle du \omterm{def:b1:carbohydrates:polysaccharide}{glycogène}, et qu'un oiseau migrateur, riche en
oxygène et limité par la masse, brûle de la \omterm{def:b1:lipids:triglyceride}{graisse}.
\end{example}

\section{Problème~: la levure avec et sans air}

\begin{problem}[{Problème du week-end --- les cuves de Pasteur
revisitées~: l'ATP compté, le glucose consommé, la biomasse produite et
les gaz mesurés, jusqu'au rapport des rendements en ATP et au rapport
P/O}]
\label{pb:b1:respiration-fermentation:1}
On cultive de la levure sur glucose dans deux fioles identiques, l'une
aérée, l'autre close. Employer des \omterm{thm:b1:respiration-fermentation:chemiosmosis}{rapports P/O} de 2.5 (NADH) et de 1.5
($\mathrm{FADH_2}$), la navette à $\mathrm{FADH_2}$ pour le NADH
cytosolique, un rendement de croissance de \qty{10.5}{g} de biomasse
sèche par mole d'\omterm{def:b1:water-small-molecules:atp}{ATP}, et un glucose à \qty{180}{g/mol}.

\medskip
\noindent\textbf{Partie I --- L'\omterm{def:b1:water-small-molecules:atp}{ATP} compté.}
\begin{enumerate}
  \item Énumérer les \omterm{def:b1:water-small-molecules:atp}{ATP}, GTP, NADH et $\mathrm{FADH_2}$ produits à
        partir d'un glucose par la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse}, la pyruvate
        déshydrogénase et le \omterm{def:b1:respiration-fermentation:krebs}{cycle de Krebs}.
  \item Calculer les protons pompés par la chaîne et par glucose (10 par
        NADH mitochondrial, 6 par $\mathrm{FADH_2}$~; le NADH cytosolique
        entre sous forme de $\mathrm{FADH_2}$).
  \item Calculer l'\omterm{def:b1:water-small-molecules:atp}{ATP} fabriqué par la synthase à \qty{4}{H^+} par \omterm{def:b1:water-small-molecules:atp}{ATP},
        et le total d'\omterm{def:b1:water-small-molecules:atp}{ATP} par glucose avec de l'air.
  \item Calculer l'\omterm{def:b1:water-small-molecules:atp}{ATP} par glucose sans air.
  \item Calculer le rapport des deux rendements.
  \item Calculer l'\omterm{prop:b1:water-small-molecules:gibbs}{énergie libre} captée avec de l'air
        ($\Delta G_{\text{ATP}}
        = \qty{-50}{kJ/mol}$) en fraction des \qty{2870}{kJ}, puis sans
        air en fraction des \qty{2870}{kJ} et de l'énergie réellement
        libérée par la \omterm{def:b1:respiration-fermentation:fermentation}{fermentation} (glucose en deux éthanols et deux
        $\mathrm{CO_2}$~: \qty{-235}{kJ/mol}).
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Glucose consommé, biomasse produite.}
La fiole close doit fabriquer de l'\omterm{def:b1:water-small-molecules:atp}{ATP} au même rythme que la fiole aérée
pour entretenir ses \omterm{def:b1:cell-unit-of-life:cell}{cellules}~: \qty{30}{mmol} d'\omterm{def:b1:water-small-molecules:atp}{ATP} par heure.
\begin{enumerate}[resume]
  \item Calculer le glucose consommé par heure dans chaque fiole.
  \item Calculer l'éthanol et le $\mathrm{CO_2}$ produits par heure dans
        la fiole close, ainsi que le $\mathrm{CO_2}$ et l'$\mathrm{O_2}$
        échangés dans la fiole aérée.
  \item Calculer la biomasse que la fiole close peut bâtir par mole de
        glucose, et par gramme de glucose.
  \item La fiole aérée bâtit \qty{0.5}{g} de biomasse par gramme de
        glucose. Calculer l'\omterm{def:b1:water-small-molecules:atp}{ATP} que cela représenterait à
        \qty{10.5}{g/mol}, et comparer aux 30 \omterm{def:b1:water-small-molecules:atp}{ATP} disponibles. Qu'est-ce
        qui limite le rendement aérobie~?
  \item Expliquer l'\omterm{prop:b1:respiration-fermentation:pasteur}{effet Pasteur} à partir des questions 7 et 9, et
        nommer l'\omterm{def:b1:enzymes:enzyme}{enzyme} dont la régulation le réalise.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Les gaz mesurés.}
Une troisième fiole, mal aérée, libère \qty{44}{mg} de $\mathrm{CO_2}$
et consomme \qty{20}{mg} d'$\mathrm{O_2}$ par heure.
\begin{enumerate}[resume]
  \item Convertir les deux en millimoles et calculer le QR.
  \item Soit $x$ le glucose respiré et $y$ le glucose fermenté par
        heure. Écrire les bilans en $\mathrm{O_2}$ et en
        $\mathrm{CO_2}$ et résoudre en $x$ et $y$.
  \item Quelle fraction du glucose est fermentée~? Quelle fraction de
        l'\omterm{def:b1:water-small-molecules:atp}{ATP} vient de la \omterm{def:b1:respiration-fermentation:fermentation}{fermentation}~?
  \item La même fiole sur palmitate au lieu de glucose (en aérobie)~:
        calculer le QR.
  \item Un étudiant mesure un QR de 0.7 chez un animal au repos après
        une nuit de jeûne et de 1.0 après un repas. Interpréter.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La turbine.}
\begin{enumerate}[resume]
  \item Calculer l'\omterm{prop:b1:water-small-molecules:gibbs}{énergie libre} d'une mole de protons traversant une
        \omterm{thm:b1:respiration-fermentation:chemiosmosis}{force protomotrice} de \qty{0.22}{V}.
  \item Calculer l'énergie de \qty{4}{H^+} et le rendement de la
