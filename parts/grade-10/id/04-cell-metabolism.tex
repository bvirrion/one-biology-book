\chapter{Metabolisme Sel}\label{ch:g10:cell-metabolism}

Rentangkan sebuah balon pada leher labu berisi air gula hangat dengan
sesendok ragi roti di dalamnya. Dalam satu jam cairannya keruh dan
mendesis, balonnya berdiri, dan labunya berbau samar seperti roti dan
bir. \omterm{def:g10:cells-common-unit:cell}{Sel} ragi itu sedang memakan gulanya dan mengubahnya menjadi
sesuatu yang lain, dan gas yang dilepaskannya adalah karbon dioksida
yang sama dengan yang kamu embuskan. Daun di bawah matahari berbuat
sebaliknya: ia menerima karbon dioksida dan melepaskan oksigen.
Keduanya adalah kimia, yang dijalankan di dalam \omterm{def:g10:cells-common-unit:cell}{sel}, pada suhu tubuh,
tanpa nyala dan tanpa asam --- dan bab ini membahas kimia itu.

\section{Metabolisme: kimia sebuah sel}

\begin{definition}[Metabolisme]\label{def:g10:cell-metabolism:metabolism}
Yang disebut \emph{metabolisme}\index{metabolisme} sebuah \omterm{def:g10:cells-common-unit:cell}{sel} adalah
himpunan seluruh reaksi kimia yang berlangsung di dalamnya: reaksi yang
membongkar molekul untuk melepaskan energi, dan reaksi yang membangun
molekul \omterm{def:g10:cells-common-unit:cell}{sel} itu sendiri dari molekul yang lebih sederhana. Tiap reaksi
metabolisme dijalankan oleh sebuah \emph{enzim}\index{enzim}, yaitu
\omterm{def:g10:chemistry-of-life:families}{protein} yang mempercepat satu reaksi tertentu tanpa terpakai olehnya.
\end{definition}

\begin{proposition}[Enzim membuat kimia sel mungkin]\label{prop:g10:cell-metabolism:enzymes}
Glukosa yang dibiarkan di udara pada \qty{37}{\celsius} tidak terbakar;
di dalam \omterm{def:g10:cells-common-unit:cell}{sel} ia teroksidasi seluruhnya dalam hitungan menit. Bedanya
terletak pada \omterm{def:g10:cell-metabolism:metabolism}{enzimnya}: masing-masing mengikat satu molekul tertentu
(\emph{substrat}-nya), mengubahnya menjadi hasil, melepaskannya dan
mulai lagi, ribuan kali tiap sekon. Karena itu \omterm{def:g10:cell-metabolism:metabolism}{metabolisme} sebuah \omterm{def:g10:cells-common-unit:cell}{sel}
ditetapkan oleh himpunan \omterm{def:g10:cell-metabolism:metabolism}{enzim} yang dimilikinya --- dan himpunan itu
ditetapkan oleh \omterm{def:g10:universal-dna:gene}{gennya}.
\end{proposition}
\admitted

\begin{example}[Satu enzim yang sedang bekerja]\label{ex:g10:cell-metabolism:catalase}
Tuangkan hidrogen peroksida pada sepotong hati mentah atau kentang:
cairannya berbuih hebat sementara oksigen dilepaskan. \omterm{def:g10:cell-metabolism:metabolism}{Enzim} katalase,
yang hadir hampir pada setiap \omterm{def:g10:cells-common-unit:cell}{sel}, memecah peroksida yang beracun itu
menjadi air dan oksigen dengan laju jutaan molekul tiap sekon tiap
molekul \omterm{def:g10:cell-metabolism:metabolism}{enzim}. Rebuslah dahulu hatinya dan tidak terjadi apa-apa:
\omterm{def:g10:cell-metabolism:metabolism}{enzimnya}, sebuah \omterm{def:g10:chemistry-of-life:families}{protein}, telah dirusak panas. Peroksida yang tidak
berbuih di atas batu menunjukkan bahwa reaksinya tidak berjalan sendiri.
\end{example}

\section{Dua cara memberi makan sebuah sel}

\begin{definition}[Autotrofi dan heterotrofi]\label{def:g10:cell-metabolism:autotrophy}
Sebuah \omterm{def:g10:cells-common-unit:cell}{sel} bersifat \emph{autotrof}\index{autotrofi} jika ia membangun
molekul organiknya sendiri dari \omterm{def:g10:chemistry-of-life:mineral-organic}{materi mineral} saja --- karbon
dioksida, air, \omterm{def:g10:chemistry-of-life:mineral-organic}{garam mineral} --- dengan memakai sumber energi dari
luar. \omterm{def:g10:cells-common-unit:cell}{Sel} tumbuhan berkloroplas, alga dan sebagian bakteri tergolong
autotrof, dan sumber energinya adalah cahaya: inilah
\emph{fotosintesis}\index{fotosintesis}. Sebuah \omterm{def:g10:cells-common-unit:cell}{sel} bersifat
\emph{heterotrof}\index{heterotrofi} jika ia harus menerima molekul
organik yang dibuat organisme lain, baik sebagai bahan bangunan maupun
sebagai sumber energi: \omterm{def:g10:cells-common-unit:cell}{sel} hewan, jamur, sebagian besar bakteri, dan
\omterm{def:g10:cells-common-unit:cell}{sel} tumbuhan yang tidak hijau (akar, misalnya) tergolong heterotrof.
\end{definition}

\begin{proposition}[Fotosintesis]\label{prop:g10:cell-metabolism:photosynthesis}
Dalam cahaya, \omterm{def:g10:cells-common-unit:cell}{sel} yang berkloroplas menyerap karbon dioksida dan air,
melepaskan oksigen, dan membangun glukosa, yang lalu diubahnya menjadi
pati, selulosa, \omterm{def:g10:chemistry-of-life:families}{lipid} dan, dengan nitrogen mineral, \omterm{def:g10:chemistry-of-life:families}{asam amino}. Secara
ringkas:
\[
6\,\mathrm{CO_2} + 6\,\mathrm{H_2O} \xrightarrow{\ \text{cahaya, klorofil}\ }
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2}.
\]
Energi cahaya tersimpan dalam ikatan kimia glukosanya.
\end{proposition}
\begin{proof}[Bukti percobaan]
Suspensi alga hijau \emph{Chlorella} dalam bejana tertutup yang
dilengkapi pengukur oksigen memperlihatkan oksigen terlarutnya naik
mantap ketika lampunya menyala dan turun ketika lampunya padam;
kenaikannya berhenti jika karbon dioksida airnya disingkirkan. Daun
yang disimpan dalam gelap selama dua hari tidak lagi terwarnai biru
kehitaman oleh iodin, dan daun yang separuhnya tertutup kertas timah
hanya terwarnai pada separuh yang terbuka: patinya dibuat di tempat
cahaya jatuh. Karbon dioksida radioaktif yang dipasok kepada tumbuhan
berakhir dalam gulanya dalam hitungan menit.
\end{proof}

\begin{example}[Euglena, keduanya sekaligus]\label{ex:g10:cell-metabolism:euglena}
\emph{Euglena} yang bersel satu punya \omterm{def:g10:cells-common-unit:organelle}{kloroplas} dan berenang dengan
bulu cambuk. Dalam cahaya, di dalam air mineral, ia tumbuh: \omterm{def:g10:cell-metabolism:autotrophy}{autotrof}.
Dalam gelap ia hanya bertahan jika molekul organik ditambahkan ke
airnya, yang lalu diserapnya: \omterm{def:g10:cell-metabolism:autotrophy}{heterotrof}. Jika lama disimpan dalam
gelap ia kehilangan \omterm{def:g10:cells-common-unit:organelle}{kloroplasnya}; jika dibawa kembali ke cahaya dengan
kesabaran beberapa hari ia membuatnya lagi. \omterm{def:g10:cell-metabolism:metabolism}{Metabolismenya} bergantung
pada \omterm{def:g10:universal-dna:gene}{gennya}, yang membolehkan kedua pilihan itu, dan pada
lingkungannya, yang memilih salah satunya.
\end{example}

\section{Memperoleh energi dari molekul organik}

\begin{proposition}[Respirasi seluler]\label{prop:g10:cell-metabolism:respiration}
Setiap \omterm{def:g10:cells-common-unit:cell}{sel}, \omterm{def:g10:cell-metabolism:autotrophy}{autotrof} maupun \omterm{def:g10:cell-metabolism:autotrophy}{heterotrof}, memperoleh energi yang terpakai
dengan membongkar molekul organik. Bila oksigen tersedia,
pembongkarannya lengkap --- inilah \emph{respirasi
seluler}\index{respirasi seluler}, di dalam \omterm{def:g10:cells-common-unit:organelle}{mitokondria}:
\[
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \longrightarrow
6\,\mathrm{CO_2} + 6\,\mathrm{H_2O} + \text{energi},
\]
sekitar \qty{2870}{kJ} tiap mol glukosa (\qty{180}{g}), yang
kira-kira 40\% darinya tertangkap dalam molekul kecil,
\emph{ATP}\index{ATP}, yang menjadi sandaran setiap proses \omterm{def:g10:cells-common-unit:cell}{sel} yang
menuntut energi --- gerakan, pengangkutan, penyusunan; sisanya
dilepaskan sebagai panas.
\end{proposition}
\begin{proof}[Bukti percobaan]
Ragi dalam bejana tertutup dengan pengukur oksigen tidak memakai
oksigen selama dibiarkan lapar; tambahkan glukosa dan oksigen
terlarutnya turun dengan laju yang mantap sementara karbon dioksida
muncul, sampai oksigennya habis. Biji yang berkecambah dalam termos
menghangatkan termosnya beberapa derajat dalam sehari --- panas
respirasinya --- sedangkan biji yang direbus tidak. \omterm{def:g10:cells-common-unit:cell}{Sel} otot yang
dilihat dengan mikroskop elektron penuh sesak dengan \omterm{def:g10:cells-common-unit:organelle}{mitokondria}, dan
racun \omterm{def:g10:cells-common-unit:organelle}{mitokondria} menghentikan ototnya.
\end{proof}

\begin{definition}[Fermentasi]\label{def:g10:cell-metabolism:fermentation}
Tanpa oksigen, sebagian \omterm{def:g10:cells-common-unit:cell}{sel} hanya membongkar glukosa sebagian, di dalam
\omterm{def:g10:cells-common-unit:cell}{sitoplasma}, menjadi molekul organik yang lebih kecil yang masih menahan
sebagian besar energinya: inilah \emph{fermentasi}\index{fermentasi}.
Ragi menjalankan fermentasi alkohol,
\[
\mathrm{C_6H_{12}O_6} \longrightarrow 2\,\mathrm{C_2H_5OH} +
2\,\mathrm{CO_2} + \text{energi},
\]
dan \omterm{def:g10:cells-common-unit:cell}{sel} otot yang kekurangan oksigen, begitu pula bakteri yoghurt,
menjalankan fermentasi laktat, yang mengubah glukosa menjadi asam
laktat. Fermentasi menghasilkan ATP kira-kira lima belas kali lebih
sedikit tiap glukosa daripada respirasi.
\end{definition}

\begin{example}[Ragi dengan udara dan tanpa udara]\label{ex:g10:cell-metabolism:pasteur}
Ragi yang dialiri gelembung udara memakai glukosa dengan hemat dan
tumbuh cepat: ia berespirasi. Ragi yang sama dalam tong tertutup
memakai glukosa dengan rakus, sedikit tumbuh dan membuat etanol: ia
berfermentasi. Pasteur, yang melukiskan hal ini pada 1861, menyebut
\omterm{def:g10:cell-metabolism:fermentation}{fermentasi} ``kehidupan tanpa udara''; \omterm{def:g10:cells-common-unit:cell}{selnya} mengalihkan \omterm{def:g10:cell-metabolism:metabolism}{metabolismenya}
ke jalur mana pun yang dibolehkan lingkungannya, dan, karena \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}
memberi energi yang sangat sedikit tiap glukosa, \omterm{def:g10:cells-common-unit:cell}{sel} itu harus
menghabiskan gula yang jauh lebih banyak untuk hidup.
\end{example}

\section{Apa yang menentukan metabolisme sebuah sel}

\begin{proposition}[Gen dan lingkungan]\label{prop:g10:cell-metabolism:decides}
\omterm{def:g10:cell-metabolism:metabolism}{Metabolisme} sebuah \omterm{def:g10:cells-common-unit:cell}{sel} bergantung pada dua hal: pada
\emph{perlengkapannya} --- \omterm{def:g10:cell-metabolism:metabolism}{enzim} dan \omterm{def:g10:cells-common-unit:organelle}{organel} yang boleh dibangunnya
menurut \omterm{def:g10:universal-dna:gene}{gennya}, sehingga \omterm{def:g10:cells-common-unit:cell}{sel} tanpa klorofil tidak akan pernah dapat
berfotosintesis --- dan pada \emph{lingkungannya} --- cahaya, oksigen,
zat gizi yang tersedia --- yang menentukan jalur mana di antara yang
mungkin itu yang sungguh berjalan. Ragi berespirasi di udara dan
berfermentasi tanpa udara; \emph{Euglena} bersifat \omterm{def:g10:cell-metabolism:autotrophy}{autotrof} dalam
cahaya dan \omterm{def:g10:cell-metabolism:autotrophy}{heterotrof} dalam gelap; \omterm{def:g10:cells-common-unit:cell}{sel} akar tumbuhan hijau bersifat
\omterm{def:g10:cell-metabolism:autotrophy}{heterotrof} seumur hidupnya, diberi makan oleh daunnya.
\end{proposition}
\admitted

\begin{method}[Membaca sebuah rekaman metabolisme]\label{met:g10:cell-metabolism:trace}
Percobaan dengan pengukur oksigen memberi kepekatan terhadap waktu.
\begin{enumerate}
  \item Kenali apa yang berubah pada tiap saat yang ditandai (lampu
        menyala, substrat ditambahkan, oksigen habis).
  \item Baca kemiringan tiap penggal yang lurus: naiknya dibagi
        mendatarnya, dalam \unit{mg/L} tiap menit. Kemiringan positif
        berarti penghasilan bersih oksigen, kemiringan negatif berarti
        pemakaian bersih.
  \item Ingatlah bahwa \omterm{def:g10:cells-common-unit:cell}{sel} yang berfotosintesis juga berespirasi:
        kemiringan dalam cahaya adalah penghasilan dikurangi
        respirasinya, sehingga laju \omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} yang sebenarnya adalah
        kemiringan terang ditambah besarnya kemiringan gelap.
  \item Setarakan jika diminta: pada kepekatan ini \qty{1}{mg/L}
        oksigen terlarut sama dengan $1/32$ milimol tiap liter.
\end{enumerate}
\end{method}

\begin{example}[Laju sebenarnya alga itu]\label{ex:g10:cell-metabolism:truerate}
Pada gambar \emph{Chlorella}, oksigennya naik \qty{2.5}{mg/L} dalam
5 menit cahaya (kemiringan $+0.5$) dan turun \qty{0.5}{mg/L} dalam 5
menit gelap (kemiringan $-0.1$). Respirasi memakai \qty{0.1}{mg/L}
tiap menit dalam cahaya juga, sehingga \omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} menghasilkan
$0.5 + 0.1 = \qty{0.6}{mg/L}$ tiap menit: enam kali yang dipakai
respirasinya. Alga itu adalah penghasil bersih oksigen dan \omterm{def:g10:chemistry-of-life:mineral-organic}{materi
organik}, dan itulah sebabnya ia tumbuh.
\end{example}

\section{Soal: Tong Pembuat Anggur}

\begin{problem}[{Soal akhir pekan --- setong sari buah anggur yang
diserahkan kepada ragi: alkohol yang akan dikandungnya, gas yang akan
dilepaskannya, panas yang akan dibuatnya, dan mengapa ruang bawah tanah
dapat menjadi tempat yang berbahaya}]
\label{pb:g10:cell-metabolism:1}
Sari buah anggur memuat sekitar \qty{200}{g} gula tiap liter, yang kita
ambil sebagai glukosa, dan tanpa oksigen terlarut begitu tongnya
ditutup. Massa molar: glukosa \qty{180}{g/mol}, etanol
$\mathrm{C_2H_5OH}$ \qty{46}{g/mol}, karbon dioksida \qty{44}{g/mol},
oksigen \qty{32}{g/mol}. Massa jenis etanol \qty{0.79}{g/mL}. Satu mol
gas menempati sekitar \qty{24}{L} pada suhu ruang bawah tanah. Tongnya
bermuatan \qty{1000}{L}.

\medskip
\noindent\textbf{Bagian I --- Apa yang dibuat raginya.}
\begin{enumerate}
  \item Jalur mana yang dipakai raginya dalam tong tertutup itu, dan
        mengapa? Tuliskan persamaannya.
  \item Berapa mol glukosa yang dimuat satu liter sarinya?
  \item Hitung massa etanol yang dihasilkan tiap liter, lalu volumenya,
        lalu kadar alkohol anggurnya dalam persen menurut volume.
  \item Hitung massa karbon dioksida yang dihasilkan tiap liter
        sarinya, dan volumenya sebagai gas.
  \item Untuk seluruh tongnya, berapa volume karbon dioksida yang
        dilepaskan? Bandingkan dengan volume ruang bawah tanah
        berukuran \qty{4}{m} kali \qty{6}{m} kali \qty{2.5}{m}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Seandainya raginya beroleh udara.}
\begin{enumerate}[resume]
  \item Tuliskan persamaan yang akan diikuti raginya jika sarinya terus
        dialiri udara.
  \item Hitung massa oksigen yang diperlukan untuk merespirasikan gula
        satu liter sarinya, dan massa karbon dioksida yang dilepaskan.
  \item Bandingkan karbon dioksida pada pertanyaan 7 dengan karbon
        dioksida pada pertanyaan 4, dan jelaskan bedanya dari sejauh
        mana glukosanya terbongkar.
  \item Respirasi menghasilkan sekitar 30 ATP tiap glukosa dan
        \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} 2. Jika raginya memerlukan ATP yang sama pada kedua
        hal itu, pada hal mana gulanya bertahan lebih lama, dan dengan faktor berapa?
  \item Jelaskan mengapa pembuat anggur tetap menjauhkan udara.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Panas tongnya.}
\omterm{def:g10:cell-metabolism:fermentation}{Fermentasi} melepaskan sekitar \qty{70}{kJ} panas tiap mol glukosa;
memanaskan \qty{1}{kg} air sebesar \qty{1}{\celsius} menuntut \qty{4.2}{kJ}.
\begin{enumerate}[resume]
  \item Hitung panas yang dilepaskan tiap liter sarinya, lalu untuk
        seluruh tongnya.
  \item Jika tidak ada satu pun panas itu yang lolos, berapa suhu
        tongnya akan naik? (Ambil sarinya sebagai air.)
  \item \omterm{def:g10:cell-metabolism:metabolism}{Enzim} ragi rusak di atas sekitar \qty{40}{\celsius}. Mulai dari
        \qty{20}{\celsius}, apakah tong tanpa pendingin akan bertahan?
        Apa yang diberitahukannya tentang mengapa tong besar didinginkan?
  \item Jelaskan, dengan \cref{prop:g10:cell-metabolism:enzymes},
        mengapa panas menghentikan \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} walaupun gulanya masih ada.
  \item Respirasi akan melepaskan sekitar \qty{2870}{kJ} tiap mol. Di
        mana energi yang \emph{tidak} dilepaskan \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} itu tinggal?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Ruang bawah tanahnya.}
\begin{enumerate}[resume]
  \item Karbon dioksida lebih rapat daripada udara. Di bagian mana
        ruang bawah tanah itu ia menumpuk, dan mengapa lilin menyala
        yang dibawa rendah menjadi uji keselamatan yang lazim?
  \item \omterm{def:g10:cells-common-unit:cell}{Sel} seorang pekerja, jika kekurangan oksigen, akan beralih ke
        jalur yang mana --- dan mengapa tubuh manusia tidak dapat hidup
        dengan jalur itu lebih dari beberapa menit?
  \item \omterm{def:g10:cell-metabolism:fermentation}{Fermentasinya} berhenti dengan sendirinya ketika alkoholnya
        mencapai sekitar 15\% menurut volume. Usulkan penjelasannya
        dalam istilah \omterm{def:g10:cell-metabolism:metabolism}{enzim} raginya.
  \item Adonan roti memakai ragi yang sama selama satu jam. Apa isi
        gelembung dalam remahnya, dan ke mana etanolnya pergi setelah
        dipanggang?
  \item Nyatakan hasilnya: untuk satu liter sarinya, kadar alkoholnya
        dan volume gas yang dihasilkan --- serta satu kenyataan
        \omterm{def:g10:cell-metabolism:metabolism}{metabolisme} yang menjelaskan keduanya.
\end{enumerate}
\end{problem}
