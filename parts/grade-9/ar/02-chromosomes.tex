\chapter{الصبغيات}\label{ch:g9:chromosomes}

حشرت \Cref{rem:g9:heredity-and-traits:question} المحددات
في \omterm{def:g6:cell-unit-of-life:parts}{النواة}. وإذا أمسكت \omterm{def:g5:first-look-at-cells:cell}{بخلية} في اللحظة المناسبة كشفت
\omterm{def:g6:cell-unit-of-life:parts}{النواة} عن يدها: فمحتوياتها تلتف حتى تصير أجسامًا تعد
وتصوّر --- هي \omterm{def:g9:chromosomes:chromosomes}{الصبغيات}. ويتبين أن عدّها، في \omterm{def:g5:first-look-at-cells:cell}{خلايا} الجسم
وفي \omterm{def:g8:sexual-reproduction:gametes}{الأمشاج}، يفسر كل ما طلبه الألبوم العائلي تقريبًا:
الأنصاف، والأزواج، والخلط، والقرعة التي تقرر أذكرًا يكون
المولود أم \omterm{def:g4:animal-reproduction:sexual}{أنثى}.

\section{النواة، ملتقطة متلبسة}

\begin{definition}[الصبغيات]\label{def:g9:chromosomes:chromosomes}
\emph{الصبغي}\index{صبغي} جسم خيطي في \omterm{def:g6:cell-unit-of-life:parts}{النواة}، يرى (إذا
صبغ جيدًا، وقت الانقسام) حين تلتف محتويات \omterm{def:g6:cell-unit-of-life:parts}{النواة}
استعدادًا للنقل. وبين الانقسامات تجري الخيوط نفسها
منفرطة غير مرئية --- وهي حالة العمل؛ وعند الانقسام
تتكاثف --- وهي حالة التنقل. وإذا صوّرت صبغيات \omterm{def:g5:first-look-at-cells:cell}{خلية} ورتبت
صارت \emph{نمطها الصبغي}\index{نمط صبغي}: مجموعة الورق
كلها، مبسوطة.
\end{definition}

\begin{proposition}[الأعداد البشرية]\label{prop:g9:chromosomes:counts}
تقدم الأنماط الصبغية ثلاثة أعداد تحمل الفصل كله:
\begin{enumerate}
  \item كل \omterm{def:g5:first-look-at-cells:cell}{خلية} من \omterm{def:g5:first-look-at-cells:cell}{خلايا} جسم الإنسان تحمل \emph{46}
        صبغيًا --- \omterm{def:g1:the-five-senses:sense}{الجلد} \omterm{prop:g7:digestion-and-nutrients:absorption}{والكبد} \omterm{def:g8:nervous-communication:neuron}{والعصبون} سواء (فانقسامات
        \cref{prop:g5:first-look-at-cells:growth} تنسخ
        المجموعة نسخًا أمينًا إلى كل \omterm{def:g5:first-look-at-cells:cell}{خلية})؛
  \item وإذا بسطت انتظمت في \emph{23 زوجًا}، يتطابق
        شريكا الزوج في الحجم والتخطيط؛
  \item و\emph{\omterm{def:g8:sexual-reproduction:gametes}{الأمشاج}} تخرق القاعدة: \omterm{def:g4:animal-reproduction:sexual}{فالبويضة} \omterm{def:g8:reproductive-systems:male}{والنطفة}
        تحملان \emph{23} --- صبغيًا واحدًا من كل زوج، ولا
        زوج كامل.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[حساب الإخصاب]\label{ex:g9:chromosomes:arithmetic}
تنطبق الأعداد بعضها على بعض كالبرهان. المشيجان 23 لكل
واحد؛ \omterm{def:g5:flower-to-fruit:steps}{والإخصاب}
(\cref{prop:g8:sexual-reproduction:core}) يجمعهما:
فتحمل \omterm{prop:g8:sexual-reproduction:core}{اللاقحة} $23 + 23 = 46$ --- وقد استعيدت الأزواج،
شريك من كل زوج من الأم، وشريك من الأب. ثم تحمل كل \omterm{def:g5:first-look-at-cells:cell}{خلية}
من \omterm{def:g5:first-look-at-cells:cell}{خلايا} جسم الولد إسهام الوالدين معًا --- وهو بعينه ما
لاحظه \cref{rem:g4:animal-reproduction:mix} من الخارج،
بالمهر والغرة، منذ جيل من الفصول.
\end{example}

\begin{remark}[لماذا وجب أن تنصّف الأمشاج]\label{rem:g9:chromosomes:whyhalve}
افترض أنها لم تفعل: لصنعت \omterm{def:g8:sexual-reproduction:gametes}{أمشاج} ذات 46 صبغيًا أولادًا
ذوي 92، وأحفادًا ذوي 184 --- فتتضاعف المجموعة كل جيل.
وانقسام \omterm{def:g8:sexual-reproduction:gametes}{الأمشاج} المنصّف الخاص --- إذ يتلقى كل واحد
\emph{شريكًا واحدًا من كل زوج} --- هو الذي يبقي عدد
\omterm{def:g5:biodiversity:species}{النوع} ثابتًا عند 46، جيلًا بعد جيل. ولاحظ المكافأة:
\emph{أي} شريك من كل زوج يتلقاه \omterm{def:g8:sexual-reproduction:gametes}{المشيج} أمر يقرر مستقلًا،
زوجًا زوجًا --- $2^{23}$ مجموعة ممكنة من والد واحد، وقد
عُدّ أخيرًا خلط
\cref{prop:g8:sexual-reproduction:shuffle}.
\end{remark}

\section{الصبغيات تحمل المحددات}

\begin{proposition}[الدليل على أن للمجموعة شأنًا]\label{prop:g9:chromosomes:evidence}
وكون \omterm{def:g9:chromosomes:chromosomes}{الصبغيات} \emph{هي} مركبات المحددات ليس افتراضًا بل
دليلًا:
\begin{itemize}
  \item فهي تسلك تمامًا كما يقتضي
        \cref{prop:g9:heredity-and-traits:conclusions}:
        حاضرة في كل \omterm{def:g5:first-look-at-cells:cell}{خلية}، منصّفة في \omterm{def:g8:sexual-reproduction:gametes}{الأمشاج}، مستعادة عند
        \omterm{def:g5:flower-to-fruit:steps}{الإخصاب}، منتقاة شريكًا شريكًا؛
  \item وحوادث \omterm{def:g9:chromosomes:chromosomes}{النمط الصبغي} تثبت الحمولة: فمجموعة فيها
        \omterm{def:g9:chromosomes:chromosomes}{صبغي} زائد أو ناقص تغير نمو الجسم كله --- فنسخة
        زائدة من الزوج الصغير 21، مثلًا، تنتج مجموعة
        \omterm{def:g9:heredity-and-traits:traits}{الصفات} المعروفة باسم متلازمة داون. خيط واحد أخطئ
        في عده، \omterm{def:g9:heredity-and-traits:traits}{وصفات} كثيرة انزاحت: إذن الخيوط تحمل
        تعليمات \omterm{def:g9:heredity-and-traits:traits}{الصفات}؛
  \item وزوج واحد يعلن حمولته في كل \omterm{def:g9:chromosomes:chromosomes}{نمط صبغي}: هما
        \emph{الصبغيان الجنسيان}.
\end{itemize}
\end{proposition}
\admitted

\begin{definition}[X وY، وقرعة النقود]\label{def:g9:chromosomes:xy}
الزوج 23 يأتي في نسختين: صبغيا \emph{X} متطابقان --- وهو
\omterm{def:g9:chromosomes:chromosomes}{النمط الصبغي} \omterm{def:g4:animal-reproduction:sexual}{للأنثى}؛ أو \omterm{def:g9:chromosomes:chromosomes}{صبغي} X \omterm{def:g9:chromosomes:chromosomes}{وصبغي} \emph{Y} صغير ---
وهو \omterm{def:g4:animal-reproduction:sexual}{الذكر}. ثم يجري التنصيف القرعة: فكل \omterm{def:g4:animal-reproduction:sexual}{بويضة} تحمل X؛
\omterm{def:g8:reproductive-systems:male}{والنطفة} تحمل X أو Y، مناصفة. \omterm{def:g5:flower-to-fruit:steps}{والإخصاب} \omterm{def:g8:reproductive-systems:male}{بنطفة} تحمل X يعطي
XX، وبنطفة تحمل Y يعطي XY --- فرصة من اثنتين (لغة الحظ
في \cref{ch:g9:heredity-and-traits}، ومعها آليتها
أخيرًا)، تقررها السابحة الواصلة، في لحظة \omterm{def:g5:flower-to-fruit:steps}{الإخصاب}.
\end{definition}

\begin{example}[أسئلة قديمة، وأجوبة قصيرة]\label{ex:g9:chromosomes:answers}
المجموعة تحسم خصومات تاريخية. فكون الولد ذكرًا أو \omterm{def:g4:animal-reproduction:sexual}{أنثى}
لا يدين بشيء لأماني أي من الوالدين ولا لطعامهما ولا
لفصول السنة --- إنما هو القرعة، تجري بآلية
\cref{def:g9:chromosomes:xy} (ولتسجيل مظالم قديمة:
الفرق الحاسم يركب في \emph{\omterm{def:g8:reproductive-systems:male}{النطفة}}). والأنصاف المتساوية
تجيب الألبوم العائلي أيضًا: فالأم والأب يسهم كل واحد
منهما بنصف مجموعة كامل --- 23 و23 --- وحساب \omterm{def:g9:heredity-and-traits:heredity}{الوراثة}
يكرّم الخطين سواء.
\end{example}

\begin{method}[قراءة نمط صبغي]\label{met:g9:chromosomes:read}
إذا سلّمت صورة \omterm{def:g9:chromosomes:chromosomes}{نمط صبغي}:
\begin{enumerate}
  \item عُدّ: 46 متوقعة في \omterm{def:g5:first-look-at-cells:cell}{خلية} جسمية، و23 في \omterm{def:g8:sexual-reproduction:gametes}{مشيج}؛
  \item وازدوج: يطابق بين الشريكين الحجم والتخطيط، 22
        زوجًا عاديًا مع الزوج الثالث والعشرين؛
  \item واقرأ الزوج الثالث والعشرين: XX أم XY؛
  \item وتحقق من الأعداد: فأي زوج فيه فرد واحد أو ثلاثة
        يفسر، وفق \cref{prop:g9:chromosomes:evidence}،
        نموًا منزاحًا.
\end{enumerate}
\end{method}

\section{مسألة: منضدة الوراثة الخلوية}

\begin{problem}[{مسألة نهاية الأسبوع --- أربعة أنماط
صبغية على منضدة إضاءة واحدة}]\label{pb:g9:chromosomes:1}
منضدة تعليمية في مختبر \omterm{def:g9:heredity-and-traits:heredity}{وراثة} خلوية بمستشفى: \omterm{def:g9:chromosomes:chromosomes}{النمط الصبغي}
أ --- 46 صبغيًا، والزوج 23 هو XY؛ وب --- 46، والزوج 23
هو XX؛ وج --- 23 صبغيًا، وفيها X واحد؛ ود --- 47، بثلاث
نسخ عند الزوج 21، والزوج 23 هو XX.

\medskip\noindent\textbf{الجزء الأول --- القراءات.}
\begin{enumerate}
  \item اقرأ أ وب وج وفق \cref{met:g9:chromosomes:read}:
        \omterm{def:g5:first-look-at-cells:cell}{خلايا} من يمكن أن يكون كل واحد منها؟
  \item عدد ج ليس خطأ. أي نوع من \omterm{def:g5:first-look-at-cells:cell}{الخلايا} هو، وأي تاريخ
        من الانقسام يسجله عدده 23؟
  \item اقرأ د: العدد، والازدواج، والنمو الذي يعلنه.
  \item \omterm{def:g8:sexual-reproduction:gametes}{مشيج} أي والد قرر الزوج 23 في أ، وما كانت
        الاحتمالات عند ذلك \omterm{def:g5:flower-to-fruit:steps}{الإخصاب}؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- أسئلة
الحساب.}
\begin{enumerate}[resume]
  \item يسأل متدرب لماذا تتفق \omterm{def:g5:first-look-at-cells:cell}{خلايا} جسم الشخص الواحد
        كلها مع النمط أ. أجب بانقسامات النسخ في
        \cref{prop:g5:first-look-at-cells:growth}.
  \item ويسأل آخر كيف صنع والدان خلاياهما 46 عدد ج
        البالغ 23. سمّ الانقسام الخاص وقاعدته.
  \item اجمع ج مع \omterm{def:g4:animal-reproduction:sexual}{بويضة} تحمل X، ومع جاره في المختبر
        الحامل \omterm{def:g9:chromosomes:chromosomes}{للصبغي} Y --- مهلًا: ج \emph{خال} من Y.
        اذكر ما يضمنه X في ج عن أي ولد يسهم في صنعه.
  \item احسب المجموعات الممكنة التي يستطيع والد واحد
        توزيعها: اذكر الاستقلال زوجًا زوجًا ونتيجة
        $2^{23}$ --- ثم أضف ازدواج \omterm{def:g5:flower-to-fruit:steps}{الإخصاب} بين توزيعين
        كهذين.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- غرفة
الإرشاد.}
\begin{enumerate}[resume]
  \item يسأل والدا طفل نمطه \omterm{def:g9:chromosomes:chromosomes}{الصبغي} د: ``ما الخطأ الذي
        ارتكبناه؟'' أعط جواب الآلية الصادق عن حوادث
        التنصيف --- وما الذي تغيره خيوط الطفل السبعة
        والأربعون وما لا تغيره في المحبة القائمة في
        الغرفة. (أجب بوصفك المرشد، والبيولوجيا صريحة.)
  \item يسأل زوجان لهما ثلاث بنات عن احتمال ابن في
        المرة القادمة. أجب بالقرعة، وصحّح غريزة سلسلة
        الحظ.
  \item يدعي زائر أن حمية ما تستطيع اختيار جنس المولود.
        دقّق الادعاء في مقابل
        \cref{def:g9:chromosomes:xy}: أين يتخذ القرار،
        وما الذي تستطيع الحمية بلوغه؟
  \item اختم المنضدة بجملة واحدة: ما الذي تثبته الصور
        الأربع مجتمعة عن موضع ركوب \omterm{def:g9:heredity-and-traits:heredity}{الوراثة}.
\end{enumerate}
\end{problem}
