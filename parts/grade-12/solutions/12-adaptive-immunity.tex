\section*{Chapter \ref{ch:g12:adaptive-immunity} --- Adaptive Immunity and Vaccination}

\begin{solution}{exo:g12:adaptive-immunity:1}
An \omterm{def:g12:adaptive-immunity:antigen}{antigen} is any molecule the adaptive system can recognise; a
\omterm{def:g12:adaptive-immunity:antigen}{lymphocyte} is a white \omterm{def:g10:cells-common-unit:cell}{cell} carrying receptors for one \omterm{def:g12:adaptive-immunity:antigen}{antigen}. Its
specificity comes from its receptor, assembled at random during
maturation and identical on all its copies.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:2}
The \omterm{def:g12:adaptive-immunity:antigen}{antigen} binds the few \omterm{def:g12:adaptive-immunity:antigen}{lymphocytes} whose receptor fits; those \omterm{def:g10:cells-common-unit:cell}{cells}
multiply into a clone over days; the clone differentiates into
effector \omterm{def:g10:cells-common-unit:cell}{cells} that fight; a minority become \omterm{prop:g12:adaptive-immunity:selection}{memory cells} that
persist.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:3}
A Y-shaped \omterm{def:g11:gene-expression:protein}{protein} secreted by plasma \omterm{def:g10:cells-common-unit:cell}{cells}, a soluble copy of the B
\omterm{def:g10:cells-common-unit:cell}{cell}'s receptor; each arm binds the \omterm{def:g12:adaptive-immunity:antigen}{antigen}. On a bacterium it forms
a coat that \omterm{def:g12:innate-immunity:phagocytosis}{phagocytes} grip, and cross-links bacteria into complexes
that are engulfed.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:4}
\omterm{prop:g12:adaptive-immunity:tcells}{Helper T cells}, activated by \omterm{prop:g12:innate-immunity:handover}{dendritic cells}, secrete the cytokines
that authorise and amplify the B \omterm{def:g10:cells-common-unit:cell}{cells} and the other T \omterm{def:g10:cells-common-unit:cell}{cells}.
Cytotoxic T \omterm{def:g10:cells-common-unit:cell}{cells} kill \omterm{def:g10:cells-common-unit:cell}{cells} displaying fragments of a virus or a
\omterm{def:g11:cancer:cancer}{tumour} \omterm{def:g11:gene-expression:protein}{protein}.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:5}
A harmless form of the pathogen's \omterm{def:g12:adaptive-immunity:antigen}{antigen} (weakened, killed, a
\omterm{def:g11:gene-expression:protein}{protein}, a toxin inactivated, or its instructions) with an adjuvant.
It provokes a primary response and leaves \omterm{prop:g12:adaptive-immunity:selection}{memory cells}, so that the
real pathogen meets a secondary response.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:6}
Primary: a week's delay, a low peak (about a tenth of the
secondary's) at two to three weeks, a decline within a month.
Secondary: two or three days' delay, a peak ten times higher within
ten days, persisting for months.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:7}
Memory is specific: the \omterm{prop:g12:adaptive-immunity:selection}{memory cells} left by the first dose of A
respond to A only; B, met for the first time, gets a primary response.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:8}
\omterm{prop:g12:adaptive-immunity:antibodies}{Antibodies} are \omterm{def:g11:gene-expression:protein}{proteins} in the fluids and cannot enter \omterm{def:g10:cells-common-unit:cell}{cells}. An
infected \omterm{def:g10:cells-common-unit:cell}{cell} displays fragments of the virus on its surface, and
cytotoxic T \omterm{def:g10:cells-common-unit:cell}{cells} recognise them and kill the \omterm{def:g10:cells-common-unit:cell}{cell}.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:9}
$1 - 1/4 = 75\%$; $1 - 1/18 \approx 94\%$.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:10}
Above the threshold each case infects fewer than one other, so the
virus cannot circulate and rarely reaches an infant; the immune
adults are the wall around them.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:11}
About 750, 520 and 180 per microlitre; \omterm{prop:g12:adaptive-immunity:hiv}{AIDS} begins around year 9,
when the count crosses 200.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:12}
The \omterm{prop:g12:adaptive-immunity:tcells}{helper T cells}' cytokines are what authorise the B \omterm{def:g10:cells-common-unit:cell}{cells} to become
plasma \omterm{def:g10:cells-common-unit:cell}{cells} and the cytotoxic T \omterm{def:g10:cells-common-unit:cell}{cells} to multiply; both arms depend
on the same coordinator, so removing it silences both.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:13}
The \omterm{prop:g12:adaptive-immunity:antibodies}{antibodies} were selected against the virus as it was; the virus
mutates its surface \omterm{def:g11:gene-expression:protein}{proteins} faster than new clones can be selected,
so each response is outrun. And the virus hides and multiplies inside
the very helper \omterm{def:g10:cells-common-unit:cell}{cells} that would coordinate the response.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:14}
\omterm{prop:g12:adaptive-immunity:selection}{Memory cells} recognise the surface \omterm{def:g11:gene-expression:protein}{proteins} the \omterm{def:g12:adaptive-immunity:vaccine}{vaccine} presented.
Influenza's mutate every year into forms the old memory does not bind;
measles' surface \omterm{def:g11:gene-expression:protein}{proteins} hardly change, so the memory of 1970 still
fits the virus of today.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:15}
What varies: the receptors of the \omterm{def:g12:adaptive-immunity:antigen}{lymphocytes}, made at random. What
selects: the \omterm{def:g12:adaptive-immunity:antigen}{antigen}, which lets only fitting clones multiply. What is
inherited: the receptor, by the clone's descendants, including the
\omterm{prop:g12:adaptive-immunity:selection}{memory cells}. Time scale: days for a response, a lifetime for memory,
against generations for a \omterm{def:g12:selection-drift-speciation:population}{population}. The analogy stops at
transmission: the selected clones die with the body and are not
passed to the next generation.
\end{solution}

\begin{solution}{pb:g12:adaptive-immunity:1}
\textbf{1.} Nothing for about a week, a modest peak at two to three
weeks, a decline over the following weeks.

\textbf{2.} \omterm{prop:g12:adaptive-immunity:antibodies}{Antibodies} within two or three days, a peak ten times
higher, lasting months: the \omterm{prop:g12:adaptive-immunity:selection}{memory cells} left by the first dose are
already a large clone, so multiplication starts from thousands of
\omterm{def:g10:cells-common-unit:cell}{cells} instead of a few.

\textbf{3.} A living virus multiplies briefly inside \omterm{def:g10:cells-common-unit:cell}{cells}, so
infected \omterm{def:g10:cells-common-unit:cell}{cells} display its fragments and cytotoxic T \omterm{def:g10:cells-common-unit:cell}{cells} are
selected as well as \omterm{prop:g12:adaptive-immunity:antibodies}{antibodies}; a killed virus trains mainly the
\omterm{prop:g12:adaptive-immunity:antibodies}{antibody} arm.

\textbf{4.} \omterm{prop:g12:adaptive-immunity:antibodies}{Antibodies} already present block much of the virus;
memory B and T \omterm{def:g10:cells-common-unit:cell}{cells} multiply within two days; infected \omterm{def:g10:cells-common-unit:cell}{cells} are
killed before the virus spreads. The infection is stopped before
symptoms.

\textbf{5.} Three days of silent multiplication, then fever, rash and
illness while the primary response builds over a week; recovery
around day ten, with \omterm{prop:g12:adaptive-immunity:selection}{memory cells} and lifelong immunity --- if he
survives the complications.

\textbf{6.} $1 - 1/15 \approx 93\%$.

\textbf{7.} $15 \times 0.10 = 1.5$: each case infects more than one
other, so the disease can spread. Not protected.

\textbf{8.} $10\% \times \num{50000} = 5000$ not immune, of whom 1500
are the infants.

\textbf{9.} 2500 susceptible; $15 \times 0.05 = 0.75$: fewer than one,
the disease dies out.

\textbf{10.} Infants cannot be vaccinated, so their only protection is
that the adults around them, by being immune, leave the virus no path
to reach them.

\textbf{11.} Generation 3: $1.5^3 \approx 3.4$ cases; generation 6:
$1.5^6 \approx 11$.

\textbf{12.} $1 + 1.5 + 2.25 + 3.4 + 5.1 + 7.6 + 11.4 \approx 32$ cases.

\textbf{13.} About 30\% of 32: some 10 infants.

\textbf{14.} Each case removes a susceptible person and adds an immune
one, so the susceptible fraction falls and with it the number each
case infects; with 5000 susceptible the outbreak would slow only after
hundreds of cases, over months --- unless vaccination intervenes.

\textbf{15.} At 95\%: $0.75^3 \approx 0.4$ and $0.75^6 \approx 0.18$
cases; total about $1 + 0.75 + 0.56 + \dots \approx 4$ cases. Ten
times fewer cases, and the outbreak extinguishes itself.

\textbf{16.} 800 in 8 years: 100 per microlitre per year, about 8 per
month.

\textbf{17.} B \omterm{def:g10:cells-common-unit:cell}{cells} need the helper \omterm{def:g10:cells-common-unit:cell}{cells}' cytokines to become plasma
\omterm{def:g10:cells-common-unit:cell}{cells}, and cytotoxic T \omterm{def:g10:cells-common-unit:cell}{cells} need them to multiply: with too few
helpers, neither arm is authorised, though their \omterm{def:g10:cells-common-unit:cell}{cells} exist.

\textbf{18.} Yes: \omterm{prop:g12:adaptive-immunity:selection}{memory cells} still need helper cytokines to expand
into a secondary response; without coordination the memory cannot be
used, and measles, harmless to a healthy adult, can be fatal.

\textbf{19.} The helper \omterm{def:g12:selection-drift-speciation:population}{population} regrows from surviving \omterm{def:g10:cells-common-unit:cell}{cells} and
the marrow, and the \omterm{prop:g12:adaptive-immunity:selection}{memory cells} of the other arms were never
destroyed --- the repertoire is intact once coordination returns. The
virus persists in hidden \omterm{def:g10:cells-common-unit:cell}{cells}, so stopping the drugs lets it resume.

\textbf{20.} 93\% coverage; about 32 cases from one traveller at 90\%;
the \omterm{prop:g12:adaptive-immunity:tcells}{helper T cell}.
\end{solution}
