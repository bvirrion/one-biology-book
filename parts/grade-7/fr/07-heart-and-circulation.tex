\chapter{Le cœur et la circulation}\label{ch:g7:heart-and-circulation}

\Cref{ch:g4:heart-and-blood} présentait la pompe et ses tournées de
livraison~; depuis, le \omterm{prop:g4:journey-of-food:absorb}{sang} a pris des responsabilités --- les
\omterm{def:g7:digestion-and-nutrients:families}{nutriments} aux \omterm{prop:g7:digestion-and-nutrients:absorption}{villosités}
(\cref{prop:g7:digestion-and-nutrients:absorption}), le \omterm{def:g7:what-is-respiration:respiration}{dioxygène}
aux \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alvéoles}
(\cref{prop:g7:lungs-and-gas-exchange:numbers}), des factures à
chaque \omterm{def:g3:bones-and-muscles:muscle}{muscle} au travail (\cref{prop:g7:organs-at-work:bill}). Il
est temps de cartographier le réseau routier pour de bon~: les
quatre cavités du \omterm{def:g4:heart-and-blood:heart}{cœur}, les deux circuits et les trois sortes de
vaisseaux.

\section{Les quatre cavités du cœur}

\begin{definition}[Oreillettes et ventricules]\label{def:g7:heart-and-circulation:rooms}
Chacun des deux côtés du \omterm{def:g4:heart-and-blood:heart}{cœur}
(\cref{def:g4:heart-and-blood:heart}) contient deux cavités~: une
\emph{oreillette}\index{oreillette} en haut, qui reçoit le \omterm{prop:g4:journey-of-food:absorb}{sang} qui
arrive, et un \emph{ventricule}\index{ventricule} en bas --- la
cavité de pompage à paroi épaisse --- qui le chasse. Entre elles et
au-delà, des \emph{valves}\index{valve (cœur)} --- des portes à sens
unique --- claquent derrière chaque poussée~; leurs claquements par
paires sont exactement le «~poum-tac~» qu'entend un stéthoscope (ou
une \omterm{def:g1:the-five-senses:sense}{oreille} posée sur une poitrine).
\end{definition}

\begin{proposition}[Un battement, dans l'ordre]\label{prop:g7:heart-and-circulation:beat}
Chaque battement déroule la même séquence des deux côtés à la fois~:
les \omterm{def:g7:heart-and-circulation:rooms}{oreillettes} se contractent et achèvent de remplir les
\omterm{def:g7:heart-and-circulation:rooms}{ventricules}~; les \omterm{def:g7:heart-and-circulation:rooms}{ventricules} se contractent et chassent le \omterm{prop:g4:journey-of-food:absorb}{sang} ---
le côté droit vers les \omterm{def:g7:how-animals-breathe:four}{poumons}, le côté gauche vers le corps ---
tandis que les \omterm{def:g7:heart-and-circulation:rooms}{valves} d'entrée claquent (poum)~; les \omterm{def:g7:heart-and-circulation:rooms}{ventricules} se
relâchent, les \omterm{def:g7:heart-and-circulation:rooms}{valves} de sortie claquent (tac), et les cavités se
remplissent de nouveau. Puis on recommence, $100000$ fois par jour
(\cref{rem:g4:heart-and-blood:rest}).
\end{proposition}
\admitted

\section{Deux circuits, trois vaisseaux}

\begin{definition}[La double circulation]\label{def:g7:heart-and-circulation:double}
Le réseau routier du \omterm{prop:g4:journey-of-food:absorb}{sang} est fait de deux boucles à travers une
seule pompe (les deux côtés de
\cref{def:g4:heart-and-blood:heart}, vus maintenant comme des
circuits)~:
\begin{itemize}
  \item le \emph{circuit des \omterm{def:g7:how-animals-breathe:four}{poumons}}~: \omterm{def:g7:heart-and-circulation:rooms}{ventricule} droit
        $\rightarrow$ \omterm{def:g7:how-animals-breathe:four}{poumons} (chargement du \omterm{def:g7:what-is-respiration:respiration}{dioxygène}, déchargement
        du \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone}) $\rightarrow$ \omterm{def:g7:heart-and-circulation:rooms}{oreillette} gauche~;
  \item le \emph{circuit du corps}~: \omterm{def:g7:heart-and-circulation:rooms}{ventricule} gauche
        $\rightarrow$ tout le corps (livraisons, collectes)
        $\rightarrow$ \omterm{def:g7:heart-and-circulation:rooms}{oreillette} droite.
\end{itemize}
Chaque goutte alterne les circuits~: \omterm{def:g7:how-animals-breathe:four}{poumons}, corps, \omterm{def:g7:how-animals-breathe:four}{poumons}, corps
--- et c'est pourquoi le \omterm{def:g7:heart-and-circulation:rooms}{ventricule} gauche, qui pousse autour de la
boucle du corps, bien plus grande, est la cavité du \omterm{def:g4:heart-and-blood:heart}{cœur} à la paroi
la plus épaisse.
\end{definition}

\begin{definition}[Artères, capillaires, veines]\label{def:g7:heart-and-circulation:vessels}
Trois sortes de vaisseaux composent chaque boucle
(\cref{def:g4:heart-and-blood:vessels}, maintenant nommés)~:
\begin{itemize}
  \item les \emph{artères}\index{artère}~: à paroi épaisse et
        élastique, elles emportent le \omterm{prop:g4:journey-of-food:absorb}{sang} \emph{loin} du \omterm{def:g4:heart-and-blood:heart}{cœur} sous
        pression --- le \omterm{prop:g4:heart-and-blood:pulse}{pouls}
        (\cref{prop:g4:heart-and-blood:pulse}) est leur étirement à
        chaque poussée~;
  \item les \emph{capillaires}\index{capillaire}~: les vaisseaux
        fins comme des cheveux qui se faufilent dans chaque organe
        --- des parois d'une seule \omterm{def:g5:first-look-at-cells:cell}{cellule} d'épaisseur, où se font
        \emph{tous} les échanges~: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alvéoles}, \omterm{prop:g7:digestion-and-nutrients:absorption}{villosités}, \omterm{def:g3:bones-and-muscles:muscle}{muscles},
        \omterm{def:g1:the-five-senses:sense}{peau}~;
  \item les \emph{veines}\index{veine}~: larges, à paroi plus mince,
        elles ramènent le \omterm{prop:g4:journey-of-food:absorb}{sang} \emph{vers} le \omterm{def:g4:heart-and-blood:heart}{cœur} à basse pression,
        avec leurs propres \omterm{def:g7:heart-and-circulation:rooms}{valves} à sens unique --- les lignes
        bleutées de \cref{rem:g4:heart-and-blood:see}.
\end{itemize}
\end{definition}

\begin{example}[Une goutte, une tournée complète]\label{ex:g7:heart-and-circulation:tour}
La tournée de livraison de \cref{ex:g4:heart-and-blood:round}, avec
les noms de la carte~: \omterm{def:g7:heart-and-circulation:rooms}{ventricule} droit --- \omterm{def:g7:heart-and-circulation:vessels}{capillaires} des \omterm{def:g7:how-animals-breathe:four}{poumons}
(le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} embarque aux \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alvéoles}, rouge vif) --- \omterm{def:g7:heart-and-circulation:rooms}{oreillette}
gauche, \omterm{def:g7:heart-and-circulation:rooms}{ventricule} gauche --- \omterm{def:g7:heart-and-circulation:vessels}{artère} --- \omterm{def:g7:heart-and-circulation:vessels}{capillaires} d'un \omterm{def:g3:bones-and-muscles:muscle}{muscle} de
cuisse (\omterm{def:g7:what-is-respiration:respiration}{dioxygène} et \omterm{prop:g7:organs-at-work:bill}{glucose} débarqués, \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} embarqué,
plus foncé) --- \omterm{def:g7:heart-and-circulation:vessels}{veine}, les \omterm{def:g7:heart-and-circulation:rooms}{valves} facilitant le trajet en montée ---
\omterm{def:g7:heart-and-circulation:rooms}{oreillette} droite. Deux circuits, en moins d'une minute, sans aucun
mauvais virage possible~: les \omterm{def:g7:heart-and-circulation:rooms}{valves} y veillent.
\end{example}

\begin{remark}[Pourquoi les capillaires sont l'essentiel]\label{rem:g7:heart-and-circulation:capillaries}
Les \omterm{def:g7:heart-and-circulation:vessels}{artères} et les \omterm{def:g7:heart-and-circulation:vessels}{veines} ne sont que des autoroutes~; toute
livraison réelle --- chacun des échanges étudiés cette année --- se
fait à travers les parois des \omterm{def:g7:heart-and-circulation:vessels}{capillaires}. Le cahier des charges de
\cref{prop:g7:how-animals-breathe:spec} habite ici aussi~: des
parois d'une seule \omterm{def:g5:first-look-at-cells:cell}{cellule} (minces), se faufilant dans chaque tissu
sur des dizaines de milliers de kilomètres
(\cref{def:g4:heart-and-blood:vessels} --- grandes), baignées dans
l'\omterm{def:g6:exploring-our-environment:conditions}{humidité} du corps. Le \omterm{def:g4:heart-and-blood:heart}{cœur} existe pour maintenir le \omterm{prop:g4:journey-of-food:absorb}{sang} en
mouvement \emph{à travers les \omterm{def:g7:heart-and-circulation:vessels}{capillaires}}~; tout le reste est de la
tuyauterie d'amenée et de retour.
\end{remark}

\section{Lire et entretenir le réseau}

\begin{example}[Des mesures au bord de la route]\label{ex:g7:heart-and-circulation:measures}
Trois lectures quotidiennes du réseau~: le \emph{\omterm{prop:g4:heart-and-blood:pulse}{pouls}} ---
l'étirement des \omterm{def:g7:heart-and-circulation:vessels}{artères}, qui compte les poussées
(\cref{met:g4:heart-and-blood:pulse})~; la \emph{pression
artérielle}, les deux nombres du brassard --- la pression dans les
\omterm{def:g7:heart-and-circulation:vessels}{artères} à la poussée et entre deux poussées~; et le
\emph{poum-tac} du stéthoscope --- les portes des \omterm{def:g7:heart-and-circulation:rooms}{valves} entendues
en train de se fermer. Trois fenêtres, sans aucune incision.
\end{example}

\begin{proposition}[Entretenir le réseau]\label{prop:g7:heart-and-circulation:keep}
On connaît les ennemis et les amis du réseau sur le long terme~:
\begin{itemize}
  \item l'\emph{exercice} renforce la pompe
        (\cref{prop:g7:organs-at-work:training}) et garde les
        autoroutes souples~;
  \item une \emph{alimentation équilibrée}
        (\cref{met:g3:balanced-diet:check}) protège la paroi des
        \omterm{def:g7:heart-and-circulation:vessels}{artères}, qu'un \omterm{prop:g4:journey-of-food:absorb}{sang} trop riche encrasse et raidit
        lentement~;
  \item la \emph{fumée} attaque les vaisseaux autant que les
        \omterm{def:g7:how-animals-breathe:four}{poumons} (\cref{prop:g7:lungs-and-gas-exchange:smoke}) ---
        elle rétrécit, raidit et vole des places~;
  \item l'approvisionnement du \omterm{def:g4:heart-and-blood:heart}{cœur} lui-même --- il se nourrit par
        ses propres petites \omterm{def:g7:heart-and-circulation:vessels}{artères}, et non du \omterm{prop:g4:journey-of-food:absorb}{sang} de ses cavités
        --- dépend des trois.
\end{itemize}
Les habitudes de \cref{prop:g5:healthy-choices:contract} sont,
concrètement, de l'entretien de vaisseaux.
\end{proposition}
\admitted

\begin{method}[Placer n'importe quel organe sur le réseau]\label{met:g7:heart-and-circulation:map}
Pour n'importe quel organe rencontré dans ce livre~:
\begin{enumerate}
  \item place-le sur le circuit du corps~: une \omterm{def:g7:heart-and-circulation:vessels}{artère} qui entre,
        une \omterm{def:g7:heart-and-circulation:vessels}{veine} qui sort, des \omterm{def:g7:heart-and-circulation:vessels}{capillaires} à l'intérieur~;
  \item énumère ce qui traverse ses parois \omterm{def:g7:heart-and-circulation:vessels}{capillaires}, dans les
        deux sens (\omterm{prop:g7:digestion-and-nutrients:absorption}{villosités}~: des \omterm{def:g7:digestion-and-nutrients:families}{nutriments} qui entrent~;
        \omterm{def:g3:bones-and-muscles:muscle}{muscle}~: du carburant et du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} qui entrent, du
        \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} qui sort~; \omterm{def:g1:the-five-senses:sense}{peau}~: de la chaleur qui
        sort)~;
  \item note sa liaison particulière, s'il en a une (le \omterm{prop:g4:journey-of-food:absorb}{sang} de
        l'intestin passe d'abord par le \omterm{prop:g7:digestion-and-nutrients:absorption}{foie}~; les \omterm{def:g7:how-animals-breathe:four}{poumons} sont sur
        leur propre circuit)~;
  \item confronte les flux au métier de l'organe --- le schéma du
        réseau devrait raconter de nouveau le chapitre de cet
        organe.
\end{enumerate}
\end{method}

\section{Problème~: la matinée de la cardiologue}

\begin{problem}[{Problème du week-end --- quatre consultations, une
carte du réseau}]\label{pb:g7:heart-and-circulation:1}
Suis une cardiologue pendant une matinée de quatre consultations
(imaginaires), la carte à la main.

\medskip\noindent\textbf{Partie I --- La référence en bonne
santé.}
\begin{enumerate}
  \item La patiente 1, une adolescente sportive, reçoit les trois
        lectures de bord de route de
        \cref{ex:g7:heart-and-circulation:measures}. Nomme chacune,
        et dis quelle partie de la machinerie chacune lit.
  \item Son \omterm{prop:g4:heart-and-blood:pulse}{pouls} au repos est de $58$. La cardiologue sourit.
        Pourquoi
        (\cref{exo:g4:heart-and-blood:11})~?
  \item Son poum-tac est net et double. Qu'est-ce qui a claqué,
        deux fois, à chaque battement~?
  \item Esquisse le trajet de son \omterm{prop:g4:journey-of-food:absorb}{sang} depuis l'\omterm{def:g7:heart-and-circulation:rooms}{oreillette} droite
        autour des deux circuits et retour, en huit étapes nommées.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Trois ennuis.}
\begin{enumerate}[resume]
  \item Le patient 2 a une valve qui fuit entre l'\omterm{def:g7:heart-and-circulation:rooms}{oreillette} et le
        \omterm{def:g7:heart-and-circulation:rooms}{ventricule} gauches~: une partie du \omterm{prop:g4:journey-of-food:absorb}{sang} reflue à chaque
        battement, avec un souffle. Prédis les conséquences pour les
        livraisons du circuit du corps et pour l'endurance du
        patient.
  \item Le patient 3, gros fumeur au régime riche, a des \omterm{def:g7:heart-and-circulation:vessels}{artères}
        encrassées et raidies. Quelles lectures de la consultation 1
        se seront dégradées, et pourquoi le \omterm{def:g7:heart-and-circulation:rooms}{ventricule} gauche
        fait-il des heures supplémentaires~?
  \item L'encrassement du patient 3 touche aussi les \omterm{def:g7:heart-and-circulation:vessels}{artères}
        nourricières du \omterm{def:g4:heart-and-blood:heart}{cœur}. Explique en une phrase l'inquiétude
        particulière de la cardiologue.
  \item La patiente 4 a les \omterm{def:g1:my-body:joint}{chevilles} gonflées après de longues
        journées debout~: le retour en montée des \omterm{def:g7:heart-and-circulation:vessels}{veines} peine.
        Quelles structures sont en défaut, et pourquoi la marche ---
        les \omterm{def:g3:bones-and-muscles:muscle}{muscles} du mollet qui pressent les \omterm{def:g7:heart-and-circulation:vessels}{veines} --- la
        soulage-t-elle~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- L'affiche murale.}
\begin{enumerate}[resume]
  \item L'affiche du cabinet montre la double boucle. Explique à un
        patient de la salle d'attente pourquoi le \omterm{prop:g4:journey-of-food:absorb}{sang} doit passer
        par les \omterm{def:g7:how-animals-breathe:four}{poumons} entre deux tours du corps.
  \item Ajoute la légende des vaisseaux~: \omterm{def:g7:heart-and-circulation:vessels}{artère}, \omterm{def:g7:heart-and-circulation:vessels}{capillaire},
