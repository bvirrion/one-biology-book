\section*{Chapter \ref{ch:g7:eco-energy} --- Energy in Ecosystems}

\begin{solution}{exo:g7:eco-energy:1}
\omterm{def:g7:eco-energy:ecosystem}{Ecosystem}: \omterm{def:g6:ecology:levels}{community} plus non-living \omterm{def:g4:traits:env}{environment}. Living: trees, deer
(examples). Non-living: \omterm{def:g4:plants-need:needs}{light}, soil \omterm{def:g4:plants-need:needs}{water}/\omterm{def:g6:nutrition:groups}{minerals}, temperature, \omterm{def:g4:plants-need:needs}{air}.
\end{solution}

\begin{solution}{exo:g7:eco-energy:2}
\omterm{def:g7:eco-energy:roles}{Producer}: makes \omterm{def:g2:food-energy:food}{food} (grass). \omterm{def:g7:eco-energy:roles}{Consumer}: eats others (rabbit). \omterm{def:g7:eco-energy:roles}{Decomposer}:
breaks down dead material (fungus/bacterium).
\end{solution}

\begin{solution}{exo:g7:eco-energy:3}
\omterm{def:g6:reproduction:plant}{Seeds} $\to$ mouse $\to$ snake $\to$ hawk.
\end{solution}

\begin{solution}{exo:g7:eco-energy:4}
Feeding step. Grass: level 1; rabbit: level 2; fox: level 3.
\end{solution}

\begin{solution}{exo:g7:eco-energy:5}
Large \omterm{def:g6:nutrition:def}{energy} losses at each transfer (heat, waste, incomplete eating) leave
little \omterm{def:g6:nutrition:def}{energy} for high levels.
\end{solution}
