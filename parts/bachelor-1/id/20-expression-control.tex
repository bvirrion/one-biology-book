\chapter{Kendali Pengungkapan Gen}\label{ch:b1:expression-control}

Sebuah bakteri yang tumbuh pada glukosa membawa lima molekul \omterm{def:b1:enzymes:enzyme}{enzim} yang
mencerna laktosa; berilah ia laktosa dan, dalam beberapa menit, ia
membawa lima ribu. Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} hati dan sebuah \omterm{def:b1:body-plans-tissues:nervous}{neuron} membawa dua puluh
ribu \omterm{def:b1:genomes:gene}{gen} yang sama lalu membuat perangkat \omterm{def:b1:proteins:peptide}{protein} yang hampir seluruhnya
berbeda, dan tak satu pun akan pernah membuat perangkat yang lain. Tak
ada dalam \cref{ch:b1:gene-expression} yang menerangkan hal ini: mesin
pengungkapannya sama bagi tiap \omterm{def:b1:genomes:gene}{gen}. Yang berbeda adalah apakah sebuah
\omterm{def:b1:genomes:gene}{gen} dibaca, seberapa sering, dan berapa lama --- keputusan yang diambil
pada promotornya oleh \omterm{def:b1:proteins:peptide}{protein} yang mengikat \omterm{def:b1:nucleic-acids:chain}{DNA}, dan sesudahnya oleh
nasib pembawa pesan serta \omterm{def:b1:proteins:peptide}{proteinnya}. Bab ini memerikan bagaimana
bakteri menyalakan dan memadamkan \omterm{def:b1:genomes:gene}{gennya} sebagai tanggapan atas
makanannya, bagaimana eukariot mengendalikan \omterm{def:b1:genomes:gene}{gennya} lewat kromatin,
\omterm{def:b1:expression-control:enhancer}{penguat} yang jauh dan gabungan faktor, serta bagaimana kendali yang
sama, yang diterapkan pada \omterm{def:b1:genomes:gene}{gen} yang berbeda dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} yang berbeda,
membuat sebuah tubuh dari satu \omterm{def:b1:genomes:genome}{genom}.

\section{Mengapa dan di mana gen dikendalikan}

\begin{proposition}[Pengungkapannya diatur, pada tiap tingkat]\label{prop:b1:expression-control:levels}
\index{pengaturan gen}\index{gen konstitutif}
Sebagian \omterm{def:b1:genomes:gene}{gen} bersifat \emph{konstitutif} --- selalu diungkapkan, pada
aras yang mantap, sebab hasilnya selalu dibutuhkan (\omterm{def:b1:proteins:peptide}{protein} \omterm{def:b1:gene-expression:ribosome}{ribosomnya},
\omterm{def:b1:enzymes:enzyme}{enzim} \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisisnya}). Sebagian besarnya \emph{diatur}: diungkapkan
hanya pada sebagian keadaan, sebagian \omterm{def:b1:cell-unit-of-life:cell}{sel}, sebagian saat. Kendalinya
bekerja pada tiap langkah aliran itu --- terutama pada awal
transkripsinya, tempat termurah untuk memutuskan, tetapi juga pada
pemrosesan, ekspor, kemantapan dan translasi pembawa pesannya, serta
pada perubahan dan penguraian \omterm{def:b1:proteins:peptide}{proteinnya} --- dan waktu tanggapnya
berkisar dari beberapa sekon (sebuah \omterm{def:b1:proteins:peptide}{protein} yang diubah) lewat
beberapa menit (sebuah \omterm{def:b1:genomes:gene}{gen} bakteri yang dinyalakan) sampai beberapa
hari (sebuah keadaan kromatin yang diubah).
\end{proposition}

\begin{example}[Membuat enzim hanya ketika dibutuhkan]\label{ex:b1:expression-control:economy}
\omterm{def:b1:enzymes:enzyme}{Enzim} $\beta$-galaktosidase, yang membelah laktosa, adalah sebuah
tetramer berisi empat rantai \qty{1024}{} residu: lima ribu salinannya
adalah \qty{3}{\%} \omterm{def:b1:proteins:peptide}{protein} sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} dan memakan biaya \qty{8e7}{} \omterm{def:b1:water-small-molecules:atp}{ATP}
tiap generasi. Sebuah bakteri yang membuatnya tanpa laktosa untuk
dicerna akan tumbuh sekitar satu persen lebih lambat daripada seorang
pesaing yang tidak membuatnya, dan dalam beberapa ratus generasi akan
kalah jumlah. Pengaturannya terpilih karena pengungkapannya mahal.
\end{example}

\section{Operon lac}

\begin{definition}[Operon, penekan, operator]\label{def:b1:expression-control:operon}
\index{operon}\index{penekan}\index{operator}\index{operon lac}
Yang disebut \emph{operon} adalah sekelompok \omterm{def:b1:genomes:gene}{gen} bakteri yang
ditranskripsikan bersama dari satu promotor menjadi satu pembawa pesan,
sehingga dikendalikan bersama. \emph{Operon lac} \emph{E. coli}
terdiri atas tiga \omterm{def:b1:genomes:gene}{gen} bagi pemakaian laktosa --- \emph{lacZ}
($\beta$-galaktosidase), \emph{lacY} (permease laktosa yang
mengimpornya) dan \emph{lacA} --- di belakang sebuah promotor dan
sebuah \emph{operator}, yaitu urutan pendek yang bertindih dengan
promotornya. Sebuah \omterm{def:b1:genomes:gene}{gen} terpisah, \emph{lacI}, membuat \emph{penekan
lac}, yaitu \omterm{def:b1:proteins:peptide}{protein} yang mengikat operatornya lalu menghalangi
polimerasenya. Laktosa, yang diubah dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya} menjadi alolaktosa,
adalah \emph{penginduksinya}: ia mengikat penekannya, mengubah
bentuknya, lalu membuatnya melepaskan operatornya. Karena itu operonnya
padam kecuali laktosanya hadir --- \emph{kendali negatif} yang diangkat
lewat \emph{induksi}.
\end{definition}

\begin{proposition}[Model operonnya]\label{prop:b1:expression-control:jacobmonod}
\omterm{def:b1:genomes:gene}{Gen} lac dikendalikan sebuah \omterm{def:b1:expression-control:operon}{penekan} yang dapat berdifusi yang bekerja
pada sebuah sisi di sebelah \omterm{def:b1:genomes:gene}{gennya}: kendalinya negatif, \omterm{def:b1:expression-control:operon}{penekannya}
bekerja secara \emph{trans} (dari mana pun dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya}), dan
\omterm{def:b1:expression-control:operon}{operatornya} bekerja secara \emph{cis} (hanya pada \omterm{def:b1:genomes:gene}{gen} di sebelahnya).
\end{proposition}
\begin{proof}[Bukti eksperimental]
Jacob dan Monod (1959--1961) menalar dari mutannya. Galur yang mutan
pada \emph{lacI} ($I^-$) membuat \omterm{def:b1:enzymes:enzyme}{enzimnya} dengan atau tanpa laktosa
(\emph{konstitutif}); sebuah \omterm{def:b1:genomes:gene}{gen} \emph{lacI} yang normal yang
dimasukkan pada salinan kedua kawasan itu memulihkan pengaturannya,
sehingga hasil \omterm{def:b1:genomes:gene}{gennya} adalah sebuah \omterm{def:b1:expression-control:operon}{penekan} yang dapat berdifusi yang
tidak dimiliki mutannya. Galur yang mutan pada \omterm{def:b1:expression-control:operon}{operatornya} ($O^c$) juga
konstitutif, tetapi sebuah \omterm{def:b1:expression-control:operon}{operator} normal kedua pada salinan yang lain
tidak memulihkan pengaturannya --- \omterm{def:b1:expression-control:operon}{operatornya} hanya mengendalikan \omterm{def:b1:genomes:gene}{gen}
yang terpaut padanya secara fisik, sehingga ia sebuah sisi, bukan
sebuah hasil. Golongan ketiga ($I^s$) sama sekali tak dapat diinduksi,
dan bersifat dominan: sebuah \omterm{def:b1:expression-control:operon}{penekan} yang tak lagi mengikat
penginduksinya. Dalam percobaan PaJaMo (1959) sebuah \omterm{def:b1:genomes:gene}{gen} \emph{lacI}
normal yang dipindahkan lewat konjugasi ke dalam sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} $I^-$
memadamkan sintesis \omterm{def:b1:enzymes:enzyme}{enzimnya} dalam beberapa menit: \omterm{def:b1:expression-control:operon}{penekannya} bekerja
cepat, dan pada transkripsinya --- pembawa pesannya kemudian terbukti
berwaktu paruh beberapa menit.
\end{proof}

\begin{definition}[Kendali positif: CAP dan cAMP]\label{def:b1:expression-control:cap}
\index{penekanan katabolit}\index{diauksi}
Promotor lac itu lemah: bahkan tanpa \omterm{def:b1:expression-control:operon}{penekannya}, polimerasenya jarang
mulai kecuali sebuah \omterm{def:b1:proteins:peptide}{protein} kedua, \emph{CAP} (\omterm{def:b1:proteins:peptide}{protein} pengaktif
katabolit), terikat tepat di hulunya, yang membengkokkan \omterm{def:b1:nucleic-acids:chain}{DNA-nya} lalu
memegang polimerasenya di tempat. CAP mengikat \omterm{def:b1:nucleic-acids:chain}{DNA} hanya ketika membawa
\emph{AMP siklik}, yang arasnya dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya} turun ketika glukosanya
berlimpah. Karena itu \omterm{def:b1:expression-control:operon}{operonnya} sepenuhnya nyala hanya ketika laktosanya
hadir \emph{dan} glukosanya tidak ada: dua isyarat, satu negatif dan
satu positif, dipadukan pada satu promotor. Bila diberi kedua gulanya,
\emph{E. coli} memakan glukosanya lebih dahulu, berhenti sejenak
sementara cAMP-nya naik dan \omterm{def:b1:enzymes:enzyme}{enzim} lac-nya dibuat, lalu memakan
laktosanya --- yaitu kurva pertumbuhan dua tahap yang disebut
\emph{diauksi}, yang diperikan Monod pada 1941 dan yang membawanya ke
jalan menuju \omterm{def:b1:expression-control:operon}{operonnya}.
\end{definition}

\begin{method}[Meramalkan sebuah fenotipe lac]\label{met:b1:expression-control:lacphenotype}
\begin{enumerate}
  \item Tuliskan genotipe tiap salinan kawasannya: $I$, $P$, $O$, $Z$
        ($^+$ normal, $^-$ tak aktif, $O^c$ \omterm{def:b1:expression-control:operon}{operator} konstitutif, $I^s$
        \omterm{def:b1:expression-control:operon}{penekan} super).
  \item \omterm{def:b1:expression-control:operon}{Penekannya} bekerja secara trans: satu $I^+$ di mana pun menekan
        tiap \omterm{def:b1:expression-control:operon}{operator} normal; $I^s$ menekan apa pun penginduksinya.
  \item \omterm{def:b1:expression-control:operon}{Operator} dan promotornya bekerja secara cis: sebuah $O^c$ hanya
        membebaskan $Z$ pada \omterm{def:b1:nucleic-acids:chain}{DNA-nya} sendiri; sebuah $P^-$ hanya
        membungkam \omterm{def:b1:genomes:gene}{gennya} sendiri.
  \item Tanyakan, dengan dan tanpa penginduksinya, apakah tiap \omterm{def:b1:genomes:gene}{gen}
        $Z^+$-nya ditranskripsikan; fenotipenya adalah dapat diinduksi,
        konstitutif atau tak dapat diinduksi sesuai dengan itu. Lalu
        tambahkan glukosa: tanpa cAMP--CAP, transkripsinya rendah apa
        pun yang diperbuat \omterm{def:b1:expression-control:operon}{penekannya}.
\end{enumerate}
\end{method}

\begin{example}[Dua diploid parsial]\label{ex:b1:expression-control:diploids}
$I^-\,O^+\,Z^+ / I^+\,O^+\,Z^-$: salinan $I^+$-nya membuat \omterm{def:b1:expression-control:operon}{penekan} yang
mengikat $O^+$ di sebelah $Z^+$-nya; dapat diinduksi --- mutasinya
dilengkapi. $I^+\,O^c\,Z^+ / I^+\,O^+\,Z^-$: satu-satunya $Z$ yang
bekerja duduk di belakang sebuah \omterm{def:b1:expression-control:operon}{operator} yang tak dapat diikat
\omterm{def:b1:expression-control:operon}{penekannya}; konstitutif --- \omterm{def:b1:expression-control:operon}{operator} normal pada salinan yang lain tak
dapat menolong, sebab ia hanya mengendalikan $Z$ yang rusak.
\end{example}

\section{Siasat bakteri yang lain}

\begin{proposition}[Penekanan, atenuasi dan faktor sigma]\label{prop:b1:expression-control:trp}
\index{operon trp}\index{atenuasi}\index{faktor sigma}
\omterm{def:b1:expression-control:operon}{Operon} bagi biosintesis dikendalikan dalam arti yang berlawanan:
\emph{\omterm{prop:b1:expression-control:trp}{operon trp}}, yaitu lima \omterm{def:b1:genomes:gene}{gen} bagi pembuatan triptofan,
ditranskripsikan kecuali triptofannya hadir --- \omterm{def:b1:proteins:aminoacid}{asam aminonya} mengikat
\omterm{def:b1:expression-control:operon}{penekan} trp lalu \emph{memungkinkannya} mengikat \omterm{def:b1:expression-control:operon}{operatornya} (sebuah
korepresor). Kendali kedua yang lebih halus, yaitu \emph{atenuasi},
memakai penggandengan transkripsi dan translasi pada bakteri: awal
pembawa pesannya menyandi sebuah peptida pendek dengan dua triptofan
berderet; bila tRNA bermuatan triptofannya berlimpah, sebuah \omterm{def:b1:gene-expression:ribosome}{ribosom}
mentranslasikannya dengan cepat dan \omterm{def:b1:nucleic-acids:chain}{RNA} di belakangnya melipat menjadi
sebuah jepit rambut pengakhir yang menghentikan polimerasenya sebelum
\omterm{def:b1:genomes:gene}{gennya}; bila triptofannya langka, \omterm{def:b1:gene-expression:ribosome}{ribosomnya} terhenti pada \omterm{def:b1:gene-expression:code}{kodon}
triptofannya dan \omterm{def:b1:nucleic-acids:chain}{RNA-nya} melipat secara lain, sehingga membiarkan
transkripsinya berlanjut. Bakteri juga menyalakan seluruh perangkat \omterm{def:b1:genomes:gene}{gen}
dengan mengganti subunit $\sigma$ polimerasenya: sebuah $\sigma$ kejut
panas mengarahkannya ke promotor \omterm{def:b1:genomes:gene}{gen} \omterm{prop:b1:proteins:denaturation}{pendamping proteinnya}; sebuah
$\sigma$ sporulasi ke promotor pembentukan sporanya.
\end{proposition}

\section{Kendali pada eukariot}

\begin{definition}[Faktor transkripsi, penguat]\label{def:b1:expression-control:enhancer}
\index{faktor transkripsi}\index{penguat}
Pada eukariot polimerasenya tak pernah mulai sendiri: faktor umum pada
promotornya (\cref{ch:b1:gene-expression}) merekrutnya, dan lajunya
ditetapkan \emph{faktor transkripsi} yang terikat pada \emph{urutan
pengatur} --- sebagian dekat promotornya, banyak dalam \emph{penguat}
yang dapat terletak ribuan \omterm{thm:b1:nucleic-acids:helix}{pasangan basa} jauhnya, di hulu, di hilir
atau di dalam sebuah \omterm{def:b1:genomes:gene}{intron}, pada arah hadap mana pun, dan bekerja
lewat \emph{penggelungan} \omterm{def:b1:nucleic-acids:chain}{DNA-nya} sehingga faktor yang terikat padanya
menyentuh kompleks promotornya lewat \omterm{def:b1:proteins:peptide}{protein} perantara. Sebuah faktor
transkripsi bersifat \omterm{prop:b1:flowering-plant-organization:modular}{bermodul}: sebuah \emph{ranah pengikat \omterm{def:b1:nucleic-acids:chain}{DNA}} yang
mengenali sebuah urutan pendek (keluarga heliks-belok-heliks, jari
seng, ritsleting leusina) dan sebuah ranah \emph{pengaktif} atau
\emph{\omterm{def:b1:expression-control:operon}{penekan}} yang merekrut atau menghalangi mesinnya. Sebuah \omterm{def:b1:genomes:gene}{gen}
dibaca ketika \emph{gabungan} faktor yang tepat hadir: beberapa ratus
faktor, dalam gabungan, mengendalikan dua puluh ribu \omterm{def:b1:genomes:gene}{gen}.
\end{definition}

\begin{proposition}[Kromatin adalah bagian kendalinya]\label{prop:b1:expression-control:chromatin}
\index{asetilasi histon}\index{metilasi DNA}
Sebuah \omterm{def:b1:genomes:gene}{gen} yang terbungkus \omterm{def:b1:genomes:nucleosome}{nukleosom} yang terkemas menjadi
heterokromatin tak dapat dibaca: faktornya tak sanggup mencapainya.
Pengaktif merekrut \omterm{def:b1:enzymes:enzyme}{enzim} yang \emph{mengasetilasi} \omterm{def:b1:genomes:nucleosome}{histonnya}, sehingga
melonggarkan cengkeramannya pada \omterm{def:b1:nucleic-acids:chain}{DNA-nya} dan menandai kawasan itu
sebagai terbuka; \omterm{def:b1:expression-control:operon}{penekan} merekrut \omterm{def:b1:enzymes:enzyme}{enzim} yang menyingkirkan gugus
asetilnya lalu menambahkan tanda yang memampatkan kromatinnya. Gugus
metil yang ditambahkan pada sitosina sebuah promotor membungkamnya
secara awet, dan tandanya disalin pada replikasinya, sehingga pola \omterm{def:b1:genomes:gene}{gen}
terbuka dan tertutup sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} diwarisi anaknya --- yaitu ingatan yang
membuat anak sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} hati tetap \omterm{def:b1:cell-unit-of-life:cell}{sel} hati (jilid Tahun ke-3 membahas
mekanisme \emph{epigenetik} ini secara mendalam).
\end{proposition}
\begin{proof}[Bukti eksperimental]
Pada \omterm{def:b1:genomes:genome}{kromosom} politen raksasa larva lalat (seribu salinan tiap
\omterm{def:b1:genomes:genome}{kromosom} berdampingan, berpita), \omterm{def:b1:genomes:gene}{gen} yang sedang ditranskripsikan
muncul sebagai \emph{puf} tempat kromatinnya membuka; hormon ekdison,
yang memicu pergantian kulit, membuat seperangkat puf tertentu muncul
dalam beberapa menit dan yang lain beberapa jam kemudian, dalam urutan
yang tetap --- transkripsi yang terlihat langsung, dan yang disakelar
sebuah isyarat. \omterm{def:b1:genomes:gene}{Gen} yang dipindahkan lewat penataan ulang \omterm{def:b1:genomes:genome}{kromosom} ke
sebelah heterokromatin dibungkam pada sebagian \omterm{def:b1:cell-unit-of-life:cell}{sel} dan tidak pada yang
lain, dan keadaannya diwarisi anaknya.
\end{proof}

\begin{example}[Sebuah isyarat menjadi sebuah pola pengungkapan]\label{ex:b1:expression-control:steroid}
Kortisol masuk ke sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} hati lalu mengikat reseptornya, yaitu
sebuah \omterm{def:b1:expression-control:enhancer}{faktor transkripsi} yang dijaga tak aktif dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya};
kompleksnya masuk ke \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}, mengikat sebuah urutan lima belas basa
yang hadir dekat seratus \omterm{def:b1:genomes:gene}{gen} --- \omterm{def:b1:genomes:gene}{gen} \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesisnya} termasuk
(\cref{ch:b1:biosyntheses-integration}) --- lalu merekrut asetilase dan
perantaranya: dalam sejam \omterm{def:b1:enzymes:enzyme}{enzim} sintesis glukosanya sedang dibuat.
Hormon yang sama dalam sebuah limfosit, yang kromatin terbukanya
memaparkan perangkat urutan itu yang berbeda, menyalakan \omterm{def:b1:genomes:gene}{gen} yang
membunuh \omterm{def:b1:cell-unit-of-life:cell}{selnya}. Satu isyarat, satu reseptor, dua tanggapan:
perbedaannya adalah \omterm{def:b1:genomes:gene}{gen} mana yang terjangkau.
\end{example}

\section{Sesudah transkripsi, dan pembuatan sebuah tubuh}

\begin{proposition}[Kendali di luar promotornya]\label{prop:b1:expression-control:posttranscriptional}
\index{kemantapan mRNA}
Sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{sel eukariot} juga memutuskan bagaimana sebuah transkrip
disambung (penyambungan berselang-selang memberi sebuah otot dan sebuah
otak \omterm{def:b1:proteins:peptide}{protein} yang berbeda dari satu \omterm{def:b1:genomes:gene}{gen}), apakah ia diekspor, berapa
lama ia bertahan (waktu paruhnya dari beberapa menit bagi pesan \omterm{def:b1:proteins:peptide}{protein}
pengatur sampai beberapa hari bagi pesan hemoglobin, yang ditetapkan
urutan pada kawasan tak tertranslasinya yang mengikat \omterm{def:b1:proteins:peptide}{protein} dan \omterm{def:b1:nucleic-acids:chain}{RNA}
pengatur yang kecil), dan apakah ia ditranslasikan (\omterm{def:b1:cell-unit-of-life:cell}{sel} yang kelaparan
besi menghalangi translasi pesan feritinnya lewat sebuah \omterm{def:b1:proteins:peptide}{protein} yang
terikat pada ujung $5'$-nya, lalu melepasnya ketika besinya hadir).
\omterm{def:b1:proteins:peptide}{Proteinnya}, begitu dibuat, dikendalikan lewat perubahan dan
penguraiannya (\cref{ch:b1:enzymes}). Tiap tingkat yang belakangan lebih
cepat dan lebih setempat daripada yang mendahuluinya, dan lebih mahal:
transkripsinya memutuskan apa yang dapat dilakukan \omterm{def:b1:cell-unit-of-life:cell}{selnya}, tingkat yang
belakangan memutuskan apa yang dilakukannya sekarang.
\end{proposition}

\begin{proposition}[Pengungkapan yang berbeda membuat sebuah tubuh]\label{prop:b1:expression-control:differentiation}
\index{diferensiasi}
Tiap \omterm{def:b1:cell-unit-of-life:cell}{sel} sebuah \omterm{def:b1:organism-environment:organism}{organisme} bersel banyak membawa \omterm{def:b1:genomes:genome}{genom} yang sama;
\omterm{def:b1:cell-unit-of-life:cell}{selnya} berbeda karena mengungkapkan bagian yang berbeda darinya. \omterm{def:b1:genomes:gene}{Gen}
\emph{rumah tangga} --- beberapa ribu bagi metabolisme, \omterm{def:b1:gene-expression:ribosome}{ribosom}, \omterm{def:b1:eukaryotic-cell:cytoskeleton}{rangka
selnya} --- nyala pada tiap \omterm{def:b1:cell-unit-of-life:cell}{sel}; selebihnya dinyalakan pada sebagian \omterm{def:b1:cell-unit-of-life:cell}{sel}
dan dipadamkan pada yang lain oleh gabungan \omterm{def:b1:expression-control:enhancer}{faktor transkripsi} yang
dibawa tiap \omterm{def:b1:cell-unit-of-life:cell}{selnya} dan keadaan kromatin yang diwarisinya. Prazat sebuah
\omterm{def:b1:cell-unit-of-life:cell}{sel} merah menyalakan globinnya lalu memadamkan hampir semua yang lain;
sebuah \omterm{def:b1:body-plans-tissues:nervous}{neuron} menyalakan salurannya lalu tak pernah membelah lagi.
\emph{Diferensiasi} adalah perolehan sebuah pola pengungkapan yang
mantap, dan \emph{perkembangan} (jilid Tahun ke-2) adalah proses yang
lewatnya isyarat antarsel menetapkan pola itu di tempat yang tepat.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} yang diambil dari sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} katak yang terdiferensiasi
lalu ditaruh dalam sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} telur yang \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya} sendiri telah
disingkirkan dapat mengarahkan perkembangan seekor berudu utuh
(Gurdon, 1962; diingat kembali dari jilid Sekolah Menengah): \omterm{def:b1:cell-unit-of-life:cell}{sel} yang
terdiferensiasi itu tidak kehilangan \omterm{def:b1:genomes:gene}{gen}, hanya memadamkannya, dan
\omterm{def:b1:cell-unit-of-life:cell}{sitoplasma} \omterm{def:b1:cell-unit-of-life:cell}{sel} telurnya menyetel ulang sakelarnya. \omterm{def:b1:proteins:peptide}{Protein} sebuah jenis
\omterm{def:b1:cell-unit-of-life:cell}{sel}, yang dipisahkan pada sebuah gel, berbeda dari \omterm{def:b1:proteins:peptide}{protein} jenis yang
lain; pembawa pesannya, begitu diurutkan, adalah bagian \omterm{def:b1:genomes:genome}{genomnya} yang
khas baginya; dan segelintir \omterm{def:b1:expression-control:enhancer}{faktor transkripsi} yang dimasukkan ke
dalam sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} kulit dapat memrogram ulangnya menjadi sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}
punca atau sebuah \omterm{def:b1:body-plans-tissues:nervous}{neuron}.
\end{proof}

\begin{example}[Sebuah gen yang dibaca pada dua organ]\label{ex:b1:expression-control:twoorgans}
\omterm{def:b1:genomes:gene}{Gen} bagi \omterm{def:b1:enzymes:enzyme}{enzim} yang membuat glukosa dari glukosa-6-fosfat diungkapkan
di hati dan di ginjal dan tidak pada \omterm{def:b1:body-plans-tissues:tissue}{jaringan} lain: promotornya membawa
sisi bagi faktor yang hanya hadir di sana, dan pada \omterm{def:b1:cell-unit-of-life:cell}{sel} yang lain
promotornya termetilasi dan kromatinnya tertutup. Sebuah otot, yang
membawa \omterm{def:b1:genomes:gene}{gennya} secara utuh, tak sanggup melepas glukosa ke dalam
darahnya (\cref{ch:b1:biosyntheses-integration}) --- bukan karena
kekurangan \omterm{def:b1:genomes:gene}{gennya} melainkan karena \omterm{def:b1:genomes:gene}{gennya} tidak dibaca. \omterm{def:b1:genomes:genome}{Genomnya} sama;
pengungkapannya adalah \omterm{def:b1:mammal-organization:organ}{organnya}.
\end{example}

\section{Soal: Diauksi}

\begin{problem}[{Soal akhir pekan --- kurva pertumbuhan dua tahap Monod
yang disusun ulang sel demi sel: glukosanya yang habis, laktosanya yang
diinduksi, enzimnya yang dihitung, mutannya yang diramalkan dan biaya
pengaturannya yang diperhitungkan, berakhir pada faktor induksi
$\beta$-galaktosidase}]
\label{pb:b1:expression-control:1}
Sebuah biakan \emph{E. coli} mulai pada $10^7$ \omterm{def:b1:cell-unit-of-life:cell}{sel} tiap mililiter dalam
sebuah medium berisi \qty{0.5}{mmol/L} glukosa dan \qty{1.0}{mmol/L}
laktosa. Pada glukosa \omterm{def:b1:cell-unit-of-life:cell}{selnya} berlipat dua tiap \qty{20}{min}, pada
laktosa tiap \qty{30}{min}; sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} memerlukan \qty{1.5e-12}{g} gula
tiap pembelahan (\qty{180}{g/mol} bagi glukosa; laktosa,
\qty{342}{g/mol}, dihitung sebagai dua glukosa). \omterm{def:b1:cell-unit-of-life:cell}{Sel} yang belum
terinduksi memegang 5 molekul $\beta$-galaktosidase; yang terinduksi
penuh, \num{5000}. Induksinya memakan \qty{30}{min}. \omterm{def:b1:enzymes:enzyme}{Enzimnya} adalah
tetramer berisi $4\times 1024$ residu; tiap residu memakan biaya 4 \omterm{def:b1:water-small-molecules:atp}{ATP};
anggaran \omterm{def:b1:water-small-molecules:atp}{ATP} sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} adalah $10^{10}$ tiap pembelahan dan \omterm{def:b1:proteins:peptide}{proteinnya}
$\qty{1.5e-13}{g}$ (\qty{110}{Da} tiap residu).

\medskip
\noindent\textbf{Bagian I --- Tahap glukosanya.}
\begin{enumerate}
  \item Hitunglah glukosa yang tersedia tiap mililiter, dalam gram.
  \item Berapa banyak pembelahan tiap mililiter yang dapat
        ditopangnya?
  \item Mulai dari $10^7$ \omterm{def:b1:cell-unit-of-life:cell}{sel}, berapa banyak \omterm{def:b1:cell-unit-of-life:cell}{sel} yang ada ketika
        glukosanya habis? (Sebuah pembelahan membuat satu \omterm{def:b1:cell-unit-of-life:cell}{sel} baru.)
  \item Berapa kali lipat dua itu, dan berapa lama tahap glukosanya
        berlangsung?
  \item Selama tahap ini, bagaimana keadaan \omterm{def:b1:expression-control:operon}{operon lac-nya}, dan
        mengapa, dalam istilah molekul (\omterm{def:b1:expression-control:operon}{penekan} dan CAP)?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Sakelarnya.}
\begin{enumerate}[resume]
  \item Ketika glukosanya habis, apa yang naik di dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya} dan apa
        yang diikatnya?
  \item Laktosanya hadir sejak semula. Mengapa \omterm{def:b1:expression-control:operon}{operonnya} tetap saja
        nyaris padam selama tahap glukosanya?
  \item Selama jeda \qty{30}{min} itu tiap \omterm{def:b1:cell-unit-of-life:cell}{sel} membuat \num{5000}
        \omterm{def:b1:enzymes:enzyme}{enzimnya}. Hitunglah molekul \omterm{def:b1:enzymes:enzyme}{enzim} yang dibuat tiap sekon tiap
        \omterm{def:b1:cell-unit-of-life:cell}{selnya}, dan residu tiap sekonnya.
  \item Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} memegang \num{20000} \omterm{def:b1:gene-expression:ribosome}{ribosom} pada \qty{20}{} residu
        tiap sekon. Berapa fraksi translasinya yang dikerahkan bagi
        $\beta$-galaktosidase selama jedanya?
  \item Hitunglah massa \num{5000} tetramernya dan fraksi \omterm{def:b1:proteins:peptide}{protein}
        \omterm{def:b1:cell-unit-of-life:cell}{selnya} yang diwakilinya.
  \item Hitunglah faktor induksinya.
  \item Mengapa permeasenya (\emph{lacY}) sama perlunya dengan \omterm{def:b1:enzymes:enzyme}{enzimnya}
        bagi pertumbuhan pada laktosa, dan apa yang terjadi pada sebuah
        mutan \emph{lacY}$^-$ yang diberi laktosa?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Tahap laktosanya.}
\begin{enumerate}[resume]
  \item Hitunglah laktosa yang tersedia tiap mililiter dalam setara
        glukosa, dalam gram.
  \item Berapa banyak pembelahan lanjutan yang ditopangnya, dan berapa
        cacah \omterm{def:b1:cell-unit-of-life:cell}{sel} akhirnya?
  \item Berapa lama tahap laktosanya berlangsung?
  \item Susunlah garis waktunya: tahap glukosa, jeda, tahap laktosa,
        seluruhnya.
  \item Jelaskan mengapa waktu lipat duanya lebih lama pada laktosa
        (dua alasan: satu tentang langkah \omterm{def:b1:enzymes:enzyme}{enzimnya}, satu tentang
        permeasenya).
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Mutan dan biayanya.}
\begin{enumerate}[resume]
  \item Sebuah mutan $I^-$: perikan kurva pertumbuhannya pada medium
        yang sama dan aras \omterm{def:b1:enzymes:enzyme}{enzimnya} sepanjang itu.
  \item Sebuah mutan CAP$^-$: perikan kurvanya dan aras \omterm{def:b1:enzymes:enzyme}{enzimnya}.
  \item Sebuah mutan $O^c$ dalam medium berlaktosa saja: adakah
        perbedaannya dari tipe liarnya? Dan dengan glukosa saja?
  \item Hitunglah biaya \omterm{def:b1:water-small-molecules:atp}{ATP} tiap generasi untuk membuat \num{5000}
        tetramernya, dan fraksinya dari anggaran \omterm{def:b1:cell-unit-of-life:cell}{selnya}.
  \item Bila fraksi itu memperpanjang waktu generasinya sebesar fraksi
        yang sama, seberapa besar laju pertumbuhan mutan $I^-$-nya
        tertinggal dari tipe liarnya pada glukosa?
  \item Mulai dari jumlah yang sama, sesudah berapa generasi pada
        glukosa mutannya menjadi sepersepuluh populasinya? ($r^n =
        0.1/0.9$ dengan $r$ rasio laju pertumbuhannya tiap generasi,
        yakni $2^{-\delta}$ bagi ketertinggalan $\delta$ dalam lipat
        dua.)
  \item Jelaskan mengapa, meskipun demikian, mutan $I^-$ lazim pada
