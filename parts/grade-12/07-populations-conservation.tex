\chapter{Populations and Conservation}\label{ch:g12:populations-conservation}

A single \omterm{def:g10:scientific-biology:organism}{organism} has a lifespan; a \omterm{def:g12:populations-conservation:pop}{population} has a trajectory. Counts
rise when births and immigration exceed deaths and emigration, then
level off when resources or enemies intervene. This chapter models
\omterm{def:g12:populations-conservation:pop}{population} \omterm{def:g10:scientific-biology:properties}{growth}, sketches \omterm{def:g12:populations-conservation:community}{community} interactions, faces the modern
\omterm{def:g12:populations-conservation:biodiversity}{biodiversity} crisis, and surveys strategies that aim to keep \omterm{def:g12:phylogeny-macroevolution:species}{species} and
\omterm{def:g10:ecosystems:levels}{ecosystems} from slipping away.

\section{Populations and growth models}

\begin{definition}[Population]\label{def:g12:populations-conservation:pop}
A \emph{population}\index{population} is a group of individuals of the
same \omterm{def:g12:phylogeny-macroevolution:species}{species} that live in the same area and can interbreed. Key
descriptors include size $N$, density, age structure and geographic
distribution.
\end{definition}

\begin{definition}[Population growth rate]
\label{def:g12:populations-conservation:rate}
In a closed \omterm{def:g12:populations-conservation:pop}{population} (no migration), the change in size depends on
births and deaths. The per capita rate of increase $r$ summarises the
balance: when $r>0$ the \omterm{def:g12:populations-conservation:pop}{population} tends to grow; when $r<0$ it shrinks.
\end{definition}

\begin{proposition}[Exponential growth]
\label{prop:g12:populations-conservation:exp}
When resources are effectively unlimited and $r$ is constant,
\[
\frac{\mathrm{d}N}{\mathrm{d}t} = rN
\qquad\Rightarrow\qquad
N(t) = N_0\,\mathrm{e}^{rt}.
\]
\omterm{def:g10:scientific-biology:properties}{Growth} is density-independent in form: the larger $N$, the faster the
absolute increase. Such phases occur after colonisation or release from
control, but cannot last indefinitely.
\end{proposition}
\admitted

\begin{definition}[Carrying capacity]
\label{def:g12:populations-conservation:K}
The \emph{carrying capacity}\index{carrying capacity} $K$ is the maximum
\omterm{def:g12:populations-conservation:pop}{population} size that an environment can sustain in the long run given
prevailing resources and conditions.
\end{definition}

\begin{proposition}[Logistic growth]
\label{prop:g12:populations-conservation:logistic}
With density dependence, a simple model is
\[
\frac{\mathrm{d}N}{\mathrm{d}t} = rN\Bigl(1-\frac{N}{K}\Bigr).
\]
\omterm{def:g10:scientific-biology:properties}{Growth} is nearly exponential when $N\ll K$, slows as $N$ approaches
$K$, and is zero at $N=K$. Real \omterm{def:g12:populations-conservation:pop}{populations} may overshoot, oscillate or
shift $K$ as environments change.
\end{proposition}
\admitted

\begin{example}[Interpreting $r$ and $K$]
\label{ex:g12:populations-conservation:rk}
A reintroduced bird \omterm{def:g12:populations-conservation:pop}{population} on an empty island may show high $r$ at
first. As nest sites and food fill, density dependence appears and $N$
levels near $K$. Improving \omterm{def:g10:ecosystems:habitat-niche}{habitat} raises $K$; disease or drought can
lower it.
\end{example}

\begin{method}[Reading a population trajectory]
\label{met:g12:populations-conservation:read}
\begin{enumerate}
  \item Is $N$ rising, falling or stable?
  \item Does the curve look exponential or levelling (logistic-like)?
  \item What resources or enemies likely set $K$?
  \item Are humans changing birth rates, death rates or \omterm{def:g10:ecosystems:habitat-niche}{habitat} area?
\end{enumerate}
\end{method}

\section{Community interactions}

\begin{definition}[Community]\label{def:g12:populations-conservation:community}
A \emph{community}\index{community (ecology)} is the set of interacting
\omterm{def:g12:populations-conservation:pop}{populations} of different \omterm{def:g12:phylogeny-macroevolution:species}{species} in an area.
\end{definition}

\begin{definition}[Major interaction types]
\label{def:g12:populations-conservation:interactions}
\begin{itemize}
  \item \emph{Competition}\index{competition}: both \omterm{def:g12:phylogeny-macroevolution:species}{species} harmed
        ($-/-$) when they share limited resources.
  \item \emph{Predation}\index{predation}, herbivory, parasitism:
        one benefits, one is harmed ($+/-$).
  \item \emph{Mutualism}\index{mutualism}: both benefit ($+/+$).
  \item \emph{Commensalism}\index{commensalism}: one benefits, the
        other little affected ($+/0$).
\end{itemize}
\end{definition}

\begin{proposition}[Competitive exclusion and coexistence]
\label{prop:g12:populations-conservation:exclusion}
Complete competitors may not coexist indefinitely on a single limiting
resource (\emph{competitive exclusion}\index{competitive exclusion});
\omterm{def:g10:ecosystems:habitat-niche}{niche} differentiation, disturbance or \omterm{def:g12:populations-conservation:interactions}{predation} can allow coexistence in
nature.
\end{proposition}
\admitted

\begin{example}[Keystone predator]
\label{ex:g12:populations-conservation:keystone}
A predator that preferentially eats a \omterm{def:g11:mendelian-genetics:dominance}{dominant} competitor can maintain
higher diversity among prey --- a \emph{keystone}\index{keystone species}
role disproportionate to its biomass.
\end{example}

\section{Biodiversity and the crisis}

\begin{definition}[Biodiversity]
\label{def:g12:populations-conservation:biodiversity}
\emph{Biodiversity}\index{biodiversity} is the variety of life at
genetic, \omterm{def:g12:phylogeny-macroevolution:species}{species} and \omterm{def:g10:ecosystems:levels}{ecosystem} levels. It underpins \omterm{def:g10:ecosystems:levels}{ecosystem} services:
\omterm{def:g12:plant-biology:flower}{pollination}, water purification, climate regulation, food and medicine.
\end{definition}

\begin{proposition}[Main drivers of modern loss]
\label{prop:g12:populations-conservation:drivers}
The principal anthropogenic drivers are: \omterm{def:g10:ecosystems:habitat-niche}{habitat} destruction and
fragmentation; overexploitation; invasive \omterm{def:g12:phylogeny-macroevolution:species}{species}; pollution; and climate
change. They often act together.
\end{proposition}
\admitted

\begin{definition}[Extinction debt and small populations]
\label{def:g12:populations-conservation:small}
Small \omterm{def:g12:populations-conservation:pop}{populations} face high extinction risk from demographic chance,
inbreeding, \omterm{def:g11:evolution-mechanisms:drift}{genetic drift} and environmental shocks. \omterm{def:g10:ecosystems:habitat-niche}{Habitat} loss can
create an \emph{extinction debt}\index{extinction debt}: \omterm{def:g12:phylogeny-macroevolution:species}{species} still
present but committed to decline.
\end{definition}

\begin{remark}[Background vs mass extinction]
\label{rem:g12:populations-conservation:mass}
Fossil background rates are low compared with estimated current
vertebrate extinction rates; many biologists describe an ongoing
\omterm{def:g12:populations-conservation:biodiversity}{biodiversity} crisis driven by human activity.
\end{remark}

\section{Conservation strategies}

\begin{definition}[Conservation biology]
\label{def:g12:populations-conservation:cons}
\emph{Conservation biology}\index{conservation biology} is the applied
\omterm{def:g10:scientific-biology:science}{science} of protecting \omterm{def:g12:phylogeny-macroevolution:species}{species}, \omterm{def:g10:biodiversity-classification:biodiversity}{genetic diversity} and \omterm{def:g10:ecosystems:levels}{ecosystems}, combining
ecology, genetics, economics and policy.
\end{definition}

\begin{method}[A toolbox of strategies]
\label{met:g12:populations-conservation:tools}
\begin{enumerate}
  \item \textbf{Protected areas}: reserves that limit destructive use;
        design matters (size, connectivity, representation of \omterm{def:g10:ecosystems:habitat-niche}{habitats}).
  \item \textbf{\omterm{def:g10:ecosystems:habitat-niche}{Habitat} restoration}: reverse degradation, re-establish
        native vegetation and hydrology.
  \item \textbf{Corridors}: reconnect fragments to allow dispersal and
        \omterm{def:g11:evolution-mechanisms:geneflow}{gene flow}.
  \item \textbf{Sustainable harvest}: manage fisheries and forests below
        replacement limits.
  \item \textbf{Ex situ}: captive breeding, \omterm{def:g12:plant-biology:flower}{seed} banks --- backups, not
        substitutes for wild \omterm{def:g10:ecosystems:habitat-niche}{habitat}.
  \item \textbf{Invasive control and pollution reduction}.
  \item \textbf{Climate-aware planning}: protect climatic refugia and
        allow range shifts.
\end{enumerate}
\end{method}

\begin{proposition}[In situ priority]
\label{prop:g12:populations-conservation:insitu}
Protecting and restoring \omterm{def:g10:ecosystems:habitat-niche}{habitats} \emph{in situ} preserves interacting
communities and evolutionary processes; zoos and banks are insurance,
not a full replacement.
\end{proposition}
\admitted

\begin{example}[Population viability]
\label{ex:g12:populations-conservation:pva}
Managers estimate whether a \omterm{def:g12:populations-conservation:pop}{population} is likely to persist for a stated
time (\omterm{def:g12:populations-conservation:pop}{population} viability analysis), using data on $r$, variance in
\omterm{def:g10:scientific-biology:properties}{growth}, catastrophes and \omterm{def:g10:ecosystems:habitat-niche}{habitat} area --- then target the limiting
threats.
\end{example}

\begin{remark}[People and incentives]
\label{rem:g12:populations-conservation:people}
Successful conservation aligns ecological goals with local livelihoods,
\omterm{def:g10:scientific-biology:theory}{law} and education. Fortress-only approaches often fail without social
support.
\end{remark}

\section{Problem: Island reserve under pressure}

\begin{problem}[{Weekend problem --- model a population, diagnose
threats and propose a conservation plan}]
\label{pb:g12:populations-conservation:1}
A flightless island bird has $N_0 = 200$ adults. Historically the island
supported about $K = 2000$ birds. Invasive rats now eat eggs; coastal
\omterm{def:g10:scientific-biology:properties}{development} removes nesting \omterm{def:g10:ecosystems:habitat-niche}{habitat}.

\medskip
\noindent\textbf{Part I --- \omterm{def:g10:scientific-biology:properties}{Growth} models.}
\begin{enumerate}
  \item If rats were absent and \omterm{def:g10:ecosystems:habitat-niche}{habitat} intact, write a logistic model
        using symbols $r$, $N$ and $K=2000$.
  \item Sketch $N(t)$ from $200$ toward $2000$ under that model.
  \item Suppose invasive \omterm{def:g12:populations-conservation:interactions}{predation} holds the \omterm{def:g12:populations-conservation:pop}{population} near $300$
        despite intact \omterm{def:g10:ecosystems:habitat-niche}{habitat} area. Is the effective $K$ still $2000$?
        Explain.
  \item Under exponential approximation with $r = 0.2\,\mathrm{year}^{-1}$,
        compute $N$ after $5$ years from $N_0=200$ (leave in terms of
        $\mathrm{e}^{1}$ if needed: $\mathrm{e}^{1}\approx 2.7$).
\end{enumerate}

\medskip
\noindent\textbf{Part II --- \omterm{def:g12:populations-conservation:community}{Community} context.}
\begin{enumerate}[resume]
  \item Classify bird--rat interaction and \omterm{def:g12:populations-conservation:interactions}{competition} among birds for
        nest sites.
  \item Name one \omterm{def:g12:populations-conservation:interactions}{mutualism} that might support the bird (e.g.\ with a
        plant) and what happens if that partner declines.
  \item How might removing rats affect other \omterm{def:g12:phylogeny-macroevolution:species}{species} (think beyond the
        bird alone)?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Crisis and strategy.}
\begin{enumerate}[resume]
  \item Which global drivers appear locally on the island?
  \item Propose a ranked action list: rat control, \omterm{def:g10:ecosystems:habitat-niche}{habitat} protection,
        captive breeding --- justify the \omterm{def:g10:biodiversity-classification:hierarchy}{order}.
  \item Design a corridor or buffer if \omterm{def:g10:scientific-biology:properties}{development} splits nesting
        beaches.
  \item How would you monitor success for $10$ years (what data)?
  \item In one paragraph, link $r$, $K$, \omterm{def:g12:populations-conservation:community}{community} interactions and
        human institutions in a single conservation narrative for the
        island.
\end{enumerate}
\end{problem}
