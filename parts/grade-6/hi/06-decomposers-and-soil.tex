\chapter{अपघटक और मिट्टी}\label{ch:g6:decomposers-and-soil}

हर पतझड़ जंगल \omterm{def:g1:plants-around-us:parts}{पत्तियों} का मोटा क़ालीन गिरा देता है --- प्रति हेक्टेयर
टनों \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}। फिर भी हर वसंत उसी जंगल में टहलो और तुम्हें पत्तों के
कूड़े का बढ़ता समुंदर पार नहीं करना पड़ता: क़ालीन साल दर साल फिर पतला
मिलता है। जंगल के मुर्दों को कोई खा रहा है। यह अध्याय उन्हीं खाने वालों
से मिलने \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} में उतरता है --- और आख़िरकार पदार्थ का घेरा पूरा करता
है।

\section{मिट्टी जीवित है}

\begin{definition}[मिट्टी]\label{def:g6:decomposers-and-soil:soil}
\emph{मिट्टी}\index{मिट्टी} धरती की जीवित \omterm{def:g1:the-five-senses:sense}{त्वचा} है: \emph{खनिज} कणों
(पिसी हुई चट्टान, बालू से लेकर चिकनी मिट्टी तक), गलने की हर अवस्था वाले
\emph{कार्बनिक} अवशेषों, पानी, हवा --- और सजीवों की एक विशाल आबादी का
मेल। उसकी ऊपरी गहरी परत, जो गले हुए पदार्थ से सबसे समृद्ध होती है, वही
\emph{ह्यूमस}\index{ह्यूमस} की परत है जिसे माली बहुत मानते हैं।
\end{definition}

\begin{example}[एक मुट्ठी, गिनी हुई]\label{ex:g6:decomposers-and-soil:handful}
जंगल की \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} की एक मुट्ठी, सफ़ेद तश्तरी पर छाँटी हुई: खनिज कण; \omterm{def:g1:plants-around-us:parts}{जड़ों}
के धागे; आधी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}, \omterm{def:g1:plants-around-us:parts}{पत्तियों} के ढाँचे और काले टुकड़े --- यानी नीचे
की ओर जाता \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ}; और गिनती --- एक केंचुआ, दो कठ-जूएँ, एक
गोजर, नन्हे सफ़ेद स्प्रिंगटेल, और धागे जैसे महीन कवक-जाल। और तश्तरी
सिर्फ़ दिखने वाले दिखाती है: \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} के एक ग्राम में \omterm{def:g1:the-five-senses:sense}{आँखों} से न दिखने
वाले उतने \omterm{def:g6:exploring-our-environment:components}{सजीव} होते हैं जितने लोग किसी बड़े शहर में भी नहीं।
\end{example}

\begin{definition}[अपघटक]\label{def:g6:decomposers-and-soil:decomposers}
\emph{अपघटक}\index{अपघटक} वे \omterm{def:g6:exploring-our-environment:components}{सजीव} हैं जो \emph{मरे हुए \omterm{def:g6:plants-are-producers:matter}{कार्बनिक
पदार्थ}} पर पलते हैं --- गिरी हुई \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}, सूखी लकड़ी, मल, लाशें:
\begin{itemize}
  \item दिखने वाला दल: केंचुए, कठ-जूएँ, गोजर, स्प्रिंगटेल, मक्खियों के
        डिंभ, गोबर के भृंग;
  \item \emph{कवक}\index{कवक}: फफूँदें और कुकुरमुत्ते, जिनका खाने वाला
        शरीर \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} और सूखी लकड़ी में फैले सफ़ेद धागों का जाल होता है
        --- \cref{exo:g1:living-or-not:10} का कुकुरमुत्ता तो उसका
        सिर्फ़ \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} है;
  \item और हर काम को पूरा करती, \omterm{def:g1:the-five-senses:sense}{आँखों} से न दिखने वाली \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} की
        जीव-सृष्टि --- यानी आख़िरी खोलने वाले।
\end{itemize}
ये \cref{def:g5:food-webs:roles} के पर्दे के पीछे काम करने वाले हैं,
जिन्हें अपना अध्याय देने का वादा था: यही वह अध्याय है।
\end{definition}

\section{अपघटन: कार्बनिक से वापस खनिज तक}

\begin{proposition}[अपघटन करता क्या है]\label{prop:g6:decomposers-and-soil:decomposition}
\emph{अपघटन}\index{अपघटन} अपघटकों का खाना ही है, और उसका नतीजा
\omterm{def:g5:food-webs:roles}{उत्पादकों} के पेशे का ठीक उलटा है: मरा हुआ \emph{कार्बनिक} पदार्थ चरण
दर चरण और खाने वाले दर खाने वाले खोला जाता है, जब तक उसका पदार्थ
\emph{खनिज} दुनिया को न लौट जाए --- पहले भुरभुरा \omterm{def:g6:decomposers-and-soil:soil}{ह्यूमस}, फिर \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} के
पानी में घुले \omterm{prop:g4:what-plants-need:needs}{खनिज लवण}, और हवा को लौटी हुई गैस। पतझड़ का क़ालीन वसंत तक
ठीक वही बन जाता है जो \omterm{def:g1:plants-around-us:parts}{जड़ें} पीती हैं
(\cref{prop:g4:what-plants-need:needs})।
\end{proposition}
\admitted

\begin{example}[पत्ती का आख़िरी साल]\label{ex:g6:decomposers-and-soil:leaf}
बलूत की एक \omterm{def:g1:plants-around-us:parts}{पत्ती} का नीचे तक पीछा करो। नवंबर: गिरी हुई, साबुत। सर्दी:
नरम पड़ी, \omterm{def:g6:decomposers-and-soil:decomposers}{कवक} के धागों से खोदी हुई, और स्प्रिंगटेलों तथा कठ-जुओं की
कुतरन से जाली जैसा ढाँचा बनी हुई। वसंत: किसी केंचुए ने ज़मीन के नीचे
घसीटी, खाई, और काम किए हुए टुकड़ों के रूप में बाहर छोड़ दी। गर्मी: उसके
आख़िरी टुकड़े काला \omterm{def:g6:decomposers-and-soil:soil}{ह्यूमस} हैं, और उसके खनिज \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} के पानी पर सवार ---
किसी \omterm{def:g1:plants-around-us:parts}{जड़} में, शायद ख़ुद उसी बलूत की \omterm{def:g1:plants-around-us:parts}{जड़} में। चार बार खाई जाकर वह \omterm{def:g1:plants-around-us:parts}{पत्ती}
अपने ही \omterm{def:g1:plants-around-us:tree}{पेड़} का भोजन बन गई है।
\end{example}

\begin{remark}[घेरा पूरा होता है]\label{rem:g6:decomposers-and-soil:circle}
अपघटकों के आ जाने से इस साल की पदार्थ-कथा एक पहिए में बंद हो जाती है:
\omterm{def:g5:food-webs:roles}{उत्पादक} खनिज से कार्बनिक बनाते हैं
(\cref{ch:g6:plants-are-producers}); \omterm{def:g5:food-webs:roles}{उपभोक्ता} उसे आगे बढ़ाते हैं
(\cref{ch:g6:matter-from-food}); और \omterm{def:g6:decomposers-and-soil:decomposers}{अपघटक} उसे फिर खनिज बना देते हैं,
जो \omterm{def:g5:food-webs:roles}{उत्पादकों} के लिए तैयार है। पदार्थ ख़र्च होकर ख़त्म नहीं होता --- वह
घूमता रहता है। इस पन्ने के काग़ज़ के परमाणु इतनी बार \omterm{def:g1:plants-around-us:parts}{पत्ती}, \omterm{def:g6:decomposers-and-soil:soil}{ह्यूमस}, \omterm{def:g1:plants-around-us:parts}{जड़}
और फिर \omterm{def:g1:plants-around-us:parts}{पत्ती} रह चुके हैं कि गिनती नहीं की जा सकती।
\end{remark}

\section{अपघटकों के साथ काम करना}

\begin{example}[खाद, आख़िरकार समझाई हुई]\label{ex:g6:decomposers-and-soil:compost}
\cref{ex:g2:caring-for-nature:compost} का खाद का डिब्बा अब एक उपकरण की
तरह पढ़ा जाता है: छिलकों से भरी अपघटकों की एक कार्यशाला। केंचुए,
कठ-जूएँ और न दिखने वाले दल रसोई के कार्बनिक कचरे को खोलकर गहरा \omterm{def:g6:decomposers-and-soil:soil}{ह्यूमस}
बना देते हैं; माली उसे बिखेरता है, और उसके खनिज क्यारियों का पोषण करते
हैं (\cref{rem:g4:what-plants-need:soil})। वह ढेर तो गरम भी रहता है ---
ख़ुद अपघटकों का जलाना
(\cref{prop:g6:matter-from-food:losses}) उनकी कार्यशाला को गरम कर देता
है।
\end{example}

\begin{proposition}[काम को क्या तेज़ करता है और क्या धीमा]\label{prop:g6:decomposers-and-soil:speed}
\omterm{def:g6:decomposers-and-soil:decomposers}{अपघटक} भी अपनी \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} वाले \omterm{def:g6:exploring-our-environment:components}{सजीव} हैं
(\cref{prop:g6:exploring-our-environment:decide}): वे \emph{गरम},
\emph{नम} और \emph{हवादार} में सबसे तेज़ काम करते हैं, और ठंड में, सूखे
में या हवा-रहित डूबी हालत में लगभग रुक जाते हैं। इसीलिए पतझड़ का क़ालीन
सर्दी भर धीरे-धीरे घटता है और वसंत में जल्दी साफ़ हो जाता है; और इसीलिए
माली की खाद की तरकीब भी यही है --- नम, हवा के लिए उलटी-पलटी जाती, और
कभी भी भीगा हुआ बंद ढेर नहीं।
\end{proposition}
\admitted

\begin{method}[मिट्टी की सेहत पढ़ना]\label{met:g6:decomposers-and-soil:read}
किसी \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} को प्रकृतिविद के ढंग से परखने के लिए:
\begin{enumerate}
  \item एक बालिश्त गहरा खोदो और देखो: गहरा भुरभुरा \omterm{def:g6:decomposers-and-soil:soil}{ह्यूमस}, या पीली
        बेजान धूल?
  \item एक मुट्ठी की गिनती सफ़ेद तश्तरी पर करो --- केंचुए, कठ-जूएँ,
        स्प्रिंगटेल: भरा-पूरा दल मतलब चक्र चल रहा है;
  \item सतह जाँचो: क्या मरा हुआ पदार्थ ऋतुओं के साथ ग़ायब होता जाता है,
        या बिना गले ढेर होता जाता है?
  \item इस दल को पशुधन की तरह सँभालो: उसे खिलाओ (जोती हुई \omterm{def:g1:plants-around-us:parts}{जड़ें}, खाद,
        पलवार), नम रखो, और ज़हरों से बचाओ।
\end{enumerate}
\end{method}

\begin{example}[केंचुआ, मालियों का माली]\label{ex:g6:decomposers-and-soil:worm}
केंचुआ दोहरी ड्यूटी करता है: \omterm{def:g6:decomposers-and-soil:decomposers}{अपघटक} --- \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} नीचे घसीटता और गला हुआ
पदार्थ खाता --- और \emph{हलवाहा} भी, जिसकी बिलें \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} को हवा देती हैं
और जिसके मल खनिज तथा कार्बनिक को आपस में गूँध देते हैं। किसी एक मैदान
में घास के नीचे उतना केंचुआ-वज़न हो सकता है जितना ऊपर गाय-वज़न भी नहीं;
पुराने प्रकृतिविद केंचुए को खेती की ज़मीन का असली बनाने वाला कहते थे,
और वे सही थे।
\end{example}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decomposers-and-soil:4}
बलूत की \omterm{def:g1:plants-around-us:parts}{पत्ती} का आख़िरी साल चारों ऋतुओं में दोहराओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decomposers-and-soil:5}
जंगल का पतझड़ वाला क़ालीन साल दर साल ढेर क्यों नहीं होता जाता?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decomposers-and-soil:6}
\omterm{def:g6:decomposers-and-soil:decomposers}{अपघटक} किन तीन \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} में सबसे तेज़ काम करते हैं? साल में यह काम
कहाँ धीमा पड़ता है, और क्यों?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:decomposers-and-soil:7}
पदार्थ का चक्र उसके तीनों पेशों --- \omterm{def:g5:food-webs:roles}{उत्पादक}, \omterm{def:g5:food-webs:roles}{उपभोक्ता}, \omterm{def:g6:decomposers-and-soil:decomposers}{अपघटक} --- और
हर जोड़ी के बीच एक-एक तीर के साथ बनाओ या उसका वर्णन करो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:decomposers-and-soil:8}
खाद के डिब्बे को अपघटकों की कार्यशाला के रूप में समझाओ: वहाँ कौन काम
करता है, किस पर, बग़ीचे के लिए क्या बनाता है --- और वह ढेर गरम क्यों

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:decomposers-and-soil:9}
केंचुए के दोनों काम बताओ, और यह भी कि पुराने प्रकृतिविद उसे खेती की
ज़मीन का बनाने वाला क्यों कहते थे।

\begin{exercise}[$\star\star$]\label{exo:g6:decomposers-and-soil:10}

\end{exercise}

\section{समस्या: पत्ती-थैलों के प्रयोग}

\begin{problem}[{सप्ताहांत समस्या --- जाली के थैले, गाड़ी हुई पत्तियाँ,
और मृदा-विज्ञान की एक सर्दी}]\label{pb:g6:decomposers-and-soil:1}
कक्षा नवंबर में स्कूल की बाड़ के किनारे \omterm{def:g1:plants-around-us:parts}{पत्तियों} के थैले गाड़ती है: जाली
के थैलों में बलूत की ताज़ा \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}, हर थैले में बराबर वज़न, और साल भर
खोदकर फिर से तौलने के लिए। थैलों के तीन प्रकार: मोटी जाली (केंचुए,
कठ-जूएँ, सब कुछ भीतर आने देती है), महीन जाली (दिखने वाले दल को रोकती
है, सिर्फ़ न दिखने वालों को आने देती है), और बंद प्लास्टिक (सब कुछ
रोकता है, और हवा भी)।

\medskip\noindent\textbf{भाग I --- तैयारी।}
\begin{enumerate}
  \item सारे थैलों में \emph{उन्हीं} \omterm{def:g1:plants-around-us:parts}{पत्तियों} का \emph{बराबर} वज़न,
        \emph{उसी} दिन, \emph{उसी} किनारे पर क्यों गाड़ा जाना चाहिए?
        उस विधि का नाम बताओ।
  \item महीन जाली वाला थैला वह कौन-सा सवाल पूछता है जो मोटी जाली वाला
        अकेले नहीं पूछ सकता?
  \item एक टोली चौथा थैला चाहती है: मोटी जाली वाला, पर स्कूल के अहाते
        की सूखी बजरी में गाड़ा हुआ।
        \cref{prop:g6:decomposers-and-soil:speed} के अनुसार वह कौन-सा
        सवाल पूछेगा?
  \item खोदने से पहले भविष्यवाणी करो: बाड़ के किनारे वाले तीनों थैलों
        को इस क्रम में लगाओ कि जून तक वे कितना पत्ती-वज़न खोएँगे।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- नतीजे।}
\begin{enumerate}[resume]
  \item जून की तौल: मोटी जाली --- ज़्यादातर वज़न ग़ायब, \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}
        ढाँचों और टुकड़ों में बदलीं; महीन जाली --- साफ़ तौर पर हलकी,
        \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} नरम और काली पड़ीं पर धीमी; बंद प्लास्टिक --- वज़न
        लगभग जितना का तितना, और भीतर खट्टी लोई। हर थैले की व्याख्या
        करो।
  \item मोटी जाली वाले थैले में कक्षा को कठ-जूएँ, एक केंचुआ,
        स्प्रिंगटेल और सफ़ेद कवक-धागे मिलते हैं। \omterm{def:g1:plants-around-us:parts}{पत्ती} को खोलने में हर
        एक की भूमिका बताओ
        (\cref{ex:g6:decomposers-and-soil:leaf} नमूना है)।
  \item महीन जाली वाले थैले ने सचमुच वज़न खोया, जबकि हर दिखने वाला \omterm{def:g1:animals-around-us:animal}{जंतु}
        रोक दिया गया था। इससे न दिखने वाली \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी} की जीव-सृष्टि के
        बारे में क्या साबित होता है?
  \item बजरी वाले अहाते का थैला (सवाल 3 वाला) अपने बाड़-किनारे वाले
        जुड़वाँ से कहीं कम खोता है। बजरी काम की कौन-सी \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ}
        छीन लेती है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- बड़ी सीख।}
\begin{enumerate}[resume]
  \item मोटी जाली वाले थैले का खोया हुआ वज़न \emph{गया कहाँ}?
        \cref{prop:g6:decomposers-and-soil:decomposition} से उत्तर दो
        --- कम से कम दो ठिकाने।
  \item बाड़ की वसंत वाली बिच्छू-बूटियाँ ठीक उसी किनारे पर सबसे हरी
        उगती हैं जहाँ कक्षा ने अपने थैले गाड़े थे। क्या यह संयोग है?
        पदार्थ के चक्र से दलील दो।
  \item बंद थैले की मदद से समझाओ कि कूड़ाघर में प्लास्टिक में हवा-बंद
        लपेटा कार्बनिक कचरा दशकों टिका क्यों रहता है, जबकि वही कचरा
        खाद में महीनों में गल जाता है।
  \item दो वाक्यों में निष्कर्ष दो: इस सर्दी ने यह क्या साबित किया कि
        जंगल का क़ालीन कौन साफ़ करता है, और वह सफ़ाई किसका पोषण करती
        है?
\end{enumerate}
\end{problem}
