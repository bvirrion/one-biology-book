\chapter{DNA: Molekul Genetik yang Semesta}\label{ch:g10:universal-dna}

Di laboratorium yang digelapkan, di bawah lampu biru, satu dari dua
mencit putih berpendar hijau dari hidung sampai ekor. Ia membawa, dalam
setiap \omterm{def:g10:cells-common-unit:cell}{selnya}, satu \omterm{def:g10:universal-dna:gene}{gen} yang diambil dari ubur-ubur yang menyala di
Pasifik --- dan \omterm{def:g10:cells-common-unit:cell}{sel} mencit itu membaca \omterm{def:g10:universal-dna:gene}{gen} ubur-ubur tersebut persis
seperti \omterm{def:g10:cells-common-unit:cell}{sel} ubur-ubur membacanya. Percobaan itu berhasil karena setiap
makhluk hidup menuliskan \omterm{def:g10:universal-dna:gene}{gennya} dalam molekul yang sama, dengan empat
huruf yang sama, dan membacanya menurut aturan yang sama. Bab ini
membahas molekul itu, \omterm{def:g10:universal-dna:information}{DNA}: bagaimana ia dibangun, apa yang dijelaskan
strukturnya, dan apa yang dibuktikan mencit yang berpendar itu.

\section{Bahan genetiknya}

\begin{definition}[Informasi genetik]\label{def:g10:universal-dna:information}
Yang disebut \emph{informasi genetik}\index{informasi genetik} sebuah
\omterm{def:g10:cells-common-unit:cell}{sel} adalah himpunan petunjuk, yang diwariskan dari sebuah \omterm{def:g10:cells-common-unit:cell}{sel} kepada
\omterm{def:g10:cells-common-unit:cell}{sel} anaknya dan dari orang tua kepada keturunannya, yang menentukan
sifat organisme dan \omterm{def:g10:chemistry-of-life:families}{protein} yang dapat dibuat \omterm{def:g10:cells-common-unit:cell}{selnya}. Pada setiap
makhluk hidup ia dibawa oleh molekul \emph{DNA}\index{DNA} (asam
deoksiribonukleat), yang terkemas bersama \omterm{def:g10:chemistry-of-life:families}{protein} menjadi
\emph{kromosom}\index{kromosom} --- di dalam inti \omterm{def:g10:cells-common-unit:prokaryote}{sel eukariotik}, bebas
dalam \omterm{def:g10:cells-common-unit:cell}{sitoplasma} bakteri.
\end{definition}

\begin{proposition}[DNA adalah molekul pewarisan sifat]\label{prop:g10:universal-dna:material}
Bahan pewarisan sifat sebuah \omterm{def:g10:cells-common-unit:cell}{sel} adalah \omterm{def:g10:universal-dna:information}{DNA-nya}, bukan \omterm{def:g10:chemistry-of-life:families}{proteinnya} atau
penyusun lainnya: memindahkan \omterm{def:g10:universal-dna:information}{DNA} murni dari satu galur bakteri ke
galur lain memindahkan sifat galur yang pertama.
\end{proposition}
\begin{proof}[Bukti percobaan]
Griffith (1928) menemukan bahwa bakteri pneumonia yang tak berbahaya,
jika dicampur dengan \omterm{def:g10:cells-common-unit:cell}{sel} mati galur yang mematikan, menjadi mematikan
dan tetap begitu selama beberapa keturunan: sesuatu dari \omterm{def:g10:cells-common-unit:cell}{sel} mati itu
telah ``mentransformasi'' mereka. Avery, MacLeod dan McCarty (1944)
memisahkan sari \omterm{def:g10:cells-common-unit:cell}{sel} mati itu menjadi keluarga molekulnya dan menguji
masing-masing: hanya pecahan \omterm{def:g10:universal-dna:information}{DNA} yang mentransformasi; memperlakukan
sarinya dengan enzim perusak \omterm{def:g10:universal-dna:information}{DNA} menghapus akibatnya, sedangkan enzim
yang merusak \omterm{def:g10:chemistry-of-life:families}{protein} atau RNA tidak. Zat pentransformasi itu, dan
dengan demikian pembawa sifat yang diwariskan, adalah \omterm{def:g10:universal-dna:information}{DNA}.
\end{proof}

\begin{example}[Di mana DNA itu berada]\label{ex:g10:universal-dna:where}
Zat warna khusus \omterm{def:g10:universal-dna:information}{DNA} mewarnai \omterm{def:g10:cells-common-unit:organelle}{inti sel} pipi dan tidak mewarnai apa pun
lagi dalam \omterm{def:g10:cells-common-unit:cell}{sitoplasmanya}; pada \omterm{def:g10:cells-common-unit:cell}{sel} yang membelah ia mewarnai \omterm{def:g10:universal-dna:information}{kromosom}
yang memadat. Bakteri menyerap zat warna itu pada daerah tengah yang
tidak berselubung. Pada keduanya, jumlah \omterm{def:g10:universal-dna:information}{DNA} berlipat dua sebelum tiap
pembelahan dan terbagi dua di antara kedua \omterm{def:g10:cells-common-unit:cell}{sel} anaknya --- perilaku
yang diharapkan dari keterangan yang harus disalin dan dibagi.
\end{example}

\section{Struktur DNA}

\begin{definition}[Nukleotida]\label{def:g10:universal-dna:nucleotide}
\omterm{def:g10:universal-dna:information}{DNA} adalah rantai \emph{nukleotida}\index{nukleotida}. Tiap nukleotida
terbuat dari tiga bagian: gugus fosfat, sebuah gula (deoksiribosa) dan
salah satu dari empat \emph{basa}\index{basa (DNA)} bernitrogen: adenin
(A), timin (T), guanin (G) dan sitosin (C). Gula dan fosfat
berselang-seling membentuk tulang punggung rantainya; basanya mencuat
darinya, satu tiap nukleotida. Keempat nukleotida hanya berbeda pada
basanya, sehingga sebuah untai \omterm{def:g10:universal-dna:information}{DNA} dilukiskan oleh urutan basanya, yang
ditulis sebagai untaian huruf: \texttt{ATGGCTTAC\dots}
\end{definition}

\begin{proposition}[Heliks ganda]\label{prop:g10:universal-dna:helix}
Molekul \omterm{def:g10:universal-dna:information}{DNA} terbuat dari dua untai yang saling melilit menjadi
\emph{heliks ganda}\index{heliks ganda} putar-kanan selebar sekitar
\qty{2}{nm}, dengan satu putaran tiap sepuluh pasangan \omterm{def:g10:universal-dna:nucleotide}{basa}
(\qty{3.4}{nm}). Kedua untainya berjalan dalam arah berlawanan dan
saling berhadapan lewat \omterm{def:g10:universal-dna:nucleotide}{basanya}, yang berpasangan menurut aturan
\emph{komplementaritas}\index{komplementaritas} yang ketat: A hanya
berpasangan dengan T, G hanya dengan C. Karena itu urutan satu untai
menetapkan urutan untai yang lain.
\end{proposition}
\begin{proof}[Bukti percobaan]
Chargaff (1950) mengukur komposisi \omterm{def:g10:universal-dna:nucleotide}{basa} \omterm{def:g10:universal-dna:information}{DNA} dari banyak spesies dan
menemukan, pada setiap spesies, adenin sebanyak timin dan guanin
sebanyak sitosin, sedangkan perbandingan A+T terhadap G+C berbeda dari
spesies ke spesies --- sebuah aturan berpasangan, bukan kebetulan
kimia. Foto sinar-X serat \omterm{def:g10:universal-dna:information}{DNA} yang dibuat Franklin (1952)
memperlihatkan tanda khas heliks bergaris tengah \qty{2}{nm} dan
berkisar \qty{3.4}{nm}. Watson dan Crick (1953) membangun satu-satunya
model yang cocok dengan keduanya: dua tulang punggung di luar, \omterm{def:g10:universal-dna:nucleotide}{basa}
berpasangan di dalam, dengan pasangan A--T dan G--C yang selebar sama
sehingga heliksnya teratur.
\end{proof}

\begin{example}[Aturan Chargaff dalam angka]\label{ex:g10:universal-dna:chargaff}
\omterm{def:g10:universal-dna:information}{DNA} manusia: A 30.9\%, T 29.4\%, G 19.9\%, C 19.8\% --- A$\,\approx\,$T
dan G$\,\approx\,$C sampai ketelitian pengukurannya, dan
A+T $= 60\%$. \emph{E.~coli}: A 24.7\%, T 23.6\%, G 26.0\%, C 25.7\%,
A+T $= 48\%$. Ragi: A+T $= 64\%$. Jika sebuah cuplikan \omterm{def:g10:universal-dna:information}{DNA} memuat 32\%
A, \omterm{prop:g10:universal-dna:helix}{komplementaritas} meramalkan 32\% T dan masing-masing $(100 - 64)/2 = 18\%$
G dan C.
\end{example}

\begin{method}[Menuliskan untai komplementernya]\label{met:g10:universal-dna:complement}
Diberikan satu untai, tuliskan di bawah tiap \omterm{def:g10:universal-dna:nucleotide}{basa} pasangannya (A$\leftrightarrow$T,
G$\leftrightarrow$C), lalu baca hasilnya dalam arah berlawanan jika
kesepakatan latihannya menuntut demikian. Untuk
\texttt{5'-ATGGCTTAC-3'} untai pasangannya, yang dibaca dalam arahnya
sendiri, adalah \texttt{5'-GTAAGCCAT-3'}. Penghitungannya: untai dengan
$n_\mathrm{A}$ adenin berhadapan dengan untai dengan $n_\mathrm{A}$
timin, sehingga dalam molekul utuhnya $n_\mathrm{A} = n_\mathrm{T}$ dan
$n_\mathrm{G} = n_\mathrm{C}$.
\end{method}

\begin{proposition}[Apa yang dijelaskan strukturnya]\label{prop:g10:universal-dna:explains}
Dua sifat bahan genetik mengikuti dari strukturnya.
\begin{itemize}
  \item \emph{Keterangan}: urutan \omterm{def:g10:universal-dna:nucleotide}{basa} di sepanjang sebuah untai tidak
        dibatasi oleh kimianya --- setiap urutan mungkin --- sehingga
        urutannya dapat membawa pesan, sebagaimana urutan huruf membawa
        teks. \omterm{def:g10:cells-common-unit:cell}{Sel} manusia memuat $\num{3.2e9}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} dalam tiap
        perangkat \omterm{def:g10:universal-dna:information}{kromosomnya}.
  \item \emph{Penyalinan}: karena tiap untai menentukan untai yang
        lain, molekulnya dapat digandakan dengan memisahkan kedua
        untainya dan membangun pasangan baru pada masing-masing.
        Mekanismenya menjadi pokok \cref{ch:g11:dna-replication}.
\end{itemize}
\end{proposition}
\admitted

\section{Gen, alel, kromosom}

\begin{definition}[Gen dan alel]\label{def:g10:universal-dna:gene}
Sebuah \emph{gen}\index{gen} adalah ruas molekul \omterm{def:g10:universal-dna:information}{DNA} --- lazimnya
beberapa ribu pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} --- yang urutannya membawa petunjuk untuk
membuat satu \omterm{def:g10:chemistry-of-life:families}{protein} (atau, kadang-kadang, satu molekul RNA yang
bekerja). Ia menempati kedudukan tetap pada \omterm{def:g10:universal-dna:information}{kromosom} tertentu. Versi
sebuah gen yang berbeda oleh beberapa \omterm{def:g10:universal-dna:nucleotide}{basa} adalah
\emph{alel}\index{alel}-nya: kedudukan sama, fungsi sama, urutan sedikit
berbeda, sehingga sering \omterm{def:g10:chemistry-of-life:families}{proteinnya} sedikit berbeda pula.
\end{definition}

\begin{example}[Angka-angkanya]\label{ex:g10:universal-dna:numbers}
Genom manusia --- satu perangkat \omterm{def:g10:universal-dna:information}{kromosom} yang lengkap --- membawa
$\num{3.2e9}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} dan sekitar $\num{20000}$ \omterm{def:g10:universal-dna:gene}{gen}; \omterm{def:g10:universal-dna:gene}{gen} hanya
menempati beberapa persen urutannya. \omterm{def:g10:cells-common-unit:cell}{Sel} \emph{E.~coli} membawa satu
molekul \omterm{def:g10:universal-dna:information}{DNA} melingkar sepanjang $\num{4.6e6}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} dan sekitar
4300 \omterm{def:g10:universal-dna:gene}{gen}. Dua manusia yang tak berkerabat berbeda pada kira-kira satu
\omterm{def:g10:universal-dna:nucleotide}{basa} dari seribu: sekitar tiga juta kedudukan, sebagian besar di luar
\omterm{def:g10:universal-dna:gene}{gen} dan tanpa akibat, beberapa di antaranya \omterm{def:g10:universal-dna:gene}{alel} yang mengubah sifat.
\end{example}

\begin{proposition}[Kromosom]\label{prop:g10:universal-dna:chromosome}
Sebuah \omterm{def:g10:universal-dna:information}{kromosom} adalah satu molekul \omterm{def:g10:universal-dna:information}{DNA}, yang terlilit pada \omterm{def:g10:chemistry-of-life:families}{protein}
pengemas dan tergulung pada dirinya sendiri. Jika direntangkan, \omterm{def:g10:universal-dna:information}{DNA}
satu \omterm{def:g10:universal-dna:information}{kromosom} manusia akan sepanjang \qty{1.5}{cm} sampai \qty{8.5}{cm};
ke-46 \omterm{def:g10:universal-dna:information}{kromosom} sebuah \omterm{def:g10:cells-common-unit:cell}{sel} bersama-sama memuat sekitar \qty{2}{m} \omterm{def:g10:universal-dna:information}{DNA}
dalam \omterm{def:g10:cells-common-unit:organelle}{inti sel} selebar \qty{6}{\micro m}. \omterm{def:g10:universal-dna:information}{Kromosom} terlihat sebagai
benda terpisah hanya selama pembelahan \omterm{def:g10:cells-common-unit:cell}{sel}, ketika ia tergulung paling
rapat; di antara pembelahan ia terurai dan mengisi \omterm{def:g10:cells-common-unit:organelle}{inti sel} sebagai kusut.
\end{proposition}
\admitted

\section{Satu molekul untuk seluruh makhluk hidup}

\begin{proposition}[Kesemestaan DNA]\label{prop:g10:universal-dna:universal}
Seluruh makhluk hidup --- bakteri, jamur, tumbuhan, hewan --- membawa
\omterm{def:g10:universal-dna:information}{informasi genetiknya} sebagai \omterm{def:g10:universal-dna:information}{DNA} yang dibangun dari empat \omterm{def:g10:universal-dna:nucleotide}{nukleotida}
yang sama, dan membaca urutan sebuah \omterm{def:g10:universal-dna:gene}{gen} menurut aturan yang sama.
Akibatnya sebuah \omterm{def:g10:universal-dna:gene}{gen} yang diambil dari satu spesies dan disisipkan ke
dalam \omterm{def:g10:universal-dna:information}{DNA} spesies lain dibaca oleh \omterm{def:g10:cells-common-unit:cell}{sel} penerimanya, yang lalu membuat
\omterm{def:g10:chemistry-of-life:families}{protein} yang disandikannya: \emph{transgenesis}\index{transgenesis}
mungkin dilakukan, dan organisme transgenik mewariskan \omterm{def:g10:universal-dna:gene}{gen} asing itu
kepada keturunannya seperti \omterm{def:g10:universal-dna:gene}{gen} mana pun.
\end{proposition}
\begin{proof}[Bukti percobaan]
\omterm{def:g10:universal-dna:gene}{Gen} untuk \omterm{def:g10:chemistry-of-life:families}{protein} fluoresen hijau (GFP) ubur-ubur \emph{Aequorea
victoria} telah disisipkan ke dalam bakteri, ragi, tumbuhan, ikan,
mencit dan kelinci: setiap satunya membuat \omterm{def:g10:chemistry-of-life:families}{protein} itu dan berpendar
hijau di bawah cahaya biru, dan sifat itu diwariskan. \omterm{def:g10:universal-dna:gene}{Gen} manusia untuk
insulin, yang disisipkan ke dalam \emph{E.~coli} pada 1978, membuat
bakterinya menghasilkan insulin manusia, yang sejak itu menjadi sumber
hormon tersebut bagi penderita diabetes. Tidak dikenal satu pun kasus
spesies yang \omterm{def:g10:universal-dna:gene}{gennya} dituliskan dalam kimia yang lain.
\end{proof}

Molekul sama, aturan baca sama: petunjuk ubur-ubur itu berjalan dalam
tubuh mencit.}
\end{omfigure}

\begin{example}[Kegunaan transgenesis]\label{ex:g10:universal-dna:uses}
Insulin dan hormon pertumbuhan manusia dari bakteri; \omterm{def:g10:universal-dna:gene}{gen} ketahanan
terhadap hama serangga yang disisipkan ke dalam jagung atau kapas; padi
yang membawa \omterm{def:g10:universal-dna:gene}{gen} untuk membuat bakal vitamin A; \omterm{def:g10:universal-dna:gene}{gen} GFP yang
ditempelkan pada \omterm{def:g10:universal-dna:gene}{gen} lain sehingga \omterm{def:g10:cells-common-unit:cell}{sel} tempat \omterm{def:g10:universal-dna:gene}{gen} itu aktif menyala di
bawah mikroskop --- kegunaan terakhir, sebagai alat penelitian, adalah
yang paling lazim di antara semuanya. Tiap kegunaan menimbulkan
pertanyaannya sendiri tentang keamanan dan akibatnya bagi lingkungan;
masing-masing bersandar pada kenyataan yang sama, yaitu kesemestaan
molekul genetiknya.
\end{example}

\begin{remark}[Kesatuan dan keragaman, pada aras molekul]\label{rem:g10:universal-dna:unity}
Kesemestaan \omterm{def:g10:universal-dna:information}{DNA} adalah wajah molekuler kesatuan kehidupan yang ditemui
pada \cref{ch:g10:chemistry-of-life}: satu kimia, satu molekul
keterangan, satu sistem pembacaan bagi setiap organisme di Bumi, dan
itulah yang diramalkan satu asal usul bersama
(\cref{ch:g10:common-ancestry}). Keragaman kehidupan pun tertulis dalam
molekul yang sama, sebagai beda urutan: antara dua \omterm{def:g10:universal-dna:gene}{alel} beberapa \omterm{def:g10:universal-dna:nucleotide}{basa},
antara dua spesies beberapa persen genomnya, antara bakteri dan manusia
seluruh penataan teksnya.
\end{remark}

\section{Soal: Gen Ubur-ubur}

\begin{problem}[{Soal akhir pekan --- satu gen dari ubur-ubur yang
diikuti sampai ke dalam mencit: panjangnya, hurufnya, mengapa mencit
dapat membacanya, dan dua meter DNA yang membawanya dalam setiap sel}]
\label{pb:g10:universal-dna:1}
\omterm{def:g10:chemistry-of-life:families}{Protein} fluoresen hijau ubur-ubur \emph{Aequorea victoria}
adalah rantai 238 \omterm{def:g10:chemistry-of-life:families}{asam amino}. \omterm{def:g10:universal-dna:gene}{Gennya}, sebagaimana dipakai di
laboratorium, adalah ruas \omterm{def:g10:universal-dna:information}{DNA} sepanjang 720 pasangan \omterm{def:g10:universal-dna:nucleotide}{basa}. Ambil
\qty{0.34}{nm} tiap pasangan \omterm{def:g10:universal-dna:nucleotide}{basa}, sepuluh pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} tiap putaran,
dan genom mencit sepanjang $\num{2.7e9}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} tiap perangkat
\omterm{def:g10:universal-dna:information}{kromosom}, dua perangkat tiap \omterm{def:g10:cells-common-unit:cell}{sel}.

\medskip
\noindent\textbf{Bagian I --- Molekulnya.}
\begin{enumerate}
  \item Hitung panjang \omterm{def:g10:universal-dna:gene}{gen} GFP dalam nanometer, dan banyaknya putaran
        heliks yang dibuatnya.
  \item Satu untai \omterm{def:g10:universal-dna:gene}{gen} itu diawali \texttt{ATGAGTAAAGGAGAAGAAC}.
        Tuliskan untai komplementernya.
  \item Dalam \omterm{def:g10:universal-dna:gene}{gen} utuhnya, adenin mewakili 27\% \omterm{def:g10:universal-dna:nucleotide}{basanya}. Berikan
        persentase ketiga \omterm{def:g10:universal-dna:nucleotide}{basa} lainnya.
  \item \omterm{def:g10:universal-dna:gene}{Gen} itu punya 720 pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} dan menyandikan 238 \omterm{def:g10:chemistry-of-life:families}{asam amino}.
        Hitung nisbahnya, dan usulkan apa yang disarankannya tentang
        berapa banyak \omterm{def:g10:universal-dna:nucleotide}{basa} yang menetapkan satu \omterm{def:g10:chemistry-of-life:families}{asam amino} (dugaan yang
        akan dibenarkan \cref{ch:g11:gene-expression}).
  \item Berapa banyak urutan 720 \omterm{def:g10:universal-dna:nucleotide}{basa} yang berbeda yang mungkin pada
        asasnya? Tuliskan jawabannya sebagai pangkat dari 4, lalu
        jelaskan dalam satu kalimat mengapa urutannya dapat disebut keterangan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Mengapa mencit dapat membacanya.}
\begin{enumerate}[resume]
  \item Nyatakan sifat \omterm{def:g10:universal-dna:information}{DNA} yang memungkinkan \omterm{def:g10:cells-common-unit:cell}{sel} mencit memakai \omterm{def:g10:universal-dna:gene}{gen}
        ubur-ubur.
  \item \omterm{def:g10:universal-dna:gene}{Gennya} disisipkan ke dalam \omterm{def:g10:cells-common-unit:cell}{sel} telur yang telah dibuahi, bukan
        ke dalam kulit mencit dewasa. Dengan teori \omterm{def:g10:cells-common-unit:cell}{sel}, jelaskan apa
        bedanya bagi tempat \omterm{def:g10:universal-dna:gene}{gen} itu berakhir.
  \item Mencit dewasanya berpendar pada setiap jaringan. Apa yang
        ditunjukkannya tentang kandungan \omterm{def:g10:universal-dna:information}{DNA} berbagai jenis \omterm{def:g10:cells-common-unit:cell}{selnya}?
  \item Mencit yang berpendar itu dikawinkan dengan mencit biasa.
        Ramalkan, dengan alasan, apakah sebagian keturunannya berpendar.
  \item Seekor mencit kontrol disuntik dengan \omterm{def:g10:chemistry-of-life:families}{protein} GFP murni, bukan
        \omterm{def:g10:universal-dna:gene}{gennya}: ia berpendar samar selama beberapa hari, lalu berhenti.
        Jelaskan beda kedua percobaan itu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Bukti di balik ceritanya.}
\begin{enumerate}[resume]
  \item Dalam percobaan Avery, sebutkan pecahan yang mentransformasi
        bakterinya dan perlakuan yang menghapus transformasinya.
  \item Jelaskan mengapa percobaan transformasi itu adalah bentuk
        alami \omterm{prop:g10:universal-dna:universal}{transgenesis}.
  \item Chargaff menemukan A$\,=\,$T dan G$\,=\,$C pada setiap spesies
        tetapi A+T berubah-ubah dari 25\% sampai 75\%. Pengamatan yang
        mana dari keduanya yang menyokong pemasangan \omterm{def:g10:universal-dna:nucleotide}{basa}, dan yang
        mana yang menunjukkan bahwa urutannya bebas?
  \item Foto Franklin memberikan garis tengah heliksnya \qty{2}{nm}.
        Jelaskan mengapa memasangkan \omterm{def:g10:universal-dna:nucleotide}{basa} besar dengan \omterm{def:g10:universal-dna:nucleotide}{basa} kecil
        (A dengan T, G dengan C) dan bukan dua \omterm{def:g10:universal-dna:nucleotide}{basa} besar membuat
        heliksnya teratur.
  \item Mengapa penemuan strukturnya seketika menyarankan cara \omterm{def:g10:universal-dna:information}{DNA} disalin?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Dua meter dalam sebuah \omterm{def:g10:cells-common-unit:organelle}{inti sel}.}
\begin{enumerate}[resume]
  \item Hitung panjang seluruh \omterm{def:g10:universal-dna:information}{DNA} dalam satu \omterm{def:g10:cells-common-unit:cell}{sel} mencit.
  \item \omterm{def:g10:cells-common-unit:organelle}{Inti selnya} selebar \qty{6}{\micro m}. Dengan faktor berapa \omterm{def:g10:universal-dna:information}{DNA}
        itu harus diperpendek lewat penggulungan agar muat?
  \item Berapa bagian \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:cells-common-unit:cell}{sel} itu yang berupa \omterm{def:g10:universal-dna:gene}{gen} GFP?
  \item Mencit punya sekitar $\num{2e4}$ \omterm{def:g10:universal-dna:gene}{gen} yang rata-rata $\num{3e4}$
        pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} termasuk bagian bukan penyandinya. Berapa bagian
        genomnya yang ditempati \omterm{def:g10:universal-dna:gene}{gen}?
  \item Nyatakan hasilnya dalam satu kalimat: berapa banyak \omterm{def:g10:universal-dna:information}{DNA} yang
        dibawa \omterm{def:g10:cells-common-unit:cell}{sel} mencit, dan kenyataan tunggal apa tentang \omterm{def:g10:universal-dna:information}{DNA} itu
        yang memungkinkan satu \omterm{def:g10:universal-dna:gene}{gen} ubur-ubur di antara ribuan \omterm{def:g10:universal-dna:gene}{gennya} dibaca?
\end{enumerate}
\end{problem}
