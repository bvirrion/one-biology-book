\chapter{Organisasi Fungsional Sebuah Tumbuhan Berbunga}\label{ch:b1:flowering-plant-organization}

Sebatang pohon ek berumur dua abad berdiri pada meter persegi tanah yang
sama tempat ia berkecambah. Ia tak pernah makan, tak pernah berpindah,
tak pernah menjaga suhunya; namun ia telah membangun empat puluh ton
kayu, dan setiap musim panas ia membentangkan lima ratus meter persegi
\omterm{def:b1:flowering-plant-organization:organs}{daun} ke udara dan permukaan \omterm{def:b1:flowering-plant-organization:organs}{akar} yang lebih luas lagi ke dalam tanah.
Sebuah tumbuhan berbunga adalah jawaban yang lain atas masalah
\cref{ch:b1:organism-environment}: autotrof yang menetap dan menemui
\omterm{def:b1:organism-environment:organism}{lingkungannya} bukan dengan memindahkan pertukarannya ke dalam melainkan
dengan menumbuhkan \omterm{prop:b1:organism-environment:surfaces}{permukaan pertukarannya} ke luar, tanpa batas, ke mana
pun cahaya dan air berada. Bab ini memerikan bagaimana tubuh semacam itu
tersusun --- \omterm{def:b1:mammal-organization:organ}{organnya}, jaringannya, permukaan yang dibentangkannya, dan
pertumbuhan yang membangunnya.

\section{Tubuh vegetatif}

\begin{definition}[Akar, batang, daun]\label{def:b1:flowering-plant-organization:organs}
Tubuh vegetatif sebuah tumbuhan berbunga tersusun dari tiga macam \omterm{def:b1:mammal-organization:organ}{organ}.
\emph{Akar}\index{akar} menambatkan tumbuhan dan menyerap air serta ion
mineral dari tanah; ia tumbuh dari ujungnya, bercabang, dan membawa
\emph{rambut akar}\index{rambut akar} tepat di belakang ujungnya.
\emph{Batang}\index{batang} membawa daun ke arah cahaya dan
menghantarkan antara akar dan daun; ia dibangun dari satuan yang
berulang, masing-masing berupa sebuah \emph{buku}\index{buku} yang
membawa sehelai daun dan satu tunas ketiak, serta sebuah
\emph{ruas}\index{ruas} di antara dua buku. \emph{Daun}\index{daun}
adalah \omterm{def:b1:mammal-organization:organ}{organ} fotosintesis: sehelai lembar rata berpermukaan luas dan
bertebal kecil, yang dipegang sebuah tangkai daun. Akar membentuk sistem
akar; batang dan daun, sistem tajuk.
\end{definition}

\begin{proposition}[Tubuh yang bermodul dan terbuka]\label{prop:b1:flowering-plant-organization:modular}
Tubuh tumbuhan bersifat \emph{bermodul}\index{pertumbuhan bermodul}:
tajuknya adalah deretan satuan yang seragam (\omterm{def:b1:flowering-plant-organization:organs}{buku}, \omterm{def:b1:flowering-plant-organization:organs}{daun}, tunas, \omterm{def:b1:flowering-plant-organization:organs}{ruas})
yang ditambahkan satu demi satu oleh tunas ujung, dan tiap tunas ketiak
dapat memulai deretan baru. Pertumbuhannya \emph{tak
tentu}\index{pertumbuhan tak tentu}: tidak ada ukuran dewasa, dan
tumbuhan itu terus menambah modul, dan dengan demikian permukaan,
selama ia hidup. Banyak dan letak modulnya bergantung pada tempatnya ---
cahaya, air, kerusakan --- sehingga jenis yang sama berbentuk lain di
tempat yang lain. Sebaliknya tubuh mamalia tertutup: \omterm{def:b1:mammal-organization:organ}{organnya} tetap
jumlahnya dan ia berhenti tumbuh pada ukuran tertentu.
\end{proposition}

\section{Jaringan tubuh primer}

\begin{definition}[Jaringan pelindung, dasar dan pengangkut]\label{def:b1:flowering-plant-organization:tissues}
Setiap \omterm{def:b1:mammal-organization:organ}{organ} tumbuhan dibangun dari tiga sistem jaringan. \emph{Jaringan
pelindung}\index{jaringan pelindung} adalah
\emph{epidermis}\index{epidermis}, selapis tunggal sel yang menyelimuti
\omterm{def:b1:mammal-organization:organ}{organ}, berlapis \emph{kutikula}\index{kutikula} berlilin di udara dan
tertembus \emph{stomata}\index{stoma}, serta diperpanjang pada \omterm{def:b1:flowering-plant-organization:organs}{akar} oleh
\omterm{def:b1:flowering-plant-organization:organs}{rambut akar}. \emph{Jaringan dasar}\index{jaringan dasar} mengisi \omterm{def:b1:mammal-organization:organ}{organ}:
\emph{parenkim}\index{parenkim} (sel hidup berdinding tipis untuk
fotosintesis dan penimbunan), \emph{kolenkim}\index{kolenkim} (sel hidup
dengan dinding menebal tak merata, penopang lentur \omterm{def:b1:flowering-plant-organization:organs}{batang} muda) dan
\emph{sklerenkim}\index{sklerenkim} (sel berdinding tebal berlignin,
mati saat dewasa, penopang kaku: serabut dan sel batu). \emph{Jaringan
pengangkut}\index{jaringan pengangkut} menghantarkan:
\emph{xilem}\index{xilem} membawa air dan mineral ke atas di dalam
\emph{trakeid}\index{trakeid} dan \emph{trakea}\index{trakea} yang
berongga, berlignin dan mati (\cref{ch:b1:plant-transport});
\emph{floem}\index{floem} membawa gula di dalam \emph{pembuluh
tapis}\index{pembuluh tapis} hidup yang dijaga tetap hidup oleh sel
pengiringnya.
\end{definition}

\begin{proposition}[Batang dan akar berbeda pada letak jaringan pengangkutnya]\label{prop:b1:flowering-plant-organization:stemroot}
Pada \omterm{def:b1:flowering-plant-organization:organs}{batang} muda \omterm{def:b1:flowering-plant-organization:tissues}{jaringan pengangkut} membentuk berkas yang tersusun
melingkar (dikotil) atau tersebar di dalam \omterm{def:b1:flowering-plant-organization:tissues}{jaringan dasar} (monokotil),
tiap berkas dengan \omterm{def:b1:flowering-plant-organization:tissues}{xilem} menghadap pusat dan \omterm{def:b1:flowering-plant-organization:tissues}{floem} menghadap sisi luar;
tidak ada \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}, dan \omterm{def:b1:flowering-plant-organization:tissues}{jaringan dasarnya} terbagi menjadi empulur di
pusat dan korteks di luar. Pada \omterm{def:b1:flowering-plant-organization:organs}{akar} muda hanya ada satu silinder pusat:
\omterm{def:b1:flowering-plant-organization:tissues}{xilem} membentuk bintang (dua sampai lima lengan pada dikotil, banyak
pada monokotil), \omterm{def:b1:flowering-plant-organization:tissues}{floem} terletak di antara lengannya, dan silinder itu
dibatasi \emph{endodermis}\index{endodermis}, selingkar sel yang dinding
radialnya membawa pita kedap air (\cref{ch:b1:plant-water-minerals}).
Letaknya sesuai dengan fungsinya: \omterm{def:b1:flowering-plant-organization:organs}{batang} menahan lenturan, dan cincin
berkas dekat permukaannya paling kaku untuk massanya; \omterm{def:b1:flowering-plant-organization:organs}{akar} menahan
tarikan, dan silinder pusat adalah susunan sebuah kabel.
\end{proposition}

\begin{method}[Mengenali sebuah sayatan tumbuhan]\label{met:b1:flowering-plant-organization:section}
\begin{enumerate}
  \item Satu silinder pengangkut di pusat dengan \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}: sebuah
        \omterm{def:b1:flowering-plant-organization:organs}{akar}. Berkas melingkar atau tersebar, dengan empulur: sebuah
        \omterm{def:b1:flowering-plant-organization:organs}{batang}. \omterm{def:b1:mammal-organization:organ}{Organ} pipih dengan dua \omterm{def:b1:flowering-plant-organization:tissues}{epidermis} dan lapisan bunga
        karang: sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun}.
  \item \omterm{def:b1:flowering-plant-organization:organs}{Akar} dengan sedikit lengan \omterm{def:b1:flowering-plant-organization:tissues}{xilem} (2--5): dikotil; banyak lengan
        mengelilingi empulur: monokotil. \omterm{def:b1:flowering-plant-organization:organs}{Batang} dengan berkas pada satu
        lingkaran: dikotil; berkas tersebar: monokotil.
  \item \omterm{def:b1:flowering-plant-organization:tissues}{Xilem} dikenali dari selnya yang besar, berdinding tebal dan
        kosong (merah dengan pewarna yang biasa); \omterm{def:b1:flowering-plant-organization:tissues}{floem} dari selnya yang
        kecil dan hidup dengan \omterm{def:b1:flowering-plant-organization:tissues}{pembuluh tapis} serta sel pengiring (hijau
        atau biru).
  \item Pada sayatan \omterm{def:b1:flowering-plant-organization:organs}{batang}, \omterm{def:b1:flowering-plant-organization:tissues}{xilem} menunjuk ke dalam dan \omterm{def:b1:flowering-plant-organization:tissues}{floem} ke luar.
        Pada \omterm{def:b1:flowering-plant-organization:organs}{akar}, \omterm{def:b1:flowering-plant-organization:tissues}{xilem} dan \omterm{def:b1:flowering-plant-organization:tissues}{floem} berselang-seling pada jari-jari yang
        sama.
\end{enumerate}
\end{method}

\begin{example}[Tempat pertukaran berlangsung]\label{ex:b1:flowering-plant-organization:where}
Karbon dioksida masuk ke sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} lewat \omterm{def:b1:flowering-plant-organization:tissues}{stomatanya}, melintasi ruang
udara lapisan bunga karang dan melarut dalam dinding basah sel palisade;
air datang di dalam \omterm{def:b1:flowering-plant-organization:tissues}{xilem} tulang \omterm{def:b1:flowering-plant-organization:organs}{daunnya} dan pergi, sebagai uap, lewat
\omterm{def:b1:flowering-plant-organization:tissues}{stomata} yang sama. Air dan ion masuk ke \omterm{def:b1:flowering-plant-organization:organs}{akar} lewat \omterm{def:b1:flowering-plant-organization:organs}{rambut akar} dan
melintasi korteks menuju \omterm{def:b1:flowering-plant-organization:tissues}{xilem} di pusat. Kedua \omterm{def:b1:mammal-organization:organ}{organ} itu menempatkan
\omterm{prop:b1:organism-environment:surfaces}{permukaan pertukarannya} di sisi luar dan jaringan penghantarnya di
tengah.
\end{example}

\section{Hidup menetap: permukaan}

\begin{definition}[Indeks luas daun]\label{def:b1:flowering-plant-organization:lai}
Yang disebut \emph{indeks luas daun}\index{indeks luas daun} $L$ sebuah
tudung tumbuhan adalah total luas \omterm{def:b1:flowering-plant-organization:organs}{daun} satu sisi tiap satuan luas tanah.
Padang rumput pada bulan Juni memiliki $L \approx 3$, hutan peluruh
\qtyrange{4}{6}{}, hutan hujan tropis sampai \qty{8}{}: beberapa lapis
\omterm{def:b1:flowering-plant-organization:organs}{daun} bertumpuk di atas tiap meter persegi tanah.
\end{definition}

\begin{proposition}[Penangkapan cahaya oleh kanopi]\label{prop:b1:flowering-plant-organization:beer}
Fraksi cahaya datang yang mencapai tanah di bawah sebuah kanopi
berindeks luas \omterm{def:b1:flowering-plant-organization:organs}{daun} $L$ meluruh secara eksponensial,
\[
\frac{I(L)}{I_0} = e^{-kL},
\]
dengan koefisien pemadaman $k$ antara $0.3$ (\omterm{def:b1:flowering-plant-organization:organs}{daun} tegak) dan $0.8$
(\omterm{def:b1:flowering-plant-organization:organs}{daun} mendatar), biasanya $0.5$. Kanopi dengan $L = 5$ dan $k = 0.5$
menangkap $1 - e^{-2.5} = \qty{92}{\%}$ cahayanya.
\end{proposition}
\begin{proof}
Menuruni kanopi, tiap lapis tipis berluas \omterm{def:b1:flowering-plant-organization:organs}{daun} $\dd L$ menangkap fraksi
$k\,\dd L$ dari cahaya yang mencapainya, dengan $k$ adalah fraksi luas
mendatar yang dinaungi satu satuan luas \omterm{def:b1:flowering-plant-organization:organs}{daun} (yang bergantung pada arah
hadap \omterm{def:b1:flowering-plant-organization:organs}{daunnya}). Karena itu $\dd I = -kI\,\dd L$, yang penyelesaiannya
adalah eksponensial tadi.
\end{proof}

\begin{proposition}[Permukaan akar melebihi permukaan daun]\label{prop:b1:flowering-plant-organization:rootsurface}
Satu \omterm{def:b1:flowering-plant-organization:organs}{batang} gandum hitam yang ditumbuhkan empat bulan dalam kotak berisi
\qty{0.05}{m^3} tanah menghasilkan \qty{620}{km} \omterm{def:b1:flowering-plant-organization:organs}{akar}, berpermukaan
\qty{240}{m^2}, dan $\num{1.4e10}$ \omterm{def:b1:flowering-plant-organization:organs}{rambut akar} yang menambahkan
\qty{400}{m^2}: permukaan sekitar seratus kali permukaan \omterm{def:b1:flowering-plant-organization:organs}{daunnya},
tersebar dalam tanah pada kerapatan sepuluh kilometer \omterm{def:b1:flowering-plant-organization:organs}{akar} tiap liter.
\omterm{def:b1:flowering-plant-organization:organs}{Rambut akar}, yang masing-masing berupa perpanjangan pipa satu sel
\omterm{def:b1:flowering-plant-organization:tissues}{epidermis} selebar \qtyrange{5}{20}{\micro m} dan sepanjang sampai satu
milimeter, adalah tempat sebagian besar penyerapan terjadi; masing-masing
hidup beberapa hari dan diganti seiring majunya ujung \omterm{def:b1:flowering-plant-organization:organs}{akar}.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Dittmer (1937) mencuci seluruh sistem \omterm{def:b1:flowering-plant-organization:organs}{akar} satu \omterm{def:b1:flowering-plant-organization:organs}{batang} gandum hitam
keluar dari kotaknya, menghitung dan mengukur cuplikan tiap orde \omterm{def:b1:flowering-plant-organization:organs}{akar}
serta \omterm{def:b1:flowering-plant-organization:organs}{rambut akarnya} di bawah mikroskop, lalu menskalakannya:
\qty{13.8}{} juta \omterm{def:b1:flowering-plant-organization:organs}{akar} sepanjang seluruhnya \qty{623}{km}, dan
\qty{14}{} miliar rambut sepanjang seluruhnya \qty{10620}{km}. Air dan
mineral tanah bergerak begitu lambat sehingga sebuah \omterm{def:b1:flowering-plant-organization:organs}{akar} harus berada
dalam satu milimeter darinya untuk menyerapnya; panjang yang luar biasa
itulah yang menempatkan sebuah \omterm{def:b1:flowering-plant-organization:organs}{akar} dalam jangkauan tiap tetesnya.
\end{proof}

\begin{definition}[Meristem, pertumbuhan primer dan sekunder]\label{def:b1:flowering-plant-organization:meristem}

menjadi bagian jilid Tahun ke-2.
\end{definition}

\begin{proposition}[Bentuk mengikuti tempat]\label{prop:b1:flowering-plant-organization:plasticity}
Karena pertumbuhannya \omterm{prop:b1:flowering-plant-organization:modular}{bermodul} dan \omterm{prop:b1:flowering-plant-organization:modular}{tak tentu}, sebuah tumbuhan
membentuk tubuhnya menurut tempatnya (\emph{keplastisan
fenotipe}\index{keplastisan fenotipe}): sebatang pohon yang tumbuh di
tempat terbuka bercabang rendah dan lebar, jenis yang sama di dalam
hutan menumbuhkan \omterm{def:b1:flowering-plant-organization:organs}{batang} jangkung yang gundul; tumbuhan di tanah kering
mengalokasikan lebih banyak pertumbuhannya ke \omterm{def:b1:flowering-plant-organization:organs}{akar}, di tempat teduh
lebih banyak ke \omterm{def:b1:flowering-plant-organization:organs}{daun}. Jenis dari habitat kering
(\emph{xerofit}\index{xerofit}) membawa \omterm{def:b1:flowering-plant-organization:tissues}{kutikula} tebal, \omterm{def:b1:flowering-plant-organization:tissues}{stomata} yang
terbenam dalam lekuk atau terlindung rambut, \omterm{def:b1:flowering-plant-organization:organs}{daun} kecil atau berair, dan
\omterm{def:b1:flowering-plant-organization:organs}{akar} yang dalam; jenis dari habitat basah
(\emph{hidrofit}\index{hidrofit}) membawa \omterm{def:b1:flowering-plant-organization:tissues}{kutikula} tipis, \omterm{def:b1:flowering-plant-organization:tissues}{stomata} pada
permukaan atas \omterm{def:b1:flowering-plant-organization:organs}{daun} yang mengapung, dan jaringan berisi udara
(\emph{aerenkim}\index{aerenkim}) yang membawa oksigen turun ke \omterm{def:b1:flowering-plant-organization:organs}{akar} di
dalam lumpur tanpa oksigen.
\end{proposition}

\begin{example}[Mamalia dan tumbuhan, berdampingan]\label{ex:b1:flowering-plant-organization:comparison}
Nutrisi: mamalia memakan materi organik; tumbuhan membuatnya dari karbon
dioksida, air dan cahaya. \omterm{prop:b1:organism-environment:surfaces}{Permukaan pertukaran}: di dalam, tetap
ukurannya, dialiri darah; di luar, terus tumbuh, diventilasi angin dan
tanah. \omterm{def:b1:organism-environment:milieu}{Lingkungan internal}: cairan yang diatur pada suhu dan susunan
tetap; tidak ada --- selnya mengatur isinya sendiri dan suhu tumbuhan
adalah suhu udara. Pertumbuhan: sampai bentuk dewasa yang tertentu;
terbuka dan \omterm{prop:b1:flowering-plant-organization:modular}{bermodul}, seumur hidup. Gerak: hewan mendatangi makanannya;
tumbuhan tumbuh ke arah cahaya dan air. Pemaduan: saraf dan hormon dalam
sekon sampai jam; hormon saja, dalam jam sampai hari. Tiap lajur adalah
satu cara yang taat asas untuk menjadi \omterm{def:b1:organism-environment:organism}{sistem terbuka}.
\end{example}

\begin{problem}[{Soal akhir pekan --- daun sebatang pohon ek yang
