\chapter{पादपों का जल और खनिज पोषण}\label{ch:b1:plant-water-minerals}

जुलाई में कोई मक्का का पौधा दिन में एक लीटर जल ऐसी मिट्टी से खींचता है जो
हाथ को सूखी लगती है, और उसके साथ नाइट्रेट के रूप में वे तीस मिलीग्राम
नाइट्रोजन, जो उसे एक दिन की वृद्धि गढ़ने के लिए चाहिए। उसके पास कोई पंप
नहीं है। जल इसलिए घुसता है क्योंकि \omterm{def:b1:flowering-plant-organization:organs}{जड़} मिट्टी से सूखी है, ठीक उसी अर्थ
में जो \cref{ch:b1:membranes-transport} में है; नाइट्रेट इसलिए घुसता है
क्योंकि \omterm{def:b1:flowering-plant-organization:organs}{जड़} की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} उस पर \omterm{def:b1:water-small-molecules:atp}{ATP} खर्च करती हैं; और दोनों को \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} तक
पहुँचने से पहले \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} की एक ही ऐसी परत पार करनी पड़ती है जिनकी
दीवारें \omterm{def:b1:lipids:wax}{मोम} से बंद हैं। यह अध्याय \omterm{def:b1:flowering-plant-organization:organs}{जड़} को एक अवशोषक \omterm{def:b1:mammal-organization:organ}{अंग} के रूप में, उन
जल-विभवों को जो जल को मिट्टी से \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} तक ठेलते हैं, पौधे को चाहिए खनिज
तत्वों और उन्हें लेने के ढंग को, नाइट्रोजन के विशेष प्रसंग को, तथा उन
कवकों और बैक्टीरिया को बताता है जिन्हें अधिकांश \omterm{def:b1:flowering-plant-organization:organs}{जड़ें} इस काम का एक हिस्सा
करने के लिए रखती हैं।

\section{अवशोषक अंग के रूप में जड़}

\begin{definition}[अपप्रवेश्य पथ, जीवद्रव्य पथ, अंतस्त्वचा]\label{def:b1:plant-water-minerals:pathways}
\index{अपप्रवेश्य पथ}\index{जीवद्रव्य पथ}\index{कैस्पेरी पट्टी}
जल और विलेय \omterm{def:b1:flowering-plant-organization:organs}{जड़} के वल्कुट को दो रास्तों से पार करते हैं: \emph{अपप्रवेश्य
पथ}, कोशिका-भित्तियों और अंतराकोशिकीय अवकाशों की सतत प्रणाली, जिससे होकर
वे किसी झिल्ली को पार किए बिना चलते हैं; और \emph{जीवद्रव्य पथ},
जीवद्रव्यतंतुओं से जुड़ी \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} का सतत \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाद्रव्य}
(\cref{ch:b1:eukaryotic-cell}), जिसमें एक जीवद्रव्य-झिल्ली पार करके घुसा
जाता है। वल्कुट की भीतरी सीमा पर \emph{\omterm{prop:b1:flowering-plant-organization:stemroot}{अंतस्त्वचा}}
(\cref{ch:b1:flowering-plant-organization}) अपप्रवेश्य पथ रोक देती है:
सुबेरिन और लिग्निन की एक पट्टी, \emph{कैस्पेरी पट्टी}, उसकी अरीय
दीवारों में समा जाती है, ताकि संवहन-स्तंभ में घुसने वाली हर चीज़ को किसी
अंतस्त्वचीय \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की झिल्ली से गुज़रना ही पड़े। इस तरह झिल्ली के वाहक
तय करते हैं कि \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} तक क्या पहुँचे, और जो \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} में लादा जा चुका है उसे
वापस रिसने से रोकते हैं।
\end{definition}

\begin{proposition}[मूल दाब]\label{prop:b1:plant-water-minerals:rootpressure}
\index{मूल दाब}\index{बिंदुस्राव}
\omterm{prop:b1:flowering-plant-organization:stemroot}{अंतस्त्वचा} और उसके भीतर की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} में आयन पंप करती हैं, जिससे
\omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} का जल-विभव गिरता है और वल्कुट से जल \omterm{def:b1:membranes-transport:osmosis}{परासरण} द्वारा भीतर खिंच आता
है; जब प्ररोह कम वाष्पोत्सर्जन करता है (रात में, नम हवा में), तो जल जमा
होता है और जाइलम-रस कुछ दसवें भाग मेगापास्कल के धनात्मक \emph{\omterm{prop:b1:plant-water-minerals:rootpressure}{मूल दाब}}
के नीचे ऊपर ठेला जाता है --- इतना कि छोटे पौधों की \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} के सिरों से
बूँदें बाहर निकल आएँ (\emph{बिंदुस्राव}) और हवा से रुकी वाहिकाएँ फिर भर
जाएँ, पर इतना नहीं कि रस किसी वृक्ष की चोटी तक चढ़ जाए। दिन में
वाष्पोत्सर्जन का खिंचाव (\cref{ch:b1:plant-transport}) सँभाल लेता है और
\omterm{prop:b1:plant-water-minerals:rootpressure}{मूल दाब} लुप्त हो जाता है।
\end{proposition}
\begin{proof}[साक्ष्य]
ज़मीन के पास से काटा गया कोई \omterm{def:b1:flowering-plant-organization:organs}{तना} अपने ठूँठ से घंटों रस बहाता है, और ठूँठ
पर लगा कोई दाबमापी \qtyrange{0.1}{0.3}{MPa} का दाब दर्ज करता है; बहा हुआ
रस मिट्टी के विलयन से आयनों में अधिक धनी होता है, और \omterm{def:b1:flowering-plant-organization:organs}{जड़ें} ठंडी कर देने
या श्वसन के किसी \omterm{def:b1:enzymes:inhibitors}{संदमक} से विषाक्त कर देने पर उसका बहाव रुक जाता है: बहाव
को सक्रिय आयन-लदान चलाता है, जल का कोई पंपन नहीं।
\end{proof}

\section{जल: मिट्टी से जाइलम तक}

\begin{proposition}[जल-विभव की सीढ़ी]\label{prop:b1:plant-water-minerals:ladder}
\index{जल विभव}
जल मिट्टी से \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} तक इसलिए चलता है कि जल-विभव
(\cref{ch:b1:membranes-transport}) हर चरण पर गिरता है:
\begin{center}
\small
\begin{tabular}{lr}
\hline
कक्ष & $\Psi$ (\unit{MPa}) \\
\hline
नम मिट्टी & $-0.03$ से $-0.3$ \\
म्लानि-बिंदु पर मिट्टी & $-1.5$ \\
\omterm{def:b1:flowering-plant-organization:organs}{जड़} की वल्कुट \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} & $-0.3$ से $-0.6$ \\
\omterm{def:b1:flowering-plant-organization:organs}{जड़} का \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} & $-0.5$ से $-0.8$ \\
\omterm{def:b1:flowering-plant-organization:organs}{पत्ती} की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} & $-1.0$ से $-2.0$ \\
\qty{50}{\%} आपेक्षिक आर्द्रता पर हवा & $-95$ \\
\hline
\end{tabular}
\end{center}
मिट्टी का विभव अधिकतर उस तनाव से तय होता है जिससे उसके छिद्र जल पकड़े
रहते हैं (केशिकत्व); \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} का उनके विलेयों और उनकी \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} से;
\omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} का वाष्पोत्सर्जन करते स्तंभ के तनाव से; हवा का उसकी आर्द्रता से,
$\Psi = (RT/V_w)\ln(\text{RH})$, जो नम हवा में भी बहुत बड़ा होता है।
कहीं अधिक तीखा चरण अंतिम वाला है, \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} से हवा तक: वहीं लगभग सारा
चालक बल खर्च होता है, और इसीलिए पौधे उसे \omterm{def:b1:flowering-plant-organization:tissues}{रंध्रों} से नियंत्रित करते
हैं।
\end{proposition}

\begin{method}[किसी जल-विभव परिच्छेदिका को पढ़ना]\label{met:b1:plant-water-minerals:profile}
\begin{enumerate}
  \item कक्षों को क्रम से और उनका $\Psi$ गिनाइए; जल केवल ऊँचे से नीचे
        बहता है, इसलिए क्रम को रास्ते भर घटना ही चाहिए वरना बहाव उलट
        जाता है।
  \item हर \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के $\Psi$ को $\Psi_s$ (विलेय, ऋणात्मक) और
        $\Psi_p$ (\omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति}, धनात्मक) में बाँटिए; $\Psi = -0.4$ वाली कोई
        जड़-कोशिका जिसका $\Psi_s = -0.9$ हो उसका $\Psi_p = +0.5$ होता
        है। \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} के जल का $\Psi_s \approx 0$ और $\Psi_p$ ऋणात्मक होता
        है: वह तनाव में है।
  \item किसी चरण के आरपार बहाव $\Psi$ के अंतर गुणा उस रोक की
        जल-चालकता है; \omterm{prop:b1:flowering-plant-organization:stemroot}{अंतस्त्वचा} और रंध्र नीची चालकता वाले वे दो चरण
        हैं जहाँ पौधा नियमन करता है।
  \item मिट्टी सुखाइए और उसका $\Psi$ गिरता है; जब वह \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} वाले तक आ
        जाए, बहाव रुक जाता है और पौधा मुरझा जाता है; जब \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} की
        \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} का $\Psi_p$ शून्य पर आ जाए तो \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} लटक जाती है।
        उबरने के लिए मिट्टी को बदलना पड़ता है, पौधे को नहीं।
\end{enumerate}
\end{method}

\begin{example}[दिन में एक लीटर]\label{ex:b1:plant-water-minerals:litre}
मक्का के किसी पौधे की महीन \omterm{def:b1:flowering-plant-organization:organs}{जड़ें} लगभग \qty{0.2}{m^2} पृष्ठ देती हैं
जिसकी जल-चालकता करीब \qty{2e-7}{m.s^{-1}.MPa^{-1}} है; \qty{-0.2}{MPa}
पर मिट्टी और \qty{-0.6}{MPa} पर \omterm{def:b1:flowering-plant-organization:organs}{जड़} के \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} के बीच अभिवाह
$2\times 10^{-7}\times 0.4\times 0.2 = \qty{1.6e-8}{m^3/s}$ है --- दिन
में डेढ़ लीटर, और वह सारा \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} से खो जाता है। \omterm{def:b1:flowering-plant-organization:organs}{जड़} उसे चलाने में कोई
काम नहीं करती: \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} का वाष्पोत्सर्जन तनाव तय करता है, और मिट्टी तब तक
जल देती रहती है जब तक उसका विभव \omterm{def:b1:flowering-plant-organization:organs}{जड़} के विभव से ऊपर रहे।
\end{example}

\section{खनिज पोषण}

\begin{definition}[आवश्यक तत्व]\label{def:b1:plant-water-minerals:elements}
\index{आवश्यक तत्व}\index{दीर्घपोषक}\index{लघुपोषक}
कोई तत्व किसी पौधे के लिए \emph{आवश्यक} है यदि पौधा उसके बिना अपना
जीवनचक्र पूरा न कर सके और कोई दूसरा तत्व उसकी जगह न ले सके। ऐसे सत्रह
हैं: हवा और जल से कार्बन, हाइड्रोजन तथा ऑक्सीजन; मिट्टी से शुष्क
द्रव्यमान के प्रति किलोग्राम ग्राम में लिए जाने वाले \emph{दीर्घपोषक} ---
नाइट्रोजन (नाइट्रेट या अमोनियम के रूप में), फ़ॉस्फ़ोरस (फ़ॉस्फ़ेट),
पोटैशियम, कैल्सियम, मैग्नीशियम, सल्फ़र (सल्फ़ेट); और मिलीग्राम या उससे
कम में चाहिए \emph{लघुपोषक} --- लोहा, मैंगनीज़, ज़िंक, ताम्र, बोरॉन,
मॉलिब्डेनम, क्लोरीन, निकल। नाइट्रोजन \omterm{def:b1:proteins:peptide}{प्रोटीन} और न्यूक्लीक अम्ल गढ़ती है;
फ़ॉस्फ़ोरस, \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} और झिल्लियाँ; पोटैशियम \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} का मुख्य धनायन
और \omterm{def:b1:flowering-plant-organization:tissues}{रंध्रों} का \omterm{def:b1:membranes-transport:osmosis}{परासरणी} कारक है; मैग्नीशियम \omterm{def:b1:photosynthesis:pigments}{पर्णहरित} में बैठा है; लोहा
साइटोक्रोमों और फ़ेरेडॉक्सिन में; मॉलिब्डेनम \omterm{prop:b1:plant-water-minerals:nitrate}{नाइट्रेट रिडक्टेज़} और
नाइट्रोजिनेज़ में।
\end{definition}

\begin{proposition}[यह सूची कैसे बनी]\label{prop:b1:plant-water-minerals:hydroponics}
\omterm{def:b1:plant-water-minerals:elements}{आवश्यक तत्व} पौधों को केवल ज्ञात लवण रखने वाले जल में उगाकर पहचाने
गए।
\end{proposition}
\begin{proof}[साक्ष्य]
ज़ाक्स और क्नोप (1860 के दशक) ने पौधों को बिना किसी मिट्टी के कुछ खनिज
लवणों के विलयनों में परिपक्वता तक उगाया, और यह सिद्ध किया कि मिट्टी
खनिजों और आधार के सिवा कुछ नहीं देती; एक-एक करके एक लवण छोड़ देने से हर
तत्व के लिए एक विशिष्ट रुग्णता मिली, और उसे वापस डालने से वह ठीक हो गई।
\omterm{def:b1:plant-water-minerals:elements}{लघुपोषक} बाद में मिले, जैसे-जैसे रसायनज्ञ लवण शुद्ध करना सीखे: एक करोड़
में एक भाग जितना चाहिए मॉलिब्डेनम से वंचित पौधा नाइट्रेट अपचयित नहीं कर
पाता। वह तकनीक, जलकृषि, अब हरितगृहों को पालती है।
\end{proof}

\includegraphics[width=0.6\linewidth]{images/book3/ai/b3-23-nutrient-deficiency.jpg}

{\small पोषक विलयनों में तीन टमाटर के पौधे: पूरा (बाएँ), नाइट्रोजन के
बिना (बीच में: बौना, पुरानी \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} पीली क्योंकि उनकी नाइट्रोजन नई
\omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} को भेज दी गई), फ़ॉस्फ़ोरस के बिना (दाएँ: गहरी \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ}, बैंगनी
आभा लिए)। हर लुप्त तत्व के अपने चिह्न हैं।}
\end{omfigure}

\begin{method}[किसी न्यूनता को पढ़ना]\label{met:b1:plant-water-minerals:deficiency}
\begin{enumerate}
  \item पूछिए कि लक्षण कहाँ प्रकट होता है। जिन तत्वों को पौधा फ्लोएम
        में चला सकता है (N, P, K, Mg) वे नई \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} को पालने के लिए
        पुरानी से खींच लिए जाते हैं: न्यूनता पहले पुरानी \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} पर
        दिखती है। जिन्हें वह नहीं चला सकता (Ca, Fe, B) वे वहीं फँसे रह
        जाते हैं जहाँ रखे गए थे: वह नई \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} और बढ़ते शीर्षों पर
        दिखती है।
  \item पूछिए कि वह तत्व करता क्या है: नाइट्रोजन नहीं, तो \omterm{def:b1:photosynthesis:pigments}{पर्णहरित} और
        \omterm{def:b1:proteins:peptide}{प्रोटीन} कम --- फीका, बौना; मैग्नीशियम या लोहा नहीं, तो \omterm{def:b1:photosynthesis:pigments}{पर्णहरित}
        विफल --- हरी शिराओं वाली पीली \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ}; पोटैशियम नहीं, तो \omterm{def:b1:flowering-plant-organization:organs}{पत्ती}
        के किनारे झुलसते हैं; कैल्सियम नहीं, तो बढ़ते शीर्ष मर जाते हैं।
  \item संदिग्ध तत्व अकेला देकर और नई वृद्धि देखकर पुष्टि कीजिए।
\end{enumerate}
\end{method}

\begin{proposition}[आयनों का सक्रिय ग्रहण]\label{prop:b1:plant-water-minerals:uptake}
\index{प्रोटॉन पंप}
मिट्टी का विलयन आयनों को माइक्रोमोल से लेकर कुछ मिलीमोल प्रति लीटर पर
रखता है; \omterm{def:b1:flowering-plant-organization:organs}{जड़} की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} पोटैशियम को \qty{100}{mmol/L} पर, फ़ॉस्फ़ेट को
\qty{10}{} पर, नाइट्रेट को कई पर रखती हैं। इसलिए ग्रहण अधिकतर चढ़ाई का
होता है और \omterm{def:b1:water-small-molecules:atp}{ATP} से चुकाया जाता है, पर परोक्ष रूप से: जीवद्रव्य-झिल्ली में
कोई \emph{\omterm{prop:b1:plant-water-minerals:uptake}{प्रोटॉन पंप}} (\cref{ch:b1:membranes-transport})
$\mathrm{H^+}$ बाहर करता है, जिससे बाहर अम्लीय और भीतर
\qtyrange{-120}{-200}{mV} ऋणात्मक हो जाता है; तब $\mathrm{K^+}$ जैसे
धनायन विद्युत प्रवणता के नीचे \omterm{def:b1:membranes-transport:transporters}{चैनलों} से घुसते हैं, और ऋणायन
($\mathrm{NO_3^-}$, $\mathrm{H_2PO_4^-}$, $\mathrm{SO_4^{2-}}$) ऐसे
\omterm{prop:b1:membranes-transport:secondary}{सहवाहकों} से जो प्रोटॉनों को वापस भीतर ले आते हैं। वाहक चयनशील हैं:
पोटैशियम उतने ही आवेश और लगभग उतने ही आकार के सोडियम से सौ गुना अधिक
सहजता से लिया जाता है, और पौधा नमकीन मिट्टी में भी सोडियम बाहर रखता है।
\omterm{def:b1:flowering-plant-organization:organs}{जड़ें} इस परिवहन पर पौधे के श्वसन का एक तिहाई खर्च करती हैं।
\end{proposition}

\section{नाइट्रोजन}

\begin{proposition}[नाइट्रेट से ऐमीनो अम्ल तक]\label{prop:b1:plant-water-minerals:nitrate}
\index{नाइट्रेट रिडक्टेज़}\index{नाइट्रोजन स्वांगीकरण}
अधिकांश पौधे नाइट्रोजन नाइट्रेट के रूप में लेते हैं। \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाद्रव्य} में
\emph{\omterm{prop:b1:plant-water-minerals:nitrate}{नाइट्रेट रिडक्टेज़}}, एक मॉलिब्डेनम \omterm{def:b1:enzymes:enzyme}{एंजाइम}, NADH से मिले
इलेक्ट्रॉनों से उसे नाइट्राइट तक अपचयित करती है; \omterm{def:b1:eukaryotic-cell:plastid}{लवक} में \emph{नाइट्राइट
रिडक्टेज़} फ़ेरेडॉक्सिन से मिले इलेक्ट्रॉनों से नाइट्राइट को अमोनियम तक
अपचयित करती है; और अमोनियम तत्काल ग्लूटामीन तथा \omterm{def:b1:biosyntheses-integration:nitrogen}{ग्लूटामेट} में बाँध दिया
जाता है (\cref{ch:b1:biosyntheses-integration}) --- अमोनियम स्वयं विषैला
है और कभी जमा नहीं होता। पूरे अपचयन में प्रति नाइट्रोजन आठ इलेक्ट्रॉन
लगते हैं, नाइट्रेट-धनी मिट्टी पर किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} के प्रकाश-संश्लेषी
इलेक्ट्रॉन-बहाव का दसवाँ भाग, और \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} उसका बहुत कुछ उजाले में, सीधे
\omterm{def:b1:eukaryotic-cell:plastid}{हरितलवक} के फ़ेरेडॉक्सिन से करती है। जहाँ मिट्टी अमोनियम देती है, वह सीधे
लिया जाता है और यह खर्च बच जाता है।
\end{proposition}

\begin{definition}[नाइट्रोजन स्थिरीकरण और ग्रंथिकाएँ]\label{def:b1:plant-water-minerals:nodules}
\index{नाइट्रोजन स्थिरीकरण}\index{मूल ग्रंथिका}\index{Rhizobium}
कोई भी \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकी} हवा की $\mathrm{N_2}$ नहीं बरत सकता। कुछ बैक्टीरिया
बरत सकते हैं: \emph{नाइट्रोजिनेज़} $\mathrm{N_2}$ का त्रिबंध तोड़ती है
और उसे प्रति अणु सोलह \omterm{def:b1:water-small-molecules:atp}{ATP} तथा आठ इलेक्ट्रॉन के मोल पर दो अमोनिया तक
अपचयित करती है, और केवल वहीं जहाँ ऑक्सीजन बाहर रखी गई हो। शिंबी
(मटर, सेम, तिपतिया) ऐसे बैक्टीरिया (\emph{Rhizobium}) को \emph{मूल
ग्रंथिकाओं} में बसाते हैं: पौधा एक कक्ष गढ़ता है, बैक्टीरिया को शर्करा
खिलाता है, और ऑक्सीजन को लेगहीमोग्लोबिन नामक एक लाल \omterm{def:b1:proteins:peptide}{प्रोटीन} से नीची रखता
है, जो उसे बैक्टीरिया के श्वसन तक इतनी नीची सांद्रता पर पहुँचाता है कि
\omterm{def:b1:enzymes:enzyme}{एंजाइम} को हानि न हो; बैक्टीरिया बदले में अमोनियम देते हैं, जिसे पौधा
\omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} में बदल लेता है। तिपतिया का एक खेत साल में प्रति हेक्टेयर सौ
किलोग्राम नाइट्रोजन स्थिर करता है, जिसके लिए पौधा अपने प्रकाश-संश्लेषण का
दसवाँ भाग चुकाता है। समूचा नाइट्रोजन चक्र वर्ष 2 के खंड का विषय है।
\end{definition}

\begin{omfigure}
\includegraphics[width=0.52\linewidth]{images/book3/ai/b3-23-root-nodules.jpg}

{\small किसी शिंबी की \omterm{def:b1:flowering-plant-organization:organs}{जड़ों} पर ग्रंथिकाएँ: हर एक पादप-ऊतक का ऐसा कक्ष है
जो बैक्टीरिया से भरा और लेगहीमोग्लोबिन से गुलाबी है, जहाँ वायुमंडलीय
नाइट्रोजन स्थिर की जाती है।}
\end{omfigure}

\begin{example}[नाइट्रोजन का मोल]\label{ex:b1:plant-water-minerals:priceN}
मक्का के किसी पौधे को दिन में करीब \qty{30}{mg} नाइट्रोजन चाहिए;
\qty{2}{mmol/L} पर नाइट्रेट से वह उतनी उस एक लीटर जल में पा लेता है जो
वह वाष्पोत्सर्जित करता है, और उसे अपचयित करने में कुछ सौ मिलीमोल
इलेक्ट्रॉन खर्च करता है। उतनी ही मात्रा स्थिर करता कोई तिपतिया का पौधा
प्रति $\mathrm{N_2}$ \qty{16}{} \omterm{def:b1:water-small-molecules:atp}{ATP} जोड़ अपचायक चुकाता है --- मोटे तौर
पर प्रति ग्राम नाइट्रोजन \qty{6}{g} शर्करा, यानी दिन में लगभग
\qty{0.2}{g}, उसके उत्पादन का दसवाँ भाग --- और बिलकुल नाइट्रेट-रहित
मिट्टी पर भी उग सकता है। किसानों ने उस नाइट्रोजन को एक खेत से दूसरे तक
ले जाने के लिए दो हज़ार साल से शिंबियों को अनाजों के साथ बदल-बदलकर बोया
है।
\end{example}

\section{मिट्टी में साझेदार}

\begin{definition}[कवकमूल]\label{def:b1:plant-water-minerals:mycorrhiza}
\index{कवकमूल}
\emph{कवकमूल} किसी \omterm{def:b1:flowering-plant-organization:organs}{जड़} और किसी कवक के बीच का साहचर्य है, जो दस में से नौ
पौधों में मिलता है। सबसे आम रूप में कवक-तंतु वल्कुट की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में
बढ़ते हैं और उनके भीतर \emph{अर्बुस्कुल} में शाखित हो जाते हैं, जो
विनिमय-पृष्ठ है, जबकि \omterm{def:b1:flowering-plant-organization:organs}{जड़} के बाहर कुछ माइक्रोमीटर चौड़े तंतुओं का एक जाल
मीटरों तक मिट्टी टटोलता है। कवक फ़ॉस्फ़ेट तथा दूसरे कम गतिशील आयन, और
जल, वहाँ तक पहुँचाता है जहाँ \omterm{def:b1:flowering-plant-organization:organs}{जड़} के अपने रोम न पहुँच पाते; पौधा शर्करा
देता है, अपने प्रकाश-संश्लेषण का दसवाँ भाग या उससे अधिक। शीतोष्ण वनों के
वृक्ष एक दूसरा रूप ढोते हैं, जिसमें कवक \omterm{def:b1:flowering-plant-organization:organs}{जड़} के सिरों पर खोल चढ़ाता है और
\omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} के बीच धागे डालता है; उसके फलनकाय ही वन-तल के छत्रक हैं।
\end{definition}
\begin{proof}[साक्ष्य]
निर्जीवाणुकृत मिट्टी में उगाए गए अंकुर खराब उगते हैं और कम फ़ॉस्फ़ेट
लेते हैं; कवक के बीजाणुओं से टीका लगाने पर वे कई गुना बड़े उगते हैं।
\omterm{def:b1:flowering-plant-organization:organs}{जड़ों} की पहुँच से परे मिट्टी में रखा रेडियोसक्रिय फ़ॉस्फ़ेट पौधे में तभी
प्रकट होता है जब तंतु दोनों को जोड़ें, और \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} को खिलाया अंकित
कार्बन एक दिन के भीतर तंतुओं में प्रकट हो जाता है। विलेय तो जाने दे पर
कवक न जाने दे, ऐसी महीन जाली से तंतु काट देने पर यह लाभ मिट जाता है।
\end{proof}

\begin{example}[फ़ॉस्फ़ेट को कवक क्यों चाहिए]\label{ex:b1:plant-water-minerals:phosphate}
फ़ॉस्फ़ेट मिट्टी के कणों से बँध जाता है और दिन में एक मिलीमीटर विसरित
होता है; कोई \omterm{def:b1:flowering-plant-organization:organs}{मूलरोम} अपनी पहुँच का सारा फ़ॉस्फ़ेट घंटों में ले लेता है और
फिर प्रतीक्षा करता है। किसी \omterm{def:b1:flowering-plant-organization:organs}{मूलरोम} से दसवें भाग जितने मोटे तंतु प्रति

हज़ारवाँ भाग पड़ता है जो एक मीटर \omterm{def:b1:flowering-plant-organization:organs}{जड़} पड़ती है। नाइट्रेट के लिए, जो

नाइट्रेट-धनी मिट्टियों पर पौधे अपना कवक घटा देते हैं; फ़ॉस्फ़ेट के लिए
कवक ही \omterm{def:b1:flowering-plant-organization:organs}{जड़} का विस्तार है।
\end{example}

\section{समस्या: मक्का की किसी जड़ की जल-विभव सीढ़ी}

\begin{problem}[{सप्ताहांत समस्या --- जुलाई की मिट्टी से मक्का के किसी
पौधे में जल और नाइट्रेट: हर चरण पर विभव, जड़ों से होकर अभिवाह, वह
नाइट्रोजन जो वह ढोता है और उसे अपचयित करने का खर्च, और अंत में जड़ में
दैनिक जल-अभिवाह}]
\label{pb:b1:plant-water-minerals:1}
जुलाई में मक्का का एक पौधा: महीन \omterm{def:b1:flowering-plant-organization:organs}{जड़ों} का पृष्ठ \qty{0.2}{m^2}, \omterm{def:b1:flowering-plant-organization:organs}{जड़} की
जल-चालकता \qty{2e-7}{m.s^{-1}.MPa^{-1}}; पर्ण-क्षेत्रफल
\qty{0.5}{m^2}। मिट्टी का जल-विभव \qty{-0.2}{MPa}; \omterm{def:b1:flowering-plant-organization:organs}{जड़} की वल्कुट
\omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} का $\Psi_s = \qty{-0.9}{MPa}$ और $\Psi_p = \qty{0.5}{MPa}$; \omterm{def:b1:flowering-plant-organization:organs}{जड़}
का \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} \qty{-0.6}{MPa}; \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} \qty{-1.3}{MPa}; \qty{50}{\%}
आपेक्षिक आर्द्रता पर हवा ($RT/V_w = \qty{137}{MPa}$)। मिट्टी में
नाइट्रेट \qty{2}{mmol/L}; पौधे को दिन में \qty{30}{mg} नाइट्रोजन चाहिए।
नाइट्रेट का अमोनियम तक अपचयन प्रति नाइट्रोजन 8 इलेक्ट्रॉन बरतता है;
प्रकाश-संश्लेषण प्रति $\mathrm{CO_2}$ 4 इलेक्ट्रॉन चलाता है और \omterm{def:b1:flowering-plant-organization:organs}{पत्ती}
\qty{12}{h} तक प्रति वर्ग मीटर प्रति सेकंड \qty{4}{\micro mol}
$\mathrm{CO_2}$ बाँधती है।

\medskip
\noindent\textbf{भाग I --- सीढ़ी।}
\begin{enumerate}
  \item \omterm{def:b1:flowering-plant-organization:organs}{जड़} की वल्कुट \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} का जल-विभव निकालिए।
  \item मिट्टी, वल्कुट, \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम}, \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} का क्रम लिखिए, और सत्यापित कीजिए
        कि हर चरण पर जल भीतर और ऊपर बहता है।
  \item हवा का जल-विभव निकालिए।
  \item मिट्टी से \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} तक की गिरावट और \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} से हवा तक की गिरावट
        निकालिए; कुल का कितना अंश अंतिम चरण है?
  \item मिट्टी सूखकर \qty{-0.9}{MPa} पर आ जाती है। वल्कुट \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में
        बहाव का, और उनकी \omterm{def:b1:eukaryotic-cell:plantcell}{स्फीति} का क्या होता है?
  \item मिट्टी सूखकर \qty{-1.5}{MPa} पर आ जाती है। \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} का क्या
        होता है, और पानी देने से वे फिर क्यों उबर आती हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- अभिवाह।}
\begin{enumerate}[resume]
  \item \omterm{def:b1:flowering-plant-organization:organs}{जड़ों} में जल-अभिवाह निकालिए (चालकता $\times$
        पृष्ठ $\times$ मिट्टी और \omterm{def:b1:flowering-plant-organization:organs}{जड़} के \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} का अंतर)।
  \item उसे लीटर प्रति दिन में बदलिए।
  \item वह सारा वाष्पोत्सर्जन से खो जाता है। प्रति वर्ग मीटर \omterm{def:b1:flowering-plant-organization:organs}{पत्ती}
        प्रति घंटा वाष्पोत्सर्जन ग्राम में निकालिए।
  \item प्रति वर्ग मीटर \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} प्रति सेकंड वाष्पोत्सर्जन को जल के
        मिलीमोल में बताइए (\qty{18}{g/mol})।
  \item दोपहर में रंध्र बंद हो जाते हैं और \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} का $\Psi$ चढ़कर
        \qty{-0.9}{MPa} हो जाता है जबकि \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} का चढ़कर \qty{-0.4}{MPa}
        हो जाता है। अभिवाह फिर से निकालिए। पौधे ने क्या किया, और क्यों?
  \item \omterm{def:b1:flowering-plant-organization:organs}{जड़} जल लेने में ऊर्जा क्यों नहीं खर्च करती, जबकि नाइट्रेट लेने
        में उसे ऊर्जा खर्च करनी ही पड़ती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- नाइट्रोजन।}
\begin{enumerate}[resume]
  \item प्रश्न 8 के जल-अभिवाह से \omterm{def:b1:flowering-plant-organization:organs}{जड़} में पहुँचा नाइट्रेट मिलीमोल में और
        नाइट्रोजन के मिलीग्राम में निकालिए।
  \item क्या वह पौधे की ज़रूरत पूरी करता है? शेष का क्या होता है?
  \item \omterm{def:b1:flowering-plant-organization:organs}{जड़} अपनी \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में नाइट्रेट को \qty{5}{mmol/L} तक जमा करती
        है। \qty{-150}{mV} के झिल्ली-विभव के विरुद्ध बाहर
        \qty{2}{mmol/L} से भीतर \qty{5}{mmol/L} तक एक मोल चलाने की
        \omterm{prop:b1:water-small-molecules:gibbs}{मुक्त ऊर्जा} निकालिए ($RT = \qty{2.48}{kJ/mol}$; कोई ऋणायन किसी
        ऋणात्मक \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में घुसता हुआ)।
  \item यह ग्रहण दो प्रोटॉनों के साथ एक सहवहन है। pH 5.5 से pH 7.5 तक
        \qty{-150}{mV} पर किसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में घुसते दो प्रोटॉन जितनी \omterm{prop:b1:water-small-molecules:gibbs}{मुक्त
        ऊर्जा} छोड़ते हैं वह निकालिए, और जाँचिए कि वह बहुत है।
  \item \qty{30}{mg} नाइट्रोजन को अमोनियम तक अपचयित करने के लिए प्रति
        दिन चाहिए इलेक्ट्रॉन निकालिए।
  \item \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} अपने प्रकाश-निकायों से प्रति दिन जितने इलेक्ट्रॉन चलाती
        है वे, और नाइट्रेट की ओर मोड़ा गया अंश निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- विकल्प।}
\begin{enumerate}[resume]
  \item यदि मिट्टी नाइट्रेट के बजाय अमोनियम देती, तो पौधा क्या बचाता,
        और अमोनियम-ग्रहण उसके pH संतुलन के लिए कौन-सी समस्या खड़ी
        करता?
  \item कोई तिपतिया उतनी ही \qty{30}{mg} नाइट्रोजन दिन में स्थिर करता
        है। प्रति $\mathrm{N_2}$ \qty{16}{} \omterm{def:b1:water-small-molecules:atp}{ATP} और \qty{8}{} इलेक्ट्रॉन
        पर, नाइट्रोजिनेज़ का प्रति दिन खर्च किया \omterm{def:b1:water-small-molecules:atp}{ATP} निकालिए।
  \item कुल मिलाकर प्रति ग्राम स्थिर नाइट्रोजन \qty{6}{g} शर्करा पर,
        तिपतिया का दैनिक शर्करा-खर्च निकालिए और उसकी तुलना प्रति दिन
        \qty{3}{g} शर्करा के उत्पादन से कीजिए। \qty{2}{mmol/L}
        नाइट्रेट वाली मिट्टी में क्या यह सौदा फ़ायदे का है? और
        नाइट्रेट-रहित मिट्टी में?
  \item मिट्टी के विलयन में फ़ॉस्फ़ेट \qty{2}{\micro mol/L} पर है।
        प्रश्न 8 का जल-अभिवाह जितना फ़ॉस्फ़ेट भीतर लाता है वह
        निकालिए, और प्रति दिन \qty{4}{mg} फ़ॉस्फ़ोरस की ज़रूरत से
        तुलना कीजिए (\qty{31}{g/mol})। पौधे को किस पर निर्भर रहना ही
        पड़ेगा?
  \item दो वाक्यों में समझाइए कि \omterm{def:b1:plant-water-minerals:pathways}{कैस्पेरी पट्टी} \omterm{def:b1:flowering-plant-organization:organs}{जड़} को यह कैसे करने
        देती है कि मिट्टी \qty{2}{} पर हो तब भी वह अपने \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} में
        नाइट्रेट \qty{5}{mmol/L} पर रखे।
  \item किसी उत्परिवर्ती में \omterm{def:b1:plant-water-minerals:pathways}{कैस्पेरी पट्टी} नहीं है। उसके जाइलम-रस के
        संघटन और नमकीन मिट्टी में उसके व्यवहार की भविष्यवाणी कीजिए।
  \item परिणाम बताइए: मक्का की \omterm{def:b1:flowering-plant-organization:organs}{जड़} में प्रति दिन जल-अभिवाह, वह
        नाइट्रोजन जो वह पहुँचाता है, और \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} के इलेक्ट्रॉनों का वह अंश
        जो उसे अपचयित करने में खर्च होता है।
\end{enumerate}
\end{problem}
