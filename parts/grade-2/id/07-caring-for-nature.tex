\chapter{Merawat Alam}\label{ch:g2:caring-for-nature}

Sesudah piknik, satu keluarga meninggalkan padang rumput persis
seperti keadaan yang ditemuinya; keluarga yang lain meninggalkan
kantong, tutup botol dan sebuah sarang semut yang terinjak-injak.
Kelinci, semut dan \omterm{def:g1:plants-around-us:parts}{bunga} aster di padang rumput itu tidak dapat
meminta kita berhati-hati --- tetapi mereka \omterm{def:g1:living-or-not:living}{makhluk hidup} yang berbagi
tempat dengan kita, dan cara kita berperilaku di \omterm{def:g2:caring-for-nature:environment}{alam} berarti bagi
mereka. Merawat \omterm{def:g2:caring-for-nature:environment}{alam} dapat dipelajari siapa pun, pada umur berapa pun.

\section{Alam adalah tempat makhluk hidup tinggal}

\begin{definition}[Alam dan lingkungan]\label{def:g2:caring-for-nature:environment}
Yang disebut \emph{alam}\index{alam} adalah dunia \omterm{def:g1:living-or-not:living}{makhluk hidup}
beserta sekelilingnya: padang rumput, hutan, kolam, laut --- serta
udara, air dan tanah yang menjadi sandarannya. Sekeliling yang menjadi
sandaran sebuah \omterm{def:g1:living-or-not:living}{makhluk hidup} disebut
\emph{lingkungan}\index{lingkungan}nya.
\end{definition}

\begin{example}[Segalanya adalah rumah seseorang]\label{ex:g2:caring-for-nature:home}
Sebatang kayu lapuk di hutan tampak seperti sampah --- padahal ia
sebuah istana: kumbang di bawah kulit kayunya, kutu kayu di bawahnya,
lumut di atasnya, seekor landak yang melewati musim dingin di
rongganya. Tendang ia berkeping-keping untuk bersenang-senang, dan
empat rumah tangga kehilangan rumahnya. Di \omterm{def:g2:caring-for-nature:environment}{alam}, hampir segalanya
adalah rumah, makanan, atau perlindungan bagi seseorang.
\end{example}

\begin{remark}[Kita bagian darinya]\label{rem:g2:caring-for-nature:part}
Manusia juga \omterm{def:g1:living-or-not:living}{makhluk hidup}: udara yang kita hirup, air yang kita minum
dan makanan yang kita makan semuanya datang dari \omterm{def:g2:caring-for-nature:environment}{alam}. Merawatnya
bukan sekadar berbaik hati kepada kelinci --- ia berarti menjaga dunia
yang darinya kita sendiri hidup, seperti dikatakan
\cref{ch:g2:where-animals-live} tentang setiap \omterm{def:g2:where-animals-live:habitat}{habitat}.
\end{remark}

\section{Apa yang merusak, apa yang menolong}

\begin{proposition}[Tindakan kecil, kerusakan nyata]\label{prop:g2:caring-for-nature:harm}
Di \omterm{def:g2:caring-for-nature:environment}{alam}, kerusakan jarang datang dari keinginan berbuat jahat --- ia
datang dari tidak berpikir:
\begin{itemize}
  \item sampah bertahan bertahun-tahun dan dapat menjerat atau
        meracuni \omterm{def:g1:animals-around-us:animal}{hewan};
  \item memetik setiap \omterm{def:g1:plants-around-us:parts}{bunga} menyisakan nol \omterm{def:g1:plants-around-us:parts}{bunga} yang membuat \omterm{def:g2:from-seed-to-plant:seed}{biji}
        --- dan nol \omterm{def:g1:plants-around-us:parts}{bunga} bagi lebah;
  \item teriakan keras dan kejar-kejaran menakuti \omterm{def:g1:animals-around-us:animal}{hewan} sehingga
        menjauh dari makanan dan anaknya;
  \item tepian yang terinjak, batu yang terbalik dan dahan yang patah
        masing-masing membatalkan sebuah rumah kecil.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Tutup botolnya]\label{ex:g2:caring-for-nature:cap}
Sebuah tutup plastik yang terjatuh tidak berarti bagi kita dan
bertahan di padang rumput selama bertahun-tahun yang amat panjang ---
cukup panjang bagi orang dewasa untuk punya anak yang menemukan tutup
yang sama. \omterm{def:g2:caring-for-nature:environment}{Alam} tidak punya truk sampah: apa pun yang kita
tinggalkan, tetap tinggal.
\end{example}

\begin{method}[Aturan bagi pengunjung]\label{met:g2:caring-for-nature:rules}
Ketika berada di \omterm{def:g2:caring-for-nature:environment}{alam}:
\begin{enumerate}
  \item bawa pulang semua sampahmu --- semuanya;
  \item amati \omterm{def:g1:plants-around-us:parts}{bunga}, serangga dan sarang alih-alih mengumpulkannya;
        petiklah paling banyak beberapa \omterm{def:g1:plants-around-us:parts}{bunga} yang biasa, jangan
        pernah seluruh rumpunnya;
  \item kembalikan setiap batu dan batang kayu yang kamu balikkan ---
        ada yang tinggal di bawahnya;
  \item tetaplah di jalan setapak di dekat tempat yang rapuh seperti
        tepian sungai;
  \item amati \omterm{def:g1:animals-around-us:animal}{hewan} dari jauh, dengan tenang, dan biarkan anaknya
        tidak terganggu.
\end{enumerate}
Tinggalkan tempatnya sehingga tidak seorang pun dapat menyadari bahwa
kamu pernah di situ.
\end{method}

\begin{example}[Menolong dengan sengaja]\label{ex:g2:caring-for-nature:help}
Merawat dapat melampaui sekadar tidak merusak. Sepiring kecil air pada
gelombang panas, satu sudut kebun yang liar dibiarkan tak dipangkas,
\omterm{def:g2:from-seed-to-plant:seed}{biji} untuk burung pada musim dingin, sebuah kolam sekolah seperti
kolam pada \cref{ex:g2:where-animals-live:schoolpond}: masing-masing
memberi sebagian \omterm{def:g1:living-or-not:living}{makhluk hidup} makanan, air atau rumah yang telah
hilang dari mereka.
\end{example}

\section{Memilah dan menghemat}

\begin{proposition}[Sampah dapat dikembalikan]\label{prop:g2:caring-for-nature:recycle}
Banyak dari apa yang kita buang dapat dipakai lagi.
\emph{Pemilahan}\index{pemilahan sampah} sampah --- kaca dengan kaca,
kertas dengan kertas, kulit sayuran ke komposnya --- membuat botol
menjadi botol baru dan kulit sayuran menjadi tanah, alih-alih semuanya
menumpuk. Dan apa yang \emph{dihemat} --- air, kertas, listrik ---
sama sekali tidak pernah menjadi sampah.
\end{proposition}
\admitted

\begin{example}[Tempat kompos]\label{ex:g2:caring-for-nature:compost}
Bonggol apel dan kulit wortel, di dalam tempat kompos, perlahan
berubah menjadi tanah gelap yang gembur --- dengan pertolongan cacing,
kutu kayu dan \omterm{def:g1:living-or-not:living}{makhluk hidup} yang terlalu kecil untuk dilihat. Musim
semi berikutnya, tanah itu memberi makan tomatnya. Tidak ada yang
terbuang; kulit sayurannya kembali ke daur kehidupan.
\end{example}

\begin{remark}[Tangan kecil ikut berarti]\label{rem:g2:caring-for-nature:count}
Mematikan keran sambil menyikat gigi, memakai kedua sisi kertas,
tempat komposnya, sampah yang dibawa pulang: masing-masing kecil, dan
jutaan orang melakukannya --- atau tidak. \omterm{def:g2:caring-for-nature:environment}{Alam} dirawat, atau dirusak,
sebagian besar melalui pilihan kecil sehari-hari seperti pilihanmu.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:1}
Apa itu \omterm{def:g2:caring-for-nature:environment}{lingkungan} sebuah \omterm{def:g1:living-or-not:living}{makhluk hidup}? Sebutkan tiga hal di
lingkunganmu yang datang dari \omterm{def:g2:caring-for-nature:environment}{alam}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:2}
Siapa saja yang dapat kehilangan rumahnya ketika sebatang kayu lapuk
ditendang berkeping-keping? Sebutkan tiga penghuni dari
\cref{ex:g2:caring-for-nature:home}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:3}
Sebutkan empat aturan pengunjung pada \cref{met:g2:caring-for-nature:rules}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:4}
Mengapa sampah harus pulang bersama kita? Jawablah dengan tutup botol
pada \cref{ex:g2:caring-for-nature:cap}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:5}
Mengapa kita tidak boleh memetik seluruh \omterm{def:g1:plants-around-us:parts}{bunga} sebuah rumpun? Berikan
kedua alasan pada \cref{prop:g2:caring-for-nature:harm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:6}
Apa yang terjadi pada bonggol apel di dalam tempat kompos? Siapa yang
mengerjakannya, dan apa yang kita peroleh pada akhirnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:7}
Sebutkan dua cara menolong \omterm{def:g2:caring-for-nature:environment}{alam} dengan sengaja, dari
\cref{ex:g2:caring-for-nature:help}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:caring-for-nature:8}
Cobalah: pada jalan-jalanmu berikutnya, hitung potongan sampah yang
kamu lihat (tanpa menyentuh yang berbahaya). Jenis mana yang paling
sering? Ke mana seharusnya tiap jenis itu pergi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:caring-for-nature:9}
Di kolam, Emma ingin membawa pulang tiga ekor \omterm{ex:g2:animals-grow-up:frog}{berudu} dalam sebuah
toples ``supaya mereka tumbuh dengan aman''. Dengan memakai bab ini
dan \cref{exo:g2:where-animals-live:10}, apa yang akan kamu sarankan
sebagai gantinya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:caring-for-nature:10}
``Satu tutup botol tidak berarti apa-apa.'' Pakai
\cref{rem:g2:caring-for-nature:count} untuk menjawab: apa yang terjadi
ketika sejuta pengunjung masing-masing memikirkan kalimat yang sama
--- ke arah yang buruk, dan ke arah yang baik?
\end{exercise}
