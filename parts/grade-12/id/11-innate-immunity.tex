\chapter{Imunitas Bawaan}\label{ch:g12:innate-immunity}

Sebuah serpihan kayu masuk ke ujung jari dan dicabut dalam semenit.
Menjelang petang tempatnya memerah, hangat, membengkak dan berdenyut;
keesokan harinya setetes nanah kuning terbentuk di bawah kulitnya; pada
hari ketiga semuanya hilang. Tak ada yang diputuskan, tak ada yang
dipelajari: keempat tanda yang sama muncul pada setiap luka, pada
setiap orang, pada setiap mamalia, dan sudah begitu jauh sebelum ada
yang tahu bahwa serpihan kayu membawa bakteri. Itulah permukaan yang
terlihat dari garis pertahanan pertama tubuh --- seperangkat \omterm{def:g10:cells-common-unit:cell}{sel} dan
molekul yang mengenali penyusup dalam beberapa menit dan menyerangnya
tanpa pernah menemuinya. Bab ini menguraikan garis itu; bab berikutnya
menguraikan pertahanan yang lebih lambat dan mampu belajar yang
dipanggilnya.

\section{Dua imunitas}

\begin{definition}[Imunitas bawaan dan imunitas adaptif]\label{def:g12:innate-immunity:immunity}
\emph{Sistem imun}\index{sistem imun} adalah kumpulan organ, \omterm{def:g10:cells-common-unit:cell}{sel} dan
molekul yang membela tubuh terhadap jangkitan dan kerusakan. Sistem itu
punya dua bagian. \emph{Imunitas bawaan}\index{imunitas bawaan}
sudah ada sejak lahir, bekerja dalam hitungan menit sampai jam,
mengenali golongan penyusup yang luas --- bakteri, virus, jamur, \omterm{def:g10:cells-common-unit:cell}{sel}
yang rusak --- lewat ciri yang mereka bagi, dan menanggapinya dengan
cara yang sama setiap kali. \emph{Imunitas adaptif}\index{imunitas adaptif}
perlu beberapa hari untuk bekerja, mengenali tiap penyusup satu per
satu, dan mengingatnya (\cref{ch:g12:adaptive-immunity}). Sistem bawaan
dimiliki semua hewan; sistem adaptif hanya ada pada \omterm{def:g10:common-ancestry:bodyplan}{vertebrata}, dan
bergantung pada sistem bawaan untuk dipicu.
\end{definition}

\begin{proposition}[Penghalang lebih dahulu]\label{prop:g12:innate-immunity:barriers}
Sebagian besar penyusup tak pernah masuk. Kulit, yang kering, asam dan
terus-menerus meluruhkan permukaannya, menghentikan hampir segalanya;
lapisan saluran napas, usus dan saluran reproduksi memang basah tetapi
berselaput lendir yang memerangkap mikroba lalu disapu keluar atau
diperbarui; air mata, air liur dan lendir membawa \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} yang memecah
dinding bakteri; asam lambung dan bakteri usus sendiri, yang menempati
ruangnya, menyisakan sedikit tempat bagi pendatang. Tanggapan imun bab
ini mulai ketika sebuah penghalang jebol.
\end{proposition}
\admitted

\section{Tanggapan radang}

\begin{proposition}[Keempat tanda dan penyebabnya]\label{prop:g12:innate-immunity:inflammation}
Jebolan --- sebuah luka, sebuah jangkitan, sebuah luka bakar ---
memancing dalam beberapa menit \emph{tanggapan
radang}\index{peradangan}: kemerahan, panas, bengkak dan nyeri.
Masing-masing punya sebab:
\begin{itemize}
  \item \emph{\omterm{def:g10:cells-common-unit:cell}{Sel} sentinel} yang bermukim di dalam jaringan ---
        makrofag, \omterm{prop:g12:innate-immunity:handover}{sel dendritik}, \omterm{def:g10:cells-common-unit:cell}{sel} mast --- membawa reseptor yang
        mengenali molekul yang lazim pada seluruh golongan mikroba
        (serpihan dinding bakteri, \omterm{def:g10:chemistry-of-life:families}{asam nukleat} virus) dan molekul yang
        dilepaskan \omterm{def:g10:cells-common-unit:cell}{sel} yang rusak. Begitu mengenalinya, \omterm{def:g10:cells-common-unit:cell}{sel} itu
        melepaskan \emph{mediator}\index{mediator radang}: histamin,
        prostaglandin, dan \omterm{def:g11:gene-expression:protein}{protein} pengisyarat yang disebut sitokina.
  \item \omterm{prop:g12:innate-immunity:inflammation}{Mediatornya} melebarkan pembuluh darah setempat dan membuat
        dindingnya bocor: lebih banyak darah (kemerahan, panas), dan
        plasma yang merembes ke jaringan (bengkak). \omterm{prop:g12:innate-immunity:inflammation}{Mediator} itu juga
        mempekakan ujung saraf (nyeri), dan membuat dinding pembuluhnya
        lengket bagi \omterm{def:g10:cells-common-unit:cell}{sel} darah putih yang lewat.
  \item \emph{\omterm{def:g12:innate-immunity:phagocytosis}{Fagosit}} --- neutrofil dari darah dalam hitungan jam,
        monosit yang menjadi makrofag dalam sehari --- menyelinap
        melewati dinding pembuluh, merangkak menaiki landaian
        \omterm{prop:g12:innate-immunity:inflammation}{mediatornya} menuju lokasi, lalu menelan dan mencerna mikroba
        serta puingnya. \omterm{def:g12:innate-immunity:phagocytosis}{Fagosit} yang mati, bakteri dan cairan itulah
        nanahnya.
\end{itemize}
Tanggapannya bersifat setempat, cepat, dan sama saja bagi serpihan kayu
maupun bakteri: ia tak memerlukan kontak sebelumnya dengan penyusupnya.
\end{proposition}
\begin{proof}[Bukti percobaan]
Menyuntikkan serpihan dinding bakteri saja ke dalam kulit, tanpa satu
pun bakteri hidup, menghasilkan tanggapan yang lengkap dalam sejam:
isyaratnya adalah sebuah pola molekul, bukan mikrobanya. Menghambat
histamin (dengan antihistamin untuk demam jerami) melenyapkan kemerahan
dan bengkak yang dini; menghambat \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} pembuat prostaglandin (dengan
aspirin atau ibuprofen) mengurangi bengkak, nyeri dan demamnya.
Mengamati sebuah luka di bawah mikroskop memperlihatkan neutrofil
menggelinding sepanjang dinding pembuluh, berhenti, menyeberanginya
lalu berkerumun ke lokasi dalam dua jam; mencit yang neutrofilnya
dihabiskan mati oleh jangkitan yang dibersihkan mencit normal dalam
semalam.
\end{proof}

\begin{definition}[Fagositosis]\label{def:g12:innate-immunity:phagocytosis}
\emph{Fagositosis}\index{fagositosis} adalah penelanan sebuah partikel
--- sebutir bakteri, sebuah \omterm{def:g10:cells-common-unit:cell}{sel} mati, sebutir debu --- oleh \omterm{def:g10:cells-common-unit:cell}{sel} yang
membungkusnya dengan membrannya, menariknya masuk dalam sebuah
vesikel, lalu mencernanya dengan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} dan zat kimia yang reaktif.
\emph{Fagosit}\index{fagosit} tanggapan bawaan adalah neutrofil, yang
berumur pendek dan berjumlah banyak, serta makrofag, yang berumur
panjang dan bermukim. Makrofag yang telah mencerna sebuah mikroba
menyimpan serpihan \omterm{def:g11:gene-expression:protein}{proteinnya} lalu memajangnya di permukaannya ---
langkah yang akan menghubungkan tanggapan ini dengan tanggapan bab
berikutnya.
\end{definition}

\begin{example}[Jadwal sebuah serpihan kayu]\label{ex:g12:innate-immunity:timetable}
Menit 0: bakteri dari serpihan kayu berada di dalam jaringan. Menit
1--10: \omterm{def:g10:cells-common-unit:cell}{sel} mast dan makrofag mengenali serpihan dindingnya lalu
melepaskan histamin dan sitokina; pembuluhnya melebar. Jam 1:
kemerahan, kehangatan, dan neutrofil pertama menyeberangi dinding
pembuluh. Jam 2--12: neutrofil berkerumun, menelan bakteri, lalu mati;
bengkak dan nyerinya memuncak. Hari 1: monosit tiba dan menjadi
makrofag, membersihkan puingnya; nanahnya terlihat. Hari 2--3:
bakterinya lenyap, \omterm{prop:g12:innate-immunity:inflammation}{mediatornya} tak lagi dilepaskan, dan pembuluhnya
kembali normal, lalu perbaikan pun mulai. Bila bakterinya berlipat lebih cepat
daripada kesanggupan \omterm{def:g12:innate-immunity:phagocytosis}{fagosit} membersihkannya, tanggapannya meluas,
demam muncul, dan tanggapan adaptif --- yang sudah mulai di kelenjar
getah bening --- mengambil alih.
\end{example}

\section{Menenangkan tanggapannya}

\begin{proposition}[Obat antiradang]\label{prop:g12:innate-immunity:drugs}
\omterm{prop:g12:innate-immunity:inflammation}{Tanggapan radang} itu berguna sekaligus mahal: bengkak dan nyerinya
adalah harga membersihkan penyusupnya, dan ketika tanggapan itu
berlebihan, salah arah (terhadap serbuk sari yang tak berbahaya, atau
terhadap \omterm{def:g10:muscles-and-joints:joint}{sendi} tubuh sendiri) atau berkepanjangan, ia menimbulkan
kerusakannya sendiri. \emph{Obat antiradang}\index{obat antiradang}
bekerja pada \omterm{prop:g12:innate-immunity:inflammation}{mediatornya}: aspirin, ibuprofen dan kerabatnya menghambat
\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} pembuat prostaglandin, sehingga mengurangi nyeri, bengkak dan
demam; kortikosteroid, yang diturunkan dari \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} kelenjar adrenal,
mematikan produksi sebagian besar \omterm{prop:g12:innate-immunity:inflammation}{mediator} dan dipakai untuk peradangan
yang berat atau menahun. Semuanya mengurangi tandanya; tak satu pun
menyingkirkan sebabnya, dan dengan meredam tanggapannya obat itu dapat
memperlambat pembersihan jangkitan yang sungguhan.
\end{proposition}
\admitted

\begin{example}[Demam]\label{ex:g12:innate-immunity:fever}
Sitokina yang dilepaskan di lokasi peradangan yang luas atau berlarut
mencapai otak, tempat sitokina itu menaikkan titik acuan suhu tubuh:
demam. Satu-dua derajat panas tambahan mempercepat \omterm{def:g12:innate-immunity:phagocytosis}{fagosit} dan
memperlambat banyak bakteri, dengan ongkos energi dan rasa tak nyaman;
di atas \qty{40}{\celsius} ongkosnya melebihi manfaatnya, dan di
situlah obat antipiretik (penghambat prostaglandin) dipakai. Demam
adalah tanggapan yang diatur, bukan kegagalan pengaturan.
\end{example}

\section{Dari bawaan ke adaptif}

\begin{proposition}[Serah terimanya]\label{prop:g12:innate-immunity:handover}
Di antara \omterm{def:g10:cells-common-unit:cell}{sel} sentinel, \emph{sel dendritik}\index{sel dendritik}
punya tugas kedua. Setelah menelan penyusupnya, \omterm{def:g10:cells-common-unit:cell}{sel} itu meninggalkan
jaringannya lewat pembuluh limfa dengan membawa serpihan \omterm{def:g11:gene-expression:protein}{protein}
penyusup yang dipajang di permukaannya, lalu berjalan menuju \emph{kelenjar
getah bening} terdekat. Di sana \omterm{def:g10:cells-common-unit:cell}{sel} itu menyajikan serpihannya kepada
limfosit sistem adaptif, dan sitokina yang dibawanya memberi tahu
limfosit itu penyusup macam apa yang datang. Jadi tanggapan bawaan
bukan hanya bertempur pada hari-hari pertama; ia juga mengenali
musuhnya dan mengantarkan pemeriannya. Tanpa serah terima itu, tak ada
tanggapan adaptif yang mulai --- itulah sebabnya sebuah vaksin memuat,
selain antigennya, zat yang memancing peradangan kecil.
\end{proposition}
\admitted

\begin{method}[Membaca sebuah peradangan]\label{met:g12:innate-immunity:read}
\begin{enumerate}
  \item Sebutkan jebolannya dan penyusup yang mungkin (atau kerusakan
        tanpa penyusup: keseleo pun meradang).
  \item Kaitkan tiap tandanya dengan \omterm{prop:g12:innate-immunity:inflammation}{mediatornya} dan perubahan
        pembuluhnya: kemerahan dan panas dengan pembuluh yang melebar,
        bengkak dengan pembuluh yang bocor, nyeri dengan saraf yang
        dipekakan.
  \item Kenali \omterm{def:g10:cells-common-unit:cell}{sel} yang hadir pada tiap tahap: sentinel lebih dahulu,
        neutrofil dalam hitungan jam, makrofag dalam sehari; nanahnya
        sisa mereka.
  \item Nilailah apakah tanggapannya sepadan: setempat dan mereda dalam
        beberapa hari (bekerja sebagaimana mestinya), meluas disertai
        demam (jangkitannya menang, tanggapan adaptif diperlukan), atau
        diarahkan kepada sesuatu yang tak berbahaya atau berlarut
        (kasus untuk pengobatan antiradang).
\end{enumerate}
\end{method}

\begin{remark}[Tua, cepat dan tumpul]\label{rem:g12:innate-immunity:old}
\omterm{def:g12:innate-immunity:immunity}{Imunitas bawaan} adalah imunitas seekor bunga karang, seekor serangga
dan sebatang pohon ek sama seperti imunitas manusia: reseptor pola dan
\omterm{def:g12:innate-immunity:phagocytosis}{fagosit} lebih tua daripada \omterm{def:g10:common-ancestry:bodyplan}{vertebrata} sebanyak ratusan juta tahun.
Imunitas itu cepat karena disiapkan sejak awal untuk golongan musuh,
dan tumpul justru karena alasan yang sama: ia tak dapat membedakan
satu bakteri dari bakteri lain, dan tak dapat memperbaiki diri. Sistem
bab berikutnya sanggup keduanya --- tetapi hanya setelah sistem ini
membunyikan tanda bahaya.
\end{remark}

\section{Soal: Tiga Hari Sebuah Serpihan Kayu}

\begin{problem}[{Soal akhir pekan --- sebuah luka diikuti jam demi jam:
selnya dicacah, mediatornya dilacak, obatnya diuji, dan serah terima
kepada kelenjar getah bening diukur waktunya}]
\label{pb:g12:innate-immunity:1}
Sebuah serpihan kayu membawa sekitar $\num{10000}$ bakteri ke ujung
jari; bakteri itu membelah tiap 40 menit. Jaringannya memuat 200
makrofag bermukim dalam radius satu milimeter dari lukanya; neutrofil
mulai berdatangan dari darah setelah 1 jam dengan laju $\num{50000}$
per jam, dan masing-masing sanggup menelan sekitar 20 bakteri sebelum
mati.

\medskip
\noindent\textbf{Bagian I --- Perlombaannya.}
\begin{enumerate}
  \item Berapa bakteri yang akan ada setelah 2 jam, lalu setelah 4 jam,
        seandainya tak ada yang membunuhnya?
  \item Kedua ratus makrofag itu masing-masing menelan sekitar 5
        bakteri sejam. Berapa yang disingkirkannya pada jam pertama,
        dan apakah itu menghentikan pertumbuhannya?
  \item Sejak jam ke-1, neutrofil berdatangan. Berapa bakteri yang
        dapat disingkirkan neutrofil jam pertama itu seluruhnya?
  \item Bandingkan dengan \omterm{def:g12:selection-drift-speciation:population}{populasi} bakteri pada jam ke-2. Siapa yang
        sedang menang?
  \item Kira-kira pada jam ke berapa lukanya bersih? Apa yang tersisa
        di lokasinya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Tandanya.}
\begin{enumerate}[resume]
  \item Kaitkan kemerahan, panas, bengkak dan nyeri dengan sebuah
        \omterm{prop:g12:innate-immunity:inflammation}{mediator} dan dengan perubahan pada pembuluh atau sarafnya.
  \item Jelaskan mengapa bengkaknya membantu neutrofilnya, dan mengapa
        bengkak itu terasa nyeri.
  \item Ujung jarinya berdenyut mengikuti tiap detak jantung. Jelaskan
        dari pembuluh yang melebar dan tekanan di dalam jaringan yang
        membengkak.
  \item Sehari kemudian setetes nanah terbentuk. Daftarkan isinya.
  \item Pada hari ke-3 tandanya lenyap. Apa yang menghentikan
        pelepasan \omterm{prop:g12:innate-immunity:inflammation}{mediatornya}?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Obatnya.}
\begin{enumerate}[resume]
  \item Antihistamin diminum pada jam ke-1. Tanda mana yang
        dikuranginya, dan mana yang tidak?
  \item Ibuprofen diminum pada jam ke-6. \omterm{prop:g12:innate-immunity:inflammation}{Mediator} mana yang
        dihambatnya, dan tanda mana yang dikuranginya?
  \item Sebuah krim kortikosteroid dioleskan. Apa yang terjadi pada
        perekrutan neutrofilnya, dan risiko apa yang mengikutinya bagi
        jangkitan itu?
  \item Jelaskan mengapa tak satu pun dari ketiga obat itu
        memperpendek jangkitannya, dan obat mana yang justru dapat
        memperpanjangnya.
  \item Pasiennya juga demam \qty{38.5}{\celsius}. Molekul mana yang
        menyebabkannya, dan untuk apa demam itu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Laporannya.}
\begin{enumerate}[resume]
  \item \omterm{prop:g12:innate-immunity:handover}{Sel dendritik} yang menelan bakteri pada jam ke-2 mencapai
        kelenjar getah bening di siku pada jam ke-20, sambil memajang
        serpihan bakterinya. Apa lagi yang dibawanya selain serpihan
        itu, dan kepada siapa ia melapor?
  \item Dari gambar dua tanggapan itu, berapa hari akan berlalu
        sebelum antibodi terhadap bakteri ini berlimpah? Apakah
        antibodi itu akan diperlukan untuk serpihan kayu ini?
  \item Bakteri yang sama masuk lewat serpihan kayu kedua sebulan
        kemudian. Mana di antara kedua tanggapan itu yang berbeda pada
        kali kedua, dan mana yang sama saja?
  \item Seseorang yang lahir tanpa neutrofil yang berfungsi bertahan
        hidup dengan \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotik} harian. Jelaskan bagian pertahanan mana
        yang digantikan \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotik} itu dan mana yang tak dapat
        digantikannya.
  \item Nyatakan hasilnya: jam ketika \omterm{def:g12:innate-immunity:phagocytosis}{fagositnya} menyalip bakterinya,
        \omterm{prop:g12:innate-immunity:inflammation}{mediator} di balik masing-masing dari keempat tanda itu, dan \omterm{def:g10:cells-common-unit:cell}{sel}
        yang membawa laporannya ke kelenjar getah bening.
\end{enumerate}
\end{problem}
