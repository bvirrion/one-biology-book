\chapter{Wat dieren eten}\label{ch:g2:what-animals-eat}

Het konijn in de wei knabbelt de hele dag gras. De vos die bij
schemering diezelfde wei oversteekt heeft geen enkele belangstelling
voor gras --- hij heeft belangstelling voor het konijn. En de merel
pakt 's ochtends wormen en 's middags kersen. Elk \omterm{def:g1:animals-around-us:animal}{dier} moet eten, maar
niet elk \omterm{def:g1:animals-around-us:animal}{dier} eet hetzelfde.

\section{Drie soorten eters}

\begin{definition}[Planteneter, vleeseter, alleseter]\label{def:g2:what-animals-eat:kinds}
\omterm{def:g1:animals-around-us:animal}{Dieren} worden ingedeeld naar wat ze eten:
\begin{itemize}
  \item een \emph{planteneter}\index{planteneter} eet \omterm{def:g1:plants-around-us:plant}{planten}: gras,
        \omterm{def:g1:plants-around-us:parts}{bladeren}, zaden, vruchten --- het konijn, de koe, de slak;
  \item een \emph{vleeseter}\index{vleeseter} eet andere \omterm{def:g1:animals-around-us:animal}{dieren}: de
        vos, de kat, het lieveheersbeestje (dat op piepkleine
        bladluizen jaagt);
  \item een \emph{alleseter}\index{alleseter} eet zowel \omterm{def:g1:plants-around-us:plant}{planten} als
        \omterm{def:g1:animals-around-us:animal}{dieren}: de merel, het wilde zwijn --- en de mens.
\end{itemize}
\end{definition}

\begin{example}[Eén wei, drie menu's]\label{ex:g2:what-animals-eat:meadow}
In één zomerse wei: het konijn scheert het gras af (\omterm{def:g2:what-animals-eat:kinds}{planteneter}); de
vos jaagt op het konijn (\omterm{def:g2:what-animals-eat:kinds}{vleeseter}); de merel pakt nu een worm, straks
een kers (\omterm{def:g2:what-animals-eat:kinds}{alleseter}). Drie buren, drie menu's --- en let op: het
avondeten van de vos at eerst gras.
\end{example}

\begin{remark}[Vleeseters zijn geen schurken]\label{rem:g2:what-animals-eat:fox}
De vos is niet ``gemeen'' omdat hij konijnen eet: jagen is nu eenmaal
de manier waarop een \omterm{def:g2:what-animals-eat:kinds}{vleeseter} zich voedt --- net zo natuurlijk als het
knabbelen van het konijn. In de natuur horen eten en gegeten worden bij
hoe het leven werkt; een later hoofdstuk knoopt al deze maaltijden aan
elkaar tot voedselketens.
\end{remark}

\section{Het menu van een dier aflezen aan zijn lichaam}

\begin{proposition}[Lichamen passen bij menu's]\label{prop:g2:what-animals-eat:match}
Het lichaam van een \omterm{def:g1:animals-around-us:animal}{dier} draagt aanwijzingen over zijn voedsel:
\begin{itemize}
  \item \omterm{def:g2:what-animals-eat:kinds}{planteneters} als het konijn en de koe hebben grote platte
        kiezen om \omterm{def:g1:plants-around-us:plant}{planten} mee te vermalen, en besteden lange uren aan
        het eten;
  \item \omterm{def:g2:what-animals-eat:kinds}{vleeseters} als de kat en de vos hebben puntige tanden om prooi
        mee te grijpen, scherpe klauwen, en \omterm{def:g1:the-five-senses:sense}{ogen} die naar voren kijken
        om te jagen;
  \item bij vogels zie je het aan de snavel: kort en dik om zaden te
        kraken (de mus), gekromd en gehaakt om vlees te scheuren (de
        arend), lang en dun om insecten uit de bast te pikken.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[De tanden van de kat]\label{ex:g2:what-animals-eat:cat}
Kijk eens naar een geeuwende kat: lange puntige tanden in de voorste
hoeken van haar bek. Dat zijn grijptanden --- de uitrusting van een
\omterm{def:g2:what-animals-eat:kinds}{vleeseter}. De koe zou er geeuwend geen enkele laten zien: haar bek zit
vol brede, platte maalkiezen voor gras.
\end{example}

\begin{method}[Een menu raden]\label{met:g2:what-animals-eat:guess}
Om te raden wat een onbekend \omterm{def:g1:animals-around-us:animal}{dier} eet:
\begin{enumerate}
  \item kijk naar de tanden of de snavel: platte maalkiezen, puntige
        grijpers, of allebei?
  \item kijk naar klauwen en \omterm{def:g1:the-five-senses:sense}{ogen}: jachtgereedschap, of niet?
  \item kijk het zo mogelijk zien eten --- de zekerste aanwijzing van alle;
  \item noem het dan: \omterm{def:g2:what-animals-eat:kinds}{planteneter}, \omterm{def:g2:what-animals-eat:kinds}{vleeseter} of \omterm{def:g2:what-animals-eat:kinds}{alleseter}.
\end{enumerate}
\end{method}

\section{Genoeg eten, dag na dag}

\begin{proposition}[Eten is een dagelijkse taak]\label{prop:g2:what-animals-eat:daily}
Voedsel is de brandstof en het bouwmateriaal van het lichaam, dus
\omterm{def:g1:animals-around-us:animal}{dieren} moeten steeds opnieuw eten, dag na dag. Hoeveel tijd dat kost,
hangt van het menu af: gras geeft per hap weinig, dus graast de koe
urenlang; vlees is rijk, dus eet de vos misschien één goede maaltijd en
rust daarna uit.
\end{proposition}
\admitted

\begin{example}[De winter verandert het menu]\label{ex:g2:what-animals-eat:winter}
Voedsel verandert met de seizoenen. De merel eet in de zomer wormen en
insecten, maar in de winter, als de grond hard is, stapt hij over op
bessen. Het brede menu van een \omterm{def:g2:what-animals-eat:kinds}{alleseter} is een grote hulp als de
tijden zwaar worden.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:1}
Geef de woorden voor: een \omterm{def:g1:animals-around-us:animal}{dier} dat \omterm{def:g1:plants-around-us:plant}{planten} eet; een \omterm{def:g1:animals-around-us:animal}{dier} dat andere
\omterm{def:g1:animals-around-us:animal}{dieren} eet; een \omterm{def:g1:animals-around-us:animal}{dier} dat allebei eet.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:2}
\omterm{def:g2:what-animals-eat:kinds}{Planteneter}, \omterm{def:g2:what-animals-eat:kinds}{vleeseter} of \omterm{def:g2:what-animals-eat:kinds}{alleseter}: (a) de koe; (b) de vos; (c) de
merel; (d) de slak; (e) de mens?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:3}
Welke tanden heeft een \omterm{def:g2:what-animals-eat:kinds}{planteneter} het hardst nodig: platte maalkiezen
of puntige grijpers? Waarom?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:4}
Koppel elke snavel aan een menu: kort en dik; gekromd en gehaakt; lang
en dun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:5}
Het lieveheersbeestje is klein en mooi --- en een \omterm{def:g2:what-animals-eat:kinds}{vleeseter}. Waarop
jaagt het?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:6}
Waarom besteedt de koe veel meer uren aan eten dan de vos? Gebruik
\cref{prop:g2:what-animals-eat:daily}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:7}
Wat eet de merel in de zomer, en wat in de winter? Waarom is een breed
menu nuttig?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:8}
Probeer het: kijk naar vogels in een park of bij een voedertafel.
Schrijf op wat elke soort pakt --- zaden, brood, insecten, bessen ---
en kijk naar hun snavels. Klopt \cref{prop:g2:what-animals-eat:match}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:9}
Een dierentuin krijgt een raadseldier: puntige voortanden, scherpe
klauwen, \omterm{def:g1:the-five-senses:sense}{ogen} die naar voren kijken. Wat is zijn waarschijnlijke menu?
Welke stappen van \cref{met:g2:what-animals-eat:guess} heb je gebruikt?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:10}
``Zonder gras zouden er al snel geen vossen meer zijn.'' De vos eet
nooit gras --- en toch klopt de zin. Leg uit, met de wei van
\cref{ex:g2:what-animals-eat:meadow}.
\end{exercise}
