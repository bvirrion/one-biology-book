\section*{Chapter \ref{ch:g2:animals-grow-up} --- How Animals Grow Up}

\begin{solution}{exo:g2:animals-grow-up:1}
From an \omterm{def:g2:animals-grow-up:born}{egg} laid by the mother (a chick, a frog, a snail), or \omterm{def:g2:animals-grow-up:born}{born
alive} from the mother's body and drinking her \omterm{def:g2:animals-grow-up:born}{milk} (a kitten, a
calf, a human baby).
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:2}
(a) From an \omterm{def:g2:animals-grow-up:born}{egg}. (b) \omterm{def:g2:animals-grow-up:born}{Born alive}. (c) From an \omterm{def:g2:animals-grow-up:born}{egg}. (d) \omterm{def:g2:animals-grow-up:born}{Born alive}.
(e) From an \omterm{def:g2:animals-grow-up:born}{egg}.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:3}
Its mother's \omterm{def:g2:animals-grow-up:born}{milk}. All the \omterm{def:g1:animals-around-us:animal}{animals} \omterm{def:g2:animals-grow-up:born}{born alive} in this chapter ---
puppies, calves, baby rabbits, human babies --- begin on \omterm{def:g2:animals-grow-up:born}{milk} too.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:4}
\omterm{def:g2:animals-grow-up:metamorphosis}{Metamorphosis} is the great change of body shape some young \omterm{def:g1:animals-around-us:animal}{animals}
go through to become adults. The tadpole becoming a frog, and the
\omterm{ex:g2:animals-grow-up:butterfly}{caterpillar} becoming a butterfly.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:5}
\omterm{def:g2:animals-grow-up:born}{Eggs}, tadpole, tadpole with \omterm{def:g1:my-body:parts}{legs}, frog.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:6}
\omterm{def:g2:animals-grow-up:born}{Egg}, \omterm{ex:g2:animals-grow-up:butterfly}{caterpillar}, \omterm{ex:g2:animals-grow-up:butterfly}{chrysalis}, butterfly. It is the \omterm{ex:g2:animals-grow-up:butterfly}{caterpillar} that
eats \omterm{def:g1:plants-around-us:parts}{leaves} and does nearly all the growing.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:7}
The foal --- a small horse from its first day. For it, growing up
means getting bigger and stronger while keeping the same shape.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:8}
To keep them warm. Inside each warm \omterm{def:g2:animals-grow-up:born}{egg} a chick is being built,
fed by the yolk, until it is ready to break the \omterm{def:g1:animals-around-us:covering}{shell} and hatch.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:9}
It is a tadpole partway through \omterm{def:g2:animals-grow-up:metamorphosis}{metamorphosis}: the back \omterm{def:g1:my-body:parts}{legs} have
grown, the front \omterm{def:g1:my-body:parts}{legs} have not yet appeared. Before, it was a
legless tadpole (and before that, an \omterm{def:g2:animals-grow-up:born}{egg}); next, front \omterm{def:g1:my-body:parts}{legs} will
appear, the tail will shrink, and it will become a small frog.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:10}
Nothing moved in --- the \omterm{ex:g2:animals-grow-up:butterfly}{caterpillar} and the butterfly are the
same \omterm{def:g1:animals-around-us:animal}{animal}. Inside the \omterm{ex:g2:animals-grow-up:butterfly}{chrysalis} the \omterm{ex:g2:animals-grow-up:butterfly}{caterpillar}'s body was
rebuilt into a butterfly's: that rebuilding is what \omterm{def:g2:animals-grow-up:metamorphosis}{metamorphosis}
means.
\end{solution}
