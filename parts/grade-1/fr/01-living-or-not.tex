\chapter{Vivant ou non vivant~?}\label{ch:g1:living-or-not}

Un escargot glisse lentement sur le chemin mouillé. À côté de lui
repose un caillou gris, aussi petit, aussi rond. L'escargot et le
caillou reçoivent la même pluie --- pourtant l'un est vivant et
l'autre non. Comment le savons-nous~? C'est la toute première
question de la \omterm{def:g1:living-or-not:living}{biologie}, la science des \omterm{def:g1:living-or-not:living}{êtres vivants}.

\section{Ce que font les êtres vivants}

\begin{definition}[Être vivant]\label{def:g1:living-or-not:living}
Un \emph{être vivant}\index{être vivant} est quelque chose qui est
\emph{vivant}\index{vivant}~: il naît, il se nourrit, il grandit, il
peut avoir des petits, et un jour il meurt. Les animaux et les
plantes sont des êtres vivants. La biologie\index{biologie} est la
science qui les étudie.
\end{definition}

\begin{example}[L'escargot et le caillou]\label{ex:g1:living-or-not:snail}
L'escargot est né d'un tout petit œuf. Il mange des feuilles de
salade. Il grandit --- sa coquille grandit avec lui. Il peut y avoir
des bébés escargots. L'escargot est donc un \omterm{def:g1:living-or-not:living}{être vivant}. Le caillou
ne mange rien, ne grandit pas du tout et n'a jamais de bébés
cailloux~: le caillou n'est pas vivant.
\end{example}

\begin{proposition}[Les signes de la vie]\label{prop:g1:living-or-not:signs}
Un \omterm{def:g1:living-or-not:living}{être vivant} montre ces signes~:
\begin{enumerate}
  \item il \emph{naît}\index{naît}~;
  \item il se \emph{nourrit}\index{nourrit} --- il prend de la
        nourriture ou la fabrique lui-même, comme le font les
        plantes avec la lumière~;
  \item il \emph{grandit}\index{grandit}~;
  \item il peut avoir des petits --- de nouveaux \omterm{def:g1:living-or-not:living}{êtres vivants}
        comme lui~;
  \item il \emph{meurt}\index{meurt} un jour.
\end{enumerate}
Une chose qui ne montre aucun de ces signes n'est pas vivante.
\end{proposition}
\admitted

\begin{remark}[Chaque signe compte]\label{rem:g1:living-or-not:allsigns}
Un seul signe ne suffit pas. Une voiture «~boit~» du carburant, et
un glaçon s'allonge quand il fait froid --- mais la voiture a été
construite, elle n'est pas née, et le glaçon n'a pas de bébés
glaçons. Pour \omterm{def:g1:living-or-not:living}{être vivant}, il faut montrer les signes de la vie
\emph{tous ensemble}~: naître, se nourrir, grandir, faire de
nouveaux \omterm{def:g1:living-or-not:living}{êtres vivants}, mourir.
\end{remark}

\section{Vivant, non vivant, jamais vivant}

\begin{definition}[Chose non vivante]\label{def:g1:living-or-not:nonliving}
Une \emph{chose non vivante}\index{chose non vivante} est une chose
qui n'est pas vivante. Il y en a deux sortes~:
\begin{itemize}
  \item les choses qui n'ont \emph{jamais été vivantes}\index{jamais
        été vivantes}~: un caillou, l'eau, le vent, un vélo~;
  \item les choses qui ne sont \emph{plus vivantes}\index{plus
        vivantes} ou qui \emph{viennent} d'un \omterm{def:g1:living-or-not:living}{être vivant}~: une
        feuille morte par terre, une table en bois, un pull en laine.
\end{itemize}
\end{definition}

\begin{example}[La table en bois]\label{ex:g1:living-or-not:table}
Une table en bois ne se nourrit pas et ne grandit pas~: elle n'est
pas vivante. Mais son bois, lui, l'a été~: il vient d'un arbre né
d'une graine, qui a grandi pendant des années avant d'être coupé. La
table est non vivante --- et elle vient d'un \omterm{def:g1:living-or-not:living}{être vivant}.
\end{example}

\begin{example}[Et un robot, alors~?]\label{ex:g1:living-or-not:robot}
Un robot jouet avance, fait du bruit, et sa lumière clignote comme
un petit œil. Est-il vivant~? Vérifie les signes~: il a été fabriqué
dans une usine, il n'est pas né~; sa pile n'est pas une nourriture
qui le fait grandir~; il n'aura jamais de bébés robots. Le robot
bouge, mais il n'est pas vivant.
\end{example}

\begin{method}[Comment décider~: vivant ou non~?]\label{met:g1:living-or-not:decide}
Pose les questions de la vie, une par une~:
\begin{enumerate}
  \item Est-ce que c'est né~? Est-ce que ça se nourrit~? Est-ce que
        ça grandit~?
  \item Peut-il y avoir des bébés~? Est-ce que ça mourra un jour~?
  \item Si les réponses sont oui~: c'est vivant.
  \item Si les réponses sont non, pose encore une question~: est-ce
        que ça \emph{vient} de quelque chose de vivant, comme le
        bois vient de l'arbre~?
\end{enumerate}
\end{method}

\begin{remark}[Les graines ont l'air endormies]\label{rem:g1:living-or-not:seed}
Une graine de haricot sèche, dans son sachet, semble aussi immobile
qu'un caillou. Mais pose-la sur du coton mouillé~: en quelques
jours, une petite racine apparaît. La graine était vivante depuis le
début, elle attendait. Les \omterm{def:g1:living-or-not:living}{êtres vivants} lents et discrets se
confondent facilement avec les \omterm{def:g1:living-or-not:nonliving}{choses non vivantes} --- les questions
de la vie nous aident à mieux regarder.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:1}
Dis les cinq signes de la vie. Lesquels peux-tu voir un chaton faire
aujourd'hui~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:2}
Vivant ou non vivant~? (a) un papillon~; (b) un vélo~; (c) une
pâquerette~; (d) un nuage.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:3}
Une feuille morte est posée sur le chemin. Est-elle vivante
maintenant~? A-t-elle déjà été vivante~? D'où vient-elle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:4}
Tom dit~: «~La rivière est vivante --- regarde, elle bouge et elle
fait du bruit~!~» Utilise les questions de
\cref{met:g1:living-or-not:decide} pour lui répondre gentiment.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:5}
Range dans les trois boîtes de la figure --- vivant, a été vivant,
jamais vivant~: une araignée, une chaussette en laine, une goutte de
pluie, une pomme sur l'arbre, une pomme de pin tombée.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:6}
La flamme d'une bougie «~mange~» la cire, grandit, et meurt quand tu
souffles dessus. La flamme est-elle un \omterm{def:g1:living-or-not:living}{être vivant}~? Quelle question
de la vie dit non~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:7}
Nomme deux choses de ta chambre qui viennent d'\omterm{def:g1:living-or-not:living}{êtres vivants}, comme
la table en bois de \cref{ex:g1:living-or-not:table}. Dis de quel
\omterm{def:g1:living-or-not:living}{être vivant} chacune vient.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:8}
À toi d'essayer~: pose un haricot sec sur du coton mouillé près
d'une fenêtre, et un caillou sur un autre coton mouillé à côté.
Regarde chaque jour pendant une semaine. Lequel change~? Qu'est-ce
que cela montre~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:9}
Un robot jouet et un escargot avancent tous les deux sur le sol.
Donne deux questions de la vie où leurs réponses sont différentes.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:10}
Les champignons ne courent pas comme les animaux et ne sont pas
verts comme la plupart des plantes. Avec les signes de
\cref{prop:g1:living-or-not:signs}, explique pourquoi un champignon
est quand même un \omterm{def:g1:living-or-not:living}{être vivant}.
\end{exercise}
