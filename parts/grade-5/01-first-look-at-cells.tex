\chapter{A First Look at Cells}\label{ch:g5:first-look-at-cells}

Look at the back of your hand: smooth \omterm{def:g1:the-five-senses:sense}{skin}, one seamless
surface. Now imagine a magnifying glass strong enough to zoom
in a thousand times. The seamless surface would dissolve into a
pavement of tiny compartments, fitted together like flagstones
--- and every \omterm{def:g1:living-or-not:living}{living thing} you have met in this book, moss to
whale, is paved the same way. \omterm{def:g1:living-or-not:living}{Biology}'s biggest secret is
small.

\section{Living things are made of cells}

\begin{definition}[Cell]\label{def:g5:first-look-at-cells:cell}
A \emph{cell}\index{cell} is the smallest living building block
of a \omterm{def:g1:living-or-not:living}{living thing}. Almost all cells are far too small to see
with the naked \omterm{def:g1:the-five-senses:sense}{eye}; a body is built of them as a wall is built
of bricks. Your own body is made of tens of thousands of
billions of cells.
\end{definition}

\begin{proposition}[The cell principle]\label{prop:g5:first-look-at-cells:principle}
Every \omterm{def:g1:living-or-not:living}{living thing} is made of \omterm{def:g5:first-look-at-cells:cell}{cells} --- every \omterm{def:g1:animals-around-us:animal}{animal}, every
\omterm{def:g1:plants-around-us:plant}{plant}, every mushroom, down to \omterm{def:g1:living-or-not:living}{living things} that are just
\emph{one} \omterm{def:g5:first-look-at-cells:cell}{cell}. This shared construction is one of the great
signs of the \emph{unity of life}\index{unity of life}: however
different the daisy and the whale, up close they are built of
the same kind of small living units.
\end{proposition}
\admitted

\begin{example}[How small is small?]\label{ex:g5:first-look-at-cells:small}
A typical \omterm{def:g5:first-look-at-cells:cell}{cell} of your body measures about a fiftieth of a
millimetre. Line up $50$ of them and they span one millimetre
--- a pencil dot could cover the parade. That is why nobody
suspected \omterm{def:g5:first-look-at-cells:cell}{cells} for most of human history: the building blocks
of life are smaller than the \omterm{def:g1:the-five-senses:sense}{eye}'s smallest visible speck.
\end{example}

\section{The instrument that revealed them}

\begin{definition}[Microscope]\label{def:g5:first-look-at-cells:microscope}
A \emph{microscope}\index{microscope} is an instrument whose
lenses magnify very small things hundreds of times. Pointed at
thin slices of living material, it revealed what no \omterm{def:g1:the-five-senses:sense}{eye} had
ever seen: the \omterm{def:g5:first-look-at-cells:cell}{cells}. The name itself was coined by an early
observer who thought the little compartments in a slice of
cork looked like monks' small rooms --- \emph{\omterm{def:g5:first-look-at-cells:cell}{cells}}.
\end{definition}

\begin{remark}[The dark spot]\label{rem:g5:first-look-at-cells:nucleus}
In both views, most \omterm{def:g5:first-look-at-cells:cell}{cells} show a small darker spot inside: the
\omterm{def:g5:first-look-at-cells:cell}{cell}'s command center, called its
\emph{nucleus}\index{nucleus}. What the nucleus commands with
--- the famous thread of instructions coiled inside it --- is
a story for the later years of this book; for now, note only
that the little rooms are not empty.
\end{remark}

\section{From one cell to a body}

\begin{proposition}[Bodies grow by their cells]\label{prop:g5:first-look-at-cells:growth}
A body grows, and repairs itself, because its \omterm{def:g5:first-look-at-cells:cell}{cells} can
\emph{divide}: one \omterm{def:g5:first-look-at-cells:cell}{cell} splits into two living \omterm{def:g5:first-look-at-cells:cell}{cells}, two into
four, and so on. Growing taller
(\cref{prop:g3:stages-of-life:animals}) is, underneath, \omterm{def:g5:first-look-at-cells:cell}{cells}
multiplying; a healing scrape
(\cref{rem:g3:bones-and-muscles:alive}) is \omterm{def:g5:first-look-at-cells:cell}{cells} rebuilding.
And every body, even the largest, began as a \emph{single
\omterm{def:g5:first-look-at-cells:cell}{cell}} --- the fertilized \omterm{def:g4:animal-reproduction:sexual}{egg cell} of
\cref{def:g4:animal-reproduction:sexual}.
\end{proposition}
\admitted

\begin{example}[From one to trillions]\label{ex:g5:first-look-at-cells:one}
The fertilized \omterm{def:g2:animals-grow-up:born}{egg} divides into $2$ \omterm{def:g5:first-look-at-cells:cell}{cells}, then $4$, $8$,
$16$ \dots{} Repeated division, over months and years, built
the tens of thousands of billions of \omterm{def:g5:first-look-at-cells:cell}{cells} reading this
sentence. Every \omterm{def:g5:first-look-at-cells:cell}{cell} of you descends from that first one.
\end{example}

\begin{remark}[Whole lives in one cell]\label{rem:g5:first-look-at-cells:onecelled}
Some \omterm{def:g1:living-or-not:living}{living things} never go beyond one \omterm{def:g5:first-look-at-cells:cell}{cell} --- and need
nothing more. A drop of pond water teems with them: single
\omterm{def:g5:first-look-at-cells:cell}{cells} that swim, feed, and divide, whole lives at \omterm{def:g5:first-look-at-cells:microscope}{microscope}
scale. The \omterm{def:g1:healthy-habits:germs}{germs} of \cref{def:g1:healthy-habits:germs} are
mostly such single-celled \omterm{def:g1:living-or-not:living}{living things} --- old acquaintances,
now with an explanation.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:1}
What is a \omterm{def:g5:first-look-at-cells:cell}{cell}? Roughly how many make up your body?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:2}
Why did nobody know about \omterm{def:g5:first-look-at-cells:cell}{cells} for most of history? Which
instrument changed that?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:3}
Describe what the \omterm{def:g5:first-look-at-cells:microscope}{microscope} shows in onion \omterm{def:g1:the-five-senses:sense}{skin}, and in a
cheek scrape. What do the two views have in common?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:4}
What is the small dark spot visible in most \omterm{def:g5:first-look-at-cells:cell}{cells} called?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:5}
About how many body \omterm{def:g5:first-look-at-cells:cell}{cells}, lined up, span one millimetre ---
and how long is one \omterm{def:g5:first-look-at-cells:cell}{cell}, in millimetres?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:6}
What are \omterm{def:g5:first-look-at-cells:cell}{cells} doing, underneath, when a child grows taller?
And when a scraped \omterm{def:g1:my-body:joint}{knee} heals?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:7}
Every human body began as how many \omterm{def:g5:first-look-at-cells:cell}{cells}? What was that
first \omterm{def:g5:first-look-at-cells:cell}{cell}, by \cref{def:g4:animal-reproduction:sexual}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:8}
A \omterm{def:g5:first-look-at-cells:cell}{cell} divides into $2$; each of those divides again, and so
on. Continue the doubling from $1$: how many \omterm{def:g5:first-look-at-cells:cell}{cells} after
$5$ divisions? After $10$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:first-look-at-cells:9}
Explain the sentence: ``the daisy and the whale show the
\omterm{prop:g5:first-look-at-cells:principle}{unity of life}.'' What do they share, by
\cref{prop:g5:first-look-at-cells:principle}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:first-look-at-cells:10}
A drop of pond water holds \omterm{def:g1:living-or-not:living}{living things} of one single \omterm{def:g5:first-look-at-cells:cell}{cell}
each. Check them against the signs of life of
\cref{prop:g1:living-or-not:signs}: which signs can a single
\omterm{def:g5:first-look-at-cells:cell}{cell} show?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:first-look-at-cells:11}
An elephant's \omterm{def:g5:first-look-at-cells:cell}{cells} are about the same size as a mouse's.
What, then, makes the elephant bigger --- larger \omterm{def:g5:first-look-at-cells:cell}{cells}, or
more \omterm{def:g5:first-look-at-cells:cell}{cells}? What does your answer say about how growing
works, by \cref{prop:g5:first-look-at-cells:growth}?
\end{exercise}
