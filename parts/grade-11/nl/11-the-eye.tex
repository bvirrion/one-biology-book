\chapter{Het oog en zijn fotoreceptoren}\label{ch:g11:the-eye}

In een maanloze nacht, ver van elke lamp, zie je het pad maar niet de
kleuren ervan; het rode jack van een vriend is een grijze vorm. Kijk recht
naar een zwakke ster en zij verdwijnt; kijk er een beetje naast en zij
komt terug. Beide eigenaardigheden komen van het netvlies, het dunne vlies
achter in het \omterm{def:g11:the-eye:parts}{oog} waar licht een zenuwsignaal wordt: twee typen
zintuigcellen, ongelijk verdeeld, waarvan de ene blind is voor kleur en de
andere blind in het donker. Dit hoofdstuk haalt het \omterm{def:g11:the-eye:parts}{oog} uit elkaar, van
het hoornvlies tot de oogzenuw, en leest in de \omterm{def:g10:universal-dna:gene}{genen} van de kleurstoffen
van het netvlies een stuk van onze evolutionaire geschiedenis.

\section{Het oog als optisch instrument}

\begin{definition}[De delen van het oog]\label{def:g11:the-eye:parts}
Het \emph{oog}\index{oog} is een bol van ongeveer \qty{24}{mm} doorsnee.
Licht komt binnen door het doorzichtige \emph{hoornvlies}, gaat door de
\emph{pupil} --- de opening in de gekleurde \emph{iris}, die zich in het
donker verwijdt en bij fel licht vernauwt --- en door de \emph{ooglens},
doorkruist het heldere gelei van het \emph{glasachtig lichaam} en bereikt
het \emph{netvlies}, de lichtgevoelige laag die de achterkant van de bol
bekleedt. Hoornvlies en ooglens vormen samen op het netvlies een klein
omgekeerd beeld van het tafereel; de ooglens, waarvan spieren de kromming
kunnen veranderen, stelt scherp van veraf naar dichtbij
(\emph{accommodatie}). De \emph{oogzenuw} verlaat de achterkant van het
oog en brengt de signalen van het netvlies naar de hersenen.
\end{definition}

\begin{example}[Scherpstellen, en de fouten daarbij]\label{ex:g11:the-eye:focus}
Een ontspannen \omterm{def:g11:the-eye:parts}{oog} stelt verre voorwerpen scherp op het netvlies; om op
\qty{25}{cm} te lezen moet de ooglens boller worden. Een \omterm{def:g11:the-eye:parts}{oog} dat iets te
lang is, stelt verre voorwerpen vóór het netvlies scherp --- bijziendheid,
te verhelpen met een negatieve brillenglas; een \omterm{def:g11:the-eye:parts}{oog} dat te kort is, of een
lens die met de jaren stijver is geworden, kan dichtbij niet meer
scherpstellen --- verziendheid, te verhelpen met een positief glas. In
beide gevallen is het netvlies gaaf: de fout is optisch, en de bril legt
het beeld terug waar de zintuigcellen zitten.
\end{example}

\section{Het netvlies}

\begin{definition}[Fotoreceptoren]\label{def:g11:the-eye:photoreceptors}
Het netvlies bevat twee typen \emph{fotoreceptoren}\index{fotoreceptor},
genoemd naar hun vorm. De \emph{staafjes}\index{staafje}, ongeveer 120
miljoen, reageren op heel zwak licht maar alle met dezelfde kleurstof: zij
geven 's nachts zicht in grijstinten. De \emph{kegeltjes}\index{kegeltje},
ongeveer 6 miljoen, hebben feller licht nodig en komen in drie typen
voor, elk met een kleurstof die het gevoeligst is voor een ander deel van
het spectrum: zij geven zicht bij daglicht en kleur. Elke fotoreceptor
bevat een \emph{fotopigment} --- een \omterm{def:g11:gene-expression:protein}{eiwit}, een \emph{opsine}\index{opsine},
dat een klein lichtabsorberend molecuul vasthoudt dat van vitamine A is
afgeleid --- waarvan de vormverandering bij het opnemen van een foton de
elektrische reactie van de \omterm{def:g10:cells-common-unit:cell}{cel} in gang zet.
\end{definition}

\begin{proposition}[Bouw van het netvlies]\label{prop:g11:the-eye:organisation}
De \omterm{def:g11:the-eye:photoreceptors}{fotoreceptoren} liggen achter in het netvlies, tegen een donkere laag
aan; het licht moet de andere lagen doorkruisen om hen te bereiken. Hun
signalen gaan naar \emph{bipolaire \omterm{def:g10:cells-common-unit:cell}{cellen}} en daarna naar
\emph{ganglioncellen}, waarvan de lange vezels --- ongeveer een miljoen
--- zich tot de oogzenuw bundelen. De verdeling is ongelijk: in de
\emph{gele vlek}, het kleine centrale kuiltje waar het \omterm{def:g11:the-eye:parts}{oog} naartoe kijkt,
zitten alleen \omterm{def:g11:the-eye:photoreceptors}{kegeltjes}, dicht opeen en elk met een eigen ganglioncel
verbonden, wat het scherpste zicht geeft; verder van het midden overheersen
de \omterm{def:g11:the-eye:photoreceptors}{staafjes} en delen er vele één ganglioncel, wat gevoeligheid geeft ten
koste van detail. Waar de oogzenuw vertrekt, zitten helemaal geen
zintuigcellen: de \emph{blinde vlek}.
\end{proposition}
\admitted

\begin{example}[De twee eigenaardigheden verklaard]\label{ex:g11:the-eye:oddities}
De zwakke ster verdwijnt wanneer je er recht naar kijkt, omdat de gele
vlek alleen \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} heeft, die meer licht nodig hebben dan de ster
levert; kijk je er opzij naar, dan valt haar licht op \omterm{def:g11:the-eye:photoreceptors}{staafjes}, die haar
wel opmerken. Het jack is 's nachts grijs omdat de \omterm{def:g11:the-eye:photoreceptors}{staafjes} die het zien
één enkele kleurstof hebben: zij kunnen melden hoeveel licht er is, niet
welke golflengte. Kleur is een vergelijking tussen de drie typen
\omterm{def:g11:the-eye:photoreceptors}{kegeltjes}, en de \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} zwijgen in het donker.
\end{example}

\section{Kleur: drie kleurstoffen}

\begin{proposition}[Kleurenzien]\label{prop:g11:the-eye:colour}
Elk type \omterm{def:g11:the-eye:photoreceptors}{kegeltje} bevat een van drie \omterm{def:g11:the-eye:photoreceptors}{opsines}, waarvan de
absorptietoppen rond \qty{420}{nm} liggen (\omterm{def:g11:the-eye:photoreceptors}{S-kegeltjes}, blauw),
\qty{530}{nm} (\omterm{def:g11:the-eye:photoreceptors}{M-kegeltjes}, groen) en \qty{560}{nm} (\omterm{def:g11:the-eye:photoreceptors}{L-kegeltjes}, rood).
Een gegeven licht prikkelt de drie typen in verhoudingen die van zijn
golflengte afhangen; de hersenen lezen de kleur uit die verhoudingen af.
Wie één type \omterm{def:g11:the-eye:photoreceptors}{kegeltje} mist, verwart de kleuren die dat type
onderscheidde: zonder werkende L- of \omterm{def:g11:the-eye:photoreceptors}{M-kegeltjes} zijn rood en groen
moeilijk uit elkaar te houden --- \emph{rood-groenkleurenblindheid}, die
ongeveer 8\% van de mannen en 0.5\% van de vrouwen treft.
\end{proposition}
\admitted

\begin{example}[De krommen lezen]\label{ex:g11:the-eye:curves}
Licht van \qty{500}{nm}: \omterm{def:g11:the-eye:photoreceptors}{S-kegeltjes} op 8\%, M op 75\%, L op 40\% --- dat
wordt als blauwgroen gezien. Licht van \qty{620}{nm}: S op 0, M op 10\%,
L op 42\% --- rood. Iemand zonder \omterm{def:g11:the-eye:photoreceptors}{L-kegeltjes} ontvangt voor allebei alleen
een S- en een M-waarde, en het tweede licht geeft S~0, M~10: nauwelijks te
onderscheiden van een zwak groen. De kleuren die het ontbrekende type uit
elkaar zou hebben gehouden, vallen samen tot één.
\end{example}

\section{De opsinegenen: een familie met een geschiedenis}

\begin{proposition}[De genen van de opsines]\label{prop:g11:the-eye:genes}
Elk \omterm{def:g11:the-eye:photoreceptors}{opsine} wordt door een eigen \omterm{def:g10:universal-dna:gene}{gen} gecodeerd. Het \omterm{def:g10:universal-dna:gene}{gen} voor het \omterm{def:g11:the-eye:photoreceptors}{S-opsine}
ligt op een gewoon \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}; de \omterm{def:g10:universal-dna:gene}{genen} voor het M- en het \omterm{def:g11:the-eye:photoreceptors}{L-opsine} liggen
naast elkaar op het \omterm{def:g11:cell-cycle-mitosis:chromosome}{X-chromosoom}. De \omterm{def:g10:chemistry-of-life:families}{eiwitten} L en M verschillen op
slechts 15 van hun 364 \omterm{def:g10:chemistry-of-life:families}{aminozuren} (96\% gelijk), het \omterm{def:g11:gene-expression:protein}{S-eiwit} op meer dan
de helft van zijn posities van elk van beide. Doordat het op de X ligt,
komt een niet werkend M- of \omterm{def:g10:universal-dna:gene}{L-allel} tot uiting bij elke man die het
draagt, en alleen bij vrouwen die er twee dragen --- het patroon van
\cref{ch:g11:genes-and-disease}, en de reden dat
rood-groenkleurenblindheid zestien keer vaker bij mannen voorkomt.
\end{proposition}
\admitted

\begin{proposition}[Een genfamilie uit verdubbelingen]\label{prop:g11:the-eye:family}
De drie opsinegenen, en dat van de kleurstof van de \omterm{def:g11:the-eye:photoreceptors}{staafjes}, zijn versies
van één voorouderlijk \omterm{def:g10:universal-dna:gene}{gen} dat door \emph{verdubbeling} is gekopieerd en
daarna door \omterm{def:g11:mutations:mutation}{mutatie} uiteen is gegroeid. Hun mate van gelijkenis dateert de
kopieën: de lijnen S en L/M gingen heel vroeg in de geschiedenis van de
\omterm{def:g10:common-ancestry:bodyplan}{gewervelden} uiteen, zo'n 500 miljoen jaar geleden; de \omterm{def:g10:universal-dna:gene}{genen} L en M
ontstonden uit één verdubbeling ongeveer 30 tot 40 miljoen jaar geleden bij
de voorouder van de apen van de Oude Wereld, wier nakomelingen --- apen,
mensapen, mensen --- drie typen \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} hebben, terwijl de meeste
andere zoogdieren er twee hebben.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
Vergelijking van de volgordes: de \omterm{def:g10:universal-dna:gene}{genen} L en M zijn voor 96\% gelijk en
liggen naast elkaar op de X --- het handschrift van een recente
verdubbeling naast elkaar; elk is voor ongeveer 43\% gelijk aan het \omterm{def:g10:universal-dna:gene}{S-gen},
en alle drie voor ongeveer 40\% aan het opsinegen van de \omterm{def:g11:the-eye:photoreceptors}{staafjes}, de
verder verwijderde kopieën. Honden, runderen en de meeste zoogdieren
hebben maar één opsinegen op de X, op de plaats waar primaten er twee
hebben; apen van de Nieuwe Wereld hebben er één, met verscheidene \omterm{def:g10:universal-dna:gene}{allelen},
en alleen vrouwtjes die twee verschillende \omterm{def:g10:universal-dna:gene}{allelen} dragen zien drie
kleuren. Het aantal \omterm{def:g10:universal-dna:gene}{genen} komt in elke lijn overeen met het aantal
\omterm{def:g11:the-eye:photoreceptors}{kegeltjes}, en de boom van de \omterm{def:g10:universal-dna:gene}{genen} komt overeen met die van de \omterm{def:g10:biodiversity-scales:species}{soorten}.
\end{proof}

\begin{example}[Wat één verdubbeling veranderde]\label{ex:g11:the-eye:duplication}
Een aap met twee typen \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} ziet het bos in twee dimensies van
kleur, en rijp fruit is moeilijk uit het gebladerte te halen. Na de
verdubbeling L/M kon de ene kopie naar langere golflengten muteren terwijl
de andere het oorspronkelijke behield --- een derde \omterm{def:g11:the-eye:photoreceptors}{kegeltje}, en rijp rood
tegen groen. De \omterm{def:g11:mutations:mutation}{mutatie} die later bij 8\% van de mannen een kopie
onbruikbaar maakt, brengt het \omterm{def:g11:the-eye:parts}{oog} eenvoudigweg terug naar de
voorouderlijke toestand met twee \omterm{def:g11:the-eye:photoreceptors}{kegeltjes}.
\end{example}

\begin{method}[Redeneren over een geval van kleurenzien]\label{met:g11:the-eye:case}
\begin{enumerate}
  \item Welk type \omterm{def:g11:the-eye:photoreceptors}{kegeltje} ontbreekt of is veranderd? Lees dat af aan
        de kleuren die worden verward (rood-groen: L of M; blauw-geel,
        zeldzaam: S).
  \item Welk \omterm{def:g10:universal-dna:gene}{gen}, en op welk \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}? L en M op de X, S elders.
  \item Pas de overerving toe: \omterm{def:g11:genes-and-disease:genetic}{recessief} op de X voor rood-groen ---
        zonen van dragende moeders in de helft van de gevallen aangedaan,
        dochters van aangedane vaders allemaal draagster.
  \item Breng het in verband met de genfamilie: een verloren kopie, of
        een L- en een M-kopie die door uitwisseling tussen de naast
        elkaar liggende \omterm{def:g10:universal-dna:gene}{genen} te veel op elkaar zijn gaan lijken.
\end{enumerate}
\end{method}

\begin{remark}[Van het oog naar de hersenen]\label{rem:g11:the-eye:tobrain}
Het netvlies ziet niet; het zet om en begint te sorteren. De miljoen
vezels van de oogzenuw brengen een gecodeerde beschrijving ---
contrasten, randen, verhoudingen tussen de reacties van de \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} ---
naar de hersenen, waar het volgende hoofdstuk haar oppikt. Wat wij zien
noemen, gebeurt daar, gebouwd op de signalen van de \omterm{def:g10:cells-common-unit:cell}{cellen} die hier zijn
beschreven en op niets anders: iemand van wie het netvlies werkt maar de
visuele hersenen zijn beschadigd, is blind, en iemand van wie de visuele
hersenen werken maar de \omterm{def:g11:the-eye:photoreceptors}{fotoreceptoren} zijn afgestorven, eveneens.
\end{remark}

\section{Probleem: Drie kleurstoffen en één verdubbeling}

\begin{problem}[{Weekendprobleem --- het kleurenzien uit elkaar gehaald:
de reacties van de kegeltjes op een spectrum, een familie met
kleurenblindheid, de opsinegenen vergeleken, en de scherpte die een gele
vlek kan halen}]
\label{pb:g11:the-eye:1}
Gebruik de absorptiefiguur. Een test toont lichtbronnen van
\qty{470}{nm}, \qty{530}{nm}, \qty{590}{nm} en \qty{650}{nm}.

\medskip
\noindent\textbf{Deel I --- De \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} reageren.}
\begin{enumerate}
  \item Lees voor elk licht de globale reactie van de S-, M- en
        \omterm{def:g11:the-eye:photoreceptors}{L-kegeltjes} af.
  \item Welk licht prikkelt alle drie de typen? Welk prikkelt er maar
        één?
  \item Iemand zonder \omterm{def:g11:the-eye:photoreceptors}{L-kegeltjes} ontvangt alleen de S- en de M-waarden.
        Welke twee van de vier lichtbronnen worden voor hem moeilijk te
        onderscheiden? Verantwoord dat met de getallen.
  \item Iemand zonder \omterm{def:g11:the-eye:photoreceptors}{M-kegeltjes}: welk paar verwart zij?
  \item Leg uit waarom geen van beiden "blind voor rood" is, maar beiden
        slecht zijn in het onderscheiden van rood en groen.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Een familie.}
Nora ziet kleuren normaal; haar vader is rood-groenkleurenblind. Zij
trouwt met Sam, die normaal ziet.
\begin{enumerate}[resume]
  \item Geef het \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} van Nora voor het opsinegen op de X, en
        verantwoord dat.
  \item Geef de \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypen} en het kleurenzien van hun mogelijke zonen en
        dochters, met de kansen.
  \item Hun dochter Ines, die normaal kleuren ziet, trouwt met een
        kleurenblinde man. Bereken de kans dat Ines draagster is, en
        vervolgens de kans dat een dochter van hen kleurenblind is.
  \item Leg uit waarom een kleurenblinde dochter altijd een
        kleurenblinde vader heeft.
  \item De moeder van Nora's vader zag kleuren normaal. Was zij
        noodzakelijk draagster? Leg dat uit.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De \omterm{def:g10:universal-dna:gene}{genen}.}
Overeenkomsten tussen de eiwitvolgordes van de \omterm{def:g11:the-eye:photoreceptors}{opsines}: L--M 96\%, L--S
43\%, M--S 43\%, S--rodopsine 41\%.
\begin{enumerate}[resume]
  \item Teken, of beschrijf, de boom van de vier \omterm{def:g10:universal-dna:gene}{genen} die uit die
        getallen volgt.
  \item Schat, als de verschillen zich met een gelijkmatig tempo ophopen
        en de verdubbeling L/M 35 miljoen jaar oud is, de ouderdom van
        de splitsing tussen S en L/M.
  \item De \omterm{def:g10:universal-dna:gene}{genen} L en M liggen naast elkaar op de X. Leg uit hoe een
        ongelijke uitwisseling tussen hen tijdens de meiose een X kan
        opleveren met slechts een van de twee, en welk kleurenzien
        daaruit volgt.
  \item Zulke uitwisselingen zijn de gewoonste oorzaak van
        rood-groenkleurenblindheid. Waarom zijn twee naast elkaar
        liggende, vrijwel gelijke \omterm{def:g10:universal-dna:gene}{genen} daar bijzonder gevoelig voor?
  \item De verdubbeling L/M komt bij alle apen van de Oude Wereld voor en
        bij geen enkel ander zoogdier. Wat zegt dat over wanneer en bij
        wie zij optrad?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De scherpte van de gele vlek.}
De \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} van de gele vlek liggen \qty{2.5}{\micro m} uit elkaar; de
brandpuntsafstand van het \omterm{def:g11:the-eye:parts}{oog} is \qty{17}{mm}. Gebruik hoek (radialen)
$=$ grootte $/$ afstand.
\begin{enumerate}[resume]
  \item Bereken de hoek tussen twee naburige \omterm{def:g11:the-eye:photoreceptors}{kegeltjes}, in radialen en in
        boogminuten ($1' = \num{2.9e-4}$ rad).
  \item Wat is het kleinste letterdetail dat op \qty{6}{m} te
        onderscheiden is? Vergelijk met de streepjes van de letters op
        een oogtestkaart, ongeveer \qty{1.7}{mm} op die afstand voor
        "normaal" zicht.
  \item Buiten de gele vlek delen 100 \omterm{def:g11:the-eye:photoreceptors}{staafjes} één ganglioncel. Schat de
        hoek die daar te onderscheiden is, en leg uit wat je er in ruil
        voor krijgt.
  \item De \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} in de gele vlek van een arend liggen twee keer zo
        dicht opeen en zijn \omterm{def:g11:the-eye:parts}{oog} is iets groter. Met welke factor is zijn
        \omterm{def:g10:cells-common-unit:magnification}{oplossend vermogen} fijner?
  \item Geef het resultaat: de drie typen \omterm{def:g11:the-eye:photoreceptors}{kegeltjes} en het licht waar
        elk blind voor is, de ene genetische gebeurtenis die de primaten
        de derde gaf, en de hoek die de gele vlek onderscheidt.
\end{enumerate}
\end{problem}
