\chapter{Biodiversity Near You}\label{ch:g6:biodiversity}

\omterm{def:g6:biodiversity:def}{Biodiversity} is not only coral reefs and rainforests. Your street trees,
school hedge and local stream hold a living mix of \omterm{def:g6:classification:species}{species} --- and face
local threats. This chapter trains you to notice, measure and protect
nearby variety of life.

\section{What biodiversity means}

\begin{definition}[Biodiversity]\label{def:g6:biodiversity:def}
\emph{Biodiversity}\index{biodiversity} is the variety of life:
different \omterm{def:g6:classification:species}{species}, genetic differences within \omterm{def:g6:classification:species}{species}, and variety of
\omterm{def:g6:ecology:levels}{ecosystems} in a region.
\end{definition}

\begin{proposition}[Local value]\label{prop:g6:biodiversity:value}
Local \omterm{def:g6:biodiversity:def}{biodiversity} supports \omterm{def:g6:reproduction:plant}{pollination}, soil \omterm{def:g2:healthy:health}{health}, clean \omterm{def:g4:plants-need:needs}{water},
pest balance and human wellbeing. Losing it weakens those services.
\end{proposition}
\admitted

\section{Measuring what is near you}

\begin{method}[Simple biodiversity survey]\label{met:g6:biodiversity:survey}
\begin{enumerate}
  \item Choose a safe sample area (quadrat on grass, length of hedge).
  \item Record \omterm{def:g6:classification:species}{species} you can identify; photograph unknowns.
  \item Count how many kinds (richness), not only total individuals.
  \item Repeat in a second \omterm{def:g4:ecosystems:habitat}{habitat} for comparison.
  \item Minimise disturbance; follow access rules.
\end{enumerate}
\end{method}

\begin{definition}[Species richness]\label{def:g6:biodiversity:richness}
\emph{Species richness}\index{species richness} is the number of
different \omterm{def:g6:classification:species}{species} in a sample. A lawn and a wildflower strip may have
similar \omterm{def:g1:plants-animals:plant}{plant} cover but very different richness.
\end{definition}

\begin{example}[Two patches]\label{ex:g6:biodiversity:patches}
A mown football pitch may show few \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g6:classification:species}{species}. A rough edge with
nettles, \omterm{def:g2:plant-parts:main}{flowers} and shrubs may host more \omterm{def:g3:insects:insect}{insects} and \omterm{def:g2:animal-groups:five}{birds} --- higher
local \omterm{def:g6:biodiversity:def}{biodiversity}.
\end{example}

\section{Threats and responses}

\begin{definition}[Local threats]\label{def:g6:biodiversity:threats}
Common pressures include \omterm{def:g4:protecting:habitat-loss}{habitat loss} and fragmentation, pollution,
invasive \omterm{def:g6:classification:species}{species}, overharvesting and climate stress. Urban lighting and
litter also matter at street scale.
\end{definition}

\begin{method}[Actions that scale to you]\label{met:g6:biodiversity:act}
\begin{enumerate}
  \item Leave a wild corner or \omterm{def:g1:plants-animals:plant}{plant} native \omterm{def:g2:plant-parts:main}{flowers} if allowed.
  \item Reduce pesticides in gardens when possible.
  \item Keep cats indoors at peak \omterm{def:g5:vertebrates:groups}{bird} times if that fits your home.
  \item Join citizen \omterm{def:g6:method:science}{science} counts (\omterm{def:g2:animal-groups:five}{birds}, butterflies) with \omterm{def:g3:life-cycles:butterfly}{adults}.
  \item Support protection of local green spaces with evidence from
        your surveys.
\end{enumerate}
\end{method}

\begin{example}[Invasive plant]\label{ex:g6:biodiversity:invasive}
A fast-spreading non-native \omterm{def:g1:plants-animals:plant}{plant} can shade out natives, cutting \omterm{def:g2:food-energy:food}{food}
for local \omterm{def:g3:insects:insect}{insects}. Controlled removal (with permission) can restore
space for natives.
\end{example}
