\section*{Bab \ref{ch:b1:organism-environment} --- Organisme: Sistem yang Berinteraksi dengan Lingkungannya}

\begin{solution}{exo:b1:organism-environment:1}
Materi (mamalia memakan makanan dan mengembuskan karbon dioksida;
tumbuhan memasukkan karbon dioksida dan melepas oksigen), energi
(mamalia mengambil energi kimia makanan dan melepas kalor; tumbuhan
mengambil cahaya dan melepas kalor), informasi (mamalia melihat seekor
pemangsa; tumbuhan membelok ke arah cahaya).
\end{solution}

\begin{solution}{exo:b1:organism-environment:2}
$S/V = 3/r$ terbagi dua. Penyerapan lewat permukaan tumbuh sebagai $r^2$
dan pemakaian sebagai $r^3$; melewati suatu jari-jari permukaan itu tidak
lagi sanggup memberi makan volumenya, sehingga sel tetap berada pada
kisaran mikrometer atau melipat permukaannya.
\end{solution}

\begin{solution}{exo:b1:organism-environment:3}
Pemeliharaan sebuah variabel \omterm{def:b1:organism-environment:milieu}{lingkungan internal} dekat pada \omterm{def:b1:organism-environment:homeostasis}{titik acuan}
lewat \omterm{def:b1:organism-environment:homeostasis}{umpan balik negatif}. Pengindera: termoreseptor kulit dan otak;
pusat pemadu: hipotalamus, yang memegang \omterm{def:b1:organism-environment:homeostasis}{titik acuan}
\qty{37}{\celsius}; efektor: otot yang menggigil, pembuluh kulit,
kelenjar keringat.
\end{solution}

\begin{solution}{exo:b1:organism-environment:4}
Kadal: dari $1.0$ ke $0.25$, faktor $4$ (dua kali lipat dua untuk
\qty{20}{\celsius}). Mencit: dari $1.0$ ke $3.4$, faktor sekitar
$3.4$.
\end{solution}

\begin{solution}{exo:b1:organism-environment:5}
Sisi \qty{1}{mm}: perlu $\qty{1}{nmol/s}$ ($V = \qty{1}{mm^3}$); pasokan
$2\times 6 = \qty{12}{nmol/s}$: bisa. Sisi \qty{10}{mm}: perlu
\qty{1000}{nmol/s}, pasokan $2\times 600 = \qty{1200}{nmol/s}$: nyaris.
Batasnya: $2\times 6L^2 = L^3$ memberi $L = \qty{12}{mm}$.
\end{solution}

\begin{solution}{exo:b1:organism-environment:6}
Kuda: $3.4\times 500^{0.75} = \qty{360}{W}$, \qty{0.72}{W/kg}. Celurut:
$3.4\times 0.005^{0.75} = \qty{0.064}{W}$, \qty{13}{W/kg}. Sehari:
$3\times 0.064\times 86400 = \qty{16.6}{kJ}$, yakni \qty{3.3}{g} makanan,
dua pertiga massanya sendiri.
\end{solution}

\begin{solution}{exo:b1:organism-environment:7}
$\qty{80}{W} = \qty{4.8}{kJ/min}$, yakni \qty{0.24}{L} $\mathrm{O_2}$
tiap menit; sehari $80\times 86400 = \qty{6.9}{MJ}$. \qty{2000}{kcal}
= \qty{8.36}{MJ} sehari = \qty{97}{W}.
\end{solution}

\begin{solution}{exo:b1:organism-environment:8}
Mencit: $100\times(0.02/70)^{0.75} = \qty{0.22}{m^2}$; gajah:
$100\times(4000/70)^{0.75} = \qty{2080}{m^2}$. Bola: mencit
$r = \qty{1.7}{cm}$, $S = \qty{0.0036}{m^2}$ (paru enam puluh kali luas
permukaan tubuh); gajah $r = \qty{0.98}{m}$, $S = \qty{12}{m^2}$ (paru
seratus tujuh puluh kali). Makin besar hewannya, makin terlipat parunya
dibandingkan sisi luarnya.
\end{solution}

\begin{solution}{exo:b1:organism-environment:9}
Efektor baru bertindak setelah pengindera mencatat sebuah simpangan,
jadi sebuah simpangan harus ada dulu sebelum dikoreksi; koreksinya lalu
sedikit melewati sasaran sebelum pengindera melaporkannya. Lebar pitanya
ditentukan oleh kepekaan pengindera (simpangan terkecil yang terdeteksi),
tundaan antara pendeteksian dan pengaruhnya, serta kekuatan efektornya.
\end{solution}

\begin{solution}{exo:b1:organism-environment:10}
$V = \qty{2e-5}{m^3}$, $r = \qty{1.68}{cm}$, $S = 4\pi r^2 =
\qty{3.6e-3}{m^2}$. Kehilangan: $5\times 3.6\times 10^{-3}\times 37 =
\qty{0.66}{W}$. Produksi basal: \qty{0.18}{W}, empat kali lebih kecil.
Si mencit harus menaikkan produksinya (menggigil, makan: tiga sampai
empat kali basal), memperkecil $h$ (sarang, berdesakan) atau membiarkan
suhunya turun (torpor).
\end{solution}

\begin{solution}{exo:b1:organism-environment:11}
Kehilangan $\propto M^{2/3}$, produksi $\propto M^{3/4}$: rasio
produksi terhadap kehilangan $\propto M^{1/12}$ naik bersama massa,
sehingga hewan kecil memiliki produksi paling sedikit tiap satuan
permukaan dan hewan besar paling banyak. Bola bermassa $M$:
$S = 4\pi(3M/4000\pi)^{2/3} = \num{0.0483}\,M^{2/3}$ (SI); kehilangan
$= 5\times 20\times 0.0483\,M^{2/3} = 4.83\,M^{2/3}$; kesamaan dengan
$3.4\,M^{3/4}$ memberi $M^{1/12} = 1.42$, $M \approx \qty{67}{kg}$. Di
bawahnya, hewan yang berbaring kedinginan tanpa penyekatan; di atasnya,
ia memanas dan harus membuang kalor.
\end{solution}

\begin{solution}{exo:b1:organism-environment:12}
Bukan lebih sederhana, melainkan tersusun secara lain. Pertukaran
tumbuhan bersifat luar (permukaan daun dan akar yang dibentangkan ke
dalam medium) dan selnya terpapar medium yang tidak diaturnya;
``pengaturannya'' berupa pertumbuhan dan bentuk, yang menyetel
permukaannya sendiri. Mamalia memindahkan pertukarannya ke dalam, di
balik cairan yang diatur, sehingga selnya tak pernah melihat dunia luar.
\omterm{def:b1:organism-environment:homeostasis}{Homeostasis} adalah sifat \omterm{def:b1:organism-environment:milieu}{lingkungan internal} hewan, bukan ukuran
kerumitan: sel tumbuhan mengatur isinya sendiri, stomatanya mengatur
kehilangan airnya, dan tubuhnya adalah \omterm{prop:b1:organism-environment:surfaces}{permukaan pertukaran} yang terus
membesar --- sebuah siasat, bukan ketiadaan.
\end{solution}

\begin{solution}{pb:b1:organism-environment:1}
\textbf{1.} $V = M/\rho$: mencit \qty{2e-5}{m^3} (\qty{20}{cm^3}),
gajah \qty{4}{m^3}.
\textbf{2.} $r = (3V/4\pi)^{1/3}$: mencit \qty{1.68}{cm}, gajah
\qty{0.985}{m}.
\textbf{3.} $S = 4\pi r^2$: mencit \qty{3.55e-3}{m^2} (\qty{35.5}{cm^2}),
gajah \qty{12.2}{m^2}.
\textbf{4.} $S/V = 3/r$: mencit \qty{178}{m^{-1}}, gajah
\qty{3.05}{m^{-1}}; rasionya $58$.
\textbf{5.} $S/V \propto 1/r \propto M^{-1/3}$, jadi rasionya adalah
$(M_e/M_m)^{1/3} = (2\times 10^5)^{1/3} = 58.5$.
\textbf{6.} Lebih besar bagi keduanya (bola memiliki permukaan terkecil
untuk volumenya); si mencit, dengan ekor, telinga dan tungkainya yang
langsing, menyimpang lebih jauh.
\textbf{7.} $P = 10\times S\times 20$: mencit \qty{0.71}{W}, gajah
\qty{2440}{W}.
\textbf{8.} Mencit $3.4\times 0.02^{0.75} = \qty{0.18}{W}$; gajah
$3.4\times 4000^{0.75} = \qty{1710}{W}$.
\textbf{9.} Kehilangan \qty{2440}{W} berhadapan dengan produksi
\qty{1710}{W} saat berbaring: kulit telanjang kira-kira memadai, dan si
gajah justru terancam kepanasan alih-alih kedinginan ketika giat atau
ketika udara lebih hangat.
\textbf{10.} Kehilangan \qty{0.71}{W} berhadapan dengan \qty{0.18}{W}:
empat kali terlalu banyak. Bulunya harus menurunkan $h$ sampai sekitar
\qty{2.5}{W.m^{-2}.K^{-1}}.
\textbf{11.} $S/V$-nya terkecil di antara semua mamalia darat;
telinganya menambah sepertiga permukaannya dan, karena tipis dan kaya
pembuluh, bekerja sebagai radiator yang membuang kalor ketika tubuh
menghasilkan lebih daripada yang dapat dilepas \qty{12}{m^2}.
\textbf{12.} $S/V \propto M^{-1/3}$ tumbuh tanpa batas seiring turunnya
$M$, sedangkan $B/M \propto M^{-1/4}$ tumbuh lebih lambat; di bawah
beberapa gram tak ada bulu dan tak ada laju makan yang sanggup menutup
kehilangannya. Membiarkan suhunya turun pada malam hari (torpor)
meniadakan suku $T_b - T_a$ selama jam-jam tanpa makanan.
\textbf{13.} $2B\times\qty{86400}{s}$: mencit \qty{31}{kJ}, gajah
\qty{295}{MJ}.
\textbf{14.} Mencit $31/18 = \qty{1.7}{g}$; gajah
$295\,000/5 = \qty{59}{kg}$.
\textbf{15.} Mencit $1.7/20 = 8.6\%$ massanya; gajah $59/4000 =
1.5\%$: si mencit makan enam kali lebih banyak tiap gram.
\textbf{16.} Rasio panjang usus $35/0.4 = 88$; rasio jari-jari $0.985/0.0168
= 59$. Usus tumbuh lebih cepat daripada ukuran linear
tubuh (di sini kira-kira sebagai $M^{0.36}$), yang memperbesar luas
permukaan serap si gajah dibandingkan volumenya.
\textbf{17.} $f_0 = 600\times 0.02^{1/4} = \qty{226}{min^{-1}}$;
$\tau_0 = 2/0.02^{1/4} = \qty{5.3}{yr}$.
\textbf{18.} $f = 226\times 4000^{-1/4} = \qty{28}{min^{-1}}$; $\tau =
5.3\times 4000^{1/4} = \qty{42}{yr}$.
\textbf{19.} Mencit $600\times 60\times 24\times 365\times 2 =
\num{6.3e8}$; gajah $28\times 60\times 24\times 365\times 42 =
\num{6.2e8}$.
\textbf{20.} Denyut seumur hidup $= f\tau \propto M^{-1/4}M^{1/4} = M^0$:
tidak bergantung pada massa, mendekati semiliar.
\textbf{21.} $f = 226\times 70^{-1/4} = \qty{78}{min^{-1}}$ (dekat
dengan 70); $\tau = 5.3\times 70^{1/4} = \qty{15}{yr}$, jauh di bawah
80: umur manusialah yang menyimpang, tiga sampai lima kali kecenderungan
mamalia, karena sebab (otak, kehidupan bermasyarakat, kedokteran) di
luar hukum penskalaan itu.
\textbf{22.} $160\times 4000^{-1/4}/0.02^{-1/4} = 160\times 0.0473 =
\qty{7.6}{min^{-1}}$; denyut tiap tarikan napas $600/160 = 28/7.6 = 3.75$
bagi keduanya, sebab kedua laju itu berbagi pangkat yang sama.
\textbf{23.} $B/M = 3.4\,M^{3/4}/M = 3.4\,M^{-1/4}$; pangkatnya $-1/4$.
\textbf{24.} Mencit $0.18/0.02 = \qty{9.0}{W/kg}$; gajah $1710/4000 =
\qty{0.43}{W/kg}$.
\textbf{25.} Rasionya $9.0/0.43 = 21$, sama dengan $(M_e/M_m)^{1/4} =
(2\times 10^5)^{1/4} = 21.1$: tiap gram si mencit membakar dua puluh
satu kali lebih cepat daripada tiap gram si gajah --- rasio metabolisme
per satuan massa adalah akar pangkat empat rasio massanya.
\end{solution}
