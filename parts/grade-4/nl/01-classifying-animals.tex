\chapter{Dieren classificeren}\label{ch:g4:classifying-animals}

Een natuurhistorisch museum bewaart miljoenen \omterm{def:g1:animals-around-us:animal}{dieren} --- laden vol
kevers, zalen vol skeletten, potten vol \omterm{prop:g3:sorting-living-things:groups}{vissen}. En toch loopt een
conservator in een paar minuten naar elk willekeurig exemplaar. Het
geheim is geen geheugen maar \emph{ordening}: elk \omterm{def:g1:animals-around-us:animal}{dier} heeft zijn plaats
in een groot stelsel van groepen binnen groepen, gebouwd op de
indelingsregels uit \cref{ch:g3:sorting-living-things} --- en dit jaar
leren we hoe dat stelsel werkt.

\section{Groepen binnen groepen}

\begin{definition}[Classificatie]\label{def:g4:classifying-animals:classification}
Levende wezens \emph{classificeren}\index{classificeren} is ze naar
gedeelde lichaamskenmerken in groepen indelen, en die groepen weer in
grotere groepen --- \emph{groepen genest binnen
groepen}\index{geneste groepen}, als dozen in dozen. Het hele stelsel
is een \emph{classificatie}\index{classificatie}.
\end{definition}

\begin{example}[De adressen van de kat]\label{ex:g4:classifying-animals:cat}
Eén kat, meerdere geneste adressen: de kat is een \emph{zoogdier}
(\omterm{def:g1:animals-around-us:covering}{vacht}, \omterm{def:g2:animals-grow-up:born}{melk} voor de jongen); elk zoogdier is een \emph{\omterm{def:g4:classifying-animals:vertebrate}{gewerveld dier}}
(een \omterm{def:g3:bones-and-muscles:skeleton}{skelet} met een \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat} vanbinnen); elk \omterm{def:g4:classifying-animals:vertebrate}{gewerveld dier} is een
\emph{\omterm{def:g1:animals-around-us:animal}{dier}}. Kat binnen de \omterm{prop:g3:sorting-living-things:groups}{zoogdieren}, \omterm{prop:g3:sorting-living-things:groups}{zoogdieren} binnen de gewervelde
\omterm{def:g1:animals-around-us:animal}{dieren}, gewervelde \omterm{def:g1:animals-around-us:animal}{dieren} binnen de \omterm{def:g1:animals-around-us:animal}{dieren}: elke doos zit in een grotere.
\end{example}

\begin{definition}[Gewervelde en ongewervelde dieren]\label{def:g4:classifying-animals:vertebrate}
Een \emph{gewerveld dier}\index{gewerveld dier} is een \omterm{def:g1:animals-around-us:animal}{dier} met een
inwendig \omterm{def:g3:bones-and-muscles:skeleton}{skelet} dat om een \emph{ruggengraat}\index{ruggengraat} heen
gebouwd is --- zoals de \omterm{def:g3:bones-and-muscles:skeleton}{wervelkolom} die je in
\cref{ex:g3:bones-and-muscles:feel} voelde. Een \omterm{def:g1:animals-around-us:animal}{dier} zonder ruggengraat
is een \emph{ongewerveld dier}\index{ongewerveld dier}: \omterm{prop:g3:sorting-living-things:groups}{insecten},
spinnen, slakken, wormen, kwallen --- de overgrote meerderheid van alle \omterm{def:g1:animals-around-us:animal}{dieren}.
\end{definition}

\section{De vijf klassen van de gewervelde dieren}

\begin{proposition}[De vijf klassen van gewervelden]\label{prop:g4:classifying-animals:classes}
Gewervelde \omterm{def:g1:animals-around-us:animal}{dieren} worden in vijf grote groepen ingedeeld, die
\emph{klassen}\index{klasse (van gewervelden)} heten:
\begin{enumerate}
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{vissen}}: \omterm{def:g1:animals-around-us:covering}{schubben}, vinnen, ademen in het water --- forel,
        haai, sardien;
  \item \emph{amfibieën}\index{amfibie}: een vochtige kale \omterm{def:g1:the-five-senses:sense}{huid}, eitjes
        in het water, een jeugd als donderpad --- kikker, pad, salamander;
  \item \emph{reptielen}\index{reptiel}: een droge schubbige \omterm{def:g1:the-five-senses:sense}{huid}, de
        meeste leggen eieren met een schaal op het land --- hagedis,
        slang, schildpad, krokodil;
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{vogels}}: \omterm{def:g1:animals-around-us:covering}{veren}, snavel, eieren met een schaal --- merel,
        struisvogel, pinguïn;
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{zoogdieren}}: \omterm{def:g1:animals-around-us:covering}{vacht} of haren, jongen die \omterm{def:g2:animals-grow-up:born}{melk} drinken ---
        kat, walvis, vleermuis, mens.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Een dierentuinlevering sorteren]\label{ex:g4:classifying-animals:zoo}
Er komen zes kratten aan: een python (droge schubbige \omterm{def:g1:the-five-senses:sense}{huid}, eieren met
een schaal --- reptiel), een pad (vochtige kale \omterm{def:g1:the-five-senses:sense}{huid} --- amfibie), een
pinguïn (\omterm{def:g1:animals-around-us:covering}{veren} --- vogel), een haai (vinnen en kieuwen --- vis), een
dolfijn (\omterm{def:g2:animals-grow-up:born}{melk} voor haar kalf --- zoogdier), en een krat reuzenslakken
(helemaal geen \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat} --- ongewervelden, in geen enkele klasse van
de gewervelden).
\end{example}

\begin{remark}[Reptiel of amfibie?]\label{rem:g4:classifying-animals:confusion}
Hagedissen en salamanders lijken op het eerste gezicht op elkaar ---
vier poten, een lange \omterm{def:g1:animals-around-us:parts}{staart}. De \omterm{def:g1:the-five-senses:sense}{huid} beslist: die van de hagedis is
droog en schubbig, die van de salamander vochtig en kaal; de eieren van
de hagedis hebben een schaal en worden op het land gelegd, die van de
salamander worden kaal in het water gelegd. Hetzelfde silhouet,
verschillende klassen.
\end{remark}

\section{Een classificatie gebruiken}

\begin{method}[Een gewerveld dier in zijn klasse plaatsen]\label{met:g4:classifying-animals:place}
Stel de vragen op volgorde en stop bij het eerste ja:
\begin{enumerate}
  \item \omterm{def:g1:animals-around-us:covering}{Veren}? --- een vogel.
  \item \omterm{def:g1:animals-around-us:covering}{Vacht} of haren (of \omterm{def:g2:animals-grow-up:born}{melk} voor de jongen)? --- een zoogdier.
  \item Een droge schubbige \omterm{def:g1:the-five-senses:sense}{huid}? --- een reptiel.
  \item Een vochtige kale \omterm{def:g1:the-five-senses:sense}{huid} en een jeugd als donderpad? --- een amfibie.
  \item Vinnen en kieuwen, een leven in het water? --- een vis.
\end{enumerate}
Helemaal geen \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}? Dan was het een \omterm{def:g4:classifying-animals:vertebrate}{ongewerveld dier}, en gelden
de vragen niet.
\end{method}

\begin{proposition}[Wat een klasse je gratis vertelt]\label{prop:g4:classifying-animals:free}
De klasse van een \omterm{def:g1:animals-around-us:animal}{dier} kennen is een bundel feiten kennen nog vóór je
kijkt: als je alleen te horen krijgt ``het is een zoogdier'', verwacht
je al \omterm{def:g1:animals-around-us:covering}{vacht}, \omterm{def:g2:animals-grow-up:born}{melk} voor de jongen en --- zoals bij bijna alle \omterm{prop:g3:sorting-living-things:groups}{zoogdieren}
--- levendgeboren jongen. Die besparing is het hele punt van
\omterm{def:g4:classifying-animals:classification}{classificeren}: leer de groep één keer, en elk lid komt half gekend aan.
\end{proposition}
\admitted

\begin{example}[Er komt een nieuwe soort aan]\label{ex:g4:classifying-animals:new}
Natuuronderzoekers ontdekken nog steeds \omterm{def:g1:animals-around-us:animal}{dieren} die nooit eerder gezien
zijn. Als er een nieuw vleermuisachtig \omterm{def:g1:animals-around-us:animal}{dier} met \omterm{def:g1:animals-around-us:covering}{vacht} en \omterm{def:g2:animals-grow-up:born}{melk} gevonden
wordt, gaat het meteen bij de \omterm{prop:g3:sorting-living-things:groups}{zoogdieren} --- en direct verwachten
biologen een \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat}, een hart met vier kamers en warm bloed, nog
voordat iets daarvan gecontroleerd is. De \omterm{def:g4:classifying-animals:classification}{classificatie} keert
voorspellingen uit.
\end{example}

\begin{remark}[Op weg naar de stamboom]\label{rem:g4:classifying-animals:tree}
Waarom reizen \omterm{def:g1:animals-around-us:covering}{vacht}, \omterm{def:g2:animals-grow-up:born}{melk} en \omterm{def:g4:classifying-animals:vertebrate}{ruggengraat} zo trouw samen? Het volledige
antwoord --- dat de leden van een groep hun gedeelde kenmerken van
gemeenschappelijke voorouders erven --- hoort bij het verhaal van
verwantschap en evolutie, dat in latere jaren aan bod komt. Voorlopig
werken de geneste dozen gewoon, en wonderbaarlijk goed.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:1}
Wat is een \omterm{def:g4:classifying-animals:vertebrate}{gewerveld dier}? Wat is een \omterm{def:g4:classifying-animals:vertebrate}{ongewerveld dier}? Geef van elk
twee voorbeelden.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:2}
Noem de vijf klassen van de gewervelde \omterm{def:g1:animals-around-us:animal}{dieren} met hun bepalende kenmerken.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:3}
Plaats elk \omterm{def:g1:animals-around-us:animal}{dier}: de sardien, de krokodil, de salamander, de
struisvogel, de walvis, de regenworm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:4}
Schrijf de geneste adressen van de kat op, van de kleinste doos naar de
grootste, zoals in \cref{ex:g4:classifying-animals:cat}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:5}
Een hagedis en een salamander lijken op elkaar. Welke twee kenmerken
scheiden hun klassen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:6}
Welke groep bevat de overgrote meerderheid van alle \omterm{def:g1:animals-around-us:animal}{dieren}? Noem vier
van haar leden.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:7}
Laat \cref{met:g4:classifying-animals:place} hardop lopen over de
pinguïn en over de dolfijn, vraag voor vraag.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:8}
Je krijgt alleen te horen: ``dit \omterm{def:g1:animals-around-us:animal}{dier} is een vogel.'' Noem drie dingen
die je nu zonder te kijken verwacht, volgens
\cref{prop:g4:classifying-animals:free}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:9}
De walvis heeft bijna geen haren en leeft als een vis. Leg met
\cref{met:g4:classifying-animals:place} en
\cref{ex:g3:sorting-living-things:surprises} uit waarom hij toch een
zoogdier is --- en waarom ``leeft in de zee'' nooit een
classificatiekenmerk is geweest.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:10}
De slang heeft geen poten, en de regenworm ook niet. Waarom is de een
een gewerveld reptiel en de ander een \omterm{def:g4:classifying-animals:vertebrate}{ongewerveld dier}? Welk enkele
kenmerk gaat boven de ontbrekende poten?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:classifying-animals:11}
Het vogelbekdier verbaasde natuuronderzoekers: \omterm{def:g1:animals-around-us:covering}{vacht} en \omterm{def:g2:animals-grow-up:born}{melk}, en toch
legt het eieren met een schaal. Welke klasse won, en waarom? Wat laat
het raadsel zien over \omterm{def:g4:classifying-animals:classification}{classificeren} naar \emph{bundels} kenmerken in
plaats van naar één enkel kenmerk?
\end{exercise}
