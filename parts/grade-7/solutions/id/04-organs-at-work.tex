\section*{Bab \ref{ch:g7:organs-at-work} --- Organ yang Bekerja}

\begin{solution}{exo:g7:organs-at-work:1}
Lebih banyak \omterm{def:g7:what-is-respiration:respiration}{oksigen} dan lebih banyak \omterm{prop:g7:organs-at-work:bill}{glukosa} yang diambil dari
\omterm{prop:g4:journey-of-food:absorb}{darahnya}; lebih banyak \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} dan lebih banyak panas yang
dilepaskan.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:2}
\omterm{def:g4:journey-of-food:digestion}{Zat gizi} bergula yang siap pakai di dalam \omterm{prop:g4:journey-of-food:absorb}{darah} --- bahan bakar
sehari-hari bagi pembakaran oleh \omterm{def:g5:first-look-at-cells:cell}{selnya}. \omterm{def:g3:bones-and-muscles:muscle}{Ototnya} mengambilnya dari
\omterm{prop:g4:journey-of-food:absorb}{darah} yang mengalir melintasinya (yang, lebih jauh ke belakang, diisi
oleh pengantaran \omterm{def:g4:journey-of-food:digestion}{pencernaannya}).
\end{solution}

\begin{solution}{exo:g7:organs-at-work:3}
\omterm{prop:g4:journey-of-food:absorb}{Darah} yang memasuki \omterm{def:g3:bones-and-muscles:muscle}{ototnya} dibandingkan dengan \omterm{prop:g4:journey-of-food:absorb}{darah} yang
meninggalkannya. Ketika beristirahat, \omterm{prop:g4:journey-of-food:absorb}{darah} yang meninggalkannya baru
menyerahkan sebagian kecil \omterm{def:g7:what-is-respiration:respiration}{oksigen} dan \omterm{prop:g7:organs-at-work:bill}{glukosanya}; ketika bekerja
keras, ia pergi dalam keadaan terkuras dari sebagian besar \omterm{def:g7:what-is-respiration:respiration}{oksigen}
yang dapat diantarkannya, dan jauh lebih kaya \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}.
Perbedaan itulah tagihan \omterm{def:g3:bones-and-muscles:muscle}{ototnya}.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:4}
\omterm{def:g7:what-is-respiration:respiration}{Pernapasannya} mendalam dan mencepat; \omterm{def:g4:heart-and-blood:heart}{jantungnya} berdenyut lebih cepat
dan lebih keras; \omterm{prop:g4:journey-of-food:absorb}{darahnya} dialihkan --- \omterm{def:g4:heart-and-blood:vessels}{pembuluh} \omterm{def:g3:bones-and-muscles:muscle}{otot} yang bekerja dan
\omterm{def:g1:the-five-senses:sense}{kulit} yang mendinginkan melebar sementara \omterm{def:g4:heart-and-blood:vessels}{pembuluh} \omterm{def:g4:journey-of-food:tract}{saluran pencernaan}
yang beristirahat menyempit.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:5}
Membakar bahan bakar selalu menghasilkan kehangatan, dan \omterm{def:g3:bones-and-muscles:muscle}{otot} yang
bekerja adalah tungku tubuhnya. Menggigil adalah \omterm{def:g3:bones-and-muscles:muscle}{otot} yang
dipekerjakan tanpa menghasilkan apa pun selain kehangatan itu ---
pemanasan sesuai permintaan.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:6}
Kembalinya \omterm{prop:g4:heart-and-blood:pulse}{denyut nadi} dan \omterm{def:g7:what-is-respiration:respiration}{pernapasan} secara bertahap sesudah usahanya
berakhir. Tubuhnya sedang memuat ulang cadangan \omterm{def:g7:what-is-respiration:respiration}{oksigennya},
membersihkan buangan usahanya, dan melepaskan panas yang menumpuk.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:7}
Istirahatnya kira-kira $72$; dataran tingginya dekat $160$;
\omterm{def:g7:organs-at-work:recovery}{pemulihannya} kasarnya lima sampai enam menit pada grafiknya. Latihan
menurunkan nilai istirahatnya, menurunkan dataran tingginya untuk
usaha yang sama, dan memperpendek \omterm{def:g7:organs-at-work:recovery}{pemulihannya}.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:8}
Keduanya adalah akibat penjaluran ulang. Sesudah makan siang, \omterm{def:g4:journey-of-food:tract}{saluran
pencernaannya} memegang bagian besar \omterm{prop:g4:journey-of-food:absorb}{darahnya}, sehingga bagian tubuh
yang lain berjalan lesu; berlari cepat waktu itu lalu memaksa kedua
lantainya bersaing memperebutkan pengantarannya --- nyeri \omterm{def:g4:journey-of-food:tract}{lambungnya}
dan aturan lama agar tidak berlomba dengan perut yang penuh.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:9}
\omterm{def:g4:heart-and-blood:vessels}{Pembuluh} \omterm{def:g1:the-five-senses:sense}{kulit} yang melebar adalah pengiriman panas tungkunya ke
permukaan radiator tubuhnya: sebuah pengantaran panas ke luar, yang
memecahkan masalah kepanasan \omterm{def:g3:bones-and-muscles:muscle}{otot} yang bekerja. Warna merahnya adalah
sistem pendinginan yang sedang berjalan, bukan yang sedang gagal.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:10}
\omterm{def:g4:heart-and-blood:heart}{Jantungnya} memindahkan lebih banyak \omterm{prop:g4:journey-of-food:absorb}{darah} tiap denyut --- tanda
sehari-harinya, \omterm{prop:g4:heart-and-blood:pulse}{denyut nadi} istirahat yang lebih rendah. \omterm{def:g3:bones-and-muscles:muscle}{Ototnya}
tumbuh dan menyimpan lebih banyak bahan bakar --- bukit yang sama
terasa lebih mudah. \omterm{def:g4:heart-and-blood:vessels}{Pembuluh} halus jaringan pasokannya berlipat ganda
dan \omterm{def:g7:what-is-respiration:respiration}{pernapasannya} mendalam --- berbicara tetap mungkin selama
usahanya, dan \omterm{def:g7:organs-at-work:recovery}{pemulihannya} memendek.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:11}
Mesin dengan istirahat $68$ dan \omterm{def:g7:organs-at-work:recovery}{pemulihan} tiga menit itulah yang
terlatih. \omterm{prop:g4:heart-and-blood:pulse}{Denyut nadi} istirahatnya menjamin sebuah \omterm{def:g4:heart-and-blood:heart}{jantung} yang
memindahkan lebih banyak tiap denyut; waktu \omterm{def:g7:organs-at-work:recovery}{pemulihannya} menjamin
jalur pasokan yang melunasi utang usahanya dengan cepat. Bersama-sama
keduanya adalah kedua ujung pembangunan ulang pada
\cref{prop:g7:organs-at-work:training}.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:12}
Ketika berlari cepat, \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g3:bones-and-muscles:muscle}{otot} \omterm{def:g1:my-body:parts}{tungkainya} membakar \omterm{prop:g7:organs-at-work:bill}{glukosa} dengan
\omterm{def:g7:what-is-respiration:respiration}{oksigen} untuk menenagai kontraksinya --- tujuan \omterm{def:g7:what-is-respiration:respiration}{respirasi} --- dan
tagihannya (lebih banyak bahan bakar dan \omterm{def:g7:what-is-respiration:respiration}{oksigen} masuk; lebih banyak
\omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} dan panas keluar) harus dibayar seketika. \omterm{prop:g4:journey-of-food:absorb}{Darahnya}
adalah layanan pembayarnya: dimuat di \omterm{def:g7:how-animals-breathe:four}{paru-paru} dan ususnya, dipompa
lebih cepat oleh \omterm{def:g4:heart-and-blood:heart}{jantung} yang berpacu, dijalurkan ulang ke \omterm{def:g1:my-body:parts}{tungkainya},
ia mengantarkan bahan bakar dan \omterm{def:g7:what-is-respiration:respiration}{oksigennya} lalu mengangkut pergi
\omterm{def:g7:what-is-respiration:respiration}{karbon dioksidanya} --- satu kali lari cepat, tiga bab, satu penjelasan
yang tunggal.
\end{solution}

\begin{solution}{pb:g7:organs-at-work:1}
\textbf{1.} Timbaan \omterm{def:g7:what-is-respiration:respiration}{oksigen} dan \omterm{prop:g7:organs-at-work:bill}{glukosanya} dari \omterm{prop:g4:journey-of-food:absorb}{darah} yang lewat
berlipat ganda; keluaran \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} dan keluaran panasnya naik
seiring --- tagihan pekerjaan yang keras.

\textbf{2.} Karena keduanya adalah dua tahap dari satu jalur pasokan:
\omterm{def:g7:what-is-respiration:respiration}{pernapasan} memuat ulang \omterm{prop:g4:journey-of-food:absorb}{darahnya} di \omterm{def:g7:how-animals-breathe:four}{paru-paru}, \omterm{def:g4:heart-and-blood:heart}{jantung} memindahkan
\omterm{prop:g4:journey-of-food:absorb}{darah} yang termuat ulang itu ke \omterm{def:g3:bones-and-muscles:muscle}{ototnya}. Permintaan yang dinaikkan di
satu ujung menaikkan laju keduanya.

\textbf{3.} \omterm{def:g3:bones-and-muscles:muscle}{Otot} yang bekerja adalah tungku; panasnya harus pergi.
\omterm{prop:g4:journey-of-food:absorb}{Darahnya} membawanya ke \omterm{def:g1:the-five-senses:sense}{kulit}, yang \omterm{def:g4:heart-and-blood:vessels}{pembuluhnya} melebar lalu memerah
dan memancarkan --- tungku di dalam, radiator di permukaannya.

\textbf{4.} Karena utangnya masih ada: cadangan \omterm{def:g7:what-is-respiration:respiration}{oksigen} yang harus
dimuat ulang, buangan yang harus dibersihkan, panas yang harus
dilepaskan. \omterm{prop:g4:heart-and-blood:pulse}{Denyut nadinya} turun hanya sementara rekeningnya dilunasi
--- penurunan itulah \omterm{def:g7:organs-at-work:recovery}{pemulihannya}.

\textbf{5.} Istirahat $84 \to 72$: sebuah \omterm{def:g4:heart-and-blood:heart}{jantung} yang dibangun ulang
untuk memindahkan lebih banyak tiap denyut. Usaha $168 \to 150$:
naik-turun bangku yang sama sekarang berharga lebih murah --- \omterm{def:g3:bones-and-muscles:muscle}{otot}
yang lebih kuat, pasokan \omterm{def:g4:heart-and-blood:vessels}{pembuluh} halus yang lebih kaya. \omterm{def:g7:organs-at-work:recovery}{Pemulihan}
$9 \to 5$ menit: seluruh rantainya melunasi utangnya lebih cepat.
Olahraga sepanjang triwulannyalah yang mengerjakan pembangunan
ulangnya.

\textbf{6.} Entah B sudah terlatih sejak awal triwulannya ---
bilangan yang dekat lantai terlatihnya memang hanya sedikit bergerak
--- atau B melewatkan olahraga triwulannya lalu sekadar
mempertahankan sebuah mesin tak terlatih yang mantap. Fakta yang
memutuskan: kegiatan B di luar kelas (olahraga di tempat lain akan
menegaskan pembacaan yang pertama; tanpa itu, yang kedua --- dan
bilangan minggu pertama B yang baik sudah condong ke situ).

\textbf{7.} Aturan penjaluran ulangnya: sesudah santapan, \omterm{def:g4:journey-of-food:tract}{saluran
pencernaannya} menuntut \omterm{prop:g4:journey-of-food:absorb}{darahnya}, sehingga \omterm{prop:g4:heart-and-blood:pulse}{denyut nadi} istirahatnya
terbaca tinggi dan daya tahan usahanya rendah. Tata caranya: ukurlah
sebelum makan siang, atau dua jam penuh sesudahnya --- selalu pada
jadwal yang sama.

\textbf{8.} Uji yang adil: hanya pengukuran yang dilakukan dengan cara
yang sama yang dapat dibandingkan, minggu melawan minggu dan murid
melawan murid.

\textbf{9.} ``Dua belas minggu menaikkan langit-langit A: usaha yang
sama sekarang berharga $150$ denyut alih-alih $168$, dari \omterm{prop:g4:heart-and-blood:pulse}{denyut nadi}
istirahat yang delapan denyut lebih rendah. Dan mesin A melunasi
utangnya dalam lima menit, sedangkan pada bulan September ia
membutuhkan sembilan --- \omterm{def:g7:organs-at-work:recovery}{pemulihannya} adalah keuntungan yang paling
jelas dari semuanya.''

\textbf{10.} Biasanya justru kebalikan dari mengkhawatirkan: \omterm{def:g4:heart-and-blood:heart}{jantung}
yang terlatih memindahkan lebih banyak \omterm{prop:g4:journey-of-food:absorb}{darah} tiap denyut sehingga
menganggur pada nilai yang rendah --- denyut $50$ yang tenang milik
sang pengendara sepeda mengerjakan pekerjaan denyut $75$ milik yang
tidak terlatih. (Kekhawatiran apa pun adalah urusan dokter, tetapi
rendah-dan-bugar adalah kisah yang lazim.)

\textbf{11.} Waktu \omterm{def:g7:organs-at-work:recovery}{pemulihan} mengukur seluruh rantainya --- \omterm{def:g4:heart-and-blood:heart}{jantung},
\omterm{def:g7:how-animals-breathe:four}{paru-paru}, \omterm{def:g4:heart-and-blood:vessels}{pembuluh}, \omterm{def:g3:bones-and-muscles:muscle}{otot} --- yang melunasi utang yang sungguhan, dan
ia membaik dengan mantap serta terlihat seiring latihannya; \omterm{prop:g4:heart-and-blood:pulse}{denyut
nadi} usaha lebih bergantung pada usaha dan suasana hati hari itu.
Sebagai satu bilangan yang dilacak, \omterm{def:g7:organs-at-work:recovery}{pemulihan} adalah rangkuman yang
lebih jujur.

\textbf{12.} Pemakaiannya sendiri: usaha, yang diulang pelan-pelan dan
teratur, adalah yang membangun ulang mesin yang menjalankannya.
\end{solution}
