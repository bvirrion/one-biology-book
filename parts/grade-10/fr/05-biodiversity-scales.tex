\chapter{La biodiversité à toutes les échelles}\label{ch:g10:biodiversity-scales}

Posez sur une pelouse un carré de ficelle d'un mètre de côté et
comptez ce qui pousse à l'intérieur~: de l'herbe, bien sûr, mais aussi
du trèfle, des pâquerettes, un pissenlit, du plantain, de la mousse
et, à la loupe, une douzaine d'insectes et un escargot. Déplacez le
carré sur le bitume du parking~: rien, ou une herbe folle dans une
fissure. Déplacez-le dans une vieille prairie jamais labourée~:
quarante \omterm{def:g10:biodiversity-scales:species}{espèces} de plantes, et des insectes sans nombre. Le mot qui
désigne ce que mesure le carré est \emph{\omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité}}, et ce chapitre
dit comment la mesurer, à quelles échelles elle existe, et pourquoi
elle n'est ni fixe ni garantie.

\section{Trois échelles de diversité}

\begin{definition}[Biodiversité]\label{def:g10:biodiversity-scales:biodiversity}
La \emph{biodiversité}\index{biodiversité} est la diversité des êtres
vivants, considérée à trois échelles~:
\begin{itemize}
  \item la diversité des \emph{écosystèmes}\index{écosystème} --- des
        communautés vivantes et des milieux qu'elles occupent (une
        mare, une hêtraie, un récif corallien, une prairie), chacun
        avec son propre jeu d'\omterm{def:g10:biodiversity-scales:species}{espèces}~;
  \item la diversité des \omterm{def:g10:biodiversity-scales:species}{espèces} au sein d'un
        écosystème ou sur la Terre --- le nombre d'\omterm{def:g10:biodiversity-scales:species}{espèces} et leur
        abondance relative~;
  \item la \emph{diversité génétique}\index{diversité génétique} au
        sein d'une \omterm{def:g10:biodiversity-scales:species}{espèce} --- le nombre d'\omterm{def:g10:universal-dna:gene}{allèles} de ses \omterm{def:g10:universal-dna:gene}{gènes} et la
        façon dont ils se répartissent entre ses individus et ses
        populations.
\end{itemize}
\end{definition}

\begin{definition}[Espèce]\label{def:g10:biodiversity-scales:species}
Une \emph{espèce}\index{espèce} est un groupe d'êtres vivants qui se ressemblent,
peuvent se reproduire entre eux et donner une descendance
fertile~; deux populations qui ne peuvent pas se croiser, ou dont les
hybrides sont stériles, sont des espèces distinctes. En pratique ---
pour les fossiles, pour les bactéries, pour les organismes qu'on n'a
jamais vus se reproduire --- les espèces se reconnaissent à leurs
caractères communs, et le \cref{ch:g12:selection-drift-speciation}
montrera que la frontière n'est pas toujours nette.
\end{definition}

\begin{example}[Cheval, âne, mulet]\label{ex:g10:biodiversity-scales:mule}
Un cheval et un âne se ressemblent et peuvent se reproduire ensemble~:
leur descendant est le mulet. Mais le mulet est stérile, donc le cheval
et l'âne sont deux \omterm{def:g10:biodiversity-scales:species}{espèces}. Un caniche et un lévrier irlandais ne se
ressemblent en rien, se reproduisent volontiers et donnent des chiots
fertiles~: une seule \omterm{def:g10:biodiversity-scales:species}{espèce}, le chien --- qui peut aussi se reproduire
avec le loup, son ancêtre.
\end{example}

\section{La diversité des espèces}

\begin{proposition}[Compter les espèces]\label{prop:g10:biodiversity-scales:count}
Environ deux millions d'\omterm{def:g10:biodiversity-scales:species}{espèces} ont été décrites et nommées. La plupart
sont des insectes (plus d'un million), puis viennent les plantes
(quelque 300\,000), les champignons, les arachnides, les mollusques~;
les vertébrés, le groupe que nous connaissons le mieux, sont environ
70\,000, dont 6\,400 mammifères. Le nombre d'\omterm{def:g10:biodiversity-scales:species}{espèces} encore non
décrites est estimé, d'après le rythme auquel on en découvre de
nouvelles, à plusieurs millions de plus --- peut-être huit à dix
millions d'\omterm{def:g10:biodiversity-scales:species}{espèces} eucaryotes en tout, plus une immense diversité de
bactéries, presque entièrement non recensée.
\end{proposition}

\begin{method}[Mesurer la diversité d'un site]\label{met:g10:biodiversity-scales:sampling}
Un site ne se compte jamais en entier~; on l'\emph{échantillonne}.
\begin{enumerate}
  \item Choisir une unité d'échantillonnage adaptée aux organismes~: un
        quadrat de \qty{1}{m^2} pour les plantes d'une prairie, un coup
        de filet fauchoir pour les insectes, un tamisage de sol pour
        les vers de terre, une longueur de transect pour les oiseaux.
  \item Placer les unités au hasard ou à intervalles réguliers, en
        nombre suffisant pour qu'une unité de plus change peu le total.
  \item Dans chacune, identifier et compter. Deux nombres décrivent
        l'échantillon~: la \emph{richesse spécifique}, le nombre
        d'\omterm{def:g10:biodiversity-scales:species}{espèces} trouvées, et l'\emph{abondance relative} de chacune
        --- un site où une \omterm{def:g10:biodiversity-scales:species}{espèce} fait 95~\% des individus est moins
        divers qu'un site où dix \omterm{def:g10:biodiversity-scales:species}{espèces} se les partagent également, à
        richesse égale.
  \item Ne comparer des sites qu'avec la même méthode et le même
        effort, à la même saison.
\end{enumerate}
\end{method}

\begin{example}[Deux pelouses]\label{ex:g10:biodiversity-scales:lawns}
Dix quadrats sur une pelouse de collège tondue~: 6 \omterm{def:g10:biodiversity-scales:species}{espèces} de plantes,
dont une graminée qui couvre 90~\%. Dix quadrats sur un talus non tondu
juste à côté~: 23 \omterm{def:g10:biodiversity-scales:species}{espèces}, aucune au-dessus de 30~\%. Le talus est plus
riche et plus équilibré~; la différence ne tient pas au sol ni au
climat, qui sont les mêmes, mais à la tonte, qui élimine toute plante
incapable de fleurir au ras du sol.
\end{example}

\section{La diversité génétique au sein d'une espèce}

\begin{proposition}[Les individus d'une espèce ne sont pas identiques]\label{prop:g10:biodiversity-scales:genetic}
Au sein d'une \omterm{def:g10:biodiversity-scales:species}{espèce}, la plupart des \omterm{def:g10:universal-dna:gene}{gènes} existent en plusieurs
\omterm{def:g10:universal-dna:gene}{allèles}, et les individus en portent des combinaisons différentes. La
\emph{fréquence} d'un \omterm{def:g10:universal-dna:gene}{allèle} dans une population --- la fraction de
tous les exemplaires du \omterm{def:g10:universal-dna:gene}{gène} qui sont cet \omterm{def:g10:universal-dna:gene}{allèle} --- varie d'une
population à l'autre. La \omterm{def:g10:biodiversity-scales:biodiversity}{diversité génétique} est ce qui permet à une
\omterm{def:g10:biodiversity-scales:species}{espèce} de répondre à un environnement changeant~: une population dont
tous les individus se ressemblent n'offre rien à sélectionner.
\end{proposition}
\admitted

\begin{example}[Les allèles des groupes sanguins dans le monde]\label{ex:g10:biodiversity-scales:abo}
Le \omterm{def:g10:universal-dna:gene}{gène} des groupes sanguins ABO a trois \omterm{def:g10:universal-dna:gene}{allèles} courants, A, B et O.
En Europe de l'Ouest, leurs fréquences valent environ 0.27, 0.07 et
0.66~; en Asie de l'Est, environ 0.20, 0.20 et 0.60~; dans certaines
populations autochtones d'Amérique du Sud, l'\omterm{def:g10:universal-dna:gene}{allèle} O atteint à lui
seul 0.98. Même \omterm{def:g10:biodiversity-scales:species}{espèce}, même \omterm{def:g10:universal-dna:gene}{gène}, distributions d'\omterm{def:g10:universal-dna:gene}{allèles}
différentes~: la \omterm{def:g10:biodiversity-scales:biodiversity}{diversité génétique} de l'humanité est réelle, et elle
est répartie entre les populations plutôt qu'elle ne les sépare ---
l'essentiel de la diversité de l'\omterm{def:g10:biodiversity-scales:species}{espèce} se retrouve à l'intérieur de
n'importe laquelle d'entre elles.
\end{example}

\begin{example}[Une espèce qui n'en a presque plus]\label{ex:g10:biodiversity-scales:cheetah}
Les guépards se ressemblent tant génétiquement que des greffes de peau
entre animaux sans lien de parenté sont acceptées comme entre jumeaux~:
l'\omterm{def:g10:biodiversity-scales:species}{espèce} est passée par un goulot d'étranglement de très peu
d'individus il y a une dizaine de milliers d'années et n'a jamais
retrouvé ses \omterm{def:g10:universal-dna:gene}{allèles}. Quelle que soit la nouvelle maladie ou le nouveau
climat, chaque guépard l'affronte avec le même équipement~; l'\omterm{def:g10:biodiversity-scales:species}{espèce}
survit, mais sans rien en réserve.
\end{example}

\section{La biodiversité change~: passé, présent}

\begin{proposition}[La biodiversité est un état, non une constante]\label{prop:g10:biodiversity-scales:changes}
L'ensemble des \omterm{def:g10:biodiversity-scales:species}{espèces} vivantes à un instant donné est une étape d'une
histoire. Le registre fossile montre des \omterm{def:g10:biodiversity-scales:species}{espèces} qui apparaissent,
persistent quelques millions d'années en moyenne et disparaissent~; la
diversité totale du vivant a augmenté au cours des 540 derniers
millions d'années, interrompue par cinq
\emph{extinctions de masse}\index{extinction de masse} au cours
desquelles plus des trois quarts des \omterm{def:g10:biodiversity-scales:species}{espèces} se sont évanouies en un
temps géologiquement court. La plus récente, il y a 66 millions
d'années, a mis fin aux dinosaures non aviens et ouvert la voie à la
diversification des mammifères.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Les décomptes de familles fossiles dans des couches de roche datées,
compilés pour tout le registre marin, donnent la courbe de la figure
suivante~: une montée de quelques centaines à près d'un millier de
familles, avec des chutes brutales il y a 445, 370, 252, 201 et 66
millions d'années. L'événement d'il y a 252 millions d'années, le plus
grand, a supprimé environ 95~\% des \omterm{def:g10:biodiversity-scales:species}{espèces} marines~; les couches juste
au-dessus sont presque vides de fossiles, et de nouveaux groupes ne les
emplissent qu'au fil des millions d'années suivants.
\end{proof}

\begin{proposition}[Le changement actuel]\label{prop:g10:biodiversity-scales:present}
Les \omterm{def:g10:biodiversity-scales:species}{espèces} disparaissent aujourd'hui à un rythme estimé à cent ou
mille fois la moyenne du registre fossile, du fait de la destruction
des \omterm{def:g10:biodiversity-scales:biodiversity}{écosystèmes} (défrichement des forêts, drainage des zones humides,
urbanisation), de la surexploitation, des \omterm{def:g10:biodiversity-scales:species}{espèces} introduites, des
pollutions et du changement climatique --- tous d'origine humaine. La
moitié des forêts tropicales de 1950 a disparu~; le tiers des \omterm{def:g10:biodiversity-scales:species}{espèces}
d'amphibiens est menacé~; la \omterm{def:g10:biodiversity-scales:biodiversity}{diversité génétique} des cultures et des
élevages s'est resserrée sur quelques variétés. La \omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité} est
une ressource --- nourriture, médicaments, pollinisation, sol,
stabilité des \omterm{def:g10:biodiversity-scales:biodiversity}{écosystèmes} --- et sa perte n'est pas réversible à
l'échelle d'une vie humaine.
\end{proposition}
\admitted

\begin{example}[Réparer une perte]\label{ex:g10:biodiversity-scales:reversing}
Là où des haies arrachées dans les années 1960 ont été replantées, la
richesse en oiseaux et en insectes revient en une décennie~; là où des
loups ont été réintroduits dans un parc national en 1995, les cerfs
qu'ils chassent ont cessé de raser les saules des berges, les saules
ont repoussé, les castors sont revenus, et avec eux les oiseaux et les
poissons qui dépendent des étangs de castors. Protéger ou restaurer un
\omterm{def:g10:biodiversity-scales:biodiversity}{écosystème} restaure la diversité aux trois échelles à la fois~; une
\omterm{def:g10:biodiversity-scales:species}{espèce} maintenue en vie dans un zoo n'en garde qu'une.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:1}
Nommer les trois échelles de la \omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité} et donner un exemple de
chacune.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:2}
Définir une \omterm{def:g10:biodiversity-scales:species}{espèce}. Le lion et le tigre --- qui peuvent donner des
hybrides stériles en captivité --- font-ils une \omterm{def:g10:biodiversity-scales:species}{espèce} ou deux~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:3}
D'après le graphique des \omterm{def:g10:biodiversity-scales:species}{espèces}, combien de fois y a-t-il plus
d'\omterm{def:g10:biodiversity-scales:species}{espèces} d'insectes décrites que d'\omterm{def:g10:biodiversity-scales:species}{espèces} de mammifères~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:4}
Qu'est-ce qu'une fréquence allélique~? Dans une population où l'\omterm{def:g10:universal-dna:gene}{allèle}
O a une fréquence de 0.66 et l'\omterm{def:g10:universal-dna:gene}{allèle} A de 0.27, quelle est la
fréquence de B~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:biodiversity-scales:5}
Combien d'\omterm{prop:g10:biodiversity-scales:changes}{extinctions de masse} le registre fossile montre-t-il, et
laquelle fut la plus grande~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:6}
Deux prairies comptent chacune 12 \omterm{def:g10:biodiversity-scales:species}{espèces} de plantes. Dans la première,
une \omterm{def:g10:biodiversity-scales:species}{espèce} fait 80~\% des plantes~; dans la seconde, aucune ne dépasse
15~\%. Laquelle est la plus diverse, et lequel des deux nombres de la
\cref{met:g10:biodiversity-scales:sampling} les distingue~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:7}
Un élève compte les insectes de 3 coups de filet et trouve 8 \omterm{def:g10:biodiversity-scales:species}{espèces}~;
un camarade fait 30 coups de filet dans le même champ et en trouve 21.
Le champ a-t-il plus d'\omterm{def:g10:biodiversity-scales:species}{espèces} pour le camarade~? Expliquer ce que la
différence montre au sujet de l'échantillonnage.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:8}
Expliquer pourquoi une \omterm{def:g10:biodiversity-scales:species}{espèce} réduite à quelques dizaines d'individus
peut rester en danger même après que ses effectifs sont remontés à
plusieurs milliers.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:9}
Sur la figure des familles fossiles, lire le nombre de familles juste
avant et juste après l'extinction d'il y a 252 millions d'années, et
calculer la fraction perdue. Pourquoi la fraction d'\emph{\omterm{def:g10:biodiversity-scales:species}{espèces}}
perdues est-elle plus grande que la fraction de familles~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:10}
À l'aide de l'\cref{ex:g10:biodiversity-scales:abo}, expliquer en quel
sens deux voisins d'une même ville diffèrent génétiquement à peu près
autant que deux personnes de continents différents.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:biodiversity-scales:11}
Une mare est asséchée pour construire un parking. Citer une perte à
chacune des trois échelles de la \omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:biodiversity-scales:12}
L'\omterm{def:g10:biodiversity-scales:species}{espèce} moyenne du registre fossile dure quelques millions d'années~;
avec deux millions d'\omterm{def:g10:biodiversity-scales:species}{espèces} connues, estimer le nombre
d'extinctions par an du «~bruit de fond~». Si le rythme actuel est
mille fois plus élevé, combien d'\omterm{def:g10:biodiversity-scales:species}{espèces} disparaîtraient par an, et
combien par siècle~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:biodiversity-scales:13}
Après l'extinction d'il y a 66 millions d'années, les mammifères se
sont diversifiés à partir de quelques formes de petite taille en
baleines, chauves-souris, éléphants et primates en quelque 20 millions
d'années. Expliquer comment une catastrophe pour la \omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité} peut
aussi être une cause de \omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité} nouvelle.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:biodiversity-scales:14}
Le blé moderne descend d'une poignée de variétés~; ses parents
sauvages poussent dans les montagnes du Proche-Orient. Expliquer, avec
la notion de \omterm{def:g10:biodiversity-scales:biodiversity}{diversité génétique}, pourquoi les banques de graines
conservent les parents sauvages alors qu'ils ne rendent rien
d'utilisable.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:biodiversity-scales:15}

\section{Problème~: le dividende de la haie}

\begin{problem}[{Problème du week-end --- un relevé de terrain sur deux
fermes, l'une avec haies et l'autre sans~: espèces comptées, abondances
comparées, allèles d'un escargot suivis, et ce que vaut une haie}]
\label{pb:g10:biodiversity-scales:1}
Deux fermes voisines cultivent le même blé sur le même sol. La ferme A
a gardé ses haies~; la ferme B les a arrachées en 1970. Une classe
échantillonne les deux en juin avec la même méthode~: 20 quadrats de
\qty{1}{m^2} le long des bords de champ pour les plantes, 20 coups de
filet fauchoir pour les insectes, et 10 minutes d'écoute en 5 points
pour les oiseaux.

\medskip
\noindent\textbf{Partie I --- Les plantes.}
Résultats, ferme A~: 31 \omterm{def:g10:biodiversity-scales:species}{espèces} de plantes dans les 20 quadrats~;
ferme B~: 9. Sur la ferme B, une graminée fait 84~\% des plantes
comptées~; sur la ferme A, l'\omterm{def:g10:biodiversity-scales:species}{espèce} la plus commune en fait 22~\%.
\begin{enumerate}
  \item Donner la richesse spécifique des bords de champ de chaque
        ferme et le rapport entre les deux.
  \item Quelle communauté végétale est la plus équilibrée~? Expliquer
        ce qu'«~équilibrée~» ajoute à «~riche~».
  \item Sur la ferme A, le nombre d'\omterm{def:g10:biodiversity-scales:species}{espèces} trouvées après 5, 10, 15
        et 20 quadrats était de 18, 25, 29 et 31. L'échantillonnage
        est-il suffisant~? Justifier.
  \item La même série sur la ferme B était 6, 8, 9, 9. Comparer les
        deux séries et dire quelle forme prend une courbe
        d'échantillonnage «~complète~».
  \item Pourquoi les deux fermes doivent-elles être échantillonnées le
        même mois~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Insectes et oiseaux.}
Insectes~: ferme A 46 \omterm{def:g10:biodiversity-scales:species}{espèces}, ferme B 14. Oiseaux~: ferme A 17
\omterm{def:g10:biodiversity-scales:species}{espèces}, ferme B 5~; sur la ferme B, 60~\% des oiseaux entendus
appartenaient à une seule \omterm{def:g10:biodiversity-scales:species}{espèce}.
\begin{enumerate}[resume]
  \item Calculer le rapport des richesses en insectes et en oiseaux
        entre les fermes. Les trois groupes (plantes, insectes,
        oiseaux) racontent-ils la même histoire~?
  \item Proposer une chaîne de causes reliant l'arrachage des haies à
        la baisse de la diversité en insectes, puis à celle des
        oiseaux.
  \item Le rendement du blé à l'hectare est le même sur les deux
        fermes. Cela signifie-t-il que les haies n'ont aucun effet sur
        la culture~? Proposer un service que peut rendre une haie et
        que le chiffre du rendement ne montre pas.
  \item Citer une \omterm{def:g10:biodiversity-scales:species}{espèce} que la classe s'attendrait à ne trouver que
        sur la ferme A et expliquer pourquoi.
  \item Quelle échelle de \omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité} a été la plus manifestement
        réduite sur la ferme B~: \omterm{def:g10:biodiversity-scales:biodiversity}{écosystèmes}, \omterm{def:g10:biodiversity-scales:species}{espèces} ou \omterm{def:g10:universal-dna:gene}{gènes}~?
        Justifier.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Les \omterm{def:g10:universal-dna:gene}{allèles} d'un escargot.}
L'escargot des haies porte un \omterm{def:g10:universal-dna:gene}{gène} à deux \omterm{def:g10:universal-dna:gene}{allèles} courants pour la
couleur de la coquille, jaune (j) et brun (b). Dans les haies de la
ferme A, 400 escargots ont été échantillonnés~: 100 ont deux \omterm{def:g10:universal-dna:gene}{allèles}
j, 200 ont un j et un b, 100 ont deux b. Sur le seul talus restant de
la ferme B, 100 escargots~: 4 avec deux j, 32 avec un de chaque, 64
avec deux b.
\begin{enumerate}[resume]
  \item Compter les exemplaires de j et de b dans chaque échantillon
        (chaque escargot en porte deux). En déduire la fréquence de j
        sur chaque ferme.
  \item Quelle population est la plus diverse génétiquement pour ce
        \omterm{def:g10:universal-dna:gene}{gène}~? Comparer la proportion d'escargots porteurs des deux
        \omterm{def:g10:universal-dna:gene}{allèles}.
  \item Les grives chassent les escargots à vue~; sur le talus nu de
        la ferme B, les coquilles brunes sont mieux dissimulées.
        Proposer une explication de l'origine des fréquences de la
        ferme B.
  \item Si une haie humide et ombragée favorisait les coquilles
        jaunes, que prévoiriez-vous pour les escargots de la ferme B
        si les haies étaient replantées et que rien d'autre ne
        changeait~?
  \item Les escargots de la ferme B sont quelques centaines, ceux de
        la ferme A des dizaines de milliers. Quelle population est la
        plus exposée à perdre un \omterm{def:g10:universal-dna:gene}{allèle} par le seul hasard~? Pourquoi~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Le dividende.}
\begin{enumerate}[resume]
  \item Sur les trois groupes, calculer la richesse spécifique totale
        de chaque ferme, puis le total de la ferme A divisé par celui
        de la ferme B.
  \item Les haies de la ferme A occupent 3~\% de sa surface. Énoncer
        en une phrase quelle fraction du terrain détient quelle
        fraction des \omterm{def:g10:biodiversity-scales:species}{espèces}.
  \item Des haies replantées sur la ferme B en 1995 ont été relevées à
        nouveau en 2010~: plantes 24, insectes 35, oiseaux 12 \omterm{def:g10:biodiversity-scales:species}{espèces}.
        Quelle fraction de la richesse de la ferme A chaque groupe
        avait-il retrouvée~?
  \item Laquelle des trois échelles de la \omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité} se rétablit le
        plus lentement après une replantation, et pourquoi~?
  \item Énoncer le résultat~: le facteur par lequel les haies
        multiplient la richesse spécifique d'une ferme, et la seule
        échelle de \omterm{def:g10:biodiversity-scales:biodiversity}{biodiversité} qu'une haie replantée ne peut pas
        ramener à elle seule.
\end{enumerate}
\end{problem}
