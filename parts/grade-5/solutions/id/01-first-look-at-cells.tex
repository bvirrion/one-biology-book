\section*{Bab \ref{ch:g5:first-look-at-cells} --- Pandangan Pertama pada Sel}

\begin{solution}{exo:g5:first-look-at-cells:1}
Balok pembangun hidup yang terkecil pada sebuah \omterm{def:g1:living-or-not:living}{makhluk hidup}. Tubuhmu
tersusun atas puluhan ribu miliar \omterm{def:g5:first-look-at-cells:cell}{sel}.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:2}
Karena \omterm{def:g5:first-look-at-cells:cell}{sel} lebih kecil daripada bintik terkecil yang dapat dilihat
\omterm{def:g1:the-five-senses:sense}{mata}. Mikroskoplah --- yang memperbesar irisan tipis ratusan kali ---
yang menyingkapkannya.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:3}
\omterm{def:g1:the-five-senses:sense}{Kulit} bawang bombai: \omterm{def:g5:first-look-at-cells:cell}{sel} mirip bata dalam deretan rapi, dinding beradu
dinding. Kerokan pipi: \omterm{def:g5:first-look-at-cells:cell}{sel} yang lebih bulat, tersebar. Persamaannya:
kedua bahan itu tersusun sepenuhnya atas \omterm{def:g5:first-look-at-cells:cell}{sel}, dan kebanyakan \omterm{def:g5:first-look-at-cells:cell}{selnya}
memperlihatkan sebuah bintik gelap kecil di dalamnya.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:4}
Inti \omterm{def:g5:first-look-at-cells:cell}{sel} --- pusat komando \omterm{def:g5:first-look-at-cells:cell}{selnya}.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:5}
Kira-kira $50$ \omterm{def:g5:first-look-at-cells:cell}{sel} membentang sepanjang satu milimeter; satu \omterm{def:g5:first-look-at-cells:cell}{sel}
berukuran kira-kira seperlima puluh milimeter --- $\qty{0.02}{mm}$.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:6}
Bertambah tinggi: \omterm{def:g5:first-look-at-cells:cell}{sel} tubuhnya sedang membelah dan berlipat ganda.
Menyembuh: \omterm{def:g5:first-look-at-cells:cell}{selnya} sedang membelah untuk membangun ulang petak yang
rusak.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:7}
Satu. \omterm{def:g5:first-look-at-cells:cell}{Sel} itu adalah \omterm{def:g4:animal-reproduction:sexual}{sel telur} yang terbuahi --- penggabungan satu \omterm{def:g4:animal-reproduction:sexual}{sel
sperma} dan satu \omterm{def:g4:animal-reproduction:sexual}{sel telur}.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:8}
Penggandaan dari $1$ memberi $2$, $4$, $8$, $16$, $32$: ada $32$ \omterm{def:g5:first-look-at-cells:cell}{sel}
sesudah $5$ pembelahan. Lima penggandaan lagi memberi $64$, $128$,
$256$, $512$, $1024$: jadi $1024$ \omterm{def:g5:first-look-at-cells:cell}{sel} sesudah $10$ pembelahan.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:9}
Sebeda apa pun dalam ukuran, bentuk dan kehidupannya, keduanya
dibangun dari jenis satuan hidup kecil yang sama --- \omterm{def:g5:first-look-at-cells:cell}{sel}. Rancangan
pembangunan bersama itu, yang dimiliki semua \omterm{def:g1:living-or-not:living}{makhluk hidup}, adalah
arti ``\omterm{prop:g5:first-look-at-cells:principle}{kesatuan kehidupan}''.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:10}
Semuanya: sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal lahir (dari \omterm{def:g5:first-look-at-cells:cell}{sel} induk yang membelah),
makan, tumbuh, membelah --- membuat \omterm{def:g1:living-or-not:living}{makhluk hidup} baru seperti dirinya
--- lalu mati. Satu \omterm{def:g5:first-look-at-cells:cell}{sel} sudah cukup untuk sebuah kehidupan yang
lengkap.
\end{solution}

\begin{solution}{exo:g5:first-look-at-cells:11}
\omterm{def:g5:first-look-at-cells:cell}{Sel} yang lebih banyak --- ukuran \omterm{def:g5:first-look-at-cells:cell}{sel} nyaris tidak berubah antara tikus
dan gajah. Kalau begitu, tumbuh bukanlah menggelembungkan \omterm{def:g5:first-look-at-cells:cell}{sel}
melainkan melipatgandakannya: tubuh menjadi besar lewat pembelahan
demi pembelahan, persis seperti kata proposisinya.
\end{solution}
