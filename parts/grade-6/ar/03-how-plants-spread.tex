\chapter{كيف تنتشر النباتات}\label{ch:g6:how-plants-spread}

يُجرَف حفر طريق جديد في آذار: تراب عارٍ خام، لا ذرّة خضراء فيه.
ثم عُد في أيلول فإذا هو مطرَّز كله بالبادرات --- خشخاش، وهندباء،
وأفرخ صفصاف، وأعشاب. لم يبذره أحد. ومع ذلك استوطنت \omterm{def:g1:plants-around-us:plant}{النباتات}، وهي
لا تستطيع أن تمشي خطوة واحدة، الأرض العارية في شهور. وهذا الفصل عن
قسم السفر عندها.

\section{النباتات تسافر بذورًا وأبواغًا}

\begin{proposition}[نباتات ثابتة وبدايات مسافرة]\label{prop:g6:how-plants-spread:travel}
\omterm{def:g1:plants-around-us:plant}{النبات} البالغ ثابت مدى الحياة
(\cref{def:g1:plants-around-us:plant}) --- \omterm{def:g5:biodiversity:species}{فالنوع} ينتشر بإرسال
\emph{بداياته}: \omterm{def:g2:from-seed-to-plant:seed}{البذور} \omterm{prop:g3:flowers-fruits-seeds:transform}{والثمار} (\cref{ch:g5:flower-to-fruit})، أو
\omterm{def:g4:plants-without-seeds:spore}{الأبواغ} (\cref{def:g4:plants-without-seeds:spore})، أو \omterm{ex:g4:plants-without-seeds:runner}{المدّادات}
وسائر الزحوف الخضرية
(\cref{def:g4:plants-without-seeds:vegetative}). ويسمى احتلال أرض
جديدة على هذا النحو \emph{الاستيطان}\index{استيطان}.
\end{proposition}
\admitted

\begin{definition}[الانتثار]\label{def:g6:how-plants-spread:dispersal}
\emph{الانتثار}\index{انتثار} هو حمل \omterm{def:g2:from-seed-to-plant:seed}{البذور} أو \omterm{prop:g3:flowers-fruits-seeds:transform}{الثمار} أو \omterm{def:g4:plants-without-seeds:spore}{الأبواغ}
بعيدًا عن \omterm{def:g1:plants-around-us:plant}{النبات} الأم. والحاملون قليلون قدماء: \emph{الريح}،
و\emph{\omterm{def:g1:animals-around-us:animal}{الحيوانات}}، و\emph{الماء} --- وبعض \omterm{prop:g3:flowers-fruits-seeds:transform}{الثمار} التي تقذف بذورها
بنفسها.
\end{definition}

\begin{example}[ولماذا مغادرة البيت أصلًا؟]\label{ex:g6:how-plants-spread:whyleave}
\omterm{def:g2:from-seed-to-plant:seed}{البذرة} الساقطة عند قدم أمها \omterm{def:g2:from-seed-to-plant:germination}{تنبت} في ظل أمها، بين \omterm{def:g1:plants-around-us:parts}{جذور} أمها ---
منافسةً على \omterm{def:g6:exploring-our-environment:conditions}{الضوء} والماء نفسهما في
\cref{prop:g4:what-plants-need:needs} خصمًا يفوقها حجمًا مئة مرة.
\omterm{def:g6:how-plants-spread:dispersal}{والانتثار} هو الفرار من تلك المباراة الخاسرة: فالمسافة أول هدية
يعطيها \omterm{def:g1:plants-around-us:plant}{النبات} لصغاره.
\end{example}

\section{الحاملون والثمار المبنية لهم}

\begin{proposition}[مبنية لتناسب الحامل]\label{prop:g6:how-plants-spread:builds}
أشكال \omterm{prop:g3:flowers-fruits-seeds:transform}{الثمار} \omterm{def:g2:from-seed-to-plant:seed}{والبذور} تناسب حامليها --- وهي ملاءمة
\cref{def:g4:adapted-to-their-world:adaptation} مطبَّقة على
الأمتعة:
\begin{itemize}
  \item \emph{الريح}: مظلات (\omterm{def:g2:from-seed-to-plant:seed}{كبذور} الهندباء المريَّشة)، أو أجنحة
        (كسمارات القيقب والدردار تدور نازلة مائلة)، أو \omterm{def:g4:plants-without-seeds:spore}{أبواغ} في
        نعومة الغبار تطفو ببساطة (كملايين \omterm{def:g4:plants-without-seeds:spore}{أبواغ} السرخس)؛
  \item و\emph{\omterm{def:g1:animals-around-us:animal}{الحيوانات} من الخارج}: الحسك --- كالأرقطيون
        واللزّيق --- وخطاطيفه تمسك \omterm{def:g1:animals-around-us:covering}{الفرو} والجوارب وتركب حتى
        تُنزع؛
  \item و\emph{\omterm{def:g1:animals-around-us:animal}{الحيوانات} من الداخل}: \omterm{prop:g3:flowers-fruits-seeds:transform}{الثمار} اللحمية الحلوة ---
        كالكرز والعليق وثمر الورد --- تُؤكل كلها؛ فتمرّ \omterm{def:g2:from-seed-to-plant:seed}{البذور}
        الغليظة القشرة سالمة وتُوضع في مكان آخر مع سماد
        (وهو جواب \cref{exo:g3:flowers-fruits-seeds:10} مكتملًا)؛
  \item و\emph{الماء}: \omterm{prop:g3:flowers-fruits-seeds:transform}{الثمار} الطافية --- كأرماث الجار الصغيرة،
        وقشرة جوز الهند العابرة للمحيط؛
  \item و\emph{القذف الذاتي}: قرون تجفّ فتشتدّ ثم تنفقع ---
        فالوزّال والبنفسج يقذفان بذورهما أمتارًا عن الأم.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[قراءة جانب الطريق]\label{ex:g6:how-plants-spread:roadside}
حفر الطريق في أيلول، مفكوكًا: الهندباء والصفصاف --- \omterm{def:g2:from-seed-to-plant:seed}{بذور} مريَّشة
طارت مع الريح؛ والخشخاش --- بذوره الصغيرة كانت في التراب المنقول
منذ الآن، تنتظر كما تفعل \omterm{def:g2:from-seed-to-plant:seed}{البذور} وحدها
(\cref{exo:g2:from-seed-to-plant:9})؛ وبادرات العليق تحت أعمدة
السياج --- وضعتها طيور جاثمة، والسماد معها؛ والأرقطيون على الممر
--- حمله كلب. فطريق وصول كل مستوطن مكتوب في عتاده.
\end{example}

\begin{method}[تسمية الحامل من الثمرة]\label{met:g6:how-plants-spread:name}
إذا أُعطيت \omterm{prop:g3:flowers-fruits-seeds:transform}{ثمرة} أو \omterm{def:g2:from-seed-to-plant:seed}{بذرة}:
\begin{enumerate}
  \item ابحث عن عتاد طيران --- \omterm{def:g1:animals-around-us:covering}{ريش}، أو أجنحة، أو نعومة غبارية:
        فهي الريح؛
  \item وابحث عن خطاطيف: فهي \omterm{def:g1:animals-around-us:covering}{فرو} \omterm{def:g1:animals-around-us:animal}{الحيوان} من الخارج؛
  \item ألحمية ملونة حلوة؟ فهي أمعاء \omterm{def:g1:animals-around-us:animal}{الحيوان} من الداخل ---
        وتحقق أن \omterm{def:g2:from-seed-to-plant:seed}{للبذور} قشورًا غليظة؛
  \item أطافية قرب الماء؟ فهو الماء؛
  \item أقرن جاف ملتوٍ مشدود؟ فهو القذف الذاتي؛
  \item واختبر حكمك: أسقطها، وأطفها، واضغطها على كمّ صوفي.
\end{enumerate}
\end{method}

\section{الاستيطان بلا بذور}

\begin{example}[الزحف الأرضي]\label{ex:g6:how-plants-spread:vegetative}
إلى جانب الواصلين جوًّا، يمشي بعض المستوطنين على أقدامهم الخضرية
(\cref{ch:g4:plants-without-seeds}): فالنجيل يزحف تحت الأرض
\omterm{def:g1:plants-around-us:parts}{بسيقان} دابّة، مترًا بعد متر؛ والعليق يتقوّس ويتجذّر حيث تلامس
أطرافه؛ \omterm{ex:g4:plants-without-seeds:runner}{ومدّادات} الفراولة تقفز على الأرض المفتوحة. بطيء قصير
المدى --- ولا يُوقف في حديقة مهملة، كما يعرف كل بستاني.
\end{example}

\begin{remark}[استيطان بسرعتين]\label{rem:g6:how-plants-spread:twospeeds}
يتقاسم النظامان العمل. \omterm{def:g2:from-seed-to-plant:seed}{فالبذور} \omterm{def:g4:plants-without-seeds:spore}{والأبواغ} خدمة \emph{بعيدة المدى}
--- هبوطات نادرة موفَّقة في البعيد، والهندباء تعبر المدينة. والزحف
الخضري خدمة \emph{قصيرة المدى} --- احتلال كثيف مضمون للأرض
المجاورة. ورقعة العليق المنتشرة من السياج تستعمل الاثنتين معًا:
سيقانًا متقوّسة للمتر، وبذورًا تحملها \omterm{prop:g3:sorting-living-things:groups}{الطيور} للميل.
\end{remark}

\section{مسألة: الحقل المهجور}

\begin{problem}[{مسألة نهاية الأسبوع --- عشر سنوات من
الاستيطان، مراقَبة من بوابة واحدة}]\label{pb:g6:how-plants-spread:1}
يُهجر حقل محروث في الخريف. فيصوّره عالم طبيعة من البوابة كل حزيران
عشر سنوات، ويحتفظ بملاحظات.

\medskip\noindent\textbf{القسم الأول --- السنة الأولى.}
\begin{enumerate}
  \item في حزيران السنة الأولى يكون التراب العاري منقّطًا بالخشخاش
        والهندباء والشوك. سمِّ لكل واحد المصدر المرجّح والطريق ---
        الخشخاش من التربة نفسها، والآخران من الجو. وأي عتاد جعل
        الوصول الجوي ممكنًا؟
  \item لا يظهر عليق في وسط الحقل في السنة الأولى، لكن ثلاث
        بادرات تقف تحت أعمدة السياج القديمة. اشرح النمط بخدمة
        الداخل الحيوانية --- ولماذا تحت الأعمدة بالضبط؟
  \item لا يجد عالم الطبيعة سرخسيات في أي مكان، مع أن حرشًا مليئًا
        بها يقع في اتجاه الريح. وقد وصلت \omterm{def:g4:plants-without-seeds:spore}{أبواغ} السرخس يقينًا
        (وقد عدّت \cref{ex:g4:plants-without-seeds:fern} ملايينها).
        فأي شرط من \omterm{def:g6:exploring-our-environment:conditions}{شروط} الحقل المفتوح رفضها وفق
        \cref{prop:g4:plants-without-seeds:damp}؟
  \item يغزو النجيل من حافة الحقل وحدها، مترًا أو مترين في السنة.
        فأي نظام استيطان هذا، ولماذا لا يستطيع أن يظهر في وسط
        الحقل كما ظهرت الهندباء؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- السنوات التالية.}
\begin{enumerate}[resume]
  \item بحلول السنة الثالثة تكون الأعشاب \omterm{def:g1:plants-around-us:plant}{والنباتات} العشبية
        الطويلة قد أغلقت الأرض العارية، ويكون الخشخاش قد اختفى من
        الغطاء القائم. فأين الخشخاش الآن، وبأي صورة؟ (تذكّر حفر
        الطريق و\cref{prop:g6:life-through-seasons:herbs}.)
  \item ومن السنة الرابعة تنقّط أقواس العليق وأشجار الزعرور
        الصغيرة الحقل، وهي أكثفها قرب الأسيجة وتحت خط السياج.
        سمِّ الخدمتين اللتين وضعتهما، والدليل في النمط.
  \item وبحلول السنة الثامنة تقف غراس الصفصاف والبتولا أطول من
        العليق --- وكلتا الشجرتين ذات \omterm{def:g2:from-seed-to-plant:seed}{بذور} صغيرة مريَّشة أو خفيفة
        كالغبار. فلماذا يظل البلوط والكستناء ذوا \omterm{def:g2:from-seed-to-plant:seed}{البذور} الثقيلة
        غائبين تقريبًا، وأي ساعٍ حيواني يغرسهما في النهاية؟
  \item تُظهر الصور طاقم الحقل يتغير سنة بعد سنة مع أن بستانيًّا لم
        يمسّه. اذكر القاعدة المحركة: ما الذي يغيّره غطاء كل سنة في
        \omterm{def:g6:exploring-our-environment:conditions}{شروط} الحقل نفسه --- \omterm{def:g6:exploring-our-environment:conditions}{والضوء} عند سطح الأرض قبل كل شيء ---
        ولمن يرجّح ذلك التغير بعدُ؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- الدرس العام.}
\begin{enumerate}[resume]
  \item رتّب المستوطنين بحسب الوصول: \omterm{def:g2:from-seed-to-plant:seed}{بذور} التربة المنتظرة،
        فراكبو الريح، فالشجيرات التي تحملها \omterm{def:g1:animals-around-us:animal}{الحيوانات}، فالأشجار
        ذات \omterm{def:g2:from-seed-to-plant:seed}{البذور} الثقيلة. واقرن كل موجة بخدمة \omterm{def:g6:how-plants-spread:dispersal}{الانتثار} التي
        وقّتتها.
  \item شقوق رصيف ساحة مدينة تُنبت طاقم السنة الأولى نفسه ---
        خشخاشًا وهندباء --- لكنها لا تبلغ شجيرات السنة الرابعة
        قط. أعطِ سببين لتوقف التعاقب هناك.
  \item يزعم عالم الطبيعة: ``إذا تُرك هذا الحقل وشأنه صار حرشًا.''
        دافع عن الزعم باتجاه السنوات العشر --- وسمِّ استثناء
