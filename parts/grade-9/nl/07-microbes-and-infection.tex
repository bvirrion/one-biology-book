\chapter{Microben en infectie}\label{ch:g9:microbes-and-infection}

Ze schaduwen dit boek al sinds de onzichtbare \omterm{def:g1:healthy-habits:germs}{ziektekiemen} van
groep drie (\cref{def:g1:healthy-habits:germs}): opgeklommen tot
personeel in de bakkerij
(\cref{ch:g6:producing-our-food}), tot afbraakploegen in de bodem,
tot evoluerende tegenstander op de ziekenhuisafdeling
(\cref{ex:g9:how-species-change:resistance}). Dit hoofdstuk legt
eindelijk volledig rekenschap af over de \omterm{def:g6:producing-our-food:fermentation}{microben}: wat ze zijn, hoe
de schadelijke minderheid aanvalt, en hoe de buitenste
verdedigingen van het lichaam --- en de onze --- de deuren houden.

\section{De wereld van de microben}

\begin{definition}[Microben, gesorteerd]\label{def:g9:microbes-and-infection:sorted}
\emph{Micro-organismen}\index{micro-organisme} --- \omterm{def:g6:producing-our-food:fermentation}{microben} ---
zijn \omterm{def:g6:exploring-our-environment:components}{levende wezens} die te klein zijn om met het blote \omterm{def:g1:the-five-senses:sense}{oog} te zien.
De belangrijkste \omterm{def:g5:biodiversity:species}{soorten}:
\begin{itemize}
  \item \emph{bacteriën}\index{bacteriën}: \omterm{def:g6:cell-unit-of-life:unicellular}{eencelligen} zonder
        echte \omterm{def:g6:cell-unit-of-life:parts}{celkern}, veel kleiner dan jouw cellen, die snel
        delen (de wereld van
        \cref{def:g6:cell-unit-of-life:unicellular}) --- overal,
        in ontelbare aantallen;
  \item \emph{eencellige \omterm{def:g6:decomposers-and-soil:decomposers}{schimmels} en andere}: \omterm{ex:g6:producing-our-food:bread}{gisten}
        (\cref{ex:g6:producing-our-food:bread}), de \omterm{def:g4:plants-without-seeds:spore}{sporen} van
        \omterm{def:g6:decomposers-and-soil:decomposers}{schimmels}, de zwemmers uit de druppel slootwater;
  \item \emph{virussen}\index{virus}: nog kleiner, en van een
        andere aard --- een verpakt snippertje genetische tekst
        (de letters van \cref{def:g9:genes-and-dna:dna}) in een
        jas, \emph{zonder \omterm{def:g5:first-look-at-cells:cell}{cel} en zonder eigen leven}: een virus
        doet niets tot het een levende \omterm{def:g5:first-look-at-cells:cell}{cel} binnengaat en de
        leesmachinerie van die \omterm{def:g5:first-look-at-cells:cell}{cel} aan het kopiëren van de tekst
        van de indringer zet. Op de tekenen van
        \cref{prop:g1:living-or-not:signs} zitten virussen op de
        uiterste rand van het leven.
\end{itemize}
\end{definition}

\begin{proposition}[Meestal onschadelijk, vaak onmisbaar]\label{prop:g9:microbes-and-infection:mostly}
De schadelijke reputatie is een minderheidsrapport. De meeste
\omterm{def:g6:producing-our-food:fermentation}{microben} negeren ons; veel ervan dienen ons: de afbraakploegen
(\cref{def:g6:decomposers-and-soil:decomposers}), het
voedselpersoneel, de scheikundigen van de bodem --- en je eigen
\emph{lichaamseigen flora}\index{lichaamseigen flora}: de biljoenen
onschadelijke \omterm{def:g9:microbes-and-infection:sorted}{bacteriën} die \omterm{def:g1:the-five-senses:sense}{huid} en darm bekleden, de \omterm{def:g4:journey-of-food:digestion}{vertering}
helpen en, alleen al door bezetting, indringers van de grond
verdringen. Slechts een kleine minderheid --- de
\emph{ziekteverwekkers}\index{ziekteverwekker} --- maakt ziek.
\end{proposition}
\admitted

\begin{example}[Eén lichaam, drie verhoudingen]\label{ex:g9:microbes-and-infection:relations}
Op je \omterm{def:g1:the-five-senses:sense}{huid}, op dit moment: bewoners bij de miljard, die het
oppervlak houden (verhouding één: bondgenootschap). In je yoghurt
bij het ontbijt: werknemers, die op bevel verzuren (verhouding
twee: dienstverband,
\cref{ex:g6:producing-our-food:yogurt}). Op de roestige spijker in
het schuurtje: \omterm{def:g4:plants-without-seeds:spore}{sporen} van de tetanusbacterie, wachtend op een diepe
snee (verhouding drie: de minderheid van de \omterm{prop:g9:microbes-and-infection:mostly}{ziekteverwekkers}, en de
reden voor \cref{ch:g9:immune-defenses} en de tetanusprik).
\end{example}

\section{Infectie: de aanval, in fasen}

\begin{definition}[De fasen van een infectie]\label{def:g9:microbes-and-infection:stages}
Een \emph{infectie}\index{infectie} verloopt in fasen, elk met haar
eigen naam:
\begin{enumerate}
  \item \emph{overdracht}: de ziekteverwekker reist ---
        aangeraakte oppervlakken en handen, gehoeste druppeltjes,
        besmet voedsel en water, \omterm{prop:g4:journey-of-food:absorb}{bloed}, en de seksuele weg waar
        \cref{def:g8:contraception:condom} op wees;
  \item \emph{binnenkomst}: door een bres --- een snee in de \omterm{def:g1:the-five-senses:sense}{huid},
        of de open deuren van het lichaam: luchtwegen, darm en de
        wegen hierboven;
  \item \emph{vermeerdering}: warm, nat en gevoed delen de
        aangekomen \omterm{def:g6:producing-our-food:fermentation}{microben} --- \omterm{def:g9:microbes-and-infection:sorted}{bacteriën} volgens de regel van de
        \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} uit
        \cref{prop:g6:decomposers-and-soil:speed}, met uren die
        hun aantal verdubbelen; \omterm{def:g9:microbes-and-infection:sorted}{virussen} door cellen in
        kopieerwinkels te veranderen;
  \item \emph{ziekte}: verschijnselen --- uit de schade zelf, uit
        bacteriële gifstoffen, en deels (volgend hoofdstuk) uit
        het gevecht van de verdediging zelf.
\end{enumerate}
\end{definition}

\begin{example}[Twee indringers, tegenover elkaar]\label{ex:g9:microbes-and-infection:contrast}
De bacterie van de snee tegenover het verkoudheidsvirus van de
winter. De bacterie: gaat de bres binnen, vermeerdert zich
\emph{tussen} de cellen in het warme natte weefsel, en haar
afvalstoffen vergiftigen de buurt --- het kloppen en de pus van een
ontstoken vinger. Het \omterm{def:g9:microbes-and-infection:sorted}{virus}: rijdt op een druppeltje naar de
bekleding van de luchtweg, gaat de bekledingscellen zelf binnen, en
elke gekaapte \omterm{def:g5:first-look-at-cells:cell}{cel} laat een menigte kopieën los voordat ze bezwijkt
--- de rauwe keel en de lopende \omterm{def:g1:the-five-senses:sense}{neus} van zijn spoor. Andere
mechaniek; het antibioticum dat de eerste uithongert kan de tweede
niet raken, want de tweede leent \emph{jouw} machinerie, en
medicijnen ertegen zijn moeilijk te richten.
\end{example}

\section{De deuren houden}

\begin{proposition}[De eerste verdedigingen]\label{prop:g9:microbes-and-infection:doors}
Nog voor enig gevecht houdt het lichaam zijn grenzen:
\begin{itemize}
  \item \emph{de \omterm{def:g1:the-five-senses:sense}{huid}}: droog, gelaagd, zichzelf vernieuwend ---
        de muur (het orgaan van
        \cref{def:g1:the-five-senses:sense}, op wachtdienst);
  \item \emph{bekledingen en hun bezems}: de schoonmakende
        oppervlakken van de luchtwegen
        (\cref{prop:g7:lungs-and-gas-exchange:smoke} liet ze
        overbelast zien), de spoelende scheikunde van tranen en
        \omterm{def:g4:journey-of-food:tract}{speeksel}, het zuurbad van de \omterm{def:g4:journey-of-food:tract}{maag}
        (\cref{pb:g7:digestion-and-nutrients:1} steriliseert
        terwijl het verteert);
  \item \emph{de bewoners}: de bezette grond van de flora, weinig
        ruimte om te landen;
\end{itemize}
en de \omterm{def:g1:healthy-habits:hygiene}{hygiëne} verlengt ze: de zeep van
\cref{met:g1:healthy-habits:hands} (nu met haar volledige
mechanisme --- de overdracht onderbroken), veilig water, gekookt en
koud bewaard voedsel (het weigeren van de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} uit
\cref{rem:g6:producing-our-food:goodbad}), de gesteriliseerde naald,
de barrière van het \omterm{def:g8:contraception:condom}{condoom}. Elke hygiëneregel uit dit boek is een
geblokkeerde fase van overdracht of binnenkomst.
\end{proposition}
\admitted

\begin{example}[De twee artsen die de cijfers verzetten]\label{ex:g9:microbes-and-infection:history}
Twee namen verankeren de geschiedenis. Semmelweis, Wenen jaren
1840: de sterfte aan kraamvrouwenkoorts op zijn kraamafdeling
stortte in toen de artsen hun handen wasten tussen de snijzaal en
de bevalling --- de overdracht onderbroken voordat iemand de
reiziger had gezien. Pasteur liet een generatie later de reizigers
zelf zien: \omterm{def:g6:producing-our-food:fermentation}{microben}, geen ``slechte lucht'', die de wijn zuur
maakten, de zijderupsen doodden en de wonden besmetten --- en hitte
(zijn \emph{pasteurisatie}) of sterilisatie die hen tegenhield. De
kiemtheorie maakte binnen één mensenleven van ziekenhuizen
verdedigingen in plaats van gevaren.
\end{example}

\begin{method}[Een infectierisico keuren]\label{met:g9:microbes-and-infection:audit}
Bij elke situatie --- keuken, wond, ziekenhuisafdeling, menigte:
\begin{enumerate}
  \item noem de aannemelijke ziekteverwekker en zijn \omterm{def:g5:biodiversity:species}{soort}
        (bacterie, \omterm{def:g9:microbes-and-infection:sorted}{virus}, \omterm{def:g6:decomposers-and-soil:decomposers}{schimmel});
  \item volg zijn weg van overdracht naar jou;
  \item vind zijn deur van binnenkomst;
  \item vraag wat hem vermeerdert --- warmte, vocht, voedsel,
        tijd;
  \item blokkeer de goedkoopste fase: wassen, koken, afdekken,
        koelen, steriliseren, vaccineren
        (\cref{ch:g9:immune-defenses} legt de laatste
        uit).
\end{enumerate}
\end{method}

\section{Probleem: Het uitbraakschrift}

\begin{problem}[{Weekendprobleem --- een schooluitbraak,
fase voor fase nagegaan}]\label{pb:g9:microbes-and-infection:1}
Een (verzonnen) uitbraak op school: binnen één week meldt de helft
van klas 3B koorts en hoest; daarna duiken er gevallen in de andere
klassen op. De schoolverpleegkundige houdt een uitbraakschrift bij
en vraagt de biologieklas het te helpen lezen.

\medskip\noindent\textbf{Deel I --- De \omterm{def:g6:how-plants-spread:dispersal}{verspreiding} lezen.}
\begin{enumerate}
  \item Het eerste geval --- de index --- zat in 3B; de volgende
        acht delen haar rij en haar tafeltennispartners. Noem de
        waarschijnlijke weg van overdracht, en de deur van
        binnenkomst.
  \item De gevallen verdubbelen in het begin om de twee dagen. De
        rekenkunde van welke fase laat zich hier zien, en wiens
        machinerie doet het kopiëren als de verwekker een \omterm{def:g9:microbes-and-infection:sorted}{virus}
        is?
  \item De checklist van de verpleegkundige: het opkomen van de
        koorts, hoest, rauwe keel. Welke fase markeren de
        verschijnselen, en uit welke twee bronnen komen ze voort?
  \item Waarom sprong de uitbraak precies bij de twee
        schoolbrede bijeenkomsten van die week naar andere
        klassen over?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De tegenaanval.}
\begin{enumerate}[resume]
  \item De maatregelen van de school: zieke leerlingen naar huis;
        ramen open
        (\cref{exo:g4:breathing:10}); handen wassen ingeoefend;
        gedeeld materiaal afgenomen. Wijs elke maatregel de fase
        toe die ze blokkeert.
  \item Een ouder eist antibiotica voor iedereen. Stel het
        antwoord van de arts op voor een virusuitbraak --- en de
        waarschuwing over selectie
        (\cref{exo:g9:how-species-change:10}) als er toch
        antibiotica rondgestrooid zouden worden.
  \item De kantine meldt tegelijkertijd twee maaggevallen, terug
        te voeren op een niet gekoeld nagerecht. Keur die
        neven-uitbraak met
        \cref{met:g9:microbes-and-infection:audit} --- ander \omterm{def:g5:biodiversity:species}{soort}
        verwekker, andere fasen.
  \item Waarom brengt de verpleegkundige in kaart \emph{wie waar
        zat} in plaats van de gevallen als toevallig te
        behandelen? Wat is een uitbraakkaart, in termen van
        overdracht?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De nabespreking.}
\begin{enumerate}[resume]
  \item Schrijf de levensbeschrijving van de uitbraak in de vier
        fasen, van indexgeval tot uitdoven.
  \item De uitbraak doofde uit voordat ze elke leerling bereikte.
        Bied twee mechanismen aan --- de blokkades van de
        maatregelen, en de hint van het volgende hoofdstuk: wie al
        immuun waren door de verwante stam van vorig jaar.
  \item Voeg de alinea over de geschiedenis toe: van welke twee
        negentiende-eeuwse lessen
        (\cref{ex:g9:microbes-and-infection:history})
        stamt het hele antwoord van de school af?
  \item Sluit het schrift af met de keuringsmethode als
        wapenspreuk: één zin over waar uitbraken het goedkoopst
        te stoppen zijn.
\end{enumerate}
\end{problem}
