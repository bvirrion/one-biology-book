\chapter{Poblaciones y demografía}\label{ch:b1:populations}

Un matraz de levadura empieza con un millón de \omterm{def:b1:cell-unit-of-life:cell}{células} y, dieciséis horas
después, alberga sesenta millones y ha dejado de crecer; una manada de
ciervos en una isla crece una décima parte cada año hasta el invierno en que
se le acaba el alimento; los linces del bosque boreal suben y se desploman
cada diez años, un año por detrás de las liebres que comen. Cada uno es una
\emph{\omterm{def:b1:populations:population}{población}} --- los individuos de una especie en un lugar --- y cada
uno obedece una aritmética que puede escribirse. Este capítulo describe cómo
se cuentan las poblaciones, cómo se tabulan sus nacimientos y sus muertes,
las dos ecuaciones del crecimiento, las fuerzas que mantienen a raya los
efectivos, y qué dice esa misma aritmética sobre nuestra propia especie.

\section{Contar una población}

\begin{definition}[Población, densidad, dispersión]\label{def:b1:populations:population}
\index{población}\index{densidad de población}
Una \emph{población} es el conjunto de individuos de una especie que viven
en un área dada en un momento dado y pueden cruzarse entre sí. Su
\emph{tamaño} $N$ es el número de individuos; su \emph{densidad}, el número
por unidad de superficie o de volumen; su \emph{dispersión}, el patrón de su
distribución en el espacio --- \emph{agregada} donde los recursos o la vida
social los reúnen (manadas, bancos, rodales de una planta), \emph{uniforme}
donde se repelen (aves territoriales, arbustos del desierto que compiten por
el agua), \emph{aleatoria} donde no actúa ninguna de las dos cosas. Las
poblaciones tienen además una \emph{estructura de edades}, una \emph{razón
de sexos} y tasas de natalidad, mortalidad, inmigración y emigración, cuyo
balance es el crecimiento.
\end{definition}

\begin{method}[Estimación de una población]\label{met:b1:populations:counting}
\index{marcaje--recaptura}
\begin{enumerate}
  \item \omterm{def:b1:organism-environment:organism}{Organismos} sésiles o lentos: cuéntense en \emph{cuadrados} de área
        conocida colocados al azar, y multiplíquese la densidad media por la
        superficie total; la dispersión de los valores entre cuadrados da la
        precisión y revela el patrón espacial.
  \item Animales móviles: \emph{marcaje--recaptura}. Captúrense $M$
        individuos, márquense y libérense; más tarde captúrense $C$, de los
        cuales $R$ están marcados. Si los marcados se mezclan libremente y
        nada ha cambiado, la fracción marcada de la segunda captura iguala a
        la fracción marcada de la \omterm{def:b1:populations:population}{población}, $R/C = M/N$, de modo que
        $N \approx MC/R$ (la estimación de Lincoln--Petersen).
  \item Compruébense los supuestos: ni nacimientos, ni muertes, ni
        migraciones entre las capturas, marcas que ni se caen ni cambian el
        comportamiento del animal, e igual capturabilidad. Cada
        incumplimiento sesga $N$ en un sentido predecible.
  \item Indicios indirectos donde contar es imposible: excrementos, huellas,
        cantos, nidos, el \omterm{def:b1:nucleic-acids:chain}{ADN} de un litro de agua de charca.
\end{enumerate}
\end{method}

\begin{example}[Topillos en un prado]\label{ex:b1:populations:voles}
Sesenta topillos capturados, marcados y liberados en una hectárea de prado;
tres noches después se capturan ochenta, de los cuales doce llevan marca:
$N \approx 60\times 80/12 = 400$ topillos, cuatro por cada cien metros
cuadrados. Si los topillos marcados, tras aprender que las trampas contienen
comida, se dejan capturar más fácilmente, $R$ queda inflado y $N$
subestimado; si el marcaje los ahuyentó de las trampas, ocurre lo contrario.
\end{example}

\section{Demografía: nacimientos y muertes por edad}

\begin{definition}[Tabla de vida]\label{def:b1:populations:lifetable}
\index{tabla de vida}\index{supervivencia}
Una \emph{tabla de vida} sigue a lo largo de su existencia a una cohorte de
individuos nacidos a la vez. Para cada edad $x$ registra $l_x$, la
\emph{supervivencia} --- la fracción de la cohorte todavía viva a la edad
$x$ --- y $m_x$, la \emph{fecundidad} --- el número medio de hijas nacidas
de una hembra de edad $x$. De ahí: la mortalidad entre edades,
$q_x = 1 - l_{x+1}/l_x$; la \emph{tasa reproductiva neta}
$R_0 = \sum_x l_x m_x$, el número de hijas que tendrá en promedio una hembra
recién nacida a lo largo de su vida; y el \emph{tiempo de generación}
$T = \sum_x x\,l_x m_x / R_0$, la edad media de las madres de esas hijas.
Una \omterm{def:b1:populations:population}{población} con $R_0 > 1$ crece y con $R_0 < 1$ decae; a lo largo de una
generación queda multiplicada por $R_0$.
\end{definition}

\begin{proposition}[Tres formas de la curva de supervivencia]\label{prop:b1:populations:survivorship}
Representar $\log l_x$ frente a la edad da tres tipos de curva. \emph{Tipo
I}: la mayoría de los individuos llega a viejo y muere en un intervalo
estrecho (grandes mamíferos, seres humanos en los países ricos). \emph{Tipo
II}: una probabilidad de morir constante a todas las edades, una recta en
escala logarítmica (muchas aves, lagartos, roedores). \emph{Tipo III}:
mortalidad enorme entre los jóvenes y vida larga para los pocos
supervivientes (ostras, peces, la mayoría de las plantas y de los insectos).
La forma se sigue del modo de vida del \omterm{def:b1:organism-environment:organism}{organismo}: una especie que invierte
poco en cada uno de muchos descendientes tiene una curva de tipo III; una
que invierte mucho en cada uno de unos pocos, de tipo I.
\end{proposition}

\begin{example}[Una tabla de vida para una manada de ciervos]\label{ex:b1:populations:deer}
De mil cervatillos, seiscientos llegan a su primer cumpleaños, y los
supervivientes pierden luego una décima parte de sus efectivos cada año
hasta que pocos pasan de los nueve; las hembras crían desde los dos años,
unas ocho décimas de hija al año en su plenitud. Sumar $l_x m_x$ da
$R_0 \approx 1.5$: cada hembra deja hija y media, y la manada crece la mitad
en cada generación de unos cuatro años --- un aumento del once por ciento al
año hasta que algo lo detiene.
\end{example}

\section{El crecimiento}

\begin{theorem}[Crecimiento exponencial]\label{thm:b1:populations:exponential}
\index{crecimiento exponencial}\index{tasa intrínseca de crecimiento}
Una \omterm{def:b1:populations:population}{población} cuyas tasas per cápita de natalidad y de mortalidad $b$ y $d$
son constantes crece a un ritmo proporcional a su tamaño,
\[
\frac{\dd N}{\dd t} = rN, \qquad r = b - d, \qquad N(t) = N_0\,e^{rt},
\]
donde $r$ es la \emph{\omterm{thm:b1:populations:exponential}{tasa intrínseca de crecimiento}}. Se duplica cada
$\ln 2/r$; una \omterm{def:b1:populations:population}{población} con $r = \qty{0.11}{yr^{-1}}$ se duplica en
\qty{6.3}{yr}, y una con $r = \qty{0.5}{h^{-1}}$, en \qty{83}{min}. A partir
de una \omterm{def:b1:populations:lifetable}{tabla de vida}, $r \approx \ln R_0 / T$.
\end{theorem}
\begin{proof}
En un intervalo corto $\dd t$ los nacimientos son $bN\,\dd t$ y las muertes
$dN\,\dd t$, de modo que $\dd N = (b - d)N\,\dd t$; la ecuación
$\dd N/\dd t = rN$ tiene la solución $N_0 e^{rt}$ (la derivada de
$e^{rt}$ es $re^{rt}$). Duplicación: $e^{rt} = 2$ da $t = \ln 2/r$. A lo
largo de una generación $T$ la \omterm{def:b1:populations:population}{población} queda multiplicada por
$R_0 = e^{rT}$, de donde $r = \ln R_0/T$.
\end{proof}

\begin{theorem}[Crecimiento logístico]\label{thm:b1:populations:logistic}
\index{crecimiento logístico}\index{capacidad de carga}
Si la tasa de crecimiento per cápita cae linealmente conforme la \omterm{def:b1:populations:population}{población}
se acerca a una \emph{\omterm{thm:b1:populations:logistic}{capacidad de carga}} $K$ --- el tamaño que el ambiente
puede sostener ---
\[
\frac{\dd N}{\dd t} = rN\left(1 - \frac{N}{K}\right), \qquad
N(t) = \frac{K}{1 + \dfrac{K - N_0}{N_0}\,e^{-rt}} .
\]
El crecimiento es casi exponencial mientras $N \ll K$, máximo en $N = K/2$
(donde $\dd N/\dd t = rK/4$), y se anula en $N = K$, valor al que la
\omterm{def:b1:populations:population}{población} se acerca como una meseta: una curva en forma de S.
\end{theorem}
\begin{proof}
Escríbase la ecuación como $\dd N/[N(1 - N/K)] = r\,\dd t$ y divídase el
miembro izquierdo en $\dd N/N + \dd N/(K - N)$; al integrar se obtiene
$\ln[N/(K - N)] = rt + \text{cte}$, y al despejar $N$ con $N(0) =
N_0$ resulta la fórmula. Derivar $rN(1 - N/K)$ respecto de
$N$ e igualar a cero da el máximo en $N = K/2$.
\end{proof}

\begin{proposition}[Dependencia de la densidad]\label{prop:b1:populations:densitydependence}
\index{dependencia de la densidad}
El crecimiento se frena al subir la densidad porque la tasa de natalidad per
cápita cae, o la de mortalidad sube, con el hacinamiento: el alimento por
cabeza disminuye, los territorios se agotan, se acumulan desechos y
enfermedades, los depredadores se concentran. Estos factores
\emph{dependientes de la densidad} regulan la \omterm{def:b1:populations:population}{población} hacia $K$. Los
factores \emph{independientes de la densidad} --- helada, inundación,
incendio, sequía --- matan una fracción sea cual sea el número y no pueden
regular, solo perturbar; las poblaciones de muchos insectos las gobierna
sobre todo el tiempo y nunca se acercan a una \omterm{thm:b1:populations:logistic}{capacidad de carga}.
\end{proposition}
\begin{proof}[Evidencia]
Gause (1934) cultivó \emph{Paramecium} en tubos de medio renovado a diario:
los recuentos siguieron una curva logística hasta una meseta fijada por el
alimento, y duplicar el alimento duplicaba la meseta. Pearl (1927) halló lo
mismo en moscas del vinagre en botellas, y Carlson (1913), en levadura. En
el campo, los gorriones cantores de una isla ponen puestas menores y pierden
más pollos cuando la \omterm{def:b1:populations:population}{población} reproductora es grande, y el número de
territorios pone tope al número de reproductores; retirar aves va seguido de
su reemplazo a partir de un excedente que había quedado excluido.
\end{proof}

\begin{definition}[Estrategias de historia vital]\label{def:b1:populations:rK}
\index{estrategia r}\index{estrategia K}
Las especies difieren en dónde se sitúan entre dos extremos. Una especie
\emph{$r$-seleccionada} --- malas hierbas, pulgones, ratones, la mayoría de
los peces --- está hecha para una $r$ alta: madura pronto, produce muchos
descendientes pequeños, invierte poco en cada uno, vive poco y explota
hábitats nuevos o perturbados antes de que llegue la competencia, con una
\omterm{def:b1:populations:lifetable}{supervivencia} de tipo III. Una especie \emph{$K$-seleccionada} --- robles,
elefantes, albatros --- está hecha para persistir cerca de $K$: madura
tarde, produce pocos descendientes grandes, los cuida, vive mucho y compite
bien en hábitats estables y saturados, con una curva de tipo I. La mayoría
de las especies mezclan ambas cosas; las etiquetas nombran los extremos de
la escala.
\end{definition}

\section{Fluctuación y regulación}

\begin{proposition}[Ciclos de depredador y presa]\label{prop:b1:populations:cycles}
\index{ciclo poblacional}
Algunas poblaciones no se estabilizan, sino que oscilan. La liebre americana
del bosque septentrional y el lince que la come suben y bajan juntos con un
periodo de unos diez años, y el lince va uno o dos años por detrás de la
liebre: muchas liebres alimentan a muchos linces, muchos linces se comen las
liebres hasta reducirlas, pocas liebres matan de hambre a los linces, pocos
linces dejan recuperarse a las liebres. El ciclo lo impulsan en parte el
depredador y en parte el propio alimento de las liebres, cuya recuperación
tras el ramoneo excesivo tarda años; los modelos de depredador y presa
acoplados (\cref{ch:b1:species-interactions}) reproducen tales oscilaciones
a partir de dos ecuaciones.
\end{proposition}
\begin{proof}[Evidencia]
Los registros de pieles de una compañía comercial, llevados durante noventa
años desde la década de 1840, recogen el número de pieles de lince y de
liebre entregadas cada año en todo un subcontinente: ambas series muestran
picos cada nueve o diez años, y el pico del lince va por detrás del de la
liebre, a lo largo de nueve ciclos completos. Los estudios de campo
posteriores, con liebres protegidas de los depredadores por vallas o
alimentadas de más, muestran que suprimir cualquiera de los dos factores
amortigua el ciclo y suprimir ambos lo elimina.
\end{proof}

\begin{example}[Nuestra propia especie]\label{ex:b1:populations:humans}
La \omterm{def:b1:populations:population}{población} humana tardó toda la historia en llegar a mil millones, hacia
1800, y dos siglos en llegar a ocho mil millones. Su tasa de crecimiento
alcanzó su máximo, cerca del \qty{2}{\%} anual, en la década de 1960
(duplicación en 35 años) y ha caído desde entonces por debajo del
\qty{1}{\%}, no por las muertes sino por los nacimientos: allí donde la
mortalidad infantil baja y las mujeres se educan, las familias se reducen en
una generación, y la pirámide de edades pasa de triángulo a columna. La
aritmética de $R_0$ se nos aplica; lo que cambió es $m_x$.
\end{example}

\section{Problema: levadura en un matraz y ciervos en una isla}

\begin{problem}[{Problema de fin de semana --- un cultivo contado, una
manada tabulada, un lago muestreado y una cosecha fijada, hasta llegar a la
capacidad de carga y la tasa intrínseca de crecimiento de la levadura}]
\label{pb:b1:populations:1}
Un cultivo de levadura se cuenta cada dos horas (\omterm{def:b1:cell-unit-of-life:cell}{células} por mililitro,
$\times 10^5$): $10, 29, 71, 175, 351, 513, 595, 641, 656$ en $t = 0,
2, \ldots, 16$ h. Una \omterm{def:b1:populations:population}{población} de ciervos de una isla tiene la \omterm{def:b1:populations:lifetable}{tabla de
vida}: $l_x = 1.00, 0.60, 0.52, 0.46, 0.40, 0.34, 0.26, 0.16, 0.06, 0$ y
$m_x = 0, 0, 0.6, 0.8, 0.8, 0.8, 0.7, 0.5, 0.3, 0$ para $x = 0$ a $9$
años. En un lago se marcan y liberan 120 percas; una semana después se
capturan 150, de las cuales 18 están marcadas.

\medskip
\noindent\textbf{Parte I --- La levadura.}
\begin{enumerate}
  \item Estímese $r$ con los dos primeros recuentos, suponiendo \omterm{thm:b1:populations:exponential}{crecimiento
        exponencial} durante las dos primeras horas.
  \item Calcúlese el tiempo de duplicación.
  \item Estímese $K$ con los últimos recuentos.
  \item Calcúlese la predicción logística en $t = 4$, $8$ y \qty{12}{h} y
        compárese con los recuentos.
  \item ¿A qué \omterm{def:b1:populations:population}{población} es máximo el crecimiento, y cuándo se alcanza?
        Calcúlese la tasa máxima de crecimiento en \omterm{def:b1:cell-unit-of-life:cell}{células} por mililitro y
        hora.
  \item ¿Qué predeciría el modelo exponencial a las \qty{12}{h}? ¿En qué
        factor se equivoca?
  \item Nómbrense dos cosas que podrían fijar $K$ en un matraz sellado de
        levadura.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Los ciervos.}
\begin{enumerate}[resume]
  \item Calcúlese la mortalidad $q_x$ del primer año y la del quinto. ¿De
        qué tipo de \omterm{def:b1:populations:lifetable}{supervivencia} se trata?
  \item Calcúlese $l_x m_x$ para cada edad y súmense: $R_0$.
  \item Calcúlese $T$.
  \item Calcúlense $r$ y el tiempo de duplicación de la manada.
  \item Hoy la isla alberga 200 ciervos. Predígase el número dentro de 10
        años con \omterm{thm:b1:populations:exponential}{crecimiento exponencial}.
  \item La isla puede alimentar a 600 ciervos. Predígase el número dentro de
        10 años con \omterm{thm:b1:populations:logistic}{crecimiento logístico}, y el año en que la manada pasa de
        500.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Las percas y la cosecha.}
\begin{enumerate}[resume]
  \item Estímese la \omterm{def:b1:populations:population}{población} de percas del lago.
  \item Si 20 de las percas marcadas perdieron sus marcas antes de la
        segunda captura, recalcúlese con el número verdadero de peces
        marcados, y dígase en qué sentido va el error de la primera
        estimación.
  \item La \omterm{def:b1:populations:population}{población} de percas tiene $K = 1000$ y $r = \qty{0.4}{yr^{-1}}$.
        Calcúlense el rendimiento máximo sostenible y la \omterm{def:b1:populations:population}{población} a la que
        se obtiene.
  \item Calcúlese el tiempo de duplicación de la \omterm{def:b1:populations:population}{población} de percas cuando
        está muy por debajo de $K$.
  \item El club decide en cambio llevarse un \qty{20}{\%} fijo de la
        existencia cada año. Calcúlense la \omterm{def:b1:populations:population}{población} de equilibrio y la
        captura anual en ese equilibrio.
  \item Un club de pescadores se lleva 120 percas al año. Muéstrese si la
        \omterm{def:b1:populations:population}{población} puede sostenerlo, calculando $\dd N/\dd t$ en $N =
        500$ y en $N = 300$.
  \item ¿Por qué es arriesgado en la práctica cosechar en $K/2$, teniendo en
        cuenta el tiempo y el error de recuento de la pregunta 14?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La regulación.}
\begin{enumerate}[resume]
  \item El $m_x$ de los ciervos cae a la mitad cuando la manada supera los
        400. Muéstrese que $R_0$ baja entonces de 1 --- ¿qué le hace esto a
        la manada, y qué clase de factor está actuando?
  \item Un invierno duro mata al \qty{40}{\%} de los ciervos sea cual sea su
        número. ¿Es este factor regulador? ¿Qué le ocurre a la manada en los
        años siguientes?
  \item Llegan lobos y se llevan 30 ciervos fijos al año. Calcúlense los
        tamaños de manada de equilibrio para el modelo logístico con
        $K = 600$ y la $r$ de la pregunta 11 (resuélvase $rN(1 - N/K) = 30$),
        y dígase qué equilibrio es estable.
  \item Explíquese en dos frases por qué el ciclo del lince va por detrás
        del de la liebre.
  \item Enúnciese el resultado: la \omterm{thm:b1:populations:logistic}{capacidad de carga} y la \omterm{thm:b1:populations:exponential}{tasa intrínseca
        de crecimiento} de la levadura, y $R_0$, $T$ y $r$ de los ciervos.
\end{enumerate}
\end{problem}
