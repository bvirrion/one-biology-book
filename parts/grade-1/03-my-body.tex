\chapter{My Body}\label{ch:g1:my-body}

Your \omterm{def:g1:my-body:body}{body} is a wonderful living machine. It has parts that work
together so you can walk, eat, breathe and think. Let us name the
main parts and say what they do.

\section{Outside the body}

\begin{definition}[Body]\label{def:g1:my-body:body}
The \emph{body}\index{body} is all of you: \omterm{def:g1:my-body:parts}{head}, \omterm{def:g1:my-body:parts}{trunk}, \omterm{def:g1:my-body:parts}{arms} and
\omterm{def:g1:my-body:parts}{legs}. The outside is covered by \emph{skin}\index{skin}, which
protects you.
\end{definition}

\omimg{my-body.jpg}{0.52\linewidth}{A child figure with main outer
parts labelled --- \omterm{def:g1:my-body:parts}{head}, \omterm{def:g1:my-body:parts}{arms}, \omterm{def:g1:my-body:parts}{hands}, \omterm{def:g1:my-body:parts}{legs} and \omterm{def:g1:my-body:parts}{feet} work as a team.}

\begin{definition}[Main outer parts]\label{def:g1:my-body:parts}
\begin{itemize}
  \item the \emph{head}\index{head} holds the brain, eyes, ears,
        nose and mouth;
  \item the \emph{trunk}\index{trunk (body)} is the middle: chest
        and belly;
  \item the \emph{arms}\index{arm} and \emph{hands}\index{hand}
        help you hold and carry;
  \item the \emph{legs}\index{leg} and \emph{feet}\index{foot} help
        you stand and walk.
\end{itemize}
\end{definition}

\begin{example}[Counting]\label{ex:g1:my-body:count}
You have one \omterm{def:g1:my-body:parts}{head}, one \omterm{def:g1:my-body:parts}{trunk}, two \omterm{def:g1:my-body:parts}{arms}, two \omterm{def:g1:my-body:parts}{hands}, two \omterm{def:g1:my-body:parts}{legs} and two
\omterm{def:g1:my-body:parts}{feet}. You have ten fingers and ten toes (usually).
\end{example}

\section{What the parts do}

\begin{method}[Match part to job]\label{met:g1:my-body:jobs}
\begin{enumerate}
  \item Eyes help you see; ears help you hear.
  \item The mouth helps you eat and speak.
  \item \omterm{def:g1:my-body:parts}{Hands} grasp; \omterm{def:g1:my-body:parts}{legs} walk and run.
  \item The chest rises when you breathe air in.
\end{enumerate}
Every part has a job. Together they keep you active and safe.
\end{method}

\begin{example}[A simple day]\label{ex:g1:my-body:day}
You use your \omterm{def:g1:my-body:parts}{legs} to walk to school. You use your \omterm{def:g1:my-body:parts}{hands} to write.
You use your mouth to talk with friends. You use your eyes to read.
Your whole \omterm{def:g1:my-body:body}{body} works all day long.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g1:my-body:1}
Name five outer parts of your \omterm{def:g1:my-body:body}{body}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:2}
How many \omterm{def:g1:my-body:parts}{arms} do you have? How many \omterm{def:g1:my-body:parts}{legs}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:3}
Which part do you use to hold a pencil? To kick a ball?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:4}
True or false? ``\omterm{def:g1:my-body:body}{Skin} covers the outside of the \omterm{def:g1:my-body:body}{body}.''
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:5}
