\chapter{Le cerveau aux commandes}\label{ch:g5:brain-in-command}

Un ballon file vers toi~; tu l'attrapes. Une demi-seconde, sans y
penser --- et pourtant, en cette demi-seconde, tes \omterm{def:g1:the-five-senses:sense}{yeux} ont signalé
un objet en vol, quelque chose a calculé sa trajectoire, et des
dizaines de \omterm{def:g3:bones-and-muscles:muscle}{muscles} ont reçu l'ordre de faire exactement la bonne
prise. Ce quelque chose est le \omterm{def:g5:brain-in-command:system}{cerveau}, et ce chapitre suit ses
ordres le long du câblage du corps.

\section{Le réseau de commande}

\begin{definition}[Le système nerveux]\label{def:g5:brain-in-command:system}
Le \emph{système nerveux}\index{système nerveux} est le réseau de
commande du corps~:
\begin{itemize}
  \item le \emph{cerveau}\index{cerveau}, à l'abri du \omterm{def:g3:bones-and-muscles:skeleton}{crâne}
        (\cref{def:g3:bones-and-muscles:skeleton}), est le centre~:
        il reçoit tous les rapports et donne tous les ordres~;
  \item la \emph{moelle épinière}\index{moelle épinière}, un gros
        câble de tissu nerveux, descend du cerveau à l'intérieur de
        la colonne protectrice~;
  \item les \emph{nerfs}\index{nerf}, fils blancs qui se ramifient
        du cerveau et de la moelle vers chaque organe, portent les
        messages --- à plus de \qty{100}{m/s} pour les plus rapides.
\end{itemize}
\end{definition}

\begin{proposition}[Signaler, décider, ordonner]\label{prop:g5:brain-in-command:loop}
Toute action suit la même boucle
(\cref{prop:g1:the-five-senses:brain} en était la première
esquisse)~:
\begin{enumerate}
  \item les \emph{\omterm{def:g1:the-five-senses:sense}{organes des sens}} signalent~: des \omterm{def:g5:brain-in-command:system}{nerfs} portent
        leurs messages \emph{vers} le \omterm{def:g5:brain-in-command:system}{cerveau}~;
  \item le \emph{\omterm{def:g5:brain-in-command:system}{cerveau}} rassemble les rapports, décide et compose
        les ordres~;
  \item des \omterm{def:g5:brain-in-command:system}{nerfs} portent les ordres \emph{depuis} le \omterm{def:g5:brain-in-command:system}{cerveau}
        jusqu'aux \emph{\omterm{def:g3:bones-and-muscles:muscle}{muscles}}
        (\cref{def:g3:bones-and-muscles:muscle}), qui se
        contractent sur commande.
\end{enumerate}
Les \omterm{def:g1:the-five-senses:sense}{organes des sens} signalent, les \omterm{def:g3:bones-and-muscles:muscle}{muscles} obéissent --- ni les uns
ni les autres ne décident jamais.
\end{proposition}
\admitted

\begin{example}[La prise, rejouée au ralenti]\label{ex:g5:brain-in-command:catch}
Rejoue la demi-seconde~: les \omterm{def:g1:the-five-senses:sense}{yeux} suivent le ballon et le signalent,
sans arrêt~; le \omterm{def:g5:brain-in-command:system}{cerveau}, en comparant rapport après rapport, calcule
où le ballon sera~; les ordres partent en flot vers les \omterm{def:g1:my-body:parts}{jambes} (un
pas à gauche), le tronc (se pencher), le \omterm{def:g1:my-body:parts}{bras} et les doigts (ouvrir
--- maintenant fermer). Aucun organe n'a «~attrapé le ballon~» à lui
seul~: les \omterm{def:g1:the-five-senses:sense}{yeux}, le \omterm{def:g5:brain-in-command:system}{cerveau} et les \omterm{def:g3:bones-and-muscles:muscle}{muscles} ont chacun joué leur
ligne de \cref{prop:g5:brain-in-command:loop}.
\end{example}

\begin{remark}[Bien plus que le mouvement]\label{rem:g5:brain-in-command:more}
Le même \omterm{def:g5:brain-in-command:system}{cerveau} qui attrape des ballons garde aussi tes souvenirs,
maintient ton attention sur cette page, rêve la nuit --- et fait
tourner des services auxquels tu ne penses jamais~: il entretient la
\omterm{def:g4:breathing:movements}{respiration} jour et nuit
(\cref{def:g4:breathing:movements}) et règle le rythme du \omterm{def:g4:heart-and-blood:heart}{cœur} sur
les besoins du corps (\cref{ex:g4:heart-and-blood:effort}). Certains
ordres, c'est à toi de les donner~; d'autres, le \omterm{def:g5:brain-in-command:system}{cerveau} les donne à
ta place.
\end{remark}

\section{Mesurer la boucle~: le temps de réaction}

\begin{definition}[Temps de réaction]\label{def:g5:brain-in-command:reaction}
La boucle est rapide, mais pas instantanée~: le rapport, la décision
et l'ordre prennent chacun leur part. Le délai entre un signal et la
réponse du corps est le \emph{temps de réaction}\index{temps de
réaction} --- pour une tâche simple (attraper, appuyer, saisir),
environ un cinquième de seconde chez une personne entraînée, un peu
plus chez les enfants.
\end{definition}

\begin{method}[Le test de la règle]\label{met:g5:brain-in-command:ruler}
Le \omterm{def:g5:brain-in-command:reaction}{temps de réaction} sans autre instrument qu'une règle~:
\begin{enumerate}
  \item un camarade tient une règle suspendue à la verticale, le
        zéro en bas, entre ton pouce et ton index ouverts~;
  \item sans prévenir, il lâche~; tu attrapes la règle le plus vite
        possible~;
  \item lis où tu l'as attrapée~: moins de centimètres ont défilé,
        plus ta boucle est rapide~;
  \item recommence plusieurs fois et garde ta valeur habituelle ---
        les essais isolés sautent dans tous les sens.
\end{enumerate}
La règle qui tombe change du temps en centimètres~: environ
$\qty{15}{cm}$ correspondent à un sixième de seconde.
\end{method}

\begin{example}[Pourquoi le passager sursaute le premier]\label{ex:g5:brain-in-command:driver}
Un cinquième de seconde paraît nul --- jusqu'à ce que la vitesse le
multiplie. Une voiture lancée sur l'autoroute parcourt environ
\qty{7}{m} pendant un \omterm{def:g5:brain-in-command:reaction}{temps de réaction} d'un cinquième de seconde~:
le conducteur n'a pas encore touché le frein, et la voiture a
traversé une salle de classe. Le \omterm{def:g5:brain-in-command:reaction}{temps de réaction} est la raison
d'être des distances entre voitures, et fait de l'attention du
conducteur un organe de sécurité.
\end{example}

\begin{remark}[Entraîner la boucle]\label{rem:g5:brain-in-command:training}
L'entraînement raccourcit et affine la boucle --- non pas la vitesse
des \omterm{def:g5:brain-in-command:system}{nerfs}, mais l'aiguillage du \omterm{def:g5:brain-in-command:system}{cerveau}. La jongleuse débutante fait
tout tomber~; un mois plus tard, le même \omterm{def:g5:brain-in-command:system}{cerveau} conduit les mêmes
mains à travers trois balles sans effort apparent. Apprendre un
savoir-faire, c'est le \omterm{def:g5:brain-in-command:system}{cerveau} qui trace de meilleurs chemins ---
une chose de plus qui grandit avec l'exercice, comme les \omterm{def:g3:bones-and-muscles:muscle}{muscles} de
\cref{met:g3:bones-and-muscles:care}.
\end{remark}

\section{Protéger le chef}

\begin{proposition}[Sommeil, nourriture, sécurité]\label{prop:g5:brain-in-command:protect}
Le \omterm{def:g5:brain-in-command:system}{cerveau} est l'organe le mieux protégé du corps --- casque du
\omterm{def:g3:bones-and-muscles:skeleton}{crâne}, \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale} --- et le plus exigeant~: bien qu'il ne
pèse qu'une petite part du corps, il consomme une grande part de son
\omterm{prop:g3:balanced-diet:jobs}{énergie} et de son dioxygène
(\cref{prop:g4:heart-and-blood:carries}), et il fait son classement
et ses réparations pendant le \emph{\omterm{prop:g1:healthy-habits:needs}{sommeil}}
(\cref{prop:g1:healthy-habits:needs}). Un \omterm{def:g5:brain-in-command:system}{cerveau} fatigué commande
mal~: réactions plus lentes, attention dispersée, décisions
grincheuses.
\end{proposition}
\admitted

\begin{example}[Le casque protège la boucle]\label{ex:g5:brain-in-command:helmet}
Les \omterm{def:g3:bones-and-muscles:skeleton}{os} se ressoudent (\cref{rem:g3:bones-and-muscles:alive})~; le
tissu du \omterm{def:g5:brain-in-command:system}{cerveau}, une fois gravement abîmé, le plus souvent non.
C'est le calcul qui justifie le casque de vélo~: le \omterm{def:g3:bones-and-muscles:skeleton}{crâne} plus de la
mousse là où la protection du corps pourrait ne pas suffire. Une
assurance bon marché pour l'organe qui dirige tout le reste.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g5:brain-in-command:1}
Nomme les trois parties du \omterm{def:g5:brain-in-command:system}{système nerveux} et le rôle de chacune.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:brain-in-command:2}
Énonce la boucle de toute action en trois étapes. Quelles parties
signalent, laquelle décide, lesquelles obéissent~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:brain-in-command:3}
Où le \omterm{def:g5:brain-in-command:system}{cerveau} et la \omterm{def:g5:brain-in-command:system}{moelle épinière} sont-ils protégés, chacun, et
par quels \omterm{def:g3:bones-and-muscles:skeleton}{os}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:brain-in-command:4}
Qu'est-ce que le \omterm{def:g5:brain-in-command:reaction}{temps de réaction}~? Donne sa valeur approximative
pour une tâche simple.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:brain-in-command:5}
Rejoue au ralenti, comme dans
\cref{ex:g5:brain-in-command:catch}~: tu marches sur quelque chose
de pointu et ton pied se retire d'un coup. Qui a signalé, qui a
décidé, qui a obéi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:brain-in-command:6}
Donne deux services que le \omterm{def:g5:brain-in-command:system}{cerveau} assure sans que tu y penses,
d'après \cref{rem:g5:brain-in-command:more}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:brain-in-command:7}
Pourquoi un \omterm{def:g5:brain-in-command:system}{cerveau} fatigué commande-t-il mal~? Nomme deux de ses
besoins d'après \cref{prop:g5:brain-in-command:protect}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:brain-in-command:8}
À toi d'essayer~: réalise le test de la règle de
\cref{met:g5:brain-in-command:ruler}, cinq essais. Donne ta distance
de prise habituelle --- et dis si l'entraînement l'a améliorée.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:brain-in-command:9}
Pourquoi une prise instantanée n'existe-t-elle pas, si habile que
soit celui qui attrape~? Réponds avec les trois étapes de la boucle.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:brain-in-command:10}
À l'aide de \cref{ex:g5:brain-in-command:driver}~: pourquoi le code
de la route exige-t-il des \emph{distances} entre voitures plutôt
que simplement de bons freins~? Que change l'attention dans la
boucle~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:brain-in-command:11}
La jongleuse de \cref{rem:g5:brain-in-command:training} s'est
améliorée sans que ses \omterm{def:g5:brain-in-command:system}{nerfs} conduisent plus vite. Quelle partie de
la boucle s'est améliorée à la place~? Compare avec l'entraînement
d'un \omterm{def:g3:bones-and-muscles:muscle}{muscle}~: qu'est-ce qui grandit dans chaque cas~?
\end{exercise}
