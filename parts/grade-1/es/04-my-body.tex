\chapter{Mi cuerpo}\label{ch:g1:my-body}

Ponte delante de un espejo y saluda con la mano: un cuerpo vivo
entero te devuelve el saludo. Lleva creciendo desde antes de que
nacieras, se dobla en unos sitios y en otros no, y es tuyo para toda
la vida. Toca mirarlo como miramos al gato y a la margarita ---
parte por parte.

\section{Las partes de mi cuerpo}

\begin{definition}[Partes del cuerpo humano]\label{def:g1:my-body:parts}
El cuerpo humano tiene tres partes principales:
\begin{enumerate}
  \item la \emph{cabeza}\index{cabeza}, que lleva la cara, los ojos,
        las orejas, la nariz y la boca;
  \item el \emph{tronco}\index{tronco (cuerpo)}, la parte de en medio
        --- pecho, barriga y espalda;
  \item las \emph{extremidades}\index{extremidad}: dos
        \emph{brazos}\index{brazo} que terminan en manos y dos
        \emph{piernas}\index{pierna} que terminan en pies.
\end{enumerate}
La cabeza se apoya en el \emph{cuello}\index{cuello}, que la une al
tronco.
\end{definition}

\begin{example}[¿El mismo plano que el gato?]\label{ex:g1:my-body:cat}
Compárate con el gato: cabeza, cuerpo, cuatro \omterm{def:g1:my-body:parts}{extremidades} --- el
plano se parece. Pero el gato anda a cuatro patas; tú te sostienes
sobre dos \omterm{def:g1:my-body:parts}{piernas} y dejas libres los dos \omterm{def:g1:my-body:parts}{brazos} para llevar cosas,
dibujar y saludar. Y donde el gato tiene cola, tú no tienes ninguna.
\end{example}

\section{Doblarse: las articulaciones}

\begin{definition}[Articulación]\label{def:g1:my-body:joint}
Una \emph{articulación}\index{articulación} es un sitio por donde el
cuerpo se puede doblar o girar. El \emph{codo}\index{codo} dobla el
\omterm{def:g1:my-body:parts}{brazo}, la \emph{rodilla}\index{rodilla} dobla la \omterm{def:g1:my-body:parts}{pierna}, el
\emph{hombro}\index{hombro} gira el \omterm{def:g1:my-body:parts}{brazo} entero, la
\emph{muñeca}\index{muñeca} y el \emph{tobillo}\index{tobillo} doblan
la mano y el pie. Entre dos articulaciones, el cuerpo se queda recto.
\end{definition}

\begin{example}[Pruébalo: andar sin doblar las rodillas]\label{ex:g1:my-body:stiff}
Prueba a andar sin doblar las \omterm{def:g1:my-body:joint}{rodillas}. Funciona --- justito --- con
pasitos tiesos de pato. Ahora prueba a coger un lápiz del suelo sin
doblar \omterm{def:g1:my-body:joint}{rodillas}, caderas ni espalda. Las articulaciones son las que
hacen que los movimientos del cuerpo sean suaves, rápidos y fáciles.
\end{example}

\begin{remark}[Derecha e izquierda]\label{rem:g1:my-body:sides}
El cuerpo tiene dos lados, derecho e izquierdo, casi como un espejo:
un \omterm{def:g1:my-body:parts}{brazo} y una \omterm{def:g1:my-body:parts}{pierna} a cada lado, un ojo y una oreja a cada lado de
la cabeza. Distinguir tu mano derecha de la izquierda también es saber
del cuerpo --- le dice a la gente hacia dónde quieres decir.
\end{remark}

\section{Mi cuerpo crece}

\begin{proposition}[Hacerse mayor]\label{prop:g1:my-body:growth}
El cuerpo de un niño \emph{crece}\index{crecimiento}: se hace más alto
y más pesado, año tras año, hasta el tamaño de una persona mayor.
Crecer es lento y constante --- demasiado lento para notarlo, fácil de
medir.
\end{proposition}
\admitted

\begin{example}[Marcas en el marco de la puerta]\label{ex:g1:my-body:doorframe}
Ponte contra el marco de la puerta; una persona mayor marca tu altura
con un lápiz. Hazlo otra vez unos meses después: ¡la marca nueva está
más arriba! Nunca te notaste crecer, pero las dos marcas, una encima
de la otra, lo enseñan sin discusión.
\end{example}

\begin{method}[Medir mi altura]\label{met:g1:my-body:measure}
\begin{enumerate}
  \item quítate los zapatos;
  \item ponte recto, con la espalda y los talones contra la pared y
        mirando al frente;
  \item alguien te apoya un libro plano en la cabeza, en horizontal y
        tocando la pared;
  \item marca el sitio de debajo del libro y escribe la fecha al lado.
\end{enumerate}
Medidas siempre de la misma manera, las marcas son de fiar --- y se
pueden comparar.
\end{method}

\begin{remark}[Cada cuerpo es distinto]\label{rem:g1:my-body:different}
En una clase, unos niños son más altos y otros más bajos; unos crecen
antes y otros después. Todo eso es normal. Los cuerpos son como las
caras: construidos con el mismo plano y cada uno distinto. Las marcas
de tu puerta son tuyas --- no son para echar una carrera con el vecino.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g1:my-body:1}
Nombra las tres partes principales del cuerpo humano. ¿Qué une la
cabeza al tronco?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:2}
¿En qué termina cada \omterm{def:g1:my-body:parts}{extremidad}: qué hay al final de un \omterm{def:g1:my-body:parts}{brazo} y al
final de una \omterm{def:g1:my-body:parts}{pierna}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:3}
¿Qué \omterm{def:g1:my-body:joint}{articulación} dobla: (a) el \omterm{def:g1:my-body:parts}{brazo} por en medio? (b) la \omterm{def:g1:my-body:parts}{pierna} por
en medio? (c) la mano al final del \omterm{def:g1:my-body:parts}{brazo}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:4}
Tócate la oreja derecha con la mano izquierda. ¿Qué articulaciones
acaba de usar tu \omterm{def:g1:my-body:parts}{brazo}? Nombra al menos dos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:5}
En el \cref{ex:g1:my-body:cat}, encuentra un parecido de tu cuerpo con
el del gato y dos diferencias.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:6}
¿Por qué la persona que ayuda en el \cref{met:g1:my-body:measure}
apoya un libro plano en tu cabeza en vez de mirar sin más?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:7}
Dos marcas en el marco de la puerta: la del año pasado y la de hoy, un
poco más arriba. ¿Qué enseñan las marcas? ¿Lo notaste mientras pasaba?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:8}
Pruébalo: anda sin doblar las \omterm{def:g1:my-body:joint}{rodillas} como en el
\cref{ex:g1:my-body:stiff} y después prueba a aplaudir sin doblar los
\omterm{def:g1:my-body:joint}{codos}. ¿Qué movimientos de todos los días resultan necesitar
articulaciones?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:9}
Mía es la más bajita de su clase y le preocupa. ¿Qué le responde la
\cref{rem:g1:my-body:different}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:10}
Un muñeco hecho de un solo palo de madera no puede sentarse en una
silla. ¿Dónde cortarías el palo y añadirías puntos de giro para que se
pueda sentar? Nombra las articulaciones del cuerpo que copian tus
cortes.
\end{exercise}
