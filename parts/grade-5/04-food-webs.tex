\chapter{Food Webs}\label{ch:g5:food-webs}

A single \omterm{def:g5:food-webs:chain-web}{food chain} is a thin story. Real \omterm{def:g4:ecosystems:levels}{ecosystems} are networks:
species eat more than one \omterm{def:g2:food-energy:food}{food} and are eaten by more than one hunter.
\omterm{def:g5:food-webs:chain-web}{Food webs} show that tangle --- and help predict what happens when one
strand breaks.

\section{Chains and webs}

\begin{definition}[Food chain and food web]\label{def:g5:food-webs:chain-web}
A \emph{food chain}\index{food chain} is one path of who-eats-whom. A
\emph{food web}\index{food web} is many linked chains in the same
\omterm{def:g4:ecosystems:levels}{community}. Arrows point toward the eater (direction of \omterm{def:g2:food-energy:energy}{energy} flow).
\end{definition}

\omimg{food-chain.jpg}{0.52\linewidth}{A \omterm{def:g5:food-webs:chain-web}{food chain} is a single pathway.}



\begin{example}[Pond links]\label{ex:g5:food-webs:pond}
Algae $\to$ tadpole $\to$ \omterm{def:g5:vertebrates:groups}{fish} $\to$ heron is one chain. The \omterm{def:g5:vertebrates:groups}{fish} may
also eat \omterm{def:g3:insects:insect}{insects}; the heron may also eat frogs --- already a mini-web.
\end{example}

\section{Roles in the web}

\begin{definition}[Trophic roles]\label{def:g5:food-webs:roles}
\emph{Producers}\index{producer} make \omterm{def:g2:food-energy:food}{food} (usually by \omterm{def:g5:photosynthesis:def}{photosynthesis}).
\emph{Herbivores}\index{herbivore} eat producers. \emph{Carnivores}
\index{carnivore} eat \omterm{def:g1:plants-animals:animal}{animals}. \emph{Omnivores}\index{omnivore} eat both
\omterm{def:g1:plants-animals:plant}{plants} and \omterm{def:g1:plants-animals:animal}{animals}. \emph{Decomposers}\index{decomposer} break down
dead matter.
\end{definition}

\begin{definition}[Predator and prey]\label{def:g5:food-webs:pred}
A \emph{predator}\index{predator} hunts and eats other \omterm{def:g1:plants-animals:animal}{animals}. Those
\omterm{def:g1:plants-animals:animal}{animals} are its \emph{prey}\index{prey}. Predators are \omterm{def:g4:ecosystems:roles}{consumers}; prey
may be \omterm{def:g5:food-webs:roles}{herbivores} or smaller \omterm{def:g5:food-webs:roles}{carnivores}.
\end{definition}

\section{Energy flow}

\begin{proposition}[Energy thins out]\label{prop:g5:food-webs:energy}
\omterm{def:g2:food-energy:energy}{Energy} enters mainly as sunlight captured by \omterm{def:g5:food-webs:roles}{producers}. At each feeding
step, much \omterm{def:g2:food-energy:energy}{energy} is used or lost as heat, so fewer individuals can be
supported at higher levels. That is why webs rest on a broad base of
\omterm{def:g1:plants-animals:plant}{plants}.
\end{proposition}
\admitted

\section{When webs change}

\begin{method}[Predicting knock-on effects]\label{met:g5:food-webs:knockon}
\begin{enumerate}
  \item Find the species that is removed or added.
  \item List what it eats and what eats it.
  \item Predict who might increase (less predation or more \omterm{def:g2:food-energy:food}{food}) and
        who might decrease (less \omterm{def:g2:food-energy:food}{food}).
  \item Remember \omterm{def:g5:food-webs:roles}{decomposers} still recycle dead material.
\end{enumerate}
\end{method}

\begin{example}[Fewer rabbits]\label{ex:g5:food-webs:rabbits}
If disease cuts the rabbit \omterm{def:g4:ecosystems:levels}{population}, foxes may go hungry or switch
\omterm{def:g5:food-webs:pred}{prey}; grass may \omterm{def:g2:what-alive:signs}{grow} denser without as much grazing.
\end{example}
