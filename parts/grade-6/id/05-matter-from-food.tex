\chapter{Materi dari Makanan}\label{ch:g6:matter-from-food}

Seekor anak sapi berbobot \qty{40}{kg} menjadi, dalam dua tahun,
seekor sapi berbobot \qty{600}{kg}. Setengah ton sapi baru --- daging,
\omterm{def:g3:bones-and-muscles:skeleton}{tulang}, kulit mentah, tanduk --- muncul dari suatu tempat. Sapinya memakan
rumput; materi barunya \emph{bukanlah rumput}; namun tanpa rumputnya
tidak satu pun darinya akan ada. Apa yang diperbuat \omterm{def:g1:animals-around-us:animal}{hewan} dengan
makanannya adalah urusan bab ini --- sisi \omterm{def:g5:food-webs:roles}{konsumen} dari pembukuan
\cref{ch:g6:plants-are-producers}.

\section{Hewan membangun materinya dari makanan}

\begin{proposition}[Konsumen mengubah materi organik]\label{prop:g6:matter-from-food:transform}
Seekor \omterm{def:g1:animals-around-us:animal}{hewan} membangun \omterm{def:g6:plants-are-producers:matter}{materi organiknya} sendiri --- \omterm{def:g3:bones-and-muscles:muscle}{otot}, \omterm{def:g3:bones-and-muscles:skeleton}{tulang},
lemak, \omterm{def:g1:animals-around-us:covering}{bulu burung}, \omterm{def:g1:animals-around-us:covering}{cangkang} --- dari \emph{\omterm{def:g6:plants-are-producers:matter}{materi organik} yang
dimakannya}. \omterm{def:g4:journey-of-food:digestion}{Pencernaan} (\cref{def:g4:journey-of-food:digestion})
membongkar makanannya menjadi \omterm{def:g4:journey-of-food:digestion}{zat gizi}; \omterm{prop:g4:journey-of-food:absorb}{darahnya} mengantarkannya
(\cref{prop:g4:heart-and-blood:carries}); dan \omterm{def:g5:first-look-at-cells:cell}{sel} tubuhnya
(\cref{prop:g5:first-look-at-cells:growth}) merangkainya kembali
menjadi zat \omterm{def:g1:animals-around-us:animal}{hewannya} sendiri. \omterm{def:g1:animals-around-us:animal}{Hewan} adalah \emph{pengubah} materi,
tidak pernah pembuat-dari-mineral: perdagangan itu milik para \omterm{def:g5:food-webs:roles}{produsen}
belaka.
\end{proposition}
\admitted

\begin{example}[Rumput menjadi sapi]\label{ex:g6:matter-from-food:cow}
Ikutilah setengah tonnya: sapinya merumput jam demi jam
(\cref{prop:g2:what-animals-eat:daily} sudah menjelaskan jam yang
panjang itu); \omterm{def:g4:journey-of-food:digestion}{pencernaannya} yang berat
(\cref{exo:g4:journey-of-food:11}) membongkar rumputnya menjadi \omterm{def:g4:journey-of-food:digestion}{zat
gizi}; \omterm{prop:g4:journey-of-food:absorb}{darahnya} membawanya ke ambing, \omterm{def:g3:bones-and-muscles:muscle}{otot} dan anak sapi yang tumbuh di
dalamnya; \omterm{def:g5:first-look-at-cells:cell}{selnya} membangun materi-sapi dari materi-rumput. Padang
penggembalaannya, secara harfiah, sedang berjalan-jalan dalam wujud
yang baru.
\end{example}

\begin{remark}[Materi berpindah tangan, bukan berganti sifat]\label{rem:g6:matter-from-food:hands}
Pada setiap anak panah sebuah \omterm{def:g3:food-chains:chain}{rantai makanan}
(\cref{def:g3:food-chains:chain}), \omterm{def:g6:plants-are-producers:matter}{materi organik} berganti pemilik lalu
dibangun ulang --- materi-rumput menjadi materi-sapi, materi-sapi
(\omterm{def:g2:animals-grow-up:born}{susunya}, dagingnya) menjadi materi-anak. Yang \emph{organik} tidak
pernah datang dari ketiadaan: \omterm{def:g5:food-webs:roles}{konsumen} meneruskannya lalu membentuk
ulangnya. Hanya pada permulaan hijau rantainya ia dibuat baru
(\cref{prop:g6:plants-are-producers:firstlink}); hanya pada ujung para
\omterm{def:g5:food-webs:roles}{pengurainya}, pada bab berikutnya, ia kembali menjadi mineral.
\end{remark}

\section{Pertumbuhan adalah produksi materi}

\begin{definition}[Produksi materi seekor hewan]\label{def:g6:matter-from-food:production}
Yang disebut \emph{produksi materi}\index{produksi materi} seekor
\omterm{def:g1:animals-around-us:animal}{hewan} adalah \omterm{def:g6:plants-are-producers:matter}{materi organik} baru yang dibangunnya: pertumbuhan si muda
(kilogram tambahan anak sapinya), pembaruan tahunan (\omterm{def:g1:animals-around-us:covering}{bulu} yang
ditanggalkan lalu ditumbuhkan lagi, ranggah, \omterm{def:g1:animals-around-us:covering}{bulu burung}), perbaikan
(\omterm{def:g3:bones-and-muscles:skeleton}{tulang} yang menyatu pada \cref{rem:g3:bones-and-muscles:alive}), dan
hasil (\omterm{def:g2:animals-grow-up:born}{susu}, \omterm{def:g2:animals-grow-up:born}{telur}, wol). Semuanya adalah makanan yang dibangun ulang.
\end{definition}

\begin{example}[Menimbang produksinya]\label{ex:g6:matter-from-food:weighing}
Petani menimbang produksinya setiap hari: seekor sapi perah memberi
sekitar \qty{25}{kg} \omterm{def:g2:animals-grow-up:born}{susu} sehari --- \omterm{def:g6:plants-are-producers:matter}{materi organik} yang harus
dibangunnya ulang dari pakannya, hari demi hari; sebutir \omterm{def:g2:animals-grow-up:born}{telur} ayam
adalah sekitar \qty{60}{g} bekal yang terkemas
(\cref{rem:g2:animals-grow-up:egg}), yang dirangkai dalam semalam dari
biji-bijiannya. Tanpa pakan, tanpa produksi: pembukuannya seimbang
tanpa ampun.
\end{example}

\begin{proposition}[Tidak semua makanan menjadi tubuh]\label{prop:g6:matter-from-food:losses}
Seorang \omterm{def:g5:food-webs:roles}{konsumen} tidak pernah mengubah seluruh makanannya menjadi
materi baru:
\begin{itemize}
  \item sebagian tidak pernah terserap, dan keluar sebagai tinja
        (\cref{prop:g4:journey-of-food:absorb});
  \item sebagian besar \emph{dibakar} --- dipadukan dengan oksigen
        dari napasnya
        (\cref{prop:g4:breathing:exchange}) --- untuk menenagai
        gerakan dan kehangatan: materi yang dibelanjakan sebagai bahan
        bakar, yang keluar sebagai gas yang diembuskan;
  \item hanya sisanya yang menjadi pertumbuhan dan hasil.
\end{itemize}
Dari rumput yang dimakan seekor sapi, hanya pecahan yang sederhana yang
berakhir sebagai sapi yang baru --- dan itulah sebabnya tiap lantai
piramida pada \cref{prop:g5:food-webs:pyramid} jauh lebih kecil
daripada lantai di bawahnya.
\end{proposition}
\admitted

\begin{example}[Piramidanya, yang akhirnya dijelaskan]\label{ex:g6:matter-from-food:pyramid}
Berton-ton rumput sebuah padang rumput memberi makan kelinci sebanyak
satu kawanan; berkilo-kilogram kelinci memberi makan satu atau dua
ekor rubah. Pada tiap anak panahnya, sebagian besar materinya dibakar
untuk hidup atau hilang, dan hanya sebuah pecahan yang diteruskan ke
atas sebagai tubuh. Pemburu besar itu langka
(\cref{prop:g5:food-webs:pyramid}) karena mereka hidup dari
\emph{sisanya sisa} --- hitung-hitungan, bukan nasib buruk.
\end{example}

\section{Daftar makanan, gigi dan satu perdagangan yang sama}

\begin{proposition}[Daftar makanan berbeda, perubahan yang sama]\label{prop:g6:matter-from-food:menus}
\omterm{def:g2:what-animals-eat:kinds}{Herbivora}, \omterm{def:g2:what-animals-eat:kinds}{karnivora}, \omterm{def:g2:what-animals-eat:kinds}{omnivora}
(\cref{def:g2:what-animals-eat:kinds}) menyebutkan \emph{sumber}
\omterm{def:g6:plants-are-producers:matter}{materi organik} yang berbeda, bukan perdagangan yang berbeda. Sapi yang
membangun ulang rumput dan rubah yang membangun ulang kelinci
sama-sama menjalankan perubahan pada
\cref{prop:g6:matter-from-food:transform} --- dengan perlengkapan yang
cocok dengan daftar makanannya
(\cref{prop:g2:what-animals-eat:match},
\cref{exo:g4:journey-of-food:11}): rumput menuntut penggilas dan
\omterm{def:g4:journey-of-food:digestion}{pencernaan} yang luas, daging memberi upah kepada penerkam dan usus
yang pendek.
\end{proposition}
\admitted

\begin{method}[Mengaudit materi seekor hewan]\label{met:g6:matter-from-food:audit}
Untuk \omterm{def:g1:animals-around-us:animal}{hewan} apa pun, susunlah pembukuan materinya:
\begin{enumerate}
  \item \emph{masuk}: \omterm{def:g6:plants-are-producers:matter}{materi organik} apa yang dimakannya, dan dari
        siapa? (anak panah \omterm{def:g5:food-webs:web}{jaring makanan} yang masuk kepadanya,
        \cref{def:g5:food-webs:web});
  \item \emph{dibangun}: pertumbuhan, pembaruan dan hasil apa yang
        dibuatnya?
  \item \emph{keluar}: tinja, dan materi yang dibakar untuk gerakan
        serta kehangatan;
  \item periksalah keseimbangannya: produksinya pastilah sisa yang
        sederhana --- kalau sebuah pengakuan membuatnya lebih besar
        (``kolam ini menghasilkan \omterm{prop:g3:sorting-living-things:groups}{materi-ikan} lebih banyak daripada
        makanan yang ditumbuhkan kolamnya''), pengakuannya keliru.
\end{enumerate}
\end{method}

\begin{example}[Auditnya ketika dipakai]\label{ex:g6:matter-from-food:audituse}
Auditlah \omterm{prop:g3:sorting-living-things:groups}{burung} hitam (\cref{ex:g2:what-animals-eat:winter}): masuk
--- cacing, \omterm{prop:g3:sorting-living-things:groups}{serangga}, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} beri; dibangun --- pertumbuhannya sendiri,
\omterm{def:g1:animals-around-us:covering}{bulu burung} barunya tiap tahun, kelompok \omterm{def:g2:animals-grow-up:born}{telurnya}; keluar --- tinja di
tembok, dan bagian besar yang terbakar yang menenagai setahun terbang
dan berkicau. Berat makanan harian seekor \omterm{prop:g3:sorting-living-things:groups}{burung} penyanyi sangat besar
bagi ukurannya justru karena terbang membakar begitu banyak bagian
pembukuannya.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:1}
Nyatakan perdagangan para \omterm{def:g5:food-webs:roles}{konsumen}, lalu pertentangkan dengan
perdagangan para \omterm{def:g5:food-webs:roles}{produsen}, masing-masing dalam satu kalimat.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:2}
Daftarkan langkah yang mengubah rumput menjadi \omterm{def:g3:bones-and-muscles:muscle}{otot} sapi, dengan
menyebutkan bab yang membahas masing-masingnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:3}
Berikan empat wujud \omterm{def:g6:matter-from-food:production}{produksi materi} seekor \omterm{def:g1:animals-around-us:animal}{hewan}, dengan satu contoh
masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:4}
Sebutkan ketiga tujuan makanan seorang \omterm{def:g5:food-webs:roles}{konsumen}, menurut
\cref{prop:g6:matter-from-food:losses}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:5}
Mengapa seekor sapi perah membutuhkan pakan setiap hari agar terus
memberi \omterm{def:g2:animals-grow-up:born}{susu}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:matter-from-food:6}
Sebutir \omterm{def:g2:animals-grow-up:born}{telur} memuat \qty{60}{g} \omterm{def:g6:plants-are-producers:matter}{materi organik}. Di mana ia dibuat ---
di dalam ayamnya, atau sebelum ayamnya? Uraikan dengan cermat.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:7}
Pakai \cref{prop:g6:matter-from-food:losses} untuk menjelaskan mengapa
piramida makanannya menyempit pada tiap lantainya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:8}
Jalankan \cref{met:g6:matter-from-food:audit} atas seekor kucing
peliharaan: masuk, dibangun, keluar --- dengan sedikitnya dua butir
per barisnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:9}
``Rubah dan sapi melakukan hal yang berbeda dengan makanannya.''
Betulkan kalimat itu dengan
\cref{prop:g6:matter-from-food:menus} --- apa yang berbeda, dan apa
yang serupa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:10}
Seorang anak tumbuh \qty{3}{kg} dalam setahun sambil memakan beberapa
ratus kilogram makanan. Ke mana perginya sisanya? Jawablah dengan
pembukuannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:matter-from-food:11}
\omterm{def:g1:animals-around-us:covering}{Cangkang} siput adalah zat mineral, namun
\cref{ex:g6:plants-are-producers:sorting} menghitungnya sebagai buatan
siputnya. Damaikanlah: apa yang dilakukan tubuh siputnya, dan dari
pasokan apa, untuk membangunnya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:matter-from-food:12}
Satu hektare tanah dapat memberi makan lebih banyak orang dengan
kentang daripada dengan daging babi yang dibesarkan dengan kentang
itu. Jelaskan dengan
\cref{prop:g6:matter-from-food:losses} --- ke mana perginya
perbedaannya ketika kentangnya menempuh jalan memutar lewat babinya?
\end{exercise}

\section{Soal: Pembukuan Ladang}

\begin{problem}[{Soal akhir pekan --- sebuah ladang kecil yang
diaudit, dari petak sampai meja makan}]\label{pb:g6:matter-from-food:1}
Lahan kecil milik Nenek: sebuah padang penggembalaan dengan satu ekor
sapi, sebuah kandang berisi sepuluh ayam yang diberi makan biji-bijian,
sepetak kebun sayur, dan dapur keluarganya. Akhir pekan ini, kamu
menyimpan pembukuan materinya.

\medskip\noindent\textbf{Bagian I --- Lantai para \omterm{def:g5:food-webs:roles}{produsennya}.}
\begin{enumerate}
  \item Daftarkan \omterm{def:g5:food-webs:roles}{produsen} di lahannya. Apa masukan mineralnya, dan
        apa sumber \omterm{prop:g3:balanced-diet:jobs}{energinya}?
  \item Padang penggembalaannya menumbuhkan sekitar \qty{10000}{kg}
        materi rumput dalam satu musim. Di dalam \omterm{def:g1:plants-around-us:parts}{daun} siapa setiap
        gramnya dibuat?
  \item Nenek menebarkan pupuk kandang dari kandang ayamnya ke petak
        sayurnya. Aturan \cref{prop:g5:people-and-nature:lasting} yang
        mana yang sedang dipegangnya, dan apa tepatnya yang diisi
        ulang pupuk kandangnya?
  \item Mengapa sapinya tidak dapat sekadar diberi \omterm{prop:g4:what-plants-need:needs}{garam mineral} dan
        air secara langsung, sehingga padang penggembalaannya tidak
        perlu?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Lantai para \omterm{def:g5:food-webs:roles}{konsumennya}.}
\begin{enumerate}[resume]
  \item Sapinya memakan kira-kira \qty{70}{kg} rumput sehari dan
        memberi sekitar \qty{20}{kg} \omterm{def:g2:animals-grow-up:born}{susu}. Sebutkan ketiga tujuan
        \qty{50}{kg} yang hilang itu.
  \item Ayamnya mengubah biji-bijian menjadi \omterm{def:g2:animals-grow-up:born}{telur}. Tuliskan pembukuan
        ayamnya --- masuk, dibangun, keluar --- lalu sebutkan mengapa
        ayam yang sedang berganti \omterm{def:g1:animals-around-us:covering}{bulu} (memperbarui seluruh \omterm{def:g1:animals-around-us:covering}{bulunya})
        bertelur lebih sedikit untuk sementara waktu.
  \item Seekor elang menyambar seekor anak ayam. Perpanjanglah \omterm{def:g3:food-chains:chain}{rantai
        makanan} lahannya sampai kepadanya, lalu nyatakan pada mata
        rantai mana materinya \emph{dibuat} dan pada mata rantai mana
        ia hanya diubah.
  \item Keluarganya memakan \omterm{def:g2:animals-grow-up:born}{telur}, \omterm{def:g2:animals-grow-up:born}{susu} dan sayur. Lacaklah \omterm{def:g6:plants-are-producers:matter}{materi
        organik} satu sarapan --- roti, \omterm{def:g2:animals-grow-up:born}{telur} rebus, \omterm{def:g2:animals-grow-up:born}{susu} ---
        masing-masing sampai ke \omterm{def:g1:plants-around-us:parts}{daun} tempat ia pertama kali dibuat.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Hitung-hitungan
lantainya.}
\begin{enumerate}[resume]
  \item Nenek berkata: ``padang penggembalaan ini memberi makan satu
        ekor sapi; ia tidak akan memberi makan lebih daripada satu
        sarang kecil rubah kalau rubah memakan kelinci yang diberi
        makan rumput di situ.'' Belalah perkiraannya dengan kehilangan
        pada tiap lantainya.
  \item Petak sayurnya memberi makan keluarganya secara langsung.
        Dengan memakai logika \cref{exo:g6:matter-from-food:12},
        jelaskan mengapa petaknya memberi makan lebih banyak orang per
        meter persegi daripada jalur padang-rumput-sapi-susu.
  \item Ketika kemarau, rumputnya berhenti tumbuh. Ramalkan, secara
        berurutan, apa yang terjadi pada \omterm{def:g2:animals-grow-up:born}{susu} sapinya, pada beratnya,
        dan pada baris ``dibangun'' pada pembukuannya --- lalu sebutkan
        mengapa tambahan sinar matahari tidak dapat menolong selama
        hujannya gagal (\cref{pb:g6:plants-are-producers:1},
        pertanyaan 7, memegang kuncinya).
  \item Tutuplah pembukuannya dengan satu kalimat per lantainya:
        \omterm{def:g5:food-webs:roles}{produsen}, \omterm{def:g5:food-webs:roles}{konsumen} --- lalu sebutkan siapa yang masih hilang
        dari pembukuannya (pokok bab berikutnya).
\end{enumerate}
\end{problem}
