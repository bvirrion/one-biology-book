\section{Satu masalah, empat mesin}

\begin{proposition}[Spesifikasi bersamanya]\label{prop:g7:how-animals-breathe:spec}
Setiap organ \omterm{def:g7:what-is-respiration:respiration}{pernapasan}, siapa pun pemiliknya, dibangun menurut satu
spesifikasi, yang sudah dijumpai pada
\cref{rem:g4:breathing:sponge}: sebuah permukaan yang \emph{luas},
\emph{tipis} dan \emph{lembap}, tempat \omterm{prop:g4:journey-of-food:absorb}{darah} \omterm{def:g1:animals-around-us:animal}{hewannya} (atau cairan
tubuhnya) mendekat cukup dekat ke sekelilingnya sehingga \omterm{def:g7:what-is-respiration:respiration}{oksigen} dapat
menyeberang masuk dan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} menyeberang keluar. Luas agar
penyeberangannya cukup; tipis agar penyeberangannya mudah; lembap
karena gasnya menyeberang dalam keadaan terlarut. Mesinnya berbeda
dalam cara ia mendekatkan permukaan itu kepada sekelilingnya.
\end{proposition}
\admitted

\begin{definition}[Keempat mesinnya]\label{def:g7:how-animals-breathe:four}
\omterm{def:g1:animals-around-us:animal}{Hewan} mendekatkan permukaan pertukarannya dengan empat cara utama:
\begin{enumerate}
  \item \emph{paru-paru}\index{paru-paru}: kantong berisi udara yang
        terlipat \emph{di dalam} tubuhnya, yang diangini oleh gerakan
        \omterm{def:g7:what-is-respiration:respiration}{pernapasan} --- \omterm{prop:g3:sorting-living-things:groups}{mamalia}, \omterm{prop:g3:sorting-living-things:groups}{burung}, \omterm{prop:g4:classifying-animals:classes}{reptil}, \omterm{prop:g3:sorting-living-things:groups}{amfibi} dewasa;
  \item \emph{insang}\index{insang}: permukaan berumbai yang
        didekatkan \emph{di luar atau di tepi tubuhnya}, yang dicuci
        oleh arus air --- \omterm{prop:g3:sorting-living-things:groups}{ikan}
        (\cref{exo:g4:breathing:11}), \omterm{ex:g2:animals-grow-up:frog}{berudu}, banyak larva air,
        kerang;
  \item \emph{trakea}\index{trakea serangga}: sebuah jaringan pipa
        udara halus yang dipipakan dari lubang di sepanjang tubuhnya
        langsung ke setiap organ --- \omterm{prop:g3:sorting-living-things:groups}{serangga}; udaranya diantarkan
        dari pintu ke pintu, sebagian besar tanpa pertolongan
        \omterm{prop:g4:journey-of-food:absorb}{darahnya};
  \item \emph{pernapasan kulit}\index{pernapasan kulit}: seluruh \omterm{def:g1:the-five-senses:sense}{kulit}
        yang lembap sebagai permukaan pertukarannya --- cacing \omterm{def:g6:decomposers-and-soil:soil}{tanah}
        sepenuhnya; \omterm{prop:g3:sorting-living-things:groups}{amfibi} sebagai tambahan besar bagi paru-parunya
        yang sederhana.
\end{enumerate}
\end{definition}

\begin{example}[Mengamati mesinnya berjalan]\label{ex:g7:how-animals-breathe:watch}
Tiap mesinnya memperlihatkan iramanya. Mulut dan tutup \omterm{def:g7:how-animals-breathe:four}{insang} \omterm{prop:g3:sorting-living-things:groups}{ikan}
troutnya memompa dalam pergantian yang mantap --- air masuk di
mulutnya, melewati \omterm{def:g7:how-animals-breathe:four}{insangnya}, keluar di bawah tutupnya. Perut belalang
yang beristirahat berdenyut --- memompa pipa udaranya. Tenggorokan
kataknya bergetar --- mendorong udara ke dalam \omterm{def:g7:how-animals-breathe:four}{paru-parunya} (tanpa
sangkar rusuk yang mengerjakannya). Hanya cacing \omterm{def:g6:decomposers-and-soil:soil}{tanahnya} yang tidak
memperlihatkan apa-apa: \omterm{def:g7:how-animals-breathe:four}{pernapasan kulit} sama sekali tidak punya
gerakan, dan justru itulah sebabnya
\cref{def:g7:what-is-respiration:respiration} membedakan
pertukarannya dari pemompaannya.
\end{example}

\section{Mesinnya cocok dengan mediumnya --- dan dengan kehidupannya}

\begin{proposition}[Mengapa tiap mesin di tempatnya]\label{prop:g7:how-animals-breathe:fits}
Mesinnya memetakan diri pada medium dan kehidupannya
(\cref{def:g4:adapted-to-their-world:adaptation} yang bekerja pada
\omterm{def:g7:what-is-respiration:respiration}{pernapasan}):
\begin{itemize}
  \item \omterm{def:g7:how-animals-breathe:four}{insang} cocok bagi air: permukaan berumbainya mengapung
        terpisah di dalamnya, dan dicuci terus-menerus --- lalu ambruk
        tanpa guna di udara (\omterm{prop:g3:sorting-living-things:groups}{ikan} trout yang tercekik pada
        \cref{exo:g4:breathing:11});
  \item \omterm{def:g7:how-animals-breathe:four}{paru-paru} cocok bagi udara: karena terlipat di dalam, ia tetap
        lembap di tempat air terbuka akan hilang --- tetapi ia tidak
        dapat memanen \omterm{def:g7:what-is-respiration:respiration}{oksigen} air yang terlarut dan langka;
  \item \omterm{def:g7:how-animals-breathe:four}{trakea} cocok bagi tubuh yang kecil: hebat pada jarak
        milimeter, ia mengantar terlalu lambat pada jarak yang panjang
        --- salah satu sebab \omterm{prop:g3:sorting-living-things:groups}{serangga} tetap kecil;
  \item \omterm{def:g7:how-animals-breathe:four}{pernapasan kulit} cocok bagi kehidupan yang kecil dan berkulit
        lembap di tempat yang lembap: seluruh permukaan cacingnya
        nyaris hanya melayani tubuhnya yang tipis, dan udara kering
        membunuhnya --- ingatlah cacing \omterm{def:g6:decomposers-and-soil:soil}{tanah} yang terdampar sesudah
        hujan.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Dua kehidupan, dua mesin dalam satu hewan]\label{ex:g7:how-animals-breathe:frog}
Katak menjalankan seluruh bab ini dalam satu kehidupan: \omterm{def:g7:how-animals-breathe:four}{insang} ketika
\omterm{ex:g2:animals-grow-up:frog}{berudu} (mesin milik penghuni air,
\cref{ex:g2:animals-grow-up:frog}), \omterm{def:g7:how-animals-breathe:four}{paru-paru} ditambah \omterm{def:g1:the-five-senses:sense}{kulit} yang
lembap ketika dewasa --- \omterm{def:g7:how-animals-breathe:four}{paru-parunya} sederhana, \omterm{def:g1:the-five-senses:sense}{kulitnya} mengerjakan
sebagian besar pekerjaannya, terutama sepanjang musim dingin yang
dingin di dasar kolam
(\cref{exo:g6:life-through-seasons:11}: katak yang dorman itu bernapas
lewat \omterm{def:g1:the-five-senses:sense}{kulitnya} belaka). \omterm{def:g2:animals-grow-up:metamorphosis}{Metamorfosis} membangun ulang bahkan permesinan
\omterm{def:g7:what-is-respiration:respiration}{pernapasannya}.
\end{example}

\begin{example}[Bernapas dengan udara sambil hidup di air]\label{ex:g7:how-animals-breathe:airinwater}
Penghuni air yang berparu-paru harus pulang pergi. Lumba-lumba dan
paus muncul ke permukaan untuk menyembur lalu \omterm{def:g4:breathing:movements}{menarik napas}
(\cref{ex:g6:classification-kinship:whale}: \omterm{def:g7:how-animals-breathe:four}{paru-paru} \omterm{prop:g3:sorting-living-things:groups}{mamalia}, yang
diwarisi dari nenek moyang daratnya); siput kolam memanjat ke
permukaan untuk mengisi ulang; kumbang selam membawa sebuah gelembung
udara keperakan di bawah tutup sayapnya --- tangki seorang penyelam,
yang diperbarui di permukaan. \omterm{def:g2:where-animals-live:habitat}{Habitat} dapat ditawar; medium mesinnya
tidak.
\end{example}

\begin{method}[Membaca pernapasan seekor hewan]\label{met:g7:how-animals-breathe:read}
Diberikan seekor \omterm{def:g1:animals-around-us:animal}{hewan} yang belum dikenal:
\begin{enumerate}
  \item di mana ia hidup --- dan dapatkah ia meninggalkannya? (air,
        udara, \omterm{def:g6:decomposers-and-soil:soil}{tanah}; pulang pergi atau menetap);
  \item carilah tanda mesinnya: tutup \omterm{def:g7:how-animals-breathe:four}{insang} atau berkas berumbai;
        lubang hidung dan dada yang bergerak; deretan lubang mungil di
        sepanjang tubuh \omterm{prop:g3:sorting-living-things:groups}{serangga}; atau tidak ada yang terlihat kecuali
        sebuah \omterm{def:g1:the-five-senses:sense}{kulit} yang lembap;
  \item amati iramanya: tutup yang memompa, perut yang berdenyut, dada
        yang naik, kunjungan ke permukaan;
  \item sebutkan mesinnya --- lalu periksalah ia terhadap mediumnya
        dengan
        \cref{prop:g7:how-animals-breathe:fits}.
\end{enumerate}
\end{method}

\begin{remark}[Spesifikasinya tidak pernah berubah]\label{rem:g7:how-animals-breathe:same}
Di balik keberagamannya, periksalah salah satu di antara keempat
mesinnya dari dekat dan \cref{prop:g7:how-animals-breathe:spec} muncul
kembali: umbai \omterm{def:g7:how-animals-breathe:four}{insangnya}, kantong \omterm{def:g7:how-animals-breathe:four}{paru-parunya}, ujung terhalus
\omterm{def:g7:how-animals-breathe:four}{trakeanya} dan \omterm{def:g1:the-five-senses:sense}{kulit} cacingnya semuanya adalah tempat pertemuan yang
luas-tipis-lembap antara sekeliling dan tubuhnya. Empat mesin, satu
spesifikasi --- \omterm{prop:g5:first-look-at-cells:principle}{kesatuan kehidupan}
(\cref{prop:g5:first-look-at-cells:principle}), yang tertulis pada
perlengkapan \omterm{def:g7:what-is-respiration:respiration}{pernapasan}.
\end{remark}

\section{Soal: Pendataan Para Pernapas di Kolam}

\begin{problem}[{Soal akhir pekan --- satu kolam, setiap mesin
bekerja}]\label{pb:g7:how-animals-breathe:1}
Sesore di kolam dengan jaring, kotak pengamat dan buku catatan.
Pendataannya: benih \omterm{prop:g3:sorting-living-things:groups}{ikan} trout dan \omterm{prop:g3:sorting-living-things:groups}{ikan} duri; \omterm{ex:g2:animals-grow-up:frog}{berudu} (sebagian sudah
berkaki belakang); katak dewasa; siput kolam yang menjelajahi selaput
permukaannya; seekor kumbang selam yang berkilat keperakan; larva
lalat sehari dengan berkas berumbai di bawah batu; dan, di halaman
rumput lembap pada tebingnya, cacing \omterm{def:g6:decomposers-and-soil:soil}{tanah}.

\medskip\noindent\textbf{Bagian I --- Menyebutkan mesinnya.}
\begin{enumerate}
  \item Berikan kepada tiap anggota pendataannya mesin \omterm{def:g7:what-is-respiration:respiration}{pernapasannya}.
  \item Berkas berumbai larva lalat seharinya melambai terus-menerus
        bahkan di air yang tenang. Berkas itu apa, dan untuk apa
        lambaiannya?
  \item Anggota pendataan mana yang harus mengunjungi permukaannya, dan
        mengapa persis mereka?
  \item \omterm{ex:g2:animals-grow-up:frog}{Berudu} yang belum berkaki dan yang sudah berkaki boleh jadi
        berbeda di dalam sebanyak perbedaannya di luar. Pembangunan
        ulang apa yang sedang berjalan, menurut
        \cref{ex:g7:how-animals-breathe:frog}?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Mesinnya di bawah
tekanan.}
\begin{enumerate}[resume]
  \item Seminggu yang panas dan tanpa angin menurunkan \omterm{def:g7:what-is-respiration:respiration}{oksigen}
        terlarut kolamnya (\cref{ch:g7:oxygen-in-the-water} akan
        menjelaskan mengapa). Urutkan pendataannya: siapa yang
        menderita lebih dahulu --- para pemakai \omterm{def:g7:how-animals-breathe:four}{insang}, atau para
        pengunjung permukaan? Mengapa?
  \item Pada minggu itu, \omterm{prop:g3:sorting-living-things:groups}{ikan} durinya menggantung tepat di bawah
        permukaan, dengan mulut bekerja pada selaputnya. Tafsirkan
        perilakunya dengan spesifikasinya: apa yang istimewa pada
        milimeter teratas airnya?
  \item Seorang pengarung yang ceroboh mengaduk awan lumpur halus.
        Mesin yang mana yang tersumbat, dan apa yang kamu harapkan
        dilakukan \omterm{def:g1:animals-around-us:animal}{hewan} yang terkena?
  \item Sesudah hujan semalam, cacing \omterm{def:g6:decomposers-and-soil:soil}{tanah} di halaman rumputnya
        tergeletak di jalan setapak. Jelaskan pemandangan klasik itu:
        mesin yang mana, dan \omterm{def:g6:exploring-our-environment:conditions}{kondisi} yang mana, yang mengusir mereka ke
        atas?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Peta yang lebih besar.}
\begin{enumerate}[resume]
  \item Susunlah pendataannya sebagai sebuah tabel: \omterm{def:g1:animals-around-us:animal}{hewan}, mesin,
        medium, pulang pergi atau menetap. Satu spesifikasi tunggal apa
        yang mengepalai setiap lajur mesinnya, menurut
        \cref{rem:g7:how-animals-breathe:same}?
  \item Kolamnya membeku di permukaannya pada bulan Januari, sehingga
        permukaannya tersegel. Ramalkan nasib musim dinginnya:
        kataknya (\cref{exo:g6:life-through-seasons:11}), siputnya,
        kumbangnya, \omterm{prop:g3:sorting-living-things:groups}{ikannya} --- siapa yang kesulitan, siapa yang
        tercukupi?
  \item Seorang pendatang baru tiba: seekor laba-laba air yang menganyam
        sebuah lonceng sutra lalu mengisinya dengan udara permukaan.
        Tempatkan kiatnya di samping kiat kumbangnya, lalu sebutkan
        mesin yang tetap dilayaninya.
  \item Simpulkan: mengapa satu kolam kecil membutuhkan empat mesin
        \omterm{def:g7:what-is-respiration:respiration}{pernapasan} yang berbeda agar terhuni sepenuhnya? Satu kalimat,
        dengan memakai \emph{medium} dan \emph{\omterm{def:g7:what-is-respiration:respiration}{oksigen}}.
\end{enumerate}
\end{problem}
