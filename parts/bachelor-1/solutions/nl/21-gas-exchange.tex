\section*{Hoofdstuk \ref{ch:b1:gas-exchange} --- Gasuitwisseling bij de ademhaling}

\begin{solution}{exo:b1:gas-exchange:1}
$\dot V = K(A/x)(P_1 - P_2)$: een groter oppervlak, een dunnere barrière,
een hogere \omterm{def:b1:gas-exchange:partial}{partiële druk} buiten (ventilatie) en een lagere binnen
(doorbloeding).
\end{solution}

\begin{solution}{exo:b1:gas-exchange:2}
Dertig keer minder zuurstof per liter (\qty{7}{} tegen \qty{210}{mL/L});
achthonderd keer dichter en vijftig keer stroperiger, dus duur om te
pompen; zuurstof diffundeert er tienduizend keer trager doorheen, zodat
het midden helemaal langs het oppervlak moet worden bewogen.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:3}
Meestroom: water en bloed vertrekken beide bij \qty{12}{kPa}; onttrekking
$(21 - 12)/21 = \qty{43}{\%}$. \omterm{def:b1:gas-exchange:gill}{Tegenstroom}: het water vertrekt bij
\qty{4}{kPa}, het bloed bij \qty{18}{kPa}; onttrekking $(21 - 4)/21 =
\qty{81}{\%}$.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:4}
Opgelost (\qty{7}{\%}), gebonden aan de aminogroepen van hemoglobine
(\qty{23}{\%}), als bicarbonaat in het \omterm{def:b1:mammal-organization:compartments}{plasma} (\qty{70}{\%}).
\end{solution}

\begin{solution}{exo:b1:gas-exchange:5}
Minuut \qty{6.75}{L/min}; alveolair $15\times 0.3 = \qty{4.5}{L/min}$;
opname $4.5\times 0.07 = \qty{0.32}{L/min}$. Oppervlakkig: minuut nog
steeds \qty{6.75}{L/min}, maar alveolair $30\times 0.075 =
\qty{2.25}{L/min}$ en opname \qty{0.16}{L/min} --- de helft, voor
hetzelfde ademwerk; de dode ruimte wordt bij elke ademteug betaald.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:6}
$2/(6\times 0.75) = \qty{0.44}{L}$ per uur, tweeëntwintig keer haar
volume van \qty{20}{mL}.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:7}
$A/x$ daalt met $2\times 4 = 8$: \qty{31}{mL/min}. Om \qty{250}{} te
herstellen moet het verschil \qty{64}{kPa} zijn --- onmogelijk, want de
alveolaire druk kan in lucht de \qty{13}{kPa} niet overschrijden en het
aderlijke bloed kan niet onder nul zakken; de patiënt kan het alleen
verhogen door zuurstof te ademen, en zelfs dan niet achtvoudig.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:8}
\omterm{prop:b1:gas-exchange:transport}{Koolzuuranhydrase} zit in de \omterm{def:b1:cell-unit-of-life:cell}{cel}, zodat het bicarbonaat daar wordt gemaakt
en zich ophoopt; het diffundeert naar buiten langs zijn gradiënt door een
wisselaar die voor elk bicarbonaat een chloride binnenlaat, waardoor de
lading in evenwicht blijft. Zonder die ruil zou het bicarbonaat binnen
blijven, samen met de partner van zijn proton, wat de osmolariteit van de
\omterm{def:b1:cell-unit-of-life:cell}{cel} zou verhogen en er water zou laten instromen: de \omterm{def:b1:cell-unit-of-life:cell}{cel} zou in de
\omterm{def:b1:body-plans-tissues:tissue}{weefsels} opzwellen (zij zwelt sowieso al een beetje).
\end{solution}

\begin{solution}{exo:b1:gas-exchange:9}
De ademprikkel komt van stijgend $\mathrm{CO_2}$. Hyperventileren verlaagt
het $\mathrm{CO_2}$ ver onder normaal zonder veel zuurstof toe te voegen
(het hemoglobine was al verzadigd); tijdens de duik daalt de zuurstof tot
het niveau van bewusteloosheid voordat het $\mathrm{CO_2}$ is teruggeklommen
tot het niveau dat een ademteug afdwingt. Zonder hyperventileren bereikt
het $\mathrm{CO_2}$ dat niveau terwijl er nog volop zuurstof is, en komt de
duiker boven.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:10}
$A = \pi(10^{-5})^2 = \qty{3.1e-10}{m^2}$; $\Delta c = JL/DA =
10^{-9}\times 10^{-3}/(2\times 10^{-5}\times 3.1\times 10^{-10}) =
\qty{160}{mol/m^3}$ --- veel meer dan de beschikbare \qty{8.7}{}, zodat
een zo nauwe buis de spier op \qty{1}{mm} niet door diffusie kan
bevoorraden; er is een wijdere buis nodig (\qty{100}{\micro m}:
$\Delta c = \qty{6.4}{mol/m^3}$, net mogelijk) of actieve ventilatie. Bij
\qty{10}{mm} is de eis opnieuw tien keer zo groot. Omdat de lengte van de
buis met de lichaamsgrootte meegroeit en de vraag met het volume, begrenst
diffusie door tracheeën de grootte van insecten tot enkele centimeters.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:11}
Vissen: water in één richting, bloed tegengesteld (\omterm{def:b1:gas-exchange:gill}{tegenstroom}),
onttrekking \qty{80}{\%}. Vogels: lucht in één richting door de
parabronchi, bloed er haaks op (dwarsstroom), een onttrekking die
tussenin ligt maar hoger is dan die van zoogdieren, met een
\omterm{prop:b1:organism-environment:surfaces}{uitwisselingsoppervlak} dat nooit oude lucht ontvangt. Een getijdenlong
mengt verse lucht met het achtergebleven gas, zodat het
\omterm{prop:b1:organism-environment:surfaces}{uitwisselingsoppervlak} hoogstens het alveolaire mengsel ziet en het bloed
zich alleen daarmee in evenwicht kan stellen: hoe groot het oppervlak ook
is, slagaderlijk bloed kan de alveolaire druk niet overschrijden, en de
alveolaire druk kan de ingeademde niet benaderen zolang er restvolume is.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:12}
$\mathrm{CO_2}$ is het betere signaal omdat zijn slagaderlijke gehalte
rechtstreeks en steil op de ventilatie reageert (halveer de ventilatie en
het verdubbelt), het via de pH nauwkeurig door de centrale receptoren wordt
gemeten, en zuurstof, die op het vlakke bovenstuk van de hemoglobinekromme
zit, over het normale ventilatiebereik weinig verandert: een
$\mathrm{CO_2}$-thermostaat houdt de zuurstof als bijproduct goed. Zij
faalt wanneer de zuurstof los van het $\mathrm{CO_2}$ daalt --- op hoogte,
waar de ingeademde zuurstof laag is maar de $\mathrm{CO_2}$-productie niet,
zodat het $\mathrm{CO_2}$-signaal ``adem minder'' zegt en de zuurstof
``adem meer''; alleen de halsslagaderlichaampjes nemen het over wanneer de
zuurstof ernstig laag is, en het compromis (hyperventilatie met alkalose)
heeft dagen van nierbijstelling nodig om tot rust te komen.
\end{solution}

\begin{solution}{pb:b1:gas-exchange:1}
\textbf{1.} Lucht $210/22.4 = \qty{9.4}{mmol/L}$; water $7/22.4 =
\qty{0.31}{mmol/L}$.
\textbf{2.} 30.
\textbf{3.} Lucht: $1.2/9.4 = \qty{0.13}{g}$; water: $1000/0.31 =
\qty{3200}{g}$ --- \num{25000} keer meer massa.
\textbf{4.} $210/7 = \qty{30}{L}$ water.
\textbf{5.} Water is te zwaar en te arm aan zuurstof om heen en weer te
worden geduwd; het moet er één keer doorheen, wat ook de \omterm{def:b1:gas-exchange:gill}{tegenstroom}
mogelijk maakt; lucht is goedkoop genoeg om twee keer te bewegen.
\textbf{6.} Minuut $12\times 0.5 = \qty{6}{L/min}$; alveolair
$12\times 0.35 = \qty{4.2}{L/min}$.
\textbf{7.} $150/500 = \qty{30}{\%}$ van elke ademteug, en van de minuutventilatie.
\textbf{8.} Afgeleverd $4.2\times 0.21 = \qty{0.88}{L/min}$; opname
$4.2\times 0.07 = \qty{0.29}{L/min}$, dicht bij \qty{250}{mL/min}.
\textbf{9.} $0.25/(6\times 0.21) = \qty{20}{\%}$ (\qty{28}{\%} van wat de
\omterm{def:b1:gas-exchange:lung}{longblaasjes} bereikt).
\textbf{10.} $0.14\times(101 - 6.3) = \qty{13.3}{kPa}$.
\textbf{11.} $250/5 = \qty{50}{mL/L}$; aderlijk \qty{150}{mL/L},
\qty{75}{\%} verzadigd.
\textbf{12.} $3000/20 = \qty{150}{mL/L}$; aderlijk \qty{50}{mL/L},
\qty{25}{\%} verzadigd.
\textbf{13.} $50/60 = \qty{0.83}{mL/min}$.
\textbf{14.} $0.83/(7\times 0.8) = \qty{0.149}{L/min}$, \qty{8.9}{L}
per uur.
\textbf{15.} $0.83/(7\times 0.43) = \qty{0.28}{L/min}$: de \omterm{def:b1:gas-exchange:gill}{tegenstroom}
halveert het te pompen water, en het werk.
\textbf{16.} \qty{8.9}{kg} water per uur; de mens verplaatst $6\times
60\times 1.2 = \qty{430}{g}$ lucht per uur --- een twintigste van de
massa, voor driehonderd keer zoveel zuurstof ($15\,000$ tegen
\qty{50}{mL} per uur).
\textbf{17.} $100\times(0.95 - 0.15) = \qty{80}{mL/L}$; hartminuutvolume
$0.83/0.080 = \qty{10.4}{mL/min}$.
\textbf{18.} Water/bloed $149/10.4 = 14$; lucht/bloed $4.2/5 = 0.84$.
De vis moet veertien volumes water per volume bloed langsvoeren omdat
elk zo weinig zuurstof bevat; de mens ongeveer één.
\textbf{19.} Opname $\times 1.5$ uit water dat er $5.5/7 = 0.79$ keer
zoveel bevat: ventilatie $\times 1.9$; de kosten van het ademen, ruwweg
evenredig met de stroom (of meer, want zij stijgen sneller dan lineair),
gaan van \qty{10}{\%} naar minstens \qty{13}{\%} van een grotere opname
--- de vis werkt harder om in warm water te ademen.
\textbf{20.} Aan het instroomeinde van het water: $21 - 18 = \qty{3}{kPa}$;
aan het uitstroomeinde: $4 - 3 = \qty{1}{kPa}$: aan beide einden positief.
\textbf{21.} Aan het instroomeinde $21 - 3 = \qty{18}{kPa}$; aan het uitstroomeinde $12 - 12 =
0$: de twee vloeistoffen staan in evenwicht en er is geen gradiënt meer,
hoewel het water nog \qty{57}{\%} van zijn zuurstof bevat.
\textbf{22.} Op de meestroom: het bloed komt in evenwicht met één enkel,
goed gemengd alveolair gas en vertrekt bij diens druk; het kan de
\qty{13.3}{kPa} niet overschrijden omdat er stroomafwaarts geen verser gas
te ontmoeten is, zoals dat bij de \omterm{def:b1:gas-exchange:gill}{tegenstroom} wel het geval is.
\textbf{23.} Forel $2000/(2\times 10^{-4}) = \num{1e7}$ (cm$^2$ per cm);
mens $10^6/(5\times 10^{-5}) = \num{2e10}$: verhouding \num{2000}. Opnamen
$250/0.83 = 300$. De forel neemt zeven keer meer zuurstof op per eenheid
$A/x$: haar gemiddelde drukverschil over een lamel moet dus groter zijn
--- wat de \omterm{def:b1:gas-exchange:gill}{tegenstroom} levert, die over het hele oppervlak een gradiënt in
stand houdt waar die van de getijdenlong tegen het eind van de
evenwichtsinstelling klein is.
\textbf{24.} $\mathrm{CO_2}$ is vijfentwintig keer beter oplosbaar in
water dan zuurstof: het water voert het even snel af als het wordt
gemaakt, en het $\mathrm{CO_2}$ in het bloed van een vis blijft zeer laag.
\textbf{25.} De \omterm{def:b1:gas-exchange:gill}{kieuw} ongeveer \qty{80}{\%}, de \omterm{def:b1:gas-exchange:lung}{long} ongeveer \qty{20}{\%}
van de zuurstof in het ingeademde midden; de eenrichtingsstroom van de
\omterm{def:b1:gas-exchange:gill}{kieuw} en haar tegenstroomopstelling van water en bloed maken het verschil.
\end{solution}
