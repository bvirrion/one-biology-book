\section*{Hoofdstuk \ref{ch:g4:heart-and-blood} --- Het hart en het bloed}

\begin{solution}{exo:g4:heart-and-blood:1}
\omterm{def:g4:journey-of-food:digestion}{Voedingsstoffen}, opgehaald bij de \omterm{def:g4:journey-of-food:tract}{dunne darm}; en zuurstof, opgehaald bij
de \omterm{def:g4:breathing:organs}{longen}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:2}
Koolstofdioxide --- afgeleverd bij de \omterm{def:g4:breathing:organs}{longen}, waar het uitgeademd wordt.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:3}
Het is een holle \omterm{def:g3:bones-and-muscles:muscle}{spier}, ongeveer zo groot als de vuist van zijn
eigenaar, midden in de borst achter de ribbenkast.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:4}
De rechterhelft pompt \omterm{prop:g4:journey-of-food:absorb}{bloed} naar de \omterm{def:g4:breathing:organs}{longen} om zuurstof te laden; de
linkerhelft pompt het zuurstofrijke \omterm{prop:g4:journey-of-food:absorb}{bloed} naar het hele lichaam.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:5}
Het golfje dat elke \omterm{def:g4:heart-and-blood:heart}{hartslag} door de vaten drijft, voelbaar aan de \omterm{def:g1:my-body:joint}{pols}
of in de \omterm{def:g1:my-body:parts}{hals}. Het tellen ervan telt de \omterm{def:g4:heart-and-blood:heart}{hartslagen} per minuut.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:6}
De duim heeft zelf een goed voelbaar klein polsje, dus zou je je duim
kunnen tellen in plaats van de \omterm{def:g1:my-body:joint}{pols} die je onderzoekt.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:7}
Sprintende \omterm{def:g1:my-body:parts}{benen} verbranden snel \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} en zuurstof en moeten
die snel vervangen krijgen; het \omterm{def:g4:heart-and-blood:heart}{hart} versnelt en versterkt zijn slagen
zodat het \omterm{prop:g4:journey-of-food:absorb}{bloed} zijn bezorgrondes vaker maakt.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:8}
Open antwoord. Meestal ongeveer $80$--$100$ in rust en $120$--$150$ na
de sprongen --- de bevoorradingsketen aan het werk.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:9}
Bij het verlaten van de \omterm{def:g4:breathing:organs}{longen} had hij net zuurstof geladen ---
helderrood. In de \omterm{def:g3:bones-and-muscles:muscle}{spier} gaf hij de zuurstof af en nam hij
koolstofdioxide op --- donkerder. De kleur volgt de lading.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:10}
De \omterm{def:g4:journey-of-food:tract}{dunne darm} brengt \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} in het \omterm{prop:g4:journey-of-food:absorb}{bloed}; de \omterm{def:g4:breathing:organs}{longen} brengen er
zuurstof in en halen er koolstofdioxide uit; het \omterm{def:g4:heart-and-blood:heart}{hart} is de pomp die het
\omterm{prop:g4:journey-of-food:absorb}{bloed} tussen beide en naar elke afnemer in beweging houdt --- \omterm{def:g3:bones-and-muscles:muscle}{spieren},
hersenen, \omterm{def:g3:bones-and-muscles:skeleton}{botten}, \omterm{def:g1:the-five-senses:sense}{huid}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:11}
Het getrainde \omterm{def:g4:heart-and-blood:heart}{hart} is een sterkere \omterm{def:g3:bones-and-muscles:muscle}{spier}: elke slag verplaatst meer
\omterm{prop:g4:journey-of-food:absorb}{bloed}. Dezelfde bevoorrading leveren vraagt dan minder slagen per minuut
--- dus doet de rustige $50$ van de wielrenner het werk van de
ongetrainde $75$, met reservekracht achter de hand voor inspanning.
\end{solution}
