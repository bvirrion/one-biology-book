\chapter{Nerves and Hormones}\label{ch:g12:nerves-hormones}

A sprinter hears the starting signal and is already leaving the blocks;
minutes later, adrenaline still tightens focus and liberates fuel. Fast
and slow control share the same goal --- coordinating the body --- but
use different languages: electrical spikes along \omterm{def:g12:nerves-hormones:neuron}{neurons}, and \omterm{def:g12:nerves-hormones:synapse}{chemical}
messengers in the blood. This chapter builds the \omterm{def:g12:nerves-hormones:neuron}{neuron} and the \omterm{def:g12:nerves-hormones:synapse}{synapse},
sketches the \omterm{def:g12:nerves-hormones:ap}{action potential}, surveys endocrine glands, and closes the
loop of \omterm{def:g12:nerves-hormones:homeostasis}{homeostasis} by \omterm{def:g12:nerves-hormones:negative}{negative feedback}.

\section{The neuron}

\begin{definition}[Neuron]\label{def:g12:nerves-hormones:neuron}
A \emph{neuron}\index{neuron} is an excitable \omterm{def:g10:the-cell:cell}{cell} specialised for
receiving, integrating and transmitting information. Typical parts: a
\emph{cell body}\index{cell body (neuron)} (soma) with the nucleus;
\emph{dendrites}\index{dendrite} that receive inputs; an \emph{axon}
\index{axon} that conducts output signals; and axon terminals that form
\omterm{def:g12:nerves-hormones:synapse}{synapses} with other \omterm{def:g10:the-cell:cell}{cells}.
\end{definition}

\begin{definition}[Glia]\label{def:g12:nerves-hormones:glia}
\emph{Glial cells}\index{glia} support \omterm{def:g12:nerves-hormones:neuron}{neurons}: insulation
(\emph{myelin}\index{myelin} sheaths), nutrient supply, defence and
regulation of the extracellular milieu. Myelination speeds conduction
in vertebrates.
\end{definition}

\begin{definition}[Resting potential]\label{def:g12:nerves-hormones:resting}
The \emph{resting membrane potential}\index{resting potential} is the
voltage difference across the \omterm{def:g10:membranes-enzymes:fluid-mosaic}{plasma membrane} of a quiet \omterm{def:g12:nerves-hormones:neuron}{neuron},
typically about \qty{-70}{mV} (inside negative). It arises from ion
gradients (high K$^{+}$ inside, high Na$^{+}$ outside) and selective
membrane permeability, maintained by the Na$^{+}$/K$^{+}$ pump.
\end{definition}

\section{The action potential}

\begin{definition}[Action potential]\label{def:g12:nerves-hormones:ap}
An \emph{action potential}\index{action potential} is a rapid, brief
reversal of membrane potential that propagates along the \omterm{def:g12:nerves-hormones:neuron}{axon} without
decrement when threshold is reached. It is the digital-like signal of
long-distance neural communication.
\end{definition}

\begin{proposition}[Phases of the spike]\label{prop:g12:nerves-hormones:phases}
A typical neuronal \omterm{def:g12:nerves-hormones:ap}{action potential} has: (1) depolarisation --- voltage-gated
Na$^{+}$ channels open, Na$^{+}$ enters, voltage rises toward positive;
(2) repolarisation --- Na$^{+}$ channels inactivate, K$^{+}$ channels
open, K$^{+}$ exits; (3) a short after-hyperpolarisation may follow
before return to rest. The \omterm{def:g10:molecules-of-life:proteins}{sequence} is all-or-none above threshold.
\end{proposition}
\admitted

\begin{definition}[Propagation and myelin]
\label{def:g12:nerves-hormones:prop}
In unmyelinated \omterm{def:g12:nerves-hormones:neuron}{axons} the spike regenerates continuously along the
membrane. In myelinated \omterm{def:g12:nerves-hormones:neuron}{axons}, regeneration occurs at \emph{nodes of
Ranvier}\index{node of Ranvier} (\emph{saltatory conduction}
\index{saltatory conduction}), which is faster and more energy-efficient.
\end{definition}

\begin{remark}[Frequency code]\label{rem:g12:nerves-hormones:freq}
Spike amplitude is roughly stereotyped; intensity of a stimulus is often
encoded by firing rate and by which \omterm{def:g12:nerves-hormones:neuron}{neurons} are recruited, not by
``bigger'' spikes.
\end{remark}

\section{The synapse}

\begin{definition}[Synapse]\label{def:g12:nerves-hormones:synapse}
A \emph{synapse}\index{synapse} is a specialised junction where a \omterm{def:g12:nerves-hormones:neuron}{neuron}
communicates with another \omterm{def:g12:nerves-hormones:neuron}{neuron} or an effector (muscle, gland). Most
synapses in the brain are \emph{chemical}\index{chemical synapse};
some are electrical (gap junctions).
\end{definition}

\begin{method}[Sequence at a chemical synapse]
\label{met:g12:nerves-hormones:synapse}
\begin{enumerate}
  \item \omterm{def:g12:nerves-hormones:ap}{Action potential} arrives at the terminal; voltage-gated Ca$^{2+}$
        channels open.
  \item Ca$^{2+}$ triggers vesicle fusion; \emph{neurotransmitter}
        \index{neurotransmitter} is released into the synaptic cleft.
  \item Transmitter binds receptors on the postsynaptic membrane.
  \item Ion channels open (or second messengers act): excitatory or
        inhibitory postsynaptic potentials result.
  \item Transmitter is removed by \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}, reuptake or enzymatic
        breakdown --- clearing the signal.
\end{enumerate}
\end{method}

\begin{example}[Neuromuscular junction]
\label{ex:g12:nerves-hormones:nmj}
Motor \omterm{def:g12:nerves-hormones:neuron}{neurons} release acetylcholine onto skeletal muscle; ligand-gated
channels depolarise the fibre and trigger contraction. Enzymatic
cleavage of acetylcholine ends the signal quickly.
\end{example}

\begin{definition}[Central and peripheral organisation]
\label{def:g12:nerves-hormones:cns}
The \emph{central nervous system}\index{central nervous system} (CNS)
comprises brain and spinal cord; the \emph{peripheral nervous system}
\index{peripheral nervous system} (PNS) links CNS to sensors and
effectors. The \emph{autonomic}\index{autonomic nervous system} division
regulates smooth muscle, heart and glands (sympathetic vs parasympathetic
balance).
\end{definition}

\section{Endocrine glands and hormones}

\begin{definition}[Hormone]\label{def:g12:nerves-hormones:hormone}
A \emph{hormone}\index{hormone} is a \omterm{def:g12:nerves-hormones:synapse}{chemical} messenger secreted into
extracellular fluid (usually blood) that acts on distant target \omterm{def:g10:the-cell:cell}{cells}
bearing specific receptors. The \emph{endocrine system}\index{endocrine
system} is the set of glands and tissues that secrete hormones.
\end{definition}

\begin{definition}[Major endocrine glands]
\label{def:g12:nerves-hormones:glands}
Key vertebrate glands include: \emph{hypothalamus}\index{hypothalamus}
and \emph{pituitary}\index{pituitary} (master control); \emph{thyroid}
\index{thyroid} (metabolic rate); \emph{adrenal}\index{adrenal gland}
(stress: cortisol, adrenaline); \emph{pancreas}\index{pancreas}
(\omterm{ex:g12:nerves-hormones:insulin}{insulin}, \omterm{ex:g12:nerves-hormones:insulin}{glucagon}); gonads (sex \omterm{def:g10:molecules-of-life:lipids}{steroids}). Many organs (gut, heart,
kidney) also secrete \omterm{def:g12:nerves-hormones:hormone}{hormones}.
\end{definition}

\begin{proposition}[Classes of hormone action]
\label{prop:g12:nerves-hormones:classes}
Water-soluble \omterm{def:g12:nerves-hormones:hormone}{hormones} (peptides, catecholamines) bind surface receptors
and use second messengers. Lipid-soluble \omterm{def:g12:nerves-hormones:hormone}{hormones} (\omterm{def:g10:molecules-of-life:lipids}{steroids}, \omterm{def:g12:nerves-hormones:glands}{thyroid}
\omterm{def:g12:nerves-hormones:hormone}{hormones}) often cross membranes and alter \omterm{def:g11:gene-expression:gene}{gene} \omterm{def:g11:gene-expression:transcription}{transcription} via
intracellular receptors. Effects differ in onset and duration.
\end{proposition}
\admitted

\begin{example}[Insulin and glucagon]\label{ex:g12:nerves-hormones:insulin}
After a \omterm{def:g10:molecules-of-life:carbs}{carbohydrate} meal, rising blood glucose stimulates pancreatic
$\beta$ \omterm{def:g10:the-cell:cell}{cells} to release \emph{insulin}\index{insulin}, promoting
glucose uptake and storage. Fasting stimulates \emph{glucagon}
\index{glucagon}, mobilising glucose from the liver. The pair stabilises
blood sugar.
\end{example}

\section{Homeostasis and feedback}

\begin{definition}[Homeostasis]\label{def:g12:nerves-hormones:homeostasis}
\emph{Homeostasis}\index{homeostasis} is the maintenance of relatively
stable internal conditions (temperature, pH, glucose, osmolarity, \ldots)
despite external and internal disturbances.
\end{definition}

\begin{definition}[Negative feedback]
\label{def:g12:nerves-hormones:negative}
In \emph{negative feedback}\index{negative feedback}, a change in a
regulated variable triggers responses that oppose the change and return
the variable toward a set point. Most homeostatic loops are negative.
\end{definition}

\begin{definition}[Positive feedback]
\label{def:g12:nerves-hormones:positive}
In \emph{positive feedback}\index{positive feedback}, a change amplifies
itself (e.g.\ oxytocin and uterine contractions in childbirth; the rising
phase of the \omterm{def:g12:nerves-hormones:ap}{action potential} via Na$^{+}$ influx). It drives rapid
completion of a process, not steady regulation.
\end{definition}

\begin{method}[Mapping a feedback loop]
\label{met:g12:nerves-hormones:loop}
\begin{enumerate}
  \item Identify the regulated variable and its sensor.
  \item Name the control centre (often \omterm{def:g12:nerves-hormones:glands}{hypothalamus} or endocrine gland).
  \item Name the effectors (muscles, glands, liver, vessels, \ldots).
  \item State whether the effector response reduces (negative) or
        amplifies (positive) the original deviation.
\end{enumerate}
\end{method}

\begin{example}[Thermoregulation]\label{ex:g12:nerves-hormones:thermo}
Hypothalamic sensors detect a fall in core temperature; shivering and
vasoconstriction raise heat production and reduce heat loss --- \omterm{def:g12:nerves-hormones:negative}{negative
feedback} toward the set point.
\end{example}

\begin{proposition}[Nervous vs endocrine timing]
\label{prop:g12:nerves-hormones:timing}
Neural signals act in milliseconds on precise targets; hormonal signals
often act in seconds to hours on broader sets of \omterm{def:g10:the-cell:cell}{cells}. The systems
interlock: the \omterm{def:g12:nerves-hormones:glands}{hypothalamus} is both neural and endocrine, and \omterm{def:g12:nerves-hormones:glands}{adrenal}
medulla release is neurally triggered.
\end{proposition}
\admitted

\section{Problem: From spike to set point}

\begin{problem}[{Weekend problem --- integrate neuronal signalling with
endocrine feedback in a single physiological story}]
\label{pb:g12:nerves-hormones:1}
A hiker steps on a snake, leaps back, then feels a pounding heart for
several minutes.

\medskip
\noindent\textbf{Part I --- Neural pathway.}
\begin{enumerate}
  \item Identify sensory input, CNS integration and motor output for the
        rapid leap (in outline).
  \item At a motor \omterm{def:g12:nerves-hormones:neuron}{neuron} terminal, list the steps that release
        acetylcholine.
  \item Sketch (axes only) an \omterm{def:g12:nerves-hormones:ap}{action potential} and mark threshold,
        depolarisation and repolarisation.
  \item Why does the leap begin in well under a second, whereas blood
        \omterm{def:g12:nerves-hormones:hormone}{hormone} levels change more slowly?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- \omterm{def:g12:nerves-hormones:ap}{Action potential} mechanism.}
\begin{enumerate}[resume]
  \item Which ions dominate the upstroke and the downstroke?
  \item What would happen if K$^{+}$ channels failed to open after the
        peak?
  \item Explain \omterm{def:g12:nerves-hormones:prop}{saltatory conduction} in a myelinated motor \omterm{def:g12:nerves-hormones:neuron}{axon}.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- \omterm{def:g12:nerves-hormones:hormone}{Hormones} and \omterm{def:g12:nerves-hormones:homeostasis}{homeostasis}.}
\begin{enumerate}[resume]
  \item Name \omterm{def:g12:nerves-hormones:hormone}{hormones} released in stress from \omterm{def:g12:nerves-hormones:glands}{adrenal} medulla and
        cortex, and one effect of each.
  \item After the scare, how does \omterm{def:g12:nerves-hormones:negative}{negative feedback} restore heart rate
        and \omterm{def:g10:scientific-biology:properties}{metabolism} toward baseline?
  \item Using \omterm{ex:g12:nerves-hormones:insulin}{insulin}/\omterm{ex:g12:nerves-hormones:insulin}{glucagon} or thermoregulation, write a full
        negative-feedback diagram in words.
  \item Give one example where \omterm{def:g12:nerves-hormones:positive}{positive feedback} is useful, and why it
        must be time-limited.
  \item In one paragraph, argue that nerves and \omterm{def:g12:nerves-hormones:hormone}{hormones} are
        \omterm{prop:g11:dna-replication:base-pairs}{complementary} control systems rather than alternatives.
\end{enumerate}
\end{problem}
