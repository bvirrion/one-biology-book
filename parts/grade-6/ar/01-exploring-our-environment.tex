\chapter{استكشاف بيئتنا}\label{ch:g6:exploring-our-environment}

اعبر ساحة المدرسة في ظهيرة تموز، فتجد الوجه الجنوبي من الجدار
حارًّا تحت يدك، والهواء فوق الإسفلت يترجرج، وليس في المرأى شيء أخضر
إلا عشبة في شقّ. ثم انتقل خلف الجدار الشمالي: هواء بارد، وتربة
ندية، وطحلب، وحشرات قمل خشبي تحت حجر. عالمان يفصل بينهما ثلاثة
أمتار. وتبدأ هذه السنة باستكشاف البيئات كما يفعل العلماء ---
بقياسها، وبالسؤال عن سبب عيش كل \omterm{def:g6:exploring-our-environment:components}{كائن حي} حيث يعيش بالضبط.

\section{مِمَّ تتكوّن البيئة}

\begin{definition}[مكوّنات البيئة]\label{def:g6:exploring-our-environment:components}
\emph{البيئة}\index{بيئة} --- ساحة، أو حرش، أو ضفة بركة --- تحتوي
ثلاثة أصناف من المكوّنات:
\begin{enumerate}
  \item \emph{الكائنات الحية}\index{كائن حي}: أي \omterm{def:g1:animals-around-us:animal}{الحيوانات}
        \omterm{def:g1:plants-around-us:plant}{والنباتات} والفطور والحياة المجهرية الموجودة؛
  \item و\emph{المادة المعدنية}\index{مادة معدنية}، وهي التي \omterm{def:g1:living-or-not:nonliving}{لم
        تكن حية قط}: الصخر، والحجر، والجزء المعدني من التربة،
        والماء، والهواء؛
  \item و\emph{آثار الكائنات الحية وبقاياها}: \omterm{def:g1:plants-around-us:parts}{الأوراق} الميتة،
        والخشب الساقط، والفضلات، \omterm{def:g1:animals-around-us:covering}{والريش}، والأصداف --- أي صندوق
        ``كان حيًّا'' في \cref{def:g1:living-or-not:nonliving}،
        وقد صار الآن عدّة قرائن للمستكشف.
\end{enumerate}
\end{definition}

\begin{example}[جرد خلف الجدار الشمالي]\label{ex:g6:exploring-our-environment:inventory}
جرد عشر دقائق في الركن البارد: الحي --- طحلب، وسرخسيات، وحشرات قمل
خشبي، وحلزون، وأم أربعة وأربعين، ولبلاب؛ والمعدني --- حجر الجدار،
وتربة ندية، وبركة ماء؛ والآثار --- \omterm{def:g1:plants-around-us:parts}{أوراق} ميتة، \omterm{def:g1:animals-around-us:covering}{وصدفة} حلزون، وفضلات
طيور على قمة الجدار. ثلاث قوائم، وقد وُصف الركن كما يوصف موقع
عالم.
\end{example}

\begin{remark}[الآثار معطيات]\label{rem:g6:exploring-our-environment:traces}
أكثر \omterm{def:g1:animals-around-us:animal}{الحيوانات} تختبئ أو تهرب، فيقرأ المستكشف \emph{الآثار} بدلها
في الغالب: فبندقة مقضومة تسمّي فأر الحرش، وشبكة عنكبوت تسمّي
صاحبها، وآثار أقدام في الطين تسمّي البلشون. والجرد الذي لا يعدّد
إلا ما سكن حتى رُئي يفوّت نصف السكان.
\end{remark}

\section{قياس الشروط}

\begin{definition}[الشروط البيئية]\label{def:g6:exploring-our-environment:conditions}
\emph{شروط}\index{شروط بيئية} \omterm{def:g6:exploring-our-environment:components}{البيئة} هي خصائصها الفيزيائية القابلة
للقياس:
\begin{itemize}
  \item \emph{درجة الحرارة}\index{درجة الحرارة}، وتُقاس بمحرار
        بالدرجات المئوية؛
  \item و\emph{الرطوبة}\index{رطوبة} --- أي مقدار ندى الهواء أو
        التربة؛
  \item و\emph{الضوء}\index{ضوء (شرط)} --- من الشمس الكاملة إلى
        الظل العميق.
\end{itemize}
والشروط تتغير من بقعة إلى بقعة --- تغيرًا حادًّا في نطاق أمتار في
الغالب --- وعبر النهار وعبر السنة.
\end{definition}

\begin{example}[الساحة مقيسةً]\label{ex:g6:exploring-our-environment:measured}
مسح ظهيرة تموز بالأدوات: عند قدم الجدار الجنوبي ---
\qty{38}{\degreeCelsius}، جفاف، شمس كاملة؛ وفي الركن الشمالي ---
\qty{22}{\degreeCelsius}، تربة ندية، ظل عميق. ثلاثة أمتار من
المسافة، وستّ عشرة درجة من الفرق. وعالما الفقرة الافتتاحية صارا
الآن \emph{سطرين في جدول} --- وذلك تقدم، لأن الجداول تُقارن.
\end{example}

\begin{proposition}[الشروط تقرر السكان]\label{prop:g6:exploring-our-environment:decide}
كائنات البقعة الحية تناسب شروطها: فكل \omterm{def:g5:biodiversity:species}{نوع} لا يزدهر إلا داخل مداه
الخاص من الحرارة \omterm{def:g6:exploring-our-environment:conditions}{والرطوبة} \omterm{def:g6:exploring-our-environment:conditions}{والضوء}. فالطحلب وحشرات القمل الخشبي
تحتاج إلى الندى (وكساء حشرة القمل الخشبي، بخلاف كساء \omterm{prop:g3:sorting-living-things:groups}{الحشرات}، يدع
الماء يفلت --- فتجفّ جفافًا قاتلًا في ساعة من الشمس)؛ أما عشبة
الشقّ فتحتمل الجفاف والوهج. وحيث تختلف \omterm{def:g6:exploring-our-environment:conditions}{الشروط} يختلف السكان ---
وهي القاعدة وراء كل \omterm{def:g2:where-animals-live:habitat}{موئل} في \cref{ch:g2:where-animals-live}.
\end{proposition}
\admitted

\begin{example}[صندوق اختيار حشرات القمل الخشبي]\label{ex:g6:exploring-our-environment:chooser}
التجربة الكلاسيكية: صندوق مغلق، نصفه مبطّن بورق ندي ونصفه بورق
جاف، وعشرات من حشرات القمل الخشبي تُطلق على خط الوسط. وفي دقائق
تكون كلها تقريبًا على الجانب الندي. وأعِدها بنصف مظلل في مقابل نصف
مضاء: فتتجمع في الظل. ولا أحد يسوقها --- بل كل \omterm{def:g1:animals-around-us:animal}{حيوان}، إذ يهيم
ويستدير، ينتهي حيث يرتاح جسمه. فخريطة حشرات القمل الخشبي
\emph{هي} خريطة الندى.
\end{example}

\begin{method}[مسح موقع]\label{met:g6:exploring-our-environment:survey}
لتمسح موقعًا صغيرًا مسح عالم ميداني:
\begin{enumerate}
  \item اختر مساحة صغيرة وعلّمها --- فالمتر المربع المفتَّش جيدًا
        يقول أكثر من هكتار يُلقى عليه نظرة؛
  \item وقِس \omterm{def:g6:exploring-our-environment:conditions}{الشروط}: الحرارة، والندى، \omterm{def:g6:exploring-our-environment:conditions}{والضوء} --- بالأدوات نفسها
        وبالطريقة نفسها في كل موقع (وهي قاعدة الإنصاف في
        \cref{met:g4:what-plants-need:fairtest})؛
  \item وجرِّد أصناف المكوّنات الثلاثة --- \omterm{def:g6:exploring-our-environment:components}{الكائنات الحية}،
        \omterm{def:g6:exploring-our-environment:components}{والمادة المعدنية}، والآثار --- مفتشًا كما ينبغي: تحت
        الحجارة (وأعدها إلى مكانها:
        \cref{met:g2:caring-for-nature:rules})، وفي الشقوق، وعلى
        وجوه \omterm{def:g1:plants-around-us:parts}{الأوراق} السفلى؛
  \item وسجّل كل شيء في الموقع --- قوائم، ورسومًا، وأعدادًا؛
        فالذاكرة ليست أداة تسجيل؛
  \item وقارن المواقع: واقرن كل فرق في السكان بفرق في \omterm{def:g6:exploring-our-environment:conditions}{الشروط}.
\end{enumerate}
\end{method}

\begin{remark}[لماذا تغلب السجلات الذكريات]\label{rem:g6:exploring-our-environment:records}
عند العودة إلى الداخل يتجادل ماسحان وثقا بالذاكرة أكانت الحلزونات
ثلاثًا أم خمسًا. والدفتر يفصل في ذلك --- أو يعترف بصدق بأن
الحلزونات لم تُعدّ. فالسجلات المكتوبة في الموقع هي التي تحوّل
النزهة معطيات؛ وكل علم في هذا الكتاب يجري عليها.
\end{remark}

\section{بيئة واحدة وعوالم دقيقة كثيرة}

\begin{definition}[الموئل الدقيق]\label{def:g6:exploring-our-environment:microhabitat}
داخل \omterm{def:g6:exploring-our-environment:components}{البيئة} الواحدة، كل بقعة صغيرة لها شروطها وسكانها --- الوجه
السفلي لحجر، وشقّ في جدار، وقدم السياج المظللة --- هي
\emph{موئل دقيق}\index{موئل دقيق}. وقد كانت طوابق الغابة في
\cref{rem:g2:where-animals-live:layers} \omterm{def:g2:where-animals-live:habitat}{موائل} دقيقة؛ والمتر المربع
عند الماسح يحوي عادةً عدة منها.
\end{definition}

\begin{example}[الحجر، أعلاه وأسفله]\label{ex:g6:exploring-our-environment:stone}
حجر واحد في الركن البارد: فوقه أشنة وذبابة تتشمس --- مضاء، جاف،
دافئ. وتحته حشرات قمل خشبي، وأم أربعة وأربعين، وجذيرات شاحبة ---
مظلم، ندي، بارد. موئلان دقيقان كاملان يفصل بينهما خمسة سنتيمترات
من الصخر، ويسكن كلًّا منهما بالضبط \omterm{def:g6:exploring-our-environment:components}{الكائنات الحية} التي توافق شروطُه
مداها.
\end{example}

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:3}
افرز في أصناف المكوّنات الثلاثة: سرخسة، \omterm{def:g1:animals-around-us:covering}{وصدفة} حلزون، وبركة ماء،
وفضلات طيور، وأم أربعة وأربعين، وحجر الجدار.

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:4}

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:5}
لماذا تموت حشرة القمل الخشبي في ساعة من الشمس، وعلام يفسّر ذلك في

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:6}

\begin{exercise}[$\star\star$]\label{exo:g6:exploring-our-environment:8}

يعدّد مسح ``بعض \omterm{prop:g3:sorting-living-things:groups}{الحشرات}، وكان الجو دافئًا نوعًا ما''. أعد كتابة

\section{مسألة: الخندقان}

\begin{problem}[{مسألة نهاية الأسبوع --- نادي علماء الطبيعة يمسح
خندقين على جانب الطريق فيجد عالمين
مختلفين}]\label{pb:g6:exploring-our-environment:1}
يمسح نادي علماء الطبيعة خندقين على الممر نفسه، يفصل بينهما $50$
مترًا. الخندق الشمالي يمتد على الجانب الشمالي من سياج نباتي عالٍ؛
والخندق الجنوبي يمتد في العراء مواجهًا الجنوب. ويعمل النادي حسب
الأصول: الأدوات نفسها، والمتر المربع نفسه لكل موقع، وزمن التفتيش
نفسه.

\medskip\noindent\textbf{القسم الأول --- القياسات.}
\begin{enumerate}
  \item يقرأ المحراران \qty{19}{\degreeCelsius} في الخندق الشمالي
        و\qty{31}{\degreeCelsius} في الجنوبي في الساعة نفسها. فأي
        صفة واحدة من صفات الموقعين تفسّر الفرق أحسن تفسير؟
  \item تربة الخندق الشمالي تلتصق بالأصابع؛ وتربة الجنوبي تتفتت
        غبارًا. ترجم الرصدين إلى معجم
        \cref{def:g6:exploring-our-environment:conditions}.
  \item يقيس النادي عند الظهر. أعطِ سببًا واحدًا يجعل المقارنة أقل
        إنصافًا لو قيس الشمالي عند الظهر والجنوبي في السادسة
        مساءً.
  \item لماذا يستعمل النادي إطار المتر المربع \emph{نفسه} وزمن
        التفتيش نفسه في الموقعين؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- الجردان.}
\begin{enumerate}[resume]
  \item قائمة الخندق الشمالي: طحلب، وسرخسيات، وحشرات قمل خشبي،
        وحلزونات، وعلجوم. وقائمة الجنوبي: أعشاب جافة، وعشبة كبيرة
        ذات \omterm{def:g1:plants-around-us:parts}{أوراق} شمعية، ونمل، وسحلية تتشمس، وجنادب. افرز أفراد
        كل قائمة في أصناف المكوّنات الثلاثة --- ولاحظ ما تفتقده
        القائمتان (راجع
        \cref{def:g6:exploring-our-environment:components}).
  \item ولكل من حشرات القمل الخشبي والعلجوم والسحلية، اذكر \omterm{def:g6:exploring-our-environment:conditions}{شروط}
        أي خندق توافق حاجاته، ولماذا (استعمل
        \cref{prop:g6:exploring-our-environment:decide} وما تعرفه
        عن كل \omterm{def:g1:animals-around-us:animal}{حيوان}).
  \item يجد النادي في الخندق الجنوبي \omterm{def:g1:animals-around-us:covering}{صدفة} حلزون لا حلزونًا،
        فيسأل: أتعيش الحلزونات هناك؟ فما الاحتمالان الصادقان
        اللذان يبقيهما أثر كهذا مفتوحين؟
  \item تحت حجر مسطح في الخندق الشمالي: ثماني حشرات قمل خشبي،
        وأم أربعة وأربعين، وجذيرات بيضاء. سمِّ \omterm{def:g6:exploring-our-environment:conditions}{شروط} \omterm{def:g6:exploring-our-environment:microhabitat}{الموئل
        الدقيق}، واذكر القاعدة التي وضعت أولئك السكان هناك.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- التفسير.}
\begin{enumerate}[resume]
  \item اقرن كل فرق في السكان بفرق في \omterm{def:g6:exploring-our-environment:conditions}{الشروط}: أكمل ``في الشمالي
        طحلب وعلاجيم لأن \dots؛ وفي الجنوبي سحالٍ وأعشاب ذات
        \omterm{def:g1:plants-around-us:parts}{أوراق} شمعية لأن \dots''.
  \item السياج مقرَّر اقتلاعه. تنبأ \omterm{def:g6:exploring-our-environment:conditions}{بشروط} الخندق الشمالي بعد
        صيفين --- وبسكانه، مستعملًا اقتران السؤال التاسع.
  \item يقترح عضو أن يُختبر التنبؤ كما ينبغي: بمسح الخندقين من
        جديد بعد ذهاب السياج. فلماذا يجب أن تُستعمل المواقع
        \emph{نفسها} والأدوات نفسها والموسم نفسه ليُعتدّ بالمسح
        الثاني؟
  \item ينتهي تقرير النادي بعبارة: ``السكان يرسمون خريطة
        \omterm{def:g6:exploring-our-environment:conditions}{الشروط}.'' اشرح الجملة بصندوق الاختيار
        (\cref{ex:g6:exploring-our-environment:chooser})، واذكر
        لماذا لا حاجة إلى سَوق ليرسم مثل هذه الخريطة نفسه.
\end{enumerate}
\end{problem}
