\section*{Bab \ref{ch:g7:what-is-respiration} --- Apa Itu Respirasi?}

\begin{solution}{exo:g7:what-is-respiration:1}
\omterm{def:g7:what-is-respiration:respiration}{Respirasi} adalah pertukaran gas yang membuat sebuah \omterm{def:g6:exploring-our-environment:components}{makhluk hidup}
mengambil \omterm{def:g7:what-is-respiration:respiration}{oksigen} dari sekelilingnya lalu melepaskan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}.
\omterm{def:g7:what-is-respiration:respiration}{Pernapasan} hanyalah gerakan memompa dada yang terlihat --- salah satu
cara melayani pertukarannya; pertukarannya sendiri berjalan di dalam
setiap \omterm{def:g5:first-look-at-cells:cell}{sel} yang hidup.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:2}
Sebuah sensor \omterm{def:g7:what-is-respiration:respiration}{oksigen} di dalam wadah tertutup --- memperlihatkan taraf
\omterm{def:g7:what-is-respiration:respiration}{oksigennya} turun; dan \omterm{prop:g7:what-is-respiration:shown}{air kapur} --- cairan bening yang berubah keruh
seperti susu ketika \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} melintasinya.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:3}
Untuk membuktikan bahwa perubahannya datang dari subjek yang hidup dan
bukan dari wadahnya, dari kehangatan harinya atau dari \omterm{prop:g7:what-is-respiration:shown}{air kapurnya}
sendiri: kembaran uji yang adil, dengan kehidupan sebagai satu-satunya
perbedaan.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:4}
\omterm{prop:g3:balanced-diet:jobs}{Energi}: di dalam setiap \omterm{def:g5:first-look-at-cells:cell}{sel}, \omterm{def:g4:journey-of-food:digestion}{zat gizi} perlahan dibakar dengan
\omterm{def:g7:what-is-respiration:respiration}{oksigennya}, sehingga melepaskan \omterm{prop:g3:balanced-diet:jobs}{energinya} bagi pekerjaan \omterm{def:g5:first-look-at-cells:cell}{selnya}.
\omterm{def:g7:what-is-respiration:respiration}{Karbon dioksida} adalah sisa yang sudah terbakar habis, yang diangkut
pergi.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:5}
Seekor tikus, seekor kutu kayu, ercis yang \omterm{def:g2:from-seed-to-plant:germination}{berkecambah}, sebuah
cendawan, siput kolam --- dan sesendok \omterm{def:g6:decomposers-and-soil:soil}{tanah} yang hidup.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:6}
Ujilah ia di dalam gelap: dengan pembuatan yang digerakkan \omterm{def:g6:exploring-our-environment:conditions}{cahaya}
sudah berhenti, \omterm{def:g7:what-is-respiration:respiration}{oksigen} toplesnya turun dan \omterm{prop:g7:what-is-respiration:shown}{air kapurnya} menjadi keruh
--- \omterm{def:g7:what-is-respiration:respiration}{respirasinya} tersingkap.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:7}
Kemiripannya: keduanya melahap \omterm{def:g7:what-is-respiration:respiration}{oksigen} lalu melepaskan \omterm{def:g7:what-is-respiration:respiration}{karbon
dioksida}; keduanya melepaskan \omterm{prop:g3:balanced-diet:jobs}{energi} bahan bakarnya. Perbedaannya:
pembakaran oleh \omterm{def:g5:first-look-at-cells:cell}{selnya} berlangsung lambat dan tanpa nyala, dalam
ribuan langkah kecil; dan \omterm{prop:g3:balanced-diet:jobs}{energinya} ditangkap untuk pekerjaan alih-alih
hilang sebagai \omterm{def:g6:exploring-our-environment:conditions}{cahaya} dan panas sekaligus.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:8}
Anak yang berlari cepat (kerja \omterm{def:g3:bones-and-muscles:muscle}{otot} yang berat) $>$ anak yang membaca
(beban tetap tubuh yang tenang) $>$ ercis yang \omterm{def:g2:from-seed-to-plant:germination}{berkecambah} (membangun
dengan cepat, tetapi kecil) $>$ ercis kering (dorman --- \omterm{def:g7:what-is-respiration:respiration}{respirasinya}
menganggur mendekati nol). Lajunya mengikuti pekerjaannya.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:9}
Sepanjang malam \omterm{def:g7:what-is-respiration:respiration}{respirasi} ercis yang bertunas itu melahap \omterm{def:g7:what-is-respiration:respiration}{oksigen}
toplesnya lalu melepaskan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}; menjelang pagi terlalu
sedikit \omterm{def:g7:what-is-respiration:respiration}{oksigen} yang tersisa untuk menjaga sebuah nyala tetap menyala.
\omterm{prop:g7:what-is-respiration:shown}{Air kapurnya} akan menjadi keruh dengan mantap sepanjang malam.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:10}
Dua wadah dengan sensor dan perangkap \omterm{prop:g7:what-is-respiration:shown}{air kapur}: satu berisi \omterm{def:g6:decomposers-and-soil:soil}{tanah}
hidup yang segar, satu berisi \omterm{def:g6:decomposers-and-soil:soil}{tanah} yang sama yang dipanggang sampai
tanpa nyawa --- itulah satu-satunya perbedaannya. Kalau \omterm{def:g7:what-is-respiration:respiration}{oksigen} toples
yang hidup turun dan \omterm{prop:g7:what-is-respiration:shown}{air kapurnya} menjadi keruh sementara kembaran
yang dipanggang tetap tidak berubah, kehidupan tanahnyalah pelakunya.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:11}
Biji-bijian yang lembap terbangun: \omterm{def:g2:from-seed-to-plant:germination}{perkecambahan} bermula, dan
\omterm{def:g7:what-is-respiration:respiration}{respirasinya} naik dari sebuah bisikan menuju laju ercis yang bertunas
--- miliaran butir membakar cadangannya sekaligus, sehingga
memanaskan tumpukannya seperti awak kompos yang bekerja. Kehangatan
mempercepat pekerjaannya lebih jauh: sebuah tumpukan yang lembap dapat
memanaskan dirinya sendiri sampai membusuk. Biji-bijian yang kering
\omterm{prop:g1:healthy-habits:needs}{tidur} dalam keadaan dingin; biji-bijian yang lembap adalah sebuah
tungku yang lambat.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:12}
Perilaku: \omterm{def:g7:what-is-respiration:respiration}{pernapasan} dapat diamati --- gerakan dadanya, lima belas
kali semenit (\cref{ex:g4:breathing:watch}). Kimia: \omterm{def:g7:what-is-respiration:respiration}{respirasi} berjalan
pada benda yang sama sekali tidak berdada --- ercis, cendawan, \omterm{def:g6:decomposers-and-soil:soil}{tanah}
--- dan hanya terdeteksi lewat uji gas. Pelayanan: \omterm{def:g7:what-is-respiration:respiration}{pernapasan} ada
untuk memperbarui udara \omterm{def:g4:breathing:organs}{paru-parunya} sehingga pertukaran gasnya dapat
berlanjut --- dan berpacu persis ketika pembakaran oleh \omterm{def:g5:first-look-at-cells:cell}{selnya} berpacu
(\cref{ex:g4:breathing:running}).
\end{solution}

\begin{solution}{pb:g7:what-is-respiration:1}
\textbf{1.} Tanpa perubahan: \omterm{def:g7:what-is-respiration:respiration}{oksigennya} tetap, \omterm{prop:g7:what-is-respiration:shown}{air kapurnya} bening.
A adalah kontrol tanpa nyawanya --- kediaman yang menjadi pembanding
bagi setiap toples yang lain.

\textbf{2.} B: perubahan yang nyaris tidak terukur --- ercis kering
itu dorman, \omterm{def:g7:what-is-respiration:respiration}{respirasinya} menganggur. C: penurunan \omterm{def:g7:what-is-respiration:respiration}{oksigen} dan
kekeruhan yang jelas --- \omterm{def:g2:from-seed-to-plant:germination}{perkecambahan} adalah pembangunan dengan
kecepatan penuh, dan pekerjaannya dibayar dengan \omterm{def:g7:what-is-respiration:respiration}{respirasi}.

\textbf{3.} D: penurunan dan kekeruhan yang tercepat di antara
semuanya --- seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia} yang hangat dan bergerak membakar dengan
keras. Jamnya dibatasi karena tikusnya sedang melahap \omterm{def:g7:what-is-respiration:respiration}{oksigen}
toplesnya; percobaannya harus berakhir jauh sebelum udaranya menipis.

\textbf{4.} E: \omterm{def:g7:what-is-respiration:respiration}{oksigennya} tetap atau naik --- pembuatan oleh \omterm{def:g1:plants-around-us:plant}{tumbuhan}
yang tersinari itu menyerap \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} lalu melepaskan \omterm{def:g7:what-is-respiration:respiration}{oksigen}
lebih banyak daripada yang dipakai \omterm{def:g7:what-is-respiration:respiration}{respirasinya}. F: \omterm{def:g7:what-is-respiration:respiration}{oksigennya} turun,
\omterm{prop:g7:what-is-respiration:shown}{air kapurnya} menjadi keruh --- di dalam gelap pembuatannya berhenti
dan \omterm{def:g7:what-is-respiration:respiration}{respirasinya}, yang tidak pernah berhenti, terlihat terang-terangan.

\textbf{5.} F memperlihatkan bahwa \omterm{def:g1:plants-around-us:plant}{tumbuhannya} selalu berespirasi ---
\omterm{def:g7:what-is-respiration:respiration}{oksigen} masuk, \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} keluar, seperti tikusnya. E
menyembunyikan \omterm{def:g7:what-is-respiration:respiration}{respirasi} yang sama di bawah perdagangan siang hari
yang lebih besar: pemasukan dan pengeluaran pembuatannya
menenggelamkannya. Bersama-sama kembarnya memisahkan kedua
pertukarannya.

\textbf{6.} Ercis yang \omterm{def:g2:from-seed-to-plant:germination}{berkecambah} itu bekerja tiga kali lebih keras
per gramnya --- membangun \omterm{def:g1:plants-around-us:parts}{akar} dan tunas dengan kecepatan penuh ---
dan \omterm{def:g7:what-is-respiration:respiration}{respirasi} adalah bunyi mesin pekerjaan itu.

\textbf{7.} \omterm{def:g3:bones-and-muscles:muscle}{Otot} dan tubuh hangat tikusnya membakar \omterm{def:g4:journey-of-food:digestion}{zat gizi} dengan
\omterm{def:g7:what-is-respiration:respiration}{oksigen}; pembakarannya melepaskan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} ke udara toplesnya;
udara yang terperangkap itu melintasi \omterm{prop:g7:what-is-respiration:shown}{air kapurnya}; \omterm{def:g7:what-is-respiration:respiration}{karbon dioksidanya}
membuatnya keruh seperti susu.

\textbf{8.} ``Setiap \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} berespirasi --- \omterm{def:g1:animals-around-us:animal}{hewan}, \omterm{def:g1:plants-around-us:plant}{tumbuhan},
\omterm{def:g2:from-seed-to-plant:seed}{biji} yang terjaga atau yang nyaris tertidur --- dengan mengambil
\omterm{def:g7:what-is-respiration:respiration}{oksigen} lalu melepaskan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}.''

\textbf{9.} \omterm{def:g7:what-is-respiration:respiration}{Oksigennya} turun, \omterm{prop:g7:what-is-respiration:shown}{air kapurnya} menjadi keruh ---
cendawannya berespirasi. Ia menambahkan bahwa aturannya meliputi \omterm{def:g6:decomposers-and-soil:decomposers}{jamur}
juga: bukan \omterm{def:g1:plants-around-us:plant}{tumbuhan} dan bukan \omterm{def:g1:animals-around-us:animal}{hewan}, dan tanpa perkecualian.

\textbf{10.} Karena sebuah lumbung adalah toples C pada skala gudang:
biji-bijian yang disimpan dalam keadaan lembap mulai \omterm{def:g2:from-seed-to-plant:germination}{berkecambah} lalu
berespirasi keras, sehingga memanaskan dan membusukkan simpanannya ---
aturan kekeringan sang pengelola adalah pengendalian \omterm{def:g7:what-is-respiration:respiration}{respirasi}.

\textbf{11.} Segala yang ada di dalamnya: siput kolam dan \omterm{def:g1:animals-around-us:animal}{hewan} kecil,
\omterm{def:g6:decomposers-and-soil:decomposers}{pengurai} di lumpurnya, \omterm{def:g1:plants-around-us:plant}{tumbuhan} dan \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal hijau pada jam
gelapnya. Hal itu berarti karena penghuni kolamnya semuanya menimba
satu persediaan kecil \omterm{def:g7:what-is-respiration:respiration}{oksigen} terlarut --- ekonomi yang dibuka
bab-bab berikutnya.

\textbf{12.} Hanya D --- tikusnya --- yang memperlihatkan \omterm{def:g7:what-is-respiration:respiration}{pernapasan}:
gerakan dada yang terlihat, yang melayani pertukaran gasnya. Setiap
toples yang ada kehidupannya memperlihatkan \omterm{def:g7:what-is-respiration:respiration}{respirasi}: B dengan samar,
lalu C, D, F dengan terang-terangan, dan E yang tertutupi pembuatan
siang harinya tetapi tetap saja berespirasi.
\end{solution}
