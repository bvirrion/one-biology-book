\chapter{पौधों को क्या चाहिए}\label{ch:g4:what-plants-need}

हलीम के दो गमले एक ही खिड़की पर रखे हैं, एक ही दिन बोए गए। एक को पानी
मिलता है, दूसरे को नहीं --- और हफ़्ते भर में फ़र्क़ हरे और पीले रंग में
लिखा दिखता है। गमलों, \omterm{def:g2:from-seed-to-plant:seed}{बीजों} और थोड़े धीरज से तुम \omterm{def:g1:plants-around-us:plant}{पौधों} से ही इस अध्याय
के सवाल का उत्तर दिलवा सकते हो: \omterm{def:g1:plants-around-us:plant}{पौधे} को जीने और बढ़ने के लिए ठीक-ठीक
क्या चाहिए?

\section{पौधों से सवाल पूछना: न्यायी जाँच}

\begin{method}[न्यायी जाँच]\label{met:g4:what-plants-need:fairtest}
यह पता करने के लिए कि \omterm{def:g1:plants-around-us:plant}{पौधे} को कोई चीज़ चाहिए या नहीं --- पानी, प्रकाश,
मिट्टी, गरमाहट:
\begin{enumerate}
  \item एक ही तरह के \omterm{def:g1:plants-around-us:plant}{पौधों} के \emph{दो} गमले लो, एक ही ढंग से बोए
        हुए;
  \item \emph{जिस एक चीज़ की जाँच हो रही है उसे छोड़कर} बाक़ी सब में
        उनके साथ बिलकुल एक जैसा बरताव करो --- एक को वह चीज़ मिले,
        दूसरे को नहीं;
  \item दिनों या हफ़्तों तक रुको, और तुलना करो;
  \item जो भी फ़र्क़ दिखे वह उसी एक चीज़ से आया होगा जिसे तुमने बदला:
        जाँच न्यायी थी।
\end{enumerate}
जिस गमले को छुआ नहीं गया उसे \emph{नियंत्रण}\index{नियंत्रण (प्रयोग)}
कहते हैं: वह दिखाता है कि कुछ भी कम न हो तो क्या होता है।
\end{method}

\begin{example}[एक नहीं, दो गमले क्यों]\label{ex:g4:what-plants-need:control}
मान लो बिना पानी वाला अकेला गमला मुरझा जाता है। क्या यह पानी न मिलने
से हुआ --- या ठंड की किसी लहर से, या ख़राब \omterm{def:g2:from-seed-to-plant:seed}{बीजों} से? पर उसके बग़ल में
उसी पैकेट से, उसी कमरे में उगा उसका पानी पाया हुआ हरा जुड़वाँ खड़ा हो,
तो शक ग़ायब हो जाता है: जुड़वाँओं में फ़र्क़ सिर्फ़ पानी का था। एक-गमला
प्रयोग किसी को नहीं मनाते, जीवविज्ञानी को तो बिलकुल नहीं।
\end{example}

\section{जाँचें क्या बताती हैं}

\begin{proposition}[पौधे की ज़रूरतें]\label{prop:g4:what-plants-need:needs}
हर ज़रूरत के लिए बारी-बारी की गई न्यायी जाँचें दिखाती हैं कि \omterm{def:g1:plants-around-us:plant}{पौधे} को
चाहिए:
\begin{itemize}
  \item \emph{पानी} --- बिना सींचा गमला मुरझाकर मर जाता है;
  \item \emph{प्रकाश} --- अलमारी वाला गमला पीला, लंबा और कमज़ोर होकर
        हार मान जाता है: प्रकाश नहीं, तो \omterm{def:g1:plants-around-us:plant}{पौधे} का भोजन भी नहीं बनता;
  \item \emph{हवा} --- ताज़ी हवा से बंद कर देने पर बढ़ना रुक जाता है;
  \item \emph{गरमाहट} --- ठंड में \omterm{def:g2:from-seed-to-plant:seed}{बीज} इंतज़ार करते हैं
        (\cref{prop:g2:from-seed-to-plant:needs}) और बढ़ना जमकर रुक
        जाता है;
  \item \emph{खनिज लवण}\index{खनिज लवण}: पोषण देने वाले वे थोड़े-से
        पदार्थ जिन्हें \omterm{def:g1:plants-around-us:parts}{जड़ें} मिट्टी के पानी में घुला हुआ पीती हैं ---
        अकेले शुद्ध पानी में \omterm{def:g1:plants-around-us:plant}{पौधे} के पास इनकी जल्द ही कमी पड़ जाती है।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[पीला क़ैदी]\label{ex:g4:what-plants-need:pale}
\cref{exo:g1:plants-around-us:8} की अलमारी वाली जाँच, अगर उजाले में
रखे जुड़वाँ के साथ ठीक से की जाए, तो चौंकाने वाला नतीजा देती है: क़ैदी
सिर्फ़ उदास नहीं, \emph{पीला} है --- उसने अपना हरा पदार्थ बनाना बंद कर
दिया --- और अजीब तरह से \emph{लंबा} भी, मानो गुम प्रकाश को ढूँढ़ता हुआ
खिंच गया हो। प्रकाश के बिना बढ़ना कमाए बिना ख़र्च करना है; \omterm{def:g2:from-seed-to-plant:seed}{बीज} का जमा
किया भंडार जलते ही वह रुक जाता है।
\end{example}

\begin{remark}[माली मिट्टी को क्यों खिलाते हैं]\label{rem:g4:what-plants-need:soil}
खाद और गोबर \omterm{def:g1:plants-around-us:plant}{पौधों} का ``खाना'' नहीं हैं --- \omterm{def:g1:plants-around-us:plant}{पौधे} किसी को खाते ही नहीं।
वे मिट्टी के उन \omterm{prop:g4:what-plants-need:needs}{खनिज लवणों} को फिर से भर देते हैं जिन्हें फ़सलें साल दर
साल उठा ले जाती हैं। \cref{ex:g2:caring-for-nature:compost} का खाद का
डिब्बा यही घेरा पूरा करता है: मिट्टी ने जो सब्ज़ियों को दिया था, छिलके
वही मिट्टी को लौटा देते हैं।
\end{remark}

\section{पौधे जो लेते हैं उसका करते क्या हैं}

\begin{proposition}[पौधे अपना पदार्थ ख़ुद बनाते हैं]\label{prop:g4:what-plants-need:make}
\omterm{def:g1:animals-around-us:animal}{जंतु} दूसरे सजीवों को खाकर बढ़ते हैं। \omterm{def:g1:plants-around-us:plant}{पौधा} किसी को नहीं खाता: मिट्टी से
मिले पानी और \omterm{prop:g4:what-plants-need:needs}{खनिज लवणों}, हवा, और अपनी हरी \omterm{def:g1:plants-around-us:parts}{पत्तियों} से पकड़ी गई
\emph{प्रकाश} की \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} से वह \emph{अपना पदार्थ ख़ुद बनाता है} ---
नए \omterm{def:g1:plants-around-us:parts}{डंठल}, \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}, \omterm{def:g1:plants-around-us:parts}{जड़ें} और \omterm{def:g2:from-seed-to-plant:seed}{बीज}। इसीलिए \omterm{def:g1:plants-around-us:plant}{पौधे} के लिए प्रकाश खाने-पीने
जितना गंभीर मामला है, और इसीलिए हरी \omterm{def:g1:plants-around-us:parts}{पत्ती} आसमान की ओर मुँह किए रहती
है।
\end{proposition}
\admitted

\begin{example}[लगभग कुछ नहीं से एक पेड़]\label{ex:g4:what-plants-need:tree}
तौली हुई मिट्टी के किसी बड़े गमले में एक \omterm{def:g2:from-seed-to-plant:seed}{बीज} बोओ, सालों रुको, और वहाँ
एक छोटा \omterm{def:g1:plants-around-us:tree}{पेड़} खड़ा मिलता है --- फिर भी मिट्टी का वज़न लगभग ज़रा भी कम
नहीं हुआ। \omterm{def:g1:plants-around-us:tree}{पेड़} की लकड़ी गमले में से निकालकर नहीं ली गई: वह \omterm{def:g1:plants-around-us:parts}{पत्तियों} की
हरी कार्यशालाओं में पानी, हवा और प्रकाश से \emph{बनाई} गई। सजीवों का
पदार्थ कहाँ से आता है, यह सवाल आगे के सालों में हमें फिर व्यस्त रखेगा।
\end{example}

\begin{remark}[हरा रंग सबका पेट भरता है]\label{rem:g4:what-plants-need:everyone}
\cref{prop:g3:food-chains:start} याद करो: हर \omterm{def:g3:food-chains:chain}{आहार शृंखला} किसी \omterm{def:g1:plants-around-us:plant}{पौधे} से
शुरू होती है। अब हम देखते हैं कि ऐसा होना ही क्यों था --- \omterm{def:g1:plants-around-us:plant}{पौधे} ही एकमात्र
कड़ी हैं जो निर्जीव सामग्री से नया पदार्थ बनाते हैं। कहीं भी, कोई भी जो
कौर खाता है, वह पहले किसी \omterm{def:g1:plants-around-us:parts}{पत्ती} में बना था।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:1}
\omterm{def:g1:plants-around-us:plant}{पौधे} की पाँचों ज़रूरतें गिनाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:2}
न्यायी जाँच में दोनों गमलों के बीच कितनी चीज़ें अलग हो सकती हैं? जिस
गमले को छुआ नहीं गया उसे क्या कहते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:3}
पंद्रह दिन अँधेरे में उगे \omterm{def:g1:plants-around-us:plant}{पौधे} की शक्ल बताओ। कौन-सी ज़रूरत गुम थी, और
वह क्या नहीं बना सका?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:4}
\omterm{prop:g4:what-plants-need:needs}{खनिज लवण} क्या हैं, और \omterm{def:g1:plants-around-us:plant}{पौधा} उन्हें कैसे लेता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:5}
अगर \omterm{def:g1:plants-around-us:plant}{पौधे} खाते ही नहीं, तो माली खाद क्यों बिखेरते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:6}
\cref{prop:g4:what-plants-need:make} के अनुसार \omterm{def:g1:plants-around-us:plant}{पौधे} के बढ़ने का ढंग
\omterm{def:g1:animals-around-us:animal}{जंतु} के ढंग से कैसे अलग है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:7}
\cref{ex:g4:what-plants-need:tree} में \omterm{def:g1:plants-around-us:tree}{पेड़} के बढ़ने पर भी गमले की
मिट्टी का वज़न मुश्किल से घटा। \omterm{def:g1:plants-around-us:tree}{पेड़} का पदार्थ कहाँ से आया?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:what-plants-need:8}
करके देखो: चित्र वाली पानी की न्यायी जाँच हलीम के साथ बनाओ और चलाओ ---
दो गमले, एक ही फ़र्क़। एक हफ़्ते बाद जो दिखे उसे लिखो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:what-plants-need:9}
लुकास ``सिद्ध'' करता है कि \omterm{def:g1:plants-around-us:plant}{पौधों} को संगीत चाहिए: उसने एक महीने तक एक
गमले के सामने गाया, और वह ख़ूब फला-फूला।
\cref{met:g4:what-plants-need:fairtest} के अनुसार उसके प्रयोग में क्या
कमी है? वह रूप बनाओ जो सचमुच इसकी जाँच करता।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:what-plants-need:10}
पानी का एक \omterm{def:g1:plants-around-us:plant}{पौधा} पूरी तरह डूबा हुआ, बिना \omterm{def:g1:plants-around-us:parts}{जड़ों} के, फिर भी स्वस्थ बढ़ता
है। पाँचों ज़रूरतों में से किसे वह साफ़ तौर पर अब भी पूरा कर रहा है,
और कैसे --- और मिट्टी की जगह उसका पानी कौन-सी ज़रूरत पूरी करता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:what-plants-need:11}
पुदीने के दो एक जैसे गमले: एक में सिर्फ़ बारिश के पानी वाली रेत, दूसरे
में बग़ीचे की मिट्टी, दोनों को एक जैसा पानी और उजाला। दो महीने बाद रेत
वाला गमला छोटा और पीला पड़ रहा है। यह न्यायी जाँच कौन-सी ज़रूरत को अलग
करके दिखाती है --- और रेत वाला \omterm{def:g1:plants-around-us:plant}{पौधा} शुरू में बढ़ा ही क्यों?
\end{exercise}
