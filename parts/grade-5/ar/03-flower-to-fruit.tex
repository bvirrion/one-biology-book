\chapter{من الزهرة إلى الثمرة}\label{ch:g5:flower-to-fruit}

شاهد \cref{ch:g3:flowers-fruits-seeds} \omterm{def:g3:flowers-fruits-seeds:parts}{زهرة} الكرز تصير كرزة، وترك
سؤالًا مفتوحًا: \emph{ماذا يحدث فعلًا} حين تقع حبة لقاح على \omterm{def:g3:flowers-fruits-seeds:parts}{المدقة}؟
وهذه السنة، وقد تسلحنا بفكرة \omterm{def:g5:flower-to-fruit:steps}{الإخصاب} في
\cref{ch:g4:animal-reproduction} وبخلايا
\cref{ch:g5:first-look-at-cells}، نستطيع أن نفتح آلة \omterm{def:g3:flowers-fruits-seeds:parts}{الزهرة} كما
ينبغي.

\section{للأزهار أجزاء ذكرية وأنثوية}

\begin{proposition}[دورا الزهرة في واحدة]\label{prop:g5:flower-to-fruit:roles}
التكاثر في \omterm{def:g1:plants-around-us:plant}{النباتات} المزهرة تكاثر \emph{جنسي}، بمعنى
\cref{def:g4:animal-reproduction:sexual} بالضبط --- وأكثر الأزهار
تحمل الدورين معًا:
\begin{itemize}
  \item \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{الأسدية}} هي الجزء الذكري: فكل حبة من
        \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{حبوب اللقاح}} تحمل \omterm{def:g5:first-look-at-cells:cell}{خلايا} \omterm{def:g1:plants-around-us:plant}{النبات} التكاثرية الذكرية؛
  \item و\emph{\omterm{def:g3:flowers-fruits-seeds:parts}{المدقة}} هي الجزء الأنثوي: ففي عمقها تقع
        \emph{البييضات}\index{بييضة}، وفي كل واحدة \omterm{def:g4:animal-reproduction:sexual}{بويضة}.
\end{itemize}
و``\omterm{def:g2:from-seed-to-plant:seed}{البذور} المستقبلية'' في \cref{def:g3:flowers-fruits-seeds:parts}
تنال الآن اسمها الحقيقي: \omterm{prop:g5:flower-to-fruit:roles}{البييضات}، تنتظر أن تُخصَّب.
\end{proposition}
\admitted

\begin{remark}[الخطة نفسها كالحيوانات]\label{rem:g5:flower-to-fruit:sameplan}
\omterm{def:g5:first-look-at-cells:cell}{خلايا} ذكرية، \omterm{def:g4:animal-reproduction:sexual}{وبويضات}، \omterm{def:g4:animal-reproduction:sexual}{وإخصاب}: معجم
\cref{ch:g4:animal-reproduction} يُترجم كلمةً كلمة. إنها \omterm{prop:g5:first-look-at-cells:principle}{وحدة
الحياة} من جديد (\cref{prop:g5:first-look-at-cells:principle}) ---
فالأقحوان والدلافين تتكاثر على خطة واحدة كامنة، مهما تختلف الآلة
في المظهر.
\end{remark}

\section{التأبير ثم الإخصاب}

\begin{definition}[الخطوتان]\label{def:g5:flower-to-fruit:steps}
صنع \omterm{def:g2:from-seed-to-plant:seed}{البذرة} يحتاج إلى خطوتين متمايزتين:
\begin{enumerate}
  \item \emph{التأبير}\index{تأبير}: تسافر حبة لقاح من \omterm{def:g3:flowers-fruits-seeds:parts}{سداة} إلى
        قمة \omterm{def:g3:flowers-fruits-seeds:parts}{مدقة} --- تحملها حشرة
        (\cref{prop:g3:flowers-fruits-seeds:transform}) أو تحملها
        الريح؛
  \item و\emph{الإخصاب}\index{إخصاب (النبات)}: تُنبت الحبة التي
        وقعت أنبوبًا دقيقًا \emph{نازلًا في \omterm{def:g3:flowers-fruits-seeds:parts}{المدقة}}؛ وعبره تبلغ
        \omterm{def:g5:first-look-at-cells:cell}{الخلية} الذكرية بييضة وتلتحم ببويضتها.
\end{enumerate}
فالتأبير هو الرحلة؛ \omterm{def:g4:animal-reproduction:sexual}{والإخصاب} هو اللقاء.
\end{definition}

\begin{example}[لماذا تسقط الأزهار غير المزارة]\label{ex:g5:flower-to-fruit:unvisited}
صارت لقاعدة بستان \cref{rem:g3:flowers-fruits-seeds:bees} آليتها
الآن: فلا زيارة، فلا لقاح على \omterm{def:g3:flowers-fruits-seeds:parts}{المدقة}؛ ولا لقاح، فلا أنبوب، فلا
\omterm{def:g4:animal-reproduction:sexual}{إخصاب}؛ \omterm{def:g3:flowers-fruits-seeds:parts}{والزهرة} غير المخصَّبة تذبل كلها --- فلا \omterm{def:g3:flowers-fruits-seeds:parts}{مدقة} تنتفخ ولا
كرزة. وكانت الخطوة الناقصة غير مرئية، رحلة \omterm{def:g5:first-look-at-cells:cell}{خلية} في أنبوب --- أما
\omterm{def:g1:plants-around-us:tree}{الغصن} الخالي في تموز فمرئي بما يكفي.
\end{example}

\section{بعد اللقاء: البذرة والثمرة}

\begin{proposition}[إلام يصير كل جزء]\label{prop:g5:flower-to-fruit:becomes}
بعد \omterm{def:g5:flower-to-fruit:steps}{الإخصاب} يجري \omterm{def:g2:animals-grow-up:metamorphosis}{التحول} في
\cref{prop:g3:flowers-fruits-seeds:transform} بدقة جديدة:
\begin{itemize}
  \item كل \emph{بييضة مخصَّبة} تصير \emph{\omterm{def:g2:from-seed-to-plant:seed}{بذرة}} --- \omterm{def:g1:plants-around-us:plant}{والنبتة}
        الصغيرة في داخلها
        (\cref{def:g2:from-seed-to-plant:seed}) تنمو من \omterm{def:g4:animal-reproduction:sexual}{البويضة}
        المخصَّبة، تمامًا كما ينمو \omterm{def:g1:animals-around-us:animal}{الحيوان} من بويضته
        (\cref{ex:g5:first-look-at-cells:one})؛
  \item و\emph{\omterm{def:g3:flowers-fruits-seeds:parts}{المدقة}} حولها تنتفخ فتصير \emph{\omterm{prop:g3:flowers-fruits-seeds:transform}{الثمرة}}؛
  \item \omterm{def:g3:flowers-fruits-seeds:parts}{والبتلات} \omterm{def:g3:flowers-fruits-seeds:parts}{والأسدية} تذبل وتسقط، وقد انتهى عملها.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[عدّ البذور]\label{ex:g5:flower-to-fruit:pips}
اقطع تفاحة عرضًا: خمس حجيرات صغيرة، في كل واحدة \omterm{def:g2:from-seed-to-plant:seed}{بذور} قليلة. وكل
\omterm{def:g2:from-seed-to-plant:seed}{بذرة} بييضة واحدة بلغها أنبوب لقاح واحد. والتفاحة المائلة ذات
الحجيرات الفارغة في جانب تحكي عن \omterm{def:g3:flowers-fruits-seeds:parts}{زهرة} رديئة \omterm{def:g5:flower-to-fruit:steps}{التأبير} --- فقد وقعت
حبات أقل من أن تخصّب \omterm{prop:g5:flower-to-fruit:roles}{البييضات} كلها، فانتفخت \omterm{prop:g3:flowers-fruits-seeds:transform}{الثمرة} انتفاخًا
متفاوتًا.
\end{example}

\begin{example}[لقاح تحمله الريح]\label{ex:g5:flower-to-fruit:wind}
الأعشاب والقمح وأشجار البندق لا \omterm{def:g3:flowers-fruits-seeds:parts}{بتلات} لها ولا رائحة ولا رحيق ---
فلا شيء تعرضه على الزائرين. ولقاحها يركب \emph{الريح}: يُصنع
بكميات هائلة، في نعومة الغبار، ويُقذف في الهواء ليقع حيث يشاء
الحظ. والخطة تبادل الدقة بالكمية --- وهي منطق سحابة \omterm{def:g4:plants-without-seeds:spore}{الأبواغ} في
\cref{rem:g4:plants-without-seeds:sporeseed}، ملعوبًا باللقاح.
(وذلك اللقاح المحمول في الهواء، عرضًا، هو الذي يدغدغ أنوف حمّى
القشّ في حزيران.)
\end{example}

\begin{method}[تشخيص نبتة لا تثمر]\label{met:g5:flower-to-fruit:diagnose}
\omterm{def:g1:plants-around-us:plant}{نبتة} أزهرت جيدًا لكنها لا تحمل ثمرًا. اسأل بالترتيب:
\begin{enumerate}
  \item أبلغها المؤبّرون؟ (أهي في الداخل، أم تحت شبكة، أم لا
        حشرات حولها، أم كانت أسابيع الإزهار باردة ممطرة؟)
  \item فإن لم يبلغوها، فأبّرها بيدك: مسّ بفرشاة ناعمة \omterm{def:g3:flowers-fruits-seeds:parts}{أسدية} \omterm{def:g3:flowers-fruits-seeds:parts}{زهرة}
        ثم \omterm{def:g3:flowers-fruits-seeds:parts}{مدقة} أخرى؛
  \item أقتل الصقيع أو الأذى الأزهار قبل أن تُخصَّب \omterm{prop:g5:flower-to-fruit:roles}{البييضات}
        (\cref{exo:g3:flowers-fruits-seeds:9})؟
  \item فإن عقدت \omterm{prop:g3:flowers-fruits-seeds:transform}{الثمرة} الآن، كانت آلة \omterm{def:g1:plants-around-us:plant}{النبتة} سليمة --- ولم يخفق
        إلا رحلة الخطوة الأولى.
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:1}
أي جزء من \omterm{def:g3:flowers-fruits-seeds:parts}{الزهرة} ذكري، وماذا يصنع؟ وأيها أنثوي، وما الذي يقع في
داخله؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:2}
عرّف \omterm{def:g5:flower-to-fruit:steps}{التأبير} \omterm{def:g5:flower-to-fruit:steps}{والإخصاب}، واذكر أيهما يأتي أولًا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:3}
احكِ رحلة حبة لقاح واحدة، من \omterm{def:g3:flowers-fruits-seeds:parts}{السداة} إلى \omterm{def:g4:animal-reproduction:sexual}{البويضة}، في أربع خطوات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:4}
بعد \omterm{def:g5:flower-to-fruit:steps}{الإخصاب}: إلام تصير كل بييضة مخصَّبة؟ وإلام تصير \omterm{def:g3:flowers-fruits-seeds:parts}{المدقة}؟ وماذا
يحدث \omterm{def:g3:flowers-fruits-seeds:parts}{للبتلات}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:5}
ما كل \omterm{def:g2:from-seed-to-plant:seed}{بذرة} تفاح بمعجم هذا الفصل؟ وعلام تدل تفاحة مائلة ذات حجيرات
فارغة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:6}
كيف تُؤبَّر الأعشاب والبندق بلا \omterm{def:g3:flowers-fruits-seeds:parts}{بتلات} ولا رائحة ولا رحيق؟ وما ثمن
طريقتها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:7}
لماذا لا تصنع \omterm{def:g3:flowers-fruits-seeds:parts}{الزهرة} التي لم تزرها حشرة ثمرًا في العادة؟ سمِّ
الخطوة الناقصة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:8}
جرّب (مع \omterm{def:g1:plants-around-us:plant}{نبتة} مزهرة في شرفة أو حافة نافذة): أبّر زهرتين بيدك
بفرشاة ناعمة، كما في \cref{met:g5:flower-to-fruit:diagnose}،
وعلّمهما بخيط. أتعقد الزهرتان المعلَّمتان ثمرًا أكثر من غيرهما؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:flower-to-fruit:9}
عدّد أوجه التوازي بين تكاثر \omterm{def:g1:plants-around-us:plant}{النبات} وتكاثر \omterm{def:g1:animals-around-us:animal}{الحيوان}: \omterm{def:g5:first-look-at-cells:cell}{الخلايا}
الذكرية، \omterm{def:g4:animal-reproduction:sexual}{والبويضات}، \omterm{def:g5:flower-to-fruit:steps}{والإخصاب}، وأول \omterm{def:g5:first-look-at-cells:cell}{خلية} للفرد الجديد. وأين يقع كل
بند منها في حالة \omterm{def:g3:flowers-fruits-seeds:parts}{الزهرة}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:flower-to-fruit:10}
كان ربيع مزارع كرز باردًا ممطرًا: فلم يكد النحل يطير أسبوعين، ولم
يأتِ صقيع، وأزهرت الأشجار إزهارًا رائعًا. تنبأ بحصاد تموز واشرحه
وفق \cref{def:g5:flower-to-fruit:steps}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:flower-to-fruit:11}
يضع مزارع طماطم في بيت زجاجي \omterm{def:g5:first-look-at-cells:cell}{خلية} نحل طنّان في الداخل، مع أنه لا
يمكن بلوغ أي \omterm{def:g3:flowers-fruits-seeds:parts}{زهرة} خارجه من البيت المغلق. استعمل هذا الفصل لتشرح
بالضبط أي خدمة يبيعها النحل الطنّان --- ولماذا تستطيع مراوح تنفخ
الهواء على \omterm{def:g1:plants-around-us:plant}{النباتات} أن تحلّ محله جزئيًّا في الطماطم (التي ينفرط
لقاحها بالهزّ)، بينما لا تستطيع ذلك في أشجار الكرز.
\end{exercise}
