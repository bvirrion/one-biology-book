\chapter{Pola Tubuh dan Nenek Moyang Bersama}\label{ch:g10:common-ancestry}

Julurkan lenganmu dan hitunglah: satu tulang dari bahu ke siku, dua
dari siku ke pergelangan, sekumpulan tulang kecil di pergelangan, lima
jari. Sekarang tengoklah rangka sayap kelelawar, sirip renang seekor
paus, tungkai depan kuda, lengan katak: satu tulang, dua tulang, kumpulan,
dan jari yang sama, yang terentang, berpadu atau memendek tetapi tidak
pernah tersusun ulang. Tidak ada perancang sayap dari nol yang akan
memulainya dari sebuah lengan. Keserupaan itu adalah tanda sesuatu yang
diwariskan, dan bab ini membahas cara membaca tanda semacam itu ---
pada rangka, pada embrio, pada molekul --- untuk menyusun kembali
kekerabatan makhluk hidup.

\section{Pola tubuh vertebrata}

\begin{definition}[Pola tubuh]\label{def:g10:common-ancestry:bodyplan}
Yang disebut \emph{pola tubuh}\index{pola tubuh} sekelompok organisme
adalah susunan bersama bagian-bagiannya: sumbu simetrinya dan kedudukan
nisbi organ utamanya, terlepas dari ukuran, bentuk atau cara hidupnya.
\emph{Vertebrata}\index{vertebrata} --- ikan, amfibi, reptil, burung,
mamalia --- berbagi satu pola: tubuh yang simetris terhadap sebuah
bidang tegak, dengan ujung depan yang membawa kepala dan ujung
belakang; rangka dalam yang sumbunya berupa lajur ruas tulang belakang
yang melindungi tali saraf punggung; saluran pencernaan yang
membentang di sisi perut, mulut di depan, dan jantung di bawahnya.
\end{definition}

\begin{example}[Pola sama, hewan berbeda]\label{ex:g10:common-ancestry:sameplan}
Ikan trout, katak, kadal, merpati dan mencit yang dibedah
berdampingan memperlihatkan polanya pada tiap hal: sumsum tulang
belakang di punggung, yang dilindungi ruas tulang belakang; saluran
pencernaan di bawahnya dengan hati di samping lambung; jantung di
depan, di bawah salurannya; tungkai (atau sirip) berpasangan yang
melekat pada gelang tulang. Bedanya --- insang atau paru-paru, sirip
atau kaki, sisik, bulu atau rambut --- semuanya berupa keragaman yang
dibangun pada susunan yang sama.
\end{example}

\section{Organ homolog}

\begin{definition}[Homologi]\label{def:g10:common-ancestry:homology}
Dua organ dari dua \omterm{def:g10:biodiversity-scales:species}{spesies} bersifat \emph{homolog}\index{homologi}
apabila keduanya punya struktur yang sama dan kedudukan yang sama dalam
\omterm{def:g10:common-ancestry:bodyplan}{pola tubuhnya} --- dibangun dari bagian yang sama dalam hubungan yang
sama --- apa pun fungsinya sekarang. Tungkai depan tetrapoda menjadi
contoh bakunya: satu tulang (humerus), dua tulang (radius dan ulna),
sekumpulan tulang pergelangan, lalu jarinya. Organ yang menjalankan
fungsi yang sama dengan struktur yang berbeda --- sayap burung dan
sayap lalat --- hanyalah \emph{analog}, dan tidak menyatakan apa pun
tentang kekerabatan.
\end{definition}

\begin{proposition}[Homologi menyingkapkan kekerabatan]\label{prop:g10:common-ancestry:kinship}
Struktur homolog yang dibagi bersama diwarisi dari nenek moyang bersama
yang memilikinya; makin banyak struktur semacam itu yang dibagi dua
\omterm{def:g10:biodiversity-scales:species}{spesies}, makin dekat kekerabatannya. Tungkai tetrapoda ditemukan pada
setiap \omterm{def:g10:common-ancestry:bodyplan}{vertebrata} darat karena semuanya berasal dari satu garis
keturunan moyang yang memilikinya; sayap kelelawar adalah lengan yang
diubah karena kelelawar berasal dari mamalia berlengan, bukan dari burung.
\end{proposition}
\begin{proof}[Bukti percobaan]
Tiga garis bukti yang saling bebas bertemu. \emph{Fosil}: urutan batuan
yang bertarikh memuat bentuk dengan gabungan ciri yang antara --- ikan
dengan sirip menyerupai tungkai dan sebuah leher 375 juta tahun lalu,
lalu hewan berkaki empat dengan ekor menyerupai ikan;
\emph{Archaeopteryx}, yang berumur 150 juta tahun, dengan bulu dan
sayap tetapi juga gigi, jari bercakar dan ekor tulang yang panjang.
\emph{Embrio}: seluruh embrio \omterm{def:g10:common-ancestry:bodyplan}{vertebrata} melewati tahap dengan kantong
insang, sebuah ekor dan sebuah notokorda --- struktur yang tetap ada
pada ikan dewasa, yang diolah ulang atau hilang pada yang lainnya.
\emph{Molekul}: sebagaimana ditunjukkan \cref{ch:g10:universal-dna},
semuanya berbagi \omterm{def:g10:universal-dna:information}{DNA} dan \omterm{def:g10:universal-dna:gene}{gen} yang sama, dan beda antara versi satu \omterm{def:g10:universal-dna:gene}{gen}
mengurutkan \omterm{def:g10:biodiversity-scales:species}{spesiesnya} persis seperti anatominya.
\end{proof}

\section{Kelompok yang bersarang}

\begin{method}[Mengelompokkan menurut ciri bersama]\label{met:g10:common-ancestry:grouping}
Kelompok kekerabatan dibangun dari ciri \emph{turunan} yang dibagi
bersama --- yaitu ciri yang muncul sekali pada seorang moyang dan
diwarisi seluruh keturunannya.
\begin{enumerate}
  \item Daftarkan \omterm{def:g10:biodiversity-scales:species}{spesiesnya} dan seperangkat ciri (lajur tulang
        belakang; empat tungkai; telur bercangkang atau padanannya,
        yaitu telur amniotik; bulu; rambut dan susu; dan seterusnya).
  \item Untuk tiap ciri, catat \omterm{def:g10:biodiversity-scales:species}{spesies} mana yang memilikinya.
  \item Ciri yang dibagi sebuah himpunan bagian menetapkan sebuah
        kelompok: seluruh \omterm{def:g10:biodiversity-scales:species}{spesies} berempat tungkai adalah tetrapoda; di
        antaranya, seluruh yang bertelur amniotik adalah amniota; di
        antara mereka, seluruh yang berambut adalah mamalia.
  \item Gambarkan kelompoknya sebagai kotak yang bersarang, atau
        sebagai pohon yang tiap simpulnya adalah moyang bersama tempat
        cirinya muncul. Sebuah kelompok harus memuat \emph{seluruh}
        keturunan moyang itu: ``ikan'' atau ``reptil'' dalam arti
        sehari-hari bukanlah kelompok semacam itu.
\end{enumerate}
\end{method}

\begin{example}[Membaca kotaknya]\label{ex:g10:common-ancestry:reading}
Mana kerabat terdekat mencit di antara ikan trout, katak, kadal dan
merpati? Kotak terkecil yang memuat mencit dan \omterm{def:g10:biodiversity-scales:species}{spesies} lain adalah
``amniota'', yang memuat kadal dan merpati: keduanya sama dekat,
sama-sama lebih dekat ke mencit daripada katak, dan katak lebih dekat
daripada ikan trout. Bahwa kadal lebih dekat ke merpati daripada ke
katak mengejutkan sebagian besar orang; telur bercangkang, sebuah ciri
yang tidak dimiliki katak mana pun, memutuskannya.
\end{example}

\section{Kekerabatan dalam molekulnya}

\begin{proposition}[Kekerabatan molekuler]\label{prop:g10:common-ancestry:molecular}
Karena seluruh makhluk hidup berasal dari nenek moyang bersama, sebuah
\omterm{def:g10:universal-dna:gene}{gen} yang hadir pada dua \omterm{def:g10:biodiversity-scales:species}{spesies} adalah salinan \omterm{def:g10:universal-dna:gene}{gen} moyang yang telah
diubah, dan jumlah beda antara kedua salinan itu tumbuh seiring waktu
sejak \omterm{def:g10:biodiversity-scales:species}{spesiesnya} berpisah. Karena itu membandingkan urutan satu \omterm{def:g10:universal-dna:gene}{gen},
atau urutan \omterm{def:g10:chemistry-of-life:families}{protein} yang disandikannya, lintas \omterm{def:g10:biodiversity-scales:species}{spesies} mengukur
kekerabatan, dan pengelompokan yang dihasilkannya sesuai dengan
pengelompokan yang ditarik dari anatominya.
\end{proposition}
\admitted

\begin{example}[Sitokrom c]\label{ex:g10:common-ancestry:cytochrome}
Sitokrom c adalah \omterm{def:g10:chemistry-of-life:families}{protein} berisi sekitar 104 \omterm{def:g10:chemistry-of-life:families}{asam amino} yang dipakai
dalam respirasi oleh setiap eukariot. Dibandingkan dengan urutan
manusia, urutan simpanse berbeda pada 0 kedudukan, urutan mencit pada
9, ayam pada 13, katak pada 18, tuna pada 21, dan ragi pada 45. Jika
diurutkan menurut bedanya, \omterm{def:g10:biodiversity-scales:species}{spesiesnya} jatuh persis ke dalam kotak
gambar sebelumnya: mamalia, lalu amniota, lalu tetrapoda, lalu
\omterm{def:g10:common-ancestry:bodyplan}{vertebrata}, lalu eukariot.
\end{example}

\begin{remark}[Kesatuan dan kekerabatan]\label{rem:g10:common-ancestry:unity}
Kesatuan komposisi (\cref{ch:g10:chemistry-of-life}), kesatuan \omterm{def:g10:cells-common-unit:cell}{sel}
(\cref{ch:g10:cells-common-unit}), kesatuan molekul genetik
(\cref{ch:g10:universal-dna}) dan kini kesatuan \omterm{def:g10:common-ancestry:bodyplan}{pola tubuh} serta urutan
\omterm{def:g10:universal-dna:gene}{gen} adalah satu kenyataan yang terlihat empat kali: makhluk hidup
saling menyerupai karena mereka berkerabat, dan mereka berkerabat
karena mereka berasal dari nenek moyang bersama. Keragaman pada
\cref{ch:g10:biodiversity-scales} adalah hasil pewarisan dengan
perubahan itu, sepanjang waktu yang diukur rekaman fosil. Bagaimana
perubahannya muncul dan menyebar menjadi pokok tahun terakhir.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:common-ancestry:1}
Sebutkan empat ciri \omterm{def:g10:common-ancestry:bodyplan}{pola tubuh} \omterm{def:g10:common-ancestry:bodyplan}{vertebrata}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:common-ancestry:2}
Rumuskan organ \omterm{def:g10:common-ancestry:homology}{homolog} dan berikan contoh bakunya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:common-ancestry:3}
Apakah sayap burung dan sayap kupu-kupu \omterm{def:g10:common-ancestry:homology}{homolog} atau analog? Berikan
alasannya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:common-ancestry:4}
Dengan gambar kotak bersarangnya, sebutkan mana yang lebih dekat ke
katak: ikan trout atau kadal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:common-ancestry:5}
Tiga garis bukti mana yang menyokong gagasan bahwa \omterm{def:g10:common-ancestry:homology}{homologi} bersama
berasal dari nenek moyang bersama?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:common-ancestry:6}
Seekor paus punya sirip renang berpola tulang tetrapoda, bernapas
dengan udara, menyusui anaknya, dan punya tulang panggul mungil yang
tidak melekat pada kaki mana pun. Termasuk kotak mana pada gambarnya?
Apa yang disarankan tulang panggul itu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:common-ancestry:7}
Tambahkan buaya (empat tungkai, telur amniotik, sisik, tanpa rambut)
dan salamander (empat tungkai, telur yang ditaruh dalam air tanpa
cangkang) ke dalam kotak bersarangnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:common-ancestry:8}
Jelaskan mengapa ``ikan'' (trout, hiu, ikan paru\dots) bukan kelompok
kekerabatan dalam arti \cref{met:g10:common-ancestry:grouping}, jika
diketahui bahwa ikan paru lebih dekat ke katak daripada ke trout.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:common-ancestry:9}
Dengan data sitokrom c, urutkan kerabat manusia dari yang terdekat ke
yang terjauh dan periksa urutannya terhadap kotak anatominya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:common-ancestry:10}
Embrio manusia berumur empat pekan punya kantong insang di lehernya dan
sebuah ekor, yang keduanya lenyap kemudian. Apa sumbangan pengamatan
ini bagi bahasan bab ini?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:common-ancestry:11}
\emph{Archaeopteryx} punya bulu dan ekor tulang yang panjang. Jelaskan
mengapa fosil dengan campuran ciri memang diharapkan menurut nenek
moyang bersama dan akan membingungkan tanpa gagasan itu.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:common-ancestry:12}
Mata gurita dan mata manusia sama-sama berupa kamera berlensa dan
berretina, namun keduanya berkembang dari jaringan yang berbeda dan
retina gurita terpasang secara terbalik. \omterm{def:g10:common-ancestry:homology}{Homolog} atau analog? Apa makna
tiap jawabannya bagi moyang bersama keduanya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:common-ancestry:13}
Sitokrom c berbeda antara manusia dan mencit pada 9 kedudukan, antara
manusia dan ayam pada 13, dan antara mencit dan ayam pada 13 juga.
Jelaskan mengapa kedua bilangan terakhir memang diharapkan sama (atau
hampir sama) jika bedanya menumpuk mantap seiring waktu.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:common-ancestry:14}
Ular tidak punya tungkai, namun ia ditempatkan di antara tetrapoda.
Atas bukti apa, dan apa lalu arti ``tetrapoda''?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:common-ancestry:15}
Seorang teman berpendapat bahwa hewan yang mirip sekadar hidup dengan
cara yang mirip, dan itulah sebabnya mereka tampak serupa. Pakailah
tungkai depan, embrio dan molekulnya untuk menjelaskan apa yang tidak
dapat dijelaskan pendapat itu.
\end{exercise}

\section{Soal: Satu Pohon, Dua Macam Bukti}

\begin{problem}[{Soal akhir pekan --- lima vertebrata, rangkanya dan
salah satu proteinnya: kekerabatan yang dibangun dari anatomi, dibangun
ulang dari molekul, dan keduanya dibandingkan}]
\label{pb:g10:common-ancestry:1}
Kelasnya mempelajari lima \omterm{def:g10:biodiversity-scales:species}{spesies}: ikan trout, katak, kadal, merpati
dan mencit. Tabel cirinya (1 = ada, 0 = tidak ada):

\begin{center}
\begin{tabular}{lccccc}
 & trout & katak & kadal & merpati & mencit \\ \hline
lajur tulang belakang & 1 & 1 & 1 & 1 & 1 \\
empat tungkai & 0 & 1 & 1 & 1 & 1 \\
telur amniotik & 0 & 0 & 1 & 1 & 1 \\
bulu & 0 & 0 & 0 & 1 & 0 \\
rambut dan susu & 0 & 0 & 0 & 0 & 1 \\
\end{tabular}
\end{center}

Jumlah beda pada sitokrom c (dibulatkan):

\begin{center}
\begin{tabular}{lcccc}
 & katak & kadal & merpati & mencit \\ \hline
trout & 19 & 20 & 21 & 21 \\
katak & & 16 & 17 & 18 \\
kadal & & & 12 & 14 \\
merpati & & & & 13 \\
\end{tabular}
\end{center}

\medskip
\noindent\textbf{Bagian I --- Rangkanya.}
\begin{enumerate}
  \item Ciri mana yang dibagi kelimanya? Kelompok mana yang
        ditetapkannya?
  \item Ciri mana yang menetapkan kelompok berisi empat, dan kelompok
        berisi tiga? Sebutkan nama kelompoknya.
  \item Bulu dan rambut masing-masing hanya ditemukan pada satu
        \omterm{def:g10:biodiversity-scales:species}{spesies}. Dapatkah ciri semacam itu menetapkan kelompok
        kekerabatan di antara kelimanya? Untuk apa ciri itu berguna?
  \item Gambarkan kotak bersarang bagi kelima \omterm{def:g10:biodiversity-scales:species}{spesiesnya}, atau pohon
        yang sepadan.
  \item Tungkai depan katak punya pola humerus, radius--ulna,
        pergelangan dan jari; sirip dada trout punya jari-jari tulang
        yang mengipas dari sebuah alas kecil. Mana yang \omterm{def:g10:common-ancestry:homology}{homolog} dengan
        lenganmu? Ciri mana pada tabelnya yang sepadan dengan ini?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Molekulnya.}
\begin{enumerate}[resume]
  \item Dari tabel kedua, dua \omterm{def:g10:biodiversity-scales:species}{spesies} mana yang menjadi pasangan
        terdekat? \omterm{def:g10:biodiversity-scales:species}{Spesies} mana yang terjauh dari semua yang lain?
  \item Urutkan kerabat mencit dari yang terdekat ke yang terjauh
        dengan memakai molekulnya saja.
  \item Lakukan hal yang sama untuk kataknya. Apakah urutannya taat
        asas dengan kotak Bagian I?
  \item Jarak kadal--merpati (12) lebih kecil daripada jarak
        kadal--mencit (14). Apa yang disarankannya tentang urutan
        percabangan di dalam amniota, dan apakah tabel cirinya
        menyatakan sesuatu tentang itu?
  \item Jelaskan mengapa jarak trout ke keempat lainnya semuanya
        kira-kira sama (19--21).
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Membandingkan keduanya.}
\begin{enumerate}[resume]
  \item Sebutkan pengelompokan yang disepakati kedua macam buktinya.
  \item Mengapa kesepakatan antara urutan anatomi dan urutan molekuler
        menjadi bukti yang kuat, padahal masing-masing sendirian dapat
        dipertanyakan?
  \item Andaikan \omterm{def:g10:biodiversity-scales:species}{spesies} keenam, seekor kelelawar, ditambahkan: tungkai
        depan berpola tetrapoda, telur amniotik, rambut dan susu,
        sitokrom c yang berbeda dari mencit pada 8 kedudukan dan dari
        merpati pada 13. Ia masuk ke mana? Mengapa sayapnya tidak
        menempatkannya bersama merpati?
  \item Sitokrom c simpanse serupa dengan sitokrom c manusia, namun
        kedua \omterm{def:g10:biodiversity-scales:species}{spesiesnya} berbeda secara kasatmata. Apa yang
        dinyatakannya tentang \omterm{def:g10:universal-dna:gene}{gen} mana yang dapat dan tidak dapat
        diterangkan perbandingan sitokrom c kepada kita?
  \item Sitokrom c ragi berbeda dari kelimanya pada sekitar 45
        kedudukan. Kotak apa, yang lebih besar daripada ``\omterm{def:g10:common-ancestry:bodyplan}{vertebrata}'',
        yang dibagi ragi itu bersama mereka?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Waktu dan pohonnya.}
Rekaman fosil menarikhkan perpisahan garis keturunan mencit dan merpati
sekitar 320 juta tahun lalu.
\begin{enumerate}[resume]
  \item Jika bedanya menumpuk mantap pada kedua garis keturunan, berapa
        beda yang muncul tiap 100 juta tahun sepanjang satu garis
        keturunan? (Pakai jarak mencit--merpati, dengan mengingat bahwa
        kedua garis keturunannya sama-sama berubah.)
  \item Pakai laju itu untuk memperkirakan umur perpisahan garis
        keturunan katak dari amniota, dari jarak katak--mencit dan
        katak--merpati.
  \item Perkirakan dengan cara yang sama perpisahan garis keturunan trout.
  \item Fosil menempatkan tetrapoda pertama sekitar 370 juta tahun lalu
        dan \omterm{def:g10:common-ancestry:bodyplan}{vertebrata} berahang pertama jauh di atas 420. Bandingkan
        dengan perkiraanmu dan beri tanggapan atas kesesuaiannya.
  \item Nyatakan hasilnya dalam satu kalimat: apa yang ditegakkan kedua
        macam bukti itu bersama-sama tentang kelima \omterm{def:g10:biodiversity-scales:species}{spesiesnya}, dan apa
        yang ditambahkan molekulnya yang tidak dapat ditambahkan rangkanya.
\end{enumerate}
\end{problem}
