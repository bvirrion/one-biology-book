\chapter{نظرة في تطور الإنسان}\label{ch:g12:human-evolution}

في طبقة من الرماد البركاني عمرها ثلاثة ملايين ونصف مليون سنة، يمتد
أثران من آثار الأقدام جنبًا إلى جنب سبعة وعشرين مترًا. تركهما مخلوقان
يمشيان منتصبين، على العقب أولًا، بقدم مقوسة وإبهام في صف سائر الأصابع
--- أي قدمين كقدمينا --- في زمن لم يكن فيه دماغ أكبر من دماغ قرد.
فالمشي المنتصب جاء أولًا، بمليوني سنة؛ وجاء الدماغ والأدوات والكلام
بعده، لا في خط مستقيم بل على أجمة من \omterm{def:g10:biodiversity-scales:species}{الأنواع}، لم يخلّف معظمها نسلًا.
ويطبّق هذا الفصل أدوات الفصلين الماضيين --- الصفات المشتركة، والأشجار،
والتواريخ، \omterm{def:g10:universal-dna:information}{وDNA} --- على السلالة المؤدية إلينا.

\section{الإنسان بين الرئيسيات}

\begin{proposition}[موضعنا من الشجرة]\label{prop:g12:human-evolution:place}
الإنسان من الرئيسيات: وهي ثدييات ذوات أيد قابضة، وأظافر بدل المخالب،
وعينين متجهتين إلى الأمام، ودماغ كبير قياسًا بحجمها. وداخل الرئيسيات،
تؤلف القردة العليا --- الأورانغوتان والغوريلا والشمبانزي والبونوبو
والإنسان --- فرعًا حيويًا، وداخله يكون الشمبانزي والبونوبو المجموعة
الشقيقة للإنسان: فيختلف \omterm{def:g10:universal-dna:information}{DNA} عندنا عن \omterm{def:g10:universal-dna:information}{DNA} عندهما في نحو 1.2\% من
المواضع، وقد عاش السلف المشترك للسلالتين قبل نحو ستة ملايين إلى سبعة
ملايين سنة، في أفريقيا. ولم يكن ذلك السلف شمبانزي ولا إنسانًا؛ فكلتا
السلالتين تطورت منذئذ.
\end{proposition}
\begin{proof}[شواهد]
مصفوفات الصفات ومقارنات \omterm{def:g10:universal-dna:information}{DNA} في \cref{ch:g12:phylogenetic-trees}: فكل
\omterm{def:g10:universal-dna:gene}{مورّثة} سلسلت تضع الشمبانزي أقرب إلى الإنسان منه إلى الغوريلا، \omterm{def:g11:cell-cycle-mitosis:chromosome}{والصبغي}
2 البشري اندماج، طرفًا إلى طرف، لصبغيين يبقيان منفصلين عند الشمبانزي
--- ولهذا كان لنا 23 زوجًا ولها 24. والساعة الجزيئية، معايرةً على
تواريخ أحافير انفصالات أقدم بين الرئيسيات، تعطي عقدة الإنسان--الشمبانزي
تاريخها.
\end{proof}

\begin{definition}[السلالة البشرية]\label{def:g12:human-evolution:lineage}
\emph{السلالة البشرية}\index{سلالة بشرية} هي \omterm{prop:g12:phylogenetic-trees:rules}{الفرع الحيوي} لكل \omterm{def:g10:biodiversity-scales:species}{الأنواع}
الأقرب إلى الإنسان الحديث منها إلى الشمبانزي: أي كل ما انحدر من الجانب
البشري لعقدة الستة ملايين سنة. وتحوي نوعًا حيًا واحدًا، هو \emph{Homo
sapiens}، وعشرين نوعًا أحفوريًا أو نحوها --- الأسترالوبيثيكات \omterm{def:g10:biodiversity-scales:species}{وأنواع}
الجنس \emph{Homo} العدة. وتنتمي أحفورة إليها إن أبدت صفات مشتقة
للسلالة: المشي المنتصب، ووجهًا قصيرًا، وأنيابًا صغيرة، ثم لاحقًا دماغًا
كبيرًا وأدوات.
\end{definition}

\section{صفات السلالة}

\begin{proposition}[المشي على قدمين جاء أولًا]\label{prop:g12:human-evolution:bipedalism}
أبكر صفة مشتقة \omterm{def:g12:human-evolution:lineage}{للسلالة البشرية} هي المشي المعتاد على رجلين. وهي تظهر في
الهيكل: فالفتحة في قاعدة الجمجمة التي يمر منها النخاع الشوكي تقع تحت
الجمجمة لا وراءها، فيتوازن الرأس على عمود فقري شاقولي؛ وللعمود الفقري
انحناء مزدوج؛ والحوض قصير على هيئة طاس؛ وعظما الفخذ يميلان إلى الداخل
نحو الركبة، فيأتيان بالقدمين تحت مركز الجسم؛ وللقدم قوس وإبهام في صف
سائر الأصابع، مصنوع للدفع لا للقبض. وهذه كلها موجودة، مع سمات قليلة
شبيهة بسمات القردة، في أحافير عمرها أربعة ملايين سنة أدمغتها ليست أكبر
من دماغ شمبانزي.
\end{proposition}
\begin{proof}[شواهد]
آثار الأقدام في الرماد البركاني، وعمرها 3.6 ملايين سنة، تبدي مشية
واسعة الخطى بضرب العقب وقدم مقوسة. وهيكل أنثى أسترالوبيثيكية عمره 3.2
ملايين سنة، وهو كامل بنسبة 40\%، له حوض وركبة شبيهان بالبشريين ودماغ
نحو \qty{400}{cm^3}. وقواعد الجماجم من العمر نفسه فتحتها الشوكية إلى
الأمام، تحت الجمجمة. فالصفات تظهر بالترتيب: المشي، ثم، بعد أكثر من
مليون سنة، نمو الدماغ.
\end{proof}

\begin{proposition}[ثم الدماغ والأدوات والوجه]\label{prop:g12:human-evolution:brain}
منذ نحو 2.5 مليون سنة تبدي السلالة طقمًا ثانيًا من الصفات المشتقة:
دماغًا ينمو من \qty{450}{cm^3} إلى \qty{1350}{cm^3}، ووجهًا وفكوكًا
تصغر تحته، وذقنًا، وأدوات حجرية --- أولًا حصى مشظّى إلى حد، ثم فؤوس يد
مشكّلة، ثم نصال ورؤوس وأدوات مركّبة على مقابض. وتضبط النار قبل مليون
سنة، ويدفن الموتى قبل مئة ألف، وترسم الصور قبل أربعين ألفًا. فالدماغ
والأدوات ينموان معًا، ويجعل كل منهما الآخر نافعًا، على مدى مليوني سنة.
\end{proposition}
\admitted

\begin{example}[قراءة جمجمة]\label{ex:g12:human-evolution:skull}
جمجمة أحفورية فتحتها الشوكية من تحت، وعلبة دماغها \qty{450}{cm^3}،
وفكوكها كبيرة ووجهها بارز، هي جمجمة أسترالوبيثيكي: منتصب صغير الدماغ.
وأخرى \qty{900}{cm^3}، بحاجب جبهة ثقيل وجبين متراجع وبلا ذقن، هي
\emph{Homo erectus}. وأخرى \qty{1400}{cm^3}، بجبين عال ووجه صغير
مندسّ تحت علبة الدماغ وذقن، هي إنسان حديث. وترتيب هذه الصفات على
الشجرة هو الترتيب الذي تظهر به في الصخور.
\end{example}

\section{أجمة لا سلّم}

\begin{proposition}[أنواع عدة في وقت واحد]\label{prop:g12:human-evolution:bush}
\omterm{def:g12:human-evolution:lineage}{السلالة البشرية} ليست خطًا واحدًا من الأسلاف والنسل بل أجمة من \omterm{def:g10:biodiversity-scales:species}{الأنواع}،
عاش عدد منها في الوقت نفسه: فنوعان أو ثلاثة من الأسترالوبيثيكات قبل
مليوني سنة إلى جانب أول \emph{Homo}؛ وانتشر \emph{Homo erectus} على
ثلاث قارات بينما نشأت \omterm{def:g10:biodiversity-scales:species}{أنواع} أخرى في أفريقيا؛ وقبل خمسين ألف سنة لا
غير، كان الإنسان الحديث يتقاسم الأرض مع النياندرتاليين في أوروبا، ومع
\omterm{def:g12:selection-drift-speciation:population}{جمهرة} أخرى في آسيا، ومع \omterm{def:g10:biodiversity-scales:species}{نوع} صغير الجسم في جزيرة. وقد انتهى معظم أفرع
الأجمة إلى الانقراض؛ وفرعنا هو الباقي.
\end{proposition}
\begin{proof}[شواهد]
توجد أحافير \omterm{def:g10:biodiversity-scales:species}{أنواع} متمايزة في طبقات من العمر نفسه في المواقع نفسها.
وتتداخل بقايا النياندرتال والإنسان الحديث في أوروبا آلاف السنين. \omterm{def:g10:universal-dna:information}{وDNA}
النياندرتاليين \omterm{def:g12:selection-drift-speciation:population}{والجمهرة} الآسيوية، مستخرجًا من عظامهم، متمايز عن \omterm{def:g10:universal-dna:information}{DNA}
عندنا --- وموجود فينا: فالناس خارج أفريقيا يحملون نحو 2\% من \omterm{def:g10:universal-dna:information}{DNA}
النياندرتال، وبعض \omterm{def:g12:selection-drift-speciation:population}{جمهرات} المحيط الهادئ عدة في المئة من \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g12:selection-drift-speciation:population}{الجمهرة}
الآسيوية، وهو أثر التناسل حين التقى النوعان. فالأفرع تلامست قبل أن
تنتهي.
\end{proof}

\begin{proposition}[الإنسان الحديث]\label{prop:g12:human-evolution:sapiens}
ظهر \emph{Homo sapiens} في أفريقيا قبل نحو 300\,000 سنة، ويعرف
بجمجمته العالية المدورة ووجهه الصغير وذقنه. وغادرت جماعات صغيرة
أفريقيا قبل نحو 60\,000 سنة، وبلغت في غضون 50\,000 سنة كل قارة عدا
القارة القطبية الجنوبية، ملتقيةً \omterm{def:g12:selection-drift-speciation:population}{بالجمهرات} الأقدم التي وجدتها وممتصةً
بعضها. وكل البشر الأحياء منحدرون من تلك \omterm{def:g12:selection-drift-speciation:population}{الجمهرة} الأفريقية: \omterm{def:g10:biodiversity-scales:biodiversity}{فالتنوع
الوراثي} \omterm{def:g10:biodiversity-scales:species}{للنوع} كله أصغر من \omterm{def:g10:biodiversity-scales:biodiversity}{التنوع الوراثي} لشمبانزيات منطقة أفريقية
واحدة، وهو أعظم ما يكون في أفريقيا، ويتناقص مع البعد عنها على طول
مسارات الانتشار --- وهي بصمة جماعات مؤسِّسة متعاقبة.
\end{proposition}
\admitted

\section{ما الذي صنعنا}

\begin{proposition}[المورّثات، ثم الثقافة]\label{prop:g12:human-evolution:culture}
نشأت صفات السلالة كما تصف الفصول السابقة: \omterm{def:g11:mutations:mutation}{بالطفرات}، ومنها تغيرات في
تنظيم \omterm{prop:g12:diversification-of-life:development}{المورّثات النمائية} (نمو أطول للدماغ، ووجه أقصر)، مفروزةً
بالانتقاء في بيئات السافانا الأفريقية، وبالانسياق في \omterm{def:g12:selection-drift-speciation:population}{جمهرات} صغيرة.
لكن السلالة أضافت نمطًا ثانيًا من التوارث --- هو السلوك المنقول في
\cref{ch:g12:diversification-of-life}، مرفوعًا إلى مقياس لم يبلغه أي
\omterm{def:g10:biodiversity-scales:species}{نوع} آخر: الأدوات، والنار، واللغة، والتعاون، والتعليم. فالثقافة تتراكم
في العمر الواحد وتنتقل إلى كل من يتعلمها، لا إلى النسل وحده؛ وفي المئة
ألف سنة الأخيرة كانت التغيرات الأهم عند البشر تغيرات ثقافية، والبيئة
التي يتطور فيها \omterm{def:g10:biodiversity-scales:species}{النوع} هي في معظمها بيئة صنعها.
\end{proposition}
\admitted

\begin{method}[وضع أحفورة في موضعها]\label{met:g12:human-evolution:placing}
\begin{enumerate}
  \item أرّخ الطبقة (بالتفكك الإشعاعي للمعادن البركانية، أو بالعمر
        المعروف للطبقات).
  \item اقرأ المشية: موضع الفتحة الشوكية، وشكل الحوض وعظم الفخذ،
        والقدم.
  \item قس علبة الدماغ؛ ولاحظ الوجه وحاجب الجبهة والذقن والأسنان.
  \item اذكر الحالات المشتقة الحاضرة والغائبة، وضع الأحفورة على
        الشجرة بقواعد \cref{ch:g12:phylogenetic-trees}: أي فرعًا
        جانبيًا قرب العقدة التي يوافقها مزيج صفاتها، لا "السلف".
  \item وحيث يحفظ العظم حفظًا كافيًا، اقرأ \omterm{def:g10:universal-dna:information}{DNA} وقارنه \omterm{def:g12:selection-drift-speciation:population}{بالجمهرات}
        الحية.
\end{enumerate}
\end{method}

\begin{remark}[لا حلقة مفقودة]\label{rem:g12:human-evolution:link}
تفترض العبارة سلسلة فيها ثغرة واحدة؛ والسجل أجمة ذات أفرع كثيرة، يعرف
معظمها من عظام قليلة، و"الحلقة" بيننا وبين الشمبانزي عقدة --- أي \omterm{def:g12:selection-drift-speciation:population}{جمهرة}
منقرضة لم تكن أحدهما --- لا مخلوق في منتصف الطريق. أما الذي تبينه
الأحافير، بالترتيب الذي تحفظها به الصخور، فهو ظهور كل صفة مشتقة في
دورها: الأقدام، ثم الأدمغة، ثم الأدوات، ثم الذقن --- وفي الخمسين ألف
سنة الأخيرة، أعمال \omterm{def:g10:biodiversity-scales:species}{نوع} صار تطوره في معظمه من صنعه.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:human-evolution:1}
أي \omterm{def:g10:biodiversity-scales:species}{الأنواع} الحية أقرب أقرباء الإنسان، وقبل كم انفصلت السلالتان؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:human-evolution:2}
اذكر أربع علامات هيكلية على المشي المنتصب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:human-evolution:3}
من شكل حجم الدماغ، أعط حجم دماغ أسترالوبيثيكي، ودماغ
\emph{Homo erectus} الباكر، ودماغ إنسان حديث.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:human-evolution:4}
لماذا وصفت \omterm{def:g12:human-evolution:lineage}{السلالة البشرية} بأنها أجمة لا سلّم؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:human-evolution:5}
أين ومتى ظهر \emph{Homo sapiens}، وماذا يبين توزع \omterm{def:g10:biodiversity-scales:biodiversity}{التنوع الوراثي}
البشري؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:human-evolution:6}
\omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} 2 البشري يقابل صبغيين عند الشمبانزي ملتحمين طرفًا إلى طرف. فسّر
كيف يفسر ذلك اختلاف عددي \omterm{def:g10:universal-dna:information}{الصبغيات}، ولماذا كان شاهدًا على القرابة لا
عليها.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:human-evolution:7}
أيهما جاء أولًا في السلالة، المشي المنتصب أم الدماغ الكبير؟ وبكم؟ اذكر
الشاهد.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:human-evolution:8}
من شكل الخط الزمني، أي \omterm{def:g10:biodiversity-scales:species}{الأنواع} تعايشت قبل مليوني سنة؟ وأيها قبل
100\,000 سنة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:human-evolution:9}
شخص من أصل أوروبي يحمل 2\% من \omterm{def:g10:universal-dna:information}{DNA} النياندرتال. فسّر أي حدث يسجل ذلك،
ولماذا لا يكاد الأفارقة جنوب الصحراء يحملون منه شيئًا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:human-evolution:10}
طبّق \cref{met:g12:human-evolution:placing} على جمجمة فتحتها الشوكية
إلى الأمام، وعلبة دماغها \qty{500}{cm^3}، وفكوكها كبيرة، وجدت في طبقة
عمرها 2.8 مليون سنة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:human-evolution:11}
فسّر لماذا تكون عبارة "الإنسان منحدر من القردة" صائبةً إذا قيلت
"الإنسان من القردة"، بمفردات الأفرع الحيوية.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:human-evolution:12}
كانت أدمغة النياندرتاليين أكبر قليلًا من أدمغتنا وصنعوا أدوات محكمة،
ومع ذلك اختفوا في غضون بضعة آلاف من السنين من وصولنا إلى أوروبا. اقترح
فرضيتين توافقان الفصل وقل أي شاهد كان يميز بينهما.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:human-evolution:13}
\omterm{def:g10:biodiversity-scales:biodiversity}{التنوع الوراثي} البشري يتناقص مع البعد عن أفريقيا على طول مسارات
الانتشار. فسّر ذلك بأثر المؤسس في
\cref{ch:g12:selection-drift-speciation}، وقل ماذا يستلزم في كيفية
وقوع الانتشار.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:human-evolution:14}
دوام اللاكتاز، وفتح البشرة في خطوط العرض العالية، ومقاومة الملاريا،
صفات بشرية تطورت في الألوف العشرة الأخيرة من السنين. فسّر كيف يستطيع
تغير ثقافي (تربية الحلوب، والهجرة، والزراعة) أن ينشئ الانتقاء الذي
يغير \omterm{def:g10:universal-dna:gene}{المورّثات}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:human-evolution:15}
"توقف التطور عند البشر، إذ يحميهم الطب والثقافة من الانتقاء." ناقش في
فقرة، بالانسياق، وبصفات التمرين السابق الثلاث، وبمعنى الانتقاء.
\end{exercise}

\section{مسألة: الماشيان في الرماد}

\begin{problem}[{مسألة نهاية الأسبوع --- أثران من آثار الأقدام يقرآن
للمشية والقامة والسرعة؛ وصف من الجماجم يقاس؛ وDNA طالبة حية يتتبع إلى
ثلاث جمهرات}]
\label{pb:g12:human-evolution:1}
أثران من آثار الأقدام، عمرهما 3.6 ملايين سنة، يمتدان متوازيين
\qty{27}{m}. الأثر A: طول أثر القدم \qty{21.5}{cm}، والخطوة (من أثر
إلى الأثر التالي للقدم نفسها) \qty{0.95}{m}. والأثر B: الطول
\qty{18.5}{cm}، والخطوة \qty{0.80}{m}. وعند الإنسان الحديث تبلغ القدم
نحو 15\% من طول القامة واقفًا، وخطوة المشي التي تقارب 0.8 من القامة
توافق مشيًا بطيئًا.

\medskip
\noindent\textbf{القسم الأول --- آثار الأقدام.}
\begin{enumerate}
  \item قدّر قامة الماشي A وقامة الماشي B.
  \item قارن كل خطوة بحاصل ضرب القامة في 0.8. أكانا يمشيان أم يعدوان؟
  \item تبدي الآثار عقبًا عميقًا، وقوسًا، وإبهامًا موازيًا لسائر
        الأصابع. ماذا تبين كل سمة عن القدم وعن المشية؟
  \item الشمبانزي إذا مشى منتصبًا ترك آثارًا بأخمص مسطح وإبهام متباعد،
        ولا يستطيع أن يديم ذلك طويلًا. فأي صفات
        \cref{prop:g12:human-evolution:bipedalism} تثبتها الآثار
        لصاحبيها، وأيها تتركها مجهولة؟
  \item أقرب الأحافير من العمر نفسه أدمغتها \qty{400}{cm^3}. فماذا
        يحسم الاجتماع --- هذه الأقدام، وذلك الدماغ --- في ترتيب صفات
        السلالة؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثاني --- الجماجم.}
خمس جماجم، بحجم علبة الدماغ والعمر والسمات:
1: \qty{450}{cm^3}، و3.0 ملايين سنة، ووجه بارز، وبلا ذقن.
2: \qty{650}{cm^3}، و1.9 مليون سنة، ووجه أصغر.
3: \qty{950}{cm^3}، ومليون سنة، وحاجب ثقيل، وبلا ذقن.
4: \qty{1450}{cm^3}، و60\,000 سنة، وحاجب ثقيل، وبلا ذقن، وجمجمة طويلة
منخفضة.
5: \qty{1350}{cm^3}، و30\,000 سنة، وجبين عال، وذقن.
\begin{enumerate}[resume]
  \item انسب كل جمجمة إلى مجموعة من مجموعات شكل الخط الزمني.
  \item ارسم، أو صف، حجم الدماغ بدلالة العمر عند الخمس. في أي مجال
        ينمو الحجم أسرع ما يكون؟
  \item الجمجمة 4 أكبر من الجمجمة 5. أيعرّف حجم الدماغ وحده الإنسان
        الحديث؟ فأي الصفات تعرّفه؟
  \item الجمجمتان 4 و5 تتداخلان في الزمن. فماذا يقول الفصل عن العلاقة
        بين جمهرتيهما، وأي شاهد يسنده؟
  \item الجمجمة 1 وآثار الأقدام في القسم الأول قد تعودان إلى \omterm{def:g10:biodiversity-scales:species}{النوع}
        نفسه. فسّر لماذا كان ذلك \omterm{def:g10:biodiversity-scales:species}{النوع} \emph{قريبًا} شبيهًا بالسلف لا
        سلفًا مبرهنًا عليه، بلغة \cref{ch:g12:phylogenetic-trees}.
\end{enumerate}

\medskip
\noindent\textbf{القسم الثالث --- مجين الطالبة.}
طالبة من أصل أوروبي وميلانيزي قورن مجينها بمجينات النياندرتاليين
\omterm{def:g12:selection-drift-speciation:population}{والجمهرة} الآسيوية المنقرضة: فطابق 1.6\% مجين النياندرتال، و1.8\% مجين
\omterm{def:g12:selection-drift-speciation:population}{الجمهرة} الآسيوية.
\begin{enumerate}[resume]
  \item فسّر ما تعنيه مطابقة 1.6\% وأي حدث تسجله.
  \item قارن أرقامها بالمخطط. أي جمهراتها السلفية أسهمت بكل حصة؟
  \item إذا وقع التناسل قبل 50\,000 سنة وكان الجيل 25 سنة، فكم جيلًا
        يفصلها عن الحدث؟ ولماذا لم تخفّ الحصة حتى تنعدم؟
  \item أسهمت سلالتان \omterm{def:g10:universal-dna:gene}{بمورّثات} في مورّثاتها مع أنهما تسميان نوعين.
        فماذا يقول ذلك عن حد \omterm{def:g10:biodiversity-scales:species}{النوع} في
        \cref{ch:g12:selection-drift-speciation} في ذلك الزمن؟
  \item \omterm{def:g10:universal-dna:information}{DNA} متقدرتها من نمط الإنسان الحديث، وكذلك هو عند كل شخص حي.
        فماذا يبين ذلك عن اتجاه التناسل أو مصيره؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الرابع --- الساعة والأجمة.}
\begin{enumerate}[resume]
  \item يختلف \omterm{def:g10:universal-dna:information}{DNA} الإنسان والشمبانزي بنسبة 1.2\%؛ وساعة الرئيسيات في
        \cref{ch:g12:phylogenetic-trees} تمشي بنسبة 0.27\% لكل مليون
        سنة. أرّخ السلف المشترك.
  \item أقدم أحافير \omterm{def:g12:human-evolution:lineage}{السلالة البشرية} المعروفة عمرها 6 إلى 7 ملايين
        سنة. أذلك متوافق مع تاريخك؟
  \item اذكر \omterm{def:g10:biodiversity-scales:species}{أنواع} الخط الزمني التي كانت حية قبل مليون سنة، واذكر كم
        منها له نسل حي.
  \item فسّر لماذا كانت المئة ألف سنة الأخيرة من السلالة يوصفها التغير
        الثقافي خيرًا مما توصفها صفات هذا الفصل.
  \item اذكر النتيجة: قامة الماشي A ومشيته، والترتيب الذي ظهرت به
        الأقدام والدماغ والذقن، وعدد \omterm{def:g12:selection-drift-speciation:population}{الجمهرات} المنقرضة التي تحمل
        الطالبة مورّثاتها.
\end{enumerate}
\end{problem}
