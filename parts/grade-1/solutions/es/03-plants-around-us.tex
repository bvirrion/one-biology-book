\section*{Capítulo \ref{ch:g1:plants-around-us} --- Las plantas que nos rodean}

\begin{solution}{exo:g1:plants-around-us:1}
\omterm{def:g1:plants-around-us:parts}{Raíces} (bajo la tierra), \omterm{def:g1:plants-around-us:parts}{tallo}, \omterm{def:g1:plants-around-us:parts}{hojas} y \omterm{def:g1:plants-around-us:parts}{flor}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:2}
(a) Las \omterm{def:g1:plants-around-us:parts}{raíces}. (b) Las \omterm{def:g1:plants-around-us:parts}{hojas}. (c) La \omterm{def:g1:plants-around-us:parts}{flor}. (d) El \omterm{def:g1:plants-around-us:parts}{tallo}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:3}
Nació de una semilla, se alimenta (agua de la tierra, luz del cielo),
crece, hace semillas que se convierten en margaritas nuevas y un día
muere. Están todos los signos de la vida; andar no es uno de ellos.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:4}
El \omterm{def:g1:plants-around-us:parts}{tallo} del \omterm{def:g1:plants-around-us:tree}{árbol} es el tronco; las \omterm{def:g1:plants-around-us:parts}{hojas} las llevan las \omterm{def:g1:plants-around-us:tree}{ramas}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:5}
Agua, luz y un sitio para sus \omterm{def:g1:plants-around-us:parts}{raíces} --- normalmente tierra.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:6}
El riego. La \omterm{def:g1:plants-around-us:plant}{planta} se ha marchitado por falta de agua; unas horas
después de regarla, sus \omterm{def:g1:plants-around-us:parts}{hojas} volverán a levantarse.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:7}
Se parecen: las dos son \omterm{def:g1:plants-around-us:plant}{plantas}, con \omterm{def:g1:plants-around-us:parts}{raíces}, \omterm{def:g1:plants-around-us:parts}{tallo}, \omterm{def:g1:plants-around-us:parts}{hojas} y \omterm{def:g1:plants-around-us:parts}{flores}, y
las dos necesitan agua y luz. Se diferencian: el \omterm{def:g1:plants-around-us:parts}{tallo} del \omterm{def:g1:plants-around-us:tree}{árbol} es un
tronco leñoso, mientras que el de la margarita es blando y verde, y el
\omterm{def:g1:plants-around-us:tree}{árbol} es mucho más grande y vive mucho más tiempo.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:8}
La \omterm{def:g1:plants-around-us:plant}{planta} de la ventana debería estar verde y erguida; la del armario,
pálida, estirada y débil, aunque se regara. La necesidad que falta es
la luz --- el agua sola no basta.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:9}
Por ejemplo: la \omterm{def:g1:plants-around-us:plant}{planta} crece --- más alta semana tras semana --- y se
gira o se inclina hacia la luz, como el girasol que sigue al día.
Despacio no es lo mismo que no hacer nada.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:10}
La \omterm{def:g1:plants-around-us:parts}{raíz}. El ``bajo tierra'' es el sitio de las \omterm{def:g1:plants-around-us:parts}{raíces} en la figura ---
la zanahoria que comemos es una \omterm{def:g1:plants-around-us:parts}{raíz} naranja, grande y gorda, que la
\omterm{def:g1:plants-around-us:plant}{planta} llenó de alimento.
\end{solution}
