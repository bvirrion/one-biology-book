\chapter{As etapas da vida}\label{ch:g3:stages-of-life}

Ponha em fila uma foto de família: o bebê no colo da vovó, a criança
de escola sem os dois dentes da frente, o primo \omterm{def:g3:stages-of-life:stages}{adolescente} de
repente mais alto que a mãe, os pais, a própria vovó. Uma família,
uma tarde --- e o \omterm{def:g2:born-grow-die:cycle}{ciclo de vida} humano inteiro sentado num sofá.
Neste ano vamos olhar de perto essas etapas, e ver como o próprio
crescimento pode ser medido.

\section{As etapas de uma vida humana}

\begin{definition}[As etapas da vida humana]\label{def:g3:stages-of-life:stages}
Uma vida humana passa por etapas:
\begin{enumerate}
  \item o \emph{bebê}\index{bebê} (mais ou menos os dois primeiros
        anos): o crescimento mais rápido de todos, os primeiros
        dentes, os primeiros passos, as primeiras palavras;
  \item a \emph{criança}\index{criança}: crescimento constante,
        \omterm{prop:g2:teeth-and-eating-well:sets}{dentes de leite} trocados por \omterm{prop:g2:teeth-and-eating-well:sets}{dentes de adulto},
        aprendizados de todo tipo;
  \item o \emph{adolescente}\index{adolescente}: um novo salto de
        crescimento rápido, e o corpo passando de corpo de criança
        a corpo de adulto;
  \item o \emph{adulto}\index{adulto}: o crescimento em altura
        acabou; os adultos podem virar pais e mães;
  \item a \emph{velhice}\index{velhice}: o cabelo embranquece, a
        \omterm{def:g1:the-five-senses:sense}{pele} enruga, a força diminui devagar --- a última etapa
        natural do ciclo.
\end{enumerate}
\end{definition}

\begin{example}[Lendo o sofá]\label{ex:g3:stages-of-life:sofa}
No sofá da família: o bebê (etapa 1), a criança de escola banguela
(etapa 2 --- a troca do \cref{ch:g2:teeth-and-eating-well} a todo
vapor), o primo de dezesseis anos (etapa 3, no meio do salto), os
pais (etapa 4), a vovó (etapa 5). Cinco etapas, uma família.
\end{example}

\begin{remark}[Crescer não é constante]\label{rem:g3:stages-of-life:bursts}
O crescimento tem um ritmo: muito rápido no bebê --- a altura
aumenta mais ou menos a metade dela de novo no primeiro ano ---,
mais lento e constante durante a infância, rápido de novo na
adolescência e então acabado. Dois saltos com um trecho calmo no
meio: essa é a forma normal de crescer.
\end{remark}

\section{Medir o crescimento}

\begin{definition}[Curva de crescimento]\label{def:g3:stages-of-life:chart}
Uma \emph{curva de crescimento}\index{curva de crescimento} é um
registro da altura (ou do peso) medida várias e várias vezes ao
longo do tempo --- as marcas no batente da porta do
\cref{ch:g1:my-body}, feitas com regularidade e com data. Os
médicos mantêm uma para cada criança.
\end{definition}

\begin{example}[Lendo a curva]\label{ex:g3:stages-of-life:reading}
Na curva, a linha sobe cerca de $\qty{25}{cm}$ logo no primeiro ano
--- e só uns $\qty{6}{cm}$ por ano durante a infância. Os dois
trechos íngremes são os dois saltos da
\cref{rem:g3:stages-of-life:bursts}; a ponta plana diz que o \omterm{def:g3:stages-of-life:stages}{adulto}
parou de ficar mais alto.
\end{example}

\begin{method}[Fazer a sua própria curva de crescimento]\label{met:g3:stages-of-life:keep}
\begin{enumerate}
  \item Meça a altura com cuidado (do jeito da parede e do livro
        dos anos anteriores), sempre sem sapatos;
  \item escreva a data e a altura num caderno, ou marque o batente
        da porta e ponha a data na marca;
  \item meça de novo a cada poucos meses --- a regularidade importa
        mais do que a frequência;
  \item para ver a história, marque os pontos: a idade embaixo, a
        altura na lateral, como na figura.
\end{enumerate}
\end{method}

\section{Os animais e as plantas também têm etapas}

\begin{proposition}[Etapas por todo o mundo vivo]\label{prop:g3:stages-of-life:animals}
Todo \omterm{def:g1:living-or-not:living}{ser vivo} passa pelas etapas da sua própria versão do ciclo ---
jovem, \omterm{def:g3:stages-of-life:stages}{adulto}, \omterm{def:g3:stages-of-life:stages}{velhice}. A etapa jovem pode se parecer com o \omterm{def:g3:stages-of-life:stages}{adulto}
(o gatinho, o potrinho) ou não se parecer nada com ele (o girino, a
\omterm{ex:g2:animals-grow-up:butterfly}{lagarta} do \cref{ch:g2:animals-grow-up}); as etapas de uma \omterm{def:g1:plants-around-us:plant}{planta}
vão de \omterm{def:g2:from-seed-to-plant:seed}{semente} a \omterm{prop:g2:from-seed-to-plant:cycle}{plântula}, \omterm{def:g1:plants-around-us:plant}{planta} jovem e adulta com \omterm{def:g1:plants-around-us:parts}{flores}. O que
nenhum deles consegue fazer é pular uma etapa ou percorrê-las de
trás para a frente.
\end{proposition}
\admitted

\begin{example}[Que rápido, que devagar]\label{ex:g3:stages-of-life:speeds}
As velocidades são muito diferentes. Uma mosca é adulta em semanas;
um gato, em mais ou menos um ano; um ser humano, em quase vinte
anos; um carvalho leva várias dezenas de anos até as primeiras
bolotas. Devagar ou rápido, a ordem das etapas nunca muda.
\end{example}

\begin{remark}[A velhice é uma etapa, não um defeito]\label{rem:g3:stages-of-life:oldage}
Cabelos brancos e rugas não são uma doença: são a última etapa do
corpo, tão natural quanto os primeiros dentes do bebê. Em muitas
espécies, e nas famílias humanas, os mais velhos carregam aquilo de
que os jovens mais precisam --- a experiência.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:1}
Diga as cinco etapas de uma vida humana, em ordem.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:2}
Em que etapa: (a) caem os \omterm{prop:g2:teeth-and-eating-well:sets}{dentes de leite}? (b) uma pessoa pode virar
pai ou mãe? (c) o crescimento é o mais rápido da vida inteira?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:3}
Quais são os dois saltos de crescimento, e o que fica entre eles?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:4}
Na \omterm{def:g3:stages-of-life:chart}{curva de crescimento}, como dá para saber, só pela linha, que um
\omterm{def:g3:stages-of-life:stages}{adulto} parou de crescer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:5}
Usando a curva, mais ou menos quantos centímetros a linha sobe entre
os $6$ e os $8$ anos? E no primeiro ano sozinho?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:6}
Por que a altura precisa ser sempre medida sem sapatos, do mesmo
jeito todas as vezes? O que poderia dar errado se não fosse assim?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:7}
Diga as etapas da vida de um carvalho, da bolota à árvore velha.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:8}
Faça a experiência: comece a \omterm{def:g3:stages-of-life:chart}{curva de crescimento} do
\cref{met:g3:stages-of-life:keep} para você, com o primeiro ponto de
hoje. Em que etapa da \cref{def:g3:stages-of-life:stages} você está
--- e que salto ainda está pela frente?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:9}
Uma mosca, um gato, um ser humano e um carvalho percorrem as mesmas
etapas em velocidades muito diferentes. Ponha os quatro em ordem, do
mais rápido ao mais lento para chegar à vida adulta --- e diga o que
as quatro ordens de etapas têm em comum.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:10}
O Tomás, de $9$ anos, se mede duas vezes num mês e não encontra
``crescimento nenhum''. Ele deve se preocupar? Responda com a
\cref{rem:g3:stages-of-life:bursts} e com o formato da curva na
idade dele.
\end{exercise}
