\section*{Chapter \ref{ch:g4:adaptations} --- Animal Adaptations}

\begin{solution}{exo:g4:adaptations:1}
An \omterm{def:g4:adaptations:def}{adaptation} is a \omterm{def:g1:my-body:body}{body} feature or behaviour that helps a \omterm{def:g1:living-nonliving:living}{living thing}
survive and \omterm{def:g2:what-alive:signs}{reproduce} in its environment.
\end{solution}

\begin{solution}{exo:g4:adaptations:2}
Examples: webbed \omterm{def:g1:my-body:parts}{feet} (structural); migration or hibernation
(behavioural). Any correct pair is fine.
\end{solution}

\begin{solution}{exo:g4:adaptations:3}
Prey: harder for predators to see. Predator: harder for prey to notice
the hunter approaching.
\end{solution}

\begin{solution}{exo:g4:adaptations:4}
Thick fur traps \omterm{def:g4:plants-need:needs}{air} and reduces heat loss, helping the \omterm{def:g1:plants-animals:animal}{animal} stay warm.
\end{solution}

\begin{solution}{exo:g4:adaptations:5}
Large \omterm{def:g1:senses:five}{ears} can help release heat (blood near the surface cools). They
may also help detect prey or danger. Accept reasoned heat-loss or
\omterm{def:g1:senses:five}{hearing} answers.
\end{solution}
