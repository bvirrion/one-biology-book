\chapter{Populasi dan Demografi}\label{ch:b1:populations}

Sebuah labu khamir bermula dengan sejuta \omterm{def:b1:cell-unit-of-life:cell}{sel} lalu, enam belas jam
kemudian, memuat enam puluh juta dan telah berhenti tumbuh; sekawanan
rusa di sebuah pulau tumbuh sepersepuluh tiap tahun sampai musim
dingin ketika makanannya habis; lynx hutan boreal naik lalu runtuh
tiap sepuluh tahun, setahun di belakang terwelu yang dimakannya.
Masing-masing adalah sebuah \emph{\omterm{def:b1:populations:population}{populasi}} --- individu satu spesies
di satu tempat --- dan masing-masing menuruti aritmetika yang dapat
dituliskan. Bab ini memerikan bagaimana \omterm{def:b1:populations:population}{populasi} dihitung, bagaimana
kelahiran dan kematiannya ditabelkan, kedua persamaan pertumbuhannya,
gaya yang menahan jumlahnya, dan apa yang dikatakan aritmetika yang
sama tentang spesies kita sendiri.

\section{Menghitung sebuah populasi}

\begin{definition}[Populasi, kerapatan, sebaran]\label{def:b1:populations:population}
\index{populasi}\index{kerapatan populasi}
Yang disebut \emph{populasi} adalah himpunan individu satu spesies
yang hidup dalam sebuah daerah tertentu pada saat tertentu dan sanggup
berbiak silang. \emph{Ukuran}nya $N$ adalah cacah individunya;
\emph{kerapatan}nya adalah cacah tiap satuan luas atau volume;
\emph{sebaran}nya adalah pola jarak antarindividunya ---
\emph{mengelompok} bila sumber daya atau kehidupan sosialnya
mengumpulkannya (kawanan, gerombolan, rumpun sebatang tumbuhan),
\emph{seragam} bila keduanya saling menolak (burung berteritori, semak
gurun yang bersaing memperebutkan air), \emph{acak} bila tak satu pun
bekerja. Populasi juga memiliki sebuah \emph{struktur umur}, sebuah
\emph{nisbah kelamin}, serta laju kelahiran, kematian, imigrasi dan
emigrasi, yang neracanya adalah pertumbuhannya.
\end{definition}

\begin{method}[Menaksir sebuah populasi]\label{met:b1:populations:counting}
\index{tanda--tangkap ulang}
\begin{enumerate}
  \item \omterm{def:b1:organism-environment:organism}{Organisme} yang menetap atau lamban: hitunglah dalam
        \emph{kuadrat} berluas tertentu yang ditaruh secara acak, lalu
        kalikan kerapatan puratanya dengan luas totalnya; sebaran
        antarkuadratnya memberi ketelitiannya lalu menyingkapkan
        sebarannya.
  \item Hewan yang bergerak: \emph{\omterm{met:b1:populations:counting}{tanda--tangkap ulang}}. Tangkaplah $M$
        individu, tandai lalu lepaskan; kemudian tangkaplah $C$, yang
        $R$ di antaranya bertanda. Bila yang bertanda bercampur bebas
        dan tak ada yang berubah, fraksi bertanda tangkapan keduanya
        setara dengan fraksi bertanda \omterm{def:b1:populations:population}{populasinya}, $R/C = M/N$, sehingga
        $N \approx MC/R$ (taksiran Lincoln--Petersen).
  \item Periksalah anggapannya: tak ada kelahiran, kematian atau
        migrasi antara kedua tangkapannya, tanda yang tidak lepas
        maupun mengubah perilaku hewannya, ketertangkapan yang setara.
        Tiap kegagalannya membiaskan $N$ pada arah yang dapat
        diramalkan.
  \item Tanda tak langsung bila menghitung mustahil: kotoran, jejak,
        panggilan, sarang, \omterm{def:b1:nucleic-acids:chain}{DNA} dalam seliter air kolam.
\end{enumerate}
\end{method}

\begin{example}[Tikus ladang di sebuah padang rumput]\label{ex:b1:populations:voles}
Enam puluh tikus ladang ditangkap, ditandai lalu dilepaskan dalam
sehektar padang rumput; tiga malam kemudian delapan puluh ditangkap,
yang dua belas di antaranya membawa tanda:
$N \approx 60\times 80/12 = 400$ tikus ladang, empat tiap seratus meter
persegi. Bila tikus ladang bertandanya, yang telah belajar bahwa
perangkapnya memuat makanan, tertangkap lebih mudah, $R$ menggembung
dan $N$ tertaksir terlalu rendah; bila penandaannya menakutinya menjauh
dari perangkapnya, sebaliknya.
\end{example}

\section{Demografi: kelahiran dan kematian menurut umur}

\begin{definition}[Tabel hayat]\label{def:b1:populations:lifetable}
\index{tabel hayat}\index{kesintasan}
Sebuah \emph{tabel hayat} mengikuti sebuah kohort individu yang lahir
bersama sepanjang hidupnya. Bagi tiap umur $x$ ia mencatat $l_x$,
yaitu \emph{kesintasan} --- fraksi kohortnya yang masih hidup pada umur
$x$ --- dan $m_x$, yaitu \emph{fekunditas} --- cacah purata anak betina
yang dilahirkan seekor betina berumur $x$. Dari keduanya: kematian
antarumurnya, $q_x = 1 - l_{x+1}/l_x$; \emph{laju reproduksi
bersih} $R_0 = \sum_x l_x m_x$, yaitu cacah anak betina yang akan
dimiliki seekor betina yang baru lahir secara purata sepanjang
hidupnya; serta \emph{waktu generasi}
$T = \sum_x x\,l_x m_x / R_0$, yaitu umur purata induk anak betina itu.
Sebuah \omterm{def:b1:populations:population}{populasi} dengan $R_0 > 1$ tumbuh, dengan $R_0 < 1$
menyusut; sepanjang satu generasi ia dikalikan $R_0$.
\end{definition}

\begin{proposition}[Tiga bentuk kesintasan]\label{prop:b1:populations:survivorship}
Menggambarkan $\log l_x$ terhadap umurnya memberi tiga jenis kurva.
\emph{Jenis I}: sebagian besar individunya sintas sampai tua lalu mati
dalam sebuah rentang yang sempit (mamalia besar, manusia di negara
kaya). \emph{Jenis II}: sebuah peluang kematian yang tetap pada tiap
umurnya, sebuah garis lurus pada skala logaritmiknya (banyak burung,
kadal, hewan pengerat). \emph{Jenis III}: kematian yang sangat besar di
antara yang muda dan umur panjang bagi segelintir yang sintas (tiram,
ikan, sebagian besar tumbuhan dan serangga). Bentuknya mengikuti cara
hidup \omterm{def:b1:organism-environment:organism}{organismenya}: sebuah spesies yang menanamkan sedikit pada tiap
anak yang banyak memiliki kurva jenis III; yang menanamkan banyak pada
tiap anak yang sedikit, jenis I.
\end{proposition}

\begin{example}[Sebuah tabel hayat bagi sekawanan rusa]\label{ex:b1:populations:deer}
Dari seribu anak rusa, enam ratus mencapai ulang tahun pertamanya, dan
penyintasnya lalu kehilangan sepersepuluh jumlahnya tiap tahun sampai
sedikit yang melampaui sembilan; rusa betinanya beranak sejak umur
dua, sekitar delapan persepuluh anak betina setahun pada masa
puncaknya. Menjumlahkan $l_x m_x$ memberi $R_0 \approx 1.5$: tiap rusa
betina meninggalkan satu setengah anak betina, dan kawanannya tumbuh
setengah tiap generasinya yang sekitar empat tahun --- sebuah
pertambahan sebelas persen tiap tahun sampai sesuatu menghentikannya.
\end{example}

\section{Pertumbuhan}

\begin{theorem}[Pertumbuhan eksponensial]\label{thm:b1:populations:exponential}
\index{pertumbuhan eksponensial}\index{laju pertambahan intrinsik}
Sebuah \omterm{def:b1:populations:population}{populasi} yang laju kelahiran dan kematian tiap kepalanya $b$
dan $d$ tetap tumbuh pada laju yang sebanding dengan ukurannya,
\[
\frac{\dd N}{\dd t} = rN, \qquad r = b - d, \qquad N(t) = N_0\,e^{rt},
\]
dengan $r$ adalah \emph{\omterm{thm:b1:populations:exponential}{laju pertambahan intrinsik}}nya. Ia berlipat
dua tiap $\ln 2/r$; sebuah \omterm{def:b1:populations:population}{populasi} dengan $r = \qty{0.11}{yr^{-1}}$
berlipat dua dalam \qty{6.3}{yr}, yang dengan $r = \qty{0.5}{h^{-1}}$
dalam \qty{83}{min}. Dari sebuah \omterm{def:b1:populations:lifetable}{tabel hayat}, $r \approx \ln R_0 / T$.
\end{theorem}
\begin{proof}
Dalam sebuah selang pendek $\dd t$ kelahirannya adalah $bN\,\dd t$ dan
kematiannya $dN\,\dd t$, sehingga $\dd N = (b - d)N\,\dd t$;
persamaan $\dd N/\dd t = rN$ memiliki penyelesaian $N_0 e^{rt}$
(turunan $e^{rt}$ adalah $re^{rt}$). Perlipatduaannya: $e^{rt} = 2$
memberi $t = \ln 2/r$. Sepanjang satu generasi $T$ \omterm{def:b1:populations:population}{populasinya}
dikalikan $R_0 = e^{rT}$, dari situ $r = \ln R_0/T$.
\end{proof}

\begin{theorem}[Pertumbuhan logistik]\label{thm:b1:populations:logistic}
\index{pertumbuhan logistik}\index{daya dukung}
Bila laju pertumbuhan tiap kepalanya turun secara linear seiring
\omterm{def:b1:populations:population}{populasinya} mendekati sebuah \emph{\omterm{thm:b1:populations:logistic}{daya dukung}} $K$ --- yaitu ukuran
yang dapat ditanggung \omterm{def:b1:organism-environment:organism}{lingkungannya} ---
\[
\frac{\dd N}{\dd t} = rN\left(1 - \frac{N}{K}\right), \qquad
N(t) = \frac{K}{1 + \dfrac{K - N_0}{N_0}\,e^{-rt}} .
\]
Pertumbuhannya nyaris eksponensial selagi $N \ll K$, paling cepat pada
$N = K/2$ (dengan $\dd N/\dd t = rK/4$), lalu lenyap pada $N = K$, yang
didekati \omterm{def:b1:populations:population}{populasinya} sebagai sebuah dataran: sebuah kurva berbentuk S.
\end{theorem}
\begin{proof}
Tuliskan persamaannya sebagai $\dd N/[N(1 - N/K)] = r\,\dd t$ lalu
belahlah \omterm{def:b1:flowering-plant-organization:organs}{ruas} kirinya menjadi $\dd N/N + \dd N/(K - N)$;
mengintegralkannya memberi $\ln[N/(K - N)] = rt + \text{const}$, dan
menyelesaikannya bagi $N$ dengan $N(0) =
N_0$ memberi rumusnya. Menurunkan $rN(1 - N/K)$ terhadap
$N$ lalu menyamakannya dengan nol memberi maksimumnya pada $N = K/2$.
\end{proof}

\begin{proposition}[Kebergantungan kerapatan]\label{prop:b1:populations:densitydependence}
\index{kebergantungan kerapatan}
Pertumbuhannya melambat seiring kerapatannya naik sebab laju
kelahiran tiap kepalanya turun atau laju kematiannya naik karena
kesesakan: makanan tiap kepalanya menurun, teritorinya menjadi tak
tersedia, sampah dan penyakitnya bertimbun, pemangsanya memusat.
Faktor yang \emph{bergantung kerapatan} ini mengatur sebuah \omterm{def:b1:populations:population}{populasi}
menuju $K$. Faktor yang \emph{bebas kerapatan} --- beku, banjir,
kebakaran, kekeringan --- membunuh sebagian tanpa memandang jumlahnya
lalu tak dapat mengatur, hanya mengganggu; \omterm{def:b1:populations:population}{populasi} banyak serangga
dikuasai sebagian besarnya oleh cuacanya lalu tak pernah mendekati
sebuah \omterm{thm:b1:populations:logistic}{daya dukung}.
\end{proposition}
\begin{proof}[Bukti]
Gause (1934) menumbuhkan \emph{Paramecium} dalam tabung berisi medium
yang diperbarui tiap hari: cacahnya mengikuti sebuah kurva logistik
sampai sebuah dataran yang ditetapkan makanannya, dan melipatduakan
makanannya melipatduakan datarannya. Pearl (1927) menemukan hal yang
sama pada lalat buah dalam botol, dan Carlson (1913) pada khamir. Di
lapangan, burung gereja penyanyi di sebuah pulau bertelur lebih
sedikit lalu kehilangan lebih banyak anak ketika \omterm{def:b1:populations:population}{populasi} berbiaknya
besar, dan cacah teritorinya membatasi cacah pembiaknya; menyingkirkan
burungnya diikuti penggantiannya dari sebuah kelebihan yang tadinya
tersisih.
\end{proof}

\begin{definition}[Siasat riwayat hidup]\label{def:b1:populations:rK}
\index{siasat r}\index{siasat K}
Spesies berbeda dalam letaknya di antara dua ujung. Sebuah spesies
\emph{terpilih $r$} --- gulma, kutu \omterm{def:b1:flowering-plant-organization:organs}{daun}, tikus, sebagian besar ikan
--- dibangun bagi $r$ yang tinggi: ia matang awal, menghasilkan banyak
anak yang kecil, menanamkan sedikit pada tiap anaknya, hidup singkat,
lalu memanfaatkan habitat yang baru atau terganggu sebelum
persaingannya tiba, dengan \omterm{def:b1:populations:lifetable}{kesintasan} jenis III. Sebuah spesies
\emph{terpilih $K$} --- pohon ek, gajah, albatros --- dibangun untuk
bertahan dekat $K$: ia matang lambat, menghasilkan sedikit anak yang
besar, mengasuhnya, hidup lama, lalu bersaing baik dalam habitat yang
sesak dan mantap, dengan kurva jenis I. Sebagian besar spesies
mencampur keduanya; namanya menyebut ujung skalanya.
\end{definition}

\section{Naik turun dan pengaturan}

\begin{proposition}[Daur pemangsa dan mangsanya]\label{prop:b1:populations:cycles}
\index{daur populasi}
Sebagian \omterm{def:b1:populations:population}{populasi} tidak mereda melainkan berayun. Terwelu sepatu salju
hutan utara dan lynx yang memakannya naik lalu turun bersama dengan
sebuah kala sekitar sepuluh tahun, lynxnya setahun atau dua tahun di
belakang terwelunya: banyak terwelu memberi makan banyak lynx,
banyak lynx memakan habis terwelunya, sedikit terwelu membuat
lynxnya kelaparan, sedikit lynx membiarkan terwelunya pulih. Daurnya dipacu
sebagian oleh pemangsanya dan sebagian oleh makanan terwelunya
sendiri, yang pulihnya sesudah penjelajahan berlebihan memakan
bertahun-tahun; model pemangsa dan mangsa yang bergandeng
(\cref{ch:b1:species-interactions}) menghasilkan ayunan semacam itu
dari dua persamaan.
\end{proposition}
\begin{proof}[Bukti]
Catatan bulu sebuah perusahaan dagang, yang disimpan selama sembilan
puluh tahun sejak 1840-an, merekam cacah kulit lynx dan terwelu yang
dibawa masuk tiap tahun di seluruh sebuah subbenua: kedua deretnya
memperlihatkan puncak tiap sembilan sampai sepuluh tahun, puncak
lynxnya tertinggal dari puncak terwelunya, sepanjang sembilan daur
penuh. Kajian lapangan sesudahnya, dengan terwelunya dipagari dari
pemangsanya atau diberi makanan tambahan, memperlihatkan bahwa
menyingkirkan salah satu faktornya meredam daurnya dan menyingkirkan
keduanya meniadakannya.
\end{proof}

mangsanya, yang tertunda.}
\end{omfigure}

\begin{example}[Spesies kita sendiri]\label{ex:b1:populations:humans}
\omterm{def:b1:populations:population}{Populasi} manusia memakan seluruh sejarahnya untuk mencapai satu
miliar, sekitar tahun 1800, lalu dua abad untuk mencapai delapan. Laju
pertumbuhannya memuncak dekat \qty{2}{\%} setahun pada 1960-an
(berlipat dua dalam 35 tahun) lalu sejak itu turun di bawah
\qty{1}{\%}, bukan lewat kematiannya melainkan lewat kelahirannya:
tempat kematian anaknya turun dan perempuannya berpendidikan,
keluarganya mengecil dalam satu generasi, dan piramida umurnya berubah
dari sebuah segitiga menjadi sebuah kolom. Aritmetika $R_0$ berlaku
bagi kita; yang berubah adalah $m_x$.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:populations:1}
Definisikan \omterm{def:b1:populations:population}{populasi}, kerapatan dan sebaran, lalu berilah sebuah
contoh bagi tiap pola sebarannya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:populations:2}
Definisikan $l_x$, $m_x$, $R_0$ dan $T$, lalu katakan apa arti $R_0 = 1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:populations:3}
Dari gambar \omterm{def:b1:populations:lifetable}{kesintasannya}, berapa fraksi sebuah kohort jenis III yang
sintas sepersepuluh pertama umurnya? Dari sebuah kohort jenis I,
berapa fraksi yang mencapai tiga perempatnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:populations:4}
Nyatakan persamaan eksponensial dan logistiknya lalu katakan apa arti
$r$ dan $K$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:5}
Sebuah biakan bakteri tumbuh dari $10^4$ menjadi $10^7$ \omterm{def:b1:cell-unit-of-life:cell}{sel} tiap
mililiter dalam \qty{5}{h}. Hitunglah $r$ dan waktu lipat duanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:6}
Sebuah \omterm{def:b1:populations:lifetable}{tabel hayat}: $l_x = 1, 0.5, 0.3, 0.1, 0$ pada umur 0 sampai 4; $m_x = 0,
1, 2, 2, 0$. Hitunglah $R_0$, $T$ dan $r$, lalu katakan apakah
\omterm{def:b1:populations:population}{populasinya} tumbuh.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:7}
Lima puluh ikan ditandai lalu dilepaskan dalam sebuah kolam; sepekan
kemudian 70 ditangkap, 7 bertanda. Taksirlah $N$. Bila sepuluh ikan
bertandanya mati akibat penanganannya, apakah taksirannya terlalu
tinggi atau terlalu rendah, dan seberapa besar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:8}
Sebuah \omterm{def:b1:populations:population}{populasi} logistik memiliki $K = 1000$ dan $r = \qty{0.2}{yr^{-1}}$.
Hitunglah $\dd N/\dd t$ pada $N = 100$, $500$ dan $900$, serta cacah
maksimum yang dapat dipanen tiap tahun tanpa penyusutan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:populations:9}
Golongkan sebagai terpilih $r$ atau terpilih $K$, dengan dua sifat
masing-masing: jombang, gajah, ikan kod, gorila, lalat rumah.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:populations:10}
Dari gambar khamirnya, taksirlah $r$ dari kedua titik pertamanya dan
$K$ dari titik terakhirnya, lalu ramalkan $N$ pada \qty{8}{h} dari
rumus logistiknya lalu bandingkanlah dengan cacahnya. Pada saat mana
laju pertumbuhannya paling tinggi, dan berapa nilainya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:populations:11}
Jelaskan mengapa sebuah faktor yang bebas kerapatan tak dapat mengatur
sebuah \omterm{def:b1:populations:population}{populasi} walaupun ia dapat menguranginya, dengan memakai
persamaannya, dan mengapa \omterm{def:b1:populations:population}{populasi} serangga yang dikuasai cuaca tetap
saja tidak tumbuh tanpa batas.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:populations:12}
``Sebuah \omterm{def:b1:populations:population}{populasi} bukanlah sebuah bilangan melainkan sebuah sebaran.''
Bahaslah dalam satu paragraf: struktur umurnya, tundaan antara turunnya
$m_x$ dan turunnya $N$, dan apa yang diramalkan sebuah piramida yang
tidak diramalkan $N$ sendirian.
\end{exercise}

\section{Soal: Khamir dalam Sebuah Labu dan Rusa di Sebuah Pulau}

\begin{problem}[{Soal akhir pekan --- sebuah biakan yang dihitung,
sekawanan yang ditabelkan, sebuah danau yang dicuplik dan sebuah panen
yang ditetapkan, berakhir pada daya dukung dan laju pertumbuhan
intrinsik khamirnya}]
\label{pb:b1:populations:1}
Sebuah biakan khamir dihitung tiap dua jam (\omterm{def:b1:cell-unit-of-life:cell}{sel} tiap mililiter,
$\times 10^5$): $10, 29, 71, 175, 351, 513, 595, 641, 656$ pada $t = 0,
2, \ldots, 16$ h. Sebuah \omterm{def:b1:populations:population}{populasi} rusa di sebuah pulau memiliki \omterm{def:b1:populations:lifetable}{tabel
hayat}: $l_x = 1.00, 0.60, 0.52, 0.46, 0.40, 0.34, 0.26, 0.16, 0.06, 0$ dan
$m_x = 0, 0, 0.6, 0.8, 0.8, 0.8, 0.7, 0.5, 0.3, 0$ bagi $x = 0$ sampai $9$
tahun. Dalam sebuah danau, 120 ikan perch ditandai lalu dilepaskan;
sepekan kemudian 150 ditangkap, 18 bertanda.

\medskip
\noindent\textbf{Bagian I --- Khamirnya.}
\begin{enumerate}
  \item Taksirlah $r$ dari kedua cacah pertamanya, dengan menganggap
        \omterm{thm:b1:populations:exponential}{pertumbuhan eksponensial} sepanjang dua jam pertamanya.
  \item Hitunglah waktu lipat duanya.
  \item Taksirlah $K$ dari cacah terakhirnya.
  \item Hitunglah ramalan logistiknya pada $t = 4$, $8$ dan \qty{12}{h}
        lalu bandingkanlah dengan cacahnya.
  \item Pada \omterm{def:b1:populations:population}{populasi} berapa pertumbuhannya paling cepat, dan kapan ia
        dicapai? Hitunglah laju pertumbuhan maksimumnya dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} tiap
        mililiter tiap jam.
  \item Apa yang akan diramalkan model eksponensialnya pada
        \qty{12}{h}? Berapa kali lipat ia keliru?
  \item Sebutkan dua hal yang dapat menetapkan $K$ dalam sebuah labu
        khamir yang tersegel.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Rusanya.}
\begin{enumerate}[resume]
  \item Hitunglah kematian $q_x$ pada tahun pertama dan pada tahun
        kelimanya. Jenis \omterm{def:b1:populations:lifetable}{kesintasan} yang mana ini?
  \item Hitunglah $l_x m_x$ bagi tiap umurnya lalu jumlahkan: $R_0$.
  \item Hitunglah $T$.
  \item Hitunglah $r$ dan waktu lipat dua kawanannya.
  \item Pulaunya memuat 200 rusa hari ini. Ramalkan cacahnya dalam 10
        tahun lewat \omterm{thm:b1:populations:exponential}{pertumbuhan eksponensialnya}.
  \item Pulaunya sanggup memberi makan 600 rusa. Ramalkan cacahnya
        dalam 10 tahun lewat \omterm{thm:b1:populations:logistic}{pertumbuhan logistiknya}, dan tahun ketika
        kawanannya melampaui 500.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Ikan perch dan panennya.}
\begin{enumerate}[resume]
  \item Taksirlah \omterm{def:b1:populations:population}{populasi} ikan perch danaunya.
  \item Bila 20 ikan perch bertandanya kehilangan tandanya sebelum
        tangkapan keduanya, hitunglah kembali dengan cacah sejati ikan
        bertandanya, lalu nyatakan arah kekeliruan taksiran
        pertamanya.
  \item \omterm{def:b1:populations:population}{Populasi} ikan perchnya memiliki $K = 1000$ dan $r = \qty{0.4}{yr^{-1}}$.
        Hitunglah hasil lestari maksimumnya dan \omterm{def:b1:populations:population}{populasi} tempat ia
        diperoleh.
  \item Hitunglah waktu lipat dua \omterm{def:b1:populations:population}{populasi} ikan perchnya ketika ia
        jauh di bawah $K$.
  \item Perkumpulannya memutuskan alih-alih mengambil \qty{20}{\%}
        tetap dari stoknya tiap tahun. Hitunglah \omterm{def:b1:populations:population}{populasi}
        kesetimbangannya dan tangkapan tahunannya pada kesetimbangan
        itu.
  \item Sebuah perkumpulan pemancing mengambil 120 ikan perch setahun.
        Tunjukkan apakah \omterm{def:b1:populations:population}{populasinya} sanggup menanggungnya, dengan
        menghitung $\dd N/\dd t$ pada $N =
        500$ dan pada $N = 300$.
  \item Mengapa memanen pada $K/2$ berisiko dalam kenyataannya, dengan
        mengingat cuacanya dan kekeliruan penghitungan pada pertanyaan
        14?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pengaturan.}
\begin{enumerate}[resume]
  \item $m_x$ rusanya turun menjadi separuh ketika kawanannya melampaui
        400. Tunjukkan bahwa $R_0$ lalu jatuh di bawah 1 --- apa yang
        dilakukan hal ini pada kawanannya, dan jenis faktor apa yang
        bekerja?
  \item Sebuah musim dingin yang keras membunuh \qty{40}{\%} rusanya
        tanpa memandang jumlahnya. Adakah faktor ini mengatur? Apa yang
        terjadi pada kawanannya pada tahun sesudahnya?
  \item Serigala tiba lalu mengambil 30 rusa setahun secara tetap.
        Hitunglah ukuran kawanan pada kesetimbangan bagi model
        logistiknya dengan $K = 600$ dan $r$ pada pertanyaan 11
        (selesaikan $rN(1 - N/K) = 30$), lalu katakan kesetimbangan
        yang mana yang mantap.
  \item Jelaskan dalam dua kalimat mengapa daur lynxnya tertinggal dari
        daur terwelunya.
  \item Nyatakan hasilnya: \omterm{thm:b1:populations:logistic}{daya dukung} dan laju pertumbuhan intrinsik
        khamirnya, serta $R_0$, $T$ dan $r$ rusanya.
\end{enumerate}
\end{problem}
