\section*{Chapter \ref{ch:g8:diffusion} --- Diffusion and Osmosis}

\begin{solution}{exo:g8:diffusion:1}
Net movement of particles from high to low concentration down a gradient
by random motion; no metabolic \omterm{def:g6:nutrition:def}{energy} required.
\end{solution}

\begin{solution}{exo:g8:diffusion:2}
Net \omterm{def:g8:diffusion:diff}{diffusion} of \omterm{def:g4:plants-need:needs}{water} across a \omterm{def:g8:diffusion:membrane}{partially permeable membrane} from higher
to lower \omterm{def:g4:plants-need:needs}{water} concentration (dilute $\to$ concentrated solution).
\end{solution}

\begin{solution}{exo:g8:diffusion:3}
Steeper gradient; higher temperature; shorter distance; larger surface
area (any three).
\end{solution}

\begin{solution}{exo:g8:diffusion:4}
\omterm{def:g6:nutrition:groups}{Water} \omterm{def:g7:tissues:plant-organs}{leaves} by \omterm{def:g8:diffusion:osmosis}{osmosis}; the \omterm{def:g7:microscope:cell}{cell} shrinks (crenates).
\end{solution}

\begin{solution}{exo:g8:diffusion:5}
\omterm{def:g8:diffusion:diff}{Diffusion}: down gradient, no \omterm{def:g7:microscope:cell}{cell} \omterm{def:g6:nutrition:def}{energy}. \omterm{def:g8:diffusion:active}{Active transport}: against
gradient, uses ATP/membrane \omterm{def:g6:nutrition:groups}{proteins}.
\end{solution}
