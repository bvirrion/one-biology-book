\section*{अध्याय \ref{ch:g3:food-chains} --- आहार शृंखलाएँ}

\begin{solution}{exo:g3:food-chains:1}
\omterm{def:g3:food-chains:chain}{``द्वारा खाया जाता है''}। वह खाए जाने वाले से खाने वाले की ओर इशारा
करता है --- यानी भोजन के पीछे-पीछे।
\end{solution}

\begin{solution}{exo:g3:food-chains:2}
कोई \omterm{def:g1:plants-around-us:plant}{पौधा} (या कोई और \omterm{def:g1:living-or-not:living}{सजीव} जो अपना भोजन ख़ुद बनाता है)। क्योंकि सिर्फ़
\omterm{def:g1:plants-around-us:plant}{पौधे} ही किसी को खाए बिना पोषण लेते हैं --- प्रकाश, पानी और हवा से; और
हर जंतु-कड़ी सीधे या दूसरों के ज़रिए उसी पहली हरी कड़ी पर जीती है।
\end{solution}

\begin{solution}{exo:g3:food-chains:3}
घास $\rightarrow$ टिड्डा $\rightarrow$ छिपकली
$\rightarrow$ बाज़।
\end{solution}

\begin{solution}{exo:g3:food-chains:4}
तीर उलटे हैं: घास $\rightarrow$ ख़रगोश
$\rightarrow$ लोमड़ी --- घास ख़रगोश द्वारा खाई जाती है, और ख़रगोश
लोमड़ी द्वारा।
\end{solution}

\begin{solution}{exo:g3:food-chains:5}
बलूत की \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} $\rightarrow$ \omterm{ex:g2:animals-grow-up:butterfly}{इल्ली} $\rightarrow$ नीली चिड़िया
$\rightarrow$ बाज़।
\end{solution}

\begin{solution}{exo:g3:food-chains:6}
सलाद घोंघे \omterm{def:g3:food-chains:chain}{द्वारा खाया जाता है}; और घोंघे को साही खाता है।
\end{solution}

\begin{solution}{exo:g3:food-chains:7}
जैसे: पानी के \omterm{def:g1:plants-around-us:plant}{पौधे} $\rightarrow$ \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} $\rightarrow$
काँटा-मछली $\rightarrow$ बगुला।
\end{solution}

\begin{solution}{exo:g3:food-chains:8}
क्योंकि उसके भोजन के भोजन का भोजन घास है: भूरी पड़ी घास पर टिड्डे कम
होते हैं, कम टिड्डों का मतलब कम छिपकलियाँ, और कम छिपकलियों का मतलब
बाज़ के लिए दुबला शिकार। पहली कड़ी का बदलाव पूरी शृंखला में ऊपर तक
सफ़र करता है।
\end{solution}

\begin{solution}{exo:g3:food-chains:9}
घोंघों को ज़हर देने से उनका खाने वाला भूखा रह जाता है: साही चला जाता
है या मर जाता है। बाद में, घोंघों के मुख्य दुश्मन साही के चले जाने पर,
बचे हुए स्लग और घोंघे पहले से भी ज़्यादा बढ़ जाते हैं: कड़ियाँ हटाना
शृंखला में ऊपर भी सफ़र करता है (भूखा साही) और नीचे भी (बिना शिकार वाले
घोंघे)।
\end{solution}

\begin{solution}{exo:g3:food-chains:10}
सही --- पर शृंखला के बीच के ज़रिए। छिपकली टिड्डे खाती है, और टिड्डे
घास खाते हैं: घास मारी जाए तो यह कमी घास से टिड्डे तक और टिड्डे से
छिपकली तक, कड़ी दर कड़ी ऊपर सफ़र करती है। छिपकली घास को कभी छूती नहीं,
और फिर भी उस पर निर्भर है।
\end{solution}
