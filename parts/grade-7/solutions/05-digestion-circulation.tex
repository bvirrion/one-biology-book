\section*{Chapter \ref{ch:g7:digestion} --- Digestion and Circulation}

\begin{solution}{exo:g7:digestion:1}
\omterm{def:g7:digestion:def}{Digestion} breaks large \omterm{def:g2:food-energy:food}{food} molecules into small soluble ones. Only small
molecules can pass efficiently into \omterm{def:g7:digestion:blood}{blood}/lymph across gut walls.
\end{solution}

\begin{solution}{exo:g7:digestion:2}
\omterm{def:g3:teeth:path}{Mouth} $\to$ oesophagus $\to$ \omterm{def:g3:teeth:path}{stomach} $\to$ small \omterm{def:g3:teeth:path}{intestine} $\to$ large
\omterm{def:g3:teeth:path}{intestine} $\to$ anus.
\end{solution}

\begin{solution}{exo:g7:digestion:3}
Small \omterm{def:g3:teeth:path}{intestine}; \omterm{def:g7:digestion:villi}{villi} (and microvilli if mentioned).
\end{solution}

\begin{solution}{exo:g7:digestion:4}
\omterm{def:g7:digestion:blood}{Arteries}: away from \omterm{def:g7:digestion:heart}{heart} (usually high pressure). \omterm{def:g7:digestion:blood}{Veins}: toward \omterm{def:g7:digestion:heart}{heart}.
\omterm{def:g7:digestion:blood}{Capillaries}: tiny thin-walled exchange vessels in \omterm{def:g7:tissues:tissue}{tissues}.
\end{solution}

\begin{solution}{exo:g7:digestion:5}
\omterm{def:g7:digestion:blood}{Blood} passes through the \omterm{def:g7:digestion:heart}{heart} twice per full \omterm{def:g1:my-body:body}{body} circuit: pulmonary
(\omterm{def:g5:circulation:breathing}{lungs}) and systemic (\omterm{def:g1:my-body:body}{body}) loops.
\end{solution}
