\chapter{استقلاب الخلية}\label{ch:g10:cell-metabolism}

شُدّ بالونًا على عنق دورق فيه ماء سكري دافئ وملعقة من خميرة الخباز.
ففي غضون ساعة يعكرّ السائل ويفور، وينتصب البالون، وتفوح من الدورق
رائحة خفيفة من الخبز والجعة. \omterm{def:g10:cells-common-unit:cell}{خلايا} الخميرة تأكل السكر وتحوله إلى شيء
آخر، والغاز الذي تطلقه هو ثاني أكسيد الكربون نفسه الذي تزفره أنت. أما
الورقة في الشمس فتفعل العكس: تأخذ ثاني أكسيد الكربون وتطلق الأكسجين.
وكلا الأمرين كيمياء، تجري داخل \omterm{def:g10:cells-common-unit:cell}{الخلايا}، عند حرارة الجسم، بلا لهب ولا
حمض --- وهذا الفصل عن تلك الكيمياء.

\section{الاستقلاب: كيمياء الخلية}

\begin{definition}[الاستقلاب]\label{def:g10:cell-metabolism:metabolism}
\emph{استقلاب}\index{استقلاب} \omterm{def:g10:cells-common-unit:cell}{خلية} ما هو مجموع التفاعلات الكيميائية
التي تجري داخلها: تلك التي تفكك الجزيئات لتحرير طاقة، وتلك التي تبني
جزيئات \omterm{def:g10:cells-common-unit:cell}{الخلية} نفسها من جزيئات أبسط. ويقود كل تفاعل من تفاعلات
الاستقلاب \emph{إنزيم}\index{إنزيم}، وهو \omterm{def:g10:chemistry-of-life:families}{بروتين} يسرّع تفاعلًا واحدًا
بعينه دون أن يستهلك فيه.
\end{definition}

\begin{proposition}[الإنزيمات تجعل كيمياء الخلية ممكنة]\label{prop:g10:cell-metabolism:enzymes}
الغلوكوز المتروك في الهواء عند \qty{37}{\celsius} لا يحترق؛ أما في
\omterm{def:g10:cells-common-unit:cell}{الخلية} فيؤكسد أكسدة تامة في دقائق. والفرق هو الإنزيمات: فكل \omterm{def:g10:cell-metabolism:metabolism}{إنزيم}
يرتبط بجزيء بعينه (هو \emph{ركيزته})، ويحوله إلى ناتج، ويطلقه، ويعيد
الكرة، آلاف المرات في الثانية. \omterm{def:g10:cell-metabolism:metabolism}{فاستقلاب} \omterm{def:g10:cells-common-unit:cell}{الخلية} إذن تثبته مجموعة
الإنزيمات التي تملكها --- وتلك المجموعة تثبتها مورّثاتها.
\end{proposition}
\admitted

\begin{example}[إنزيم واحد في العمل]\label{ex:g10:cell-metabolism:catalase}
صب ماء أكسجينيًا على قطعة كبد نيئة أو بطاطا: فترغي بعنف إذ ينطلق
الأكسجين. \omterm{def:g10:cell-metabolism:metabolism}{فإنزيم} الكاتالاز، الموجود في كل \omterm{def:g10:cells-common-unit:cell}{خلية} تقريبًا، يشطر الماء
الأكسجيني السام إلى ماء وأكسجين بمعدل ملايين الجزيئات في الثانية لكل
جزيء \omterm{def:g10:cell-metabolism:metabolism}{إنزيم}. واسلق الكبد أولًا فلا يحدث شيء: \omterm{def:g10:cell-metabolism:metabolism}{فالإنزيم}، وهو \omterm{def:g10:chemistry-of-life:families}{بروتين}، قد
هدمته الحرارة. أما الماء الأكسجيني على حجر فلا يرغي، وهذا يبين أن
التفاعل لا يجري من تلقاء نفسه.
\end{example}

\section{طريقتان في تغذية الخلية}

\begin{definition}[التغذي الذاتي والتغذي غير الذاتي]\label{def:g10:cell-metabolism:autotrophy}
\omterm{def:g10:cells-common-unit:cell}{الخلية} \emph{ذاتية التغذي}\index{تغذ ذاتي} إذا بنت جزيئاتها العضوية من
\omterm{def:g10:chemistry-of-life:mineral-organic}{المادة المعدنية} وحدها --- ثاني أكسيد الكربون والماء \omterm{def:g10:chemistry-of-life:mineral-organic}{والأملاح المعدنية}
--- مستعملة مصدر طاقة خارجيًا. \omterm{def:g10:cells-common-unit:cell}{والخلايا} النباتية ذات \omterm{def:g10:cells-common-unit:organelle}{البلاستيدات
الخضراء} والطحالب وبعض البكتيريا ذاتية التغذي، ومصدر طاقتها الضوء: وهذا
هو \emph{التركيب الضوئي}\index{تركيب ضوئي}. أما \omterm{def:g10:cells-common-unit:cell}{الخلية} \emph{غير ذاتية
التغذي}\index{تغذ غير ذاتي} فعليها أن تأخذ جزيئات عضوية صنعتها كائنات
أخرى، مادةَ بناء ومصدرَ طاقة معًا: \omterm{def:g10:cells-common-unit:cell}{كالخلايا} الحيوانية والفطريات ومعظم
البكتيريا \omterm{def:g10:cells-common-unit:cell}{وخلايا} النبات غير الخضراء (\omterm{def:g10:cells-common-unit:cell}{خلايا} الجذر مثلًا).
\end{definition}

\begin{proposition}[التركيب الضوئي]\label{prop:g10:cell-metabolism:photosynthesis}
في الضوء، تمتص \omterm{def:g10:cells-common-unit:cell}{الخلية} ذات \omterm{def:g10:cells-common-unit:organelle}{البلاستيدات الخضراء} ثاني أكسيد الكربون
والماء، وتطلق الأكسجين، وتبني الغلوكوز، ثم تحوله إلى نشاء وسللوز
ودهون، وإلى حموض أمينية بنيتروجين معدني. وباختصار:
\[
6\,\mathrm{CO_2} + 6\,\mathrm{H_2O} \xrightarrow{\ \text{ضوء، يخضور}\ }
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2}.
\]
وتخزن طاقة الضوء في الروابط الكيميائية للغلوكوز.
\end{proposition}
\begin{proof}[شواهد]
معلق من الطحلب الأخضر \emph{Chlorella} في وعاء محكم مزود بمسبار
أكسجين يبين الأكسجين المنحل يرتفع باطراد حين يضاء المصباح ويهبط حين
يطفأ؛ ويتوقف الارتفاع إذا نزع ثاني أكسيد الكربون من الماء. والأوراق
التي تبقى في الظلام يومين لا تعود تعطي اللون الأزرق الأسود مع اليود،
والورقة المغطى نصفها برقاقة معدنية لا تتلون إلا في نصفها المكشوف:
فالنشاء يصنع حيث يقع الضوء. وثاني أكسيد الكربون المشع الذي يقدم لنبات
ينتهي في سكرياته في دقائق.
\end{proof}

\begin{example}[الأوغلينا، الأمران معًا]\label{ex:g10:cell-metabolism:euglena}
\emph{الأوغلينا} الوحيدة \omterm{def:g10:cells-common-unit:cell}{الخلية} لها بلاستيدات خضراء وتسبح بسوط. في
الضوء، في ماء معدني، تنمو: فهي \omterm{def:g10:cell-metabolism:autotrophy}{ذاتية التغذي}. وفي الظلام لا تبقى حية
إلا إذا أضيفت جزيئات عضوية إلى الماء تمتصها: فهي \omterm{def:g10:cell-metabolism:autotrophy}{غير ذاتية التغذي}.
وإذا طال بقاؤها في الظلام فقدت بلاستيداتها الخضراء؛ وإذا أعيدت إلى
الضوء صنعتها من جديد بعد بضعة أيام من الصبر. فاستقلابها يتوقف على
مورّثاتها، وهي تتيح الخيارين، وعلى محيطها، وهو ينتقي أحدهما.
\end{example}

\section{استخلاص الطاقة من الجزيئات العضوية}

\begin{proposition}[التنفس الخلوي]\label{prop:g10:cell-metabolism:respiration}
كل \omterm{def:g10:cells-common-unit:cell}{خلية}، \omterm{def:g10:cell-metabolism:autotrophy}{ذاتية التغذي} كانت أو غير ذاتية، تحصل على طاقة قابلة للاستعمال
بتفكيك جزيئات عضوية. وحين يتوفر الأكسجين يكون التفكيك تامًا --- وهذا
هو \emph{التنفس الخلوي}\index{تنفس خلوي}، في \omterm{def:g10:cells-common-unit:organelle}{المتقدرات}:
\[
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \longrightarrow
6\,\mathrm{CO_2} + 6\,\mathrm{H_2O} + \text{طاقة},
\]
أي نحو \qty{2870}{kJ} لكل مول من الغلوكوز (\qty{180}{g})، يقتنص منها
نحو 40\% في جزيء صغير هو \emph{ATP}\index{ATP}، تسحب منه كل عملية
تحتاج إلى طاقة في \omterm{def:g10:cells-common-unit:cell}{الخلية} --- حركة ونقل وتركيب --- ؛ أما الباقي فيتحرر
حرارةً.
\end{proposition}
\begin{proof}[شواهد]
الخميرة في وعاء محكم مزود بمسبار أكسجين لا تستهلك أكسجينًا وهي جائعة؛
أضف الغلوكوز فيهبط الأكسجين المنحل بنسبة ثابتة بينما يظهر ثاني أكسيد
الكربون، حتى ينفد الأكسجين. والبذور المنبتة في وعاء عازل ترفع حرارته
عدة درجات في يوم --- وهي حرارة التنفس --- بينما لا ترفعها البذور
المسلوقة. \omterm{def:g10:cells-common-unit:cell}{وخلايا} العضلة تحت المجهر الإلكتروني مكتظة \omterm{def:g10:cells-common-unit:organelle}{بالمتقدرات}، وسم
\omterm{def:g10:cells-common-unit:organelle}{المتقدرات} يوقف العضلة.
\end{proof}

\begin{definition}[التخمر]\label{def:g10:cell-metabolism:fermentation}
من دون أكسجين، تفكك بعض \omterm{def:g10:cells-common-unit:cell}{الخلايا} الغلوكوز تفكيكًا جزئيًا فقط، في
\omterm{def:g10:cells-common-unit:cell}{السيتوبلازم}، إلى جزيئات عضوية أصغر لا تزال تحمل معظم الطاقة: وهذا هو
\emph{التخمر}\index{تخمر}. فالخميرة تجري التخمر الكحولي،
\[
\mathrm{C_6H_{12}O_6} \longrightarrow 2\,\mathrm{C_2H_5OH} +
2\,\mathrm{CO_2} + \text{طاقة},
\]
\omterm{def:g10:cells-common-unit:cell}{وخلايا} العضلة الشحيحة الأكسجين، وكذلك بكتيريا اللبن الرائب، تجري
التخمر اللبني، فتحول الغلوكوز إلى حمض لبني. والتخمر يعطي من ATP لكل
غلوكوز نحو خمسة عشر ضعفًا أقل مما يعطيه التنفس.
\end{definition}

\begin{example}[الخميرة في الهواء وخارجه]\label{ex:g10:cell-metabolism:pasteur}
الخميرة التي يمرر فيها الهواء تستعمل الغلوكوز باقتصاد وتنمو سريعًا:
فهي تتنفس. والخميرة نفسها في خزان محكم تستعمل الغلوكوز بنهم، وتنمو
قليلًا، وتصنع الإيثانول: فهي تخمّر. وقد سمى باستور، الذي وصف ذلك سنة
1861، \omterm{def:g10:cell-metabolism:fermentation}{التخمر} «حياةً بلا هواء»؛ \omterm{def:g10:cells-common-unit:cell}{فالخلايا} تحوّل استقلابها إلى السبيل
الذي يتيحه محيطها، ولما كان \omterm{def:g10:cell-metabolism:fermentation}{التخمر} يعطي طاقة ضئيلة لكل غلوكوز، وجب أن
تحرق سكرًا أكثر بكثير لتحيا.
\end{example}

\section{ما الذي يقرر استقلاب الخلية}

\begin{proposition}[المورّثات والمحيط]\label{prop:g10:cell-metabolism:decides}
يتوقف \omterm{def:g10:cell-metabolism:metabolism}{استقلاب} \omterm{def:g10:cells-common-unit:cell}{الخلية} على أمرين: على \emph{عدتها} --- أي الإنزيمات
\omterm{def:g10:cells-common-unit:organelle}{والعضيات} التي تتيح لها مورّثاتها بناءها، حتى إن \omterm{def:g10:cells-common-unit:cell}{خلية} بلا يخضور لا
تستطيع \omterm{def:g10:cell-metabolism:autotrophy}{التركيب الضوئي} أبدًا --- وعلى \emph{محيطها} --- من ضوء وأكسجين
ومغذيات متوفرة --- وهو الذي يقرر أي السبل الممكنة يجري فعلًا. فالخميرة
متنفسة في الهواء ومخمّرة من دونه؛ و\emph{الأوغلينا} \omterm{def:g10:cell-metabolism:autotrophy}{ذاتية التغذي} في
الضوء وغير ذاتية في الظلام؛ \omterm{def:g10:cells-common-unit:cell}{وخلية} جذر نبات أخضر \omterm{def:g10:cell-metabolism:autotrophy}{غير ذاتية التغذي} طول
حياتها، تغذيها الأوراق.
\end{proposition}
\admitted

\begin{method}[قراءة أثر استقلابي]\label{met:g10:cell-metabolism:trace}
تعطي تجارب مسبار الأكسجين تركيزًا بدلالة الزمن.
\begin{enumerate}
  \item عيّن ما الذي تغير عند كل لحظة معلَّمة (إضاءة، إضافة ركيزة،
        نفاد أكسجين).
  \item اقرأ ميل كل قطعة مستقيمة: الارتفاع على الامتداد، بوحدة
        \unit{mg/L} في الدقيقة. فالميل الموجب يعني إنتاجًا صافيًا
        للأكسجين، والسالب استهلاكًا صافيًا.
  \item تذكّر أن \omterm{def:g10:cells-common-unit:cell}{الخلية} التي تجري \omterm{def:g10:cell-metabolism:autotrophy}{التركيب الضوئي} تتنفس أيضًا: فالميل
        في الضوء هو الإنتاج ناقص التنفس، والنسبة الحقيقية \omterm{def:g10:cell-metabolism:autotrophy}{للتركيب
        الضوئي} هي ميل الضوء زائد قيمة ميل الظلام.
  \item حوّل إذا طلب منك: عند هذه التراكيز، \qty{1}{mg/L} من الأكسجين
        المنحل هي $1/32$ مليمول في اللتر.
\end{enumerate}
\end{method}

\begin{example}[النسبة الحقيقية عند الطحلب]\label{ex:g10:cell-metabolism:truerate}
في شكل \emph{Chlorella} يرتفع الأكسجين \qty{2.5}{mg/L} في 5 دقائق من
الضوء (الميل $+0.5$) ويهبط \qty{0.5}{mg/L} في 5 دقائق من الظلام
(الميل $-0.1$). والتنفس يستهلك \qty{0.1}{mg/L} في الدقيقة في الضوء
كذلك، \omterm{def:g10:cell-metabolism:autotrophy}{فالتركيب الضوئي} ينتج $0.5 + 0.1 = \qty{0.6}{mg/L}$ في الدقيقة:
أي ستة أمثال ما يستهلكه التنفس. فالطحلب منتج صاف للأكسجين وللمادة
العضوية، ولذلك ينمو.
\end{example}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:cell-metabolism:2}
اكتب المعادلة الإجمالية \omterm{prop:g10:cell-metabolism:respiration}{للتنفس الخلوي} وللتركيب الضوئي. بأي معنى يكون

\begin{exercise}[$\star$]\label{exo:g10:cell-metabolism:3}
صنّف إلى ذاتي التغذي وغير ذاتي: فطر، وطحلب، \omterm{def:g10:cells-common-unit:cell}{وخلية} جذر جزرة، وكلوريلا،

\begin{exercise}[$\star$]\label{exo:g10:cell-metabolism:4}
في تجربة الخميرة والبالون، ما الغاز الذي ينفخ البالون، وما الناتج
السائل الذي يتراكم، وما الذي يفتقر إليه الدورق مما كان ليغير النتيجة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:cell-metabolism:5}
لماذا لم يعد الكبد المسلوق يرغي الماء الأكسجيني؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cell-metabolism:6}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cell-metabolism:8}

\begin{exercise}[$\star\star$]\label{exo:g10:cell-metabolism:9}

\begin{exercise}[$\star\star$]\label{exo:g10:cell-metabolism:11}

\begin{exercise}[$\star\star\star$]\label{exo:g10:cell-metabolism:12}

\section{مسألة: خزان صانع النبيذ}

\begin{problem}[{مسألة نهاية الأسبوع --- خزان من عصير العنب يترك
للخميرة: الكحول الذي سيحمله، والغاز الذي سيطلقه، والحرارة التي
سيصنعها، ولماذا يكون القبو مكانًا خطرًا}]
\label{pb:g10:cell-metabolism:1}
يحوي عصير العنب نحو \qty{200}{g} من السكر في اللتر، ونأخذه غلوكوزًا،
ولا أكسجين منحلًا فيه حالما يغلق الخزان. الكتل المولية: الغلوكوز
\qty{180}{g/mol}، والإيثانول $\mathrm{C_2H_5OH}$ \qty{46}{g/mol}،
وثاني أكسيد الكربون \qty{44}{g/mol}، والأكسجين \qty{32}{g/mol}.
وكثافة الإيثانول \qty{0.79}{g/mL}. ويشغل مول واحد من الغاز نحو
\qty{24}{L} عند حرارة القبو. وسعة الخزان \qty{1000}{L}.

\medskip
\noindent\textbf{الجزء الأول --- ما تصنعه الخميرة.}
\begin{enumerate}
  \item أي سبيل تسلكه الخميرة في الخزان المغلق، ولماذا؟ اكتب معادلته.
  \item كم مولًا من الغلوكوز يحوي لتر واحد من العصير؟
  \item احسب كتلة الإيثانول المنتجة لكل لتر، ثم حجمها، ثم درجة كحول
        النبيذ بالمئة حجمًا.
  \item احسب كتلة ثاني أكسيد الكربون المنتجة لكل لتر من العصير، وحجمها
        غازًا.
  \item ما حجم ثاني أكسيد الكربون المنطلق من الخزان كله؟ قارنه بحجم
        قبو أبعاده \qty{4}{m} في \qty{6}{m} في \qty{2.5}{m}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- لو كان للخميرة هواء.}
\begin{enumerate}[resume]
  \item اكتب المعادلة التي كانت الخميرة لتتبعها لو هوي العصير
        باستمرار.
  \item احسب كتلة الأكسجين اللازمة لتنفس سكر لتر واحد من العصير، وكتلة
        ثاني أكسيد الكربون المنطلقة.
  \item قارن ثاني أكسيد الكربون في السؤال 7 بمثيله في السؤال 4، وفسّر
        الفرق بدلالة مدى تفكيك الغلوكوز.
  \item التنفس يعطي نحو 30 من ATP لكل غلوكوز، \omterm{def:g10:cell-metabolism:fermentation}{والتخمر} 2. فإذا احتاجت
        الخميرة إلى العدد نفسه من ATP في الحالتين، ففي أي حالة يدوم
        السكر أطول، وبأي عامل؟
  \item فسّر لماذا يبعد صانع النبيذ الهواء مع ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- حرارة الخزان.}
يحرر \omterm{def:g10:cell-metabolism:fermentation}{التخمر} نحو \qty{70}{kJ} من الحرارة لكل مول من الغلوكوز؛ وتسخين
\qty{1}{kg} من الماء \qty{1}{\celsius} يحتاج إلى \qty{4.2}{kJ}.
\begin{enumerate}[resume]
  \item احسب الحرارة المحررة لكل لتر من العصير، ثم للخزان كله.
  \item لو لم يفلت شيء منها، فبكم ترتفع حرارة الخزان؟ (خذ العصير
        ماءً.)
  \item إنزيمات الخميرة تهدم فوق نحو \qty{40}{\celsius}. فانطلاقًا من
        \qty{20}{\celsius}، هل يبقى خزان غير مبرد سليمًا؟ وما الذي
        يعلمك إياه ذلك عن سبب تبريد الخزانات الكبيرة؟
  \item فسّر، باستعمال \cref{prop:g10:cell-metabolism:enzymes}، لماذا
        توقف الحرارة \omterm{def:g10:cell-metabolism:fermentation}{التخمر} مع أن السكر لا يزال موجودًا.
  \item كان التنفس ليحرر نحو \qty{2870}{kJ} لكل مول. فأين تبقى الطاقة
        التي \emph{لا} يحررها \omterm{def:g10:cell-metabolism:fermentation}{التخمر}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- القبو.}
\begin{enumerate}[resume]
  \item ثاني أكسيد الكربون أكثف من الهواء. فأين يتراكم في القبو،
        ولماذا كانت شمعة مشتعلة تحمل منخفضة اختبار سلامة تقليديًا؟
  \item \omterm{def:g10:cells-common-unit:cell}{خلايا} عامل حرمت الأكسجين تتحول إلى أي سبيل --- ولماذا لا
        يستطيع جسم الإنسان أن يحيا عليه أكثر من دقائق؟
  \item يتوقف \omterm{def:g10:cell-metabolism:fermentation}{التخمر} من تلقاء نفسه حين يبلغ الكحول نحو 15\% حجمًا.
        اقترح تفسيرًا بدلالة إنزيمات الخميرة.
  \item عجينة الخبز تستعمل الخميرة نفسها ساعة. فما الذي تحويه فقاعات
        اللباب، وأين ذهب الإيثانول بعد الخبز؟
  \item اذكر النتيجة: لكل لتر من العصير، درجة الكحول وحجم الغاز
        المنتج --- والحقيقة الاستقلابية الواحدة التي تفسر الأمرين.
\end{enumerate}
\end{problem}
