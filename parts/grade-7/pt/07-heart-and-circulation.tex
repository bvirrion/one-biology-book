\chapter{O coração e a circulação}\label{ch:g7:heart-and-circulation}

O \cref{ch:g4:heart-and-blood} apresentou a bomba e as rodadas de
entrega dela; desde então o \omterm{prop:g4:journey-of-food:absorb}{sangue} ganhou responsabilidades ---
\omterm{def:g7:digestion-and-nutrients:families}{nutrientes} nas \omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades}
(\cref{prop:g7:digestion-and-nutrients:absorption}), \omterm{def:g7:what-is-respiration:respiration}{oxigênio} nos
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{alvéolos} (\cref{prop:g7:lungs-and-gas-exchange:numbers}), contas em
cada \omterm{def:g3:bones-and-muscles:muscle}{músculo} em atividade (\cref{prop:g7:organs-at-work:bill}). É hora
de mapear direito a rede de estradas: os quatro cômodos do \omterm{def:g4:heart-and-blood:heart}{coração}, os
dois circuitos e os três tipos de vaso.

\section{Os quatro cômodos do coração}

\begin{definition}[Átrios e ventrículos]\label{def:g7:heart-and-circulation:rooms}
Os dois lados do \omterm{def:g4:heart-and-blood:heart}{coração} (\cref{def:g4:heart-and-blood:heart}) têm
dois cômodos cada: em cima um \emph{átrio}\index{átrio}, que recebe o
\omterm{prop:g4:journey-of-food:absorb}{sangue} que chega, e embaixo um \emph{ventrículo}\index{ventrículo} ---
o cômodo de bombeamento, de parede grossa ---, que empurra o \omterm{prop:g4:journey-of-food:absorb}{sangue}
para fora. Entre eles e depois deles ficam as
\emph{valvas}\index{valva (cardíaca)} --- portas de mão única --- que se
fecham de estalo atrás de cada onda; os estalos em par são exatamente
o ``tum-tá'' que um estetoscópio (ou um \omterm{def:g1:the-five-senses:sense}{ouvido} encostado num peito)
escuta.
\end{definition}

\begin{proposition}[Uma batida, em ordem]\label{prop:g7:heart-and-circulation:beat}
Cada batida executa a mesma sequência nos dois lados ao mesmo tempo:
os \omterm{def:g7:heart-and-circulation:rooms}{átrios} se contraem, completando os \omterm{def:g7:heart-and-circulation:rooms}{ventrículos}; os \omterm{def:g7:heart-and-circulation:rooms}{ventrículos} se
contraem, empurrando o \omterm{prop:g4:journey-of-food:absorb}{sangue} para fora --- o lado direito rumo aos
\omterm{def:g7:how-animals-breathe:four}{pulmões}, o lado esquerdo rumo ao corpo --- enquanto as \omterm{def:g7:heart-and-circulation:rooms}{valvas} de
entrada batem (tum); os \omterm{def:g7:heart-and-circulation:rooms}{ventrículos} relaxam, as \omterm{def:g7:heart-and-circulation:rooms}{valvas} de saída batem
(tá), e os cômodos se enchem de novo. E de novo, $100000$ vezes por
dia (\cref{rem:g4:heart-and-blood:rest}).
\end{proposition}
\admitted

\section{Dois circuitos, três vasos}

\begin{definition}[A circulação dupla]\label{def:g7:heart-and-circulation:double}
A rede de estradas do \omterm{prop:g4:journey-of-food:absorb}{sangue} são dois circuitos por uma bomba só (os
dois lados da \cref{def:g4:heart-and-blood:heart}, agora como
circuitos):
\begin{itemize}
  \item o \emph{circuito pulmonar}: \omterm{def:g7:heart-and-circulation:rooms}{ventrículo} direito $\rightarrow$
        \omterm{def:g7:how-animals-breathe:four}{pulmões} (carregando \omterm{def:g7:what-is-respiration:respiration}{oxigênio}, descarregando \omterm{def:g7:what-is-respiration:respiration}{gás carbônico})
        $\rightarrow$ \omterm{def:g7:heart-and-circulation:rooms}{átrio} esquerdo;
  \item o \emph{circuito corporal}: \omterm{def:g7:heart-and-circulation:rooms}{ventrículo} esquerdo $\rightarrow$
        o corpo inteiro (entregando, recolhendo) $\rightarrow$ \omterm{def:g7:heart-and-circulation:rooms}{átrio}
        direito.
\end{itemize}
Cada gota alterna os circuitos: \omterm{def:g7:how-animals-breathe:four}{pulmões}, corpo, \omterm{def:g7:how-animals-breathe:four}{pulmões}, corpo --- e é
por isso que o \omterm{def:g7:heart-and-circulation:rooms}{ventrículo} esquerdo, empurrando pelo circuito corporal,
muito maior, é o cômodo de parede mais grossa do \omterm{def:g4:heart-and-blood:heart}{coração}.
\end{definition}

\begin{definition}[Artérias, capilares, veias]\label{def:g7:heart-and-circulation:vessels}
Três tipos de vaso formam cada circuito
(\cref{def:g4:heart-and-blood:vessels}, agora com nome):
\begin{itemize}
  \item as \emph{artérias}\index{artéria}: de parede grossa,
        elásticas, levando o \omterm{prop:g4:journey-of-food:absorb}{sangue} para \emph{longe} do \omterm{def:g4:heart-and-blood:heart}{coração} sob
        pressão --- o \omterm{prop:g4:heart-and-blood:pulse}{pulso}
        (\cref{prop:g4:heart-and-blood:pulse}) é o esticamento delas a
        cada onda;
  \item os \emph{capilares}\index{capilar}: os \omterm{def:g4:heart-and-blood:vessels}{vasos} finos como fio de
        cabelo que se enfiam em cada órgão --- de paredes com a
        espessura de uma \omterm{def:g5:first-look-at-cells:cell}{célula}, onde acontecem \emph{todas} as
        trocas: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alvéolos}, \omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades}, \omterm{def:g3:bones-and-muscles:muscle}{músculos}, \omterm{def:g1:the-five-senses:sense}{pele};
  \item as \emph{veias}\index{veia}: largas, de parede mais fina,
        devolvendo o \omterm{prop:g4:journey-of-food:absorb}{sangue} \emph{ao} \omterm{def:g4:heart-and-blood:heart}{coração} a baixa pressão, com
        \omterm{def:g7:heart-and-circulation:rooms}{valvas} de mão única próprias --- as linhas azuladas da
        \cref{rem:g4:heart-and-blood:see}.
\end{itemize}
\end{definition}

\begin{example}[Uma gota, uma volta completa]\label{ex:g7:heart-and-circulation:tour}
A rodada de entrega do \cref{ex:g4:heart-and-blood:round}, com os
nomes do mapa: \omterm{def:g7:heart-and-circulation:rooms}{ventrículo} direito --- \omterm{def:g7:heart-and-circulation:vessels}{capilares} do \omterm{def:g7:how-animals-breathe:four}{pulmão} (\omterm{def:g7:what-is-respiration:respiration}{oxigênio} a
bordo nos \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alvéolos}, vermelho vivo) --- \omterm{def:g7:heart-and-circulation:rooms}{átrio} esquerdo, \omterm{def:g7:heart-and-circulation:rooms}{ventrículo}
esquerdo --- \omterm{def:g7:heart-and-circulation:vessels}{artéria} --- \omterm{def:g7:heart-and-circulation:vessels}{capilares} de um \omterm{def:g3:bones-and-muscles:muscle}{músculo} da coxa (\omterm{def:g7:what-is-respiration:respiration}{oxigênio} e
\omterm{prop:g7:organs-at-work:bill}{glicose} descem, \omterm{def:g7:what-is-respiration:respiration}{gás carbônico} embarca, mais escuro) --- \omterm{def:g7:heart-and-circulation:vessels}{veia}, com as
\omterm{def:g7:heart-and-circulation:rooms}{valvas} facilitando a subida --- \omterm{def:g7:heart-and-circulation:rooms}{átrio} direito. Dois circuitos, em menos
de um minuto, sem nenhuma conversão errada disponível: as \omterm{def:g7:heart-and-circulation:rooms}{valvas}
cuidam disso.
\end{example}

\begin{remark}[Por que os capilares são o ponto]\label{rem:g7:heart-and-circulation:capillaries}
As \omterm{def:g7:heart-and-circulation:vessels}{artérias} e as \omterm{def:g7:heart-and-circulation:vessels}{veias} são só rodovias; toda entrega de verdade ---
toda troca que este ano estudou --- acontece através das paredes dos
\omterm{def:g7:heart-and-circulation:vessels}{capilares}. A especificação da
\cref{prop:g7:how-animals-breathe:spec} mora aqui também: paredes de
uma \omterm{def:g5:first-look-at-cells:cell}{célula} (finas), enfiando-se em cada tecido ao longo de dezenas de
milhares de quilômetros (\cref{def:g4:heart-and-blood:vessels} ---
grandes), banhadas pela \omterm{def:g6:exploring-our-environment:conditions}{umidade} do corpo. O \omterm{def:g4:heart-and-blood:heart}{coração} existe para manter
o \omterm{prop:g4:journey-of-food:absorb}{sangue} circulando \emph{pelos \omterm{def:g7:heart-and-circulation:vessels}{capilares}}; todo o resto é encanamento
de ida e de volta.
\end{remark}

\section{Ler e conservar a rede}

\begin{example}[Medidas à beira da estrada]\label{ex:g7:heart-and-circulation:measures}
Três leituras cotidianas da rede: o \emph{\omterm{prop:g4:heart-and-blood:pulse}{pulso}} --- o esticamento das
\omterm{def:g7:heart-and-circulation:vessels}{artérias}, contando as ondas
(\cref{met:g4:heart-and-blood:pulse}); a \emph{pressão arterial}, os
dois números do manguito --- a pressão nas \omterm{def:g7:heart-and-circulation:vessels}{artérias} na onda e entre as
ondas; e o \emph{tum-tá} do estetoscópio --- as portas das \omterm{def:g7:heart-and-circulation:rooms}{valvas}
ouvidas se fechando. Três janelas, sem nenhum corte.
\end{example}

\begin{proposition}[Conservar a rede]\label{prop:g7:heart-and-circulation:keep}
Os inimigos e os amigos de longo prazo da rede são conhecidos:
\begin{itemize}
  \item o \emph{exercício} fortalece a bomba
        (\cref{prop:g7:organs-at-work:training}) e mantém as rodovias
        flexíveis;
  \item a \emph{alimentação equilibrada}
        (\cref{met:g3:balanced-diet:check}) protege as paredes das
        \omterm{def:g7:heart-and-circulation:vessels}{artérias}, que um \omterm{prop:g4:journey-of-food:absorb}{sangue} rico demais aos poucos entope e
        enrijece;
  \item a \emph{fumaça} ataca os \omterm{def:g4:heart-and-blood:vessels}{vasos} além dos \omterm{def:g7:how-animals-breathe:four}{pulmões}
        (\cref{prop:g7:lungs-and-gas-exchange:smoke}) --- estreitando,
        enrijecendo, roubando assentos;
  \item o abastecimento do próprio \omterm{def:g4:heart-and-blood:heart}{coração} --- ele se alimenta pelas
        arteriazinhas dele, e não pelo \omterm{prop:g4:journey-of-food:absorb}{sangue} dos cômodos dele ---
        depende dos três itens acima.
\end{itemize}
Os hábitos da \cref{prop:g5:healthy-choices:contract} são,
concretamente, manutenção de \omterm{def:g4:heart-and-blood:vessels}{vasos}.
\end{proposition}
\admitted

\begin{method}[Situar qualquer órgão na rede]\label{met:g7:heart-and-circulation:map}
Para qualquer órgão que este livro encontrar:
\begin{enumerate}
  \item situe o órgão no circuito corporal: \omterm{def:g7:heart-and-circulation:vessels}{artéria} de entrada, \omterm{def:g7:heart-and-circulation:vessels}{veia}
        de saída, \omterm{def:g7:heart-and-circulation:vessels}{capilares} por dentro;
  \item liste o que atravessa as paredes dos \omterm{def:g7:heart-and-circulation:vessels}{capilares} dele, para
        dentro e para fora (\omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades}: \omterm{def:g7:digestion-and-nutrients:families}{nutrientes} para dentro;
        \omterm{def:g3:bones-and-muscles:muscle}{músculo}: combustível e \omterm{def:g7:what-is-respiration:respiration}{oxigênio} para dentro, \omterm{def:g7:what-is-respiration:respiration}{gás carbônico}
        para fora; \omterm{def:g1:the-five-senses:sense}{pele}: calor para fora);
  \item anote a ligação especial dele, se houver (o \omterm{prop:g4:journey-of-food:absorb}{sangue} do
        intestino passa antes pelo \omterm{prop:g7:digestion-and-nutrients:absorption}{fígado}; os \omterm{def:g7:how-animals-breathe:four}{pulmões} ficam num
        circuito próprio);
  \item confira os fluxos com o trabalho do órgão --- o diagrama da
        rede deve recontar o capítulo daquele órgão.
\end{enumerate}
\end{method}

\section{Problema: A manhã da cardiologista}

\begin{problem}[{Problema de fim de semana --- quatro consultas, um
mapa da rede}]\label{pb:g7:heart-and-circulation:1}
Acompanhe uma cardiologista por uma manhã de quatro consultas
(inventadas), com o mapa na mão.

\medskip\noindent\textbf{Parte I --- A referência
saudável.}
\begin{enumerate}
  \item A paciente 1, uma \omterm{def:g3:stages-of-life:stages}{adolescente} em forma, recebe as três
        leituras à beira da estrada do
        \cref{ex:g7:heart-and-circulation:measures}. Diga o nome de
        cada uma e que parte da maquinaria cada uma lê.
  \item O \omterm{prop:g4:heart-and-blood:pulse}{pulso} de repouso dela é $58$. A cardiologista sorri. Por
        quê (\cref{exo:g4:heart-and-blood:11})?
  \item O tum-tá dela é nítido e duplo. O que exatamente estalou,
        duas vezes, em cada batida?
  \item Esboce o caminho do \omterm{prop:g4:journey-of-food:absorb}{sangue} dela, do \omterm{def:g7:heart-and-circulation:rooms}{átrio} direito pelos dois
        circuitos e de volta, em oito paradas com nome.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Três problemas.}
\begin{enumerate}[resume]
  \item O paciente 2 tem uma valva com vazamento entre o \omterm{def:g7:heart-and-circulation:rooms}{átrio} e o
        \omterm{def:g7:heart-and-circulation:rooms}{ventrículo} esquerdos: parte do \omterm{prop:g4:journey-of-food:absorb}{sangue} reflui a cada batida,
        com um sopro sibilante. Preveja as consequências para as
        entregas do circuito corporal e para o fôlego do paciente.
  \item O paciente 3, fumante pesado e de alimentação rica, tem
        \omterm{def:g7:heart-and-circulation:vessels}{artérias} entupidas e enrijecidas. Que leituras da consulta 1
        vão ter piorado, e por que o \omterm{def:g7:heart-and-circulation:rooms}{ventrículo} esquerdo faz hora
        extra?
  \item O entupimento do paciente 3 inclui as arteriazinhas de
        alimentação do próprio \omterm{def:g4:heart-and-blood:heart}{coração}. Explique numa frase a
        preocupação especial da cardiologista.
  \item O paciente 4 tem \omterm{def:g1:my-body:joint}{tornozelos} inchados depois de dias longos em
        pé: o retorno das \omterm{def:g7:heart-and-circulation:vessels}{veias}, morro acima, está sofrendo. Que
        estruturas estão falhando, e por que caminhar --- com os
        \omterm{def:g3:bones-and-muscles:muscle}{músculos} da panturrilha espremendo as \omterm{def:g7:heart-and-circulation:vessels}{veias} --- alivia?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O cartaz da parede.}
\begin{enumerate}[resume]
  \item O cartaz da clínica mostra o circuito duplo. Explique a um
        paciente na espera por que o \omterm{prop:g4:journey-of-food:absorb}{sangue} precisa visitar os \omterm{def:g7:how-animals-breathe:four}{pulmões}
        entre duas voltas pelo corpo.
  \item Acrescente a legenda dos \omterm{def:g4:heart-and-blood:vessels}{vasos}: \omterm{def:g7:heart-and-circulation:vessels}{artéria}, \omterm{def:g7:heart-and-circulation:vessels}{capilar}, \omterm{def:g7:heart-and-circulation:vessels}{veia} --- uma
        linha para cada, sobre construção, pressão e trabalho.
  \item O rodapé do cartaz diz: ``todo tratamento serve aos
        \omterm{def:g7:heart-and-circulation:vessels}{capilares}''. Justifique isso com a
        \cref{rem:g7:heart-and-circulation:capillaries}.
  \item Encerre a manhã com a receita de prevenção que a
        cardiologista entrega a todo paciente --- três hábitos, cada
        um ligado ao mecanismo dele na
        \cref{prop:g7:heart-and-circulation:keep}.
\end{enumerate}
