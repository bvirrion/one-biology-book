\chapter{Caring for Living Things}\label{ch:g1:caring}

\omterm{def:g1:living-nonliving:living}{Living things} need care. \omterm{def:g1:plants-animals:plant}{Plants} need \omterm{def:g1:caring:needs}{water} and light. \omterm{def:g1:plants-animals:animal}{Animals} need
\omterm{def:g1:caring:needs}{food}, \omterm{def:g1:caring:needs}{water} and a safe \omterm{def:g1:caring:needs}{place}. People need kindness, rest and
healthy habits. Caring is part of biology in everyday life.

\section{What living things need}

\omimg{caring-for-life.jpg}{0.55\linewidth}{Caring for \omterm{def:g1:plants-animals:plant}{plants} and \omterm{def:g1:plants-animals:animal}{animals}: \omterm{def:g1:caring:needs}{water}, \omterm{def:g1:caring:needs}{food}, \omterm{def:g1:caring:needs}{shelter} and kindness.}

\begin{definition}[Needs of living things]\label{def:g1:caring:needs}
\omterm{def:g1:living-nonliving:living}{Living things} share basic \emph{needs}\index{need (living thing)}:
\begin{itemize}
  \item \emph{water}\index{water} to drink or take up;
  \item \emph{food}\index{food} (\omterm{def:g1:plants-animals:plant}{plants} make food; \omterm{def:g1:plants-animals:animal}{animals} eat);
  \item \emph{air}\index{air} to breathe;
  \item a safe \emph{place}\index{shelter} to live;
  \item the right temperature --- not too hot, not too cold.
\end{itemize}
\end{definition}

\section{Caring for plants and animals}

\begin{method}[Care for a plant]\label{met:g1:caring:plant}
\begin{enumerate}
  \item Give it light (a sunny window or garden).
  \item \omterm{def:g1:caring:needs}{Water} when the soil is dry --- not a flood every day.
  \item Do not break stems or pick all the leaves.
  \item Keep it in a pot or bed with soil.
\end{enumerate}
\end{method}

\begin{method}[Care for a pet]\label{met:g1:caring:pet}
\begin{enumerate}
  \item Give clean \omterm{def:g1:caring:needs}{water} every day.
  \item Give the right \omterm{def:g1:caring:needs}{food} (ask an adult what and how much).
  \item Give a clean bed or cage and time to move.
  \item Be gentle: soft voice, soft \omterm{def:g1:my-body:parts}{hands}.
\end{enumerate}
\end{method}

\begin{example}[Class plant]\label{ex:g1:caring:class}
The class has a pot \omterm{def:g1:plants-animals:plant}{plant}. One child \omterm{def:g1:caring:needs}{waters} it on Monday. Another
turns it toward the light on Wednesday. Everyone watches new leaves
grow. Care is a team job.
\end{example}

\begin{example}[Kind hands]\label{ex:g1:caring:kind}
Holding a chick too tight can hurt it. Shouting at a dog can scare
it. Gentle care keeps \omterm{def:g1:plants-animals:animal}{animals} calm and safe.
\end{example}
