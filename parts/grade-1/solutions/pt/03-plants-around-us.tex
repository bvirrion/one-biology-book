\section*{Capítulo \ref{ch:g1:plants-around-us} --- As plantas ao nosso redor}

\begin{solution}{exo:g1:plants-around-us:1}
\omterm{def:g1:plants-around-us:parts}{Raízes} (debaixo da terra), \omterm{def:g1:plants-around-us:parts}{caule}, \omterm{def:g1:plants-around-us:parts}{folhas}, \omterm{def:g1:plants-around-us:parts}{flor}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:2}
(a) As \omterm{def:g1:plants-around-us:parts}{raízes}. (b) As \omterm{def:g1:plants-around-us:parts}{folhas}. (c) A \omterm{def:g1:plants-around-us:parts}{flor}. (d) O \omterm{def:g1:plants-around-us:parts}{caule}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:3}
Ela nasceu de uma semente, se alimenta (água da terra, luz do céu),
cresce, faz sementes que viram margaridas novas e morre um dia.
Todos os sinais da vida estão lá; andar não é um deles.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:4}
O \omterm{def:g1:plants-around-us:parts}{caule} da \omterm{def:g1:plants-around-us:tree}{árvore} é o tronco; as \omterm{def:g1:plants-around-us:parts}{folhas} são carregadas pelos \omterm{def:g1:plants-around-us:tree}{galhos}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:5}
Água, luz e um lugar para as \omterm{def:g1:plants-around-us:parts}{raízes} --- quase sempre a terra.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:6}
A rega. A \omterm{def:g1:plants-around-us:plant}{planta} murchou por falta de água; poucas horas depois de
ser regada, as \omterm{def:g1:plants-around-us:parts}{folhas} dela vão voltar a ficar em pé.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:7}
Iguais: as duas são \omterm{def:g1:plants-around-us:plant}{plantas}, com \omterm{def:g1:plants-around-us:parts}{raízes}, \omterm{def:g1:plants-around-us:parts}{caule}, \omterm{def:g1:plants-around-us:parts}{folhas} e \omterm{def:g1:plants-around-us:parts}{flores}, e
as duas precisam de água e de luz. Diferentes: o \omterm{def:g1:plants-around-us:parts}{caule} da \omterm{def:g1:plants-around-us:tree}{árvore} é
um tronco de madeira, enquanto o \omterm{def:g1:plants-around-us:parts}{caule} da margarida é macio e verde,
e a \omterm{def:g1:plants-around-us:tree}{árvore} é bem maior e vive bem mais tempo.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:8}
A \omterm{def:g1:plants-around-us:plant}{planta} da janela deve estar verde e em pé; a do armário, pálida,
esticada e fraca, mesmo tendo sido regada. A necessidade que faltou
é a luz --- só água não basta.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:9}
Por exemplo: a \omterm{def:g1:plants-around-us:plant}{planta} cresce --- mais alta semana após semana --- e
se vira ou se inclina para a luz, como o girassol acompanhando o
dia. Devagar não é o mesmo que não fazer nada.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:10}
A \omterm{def:g1:plants-around-us:parts}{raiz}. ``Debaixo da terra'' é o lugar das \omterm{def:g1:plants-around-us:parts}{raízes} na figura --- a
cenoura que a gente come é uma \omterm{def:g1:plants-around-us:parts}{raiz} laranja grande e gorda que a
\omterm{def:g1:plants-around-us:plant}{planta} encheu de alimento.
\end{solution}
