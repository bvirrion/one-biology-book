\chapter{النوكليوتيدات والحموض النووية}\label{ch:b1:nucleic-acids}

صبّ إيثانولًا باردًا على محلول من \omterm{def:b1:cell-unit-of-life:cell}{خلايا} مكسورة فتظهر سحابة بيضاء من
الألياف يمكن لفها على قضيب زجاجي: إنه \omterm{def:b1:nucleic-acids:chain}{DNA}، الجزيء الذي يحمل كل تعليمة
من تعليمات \omterm{def:b1:organism-environment:organism}{الكائن الحي}، مرئيًا بالعين المجردة. مترانِ منه مطويان في
كل نواة بشرية؛ وهو مكتوب كله بأبجدية من أربعة حروف، يُقرأ في اتجاه
واحد، ويُنسخ بفصل خصلتين تتوافقان توافق اليد وقفازها. يصف هذا الفصل
\omterm{def:b1:nucleic-acids:nucleotide}{النوكليوتيدات} --- وهي حروف وعملة طاقة ومرافقات إنزيمية في آن ---
والسلاسل التي تكوّنها، \omterm{thm:b1:nucleic-acids:helix}{واللولب المزدوج} وشواهده، وما يحدث حين يُصهر
ثم يُعاد التئامه، وأنواع \omterm{def:b1:nucleic-acids:chain}{RNA} التي تحمل رسالته إلى \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.

\section{النوكليوتيدات}

\begin{definition}[النوكليوتيد]\label{def:b1:nucleic-acids:nucleotide}
\emph{النوكليوتيد}\index{نوكليوتيد} مكون من ثلاثة أجزاء:
\emph{قاعدة}\index{قاعدة نتروجينية} نتروجينية --- وهي
\emph{بورين}\index{بورين} (أدنين A، وغوانين G: حلقتان ملتحمتان) أو
\emph{بيريميدين}\index{بيريميدين} (سيتوزين C، وثيمين T، ويوراسيل U:
حلقة واحدة)؛ وسكر من خمس ذرات كربون، هو \emph{الريبوز} في \omterm{def:b1:nucleic-acids:chain}{RNA} أو
\emph{الريبوز 2$'$ منقوص الأكسجين} في \omterm{def:b1:nucleic-acids:chain}{DNA} (وهو يفتقر إلى الهيدروكسيل
على الكربون $2'$)؛ وزمرة \emph{فوسفات} إلى ثلاث زمر على الكربون
$5'$ من السكر. والقاعدة مرتبطة بالكربون $1'$؛ وكربونات السكر مرقمة
بعلامات تفريق لتمييزها من كربونات القاعدة. والقاعدة مع السكر
\emph{نوكليوزيد} (أدنوزين، وغوانوزين، وسيتيدين، وثيميدين، ويوريدين)؛
ومع الفوسفات نوكليوتيد (AMP، وADP، \omterm{def:b1:water-small-molecules:atp}{وATP} وأشباهها).
\end{definition}

\begin{proposition}[للنوكليوتيدات ثلاث وظائف]\label{prop:b1:nucleic-acids:threejobs}
\omterm{def:b1:nucleic-acids:nucleotide}{النوكليوتيدات}، فضلًا عن كونها وحيدات \omterm{def:b1:nucleic-acids:chain}{DNA} \omterm{def:b1:nucleic-acids:chain}{وRNA}، هي \emph{\omterm{def:b1:water-small-molecules:atp}{عملة الطاقة}}
في \omterm{def:b1:cell-unit-of-life:cell}{الخلية} --- \omterm{def:b1:water-small-molecules:atp}{ATP} وGTP، اللذان تدفع روابطهما الفوسفوأنهيدريدية
(\cref{ch:b1:water-small-molecules}) ثمن التركيب والنقل والحركة؛
و\emph{مرافقاتها الإنزيمية} --- $\mathrm{NAD^+}$ و$\mathrm{NADP^+}$
وFAD، التي تحمل الإلكترونات في التنفس والتركيب الضوئي
(\cref{ch:b1:respiration-fermentation})، \omterm{def:b1:water-small-molecules:ph}{والمرافق} A، الذي يحمل زمر
الأسيل؛ و\emph{رسلها} --- AMP الحلقي، الذي ينقل إشارات الهرمونات داخل
\omterm{def:b1:cell-unit-of-life:cell}{الخلايا}. وكلها \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيدات} أدنين أو مبنية عليها: فالجزيء نفسه الذي
يهجئ الرسالة يمد تنفيذها بالطاقة.
\end{proposition}

\section{سلسلة عديد النوكليوتيد}

\begin{definition}[الرابطة الفوسفوديإستيرية والاتجاه]\label{def:b1:nucleic-acids:chain}
تُوصل \omterm{def:b1:nucleic-acids:nucleotide}{النوكليوتيدات} في \emph{عديد نوكليوتيد}\index{عديد نوكليوتيد}
\emph{بروابط فوسفوديإستيرية}\index{رابطة فوسفوديإستيرية}: إذ يُربط
الفوسفات على الكربون $5'$ من \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيد} بهيدروكسيل $3'$ من التالي.
فيكون للسلسلة إذن \emph{هيكل} سكري فوسفاتي، منتظم ومشحون سالبًا
(شحنة واحدة لكل فوسفات)، تبرز منه القواعد تسلسلًا؛ ويكون لها اتجاه:
\emph{نهاية $5'$}\index{5end@نهاية $5'$} تحمل فوسفاتًا حرًا،
و\emph{نهاية $3'$}\index{3end@نهاية $3'$} تحمل هيدروكسيلًا حرًا.
وتُكتب التسلسلات وتُقرأ من $5' \to 3'$، وهو الاتجاه الذي تُصنَّع فيه
السلاسل. و\emph{DNA}\index{DNA} (\omterm{def:b1:water-small-molecules:ph}{الحمض} النووي الريبي منقوص الأكسجين)
يستعمل الريبوز منقوص الأكسجين والقواعد A وG وC وT؛
و\emph{RNA}\index{RNA} (\omterm{def:b1:water-small-molecules:ph}{الحمض} النووي الريبي) يستعمل الريبوز وA وG وC
وU.
\end{definition}

\begin{example}[RNA هش وDNA متين]\label{ex:b1:nucleic-acids:fragile}
يستطيع هيدروكسيل $2'$ في الريبوز أن يهاجم \omterm{def:b1:nucleic-acids:chain}{الرابطة الفوسفوديإستيرية}
المجاورة: ففي قلوي خفيف يُحلمه \omterm{def:b1:nucleic-acids:chain}{RNA} إلى \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيدات} في غضون ساعات،
بينما ينجو \omterm{def:b1:nucleic-acids:chain}{DNA}، وهو يفتقر إلى ذلك الهيدروكسيل، من الغليان في القلوي
وقد استُخرج سليمًا من عظام عمرها عشرات الآلاف من السنين. واليوراسيل في
\omterm{def:b1:nucleic-acids:chain}{RNA} ثيمين بلا زمرة ميثيل؛ \omterm{def:b1:nucleic-acids:chain}{وDNA} يستعمل الثيمين حتى يمكن التعرف على
السيتوزين، الذي يفقد زمرته الأمينية ببطء فيصير يوراسيلًا، بوصفه تالفًا
فيُصلَح --- فاليوراسيل في \omterm{def:b1:nucleic-acids:chain}{DNA} خطأ دائمًا.
\end{example}

\section{اللولب المزدوج}

\begin{proposition}[قانونا شارغاف]\label{prop:b1:nucleic-acids:chargaff}
\index{قانونا شارغاف}
في \omterm{def:b1:nucleic-acids:chain}{DNA} أي \omterm{def:b1:organism-environment:organism}{كائن حي} تساوي كمية الأدنين كمية الثيمين وتساوي كمية
الغوانين كمية السيتوزين ($A = T$ و$G = C$، ومن ثم البورينات $=$
البيريميدينات)، بينما تتغير النسبة $(G + C)/(A + T)$ من نوع إلى نوع
(من \qty{25}{\%} إلى \qty{75}{\%} $G + C$).
\end{proposition}
\begin{proof}[شواهد]
حلمه شارغاف (1950) \omterm{def:b1:nucleic-acids:chain}{DNA} من \omterm{def:b1:organism-environment:organism}{كائنات} كثيرة وفصل القواعد بالاستشراب،
وقاس كلًا منها بامتصاصها فوق البنفسجي. فصدقت المساواتان في حدود الخطأ
التجريبي عند كل \omterm{def:b1:nucleic-acids:chain}{DNA}؛ ولم يصدق محتوى $G + C$، وكان هو نفسه في كل \omterm{def:b1:body-plans-tissues:tissue}{نسيج}
من الكائن الواحد.
\end{proof}

\begin{theorem}[لولب واطسون وكريك المزدوج]\label{thm:b1:nucleic-acids:helix}
\index{لولب مزدوج}\index{زوج قواعد}\index{متعاكسا الاتجاه}
\omterm{def:b1:nucleic-acids:chain}{DNA} سلسلتا \omterm{def:b1:nucleic-acids:chain}{عديد نوكليوتيد} ملتفتان حول محور مشترك في \emph{\omterm{thm:b1:nucleic-acids:helix}{لولب
مزدوج}} يميني: الهيكلان السكريان الفوسفاتيان في الخارج، والقواعد في
الداخل، مسطحة مكدسة عموديًا على المحور. والسلسلتان
\emph{متعاكستا الاتجاه} (إحداهما تجري $5' \to 3'$ صعودًا والأخرى
هبوطًا) و\emph{متتامتان}: فمقابل كل A ثيمين T، تمسكهما رابطتان
هيدروجينيتان، ومقابل كل G سيتوزين C، تمسكهما ثلاث؛ وكل \emph{\omterm{thm:b1:nucleic-acids:helix}{زوج
قواعد}} كهذا له العرض نفسه، فيكون اللولب منتظمًا مهما كان التسلسل. وفي
الصورة B الشائعة يبلغ عرض اللولب \qty{2}{nm}، ويرتفع \qty{0.34}{nm}
لكل \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} و\qty{3.4}{nm} لكل دورة من عشرة أزواج، ويُظهر
\emph{أخدودًا كبيرًا} واسعًا و\emph{أخدودًا صغيرًا} ضيقًا تنكشف فيهما
حواف القواعد للبروتينات.
\end{theorem}
\begin{proof}[شواهد]
أظهرت صور فرانكلن لانعراج الأشعة السينية على ألياف \omterm{def:b1:nucleic-acids:chain}{DNA} (1952) صليب
اللولب بتكرار قدره \qty{3.4}{nm} على امتداد المحور، وانعكاسًا قويًا
عند \qty{0.34}{nm} (وحدات مسطحة مكدسة)، وقطرًا قدره \qty{2}{nm}؛ ولا
بد أن تكون الفوسفات، وهي أثقل الذرات، في الخارج. ثم وجد واطسون وكريك
(1953) أن اقتران \omterm{def:b1:nucleic-acids:nucleotide}{بورين} \omterm{def:b1:nucleic-acids:nucleotide}{ببيريميدين} --- وA مع T وG مع C وحدها ---
يعطي أزواجًا متساوية العرض توافق قطر \qty{2}{nm}، فيفسر قانوني
شارغاف، وأن سلسلتين متعاكستي الاتجاه مقترنتين هكذا تبنيان لولبًا
منتظمًا بأبعاد الليف. وتنبأ النموذج بأن كل سلسلة قالب للأخرى؛ وأكد
ذلك إثبات مزلسون وستال للتضاعف نصف المحافظ (المستعاد من كتاب المرحلة
الثانوية، وفي \cref{ch:b1:replication-mitosis}).
\end{proof}

\begin{example}[قراءة خصلة]\label{ex:b1:nucleic-acids:complement}
إذا قرأت إحدى الخصلتين $5'$-ATGGCATTC-$3'$، كانت شريكتها،
مكتوبةً من $5' \to 3'$، هي $5'$-GAATGCCAT-$3'$: فتمّم كل \omterm{def:b1:nucleic-acids:nucleotide}{قاعدة}، ثم
اعكس الترتيب. وهي تسعة أزواج، أي \qty{3.1}{nm} من اللولب، واثنتان
وعشرون \omterm{def:b1:water-small-molecules:water}{رابطة هيدروجينية} (خمس G--C وأربع A--T). وصبغي بشري من
\num{2.5e8} زوج طوله \qty{8.5}{cm} في هذه الصورة ولا بد أن يُطوى
مئة ألف مرة ليسع في نواة (\cref{ch:b1:genomes}).
\end{example}

\begin{proposition}[ما الذي يمسك اللولب]\label{prop:b1:nucleic-acids:stability}
\omterm{def:b1:water-small-molecules:water}{الروابط الهيدروجينية} بين القواعد المقترنة تمنح النوعية؛
و\emph{تكديس}\index{تكديس القواعد} القواعد المسطحة بعضها على بعض ---
وهو تماسات \omterm{def:b1:water-small-molecules:bonds}{فان دير فالس} \omterm{prop:b1:water-small-molecules:properties}{والأثر الكاره للماء} الذي يبقي القواعد خارج
الماء --- يمنح معظم الاستقرار؛ والهيكلان المشحونان سالبًا يتنافران،
ولا يستقر اللولب إلا بوجود كاتيونات (وهي فيزيولوجيًا
$\mathrm{Mg^{2+}}$ و$\mathrm{Na^+}$، أو بروتينات الصبغين الأساسية)
تحجب الشحنة. والزوج G--C، بروابطه الهيدروجينية الثلاث وتكديسه الأقوى،
يمسك أحسن من الزوج A--T.
\end{proposition}

\section{الصهر والتهجين}

\begin{definition}[المسخ ودرجة حرارة الانصهار]\label{def:b1:nucleic-acids:melting}
التسخين، أو pH المتطرف، يفصل خصلتي \omterm{def:b1:nucleic-acids:chain}{DNA}:
\emph{المسخ}\index{مسخ DNA} (الانصهار)، وهو انتقال تعاوني على مدى
بضع درجات، منتصفه \emph{درجة حرارة
الانصهار}\index{درجة حرارة الانصهار} $T_m$. ويتبعه \emph{الأثر فوق
اللوني}: إذ تمتص الخصلتان المفردتان من الضوء فوق البنفسجي عند
\qty{260}{nm} نحو \qty{40}{\%} أكثر مما تمتصه القواعد نفسها مكدسة في
لولب. وترتفع $T_m$ مع محتوى $G + C$ (\qty{0.4}{\celsius} لكل واحد في
المئة)، ومع تركيز الملح، ومع الطول؛ ففي ملح \qty{0.15}{mol/L} ينصهر
\omterm{def:b1:nucleic-acids:chain}{DNA} طويل فيه \qty{50}{\%} $G + C$ قرب \qty{90}{\celsius}. وإذا
بُرِّد ببطء وجدت الخصلتان متمماتهما وأعادتا تكوين اللولب: وهو
\emph{إعادة التطبيع}، أو \emph{التهجين}\index{تهجين} حين تأتي
الخصلتان من مصدرين مختلفين.
\end{definition}

\begin{method}[قياس الحموض النووية بالامتصاصية]\label{met:b1:nucleic-acids:absorbance}
\begin{enumerate}
  \item تمتص القواعد الضوء فوق البنفسجي بأعظمية عند \qty{260}{nm}.
        وفي \omterm{def:b1:cell-unit-of-life:cell}{خلية} سماكتها \qty{1}{cm}، تقابل امتصاصية قدرها $1.0$
        \qty{50}{\micro g/mL} من \omterm{def:b1:nucleic-acids:chain}{DNA} المزدوج الخصلة
        (و\qty{40}{\micro g/mL} من \omterm{def:b1:nucleic-acids:chain}{RNA}، و\qty{33}{\micro g/mL} من
        \omterm{def:b1:nucleic-acids:chain}{DNA} المفرد الخصلة).
  \item والنسبة $A_{260}/A_{280}$ هي $1.8$ عند \omterm{def:b1:nucleic-acids:chain}{DNA} النقي و$2.0$ عند
        \omterm{def:b1:nucleic-acids:chain}{RNA} النقي؛ والبروتينات تمتص عند \qty{280}{nm} فتخفضها.
  \item ولمتابعة الانصهار، سجّل $A_{260}$ في أثناء التسخين البطيء؛
        فمنتصف المنحني هو $T_m$ وانحداره مقياس لتجانس العينة.
  \item وللكشف عن تسلسل، امسخ العينة وأضف خصلة مفردة موسومة متممة له
        (وهي \emph{مسبار})؛ فتتهجن حيث تكون متممتها وحدها، ويعلّم
        الوسم الموضع --- على هلام، أو على صبغي، أو في \omterm{def:b1:body-plans-tissues:tissue}{نسيج}.
\end{enumerate}
\end{method}

\section{RNA}

\begin{definition}[أنواع RNA]\label{def:b1:nucleic-acids:rnas}
\omterm{def:b1:nucleic-acids:chain}{RNA} مفرد الخصلة عادة، وهو ينطوي على نفسه حيثما سمحت مقاطع قصيرة
متممة، في دبابيس وعرى وانتفاخات --- أي \emph{بنية ثانوية} --- ترتص
بعد ذلك في شكل. وثلاثة أصناف تحمل جريان المعلومة الوراثية
(\cref{ch:b1:gene-expression}): \emph{RNA الرسول}\index{RNA الرسول}
(mRNA)، وهو نسخة من مورّثة يقرأها الريبوزوم؛ و\emph{RNA
الناقل}\index{RNA الناقل} (tRNA)، وهو نحو \qty{75}{} نوكليوتيدًا
مطويًا في \omterm{def:b1:flowering-plant-organization:organs}{ورقة} برسيم، يحمل حمضًا أمينيًا عند نهايته $3'$
و\emph{مضاد رمزة} يقترن بالرسالة؛ و\emph{RNA
الريبوزومي}\index{RNA الريبوزومي} (rRNA)، وهو اللب البنيوي والحفاز
للريبوزوم، وأربعة أخماس \omterm{def:b1:nucleic-acids:chain}{RNA} \omterm{def:b1:cell-unit-of-life:cell}{الخلية}. وثمة أنواع أخرى كثيرة تنظّم
وتشذّب وتوجّه (كتاب السنة الثالثة).
\end{definition}

\begin{remark}[RNA يمكن أن يكون حفازًا]\label{rem:b1:nucleic-acids:ribozyme}
بعض أنواع \omterm{def:b1:nucleic-acids:chain}{RNA} إنزيمات (\emph{ريبوزيمات}): فالرابطة الببتيدية في كل
بروتين يكوّنها \omterm{def:b1:nucleic-acids:rnas}{RNA الريبوزومي} لا بروتين، وبعض أنواع \omterm{def:b1:nucleic-acids:chain}{RNA} تقص نفسها
وتشذبها. وجزيء يحمل المعلومة ويحفّز التفاعلات في آن هو ما قد يكون أول
نظام حي مصنوعًا منه؛ وعالم \omterm{def:b1:nucleic-acids:chain}{DNA} والبروتين يكون عندئذ تقسيم عمل لاحقًا،
مع بقاء \omterm{def:b1:nucleic-acids:chain}{RNA} في مركزه.
\end{remark}

\section{مسألة: مورثوم فيروس}

\begin{problem}[{مسألة نهاية الأسبوع --- DNA العاثية $\lambda$ يُقاس
ويُوزن ويُعدّ ويُصهر ويُحشر في قفيصته، وصولًا إلى درجة حرارة انصهاره
وطوله الممتد}]
\label{pb:b1:nucleic-acids:1}
مورثوم العاثية $\lambda$ جزيء \omterm{def:b1:nucleic-acids:chain}{DNA} خطي واحد مزدوج الخصلة من
\num{48502} \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} ومحتوى $G + C$ فيه \qty{49.9}{\%}. وهو محشور
في قفيصة عشرينية الوجوه قطرها الداخلي \qty{55}{nm}. خذ \qty{0.34}{nm}
لكل \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد}، و\qty{2.0}{nm} لقطر اللولب، و\qty{650}{Da} لكل \omterm{thm:b1:nucleic-acids:helix}{زوج
قواعد}، و$T_m = 81.5 + 16.6\log_{10}[\mathrm{Na^+}]
+ 0.41\,(\%\,G{+}C) - 500/L$ بالدرجات المئوية، حيث $[\mathrm{Na^+}]$
بوحدة \unit{mol/L} و$L$ الطول بأزواج القواعد.

\medskip
\noindent\textbf{الجزء الأول --- مقيسًا وموزونًا.}
\begin{enumerate}
  \item احسب الطول الممتد للجزيء بالميكرومتر.
  \item احسب عدد دورات اللولب.
  \item احسب كتلته المولية بالدالتون.
  \item احسب كتلة جزيء واحد بالغرام.
  \item احسب عدد زمر الفوسفات ومن ثم الشحنة الصافية للجزيء.
  \item احسب عدد \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيدات} كل \omterm{def:b1:nucleic-acids:nucleotide}{قاعدة} (A وT وG وC).
  \item احسب العدد الكلي \omterm{def:b1:water-small-molecules:water}{للروابط الهيدروجينية} التي تمسك الخصلتين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- مصهورًا.}
\begin{enumerate}[resume]
  \item احسب $T_m$ في صوديوم \qty{0.1}{mol/L}.
  \item احسب $T_m$ في صوديوم \qty{0.01}{mol/L}، وفسّر لماذا يثبّت
        الملح اللولب.
  \item يحوي مورثوم $\lambda$ منطقة من \qty{1000}{} زوج عند
        \qty{35}{\%} $G + C$ وأخرى عند \qty{62}{\%}. احسب $T_m$ لكل
        منطقة في صوديوم \qty{0.1}{mol/L} وصف شكل منحني انصهار الجزيء
        كله.
  \item محلول من \omterm{def:b1:nucleic-acids:chain}{DNA} العاثية $\lambda$ له $A_{260} = 0.50$ قبل
        الانصهار. فما تركيزه، وماذا يصير $A_{260}$ بعد الانصهار التام؟
  \item كم جزيء $\lambda$ في \qty{1}{mL} من ذلك المحلول؟
  \item نهايتا مورثوم $\lambda$ نتوءان مفردا الخصلة من 12 نوكليوتيدًا،
        متممان أحدهما للآخر، يتحلق بهما الجزيء في المضيف. احسب $T_m$
        لهما (باستعمال $L = 12$) وقل لماذا تصمد الحلقة مع ذلك عند
        \qty{37}{\celsius} داخل \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- محشورًا.}
\begin{enumerate}[resume]
  \item احسب حجم \omterm{def:b1:nucleic-acids:chain}{DNA} بوصفه أسطوانة.
  \item احسب الحجم الداخلي للقفيصة.
  \item أي كسر من القفيصة يملؤه \omterm{def:b1:nucleic-acids:chain}{DNA}؟ قارنه بأوثق ارتصاص لأسطوانات
        متوازية (\qty{91}{\%}).
  \item يحمل \omterm{def:b1:nucleic-acids:chain}{DNA} \num{97000} شحنة سالبة في كرة قطرها \qty{55}{nm}.
        فسّر ما الذي يجب أن يرافق \omterm{def:b1:nucleic-acids:chain}{DNA} إلى القفيصة، ولماذا يمارس
        \omterm{def:b1:nucleic-acids:chain}{DNA} المحشور ضغطًا على الغلاف (عشرات الأجواء) يساعد على حقنه
        في بكتيريا.
  \item لا بد أن يُثنى الجزيء في عرى نصف قطرها من رتبة نصف قطر
        القفيصة. \omterm{def:b1:nucleic-acids:chain}{وDNA} يقاوم الثني دون نصف قطر يناهز \qty{50}{nm}.
        علّق.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- مقروءًا.}
\begin{enumerate}[resume]
  \item تبدأ إحدى خصلتي المورثوم $5'$-GGGCGGCGACCT-$3'$. اكتب الخصلة
        المتممة $5' \to 3'$.
  \item إذا استُنسخت مورّثة من \num{1200} \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} إلى mRNA، فما طول
        هذا mRNA \omterm{def:b1:nucleic-acids:nucleotide}{بالنوكليوتيدات} وما كتلته عند \qty{330}{Da} لكل
        \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيد}؟
  \item في المورثوم \qty{75}{} مورّثة في \num{48502} زوج. فأي كسر من
        المورثوم مشفِّر إذا كان متوسط المورّثة \num{600} زوج؟ قارنه
        بمورثوم بشري (\qty{1.5}{\%}).
  \item يتضاعف المورثوم في \qty{20}{min} عند \qty{37}{\celsius}
        بإنزيمات المضيف. احسب المعدل بأزواج القواعد في الثانية إذا
        بدأ التضاعف عند منشأ واحد ومضى في الاتجاهين.
  \item كم \omterm{def:b1:water-small-molecules:water}{رابطة هيدروجينية} تُكسر في الثانية عند الشوكتين في أثناء
        التضاعف (استعمل المتوسط لكل زوج من السؤال 7)؟
  \item طفرة تحوّل زوج G--C واحدًا إلى A--T. فبكم تتغير $T_m$ للجزيء
        كله؟ وهل يستطيع الانصهار كشفها؟
  \item اذكر النتيجة: درجة حرارة انصهار \omterm{def:b1:nucleic-acids:chain}{DNA} العاثية $\lambda$ في
        صوديوم \qty{0.1}{mol/L} وطوله الممتد، والضغط الذي يوسّعه في
        القفيصة.
\end{enumerate}
\end{problem}
