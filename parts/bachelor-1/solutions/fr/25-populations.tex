\section*{Chapitre \ref{ch:b1:populations} --- Populations et démographie}

\begin{solution}{exo:b1:populations:1}
\omterm{def:b1:populations:population}{Population}~: les individus d'une même espèce en un même lieu et à un
même moment, capables de se reproduire entre eux. Densité~: individus
par unité d'aire ou de volume. Dispersion~: leur motif spatial ---
agrégée (un troupeau, une tache d'orties), uniforme (des fous de Bassan
nichant à un coup de bec l'un de l'autre, des créosotiers), aléatoire
(des pissenlits sur une pelouse).
\end{solution}

\begin{solution}{exo:b1:populations:2}
$l_x$~: fraction d'une cohorte vivante à l'âge $x$. $m_x$~: filles par
femelle d'âge $x$. $R_0 = \sum l_x m_x$~: filles par femelle nouveau-née
au cours de sa vie. $T = \sum x l_x m_x / R_0$~: âge moyen des mères.
$R_0 = 1$~: chaque femelle se remplace~; la \omterm{def:b1:populations:population}{population} est stable.
\end{solution}

\begin{solution}{exo:b1:populations:3}
Type III~: environ 15 sur 1000, \qty{1.5}{\%}. Type I~: environ 700 sur
1000, \qty{70}{\%}.
\end{solution}

\begin{solution}{exo:b1:populations:4}
$\dd N/\dd t = rN$ et $\dd N/\dd t = rN(1 - N/K)$. $r$ est le taux de
croissance par tête quand les ressources sont illimitées (naissances
moins morts par individu et par unité de temps)~; $K$ est la \omterm{thm:b1:populations:logistic}{capacité
de charge}, l'effectif auquel la croissance s'arrête.
\end{solution}

\begin{solution}{exo:b1:populations:5}
$r = \ln(1000)/5 = \qty{1.38}{h^{-1}}$~; doublement $\ln 2/1.38 =
\qty{0.5}{h}$.
\end{solution}

\begin{solution}{exo:b1:populations:6}
$l_x m_x = 0, 0.5, 0.6, 0.2, 0$~: $R_0 = 1.3$. $T = (1\times 0.5 +
2\times 0.6 + 3\times 0.2)/1.3 = 2.3/1.3 = 1.77$. $r = \ln 1.3/1.77 =
\qty{0.15}{}$ par unité de temps~: elle croît.
\end{solution}

\begin{solution}{exo:b1:populations:7}
$N = 50\times 70/7 = 500$. Seuls 40 poissons marqués subsistaient~: $N = 40\times
70/7 = 400$~; la première estimation était trop haute d'un quart (la
fraction marquée de la \omterm{def:b1:populations:population}{population} était plus petite que supposé).
\end{solution}

\begin{solution}{exo:b1:populations:8}
$0.2\times 100\times 0.9 = 18$~; $0.2\times 500\times 0.5 = 50$~;
$0.2\times 900\times 0.1 = 18$ par an. Maximum $rK/4 = 50$ par an, en
$N = 500$.
\end{solution}

\begin{solution}{exo:b1:populations:9}
Pissenlit~: $r$ (des milliers de graines, annuel). Éléphant~: $K$ (un
petit tous les quatre ans, longue vie, soins). Morue~: $r$ (des millions
d'œufs, aucun soin). Gorille~: $K$ (un petit tous les quatre ans, des
décennies de vie). Mouche domestique~: $r$ (des centaines d'œufs,
maturité en dix jours).
\end{solution}

\begin{solution}{exo:b1:populations:10}
$r = \ln(29/10)/2 = \qty{0.53}{h^{-1}}$~; $K \approx 660$. À \qty{8}{h}~:
$660/(1 + 65e^{-4.24}) = 660/(1 + 0.94) = 340$, contre 351 comptées.
Maximum en $K/2 = 330$, atteint en $t = \ln 65/0.53 = \qty{7.9}{h}$, où
$\dd N/\dd t = rK/4 = 87\times 10^5$ \omterm{def:b1:cell-unit-of-life:cell}{cellules} par millilitre et par
heure.
\end{solution}

\begin{solution}{exo:b1:populations:11}
La régulation exige que le taux par tête baisse quand $N$ monte, de
sorte que $N$ revienne vers un équilibre après une perturbation~; un
facteur qui tue une fraction fixe change $N$ mais non la dépendance du
taux envers $N$, si bien qu'après lui la \omterm{def:b1:populations:population}{population} reprend la même
croissance. Le temps qu'il fait limite les insectes en frappant assez
souvent, et au hasard, pour que la \omterm{def:b1:populations:population}{population} soit rejetée en arrière
avant d'approcher le moindre plafond~: une \omterm{def:b1:populations:population}{population} non régulée,
maintenue basse par des perturbations répétées, et dont l'effectif
moyen dépend de la fréquence des mauvaises saisons plutôt que d'une
\omterm{thm:b1:populations:logistic}{capacité de charge}.
\end{solution}

\begin{solution}{exo:b1:populations:12}
Deux \omterm{def:b1:populations:population}{populations} de même effectif, l'une à large base de jeunes et
l'autre à classes d'âge égales, ont des avenirs différents~: la première
croîtra des décennies durant même si son $m_x$ tombe dès maintenant au
seuil de remplacement, parce que ses nombreux jeunes n'ont pas encore
procréé (inertie démographique)~; la seconde non. $N$ enregistre le
passé~; c'est la distribution des âges et le calendrier de $m_x$ qui la
parcourt que les équations font agir, et la pyramide prédit l'effectif
de la génération suivante, ce dont $N$ seul ne dit rien.
\end{solution}

\begin{solution}{pb:b1:populations:1}
\textbf{1.} $r = \ln(29/10)/2 = \qty{0.53}{h^{-1}}$.
\textbf{2.} $\ln 2/0.53 = \qty{1.3}{h}$.
\textbf{3.} $K \approx 660\times 10^5 = \qty{6.6e7}{}$ \omterm{def:b1:cell-unit-of-life:cell}{cellules} par
millilitre.
\textbf{4.} $N(t) = 660/(1 + 65e^{-0.53t})$~: à 4 h, $660/(1 + 7.8) =
75$ (71 comptées)~; à 8 h, 340 (351)~; à 12 h, $660/(1 + 0.112) = 594$
(595).
\textbf{5.} En $K/2 = 330$, atteint en $t = \ln 65/0.53 = \qty{7.9}{h}$~;
vitesse maximale $rK/4 = 0.53\times 660/4 = 87\times 10^5$ \omterm{def:b1:cell-unit-of-life:cell}{cellules} par
millilitre et par heure.
\textbf{6.} $10e^{0.53\times 12} = 10\times 578 = 5780$~: dix fois les
595 observées.
\textbf{7.} L'épuisement du sucre (ou de l'oxygène)~; l'accumulation
d'éthanol et d'acides qui empoisonnent les \omterm{def:b1:cell-unit-of-life:cell}{cellules}.
\textbf{8.} $q_0 = 0.40$~; $q_4 = 1 - 0.34/0.40 = 0.15$~: une forte
mortalité précoce puis un taux plus régulier --- entre les types II et
III, comme chez la plupart des grands mammifères sauvages.
\textbf{9.} $l_x m_x = 0, 0, 0.312, 0.368, 0.320, 0.272, 0.182, 0.080,
0.018, 0$~: $R_0 = 1.55$.
\textbf{10.} $\sum x l_x m_x = 0.624 + 1.104 + 1.280 + 1.360 + 1.092 +
0.560 + 0.144 = 6.16$~; $T = 6.16/1.55 = \qty{4.0}{yr}$.
\textbf{11.} $r = \ln 1.55/4.0 = \qty{0.11}{yr^{-1}}$~; doublement en
\qty{6.3}{yr}.
\textbf{12.} $200e^{1.1} = 600$ cerfs.
\textbf{13.} $N(10) = 600/(1 + 2e^{-1.1}) = 600/1.67 = 360$~; le
troupeau dépasse 500 quand $(K - N_0)/N_0\,e^{-rt} = 0.2$, c'est-à-dire
$e^{-rt} = 0.1$, $t = \ln
10/0.11 = 21$ ans.
\textbf{14.} $120\times 150/18 = 1000$ perches.
\textbf{15.} 100 marquées~: $100\times 150/18 = 833$~; la première
estimation était trop haute (moins de poissons marqués circulaient que
supposé).
\textbf{16.} $rK/4 = 100$ par an, en $N = 500$.
\textbf{17.} $T_2 = \ln 2/0.4 = \qty{1.7}{yr}$.
\textbf{18.} Prise $hN$ avec $h = 0.2$~: $rN(1 - N/K) = hN$ donne $N^* =
K(1 - h/r) = 1000\times 0.5 = 500$ et une prise de \num{100} par an ---
en dessous du rendement soutenable maximal, mais une fraction s'ajuste
d'elle-même au stock~: si le stock diminue de moitié, la prise aussi, et
la \omterm{def:b1:populations:population}{population} revient.
\textbf{19.} En 500~: $0.4\times 500\times 0.5 = 100 < 120$~: la
\omterm{def:b1:populations:population}{population} décline. En 300~: $0.4\times 300\times 0.7 = 84 < 120$~: elle
décline plus vite --- une fois sous $K/2$, un prélèvement fixe supérieur
au rendement maximal la mène à l'extinction.
\textbf{20.} En $K/2$, le surplus est maximal mais la moindre erreur ---
une \omterm{def:b1:populations:population}{population} surestimée d'un cinquième, une mauvaise année qui réduit
$r$ --- transforme un prélèvement soutenable en déclin, et sous $K/2$ le
surplus se réduit à mesure que la \omterm{def:b1:populations:population}{population} se réduit~: l'équilibre est
instable au surprélèvement. Prélever un peu au-dessus de $K/2$ laisse
une marge.
\textbf{21.} $R_0$ tombe de moitié à $0.78 < 1$~: le troupeau se réduit
jusqu'à repasser sous 400, où $m_x$ se rétablit --- un facteur
densité-dépendant, qui régule le troupeau autour de 400.
\textbf{22.} Non~: il retire \qty{40}{\%} quel que soit l'effectif et
laisse le taux par tête inchangé~; les années suivantes, le troupeau
recroît au même $r$ depuis un point de départ plus bas --- une
perturbation, non une régulation.
\textbf{23.} $0.11N(1 - N/600) = 30$~: $N^2 - 600N + 163\,600 = 0$,
$N = 300\pm\sqrt{90\,000 - 163\,600}$ --- aucune solution réelle~: 30 par
an dépasse le surplus maximal $rK/4 = 16.5$, et les loups mènent le
troupeau à l'extinction. Avec 15 par an~: $N = 300\pm\sqrt{90\,000 -
81\,800} = 300\pm 91$, soit 209 et 391~; le plus haut est stable
(au-dessus, le surplus est inférieur à 15 et le troupeau redescend~;
au-dessous, il est supérieur et le troupeau remonte), le plus bas est
instable.
\textbf{24.} Les effectifs de lynx répondent à la nourriture avec un
retard~: une hausse des lièvres élève les naissances et la \omterm{def:b1:populations:lifetable}{survie} des
lynx au cours de l'année ou des deux suivantes, et un effondrement des
lièvres n'affame les lynx qu'une fois leur \omterm{def:b1:lipids:triglyceride}{graisse} et les petits de la
saison épuisés.
\textbf{25.} Levure~: $K = \qty{6.6e7}{}$ \omterm{def:b1:cell-unit-of-life:cell}{cellules} par millilitre, $r =
\qty{0.53}{h^{-1}}$. Cerfs~: $R_0 = 1.55$, $T = \qty{4.0}{yr}$, $r =
\qty{0.11}{yr^{-1}}$.
\end{solution}
