\chapter{हमारे आस-पास के जंतु}\label{ch:g1:animals-around-us}

एक कलचिड़ी लॉन पर फुदक रही है, दीवार से एक बिल्ली उसे ताक रही है, और
एक चींटी दरवाज़े की चौखट पर अपने से बड़ा टुकड़ा ढो रही है। \omterm{def:g1:animals-around-us:animal}{जंतु} हमारे
चारों ओर हैं --- बग़ीचे में, पार्क में, घर में। आओ, उन्हें उसी तरह
देखना सीखें जैसे कोई जीवविज्ञानी देखता है: उनके शरीर को, और उनके चलने
के ढंग को।

\section{जंतु किसे कहते हैं?}

\begin{definition}[जंतु]\label{def:g1:animals-around-us:animal}
\emph{जंतु}\index{जंतु} वह \omterm{def:g1:living-or-not:living}{सजीव} है जो दूसरे सजीवों को --- पौधों को,
या दूसरे जंतुओं को --- खाकर पोषण लेता है और जो आम तौर पर अपने आप
इधर-उधर जा सकता है। बिल्ली, कलचिड़ी, मछली और चींटी, सब जंतु हैं। हम
भी जंतु हैं।
\end{definition}

\begin{example}[एक ही बग़ीचे के जंतु]\label{ex:g1:animals-around-us:garden}
एक छोटे-से बग़ीचे में बहुत-से \omterm{def:g1:animals-around-us:animal}{जंतुओं} का घर हो सकता है: ज़मीन से केंचुआ
खींचती कलचिड़ी, वह केंचुआ ख़ुद, पत्ती पर बैठा घोंघा, जाले में मकड़ी,
फूल में मधुमक्खी, और पड़ोसी की बिल्ली। बड़े हों या नन्हे, ये सब पोषण
लेते हैं और सब चलते-फिरते हैं।
\end{example}

\begin{remark}[छोटा होने का मतलब मामूली होना नहीं]\label{rem:g1:animals-around-us:small}
चींटी उतनी ही \omterm{def:g1:animals-around-us:animal}{जंतु} है जितना हाथी। शरीर का आकार यह तय नहीं करता कि कौन
\omterm{def:g1:animals-around-us:animal}{जंतु} है: खाना और चलना तय करते हैं। हमारे आस-पास के कुछ सबसे अचरज भरे
\omterm{def:g1:animals-around-us:animal}{जंतु} सबसे नन्हे ही हैं।
\end{remark}

\section{जंतु के शरीर के भाग}

\begin{definition}[शरीर के भाग]\label{def:g1:animals-around-us:parts}
ज़्यादातर \omterm{def:g1:animals-around-us:animal}{जंतुओं} का एक \emph{सिर}\index{सिर} होता है (जिसमें मुँह और
ज़्यादातर ज्ञानेंद्रियाँ होती हैं), एक धड़, और \emph{उपांग}\index{उपांग}
--- टाँगें, डैने या मीन-पंख --- जिनसे वे चलते हैं। कइयों की एक
\emph{पूँछ}\index{पूँछ} भी होती है।
\end{definition}

\end{omfigure}

\begin{example}[टाँगें गिनो]\label{ex:g1:animals-around-us:legs}
टाँगों की गिनती करने लायक़ होती है। कलचिड़ी $2$ टाँगों पर चलती है,
बिल्ली $4$ पर, गुबरैला $6$ पर, मकड़ी $8$ पर --- और केंचुए तथा घोंघे
की तो एक भी टाँग नहीं होती, फिर भी वे जहाँ जाना हो वहाँ पहुँच ही जाते
हैं।
\end{example}

\begin{definition}[शरीर का आवरण]\label{def:g1:animals-around-us:covering}
\omterm{def:g1:animals-around-us:animal}{जंतु} का शरीर एक आवरण से सुरक्षित रहता है: बिल्ली पर
\emph{रोम}\index{रोम} (बाल), कलचिड़ी पर \emph{पंख}\index{पंख}, मछली
पर \emph{शल्क}\index{शल्क}, घोंघे पर \emph{खोल}\index{खोल}, और मेंढक
पर चिकनी गीली \emph{त्वचा}\index{त्वचा}।
\end{definition}

\begin{example}[किसका कौन-सा लिबास]\label{ex:g1:animals-around-us:covering}
बिल्ली पर हाथ फेरो: \omterm{def:g1:animals-around-us:covering}{रोम}। गिरा हुआ कलचिड़ी का \omterm{def:g1:animals-around-us:covering}{पंख} उठाओ: हलका और मुलायम।
बाज़ार में मछली को ग़ौर से देखो: चमकीले \omterm{def:g1:animals-around-us:covering}{शल्कों} की कतारें, छोटी-छोटी
छत की टाइलों जैसी। तीन \omterm{def:g1:animals-around-us:animal}{जंतु}, तीन अलग लिबास।
\end{example}

\section{जंतु कैसे चलते हैं}

\begin{proposition}[चलने के ढंग]\label{prop:g1:animals-around-us:moving}
\omterm{def:g1:animals-around-us:animal}{जंतु} अपने उपांगों के अनुसार चलते हैं:
\begin{itemize}
  \item टाँगों वाले \omterm{def:g1:animals-around-us:animal}{जंतु} \emph{चलते}, दौड़ते, फुदकते या रेंगते हैं ---
        बिल्ली, लॉन पर कलचिड़ी, चींटी;
  \item डैनों वाले \omterm{def:g1:animals-around-us:animal}{जंतु} \emph{उड़ते}\index{उड़ना} हैं --- कलचिड़ी,
        मधुमक्खी, तितली;
  \item मीन-पंखों वाले \omterm{def:g1:animals-around-us:animal}{जंतु} \emph{तैरते}\index{तैरना} हैं --- मछली;
  \item बिना \omterm{def:g1:animals-around-us:parts}{उपांग} वाले \omterm{def:g1:animals-around-us:animal}{जंतु} सरकते या बल खाते हैं --- घोंघा अपने एक
        मुलायम पैर पर, केंचुआ लंबा-छोटा होते हुए।
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[एक जंतु, कई ढंग]\label{rem:g1:animals-around-us:several}
बहुत-से \omterm{def:g1:animals-around-us:animal}{जंतुओं} के चलने के एक से ज़्यादा ढंग होते हैं। कलचिड़ी चलती
\emph{भी} है और उड़ती भी; मेंढक तैरता \emph{भी} है और फुदकता भी;
बत्तख़ चलती है, तैरती \emph{भी} है और उड़ती भी। किसी \omterm{def:g1:animals-around-us:animal}{जंतु} की चाल तय
करने से पहले उसे कुछ देर देखो।
\end{remark}

\begin{method}[जंतु का प्रेक्षण]\label{met:g1:animals-around-us:observe}
किसी \omterm{def:g1:animals-around-us:animal}{जंतु} का वर्णन जीवविज्ञानी की तरह करने के लिए:
\begin{enumerate}
  \item भागों को देखो: \omterm{def:g1:animals-around-us:parts}{सिर}, धड़, \omterm{def:g1:animals-around-us:parts}{उपांग}, पूँछ;
  \item टाँगें गिनो, अंदाज़े से नहीं;
  \item आवरण का नाम बताओ: \omterm{def:g1:animals-around-us:covering}{रोम}, \omterm{def:g1:animals-around-us:covering}{पंख}, \omterm{def:g1:animals-around-us:covering}{शल्क}, \omterm{def:g1:animals-around-us:covering}{खोल}, \omterm{def:g1:animals-around-us:covering}{त्वचा};
  \item देखो कि वह कैसे चलता है: चलना, फुदकना, उड़ना, तैरना, सरकना;
  \item बताओ कि वह कहाँ मिला: लॉन, पेड़, तालाब, पत्थर के नीचे।
\end{enumerate}
\omterm{def:g1:animals-around-us:animal}{जंतु} को कभी चोट मत पहुँचाओ: देखो, फिर उसे अपने रास्ते जाने दो।
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:1}
चित्र में बिल्ली के शरीर के चार मुख्य भागों के नाम बताओ। कौन-सा भाग
आँखें और मुँह लिए रहता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:2}
कितनी टाँगें: (क) मुर्ग़ी? (ख) कुत्ता? (ग) गुबरैला? (घ) मकड़ी?
(ङ) केंचुआ?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:3}
हर \omterm{def:g1:animals-around-us:animal}{जंतु} का आवरण बताओ: भेड़, बत्तख़, ट्राउट, घोंघा।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:4}
इस अध्याय के कौन-से \omterm{def:g1:animals-around-us:animal}{जंतु} उड़ सकते हैं? अगली टाँगों या बाँहों की जगह
उन सबके पास क्या है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:5}
घोंघे और केंचुए की कोई टाँग नहीं होती। इनमें से हर एक कैसे चलता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:6}
बत्तख़ तीन ढंग से चल सकती है। उनके नाम बताओ, और हर एक के लिए बताओ कि
वह शरीर के किस भाग का इस्तेमाल करती है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:7}
\cref{def:g1:animals-around-us:animal} के अनुसार चींटी \omterm{def:g1:animals-around-us:animal}{जंतु} क्यों है?
परिभाषा की दोनों बातों से उत्तर दो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:8}
करके देखो: बाहर किसी एक \omterm{def:g1:animals-around-us:animal}{जंतु} को कुछ देर देखो --- कोई चिड़िया, कोई
कीट, कोई घोंघा। \cref{met:g1:animals-around-us:observe} के पाँच चरणों
से उसका वर्णन करो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:9}
एक साथी कहता है: ``मछलियाँ \omterm{def:g1:animals-around-us:animal}{जंतु} नहीं हो सकतीं --- वे चलती ही नहीं!''
तुम क्या उत्तर दोगे? \cref{prop:g1:animals-around-us:moving} का
इस्तेमाल करो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:10}
पहेली वाला \omterm{def:g1:animals-around-us:animal}{जंतु}: उसकी $2$ टाँगें हैं, $2$ डैने हैं, \omterm{def:g1:animals-around-us:covering}{पंख} हैं, और वह
उड़ नहीं सकता, पर बहुत तेज़ दौड़ता है। क्या वह पक्षी है? क्या वह
तुम्हारे मोहल्ले का सबसे तेज़ धावक हो सकता है? (शुतुरमुर्ग़ के बारे
में सोचो।)
\end{exercise}
