\section*{अध्याय \ref{ch:g4:heart-and-blood} --- हृदय और रक्त}

\begin{solution}{exo:g4:heart-and-blood:1}
\omterm{def:g4:journey-of-food:digestion}{पोषक तत्व}, जो \omterm{def:g4:journey-of-food:tract}{छोटी आँत} पर बटोरे जाते हैं; और ऑक्सीजन, जो फेफड़ों पर
बटोरी जाती है।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:2}
कार्बन डाइऑक्साइड --- जिसे वह फेफड़ों पर छोड़ता है, जहाँ से वह साँस के
साथ बाहर निकल जाती है।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:3}
वह एक खोखली \omterm{def:g3:bones-and-muscles:muscle}{पेशी} है, लगभग अपने मालिक की मुट्ठी जितनी बड़ी, जो छाती के
बीच में \omterm{def:g3:bones-and-muscles:skeleton}{पसलियों} के पिंजरे के पीछे बैठी है।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:4}
दायाँ पक्ष \omterm{prop:g4:journey-of-food:absorb}{रक्त} को ऑक्सीजन लादने के लिए फेफड़ों तक पंप करता है; बायाँ
पक्ष ऑक्सीजन से भरे \omterm{prop:g4:journey-of-food:absorb}{रक्त} को पूरे शरीर में पंप करता है।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:5}
वह छोटी लहर जो हर \omterm{def:g4:heart-and-blood:heart}{धड़कन} \omterm{def:g4:heart-and-blood:vessels}{वाहिकाओं} में दौड़ा देती है, और जो \omterm{def:g1:my-body:joint}{कलाई} या गरदन
पर महसूस होती है। उसे गिनने से मिनट भर की धड़कनें गिनी जाती हैं।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:6}
अँगूठे की अपनी एक साफ़ महसूस होने वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} होती है, इसलिए तुम जिस
\omterm{def:g1:my-body:joint}{कलाई} की जाँच कर रहे हो उसके बजाय अपना अँगूठा गिन बैठ सकते हो।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:7}
दौड़ती \omterm{def:g1:my-body:parts}{टाँगें} \omterm{def:g4:journey-of-food:digestion}{पोषक तत्व} और ऑक्सीजन तेज़ी से जलाती हैं और उन्हें तेज़ी
से वापस चाहिए; इसलिए \omterm{def:g4:heart-and-blood:heart}{हृदय} अपनी धड़कनें तेज़ और ज़ोरदार कर देता है ताकि
\omterm{prop:g4:journey-of-food:absorb}{रक्त} अपने चक्कर ज़्यादा बार लगा सके।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:8}
उत्तर खुला है। आम तौर पर आराम में लगभग $80$--$100$, और तारे-कूद के बाद
$120$--$150$ --- यानी आपूर्ति-शृंखला काम पर।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:9}
फेफड़ों से निकलते समय उसने अभी-अभी ऑक्सीजन लादी थी --- चटक लाल। \omterm{def:g3:bones-and-muscles:muscle}{पेशी}
में उसने ऑक्सीजन सौंप दी और कार्बन डाइऑक्साइड ले ली --- गहरी। रंग सामान
के पीछे चलता है।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:10}
\omterm{def:g4:journey-of-food:tract}{छोटी आँत} \omterm{prop:g4:journey-of-food:absorb}{रक्त} में \omterm{def:g4:journey-of-food:digestion}{पोषक तत्व} डालती है; \omterm{def:g4:breathing:organs}{फेफड़े} उसमें ऑक्सीजन डालते और
कार्बन डाइऑक्साइड निकालते हैं; और \omterm{def:g4:heart-and-blood:heart}{हृदय} वह पंप है जो \omterm{prop:g4:journey-of-food:absorb}{रक्त} को इनके बीच
और हर उपभोक्ता तक --- \omterm{def:g3:bones-and-muscles:muscle}{पेशियों}, मस्तिष्क, \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियों}, \omterm{def:g1:the-five-senses:sense}{त्वचा} तक --- चलता
रखता है।
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:11}
सधा हुआ \omterm{def:g4:heart-and-blood:heart}{हृदय} ज़्यादा मज़बूत \omterm{def:g3:bones-and-muscles:muscle}{पेशी} है: उसकी हर \omterm{def:g4:heart-and-blood:heart}{धड़कन} ज़्यादा \omterm{prop:g4:journey-of-food:absorb}{रक्त} हिलाती
है। तब उतनी ही आपूर्ति पहुँचाने के लिए मिनट में कम धड़कनें चाहिए ---
इसलिए साइकिल-चालक की इत्मीनान वाली $50$ वही काम कर देती है जो बिना सधे
की $75$, और मेहनत के लिए ताक़त बची भी रहती है।
\end{solution}
