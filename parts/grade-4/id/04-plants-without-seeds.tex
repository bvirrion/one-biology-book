\chapter{Tumbuhan Tanpa Biji}\label{ch:g4:plants-without-seeds}

Sisi teduh tembok tua itu berlapis \omterm{prop:g4:plants-without-seeds:damp}{lumut}; di bawah pohon beech,
\omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} membuka gulungan \omterm{def:g1:plants-around-us:parts}{daunnya} yang hijau dan besar. Carilah
sepanjang musim panas: kamu tidak akan menemukan \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}, atau
\omterm{def:g2:from-seed-to-plant:seed}{biji} --- tidak pernah. Namun \omterm{prop:g4:plants-without-seeds:damp}{lumut} dan \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} menyebar dengan
sangat baik. \omterm{def:g1:plants-around-us:plant}{Tumbuhan}, ternyata, punya lebih dari satu cara membuat
generasi berikutnya.

\section{Tumbuhan paku dan lumut: berkembang biak dengan spora}

\begin{definition}[Spora]\label{def:g4:plants-without-seeds:spore}
Suatu \emph{spora}\index{spora} adalah butir renik, yang dibuat dalam
jumlah amat besar, dan darinya sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan} baru dapat tumbuh.
Berbeda dengan \omterm{def:g2:from-seed-to-plant:seed}{biji} (\cref{def:g2:from-seed-to-plant:seed}), sebutir
spora adalah satu bintik \emph{tanpa \omterm{def:g3:animals-through-seasons:active}{cadangan makanan} dan tanpa
\omterm{def:g1:plants-around-us:plant}{tumbuhan} mungil yang sudah jadi di dalamnya} --- sebuah permulaan
dalam wujudnya yang paling telanjang.
\end{definition}

\begin{example}[Di bawah daun tumbuhan paku]\label{ex:g4:plants-without-seeds:fern}
Baliklah sehelai \omterm{def:g1:plants-around-us:parts}{daun} \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} pada akhir musim panas: deretan
rapi bintik cokelat berbaris di sisi bawahnya. Tiap bintik adalah
sebuah bungkusan berisi ribuan \omterm{def:g4:plants-without-seeds:spore}{spora}; ketika masak, bungkusannya
terbelah, dan \omterm{def:g4:plants-without-seeds:spore}{sporanya} melayang pergi di udara seperti debu yang
paling halus. Sebutir \omterm{def:g4:plants-without-seeds:spore}{spora} yang mendarat di tanah teduh yang lembap
dapat tumbuh --- bertahap --- menjadi \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} yang baru.
\end{example}

\begin{omfigure}
\includegraphics[width=0.56\linewidth]{images/book1/ai/g4-ferns-sori.jpg}

{\small Di bawah \omterm{def:g1:plants-around-us:parts}{daunnya}: deretan rapi bintik cokelat, masing-masing
sebuah bungkusan berisi ribuan \omterm{def:g4:plants-without-seeds:spore}{spora}.}
\end{omfigure}

\begin{proposition}[Tumbuhan berspora membutuhkan kelembapan]\label{prop:g4:plants-without-seeds:damp}
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} berspora --- \emph{lumut}\index{lumut} dan \emph{tumbuhan
paku}\index{tumbuhan paku} --- hanya dapat berkembang biak di tempat
yang \emph{lembap}: \omterm{def:g4:plants-without-seeds:spore}{sporanya}, dan tahap halus yang tumbuh darinya,
mengering lalu mati di udara yang kering. Itulah sebabnya \omterm{prop:g4:plants-without-seeds:damp}{lumut}
menghampari tembok utara yang teduh tetapi tidak pernah tembok selatan
yang tersinari, dan sebabnya \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} berdesak di hutan yang
lembap, tepian sungai dan sumur.
\end{proposition}
\admitted

\begin{remark}[Spora bukanlah biji]\label{rem:g4:plants-without-seeds:sporeseed}
Keduanya adalah permulaan yang mengembara, tetapi \omterm{def:g2:from-seed-to-plant:seed}{biji} adalah
perlengkapan bertahan hidup --- \omterm{def:g1:the-five-senses:sense}{kulit}, \omterm{def:g3:animals-through-seasons:active}{cadangan makanan}, \omterm{def:g1:plants-around-us:plant}{tumbuhan}
mungil yang sudah jadi --- sedangkan \omterm{def:g4:plants-without-seeds:spore}{spora} telanjang. Karena itulah
jumlahnya: jutaan \omterm{def:g4:plants-without-seeds:spore}{spora} \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} melawan selusin \omterm{def:g2:from-seed-to-plant:seed}{biji} dalam
polong kacang. Permulaan yang telanjang murah dibuat dan mudah hilang;
kedua rencana itu menggemakan \omterm{prop:g3:sorting-living-things:groups}{ikan} kod dan gajah pada
\cref{prop:g4:animal-reproduction:strategies}.
\end{remark}

\section{Tumbuhan baru tanpa penyemaian}

\begin{definition}[Perkembangbiakan vegetatif]\label{def:g4:plants-without-seeds:vegetative}
Banyak \omterm{def:g1:plants-around-us:plant}{tumbuhan} juga dapat membuat \omterm{def:g1:plants-around-us:plant}{tumbuhan} baru \emph{langsung dari
sepotong dirinya sendiri} --- tanpa \omterm{def:g4:plants-without-seeds:spore}{spora}, tanpa \omterm{def:g2:from-seed-to-plant:seed}{biji}, hanya satu
induk. Inilah \emph{perkembangbiakan vegetatif}\index{perkembangbiakan
vegetatif}. \omterm{def:g1:plants-around-us:plant}{Tumbuhan} barunya adalah kelanjutan hidup \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang
lama, dan menyerupainya dengan persis.
\end{definition}

\begin{example}[Geragih stroberi]\label{ex:g4:plants-without-seeds:runner}
Tanaman stroberi menjulurkan \omterm{def:g1:plants-around-us:parts}{batang} mendatar yang panjang dan tipis
--- \emph{geragih}\index{geragih} --- melintasi tanah. Di tempat
sebuah \omterm{ex:g4:plants-without-seeds:runner}{geragih} menyentuh tanah, ia berakar, menumbuhkan \omterm{def:g1:plants-around-us:parts}{daun}, lalu
menjadi tanaman stroberi baru yang lengkap; \omterm{ex:g4:plants-without-seeds:runner}{geragihnya} kemudian layu,
dan anaknya berdiri bebas. Satu tanaman menjadi sepetak tanpa satu
butir \omterm{def:g2:from-seed-to-plant:seed}{biji} pun disemai.
\end{example}

\begin{example}[Umbi lapis dan umbi batang]\label{ex:g4:plants-without-seeds:bulbtuber}
\emph{Umbi lapis}\index{umbi lapis} tulip dan \emph{umbi
batang}\index{umbi batang} kentang adalah gudang bawah tanah --- dan
sekaligus perlengkapan awal. Kalau ditinggalkan di dalam tanah, sebuah
\omterm{ex:g4:plants-without-seeds:bulbtuber}{umbi lapis} terbelah menjadi umbi anakan, masing-masing sebuah tulip
masa depan; setiap ``\omterm{def:g1:the-five-senses:sense}{mata}'' kentang yang bertunas dapat tumbuh menjadi
satu tanaman baru yang utuh. Tukang kebun menanam kentang, bukan \omterm{def:g2:from-seed-to-plant:seed}{biji}
kentang.
\end{example}

\begin{method}[Mengambil stek]\label{met:g4:plants-without-seeds:cutting}
Tukang kebun meniru \omterm{def:g4:plants-without-seeds:vegetative}{perkembangbiakan vegetatif} dengan
\emph{stek}\index{stek}:
\begin{enumerate}
  \item potong sebuah tunas muda yang sehat, sepanjang satu telapak
        tangan, dari tanaman induknya (\omterm{def:g1:plants-around-us:parts}{daun} mint, dedalu dan geranium
        cocok sekali);
  \item buang \omterm{def:g1:plants-around-us:parts}{daun} bagian bawahnya lalu tegakkan tunasnya di dalam air
        atau tanah yang lembap;
  \item tunggulah: dalam hitungan hari atau minggu, \omterm{def:g1:plants-around-us:parts}{akar} tumbuh dari
        \omterm{def:g1:plants-around-us:parts}{batang} yang dipotong itu;
  \item tanamlah ia --- sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan} baru yang lengkap, serupa
        dengan satu induknya.
\end{enumerate}
\end{method}

\begin{remark}[Satu induk, salinan yang persis]\label{rem:g4:plants-without-seeds:copy}
\omterm{def:g2:from-seed-to-plant:seed}{Biji} mencampurkan dua induk
(\cref{rem:g4:animal-reproduction:mix} berlaku bagi \omterm{def:g1:plants-around-us:plant}{tumbuhan} juga,
lewat \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari}); \omterm{def:g4:plants-without-seeds:vegetative}{perkembangbiakan vegetatif} menyalin satu induk.
Itulah sebabnya setiap tanaman dari sebuah jenis apel bernama
ditumbuhkan dari \omterm{met:g4:plants-without-seeds:cutting}{stek} dan sambungan, tidak pernah dari \omterm{def:g2:from-seed-to-plant:seed}{bijinya}:
kebunnya menginginkan salinan yang persis dari satu pohon yang
istimewa. Harga penyalinan adalah keseragaman --- sebuah kebun berisi
pohon yang serupa berbagi kelemahan yang serupa pula, seperti akan
ditunjukkan pada tahun berikutnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:1}
Sebutkan dua \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang tidak pernah membuat \omterm{def:g2:from-seed-to-plant:seed}{biji}. Bagaimana
keduanya berkembang biak?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:2}
Berikan dua perbedaan antara sebutir \omterm{def:g4:plants-without-seeds:spore}{spora} dan sebutir \omterm{def:g2:from-seed-to-plant:seed}{biji}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:3}
Di mana \omterm{def:g4:plants-without-seeds:spore}{spora} \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku} dibuat? Lukiskan apa yang kamu temukan di
bawah sehelai \omterm{def:g1:plants-around-us:parts}{daunnya} pada akhir musim panas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:4}
Mengapa \omterm{prop:g4:plants-without-seeds:damp}{lumut} tumbuh di tembok utara yang teduh dan tidak di tembok
selatan yang tersinari?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:5}
Apa itu \omterm{def:g4:plants-without-seeds:vegetative}{perkembangbiakan vegetatif}? Berapa banyak induk yang
dibutuhkannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:6}
Ceritakan kisah sebuah \omterm{ex:g4:plants-without-seeds:runner}{geragih} stroberi, dari \omterm{def:g1:plants-around-us:parts}{batang} pertamanya sampai
tanaman anak yang berdiri bebas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:7}
Mengapa tukang kebun menanam kentang alih-alih menyemai \omterm{def:g2:from-seed-to-plant:seed}{biji} kentang?
Apa yang dapat dilakukan tiap ``\omterm{def:g1:the-five-senses:sense}{mata}''?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:8}
Cobalah: ambil sebuah \omterm{met:g4:plants-without-seeds:cutting}{stek} \omterm{def:g1:plants-around-us:parts}{daun} mint menurut
\cref{met:g4:plants-without-seeds:cutting} lalu tegakkan ia di dalam
segelas air di ambang jendela. Apa yang muncul, dan sesudah kira-kira
berapa lama?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:9}
\omterm{prop:g4:plants-without-seeds:damp}{Tumbuhan paku} membuat jutaan \omterm{def:g4:plants-without-seeds:spore}{spora}; polong kacang, selusin \omterm{def:g2:from-seed-to-plant:seed}{biji}.
Jelaskan perbedaan jumlahnya dengan
\cref{rem:g4:plants-without-seeds:sporeseed} --- lalu sebutkan
pasangan \omterm{def:g1:animals-around-us:animal}{hewan} yang siasatnya digemakan keduanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:10}
Seorang tukang kebun mendapati tanaman stroberi barunya serupa dengan
induknya, sementara tanaman kacangnya sedikit berbeda dari induknya.
Jelaskan kedua pengamatan itu dengan
\cref{rem:g4:plants-without-seeds:copy}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:plants-without-seeds:11}
Sebuah dahan dedalu yang patah oleh banjir musim dingin, terbawa arus
lalu terdampar di sebuah gundukan lumpur, sering tumbuh menjadi pohon
dedalu yang baru. Perkembangbiakan jenis apa ini --- dan mengapa
sungai begitu sering dipagari pohon dedalu? Apa yang dilakukan
banjirnya, yang dikerjakan tukang kebun pada
\cref{met:g4:plants-without-seeds:cutting} dengan gunting?
\end{exercise}
