\chapter{De la fleur au fruit}\label{ch:g5:flower-to-fruit}

\Cref{ch:g3:flowers-fruits-seeds} a regardé la \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} de cerisier
devenir une cerise et a laissé une question ouverte~: \emph{que se
passe-t-il vraiment} quand un grain de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} se pose sur le
\omterm{def:g3:flowers-fruits-seeds:parts}{pistil}~? Cette année, munis de l'idée de \omterm{def:g5:flower-to-fruit:steps}{fécondation} de
\cref{ch:g4:animal-reproduction} et des \omterm{def:g5:first-look-at-cells:cell}{cellules} de
\cref{ch:g5:first-look-at-cells}, nous pouvons ouvrir pour de bon la
machinerie de la \omterm{def:g3:flowers-fruits-seeds:parts}{fleur}.

\section{Les fleurs ont des parties mâles et femelles}

\begin{proposition}[Les deux rôles de la fleur réunis]\label{prop:g5:flower-to-fruit:roles}
La reproduction des \omterm{def:g1:plants-around-us:plant}{plantes} à \omterm{def:g3:flowers-fruits-seeds:parts}{fleurs} est une reproduction
\emph{sexuée}, exactement au sens de
\cref{def:g4:animal-reproduction:sexual} --- et la plupart des
\omterm{def:g3:flowers-fruits-seeds:parts}{fleurs} portent les deux rôles à la fois~:
\begin{itemize}
  \item les \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{étamines}} sont la partie \omterm{def:g4:animal-reproduction:sexual}{mâle}~: chaque grain de
        \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{pollen}} porte les \omterm{def:g5:first-look-at-cells:cell}{cellules} reproductrices \omterm{def:g4:animal-reproduction:sexual}{mâles} de la
        \omterm{def:g1:plants-around-us:plant}{plante}~;
  \item le \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{pistil}} est la partie \omterm{def:g4:animal-reproduction:sexual}{femelle}~: tout au fond
        reposent les \emph{ovules}\index{ovule}, chacun contenant
        une \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}.
\end{itemize}
Les «~futures \omterm{prop:g3:flowers-fruits-seeds:transform}{graines}~» de
\cref{def:g3:flowers-fruits-seeds:parts} ont maintenant leur vrai
nom~: des \omterm{prop:g5:flower-to-fruit:roles}{ovules}, qui attendent d'être fécondés.
\end{proposition}
\admitted

\begin{remark}[Le même plan que chez les animaux]\label{rem:g5:flower-to-fruit:sameplan}
\omterm{def:g4:animal-reproduction:sexual}{Cellules mâles}, \omterm{def:g4:animal-reproduction:sexual}{cellules œufs}, \omterm{def:g5:flower-to-fruit:steps}{fécondation}~: le vocabulaire de
\cref{ch:g4:animal-reproduction} se traduit mot pour mot. Encore
l'\omterm{prop:g5:first-look-at-cells:principle}{unité du vivant} (\cref{prop:g5:first-look-at-cells:principle}) ---
pâquerettes et dauphins se reproduisent sur un même plan de fond,
si différente que paraisse la machinerie.
\end{remark}

\section{Pollinisation, puis fécondation}

\begin{definition}[Les deux étapes]\label{def:g5:flower-to-fruit:steps}
Faire une \omterm{def:g2:from-seed-to-plant:seed}{graine} demande deux étapes distinctes~:
\begin{enumerate}
  \item la \emph{pollinisation}\index{pollinisation}~: un grain de
        \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} voyage d'une \omterm{def:g3:flowers-fruits-seeds:parts}{étamine} jusqu'au sommet d'un \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} ---
        porté par un insecte
        (\cref{prop:g3:flowers-fruits-seeds:transform}) ou par le
        vent~;
  \item la \emph{fécondation}\index{fécondation (plantes)}~: le
        grain posé fait pousser un fin tube \emph{à travers le
        \omterm{def:g3:flowers-fruits-seeds:parts}{pistil}}, vers le bas~; par ce tube, la \omterm{def:g4:animal-reproduction:sexual}{cellule mâle} atteint
        un ovule et rejoint sa \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}.
\end{enumerate}
La pollinisation est le voyage~; la fécondation est la rencontre.
\end{definition}

\begin{example}[Pourquoi les fleurs non visitées tombent]\label{ex:g5:flower-to-fruit:unvisited}
La règle du verger de \cref{rem:g3:flowers-fruits-seeds:bees} a
maintenant son mécanisme~: pas de visite, pas de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} sur le
\omterm{def:g3:flowers-fruits-seeds:parts}{pistil}~; pas de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen}, pas de tube, pas de \omterm{def:g5:flower-to-fruit:steps}{fécondation}~; et une
\omterm{def:g3:flowers-fruits-seeds:parts}{fleur} non fécondée se fane tout entière --- pas de \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} qui
gonfle, pas de cerise. L'étape manquante était invisible, le voyage
d'une \omterm{def:g5:first-look-at-cells:cell}{cellule} dans un tube --- mais la \omterm{def:g1:plants-around-us:tree}{branche} vide de juillet est
assez visible.
\end{example}

\section{Après la rencontre~: la graine et le fruit}

\begin{proposition}[Ce que devient chaque partie]\label{prop:g5:flower-to-fruit:becomes}
Après la \omterm{def:g5:flower-to-fruit:steps}{fécondation}, la transformation de
\cref{prop:g3:flowers-fruits-seeds:transform} se déroule avec une
précision nouvelle~:
\begin{itemize}
  \item chaque \emph{ovule fécondé} devient une \emph{\omterm{def:g2:from-seed-to-plant:seed}{graine}} ---
        la petite \omterm{def:g1:plants-around-us:plant}{plante} qu'elle contient
        (\cref{def:g2:from-seed-to-plant:seed}) pousse à partir de
        la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée, exactement comme un \omterm{def:g1:animals-around-us:animal}{animal} pousse
        à partir de la sienne
        (\cref{ex:g5:first-look-at-cells:one})~;
  \item le \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{pistil}} qui les entoure gonfle en \emph{\omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}}~;
  \item les \omterm{def:g3:flowers-fruits-seeds:parts}{pétales} et les \omterm{def:g3:flowers-fruits-seeds:parts}{étamines}, leur travail fait, se fanent
        et tombent.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Compter les pépins]\label{ex:g5:flower-to-fruit:pips}
Coupe une pomme en travers~: cinq petites loges, quelques pépins
dans chacune. Chaque pépin est un ovule qu'un tube pollinique a
atteint. Une pomme de travers, avec des loges vides d'un côté,
raconte une \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} mal pollinisée --- trop peu de grains s'y sont
posés pour féconder tous les \omterm{prop:g5:flower-to-fruit:roles}{ovules}, et le \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit} a gonflé de façon
inégale.
\end{example}

\begin{example}[Le pollen porté par le vent]\label{ex:g5:flower-to-fruit:wind}
Les graminées, le blé et les noisetiers n'ont ni \omterm{def:g3:flowers-fruits-seeds:parts}{pétales}, ni parfum,
ni nectar --- rien à offrir aux visiteurs. Leur \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} voyage par le
\emph{vent}~: fabriqué en quantités énormes, fin comme de la
poussière, jeté en l'air pour se poser où la chance le mène. Cette
stratégie échange la précision contre la quantité --- la logique du
nuage de \omterm{def:g4:plants-without-seeds:spore}{spores} de
\cref{rem:g4:plants-without-seeds:sporeseed}, jouée avec du \omterm{def:g3:flowers-fruits-seeds:parts}{pollen}.
(C'est ce \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} en suspension qui, au passage, chatouille les nez
allergiques en juin.)
\end{example}

\begin{method}[Diagnostiquer une plante sans fruits]\label{met:g5:flower-to-fruit:diagnose}
Une \omterm{def:g1:plants-around-us:plant}{plante} a bien fleuri mais ne porte pas de \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits}. Demande-toi,
dans l'ordre~:
\begin{enumerate}
  \item Les pollinisateurs l'ont-ils atteinte~? (À l'intérieur,
        sous filet, sans \omterm{prop:g3:sorting-living-things:groups}{insectes} alentour, ou pendant des semaines
        de floraison froides et pluvieuses~?)
  \item Sinon, pollinise à la main~: touche avec un pinceau doux
        les \omterm{def:g3:flowers-fruits-seeds:parts}{étamines} d'une \omterm{def:g3:flowers-fruits-seeds:parts}{fleur}, puis le \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} d'une autre~;
  \item le gel ou un dégât ont-ils tué les \omterm{def:g3:flowers-fruits-seeds:parts}{fleurs} avant que les
        \omterm{prop:g5:flower-to-fruit:roles}{ovules} aient pu être fécondés
        (\cref{exo:g3:flowers-fruits-seeds:9})~?
  \item Si des \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} se forment maintenant, la machinerie de la
        \omterm{def:g1:plants-around-us:plant}{plante} était saine --- seul le voyage de l'étape 1 avait
        échoué.
\end{enumerate}
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:1}
Quelle partie de la \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} est \omterm{def:g4:animal-reproduction:sexual}{mâle}, et que fabrique-t-elle~?
Laquelle est \omterm{def:g4:animal-reproduction:sexual}{femelle}, et que contient-elle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:2}
Définis la \omterm{def:g5:flower-to-fruit:steps}{pollinisation} et la \omterm{def:g5:flower-to-fruit:steps}{fécondation}, et dis laquelle vient en
premier.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:3}
Raconte le voyage d'un grain de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen}, de l'\omterm{def:g3:flowers-fruits-seeds:parts}{étamine} à la \omterm{def:g4:animal-reproduction:sexual}{cellule
œuf}, en quatre étapes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:4}
Après la \omterm{def:g5:flower-to-fruit:steps}{fécondation}~: que devient chaque ovule fécondé~? Que
devient le \omterm{def:g3:flowers-fruits-seeds:parts}{pistil}~? Qu'arrive-t-il aux \omterm{def:g3:flowers-fruits-seeds:parts}{pétales}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:5}
Qu'est-ce que chaque pépin de pomme, dans le vocabulaire de ce
chapitre~? Que raconte une pomme de travers avec des loges vides~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:6}
Comment les graminées et les noisetiers se pollinisent-ils sans
\omterm{def:g3:flowers-fruits-seeds:parts}{pétales}, sans parfum et sans nectar~? Quel est le prix de leur
méthode~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:7}
Pourquoi une \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} qu'aucun insecte n'a visitée ne donne-t-elle
d'ordinaire pas de \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}~? Nomme l'étape manquante.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:8}
À toi d'essayer (avec une \omterm{def:g1:plants-around-us:plant}{plante} en \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} de balcon ou de fenêtre)~:
pollinise deux \omterm{def:g3:flowers-fruits-seeds:parts}{fleurs} à la main avec un pinceau doux, comme dans
\cref{met:g5:flower-to-fruit:diagnose}, et marque-les d'un fil. Les
\omterm{def:g3:flowers-fruits-seeds:parts}{fleurs} marquées donnent-elles des \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} plus souvent que les
autres~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:flower-to-fruit:9}
Énumère les correspondances entre la reproduction des \omterm{def:g1:plants-around-us:plant}{plantes} et
celle des \omterm{def:g1:animals-around-us:animal}{animaux}~: \omterm{def:g4:animal-reproduction:sexual}{cellules mâles}, \omterm{def:g4:animal-reproduction:sexual}{cellules œufs}, \omterm{def:g5:flower-to-fruit:steps}{fécondation},
première \omterm{def:g5:first-look-at-cells:cell}{cellule} du nouvel individu. Où loge chacun de ces éléments,
dans le cas de la \omterm{def:g3:flowers-fruits-seeds:parts}{fleur}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:flower-to-fruit:10}
Le printemps d'un producteur de cerises a été froid et pluvieux~:
les abeilles ont à peine volé pendant deux semaines, sans qu'il y
ait eu de gel et alors que les arbres ont magnifiquement fleuri.
Prédis la récolte de juillet et explique-la avec
\cref{def:g5:flower-to-fruit:steps}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:flower-to-fruit:11}
Un producteur de tomates sous serre installe une ruche de bourdons à
l'intérieur, alors qu'aucune \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} du dehors n'est accessible depuis
la serre fermée. À l'aide de ce chapitre, explique exactement quel
service vendent les bourdons --- et pourquoi des ventilateurs qui
soufflent de l'air sur les plants peuvent en partie les remplacer
pour les tomates (dont le \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} se décroche facilement), alors que
pour les cerisiers ils ne le pourraient pas.
\end{exercise}
