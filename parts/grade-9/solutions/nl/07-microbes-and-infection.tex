\section*{Hoofdstuk \ref{ch:g9:microbes-and-infection} --- Microben en infectie}

\begin{solution}{exo:g9:microbes-and-infection:1}
\omterm{def:g9:microbes-and-infection:sorted}{Bacteriën} (\omterm{def:g6:cell-unit-of-life:unicellular}{eencelligen} zonder kern), eencellige \omterm{def:g6:decomposers-and-soil:decomposers}{schimmels} zoals
\omterm{ex:g6:producing-our-food:bread}{gist} en het overige eencellige leven uit de sloot --- en \omterm{def:g9:microbes-and-infection:sorted}{virussen},
apart van aard: geen \omterm{def:g5:first-look-at-cells:cell}{cel}, geen eigen leven, alleen verpakte
genetische tekst die de machinerie van een levende \omterm{def:g5:first-look-at-cells:cell}{cel} leent.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:2}
De biljoenen onschadelijke \omterm{def:g9:microbes-and-infection:sorted}{bacteriën} die op de \omterm{def:g1:the-five-senses:sense}{huid} en in de darm
wonen. Diensten: de \omterm{def:g4:journey-of-food:digestion}{vertering} helpen, en de grond bezetten ---
\omterm{prop:g9:microbes-and-infection:mostly}{ziekteverwekkers} van de landingsplaatsen verdringen.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:3}
Overdracht (de druppeltjes van een hoest); binnenkomst (een snee,
de luchtwegen); vermeerdering (delingen die per uur verdubbelen, of
gekaapte cellen die kopiëren); ziekte (het kloppen van de ontstoken
vinger, de rauwe keel van de verkoudheid).
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:4}
De bacterie van de snee vermeerdert zich tussen de cellen in het
warme natte weefsel en vergiftigt haar buurt. Het
verkoudheidsvirus vermeerdert zich binnen in de cellen van de
luchtwegbekleding, waarbij elke gekaapte \omterm{def:g5:first-look-at-cells:cell}{cel} een menigte kopieën
loslaat.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:5}
De \omterm{def:g1:the-five-senses:sense}{huid}: de muur bij de binnenkomst. De bezems en baden van de
bekledingen (de schoonmaak van de luchtwegen, tranen, maagzuur):
binnenkomst en vroege vermeerdering. De \omterm{prop:g9:microbes-and-infection:mostly}{lichaamseigen flora}:
binnenkomst --- de grond bezet. De zeep, het koken en de kou van de
\omterm{def:g1:healthy-habits:hygiene}{hygiëne}: overdracht en vermeerdering.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:6}
De overdracht --- de reiziger van de koorts, op de handen van
artsen meegedragen van de snijzaal naar de kraamafdeling. Pasteur
liet de reizigers zelf zien: \omterm{def:g6:producing-our-food:fermentation}{microben} als de verwekkers van zuur
worden, ziekte en \omterm{def:g9:microbes-and-infection:stages}{infectie} --- en hitte als hun stopper.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:7}
Antibiotica hongeren \omterm{def:g9:microbes-and-infection:sorted}{bacteriën} uit en breken ze --- zelfstandige
cellen met een eigen machinerie om te vergiftigen. Een
verkoudheidsvirus heeft geen eigen machinerie: het draait op de
\emph{jouwe}, en een medicijn dat op zijn werking mikt, mikt op die
van jouw cellen --- en daarom laat het medicijn van de bacterie het
\omterm{def:g9:microbes-and-infection:sorted}{virus} onaangeroerd.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:8}
Wassen voor het eten: blokkeert de overdracht van hand naar \omterm{def:g4:journey-of-food:tract}{mond}.
Wassen na het toilet: blokkeert de lus van uitgang naar ingang van
de darmweg. Hoesten in een zakdoek (en niezen in de \omterm{def:g1:my-body:joint}{elleboog}):
blokkeert de druppeloverdracht bij de bron. (De regels over tanden
en voedsel blokkeren de vermeerderingsplekken in de \omterm{def:g4:journey-of-food:tract}{mond}.)
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:9}
Ziekteverwekker: voedselvergiftigende \omterm{def:g9:microbes-and-infection:sorted}{bacteriën}, op de kip.
Overdracht: het gerecht zelf. Binnenkomst: de darm. Vermeerderd
door: drie warme uren in de auto --- de regel van de
\omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} voedt een verdubbeling om het halfuur of zo.
Goedkoopste blokkade: de koudeketen --- een koeltas (of het vers
opeten); daarna het goedkoopst: vooraf goed doorbakken.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:10}
Omdat handen koeriers tussen afdelingen zijn: de artsen van
Semmelweis waren naar elke dagelijkse maatstaf schoon en droegen de
moordenaar van de afdeling toch van lijk naar moeder. De wasbeurt
die telt is die \emph{op de grens} --- van rauwe kip naar salade,
van toilet naar tafel --- waar de weg zich anders zou sluiten.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:11}
Het antibioticum valt de vermeerdering aan --- het breekt de
delende \omterm{def:g9:microbes-and-infection:sorted}{bacteriën}. Rafelig gebruikt wordt het een selectierem die
halverwege blijft steken: de gevoelige \omterm{def:g6:producing-our-food:fermentation}{microben} sterven, de
resistente minderheid erft de afdeling --- en fokt de
ziekteverwekker die de volgende patiënt tegenkomt.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:12}
De tekst: een snippertje genetische letters, hetzelfde alfabet als
dat van elke \omterm{def:g5:first-look-at-cells:cell}{cel} (\cref{rem:g9:genes-and-dna:universal}). De
drukpers: de lees- en bouwmachinerie van een levende \omterm{def:g5:first-look-at-cells:cell}{cel}, die de
binnengekomen tekst aan het drukken van kopieën van zichzelf zet.
De rand van het leven: alleen doet het \omterm{def:g9:microbes-and-infection:sorted}{virus} niets van de lijst van
\cref{prop:g1:living-or-not:signs} --- geen voeden, geen groeien,
geen delen; geleend draait het een parodie op voortplanting.
Moeilijk met medicijnen te raken: er is geen virusmachinerie om te
vergiftigen --- alleen de drukpers, en de drukpers ben jij.
\end{solution}

\begin{solution}{pb:g9:microbes-and-infection:1}
\textbf{1.} Overdracht via druppeltjes --- gehoest en gesproken ---
met hulp van gedeelde oppervlakken bij de tafeltennisbatjes;
binnenkomst via de luchtwegen.

\textbf{2.} Die van de vermeerdering: de gekaapte bekledingscellen
van elk geval drukken kopieën die de volgende gevallen inzaaien ---
de verdubbeling is de geleende machinerie van het \omterm{def:g9:microbes-and-infection:sorted}{virus}, die zich
opstapelt.

\textbf{3.} De ziekte. Uit de schade van gekaapte, bezwijkende
bekledingscellen --- en deels uit het gevecht van de verdediging
zelf, zoals het volgende hoofdstuk uitwerkt.

\textbf{4.} Omdat bijeenkomsten het druppelbereik van elke klas een
uur lang met elkaar verbinden: de overdracht schaalt met de
drukte, en de twee dichtste mengmomenten van de school waren de weg
tussen de afdelingen.

\textbf{5.} Zieke leerlingen naar huis: de overdracht bij de bron
weggenomen. Ramen: de druppeldoses verdund --- overdracht. Handen
wassen: de weg via oppervlakken doorgesneden --- overdracht.
Materiaal afnemen: de weg via gedeelde voorwerpen doorgesneden ---
overdracht. (Allemaal op dezelfde goedkoopste fase gericht.)

\textbf{6.} ``Antibiotica doden \omterm{def:g9:microbes-and-infection:sorted}{bacteriën}; deze verwekker is een
\omterm{def:g9:microbes-and-infection:sorted}{virus} dat op de eigen cellen van de kinderen draait --- het
medicijn kan er niet bij, en het toch rondstrooien zou alleen een
selectiezeef over de lichaamseigen \omterm{def:g9:microbes-and-infection:sorted}{bacteriën} van iedereen laten
lopen en resistentie fokken voor de dag waarop antibiotica echt
nodig zijn.''

\textbf{7.} Verwekker: voedselvergiftigende \omterm{def:g9:microbes-and-infection:sorted}{bacteriën}. Overdracht:
het nagerecht. Binnenkomst: de darm. Vermeerdering: ongekoelde uren
--- warmte, vocht, voedsel en tijd, alles toegestaan. Blokkade: de
koelkast --- de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} geweigerd; de twee uitbraken delen
een school en verder niets.

\textbf{8.} Een uitbraakkaart is de overdracht zichtbaar gemaakt:
wie waar zat tekent de weg --- rijen, partners, bijeenkomsten ---
en de weg laat, eenmaal getekend, zien waar je moet knippen.
Willekeurig behandelen bestrijdt gevallen; de kaart bestrijdt de
\omterm{def:g6:how-plants-spread:dispersal}{verspreiding}.

\textbf{9.} Overdracht: de druppeltjes van het indexgeval, eerst de
rij en de partners, via de bijeenkomsten de school in. Binnenkomst:
de luchtwegen, klas voor klas. Vermeerdering: verdubbelingen om de
twee dagen, week één door. Ziekte en uitdoven: de koortsen bereiken
hun top, de maatregelen knippen de wegen door, de vijver van
vatbaren dunt uit --- de curve buigt omlaag.

\textbf{10.} De blokkades van de maatregelen hongerden de wegen
uit --- en de vijver zelf was kleiner dan de lijst: leerlingen die
de immuniteit van de verwante stam van vorige winter meedroegen
waren niet te ontsteken, en een uitbraak sterft wanneer er te
weinig vatbaren over zijn: het geheugen van het volgende
hoofdstuk, alvast getoond.

\textbf{11.} Die van Semmelweis: knip de koerierswegen door ---
thuisblijven, gewassen handen, afgenomen batjes. Die van Pasteur:
ken de reiziger --- noem de verwekker, weiger hem de
\omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}, en richt het middel op zijn \omterm{def:g5:biodiversity:species}{soort} en niet op de
gewoonte.

\textbf{12.} ``Uitbraken zijn het goedkoopst te stoppen bij de
overdracht --- zeep, lucht, afstand en een gesloten koeltas
verslaan elk medicijn dat op de vermeerdering wacht.''
\end{solution}
