\chapter{رحلة الغذاء}\label{ch:g4:journey-of-food}

لقمة خبز واحدة: تُمضغ، وتُبتلع --- وتختفي. فإلى أين؟ في هذه الليلة
سيكون جزء من ذلك الخبز يعين رجليك على الجري وعظامك على النمو، وهذا
يعني أنه لا بد أن يخرج بطريقة ما \emph{من} الأنبوب الذي ابتُلع
فيه \emph{إلى} بقية الجسم. فتتبّع اللقمة: رحلتها من أجمل حكايات
الجسم.

\section{الأنبوب الهضمي}

\begin{definition}[الأنبوب الهضمي]\label{def:g4:journey-of-food:tract}
يسافر الغذاء عبر \emph{الأنبوب الهضمي}\index{أنبوب هضمي}، وهو
أنبوب واحد طويل يمتد في الجسم من طرف إلى طرف:
\begin{enumerate}
  \item \emph{الفم}\index{فم}، حيث تطحن الأسنان
        (\cref{ch:g2:teeth-and-eating-well}) ويرطّب
        \emph{اللعاب}\index{لعاب}؛
  \item و\emph{المريء}\index{مريء}، وهو المزلق من الحلق إلى
        المعدة؛
  \item و\emph{المعدة}\index{معدة}، وهي كيس عضلي يُخضّ فيه الغذاء
        ويُمزج \omterm{def:g4:journey-of-food:digestion}{بالعصارات الهاضمة} ساعات؛
  \item و\emph{المعي الدقيق}\index{معي دقيق}، وهو أنبوب مطوي طوله
        أمتار عدة، وفيه يتم \omterm{def:g4:journey-of-food:digestion}{الهضم} ويعبر خير الغذاء إلى الجسم؛
  \item و\emph{المعي الغليظ}\index{معي غليظ}، وفيه يُستردّ الماء
        مما بقي، قبل أن تغادر البقايا الجسم.
\end{enumerate}
\end{definition}

\begin{example}[توقيت الرحلة]\label{ex:g4:journey-of-food:timing}
تقضي لقمة الخبز ثوانيَ في الفم، وثانيتين تنزلق في \omterm{def:g4:journey-of-food:tract}{المريء}، وساعات
تُخضّ في \omterm{def:g4:journey-of-food:tract}{المعدة}، وساعات أخرى تشقّ طريقها في \omterm{def:g4:journey-of-food:tract}{المعي الدقيق}، ويومًا أو
أكثر تتم فيه في \omterm{def:g4:journey-of-food:tract}{المعي الغليظ}. والرحلة كلها تستغرق يومًا أو يومين
في العادة.
\end{example}

\section{الهضم: من الغذاء إلى المغذيات}

\begin{definition}[الهضم والمغذيات]\label{def:g4:journey-of-food:digestion}
\emph{الهضم}\index{هضم} هو تحويل الغذاء، على امتداد الأنبوب، إلى
مكوّنات صغيرة بسيطة بما يكفي لدخول الجسم: وهي
\emph{المغذيات}\index{مغذيات}. فالأسنان وخضّ \omterm{def:g4:journey-of-food:tract}{المعدة} تفتّت الغذاء؛
و\emph{العصارات الهاضمة}\index{عصارة هاضمة} --- ومنها \omterm{def:g4:journey-of-food:tract}{اللعاب} ---
تفكّكه أكثر، قطعة قطعة، حتى يصير ما كان خبزًا مزيجًا سائلًا من
المغذيات.
\end{definition}

\begin{example}[الخبز يحلو]\label{ex:g4:journey-of-food:sweet}
امضغ قطعة خبز عاديًّا مضغًا طويلًا جدًّا --- عدًّا بطيئًا إلى الستين
--- فتظهر حلاوة خفيفة. ولم يُضف شيء: بل \omterm{def:g4:journey-of-food:tract}{اللعاب} يفكّك الخبز منذ
الآن إلى قطع أبسط وأحلى. وهكذا تكون قد ذقت \omterm{def:g4:journey-of-food:digestion}{الهضم} وهو يبدأ.
\end{example}

\begin{proposition}[العبور إلى الجسم]\label{prop:g4:journey-of-food:absorb}
داخل الأنبوب ما زال، بمعنى ما، خارج الجسم --- كثقب كعكة الدونات.
والعبور يحدث في \emph{\omterm{def:g4:journey-of-food:tract}{المعي الدقيق}}: فعبر جدرانه الرقيقة الشديدة
الطيّ تنتقل \omterm{def:g4:journey-of-food:digestion}{المغذيات} إلى \emph{الدم}\index{دم}، وهو يحملها إلى كل
جزء من الجسم (وهي خدمة التوصيل التي يستكشفها
\cref{ch:g4:heart-and-blood}). أما ما لا يستطيع العبور --- كالألياف
والقشور --- فيسافر أبعد، ويفقد ماءه في \omterm{def:g4:journey-of-food:tract}{المعي الغليظ}، ويغادر
فضلات.
\end{proposition}
\admitted

\begin{remark}[لماذا الطيّات والأمتار]\label{rem:g4:journey-of-food:surface}
العبور يحتاج إلى مساحة جدار، \omterm{def:g4:journey-of-food:tract}{والمعي الدقيق} مبني لذلك: أمتار من
الطول، وجدران مطوية ومعادة الطيّ في الداخل. ولو بُسطت مسطحة لفرشت
مساحة العبور غرفة صغيرة --- وهي محشورة في وسطك. فالأعمال الكبيرة
في الجسم تنال مساحات كبيرة مطوية في حيّز صغير.
\end{remark}

\section{إعانة الهضم على عمله}

\begin{method}[عادات صديقة للهضم]\label{met:g4:journey-of-food:habits}
\begin{enumerate}
  \item امضغ كما ينبغي: \omterm{def:g4:journey-of-food:tract}{فالمعدة} لا أسنان لها --- وما يتخطاه الفم
        تعاني منه \omterm{def:g4:journey-of-food:tract}{المعدة}؛
  \item وكُلْ في هدوء، في مواعيد منتظمة؛
  \item واشرب الماء مع الوجبات؛
  \item وكُلْ أطعمة غنية بالألياف --- خضرًا وفواكه وخبزًا كاملًا:
        فالألياف تُبقي البقايا متحركة؛
  \item وأجّل اللعب الشاق أو السباحة إلى ما بعد استقرار الوجبة:
        \omterm{def:g4:journey-of-food:tract}{فالمعدة} الكادحة \omterm{def:g1:my-body:parts}{والرجلان} الكادحتان تتنافسان.
\end{enumerate}
\end{method}

\begin{example}[محقّق وجع البطن]\label{ex:g4:journey-of-food:ache}
غداء ازدُرد ازدرادًا، ثم عَدْو: وهذا وجع البطن الكلاسيكي بعد
الظهر. فقد ابتُلعت الوجبة نصف ممضوغة (والقاعدة الأولى مخروقة)،
واستُعجلت \omterm{def:g4:journey-of-food:tract}{المعدة} الهاضمة إلى سباق (والقاعدة الخامسة كذلك). والعلاج
ليس دواءً بل أدبًا --- مع أنبوب المرء الهضمي.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:1}
سمِّ أجزاء \omterm{def:g4:journey-of-food:tract}{الأنبوب الهضمي} بالترتيب، من الفم إلى المخرج.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:2}
ماذا يحدث للغذاء في \omterm{def:g4:journey-of-food:tract}{المعدة}، وكم يبقى فيها تقريبًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:3}
ما \omterm{def:g4:journey-of-food:digestion}{الهضم}؟ وما \omterm{def:g4:journey-of-food:digestion}{المغذيات}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:4}
في أي جزء تعبر \omterm{def:g4:journey-of-food:digestion}{المغذيات} إلى \omterm{prop:g4:journey-of-food:absorb}{الدم}؟ وماذا يحدث لما لا يستطيع
العبور؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:5}
لماذا يحلو الخبز حلاوة خفيفة إذا مُضغ طويلًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:6}
لماذا كان \omterm{def:g4:journey-of-food:tract}{المعي الدقيق} بهذا الطول وبهذا الطيّ في داخله؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:7}
أعطِ ثلاث عادات صديقة \omterm{def:g4:journey-of-food:digestion}{للهضم} وسبب كل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:8}
جرّب (على المائدة): وقّت وجبتك القادمة. كم يقضي غذاؤك في فمك ---
وكيف يقارَن ذلك بالأيام التي تنتظره؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:9}
شخّص وجع البطن في \cref{ex:g4:journey-of-food:ache}: أي عادتين
خُرقتا، وما الذي تغيّره غدًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:10}
``الغذاء يدخل الجسم لحظة ابتلاعه.'' صحّح ذلك وفق
\cref{prop:g4:journey-of-food:absorb}: أين يدخل الغذاء الجسم
حقًّا، وما الذي يجب أن يحدث له قبل ذلك؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:journey-of-food:11}
عشب البقرة غذاء أقسى بكثير من خبزك --- وللبقرة أربع حجرات معدية،
وهي تمضغ غذاءها مرتين، وتؤوي كائنات حية معينة في أمعائها تفكّك
العشب. استعمل \cref{def:g4:journey-of-food:digestion}
و\cref{prop:g2:what-animals-eat:match} لتشرح لماذا يحتاج آكل العشب
إلى هذه الآلات الأثقل.
\end{exercise}
