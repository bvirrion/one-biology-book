\section*{अध्याय \ref{ch:g7:organs-at-work} --- काम पर लगे अंग}

\begin{solution}{exo:g7:organs-at-work:1}
\omterm{prop:g4:journey-of-food:absorb}{रक्त} से ज़्यादा \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} और ज़्यादा \omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़} ली जाती है; और ज़्यादा
\omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} तथा ज़्यादा ऊष्मा छोड़ी जाती है।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:2}
\omterm{prop:g4:journey-of-food:absorb}{रक्त} का तैयार शर्करा-पोषक --- यानी \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} के जलने का रोज़मर्रा का
ईंधन। \omterm{def:g3:bones-and-muscles:muscle}{पेशी} उसे अपने भीतर से बहते \omterm{prop:g4:journey-of-food:absorb}{रक्त} से लेती है (और उसमें, उससे पीछे,
\omterm{def:g4:journey-of-food:digestion}{पाचन} की पहुँचाई हुई ख़ुराक भरी होती है)।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:3}
\omterm{def:g3:bones-and-muscles:muscle}{पेशी} में घुसते \omterm{prop:g4:journey-of-food:absorb}{रक्त} की तुलना उसमें से निकलते \omterm{prop:g4:journey-of-food:absorb}{रक्त} से की जाती है। आराम
में निकलता \omterm{prop:g4:journey-of-food:absorb}{रक्त} \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} और \omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़} का बस एक मामूली हिस्सा छोड़ आया
होता है; कड़ी मेहनत में वह पहुँचाने लायक़ ज़्यादातर \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} गँवाकर, और
\omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} से कहीं ज़्यादा भरकर निकलता है। यह फ़र्क़ ही \omterm{def:g3:bones-and-muscles:muscle}{पेशी} का
बिल है।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:4}
साँस गहरी और तेज़ होती है; \omterm{def:g4:heart-and-blood:heart}{दिल} तेज़ और ज़ोर से धड़कता है; और \omterm{prop:g4:journey-of-food:absorb}{रक्त} का
रुख़ मोड़ा जाता है --- काम करती \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} तथा ठंडक करती \omterm{def:g1:the-five-senses:sense}{त्वचा} की \omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ}
चौड़ी होती हैं, जबकि आराम करती पाचन-नली की सिकुड़ जाती हैं।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:5}
ईंधन जलने से हमेशा गरमाहट निकलती है, और काम करती \omterm{def:g3:bones-and-muscles:muscle}{पेशी} शरीर की भट्ठी है।
कँपकँपी वे \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} हैं जो उसी गरमाहट के सिवा कुछ न पैदा करने के लिए काम
पर लगा दी गई हैं --- यानी माँग पर तपिश।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:6}
मेहनत ख़त्म होने के बाद \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} और साँस की धीरे-धीरे वापसी। शरीर \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन}
के भंडार फिर से भर रहा होता है, मेहनत का कचरा साफ़ कर रहा होता है, और
जमा हुई ऊष्मा उतार रहा होता है।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:7}
आराम लगभग $72$; पठार $160$ के पास; और आलेख पर \omterm{def:g7:organs-at-work:recovery}{बहाली} मोटे तौर पर पाँच से
छह मिनट। प्रशिक्षण आराम वाला मान घटाता है, उसी मेहनत के लिए पठार नीचा
करता है, और \omterm{def:g7:organs-at-work:recovery}{बहाली} छोटी कर देता है।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:8}
दोनों ही रुख़ मोड़ने के असर हैं। खाने के बाद पाचन-नली \omterm{prop:g4:journey-of-food:absorb}{रक्त} का एक बड़ा
हिस्सा थामे रहती है, इसलिए बाक़ी शरीर सुस्त चलता है; और तब दौड़ लगाने से
दोनों मंज़िलें पहुँचाई जाने वाली चीज़ों के लिए होड़ करने लगती हैं ---
यानी वह चुभन, और भरे पेट दौड़ने के ख़िलाफ़ वह पुराना नियम।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:9}
चौड़ी हुई \omterm{def:g1:the-five-senses:sense}{त्वचा} की \omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ} भट्ठी की ऊष्मा शरीर की उस सतह तक भेजने का
फ़रमान हैं जो ऊष्मा-पंखा है: यानी ऊष्मा का बाहर की ओर पहुँचाया जाना, जो
काम करती \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} की ज़्यादा तपने वाली समस्या हल करता है। वह लाली ठंडक
वाले तंत्र का चलना है, नाकाम होना नहीं।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:10}
\omterm{def:g4:heart-and-blood:heart}{दिल} हर धड़कन पर ज़्यादा \omterm{prop:g4:journey-of-food:absorb}{रक्त} चलाता है --- रोज़मर्रा की निशानी, कम आराम
वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी}। \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} बढ़ती हैं और ज़्यादा ईंधन जमा करती हैं --- वही चढ़ाई
आसान लगती है। आपूर्ति के जाल की महीन \omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ} बढ़ती हैं और साँस गहरी
होती है --- मेहनत के दौरान बोलना मुमकिन रहता है, और \omterm{def:g7:organs-at-work:recovery}{बहाली} छोटी हो जाती
है।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:11}
$68$ वाली आराम की \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} और तीन मिनट वाली मशीन प्रशिक्षित है। आराम वाली
\omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} उस \omterm{def:g4:heart-and-blood:heart}{दिल} की गवाही देती है जो हर धड़कन पर ज़्यादा चलाता है; और \omterm{def:g7:organs-at-work:recovery}{बहाली}
का समय उन आपूर्ति-लकीरों की, जो मेहनत के उधार जल्दी चुकता कर देती हैं।
दोनों मिलकर \cref{prop:g7:organs-at-work:training} वाली दोबारा बनावट के
दो सिरे हैं।
\end{solution}

\begin{solution}{exo:g7:organs-at-work:12}
दौड़ में \omterm{def:g1:my-body:parts}{टाँग} की \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} की \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} सिकुड़न को ताक़त देने के लिए
\omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़} को \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} के साथ जलाती हैं --- यही \omterm{def:g7:what-is-respiration:respiration}{श्वसन} का मक़सद है --- और
उनका बिल (ज़्यादा ईंधन, ज़्यादा \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} भीतर; ज़्यादा \omterm{def:g7:what-is-respiration:respiration}{कार्बन
डाइऑक्साइड}, ऊष्मा बाहर) उसी वक़्त चुकाना पड़ता है। चुकाने वाली सेवा \omterm{prop:g4:journey-of-food:absorb}{रक्त}
है: फेफड़ों और आँत पर लदा हुआ, दौड़ते \omterm{def:g4:heart-and-blood:heart}{दिल} से तेज़ पंप किया हुआ, \omterm{def:g1:my-body:parts}{टाँगों}
की ओर मोड़ा हुआ --- वह ईंधन और \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} पहुँचाता है और \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड}
ढोकर ले जाता है --- एक दौड़, तीन अध्याय, और हिसाब एक।
\end{solution}

\begin{solution}{pb:g7:organs-at-work:1}
\textbf{1.} गुज़रते \omterm{prop:g4:journey-of-food:absorb}{रक्त} से उसकी \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} और \omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़} की खींच कई गुना हो
जाती है; और उसी क़दम से उसकी \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} तथा ऊष्मा की पैदावार
चढ़ जाती है --- यही कड़ी मेहनत का बिल है।

\textbf{2.} क्योंकि वे एक ही आपूर्ति-लकीर की दो मंज़िलें हैं: साँस
फेफड़ों पर \omterm{prop:g4:journey-of-food:absorb}{रक्त} फिर से लादती है, और \omterm{def:g4:heart-and-blood:heart}{दिल} उस लदे हुए \omterm{prop:g4:journey-of-food:absorb}{रक्त} को \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} तक
पहुँचाता है। एक सिरे पर बढ़ी माँग दोनों की रफ़्तार बढ़ा देती है।

\textbf{3.} काम करती \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} भट्ठियाँ हैं; उनकी ऊष्मा को निकलना ही है।
\omterm{prop:g4:journey-of-food:absorb}{रक्त} उसे \omterm{def:g1:the-five-senses:sense}{त्वचा} तक ले जाता है, जिसकी चौड़ी हुई \omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ} लाल पड़कर उसे
हवा में उतार देती हैं --- भीतर भट्ठी, सतह पर ऊष्मा-पंखा।

\textbf{4.} क्योंकि उधार बाक़ी हैं: \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} के भंडार फिर भरने हैं, कचरा
साफ़ करना है, ऊष्मा उतारनी है। \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} उतनी ही गिरती है जितने हिसाब चुकते
जाते हैं --- और वही गिरना \omterm{def:g7:organs-at-work:recovery}{बहाली} है।

\textbf{5.} आराम $84 \to 72$: एक ऐसा \omterm{def:g4:heart-and-blood:heart}{दिल}, जो हर धड़कन पर ज़्यादा चलाने
के लिए दोबारा बना है। मेहनत $168 \to 150$: वही सीढ़ियाँ अब कम में पड़
रही हैं --- यानी मज़बूत \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} और महीन \omterm{def:g4:heart-and-blood:vessels}{वाहिकाओं} की भरपूर आपूर्ति। \omterm{def:g7:organs-at-work:recovery}{बहाली}
$9 \to 5$ मिनट: पूरी शृंखला अपने उधार जल्दी चुकाती है। यह दोबारा बनावट
सत्र भर की कसरत ने की।

\textbf{6.} या तो छात्र ख सत्र की शुरुआत में ही प्रशिक्षित था --- अपनी
प्रशिक्षित तली के पास पहुँची संख्याएँ कम ही हिलती हैं --- या फिर उसने
सत्र की कसरत छोड़ दी और बस एक टिकी हुई बिना प्रशिक्षण वाली मशीन बनाए
रखी। फ़ैसला करने वाला तथ्य: कक्षा के बाहर उसकी हरकत (कहीं और खेल हो तो
पहली पढ़त पक्की; कुछ न हो तो दूसरी --- और हफ़्ता 1 की उसकी अच्छी
संख्याएँ पहले ही एक ओर झुकी हुई हैं)।

\textbf{7.} रुख़ मोड़ने वाला नियम: खाने के बाद पाचन-नली \omterm{prop:g4:journey-of-food:absorb}{रक्त} पर दावा
कर लेती है, इसलिए आराम वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} ऊँची और मेहनत सहने की ताक़त कम पढ़ी
जाती है। कार्य-विधि: दोपहर के खाने से पहले नापो, या उसके पूरे दो घंटे
बाद --- और हमेशा वही समय।

\textbf{8.} निष्पक्ष परीक्षण: सिर्फ़ एक ही तरह से लिए गए मापों की तुलना
की जा सकती है --- हफ़्ते से हफ़्ते की, और विद्यार्थी से विद्यार्थी की।

\textbf{9.} ``बारह हफ़्तों ने छात्र क की छत ऊपर उठा दी: वही मेहनत अब
$150$ धड़कनों में पड़ती है, $168$ में नहीं, और वह भी आठ धड़कन नीची आराम
वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} से। और उसकी मशीन अब पाँच मिनट में अपने उधार चुका देती है,
जबकि सितंबर में नौ लगते थे --- \omterm{def:g7:organs-at-work:recovery}{बहाली} ही सबसे साफ़ कमाई है।''

\textbf{10.} आम तौर पर चिंता से ठीक उलटी बात: प्रशिक्षित \omterm{def:g4:heart-and-blood:heart}{दिल} हर धड़कन
पर ज़्यादा \omterm{prop:g4:journey-of-food:absorb}{रक्त} चलाता है, इसलिए वह नीचे ही सुस्ताता है --- साइकिल-सवार
की शांत $50$ वही काम करती है जो बिना प्रशिक्षण वाले की $75$। (कोई भी
फ़िक्र डॉक्टर की चीज़ है, पर नीची-और-चुस्त ही आम कहानी है।)

\textbf{11.} \omterm{def:g7:organs-at-work:recovery}{बहाली} का समय पूरी शृंखला को नापता है --- \omterm{def:g4:heart-and-blood:heart}{दिल}, \omterm{def:g7:how-animals-breathe:four}{फेफड़े},
\omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ}, \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} --- और वह भी असली उधार चुकाते हुए; साथ ही वह
प्रशिक्षण के साथ लगातार और साफ़ दिखता हुआ सुधरता है। जबकि मेहनत वाली
\omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} उस दिन की ठीक-ठीक मेहनत और मिज़ाज पर ज़्यादा टिकी होती है। नज़र
रखने के लिए एक ही संख्या चाहिए, तो \omterm{def:g7:organs-at-work:recovery}{बहाली} ज़्यादा ईमानदार सार है।

\textbf{12.} उसका अपना इस्तेमाल: आराम से और नियमित रूप से दोहराई गई
मेहनत ही वह चीज़ है जिसने उसे करने वाली मशीन को दोबारा बनाया।
\end{solution}
