\chapter{التواصل العصبي}\label{ch:g8:nervous-communication}

رسم \Cref{ch:g5:brain-in-command} الحلقة --- بلاغ، وقرار،
وأمر --- وقاس خمس الثانية الذي تستغرقه. وفي هذه السنة
نفتح الكابلات. ما الذي يسافر فعلًا على \omterm{def:g8:nervous-communication:neuron}{العصب}؟ ومم تتكون
\omterm{def:g5:brain-in-command:system}{الأعصاب}؟ وماذا يجري عند الفجوات التي ينتهي عندها سلك
ويبدأ الذي يليه --- الفجوات التي سيتبين أن الذاكرة
وكثيرًا من السموم تعمل فيها جميعًا؟

\section{الأسلاك وخلاياها}

\begin{definition}[العصبونات والرسائل العصبية]\label{def:g8:nervous-communication:neuron}
\omterm{def:g5:first-look-at-cells:cell}{الخلايا} العاملة في \omterm{def:g5:brain-in-command:system}{الجهاز العصبي} هي
\emph{العصبونات}\index{عصبون}
(\cref{rem:g6:cell-unit-of-life:twoways} عددها): لكل
واحد منها جسم خلوي وليف طويل رفيع --- وبعض الألياف يمتد
من \omterm{def:g3:bones-and-muscles:skeleton}{العمود الفقري} إلى إصبع القدم، فهي أطول \omterm{def:g5:first-look-at-cells:cell}{خلايا} الجسم.
و\emph{العصب}\index{عصب} كابل يحزم آلاف هذه الألياف.
وعلى كل ليف تسافر \emph{الرسائل
العصبية}\index{رسالة عصبية}: إشارات كهربائية وجيزة،
تنطلق بسرعة تبلغ \qty{100}{m/s}
(رقم \cref{def:g5:brain-in-command:system}، ومعه حامله
الآن).
\end{definition}

\begin{proposition}[الترميز بالتواتر]\label{prop:g8:nervous-communication:code}
الإشارات نفسها متشابهة كلها --- والذي يتغير هو
\emph{معدلها}. فاللمسة اللطيفة ترسل نقرًا بطيئًا صاعدًا
في الليف الحسي؛ والقوية ترسل قرعًا سريعًا. للغة \omterm{def:g5:brain-in-command:system}{الجهاز
العصبي} كلمة واحدة تكرر بسرعات مختلفة: فالشدة ترمز
تواترًا، على كل خط، ذاهبًا وآيبًا.
\end{proposition}
\admitted

\begin{example}[ثلاثة خطوط، لغة واحدة]\label{ex:g8:nervous-communication:lines}
شخوص الحلقة (\cref{prop:g5:brain-in-command:loop})، وقد
أعيد توصيلهم: \emph{\omterm{def:g8:nervous-communication:neuron}{العصبونات} الحسية} تحمل البلاغات ---
من \omterm{def:g1:the-five-senses:sense}{الجلد} إلى \omterm{def:g5:brain-in-command:system}{النخاع الشوكي}، ومن \omterm{def:g1:the-five-senses:sense}{العين} إلى \omterm{def:g5:brain-in-command:system}{الدماغ}؛
و\emph{\omterm{def:g8:nervous-communication:neuron}{العصبونات} الحركية} تحمل الأوامر --- من النخاع إلى
\omterm{def:g3:bones-and-muscles:muscle}{العضلة}، وتنتهي حيث يلقى الليف \omterm{def:g3:bones-and-muscles:muscle}{العضلة} فيحدد قرعه شدة
\omterm{def:g3:bones-and-muscles:muscle}{الانقباض} (\cref{def:g3:bones-and-muscles:muscle})؛
وبينهما، في \omterm{def:g5:brain-in-command:system}{الدماغ} والنخاع، \emph{\omterm{def:g8:nervous-communication:neuron}{عصبونات} رابطة} لا تحصى
تتولى القرار. منعكسًا كان أم سيمفونية، فالمرور هو
الإشارات نفسها بمعدلات مختلفة على خطوط مختلفة.
\end{example}

\section{الفجوات: المشابك}

\begin{definition}[المشبك]\label{def:g8:nervous-communication:synapse}
\omterm{def:g8:nervous-communication:neuron}{العصبونات} لا يلمس بعضها بعضًا. وحيث يلقى طرف ليف \omterm{def:g8:nervous-communication:neuron}{عصبون}
\omterm{def:g5:first-look-at-cells:cell}{الخلية} التالية تقع فجوة مجهرية --- هي
\emph{المشبك}\index{مشبك}. والإشارة الكهربائية لا تستطيع
قفزها؛ بل يطلق الليف الواصل \emph{رسولًا كيميائيًا} في
الفجوة، فيعبرها، ويرسو على الضفة الأخرى، ثم --- إن وصل
منه ما يكفي --- يبدأ الإشارة من جديد في \omterm{def:g8:nervous-communication:neuron}{العصبون} التالي.
كهربائي على السلك، كيميائي عند كل وصلة: والجهاز كله
يناوب بين الاثنين.
\end{definition}

\begin{proposition}[لماذا فجوات لا لحام]\label{prop:g8:nervous-communication:whygaps}
الفجوات تبدو عيبًا وهي عبقرية الجهاز:
\begin{itemize}
  \item فهي تجعل الوصلات \emph{قابلة للضبط} --- \omterm{def:g8:nervous-communication:synapse}{فالمشبك}
        الذي يعبر كثيرًا يقوى، والذي لا يستعمل يخبو:
        وذلك هو التعلم (``مسالك أفضل'' في
        \cref{rem:g5:brain-in-command:training}، وقد
        وجدت --- إنها \omterm{def:g8:nervous-communication:synapse}{مشابك} مقواة)؛
  \item وهي تجعل الإشارات \emph{وحيدة الاتجاه} ---
        فالرسل لا يطلقون إلا من جهة واحدة، فلا يستطيع
        المرور أن يعود إلى الوراء؛
  \item وهي تتيح \omterm{def:g8:nervous-communication:neuron}{للعصبون} الواحد أن \emph{يزن} آلاف
        المدخلات --- أصواتًا منشطة وأخرى مثبطة تجمع عند
        كل \omterm{def:g5:first-look-at-cells:cell}{خلية} قبل أن تطلق: فقرار الحلقة يؤدى عند
        الوصلات.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[مشابك المشعوذ]\label{ex:g8:nervous-communication:juggler}
مشعوذ \Cref{rem:g5:brain-in-command:training}، وقد فُسّر
أخيرًا: شهر من التمرين أجرى الإشارات نفسها عبر الوصلات
نفسها حتى قويت تلك \omterm{def:g8:nervous-communication:synapse}{المشابك} --- فالتوجيه الذي كان يوزن
بجهد صار يجري موزونًا سلفًا. المهارة كيمياء ترسخت عند
الفجوات؛ و``التمرين يجعل الأمر دائمًا'' قول عن \omterm{def:g8:nervous-communication:synapse}{المشابك}.
\end{example}

\begin{remark}[الفجوات هي النقاط الضعيفة]\label{rem:g8:nervous-communication:vulnerable}
كل ما يؤثر في \omterm{def:g5:brain-in-command:system}{الجهاز العصبي} من الخارج إنما يؤثر عند
\omterm{def:g8:nervous-communication:synapse}{المشابك} --- فهي الموضع الوحيد الذي تصير فيه الإشارة
كيمياء، تُرسى ويمكن خداعها. فمواد التخدير تسكّنها؛
والسموم تعطلها (فبعضها يشل بسد رسول وصلة \omterm{def:g3:bones-and-muscles:muscle}{العضلة}، وبعضها
بإغراقها)؛ والكحول في
\cref{prop:g5:healthy-choices:substances}، وكل مخدر يغير
المزاج أو الإدراك، يعمل بمحاكاة رسل الفجوات أو سدها أو
استنزافها. و\cref{ch:g8:protecting-the-nervous-system}
يبني على هذه النقطة بعينها.
\end{remark}

\section{أسلاك سريعة وبريد بطيء: مقارنة بين جهازين}

\begin{proposition}[الأعصاب في مقابل الهرمونات]\label{prop:g8:nervous-communication:compare}
يظهر الجسم الآن جهازي تواصل، بنيا لعملين متقابلين:
\begin{itemize}
  \item \emph{العصبي}: إشارات في أجزاء من الألف من
        الثانية، على خطوط مخصصة، إلى عناوين دقيقة ---
        \omterm{def:g3:bones-and-muscles:muscle}{عضلة} واحدة، غدة واحدة --- وتزول متى توقفت
        الإشارة: جهاز الالتقاط والكلام والمنعكسات؛
  \item \emph{الهرموني}
        (\cref{def:g8:puberty:hormones}): رسل في دقائق
        إلى أشهر، يذاعون \omterm{prop:g4:journey-of-food:absorb}{بالدم} إلى كل عضو بني ليسمع،
        يأمرون ببرامج بطيئة تشمل الجسم كله --- النمو،
        والدورات، وإعادة بناء \omterm{def:g8:puberty:puberty}{البلوغ}.
\end{itemize}
برق وبريد: الجسم يشغّل الاثنين،
و\cref{ch:g8:hormonal-communication} يفرد للبريد فصلًا
خاصًا به.
\end{proposition}
\admitted

\begin{method}[تشخيص تواصل، جهازًا جهازًا]\label{met:g8:nervous-communication:diagnose}
أمام أي تنسيق في الجسم، عيّن جهازه:
\begin{enumerate}
  \item قس زمنه: أجزاء من الألف من الثانية --- عصبي؛
        دقائق إلى سنوات --- هرموني (أو كلاهما، بالتناوب)؛
  \item وارسم مداه: عنوان واحد دقيق --- عصبي؛ الجسم كله
        --- هرموني؛
  \item وجد طريقه: مسلك أسلاك يمكن تسميته --- عصبي؛
        إذاعة في مجرى \omterm{prop:g4:journey-of-food:absorb}{الدم} --- هرموني؛
  \item وتحقق من دوامه: يزول مع الإشارة --- عصبي؛ يجري
        ما دام الرسول يدور --- هرموني.
\end{enumerate}
\end{method}

\begin{example}[الطريقة على ثلاث حالات]\label{ex:g8:nervous-communication:cases}
الرمشة عند نفخة هواء: أجزاء من الألف من الثانية، وحلقة
عضلية واحدة، ومسلك مسلوك، وتزول في الحال --- عصبي. ودفقة
النمو: سنوات، والجسم كله، ومجرى \omterm{prop:g4:journey-of-food:absorb}{الدم}، وتجري فصولًا ---
هرموني. \omterm{def:g4:heart-and-blood:heart}{والقلب} المتسارع في العدو
(\cref{prop:g7:organs-at-work:supply}): كلاهما بالتناوب
--- فإشارات \omterm{def:g5:brain-in-command:system}{الأعصاب} ترفع الإيقاع في غضون دقة واحدة،
ورسول في مجرى \omterm{prop:g4:journey-of-food:absorb}{الدم} (الاندفاع المشهور الذي تحسه موجة
ساخنة عند الفزع) يبقي تأهب الجسم كله دقائق. البرق أولًا،
والبريد يتبع.
\end{example}

\section{مسألة: منضدة المنعكسات}

\begin{problem}[{مسألة نهاية الأسبوع --- منعكس الركبة،
مزودًا بالأجهزة}]\label{pb:g8:nervous-communication:1}
العرض الكلاسيكي على المنضدة: المتطوع جالسًا، والمطرقة
الصغيرة تحت الرضفة، وركلة القدم اللاإرادية --- وقد زودت
الآن بمؤقتات وبهذا الفصل.

\medskip\noindent\textbf{الجزء الأول --- الدارة.}
\begin{enumerate}
  \item تمد المطرقة وترًا؛ فتركل القدم بعد نحو جزء من
        خمسين من الثانية. اعرض الدارة: أي \omterm{def:g5:biodiversity:species}{أنواع}
        \omterm{def:g8:nervous-communication:neuron}{العصبونات}، وعبر أين --- ولماذا \omterm{def:g5:brain-in-command:system}{الدماغ} ليس على
        هذه الحلقة.
  \item جزء من خمسين من الثانية أسرع من خمس الثانية في
        التقاط الكرة. علّل الفرق بطول الدارة وعدد
        وصلاتها.
  \item لا يستطيع المتطوع كبح الركلة، وإن حذر منها. ماذا
        يقول هذا عن موضع اتخاذ قرار المنعكس --- وعن
        المرور وحيد الاتجاه عند \omterm{def:g8:nervous-communication:synapse}{المشابك}؟
  \item بعد الركلة مباشرة \emph{يحس} المتطوع بالنقرة.
        فسّر رحلة التأخير الثانية الأبطأ.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- تنويعات المنضدة.}
\begin{enumerate}[resume]
  \item انقر بقوة أشد: تشتد الركلة. أي قاعدة ترميز في
        \cref{prop:g8:nervous-communication:code} ظهرت؟
  \item يشرح طبيب الصف أن هذا المنعكس يفحص في كل زيارة
        دورية. ما الذي يشهد به منعكس \omterm{def:g1:my-body:joint}{ركبة} حاضر ومتناظر
        عن النخاع \omterm{def:g5:brain-in-command:system}{والأعصاب} والوصلات؟
  \item المتطوع الذي يطبق فكيه يركل أشد --- وهو تضخيم
        معروف. ماذا يكشف هذا عن \omterm{def:g8:puberty:puberty}{بلوغ} أصوات \omterm{def:g8:nervous-communication:neuron}{العصبونات}
        الرابطة حلقات ``بلا \omterm{def:g5:brain-in-command:system}{دماغ}'' أيضًا؟
  \item يلغي مخدر موضعي في \omterm{def:g1:plants-around-us:parts}{الساق} الركلة والإحساس معًا.
        سمّ نقطة الهجوم، وفق
        \cref{rem:g8:nervous-communication:vulnerable}.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- التقرير.}
\begin{enumerate}[resume]
  \item ارسم المنعكس بوصفه حلقة الخطوات الثلاث في
        \cref{prop:g5:brain-in-command:loop}، مع تحديد
        موضع اختصاره هنا.
  \item قارن بين المنعكس والتقاط الكرة في جدول من
        عمودين: المسلك، والوصلات، والزمن، وقابلية الكبح،
        ودور التمرين.
  \item يسأل القسم الأخير من التقرير: ``لماذا تتخطى
        الحلقات الواقية السريعة \omterm{def:g5:brain-in-command:system}{الدماغ}؟'' أجب بمنطق
        المقلاة الساخنة
        (\cref{exo:g5:brain-in-command:5}).
  \item جملة واحدة للختام: ما الذي أظهرته المنضدة عن لغة
        \omterm{def:g5:brain-in-command:system}{الجهاز العصبي} وأسلاكه ووصلاته في وقت واحد.
\end{enumerate}
\end{problem}
