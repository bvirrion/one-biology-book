\chapter{Otot dan Sendi}\label{ch:g10:muscles-and-joints}

Seorang pemain bola menancapkan kakinya, berputar, lalu jatuh sambil
mencengkeram lututnya. Diagnosisnya, satu jam kemudian: \omterm{def:g10:muscles-and-joints:joint}{ligamen} yang
robek. Tidak ada tulang yang patah, tidak ada otot yang gagal; sebuah
tali jaringan sepanjang beberapa sentimeter, yang tugasnya menjaga dua
tulang tetap sejajar, telah menyerah oleh puntiran yang tidak pernah
menjadi rancangannya. Untuk memahami apa yang robek, dan mengapa
lututnya akan menuntut berbulan-bulan untuk pulih, kita perlu tahu
\omterm{def:g10:muscles-and-joints:joint}{sendi} itu terbuat dari apa, bagaimana otot menariknya, dan gaya apa
yang mengalir lewatnya ketika tubuh bermassa tujuh puluh kilogram
mendarat pada satu kaki.

\section{Sendinya}

\begin{definition}[Sendi]\label{def:g10:muscles-and-joints:joint}
Sebuah \emph{sendi}\index{sendi} adalah tempat dua tulang bertemu. Pada
sendi yang dapat bergerak seperti lutut, pinggul, bahu atau siku, ujung
tulangnya terlapisi selapis \emph{tulang rawan}\index{tulang rawan}
yang licin, setebal beberapa milimeter, dan terkurung dalam kantong
tertutup, yaitu \emph{kapsul}, yang selaputnya mengeluarkan
\emph{cairan sinovial} yang melumasi permukaannya. Pita jaringan yang
liat, yaitu \emph{ligamen}\index{ligamen}, membentang dari tulang ke
tulang melintasi sendinya dan membatasi gerakannya pada gerakan yang
dituju. Pada lutut, dua sabit tulang rawan, yaitu \emph{meniskus},
duduk di antara tulangnya dan menyebarkan bebannya.
\end{definition}

\begin{proposition}[Untuk apa masing-masing bagiannya]\label{prop:g10:muscles-and-joints:parts}
\omterm{def:g10:muscles-and-joints:joint}{Tulang rawan} dan cairan sinovial membuat permukaannya meluncur dengan
gesekan yang lebih kecil daripada es di atas es; meniskusnya menyebarkan
beban yang jika tidak akan terpusat pada beberapa milimeter persegi;
\omterm{def:g10:muscles-and-joints:joint}{ligamennya} membolehkan lutut membengkok dan lurus tetapi menahan
puntiran dan pembengkokan menyamping; kapsulnya menahan cairannya tetap
di dalam. Tiap bagian punya kegagalannya sendiri: \omterm{def:g10:muscles-and-joints:joint}{tulang rawan} aus
(artrosis), meniskus robek oleh puntiran ketika lututnya membengkok,
\omterm{def:g10:muscles-and-joints:joint}{ligamen} terkilir atau putus ketika \omterm{def:g10:muscles-and-joints:joint}{sendinya} dipaksa melampaui
jangkauannya.
\end{proposition}
\admitted

\begin{example}[Terkilir, robek, lepas sendi]\label{ex:g10:muscles-and-joints:injuries}
Puntirlah pergelangan kaki di tepi trotoar: \omterm{def:g10:muscles-and-joints:joint}{ligamen} di sisi luarnya
terentang melampaui batasnya --- itulah \emph{keseleo}; \omterm{def:g10:muscles-and-joints:joint}{sendinya}
membengkak karena kapsulnya berdarah dan cairannya menumpuk. Tariklah
kuat-kuat otot paha belakang yang dingin: sebagian seratnya putus ---
itulah \emph{robekan} otot. Jatuhlah dengan lengan terjulur: kepala
humerusnya dapat meninggalkan mangkuknya --- itulah \emph{lepas \omterm{def:g10:muscles-and-joints:joint}{sendi}},
dengan \omterm{def:g10:muscles-and-joints:joint}{ligamen} dan kapsulnya terentang atau robek seiring lepasnya.
Tidak satu pun berupa tulang patah; semuanya dapat memakan waktu lebih
lama untuk sembuh daripada tulang patah.
\end{example}

\section{Otot menarik, secara berpasangan}

\begin{definition}[Otot rangka dan tendon]\label{def:g10:muscles-and-joints:muscle}
Sebuah \emph{otot rangka}\index{otot rangka} adalah organ yang terbuat
dari ribuan \omterm{def:g10:cells-common-unit:cell}{sel} panjang, yaitu \emph{serat otot}, yang terikat menjadi
berkas; tiap seratnya memendek ketika dirangsang sebuah saraf. Pada
kedua ujungnya ototnya menyempit menjadi \emph{tendon}\index{tendon},
yaitu tali jaringan yang tidak dapat berkerut dan melekat pada tulang.
Sebuah otot hanya dapat \emph{menarik}, dengan memendek, dan hanya
sepanjang panjangnya sendiri: ia menggerakkan \omterm{def:g10:muscles-and-joints:joint}{sendi} dengan menarik
tulang di seberangnya.
\end{definition}

\begin{proposition}[Pasangan yang berlawanan]\label{prop:g10:muscles-and-joints:antagonists}
Karena otot tidak dapat mendorong, setiap gerakan sebuah \omterm{def:g10:muscles-and-joints:joint}{sendi} menuntut
dua otot atau dua kelompok yang bekerja berlawanan: sebuah
\emph{fleksor} yang membengkokkan \omterm{def:g10:muscles-and-joints:joint}{sendinya} dan sebuah \emph{ekstensor}
yang meluruskannya. Ketika yang satu berkerut, yang lain mengendur dan
terentang; untuk menahan sebuah sikap, keduanya berkerut bersama. Bisep
membengkokkan siku dan trisep meluruskannya; kuadrisep di depan paha
meluruskan lutut, otot paha belakang di belakangnya membengkokkannya.
\end{proposition}
\admitted

\begin{proposition}[Tuas dan gaya]\label{prop:g10:muscles-and-joints:lever}
Tulang yang berputar pada sebuah \omterm{def:g10:muscles-and-joints:joint}{sendi} adalah sebuah tuas. \omterm{def:g10:muscles-and-joints:muscle}{Tendon}
ototnya melekat pada jarak pendek $d$ dari titik tumpunya, sedangkan
bebannya bekerja pada jarak $D$ yang lebih besar; agar seimbang ototnya
harus menarik dengan gaya
\[
F_{\text{otot}} = F_{\text{beban}} \times \frac{D}{d},
\]
beberapa kali bebannya. Susunan itu menukar gaya dengan kecepatan dan
jangkauan: pemendekan kecil ototnya menghasilkan gerakan tangan atau
kaki yang besar dan cepat, dengan harga berupa gaya yang besar dalam
\omterm{def:g10:muscles-and-joints:muscle}{tendon} dan \omterm{def:g10:muscles-and-joints:joint}{sendinya}.
\end{proposition}
\admitted

\begin{example}[Menjinjing sebuah tas]\label{ex:g10:muscles-and-joints:bag}
Sebuah tas \qty{5}{kg} (beratnya \qty{49}{N}) tergantung pada tangan,
\qty{35}{cm} dari siku; \omterm{def:g10:muscles-and-joints:muscle}{tendon} bisepnya melekat \qty{5}{cm} darinya.
Bisepnya menarik dengan $49 \times 35/5 = \qty{343}{N}$ --- tujuh kali
bebannya, yaitu berat \qty{35}{kg} --- dan \omterm{def:g10:muscles-and-joints:joint}{sendi} sikunya tertekan
menyatu dengan sekitar \qty{300}{N}. Otot, \omterm{def:g10:muscles-and-joints:muscle}{tendon} dan \omterm{def:g10:muscles-and-joints:joint}{tulang rawan}
memang dibangun untuk gaya sebesar ini; yang merusaknya adalah gaya
mendadak dalam arah yang salah, atau beban yang berulang melampaui apa
yang dapat diperbaiki jaringannya.
\end{example}

\section{Di dalam otot}

\begin{proposition}[Struktur sebuah serat otot]\label{prop:g10:muscles-and-joints:fibre}
\omterm{def:g10:muscles-and-joints:muscle}{Serat otot} adalah satu \omterm{def:g10:cells-common-unit:cell}{sel} raksasa, selebar \num{10}--\qty{100}{\micro m}
dan panjangnya sampai beberapa sentimeter, dengan banyak \omterm{def:g10:cells-common-unit:organelle}{inti sel}. Ia
padat dengan \emph{miofibril} yang sejajar, yang masing-masing berupa
rantai satuan berulang tempat dua jenis filamen \omterm{def:g10:chemistry-of-life:families}{protein}, yang tebal dan
yang tipis, saling menyisip. Kerutannya adalah meluncurnya filamen
tipis di antara filamen tebal, yang dijalankan ATP: tiap satuannya
memendek sepersekian, miofibrilnya sebanyak jumlahnya, dan seratnya
ikut bersamanya. \omterm{def:g10:cells-common-unit:organelle}{Mitokondria} di antara miofibrilnya memasok ATP-nya;
ujung saraf pada tiap seratnya memberi perintahnya.
\end{proposition}
\admitted

\begin{definition}[Jenis serat]\label{def:g10:muscles-and-joints:types}
\omterm{def:g10:muscles-and-joints:muscle}{Serat otot} hadir dalam dua jenis utama. \emph{Serat lambat} berkerut
dengan lambat dan lemah tetapi tanpa lelah: kaya \omterm{def:g10:cells-common-unit:organelle}{mitokondria} dan
kapiler, merah oleh \omterm{def:g10:chemistry-of-life:families}{protein} penyimpan oksigen, seratnya berjalan dengan
respirasi dan berkuasa pada otot penegak tubuh serta pada atlet
ketahanan. \emph{Serat cepat} berkerut dengan cepat dan kuat tetapi
lelah dalam satu menit: lebih miskin \omterm{def:g10:cells-common-unit:organelle}{mitokondria}, lebih pucat, seratnya
bersandar pada simpanan fosfat dan \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} serta berkuasa pada
pelari cepat. Perbandingannya dalam sebuah otot sebagian besar
diwariskan; latihan lebih mengubah ukuran dan perlengkapan seratnya
daripada jumlahnya.
\end{definition}

\begin{example}[Membaca sebuah biopsi otot]\label{ex:g10:muscles-and-joints:biopsy}
Pewarna bagi \omterm{def:g10:cell-metabolism:metabolism}{enzim} \omterm{def:g10:cells-common-unit:organelle}{mitokondria} pada irisan tipis otot paha
memperlihatkan mosaik: serat gelap (lambat) dan serat pucat (cepat).
Cuplikan seorang pelari maraton 80\% gelap; cuplikan pelari cepat 30\%;
cuplikan orang tak terlatih kira-kira separuh-separuh. Pewarna yang
sama pada pelari yang sama setelah setahun latihan lari cepat
memperlihatkan serat pucatnya membesar tetapi tidak bertambah banyak.
\end{example}

\section{Kelelahan, kerusakan, perbaikan}

\begin{proposition}[Batas ototnya]\label{prop:g10:muscles-and-joints:limits}
Otot gagal dengan tiga cara yang berbeda. \emph{Kelelahan}: gayanya
menurun ketika pasokan ATP-nya tidak dapat mengimbangi --- dalam satu
menit pada serat cepat, sepanjang berjam-jam pada serat lambat --- dan
pulih dengan istirahat. \emph{Kram}: kerutan mendadak, berkepanjangan
dan tak disengaja, yang sering berasal dari kekurangan cairan atau
hilangnya garam. \emph{Cedera}: serat yang robek oleh gaya yang
melampaui kekuatannya, lazimnya ketika ototnya dingin, lelah, atau
terentang sambil berkerut. Serat diperbaiki dalam hitungan pekan dari
\omterm{def:g10:cells-common-unit:cell}{sel} cadangan dalam ototnya; \omterm{def:g10:muscles-and-joints:muscle}{tendon} atau \omterm{def:g10:muscles-and-joints:joint}{ligamen}, yang buruk pasokan
darahnya, diperbaiki dalam hitungan bulan.
\end{proposition}
\admitted

\begin{method}[Menganalisis sebuah cedera otot dan sendi]\label{met:g10:muscles-and-joints:analyse}
\begin{enumerate}
  \item Kenali gerakan yang menyebabkannya dan arahnya: apakah \omterm{def:g10:muscles-and-joints:joint}{sendinya}
        digerakkan dalam jangkauannya (persoalan otot) atau melampauinya
        (persoalan \omterm{def:g10:muscles-and-joints:joint}{ligamen} atau kapsul)?
  \item Kenali jaringannya: \omterm{def:g10:muscles-and-joints:muscle}{serat otot} robek (nyeri pada perut ototnya,
        lemas); \omterm{def:g10:muscles-and-joints:joint}{ligamen} terkilir (nyeri dan bengkak pada \omterm{def:g10:muscles-and-joints:joint}{sendinya}, tidak
        mantap); \omterm{def:g10:muscles-and-joints:muscle}{tendon} meradang di bawah beban yang berulang; \omterm{def:g10:muscles-and-joints:joint}{tulang
        rawan} aus diam-diam selama bertahun-tahun.
  \item Perkirakan gayanya: pakai aturan tuasnya; ingatlah bahwa
        mendarat atau mengerem melipatgandakan berat tubuh beberapa kali.
  \item Ramalkan pemulihannya dari pasokan darah jaringannya: otot
        dalam hitungan pekan, \omterm{def:g10:muscles-and-joints:joint}{ligamen} dan \omterm{def:g10:muscles-and-joints:muscle}{tendon} dalam hitungan bulan,
        \omterm{def:g10:muscles-and-joints:joint}{tulang rawan} nyaris tidak sama sekali.
\end{enumerate}
\end{method}

\begin{remark}[Mengapa pemanasan berhasil]\label{rem:g10:muscles-and-joints:warmup}
Sepuluh menit gerakan lembut menaikkan suhu ototnya satu atau dua
derajat, yang membuat \omterm{def:g10:chemistry-of-life:families}{proteinnya} lebih dapat merentang dan \omterm{def:g10:cell-metabolism:metabolism}{enzimnya}
lebih cepat; pemanasan itu membuka kapiler yang akan mengantarkan
oksigen \cref{ch:g10:heart-lungs-effort}, dan menebalkan selaput cairan
sinovialnya pada \omterm{def:g10:muscles-and-joints:joint}{tulang rawannya}. Otot dingin yang diminta berkerut
maksimal adalah penyebab robekan yang terlazim; \omterm{def:g10:muscles-and-joints:joint}{sendi} yang dibebani
sebelum cairannya menyebar akan lebih cepat aus.
\end{remark}

\section{Soal: Gaya dalam Sebuah Lutut}

\begin{problem}[{Soal akhir pekan --- lutut seorang pemain bola yang
dibongkar: tuas kakinya, tarikan otot pahanya, ligamen yang robek, dan
angka yang dibawa sebuah tendon pada setiap langkah}]
\label{pb:g10:muscles-and-joints:1}
Karim, \qty{70}{kg}, mencederai lututnya ketika berputar tajam dengan
kaki tertancap. Ambil $g = \qty{9.8}{N/kg}$. \omterm{def:g10:muscles-and-joints:muscle}{Tendon} kuadrisepnya
melekat pada tibia \qty{5}{cm} dari titik tumpu lututnya; dorongan
tanah pada kakinya bekerja sekitar \qty{20}{cm} darinya ketika lututnya
membengkok dalam, seperti saat mendarat.

\medskip
\noindent\textbf{Bagian I --- Berdiri dan melangkah.}
\begin{enumerate}
  \item Hitung berat Karim.
  \item Ketika berdiri pada satu kaki dengan lutut sedikit membengkok,
        tanah mendorong kakinya ke atas dengan seluruh beratnya, yang
        bekerja \qty{10}{cm} dari titik tumpu lututnya. Hitung gaya
        dalam \omterm{def:g10:muscles-and-joints:muscle}{tendon} kuadrisepnya.
  \item Ketika menaiki tangga, lututnya membengkok lebih jauh dan gaya
        tanahnya bekerja \qty{20}{cm} dari titik tumpunya. Hitung lagi
        gaya \omterm{def:g10:muscles-and-joints:muscle}{tendonnya}.
  \item Nyatakan kedua hasilnya sebagai kelipatan berat tubuhnya, dan
        jelaskan mengapa tangga melelahkan pahanya.
  \item Kelompok otot mana yang menjadi lawan kuadrisepnya, dan apa
        yang dikerjakannya selama langkahnya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Pendaratannya.}
\begin{enumerate}[resume]
  \item Ketika mendarat dari sundulan, tanah mendorong dengan tiga kali
        berat Karim, \qty{20}{cm} dari titik tumpunya. Hitung gaya
        \omterm{def:g10:muscles-and-joints:muscle}{tendonnya}.
  \item Penampang \omterm{def:g10:muscles-and-joints:muscle}{tendonnya} \qty{1.2}{cm^2}. Hitung tegangan di
        dalamnya, dalam newton tiap milimeter persegi, dan bandingkan
        dengan tegangan putusnya yang sekitar \qty{100}{N/mm^2}.
  \item Permukaan \omterm{def:g10:muscles-and-joints:joint}{sendinya} tertekan menyatu oleh kira-kira gaya
        \omterm{def:g10:muscles-and-joints:muscle}{tendonnya} ditambah gaya tanahnya. Hitung beban itu, lalu
        tekanan pada \omterm{def:g10:muscles-and-joints:joint}{tulang rawannya} jika luas sentuhnya \qty{6}{cm^2}
        dengan meniskusnya dan \qty{1}{cm^2} tanpa meniskusnya.
  \item Turunkan mengapa lutut yang telah kehilangan meniskusnya
        mengauskan \omterm{def:g10:muscles-and-joints:joint}{tulang rawannya} lebih cepat.
  \item Jelaskan apa yang disumbangkan cairan sinovialnya selama
        pendaratan itu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Cederanya.}
\begin{enumerate}[resume]
  \item Kaki Karim tertancap dan tubuhnya berputar: arah gerakan mana
        yang dipaksakan pada lututnya, dan jaringan mana yang dibangun
        untuk menahannya?
  \item Mengapa kuadrisepnya, yang cukup kuat untuk menarik dengan
        beberapa kilonewton, tidak melindungi \omterm{def:g10:muscles-and-joints:joint}{ligamennya}?
  \item Lututnya membengkak dalam satu jam. Apa yang mengisinya, dan
        bagian \omterm{def:g10:muscles-and-joints:joint}{sendi} mana yang jebol?
  \item Pemindaian memperlihatkan \omterm{def:g10:muscles-and-joints:joint}{ligamennya} putus dan sebuah meniskus
        robek. Dengan \cref{met:g10:muscles-and-joints:analyse},
        ramalkan waktu pemulihan masing-masing dan berikan alasannya
        dari pasokan darahnya.
  \item Karim bertanya mengapa ahli bedahnya ingin menguatkan otot paha
        belakangnya sebelum operasinya. Jawablah dengan
        \cref{prop:g10:muscles-and-joints:antagonists}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Tiap langkah berarti.}
\begin{enumerate}[resume]
  \item Karim melangkah sekitar 8000 kali sehari; separuhnya pada kaki
        yang cedera. Dengan gaya pada pertanyaan 2, berapa kali sehari
        \omterm{def:g10:muscles-and-joints:muscle}{tendonnya} membawa sedikitnya beban itu?
  \item Sepanjang setahun, berapa jumlah pembebanannya seluruhnya?
        Jelaskan mengapa \omterm{def:g10:muscles-and-joints:muscle}{tendon} yang meradang oleh pemakaian berlebih
        (tendinitis) memerlukan istirahat alih-alih kekuatan.
  \item Selama satu pertandingan ia mendarat dari lompatan sekitar 40
        kali. Bandingkan beban seluruhnya yang terbawa dalam pendaratan
        (pertanyaan 6) dengan beban yang terbawa dalam langkah
        sehariannya.
  \item Berlari melipatduakan gaya tanahnya dibandingkan berjalan.
        Mengapa dokter menganjurkan bersepeda dan berenang alih-alih
        berlari bagi lutut yang sedang sembuh?
  \item Nyatakan hasilnya: kelipatan berat tubuh yang dibawa \omterm{def:g10:muscles-and-joints:muscle}{tendon}
        kuadrisepnya di tangga dan saat mendarat, dan satu jaringan
        lutut yang kekuatannya tidak pernah diuji angka itu --- sampai
        putarannya.
\end{enumerate}
\end{problem}
