\chapter{Mendel's Discoveries}\label{ch:g9:mendel}

Gregor Mendel's pea experiments revealed patterns of inheritance long
before \omterm{def:g9:dna:dna}{DNA} was known. This chapter uses \omterm{def:g9:mendel:alleles}{dominant} and \omterm{def:g9:mendel:alleles}{recessive alleles},
Punnett squares and simple ratios.

\section{Mendel's approach}

\begin{definition}[True-breeding]\label{def:g9:mendel:true}
A \emph{true-breeding}\index{true-breeding} line always produces \omterm{def:g2:growing:offspring}{offspring}
like the parents for a \omterm{def:g7:heredity:trait}{trait} when self-fertilised (genetically \omterm{def:g9:mendel:alleles}{homozygous}
for that \omterm{def:g7:heredity:trait}{trait} in modern terms).
\end{definition}

\begin{proposition}[Particulate inheritance]\label{prop:g9:mendel:part}
\omterm{def:g7:heredity:trait}{Traits} are passed as discrete hereditary factors (\omterm{def:g9:dna:gene}{genes}), not by permanent
blending of parental essences. Factors can be hidden in one generation and
reappear in the next.
\end{proposition}
\admitted

\begin{example}[Pea traits]\label{ex:g9:mendel:pea}
Mendel tracked \omterm{def:g6:reproduction:plant}{seed} shape, \omterm{def:g6:reproduction:plant}{seed} colour, \omterm{def:g2:plant-parts:main}{flower} colour and more --- \omterm{def:g7:heredity:trait}{traits}
with clear alternative forms.
\end{example}

\section{Dominant and recessive}

\begin{definition}[Alleles and zygosity]\label{def:g9:mendel:alleles}
For a \omterm{def:g9:dna:gene}{gene} with two \omterm{def:g7:heredity:allele}{alleles}, an individual may be
\emph{homozygous}\index{homozygous} (two the same) or
\emph{heterozygous}\index{heterozygous} (two different). A
\emph{dominant}\index{dominant allele} \omterm{def:g8:genetics:gene}{allele} masks a
\emph{recessive}\index{recessive allele} in the heterozygote's \omterm{def:g8:genetics:pheno}{phenotype}.
\end{definition}

\begin{definition}[Parental and F generations]\label{def:g9:mendel:gen}
Crossing pure parental lines gives the $F_1$ generation; crossing $F_1$
among themselves gives the $F_2$.
\end{definition}

\begin{proposition}[Classic monohybrid ratio]\label{prop:g9:mendel:3to1}
For a single \omterm{def:g9:dna:gene}{gene} with complete dominance, $F_2$ \omterm{def:g8:genetics:pheno}{phenotypes} often
approximate $3$ \omterm{def:g9:mendel:alleles}{dominant} : $1$ \omterm{def:g9:mendel:alleles}{recessive} when many \omterm{def:g2:growing:offspring}{offspring} are counted.
\end{proposition}
\admitted

\section{Punnett squares}

\begin{method}[Monohybrid Punnett square]\label{met:g9:mendel:punnett}
\begin{enumerate}
  \item Write parent \omterm{def:g8:genetics:pheno}{genotypes} (e.g.\ $Aa \times Aa$).
  \item Put possible \omterm{def:g6:reproduction:sexual}{gamete} \omterm{def:g7:heredity:allele}{alleles} on top and side.
  \item Fill boxes with \omterm{def:g2:growing:offspring}{offspring} \omterm{def:g8:genetics:pheno}{genotypes}.
  \item Convert to \omterm{def:g8:genetics:pheno}{phenotype} counts using dominance rules.
\end{enumerate}
\end{method}

\begin{example}[Seed shape]\label{ex:g9:mendel:round}
$R$ round \omterm{def:g9:mendel:alleles}{dominant} to $r$ wrinkled. $Rr \times Rr$ yields \omterm{def:g8:genetics:pheno}{genotypes}
$1\,RR : 2\,Rr : 1\,rr$ and \omterm{def:g8:genetics:pheno}{phenotypes} about $3$ round : $1$ wrinkled.
\end{example}

\begin{definition}[Test cross]\label{def:g9:mendel:test}
A \emph{test cross}\index{test cross} mates an individual of \omterm{def:g9:mendel:alleles}{dominant}
\omterm{def:g8:genetics:pheno}{phenotype} with a \omterm{def:g9:mendel:alleles}{homozygous} \omterm{def:g9:mendel:alleles}{recessive} to reveal whether the unknown is
\omterm{def:g9:mendel:alleles}{homozygous} or \omterm{def:g9:mendel:alleles}{heterozygous}.
\end{definition}
