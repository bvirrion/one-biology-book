\chapter{A viagem do alimento}\label{ch:g4:journey-of-food}

Uma mordida de pão: mastigada, engolida --- sumiu. Sumiu para onde?
Até hoje à noite, parte desse pão vai estar ajudando as suas \omterm{def:g1:my-body:parts}{pernas}
a correr e os seus \omterm{def:g3:bones-and-muscles:skeleton}{ossos} a crescer, o que quer dizer que ele precisa
de algum jeito \emph{sair} do tubo em que foi engolido e \emph{entrar}
no resto do corpo. Siga a mordida: a viagem dela é uma das melhores
histórias do corpo.

\section{O tubo digestivo}

\begin{definition}[O tubo digestivo]\label{def:g4:journey-of-food:tract}
O alimento viaja pelo \emph{tubo digestivo}\index{tubo digestivo},
um tubo comprido que atravessa o corpo inteiro:
\begin{enumerate}
  \item a \emph{boca}\index{boca}, onde os dentes trituram
        (\cref{ch:g2:teeth-and-eating-well}) e a
        \emph{saliva}\index{saliva} umedece;
  \item o \emph{esôfago}\index{esôfago}, o escorregador da garganta
        até o estômago;
  \item o \emph{estômago}\index{estômago}, uma bolsa musculosa onde
        o alimento é batido e misturado com \omterm{def:g4:journey-of-food:digestion}{sucos digestivos} durante
        horas;
  \item o \emph{intestino delgado}\index{intestino delgado}, um tubo
        dobrado de vários metros, onde a \omterm{def:g4:journey-of-food:digestion}{digestão} termina e as coisas
        boas do alimento passam para o corpo;
  \item o \emph{intestino grosso}\index{intestino grosso}, onde a
        água é recuperada do que sobra, antes de os restos saírem do
        corpo.
\end{enumerate}
\end{definition}

\begin{example}[Cronometrando a viagem]\label{ex:g4:journey-of-food:timing}
A mordida de pão passa segundos na boca, uns dois segundos
escorregando pelo \omterm{def:g4:journey-of-food:tract}{esôfago}, algumas horas sendo batida no \omterm{def:g4:journey-of-food:tract}{estômago},
várias horas a mais atravessando o \omterm{def:g4:journey-of-food:tract}{intestino delgado} e até um dia ou
mais terminando no \omterm{def:g4:journey-of-food:tract}{intestino grosso}. A viagem inteira costuma levar
de um a dois dias.
\end{example}

\section{Digestão: do alimento aos nutrientes}

\begin{definition}[Digestão e nutrientes]\label{def:g4:journey-of-food:digestion}
A \emph{digestão}\index{digestão} é a transformação do alimento, ao
longo do tubo, em componentes pequenos e simples o bastante para
entrar no corpo: os \emph{nutrientes}\index{nutriente}. Os dentes e
o batimento do \omterm{def:g4:journey-of-food:tract}{estômago} quebram o alimento; os \emph{sucos
digestivos}\index{suco digestivo} --- a \omterm{def:g4:journey-of-food:tract}{saliva} entre eles ---
desmontam o alimento ainda mais, pedaço por pedaço, até que aquilo
que era pão vire uma mistura líquida de nutrientes.
\end{definition}

\begin{example}[O pão fica doce]\label{ex:g4:journey-of-food:sweet}
Mastigue um pedaço de pão puro por muito tempo --- uma contagem
lenta até sessenta --- e um docinho aparece. Nada foi acrescentado:
a \omterm{def:g4:journey-of-food:tract}{saliva} já está desmontando o pão em pedaços mais simples e mais
doces. Você acabou de provar o gosto da \omterm{def:g4:journey-of-food:digestion}{digestão} começando.
\end{example}

\begin{proposition}[Atravessar para dentro do corpo]\label{prop:g4:journey-of-food:absorb}
O interior do tubo é, num certo sentido, ainda do lado de fora do
corpo --- como o buraco de uma rosquinha. A travessia acontece no
\emph{\omterm{def:g4:journey-of-food:tract}{intestino delgado}}: pelas paredes finas e muito dobradas dele,
os \omterm{def:g4:journey-of-food:digestion}{nutrientes} passam para o \emph{sangue}\index{sangue}, que os leva
a todas as partes do corpo (o serviço de entrega que o
\cref{ch:g4:heart-and-blood} explora). O que não consegue atravessar
--- fibras, cascas --- segue viagem, perde a água no \omterm{def:g4:journey-of-food:tract}{intestino grosso}
e sai como fezes.
\end{proposition}
\admitted

\begin{remark}[Por que as dobras e os metros]\label{rem:g4:journey-of-food:surface}
Atravessar exige superfície de parede, e o \omterm{def:g4:journey-of-food:tract}{intestino delgado} é
construído para isso: metros de comprimento e paredes dobradas e
redobradas por dentro. Esticada, a superfície de travessia forraria
um cômodo pequeno --- e está compactada dentro da sua barriga. No
corpo, trabalhos grandes ganham superfícies grandes dobradas em
pouco espaço.
\end{remark}

\section{Ajudar a digestão a fazer o trabalho dela}

\begin{method}[Hábitos amigos da digestão]\label{met:g4:journey-of-food:habits}
\begin{enumerate}
  \item Mastigue direito: o \omterm{def:g4:journey-of-food:tract}{estômago} não tem dentes --- o que a boca
        pula, o \omterm{def:g4:journey-of-food:tract}{estômago} tem que enfrentar;
  \item coma com calma, em horários regulares;
  \item beba água nas refeições;
  \item coma alimentos ricos em fibras --- verduras, frutas, pão
        integral: a fibra mantém os restos em movimento;
  \item deixe a brincadeira pesada ou a natação para depois de a
        refeição assentar: um \omterm{def:g4:journey-of-food:tract}{estômago} trabalhando duro e \omterm{def:g1:my-body:parts}{pernas}
        trabalhando duro competem entre si.
\end{enumerate}
\end{method}

\begin{example}[O detetive da dor de barriga]\label{ex:g4:journey-of-food:ache}
Almoço devorado às pressas e depois uma corrida: a clássica dor de
barriga da tarde. A refeição foi engolida meio mastigada (regra 1
quebrada) e o \omterm{def:g4:journey-of-food:tract}{estômago} em plena \omterm{def:g4:journey-of-food:digestion}{digestão} foi obrigado a entrar numa
corrida (regra 5). O conserto não é remédio, é boa educação --- com o
próprio \omterm{def:g4:journey-of-food:tract}{tubo digestivo}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:1}
Diga as partes do \omterm{def:g4:journey-of-food:tract}{tubo digestivo} em ordem, da boca até a saída.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:2}
O que acontece com o alimento no \omterm{def:g4:journey-of-food:tract}{estômago}, e quanto tempo mais ou
menos ele fica lá?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:3}
O que é a \omterm{def:g4:journey-of-food:digestion}{digestão}? O que são os \omterm{def:g4:journey-of-food:digestion}{nutrientes}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:4}
Em que parte os \omterm{def:g4:journey-of-food:digestion}{nutrientes} atravessam para o \omterm{prop:g4:journey-of-food:absorb}{sangue}? O que acontece
com o que não consegue atravessar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:5}
Por que o pão mastigado por muito tempo fica levemente doce?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:6}
Por que o \omterm{def:g4:journey-of-food:tract}{intestino delgado} é tão comprido e tão dobrado por dentro?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:7}
Dê três hábitos amigos da \omterm{def:g4:journey-of-food:digestion}{digestão} e o motivo de cada um.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:8}
Faça a experiência (à mesa): cronometre a sua próxima refeição.
Quanto tempo o seu alimento passa na sua boca --- e como isso se
compara com os dias que ele tem pela frente?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:9}
Diagnostique a dor de barriga do \cref{ex:g4:journey-of-food:ache}:
que dois hábitos foram quebrados, e o que você mudaria amanhã?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:10}
``O alimento entra no corpo no instante em que você engole.''
Corrija isso com a \cref{prop:g4:journey-of-food:absorb}: onde o
alimento entra no corpo de verdade, e o que precisa acontecer com
ele antes?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:journey-of-food:11}
O capim da vaca é um alimento bem mais duro que o seu pão --- e a
vaca tem quatro compartimentos no \omterm{def:g4:journey-of-food:tract}{estômago}, mastiga o alimento duas
vezes e hospeda \omterm{def:g1:living-or-not:living}{seres vivos} ajudantes no intestino para desmontar o
capim. Usando a \cref{def:g4:journey-of-food:digestion} e a
\cref{prop:g2:what-animals-eat:match}, explique por que quem come
capim precisa dessa maquinaria mais pesada.
\end{exercise}
