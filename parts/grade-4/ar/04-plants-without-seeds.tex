\chapter{نباتات بلا بذور}\label{ch:g4:plants-without-seeds}

الوجه المظلل من الجدار العتيق مفروش بالطحلب؛ وتحت أشجار الزان تنشر
\omterm{prop:g4:plants-without-seeds:damp}{السرخسيات} سعفها الأخضر الكبير. فتّش فيها الصيف كله: فلن تجد \omterm{def:g3:flowers-fruits-seeds:parts}{زهرة}
ولا \omterm{prop:g3:flowers-fruits-seeds:transform}{ثمرة} ولا \omterm{def:g2:from-seed-to-plant:seed}{بذرة} --- أبدًا. ومع ذلك ينتشر الطحلب والسرخس انتشارًا
حسنًا. وتبين أن \omterm{def:g1:plants-around-us:plant}{للنباتات} أكثر من طريقة لصنع الجيل التالي.

\section{السرخسيات والطحالب: التكاثر بالأبواغ}

\begin{definition}[البوغ]\label{def:g4:plants-without-seeds:spore}
\emph{البوغ}\index{بوغ} حبيبة مجهرية تُصنع بأعداد هائلة، ويمكن أن
ينمو منها \omterm{def:g1:plants-around-us:plant}{نبات} جديد. وهو، بخلاف \omterm{def:g2:from-seed-to-plant:seed}{البذرة}
(\cref{def:g2:from-seed-to-plant:seed})، ذرّة واحدة \emph{لا \omterm{def:g3:animals-through-seasons:active}{مخزن
غذاء} فيها ولا \omterm{def:g1:plants-around-us:plant}{نبتة} صغيرة جاهزة} --- إنه البداية في أعرى صورها.
\end{definition}

\begin{example}[تحت سعف السرخس]\label{ex:g4:plants-without-seeds:fern}
اقلب سعفة سرخس في أواخر الصيف: فصفوف أنيقة من النقط البنية تصطف
على وجهها السفلي. وكل نقطة كيس فيه آلاف \omterm{def:g4:plants-without-seeds:spore}{الأبواغ}؛ فإذا نضجت انشقّت
الأكياس، وطارت \omterm{def:g4:plants-without-seeds:spore}{الأبواغ} في الهواء كأدقّ الغبار. \omterm{def:g4:plants-without-seeds:spore}{والبوغ} الذي يقع على
أرض ندية مظللة يمكن أن ينمو --- على مراحل --- سرخسة جديدة.
\end{example}

\begin{omfigure}
\includegraphics[width=0.56\linewidth]{images/book1/ai/g4-ferns-sori.jpg}

{\small تحت السعفة: صفوف النقط البنية الأنيقة، كل واحدة كيس فيه
آلاف \omterm{def:g4:plants-without-seeds:spore}{الأبواغ}.}
\end{omfigure}

\begin{proposition}[نباتات الأبواغ تحتاج إلى الندى]\label{prop:g4:plants-without-seeds:damp}
\omterm{def:g1:plants-around-us:plant}{نباتات} \omterm{def:g4:plants-without-seeds:spore}{الأبواغ} --- \emph{الطحالب}\index{طحلب}
و\emph{السرخسيات}\index{سرخس} --- لا تستطيع أن تتكاثر إلا حيث
يكون \emph{نديًّا}: فأبواغها، والمراحل الرقيقة التي تنمو منها، تجفّ
وتموت في الهواء الجاف. ولهذا يفرش الطحلب الجدار الشمالي المظلل ولا
يفرش الجنوبي المشمس قط، ولهذا تزدحم \omterm{prop:g4:plants-without-seeds:damp}{السرخسيات} في الغابات الندية
وضفاف الجداول والآبار.
\end{proposition}
\admitted

\begin{remark}[البوغ ليس بذرة]\label{rem:g4:plants-without-seeds:sporeseed}
كلاهما بداية تسافر، لكن \omterm{def:g2:from-seed-to-plant:seed}{البذرة} عدة نجاة --- غلاف، \omterm{def:g3:animals-through-seasons:active}{ومخزن غذاء}،
\omterm{def:g1:plants-around-us:plant}{ونبتة} جاهزة --- أما \omterm{def:g4:plants-without-seeds:spore}{البوغ} فعارٍ. ومن هنا الأعداد: ملايين \omterm{def:g4:plants-without-seeds:spore}{أبواغ}
السرخس في مقابل عشرات \omterm{def:g2:from-seed-to-plant:seed}{بذور} قرن الفول. فالبدايات العارية رخيصة
الصنع سهلة الضياع؛ والخطتان صدى لسمكة القدّ والفيلة في
\cref{prop:g4:animal-reproduction:strategies}.
\end{remark}

\section{نباتات جديدة بلا أي بذار}

\begin{definition}[التكاثر الخضري]\label{def:g4:plants-without-seeds:vegetative}
كثير من \omterm{def:g1:plants-around-us:plant}{النباتات} يستطيع أيضًا أن يصنع \omterm{def:g1:plants-around-us:plant}{نباتات} جديدة \emph{مباشرة من
قطعة من نفسه} --- بلا \omterm{def:g4:plants-without-seeds:spore}{أبواغ} ولا \omterm{def:g2:from-seed-to-plant:seed}{بذور}، ووالد واحد فحسب. وهذا هو
\emph{التكاثر الخضري}\index{تكاثر خضري}. \omterm{def:g1:plants-around-us:plant}{والنبات} الجديد امتداد حي
للقديم، ويشبهه تمامًا.
\end{definition}

\begin{example}[مدّادات الفراولة]\label{ex:g4:plants-without-seeds:runner}
ترسل \omterm{def:g1:plants-around-us:plant}{نبتة} الفراولة سيقانًا أفقية طويلة رفيعة ---
\emph{مدّادات}\index{مدّاد} --- على التربة. وحيث يلامس المدّاد
الأرض يتجذّر ويطلع أوراقًا، فيصير \omterm{def:g1:plants-around-us:plant}{نبتة} فراولة جديدة كاملة؛ ثم يذبل
المدّاد فيقف الابن حرًّا. فتصير \omterm{def:g1:plants-around-us:plant}{نبتة} واحدة رقعةً كاملة من غير أن
تُبذر \omterm{def:g2:from-seed-to-plant:seed}{بذرة} واحدة.
\end{example}

\begin{example}[الأبصال والدرنات]\label{ex:g4:plants-without-seeds:bulbtuber}
\emph{بصلة}\index{بصلة} الخزامى و\emph{درنة}\index{درنة} البطاطا
مخازن تحت الأرض --- وعُدد بدء أيضًا. فإذا تُركت \omterm{ex:g4:plants-without-seeds:bulbtuber}{البصلة} في الأرض
انقسمت إلى أبصال بنات، كل واحدة خزامى مستقبلية؛ وكل ``\omterm{def:g1:the-five-senses:sense}{عين}'' من
عيون بطاطا نابتة تستطيع أن تنمو \omterm{def:g1:plants-around-us:plant}{نبتة} جديدة كاملة. ولذلك يزرع
البستانيون بطاطا لا \omterm{def:g2:from-seed-to-plant:seed}{بذور} بطاطا.
\end{example}

\begin{method}[أخذ عقلة]\label{met:g4:plants-without-seeds:cutting}
يحاكي البستانيون \omterm{def:g4:plants-without-seeds:vegetative}{التكاثر الخضري} بأخذ
\emph{العُقَل}\index{عقلة}:
\begin{enumerate}
  \item اقطع فرعًا صغيرًا سليمًا بطول شبر من \omterm{def:g1:plants-around-us:plant}{النبتة} الأم (والنعناع
        والصفصاف والغرنوقي تصلح جيدًا)؛
  \item وانزع \omterm{def:g1:plants-around-us:parts}{الأوراق} السفلى وضع الفرع في ماء أو تربة ندية؛
  \item وانتظر: ففي أيام أو أسابيع تنمو \omterm{def:g1:plants-around-us:parts}{جذور} من \omterm{def:g1:plants-around-us:parts}{الساق} المقطوعة؛
  \item ثم اغرسه --- فإذا \omterm{def:g1:plants-around-us:plant}{نبتة} جديدة كاملة مطابقة لوالدها
        الواحد.
\end{enumerate}
\end{method}

\begin{remark}[والد واحد ونسخ مطابقة]\label{rem:g4:plants-without-seeds:copy}
\omterm{def:g2:from-seed-to-plant:seed}{البذور} تمزج والدين
(\cref{rem:g4:animal-reproduction:mix} تصدق على \omterm{def:g1:plants-around-us:plant}{النباتات} أيضًا
عبر \omterm{def:g3:flowers-fruits-seeds:parts}{حبوب اللقاح})؛ أما \omterm{def:g4:plants-without-seeds:vegetative}{التكاثر الخضري} فينسخ واحدًا. ولهذا تُنبَت كل
شجرة من صنف تفاح مسمّى من \omterm{met:g4:plants-without-seeds:cutting}{العُقَل} والتطعيم لا من \omterm{def:g2:from-seed-to-plant:seed}{البذور}: فالبستان
يريد نسخًا مطابقة لشجرة ممتازة واحدة. وثمن النسخ التشابه ---
فبستان من أشجار متطابقة يتشارك مواطن ضعف متطابقة، كما ستبين سنة
لاحقة.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:1}
سمِّ نباتين لا يصنعان بذورًا أبدًا. وكيف يتكاثران؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:2}
أعطِ فرقين بين \omterm{def:g4:plants-without-seeds:spore}{البوغ} \omterm{def:g2:from-seed-to-plant:seed}{والبذرة}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:3}
أين تُصنع \omterm{def:g4:plants-without-seeds:spore}{أبواغ} السرخس؟ صف ما تجده تحت سعفة في أواخر الصيف.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:4}
لماذا ينمو الطحلب على الجدار الشمالي المظلل لا على الجنوبي
المشمس؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:5}
ما \omterm{def:g4:plants-without-seeds:vegetative}{التكاثر الخضري}؟ وكم والدًا يقتضي؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:6}
احكِ حكاية مدّاد فراولة، من أول \omterm{def:g1:plants-around-us:parts}{ساق} إلى \omterm{def:g1:plants-around-us:plant}{نبتة} ابنة قائمة بنفسها.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:7}
لماذا يزرع البستانيون بطاطا بدل أن يبذروا \omterm{def:g2:from-seed-to-plant:seed}{بذور} بطاطا؟ وماذا
تستطيع كل ``\omterm{def:g1:the-five-senses:sense}{عين}'' أن تفعل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:8}
جرّب: خذ عقلة نعناع وفق \cref{met:g4:plants-without-seeds:cutting}
وضعها في كوب ماء على حافة النافذة. فماذا يظهر، وبعد كم من الوقت
تقريبًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:9}
السرخس يصنع ملايين \omterm{def:g4:plants-without-seeds:spore}{الأبواغ}؛ وقرن الفول يصنع عشرات \omterm{def:g2:from-seed-to-plant:seed}{البذور}. اشرح
فرق الأعداد وفق \cref{rem:g4:plants-without-seeds:sporeseed} ---
وسمِّ زوج \omterm{def:g1:animals-around-us:animal}{الحيوانات} الذي تردّد خطتاهما صداهما.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:10}
تجد بستانية \omterm{def:g1:plants-around-us:plant}{نبتات} الفراولة الجديدة مطابقة للأم، بينما تختلف \omterm{def:g1:plants-around-us:plant}{نبتات}
الفول عندها اختلافًا يسيرًا عن والديها. اشرح الملاحظتين وفق
\cref{rem:g4:plants-without-seeds:copy}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:plants-without-seeds:11}
\omterm{def:g1:plants-around-us:tree}{غصن} صفصاف كسره فيضان شتوي، فحملته المياه إلى أسفل النهر وألقته
على ضفة طينية، كثيرًا ما ينمو شجرة صفصاف جديدة. فأي صنف من
التكاثر هذا --- ولماذا تصطفّ الصفصافات على الأنهار كثيرًا؟ وما
الذي فعله الفيضان مما يفعله بستاني
\cref{met:g4:plants-without-seeds:cutting} بالمقصّ؟
\end{exercise}
