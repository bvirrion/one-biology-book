\chapter{Os sistemas reprodutores}\label{ch:g8:reproductive-systems}

O \cref{ch:g8:sexual-reproduction} precisou de \omterm{def:g8:sexual-reproduction:gametes}{gametas}; o
\cref{ch:g8:puberty} ligou as fábricas deles. Este capítulo mapeia as
próprias fábricas: os sistemas reprodutores masculino e feminino --- os
órgãos, os \omterm{def:g8:sexual-reproduction:gametes}{gametas} que cada um produz e o notável ritmo mensal do
sistema feminino. Anatomia direta, com nomes diretos: estes são órgãos
como o \omterm{def:g4:heart-and-blood:heart}{coração} e os \omterm{def:g7:how-animals-breathe:four}{pulmões}, e este é o capítulo deles.

\section{O sistema masculino}

\begin{definition}[O sistema reprodutor masculino]\label{def:g8:reproductive-systems:male}
Os órgãos do sistema masculino:
\begin{itemize}
  \item os dois \emph{testículos}\index{testículo}, mantidos fora do
        calor do corpo, no \emph{escroto} --- as fábricas de \omterm{def:g8:sexual-reproduction:gametes}{gameta}:
        da \omterm{def:g8:puberty:puberty}{puberdade} em diante eles produzem
        \emph{espermatozoides}\index{espermatozoide} sem parar, às
        centenas de milhões;
  \item canais que armazenam e transportam os espermatozoides, aos
        quais se juntam pelo caminho glândulas cujos líquidos os
        alimentam e os carregam --- formando juntos o \emph{sêmen};
  \item o \emph{pênis}\index{pênis}, por onde o sêmen sai do corpo ---
        o mesmo órgão que, no outro serviço dele, carrega a \omterm{def:g7:removing-wastes:kidneys}{urina}
        (\cref{def:g7:removing-wastes:kidneys}); um arranjo de \omterm{def:g7:heart-and-circulation:rooms}{valvas}
        mantém os dois serviços rigorosamente separados.
\end{itemize}
\end{definition}

\begin{example}[O espermatozoide, de perto]\label{ex:g8:reproductive-systems:sperm}
Ao \omterm{def:g5:first-look-at-cells:microscope}{microscópio} (\cref{def:g5:first-look-at-cells:microscope}), um
\omterm{def:g8:reproductive-systems:male}{espermatozoide} é uma \omterm{def:g5:first-look-at-cells:cell}{célula} despida para a viagem: uma cabeça que é
quase só \omterm{def:g6:cell-unit-of-life:parts}{núcleo} --- a contribuição do pai, embalada
(\cref{rem:g5:first-look-at-cells:nucleus}) --- e uma cauda que chicoteia.
É a menor \omterm{def:g5:first-look-at-cells:cell}{célula} do corpo, construída para nadar e provisionada para
dias. A produção corre da \omterm{def:g8:puberty:puberty}{puberdade} até a \omterm{def:g3:stages-of-life:stages}{velhice}: a fábrica masculina
é contínua.
\end{example}

\section{O sistema feminino}

\begin{definition}[O sistema reprodutor feminino]\label{def:g8:reproductive-systems:female}
Os órgãos do sistema feminino, todos na parte baixa do abdome:
\begin{itemize}
  \item os dois \emph{ovários}\index{ovário} --- as reservas e as
        fábricas de \omterm{def:g8:sexual-reproduction:gametes}{gameta}: ao contrário dos \omterm{def:g8:reproductive-systems:male}{testículos}, eles guardam
        o estoque de futuras \emph{\omterm{def:g4:animal-reproduction:sexual}{células reprodutoras femininas}}
        desde o \omterm{def:g2:born-grow-die:cycle}{nascimento} e, a partir da \omterm{def:g8:puberty:puberty}{puberdade}, liberam essas
        \omterm{def:g5:first-look-at-cells:cell}{células} uma de cada vez;
  \item as duas \emph{tubas uterinas}\index{tuba uterina}, tubos com
        franjas que capturam a \omterm{def:g5:first-look-at-cells:cell}{célula} liberada --- o lugar habitual do
        encontro do passo 2 da
        \cref{prop:g8:sexual-reproduction:core};
  \item o \emph{\omterm{prop:g5:human-reproduction:plan}{útero}} (aquele da
        \cref{prop:g5:human-reproduction:plan}), cuja parede muscular
        grossa e cujo revestimento renovável são a hospedagem do
        \omterm{prop:g8:sexual-reproduction:core}{zigoto};
  \item a \emph{vagina}, que liga o \omterm{prop:g5:human-reproduction:plan}{útero} ao lado de fora --- onde o
        sêmen é recebido, e o canal de parto do
        \cref{ex:g5:human-reproduction:birth}.
\end{itemize}
\end{definition}

\begin{example}[A célula reprodutora feminina, de perto]\label{ex:g8:reproductive-systems:egg}
A \omterm{def:g4:animal-reproduction:sexual}{célula reprodutora feminina} é o oposto do \omterm{def:g8:reproductive-systems:male}{espermatozoide} em tudo,
menos no papel: é a \emph{maior} \omterm{def:g5:first-look-at-cells:cell}{célula} do corpo --- um décimo de
milímetro, no limite do que o \omterm{def:g1:the-five-senses:sense}{olho} vê ---, redonda, imóvel, abastecida
de provisões para os primeiros dias da viagem (o princípio da gema da
\cref{rem:g2:animals-grow-up:egg}, na escala da \omterm{def:g5:first-look-at-cells:cell}{célula}). Uma fábrica
produz milhões sem parar; a outra libera uma única \omterm{def:g5:first-look-at-cells:cell}{célula} provisionada
por mês: o números-contra-cuidado da
\cref{prop:g4:animal-reproduction:strategies}, escrito nos próprios
\omterm{def:g8:sexual-reproduction:gametes}{gametas}.
\end{example}

\section{O ciclo}

\begin{proposition}[O ciclo menstrual]\label{prop:g8:reproductive-systems:cycle}
Da \omterm{def:g8:puberty:puberty}{puberdade} até por volta dos cinquenta anos, o sistema feminino roda
um programa que se repete a cada \emph{28 dias}, mais ou menos --- o
\emph{ciclo menstrual}\index{ciclo menstrual}, comandado pela cadeia
hormonal da \cref{def:g8:puberty:hormones}:
\begin{enumerate}
  \item o revestimento do \omterm{prop:g5:human-reproduction:plan}{útero} se reconstrói, grosso e cheio de
        \omterm{prop:g4:journey-of-food:absorb}{sangue} --- a hospedagem preparada;
  \item no meio do ciclo, por volta do dia 14, um \omterm{def:g8:reproductive-systems:female}{ovário} libera uma
        \omterm{def:g4:animal-reproduction:sexual}{célula reprodutora feminina} --- é a
        \emph{ovulação}\index{ovulação}; a \omterm{def:g8:reproductive-systems:female}{tuba uterina} a captura e,
        por cerca de um dia, ela pode ser fecundada;
  \item sem \omterm{def:g5:flower-to-fruit:steps}{fecundação}: o revestimento não é necessário; ele se
        desfaz e sai pela vagina ao longo de alguns dias --- é a
        \emph{menstruação}\index{menstruação} --- e o programa
        recomeça;
  \item com \omterm{def:g5:flower-to-fruit:steps}{fecundação}: o \omterm{prop:g8:sexual-reproduction:core}{zigoto} se instala no revestimento preparado
        (\cref{def:g5:human-reproduction:pregnancy}), o ciclo é
        suspenso e a \omterm{def:g5:human-reproduction:pregnancy}{gravidez} começa.
\end{enumerate}
\end{proposition}
\admitted

\begin{remark}[Conviver com o ciclo]\label{rem:g8:reproductive-systems:living}
Coisas práticas, ditas com clareza: os ciclos são irregulares nos
primeiros anos --- a maquinaria está se afinando; a duração varia de
pessoa para pessoa e de mês para mês; a \omterm{prop:g8:reproductive-systems:cycle}{menstruação} pode trazer
cólicas (o \omterm{prop:g5:human-reproduction:plan}{útero} é um \omterm{def:g3:bones-and-muscles:muscle}{músculo},
\cref{def:g3:bones-and-muscles:muscle}, e ele trabalha durante a
descamação) --- calor, movimento e, quando for preciso, o conselho do
enfermeiro da escola ajudam; e absorventes, tampões e coletores são
simplesmente o equipamento prático da \omterm{prop:g8:reproductive-systems:cycle}{menstruação}, uma prateleira
normal de qualquer farmácia. Nada disso é doença: o ciclo é o sistema
funcionando exatamente como foi construído.
\end{remark}

\begin{method}[Ler os dois sistemas lado a lado]\label{met:g8:reproductive-systems:sides}
Para manter o mapa em ordem, compare por função:
\begin{enumerate}
  \item fábrica de \omterm{def:g8:sexual-reproduction:gametes}{gameta}: \omterm{def:g8:reproductive-systems:male}{testículos} --- \omterm{def:g8:reproductive-systems:female}{ovários};
  \item \omterm{def:g8:sexual-reproduction:gametes}{gameta}: milhões contínuos que nadam --- uma \omterm{def:g5:first-look-at-cells:cell}{célula}
        provisionada por mês;
  \item lugar do encontro: a \omterm{def:g8:reproductive-systems:female}{tuba uterina}, alcançada pela nadada do
        \omterm{def:g8:reproductive-systems:male}{espermatozoide} da vagina através do \omterm{prop:g5:human-reproduction:plan}{útero};
  \item hospedagem: nenhuma no mapa masculino --- o \omterm{prop:g5:human-reproduction:plan}{útero}, preparado
        todo mês, no feminino;
  \item calendário: contínuo desde a \omterm{def:g8:puberty:puberty}{puberdade} --- cíclico, da
        \omterm{def:g8:puberty:puberty}{puberdade} até por volta dos cinquenta.
\end{enumerate}
\end{method}

\section{Problema: Os modelos didáticos da clínica}

\begin{problem}[{Problema de fim de semana --- explicar os sistemas
com dois modelos e um calendário}]\label{pb:g8:reproductive-systems:1}
Um dia aberto de uma unidade de saúde usa três recursos didáticos: um
modelo do sistema masculino, um modelo do sistema feminino e um
calendário grande de 28 dias. Você é quem explica, como voluntário.

\medskip\noindent\textbf{Parte I --- Os dois modelos.}
\begin{enumerate}
  \item Uma visitante percorre o modelo masculino dos \omterm{def:g8:reproductive-systems:male}{testículos} até
        a saída. Narre o trajeto, dizendo o que é acrescentado pelo
        caminho e como funciona o arranjo do órgão de serviço duplo.
  \item No modelo feminino, aponte onde o estoque de \omterm{def:g8:sexual-reproduction:gametes}{gametas} espera,
        onde a \omterm{def:g5:flower-to-fruit:steps}{fecundação} costuma acontecer e onde uma \omterm{def:g5:human-reproduction:pregnancy}{gravidez} se
        hospeda.
  \item Um visitante pergunta por que as fábricas de \omterm{def:g8:sexual-reproduction:gametes}{gameta} dos
        modelos diferem tanto --- milhões por dia contra uma por mês.
        Responda com a frase final do
        \cref{ex:g8:reproductive-systems:egg}.
  \item Outra pergunta o que os sistemas estavam fazendo antes da
        \omterm{def:g8:puberty:puberty}{puberdade}. Responda com a maquinaria do \cref{ch:g8:puberty}
        numa linha.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- O calendário.}
\begin{enumerate}[resume]
  \item Percorra o calendário: situe a \omterm{prop:g8:reproductive-systems:cycle}{menstruação}, a reconstrução, a
        \omterm{prop:g8:reproductive-systems:cycle}{ovulação} e o dia fértil, com as datas aproximadas deles.
  \item Um visitante presume que o dia 28 é o mesmo para todo mundo.
        Corrija a suposição com a
        \cref{rem:g8:reproductive-systems:living} --- duas fontes de
        variação.
  \item Mostre no calendário o que muda se a \omterm{def:g5:flower-to-fruit:steps}{fecundação} acontecer
        perto do dia 14 --- que acontecimentos são cancelados e quais
        começam (a \cref{def:g5:human-reproduction:pregnancy} continua
        a história).
  \item Uma \omterm{def:g3:stages-of-life:stages}{adolescente} pergunta, baixinho, se as cólicas querem
        dizer que algo está errado. Dê a resposta do \omterm{def:g3:bones-and-muscles:muscle}{músculo} e as duas
        ajudas --- e o endereço para qualquer coisa além disso.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O ofício de quem
explica.}
\begin{enumerate}[resume]
  \item Alguns visitantes riem sem jeito dos modelos. A sua frase de
        abertura trata os sistemas como o primeiro parágrafo do
        \cref{ch:g8:reproductive-systems} trata. Escreva essa frase.
  \item Resuma os dois sistemas na comparação de cinco linhas do
        \cref{met:g8:reproductive-systems:sides}, de memória.
  \item Um visitante pergunta onde o começo dos bebês se encaixa nos
        recursos didáticos. Encadeie os três recursos nos três passos
        da \cref{prop:g8:sexual-reproduction:core}.
  \item Encerre o dia aberto com uma frase sobre por que as unidades
        de saúde ensinam essa anatomia --- conhecimento contra lenda,
        como o \cref{ex:g8:puberty:questions} defendeu.
\end{enumerate}
\end{problem}
