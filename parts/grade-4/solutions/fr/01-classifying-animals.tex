\section*{Chapitre \ref{ch:g4:classifying-animals} --- Classer les animaux}

\begin{solution}{exo:g4:classifying-animals:1}
Un \omterm{def:g4:classifying-animals:vertebrate}{vertébré} possède un \omterm{def:g3:bones-and-muscles:skeleton}{squelette} intérieur bâti autour d'une \omterm{def:g4:classifying-animals:vertebrate}{colonne
vertébrale} (la truite, le chat)~; un \omterm{def:g4:classifying-animals:vertebrate}{invertébré} n'a pas de \omterm{def:g4:classifying-animals:vertebrate}{colonne
vertébrale} (l'escargot, l'abeille, le ver de terre, la méduse).
\end{solution}

\begin{solution}{exo:g4:classifying-animals:2}
\omterm{prop:g3:sorting-living-things:groups}{Poissons}~: \omterm{def:g1:animals-around-us:covering}{écailles}, nageoires, respiration dans l'eau. \omterm{prop:g3:sorting-living-things:groups}{Amphibiens}~:
\omterm{def:g1:the-five-senses:sense}{peau} nue et humide, \omterm{def:g2:animals-grow-up:born}{œufs} dans l'eau, enfance de têtard. \omterm{prop:g4:classifying-animals:classes}{Reptiles}~:
\omterm{def:g1:the-five-senses:sense}{peau} sèche et écailleuse, la plupart pondent des \omterm{def:g2:animals-grow-up:born}{œufs} à \omterm{def:g1:animals-around-us:covering}{coquille} sur
la terre ferme. \omterm{prop:g3:sorting-living-things:groups}{Oiseaux}~: \omterm{def:g1:animals-around-us:covering}{plumes}, bec, \omterm{def:g2:animals-grow-up:born}{œufs} à \omterm{def:g1:animals-around-us:covering}{coquille}.
\omterm{prop:g3:sorting-living-things:groups}{Mammifères}~: poils, petits nourris au \omterm{def:g2:animals-grow-up:born}{lait}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:3}
Sardine~: poisson. Crocodile~: reptile. Triton~: amphibien.
Autruche~: oiseau. Baleine~: mammifère. Ver de terre~: \omterm{def:g4:classifying-animals:vertebrate}{invertébré}
--- pas de \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale}, donc aucune classe de \omterm{def:g4:classifying-animals:vertebrate}{vertébrés}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:4}
Le chat, dans les \omterm{prop:g3:sorting-living-things:groups}{mammifères}, dans les \omterm{def:g4:classifying-animals:vertebrate}{vertébrés}, dans les \omterm{def:g1:animals-around-us:animal}{animaux}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:5}
La \omterm{def:g1:the-five-senses:sense}{peau} --- sèche et écailleuse chez le lézard, nue et humide chez
le triton --- et les \omterm{def:g2:animals-grow-up:born}{œufs}~: à \omterm{def:g1:animals-around-us:covering}{coquille} et pondus sur la terre ferme
chez le lézard, nus et pondus dans l'eau chez le triton.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:6}
Les \omterm{def:g4:classifying-animals:vertebrate}{invertébrés}. Par exemple~: \omterm{prop:g3:sorting-living-things:groups}{insectes}, araignées, escargots, vers
(et méduses).
\end{solution}

\begin{solution}{exo:g4:classifying-animals:7}
Manchot~: des \omterm{def:g1:animals-around-us:covering}{plumes}~? Oui --- un oiseau~; on s'arrête à la première
question. Dauphin~: des \omterm{def:g1:animals-around-us:covering}{plumes}~? Non. Des poils ou du \omterm{def:g2:animals-grow-up:born}{lait}~? Il
nourrit son petit au \omterm{def:g2:animals-grow-up:born}{lait} --- un mammifère~; on s'arrête là.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:8}
Par exemple~: des \omterm{def:g1:animals-around-us:covering}{plumes}, un bec, des ailes, des \omterm{def:g2:animals-grow-up:born}{œufs} à \omterm{def:g1:animals-around-us:covering}{coquille} ---
et une \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale}, puisque les \omterm{prop:g3:sorting-living-things:groups}{oiseaux} sont des \omterm{def:g4:classifying-animals:vertebrate}{vertébrés}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:9}
Les questions s'arrêtent à «~du \omterm{def:g2:animals-grow-up:born}{lait} pour les petits~»~: la baleine
allaite son baleineau, donc c'est un mammifère, presque sans poils
ou non. «~Vit dans la mer~» est un milieu, pas un caractère du corps
--- et la \omterm{def:g4:classifying-animals:classification}{classification} a été bâtie sur ce qu'un \omterm{def:g1:animals-around-us:animal}{animal}
\emph{possède} précisément parce que des habitudes et des maisons
peuvent être partagées par des étrangers~: le requin et le dauphin
partagent une adresse, pas une classe.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:10}
La \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale}. Le serpent a une colonne de centaines de
petits \omterm{def:g3:bones-and-muscles:skeleton}{os} --- un \omterm{def:g4:classifying-animals:vertebrate}{vertébré}, avec la \omterm{def:g1:the-five-senses:sense}{peau} sèche et écailleuse d'un
reptile~; le ver de terre n'a aucune \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale} --- un
\omterm{def:g4:classifying-animals:vertebrate}{invertébré}. Des pattes manquantes ne sont qu'un caractère absent~;
la question de la \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale} se pose la première.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:11}
Les \omterm{prop:g3:sorting-living-things:groups}{mammifères} l'ont emporté~: les poils et le \omterm{def:g2:animals-grow-up:born}{lait} pour les petits
sont le paquet des \omterm{prop:g3:sorting-living-things:groups}{mammifères}, et l'ornithorynque a les deux ---
l'\omterm{def:g2:animals-grow-up:born}{œuf} est un caractère à contretemps. La \omterm{def:g4:classifying-animals:classification}{classification} pèse le
paquet entier au lieu de laisser un seul caractère décider~; un
\omterm{def:g1:animals-around-us:animal}{animal} bizarre sur un point se range quand même là où pointent la
plupart de ses caractères. (L'ornithorynque est assez singulier pour
que les \omterm{prop:g3:sorting-living-things:groups}{mammifères} qui pondent des \omterm{def:g2:animals-grow-up:born}{œufs} forment leur propre petit
sous-groupe.)
\end{solution}
