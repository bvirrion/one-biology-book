\section*{Chapter \ref{ch:g7:tissues} --- Tissues and Organs}

\begin{solution}{exo:g7:tissues:1}
\omterm{def:g7:tissues:tissue}{Tissue}: group of similar \omterm{def:g7:microscope:cell}{cells} with a shared function (e.g.\ \omterm{def:g7:tissues:animal}{muscle
tissue}). \omterm{def:g7:tissues:organ}{Organ}: several \omterm{def:g7:tissues:tissue}{tissues} cooperating (e.g.\ \omterm{def:g5:circulation:heart}{heart}). \omterm{def:g7:tissues:system}{Organ system}:
\omterm{def:g7:tissues:organ}{organs} cooperating for a major function (e.g.\ \omterm{def:g4:body-systems:circ}{circulatory system}).
\end{solution}

\begin{solution}{exo:g7:tissues:2}
Cardiac \omterm{def:g3:bones:muscle}{muscle} \omterm{def:g7:microscope:cell}{cell} $\to$ \omterm{def:g7:tissues:animal}{muscle tissue} $\to$ \omterm{def:g5:circulation:heart}{heart} $\to$ \omterm{def:g4:body-systems:circ}{circulatory
system} $\to$ human.
\end{solution}

\begin{solution}{exo:g7:tissues:3}
Epithelial: cover/line. Connective: support/bind (\omterm{def:g3:bones:bone}{bone}, \omterm{def:g5:circulation:blood}{blood},\ldots).
\omterm{def:g3:bones:muscle}{Muscle}: contract/move. Nervous: carry signals.
\end{solution}

\begin{solution}{exo:g7:tissues:4}
A \omterm{def:g7:tissues:plant-organs}{leaf} contains several \omterm{def:g7:tissues:tissue}{tissue} types (epidermis, mesophyll, vascular
\omterm{def:g7:tissues:tissue}{tissue} in \omterm{def:g5:circulation:heart}{veins}) working together --- the definition of an \omterm{def:g7:tissues:organ}{organ}.
\end{solution}

\begin{solution}{exo:g7:tissues:5}
Example: \omterm{def:g7:tissues:plant-organs}{root} hair \omterm{def:g7:microscope:cell}{cell} --- large surface area to absorb \omterm{def:g4:plants-need:needs}{water} and
\omterm{def:g6:nutrition:groups}{minerals}; or \omterm{def:g7:tissues:plant-tissue}{xylem} vessel --- transport \omterm{def:g4:plants-need:needs}{water}; or palisade \omterm{def:g7:microscope:cell}{cell} ---
\omterm{def:g5:photosynthesis:def}{photosynthesis}.
\end{solution}
