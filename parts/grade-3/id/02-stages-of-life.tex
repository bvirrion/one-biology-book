\chapter{Tahap-Tahap Kehidupan}\label{ch:g3:stages-of-life}

Berjajarkan sebuah foto keluarga: bayi di pangkuan Nenek, anak sekolah
yang ompong dua gigi depannya, sepupu \omterm{def:g3:stages-of-life:stages}{remaja} yang tiba-tiba lebih
tinggi daripada ibunya, kedua orang tua, dan Nenek sendiri. Satu
keluarga, satu sore --- dan seluruh \omterm{def:g2:born-grow-die:cycle}{daur hidup} manusia duduk di satu
sofa. Tahun ini kita mengamati tahap-tahap itu dengan teliti, dan
mengamati bagaimana pertumbuhan itu sendiri dapat diukur.

\section{Tahap-tahap kehidupan manusia}

\begin{definition}[Tahap-tahap kehidupan manusia]\label{def:g3:stages-of-life:stages}
Kehidupan manusia melewati beberapa tahap:
\begin{enumerate}
  \item masa \emph{bayi}\index{bayi} (kira-kira dua tahun pertama):
        pertumbuhan tercepat sepanjang hidup, gigi pertama, langkah
        pertama, kata pertama;
  \item masa \emph{anak}\index{anak}: pertumbuhan yang tetap, \omterm{prop:g2:teeth-and-eating-well:sets}{gigi
        susu} ditukar dengan \omterm{prop:g2:teeth-and-eating-well:sets}{gigi tetap}, pembelajaran segala macam;
  \item masa \emph{remaja}\index{remaja}: ledakan baru pertumbuhan
        cepat, dan tubuh berubah dari tubuh anak menjadi tubuh orang
        dewasa;
  \item masa \emph{dewasa}\index{dewasa}: pertumbuhan tinggi badan
        berakhir; orang dewasa dapat menjadi orang tua;
  \item \emph{usia tua}\index{usia tua}: rambut memutih, \omterm{def:g1:the-five-senses:sense}{kulit}
        berkerut, kekuatan perlahan menurun --- tahap terakhir daurnya
        yang wajar.
\end{enumerate}
\end{definition}

\begin{example}[Membaca sofanya]\label{ex:g3:stages-of-life:sofa}
Di sofa keluarga: bayinya (tahap 1), anak sekolah yang ompong (tahap 2
--- pergantian \cref{ch:g2:teeth-and-eating-well} yang sedang berjalan
penuh), sepupu berumur enam belas tahun (tahap 3, di tengah
ledakannya), kedua orang tuanya (tahap 4), Nenek (tahap 5). Lima
tahap, satu keluarga.
\end{example}

\begin{remark}[Tumbuh tidaklah rata]\label{rem:g3:stages-of-life:bursts}
Pertumbuhan punya iramanya: sangat cepat bagi bayi --- tinggi badan
kira-kira bertambah setengahnya lagi pada tahun pertama --- lebih
lambat dan tetap sepanjang masa anak, cepat lagi bagi \omterm{def:g3:stages-of-life:stages}{remaja}, lalu
berakhir. Dua ledakan dengan rentang tenang di antaranya: itulah
bentuk wajar pertumbuhan seorang anak.
\end{remark}

\section{Mengukur pertumbuhan}

\begin{definition}[Grafik pertumbuhan]\label{def:g3:stages-of-life:chart}
Suatu \emph{grafik pertumbuhan}\index{grafik pertumbuhan} adalah
catatan tinggi badan (atau berat badan) yang diukur berulang-ulang
sepanjang waktu --- tanda kusen pintu pada \cref{ch:g1:my-body}, yang
dibuat teratur dan bertanggal. Dokter menyimpan satu grafik untuk
setiap anak.
\end{definition}

\begin{example}[Membaca grafiknya]\label{ex:g3:stages-of-life:reading}
Pada grafiknya, garisnya naik kira-kira $\qty{25}{cm}$ pada tahun yang
paling awal --- dan hanya kira-kira $\qty{6}{cm}$ setahun sepanjang
masa anak. Kedua bagian yang curam adalah dua ledakan pada
\cref{rem:g3:stages-of-life:bursts}; ujungnya yang mendatar berkata
bahwa orang \omterm{def:g3:stages-of-life:stages}{dewasanya} sudah berhenti bertambah tinggi.
\end{example}

\begin{method}[Menyimpan grafik pertumbuhanmu sendiri]\label{met:g3:stages-of-life:keep}
\begin{enumerate}
  \item Ukur tinggi badan dengan cermat (cara dinding-dan-buku pada
        tahun-tahun sebelumnya), selalu tanpa sepatu;
  \item tuliskan tanggal dan tinggi badannya di sebuah buku catatan,
        atau tandai kusen pintunya lalu beri tanggal pada tandanya;
  \item ukur lagi setiap beberapa bulan --- keteraturan lebih penting
        daripada kekerapan;
  \item untuk melihat kisahnya, gambarkan titik-titiknya: umur pada
        sumbu mendatar, tinggi badan pada sumbu tegak, seperti pada
        gambarnya.
\end{enumerate}
\end{method}

\section{Hewan dan tumbuhan pun punya tahap}

\begin{proposition}[Tahap di seluruh dunia yang hidup]\label{prop:g3:stages-of-life:animals}
Setiap \omterm{def:g1:living-or-not:living}{makhluk hidup} melewati tahap-tahap daurnya sendiri --- muda,
\omterm{def:g3:stages-of-life:stages}{dewasa}, \omterm{def:g3:stages-of-life:stages}{usia tua}. Tahap mudanya boleh jadi mirip yang \omterm{def:g3:stages-of-life:stages}{dewasa} (anak
kucing, anak kuda) atau sama sekali tidak mirip (\omterm{ex:g2:animals-grow-up:frog}{berudu}, \omterm{ex:g2:animals-grow-up:butterfly}{ulat} pada
\cref{ch:g2:animals-grow-up}); tahap sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan} berjalan dari
\omterm{def:g2:from-seed-to-plant:seed}{biji}, \omterm{prop:g2:from-seed-to-plant:cycle}{kecambah}, \omterm{def:g1:plants-around-us:plant}{tumbuhan} muda, sampai \omterm{def:g1:plants-around-us:plant}{tumbuhan} \omterm{def:g3:stages-of-life:stages}{dewasa} yang berbunga.
Yang tidak dapat dilakukan satu pun di antaranya adalah melompati satu
tahap atau menjalankannya terbalik.
\end{proposition}
\admitted

\begin{example}[Secepat apa, selambat apa]\label{ex:g3:stages-of-life:speeds}
Kecepatannya berbeda-beda luar biasa. Seekor lalat sudah \omterm{def:g3:stages-of-life:stages}{dewasa} dalam
hitungan minggu; seekor kucing kira-kira dalam setahun; seorang
manusia dalam hampir dua puluh tahun; sebatang pohon ek membutuhkan
beberapa puluh tahun sebelum \omterm{def:g2:from-seed-to-plant:seed}{biji} eknya yang pertama. Lambat atau
cepat, urutan tahapnya tidak pernah berubah.
\end{example}

\begin{remark}[Usia tua adalah sebuah tahap, bukan cacat]\label{rem:g3:stages-of-life:oldage}
Rambut putih dan kerut bukanlah penyakit: keduanya adalah tahap
terakhir tubuh, sewajar gigi pertama seorang bayi. Pada banyak spesies,
dan di dalam keluarga manusia, yang tua membawa apa yang paling
dibutuhkan yang muda --- pengalaman.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:1}
Sebutkan kelima tahap kehidupan manusia, berurutan.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:2}
Pada tahap mana: (a) \omterm{prop:g2:teeth-and-eating-well:sets}{gigi susu} tanggal? (b) seseorang dapat menjadi
orang tua? (c) pertumbuhan paling cepat sepanjang hidup?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:3}
Apa kedua ledakan pertumbuhan itu, dan apa yang terletak di antaranya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:4}
Pada \omterm{def:g3:stages-of-life:chart}{grafik pertumbuhannya}, bagaimana kamu dapat mengetahui hanya dari
garisnya bahwa seorang \omterm{def:g3:stages-of-life:stages}{dewasa} sudah berhenti tumbuh?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:5}
Dengan memakai grafiknya, kira-kira berapa sentimeter garisnya naik
antara umur $6$ dan $8$? Dan pada tahun pertama saja?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:6}
Mengapa tinggi badan harus selalu diukur tanpa sepatu, dengan cara
yang sama setiap kali? Apa yang bisa keliru kalau tidak begitu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:7}
Sebutkan tahap-tahap kehidupan sebatang pohon ek, dari \omterm{def:g2:from-seed-to-plant:seed}{biji} ek sampai
pohon tua.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:8}
Cobalah: mulailah \omterm{def:g3:stages-of-life:chart}{grafik pertumbuhan} pada
\cref{met:g3:stages-of-life:keep} untuk dirimu sendiri, dengan titik
pertama hari ini. Kamu berada pada tahap mana menurut
\cref{def:g3:stages-of-life:stages} --- dan ledakan mana yang masih
ada di depanmu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:9}
Seekor lalat, seekor kucing, seorang manusia dan sebatang pohon ek
menjalani tahap yang sama dengan kecepatan yang sangat berbeda.
Urutkan dari yang tercepat sampai yang terlambat mencapai \omterm{def:g3:stages-of-life:stages}{dewasa} ---
lalu sebutkan apa persamaan keempat urutan tahapnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:10}
Tomi, berumur $9$ tahun, mengukur dirinya dua kali dalam satu bulan
dan mendapati ``sama sekali tidak tumbuh''. Perlukah ia cemas?
Jawablah dengan \cref{rem:g3:stages-of-life:bursts} dan bentuk
grafiknya pada umurnya.
\end{exercise}
