\section*{Bab \ref{ch:g10:exercise-and-energy} --- Olahraga dan Kebutuhan Energi Tubuh}

\begin{solution}{exo:g10:exercise-and-energy:1}
Energi yang dikeluarkan dalam keadaan istirahat penuh untuk menjaga
tubuh tetap hidup --- jantung, otak, suhu, perawatan \omterm{def:g10:cells-common-unit:cell}{sel}: sekitar
\qty{80}{W}, sekitar \qty{7000}{kJ} sehari.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:2}
$2.0 \times 20 = \qty{40}{kJ}$ tiap menit, yaitu $40000/60 \approx
\qty{670}{W}$.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:3}
Simpanan fosfat ototnya (sekitar sepuluh sekon); \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} laktat
(satu sampai dua menit usaha habis-habisan); \omterm{prop:g10:cell-metabolism:respiration}{respirasi seluler} (berjam-jam).
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:4}
Laju tertinggi yang dengannya tubuh dapat menyerap dan memakai oksigen.
$3500/70 = \qty{50}{mL/min}$ tiap kilogram.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:5}
Otot, sekitar \qty{400}{g}, yang menjadi bahan bakar otot yang
menyimpannya; hati, sekitar \qty{100}{g}, yang dilepaskan sebagai
glukosa ke dalam darah bagi otak dan jaringan lainnya.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:6}
\qty{2.7}{L/min}: $2.7 \times 20 = \qty{54}{kJ}$ tiap menit, yaitu
\qty{900}{W}. Kalornya: $900 - 200 = \qty{700}{W}$.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:7}
Usahanya $mgh = 60 \times 9.8 \times 800 \approx \qty{470}{kJ}$. Pada
efisiensi 25\% ototnya mengeluarkan sekitar \qty{1900}{kJ}. Sepanjang
\qty{7200}{s}: sekitar \qty{260}{W} di atas keadaan istirahat.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:8}
Lari cepat 100 meter berjalan hampir seluruhnya dengan simpanan fosfat:
tanpa utang oksigen, sedikit asam laktat. Lomba 400 meter berlangsung
sekitar satu menit dan menarik separuh energinya dari \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}
laktat: asam laktatnya menumpuk (itulah nyerinya) dan respirasi yang
harus membersihkannya membuat pelarinya terus terengah selama beberapa
menit sesudahnya.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:9}
Energi \omterm{def:g10:chemistry-of-life:families}{karbohidratnya} $0.6 \times 3000 = \qty{1800}{kJ}$; \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya}
$1800/17 \approx \qty{106}{g}$; air yang dilepaskan sekitar \qty{320}{g}.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:10}
Daya \omterm{def:g10:cell-metabolism:metabolism}{metabolismenya} sekitar \qty{600}{W}: $\num{1.1e6}/600 \approx
\qty{1830}{s}$, sekitar 30 menit.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:11}
Tiga perempat energi yang dilepaskan dalam ototnya berupa kalor,
beberapa ratus watt selama olahraga berat. Berkeringat
menyingkirkannya: menguapkan air menuntut sekitar \qty{2.4}{kJ} tiap
gram, dan aliran darah ke kulit membawa kalornya ke sana. Tanpa
berkeringat, suhu tubuhnya akan naik satu derajat tiap beberapa menit.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:12}
Ketika menanjak, daya yang diperlukan sebanding dengan massa tubuhnya:
subjek \qty{60}{kg} punya \qty{60}{mL/min} tiap kilogram berhadapan
dengan \num{40}, dan menanjak lebih cepat. Pada sepeda laboratorium
dayanya tidak bergantung pada massa, keduanya punya \qty{3.6}{L/min}
yang sama, dan keduanya dapat mempertahankan watt yang sama.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:13}
Kekurangan \qty{0.4}{L/min} setara oksigen itu dipasok oleh \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}
laktat dalam ototnya. Asam laktatnya menumpuk; dalam beberapa menit
ototnya terbakar dan pelarinya terpaksa melambat ke kecepatan yang
dapat ditutupi respirasi sendirian.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:14}
Letupan maksimal yang singkat berjalan dengan simpanan fosfat dan
\omterm{def:g10:cell-metabolism:fermentation}{fermentasi}, yang sama sekali tidak memakai lemak; lemak hanya dibakar
lewat respirasi, yang menuntut beberapa menit untuk mencapai laju penuh
dan hanya berkuasa dalam olahraga sedang yang berkepanjangan. Energi
seluruh beberapa letupan itu juga kecil dibandingkan usaha mantap
selama satu jam.
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:15}
Kekurangannya $25000 - 8500 - 6000 = \qty{10500}{kJ}$, yaitu sekitar
\qty{280}{g} lemak. Makan selama etapenya menjaga glukosa darahnya
tetap tinggi bagi otaknya dan menghemat \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya} untuk tanjakan,
tempat daya yang diperlukan jauh di atas apa yang dapat dipasok lemak
sendirian; lemaknya menutupi bagian hari yang mantap.
\end{solution}

\begin{solution}{pb:g10:exercise-and-energy:1}
\textbf{1.} $4.0 \times 60 \times 42.2 \approx \qty{10100}{kJ}$.

\textbf{2.} $\qty{3}{h}\,\qty{30}{min} = \qty{12600}{s}$:
$\num{1.01e7}/12600 \approx \qty{800}{W}$.

\textbf{3.} $10100/20 \approx \qty{506}{L}$; sepanjang \qty{210}{min},
sekitar \qty{2.4}{L/min}.

\textbf{4.} $2400/60 = \qty{40}{mL/min}$ tiap kilogram, yaitu $40/52
\approx 77\%$ nilai maksimalnya.

\textbf{5.} $(2.4 - 0.2)/2.4 \approx 92\%$.

\textbf{6.} $380 \times 17 = \qty{6460}{kJ}$.

\textbf{7.} Lombanya menuntut \qty{240}{kJ} tiap kilometer: $6460/240
\approx \qty{27}{km}$.

\textbf{8.} Pemakaian \omterm{def:g10:chemistry-of-life:families}{karbohidratnya} $0.8 \times 240 = \qty{192}{kJ/km}$:
$6460/192 \approx \qty{34}{km}$ --- itulah temboknya.

\textbf{9.} Energi lemaknya $0.2 \times 10100 \approx \qty{2020}{kJ}$,
yaitu $2020/38 \approx \qty{53}{g}$: sekitar 0.6\% dari \qty{9}{kg}-nya.

\textbf{10.} Lemak hanya dibakar lewat respirasi dan dengan laju yang
terbatas: ia hanya dapat memasok kira-kira separuh daya yang dituntut
kecepatannya. Tanpa \omterm{def:g10:exercise-and-energy:glycogen}{glikogen}, ototnya tidak dapat membuat ATP cukup
cepat, dan kecepatannya turun.

\textbf{11.} $60 \times 3.5 = \qty{210}{g}$, yaitu $210 \times 17 \approx
\qty{3570}{kJ}$, yang menutupi $3570/192 \approx \qty{19}{km}$
pemakaian \omterm{def:g10:chemistry-of-life:families}{karbohidrat}.

\textbf{12.} \omterm{def:g10:chemistry-of-life:families}{Karbohidrat} yang tersedia $6460 + 3570 = \qty{10030}{kJ}$:
$10030/192 \approx \qty{52}{km}$, melampaui garis akhirnya. Ya.

\textbf{13.} $+\qty{200}{g}$ \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} $+ \qty{600}{g}$ air:
\qty{0.8}{kg}, yang menuntut tambahan $0.8 \times 4.0 \approx \qty{3}{kJ}$
tiap kilometer --- dapat diabaikan terhadap 240.

\textbf{14.} \omterm{def:g10:exercise-and-energy:glycogen}{Glikogennya} $580 \times 17 = \qty{9860}{kJ}$ ditambah
minumannya: sekitar \qty{13400}{kJ}; pemakaian \omterm{def:g10:chemistry-of-life:families}{karbohidratnya} $0.55 \times 240 =
\qty{132}{kJ/km}$: lebih dari \qty{100}{km}. Temboknya
telah berpindah jauh melampaui garis akhir, dengan selisih yang lebar.

\textbf{15.} Otak hanya membakar glukosa, yang dipasok \omterm{def:g10:exercise-and-energy:glycogen}{glikogen}
hatinya. Ketika simpanan itu kosong, glukosa darahnya turun, dan otak
mengisyaratkan keadaan darurat itu sebagai pusing dan lelah yang luar
biasa, seberapa pun yang masih dimiliki ototnya.

\textbf{16.} $3.6 \times 55 \times 42.2 \approx \qty{8360}{kJ}$;
$8360/20 = \qty{418}{L}$ sepanjang \qty{125}{min}: sekitar \qty{3.3}{L/min}.

\textbf{17.} $3340/55 \approx \qty{61}{mL/min}$ tiap kilogram, yaitu
$61/80 \approx 76\%$ --- bagian yang sama dengan Nadia. Yang berbeda
adalah batas atasnya, bukan bagian yang terpakai darinya.

\textbf{18.} Kehematan adalah biaya tiap kilogram tiap kilometer. Bagi
pelari \qty{55}{kg}, $(4.0 - 3.6) \times 55 \times 42.2 \approx
\qty{930}{kJ}$ terhemat sepanjang lombanya, sekitar 9\%.

\textbf{19.} \omterm{def:g10:exercise-and-energy:glycogen}{Glikogennya} \qty{8500}{kJ}; \omterm{def:g10:chemistry-of-life:families}{karbohidrat} yang diperlukan $0.7 \times
8360 \approx \qty{5850}{kJ}$: ya, simpanannya mencukupi tanpa minum.

\textbf{20.} Tanpa persiapan, \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} Nadia habis sekitar kilometer
34; pemuatan \omterm{def:g10:chemistry-of-life:families}{karbohidrat} sebelum lomba, minum \omterm{def:g10:chemistry-of-life:families}{karbohidrat} selama lomba,
dan latihan yang menaikkan bagian lemak yang terbakar masing-masing
mendorong temboknya melampaui garis akhir.
\end{solution}
