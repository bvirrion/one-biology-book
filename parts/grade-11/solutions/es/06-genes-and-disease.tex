\section*{Capítulo \ref{ch:g11:genes-and-disease} --- Variación genética y enfermedad}

\begin{solution}{exo:g11:genes-and-disease:1}
\omterm{def:g11:genes-and-disease:genetic}{Recesivo}: la enfermedad solo aparece con dos copias del \omterm{def:g10:universal-dna:gene}{alelo};
\omterm{def:g11:genes-and-disease:genetic}{dominante}: basta una copia. Un \omterm{def:g11:genes-and-disease:genetic}{portador} lleva un \omterm{def:g10:universal-dna:gene}{alelo} mutante \omterm{def:g11:genes-and-disease:genetic}{recesivo}
y está sano.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:2}
Afectado $1/4$, \omterm{def:g11:genes-and-disease:genetic}{portador} $1/2$, ninguna de las dos cosas $1/4$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:3}
I-1 $Aa$, I-2 $Aa$ (padres de un hijo afectado), II-2 $aa$, III-2 $aa$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:4}
Un varón tiene un solo \omterm{def:g11:cell-cycle-mitosis:chromosome}{cromosoma} X: un \omterm{def:g10:universal-dna:gene}{alelo} \omterm{def:g11:genes-and-disease:genetic}{recesivo} situado en él se
expresa, porque no hay segunda copia que lo enmascare. Una mujer
necesita el \omterm{def:g10:universal-dna:gene}{alelo} en sus dos \omterm{def:g10:universal-dna:information}{cromosomas} X.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:5}
Una enfermedad cuyo riesgo depende de muchos \omterm{def:g10:universal-dna:gene}{alelos} de efecto pequeño y
del ambiente: la diabetes de tipo 2, las enfermedades del corazón
(también el asma y la mayoría de los cánceres).
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:6}
$aa \times AA$: todos los hijos $Aa$, \omterm{def:g11:genes-and-disease:genetic}{portadores} sanos. Un hijo
\omterm{def:g11:genes-and-disease:genetic}{portador} con un \omterm{def:g11:genes-and-disease:genetic}{portador}: $1/4$ afectados, $1/2$ \omterm{def:g11:genes-and-disease:genetic}{portadores}, $1/4$ no
\omterm{def:g11:genes-and-disease:genetic}{portadores}.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:7}
II-1 es una hija sana de dos \omterm{def:g11:genes-and-disease:genetic}{portadores}: \omterm{def:g11:genes-and-disease:genetic}{portadora} con probabilidad
$2/3$. Riesgo con un varón de la población: $2/3 \times 1/25 \times 1/4
= 1/150$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:8}
$1/2$ para cada hijo. Ninguno de los cuatro: $(1/2)^4 = 1/16$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:9}
$X^hY \times X^HX^H$: todas las hijas $X^HX^h$, \omterm{def:g11:genes-and-disease:genetic}{portadoras} sanas; todos
los hijos $X^HY$, sanos. Una hija \omterm{def:g11:genes-and-disease:genetic}{portadora} con un hombre sano: la
mitad de sus hijos varones hemofílicos y la mitad de sus hijas
\omterm{def:g11:genes-and-disease:genetic}{portadoras}.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:10}
Un \omterm{def:g10:universal-dna:gene}{alelo} da 400 y 200; el otro, con la deleción adicional, da 300 y
200. Bandas de 400, 300 y 200 --- y ninguna de 600.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:11}
La frecuencia de un \omterm{def:g10:universal-dna:gene}{alelo} en una población depende de su historia y no
solo de la tasa de \omterm{def:g11:mutations:mutation}{mutación}: el \omterm{def:g10:universal-dna:gene}{alelo} europeo surgió hace mucho y se
extendió, quizá porque los \omterm{def:g11:genes-and-disease:genetic}{portadores} tenían alguna ventaja (se
sospecha de una resistencia a una enfermedad diarreica); en el este de
Asia no ocurrió.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:12}
Cada hijo es una tirada independiente: $1/4$ para el segundo. Al menos
uno de los dos siguientes: $1 - (3/4)^2 = 7/16$. Los \omterm{def:g10:universal-dna:gene}{alelos} se reparten
de nuevo en cada concepción; un hijo afectado anterior no baja el
riesgo del siguiente.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:13}
Antes de la prueba, \omterm{def:g11:genes-and-disease:genetic}{portadora} con probabilidad $1/25$; la prueba se le
escapa al 10\% de los \omterm{def:g11:genes-and-disease:genetic}{portadores}, así que de 1000 mujeres, 40 son
\omterm{def:g11:genes-and-disease:genetic}{portadoras} y 4 dan negativo, mientras que 960 no \omterm{def:g11:genes-and-disease:genetic}{portadoras} dan
negativo: probabilidad de ser \omterm{def:g11:genes-and-disease:genetic}{portadora} $4/964 \approx 1/240$. Riesgo
con un \omterm{def:g11:genes-and-disease:genetic}{portador} conocido: $1/240 \times 1/4
\approx 1/960$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:14}
De muchos a pocos \omterm{def:g10:universal-dna:gene}{alelos}, siendo sedentario: del 34\% al 9\%; de
sedentario a activo, con muchos \omterm{def:g10:universal-dna:gene}{alelos}: del 34\% al 12\%. El cambio que
sí se puede hacer da casi la misma reducción que el que no. Una
puntuación de riesgo sirve para identificar a quién le compensa más
cambiar de modo de vida, no como un veredicto.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:15}
La prueba ofrece certeza --- librarse de una sombra del cincuenta por
ciento si es negativa, poder planificar una vida y una familia si es
positiva --- y cuesta, si es positiva, saber que hay una enfermedad
incurable décadas antes, con efectos sobre el ánimo, los seguros y los
familiares que comparten el \omterm{def:g10:universal-dna:gene}{gen}. Del resultado no se sigue ningún
tratamiento, así que la única razón para saberlo es la de la propia
persona; por eso la elección, y su momento, son suyas.
\end{solution}

\begin{solution}{pb:g11:genes-and-disease:1}
\textbf{1.} \omterm{def:g11:genes-and-disease:genetic}{Recesiva}: Zoé está afectada y sus dos padres están sanos,
así que cada uno lleva oculto un \omterm{def:g10:universal-dna:gene}{alelo} mutante.

\textbf{2.} Lea $Aa$, Tom $Aa$, Zoé $aa$.

\textbf{3.} $AA$ con probabilidad $1/3$ y $Aa$ con $2/3$ (está sano,
así que $aa$ queda descartado).

\textbf{4.} Tom es \omterm{def:g11:genes-and-disease:genetic}{portador}, así que al menos uno de sus padres lo es;
como en esa generación no hay ninguna persona afectada, Ana recibió el
\omterm{def:g10:universal-dna:gene}{alelo} de ese progenitor con probabilidad $1/2$: $1/2$ (tomando al otro
progenitor como no \omterm{def:g11:genes-and-disease:genetic}{portador}).

\textbf{5.} $1/25$, la cifra de la población.

\textbf{6.} $1/2 \times 1/25 \times 1/4 = 1/200$.

\textbf{7.} Afectado $1/4$; \omterm{def:g11:genes-and-disease:genetic}{portador} $1/2$.

\textbf{8.} $2/3 \times 1/25 \times 1/4 = 1/150$.

\textbf{9.} Zoé es $aa$; su pareja es $Aa$ con probabilidad $1/25$.
Afectado: $1/25 \times 1/2 = 1/50$. \omterm{def:g11:genes-and-disease:genetic}{Portador}: todos los hijos reciben
$a$ de Zoé, así que serán \omterm{def:g11:genes-and-disease:genetic}{portadores} con probabilidad $24/25 \times 1 + 1/25 \times
1/2 \approx 0.98$.

\textbf{10.} Lou es \omterm{def:g11:genes-and-disease:genetic}{portadora} con probabilidad $2/3$; el primo recibió
$a$ de Ana (\omterm{def:g11:genes-and-disease:genetic}{portadora} con probabilidad $1/2$) con probabilidad $1/2$,
es decir es \omterm{def:g11:genes-and-disease:genetic}{portador} con probabilidad $1/4$ (sin contar a Karim):
$2/3 \times 1/4
\times 1/4 = 1/24$, cuatro veces más que en el caso de
Max, porque el primo procede de la misma familia y es mucho más
probable que un desconocido que lleve el mismo \omterm{def:g10:universal-dna:gene}{alelo}.

\textbf{11.} Zoé: solo 600 y 300. Tom: 900, 600 y 300. No \omterm{def:g11:genes-and-disease:genetic}{portadora}:
solo 900.

\textbf{12.} $AA$: no es \omterm{def:g11:genes-and-disease:genetic}{portador}; el riesgo para su hijo baja de
$1/150$ a cero (salvo una \omterm{def:g11:mutations:mutation}{mutación} nueva).

\textbf{13.} Ana es $Aa$: riesgo $1 \times 1/25 \times 1/4 = 1/100$.

\textbf{14.} De 1000 personas como Karim, 40 son \omterm{def:g11:genes-and-disease:genetic}{portadoras} y 4 de
ellas dan negativo, y 960 no \omterm{def:g11:genes-and-disease:genetic}{portadoras} dan negativo: $4/964 \approx
1/240$. Riesgo: $1 \times 1/240 \times 1/4 \approx 1/960$.

\textbf{15.} $Aa$: un \omterm{def:g11:genes-and-disease:genetic}{portador} sano, como su madre.

\textbf{16.} $(1/25)^2 = 1/625$ de las parejas son dos \omterm{def:g11:genes-and-disease:genetic}{portadores}; una
cuarta parte de sus hijos están afectados: $1/2500$.

\textbf{17.} Exactamente la frecuencia observada: la frecuencia de
\omterm{def:g11:genes-and-disease:genetic}{portadores} y la regla \omterm{def:g11:genes-and-disease:genetic}{recesiva} dan cuenta de ella.

\textbf{18.} Con una frecuencia de \omterm{def:g11:genes-and-disease:genetic}{portadores} de $1/25$ y una
frecuencia de afectados de $1/2500$, los \omterm{def:g10:universal-dna:gene}{alelos} mutantes que llevan los
\omterm{def:g11:genes-and-disease:genetic}{portadores} sanos superan a los de las personas afectadas en la
proporción de $2 \times 1/25$ frente a $2 \times
1/2500$, cien a uno.
La muerte de las personas afectadas retira solo el 1\% de las copias
por generación; el resto lo transmiten sin verse los \omterm{def:g11:genes-and-disease:genetic}{portadores}.

\textbf{19.} La frecuencia del \omterm{def:g10:universal-dna:gene}{alelo} subirá, pero muy despacio: el 1\%
de las copias que antes se perdía en cada generación se conserva ahora,
así que el cambio es del orden del 1\% de la frecuencia por generación.

\textbf{20.} Dos \omterm{def:g11:genes-and-disease:genetic}{portadores} arriesgan un hijo afectado de cada cuatro;
un hermano sano de un hijo afectado es \omterm{def:g11:genes-and-disease:genetic}{portador} dos veces de cada tres;
el gel convirtió las probabilidades en \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotipos} --- descartando a Max,
confirmando a Ana, tranquilizando pero sin descartar a Karim --- e
identificó al feto como \omterm{def:g11:genes-and-disease:genetic}{portador} sano.
\end{solution}
