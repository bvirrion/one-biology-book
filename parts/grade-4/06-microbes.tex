\chapter{Microbes Around Us}\label{ch:g4:microbes}

Not all \omterm{def:g1:living-nonliving:living}{living things} are big enough to see. \omterm{def:g4:microbes:def}{Microbes} are tiny
\omterm{def:g4:ecosystems:levels}{organisms} --- some helpful, some harmful, many neither. Learning about
them helps you cook safely, wash wisely and understand yoghurt, soil
and illness.

\section{What are microbes?}

\begin{definition}[Microbe]\label{def:g4:microbes:def}
A \emph{microbe}\index{microbe} (microorganism) is a \omterm{def:g1:living-nonliving:living}{living thing} too
small to see clearly without a \omterm{prop:g4:cells:building}{microscope}. Microbes include many
\emph{bacteria}\index{bacteria}, some \emph{fungi}\index{fungi} (such as
yeasts), and other tiny life forms. \emph{Viruses}\index{virus} are even
smaller particles that can infect \omterm{def:g4:cells:cell}{cells}; they are not \omterm{def:g4:cells:cell}{cells} themselves.
\end{definition}

\omimg{microscope.jpg}{0.52\linewidth}{\omterm{prop:g4:cells:building}{Microscopes} opened the door to
the microbial world that covers \omterm{def:g1:senses:five}{skin}, soil, \omterm{def:g4:plants-need:needs}{water} and \omterm{def:g2:food-energy:food}{food}.}

\begin{example}[Everywhere]\label{ex:g4:microbes:where}
\omterm{def:g4:microbes:def}{Microbes} live in soil, rivers, \omterm{def:g2:habitats:kinds}{oceans}, on \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g2:plant-parts:main}{leaves}, on your \omterm{def:g1:senses:five}{skin}
and in your gut. A clean-looking surface can still hold many \omterm{def:g4:microbes:def}{microbes}.
\end{example}

\section{Helpful microbes}

\begin{proposition}[Not the enemy by default]\label{prop:g4:microbes:helpful}
Many \omterm{def:g4:microbes:def}{microbes} are useful or harmless. Only some cause disease under
certain conditions.
\end{proposition}
\admitted

\begin{example}[Food and soil]\label{ex:g4:microbes:food}
Yeast helps bread rise. \omterm{def:g4:microbes:def}{Bacteria} help make yoghurt, cheese and some
pickles. Soil \omterm{def:g4:microbes:def}{microbes} break down dead \omterm{def:g2:plant-parts:main}{leaves} and free \omterm{def:g4:plants-need:needs}{nutrients} for
\omterm{def:g1:plants-animals:plant}{plants}. Gut \omterm{def:g4:microbes:def}{microbes} help digest some \omterm{def:g2:food-energy:food}{foods}.
\end{example}

\section{Harmful microbes and hygiene}

\begin{definition}[Pathogen]\label{def:g4:microbes:pathogen}
A \emph{pathogen}\index{pathogen} is a \omterm{def:g4:microbes:def}{microbe} (or \omterm{def:g4:microbes:def}{virus}) that can cause
disease. Spread may occur by dirty \omterm{def:g1:my-body:parts}{hands}, uncooked \omterm{def:g2:food-energy:food}{food}, coughs and
sneezes, or unclean \omterm{def:g4:plants-need:needs}{water} --- depending on the pathogen.
\end{definition}

\begin{method}[Everyday protection]\label{met:g4:microbes:hygiene}
\begin{enumerate}
  \item Wash \omterm{def:g1:my-body:parts}{hands} with soap and \omterm{def:g4:plants-need:needs}{water} before eating and after using
        the toilet.
  \item Cover coughs and sneezes; bin used tissues.
  \item Cook \omterm{def:g2:food-energy:food}{foods} such as meat thoroughly; keep cold \omterm{def:g2:food-energy:food}{food} cold.
  \item Clean cuts and cover them when needed.
  \item Follow local advice on vaccines and medicines from \omterm{def:g2:healthy:health}{health}
        professionals.
\end{enumerate}
\end{method}

\begin{example}[Soap matters]\label{ex:g4:microbes:soap}
Rinsing with \omterm{def:g4:plants-need:needs}{water} alone removes some dirt. Soap helps lift oils and
many \omterm{def:g4:microbes:def}{microbes} off the \omterm{def:g1:senses:five}{skin} so they rinse away.
\end{example}

\section{Growing and observing safely}

\begin{example}[Yeast balloons]\label{ex:g4:microbes:yeast}
Warm \omterm{def:g4:plants-need:needs}{water}, sugar and yeast in a bottle can inflate a balloon as yeast
produces gas --- a visible sign of microbial activity when done as a
supervised class demonstration.
\end{example}
