\section*{अध्याय \ref{ch:g9:immune-defenses} --- प्रतिरक्षा की हिफ़ाज़तें}

\begin{solution}{exo:g9:immune-defenses:1}
\omterm{def:g9:immune-defenses:standing}{श्वेत रक्त कोशिकाएँ}, जो \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} की मज्जा में बनती हैं और \omterm{prop:g4:journey-of-food:absorb}{रक्त} तथा ऊतकों में
गश्त लगाती हैं। सबसे पहले पहुँचने वाले: भक्षक, जो घुसपैठियों के चारों ओर
बहकर उन्हें पचा डालते हैं।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:2}
लाली और गरमाहट: \omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ} चौड़ी, और उस जगह ज़्यादा \omterm{prop:g4:journey-of-food:absorb}{रक्त}। सूजन: रिसती हुई
दीवारें, जो तरल तथा फ़ौज को बाहर आने देती हैं। टीस: वह भीड़ भरा,
इशारे करता मैदान। (और पीप, अगर आए तो: चुके हुए भक्षकों और मरे हुए
हमलावरों का हिसाब।)
\end{solution}

\begin{solution}{exo:g9:immune-defenses:3}
\omterm{prop:g9:immune-defenses:specific}{प्रतिजन} किसी हमलावर की सतह का निशान है --- यानी वह आणविक आकार, जो पराया
पढ़ा जाता है। \omterm{prop:g9:immune-defenses:specific}{प्रतिरक्षी} वह बैठने वाला अणु है, जिसे विशेषज्ञ \omterm{prop:g4:journey-of-food:absorb}{रक्त} में
उँडेल देते हैं। रिश्ता: उस \omterm{prop:g9:immune-defenses:specific}{प्रतिरक्षी} का आकार उसी \omterm{prop:g9:immune-defenses:specific}{प्रतिजन} पर बैठता है, और
उसके ढोने वाले को भक्षकों के लिए जकड़कर गुच्छों में बाँध देता है।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:4}
क्योंकि \omterm{def:g9:microbes-and-infection:sorted}{विषाणु} शरीर की \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} के \emph{भीतर}, उन्हीं की मशीनरी पर बढ़ता
है: आज़ाद \omterm{def:g9:microbes-and-infection:sorted}{विषाणु} पर किया गया कोई भी हमला उस छपाई को नहीं रोकता। इसलिए
हड़पी हुई छापाख़ाना-कोशिकाओं को नष्ट करना ही उत्पादन रोकने का अकेला
तरीक़ा है।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:5}
लंबी उम्र वाला वह विशेषज्ञ, जो किसी हारे हुए हमलावर का हुलिया थामे रहता
है। दूसरी उतराई एक तैयार टुकड़ी से मिलती है: जवाब घंटों में, और \omterm{def:g9:heredity-and-traits:traits}{लक्षणों}
से पहले ही सफ़ाया --- यानी \omterm{def:g9:immune-defenses:memory}{प्रतिरक्षा}।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:6}
\omterm{prop:g9:immune-defenses:vaccination}{टीकाकरण} \omterm{def:g9:immune-defenses:memory}{प्रतिरक्षा} के विशेषज्ञों के सामने किसी हमलावर का बेज़रर हुलिया पेश
करता है, ताकि याददाश्त बिना बीमारी के बन जाए।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:7}
क्योंकि बहुत-से \omterm{prop:g9:microbes-and-infection:mostly}{रोगजनक} गरमी में कम बढ़ते हैं ---
\cref{prop:g6:decomposers-and-soil:speed} वाला परिस्थिति-नियम, हमलावर के
ख़िलाफ़ मोड़ा हुआ: शरीर जान-बूझकर गरम चलता है ताकि अपनी नक़ल-फ़ौज खड़ी
होने तक दुश्मन का दोगुना होना धीमा कर सके। क़ीमत, हाँ; ख़राबी, नहीं।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:8}
खसरे के \omterm{prop:g9:immune-defenses:specific}{प्रतिजन} टिकाऊ हैं: एक जीत की याददाश्त हर बाद की मुलाक़ात पर बैठ
जाती है --- खसरा एक बार, दो बार कभी नहीं। और ज़ुकाम के \omterm{def:g9:microbes-and-infection:sorted}{विषाणु} तेज़ी से
उत्परिवर्तित होते हैं: हर सर्दी की नस्लें ऐसे हुलिये पहनती हैं जो
याददाश्त के पास हैं ही नहीं, इसलिए हर ज़ुकाम एक पहली उतराई है --- यानी
ज़ुकाम हमेशा, और फिर भी हर एक याद रखा हुआ।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:9}
विविधता: अनगिनत अलग-अलग ग्राहियों वाले विशेषज्ञ तैयार खड़े हैं। रोक:
\omterm{prop:g9:immune-defenses:specific}{प्रतिजन} का संपर्क --- सिर्फ़ बैठने वाले गिनती के ही ``चुने'' जाते हैं।
असमान बढ़त: चुने हुए बढ़कर नक़ल-फ़ौज बन जाते हैं, और वही बैठने वाला ग्राही
विरासत में देते हैं। यानी चयन के तीनों तथ्य, एक ही शरीर के भीतर दिनों में
चले हुए।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:10}
किसी फैलाव को पकड़ में आने लायक़ मेज़बान चाहिए: टीका लगा हर पड़ोसी संचरण
के लिए एक बंद गली है, और काफ़ी सारी बंद गलियाँ उस फैलाव को हर किसी तक
पहुँचने से पहले ही भूखा मार देती हैं --- यही वह बुझना है। और टिके हुए
लोग: वे, जिन्हें सिखाया ही नहीं जा सकता --- बहुत छोटे, बहुत बीमार ---
और जो या तो उस साझी ढाल के भीतर खड़े हैं या कहीं नहीं।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:11}
(क) तय करके कराई गई दूसरी उतराई: वह पुनर्बल ख़ुराक टिटनेस का हुलिया
दोबारा पेश करती है और फीकी पड़ती याददाश्त फिर भर देती है --- यानी बचाव,
किसी ताज़ा प्रवेश-घाव के हिसाब से सधा हुआ। (ख) पहली उतराई, विशेषज्ञों
वाला चरण: वे \omterm{def:g8:hormonal-communication:glands}{ग्रंथियाँ} ही छावनियाँ हैं, और गले के उस हमलावर के ख़िलाफ़
खड़ी की जा रही नक़ल-फ़ौज से खचाखच भरी हैं --- यानी हिफ़ाज़त काम पर, कोई
आफ़त नहीं।
\end{solution}

\begin{solution}{exo:g9:immune-defenses:12}
वह \omterm{def:g9:microbes-and-infection:sorted}{विषाणु}: एक इबारत, जो सिर्फ़ इंसानी छापाख़ाने उधार लेती है --- न कोई
जंतु-भंडार, न कहीं और छिपने की जगह। वह याददाश्त: जेनर की सधी हुई टुकड़ी,
जो निखरकर एक टिकाऊ टीका बनी और जिसकी हिफ़ाज़त जीवन भर की थी। वह साझापन:
एक विश्व-अभियान, जिसने घेरा-दर-घेरा टीके लगाए, यहाँ तक कि किसी उतराई को
पकड़ में आने लायक़ मेज़बान मिलना ही बंद हो गया --- यानी संचरण एक ही साथ हर
जगह भूखा मारा गया। और उन्मूलन विलुप्ति ही है:
\cref{prop:g5:biodiversity:loss} वाला हमेशा-के-लिए-गई \omterm{def:g5:biodiversity:species}{प्रजाति} वाला मामला,
जान-बूझकर हासिल किया हुआ, और जिसका शोक किसी ने नहीं मनाया --- यानी वह
अकेली विलुप्ति, जिसके लिए मानवता ने मेहनत की, और इस बात का सबूत कि साझी
याददाश्त क्या कर सकती है।
\end{solution}

\begin{solution}{pb:g9:immune-defenses:1}
\textbf{1.} हमलावरों के हुलिये --- यानी कमज़ोर किए गए सूक्ष्मजीवों और
बेज़रर टुकड़ों पर लगे \omterm{prop:g9:immune-defenses:specific}{प्रतिजन}। और पूरा चुनाव-और-याददाश्त वाला कार्यक्रम:
बैठने वाले विशेषज्ञ चुने गए, नक़ल-फ़ौजें खड़ी हुईं, और \omterm{def:g9:immune-defenses:memory}{स्मृति-कोशिकाएँ} एक
सधी हुई टुकड़ी के रूप में बैठ गईं।

\textbf{2.} सफ़ाई पहली हिफ़ाज़तों के काम आती है --- यानी प्रवेश पर
बुहारा गया दरवाज़ा। और पुनर्बल ख़ुराक वह हुलिया दोबारा पेश करती है तथा
याददाश्त को पूरी ताक़त तक भर देती है। टिटनेस जुआ इसलिए नहीं खेलने देता कि
उसकी पहली लड़ाई किसी गहरे, हवा-रहित घाव से आते लक़वा मारने वाले ज़हर के
ख़िलाफ़ लड़ी जाती है --- और ऐसी लड़ाइयाँ शरीर हार जाते हैं; इसलिए सिखाना
उससे पहले ही होना चाहिए।

\textbf{3.} पहले से तीसरे दिन: तेज़ जवाब पीछे छूटा, और \omterm{def:g9:microbes-and-infection:sorted}{विषाणु} हवा-मार्गों
तथा \omterm{def:g1:the-five-senses:sense}{त्वचा} में छपाई करता हुआ। पहले हफ़्ते का बुख़ार और ददोरे: नक़ल-फ़ौज
खड़ी होती हुई --- यानी उस लड़ाई का ख़र्च, दिखता हुआ। दूसरा हफ़्ता:
\omterm{prop:g9:immune-defenses:specific}{प्रतिरक्षी} और मारने वाले उन छापाख़ानों को साफ़ करते हैं; और \omterm{def:g7:organs-at-work:recovery}{बहाली}। और
बैठ जाती है: जीवन भर की एक स्मृति-चौकी --- खसरा एक बार, दो बार कभी
नहीं।

\textbf{4.} टीका लगे हुए: सधे हुए --- उनके लिए कोई भी उतराई दूसरी उतराई
है, जो घंटों में साफ़ हो जाती है। ठीक हो चुके: वही हैसियत, बस याददाश्त
किसी असली पहली लड़ाई से। और वह शिशु: बिना सधा हुआ --- उसकी पहली उतराई
पूरा, ख़तरनाक कार्यक्रम चलाती; इसीलिए वह चिंता भरी नज़र।

\textbf{5.} इसकी कि उससे मेल खाती याददाश्त खड़ी है: \omterm{prop:g9:immune-defenses:specific}{प्रतिरक्षी} का स्तर
\omterm{prop:g4:journey-of-food:absorb}{रक्त} में उस चौकी के दस्तख़त हैं, और वह बताता है कि \omterm{def:g9:immune-defenses:memory}{स्मृति-कोशिकाएँ} भरी
हुई हैं तथा संपर्क होते ही और उँडेलने को तैयार।

\textbf{6.} स्मृति-चौकियाँ सालों के साथ पतली पड़ जाती हैं --- और स्तर
फ़ौरन सफ़ाई वाली लकीर से नीचे उतर आते हैं। और पुनर्बल ख़ुराक एक ताज़ा
पेशकश है: बची हुई याददाश्त नए सिरे से बढ़ती है और उस चौकी को फिर पूरी
ताक़त तक भर देती है।

\textbf{7.} उसका तेज़ \omterm{prop:g9:genes-and-dna:explains}{उत्परिवर्तन}: हर मौसम की नस्लें दोबारा लिखे हुए
हुलिये ढोती हैं, इसलिए पिछले साल की सिखाई इस साल के \omterm{prop:g9:immune-defenses:specific}{प्रतिजनों} पर बैठती ही
नहीं --- और उस टुकड़ी को हर साल मौजूदा हुलिये पर दोबारा सिखाना पड़ता है।

\textbf{8.} क्योंकि कर्मचारी ही उस इमारत के जोड़ हैं --- रोज़ बीमारों और
नाज़ुक लोगों के बग़ल में, यानी ठीक उन्हीं के, जिनकी अपनी सिखाई या तो है
ही नहीं या कमज़ोर है। टीका लगे कर्मचारी बिस्तरों के बीच बंद गलियाँ हैं:
उनकी साझी हिफ़ाज़त वार्ड के सबसे टिके हुए लोगों पर ढाल बन जाती है।

\textbf{9.} सिखाने की क़ीमत: एक दुखती \omterm{def:g1:my-body:parts}{बाँह} और एक दिन की खीझ, और
याददाश्त बिना किसी लड़ाई के हासिल। और खसरे के लिए पहली लड़ाई की क़ीमत:
अच्छे से अच्छे में भी हफ़्ते भर का तेज़ बुख़ार, और उसके साथ निमोनिया का एक
असली हिस्सा, तथा उससे भी बुरा --- और वह भी ऐसे बच्चे पर, जिसके पास
याददाश्त है ही नहीं। ``क़ुदरती'' उसी याददाश्त तक जाने वाली महँगी सड़क का
नाम है, बस सुरक्षा घटाकर।

\textbf{10.} उसके इर्द-गिर्द के टीका लगे लोग --- परिवार, कक्षाएँ, पूरा
क़स्बा: यानी साझी हिफ़ाज़त, वही ढाल जो उन पर तनी रहती है जो सधने के लिए
बहुत छोटे हैं। और उस प्रतिज्ञप्ति में उस फ़र्ज़ का नाम: हिफ़ाज़त का साझा
होना --- यानी न सिखाए जा सकने वालों पर तनी हुई ढाल।

\textbf{11.} मंगलवार ने पास्तूर की लकीर चलाई: जान-बूझकर कमज़ोर या
निष्क्रिय की गई पेशकश, जो निखरकर पुनर्बल ख़ुराकें बनी। और बुधवार के जाँचे
गए संपर्कों तथा भूखे मरते फैलाव ने जेनर की: सधी हुई याददाश्त, जो किसी
\omterm{def:g9:microbes-and-infection:sorted}{विषाणु} और उसके अगले मेज़बान के बीच खड़ी है।

\textbf{12.} ``यह केंद्र याददाश्त देता है: सधी हुई, पुनर्बल की हुई, और
साझी --- यानी वह अकेली दवा, जो दुश्मन के आने से पहले काम करती है।''
\end{solution}
