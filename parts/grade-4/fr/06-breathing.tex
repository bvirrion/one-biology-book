\chapter{La respiration}\label{ch:g4:breathing}

Retiens ton souffle et commence à compter. Bien avant une minute, ta
poitrine réclame de bouger de nouveau, de plus en plus fort. On peut
sauter un repas, le \omterm{prop:g1:healthy-habits:needs}{sommeil} peut attendre ce soir --- mais l'air, il
le faut \emph{maintenant}, jour et nuit, environ quinze fois par
minute. Qu'y a-t-il de si urgent dans une bouffée d'air~?

\section{Les mouvements respiratoires}

\begin{definition}[Inspirer et expirer]\label{def:g4:breathing:movements}
La \emph{respiration}\index{respiration} est l'alternance sans fin
de deux mouvements~:
\begin{itemize}
  \item \emph{inspirer} (l'inspiration)\index{inspiration}~: la cage
        thoracique se dilate et l'air entre par le nez ou la
        bouche~;
  \item \emph{expirer} (l'expiration)\index{expiration}~: la cage
        thoracique se relâche et l'air ressort.
\end{itemize}
Au repos, la paire se répète environ $15$ fois par minute --- plus
vite chez les bébés, plus vite à l'effort --- sans que tu aies
jamais besoin d'y penser.
\end{definition}

\begin{example}[Observer les mouvements]\label{ex:g4:breathing:watch}
Les mains à plat sur les \omterm{def:g3:bones-and-muscles:skeleton}{côtes}~: à l'\omterm{def:g4:breathing:movements}{inspiration}, la cage thoracique
se soulève et s'élargit sous tes paumes~; à l'\omterm{def:g4:breathing:movements}{expiration}, elle
retombe. Le flanc d'un chat qui dort montre la même montée et la
même descente. Les mouvements ne s'arrêtent jamais --- éveillé,
endormi, du premier cri au dernier jour.
\end{example}

\section{Où va l'air}

\begin{definition}[Les organes de la respiration]\label{def:g4:breathing:organs}
L'air inspiré passe par le nez (ou la bouche), descend la
\emph{trachée}\index{trachée} --- le tube ferme et annelé que tu
peux sentir devant ton cou --- qui se divise en deux \omterm{def:g1:plants-around-us:tree}{branches}, une
vers chaque \emph{poumon}\index{poumon}. Les poumons, deux organes
spongieux qui remplissent la cage thoracique (derrière les \omterm{def:g3:bones-and-muscles:skeleton}{côtes} de
\cref{def:g3:bones-and-muscles:skeleton}), se terminent par des
millions de minuscules poches d'air, chacune enveloppée de fins
vaisseaux sanguins.
\end{definition}

\begin{remark}[Spongieux exprès]\label{rem:g4:breathing:sponge}
Des millions de poches au lieu de deux simples sacs~: c'est de
nouveau le tour de force de l'\omterm{def:g4:journey-of-food:tract}{intestin grêle}
(\cref{rem:g4:journey-of-food:surface}). Toutes ces parois de poches
s'additionnent en une énorme surface de rencontre entre l'air et le
\omterm{prop:g4:journey-of-food:absorb}{sang} --- la taille d'un demi-court de tennis, repliée dans ta
poitrine.
\end{remark}

\section{À quoi sert la respiration}

\begin{proposition}[Prendre du dioxygène, rendre du dioxyde de carbone]\label{prop:g4:breathing:exchange}
L'air est un mélange de gaz, et le corps en veut un~: le
\emph{dioxygène}\index{dioxygène}. Dans les poches des \omterm{def:g4:breathing:organs}{poumons}, le
dioxygène de l'air frais passe dans le \omterm{prop:g4:journey-of-food:absorb}{sang}, qui le livre à toutes
les parties du corps~; il y est utilisé, avec les \omterm{def:g4:journey-of-food:digestion}{nutriments}, pour
libérer l'\omterm{prop:g3:balanced-diet:jobs}{énergie} de \cref{prop:g3:balanced-diet:jobs}. Le gaz de
rebut du corps, le \emph{dioxyde de carbone}\index{dioxyde de
carbone}, fait le voyage de retour et est expiré. L'air expiré
contient moins de dioxygène et plus de dioxyde de carbone que l'air
inspiré --- cette différence est ce à quoi la \omterm{def:g4:breathing:movements}{respiration}
\emph{sert}.
\end{proposition}
\admitted

\begin{example}[Pourquoi la course coupe le souffle]\label{ex:g4:breathing:running}
Quand tu sprintes, tes \omterm{def:g1:my-body:parts}{jambes} brûlent de l'\omterm{prop:g3:balanced-diet:jobs}{énergie} vite --- il leur
faut donc du dioxygène vite. La \omterm{def:g4:breathing:movements}{respiration} s'exécute~: d'environ
$15$ souffles calmes par minute, on passe à $40$ ou plus, profonds,
pendant que le cœur (\cref{ch:g4:heart-and-blood} raconte son côté
de l'histoire) s'emballe pour accélérer les livraisons. Le halètement
après la course, c'est le corps qui solde ses comptes de dioxygène.
\end{example}

\begin{example}[Le miroir embué]\label{ex:g4:breathing:mirror}
Souffle sur un miroir froid~: de la buée. L'air expiré revient
changé --- réchauffé, humidifié, et, invisiblement, plus pauvre en
dioxygène et plus riche en dioxyde de carbone. Le miroir montre
l'humidité~; l'échange de gaz qu'il ne peut pas montrer est le plus
important.
\end{example}

\begin{method}[Prendre soin de son souffle]\label{met:g4:breathing:care}
\begin{enumerate}
  \item Aère les pièces chaque jour --- beaucoup de respirants dans
        une pièce fermée, cela fait un air vicié et riche en
        dioxyde de carbone~;
  \item respire par le nez dans l'air froid ou poussiéreux~: il
        réchauffe, humidifie et filtre~;
  \item cours, nage, joue --- les \omterm{def:g4:breathing:organs}{poumons} et les \omterm{def:g3:bones-and-muscles:muscle}{muscles}
        respiratoires exercés deviennent plus forts et plus
        amples~;
  \item tiens la fumée loin de tes \omterm{def:g4:breathing:organs}{poumons}~: elle salit les
        délicates poches, qui ne se nettoient que lentement.
\end{enumerate}
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g4:breathing:1}
Nomme les deux mouvements respiratoires et ce que fait la cage
thoracique dans chacun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:2}
Retrace le trajet d'une bouffée d'air, du nez aux minuscules poches
d'air.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:3}
Combien de fois par minute respire à peu près un enfant calme au
repos~? Qu'arrive-t-il à ce nombre pendant un sprint, et pourquoi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:4}
Quel gaz le corps prend-il à l'air, et lequel rend-il~? Où
exactement se fait l'échange~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:5}
En quoi l'air expiré diffère-t-il de l'air inspiré~? Nomme trois
différences.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:6}
Pourquoi les \omterm{def:g4:breathing:organs}{poumons} sont-ils bâtis en millions de minuscules poches
plutôt qu'en deux simples sacs~? Quel autre organe de cette année
emploie le même tour de force~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:7}
Pourquoi vaut-il mieux respirer par le nez un jour de gel ou de
poussière~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:8}
À toi d'essayer~: assis calmement, compte tes \omterm{def:g4:breathing:movements}{respirations} pendant
une minute. Fais ensuite vingt sauts avec écart et recompte. Donne
les deux nombres et explique le changement.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:9}
Les repas peuvent attendre des heures~; la \omterm{def:g4:breathing:movements}{respiration} ne peut pas
attendre une minute. Que dit cela sur la quantité de dioxygène que
le corps garde en réserve, comparée à ses \omterm{def:g3:animals-through-seasons:active}{réserves de nourriture}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:10}
Une salle de classe est «~étouffante~» en fin de matinée. Qu'a
changé dans son air, d'après \cref{prop:g4:breathing:exchange}, et
que prescrit \cref{met:g4:breathing:care}~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:breathing:11}
La truite n'a pas de \omterm{def:g4:breathing:organs}{poumons}~: l'eau passe sur ses branchies
plumeuses, et le dioxygène \emph{dissous dans l'eau} y passe dans
son \omterm{prop:g4:journey-of-food:absorb}{sang}. Quel travail les branchies et les \omterm{def:g4:breathing:organs}{poumons} partagent-ils~?
Et pourquoi une truite étouffe-t-elle à l'air libre, où le dioxygène
abonde~? (Indice~: hors de l'eau, les surfaces plumeuses des
branchies s'affaissent et se collent.)
\end{exercise}
