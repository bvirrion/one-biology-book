\section*{Chapitre \ref{ch:b1:plant-water-minerals} --- Nutrition hydrique et minérale des plantes}

\begin{solution}{exo:b1:plant-water-minerals:1}
\omterm{def:b1:plant-water-minerals:pathways}{Apoplasme}~: les parois et les espaces continus, traversés sans franchir
de membrane. \omterm{def:b1:plant-water-minerals:pathways}{Symplasme}~: le \omterm{def:b1:cell-unit-of-life:cell}{cytoplasme} connecté des \omterm{def:b1:cell-unit-of-life:cell}{cellules}, où l'on
entre par une membrane et que l'on parcourt par les \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesmes}. Le
\omterm{def:b1:plant-water-minerals:pathways}{cadre de Caspary} barre l'\omterm{def:b1:plant-water-minerals:pathways}{apoplasme} au niveau de l'endoderme, ce qui
force tout à passer par une membrane et empêche le reflux depuis le
\omterm{def:b1:flowering-plant-organization:tissues}{xylème}.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:2}
N (\omterm{def:b1:proteins:peptide}{protéines}, acides nucléiques), P (\omterm{def:b1:nucleic-acids:nucleotide}{nucléotides}, membranes), K (agent
osmotique, \omterm{def:b1:flowering-plant-organization:tissues}{stomates}), Ca (parois, signalisation), Mg (\omterm{def:b1:photosynthesis:pigments}{chlorophylle}),
S (\omterm{def:b1:proteins:aminoacid}{acides aminés} cystéine et méthionine).
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:3}
Sol \qty{-0.2}{}, \omterm{def:b1:flowering-plant-organization:tissues}{xylème} racinaire \qty{-0.6}{}, \omterm{def:b1:cell-unit-of-life:cell}{cellule} foliaire
\qty{-1.3}{MPa}~: du sol vers la \omterm{def:b1:flowering-plant-organization:organs}{racine}, du \omterm{def:b1:flowering-plant-organization:tissues}{xylème} racinaire vers la
\omterm{def:b1:flowering-plant-organization:organs}{feuille} --- toujours vers la valeur la plus négative.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:4}
L'azote est mobile dans le phloème~: la plante le retire des vieilles
\omterm{def:b1:flowering-plant-organization:organs}{feuilles} pour nourrir les jeunes, si bien que les vieilles jaunissent
les premières. Le fer n'est pas remobilisé~: ce sont les jeunes
\omterm{def:b1:flowering-plant-organization:organs}{feuilles}, bâties sans lui, qui jaunissent.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:5}
\omterm{def:b1:cell-unit-of-life:cell}{Cellule} $\Psi = -0.9 + 0.5 = \qty{-0.4}{MPa}$~: sol ($-0.2$) $>$ \omterm{def:b1:cell-unit-of-life:cell}{cellule}
($-0.4$) $>$ \omterm{def:b1:flowering-plant-organization:tissues}{xylème} ($-0.6$), donc l'eau va du sol $\to$ à la \omterm{def:b1:cell-unit-of-life:cell}{cellule}
$\to$ au \omterm{def:b1:flowering-plant-organization:tissues}{xylème}. L'absorption cesse quand la \omterm{def:b1:cell-unit-of-life:cell}{cellule} atteint
\qty{-0.2}{MPa}, c'est-à-dire $\Psi_p = -0.2 + 0.9 = \qty{0.7}{MPa}$.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:6}
$\Psi = 137\ln(\text{HR})$~: \qty{-14}{MPa}, \qty{-95}{MPa},
\qty{-315}{MPa}. Même un air humide est dix fois plus sec, en ce sens,
que le sol au point de flétrissement~: une \omterm{def:b1:flowering-plant-organization:organs}{feuille} perd toujours de
l'eau vers l'air~; la seule question est de savoir à quelle vitesse.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:7}
$E_K = 25.7\ln(0.1/100) = \qty{-178}{mV}$~: à \qty{-150}{mV}, la \omterm{def:b1:cell-unit-of-life:cell}{cellule}
est un peu moins négative qu'il ne faudrait, donc une accumulation d'un
facteur mille demande un peu plus que des canaux --- mais un rapport de
100 serait à l'équilibre~: les canaux en font presque tout. Nitrate~:
$E = -25.7\ln(0.5/5)
= \qty{+59}{mV}$~; la \omterm{def:b1:cell-unit-of-life:cell}{cellule} est à \qty{-150}{}, à \qty{209}{mV} de là
et dans le mauvais sens~: un anion doit être pompé, par symport de
protons.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:8}
La nuit, sans transpiration, le chargement d'ions dans le \omterm{def:b1:flowering-plant-organization:tissues}{xylème} y
attire l'eau et une \omterm{prop:b1:plant-water-minerals:rootpressure}{poussée racinaire} de quelques dixièmes de
mégapascal pousse la sève de quelques mètres au plus --- assez pour
atteindre la pointe des brins d'herbe et en faire sourdre des
gouttelettes, non pour atteindre la cime d'un arbre. Quand les
\omterm{def:b1:flowering-plant-organization:tissues}{stomates} s'ouvrent, la transpiration met le \omterm{def:b1:flowering-plant-organization:tissues}{xylème} sous tension et la
\omterm{prop:b1:plant-water-minerals:rootpressure}{poussée racinaire} disparaît.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:9}
Sans molybdène, la \omterm{prop:b1:plant-water-minerals:nitrate}{nitrate réductase} ne fonctionne pas~: sur nitrate,
la plante ne peut fabriquer d'\omterm{def:b1:proteins:aminoacid}{acides aminés} et montre une carence
azotée alors que le nitrate abonde. Sur ammonium, elle pousse
normalement, puisque l'ammonium entre dans les \omterm{def:b1:proteins:aminoacid}{acides aminés} sans
réduction. La différence localise l'élément dans une seule \omterm{def:b1:enzymes:enzyme}{enzyme}.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:10}
\qty{30}{mg} de N $= \qty{2.14}{mmol}$~; $\times 8 = \qty{17}{mmol}$
d'électrons, \qty{8.6}{mmol} d'équivalents \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}. \omterm{def:b1:flowering-plant-organization:organs}{Feuille}~: $0.05\times
4\times 10^{-6}\times 43\,200 = \qty{8.6}{mmol}$ de $\mathrm{CO_2}$,
\qty{35}{mmol} d'électrons~: la réduction du nitrate prend, à ce compte,
deux fois moins d'électrons que la fixation du carbone --- une grande
fraction, ce qui explique que les plantes la fassent à la lumière et
que les \omterm{def:b1:flowering-plant-organization:organs}{feuilles} à croissance rapide consomment tant de réducteur.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:11}
\qty{2.14}{mmol} de N $= \qty{1.07}{mmol}$ de $\mathrm{N_2}$~: \omterm{def:b1:water-small-molecules:atp}{ATP}
$16\times 1.07 = \qty{17}{mmol}$, soit \qty{0.57}{mmol} de glucose,
\qty{0.10}{g}~; plus 8 électrons par $\mathrm{N_2}$ (encore
\qty{0.05}{g} de glucose). La règle des \qty{6}{g/g} donne
\qty{0.18}{g} par jour~: la nitrogénase elle-même en fait l'essentiel,
et le reste sert à bâtir et entretenir les \omterm{def:b1:plant-water-minerals:nodules}{nodosités}, leur
léghémoglobine, et la respiration des bactéries qui tient l'oxygène bas.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:12}
La \omterm{def:b1:flowering-plant-organization:organs}{racine} prend elle-même l'eau et les ions mobiles (nitrate,
potassium) que l'eau du sol amène à ses poils, en ne dépensant d'\omterm{def:b1:water-small-molecules:atp}{ATP}
que pour la \omterm{prop:b1:plant-water-minerals:uptake}{pompe à protons}. Pour le phosphate, qui ne viendra pas à
elle, elle paie un champignon en sucre pour étendre sa portée~; pour
l'azote, dans un sol sans nitrate, elle paie des bactéries en sucre
pour le fixer. L'honoraire est d'un dixième ou plus de sa
photosynthèse, et la plante cesse de payer quand le service est inutile~:
sur sol riche en phosphate les \omterm{def:b1:plant-water-minerals:mycorrhiza}{mycorhizes} se réduisent, sur sol riche
en nitrate les légumineuses font moins de \omterm{def:b1:plant-water-minerals:nodules}{nodosités} --- on ne garde des
sous-traitants que tant qu'ils coûtent moins cher que le travail seul.
\end{solution}

\begin{solution}{pb:b1:plant-water-minerals:1}
\textbf{1.} $-0.9 + 0.5 = \qty{-0.4}{MPa}$.
\textbf{2.} $-0.2 > -0.4 > -0.6 > -1.3$~: chaque marche plus basse,
l'eau va du sol $\to$ au cortex $\to$ au \omterm{def:b1:flowering-plant-organization:tissues}{xylème} $\to$ à la \omterm{def:b1:flowering-plant-organization:organs}{feuille}.
\textbf{3.} $137\ln 0.5 = \qty{-95}{MPa}$.
\textbf{4.} Du sol à la \omterm{def:b1:flowering-plant-organization:organs}{feuille} \qty{1.1}{MPa}~; de la \omterm{def:b1:flowering-plant-organization:organs}{feuille} à l'air
\qty{94}{MPa}~: \qty{99}{\%} du total tient dans la dernière marche.
\textbf{5.} Un sol à $-0.9$ est plus bas que le $-0.4$ des \omterm{def:b1:cell-unit-of-life:cell}{cellules} du
cortex~: l'eau sort des \omterm{def:b1:cell-unit-of-life:cell}{cellules} jusqu'à ce qu'elles tombent à $-0.9$,
c'est-à-dire que leur \omterm{def:b1:eukaryotic-cell:plantcell}{turgescence} s'annule~; l'absorption cesse jusqu'à
ce que les \omterm{def:b1:cell-unit-of-life:cell}{cellules} abaissent leur $\Psi_s$ en accumulant des solutés.
\textbf{6.} Le sol est désormais sous le $-1.3$ des \omterm{def:b1:flowering-plant-organization:organs}{feuilles}~: l'eau
sort de la plante~; les \omterm{def:b1:flowering-plant-organization:organs}{feuilles} perdent leur \omterm{def:b1:eukaryotic-cell:plantcell}{turgescence} et flétrissent.
L'arrosage relève le $\Psi$ du sol au-dessus de celui de la \omterm{def:b1:flowering-plant-organization:organs}{racine} et
le flux reprend~; les \omterm{def:b1:flowering-plant-organization:organs}{feuilles}, dont les parois étaient intactes, se
regonflent.
\textbf{7.} $2\times 10^{-7}\times 0.2\times(-0.2 - (-0.6)) =
\qty{1.6e-8}{m^3/s}$.
\textbf{8.} $1.6\times 10^{-8}\times 86\,400 = \qty{1.4e-3}{m^3} =
\qty{1.4}{L}$ par jour.
\textbf{9.} \qty{1400}{g} sur \qty{0.5}{m^2} et \qty{12}{h} de jour
(pour l'essentiel)~: environ \qty{230}{g} par mètre carré et par heure.
\textbf{10.} $230/18 = \qty{12.8}{mol}$ par mètre carré et par heure, \qty{3.5}{mmol} par mètre carré et par seconde --- mille molécules d'eau pour chaque $\mathrm{CO_2}$ fixé à \qty{4}{\micro mol}.
\textbf{11.} Différence $-0.2 - (-0.4) = 0.2$~: flux divisé par deux,
\qty{0.7}{L} par jour. La fermeture des \omterm{def:b1:flowering-plant-organization:tissues}{stomates} a réduit la perte, la
tension s'est relâchée et toute l'échelle est remontée~: la plante
échange du carbone contre de l'eau quand l'air est le plus sec.
\textbf{12.} L'eau descend son propre gradient de potentiel, créé par
l'évaporation de la \omterm{def:b1:flowering-plant-organization:organs}{feuille}, de sorte que la \omterm{def:b1:flowering-plant-organization:organs}{racine} est passive~; le
nitrate doit être concentré contre un gradient de concentration et un
gradient électrique, ce que seul le couplage à la \omterm{prop:b1:plant-water-minerals:uptake}{pompe à protons} peut
payer.
\textbf{13.} $\qty{1.4}{L}\times\qty{2}{mmol/L} = \qty{2.8}{mmol}$,
\qty{39}{mg} d'azote.
\textbf{14.} Oui, avec un tiers de marge~; le surplus est stocké dans
les \omterm{def:b1:eukaryotic-cell:plantcell}{vacuoles} sous forme de nitrate ou exclu à l'endoderme.
\textbf{15.} $\Delta G = RT\ln(5/2) + zFV = 2.48\times 0.92 + (-1)(96.5)(-0.15)
= 2.3 + 14.5 = \qty{16.8}{kJ/mol}$.
\textbf{16.} Par proton~: $RT\ln(10^{-7.5}/10^{-5.5}) + F(-0.15) =
-11.4 - 14.5 = \qty{-25.9}{kJ}$~; deux protons~: \qty{-52}{kJ}, trois
fois ce qu'il faut.
\textbf{17.} $\qty{2.14}{mmol}\times 8 = \qty{17}{mmol}$ d'électrons.
\textbf{18.} $0.5\times 4\times 10^{-6}\times 43\,200 = \qty{86}{mmol}$
de $\mathrm{CO_2}$, \qty{346}{mmol} d'électrons~: \qty{5}{\%} pour le
nitrate.
\textbf{19.} Elle économiserait les 8 électrons par azote~; mais
absorber un cation et libérer un proton pour chacun acidifie le sol
autour de la \omterm{def:b1:flowering-plant-organization:organs}{racine} et oblige la \omterm{def:b1:cell-unit-of-life:cell}{cellule} à exporter de l'acide, et
l'ammonium est toxique s'il s'accumule, donc il doit être assimilé
aussitôt dans la \omterm{def:b1:flowering-plant-organization:organs}{racine}.
\textbf{20.} \qty{1.07}{mmol} de $\mathrm{N_2}$ $\times 16 =
\qty{17}{mmol}$ d'\omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{21.} $6\times 0.03 = \qty{0.18}{g}$ de sucre, \qty{6}{\%} de
\qty{3}{g}. Avec \qty{2}{mmol/L} de nitrate disponible pour le coût de
sa réduction (\qty{5}{\%} des électrons), la fixation est un mauvais
marché et une légumineuse fait peu de \omterm{def:b1:plant-water-minerals:nodules}{nodosités}~; sans nitrate, c'est
la seule voie, et \qty{6}{\%} de la production est peu cher pour la
totalité des \omterm{def:b1:proteins:peptide}{protéines} de la plante.
\textbf{22.} $1.4\times 2\times 10^{-6} = \qty{2.8}{\micro mol}$,
\qty{0.09}{mg} de phosphore --- un quarantième du besoin~; la plante
doit compter sur la diffusion vers ses \omterm{def:b1:flowering-plant-organization:organs}{poils absorbants}, qui épuisent
vite leur voisinage, et surtout sur son champignon mycorhizien.
\textbf{23.} Le cadre barre l'\omterm{def:b1:plant-water-minerals:pathways}{apoplasme}, de sorte que le nitrate ne
peut entrer dans le \omterm{def:b1:flowering-plant-organization:tissues}{xylème} que par des membranes endodermiques qui l'y
pompent~; et il empêche la sève brute concentrée de refluer le long des
parois, si bien que le gradient tient.
\textbf{24.} Sa sève brute ressemblerait à la solution du sol ---
diluée, non triée, avec du sodium et d'autres ions indésirables --- et
ne pourrait être tenue concentrée~; en sol salé, le sodium atteindrait
librement les \omterm{def:b1:flowering-plant-organization:organs}{feuilles} et la plante serait empoisonnée, là où une
plante normale l'exclut à l'endoderme.
\textbf{25.} Environ \qty{1.4}{L} d'eau par jour à travers
\qty{0.2}{m^2} de \omterm{def:b1:flowering-plant-organization:organs}{racine}, portant \qty{39}{mg} d'azote pour un besoin
de \qty{30}{}, dont la réduction prend environ \qty{5}{\%} des
électrons photosynthétiques de la \omterm{def:b1:flowering-plant-organization:organs}{feuille}.
\end{solution}
