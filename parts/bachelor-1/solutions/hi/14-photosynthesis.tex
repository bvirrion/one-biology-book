\section*{अध्याय \ref{ch:b1:photosynthesis} --- प्रकाश-संश्लेषण और स्वपोषण}

\begin{solution}{exo:b1:photosynthesis:1}
बाहरी और भीतरी आवरण-झिल्ली, और भीतर \omterm{def:b1:photosynthesis:chloroplast}{थाइलेकॉइड} झिल्ली।
प्रकाश-अभिक्रियाएँ: \omterm{def:b1:photosynthesis:chloroplast}{थाइलेकॉइड} झिल्ली में (प्रोटॉन गुहा में पंप किए जाते
हैं)। \omterm{def:b1:photosynthesis:fixation}{कैल्विन चक्र}: \omterm{def:b1:photosynthesis:chloroplast}{पीठिका} में।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:2}
$2\,\mathrm{H_2O} + 2\,\mathrm{NADP^+} + 3\,\mathrm{ADP} + 3\,\mathrm{P_i}
\to \mathrm{O_2} + 2\,\mathrm{NADPH} + 2\,\mathrm{H^+} + 3\,\mathrm{ATP}$
आठ फ़ोटॉनों के साथ। ऑक्सीजन जल से आती है: जिन शैवाल को
$\mathrm{H_2^{18}O}$ दिया जाए वे $\mathrm{^{18}O_2}$ छोड़ते हैं; और जिन्हें
$\mathrm{C^{18}O_2}$ दिया जाए वे नहीं (रूबेन और केमेन)।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:3}
स्थिरीकरण (\omterm{def:b1:photosynthesis:fixation}{RuBisCO}), अपचयन (G3P तक), पुनर्जनन (RuBP का)। प्रति
$\mathrm{CO_2}$: 3 \omterm{def:b1:water-small-molecules:atp}{ATP} और 2 NADPH.
\end{solution}

\begin{solution}{exo:b1:photosynthesis:4}

आसपास \omterm{def:b1:photosynthesis:pigments}{कैरोटिनॉइड} अवशोषित करते हैं और ऊर्जा \omterm{def:b1:photosynthesis:pigments}{पर्णहरित} को दे देते हैं, इसलिए
ऑक्सीजन छूटती है यद्यपि वहाँ \omterm{def:b1:photosynthesis:pigments}{पर्णहरित} स्वयं कम अवशोषित करता है।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:5}
अकेला सुदूर-लाल प्रकाश (\qty{700}{nm}) कम ही प्रकाश-संश्लेषण चलाता है;
\qty{680}{nm} प्रकाश में जोड़ने पर वह दोनों दरों के योग से अधिक देता है।
भिन्न अवशोषण-उच्चिष्ठों वाले दो वर्णक-निकायों को श्रेणी में सहयोग करना ही
चाहिए, हर एक को अपने फ़ोटॉन चाहिए: प्रकाश-निकाय I (P700) और
प्रकाश-निकाय II (P680)।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:6}
प्रोटॉनों का $0.18\times 96\,500 = \qty{17.4}{kJ/mol}$; $50/17.4 = 2.9$
प्रोटॉन कम से कम; सिंथेज़ चार बरतती है, यानी \qty{72}{\%}

\end{solution}

\begin{solution}{exo:b1:photosynthesis:7}
वह सिद्ध करता है कि \omterm{def:b1:photosynthesis:chloroplast}{थाइलेकॉइड} झिल्ली के आरपार कोई प्रोटॉन-प्रवणता
ATP-संश्लेषण चलाने के लिए बहुत है, बिना किसी प्रकाश और बिना किसी
इलेक्ट्रॉन-परिवहन के: युग्मन प्रवणता से होता है। वह स्वयं यह सिद्ध नहीं
करता कि \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} में प्रकाश-अभिक्रियाएँ \omterm{def:b1:water-small-molecules:atp}{ATP} इसी तरह बनाती हैं (उसके लिए
असंयोजक और pH की मापें चाहिए)। किसी असंयोजक के साथ सिंथेज़ के बरतने से
पहले ही प्रवणता ढह जाती है: कोई \omterm{def:b1:water-small-molecules:atp}{ATP} नहीं।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:8}
$3\times 5 + 3\times 1 = 18$ कार्बन; छह 3-PGA $= 18$; छह G3P $= 18$;
एक G3P बाहर (3) और पाँच G3P (15) तीन RuBP (15) पुनर्जनित करते हैं: $3 +
15 = 18$। संतुलित।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:9}
$\mathrm{CO_2}$ के बिना \omterm{def:b1:photosynthesis:fixation}{RuBisCO} रुक जाती है: RuBP अब खर्च नहीं होता
(चढ़ता है) और 3-PGA अब नहीं बनता (गिरता है) जबकि प्रकाश उसे
अपचयित करके हटाता रहता है। प्रकाश के बिना कोई \omterm{def:b1:water-small-molecules:atp}{ATP} या NADPH नहीं: 3-PGA अब
अपचयित नहीं होता (चढ़ता है) और RuBP अब पुनर्जनित नहीं होता (गिरता है)
जबकि स्थिरीकरण थोड़ी देर चलता रहता है।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:10}
$E = hc/\lambda = 6.63\times 10^{-34}\times 3\times 10^8/6.8\times
10^{-7} = \qty{2.92e-19}{J}$; प्रति मोल \qty{176}{kJ}। 48 फ़ोटॉन:
\qty{8450}{kJ}; दक्षता $2870/8450 = \qty{34}{\%}$। कोई फ़सल वह
प्रकाश खोती है जो अवशोषित नहीं होता (आधा वर्णक्रम, परावर्तन, पारगमन,
पौधों के बीच की खाली जगहें), प्रकाश-श्वसन, \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों}, \omterm{def:b1:flowering-plant-organization:organs}{तनों} तथा \omterm{def:b1:flowering-plant-organization:organs}{जड़ों} का
श्वसन, भरपूर धूप में \omterm{def:b1:enzymes:enzyme}{एंजाइमों} की प्रकाश-संतृप्ति, और वे ऋतुएँ जब \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ}
ही नहीं होतीं।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:11}
प्रति 100 कार्बॉक्सिलन, 33 ऑक्सीजनन जो $16.7$ कार्बन छोड़ते हैं: शुद्ध
$83.3$, यानी \qty{17}{\%} की हानि। उससे बचने की लागत: 2 \omterm{def:b1:water-small-molecules:atp}{ATP} $\approx

कीमत चुकाती है जब हानि लगभग पाँचवें भाग से अधिक हो --- यानी गरम, सूखी
दशाओं में, ठंडी दशाओं में नहीं।
\end{solution}

\begin{solution}{exo:b1:photosynthesis:12}
दोनों ऐसी झिल्लियाँ बरतते हैं जो इलेक्ट्रॉन-परिवहन की ऊर्जा से प्रोटॉन
पंप करती हैं, और ऐसी \omterm{thm:b1:photosynthesis:chemiosmosis}{ATP सिंथेज़} जो प्रवणता खर्च करती है: क्रियाविधि एक ही
है, और दोनों सिंथेज़ समजात हैं। पर इलेक्ट्रॉन उलटी दिशाओं में चलते हैं
--- \omterm{def:b1:photosynthesis:chloroplast}{हरितलवक} में जल से NADPH तक, प्रकाश से उठाए हुए; \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका} में NADH से
ऑक्सीजन तक, गिरते हुए
(\cref{ch:b1:respiration-fermentation}) --- और कार्बन के चक्र
एक-दूसरे के उलट नहीं हैं: \omterm{def:b1:photosynthesis:fixation}{कैल्विन चक्र} और क्रेब्स चक्र कोई \omterm{def:b1:enzymes:enzyme}{एंजाइम} साझा
नहीं करते, और $\mathrm{CO_2}$ का अपचयन NADPH बरतता है जबकि उसका
छूटना $\mathrm{NAD^+}$। समग्र समीकरण उलट हैं; पर
यह तंत्र एक साझी विरासत है जो दो दिशाओं में बरती जाती है, और कार्बन वाले
सिरे पर उसका रसायन अलग है।
\end{solution}

\begin{solution}{pb:b1:photosynthesis:1}
\textbf{1.} $hc/\lambda = \qty{2.92e-19}{J}$; $\times N_A = \qty{176}{kJ/mol}$।
\textbf{2.} \qty{700}{nm}: \qty{171}{kJ/mol}; \qty{450}{nm}:
\qty{266}{kJ/mol}। कोई नीला फ़ोटॉन \omterm{def:b1:photosynthesis:pigments}{पर्णहरित} को किसी ऊँची अवस्था तक
उत्तेजित करता है जो पिकोसेकंडों में उसी न्यूनतम उत्तेजित अवस्था तक क्षय
हो जाती है जहाँ कोई लाल फ़ोटॉन पहुँचाता है; अतिरिक्त ऊष्मा बनकर खो जाता है,
और रसायन दोनों ही तरह एक ही फ़ोटॉन देखता है।
\textbf{3.} प्रति जूल $1/2.92\times 10^{-19} = \num{3.4e18}$ फ़ोटॉन।
\textbf{4.} $\Delta E = 0.82 - (-0.32) = \qty{1.14}{V}$: प्रति मोल
इलेक्ट्रॉन $\Delta G =
96\,500\times 1.14 = \qty{110}{kJ}$।
\textbf{5.} दो फ़ोटॉन: लगभग $176 + 171 = \qty{347}{kJ}$; संचित
\qty{110}{kJ}: \qty{32}{\%}।
\textbf{6.} प्रति $\mathrm{O_2}$ चार इलेक्ट्रॉन (दो जल-अणु),
जो दो NADPH देते हैं: यानी आठ फ़ोटॉन।
\textbf{7.} 12 NADPH: 24 इलेक्ट्रॉन, 48 फ़ोटॉन।
\textbf{8.} $2.303\times 8.314\times 298\times 3/96\,500 = \qty{0.177}{V}$;
प्रति मोल प्रोटॉन $\qty{17.1}{kJ}$।
\textbf{9.} $4 + 8 = 12$ प्रोटॉन प्रति $\mathrm{O_2}$: 3 \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{10.} $6\times 3 = 18$ \omterm{def:b1:water-small-molecules:atp}{ATP}: इस हिसाब से ठीक चक्र की ज़रूरत जितना;
व्यवहार में उपज प्रति $\mathrm{O_2}$ 2.6 के पास होती है
(\omterm{def:b1:photosynthesis:chloroplast}{हरितलवक} में प्रति \omterm{def:b1:water-small-molecules:atp}{ATP} \qty{4.7}{H^+}) और प्रकाश-निकाय I के चारों ओर
चक्रीय बहाव बाकी देता है।
\textbf{11.} \qty{50}{kJ} के एक \omterm{def:b1:water-small-molecules:atp}{ATP} के लिए $4\times 17.1 = \qty{68}{kJ}$:
\qty{73}{\%}।
\textbf{12.} गुहा में $10^{-5}\,\mathrm{mol/L}\times 10^{-15}\,\mathrm{L}\times
6\times 10^{23} = 6$ मुक्त प्रोटॉन, जबकि सिंथेज़ों से होकर प्रति सेकंड
$10^5$: यह अभिवाह गुहा के \omterm{def:b1:proteins:peptide}{प्रोटीनों} तथा \omterm{def:b1:lipids:fattyacid}{लिपिडों} के प्रतिरोधी समूहों से
बँधे प्रोटॉन ढोते हैं, जो उन्हें उतनी ही तेज़ी से छोड़ते हैं जितनी तेज़ी
से वे निकलते हैं।
\textbf{13.} $18\times 50 + 12\times 220 = 900 + 2640 = \qty{3540}{kJ}$
जबकि संचित \qty{2870}{kJ}:
यानी \qty{670}{kJ} (पाँचवाँ भाग) चक्र की अभिक्रियाओं में ऊष्मा बनकर
बिखर जाता है, जिन्हें चलने के लिए ढलान वाला होना ही पड़ता है।
\textbf{14.} $48\times 176 = \qty{8450}{kJ}$; $2870/8450 = \qty{34}{\%}$।
\textbf{15.} $0.34\times 0.45\times 0.9 = \qty{13.8}{\%}$।
\textbf{16.} $400\times 0.005\times 36\,000 = \qty{72}{kJ}$; ग्लूकोज
$0.138\times 72\,000/2\,870\,000 = \qty{3.5e-3}{mol} = \qty{0.62}{g}$।
\textbf{17.} शुद्ध $0.6\times 0.62 = \qty{0.37}{g}$ ग्लूकोज,
यानी $\qty{0.15}{g}$ कार्बन ($72/180$)।
\textbf{18.} पौधों के बीच या मिट्टी पर गिरता प्रकाश; दूसरी \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} की
छाया में क्षमता से नीचे काम करती \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ}, और भरपूर धूप में संतृप्त
\omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} (\omterm{def:b1:enzymes:enzyme}{एंजाइम} सारे फ़ोटॉन बरत ही नहीं सकते); \omterm{def:b1:flowering-plant-organization:organs}{जड़ों}, \omterm{def:b1:flowering-plant-organization:organs}{तनों} का और रात का
श्वसन; प्रकाश-श्वसन; और वे हफ़्ते जब तक छत्रछाया बंद नहीं होती और उसके
जीर्ण होने के बाद।
\textbf{19.} 33 ऑक्सीजनन, 16.7 कार्बन खोए, शुद्ध 83.3 कमाए।
\textbf{20.} उद्धार $33\times 3.5 = 117$ \omterm{def:b1:water-small-molecules:atp}{ATP}; चक्र 300 \omterm{def:b1:water-small-molecules:atp}{ATP}; कुल
83.3 कार्बन के लिए 417 \omterm{def:b1:water-small-molecules:atp}{ATP}: यानी प्रति शुद्ध कार्बन 5.0 \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{21.} प्रति कार्बन $3 + 2 = 5$ \omterm{def:b1:water-small-molecules:atp}{ATP}, और कोई कार्बन खोए बिना ---
यानी \omterm{def:b1:water-small-molecules:atp}{ATP} में बराबर, पर C3 पौधा अपनी कार्बॉक्सिलन-क्षमता का
\qty{17}{\%} भी खो चुका है; \qty{25}{\celsius} पर मोटे तौर पर बराबरी।
\textbf{22.} 50 ऑक्सीजनन, 25 कार्बन खोए, शुद्ध 75; उद्धार 175 \omterm{def:b1:water-small-molecules:atp}{ATP};
C4 पौधे के 5 के सामने $475/75 = 6.3$ \omterm{def:b1:water-small-molecules:atp}{ATP} प्रति शुद्ध कार्बन, और
कार्बन का एक-चौथाई खोया हुआ: C4 साफ़ जीतता है।
\textbf{23.} $\mathrm{CO_2}$ तिगुनी करने से \omterm{def:b1:photosynthesis:fixation}{RuBisCO} की होड़
कार्बॉक्सिलन की ओर खिसक जाती है, जिससे प्रकाश-श्वसन कटता है और C3 पौधों
में शुद्ध स्थिरीकरण चढ़ता है; C4 पौधे \omterm{def:b1:photosynthesis:fixation}{RuBisCO} को पहले ही सांद्रित
$\mathrm{CO_2}$ से संतृप्त रखते हैं और जल की थोड़ी बचत के सिवा कुछ नहीं
पाते।

हैं।
\end{solution}
