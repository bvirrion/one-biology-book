\chapter{Asking Questions in Biology}\label{ch:g6:method}

\omterm{def:g6:method:science}{Biology} is more than memorising names. It is a way of asking questions
about \omterm{def:g1:living-nonliving:living}{living things} and checking answers with evidence. This chapter
builds the habits you will use in every later topic.

\section{Science as a method}

\begin{definition}[Science and biology]\label{def:g6:method:science}
\emph{Science}\index{science} is a public way of building reliable
knowledge about the natural world using testable ideas and evidence.
\emph{Biology}\index{biology} is the science of life and living
\omterm{def:g4:ecosystems:levels}{organisms}.
\end{definition}

\begin{definition}[Observation and inference]\label{def:g6:method:obs}
An \emph{observation}\index{observation} records what can be sensed or
measured. An \emph{inference}\index{inference} is a reasoned conclusion
from observations; it might be wrong even if the observations are
right.
\end{definition}

\begin{example}[Pondweed bubbles]\label{ex:g6:method:bubbles}
\omterm{def:g6:method:obs}{Observation}: bubbles rise from green pondweed in bright \omterm{def:g4:plants-need:needs}{light}.
\omterm{def:g6:method:obs}{Inference}: the bubbles might be oxygen from \omterm{def:g5:photosynthesis:def}{photosynthesis}. A gas test
would check the \omterm{def:g6:method:obs}{inference}.
\end{example}

\section{Hypotheses and experiments}

\begin{definition}[Hypothesis]\label{def:g6:method:hypothesis}
A \emph{hypothesis}\index{hypothesis} is a clear, testable explanation
or prediction about a pattern in nature. If nothing could prove it
wrong, it is not a scientific hypothesis.
\end{definition}

\begin{definition}[Variables]\label{def:g6:method:variables}
In an experiment the \emph{independent variable}\index{variable!independent}
is what you change. The \emph{dependent variable}\index{variable!dependent}
is what you measure. \emph{Controlled variables}\index{variable!controlled}
are factors kept the same.
\end{definition}

\begin{definition}[Control]\label{def:g6:method:control}
A \emph{control}\index{control (experiment)} (control group or
treatment) provides a baseline without the change being tested, so you
can compare fairly.
\end{definition}

\begin{method}[Designing a fair biological test]\label{met:g6:method:fair}
\begin{enumerate}
  \item Write a focused question about a living system.
  \item State a testable \omterm{def:g6:method:hypothesis}{hypothesis}.
  \item Name independent, dependent and controlled variables.
  \item Include a \omterm{def:g6:method:control}{control} or comparison.
  \item Replicate (several \omterm{def:g4:ecosystems:levels}{organisms} or trials).
  \item Record data, then conclude carefully --- and note limits.
\end{enumerate}
\end{method}

\section{Data and honesty}

\begin{proposition}[Evidence rules]\label{prop:g6:method:evidence}
Scientific claims stand only as long as evidence supports them.
Unexpected results are useful: they may falsify a \omterm{def:g6:method:hypothesis}{hypothesis} or reveal
a design flaw.
\end{proposition}
\admitted

\begin{example}[Weak design]\label{ex:g6:method:weak}
``Music makes \omterm{def:g1:plants-animals:plant}{plants} \omterm{def:g2:what-alive:signs}{grow}'' based on one \omterm{def:g1:plants-animals:plant}{plant} by a speaker and one in
a cupboard confuses \omterm{def:g4:plants-need:needs}{light}, temperature and music. A fair test isolates
one factor.
\end{example}
