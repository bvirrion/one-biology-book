\chapter{De reis van het voedsel}\label{ch:g4:journey-of-food}

Eén hap brood: gekauwd, doorgeslikt --- weg. Waarheen? Vanavond helpt
een deel van dat brood je \omterm{def:g1:my-body:parts}{benen} rennen en je \omterm{def:g3:bones-and-muscles:skeleton}{botten} groeien, en dus moet
het op de een of andere manier \emph{uit} de buis komen waarin het
doorgeslikt is en \emph{in} de rest van het lichaam. Volg de hap: zijn
reis is een van de mooiste verhalen van het lichaam.

\section{Het spijsverteringskanaal}

\begin{definition}[Het spijsverteringskanaal]\label{def:g4:journey-of-food:tract}
Voedsel reist door het
\emph{spijsverteringskanaal}\index{spijsverteringskanaal}, één lange
buis dwars door het lichaam:
\begin{enumerate}
  \item de \emph{mond}\index{mond}, waar de tanden vermalen
        (\cref{ch:g2:teeth-and-eating-well}) en het
        \emph{speeksel}\index{speeksel} bevochtigt;
  \item de \emph{slokdarm}\index{slokdarm}, de glijbaan van keel naar maag;
  \item de \emph{maag}\index{maag}, een gespierde zak waarin het voedsel
        urenlang gekneed en met \omterm{def:g4:journey-of-food:digestion}{verteringssappen} gemengd wordt;
  \item de \emph{dunne darm}\index{dunne darm}, een opgevouwen buis van
        meerdere meters, waar de \omterm{def:g4:journey-of-food:digestion}{vertering} afgemaakt wordt en het goede
        uit het voedsel het lichaam in gaat;
  \item de \emph{dikke darm}\index{dikke darm}, waar het water uit de
        rest teruggewonnen wordt voordat de resten het lichaam verlaten.
\end{enumerate}
\end{definition}

\begin{example}[De reis klokken]\label{ex:g4:journey-of-food:timing}
De hap brood brengt seconden in de \omterm{def:g4:journey-of-food:tract}{mond} door, een paar seconden
glijdend door de \omterm{def:g4:journey-of-food:tract}{slokdarm}, enkele uren gekneed in de \omterm{def:g4:journey-of-food:tract}{maag}, nog eens
enkele uren op weg door de \omterm{def:g4:journey-of-food:tract}{dunne darm}, en tot een dag of meer om af te
ronden in de \omterm{def:g4:journey-of-food:tract}{dikke darm}. De hele reis duurt meestal een tot twee dagen.
\end{example}

\section{Vertering: van voedsel naar voedingsstoffen}

\begin{definition}[Vertering en voedingsstoffen]\label{def:g4:journey-of-food:digestion}
\emph{Vertering}\index{vertering} is de omzetting van voedsel, langs het
kanaal, in bestanddelen die klein en eenvoudig genoeg zijn om het
lichaam binnen te gaan: de
\emph{voedingsstoffen}\index{voedingsstof}. Tanden en het kneden van de
\omterm{def:g4:journey-of-food:tract}{maag} breken het voedsel af; \emph{verteringssappen}\index{verteringssap}
--- het \omterm{def:g4:journey-of-food:tract}{speeksel} is er een van --- halen het stuk voor stuk verder uit
elkaar, tot wat brood was een vloeibaar mengsel van voedingsstoffen is.
\end{definition}

\begin{example}[Het brood wordt zoet]\label{ex:g4:journey-of-food:sweet}
Kauw heel lang op een stuk brood zonder beleg --- een langzaam telletje
tot zestig --- en er komt een vage zoetheid op. Er is niets aan
toegevoegd: het \omterm{def:g4:journey-of-food:tract}{speeksel} is het brood al in eenvoudiger, zoetere stukken
aan het afbreken. Je hebt zojuist geproefd hoe de \omterm{def:g4:journey-of-food:digestion}{vertering} begint.
\end{example}

\begin{proposition}[De oversteek het lichaam in]\label{prop:g4:journey-of-food:absorb}
De binnenkant van het kanaal ligt in zekere zin nog buiten het lichaam
--- als het gat van een donut. De oversteek gebeurt in de \emph{\omterm{def:g4:journey-of-food:tract}{dunne
darm}}: door haar dunne, diep gevouwen wanden gaan de \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen}
het \emph{bloed}\index{bloed} in, dat ze naar elk deel van het lichaam
brengt (de bezorgdienst die \cref{ch:g4:heart-and-blood} onderzoekt).
Wat niet kan oversteken --- vezels, vliesjes --- reist verder, verliest
zijn water in de \omterm{def:g4:journey-of-food:tract}{dikke darm} en verlaat het lichaam als ontlasting.
\end{proposition}
\admitted

\begin{remark}[Waarom de plooien en de meters]\label{rem:g4:journey-of-food:surface}
Oversteken kost wandoppervlak, en de \omterm{def:g4:journey-of-food:tract}{dunne darm} is daarvoor gebouwd:
meters lengte, en wanden die vanbinnen geplooid en herplooid zijn. Plat
uitgespreid zou het oversteekoppervlak een kleine kamer bekleden ---
opgevouwen in je buik. Grote klussen krijgen in het lichaam grote
oppervlakken, klein opgevouwen.
\end{remark}

\section{De vertering helpen bij haar werk}

\begin{method}[Gewoonten waar de vertering blij van wordt]\label{met:g4:journey-of-food:habits}
\begin{enumerate}
  \item Kauw goed: de \omterm{def:g4:journey-of-food:tract}{maag} heeft geen tanden --- waar de \omterm{def:g4:journey-of-food:tract}{mond}
        overheen gaat, moet de \omterm{def:g4:journey-of-food:tract}{maag} mee worstelen;
  \item eet rustig, op vaste eettijden;
  \item drink water bij de maaltijd;
  \item eet vezelrijk voedsel --- groente, fruit, volkorenbrood: vezels
        houden de resten in beweging;
  \item bewaar wild spelen of zwemmen tot de maaltijd gezakt is: een
        hardwerkende \omterm{def:g4:journey-of-food:tract}{maag} en hardwerkende \omterm{def:g1:my-body:parts}{benen} concurreren met elkaar.
\end{enumerate}
\end{method}

\begin{example}[De buikpijndetective]\label{ex:g4:journey-of-food:ache}
Lunch naar binnen geschrokt, dan een sprint: de klassieke buikpijn van
de middag. De maaltijd werd halfgekauwd doorgeslikt (regel 1 gebroken)
en de verterende \omterm{def:g4:journey-of-food:tract}{maag} werd een race in gejaagd (regel 5). De oplossing
is geen medicijn maar goede manieren --- tegenover je eigen
\omterm{def:g4:journey-of-food:tract}{spijsverteringskanaal}.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:1}
Noem de delen van het \omterm{def:g4:journey-of-food:tract}{spijsverteringskanaal} op volgorde, van \omterm{def:g4:journey-of-food:tract}{mond} tot uitgang.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:2}
Wat gebeurt er met voedsel in de \omterm{def:g4:journey-of-food:tract}{maag}, en hoe lang blijft het daar
ongeveer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:3}
Wat is \omterm{def:g4:journey-of-food:digestion}{vertering}? Wat zijn \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:4}
In welk deel steken de \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} naar het \omterm{prop:g4:journey-of-food:absorb}{bloed} over? Wat gebeurt
er met wat niet kan oversteken?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:5}
Waarom wordt lang gekauwd brood vaag zoet?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:6}
Waarom is de \omterm{def:g4:journey-of-food:tract}{dunne darm} zo lang en vanbinnen zo geplooid?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:7}
Geef drie verteringsvriendelijke gewoonten en van elk de reden.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:8}
Probeer het (aan tafel): klok je volgende maaltijd. Hoe lang blijft je
eten in je \omterm{def:g4:journey-of-food:tract}{mond} --- en hoe verhoudt zich dat tot de dagen die het nog
voor de boeg heeft?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:9}
Stel de diagnose bij de buikpijn uit \cref{ex:g4:journey-of-food:ache}:
welke twee gewoonten werden gebroken, en wat zou jij morgen anders doen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:10}
``Voedsel gaat het lichaam in op het moment dat je het doorslikt.''
Verbeter dit met \cref{prop:g4:journey-of-food:absorb}: waar gaat
voedsel echt het lichaam in, en wat moet er eerst mee gebeuren?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:journey-of-food:11}
Het gras van de koe is veel taaier voedsel dan jouw brood --- en de koe
heeft vier maagkamers, kauwt haar voedsel twee keer, en herbergt in haar
darmen behulpzame levende wezens die gras uit elkaar halen. Leg met
\cref{def:g4:journey-of-food:digestion} en
\cref{prop:g2:what-animals-eat:match} uit waarom een graseter deze
zwaardere machinerie nodig heeft.
\end{exercise}
