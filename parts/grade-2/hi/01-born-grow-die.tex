\chapter{जन्म, वृद्धि, मृत्यु}\label{ch:g2:born-grow-die}

वसंत में फाटक के पास वाला चेरी का \omterm{def:g1:plants-around-us:tree}{पेड़} फूलों से लद जाता है; उसके नीचे
एक बिल्ली अपने नवजात बच्चों को दूध पिलाती है; और दादाजी अपनी उस समय
की तस्वीर दिखाते हैं जब वे तुम्हारी उम्र के थे --- छोटे, दाँत टूटे
हुए, और तभी से चेरी के शौक़ीन। हर \omterm{def:g1:living-or-not:living}{सजीव} इसी एक रास्ते पर चलता है: वह
शुरू होता है, बढ़ता है, और एक दिन ख़त्म हो जाता है। इसी रास्ते को
\omterm{def:g2:born-grow-die:cycle}{जीवन चक्र} कहते हैं।

\section{जीवन की अवस्थाएँ}

\begin{definition}[जीवन चक्र]\label{def:g2:born-grow-die:cycle}
किसी \omterm{def:g1:living-or-not:living}{सजीव} का \emph{जीवन चक्र}\index{जीवन चक्र} उन
\emph{अवस्थाओं}\index{अवस्था} की कतार है जिनसे उसका जीवन गुज़रता है:
\begin{enumerate}
  \item वह \emph{जन्म लेता है}\index{जन्म लेना} --- यही शुरुआत है;
  \item वह \emph{शावक}\index{शावक} होता है --- तेज़ी से बढ़ता और
        बदलता है;
  \item वह \emph{वयस्क}\index{वयस्क} होता है --- पूरा बढ़ चुका;
        वयस्कों के अपने बच्चे हो सकते हैं;
  \item वह \emph{बूढ़ा}\index{बुढ़ापा} होता जाता है, और अंत में वह
        \emph{मरता है}\index{मृत्यु}।
\end{enumerate}
\end{definition}

\begin{example}[तीन जीवन, एक ही रास्ता]\label{ex:g2:born-grow-die:three}
बिल्ली का बच्चा जन्म लेता है, खेलता है और बड़ी बिल्ली बन जाता है,
जिसके अपने बच्चे हो सकते हैं, और फिर वह बूढ़ी होती है। चेरी का बीज
फूटता है, नन्हा \omterm{def:g1:plants-around-us:tree}{पेड़} सालों तक ऊँचा होता जाता है, \omterm{def:g2:born-grow-die:cycle}{वयस्क} \omterm{def:g1:plants-around-us:tree}{पेड़} \omterm{def:g1:plants-around-us:parts}{फूल} और
चेरियाँ बनाता है, और कई दशकों बाद वह मर जाता है। और तुम: शिशु, बालक,
और फिर एक दिन बड़े। बिल्ली का बच्चा, चेरी का \omterm{def:g1:plants-around-us:tree}{पेड़}, बालक --- वही चार
अवस्थाएँ।
\end{example}

\begin{remark}[इसे चक्र क्यों कहते हैं]\label{rem:g2:born-grow-die:wheel}
हर अकेला जीवन जन्म से मृत्यु तक चलता है, दो सिरों वाली सड़क की तरह।
पर अंत से पहले वयस्कों के बच्चे होते हैं, और वे बच्चे यह रास्ता फिर
से शुरू कर देते हैं। कई पीढ़ियों के पैमाने पर देखो तो जीवन पहिए की
तरह घूमता है --- इसीलिए हम जीवन \emph{चक्र} कहते हैं।
\end{remark}

\section{बढ़ना और बदलना}

\begin{proposition}[बड़े होना सिर्फ़ बड़ा होना नहीं है]\label{prop:g2:born-grow-die:change}
बढ़ते समय कोई नन्हा \omterm{def:g1:living-or-not:living}{सजीव} \emph{बदलता} भी है: शिशु के दाँत आते हैं,
गिरते हैं, और उनकी जगह मज़बूत दाँत आते हैं; चूज़े का पीला रोआँ असली
\omterm{def:g1:animals-around-us:covering}{पंखों} में बदल जाता है; नन्हे \omterm{def:g1:plants-around-us:tree}{पेड़} का मुलायम हरा \omterm{def:g1:plants-around-us:parts}{डंठल} सख़्त लकड़ी बन
जाता है। बढ़ना यानी बड़ा होना \emph{और} अलग होते जाना।
\end{proposition}
\admitted

\begin{example}[तस्वीरों की जाँच]\label{ex:g2:born-grow-die:photo}
दादाजी की पुरानी तस्वीरों में एक ही व्यक्ति को हर \omterm{def:g2:born-grow-die:cycle}{अवस्था} में देखा जा
सकता है: शिशु, स्कूल जाता बालक, नौजवान, पिता, दादा। वही व्यक्ति,
बदलता हुआ शरीर। तस्वीरें \omterm{def:g2:born-grow-die:cycle}{जीवन चक्र} देखने की छोटी समय-मशीनें हैं।
\end{example}

\begin{method}[जीवन की अवस्थाओं को क्रम में लगाना]\label{met:g2:born-grow-die:order}
किसी \omterm{def:g1:living-or-not:living}{सजीव} के जीवन के चित्र गड्डमड्ड मिले हों, तो:
\begin{enumerate}
  \item जन्म वाला चित्र ढूँढ़ो --- अंडा, बीज या नवजात शिशु;
  \item \omterm{def:g2:born-grow-die:cycle}{वयस्क} ढूँढ़ो --- पूरा बढ़ा हुआ, बच्चे कर सकने वाला;
  \item बीच में नन्हों को रखो, छोटे से बड़े की ओर;
  \item अपनी कतार जाँचो: क्या हर चित्र अगले तक ले जाता है?
\end{enumerate}
\end{method}

\section{हर जीवन ख़त्म होता है --- और जीवन चलता रहता है}

\begin{proposition}[जीवन-अवधि]\label{prop:g2:born-grow-die:lifespan}
अलग-अलग \omterm{def:g1:living-or-not:living}{सजीव} अलग-अलग लंबाई का जीवन जीते हैं --- यही
\emph{जीवन-अवधि}\index{जीवन-अवधि} है। तितली शायद कुछ हफ़्ते जिए;
बिल्ली और कुत्ता दस से बीस साल; व्यक्ति अकसर अस्सी साल से भी ज़्यादा;
कुछ \omterm{def:g1:plants-around-us:tree}{पेड़} सैकड़ों साल। लंबा हो या छोटा, हर जीवन का एक अंत होता है।
\end{proposition}
\admitted

\begin{example}[सेब का बूढ़ा पेड़]\label{ex:g2:born-grow-die:oldtree}
बग़ीचे के सिरे पर झुका हुआ सेब का बूढ़ा \omterm{def:g1:plants-around-us:tree}{पेड़} हर साल कम सेब देता है ---
वह अपने बुढ़ापे में है। पर उसके सेबों में बीज हैं, और एक बीज बोया जाए
तो वह पहले से ही सेब का एक नन्हा \omterm{def:g1:plants-around-us:tree}{पेड़} है जो रास्ता शुरू कर रहा है।
बूढ़ा \omterm{def:g1:plants-around-us:tree}{पेड़} मरेगा तब भी सेब के \omterm{def:g1:plants-around-us:tree}{पेड़} ख़त्म नहीं होंगे: उसके बच्चे आगे
चलते रहेंगे।
\end{example}

\begin{remark}[मृत्यु की बात]\label{rem:g2:born-grow-die:death}
मरना हर \omterm{def:g2:born-grow-die:cycle}{जीवन चक्र} का हिस्सा है, और उसके बारे में दुख महसूस होना
बिलकुल ठीक है --- लोगों को होता ही है, ख़ास तौर पर पालतू जानवरों के
और अपने प्रियजनों के लिए। \omterm{def:g1:living-or-not:living}{जीवविज्ञान} एक तसल्ली और जोड़ता है: चूँकि
\omterm{def:g2:born-grow-die:cycle}{वयस्क} मरने से पहले बच्चे कर जाते हैं, इसलिए जीवन ख़ुद चलता रहता है,
चक्र दर चक्र।
\end{remark}

मृत्यु तक सड़क की तरह चलता है? चित्र के बैंगनी तीर का इस्तेमाल करो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:8}

\end{exercise}
