\section*{Chapter \ref{ch:b1:replication-mitosis} --- DNA Replication and Mitosis}

\begin{solution}{exo:b1:replication-mitosis:1}
Polymerases add \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides} only to a $3'$ end, so new strands grow
$5' \to 3'$; the two templates are antiparallel, so at a fork one new
strand can grow toward the fork continuously while the other must
grow away from it, restarting in fragments as more template is
exposed.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:2}
Helicase (unwinds), single-strand binding \omterm{def:b1:proteins:peptide}{proteins} (hold the strands
apart), topoisomerase (relieves twist ahead), primase (\omterm{def:b1:nucleic-acids:chain}{RNA} primers),
\omterm{def:b1:replication-mitosis:fork}{DNA polymerase} (extends primers, proofreads), a second polymerase
(replaces primers), ligase (seals fragments).
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:3}
G$_1$: $q$, 2n \omterm{def:b1:genomes:genome}{chromosomes} of one \omterm{def:b1:replication-mitosis:mitosis}{chromatid}. G$_2$: $2q$, 2n
\omterm{def:b1:genomes:genome}{chromosomes} of two \omterm{def:b1:replication-mitosis:mitosis}{chromatids}. After \omterm{def:b1:replication-mitosis:mitosis}{mitosis}: $q$ per daughter, 2n
\omterm{def:b1:genomes:genome}{chromosomes} each.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:4}
Prophase (condensation, \omterm{def:b1:replication-mitosis:mitosis}{spindle} forms), prometaphase (envelope breaks,
kinetochores attach), metaphase (alignment at the equator), anaphase
(\omterm{def:b1:replication-mitosis:mitosis}{chromatids} separate), telophase (envelopes re-form), then
cytokinesis.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:5}
Half the \omterm{def:b1:genomes:genome}{genome} is lagging-strand synthesis: $\num{6.4e9}/150 =
\num{4.3e7}$ fragments; as many primers, removals and ligations.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:6}
$\num{6.4e9}\times 10^{-5} = \num{64000}$; $\times 10^{-7}$: 640;
$\times 10^{-9}$: about 6 mutations per division.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:7}
$5000/80 = 62$ divisions. A tumour must divide without limit; without
telomerase its \omterm{def:b1:genomes:karyotype}{telomeres} would run out and its \omterm{def:b1:cell-unit-of-life:cell}{cells} would arrest or
die, so nearly all cancers reactivate the \omterm{def:b1:enzymes:enzyme}{enzyme}.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:8}
Cycle $= 1\,\mathrm{h}/0.05 = \qty{20}{h}$; S $= 0.30\times 20 =
\qty{6}{h}$.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:9}
G$_1$: 8 \omterm{def:b1:genomes:genome}{chromosomes}, 8 \omterm{def:b1:replication-mitosis:mitosis}{chromatids}, $q$. G$_2$: 8, 16, $2q$. Metaphase:
8, 16, $2q$. Daughter: 8, 8, $q$.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:10}
New rounds start every \qty{20}{min} at the origin before the
previous round ends: replication is overlapping, and a newborn \omterm{def:b1:cell-unit-of-life:cell}{cell}
already has partly replicated \omterm{def:b1:nucleic-acids:chain}{DNA}. At division the \omterm{def:b1:genomes:genome}{chromosome} finishing
its round was started \qty{40}{min} earlier, a second round started
\qty{20}{min} earlier has forks halfway, and a third is just
beginning: 4 origins and 6 forks in the dividing \omterm{def:b1:cell-unit-of-life:cell}{cell}.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:11}
No \omterm{def:b1:replication-mitosis:mitosis}{spindle} forms: the \omterm{def:b1:genomes:genome}{chromosomes} condense but cannot align or
separate, and the \omterm{def:b1:cell-unit-of-life:cell}{cell} arrests at the \omterm{def:b1:replication-mitosis:mitosis}{spindle} \omterm{prop:b1:replication-mitosis:checkpoints}{checkpoint} in a
metaphase-like state. If it slips back into \omterm{def:b1:replication-mitosis:cycle}{interphase} without
dividing it has 4n \omterm{def:b1:genomes:genome}{chromosomes} (tetraploid). Arrested \omterm{def:b1:cell-unit-of-life:cell}{cells} with
condensed, separate \omterm{def:b1:genomes:genome}{chromosomes} are exactly what a \omterm{def:b1:genomes:karyotype}{karyotype} needs:
the drug is applied to accumulate them.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:12}
Replication guarantees that each \omterm{def:b1:genomes:genome}{chromosome} becomes two identical
\omterm{def:b1:replication-mitosis:mitosis}{chromatids}, to one error in a billion; the \omterm{def:b1:replication-mitosis:mitosis}{spindle} guarantees that
each daughter receives one of each pair, by attaching every \omterm{def:b1:genomes:genome}{chromosome}
from both poles and holding anaphase until every one is under tension.
The G$_2$ \omterm{prop:b1:replication-mitosis:checkpoints}{checkpoint} ensures copying is finished before counting
begins; the \omterm{def:b1:replication-mitosis:mitosis}{spindle} \omterm{prop:b1:replication-mitosis:checkpoints}{checkpoint} ensures counting is set up before
separation. A \omterm{def:b1:cell-unit-of-life:cell}{cell} that mis-segregates ends with one \omterm{def:b1:genomes:genome}{chromosome} too
many or too few (aneuploidy): a copy correct to the letter, delivered
to the wrong address --- which kills most such \omterm{def:b1:cell-unit-of-life:cell}{cells} and characterises
most cancers.
\end{solution}

\begin{solution}{pb:b1:replication-mitosis:1}
\textbf{1.} $50\times 28\,800 = \qty{1.44e6}{bp}$.
\textbf{2.} \qty{2.88e6}{bp}.
\textbf{3.} $\num{6.4e9}/\num{2.88e6} = 2220$ origins.
\textbf{4.} \qty{2.9}{Mb}, i.e.\ $2.9\times 10^6\times 0.34\,\mathrm{nm} =
\qty{0.98}{mm}$ of \omterm{def:b1:nucleic-acids:chain}{DNA} between origins.
\textbf{5.} $\num{6.4e9}/\num{40000} = \qty{160}{kb}$ spacing; each
fork travels about \qty{80}{kb}, half an hour's work.
\textbf{6.} \qty{125}{Mb} per fork at \qty{50}{bp/s}: $\num{2.5e6}$ s,
29 days.
\textbf{7.} The fork speed is limited by the chemistry and by
chromatin (a twentyfold slowing relative to bacteria); origins can be
multiplied without limit, so S phase is shortened by adding starting
points, not by racing the forks.
\textbf{8.} Lagging synthesis covers half the \omterm{def:b1:genomes:genome}{genome}, $\num{3.2e9}$
\omterm{def:b1:nucleic-acids:nucleotide}{nucleotides}: $\num{2.1e7}$ fragments (each origin's two forks each have
a lagging strand, and the halves add to one \omterm{def:b1:genomes:genome}{genome}'s worth).
\textbf{9.} $\num{2.1e7} + 2\times\num{40000} = \num{2.1e7}$ primers.
\textbf{10.} About $\num{2.1e7}$ ligations (one per fragment) plus one
per meeting of forks.
\textbf{11.} $2\times\num{6.4e9} = \num{1.28e10}$ \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides}.
\textbf{12.} $\num{2.56e10}$ \omterm{def:b1:water-small-molecules:atp}{ATP}; over \qty{28800}{s}, $\num{8.9e5}$
per second.
\textbf{13.} About \qty{0.1}{\%} of the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s \omterm{def:b1:water-small-molecules:atp}{ATP} turnover:
polymerisation is cheap; what is expensive is making the \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides}
and the \omterm{def:b1:genomes:nucleosome}{histones}.
\textbf{14.} \num{64000}, 640 and 6.4 errors.
\textbf{15.} $1500\times 10^{-9} = \num{1.5e-6}$: one \omterm{def:b1:genomes:gene}{gene} in seven
hundred thousand per division.
\textbf{16.} $10^{16}\times 6.4 = \num{6.4e16}$ mutations --- every
possible single change of the \omterm{def:b1:genomes:genome}{genome} many times over, scattered among
$10^{13}$ \omterm{def:b1:cell-unit-of-life:cell}{cells}: almost all fall in non-coding \omterm{def:b1:nucleic-acids:chain}{DNA} or in \omterm{def:b1:cell-unit-of-life:cell}{cells} that
are shed, and a mutated \omterm{def:b1:cell-unit-of-life:cell}{cell} is one among billions; only a few
combinations in one \omterm{def:b1:cell-unit-of-life:cell}{cell} matter.
\textbf{17.} A hundredfold ($10^{-7}$): the sequence of mutations that
makes a cancer, which takes decades to accumulate at $10^{-9}$,
accumulates far sooner --- inherited defects in mismatch repair cause
colon cancer in early adulthood.
\textbf{18.} $\num{2.3e6}$ pairs per fork at \qty{1000}{bp/s}:
\qty{2300}{s}, \qty{38}{min}.
\textbf{19.} About two rounds overlapping; a newborn \omterm{def:b1:cell-unit-of-life:cell}{cell} carries
two origins (each already replicated once) and forks halfway along:
its \omterm{def:b1:genomes:genome}{chromosome} is partly replicated at birth.
\textbf{20.} $\num{4.6e6}/1500 = 3070$ fragments per round, about 1.3
per second.
\textbf{21.} $\num{4.6e6}\times 10^{-9} = \num{4.6e-3}$ errors per
round: about one daughter in 200 carries a new mutation.
\textbf{22.} $10^9\times\num{4.6e-3} = \num{4.6e6}$ new mutations per
millilitre; a given \qty{1}{kb} \omterm{def:b1:genomes:gene}{gene} is hit $\num{4.6e6}\times
1000/\num{4.6e6} = 1000$ times.
\textbf{23.} With a thousand independent mutations in any given \omterm{def:b1:genomes:gene}{gene}
per millilitre per division, every possible single change --- including
the ones that confer resistance --- is already present in a large
culture before the antibiotic is applied; the drug selects, it does
not create.
\textbf{24.} Human: $\num{6.4e9}$ pairs, \num{40000} origins,
\qty{50}{bp/s}, \qty{8}{h}. Bacterium: $\num{4.6e6}$ pairs, one origin,
\qty{1000}{bp/s}, \qty{40}{min}. A \omterm{def:b1:genomes:genome}{genome} a thousand times larger,
copied by forks twenty times slower, in only twelve times the time:
the difference is the tens of thousands of origins working in
parallel.
\textbf{25.} About \num{2200} origins at minimum, if all fired at
once; in reality some \num{40000}, one every \qty{160}{kb}, firing in
succession so that each fork travels about \qty{80}{kb}.
\end{solution}
