\chapter{जीन अभिव्यक्ति: अनुलेखन और अनुवादन}\label{ch:b1:gene-expression}

तीस हज़ार \omterm{thm:b1:nucleic-acids:helix}{क्षारक युग्मों} का कोई \omterm{def:b1:genomes:gene}{जीन} चौथाई घंटे में किसी कार्यशील प्रति
में पढ़ लिया जाता है, दो हज़ार अक्षरों तक काट दिया जाता है, केंद्रक से
बाहर भेजा जाता है, और एक साथ पच्चीस \omterm{def:b1:gene-expression:ribosome}{राइबोसोमों} द्वारा पाँच सौ \omterm{def:b1:proteins:aminoacid}{ऐमीनो
अम्लों} के किसी \omterm{def:b1:proteins:peptide}{प्रोटीन} में अनूदित किया जाता है --- हर चार सेकंड में एक,
जब तक वह प्रति चले। \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} इस पर उससे अधिक ऊर्जा ख़र्च करती है जितनी और
किसी काम पर। यह अध्याय जानकारी का \omterm{def:b1:nucleic-acids:chain}{DNA} से \omterm{def:b1:proteins:peptide}{प्रोटीन} तक पीछा करता है: किसी
\omterm{def:b1:genomes:gene}{जीन} की \omterm{def:b1:nucleic-acids:chain}{RNA} में नक़ल, \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकियों} में उस \omterm{def:b1:nucleic-acids:chain}{RNA} का संपादन, वह कूट जो
क्षारकों के त्रिक \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} पर मानचित्रित करता है, वह \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} जो उसे
पढ़ता है, और किसी \omterm{def:b1:proteins:peptide}{प्रोटीन} का बन जाने के बाद क्या होता है।

\section{जीन से प्रोटीन तक}

\begin{proposition}[जानकारी का प्रवाह]\label{prop:b1:gene-expression:dogma}
\index{जीन अभिव्यक्ति}\index{अनुलेखन}\index{अनुवादन}
आनुवंशिक जानकारी \omterm{def:b1:nucleic-acids:chain}{DNA} से \omterm{def:b1:nucleic-acids:chain}{RNA} तक और \omterm{def:b1:nucleic-acids:chain}{RNA} से \omterm{def:b1:proteins:peptide}{प्रोटीन} तक बहती है।
\emph{अनुलेखन} किसी \omterm{def:b1:genomes:gene}{जीन} की एक लड़ी को ऐसे \omterm{def:b1:nucleic-acids:chain}{RNA} में नक़ल करता है जिसका
अनुक्रम दूसरी लड़ी जैसा है (T की जगह U); और \emph{अनुवादन}
\omterm{def:b1:nucleic-acids:rnas}{संदेशवाहक RNA} को एक बार में तीन क्षारक पढ़ता है और उनसे मेल खाते \omterm{def:b1:proteins:aminoacid}{ऐमीनो
अम्लों} को किसी \omterm{def:b1:proteins:peptide}{बहुपेप्टाइड} में जोड़ देता है। हर चरण क्षारकों के युग्मन
पर चलता है: अनुलेखन में \omterm{def:b1:nucleic-acids:chain}{DNA} का \omterm{def:b1:nucleic-acids:chain}{RNA} से, और अनुवादन में संदेशवाहक का
\omterm{def:b1:gene-expression:trna}{अंतरण RNA} से। इस तरह किसी \omterm{def:b1:proteins:peptide}{प्रोटीन} का अनुक्रम उसके \omterm{def:b1:genomes:gene}{जीन} के अनुक्रम की
नक़ल है, और एक क्षारक का परिवर्तन एक \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} बदल सकता है --- यही
उत्परिवर्तन और लक्षणप्ररूप के बीच की कड़ी है।
जानकारी \omterm{def:b1:proteins:peptide}{प्रोटीन} से न्यूक्लिक अम्ल की ओर वापस नहीं बहती; और वह इकलौता
उल्टा चरण, यानी प्रतिवर्ती \omterm{def:b1:genomes:virus}{विषाणुओं} द्वारा \omterm{def:b1:nucleic-acids:chain}{RNA} की \omterm{def:b1:nucleic-acids:chain}{DNA} में नक़ल, \omterm{def:b1:proteins:peptide}{प्रोटीन}
को छूता ही नहीं।
\end{proposition}

\section{अनुलेखन}

\begin{definition}[RNA पॉलिमरेज़, उन्नायक, अनुलेखन]\label{def:b1:gene-expression:transcription}
\index{RNA पॉलिमरेज़}\index{उन्नायक}
\emph{\omterm{def:b1:nucleic-acids:chain}{RNA} पॉलिमरेज़} किसी \omterm{def:b1:nucleic-acids:chain}{DNA} साँचे पर \omterm{def:b1:nucleic-acids:chain}{RNA} संश्लेषित करता है, $5' \to 3'$,
चारों राइबोन्यूक्लियोसाइड ट्राइफ़ॉस्फ़ेटों से, बिना किसी प्राइमर के। वह
किसी \emph{उन्नायक} से शुरू होता है, यानी \omterm{def:b1:genomes:gene}{जीन} से ठीक ऊपर-प्रवाह के उस
अनुक्रम से जिसे वह पहचानता और बाँधता है; जीवाणुओं में
$\sigma$ (\emph{सिग्मा}) नाम की एक उपइकाई यह पहचान करती है, आरंभ से दस
और पैंतीस युग्म पहले के दो संरक्षित खंडों पर। \omterm{def:b1:enzymes:enzyme}{एंजाइम} कुंडली के लगभग
पंद्रह युग्म खोलकर एक \emph{अनुलेखन बुलबुला} बनाता है,
\emph{साँचा लड़ी} ($3' \to 5'$ पढ़ी जाती) को ऐसे \omterm{def:b1:nucleic-acids:chain}{RNA} में नक़ल करता है जो
अनुक्रम में दूसरी, यानी \emph{कूटलेखी} लड़ी जैसा है, और लगभग पचास
\omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} प्रति सेकंड की चाल से आगे बढ़ता है तथा अपने पीछे कुंडली
फिर बंद करता जाता है; और \omterm{def:b1:nucleic-acids:chain}{RNA} बनते ही छिलकर अलग होता जाता है। वह किसी
\emph{समापक} पर रुकता है: जीवाणुओं में यह कोई स्व-पूरक अनुक्रम है जिसका
\omterm{def:b1:nucleic-acids:chain}{RNA} किसी हेयरपिन में मुड़कर अनुलेख को खींचकर मुक्त कर देता है। एक ही \omterm{def:b1:genomes:gene}{जीन}
के सहारे अनेक पॉलिमरेज़ एक-दूसरे के पीछे चल सकते हैं, इसलिए हर सेकंड एक
अनुलेख शुरू किया जा सकता है।
\end{definition}

\begin{proposition}[सुकेंद्रकी अनुलेखन और RNA प्रसंस्करण]\label{prop:b1:gene-expression:processing}
\index{संबंधन}\index{संदेशवाहक RNA}\index{टोपी}\index{पॉली-A पुच्छ}
\omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकियों} में तीन पॉलिमरेज़ हैं: बड़े \omterm{def:b1:nucleic-acids:rnas}{राइबोसोमी RNA} के लिए I,
\omterm{def:b1:nucleic-acids:rnas}{संदेशवाहक RNA} के लिए II, और अंतरण तथा छोटे \omterm{def:b1:nucleic-acids:chain}{RNA} के लिए III। पॉलिमरेज़ II
अपना \omterm{def:b1:gene-expression:transcription}{उन्नायक} अकेले नहीं पहचानता: उस पर \emph{सामान्य अनुलेखन कारकों}
का एक समुच्चय जमता है (अनेक \omterm{def:b1:genomes:gene}{जीनों} में लगभग तीस युग्म ऊपर-प्रवाह के किसी
TATA अनुक्रम पर) और वह \omterm{def:b1:enzymes:enzyme}{एंजाइम} को बुला लाता है, तथा पास या दूर बँधे नियामक
\omterm{def:b1:proteins:peptide}{प्रोटीन} उस दर को घटाते-बढ़ाते हैं
(\cref{ch:b1:expression-control})। प्राथमिक अनुलेख का केंद्रक में
\emph{प्रसंस्करण} होता है: $5'$ सिरे पर एक रूपांतरित ग्वानीन की
\emph{टोपी} जोड़ी जाती है, $3'$ सिरे पर कोई दो सौ ऐडेनीनों की एक पुच्छ
(\emph{पॉली-A}), और \omterm{def:b1:genomes:gene}{इंट्रॉन} \emph{संबंधन} से हटा दिए जाते हैं --- छोटे
\omterm{def:b1:nucleic-acids:chain}{RNA} तथा \omterm{def:b1:proteins:peptide}{प्रोटीनों} का एक संकुल, यानी \emph{संबंधनकाय}, हर \omterm{def:b1:genomes:gene}{इंट्रॉन} के आरंभ
के GU और उसके अंत के AG को पहचानता है, काटता है, और \omterm{def:b1:genomes:gene}{एक्सॉन} जोड़ देता है।
तभी परिपक्व संदेशवाहक \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} को भेजा जाता है।
अनेक \omterm{def:b1:genomes:gene}{जीन} एक से अधिक ढंगों से संबंधित होते हैं (\emph{वैकल्पिक
संबंधन}), इसलिए एक ही \omterm{def:b1:genomes:gene}{जीन} कई \omterm{def:b1:proteins:peptide}{प्रोटीन} देता है; और बीस
हज़ार मानव \omterm{def:b1:genomes:gene}{जीन} शायद एक लाख बनाते हैं।
\end{proposition}
\begin{proof}[साक्ष्य]
शार्प और रॉबर्ट्स (1977) ने किसी \omterm{def:b1:genomes:virus}{विषाणु} \omterm{def:b1:nucleic-acids:rnas}{संदेशवाहक RNA} को उसके \omterm{def:b1:genomes:gene}{जीन} के \omterm{def:b1:nucleic-acids:chain}{DNA}
से संकरित किया और उन संकरों को \omterm{prop:b1:cell-unit-of-life:microscopes}{इलेक्ट्रॉन सूक्ष्मदर्शी} में देखा: \omterm{def:b1:nucleic-acids:chain}{RNA}
\omterm{def:b1:nucleic-acids:chain}{DNA} से कई खंडों में युग्मित था, और उनके बीच \omterm{def:b1:nucleic-acids:chain}{DNA} अयुग्मित फंदों के रूप में
बाहर निकला था --- यानी \omterm{def:b1:genomes:gene}{इंट्रॉन}, जो \omterm{def:b1:genomes:gene}{जीन} में मौजूद हैं और संदेश से अनुपस्थित।
इसी तरह जाँचे गए जीवाणु \omterm{def:b1:genomes:gene}{जीनों} ने कोई फंदा नहीं दिया। केंद्रक में मापे गए
किसी \omterm{def:b1:genomes:gene}{जीन} के प्राथमिक अनुलेख का आकार \omterm{def:b1:genomes:gene}{जीन} से मेल खाता है; और
कोशिकाद्रव्यी संदेशवाहक का आकार अकेले \omterm{def:b1:genomes:gene}{एक्सॉनों} से।
\end{proof}

\section{आनुवंशिक कूट}

\begin{definition}[आनुवंशिक कूट]\label{def:b1:gene-expression:code}
\index{आनुवंशिक कूट}\index{कोडॉन}\index{वाचन-चौखटा}
\emph{आनुवंशिक कूट} संदेशवाहक के क्षारकों के त्रिकों, यानी
\emph{कोडॉनों} को, \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} पर मानचित्रित करता है। 64 कोडॉनों में से
61 बीस \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} बताते हैं और तीन (UAA, UAG, UGA) \emph{विराम} संकेत
हैं; AUG मेथिओनीन बताता है और साथ ही \emph{आरंभ} कोडॉन भी है, इसलिए हर
नया \omterm{def:b1:proteins:peptide}{बहुपेप्टाइड} मेथिओनीन से शुरू होता है। यह कूट
\emph{अपह्रासी} है --- अधिकांश \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} के कई कोडॉन हैं, जो अधिकतर
तीसरे स्थान पर भिन्न हैं --- \emph{अनतिव्यापी} है, और बिना किसी
विरामचिह्न के उस \emph{वाचन-चौखटे} में पढ़ा जाता है जिसे आरंभ कोडॉन तय
करता है, और वह लगभग \emph{सार्वत्रिक} है, यानी जीवाणुओं, पौधों और जंतुओं
में वही, और \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाओं} में मामूली भेदों के साथ: किसी जीवाणु में डाला
गया कोई मानव \omterm{def:b1:genomes:gene}{जीन} ठीक-ठीक पढ़ा जाता है।
\end{definition}

\begin{proposition}[कूट कैसे पढ़ा गया]\label{prop:b1:gene-expression:nirenberg}
यह कूट अनतिव्यापी त्रिक कूट है, और हर \omterm{def:b1:gene-expression:code}{कोडॉन} का अर्थ रसायन से तय किया जा
सकता है।
\end{proposition}
\begin{proof}[साक्ष्य]
क्रिक और ब्रेनर (1961) ने किसी फ़ेज \omterm{def:b1:genomes:gene}{जीन} में ऐसे उत्परिवर्तन किए जो एक
क्षारक जोड़ते या हटाते थे: ऐसा एक परिवर्तन \omterm{def:b1:genomes:gene}{जीन} का कार्य नष्ट कर देता था,
और दो भी, पर पास-पास तीन प्रवेश (या तीन विलोपन) उसे लौटा देते थे --- यानी
संदेश किसी नियत आरंभ-बिंदु से तीन-तीन में पढ़ा जाता है, और कोई भी प्रवेश
नीचे-प्रवाह की हर चीज़ का चौखटा खिसका देता है।
नीरनबर्ग और मथाई (1961) ने \emph{E. coli} के किसी कोशिका-मुक्त निष्कर्ष
में, बीसों \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} के साथ, केवल यूरैसिलों का कोई कृत्रिम \omterm{def:b1:nucleic-acids:chain}{RNA} मिलाया:
उसने केवल फ़ेनिलऐलानीन की शृंखला बनाई, इसलिए UUU का अर्थ Phe है; और दूसरे
कृत्रिम संदेशवाहकों ने, तथा फिर \omterm{def:b1:gene-expression:ribosome}{राइबोसोमों} पर उनके \omterm{def:b1:gene-expression:trna}{अंतरण RNA} के साथ एकल
त्रिन्यूक्लियोटाइडों के बंधन ने, 1966 तक चौंसठों तय कर दिए।
\end{proof}

\begin{example}[कोई संदेश पढ़ना]\label{ex:b1:gene-expression:reading}
संदेशवाहक $5'$-\ldots{}GC\underline{AUG}GCUUUCGGAUAA\ldots{}-$3'$
AUG से पढ़ा जाता है: AUG GCU UUC GGA UAA --- Met-Ala-Phe-Gly-विराम:
यानी चार अवशेषों का एक पेप्टाइड। AUG के बाद का पहला G हटा दीजिए और चौखटा
खिसक जाता है: AUG CUU UCG GAU AA\ldots{} --- Met-Leu-Ser-Asp\ldots, यानी
अलग लंबाई का एक अलग \omterm{def:b1:proteins:peptide}{प्रोटीन}। UUC को UUU कर दीजिए और कुछ नहीं
बदलता: दोनों Phe हैं, यानी एक \emph{मौन} उत्परिवर्तन।
\end{example}

\section{अनुवादन}

\begin{definition}[अंतरण RNA और उसके सिंथेटेज़]\label{def:b1:gene-expression:trna}
\index{अंतरण RNA}\index{प्रतिकोडॉन}\index{ऐमीनोऐसिल-tRNA सिंथेटेज़}
\emph{अंतरण \omterm{def:b1:nucleic-acids:chain}{RNA}} (\cref{ch:b1:nucleic-acids}) \omterm{def:b1:gene-expression:code}{कोडॉन} और \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} के
बीच का अनुकूलक है: उसका \emph{प्रतिकोडॉन}, यानी किसी फंदे में तीन
क्षारक, संदेश के किसी \omterm{def:b1:gene-expression:code}{कोडॉन} से प्रतिसमांतर युग्म बनाता है, और उसका $3'$
सिरा उससे मेल खाता \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} ढोता है। \omterm{def:b1:gene-expression:code}{कोडॉन} के तीसरे स्थान पर युग्मन
ढीला होता है (\emph{डगमग}), इसलिए इकसठ \omterm{def:b1:gene-expression:code}{कोडॉनों} के लिए लगभग चालीस tRNA
काफ़ी हैं। हर tRNA को उसका अपना
\emph{ऐमीनोऐसिल-tRNA सिंथेटेज़} लादता है, यानी वह \omterm{def:b1:enzymes:enzyme}{एंजाइम} जो \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल}
और tRNA दोनों को पहचानता है और उन्हें एक \omterm{def:b1:water-small-molecules:atp}{ATP} की क़ीमत पर दो चरणों में
जोड़ देता है (जो AMP तक टूटता है: यानी दो उच्च-ऊर्जा बंध), और ऐसा करते
हुए \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} का प्रूफ़-पठन भी करता है। ये बीस \omterm{def:b1:enzymes:enzyme}{एंजाइम} ही वे जगहें हैं
जहाँ कूट सचमुच लागू होता है: \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} यह नहीं जाँचता कि कोई tRNA कौन-सा
\omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} ढो रहा है, केवल यह कि उसका प्रतिकोडॉन मेल खाता है।
\end{definition}

\begin{definition}[राइबोसोम]\label{def:b1:gene-expression:ribosome}
\index{राइबोसोम}
\emph{राइबोसोम} दो उपइकाइयों का एक कण है, और हर उपइकाई \omterm{def:b1:nucleic-acids:rnas}{राइबोसोमी
RNA} तथा \omterm{def:b1:proteins:peptide}{प्रोटीनों} की है (जीवाणुओं में: एक छोटी 30S उपइकाई और एक बड़ी 50S
उपइकाई, जो मिलकर 70S, \qty{2.5}{MDa}; \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकी} राइबोसोम
बड़े हैं, 80S)। छोटी उपइकाई संदेशवाहक को बाँधती है और उसे पढ़ती है;
और बड़ी उपइकाई tRNA थामती है तथा \omterm{def:b1:proteins:peptide}{पेप्टाइड बंध} बनाती है --- और
उत्प्रेरक स्वयं \omterm{def:b1:nucleic-acids:rnas}{राइबोसोमी RNA} है। तीन tRNA स्थल दोनों उपइकाइयों पर
फैले हैं: A (ऐमीनोऐसिल, जहाँ अगला लदा हुआ tRNA घुसता है), P
(पेप्टाइडिल, जो बढ़ती शृंखला थामे है), E (निकास)।
\end{definition}

\begin{proposition}[दीर्घीकरण का चक्र]\label{prop:b1:gene-expression:elongation}
अनुवादन तब शुरू होता है जब छोटी उपइकाई आरंभ \omterm{def:b1:gene-expression:code}{कोडॉन} ढूँढ़ लेती है ---
जीवाणुओं में AUG से ठीक ऊपर-प्रवाह के किसी स्थल से किसी \omterm{def:b1:nucleic-acids:rnas}{राइबोसोमी RNA}
अनुक्रम का युग्म बनाकर, और \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकियों} में टोपी बाँधकर तथा पहले AUG तक
बुहारते हुए --- और आरंभक tRNA (मेथिओनीन) P स्थल में बैठ जाता है; फिर बड़ी
उपइकाई आ जुड़ती है। \emph{दीर्घीकरण} का हर चक्कर एक अवशेष जोड़ता है: कोई
लदा हुआ tRNA, जिसका \omterm{def:b1:gene-expression:trna}{प्रतिकोडॉन} A-स्थल के \omterm{def:b1:gene-expression:code}{कोडॉन} से मेल खाता है, किसी
दीर्घीकरण कारक द्वारा पहुँचाया और जाँचा जाता है (एक GTP); बड़ी उपइकाई का
\omterm{def:b1:nucleic-acids:rnas}{राइबोसोमी RNA} बढ़ती शृंखला को P-स्थल के tRNA से A-स्थल के tRNA के \omterm{def:b1:proteins:aminoacid}{ऐमीनो
अम्ल} पर हस्तांतरित करके \omterm{def:b1:proteins:peptide}{पेप्टाइड बंध} बनाता है; \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} एक \omterm{def:b1:gene-expression:code}{कोडॉन} आगे
खिसकता है (एक दूसरा GTP), और tRNA P तथा E पर सरक जाते हैं, और ख़ाली वाला
निकल जाता है। किसी विराम \omterm{def:b1:gene-expression:code}{कोडॉन} पर कोई \emph{मुक्तिकारक कारक} A स्थल में
घुसता है और शृंखला \omterm{prop:b1:water-small-molecules:families}{जल-अपघटन} से मुक्त हो जाती है। जीवाणु \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} प्रति
सेकंड पंद्रह से बीस अवशेष जोड़ते हैं, \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकी} दो से पाँच; और एक ही
संदेश को कई \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} एक साथ पढ़ते हैं, जिससे कोई \emph{बहुसूत्र} बनता है।
\end{proposition}

\begin{method}[किसी प्रोटीन के संश्लेषण का हिसाब]\label{met:b1:gene-expression:reckon}
\begin{enumerate}
  \item लंबाई: $n$ अवशेषों के किसी \omterm{def:b1:proteins:peptide}{प्रोटीन} को
        $3n + 3$ \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइडों} का कूटलेखी अनुक्रम चाहिए (विराम \omterm{def:b1:gene-expression:code}{कोडॉन}
        सहित), जो किसी ऐसे संदेशवाहक के भीतर है जो अपने अनूदित न होने
        वाले सिरों से और, \omterm{def:b1:genomes:gene}{जीन} में, अपने \omterm{def:b1:genomes:gene}{इंट्रॉनों} से लंबा है।
  \item समय: $n$ को दीर्घीकरण दर से भाग दीजिए (जीवाणुओं में प्रति सेकंड
        \qtyrange{15}{20}{}, \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकियों} में \qtyrange{2}{5}{});
        संदेश को हर \qtyrange{80}{100}{} \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} पर एक \omterm{def:b1:gene-expression:ribosome}{राइबोसोम}
        पढ़ रहा होता है, इसलिए प्रति संदेश उत्पादन हर
        (दूरी / दर) सेकंड में एक शृंखला है।
  \item ऊर्जा: प्रति अवशेष चार उच्च-ऊर्जा फ़ॉस्फ़ेट बंध (tRNA लादने के
        लिए दो, \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} पर दो GTP), और साथ में संदेश का अनुलेखन प्रति
        \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} दो पर, जो उस संदेश से बनी सारी शृंखलाओं में बँट
        जाता है।
  \item सत्यनिष्ठा: $10^4$ \omterm{def:b1:gene-expression:code}{कोडॉनों} में लगभग एक ग़लत \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल}; इसलिए
        \qty{500}{} अवशेषों का कोई \omterm{def:b1:proteins:peptide}{प्रोटीन} बीस में से एक प्रति में कहीं
        न कहीं ग़लत है --- और यह सह्य है, क्योंकि \omterm{def:b1:proteins:peptide}{प्रोटीन} बदले जा सकते
        हैं और भूलें वंशागत नहीं होतीं।
\end{enumerate}
\end{method}

\section{अनुवादन के बाद}

\begin{proposition}[किसी नए बहुपेप्टाइड का क्या होता है]\label{prop:b1:gene-expression:after}
\index{संकेत पेप्टाइड}\index{अनुवादनोत्तर रूपांतरण}
शृंखला निकलते-निकलते ही मुड़ने लगती है, और \omterm{prop:b1:proteins:denaturation}{परिचारक} उसकी मदद करते हैं
(\cref{ch:b1:proteins})। उसका गंतव्य उसके अनुक्रम में ही लिखा है:
N-सिरे पर कोई \emph{\omterm{prop:b1:gene-expression:after}{संकेत पेप्टाइड}} किसी \emph{संकेत
\omterm{prop:b1:water-small-molecules:weak}{अभिज्ञान} कण} से बँधता है जो अनुवादन रोक देता है, \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} को
\omterm{def:b1:eukaryotic-cell:endomembrane}{अंतर्द्रव्यी जालिका} पर टिका देता है, और शृंखला को उसकी अवकाशिका में पिरो
देता है (\cref{ch:b1:eukaryotic-cell}); दूसरे अनुक्रम \omterm{def:b1:proteins:peptide}{प्रोटीनों} को
संश्लेषण के बाद \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाओं}, \omterm{def:b1:eukaryotic-cell:plastid}{हरितलवकों}, केंद्रक या \omterm{def:b1:eukaryotic-cell:peroxisome}{परऑक्सीकायों} में
भेजते हैं; और जिस \omterm{def:b1:proteins:peptide}{प्रोटीन} पर कोई संकेत न हो वह \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} में ही रह जाता
है। अनेक \omterm{def:b1:proteins:peptide}{प्रोटीन} \emph{रूपांतरित} होते हैं: आरंभिक मेथिओनीन हटा दिया
जाता है, ER तथा \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} में शर्कराएँ जोड़ी जाती हैं, काइनेज़ों तथा
फ़ॉस्फ़ेटेज़ों से फ़ॉस्फ़ेट जोड़े और हटाए जाते हैं, \omterm{def:b1:lipids:fattyacid}{लिपिड} जोड़े जाते हैं,
टुकड़े काटे जाते हैं (इंसुलिन एक ही शृंखला के रूप में बनकर दो में काटा
जाता है; ज़ाइमोजन सक्रिय करने के लिए काटे जाते हैं)। और हर \omterm{def:b1:proteins:peptide}{प्रोटीन} अंततः
\emph{तोड़ा} जाता है: छोटे \omterm{def:b1:proteins:peptide}{प्रोटीन} \emph{यूबिक्विटिन} से चिह्नित होकर
\emph{प्रोटिएज़ोम} में खोला और पचाया जाता है, मिनटों (नियामक \omterm{def:b1:proteins:peptide}{प्रोटीन}) से
लेकर महीनों (हीमोग्लोबिन) के जीवन के बाद --- ताकि जो \omterm{def:b1:proteins:peptide}{प्रोटीन} मौजूद हों वे
वही हों जिन्हें \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} इस समय बना रही है।
\end{proposition}

\begin{example}[किसी कोशिका की संख्याएँ]\label{ex:b1:gene-expression:numbers}
बढ़ता हुआ कोई \emph{E. coli} कोई \num{20000} \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} रखता है, और हर एक
प्रति सेकंड बीस अवशेष जोड़ता है: यानी प्रति सेकंड $\num{4e5}$ अवशेष, और
घंटे भर में \qty{300}{} अवशेषों के दस लाख \omterm{def:b1:proteins:peptide}{प्रोटीन} --- यानी उसकी अपनी
अंतर्वस्तु, और हर घंटे दुगुना होने के लिए यही चाहिए। उसकी आधी
ऊर्जा \omterm{def:b1:proteins:peptide}{प्रोटीन} संश्लेषण में जाती है। किसी यकृत \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में एक करोड़
\omterm{def:b1:gene-expression:ribosome}{राइबोसोम} हैं और वह विभाजन के लिए नहीं, हफ़्तों के उपयोग के लिए \omterm{def:b1:proteins:peptide}{प्रोटीन}
बनाती है; और प्रतिरक्षी स्रावित करती कोई प्लाज़्मा \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} उनमें से लगभग
सब एक ही \omterm{def:b1:proteins:peptide}{प्रोटीन} पर लगा देती है और प्रति सेकंड दो हज़ार अणु उँड़ेलती है।
\end{example}

\section{समस्या: किसी जीन से किसी प्रोटीन तक}

\begin{problem}[{सप्ताहांत समस्या --- तीस किलोक्षारक का एक जीन पाँच सौ
अवशेषों के प्रोटीन तक पीछा किया गया: अनुलेखन समयबद्ध, संबंधन तौला गया,
राइबोसोम गिने गए, ऊर्जा तथा भूलें आँकी गईं, और अंत में एक प्रोटीन की
ATP में क़ीमत}]
\label{pb:b1:gene-expression:1}
कोई मानव \omterm{def:b1:genomes:gene}{जीन} \qty{30}{kb} में फैला है जिसमें आठ \omterm{def:b1:genomes:gene}{एक्सॉन} मिलकर परिपक्व
संदेशवाहक के \qty{2000}{} \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} बनाते हैं, जिनमें से \qty{1503}{}
कूट करते हैं (500 अवशेष और विराम)। पॉलिमरेज़ II प्रति सेकंड
\qty{30}{} \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} अनुलेखित करता है; \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} प्रति सेकंड \qty{4}{}
अवशेष दीर्घित करते हैं और प्रति \qty{80}{} \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} एक की दूरी पर
बैठते हैं; और संदेशवाहक का अर्ध-आयु \qty{2}{h} है। लागत: प्रति अनुलेखित
\omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} दो उच्च-ऊर्जा बंध, और प्रति अनूदित अवशेष चार;
एक उच्च-ऊर्जा बंध को एक \omterm{def:b1:water-small-molecules:atp}{ATP} मानिए। भूल-दरें:
प्रति अनुलेखित \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} $10^{-5}$, प्रति अनूदित \omterm{def:b1:gene-expression:code}{कोडॉन} $10^{-4}$।

\medskip
\noindent\textbf{भाग I --- अनुलेखन और संबंधन।}
\begin{enumerate}
  \item एक पॉलिमरेज़ को \omterm{def:b1:genomes:gene}{जीन} अनुलेखित करने में कितना समय लगता है?
  \item संबंधन प्राथमिक अनुलेख का कितना भिन्न हटा देता है?
  \item परिपक्व संदेशवाहक के हर \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} के लिए कितने
        \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} अनुलेखित होते हैं?
  \item एक प्राथमिक अनुलेख के अनुलेखन पर ख़र्च \omterm{def:b1:water-small-molecules:atp}{ATP} निकालिए।
  \item यदि हर \qty{10}{s} में एक पॉलिमरेज़ शुरू हो, तो एक साथ \omterm{def:b1:genomes:gene}{जीन} पर
        कितने पॉलिमरेज़ होते हैं, और वह \omterm{def:b1:genomes:gene}{जीन} प्रति घंटा कितने संदेशवाहक
        बनाता है?
  \item \omterm{def:b1:genomes:gene}{इंट्रॉनों} की संख्या और उनकी माध्य लंबाई निकालिए।
  \item संबंधनकाय हर \omterm{def:b1:genomes:gene}{इंट्रॉन} लगभग एक मिनट में, समांतर में हटाता है।
        \omterm{def:b1:genomes:gene}{जीन} से संदेशवाहक तक का समय संबंधन तय करता है या अनुलेखन?
  \item \omterm{def:b1:genomes:gene}{जीन} की और परिपक्व संदेशवाहक की भौतिक लंबाई निकालिए
        (प्रति \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} \qty{0.34}{nm})।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- अनुवादन।}
\begin{enumerate}[resume]
  \item एक \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} को वह \omterm{def:b1:proteins:peptide}{प्रोटीन} बनाने में कितना समय लगता है?
  \item एक संदेशवाहक को एक साथ कितने \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} पढ़ते हैं?
  \item एक संदेशवाहक प्रति घंटा कितने \omterm{def:b1:proteins:peptide}{प्रोटीन} अणु देता है?
  \item अपने \qty{2}{h} के अर्ध-आयु में वह कुल कितने देता है
        (अर्ध-आयु $t_{1/2}$ का कोई संदेश औसतन
        $t_{1/2}/\ln 2$ घंटे का पूरा उत्पादन देता है)?
  \item एक \omterm{def:b1:proteins:peptide}{प्रोटीन} के अनुवादन पर ख़र्च \omterm{def:b1:water-small-molecules:atp}{ATP} निकालिए।
  \item प्रति \omterm{def:b1:proteins:peptide}{प्रोटीन} अनुलेखन पर ख़र्च \omterm{def:b1:water-small-molecules:atp}{ATP} निकालिए, और अनुलेख की लागत
        प्रश्न 12 के \omterm{def:b1:proteins:peptide}{प्रोटीनों} में बाँटिए।
  \item एक \omterm{def:b1:proteins:peptide}{प्रोटीन} की कुल लागत और उसका वह भिन्न निकालिए जो अनुवादन का
        है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- भूलें।}
\begin{enumerate}[resume]
  \item यह प्रायिकता निकालिए कि किसी दिए गए संदेशवाहक के \qty{1503}{}
        कूटलेखी \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइडों} में कम से कम एक अनुलेखन भूल है।
  \item यह प्रायिकता निकालिए कि किसी दिए गए \omterm{def:b1:proteins:peptide}{प्रोटीन} अणु में कम से कम एक
        अनुवादन भूल है।
  \item \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} परिवर्तनों का लगभग एक-चौथाई मौन है और ऐमीनो-अम्ल
        प्रतिस्थापनों का एक-तिहाई हानिरहित। बने हुए \omterm{def:b1:proteins:peptide}{प्रोटीन} अणुओं का
        कितना भिन्न दोषपूर्ण है?
  \item कोई दोषपूर्ण संदेशवाहक दो घंटे तक दोषपूर्ण \omterm{def:b1:proteins:peptide}{प्रोटीन} देता है; और
        कोई दोषपूर्ण \omterm{def:b1:proteins:peptide}{प्रोटीन} एक ही अणु है। समझाइए कि \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} अनुवादन की
        भूल-दर अनुलेखन की दर से दस गुना, और दोनों को प्रतिकृतिकरण की दर
        से कहीं ऊपर, क्यों सह सकती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} का बजट।}
\omterm{def:b1:cell-unit-of-life:cell}{कोशिका} इस \omterm{def:b1:proteins:peptide}{प्रोटीन} की \num{3000} प्रतियाँ प्रति घंटा बनाती है, और कुल
मिलाकर प्रतिदिन \omterm{def:b1:proteins:peptide}{प्रोटीन} के \num{3e9} अवशेष।
\begin{enumerate}[resume]
  \item इस \omterm{def:b1:genomes:gene}{जीन} के कितने संदेशवाहक एक साथ मौजूद होने चाहिए
        (हर एक प्रश्न 11 की संख्या देता है)?
  \item इस \omterm{def:b1:proteins:peptide}{प्रोटीन} में कितने \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} लगे रहते हैं?
  \item \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} अपने सारे \omterm{def:b1:proteins:peptide}{प्रोटीन} संश्लेषण पर प्रतिदिन जितना \omterm{def:b1:water-small-molecules:atp}{ATP} ख़र्च
        करती है वह (प्रति अवशेष चार) निकालिए, और \qty{50}{kJ/mol} पर
        \qty{2000}{\micro m^3} की किसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के लिए शक्ति वाट में।
  \item यदि \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} प्रति सेकंड \qty{1e9}{} \omterm{def:b1:water-small-molecules:atp}{ATP} खपाती हो तो उसकी कुल
        शक्ति से तुलना कीजिए।
  \item कोई दवा संबंधनकाय रोक देती है। इस \omterm{def:b1:proteins:peptide}{प्रोटीन} पर और किसी जीवाणु
        \omterm{def:b1:proteins:peptide}{प्रोटीन} पर उसके प्रभाव की भविष्यवाणी कीजिए।
  \item परिणाम बताइए: उस \omterm{def:b1:proteins:peptide}{प्रोटीन} के एक अणु की \omterm{def:b1:water-small-molecules:atp}{ATP} लागत, अनुवादन तथा
        अनुलेखन में बँटी हुई, और अनुलेखन शुरू होने से पहले पूरे \omterm{def:b1:proteins:peptide}{प्रोटीन}
        तक का समय।
\end{enumerate}
\end{problem}
