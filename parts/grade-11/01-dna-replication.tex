\chapter{DNA Structure and Replication}\label{ch:g11:dna-replication}

Every living \omterm{def:g10:the-cell:cell}{cell} carries a chemical library written in a four-letter alphabet.
That library is \omterm{def:g11:dna-replication:helix}{DNA}: long, double-stranded molecules packed into \omterm{def:g11:dna-replication:chromosome}{chromosomes},
copied before every division, and read whenever a \omterm{def:g10:molecules-of-life:proteins}{protein} is made. This chapter
builds the structure of \omterm{def:g11:dna-replication:helix}{DNA} and then the mechanism that copies it ---
\omterm{def:g11:dna-replication:semi}{semi-conservative replication} --- with the main \omterm{def:g10:membranes-enzymes:enzyme}{enzymes} named and placed.

\section{Nucleotides and the double helix}

\begin{definition}[Nucleotide]\label{def:g11:dna-replication:nucleotide}
A \emph{nucleotide}\index{nucleotide} is the monomeric unit of a \omterm{def:g10:molecules-of-life:nucleic}{nucleic acid}.
Each nucleotide has three parts: a five-carbon sugar, a phosphate group, and a
nitrogenous base. In \omterm{def:g11:dna-replication:helix}{DNA} the sugar is \emph{deoxyribose}\index{deoxyribose};
in \omterm{def:g10:molecules-of-life:nucleic}{RNA} it is ribose. \omterm{def:g11:dna-replication:helix}{DNA} uses four bases: adenine (A), thymine (T), cytosine (C)
and guanine (G).
\end{definition}

\begin{definition}[DNA double helix]\label{def:g11:dna-replication:helix}
\emph{Deoxyribonucleic acid}\index{DNA}\index{deoxyribonucleic acid} (DNA) is a
\omterm{def:g10:molecules-of-life:polymer}{polymer} of deoxyribonucleotides. Two strands wind into a right-handed
\emph{double helix}\index{double helix}: the sugar--phosphate backbones form the
outer rails, and the bases point inward, pairing by \omterm{def:g10:molecules-of-life:water}{hydrogen bonds}.
\end{definition}

\begin{proposition}[Complementary base pairing]\label{prop:g11:dna-replication:base-pairs}
In natural \omterm{def:g11:dna-replication:helix}{DNA}, adenine pairs only with thymine (two \omterm{def:g10:molecules-of-life:water}{hydrogen bonds}) and
cytosine only with guanine (three \omterm{def:g10:molecules-of-life:water}{hydrogen bonds}). The two strands are therefore
\emph{complementary}\index{complementary strands}: the \omterm{def:g10:molecules-of-life:proteins}{sequence} of one determines
the \omterm{def:g10:molecules-of-life:proteins}{sequence} of the other.
\end{proposition}
\admitted

\begin{remark}
Watson and Crick (1953) used Chargaff's \omterm{def:g10:scientific-biology:observation}{observation} that A equals T and C equals
G in cellular \omterm{def:g11:dna-replication:helix}{DNA}, together with X-ray patterns from Franklin and Wilkins, to
propose the \omterm{def:g11:dna-replication:helix}{double helix}. Complementarity is the structural key to both
\omterm{def:g11:dna-replication:semi}{replication} and repair.
\end{remark}

\begin{definition}[Antiparallel strands]\label{def:g11:dna-replication:antiparallel}
Each \omterm{def:g11:dna-replication:helix}{DNA} strand has direction: the carbons of \omterm{def:g11:dna-replication:nucleotide}{deoxyribose} are numbered, and the
chain runs from the \emph{$5'$ end}\index{5' end} (phosphate attached to carbon~5)
to the \emph{$3'$ end}\index{3' end} (free hydroxyl on carbon~3). The two strands
of a duplex run in opposite directions: they are
\emph{antiparallel}\index{antiparallel}.
\end{definition}

\begin{example}[Reading a short duplex]\label{ex:g11:dna-replication:read}
If one strand (written $5'$ to $3'$) is \texttt{5'-ATGCC-3'}, its complement
(also written $5'$ to $3'$) is \texttt{5'-GGCAT-3'}. Written as a duplex with
the top strand $5'\to 3'$ left to right:
\begin{center}
\texttt{5'-A T G C C-3'}\\
\texttt{3'-T A C G G-5'}
\end{center}
\end{example}

\section{Chromosomes: DNA in the cell}

\begin{definition}[Chromosome and chromatin]\label{def:g11:dna-replication:chromosome}
A \emph{chromosome}\index{chromosome} is one \omterm{def:g11:dna-replication:helix}{DNA} molecule plus the \omterm{def:g10:molecules-of-life:proteins}{proteins} that
package it. In \omterm{def:g10:the-cell:prok-euk}{eukaryotes}, \omterm{def:g11:dna-replication:helix}{DNA} wraps around histone \omterm{def:g10:molecules-of-life:proteins}{proteins} to form
\emph{chromatin}\index{chromatin}; further folding produces the condensed
chromosomes visible at mitosis.
\end{definition}

\begin{remark}
A typical human \omterm{def:g10:the-cell:cell}{cell} holds about \qty{2}{m} of \omterm{def:g11:dna-replication:helix}{DNA} in a nucleus a few
micrometres across. Packaging is not mere storage: which regions of \omterm{def:g11:dna-replication:chromosome}{chromatin}
are open or closed also controls which genes can be read
(\cref{ch:g11:gene-expression}).
\end{remark}

\begin{proposition}[Genome and ploidy sketch]\label{prop:g11:dna-replication:genome}
The complete set of genetic material of an \omterm{def:g10:scientific-biology:organism}{organism} is its
\emph{genome}\index{genome}. Human somatic \omterm{def:g10:the-cell:cell}{cells} are diploid: two sets of
$23$ \omterm{def:g11:dna-replication:chromosome}{chromosomes} (one maternal, one paternal). Before division each \omterm{def:g11:dna-replication:chromosome}{chromosome}
is duplicated so that the \omterm{def:g10:the-cell:cell}{cell} temporarily carries two identical sister chromatids
per \omterm{def:g11:dna-replication:chromosome}{chromosome} --- the subject of \cref{ch:g11:cell-cycle-mitosis}.
\end{proposition}
\admitted

\section{Semi-conservative replication}

\begin{definition}[Semi-conservative replication]\label{def:g11:dna-replication:semi}
\emph{Replication}\index{DNA replication}\index{replication} is the process that
produces two \omterm{def:g11:dna-replication:helix}{DNA} duplexes from one. It is
\emph{semi-conservative}\index{semi-conservative replication}: each daughter
duplex keeps one parental strand and gains one newly synthesized strand.
\end{definition}

\begin{proposition}[Why complementarity enables copying]\label{prop:g11:dna-replication:template}
Because bases pair specifically, each parental strand is a template: opposite
every A the new strand receives T, opposite every C it receives G, and so on.
The \omterm{def:g10:molecules-of-life:proteins}{sequence} information is preserved without the \omterm{def:g10:the-cell:cell}{cell} ``knowing'' the meaning
of the \omterm{def:g10:molecules-of-life:proteins}{sequence} --- chemistry does the matching.
\end{proposition}
\admitted

\section{The replication fork and enzymes}

\begin{definition}[Replication fork]\label{def:g11:dna-replication:fork}
\omterm{def:g11:dna-replication:semi}{Replication} begins at specific sites called
\emph{origins of replication}\index{origin of replication}. The two strands
separate locally, forming a Y-shaped
\emph{replication fork}\index{replication fork} that moves as synthesis
proceeds. Eukaryotic \omterm{def:g11:dna-replication:chromosome}{chromosomes} have many origins so that long \omterm{def:g11:dna-replication:helix}{DNA} can be
copied in a reasonable time.
\end{definition}

\begin{definition}[Main replication enzymes]\label{def:g11:dna-replication:enzymes}
Several \omterm{def:g10:molecules-of-life:proteins}{proteins} act at the fork:
\begin{itemize}
  \item \emph{Helicase}\index{helicase} unwinds the \omterm{def:g11:dna-replication:helix}{double helix}, separating
        the parental strands.
  \item \emph{Single-strand binding proteins}\index{single-strand binding
        protein} coat the opened strands and keep them from re-annealing.
  \item \emph{Primase}\index{primase} lays down a short \omterm{def:g10:molecules-of-life:nucleic}{RNA}
        \emph{primer}\index{primer} so that \omterm{def:g11:dna-replication:helix}{DNA} polymerase has a free $3'$ end
        to extend.
  \item \emph{DNA polymerase}\index{DNA polymerase} adds deoxyribonucleotides
        to the growing strand, always in the $5'\to 3'$ direction, using the
        parental strand as template. It also proofreads many mistakes.
  \item \emph{DNA ligase}\index{DNA ligase} seals nicks (joins \omterm{prop:g11:dna-replication:leading-lagging}{Okazaki fragments})
        by forming a phosphodiester bond.
\end{itemize}
\end{definition}

\begin{proposition}[Leading and lagging strands]\label{prop:g11:dna-replication:leading-lagging}
Because \omterm{def:g11:dna-replication:enzymes}{DNA polymerase} synthesizes only $5'\to 3'$ and the parental strands are
\omterm{def:g11:dna-replication:antiparallel}{antiparallel}, the two new strands behave differently at a moving fork:
\begin{itemize}
  \item the \emph{leading strand}\index{leading strand} is made continuously
        toward the fork;
  \item the \emph{lagging strand}\index{lagging strand} is made as short
        \emph{Okazaki fragments}\index{Okazaki fragment}, each begun with a
        \omterm{def:g11:dna-replication:enzymes}{primer} and later joined by ligase.
\end{itemize}
\end{proposition}
\admitted

\begin{method}[Tracking a short replication event]\label{met:g11:dna-replication:track}
\begin{enumerate}
  \item Identify the template strand and write it $3'\to 5'$ if the new strand
        will be written $5'\to 3'$ beneath it.
  \item Apply base-pairing rules: A$\leftrightarrow$T, G$\leftrightarrow$C
        (in \omterm{def:g11:dna-replication:helix}{DNA}; \omterm{def:g10:molecules-of-life:nucleic}{RNA} uses U instead of T --- next chapter).
  \item Mark \omterm{def:g11:dna-replication:enzymes}{primers} as \omterm{def:g10:molecules-of-life:nucleic}{RNA} segments that will later be replaced by \omterm{def:g11:dna-replication:helix}{DNA} and
        sealed by ligase.
\end{enumerate}
\end{method}

\begin{example}[Complement of a template]\label{ex:g11:dna-replication:complement}
Template (parental) segment, written $3'\to 5'$:
\texttt{3'-TACGGAT-5'}. The new \omterm{def:g11:dna-replication:helix}{DNA} strand made $5'\to 3'$ is
\texttt{5'-ATGCCTA-3'}.
\end{example}

\section{Fidelity and the need for accuracy}

\begin{proposition}[Error rate and proofreading]\label{prop:g11:dna-replication:fidelity}
\omterm{def:g11:dna-replication:enzymes}{DNA polymerases} insert the wrong \omterm{def:g11:dna-replication:nucleotide}{nucleotide} occasionally; proofreading
exonuclease activity removes many mismatches immediately, and separate repair
systems catch others. The residual error rate is low enough that most
daughter \omterm{def:g10:the-cell:cell}{cells} inherit a faithful \omterm{prop:g11:dna-replication:genome}{genome} --- yet high enough that mutations
accumulate and supply raw material for evolution
(\cref{ch:g11:evolution-mechanisms}).
\end{proposition}
\admitted

\begin{remark}[Why both strands must be copied]
If only one strand were used as a permanent archive, the other would drift
away chemically. \omterm{def:g11:dna-replication:semi}{Semi-conservative} copying of \emph{both} strands keeps two
complete, \omterm{prop:g11:dna-replication:base-pairs}{complementary} records after every division.
\end{remark}

\begin{problem}[A bacterial chromosome is copied]\label{pb:g11:dna-replication:1}
\emph{Escherichia coli} has a circular \omterm{def:g11:dna-replication:chromosome}{chromosome} of roughly
\num{4.6e6} base pairs and typically starts bidirectional \omterm{def:g11:dna-replication:semi}{replication} from a
single origin.
\begin{enumerate}
  \item Sketch a circle, mark an origin, and draw two \omterm{def:g11:dna-replication:fork}{replication forks}
        travelling in opposite directions until they meet.
  \item Why is bidirectional \omterm{def:g11:dna-replication:semi}{replication} faster than unidirectional \omterm{def:g11:dna-replication:semi}{replication}
        from the same origin?
  \item At each fork, label which new strand is leading and which is lagging
        for one parental strand orientation of your choice.
  \item \omterm{def:g11:dna-replication:enzymes}{DNA polymerase} synthesizes at the \omterm{def:g10:biodiversity-classification:hierarchy}{order} of \qty{1000}{\omterm{def:g10:molecules-of-life:nucleic}{nucleotides}/s}
        in bacteria. Estimate roughly how long one full round of \omterm{def:g11:dna-replication:semi}{replication}
        would take if a \emph{single} fork had to copy the entire
        \num{4.6e6}~bp \omterm{def:g11:dna-replication:chromosome}{chromosome} alone (one direction only). Give the answer
        in minutes (\omterm{def:g10:biodiversity-classification:hierarchy}{order} of magnitude is enough).
  \item With two forks working from the origin, how does your estimate change?
  \item Name three \omterm{def:g10:molecules-of-life:proteins}{proteins} required at each fork and one sentence on each
        role.
  \item After one complete round of \omterm{def:g11:dna-replication:semi}{semi-conservative replication}, how many
        of the two daughter circles contain a parental strand?
\end{enumerate}
\end{problem}
