\chapter{Defending the Body}\label{ch:g8:immune}

\omterm{def:g8:microbiology:pathogen}{Pathogens} meet \omterm{def:g8:immune:barriers}{barriers}, rapid general defences and slower targeted
\omterm{def:g7:nervous:stim}{responses}. This chapter surveys the immune system at middle-school depth
--- enough to understand cuts, vaccinations and allergies as ideas.

\section{First lines of defence}

\begin{definition}[Barriers]\label{def:g8:immune:barriers}
Physical and chemical \emph{barriers}\index{barrier (immune)} include
intact \omterm{def:g5:nerves:senses}{skin}, mucus and cilia in airways, \omterm{def:g3:teeth:path}{stomach} acid, tears and
beneficial \omterm{def:g4:microbes:def}{microbes} on surfaces. They stop many \omterm{def:g8:microbiology:pathogen}{pathogens} before invasion.
\end{definition}

\begin{example}[Cut skin]\label{ex:g8:immune:cut}
A cut breaches the barrier; clotting helps seal it while immune \omterm{def:g7:microscope:cell}{cells}
respond to any \omterm{def:g4:microbes:def}{microbes} that enter.
\end{example}

\section{Innate (general) responses}

\begin{definition}[Innate immunity]\label{def:g8:immune:innate}
\emph{Innate immunity}\index{innate immunity} is fast and non-specific:
phagocytic white \omterm{def:g7:digestion:blood}{blood} \omterm{def:g7:microscope:cell}{cells} engulf \omterm{def:g8:microbiology:pathogen}{pathogens}; inflammation brings \omterm{def:g7:digestion:blood}{blood}
flow and defences to damaged \omterm{def:g7:tissues:tissue}{tissue}; fever can hinder some \omterm{def:g4:microbes:def}{microbes}.
\end{definition}

\section{Adaptive (specific) responses}

\begin{definition}[Adaptive immunity]\label{def:g8:immune:adaptive}
\emph{Adaptive immunity}\index{adaptive immunity} targets specific
\omterm{def:g8:microbiology:pathogen}{pathogens}. \emph{Lymphocytes}\index{lymphocyte} include B \omterm{def:g7:microscope:cell}{cells} that can
produce \emph{antibodies}\index{antibody} and T \omterm{def:g7:microscope:cell}{cells} that help coordinate
\omterm{def:g7:nervous:stim}{responses} or kill infected \omterm{def:g7:microscope:cell}{cells}.
\end{definition}

\begin{definition}[Antibody and antigen]\label{def:g8:immune:ab}
An \emph{antigen}\index{antigen} is a molecule that the immune system can
recognise as foreign. An \emph{antibody}\index{antibody} is a Y-shaped
\omterm{def:g6:nutrition:groups}{protein} that binds a specific antigen, marking \omterm{def:g8:microbiology:pathogen}{pathogens} for destruction
or blocking them.
\end{definition}

\begin{definition}[Memory]\label{def:g8:immune:memory}
After an infection or vaccination, \emph{memory cells}\index{memory cell
(immune)} remain. A second meeting with the same \omterm{def:g8:microbiology:pathogen}{pathogen} triggers a
faster, stronger \omterm{def:g7:nervous:stim}{response} --- often preventing illness.
\end{definition}

\begin{definition}[Vaccine]\label{def:g8:immune:vaccine}
A \emph{vaccine}\index{vaccine} presents safe \omterm{def:g8:immune:ab}{antigen} material (or
instructions to make it) so the adaptive system builds memory without the
full disease.
\end{definition}

\begin{proposition}[Why vaccines work]\label{prop:g8:immune:vax}
Vaccination trains \omterm{def:g8:immune:adaptive}{adaptive immunity} ahead of time. If the real \omterm{def:g8:microbiology:pathogen}{pathogen}
arrives, memory \omterm{def:g7:nervous:stim}{responses} can neutralise it quickly.
\end{proposition}
\admitted
