\chapter{Variation génétique et maladies}\label{ch:g11:genes-and-disease}

Un couple en parfaite santé a un premier enfant qui, dès les premières
semaines, tousse, ne prend pas de poids, et dont la sueur est si salée
qu'un baiser sur le front en a le goût. Le diagnostic est la
mucoviscidose~: les deux parents, sans le savoir, portaient chacun une
copie non fonctionnelle du même \omterm{def:g10:universal-dna:gene}{gène}, et l'enfant a reçu les deux. Une
personne sur vingt-cinq d'ascendance européenne est un tel \omterm{def:g11:genes-and-disease:genetic}{porteur}~;
les parents auraient pu le savoir, avant la grossesse, par un test sur
une goutte de sang. Ce chapitre porte sur les maladies écrites dans les
\omterm{def:g10:universal-dna:gene}{allèles}~: comment elles se transmettent, comment le risque se calcule
sur un \omterm{def:g11:genes-and-disease:pedigree}{arbre généalogique}, comment un laboratoire lit le \omterm{def:g10:universal-dna:gene}{gène}, et
pourquoi la plupart des maladies fréquentes ne sont génétiques qu'en
partie.

\section{Les maladies d'un seul gène}

\begin{definition}[Maladie génétique, dominant et récessif]\label{def:g11:genes-and-disease:genetic}
Une \emph{maladie génétique}\index{maladie génétique} est une maladie
causée par un ou plusieurs \omterm{def:g10:universal-dna:gene}{allèles} mutés. Pour une maladie d'un seul
\omterm{def:g10:universal-dna:gene}{gène}, un \omterm{def:g10:universal-dna:gene}{allèle} est \emph{récessif}\index{récessif} si la maladie
n'apparaît que chez les personnes qui en portent deux copies --- une
personne qui n'en porte qu'une est un \emph{porteur}\index{porteur}
sain --- et \emph{dominant}\index{dominant} si une copie suffit. Par
convention, on note les \omterm{def:g10:universal-dna:gene}{allèles} d'un \omterm{def:g10:universal-dna:gene}{gène} par une lettre~: $A$ pour
l'\omterm{def:g10:universal-dna:gene}{allèle} ordinaire, $a$ pour un \omterm{def:g10:universal-dna:gene}{allèle} muté récessif, si bien qu'un
porteur est $Aa$ et une personne atteinte $aa$.
\end{definition}

\begin{example}[La mucoviscidose]\label{ex:g11:genes-and-disease:cf}
Le \omterm{def:g10:universal-dna:gene}{gène} CFTR code une \omterm{def:g11:gene-expression:protein}{protéine} qui laisse les ions chlorure traverser
la membrane des \omterm{def:g10:cells-common-unit:cell}{cellules} qui tapissent les voies respiratoires,
l'intestin et les glandes sudoripares. L'\omterm{def:g10:universal-dna:gene}{allèle} muté le plus fréquent
est privé de trois \omterm{def:g10:universal-dna:nucleotide}{nucléotides}, donc d'un \omterm{def:g10:chemistry-of-life:families}{acide aminé}~; la \omterm{def:g11:gene-expression:protein}{protéine} se
replie mal et est détruite. Sans elle, le mucus des voies respiratoires
est épais et collant, les infections s'installent dans les poumons, et
les canaux du pancréas s'obstruent. La maladie est récessive~: environ
un Européen sur 25 est \omterm{def:g11:genes-and-disease:genetic}{porteur} sain, et un nouveau-né sur 2500 est
atteint. La drépanocytose (\cref{ch:g11:enzymes-and-phenotype}) est de
même récessive~; la maladie de Huntington, une dégénérescence du
cerveau qui débute vers quarante ans, est dominante --- un seul \omterm{def:g10:universal-dna:gene}{allèle}
muté suffit, et toute personne atteinte avait un parent atteint.
\end{example}

\begin{proposition}[Transmission d'une maladie récessive]\label{prop:g11:genes-and-disease:recessive}
Chaque parent transmet à chaque enfant l'un de ses deux \omterm{def:g10:universal-dna:gene}{allèles}, tiré
au hasard. Deux \omterm{def:g11:genes-and-disease:genetic}{porteurs} $Aa \times Aa$ ont donc, pour chaque enfant,
une chance sur $1/4$ d'avoir un enfant atteint $aa$, $1/2$ d'avoir un
\omterm{def:g11:genes-and-disease:genetic}{porteur} $Aa$ et $1/4$ d'avoir un \omterm{def:g11:genes-and-disease:genetic}{non-porteur} $AA$. Un \omterm{def:g11:genes-and-disease:genetic}{porteur} et un
\omterm{def:g11:genes-and-disease:genetic}{non-porteur} $Aa \times AA$ n'ont aucun enfant atteint et une chance sur
deux d'avoir un \omterm{def:g11:genes-and-disease:genetic}{porteur}. Une personne atteinte et un \omterm{def:g11:genes-and-disease:genetic}{non-porteur}
$aa \times AA$ n'ont que des enfants \omterm{def:g11:genes-and-disease:genetic}{porteurs}.
\end{proposition}
\admitted

\section{Lire un arbre généalogique}

\begin{definition}[Arbre généalogique]\label{def:g11:genes-and-disease:pedigree}
Un \emph{arbre généalogique}\index{arbre généalogique} est un schéma
d'une famille sur plusieurs générations~: des carrés pour les hommes,
des cercles pour les femmes, un trait horizontal réunissant un couple,
un trait vertical descendant vers leurs enfants, des symboles pleins
pour les personnes atteintes. Les générations sont numérotées I, II,
III depuis le haut~; les individus sont numérotés depuis la gauche.
\end{definition}

\begin{method}[Analyser un arbre généalogique]\label{met:g11:genes-and-disease:analyse}
\begin{enumerate}
  \item \emph{\omterm{def:g11:genes-and-disease:genetic}{Dominant} ou \omterm{def:g11:genes-and-disease:genetic}{récessif}~?} Un enfant atteint né de deux
        parents sains signifie \omterm{def:g11:genes-and-disease:genetic}{récessif} (les parents sont \omterm{def:g11:genes-and-disease:genetic}{porteurs}).
        Une personne atteinte ayant un parent atteint à chaque
        génération suggère un \omterm{def:g10:universal-dna:gene}{allèle} \omterm{def:g11:genes-and-disease:genetic}{dominant}.
  \item \emph{Sur un \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} sexuel~?} Si seuls des hommes sont
        atteints et que les frères ou le père de leur mère le sont
        aussi, soupçonner le \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} X (voir plus bas). Sinon, le
        \omterm{def:g10:universal-dna:gene}{gène} est sur un \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} ordinaire.
  \item \emph{Attribuer les \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotypes}}~: dans une maladie récessive,
        les personnes atteintes sont $aa$~; leurs parents et leurs
        enfants portent au moins un $a$~; les frères et sœurs sains
        d'une personne atteinte sont $AA$ ou $Aa$.
  \item \emph{Calculer le risque} pour un enfant à venir en combinant
        les \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotypes} des parents dans un échiquier de croisement~;
        quand le \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotype} d'un parent est incertain, pondérer chaque
        possibilité par sa probabilité.
\end{enumerate}
\end{method}

\begin{example}[Les deux tiers]\label{ex:g11:genes-and-disease:twothirds}
Pourquoi le frère sain d'un enfant atteint est-il \omterm{def:g11:genes-and-disease:genetic}{porteur} avec une
probabilité de $2/3$ et non de $1/2$~? Les parents sont
$Aa \times Aa$~; un enfant est $AA$, $Aa$, $aA$ ou $aa$ avec des
chances égales. Savoir qu'il est sain écarte $aa$~: il reste trois cas,
dont deux sont \omterm{def:g11:genes-and-disease:genetic}{porteurs}. S'il épouse une femme de la population
générale (porteuse avec une probabilité de $1/25$), le risque d'un
premier enfant atteint vaut
$2/3 \times 1/25 \times 1/4 = 1/150$.
\end{example}

\section{Les gènes portés par le chromosome X}

\begin{proposition}[La transmission liée à l'X]\label{prop:g11:genes-and-disease:xlinked}
Les femmes portent deux \omterm{def:g10:universal-dna:information}{chromosomes} X, les hommes un X et un Y. Un
\omterm{def:g10:universal-dna:gene}{allèle} \omterm{def:g11:genes-and-disease:genetic}{récessif} porté par le \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} X est masqué chez une femme par
son second X, mais un homme, qui n'a pas de second X, l'exprime. Une
telle maladie touche donc surtout les hommes~; les filles d'un homme
atteint sont toutes porteuses~; une mère porteuse transmet l'\omterm{def:g10:universal-dna:gene}{allèle} à
la moitié de ses fils, qui sont atteints, et à la moitié de ses filles,
qui sont porteuses. L'hémophilie, où le sang ne coagule pas, en est
l'exemple classique~; le daltonisme rouge-vert (\cref{ch:g11:the-eye})
en est un autre.
\end{proposition}
\admitted

\section{Lire le gène lui-même}

\begin{proposition}[Les tests génétiques]\label{prop:g11:genes-and-disease:testing}
Les \omterm{def:g10:universal-dna:gene}{allèles} que porte une personne se lisent directement sur son \omterm{def:g10:universal-dna:information}{ADN},
extrait de quelques \omterm{def:g10:cells-common-unit:cell}{cellules} (sang, salive et, pour un fœtus, un
prélèvement de placenta ou du liquide qui l'entoure). La région du \omterm{def:g10:universal-dna:gene}{gène}
est copiée des millions de fois par une réaction qui imite la
réplication (\cref{ch:g11:dna-replication}) dans un tube à essai, puis
séquencée, ou bien coupée par des \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymes} qui reconnaissent des
séquences précises et triée par taille dans un gel~: un \omterm{def:g10:universal-dna:gene}{allèle} muté qui
a gagné, perdu ou modifié un site de coupure donne des fragments de
longueurs différentes, visibles sous forme de bandes. De tels tests
identifient les \omterm{def:g11:genes-and-disease:genetic}{porteurs} avant même qu'un enfant soit conçu,
diagnostiquent un fœtus et détectent la \omterm{def:g11:mutations:mutation}{mutation} chez un nouveau-né
avant l'apparition des symptômes.
\end{proposition}
\admitted

\begin{example}[Ce qu'un test peut dire et ne peut pas dire]\label{ex:g11:genes-and-disease:testsays}
Un test de dépistage des \omterm{def:g11:genes-and-disease:genetic}{porteurs} de la mucoviscidose recherche les
cinquante \omterm{def:g10:universal-dna:gene}{allèles} mutés les plus fréquents et trouve environ 90~\% des
\omterm{def:g11:genes-and-disease:genetic}{porteurs}~: un résultat négatif abaisse la probabilité d'être \omterm{def:g11:genes-and-disease:genetic}{porteur} de
$1/25$ à environ $1/250$, mais ne l'annule pas. Un diagnostic prénatal
sur le fœtus de deux \omterm{def:g11:genes-and-disease:genetic}{porteurs} connus est quasi certain, et met les
parents devant une décision que la biologie ne prend pas à leur place.
Un test de la maladie de Huntington dit à une personne de trente ans en
bonne santé si la maladie viendra~; un tiers de ceux qui y ont droit
choisissent de ne pas savoir.
\end{example}

\section{Maladies de plusieurs gènes et d'un mode de vie}

\begin{proposition}[Les maladies multifactorielles]\label{prop:g11:genes-and-disease:multifactorial}
La plupart des maladies fréquentes --- diabète de type 2, maladies du
cœur et des artères, la plupart des cancers, asthme --- sont
\emph{multifactorielles}\index{maladie multifactorielle}~: des dizaines
ou des centaines d'\omterm{def:g10:universal-dna:gene}{allèles} déplacent chacun un peu le risque, et
l'environnement (régime, activité, tabac, infections) le déplace
beaucoup. Une telle maladie se retrouve dans les familles sans suivre
les règles d'un \omterm{def:g10:universal-dna:gene}{gène} unique, et une prédisposition génétique est un
risque, non une sentence~: les mêmes \omterm{def:g10:universal-dna:gene}{allèles} donnent la maladie dans un
mode de vie et pas dans un autre.
\end{proposition}
\begin{proof}[Faits expérimentaux]
De vrais jumeaux partagent tous leurs \omterm{def:g10:universal-dna:gene}{allèles}~; quand l'un a un diabète
de type 2, l'autre l'a dans environ 70~\% des cas --- bien au-dessus des
8~\% de la population, donc les \omterm{def:g10:universal-dna:gene}{gènes} comptent --- mais pas 100~\%,
donc les \omterm{def:g10:universal-dna:gene}{gènes} ne décident pas. Des populations passées en une
génération d'un régime rural à un régime urbain ont vu la maladie
augmenter plusieurs fois avec les mêmes \omterm{def:g10:universal-dna:gene}{gènes}. Parmi les personnes qui
portent beaucoup d'\omterm{def:g10:universal-dna:gene}{allèles} à risque, celles qui restent actives et
minces développent la maladie à une fraction du rythme des autres.
\end{proof}

\begin{remark}[Ce que «~génétique~» veut dire, alors]\label{rem:g11:genes-and-disease:meaning}
La mucoviscidose est génétique au sens strict~: le \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotype} détermine
la maladie, quel que soit l'environnement, et l'environnement ne change
que la façon dont elle est vécue. Le diabète de type 2 est génétique en
un sens plus faible~: le \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotype} fixe une susceptibilité que
l'environnement réalise ou non. Entre les deux se placent la
phénylcétonurie, où un régime annule l'effet du \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotype}, et le \omterm{def:g11:genes-and-disease:genetic}{porteur}
sain de la drépanocytose, dont l'\omterm{def:g10:universal-dna:gene}{allèle} est une maladie ici et une
protection ailleurs. «~Est-ce génétique~?~» est rarement une question
à laquelle on répond par oui ou par non~; «~dans quelle mesure, et dans
quelles conditions~?~» est la bonne question.
\end{remark}

\section{Problème~: une famille, un gène}

\begin{problem}[{Problème du week-end --- une famille touchée par la
mucoviscidose suivie sur trois générations~: génotypes attribués,
risques calculés pour chaque enfant à venir, un test de laboratoire lu,
et l'arithmétique d'une population de porteurs}]
\label{pb:g11:genes-and-disease:1}
Léa et Tom, tous deux en bonne santé, ont trois enfants~: Max (sain),
Zoé (mucoviscidose) et Lou (sain). La sœur de Tom, Ana, est en bonne
santé et mariée à Karim, dont la famille n'a aucun antécédent de la
maladie~; ils attendent un enfant. Dans la population, une personne sur
25 est porteuse.

\medskip
\noindent\textbf{Partie I --- Les \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotypes}.}
\begin{enumerate}
  \item La maladie est-elle dominante ou récessive~? Justifier à partir
        de la famille.
  \item Donner les \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotypes} de Léa, Tom et Zoé.
  \item Quels sont les \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotypes} possibles de Max, avec quelles
        probabilités~?
  \item Quelle est la probabilité qu'Ana soit porteuse~? (Considérer
        ses parents.)
  \item Quelle est la probabilité que Karim soit \omterm{def:g11:genes-and-disease:genetic}{porteur}~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Les risques.}
\begin{enumerate}[resume]
  \item Calculer la probabilité que l'enfant d'Ana et de Karim soit
        atteint.
  \item Léa et Tom envisagent un quatrième enfant. Probabilité qu'il
        soit atteint~? Qu'il soit \omterm{def:g11:genes-and-disease:genetic}{porteur}~?
  \item Max, adulte, épouse une femme de la population générale.
        Probabilité d'un premier enfant atteint~?
  \item Zoé, adulte, épouse un homme de la population générale.
        Probabilité d'un enfant atteint~? D'un enfant \omterm{def:g11:genes-and-disease:genetic}{porteur}~?
  \item Lou épouse un cousin germain, enfant d'Ana et de Karim.
        Calculer la probabilité d'un enfant atteint. Pourquoi est-elle
        plus élevée que pour Max~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le test.}
L'\omterm{def:g10:universal-dna:gene}{allèle} muté porté dans cette famille ajoute un site de coupure~:
l'\omterm{def:g10:universal-dna:gene}{allèle} ordinaire donne un fragment de 900 paires de \omterm{def:g10:universal-dna:nucleotide}{bases}, l'\omterm{def:g10:universal-dna:gene}{allèle}
muté en donne un de 600 et un de 300.
\begin{enumerate}[resume]
  \item Dessiner, ou décrire, les bandes attendues pour Zoé, pour Tom
        et pour un \omterm{def:g11:genes-and-disease:genetic}{non-porteur}.
  \item Max est testé~: une seule bande à 900. Quel est désormais son
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotype}, et le nouveau risque pour son premier enfant
        (question 8)~?
  \item Ana est testée~: bandes à 900, 600 et 300. Recalculer le risque
        pour son enfant.
  \item Le test de Karim ne trouve aucun des cinquante \omterm{def:g10:universal-dna:gene}{allèles} mutés
        les plus fréquents, qui couvrent 90~\% des \omterm{def:g11:genes-and-disease:genetic}{porteurs}. Calculer
        la probabilité qui lui reste d'être \omterm{def:g11:genes-and-disease:genetic}{porteur} et recalculer le
        risque.
  \item Un test prénatal sur le fœtus d'Ana et de Karim montre des
        bandes à 900, 600 et 300. Quels sont le \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotype} et le
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{phénotype} du fœtus~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La population.}
\begin{enumerate}[resume]
  \item Si une personne sur 25 est porteuse, quelle est la probabilité
        que les deux membres d'un couple pris au hasard soient
        \omterm{def:g11:genes-and-disease:genetic}{porteurs}, et donc la fréquence des nouveau-nés atteints~?
  \item Comparer au un sur 2500 observé. Commenter.
  \item Presque toutes les personnes atteintes mouraient autrefois dans
        l'enfance. Expliquer pourquoi l'\omterm{def:g10:universal-dna:gene}{allèle} muté est néanmoins resté
        à une fréquence d'une copie sur 50, en utilisant le nombre de
        copies portées par les \omterm{def:g11:genes-and-disease:genetic}{porteurs} sains face à celles portées par
        les personnes atteintes.
  \item Les traitements permettent aujourd'hui à la plupart des
        personnes atteintes d'atteindre l'âge adulte et d'avoir des
        enfants. Prévoir l'effet sur la fréquence de l'\omterm{def:g10:universal-dna:gene}{allèle} au fil
        des générations à venir, et son ampleur.
  \item Énoncer le résultat~: le un sur quatre de deux \omterm{def:g11:genes-and-disease:genetic}{porteurs}, le
        deux sur trois d'un frère ou d'une sœur sain, et ce que le gel
        a ajouté à l'arithmétique pour cette famille.
\end{enumerate}
\end{problem}
