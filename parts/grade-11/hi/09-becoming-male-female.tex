\chapter{नर या मादा बनना}\label{ch:g11:becoming-male-female}

निषेचन के सात हफ़्ते बाद किसी मानव भ्रूण में जननग्रंथियों का एक जोड़ा होता
है जो न वृषण है न अंडाशय, और साथ-साथ वाहिनियों के दो पूरे जत्थे --- एक जो
नर तंत्र बन सकता है, और एक जो मादा तंत्र। जो भ्रूण लड़का बनेगा उसे लड़की
बनने वाले से अलग बताने लायक़ कुछ भी दिखता नहीं। अगले हफ़्तों में
वाहिनियों का एक जत्था बढ़ता है और दूसरा सूख जाता है, जननग्रंथियाँ कोई एक
रूप ले लेती हैं, और चौथे महीने तक शरीर-रचना तय हो जाती है। यह फ़ैसला अकेला
एक \omterm{def:g10:universal-dna:gene}{जीन} करता है, और उसे दो हॉर्मोन अंजाम देते हैं। यह अध्याय उसी क्रम का
पीछा करता है, निषेचन पर मिले \omterm{def:g10:universal-dna:information}{गुणसूत्रों} से लेकर \omterm{prop:g11:becoming-male-female:puberty}{यौवनारंभ} के बदलावों तक।

\section{लिंग गुणसूत्र}

\begin{definition}[लिंग गुणसूत्र]\label{def:g11:becoming-male-female:chromosomes}
मानव \omterm{def:g10:universal-dna:information}{गुणसूत्रों} के 23 जोड़ों में से 22 दोनों लिंगों में एक जैसे हैं; और
23वाँ जोड़ा \emph{लिंग गुणसूत्रों}\index{लिंग गुणसूत्र} का बना है: मादा में
दो X \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} (XX), और नर में एक X तथा एक Y (XY)। Y छोटा है, चंद ही \omterm{def:g10:universal-dna:gene}{जीन}
ढोता है, और केवल पिता से पुत्र तक जाता है; हर अंडाणु एक X ढोता है, और कोई
शुक्राणु या तो X ढोता है या Y, इसलिए बच्चे का गुणसूत्रीय लिंग पिता का
शुक्राणु ही तय करता है, और वह भी बराबर संभावनाओं से।
\end{definition}

\begin{proposition}[अविभेदित अवस्था]\label{prop:g11:becoming-male-female:undifferentiated}
सातवें हफ़्ते तक दोनों गुणसूत्रीय लिंगों के भ्रूण अपनी जनन-रचना में एक
जैसे होते हैं: अविभेदित जननग्रंथियों का एक जोड़ा, और हर एक के बग़ल में दो
वाहिनियाँ --- \emph{वुल्फ़ियन वाहिनी}, जो नर तंत्र (अधिवृषण, शुक्रवाहिका)
की पूर्ववर्ती है, और \emph{म्यूलेरियन वाहिनी}, जो मादा तंत्र (अंडवाहिनी,
गर्भाशय) की पूर्ववर्ती है। लैंगिक परिवर्धन इसी साझा योजना का दो में से एक
रूप में बदल जाना है; और उसे शुरू जननग्रंथि का विभेदीकरण करता है।
\end{proposition}
\admitted

\section{वह जीन जो फ़ैसला करता है}

\begin{proposition}[SRY और जननग्रंथि]\label{prop:g11:becoming-male-female:sry}
Y \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} एक \omterm{def:g10:universal-dna:gene}{जीन} ढोता है, \emph{SRY}\index{SRY जीन}, जो सातवें हफ़्ते में
अविभेदित जननग्रंथि में अभिव्यक्त होता है। उसका \omterm{def:g11:gene-expression:protein}{प्रोटीन} उन \omterm{def:g10:universal-dna:gene}{जीनों} को चालू कर
देता है जो उस जननग्रंथि को \emph{वृषण} में बदल देते हैं। उसके अभाव में ---
यानी किसी XX भ्रूण में --- वह जननग्रंथि कुछ हफ़्ते बाद \emph{अंडाशय} बन
जाती है। जननग्रंथि का फ़ैसला \omterm{prop:g11:becoming-male-female:sry}{SRY} की मौजूदगी करती है, X \omterm{def:g10:universal-dna:information}{गुणसूत्रों} की संख्या
नहीं।
\end{proposition}
\begin{proof}[साक्ष्य]
जिन विरले XY व्यक्तियों का \omterm{prop:g11:becoming-male-female:sry}{SRY जीन} कोई \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} ढोता है उनमें अंडाशय और
मादा शरीर विकसित होते हैं; और जो विरले XX व्यक्ति किसी X पर अंतरित Y का
\omterm{prop:g11:becoming-male-female:sry}{SRY} वाला टुकड़ा ढोते हैं उनमें वृषण और नर शरीर। चूहों में: जिस XX भ्रूण में
चूहे का \emph{Sry} \omterm{def:g10:universal-dna:gene}{जीन} डाल दिया गया हो वह नर की तरह विकसित होता है। यानी Y
का अकेला एक \omterm{def:g10:universal-dna:gene}{जीन} वृषण बनाने के लिए पर्याप्त भी है और ज़रूरी भी।
\end{proof}

\begin{proposition}[गर्भस्थ वृषण के हॉर्मोन]\label{prop:g11:becoming-male-female:hormones}
बन जाने के बाद वृषण दो हॉर्मोन स्रावित करता है।
\emph{टेस्टोस्टेरोन}\index{टेस्टोस्टेरोन} वुल्फ़ियन वाहिनियों को बनाए रखता
है और उन्हें नर के भीतरी तंत्र में विकसित करता है, तथा बाहरी जननांगों को
नर रूप में गढ़ता है। \emph{प्रति-म्यूलेरियन हॉर्मोन} (AMH) म्यूलेरियन
वाहिनियों को सिकोड़ देता है। इस अवस्था पर अंडाशय इनमें से कोई नहीं स्रावित
करता: इसलिए \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} के बिना वुल्फ़ियन वाहिनियाँ सिकुड़ जाती हैं, और
AMH के बिना म्यूलेरियन वाहिनियाँ अंडवाहिनियाँ, गर्भाशय तथा योनि का ऊपरी
हिस्सा बन जाती हैं। इस तरह मादा तंत्र वही रूप है जो वे वाहिनियाँ वृषण के
हॉर्मोनों के अभाव में ले लेती हैं।
\end{proposition}
\begin{proof}[साक्ष्य]
जोस्त (1947) ने ख़रगोश के गर्भस्थ शिशुओं की जननग्रंथियाँ विभेदीकरण से पहले ही
गर्भाशय में हटा दीं: हर गर्भस्थ शिशु में, चाहे XX हो या XY, मादा तंत्र
विकसित हुआ। जिन शिशुओं की जननग्रंथियाँ हटाकर उनके बग़ल में वृषण प्रत्यारोपित
किया गया उनमें नर तंत्र बना; और जिनमें अकेले \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} का एक क्रिस्टल
बिठाया गया उनमें वुल्फ़ियन वाहिनियाँ तो बची रहीं पर म्यूलेरियन वाहिनियाँ
हटीं नहीं और उनके बग़ल में टिकी रहीं। इसलिए वृषण दो उत्पाद बनाता है ---
एक जो नर वाहिनियाँ बनाता है (\omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन}) और एक, जिसे बाद में AMH के रूप
में पहचाना गया, जो मादा वाली हटा देता है --- और मादा तंत्र बनने के लिए
अंडाशय की ज़रूरत ही नहीं।
\end{proof}

\begin{example}[क्रम से पूरी शृंखला]\label{ex:g11:becoming-male-female:cascade}
Y ढोते शुक्राणु से निषेचन; हफ़्ता 7, \omterm{prop:g11:becoming-male-female:sry}{SRY} अभिव्यक्त, जननग्रंथि वृषण बनी;
हफ़्ते 8 से 12, \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} और AMH: वुल्फ़ियन वाहिनियाँ बचीं,
म्यूलेरियन वाहिनियाँ गईं, बाहरी जननांग नर रूप में गढ़े गए; और फिर
\omterm{prop:g11:becoming-male-female:puberty}{यौवनारंभ} तक सन्नाटा। X ढोते शुक्राणु से निषेचन: कोई \omterm{prop:g11:becoming-male-female:sry}{SRY} नहीं; हफ़्ते 10 से
12, जननग्रंथि अंडाशय बनी; वृषण के कोई हॉर्मोन नहीं: म्यूलेरियन वाहिनियाँ
विकसित हुईं, वुल्फ़ियन सिकुड़ीं, और बाहरी जननांगों ने मादा रूप लिया। हर
क़दम अपने पिछले पर टिका है, और किसी भी क़दम की चूक उसके आगे का सब कुछ बदल
देती है।
\end{example}

\section{यौवनारंभ}

\begin{proposition}[यौवनारंभ]\label{prop:g11:becoming-male-female:puberty}
जन्म से पहले बन चुके जनन अंग बचपन भर निष्क्रिय रहते हैं।
\emph{यौवनारंभ}\index{यौवनारंभ} पर --- यानी क़रीब ग्यारह से तेरह साल की
उम्र में, और लड़कियों में उससे पहले --- मस्तिष्क का एक केंद्र धड़कनों में
ऐसा हॉर्मोन छोड़ने लगता है जो पीयूष ग्रंथि से दो हॉर्मोन, LH और FSH, रक्त
में स्रावित कराता है। ये जननग्रंथियों पर काम करते हैं: वृषण शुक्राणु बनाने
और वयस्क मात्रा में \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} स्रावित करने लगता है; और अंडाशय अपने चक्र
शुरू कर देता है तथा एस्ट्रोजन स्रावित करता है। फिर लैंगिक हॉर्मोन
\emph{गौण लैंगिक लक्षण} पैदा करते हैं: जनन अंगों की वृद्धि, शरीर के बाल,
आवाज़, स्तन, पेशी तथा चर्बी का बँटवारा, वृद्धि-उछाल और फिर हड्डियों की
वृद्धि पट्टिकाओं का बंद हो जाना। \omterm{prop:g11:becoming-male-female:puberty}{यौवनारंभ} तब पूरा होता है जब वह व्यक्ति
जनन कर सके।
\end{proposition}
\admitted

\section{लिंग, उसके कई अर्थों में}

\begin{proposition}[स्तरों को अलग करना]\label{prop:g11:becoming-male-female:levels}
``नर'' और ``मादा'' कई ऐसे स्तरों पर लागू होते हैं जो आमतौर पर आपस में मेल
खाते हैं पर अलग-अलग हैं:
\begin{itemize}
  \item \emph{गुणसूत्रीय} लिंग: XX या XY, जो निषेचन पर तय हो जाता है;
  \item \emph{जननग्रंथीय} लिंग: अंडाशय या वृषण, जिसका फ़ैसला \omterm{prop:g11:becoming-male-female:sry}{SRY} करता है;
  \item \emph{शारीरिक} लिंग: वे तंत्र और बाहरी जननांग, जिन्हें गर्भस्थ
        जननग्रंथि के हॉर्मोन गढ़ते हैं, तथा वे गौण लक्षण जो \omterm{prop:g11:becoming-male-female:puberty}{यौवनारंभ} पर
        गढ़े जाते हैं;
  \item \emph{लैंगिक पहचान}, यानी किसी व्यक्ति का यह बोध कि वह पुरुष है,
        स्त्री है या इनमें से कोई नहीं, और \emph{यौन अभिविन्यास}, यानी वह
        किसकी ओर आकर्षित होता है --- और ये दोनों उस व्यक्ति के बारे में
        ऐसे तथ्य हैं जो पिछले स्तरों से निकाले नहीं जाते और जो चुने नहीं
        जाते।
\end{itemize}
इस अध्याय का जीवविज्ञान पहले तीन का वर्णन करता है; वह आख़िरी को न तय करता
है, न उस पर कोई फ़ैसला सुनाता है।
\end{proposition}
\admitted

\begin{example}[जब स्तर आपस में मेल न खाएँ]\label{ex:g11:becoming-male-female:disagree}
कुछ हज़ार में लगभग एक व्यक्ति ऐसे मेल के साथ जन्म लेता है जो वह शृंखला
आमतौर पर नहीं बनाती: जिस XY व्यक्ति की \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} का ग्राही
न हो उसमें मादा शरीर विकसित होता है, ऐसे वृषणों के साथ जो कभी नीचे उतरे ही
नहीं, और बिना गर्भाशय के; जिस XX व्यक्ति में \omterm{prop:g11:becoming-male-female:sry}{SRY} का वृषण-निर्माता
स्थानांतरण हो उसमें नर शरीर; और जिस XY व्यक्ति का \omterm{prop:g11:becoming-male-female:sry}{SRY} काम न करता हो वह मादा
के रूप में विकसित होता है। ये विविधताएँ साधारण मामले से भी बेहतर दिखाती हैं
कि हर स्तर किसी एक ख़ास क़दम का --- किसी \omterm{def:g10:universal-dna:gene}{जीन}, किसी हॉर्मोन, किसी ग्राही का
--- नतीजा है, और वे स्तर अलग-अलग तथ्य हैं।
\end{example}

\begin{method}[किसी मामले पर तर्क करना]\label{met:g11:becoming-male-female:case}
\begin{enumerate}
  \item \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र}: क्या \omterm{prop:g11:becoming-male-female:sry}{SRY} मौजूद है और काम कर रहा है?
  \item जननग्रंथि: \omterm{prop:g11:becoming-male-female:sry}{SRY} काम करे तो वृषण, वरना अंडाशय।
  \item गर्भस्थ जननग्रंथि के हॉर्मोन: \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} और AMH, या इनमें से
        कोई नहीं; और क्या ऊतक जवाब दे सकते हैं (ग्राही)?
  \item वाहिनियाँ: दोनों हॉर्मोन काम करें तो वुल्फ़ियन बची और म्यूलेरियन
        हटी; कोई भी काम न करे तो म्यूलेरियन विकसित; और \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} बिना
        AMH के काम करे तो दोनों।
  \item \omterm{prop:g11:becoming-male-female:puberty}{यौवनारंभ}: जननग्रंथि जो लैंगिक हॉर्मोन स्रावित करे वही गौण लक्षण तय
        करता है --- और वह भी ग्राहियों के ज़रिए।
\end{enumerate}
\end{method}

\begin{remark}[एक जीन, और उसके बाद सब कुछ]\label{rem:g11:becoming-male-female:onegene}
दोनों शरीर-रचनाओं का पूरा फ़र्क़ लौटकर इसी पर टिकता है कि सातवें हफ़्ते में
किसी एक \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} पर एक \omterm{def:g10:universal-dna:gene}{जीन} मौजूद है या नहीं। उसके बाद जो आता है वह साधारण
कोशिका-जीवविज्ञान की एक शृंखला है --- \omterm{def:g10:universal-dna:gene}{जीन} अभिव्यक्ति, हॉर्मोन, ग्राही ---
यानी वही जिसका वर्णन इस वर्ष के हर अध्याय ने किया है। वह शृंखला जिस योजना
से शुरू होती है वह दोनों लिंगों में साझा है, और मादा रूप वही है जो वह
शृंखला तब बनाती है जब वृषण दख़ल न दे।
\end{remark}

\section{समस्या: जोस्त के ख़रगोश}

\begin{problem}[{सप्ताहांत समस्या --- वृषण के दोनों हॉर्मोन खोजने वाले
प्रयोग क़दम-दर-क़दम दोबारा खड़े किए गए, फिर तीन मानव मामलों पर और
यौवनारंभ के समय पर लागू किए गए}]
\label{pb:g11:becoming-male-female:1}
ख़रगोश में जननग्रंथियाँ गर्भावधि के लगभग 19वें दिन विभेदित होती हैं और
वाहिनियाँ 22वें दिन के आसपास। जोस्त ने गर्भाशय में पड़े शिशुओं पर 19वें
दिन शल्यक्रिया की और उन्हें जन्म पर (30वें दिन) देखा।

\medskip
\noindent\textbf{भाग I --- प्रयोग।}
\begin{enumerate}
  \item दोनों गुणसूत्रीय लिंगों के शिशु, जिनकी जननग्रंथियाँ 19वें दिन हटा
        दी गईं: सब अंडवाहिनियों और गर्भाशय के साथ, तथा नर वाहिनियों के
        बिना जन्म लेते हैं। इससे अंडाशय की भूमिका के बारे में, और वृषण की
        भूमिका के बारे में, क्या पता चलता है?
  \item 22वें दिन से पहले शल्यक्रिया करना क्यों ज़रूरी था?
  \item 19वें दिन जननग्रंथियाँ हटाकर पेट में वृषण प्रत्यारोपित किए गए
        शिशु: नर वाहिनियाँ, कोई मादा वाहिनी नहीं। वह प्रत्यारोपण क्या देता
        है, और वह वाहिनियों तक पहुँचता कैसे है?
  \item 19वें दिन जननग्रंथियाँ हटाकर \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} का क्रिस्टल बिठाए गए
        शिशु: नर वाहिनियाँ मौजूद, मादा वाहिनियाँ भी मौजूद। \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन}
        क्या करता है, और क्या नहीं करता?
  \item इससे निकालिए कि वृषण को कोई दूसरा पदार्थ भी बनाना चाहिए, और उसका
        असर बताइए। उसका होना बाद में कैसे पक्का किया जा सकता था?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- एक जननग्रंथि हटाई गई।}
\begin{enumerate}[resume]
  \item किसी XY शिशु से 19वें दिन एक ही वृषण हटाया जाता है। यह जानते हुए
        कि हॉर्मोन ज़्यादातर अपनी ही ओर काम करते हैं, शरीर के दोनों ओर की
        वाहिनियों का अनुमान लगाइए।
  \item किसी XX शिशु के एक अंडाशय के बग़ल में वृषण प्रत्यारोपित किया जाता
        है। दोनों ओर का अनुमान लगाइए।
  \item किसी XX शिशु के एक अंडाशय के बग़ल में \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} का क्रिस्टल
        बिठाया जाता है। दोनों ओर का अनुमान लगाइए।
  \item ये एक-तरफ़ा प्रयोग भाग I के निष्कर्ष को और मज़बूत क्यों करते हैं?
  \item समझाइए कि वृषण प्रत्यारोपण वाला XX शिशु फिर भी वयस्क होकर बंध्य
        क्यों रहता।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- तीन मानव मामले।}
हर एक पर \cref{met:g11:becoming-male-female:case} लगाइए।
\begin{enumerate}[resume]
  \item कोई XX व्यक्ति अपने एक X \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} पर \omterm{prop:g11:becoming-male-female:sry}{SRY} वाला Y का टुकड़ा ढोता
        है।
  \item कोई XY व्यक्ति ऐसा \omterm{prop:g11:becoming-male-female:sry}{SRY} ढोता है जो काम नहीं करता।
  \item सामान्य वृषण वाले किसी XY व्यक्ति की \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} AMH का जवाब नहीं दे
        पातीं।
  \item इन तीनों में से किस मामले में वह व्यक्ति उपजाऊ होने की संभावना
        रखता है? कारण बताइए।
  \item ये तीनों मामले गुणसूत्रीय लिंग और शारीरिक लिंग के रिश्ते के बारे
        में क्या दिखाते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- अंकों में \omterm{prop:g11:becoming-male-female:puberty}{यौवनारंभ}।}
हॉर्मोन वाले चित्र से यह मानिए कि वृद्धि-उछाल हॉर्मोन के अपने वयस्क स्तर
का 20\% छूने के साल भर बाद शुरू होता है और हड्डियाँ तब बढ़ना बंद करती हैं
जब वह 100\% पर पहुँच जाए।
\begin{enumerate}[resume]
  \item आँकिए कि लड़कियों में और लड़कों में वृद्धि-उछाल किस उम्र पर शुरू
        होता है, और वृद्धि किस उम्र पर रुक जाती है।
  \item उछाल के आरंभ से उसके अंत तक हर लिंग को कितने साल की वृद्धि मिलती
        है?
  \item कोई लड़का उछाल के दौरान साल भर में \qty{8}{cm} बढ़ता है और कोई
        लड़की \qty{7}{cm}। उछाल के दौरान हर एक की पाई हुई लंबाई आँकिए, और
        वयस्क लंबाई के औसत फ़र्क़ (लगभग \qty{13}{cm}) पर टिप्पणी कीजिए।
  \item जिस लड़की के एस्ट्रोजन 7 पर चढ़ने लगें वह जल्दी तेज़ बढ़ेगी और
        जल्दी रुक जाएगी। समझाइए कि डॉक्टर बहुत जल्दी आए \omterm{prop:g11:becoming-male-female:puberty}{यौवनारंभ} को दवाओं
        से कभी-कभी क्यों टालते हैं।
  \item परिणाम लिखिए: जोस्त के ख़रगोशों ने कौन-से दो हॉर्मोन उजागर किए,
        हर एक का असर क्या है, और वह अकेला \omterm{def:g10:universal-dna:gene}{जीन} कौन-सा है जो उन्हें बनाने
        वाला अंग चालू करता है।
\end{enumerate}
\end{problem}
