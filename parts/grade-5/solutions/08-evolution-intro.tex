\section*{Chapter \ref{ch:g5:evolution} --- Living Things Change Over Time}

\begin{solution}{exo:g5:evolution:1}
\omterm{def:g5:evolution:def}{Evolution} is change in the \omterm{def:g4:traits:inherited}{inherited traits} of \omterm{def:g4:ecosystems:levels}{populations} over many
generations.
\end{solution}

\begin{solution}{exo:g5:evolution:2}
Differences among individuals of the same species.
\end{solution}

\begin{solution}{exo:g5:evolution:3}
A preserved remain or trace of a once-living \omterm{def:g4:ecosystems:levels}{organism}.
\end{solution}

\begin{solution}{exo:g5:evolution:4}
No. \omterm{def:g5:evolution:def}{Evolution} in this \omterm{def:g1:senses:sense}{sense} is \omterm{def:g4:ecosystems:levels}{population} change across generations, not
one \omterm{def:g1:plants-animals:animal}{animal} transforming into a new species in its lifetime.
\end{solution}

\begin{solution}{exo:g5:evolution:5}
Individuals with \omterm{def:g4:traits:inherited}{inherited traits} that fit the \omterm{def:g4:traits:env}{environment} better tend
to survive and \omterm{def:g2:what-alive:signs}{reproduce} more, so those \omterm{def:g4:traits:trait}{traits} become more common over
time.
\end{solution}
