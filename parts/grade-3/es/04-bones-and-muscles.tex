\chapter{Huesos y músculos}\label{ch:g3:bones-and-muscles}

Salta y cae: algo dentro de ti acaba de funcionar como una máquina
perfecta. Unas varillas rígidas mantuvieron tu forma, unas bisagras
dejaron que tus \omterm{def:g1:my-body:parts}{piernas} se plegaran y saltaran, y cientos de motorcitos
tiraron justo cuando hacía falta. Las varillas son tus \omterm{def:g3:bones-and-muscles:skeleton}{huesos}, las
bisagras tus \omterm{prop:g3:bones-and-muscles:joints}{articulaciones} y los motores tus \omterm{def:g3:bones-and-muscles:muscle}{músculos} --- el equipo
del cuerpo para todos los movimientos que hará en su vida.

\section{El esqueleto}

\begin{definition}[Esqueleto y huesos]\label{def:g3:bones-and-muscles:skeleton}
El \emph{esqueleto}\index{esqueleto} es el armazón de unos $200$
\emph{huesos}\index{hueso} que hay dentro del cuerpo. Sus regiones
principales:
\begin{itemize}
  \item el \emph{cráneo}\index{cráneo}, un casco macizo de hueso
        alrededor del cerebro;
  \item la \emph{columna vertebral}\index{columna vertebral}, una
        columna flexible de huesecillos apilados desde el cuello hasta
        las caderas, que mantiene el tronco erguido;
  \item las \emph{costillas}\index{costilla}, huesos curvos que forman
        una jaula alrededor del corazón y de los pulmones;
  \item los \emph{huesos de las \omterm{def:g1:my-body:parts}{extremidades}}: huesos largos en \omterm{def:g1:my-body:parts}{brazos}
        y \omterm{def:g1:my-body:parts}{piernas} --- el fémur es el más largo del cuerpo.
\end{itemize}
\end{definition}

\begin{example}[Nota tu armazón]\label{ex:g3:bones-and-muscles:feel}
No hace falta radiografía: aprieta con cuidado y el \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto}
responde. Los nudillos; el filo de la espinilla; la cadena de bultitos
que baja por el medio de la espalda --- la columna, un huesecillo
detrás de otro; las \omterm{def:g3:bones-and-muscles:skeleton}{costillas}, que puedes contar con los dedos. El
armazón está justo debajo de la \omterm{def:g1:the-five-senses:sense}{piel}.
\end{example}

\begin{remark}[Duro por fuera, vivo por dentro]\label{rem:g3:bones-and-muscles:alive}
Un \omterm{def:g3:bones-and-muscles:skeleton}{hueso} no es material muerto como una tiza: está vivo, lo alimenta
la sangre, crece mientras tú creces --- y es capaz de repararse solo.
Un \omterm{def:g1:my-body:parts}{brazo} roto se suelda dentro de la escayola porque el propio \omterm{def:g3:bones-and-muscles:skeleton}{hueso}
reconstruye la rotura poco a poco. Las tizas no hacen eso nunca.
\end{remark}

\section{Las articulaciones, donde el armazón se dobla}

\begin{proposition}[Varillas rígidas, sitios que se doblan]\label{prop:g3:bones-and-muscles:joints}
Los \omterm{def:g3:bones-and-muscles:skeleton}{huesos} no se doblan nunca --- un \omterm{def:g3:bones-and-muscles:skeleton}{hueso} doblado es un \omterm{def:g3:bones-and-muscles:skeleton}{hueso} roto.
Todo lo que el cuerpo dobla lo dobla en las
\emph{articulaciones}\index{articulación}, donde se juntan dos \omterm{def:g3:bones-and-muscles:skeleton}{huesos}:
\omterm{def:g1:my-body:joint}{hombro}, \omterm{def:g1:my-body:joint}{codo}, \omterm{def:g1:my-body:joint}{muñeca}, cadera, \omterm{def:g1:my-body:joint}{rodilla}, \omterm{def:g1:my-body:joint}{tobillo} y las muchas
\omterm{prop:g3:bones-and-muscles:joints}{articulaciones} pequeñas de la columna, cuyos doblecitos suman una
espalda flexible.
\end{proposition}
\admitted

\begin{example}[El truco de la columna]\label{ex:g3:bones-and-muscles:spinetrick}
Ningún \omterm{def:g3:bones-and-muscles:skeleton}{hueso} de tu espalda se puede enroscar --- y, aun así, puedes
hacerte casi una bola hacia delante. La columna es un montón de más de
$20$ huesecillos, cada uno girando unos pocos grados sobre sus vecinos:
muchos doblecitos que hacen uno grande. Por eso también hay que doblar
las \omterm{def:g1:my-body:joint}{rodillas}, y no la espalda, para levantar algo pesado --- las
\omterm{def:g1:my-body:joint}{rodillas} están hechas para plegarse mucho, y la columna para poco.
\end{example}

\section{Los músculos, los que mueven}

\begin{definition}[Músculo]\label{def:g3:bones-and-muscles:muscle}
Un \emph{músculo}\index{músculo} es un motor carnoso sujeto a los
\omterm{def:g3:bones-and-muscles:skeleton}{huesos}. Un músculo puede \emph{contraerse}\index{contraerse} ---
acortarse y engordar --- y al contraerse \emph{tira} del \omterm{def:g3:bones-and-muscles:skeleton}{hueso} al que
está sujeto. Los músculos solo tiran; ningún músculo puede empujar.
\end{definition}

\begin{proposition}[Trabajar por parejas]\label{prop:g3:bones-and-muscles:pairs}
Como los \omterm{def:g3:bones-and-muscles:muscle}{músculos} solo tiran, mover una \omterm{def:g1:my-body:joint}{articulación} en los dos
sentidos exige una \emph{pareja}: un \omterm{def:g3:bones-and-muscles:muscle}{músculo} delante del \omterm{def:g1:my-body:parts}{brazo} dobla
el \omterm{def:g1:my-body:joint}{codo} y su compañero de detrás lo estira; cada uno se relaja
mientras el otro tira. Casi todas las \omterm{prop:g3:bones-and-muscles:joints}{articulaciones} del cuerpo las
mueven parejas así.
\end{proposition}
\admitted

\begin{example}[Nota la pareja trabajando]\label{ex:g3:bones-and-muscles:pair}
Agárrate el \omterm{def:g1:my-body:parts}{brazo} con la otra mano y dobla el \omterm{def:g1:my-body:joint}{codo} despacio: el
\omterm{def:g3:bones-and-muscles:muscle}{músculo} de delante se hincha y se pone duro --- se está contrayendo.
Ahora estira el \omterm{def:g1:my-body:parts}{brazo}: el de delante se ablanda y, detrás del \omterm{def:g1:my-body:parts}{brazo},
su compañero se tensa. Acabas de notar cómo ocurre la
\cref{prop:g3:bones-and-muscles:pairs}.
\end{example}

\begin{method}[Cuidar los huesos y los músculos]\label{met:g3:bones-and-muscles:care}
\begin{enumerate}
  \item muévete cada día --- los \omterm{def:g3:bones-and-muscles:skeleton}{huesos} se hacen más fuertes con el
        uso, y los \omterm{def:g3:bones-and-muscles:muscle}{músculos} también;
  \item come los alimentos que construyen: lácteos para los \omterm{def:g3:bones-and-muscles:skeleton}{huesos} y
        \omterm{prop:g1:healthy-habits:needs}{comidas variadas} (\cref{ch:g2:teeth-and-eating-well});
  \item calienta antes del ejercicio fuerte, para que los \omterm{def:g3:bones-and-muscles:muscle}{músculos} se
        estiren en vez de romperse;
  \item levanta las cosas pesadas con las \omterm{def:g1:my-body:joint}{rodillas} dobladas y la
        espalda recta;
  \item casco y protecciones donde sean probables las caídas: los
        \omterm{def:g3:bones-and-muscles:skeleton}{huesos} se reparan, pero despacio.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:1}
Nombra las cuatro regiones del \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto} de la
\cref{def:g3:bones-and-muscles:skeleton} y di qué protege o sostiene
cada una.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:2}
¿Cuántos \omterm{def:g3:bones-and-muscles:skeleton}{huesos} tiene más o menos el \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto}? ¿Cuál es el más largo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:3}
¿Se puede doblar un \omterm{def:g3:bones-and-muscles:skeleton}{hueso}? ¿Dónde se dobla todo lo que dobla el cuerpo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:4}
La columna se puede enroscar aunque ninguno de sus \omterm{def:g3:bones-and-muscles:skeleton}{huesos} pueda.
Explica su truco.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:5}
¿Qué hace un \omterm{def:g3:bones-and-muscles:muscle}{músculo} cuando se contrae? ¿Qué es lo que ningún \omterm{def:g3:bones-and-muscles:muscle}{músculo}
puede hacer nunca?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:6}
¿Por qué hacen falta dos \omterm{def:g3:bones-and-muscles:muscle}{músculos} para doblar y estirar el \omterm{def:g1:my-body:joint}{codo}?
Cuenta qué hace cada uno en cada momento.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:7}
¿Cómo se suelda un \omterm{def:g1:my-body:parts}{brazo} roto dentro de una escayola? ¿Qué demuestra
eso sobre el \omterm{def:g3:bones-and-muscles:skeleton}{hueso}, según la \cref{rem:g3:bones-and-muscles:alive}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:8}
Pruébalo: hazte en tu propio \omterm{def:g1:my-body:parts}{brazo} la prueba del
\cref{ex:g3:bones-and-muscles:pair}. Describe qué hace la parte de
delante del \omterm{def:g1:my-body:parts}{brazo} cuando doblas y cuando estiras.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:9}
¿Por qué dicen las personas mayores ``levanta con las \omterm{def:g1:my-body:joint}{rodillas}, no con
la espalda''? Responde con el \cref{ex:g3:bones-and-muscles:spinetrick}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:10}
A un títere de hilos se le puede hacer saludar: cada hilo solo puede
\emph{tirar} de su \omterm{def:g1:my-body:parts}{brazo} en un sentido. ¿En qué se parece esto
exactamente a la situación de los \omterm{def:g3:bones-and-muscles:muscle}{músculos} y cuál es la respuesta del
titiritero? Compárala con la \cref{prop:g3:bones-and-muscles:pairs}.
\end{exercise}
