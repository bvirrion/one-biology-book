\chapter{الكربوهيدرات}\label{ch:b1:carbohydrates}

حبة بطاطا \omterm{def:b1:flowering-plant-organization:organs}{وورقة} ورق مصنوعتان من الجزيء نفسه، الغلوكوز، منظومًا في
سلاسل. إحداهما وجبة والأخرى ليست كذلك، لأن السلاسل موصولة برابطتين
مختلفتين: فرابطة $\alpha$ في \omterm{def:b1:carbohydrates:polysaccharide}{النشا} تلوي السلسلة لولبًا تفتحه
إنزيماتنا، ورابطة $\beta$ في \omterm{def:b1:carbohydrates:polysaccharide}{السليولوز} تمدها شرائط ترتص في ألياف لا
يستطيع ثديي أن يهضمها. والكبد يدخر مئة غرام من الغلوكوز غليكوجينًا،
والشجرة تسند نفسها \omterm{def:b1:carbohydrates:polysaccharide}{بالسليولوز}، والخنفساء تلبس \omterm{def:b1:carbohydrates:polysaccharide}{الكيتين}، وكل \omterm{def:b1:cell-unit-of-life:cell}{خلية} تعرض
سكريات على سطحها هويةً لها. يصف هذا الفصل السكريات البسيطة، والروابط
التي تصلها، والسكريات المتعددة التي تبنيها، والجزيئات المزينة
بالسكر في سطح \omterm{def:b1:cell-unit-of-life:cell}{الخلية} وجدار النبات.

\section{السكريات الأحادية}

\begin{definition}[السكر الأحادي]\label{def:b1:carbohydrates:monosaccharide}
\emph{الكربوهيدرات}\index{كربوهيدرات} جزيئات تركيبها
$(\mathrm{CH_2O})_n$ ومشتقاتها. و\emph{السكر الأحادي}\index{سكر أحادي}
(السكر البسيط) سلسلة من ثلاث إلى سبع ذرات كربون، إحداها تحمل زمرة
كربونيل وسائرها هيدروكسيلات: فهو \emph{ألدوز}\index{ألدوز} حين يكون
الكربونيل طرفيًا (ألدهيد: الغلوكوز، والغالاكتوز، والريبوز)،
و\emph{كيتوز}\index{كيتوز} حين يكون داخليًا (كيتون: الفركتوز).
وبحسب الطول: سكريات ثلاثية ($n = 3$: الغليسرألدهيد)، وخماسية
($n = 5$: الريبوز، والريبوز منقوص الأكسجين)، وسداسية ($n = 6$:
الغلوكوز، والفركتوز، والغالاكتوز). وكل كربون يحمل أربع زمر مختلفة
مركز \emph{لاتناظر}، فتقوم كل صيغة مقام عدة \emph{مصاوغات
فراغية}: فللغلوكوز أربعة كربونات لامتناظرة وستة عشر مصاوغًا، لا
تستعمل الأحياء منها إلا واحدًا، هو D-غلوكوز. و\emph{الإبيمرات} تختلف
عند كربون واحد فقط (الغلوكوز والغالاكتوز عند C4).
\end{definition}

\begin{proposition}[الصور الحلقية والأنومرات]\label{prop:b1:carbohydrates:rings}
\index{أنومر}\index{إسقاط هاوورث}
في الماء يكون السكر الخماسي أو السداسي حلقيًا كله تقريبًا: إذ يتفاعل
كربون الكربونيل مع هيدروكسيل من الجزيء نفسه (C5 في الغلوكوز) فيغلق
حلقة \emph{بيرانوزية} سداسية أو حلقة \emph{فورانوزية} خماسية
(الفركتوز، والريبوز). ويصير كربون الكربونيل مركز لاتناظر جديدًا، هو
\emph{الكربون الأنوميري}، ويمكن أن يقع هيدروكسيله تحت مستوي الحلقة
(الأنومر $\alpha$) أو فوقه (الأنومر $\beta$) في رسم هاوورث؛ والصورتان
تتحولان إحداهما إلى الأخرى عبر السلسلة المفتوحة في دقائق. وفي
المحلول يكون الغلوكوز \qty{36}{\%} $\alpha$ و\qty{64}{\%} $\beta$
وأقل من \qty{0.1}{\%} سلسلة مفتوحة. والكربون الأنوميري الحر هو ما
يجعل السكر \emph{مرجعًا}: إذ يستطيع أن ينفتح ويرجع كاشفًا نحاسيًا أو
فضيًا.
\end{proposition}

\begin{method}[قراءة إسقاط هاوورث]\label{met:b1:carbohydrates:haworth}
\begin{enumerate}
  \item ابحث عن أكسجين الحلقة؛ فالكربون الأنوميري (C1 في \omterm{def:b1:carbohydrates:monosaccharide}{الألدوز}،
        وC2 في \omterm{def:b1:carbohydrates:monosaccharide}{الكيتوز}) هو كربون الحلقة الواقع يمينه، وهو يحمل
        هيدروكسيلًا وأكسجين الحلقة معًا.
  \item والكربون الحامل لزمرة $\mathrm{CH_2OH}$ خارج الحلقة (C5 في
        الغلوكوز) يقع يسار أكسجين الحلقة؛ وزمرة $\mathrm{CH_2OH}$
        عنده تشير إلى أعلى في سكر من نوع D.
  \item الهيدروكسيل الأنوميري إلى أسفل: $\alpha$؛ وإلى أعلى (في جهة
        $\mathrm{CH_2OH}$ نفسها): $\beta$.
  \item قارن الهيدروكسيلات الأخرى بهيدروكسيلات الغلوكوز (أسفل، ثم
        أعلى، ثم أسفل عند C2 وC3 وC4): فاختلاف واحد يعيّن إبيمرًا
        (الغالاكتوز: C4 إلى أعلى).
\end{enumerate}
\end{method}

\begin{example}[سكريات ليست $(\mathrm{CH_2O})_n$]\label{ex:b1:carbohydrates:derivatives}
الريبوز منقوص الأكسجين يفتقر إلى الهيدروكسيل عند C2
(\cref{ch:b1:nucleic-acids})؛ والغلوكوزامين وأسيتيل الغلوكوزامين
يحملان أمينًا بدل هيدروكسيل C2 (\omterm{def:b1:carbohydrates:polysaccharide}{الكيتين}، \omterm{def:b1:carbohydrates:polysaccharide}{والببتيدوغليكان})؛ \omterm{def:b1:water-small-molecules:ph}{وحمض}
الغلوكورونيك له كربوكسيل عند C6 (سكريات \omterm{def:b1:body-plans-tissues:connective}{المادة خارج الخلوية})؛ وفوسفات
السكريات (غلوكوز-6-فوسفات، وريبوز-5-فوسفات) هي الصور التي تدخل بها
السكريات الاستقلاب؛ وكحولات السكر (الغليسرول، والسوربيتول) لا كربونيل
لها. واللب واحد: سلسلة كربونية صغيرة كثيرة الهيدروكسيلات يحبها الماء.
\end{example}

\section{الرابطة الغليكوزيدية}

\begin{definition}[الرابطة الغليكوزيدية والسكر الثنائي]\label{def:b1:carbohydrates:glycosidic}
\emph{الرابطة الغليكوزيدية}\index{رابطة غليكوزيدية} تصل الهيدروكسيل
الأنوميري لسكر بهيدروكسيل سكر آخر (أو كحول، أو أمين، أو \omterm{def:b1:water-small-molecules:ph}{أساس})
\omterm{prop:b1:water-small-molecules:families}{بالتكاثف}، فتثبّت الكربون الأنوميري في تشكيله $\alpha$ أو $\beta$؛
وتُسمى بالتشكيل وبالكربونين الموصولين: $\alpha(1{\to}4)$،
و$\beta(1{\to}4)$، و$\alpha(1{\to}6)$. وسكران موصولان هكذا يكونان
\emph{سكرًا ثنائيًا}\index{سكر ثنائي}: \emph{المالتوز} (غلوكوز
$\alpha(1{\to}4)$ غلوكوز، من هضم \omterm{def:b1:carbohydrates:polysaccharide}{النشا})، و\emph{اللاكتوز} (غالاكتوز
$\beta(1{\to}4)$ غلوكوز، وهو سكر اللبن)، و\emph{السكروز} (غلوكوز
$\alpha(1{\to}2)\beta$ فركتوز، وهو سكر عصارة النبات وسكر المطبخ). وفي
السكروز يكون الكربونان الأنوميريان كلاهما مرتبطين: فهو غير مرجع ولا
ينفتح، ولهذا تنقله النباتات (\cref{ch:b1:plant-transport}).
\end{definition}

\begin{method}[الكشف عن سكر مرجع]\label{met:b1:carbohydrates:reducing}
\begin{enumerate}
  \item سخّن المحلول مع كاشف نحاسي قلوي (بندكت أو فهلنغ).
  \item ينفتح الكربون الأنوميري الحر إلى ألدهيد، فيرجع
        $\mathrm{Cu^{2+}}$ الأزرق إلى راسب أحمر آجري من
        $\mathrm{Cu_2O}$؛ وشدة اللون تتناسب مع كمية السكر.
  \item الغلوكوز، والفركتوز، والمالتوز، واللاكتوز: إيجابية.
        والسكروز: سلبي --- إلا إذا حُلمه أولًا \omterm{def:b1:water-small-molecules:ph}{بحمض} أو بإنزيم
        الإنفرتاز، فيتفاعل عندئذ غلوكوزه وفركتوزه.
  \item \omterm{def:b1:carbohydrates:polysaccharide}{والنشا} لا يتفاعل (فطرفه المرجع الوحيد واحد من آلاف)؛ ويُكشف
        عنه باليود، الذي يدخل لولب الأميلوز فيصير أزرق داكنًا.
\end{enumerate}
\end{method}

\section{السكريات المتعددة}

\begin{definition}[السكر المتعدد]\label{def:b1:carbohydrates:polysaccharide}
\emph{السكر المتعدد}\index{سكر متعدد} سلسلة من مئات إلى عشرات آلاف
السكريات الأحادية موصولة بروابط غليكوزيدية، خطية أو متفرعة. وسكريات
الادخار المتعددة: \emph{النشا}\index{نشا} في النبات ---
و\emph{الأميلوز} غير متفرع، غلوكوز $\alpha(1{\to}4)$، بنسبة
\qty{20}{\%}، و\emph{الأميلوبكتين} $\alpha(1{\to}4)$ بتفرعات
$\alpha(1{\to}6)$ كل \qtyrange{24}{30}{} بقية، بنسبة \qty{80}{\%} ---
مرصوصة في حبيبات؛ و\emph{الغليكوجين}\index{غليكوجين} في الحيوان
والفطر، وهو كالأميلوبكتين لكنه متفرع كل \qtyrange{8}{12}{} بقية، في
حبيبات عصارية في الكبد والعضلات. وسكريات البنية المتعددة:
\emph{السليولوز}\index{سليولوز}، غلوكوز $\beta(1{\to}4)$ غير متفرع،
من \qtyrange{2000}{15000}{} بقية، في جدر النبات؛
و\emph{الكيتين}\index{كيتين}، أسيتيل غلوكوزامين $\beta(1{\to}4)$، في
جدر الفطر وأديمات المفصليات؛ و\emph{الببتيدوغليكان}\index{ببتيدوغليكان}،
وهو أسيتيل غلوكوزامين \omterm{def:b1:water-small-molecules:ph}{وحمض} أسيتيل الموراميك متناوبان تربطهما ببتيدات
قصيرة، وهو الشبكة ذات الجزيء الواحد التي هي جدار البكتيريا.
\end{definition}

\begin{proposition}[الرابطة تحسم الشكل، والشكل يحسم الوظيفة]\label{prop:b1:carbohydrates:alphabeta}
في سلسلة $\alpha(1{\to}4)$ يكون كل غلوكوز موجهًا في جهة سابقه فتلتف
السلسلة في لولب من ست بقايا في الدورة، وهو ملف مفتوح مماه تدخله
الإنزيمات بسهولة: أي صورة \emph{المخزن}. أما في سلسلة
$\beta(1{\to}4)$ فكل غلوكوز مقلوب $180^\circ$ بالنسبة إلى جاره
والسلسلة شريط مستقيم مسطح؛ وتصطف الشرائط جنبًا إلى جنب وترتبط
\omterm{def:b1:water-small-molecules:water}{بروابط هيدروجينية} في \emph{ليفيفات دقيقة} من ثلاثين إلى مئة سلسلة،
بلورية، غير منحلة، مقاومتها للشد كمقاومة الفولاذ لواحدة الكتلة: أي
صورة \emph{الليف}. والإنزيمات التي تحلمه روابط $\alpha$ (الأميلازات)
لا تستطيع مس روابط $\beta$؛ و\emph{السليولازات} موجودة عند البكتيريا
والفطريات وقليل من الحيوانات، والثدييات التي تعيش على العشب تفعل ذلك
بمتعايشات (\cref{ch:b1:digestion-absorption}).
\end{proposition}

\begin{example}[لماذا كان الغليكوجين متفرعًا]\label{ex:b1:carbohydrates:branching}
يُهدَم \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} من أطرافه غير المرجعة، غلوكوزًا واحدًا في كل مرة،
بفسفوريلاز \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين}. وسلسلة غير متفرعة من \num{50000} غلوكوز كان
لها طرف واحد كهذا وكانت لتطلق غلوكوزًا واحدًا لكل دورة إنزيمية؛ أما
جسيمة \omterm{def:b1:carbohydrates:polysaccharide}{غليكوجين} بالقياس نفسه، متفرعة كل اثنتي عشرة بقية، فلها نحو
\num{2000} طرف وتطلق ألفي غلوكوز دفعة واحدة. والتفرع يبقي الجسيمة
أيضًا مضغوطة منحلة. ومئة غرام \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} في الكبد يمكن حشدها بمعدل
عشرة غرامات في الساعة؛ أما الغلوكوز نفسه مخزونًا سلسلة واحدة طويلة
فكان فك لفه ليستغرق سنوات.
\end{example}

\begin{example}[لماذا لا يُدخَّر الغلوكوز نفسه]\label{ex:b1:carbohydrates:osmotic}
مئة غرام من الغلوكوز (\qty{0.55}{mol}) منحلة في \qty{1}{L} من ماء
\omterm{def:b1:cell-unit-of-life:cell}{خلايا} الكبد كانت لتضيف \qty{0.55}{osmol/L} إلى عصارة عند
\qty{0.3}{osmol/L}: فتنتفخ \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} إلى ثلاثة أمثال حجمها وتنفجر. أما
غليكوجينًا فالغلوكوز نفسه $\num{7e18}$ جسيمة، أي \qty{1e-5}{mol/L}،
غير مرئية تناضحيًا. فالبوليمر طريقة في ادخار جزيئات كثيرة جدًا في
جزيء واحد.
\end{example}

\section{المقترنات السكرية وجدار النبات}

\begin{definition}[البروتينات السكرية والبروتيوغليكانات والغليكوليبيدات]\label{def:b1:carbohydrates:glycoconjugate}
تُوصل السكريات تساهميًا بالبروتينات \omterm{def:b1:lipids:fattyacid}{والدهون} فتكوّن
\emph{مقترنات سكرية}\index{مقترن سكري}. و\emph{البروتين
السكري}\index{بروتين سكري} يحمل سلاسل قصيرة متفرعة (نحو اثني عشر
سكرًا) على بعض بقايا الأسبارجين أو السيرين أو الثريونين فيه، تُضاف في
\omterm{def:b1:eukaryotic-cell:endomembrane}{الشبكة الإندوبلازمية} وغولجي (\cref{ch:b1:eukaryotic-cell}): وكل
بروتين مفرز أو غشائي يكاد يكون واحدًا منها.
و\emph{البروتيوغليكان}\index{بروتيوغليكان} لب بروتيني يحمل سلاسل
طويلة غير متفرعة من سكريات ثنائية حمضية متكررة، هي
\emph{الغليكوزامينوغليكانات}\index{غليكوزامينوغليكان} (الهيالورونان،
وكبريتات الكوندرويتين، والهيبارين)، وهي تربط كميات هائلة من الماء
وتمنح الغضروف والجسم الزجاجي \omterm{def:b1:body-plans-tissues:connective}{والمادة خارج الخلوية} مرونتها.
و\emph{الغليكوليبيدات}\index{غليكوليبيد} تحمل سكريات على ذيل دهني في
الصفيحة الخارجية للغشاء البلازمي. وهي مع سكريات البروتينات السكرية
تكوّن \emph{الغلاف السكري}\index{غلاف سكري}، وهو الكساء السكري الذي
تُعرف به \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}.
\end{definition}

\begin{example}[زمر الدم]\label{ex:b1:carbohydrates:abo}
زمر الدم ABO ثلاث صور لسلسلة سكرية واحدة على غليكوليبيدات سطح الكرية
الحمراء وبروتيناتها السكرية: فسلسلة O تنتهي بالفوكوز؛ وA تضيف إليها
أسيتيل غالاكتوزامين، وB تضيف غالاكتوزًا، بصورتين من إنزيم واحد؛
\omterm{def:b1:cell-unit-of-life:cell}{وخلايا} AB تحمل الاثنتين. فبقية سكرية واحدة تكفي \omterm{def:b1:mammal-organization:organ}{جهاز} المناعة ليميز
الذات من الغريب، ولينجح نقل الدم أو يقتل.
\end{example}

\begin{proposition}[جدار الخلية النباتية]\label{prop:b1:carbohydrates:wall}
\index{جدار خلوي}
الجدار الأولي \omterm{def:b1:cell-unit-of-life:cell}{للخلية} النباتية مركّب مقوّى بالألياف: ليفيفات \omterm{def:b1:carbohydrates:polysaccharide}{سليولوز}
دقيقة (ربع الكتلة، وهي مقاومة الشد) موضوعة في طبقات، تربطها معًا
\emph{الهيميسليولوزات} (سكريات متعددة متفرعة ترتبط هيدروجينيًا
بالليفيفات وتجسر بينها) ومغروسة في هلام من \emph{البكتينات} (سكريات
متعددة حمضية تحبس الماء والكالسيوم وتلحم \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} بعضها ببعض في
الصفيحة الوسطى)، مع بضعة في المئة من البروتين. وهو رقيق
(\qtyrange{0.1}{1}{\micro m})، نفوذ للماء وللمذابات الصغيرة، وقوي بما
يكفي لتحمل \qty{1}{MPa} من \omterm{def:b1:eukaryotic-cell:plantcell}{الامتلاء}. \omterm{def:b1:cell-unit-of-life:cell}{والخلايا} التي تكف عن النمو قد
تضيف \emph{جدارًا ثانويًا} سميكًا \omterm{def:b1:carbohydrates:polysaccharide}{بسليولوز} أكثر و\emph{لغنين}، وهو
بوليمر عطري يكتم الماء ويصلّبه: \omterm{def:b1:flowering-plant-organization:tissues}{والخشب} جدر ثانوية.
\end{proposition}

\section{السكريات وقودًا وكربونًا}

\begin{proposition}[مكانة الغلوكوز]\label{prop:b1:carbohydrates:fuel}
تعطي \omterm{def:b1:carbohydrates:monosaccharide}{الكربوهيدرات} \qty{17}{kJ/g} بالأكسدة وهي الوقود الذي تستطيع كل
\omterm{def:b1:cell-unit-of-life:cell}{خلية} أن تستعمله، بالأكسجين أو دونه
(\cref{ch:b1:respiration-fermentation})؛ والغلوكوز سكر الدم
(\qty{5}{mmol/L})، والسكروز سكر عصارة النبات، \omterm{def:b1:carbohydrates:polysaccharide}{والنشا} \omterm{def:b1:carbohydrates:polysaccharide}{والغليكوجين}
مخزناهما. والسكريات أيضًا هي الكربون الذي تصنع منه \omterm{def:b1:cell-unit-of-life:cell}{الخلية} كل ما
عداها: فخماسيات النوكليوتيدات، وغليسرول \omterm{def:b1:lipids:fattyacid}{الدهون}، والهياكل الكربونية
للحموض الأمينية كلها تغادر سبل السكريات
(\cref{ch:b1:biosyntheses-integration}). فالتركيب الضوئي يصنع السكر
أولًا؛ وما عداه يتبع.
\end{proposition}

\begin{example}[سكر يوم]\label{ex:b1:carbohydrates:day}
يأكل الإنسان نحو \qty{300}{g} من \omterm{def:b1:carbohydrates:monosaccharide}{الكربوهيدرات} في اليوم، جلها \omterm{def:b1:carbohydrates:polysaccharide}{نشا}
وسكروز؛ ويحلمهها المعي إلى غلوكوز وفركتوز وغالاكتوز، ويحول الكبد
الأخيرين إلى غلوكوز، ويدور الغلوكوز عند \qty{5}{g} في الدم كله ---
أي إمداد عشرين دقيقة للدماغ، الذي يأخذ \qty{120}{g} في اليوم، ويُملأ
باستمرار من \omterm{def:b1:carbohydrates:polysaccharide}{غليكوجين} الكبد. وعدسة العين والكريات الحمراء ولب الكلية
تعيش على الغلوكوز وحده؛ ولا شيء آخر في الغذاء يحل محله، وحين ينفد
يصنعه الكبد من البروتين.
\end{example}

\begin{exercise}[$\star$]\label{exo:b1:carbohydrates:4}
من شكل هاوورث، اذكر مواضع الهيدروكسيلات (أعلى أو أسفل) على الكربونات

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:5}
فسّر، انطلاقًا من هندسة الرابطتين $\alpha$ و$\beta$، لماذا يكوّن
الأميلوز لولبًا \omterm{def:b1:carbohydrates:polysaccharide}{والسليولوز} شريطًا، ولماذا ينجح اختبار اليود على

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:6}
ثلاثة أنابيب فيها غلوكوز وسكروز \omterm{def:b1:carbohydrates:polysaccharide}{ونشا}. صمّم تتابعًا من الاختبارات
(بندكت، واليود، والحلمهة الحمضية) يعيّن كلًا منها، مع ذكر النتيجة
المنتظرة من كل اختبار.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:7}
جسيمة \omterm{def:b1:carbohydrates:polysaccharide}{غليكوجين} تحمل \num{50000} بقية غلوكوز متفرعة كل 12. قدّر عدد
الأطراف غير المرجعة وقارنها بجزيء أميلوز بالقياس نفسه. وإذا كان
الفسفوريلاز يطلق غلوكوزًا واحدًا في الثانية من كل طرف، فكم يستغرق كل
منهما لإطلاق \qty{10}{\%} من غلوكوزه؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:8}
لليفيفات \omterm{def:b1:carbohydrates:polysaccharide}{السليولوز} الدقيقة مقاومة شد تناهز \qty{1}{GPa}. \omterm{def:b1:cell-unit-of-life:cell}{وخلية} قطرها
\qty{50}{\micro m} وجدارها \qty{0.5}{\micro m} سماكةً تحمل امتلاءً
قدره \qty{0.8}{MPa}. احسب إجهاد الجدار ($\sigma = Pr/2t$ من أجل كرة)
وقارنه بالمقاومة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:9}
عدم تحمل اللاكتوز: البالغون الذين يفتقرون إلى لاكتاز المعي يصيبهم
مغص وإسهال بعد اللبن. فسّر، مستعملًا الرابطة التي يكسرها اللاكتاز،
ما الذي يحل باللاكتوز غير المهضوم في القولون (البكتيريا، \omterm{def:b1:membranes-transport:osmosis}{والتناضح}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:carbohydrates:10}
بقرة تأكل \qty{10}{kg} من المادة الجافة للعشب في اليوم، \qty{40}{\%}
منها \omterm{def:b1:carbohydrates:polysaccharide}{سليولوز}. ولا سليولاز لها. فسّر كيف تحصل مع ذلك على طاقة من
\omterm{def:b1:carbohydrates:polysaccharide}{السليولوز}، ولماذا يجب أن تجري العملية قبل المعي الدقيق، وقدّر الطاقة
المعنية إذا خُمّر \qty{60}{\%} من \omterm{def:b1:carbohydrates:polysaccharide}{السليولوز} وأعطى \qty{12}{kJ} من
النواتج القابلة للاستعمال لكل غرام.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:carbohydrates:11}
مولدا الضد A وB في زمر الدم يختلفان بسكر واحد تضيفه صورتان من ناقلة؛
والزمرة O لا تحمل أيًا منهما. فسّر لماذا يستطيع صاحب الزمرة O أن
يتبرع بكرياته الحمراء لأي أحد ولا يتلقى إلا من O، وما الذي يجب أن
تحويه بلازما صاحب الزمرة A.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:carbohydrates:12}
«الغلوكوز الوقود الشامل، \omterm{def:b1:carbohydrates:polysaccharide}{والسليولوز} مادة البناء الشاملة، وهما
الجزيء نفسه.» ناقش في فقرة: ما الذي تصيبه الجملة، وما الذي تغيره
رابطة واحدة، وما الذي تقوله عن العلاقة بين الكيمياء والبيولوجيا.
\end{exercise}

\section{مسألة: شجرة الغليكوجين}

\begin{problem}[{مسألة نهاية الأسبوع --- جسيمة غليكوجين تُعدّ فرعًا
فرعًا، ومخزن الكبد يُوزن ويُوقَّت إطلاقه، والثمن التناضحي الذي
تتجنبه البلمرة، وصولًا إلى معدل إطلاق الكبد للغلوكوز}]
\label{pb:b1:carbohydrates:1}
جسيمة \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} شجرة من السلاسل: سلسلة داخلية من \qty{13}{} بقية
غلوكوز تحمل فرعين، كل منهما من \qty{13}{} بقية ويحمل بدوره فرعين،
وهكذا على \qty{12}{} طبقة؛ وسلاسل الطبقة الخارجية غير متفرعة. وكتلة
كل بقية غلوكوز في البوليمر \qty{162}{g/mol}. وكبد إنسان مستريح كتلته
\qty{1.5}{kg} يحمل \qty{100}{g} من \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} في \qty{1.0}{L} من ماء
\omterm{def:b1:cell-unit-of-life:cell}{الخلايا}، ويطلق الغلوكوز إلى الدم بمعدل \qty{10}{g/h} بين الوجبات.
وفسفوريلاز \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} ينزع غلوكوزًا واحدًا من طرف غير مرجع في كل
دورة، بمعدل \qty{20}{} دورة في الثانية لكل جزيء إنزيم.

\medskip
\noindent\textbf{الجزء الأول --- عدّ الشجرة.}
\begin{enumerate}
  \item كم سلسلة تحوي الطبقة $t$ ($t = 1$ للسلسلة الداخلية)؟ أعطِ
        العدد من أجل الطبقات 1 و2 و3 و12.
  \item احسب العدد الكلي للسلاسل في الجسيمة.
  \item احسب العدد الكلي لبقايا الغلوكوز.
  \item احسب كتلة الجسيمة بالدالتون وبالغرام.
  \item أي كسر من السلاسل كلها في الطبقة الخارجية؟ وكم طرفًا غير
        مرجع تقدم الجسيمة؟
  \item جزيء أميلوز بالعدد نفسه من البقايا كم طرفًا غير مرجع له؟
  \item إذا أضافت كل طبقة \qty{1.9}{nm} إلى نصف القطر، فاحسب قطر
        الجسيمة وقارنه بريبوزوم (\qty{25}{nm}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- مخزن الكبد.}
\begin{enumerate}[resume]
  \item احسب عدد الجسيمات في الكبد.
  \item احسب الغلوكوز الذي يمثله المخزن بالمول وبالغرام من الغلوكوز
        الحر (\qty{180}{g/mol}).
  \item يربط \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} \qty{3}{g} من الماء لكل غرام. فأي كتلة من
        الكبد \omterm{def:b1:carbohydrates:polysaccharide}{غليكوجين} مع مائه؟
  \item احسب معدل الإطلاق البالغ \qty{10}{g/h} بجزيئات الغلوكوز في
        الثانية.
  \item كم ساعة يدوم المخزن عند ذلك المعدل؟ وأي \omterm{def:b1:mammal-organization:organ}{عضو} تخدم حاجته هذا
        الإطلاق أساسًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الأطراف والإنزيمات.}
\begin{enumerate}[resume]
  \item احسب العدد الكلي للأطراف غير المرجعة في الكبد.
  \item لو هوجم كل طرف دفعة واحدة بمعدل \qty{20}{} غلوكوزًا في
        الثانية، فما معدل الإطلاق، بالغرام في الثانية؟ قارنه بالمعدل
        الفعلي.
  \item كم جزيء فسفوريلاز، يعمل كل منها بمعدل \qty{20}{} في الثانية،
        يلزم للمعدل الفعلي؟
  \item افترض أن \qty{100}{g} نفسها خُزنت سلاسل أميلوز بالقياس نفسه
        الذي للجسيمات. احسب عدد الأطراف ومعدل الإطلاق الأقصى.
  \item فسّر لماذا كانت نقاط التفرع مفتاح استجابة الكبد لهبوط غلوكوز
        الدم في دقائق.
\end{enumerate}

\noindent\textbf{الجزء الرابع --- الثمن التناضحي.}
\begin{enumerate}[resume]
  \item لو أُذيبت \qty{100}{g} غلوكوزًا حرًا في \qty{1.0}{L} من ماء
        \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}، فأي أسمولية كانت لتضيف؟
  \item قارن مع \qty{0.3}{osmol/L} في \omterm{def:b1:eukaryotic-cell:organelle}{العصارة الخلوية} وتنبأ بما كان
        ليحل \omterm{def:b1:cell-unit-of-life:cell}{بالخلايا}.
  \item احسب الأسمولية التي تسهم بها جسيمات \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} نفسها.
  \item احسب \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي} الذي كان الغلوكوز الحر ليحدثه
        ($\Psi_s = -RTc$، و$RT = \qty{2.58}{MPa.L/mol}$) والضغط الذي
        كان جدار من النوع النباتي ليحتاج إلى تحمله.
  \item تدخر العضلة \qty{400}{g} من \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} في \qty{30}{kg} من
        العضل لكنها لا تستطيع إطلاق الغلوكوز إلى الدم. اقترح سببًا،
        بحسب ما يفيده \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} في عضلة.
  \item البطاطا تدخر \omterm{def:b1:carbohydrates:polysaccharide}{النشا} في حبيبات قدرها \qty{50}{\micro m}، لا في
        جسيمات \omterm{def:b1:carbohydrates:polysaccharide}{غليكوجين} قدرها \qty{40}{nm}. اربط الفرق بالمقياس
        الزمني للحشد الذي يحتاج إليه كل كائن.
  \item بعد وجبة يعيد الكبد بناء \qty{100}{g} من \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} في أربع
        ساعات. احسب المعدل ببقايا الغلوكوز في الثانية وكلفة \omterm{def:b1:water-small-molecules:atp}{ATP} إذا
        كانت إضافة بقية واحدة تكلف مكافئي \omterm{def:b1:water-small-molecules:atp}{ATP}.
  \item اذكر النتيجة: معدل إطلاق الكبد للغلوكوز بالجزيئات في الثانية،
        وعدد الأطراف غير المرجعة الذي يتيحه، والأسمولية التي يوفرها
        البوليمر على \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}.
\end{enumerate}
\end{problem}
