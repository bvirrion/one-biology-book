\section*{Bab \ref{ch:b1:expression-control} --- Kendali Pengungkapan Gen}

\begin{solution}{exo:b1:expression-control:1}
\emph{lacI} (\omterm{def:b1:genomes:gene}{gen} \omterm{def:b1:expression-control:operon}{penekannya}), promotor, \omterm{def:b1:expression-control:operon}{operator}, \emph{lacZ},
\emph{lacY}, \emph{lacA}, dan sisi CAP-nya. Tanpa laktosa: padam
(\omterm{def:b1:expression-control:operon}{penekannya} terikat). Laktosa ada, glukosa ada: nyaris tak nyala
(\omterm{def:b1:expression-control:operon}{penekannya} lepas, tetapi tanpa cAMP--CAP). Laktosa ada, tanpa glukosa:
nyala penuh. Tanpa kedua gulanya: padam.
\end{solution}

\begin{solution}{exo:b1:expression-control:2}
Sebuah hasil yang bekerja secara trans adalah molekul yang dapat
berdifusi yang dapat bekerja pada salinan sasarannya mana pun dalam
\omterm{def:b1:cell-unit-of-life:cell}{selnya}: \omterm{def:b1:expression-control:operon}{penekannya}. Sebuah sisi yang bekerja secara cis adalah urutan
\omterm{def:b1:nucleic-acids:chain}{DNA} yang hanya memengaruhi \omterm{def:b1:genomes:gene}{gen} yang terpaut padanya secara fisik:
\omterm{def:b1:expression-control:operon}{operatornya}.
\end{solution}

\begin{solution}{exo:b1:expression-control:3}
Jedanya sekitar \qty{36}{min} (dari \qty{2}{h} ke \qty{2.6}{h}); lipat
duanya \qty{20}{min} pada glukosa, \qty{30}{min} pada laktosa.
\end{solution}

\begin{solution}{exo:b1:expression-control:4}
\omterm{def:b1:expression-control:enhancer}{Penguat}: sebuah urutan pengatur, sering jauh, yang menaikkan
transkripsi sebuah \omterm{def:b1:genomes:gene}{gen} ketika faktor mengikatnya. \omterm{def:b1:expression-control:enhancer}{Faktor transkripsi}:
sebuah \omterm{def:b1:proteins:peptide}{protein} dengan ranah pengikat \omterm{def:b1:nucleic-acids:chain}{DNA} dan ranah pengaktif atau
\omterm{def:b1:expression-control:operon}{penekan} yang mengatur transkripsinya. \omterm{def:b1:genomes:gene}{Gen} rumah tangga: \omterm{def:b1:genomes:gene}{gen} yang
diungkapkan pada tiap \omterm{def:b1:cell-unit-of-life:cell}{sel} pada aras yang mantap.
\end{solution}

\begin{solution}{exo:b1:expression-control:5}
$I^-$: konstitutif. $O^c$: konstitutif. $I^s$: tak dapat diinduksi.
$I^-/I^+$: dapat diinduksi (\omterm{def:b1:expression-control:operon}{penekan} $I^+$-nya bekerja secara trans).
$I^s/I^+$: tak dapat diinduksi ($I^s$ dominan).
$I^+\,O^c\,Z^- / I^+\,O^+\,Z^+$: dapat diinduksi --- satu-satunya $Z$
yang bekerja berada di belakang sebuah \omterm{def:b1:expression-control:operon}{operator} yang normal.
\end{solution}

\begin{solution}{exo:b1:expression-control:6}
Lac: gulanya (penginduksi) mengikat \omterm{def:b1:expression-control:operon}{penekannya} lalu
\emph{melepaskannya} dari \omterm{def:b1:expression-control:operon}{operatornya} --- \omterm{def:b1:enzymes:enzyme}{enzimnya} dibuat ketika
\omterm{def:b1:enzymes:enzyme}{substratnya} hadir. Trp: \omterm{def:b1:proteins:aminoacid}{asam aminonya} (korepresor) mengikat \omterm{def:b1:expression-control:operon}{penekannya}
lalu \emph{memungkinkannya} mengikat --- \omterm{def:b1:enzymes:enzyme}{enzimnya} dibuat ketika hasilnya
tidak ada. Lintasan katabolik dinyalakan oleh masukannya, lintasan
anabolik dipadamkan oleh keluarannya.
\end{solution}

\begin{solution}{exo:b1:expression-control:7}
Pada glukosa: pertumbuhan normal (glukosa tak memerlukan CAP). Pada
laktosa saja: nyaris tak tumbuh --- tanpa cAMP, CAP tak dapat
mengaktifkan promotor lac-nya, sehingga \omterm{def:b1:enzymes:enzyme}{enzimnya} tinggal pada beberapa
persen aras terinduksinya bahkan dengan \omterm{def:b1:expression-control:operon}{penekannya} terlepas. Pada
keduanya: tumbuh pada glukosa, lalu berhenti; tak ada tahap kedua.
\omterm{def:b1:enzymes:enzyme}{Enzimnya} rendah pada tiap halnya.
\end{solution}

\begin{solution}{exo:b1:expression-control:8}
Atenuasi menuntut \omterm{def:b1:gene-expression:ribosome}{ribosomnya} mentranslasikan pembawa pesannya sementara
polimerasenya masih mentranskripsikannya, sehingga letak \omterm{def:b1:gene-expression:ribosome}{ribosomnya}
membentuk \omterm{def:b1:nucleic-acids:chain}{RNA} di belakang polimerasenya. Pada eukariot transkripsinya
berada dalam \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya} dan translasinya dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya}, dan pesannya
tidak ditranslasikan sampai ia diproses lalu diekspor.
\end{solution}

\begin{solution}{exo:b1:expression-control:9}
\omterm{def:b1:expression-control:enhancer}{Penguat} bekerja dari jauh, pada arah hadap mana pun dan pada sisi mana
pun, sehingga ia tak dapat bekerja lewat dibaca atau lewat menempatkan
polimerasenya secara langsung; ia bekerja lewat mengikat faktor yang
menyentuh promotornya lewat sebuah gelung \omterm{def:b1:nucleic-acids:chain}{DNA}, yang jarak dan arah
hadapnya sedikit berpengaruh.
\end{solution}

\begin{solution}{exo:b1:expression-control:10}
$5000\times 4096\times 4 = \num{8.2e7}$ \omterm{def:b1:water-small-molecules:atp}{ATP}: \qty{0.8}{\%} anggarannya.
Laju pertumbuhannya lebih rendah \qty{0.8}{\%}: tiap generasi bagian
mutannya turun sebesar faktor $2^{-0.008} = 0.9945$. Dari $1:1$ ke
$1:99$: $0.9945^n = 1/99$, $n = \ln 99/0.00555 = 830$ generasi ---
beberapa minggu biakan sinambung.
\end{solution}

\begin{solution}{exo:b1:expression-control:11}
Tiap jenis \omterm{def:b1:cell-unit-of-life:cell}{sel} membawa seperangkat \omterm{def:b1:expression-control:enhancer}{faktor transkripsi} yang khas, yang
mengikat \omterm{def:b1:expression-control:enhancer}{penguat} \omterm{def:b1:genomes:gene}{gennya} lalu merekrut asetilase yang membuka
kromatinnya, sementara \omterm{def:b1:genomes:gene}{gen} jenis yang lain termetilasi dan terkemas
menjadi heterokromatin yang tak terjangkau faktor mana pun. Faktornya
saling memelihara pengungkapan satu sama lain (sebuah \omterm{def:b1:body-plans-tissues:tissue}{jaringan} dengan
keadaan yang mantap), dan tanda kromatinnya disalin pada replikasinya,
sehingga anaknya mewarisi faktornya maupun kawasan terbuka dan
tertutupnya. \omterm{def:b1:genomes:genome}{Genomnya} sama; \omterm{def:b1:genomes:genome}{genom} yang terjangkau tidak, dan
ketakterjangkauannya diwariskan.
\end{solution}

\begin{solution}{exo:b1:expression-control:12}
Kendali sebuah bakteri disetel menurut mediumnya: sebuah gula, sebuah
\omterm{def:b1:proteins:aminoacid}{asam amino}, sebuah suhu mengikat sebuah \omterm{def:b1:expression-control:operon}{penekan} atau sebuah \omterm{prop:b1:expression-control:trp}{faktor
sigma} dan tanggapannya lengkap dalam beberapa menit lalu dibalik begitu
isyaratnya pergi --- pengungkapannya mengikuti \omterm{def:b1:organism-environment:organism}{lingkungannya}. Sebuah
\omterm{def:b1:cell-unit-of-life:cell}{sel} sebuah tubuh juga menjawab isyarat, tetapi keputusan utamanya
diambil selama perkembangannya lalu dikunci dalam kromatinnya: apa yang
dapat diungkapkannya ditetapkan faktor yang diwarisinya dan tanda pada
\omterm{def:b1:nucleic-acids:chain}{DNA-nya}, yang hanya berubah lewat pembelahan dan sebagian besarnya tak
berbalik. Bakterinya adalah pembaca masa kini; \omterm{def:b1:cell-unit-of-life:cell}{sel} yang terdiferensiasi
memiliki sebuah ingatan yang menjadi jati dirinya.
\end{solution}

\begin{solution}{pb:b1:expression-control:1}
\textbf{1.} $0.5\times 10^{-3}\times 180 = \qty{0.09}{g/L} =
\qty{9e-5}{g/mL}$.
\textbf{2.} $9\times 10^{-5}/1.5\times 10^{-12} = \num{6e7}$ pembelahan.
\textbf{3.} $10^7 + \num{6e7} = \num{7e7}$ \omterm{def:b1:cell-unit-of-life:cell}{sel} tiap mL.
\textbf{4.} $7 = 2^n$: $n = 2.8$ kali lipat dua, \qty{56}{min}.
\textbf{5.} Padam: laktosanya (alolaktosa) telah melepaskan
\omterm{def:b1:expression-control:operon}{penekannya}, tetapi glukosanya menjaga cAMP tetap rendah, sehingga CAP
tidak terikat dan promotornya nyaris bisu; lagi pula permeasenya
langka, sehingga sedikit laktosa yang masuk.
\textbf{6.} cAMP naik lalu mengikat CAP, yang mengikat promotornya.
\textbf{7.} Kendali negatifnya diangkat tetapi kendali positifnya tak
ada: sebuah promotor lemah tanpa CAP jarang mulai (\omterm{def:b1:expression-control:cap}{penekanan
katabolit}).
\textbf{8.} $5000/1800 = 2.8$ tetramer tiap sekon; $2.8\times 4096 =
\num{11400}$ residu tiap sekon.
\textbf{9.} Kemampuannya $20\,000\times 20 = \num{4e5}$ residu tiap sekon:
\qty{3}{\%} \omterm{def:b1:gene-expression:ribosome}{ribosomnya}.
\textbf{10.} $5000\times 4096\times 110\times 1.66\times 10^{-24} =
\qty{3.7e-15}{g}$: \qty{2.5}{\%} \omterm{def:b1:proteins:peptide}{protein} \omterm{def:b1:cell-unit-of-life:cell}{selnya}.
\textbf{11.} $5000/5 = 1000$.
\textbf{12.} Laktosa tak dapat melintasi membrannya tanpa permeasenya;
sebuah mutan \emph{lacY}$^-$ memiliki \omterm{def:b1:enzymes:enzyme}{enzimnya} tetapi tak memiliki
\omterm{def:b1:enzymes:enzyme}{substrat} di dalamnya dan tidak tumbuh pada laktosa (dan tidak
terinduksi dengan baik pula, sebab penginduksinya dibuat dari laktosa
di dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya}).
\textbf{13.} $1.0\times 10^{-3}\times 342 = \qty{0.342}{g/L}$, yang
senilai $2\times 180 = \qty{360}{g}$ glukosa tiap mol, yakni
\qty{0.36}{g/L}
$= \qty{3.6e-4}{g/mL}$.
\textbf{14.} $3.6\times 10^{-4}/1.5\times 10^{-12} = \num{2.4e8}$
pembelahan; akhirnya $\num{7e7} + \num{2.4e8} = \num{3.1e8}$ tiap mL.
\textbf{15.} $3.1/0.7 = 4.4 = 2^n$: $n = 2.1$ kali lipat dua,
\qty{64}{min}.
\textbf{16.} Glukosa \qty{56}{min}, jeda \qty{30}{min}, laktosa
\qty{64}{min}: \qty{150}{min} seluruhnya.
\textbf{17.} Laktosanya harus dihidrolisis lebih dahulu, sebuah langkah
\omterm{def:b1:enzymes:enzyme}{enzim} tambahan yang lajunya membatasi pasokan glukosanya; dan
permeasenya memakai gradien protonnya untuk mengimpornya, sebuah biaya
yang tidak dibayar pengangkutan glukosanya.
\textbf{18.} $I^-$: \omterm{def:b1:enzymes:enzyme}{enzimnya} \num{5000} sepanjang itu (konstitutif,
begitu glukosanya habis; pada glukosa CAP masih membatasinya, sehingga
arasnya pertengahan); jedanya lebih pendek atau tak ada, sehingga
kurvanya memperlihatkan jeda yang lebih kecil; biayanya sedikit
memperlambatnya pada glukosa.
\textbf{19.} CAP$^-$: tumbuh pada glukosa sampai $\num{7e7}$, lalu
berhenti: \omterm{def:b1:expression-control:operon}{operon lac-nya} tak dapat diaktifkan, \omterm{def:b1:enzymes:enzyme}{enzimnya} tinggal dekat 5
molekul, dan laktosanya tak pernah terpakai.
\textbf{20.} Pada laktosa saja: tak ada perbedaan pada aras \omterm{def:b1:enzymes:enzyme}{enzimnya}
begitu terinduksi (keduanya nyala penuh) dan pada dasarnya tak ada pada
pertumbuhannya; $O^c$ sekadar tidak mengalami tundaan induksinya. Pada
glukosa saja: $O^c$ membuat \omterm{def:b1:enzymes:enzyme}{enzimnya} tanpa guna (sejauh yang diizinkan
CAP), tipe liarnya tidak.
\textbf{21.} $5000\times 4096\times 4 = \num{8.2e7}$ \omterm{def:b1:water-small-molecules:atp}{ATP}: \qty{0.8}{\%}
dari $10^{10}$.
\textbf{22.} Waktu generasinya lebih panjang \qty{0.8}{\%}: laju
pertumbuhannya lebih rendah \qty{0.8}{\%}, $\delta = 0.008$ kali lipat
dua tiap generasi.
\textbf{23.} $r = 2^{-0.008} = 0.99447$; $r^n = 0.111$: $n =
\ln 0.111/\ln 0.99447 = 2.20/0.00555 = 400$ generasi.
\textbf{24.} Pada laktosa \omterm{def:b1:enzymes:enzyme}{enzimnya} memang dibutuhkan, sehingga mutannya
tidak membayar apa pun lalu memperoleh jedanya: ia tidak terpilih
lawan, dan kadang terpilih menguntungkan; lagi pula galur laboratorium
ditumbuhkan demi kemudahan, bukan demi persaingan.
\textbf{25.} Faktor induksinya 1000 (5 ke \num{5000} molekul);
sintesis yang tak perlu memakan biaya \qty{0.8}{\%} anggaran \omterm{def:b1:water-small-molecules:atp}{ATP-nya};
percobaannya berlangsung sekitar dua setengah jam.
\end{solution}
