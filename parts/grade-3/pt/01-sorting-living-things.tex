\chapter{Classificar os seres vivos}\label{ch:g3:sorting-living-things}

Despeje uma caixa de botões misturados na mesa e as suas mãos
começam a separar sozinhas: os grandes com os grandes, os vermelhos
com os vermelhos, os de quatro furos com os de quatro furos. Os
naturalistas sentiram a mesma vontade diante da bagunça de besouros,
\omterm{prop:g3:sorting-living-things:groups}{aves}, samambaias e \omterm{prop:g3:sorting-living-things:groups}{peixes} do mundo vivo --- e classificar os \omterm{def:g1:living-or-not:living}{seres
vivos} com cuidado acabou virando uma das ideias mais poderosas da
\omterm{def:g1:living-or-not:living}{biologia}.

\section{Classificar precisa de uma regra}

\begin{definition}[Classificação e critérios]\label{def:g3:sorting-living-things:criterion}
\emph{Classificar}\index{classificação} uma coleção é juntar as
coisas que combinam entre si. Um \emph{critério}\index{critério} é a
regra usada na classificação --- ``tem seis patas'', ``tem \omterm{def:g1:animals-around-us:covering}{penas}''.
Um bom critério para \omterm{def:g1:living-or-not:living}{seres vivos} é uma característica do corpo que
dá para \emph{observar}: algo que o \omterm{def:g1:animals-around-us:animal}{animal} ou a \omterm{def:g1:plants-around-us:plant}{planta} \emph{tem}, e
não o lugar onde vive ou o que está fazendo naquele momento.
\end{definition}

\begin{example}[Duas classificações dos mesmos animais]\label{ex:g3:sorting-living-things:two}
Pegue o gato, o pato, a truta e a joaninha. Classificados por ``vive
perto da água'': pato e truta juntos. Classificados por ``tem
\omterm{def:g1:animals-around-us:covering}{penas}'': o pato fica sozinho. As duas são classificações --- mas só
a segunda usa o próprio corpo do \omterm{def:g1:animals-around-us:animal}{animal}, e é esse o tipo em que os
biólogos confiam: um \omterm{def:g1:animals-around-us:animal}{animal} pode mudar de casa, mas não pode trocar
as \omterm{def:g1:animals-around-us:covering}{penas} por escamas.
\end{example}

\begin{remark}[Por que não classificar por casa ou por cardápio?]\label{rem:g3:sorting-living-things:why}
O \omterm{def:g2:where-animals-live:habitat}{hábitat} e o cardápio mudam com as estações e com a sorte --- o
sabiá do \cref{ch:g2:what-animals-eat} troca minhocas por frutinhas
sem virar outro \omterm{def:g1:animals-around-us:animal}{animal}. As características do corpo ficam. É por
isso que ``o que ele tem'' vence ``o que ele faz'' como regra de
\omterm{def:g3:sorting-living-things:criterion}{classificação}.
\end{remark}

\section{Os grandes grupos de animais}

\begin{proposition}[Alguns grandes grupos]\label{prop:g3:sorting-living-things:groups}
\omterm{def:g3:sorting-living-things:criterion}{Classificar} por características do corpo reúne os \omterm{def:g1:animals-around-us:animal}{animais} em grandes
grupos, entre eles:
\begin{itemize}
  \item os \emph{insetos}\index{inseto}: seis patas, corpo em três
        partes --- a formiga, a joaninha, a abelha, a borboleta;
  \item os \emph{peixes}\index{peixe}: nadadeiras e escamas, vida
        passada na água --- a truta, o peixinho dourado;
  \item as \emph{aves}\index{ave}: \omterm{def:g1:animals-around-us:covering}{penas}, asas e bico --- o pato, o
        sabiá, a ema;
  \item os \emph{mamíferos}\index{mamífero}: pelo, e mães que
        alimentam os filhotes com \omterm{def:g2:animals-grow-up:born}{leite} --- o gato, a vaca, o
        morcego, a baleia --- e os seres humanos;
  \item os \emph{anfíbios}\index{anfíbio}: \omterm{def:g1:the-five-senses:sense}{pele} úmida, \omterm{def:g2:animals-grow-up:born}{ovos} postos
        na água, uma infância de girino --- o sapo, a salamandra.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Surpresas resolvidas direito]\label{ex:g3:sorting-living-things:surprises}
Os \omterm{def:g3:sorting-living-things:criterion}{critérios} nos salvam de palpites preguiçosos. O morcego voa como
uma ave --- mas tem pelo, não tem \omterm{def:g1:animals-around-us:covering}{pena} e alimenta os filhotes com
\omterm{def:g2:animals-grow-up:born}{leite}: é um mamífero. A baleia nada como um peixe --- com o pelo
reduzido a quase nada, mas com \omterm{def:g2:animals-grow-up:born}{leite} para o filhote e sem escamas:
também é um mamífero. A ema não voa de jeito nenhum e, mesmo assim,
as \omterm{def:g1:animals-around-us:covering}{penas} e o bico dizem: ave. Quem decide é o \omterm{def:g3:sorting-living-things:criterion}{critério}, não a
impressão.
\end{example}

\begin{method}[Classificar um animal desconhecido]\label{met:g3:sorting-living-things:sort}
\begin{enumerate}
  \item Observe o corpo: \omterm{def:g1:the-five-senses:sense}{pele}, pelo, \omterm{def:g1:animals-around-us:covering}{penas} ou escamas? quantas
        patas? asas? bico?
  \item Pergunte sobre os filhotes, se der: \omterm{def:g2:animals-grow-up:born}{leite}? \omterm{def:g2:animals-grow-up:born}{ovos}? uma fase
        de girino?
  \item Compare com os grupos da
        \cref{prop:g3:sorting-living-things:groups} e ponha o
        \omterm{def:g1:animals-around-us:animal}{animal} onde as \emph{características} dele encaixam;
  \item se uma impressão (``ele voa!'') discordar de uma
        característica (``ele tem pelo''), confie na característica.
\end{enumerate}
\end{method}

\section{As plantas também são classificadas}

\begin{proposition}[Classificar plantas]\label{prop:g3:sorting-living-things:plants}
As \omterm{def:g1:plants-around-us:plant}{plantas} são classificadas pelas características delas, como os
\omterm{def:g1:animals-around-us:animal}{animais}. \omterm{def:g3:sorting-living-things:criterion}{Critérios} úteis: ela faz \emph{\omterm{def:g1:plants-around-us:parts}{flores}} e \omterm{def:g2:from-seed-to-plant:seed}{sementes} (a
margarida, o castanheiro) ou nunca faz (a samambaia, o musgo)? O
\omterm{def:g1:plants-around-us:parts}{caule} dela é macio e verde, ou lenhoso como o tronco de uma
\emph{árvore}? As \omterm{def:g1:plants-around-us:parts}{folhas} são largas ou são agulhas, como as do
pinheiro?
\end{proposition}
\admitted

\begin{example}[Três cantos do parque]\label{ex:g3:sorting-living-things:park}
Num mesmo parque: a roseira --- \omterm{def:g1:plants-around-us:parts}{flores}, \omterm{def:g1:plants-around-us:parts}{caules} lenhosos; o musgo do
muro voltado para o sul --- nunca dá \omterm{def:g1:plants-around-us:parts}{flores}, nem tem \omterm{def:g1:plants-around-us:parts}{caule} de
verdade; o pinheiro --- uma árvore de agulhas e pinhas. Três
\omterm{def:g1:plants-around-us:plant}{plantas}, três respostas diferentes aos mesmos \omterm{def:g3:sorting-living-things:criterion}{critérios}.
\end{example}

\begin{remark}[Classificar é o primeiro passo de uma ciência]\label{rem:g3:sorting-living-things:science}
Grupos bem classificados deixam as pessoas fazerem perguntas
melhores: tudo o que é verdade dos \omterm{prop:g3:sorting-living-things:groups}{insetos} em geral --- seis patas,
corpo em três partes, muitas vezes asas --- passa a ser conhecido
para centenas de milhares de espécies de uma vez. Capítulos mais
adiante vão transformar a \omterm{def:g3:sorting-living-things:criterion}{classificação} numa árvore de parentesco de
todos os \omterm{def:g1:living-or-not:living}{seres vivos} --- por enquanto, aprenda a classificar pelo
que dá para ver.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:1}
O que é um \omterm{def:g3:sorting-living-things:criterion}{critério}? Dê um bom \omterm{def:g3:sorting-living-things:criterion}{critério} para classificar \omterm{def:g1:animals-around-us:animal}{animais} e
um que os biólogos evitam.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:2}
Diga a característica (ou as características) que define cada grupo:
\omterm{prop:g3:sorting-living-things:groups}{insetos}, \omterm{prop:g3:sorting-living-things:groups}{peixes}, \omterm{prop:g3:sorting-living-things:groups}{aves}, \omterm{prop:g3:sorting-living-things:groups}{mamíferos}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:3}
Ponha cada \omterm{def:g1:animals-around-us:animal}{animal} no grupo dele: a abelha, o peixinho dourado, a
ema, a vaca, o sapo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:4}
O morcego voa. Por que ele é um mamífero e não uma ave? Que
características decidem?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:5}
A baleia vive no mar. Por que ela não é um peixe? Que
características decidem?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:6}
Uma aranha tem oito patas. Ela pode ser um inseto? O que o \omterm{def:g3:sorting-living-things:criterion}{critério}
responde?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:7}
Classifique estas \omterm{def:g1:plants-around-us:plant}{plantas} com os \omterm{def:g3:sorting-living-things:criterion}{critérios} da
\cref{prop:g3:sorting-living-things:plants}: a margarida, o musgo, o
pinheiro, a roseira.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:8}
Faça a experiência: classifique os \omterm{def:g1:animals-around-us:animal}{animais} da sua rua ou do seu
quintal --- todos os que você conseguir ver num passeio --- nos
grupos deste capítulo. Que grupo ganha em número?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sorting-living-things:9}
A Lea classifica os \omterm{def:g1:animals-around-us:animal}{animais} em ``de estimação'' e ``selvagens''. A
amiga dela classifica os mesmos \omterm{def:g1:animals-around-us:animal}{animais} em ``\omterm{prop:g3:sorting-living-things:groups}{mamíferos}'' e ``\omterm{prop:g3:sorting-living-things:groups}{aves}''.
Qual das duas classificações é do tipo do biólogo, e por quê? Use a
\cref{rem:g3:sorting-living-things:why}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sorting-living-things:10}
Um pinguim nada muito bem, não voa e é coberto por uma coisa que
parece telhinha brilhante. Um naturalista olha de perto: as
``telhinhas'' são \omterm{def:g1:animals-around-us:covering}{penas} bem juntinhas, e há um bico. Rode o
\cref{met:g3:sorting-living-things:sort} no pinguim, passo a passo.
\end{exercise}
