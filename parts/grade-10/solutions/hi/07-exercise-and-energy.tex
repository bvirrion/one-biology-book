\section*{अध्याय \ref{ch:g10:exercise-and-energy} --- व्यायाम और शरीर की ऊर्जा-ज़रूरतें}

\begin{solution}{exo:g10:exercise-and-energy:1}
पूरे विश्राम में शरीर को जीवित रखने पर ख़र्च होने वाली ऊर्जा --- हृदय,
मस्तिष्क, ताप, \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} का रखरखाव: लगभग \qty{80}{W}, यानी दिन भर में
क़रीब \qty{7000}{kJ}।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:2}
प्रति मिनट $2.0 \times 20 = \qty{40}{kJ}$, यानी $40000/60 \approx
\qty{670}{W}$।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:3}
पेशी का फ़ॉस्फ़ेट भंडार (लगभग दस सेकंड); लैक्टिक \omterm{def:g10:cell-metabolism:fermentation}{किण्वन} (पूरे ज़ोर की
कोशिश के एक से दो मिनट); \omterm{prop:g10:cell-metabolism:respiration}{कोशिकीय श्वसन} (घंटों)।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:4}
वह सबसे ऊँची दर जिस पर शरीर ऑक्सीजन ले और इस्तेमाल कर सकता है। प्रति
किलोग्राम $3500/70 = \qty{50}{mL/min}$।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:5}
पेशियाँ, लगभग \qty{400}{g}, जो उसी पेशी को ईंधन देता है जो उसे संचित करती
है; और यकृत, लगभग \qty{100}{g}, जो मस्तिष्क तथा बाक़ी ऊतकों के लिए
ग्लूकोज़ बनकर रक्त में छोड़ा जाता है।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:6}
\qty{2.7}{L/min}: प्रति मिनट $2.7 \times 20 = \qty{54}{kJ}$, यानी
\qty{900}{W}। ऊष्मा: $900 - 200 = \qty{700}{W}$।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:7}
कार्य $mgh = 60 \times 9.8 \times 800 \approx \qty{470}{kJ}$। 25\%
दक्षता पर पेशियों ने लगभग \qty{1900}{kJ} ख़र्च किए। \qty{7200}{s} में:
यानी विश्राम से ऊपर लगभग \qty{260}{W}।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:8}
100 मीटर की फ़र्राटा दौड़ लगभग पूरी तरह फ़ॉस्फ़ेट भंडार पर चलती है: न
ऑक्सीजन का उधार, न कोई ख़ास लैक्टिक अम्ल। 400 मीटर की दौड़ लगभग एक मिनट
चलती है और अपनी आधी ऊर्जा लैक्टिक \omterm{def:g10:cell-metabolism:fermentation}{किण्वन} से खींचती है: लैक्टिक अम्ल जमा
होता है (वही दर्द), और जिस श्वसन को उसे हटाना है वह धावक को बाद में भी
कई मिनट हाँफता रखता है।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:9}
\omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} से ऊर्जा $0.6 \times 3000 = \qty{1800}{kJ}$; \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन}
$1800/17 \approx \qty{106}{g}$; और छूटा हुआ जल लगभग \qty{320}{g}।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:10}
उपापचयी शक्ति लगभग \qty{600}{W}: $\num{1.1e6}/600 \approx
\qty{1830}{s}$, यानी क़रीब 30 मिनट।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:11}
पेशियों में छूटी ऊर्जा का तीन-चौथाई ऊष्मा है, यानी कड़ी मेहनत के दौरान
कई सौ वाट। पसीना उसे हटाता है: जल के वाष्पित होने में प्रति ग्राम लगभग
\qty{2.4}{kJ} लगते हैं, और त्वचा तक जाता रक्त ऊष्मा वहाँ पहुँचाता है।
पसीने के बिना शरीर का ताप हर कुछ मिनट में एक अंश बढ़ जाता।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:12}
चढ़ाई पर ज़रूरी शक्ति शरीर-द्रव्यमान के अनुपात में होती है: \qty{60}{kg}
वाले व्यक्ति के पास प्रति किलोग्राम \qty{60}{mL/min} हैं, जबकि दूसरे के
पास \num{40}, इसलिए वही तेज़ चढ़ता है। प्रयोगशाला की साइकिल पर शक्ति
द्रव्यमान पर निर्भर नहीं करती, दोनों के पास वही \qty{3.6}{L/min} है, और
दोनों उतने ही वाट टिका सकते हैं।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:13}
ऑक्सीजन-तुल्यांक के जो \qty{0.4}{L/min} कम पड़ रहे हैं, वे पेशियों के
लैक्टिक \omterm{def:g10:cell-metabolism:fermentation}{किण्वन} से आते हैं। लैक्टिक अम्ल जमा होता है; कुछ ही मिनटों में
पेशियाँ जलने लगती हैं और धावक को उस रफ़्तार पर धीमा होना पड़ता है जिसे
अकेला श्वसन ढँक सके।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:14}
छोटे-छोटे अत्यधिक तीव्र झोंके फ़ॉस्फ़ेट भंडार और \omterm{def:g10:cell-metabolism:fermentation}{किण्वन} पर चलते हैं, जो
चर्बी बिलकुल इस्तेमाल नहीं करते; चर्बी केवल श्वसन जलाता है, जिसे पूरी दर
तक पहुँचने में मिनट लगते हैं और जिसका बोलबाला केवल लंबे मध्यम व्यायाम में
होता है। और चंद झोंकों की कुल ऊर्जा भी घंटे भर की लगातार मेहनत के मुक़ाबले
कम ही होती है।
\end{solution}

\begin{solution}{exo:g10:exercise-and-energy:15}
घाटा $25000 - 8500 - 6000 = \qty{10500}{kJ}$, यानी लगभग \qty{280}{g}
चर्बी। चरण के दौरान खाते रहना मस्तिष्क के लिए रक्त-ग्लूकोज़ ऊपर बनाए
रखता है और चढ़ाइयों के लिए \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} बचा लेता है, जहाँ ज़रूरी शक्ति उससे
कहीं ऊपर होती है जितनी अकेली चर्बी दे सकती है; चर्बी दिन के ठहरे हुए
हिस्से को ढँकती है।
\end{solution}

\begin{solution}{pb:g10:exercise-and-energy:1}
\textbf{1.} $4.0 \times 60 \times 42.2 \approx \qty{10100}{kJ}$।

\textbf{2.} $\qty{3}{h}\,\qty{30}{min} = \qty{12600}{s}$:
$\num{1.01e7}/12600 \approx \qty{800}{W}$।

\textbf{3.} $10100/20 \approx \qty{506}{L}$; और \qty{210}{min} में, यानी
लगभग \qty{2.4}{L/min}।

\textbf{4.} प्रति किलोग्राम $2400/60 = \qty{40}{mL/min}$, यानी उसकी
अधिकतम का $40/52
\approx 77\%$।

\textbf{5.} $(2.4 - 0.2)/2.4 \approx 92\%$।

\textbf{6.} $380 \times 17 = \qty{6460}{kJ}$।

\textbf{7.} दौड़ का दाम प्रति किलोमीटर \qty{240}{kJ} है: $6460/240
\approx \qty{27}{km}$।

\textbf{8.} \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} की खपत $0.8 \times 240 = \qty{192}{kJ/km}$:
$6460/192 \approx \qty{34}{km}$ --- यही दीवार है।

\textbf{9.} चर्बी से ऊर्जा $0.2 \times 10100 \approx \qty{2020}{kJ}$,
यानी $2020/38 \approx \qty{53}{g}$: उसके \qty{9}{kg} का क़रीब 0.6\%।

\textbf{10.} चर्बी केवल श्वसन से और एक सीमित दर पर ही जलती है: वह मोटे
तौर पर उसकी रफ़्तार की माँगी हुई शक्ति का आधा ही दे सकती है। \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन}
के बिना पेशियाँ इतनी तेज़ी से ATP नहीं बना पातीं, और रफ़्तार गिर जाती है।

\textbf{11.} $60 \times 3.5 = \qty{210}{g}$, यानी $210 \times 17 \approx
\qty{3570}{kJ}$, जो \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} खपत के $3570/192 \approx \qty{19}{km}$
ढँक लेता है।

\textbf{12.} उपलब्ध \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} $6460 + 3570 = \qty{10030}{kJ}$:
$10030/192 \approx \qty{52}{km}$, यानी अंतिम रेखा से आगे। हाँ।

\textbf{13.} $+\qty{200}{g}$ \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} और $+ \qty{600}{g}$ जल:
\qty{0.8}{kg}, जिसका दाम प्रति किलोमीटर अतिरिक्त
$0.8 \times 4.0 \approx \qty{3}{kJ}$ --- 240 के सामने नगण्य।

\textbf{14.} \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} $580 \times 17 = \qty{9860}{kJ}$ और उसके ऊपर
घोल: कुल मिलाकर लगभग \qty{13400}{kJ}; \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} की खपत
$0.55 \times 240 =
\qty{132}{kJ/km}$: यानी \qty{100}{km} से ऊपर। दीवार अंतिम रेखा से बहुत
आगे खिसक गई है, और अच्छे-ख़ासे अंतर से।

\textbf{15.} मस्तिष्क केवल ग्लूकोज़ जलाता है, जो यकृत के \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} से
आता है। वह भंडार ख़ाली होते ही रक्त-ग्लूकोज़ गिरता है, और मस्तिष्क इस
आपात स्थिति का संकेत चक्कर और भारी थकावट के रूप में देता है, चाहे
पेशियों के पास कुछ भी बचा हो।

\textbf{16.} $3.6 \times 55 \times 42.2 \approx \qty{8360}{kJ}$;
\qty{125}{min} में $8360/20 = \qty{418}{L}$: यानी लगभग \qty{3.3}{L/min}।

\textbf{17.} प्रति किलोग्राम $3340/55 \approx \qty{61}{mL/min}$, यानी
$61/80 \approx 76\%$ --- यानी नादिया जितना ही अंश। छत अलग है, उसका
इस्तेमाल किया गया अंश नहीं।

\textbf{18.} मितव्ययिता प्रति किलोग्राम प्रति किलोमीटर का दाम है।
\qty{55}{kg} के धावक के लिए पूरी दौड़ में
$(4.0 - 3.6) \times 55 \times 42.2 \approx
\qty{930}{kJ}$ की बचत, यानी लगभग 9\%।

\textbf{19.} \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} \qty{8500}{kJ}; ज़रूरी \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} $0.7 \times
8360 \approx \qty{5850}{kJ}$: हाँ, भंडार बिना कुछ पिए काफ़ी हैं।

\textbf{20.} बिना तैयारी के नादिया का \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} किलोमीटर 34 के आसपास
चुक जाता है; दौड़ से पहले \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} भरना, दौड़ के दौरान
\omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} पीना, और वह प्रशिक्षण जो जलाई गई चर्बी का हिस्सा बढ़ाता
है --- तीनों दीवार को अंतिम रेखा के पार धकेल देते हैं।
\end{solution}
