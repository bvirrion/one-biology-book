\chapter{एंजाइम और जैवरासायनिक उत्प्रेरण}\label{ch:b1:enzymes}

कटे हुए किसी आलू पर हाइड्रोजन परॉक्साइड उँड़ेलिए और वह झाग उठता है: आलू
की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} का कैटेलेज़ का एक ही अणु प्रति सेकंड चार करोड़ परॉक्साइड अणु
तोड़ देता है, और यही अभिक्रिया अपने भरोसे छोड़ दी जाए तो वर्षों लेती।
परॉक्साइड को वैसे भी अपघटित होना ही था --- अभिक्रिया ढलान वाली है ---
पर किसी काम के समय में नहीं। \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की हर अभिक्रिया ऐसी ही है:
ऊष्मागतिकी बताती है कि वह किस ओर जा सकती है, और कोई
\emph{\omterm{def:b1:enzymes:enzyme}{एंजाइम}} बताता है कि वह अभी जाएगी या नहीं। यह अध्याय बताता है कि
कोई \omterm{def:b1:enzymes:enzyme}{एंजाइम} किसी अभिक्रिया का क्या करता है, वह बलगतिकी जिससे \omterm{def:b1:enzymes:enzyme}{एंजाइम} मापे
और तुले जाते हैं, वे ढंग जिनसे वे संदमित होते हैं, और वे ढंग जिनसे \omterm{def:b1:cell-unit-of-life:cell}{कोशिका}
उन्हें चालू और बंद करती है।

\section{उत्प्रेरण}

\begin{definition}[एंजाइम, क्रियाधार, सक्रिय स्थल]\label{def:b1:enzymes:enzyme}
\emph{एंजाइम}\index{एंजाइम} वह \omterm{def:b1:proteins:peptide}{प्रोटीन} (कभी-कभार कोई \omterm{def:b1:nucleic-acids:chain}{RNA}) है जो किसी
अभिक्रिया को \emph{उत्प्रेरित} करता है: वह दर बढ़ाता है, बिना स्वयं
ख़र्च हुए और बिना साम्य बदले। जिन अणुओं पर वह क्रिया करता है वे उसके
\emph{क्रियाधार}\index{क्रियाधार} हैं; वे \emph{सक्रिय
स्थल}\index{सक्रिय स्थल} में बँधते हैं, यानी उस दरार में जिसे कुछ ही
अवशेष अस्तरित करते हैं और जहाँ रसायन होता है। एंजाइम \emph{विशिष्ट} होते
हैं --- किसी एक क्रियाधार या किसी कुल के लिए, किसी एक बंध के लिए, किसी एक
त्रिविम समावयव के लिए --- और उनके नाम उनके क्रियाधार तथा अभिक्रिया पर
\emph{-एज़} प्रत्यय के साथ रखे जाते हैं (लैक्टेज़, \omterm{def:b1:nucleic-acids:chain}{DNA} पॉलिमरेज़,
सक्सिनेट डिहाइड्रोजनेज़), और वे छह वर्गों में हैं: ऑक्सीडोरिडक्टेज़,
हस्तांतरेज़, हाइड्रोलेज़, लायेज़, आइसोमरेज़, लाइगेज़। अनेकों को कोई
\omterm{def:b1:proteins:peptide}{अ-प्रोटीन} \emph{सहकारक}\index{सहकारक} चाहिए: कोई धातु आयन (Zn, Mg, Fe),
या कोई कार्बनिक \emph{सहएंजाइम}\index{सहएंजाइम}
--- $\mathrm{NAD^+}$, FAD, सहएंजाइम A --- जिनमें से अधिकांश विटामिनों से
बनते हैं, और विटामिन इसीलिए हैं।
\end{definition}

\begin{proposition}[एंजाइम क्या बदलता है और क्या नहीं]\label{prop:b1:enzymes:whatitdoes}
\index{सक्रियण ऊर्जा}\index{संक्रमण अवस्था}
कोई अभिक्रिया किसी \emph{\omterm{prop:b1:enzymes:whatitdoes}{संक्रमण अवस्था}} से होकर जाती है, यानी परमाणुओं
की उस तनी हुई सजावट से जिसकी \omterm{prop:b1:water-small-molecules:gibbs}{मुक्त ऊर्जा} अभिकारकों या उत्पादों से
\emph{\omterm{prop:b1:enzymes:whatitdoes}{सक्रियण ऊर्जा}} $E_a$ जितनी अधिक है; और केवल वही अणु अभिक्रिया करते
हैं जिन्हें ऊष्मीय हलचल इस अवरोध के पार ले जाती है, और उनका भिन्न
$e^{-E_a/RT}$ के समानुपाती होता है। कोई \omterm{def:b1:enzymes:enzyme}{एंजाइम} \omterm{prop:b1:enzymes:whatitdoes}{संक्रमण अवस्था} को
\omterm{def:b1:enzymes:enzyme}{क्रियाधार} से अधिक कसकर बाँधता है --- उसका \omterm{def:b1:enzymes:enzyme}{सक्रिय स्थल} उस \omterm{def:b1:flowering-plant-organization:organs}{तने} हुए रूप का
पूरक होता है --- और इस तरह $E_a$ घटा देता है: उसे \qty{37}{\celsius} पर
\qty{34}{kJ/mol} (कुछ \omterm{def:b1:water-small-molecules:water}{हाइड्रोजन बंधों} जितना) घटाने पर दर
$e^{34\,000/2580} \approx
10^6$ गुना हो जाती है। वह $\Delta G$ या साम्य स्थिरांक नहीं बदलता, जो
केवल अभिकारकों और उत्पादों पर निर्भर हैं: कोई \omterm{def:b1:enzymes:enzyme}{एंजाइम} अग्र और पश्च दोनों
अभिक्रियाएँ बराबर तेज़ करता है और तंत्र को उसी साम्य पर जल्दी पहुँचा देता
है।
\end{proposition}

\begin{proposition}[सक्रिय स्थल कैसे काम करता है]\label{prop:b1:enzymes:mechanisms}
कोई \omterm{def:b1:enzymes:enzyme}{सक्रिय स्थल} अपनी अभिक्रिया कई साधनों से एक साथ तेज़ करता है: वह
\omterm{def:b1:enzymes:enzyme}{क्रियाधारों} को \emph{बाँधता और अनुस्थापित करता} है ताकि क्रिया करते समूह
सही ज्यामिति में मिलें, और इस तरह किसी दुर्लभ टक्कर की जगह एक निश्चित
टक्कर आ जाए; वह \omterm{def:b1:enzymes:enzyme}{क्रियाधार} को \omterm{prop:b1:enzymes:whatitdoes}{संक्रमण अवस्था} की ओर \emph{तानता} है
(\emph{प्रेरित अनुरूपण}\index{प्रेरित अनुरूपण}: \omterm{def:b1:enzymes:enzyme}{एंजाइम} \omterm{def:b1:enzymes:enzyme}{क्रियाधार} के
चारों ओर सिमट जाता है, और अनुरूपण \omterm{prop:b1:enzymes:whatitdoes}{संक्रमण अवस्था} के लिए सबसे अच्छा है);
उसकी \omterm{def:b1:proteins:aminoacid}{पार्श्व शृंखलाएँ} \emph{अम्ल और क्षारक} की तरह काम करती हैं, और सही
क्षण पर \omterm{def:b1:enzymes:enzyme}{क्रियाधार} को प्रोटॉन देती-लेती हैं (हिस्टिडीन, जिसका p$K_a$ 7 के
पास है, अनेक \omterm{def:b1:enzymes:enzyme}{एंजाइमों} में यही करता है); कुछ \omterm{def:b1:enzymes:enzyme}{क्रियाधार} से एक क्षणिक
\emph{\omterm{def:b1:water-small-molecules:bonds}{सहसंयोजी बंध}} बनाते हैं (सेरीन प्रोटिएज़); और उसके धातु आयन बंधों
को ध्रुवित करते हैं तथा आवेशों को स्थायी करते हैं। जल उस स्थल से बड़े
पैमाने पर बाहर रखा जाता है, ताकि आवेश और \omterm{def:b1:water-small-molecules:water}{हाइड्रोजन बंध} पूरी शक्ति से काम
करें।
\end{proposition}

\begin{example}[तीन एंजाइम]\label{ex:b1:enzymes:three}
लाइसोज़ाइम, जो आँसुओं और अंडे के सफ़ेद भाग में है, जीवाणु भित्ति के किसी
शर्करा वलय को \omterm{prop:b1:enzymes:whatitdoes}{संक्रमण अवस्था} की आकृति में तान देता है और शृंखला काट देता
है: $10^8$ गुना त्वरण। कार्बोनिक ऐनहाइड्रेज़, जो लाल \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में है, एक
ज़िंक आयन थामे है जो जल को ऐसा हाइड्रॉक्साइड बना देता है जो
$\mathrm{CO_2}$ पर हमला करने को तैयार है: प्रति सेकंड दस लाख अणु, यानी
कैटेलेज़ के बाद सबसे तेज़ \omterm{def:b1:enzymes:enzyme}{एंजाइम}। काइमोट्रिप्सिन, जो अग्न्याशयी रस में
है, एक हिस्टिडीन से क्रियाशील बनाई गई सेरीन से ऐरोमैटिक अवशेषों के बाद
\omterm{def:b1:proteins:peptide}{प्रोटीन} काटता है: प्रति सेकंड सौ, और एक सहसंयोजी मध्यवर्ती के साथ।
\end{example}

\section{एंजाइम बलगतिकी}

\begin{definition}[आरंभिक दर, संतृप्ति]\label{def:b1:enzymes:rate}
किसी \omterm{def:b1:enzymes:enzyme}{एंजाइम} अभिक्रिया की \emph{दर}\index{अभिक्रिया दर} $v$ प्रति एकांक
समय बने उत्पाद की मात्रा है (\unit{mol/s}, या \omterm{def:b1:enzymes:enzyme}{एंजाइम} के मात्रक:
\qty{1}{\micro mol/min})। उसे \emph{आरंभिक
दर} $v_0$ के रूप में मापा जाता है, यानी \omterm{def:b1:enzymes:enzyme}{क्रियाधार} चुकने या उत्पाद जमा
होने से पहले। किसी नियत \omterm{def:b1:enzymes:enzyme}{एंजाइम} सांद्रता पर $v_0$ \omterm{def:b1:enzymes:enzyme}{क्रियाधार} की सांद्रता
$[S]$ के साथ बढ़ता है और फिर किसी अधिकतम $V_{\max}$ पर
\emph{संतृप्त} हो जाता है: \omterm{def:b1:enzymes:enzyme}{एंजाइम} पूरा घिर चुका है और इससे तेज़ नहीं जा
सकता।
\end{definition}

\begin{theorem}[माइकेलिस--मेंटन]\label{thm:b1:enzymes:mm}
\index{माइकेलिस--मेंटन समीकरण}\index{माइकेलिस स्थिरांक}
किसी \omterm{def:b1:enzymes:enzyme}{एंजाइम} $E$ के लिए जो अपने \omterm{def:b1:enzymes:enzyme}{क्रियाधार} को उत्क्रमणीय रूप से बाँधता और बदलता है,
\[
E + S \underset{k_{-1}}{\overset{k_1}{\rightleftharpoons}} ES
\overset{k_{\mathrm{cat}}}{\longrightarrow} E + P ,
\]
\omterm{def:b1:enzymes:enzyme}{क्रियाधार} सांद्रता $[S]$ पर आरंभिक दर यह है
\[
v_0 = \frac{V_{\max}\,[S]}{K_m + [S]}, \qquad V_{\max} = k_{\mathrm{cat}}\,[E]_{\text{कुल}},
\qquad K_m = \frac{k_{-1} + k_{\mathrm{cat}}}{k_1} .
\]
$K_m$, यानी \emph{\omterm{thm:b1:enzymes:mm}{माइकेलिस स्थिरांक}}, वह \omterm{def:b1:enzymes:enzyme}{क्रियाधार} सांद्रता है
जिस पर दर अपनी अधिकतम की आधी होती है, और वह मापता है कि \omterm{def:b1:enzymes:enzyme}{एंजाइम} को कितना
\omterm{def:b1:enzymes:enzyme}{क्रियाधार} चाहिए; $k_{\mathrm{cat}}$, यानी \emph{परिवर्तन
संख्या}\index{परिवर्तन संख्या}, संतृप्त होने पर एक \omterm{def:b1:enzymes:enzyme}{एंजाइम} प्रति सेकंड
जितने \omterm{def:b1:enzymes:enzyme}{क्रियाधार} अणु बदलता है वह संख्या है; और अनुपात
$k_{\mathrm{cat}}/K_m$ \emph{उत्प्रेरक दक्षता} है, यानी कम \omterm{def:b1:enzymes:enzyme}{क्रियाधार} पर दर
स्थिरांक, जिसकी ऊपरी सीमा वह दर है जिससे \omterm{def:b1:enzymes:enzyme}{क्रियाधार} विसरण द्वारा \omterm{def:b1:enzymes:enzyme}{एंजाइम}
तक पहुँच सकता है, लगभग
$10^8$--$10^9$ \unit{L.mol^{-1}.s^{-1}}।
\end{theorem}
\begin{proof}
ऐसी \emph{स्थायी अवस्था} मानिए जिसमें $ES$ उतनी ही तेज़ी से बनता है
जितनी तेज़ी से लुप्त होता है (जो पहले कुछ मिलीसेकंडों के बाद, जब तक
$[S] \gg [E]$ है, सही है):
$k_1[E][S] = (k_{-1} + k_{\mathrm{cat}})[ES]$, इसलिए $[E][S] = K_m[ES]$
जिसमें $K_m$ ऊपर की परिभाषा वाला है। संरक्षण से $[E] = [E]_{\text{कुल}} -
[ES]$ मिलता है; और यह रखने पर $([E]_{\text{कुल}} - [ES])[S] = K_m[ES]$,
इसलिए $[ES] = [E]_{\text{कुल}}[S]/(K_m + [S])$। दर
$v_0 = k_{\mathrm{cat}}[ES]$ है, जिससे सूत्र निकल आता है; $[S] \gg K_m$ पर
$v_0 \to k_{\mathrm{cat}}[E]_{\text{कुल}} = V_{\max}$, और $[S] = K_m$ पर
$v_0 = V_{\max}/2$।
\end{proof}

\begin{method}[$K_m$ और $V_{\max}$ मापना]\label{met:b1:enzymes:lb}
\begin{enumerate}
  \item $K_m/4$ से $10K_m$ तक फैली \omterm{def:b1:enzymes:enzyme}{क्रियाधार} सांद्रताओं की एक शृंखला
        बनाइए, और \omterm{def:b1:enzymes:enzyme}{एंजाइम} की सांद्रता वही रखिए; पहले मिनट तक उत्पाद
        (रंग, अवशोषकता, गैस) का पीछा कीजिए और प्रवणता को $v_0$ मानिए।
  \item $v_0$ को $[S]$ के सापेक्ष आलेखित कीजिए: अतिपरवलय एक मोटा
        $V_{\max}$ (पठार) और $K_m$ (उसकी आधी पर की सांद्रता) दे देता
        है।
  \item परिशुद्धता के लिए उलटिए: $1/v_0 = (K_m/V_{\max})(1/[S]) +
        1/V_{\max}$ (\emph{लाइनवीवर--बर्क})। रेखा का $y$ अक्ष पर
        अंतःखंड $1/V_{\max}$ है, $x$ अक्ष पर $-1/K_m$, और उसकी
        प्रवणता $K_m/V_{\max}$।
  \item $V_{\max}$ को \omterm{def:b1:enzymes:enzyme}{एंजाइम} की सांद्रता से भाग दीजिए और
        $k_{\mathrm{cat}}$ पाइए; और दक्षता के लिए $K_m$ से भाग दीजिए। किसी \omterm{def:b1:enzymes:inhibitors}{संदमक} के साथ यही
        दोहराइए ताकि पता चले कि वह कौन-सा प्राचल बदलता है।
\end{enumerate}
\end{method}

\section{संदमन}

\begin{definition}[संदमक]\label{def:b1:enzymes:inhibitors}
कोई \emph{संदमक}\index{एंजाइम संदमक} किसी \omterm{def:b1:enzymes:enzyme}{एंजाइम} की दर घटाता है।
\emph{अनुत्क्रमणीय} संदमक सहसंयोजी रूप से बँधते हैं और \omterm{def:b1:enzymes:enzyme}{एंजाइम} की
क्रियाशीलता सदा के लिए नष्ट कर देते हैं (ऐसीटिलकोलीनएस्टरेज़ पर तंत्रिका
गैस, जीवाणु भित्ति को आपस में बाँधने वाले \omterm{def:b1:enzymes:enzyme}{एंजाइम} पर पेनिसिलिन,
प्रोस्टाग्लैंडिन बनाने वाले \omterm{def:b1:enzymes:enzyme}{एंजाइम} पर ऐस्पिरिन)। \emph{उत्क्रमणीय} संदमक
दुर्बल बंधों से बँधते हैं और धोकर हटाए जा सकते हैं; और वे कहाँ बँधते हैं
इसके अनुसार:
\begin{itemize}
  \item \emph{स्पर्धी}\index{स्पर्धी संदमन}: संदमक \omterm{def:b1:enzymes:enzyme}{क्रियाधार} जैसा दिखता
        है और \omterm{def:b1:enzymes:enzyme}{सक्रिय स्थल} घेर लेता है; वह आभासी $K_m$ बढ़ा देता है
        (गुणक $1 + [I]/K_i$ से) और $V_{\max}$ अपरिवर्तित छोड़ देता है
        --- पर्याप्त \omterm{def:b1:enzymes:enzyme}{क्रियाधार} उसे हटा देता है;
  \item \emph{अस्पर्धी}: संदमक कहीं और बँधता है और उत्प्रेरण बिगाड़
        देता है, चाहे \omterm{def:b1:enzymes:enzyme}{क्रियाधार} बँधा हो या नहीं; वह $V_{\max}$ घटा देता
        है और $K_m$ अपरिवर्तित छोड़ देता है --- कितना भी
        \omterm{def:b1:enzymes:enzyme}{क्रियाधार} उस पर भारी नहीं पड़ता;
  \item \emph{अप्रतिस्पर्धी}: संदमक केवल $ES$ संकुल से बँधता है;
        $V_{\max}$ और $K_m$ दोनों घट जाते हैं।
\end{itemize}
$K_i$, यानी संदमक का वियोजन स्थिरांक, उसकी प्रबलता मापता है: जितना कम,
उतना प्रबल।
\end{definition}

\section{नियमन}

\begin{definition}[अन्यस्थानिक एंजाइम]\label{def:b1:enzymes:allosteric}
किसी \emph{अन्यस्थानिक एंजाइम}\index{अन्यस्थानिक एंजाइम} में कई
उपइकाइयाँ होती हैं और, अपने \omterm{def:b1:enzymes:enzyme}{सक्रिय स्थलों} के अलावा, \emph{नियामक स्थल}
भी, जहाँ प्रभावक बँधते हैं: \emph{सक्रियक} उसे उसके सक्रिय संरूपण की ओर
खिसकाते हैं, और \emph{\omterm{def:b1:enzymes:inhibitors}{संदमक}} निष्क्रिय संरूपण की ओर। $[S]$ के सापेक्ष
उसकी दर अंग्रेज़ी S जैसी होती है, अतिपरवलयी नहीं (\omterm{def:b1:enzymes:enzyme}{सक्रिय स्थलों} के बीच
\omterm{def:b1:proteins:allostery}{सहकारिता}, जैसे हीमोग्लोबिन में, \cref{ch:b1:proteins}), इसलिए तीखे भाग
के पास \omterm{def:b1:enzymes:enzyme}{क्रियाधार} में छोटा-सा परिवर्तन भी दर बहुत बदल देता है; और प्रभावक
वक्र को अगल-बगल खिसकाते हैं (चाहिए क्रियाधार-सांद्रता बदलकर) या ऊपर-नीचे
(अधिकतम दर बदलकर)। \omterm{def:b1:proteins:allostery}{अन्यस्थानिक} \omterm{def:b1:enzymes:enzyme}{एंजाइम} उपापचय के शाखा-बिंदुओं पर खड़े होते
हैं, और प्रभावक स्वयं उस पथ के उत्पाद तथा \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के ऊर्जा-संकेत होते हैं
(\omterm{def:b1:water-small-molecules:atp}{ATP}, ADP, AMP)।
\end{definition}

\begin{proposition}[कोशिका किसी एंजाइम को नियंत्रित करने के चार ढंग]\label{prop:b1:enzymes:control}
\begin{enumerate}
  \item \emph{प्रतिपुष्टि संदमन}\index{प्रतिपुष्टि संदमन}: किसी पथ का
        अंतिम उत्पाद उसके पहले प्रतिबद्ध \omterm{def:b1:enzymes:enzyme}{एंजाइम} का \omterm{def:b1:proteins:allostery}{अन्यस्थानिक} \omterm{def:b1:enzymes:inhibitors}{संदमक}
        होता है, इसलिए पथ उतना ही तेज़ चलता है जितनी तेज़ी से उत्पाद
        काम में आता है (थ्रिओनीन डिऐमिनेज़ पर आइसोल्यूसीन; फ़ॉस्फ़ो-फ्रक्टोकाइनेज़
        पर \omterm{def:b1:water-small-molecules:atp}{ATP}, \cref{ch:b1:respiration-fermentation})।
        अनुक्रिया-काल: मिलीसेकंड।
  \item \emph{सहसंयोजी रूपांतरण}\index{सहसंयोजी रूपांतरण}:
        कोई काइनेज़ \omterm{def:b1:enzymes:enzyme}{एंजाइम} के किसी सेरीन, थ्रिओनीन या टाइरोसीन पर एक
        फ़ॉस्फ़ेट लगाता है, कोई फ़ॉस्फ़ेटेज़ उसे हटाता है, और दोनों रूप
        क्रियाशीलता में भिन्न होते हैं (एक ही हॉर्मोनी संकेत से \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन}
        फ़ॉस्फ़ोरिलेज़ फ़ॉस्फ़ोरिलीकरण से चालू होती है और \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन}
        सिंथेज़ बंद)। सेकंडों से मिनटों में; उत्क्रमणीय; और शृंखलाओं में
        प्रवर्धनीय।
  \item \emph{प्रोटीनअपघटनी सक्रियण}: कुछ \omterm{def:b1:enzymes:enzyme}{एंजाइम} निष्क्रिय पूर्ववर्तियों
        के रूप में बनते हैं (\emph{ज़ाइमोजन}: ट्रिप्सिनोजन,
        पेप्सिनोजन, स्कंदन कारक) और एक पेप्टाइड काटकर, एक ही बार और सदा
        के लिए, वहीं और तभी चालू किए जाते हैं जहाँ और जब वे चाहिए।
  \item \emph{मात्रा}: \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} उस \omterm{def:b1:enzymes:enzyme}{एंजाइम} का जीन नियंत्रित करके
        (\cref{ch:b1:expression-control}) उसे अधिक या कम बनाती है और
        उसे तेज़ या धीमे तोड़ती है। मिनटों से घंटों में।
\end{enumerate}
\end{proposition}

\begin{example}[सम-एंजाइम]\label{ex:b1:enzymes:isoenzymes}
लैक्टेट डिहाइड्रोजनेज़ पाँच रूपों में मिलती है, जो दो प्रकार की
उपइकाइयों के सारे संयोजनों के चतुष्क हैं; हृदय के रूप का लैक्टेट के लिए
$K_m$ कम है और उसे पाइरुवेट संदमित करता है (वह क्रेब्स चक्र को खिलाने के
लिए लैक्टेट का ऑक्सीकरण करता है), जबकि पेशी के रूप का $V_{\max}$ ऊँचा है
और वह पाइरुवेट सह लेता है (वह किसी दौड़ में लैक्टेट बनाता है)। रक्त में
रूपों का प्रतिरूप बता देता है कि कौन-सा \omterm{def:b1:mammal-organization:organ}{अंग} क्षतिग्रस्त हुआ है --- कोई
हृदयाघात हृदय का रूप छोड़ता है।
\end{example}

\section{ताप और pH}

\begin{proposition}[एंजाइम और उनका परिवेश]\label{prop:b1:enzymes:environment}
किसी \omterm{def:b1:enzymes:enzyme}{एंजाइम} अभिक्रिया की दर हर \qty{10}{\celsius} पर मोटे तौर पर
दुगुनी हो जाती है ($Q_{10} \approx 2$), क्योंकि ऊष्मीय ऊर्जा अधिक अणुओं
को अवरोध के पार ले जाती है, जब तक \omterm{def:b1:enzymes:enzyme}{एंजाइम} खुलना शुरू न कर दे
(अधिकांश के लिए \qtyrange{40}{60}{\celsius}; गरम झरनों के जीवाणुओं के
\omterm{def:b1:enzymes:enzyme}{एंजाइमों} के लिए \qty{100}{\celsius}) और दर ढह न जाए। हर \omterm{def:b1:enzymes:enzyme}{एंजाइम} का एक
\emph{pH इष्टतम} होता है, जहाँ उसके उत्प्रेरक अवशेषों तथा उसके \omterm{def:b1:enzymes:enzyme}{क्रियाधार}
का आयनीकरण ठीक हो: आमाशय में पेप्सिन pH 2 के पास;
आँत में ट्रिप्सिन 8 के पास; और अधिकांश अंतःकोशिकीय \omterm{def:b1:enzymes:enzyme}{एंजाइम} 7 के पास।
इष्टतम से दूर दर गिरती है, और उससे बहुत दूर \omterm{def:b1:proteins:peptide}{प्रोटीन} विकृत हो जाता है।
ताप और pH \omterm{def:b1:proteins:peptide}{प्रोटीन} पर काम करते हैं; और \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} दोनों को उसी परास में रखती
है जहाँ उसके हज़ार \omterm{def:b1:enzymes:enzyme}{एंजाइम} सब काम करते हैं।
\end{proposition}

\begin{example}[एक ज्वर और एक गरम झरना]\label{ex:b1:enzymes:fever}
\qty{40}{\celsius} का ज्वर शरीर की हर अभिक्रिया को एक-चौथाई तेज़ कर देता
है और सबसे नाज़ुक \omterm{def:b1:proteins:peptide}{प्रोटीनों} को खोलना शुरू कर देता है;
\qty{42}{\celsius} पर मस्तिष्क के \omterm{def:b1:proteins:peptide}{प्रोटीन} चूक जाते हैं। \qty{75}{\celsius}
के किसी झरने के जीवाणु की \omterm{def:b1:nucleic-acids:chain}{DNA} पॉलिमरेज़ \qty{95}{\celsius} सह लेती है,
और इसीलिए उसे \omterm{def:b1:nucleic-acids:chain}{DNA} के पिघलने के हज़ारों चक्रों से गुज़ारा जा सकता है
और वह पॉलिमरेज़ शृंखला अभिक्रिया का \omterm{def:b1:enzymes:enzyme}{एंजाइम} बन गई (वर्ष 3 का
खंड): तापस्थायित्व अनुक्रम का गुण है, उत्प्रेरित रसायन का नहीं।
\end{example}

\end{exercise}

\end{exercise}

\section{समस्या: एक मापा गया एंजाइम}

\begin{problem}[{सप्ताहांत समस्या --- छह \omterm{def:b1:enzymes:enzyme}{क्रियाधार} सांद्रताओं पर
आमापित एक एस्टरेज़, अकेले और दो \omterm{def:b1:enzymes:inhibitors}{संदमकों} के साथ, उसके स्थिरांक निकाले गए
और \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में उसका व्यवहार आँका गया, और अंत में उसका $K_m$,
$V_{\max}$ तथा $K_i$}]
\label{pb:b1:enzymes:1}
किसी एस्टरेज़ का \qty{25}{\celsius}, pH 7.5 पर \qty{1.0}{mL} में
\qty{10}{nmol/L} \omterm{def:b1:enzymes:enzyme}{एंजाइम} के साथ आमापन किया जाता है। छह \omterm{def:b1:enzymes:enzyme}{क्रियाधार}
सांद्रताओं पर आरंभिक दरें (\unit{\micro mol/min}), अकेले और
\qty{1.0}{mmol/L} \omterm{def:b1:enzymes:inhibitors}{संदमक} I या \qty{1.0}{mmol/L} \omterm{def:b1:enzymes:inhibitors}{संदमक} J की उपस्थिति में:

\begin{center}
\small
\begin{tabular}{lcccccc}
\hline
$[S]$ (\unit{mmol/L}) & 0.5 & 1 & 2 & 5 & 10 & 20 \\
\hline
$v_0$, कोई \omterm{def:b1:enzymes:inhibitors}{संदमक} नहीं & 0.200 & 0.333 & 0.500 & 0.714 & 0.833 & 0.909 \\
$v_0$, I के साथ & 0.111 & 0.200 & 0.333 & 0.556 & 0.714 & 0.833 \\
$v_0$, J के साथ & 0.100 & 0.167 & 0.250 & 0.357 & 0.417 & 0.455 \\
\hline
\end{tabular}
\end{center}

\medskip
\noindent\textbf{भाग I --- अकेला \omterm{def:b1:enzymes:enzyme}{एंजाइम}।}
\begin{enumerate}
  \item बिना \omterm{def:b1:enzymes:inhibitors}{संदमक} वाले छह बिंदुओं के लिए $1/[S]$ और $1/v_0$
        निकालिए।
  \item दिखाइए कि ये बिंदु एक सरल रेखा पर हैं और उसकी प्रवणता तथा
        अंतःखंड निकालिए।
  \item $V_{\max}$ और $K_m$ निकालिए।
  \item \omterm{thm:b1:enzymes:mm}{माइकेलिस--मेंटन समीकरण} से जाँचिए, $[S] =
        \qty{5}{mmol/L}$ पर।
  \item आमापन में \omterm{def:b1:enzymes:enzyme}{एंजाइम} की मात्रा (मोल) और \unit{s^{-1}} में
        $k_{\mathrm{cat}}$ निकालिए।
  \item $k_{\mathrm{cat}}/K_m$ निकालिए और विसरण की सीमा से तुलना
        कीजिए।
  \item किस \omterm{def:b1:enzymes:enzyme}{क्रियाधार} सांद्रता पर \omterm{def:b1:enzymes:enzyme}{एंजाइम} $V_{\max}$ के \qty{95}{\%} पर
        काम कर रहा है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{def:b1:enzymes:inhibitors}{संदमक} I।}
\begin{enumerate}[resume]
  \item I के साथ छह बिंदुओं के लिए $1/v_0$ निकालिए और उनकी रेखा की
        प्रवणता तथा अंतःखंड निकालिए।
  \item I के साथ आभासी $V_{\max}$ और $K_m$ निकालिए।
  \item कौन-सा प्राचल बदला? \omterm{def:b1:enzymes:inhibitors}{संदमक} का वर्गीकरण कीजिए।
  \item $K_i$ निकालिए, $K_m^{\text{आभासी}} = K_m(1 + [I]/K_i)$ से।
  \item $[S] = \qty{2}{mmol/L}$ पर I की कौन-सी सांद्रता दर आधी कर देती
        है? और $[S] = \qty{20}{mmol/L}$ पर?
  \item सुझाइए कि I संरचना में क्या हो सकता है और वह कहाँ बँधता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:b1:enzymes:inhibitors}{संदमक} J।}
\begin{enumerate}[resume]
  \item J के साथ रेखा की प्रवणता तथा अंतःखंड निकालिए और आभासी
        $V_{\max}$ तथा $K_m$ निकालिए।
  \item J का वर्गीकरण कीजिए और उसका $K_i$ निकालिए,
        $V_{\max}^{\text{आभासी}} =
        V_{\max}/(1 + [J]/K_i)$ से।
  \item समझाइए कि \omterm{def:b1:enzymes:enzyme}{क्रियाधार} बढ़ाकर J पर काबू क्यों नहीं पाया जा सकता।
  \item J की उपस्थिति में \omterm{def:b1:enzymes:enzyme}{एंजाइम} की सांद्रता दुगुनी करने से दर का क्या
        होता है? और I की उपस्थिति में?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में।}
\omterm{def:b1:cell-unit-of-life:cell}{कोशिका} \omterm{def:b1:enzymes:enzyme}{क्रियाधार} को \qty{0.5}{mmol/L} पर और \omterm{def:b1:enzymes:enzyme}{एंजाइम} को
\qty{10}{nmol/L} पर, \qty{1000}{\micro m^3} के किसी आयतन में,
\qty{37}{\celsius} पर रखती है, और $Q_{10} = 2$ है।
\begin{enumerate}[resume]
  \item \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में दर निकालिए (पूरी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में प्रति सेकंड उत्पाद के
        अणु)।
  \item उत्पाद प्रति सेकंड \qty{3e6}{} अणुओं की दर से चाहिए।
        \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को दर कितने गुणक से बढ़ानी पड़ेगी, और वह ऐसा कैसे कर
        सकती है, इसके दो ढंग बताइए।
  \item उत्पाद $K_i = \qty{0.2}{mmol/L}$ वाला उस \omterm{def:b1:enzymes:enzyme}{एंजाइम} का स्पर्धी
        \omterm{def:b1:enzymes:inhibitors}{संदमक} है, और वह जमा होकर \qty{0.4}{mmol/L} हो जाता है। तब की दर
        निकालिए, और टिप्पणी कीजिए कि इससे क्या सधता है।
  \item \omterm{def:b1:enzymes:enzyme}{एंजाइम} को कोई काइनेज़ फ़ॉस्फ़ोरिलित कर देता है, जिससे उसका
        $K_m$ \qty{0.4}{mmol/L} हो जाता है। \qty{0.5}{mmol/L} \omterm{def:b1:enzymes:enzyme}{क्रियाधार}
        पर पहले और बाद की दर निकालिए।
  \item \omterm{def:b1:enzymes:enzyme}{एंजाइम} का pH इष्टतम 7.5 है और उसकी क्रियाशीलता pH
        6.5 पर आधी हो जाती है। कोई \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकाय} pH 5 पर है। वहाँ उसकी
        क्रियाशीलता की गुणात्मक भविष्यवाणी कीजिए और आयनीकरणीय अवशेषों
        से समझाइए।
  \item कोई उत्परिवर्तन \omterm{def:b1:enzymes:enzyme}{सक्रिय स्थल} के हिस्टिडीन की जगह ऐलानीन रख देता
        है। $K_m$ पर और $k_{\mathrm{cat}}$ पर पड़ते प्रभाव की, हर एक के
        कारण सहित, भविष्यवाणी कीजिए।
  \item समझाइए कि \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} \omterm{def:b1:enzymes:enzyme}{क्रियाधार} को $K_m$ से बहुत ऊपर रखने के बजाय
        उसके पास क्यों रखती है।
  \item परिणाम बताइए: एस्टरेज़ का $K_m$, $V_{\max}$, $k_{\mathrm{cat}}$,
        और हर \omterm{def:b1:enzymes:inhibitors}{संदमक} का प्रकार तथा $K_i$।
\end{enumerate}
\end{problem}
