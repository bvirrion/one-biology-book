\section*{Hoofdstuk \ref{ch:g2:animals-grow-up} --- Hoe dieren opgroeien}

\begin{solution}{exo:g2:animals-grow-up:1}
Uit een \omterm{def:g2:animals-grow-up:born}{ei} dat de moeder legt (een kuiken, een kikker, een slak), of
\omterm{def:g2:animals-grow-up:born}{levend geboren} uit het lichaam van de moeder en drinkend van haar \omterm{def:g2:animals-grow-up:born}{melk}
(een kitten, een kalf, een mensenbaby).
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:2}
(a) Uit een \omterm{def:g2:animals-grow-up:born}{ei}. (b) \omterm{def:g2:animals-grow-up:born}{Levend geboren}. (c) Uit een \omterm{def:g2:animals-grow-up:born}{ei}. (d) \omterm{def:g2:animals-grow-up:born}{Levend geboren}.
(e) Uit een \omterm{def:g2:animals-grow-up:born}{ei}.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:3}
De \omterm{def:g2:animals-grow-up:born}{melk} van zijn moeder. Alle \omterm{def:g1:animals-around-us:animal}{dieren} uit dit hoofdstuk die \omterm{def:g2:animals-grow-up:born}{levend
geboren} worden --- puppy's, kalveren, jonge konijnen, mensenbaby's ---
beginnen ook met \omterm{def:g2:animals-grow-up:born}{melk}.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:4}
\omterm{def:g2:animals-grow-up:metamorphosis}{Gedaanteverwisseling} is de grote verandering van lichaamsvorm die
sommige jonge \omterm{def:g1:animals-around-us:animal}{dieren} doormaken om volwassen te worden. De donderpad die
een kikker wordt, en de \omterm{ex:g2:animals-grow-up:butterfly}{rups} die een vlinder wordt.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:5}
Eitjes, donderpad, donderpad met poten, kikker.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:6}
\omterm{def:g2:animals-grow-up:born}{Ei}, \omterm{ex:g2:animals-grow-up:butterfly}{rups}, \omterm{ex:g2:animals-grow-up:butterfly}{pop}, vlinder. Het is de \omterm{ex:g2:animals-grow-up:butterfly}{rups} die \omterm{def:g1:plants-around-us:parts}{bladeren} eet en bijna al het
groeien doet.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:7}
Het veulen --- vanaf zijn eerste dag een klein paard. Voor hem betekent
opgroeien groter en sterker worden met dezelfde vorm.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:8}
Om ze warm te houden. In elk warm \omterm{def:g2:animals-grow-up:born}{ei} wordt een kuiken gebouwd, gevoed
door de dooier, tot het klaar is om de schaal te breken en uit te komen.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:9}
Het is een donderpad halverwege de \omterm{def:g2:animals-grow-up:metamorphosis}{gedaanteverwisseling}: de
achterpoten zijn er, de voorpoten nog niet. Daarvoor was het een
donderpad zonder poten (en daarvóór een eitje); hierna komen de
voorpoten, krimpt de \omterm{def:g1:animals-around-us:parts}{staart}, en wordt het een kikkertje.
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:10}
Er is niemand in getrokken --- de \omterm{ex:g2:animals-grow-up:butterfly}{rups} en de vlinder zijn hetzelfde
\omterm{def:g1:animals-around-us:animal}{dier}. In de \omterm{ex:g2:animals-grow-up:butterfly}{pop} werd het lichaam van de \omterm{ex:g2:animals-grow-up:butterfly}{rups} omgebouwd tot dat van een
vlinder: dat ombouwen is precies wat \omterm{def:g2:animals-grow-up:metamorphosis}{gedaanteverwisseling} betekent.
\end{solution}
