\chapter{Les microbes et l'infection}\label{ch:g9:microbes-and-infection}

Ils accompagnent ce livre depuis les \omterm{def:g6:producing-our-food:fermentation}{microbes} invisibles de
l'année 1 (\cref{def:g1:healthy-habits:germs})~: promus
main-d'œuvre à la boulangerie
(\cref{ch:g6:producing-our-food}), équipes de \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs}
dans le \omterm{def:g6:decomposers-and-soil:soil}{sol}, adversaire qui évolue dans le service
hospitalier (\cref{ex:g9:how-species-change:resistance}). Ce
chapitre donne enfin aux \omterm{def:g6:producing-our-food:fermentation}{microbes} leur compte rendu
complet~: ce qu'ils sont, comment la minorité nuisible
attaque, et comment les défenses extérieures du corps --- et
les nôtres --- tiennent les portes.

\section{Le monde microbien}

\begin{definition}[Les microbes, triés]\label{def:g9:microbes-and-infection:sorted}
Les \emph{micro-organismes}\index{micro-organisme} --- les
\omterm{def:g6:producing-our-food:fermentation}{microbes} --- sont des \omterm{def:g6:exploring-our-environment:components}{êtres vivants} trop petits pour être
vus à l'\omterm{def:g1:the-five-senses:sense}{œil} nu. Les principales sortes~:
\begin{itemize}
  \item les \emph{bactéries}\index{bactérie}~: des \omterm{def:g5:first-look-at-cells:cell}{cellules}
        uniques sans véritable \omterm{def:g6:cell-unit-of-life:parts}{noyau}, bien plus petites que
        les tiennes, qui se divisent vite (le monde de
        \cref{def:g6:cell-unit-of-life:unicellular}) ---
        partout, en nombres qui dépassent tout compte~;
  \item les \emph{\omterm{def:g6:decomposers-and-soil:decomposers}{champignons} \omterm{def:g6:cell-unit-of-life:unicellular}{unicellulaires} et autres}~:
        les \omterm{ex:g6:producing-our-food:bread}{levures} (\cref{ex:g6:producing-our-food:bread}),
        les \omterm{def:g4:plants-without-seeds:spore}{spores} des moisissures, les nageurs de la goutte
        de mare~;
  \item les \emph{virus}\index{virus}~: plus petits encore,
        et différents par nature --- un fragment de texte
        génétique emballé (les lettres de
        \cref{def:g9:genes-and-dna:dna}) dans un manteau,
        \emph{sans \omterm{def:g5:first-look-at-cells:cell}{cellule} et sans vie propre}~: un virus ne
        fait rien tant qu'il n'est pas entré dans une
        \omterm{def:g5:first-look-at-cells:cell}{cellule} vivante et n'a pas tourné la machinerie de
        lecture de cette \omterm{def:g5:first-look-at-cells:cell}{cellule} vers la copie du texte de
        l'intrus. Au regard des signes de
        \cref{prop:g1:living-or-not:signs}, les virus se
        tiennent tout au bord de la vie.
\end{itemize}
\end{definition}

\begin{proposition}[Le plus souvent inoffensifs, souvent indispensables]\label{prop:g9:microbes-and-infection:mostly}
La réputation de nuisibles est un rapport de minorité. La
plupart des \omterm{def:g6:producing-our-food:fermentation}{microbes} nous ignorent~; beaucoup nous
servent~: les équipes de \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs}
(\cref{def:g6:decomposers-and-soil:decomposers}), les
main-d'œuvre de l'alimentation, les chimistes du \omterm{def:g6:decomposers-and-soil:soil}{sol} --- et
ta propre \emph{flore résidente}\index{flore résidente}~:
les milliers de milliards de \omterm{def:g9:microbes-and-infection:sorted}{bactéries} inoffensives qui
tapissent la \omterm{def:g1:the-five-senses:sense}{peau} et l'intestin, aident la \omterm{def:g4:journey-of-food:digestion}{digestion} et,
par simple occupation, chassent les intrus du terrain.
Seule une petite minorité --- les \emph{agents
pathogènes}\index{agent pathogène} --- provoque des
maladies.
\end{proposition}
\admitted

\begin{example}[Un seul corps, trois relations]\label{ex:g9:microbes-and-infection:relations}
Sur ta \omterm{def:g1:the-five-senses:sense}{peau} en ce moment~: des résidents par milliards, qui
tiennent la surface (relation une~: l'alliance). Dans ton
yaourt du petit-déjeuner~: des ouvriers, qui acidifient sur
commande (relation deux~: l'emploi,
\cref{ex:g6:producing-our-food:yogurt}). Sur le clou
rouillé de la remise~: des \omterm{def:g4:plants-without-seeds:spore}{spores} de la \omterm{def:g9:microbes-and-infection:sorted}{bactérie} du tétanos,
qui attendent une coupure profonde (relation trois~: la
minorité pathogène, et la raison d'être de
\cref{ch:g9:immune-defenses} et du rappel du tétanos).
\end{example}

\section{L'infection~: l'attaque, mise en scène}

\begin{definition}[Les étapes de l'infection]\label{def:g9:microbes-and-infection:stages}
L'\emph{infection}\index{infection} se déroule en étapes,
chacune avec son nom~:
\begin{enumerate}
  \item la \emph{transmission}~: l'\omterm{prop:g9:microbes-and-infection:mostly}{agent pathogène} voyage
        --- surfaces touchées et mains, gouttelettes de
        toux, aliments et eau contaminés, \omterm{prop:g4:journey-of-food:absorb}{sang}, et la voie
        sexuelle que signalait
        \cref{def:g8:contraception:condom}~;
  \item l'\emph{entrée}~: par une brèche --- \omterm{def:g1:the-five-senses:sense}{peau} coupée,
        ou les portes ouvertes du corps~: voies aériennes,
        intestin, et les voies ci-dessus~;
  \item la \emph{multiplication}~: au chaud, au mouillé et
        nourris, les arrivants se divisent --- les
        \omterm{def:g9:microbes-and-infection:sorted}{bactéries} selon la règle des \omterm{def:g6:exploring-our-environment:conditions}{conditions} de
        \cref{prop:g6:decomposers-and-soil:speed}, quelques
        heures doublant leur nombre~; les \omterm{def:g9:microbes-and-infection:sorted}{virus} en changeant
        des \omterm{def:g5:first-look-at-cells:cell}{cellules} en ateliers de copie~;
  \item la \emph{maladie}~: les symptômes --- venus des
        dégâts eux-mêmes, des poisons bactériens, et en
        partie (prochain chapitre) de la bataille menée par
        la défense elle-même.
\end{enumerate}
\end{definition}

\begin{example}[Deux envahisseurs, opposés]\label{ex:g9:microbes-and-infection:contrast}
La \omterm{def:g9:microbes-and-infection:sorted}{bactérie} de la coupure face au \omterm{def:g9:microbes-and-infection:sorted}{virus} du rhume d'hiver.
La \omterm{def:g9:microbes-and-infection:sorted}{bactérie}~: elle entre par la brèche, se multiplie
\emph{entre} les \omterm{def:g5:first-look-at-cells:cell}{cellules} dans le tissu chaud et humide, et
ses déchets empoisonnent le voisinage --- le doigt
infecté qui bat et qui suppure. Le \omterm{def:g9:microbes-and-infection:sorted}{virus}~: il chevauche une
gouttelette jusqu'au revêtement des voies aériennes, entre
dans les \omterm{def:g5:first-look-at-cells:cell}{cellules} de ce revêtement, et chaque \omterm{def:g5:first-look-at-cells:cell}{cellule}
détournée libère une foule de copies avant de céder --- la
gorge à vif et le nez qui coule sur son passage. Des
mécaniques différentes~; l'antibiotique qui affame la
première ne peut rien contre le second, car le second
emprunte \emph{ta} machinerie, et les médicaments contre lui
sont difficiles à viser.
\end{example}

\section{Tenir les portes}

\begin{proposition}[Les premières défenses]\label{prop:g9:microbes-and-infection:doors}
Avant toute bataille, le corps tient ses frontières~:
\begin{itemize}
  \item la \emph{\omterm{def:g1:the-five-senses:sense}{peau}}~: sèche, en couches, qui se
        renouvelle toute seule --- le mur (l'organe de
        \cref{def:g1:the-five-senses:sense}, de garnison)~;
  \item les \emph{revêtements et leurs balais}~: les
        surfaces de nettoyage des voies aériennes
        (\cref{prop:g7:lungs-and-gas-exchange:smoke} les a
        montrées débordées), la chimie qui rince des larmes
        et de la \omterm{def:g4:journey-of-food:tract}{salive}, le bain d'acide de l'\omterm{def:g4:journey-of-food:tract}{estomac}
        (\cref{pb:g7:digestion-and-nutrients:1} stérilise en
        digérant)~;
  \item \emph{les résidents}~: le terrain occupé par la
        flore, peu de place pour un débarquement~;
\end{itemize}
et l'\omterm{def:g1:healthy-habits:hygiene}{hygiène} les prolonge~: le savon de
\cref{met:g1:healthy-habits:hands} (avec désormais tout son
mécanisme --- la transmission interrompue), une eau sûre,
des aliments cuits et gardés au froid (le refus des
\omterm{def:g6:exploring-our-environment:conditions}{conditions} de \cref{rem:g6:producing-our-food:goodbad}),
l'aiguille stérilisée, la barrière du \omterm{def:g8:contraception:condom}{préservatif}. Chaque
règle d'\omterm{def:g1:healthy-habits:hygiene}{hygiène} de ce livre est une étape de transmission ou
d'entrée, bloquée.
\end{proposition}
\admitted

\begin{example}[Les deux médecins qui ont fait bouger les chiffres]\label{ex:g9:microbes-and-infection:history}
Deux noms tiennent cette histoire. Semmelweis, à Vienne
dans les années 1840~: les morts par fièvre puerpérale se
sont effondrées dans sa maternité quand les médecins se sont
lavé les mains entre la salle d'autopsie et
l'accouchement --- transmission interrompue avant que
personne n'ait vu le voyageur. Pasteur, une génération plus
tard, a montré les voyageurs eux-mêmes~: des \omterm{def:g6:producing-our-food:fermentation}{microbes}, et
non un «~mauvais air~», qui aigrissaient le vin, tuaient les
vers à soie, infectaient les plaies --- et la chaleur (sa
\emph{pasteurisation}) ou la stérilisation qui les
arrêtaient. La théorie microbienne a changé les hôpitaux de
dangers en défenses en l'espace d'une vie.
\end{example}

\begin{method}[Passer au crible un risque d'infection]\label{met:g9:microbes-and-infection:audit}
Pour toute situation --- cuisine, plaie, service, foule~:
\begin{enumerate}
  \item nomme l'\omterm{prop:g9:microbes-and-infection:mostly}{agent pathogène} plausible et sa sorte
        (\omterm{def:g9:microbes-and-infection:sorted}{bactérie}, \omterm{def:g9:microbes-and-infection:sorted}{virus}, \omterm{def:g6:decomposers-and-soil:decomposers}{champignon})~;
  \item suis sa voie de transmission jusqu'à toi~;
  \item trouve sa porte d'entrée~;
  \item demande ce qui le multiplie --- chaleur, \omterm{def:g6:exploring-our-environment:conditions}{humidité},
        nourriture, temps~;
  \item bloque l'étape la moins chère~: laver, cuire,
        couvrir, réfrigérer, stériliser, vacciner
        (\cref{ch:g9:immune-defenses} explique la
        dernière).
\end{enumerate}
\end{method}

\section{Problème~: le carnet d'épidémie}

\begin{problem}[{Problème du week-end --- une épidémie
scolaire, suivie étape par étape}]\label{pb:g9:microbes-and-infection:1}
Une épidémie scolaire (fictive)~: en une semaine, la moitié
de la classe de 9B déclare fièvre et toux~; des cas
parsèment ensuite les autres classes. L'infirmière scolaire
tient un carnet d'épidémie et demande à la classe de
\omterm{def:g1:living-or-not:living}{biologie} de l'aider à le lire.

\medskip\noindent\textbf{Partie I --- Lire la propagation.}
\begin{enumerate}
  \item Le premier cas --- le cas index --- était en 9B~;
        les huit suivants partagent sa rangée et ses
        partenaires de tennis de table. Nomme la voie de
        transmission probable, et la porte d'entrée.
  \item Les cas doublent tous les deux jours au début. De
        quelle étape voit-on l'arithmétique, et de qui est
        la machinerie qui fait les copies si l'agent est un
        \omterm{def:g9:microbes-and-infection:sorted}{virus}~?
  \item La liste de contrôle de l'infirmière~: début de la
        fièvre, toux, gorge à vif. Quelle étape les
        symptômes marquent-ils, et de quelles deux sources
        naissent-ils~?
  \item Pourquoi l'épidémie a-t-elle sauté de classe en
        classe exactement lors des deux assemblées de tout
        l'établissement de la semaine~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- La contre-attaque.}
\begin{enumerate}[resume]
  \item Les mesures de l'établissement~: les élèves malades
        chez eux~; les fenêtres ouvertes
        (\cref{exo:g4:breathing:10})~; le lavage des mains
        répété~; le matériel partagé essuyé. Attribue à
        chaque mesure l'étape qu'elle bloque.
  \item Un parent exige des antibiotiques pour tous. Rédige
        la réponse du médecin pour une épidémie virale ---
        et l'avertissement sur la sélection
        (\cref{exo:g9:how-species-change:10}) si l'on
        répandait quand même des antibiotiques.
  \item La cantine signale au même moment deux cas
        gastriques rattachés à un dessert non réfrigéré.
        Passe cette épidémie annexe au crible de
        \cref{met:g9:microbes-and-infection:audit} ---
        autre sorte d'agent, autres étapes.
  \item Pourquoi l'infirmière cartographie-t-elle
        \emph{qui était assis où} au lieu de traiter les
        cas comme des cas au hasard~? Qu'est-ce qu'une
        carte d'épidémie, en termes de transmission~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le bilan.}
\begin{enumerate}[resume]
  \item Écris la biographie de l'épidémie selon les quatre
        étapes, du cas index à l'extinction.
  \item L'épidémie s'est arrêtée avant d'atteindre tous les
        élèves. Propose deux mécanismes --- les blocages
        des mesures, et l'indice du prochain chapitre~:
        ceux qui étaient déjà immunisés par la souche
        cousine de l'an dernier.
  \item Ajoute le paragraphe d'histoire~: de quelles deux
        leçons du dix-neuvième siècle
        (\cref{ex:g9:microbes-and-infection:history})
        descendait toute la réponse de l'établissement~?
  \item Ferme le carnet sur la méthode du crible en guise
        de devise~: une phrase sur l'endroit où les
        épidémies sont les moins chères à arrêter.
\end{enumerate}
\end{problem}
