\chapter{الإنزيمات والنمط الظاهري}\label{ch:g11:enzymes-and-phenotype}

القط السيامي كريمي اللون في معظم بدنه وبني داكن في أذنيه وخطمه وكفوفه
وذنبه --- أي في الأجزاء الباردة. احلق رقعة على ظهره، وأبقها باردة تحت
كيس ثلج بينما ينمو الفرو من جديد، فيأتي الفرو الجديد داكنًا. ولم تتغير
\omterm{def:g10:universal-dna:gene}{مورّثات} القط؛ وإنما إنزيمه الصانع للصباغ يعمل دون \qty{35}{\celsius}
ولا يعمل فوقها. فبين \omterm{def:g10:universal-dna:gene}{المورّثة} والصفة المرئية يقف، دائمًا تقريبًا،
\omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيم}؛ وبين \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} والصفة يقف المحيط الذي يعمل فيه. وهذا الفصل عن
هذين الوسيطين.

\section{النمط الظاهري على ثلاثة سلالم}

\begin{definition}[النمط الوراثي والنمط الظاهري]\label{def:g11:enzymes-and-phenotype:phenotype}
\emph{النمط الوراثي}\index{نمط وراثي} لفرد ما هو مجموعة \omterm{def:g10:universal-dna:gene}{الأليلات} التي
يحملها؛ و\emph{نمطه الظاهري}\index{نمط ظاهري} هو مجموعة خصائصه القابلة
للمشاهدة. ويمكن وصف النمط الظاهري على ثلاثة سلالم:
\begin{itemize}
  \item السلّم \emph{الجزيئي}: أي \omterm{def:g10:chemistry-of-life:families}{البروتينات} الموجودة، وبأي صورة،
        وبأي نشاط؛
  \item السلّم \emph{الخلوي}: أي شكل \omterm{def:g10:cells-common-unit:cell}{الخلايا} ومحتواها وسلوكها؛
  \item السلّم \emph{العضوي}: أي الصفات المرئية، من لون العينين إلى
        أعراض مرض.
\end{itemize}
وكل سلّم نتيجة للسلّم الذي تحته.
\end{definition}

\begin{example}[فقر الدم المنجلي على ثلاثة سلالم]\label{ex:g11:enzymes-and-phenotype:sickle}
جزيئيًا: \omterm{def:g10:chemistry-of-life:families}{حمض أميني} واحد من سلسلة الهيموغلوبين (الغلوتامات، وهو
السادس) يحل محله الفالين؛ فتلتصق الجزيئات المتغيرة بعضها ببعض في
ألياف طويلة حين ينقص الأكسجين. وخلويًا: تشوه الألياف الكرية الحمراء
فتصير هلالًا صلبًا يسد الأوعية الصغيرة وينكسر بسهولة. وعضويًا: فقر دم،
ونوبات ألم حين تنسد الأوعية، وأذى في الطحال والكليتين. فاستبدال واحد
في \omterm{def:g11:gene-expression:code}{رامزة} واحدة من \omterm{def:g10:universal-dna:gene}{مورّثة} واحدة، وثلاثة أوصاف لعاقبته.
\end{example}

\begin{proposition}[البروتينات تصنع النمط الظاهري]\label{prop:g11:enzymes-and-phenotype:proteins}
كل مستوى من \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الظاهري} تنتجه \omterm{def:g10:chemistry-of-life:families}{البروتينات}: \omterm{def:g10:universal-dna:gene}{فالأليل} يحدد بروتينًا
(\cref{ch:g11:gene-expression})؛ ونشاط \omterm{def:g11:gene-expression:protein}{البروتين} يحدد كيمياء \omterm{def:g10:cells-common-unit:cell}{الخلية}
وبنيتها؛ \omterm{def:g10:cells-common-unit:cell}{والخلايا} تحدد صفات الكائن. ومعظم تلك \omterm{def:g10:chemistry-of-life:families}{البروتينات}
\emph{\omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيمات}}، ومن ثم فمعظم الصفات حصيلة سلاسل من التفاعلات التي
تديرها \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيمات}.
\end{proposition}
\admitted

\section{الإنزيمات}

\begin{definition}[الإنزيم والركيزة والموقع الفعال]\label{def:g11:enzymes-and-phenotype:enzyme}
\emph{الإنزيم}\index{إنزيم} \omterm{def:g11:gene-expression:protein}{بروتين} يحفّز تفاعلًا كيميائيًا واحدًا: فهو
يجعل التفاعل يجري أسرع ملايين المرات من دون أن يستهلك. والجزيء الذي
يحوله هو \emph{ركيزته}\index{ركيزة}؛ ويجري التفاعل في جيب من الإنزيم
هو \emph{الموقع الفعال}\index{موقع فعال}، يوافق شكله وكيميته تلك
الركيزة دون غيرها. وهذه \emph{النوعية} تتبع من طي الإنزيم الثلاثي
الأبعاد، ومنه من متتالية حموضه الأمينية، ومنها من مورّثته.
\end{definition}

\begin{example}[اللاكتاز]\label{ex:g11:enzymes-and-phenotype:lactase}
سكر الحليب، وهو اللاكتوز، يشطر إلى سكرين بسيطين \omterm{def:g11:enzymes-and-phenotype:enzyme}{بإنزيم} اللاكتاز في
بطانة المعي الدقيق؛ ومن دون الشطر يمر غير مهضوم إلى القولون، فتخمّره
البكتيريا، فينتج غاز ومغص وإسهال. وكل الرضّع يصنعون اللاكتاز؛ وتطفأ
مورّثته عند معظم البشر بعد الفطام، فيصير البالغون غير محتملين
للاكتوز. وفي الجمهرات التي لها تاريخ طويل في تربية الماشية الحلوب،
يشيع \omterm{def:g10:universal-dna:gene}{أليل} يبقي \omterm{def:g10:universal-dna:gene}{المورّثة} نشطة مدى الحياة --- 90\% في شمال أوروبا، و10\%
في شرق آسيا. وقطرات اللاكتاز التي تباع لغير المحتملين تحوي \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم}
نفسه، تصنعه فطريات.
\end{example}

\begin{proposition}[شروط النشاط]\label{prop:g11:enzymes-and-phenotype:conditions}
يتوقف نشاط \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} على محيطه. فهو يرتفع مع الحرارة حتى يبدأ \omterm{def:g11:gene-expression:protein}{البروتين}
بالانتشار، ثم ينهار --- \omterm{def:g11:enzymes-and-phenotype:enzyme}{والإنزيمات} البشرية تعمل خير ما تعمل قرب
\qty{37}{\celsius} وتهدم فوق نحو \num{45}؛ ويعمل كل \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيم} ضمن مدى من
الحموضة يوافق موضعه (البيبسين في حمض المعدة، والتربسين في المعي
المتعادل)؛ وترتفع نسبته مع تركيز ركيزته حتى حد أقصى، حين ينشغل كل
\omterm{def:g11:enzymes-and-phenotype:enzyme}{موقع فعال}.
\end{proposition}
\admitted

\section{سلاسل التفاعلات}

\begin{proposition}[السبل الاستقلابية]\label{prop:g11:enzymes-and-phenotype:pathway}
قلما تصنع \omterm{def:g10:cells-common-unit:cell}{الخلية} ناتجًا في خطوة واحدة: فجزيء ابتدائي يحوَّل بتتابع من
\omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيمات}، يعمل كل منها على ناتج الذي قبله، على طول \emph{سبيل
استقلابي}\index{سبيل استقلابي}. ولكل \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيم} مورّثته الخاصة. \omterm{def:g10:universal-dna:gene}{وأليل} غير
عامل لأي واحد منها يحصر السبيل عند تلك الخطوة: فلا يصنع الناتج، وقد
تتراكم \omterm{def:g11:enzymes-and-phenotype:enzyme}{ركيزة} \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} المحصور. ومن ثم تستطيع مورّثتان مختلفتان أن
تنتجا العيب المرئي نفسه، ويستطيع عيب \omterm{def:g10:universal-dna:gene}{مورّثة} واحدة أن ينتج عدة عيوب.
\end{proposition}
\admitted

\section{النمط الوراثي والمحيط والنمط الظاهري}

\begin{proposition}[المحيط يشكّل النمط الظاهري]\label{prop:g11:enzymes-and-phenotype:environment}
\omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الظاهري} حصيلة \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي} \emph{و}المحيط. والمحيط يؤثر على
السلّم الجزيئي (فالحرارة والحموضة تضبطان نشاط \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم}، ووجود \omterm{def:g11:enzymes-and-phenotype:enzyme}{الركيزة}
يقرر هل يجري السبيل)، وعلى السلّم الخلوي (\omterm{def:g10:cells-common-unit:cell}{فالخلايا} تستجيب للإشارات
والمغذيات والضوء)، وعلى السلّم العضوي (الغذاء والحركة والمرض). فأفراد
متطابقو \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي} --- توأمان متطابقان، أو عقل من نبتة واحدة ---
يختلفون في \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الظاهري} بحسب تاريخهم؛ وأفراد مختلفو الأنماط الوراثية
يستطيعون التقارب على \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الظاهري} نفسه في ظروف مناسبة.
\end{proposition}
\begin{proof}[شواهد]
تيروزيناز القط السيامي الحساس للحرارة: فالصباغ لا يتكون إلا في
الأطراف الباردة، والرقعة المحلوقة المبردة ينمو فرقها داكنًا. وبيلة
الفينيل كيتون تضبط بالغذاء. والتوأمان المتطابقان اللذان يربيان
متفرقين يختلفان في الطول عدة سنتيمترات وفي كتلة الجسم أكثر. ونبتة من
النسيلة نفسها تنمى عند مستوى البحر وعلى ارتفاع فيختلف طولها عشرة
أضعاف، وترجع إذا نقلت. ففي كل حالة يكون \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي} ثابتًا \omterm{def:g11:enzymes-and-phenotype:phenotype}{والنمط
الظاهري} يتحرك مع الظروف.
\end{proof}

\begin{method}[الاستدلال من المورّثة إلى الصفة]\label{met:g11:enzymes-and-phenotype:reason}
\begin{enumerate}
  \item عيّن \omterm{def:g11:gene-expression:protein}{البروتين}: أي \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيم}، في أي سبيل، وماذا يفعل.
  \item عيّن أثر \omterm{def:g10:universal-dna:gene}{الأليل} في \omterm{def:g11:gene-expression:protein}{البروتين}: طبيعي، أو منقوص، أو غائب، أو
        متغير في شروط نشاطه.
  \item تابع العاقبة صعودًا على السلالم: ما الذي تفتقر إليه \omterm{def:g10:cells-common-unit:cell}{الخلية} أو
        تراكمه، ثم ما الذي يبينه الكائن.
  \item اسأل كم أليلًا عاملًا يلزم: فمعظم \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيمات} يكفيها واحد، فلا
        يظهر العيب إلا مع \omterm{def:g10:universal-dna:gene}{أليلين} غير عاملين (أي صفة متنحية).
  \item اسأل ما الذي يضيفه المحيط: \omterm{def:g11:enzymes-and-phenotype:enzyme}{ركيزة} موجودة أو غائبة، وحرارة،
        وعلاج.
\end{enumerate}
\end{method}

\begin{remark}[لماذا يكفي عادةً أليل عامل واحد]\label{rem:g11:enzymes-and-phenotype:recessive}
\omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيمات} محفزات: فكمية صغيرة منها تقلب كمية كبيرة من \omterm{def:g11:enzymes-and-phenotype:enzyme}{الركيزة}. ونصف
كمية التيروزيناز الطبيعية لا يزال يصنع ميلانينًا كافيًا، ونصف كمية
اللاكتاز الطبيعية لا يزال يهضم كأس حليب. ولذلك لا يكاد يتميز حامل
\omterm{def:g10:universal-dna:gene}{أليل} واحد غير عامل عن غير الحامل --- وهو «المتنحي» في كتاب المرحلة
الأولى، وقد فسّر الآن --- ولا تظهر الصفة إلا حين يخفق الأليلان معًا.
أما \omterm{def:g10:chemistry-of-life:families}{البروتينات} التي ليست \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيمات}، والتي تلزم بكميات أو بأشكال دقيقة،
فكثيرًا ما تكون الاستثناء: فهيموغلوبين نصف طبيعي يحدث فرقًا.
\end{remark}

\section{مسألة: القط ذو الأذنين الباردتين}

\begin{problem}[{مسألة نهاية الأسبوع --- فرو قط سيامي يفسَّر إنزيمًا
إنزيمًا: التيروزيناز الحساس للحرارة، وخريطة حرارات الجسم، والرقعة
المحلوقة، وغذاء يعيد كتابة نمط ظاهري}]
\label{pb:g11:enzymes-and-phenotype:1}
\omterm{def:g10:universal-dna:gene}{الأليل} السيامي \omterm{def:g10:universal-dna:gene}{لمورّثة} التيروزيناز يرمز \omterm{def:g11:enzymes-and-phenotype:enzyme}{لإنزيم} نشاطه، مقيسًا في
المختبر، كالآتي: \qty{100}{\percent} عند \qty{30}{\celsius}،
و\qty{80}{\percent} عند \qty{33}{\celsius}، و\qty{40}{\percent} عند
\qty{35}{\celsius}، و\qty{5}{\percent} عند \qty{37}{\celsius}،
و\qty{0}{\percent} عند \qty{39}{\celsius}. أما \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيم} \omterm{def:g10:universal-dna:gene}{الأليل} العادي
فيبقي \qty{100}{\percent} حتى \qty{40}{\celsius}. وحرارات جلد القط:
البدن \qty{38}{\celsius}، والأرجل \qty{36}{\celsius}، والكفوف
\qty{33}{\celsius}، والأذنان والخطم \qty{31}{\celsius}. والميلانين
يرى في الفرو حين يعمل \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} بأكثر من نحو \qty{30}{\percent}.

\medskip
\noindent\textbf{الجزء الأول --- الخريطة.}
\begin{enumerate}
  \item لكل منطقة من الجسم، اقرأ نشاط \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} السيامي وقل هل الفرو
        متصبغ.
  \item ارسم الفرو الناتج أو صفه. فهل يوافق القط السيامي؟
  \item افعل ذلك نفسه لقط يحمل \omterm{def:g10:universal-dna:gene}{الأليل} العادي.
  \item تولد الهريرة السيامية كريمية اللون كلها. فسّر ذلك، علمًا بأن
        الرحم عند \qty{38}{\celsius}.
  \item عند أي حرارة جلد يقع الحد بين الفرو الداكن والفاتح؟ وأي جزء
        من منحني \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} يثبّته؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- التجربة.}
\begin{enumerate}[resume]
  \item تحلق رقعة على البدن وتبقى عند \qty{30}{\celsius} تحت كمادة
        باردة بينما ينمو الفرو من جديد. تنبأ بلونها.
  \item تحلق رقعة على كف وتبقى دافئة عند \qty{38}{\celsius} بضمادة.
        تنبأ.
  \item بعد رفع الكمادات، يحفظ الفرو النامي لونه أشهرًا، ثم يستبدل
        بالتدريج. فسّر الأمرين.
  \item فسّر لماذا تبين هاتان التجربتان أن المحيط يؤثر على السلّم
        الجزيئي، لا في \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي}.
  \item يقترح تلميذ أن الأطراف الداكنة سببها \omterm{def:g10:universal-dna:gene}{أليل} مختلف في تلك
        \omterm{def:g10:cells-common-unit:cell}{الخلايا}. أعطِ حجتين ضد ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- السلالم الثلاثة.}
\begin{enumerate}[resume]
  \item صف \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الظاهري} السيامي على السلالم الجزيئي والخلوي والعضوي.
  \item يختلف \omterm{def:g10:universal-dna:gene}{الأليل} السيامي عن العادي باستبدال واحد يغير حمضًا
        أمينيًا واحدًا. فسّر كيف يستطيع \omterm{def:g10:chemistry-of-life:families}{حمض أميني} واحد أن يجعل إنزيمًا
        حساسًا للحرارة.
  \item قط يحمل أليلًا سياميًا وأليلًا عاديًا متصبغ تصبغًا تامًا.
        فسّر بالاعتماد على
        \cref{rem:g11:enzymes-and-phenotype:recessive}.
  \item قط أمهق له أليلان \omterm{def:g10:universal-dna:gene}{للمورّثة} نفسها لا يرمزان لأي \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيم} عامل عند
        أي حرارة. قارن نمطه الظاهري بالنمط السيامي على السلالم
        الثلاثة.
  \item الأرانب الهيمالائية وبعض الفئران تبين النمط نفسه. فبم يوحي
        ذلك عن \omterm{def:g10:universal-dna:gene}{مورّثة} التيروزيناز عندها، وبم تتوقع في \omterm{def:g11:mutations:mutation}{الطفرة}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- غذاء يعيد كتابة \omterm{def:g11:enzymes-and-phenotype:phenotype}{نمط ظاهري}.}
يكتشف عند وليد أنه مصاب ببيلة الفينيل كيتون: فينيل ألانين الدم
\qty{1200}{\micro mol/L} بدل \num{60}. والمستوى الآمن لنمو الدماغ دون
\qty{360}{\micro mol/L}.
\begin{enumerate}[resume]
  \item بأي عامل ارتفع الفينيل ألانين، وأي \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيم} من شكل السبيل مفقود؟
  \item فسّر لماذا كان دماغ الطفل، لا جلده مثلًا، هو العضو المعرض
        للخطر.
  \item يخفض الغذاء وارد الفينيل ألانين إلى خمس الطبيعي؛ ويهبط مستوى
        الدم تناسبيًا. فهل الطفل آمن؟ وما الذي يقوله ذلك عن العلاقة
        بين إمداد \omterm{def:g11:enzymes-and-phenotype:enzyme}{الركيزة} \omterm{def:g11:enzymes-and-phenotype:phenotype}{والنمط الظاهري} حين يغيب \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم}؟
  \item ومع ذلك فبعض الفينيل ألانين لا غنى عنه، لأن الجسم لا يستطيع
        صنعه. فسّر لماذا يجب ضبط الغذاء لا استبعاد الفينيل ألانين
        ببساطة.
  \item اذكر النتيجة: عند القط السيامي وعند الطفل، ما الذي ثبّته
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي}، وما الذي ضبطه المحيط، والجملة الواحدة التي
        تربط \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي} بالمحيط \omterm{def:g11:enzymes-and-phenotype:phenotype}{بالنمط الظاهري}.
\end{enumerate}
\end{problem}
