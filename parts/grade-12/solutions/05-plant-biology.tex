\section*{Chapter \ref{ch:g12:plant-biology} --- Plant Biology}

\begin{solution}{exo:g12:plant-biology:1}
\omterm{def:g12:plant-biology:organs}{Roots} take up water and minerals and anchor the plant. \omterm{def:g12:plant-biology:organs}{Stems} support leaves
and \omterm{def:g12:plant-biology:flower}{flowers} and provide long-distance transport. Leaves are the main sites of
\omterm{def:g10:photosynthesis:def}{photosynthesis}, with large \omterm{def:g10:the-cell:sa-v}{surface area} and stomata for \omterm{def:g12:animal-exchange:gas}{gas exchange}.
\end{solution}

\begin{solution}{exo:g12:plant-biology:2}
\omterm{def:g12:plant-biology:xylem}{Xylem} transports water and dissolved minerals from \omterm{def:g12:plant-biology:organs}{roots} toward shoots. The
main conducting \omterm{def:g10:the-cell:cell}{cells} (tracheids and vessel elements) are dead at maturity,
forming hollow tubes suited to tension-driven flow.
\end{solution}

\begin{solution}{exo:g12:plant-biology:3}
\omterm{def:g12:plant-biology:phloem}{Phloem} transports organic solutes, especially sugars, from sources (e.g.\
photosynthesising leaves) to sinks (growing tips, \omterm{def:g12:plant-biology:organs}{roots}, fruits). Direction is
set by source--sink relations, not simply ``downward.''
\end{solution}

\begin{solution}{exo:g12:plant-biology:4}
A \omterm{def:g12:plant-biology:stoma}{stoma} is an epidermal pore flanked by \omterm{def:g12:plant-biology:stoma}{guard cells}. Opening allows
$\mathrm{CO_2}$ entry for \omterm{def:g10:photosynthesis:def}{photosynthesis} but also permits water vapour loss
(transpiration); closing saves water at the cost of carbon gain.
\end{solution}

\begin{solution}{exo:g12:plant-biology:5}
Transpiration from leaves creates tension in continuous water columns. Cohesion
between water molecules and adhesion to \omterm{def:g12:plant-biology:xylem}{xylem} walls help maintain the column
so water is pulled up from \omterm{def:g12:plant-biology:organs}{roots} --- the cohesion--tension mechanism.
\end{solution}

\begin{solution}{exo:g12:plant-biology:6}
At a source, sugar loading into \omterm{def:g12:plant-biology:phloem}{phloem} raises osmolarity; water enters and
builds hydrostatic pressure. At a sink, sugar unloading lowers pressure. \omterm{def:g12:animal-exchange:bulk}{Bulk
flow} moves sap from high to low pressure along the sieve tubes.
\end{solution}

\begin{solution}{exo:g12:plant-biology:7}
Uneven light redistributes \omterm{def:g12:plant-biology:auxin}{auxin} toward the shaded side of a shoot. \omterm{def:g10:the-cell:cell}{Cells} on
the shaded side elongate more, so the shoot bends toward the light ---
\omterm{prop:g12:plant-biology:photo}{phototropism} by differential \omterm{def:g10:scientific-biology:properties}{growth}, not by ``seeking'' light consciously.
\end{solution}

\begin{solution}{exo:g12:plant-biology:8}
Stomatal closure reduces water loss under drought, which is protective for
\omterm{def:g12:plant-biology:xylem}{xylem} continuity and turgor. It also restricts $\mathrm{CO_2}$ \omterm{def:g10:membranes-enzymes:diffusion}{diffusion} into
the \omterm{def:g12:plant-biology:organs}{leaf}, so Calvin-cycle carbon fixation slows --- the classic water versus
carbon trade-off.
\end{solution}

\begin{solution}{exo:g12:plant-biology:9}
The \omterm{def:g12:plant-biology:alternation}{sporophyte} is the \omterm{def:g11:meiosis-life-cycles:ploidy}{diploid} generation and produces spores by \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}. The
\omterm{def:g12:plant-biology:alternation}{gametophyte} is \omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} and produces gametes by \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}. \omterm{def:g11:meiosis-life-cycles:alt}{Alternation of
generations} links the two multicellular stages in the plant life cycle.
\end{solution}

\begin{solution}{exo:g12:plant-biology:10}
A complete \omterm{def:g12:plant-biology:phloem}{phloem} girdle stops sugar transport to \omterm{def:g12:plant-biology:organs}{roots}. \omterm{def:g12:plant-biology:organs}{Roots} starve for
\omterm{def:g10:molecules-of-life:carbs}{carbohydrate}, fail to maintain uptake of water and minerals, and the shoot
eventually declines despite intact \omterm{def:g12:plant-biology:xylem}{xylem} --- both systems are interdependent.
\end{solution}

\begin{solution}{exo:g12:plant-biology:11}
Abscisic acid (ABA) promotes stomatal closure and other drought responses.
\omterm{def:g12:plant-biology:auxin}{Auxin} mediates phototropic bending under uneven light (among many other roles
in \omterm{def:g10:scientific-biology:properties}{development}). Different \omterm{def:g12:nerves-hormones:hormone}{hormones} dominate different environmental and
developmental cues.
\end{solution}

\begin{solution}{exo:g12:plant-biology:12}
In \omterm{prop:g12:plant-biology:double}{double fertilisation}, one sperm fuses with the egg to form the zygote
(embryo); the other fuses with \omterm{def:g10:molecules-of-life:water}{polar} nuclei to form endosperm, which
nourishes the embryo. Both products are part of the \omterm{def:g12:plant-biology:flower}{seed} package in
angiosperms.
\end{solution}

\begin{solution}{pb:g12:plant-biology:1}
\textbf{Part I.} (1)~\omterm{def:g12:plant-biology:tissues}{Dermal tissue} waterproofs and regulates stomata; \omterm{def:g12:plant-biology:tissues}{ground
tissue} performs much \omterm{def:g10:photosynthesis:def}{photosynthesis} and storage; \omterm{def:g12:plant-biology:tissues}{vascular tissue} imports and
exports materials. (2)~Continuous \omterm{def:g12:plant-biology:xylem}{xylem} and \omterm{def:g12:plant-biology:phloem}{phloem} conduits enable bulk
transport over plant-scale distances.

\textbf{Part II.} (3)~Soil water $\to$ \omterm{def:g12:plant-biology:organs}{root} $\to$ \omterm{def:g12:plant-biology:xylem}{xylem} $\to$ mesophyll $\to$
\omterm{def:g12:plant-biology:stoma}{stoma} $\to$ air. (4)~Stomata tend to close in drought: less water loss, less
$\mathrm{CO_2}$. (5)~Pressure-flow moves sugar from source \omterm{def:g12:plant-biology:organs}{leaf} to fruit sink.
(6)~Weaker transpiration pull reduces ascent of \omterm{def:g12:plant-biology:xylem}{xylem} sap.

\textbf{Part III.} (7)~Shaded-side \omterm{def:g12:plant-biology:auxin}{auxin} accumulation and greater elongation
bend the shoot. (8)~Ethylene in ripening; ABA in drought. (9)~Removing the tip
releases lateral buds from apical \omterm{def:g12:plant-biology:auxin}{auxin} inhibition.

\textbf{Part IV.} (10)--(11)~Standard angiosperm \omterm{def:g10:molecules-of-life:proteins}{sequence}; endosperm feeds the
embryo. (12)~Land-plant package: plumbing, controlled \omterm{def:g12:animal-exchange:gas}{gas exchange}, and
protected offspring in \omterm{def:g12:plant-biology:flower}{seeds} or spores.
\end{solution}
