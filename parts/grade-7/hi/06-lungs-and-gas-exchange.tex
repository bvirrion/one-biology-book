\chapter{फेफड़े और गैसों की अदला-बदली}\label{ch:g7:lungs-and-gas-exchange}

तुम्हारे जीवन के हर मिनट में लगभग आठ लीटर हवा उन दो अंगों में से
गुज़रती है जिन्हें तुमने कभी देखा तक नहीं। \Cref{ch:g4:breathing} ने
रास्ते और अदला-बदली का ख़ाका खींचा था;
\cref{ch:g7:how-animals-breathe} ने इंसानी मशीन को जंतु-जगत की चारों
मशीनों के बीच जगह दी। यह अध्याय तस्वीर पूरी करता है: हवा चलाने वाली
मशीनरी, हवा की कोठरियाँ अपने ठीक नाम से, अदला-बदली की संख्याएँ --- और
यह भी कि धुआँ इस पूरे कारख़ाने का क्या हाल करता है।

\section{हवा चलाने वाली मशीनरी}

\begin{proposition}[हवा कैसे चलाई जाती है]\label{prop:g7:lungs-and-gas-exchange:ventilation}
फेफड़ों की अपनी कोई \omterm{def:g3:bones-and-muscles:muscle}{पेशी} नहीं होती; छाती उन्हें धौंकनी की तरह पंप करती
है:
\begin{itemize}
  \item साँस लेना \emph{मेहनत} है: \omterm{def:g3:bones-and-muscles:skeleton}{पसलियों} की \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} पिंजरे को उठाकर
        चौड़ा करती हैं, और \emph{मध्यपट}\index{मध्यपट} --- यानी
        फेफड़ों के नीचे की चौड़ी \omterm{def:g3:bones-and-muscles:muscle}{पेशी} वाली फ़र्श --- नीचे की ओर चपटा
        हो जाता है; चौड़ी हुई छाती फेफड़ों को तान देती है, और उन्हें
        भरने के लिए हवा भीतर बह आती है;
  \item साँस छोड़ना, आराम में, ज़्यादातर \emph{ढील} है: \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} ढीली
        पड़ती हैं, तनी हुई छाती वापस उछलती है, और हवा बाहर बह जाती है।
        (मोमबत्तियाँ बुझाना उस उछाल में \omterm{def:g3:bones-and-muscles:muscle}{पेशी} भी जोड़ देता है।)
\end{itemize}
आराम में मिनट में लगभग $15$ बार
(\cref{def:g4:breathing:movements}), बिना माँगे
(\cref{rem:g5:brain-in-command:more}) --- और माँग पर गहरी तथा तेज़ की
हुई (\cref{prop:g7:organs-at-work:supply})।
\end{proposition}
\admitted

\begin{example}[धौंकनी को महसूस करना]\label{ex:g7:lungs-and-gas-exchange:feel}
हाथ \omterm{def:g3:bones-and-muscles:skeleton}{पसलियों} पर रखो और गहरी साँस लो: पिंजरा तुम्हारी हथेलियों के नीचे
उठता और चौड़ा होता है (\cref{ex:g4:breathing:watch})। अब कुत्ते की तरह
हाँफो: वे छोटी-छोटी उथली फूँकें लगभग पूरी की पूरी \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{मध्यपट} की हैं। और
हिचकियाँ, पूरी बात के लिए, \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{मध्यपट} की फड़कती झूठी शुरुआतें हैं ---
धौंकनी अपनी ही मरज़ी के वक़्त पर मिसफ़ायर करती हुई।
\end{example}

\section{कूपिकाएँ और अदला-बदली}

\begin{definition}[कूपिकाएँ]\label{def:g7:lungs-and-gas-exchange:alveoli}
\cref{def:g4:breathing:organs} वाली नन्ही हवा-कोठरियाँ ही
\emph{कूपिकाएँ}\index{कूपिका} हैं: हवा-मार्गों के सबसे महीन सिरों पर
गुच्छों में लगे करोड़ों पतली दीवार वाले बुलबुले, और हर एक महीन
रक्त-वाहिकाओं में लिपटा हुआ। उन सबकी जोड़ी हुई सतह
\cref{rem:g4:breathing:sponge} वाला आधा टेनिस मैदान है --- यानी
\cref{prop:g7:how-animals-breathe:spec} वाली बड़ी-पतली-नम विशेष-सूची की
इंसानी नक़ल।
\end{definition}

\begin{proposition}[अदला-बदली, संख्याओं में]\label{prop:g7:lungs-and-gas-exchange:numbers}
भीतर आती हवा की तुलना बाहर जाती हवा से करो: भीतर ली गई हवा लगभग $21$
फ़ीसदी \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} होती है और \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} से क़रीब-क़रीब ख़ाली;
बाहर छोड़ी गई हवा में \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} घटकर लगभग $16$ फ़ीसदी रह जाती है और
\omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} बढ़कर लगभग $4$ फ़ीसदी हो जाती है (और साथ में वह गरम
तथा नम भी होती है --- \cref{ex:g4:breathing:mirror})। ये फ़र्क़ ही
अदला-बदली हैं: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाओं} की दीवारों के पार \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} \omterm{prop:g4:journey-of-food:absorb}{रक्त} की ओर जाती है
और \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} वापस आती है
(\cref{prop:g4:breathing:exchange}), और वह भी हर दिन के हर मिनट।
\end{proposition}
\admitted

\begin{example}[हिसाब मिल जाता है]\label{ex:g7:lungs-and-gas-exchange:accounting}
मिनट के आठ लीटर, और हर साँस की \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} का बीसवाँ हिस्सा रखा हुआ: यह
गणित मोटे तौर पर आराम करते शरीर का \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} वाला बिल जुटा देता है ---
और मेहनत में, जब साँस तिगुनी तथा गहरी हो जाती है और \omterm{def:g4:heart-and-blood:heart}{दिल} के चक्कर तेज़
(\cref{prop:g7:organs-at-work:supply}), तो पहुँचाना भी \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} के बिल
से मेल खाने के लिए कई गुना हो जाता है। मशीन की संख्याएँ और शरीर का
बहीखाता आपस में मिल जाते हैं; वे दो काउंटरों पर रखी एक ही किताबें हैं।
\end{example}

\section{मशीनरी से बदसुलूकी: धुआँ}

\begin{proposition}[धुआँ इस कारख़ाने का क्या करता है]\label{prop:g7:lungs-and-gas-exchange:smoke}
तंबाकू का धुआँ इस अध्याय की हर मंज़िल पर हमला करता है
(\cref{prop:g5:healthy-choices:substances} ने चेतावनी दे दी थी; यहाँ
उसकी कारीगरी है):
\begin{itemize}
  \item उसका तारकोल हवा-मार्गों पर परत जमाता और उन्हें खिजाता है:
        सफ़ाई करती सतहें लड़खड़ाने लगती हैं, और ``धूम्रपान वाली खाँसी''
        हवा-मार्गों की लदी हुई बुहारी है;
  \item लगातार खिजी हुई \omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाओं} की दीवारें टूट जाती हैं: कोठरियाँ
        आपस में मिलकर कम, बड़ी और ढीली जगहें बन जाती हैं ---
        \emph{सतह चली जाती है}, और उसके साथ अदला-बदली भी (यानी
        विशेष-सूची, टुकड़े-टुकड़े);
  \item धुएँ की एक गैस \omterm{prop:g4:journey-of-food:absorb}{रक्त} की \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} वाली सीटों पर सवार हो जाती है,
        और \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} को उसकी अपनी ही सवारी से धकेल देती है;
  \item और हवा-मार्ग सिकुड़ जाते हैं --- धूम्रपान करने वाले की छाती से
        ली गई हर साँस आधी बंद धौंकनी से ली गई साँस है।
\end{itemize}
यह नुक़सान धीमा, जुड़ता जाने वाला और सिर्फ़ आंशिक रूप से पलटने लायक़
है: जो \omterm{def:g7:how-animals-breathe:four}{फेफड़ा} कभी शुरू ही नहीं करता, वह अपना आधा टेनिस मैदान साबुत रखता
है (\cref{rem:g5:healthy-choices:never})।
\end{proposition}
\admitted

\begin{example}[सीढ़ी की दो परीक्षाएँ]\label{ex:g7:lungs-and-gas-exchange:stairs}
\cref{ch:g7:organs-at-work} वाली तीन मंज़िलें, बराबर उम्र और बराबर
कसरत की आदतों वाले दो बड़ों ने चढ़ीं, जिनमें एक बरसों से धूम्रपान करता
है: धूम्रपान करने वाला ऊपर हाँफता हुआ पहुँचता है, और वह भी उस \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} पर
जिस तक दूसरा सिर्फ़ दौड़ लगाकर पहुँचता है। कम सतह, घिरी हुई सीटें,
सिकुड़े हवा-मार्ग --- आपूर्ति की शृंखला
\cref{prop:g7:organs-at-work:bill} वाला बिल एक गला घोंटी हुई खींच से
चुकाती है। चुस्ती की परीक्षाएँ फेफड़ों का इतिहास किसी भी जाँच जितने
पक्के तौर पर पढ़ लेती हैं।
\end{example}

\begin{method}[मशीनरी सँभालना]\label{met:g7:lungs-and-gas-exchange:keep}
\Cref{met:g4:breathing:care}, इस अध्याय के कारणों के साथ और ऊँचा किया
हुआ:
\begin{enumerate}
  \item कोई धुआँ नहीं --- न अपना, न दूसरों का: गई हुई सतह मुश्किल से
        ही वापस आती है;
  \item कमरों में हवा आने दो (\cref{exo:g4:breathing:10}) और खेल के
        लिए बाहर की साफ़ हवा चुनो;
  \item धौंकनी की कसरत करो: गहरी साँस वाली मेहनत साँस की \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ}
        बनाती है और पूरी सतह में हवा पहुँचाए रखती है;
  \item ठंड और धूल में नाक से साँस लो (\cref{exo:g4:breathing:7})
        --- छन्ना बुहारी की हिफ़ाज़त करता है।
\end{enumerate}
\end{method}

``\omterm{def:g7:how-animals-breathe:four}{फेफड़ा} जूते के डिब्बे में तह किया हुआ आधे टेनिस मैदान का सरहदी इलाक़ा
है, और धुआँ उस सरहद पर लगी एक धीमी आग।'' इस अध्याय की संख्याओं और
कार्य-प्रणालियों से दोनों तस्वीरें सही ठहराओ।
\end{exercise}

\section{समस्या: साँस लेने वाली मशीन की मिसल}

\begin{problem}[{सप्ताहांत समस्या --- इंसानी धौंकनी की एक इंजीनियर वाली
जाँच}]\label{pb:g7:lungs-and-gas-exchange:1}
इंजीनियरी की एक कक्षा इंसानी साँस वाली मशीन की काग़ज़ पर ``जाँच'' करती
है: खींच, पंप, अदला-बदली की सतह, ढुलाई का जोड़, और रखरखाव का लेखा।

\medskip\noindent\textbf{भाग I --- पंप।}
\begin{enumerate}
  \item पंप का चक्र तैयार करो: दोनों चरण, हर एक के चलाने वाले, और यह
        भी कि आराम में कौन-सा चरण \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} ख़र्च करता है।
  \item पंप का जिस अंग को वह पंप करता है उसकी अपनी \omterm{def:g3:bones-and-muscles:muscle}{पेशी} से कोई
        संपर्क ही नहीं --- फेफड़ों के पास \omterm{def:g3:bones-and-muscles:muscle}{पेशी} है ही नहीं। फिर छाती
        का चौड़ा होना हवा चलाता कैसे है?
  \item पंप की आराम वाली दर बताओ, और मेहनत वाले उसके दोनों फेरबदल भी
        (\cref{prop:g7:organs-at-work:supply})।
  \item हाँफना, गहरी आह भरना, मोमबत्तियाँ बुझाना: हर एक को चक्र की
        कार्य-प्रणाली में उसकी जगह दो।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- अदला-बदली की सतह।}
\begin{enumerate}[resume]
  \item सतह की विशेष-सूची लिखो: नाम, गिनती, दीवार, लपेटा, और जोड़ा
        हुआ आकार।
  \item उस विशेष-सूची को \cref{prop:g7:how-animals-breathe:spec} के
        सामने, एक-एक मद करके जाँचो।
  \item भीतर आती हवा और बाहर जाती हवा अध्याय वाले फ़ीसदी से अलग हैं।
        \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} के फ़र्क़ को भीतर आती हवा के $21$ फ़ीसदी के हिस्से के
        रूप में निकालो --- भीतर ली गई \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} का मोटे तौर पर कितना
        हिस्सा रखा जाता है?
  \item बाहर जाती हवा को गरम और नम क्यों रहना ही पड़ता है --- वह सतह
        के किस गुण की क़ीमत है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- रखरखाव का लेखा।}
\begin{enumerate}[resume]
  \item धुएँ के चारों हमलों को इनके नीचे दर्ज करो: खींच, पंप, सतह,
        ढुलाई। हर एक पर एक पंक्ति।
  \item मिसल में लिखा है: ``सतह के नुक़सान पूँजी के नुक़सान हैं।''
        उन खाँसियों और ज़ुकामों के सामने रखकर इसे समझाओ, जो चलते
        रहने वाली मरम्मतें हैं।
  \item रखरखाव की समय-सारणी --- यानी
        \cref{met:g7:lungs-and-gas-exchange:keep} --- इंजीनियर के
        चार फ़रमानों के रूप में, कारणों के साथ लिखो।
  \item जाँच का समापन करो: एक वाक्य में बताओ कि शरीर के पंपों और
        सरहदों में अकेली यही मशीन आधी मालिक की अपनी मरज़ी के नीचे
        क्यों है --- और वह क़ाबू किस काम आता है (बोली, गीत, और
        ग़ोताख़ोर की धीमी साँस)।
\end{enumerate}
\end{problem}
