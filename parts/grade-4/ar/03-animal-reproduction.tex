\chapter{كيف تتكاثر الحيوانات}\label{ch:g4:animal-reproduction}

في أيار تضجّ البركة بغناء الضفادع؛ وفي حزيران تسودّ \omterm{ex:g2:animals-grow-up:frog}{بالشراغيف}.
وتبين أن الغناء كان بداية الحكاية: فالتكاثر --- وهو كيف تصنع
\omterm{def:g1:animals-around-us:animal}{الحيوانات} الجيل التالي --- يتبع قواعد تصدق من الضفادع إلى الفيلة،
مع تنويعات رائعة في التفاصيل.

\section{لا بد من اثنين}

\begin{definition}[التكاثر الجنسي]\label{def:g4:animal-reproduction:sexual}
يحتاج صنع الصغار في \omterm{def:g1:animals-around-us:animal}{الحيوانات} إلى والدين في العادة:
\emph{أنثى}\index{أنثى} و\emph{ذكر}\index{ذكر}. فالأنثى تنتج
\emph{بويضات}\index{بويضة}؛ والذكر ينتج \emph{نطفًا}\index{نطفة}.
ويبدأ \omterm{def:g1:animals-around-us:animal}{الحيوان} الصغير حين تلتحم نطفة واحدة ببويضة واحدة --- وذلك
الالتحام هو \emph{الإخصاب}\index{إخصاب}. ويسمى التكاثر بوالدين
\emph{التكاثر الجنسي}\index{تكاثر جنسي}.
\end{definition}

\begin{example}[لماذا تغني الضفادع]\label{ex:g4:animal-reproduction:song}
جوقة أيار هي الذكور تنادي الإناث إلى البركة. ثم يطلق كل زوج خلاياه
في الماء --- \omterm{def:g4:animal-reproduction:sexual}{بويضات} \omterm{def:g4:animal-reproduction:sexual}{الأنثى} ونطف \omterm{def:g4:animal-reproduction:sexual}{الذكر} --- حيث يحدث \omterm{def:g4:animal-reproduction:sexual}{الإخصاب}. والنقط
السوداء في الهلام بعد أيام قليلة هي الصغار وقد بدأت تنمو: فالغناء
كان الخطوة الأولى من دورة الضفدع في
\cref{ch:g2:animals-grow-up}.
\end{example}

{\small بعد الجوقة بأيام قليلة: \omterm{def:g2:animals-grow-up:born}{بيض} الضفادع بين القصب، وكل كرة

\begin{remark}[والدان وحظ ممزوج]\label{rem:g4:animal-reproduction:mix}
ينال \omterm{def:g1:animals-around-us:animal}{الحيوان} الصغير شيئًا من كل واحد من والديه --- ولهذا قد يكون
للمهر لون أمه وغرّة أبيه البيضاء. أما ما يسلّمه كل والد بالضبط،
وكيف، فمن الأسئلة الكبرى التي تنتظرنا في هذا الكتاب؛ وحسبنا الآن
أن نلاحظ أن لكل \omterm{def:g1:animals-around-us:animal}{حيوان} أُنتج جنسيًّا والدين، وأنه يشبههما معًا.
\end{remark}

\section{بيض يوضع، أو صغار تولد حية}

\begin{definition}[البيوض والولود]\label{def:g4:animal-reproduction:ovivivi}
تنال الآن طريقتا \omterm{def:g2:born-grow-die:cycle}{الولادة} في \cref{def:g2:animals-grow-up:born}
اسميهما العلميين:
\begin{itemize}
  \item يكون \omterm{def:g1:animals-around-us:animal}{الحيوان} \emph{بيوضًا}\index{بيوض} إذا وضعت \omterm{def:g4:animal-reproduction:sexual}{الأنثى}
        \emph{بيضًا}، وأتمّ الصغير نموه داخل \omterm{def:g2:animals-grow-up:born}{البيضة} خارج جسمها:
        \omterm{prop:g3:sorting-living-things:groups}{كالطيور}، وأكثر \omterm{prop:g3:sorting-living-things:groups}{الأسماك}، وأكثر \omterm{prop:g4:classifying-animals:classes}{الزواحف}، \omterm{prop:g3:sorting-living-things:groups}{والبرمائيات}،
        \omterm{prop:g3:sorting-living-things:groups}{والحشرات}، والحلزونات؛
  \item ويكون \emph{ولودًا}\index{ولود} إذا نما الصغير
        \emph{داخل جسم الأم}، مغذًّى منها، ثم \emph{وُلد حيًّا}:
        \omterm{prop:g3:sorting-living-things:groups}{كالثدييات} كلها تقريبًا.
\end{itemize}
\end{definition}

\begin{example}[مهدان]\label{ex:g4:animal-reproduction:nurseries}
كتكوت الدجاجة يقضي ثلاثة أسابيع دافئة في \omterm{def:g2:animals-grow-up:born}{بيضة} مقشورة، عائشًا على
المحّ (\cref{rem:g2:animals-grow-up:egg})؛ أما المهر فيقضي أحد عشر
شهرًا داخل الفرس، تغذّيه بجسمها باستمرار، ثم يولد مستعدًّا للوقوف في
غضون ساعة. الغاية نفسها ومهدان: غداء محزوم في قشرة، وخدمة غرف في
الداخل.
\end{example}

\begin{proposition}[أين يحدث الإخصاب]\label{prop:g4:animal-reproduction:where}
عند سكان الماء \omterm{prop:g3:sorting-living-things:groups}{كالأسماك} والضفادع يحدث \omterm{def:g4:animal-reproduction:sexual}{الإخصاب} في الغالب \emph{في
الماء}، خارج جسم \omterm{def:g4:animal-reproduction:sexual}{الأنثى} --- إذ تُطلق \omterm{def:g4:animal-reproduction:sexual}{البويضات} والنطف معًا فتلتقي
بالآلاف. أما \omterm{def:g1:animals-around-us:animal}{حيوانات} اليابسة --- \omterm{prop:g3:sorting-living-things:groups}{الطيور} \omterm{prop:g4:classifying-animals:classes}{والزواحف} \omterm{prop:g3:sorting-living-things:groups}{والثدييات}
\omterm{prop:g3:sorting-living-things:groups}{والحشرات} --- فخلاياها تجفّ في الهواء الطلق، ولذلك يضع \omterm{def:g4:animal-reproduction:sexual}{الذكر} نطفه
\emph{داخل جسم \omterm{def:g4:animal-reproduction:sexual}{الأنثى}} في أثناء \emph{التزاوج}\index{تزاوج}،
فيحدث \omterm{def:g4:animal-reproduction:sexual}{الإخصاب} هناك.
\end{proposition}
\admitted

\section{صغار قليلة وعناية كثيرة، أو كثيرة وبلا عناية}

\begin{proposition}[خطتان للجيل التالي]\label{prop:g4:animal-reproduction:strategies}
قارن الأعداد: سمكة القدّ تطلق \emph{ملايين} \omterm{def:g4:animal-reproduction:ovivivi}{البيوض} في السنة ولا
تعتني بواحدة منها؛ والفيلة لها \emph{عجل واحد} كل بضع سنوات
تحرسه عقدًا كاملًا. وبين هذين الطرفين تصدق قاعدة في عالم \omterm{def:g1:animals-around-us:animal}{الحيوان}
كله: كلما أنتج النوع صغارًا أكثر قلّت العناية بكل واحد ---
وكلما أنتج أقلّ زادت حماية كل واحد وتغذيته. والخطتان كلتاهما
تنجحان: إذ يبقى في المتوسط من الصغار ما يكفي ليحلّ محل الوالدين.
\end{proposition}
\admitted

\begin{example}[عدّ البركة]\label{ex:g4:animal-reproduction:pondcount}
حضنة زوج ضفادع واحد تحمل آلاف \omterm{def:g4:animal-reproduction:ovivivi}{البيوض}. والبلاشين \omterm{prop:g3:sorting-living-things:groups}{والأسماك} الشوكية
\omterm{ex:g2:animals-grow-up:butterfly}{ويرقات} الرعّاشات (انظر \cref{ex:g3:food-chains:three}) تأكل
\omterm{ex:g2:animals-grow-up:frog}{الشراغيف} طول الربيع --- ومن بين الآلاف تتسلق حفنة من الضفيدعات
الضفة في تموز. والحضنة الهائلة ليست هدرًا: بل هي جواب الضفدع عن
أخطار البركة، بلا حراسة أبوية البتة.
\end{example}

\begin{example}[الطرف الآخر]\label{ex:g4:animal-reproduction:whalecalf}
عجل الحوت يولد وحده، فيسبح في الحال بجانب أمه، ويشرب حليبها الغني
شهورًا كثيرة، ويُدافَع عنه أمام كل خطر. صغير واحد في المرة ---
لكنه ينال كل شيء. \omterm{prop:g3:sorting-living-things:groups}{والثدييات} بحليبها
(\cref{def:g2:animals-grow-up:born}) هي كبرى المتخصصات في خطة
القليل المعتنى به.
\end{example}

\begin{method}[قراءة خطة نوع]\label{met:g4:animal-reproduction:read}
إذا أُعطيت نوعًا فاسأل:
\begin{enumerate}
  \item كم صغيرًا في المرة --- واحد، أم عشرة، أم آلاف؟
  \item ومن يحرسها ويغذّيها --- الوالدان معًا، أم الأم، أم لا
        أحد؟
  \item وأين تبدأ --- في \omterm{def:g2:animals-grow-up:born}{بيضة} خارجية، أم نموًّا في الداخل؟
  \item وراجع ميزان
        \cref{prop:g4:animal-reproduction:strategies}: أصغار
        كثيرة بعناية قليلة، أم قليلة بعناية كثيرة.
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:1}
ما \omterm{def:g4:animal-reproduction:sexual}{الإخصاب}؟ وأي خلية تأتي من كل واحد من الوالدين؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:2}
عرّف \omterm{def:g4:animal-reproduction:ovivivi}{البيوض} \omterm{def:g4:animal-reproduction:ovivivi}{والولود}، مع حيوانين من كل صنف.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:3}
لماذا كانت الضفادع تغني في أيار؟ أعد حكاية البركة في ثلاث خطوات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:4}
أين يحدث \omterm{def:g4:animal-reproduction:sexual}{الإخصاب} عند السلمون المرقط، وأين عند الدجاجة؟ ولماذا هذا
الفرق وفق \cref{prop:g4:animal-reproduction:where}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:5}
قارن مهد الكتكوت بمهد المهر: أين ينمو كل واحد، وما الذي يغذّيه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:6}
اذكر القاعدة التي تربط عدد الصغار بالعناية التي تنالها. وأعطِ
مثالي الفصل المتطرفين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:7}
ملايين \omterm{def:g4:animal-reproduction:ovivivi}{بيوض} سمكة القدّ لا تصير أسماك قدّ بالغة في أكثرها. فلماذا
كانت الخطة ناجحة مع ذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:8}
طبّق \cref{met:g4:animal-reproduction:read} على القطة: بطن معتاد
من نحو خمسة، تُرضعها الأم وتحرسها أسابيع، وقد نمت في داخلها. فإلى
أي طرفي الخطة هي أقرب؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:animal-reproduction:9}
كثيرًا ما يشبه المهر والديه معًا. فعلام يدل هذا في شأن ما يناله
الصغير، وفق \cref{rem:g4:animal-reproduction:mix}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:animal-reproduction:10}
\omterm{def:g4:animal-reproduction:sexual}{ذكر} السمكة الشوكية يبني عشًّا، ويروّح \omterm{def:g2:animals-grow-up:born}{البيض}، ويطرد المتطفلين ---
وهو تفانٍ نادر عند سمكة. فباستعمال
\cref{prop:g4:animal-reproduction:strategies}، بم تتنبأ في شأن
حجم حضنته، بالقياس إلى ملايين سمكة القدّ؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:animal-reproduction:11}
تزحف السلاحف البحرية إلى الشاطئ لتدفن بيضها في الرمل الدافئ، ثم
تعود إلى البحر. أبيوض هي أم \omterm{def:g4:animal-reproduction:ovivivi}{ولود}؟ وأمعتنًى ببيضها أم لا؟ ولماذا
يجب على هذا الزاحف، وهو سابح طول حياته، أن يضع بيضه على
\emph{اليابسة} مع ذلك؟ (استعمل صفات \omterm{prop:g4:classifying-animals:classes}{الزواحف} في
\cref{prop:g4:classifying-animals:classes}.)
\end{exercise}
