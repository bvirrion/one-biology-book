\chapter{Aangepast aan hun wereld}\label{ch:g4:adapted-to-their-world}

Zet een ijsbeer in de Sahara en een kameel op het pakijs, en geen van
beide houdt het een seizoen uit. En toch is elk van beide thuis een
kampioen: de beer zwemt op zijn gemak in ijskoude zeeën, de kameel
doorkruist woestijnen waar een paard binnen dagen bezwijkt. Levende
wezens passen bij hun wereld met een nauwkeurigheid die het bestuderen
waard is --- kenmerk voor kenmerk.

\section{Wat aanpassing betekent}

\begin{definition}[Aanpassing]\label{def:g4:adapted-to-their-world:adaptation}
Een \emph{aanpassing}\index{aanpassing} is een kenmerk van het lichaam
of het gedrag van een \omterm{def:g1:living-or-not:living}{levend wezen} dat het geschikt maakt voor de
omstandigheden van zijn \omterm{def:g2:caring-for-nature:environment}{leefomgeving} --- voor de kou of de droogte, voor
zijn voedsel, voor zijn jagers. \Cref{prop:g2:where-animals-live:fit}
liet \omterm{def:g1:animals-around-us:animal}{dieren} zien die bij hun \omterm{def:g2:where-animals-live:habitat}{leefgebied} passen; aanpassing is dat
passen, stuk voor stuk bekeken.
\end{definition}

\begin{example}[De kameel, stuk voor stuk]\label{ex:g4:adapted-to-their-world:camel}
De problemen van de woestijn: hitte, geen water, zacht zand. De
antwoorden van de kameel: bulten vet --- een \omterm{def:g3:animals-through-seasons:active}{voedselvoorraad} voor magere
tochten; het vermogen om in minuten honderd liter te drinken en dan
dagen zonder te kunnen; brede tweetenige voeten die niet in het zand
wegzakken; neusgaten die zich tegen opwaaiend stof sluiten en lange
dubbele wimpers erbij. Elk woestijnprobleem, beantwoord met een kenmerk.
\end{example}

\begin{example}[De ijsbeer, stuk voor stuk]\label{ex:g4:adapted-to-their-world:bear}
De problemen van het pakijs: kou, ijskoud water, prooi die op wit ijs
moeilijk te benaderen is. De antwoorden van de beer: dikke \omterm{def:g1:animals-around-us:covering}{vacht} boven
een diepe laag vet --- dubbele isolatie; brede, licht van zwemvliezen
voorziene poten --- sneeuwschoenen en peddels in één; een witte \omterm{def:g1:animals-around-us:covering}{vacht} ---
onzichtbaar tegen de sneeuw waarop hij jaagt. Dezelfde nauwkeurigheid,
afgestemd op de tegenovergestelde wereld.
\end{example}

\begin{proposition}[Aanpassingen beantwoorden problemen]\label{prop:g4:adapted-to-their-world:answers}
Waar de omgeving een moeilijk probleem stelt, dragen de levende wezens
die daar thuis zijn kenmerken die erop antwoorden:
\begin{itemize}
  \item kou: een dikke \omterm{def:g1:animals-around-us:covering}{vacht}, vetlagen, kleine \omterm{def:g1:the-five-senses:sense}{oren} die weinig warmte
        verliezen;
  \item droogte: wateropslag, wasachtige \omterm{def:g1:the-five-senses:sense}{huiden}, diepe \omterm{def:g1:plants-around-us:parts}{wortels};
  \item leven in het water: gestroomlijnde lichamen, vinnen en
        zwemvliezen, \omterm{def:g1:animals-around-us:covering}{vachten} waar het water van afloopt;
  \item jagers overal om je heen: schutkleuren, pantsers, stekels,
        snelheid, en \omterm{def:g1:the-five-senses:sense}{ogen} die ver uit elkaar staan om alle kanten op te
        kijken;
  \item de kost verdienen met jagen: scherpe \omterm{def:g1:the-five-senses:sense}{zintuigen}, \omterm{def:g1:the-five-senses:sense}{ogen} naar voren,
        klauwen en snelheid
        (\cref{prop:g2:what-animals-eat:match}).
\end{itemize}
\end{proposition}
\admitted

\section{Planten zijn ook aangepast}

\begin{example}[De cactus]\label{ex:g4:adapted-to-their-world:cactus}
De woestijn beproeft \omterm{def:g1:plants-around-us:plant}{planten} het hardst --- zij kunnen niet naar het
water lopen. De cactus antwoordt: een dikke, tonvormige \omterm{def:g1:plants-around-us:parts}{stengel} die
water opslaat en het lichtvangwerk van de \omterm{def:g1:plants-around-us:parts}{bladeren} overneemt; \omterm{def:g1:plants-around-us:parts}{bladeren}
die tot stekels geslonken zijn en bijna geen water verliezen --- en die
de ton tegen dorstige \omterm{def:g1:animals-around-us:animal}{dieren} verdedigen; \omterm{def:g1:plants-around-us:parts}{wortels} die breed en ondiep
uitwaaieren om een hele zeldzame bui in één keer op te vangen.
\end{example}

\begin{example}[Alpiene kussens]\label{ex:g4:adapted-to-their-world:alpine}
Hooggebergteplanten staan tegenover ijzige wind. Veel ervan groeien als
strakke kussens die zich tegen de grond drukken: binnen het kussen is de
wind gebroken en blijft de lucht milder --- een vorm die schuilplaats is.
Laaglandneven van dezelfde \omterm{def:g1:plants-around-us:plant}{planten} groeien hoog en los; de bergvormen
klemmen zich aan de rots.
\end{example}

\begin{method}[De wereld van een organisme aflezen aan zijn lichaam]\label{met:g4:adapted-to-their-world:read}
Bij een onbekend \omterm{def:g1:animals-around-us:animal}{dier} of een onbekende \omterm{def:g1:plants-around-us:plant}{plant}:
\begin{enumerate}
  \item noem zijn opvallende kenmerken --- \omterm{def:g1:animals-around-us:covering}{vacht}, poten, \omterm{def:g1:the-five-senses:sense}{ogen},
        lichaamsvorm, \omterm{def:g1:plants-around-us:parts}{bladeren}, \omterm{def:g1:plants-around-us:parts}{wortels};
  \item vraag je bij elk af: \emph{welk probleem zou dit oplossen?}
  \item voeg de antwoorden samen tot een beeld van zijn thuis --- koud
        of heet, nat of droog, jager of bejaagd;
  \item toets je gok aan waar het organisme werkelijk leeft.
\end{enumerate}
Dit is \cref{met:g2:where-animals-live:matching} volwassen geworden: van
een \omterm{def:g2:where-animals-live:habitat}{leefgebied} zoeken naar een hele wereld reconstrueren.
\end{method}

\begin{example}[De methode op een raadselachtige huid]\label{ex:g4:adapted-to-their-world:mystery}
In een museumla ligt een \omterm{def:g1:the-five-senses:sense}{huid} zonder etiket: dichte zachte \omterm{def:g1:animals-around-us:covering}{vacht},
piepkleine \omterm{def:g1:the-five-senses:sense}{ogen}, enorme schepvormige voorklauwen. De \omterm{def:g1:animals-around-us:covering}{vacht} zegt
zoogdier; de scheppen zeggen graven; de nutteloze \omterm{def:g1:the-five-senses:sense}{ogen} zeggen een leven
in het donker. Vonnis: een ondergrondse graver --- en inderdaad is het
een mol (\cref{ex:g2:where-animals-live:mole}), correct afgelezen uit
drie kenmerken alleen.
\end{example}

\begin{remark}[Hoe komt dat passen tot stand?]\label{rem:g4:adapted-to-their-world:how}
Een kameel \emph{besluit} nooit brede voeten te laten groeien, en geen
enkele ijsbeer kiest zijn witte \omterm{def:g1:animals-around-us:covering}{vacht}. Hoe komt het dan dat levende
wezens zo goed bij hun wereld passen? Het is een van de diepste vragen
van de \omterm{def:g1:living-or-not:living}{biologie}, en het volledige antwoord --- hoe dat passen zich
generatie na generatie opbouwt --- is een verhaal dat dit boek voor een
later jaar bewaart. Leer voorlopig het passen te \emph{zien}: dat zien
is wat de vraag stelbaar maakte.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:1}
Wat is een \omterm{def:g4:adapted-to-their-world:adaptation}{aanpassing}? Geef één voorbeeld bij een \omterm{def:g1:animals-around-us:animal}{dier} en één bij een \omterm{def:g1:plants-around-us:plant}{plant}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:2}
Noem drie woestijnproblemen en het antwoord van de kameel op elk.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:3}
Noem drie problemen van het pakijs en het antwoord van de ijsbeer op elk.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:4}
Welke taken doen de stekels van de cactus? Wat slaat zijn tonvormige
\omterm{def:g1:plants-around-us:parts}{stengel} op, en welk werk heeft die \omterm{def:g1:plants-around-us:parts}{stengel} overgenomen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:5}
Waarom groeien veel hooggebergteplanten als strakke kussens?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:6}
Geef twee \omterm{def:g4:adapted-to-their-world:adaptation}{aanpassingen} van bejaagde \omterm{def:g1:animals-around-us:animal}{dieren} en twee van jagers, uit
\cref{prop:g4:adapted-to-their-world:answers}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:7}
Vertel \cref{ex:g4:adapted-to-their-world:mystery} na: welke drie
kenmerken verrieden de mol, en wat zei elk ervan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:8}
Probeer het: kies een \omterm{def:g1:animals-around-us:animal}{dier} dat je kunt bekijken --- een eend, een kat,
een spin --- en laat \cref{met:g4:adapted-to-their-world:read} erover
lopen: drie kenmerken, drie opgeloste problemen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:9}
De woestijnvos heeft enorme \omterm{def:g1:the-five-senses:sense}{oren}; de poolvos piepkleine. Verklaar
allebei met de regel over kou uit
\cref{prop:g4:adapted-to-their-world:answers} --- wat moeten grote \omterm{def:g1:the-five-senses:sense}{oren}
met warmte doen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:10}
Een raadseldier: een gestroomlijnd lichaam, zwemvliezen aan de
achterpoten, dichte \omterm{def:g1:animals-around-us:covering}{vacht} waar het water van afloopt, snorharen, en het
kan zijn adem goed inhouden. Reconstrueer zijn wereld met
\cref{met:g4:adapted-to-their-world:read}. (Het bouwt dammen.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:adapted-to-their-world:11}
De walvis (\cref{ex:g3:sorting-living-things:surprises}) is een zoogdier
met de vorm van een vis: gestroomlijnd, met vinnen, glad. De forel is
een vis met dezelfde vorm. Wat laat deze gelijkenis tussen twee
\emph{niet-verwante} zwemmers zien over \omterm{def:g4:adapted-to-their-world:adaptation}{aanpassingen} --- wie bepaalde de
vorm, de familie of het water?
\end{exercise}
