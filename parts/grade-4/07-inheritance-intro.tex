\chapter{Traits Run in Families}\label{ch:g4:traits}

Why do relatives sometimes share freckles, curly hair or a love of
music? Some features are inherited; others come from learning and
\omterm{def:g4:traits:env}{environment}. This chapter introduces \omterm{def:g4:traits:trait}{traits}, family patterns and the
difference between what you are born with and what you practise.

\section{What is a trait?}

\begin{definition}[Trait]\label{def:g4:traits:trait}
A \emph{trait}\index{trait} is a feature of a \omterm{def:g1:living-nonliving:living}{living thing} --- for
example \omterm{def:g1:senses:five}{eye} colour, height, \omterm{def:g4:plants-need:parts}{leaf} shape, or \omterm{def:g2:plant-parts:main}{flower} colour. Traits can be
described and compared among individuals.
\end{definition}

\begin{example}[People and peas]\label{ex:g4:traits:examples}
Humans show \omterm{def:g4:traits:trait}{traits} such as attached or free earlobes (with care: many
\omterm{def:g4:traits:trait}{traits} are not simple). Pea \omterm{def:g1:plants-animals:plant}{plants} may have purple or white \omterm{def:g2:plant-parts:main}{flowers}.
Dogs may have short or long coats.
\end{example}

\section{Inherited and acquired}

\begin{definition}[Inherited trait]\label{def:g4:traits:inherited}
An \emph{inherited trait}\index{trait!inherited} is a feature passed
from parents to \omterm{def:g2:growing:offspring}{offspring} through information in reproductive \omterm{def:g4:cells:cell}{cells}.
Children resemble parents partly because of inheritance.
\end{definition}

\begin{definition}[Acquired trait]\label{def:g4:traits:acquired}
An \emph{acquired trait}\index{trait!acquired} develops during life
because of experience, practice, injury or \omterm{def:g4:traits:env}{environment} --- for example
a scar, a language you learn, or stronger \omterm{def:g3:bones:muscle}{muscles} from training.
Acquired \omterm{def:g4:traits:trait}{traits} are not written into the next generation the way
\omterm{def:g4:traits:inherited}{inherited traits} are.
\end{definition}

\begin{example}[Football]\label{ex:g4:traits:sport}
Being tall may be influenced by inheritance. Being a skilled goalkeeper
requires practice. Both matter in sport; only the inherited part is
passed biologically.
\end{example}

\section{Family resemblance}

\begin{proposition}[Patterns in families]\label{prop:g4:traits:family}
\omterm{def:g2:growing:offspring}{Offspring} often share \omterm{def:g4:traits:trait}{traits} with parents and other relatives because
they inherit information from their parents. Not every \omterm{def:g4:traits:trait}{trait} appears in
every child, and siblings can look different.
\end{proposition}
\admitted

\begin{method}[Family trait chart (optional)]\label{met:g4:traits:chart}
\begin{enumerate}
  \item Choose a harmless \omterm{def:g4:traits:trait}{trait} (for example freckles: present/absent).
  \item With permission, note the \omterm{def:g4:traits:trait}{trait} in family members.
  \item Look for patterns without assuming simple rules for every
        feature.
  \item Respect privacy: \omterm{def:g2:healthy:health}{health} information is personal.
\end{enumerate}
\end{method}

\section{Environment shapes living things}

\begin{definition}[Environment and traits]\label{def:g4:traits:env}
The \emph{environment}\index{environment (traits)} includes \omterm{def:g2:food-energy:food}{food},
sunlight, training, climate and care. It can change how \omterm{def:g4:traits:trait}{traits} show
(height depends on genes and nutrition) and can create acquired
features.
\end{definition}

\begin{example}[Identical starting points]\label{ex:g4:traits:plants}
Two cuttings from one \omterm{def:g1:plants-animals:plant}{plant} share the same inherited information. Grown
in deep shade versus bright \omterm{def:g4:plants-need:needs}{light}, they may look different --- same
inheritance, different \omterm{def:g4:traits:env}{environment}.
\end{example}
