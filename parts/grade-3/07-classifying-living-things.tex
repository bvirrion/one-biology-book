\chapter{Classifying Living Things}\label{ch:g3:classify}

Sorting is a scientist's habit. We have sorted living and nonliving,
\omterm{def:g1:plants-animals:plant}{plants} and \omterm{def:g1:plants-animals:animal}{animals}, \omterm{def:g2:animal-groups:vert}{vertebrates} and \omterm{def:g3:insects:insect}{insects}. Now we practise
\emph{\omterm{def:g3:classify:class}{classification}}: putting \omterm{def:g2:what-alive:organism}{organisms} into groups that share
features.

\section{Why classify?}

\begin{definition}[Classification]\label{def:g3:classify:class}
\emph{Classification}\index{classification} is sorting \omterm{def:g1:living-nonliving:living}{living things}
into groups by shared \omterm{def:g3:classify:char}{characteristics}. Good groups help us name,
compare and remember \omterm{def:g2:what-alive:organism}{organisms}.
\end{definition}

\begin{example}[Everyday sorting]\label{ex:g3:classify:everyday}
You sort socks by colour and books by topic. Biologists sort
\omterm{def:g2:what-alive:organism}{organisms} by \omterm{def:g1:my-body:body}{body} plan, how they get \omterm{def:g2:food-energy:food}{food}, and how they have young.
\end{example}

\section{Keys and clues}

\begin{method}[Simple yes/no key]\label{met:g3:classify:key}
To identify an \omterm{def:g2:what-alive:organism}{organism}, ask yes/no questions in order:
\begin{enumerate}
  \item Does it make its own \omterm{def:g2:food-energy:food}{food} from light? $\to$ \omterm{def:g1:plants-animals:plant}{plant} side.
  \item If \omterm{def:g1:plants-animals:animal}{animal}: does it have a backbone?
  \item If \omterm{def:g2:animal-groups:vert}{vertebrate}: fur, feathers, scales, or moist \omterm{def:g1:senses:five}{skin}?
  \item If \omterm{def:g2:animal-groups:invert}{invertebrate}: six \omterm{def:g1:my-body:parts}{legs}? eight \omterm{def:g1:my-body:parts}{legs}? no \omterm{def:g1:my-body:parts}{legs}?
\end{enumerate}
Each answer narrows the group.
\end{method}

\begin{example}[Use the key]\label{ex:g3:classify:use}
A robin: \omterm{def:g1:plants-animals:animal}{animal} $\to$ backbone $\to$ feathers $\to$ \omterm{def:g2:animal-groups:five}{bird}. A daisy:
makes \omterm{def:g2:food-energy:food}{food} from light $\to$ \omterm{def:g1:plants-animals:plant}{plant}. An ant: \omterm{def:g1:plants-animals:animal}{animal} $\to$ no backbone
$\to$ six \omterm{def:g1:my-body:parts}{legs} $\to$ \omterm{def:g3:insects:insect}{insect}.
\end{example}

\begin{definition}[Characteristic]\label{def:g3:classify:char}
A \emph{characteristic}\index{characteristic} is a feature you can
observe: number of \omterm{def:g1:my-body:parts}{legs}, \omterm{def:g1:my-body:body}{body} covering, \omterm{def:g2:habitats:habitat}{habitat}, how it feeds.
Choose characteristics that stay true for the group.
\end{definition}

\begin{proposition}[Shared features]\label{prop:g3:classify:shared}
\omterm{def:g2:what-alive:organism}{Organisms} in the same group share important \omterm{def:g3:classify:char}{characteristics}. The
closer the group, the more they share.
\end{proposition}
\admitted
