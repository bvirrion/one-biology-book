\chapter{How Plants Spread}\label{ch:g6:how-plants-spread}

A new road cutting is bulldozed in March: raw, bare earth, not
a green speck. Come back in September and it is stitched all
over with \omterm{prop:g2:from-seed-to-plant:cycle}{seedlings} --- poppies, dandelions, willow shoots,
grasses. Nobody sowed it. Yet \omterm{def:g1:plants-around-us:plant}{plants}, which cannot walk a
single step, colonized the bare ground within months. This
chapter is about their travel department.

\section{Plants travel as seeds and spores}

\begin{proposition}[Sedentary plants, traveling beginnings]\label{prop:g6:how-plants-spread:travel}
An adult \omterm{def:g1:plants-around-us:plant}{plant} is fixed for life
(\cref{def:g1:plants-around-us:plant}) --- so a \omterm{def:g5:biodiversity:species}{species}
spreads by sending out its \emph{beginnings}: \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} and
\omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} (\cref{ch:g5:flower-to-fruit}), \omterm{def:g4:plants-without-seeds:spore}{spores}
(\cref{def:g4:plants-without-seeds:spore}), or \omterm{ex:g4:plants-without-seeds:runner}{runners} and
other vegetative advances
(\cref{def:g4:plants-without-seeds:vegetative}). Occupying
new ground this way is called
\emph{colonization}\index{colonization}.
\end{proposition}
\admitted

\begin{definition}[Dispersal]\label{def:g6:how-plants-spread:dispersal}
\emph{Dispersal}\index{dispersal} is the carrying of \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds},
\omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} or \omterm{def:g4:plants-without-seeds:spore}{spores} away from the parent \omterm{def:g1:plants-around-us:plant}{plant}. The carriers are
few and ancient: the \emph{wind}, \emph{\omterm{def:g1:animals-around-us:animal}{animals}},
\emph{water} --- and some \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} that fling their \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}
themselves.
\end{definition}

\begin{example}[Why leave home at all?]\label{ex:g6:how-plants-spread:whyleave}
A \omterm{def:g2:from-seed-to-plant:seed}{seed} dropped at the parent's foot \omterm{def:g2:from-seed-to-plant:germination}{germinates} in the
parent's shade, among the parent's \omterm{def:g1:plants-around-us:parts}{roots} --- competing for
the very \omterm{def:g6:exploring-our-environment:conditions}{light} and water of
\cref{prop:g4:what-plants-need:needs} with a rival a
hundred times its size. \omterm{def:g6:how-plants-spread:dispersal}{Dispersal} is the escape from that
losing match: distance is the first gift a \omterm{def:g1:plants-around-us:plant}{plant} gives its
young.
\end{example}

\section{The carriers, and the fruits built for them}

\begin{proposition}[Built to match the carrier]\label{prop:g6:how-plants-spread:builds}
\omterm{prop:g3:flowers-fruits-seeds:transform}{Fruit} and \omterm{def:g2:from-seed-to-plant:seed}{seed} shapes match their carriers ---
\cref{def:g4:adapted-to-their-world:adaptation}'s fit,
applied to luggage:
\begin{itemize}
  \item \emph{wind}: parachutes (the dandelion's plumed
        \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}), wings (maple and ash keys that spin down
        slantwise), or dust-fine \omterm{def:g4:plants-without-seeds:spore}{spores} that simply float
        (the fern's millions);
  \item \emph{\omterm{def:g1:animals-around-us:animal}{animals}, outside}: burrs --- burdock,
        cleavers --- whose hooks grip \omterm{def:g1:animals-around-us:covering}{fur} and socks and ride
        until groomed off;
  \item \emph{\omterm{def:g1:animals-around-us:animal}{animals}, inside}: sweet fleshy \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} ---
        cherries, blackberries, rose hips --- eaten whole;
        the tough-coated \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} pass through unharmed and are
        deposited elsewhere, with fertilizer
        (\cref{exo:g3:flowers-fruits-seeds:10}'s answer,
        completed);
  \item \emph{water}: floating \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} --- the alder's tiny
        rafts, the coconut's ocean-going \omterm{def:g1:animals-around-us:covering}{shell};
  \item \emph{self-flinging}: pods that dry, tense and snap
        --- gorse and violets shoot \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} metres from the
        parent.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Reading the roadside]\label{ex:g6:how-plants-spread:roadside}
September's road cutting, decoded: dandelions and willows ---
plumed \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}, flown in on the wind; poppies --- their tiny
\omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} lay already in the moved earth, waiting as only \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}
can (\cref{exo:g2:from-seed-to-plant:9}); blackberry
\omterm{prop:g2:from-seed-to-plant:cycle}{seedlings} under the fence posts --- deposited by perching
\omterm{prop:g3:sorting-living-things:groups}{birds}, fertilizer included; burdock by the path --- carried
in on a dog. Every colonist's arrival route is written in
its equipment.
\end{example}

\begin{method}[Naming the carrier from the fruit]\label{met:g6:how-plants-spread:name}
Given a \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit} or \omterm{def:g2:from-seed-to-plant:seed}{seed}:
\begin{enumerate}
  \item look for flight gear --- plumes, wings, dust
        fineness: the wind;
  \item look for hooks: \omterm{def:g1:animals-around-us:animal}{animal} \omterm{def:g1:animals-around-us:covering}{fur}, outside;
  \item fleshy, colored, sweet? \omterm{def:g1:animals-around-us:animal}{animal} gut, inside ---
        check the \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} for tough coats;
  \item buoyant, near water? water;
  \item a dried, twisted, sprung pod? self-flinging;
  \item test your verdict: drop it, float it, press it to a
        woolly sleeve.
\end{enumerate}
\end{method}

\section{Colonizing without seeds}

\begin{example}[The ground advance]\label{ex:g6:how-plants-spread:vegetative}
Alongside the airborne arrivals, some colonists walk on
their own vegetative feet
(\cref{ch:g4:plants-without-seeds}): couch grass advances
underground by creeping \omterm{def:g1:plants-around-us:parts}{stems}, metre by metre; brambles arch
over and \omterm{def:g1:plants-around-us:parts}{root} where their tips touch; the strawberry's
\omterm{ex:g4:plants-without-seeds:runner}{runners} hop across open ground. Slow, short-range --- and
unstoppable across a neglected garden, as any gardener
knows.
\end{example}

\begin{remark}[Two-speed colonization]\label{rem:g6:how-plants-spread:twospeeds}
The two systems divide the work. \omterm{prop:g3:flowers-fruits-seeds:transform}{Seeds} and \omterm{def:g4:plants-without-seeds:spore}{spores} are the
\emph{long-range} service --- rare lucky landings far away,
the dandelion crossing the town. Vegetative advance is the
\emph{short-range} service --- dense, sure occupation of the
ground next door. The bramble patch spreading from the hedge
uses both at once: arching \omterm{def:g1:plants-around-us:parts}{stems} for the metre, bird-carried
\omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} for the mile.
\end{remark}

\section{Problem: The Abandoned Field}

\begin{problem}[{Weekend problem --- ten years of
colonization, watched from one gate}]\label{pb:g6:how-plants-spread:1}
A ploughed field is abandoned in autumn. A naturalist
photographs it from the gate every June for ten years, and
keeps notes.

\medskip\noindent\textbf{Part I --- Year one.}
\begin{enumerate}
  \item June of year one: the bare earth is dotted with
        poppies, dandelions and thistles. For each, name
        the likely source and route --- poppies from the
        soil itself, the others airborne. What equipment
        made the airborne arrivals possible?
  \item No blackberries appear in the field's middle in
        year one, but three \omterm{prop:g2:from-seed-to-plant:cycle}{seedlings} stand under the old
        fence posts. Explain the pattern with the
        \omterm{def:g1:animals-around-us:animal}{inside-animal} service --- why exactly under the
        posts?
  \item The naturalist finds no \omterm{prop:g4:plants-without-seeds:damp}{ferns} anywhere, though a
        wood full of them lies upwind. \omterm{prop:g4:plants-without-seeds:damp}{Ferns}' \omterm{def:g4:plants-without-seeds:spore}{spores}
        certainly arrived (\cref{ex:g4:plants-without-seeds:fern}
        counts their millions). What condition of the open
        field, by \cref{prop:g4:plants-without-seeds:damp},
        refused them?
  \item Couch grass invades from the field's edge only, a
        metre or two a year. Which \omterm{prop:g6:how-plants-spread:travel}{colonization} system is
        this, and why can it not appear in the field's
        middle the way dandelions did?
\end{enumerate}

\medskip\noindent\textbf{Part II --- The years after.}
\begin{enumerate}[resume]
  \item By year three, grasses and tall herbs have closed
        the bare ground, and the poppies have vanished from
        the standing vegetation. Where are the poppies now,
        and in what form? (Recall the road cutting and
        \cref{prop:g6:life-through-seasons:herbs}.)
  \item From year four, bramble arcs and young hawthorns
        dot the field, thickest near the hedges and under
        the fence line. Name the two services that placed
        them, and the evidence in the pattern.
  \item By year eight, willow and birch saplings stand
        taller than the brambles --- both trees with tiny
        plumed or \omterm{def:g6:exploring-our-environment:conditions}{dust-light} \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}. Why are the heavy-seeded
        oaks and chestnuts still almost absent, and what
        \omterm{def:g1:animals-around-us:animal}{animal} courier eventually \omterm{def:g1:plants-around-us:plant}{plants} even them?
  \item The photographs show the field's cast changing
        year by year although no gardener touched it. State
        the driving rule: what does each year's vegetation
        change about the field's own \omterm{def:g6:exploring-our-environment:conditions}{conditions} --- \omterm{def:g6:exploring-our-environment:conditions}{light}
        at ground level above all --- and who does that
        change favor next?
\end{enumerate}

\medskip\noindent\textbf{Part III --- The general lesson.}
\begin{enumerate}[resume]
  \item Order the colonists by arrival: soil's waiting
        \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}, wind-riders, animal-carried shrubs,
        heavy-seeded trees. Pair each wave with the
        \omterm{def:g6:how-plants-spread:dispersal}{dispersal} service that timed it.
  \item A town square's pavement cracks sprout the same
        year-one cast --- poppies, dandelions --- but never
        reach year four's shrubs. Give two reasons the
        succession stalls there.
  \item The naturalist claims: ``left alone, this field
        will be woodland.'' Defend the claim with the
        ten-year trend --- and name the meadow exception of
        \cref{prop:g5:people-and-nature:managed} that
        proves ``left alone'' is the operative phrase.
  \item Summarize the whole problem in two sentences using
        the words \emph{\omterm{def:g6:how-plants-spread:dispersal}{dispersal}}, \emph{\omterm{def:g6:exploring-our-environment:conditions}{conditions}} and
        \emph{\omterm{prop:g6:how-plants-spread:travel}{colonization}}.
\end{enumerate}
\end{problem}
