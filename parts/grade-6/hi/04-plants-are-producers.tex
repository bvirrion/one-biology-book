\chapter{पौधे उत्पादक हैं}\label{ch:g6:plants-are-producers}

सींचे हुए एक गमले को तौलो, उसमें बलूत का एक \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} बोओ, और सालों बाद लौटकर
उस नन्हे बलूत को तौलो: किलो भर लकड़ी, छाल, \omterm{def:g1:plants-around-us:parts}{पत्ती} और \omterm{def:g1:plants-around-us:parts}{जड़} --- जबकि गमले की
मिट्टी का वज़न मुश्किल से एक ग्राम भी घटा है
(\cref{ex:g4:what-plants-need:tree} ने पहली बार यह अचरज सामने रखा था)।
किलो भर \omterm{def:g1:plants-around-us:tree}{पेड़}, आख़िर किससे बना? यह अध्याय उसी उत्तर को पक्का करता है ---
और \omterm{def:g1:plants-around-us:plant}{पौधों} के उस अनोखे पेशे को नाम देता है।

\section{कार्बनिक और खनिज पदार्थ}

\begin{definition}[खनिज और कार्बनिक पदार्थ]\label{def:g6:plants-are-producers:matter}
जीवविज्ञानी के लिए पदार्थ दो तरह का होता है:
\begin{itemize}
  \item \emph{खनिज पदार्थ}\index{खनिज पदार्थ}: निर्जीव दुनिया का
        पदार्थ --- पानी, हवा की गैसें, चट्टान, और
        \cref{prop:g4:what-plants-need:needs} के घुले हुए \omterm{prop:g4:what-plants-need:needs}{खनिज लवण};
  \item \emph{कार्बनिक पदार्थ}\index{कार्बनिक पदार्थ}: सजीवों का बनाया
        और उन्हीं के शरीर बनाने वाला पदार्थ --- लकड़ी, \omterm{def:g1:plants-around-us:parts}{पत्ती}, \omterm{def:g3:bones-and-muscles:muscle}{पेशी},
        चरबी, शक्कर, ऊन। जलने वाला, गलने वाला, और हमेशा किसी जीवित
        बनाने वाले तक ले जाने वाला।
\end{itemize}
\end{definition}

\begin{example}[एक जेब को छाँटना]\label{ex:g6:plants-are-producers:sorting}
सैर के बाद कोट की जेब से निकला: एक कंकड़ --- खनिज; एक हेज़लनट ---
कार्बनिक (उसे हेज़ल ने बनाया); प्लास्टिक का एक बटन --- अजीब बात है, पर
मूल से कार्बनिक (प्लास्टिक तेल से बनता है, और तेल पुराने सजीवों के बदले
हुए अवशेष हैं); बारिश की एक बूँद --- खनिज; ऊन का एक दस्ताना ---
कार्बनिक (वह ऊन किसी भेड़ ने बनाई थी)।
\end{example}

\section{उत्पादकों का पेशा}

\begin{proposition}[पौधे खनिज पदार्थ से कार्बनिक पदार्थ बनाते हैं]\label{prop:g6:plants-are-producers:produce}
हरे \omterm{def:g1:plants-around-us:plant}{पौधे} को सिर्फ़ \emph{खनिज} सामग्री दो --- मिट्टी से पानी और घुले
हुए लवण, और \emph{हवा} से ली गई एक गैस --- और साथ में \emph{\omterm{def:g6:exploring-our-environment:conditions}{प्रकाश}} की
\omterm{prop:g3:balanced-diet:jobs}{ऊर्जा}, तो वह \emph{कार्बनिक} पदार्थ बना देता है: अपनी शक्कर, और उससे
लकड़ी, \omterm{def:g1:plants-around-us:parts}{पत्ती} तथा \omterm{def:g1:plants-around-us:parts}{जड़}। यही वह पेशा है जिसे समझाने का वादा
\cref{prop:g4:what-plants-need:make} ने किया था: \omterm{def:g1:plants-around-us:plant}{पौधे}
\emph{उत्पादक}\index{उत्पादक} हैं --- यानी विशुद्ध खनिज सामग्री से नया
\omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} बनाने वाले अकेले कारीगर। \omterm{def:g1:plants-around-us:parts}{पत्तियों} का हरा पदार्थ \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश}
पकड़ने की मशीनरी है; और निर्माण दिन में, \omterm{def:g1:plants-around-us:parts}{पत्ती} में ही होता है।
\end{proposition}
\admitted

\begin{example}[सबूत, एक जगह जुटे हुए]\label{ex:g6:plants-are-producers:evidence}
पहले के तीन प्रेक्षण अब आपस में जुड़ जाते हैं। तौला हुआ गमला: बलूत के
किलो मिट्टी से नहीं आए --- वे ज़्यादातर हवा और पानी से आए।
\cref{ex:g4:what-plants-need:pale} का पीला क़ैदी: \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} नहीं, तो
निर्माण नहीं --- अँधेरे में \omterm{def:g1:plants-around-us:plant}{पौधा} खनिजों से भरे भंडार के बावजूद भूखा मर
जाता है। और \cref{exo:g4:what-plants-need:11} का खनिज-भूखा पुदीना: लवण
ज़रूरी तो हैं --- पर थोड़ी मात्रा के रूप में, थोक के रूप में नहीं। थोक
हवा और पानी से, \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} से, और नमक-मिर्च मिट्टी से।
\end{example}

\begin{remark}[गैसों की अदला-बदली, दोनों तरफ़]\label{rem:g6:plants-are-producers:gas}
\omterm{def:g1:plants-around-us:parts}{पत्ती} हवा से जो गैस लेती है वह \emph{कार्बन डाइऑक्साइड} है --- ठीक वही
गैस जो साँस लेने वाला हर जीव छोड़ता है
(\cref{prop:g4:breathing:exchange}); और यह निर्माण हवा में
\emph{ऑक्सीजन} छोड़ता है --- ठीक वही गैस जो साँस लेने वाले हर जीव को
चाहिए। \omterm{def:g1:plants-around-us:plant}{पौधे} भी सारे सजीवों की तरह दिन-रात साँस लेते हैं; पर दिन में
काम करती \omterm{def:g1:plants-around-us:parts}{पत्ती} का निर्माण उसकी साँस से कहीं आगे निकल जाता है। दुनिया की
साँस लेने लायक़ हवा \omterm{def:g5:food-webs:roles}{उत्पादकों} का उपोत्पाद है --- हरे रंग का एक और क़र्ज़,
जो हर \omterm{def:g1:animals-around-us:animal}{जंतु} पर चढ़ा है।
\end{remark}

\section{उत्पादक सबका पेट भरते हैं --- और अपने लिए जमा भी करते हैं}

\begin{proposition}[पहली कड़ी, समझाई हुई]\label{prop:g6:plants-are-producers:firstlink}
\Cref{prop:g3:food-chains:start} ने तय किया था कि हर \omterm{def:g3:food-chains:chain}{आहार शृंखला} किसी
\omterm{def:g1:plants-around-us:plant}{पौधे} से शुरू होती है; अब उस फ़ैसले का कारण भी है। सिर्फ़ \omterm{def:g5:food-webs:roles}{उत्पादक} ही \omterm{def:g6:plants-are-producers:matter}{खनिज
पदार्थ} से \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} बनाते हैं; हर \omterm{def:g5:food-webs:roles}{उपभोक्ता}
(\cref{def:g5:food-webs:roles}) खाए हुए \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} को सिर्फ़
\emph{बदल} सकता है। इसलिए हर जीवित शरीर का सारा \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} ---
ख़रगोश, लोमड़ी, बाज़, तुम --- पहले किसी \omterm{def:g1:plants-around-us:parts}{पत्ती} में ही बना था, और
\cref{prop:g5:food-webs:pyramid} का आहार-पिरामिड हरी मंज़िल पर इसीलिए
खड़ा है कि पदार्थ जीवित दुनिया में वहीं से घुसता है।
\end{proposition}
\admitted

\begin{example}[पौधे का अपना भंडार-घर]\label{ex:g6:plants-are-producers:pantry}
\omterm{def:g1:plants-around-us:plant}{पौधा} दिन की ज़रूरत से जो ज़्यादा बनाता है, उसे कार्बनिक भंडार के रूप में
रख लेता है: आलू का \omterm{ex:g4:plants-without-seeds:bulbtuber}{कंद} और ट्यूलिप का \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद}
(\cref{ex:g4:plants-without-seeds:bulbtuber}), सेम के दो मोटे टुकड़े
(\cref{def:g2:from-seed-to-plant:seed}), गाजर की मोटी \omterm{def:g1:plants-around-us:parts}{जड़}
(\cref{exo:g1:plants-around-us:10}), और चुकंदर की मिठास। इनमें से हर एक
गर्मियों का \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} है, जो पदार्थ के रूप में जमा किया गया --- और ठीक यही
वह चीज़ है जिसे किसान तथा खाने वाले काटते हैं।
\end{example}

\begin{method}[किसी भी भोजन को उसकी पत्ती तक पीछा करना]\label{met:g6:plants-are-producers:trace}
मेज़ पर रखा कोई भी भोजन लो:
\begin{enumerate}
  \item अगर वह \omterm{def:g1:plants-around-us:plant}{पौधे} का पदार्थ है --- रोटी का गेहूँ, आलू, सेब --- तो
        वह सीधे \omterm{def:g5:food-webs:roles}{उत्पादक} का बनाया हुआ है: उस \omterm{def:g1:plants-around-us:plant}{पौधे} और उसके भंडार-अंग का
        नाम बताओ;
  \item अगर वह \omterm{def:g1:animals-around-us:animal}{जंतु} का पदार्थ है --- पनीर, \omterm{def:g2:animals-grow-up:born}{अंडा}, माँस --- तो शृंखला
        में नीचे उतरो: उस \omterm{def:g1:animals-around-us:animal}{जंतु} ने क्या खाया? और उस भोजन का बनाने वाला
        कौन? \omterm{def:g1:plants-around-us:plant}{पौधे} तक चलते जाओ;
  \item शृंखला को तीरों के साथ लिखो
        (\cref{def:g3:food-chains:chain});
  \item निष्कर्ष हमेशा वही: रास्ता किसी \omterm{def:g1:plants-around-us:parts}{पत्ती} पर, \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} में जाकर
        ख़त्म होता है।
\end{enumerate}
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:1}
\omterm{def:g6:plants-are-producers:matter}{खनिज पदार्थ} और \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} की परिभाषा दो, हर एक के दो उदाहरणों के

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:3}
खनिज या कार्बनिक में छाँटो: ओस, घोंघे का \omterm{def:g1:animals-around-us:covering}{खोल}, मोम, ग्रेनाइट, खाने का

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:4}
अगर बलूत के किलो मिट्टी से मुश्किल से आए, तो वे ज़्यादातर कहाँ से आए?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:5}
काम करती \omterm{def:g1:plants-around-us:parts}{पत्ती} हवा से कौन-सी गैस लेती है, और कौन-सी छोड़ती है? पहली को
कौन छोड़ता है और दूसरी किसे चाहिए?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:6}
\cref{ex:g6:plants-are-producers:pantry} से \omterm{def:g1:plants-around-us:plant}{पौधों} के चार भंडार-अंग बताओ,
और यह भी कि हर एक क्या जमा रखता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:7}
अँधेरी अलमारी वाला \omterm{def:g1:plants-around-us:plant}{पौधा} \omterm{prop:g4:what-plants-need:needs}{खनिज लवणों} से भरे भंडार के बावजूद भूखा क्यों

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:8}

\omterm{def:g5:food-webs:roles}{उत्पादक} और \omterm{def:g5:food-webs:roles}{उपभोक्ता} के भेद से उत्तर दो।

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:9}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:10}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:11}
पानी के \omterm{def:g1:plants-around-us:plant}{पौधों} और \omterm{prop:g3:sorting-living-things:groups}{मछलियों} वाला बंद, उजाले में रखा काँच-घर महीनों चल सकता
है; पर बंद \emph{अँधेरा} काँच-घर दिनों में बिगड़ जाता है। दोनों नतीजे
\cref{rem:g6:plants-are-producers:gas} से समझाओ।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:plants-are-producers:12}
एक पुराना प्रयोग: टब में पाँच साल उगाए गए बेंत के \omterm{def:g1:plants-around-us:plant}{पौधे} का वज़न कई किलो
बढ़ा; और टब की सूखी मिट्टी का वज़न सिर्फ़ कुछ ग्राम घटा। प्रयोग करने
वाले ने निष्कर्ष निकाला कि ``\omterm{def:g1:plants-around-us:tree}{पेड़} अकेले पानी से बना है''। उसके निष्कर्ष
का कौन-सा हिस्सा सही था, कौन-सा अधूरा --- और आज का जीवविज्ञानी कौन-सी
दूसरी खनिज सामग्री जोड़ेगा, जो उसे मालूम नहीं थी?
\end{exercise}

\section{समस्या: सब्ज़ी उगाने वाले का साल}

\begin{problem}[{सप्ताहांत समस्या --- किसी सब्ज़ी-बग़ीचे का प्रकाश,
पत्तियाँ और बहीखाता}]\label{pb:g6:plants-are-producers:1}
सब्ज़ी उगाने वाला एक किसान आलू, गाजर, सेम और सलाद उगाता है, और जो भीतर
जाता है तथा जो बाहर आता है, उसका पक्का बहीखाता रखता है।

\medskip\noindent\textbf{भाग I --- बहीखाते की पहेली।}
\begin{enumerate}
  \item बग़ीचा साल में दो टन सब्ज़ियाँ देता है; और किसान सिर्फ़ खाद की
        बोरियाँ बिखेरता है। पहेली तौलो: अगर वे टन बोरियों से नहीं आते,
        तो कहाँ से आते हैं? (उत्तर \omterm{def:g5:food-webs:roles}{उत्पादकों} के पेशे से दो।)
  \item उपज को वर्गीकृत करो: आलू, गाजर, सेम का \omterm{def:g2:from-seed-to-plant:seed}{बीज}, सलाद की \omterm{def:g1:plants-around-us:parts}{पत्ती} ---
        हर एक की \omterm{def:g1:plants-around-us:plant}{पौधे} के लिए भूमिका बताओ (भंडार-अंग, काम करता अंग,
        \omterm{def:g2:from-seed-to-plant:seed}{बीज} का भंडार)।
  \item किसान कहता है: ``मेरा असली आपूर्तिकर्ता दिन के उजाले में
        सामान पहुँचाता है।'' इसे
        \cref{prop:g6:plants-are-producers:produce} की भाषा में
        बदलो।
  \item अगर टनों के लिए नहीं, तो किसान को खाद की ज़रूरत आख़िर है ही
        क्यों? (रेत वाला पुदीना याद करो।)
\end{enumerate}

\medskip\noindent\textbf{भाग II --- ऋतुओं को पढ़ना।}
\begin{enumerate}[resume]
  \item जून के सलाद, जो एक हफ़्ते तक जाँच के लिए कपड़े के नीचे छाया में
        रखे गए, पीले और खिंचे हुए निकलते हैं। बताओ कि यह किस प्रयोग को
        दोहराता है, और उसने कौन-सी सामग्री हटा दी थी।
  \item आलू की क़तारों पर मिट्टी चढ़ाई जाती है और उनके घने ऊपरी हिस्सों
        की तारीफ़ होती है: ``अच्छे ऊपरी हिस्से, अच्छे \omterm{ex:g4:plants-without-seeds:bulbtuber}{कंद}।'' इस कहावत
        को समझाओ --- ऊपर काम करती \omterm{def:g1:plants-around-us:parts}{पत्तियों} और नीचे फूलते भंडार के बीच
        क्या संबंध है?
  \item अगस्त का सूखा बढ़ना रोक देता है, जबकि सूरज पूरी आग बरसा रहा है।
        कौन-सी खनिज सामग्री कम पड़ गई, और \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} उसकी जगह क्यों नहीं ले
        सकता?
  \item सितंबर में किसान सेम के \omterm{def:g1:plants-around-us:plant}{पौधे} काटकर हटा देता है पर उनकी \omterm{def:g1:plants-around-us:parts}{जड़ें}
        मिट्टी में ही जोत देता है। एक पड़ोसी इसे बरबादी कहता है। पदार्थ
        की परिभाषाओं से इस चलन का बचाव करो --- मिट्टी को क्या लौटाया
        जा रहा है, और किसके लिए?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- बग़ीचे से आगे।}
\begin{enumerate}[resume]
  \item किसान का सूअर छोटे आलू खाता है; और परिवार सूअर को खाता है।
        शृंखला लिखो, और बताओ कि उसका \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} मूल रूप से कहाँ
        बना था।
  \item एक ग्राहक दावा करती है कि उसका सलाद ``ज़्यादातर धूप'' है। इस
        दावे की जाँच करो: सलाद की \omterm{def:g1:plants-around-us:parts}{पत्ती} की असली सामग्री गिनाओ और सूरज
        की ठीक-ठीक भूमिका भी।
  \item पाला रात भर में आख़िरी सलादों को मार देता है; और नवंबर तक वे
        वहीं पड़े-पड़े गल जाते हैं। उनका \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} ग़ायब नहीं
        होता। \cref{ch:g6:decomposers-and-soil} का अंदाज़ा लगाओ: उसे
        कौन लेता है, और किस अवस्था की ओर?
  \item बग़ीचे का सार एक वाक्य में लिखो, जिसमें ये शब्द हों: \omterm{def:g5:food-webs:roles}{उत्पादक},
        \omterm{def:g6:plants-are-producers:matter}{खनिज पदार्थ}, \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ}, \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश}।
\end{enumerate}
\end{problem}
