\chapter{Gezonde keuzes}\label{ch:g5:healthy-choices}

Inmiddels heeft dit boek de meeste grote diensten van je lichaam
bezocht: het \omterm{def:g4:journey-of-food:tract}{spijsverteringskanaal}, de \omterm{def:g4:breathing:organs}{longen}, het \omterm{def:g4:heart-and-blood:heart}{hart} met zijn
bloedrivieren, de \omterm{def:g3:bones-and-muscles:muscle}{spieren} en \omterm{def:g3:bones-and-muscles:skeleton}{botten}, de commanderende \omterm{def:g5:brain-in-command:system}{hersenen}. Dit
slothoofdstuk van het jaar stelt een praktische vraag: nu je al die
machinerie kent, hoe laat je haar goed draaien --- en wie beslist? Want
vanaf ongeveer jouw leeftijd worden steeds meer beslissingen die van jou.

\section{Waar de machinerie om vraagt}

\begin{proposition}[Het dagelijkse contract van het lichaam]\label{prop:g5:healthy-choices:contract}
Zet de verzoeken van de vorige hoofdstukken naast elkaar en er verschijnt
een dagelijks contract:
\begin{itemize}
  \item \emph{\omterm{prop:g1:healthy-habits:needs}{gevarieerd eten} en water} --- brandstof en bouwmateriaal
        (\cref{prop:g3:balanced-diet:jobs}), door het \omterm{prop:g4:journey-of-food:absorb}{bloed} bij elke \omterm{def:g5:first-look-at-cells:cell}{cel}
        bezorgd;
  \item \emph{beweging} --- \omterm{def:g3:bones-and-muscles:muscle}{spieren}, \omterm{def:g3:bones-and-muscles:skeleton}{botten}, \omterm{def:g4:heart-and-blood:heart}{hart} en \omterm{def:g4:breathing:organs}{longen} worden
        allemaal sterker van gebruik
        (\cref{met:g3:bones-and-muscles:care},
        \cref{rem:g4:heart-and-blood:rest});
  \item \emph{\omterm{prop:g1:healthy-habits:needs}{slaap}} --- het opruimwerk van de \omterm{def:g5:brain-in-command:system}{hersenen} en het herstel
        van het lichaam (\cref{prop:g5:brain-in-command:protect}); een
        kind van deze leeftijd heeft ongeveer tien uur nodig;
  \item \emph{schone lucht, schone handen, verzorgde tanden} --- de
        verdedigingen uit \cref{ch:g1:healthy-habits} en
        \cref{met:g4:breathing:care}, nog steeds in dienst.
\end{itemize}
Niets nieuws --- en dat is precies het punt. Gezondheid is de oude
regels, volgehouden.
\end{proposition}
\admitted

\begin{example}[Eén dag, van binnenuit gezien]\label{ex:g5:healthy-choices:day}
Een goede dag, gelezen als lichaamsbeheer: het ontbijt vult aan wat de
nacht heeft uitgegeven (\cref{ex:g3:balanced-diet:breakfast}); de
wandeling naar school warmt \omterm{def:g4:heart-and-blood:heart}{hart} en \omterm{def:g4:breathing:organs}{longen} op; het spelen tussen de
middag is training vermomd als plezier; de groente bij het avondeten
vult de groep van de versheid aan; en de vroege nacht geeft de \omterm{def:g5:brain-in-command:system}{hersenen}
hun opruimtijd. Geen enkel onderdeel is heldhaftig; samen zijn ze
gezondheid.
\end{example}

\section{Nieuwe keuzes, nieuwe druk}

\begin{proposition}[Schermen en slaap]\label{prop:g5:healthy-choices:screens}
Schermen --- telefoons, tablets, spelletjes --- zijn geen gif en geen
voedsel; het is \emph{tijd}. De uren die ze opslokken worden ergens
weggehaald, en de gebruikelijke slachtoffers zijn precies de punten van
het contract: buiten bewegen en slapen. Erger nog: felle schermen laat op
de avond vertellen de \omterm{def:g5:brain-in-command:system}{hersenen} ``het is nog dag'' en stellen de \omterm{prop:g1:healthy-habits:needs}{slaap} uit
tot voorbij de uren die je al had opgegeven. De keuze is niet
schermen-of-geen-schermen, maar welke uren ze mogen hebben --- en het
laatste uur van de avond is het slechtste om ze te geven.
\end{proposition}
\admitted

\begin{example}[De elfuurproef]\label{ex:g5:healthy-choices:latenight}
Probeer het als een eerlijke proef
(\cref{met:g4:what-plants-need:fairtest}) op jezelf: een week met de
schermen een uur voor bedtijd uit, tegenover een gewone week. De meeste
mensen vallen sneller in \omterm{prop:g1:healthy-habits:needs}{slaap}, worden gemakkelijker wakker en merken
betere ochtenden op in de eerste week --- de controleweek is het bewijs.
\end{example}

\begin{proposition}[Stoffen die het lichaam misleiden]\label{prop:g5:healthy-choices:substances}
Sommige stoffen beschadigen precies de machinerie die dit boek heeft
rondgeleid:
\begin{itemize}
  \item \emph{tabaksrook} maakt de tere zakjes van de \omterm{def:g4:breathing:organs}{longen} vuil
        (\cref{met:g4:breathing:care}) en belast het \omterm{def:g4:heart-and-blood:heart}{hart}; hij wordt
        verkocht door de eerste sigaretten volwassen te laten lijken ---
        en hij houdt zijn kopers vast met
        \emph{verslaving}\index{verslaving}: het lichaam dat eraan
        gewend is geraakt, protesteert zonder;
  \item \emph{alcohol} bereikt binnen minuten de \omterm{def:g5:brain-in-command:system}{hersenen} --- de
        commandant zelf
        (\cref{ch:g5:brain-in-command}) --- vertraagt de lus en maakt
        beslissingen wazig; jonge \omterm{def:g5:brain-in-command:system}{hersenen}, die hun paden nog aanleggen,
        raken gemakkelijker beschadigd dan volwassen \omterm{def:g5:brain-in-command:system}{hersenen};
  \item allebei waren al aangemerkt als gevaar voor de \omterm{def:g5:human-reproduction:pregnancy}{foetus}
        (\cref{prop:g5:human-reproduction:care}) --- dezelfde stoffen,
        dezelfde reden.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Waarom nooit-beginnen het wint van later-stoppen]\label{rem:g5:healthy-choices:never}
\omterm{prop:g5:healthy-choices:substances}{Verslaving} is een val met een makkelijke ingang en een moeilijke
uitgang: beginnen is één keuze, stoppen is een gevecht dat je jarenlang
herhaalt. Die scheefheid is het koelste feit van dit hoofdstuk, en het
nuttigste: het goedkoopste moment om het gevecht te winnen is voordat het
begint. Wie een eerste sigaret aanbiedt, biedt er niet één aan --- hij
biedt alle volgende aan.
\end{remark}

\section{Zelf beslissen}

\begin{method}[Je eigen contract beheren]\label{met:g5:healthy-choices:run}
Vanaf nu, elk jaar een beetje meer, ben jij de beheerder van het contract
van je lichaam:
\begin{enumerate}
  \item ken de punten van het contract ---
        \cref{prop:g5:healthy-choices:contract} is de lijst;
  \item kijk jezelf af en toe eerlijk na, zoals in
        \cref{met:g3:balanced-diet:check} --- wat ontbreekt, wat zit
        boven zijn deel?
  \item verander één ding tegelijk, en geef het de week van een eerlijke
        proef;
  \item beslis bij stoffen je ``nee'' \emph{voordat} het moment daar is
        (\cref{rem:g5:healthy-choices:never}) --- een beslissing die je
        rustig van tevoren neemt, houd je gemakkelijker vol dan een die
        je onder vriendelijke druk improviseert;
  \item en schakel de volwassenen in: artsen, verpleegkundigen, ouders
        en leraren staan bij dit contract aan jouw kant.
\end{enumerate}
\end{method}

\begin{example}[Antwoorden op zak]\label{ex:g5:healthy-choices:answers}
De beslissing vooraf uit stap 4 is in de praktijk een zin die klaar in je
zak zit: ``Nee, dank je --- niks voor mij.'' Luchtig gezegd, één keer,
maakt hij er meestal een eind aan; en vrienden die de naam waard zijn
accepteren hem de eerste keer. De zin kost niets om voor te bereiden en
is juist op het moment dat je hem nodig hebt het moeilijkst te bedenken
--- dus bedenk je hem nu.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:1}
Noem de vier punten van het dagelijkse contract van het lichaam, elk met
het orgaan of de dienst die het vooral bedient.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:2}
Ongeveer hoeveel \omterm{prop:g1:healthy-habits:needs}{slaap} heeft een kind van jouw leeftijd nodig, en welke
twee taken doet het lichaam daarmee?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:3}
In welke zin worden schermuren ``ergens weggehaald''? Noem de twee
gebruikelijke slachtoffers.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:4}
Waarom is het laatste uur van de avond het slechtste om aan een fel
scherm te geven?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:5}
Welke organen beschadigt tabaksrook het eerst, volgens
\cref{prop:g5:healthy-choices:substances}? En alcohol?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:6}
Wat is \omterm{prop:g5:healthy-choices:substances}{verslaving}, en waarom maakt ze ``nooit beginnen'' zoveel
goedkoper dan ``later stoppen''?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:7}
Wat is het nut van een ``nee'' vooraf beslissen, volgens stap 4 van
\cref{met:g5:healthy-choices:run}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:8}
Probeer het: doe de elfuurproef uit
\cref{ex:g5:healthy-choices:latenight} een week lang en vertel ---

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:healthy-choices:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:healthy-choices:10}

\cref{rem:g5:healthy-choices:never}: wat wordt er met de eerste

\end{exercise}
