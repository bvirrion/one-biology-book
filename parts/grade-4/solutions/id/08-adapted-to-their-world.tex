\section*{Bab \ref{ch:g4:adapted-to-their-world} --- Menyesuaikan Diri dengan Dunianya}

\begin{solution}{exo:g4:adapted-to-their-world:1}
Ciri tubuh atau perilaku yang mencocokkan sebuah \omterm{def:g1:living-or-not:living}{makhluk hidup} dengan
keadaan \omterm{def:g2:caring-for-nature:environment}{lingkungannya}. \omterm{def:g1:animals-around-us:animal}{Hewan}: penyekat rangkap berupa \omterm{def:g1:animals-around-us:covering}{bulu} dan lemak
milik beruang kutub. \omterm{def:g1:plants-around-us:plant}{Tumbuhan}: \omterm{def:g1:plants-around-us:parts}{batang} penyimpan air milik kaktus.
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:2}
Tanpa air: meminum seratus liter sekaligus lalu bertahan berhari-hari
tanpa minum, ditambah cadangan lemak di punuknya. Pasir yang lembut:
kaki lebar berjari dua. Debu yang beterbangan: lubang hidung yang
dapat menutup dan \omterm{def:g1:animals-around-us:covering}{bulu} \omterm{def:g1:the-five-senses:sense}{mata} rangkap yang panjang. (Panas dan lapar:
punuknya lagi --- lemak bagi lintasan yang kurus.)
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:3}
Dingin: \omterm{def:g1:animals-around-us:covering}{bulu} tebal di atas lapisan lemak yang dalam. Air yang
membekukan dan salju: telapak kaki yang lebar dan sedikit berselaput
--- sepatu salju dan dayung. Mangsa yang waspada di atas es putih:
penyamaran berupa mantel putih.
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:4}
Durinya adalah \omterm{def:g1:plants-around-us:parts}{daun} yang menyusut: ia nyaris tidak kehilangan air dan
membela \omterm{def:g1:plants-around-us:plant}{tumbuhannya} terhadap \omterm{def:g1:animals-around-us:animal}{hewan} yang kehausan. \omterm{def:g1:plants-around-us:parts}{Batang} tongnya
menyimpan air, dan sudah mengambil alih pekerjaan penangkapan cahaya
milik \omterm{def:g1:plants-around-us:parts}{daun}.
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:5}
Bentuk bantalannya mematahkan angin yang membekukan: di dalamnya
udaranya tetap lebih lunak. Bentuknya sendirilah tempat berlindungnya.
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:6}
Yang diburu: warna penyamaran, baju pelindung atau duri, kecepatan,
\omterm{def:g1:the-five-senses:sense}{mata} berjauhan yang mengawasi ke segala arah (dua saja cukup). Pemburu:
\omterm{def:g1:the-five-senses:sense}{mata} yang menghadap ke depan, indra yang tajam, cakar, kecepatan (dua
saja cukup).
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:7}
\omterm{def:g1:animals-around-us:covering}{Bulu} lembut yang lebat --- seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia}; cakar depan raksasa
berbentuk sekop --- seorang penggali; \omterm{def:g1:the-five-senses:sense}{mata} mungil yang lemah --- hidup
dalam gelap. Bersama-sama: seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia} penggali bawah tanah, si
tikus tanah.
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:8}
Jawaban terbuka. Contoh untuk bebek: kaki berselaput (mengayuh), \omterm{def:g1:animals-around-us:covering}{bulu}
yang menolak air (tetap kering dan hangat ketika mengapung), paruh
pipih penyaring (mencari makan dari air).
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:9}
\omterm{def:g1:the-five-senses:sense}{Telinga} adalah radiator: \omterm{def:g1:the-five-senses:sense}{telinga} yang besar dan tipis melepaskan
kehangatan tubuh. Di gurun, melepaskan kehangatan adalah masalah yang
harus dipecahkan --- \omterm{def:g1:the-five-senses:sense}{telinga} yang sangat besar mendinginkan rubahnya.
Di atas es, menahan kehangatan adalah segalanya --- \omterm{def:g1:the-five-senses:sense}{telinga} yang
mungil paling sedikit kehilangannya. Satu organ, dua iklim, dua ukuran
yang berlawanan.
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:10}
Bentuk yang ramping, kaki berselaput dan kemampuan menahan napas
berkata: seorang perenang dan penyelam; \omterm{def:g1:animals-around-us:covering}{bulu} lebat yang menolak air
berkata: seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia} yang hidup separuhnya di air dingin; kumisnya
berkata: meraba jalannya di air yang gelap atau keruh. Seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia}
penghuni air pembangun bendungan di sungai dan danau --- berang-berang.
\end{solution}

\begin{solution}{exo:g4:adapted-to-their-world:11}
Airnya yang menetapkannya. Paus dan \omterm{prop:g3:sorting-living-things:groups}{ikan} trout tidak mewarisi apa pun
yang berbentuk perenang dari keluarga bersama --- yang satu \omterm{prop:g3:sorting-living-things:groups}{mamalia},
yang lain \omterm{prop:g3:sorting-living-things:groups}{ikan} --- namun masalah yang sama, yaitu bergerak cepat
menembus air, mendukung jawaban ramping yang sama pada keduanya.
\omterm{def:g4:adapted-to-their-world:adaptation}{Adaptasi} menjawab masalah \omterm{def:g2:caring-for-nature:environment}{lingkungannya}; masalah yang serupa dapat
menarik bentuk yang serupa dari keluarga yang sangat berbeda.
\end{solution}
