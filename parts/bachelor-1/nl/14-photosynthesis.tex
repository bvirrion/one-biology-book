\chapter{Fotosynthese en autotrofie}\label{ch:b1:photosynthesis}

Elke gram koolstof in het konijn, de eik, de lezer en deze bladzijde was ooit
koolstofdioxide in de lucht, en is door een \omterm{def:b1:flowering-plant-organization:organs}{blad} --- of door een alg of een
cyanobacterie --- met de energie van zonlicht tot suiker vastgelegd. De reactie,
zes $\mathrm{CO_2}$ en zes water tot één glucose en zes zuurstof, gaat
\qty{2870}{kJ/mol} bergop; een \omterm{def:b1:flowering-plant-organization:organs}{blad} laat haar lopen door water met licht te
splitsen, de elektronen op NADPH en de energie op \omterm{def:b1:water-small-molecules:atp}{ATP} op te slaan, en beide uit te
geven in een cyclus die suiker bouwt. Dit hoofdstuk beschrijft de \omterm{def:b1:photosynthesis:chloroplast}{chloroplast}, de
kleurstoffen en de \omterm{prop:b1:photosynthesis:lightreactions}{lichtreacties} die \omterm{def:b1:water-small-molecules:atp}{ATP} en NADPH maken, de \omterm{def:b1:photosynthesis:fixation}{calvincyclus} die
koolstof vastlegt, het lek dat fotorespiratie heet en de twee ontwerpen die het
dichten, en wat \omterm{def:b1:photosynthesis:autotrophy}{autotrofie} in het algemeen is.

\section{De chloroplast en haar kleurstoffen}

\begin{definition}[Chloroplast]\label{def:b1:photosynthesis:chloroplast}
De \emph{chloroplast}\index{chloroplast} is een lensvormig organel van
\qtyrange{3}{10}{\micro m} lang, begrensd door twee envelopmembranen; haar
binnenste, het \emph{stroma}\index{stroma}, bevat de \omterm{def:b1:enzymes:enzyme}{enzymen} van de
koolstofvastlegging, zetmeelkorrels, ribosomen en een ringvormig \omterm{def:b1:nucleic-acids:chain}{DNA}; binnen het
stroma vormt een derde membraanstelsel, de \emph{thylakoïden}\index{thylakoïde},
afgeplatte zakjes die tot \emph{grana} zijn gestapeld en door lamellen zijn
verbonden, en die één aaneengesloten \emph{thylakoïdelumen} omsluiten. De
\omterm{prop:b1:photosynthesis:lightreactions}{lichtreacties} vinden plaats in de thylakoïdemembraan, de koolstofvastlegging in het
stroma. Een bladcel bevat \qtyrange{20}{100}{} \omterm{def:b1:eukaryotic-cell:plastid}{chloroplasten}; een vierkante
millimeter \omterm{def:b1:flowering-plant-organization:organs}{blad} een half miljoen.
\end{definition}

\begin{definition}[Fotosynthetische kleurstoffen]\label{def:b1:photosynthesis:pigments}
\emph{Chlorofyl}\index{chlorofyl} $a$ en $b$ zijn porfyrineringen rond een
magnesiumatoom met een lange hydrofobe staart die hen in de thylakoïdemembraan
verankert; zij nemen blauw (\qtyrange{430}{450}{nm}) en rood
(\qtyrange{640}{680}{nm}) licht op en weerkaatsen groen.
\emph{Carotenoïden}\index{carotenoïde} (\omterm{def:b1:lipids:steroids}{isoprenoïden}, \cref{ch:b1:lipids}) nemen
blauwgroen licht op en geven de energie aan chlorofyl door, en beschermen het door
overtollige aanslag te doven. Enkele honderden kleurstofmoleculen, op \omterm{def:b1:proteins:peptide}{eiwitten}
gehouden als een \emph{antenne} (lichtoogstcomplex), trechteren de energie van elk
opgenomen foton naar één bijzonder paar chlorofyl-$a$-moleculen, het
\emph{reactiecentrum}, waar zij in chemie wordt omgezet.
\end{definition}

\begin{proposition}[Het werkingsspectrum volgt de kleurstoffen]\label{prop:b1:photosynthesis:engelmann}
De fotosynthese wordt aangedreven door het licht dat de kleurstoffen opnemen: zij
is het snelst in rood en blauw licht en het traagst in groen.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
Engelmann (1882) legde een draad alg dwars over een spectrum dat hij onder zijn
microscoop projecteerde en voegde bacteriën toe die naar zuurstof zoeken: zij
verzamelden zich waar het rode en het blauwe licht viel, niet in het groen (bekend
uit het deel over de bovenbouw). Met een zuurstofelektrode gemeten geeft de
snelheid bij elke golflengte het werkingsspectrum hierboven, dat de opname door de
chlorofyllen volgt; het overschot in het blauwgroen is de bijdrage van de
\omterm{def:b1:photosynthesis:pigments}{carotenoïden}, wat aantoont dat zij hun energie doorgeven.
\end{proof}

\section{De lichtreacties}

\begin{proposition}[Wat de lichtreacties doen]\label{prop:b1:photosynthesis:lightreactions}
\index{lichtreacties}
In de thylakoïdemembraan wordt lichtenergie gebruikt om elektronen van water naar
$\mathrm{NADP^+}$ te brengen --- bergop, tegen een verschil in redoxpotentiaal van
\qty{1.1}{V} in --- en om protonen het lumen in te pompen:
\[
2\,\mathrm{H_2O} + 2\,\mathrm{NADP^+} + 3\,\mathrm{ADP} + 3\,\mathrm{P_i}
\xrightarrow{\ 8\ \text{fotonen}\ }
\mathrm{O_2} + 2\,\mathrm{NADPH} + 2\,\mathrm{H^+} + 3\,\mathrm{ATP} .
\]
De vrijgekomen zuurstof is de zuurstof van het water, niet die van de
koolstofdioxide; het NADPH draagt de reducerende kracht en het \omterm{def:b1:water-small-molecules:atp}{ATP} de energie die
de \omterm{def:b1:photosynthesis:fixation}{calvincyclus} zal uitgeven.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
Hill (1937) toonde aan dat losgemaakte \omterm{def:b1:eukaryotic-cell:plastid}{chloroplasten} in het licht zuurstof
afgeven zonder $\mathrm{CO_2}$ als er een kunstmatige elektronenontvanger aanwezig
is: het splitsen van water is los te maken van de koolstofvastlegging. Ruben en
Kamen (1941) gaven algen water dat met zware zuurstof was gemerkt en vonden het
merk terug in de vrijgekomen $\mathrm{O_2}$, en niet wanneer het merk in het
$\mathrm{CO_2}$ zat. Emerson (1957) vond dat rood licht van \qty{700}{nm}, alleen
vrijwel nutteloos, samen met licht van \qty{680}{nm} meer opleverde dan de som van
hun afzonderlijke snelheden: er moeten twee \omterm{def:b1:photosynthesis:zscheme}{fotosystemen} met verschillende
kleurstoffen in serie werken.
\end{proof}

\begin{definition}[Fotosystemen en het Z-schema]\label{def:b1:photosynthesis:zscheme}
\index{fotosysteem}\index{Z-schema}
Twee \emph{fotosystemen} liggen in de thylakoïdemembraan. In \emph{fotosysteem II}
(PSII, reactiecentrum P680) tilt een opgenomen foton een elektron van het
bijzondere paar naar een hoge energie; het elektron vertrekt, en het geoxideerde
P680, het sterkste biologische oxidatiemiddel, neemt een elektron van water: een
mangaancluster splitst twee watermoleculen in $\mathrm{O_2}$, vier protonen (die in
het lumen vrijkomen) en vier elektronen, telkens één foton tegelijk. Het elektron
daalt af langs een \emph{elektronentransportketen} --- plastochinon, het
\emph{cytochroom-$b_6f$-complex}, plastocyanine --- en de energie die het verliest
pompt protonen het lumen in. In \emph{fotosysteem I} (PSI, P700) tilt een tweede
foton het opnieuw op, tot een potentiaal die laag genoeg is om ferredoxine en
vervolgens $\mathrm{NADP^+}$ tot NADPH te reduceren. Uitgezet op een schaal van
redoxpotentiaal is de weg een Z: twee keer omhoog door licht, twee keer omlaag door
chemie.
\end{definition}

\begin{theorem}[Chemiosmotische ATP-synthese]\label{thm:b1:photosynthesis:chemiosmosis}
\index{chemiosmose}\index{ATP-synthase}
De protonen die bij het splitsen van water vrijkomen en die het
cytochroom-$b_6f$-complex pompt, hopen zich op in het thylakoïdelumen, dat in het
licht tot pH 5 daalt terwijl het \omterm{def:b1:photosynthesis:chloroplast}{stroma} tot pH 8 stijgt: een verschil van drie
eenheden, een protonmotorische kracht van ongeveer \qty{0.18}{V}, gelijk aan
\qty{17}{kJ} per mol protonen. Het \emph{\omterm{thm:b1:photosynthesis:chemiosmosis}{ATP-synthase}}, een draaimotor die de
thylakoïdemembraan doorkruist, laat de protonen naar het \omterm{def:b1:photosynthesis:chloroplast}{stroma} terugstromen en
gebruikt de energie om ADP te fosforyleren: ongeveer \qty{4}{H^+} per \omterm{def:b1:water-small-molecules:atp}{ATP}.
Elektronentransport en ATP-synthese zijn alleen via deze gradiënt gekoppeld (de
chemiosmotische theorie van Mitchell, 1961).
\end{theorem}
\begin{proof}[Bewijsmateriaal]
Jagendorf (1966) liet \omterm{def:b1:photosynthesis:chloroplast}{thylakoïden} in het donker weken in een \omterm{def:b1:water-small-molecules:ph}{zuur} bad (pH 4) tot
hun lumen \omterm{def:b1:water-small-molecules:ph}{zuur} was, en bracht ze toen plotseling over naar pH 8 met ADP en fosfaat:
er werd in het donker \omterm{def:b1:water-small-molecules:atp}{ATP} gemaakt, uit de gradiënt alleen. Ontkoppelaars ---
vetoplosbare zwakke zuren die protonen door membranen dragen --- heffen de
ATP-synthese op terwijl het elektronentransport en de zuurstofafgifte doorgaan en
zelfs versnellen. Losgemaakt \omterm{thm:b1:photosynthesis:chemiosmosis}{ATP-synthase}, in kunstmatige blaasjes teruggebracht,
maakt \omterm{def:b1:water-small-molecules:atp}{ATP} wanneer er een pH-gradiënt wordt opgelegd, en hydrolyseert \omterm{def:b1:water-small-molecules:atp}{ATP} om
protonen te pompen wanneer dat niet gebeurt. De draaiing van het \omterm{def:b1:enzymes:enzyme}{enzym} is sindsdien
rechtstreeks waargenomen, één molecuul tegelijk, onder de microscoop.
\end{proof}

\begin{example}[Cyclische en niet-cyclische stroom]\label{ex:b1:photosynthesis:cyclic}
De lineaire weg van water naar NADPH levert per $\mathrm{O_2}$ twee NADPH en
ongeveer drie \omterm{def:b1:water-small-molecules:atp}{ATP}; de \omterm{def:b1:photosynthesis:fixation}{calvincyclus} heeft precies een verhouding van drie \omterm{def:b1:water-small-molecules:atp}{ATP} op
twee NADPH nodig, en het overige werk van de \omterm{def:b1:photosynthesis:chloroplast}{chloroplast} vraagt meer \omterm{def:b1:water-small-molecules:atp}{ATP}. Wanneer
er \omterm{def:b1:water-small-molecules:atp}{ATP} tekortschiet, keren elektronen van ferredoxine terug naar plastochinon in
plaats van naar $\mathrm{NADP^+}$ te gaan (\emph{cyclische fotofosforylering}):
PSI alleen, dat protonen pompt en \omterm{def:b1:water-small-molecules:atp}{ATP} maakt zonder zuurstof of NADPH. De twee
werkwijzen worden op de vraag afgestemd.
\end{example}

\section{De calvincyclus}

\begin{definition}[Koolstofvastlegging]\label{def:b1:photosynthesis:fixation}
\index{calvincyclus}\index{rubisco}
De \emph{calvincyclus} in het \omterm{def:b1:photosynthesis:chloroplast}{stroma} legt $\mathrm{CO_2}$ in drie fasen tot suiker
vast. \emph{Vastlegging}: \emph{rubisco} (ribulosebisfosfaat
carboxylase/oxygenase) voegt $\mathrm{CO_2}$ toe aan ribulose-1,5-bisfosfaat (RuBP,
vijf koolstoffen), en het product met zes koolstoffen splitst meteen in twee
moleculen 3-fosfoglyceraat (3-PGA, drie koolstoffen). \emph{Reductie}: elk 3-PGA
wordt door \omterm{def:b1:water-small-molecules:atp}{ATP} gefosforyleerd en door NADPH gereduceerd tot
glyceraldehyde-3-fosfaat (G3P), de eerste suiker. \emph{Regeneratie}: vijf van elke
zes G3P worden, ten koste van \omterm{def:b1:water-small-molecules:atp}{ATP}, weer tot drie RuBP herschikt, en één G3P
verlaat de cyclus als het netto product. De stoichiometrie voor één G3P (drie
koolstoffen) is
\[
3\,\mathrm{CO_2} + 9\,\mathrm{ATP} + 6\,\mathrm{NADPH} + 6\,\mathrm{H^+}
\to \text{G3P} + 9\,\mathrm{ADP} + 8\,\mathrm{P_i} + 6\,\mathrm{NADP^+} ;
\]
twee G3P maken één glucose: \qty{18}{} \omterm{def:b1:water-small-molecules:atp}{ATP} en \qty{12}{NADPH} per glucose.
\end{definition}

\begin{proposition}[Het bewijsmateriaal van Calvin]\label{prop:b1:photosynthesis:calvin}
Het eerste stabiele product van de $\mathrm{CO_2}$-vastlegging is
3-fosfoglyceraat, en de ontvanger is ribulosebisfosfaat.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
Calvin en Benson (1948--1954) belichtten algen in een platte kolf (de
``lolly''), spoten $\mathrm{^{14}CO_2}$ in, en doodden na enkele seconden monsters
in kokende alcohol; tweedimensionale papierchromatografie en autoradiografie
toonden waar het merk zat. Na vijf seconden zat vrijwel alles in 3-PGA; bij langere
blootstelling verspreidde het zich over de suikerfosfaten en vervolgens naar
sucrose en \omterm{def:b1:carbohydrates:polysaccharide}{zetmeel}. Werd het $\mathrm{CO_2}$ plotseling weggenomen, dan hoopte RuBP
zich op en daalde 3-PGA; werd het licht uitgezet, dan hoopte 3-PGA zich op en
daalde RuBP --- dus is RuBP de ontvanger die door de vastlegging wordt verbruikt,
en 3-PGA het product dat door het \omterm{def:b1:water-small-molecules:atp}{ATP} en het NADPH van het licht wordt verbruikt.
\end{proof}

\begin{method}[Een fotosynthesebalans opstellen]\label{met:b1:photosynthesis:budget}
\begin{enumerate}
  \item Tel koolstoffen: elke vastgelegde $\mathrm{CO_2}$ levert één
        koolstof product; één glucose vraagt zes rondes van \omterm{def:b1:photosynthesis:fixation}{rubisco}.
  \item Tel dragers: elke ronde kost 3 \omterm{def:b1:water-small-molecules:atp}{ATP} en 2 NADPH (9 en 6 per G3P);
        één glucose kost 18 \omterm{def:b1:water-small-molecules:atp}{ATP} en 12 NADPH.
  \item Tel fotonen: de lineaire stroom geeft per $\mathrm{O_2}$ 2 NADPH en
        ongeveer 3 \omterm{def:b1:water-small-molecules:atp}{ATP}, voor 8 fotonen (4 per \omterm{def:b1:photosynthesis:zscheme}{fotosysteem}); 12 NADPH
        vragen 6 $\mathrm{O_2}$ en 48 fotonen; het extra \omterm{def:b1:water-small-molecules:atp}{ATP} komt uit de
        cyclische stroom (nog enkele fotonen). Ongeveer 8 fotonen per
        $\mathrm{CO_2}$: de kwantumbehoefte.
  \item Tel energie: een mol fotonen van \qty{680}{nm} is \qty{176}{kJ};
        48 fotonen zijn \qty{8450}{kJ} voor \qty{2870}{kJ} opgeslagen in
        glucose: hoogstens \qty{34}{\%}, nog vóór enig verlies door
        weerkaatsing, ademhaling en het deel van het spectrum dat niet
        wordt opgenomen.
\end{enumerate}
\end{method}

\section{Fotorespiratie, C4 en CAM}

\begin{proposition}[De zwakke plek van rubisco]\label{prop:b1:photosynthesis:photorespiration}
\index{fotorespiratie}
\omterm{def:b1:photosynthesis:fixation}{Rubisco} neemt ook $\mathrm{O_2}$ aan in plaats van $\mathrm{CO_2}$: de oxygenatie
van RuBP levert één 3-PGA en één fosfoglycolaat met twee koolstoffen op, dat de
\omterm{def:b1:cell-unit-of-life:cell}{cel} langs een kostbare route over drie \omterm{def:b1:cell-unit-of-life:prokeuk}{organellen} moet zien te redden, waarbij
$\mathrm{CO_2}$ vrijkomt en \omterm{def:b1:water-small-molecules:atp}{ATP} wordt verbruikt --- \emph{fotorespiratie}. Het
\omterm{def:b1:enzymes:enzyme}{enzym} onderscheidt slecht (bij gelijke concentraties ongeveer \qty{80}{} tegen 1
ten gunste van $\mathrm{CO_2}$), en in lucht is $\mathrm{O_2}$ vijfhonderd keer zo
overvloedig als $\mathrm{CO_2}$; bij \qty{25}{\celsius} gaat een kwart van de
vastleggingen verloren, meer bij warm droog weer wanneer de huidmondjes sluiten en
het $\mathrm{CO_2}$ in het \omterm{def:b1:flowering-plant-organization:organs}{blad} daalt. \omterm{def:b1:photosynthesis:fixation}{Rubisco}, ontstaan toen de lucht geen
zuurstof bevatte, is bovendien traag (drie omzettingen per seconde) en wordt door
overvloed gecompenseerd: het is het meest voorkomende \omterm{def:b1:proteins:peptide}{eiwit} op aarde, de helft van
het oplosbare \omterm{def:b1:proteins:peptide}{eiwit} van een \omterm{def:b1:flowering-plant-organization:organs}{blad}.
\end{proposition}

\begin{definition}[C4- en CAM-planten]\label{def:b1:photosynthesis:c4cam}
\index{C4-plant}\index{CAM-plant}
\emph{C4-planten} (mais, suikerriet, sorghum, veel tropische grassen) dichten het
lek door $\mathrm{CO_2}$ te concentreren: in de mesofylcellen legt een \omterm{def:b1:enzymes:enzyme}{enzym} zonder
affiniteit voor $\mathrm{O_2}$ (PEP-carboxylase) bicarbonaat vast in een \omterm{def:b1:water-small-molecules:ph}{zuur} met
vier koolstoffen, dat naar de \emph{bundelscheedecellen} rond de nerven wordt
gebracht en daar wordt gedecarboxyleerd, wat het $\mathrm{CO_2}$ rond \omterm{def:b1:photosynthesis:fixation}{rubisco}
tienvoudig verhoogt en de oxygenatie onderdrukt --- ten koste van twee extra \omterm{def:b1:water-small-molecules:atp}{ATP}
per $\mathrm{CO_2}$. De twee celtypen vormen de \emph{kranz}-anatomie (``krans'').
\emph{CAM-planten} (cactussen, ananas, agaven) scheiden diezelfde twee stappen in
de tijd in plaats van in de ruimte: zij openen hun huidmondjes 's nachts, slaan
$\mathrm{CO_2}$ als appelzuur op in de \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole}, en geven het overdag achter
gesloten huidmondjes aan \omterm{def:b1:photosynthesis:fixation}{rubisco} af, waarbij zij per vastgelegde koolstof een
tiende verliezen van het water dat een C3-plant verliest.
\end{definition}

\begin{omfigure}
\includegraphics[width=0.6\linewidth]{images/book3/ai/b3-14-maize-kranz.jpg}

{\small Kranz-anatomie in een maisblad: elke nerf is omkranst door grote
bundelscheedecellen die volgepakt zijn met \omterm{def:b1:eukaryotic-cell:plastid}{chloroplasten}, en die zelf door
mesofylcellen worden omringd. $\mathrm{CO_2}$ wordt in de buitenste ring
opgevangen en geconcentreerd afgeleverd bij \omterm{def:b1:photosynthesis:fixation}{rubisco} in de binnenste.}
\end{omfigure}

\begin{example}[Welke plant waar]\label{ex:b1:photosynthesis:where}
Bij \qty{20}{\celsius} in vochtige lucht leggen een C3-tarweblad en een
C4-maisblad koolstof met vergelijkbare snelheid vast, en betaalt de mais per
koolstof twee \omterm{def:b1:water-small-molecules:atp}{ATP} meer. Bij \qty{35}{\celsius} in droge lucht verliest de tarwe een
derde van haar vastlegging aan fotorespiratie terwijl de mais niets verliest:
C4-grassen overheersen in warm open land, C3-planten in het koele en beschaduwde.
Een cactus legt naar beide maatstaven traag vast maar overleeft waar de andere van
dorst zouden sterven.
\end{example}

\section{Autotrofie}

\begin{definition}[Autotrofie]\label{def:b1:photosynthesis:autotrophy}
Een \omterm{def:b1:organism-environment:organism}{organisme} is \emph{autotroof}\index{autotrofie} wanneer het zijn organische
stof uit $\mathrm{CO_2}$ opbouwt (\cref{ch:b1:organism-environment}). Het heeft een
energiebron en een elektronenbron nodig. \emph{\omterm{def:b1:organism-environment:trophy}{Fotoautotrofen}} halen hun energie
uit licht: planten, algen en cyanobacteriën halen hun elektronen uit water en geven
zuurstof af (\emph{oxygeen}); purper- en groenbacteriën halen ze uit
waterstofsulfide of organische zuren en geven er geen af. \emph{\omterm{def:b1:organism-environment:trophy}{Chemoautotrofen}}
halen zowel energie als elektronen uit de oxidatie van anorganische verbindingen
--- ammoniak, nitriet, sulfide, waterstof, tweewaardig ijzer --- geheel zonder
licht: de nitrificerende bacteriën van de bodem, en de bacteriën die de
ecosystemen van diepzeebronnen voeden. Alle gebruiken de \omterm{def:b1:photosynthesis:fixation}{calvincyclus}, of een
verwante cyclus, om de koolstof vast te leggen; alle reduceren haar met NADPH en
betalen met \omterm{def:b1:water-small-molecules:atp}{ATP}; zij verschillen alleen in waar het \omterm{def:b1:water-small-molecules:atp}{ATP} en de elektronen vandaan
komen.
\end{definition}

\begin{example}[De balans van de aarde]\label{ex:b1:photosynthesis:global}
De fotosynthese legt per jaar ongeveer \qty{120}{Gt} koolstof vast op het land en
\qty{50}{Gt} in de oceanen, waarvan de helft door eencellige algen en
cyanobacteriën die voor het oog onzichtbaar zijn; zij ververst de koolstofdioxide
van de atmosfeer elke zeven jaar en haar zuurstof elke tweeduizend jaar. De
zuurstof kwam op dezelfde manier tot stand: twee miljard jaar lang gaven
cyanobacteriën haar af aan een atmosfeer die er geen had, en elk aeroob \omterm{def:b1:organism-environment:organism}{organisme},
van het \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrion} af, is gebouwd op het lek van één enzymstelsel dat water
splitst.
\end{example}

\section{Vraagstuk: de prijs van een molecuul glucose}

\begin{problem}[{Weekendvraagstuk --- fotonen geteld, elektronen en protonen door
de thylakoïde gevolgd, ATP en NADPH becijferd en uitgegeven, de dagopbrengst van
een blad gewogen, eindigend op het energierendement van de fotosynthese}]
\label{pb:b1:photosynthesis:1}
Constanten: $h = \qty{6.63e-34}{J.s}$, $c = \qty{3.00e8}{m/s}$,
$N_A = \num{6.02e23}$, $F = \qty{96500}{C/mol}$, $R = \qty{8.314}{J.mol^{-1}.K^{-1}}$,
$T = \qty{298}{K}$. De \omterm{prop:b1:water-small-molecules:gibbs}{vrije energie} van de verbranding van glucose is
\qty{2870}{kJ/mol}; ATP-synthese in de \omterm{def:b1:photosynthesis:chloroplast}{chloroplast} kost \qty{50}{kJ/mol}; NADPH
bevat \qty{220}{kJ/mol} aan reducerende kracht.

\medskip
\noindent\textbf{Deel I --- Fotonen en elektronen.}
\begin{enumerate}
  \item Bereken de energie van één foton van \qty{680}{nm} en van een mol
        ervan.
  \item Bereken de energie van een mol fotonen van \qty{700}{nm} en van
        \qty{450}{nm}. Waarom levert blauw licht per foton niet meer
        fotosynthese op dan rood?
  \item Hoeveel fotonen van \qty{680}{nm} dragen één joule?
  \item Hoeveel \omterm{prop:b1:water-small-molecules:gibbs}{vrije energie} per mol elektronen vraagt het verplaatsen van
        één elektron van water ($E'_0 = \qty{+0.82}{V}$) naar
        $\mathrm{NADP^+}$ ($E'_0 = \qty{-0.32}{V}$) ($\Delta G = -nF\Delta E$)?
  \item Er worden per elektron twee fotonen gebruikt (één in elk
        \omterm{def:b1:photosynthesis:zscheme}{fotosysteem}). Bereken de energie die per mol elektronen wordt
        geleverd en het deel daarvan dat in de redoxverandering wordt
        opgeslagen.
  \item Hoeveel elektronen, en dus hoeveel fotonen, zijn er nodig om één
        $\mathrm{O_2}$ en twee NADPH te maken?
  \item Hoeveel fotonen per glucose, als er 12 NADPH nodig zijn?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Protonen en \omterm{def:b1:water-small-molecules:atp}{ATP}.}
Voor elke vrijgekomen $\mathrm{O_2}$ komen er vier protonen in het lumen vrij door
het splitsen van water en worden er acht door het cytochroom-$b_6f$-complex
gepompt; het \omterm{thm:b1:photosynthesis:chemiosmosis}{ATP-synthase} gebruikt \qty{4}{H^+} per \omterm{def:b1:water-small-molecules:atp}{ATP}. Het lumen zit op pH 5 en
het \omterm{def:b1:photosynthesis:chloroplast}{stroma} op pH 8; de \omterm{def:b1:membranes-transport:potential}{membraanpotentiaal} over de \omterm{def:b1:photosynthesis:chloroplast}{thylakoïde} is verwaarloosbaar.
\begin{enumerate}[resume]
  \item Bereken de protonmotorische kracht uit het pH-verschil, in volt
        ($\Delta\mu = 2.303\,RT\,\Delta\mathrm{pH}/F$), en de \omterm{prop:b1:water-small-molecules:gibbs}{vrije energie}
        per mol protonen.
  \item Bereken de protonen die per $\mathrm{O_2}$ in het lumen worden
        afgeleverd en het \omterm{def:b1:water-small-molecules:atp}{ATP} dat zij kunnen maken.
  \item Bereken het \omterm{def:b1:water-small-molecules:atp}{ATP} dat per glucose door de lineaire stroom wordt
        gemaakt (6 $\mathrm{O_2}$), en vergelijk dat met de 18 die de
        \omterm{def:b1:photosynthesis:fixation}{calvincyclus} nodig heeft. Hoe wordt het tekort aangevuld?
  \item Bereken de \omterm{prop:b1:water-small-molecules:gibbs}{vrije energie} die de protonen per \omterm{def:b1:water-small-molecules:atp}{ATP} leveren
        (\qty{4}{H^+}) en het rendement van het synthase.
  \item Het lumen van een \omterm{def:b1:photosynthesis:chloroplast}{chloroplast} heeft een volume van ongeveer
        \qty{1}{\micro m^3}. Hoeveel vrije protonen bevat het bij pH 5?
        Vergelijk dat met de ruwweg $10^5$ protonen die per seconde door de
        synthasen gaan, en concludeer wat het lumen buffert.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Suiker.}
\begin{enumerate}[resume]
  \item Schrijf de energie die per glucose wordt opgeslagen als 18 \omterm{def:b1:water-small-molecules:atp}{ATP}
        $\times$ \qty{50}{kJ} plus 12 NADPH $\times$ \qty{220}{kJ}, en
        vergelijk die met de \qty{2870}{kJ} van glucose. Waar gaat het
        verschil naartoe?
  \item Bereken de energie van de 48 fotonen uit vraag 6 en het rendement
        van de fotosynthese van foton tot glucose.
  \item Slechts \qty{45}{\%} van het zonlicht zit in de golflengten die de
        kleurstoffen opnemen, en daarvan wordt een tiende weerkaatst of
        doorgelaten. Bereken het rendement van invallend zonlicht tot
        glucose, vóór de ademhaling.
  \item Een \omterm{def:b1:flowering-plant-organization:organs}{blad} van \qty{50}{cm^2} ontvangt \qty{10}{h} lang
        \qty{400}{W/m^2} zonlicht. Bereken de ontvangen energie en, bij het
        rendement uit vraag 14, de gemaakte glucose in gram
        (\qty{180}{g/mol}).
  \item Het \omterm{def:b1:flowering-plant-organization:organs}{blad} verademt \qty{40}{\%} van die glucose om in leven te
        blijven. Wat is zijn netto winst, in gram glucose en in gram
        koolstof?
  \item Een gewas haalt over een seizoen \qty{1}{\%} totaalrendement. Noem
        drie verliezen, bovenop de reeds getelde, die het cijfer van het
        \omterm{def:b1:flowering-plant-organization:organs}{blad} van dat van het gewas scheiden.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het lek.}
\omterm{def:b1:photosynthesis:fixation}{Rubisco} voert in lucht bij \qty{25}{\celsius} één oxygenatie per drie
carboxyleringen uit; elke oxygenatie kost het equivalent van een halve vastgelegde
$\mathrm{CO_2}$ en 3,5 \omterm{def:b1:water-small-molecules:atp}{ATP}.
\begin{enumerate}[resume]
  \item Bereken per 100 carboxyleringen de oxygenaties, de verloren
        koolstof en de netto gewonnen koolstof.
  \item Bereken het \omterm{def:b1:water-small-molecules:atp}{ATP} dat per 100 carboxyleringen aan de redding wordt
        besteed en het totale \omterm{def:b1:water-small-molecules:atp}{ATP} per netto gewonnen koolstof (\omterm{def:b1:photosynthesis:fixation}{calvincyclus}
        3 \omterm{def:b1:water-small-molecules:atp}{ATP} per carboxylering plus de redding).
  \item Een \omterm{def:b1:photosynthesis:c4cam}{C4-plant} kent geen oxygenatie maar geeft per afgeleverde
        $\mathrm{CO_2}$ 2 extra \omterm{def:b1:water-small-molecules:atp}{ATP} uit. Bereken haar \omterm{def:b1:water-small-molecules:atp}{ATP} per netto
        koolstof en vergelijk.
  \item Bij \qty{35}{\celsius} stijgt de oxygenatieverhouding naar één op
        twee. Bereken de C3-cijfers opnieuw en zeg welk ontwerp wint.
  \item Leg uit waarom het verdrievoudigen van het $\mathrm{CO_2}$ in een
        kas de opbrengst van C3-gewassen (tomaat, tarwe) verhoogt maar die
        van C4-gewassen (mais) nauwelijks.
  \item Een C3-blad verliest per vastgelegde $\mathrm{CO_2}$ ongeveer 250
        watermoleculen, een \omterm{def:b1:photosynthesis:c4cam}{CAM-plant} ongeveer 25. Bereken hoeveel water elk
        verliest om \qty{1}{g} koolstof vast te leggen.
  \item Geef het resultaat: de kwantumbehoefte per $\mathrm{CO_2}$, het
        rendement van foton tot glucose, en het rendement van zonlicht tot
        de netto suiker van het \omterm{def:b1:flowering-plant-organization:organs}{blad}.
\end{enumerate}
\end{problem}
