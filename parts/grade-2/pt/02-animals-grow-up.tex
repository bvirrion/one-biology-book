\chapter{Como os animais crescem}\label{ch:g2:animals-grow-up}

Um gatinho já parece um gato pequeno desde o primeiro dia. Mas o
sapo da lagoa começou como um girino em forma de peixe, sem \omterm{def:g1:my-body:parts}{pernas},
e a borboleta pousada na \omterm{def:g1:plants-around-us:parts}{flor} passou semanas como uma \omterm{ex:g2:animals-grow-up:butterfly}{lagarta}
rastejante. Nem todo filhote se parece com os pais --- alguns
precisam se transformar por inteiro no caminho até a vida adulta.

\section{Do ovo ou de barriga}

\begin{definition}[Dois jeitos de nascer]\label{def:g2:animals-grow-up:born}
Os filhotes de \omterm{def:g1:animals-around-us:animal}{animais} chegam ao mundo de dois jeitos:
\begin{itemize}
  \item a maioria sai de um \emph{ovo}\index{ovo} posto pela mãe:
        aves, peixes, sapos, caracóis, insetos;
  \item outros \emph{nascem de barriga}\index{nascer de barriga},
        saindo do corpo da mãe já formados, e bebem o
        \emph{leite}\index{leite} dela: gatos, cachorros, vacas,
        coelhos --- e os seres humanos.
\end{itemize}
\end{definition}

\begin{example}[No galinheiro e na cestinha]\label{ex:g2:animals-grow-up:henhouse}
A galinha põe \omterm{def:g2:animals-grow-up:born}{ovos} e choca eles para mantê-los quentinhos; depois de
três semanas, os pintinhos quebram a casca e saem. Na cestinha ao
lado do fogão, os gatinhos da gata nasceram de barriga na mesma
noite, e acharam o \omterm{def:g2:animals-grow-up:born}{leite} dela em menos de uma hora.
\end{example}

\begin{remark}[Um ovo não é uma pedra]\label{rem:g2:animals-grow-up:egg}
Dentro de um \omterm{def:g2:animals-grow-up:born}{ovo} chocado, um pintinho está sendo montado em silêncio,
alimentado pela gema amarela. O \omterm{def:g2:animals-grow-up:born}{ovo} não está esperando --- ele está
trabalhando. É por isso que a galinha mantém os \omterm{def:g2:animals-grow-up:born}{ovos} quentinhos, e
por isso que um \omterm{def:g2:animals-grow-up:born}{ovo} da geladeira, que nunca ia chocar mesmo, não
pode culpar o frio.
\end{remark}

\section{Crescer sem se transformar}

\begin{definition}[Crescer parecido com os pais]\label{def:g2:animals-grow-up:direct}
Muitos filhotes são \emph{miniaturas dos pais}\index{miniatura dos
pais}: o gatinho é um gato pequeno, o potrinho é um cavalo pequeno,
o pintinho é uma galinha pequena. Para eles, crescer quer dizer
ficar maior e mais forte --- a forma do corpo continua a mesma.
\end{definition}

\begin{example}[De gatinho a gato]\label{ex:g2:animals-grow-up:kitten}
O gatinho recém-nascido mama de \omterm{def:g1:the-five-senses:sense}{olhos} fechados. Com um mês, ele
brinca e dá botes; com seis meses, caça como um gato adulto; com um
ano, ele é adulto, talvez já com gatinhos próprios. Gato pequeno,
gato maior, gato: a mesma forma o caminho inteiro.
\end{example}

\section{Crescer se transformando}

\begin{definition}[Metamorfose]\label{def:g2:animals-grow-up:metamorphosis}
Alguns filhotes não se parecem nada com os pais e precisam mudar a
forma do corpo inteiro para virar adultos. Essa grande mudança se
chama \emph{metamorfose}\index{metamorfose}. O girino vira sapo; a
\omterm{ex:g2:animals-grow-up:butterfly}{lagarta} vira borboleta.
\end{definition}

\begin{example}[A estrada do sapo]\label{ex:g2:animals-grow-up:frog}
Dos \omterm{def:g2:animals-grow-up:born}{ovos} do sapo na lagoa saem os \emph{girinos}\index{girino}: de
cabeça redonda, rabo comprido, sem \omterm{def:g1:my-body:parts}{pernas}, respirando na água como
peixes. Depois, semana após semana: aparecem as \omterm{def:g1:my-body:parts}{pernas} de trás,
depois as da frente; o rabo vai encolhendo; e um dia um sapinho sobe
na margem. O mesmo \omterm{def:g1:animals-around-us:animal}{animal} --- reconstruído.
\end{example}

\begin{example}[A estrada da borboleta]\label{ex:g2:animals-grow-up:butterfly}
Do \omterm{def:g2:animals-grow-up:born}{ovo} da borboleta sai uma \emph{lagarta}\index{lagarta}, que quase
só faz comer \omterm{def:g1:plants-around-us:parts}{folhas} e crescer. Um dia ela se enrola num estojo, a
\emph{crisálida}\index{crisálida}, e para de se mexer. Lá dentro, o
corpo é reconstruído; depois de algumas semanas o estojo se abre, e
de lá sai uma borboleta, abrindo asas novas e molhadas. \omterm{def:g2:animals-grow-up:born}{Ovo}, \omterm{ex:g2:animals-grow-up:butterfly}{lagarta},
\omterm{ex:g2:animals-grow-up:butterfly}{crisálida}, borboleta: quatro etapas de um único e mesmo \omterm{def:g1:animals-around-us:animal}{animal}.
\end{example}

\begin{method}[Ler o ciclo de vida de um animal]\label{met:g2:animals-grow-up:read}
Para qualquer \omterm{def:g1:animals-around-us:animal}{animal} que você encontrar:
\begin{enumerate}
  \item pergunte como ele nasce: de um \omterm{def:g2:animals-grow-up:born}{ovo} ou de barriga?
  \item pergunte com o que o filhote se parece: uma \omterm{def:g2:animals-grow-up:direct}{miniatura dos
        pais} ou uma forma diferente?
  \item se for uma forma diferente, ele vai se transformar: procure
        as etapas da \omterm{def:g2:animals-grow-up:metamorphosis}{metamorfose} dele;
  \item desenhe o ciclo como uma roda, com uma seta do adulto de
        volta aos \omterm{def:g2:animals-grow-up:born}{ovos}.
\end{enumerate}
\end{method}

\begin{remark}[Leite, ou direto ao trabalho]\label{rem:g2:animals-grow-up:care}
Os gatinhos são alimentados e lavados pela mãe durante semanas; um
girino recém-saído do \omterm{def:g2:animals-grow-up:born}{ovo} não recebe ajuda nenhuma e precisa se
alimentar sozinho desde o primeiro dia. Alguns filhotes são cuidados,
outros começam a vida sozinhos --- as duas estradas levam à vida
adulta.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:1}
Diga os dois jeitos de os filhotes chegarem ao mundo, com um exemplo
de cada.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:2}
De \omterm{def:g2:animals-grow-up:born}{ovo} ou de barriga: (a) um pintinho; (b) um gatinho; (c) um sapo;
(d) um coelho; (e) um caracol?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:3}
O que um gatinho bebe primeiro? Quais \omterm{def:g1:animals-around-us:animal}{animais} deste capítulo bebem a
mesma coisa ao nascer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:4}
O que é \omterm{def:g2:animals-grow-up:metamorphosis}{metamorfose}? Diga os dois exemplos famosos deste capítulo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:5}
Ponha as etapas do sapo em ordem: sapo --- \omterm{def:g2:animals-grow-up:born}{ovos} --- girino com
\omterm{def:g1:my-body:parts}{pernas} --- girino.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:6}
Ponha as etapas da borboleta em ordem: borboleta --- \omterm{ex:g2:animals-grow-up:butterfly}{lagarta} --- \omterm{def:g2:animals-grow-up:born}{ovo}
--- \omterm{ex:g2:animals-grow-up:butterfly}{crisálida}. Em que etapa o \omterm{def:g1:animals-around-us:animal}{animal} come \omterm{def:g1:plants-around-us:parts}{folhas} e cresce mais?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:7}
Qual filhote é uma \omterm{def:g2:animals-grow-up:direct}{miniatura dos pais}: o girino ou o potrinho? O que
crescer quer dizer para esse aí?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:8}
Por que a galinha choca os \omterm{def:g2:animals-grow-up:born}{ovos}? O que está acontecendo dentro
deles, segundo a \cref{rem:g2:animals-grow-up:egg}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:animals-grow-up:9}
Na lagoa, a Zoe encontra um bichinho de cabeça redonda, rabo
comprido, duas \omterm{def:g1:my-body:parts}{pernas} de trás e nenhuma \omterm{def:g1:my-body:parts}{perna} da frente. Usando a
figura do ciclo do sapo, diga exatamente onde ele está na estrada
dele, o que ele era antes e o que vem depois.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:animals-grow-up:10}
``A \omterm{ex:g2:animals-grow-up:butterfly}{lagarta} morreu dentro do estojo e uma borboleta se mudou para
lá.'' O que aconteceu de verdade dentro da \omterm{ex:g2:animals-grow-up:butterfly}{crisálida}? Responda com a
\cref{def:g2:animals-grow-up:metamorphosis} e o
\cref{ex:g2:animals-grow-up:butterfly}.
\end{exercise}
