\section*{Chapitre \ref{ch:g12:respiration-fermentation} --- La respiration cellulaire et la fermentation}

\begin{solution}{exo:g12:respiration-fermentation:1}
Un \omterm{def:g10:universal-dna:nucleotide}{nucléotide} portant trois phosphates à la file, la monnaie
énergétique de la \omterm{def:g10:cells-common-unit:cell}{cellule}. L'utiliser retire le dernier phosphate, ce
qui libère environ \qty{30}{kJ} par mole pour entraîner un processus,
et laisse de l'ADP à recharger.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:2}
\omterm{prop:g12:respiration-fermentation:glycolysis}{Glycolyse}, dans le \omterm{def:g10:cells-common-unit:cell}{cytoplasme}~: 2 \omterm{def:g12:respiration-fermentation:atp}{ATP} et 2 NAD réduits, deux
pyruvates. \omterm{prop:g12:respiration-fermentation:mitochondrion}{Cycle de Krebs}, dans la matrice mitochondriale~: du dioxyde
de carbone, des transporteurs chargés, 2 \omterm{def:g12:respiration-fermentation:atp}{ATP}. Chaîne respiratoire,
dans la membrane interne~: les transporteurs déchargés sur l'oxygène,
de l'eau, environ 26 \omterm{def:g12:respiration-fermentation:atp}{ATP}.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:3}
Deux membranes, l'interne repliée en crêtes, enfermant la matrice.
Matrice~: le \omterm{prop:g12:respiration-fermentation:mitochondrion}{cycle de Krebs}. Membrane interne~: la chaîne respiratoire
et la synthèse d'\omterm{def:g12:respiration-fermentation:atp}{ATP}.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:4}
Elle décharge sur le pyruvate le NAD réduit de la \omterm{prop:g12:respiration-fermentation:glycolysis}{glycolyse}, si bien
que la \omterm{prop:g12:respiration-fermentation:glycolysis}{glycolyse} peut continuer sans oxygène. Rendement~: les 2 \omterm{def:g12:respiration-fermentation:atp}{ATP} de
la \omterm{prop:g12:respiration-fermentation:glycolysis}{glycolyse} seulement, plus du lactate ou de l'éthanol et du
$\mathrm{CO_2}$.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:5}
Environ 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} contre 2. La respiration laisse du dioxyde de carbone
et de l'eau~; la \omterm{def:g10:cell-metabolism:fermentation}{fermentation} laisse du lactate, ou de l'éthanol et du
dioxyde de carbone, encore riches en énergie.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:6}
Glucose~: aucun effet --- les \omterm{def:g10:cells-common-unit:organelle}{mitochondries} ne savent pas
l'utiliser. Pyruvate~: la consommation d'oxygène démarre --- c'est le
\omterm{def:g11:enzymes-and-phenotype:enzyme}{substrat} de la \omterm{def:g10:cells-common-unit:organelle}{mitochondrie}. ADP~: la consommation
s'accélère --- l'usage de l'oxygène est couplé à la synthèse d'\omterm{def:g12:respiration-fermentation:atp}{ATP}.
Cyanure~: la consommation s'arrête --- la chaîne respiratoire est le
lieu où l'oxygène est utilisé.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:7}
La \omterm{prop:g12:respiration-fermentation:glycolysis}{glycolyse}, qui convertit le glucose en pyruvate, se déroule dans le
\omterm{def:g10:cells-common-unit:cell}{cytoplasme}~; la \omterm{def:g10:cells-common-unit:organelle}{mitochondrie} n'en possède pas les \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymes} et fait
entrer du pyruvate, non du glucose.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:8}
Chaque glucose charge deux NAD~; la \omterm{def:g10:cells-common-unit:cell}{cellule} ne dispose que d'un petit
stock de NAD, et une fois celui-ci entièrement chargé, la \omterm{prop:g12:respiration-fermentation:glycolysis}{glycolyse}
n'a plus de transporteur à qui remettre son hydrogène et s'arrête. La
\omterm{def:g10:cell-metabolism:fermentation}{fermentation} décharge le NAD sur le pyruvate et maintient la voie en
marche.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:9}
Respiration~: $60/30 = 2$ glucoses par seconde. \omterm{def:g10:cell-metabolism:fermentation}{Fermentation}~:
$60/2 = 30$ glucoses par seconde, quinze fois plus.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:10}
Le cyanure bloque la chaîne respiratoire~: aucun électron n'atteint
plus l'oxygène et environ 26 des 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} par glucose sont perdus
aussitôt~; la \omterm{def:g10:cell-metabolism:fermentation}{fermentation} ne peut pas combler le manque. Le cerveau
et le cœur consomment l'\omterm{def:g12:respiration-fermentation:atp}{ATP} le plus vite et disposent des plus faibles
réserves~: ils défaillent en quelques minutes.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:11}
Le dioxyde de carbone est libéré dans la matrice mitochondriale, par
le \omterm{prop:g12:respiration-fermentation:mitochondrion}{cycle de Krebs} (et par l'étape qui y fait entrer le pyruvate)~:
aucun dans la \omterm{prop:g12:respiration-fermentation:glycolysis}{glycolyse}. L'eau est produite à la fin de la chaîne
respiratoire, quand les électrons et les protons rejoignent l'oxygène.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:12}
Le muscle a obtenu 2 \omterm{def:g12:respiration-fermentation:atp}{ATP} par glucose fermenté~; reconstruire ce
glucose à partir du lactate coûte au foie environ 6 \omterm{def:g12:respiration-fermentation:atp}{ATP}, payés en
respirant un autre carburant. L'organisme a dépensé 6 pour en obtenir
2~: la \omterm{def:g10:cell-metabolism:fermentation}{fermentation} emprunte une énergie qu'il faut rembourser avec
intérêts.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:13}
À l'air, la respiration donne 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} par glucose~: il faut donc peu de
glucose et rien n'est détourné vers l'éthanol~; enfermée, la
\omterm{def:g10:cell-metabolism:fermentation}{fermentation} en donne 2~: on consomme quinze fois plus de glucose et
il se fait de l'éthanol. La pâte lève grâce au dioxyde de carbone, que
les deux voies produisent (six par glucose en respiration, deux en
\omterm{def:g10:cell-metabolism:fermentation}{fermentation})~; et dans une pâte dense, l'oxygène vient de toute façon
vite à manquer.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:14}
De la chaleur~: l'énergie de la chaîne, n'étant plus captée sous forme
d'\omterm{def:g12:respiration-fermentation:atp}{ATP}, est libérée directement. Un nouveau-né perd sa chaleur vite et
frissonne mal~; la graisse brune est un chauffage intégré.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:15}
Électrons~: la \omterm{def:g10:cell-metabolism:autotrophy}{photosynthèse} les prend à l'eau et les met dans le
sucre~; la respiration les prend au sucre et les donne à l'oxygène.
Gaz~: la \omterm{def:g10:cell-metabolism:autotrophy}{photosynthèse} consomme du $\mathrm{CO_2}$ et libère du
$\mathrm{O_2}$~; la respiration fait l'inverse. Énergie~: la
\omterm{def:g10:cell-metabolism:autotrophy}{photosynthèse} fait entrer de la lumière et la stocke dans le sucre~;
la respiration la tire du sucre sous forme d'\omterm{def:g12:respiration-fermentation:atp}{ATP} et de chaleur.
\omterm{def:g10:cells-common-unit:organelle}{Organites}~: le \omterm{def:g12:photosynthesis:chloroplast}{chloroplaste} et la \omterm{def:g10:cells-common-unit:organelle}{mitochondrie}. Chacune consomme ce
que l'autre produit~: la \omterm{def:g10:cell-metabolism:autotrophy}{photosynthèse} épuiserait le $\mathrm{CO_2}$
et la respiration le $\mathrm{O_2}$ et la nourriture.
\end{solution}

\begin{solution}{pb:g12:respiration-fermentation:1}
\textbf{1.} Le tiers de \qty{1000}{W} fait \qty{300}{W}~:
\qty{18}{kJ} par minute, soit \qty{0.6}{mol} d'\omterm{def:g12:respiration-fermentation:atp}{ATP} par minute.

\textbf{2.} Environ \qty{300}{g} par minute --- contre quelques
grammes détenus~: le stock est recyclé en continu.

\textbf{3.} \qty{50}{g} font \qty{0.1}{mol}~: chaque molécule est
rechargée environ 6 fois par minute.

\textbf{4.} $0.6/30 = \qty{0.02}{mol}$ de glucose par minute (environ
\qty{3.6}{g}).

\textbf{5.} $6 \times 0.02 = \qty{0.12}{mol}$ de $\mathrm{O_2}$, soit
environ \qty{2.9}{L} par minute --- au-dessous de son maximum de
\qty{4}{L/min}, et cohérent avec les \qty{20}{kJ} par litre d'oxygène
du \cref{ch:g10:exercise-and-energy}~: les \qty{1000}{W} peuvent être
soutenus par la respiration.

\textbf{6.} 2 dans le \omterm{def:g10:cells-common-unit:cell}{cytoplasme}, 28 dans la \omterm{def:g10:cells-common-unit:organelle}{mitochondrie}~: environ
93\% de l'\omterm{def:g12:respiration-fermentation:atp}{ATP}.

\textbf{7.} $6 \times 0.02 = \qty{0.12}{mol}$ de $\mathrm{CO_2}$ par
minute, provenant toutes du \omterm{prop:g12:respiration-fermentation:mitochondrion}{cycle de Krebs} et de l'entrée du pyruvate
dans la matrice.

\textbf{8.} $30 \times 30 = \qty{900}{kJ}$ sur les \qty{2870}{kJ},
environ un tiers~; le reste part en chaleur, qu'elle évacue par la
sueur et par le flux sanguin vers sa peau.

\textbf{9.} Les \omterm{def:g10:chemistry-of-life:families}{protéines} de la chaîne et les \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymes} qui fabriquent
l'\omterm{def:g12:respiration-fermentation:atp}{ATP} siègent dans la membrane~; la vitesse de synthèse de l'\omterm{def:g12:respiration-fermentation:atp}{ATP} est
proportionnelle à la surface qui les porte.

\textbf{10.} Cela élève la vitesse à laquelle la respiration peut
fabriquer de l'\omterm{def:g12:respiration-fermentation:atp}{ATP} (et la part de graisse qu'elle peut brûler), donc
la puissance qu'elle peut soutenir sans \omterm{def:g10:cell-metabolism:fermentation}{fermentation}~; cela ne change
ni le rendement par glucose ni l'oxygène par glucose.

\textbf{11.} $800 \times 30 = \qty{24}{kJ}$~: \qty{0.8}{mol} d'\omterm{def:g12:respiration-fermentation:atp}{ATP}.

\textbf{12.} Respiration~: $400 \times 30 = \qty{12}{kJ}$, soit
\qty{0.4}{mol}~; la \omterm{def:g10:cell-metabolism:fermentation}{fermentation} doit fournir les \qty{0.4}{mol}
restantes.

\textbf{13.} $0.4/2 = \qty{0.2}{mol}$ de glucose, laissant
\qty{0.4}{mol} de lactate.

\textbf{14.} Respiration~: $1/30$ de glucose par \omterm{def:g12:respiration-fermentation:atp}{ATP}~; \omterm{def:g10:cell-metabolism:fermentation}{fermentation}~:
$1/2$ --- quinze fois plus de glucose. Le muscle l'accepte parce qu'il
lui faut de l'\omterm{def:g12:respiration-fermentation:atp}{ATP} tout de suite et que le \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} est là~; la dette
se rembourse plus tard.

\textbf{15.} Le lactate doit être oxydé ou reconstruit en glucose, ce
qui coûte de l'\omterm{def:g12:respiration-fermentation:atp}{ATP} fabriqué par la respiration~: la consommation
d'oxygène reste au-dessus du repos tant que la dette n'est pas
soldée.

\textbf{16.} \qty{500}{W} pendant une heure font \qty{1800}{kJ}~: $1800/38
\approx \qty{47}{g}$ de graisse.

\textbf{17.} La graisse est plus réduite --- plus riche en
hydrogène ---, si bien que chaque gramme exige plus d'oxygène pour
être entièrement oxydé~; par litre d'oxygène, elle rend un peu moins
d'énergie que le glucose.

\textbf{18.} La \omterm{prop:g12:respiration-fermentation:glycolysis}{glycolyse} et sa \omterm{def:g10:cell-metabolism:fermentation}{fermentation}~: la graisse ne peut pas
être fermentée, de sorte qu'un sprint, qui repose sur la \omterm{def:g10:cell-metabolism:fermentation}{fermentation},
ne peut pas du tout l'utiliser --- seulement du glucose et du
\omterm{def:g10:exercise-and-energy:glycogen}{glycogène}.

\textbf{19.} Le foie libère du glucose à partir de son \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} et en
reconstruit une partie à partir du lactate, maintenant la glycémie
près de \qty{1}{g/L} (\cref{ch:g12:glucose-and-diabetes})~; l'apport
du cerveau est protégé.

\textbf{20.} Environ \qty{0.6}{mol} d'\omterm{def:g12:respiration-fermentation:atp}{ATP} par minute~; environ
\qty{3}{L} d'oxygène par minute les paient~; la chaîne respiratoire de
la membrane interne mitochondriale en fabrique quelque neuf dixièmes.
\end{solution}
