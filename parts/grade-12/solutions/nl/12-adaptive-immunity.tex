\section*{Hoofdstuk \ref{ch:g12:adaptive-immunity} --- Verworven afweer en vaccinatie}

\begin{solution}{exo:g12:adaptive-immunity:1}
Een \omterm{def:g12:adaptive-immunity:antigen}{antigeen} is elke molecule die de \omterm{def:g12:innate-immunity:immunity}{verworven afweer} kan herkennen; een
\omterm{def:g12:adaptive-immunity:antigen}{lymfocyt} is een witte bloedcel die ontvangers voor één \omterm{def:g12:adaptive-immunity:antigen}{antigeen} draagt.
Haar gerichtheid komt van haar ontvanger, die tijdens het rijpen
willekeurig in elkaar is gezet en op al haar kopieën gelijk is.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:2}
Het \omterm{def:g12:adaptive-immunity:antigen}{antigeen} bindt de weinige \omterm{def:g12:adaptive-immunity:antigen}{lymfocyten} waarvan de ontvanger past; die
\omterm{def:g10:cells-common-unit:cell}{cellen} vermenigvuldigen zich over enkele dagen tot een kloon; de kloon
splitst zich in uitvoerende \omterm{def:g10:cells-common-unit:cell}{cellen} die vechten; een minderheid wordt
\omterm{prop:g12:adaptive-immunity:selection}{geheugencellen} die blijven bestaan.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:3}
Een Y-vormig \omterm{def:g11:gene-expression:protein}{eiwit} dat door plasmacellen wordt afgescheiden, een oplosbare
kopie van de ontvanger van de \omterm{def:g10:cells-common-unit:cell}{B-cel}; elke arm bindt het \omterm{def:g12:adaptive-immunity:antigen}{antigeen}. Op een
bacterie vormt zij een laag waar \omterm{def:g12:innate-immunity:phagocytosis}{fagocyten} vat op hebben, en zij knoopt
bacteriën aaneen tot complexen die worden opgeslokt.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:4}
\omterm{prop:g12:adaptive-immunity:tcells}{Helper-T-cellen}, geactiveerd door \omterm{prop:g12:innate-immunity:handover}{dendritische cellen}, scheiden de
cytokinen af die de \omterm{def:g10:cells-common-unit:cell}{B-cellen} en de andere \omterm{def:g10:cells-common-unit:cell}{T-cellen} toestaan en versterken.
Cytotoxische \omterm{def:g10:cells-common-unit:cell}{T-cellen} doden \omterm{def:g10:cells-common-unit:cell}{cellen} die brokstukken van een virus of van
een gezwelseiwit tonen.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:5}
Een onschadelijke vorm van het \omterm{def:g12:adaptive-immunity:antigen}{antigeen} van de ziekteverwekker (verzwakt,
gedood, een \omterm{def:g11:gene-expression:protein}{eiwit}, een onwerkzaam gemaakt gif, of de instructies ervoor)
met een adjuvans. Het lokt een primaire reactie uit en laat \omterm{prop:g12:adaptive-immunity:selection}{geheugencellen}
achter, zodat de echte ziekteverwekker een secundaire reactie ontmoet.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:6}
Primair: een week vertraging, een lage piek (ongeveer een tiende van die
van de secundaire) na twee tot drie weken, een daling binnen een maand.
Secundair: twee of drie dagen vertraging, een tien keer hogere piek binnen
tien dagen, die maanden aanhoudt.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:7}
Het geheugen is gericht: de \omterm{prop:g12:adaptive-immunity:selection}{geheugencellen} die de eerste dosis A achterliet
reageren alleen op A; B, dat voor het eerst wordt ontmoet, krijgt een
primaire reactie.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:8}
\omterm{prop:g12:adaptive-immunity:antibodies}{Antistoffen} zijn \omterm{def:g10:chemistry-of-life:families}{eiwitten} in de lichaamsvochten en kunnen geen \omterm{def:g10:cells-common-unit:cell}{cellen}
binnengaan. Een besmette \omterm{def:g10:cells-common-unit:cell}{cel} toont brokstukken van het virus op haar
oppervlak, en cytotoxische \omterm{def:g10:cells-common-unit:cell}{T-cellen} herkennen die en doden de \omterm{def:g10:cells-common-unit:cell}{cel}.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:9}
$1 - 1/4 = 75\%$; $1 - 1/18 \approx 94\%$.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:10}
Boven de drempel besmet elk geval minder dan één ander, dus kan het virus
niet rondgaan en bereikt het zelden een zuigeling; de volwassenen met
afweer zijn de muur om hen heen.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:11}
Ongeveer 750, 520 en 180 per microliter; \omterm{prop:g12:adaptive-immunity:hiv}{aids} begint rond jaar 9, wanneer
het aantal onder de 200 zakt.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:12}
De cytokinen van de helper-T-cellen zijn wat de \omterm{def:g10:cells-common-unit:cell}{B-cellen} toestaat
plasmacellen te worden en de cytotoxische \omterm{def:g10:cells-common-unit:cell}{T-cellen} om zich te
vermenigvuldigen; beide armen hangen van dezelfde coördinator af, dus legt
hem weghalen ze allebei stil.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:13}
De \omterm{prop:g12:adaptive-immunity:antibodies}{antistoffen} waren geselecteerd tegen het virus zoals het was; het virus
muteert zijn \omterm{def:g10:chemistry-of-life:families}{oppervlakte-eiwitten} sneller dan er nieuwe klonen kunnen
worden geselecteerd, dus wordt elke reactie voorbijgestreefd. En het virus
verbergt en vermenigvuldigt zich juist in de helpercellen die de reactie
zouden coördineren.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:14}
\omterm{prop:g12:adaptive-immunity:selection}{Geheugencellen} herkennen de \omterm{def:g10:chemistry-of-life:families}{oppervlakte-eiwitten} die het \omterm{def:g12:adaptive-immunity:vaccine}{vaccin} heeft
getoond. Die van de griep muteren elk jaar tot vormen die het oude
geheugen niet bindt; de \omterm{def:g10:chemistry-of-life:families}{oppervlakte-eiwitten} van de mazelen veranderen
nauwelijks, zodat het geheugen van 1970 nog altijd op het virus van
vandaag past.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:15}
Wat varieert: de ontvangers van de \omterm{def:g12:adaptive-immunity:antigen}{lymfocyten}, willekeurig gemaakt. Wat
selecteert: het \omterm{def:g12:adaptive-immunity:antigen}{antigeen}, dat alleen passende klonen laat vermenigvuldigen.
Wat wordt geërfd: de ontvanger, door de nakomelingen van de kloon, met de
\omterm{prop:g12:adaptive-immunity:selection}{geheugencellen} erbij. Tijdschaal: dagen voor een reactie, een leven lang
voor het geheugen, tegenover generaties voor een \omterm{def:g12:selection-drift-speciation:population}{populatie}. De vergelijking
houdt op bij de overdracht: de geselecteerde klonen sterven met het lichaam
en gaan niet naar de volgende generatie.
\end{solution}

\begin{solution}{pb:g12:adaptive-immunity:1}
\textbf{1.} Ongeveer een week niets, een bescheiden piek na twee tot drie
weken, een daling in de weken daarna.

\textbf{2.} \omterm{prop:g12:adaptive-immunity:antibodies}{Antistoffen} binnen twee of drie dagen, een tien keer hogere
piek, die maanden aanhoudt: de \omterm{prop:g12:adaptive-immunity:selection}{geheugencellen} die de eerste dosis
achterliet vormen al een grote kloon, dus begint het vermenigvuldigen bij
duizenden \omterm{def:g10:cells-common-unit:cell}{cellen} in plaats van bij een handvol.

\textbf{3.} Een levend virus vermenigvuldigt zich kort binnen in \omterm{def:g10:cells-common-unit:cell}{cellen},
zodat besmette \omterm{def:g10:cells-common-unit:cell}{cellen} zijn brokstukken tonen en er naast \omterm{prop:g12:adaptive-immunity:antibodies}{antistoffen} ook
cytotoxische \omterm{def:g10:cells-common-unit:cell}{T-cellen} worden geselecteerd; een gedood virus oefent vooral
de arm van de \omterm{prop:g12:adaptive-immunity:antibodies}{antistoffen}.

\textbf{4.} \omterm{prop:g12:adaptive-immunity:antibodies}{Antistoffen} die er al zijn houden veel van het virus tegen;
\omterm{prop:g12:adaptive-immunity:selection}{geheugencellen} B en T vermenigvuldigen zich binnen twee dagen; besmette
\omterm{def:g10:cells-common-unit:cell}{cellen} worden gedood voordat het virus zich verspreidt. De infectie wordt
gestopt voordat er klachten zijn.

\textbf{5.} Drie dagen stille vermeerdering, dan koorts, uitslag en
ziekte terwijl de primaire reactie zich over een week opbouwt; herstel
rond dag tien, met \omterm{prop:g12:adaptive-immunity:selection}{geheugencellen} en levenslange afweer --- als hij de
verwikkelingen overleeft.

\textbf{6.} $1 - 1/15 \approx 93\%$.

\textbf{7.} $15 \times 0.10 = 1.5$: elk geval besmet meer dan één ander,
dus kan de ziekte zich verspreiden. Niet beschermd.

\textbf{8.} $10\% \times \num{50000} = 5000$ zonder afweer, waarvan er
1500 zuigeling zijn.

\textbf{9.} 2500 vatbaar; $15 \times 0.05 = 0.75$: minder dan één, de
ziekte sterft uit.

\textbf{10.} Zuigelingen kunnen niet worden ingeënt, dus hun enige
bescherming is dat de volwassenen om hen heen, doordat zij afweer hebben,
het virus geen weg naar hen laten.

\textbf{11.} Generatie 3: $1.5^3 \approx 3.4$ gevallen; generatie 6:
$1.5^6 \approx 11$.

\textbf{12.} $1 + 1.5 + 2.25 + 3.4 + 5.1 + 7.6 + 11.4 \approx 32$ gevallen.

\textbf{13.} Ongeveer 30\% van 32: zo'n 10 zuigelingen.

\textbf{14.} Elk geval haalt een vatbaar mens weg en voegt er een met
afweer bij, dus daalt het vatbare deel en daarmee het aantal dat elk geval
besmet; met 5000 vatbaren zou de uitbraak pas na honderden gevallen
vertragen, over maanden --- tenzij er wordt ingeënt.

\textbf{15.} Bij 95\%: $0.75^3 \approx 0.4$ en $0.75^6 \approx 0.18$
gevallen; in totaal ongeveer $1 + 0.75 + 0.56 + \dots \approx 4$ gevallen.
Tien keer minder gevallen, en de uitbraak dooft zichzelf uit.

\textbf{16.} 800 in 8 jaar: 100 per microliter per jaar, ongeveer 8 per
maand.

\textbf{17.} \omterm{def:g10:cells-common-unit:cell}{B-cellen} hebben de cytokinen van de helpercellen nodig om
plasmacellen te worden, en cytotoxische \omterm{def:g10:cells-common-unit:cell}{T-cellen} hebben hen nodig om zich
te vermenigvuldigen: met te weinig helpers wordt geen van beide armen
toegestaan, ook al bestaan hun \omterm{def:g10:cells-common-unit:cell}{cellen}.

\textbf{18.} Ja: \omterm{prop:g12:adaptive-immunity:selection}{geheugencellen} hebben nog altijd cytokinen van helpers
nodig om tot een secundaire reactie uit te groeien; zonder coördinatie kan
het geheugen niet worden gebruikt, en mazelen, onschuldig voor een gezonde
volwassene, kunnen dodelijk zijn.

\textbf{19.} De helperpopulatie groeit weer aan uit overlevende \omterm{def:g10:cells-common-unit:cell}{cellen} en
uit het merg, en de \omterm{prop:g12:adaptive-immunity:selection}{geheugencellen} van de andere armen zijn nooit
vernietigd --- het repertoire is heel zodra de coördinatie terug is. Het
virus blijft in verborgen \omterm{def:g10:cells-common-unit:cell}{cellen} zitten, dus laat de middelen stoppen het
weer opleven.

\textbf{20.} 93\% dekking; ongeveer 32 gevallen uit één reiziger bij 90\%;
de helper-T-cel.
\end{solution}
