\begin{solution}{exo:g7:plant-structure:1}

\begin{solution}{exo:g7:plant-structure:2}
Extensions of \omterm{def:g7:plant-structure:organs}{root} epidermal \omterm{def:g7:microscope:cell}{cells}; increase surface area for absorption
of \omterm{def:g4:plants-need:needs}{water} and \omterm{def:g6:nutrition:groups}{minerals}.
\end{solution}

\begin{solution}{exo:g7:plant-structure:3}
\omterm{def:g7:plant-structure:leaf}{Palisade mesophyll}. It sits near the upper surface where \omterm{def:g4:plants-need:needs}{light} is
strongest, so packing \omterm{def:g7:microscope:parts}{chloroplasts} there maximises \omterm{def:g4:plants-need:needs}{light} capture.
\end{solution}

\begin{solution}{exo:g7:plant-structure:4}
