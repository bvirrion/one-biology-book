\chapter{الخلايا: وحدة الحياة المشتركة}\label{ch:g10:cells-common-unit}

اكشط باطن خدك بعود قطن، وافرشه على شريحة زجاجية، وأضف قطرة من صباغ
أزرق، وانظر بالمجهر \omterm{def:g10:cells-common-unit:magnification}{بتكبير} أربعمئة مرة: بلاطات مسطحة غير منتظمة
متناثرة، في كل منها بقعة أغمق قرب مركزها. اسلخ الغشاء الرقيق من باطن
حرشفة بصل وانظر من جديد: قوالب مستطيلة مرتبة، لكل منها بقعتها. وماء
البركة يبين مخلوقات بيضوية تجدف مارّة بإطار من الأهداب. والعينات
الثلاث لا شيء يجمعها بالعين المجردة. أما تحت العدسة فهي مبنية من
الشيء نفسه --- \omterm{def:g10:cells-common-unit:cell}{الخلية} --- وهذا الفصل عما هو ذلك الشيء، وكيف يرى،
ولماذا كان كل كائن حي مصنوعًا منه.

\section{النظرية الخلوية}

\begin{definition}[الخلية]\label{def:g10:cells-common-unit:cell}
\emph{الخلية}\index{خلية} هي أصغر وحدة في الكائن الحي تكون هي نفسها
حية: حجم من محتوى مائي، هو \emph{السيتوبلازم}\index{سيتوبلازم}، يحيط
به \emph{غشاء بلازمي}\index{غشاء بلازمي} رقيق يفصلها عن محيطها ويسمح
مع ذلك بعبور مواد منتقاة. والخلية تأخذ مادة وطاقة، وتحولهما، وتنمو،
وتنقسم إلى خليتين.
\end{definition}

\begin{proposition}[النظرية الخلوية]\label{prop:g10:cells-common-unit:theory}
ثلاث عبارات، أرساها الفحص المجهري في القرن التاسع عشر، تكوّن النظرية
الخلوية:
\begin{enumerate}
  \item كل كائن حي مصنوع من \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة أو أكثر؛
  \item \omterm{def:g10:cells-common-unit:cell}{الخلية} هي وحدة البنية والوظيفة الأساسية في الكائنات الحية ---
        فما هو أصغر منها لا يحيا وحده؛
  \item كل \omterm{def:g10:cells-common-unit:cell}{خلية} تأتي من \omterm{def:g10:cells-common-unit:cell}{خلية} سابقة لها، بالانقسام.
\end{enumerate}
\end{proposition}
\begin{proof}[شواهد]
تقوم العبارة الأولى على الفحص المنهجي للنباتات (شلايدن، 1838)
وللحيوانات (شفان، 1839) بمجاهر محسنة: فكل نسيج فحص، مهما اختلف في
المظهر، انحل إلى \omterm{def:g10:cells-common-unit:cell}{خلايا}. أما العبارة الثالثة فأرساها فيرشو (1855)
ومشاهدة \omterm{def:g10:cells-common-unit:cell}{الخلايا} المنقسمة في الأنسجة النامية؛ والبديل --- \omterm{def:g10:cells-common-unit:cell}{خلايا} تنشأ
تلقائيًا من سائل --- لم يشاهد قط، وتجارب باستور على الميكروبات (يعود
إليها \cref{ch:g11:antibiotic-resistance}) أغلقت المسألة. وتتبع
العبارة الثانية من الأولى ومن أن \omterm{def:g10:cells-common-unit:cell}{الخلايا} المعزولة، إذا حفظت في سائل
مناسب، تظل حية، بينما لا يظل كذلك أي جزء من \omterm{def:g10:cells-common-unit:cell}{خلية}.
\end{proof}

\begin{example}[خلية واحدة أم خلايا كثيرة]\label{ex:g10:cells-common-unit:onemany}
الخميرة أو البراميسيوم أو البكتيريا \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة تفعل كل شيء --- تتغذى
وتتحرك وتنقسم. أما البلوطة أو الإنسان فتعاونية من \omterm{def:g10:cells-common-unit:cell}{خلايا} كثيرة
الأصناف، كل صنف يؤدي نصيبه من العمل: فجسم الإنسان يحوي نحو
$\num{3e13}$ \omterm{def:g10:cells-common-unit:cell}{خلية}، من نحو مئتي صنف. وبين الحالتين تقع المستعمرات
والكائنات العديدة \omterm{def:g10:cells-common-unit:cell}{الخلايا} البسيطة، لكن ما من كائن حي مبني من شيء غير
\omterm{def:g10:cells-common-unit:cell}{الخلايا}.
\end{example}

\section{رؤية الخلايا: السلالم والمجاهر}

\begin{definition}[التكبير والقدرة الفاصلة]\label{def:g10:cells-common-unit:magnification}
\emph{تكبير}\index{تكبير} أداة ما هو نسبة حجم الصورة إلى حجم الشيء.
أما \emph{قدرتها الفاصلة}\index{قدرة فاصلة} فهي أصغر مسافة بين نقطتين
تظل تبينهما منفصلتين. وتكبير صورة أكثر مما تبرره القدرة الفاصلة يكبّر
التشوش لا التفصيل.
\end{definition}

\begin{proposition}[أداتان، سلّمان]\label{prop:g10:cells-common-unit:microscopes}
يكبر المجهر الضوئي حتى نحو $\times 1500$ \omterm{def:g10:cells-common-unit:magnification}{بقدرة فاصلة} تبلغ نحو
\qty{0.2}{\micro m}: فهو يبين \omterm{def:g10:cells-common-unit:cell}{الخلايا} والأنوية \omterm{def:g10:cells-common-unit:organelle}{والبلاستيدات الخضراء}
والبكتيريا، بالألوان وحية. أما المجهر الإلكتروني فيستعمل حزمة من
الإلكترونات بدل الضوء: \omterm{def:g10:cells-common-unit:magnification}{قدرة فاصلة} دون \qty{1}{nm}، \omterm{def:g10:cells-common-unit:magnification}{وتكبير} حتى
$\times 10^6$؛ وهو يكشف البنية الداخلية \omterm{def:g10:cells-common-unit:cell}{للخلية} --- الأغشية،
\omterm{def:g10:cells-common-unit:organelle}{والمتقدرات}، \omterm{def:g10:cells-common-unit:organelle}{والريبوزومات}، بل الجزيئات الكبيرة --- لكن على عينة ميتة
مجففة مقطعة شرائح أرق من الميكرومتر.
\end{proposition}
\admitted

\begin{method}[قياس شيء على صورة مجهرية]\label{met:g10:cells-common-unit:measure}
تحمل الصورة المجهرية إما تكبيرًا وإما مسطرة سلّم.
\begin{enumerate}
  \item مع مسطرة السلّم: قس المسطرة على الصفحة (\qty{12}{mm} مثلًا
        لمسطرة مكتوب عليها \qty{10}{\micro m}) ثم قس الشيء
        (\qty{30}{mm})؛ فالحجم الحقيقي هو $30 \times 10/12 =
        \qty{25}{\micro m}$.
  \item مع \omterm{def:g10:cells-common-unit:magnification}{التكبير}: الحجم الحقيقي $=$ الحجم المقيس مقسومًا على
        \omterm{def:g10:cells-common-unit:magnification}{التكبير}؛ \omterm{def:g10:cells-common-unit:cell}{فخلية} مرسومة بطول \qty{20}{mm} عند $\times 400$
        طولها $20/400 = \qty{0.05}{mm} = \qty{50}{\micro m}$.
  \item اذكر الوحدة دائمًا، وحوّل: $\qty{1}{mm} =
        \qty{1000}{\micro m}$، و$\qty{1}{\micro m} = \qty{1000}{nm}$.
\end{enumerate}
\end{method}

\begin{example}[خلايا الخد]\label{ex:g10:cells-common-unit:cheek}
\omterm{def:g10:cells-common-unit:cell}{خلية} خد مرسومة بعرض \qty{24}{mm} من صورة \omterm{def:g10:cells-common-unit:magnification}{بتكبير} $\times 400$ عرضها
$24/400 = \qty{0.06}{mm} = \qty{60}{\micro m}$، ونواتها، المرسومة
\qty{3}{mm}، نحو \qty{8}{\micro m}. والكرية الحمراء
\qty{7}{\micro m}؛ أما بكتيريا الفم، وطولها \qty{1}{\micro m}، فتظهر
نقاطًا تكاد لا تنفصل عند \omterm{def:g10:cells-common-unit:magnification}{التكبير} نفسه.
\end{example}

\section{داخل الخلية}

\begin{definition}[العضية]\label{def:g10:cells-common-unit:organelle}
\emph{العضية}\index{عضية} بنية داخل \omterm{def:g10:cells-common-unit:cell}{الخلية}، يحيط بها غشاؤها الخاص،
تؤدي مهمة محددة. وأهم العضيات:
\begin{itemize}
  \item \emph{النواة}\index{نواة}، وهي تحوي الصبغيات --- أي DNA
        \cref{ch:g10:universal-dna} --- وتتحكم في نشاط \omterm{def:g10:cells-common-unit:cell}{الخلية}؛
  \item \emph{المتقدرات}\index{متقدرة}، وطولها نحو
        \qty{1}{\micro m}، وفيها تفكك الجزيئات العضوية بالأكسجين
        لتحرير طاقة قابلة للاستعمال (التنفس الخلوي،
        \cref{ch:g10:cell-metabolism})؛
  \item في \omterm{def:g10:cells-common-unit:cell}{الخلايا} النباتية، \emph{البلاستيدات
        الخضراء}\index{بلاستيدة خضراء}، وهي خضراء باليخضور، وفيها يجري
        التركيب الضوئي، و\emph{فجوة}\index{فجوة} كبيرة مملوءة ماءً
        تبقي \omterm{def:g10:cells-common-unit:cell}{الخلية} مشدودة.
\end{itemize}
وخارج الغشاء، تحبس \omterm{def:g10:cells-common-unit:cell}{الخلية} النباتية في \emph{جدار خلوي}\index{جدار
خلوي} صلب من السللوز؛ ولا جدار \omterm{def:g10:cells-common-unit:cell}{للخلايا} الحيوانية. ويحوي \omterm{def:g10:cells-common-unit:cell}{سيتوبلازم} كل
\omterm{def:g10:cells-common-unit:cell}{خلية} أيضًا ملايين \emph{الريبوزومات}\index{ريبوزوم}، وهي جسيمات نحو
\qty{25}{nm} تركب \omterm{def:g10:chemistry-of-life:families}{البروتينات}، ولا ترى إلا بالمجهر الإلكتروني.
\end{definition}

\begin{definition}[بدائيات النواة وحقيقيات النواة]\label{def:g10:cells-common-unit:prokaryote}
\omterm{def:g10:cells-common-unit:cell}{الخلية} التي يحبس DNA فيها داخل \omterm{def:g10:cells-common-unit:organelle}{نواة} هي \emph{خلية حقيقية
النواة}\index{خلية حقيقية النواة}: فالحيوانات والنباتات والفطريات
وكثير من الكائنات الوحيدة \omterm{def:g10:cells-common-unit:cell}{الخلية} حقيقيات \omterm{def:g10:cells-common-unit:organelle}{نواة}. أما \emph{الخلية
البدائية النواة}\index{خلية بدائية النواة} فلا \omterm{def:g10:cells-common-unit:organelle}{نواة} لها ولا \omterm{def:g10:cells-common-unit:organelle}{عضيات}
محاطة بأغشية: يقع DNA فيها حرًا في \omterm{def:g10:cells-common-unit:cell}{السيتوبلازم}، وحجمها نحو
\qty{1}{\micro m}، ويحيط بها جدار من نوع خاص. والبكتيريا بدائيات
\omterm{def:g10:cells-common-unit:organelle}{نواة}.
\end{definition}

\begin{example}[قراءة بكتيريا]\label{ex:g10:cells-common-unit:bacterium}
تبين صورة مجهرية إلكترونية لبكتيريا المعي \emph{Escherichia coli}
عصية طولها \qty{2}{\micro m} وعرضها \qty{0.5}{\micro m}: جدار، وغشاء

\section{خلايا كثيرة وأشكال كثيرة: التخصص}

\begin{proposition}[الخلايا المتخصصة]\label{prop:g10:cells-common-unit:specialised}
في الكائن العديد \omterm{def:g10:cells-common-unit:cell}{الخلايا} تتقاسم \omterm{def:g10:cells-common-unit:cell}{الخلايا} المخطط الأساسي نفسه وDNA
نفسه، لكن لكل صنف شكلًا وعدة يوافقان مهمته: فالعصبون يمد ليفًا يبلغ
طوله المتر لينقل الإشارات؛ والكرية الحمراء فقدت نواتها وصارت قرصًا
مسطحًا مكتظًا بالهيموغلوبين؛ والليف العضلي \omterm{def:g10:cells-common-unit:cell}{خلية} عملاقة مملوءة
بروتينات قابضة؛ والشعيرة الماصة أنبوب رفيع يعظم السطح لامتصاص الماء؛
\omterm{def:g10:cells-common-unit:cell}{والخلية} الحارسة في الورقة تغير شكلها لتفتح ثغرًا أو تغلقه. فالتخصص
مسألة أي أجزاء المخطط المشترك تنمّى.
\end{proposition}
\admitted

\begin{example}[السطح والحجم]\label{ex:g10:cells-common-unit:surface}
لماذا كانت \omterm{def:g10:cells-common-unit:cell}{الخلايا} بهذا الصغر؟ كل ما تأخذه \omterm{def:g10:cells-common-unit:cell}{الخلية} أو تخرجه يعبر
غشاءها، ومساحته تنمو مع مربع الحجم الخطي، بينما تنمو الحاجات مع
الحجم، أي مع المكعب. \omterm{def:g10:cells-common-unit:cell}{فالخلية} المكعبة التي ضلعها \qty{10}{\micro m}
سطحها \qty{600}{\micro m^2} لحجم \qty{1000}{\micro m^3}: أي
\qty{0.6}{\micro m^2} من الغشاء لكل ميكرومتر مكعب. وعند ضلع
\qty{100}{\micro m} تهبط النسبة إلى \qty{0.06}{\micro m^2}: أي غشاء
أقل عشر مرات لكل وحدة محتوى. \omterm{def:g10:cells-common-unit:cell}{والخلية} الكبيرة تجوع في مركزها؛ أما
\omterm{def:g10:cells-common-unit:cell}{الخلايا} الكبيرة فعلًا (بيضة ضفدع، ليف عضلي) فهي إما مدخر خامل في
معظمها وإما ممدودة حتى لا تبعد نقطة فيها عن السطح.
\end{example}

\begin{remark}[الخلية نظام مفتوح]\label{rem:g10:cells-common-unit:open}
ليست \omterm{def:g10:cells-common-unit:cell}{الخلية} كيسًا محكم الإغلاق. فالماء والأكسجين والغلوكوز والشوارد
تعبر غشاءها باستمرار، وتخرج فضلاتها من الطريق نفسه؛ \omterm{def:g10:cells-common-unit:cell}{وخلية} خد توضع في
ماء نقي تنتفخ وتنفجر في دقائق، وأخرى توضع في ماء شديد الملوحة تذوي.
أما تحكم الغشاء فيما يدخل ويخرج فموضوع كتاب السنة الأولى الجامعية؛
والمهم الآن أن تتذكر أن \omterm{def:g10:cells-common-unit:cell}{الخلية} تحيا بالتبادل مع محيطها، وتموت حين
يتوقف التبادل.
\end{remark}

\section{مسألة: عدّ خلايا الجسم}

\begin{problem}[{مسألة نهاية الأسبوع --- حصة مجهر، من قشرة البصل على
الشريحة إلى تقدير لعدد الخلايا في جسم إنسان}]
\label{pb:g10:cells-common-unit:1}
يقضي صف بعد ظهر يوم أمام المجهر. والعدسات الشيئية المتوفرة تعطي
تكبيرات كلية قدرها $\times 40$ و$\times 100$ و$\times 400$؛ وعرض
حقل الرؤية عند $\times 100$ هو \qty{1.8}{mm}.

\medskip
\noindent\textbf{الجزء الأول --- البصلة.}
\begin{enumerate}
  \item عند $\times 100$، تتسع 9 \omterm{def:g10:cells-common-unit:cell}{خلايا} بصل تقريبًا جنبًا إلى جنب عبر
        حقل الرؤية. قدّر عرض \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة، بالميكرومترات.
  \item عند $\times 400$ يصير حقل الرؤية أضيق أربع مرات. فما عرضه، وكم
        \omterm{def:g10:cells-common-unit:cell}{خلية} من هذه تتسع عبره؟
  \item يرسم تلميذ \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة بعرض \qty{28}{mm} من الصورة عند
        $\times 400$. ما \omterm{def:g10:cells-common-unit:magnification}{تكبير} الرسم قياسًا إلى \omterm{def:g10:cells-common-unit:cell}{الخلية} الحقيقية؟
  \item يلون اليود \omterm{def:g10:cells-common-unit:organelle}{النواة} بالأصفر البني؛ ويقيسها التلميذ
        \qty{4}{mm} على الرسم. فما قطرها الحقيقي؟
  \item سمِّ البنى التي على التلميذ أن يكتبها على الرسم، وبنية واحدة
        موجودة لا يمكن رؤيتها عند هذا \omterm{def:g10:cells-common-unit:magnification}{التكبير}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الخد والبركة.}
\begin{enumerate}[resume]
  \item تظهر \omterm{def:g10:cells-common-unit:cell}{خلايا} الخد الملونة بأزرق الميثيلين مضلعات مسطحة عرضها نحو
        \qty{60}{\micro m} لكن سمكها \qty{5}{\micro m} فقط. فلماذا هي
        مسطحة، بالنظر إلى الموضع الذي تأتي منه؟
  \item تبين قطرة من ماء بركة براميسيومًا طوله \qty{240}{\micro m}.
        فهل يرى بالعين المجردة؟ وعند أي \omterm{def:g10:cells-common-unit:magnification}{تكبير} يملأ ربع الحقل؟
  \item تُرى نقاط متحركة دقيقة طولها نحو \qty{1}{\micro m} قرب
        البراميسيوم عند $\times 400$. فما هي، على الأرجح، وماذا يضيف
        المجهر الإلكتروني؟
  \item البراميسيوم \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة؛ \omterm{def:g10:cells-common-unit:cell}{وخلية} الخد واحدة من تريليونات. قل في
        جملة عن كل منهما ما معنى «متخصصة» في \omterm{def:g10:cells-common-unit:cell}{خلية} الخد ولماذا لا
        تنطبق على البراميسيوم.
  \item يضيف تلميذ قطرة ماء مالح إلى \omterm{def:g10:cells-common-unit:cell}{خلايا} الخد فيراها تذوي. فسّر بالاعتماد على
        \cref{rem:g10:cells-common-unit:open}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- حجم \omterm{def:g10:cells-common-unit:cell}{الخلية}.}
\begin{enumerate}[resume]
  \item خذ \omterm{def:g10:cells-common-unit:cell}{خلية} بشرية نموذجية مكعبًا ضلعه \qty{20}{\micro m}. احسب
        حجمها بالميكرومترات المكعبة، ثم بالمليمترات المكعبة
        ($\qty{1}{mm^3} = 10^9\,\unit{\micro m^3}$).
  \item احسب مساحة سطحها ونسبة السطح إلى الحجم.
  \item أعد الحساب لبكتيريا تؤخذ مكعبًا ضلعه \qty{1}{\micro m}. بأي
        عامل تكبر نسبتها؟
  \item تأخذ \omterm{def:g10:cells-common-unit:cell}{الخلايا} كتلة تساوي تقريبًا كتلة حجم مماثل من الماء
        ($\qty{1}{g}$ لكل \unit{cm^3}). قدّر كتلة \omterm{def:g10:cells-common-unit:cell}{خلية}
        \qty{20}{\micro m} بالنانوغرامات
        ($\qty{1}{g} = 10^9\,\unit{ng}$).
  \item الكرية الحمراء قرص قطره \qty{7}{\micro m} وسمكه
        \qty{2}{\micro m}. قدّر حجمها (عاملها أسطوانة،
        $V = \pi r^2 h$) وقارنه بمكعب السؤال 11.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- عدد \omterm{def:g10:cells-common-unit:cell}{الخلايا} في الجسم.}
\begin{enumerate}[resume]
  \item تلميذ كتلته \qty{60}{kg} حجمه نحو \qty{60}{L}. فلو كان الجسم
        مصنوعًا من \omterm{def:g10:cells-common-unit:cell}{خلايا} مكعبة ضلعها \qty{20}{\micro m} وحدها، فكم
        \omterm{def:g10:cells-common-unit:cell}{خلية} يحوي؟
  \item والحق أن نحو ثلث حجم الجسم سائل بين \omterm{def:g10:cells-common-unit:cell}{الخلايا} ومواد أخرى غير
        خلوية. صحح التقدير.
  \item الكريات الحمراء وحدها عددها نحو $\num{2.5e13}$. باستعمال حجم
        السؤال 15، ما الحجم الذي تشغله؟ وهل يتوافق ذلك مع حجم دم يبلغ
        نحو \qty{4.5}{L} تشكل \omterm{def:g10:cells-common-unit:cell}{الخلايا} 45\% منه؟
  \item اجمع السؤالين 17 و18: أعطِ رتبة قدر للعدد الكلي \omterm{def:g10:cells-common-unit:cell}{لخلايا} الجسم،
        وقل لماذا يستحيل العد الدقيق.
  \item اذكر النتيجة: كم \omterm{def:g10:cells-common-unit:cell}{خلية}، على وجه التقريب، تصنع إنسانًا، وبأي
        عملية --- ومن أي عدد ابتدائي --- أنتجت كلها؟
\end{enumerate}
\end{problem}
