\chapter{Biosynthèses et cellule intégrée}\label{ch:b1:biosyntheses-integration}

À huit heures du matin, un foie transforme le sucre du petit déjeuner en
\omterm{def:b1:carbohydrates:polysaccharide}{glycogène} et en \omterm{def:b1:lipids:triglyceride}{graisse}~; à trois heures la nuit suivante, le même foie
fabrique du sucre à partir des \omterm{def:b1:proteins:aminoacid}{acides aminés} du muscle et le déverse
dans le sang pour un encéphale qui n'a rien mangé depuis neuf heures.
Rien n'a changé dans les \omterm{def:b1:enzymes:enzyme}{enzymes}~; ce qui a changé, c'est lesquelles
sont allumées. Les chapitres précédents ont démonté les molécules de la
\omterm{def:b1:cell-unit-of-life:cell}{cellule}~; celui-ci les assemble --- synthèses des sucres, du \omterm{def:b1:carbohydrates:polysaccharide}{glycogène},
des \omterm{def:b1:lipids:triglyceride}{graisses} et des \omterm{def:b1:proteins:aminoacid}{acides aminés} --- puis demande comment une \omterm{def:b1:cell-unit-of-life:cell}{cellule},
et un \omterm{def:b1:organism-environment:organism}{organisme}, décident à chaque instant quelles voies tournent. La
réponse tient en un petit jeu de monnaies communes, quelques \omterm{def:b1:enzymes:enzyme}{enzymes} aux
carrefours, et des hormones qui disent à chaque \omterm{def:b1:cell-unit-of-life:cell}{cellule} l'état de
l'ensemble.

\section{Ce qu'exige une synthèse}

\begin{proposition}[Les trois exigences de l'anabolisme]\label{prop:b1:biosyntheses-integration:needs}
\index{anabolisme}
Toute biosynthèse exige des \emph{squelettes carbonés}, prélevés sur une
poignée d'intermédiaires de la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} et du \omterm{def:b1:respiration-fermentation:krebs}{cycle de Krebs}~; du
\emph{pouvoir réducteur}, fourni sous forme de \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}~; et de
l'\emph{énergie}, fournie sous forme d'\omterm{def:b1:water-small-molecules:atp}{ATP}. Catabolisme et anabolisme se
rencontrent donc aux mêmes carrefours --- glucose-6-phosphate, triose
phosphate, pyruvate, \omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA}, $\alpha$-cétoglutarate, oxaloacétate
--- mais empruntent des \omterm{def:b1:enzymes:enzyme}{enzymes} distinctes aux étapes irréversibles, de
sorte que chaque sens peut être commandé indépendamment, et ils
emploient des \omterm{def:b1:enzymes:enzyme}{coenzymes} distincts pour leurs électrons~: le NADH, tenu
oxydé, pour emporter les électrons dans le catabolisme~; le \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}, tenu
réduit, pour les céder dans la synthèse.
\end{proposition}

\begin{definition}[La voie des pentoses phosphates]\label{def:b1:biosyntheses-integration:ppp}
\index{voie des pentoses phosphates}\index{NADPH}
La \emph{voie des pentoses phosphates} du \omterm{def:b1:eukaryotic-cell:organelle}{cytosol} oxyde le
glucose-6-phosphate en ribulose-5-phosphate et en $\mathrm{CO_2}$, en
réduisant deux $\mathrm{NADP^+}$ en NADPH~; sa branche non oxydative
interconvertit ensuite des sucres phosphorylés à cinq, quatre, six et
sept carbones, de sorte que la \omterm{def:b1:cell-unit-of-life:cell}{cellule} peut fabriquer du
ribose-5-phosphate pour les \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} quand il lui faut des pentoses,
du NADPH quand il lui faut du pouvoir réducteur (synthèse des \omterm{def:b1:lipids:triglyceride}{graisses},
défense contre les oxydants), ou les deux, et renvoyer le reste à la
\omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse}. C'est la principale source de NADPH des \omterm{def:b1:cell-unit-of-life:cell}{cellules} animales~;
chez les végétaux, la \omterm{prop:b1:photosynthesis:lightreactions}{phase photochimique} du chloroplaste fournit le
NADPH le jour.
\end{definition}

\section{Fabriquer du glucose~: la néoglucogenèse}

\begin{definition}[Néoglucogenèse]\label{def:b1:biosyntheses-integration:gluconeogenesis}
\index{neoglucogenese@néoglucogenèse}
La \emph{néoglucogenèse} fabrique du glucose à partir de précurseurs non
glucidiques --- lactate, pyruvate, glycérol et squelettes carbonés de la
plupart des \omterm{def:b1:proteins:aminoacid}{acides aminés} --- dans le foie (et un peu dans le rein).
Elle remonte la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} par ses sept étapes réversibles et contourne
les trois irréversibles avec d'autres \omterm{def:b1:enzymes:enzyme}{enzymes}~: le pyruvate est
carboxylé en oxaloacétate (dans la \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrie}, un \omterm{def:b1:water-small-molecules:atp}{ATP}) puis
décarboxylé en phosphoénolpyruvate (un GTP)~; le
fructose-1,6-bisphosphate est hydrolysé en fructose-6-phosphate~; le
glucose-6-phosphate est hydrolysé en glucose libre, qui quitte la
\omterm{def:b1:cell-unit-of-life:cell}{cellule}. Par glucose~: 6 équivalents \omterm{def:b1:water-small-molecules:atp}{ATP} et 2 NADH, contre les 2 \omterm{def:b1:water-small-molecules:atp}{ATP} que
rapporte la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} --- le prix à payer pour remonter une voie qui
descend.
\end{definition}

\begin{proposition}[Ce qui peut et ce qui ne peut pas devenir du glucose]\label{prop:b1:biosyntheses-integration:cannot}
Le lactate, le glycérol et les \omterm{def:b1:proteins:aminoacid}{acides aminés} qui donnent du pyruvate ou
des intermédiaires du \omterm{def:b1:respiration-fermentation:krebs}{cycle de Krebs} (tous sauf la leucine et la lysine)
peuvent être convertis en glucose. Les \omterm{def:b1:lipids:fattyacid}{acides gras}, chez les animaux, ne
le peuvent pas~: leur $\beta$-oxydation donne de l'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA}, dont les
deux carbones entrent dans le cycle et en sortent en deux
$\mathrm{CO_2}$ avant tout gain d'oxaloacétate, de sorte qu'il n'y a
aucune voie nette de l'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} au sucre. Les végétaux, les
champignons et les bactéries possèdent le \emph{cycle du glyoxylate},
raccourci qui saute les deux décarboxylations et permet à une graine
oléagineuse en germination de convertir sa \omterm{def:b1:lipids:triglyceride}{graisse} en sucre pour sa
plantule. Un mammifère qui jeûne doit au contraire fabriquer son glucose
à partir de \omterm{def:b1:proteins:peptide}{protéines}.
\end{proposition}

\begin{example}[Le cycle de Cori]\label{ex:b1:biosyntheses-integration:cori}
Un muscle qui sprinte fermente son \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} en lactate, qui gagne le
sang~; le foie capte le lactate, l'oxyde en pyruvate, en fabrique du
glucose à 6 \omterm{def:b1:water-small-molecules:atp}{ATP} par glucose, et rend le glucose au sang pour que le
muscle s'en resserve. Le muscle a gagné 2 \omterm{def:b1:water-small-molecules:atp}{ATP} par glucose sans oxygène
et a passé une facture de 6 au foie, qui la paie avec de l'oxygène à
loisir~: la «~dette d'oxygène~» d'un sprint se rembourse pour
l'essentiel dans le foie.
\end{example}

\section{Stocker~: glycogène et graisse}

\begin{definition}[Synthèse du glycogène]\label{def:b1:biosyntheses-integration:glycogensynthesis}
Le glucose est stocké en \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} par activation du
glucose-6-phosphate en \emph{UDP-glucose} (un UTP), que la
\emph{\omterm{def:b1:carbohydrates:polysaccharide}{glycogène} synthase} ajoute aux extrémités non réductrices de
l'arbre existant~; une \emph{\omterm{def:b1:enzymes:enzyme}{enzyme} branchante} coupe et raccroche de
courtes chaînes pour faire les branchements $\alpha(1{\to}6)$
(\cref{ch:b1:carbohydrates}). Synthase et phosphorylase sont réglées en
sens opposés par les mêmes signaux~: la phosphorylation qui allume la
phosphorylase éteint la synthase, de sorte que l'arbre n'est jamais bâti
et démonté à la fois.
\end{definition}

\begin{definition}[Synthèse des acides gras]\label{def:b1:biosyntheses-integration:fasynthesis}
\index{synthese des acides gras@synthèse des acides gras}
Les \omterm{def:b1:lipids:fattyacid}{acides gras} sont fabriqués dans le \omterm{def:b1:eukaryotic-cell:organelle}{cytosol} à partir d'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA}
exporté de la \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrie}. L'\emph{\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} carboxylase}, l'étape
réglée, le carboxyle (un \omterm{def:b1:water-small-molecules:atp}{ATP}) en malonyl-CoA~; l'\emph{\omterm{def:b1:lipids:fattyacid}{acide gras}
synthase} ajoute ensuite deux carbones à la fois, pris au malonyl-CoA,
à une chaîne en croissance, en réduisant chaque ajout avec deux \omterm{def:b1:biosyntheses-integration:ppp}{NADPH},
jusqu'à ce que le palmitate (C16) soit libéré~: 8 \omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA}, 7 \omterm{def:b1:water-small-molecules:atp}{ATP} et
14 \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} par palmitate. L'élongation et la désaturation dans le
\omterm{def:b1:eukaryotic-cell:endomembrane}{réticulum endoplasmique} donnent les autres acides~; l'estérification
avec du glycérol-3-phosphate (issu du triose phosphate) donne les
\omterm{def:b1:lipids:triglyceride}{triglycérides}. La voie n'est pas la $\beta$-oxydation à l'envers~:
\omterm{def:b1:enzymes:enzyme}{enzymes} différentes, compartiment différent, malonyl-CoA au lieu
d'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA}, \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} au lieu de $\mathrm{FADH_2}$ et de NADH --- et le
malonyl-CoA bloque le transport des \omterm{def:b1:lipids:fattyacid}{acides gras} dans la \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrie}, de
sorte qu'une \omterm{def:b1:cell-unit-of-life:cell}{cellule} ne fabrique pas de la \omterm{def:b1:lipids:triglyceride}{graisse} et n'en brûle pas en
même temps.
\end{definition}

\begin{example}[De la graisse à partir de sucre]\label{ex:b1:biosyntheses-integration:fatfromsugar}
Un foie qui reçoit plus de glucose qu'il ne peut en stocker en \omterm{def:b1:carbohydrates:polysaccharide}{glycogène}
fabrique de la \omterm{def:b1:lipids:triglyceride}{graisse}~: glucose en pyruvate en \omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} (avec perte
du tiers du carbone en $\mathrm{CO_2}$ et gain d'\omterm{def:b1:water-small-molecules:atp}{ATP}), \omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} en
palmitate aux dépens de cet \omterm{def:b1:water-small-molecules:atp}{ATP} et du \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} de la \omterm{def:b1:biosyntheses-integration:ppp}{voie des pentoses
phosphates}, palmitate en \omterm{def:b1:lipids:triglyceride}{triglycéride} expédié au \omterm{def:b1:body-plans-tissues:tissue}{tissu} adipeux. Quelque
\qty{15}{\%} de l'énergie du sucre est perdu dans la conversion~; le
reste est stocké neuf fois plus compactement que sous forme de \omterm{def:b1:carbohydrates:polysaccharide}{glycogène}
(\cref{ch:b1:lipids}). L'inverse --- \omterm{def:b1:lipids:triglyceride}{graisse} en sucre --- est
impossible.
\end{example}

\section{L'azote~: acides aminés et nucléotides}

\begin{definition}[Assimilation de l'azote, transamination]\label{def:b1:biosyntheses-integration:nitrogen}
\index{transamination}\index{glutamate}
L'ammonium entre dans la matière organique presque uniquement par le
\emph{glutamate}~: la glutamate déshydrogénase l'ajoute à
l'$\alpha$-cétoglutarate, et la glutamine synthétase en ajoute un
second à la \omterm{def:b1:proteins:aminoacid}{chaîne latérale} du glutamate (un \omterm{def:b1:water-small-molecules:atp}{ATP}), ce qui donne la
glutamine. À partir de ces deux donneurs, les \emph{transaminases}
(aminotransférases, à \omterm{def:b1:enzymes:enzyme}{coenzyme} dérivé de la vitamine B$_6$) déplacent le
groupe aminé sur n'importe quel acide $\alpha$-cétonique, bâtissant des
\omterm{def:b1:proteins:aminoacid}{acides aminés} à partir des squelettes carbonés des voies centrales~: la
famille du glutamate depuis l'$\alpha$-cétoglutarate, la famille de
l'aspartate depuis l'oxaloacétate, l'alanine et la sérine depuis le
pyruvate et le 3-phosphoglycérate. Les végétaux et les bactéries les
fabriquent tous les vingt~; les animaux ont perdu les longues voies
menant à neuf d'entre eux, les \omterm{def:b1:proteins:aminoacid}{acides aminés} \emph{indispensables},
qu'ils prennent dans leur nourriture. Les mêmes transaminases, à
l'envers, ôtent l'azote des \omterm{def:b1:proteins:aminoacid}{acides aminés} excédentaires pour l'excréter
en ammoniac (poissons), en urée (mammifères) ou en acide urique
(oiseaux, insectes).
\end{definition}

\begin{example}[Les nucléotides]\label{ex:b1:biosyntheses-integration:nucleotides}
Un cycle purique est assemblé sur du ribose-5-phosphate à partir de
glycine, d'aspartate, de deux glutamines, de deux unités à un carbone et
de $\mathrm{CO_2}$, au prix de six \omterm{def:b1:water-small-molecules:atp}{ATP}~; une \omterm{def:b1:nucleic-acids:nucleotide}{pyrimidine} à partir
d'aspartate et de carbamoyl phosphate. Les désoxynucléotides sont faits
à partir des ribonucléotides par réduction du sucre, avec du \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}. Une
\omterm{def:b1:cell-unit-of-life:cell}{cellule} sur le point de se diviser fabrique quelque $10^{10}$
\omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} en une heure~; les médicaments qui bloquent ces synthèses
--- méthotrexate, 5-fluorouracile --- arrêtent d'abord les \omterm{def:b1:cell-unit-of-life:cell}{cellules} en
division, et c'est pourquoi ils traitent les cancers.
\end{example}

\section{Le mammifère intégré}

\begin{proposition}[Nourri et à jeun]\label{prop:b1:biosyntheses-integration:fedfasting}
\index{insuline}\index{glucagon}
Les \omterm{def:b1:mammal-organization:organ}{organes} se partagent le travail. Après un repas, l'\emph{insuline}
du pancréas dit au foie de stocker le glucose en \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} et de
convertir l'excédent en \omterm{def:b1:lipids:triglyceride}{graisse}, aux muscles de capter le glucose et de
stocker du \omterm{def:b1:carbohydrates:polysaccharide}{glycogène}, au \omterm{def:b1:body-plans-tissues:tissue}{tissu} adipeux de capter la \omterm{def:b1:lipids:triglyceride}{graisse}~; la
glycémie, montée à \qty{8}{mmol/L}, revient à \qty{5}{} en deux heures.
Entre les repas, le \emph{glucagon} dit au foie de démonter son
\omterm{def:b1:carbohydrates:polysaccharide}{glycogène} et, à mesure qu'il s'épuise, de fabriquer du glucose à partir
de lactate, de glycérol et d'\omterm{def:b1:proteins:aminoacid}{acides aminés}, et au \omterm{def:b1:body-plans-tissues:tissue}{tissu} adipeux de
libérer des \omterm{def:b1:lipids:fattyacid}{acides gras}, que les muscles et le foie brûlent à la place
du glucose~; l'encéphale, qui ne sait pas brûler de \omterm{def:b1:lipids:triglyceride}{graisse}, garde ses
\qty{120}{g} de glucose par jour. Dans un jeûne de plusieurs jours, le
foie convertit les \omterm{def:b1:lipids:fattyacid}{acides gras} en \emph{corps cétoniques}, petits acides
que l'encéphale sait employer, et le besoin de glucose --- et de la
\omterm{def:b1:proteins:peptide}{protéine} qui le fabrique --- baisse des deux tiers. Les mécanismes
hormonaux relèvent du volume de deuxième année~; la logique métabolique
est celle de ce chapitre.
\end{proposition}

\begin{method}[Lire un état métabolique]\label{met:b1:biosyntheses-integration:state}
\begin{enumerate}
  \item Demander quel carburant abonde~: une glycémie élevée signifie
        stockage (\omterm{def:b1:carbohydrates:polysaccharide}{glycogène}, puis \omterm{def:b1:lipids:triglyceride}{graisse})~; une glycémie basse signifie
        mobilisation.
  \item Suivre les signaux~: l'insuline active les synthases et inactive
        la phosphorylase et la lipase~; le glucagon fait l'inverse ---
        surtout par phosphorylation des mêmes \omterm{def:b1:enzymes:enzyme}{enzymes}
        (\cref{ch:b1:enzymes}).
  \item Regarder les carrefours~: l'\omterm{def:b1:water-small-molecules:atp}{ATP} et le citrate bloquent la PFK et
        ouvrent la \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{néoglucogenèse}~; l'AMP fait l'inverse~; le
        malonyl-CoA bloque la combustion des \omterm{def:b1:lipids:triglyceride}{graisses} pendant qu'on en
        fabrique~; l'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} active la carboxylase qui amorce la
        \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{néoglucogenèse}.
  \item Répartir les \omterm{def:b1:mammal-organization:organ}{organes}~: le foie (stocke, fabrique et exporte le
        glucose~; fabrique les corps cétoniques et l'urée), le muscle
        (stocke du \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} pour lui-même, exporte lactate et alanine),
        le \omterm{def:b1:body-plans-tissues:tissue}{tissu} adipeux (stocke et libère la \omterm{def:b1:lipids:triglyceride}{graisse}), l'encéphale
        (brûle du glucose, puis des corps cétoniques).
\end{enumerate}
\end{method}

\section{Le végétal intégré}

\begin{proposition}[Source et puits]\label{prop:b1:biosyntheses-integration:sourcesink}
Le jour, les \omterm{def:b1:eukaryotic-cell:plastid}{chloroplastes} d'une \omterm{def:b1:flowering-plant-organization:organs}{feuille} exportent du triose phosphate
vers le \omterm{def:b1:eukaryotic-cell:organelle}{cytosol}, où il est converti en \emph{saccharose} pour être
exporté dans le phloème (\cref{ch:b1:plant-transport}) vers les \omterm{def:b1:flowering-plant-organization:organs}{racines},
les fruits et les extrémités en croissance~; ce que le phloème ne peut
pas emporter est gardé dans le chloroplaste sous forme d'\emph{\omterm{def:b1:carbohydrates:polysaccharide}{amidon}},
réserve transitoire, mobilisée la nuit pour que le saccharose continue
de couler. Là où le foie d'un mammifère arbitre entre \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} et
exportation par des hormones, une \omterm{def:b1:flowering-plant-organization:organs}{feuille} arbitre entre \omterm{def:b1:carbohydrates:polysaccharide}{amidon} et
saccharose par le taux de phosphate du \omterm{def:b1:eukaryotic-cell:organelle}{cytosol}, qui commande le
transporteur de l'enveloppe du chloroplaste. Un végétal se règle par les
concentrations de ses propres métabolites, \omterm{def:b1:mammal-organization:organ}{organe} par \omterm{def:b1:mammal-organization:organ}{organe}~; le
mammifère y ajoute une commande nerveuse et hormonale sur l'ensemble.
\end{proposition}

\begin{omfigure}
\includegraphics[width=0.6\linewidth]{images/book3/ai/b3-16-variegated-leaf-iodine.jpg}

{\small Une \omterm{def:b1:flowering-plant-organization:organs}{feuille} panachée avant et après le test à l'eau iodée~:
l'\omterm{def:b1:carbohydrates:polysaccharide}{amidon} (bleu-noir) n'a été fabriqué que là où il y avait de la
\omterm{def:b1:photosynthesis:pigments}{chlorophylle}. Les régions blanches, qui ont importé du saccharose des
régions vertes, n'ont pas fabriqué d'\omterm{def:b1:carbohydrates:polysaccharide}{amidon}.}
\end{omfigure}

\begin{example}[L'amidon d'une nuit]\label{ex:b1:biosyntheses-integration:nightstarch}
Une \omterm{def:b1:flowering-plant-organization:organs}{feuille} met de côté en \omterm{def:b1:carbohydrates:polysaccharide}{amidon} environ la moitié de ce qu'elle fixe
le jour et le dégrade à un rythme presque constant pendant toute la
nuit, de sorte que la réserve s'épuise juste avant l'aube~: raccourcissez
artificiellement la nuit et il reste de l'\omterm{def:b1:carbohydrates:polysaccharide}{amidon}~; allongez-la et la
plante manque dans les dernières heures. Le rythme est ajusté dès la
première heure d'obscurité à la taille de la réserve et à la durée
attendue de la nuit --- un calcul mené avec des \omterm{def:b1:enzymes:enzyme}{enzymes} et une horloge,
dans une \omterm{def:b1:cell-unit-of-life:cell}{cellule} sans cerveau.
\end{example}

\section{Problème~: vingt-quatre heures d'un humain}

\begin{problem}[{Problème du week-end --- une journée sans petit
déjeuner~: les réserves pesées, l'encéphale nourri, la protéine comptée
et la graisse mobilisée, jusqu'aux heures de jeûne que le glycogène du
foie peut couvrir}]
\label{pb:b1:biosyntheses-integration:1}
Un adulte de \qty{70}{kg} dépense \qty{8.4}{MJ} par jour
(\qty{100}{W} en moyenne) et détient~: \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} hépatique \qty{100}{g},
\omterm{def:b1:carbohydrates:polysaccharide}{glycogène} musculaire \qty{400}{g}, \omterm{def:b1:lipids:triglyceride}{graisse} \qty{12}{kg}, \omterm{def:b1:proteins:peptide}{protéines}
\qty{10}{kg} (dont \qty{6}{kg} dans les muscles). L'encéphale consomme
\qty{120}{g} de glucose par jour (\qty{5}{g/h})~; les autres \omterm{def:b1:body-plans-tissues:tissue}{tissus}
glucodépendants (globules rouges, médulla rénale) \qty{40}{g}. Énergie~:
\omterm{def:b1:carbohydrates:monosaccharide}{glucides} \qty{17}{kJ/g}, \omterm{def:b1:lipids:triglyceride}{graisse} \qty{38}{kJ/g}, \omterm{def:b1:proteins:peptide}{protéines}
\qty{17}{kJ/g}~; \qty{1}{g} de \omterm{def:b1:proteins:peptide}{protéines} rend \qty{0.55}{g} de glucose
par \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{néoglucogenèse}~; \qty{1}{g} de \omterm{def:b1:lipids:triglyceride}{graisse} rend \qty{0.1}{g} de glucose
(par son glycérol).

\medskip
\noindent\textbf{Partie I --- Les réserves.}
\begin{enumerate}
  \item Calculer l'énergie contenue dans chacune des quatre réserves, en
        mégajoules, et le nombre de jours de dépense que chacune
        représente.
  \item Quelle réserve peut fournir du glucose au sang directement~?
        Pourquoi pas le \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} musculaire~?
  \item Calculer le glucose total dont le corps a besoin par jour pour
        ses \omterm{def:b1:body-plans-tissues:tissue}{tissus} glucodépendants.
  \item Calculer le glucose du sang lui-même (\qty{5}{mmol/L},
        \qty{5}{L}, \qty{180}{g/mol}) et le nombre de minutes qu'il
        nourrirait le seul encéphale.
  \item Quelle fraction de la dépense quotidienne représente le glucose
        de l'encéphale~?
  \item Après un repas, le \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} du foie est plein et la glycémie
        monte encore. Nommer le sort suivant du glucose et l'hormone qui
        le commande.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La nuit.}
Le dernier repas s'achève à 20~h~; à partir de là, le foie seul fournit
le glucose du sang.
\begin{enumerate}[resume]
  \item À \qty{160}{g} de glucose par jour, quelle est la demande
        horaire des \omterm{def:b1:body-plans-tissues:tissue}{tissus} glucodépendants~?
  \item Combien de temps \qty{100}{g} de \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} hépatique durent-ils
        à ce rythme, si rien d'autre ne fournit de glucose~?
  \item En réalité, la \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{néoglucogenèse} fournit environ le tiers du
        glucose dès les premières heures. Recalculer la durée du
        \omterm{def:b1:carbohydrates:polysaccharide}{glycogène}.
  \item À quelle heure du matin le \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} du foie s'épuise-t-il selon
        chaque estimation~?
  \item Les muscles, pendant ce temps, brûlent des \omterm{def:b1:lipids:fattyacid}{acides gras}.
        Calculer la \omterm{def:b1:lipids:triglyceride}{graisse} oxydée pendant la nuit (\qty{12}{h}) s'ils
        représentent \qty{60}{\%} de la dépense de repos et ne brûlent
        que de la \omterm{def:b1:lipids:triglyceride}{graisse}.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le lendemain, sans manger.}
\begin{enumerate}[resume]
  \item Le \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} épuisé, les \qty{160}{g} de glucose doivent venir
        de la \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{néoglucogenèse}. Calculer la \omterm{def:b1:proteins:peptide}{protéine} nécessaire par jour,
        et la masse de muscle qu'elle représente.
  \item Calculer l'\omterm{def:b1:water-small-molecules:atp}{ATP} que le foie dépense pour fabriquer \qty{160}{g}
        de glucose à partir de pyruvate (6 \omterm{def:b1:water-small-molecules:atp}{ATP} par glucose) et son
        énergie à \qty{50}{kJ/mol}.
  \item Calculer l'apport du glycérol de la \omterm{def:b1:lipids:triglyceride}{graisse}~: quelle masse de
        \omterm{def:b1:lipids:triglyceride}{graisse} est oxydée par jour si elle couvre toute la dépense non
        glucidique, et combien de glucose son glycérol rend-il~?
  \item Recalculer la \omterm{def:b1:proteins:peptide}{protéine} nécessaire une fois l'apport du glycérol
        retranché.
  \item À ce rythme, combien de jours jusqu'à ce que la moitié du muscle
        ait disparu~?
  \item Dès le troisième jour, le foie fabrique des corps cétoniques et
        l'encéphale en tire les deux tiers de son énergie~; le besoin de
        glucose tombe à \qty{40}{g} pour l'encéphale plus \qty{40}{g}
        pour le reste. Recalculer la perte quotidienne de \omterm{def:b1:proteins:peptide}{protéines} et
        le nombre de jours pour perdre la moitié du muscle.
  \item Calculer le nombre de jours que dure la réserve de \omterm{def:b1:lipids:triglyceride}{graisse} à
        \qty{8.4}{MJ} par jour (elle tombe à \qty{6.5}{MJ} dans un jeûne
        prolongé~: recalculer).
  \item Expliquer pourquoi une personne qui jeûne meurt de la perte de
        ses \omterm{def:b1:proteins:peptide}{protéines} avant que la \omterm{def:b1:lipids:triglyceride}{graisse} ne soit épuisée, et ce que
        les corps cétoniques retardent.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Les vannes.}
\begin{enumerate}[resume]
  \item Nommer l'\omterm{def:b1:enzymes:enzyme}{enzyme} du foie qui doit être active à 3~h et inactive à
        9~h, et celle qui doit faire l'inverse, pour le \omterm{def:b1:carbohydrates:polysaccharide}{glycogène}.
  \item Nommer le signal (l'hormone) qui les règle dans chaque sens, et
        la modification chimique par laquelle il agit.
  \item Une \omterm{def:b1:cell-unit-of-life:cell}{cellule} hépatique à 3~h a beaucoup d'\omterm{def:b1:respiration-fermentation:pdh}{acétyl-CoA} issu de
        l'oxydation des \omterm{def:b1:lipids:triglyceride}{graisses}. Nommer l'\omterm{def:b1:enzymes:enzyme}{enzyme} de la \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{néoglucogenèse}
        que cela active et l'\omterm{def:b1:enzymes:enzyme}{enzyme} de la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse} que l'\omterm{def:b1:water-small-molecules:atp}{ATP} inhibe au
        même moment.
  \item Expliquer pourquoi la \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{néoglucogenèse} et la \omterm{def:b1:respiration-fermentation:glycolysis}{glycolyse}, tournant à
        la fois dans la même \omterm{def:b1:cell-unit-of-life:cell}{cellule}, ne feraient que convertir de l'\omterm{def:b1:water-small-molecules:atp}{ATP}
        en chaleur, et comment les \omterm{def:b1:enzymes:enzyme}{enzymes} distinctes des trois
        contournements l'empêchent.
  \item Un diabétique privé d'insuline a une glycémie élevée et pourtant
        le foie continue de fabriquer du glucose et des corps
        cétoniques. Expliquer ce paradoxe à partir des vannes.
  \item Énoncer le résultat~: les heures de jeûne que couvre le
        \omterm{def:b1:carbohydrates:polysaccharide}{glycogène} du foie (avec et sans \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{néoglucogenèse}), la perte
        quotidienne de \omterm{def:b1:proteins:peptide}{protéines} avant et après le passage aux corps
        cétoniques, et le nombre de jours que dure la réserve de
        \omterm{def:b1:lipids:triglyceride}{graisse}.
\end{enumerate}
\end{problem}
