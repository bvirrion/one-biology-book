\section*{अध्याय \ref{ch:g11:hormones-and-reproduction} --- जनन का हॉर्मोनी नियंत्रण}

\begin{solution}{exo:g11:hormones-and-reproduction:1}
\omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} वह अणु है जो कोई ग्रंथि रक्त में स्रावित करती है और जो उसका ग्राही
ढोती दूर की \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} पर काम करता है। \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} शृंखला के आरंभ की
ग्रंथियों पर अंतिम उत्पाद की क्रिया है: ऋणात्मक तब जब वह आदेश घटा दे (और
किसी स्तर को स्थिर रखे), और धनात्मक तब जब वह उसे बढ़ा दे (और कोई उछाल पैदा
करे)।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:2}
अधश्चेतक (GnRH), पीयूष ग्रंथि (LH और FSH), जननग्रंथि (वृषण में
\omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन}; अंडाशय में एस्ट्रोजन और प्रोजेस्टेरोन)।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:3}
शुक्रजनक नलिकाएँ शुक्राणु बनाती हैं; और उनके बीच की \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}
\omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} स्रावित करती हैं।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:4}
पुटकीय प्रावस्था (दिन 1--13): बढ़ता पुटक, एस्ट्रोजन। अंडोत्सर्ग (दिन 14):
LH का उछाल। पीत प्रावस्था (दिन 15--28): पीत पिंड, प्रोजेस्टेरोन और
एस्ट्रोजन।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:5}
चक्र के अंत में पीत पिंड का क्षीण होना: प्रोजेस्टेरोन और एस्ट्रोजन के
गिरने से गर्भाशय की मोटी हुई परत टूटकर बह जाती है।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:6}
एस्ट्रोजन की चोटी दिन 12, LH की चोटी दिन 13, अंडोत्सर्ग दिन 14, और
प्रोजेस्टेरोन की चोटी दिन 21.
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:7}
जननग्रंथियाँ हटाने से \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} हट जाता है, और उसके साथ अधश्चेतक तथा
पीयूष ग्रंथि पर उसका दबाव भी: इसलिए LH चढ़ जाता है। इंजेक्ट किया गया
\omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} वह दबाव लौटा देता है और LH गिर जाता है --- यानी दोनों दिशाओं
में ऋणात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि}।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:8}
दिन 12 पर एस्ट्रोजन का स्तर ऊँचा और टिका हुआ है, जो \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} को पलटकर
धनात्मक कर देता है और वह उछाल भड़का देता है। और दिन 20 पर एस्ट्रोजन मध्यम
हैं तथा प्रोजेस्टेरोन ऊँचा, और \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} ऋणात्मक है: इसलिए LH दबा रहता
है।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:9}
दिन 21: चक्र की लंबाई चाहे जो हो, पीत प्रावस्था लगभग 14 दिन चलती है, इसलिए
अंडोत्सर्ग अगले मासिक धर्म से 14 दिन पहले होता है; और वह अतिरिक्त हफ़्ता
पुटकीय प्रावस्था में है।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:10}
गोली का ठहरा हुआ, मध्यम एस्ट्रोजन और प्रोजेस्टेरोन जैसा \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} महीने भर
\omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} ऋणात्मक बनाए रखते हैं: FSH किसी पुटक के परिपक्व होने के लिए
बहुत नीचा रहता है, इसलिए एस्ट्रोजन की कोई चोटी नहीं आती, कोई धनात्मक
\omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} नहीं, न LH का उछाल, न अंडोत्सर्ग।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:11}
बाहर से आया \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} उसके अधश्चेतक और पीयूष ग्रंथि को दबा देता है; LH
तथा FSH गिर जाते हैं; और उसके वृषण, जिन्हें अब कोई उकसाता नहीं, स्रावित
करना तथा शुक्राणु बनाना बंद कर देते हैं और सिकुड़ जाते हैं।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:12}
एस्ट्रोजन के दबाव के बिना GnRH और FSH चढ़ जाते हैं; पुटक परिपक्व होते हैं;
और उनकी एस्ट्रोजन की चोटी (जो अब भी जवाब देने लायक़ पीयूष ग्रंथि पर काम
करती है) LH का उछाल तथा अंडोत्सर्ग करा देती है। यह तब इस्तेमाल होती है जब
अंडोत्सर्ग इसलिए चूकता हो कि पुटकों को बहुत कम FSH मिलता है।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:13}
पुटक बढ़ता और स्रावित करता है (यानी अंडाशय तथा FSH काम कर रहे हैं), पर
एस्ट्रोजन की चोटी कोई उछाल पैदा नहीं करती: यानी धनात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि}
अधश्चेतक में या पीयूष ग्रंथि में चूक रही है। चोटी के समय LH जैसे किसी
\omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} का इंजेक्शन अंडोत्सर्ग करा सकता है।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:14}
ज़िंदा रखा गया पीत पिंड प्रोजेस्टेरोन स्रावित करता रहता है: इसलिए वह परत
बनी रहती है (कोई मासिक धर्म नहीं) और ऋणात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} LH तथा FSH को दबाए
रखती है (कोई नया पुटक या अंडोत्सर्ग नहीं)। वह \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} रोपण के कुछ ही दिनों
के भीतर रक्त और मूत्र में उभर आता है, और जाँचें उसी को पकड़ती हैं।
\end{solution}

\begin{solution}{exo:g11:hormones-and-reproduction:15}
$12 \times 0.6 \times 0.4 \approx 2.9$: यानी लगभग 3 भ्रूण। तीन
स्थानांतरणों में कोई जन्म न होने की प्रायिकता: $0.65^3 \approx 0.27$; और
कम से कम एक जन्म की: लगभग 73\%।
\end{solution}

\begin{solution}{pb:g11:hormones-and-reproduction:1}
\textbf{1.} दिन 14: दिन 13 का LH उछाल अंडोत्सर्ग से एक दिन पहले आता है,
और प्रोजेस्टेरोन, जो केवल अंडोत्सर्ग के बाद उभरता है, दिन 14 से चढ़ने लगता
है।

\textbf{2.} पुटकीय प्रावस्था दिन 1--13 (13 दिन); और पीत प्रावस्था दिन
14--27 या 28 (लगभग 14 दिन)।

\textbf{3.} दिन 10 पर बढ़ता पुटक; और दिन 22 पर पीत पिंड, एस्ट्रोजन तथा
प्रोजेस्टेरोन दोनों के लिए।

\textbf{4.} दिन 6: पतली, और एस्ट्रोजन के नीचे मोटी होने लगी। दिन 22:
प्रोजेस्टेरोन के नीचे मोटी तथा स्रावी, किसी भ्रूण के लिए तैयार। दिन 28:
टूटती हुई, बहती हुई, क्योंकि दोनों \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} गिर चुके हैं।

\textbf{5.} आरंभ में पिछले चक्र के \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} गिर चुके होते हैं और उनके साथ
उनका दबाव भी, इसलिए FSH चढ़ जाता है; और पीत प्रावस्था में प्रोजेस्टेरोन तथा
एस्ट्रोजन उसे ज़ोर से दबाते हैं।

\textbf{6.} ऋणात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि}: पुटक के चढ़ते एस्ट्रोजन पीयूष ग्रंथि को
दबाते हैं, और FSH गिर जाता है।

\textbf{7.} धनात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि}: लगभग दो दिन तक अपने अधिकतम पर बने
एस्ट्रोजन अधश्चेतक तथा पीयूष ग्रंथि को दबाने के बजाय उकसाते हैं, और LH
उछल पड़ता है।

\textbf{8.} ऋणात्मक: मध्यम एस्ट्रोजन के साथ प्रोजेस्टेरोन पीयूष ग्रंथि को
दबाता है; और एस्ट्रोजन की किसी टिकी हुई चोटी के बिना न धनात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि}
होती है, न कोई दूसरा उछाल।

\textbf{9.} प्रोजेस्टेरोन और एस्ट्रोजन ढह जाते हैं; वह परत अगले चक्र के
दिन 1 पर टूटकर बह जाती है; और FSH, दबाव से छूटकर, चढ़ जाता है।

\textbf{10.} पीत प्रावस्था के \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोनों} का गिरना ऋणात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} हटा
देता है, FSH चढ़ जाता है, और कोई नया पुटक बढ़ने लगता है: यानी एक चक्र का
अंत ही अगले का संकेत है।

\textbf{11.} महीने भर FSH और LH नीचे (क्योंकि गोली के \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} ऋणात्मक
\omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} बनाए रखते हैं); और कोई पुटक परिपक्व नहीं होता।

\textbf{12.} कोई चोटी नहीं (कोई परिपक्व होता पुटक ही नहीं), कोई उछाल नहीं,
कोई अंडोत्सर्ग नहीं।

\textbf{13.} गोली के \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} गिर जाते हैं और वह पतली परत टूटकर बह जाती
है; यह किसी पीत प्रावस्था का अंत नहीं, बल्कि \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} हटने से हुआ रक्तस्राव
है --- क्योंकि कोई पीत पिंड था ही नहीं।

\textbf{14.} गोली के \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोनों} के बिना वह \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} उठ जाती है, FSH चढ़
जाता है, और कोई पुटक बढ़ने तथा स्रावित करने लगता है; और वह अपनी एस्ट्रोजन
चोटी तक पहुँच जाए तो LH का उछाल और अंडोत्सर्ग आ सकते हैं, तथा पतली परत और
गाढ़ा श्लेष्मा भी अब बने नहीं रहते।

\textbf{15.} अकेला प्रोजेस्टेरोन भी अधश्चेतक तथा पीयूष ग्रंथि को इतना दबा
देता है कि LH का उछाल रुक जाए, और वह ग्रीवा का श्लेष्मा गाढ़ा भी कर देता
है।

\textbf{16.} अधश्चेतक या पीयूष ग्रंथि: अंडाशय को कोई उकसावा मिल ही नहीं
रहा। GnRH के स्पंद (यदि पीयूष ग्रंथि काम करती हो) या FSH तथा LH के
इंजेक्शन उस चूकती मंज़िल को लाँघकर पुटकों की वृद्धि और अंडोत्सर्ग लौटा देते
हैं।

\textbf{17.} पात्र-निषेचन: अंडाणु अंडाशय से इकट्ठे किए जाते हैं, किसी
तश्तरी में निषेचित होते हैं, और भ्रूण गर्भाशय में रख दिया जाता है ---
यानी युग्मकों का मिलना और वह परिवहन, जो अंडवाहिनी करती।

\textbf{18.} प्राकृतिक LH उछाल की; और अंडोत्सर्ग उस उछाल के लगभग 36 घंटे
बाद आता है, इसलिए अंडाणु ठीक उससे पहले इकट्ठे कर लिए जाते हैं जब वे छूटकर
खो जाते।

\textbf{19.} प्रकृति में अगुआ पुटक के चढ़ते एस्ट्रोजन FSH गिरा देते हैं,
और छोटे पुटक, FSH से वंचित होकर, क्षीण हो जाते हैं; केवल सबसे बड़ा बचता
है। इंजेक्ट किया गया FSH इस छँटाई को दरकिनार कर देता है और उन सबको बढ़ता
रखता है।

\textbf{20.} अंडोत्सर्ग दिन 14 पर; वह ऋणात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} जिसने पुटक के
बढ़ने के साथ FSH गिराया, और फिर वह धनात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} जिसने एस्ट्रोजन की
चोटी को LH के उछाल में बदल दिया; और गोली ने महीने भर \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} ऋणात्मक
रखी, इसलिए न कोई पुटक परिपक्व हुआ, न कोई उछाल आया।
\end{solution}
