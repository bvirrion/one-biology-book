\section*{Hoofdstuk \ref{ch:g4:plants-without-seeds} --- Planten zonder zaden}

\begin{solution}{exo:g4:plants-without-seeds:1}
\omterm{prop:g4:plants-without-seeds:damp}{Mossen} en \omterm{prop:g4:plants-without-seeds:damp}{varens}. Ze \omterm{def:g1:plants-around-us:plant}{planten} zich voort met sporen --- microscopische
korreltjes die in enorme aantallen worden losgelaten.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:2}
Een \omterm{def:g4:plants-without-seeds:spore}{spore} is microscopisch en kaal --- geen \omterm{def:g3:animals-through-seasons:active}{voedselvoorraad}, geen
kant-en-klaar plantje; een \omterm{def:g2:from-seed-to-plant:seed}{zaad} draagt allebei, binnen een beschermende
schil. En \omterm{def:g4:plants-without-seeds:spore}{sporen} worden bij miljoenen gemaakt, \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} bij dozijnen.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:3}
Aan de onderkant van de \omterm{def:g1:plants-around-us:parts}{bladeren}: nette rijen bruine stippen, elk een
pakketje met duizenden \omterm{def:g4:plants-without-seeds:spore}{sporen} dat opensplijt als het rijp is en ze als
stof laat wegzweven.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:4}
Mos \omterm{def:g1:plants-around-us:plant}{plant} zich met sporen voort, en \omterm{def:g4:plants-without-seeds:spore}{sporen} en hun tere jonge stadia
hebben vocht nodig --- ze drogen uit en gaan dood in droge lucht. De
schaduwrijke noordmuur blijft vochtig; de zonnige zuidmuur bakt.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:5}
Een nieuwe \omterm{def:g1:plants-around-us:plant}{plant} maken rechtstreeks uit een stuk van de moederplant ---
uitloper, \omterm{ex:g4:plants-without-seeds:bulbtuber}{bol}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{knol}, stek. Slechts één ouder.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:6}
De \omterm{def:g1:plants-around-us:plant}{plant} stuurt een lange dunne \omterm{def:g1:plants-around-us:parts}{stengel} --- de uitloper --- over de
grond; waar hij de grond raakt, wortelt hij en krijgt hij blaadjes, en
wordt hij een volledig plantje; de uitloper verdort daarna en het kind
staat vrij, identiek aan zijn ouder.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:7}
Een gepote aardappel (een \omterm{ex:g4:plants-without-seeds:bulbtuber}{knol}) loopt betrouwbaar uit vanuit zijn
``\omterm{def:g1:the-five-senses:sense}{ogen}'', die elk tot een hele \omterm{def:g1:plants-around-us:plant}{plant} kunnen uitgroeien, identiek aan
het goede ras dat gepoot is --- sneller en zekerder dan \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:8}
Open antwoord. Er verschijnen witte worteltjes aan de ondergedompelde
\omterm{def:g1:plants-around-us:parts}{stengel}, meestal na een of twee weken; de stek is dan klaar om te
\omterm{def:g1:plants-around-us:plant}{planten} --- een volledige nieuwe munt, uit één ouder.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:9}
De \omterm{def:g4:plants-without-seeds:spore}{spore} is kaal en goedkoop: zonder \omterm{def:g3:animals-through-seasons:active}{voedselvoorraad} of plantje gaat
bijna elke \omterm{def:g4:plants-without-seeds:spore}{spore} verloren, dus moeten er miljoenen gemaakt worden. Het
\omterm{def:g2:from-seed-to-plant:seed}{zaad} is een duur overlevingspakket, dus volstaat een stuk of twaalf goed
uitgeruste \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}. Ze echoën de kabeljauw (miljoenen, onverzorgd) en de
olifant (weinig, goed voorzien).
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:10}
De aardbeien kwamen via \omterm{ex:g4:plants-without-seeds:runner}{uitlopers} --- \omterm{def:g4:plants-without-seeds:vegetative}{vegetatieve voortplanting}, één
ouder, exacte kopieën. De bonen kwamen uit \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} die door \omterm{def:g3:flowers-fruits-seeds:parts}{bloemen}
gemaakt zijn en die de bijdragen van twee ouders mengen (\omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel} van
de ene \omterm{def:g3:flowers-fruits-seeds:parts}{bloem}, \omterm{def:g4:animal-reproduction:sexual}{eicellen} in de andere), dus verschilt elke bonenplant een
beetje.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:11}
\omterm{def:g4:plants-without-seeds:vegetative}{Vegetatieve voortplanting} --- de tak is een natuurlijke stek. De
overstroming deed het schaarwerk van de tuinder: ze sneed de scheut af,
voerde hem mee en plantte hem in natte modder, het ideale wortelbed van
de wilg. Rivieren zetten zichzelf vol wilgen door ze in één beweging af
te breken en te \omterm{def:g1:plants-around-us:plant}{planten}.
\end{solution}
