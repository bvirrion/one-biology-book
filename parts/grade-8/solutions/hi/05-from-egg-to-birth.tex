\section*{अध्याय \ref{ch:g8:from-egg-to-birth} --- अंडाणु से जन्म तक}

\begin{solution}{exo:g8:from-egg-to-birth:1}
\omterm{def:g5:flower-to-fruit:steps}{निषेचन} के एक दिन के भीतर बँटना; चौथे दिन तक दर्जनों \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} की एक
शहतूत जैसी गेंद, जो \omterm{def:g8:reproductive-systems:female}{अंडवाहिनी} में नीचे बहती जाती है; और सातवें दिन के
आस-पास \omterm{prop:g5:human-reproduction:plan}{गर्भाशय} की परत में आरोपण।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:2}
ख़ुद \omterm{def:g4:animal-reproduction:sexual}{अंडाणु} के साज़-सामान पर --- यानी वह भंडार-कोठरी, जिसके रूप में उस
बड़े \omterm{def:g8:sexual-reproduction:gametes}{युग्मक} को भरा गया था। आरोपण यह इंतज़ाम ख़त्म कर देता है: किरायेदार
परत की आपूर्ति से जुड़ जाता है, जो जल्द ही \omterm{def:g8:from-egg-to-birth:placenta}{नाल} के रूप में सज जाती है।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:3}
\omterm{prop:g5:human-reproduction:plan}{गर्भाशय} की दीवार में उगी वह जीवन-आधार वाली तश्तरी, जहाँ दोनों \omterm{prop:g4:journey-of-food:absorb}{रक्त} बिना
मिले क़रीब आ जाते हैं। \omterm{def:g5:human-reproduction:pregnancy}{भ्रूण} की ओर: \omterm{def:g7:digestion-and-nutrients:families}{पोषक तत्व} और \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन}। वापस: कचरा और
\omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड}। और कभी पार नहीं होते: ख़ुद दोनों \omterm{prop:g4:journey-of-food:absorb}{रक्त}।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:4}
हफ़्ते 1--8, यानी \omterm{def:g5:human-reproduction:pregnancy}{भ्रूण}: बँटती, जगह बदलती और विशेषज्ञ बनती \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} से
शरीर का नक़्शा बिछाया जाता है। हफ़्ते 9--40, यानी \omterm{def:g5:human-reproduction:pregnancy}{गर्भस्थ शिशु}: नक़्शा
बन चुका --- अब बढ़त और पकाई, \omterm{def:g5:brain-in-command:system}{मस्तिष्क} की जुड़ाई, और फेफड़ों की
रिहर्सल।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:5}
क्योंकि पूरी बनावट पहुँचाई हुई चीज़ों पर चलती है: \omterm{def:g8:from-egg-to-birth:placenta}{नाल} वाली अर्थव्यवस्था
शुरू होते ही किसी पंप को \omterm{def:g5:human-reproduction:pregnancy}{भ्रूण} का \omterm{prop:g4:journey-of-food:absorb}{रक्त} रज्जु तथा शरीर भर चलाना ही पड़ता
है --- आपूर्ति की लाइन बाक़ी नक़्शे का इंतज़ार नहीं कर सकती।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:6}
प्रसव-पीड़ा के बढ़ते हुए \omterm{def:g3:bones-and-muscles:muscle}{संकुचन}; जन्म, आम तौर पर \omterm{def:g1:my-body:parts}{सिर} पहले; अदला-बदली ---
पहली रुलाई, रज्जु का कटना, और नवजात की अपनी सरहदों का ज़िम्मा सँभालना;
और आँवल --- \omterm{def:g8:from-egg-to-birth:placenta}{नाल} का अलग होकर पीछे-पीछे निकल आना।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:7}
फेफड़ों का (\omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} भीतर, \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} बाहर ---
\cref{ch:g7:lungs-and-gas-exchange}), आँत का (\omterm{def:g7:digestion-and-nutrients:families}{पोषक तत्व} भीतर ---
\cref{ch:g7:digestion-and-nutrients}), और \omterm{def:g7:removing-wastes:kidneys}{गुर्दों} का (कचरा बाहर ---
\cref{ch:g7:removing-wastes})। तीन सरहदें, एक तश्तरी।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:8}
क्योंकि नक़्शा बिछाने वाले हफ़्ते ही वे हफ़्ते हैं जब एक भी बिगड़ा हुआ
निर्देश पूरी बनावट में गूँज जाता है --- हर अंग तब विशेषज्ञ बनती
\omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} की एक छोटी-सी भीड़ होता है; और इसलिए भी कि वे हफ़्ते
\omterm{def:g5:human-reproduction:pregnancy}{गर्भावस्था} का शक तक होने से पहले बीत सकते हैं --- सो देखभाल पहली जाँच
से नहीं, पहली संभावना से शुरू होनी चाहिए।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:9}
साँस और रुलाई: फेफड़ों की वह सरहद, जो मिनट भर पुरानी है। \omterm{def:g4:heart-and-blood:heart}{दिल} की धड़कन:
दोनों चक्कर चलाता वह पंप। रंग: पहुँचाने वाली सेवा का सचमुच पहुँचाना ---
यानी \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} से भरपूर \omterm{prop:g4:journey-of-food:absorb}{रक्त} का \omterm{def:g1:the-five-senses:sense}{त्वचा} की \omterm{def:g7:heart-and-circulation:vessels}{केशिकाओं} तक पहुँचना। और हरकत
सेनापति की जाँच पूरी कर देती है। यह अंक-सूची वही फंदा है, जिसकी जाँच
जीते-जागते हो रही है।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:10}
पहले बँटवारों में --- यानी \omterm{prop:g8:sexual-reproduction:core}{युग्मनज} की गेंद का पहले ही दिनों में दो गेंदों
में बँट जाना। उसके बाद दोनों जुड़वाँ उसी एक शुरुआती \omterm{def:g5:first-look-at-cells:cell}{कोशिका} की विरासत से
बनते हैं (\cref{ex:g5:first-look-at-cells:one}), इसलिए दोनों बनावटें
एक ही नक़्शे चलाती हैं: नक़लें, क्योंकि रास्ता बँटवारे के बाद फूटा।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:11}
\omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ}: \omterm{def:g5:human-reproduction:pregnancy}{गर्भस्थ शिशु} वाला चरण --- अंग बन चुके, अब बढ़ रहे हैं; और
\omterm{def:g3:bones-and-muscles:muscle}{पेशी} तथा \omterm{def:g5:brain-in-command:system}{तंत्रिका} की \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} आपस में जुड़ रही हैं। आपूर्ति: \omterm{def:g8:from-egg-to-birth:placenta}{नाल} वाली
अर्थव्यवस्था पूरी रफ़्तार पर --- \omterm{def:g7:digestion-and-nutrients:families}{पोषक तत्व} और \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} भीतर, कचरा बाहर,
और माँ के \omterm{prop:g4:journey-of-food:absorb}{रक्त} में कोई नुक़सानदेह चीज़ नहीं। \omterm{def:g8:puberty:hormones}{हॉर्मोन}: \omterm{prop:g5:human-reproduction:plan}{गर्भाशय} को शांत
और चक्र को ठहरा हुआ थामे हुए। देखभाल: माँ की थाली और आराम, जो दो
सरहदों जितनी आवाजाही को खिलाते हैं; और वे लातें ख़ुद इस चलते हुए फंदे की
तसल्ली देने वाली निशानी हैं।
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:12}
\omterm{def:g8:from-egg-to-birth:placenta}{नाल} \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} की लदाई, \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} का हटाव, पोषक तत्वों की आपूर्ति
और कचरे की सफ़ाई सँभालती थी। पहली रुलाई पर दोनों गैसें \omterm{def:g7:how-animals-breathe:four}{फेफड़े} ले लेते
हैं; आपूर्ति आँत (पहले \omterm{def:g2:animals-grow-up:born}{दूध} से) ले लेती है; और सफ़ाई \omterm{def:g7:removing-wastes:kidneys}{गुर्दे} --- और यह सब
रज्जु कटने के मिनटों के भीतर, यानी सरहदों का वह हस्तांतरण, जो हर अंग को
पहली ही कोशिश में पास करना होता है, और ठीक इसी पर प्रसव-कक्ष की
जाँच-सूची नज़र रखती है।
\end{solution}

\begin{solution}{pb:g8:from-egg-to-birth:1}
\textbf{1.} वे \omterm{def:g5:human-reproduction:pregnancy}{गर्भावस्था} के \omterm{def:g8:puberty:hormones}{हॉर्मोन}, जो आरोपण के साथ शुरू होते हैं ---
यानी जम चुके किरायेदार का रासायनिक इशारा; और सातवें दिन के आस-पास से वे
पढ़े जाने लायक़ मौजूद होते हैं।

\textbf{2.} \omterm{def:g5:human-reproduction:pregnancy}{भ्रूण} वाला चरण बंद होता है: हफ़्ते 1--8 ने पूरा नक़्शा बिछा
दिया --- हर अंग शुरू, और आकार सेम के दाने जितना। अब \omterm{def:g5:human-reproduction:pregnancy}{गर्भस्थ शिशु} वाला
चरण --- यानी बनावट और पकाई --- शुरू होता है।

\textbf{3.} शराब और धुआँ उस मुसाफ़िर तक पार कर जाते हैं, और नक़्शा
बिछाने वाले हफ़्ते बिगड़े निर्देशों के आगे सबसे कमज़ोर होते हैं; फ़ोलिक
अम्ल वह सामान है जिसे सबसे शुरुआती उसारी खींचती है --- और चूँकि वे
हफ़्ते बिना पता चले बीत सकते हैं, इसलिए उस विटामिन का ``शुरुआत से भी
पहले'' वही गणित है जो उन परहेज़ों का है।

\textbf{4.} \omterm{def:g8:from-egg-to-birth:placenta}{नाल} वाली अर्थव्यवस्था को अपना पंप शुरू से चाहिए था:
पहुँचाई जाने वाली चीज़ें बिना धड़कते \omterm{def:g4:heart-and-blood:heart}{दिल} के घूम ही नहीं सकतीं, इसलिए
सबसे पहले वही मशीन चालू होती है --- और बारहवें हफ़्ते के अल्ट्रासाउंड
की वह फड़कन वही है, अपनी दौड़ के आठ हफ़्ते पूरे किए हुए।

\textbf{5.} \omterm{def:g5:human-reproduction:pregnancy}{गर्भस्थ शिशु} वाला चरण: बिछे हुए नक़्शे की बढ़त और पकाई। और
\omterm{def:g8:from-egg-to-birth:placenta}{नाल} की जाँच \omterm{def:g7:how-animals-breathe:four}{फेफड़े} (गैसों की अदला-बदली), आँत (आपूर्ति) तथा \omterm{def:g7:removing-wastes:kidneys}{गुर्दे}
(सफ़ाई) के रूप में होती है --- यानी उस मुसाफ़िर की तीनों सरहदें एक ही
तश्तरी में, और साथ ही उसकी जगह भी, कि वह निकास से हटकर हो।

\textbf{6.} ख़बर-फ़ैसला-हुक्म वाला वह फंदा: \omterm{def:g5:brain-in-command:system}{तंत्रिका} के हुक्म पर सिकुड़ती
\omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} --- यानी सेनापति की जुड़ाई और उसकी मोटरें, \omterm{prop:g5:human-reproduction:plan}{गर्भाशय} की दीवार से
होकर अपना ऐलान करती हुईं।

\textbf{7.} क्योंकि वह दो ग्राहकों की सरहदें चला रहा है: ख़ुद उसके अपने
शरीर की, और --- \omterm{def:g8:from-egg-to-birth:placenta}{नाल} की आवाजाही से होकर --- उस मुसाफ़िर की भी। शर्करा और
दबाव यही पढ़ते हैं कि यह दोगुना बोझ कैसे ढोया जा रहा है।

\textbf{8.} बिना हवा के साँस वाली गतियाँ --- यानी धौंकनी की ख़ाली
प्रैक्टिस, जिसमें \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} और प्रतिवर्त पहली रुलाई के उस अकेले पल के लिए
तैयार होते हैं, जब उस अंग को पहली ही कोशिश में काम करना है।

\textbf{9.} \omterm{prop:g5:human-reproduction:plan}{गर्भाशय} --- यानी शरीर की सबसे ताक़तवर \omterm{def:g3:bones-and-muscles:muscle}{पेशी}; उसके काम:
रास्ता खोलना, और शिशु को नली में नीचे धकेलना; और हुक्म हॉर्मोनों के रूप
में आए, यानी शिशु और माँ के रसायन की आपसी बातचीत से।

\textbf{10.} रुलाई: \omterm{def:g7:how-animals-breathe:four}{फेफड़े} भरे और चालू --- यानी पहली सरहद हस्तांतरित।
रज्जु का बाँधा और काटा जाना: \omterm{def:g8:from-egg-to-birth:placenta}{नाल} की आवाजाही ख़त्म। \omterm{def:g8:from-egg-to-birth:placenta}{नाल} का पूरा निकलना:
वह एक ही इस्तेमाल वाला अंग पूरी तरह अलग --- यानी अदला-बदली बंद, और पीछे
कुछ भी छूटा नहीं।

\textbf{11.} यह कि आपूर्ति-और-हटाव वाला फंदा --- \omterm{def:g7:how-animals-breathe:four}{फेफड़े}, पंप, पहुँचाना,
सेनापति --- अकेले चल रहा है: यानी उस मशीन की पहली जाँच, जिसका नक़्शा इस
किताब ने दो कक्षाएँ लगाकर बनाया।

\textbf{12.} ``चालीस हफ़्तों ने एक ही निषेचित \omterm{def:g4:animal-reproduction:sexual}{अंडाणु} से तीन हज़ार अरब
विशेषज्ञ \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} बना दीं --- और वह भी बँटने, विशेषज्ञ बनने तथा
हॉर्मोनों से चले दफ़्तर के ज़रिए, और एक उल्लेखनीय तश्तरी के पार आपूर्ति
पाकर।''
\end{solution}
