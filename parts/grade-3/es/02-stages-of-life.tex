\chapter{Las etapas de la vida}\label{ch:g3:stages-of-life}

Pon en fila una foto de familia: el bebé en las \omterm{def:g1:my-body:joint}{rodillas} de la abuela,
el escolar al que le faltan dos dientes de delante, el primo
\omterm{def:g3:stages-of-life:stages}{adolescente} de pronto más alto que su madre, los padres, la abuela
misma. Una familia, una tarde --- y el \omterm{def:g2:born-grow-die:cycle}{ciclo de vida} humano entero
sentado en un sofá. Este curso miramos de cerca esas etapas, y también
cómo se puede medir el crecimiento.

\section{Las etapas de una vida humana}

\begin{definition}[Las etapas de la vida humana]\label{def:g3:stages-of-life:stages}
Una vida humana pasa por varias etapas:
\begin{enumerate}
  \item el \emph{bebé}\index{bebé} (más o menos los dos primeros
        años): el crecimiento más rápido de todos, los primeros
        dientes, los primeros pasos, las primeras palabras;
  \item el \emph{niño}\index{niño}: crecimiento constante, \omterm{prop:g2:teeth-and-eating-well:sets}{dientes de
        leche} cambiados por \omterm{prop:g2:teeth-and-eating-well:sets}{dientes definitivos}, aprendizajes de todo
        tipo;
  \item el \emph{adolescente}\index{adolescente}: un nuevo estirón de
        crecimiento rápido, y el cuerpo que pasa de ser de niño a ser
        de adulto;
  \item el \emph{adulto}\index{adulto}: se acabó el crecer en altura;
        los adultos pueden ser padres o madres;
  \item la \emph{vejez}\index{vejez}: el cabello se vuelve blanco, la
        \omterm{def:g1:the-five-senses:sense}{piel} se arruga, la fuerza va bajando poco a poco --- la última
        etapa natural del ciclo.
\end{enumerate}
\end{definition}

\begin{example}[Leer el sofá]\label{ex:g3:stages-of-life:sofa}
En el sofá familiar: el bebé (etapa 1), el escolar mellado (etapa 2
--- el relevo del \cref{ch:g2:teeth-and-eating-well} en plena marcha),
el primo de dieciséis años (etapa 3, en pleno estirón), los padres
(etapa 4), la abuela (etapa 5). Cinco etapas, una familia.
\end{example}

\begin{remark}[Crecer no es constante]\label{rem:g3:stages-of-life:bursts}
El crecimiento tiene su ritmo: muy rápido en el bebé --- la altura
aumenta como la mitad otra vez en el primer año ---, más lento y
constante durante la infancia, otra vez rápido en el \omterm{def:g3:stages-of-life:stages}{adolescente}, y
después se acaba. Dos estirones con un tramo tranquilo en medio: esa
es la forma normal de hacerse mayor.
\end{remark}

\section{Medir el crecimiento}

\begin{definition}[Curva de crecimiento]\label{def:g3:stages-of-life:chart}
Una \emph{curva de crecimiento}\index{curva de crecimiento} es el
registro de la altura (o del peso) medida una y otra vez a lo largo
del tiempo --- las marcas del marco de la puerta del
\cref{ch:g1:my-body}, hechas con regularidad y con su fecha. Los
médicos llevan una de cada niño.
\end{definition}

\begin{example}[Leer la curva]\label{ex:g3:stages-of-life:reading}
En la curva, la línea sube unos $\qty{25}{cm}$ en el primer año --- y
solo unos $\qty{6}{cm}$ al año durante la infancia. Los dos tramos
empinados son los dos estirones de la
\cref{rem:g3:stages-of-life:bursts}; el final plano dice que el \omterm{def:g3:stages-of-life:stages}{adulto}
ha dejado de crecer en altura.
\end{example}

\begin{method}[Llevar tu propia curva de crecimiento]\label{met:g3:stages-of-life:keep}
\begin{enumerate}
  \item mide la altura con cuidado (a la manera de la pared y el libro
        de los cursos anteriores), siempre sin zapatos;
  \item apunta la fecha y la altura en un cuaderno, o marca el marco
        de la puerta y ponle la fecha a la marca;
  \item vuelve a medir cada pocos meses --- importa más la regularidad
        que la frecuencia;
  \item para ver la historia, dibuja los puntos: la edad abajo, la
        altura en el lado, como en la figura.
\end{enumerate}
\end{method}

\section{Los animales y las plantas también tienen etapas}

\begin{proposition}[Etapas en todo el mundo vivo]\label{prop:g3:stages-of-life:animals}
Todos los \omterm{def:g1:living-or-not:living}{seres vivos} pasan por las etapas de su propia versión del
ciclo --- juventud, edad adulta, \omterm{def:g3:stages-of-life:stages}{vejez}. La etapa joven puede
parecerse al \omterm{def:g3:stages-of-life:stages}{adulto} (el gatito, el potro) o no parecerse en nada (el
renacuajo, la \omterm{ex:g2:animals-grow-up:butterfly}{oruga} del \cref{ch:g2:animals-grow-up}); las etapas de
una \omterm{def:g1:plants-around-us:plant}{planta} son \omterm{def:g2:from-seed-to-plant:seed}{semilla}, \omterm{prop:g2:from-seed-to-plant:cycle}{plántula}, \omterm{def:g1:plants-around-us:plant}{planta} joven, adulta en \omterm{def:g1:plants-around-us:parts}{flor}. Lo que
ninguno puede hacer es saltarse una etapa ni recorrerlas al revés.
\end{proposition}
\admitted

\begin{example}[Qué deprisa, qué despacio]\label{ex:g3:stages-of-life:speeds}
Las velocidades son muy distintas. Una mosca es adulta en semanas; un
gato, en un año más o menos; una persona, en casi veinte años; un
roble tarda varias decenas de años en dar sus primeras bellotas.
Despacio o deprisa, el orden de las etapas no cambia nunca.
\end{example}

\begin{remark}[La vejez es una etapa, no un defecto]\label{rem:g3:stages-of-life:oldage}
El cabello blanco y las arrugas no son una enfermedad: son la última
etapa del cuerpo, tan natural como los primeros dientes del bebé. En
muchas especies, y en las familias humanas, los mayores llevan
consigo lo que más necesitan los jóvenes --- la experiencia.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:1}
Nombra las cinco etapas de una vida humana, en orden.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:2}
¿En qué etapa: (a) se caen los \omterm{prop:g2:teeth-and-eating-well:sets}{dientes de leche}? (b) se puede llegar a
ser padre o madre? (c) es el crecimiento el más rápido de toda la vida?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:3}
¿Cuáles son los dos estirones de crecimiento y qué hay entre ellos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:4}
En la \omterm{def:g3:stages-of-life:chart}{curva de crecimiento}, ¿cómo se nota solo con mirar la línea que
un \omterm{def:g3:stages-of-life:stages}{adulto} ha dejado de crecer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:5}
Con la curva, ¿cuántos centímetros sube más o menos la línea entre los
$6$ y los $8$ años? ¿Y en el primer año solo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:6}
¿Por qué hay que medir la altura siempre sin zapatos y de la misma
manera? ¿Qué podría salir mal si no?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:7}
Da las etapas de la vida de un roble, de la bellota al árbol viejo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:8}
Pruébalo: empieza para ti la \omterm{def:g3:stages-of-life:chart}{curva de crecimiento} del
\cref{met:g3:stages-of-life:keep}, con el primer punto de hoy. ¿En qué
etapa de la \cref{def:g3:stages-of-life:stages} estás --- y qué
estirón tienes todavía por delante?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:9}
Una mosca, un gato, una persona y un roble recorren las mismas etapas
a velocidades muy distintas. Ordénalos del más rápido al más lento en
llegar a la edad adulta --- y di qué tienen en común los cuatro
órdenes de etapas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:10}
Tomás, de $9$ años, se mide dos veces en un mes y no encuentra
``ningún crecimiento''. ¿Debería preocuparse? Responde con la
\cref{rem:g3:stages-of-life:bursts} y con la forma que tiene la curva
a su edad.
\end{exercise}
