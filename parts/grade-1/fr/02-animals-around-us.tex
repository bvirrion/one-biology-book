\chapter{Les animaux autour de nous}\label{ch:g1:animals-around-us}

Un merle sautille sur la pelouse, un chat le surveille du haut du
mur, une fourmi transporte sur le pas de la porte une miette plus
grosse qu'elle. Les \omterm{def:g1:animals-around-us:animal}{animaux} sont partout autour de nous --- dans le
jardin, dans le parc, dans la maison. Apprenons à les regarder comme
le fait un biologiste~: leur corps, et leur façon de se déplacer.

\section{Qu'est-ce qu'un animal~?}

\begin{definition}[Animal]\label{def:g1:animals-around-us:animal}
Un \emph{animal}\index{animal} est un \omterm{def:g1:living-or-not:living}{être vivant} qui se nourrit en
mangeant d'autres \omterm{def:g1:living-or-not:living}{êtres vivants} --- des plantes, ou d'autres animaux
--- et qui peut le plus souvent se déplacer tout seul. Un chat, un
merle, un poisson et une fourmi sont des animaux. Nous aussi, nous
sommes des animaux.
\end{definition}

\begin{example}[Les animaux d'un jardin]\label{ex:g1:animals-around-us:garden}
Un seul petit jardin peut abriter beaucoup d'\omterm{def:g1:animals-around-us:animal}{animaux}~: un merle qui
tire un ver de terre du sol, le ver lui-même, un escargot sur une
feuille, une araignée dans sa toile, une abeille dans une fleur et
le chat du voisin. Grands ou minuscules, tous se nourrissent et tous
se déplacent.
\end{example}

\begin{remark}[Petit ne veut pas dire sans importance]\label{rem:g1:animals-around-us:small}
Une fourmi est un \omterm{def:g1:animals-around-us:animal}{animal} tout autant qu'un éléphant. La taille du
corps ne décide pas qui est un \omterm{def:g1:animals-around-us:animal}{animal}~: manger et se déplacer, oui.
Certains des \omterm{def:g1:animals-around-us:animal}{animaux} les plus étonnants qui nous entourent sont les
plus minuscules.
\end{remark}

\section{Les parties du corps d'un animal}

\begin{definition}[Les parties du corps]\label{def:g1:animals-around-us:parts}
La plupart des \omterm{def:g1:animals-around-us:animal}{animaux} ont une \emph{tête}\index{tête} (avec la
bouche et la plupart des organes des sens), un corps, et des
\emph{membres}\index{membre} --- pattes, ailes ou nageoires --- qui
leur servent à se déplacer. Beaucoup ont aussi une
\emph{queue}\index{queue}.
\end{definition}

\begin{example}[Compte les pattes]\label{ex:g1:animals-around-us:legs}
Les pattes viennent en nombres qui valent la peine d'être comptés.
Le merle marche sur $2$ pattes, le chat sur $4$, la coccinelle sur
$6$, l'araignée sur $8$ --- et le ver de terre et l'escargot n'ont
aucune patte, et arrivent pourtant là où ils vont.
\end{example}

\begin{definition}[Le revêtement du corps]\label{def:g1:animals-around-us:covering}
Le corps d'un \omterm{def:g1:animals-around-us:animal}{animal} est protégé par un revêtement~: le
\emph{pelage}\index{pelage} (les poils) du chat, les
\emph{plumes}\index{plume} du merle, les \emph{écailles}\index{écaille}
du poisson, la \emph{coquille}\index{coquille} de l'escargot, et la
\emph{peau}\index{peau} lisse et humide de la grenouille.
\end{definition}

\begin{example}[À chacun son habit]\label{ex:g1:animals-around-us:covering}
Caresse un chat~: du \omterm{def:g1:animals-around-us:covering}{pelage}. Ramasse une \omterm{def:g1:animals-around-us:covering}{plume} de merle tombée~:
légère et douce. Regarde de près un poisson à l'étal~: des rangées
d'\omterm{def:g1:animals-around-us:covering}{écailles} brillantes, comme de petites tuiles. Trois \omterm{def:g1:animals-around-us:animal}{animaux}, trois
habits différents.
\end{example}

\section{Comment les animaux se déplacent}

\begin{proposition}[Les façons de se déplacer]\label{prop:g1:animals-around-us:moving}
Les \omterm{def:g1:animals-around-us:animal}{animaux} se déplacent d'une façon qui va avec leurs \omterm{def:g1:animals-around-us:parts}{membres}~:
\begin{itemize}
  \item les \omterm{def:g1:animals-around-us:animal}{animaux} à pattes \emph{marchent}, courent, sautillent ou
        rampent --- le chat, le merle sur la pelouse, la fourmi~;
  \item les \omterm{def:g1:animals-around-us:animal}{animaux} à ailes \emph{volent}\index{voler} --- le merle,
        l'abeille, le papillon~;
  \item les \omterm{def:g1:animals-around-us:animal}{animaux} à nageoires \emph{nagent}\index{nager} --- le
        poisson~;
  \item les \omterm{def:g1:animals-around-us:animal}{animaux} sans \omterm{def:g1:animals-around-us:parts}{membres} glissent ou se tortillent ---
        l'escargot sur son unique pied mou, le ver de terre qui
        s'allonge et se raccourcit.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Un animal, plusieurs façons]\label{rem:g1:animals-around-us:several}
Beaucoup d'\omterm{def:g1:animals-around-us:animal}{animaux} ont plusieurs façons de se déplacer. Le merle
marche \emph{et} vole~; la grenouille nage \emph{et} saute~; le
canard marche, nage \emph{et} vole. Observe un \omterm{def:g1:animals-around-us:animal}{animal} un moment
avant de décider comment il se déplace.
\end{remark}

\begin{method}[Observer un animal]\label{met:g1:animals-around-us:observe}
Pour décrire un \omterm{def:g1:animals-around-us:animal}{animal} comme un biologiste~:
\begin{enumerate}
  \item regarde les parties~: tête, corps, \omterm{def:g1:animals-around-us:parts}{membres}, queue~;
  \item compte les pattes, sans deviner~;
  \item nomme le revêtement~: \omterm{def:g1:animals-around-us:covering}{pelage}, \omterm{def:g1:animals-around-us:covering}{plumes}, \omterm{def:g1:animals-around-us:covering}{écailles}, \omterm{def:g1:animals-around-us:covering}{coquille},
        \omterm{def:g1:animals-around-us:covering}{peau}~;
  \item observe comment il se déplace~: marcher, sauter, voler,
        nager, glisser~;
  \item dis où tu l'as trouvé~: pelouse, arbre, mare, sous une
        pierre.
\end{enumerate}
Ne fais jamais de mal à l'\omterm{def:g1:animals-around-us:animal}{animal}~: regarde, puis laisse-le repartir.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:1}
Nomme les quatre parties principales du corps du chat de la figure.
Quelle partie porte les yeux et la bouche~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:2}
Combien de pattes~: (a) une poule~? (b) un chien~? (c) une
coccinelle~? (d) une araignée~? (e) un ver de terre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:3}
Donne le revêtement de chaque \omterm{def:g1:animals-around-us:animal}{animal}~: un mouton, un canard, une
truite, un escargot.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:4}
Quels \omterm{def:g1:animals-around-us:animal}{animaux} de ce chapitre savent voler~? Qu'ont-ils tous, à la
place des pattes avant ou des bras~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:5}
L'escargot et le ver de terre n'ont pas de pattes. Comment chacun
se déplace-t-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:6}
Un canard peut se déplacer de trois façons. Nomme-les, et dis quelle
partie de son corps il utilise pour chacune.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:7}
Pourquoi une fourmi est-elle un \omterm{def:g1:animals-around-us:animal}{animal}, d'après
\cref{def:g1:animals-around-us:animal}~? Réponds avec les deux idées
de la définition.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:8}
À toi d'essayer~: observe un moment un \omterm{def:g1:animals-around-us:animal}{animal} dehors --- un oiseau,
un insecte, un escargot. Décris-le avec les cinq étapes de
\cref{met:g1:animals-around-us:observe}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:9}
Une amie dit~: «~Les poissons ne peuvent pas être des \omterm{def:g1:animals-around-us:animal}{animaux} ---
ils ne marchent pas~!~» Que lui réponds-tu~? Utilise
\cref{prop:g1:animals-around-us:moving}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:10}
\omterm{def:g1:animals-around-us:animal}{Animal} mystère~: il a $2$ pattes, $2$ ailes, des \omterm{def:g1:animals-around-us:covering}{plumes}, et il ne
sait pas voler, mais il court très vite. Est-ce un oiseau~?
Pourrait-il être le meilleur coureur de ton quartier~? (Pense à
l'autruche.)
\end{exercise}
