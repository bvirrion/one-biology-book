\chapter{Replikasi DNA}\label{ch:g11:dna-replication}

Setiap hari sumsum tulangmu menghasilkan dua ratus miliar \omterm{def:g10:cells-common-unit:cell}{sel} darah
merah yang baru, lapisan ususmu mengganti dirinya tiap beberapa hari,
dan tiap \omterm{def:g10:cells-common-unit:cell}{sel} baru itu menerima satu salinan lengkap dua meter \omterm{def:g10:universal-dna:information}{DNA} yang
dilukiskan bab sebelumnya sebagai teks \omterm{def:g10:cells-common-unit:cell}{selnya}. Menyalin tiga miliar
huruf, tanpa contoh yang dapat ditengok, dalam beberapa jam, dengan
kesalahan yang lebih sedikit daripada kesalahan seorang juru ketik
mahir dalam satu halaman: bagaimana itu dilakukan? Jawabannya diduga
dari struktur molekulnya sendiri pada 1953, dan dibuktikan lima tahun
kemudian oleh salah satu percobaan paling anggun dalam biologi.

\section{Persoalan penyalinannya}

\begin{definition}[Replikasi]\label{def:g11:dna-replication:replication}
Yang disebut \emph{replikasi DNA}\index{replikasi DNA} adalah proses
yang dengannya sebuah \omterm{def:g10:cells-common-unit:cell}{sel} membuat, dari satu molekul \omterm{def:g10:universal-dna:information}{DNA} berutai ganda,
dua molekul yang serupa dengannya dan satu sama lain, sebelum membelah.
Replikasi berlangsung pada kurun tertentu dalam daur \omterm{def:g10:cells-common-unit:cell}{sel}, yaitu
\emph{fase S} (\cref{ch:g11:cell-cycle-mitosis}), yang selama itu
jumlah \omterm{def:g10:universal-dna:information}{DNA} dalam \omterm{def:g10:cells-common-unit:organelle}{inti selnya} berlipat dua.
\end{definition}

\begin{proposition}[Asas cetakannya]\label{prop:g11:dna-replication:template}
Karena kedua untai molekul \omterm{def:g10:universal-dna:information}{DNA} saling komplementer --- A berhadapan
dengan T, G berhadapan dengan C --- tiap untai membawa seluruh
keterangan yang diperlukan untuk membangun kembali untai yang lain.
Karena itu replikasi berjalan dengan memisahkan kedua untainya dan
memakai masing-masing sebagai \emph{cetakan} yang di atasnya sebuah
untai komplementer baru dirakit, \omterm{def:g10:universal-dna:nucleotide}{nukleotida} demi \omterm{def:g10:universal-dna:nucleotide}{nukleotida}.
\end{proposition}
\admitted

\begin{example}[Tiga hasil yang terbayangkan]\label{ex:g11:dna-replication:models}
Mengingat asas cetakannya, tiga cara membagikan untai lama dan untai
baru diusulkan pada 1950-an:
\begin{itemize}
  \item \emph{semikonservatif}: tiap molekul anaknya menyimpan satu
        untai lama dan memperoleh satu untai baru;
  \item \emph{konservatif}: molekul aslinya tersimpan utuh dan satu
        untai ganda yang sepenuhnya baru dibangun di sampingnya;
  \item \emph{dispersif}: ruas lama dan ruas baru berselang-seling di
        sepanjang tiap untai kedua anaknya.
\end{itemize}
Ketiganya tidak dapat dibedakan dengan melihat; ketiganya dapat
dibedakan dengan menimbang.
\end{example}

\section{Percobaan Meselson--Stahl}

\begin{proposition}[Replikasi bersifat semikonservatif]\label{prop:g11:dna-replication:semiconservative}
Tiap molekul \omterm{def:g10:universal-dna:information}{DNA} anaknya memuat satu untai molekul induknya dan satu
untai yang baru disusun.
\end{proposition}
\begin{proof}[Bukti percobaan]
Meselson dan Stahl (1958) menumbuhkan bakteri selama banyak generasi
dalam medium yang satu-satunya nitrogennya adalah isotop berat
$^{15}$N, sehingga seluruh \omterm{def:g10:universal-dna:information}{DNA-nya} berat. Mereka lalu memindahkan
bakterinya ke medium $^{14}$N biasa dan mencuplik biakannya setelah tiap
generasi (tiap dua puluh menit). \omterm{def:g10:universal-dna:information}{DNA} yang disari dari tiap cuplikan
diputar berjam-jam dalam larutan garam yang pekat, tempat sebuah
molekul mengapung pada aras yang sepadan dengan massa jenisnya sendiri,
lalu dipotret di bawah cahaya ultraviolet: \omterm{def:g10:universal-dna:information}{DNA} berat dan \omterm{def:g10:universal-dna:information}{DNA} ringan
membentuk pita yang berjarak terukur. Hasilnya: sebelum pemindahan,
satu pita berat; setelah satu generasi, satu pita tunggal tepat di
tengah antara berat dan ringan --- setiap molekulnya hibrid; setelah
dua generasi, dua pita yang sama pekatnya, satu hibrid dan satu ringan;
setelah tiga generasi, pita ringannya tiga kali pita hibridnya. Model
konservatif meramalkan satu pita berat dan satu pita ringan pada
generasi pertama, tidak pernah pita hibrid; model dispersif meramalkan
satu pita tunggal yang menghanyut makin ringan tiap generasi, tidak
pernah terbelah menjadi dua. Hanya model semikonservatif yang memberi
satu pita hibrid, lalu hibrid ditambah ringan dalam perbandingan yang
teramati.
\end{proof}

\begin{example}[Meramalkan generasi keempat]\label{ex:g11:dna-replication:fourth}
Setelah $n$ generasi dalam medium ringan, kedua untai berat aslinya
masih ada, satu dalam masing-masing dua molekul hibrid; setiap molekul
lainnya sepenuhnya ringan. Dari $2^n$ molekul, 2 berupa hibrid dan
$2^n - 2$ berupa ringan: pada generasi 4, 2 hibrid berbanding 14 ringan
--- yaitu pita ringan tujuh kali pita hibridnya, dan pita hibridnya
tidak pernah lenyap, hanya memudar. Kedua untai aslinya, pada
kenyataannya, abadi.
\end{example}

\section{Permesinannya}

\begin{proposition}[Garpu replikasi]\label{prop:g11:dna-replication:forks}
Replikasi mulai pada tempat tertentu, yaitu \emph{titik awal}, tempat
kedua untainya ditarik terpisah menjadi sebuah gelembung. Pada tiap
tepi gelembungnya, sebuah \emph{garpu replikasi}\index{garpu replikasi}
maju: sebuah \omterm{def:g10:cell-metabolism:metabolism}{enzim}, yaitu \emph{DNA polimerase}\index{DNA polimerase},
bergerak sepanjang tiap untai yang terbuka dan menambahkan, satu demi
satu, \omterm{def:g10:universal-dna:nucleotide}{nukleotida} yang komplementer dengannya, lalu merantaikannya
menjadi untai yang baru. Kedua garpu sebuah gelembung berjalan dalam
arah berlawanan sampai bertemu garpu gelembung tetangganya; \omterm{def:g10:universal-dna:information}{kromosom}
bakteri, yang berupa satu lingkaran tunggal, punya satu titik awal dan
dua garpu; \omterm{def:g10:universal-dna:information}{kromosom} manusia punya ribuan titik awal yang bekerja
sekaligus.
\end{proposition}
\admitted

\begin{example}[Kecepatan dan skalanya]\label{ex:g11:dna-replication:speed}
Garpu bakteri menambahkan sekitar 1000 \omterm{def:g10:universal-dna:nucleotide}{nukleotida} tiap sekon; garpu
manusia, sekitar 50. \omterm{def:g10:universal-dna:information}{Kromosom} \emph{E.~coli}, yang $\num{4.6e6}$
pasangan \omterm{def:g10:universal-dna:nucleotide}{basa}, jika disalin dua garpu dari satu titik awal, memakan waktu $\num{4.6e6}/(2 \times
1000) \approx \qty{2300}{s}$, yaitu sekitar 40 menit
--- yaitu lama daur \omterm{def:g10:cells-common-unit:cell}{sel} bakterinya dalam medium yang kaya. \omterm{def:g10:cells-common-unit:cell}{Sel} manusia,
dengan $\num{6.4e9}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} untuk disalin dalam fase S sekitar 8
jam, memerlukan ribuan titik awalnya: satu garpu saja akan memakan
lebih dari dua tahun.
\end{example}

\begin{proposition}[Kesetiaan]\label{prop:g11:dna-replication:fidelity}
\omterm{prop:g11:dna-replication:forks}{DNA polimerase} memasangkan \omterm{def:g10:universal-dna:nucleotide}{nukleotida} yang keliru sekitar sekali dalam
$10^5$; ia memeriksa tiap \omterm{def:g10:universal-dna:nucleotide}{nukleotida} sambil menambahkannya dan
menyingkirkan sebagian besar ketidakcocokannya seketika, dan \omterm{def:g10:cell-metabolism:metabolism}{enzim} lain
membetulkan sebagian besar sisanya tepat di belakang garpunya. Laju
kesalahan akhirnya sekitar satu tiap $10^9$ \omterm{def:g10:universal-dna:nucleotide}{nukleotida}: \omterm{def:g10:cells-common-unit:cell}{sel} manusia,
yang menyalin $\num{6.4e9}$ \omterm{def:g10:universal-dna:nucleotide}{nukleotida}, rata-rata membuat sekitar
setengah lusin kesalahan tiap pembelahan. Yang lolos disebut
\emph{mutasi}, yang menjadi pokok \cref{ch:g11:mutations}.
\end{proposition}
\admitted

\begin{method}[Hitungan replikasi]\label{met:g11:dna-replication:calc}
\begin{enumerate}
  \item \emph{Waktu}: panjang yang harus disalin (pasangan \omterm{def:g10:universal-dna:nucleotide}{basa}) dibagi
        kecepatan seluruh garpu yang bekerja (jumlah garpu $\times$
        \omterm{def:g10:universal-dna:nucleotide}{nukleotida} tiap sekon tiap garpu).
  \item \emph{Titik awal yang diperlukan}: panjang seluruhnya dibagi
        apa yang dapat disalin satu titik awal (dua garpu) dalam waktu
        yang tersedia.
  \item \emph{Kesalahan}: \omterm{def:g10:universal-dna:nucleotide}{nukleotida} yang disalin $\times$ laju
        kesalahannya; ingatlah bahwa sebuah \omterm{def:g10:cells-common-unit:cell}{sel} menyalin kedua untainya,
        yaitu dua kali jumlah pasangan \omterm{def:g10:universal-dna:nucleotide}{basanya}.
  \item \emph{Percobaan isotop}: lacaklah kedua untai aslinya; setelah
        $n$ generasi keduanya duduk dalam 2 molekul hibrid dari
        $2^n$ molekul.
\end{enumerate}
\end{method}

\begin{remark}[Mengapa struktur dan mekanismenya cocok]\label{rem:g11:dna-replication:fit}
Watson dan Crick mengakhiri makalahnya pada 1953 dengan mencatat bahwa
pemasangan \omterm{def:g10:universal-dna:nucleotide}{basanya} ``seketika menyarankan sebuah mekanisme penyalinan
yang mungkin''. Meselson dan Stahl menunjukkan bahwa itulah mekanisme
yang sesungguhnya; enzimologi yang menyusulnya menunjukkan caranya.
Inilah salah satu hal yang jarang dalam biologi ketika bentuk sebuah
molekul, yang terlihat sekali saja, menentukan proses yang memakainya.
\end{remark}

\section{Soal: Menimbang Sebuah Molekul untuk Menyaksikannya Menyalin}

\begin{problem}[{Soal akhir pekan --- percobaan Meselson--Stahl yang
disusun ulang: isotopnya, ketiga model beserta ramalannya, pita yang
memutuskan, dan garpu yang menyalin sebuah kromosom dalam empat puluh
menit}]
\label{pb:g11:dna-replication:1}
Nitrogen biasa adalah $^{14}$N; isotop beratnya $^{15}$N adalah 7\%
lebih berat tiap atom. Nitrogen sekitar 16\% massa \omterm{def:g10:universal-dna:information}{DNA}. Bakteri yang
ditumbuhkan selama empat belas generasi dalam medium $^{15}$N
dipindahkan ke medium $^{14}$N pada saat nol dan membelah tiap 20 menit.

\medskip
\noindent\textbf{Bagian I --- Berat dan ringan.}
\begin{enumerate}
  \item Berapa bagian \omterm{def:g10:universal-dna:information}{DNA} yang sepenuhnya berat lebih besar massa
        jenisnya daripada \omterm{def:g10:universal-dna:information}{DNA} ringan? (Anggaplah massa jenisnya
        mengikuti massanya.) Mengapa beda sekecil itu cukup untuk
        memisahkan pitanya?
  \item Mengapa bakterinya ditumbuhkan selama empat belas generasi
        dalam medium berat sebelum dipindahkan, alih-alih satu atau dua
        generasi?
  \item Sebuah molekul hibrid punya satu untai berat dan satu untai
        ringan. Di mana ia mengapung terhadap pita berat dan pita ringannya?
  \item Jelaskan mengapa \omterm{def:g10:universal-dna:information}{DNA-nya} harus disari dari sejumlah besar
        bakteri agar pitanya terlihat.
  \item Bagaimana para pelaku percobaannya dapat yakin bahwa
        pemindahan ke medium ringan tidak dengan sendirinya mengubah
        \omterm{def:g10:universal-dna:information}{DNA} bakterinya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Tiga ramalan.}
Untuk tiap modelnya, sebutkan pitanya (kedudukan dan jumlah nisbi) yang
diharapkan setelah satu dan setelah dua generasi.
\begin{enumerate}[resume]
  \item Model semikonservatif.
  \item Model konservatif.
  \item Model dispersif.
  \item Setelah satu generasi muncul satu pita tunggal di tengah antara
        berat dan ringan. Model mana yang seketika tersingkir, dan dua
        model mana yang bertahan?
  \item Setelah dua generasi ada dua pita, hibrid dan ringan, yang sama
        pekatnya. Model mana yang bertahan, dan mengapa model yang lain
        gagal?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Menghitung untainya.}
\begin{enumerate}[resume]
  \item Setelah $n$ generasi, berapa molekul \omterm{def:g10:universal-dna:information}{DNA} yang berasal dari satu
        molekul asli, dan berapa di antaranya yang memuat untai berat
        yang asli?
  \item Hitung nisbah ringan : hibrid pada generasi 3, 4 dan 6.
  \item Pada generasi keberapa pita hibridnya jatuh di bawah 5\%
        \omterm{def:g10:universal-dna:information}{DNA-nya}?
  \item Para pelaku percobaannya memanaskan \omterm{def:g10:universal-dna:information}{DNA} hibridnya untuk
        memisahkan kedua untainya lalu mensentrifugasi untai
        tunggalnya. Ramalkan pitanya. Model mana yang diuji hal ini?
  \item Jelaskan bagaimana percobaan yang sama, yang dilakukan dalam
        arah terbalik (ringan ke berat), akan membenarkan kesimpulannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Garpunya.}
\omterm{def:g10:universal-dna:information}{Kromosom} bakterinya berupa lingkaran sepanjang $\num{4.6e6}$ pasangan
\omterm{def:g10:universal-dna:nucleotide}{basa} dengan satu titik awal; tiap garpu menambahkan 1000 \omterm{def:g10:universal-dna:nucleotide}{nukleotida} tiap sekon.
\begin{enumerate}[resume]
  \item Hitung waktu yang diperlukan untuk menyalin \omterm{def:g10:universal-dna:information}{kromosomnya} dengan
        kedua garpunya.
  \item Dengan satu garpu saja, berapa lama waktunya? Mengapa satu
        titik awal dengan dua garpu menjadi yang paling sedikit bagi
        sebuah lingkaran?
  \item \omterm{def:g10:cells-common-unit:cell}{Sel} manusia menyalin $\num{6.4e9}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} dalam 8 jam
        dengan garpu 50 \omterm{def:g10:universal-dna:nucleotide}{nukleotida} tiap sekon. Berapa jumlah titik awal
        paling sedikit yang harus menyala, jika semuanya mulai sekaligus?
  \item Pada satu kesalahan tiap $10^9$ \omterm{def:g10:universal-dna:nucleotide}{nukleotida}, berapa kesalahan
        yang dibuat bakterinya tiap replikasi, dan \omterm{def:g10:cells-common-unit:cell}{sel} manusianya?
        Mengapa angka manusianya tidak sebanding lebih buruk bagi
        organismenya?
  \item Nyatakan hasilnya: apa yang dibuktikan pita dua generasinya,
        dan waktu yang diperlukan kedua garpunya untuk menyalin \omterm{def:g10:universal-dna:information}{kromosom}
        bakterinya --- dibandingkan dengan waktu generasi bakterinya.
\end{enumerate}
\end{problem}
