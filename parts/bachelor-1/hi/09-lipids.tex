\chapter{लिपिड}\label{ch:b1:lipids}

मरुस्थल पार करता कोई ऊँट अपने कूबड़ में तीस किलोग्राम \omterm{def:b1:lipids:triglyceride}{वसा} ढोता है और
वहाँ जल बिलकुल नहीं; \qty{4}{\celsius} की किसी पहाड़ी धारा की ट्राउट की
झिल्लियाँ उतनी ही तरल हैं जितनी किसी गरम तालाब की कार्प की; और जल में
हिलाई गई ज़ैतून के \omterm{def:b1:lipids:triglyceride}{तेल} की एक बूँद बूँदों के बादल में बिखर जाती है जो,
छोड़ दिए जाने पर, फिर एक हो जाती हैं। तीनों में \omterm{def:b1:lipids:fattyacid}{लिपिड} काम पर हैं: वह
सबसे सघन रासायनिक ऊर्जा-भंडार जो कोई \omterm{def:b1:organism-environment:organism}{जीव} ढो सकता है, वह झिल्ली जो हर
\omterm{def:b1:cell-unit-of-life:cell}{कोशिका} और हर \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग} की सीमा बनाती है, और अणुओं का वह वर्ग जिसे जल
घोलने से इनकार कर देता है। यह अध्याय \omterm{def:b1:lipids:fattyacid}{वसा अम्लों} और उनसे बनी \omterm{def:b1:lipids:triglyceride}{वसाओं} का, उन
\omterm{def:b1:lipids:amphiphilic}{उभयरागी} \omterm{def:b1:lipids:fattyacid}{लिपिडों} का जो झिल्लियों में जुड़ जाते हैं, इस बात का कि झिल्ली को
अधिक या कम तरल क्या बनाता है, और उन दूसरे \omterm{def:b1:lipids:fattyacid}{लिपिडों} का --- \omterm{def:b1:lipids:amphiphilic}{स्टेरॉल}, वर्णक,
हॉर्मोन --- वर्णन करता है जो कुछ कार्बन वलयों और शृंखलाओं में सजकर बन
सकते हैं।

\section{वसा अम्ल}

\begin{definition}[लिपिड, वसा अम्ल]\label{def:b1:lipids:fattyacid}
\emph{लिपिड}\index{लिपिड} वे जैविक अणु हैं जिनकी परिभाषा किसी साझा
संरचना से नहीं बल्कि किसी साझा गुण से होती है: वे जल में अघुलनशील हैं और
अध्रुवीय विलायकों में घुलनशील। अधिकांश \emph{वसा अम्लों}\index{वसा अम्ल}
पर बने हैं: यानी ऐसे कार्बोक्सिलिक अम्ल जिनकी हाइड्रोकार्बन शृंखला
अशाखित है और \qtyrange{12}{24}{} कार्बनों की, और लगभग सदा सम संख्या में
(वे एक बार में दो कार्बन जोड़कर बनाए जाते हैं,
\cref{ch:b1:biosyntheses-integration})। किसी \emph{संतृप्त}\index{संतृप्त वसा अम्ल} \omterm{def:b1:lipids:triglyceride}{वसा} अम्ल में केवल एकल बंध होते हैं और शृंखला सीधी तथा लचीली
होती है; किसी \emph{असंतृप्त}\index{असंतृप्त वसा अम्ल} में एक या अधिक
द्विबंध होते हैं, \emph{सिस} विन्यास में, और हर एक शृंखला में लगभग
$30^\circ$ का कड़ा मोड़ डाल देता है। संकेतन: C18:1 $\Delta^9$
वह 18-कार्बन अम्ल है जिसमें कार्बन 9 के बाद एक द्विबंध है (ओलेइक अम्ल);
और किसी \emph{ओमेगा-3} अम्ल का अंतिम द्विबंध मेथिल सिरे से तीन कार्बन
दूर होता है।
\end{definition}

\begin{proposition}[गलनांक]\label{prop:b1:lipids:melting}
किसी \omterm{def:b1:lipids:fattyacid}{वसा अम्ल} का गलनांक, और उससे बने किसी भी लिपिड-संयोजन की \omterm{def:b1:lipids:fluidity}{तरलता}, इससे
तय होती है कि शृंखलाएँ कितनी पास-पास जम सकती हैं: वह शृंखला की लंबाई के
साथ बढ़ता है (C12:0 के लिए \qty{44}{\celsius}, C16:0 के लिए
\qty{63}{\celsius}, C18:0 के लिए \qty{70}{\celsius}) और हर \emph{सिस}
द्विबंध के साथ तेज़ी से गिरता है (C18:0 \qty{70}{\celsius},
C18:1 \qty{13}{\celsius}, C18:2 \qty{-5}{\celsius}, C18:3
\qty{-11}{\celsius})। जंतु \omterm{def:b1:lipids:triglyceride}{वसाएँ}, जो संतृप्त अम्लों से समृद्ध हैं, कमरे
के ताप पर ठोस होती हैं; और पादप तथा मछली के \omterm{def:b1:lipids:triglyceride}{तेल}, जो \omterm{def:b1:lipids:fattyacid}{असंतृप्त} अम्लों से
समृद्ध हैं, द्रव।
\end{proposition}
\begin{proof}
सीधी शृंखलाएँ अगल-बगल लेटती हैं और हर $\mathrm{CH_2}$ अपने पड़ोसियों से
\omterm{def:b1:water-small-molecules:bonds}{वान डर वाल्स} संपर्क बनाता है, जिनमें से हर एक कुछ किलोजूल प्रति मोल का
है और जो शृंखला भर जुड़ते जाते हैं: अधिक कार्बन, अधिक संपर्क, और उन्हें
अलग करने को अधिक ऊष्मा। कोई मोड़ हर ओर कई कार्बनों तक निकट संपर्क रोक
देता है और एक साथ अनेक संपर्क हटा देता है।
\end{proof}

\section{वसाएँ: ऊर्जा का भंडार}

\begin{definition}[ट्राइग्लिसराइड]\label{def:b1:lipids:triglyceride}
\emph{ट्राइग्लिसराइड}\index{ट्राइग्लिसराइड} (ट्राइऐसिलग्लिसरॉल, ठोस
होने पर \emph{वसा}\index{वसा}, द्रव होने पर \emph{तेल}\index{तेल})
ग्लिसरॉल है --- यानी तीन कार्बन का कोई ऐल्कोहॉल --- जिसके तीनों
हाइड्रॉक्सिलों का तीन \omterm{def:b1:lipids:fattyacid}{वसा अम्लों} से एस्टरीकरण हो चुका है। उस पर न कोई
आवेश है न कोई \omterm{def:b1:water-small-molecules:water}{ध्रुवीय} समूह बचा है: वह पूरी तरह जलविरागी है, और
\emph{वसाकोशिकाओं}\index{वसाकोशिका} में निर्जल बूँदों के रूप में संचित
रहता है, यानी वसा \omterm{def:b1:body-plans-tissues:tissue}{ऊतक} की उन \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में जो वसा की एक बूँद और एक ओर
ठेले गए केंद्रक से कुछ अधिक नहीं हैं।
\end{definition}

पहुँचाती और इकट्ठा करती हैं।}

\begin{proposition}[वसा सबसे सघन ऊर्जा-भंडार है]\label{prop:b1:lipids:energystore}
\qty{1}{g} \omterm{def:b1:lipids:triglyceride}{वसा} का ऑक्सीकरण \qty{38}{kJ} छोड़ता है, जबकि \qty{1}{g}
कार्बोहाइड्रेट या प्रोटीन \qty{17}{kJ}: किसी \omterm{def:b1:lipids:fattyacid}{वसा अम्ल} के कार्बन अधिक
अपचयित होते हैं (अधिक C--H बंध, कम C--O) और इसलिए ऑक्सीजन को अधिक
इलेक्ट्रॉन देते हैं। इसके अलावा \omterm{def:b1:lipids:triglyceride}{वसा} सूखी संचित होती है, जबकि ग्लाइकोजन
प्रति ग्राम लगभग \qty{3}{g} जल बाँधता है: \qty{1}{g} संचित ग्लाइकोजन
प्रति ग्राम संचित द्रव्यमान \qty{4}{kJ} देता है, और \omterm{def:b1:lipids:triglyceride}{वसा} नौ गुना अधिक। कोई
दुबला मनुष्य \qty{0.5}{kg} ग्लाइकोजन (एक दिन की ऊर्जा) और \qty{12}{kg}
\omterm{def:b1:lipids:triglyceride}{वसा} (एक महीने से अधिक) ढोता है।
\end{proposition}

\begin{example}[सब कुछ वसा के रूप में क्यों न संचित करें]\label{ex:b1:lipids:whyglycogen}
\omterm{def:b1:lipids:triglyceride}{वसा} केवल ऑक्सीजन के साथ, धीरे-धीरे जलाई जा सकती है, और जंतुओं में उसे
वापस ग्लूकोज़ में नहीं बदला जा सकता (\cref{ch:b1:biosyntheses-integration});
मस्तिष्क और लाल \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} को ग्लूकोज़ चाहिए, और दौड़ती पेशी को वह उससे
तेज़ चाहिए जितना \omterm{def:b1:lipids:triglyceride}{वसा} दे सकती है। ग्लाइकोजन घंटों के लिए तेज़,
ऑक्सीजन-मुक्त ग्लूकोज़ भंडार है; और \omterm{def:b1:lipids:triglyceride}{वसा} हफ़्तों के लिए सघन भंडार। कोई
प्रवासी पक्षी, कोई शीतनिद्रा में पड़ा छछूँदर और कोई ऊँट \omterm{def:b1:lipids:triglyceride}{वसा} हैं; किसी
धावक की टाँगें ग्लाइकोजन।
\end{example}

\begin{definition}[मोम]\label{def:b1:lipids:wax}
\emph{मोम}\index{मोम} किसी \omterm{def:b1:lipids:fattyacid}{वसा अम्ल} और किसी लंबी-शृंखला ऐल्कोहॉल के
एस्टर हैं: ठोस, पूरी तरह जलविरागी, और लेप के रूप में काम आते हैं ---
\omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} की \omterm{def:b1:flowering-plant-organization:tissues}{उपचर्म} (\cref{ch:b1:flowering-plant-organization}), कीटों की
सतह, पंखों और रोमों का जलरोधन, और मधुमक्खियों का छत्ता।
\end{definition}

\section{झिल्ली-लिपिड और स्वयं-संयोजन}

\begin{definition}[उभयरागी लिपिड]\label{def:b1:lipids:amphiphilic}
झिल्लियों के \omterm{def:b1:lipids:fattyacid}{लिपिड} \emph{उभयरागी}\index{उभयरागी अणु} होते हैं: एक \omterm{def:b1:water-small-molecules:water}{ध्रुवीय} \emph{सिर} जिसे जल विलेयित करता है और एक या दो
अध्रुवीय \emph{पूँछें} जिन्हें वह बाहर करता है।
\emph{ग्लिसरोफ़ॉस्फ़ोलिपिड}\index{फ़ॉस्फ़ोलिपिड} ग्लिसरॉल हैं जो दो
\omterm{def:b1:lipids:fattyacid}{वसा अम्ल} धारण करते हैं और, तीसरे कार्बन पर, किसी छोटे \omterm{def:b1:water-small-molecules:water}{ध्रुवीय} ऐल्कोहॉल
(कोलीन, एथेनोलामीन, सेरीन, इनोसिटॉल) से जुड़ा एक फ़ॉस्फ़ेट:
फ़ॉस्फ़ेटिडिल-कोलीन जंतु झिल्लियों का सबसे आम \omterm{def:b1:lipids:fattyacid}{लिपिड} है।
\emph{स्फ़िंगोलिपिड}\index{स्फ़िंगोलिपिड}
ग्लिसरॉल के बजाय ऐमीनो-ऐल्कोहॉल स्फ़िंगोसिन पर बने हैं, जिनमें एक \omterm{def:b1:lipids:fattyacid}{वसा
अम्ल} है और सिर फ़ॉस्फ़ोकोलीन (स्फ़िंगोमाइलिन) या शर्कराओं
(ग्लाइकोलिपिड) का, और वे बाहरी पर्ण में तथा तंत्रिका आवरणों में समृद्ध
होते हैं। \emph{स्टेरॉल}\index{स्टेरॉल} --- जंतुओं में \emph{कोलेस्टेरॉल}\index{कोलेस्टेरॉल}
और पौधों तथा कवकों में उससे जुड़े स्टेरॉल --- चार वलयों वाले कड़े अणु
हैं जिनका सिर एक ही हाइड्रॉक्सिल है, और जो दूसरे \omterm{def:b1:lipids:fattyacid}{लिपिडों} की पूँछों के बीच
लेटे रहते हैं।
\end{definition}

\begin{proposition}[स्वयं-संयोजन]\label{prop:b1:lipids:selfassembly}
\index{स्वयं-संयोजन}\index{मिसेल}\index{लिपोसोम}
जल में \omterm{def:b1:lipids:amphiphilic}{उभयरागी अणु} स्वतःस्फूर्त रूप से इस तरह जुड़ते हैं कि अपनी पूँछें
छिपा लें और अपने सिर उघाड़ रखें। संरचना अणु की आकृति तय करती है:
शंकु जैसी आकृति के एकपुच्छी \omterm{def:b1:lipids:fattyacid}{लिपिड} (\omterm{def:b1:lipids:fattyacid}{वसा अम्ल}, अपमार्जक)
\emph{मिसेल} बनाते हैं --- यानी कुछ नैनोमीटर चौड़े गोले जिनके भीतर
पूँछें हैं; और बेलन जैसी आकृति के द्विपुच्छी \omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिड}
\emph{द्विस्तर} बनाते हैं, जो कोई किनारा उघाड़ा न छूटे इसलिए अपने ही ऊपर
मुड़कर \emph{लिपोसोम} (पुटिकाएँ) बन जाते हैं। इस संयोजन को \omterm{prop:b1:water-small-molecules:properties}{जलविरागी
प्रभाव} चलाता है (\cref{ch:b1:water-small-molecules}):
इसमें ऊर्जा नहीं लगती और कोई एंजाइम नहीं चाहिए, और कोई फटा द्विस्तर अपने
आप फिर सील हो जाता है। झिल्लियाँ स्वयं भरने वाली चादरें हैं जो नए \omterm{def:b1:lipids:fattyacid}{लिपिड}
घुसाकर बढ़ती हैं और कभी शून्य से नहीं बनतीं: हर झिल्ली किसी झिल्ली से
आती है।
\end{proposition}
\begin{proof}[साक्ष्य]
जल में फैलाया गया सूखा \omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिड} ऐसी बंद पुटिकाएँ बनाता है जिनकी
द्विस्तरीय भित्तियाँ \omterm{prop:b1:cell-unit-of-life:microscopes}{इलेक्ट्रॉन सूक्ष्मदर्शी} में दिखती हैं (बैंघम, 1965);
आयनों और जल के लिए उनकी पारगम्यता \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} झिल्लियों की पारगम्यता से मेल
खाती है, और उन्हें दवाओं से भरा तथा \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} से जोड़ा जा सकता है। कोई
अपमार्जक (एकपुच्छी, शंकु जैसा) मिलाने पर द्विस्तर घुलकर मिश्रित मिसेल बन
जाता है; और उसे डायलिसिस से हटाने पर द्विस्तर फिर बन जाता है।
\end{proof}

\section{झिल्ली की तरलता}

\begin{definition}[तरलता, प्रावस्था संक्रमण]\label{def:b1:lipids:fluidity}
किसी \omterm{def:b1:membranes-transport:membrane}{लिपिड द्विस्तर} का एक \emph{प्रावस्था संक्रमण ताप}\index{प्रावस्था संक्रमण ताप} $T_m$ होता है: उससे नीचे शृंखलाएँ किसी सुव्यवस्थित,
जेल जैसी अवस्था में जमी रहती हैं जिसमें \omterm{def:b1:lipids:fattyacid}{लिपिड} मुश्किल से चलते हैं; और
उससे ऊपर वे अव्यवस्थित हो जाती हैं और द्विस्तर एक द्विविमीय तरल बन जाता
है जिसमें कोई \omterm{def:b1:lipids:fattyacid}{लिपिड} प्रति सेकंड दस लाख बार अपने पड़ोसी से जगह बदलता है
और झिल्ली के तल में \qty{1}{\micro m/s} से चलता है। \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} अपनी
झिल्लियाँ $T_m$ से ऊपर रखती हैं: \emph{तरलता}\index{झिल्ली तरलता}
प्रोटीनों को विसरित और घूमने देती है, पुटिकाओं को निकलने और जुड़ने, और
द्विस्तर को फिर सील होने।
\end{definition}

\begin{proposition}[तरलता किससे तय होती है]\label{prop:b1:lipids:fluidityfactors}
$T_m$ शृंखलाओं की लंबाई के साथ बढ़ता है और उनकी असंतृप्तता के साथ घटता
है, जैसा मुक्त \omterm{def:b1:lipids:fattyacid}{वसा अम्लों} के लिए होता है; C18:0 शृंखलाओं का कोई द्विस्तर
\qty{55}{\celsius} पर पिघलता है और C18:1 शृंखलाओं का
\qty{-20}{\celsius} पर। पूँछों के बीच घुसा \omterm{def:b1:lipids:amphiphilic}{कोलेस्टेरॉल} दोहरा असर करता
है: $T_m$ से ऊपर उसके कड़े वलय शृंखलाओं की गति रोकते हैं और झिल्ली को
कड़ा करते हैं; और $T_m$ से नीचे वे शृंखलाओं को जमने से रोकते हैं और उसे
जमने नहीं देते। वह संक्रमण को चौड़ा करके एक क्रमिक परिवर्तन बना देता है
और ताप के एक विस्तृत परास में \omterm{def:b1:lipids:fluidity}{तरलता} को लगभग अचर रखता है --- यानी \omterm{def:b1:lipids:fluidity}{तरलता} का
एक \omterm{def:b1:water-small-molecules:buffer}{बफ़र}। जंतुओं की प्लाज़्मा झिल्लियों में हर \omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिड} पर एक तक
\omterm{def:b1:lipids:amphiphilic}{कोलेस्टेरॉल} होता है।
\end{proposition}

\begin{proposition}[समश्यानता अनुकूलन]\label{prop:b1:lipids:homeoviscous}
\index{समश्यानता अनुकूलन}
जो \omterm{def:b1:organism-environment:organism}{जीव} अपना ताप नियंत्रित नहीं कर सकते वे \omterm{def:b1:lipids:fluidity}{तरलता} अचर रखने के लिए अपनी
झिल्लियों का संघटन समायोजित करते हैं: ताप गिरने पर वे संतृप्त की जगह
\omterm{def:b1:lipids:fattyacid}{असंतृप्त वसा अम्ल} और लंबी शृंखलाओं की जगह छोटी शृंखलाएँ रख लेते हैं, और
ताप बढ़ने पर उल्टा।
\end{proposition}
\begin{proof}[साक्ष्य]
\qty{10}{\celsius} पर उगाई गई \emph{E. coli} में \omterm{def:b1:lipids:fattyacid}{असंतृप्त} \omterm{def:b1:lipids:fattyacid}{वसा अम्लों} का
अनुपात \qty{40}{\celsius} पर उगाई गई उसी नस्ल से दुगुना होता है,
और दोनों संवर्धनों की झिल्लियाँ, किसी प्रोब की गतिशीलता से मापने पर,
अपने-अपने वृद्धि-तापों पर बराबर तरल हैं। \qty{5}{\celsius} की अभ्यस्त
ट्राउट अपने झिल्ली-लिपिडों में \qty{20}{\celsius} की ट्राउट से अधिक
बहुअसंतृप्त अम्ल ढोती हैं; और बर्फ़ के पास रहने वाली बारहसिंगे की टाँगों
की \omterm{def:b1:lipids:triglyceride}{वसा} उसके धड़ की \omterm{def:b1:lipids:triglyceride}{वसा} से अधिक \omterm{def:b1:lipids:fattyacid}{असंतृप्त} होती है। पाला सह लेने वाले पौधे
शरद ऋतु में अपनी झिल्लियों को \omterm{def:b1:lipids:fattyacid}{असंतृप्त} \omterm{def:b1:lipids:fattyacid}{लिपिडों} से समृद्ध कर लेते हैं, और
जो उत्परिवर्ती ऐसा नहीं कर पाते वे ठंड पड़ते ही मर जाते हैं।
\end{proof}

\begin{example}[मक्खन, ज़ैतून का तेल, मछली का तेल]\label{ex:b1:lipids:kitchen}
मक्खन (\qty{65}{\%} संतृप्त) रेफ़्रिजरेटर में ठोस और कमरे के ताप पर नरम
रहता है; ज़ैतून का \omterm{def:b1:lipids:triglyceride}{तेल} (\qty{75}{\%} ओलेइक अम्ल) कमरे के ताप पर द्रव
रहता है और रेफ़्रिजरेटर में धुँधला हो जाता है; और मछली का \omterm{def:b1:lipids:triglyceride}{तेल}, जो पाँच
तथा छह द्विबंधों वाले ओमेगा-3 अम्लों से समृद्ध है, \qty{-20}{\celsius}
पर भी द्रव रहता है, और उत्तरी अटलांटिक में किसी कॉड की झिल्लियों को यही
चाहिए।
\end{example}

\section{स्टेरॉल, वर्णक और संकेत}

\begin{definition}[स्टेरॉइड, आइसोप्रीनॉइड]\label{def:b1:lipids:steroids}
\emph{स्टेरॉइड}\index{स्टेरॉइड} \omterm{def:b1:lipids:amphiphilic}{कोलेस्टेरॉल} के चार संगलित वलय साझा
करते हैं: स्वयं \omterm{def:b1:lipids:amphiphilic}{कोलेस्टेरॉल} (झिल्लियाँ, और बाक़ी सबका पूर्ववर्ती), पित्त
लवण, विटामिन D, और स्टेरॉइड हॉर्मोन --- कोर्टिसोल, ऐल्डोस्टेरोन,
लिंग हॉर्मोन --- यानी छोटे, जलविरागी अणु जो झिल्लियाँ स्वतंत्र रूप से
पार करते हैं और \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के भीतर के ग्राहियों पर क्रिया करते हैं।
\emph{आइसोप्रीनॉइड}\index{आइसोप्रीनॉइड}
(टर्पीन) पाँच-कार्बन आइसोप्रीन इकाइयों की शृंखलाएँ और वलय हैं: वे
\emph{कैरोटिनॉइड} जो गाजर को रंग देते हैं और \omterm{def:b1:eukaryotic-cell:plastid}{हरितलवकों} की रक्षा करते हैं
(\cref{ch:b1:photosynthesis}), पर्णहरित की पार्श्व शृंखला,
इलेक्ट्रॉन-अभिगमन शृंखलाओं के क्विनोन, रबर, और वसा-घुलनशील
विटामिन A, E तथा K।
\end{definition}

\begin{proposition}[लिपिड क्या करते हैं]\label{prop:b1:lipids:functions}
ऊर्जा-संचय (\omterm{def:b1:lipids:triglyceride}{ट्राइग्लिसराइड}); झिल्लियाँ (\omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिड}, \omterm{def:b1:lipids:amphiphilic}{स्फ़िंगोलिपिड},
\omterm{def:b1:lipids:amphiphilic}{स्टेरॉल}); कुचालकता और जलरोधन (\omterm{def:b1:lipids:triglyceride}{वसा}, \omterm{def:b1:lipids:wax}{मोम}); प्रकाश का अवशोषण तथा रक्षा
(कैरोटिनॉइड); इलेक्ट्रॉन-वहन (क्विनोन); \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} के बीच संकेतन
(\omterm{def:b1:lipids:steroids}{स्टेरॉइड} हॉर्मोन, प्रोस्टाग्लैंडिन) और उनके भीतर संकेतन
(इनोसिटॉल \omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिड}, डाइऐसिलग्लिसरॉल); विटामिन। एक ही गुण ---
जल में अघुलनशीलता --- इन सबके नीचे है: कोई \omterm{def:b1:lipids:fattyacid}{लिपिड} वहीं रहता है जहाँ उसे
रखा जाए, किसी बूँद, किसी झिल्ली या किसी परत में, या फिर बिना किसी वाहक के
कोई झिल्ली पार कर जाता है।
\end{proposition}

\begin{example}[वह हॉर्मोन जिसे सतह पर ग्राही नहीं चाहिए]\label{ex:b1:lipids:steroidhormone}
अधिवृक्क ग्रंथि से स्रावित कोर्टिसोल रक्त में किसी वाहक प्रोटीन से बँधा
चलता है, किसी लक्ष्य \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} पर उसे छोड़ देता है, सेकंडों में प्लाज़्मा
झिल्ली से घुलकर पार हो जाता है, और \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} के किसी ग्राही से बँधता है
जो फिर केंद्रक में घुसकर जीन चालू कर देता है
(\cref{ch:b1:expression-control})। इंसुलिन जैसा कोई प्रोटीन हॉर्मोन, जो
जल-घुलनशील है और पार करने के लिए बहुत बड़ा, बाहर किसी ग्राही से बँधकर
अपना संदेश भीतर पहुँचवाना पड़ता है। संदेशवाहक का रसायन ही संदेश का
रास्ता तय करता है।
\end{example}

\end{exercise}

\section{समस्या: ठंडा पानी और मरुस्थली वसा}

\begin{problem}[{सप्ताहांत समस्या --- किसी पहाड़ी धारा में एक ट्राउट की
झिल्लियाँ और मरुस्थल में एक ऊँट का कूबड़: असंतृप्तता, संक्रमण ताप,
ऊर्जा-घनत्व और उपापचयी जल, और अंत में वह द्रव्यमान जो ग्लाइकोजन के बजाय
वसा संचित करने से बचता है}]
\label{pb:b1:lipids:1}
भाग I एक ट्राउट के बारे में है जिसके झिल्ली-फ़ॉस्फ़ोलिपिड C16:0, C18:1
और C22:6 (छह द्विबंधों वाला एक ओमेगा-3 अम्ल) की शृंखलाएँ धारण करते हैं।
भाग II \qty{500}{kg} के एक ऊँट के बारे में है जिसका
\omterm{def:b1:lipids:triglyceride}{ट्राइग्लिसराइड} का कूबड़ \qty{30}{kg} का है और जो
\qty{60}{MJ/d} की \omterm{def:b1:organism-environment:allometry}{उपापचय दर} पर मरुस्थल पार कर रहा है। ऊर्जा: \omterm{def:b1:lipids:triglyceride}{वसा}
\qty{38}{kJ/g}, सूखा ग्लाइकोजन \qty{17}{kJ/g}, जो प्रति ग्राम
\qty{3}{g} जल के साथ संचित रहता है। उपापचयी जल: \omterm{def:b1:lipids:triglyceride}{वसा} का
\qty{1.07}{g/g}। ऑक्सीजन: प्रति ग्राम \omterm{def:b1:lipids:triglyceride}{वसा} \qty{2.0}{L}; हवा
\qty{21}{\%} ऑक्सीजन है; ऊँट जितनी ऑक्सीजन साँस में लेता है उसका
\qty{5}{\%} निकालता है; और छोड़ी गई हवा भीतर ली गई मरुस्थली हवा से
\qty{30}{g/m^3} अधिक जल ढोती है।

\medskip
\noindent\textbf{भाग I --- ट्राउट।}
\qty{20}{\celsius} की अभ्यस्त ट्राउट की झिल्ली-शृंखलाएँ
\qty{40}{\%} C16:0, \qty{45}{\%} C18:1 और \qty{15}{\%} C22:6 हैं; और
\qty{5}{\celsius} की ट्राउट की \qty{25}{\%}, \qty{40}{\%} तथा \qty{35}{\%}।
\begin{enumerate}
  \item तीनों अम्लों का संकेतन लिखिए और हर एक में प्रति शृंखला द्विबंध
        गिनिए।
  \item हर ताप पर प्रति शृंखला द्विबंधों की माध्य संख्या निकालिए।
  \item शृंखलाओं के जमाव से समझाइए कि यह बदलाव झिल्ली का संक्रमण ताप
        क्यों घटाता है।
  \item किसी द्विस्तर का $T_m$ औसतन प्रति शृंखला हर अतिरिक्त
        \qty{0.5}{} द्विबंध पर लगभग \qty{20}{\celsius} गिरता है। दोनों
        ट्राउटों के बीच $T_m$ का खिसकाव आँकिए।
  \item गरम पानी की ट्राउट की झिल्ली \qty{20}{\celsius} पर तरल है पर
        \qty{5}{\celsius} पर जेल बन जाती। किसी अचानक ठंडी रात में उसके
        अभिगमन प्रोटीनों, उसके पंपों और उसके तंत्रिका-चालन का क्या होता?
  \item यह अनुकूलन दिनों में होता है। बताइए कि ट्राउट को किन एंजाइमों की
        मात्रा बढ़ानी पड़ती है, और वे \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में कहाँ काम करते हैं (\cref{ch:b1:eukaryotic-cell})।
  \item दोनों ट्राउटों में \omterm{def:b1:lipids:amphiphilic}{कोलेस्टेरॉल} हर दो \omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिडों} पर एक अणु
        है। समझाइए कि वह हर स्थिति में क्या देता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ऊँट की ऊर्जा।}
\begin{enumerate}[resume]
  \item कूबड़ में संचित ऊर्जा निकालिए।
  \item वह यात्रा के कितने दिनों के लिए पर्याप्त है?
  \item उतनी ही ऊर्जा रखने वाले सूखे ग्लाइकोजन का द्रव्यमान निकालिए,
        और जलयोजित ग्लाइकोजन का द्रव्यमान भी।
  \item ग्लाइकोजन के बजाय \omterm{def:b1:lipids:triglyceride}{वसा} संचित करने से बचा द्रव्यमान निकालिए, और
        उसे ऊँट के शरीर-द्रव्यमान के भिन्न के रूप में लिखिए।
  \item \omterm{def:b1:lipids:triglyceride}{वसा} त्वचा के नीचे फैलाकर नहीं, बल्कि किसी कूबड़ में संचित होती
        है। \cref{ch:b1:mammal-organization} के \omterm{prop:b1:mammal-organization:heatbudget}{ऊष्मा-बजट} से इसका कोई
        कारण सुझाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- क्या कूबड़ जल का भंडार है?}
\begin{enumerate}[resume]
  \item पूरे कूबड़ के ऑक्सीकरण से बना उपापचयी जल निकालिए।
  \item उसके लिए चाहिए ऑक्सीजन लीटर में निकालिए।
  \item उसे पाने के लिए ऊँट को जितनी हवा संवातित करनी पड़ेगी उसका आयतन
        निकालिए।
  \item वह हवा छोड़ने में खोया जल निकालिए।
  \item कमाए गए जल की तुलना खोए जल से कीजिए, और निष्कर्ष निकालिए।
  \item पसीना बहाने के बजाय ऊँट दिन में अपने शरीर का ताप
        \qty{6}{\celsius} बढ़ने देता है और रात में उसे ठंडा कर लेता है।
        इस तरह वह जितनी ऊष्मा संचित करता है वह निकालिए (ऊष्माधारिता
        \qty{3.5}{kJ.kg^{-1}.K^{-1}}) और वह जल भी जो उतनी ही ऊष्मा
        वाष्पन से हटाने में लगता (\qty{2.4}{kJ/g})।
  \item कोई ऊँट अपने शरीर के जल का \qty{25}{\%} खोकर भी जी सकता है, और
        जब वह दस मिनट में \qty{100}{L} पानी पी लेता है तब उसकी लाल
        \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} फटे बिना अपने आयतन की दुगुनी हो जाती हैं। दूसरे गुण को
        \cref{ch:b1:membranes-transport} की झिल्लियों से जोड़िए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- द्विस्तर का अंकगणित।}
कोई \omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिड} सिर \qty{0.65}{nm^2} घेरता है; कोई द्विस्तर
\qty{5}{nm} मोटा है; ऊँट की लाल \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} की सतह
\qty{80}{\micro m^2} है और वे प्रति लीटर रक्त $\num{8e12}$ हैं।
\begin{enumerate}[resume]
  \item एक लाल-कोशिका झिल्ली में \omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिडों} की संख्या निकालिए।
  \item \qty{40}{L} रक्त की सारी लाल \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में उनकी संख्या निकालिए।
  \item \qty{750}{g/mol} पर उस \omterm{def:b1:lipids:fattyacid}{लिपिड} का द्रव्यमान ग्राम में निकालिए।
  \item कोई \omterm{def:b1:lipids:amphiphilic}{फ़ॉस्फ़ोलिपिड} स्वतःस्फूर्त रूप से एक पर्ण से दूसरे में
        हफ़्ते में लगभग एक बार पलटता है, पर पार्श्व दिशा में सेकंड भर
        में \qty{1}{\micro m} विसरित हो जाता है। \omterm{prop:b1:water-small-molecules:properties}{जलविरागी प्रभाव} से
        दोनों संख्याएँ समझाइए।
  \item समझाइए कि पलटने की इस दर को देखते हुए किसी झिल्ली के दोनों पर्ण
        \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के पूरे जीवन भर संघटन में भिन्न कैसे बने रह सकते हैं।
  \item परिणाम बताइए: उतनी ही ऊर्जा के ग्लाइकोजन के बजाय \qty{30}{kg}
        \omterm{def:b1:lipids:triglyceride}{वसा} ढोकर ऊँट जो द्रव्यमान बचाता है, और जल-भंडार के रूप में
        कूबड़ पर निर्णय।
\end{enumerate}
\end{problem}
