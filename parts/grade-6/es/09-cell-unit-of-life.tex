\chapter{La célula, unidad de la vida}\label{ch:g6:cell-unit-of-life}

El \cref{ch:g5:first-look-at-cells} terminaba con una promesa: las
celditas no están vacías. Este curso vuelve a salir el \omterm{def:g5:first-look-at-cells:microscope}{microscopio} del
colegio --- con mejores preparaciones y preguntas más afiladas. ¿Qué
hay exactamente dentro de una \omterm{def:g5:first-look-at-cells:cell}{célula}? ¿Se diferencian las \omterm{def:g5:first-look-at-cells:cell}{células}
vegetales de las \omterm{def:g1:animals-around-us:animal}{animales}? ¿Y es una \omterm{def:g5:first-look-at-cells:cell}{célula} una \emph{parte} de un \omterm{def:g6:exploring-our-environment:components}{ser
vivo}, o puede una sola \omterm{def:g5:first-look-at-cells:cell}{célula} ser un \omterm{def:g6:exploring-our-environment:components}{ser vivo} entero? La unidad más
pequeña de la \omterm{def:g1:living-or-not:living}{biología} merece un capítulo a fondo.

\section{Por dentro de la célula}

\begin{definition}[Las partes de la célula]\label{def:g6:cell-unit-of-life:parts}
Bien mirada, toda \omterm{def:g5:first-look-at-cells:cell}{célula} enseña el mismo equipo básico:
\begin{enumerate}
  \item la \emph{membrana}\index{membrana}, la envoltura exterior fina
        de la \omterm{def:g5:first-look-at-cells:cell}{célula}, que la mantiene unida y controla lo que entra y
        lo que sale;
  \item el \emph{citoplasma}\index{citoplasma}, el relleno gelatinoso
        donde se hace el trabajo de la \omterm{def:g5:first-look-at-cells:cell}{célula};
  \item el \emph{núcleo}\index{núcleo} (el punto oscuro de la
        \cref{rem:g5:first-look-at-cells:nucleus}), el centro de mando
        que dirige la vida de la \omterm{def:g5:first-look-at-cells:cell}{célula} y su división.
\end{enumerate}
Membrana, citoplasma y núcleo: el equipo de tres piezas de casi todas
las \omterm{def:g5:first-look-at-cells:cell}{células} de todas las \omterm{def:g1:plants-around-us:plant}{plantas}, todos los \omterm{def:g1:animals-around-us:animal}{animales} y todos los \omterm{def:g6:decomposers-and-soil:decomposers}{hongos}.
\end{definition}

\begin{proposition}[Las células vegetales llevan extras]\label{prop:g6:cell-unit-of-life:plantextras}
Las \omterm{def:g5:first-look-at-cells:cell}{células} vegetales añaden equipo propio:
\begin{itemize}
  \item una \emph{pared celular}\index{pared celular} rígida por fuera
        de la \omterm{def:g6:cell-unit-of-life:parts}{membrana} --- las paredes que hacen el enladrillado de la
        \omterm{def:g1:the-five-senses:sense}{piel} de cebolla (\cref{ex:g5:first-look-at-cells:classics}) y
        que les dan a las \omterm{def:g1:plants-around-us:plant}{plantas} su rigidez para sostenerse sin
        \omterm{def:g3:bones-and-muscles:skeleton}{huesos};
  \item unos granos verdes de la sustancia que atrapa la \omterm{def:g6:exploring-our-environment:conditions}{luz}, en el
        \omterm{def:g6:cell-unit-of-life:parts}{citoplasma} --- los talleres de las \omterm{def:g5:food-webs:roles}{productoras}
        (\cref{prop:g6:plants-are-producers:produce}), ausentes de
        todas las \omterm{def:g5:first-look-at-cells:cell}{células} \omterm{def:g1:animals-around-us:animal}{animales};
  \item a menudo una bolsa grande llena de agua cuya presión mantiene
        turgente la \omterm{def:g5:first-look-at-cells:cell}{célula} --- marchitarse
        (\cref{ex:g1:plants-around-us:wilt}) es que esas bolsas se
        queden flojas.
\end{itemize}
\end{proposition}
\admitted

\section{Vidas enteras en una sola célula}

\begin{definition}[Seres vivos unicelulares]\label{def:g6:cell-unit-of-life:unicellular}
Un \omterm{def:g6:exploring-our-environment:components}{ser vivo} \emph{unicelular}\index{unicelular} es un organismo
completo de exactamente una \omterm{def:g5:first-look-at-cells:cell}{célula}
(\cref{rem:g5:first-look-at-cells:onecelled}). Dentro de una sola
\omterm{def:g6:cell-unit-of-life:parts}{membrana} se alimenta, se mueve, percibe y se reproduce --- dividiéndose
en dos (la división de la \cref{prop:g5:first-look-at-cells:growth},
como historia de vida completa). La \omterm{ex:g6:producing-our-food:bread}{levadura} del
\cref{ex:g6:producing-our-food:bread} es uno; también lo son los
nadadores de cualquier gota de charca y casi todos los \omterm{def:g6:producing-our-food:fermentation}{microbios} del
\omterm{def:g6:decomposers-and-soil:soil}{suelo}, del yogur y de las manos.
\end{definition}

\begin{example}[Una gota de agua de charca, recorrida]\label{ex:g6:cell-unit-of-life:drop}
Una gota de la pared verde del acuario, bajo la lente: una \omterm{def:g5:first-look-at-cells:cell}{célula} con
forma de zapatilla que pasa remando con pelillos que baten, más rápida
que ninguna otra cosa a la vista; \omterm{def:g5:first-look-at-cells:cell}{células} verdes sueltas ---
\omterm{def:g5:food-webs:roles}{productoras} nadadoras, que atrapan la \omterm{def:g6:exploring-our-environment:conditions}{luz} como cualquier \omterm{def:g1:plants-around-us:parts}{hoja}; una
mancha que rodea su comida y se la traga entera. Cada una se alimenta,
se mueve, esquiva --- la vida completa de un \omterm{def:g1:animals-around-us:animal}{animal} o de una \omterm{def:g1:plants-around-us:plant}{planta},
metida en el presupuesto de una sola \omterm{def:g5:first-look-at-cells:cell}{célula}.
\end{example}

\begin{remark}[Una célula o muchas: dos maneras de estar vivo]\label{rem:g6:cell-unit-of-life:twoways}
Un organismo \omterm{def:g6:cell-unit-of-life:unicellular}{unicelular} lo hace todo con una \omterm{def:g5:first-look-at-cells:cell}{célula}; tu cuerpo lo hace
todo con decenas de miles de millones de millones,
\emph{especializadas} --- \omterm{def:g5:first-look-at-cells:cell}{células} de mejilla para revestir, \omterm{def:g5:first-look-at-cells:cell}{células}
musculares que se contraen
(\cref{def:g3:bones-and-muscles:muscle}), \omterm{def:g5:first-look-at-cells:cell}{células} nerviosas que llevan
los mensajes de la \cref{prop:g5:brain-in-command:loop}. Muchas \omterm{def:g5:first-look-at-cells:cell}{células}
permiten repartirse el trabajo; una \omterm{def:g5:first-look-at-cells:cell}{célula} significa todos los oficios
en una sola habitación. Los dos planes funcionan: los \omterm{def:g6:cell-unit-of-life:unicellular}{unicelulares} son
los \omterm{def:g6:exploring-our-environment:components}{seres vivos} más numerosos de la Tierra.
\end{remark}

\section{La teoría celular}

\begin{proposition}[La teoría celular]\label{prop:g6:cell-unit-of-life:theory}
Dos frases, con siglos de elaboración, resumen este capítulo y el
\cref{ch:g5:first-look-at-cells}:
\begin{enumerate}
  \item todos los \omterm{def:g6:exploring-our-environment:components}{seres vivos} están hechos de una o más \omterm{def:g5:first-look-at-cells:cell}{células}, las
        unidades más pequeñas de la vida;
  \item todas las \omterm{def:g5:first-look-at-cells:cell}{células} proceden de la división de una \omterm{def:g5:first-look-at-cells:cell}{célula} que ya
        existía.
\end{enumerate}
No se ha encontrado nunca ninguna excepción --- ni en el mar más
profundo ni en la charca de roca más antigua. La segunda frase tiene
una consecuencia enorme: todas las \omterm{def:g5:first-look-at-cells:cell}{células} vivas de hoy continúan una
cadena ininterrumpida de divisiones que llega hasta la vida más
temprana --- el \omterm{prop:g6:classification-kinship:kinship}{parentesco}
(\cref{prop:g6:classification-kinship:kinship}) escrito a escala de
\omterm{def:g5:first-look-at-cells:cell}{célula}.
\end{proposition}
\admitted

\begin{method}[Leer cualquier organismo a escala celular]\label{met:g6:cell-unit-of-life:read}
Ante cualquier \omterm{def:g6:exploring-our-environment:components}{ser vivo}:
\begin{enumerate}
  \item pregunta primero: ¿una \omterm{def:g5:first-look-at-cells:cell}{célula} o muchas?
  \item si son muchas: espera clases especializadas --- de revestir, de
        transportar, de \omterm{def:g3:bones-and-muscles:muscle}{contraerse}, de mandar;
  \item ¿vegetal o \omterm{def:g1:animals-around-us:animal}{animal}? busca paredes y granos verdes al \omterm{def:g5:first-look-at-cells:microscope}{microscopio};
  \item cualquier crecimiento, reparación o reproducción que observes:
        tradúcelo a divisiones de \omterm{def:g5:first-look-at-cells:cell}{células} --- la frase 2 garantiza que
        la traducción existe.
\end{enumerate}
\end{method}

\begin{example}[La teoría en marcha dentro de este libro]\label{ex:g6:cell-unit-of-life:atwork}
La frase 1 avaló la \omterm{prop:g5:first-look-at-cells:principle}{unidad de la vida}
(\cref{prop:g5:first-look-at-cells:principle}); la frase 2 avaló el
crecimiento por división (\cref{ex:g5:first-look-at-cells:one}), la
\omterm{ex:g6:producing-our-food:bread}{levadura} que dobla la masa, los \omterm{def:g6:decomposers-and-soil:decomposers}{descomponedores} que se multiplican en
el montón caliente y las plantillas de \omterm{def:g6:producing-our-food:fermentation}{microbios} del
\cref{ch:g6:producing-our-food}. Una teoría pequeñita que sostiene una
buena parte de la \omterm{def:g1:living-or-not:living}{biología} de este curso.
\end{example}

\section{Problema: La tarde de microscopía}

\begin{problem}[{Problema de fin de semana --- cuatro preparaciones,
una teoría}]\label{pb:g6:cell-unit-of-life:1}
La tarde de microscopía de la clase: cuatro puestos y cuatro
preparaciones --- \omterm{def:g1:the-five-senses:sense}{piel} de cebolla (A), raspado de mejilla (B), una gota
de agua de charca (C) y una muestra de yogur (D). Cada equipo recorre
los cuatro y va guardando sus dibujos.

\medskip\noindent\textbf{Parte I --- Los puestos A y B.}
\begin{enumerate}
  \item En el puesto A el equipo dibuja ``ladrillos'' de paredes
        rectas, cada uno con un punto oscuro apretado a un lado.
        Nombra la pared, el punto y al culpable del apretujón.
  \item En el puesto B: \omterm{def:g5:first-look-at-cells:cell}{células} redondeadas, sin paredes y con los
        puntos en el centro. Enumera el equipo que tienen las \omterm{def:g5:first-look-at-cells:cell}{células}
        de A y que les falta a las de B --- y el equipo que comparten
        las dos.
  \item Un compañero pregunta por qué las \omterm{def:g5:first-look-at-cells:cell}{células} de cebolla, de un
        \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbo} criado bajo tierra, no enseñan granos verdes. Responde
        con la tarea de los granos y con la dirección del \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbo}.
  \item ¿De quién son las \omterm{def:g5:first-look-at-cells:cell}{células} de la preparación B? Di qué implica
        eso según la frase 2 de la teoría celular: ¿de dónde salió cada
        una de esas \omterm{def:g5:first-look-at-cells:cell}{células}?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Los puestos C y D.}
\begin{enumerate}[resume]
  \item Puesto C: la remera con forma de zapatilla, los nadadores
        verdes, la mancha que fluye. Nombra de cada uno las actividades
        vitales que se ven en un minuto de observación --- y concluye
        qué es cada uno.
  \item Los nadadores verdes son \omterm{def:g5:first-look-at-cells:cell}{células} sueltas y aun así son
        \omterm{def:g5:food-webs:roles}{productoras}. ¿Qué pieza del equipo tiene que enseñar el dibujo
        para demostrarlo?
  \item La muestra de yogur del puesto D enseña \omterm{def:g5:first-look-at-cells:cell}{células} diminutas por
        parejas y en cadenas, en cantidades enormes. Relaciónalas con
        el \cref{ex:g6:producing-our-food:yogurt}: ¿qué le hicieron
        esas \omterm{def:g5:first-look-at-cells:cell}{células} a la \omterm{def:g2:animals-grow-up:born}{leche} y qué son, en el vocabulario de este
        capítulo?
  \item Un equipo afirma que la preparación D no enseña ``ningún \omterm{def:g6:exploring-our-environment:components}{ser
        vivo}, solo motas''. ¿Qué única observación --- con paciencia o
        con una preparación templada --- zanjaría si las motas están
        vivas, según la frase 2 de la teoría celular?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La redacción.}
\begin{enumerate}[resume]
  \item La redacción tiene que clasificar las cuatro preparaciones
        según: ¿una \omterm{def:g5:first-look-at-cells:cell}{célula} o muchas? ¿vegetal, \omterm{def:g1:animals-around-us:animal}{animal} o ninguna de las
        dos? Hazlo.
  \item Todos los dibujos de la tarde enseñan el mismo equipo de tres
        piezas. Enuncia el equipo y de qué es prueba que sea universal
        (\cref{prop:g5:first-look-at-cells:principle}).
  \item El profesor cierra: ``todo lo que habéis dibujado hoy fue en su
        día una sola \omterm{def:g5:first-look-at-cells:cell}{célula}''. Defiende la frase para la cebolla, para
        la dueña de las \omterm{def:g5:first-look-at-cells:cell}{células} de mejilla y para las cadenas del yogur.
  \item Formula el hallazgo de la tarde como la teoría celular --- dos
        frases --- y señala, para cada preparación, qué frase ilustró
        mejor.
\end{enumerate}
\end{problem}
