\chapter{A vida ao longo das estações}\label{ch:g6:life-through-seasons}

Fotografe o mesmo castanheiro no primeiro dia de cada mês e pregue as
doze fotos numa fileira: \omterm{def:g1:plants-around-us:tree}{galhos} nus, \omterm{def:g6:life-through-seasons:buds}{brotos}, \omterm{def:g1:plants-around-us:parts}{folhas}, velas de \omterm{def:g3:flowers-fruits-seeds:parts}{flores},
castanhas, ouro, \omterm{def:g1:plants-around-us:tree}{galhos} nus de novo. Nada se moveu --- a árvore ficou
parada o ano inteiro --- e mesmo assim tudo mudou. O
\cref{ch:g3:animals-through-seasons} acompanhou os \omterm{def:g1:animals-around-us:animal}{animais} ao longo
do ano; agora, com \omterm{def:g1:the-five-senses:sense}{olhos} de quem levanta um sítio, acompanhamos o
\omterm{def:g6:exploring-our-environment:components}{ambiente} inteiro, começando pelas \omterm{def:g1:plants-around-us:plant}{plantas}.

\section{O ambiente muda com as estações}

\begin{proposition}[Mudança sazonal das condições]\label{prop:g6:life-through-seasons:conditions}
Ao longo do ano, as \omterm{def:g6:exploring-our-environment:conditions}{condições} de um \omterm{def:g6:exploring-our-environment:components}{ambiente}
(\cref{def:g6:exploring-our-environment:conditions}) oscilam bastante:
os dias encurtam e alongam, as \omterm{def:g6:exploring-our-environment:conditions}{temperaturas} caem e sobem, a \omterm{def:g6:exploring-our-environment:conditions}{umidade}
do solo acompanha a chuva e a seca. Um sítio levantado em janeiro e
em junho dá duas linhas diferentes na tabela --- então, pela
\cref{prop:g6:exploring-our-environment:decide}, espere dois elencos
diferentes de moradores.
\end{proposition}
\admitted

\begin{example}[Uma cerca viva, dois levantamentos]\label{ex:g6:life-through-seasons:hedge}
A cerca viva da estradinha, levantada duas vezes. Em junho: \omterm{def:g1:plants-around-us:parts}{folhas}
densas, dez \omterm{def:g5:biodiversity:species}{espécies} de \omterm{def:g1:plants-around-us:plant}{plantas} florindo no pé dela, abelhas
barulhentas, sabiás fazendo ninho, caracóis depois da chuva. Em
janeiro: \omterm{def:g1:plants-around-us:tree}{galhos} nus, o verde reduzido à hera e ao musgo, nenhum
inseto voando, sabiás procurando comida na camada de \omterm{def:g1:plants-around-us:parts}{folhas},
caracóis lacrados nas \omterm{def:g1:animals-around-us:covering}{conchas} deles fundo no barranco. O mesmo lugar,
a mesma lista de componentes --- vivos, minerais, vestígios --- mas a
\omterm{def:g3:bones-and-muscles:skeleton}{coluna} dos vivos se transformou.
\end{example}

\section{Como as plantas atravessam a estação ruim}

\begin{definition}[Caducifólias e perenifólias]\label{def:g6:life-through-seasons:trees}
As árvores se dividem pela resposta delas ao inverno:
\begin{itemize}
  \item as árvores \emph{caducifólias}\index{caducifólia} ---
        carvalho, castanheiro, faia --- soltam todas as \omterm{def:g1:plants-around-us:parts}{folhas} no
        outono e ficam nuas;
  \item as árvores \emph{perenifólias}\index{perenifólia} ---
        pinheiro, azevinho, a hera do muro --- mantêm \omterm{def:g1:plants-around-us:parts}{folhas}
        trabalhando o inverno inteiro, em geral \omterm{def:g1:plants-around-us:parts}{folhas} duras,
        enceradas ou em forma de agulha (a
        \cref{prop:g3:sorting-living-things:plants} já tinha
        encontrado as agulhas).
\end{itemize}
\end{definition}

\begin{definition}[Brotos]\label{def:g6:life-through-seasons:buds}
Uma árvore \omterm{def:g6:life-through-seasons:trees}{caducifólia} no inverno não está morta: a primavera
seguinte já está embalada nos galhinhos dela na forma de
\emph{brotos}\index{broto} --- ramos completos em miniatura, com as
\omterm{def:g1:plants-around-us:parts}{folhas} dobradas pequenininhas, embrulhadas em escamas à prova de
tempo. As \omterm{def:g1:plants-around-us:parts}{folhas} ``novas'' da primavera foram construídas no verão
anterior; o broto é o depósito de inverno delas.
\end{definition}

\begin{example}[Abrindo um broto de castanheiro-da-índia]\label{ex:g6:life-through-seasons:bud}
Os \omterm{def:g6:life-through-seasons:buds}{brotos} grandes e grudentos do castanheiro-da-índia se abrem bem
com uma lâmina (trabalho para um adulto): debaixo das escamas
envernizadas, embaladas numa penugem de algodão, estão \omterm{def:g1:plants-around-us:parts}{folhas}
dobradas em miniatura --- perfeitas, claras, esperando. Um galhinho
cortado em fevereiro e posto na água dentro de casa ``se enfolha''
semanas antes: os ramos estavam prontos; só faltava calor.
\end{example}

\begin{proposition}[As respostas de inverno da camada rasteira]\label{prop:g6:life-through-seasons:herbs}
As \omterm{def:g1:plants-around-us:plant}{plantas} menores têm as estratégias delas:
\begin{itemize}
  \item as \emph{anuais}\index{planta anual} --- a papoula, o feijão
        --- morrem por inteiro no outono e passam o inverno como
        \emph{\omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}} (\cref{prop:g2:from-seed-to-plant:needs}: uma
        \omterm{def:g2:from-seed-to-plant:seed}{semente} atravessa o frio em segurança, esperando);
  \item as \emph{perenes}\index{planta perene} --- o narciso, o
        dente-de-leão --- deixam a parte de cima morrer e esperam
        debaixo da terra como \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbos}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{tubérculos} ou \omterm{def:g1:plants-around-us:parts}{raízes} duras (as
        despensas do \cref{ex:g4:plants-without-seeds:bulbtuber}, em
        serviço de inverno);
  \item as \omterm{def:g6:life-through-seasons:trees}{perenifólias} do chão --- musgo, hera --- simplesmente
        aguentam.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Para onde foram os narcisos?]\label{ex:g6:life-through-seasons:daffodil}
O canteiro de narcisos floresce em março, amarela em maio e, em
julho, sumiu sem deixar vestígio --- e mesmo assim, em março
seguinte, ele volta mais cerrado. O verão, o outono e o inverno
inteiros ele existiu como \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbos} embaixo do gramado: a \omterm{def:g1:plants-around-us:plant}{planta} nunca
foi embora; ela se recolheu ao depósito.
\end{example}

\section{O calendário dos animais, revisitado}

\begin{proposition}[As respostas dos animais, em termos de levantamento]\label{prop:g6:life-through-seasons:animals}
As três estratégias do \cref{ch:g3:animals-through-seasons} ---
\omterm{def:g3:animals-through-seasons:migration}{migrar}, \omterm{def:g3:animals-through-seasons:hibernation}{hibernar}, continuar ativo --- agora se leem como respostas às
mudanças levantadas: cada \omterm{def:g5:biodiversity:species}{espécie} persegue as \omterm{def:g6:exploring-our-environment:conditions}{condições} e o alimento
de que precisa.
\begin{itemize}
  \item a andorinha acompanha o estoque de alimento dela até onde as
        \omterm{def:g6:exploring-our-environment:conditions}{condições} mantêm os \omterm{prop:g3:sorting-living-things:groups}{insetos} voando;
  \item o ouriço e o caracol lacrado esperam passar os meses em que
        os corpos deles não conseguem trabalhar;
  \item os \omterm{prop:g3:sorting-living-things:groups}{insetos} acrescentam uma quarta resposta: quase todos
        passam o inverno como \emph{\omterm{def:g2:animals-grow-up:born}{ovos}, larvas ou \omterm{ex:g2:animals-grow-up:butterfly}{crisálidas}}
        (\cref{ex:g2:animals-grow-up:butterfly}) enfiados na terra,
        na casca das árvores ou na camada de \omterm{def:g1:plants-around-us:parts}{folhas} --- o adulto
        morre, e a \omterm{def:g5:biodiversity:species}{espécie} passa o inverno nas fases jovens dela.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Onde estão em janeiro os insetos de junho]\label{ex:g6:life-through-seasons:insects}
A cerca viva de janeiro não mostra nenhum inseto --- e, mesmo assim,
todos os de junho estão presentes: as borboletas como \omterm{ex:g2:animals-grow-up:butterfly}{crisálidas}
debaixo das pedras da cumeeira, os gafanhotos como \omterm{def:g2:animals-grow-up:born}{ovos} na terra, as
joaninhas como adultas dormindo amontoadas no fundo da camada de
\omterm{def:g1:plants-around-us:parts}{folhas}, as abelhas como um cacho apertado e quentinho na colmeia. O
levantamento de junho anda por aí; o de janeiro precisa cavar,
levantar e espiar --- os moradores mudaram de \emph{forma}, e não de
endereço.
\end{example}

\begin{method}[Levantar ao longo do ano]\label{met:g6:life-through-seasons:yearsurvey}
Para estudar a vida sazonal de um sítio:
\begin{enumerate}
  \item levante o mesmo sítio marcado
        (\cref{met:g6:exploring-our-environment:survey}) uma vez em
        cada estação --- mesmo quadrado, mesmos instrumentos, mesmo
        tempo de busca;
  \item registre as \omterm{def:g6:exploring-our-environment:conditions}{condições} \emph{e} os moradores em cada vez;
        fotografe o sítio sempre do mesmo ponto;
  \item procure de propósito as formas escondidas do inverno:
        \omterm{def:g6:life-through-seasons:buds}{brotos}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbos}, \omterm{def:g2:animals-grow-up:born}{ovos}, \omterm{ex:g2:animals-grow-up:butterfly}{crisálidas}, caracóis lacrados --- e
        recoloque tudo o que levantar;
  \item ponha os quatro levantamentos lado a lado: emparelhe cada
        mudança de moradores com uma mudança de \omterm{def:g6:exploring-our-environment:conditions}{condições} e nomeie a
        forma de inverno ou o destino de cada ausente.
\end{enumerate}
\end{method}

\begin{remark}[Nada simplesmente some]\label{rem:g6:life-through-seasons:vanish}
A tabela do ano inteiro ensina uma regra profunda: nenhuma \omterm{def:g5:biodiversity:species}{espécie} do
levantamento de junho simplesmente deixou de existir. Cada uma está
em outro lugar (migrou), dormindo (hibernando), no depósito (\omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbo},
\omterm{def:g2:from-seed-to-plant:seed}{semente}) ou em outra fase da vida (\omterm{def:g2:animals-grow-up:born}{ovo}, \omterm{ex:g2:animals-grow-up:butterfly}{crisálida}). ``Sumiu'' nunca é
uma resposta completa na \omterm{def:g1:living-or-not:living}{biologia} --- a pergunta seguinte é sempre
\emph{sumiu para onde, e como o quê?}
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:g6:life-through-seasons:10}

\begin{exercise}[$\star\star$]\label{exo:g6:life-through-seasons:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:life-through-seasons:12}

\omterm{def:g1:plants-around-us:parts}{folhas} \omterm{def:g6:life-through-seasons:trees}{caducifólias} são largas e finas. Proponha que problema do

de água do tatuzinho ao contrário) e por que a árvore \omterm{def:g6:life-through-seasons:trees}{caducifólia}
prefere soltar as \omterm{def:g1:plants-around-us:parts}{folhas} finas dela.
\end{exercise}

\section{Problema: As doze fotografias}

\begin{problem}[{Problema de fim de semana --- um ano do castanheiro,
lido quadro a quadro}]\label{pb:g6:life-through-seasons:1}
A Lena fotografa o castanheiro do portão no primeiro dia de cada mês
durante um ano e prega as doze cópias numa fileira. Embaixo de cada
cópia ela anota a duração do dia e a \omterm{def:g6:exploring-our-environment:conditions}{temperatura} da manhã.

\medskip\noindent\textbf{Parte I --- Lendo a fileira.}
\begin{enumerate}
  \item Janeiro e dezembro mostram \omterm{def:g1:plants-around-us:tree}{galhos} nus. O que, nos galhinhos,
        prova que a árvore está viva e pronta para a primavera nos
        dois quadros?
  \item Entre que dois quadros os \omterm{def:g6:life-through-seasons:buds}{brotos} se abrem? As anotações da
        Lena ali dizem: dias de $11$ a $13$ horas, manhãs de $8$ a
        \qty{12}{\degreeCelsius}. A que \omterm{def:g6:exploring-our-environment:conditions}{condições} em mudança a
        árvore está respondendo?
  \item O quadro de maio mostra ``velas'' eretas de \omterm{def:g3:flowers-fruits-seeds:parts}{flores} brancas.
        Usando o \cref{ch:g3:flowers-fruits-seeds} (ou o
        \cref{ch:g5:flower-to-fruit}), o que precisa visitar a
        árvore entre os quadros de maio e de setembro para as
        castanhas de outubro existirem --- e o que é uma castanha,
        exatamente?
  \item O quadro de outubro mostra \omterm{def:g1:plants-around-us:parts}{folhas} douradas; o de novembro,
        \omterm{def:g1:plants-around-us:tree}{galhos} nus e um tapete marrom embaixo. Diga o tipo da árvore
        segundo a \cref{def:g6:life-through-seasons:trees} e onde a
        primavera seguinte dela fica guardada nos quadros que vêm
        depois.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Os vizinhos da árvore.}
\begin{enumerate}[resume]
  \item Embaixo da árvore, o quadro de março flagra um canteiro de
        narcisos que não aparece no quadro de julho. Onde está o
        canteiro em julho, e em que forma?
  \item As papoulas margeiam o caminho em junho; no quadro de
        janeiro, o lugar delas é terra nua. Em que forma a papoula
        está presente em janeiro, e por qual estratégia da
        \cref{prop:g6:life-through-seasons:herbs}?
  \item O quadro de junho flagra uma andorinha acima da árvore e
        abelhas nas \omterm{def:g3:flowers-fruits-seeds:parts}{flores} embaixo; nenhuma das duas aparece de
        novembro a fevereiro. Para cada uma, diga onde ela está
        nessa época, e como o quê.
  \item Uma pega aparece em todos os quadros, sem exceção. Que
        estratégia \omterm{def:g1:animals-around-us:animal}{animal} é essa, e o que a pega precisa administrar
        o inverno inteiro e a andorinha escapa indo embora?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- A regra da fileira.}
\begin{enumerate}[resume]
  \item A Lena resume: ``o ano da árvore acompanha o ano das
        \omterm{def:g6:exploring-our-environment:conditions}{condições}''. Apoie a afirmação dela com dois emparelhamentos
        de uma mudança da árvore de um quadro para o outro contra uma
        mudança nas anotações de duração do dia e de \omterm{def:g6:exploring-our-environment:conditions}{temperatura}.
  \item O irmãozinho dela diz que o canteiro de narcisos de julho
        ``morreu''. Corrija ele com a
        \cref{rem:g6:life-through-seasons:vanish} --- e enuncie a
        pergunta seguinte que a palavra ``sumiu'' sempre merece.
  \item Preveja o quadro 13 --- a fotografia de janeiro do ano
        seguinte --- com detalhes: \omterm{def:g1:plants-around-us:tree}{galhos}, galhinhos, chão embaixo.
        Que partes da sua previsão são certas o bastante para
        apostar, e por quê?
  \item Um galhinho de castanheiro, cortado em fevereiro e posto em
        água morna dentro de casa, se enfolha em três semanas. O que
        isso mostra sobre \emph{qual} condição os \omterm{def:g6:life-through-seasons:buds}{brotos} estavam
        esperando --- e planeje o teste justo de dois galhinhos
        (\cref{met:g4:what-plants-need:fairtest}) que fecharia a
        questão.
\end{enumerate}
\end{problem}
