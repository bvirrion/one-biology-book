\chapter{La cellule, unité commune du vivant}\label{ch:g10:cells-common-unit}

Grattez l'intérieur de votre joue avec un coton-tige, étalez le
prélèvement sur une lame, ajoutez une goutte de colorant bleu et
regardez au microscope à quatre cents fois~: une dispersion de tuiles
plates et irrégulières, chacune avec une tache plus sombre près de son
centre. Détachez la fine pellicule qui tapisse une écaille d'oignon et
regardez de nouveau~: de sages briques rectangulaires, chacune avec sa
tache. L'eau d'une mare montre des créatures ovales qui passent en
ramant, frangées de cils. Les trois échantillons n'ont rien de commun
à l'œil nu. Sous l'objectif, ils sont bâtis avec le même objet --- la
\omterm{def:g10:cells-common-unit:cell}{cellule} --- et ce chapitre dit ce qu'est cet objet, comment on le voit,
et pourquoi tout être vivant en est fait.

\section{La théorie cellulaire}

\begin{definition}[Cellule]\label{def:g10:cells-common-unit:cell}
Une \emph{cellule}\index{cellule} est la plus petite unité d'un être
vivant qui soit elle-même vivante~: un volume de contenu aqueux, le
\emph{cytoplasme}\index{cytoplasme}, enfermé par une fine
\emph{membrane plasmique}\index{membrane plasmique} qui le sépare de
son environnement tout en laissant passer certaines substances. Une
cellule prend de la matière et de l'énergie, les transforme, grandit,
et se divise en deux cellules.
\end{definition}

\begin{proposition}[La théorie cellulaire]\label{prop:g10:cells-common-unit:theory}
Trois énoncés, établis au microscope au dix-neuvième siècle, forment la
théorie cellulaire~:
\begin{enumerate}
  \item tout être vivant est fait d'une ou de plusieurs \omterm{def:g10:cells-common-unit:cell}{cellules}~;
  \item la \omterm{def:g10:cells-common-unit:cell}{cellule} est l'unité de structure et de fonction des êtres
        vivants --- rien de plus petit n'est vivant par soi-même~;
  \item toute \omterm{def:g10:cells-common-unit:cell}{cellule} provient d'une \omterm{def:g10:cells-common-unit:cell}{cellule} préexistante, par division.
\end{enumerate}
\end{proposition}
\begin{proof}[Faits expérimentaux]
Le premier énoncé repose sur l'examen systématique des végétaux
(Schleiden, 1838) et des animaux (Schwann, 1839) avec des microscopes
améliorés~: chaque tissu examiné, si différent fût-il d'aspect, se
résolvait en \omterm{def:g10:cells-common-unit:cell}{cellules}. Le troisième a été établi par Virchow (1855) et
par l'observation de \omterm{def:g10:cells-common-unit:cell}{cellules} en division dans les tissus en
croissance~; la solution de rechange --- des \omterm{def:g10:cells-common-unit:cell}{cellules} se formant
spontanément à partir d'un liquide --- n'a jamais été observée, et les
expériences de Pasteur sur les microbes (le
\cref{ch:g11:antibiotic-resistance} y revient) ont clos la question. Le
deuxième découle du premier et du fait que des \omterm{def:g10:cells-common-unit:cell}{cellules} isolées,
maintenues dans un liquide convenable, continuent de vivre, alors
qu'aucun fragment de \omterm{def:g10:cells-common-unit:cell}{cellule} ne le fait.
\end{proof}

\begin{example}[Une cellule ou beaucoup]\label{ex:g10:cells-common-unit:onemany}
Une levure, une paramécie ou une bactérie est une \omterm{def:g10:cells-common-unit:cell}{cellule} unique qui
fait tout --- se nourrir, se déplacer, se diviser. Un chêne ou un
humain est une coopérative de \omterm{def:g10:cells-common-unit:cell}{cellules} de nombreuses sortes, chacune
prenant sa part du travail~: un corps humain contient quelque
$\num{3e13}$ \omterm{def:g10:cells-common-unit:cell}{cellules}, d'environ deux cents sortes. Entre les deux se
placent des colonies et des organismes pluricellulaires simples, mais
il n'existe aucun être vivant bâti avec autre chose que des \omterm{def:g10:cells-common-unit:cell}{cellules}.
\end{example}

\section{Voir les cellules~: échelles et microscopes}

\begin{definition}[Grossissement et résolution]\label{def:g10:cells-common-unit:magnification}
Le \emph{grossissement}\index{grossissement} d'un instrument est le
rapport de la taille de l'image à la taille de l'objet. Sa
\emph{résolution}\index{résolution} est la plus petite distance entre
deux points qu'il montre encore séparés. Grossir une image au-delà de
ce que la résolution justifie agrandit le flou, non le détail.
\end{definition}

\begin{proposition}[Deux instruments, deux échelles]\label{prop:g10:cells-common-unit:microscopes}
Le microscope optique grossit jusqu'à environ $\times 1500$ avec une
\omterm{def:g10:cells-common-unit:magnification}{résolution} d'environ \qty{0.2}{\micro m}~: il montre les \omterm{def:g10:cells-common-unit:cell}{cellules}, les
\omterm{def:g10:cells-common-unit:organelle}{noyaux}, les \omterm{def:g10:cells-common-unit:organelle}{chloroplastes} et les bactéries, en couleur et vivants. Le
microscope électronique utilise un faisceau d'électrons au lieu de la
lumière~: \omterm{def:g10:cells-common-unit:magnification}{résolution} inférieure au \qty{1}{nm}, \omterm{def:g10:cells-common-unit:magnification}{grossissement} jusqu'à
$\times 10^6$~; il révèle la structure interne de la \omterm{def:g10:cells-common-unit:cell}{cellule} ---
membranes, \omterm{def:g10:cells-common-unit:organelle}{mitochondries}, \omterm{def:g10:cells-common-unit:organelle}{ribosomes}, et jusqu'aux grosses molécules ---
mais sur un échantillon mort, déshydraté et découpé en tranches plus
minces qu'un micromètre.
\end{proposition}
\admitted

\begin{method}[Mesurer un objet sur une micrographie]\label{met:g10:cells-common-unit:measure}
Une micrographie porte soit un \omterm{def:g10:cells-common-unit:magnification}{grossissement}, soit une échelle.
\begin{enumerate}
  \item Avec une échelle~: mesurer la barre sur la page (disons
        \qty{12}{mm} pour une barre annotée \qty{10}{\micro m}) et
        l'objet (\qty{30}{mm})~; la taille réelle est $30 \times 10/12 =
        \qty{25}{\micro m}$.
  \item Avec un \omterm{def:g10:cells-common-unit:magnification}{grossissement}~: taille réelle $=$ taille mesurée
        divisée par le \omterm{def:g10:cells-common-unit:magnification}{grossissement}~; une \omterm{def:g10:cells-common-unit:cell}{cellule} dessinée sur
        \qty{20}{mm} de long à
        $\times 400$ mesure $20/400 = \qty{0.05}{mm} = \qty{50}{\micro m}$.
  \item Toujours préciser l'unité, et convertir~: $\qty{1}{mm} =
        \qty{1000}{\micro m}$, $\qty{1}{\micro m} = \qty{1000}{nm}$.
\end{enumerate}
\end{method}

\begin{example}[Cellules de la joue]\label{ex:g10:cells-common-unit:cheek}
Une \omterm{def:g10:cells-common-unit:cell}{cellule} de joue dessinée sur \qty{24}{mm} de large à partir d'une
image à $\times 400$ mesure
$24/400 = \qty{0.06}{mm} = \qty{60}{\micro m}$ en travers, et son
\omterm{def:g10:cells-common-unit:organelle}{noyau}, dessiné sur \qty{3}{mm}, environ \qty{8}{\micro m}. Un globule
rouge mesure \qty{7}{\micro m}~; les bactéries de la bouche, longues de
\qty{1}{\micro m}, n'apparaissent au même \omterm{def:g10:cells-common-unit:magnification}{grossissement} que comme des
points à peine résolus.
\end{example}

\section{À l'intérieur de la cellule}

\begin{definition}[Organite]\label{def:g10:cells-common-unit:organelle}
Un \emph{organite}\index{organite} est une structure interne à la
\omterm{def:g10:cells-common-unit:cell}{cellule}, enfermée par sa propre membrane, qui accomplit une tâche
déterminée. Les principaux sont~:
\begin{itemize}
  \item le \emph{noyau}\index{noyau}, qui contient les chromosomes ---
        l'ADN du \cref{ch:g10:universal-dna} --- et commande l'activité
        de la \omterm{def:g10:cells-common-unit:cell}{cellule}~;
  \item les \emph{mitochondries}\index{mitochondrie}, longues d'environ
        \qty{1}{\micro m}, où les molécules organiques sont dégradées
        avec de l'oxygène pour libérer de l'énergie utilisable (la
        respiration cellulaire, \cref{ch:g10:cell-metabolism})~;
  \item dans les \omterm{def:g10:cells-common-unit:cell}{cellules} végétales, les
        \emph{chloroplastes}\index{chloroplaste}, verts de
        chlorophylle, où se fait la photosynthèse, et une grande
        \emph{vacuole}\index{vacuole} pleine d'eau qui maintient la
        \omterm{def:g10:cells-common-unit:cell}{cellule} turgescente.
\end{itemize}
En dehors de la membrane, une \omterm{def:g10:cells-common-unit:cell}{cellule} végétale est enfermée dans une
\emph{paroi}\index{paroi} rigide de cellulose~; les \omterm{def:g10:cells-common-unit:cell}{cellules} animales
n'en ont pas. Le \omterm{def:g10:cells-common-unit:cell}{cytoplasme} de toute \omterm{def:g10:cells-common-unit:cell}{cellule} contient aussi des
millions de \emph{ribosomes}\index{ribosome}, particules d'environ
\qty{25}{nm} qui assemblent les \omterm{def:g10:chemistry-of-life:families}{protéines}, visibles seulement au
microscope électronique.
\end{definition}

\begin{definition}[Procaryotes et eucaryotes]\label{def:g10:cells-common-unit:prokaryote}
Une \omterm{def:g10:cells-common-unit:cell}{cellule} dont l'ADN est enfermé dans un \omterm{def:g10:cells-common-unit:organelle}{noyau} est une
\emph{cellule eucaryote}\index{cellule eucaryote}~: les animaux, les
végétaux, les champignons et beaucoup d'organismes unicellulaires sont
des eucaryotes. Une
\emph{cellule procaryote}\index{cellule procaryote} n'a ni \omterm{def:g10:cells-common-unit:organelle}{noyau} ni
\omterm{def:g10:cells-common-unit:organelle}{organite} entouré d'une membrane~: son ADN baigne librement dans le
\omterm{def:g10:cells-common-unit:cell}{cytoplasme}, sa taille est de l'ordre du \qty{1}{\micro m}, et elle est
entourée d'une paroi qui lui est propre. Les bactéries sont des
procaryotes.
\end{definition}

\begin{example}[Lire une bactérie]\label{ex:g10:cells-common-unit:bacterium}
Une micrographie électronique de la bactérie intestinale
\emph{Escherichia coli} montre un bâtonnet long de \qty{2}{\micro m} et
large de \qty{0.5}{\micro m}~: une paroi, une membrane juste en
dessous, un \omterm{def:g10:cells-common-unit:cell}{cytoplasme} granuleux bourré de \omterm{def:g10:cells-common-unit:organelle}{ribosomes}, et une région
centrale plus claire où repose l'unique molécule d'ADN circulaire. Pas
de \omterm{def:g10:cells-common-unit:organelle}{noyau}, pas de \omterm{def:g10:cells-common-unit:organelle}{mitochondrie}. Et pourtant la \omterm{def:g10:cells-common-unit:cell}{cellule} se nourrit,
grandit et se divise toutes les vingt minutes dans un bouillon riche~:
tout ce que la théorie cellulaire exige, dans un millième du volume
d'une \omterm{def:g10:cells-common-unit:cell}{cellule} de joue.
\end{example}

\section{Beaucoup de cellules, beaucoup de formes~: la spécialisation}

\begin{proposition}[Cellules spécialisées]\label{prop:g10:cells-common-unit:specialised}
Dans un organisme pluricellulaire, les \omterm{def:g10:cells-common-unit:cell}{cellules} partagent le même plan
de base et le même ADN, mais chaque sorte a une forme et un équipement
accordés à sa tâche~: un neurone étend une fibre longue jusqu'à un
mètre pour porter les signaux~; un globule rouge a perdu son \omterm{def:g10:cells-common-unit:organelle}{noyau} et
est un disque aplati bourré d'hémoglobine~; une fibre musculaire est
une \omterm{def:g10:cells-common-unit:cell}{cellule} géante pleine de \omterm{def:g10:chemistry-of-life:families}{protéines} contractiles~; un poil absorbant
est un tube fin qui maximise la surface d'absorption de l'eau~; une
\omterm{def:g10:cells-common-unit:cell}{cellule} de garde d'une feuille change de forme pour ouvrir ou fermer un
pore. La spécialisation est affaire des parties du plan commun que
l'on développe.
\end{proposition}
\admitted

\begin{example}[Surface et volume]\label{ex:g10:cells-common-unit:surface}
Pourquoi les \omterm{def:g10:cells-common-unit:cell}{cellules} sont-elles si petites~? Tout ce qu'une \omterm{def:g10:cells-common-unit:cell}{cellule}
prend ou rejette traverse sa membrane, dont l'aire croît comme le carré
de la taille, tandis que les besoins croissent comme le volume, donc
comme le cube. Une \omterm{def:g10:cells-common-unit:cell}{cellule} cubique de côté \qty{10}{\micro m} a une
surface de \qty{600}{\micro m^2} pour un volume de
\qty{1000}{\micro m^3}~: \qty{0.6}{\micro m^2} de membrane par
micromètre cube. À \qty{100}{\micro m} de côté, le rapport tombe à
\qty{0.06}{\micro m^2}~: dix fois moins de membrane par unité de
contenu. Une grande \omterm{def:g10:cells-common-unit:cell}{cellule} mourrait de faim en son centre~; les
\omterm{def:g10:cells-common-unit:cell}{cellules} qui sont grandes (un œuf de grenouille, une fibre musculaire)
sont soit surtout de la réserve inerte, soit allongées de sorte
qu'aucun point ne soit loin de la surface.
\end{example}

\begin{omfigure}
\includegraphics[width=0.62\linewidth]{images/book2/ai/g10-cells-paramecium.jpg}

{\small Une goutte d'eau de mare au microscope optique~: une paramécie,
\omterm{def:g10:cells-common-unit:cell}{cellule} unique longue d'un quart de millimètre, passe en ramant devant
des algues vertes sous le battement de milliers de cils minuscules. Une
\omterm{def:g10:cells-common-unit:cell}{cellule}, et un organisme entier.}
\end{omfigure}

\begin{remark}[La cellule, système ouvert]\label{rem:g10:cells-common-unit:open}
Une \omterm{def:g10:cells-common-unit:cell}{cellule} n'est pas un sac étanche. L'eau, l'oxygène, le glucose et
les ions traversent sa membrane en permanence, et ses déchets sortent
par le même chemin~; une \omterm{def:g10:cells-common-unit:cell}{cellule} de joue placée dans de l'eau pure
gonfle et éclate en quelques minutes, et une \omterm{def:g10:cells-common-unit:cell}{cellule} placée dans de
l'eau très salée se ratatine. Le contrôle exercé par la membrane sur ce
qui entre et sort fait l'objet du volume de Licence 1\textsuperscript{re}
année~; pour l'instant, retenez qu'une \omterm{def:g10:cells-common-unit:cell}{cellule} vit en échangeant avec
son environnement, et meurt quand les échanges s'arrêtent.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g10:cells-common-unit:1}
Énoncer les trois affirmations de la théorie cellulaire.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:cells-common-unit:2}
Convertir~: \qty{0.05}{mm} en micromètres~; \qty{2500}{nm} en
micromètres~; \qty{7}{\micro m} en millimètres.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:cells-common-unit:3}
Une \omterm{def:g10:cells-common-unit:cell}{cellule} est dessinée sur \qty{36}{mm} de long à partir d'une image
à $\times 600$. Quelle est sa longueur réelle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:cells-common-unit:4}
Nommer trois structures présentes dans une \omterm{def:g10:cells-common-unit:cell}{cellule} végétale et absentes
d'une \omterm{def:g10:cells-common-unit:cell}{cellule} animale, et une structure présente dans toute \omterm{def:g10:cells-common-unit:cell}{cellule}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:cells-common-unit:5}
Quelle est la différence entre une \omterm{def:g10:cells-common-unit:prokaryote}{cellule procaryote} et une \omterm{def:g10:cells-common-unit:prokaryote}{cellule
eucaryote}~? Donner un exemple de chacune.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cells-common-unit:6}
Sur une micrographie, une échelle annotée \qty{5}{\micro m} mesure
\qty{15}{mm}~; une \omterm{def:g10:cells-common-unit:organelle}{mitochondrie} mesure \qty{6}{mm} de long. Calculer la
longueur réelle de la \omterm{def:g10:cells-common-unit:organelle}{mitochondrie} et le \omterm{def:g10:cells-common-unit:magnification}{grossissement} de l'image.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cells-common-unit:7}
Pourquoi un microscope optique ne peut-il pas montrer les \omterm{def:g10:cells-common-unit:organelle}{ribosomes},
quel que soit son \omterm{def:g10:cells-common-unit:magnification}{grossissement}~? Utiliser les deux mots de la
\cref{def:g10:cells-common-unit:magnification}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cells-common-unit:8}
De l'eau iodée est ajoutée à une tranche de pomme de terre sous le
microscope~: la couleur bleu-noir apparaît dans des grains ovales à
l'intérieur des \omterm{def:g10:cells-common-unit:cell}{cellules}. À l'aide du
\cref{ch:g10:chemistry-of-life}, dire ce que sont ces grains et ce que
cela montre sur le lieu où une plante stocke ses réserves.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cells-common-unit:9}
Les \omterm{def:g10:cells-common-unit:cell}{cellules} de l'épiderme d'oignon montrent un \omterm{def:g10:cells-common-unit:organelle}{noyau} et une paroi mais
aucun \omterm{def:g10:cells-common-unit:organelle}{chloroplaste}, alors que les \omterm{def:g10:cells-common-unit:cell}{cellules} d'une feuille de mousse en
sont pleines. Expliquer la différence à partir du lieu où vit chaque
tissu.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cells-common-unit:10}
Calculer le rapport surface/volume d'une \omterm{def:g10:cells-common-unit:cell}{cellule} cubique de côté
\qty{1}{\micro m} (une bactérie) et de côté \qty{20}{\micro m} (une
\omterm{def:g10:cells-common-unit:cell}{cellule} animale ordinaire). Laquelle peut se contenter de la simple
diffusion à travers sa membrane~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:cells-common-unit:11}
Une bactérie en bouillon riche se divise toutes les \qty{20}{min}. À
partir d'une \omterm{def:g10:cells-common-unit:cell}{cellule}, combien y en a-t-il après \qty{2}{h}~? Après
\qty{4}{h}~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:cells-common-unit:12}
Un globule rouge n'a ni \omterm{def:g10:cells-common-unit:organelle}{noyau} ni \omterm{def:g10:cells-common-unit:organelle}{mitochondrie}. Satisfait-il à la
\cref{def:g10:cells-common-unit:cell}~? Discuter, sachant qu'il vit
environ 120 jours, ne peut pas se diviser, et est produit par des
\omterm{def:g10:cells-common-unit:cell}{cellules} nucléées de la moelle osseuse.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:cells-common-unit:13}
En 1665, Hooke voyait des «~\omterm{def:g10:cells-common-unit:cell}{cellules}~» dans une tranche de liège ---
des boîtes vides aux parois épaisses. Expliquer ce qu'il a réellement
vu, dans le vocabulaire de ce chapitre, et pourquoi il n'aurait pas pu
y voir de \omterm{def:g10:cells-common-unit:organelle}{noyau}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:cells-common-unit:14}
Les virus sont des particules de \qty{20}{nm} à \qty{300}{nm} qui
contiennent un \omterm{def:g10:chemistry-of-life:families}{acide nucléique} et des \omterm{def:g10:chemistry-of-life:families}{protéines}, ne peuvent ni croître
ni se diviser seules, et ne se multiplient qu'à l'intérieur d'une
\omterm{def:g10:cells-common-unit:cell}{cellule}. Sont-ils des \omterm{def:g10:cells-common-unit:cell}{cellules}~? Sont-ils vivants~? Argumenter à
partir de la théorie cellulaire.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:cells-common-unit:15}
Une fibre musculaire est une \omterm{def:g10:cells-common-unit:cell}{cellule} unique longue de \qty{3}{cm} et

Calculer son volume et le comparer à celui d'une \omterm{def:g10:cells-common-unit:cell}{cellule} cubique de
\qty{20}{\micro m}~; puis expliquer, à partir de
l'\cref{ex:g10:cells-common-unit:surface}, pourquoi sa forme et ses
nombreux \omterm{def:g10:cells-common-unit:organelle}{noyaux} rendent une telle taille viable.
\end{exercise}

\section{Problème~: compter les cellules d'un corps}

\begin{problem}[{Problème du week-end --- une séance de microscopie, de
l'épiderme d'oignon posé sur la lame à une estimation du nombre de
cellules d'un corps humain}]
\label{pb:g10:cells-common-unit:1}
Une classe passe un après-midi au microscope. Les objectifs disponibles
donnent des \omterm{def:g10:cells-common-unit:magnification}{grossissements} totaux de $\times 40$, $\times 100$ et
$\times 400$~; le champ de vision à $\times 100$ mesure \qty{1.8}{mm}
en travers.

\medskip
\noindent\textbf{Partie I --- L'oignon.}
\begin{enumerate}
  \item À $\times 100$, environ 9 \omterm{def:g10:cells-common-unit:cell}{cellules} d'oignon tiennent côte à
        côte en travers du champ. Estimer la largeur d'une \omterm{def:g10:cells-common-unit:cell}{cellule}, en
        micromètres.
  \item À $\times 400$, le champ est quatre fois plus étroit. Quelle
        est sa largeur, et combien de ces \omterm{def:g10:cells-common-unit:cell}{cellules} y tiennent~?
  \item Un élève dessine une \omterm{def:g10:cells-common-unit:cell}{cellule} sur \qty{28}{mm} de large à partir
        de l'image à $\times 400$. Quel est le \omterm{def:g10:cells-common-unit:magnification}{grossissement} du dessin
        par rapport à la \omterm{def:g10:cells-common-unit:cell}{cellule} réelle~?
  \item L'eau iodée colore le \omterm{def:g10:cells-common-unit:organelle}{noyau} en jaune-brun~; l'élève le mesure à
        \qty{4}{mm} sur le dessin. Diamètre réel~?
  \item Nommer les structures que l'élève devrait annoter sur le
        dessin, et une structure présente qui ne peut pas être vue à ce
        \omterm{def:g10:cells-common-unit:magnification}{grossissement}.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La joue et la mare.}
\begin{enumerate}[resume]
  \item Les \omterm{def:g10:cells-common-unit:cell}{cellules} de joue colorées au bleu de méthylène
        apparaissent comme des polygones plats d'environ
        \qty{60}{\micro m} de large mais de \qty{5}{\micro m}
        d'épaisseur seulement. Pourquoi sont-elles plates, vu leur
        provenance~?
  \item Une goutte d'eau de mare montre une paramécie longue de
        \qty{240}{\micro m}. Peut-elle être vue à l'œil nu~? À quel
        \omterm{def:g10:cells-common-unit:magnification}{grossissement} remplit-elle le quart du champ~?
  \item De minuscules points mobiles longs d'environ \qty{1}{\micro m}
        sont visibles près de la paramécie à $\times 400$. Que
        sont-ils, très probablement, et qu'ajouterait un microscope
        électronique~?
  \item La paramécie est une \omterm{def:g10:cells-common-unit:cell}{cellule} unique~; la \omterm{def:g10:cells-common-unit:cell}{cellule} de joue est
        l'une de milliers de milliards. Dire en une phrase chacun ce
        que «~spécialisée~» signifie pour la \omterm{def:g10:cells-common-unit:cell}{cellule} de joue et
        pourquoi le mot ne peut pas s'appliquer à la paramécie.
  \item Un élève ajoute une goutte d'eau salée aux \omterm{def:g10:cells-common-unit:cell}{cellules} de joue et
        les voit se ratatiner. Expliquer avec la
        \cref{rem:g10:cells-common-unit:open}.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La taille d'une \omterm{def:g10:cells-common-unit:cell}{cellule}.}
\begin{enumerate}[resume]
  \item Prendre une \omterm{def:g10:cells-common-unit:cell}{cellule} humaine ordinaire pour un cube de côté
        \qty{20}{\micro m}. Calculer son volume en micromètres cubes,
        puis en millimètres cubes
        ($\qty{1}{mm^3} = 10^9\,\unit{\micro m^3}$).
  \item Calculer son aire et le rapport surface/volume.
  \item Refaire le calcul pour une bactérie prise pour un cube de côté
        \qty{1}{\micro m}. De quel facteur son rapport est-il plus
        grand~?
  \item Les \omterm{def:g10:cells-common-unit:cell}{cellules} ont une masse à peu près égale à celle d'un même
        volume d'eau ($\qty{1}{g}$ par \unit{cm^3}). Estimer la masse
        de la \omterm{def:g10:cells-common-unit:cell}{cellule} de \qty{20}{\micro m} en nanogrammes
        ($\qty{1}{g} = 10^9\,\unit{ng}$).
  \item Un globule rouge est un disque de \qty{7}{\micro m} de
        diamètre et \qty{2}{\micro m} d'épaisseur. Estimer son volume
        (le traiter comme un cylindre, $V = \pi r^2 h$) et le comparer
        au cube de la question 11.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Le nombre de \omterm{def:g10:cells-common-unit:cell}{cellules} d'un corps.}
\begin{enumerate}[resume]
  \item Un élève de masse \qty{60}{kg} a un volume d'environ
        \qty{60}{L}. Si le corps n'était fait que de \omterm{def:g10:cells-common-unit:cell}{cellules} cubiques
        de \qty{20}{\micro m}, combien en contiendrait-il~?
  \item En réalité, environ un tiers du volume du corps est du liquide
        entre les \omterm{def:g10:cells-common-unit:cell}{cellules} et d'autres matériaux non cellulaires.
        Corriger l'estimation.
  \item Les globules rouges comptent à eux seuls pour environ
        $\num{2.5e13}$. À l'aide du volume de la question 15, quel
        volume occupent-ils~? Est-ce compatible avec un volume sanguin
        d'environ \qty{4.5}{L} dont 45~\% de \omterm{def:g10:cells-common-unit:cell}{cellules}~?
  \item Combiner les questions 17 et 18~: donner un ordre de grandeur
        du nombre total de \omterm{def:g10:cells-common-unit:cell}{cellules} du corps, et dire pourquoi un
        compte précis est impossible.
  \item Énoncer le résultat~: combien de \omterm{def:g10:cells-common-unit:cell}{cellules}, en gros, font un
        humain, et par quel processus --- à partir de quel nombre
        initial --- ont-elles toutes été produites~?
\end{enumerate}
\end{problem}
