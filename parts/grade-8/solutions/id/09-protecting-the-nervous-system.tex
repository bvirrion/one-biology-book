\section*{Bab \ref{ch:g8:protecting-the-nervous-system} --- Melindungi Sistem Saraf}

\begin{solution}{exo:g8:protecting-the-nervous-system:1}
Ia memadatkan \omterm{def:g8:nervous-communication:synapse}{sinapsis} yang menguat pada hari itu ---
mengarsipkan pelajarannya --- lalu menjalankan tata graha
kimiawi \omterm{def:g5:brain-in-command:system}{otaknya}. \omterm{def:g5:brain-in-command:system}{Otak} \omterm{def:g3:stages-of-life:stages}{remajanya} paling membutuhkan jamnya:
sedang dirangkai ulang, dan jamnya hanyut makin malam
melawan sekolah yang pagi.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:2}
Karena penerima indera \omterm{def:g1:the-five-senses:sense}{telinganya} mati menurut takaran ---
kenyaringan dikali lama paparannya --- dan tidak pernah
tumbuh kembali. Yang dibelanjakan: modal penerimanya.
Peringatannya: desingan sesudah sebuah malam yang keras.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:3}
Di sinapsisnya --- menirukan sebuah pengantar pesan,
menghadang salah satunya, atau memaksakan pelepasannya.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:4}
Ia meredam simpangnya di seluruh sistem: pertimbangan
(penimbangannya), \omterm{def:g5:brain-in-command:reaction}{waktu reaksi} (jam gelungnya), keseimbangan
dan ingatan --- lebih lambat dan lebih tumpul di setiap
perhentiannya.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:5}
Takaran yang berulang membuat simpang ganjarannya
menyesuaikan diri --- reseptornya dipangkas, ambangnya
dinaikkan --- sampai zatnya dibutuhkan sekadar untuk merasa
biasa. Ketiadaannya lalu menjadi protes sistemnya: pintu
keluarnya mendaki melawan \omterm{def:g8:nervous-communication:synapse}{sinapsis} yang sudah dirangkai ulang.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:6}
Karena ia melatih simpang ganjaran yang sama, tanpa sebuah
molekul: hadiah kecil yang tak terduga adalah jadwal yang
paling keras menguatkan kabelnya, dan ongkosnya mendarat
pada \omterm{prop:g1:healthy-habits:needs}{tidur} dan perhatian yang sama yang dilindungi babnya.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:7}
Belajar ulang menguatkan simpangnya; \omterm{prop:g1:healthy-habits:needs}{tidur} memadatkannya.
Murid tengah-malam-sampai-pukul-dua itu membeli
penyeberangan mentah lalu menjual pengarsipannya; yang \omterm{prop:g1:healthy-habits:needs}{tidur}
pukul sepuluh mempertahankan keduanya. Bahan yang tak
dipadatkan adalah belajar ulang yang setengah dimiliki.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:8}
\omterm{def:g5:brain-in-command:system}{Otak} \omterm{def:g3:stages-of-life:stages}{remajanya} sedang dibangun --- petunjuk yang terganggu
bergaung menembus pembangunannya (perangkaian ulang pada
\cref{rem:g8:puberty:moods}) --- dan simpangnya berada pada
keadaan yang paling dapat disetel, sehingga perangkaian
ulang ganjaran pada \omterm{prop:g5:healthy-choices:substances}{kecanduannya} mengeras paling cepat
justru di situ.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:9}
Meter yang ditempuh sebuah mobil sebelum remnya adalah \omterm{def:g5:brain-in-command:reaction}{waktu
reaksi} yang dijadikan jarak --- dan alkohol memanjangkan
gelungnya pada simpangnya yang teredam: melapor, menimbang
dan memerintah semuanya lebih lambat. Batas undang-undangnya
adalah, secara mekanis, sebuah langit-langit atas meter yang
ditambahkan.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:10}
Tidurlah sekian jamnya: pemadatannya berjalan pada malam
hari. Anggarkan kenyaringannya: penerimanya adalah modal
yang tak terbarukan. Tanpa pemerdaya simpang: zatnya
merangkai ulang simpang yang dapat disetel menuju jebakannya
--- dan \omterm{def:g5:brain-in-command:system}{otak} yang masih dibangun merangkai ulang paling
cepat. (Helm: kabelnya tidak pulih seperti \omterm{def:g3:bones-and-muscles:skeleton}{tulang}; pakailah
mesinnya: simpangnya menguat oleh lalu lintas.)
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:11}
Mencari pertolongan menjadi perawatan, bukan pengakuan
dosa: sebuah keadaan kabel punya pengobatan, alamat dan
tanpa rasa malu --- sehingga yang terjebak mencarinya lebih
awal. Menawarkannya menjadi praktis: masalah seorang teman
adalah sesuatu yang dibawa berjalan ke pintu perawatnya,
bukan rahasia yang disimpan atau vonis yang dijatuhkan.
\end{solution}

\begin{solution}{exo:g8:protecting-the-nervous-system:12}
Celah yang menyesuaikan diri itulah sebabnya mesinnya dapat
belajar: pemakaian menguatkan penyeberangannya, dan
keterampilan, ingatan serta setiap gelung yang terlatih
adalah simpang yang dipadatkan --- sebulan sang penyulap
bola. Keterstelan yang sama itulah kerawanannya: zat dan
jadwal hadiah yang menakar simpang ganjarannya juga
menyetelnya, ke arah membutuhkan takarannya --- \omterm{prop:g5:healthy-choices:substances}{kecanduan}
adalah mekanisme belajarnya, yang diajari pelajaran yang
keliru. Satu sifat, dua wajah: lindungilah keterstelannya,
dan ia berbunga menjadi keterampilan; serahkan ia kepada
pemerdaya simpang, dan ia berbunga menjadi jebakan.
\end{solution}

\begin{solution}{pb:g8:protecting-the-nervous-system:1}
\textbf{1.} Ramalannya: sukarelawan yang beristirahat cukup
menangkap beberapa sentimeter lebih awal, dan lebih mantap
di seluruh percobaannya. Mekanismenya: \omterm{prop:g1:healthy-habits:needs}{tidur} memadatkan
simpang hari itu lalu membersihkan keausan kimiawinya ---
gelung yang bermalam pendek menimbang dengan lamban di
setiap sinapsisnya.

\textbf{2.} ``Meternya memperlihatkan laju takarannya;
sumbatnya memotongnya. Penerima indera \omterm{def:g1:the-five-senses:sense}{telinganya} mati
menurut takaran yang menumpuk lalu tidak pernah tumbuh
kembali --- pendengaran adalah modal, yang dibelanjakan
paling keras di konser dan pada penyuara \omterm{def:g1:the-five-senses:sense}{telinga} bervolume
penuh, dan dibela dengan lebih rendah, lebih pendek, dan
tersumbat.''

\textbf{3.} Menirukan: nikotin --- menyamar sebagai
pengantar pesan alami di simpang ganjarannya. Menghadang:
bisa yang melumpuhkan itu --- pengantar pesan simpang
\omterm{def:g3:bones-and-muscles:muscle}{ototnya} ditolak. Memaksa: alkohol cocok pada bagannya
sebagai peredam seluruh sistemnya (panahnya: membekap
percakapan simpangnya dari ujung ke ujung).

\textbf{4.} Sasarannya: simpang ganjaran yang sama, dilatih
oleh jadwal hadiah-tak-terduga yang sama --- peragaan
pemberitahuannya adalah peragaan penakaran, dengan \omterm{def:g6:exploring-our-environment:conditions}{cahaya}
dan bunyi sebagai ganti molekulnya.

\textbf{5.} Pada usia empat belas simpangnya masih dibangun
dan berada pada keadaan yang paling dapat disetel: takaran
yang sama merangkai ulang lebih dalam lalu mengeras lebih
cepat daripada pada \omterm{def:g5:brain-in-command:system}{otak} yang sudah jadi --- ``ternyata
baik-baik saja'' dimainkan pada tingkat kesukaran yang
lebih mudah.

\textbf{6.} Harga rokok pertamanya adalah semua rokok yang
menyusulnya: tiap takaran memulai penyesuaian simpang
ganjarannya, dan pintu masuknya persis selebar satu
keputusan --- pintu keluarnya, bertahun-tahun protes yang
dirangkai ulang. Yang ditawarkan adalah ambang pintu
jebakannya, bukan sebatang rokok.

\textbf{7.} Televisi menjadwalkan ganjarannya; umpannya
membagikannya secara tak terduga, butir demi butir --- dan
hadiah kecil yang tak terduga adalah jadwal yang paling
keras melatih simpangnya, itulah sebabnya kebiasaan
memeriksanya sukar dihentikan seperti tak pernah terjadi
pada acaranya.

\textbf{8.} Bahwa berhenti sendirian berarti melawan simpang
yang sudah dirangkai ulang, dan tidak ada yang harus
begitu: perawat dan dokternya mengobati keadaan kabel itu
dengan rahasia terjaga --- termasuk bagi anak di bawah umur
--- dan penyesuaian ulangnya berhasil. Langkah pertamanya:
satu orang dewasa yang dipercaya, hari ini, sebab diam
adalah satu-satunya langkah yang kalah.

\textbf{9.} ``Sebulan menyulap bola menguatkan simpangnya
sampai tiga bola terbang sendiri --- itulah belajar, dan
itulah gunanya sinapsismu. \omterm{prop:g5:healthy-choices:substances}{Kecanduan} adalah mekanisme yang
sama persis: takarannya menguatkan gelung yang keliru sampai
ia berjalan sendiri. \omterm{def:g5:brain-in-command:system}{Otaknya} tak dapat menawarkan yang satu
tanpa mempertaruhkan yang lain --- keterstelannya adalah
seluruh hadiahnya. Jadi pertanyaannya tidak pernah apakah
simpangmu akan dilatih, melainkan oleh apa.''

\textbf{10.} Karena saat tawarannya datang adalah saat yang
paling buruk untuk mulai menimbang --- tekanannya membebani
justru simpang yang sedang memutuskan. Sebuah kalimat yang
disusun dengan tenang lebih dahulu mengubah penolakannya
menjadi gerak refleks: perutean yang sudah bertimbangan,
siap sebelum dibutuhkan.

\textbf{11.} Sebuah urutan yang dapat dipertahankan: \omterm{prop:g1:healthy-habits:needs}{tidur}
lebih dahulu (ia merawat perolehan semua butir lainnya),
lalu tanpa pemerdaya simpang (jebakan yang tak
terbalikkan), lalu kenyaringannya (modal yang tak
tergantikan), lalu jam layarnya, lalu helmnya. Pilihan
pertamanya dipertahankan: manfaat setiap butir lainnya
dipadatkan semalaman --- tidurlah perawatan yang
disandarkan oleh selebihnya.

\textbf{12.} ``Kita sedang melindungi simpang yang dapat
disetel, yang akan sudah mempelajari segala yang pernah kamu
ketahui --- dan usia empat belas duduk di tengah dasawarsa
tahun-tahunnya yang paling cepat, paling terlatihkan, dan
paling terjebakkan.''
\end{solution}
