\chapter{दिल और रक्त का चक्कर}\label{ch:g7:heart-and-circulation}

\Cref{ch:g4:heart-and-blood} ने उस पंप और उसके पहुँचाने वाले चक्करों
से मिलवाया था; तब से \omterm{prop:g4:journey-of-food:absorb}{रक्त} पर ज़िम्मेदारियाँ चढ़ती गई हैं --- \omterm{prop:g7:digestion-and-nutrients:absorption}{रसांकुरों} पर
\omterm{def:g7:digestion-and-nutrients:families}{पोषक तत्व} (\cref{prop:g7:digestion-and-nutrients:absorption}),
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाओं} पर \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन}
(\cref{prop:g7:lungs-and-gas-exchange:numbers}), और हर काम करती \omterm{def:g3:bones-and-muscles:muscle}{पेशी}
पर बिल (\cref{prop:g7:organs-at-work:bill})। अब सड़कों का जाल ठीक से
नक़्शे में उतारने का वक़्त है: \omterm{def:g4:heart-and-blood:heart}{दिल} के चार कमरे, दोनों चक्कर, और तीन तरह
की \omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ}।

\section{दिल के चार कमरे}

\begin{definition}[अलिंद और निलय]\label{def:g7:heart-and-circulation:rooms}
\omterm{def:g4:heart-and-blood:heart}{दिल} के दोनों पहलुओं (\cref{def:g4:heart-and-blood:heart}) में से हर एक
के दो कमरे हैं: ऊपर वाला \emph{अलिंद}\index{अलिंद}, जो आता हुआ \omterm{prop:g4:journey-of-food:absorb}{रक्त}
लेता है, और नीचे वाला \emph{निलय}\index{निलय} --- यानी मोटी दीवार वाला
पंप करने का कमरा --- जो उसे बाहर धकेलता है। उनके बीच और उनके आगे
\emph{कपाट}\index{कपाट (दिल के)} --- यानी एकतरफ़ा दरवाज़े --- हर लहर के
पीछे चटाक से बंद हो जाते हैं; और उनकी जोड़ियों की वही चटाक ठीक-ठीक वह
``लब-डब'' है जो स्टेथस्कोप (या छाती पर रखा \omterm{def:g1:the-five-senses:sense}{कान}) सुनता है।
\end{definition}

\begin{proposition}[एक धड़कन, क्रम से]\label{prop:g7:heart-and-circulation:beat}
हर धड़कन दोनों पहलुओं पर एक ही साथ वही क्रम चलाती है: \omterm{def:g7:heart-and-circulation:rooms}{अलिंद} सिकुड़ते
हैं और \omterm{def:g7:heart-and-circulation:rooms}{निलयों} को ऊपर तक भर देते हैं; \omterm{def:g7:heart-and-circulation:rooms}{निलय} सिकुड़ते हैं और \omterm{prop:g4:journey-of-food:absorb}{रक्त} बाहर
धकेलते हैं --- दायाँ पहलू फेफड़ों की ओर, बायाँ पहलू शरीर की ओर ---
जबकि भीतर आने वाले \omterm{def:g7:heart-and-circulation:rooms}{कपाट} चटकते हैं (लब); फिर \omterm{def:g7:heart-and-circulation:rooms}{निलय} ढीले पड़ते हैं, बाहर
जाने वाले \omterm{def:g7:heart-and-circulation:rooms}{कपाट} चटकते हैं (डब), और कमरे फिर भर जाते हैं। और फिर वही, दिन
में $100000$ बार (\cref{rem:g4:heart-and-blood:rest})।
\end{proposition}
\admitted

\section{दो चक्कर, तीन वाहिकाएँ}

\begin{definition}[दोहरा रक्त-चक्र]\label{def:g7:heart-and-circulation:double}
\omterm{prop:g4:journey-of-food:absorb}{रक्त} का सड़क-जाल एक ही पंप में से गुज़रते दो फंदे हैं
(\cref{def:g4:heart-and-blood:heart} वाले दोनों पहलू, अब चक्करों के
रूप में):
\begin{itemize}
  \item \emph{फेफड़ों वाला चक्कर}: दायाँ \omterm{def:g7:heart-and-circulation:rooms}{निलय}
        $\rightarrow$ \omterm{def:g7:how-animals-breathe:four}{फेफड़े} (\omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} लादते, \omterm{def:g7:what-is-respiration:respiration}{कार्बन
        डाइऑक्साइड} उतारते) $\rightarrow$ बायाँ \omterm{def:g7:heart-and-circulation:rooms}{अलिंद};
  \item \emph{शरीर वाला चक्कर}: बायाँ \omterm{def:g7:heart-and-circulation:rooms}{निलय}
        $\rightarrow$ पूरा शरीर (पहुँचाते,
        बटोरते) $\rightarrow$ दायाँ \omterm{def:g7:heart-and-circulation:rooms}{अलिंद}।
\end{itemize}
हर बूँद बारी-बारी से चक्कर बदलती है: \omterm{def:g7:how-animals-breathe:four}{फेफड़े}, शरीर, \omterm{def:g7:how-animals-breathe:four}{फेफड़े}, शरीर ---
और इसीलिए बायाँ \omterm{def:g7:heart-and-circulation:rooms}{निलय}, जो कहीं बड़े शरीर वाले फंदे भर धकेलता है, \omterm{def:g4:heart-and-blood:heart}{दिल} का
सबसे मोटी दीवार वाला कमरा है।
\end{definition}

\begin{definition}[धमनियाँ, केशिकाएँ, शिराएँ]\label{def:g7:heart-and-circulation:vessels}
हर फंदा तीन तरह की \omterm{def:g4:heart-and-blood:vessels}{वाहिकाओं} से बनता है
(\cref{def:g4:heart-and-blood:vessels}, अब नाम लेकर):
\begin{itemize}
  \item \emph{धमनियाँ}\index{धमनी}: मोटी दीवार वाली,
        लचीली, जो \omterm{prop:g4:journey-of-food:absorb}{रक्त} को दबाव में \omterm{def:g4:heart-and-blood:heart}{दिल} से \emph{दूर} ले
        जाती हैं --- \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी}
        (\cref{prop:g4:heart-and-blood:pulse}) हर लहर पर उनका
        ही तनना है;
  \item \emph{केशिकाएँ}\index{केशिका}: वे
        बाल जितनी महीन \omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ}, जो हर अंग में पिरोई हुई हैं ---
        दीवारें एक \omterm{def:g5:first-look-at-cells:cell}{कोशिका} भर पतली, और वहीं \emph{सारी}
        अदला-बदलियाँ होती हैं: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाएँ}, \omterm{prop:g7:digestion-and-nutrients:absorption}{रसांकुर}, \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ}, \omterm{def:g1:the-five-senses:sense}{त्वचा};
  \item \emph{शिराएँ}\index{शिरा}: चौड़ी, पतली दीवार वाली,
        जो \omterm{prop:g4:journey-of-food:absorb}{रक्त} को कम दबाव पर \omterm{def:g4:heart-and-blood:heart}{दिल} की \emph{ओर} लौटाती हैं, और
        उनके अपने एकतरफ़ा \omterm{def:g7:heart-and-circulation:rooms}{कपाट} भी होते हैं ---
        \cref{rem:g4:heart-and-blood:see} वाली वे नीली-सी लकीरें।
\end{itemize}
\end{definition}

\begin{example}[एक बूँद, एक पूरा दौरा]\label{ex:g7:heart-and-circulation:tour}
\Cref{ex:g4:heart-and-blood:round} वाला पहुँचाने का चक्कर, अब नक़्शे
के नामों के साथ: दायाँ \omterm{def:g7:heart-and-circulation:rooms}{निलय} --- फेफड़ों की \omterm{def:g7:heart-and-circulation:vessels}{केशिकाएँ} (\omterm{def:g7:lungs-and-gas-exchange:alveoli}{कूपिकाओं} पर
\omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} सवार, चटक लाल) --- बायाँ \omterm{def:g7:heart-and-circulation:rooms}{अलिंद}, बायाँ \omterm{def:g7:heart-and-circulation:rooms}{निलय} --- \omterm{def:g7:heart-and-circulation:vessels}{धमनी} --- जाँघ
की किसी \omterm{def:g3:bones-and-muscles:muscle}{पेशी} की \omterm{def:g7:heart-and-circulation:vessels}{केशिकाएँ} (\omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} और \omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़} उतरे, \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड}
चढ़ी, रंग गहरा) --- \omterm{def:g7:heart-and-circulation:vessels}{शिरा}, जहाँ \omterm{def:g7:heart-and-circulation:rooms}{कपाट} चढ़ाई वाले सफ़र को आसान करते हैं
--- दायाँ \omterm{def:g7:heart-and-circulation:rooms}{अलिंद}। दो चक्कर, एक मिनट से भी कम में, और ग़लत मोड़ लेने की
कोई गुंजाइश नहीं: उसका ज़िम्मा \omterm{def:g7:heart-and-circulation:rooms}{कपाटों} का है।
\end{example}

\begin{remark}[केशिकाएँ ही असल बात क्यों हैं]\label{rem:g7:heart-and-circulation:capillaries}
\omterm{def:g7:heart-and-circulation:vessels}{धमनियाँ} और \omterm{def:g7:heart-and-circulation:vessels}{शिराएँ} सिर्फ़ राजमार्ग हैं; असल में पहुँचाना --- यानी इस
साल पढ़ी हर अदला-बदली --- \omterm{def:g7:heart-and-circulation:vessels}{केशिकाओं} की दीवारों के पार ही होता है।
\cref{prop:g7:how-animals-breathe:spec} वाली विशेष-सूची यहाँ भी बसी है:
एक \omterm{def:g5:first-look-at-cells:cell}{कोशिका} भर की दीवारें (पतली), हर ऊतक में दसियों हज़ार किलोमीटर पिरोई
हुई (\cref{def:g4:heart-and-blood:vessels} --- बड़ी), और शरीर की नमी
में नहाई हुई। \omterm{def:g4:heart-and-blood:heart}{दिल} इसीलिए है कि \omterm{prop:g4:journey-of-food:absorb}{रक्त} \emph{\omterm{def:g7:heart-and-circulation:vessels}{केशिकाओं} में से होकर} चलता
रहे; बाक़ी सब वहाँ तक और वहाँ से की नलसाज़ी है।
\end{remark}

\section{जाल को पढ़ना और सँभालना}

\begin{example}[सड़क के किनारे लिए गए माप]\label{ex:g7:heart-and-circulation:measures}
जाल की तीन रोज़मर्रा की पढ़तें: \emph{\omterm{prop:g4:heart-and-blood:pulse}{नाड़ी}} --- यानी \omterm{def:g7:heart-and-circulation:vessels}{धमनी} का तनना,
लहरें गिनते हुए (\cref{met:g4:heart-and-blood:pulse});
\emph{रक्तचाप}, यानी पट्टी वाली दो संख्याएँ --- लहर पर और लहरों के बीच
\omterm{def:g7:heart-and-circulation:vessels}{धमनी} का दबाव; और स्टेथस्कोप की वह \emph{लब-डब} --- यानी \omterm{def:g7:heart-and-circulation:rooms}{कपाटों} के
दरवाज़े बंद होते हुए सुने गए। तीन खिड़कियाँ, और कोई चीरा नहीं।
\end{example}

\begin{proposition}[जाल सँभालना]\label{prop:g7:heart-and-circulation:keep}
इस जाल के लंबी दौड़ के दुश्मन और दोस्त जाने-पहचाने हैं:
\begin{itemize}
  \item \emph{कसरत} पंप को मज़बूत करती है
        (\cref{prop:g7:organs-at-work:training}) और राजमार्गों को
        लचीला रखती है;
  \item \emph{संतुलित खाना}
        (\cref{met:g3:balanced-diet:check}) \omterm{def:g7:heart-and-circulation:vessels}{धमनियों} की दीवारों की
        हिफ़ाज़त करता है, जिन पर हद से ज़्यादा गाढ़ा \omterm{prop:g4:journey-of-food:absorb}{रक्त} धीरे-धीरे
        परत जमाता और उन्हें अकड़ा देता है;
  \item \emph{धुआँ} फेफड़ों के साथ-साथ \omterm{def:g4:heart-and-blood:vessels}{वाहिकाओं} पर भी हमला करता है
        (\cref{prop:g7:lungs-and-gas-exchange:smoke}) --- सिकोड़ते,
        अकड़ाते, और सीटें छीनते हुए;
  \item और \omterm{def:g4:heart-and-blood:heart}{दिल} की अपनी आपूर्ति --- वह अपने ही कमरों के \omterm{prop:g4:journey-of-food:absorb}{रक्त} से नहीं,
        अपनी छोटी \omterm{def:g7:heart-and-circulation:vessels}{धमनियों} से खाता है --- इन तीनों पर टिकी है।
\end{itemize}
\cref{prop:g5:healthy-choices:contract} वाली आदतें ठोस रूप में
\omterm{def:g4:heart-and-blood:vessels}{वाहिकाओं} का रखरखाव ही हैं।
\end{proposition}
\admitted

\begin{method}[किसी भी अंग को जाल में उतारना]\label{met:g7:heart-and-circulation:map}
इस किताब में मिले किसी भी अंग के लिए:
\begin{enumerate}
  \item उसे शरीर वाले चक्कर पर रखो: \omterm{def:g7:heart-and-circulation:vessels}{धमनी} भीतर, \omterm{def:g7:heart-and-circulation:vessels}{शिरा}
        बाहर, और भीतर \omterm{def:g7:heart-and-circulation:vessels}{केशिकाएँ};
  \item गिनाओ कि उसकी \omterm{def:g7:heart-and-circulation:vessels}{केशिकाओं} की दीवारों के पार क्या-क्या जाता और
        आता है (\omterm{prop:g7:digestion-and-nutrients:absorption}{रसांकुर}: \omterm{def:g7:digestion-and-nutrients:families}{पोषक तत्व} भीतर; \omterm{def:g3:bones-and-muscles:muscle}{पेशी}: ईंधन और
        \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} भीतर, \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} बाहर; \omterm{def:g1:the-five-senses:sense}{त्वचा}: ऊष्मा बाहर);
  \item उसका ख़ास जोड़, अगर कोई हो, लिखो (आँत का \omterm{prop:g4:journey-of-food:absorb}{रक्त} पहले \omterm{prop:g7:digestion-and-nutrients:absorption}{यकृत} पर
        रुकता है; \omterm{def:g7:how-animals-breathe:four}{फेफड़े} अपने ही चक्कर पर बैठे हैं);
  \item और बहावों को अंग के काम के सामने रखकर जाँचो --- जाल का
        नक़्शा उस अंग का अध्याय दोबारा सुना देना चाहिए।
\end{enumerate}
\end{method}

\section{समस्या: हृदय-रोग विशेषज्ञ की एक सुबह}

\begin{problem}[{सप्ताहांत समस्या --- चार परामर्श, और जाल का एक
नक़्शा}]\label{pb:g7:heart-and-circulation:1}
हाथ में नक़्शा लेकर, चार (कल्पित) परामर्शों वाली एक सुबह किसी
हृदय-रोग विशेषज्ञ के साथ बिताओ।

\medskip\noindent\textbf{भाग I --- स्वस्थ आधार-रेखा।}
\begin{enumerate}
  \item पहली मरीज़, एक चुस्त किशोरी, की
        \cref{ex:g7:heart-and-circulation:measures} वाली तीनों
        सड़क-किनारे की पढ़तें ली जाती हैं। हर एक का नाम बताओ, और यह
        भी कि हर एक मशीनरी का कौन-सा हिस्सा पढ़ती है।
  \item उसकी आराम वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} $58$ है। विशेषज्ञ मुस्कुरा देती हैं।
        क्यों (\cref{exo:g4:heart-and-blood:11})?
  \item उसकी लब-डब साफ़ और दोहरी है। हर धड़कन में ठीक-ठीक क्या,
        दो बार चटका?
  \item उसके \omterm{prop:g4:journey-of-food:absorb}{रक्त} का रास्ता दाएँ \omterm{def:g7:heart-and-circulation:rooms}{अलिंद} से दोनों चक्करों भर घूमकर
        वापस तक, आठ नामज़द पड़ावों में खींचो।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- तीन तकलीफ़ें।}
\begin{enumerate}[resume]
  \item दूसरे मरीज़ के बाएँ \omterm{def:g7:heart-and-circulation:rooms}{अलिंद} और \omterm{def:g7:heart-and-circulation:rooms}{निलय} के बीच का \omterm{def:g7:heart-and-circulation:rooms}{कपाट} रिसता है:
        हर धड़कन पर कुछ \omterm{prop:g4:journey-of-food:absorb}{रक्त} उलटा उछल जाता है, और साथ में सनसनाती
        घरघराहट। शरीर वाले चक्कर के पहुँचाने और मरीज़ के दमख़म पर
        इसके नतीजों की भविष्यवाणी करो।
  \item तीसरा मरीज़, भारी धूम्रपान करने वाला और गाढ़ा खाने वाला,
        परत जमी और अकड़ी \omterm{def:g7:heart-and-circulation:vessels}{धमनियाँ} रखता है। पहले परामर्श की कौन-सी
        पढ़तें बिगड़ी होंगी, और बायाँ \omterm{def:g7:heart-and-circulation:rooms}{निलय} ओवरटाइम क्यों करता है?
  \item तीसरे मरीज़ की परत उसके \omterm{def:g4:heart-and-blood:heart}{दिल} की अपनी खिलाने वाली \omterm{def:g7:heart-and-circulation:vessels}{धमनियों} में
        भी है। विशेषज्ञ की ख़ास फ़िक्र एक वाक्य में समझाओ।
  \item चौथे मरीज़ के, लंबे-लंबे खड़े रहने वाले दिनों के बाद, टख़ने
        सूज जाते हैं: \omterm{def:g7:heart-and-circulation:vessels}{शिराओं} की चढ़ाई वाली वापसी लड़खड़ा रही है।
        कौन-सी रचनाएँ नाकाम हो रही हैं, और चलना --- यानी पिंडली की
        \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} का \omterm{def:g7:heart-and-circulation:vessels}{शिराओं} को दबाना --- इसमें राहत क्यों देता है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- दीवार का चार्ट।}
\begin{enumerate}[resume]
  \item दवाख़ाने की दीवार का चार्ट वह दोहरा फंदा दिखाता है। इंतज़ार
        करते किसी मरीज़ को समझाओ कि \omterm{prop:g4:journey-of-food:absorb}{रक्त} को शरीर के हर दो दौरों के
        बीच फेफड़ों पर क्यों जाना ही पड़ता है।
  \item \omterm{def:g4:heart-and-blood:vessels}{वाहिकाओं} की सूची जोड़ो: \omterm{def:g7:heart-and-circulation:vessels}{धमनी}, \omterm{def:g7:heart-and-circulation:vessels}{केशिका}, \omterm{def:g7:heart-and-circulation:vessels}{शिरा} --- और हर एक पर
        बनावट, दबाव तथा काम की एक-एक पंक्ति।
  \item चार्ट के नीचे लिखा है: ``हर इलाज \omterm{def:g7:heart-and-circulation:vessels}{केशिकाओं} की सेवा करता
        है।'' इसे \cref{rem:g7:heart-and-circulation:capillaries}
        से सही ठहराओ।
  \item सुबह का समापन उस बचाव वाले नुस्ख़े से करो जो विशेषज्ञ हर
        मरीज़ को थमाती हैं --- तीन आदतें, और हर एक
        \cref{prop:g7:heart-and-circulation:keep} में अपनी
        कार्य-प्रणाली से बँधी हुई।
\end{enumerate}
\end{problem}
