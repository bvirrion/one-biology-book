\chapter{Living Things Change Over Time}\label{ch:g5:evolution}

\omterm{def:g5:evolution:fossil}{Fossils}, family resemblances among species, and the fit between
\omterm{def:g4:ecosystems:levels}{organisms} and their \omterm{def:g4:ecosystems:habitat}{habitats} all hint at a deep story: \omterm{def:g1:living-nonliving:living}{living things}
change over long periods. This chapter introduces \omterm{def:g5:evolution:def}{evolution} gently for
young scientists.

\section{Change across generations}

\begin{definition}[Evolution]\label{def:g5:evolution:def}
\emph{Evolution}\index{evolution} is the change in the \omterm{def:g4:traits:inherited}{inherited traits}
of \omterm{def:g4:ecosystems:levels}{populations} over many generations. Individuals do not evolve in this
\omterm{def:g1:senses:sense}{sense}; \omterm{def:g4:ecosystems:levels}{populations} do.
\end{definition}

\section{Variation and survival}

\begin{definition}[Variation]\label{def:g5:evolution:variation}
\emph{Variation}\index{variation} means differences among individuals
of the same species --- size, colour, speed, disease resistance and
more. Some variation is inherited.
\end{definition}

\begin{definition}[Natural selection (idea)]\label{def:g5:evolution:ns}
\emph{Natural selection}\index{natural selection} is the process in
which individuals with \omterm{def:g4:traits:inherited}{inherited traits} that better fit their
\omterm{def:g4:traits:env}{environment} tend to survive and \omterm{def:g2:what-alive:signs}{reproduce} more, so those \omterm{def:g4:traits:trait}{traits} become
more common over generations.
\end{definition}

\begin{example}[Camouflage moths (story form)]\label{ex:g5:evolution:moth}
If tree bark is dark, better-camouflaged dark moths may be eaten less by
\omterm{def:g2:animal-groups:five}{birds}. Over generations, dark forms can become more common where bark is
dark --- a classic classroom illustration of selection on \omterm{def:g5:evolution:variation}{variation}.
\end{example}

\section{Evidence we can see}

\begin{definition}[Fossil]\label{def:g5:evolution:fossil}
A \emph{fossil}\index{fossil} is a preserved remain or trace of a once-
living \omterm{def:g4:ecosystems:levels}{organism}. Fossils show that life in the past was often different
from life today.
\end{definition}

\begin{example}[Dinosaurs to birds]\label{ex:g5:evolution:dino}
\omterm{def:g5:evolution:fossil}{Fossil} \omterm{def:g3:bones:bone}{skeletons} and living \omterm{def:g2:animal-groups:five}{birds} share features that help scientists
reconstruct how groups are related. Details \omterm{def:g2:what-alive:signs}{grow} richer in later grades.
\end{example}

\begin{proposition}[Adaptation revisited]\label{prop:g5:evolution:adapt}
\omterm{def:g4:adaptations:def}{Adaptations} (Grade~4) make \omterm{def:g1:senses:sense}{sense} as outcomes of long-term selection:
features that helped ancestors survive were passed on more often.
\end{proposition}
\admitted

\section{What evolution is not}

\begin{method}[Talking carefully about change]\label{met:g5:evolution:talk}
\begin{enumerate}
  \item Speak about \omterm{def:g4:ecosystems:levels}{populations} and generations, not one \omterm{def:g1:plants-animals:animal}{animal}
        deciding.
  \item Link \omterm{def:g4:traits:trait}{traits} to survival or \omterm{def:g5:reproduction:def}{reproduction} in an \omterm{def:g4:traits:env}{environment}.
  \item Use evidence: \omterm{def:g5:evolution:fossil}{fossils}, similarities, experiments on short-lived
        species.
  \item Stay curious: science revises details as new evidence arrives.
\end{enumerate}
\end{method}
