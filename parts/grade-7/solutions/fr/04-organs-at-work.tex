\section*{Chapitre \ref{ch:g7:organs-at-work} --- Les organes au travail}

\begin{solution}{exo:g7:organs-at-work:1}
Davantage de \omterm{def:g7:what-is-respiration:respiration}{dioxygène} et de \omterm{prop:g7:organs-at-work:bill}{glucose} prélevés dans le \omterm{prop:g4:journey-of-food:absorb}{sang}~;
davantage de \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} et de chaleur rejetés.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:2}
Le \omterm{def:g4:journey-of-food:digestion}{nutriment} sucré tout prêt du \omterm{prop:g4:journey-of-food:absorb}{sang} --- le carburant de tous les
jours de la combustion des \omterm{def:g5:first-look-at-cells:cell}{cellules}. Le \omterm{def:g3:bones-and-muscles:muscle}{muscle} le prélève dans le
\omterm{prop:g4:journey-of-food:absorb}{sang} qui le traverse (approvisionné, plus en amont, par les
livraisons de la \omterm{def:g4:journey-of-food:digestion}{digestion}).
\end{solution}

\begin{solution}{exo:g7:organs-at-work:3}
On compare le \omterm{prop:g4:journey-of-food:absorb}{sang} qui entre dans le \omterm{def:g3:bones-and-muscles:muscle}{muscle} et le \omterm{prop:g4:journey-of-food:absorb}{sang} qui en sort.
Au repos, le \omterm{prop:g4:journey-of-food:absorb}{sang} sortant n'a cédé qu'une part modeste de \omterm{def:g7:what-is-respiration:respiration}{dioxygène}
et de \omterm{prop:g7:organs-at-work:bill}{glucose}~; lors d'un travail intense, il ressort dépouillé de
l'essentiel de son \omterm{def:g7:what-is-respiration:respiration}{dioxygène} livrable, et bien plus riche en \omterm{def:g7:what-is-respiration:respiration}{dioxyde
de carbone}. La différence est la facture du \omterm{def:g3:bones-and-muscles:muscle}{muscle}.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:4}
La \omterm{def:g7:what-is-respiration:respiration}{ventilation} s'approfondit et s'accélère~; le \omterm{def:g4:heart-and-blood:heart}{cœur} bat plus vite
et plus fort~; le \omterm{prop:g4:journey-of-food:absorb}{sang} est redirigé --- les vaisseaux des \omterm{def:g3:bones-and-muscles:muscle}{muscles} au
travail et de la \omterm{def:g1:the-five-senses:sense}{peau} qui refroidit s'élargissent tandis que ceux du
\omterm{def:g4:journey-of-food:tract}{tube digestif} au repos se rétrécissent.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:5}
Brûler du carburant dégage toujours de la chaleur, et le \omterm{def:g3:bones-and-muscles:muscle}{muscle} au
travail est la chaufferie du corps. Frissonner, ce sont des \omterm{def:g3:bones-and-muscles:muscle}{muscles}
mis au travail pour ne produire rien d'autre que cette chaleur ---
un chauffage à la demande.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:6}
Le retour progressif du \omterm{prop:g4:heart-and-blood:pulse}{pouls} et de la \omterm{def:g7:what-is-respiration:respiration}{ventilation} après la fin de
l'effort. Le corps recharge ses réserves de \omterm{def:g7:what-is-respiration:respiration}{dioxygène}, évacue les
déchets de l'effort et se débarrasse de la chaleur accumulée.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:7}
Repos autour de $72$~; plateau près de $160$~; \omterm{def:g7:organs-at-work:recovery}{récupération} de cinq
à six minutes environ sur le graphique. L'entraînement abaisse la
valeur de repos, abaisse le plateau pour le même effort, et raccourcit
la \omterm{def:g7:organs-at-work:recovery}{récupération}.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:8}
Les deux sont des effets de réacheminement. Après le déjeuner, le
\omterm{def:g4:journey-of-food:tract}{tube digestif} retient une grande part du \omterm{prop:g4:journey-of-food:absorb}{sang}, si bien que le reste
du corps tourne au ralenti~; sprinter oblige alors les deux étages à
se disputer les livraisons --- d'où le point de côté et la vieille
règle contre la course le ventre plein.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:9}
Les vaisseaux élargis de la \omterm{def:g1:the-five-senses:sense}{peau} sont l'expédition de la chaleur de
la chaufferie vers la surface radiateur du corps~: une livraison de
chaleur vers l'extérieur, qui résout le problème de surchauffe des
\omterm{def:g3:bones-and-muscles:muscle}{muscles} au travail. La rougeur est le refroidissement qui
fonctionne, pas qui échoue.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:10}
Le \omterm{def:g4:heart-and-blood:heart}{cœur} déplace plus de \omterm{prop:g4:journey-of-food:absorb}{sang} à chaque battement --- signe
quotidien~: un \omterm{prop:g4:heart-and-blood:pulse}{pouls} de repos plus bas. Les \omterm{def:g3:bones-and-muscles:muscle}{muscles} grossissent et
stockent plus de carburant --- la même \omterm{def:g3:bones-and-muscles:skeleton}{côte} paraît plus facile. Les
fins vaisseaux du réseau se multiplient et la \omterm{def:g7:what-is-respiration:respiration}{ventilation} gagne en
ampleur --- la parole reste possible à l'effort, et la \omterm{def:g7:organs-at-work:recovery}{récupération}
raccourcit.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:11}
La machine à $68$ de repos et trois minutes est l'entraînée. Le
\omterm{prop:g4:heart-and-blood:pulse}{pouls} de repos certifie un \omterm{def:g4:heart-and-blood:heart}{cœur} qui déplace davantage à chaque
battement~; le temps de \omterm{def:g7:organs-at-work:recovery}{récupération} certifie des lignes
d'approvisionnement qui soldent vite les dettes de l'effort.
Ensemble, ce sont les deux bouts de la reconstruction de
\cref{prop:g7:organs-at-work:training}.
\end{solution}

\begin{solution}{exo:g7:organs-at-work:12}
Dans un sprint, les \omterm{def:g5:first-look-at-cells:cell}{cellules} des \omterm{def:g3:bones-and-muscles:muscle}{muscles} des \omterm{def:g1:my-body:parts}{jambes} brûlent du
\omterm{prop:g7:organs-at-work:bill}{glucose} avec du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} pour alimenter la contraction --- le but
de la \omterm{def:g7:what-is-respiration:respiration}{respiration} --- et leur facture (plus de carburant et de
\omterm{def:g7:what-is-respiration:respiration}{dioxygène} en entrée~; plus de \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} et de chaleur en
sortie) doit être payée sur-le-champ. Le \omterm{prop:g4:journey-of-food:absorb}{sang} est le service
payeur~: chargé aux \omterm{def:g7:how-animals-breathe:four}{poumons} et à l'intestin, poussé plus vite par le
\omterm{def:g4:heart-and-blood:heart}{cœur} emballé, réacheminé vers les \omterm{def:g1:my-body:parts}{jambes}, il livre le carburant et
le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} et emporte le \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} --- un sprint, trois
chapitres, un seul et même compte.
\end{solution}

\begin{solution}{pb:g7:organs-at-work:1}
\textbf{1.} Son prélèvement de \omterm{def:g7:what-is-respiration:respiration}{dioxygène} et de \omterm{prop:g7:organs-at-work:bill}{glucose} dans le \omterm{prop:g4:journey-of-food:absorb}{sang}
qui passe se multiplie~; son rejet de \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} et sa
production de chaleur montent d'autant --- la facture du travail
intense.

\textbf{2.} Parce que ce sont deux étapes d'une même ligne
d'approvisionnement~: la \omterm{def:g7:what-is-respiration:respiration}{ventilation} recharge le \omterm{prop:g4:journey-of-food:absorb}{sang} aux \omterm{def:g7:how-animals-breathe:four}{poumons},
le \omterm{def:g4:heart-and-blood:heart}{cœur} porte ce \omterm{prop:g4:journey-of-food:absorb}{sang} rechargé jusqu'aux \omterm{def:g3:bones-and-muscles:muscle}{muscles}. La demande
augmentée à un bout accélère le rythme des deux.

\textbf{3.} Les \omterm{def:g3:bones-and-muscles:muscle}{muscles} au travail sont des chaufferies~; leur
chaleur doit sortir. Le \omterm{prop:g4:journey-of-food:absorb}{sang} la porte jusqu'à la \omterm{def:g1:the-five-senses:sense}{peau}, dont les
vaisseaux élargis rougissent et rayonnent --- la chaufferie à
l'intérieur, le radiateur à la surface.

\textbf{4.} Parce que des dettes restent~: des réserves de \omterm{def:g7:what-is-respiration:respiration}{dioxygène}
à recharger, des déchets à évacuer, de la chaleur à évacuer. Le
\omterm{prop:g4:heart-and-blood:pulse}{pouls} ne baisse qu'à mesure que les comptes se soldent --- cette
baisse, c'est la \omterm{def:g7:organs-at-work:recovery}{récupération}.

\textbf{5.} Repos $84 \to 72$~: un \omterm{def:g4:heart-and-blood:heart}{cœur} rebâti pour déplacer
davantage à chaque battement. Effort $168 \to 150$~: les mêmes
montées coûtent maintenant moins --- des \omterm{def:g3:bones-and-muscles:muscle}{muscles} plus forts, un
réseau de fins vaisseaux plus riche. \omterm{def:g7:organs-at-work:recovery}{Récupération} $9 \to 5$
minutes~: toute la chaîne solde ses dettes plus vite. C'est
l'exercice du trimestre qui a fait la reconstruction.

\textbf{6.} Ou bien B a commencé le trimestre déjà entraîné --- des
nombres proches de leur plancher d'entraîné bougent peu --- ou bien
B a sauté l'exercice du trimestre et n'a fait que conserver une
machine non entraînée mais stable. Le fait qui tranche~: l'activité
de B hors de la classe (du sport ailleurs confirmerait la première
lecture~; aucun sport, la seconde --- et les bons nombres de B en
semaine 1 penchent déjà de ce côté).

\textbf{7.} La règle du réacheminement~: après un repas, le \omterm{def:g4:journey-of-food:tract}{tube
digestif} accapare le \omterm{prop:g4:journey-of-food:absorb}{sang}, si bien que le \omterm{prop:g4:heart-and-blood:pulse}{pouls} de repos est élevé
et la tolérance à l'effort basse. Protocole~: mesurer avant le
déjeuner, ou deux bonnes heures après --- toujours au même moment.

\textbf{8.} L'expérience honnête~: seules des mesures prises de la
même façon se comparent, d'une semaine à l'autre et d'un élève à
l'autre.

\textbf{9.} «~Douze semaines ont relevé le plafond de A~: le même
effort coûte maintenant $150$ battements au lieu de $168$, à partir
d'un \omterm{prop:g4:heart-and-blood:pulse}{pouls} de repos inférieur de huit battements. Et la machine de A
solde ses dettes en cinq minutes là où septembre en demandait neuf
--- la \omterm{def:g7:organs-at-work:recovery}{récupération} est le plus net de tous les gains.~»

\textbf{10.} D'ordinaire, c'est le contraire d'inquiétant~: un \omterm{def:g4:heart-and-blood:heart}{cœur}
entraîné déplace plus de \omterm{prop:g4:journey-of-food:absorb}{sang} à chaque battement et tourne donc bas
au repos --- les $50$ tranquilles du cycliste faisant le travail des
$75$ d'un non-entraîné. (Toute inquiétude relève d'un médecin, mais
bas et en forme est l'histoire la plus courante.)

\textbf{11.} Le temps de \omterm{def:g7:organs-at-work:recovery}{récupération} mesure toute la chaîne ---
\omterm{def:g4:heart-and-blood:heart}{cœur}, \omterm{def:g7:how-animals-breathe:four}{poumons}, vaisseaux, \omterm{def:g3:bones-and-muscles:muscle}{muscles} --- en train de solder des dettes
réelles, et il s'améliore régulièrement et visiblement avec
l'entraînement~; le \omterm{prop:g4:heart-and-blood:pulse}{pouls} d'effort dépend davantage de l'effort
exact du jour et de l'humeur. Comme nombre unique à suivre, la
\omterm{def:g7:organs-at-work:recovery}{récupération} est le résumé le plus honnête.

\textbf{12.} Son propre usage~: l'effort, répété doucement et
régulièrement, est ce qui a rebâti la machine qui l'accomplit.
\end{solution}
