\section*{Bab \ref{ch:b1:biosyntheses-integration} --- Biosintesis dan Sel yang Terpadu}

\begin{solution}{exo:b1:biosyntheses-integration:1}
Kerangka karbon (dari lintasan pusatnya), daya reduksi (\omterm{def:b1:biosyntheses-integration:ppp}{NADPH}), energi
(\omterm{def:b1:water-small-molecules:atp}{ATP}). \omterm{def:b1:biosyntheses-integration:ppp}{Lintasan pentosa fosfat}.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:2}
Heksokinase, dipintasi glukosa-6-fosfatase; fosfofruktokinase, oleh
fruktosa-1,6-bisfosfatase; piruvat kinase, oleh piruvat karboksilase
ditambah PEP karboksikinase.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:3}
Puncaknya sekitar \qty{7.5}{mmol/L} sejam sesudah santapannya; kembali
ke \qty{5}{} sesudah sekitar tiga jam; terendahnya sekitar
\qty{4.5}{mmol/L} sebelum sarapan.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:4}
Bijinya memiliki daur glioksilat, yang mengubah dua \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA} menjadi
satu asam berkarbon empat tanpa kehilangan karbon sebagai
$\mathrm{CO_2}$, lalu dari asam itu membuat gula. Hewan tidak memiliki
kedua \omterm{def:b1:enzymes:enzyme}{enzim} jalan pintas itu: \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA-nya} masuk ke \omterm{def:b1:respiration-fermentation:krebs}{daur Krebs} dan
kedua karbonnya pergi sebagai $\mathrm{CO_2}$ sebelum ada oksaloasetat
bersih yang muncul.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:5}
Enam setara \omterm{def:b1:water-small-molecules:atp}{ATP} (dua karboksilasi, dua GTP, dua \omterm{def:b1:water-small-molecules:atp}{ATP} pada langkah
fosfogliseratnya) tiap glukosa. Bersih tiap daurnya: $2 - 6 = -4$ \omterm{def:b1:water-small-molecules:atp}{ATP}.
Hati yang membayar, dengan oksigen, sesudah lari cepatnya --- ototnya
meminjam \omterm{def:b1:water-small-molecules:atp}{ATP} yang tak sanggup dibuatnya secara aerob pada waktunya.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:6}
Tiap glukosa memberikan 2 \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA}: 4 glukosa. Tiap glukosa lewat
cabang oksidatifnya memberikan 2 \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}: 7 glukosa. Sebelas glukosa bagi
satu palmitat --- empat memasok karbon dan tujuh memasok elektron.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:7}
\omterm{def:b1:enzymes:enzyme}{Enzim} yang berbeda membuat tiap arahnya dapat dinyalakan atau
dipadamkan sendiri-sendiri; ruang yang berbeda (\omterm{def:b1:eukaryotic-cell:organelle}{sitosol} bagi
sintesisnya, \omterm{def:b1:eukaryotic-cell:mitochondrion}{matriks} bagi oksidasinya) memisahkan kedua lubuk zat
antaranya; \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} dijaga tereduksi bagi sintesisnya sedangkan \omterm{def:b1:respiration-fermentation:carriers}{NAD} dijaga
teroksidasi bagi penguraiannya, sehingga tiap arahnya lebih disukai
secara termodinamika di tempatnya sendiri. Malonil-KoA, zat antara
pertama yang menentukan sintesisnya, menghambat ulang alik karnitin
yang membawa \omterm{def:b1:lipids:fattyacid}{asam lemak} ke dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria}: selama \omterm{def:b1:lipids:triglyceride}{lemak} sedang
dibuat, tak ada yang dibakar.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:8}
Alanina $+ \alpha$-ketoglutarat $\rightleftharpoons$ piruvat $+$
\omterm{def:b1:biosyntheses-integration:nitrogen}{glutamat}. Sesudah santapan berprotein, kelebihan alanina menyerahkan
nitrogennya kepada \omterm{def:b1:biosyntheses-integration:nitrogen}{glutamat} (ke kanan), untuk dibuang sebagai urea dan
karbonnya bagi bahan bakar atau glukosa. Dalam puasa, \omterm{def:b1:proteins:peptide}{protein} otot
diuraikan dan gugus aminonya dikumpulkan pada piruvat sebagai alanina
(ke kiri dalam otot), dikirim ke hati, lalu di sana diubah kembali
menjadi piruvat dan glukosa.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:9}
Hatinya tak sanggup melepas glukosa bebas: glukosa darahnya turun
berbahaya di antara waktu makan (hipoglikemia). Glukosa-6-fosfatnya
menumpuk lalu dialihkan ke dalam \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}, yang tak sanggup digerakkan
hatinya ke darah: hatinya membesar oleh \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}. Kelebihan
glukosa-6-fosfat itu juga menuruni \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis} menjadi laktat, yang naik
dalam darahnya.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:10}
$120/0.55 = \qty{218}{g}$ \omterm{def:b1:proteins:peptide}{protein} tiap hari, sekitar \qty{1.1}{kg} otot.
Dengan badan keton, $40/0.55 = \qty{73}{g}$ \omterm{def:b1:proteins:peptide}{protein}, \qty{360}{g} otot:
kehilangannya turun dua pertiga, dan ketahanan hidup pada simpanan
\omterm{def:b1:lipids:triglyceride}{lemaknya} diperpanjang dari beberapa minggu menjadi beberapa bulan.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:11}
Untaian fosforilasi yang sama mengaktifkan yang satu dan menonaktifkan
yang lain, sehingga isyarat yang membuka satu katup menutup yang lain.
Seandainya keduanya aktif, tiap glukosa yang ditambahkan (satu UTP,
setara satu \omterm{def:b1:water-small-molecules:atp}{ATP}) dan disingkirkan (gratis) akan memakan biaya satu \omterm{def:b1:water-small-molecules:atp}{ATP}
tiap putaran dan tidak menghasilkan apa-apa selain kalor. Lebah madu
menjalankan justru daur semacam itu (pada fruktosa-6-fosfat) dalam otot
terbangnya sebelum lepas landas pada pagi yang dingin, dengan membakar
\omterm{def:b1:water-small-molecules:atp}{ATP} sebagai pemanas sampai ototnya cukup hangat untuk terbang.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:12}
Tiap \omterm{def:b1:enzymes:enzyme}{enzim} persimpangan hanya menanggapi molekul di sekelilingnya ---
\omterm{def:b1:water-small-molecules:atp}{ATP}, AMP, sitrat, \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA}, sebuah fosfat yang ditaruh sebuah kinase
padanya --- lalu membuka atau menutup sesuai dengan itu; \omterm{def:b1:enzymes:enzyme}{enzim} yang
terpisah bagi kedua arahnya berarti membuka satu arah dapat menutup
yang lain. Tak ada \omterm{def:b1:enzymes:enzyme}{enzim} yang tahu tentang sarapan atau otaknya; namun
jumlah tanggapan setempat itu adalah bahwa hati menimbun sesudah
santapan, melepas pada malam hari, menghemat \omterm{def:b1:proteins:peptide}{protein} dalam puasa:
sebuah siasat yang taat asas, tanpa ahli siasat. ``Rencananya'' adalah
susunan kabel katupnya, yang dipilih seleksi, dan hormon yang menyetel
banyak katup sekaligus itulah yang membuat \omterm{def:b1:organism-environment:organism}{organismenya} bertindak
sebagai satu.
\end{solution}

\begin{solution}{pb:b1:biosyntheses-integration:1}
\textbf{1.} \omterm{def:b1:carbohydrates:polysaccharide}{Glikogen} hati $100\times 17 = \qty{1.7}{MJ}$, 0.2 hari;
\omterm{def:b1:carbohydrates:polysaccharide}{glikogen} otot \qty{6.8}{MJ}, 0.8 hari; \omterm{def:b1:lipids:triglyceride}{lemak} $12\,000\times 38 =
\qty{456}{MJ}$, 54 hari; \omterm{def:b1:proteins:peptide}{protein} \qty{170}{MJ}, 20 hari (tak pernah
seluruhnya dapat dipakai).
\textbf{2.} \omterm{def:b1:carbohydrates:polysaccharide}{Glikogen} hati: hatinya memiliki glukosa-6-fosfatase.
Otot tidak memilikinya, sehingga \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya} hanya menghasilkan
glukosa-6-fosfat bagi \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisisnya} sendiri.
\textbf{3.} $120 + 40 = \qty{160}{g}$ tiap hari.
\textbf{4.} $0.005\times 5\times 180 = \qty{4.5}{g}$: pada \qty{5}{g/h}, sekitar \qty{54}{min}.
\textbf{5.} $120\times 17 = \qty{2.0}{MJ}$, seperempat dari \qty{8.4}{MJ}.
\textbf{6.} Pengubahannya menjadi \omterm{def:b1:lipids:fattyacid}{asam lemak} dan \omterm{def:b1:lipids:triglyceride}{trigliserida} dalam
hatinya, yang diekspor ke \omterm{def:b1:body-plans-tissues:tissue}{jaringan} \omterm{def:b1:lipids:triglyceride}{lemak}; insulin.
\textbf{7.} $160/24 = \qty{6.7}{g/h}$.
\textbf{8.} $100/6.7 = \qty{15}{h}$.
\textbf{9.} \omterm{def:b1:carbohydrates:polysaccharide}{Glikogennya} memasok dua pertiga, \qty{4.4}{g/h}: $100/4.4 =
\qty{22}{h}$.
\textbf{10.} Pukul 11:00 pagi berikutnya menurut taksiran pertama,
18:00 menurut yang kedua --- sepanjang malam pada kedua halnya, dengan
selisihnya bergantung pada seberapa dini \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesisnya} mengambil
alih.
\textbf{11.} Pengeluaran istirahat selama \qty{12}{h}: \qty{4.2}{MJ};
\qty{60}{\%}-nya \qty{2.5}{MJ}; $2500/38 = \qty{66}{g}$ \omterm{def:b1:lipids:triglyceride}{lemak}.
\textbf{12.} $160/0.55 = \qty{290}{g}$ \omterm{def:b1:proteins:peptide}{protein}, yakni \qty{1.45}{kg}
otot sehari.
\textbf{13.} $160/180 = \qty{0.89}{mol}$ glukosa; pada 6 \omterm{def:b1:water-small-molecules:atp}{ATP}
masing-masing itu \qty{5.3}{mol} \omterm{def:b1:water-small-molecules:atp}{ATP}, dan pada \qty{50}{kJ/mol} sekitar
\qty{265}{kJ} --- sekitar \qty{3}{\%} dari \qty{8.4}{MJ} sehari:
membuat glukosanya murah, dan yang kurang pada tubuhnya adalah kerangka
karbonnya, bukan energinya.
\textbf{14.} Pengeluaran bukan glukosanya $8.4 - 160\times 0.017 =
\qty{5.7}{MJ}$: $5700/38 = \qty{150}{g}$ \omterm{def:b1:lipids:triglyceride}{lemak}, yang gliserolnya
memberikan \qty{15}{g} glukosa.
\textbf{15.} $(160 - 15)/0.55 = \qty{264}{g}$ \omterm{def:b1:proteins:peptide}{protein}, \qty{1.3}{kg}
otot sehari.
\textbf{16.} \qty{3}{kg} \omterm{def:b1:proteins:peptide}{protein} otot pada \qty{264}{g} sehari: sekitar
\qty{11}{d}.
\textbf{17.} $(80 - 15)/0.55 = \qty{118}{g}$ \omterm{def:b1:proteins:peptide}{protein}, \qty{0.6}{kg}
otot sehari; separuh ototnya dalam \qty{25}{d} --- dan dalam praktik
kehilangannya turun lebih jauh, ke \qtyrange{20}{30}{g} sehari, seiring
tubuhnya berhemat.
\textbf{18.} $456/8.4 = 54$ hari; pada \qty{6.5}{MJ}, 70 hari.
\textbf{19.} \omterm{def:b1:proteins:peptide}{Protein} bukanlah sebuah simpanan melainkan mesinnya:
kehilangan sepertiganya (otot pernapasan, jantung, \omterm{def:b1:proteins:peptide}{protein} kekebalan)
bersifat fatal, dan pada \qty{100}{g} sehari atau lebih hal itu terjadi
dalam beberapa minggu, sedangkan \omterm{def:b1:lipids:triglyceride}{lemaknya} akan bertahan dua bulan.
Badan keton membuat otaknya dapat berjalan pada \omterm{def:b1:lipids:triglyceride}{lemak}, memangkas
permintaan glukosanya dan dengan itu kehilangan \omterm{def:b1:proteins:peptide}{proteinnya} menjadi
sepertiga, serta merentangkan ketahanan hidupnya ke arah batas yang
ditetapkan \omterm{def:b1:lipids:triglyceride}{lemaknya}.
\textbf{20.} \omterm{def:b1:carbohydrates:polysaccharide}{Glikogen} fosforilase aktif pada pukul 03:00, \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}
sintase pada pukul 09:00.
\textbf{21.} Glukagon (03:00) dan insulin (09:00); fosforilasi kedua
\omterm{def:b1:enzymes:enzyme}{enzimnya} oleh untaian kinase, yang dibalik sebuah fosfatase.
\textbf{22.} \omterm{def:b1:respiration-fermentation:pdh}{Asetil-KoA} mengaktifkan piruvat karboksilase; \omterm{def:b1:water-small-molecules:atp}{ATP} (dan
sitrat) menghambat fosfofruktokinase.
\textbf{23.} Glukosa menjadi piruvat menghasilkan 2 \omterm{def:b1:water-small-molecules:atp}{ATP}; piruvat
kembali menjadi glukosa memakan biaya 6: tiap perjalanan pulang pergi
membakar 4 \omterm{def:b1:water-small-molecules:atp}{ATP} tanpa hasil. Dengan \omterm{def:b1:enzymes:enzyme}{enzim} yang berbeda pada ketiga
pintasannya, isyarat yang sama (\omterm{def:b1:water-small-molecules:atp}{ATP}, \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA}, fosforilasi)
mengaktifkan satu perangkat dan menghambat yang lain, sehingga hanya
satu arah yang terbuka.
\textbf{24.} Tanpa insulin katupnya disetel seperti dalam puasa apa pun
glukosanya: fosforilase, \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis} dan lipolisisnya menyala,
sintesisnya padam, dan hatinya, yang ``mempercayai'' tubuhnya
kelaparan, mencurahkan glukosa serta keton ke dalam darah yang sudah
penuh glukosa yang tak sanggup diserap \omterm{def:b1:body-plans-tissues:tissue}{jaringan} yang bergantung
insulin.
\textbf{25.} Sekitar \qty{15}{h} tanpa \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis}, \qty{22}{h}
dengannya --- semalam dan sepagi; kehilangan \omterm{def:b1:proteins:peptide}{proteinnya} sekitar
\qty{260}{g} sehari mula-mula dan \qty{120}{g} (yang turun lebih jauh)
begitu badan keton memberi makan otaknya; simpanan \omterm{def:b1:lipids:triglyceride}{lemaknya} bertahan
sekitar \qtyrange{55}{70}{d}.
\end{solution}
