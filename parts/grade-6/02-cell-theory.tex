\chapter{Cell Theory}\label{ch:g6:cell-theory}

Three big claims organise everything we know about \omterm{def:g6:cell-theory:cell}{cells}. This chapter
states \omterm{def:g6:cell-theory:cell}{cell} theory, \omterm{def:g1:caring:needs}{places} it in history, and explores what it means for
how life \omterm{def:g2:what-alive:signs}{grows} and repairs itself.

\section{The three claims}

\begin{definition}[Cell]\label{def:g6:cell-theory:cell}
A \emph{cell}\index{cell} is the basic membrane-bounded unit of structure
and function in living \omterm{def:g4:ecosystems:levels}{organisms}.
\end{definition}
\omimg{animal-cell.jpg}{0.50\linewidth}{All animals and plants are made of cells.}


\begin{proposition}[Cell theory]\label{prop:g6:cell-theory:three}
\begin{enumerate}
  \item All living \omterm{def:g4:ecosystems:levels}{organisms} are made of one or more \omterm{def:g6:cell-theory:cell}{cells}.
  \item The \omterm{def:g6:cell-theory:cell}{cell} is the basic unit of structure and function in
        \omterm{def:g4:ecosystems:levels}{organisms}.
  \item All \omterm{def:g6:cell-theory:cell}{cells} arise from pre-existing \omterm{def:g6:cell-theory:cell}{cells} by division.
\end{enumerate}
\end{proposition}
\admitted

\omimg{microscope.jpg}{0.52\linewidth}{\omterm{prop:g4:cells:building}{Microscopes} made \omterm{def:g6:cell-theory:cell}{cell} theory
possible by revealing life's microscopic units.}

\section{Why the third claim matters}

\begin{proposition}[No spontaneous modern cells]\label{prop:g6:cell-theory:from-cells}
Under conditions we study in ordinary \omterm{def:g6:method:science}{biology}, new \omterm{def:g6:cell-theory:cell}{cells} come from
division of existing \omterm{def:g6:cell-theory:cell}{cells}. \omterm{def:g2:growing:growth}{Growth}, healing and \omterm{def:g5:reproduction:def}{reproduction} of
cellular life depend on this fact.
\end{proposition}
\admitted

\begin{example}[Skin repair]\label{ex:g6:cell-theory:skin}
A cut heals as nearby \omterm{def:g6:cell-theory:cell}{cells} divide and fill the gap. The new \omterm{def:g6:cell-theory:cell}{cells} are
descendants of \omterm{def:g6:cell-theory:cell}{cells} already in the \omterm{def:g1:my-body:body}{body}.
\end{example}

\section{One cell or many}

\begin{definition}[Unicellular and multicellular]\label{def:g6:cell-theory:uni-multi}
A \emph{unicellular}\index{unicellular} \omterm{def:g4:ecosystems:levels}{organism} is a single \omterm{def:g6:cell-theory:cell}{cell} that
carries out all life functions. A \emph{multicellular}\index{multicellular}
\omterm{def:g4:ecosystems:levels}{organism} has many \omterm{def:g6:cell-theory:cell}{cells}, often specialised and organised into tissues.
\end{definition}



\begin{example}[Both fit the theory]\label{ex:g6:cell-theory:both}
A bacterium is one \omterm{def:g6:cell-theory:cell}{cell} --- and alive. A whale has enormous numbers of
\omterm{def:g6:cell-theory:cell}{cells} --- and is still built from \omterm{def:g6:cell-theory:cell}{cells} as units.
\end{example}

\section{Borderline cases}

\begin{method}[Using cell theory as a checklist]\label{met:g6:cell-theory:check}
\begin{enumerate}
  \item Is the specimen made of \omterm{def:g6:cell-theory:cell}{cells} (or a single \omterm{def:g6:cell-theory:cell}{cell})?
  \item Can you link structures you see to functions of life?
  \item Where did the \omterm{def:g6:cell-theory:cell}{cells} come from? (division / parent \omterm{def:g4:ecosystems:levels}{organism})
  \item If something lacks \omterm{def:g6:cell-theory:cell}{cell} structure, is it a full \omterm{def:g4:ecosystems:levels}{organism}?
\end{enumerate}
\end{method}
