\chapter{النظام الغذائي المتوازن}\label{ch:g3:balanced-diet}

ورشة البنّاء تحتاج إلى قرميد وإسمنت وألواح --- وإلى وقود
لآلاتها. وجسمك ورشة لا تغلق أبدًا: فهو ينمو، ويصلح نفسه، ويجري
طول النهار. وتصل توريداته ثلاث مرات أو أربعًا في اليوم، على صحن.
فما الذي يجب أن تحتويه التوريدات؟ ذلك هو سؤال النظام الغذائي.

\section{لماذا يحتاج الجسم إلى الغذاء}

\begin{proposition}[عملا الغذاء]\label{prop:g3:balanced-diet:jobs}
للغذاء عملان مختلفان:
\begin{enumerate}
  \item يعطي \emph{الطاقة}\index{طاقة (من الغذاء)} --- وهي القدرة
        على الحركة، وعلى البقاء دافئًا، وعلى إبقاء القلب والدماغ
        يعملان، حتى في أثناء \omterm{prop:g1:healthy-habits:needs}{النوم}؛
  \item ويعطي \emph{مادة البناء} --- وهي ما يستعمله الجسم ليطول،
        وليبني العضل \omterm{def:g3:bones-and-muscles:skeleton}{والعظم}، وليصلح الخدوش ويعوّض الأجزاء
        البالية.
\end{enumerate}
وعلى النظام الغذائي أن يوصّل الاثنين، كل يوم.
\end{proposition}
\admitted

\begin{example}[وقود حتى في الراحة]\label{ex:g3:balanced-diet:rest}
حتى وأنت مستلقٍ ساكن تمامًا تنفق \omterm{prop:g3:balanced-diet:jobs}{طاقة}: فالقلب ينبض، والصدر يعلو
ويهبط، والجسم يبقى دافئًا عند نحو
$\qty{37}{\degreeCelsius}$ مهما يكن الطقس. وتلك النفقة الثابتة
هي سبب ما يورثه تفويت الوجبات من ضعف وبرد، لا مجرد جوع.
\end{example}

\section{الأسر الغذائية}

\begin{definition}[الأسر الغذائية]\label{def:g3:balanced-diet:families}
تُفرز الأطعمة في \emph{أسر غذائية}\index{أسرة غذائية} بحسب ما
تجلبه:
\begin{itemize}
  \item \emph{الفواكه والخضر} --- فيتامينات وألياف وماء؛ وهي أسرة
        الطزاجة؛
  \item و\emph{الحبوب والنشويات} (الخبز، والمعكرونة، والأرز،
        والبطاطا) --- وهي أسرة \omterm{prop:g3:balanced-diet:jobs}{الطاقة} الثابتة؛
  \item و\emph{الألبان} (\omterm{def:g2:animals-grow-up:born}{الحليب}، واللبن، والجبن) --- وهي أسرة بناء
        \omterm{def:g3:bones-and-muscles:skeleton}{العظام}؛
  \item و\emph{اللحم والسمك \omterm{def:g2:animals-grow-up:born}{والبيض}} --- وهي أسرة بناء الجسم؛
  \item و\emph{الدهون} (الزبدة، والزيت) --- \omterm{prop:g3:balanced-diet:jobs}{طاقة} مركزة، يُحتاج
        إليها بمقادير صغيرة؛
  \item و\emph{السكريات} --- متعة سريعة، لا يُحتاج إليها البتة:
        وهي أسرة الحلوى؛
  \item و\emph{الماء}\index{ماء (شراب)} --- وهو الشراب الوحيد الذي
        يطلبه الجسم، طول النهار.
\end{itemize}
\end{definition}

\begin{example}[موازنة غداء]\label{ex:g3:balanced-diet:lunch}
معكرونة بصلصة الطماطم، وقطعة سمك، وفاصولياء خضراء، ولبن، وتفاحة،
وماء: كل أسرة حاضرة، والحلوى غائبة --- غداء متوازن. أما البطاطا
المقلية والمشروب الغازي ولوح الشوكولاتة فتغذّي أسرتين لا غير
(الدهون والسكر) وتتخطى الأسر التي كانت الورشة تنتظرها.
\end{example}

\begin{remark}[لا أطعمة محرّمة، بل أطعمة منسية]\label{rem:g3:balanced-diet:balance}
التوازن يُقاس على أيام لا على لقمات مفردة. فقطعة من كعكة عيد
الميلاد لا تخرق قاعدة --- والمشكلة ليست حلوى واحدة قط، بل الحلوى
\emph{بدلًا من} الأسر، يومًا بعد يوم. والسؤال العملي عند كل وجبة
ليس ``أهذا الطعام سيئ؟'' بل ``أي أسرة نسيتها اليوم؟''
\end{remark}

\section{ملاءمة الغذاء لليوم}

\begin{proposition}[الحاجات تتغير باليوم وبالشخص]\label{prop:g3:balanced-diet:needs}
مقدار \omterm{prop:g3:balanced-diet:jobs}{الطاقة} التي يحتاج إليها الجسم يتوقف على صاحبه وعلى ما
يفعله: فالطفل النامي يحتاج إلى مادة بناء أكثر نسبيًّا من الكبير؛
ويوم من الرياضة أو من برد الشتاء يحرق \omterm{prop:g3:balanced-diet:jobs}{طاقة} أكثر من أصيل من
القراءة. والجوع ينمو مع إنفاق الجسم.
\end{proposition}
\admitted

\begin{example}[لماذا يهمّ الفطور]\label{ex:g3:balanced-diet:breakfast}
مع الصباح يكون الجسم قد عمل الليل كله --- قلبًا ونفَسًا ودفئًا ---
بلا توريدات البتة. والفطور يعيد تموينه لدروس الصباح وجريه. فإن
فوّته نقصت الورشة في وسط الصباح: قرقرة في المعدة، وشرود في الانتباه،
ولا قوة في اللعب.
\end{example}

\begin{method}[مراجعة توازن يوم]\label{met:g3:balanced-diet:check}
في آخر اليوم، راجع القائمة:
\begin{enumerate}
  \item أفواكه أو خضر في كل وجبة؟ (والمقصد نحو خمس حصص في اليوم)
  \item أنشويات في كل وجبة \omterm{prop:g3:balanced-diet:jobs}{لطاقة} ثابتة؟
  \item أالبنّاؤون حاضرون --- ألبان مرات قليلة، ولحم أو سمك أو
        \omterm{def:g2:animals-grow-up:born}{بيض} مرة أو مرتين؟
  \item أالماء شراب اليوم؟
  \item أالحلويات قليلة المقدار لا جزء رئيس؟
\end{enumerate}
وأي أسرة غابت فادعُها غدًا.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:1}
ما عملا الغذاء؟ أعطِ \omterm{def:g3:balanced-diet:families}{أسرة غذائية} واحدة تخدم كل عمل في المقام
الأول.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:2}
سمِّ \omterm{def:g3:balanced-diet:families}{الأسر الغذائية}، والشراب الوحيد الذي يطلبه الجسم.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:3}
افرز هذه المشتريات في أسر: أرز، ولبن، وجزر، \omterm{def:g2:animals-grow-up:born}{وبيض}، وزبدة، وتفاح،
وخبز.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:4}
لماذا ينفق الجسم \omterm{prop:g3:balanced-diet:jobs}{طاقة} حتى في أثناء \omterm{prop:g1:healthy-habits:needs}{النوم}؟ سمِّ نفقتين من نفقات
الليل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:5}
راجع الغداء ``معكرونة، وسمك، وفاصولياء خضراء، ولبن، وتفاحة،
وماء'' في مقابل الصحن المتوازن في الشكل: أي جزء من الصحن يمثّل كل
صنف؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:6}
لماذا يهمّ الفطور خاصةً؟ وماذا يحدث في وسط الصباح من غيره؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:7}
من يحتاج إلى مادة بناء أكثر نسبيًّا: الطفل أم الكبير؟ ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:8}
جرّب: طبّق \cref{met:g3:balanced-diet:check} على يومك أنت، بصدق.
أي أسرة نسيت --- وأيها ظهر أكثر من حصته؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:balanced-diet:9}
``الحلوى محرّمة!'' أهذا ما يقوله هذا الفصل؟ أعطِ القاعدة الأدق في
\cref{rem:g3:balanced-diet:balance}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:balanced-diet:10}
في عطلة تزلج يكون الجميع أجوع منهم في البيت. اشرح ذلك وفق
\cref{prop:g3:balanced-diet:needs} --- وسمِّ النفقتين الزائدتين
اللتين يضيفهما يوم بارد رياضي.
\end{exercise}
