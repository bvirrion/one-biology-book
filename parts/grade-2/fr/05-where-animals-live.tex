\chapter{Où vivent les animaux}\label{ch:g2:where-animals-live}

Tu ne rencontreras pas de truite dans la cour de l'école, ni
d'écureuil au fond de la mare. Chaque \omterm{def:g1:animals-around-us:animal}{animal} vit dans le genre
d'endroit qui lui convient --- là où il trouve sa nourriture, son
abri et ses voisins. Les biologistes ont un nom pour le pays d'un
\omterm{def:g1:animals-around-us:animal}{animal}.

\section{Les milieux de vie}

\begin{definition}[Milieu de vie]\label{def:g2:where-animals-live:habitat}
Le \emph{milieu de vie}\index{milieu de vie} d'un \omterm{def:g1:animals-around-us:animal}{animal} est le
genre d'endroit où il vit naturellement et où il trouve tout ce
qu'il lui faut~: nourriture, eau, abri, et un coin sûr pour ses
petits. La mare, la forêt, le pré, la haie et le bord de mer sont
des milieux de vie.
\end{definition}

\begin{example}[Quatre milieux, quatre quartiers]\label{ex:g2:where-animals-live:four}
Dans la \emph{mare}~: grenouilles, libellules, épinoches, limnées.
Dans la \emph{forêt}~: écureuils, chevreuils, pics, fourmis rousses.
Dans le \emph{pré}~: lapins, sauterelles, papillons, mulots. Sous
les pierres d'un vieux \emph{mur}~: cloportes, araignées, parfois un
lézard qui se chauffe au soleil sur le dessus.
\end{example}

\begin{remark}[Un milieu, beaucoup de logements]\label{rem:g2:where-animals-live:layers}
Un \omterm{def:g2:where-animals-live:habitat}{milieu de vie} a bien des recoins. Dans une même forêt, le pic vit
haut sur les troncs, l'écureuil dans les \omterm{def:g1:plants-around-us:tree}{branches}, le chevreuil
parmi les buissons, et les fourmis rousses au sol, dans leur nid en
dôme. Même forêt, étages différents --- ainsi beaucoup d'\omterm{def:g1:animals-around-us:animal}{animaux}
peuvent partager un milieu sans se gêner.
\end{remark}

\section{Fait pour son milieu}

\begin{proposition}[Les animaux sont faits pour leur milieu]\label{prop:g2:where-animals-live:fit}
Le corps et les habitudes d'un \omterm{def:g1:animals-around-us:animal}{animal} vont avec l'endroit où il vit~:
\begin{itemize}
  \item les pieds palmés de la grenouille poussent l'eau --- faits
        pour la mare~;
  \item les griffes agrippantes de l'écureuil et sa grande queue
        d'équilibriste sont faites pour les arbres~;
  \item les pattes fouisseuses du lapin et ses longues \omterm{def:g1:the-five-senses:sense}{oreilles}
        d'alerte sont faites pour le pré ouvert, avec des terriers
        dessous~;
  \item les \omterm{def:g1:animals-around-us:covering}{plumes} du canard laissent glisser l'eau, et ses pieds
        palmés rament.
\end{itemize}
Déplacé dans le mauvais milieu, un \omterm{def:g1:animals-around-us:animal}{animal} perd ces avantages.
\end{proposition}
\admitted

\begin{example}[L'équipement de la taupe]\label{ex:g2:where-animals-live:mole}
La taupe vit sous terre, dans des galeries qu'elle creuse sous le
pré. Ses mains sont de larges pelles à creuser, sa fourrure se
couche dans les deux sens pour que la terre ne la rebrousse jamais,
et ses \omterm{def:g1:the-five-senses:sense}{yeux} faibles n'ont guère d'importance dans le noir. À l'air
libre elle est maladroite~; dans son milieu, c'est une championne.
\end{example}

\begin{method}[Rattacher un animal à son milieu]\label{met:g2:where-animals-live:matching}
Devant un \omterm{def:g1:animals-around-us:animal}{animal} que tu ne connais pas~:
\begin{enumerate}
  \item regarde son équipement~: pieds palmés~? griffes~? ailes~?
        pattes fouisseuses~?
  \item demande-toi ce qu'il mange, et où cette nourriture se
        trouve~;
  \item demande-toi où il peut s'abriter et élever ses petits~;
  \item le milieu qui répond aux trois est sans doute son foyer.
\end{enumerate}
\end{method}

\section{Les milieux de vie peuvent disparaître}

\begin{proposition}[Pas de milieu, pas d'habitants]\label{prop:g2:where-animals-live:loss}
Un \omterm{def:g1:animals-around-us:animal}{animal} ne peut pas vivre sans son milieu. Si une mare est
comblée, ses grenouilles et ses libellules disparaissent avec elle~;
si une haie est arrachée, les oiseaux qui y nichaient perdent leur
abri. Protéger les \omterm{def:g1:animals-around-us:animal}{animaux} commence par protéger les endroits où ils
vivent.
\end{proposition}
\admitted

\begin{example}[La mare de l'école]\label{ex:g2:where-animals-live:schoolpond}
Une école creuse une petite mare dans un coin de la cour. En deux
ans, des libellules y patrouillent, des gerris marchent dessus, et
des grenouilles s'y sont installées --- personne ne les a apportées.
Fais le milieu, et les habitants le trouvent.
\end{example}

\begin{remark}[Regarder sans déranger]\label{rem:g2:where-animals-live:respect}
Quand on visite un \omterm{def:g2:where-animals-live:habitat}{milieu de vie}, on est invité. Observe en
silence, remets les pierres comme tu les as trouvées, laisse les
\omterm{def:g2:animals-grow-up:born}{œufs} et les nids tranquilles, et remporte tes déchets --- le milieu
doit oublier que tu es passé. Les chapitres suivants donnent
d'autres façons de prendre soin de la nature.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:1}
Qu'est-ce qu'un \omterm{def:g2:where-animals-live:habitat}{milieu de vie}~? Nomme les quatre choses qu'un \omterm{def:g1:animals-around-us:animal}{animal}
y trouve.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:2}
Donne deux \omterm{def:g1:animals-around-us:animal}{animaux} de la mare, deux de la forêt et deux du pré.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:3}
Quel milieu convient à chaque \omterm{def:g1:animals-around-us:animal}{animal}~: l'épinoche, le pic, la
sauterelle, le cloporte~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:4}
Quel est l'équipement de nage de la grenouille~? Et l'équipement de
grimpe de l'écureuil~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:5}
Énumère les trois pièces de l'équipement souterrain de la taupe,
d'après \cref{ex:g2:where-animals-live:mole}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:6}
Dans \cref{rem:g2:where-animals-live:layers}, quel «~étage~» de la
forêt chaque \omterm{def:g1:animals-around-us:animal}{animal} occupe-t-il~: le pic, l'écureuil, le chevreuil,
la fourmi rousse~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:7}
Qu'arrive-t-il aux grenouilles et aux libellules quand une mare est
comblée~? Que nous apprend cela sur la protection des \omterm{def:g1:animals-around-us:animal}{animaux}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:8}
À toi d'essayer~: choisis un petit milieu près de chez toi --- sous
une pierre, un buisson, un massif de \omterm{def:g1:plants-around-us:parts}{fleurs}. Observe en silence et
note tous les \omterm{def:g1:animals-around-us:animal}{animaux} que tu vois pendant un temps fixé. Remets
chaque pierre en place. Combien d'habitants ton petit milieu
avait-il~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:where-animals-live:9}
Un \omterm{def:g1:animals-around-us:animal}{animal} mystère a des pieds palmés, un plumage dense qui laisse
glisser l'eau, et un bec plat qui filtre. Sers-toi de
\cref{met:g2:where-animals-live:matching} pour nommer son milieu ---
et peut-être l'\omterm{def:g1:animals-around-us:animal}{animal}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:where-animals-live:10}
Tom veut «~aider~» une grenouille en la ramenant chez lui dans un
terrarium sec bien garni de nourriture. À l'aide de
\cref{prop:g2:where-animals-live:fit} et de
\cref{prop:g2:where-animals-live:loss}, explique pourquoi cela
n'aiderait pas du tout la grenouille.
\end{exercise}
