\chapter{Digestão e nutrientes}\label{ch:g7:digestion-and-nutrients}

O \cref{ch:g4:journey-of-food} acompanhou a mordida de pão tubo abaixo
e prometeu que ``os \omterm{def:g4:journey-of-food:digestion}{sucos digestivos} a desmontam''. Neste ano abrimos
as portas da oficina: o que exatamente os sucos fazem, quais são os
\omterm{def:g7:digestion-and-nutrients:families}{nutrientes} que eles produzem e como a parede do \omterm{def:g4:journey-of-food:tract}{intestino delgado} dá
conta da maior travessia de fronteira do corpo? A mordida de pão está
de volta --- com instrumentos melhores.

\section{Os sucos em ação}

\begin{proposition}[A digestão é uma desmontagem química]\label{prop:g7:digestion-and-nutrients:chemical}
Mastigar e bater só quebram o alimento em pedaços \emph{menores}; a
desmontagem de verdade é química. Os \emph{sucos
digestivos}\index{suco digestivo} --- a \omterm{def:g4:journey-of-food:tract}{saliva}, o suco do \omterm{def:g4:journey-of-food:tract}{estômago}, os
sucos despejados no \omterm{def:g4:journey-of-food:tract}{intestino delgado} --- contêm substâncias
trabalhadoras que cortam os componentes grandes do alimento em
componentes pequenos: o amido em açúcares, a matéria de construção da
carne e do \omterm{def:g2:animals-grow-up:born}{ovo} nas unidades pequenas dela, as gorduras nas unidades
delas. Cada suco tem as especialidades dele, e cada um trabalha melhor
nas \omterm{def:g6:exploring-our-environment:conditions}{condições} da estação dele.
\end{proposition}
\admitted

\begin{example}[A versão de bancada]\label{ex:g7:digestion-and-nutrients:bench}
A experiência clássica roda a \omterm{def:g4:journey-of-food:digestion}{digestão} dentro do vidro. Dois tubos com
pasta de amido cozido à \omterm{def:g6:exploring-our-environment:conditions}{temperatura} do corpo: um recebe \omterm{def:g4:journey-of-food:tract}{saliva} fresca,
o outro recebe \omterm{def:g4:journey-of-food:tract}{saliva} fervida (estragada). Uma hora depois, o teste do
amido acha o tubo 2 inalterado --- ainda amido --- enquanto a pasta do
tubo 1 virou açúcares, exatamente o docinho do
\cref{ex:g4:journey-of-food:sweet}, produzido sem boca nenhuma. O suco
sozinho fez a desmontagem --- e o controle fervido mostra que a
substância trabalhadora dele é destruída pelo calor, como aconteceria
com uma maquinaria delicada.
\end{example}

\begin{remark}[Por que desmontar?]\label{rem:g7:digestion-and-nutrients:why}
Os componentes grandes do alimento não conseguem atravessar a parede do
intestino, assim como um guarda-roupa não passa por uma caixa de
correio. E o corpo não os quer inteiros: ele quer \emph{peças} ---
unidades pequenas que ele possa reconstruir na substância dele
(\cref{prop:g6:matter-from-food:transform}). A \omterm{def:g4:journey-of-food:digestion}{digestão} é desmontagem
para o transporte \emph{e} para a remontagem: móveis viram kit
desmontado, em cada refeição.
\end{remark}

\section{Os nutrientes}

\begin{definition}[As famílias de nutrientes]\label{def:g7:digestion-and-nutrients:families}
A desmontagem produz os \emph{nutrientes}\index{nutriente} --- os
componentes pequenos, prontos para o \omterm{prop:g4:journey-of-food:absorb}{sangue}:
\begin{itemize}
  \item os \emph{açúcares} (a \omterm{prop:g7:organs-at-work:bill}{glicose} acima de tudo, o combustível da
        \cref{prop:g7:organs-at-work:bill}), vindos dos alimentos
        cheios de amido e dos doces;
  \item as unidades pequenas da \emph{matéria de construção}, vindas
        de carne, peixe, \omterm{def:g2:animals-grow-up:born}{ovo}, laticínios e leguminosas --- os tijolos
        da construção da \cref{prop:g3:balanced-diet:jobs};
  \item as unidades das \emph{gorduras} --- combustível denso e
        material de construção ao mesmo tempo;
  \item mais os passageiros que não precisam de desmontagem: as
        \emph{vitaminas}, os \emph{\omterm{prop:g4:what-plants-need:needs}{sais minerais}} e a própria água.
\end{itemize}
As \omterm{def:g3:balanced-diet:families}{famílias de alimentos} da \cref{def:g3:balanced-diet:families} são,
no fundo, fornecedoras dessas famílias de nutrientes --- a lista de
compras por trás da lista de compras.
\end{definition}

\begin{example}[Um almoço, item por item]\label{ex:g7:digestion-and-nutrients:lunch}
O almoço equilibrado do \cref{ex:g3:balanced-diet:lunch}, relido como
\omterm{def:g7:digestion-and-nutrients:families}{nutrientes}: a massa --- amido, desmontado em \omterm{prop:g7:organs-at-work:bill}{glicose} aos litros; o
peixe --- matéria de construção, cortada nas unidades dela; o óleo da
vagem --- unidades de gordura; a vagem e a maçã --- vitaminas,
minerais, água e ainda fibras. Repare que a fibra é a exceção que
confirma o sistema: nenhum suco nosso consegue cortá-la, então ela não
atravessa nada e segue viagem --- o volume dos restos
(\cref{prop:g4:journey-of-food:absorb}), fazendo a varrição útil dela.
\end{example}

\section{A grande travessia}

\begin{proposition}[A absorção nas vilosidades]\label{prop:g7:digestion-and-nutrients:absorption}
A travessia prometida na \cref{prop:g4:journey-of-food:absorb}
acontece através da parede interna do \omterm{def:g4:journey-of-food:tract}{intestino delgado}, e essa parede
é construída para isso: dobrada e forrada com milhões de
\emph{vilosidades}\index{vilosidade} --- saliências em forma de dedo,
de um milímetro de altura, cada uma envolvendo os próprios \omterm{def:g4:heart-and-blood:vessels}{vasos
sanguíneos} finos. As \omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades} são o motivo de a superfície de
travessia chegar ao tamanho do carpete de um cômodo
(\cref{rem:g4:journey-of-food:surface}): dobras sobre dobras sobre
dedos. Os \omterm{def:g7:digestion-and-nutrients:families}{nutrientes} atravessam as paredes finas delas para o \omterm{prop:g4:journey-of-food:absorb}{sangue},
e o \omterm{prop:g4:journey-of-food:absorb}{sangue} os leva primeiro a um grande órgão de triagem e estoque ---
o \emph{fígado}\index{fígado} --- antes da entrega geral.
\end{proposition}
\admitted

\begin{example}[A especificação, de novo]\label{ex:g7:digestion-and-nutrients:spec}
Grandes, finas, úmidas, forradas de \omterm{prop:g4:journey-of-food:absorb}{sangue}: as \omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades} cumprem
exatamente a especificação da
\cref{prop:g7:how-animals-breathe:spec} --- só que para \omterm{def:g7:digestion-and-nutrients:families}{nutrientes} em
vez de \omterm{def:g7:what-is-respiration:respiration}{oxigênio}. As bolsas dos \omterm{def:g7:how-animals-breathe:four}{pulmões}, as plumas das \omterm{def:g7:how-animals-breathe:four}{brânquias}, as
\omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades} do intestino: onde quer que o corpo precise mover muita
coisa por uma fronteira, ele constrói a mesma máquina. Reconheça a
especificação uma vez e você a reconhece em toda parte.
\end{example}

\begin{method}[Rastrear um nutriente de ponta a ponta]\label{met:g7:digestion-and-nutrients:trace}
Para qualquer garfada, rastreie um \omterm{def:g7:digestion-and-nutrients:families}{nutriente}:
\begin{enumerate}
  \item diga a família do alimento e os componentes principais dele;
  \item estação por estação, anote onde a desmontagem dele começa e
        termina, e que sucos trabalharam nele;
  \item atravesse com ele nas \omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades}, para o \omterm{prop:g4:journey-of-food:absorb}{sangue}, passando
        pelo \omterm{prop:g7:digestion-and-nutrients:absorption}{fígado};
  \item entregue esse \omterm{def:g7:digestion-and-nutrients:families}{nutriente}: a uma \omterm{def:g5:first-look-at-cells:cell}{célula} em atividade, como
        combustível (\cref{prop:g7:organs-at-work:bill}), ou a um
        canteiro de obras, como material
        (\cref{def:g6:matter-from-food:production}).
\end{enumerate}
\end{method}

\begin{example}[O rastreamento, feito no pão]\label{ex:g7:digestion-and-nutrients:breadtrace}
Pão: família dos alimentos com amido, principalmente amido. A
desmontagem começa na boca (a \omterm{def:g4:journey-of-food:tract}{saliva} --- o docinho da mastigação
longa), faz uma pausa no ácido do \omterm{def:g4:journey-of-food:tract}{estômago} e termina nos sucos do
\omterm{def:g4:journey-of-food:tract}{intestino delgado}: \omterm{prop:g7:organs-at-work:bill}{glicose}. Travessia: \omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades}, \omterm{prop:g4:journey-of-food:absorb}{sangue}, \omterm{prop:g7:digestion-and-nutrients:absorption}{fígado} ---
que deposita parte da \omterm{prop:g7:organs-at-work:bill}{glicose} para depois, um dos muitos serviços
dele. Entrega: um \omterm{def:g3:bones-and-muscles:muscle}{músculo} da coxa no meio de uma corrida a queima em
menos de uma hora. A mordida do ano quatro, totalmente contabilizada.
\end{example}

\section{Problema: O intestino de vidro}

\begin{problem}[{Problema de fim de semana --- montar a digestão na
bancada, tubo a tubo}]\label{pb:g7:digestion-and-nutrients:1}
A turma monta um ``intestino de vidro'': uma estante de tubos à
\omterm{def:g6:exploring-our-environment:conditions}{temperatura} do corpo, cada um simulando uma estação do \omterm{def:g4:journey-of-food:tract}{tubo digestivo}.
Tubo M (boca): pasta de amido mais \omterm{def:g4:journey-of-food:tract}{saliva} fresca. Tubo M$'$: pasta de
amido mais \omterm{def:g4:journey-of-food:tract}{saliva} fervida. Tubo S (\omterm{def:g4:journey-of-food:tract}{estômago}): cubinhos de clara de \omterm{def:g2:animals-grow-up:born}{ovo}
mais suco gástrico da farmácia. Tubo S$'$: os mesmos cubinhos em água
pura. Todos ficam duas horas; depois vêm os testes --- o do amido, o
dos açúcares e uma olhada nos cubinhos.

\medskip\noindent\textbf{Parte I --- Previsões e
resultados.}
\begin{enumerate}
  \item Preveja os resultados de duas horas dos quatro tubos.
  \item M fica açucarado, M$'$ não. Enuncie a conclusão e o papel
        exato de M$'$.
  \item Os cubinhos de S encolhem, amolecem e ficam transparentes; os
        de S$'$ ficam inalterados. O que o suco gástrico fez --- e que
        família de \omterm{def:g7:digestion-and-nutrients:families}{nutrientes} está sendo produzida?
  \item Nenhum tubo foi mastigado nem batido. O que a estante prova
        sobre onde está o trabalho de verdade da \omterm{def:g4:journey-of-food:digestion}{digestão}?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Melhorando o modelo.}
\begin{enumerate}[resume]
  \item Um aluno propõe rodar o tubo M na \omterm{def:g6:exploring-our-environment:conditions}{temperatura} da geladeira
        ``para ficar mais organizado''. Faça a objeção, com o padrão
        da \cref{prop:g6:decomposers-and-soil:speed} aplicado aos
        sucos.
  \item Outro propõe um único tubo grande com os dois sucos
        misturados. Que fato de estação por estação da
        \cref{prop:g7:digestion-and-nutrients:chemical} a mistura
        embaralharia?
  \item O intestino de vidro não tem parede: nada é absorvido. Que
        estrutura um modelo completo precisaria ter em seguida, e que
        duas propriedades o material dela precisa ter?
  \item Acrescente fibra --- flocos de farelo --- a um quinto tubo com
        todos os sucos disponíveis. Preveja o estado dele em duas
        horas e o significado disso para o modelo.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Do vidro ao intestino.}
\begin{enumerate}[resume]
  \item No corpo, os açúcares de M encontrariam em seguida as
        \omterm{prop:g7:digestion-and-nutrients:absorption}{vilosidades}. Continue a viagem deles em quatro paradas.
  \item O suco do \omterm{def:g4:journey-of-food:tract}{estômago} trabalha em ácido forte; os sucos do
        intestino precisam que esse ácido seja neutralizado. O que
        isso diz sobre ``cada suco nas \omterm{def:g6:exploring-our-environment:conditions}{condições} da estação dele'' ---
        e sobre atravessar as estações em ordem?
  \item Uma indústria farmacêutica vende cápsulas de sucos para
        pessoas cujas próprias estações funcionam mal. Explique, com a
        estante, por que uma terapia dessas é sequer possível.
  \item Resuma a prática numa frase contendo \emph{sucos},
        \emph{\omterm{def:g7:digestion-and-nutrients:families}{nutrientes}} e \emph{tubos de controle}.
\end{enumerate}
\end{problem}
