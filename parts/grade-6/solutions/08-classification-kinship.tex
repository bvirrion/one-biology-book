\section*{Chapter \ref{ch:g6:classification-kinship} --- Classification and Kinship}

\begin{solution}{exo:g6:classification-kinship:1}
A feature an organism possesses, stated checkably: has a
\omterm{def:g4:classifying-animals:vertebrate}{backbone}; has four \omterm{def:g1:my-body:parts}{limbs}; has \omterm{def:g1:animals-around-us:covering}{fur} and feeds its young \omterm{def:g2:animals-grow-up:born}{milk}
(or: has \omterm{def:g1:animals-around-us:covering}{feathers}).
\end{solution}

\begin{solution}{exo:g6:classification-kinship:2}
Each \omterm{def:g6:classification-kinship:attribute}{attribute}'s set of owners sits wholly inside another's:
all fur-bearers have four \omterm{def:g1:my-body:parts}{limbs}, all four-limbed \omterm{def:g1:animals-around-us:animal}{animals}
have \omterm{def:g4:classifying-animals:vertebrate}{backbones} --- never the reverse, never crossing. Boxes
inside boxes, with no straddlers.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:3}
Because they inherited them from a common ancestor: an
\omterm{def:g6:classification-kinship:attribute}{attribute} travels down a family line, so its owners are that
ancestor's descendants --- a family.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:4}
The trout stands beyond only the \omterm{def:g4:classifying-animals:vertebrate}{backbone} dot. The whale
stands beyond \omterm{def:g4:classifying-animals:vertebrate}{backbone}, four \omterm{def:g1:my-body:parts}{limbs}, and fur-and-milk ---
all three.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:5}
Whale and cat: they share the fur-and-milk fork, the newest
one drawn; the trout parted company at the oldest fork.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:6}
They are the same information twice: a box is a \omterm{def:g1:plants-around-us:tree}{branch} with
all its twigs --- everything past a fork inherits the fork's
\omterm{def:g6:classification-kinship:attribute}{attribute} and sits in its box.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:7}
Fur-and-milk is inherited down one family line --- it marks
ancestry. ``Shaped like a \omterm{prop:g3:sorting-living-things:groups}{fish}'' can be stamped on strangers
by the same watery way of life, as the whale shows: shape
borrowed, family elsewhere. \omterm{def:g4:classifying-animals:classification}{Classification} sorts families,
so it trusts the heirloom and distrusts the borrowed coat.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:8}
Trout: \omterm{def:g4:classifying-animals:vertebrate}{backbone} only. Frog: \omterm{def:g4:classifying-animals:vertebrate}{backbone}, four \omterm{def:g1:my-body:parts}{limbs}. Cat:
\omterm{def:g4:classifying-animals:vertebrate}{backbone}, four \omterm{def:g1:my-body:parts}{limbs}, fur-and-milk. Blackbird: \omterm{def:g4:classifying-animals:vertebrate}{backbone},
four \omterm{def:g1:my-body:parts}{limbs}, \omterm{def:g1:animals-around-us:covering}{feathers}. The nesting shows in the ticks: every
row's ticks include the \omterm{def:g4:classifying-animals:vertebrate}{backbone}; every coat-tick row also
ticks four \omterm{def:g1:my-body:parts}{limbs}.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:9}
It passes \omterm{def:g4:classifying-animals:vertebrate}{backbone} and four \omterm{def:g1:my-body:parts}{limbs}, and \omterm{def:g1:plants-around-us:parts}{leaves} on its own
\omterm{def:g1:plants-around-us:tree}{branch} with its dry scaly \omterm{def:g1:the-five-senses:sense}{skin} --- beside, not beyond, the
\omterm{def:g1:animals-around-us:covering}{fur} and \omterm{def:g1:animals-around-us:covering}{feather} forks.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:10}
Nursing marks \omterm{prop:g6:classification-kinship:kinship}{kinship}: fur-and-milk is the mammal family's
heirloom, so the bat \omterm{def:g1:plants-around-us:tree}{branches} with the cat. Flying is
borrowed by way of life --- wings arose separately on the
bird \omterm{def:g1:plants-around-us:tree}{branch} and the bat \omterm{def:g1:plants-around-us:tree}{branch}, as streamlining did in
whale and trout.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:11}
It belongs. The platypus descends from the \omterm{prop:g3:sorting-living-things:groups}{mammals}' common
ancestor and carries the family heirlooms (\omterm{def:g1:animals-around-us:covering}{fur}, \omterm{def:g2:animals-grow-up:born}{milk}); its
\omterm{def:g2:animals-grow-up:born}{egg} is an old feature its early \omterm{def:g1:plants-around-us:tree}{branch} kept while the
others lost it. The family is the ancestor and \emph{all}
descendants --- oddities included; the bundle
(\cref{exo:g4:classifying-animals:11}) already voted it in.
\end{solution}

\begin{solution}{exo:g6:classification-kinship:12}
If every pair of \omterm{def:g1:plants-around-us:tree}{branches} meets, then \omterm{def:g5:biodiversity:species}{species} must change
across generations over long time --- new \omterm{def:g6:classification-kinship:attribute}{attributes}
arising and being inherited --- so that one ancestral line
could \omterm{def:g1:plants-around-us:tree}{branch} into trout, cat and daisy. Implication in one
sentence: all \omterm{def:g6:exploring-our-environment:components}{living things} descend, with change, from
common ancestors. Evidence to demand: traces of the change
--- intermediate forms, preserved remains in old rocks,
and family resemblances placed exactly where the \omterm{def:g6:classification-kinship:tree}{tree}
predicts (the business of a later year).
\end{solution}

\begin{solution}{pb:g6:classification-kinship:1}
\textbf{1.} Trout: \omterm{def:g4:classifying-animals:vertebrate}{backbone}. Frog: \omterm{def:g4:classifying-animals:vertebrate}{backbone}, four \omterm{def:g1:my-body:parts}{limbs}.
Lizard: \omterm{def:g4:classifying-animals:vertebrate}{backbone}, four \omterm{def:g1:my-body:parts}{limbs}, dry scaly \omterm{def:g1:the-five-senses:sense}{skin}. Pigeon:
\omterm{def:g4:classifying-animals:vertebrate}{backbone}, four \omterm{def:g1:my-body:parts}{limbs}, \omterm{def:g1:animals-around-us:covering}{feathers}. Mouse: \omterm{def:g4:classifying-animals:vertebrate}{backbone}, four
\omterm{def:g1:my-body:parts}{limbs}, fur-and-milk. Dolphin: \omterm{def:g4:classifying-animals:vertebrate}{backbone}, four \omterm{def:g1:my-body:parts}{limbs},
fur-and-milk (the label's testimony on \omterm{def:g2:animals-grow-up:born}{milk}; features
decide, and \omterm{def:g2:animals-grow-up:born}{milk} is the deciding feature).

\textbf{2.} The \omterm{def:g4:classifying-animals:vertebrate}{backbone} --- all six are \omterm{def:g4:classifying-animals:vertebrate}{vertebrates}. The
widest sharing marks the oldest common ancestor: all six
are one family at that depth.

\textbf{3.} Frog, lizard, pigeon, mouse, dolphin. The
pigeon's wings and the dolphin's flippers are four \omterm{def:g1:my-body:parts}{limbs} in
working dress: the same inherited \omterm{def:g1:my-body:parts}{limbs}, shaped by
different lives --- the \omterm{def:g6:classification-kinship:tree}{tree} counts the inheritance, not
the dress.

\textbf{4.} Because a shared address is not a shared
family: ``lives in water'' is a \omterm{def:g2:where-animals-live:habitat}{habitat}, borrowable by
strangers, not an \omterm{def:g6:classification-kinship:attribute}{attribute} of the body's ancestry ---
\cref{def:g3:sorting-living-things:criterion}'s old rule,
now with its reason.

\textbf{5.} Trunk $\rightarrow$ \omterm{def:g4:classifying-animals:vertebrate}{backbone} fork (trout \omterm{def:g1:plants-around-us:parts}{leaves}
after it) $\rightarrow$ \omterm{def:g1:my-body:parts}{four-limbs} fork $\rightarrow$ three
coat \omterm{def:g1:plants-around-us:tree}{branches}: dry scales (lizard), \omterm{def:g1:animals-around-us:covering}{feathers} (pigeon),
fur-and-milk (mouse and dolphin together on the newest
fork); the frog \omterm{def:g1:plants-around-us:parts}{leaves} after \omterm{def:g1:my-body:parts}{four-limbs}, before every coat.

\textbf{6.} The mouse's closest drawer-cousin is the
dolphin (they share the fur-and-milk fork). The trout's
closest is everyone equally --- it parted at the first
fork, so all five others are its equally distant cousins.

\textbf{7.} An amphibian --- four-limbed, bare moist \omterm{def:g1:the-five-senses:sense}{skin},
no coat fork of its own; its \omterm{def:g1:plants-around-us:tree}{branch} \omterm{def:g1:plants-around-us:parts}{leaves} between the
\omterm{def:g1:my-body:parts}{four-limbs} fork and the coats.

\textbf{8.} Past \omterm{def:g4:classifying-animals:vertebrate}{backbone} and four \omterm{def:g1:my-body:parts}{limbs}, onto the dry
scaly \omterm{def:g1:plants-around-us:tree}{branch}: a reptile --- classified sight unseen,
because \omterm{def:g6:classification-kinship:attribute}{attributes}, not labels, decide.

\textbf{9.} Dolphin and mouse sit closer: fur-and-milk is
an inherited fork. The streamlining is the water's uniform,
issued alike to unrelated swimmers
(\cref{exo:g4:adapted-to-their-world:11}); counted, it
would file borrowed dress as family and wreck the nesting.

\textbf{10.} A single common ancestor \omterm{def:g5:biodiversity:species}{species} --- an early
four-limbed \omterm{def:g4:classifying-animals:vertebrate}{vertebrate} --- from which frog, lizard, pigeon,
mouse and dolphin all descend, each carrying its four
\omterm{def:g1:my-body:parts}{limbs} in different dress.

\textbf{11.} That all six --- and the class drawing them
--- belong to one branching family, descended with change
from common ancestors, humans on the fur-and-milk \omterm{def:g1:plants-around-us:tree}{branch}
among the rest.

\textbf{12.} The nesting is a fact about the living world
--- \omterm{def:g6:classification-kinship:attribute}{attributes} really do fall into boxes-in-boxes, which
no filing clerk decreed. The \omterm{def:g6:classification-kinship:tree}{tree} explains the fact:
inheritance from common ancestors \emph{must} produce
nested \omterm{def:g6:classification-kinship:attribute}{attributes}, while mere filing would leave the
world's obliging tidiness a miracle.
\end{solution}
