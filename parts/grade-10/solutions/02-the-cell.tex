\section*{Chapter \ref{ch:g10:the-cell} --- The Cell}

\begin{solution}{exo:g10:the-cell:1}
(1) Free-living \omterm{def:g10:scientific-biology:organism}{organisms} are cellular; (2) \omterm{def:g10:the-cell:cell}{cell} is basic unit of
structure and function; (3) \omterm{def:g10:the-cell:cell}{cells} from \omterm{def:g10:the-cell:cell}{cells}. Viruses are not free-living
\omterm{def:g10:scientific-biology:organism}{organisms}: they lack independent \omterm{def:g10:scientific-biology:properties}{metabolism} and \omterm{def:g10:the-cell:cell}{cell} structure, so claim
1 as worded is not overturned.
\end{solution}

\begin{solution}{exo:g10:the-cell:3}
Nucleus: genetic control. \omterm{def:g10:the-cell:energy-organelles}{Mitochondrion}: ATP via respiration.
\omterm{def:g10:the-cell:energy-organelles}{Chloroplast}: photosynthesis. Rough ER: protein synthesis/folding for
export. Golgi: modify/sort/ship. \omterm{def:g10:the-cell:endomembrane}{Lysosome}: digestion. \omterm{def:g10:the-cell:support}{Cell wall}: rigid
support and protection (plants/fungi/many \omterm{def:g10:the-cell:prok-euk}{prokaryotes}).
\end{solution}

\begin{solution}{exo:g10:the-cell:4}
Plant not animal: \omterm{def:g10:the-cell:support}{cell wall}, \omterm{def:g10:the-cell:energy-organelles}{chloroplasts}, large central \omterm{def:g10:the-cell:endomembrane}{vacuole}

\end{solution}

\begin{solution}{exo:g10:the-cell:5}
For $L=2$: $S=6\times 4=24\,\unit{\micro m^2}$, $V=8\,\unit{\micro m^3}$,
$S/V=3\,\unit{\per\micro m}$. For $L=20$: $S=6\times 400=2400$,
$V=8000$, $S/V=0.3$. The smaller \omterm{def:g10:the-cell:cell}{cell} has higher $S/V$ and exchanges more
efficiently per unit \omterm{def:g10:the-cell:sa-v}{volume}.
\end{solution}

\begin{solution}{exo:g10:the-cell:6}
Microvilli fold the apical membrane, greatly increasing \omterm{def:g10:the-cell:sa-v}{surface area} for
absorption without a proportional \omterm{def:g10:the-cell:sa-v}{volume} increase --- raising effective
$S/V$ for nutrient uptake.
\end{solution}

\begin{solution}{exo:g10:the-cell:7}

\omterm{def:g10:the-cell:microscope}{magnification} $= \qty{8}{mm}/400 = \qty{0.02}{mm} = \qty{20}{\micro m}$.
\end{solution}

\begin{solution}{exo:g10:the-cell:8}
Endosymbiosis: free-living bacteria engulfed by ancestral hosts became
symbiotic; double membranes reflect engulfment; residual DNA is a remnant
of the bacterial genome. \omterm{def:g10:the-cell:energy-organelles}{Mitochondria} from aerobic bacteria; \omterm{def:g10:the-cell:energy-organelles}{chloroplasts}
from cyanobacteria-like ancestors.
\end{solution}

\begin{solution}{exo:g10:the-cell:9}
Start low power, coarse then fine; raise power stepwise; only fine focus
on high power. Working distance is tiny at high power: coarse focus can
drive the objective into the slide.
\end{solution}

\begin{solution}{exo:g10:the-cell:10}
Peel epidermis, mount in water, stain lightly if needed. Calibrate
eyepiece graticule with stage micrometer at the chosen power. Measure many
\omterm{def:g10:the-cell:cell}{cells} along a line; mean diameter $=$ mean graticule units $\times$
calibration. Errors: \omterm{def:g10:the-cell:cell}{cell} edges fuzzy, non-spherical \omterm{def:g10:the-cell:cell}{cells}, dehydration,
miscalibration, sampling bias toward larger \omterm{def:g10:the-cell:cell}{cells}.
\end{solution}

\begin{solution}{exo:g10:the-cell:11}
Strategies for large \omterm{def:g10:the-cell:cell}{cells}: \omterm{def:g10:the-cell:endomembrane}{vacuoles} fill \omterm{def:g10:the-cell:sa-v}{volume} with low metabolic cost;
multiple nuclei; flattened shape; cytoplasmic streaming. Multicellular
strategy: many small specialised \omterm{def:g10:the-cell:cell}{cells} instead of one huge \omterm{def:g10:the-cell:cell}{cell}, keeping
each \omterm{def:g10:the-cell:cell}{cell}'s $S/V$ workable.
\end{solution}

\begin{solution}{pb:g10:the-cell:1}
\textbf{Part I.} Labels should include plasma membrane, nucleus,
mitochondria, ER, Golgi, \omterm{def:g10:the-cell:support}{ribosomes}, \omterm{def:g10:the-cell:support}{cytoskeleton}; plant adds wall,
\omterm{def:g10:the-cell:energy-organelles}{chloroplasts}, large \omterm{def:g10:the-cell:endomembrane}{vacuole}. Asterisk protein-secretion path: rough ER,
transport vesicles, Golgi, secretory vesicles, plasma membrane.

\textbf{Part II.} Plant-only: wall, \omterm{def:g10:the-cell:energy-organelles}{chloroplasts}, large central \omterm{def:g10:the-cell:endomembrane}{vacuole}.
Animal-only (typical): \omterm{def:g10:the-cell:endomembrane}{lysosomes} emphasised, centrioles in many animals,
no wall. Shared: nucleus, mitochondria, endomembrane, \omterm{def:g10:the-cell:support}{ribosomes}.
Payoffs: wall (support), \omterm{def:g10:the-cell:energy-organelles}{chloroplast} (photosynthesis), \omterm{def:g10:the-cell:endomembrane}{vacuole} (turgor,
storage).

\textbf{Part III.} \omterm{def:g10:the-cell:energy-organelles}{Mitochondria}: cardiac muscle $\gg$ fibroblast $\gg$
mammalian RBC (none when mature). Rough ER: plasma \omterm{def:g10:the-cell:cell}{cell} $\gg$ adipocyte.
Paragraph should link abundant mitochondria to continuous ATP demand in
heart muscle, or abundant rough ER to mass antibody secretion.
\end{solution}
