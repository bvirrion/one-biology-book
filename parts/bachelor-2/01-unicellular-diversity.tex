\chapter{Diversity of Unicellular Organisms}\label{ch:b2:unicellular-diversity}

A drop of pond water under the microscope holds a green cell rowing
with two flagella, a slipper-shaped ciliate sweeping bacteria into its
gullet, a glassy diatom sliding along the slide, and, too small to
see well, a thousand bacteria. Each is one cell and each is a whole
organism: it feeds, moves, senses, grows, reproduces and dies as a
single cell. Most of the living world has always been like this. This
chapter takes the \omterm{def:b2:unicellular-diversity:unicellular}{unicellular organism} as a design problem --- what
one cell can and cannot do --- and surveys the answers that
\omterm{def:b2:unicellular-diversity:prokaryote}{prokaryotes} and eukaryotes have found, up to the point where cells
begin to stay together.

\section{One cell, all the functions}

\begin{definition}[Unicellular, colonial, multicellular]\label{def:b2:unicellular-diversity:unicellular}
An organism is \emph{unicellular}\index{unicellular organism} when a
single cell carries out every function of life --- nutrition,
exchange, movement, reproduction --- and lives on its own. It is
\emph{colonial}\index{colonial organism} when cells of one kind stay
attached after dividing but each remains able to live alone if
separated. It is \emph{multicellular}\index{multicellular organism}
when its cells are of several types, depend on one another, and only
a few of them (the germ line) leave descendants. Unicellular
organisms include all archaea, nearly all bacteria, and most
lineages of eukaryotes; the informal name \emph{\omterm{def:b2:unicellular-diversity:protist}{protist}} covers the
unicellular eukaryotes, which are not one group but many.
\end{definition}

\begin{example}[Six inhabitants of a drop]\label{ex:b2:unicellular-diversity:drop}
\emph{Escherichia coli}, a rod \qty{2}{\micro m} long, swims with a
tuft of flagella and divides every twenty minutes when fed.
\emph{Chlamydomonas}, a green alga \qty{10}{\micro m} across, rows
with two flagella toward the light it detects with an eyespot.
\emph{Paramecium}, a ciliate of \qty{200}{\micro m}, is covered with
thousands of cilia, feeds through a gullet and bails water out with
two \omterm{prop:b2:unicellular-diversity:osmoregulation}{contractile vacuoles}. A diatom lives in a two-part shell of
silica, ornamented with pores. \emph{Amoeba} flows by extending
pseudopods and engulfs what it meets. A yeast cell buds a daughter
from its side. Six answers, of widely different size and machinery,
to one problem.
\end{example}

\begin{proposition}[Surface, volume and the limit on size]\label{prop:b2:unicellular-diversity:surface}
\index{surface-to-volume ratio}
A cell exchanges with its surroundings through its surface and
consumes in proportion to its volume. For a sphere of radius $r$,
the \emph{\omterm{prop:b2:unicellular-diversity:surface}{surface-to-volume ratio}} is
\[
\frac{S}{V} = \frac{4\pi r^2}{\tfrac{4}{3}\pi r^3} = \frac{3}{r},
\]
so that doubling the radius halves the surface available per unit of
cytoplasm. A cell that relies on diffusion across its membrane to
feed its interior therefore cannot grow indefinitely: at some radius
the supply through the surface no longer meets the demand of the
volume. The same law explains why small cells are metabolically the
most active per gram, and why large cells are flattened, elongated,
vacuolated or full of internal membranes.
\end{proposition}

\begin{theorem}[Time to diffuse a distance]\label{thm:b2:unicellular-diversity:diffusion}
\index{diffusion time}
A molecule with diffusion coefficient $D$ wanders in random steps;
after a time $t$ its mean square displacement along one axis is
$\langle x^2\rangle = 2Dt$. The time needed to cover a distance $x$
by diffusion alone is therefore
\[
t \approx \frac{x^2}{2D},
\]
which grows with the \emph{square} of the distance. With
$D \approx \qty{1e-9}{m^2/s}$ for a small molecule in water, crossing
a bacterium (\qty{1}{\micro m}) takes half a millisecond; crossing a
\qty{1}{mm} cell would take eight minutes, and a metre, sixteen
years.
\end{theorem}
\begin{proof}
Let the molecule take steps of length $\ell$ at random to the left
or right every $\tau$ seconds. After $n$ steps its displacement is
$x = \sum_i \varepsilon_i \ell$ with $\varepsilon_i = \pm 1$
independent, so $\langle x\rangle = 0$ and $\langle x^2\rangle =
\sum_i \langle\varepsilon_i^2\rangle \ell^2 = n\ell^2$ (the cross
terms average to zero). With $n = t/\tau$ this is $\langle x^2\rangle
= (\ell^2/\tau)\,t$, and writing $D = \ell^2/2\tau$ gives
$\langle x^2\rangle = 2Dt$. Setting $\langle x^2\rangle = x^2$ gives
the time.
\end{proof}

\begin{theorem}[Diffusive supply to a spherical cell]\label{thm:b2:unicellular-diversity:supply}
\index{diffusion-limited uptake}
A spherical cell of radius $r_0$ that absorbs every molecule of a
solute reaching its surface, in a medium where the concentration far
away is $c_\infty$, receives at steady state a flux (molecules per
second) of
\[
J = 4\pi D r_0\, c_\infty .
\]
The supply grows only with $r_0$, while the demand grows with
$r_0^3$: beyond a certain radius a cell fed by diffusion starves.
\end{theorem}
\begin{proof}
At steady state the number of molecules crossing every sphere of
radius $r > r_0$ per second is the same, $J = 4\pi r^2 D\,
(\mathrm{d}c/\mathrm{d}r)$ (Fick's law, the flux directed inward).
So $r^2\,\mathrm{d}c/\mathrm{d}r = J/4\pi D$ is constant; integrating
from $r_0$, where $c = 0$, to infinity, where $c = c_\infty$:
$c_\infty = (J/4\pi D)\int_{r_0}^{\infty} r^{-2}\,\mathrm{d}r =
J/(4\pi D r_0)$, whence $J = 4\pi D r_0 c_\infty$.
\end{proof}

\begin{example}[How big can a diffusion-fed cell be?]\label{ex:b2:unicellular-diversity:howbig}
An aerobic cell consumes oxygen at about $q = \qty{0.02}{mol/m^3/s}$
of cytoplasm (a brisk rate). Its demand is $q \cdot \tfrac{4}{3}\pi
r_0^3$; the supply from air-saturated water ($c_\infty =
\qty{0.25}{mol/m^3}$, $D = \qty{2e-9}{m^2/s}$) is $4\pi D r_0
c_\infty$. Equating, $r_0^2 = 3Dc_\infty/q = 3\times 2\times 10^{-9}
\times 0.25/0.02 = \qty{7.5e-8}{m^2}$, so $r_0 \approx
\qty{270}{\micro m}$. Real active cells stay well below this because
oxygen must also diffuse \emph{inside} them; the \qty{750}{\micro m}
sulfur bacterium \emph{Thiomargarita} gets round the limit by being
almost entirely a vacuole, its cytoplasm a thin shell a few
micrometres deep.
\end{example}

\section{Prokaryotic diversity}

\begin{definition}[The prokaryotic cell]\label{def:b2:unicellular-diversity:prokaryote}
A \emph{prokaryote}\index{prokaryote} (the bacteria and the archaea,
two domains as distinct from each other as either is from us) is a
cell without a nucleus or membrane-bound organelles: a single
circular chromosome in a \emph{nucleoid}\index{nucleoid}, often with
small extra circles called \emph{plasmids}\index{plasmid}, ribosomes
free in a cytoplasm of \qtyrange{0.5}{2}{\micro m}, a plasma membrane
that carries the respiratory or photosynthetic chains, and a rigid
\emph{cell wall}\index{cell wall!bacterial} that resists the turgor
of a cytoplasm several bars above its surroundings. Shapes are few
and named: \emph{coccus} (sphere), \emph{bacillus} (rod),
\emph{spirillum} and \emph{spirochaete} (helices), \emph{vibrio}
(comma); cells may stay in pairs, chains, tetrads or clusters after
dividing.
\end{definition}

\begin{proposition}[Two envelopes: Gram-positive and Gram-negative]\label{prop:b2:unicellular-diversity:gram}
\index{Gram stain}\index{peptidoglycan}
The bacterial wall is built of \emph{peptidoglycan}, chains of two
alternating sugars cross-linked by short peptides into one giant
molecule around the cell. \emph{Gram-positive} bacteria have a thick
layer (\qtyrange{20}{80}{nm}) outside a single membrane and retain
the crystal-violet stain of the Gram procedure; \emph{Gram-negative}
bacteria have a thin layer (a few nanometres) sandwiched between the
plasma membrane and an \emph{outer membrane} whose outer leaflet is
made of lipopolysaccharide, and they lose the stain. The outer
membrane, pierced by porins, is a second barrier: it keeps out many
antibiotics and detergents, which is why Gram-negative infections
are harder to treat. Archaea have neither peptidoglycan nor the
ester-linked lipids of every other cell: their membranes are built
of ether-linked isoprenoid chains, sometimes spanning the whole
bilayer as a monolayer, which keeps them stable in boiling acid.
\end{proposition}

\begin{proposition}[The bacterial flagellum and chemotaxis]\label{prop:b2:unicellular-diversity:flagellum}
\index{flagellum!bacterial}\index{chemotaxis}
The bacterial \emph{flagellum} is a rigid helical filament of the
protein flagellin, \qty{20}{nm} thick and several body lengths long,
turned by a rotary motor set in the envelope. The motor is driven not
by ATP but by protons flowing down the electrochemical gradient
across the membrane, at some \num{1000} protons per turn and up to
\num{300} turns per second. In \emph{E.~coli} the several flagella
bundle together when they all turn counter-clockwise and push the
cell in a straight \emph{run}; when one or more reverse, the bundle
flies apart and the cell \emph{tumbles} to a new random direction.
\emph{Chemotaxis} is the bias of this random walk: the cell measures
whether an attractant concentration has risen over the last second
and, if so, suppresses tumbling. It steers by comparing the present
with the recent past --- along its path --- because across its own
\qty{1}{\micro m} of length no gradient is measurable.
\end{proposition}
\begin{proof}[Evidence]
Berg and Brown (1972) followed single \emph{E.~coli} cells in three
dimensions with a tracking microscope: runs of about a second at
\qty{20}{\micro m/s}, tumbles of a tenth of a second, and in a
gradient of serine the runs up the gradient were longer while the
runs down it were not shortened. Tethering a cell to a glass slide by
one flagellum made the whole body spin, which proved the motor is
rotary; cells with the motor uncoupled from ATP synthesis still swam
as long as a proton gradient could be imposed, which identified the
fuel.
\end{proof}

\begin{example}[Cyanobacteria and the heterocyst]\label{ex:b2:unicellular-diversity:cyanobacteria}
Cyanobacteria are the bacteria that photosynthesise as plants do,
splitting water and releasing oxygen on stacked internal membranes
(the thylakoids, which the chloroplast inherited from them). Many
grow as filaments of cells sharing a sheath. In \emph{Anabaena},
when combined nitrogen runs out, every tenth cell or so becomes a
\emph{heterocyst}: it dismantles its oxygen-producing photosystem,
thickens its wall against oxygen, and fixes atmospheric nitrogen with
an enzyme that oxygen would destroy, exporting glutamine to its
neighbours in exchange for sugar. A filament of \emph{Anabaena} is a
first sketch of a division of labour between cells.
\end{example}

\begin{remark}[Dormancy]\label{rem:b2:unicellular-diversity:spores}
Some Gram-positive rods (\emph{Bacillus}, \emph{Clostridium}) answer
starvation by building an \emph{endospore}: a copy of the chromosome
wrapped in several dehydrated coats, metabolically inert, resistant
to boiling, radiation and centuries of drought, which germinates when
conditions return. Many \omterm{def:b2:unicellular-diversity:protist}{protists} do the same with a \emph{cyst}.
Dormancy is the other way of surviving a bad season, alongside
motility.
\end{remark}

\section{Eukaryotic unicellular organisms}

\begin{definition}[Protists: a grade, not a group]\label{def:b2:unicellular-diversity:protist}
\emph{Protists}\index{protist} are the eukaryotes that are neither
animals, nor land plants, nor fungi --- mostly \omterm{def:b2:unicellular-diversity:unicellular}{unicellular}, some
\omterm{def:b2:unicellular-diversity:unicellular}{colonial}, a few (the kelps) \omterm{def:b2:unicellular-diversity:unicellular}{multicellular}. They share the eukaryotic
cell of the Year 1 volume (nucleus, mitochondria, endomembranes,
cytoskeleton, $9+2$ cilia) and nothing else in particular: the
lineages that contain them are as distant from one another as animals
are from plants. Green algae belong with the land plants; diatoms and
brown algae form a lineage of their own; ciliates, dinoflagellates and
the malaria parasite form another; amoebae and slime moulds are
closer to animals and fungi than to any of these. The word names a
level of organisation --- the eukaryotic cell living alone --- and
not a branch of the tree.
\end{definition}

\begin{example}[Four ways to be a eukaryotic cell alone]\label{ex:b2:unicellular-diversity:four}
\emph{Chlamydomonas}: an ovoid green alga with a cellulose-free
glycoprotein wall, one cup-shaped chloroplast, an orange eyespot
that shades a light sensor so that the cell knows which way the
light is as it rotates, and two anterior flagella that beat in a
breaststroke; it is a phototroph. \emph{Paramecium}: a ciliate, its
whole surface beating with cilia in coordinated waves, a mouth-like
oral groove that sweeps bacteria into food vacuoles, two \omterm{prop:b2:unicellular-diversity:osmoregulation}{contractile
vacuoles}, and two kinds of nucleus --- a small germinal
\emph{micronucleus} and a large working \emph{macronucleus} with
hundreds of copies of each gene; it is a phagotroph. \emph{Amoeba}:
no wall, no fixed shape, moving and feeding by \emph{pseudopods}
that actin polymerisation pushes out. The diatom: a photosynthetic
cell enclosed in two overlapping valves of silica like a box and its
lid, so ornamented with pores that the shells were once used to test
microscope lenses; when it divides, each daughter keeps one valve and
makes a new, smaller, inner one.
\end{example}

\begin{definition}[Modes of nutrition]\label{def:b2:unicellular-diversity:nutrition}
\index{phagotrophy}\index{osmotrophy}\index{mixotrophy}
A \omterm{def:b2:unicellular-diversity:unicellular}{unicellular} eukaryote feeds in one of three ways, or in two at
once. \emph{Phototrophy}: it has chloroplasts and makes its own
organic matter from light and $\mathrm{CO_2}$ (green algae, diatoms,
dinoflagellates). \emph{Phagotrophy}: it engulfs particles ---
bacteria, other \omterm{def:b2:unicellular-diversity:protist}{protists} --- into a food vacuole where lysosomal
enzymes digest them (amoebae, ciliates, choanoflagellates).
\emph{Osmotrophy}: it absorbs dissolved molecules across its
membrane, often after secreting enzymes outside (yeasts, many
parasites). \emph{Mixotrophs} such as \emph{Euglena} photosynthesise
in the light and eat in the dark. \omterm{def:b2:unicellular-diversity:prokaryote}{Prokaryotes} cannot phagocytose: the
wall forbids it, which is why a \omterm{def:b2:unicellular-diversity:prokaryote}{prokaryote} never became a predator
by engulfing.
\end{definition}

\begin{proposition}[Osmoregulation in fresh water]\label{prop:b2:unicellular-diversity:osmoregulation}
\index{contractile vacuole}
A freshwater \omterm{def:b2:unicellular-diversity:protist}{protist} is far saltier inside (some \qty{100}{mmol/L}
of solutes) than the pond (about \qty{1}{mmol/L}), and its membrane
lets water through. Water therefore enters continuously by osmosis,
at a rate proportional to the surface area and to the osmotic
pressure difference $\Delta\Pi = RT\,\Delta c$ --- about
\qty{2.4}{bar} for the figures above. A walled cell resists by
turgor; a naked cell would swell and burst within minutes. The
\emph{\omterm{prop:b2:unicellular-diversity:osmoregulation}{contractile vacuole}} is the answer: a reservoir fed by radial
canals that fills with water pumped in by proton-driven transporters
and then contracts, expelling the water through a pore, every few
seconds to every few minutes. A \emph{Paramecium} passes its own
volume of water every half hour; marine relatives, in a medium as
salty as themselves, have no \omterm{prop:b2:unicellular-diversity:osmoregulation}{contractile vacuole} at all.
\end{proposition}

\section{The physics of a small swimmer}

\begin{proposition}[Life at low Reynolds number]\label{prop:b2:unicellular-diversity:reynolds}
\index{Reynolds number}
The \emph{\omterm{prop:b2:unicellular-diversity:reynolds}{Reynolds number}} $Re = \rho v L/\eta$ compares inertial to
viscous forces for a body of size $L$ moving at speed $v$ in a fluid
of density $\rho$ and viscosity $\eta$. For a human swimmer it is
about $10^6$; for \emph{Paramecium} (\qty{200}{\micro m},
\qty{1}{mm/s}) it is 0.2, and for a bacterium $10^{-5}$. Below
$Re \approx 1$ inertia is irrelevant: a cell that stops beating
stops moving within a nanometre, water feels like thick syrup, and a
reciprocal motion (the same stroke forwards and backwards) produces
no net displacement. Small swimmers therefore use non-reciprocal
strokes: the rotating helix of the bacterial flagellum, the
asymmetric beat of a cilium (stiff power stroke, curled recovery
stroke), the breaststroke of \emph{Chlamydomonas}.
\end{proposition}

\begin{theorem}[Stokes' law and sinking]\label{thm:b2:unicellular-diversity:stokes}
\index{Stokes' law}
A sphere of radius $r$ and density $\rho_{\text{cell}}$ moving slowly
through a fluid of viscosity $\eta$ feels a drag $F = 6\pi\eta r v$.
Left to itself in water of density $\rho$ it sinks at the terminal
speed
\[
v_s = \frac{2\,r^2\,(\rho_{\text{cell}} - \rho)\,g}{9\eta},
\]
proportional to the square of its radius: a cell ten times larger
sinks a hundred times faster. This is why the phytoplankton is small,
why diatoms carry spines and oil droplets, and why a
\emph{Chlamydomonas} that stops swimming sinks out of the light in a
few days.
\end{theorem}
\begin{proof}
At terminal speed the drag balances the apparent weight (weight
minus buoyancy): $6\pi\eta r v_s = \tfrac{4}{3}\pi r^3
(\rho_{\text{cell}} - \rho) g$. Solving for $v_s$ gives the formula.
The drag law itself is derived in the physics series for
$Re \ll 1$; here it is taken as given.
\end{proof}

\begin{example}[A diatom's fall]\label{ex:b2:unicellular-diversity:fall}
A centric diatom of radius \qty{10}{\micro m}, \qty{100}{kg/m^3}
denser than sea water ($\eta = \qty{1e-3}{Pa.s}$), sinks at
$v_s = 2\times 10^{-10}\times 100\times 9.81/(9\times 10^{-3}) =
\qty{2.2e-5}{m/s}$, about \qty{2}{m} per day: it would leave the
\qty{50}{m} sunlit layer in under a month. Its populations survive
because turbulence stirs the water, because oil droplets and a low
density trim $\rho_{\text{cell}} - \rho$, and because a cell that
sinks and divides on the way leaves descendants that are lifted
again.
\end{example}

\section{Reproduction and sex in one cell}

\begin{definition}[Modes of division]\label{def:b2:unicellular-diversity:division}
\index{binary fission}\index{budding}\index{multiple fission}
A \omterm{def:b2:unicellular-diversity:unicellular}{unicellular organism} reproduces by dividing. \emph{Binary
fission}: the cell grows to twice its size and splits into two
equal daughters (bacteria, \emph{Paramecium}, \emph{Amoeba}).
\emph{Budding}: a small daughter grows out of the mother and detaches
(yeast; the mother keeps a scar and can bud some twenty times).
\emph{Multiple fission}: the cell grows large, then divides several
times in quick succession inside the mother wall and releases 4, 8,
16 or more daughters at once (\emph{Chlamydomonas}, the malaria
parasite in a red cell). Under constant conditions the number of
cells grows exponentially, $N(t) = N_0\,2^{t/T}$ with $T$ the
\emph{generation time}\index{generation time}: twenty minutes for
\emph{E.~coli} in rich medium, a day for many algae, a week for some
ciliates. (Growth as a function of food is treated in
\cref{ch:b2:microbial-metabolism}.)
\end{definition}

\begin{proposition}[Sex without reproduction: conjugation in \emph{Paramecium}]\label{prop:b2:unicellular-diversity:conjugation}
\index{conjugation!ciliates}
Two \emph{Paramecium} of complementary mating types join by their
oral surfaces. In each, the diploid micronucleus undergoes meiosis to
four haploid nuclei, three of which degenerate; the survivor divides
by mitosis into a stationary and a migratory nucleus, and the two
cells exchange their migratory nuclei across the junction. Each cell
then fuses its stationary nucleus with the one it received, and the
two partners separate, each with a new diploid micronucleus of the
same genotype as its partner's. The old macronucleus breaks down and
a new one is built from the new micronucleus. No new cell was made:
conjugation is genetic mixing --- meiosis and fertilisation,
\cref{ch:b2:meiosis-heredity} --- decoupled from multiplication.
Bacterial conjugation, which transfers a \omterm{def:b2:unicellular-diversity:prokaryote}{plasmid} one way from donor
to recipient, is a different process with the same name
(\cref{ch:b2:genome-diversification}).
\end{proposition}

\section{Toward multicellularity}

\begin{proposition}[The volvocine series]\label{prop:b2:unicellular-diversity:volvox}
\index{Volvox@\emph{Volvox}}\index{multicellularity, origin of}
Among green algae one lineage displays, in living species, the steps
from one cell to a body. \emph{Chlamydomonas} lives alone.
\emph{Gonium} is a flat plate of 4 to 16 identical cells that swim
together. \emph{Pandorina} and \emph{Eudorina} are hollow balls of 16
to 32 cells, still all alike, each able to found a new colony.
\emph{Pleodorina} has some small cells that only swim and never
divide. \emph{Volvox} is a sphere of \num{500} to \num{50000} small
somatic cells, flagellated, which will die with the colony, around a
handful of large germ cells (gonidia) that divide to form daughter
spheres inside the mother. Somatic and germinal cells have appeared:
this is the threshold of multicellularity, crossed in this lineage
some \num{200} million years ago and, independently, at least
twenty-five times elsewhere in the tree of life --- in animals,
plants, fungi, brown algae, red algae and several bacterial groups.
\end{proposition}
\begin{proof}[Evidence]
The molecular phylogeny of the volvocine algae, built from dozens of
genes, places \emph{Chlamydomonas} at the base and \emph{Volvox}
at the tips, with the \omterm{def:b2:unicellular-diversity:unicellular}{colonial} forms between and the number of cells
increasing along the branches; the genome of \emph{Volvox carteri}
(2010) differs from that of \emph{Chlamydomonas} by a few hundred
genes, chiefly those of the extracellular matrix that holds the cells
and of the regulators that silence division in somatic cells. The
transition needed remarkably little new genetic material.
\end{proof}

\begin{remark}[Why divide the labour?]\label{rem:b2:unicellular-diversity:whydivide}
In these algae a cell cannot swim and divide at the same time: the
basal bodies that anchor the flagella are needed to organise the
spindle. A single cell alternates; a small colony can afford to stop
swimming while all its cells divide; a large colony would sink out of
the light (\omterm{thm:b2:unicellular-diversity:stokes}{Stokes' law}, \cref{thm:b2:unicellular-diversity:stokes})
during the hours its division takes. Beyond a certain size, keeping
some cells swimming while others divide is worth the loss of their
descendants --- and a soma is born. The choanoflagellates, collared
flagellates whose cell is a copy of the sponge's feeding cell, show
the same first steps on the branch that led to animals.
\end{remark}

\section{Problem: A Drop of Pond Water}

\begin{problem}[{Weekend problem --- a sample of pond water is counted,
grown, and analysed for the physics that governs its cells, ending on
the water and energy budget of a green alga}]
\label{pb:b2:unicellular-diversity:1}
A pond of \qty{1000}{m^3} is sampled in spring. Under the microscope,
a counting chamber whose grid covers \qty{1}{mm}$\times$\qty{1}{mm}
under a coverslip held \qty{0.1}{mm} above it shows 24 cells of
\emph{Chlamydomonas} (spheres of radius \qty{5}{\micro m},
density \qty{1050}{kg/m^3}). Take $\rho = \qty{1000}{kg/m^3}$,
$\eta = \qty{1e-3}{Pa.s}$, $g = \qty{9.81}{m/s^2}$, $R =
\qty{8.314}{J/mol/K}$, $T = \qty{293}{K}$, $D = \qty{1.9e-9}{m^2/s}$
for $\mathrm{CO_2}$ in water.

\medskip
\noindent\textbf{Part I --- Counting.}
\begin{enumerate}
  \item Compute the volume of water under the grid, in microlitres.
  \item Deduce the density of \emph{Chlamydomonas} in the pond, in
        cells per millilitre.
  \item Compute the volume and the \omterm{prop:b2:unicellular-diversity:surface}{surface-to-volume ratio} of one
        cell.
  \item What fraction of the pond water's volume is occupied by these
        cells?
  \item How many \emph{Chlamydomonas} does the pond hold?
  \item A filamentous cyanobacterium is also present. Give two
        features visible under the microscope that distinguish it
        from the alga.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Growth.}
The sample is kept in the light and counted again after \qty{48}{h}:
\num{384} cells under the grid.
\begin{enumerate}[resume]
  \item Compute the \omterm{def:b2:unicellular-diversity:division}{generation time} $T$.
  \item Compute the growth rate constant $k$ in $N = N_0 e^{kt}$, in
        \unit{h^{-1}}.
  \item How long would it take the pond population to reach
        \qty{1e8}{} cells per millilitre at this rate?
  \item Give three reasons why it will not.
  \item If the cells grow only during the \qty{12}{h} of daylight and
        not at all at night, how long does the same increase take?
  \item \emph{Chlamydomonas} divides by \omterm{def:b2:unicellular-diversity:division}{multiple fission}, releasing
        4, 8 or 16 daughters at once. Explain why a \omterm{def:b2:unicellular-diversity:division}{generation time}
        is nevertheless well defined by the count.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The physics of the cell.}
\begin{enumerate}[resume]
  \item Compute the \omterm{prop:b2:unicellular-diversity:reynolds}{Reynolds number} of a cell swimming at
        \qty{100}{\micro m/s}.
  \item A cell of mass $m$ that stops beating its flagella coasts a
        distance $mv/(6\pi\eta r)$. Compute it.
  \item Compute the sinking speed of a cell that has stopped
        swimming.
  \item Compare the time to sink \qty{1}{m} with the time to swim up
        \qty{1}{m}.
  \item The pond holds \qty{10}{\micro mol/L} of dissolved
        $\mathrm{CO_2}$. Compute the maximal diffusive supply of
        $\mathrm{CO_2}$ to a cell, in molecules per second.
  \item A cell contains \qty{100}{pg} of carbon and doubles it in
        one generation. Compute its mean carbon demand in molecules
        per second, and the ratio of supply to demand.
  \item Redo the comparison for a hypothetical cell of radius
        \qty{50}{\micro m} with the same carbon density. Conclude.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Water and energy.}
The cytoplasm holds \qty{100}{mmol/L} of solutes, the pond
\qty{1}{mmol/L}. The membrane's hydraulic conductivity is $L_p =
\qty{1e-14}{m.s^{-1}.Pa^{-1}}$ (water volume flux per unit area per
unit pressure difference). Hydrolysing one ATP yields
\qty{50}{kJ/mol}; fixing one carbon atom costs three ATP.
\begin{enumerate}[resume]
  \item Compute the osmotic pressure difference across the membrane.
  \item Compute the volume of water entering the cell per second, in
        \unit{\micro m^3/s}.
  \item Without a \omterm{prop:b2:unicellular-diversity:osmoregulation}{contractile vacuole}, how long would the cell take
        to double its volume?
  \item Compute the minimal power needed to expel this water against
        the osmotic pressure.
  \item Convert it to ATP molecules per second and compare with the
