\section*{Chapter \ref{ch:g1:plants-animals} --- Plants and Animals}

\begin{solution}{exo:g1:plants-animals:1}
Answers vary. \omterm{def:g1:plants-animals:plant}{Plants}: rose, apple tree, grass, fern. \omterm{def:g1:plants-animals:animal}{Animals}: dog,
bird, fish, bee.
\end{solution}

\begin{solution}{exo:g1:plants-animals:2}
\omterm{def:g1:plants-animals:animal}{Animal}. It moves by flying and it eats (nectar, other food). It
does not stay rooted or make food from sunlight like a green \omterm{def:g1:plants-animals:plant}{plant}.
\end{solution}

\begin{solution}{exo:g1:plants-animals:3}
Sample: \omterm{def:g1:plants-animals:plant}{Plants} usually stay in one place / make food from light.
\omterm{def:g1:plants-animals:animal}{Animals} usually move / eat food.
\end{solution}

\begin{solution}{exo:g1:plants-animals:4}
False. Mushrooms (and some other \omterm{def:g1:living-nonliving:living}{living things}) are neither green
\omterm{def:g1:plants-animals:plant}{plants} nor \omterm{def:g1:plants-animals:animal}{animals}. At this level it is enough to know that \omterm{def:g1:plants-animals:plant}{plants}
and \omterm{def:g1:plants-animals:animal}{animals} are two big groups, not the only ones.
\end{solution}

\begin{solution}{exo:g1:plants-animals:5}
\omterm{def:g1:plants-animals:plant}{Plants} make their own food using light, air and water. \omterm{def:g1:plants-animals:animal}{Animals} eat
\omterm{def:g1:plants-animals:plant}{plants} or other \omterm{def:g1:plants-animals:animal}{animals}.
\end{solution}
