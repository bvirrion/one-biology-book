\chapter{Perjalanan Makanan}\label{ch:g4:journey-of-food}

Satu gigitan roti: dikunyah, ditelan --- lenyap. Lenyap ke mana?
Menjelang malam ini, sebagian roti itu akan menolong tungkaimu berlari
dan tulangmu tumbuh, yang berarti roti itu entah bagaimana harus
\emph{keluar} dari pipa tempat ia ditelan lalu \emph{masuk} ke bagian
tubuh yang lain. Ikutilah gigitannya: perjalanannya adalah salah satu
kisah terbaik tubuh.

\section{Saluran pencernaan}

\begin{definition}[Saluran pencernaan]\label{def:g4:journey-of-food:tract}
Makanan mengembara melalui \emph{saluran
pencernaan}\index{saluran pencernaan}, satu pipa panjang yang
menembus tubuh:
\begin{enumerate}
  \item \emph{mulut}\index{mulut}, tempat gigi menggilas
        (\cref{ch:g2:teeth-and-eating-well}) dan \emph{air
        liur}\index{air liur} membasahi;
  \item \emph{kerongkongan}\index{kerongkongan}, perosotan dari
        tenggorokan ke lambung;
  \item \emph{lambung}\index{lambung}, sebuah kantong berotot tempat
        makanan diaduk dan dicampur dengan \omterm{def:g4:journey-of-food:digestion}{getah pencernaan} selama
        berjam-jam;
  \item \emph{usus halus}\index{usus halus}, sebuah pipa terlipat
        sepanjang beberapa meter, tempat \omterm{def:g4:journey-of-food:digestion}{pencernaan} diselesaikan dan
        kebaikan makanannya menyeberang ke dalam tubuh;
  \item \emph{usus besar}\index{usus besar}, tempat air diambil
        kembali dari yang tersisa, sebelum sisanya meninggalkan
        tubuh.
\end{enumerate}
\end{definition}

\begin{example}[Menghitung waktu perjalanannya]\label{ex:g4:journey-of-food:timing}
Gigitan roti itu melewatkan beberapa detik di mulut, sekitar dua detik
meluncur menuruni \omterm{def:g4:journey-of-food:tract}{kerongkongan}, beberapa jam diaduk di dalam \omterm{def:g4:journey-of-food:tract}{lambung},
beberapa jam lagi menempuh \omterm{def:g4:journey-of-food:tract}{usus halus}, dan sampai sehari atau lebih
menyelesaikannya di \omterm{def:g4:journey-of-food:tract}{usus besar}. Seluruh perjalanannya lazimnya
memakan waktu satu sampai dua hari.
\end{example}

\section{Pencernaan: dari makanan menjadi zat gizi}

\begin{definition}[Pencernaan dan zat gizi]\label{def:g4:journey-of-food:digestion}
Yang disebut \emph{pencernaan}\index{pencernaan} adalah perubahan
makanan, di sepanjang salurannya, menjadi bagian-bagian yang cukup
kecil dan cukup sederhana untuk masuk ke dalam tubuh: yaitu \emph{zat
gizi}\index{zat gizi}. Gigi dan pengadukan \omterm{def:g4:journey-of-food:tract}{lambung} memecah makanannya;
\emph{getah pencernaan}\index{getah pencernaan} --- \omterm{def:g4:journey-of-food:tract}{air liur} salah
satunya --- membongkarnya lebih jauh, kepingan demi kepingan, sampai
apa yang dahulu roti menjadi campuran cair berisi zat gizi.
\end{definition}

\begin{example}[Rotinya berubah manis]\label{ex:g4:journey-of-food:sweet}
Kunyahlah sepotong roti tawar lama sekali --- hitungan pelan sampai
enam puluh --- dan rasa manis yang samar muncul. Tidak ada yang
ditambahkan: \omterm{def:g4:journey-of-food:tract}{air liur} sudah membongkar rotinya menjadi kepingan yang
lebih sederhana dan lebih manis. Kamu baru saja mengecap permulaan
\omterm{def:g4:journey-of-food:digestion}{pencernaan}.
\end{example}

\begin{proposition}[Menyeberang ke dalam tubuh]\label{prop:g4:journey-of-food:absorb}
Bagian dalam salurannya, dalam satu pengertian, masih berada di luar
tubuh --- seperti lubang sebuah donat. Penyeberangannya terjadi di
\emph{\omterm{def:g4:journey-of-food:tract}{usus halus}}: melalui dindingnya yang tipis dan berlipat dalam,
\omterm{def:g4:journey-of-food:digestion}{zat gizi} menyeberang ke dalam \emph{darah}\index{darah}, yang
membawanya ke setiap bagian tubuh (layanan pengantaran yang
ditelusuri \cref{ch:g4:heart-and-blood}). Apa yang tidak dapat
menyeberang --- serat, sekam --- meneruskan perjalanan, kehilangan
airnya di \omterm{def:g4:journey-of-food:tract}{usus besar}, lalu keluar sebagai tinja.
\end{proposition}
\admitted

\begin{remark}[Mengapa lipatan dan meter-meternya]\label{rem:g4:journey-of-food:surface}
Penyeberangan membutuhkan luas dinding, dan \omterm{def:g4:journey-of-food:tract}{usus halus} dibangun
untuknya: panjangnya bermeter-meter, dan dindingnya dilipat lalu
dilipat lagi di dalamnya. Kalau dibentangkan datar, permukaan
penyeberangannya akan menghampari sebuah kamar kecil --- terkemas di
tengah tubuhmu. Pekerjaan yang besar, di dalam tubuh, memperoleh
permukaan besar yang terlipat kecil.
\end{remark}

\section{Menolong pencernaan mengerjakan tugasnya}

\begin{method}[Kebiasaan yang bersahabat dengan pencernaan]\label{met:g4:journey-of-food:habits}
\begin{enumerate}
  \item Kunyahlah dengan benar: \omterm{def:g4:journey-of-food:tract}{lambung} tidak punya gigi --- apa yang
        dilewatkan mulut, harus diperjuangkan \omterm{def:g4:journey-of-food:tract}{lambung};
  \item makanlah dengan tenang, pada waktu makan yang teratur;
  \item minumlah air ketika makan;
  \item makanlah makanan yang kaya serat --- sayur, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}, roti gandum
        utuh: serat menjaga sisanya tetap bergerak;
  \item tundalah permainan yang berat atau berenang sampai santapannya
        mengendap: \omterm{def:g4:journey-of-food:tract}{lambung} yang bekerja keras dan \omterm{def:g1:my-body:parts}{tungkai} yang bekerja
        keras saling bersaing.
\end{enumerate}
\end{method}

\begin{example}[Detektif sakit perut]\label{ex:g4:journey-of-food:ache}
Makan siang yang dilahap tergesa-gesa, lalu lari cepat: sakit perut
sore hari yang klasik. Santapannya ditelan setengah terkunyah (aturan
1 dilanggar) dan \omterm{def:g4:journey-of-food:tract}{lambung} yang sedang mencerna dipaksa ikut berlomba
(aturan 5). Perbaikannya bukan obat melainkan tata krama --- terhadap
\omterm{def:g4:journey-of-food:tract}{saluran pencernaannya} sendiri.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:1}
Sebutkan bagian-bagian \omterm{def:g4:journey-of-food:tract}{saluran pencernaan} secara berurutan, dari mulut
sampai jalan keluar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:2}
Apa yang terjadi pada makanan di dalam \omterm{def:g4:journey-of-food:tract}{lambung}, dan kira-kira berapa
lama ia tinggal di situ?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:3}
Apa itu \omterm{def:g4:journey-of-food:digestion}{pencernaan}? Apa itu \omterm{def:g4:journey-of-food:digestion}{zat gizi}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:4}
Di bagian mana \omterm{def:g4:journey-of-food:digestion}{zat gizi} menyeberang ke dalam \omterm{prop:g4:journey-of-food:absorb}{darah}? Apa yang terjadi
pada yang tidak dapat menyeberang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:5}
Mengapa roti yang lama dikunyah berubah terasa samar manis?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:6}
Mengapa \omterm{def:g4:journey-of-food:tract}{usus halus} begitu panjang dan begitu berlipat di dalamnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:7}
Berikan tiga kebiasaan yang bersahabat dengan \omterm{def:g4:journey-of-food:digestion}{pencernaan} beserta
alasan masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:8}
Cobalah (di meja makan): hitunglah waktu santapanmu berikutnya. Berapa
lama makananmu berada di dalam mulutmu --- dan bagaimana perbandingannya
dengan hari-hari yang menantinya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:9}
Diagnosislah sakit perut pada \cref{ex:g4:journey-of-food:ache}: dua
kebiasaan mana yang dilanggar, dan apa yang akan kamu ubah besok?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:10}
``Makanan masuk ke dalam tubuh pada saat kamu menelannya.'' Betulkan
kalimat ini dengan \cref{prop:g4:journey-of-food:absorb}: di mana
makanan sungguh-sungguh masuk ke dalam tubuh, dan apa yang harus
terjadi padanya lebih dahulu?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:journey-of-food:11}
Rumput sang sapi adalah makanan yang jauh lebih liat daripada rotimu
--- dan sapi punya empat bilik \omterm{def:g4:journey-of-food:tract}{lambung}, mengunyah makanannya dua kali,
serta menampung \omterm{def:g1:living-or-not:living}{makhluk hidup} penolong di dalam ususnya untuk
membongkar rumputnya. Dengan memakai
\cref{def:g4:journey-of-food:digestion} dan
\cref{prop:g2:what-animals-eat:match}, jelaskan mengapa seorang
pemakan rumput membutuhkan permesinan yang lebih berat ini.
\end{exercise}
