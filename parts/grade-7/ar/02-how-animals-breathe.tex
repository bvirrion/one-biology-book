\chapter{كيف تتنفس الحيوانات}\label{ch:g7:how-animals-breathe}

السلمون المرقط تحت الجسر، ودودة الأرض تحت العشب، والرعّاشة فوقه،
وأنت عليه، كلكم تفعلون الشيء نفسه في هذه الدقيقة: تأخذون \omterm{def:g7:what-is-respiration:respiration}{الأكسجين}
من محيطكم (\cref{def:g7:what-is-respiration:respiration}). لكن
محيطاتكم تختلف --- هواء، \omterm{def:g6:decomposers-and-soil:soil}{وتربة}، وماء --- ويختلف العتاد كذلك. وهذا
الفصل جولة في آلات \omterm{def:g7:what-is-respiration:respiration}{التنفس} عند مملكة \omterm{def:g1:animals-around-us:animal}{الحيوان}.

\section{مشكلة واحدة وأربع آلات}

\begin{proposition}[المواصفة المشتركة]\label{prop:g7:how-animals-breathe:spec}
كل عضو \omterm{def:g7:what-is-respiration:respiration}{تنفس}، مهما يكن صاحبه، مبني على مواصفة واحدة لقيناها من قبل
في \cref{rem:g4:breathing:sponge}: سطح \emph{كبير}
\emph{رقيق} \emph{رطب} يقترب عنده دم \omterm{def:g1:animals-around-us:animal}{الحيوان} (أو سائل جسمه) من
محيطه اقترابًا يكفي لعبور \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} إلى الداخل \omterm{def:g7:what-is-respiration:respiration}{وثاني أكسيد الكربون}
إلى الخارج. كبير ليكفي العبور؛ ورقيق ليسهل العبور؛ ورطب لأن
الغازين يعبران مذابين. والآلات تختلف في كيفية إمساكها بذلك السطح
أمام المحيط.
\end{proposition}
\admitted

\begin{definition}[الآلات الأربع]\label{def:g7:how-animals-breathe:four}
تمسك \omterm{def:g1:animals-around-us:animal}{الحيوانات} سطوح تبادلها بأربع طرق رئيسة:
\begin{enumerate}
  \item \emph{الرئتان}\index{رئة}: جيوب ملأى بالهواء مطوية
        \emph{داخل} الجسم، تهوّيها حركات \omterm{def:g7:what-is-respiration:respiration}{التنفس} --- عند
        \omterm{prop:g3:sorting-living-things:groups}{الثدييات}، \omterm{prop:g3:sorting-living-things:groups}{والطيور}، \omterm{prop:g4:classifying-animals:classes}{والزواحف}، \omterm{prop:g3:sorting-living-things:groups}{والبرمائيات} البالغة؛
  \item و\emph{الغلاصم}\index{غلصمة}: سطوح ريشية ممسوكة
        \emph{خارج الجسم أو عند حافته}، يغسلها تيار من الماء ---
        عند \omterm{prop:g3:sorting-living-things:groups}{الأسماك}
        (\cref{exo:g4:breathing:11})، \omterm{ex:g2:animals-grow-up:frog}{والشراغيف}، وكثير من \omterm{ex:g2:animals-grow-up:butterfly}{يرقات}
        الماء، وبلح البحر؛
  \item و\emph{القصيبات الهوائية}\index{قصيبة هوائية}: شبكة من
        أنابيب هواء دقيقة تمتد من فتحات على امتداد الجسم مباشرة
        إلى كل عضو --- عند \omterm{prop:g3:sorting-living-things:groups}{الحشرات}؛ فالهواء يُوصَّل من باب إلى
        باب، بلا عون \omterm{prop:g4:journey-of-food:absorb}{الدم} في أكثره؛
  \item و\emph{التنفس الجلدي}\index{تنفس جلدي}: \omterm{def:g1:the-five-senses:sense}{الجلد} الرطب كله
        سطحَ تبادل --- عند دودة الأرض كلها؛ وعند \omterm{prop:g3:sorting-living-things:groups}{البرمائيات}
        مكمّلًا كبيرًا لرئتيها البسيطتين.
\end{enumerate}
\end{definition}

\begin{example}[مشاهدة الآلات تعمل]\label{ex:g7:how-animals-breathe:watch}
لكل آلة إيقاعها. ففم السلمون المرقط وأغطية غلاصمه تضخّ بتناوب ثابت
--- ماء يدخل من الفم، فيمرّ على \omterm{def:g7:how-animals-breathe:four}{الغلاصم}، فيخرج من تحت الأغطية.
وبطن الجندب المستريح ينبض --- إذ يضخّ أنابيب هوائه. وحلق الضفدع
يرفّ --- إذ يدفع الهواء إلى رئتيه (فلا قفص صدري يقوم بذلك). ولا
تُظهر دودة الأرض وحدها شيئًا: \omterm{def:g7:how-animals-breathe:four}{فالتنفس الجلدي} لا حركات له البتة،
وهذا بالضبط ما جعل
\cref{def:g7:what-is-respiration:respiration} تميّز التبادل من
الضخّ.
\end{example}

\section{الآلة تناسب الوسط --- وتناسب الحياة}

\begin{proposition}[لماذا كل آلة في مكانها]\label{prop:g7:how-animals-breathe:fits}
تنطبق الآلات على الأوساط والحيوات
(وهو عمل \cref{def:g4:adapted-to-their-world:adaptation} على
\omterm{def:g7:what-is-respiration:respiration}{التنفس}):
\begin{itemize}
  \item \omterm{def:g7:how-animals-breathe:four}{فالغلاصم} تناسب الماء: إذ تطفو سطوحها الريشية متباعدة فيه،
        مغسولة باستمرار --- وتنهار بلا فائدة في الهواء (وهو
        سلمون \cref{exo:g4:breathing:11} المختنق)؛
  \item \omterm{def:g7:how-animals-breathe:four}{والرئتان} تناسبان الهواء: فهما مطويتان في الداخل، تبقيان
        رطبتين حيث كان الماء المكشوف سيُفقد --- لكنهما لا تستطيعان
        حصاد \omterm{def:g7:what-is-respiration:respiration}{أكسجين} الماء المذاب الشحيح؛
  \item \omterm{def:g7:how-animals-breathe:four}{والقصيبات الهوائية} تناسب الأجسام الصغيرة: فهي فائقة على
        مسافات المليمترات، وتوصّل ببطء شديد على المسافات الطويلة
        --- وهو أحد أسباب بقاء \omterm{prop:g3:sorting-living-things:groups}{الحشرات} صغيرة؛
  \item \omterm{def:g7:how-animals-breathe:four}{والتنفس الجلدي} يناسب الحيوات الصغيرة الندية \omterm{def:g1:the-five-senses:sense}{الجلد} في
        الأماكن الندية: فسطح الدودة كله بالكاد يخدم جسمها الرقيق،
        والهواء الجاف يقتلها --- وتذكّر ديدان الأرض المستلقية بعد
        المطر.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[حياتان وآلتان في حيوان واحد]\label{ex:g7:how-animals-breathe:frog}
الضفدع يجري الفصل كله في حياة واحدة: \omterm{def:g7:how-animals-breathe:four}{غلاصم} وهو شرغوف (آلة ساكن
الماء، \cref{ex:g2:animals-grow-up:frog})، ورئتان مع \omterm{def:g1:the-five-senses:sense}{جلد} رطب وهو
بالغ --- \omterm{def:g7:how-animals-breathe:four}{والرئتان} بسيطتان، \omterm{def:g1:the-five-senses:sense}{والجلد} يؤدي كثيرًا من العمل، وخاصة عبر
شتاء قاع البركة البارد
(\cref{exo:g6:life-through-seasons:11}: فذلك الضفدع الخامد يتنفس
بجلده وحده). \omterm{def:g2:animals-grow-up:metamorphosis}{فالتحول} يعيد بناء آلات \omterm{def:g7:what-is-respiration:respiration}{التنفس} أيضًا.
\end{example}

\begin{example}[تنفس الهواء والعيش في الماء]\label{ex:g7:how-animals-breathe:airinwater}
سكان الماء ذوو \omterm{def:g7:how-animals-breathe:four}{الرئتين} عليهم أن يتنقلوا. فالدلفين والحوت يطفوان
لينفخا ويستنشقا
(\cref{ex:g6:classification-kinship:whale}: رئتا ثديي، موروثتان عن
أسلاف يابسة)؛ وحلزون البركة يتسلق إلى السطح ليعيد التعبئة؛ وخنفساء
الغطس تحمل فقاعة هواء فضية تحت غطاءي جناحيها --- خزّان غطّاس،
يُجدَّد عند السطح. \omterm{def:g2:where-animals-live:habitat}{فالموئل} قابل للتفاوض؛ أما وسط الآلة فلا.
\end{example}

\begin{method}[قراءة تنفس حيوان]\label{met:g7:how-animals-breathe:read}
إذا أُعطيت حيوانًا غير مألوف:
\begin{enumerate}
  \item فأين يعيش --- وأيستطيع أن يغادر؟ (ماء، أو هواء، أو \omterm{def:g6:decomposers-and-soil:soil}{تربة}؛
        متنقل أم مقيم)؛
  \item وابحث عن علامات الآلة: أغطية \omterm{def:g7:how-animals-breathe:four}{غلاصم} أو خصلات ريشية؛
        ومنخران وصدر يتحرك؛ وصفوف من الفتحات الصغيرة على جسم
        حشرة؛ أو لا شيء مرئيًّا سوى \omterm{def:g1:the-five-senses:sense}{جلد} رطب؛
  \item وراقب الإيقاع: أغطية تضخّ، أو بطن ينبض، أو صدر يرتفع، أو
        زيارات إلى السطح؛
  \item وسمِّ الآلة --- وراجعها في مقابل الوسط بما في
        \cref{prop:g7:how-animals-breathe:fits}.
\end{enumerate}
\end{method}

\begin{remark}[المواصفة لا تتغير أبدًا]\label{rem:g7:how-animals-breathe:same}
تحت هذا التنوع، افحص أيًّا من الآلات الأربع عن قرب فتظهر
\cref{prop:g7:how-animals-breathe:spec} من جديد: \omterm{def:g1:animals-around-us:covering}{فريش} \omterm{def:g7:how-animals-breathe:four}{الغلصمة}،
وجيوب \omterm{def:g7:how-animals-breathe:four}{الرئة}، وأدقّ \omterm{def:g1:animals-around-us:parts}{أطراف} القصيبات، \omterm{def:g1:the-five-senses:sense}{وجلد} الدودة كلها مواضع لقاء
كبيرة رقيقة رطبة بين المحيط والجسم. أربع آلات ومواصفة واحدة ---
\omterm{prop:g5:first-look-at-cells:principle}{وحدة الحياة}
(\cref{prop:g5:first-look-at-cells:principle})، مكتوبةً في عتاد
\omterm{def:g7:what-is-respiration:respiration}{التنفس}.
\end{remark}

\section{مسألة: إحصاء المتنفسين في البركة}

\begin{problem}[{مسألة نهاية الأسبوع --- بركة واحدة، وكل آلة
تعمل}]\label{pb:g7:how-animals-breathe:1}
أصيل في البركة بشبكة وصندوق رؤية ودفتر. والإحصاء: صغار السلمون
المرقط وأسماك شوكية؛ \omterm{ex:g2:animals-grow-up:frog}{وشراغيف} (بعضها \omterm{def:g1:my-body:parts}{بأرجل} خلفية)؛ وضفادع بالغة؛
وحلزونات بركة تجوب \omterm{def:g6:cell-unit-of-life:parts}{غشاء} السطح؛ وخنفساء غطس تلمع فضةً؛ \omterm{ex:g2:animals-grow-up:butterfly}{ويرقات} ذباب
مايو بخصلات ريشية تحت الحجارة؛ وفي عشب الضفة الندي ديدان أرض.

\medskip\noindent\textbf{القسم الأول --- تسمية الآلات.}
\begin{enumerate}
  \item أعطِ كل فرد من أفراد الإحصاء آلة تنفسه.
  \item خصلات \omterm{ex:g2:animals-grow-up:butterfly}{يرقة} ذباب مايو الريشية تتماوج باستمرار حتى في الماء
        الساكن. فما هي، وعلامَ التماوج؟
  \item وأي أفراد الإحصاء عليهم أن يزوروا السطح، ولماذا هؤلاء
        بالضبط؟
  \item \omterm{ex:g2:animals-grow-up:frog}{الشراغيف} بلا \omterm{def:g1:my-body:parts}{أرجل} وذوات \omterm{def:g1:my-body:parts}{الأرجل} قد تختلف في داخلها بقدر
        اختلافها في ظاهرها. فأي إعادة بناء جارية، وفق
        \cref{ex:g7:how-animals-breathe:frog}؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- الآلات تحت
الضغط.}
\begin{enumerate}[resume]
  \item أسبوع حارّ بلا ريح يخفض \omterm{def:g7:what-is-respiration:respiration}{أكسجين} البركة المذاب
        (وسيشرح \cref{ch:g7:oxygen-in-the-water} السبب). رتّب
        الإحصاء: من يعاني أولًا --- أصحاب \omterm{def:g7:how-animals-breathe:four}{الغلاصم} أم المتنقلون
        إلى السطح؟ ولماذا؟
  \item في ذلك الأسبوع تتعلق \omterm{prop:g3:sorting-living-things:groups}{الأسماك} الشوكية تحت السطح مباشرة،
        وأفواهها تعمل عند \omterm{def:g6:cell-unit-of-life:parts}{الغشاء}. فسّر السلوك بالمواصفة: ما
        الخاصّ في المليمتر الأعلى من الماء؟
  \item خائض مهمل يثير سحبًا من الطين الناعم. فأي آلة تنسدّ،
        وماذا تتوقع أن تفعل \omterm{def:g1:animals-around-us:animal}{الحيوانات} المتضررة؟
  \item وبعد ليلة مطر تستلقي ديدان أرض العشب على الممر. اشرح
        المشهد الكلاسيكي: أي آلة، وأي شرط، دفعها إلى أعلى؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- الخريطة الأكبر.}
\begin{enumerate}[resume]
  \item حرّر الإحصاء جدولًا: \omterm{def:g1:animals-around-us:animal}{الحيوان}، والآلة، والوسط، ومتنقل أم
        مقيم. وأي مواصفة واحدة تتصدّر كل عمود آلة، وفق
        \cref{rem:g7:how-animals-breathe:same}؟
  \item تتجمد البركة في كانون الثاني فيُختم سطحها. تنبأ بمصائر
        الشتاء: الضفادع
        (\cref{exo:g6:life-through-seasons:11})، والحلزونات،
        والخنفساء، \omterm{prop:g3:sorting-living-things:groups}{والأسماك} --- من المتضايق ومن المكفيّ؟
  \item يصل وافد جديد: عنكبوت ماء ينسج جرسًا حريريًّا ويملؤه بهواء
        السطح. ضع حيلته بجانب حيلة الخنفساء، وسمِّ الآلة التي
        يواصل خدمتها.
  \item اختم: لماذا تحتاج بركة صغيرة واحدة إلى أربع آلات \omterm{def:g7:what-is-respiration:respiration}{تنفس}
        مختلفة لتُسكن سكنًا كاملًا؟ في جملة واحدة، مستعملًا كلمتَي
        \emph{الوسط} و\emph{\omterm{def:g7:what-is-respiration:respiration}{الأكسجين}}.
\end{enumerate}
