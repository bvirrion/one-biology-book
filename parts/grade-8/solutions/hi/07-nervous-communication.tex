\section*{अध्याय \ref{ch:g8:nervous-communication} --- तंत्रिका वाला संवाद}

\begin{solution}{exo:g8:nervous-communication:1}
\omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} \omterm{def:g5:brain-in-command:system}{तंत्रिका तंत्र} की काम करती \omterm{def:g5:first-look-at-cells:cell}{कोशिका} है: एक कोशिका-काया और उसके साथ
एक लंबा पतला तंतु --- कुछ \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} से पैर की उँगली तक। और \omterm{def:g8:nervous-communication:neuron}{तंत्रिका} ऐसे
हज़ारों तंतुओं को बाँधने वाली एक केबल है।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:2}
छोटे-छोटे बिजली के इशारे, \qty{100}{m/s} तक की रफ़्तार से। वे इशारे सब
एक जैसे हैं; तीव्रता की कूट-भाषा उनकी दर है --- हलकी थपकियों से लेकर
तेज़ ढोल तक।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:3}
ख़बर वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन}: ख़बर की लाइनें, \omterm{def:g1:the-five-senses:sense}{ज्ञानेंद्रिय} से \omterm{def:g5:brain-in-command:system}{मेरुरज्जु} और \omterm{def:g5:brain-in-command:system}{मस्तिष्क}
तक। गति वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन}: हुक्म की लाइनें, \omterm{def:g5:brain-in-command:system}{मेरुरज्जु} से \omterm{def:g3:bones-and-muscles:muscle}{पेशी} तक। जोड़ने वाले
\omterm{def:g8:nervous-communication:neuron}{न्यूरॉन}: फ़ैसला, \omterm{def:g5:brain-in-command:system}{मस्तिष्क} और \omterm{def:g5:brain-in-command:system}{मेरुरज्जु} के भीतर।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:4}
वह \omterm{def:g5:first-look-at-cells:microscope}{सूक्ष्मदर्शी} ख़ाली जगह, जहाँ एक \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} के तंतु का सिरा अगली \omterm{def:g5:first-look-at-cells:cell}{कोशिका} से
मिलता है। एक रासायनिक संदेशवाहक उसे पार करता है, जुड़ता है, और --- अगर
काफ़ी मात्रा में पहुँचे --- अगले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} में इशारा फिर से शुरू कर देता
है।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:5}
सधने की गुंजाइश (\omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियाँ} इस्तेमाल से मज़बूत होती हैं और बिना
इस्तेमाल मुरझा जाती हैं --- यानी सीखना); एकतरफ़ा आवाजाही (छूटना सिर्फ़
एक ही तरफ़ होता है); और तौलना (हर \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} चलने से पहले हज़ारों उकसाती
तथा दबाती आवाज़ें जोड़ता है)।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:6}
\omterm{def:g5:brain-in-command:system}{तंत्रिकाएँ}: मिलीसेकंड, एक ठीक-ठीक पता, अपने ही तार वाले रास्ते, और असर
इशारे के साथ ही ख़त्म। \omterm{def:g8:puberty:hormones}{हॉर्मोन}: मिनटों से साल, पूरा शरीर, रक्त-धारा का
प्रसारण, और असर तब तक चलता हुआ जब तक संदेशवाहक घूमता रहे।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:7}
\omterm{def:g1:the-five-senses:sense}{आँखों} के ख़बर वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} गेंद की जगहें \omterm{def:g5:brain-in-command:system}{मस्तिष्क} तक ढोल की तरह भेजते
हैं; जोड़ने वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} उन धाराओं को अपनी \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियों} पर तौलते हैं
--- यानी निकाला गया रास्ता जोड़-दर-जोड़ जोड़े गए आवक इशारे ही हैं; और
गति वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} हुक्म बाहर की ओर ढोलते हैं, और उनकी दरें क़दम, झुकाव तथा
हाथ के बंद होने के लिए हर \omterm{def:g3:bones-and-muscles:muscle}{पेशी} की ताक़त तय करती हैं।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:8}
हर सधा हुआ फेर वही इशारे उन्हीं जोड़ों के पार भेजता है, और पार किए गए
जोड़ मज़बूत होते जाते हैं: महीने भर बाद बाज़ीगरी का रास्ता पहले से तुला
हुआ है --- जिसे मेहनत से तौलना पड़ता था वह अब अपने आप चलता है। यह
पक्कापन मज़बूत हुई \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियाँ} ही हैं।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:9}
क्योंकि \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधि} वही जगह है जहाँ इशारा रसायन बन जाता है --- यानी एक
छूटा हुआ संदेशवाहक, जुड़ने की जगहें, और हर आते हुए अणु के लिए खुली एक
ख़ाली जगह। किसी तार की बिजली की नक़ल मुश्किल है; पर किसी रासायनिक हरकारे
की नक़ल की जा सकती है, उसे रोका जा सकता है या चुकाया जा सकता है --- इसलिए
निश्चेतक, विष और नशे, सब वहीं असर करते हैं।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:10}
(क) मिलीसेकंड, एक \omterm{def:g1:my-body:parts}{बाँह} की \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ}, तार वाला रास्ता, और फ़ौरन ख़त्म:
\omterm{def:g8:nervous-communication:neuron}{तंत्रिका} वाला। (ख) साल, पूरा शरीर, रक्त-धारा, और मौसमों तक चलता हुआ:
\omterm{def:g8:puberty:hormones}{हॉर्मोन} वाला। (ग) रिले: \omterm{def:g8:nervous-communication:neuron}{तंत्रिका} वाली पल भर की शुरुआत, और फिर डर के बाद
भी मिनटों तक चौकसी थामे रखता रक्त-धारा का संदेशवाहक --- यानी दोनों, इसी
क्रम में।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:11}
हुक्म जोड़ पर पहुँचता है और पार नहीं कर पाता: \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} को कुछ मिलता ही
नहीं। शिकार धीरे-धीरे लक़वाग्रस्त होता जाता है --- होश में, महसूस करता
हुआ, फ़ैसला करता हुआ --- और फंदा अपनी आख़िरी कड़ी पर कटा हुआ, यानी
हुक्म-से-पेशी पर।
\end{solution}

\begin{solution}{exo:g8:nervous-communication:12}
सीखना: हुनर इस्तेमाल से मज़बूत हुई \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियाँ} हैं --- बाज़ीगर का
महीना जोड़ों पर ही बसता है। फ़ैसला: हर \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन} अपने जोड़ों की आवाज़ें
जोड़कर चलता है या नहीं चलता --- यानी फंदे का ``फ़ैसला'' जोड़ों का गणित
है। कमज़ोरी: निश्चेतक, विष और नशे वहीं असर करते हैं जहाँ इशारा रसायन बन
जाता है --- यानी उसी ख़ाली जगह पर। तार सिर्फ़ ढोते हैं; जोड़ सीखते हैं,
चुनते हैं, और ठगे भी जा सकते हैं।
\end{solution}

\begin{solution}{pb:g8:nervous-communication:1}
\textbf{1.} कंडरा का खिंचाव ख़बर वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉनों} को चलाता है; वे
\emph{\omterm{def:g5:brain-in-command:system}{मेरुरज्जु}} तक जाते हैं, वहीं सीधे गति वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉनों} पर
\omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधि} बनाते हैं, और गति वाला ढोल जाँघ की \omterm{def:g3:bones-and-muscles:muscle}{पेशी} से ठोकर मरवा देता
है। दो तरह के \omterm{def:g8:nervous-communication:neuron}{न्यूरॉन}, एक जोड़, और सब \omterm{def:g5:brain-in-command:system}{मेरुरज्जु} के दर्जे पर --- \omterm{def:g5:brain-in-command:system}{मस्तिष्क}
इसलिए छूट जाता है कि हिफ़ाज़त सोच-विचार का इंतज़ार नहीं कर सकती।

\textbf{2.} कैच वाला फंदा \omterm{def:g1:the-five-senses:sense}{आँख} से \omterm{def:g5:brain-in-command:system}{मस्तिष्क} से \omterm{def:g1:my-body:parts}{बाँह} तक चलता है --- यानी
मीटरों का तंतु और तौलने वाले बहुत-से जोड़; जबकि घुटने की उछाल कंडरा से
\omterm{def:g5:brain-in-command:system}{मेरुरज्जु} से \omterm{def:g3:bones-and-muscles:muscle}{पेशी} तक चलती है --- यानी सेंटीमीटर और असल में एक ही
\omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधि}। छोटा तार, कम जोड़, और दसवाँ हिस्सा समय।

\textbf{3.} फ़ैसला \omterm{def:g5:brain-in-command:system}{मेरुरज्जु} में होता है, इससे पहले कि \omterm{def:g5:brain-in-command:system}{मस्तिष्क} की कोई
मनाही वहाँ पहुँच सके --- और \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियों} की एकतरफ़ा आवाजाही का मतलब
है कि \omterm{def:g5:brain-in-command:system}{मस्तिष्क} का बाद का एतराज़ ख़बर वाली लाइन पर उलटा नहीं दौड़ सकता।
चेतावनी से कुछ नहीं बदलता: यह फंदा मरज़ी के नीचे ही बंद हो जाता है।

\textbf{4.} वह महसूस होना एक दूसरा सफ़र है: उसी चोट की ख़बर \omterm{def:g5:brain-in-command:system}{मेरुरज्जु} से
होकर \omterm{def:g5:brain-in-command:system}{मस्तिष्क} तक भी चढ़ती है --- यानी ज़्यादा जोड़ों वाला एक लंबा रास्ता
--- और वह तब पहुँचती है जब ठोकर इतिहास बन चुकी होती है।

\textbf{5.} आवृत्ति वाली कूट-भाषा: ज़्यादा ज़ोरदार खिंचाव तेज़ ढोल भेजता
है, और गति वाली लाइन भी उसी तरह जवाब देती है --- ऊँची दर से ज़्यादा
ज़ोरदार सिकुड़न, यानी दोनों सिरों पर वही एक-शब्द वाली भाषा।

\textbf{6.} इस बात की कि पूरा परिपथ दुरुस्त है: ख़बर वाली लाइन,
\omterm{def:g5:brain-in-command:system}{मेरुरज्जु} का जोड़, गति वाली लाइन और \omterm{def:g3:bones-and-muscles:muscle}{पेशी} --- और दोनों तरफ़ की बराबरी यह
गवाही देती है कि दोनों पहलू एक जैसे हैं। एक ठोकर चार पुर्ज़ों की जाँच कर
देती है।

\textbf{7.} यह कि तंत्र में कहीं और से जोड़ने वाले \omterm{def:g8:nervous-communication:neuron}{न्यूरॉनों} की आवाज़ें
इस फंदे के जोड़ तक भी पहुँचती हैं और उसकी तैयारी बढ़ा देती हैं: यानी
``बिना \omterm{def:g5:brain-in-command:system}{मस्तिष्क} वाला'' प्रतिवर्त भी पूरे तंत्र से जुड़ा है, और उसकी
\omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधि} कंडरा के अलावा और भी बहुत कुछ सुनती है।

\textbf{8.} \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियाँ} --- और आम तौर पर तंतुओं का इशारा भेजना भी:
निश्चेतक जोड़ों का रसायन चुप करा देता है, और उसके साथ संदेशों की आवाजाही
भी, यानी ठोकर और महसूस होना, दोनों एक साथ।

\textbf{9.} ख़बर (कंडरा का खिंचाव) --- फ़ैसला (\omterm{def:g5:brain-in-command:system}{मेरुरज्जु} का एक जोड़)
--- हुक्म (गति वाली लाइन): यानी वही फंदा, पर उसका बीच वाला क़दम घटकर एक
ही \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधि} रह गया, और \omterm{def:g5:brain-in-command:system}{मस्तिष्क} का तौलना छूट गया।

\textbf{10.} प्रतिवर्त: कंडरा-मेरुरज्जु-पेशी, एक जोड़, सेकंड का पचासवाँ
हिस्सा, दबाया न जा सकने वाला, और प्रैक्टिस से बेपरवाह। कैच:
आँख-मस्तिष्क-बाँह, अनगिनत जोड़, सेकंड का पाँचवाँ हिस्सा, पूरी तरह मरज़ी
वाला, और प्रैक्टिस से बना हुआ।

\textbf{11.} क्योंकि \omterm{def:g5:brain-in-command:system}{मस्तिष्क} के तौलने में वक़्त लगता है, और जलन हर
मिलीसेकंड पर गहरी होती जाती है: ऊतक बचाने वाले फंदे छोटे टाँके जाते हैं,
\omterm{def:g5:brain-in-command:system}{मेरुरज्जु} के दर्जे पर, और \omterm{def:g5:brain-in-command:system}{मस्तिष्क} को बाद में ख़बर करते हैं --- यानी
महसूस देर से, और बचाव पहले ही।

\textbf{12.} ``हथौड़े की एक ही चोट ने वह एक-शब्द वाली आवृत्ति की भाषा,
\omterm{def:g5:brain-in-command:system}{मेरुरज्जु} से होकर जाती दो लाइनों की जुड़ाई, और वह जोड़ दिखा दिया जो अपने
ही मालिक से तेज़ फ़ैसला कर लेता है।''
\end{solution}
