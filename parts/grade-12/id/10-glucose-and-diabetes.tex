\chapter{Glukosa Darah dan Diabetes}\label{ch:g12:glucose-and-diabetes}

Lima liter darah, satu gram glukosa dalam tiap liternya: lima gram
seluruhnya, kira-kira sebuah gula batu. Otak saja membakar sebanyak itu
setiap jam, siang dan malam, dan tak dapat memakai apa pun yang lain.
Namun orang yang tak makan apa-apa seharian, atau memakan sepotong kue
utuh sekaligus, tetap menjaga kepekatannya dalam beberapa persepuluh
gram per liter dari nilai yang sama --- dan orang yang pankreasnya
telah gagal tidak, dengan akibat mulai dari haus sampai koma. Bab ini
tentang angka yang paling ketat diawasi di dalam tubuh, organ yang
menjaganya tetap, dan apa yang terjadi, pada kedua bentuk \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes},
ketika organ itu tak sanggup.

\section{Sebuah tetapan yang diatur}

\begin{definition}[Glikemia]\label{def:g12:glucose-and-diabetes:glycaemia}
\emph{Glikemia}\index{glikemia} adalah kepekatan glukosa di dalam
plasma darah: sekitar \qty{1}{g/L} (\qty{5.5}{mmol/L}) sebelum makan
pada orang sehat, naik menjadi \qty{1.4}{g/L} dalam sejam setelah makan
dan kembali dalam dua jam, lalu turun menjadi \qty{0.7}{g/L} setelah
sehari berpuasa. Di bawah \qty{0.5}{g/L} otak gagal (bingung, lalu
koma); di atas \qty{1.8}{g/L} glukosa meluber ke dalam air seni, dan
bertahun-tahun di atas \qty{1.3}{g/L} merusak pembuluh dan saraf.
Glikemia adalah besaran yang \emph{diatur}: sebuah nilai yang dijaga
dekat titik acuan oleh sistem yang mengukurnya dan membetulkannya.
\end{definition}

\begin{proposition}[Lubuk glukosa dan alirannya]\label{prop:g12:glucose-and-diabetes:pool}
\qty{5}{g} glukosa di dalam darah adalah sebuah lubuk yang dilewati
aliran yang besar. \emph{Masukannya}: usus setelah makan (sampai
\qty{100}{g} dalam sejam), dan di antara waktu makan \emph{hati}, yang
melepaskan glukosa dari simpanan \emph{\omterm{def:g10:exercise-and-energy:glycogen}{glikogennya}} (\qty{100}{g},
pasokan setengah hari) serta membuat glukosa baru dari \omterm{def:g10:chemistry-of-life:families}{asam amino} dan
laktat. \emph{Keluarannya}: otak (\qty{5}{g/h}, tetap), otot (dari
beberapa gram sampai \qty{100}{g/h} saat bekerja), jaringan lain, dan
penyimpanan sebagai \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} di hati serta otot atau sebagai lemak.
Lubuk sebesar \qty{5}{g} yang diisi dan dikuras pada puluhan gram per
jam hanya mantap karena alirannya diatur menit demi menit.
\end{proposition}
\admitted

\begin{example}[Sehari glikemia]\label{ex:g12:glucose-and-diabetes:day}
Sarapan pukul 8: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} naik dari \num{0.9} menjadi \qty{1.3}{g/L}
pada pukul 9 dan kembali ke \num{1.0} pada pukul 10, dengan
kelebihannya masuk ke \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} hati dan otot. Dari pukul 10 sampai 13
hati melepaskan \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} dan tarafnya bertahan dekat \num{0.9}. Lari
pukul 17 menarik \qty{60}{g} ke dalam otot dalam sejam; hati
mengosongkan separuh simpanannya untuk mengimbanginya, dan tarafnya
melandai ke \num{0.8}. Semalaman, hati membuat glukosa baru dari \omterm{def:g10:chemistry-of-life:families}{asam
amino}, dan pukul 7 tarafnya \qty{0.9}{g/L}. Tiga ratus gram telah
melewati sebuah lubuk berisi lima gram.
\end{example}

\section{Pankreas menjaga keseimbangannya}

\begin{proposition}[Dua hormon dari pulau pankreas]\label{prop:g12:glucose-and-diabetes:hormones}
Tersebar di dalam pankreas, di antara \omterm{def:g10:cells-common-unit:cell}{sel} yang membuat \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}
pencernaan, terletak sejuta gugusan kecil \omterm{def:g10:cells-common-unit:cell}{sel}, yaitu \emph{pulau
pankreas}\index{pulau pankreas}. \omterm{def:g10:cells-common-unit:cell}{Selnya} mengukur \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} darah yang
mengalir lewat dan menyekresikan dua \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} ke dalamnya:
\begin{itemize}
  \item \emph{insulin}\index{insulin}, dari \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$, ketika
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} naik: ia membuat \omterm{def:g10:cells-common-unit:cell}{sel} otot dan \omterm{def:g10:cells-common-unit:cell}{sel} lemak menyerap
        glukosa, serta hati dan otot menyimpannya sebagai \omterm{def:g10:exercise-and-energy:glycogen}{glikogen};
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} pun turun;
  \item \emph{glukagon}\index{glukagon}, dari \omterm{def:g10:cells-common-unit:cell}{sel} $\alpha$, ketika
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} turun: ia membuat hati membongkar \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} dan
        melepaskan glukosa; \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} pun naik.
\end{itemize}
Keduanya bekerja pada organ yang sama ke arah yang berlawanan; dan
keseimbangannya, yang disetel ulang tiap beberapa menit oleh \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia}
yang terukur, itulah pengaturannya. \omterm{prop:g12:glucose-and-diabetes:hormones}{Insulin} satu-satunya \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} yang
menurunkan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia}; beberapa \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} menaikkannya.
\end{proposition}
\begin{proof}[Bukti percobaan]
Membuang pankreas seekor anjing (1889) membuatnya \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes} dalam
sehari: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} melipat tiga dan glukosa muncul di dalam air
seninya; mengikat saluran yang membawa \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} pencernaannya tidak
berakibat demikian, jadi pengaruhnya bukan pencernaan. Menyuntikkan
ekstrak \omterm{prop:g12:glucose-and-diabetes:hormones}{pulau pankreas} (1921) menurunkan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} seekor anjing
\omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes}, dan seorang anak \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes}, dalam hitungan jam; zat aktifnya,
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}, dimurnikan pada tahun yang sama. \omterm{prop:g12:glucose-and-diabetes:hormones}{Pulau pankreas} terasing di
dalam cawan menyekresikan \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} ketika glukosa mediumnya dinaikkan
dan \omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon} ketika glukosanya diturunkan. \omterm{prop:g12:glucose-and-diabetes:hormones}{Glukagon} yang disuntikkan
menaikkan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} orang yang berpuasa dalam beberapa menit, dan hanya
bila hatinya memuat \omterm{def:g10:exercise-and-energy:glycogen}{glikogen}.
\end{proof}

\begin{example}[Gelung itu bekerja]\label{ex:g12:glucose-and-diabetes:loop}
Setelah makan, \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} menanjak; dalam beberapa menit \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$
melepaskan \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}, otot dan \omterm{def:g10:cells-common-unit:cell}{sel} lemak membuka pengangkut glukosanya,
hati beralih membuat \omterm{def:g10:exercise-and-energy:glycogen}{glikogen}, dan tarafnya pun jatuh kembali --- dan
pada titik itu sekresi \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} ikut turun. Di antara waktu makan,
hanyutan menurunnya disambut \omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon}, \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya} dibongkar, dan
tarafnya bertahan. Tak satu pun \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} itu pernah benar-benar absen;
nisbahnyalah yang bergerak, dan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} mengikutinya dalam pita yang
sempit.
\end{example}

\section{Ketika gelungnya gagal: diabetes}

\begin{definition}[Diabetes]\label{def:g12:glucose-and-diabetes:diabetes}
\emph{Diabetes}\index{diabetes} adalah kelebihan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} yang menahun:
di atas \qty{1.26}{g/L} saat puasa, atau di atas \qty{2}{g/L} dua jam
setelah beban glukosa baku. Tanda awalnya mengikuti kelebihan itu:
glukosa di dalam air seni, yang menyeret air bersamanya (air seni
berlimpah, haus), dan turunnya berat badan ketika \omterm{def:g10:cells-common-unit:cell}{sel} tak dapat
memakai glukosa di sekelilingnya. Kerusakan lanjutnya --- pada pembuluh
kecil \omterm{def:g11:the-eye:parts}{mata} dan ginjal, pada saraf, pada arteri --- datang dari
bertahun-tahun glukosa bereaksi dengan \omterm{def:g11:gene-expression:protein}{protein}. Dua penyakit berbagi
satu nama ini.
\end{definition}

\begin{proposition}[Tipe 1 dan tipe 2]\label{prop:g12:glucose-and-diabetes:types}
\begin{itemize}
  \item \emph{\omterm{def:g12:glucose-and-diabetes:diabetes}{Diabetes} tipe 1} (sekitar 10\% kasus) adalah rusaknya
        \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$ oleh sistem imun pasiennya sendiri, biasanya pada
        masa kanak atau remaja: \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} tidak ada. \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glikemianya} naik
        tanpa kendali, lemak dibakar karena kekurangan glukosa yang
        dapat dipakai dan produk asamnya menumpuk; tanpa suntikan
        \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} penyakit itu mematikan dalam beberapa bulan.
        Pengobatannya adalah \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}, beberapa kali sehari, dengan
        dosis menurut makanan dan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} yang terukur, seumur hidup.
  \item \emph{\omterm{def:g12:glucose-and-diabetes:diabetes}{Diabetes} tipe 2} (sekitar 90\%) berkembang pada orang
        dewasa, paling sering yang kelebihan berat dan kurang gerak:
        jaringannya makin lama makin kurang menanggapi \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}
        (\emph{resistansi \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}}), \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$ mengimbanginya dengan
        menyekresikan lebih banyak, dan setelah bertahun-tahun \omterm{def:g10:cells-common-unit:cell}{sel} itu
        kehabisan tenaga. \omterm{prop:g12:glucose-and-diabetes:hormones}{Insulinnya} ada tetapi tak mencukupi.
        Kecenderungannya sebagian bersifat genetik
        (\cref{ch:g11:genes-and-disease}); pemicunya adalah cara hidup,
        dan pengobatan pertamanya adalah mengubah cara hidup itu ---
        menurunkan berat badan dan berolahraga --- sebelum obat dan,
        pada tahap lanjut, \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}.
\end{itemize}
\end{proposition}
\begin{proof}[Bukti percobaan]
Pasien tipe 1 punya antibodi terhadap \omterm{def:g10:cells-common-unit:cell}{sel} \omterm{prop:g12:glucose-and-diabetes:hormones}{pulau pankreasnya} sendiri
bertahun-tahun sebelum gejalanya muncul, dan pada saat didiagnosis
\omterm{prop:g12:glucose-and-diabetes:hormones}{pulau pankreasnya} nyaris kosong dari \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$ serta disusupi \omterm{def:g10:cells-common-unit:cell}{sel}
darah putih; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} darahnya tak terdeteksi dan mereka menanggapi
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} suntikan seketika. Pasien tipe 2 punya \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} yang normal atau
tinggi saat didiagnosis; \omterm{def:g10:cells-common-unit:cell}{sel} otot dan \omterm{def:g10:cells-common-unit:cell}{sel} lemaknya menyerap lebih
sedikit glukosa untuk dosis \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} yang sama daripada \omterm{def:g10:cells-common-unit:cell}{sel} orang sehat;
kembar identik berbagi penyakit itu pada 70\% kasus, dan menurunkan
berat badan atau program jalan kaki saja memulihkan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} normal
pada banyak kasus dini.
\end{proof}

\begin{example}[Mengelola sehari dengan tipe 1]\label{ex:g12:glucose-and-diabetes:type1}
Seorang remaja dengan tipe 1 mengukur \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} sebelum tiap makan
dengan setetes darah, menghitung \omterm{def:g10:chemistry-of-life:families}{karbohidrat} di piringnya, dan
menyuntikkan dosis \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} cepat yang sebanding dengannya --- sekitar
satu unit per \qty{10}{g} --- ditambah \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} lambat sekali sehari
untuk latarnya. \omterm{prop:g12:glucose-and-diabetes:hormones}{Insulin} yang terlalu banyak, atau makan yang
terlewat, dan tarafnya jatuh di bawah \qty{0.6}{g/L}: berkeringat,
gemetar, bingung, yang dibetulkan dengan gula dalam beberapa menit.
Terlalu sedikit, dan tarafnya bertahan di atas \qty{2}{g/L} dengan haus
dan letih, dan pembuluhnya menanggung kerusakan diam-diam. Ia sedang
mengerjakan, dengan tangan, apa yang dahulu dikerjakan \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$-nya
dengan \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik}.
\end{example}

\begin{method}[Membaca sebuah rekaman glikemia]\label{met:g12:glucose-and-diabetes:read}
\begin{enumerate}
  \item Tentukan titik acuan dan pitanya: saat puasa \qtyrange{0.7}{1.1}{g/L},
        setelah makan di bawah \num{1.4}; dua jam setelah beban glukosa
        di bawah \num{1.4} (sehat), \qtyrange{1.4}{2}{} (terganggu),
        di atas \qty{2}{g/L} (\omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes}).
  \item Untuk tiap kenaikan, tanyakan apa yang memasukkan glukosa
        (makanan, hati) dan apa yang seharusnya mengeluarkannya
        (sasaran \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}); untuk tiap penurunan, apa yang
        mengeluarkannya (otak, otot) dan apa yang seharusnya
        mengembalikannya (\omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon}, hati).
  \item Bacalah \omterm{def:g11:hormones-and-reproduction:hormone}{hormonnya} bersama glukosanya: \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} seharusnya naik
        bersama \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} dan \omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon} turun; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} yang tak ada
        berarti tipe 1, \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} tinggi dengan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} tinggi berarti
        resistansi.
  \item Ubahlah satuannya bila perlu: \qty{1}{g/L} glukosa setara
        \qty{5.55}{mmol/L}.
\end{enumerate}
\end{method}

\begin{remark}[Sebuah model pengaturan]\label{rem:g12:glucose-and-diabetes:model}
Sensor, titik acuan, dua efektor yang berlawanan, \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} negatif:
kendali \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} berancangan sama dengan kendali testosteron pada
\cref{ch:g11:hormones-and-reproduction}, dan sama dengan rancangan
sebagian besar tetapan tubuh --- suhu, air, tekanan darah, keasaman.
\omterm{def:g12:glucose-and-diabetes:diabetes}{Diabetes} adalah rupa sebuah tetapan yang diatur ketika gelungnya
putus: nilainya masih menanggapi masukan, tetapi tak ada yang
mengembalikannya. Membaca kurva bab ini adalah latihan untuk membaca
kurva mana pun di antaranya.
\end{remark}

\section{Soal: Lima Gram Gula}

\begin{problem}[{Soal akhir pekan --- glukosa darah dipertanggungjawabkan
sepanjang sehari: lubuknya dan alirannya, sebuah uji toleransi dibaca,
sebuah dosis insulin dihitung, dan sebuah gelung yang harus digantikan
dengan tangan}]
\label{pb:g12:glucose-and-diabetes:1}
Ambillah volume darah \qty{5}{L}, pemakaian otak \qty{5}{g/h} glukosa,
simpanan \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} hati \qty{100}{g}, dan \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} otot \qty{400}{g}.
Satu unit \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} cepat menangani sekitar \qty{10}{g} \omterm{def:g10:chemistry-of-life:families}{karbohidrat}.

\medskip
\noindent\textbf{Bagian I --- Lubuknya.}
\begin{enumerate}
  \item Hitunglah massa glukosa di dalam darah pada \qty{1}{g/L}, dan
        kepekatan yang sama dalam \unit{mmol/L} (glukosa
        \qty{180}{g/mol}).
  \item Berapa lama glukosa darah itu akan mencukupi otak saja,
        seandainya tak ada yang menggantikannya?
  \item Di antara waktu makan, hati menggantikannya. Berapa lama
        \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} hati mencukupi otak? Apa yang terjadi sesudahnya?
  \item \omterm{def:g10:exercise-and-energy:glycogen}{Glikogen} otot empat kali \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} hati, tetapi otot tak dapat
        melepaskan glukosa ke darah. Jelaskan mengapa \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} otot tak
        membantu otak, dan untuk apa \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} itu sebenarnya.
  \item Sebuah santapan mengantarkan \qty{60}{g} glukosa selama sejam.
        Jika tak ada yang menyingkirkannya, \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} akan mencapai
        berapa? Apa yang sebenarnya menyingkirkannya, dan ke mana
        perginya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Ujinya.}
Seorang pasien meminum \qty{75}{g} glukosa. \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glikemia} dan \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} yang
terukur:

\begin{center}
\begin{tabular}{lcccccc}
menit & 0 & 30 & 60 & 90 & 120 & 180 \\ \hline
\omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} (g/L) & 1.15 & 1.8 & 2.05 & 2.0 & 1.85 & 1.6 \\
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} (\% puncak orang sehat) & 90 & 140 & 180 & 200 & 190 & 160 \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item Golongkan pasien itu dengan ambang pada
        \cref{met:g12:glucose-and-diabetes:read}.
  \item Apakah \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} kurang? Kalau begitu, apa yang gagal?
  \item Sebutkan nama penyakitnya, dan berikan dua ciri riwayat pasien
        yang kamu harapkan ada.
  \item Bagaimana rupa tabel yang sama bagi pasien tipe 1 tanpa
        pengobatan?
  \item Pengobatan pertama apa yang diresepkan, dan lewat mekanisme apa
        pengobatan itu bekerja?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Menjalankan gelungnya dengan tangan.}
\omterm{def:g12:glucose-and-diabetes:glycaemia}{Glikemia} seorang pasien tipe 1 adalah \qty{1.0}{g/L} sebelum makan
malam; santapannya memuat \qty{90}{g} \omterm{def:g10:chemistry-of-life:families}{karbohidrat}.
\begin{enumerate}[resume]
  \item Hitunglah dosis \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} cepat untuk santapan itu.
  \item Ia menyuntikkannya tetapi hanya memakan separuh santapannya.
        Taksirlah seberapa jauh \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} dapat jatuh, lalu sebutkan
        gejalanya dan penawarnya.
  \item Ia tak menyuntikkan apa pun dan memakan seluruh santapannya.
        Taksirlah \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} seandainya \qty{90}{g} itu semuanya masuk
        ke lubuk \qty{5}{L}, lalu katakan apa yang dilakukan ginjal di
        atas \qty{1.8}{g/L}.
  \item Mengapa ia juga harus menyuntikkan \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} lambat pada malam
        hari, ketika ia tak makan apa pun?
  \item Jelaskan dalam arti apa meter dan pena itu menggantikan \omterm{def:g10:cells-common-unit:cell}{sel}
        $\beta$, dan apa yang tak dapat digantikannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Gelungnya, dinalar.}
\begin{enumerate}[resume]
  \item Sebuah obat merangsang \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$ agar menyekresikan lebih
        banyak \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}. Untuk tipe \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes} yang mana obat itu dapat
        bekerja, dan mengapa tidak untuk yang lain?
  \item Seorang penyandang tipe 2 menurunkan \qty{10}{kg} berat badan
        dan berjalan kaki sejam sehari; enam bulan kemudian uji
        toleransinya normal. Jelaskan bagian gelung mana yang
        diperbaiki.
  \item Jelaskan mengapa \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} dan \omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon} orang sehat tak pernah
        sama-sama tinggi, dan apa yang dilakukan sebuah \omterm{prop:g12:glucose-and-diabetes:hormones}{pulau pankreas}
        di dalam cawan ketika glukosa mediumnya disetel pada
        \qty{1}{g/L}.
  \item \omterm{prop:g12:glucose-and-diabetes:hormones}{Glukagon}, \omterm{prop:g10:heart-lungs-effort:control}{adrenalin}, kortisol dan \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} pertumbuhan semuanya
        menaikkan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia}; hanya \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} yang menurunkannya. Usulkan
        mengapa tubuh punya beberapa pengaman terhadap \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} rendah
        dan satu terhadap \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} tinggi.
  \item Nyatakan hasilnya: glukosa di dalam darah, berapa jam yang
        dapat ditutup hati, diagnosis Bagian II, dan dosis Bagian III.
\end{enumerate}
\end{problem}
