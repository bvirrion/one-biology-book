\section*{Bab \ref{ch:g12:glucose-and-diabetes} --- Glukosa Darah dan Diabetes}

\begin{solution}{exo:g12:glucose-and-diabetes:1}
Sekitar \qty{1}{g/L}, yaitu \qty{5.5}{mmol/L}. Di bawah \qty{0.5}{g/L}
otak gagal; di atas \qty{1.8}{g/L} glukosa lolos ke air seni (dan
bertahun-tahun di atas \qty{1.3}{g/L} merusak pembuluhnya).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:2}
\omterm{prop:g12:glucose-and-diabetes:hormones}{Insulin}, dari \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$, menurunkan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} (penyerapan dan
penyimpanan oleh otot, lemak dan hati); \omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon}, dari \omterm{def:g10:cells-common-unit:cell}{sel} $\alpha$,
menaikkannya (pelepasan glukosa oleh hati).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:3}
Hati, sebagai \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} (dan hati juga dapat membuat glukosa baru dari
\omterm{def:g10:chemistry-of-life:families}{asam amino} serta laktat).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:4}
Tipe 1: \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$-nya rusak dan \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} tidak ada. Tipe 2:
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} ada tetapi jaringannya kurang menanggapinya, dan \omterm{def:g10:cells-common-unit:cell}{sel}
$\beta$-nya akhirnya kehabisan tenaga.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:5}
Sehat sekitar \qty{0.95}{g/L}; toleransi terganggu sekitar \qty{1.7}{g/L}
(antara 1.4 dan 2); tipe 2 sekitar \qty{2.7}{g/L} (di atas 2).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:6}
Hati melepaskan glukosa dari \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya} pada laju yang sama dengan
laju otak menyingkirkannya, di bawah kendali \omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon}, sehingga
lubuknya bertahan dekat \qty{5}{g} walaupun isinya diperbarui tiap jam.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:7}
Salurannya membawa \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} pencernaan ke usus; menyumbatnya membiarkan
\omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} normal, jadi pengaruh pankreas terhadap glukosa bukan lewat
pencernaan. Membuang seluruh organnya berarti membuang \omterm{prop:g12:glucose-and-diabetes:hormones}{pulau
pankreasnya}: \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} dibuat di dalam pankreas dan mencapai sasarannya
lewat darah, bukan lewat salurannya.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:8}
\omterm{prop:g12:glucose-and-diabetes:hormones}{Glukagon} bekerja dengan membuat hati membongkar \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya}; setelah
puasa semalam simpanannya masih ada dan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} naik; setelah tiga
hari \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya} habis dan \omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon} tak punya apa pun untuk dilepaskan.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:9}
Di atas sekitar \qty{1.8}{g/L} ginjal tak dapat menyerap kembali semua
glukosa yang disaringnya; glukosa lolos ke air seni dan, secara
osmosis, menyeret air bersamanya. Air yang hilang harus diganti: haus.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:10}
\omterm{prop:g12:glucose-and-diabetes:hormones}{Insulinnya} mendorong glukosa masuk ke otot dan \omterm{def:g10:cells-common-unit:cell}{sel} lemak serta
menghentikan pelepasan oleh hati, tetapi tak ada glukosa yang masuk:
\omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} jatuh di bawah \qty{0.6}{g/L} dalam satu-dua jam, disertai
keringat, gemetar dan bingung; gula yang segera diambil membetulkannya.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:11}
Otot yang bekerja menyerap glukosa dari darah lewat jalan yang tak
memerlukan \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}, dan otot terlatih memulihkan kepekaannya terhadap
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} di antara sesi latihan; \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} turun selama dan setelah
olahraga, dan resistansi jaringan, yang menjadi sebab penyakitnya, pun
berkurang.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:12}
Tanpa \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}: tipe 1, \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$ \omterm{prop:g12:glucose-and-diabetes:hormones}{pulau pankreasnya} rusak dan tak
menyekresikan apa-apa. \omterm{prop:g12:glucose-and-diabetes:hormones}{Insulin} tiga kali normal: tipe 2, \omterm{prop:g12:glucose-and-diabetes:hormones}{pulau
pankreasnya} menyekresikan sekuat tenaga melawan jaringan yang tak lagi
menanggapi.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:13}
Sehat: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} naik menjadi sekitar \qty{1.3}{g/L} pada menit
30--60 lalu kembali pada jam ke-2; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} naik bersamanya dan turun
bersamanya. Tipe 1 tanpa pengobatan: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} menanjak di atas
\qty{2}{g/L} lalu bertahan; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} datar di nol --- tak ada yang
menyingkirkan glukosanya. Tipe 2: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} naik tinggi lalu turun
pelan; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} naik lebih tinggi daripada normal dan bertahan
tinggi --- banyak \omterm{def:g11:hormones-and-reproduction:hormone}{hormon}, sedikit pengaruh.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:14}
$5 + 80 = \qty{85}{g}$ di dalam \qty{5}{L}: \qty{17}{g/L}. Hal itu tak
pernah terjadi karena penyerapannya memakan satu sampai dua jam, dan
begitu glukosanya tiba \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} mengirimkannya ke \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} hati,
\omterm{def:g10:exercise-and-energy:glycogen}{glikogen} otot dan lemak, sementara otak dan jaringannya membakar
sebagian: lubuknya tak pernah memuat lebih dari beberapa gram di atas
normal.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:15}
Keduanya punya: sebuah sensor (\omterm{def:g10:cells-common-unit:cell}{sel} \omterm{prop:g12:glucose-and-diabetes:hormones}{pulau pankreas}; hipotalamus dan
hipofisis), sebuah titik acuan, efektor, dan \omterm{def:g11:hormones-and-reproduction:hormone}{umpan balik} negatif yang
menyingkirkan isyaratnya. \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glikemia} punya dua \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} berlawanan yang
bekerja pada organ yang sama, sehingga peubahnya dapat didorong naik
maupun turun secara aktif; gelung testosteron punya satu \omterm{def:g11:hormones-and-reproduction:hormone}{hormon} dan
membetulkannya terutama dengan lebih banyak atau lebih sedikit \omterm{def:g11:hormones-and-reproduction:hormone}{hormon}
itu. Dua efektor yang berlawanan memberi kendali yang lebih cepat dan
lebih ketat bagi peubah yang berubah cepat ke dua arah.
\end{solution}

\begin{solution}{pb:g12:glucose-and-diabetes:1}
\textbf{1.} \qty{5}{g}; $\qty{1}{g/L} \div \qty{180}{g/mol} = \qty{5.55}{mmol/L}$.

\textbf{2.} Satu jam.

\textbf{3.} $100/5 = 20$ jam. Sesudah itu hati membuat glukosa baru
dari \omterm{def:g10:chemistry-of-life:families}{asam amino} (dan laktat), dengan ongkos \omterm{def:g11:gene-expression:protein}{protein} tubuhnya.

\textbf{4.} \omterm{def:g10:cells-common-unit:cell}{Sel} otot tak punya \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} yang melepaskan glukosa bebas dari
\omterm{def:g10:exercise-and-energy:glycogen}{glikogen} ke dalam darah; \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya} dibakar di tempat, untuk
kontraksinya sendiri.

\textbf{5.} $5 + 60 = \qty{65}{g}$ di dalam \qty{5}{L}: \qty{13}{g/L}.
\omterm{prop:g12:glucose-and-diabetes:hormones}{Insulin} mengirimkan glukosanya ke \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} hati dan otot serta ke
lemak, dan jaringannya membakar sebagian; \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemianya} memuncak sekitar
\qty{1.4}{g/L}.

\textbf{6.} Saat puasa \qty{1.15}{g/L} (di bawah 1.26) tetapi
\qty{1.85}{g/L} pada menit ke-120 (antara 1.4 dan 2): toleransi
terganggu, di ambang \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes}.

\textbf{7.} Tidak: \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} dua kali puncak orang sehat. Tanggapan
jaringan terhadapnyalah yang gagal --- resistansi \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}.

\textbf{8.} Tipe 2 (atau tahap peringatannya). Yang diharapkan: dewasa,
kelebihan berat, kurang gerak, sering dengan kerabat penyandang \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes}.

\textbf{9.} \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glikemia} yang mulai di atas \qty{1.5}{g/L}, naik di atas
\num{2.5} dan tak turun; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} dekat nol sepanjang waktu.

\textbf{10.} Menurunkan berat badan dan berolahraga teratur: keduanya
memulihkan kepekaan jaringan terhadap \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} dan membiarkan otot yang
bekerja menyerap glukosa tanpa \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}.

\textbf{11.} $90/10 = 9$ unit.

\textbf{12.} Dosisnya menangani \qty{90}{g} tetapi hanya \qty{45}{g}
yang tiba: kelebihan \omterm{prop:g12:glucose-and-diabetes:hormones}{insulinnya} menurunkan \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} setara \qty{45}{g}
di dalam lubuk dan simpanannya --- jauh di bawah \qty{0.6}{g/L}:
berkeringat, gemetar, bingung; gula seketika.

\textbf{13.} $(5 + 90)/5 = \qty{19}{g/L}$ seandainya tak ada yang
disingkirkan; kenyataannya beberapa \unit{g/L}. Di atas \qty{1.8}{g/L}
ginjal meloloskan glukosa ke air seni, beserta airnya: air seni
berlimpah dan haus.

\textbf{14.} Di antara waktu makan, hati akan melepaskan glukosa tanpa
kendali (\omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon} bekerja, tak ada yang melawannya) dan jaringannya
akan membakar lemak dengan produk asamnya: latar \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} diperlukan
untuk keseimbangan dasarnya, bukan hanya untuk waktu makan.

\textbf{15.} Meternya mengukur \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} seperti dahulu dikerjakan \omterm{def:g10:cells-common-unit:cell}{sel}
$\beta$; penanya mengantarkan \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} yang dahulu disekresikannya.
Keduanya tak dapat menggantikan penyetelan yang malar dari menit ke
menit: pasiennya mengukur beberapa kali sehari lalu menaksir, sedangkan
\omterm{def:g10:cells-common-unit:cell}{selnya} mengukur terus-menerus dan menanggapi dengan tepat.

\textbf{16.} Tipe 2, yang \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$-nya masih ada dan dapat didorong.
Pada tipe 1 tak ada \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$ yang dapat dirangsang.

\textbf{17.} Tanggapan jaringan terhadap \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} --- sisi efektor
gelungnya: dengan lemak yang lebih sedikit dan otot yang aktif,
isyarat \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} yang normal kembali menghasilkan penyerapan yang
normal, dan \omterm{def:g10:cells-common-unit:cell}{sel} $\beta$-nya pun lega.

\textbf{18.} Masing-masing disekresikan sebagai tanggapan terhadap
penyimpangan yang berlawanan dari peubah yang sama: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glikemia} yang
tinggi memanggil \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} dan membungkam \omterm{prop:g12:glucose-and-diabetes:hormones}{glukagon}, yang rendah
sebaliknya. Pada \qty{1}{g/L} pulau itu menyekresikan keduanya pada
laju rendah yang seimbang --- keadaan istirahatnya.

\textbf{19.} \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glikemia} yang rendah membunuh otak dalam beberapa menit,
yang tinggi merusak pelan-pelan sepanjang tahun: bahaya akutnya punya
beberapa penjaga yang saling bebas, yang lambat hanya punya satu ---
itulah sebabnya gagalnya \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} saja sudah cukup menimbulkan sebuah
penyakit, dan itulah sebabnya \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes} begitu umum.

\textbf{20.} Sekitar \qty{5}{g} glukosa di dalam darah; \omterm{def:g10:exercise-and-energy:glycogen}{glikogen} hati
menutup sekitar 20 jam; pasien Bagian II bertoleransi terganggu di tepi
\omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes} tipe 2; dan dosis Bagian III adalah 9 unit.
\end{solution}
