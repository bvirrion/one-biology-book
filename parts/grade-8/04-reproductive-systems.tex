\chapter{The Reproductive Systems}\label{ch:g8:reproductive-systems}

\Cref{ch:g8:sexual-reproduction} needed \omterm{def:g8:sexual-reproduction:gametes}{gametes};
\cref{ch:g8:puberty} switched their factories on. This
chapter maps the factories themselves: the \omterm{def:g4:animal-reproduction:sexual}{male} and \omterm{def:g4:animal-reproduction:sexual}{female}
reproductive systems --- the organs, the \omterm{def:g8:sexual-reproduction:gametes}{gametes} each
produces, and the \omterm{def:g4:animal-reproduction:sexual}{female} system's remarkable monthly
rhythm. Plain anatomy, plainly named: these are organs like
\omterm{def:g4:heart-and-blood:heart}{heart} and \omterm{def:g7:how-animals-breathe:four}{lungs}, and this is their chapter.

\section{The male system}

\begin{definition}[The male reproductive system]\label{def:g8:reproductive-systems:male}
The \omterm{def:g4:animal-reproduction:sexual}{male} system's organs:
\begin{itemize}
  \item the two \emph{testes}\index{testis}, held outside
        the body's warmth in the \emph{scrotum} --- the
        \omterm{def:g8:sexual-reproduction:gametes}{gamete} factories: from \omterm{def:g8:puberty:puberty}{puberty} on they produce
        \emph{sperm cells}\index{sperm cell} continuously,
        by the hundreds of millions;
  \item ducts that store and carry the sperm, joined on
        the way by glands whose fluids feed and carry
        them --- together making \emph{semen};
  \item the \emph{penis}\index{penis}, through which semen
        \omterm{def:g1:plants-around-us:parts}{leaves} the body --- the same organ, on its other
        duty, carrying \omterm{def:g7:removing-wastes:kidneys}{urine}
        (\cref{def:g7:removing-wastes:kidneys}); a valve
        arrangement keeps the two duties strictly apart.
\end{itemize}
\end{definition}

\begin{example}[The sperm cell, up close]\label{ex:g8:reproductive-systems:sperm}
Under the \omterm{def:g5:first-look-at-cells:microscope}{microscope}
(\cref{def:g5:first-look-at-cells:microscope}), a \omterm{def:g8:reproductive-systems:male}{sperm
cell} is a \omterm{def:g5:first-look-at-cells:cell}{cell} stripped for travel: a head that is almost
all \omterm{def:g6:cell-unit-of-life:parts}{nucleus} --- the father's contribution, packed
(\cref{rem:g5:first-look-at-cells:nucleus}) --- and a
lashing tail. Smallest \omterm{def:g5:first-look-at-cells:cell}{cell} in the body, built to swim,
provisioned for days. Production runs from \omterm{def:g8:puberty:puberty}{puberty} into
\omterm{def:g3:stages-of-life:stages}{old age}: the \omterm{def:g4:animal-reproduction:sexual}{male} factory is continuous.
\end{example}

\section{The female system}

\begin{definition}[The female reproductive system]\label{def:g8:reproductive-systems:female}
The \omterm{def:g4:animal-reproduction:sexual}{female} system's organs, all within the lower belly:
\begin{itemize}
  \item the two \emph{ovaries}\index{ovary} --- the \omterm{def:g8:sexual-reproduction:gametes}{gamete}
        stores and factories: unlike the \omterm{def:g8:reproductive-systems:male}{testes}, they hold
        their stock of future \emph{\omterm{def:g4:animal-reproduction:sexual}{egg cells}} from \omterm{def:g2:born-grow-die:cycle}{birth},
        and from \omterm{def:g8:puberty:puberty}{puberty} release them one at a time;
  \item the two \emph{oviducts}\index{oviduct}, fringed
        tubes that catch the released \omterm{def:g4:animal-reproduction:sexual}{egg cell} --- the
        usual meeting place of
        \cref{prop:g8:sexual-reproduction:core}'s step 2;
  \item the \emph{uterus} (the \omterm{prop:g5:human-reproduction:plan}{womb} of
        \cref{prop:g5:human-reproduction:plan}), whose
        thick muscular wall and renewable lining are the
        \omterm{prop:g8:sexual-reproduction:core}{zygote}'s lodging;
  \item the \emph{vagina}, connecting the uterus to the
        outside --- where semen is received, and the
        \omterm{def:g2:born-grow-die:cycle}{birth} canal of \cref{ex:g5:human-reproduction:birth}.
\end{itemize}
\end{definition}

\begin{example}[The egg cell, up close]\label{ex:g8:reproductive-systems:egg}
The \omterm{def:g4:animal-reproduction:sexual}{egg cell} is the sperm's opposite in everything but
role: the body's \emph{largest} \omterm{def:g5:first-look-at-cells:cell}{cell} --- a tenth of a
millimetre, at the \omterm{def:g1:the-five-senses:sense}{eye}'s edge --- round, unmoving,
stocked with provisions for the journey's first days
(the yolk principle of \cref{rem:g2:animals-grow-up:egg},
at \omterm{def:g5:first-look-at-cells:cell}{cell} scale). One factory runs continuous millions; the
other releases a single provisioned \omterm{def:g5:first-look-at-cells:cell}{cell} a month:
\cref{prop:g4:animal-reproduction:strategies}'s
numbers-against-care, written into the \omterm{def:g8:sexual-reproduction:gametes}{gametes}
themselves.
\end{example}

\section{The cycle}

\begin{proposition}[The menstrual cycle]\label{prop:g8:reproductive-systems:cycle}
From \omterm{def:g8:puberty:puberty}{puberty} to about fifty, the \omterm{def:g4:animal-reproduction:sexual}{female} system runs a
repeating program of roughly \emph{28 days} --- the
\emph{menstrual cycle}\index{menstrual cycle}, ordered by
the \omterm{def:g8:puberty:hormones}{hormone} chain of \cref{def:g8:puberty:hormones}:
\begin{enumerate}
  \item the uterus lining rebuilds itself, thick and
        blood-rich --- the lodging prepared;
  \item mid-cycle, around day 14, an \omterm{def:g8:reproductive-systems:female}{ovary} releases one
        \omterm{def:g4:animal-reproduction:sexual}{egg cell} --- \emph{ovulation}\index{ovulation};
        the \omterm{def:g8:reproductive-systems:female}{oviduct} catches it, and for about a day it can
        be fertilized;
  \item no \omterm{def:g5:flower-to-fruit:steps}{fertilization}: the lining is not needed; it
        breaks down and \omterm{def:g1:plants-around-us:parts}{leaves} through the vagina over a
        few days --- the \emph{menstrual
        period}\index{menstrual period} --- and the
        program restarts;
  \item \omterm{def:g5:flower-to-fruit:steps}{fertilization}: the \omterm{prop:g8:sexual-reproduction:core}{zygote} settles into the
        prepared lining
        (\cref{def:g5:human-reproduction:pregnancy}), the
        cycle suspends, and \omterm{def:g5:human-reproduction:pregnancy}{pregnancy} begins.
\end{enumerate}
\end{proposition}
\admitted

\begin{remark}[Living with the cycle]\label{rem:g8:reproductive-systems:living}
Practicalities, said plainly: cycles are irregular for the
first years --- the machinery is tuning; length varies
from person to person and month to month; periods can
bring cramps (the uterus is a \omterm{def:g3:bones-and-muscles:muscle}{muscle},
\cref{def:g3:bones-and-muscles:muscle}, and it works
during shedding) --- warmth, movement and, when needed,
the school nurse's advice all help; and pads, tampons and
cups are simply the period's practical equipment, a
normal aisle of every pharmacy. None of this is illness:
the cycle is the system running exactly as built.
\end{remark}

\begin{method}[Reading the two systems side by side]\label{met:g8:reproductive-systems:sides}
To keep the map straight, compare by function:
\begin{enumerate}
  \item \omterm{def:g8:sexual-reproduction:gametes}{gamete} factory: \omterm{def:g8:reproductive-systems:male}{testes} --- \omterm{def:g8:reproductive-systems:female}{ovaries};
  \item \omterm{def:g8:sexual-reproduction:gametes}{gamete}: continuous swimming millions --- one
        provisioned \omterm{def:g4:animal-reproduction:sexual}{egg cell} a month;
  \item meeting place: the \omterm{def:g8:reproductive-systems:female}{oviduct}, reached by the sperm's
        swim from the vagina through the uterus;
  \item lodging: none in the \omterm{def:g4:animal-reproduction:sexual}{male} map --- the uterus,
        monthly prepared, in the \omterm{def:g4:animal-reproduction:sexual}{female};
  \item timetable: continuous from \omterm{def:g8:puberty:puberty}{puberty} --- cyclic,
        \omterm{def:g8:puberty:puberty}{puberty} to about fifty.
\end{enumerate}
\end{method}

\section{Problem: The Clinic's Teaching Models}

\begin{problem}[{Weekend problem --- explaining the
systems with two models and a calendar}]\label{pb:g8:reproductive-systems:1}
A health center's open day uses three teaching props: a
male-system model, a female-system model, and a large
28-day cycle calendar. You are the volunteer explainer.

\medskip\noindent\textbf{Part I --- The two models.}
\begin{enumerate}
  \item A visitor traces the \omterm{def:g4:animal-reproduction:sexual}{male} model from \omterm{def:g8:reproductive-systems:male}{testes} to
        exit. Narrate the route, naming what is added on
        the way and the double-duty organ's arrangement.
  \item On the \omterm{def:g4:animal-reproduction:sexual}{female} model, point out where the \omterm{def:g8:sexual-reproduction:gametes}{gamete}
        stock waits, where \omterm{def:g5:flower-to-fruit:steps}{fertilization} usually
        happens, and where a \omterm{def:g5:human-reproduction:pregnancy}{pregnancy} lodges.
  \item A visitor asks why the models' \omterm{def:g8:sexual-reproduction:gametes}{gamete} factories
        differ so sharply --- millions daily against one
        a month. Answer with
        \cref{ex:g8:reproductive-systems:egg}'s closing
        sentence.
  \item Another asks what the systems were doing before
        \omterm{def:g8:puberty:puberty}{puberty}. Answer with \cref{ch:g8:puberty}'s
        machinery in one line.
\end{enumerate}

\medskip\noindent\textbf{Part II --- The calendar.}
\begin{enumerate}[resume]
  \item Walk the calendar: place the period, the
        rebuild, \omterm{prop:g8:reproductive-systems:cycle}{ovulation} and the fertile day, with
        their rough dates.
  \item A visitor assumes day 28 is the same for
        everyone. Correct the assumption with
        \cref{rem:g8:reproductive-systems:living} ---
        two variabilities.
  \item Show on the calendar what changes if
        \omterm{def:g5:flower-to-fruit:steps}{fertilization} happens near day 14 --- which
        events are cancelled, which begin
        (\cref{def:g5:human-reproduction:pregnancy}
        continues the story).
  \item A teenager asks, quietly, whether cramps mean
        something is wrong. Give the \omterm{def:g3:bones-and-muscles:muscle}{muscle} answer and
        the two helps --- and the address for anything
        beyond them.
\end{enumerate}

\medskip\noindent\textbf{Part III --- The explainer's
craft.}
\begin{enumerate}[resume]
  \item Visitors giggle at the models. Your opening
        sentence treats the systems as
        \cref{ch:g8:reproductive-systems}'s first
        paragraph does. Write it.
  \item Summarize the two systems in
        \cref{met:g8:reproductive-systems:sides}'s
        five-line comparison, from memory.
  \item A visitor asks where babies' beginnings fit on
        the props. Chain the three props into
        \cref{prop:g8:sexual-reproduction:core}'s three
        steps.
  \item Close the open day with one sentence on why
        health centers teach this anatomy at all ---
        knowledge against legend, as
        \cref{ex:g8:puberty:questions} argued.
\end{enumerate}
\end{problem}
