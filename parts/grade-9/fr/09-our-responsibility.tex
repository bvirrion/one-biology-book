\chapter{Notre responsabilité}\label{ch:g9:our-responsibility}

Neuf années de ce livre s'achèvent ici, et ses fils se
rejoignent sur un fait tout simple~: l'\omterm{def:g5:biodiversity:species}{espèce} capable de
lire cette page est la seule \omterm{def:g5:biodiversity:species}{espèce} qui comprenne la
machinerie à laquelle elle appartient --- et comprendre,
comme le disait déjà \cref{ch:g8:contraception}, donne le
choix. Ce chapitre de clôture rassemble les responsabilités
que cette compréhension crée~: envers sa propre santé,
envers celle des autres, et envers la toile des \omterm{def:g6:exploring-our-environment:components}{êtres
vivants} qui nous porte tous.

\section{La responsabilité envers sa propre machinerie}

\begin{proposition}[La santé comme gestion de soi éclairée]\label{prop:g9:our-responsibility:self}
Les fils «~santé~» du livre, noués~:
\begin{itemize}
  \item le contrat quotidien
        (\cref{prop:g5:healthy-choices:contract}) ---
        manger, bouger, dormir, se laver --- est un
        entretien d'organes que tu peux désormais justifier
        système par système
        (\cref{prop:g7:heart-and-circulation:keep},
        \cref{met:g7:lungs-and-gas-exchange:keep},
        \cref{met:g8:protecting-the-nervous-system:manual})~;
  \item les mécanismes des substances sont connus
        (\cref{prop:g8:protecting-the-nervous-system:substances}),
        et l'asymétrie du piège est chiffrée
        (\cref{rem:g5:healthy-choices:never})~;
  \item les choix de la reproduction ont leur carte et
        leurs adresses (\cref{ch:g8:contraception})~;
  \item et l'histoire du corps est en partie distribuée
        (\cref{ch:g9:heredity-and-traits})~: connaître les
        tendances d'une famille --- les cheveux blancs
        précoces, les \omterm{def:g4:heart-and-blood:heart}{cœurs} qui s'encrassent --- n'est pas
        un destin, mais un calendrier d'entretien
        (\cref{met:g9:unique-individuals:claims}~: des
        marges, vécues).
\end{itemize}
\end{proposition}
\admitted

\begin{example}[La visite médicale, lue comme de la biologie]\label{ex:g9:our-responsibility:checkup}
Une visite médicale de routine, c'est ce livre en un
rendez-vous~: le \omterm{prop:g4:heart-and-blood:pulse}{pouls} et la tension (le réseau lu au bord
de la route, \cref{ex:g7:heart-and-circulation:measures}),
le marteau à réflexes
(\cref{pb:g8:nervous-communication:1}), le carnet de
\omterm{prop:g9:immune-defenses:vaccination}{vaccination} (\cref{pb:g9:immune-defenses:1}), les
antécédents familiaux (les tendances distribuées), et les
conseils habituels --- le contrat, encore. La médecine est
pour l'essentiel l'entretien appliqué de mécanismes dont tu
possèdes maintenant les manuels.
\end{example}

\section{La responsabilité envers les autres}

\begin{proposition}[La santé est une machinerie partagée]\label{prop:g9:our-responsibility:others}
Trois mécanismes font de la santé un compte partagé~:
\begin{itemize}
  \item la \emph{transmission}
        (\cref{def:g9:microbes-and-infection:stages})~: ton
        \omterm{def:g1:healthy-habits:hygiene}{hygiène}, ta toux couverte et ta journée de maladie
        passée chez toi sont la protection des autres~;
  \item l'\emph{\omterm{def:g9:immune-defenses:memory}{immunité} mise en commun}
        (\cref{prop:g9:immune-defenses:vaccination})~: ta
        \omterm{prop:g9:immune-defenses:vaccination}{vaccination} garde, à travers toi, ceux qu'on ne peut
        pas entraîner~;
  \item \emph{les biens communs de la médecine}~: des
        antibiotiques qui marchent encore
        (\cref{exo:g9:how-species-change:10}) et du \omterm{prop:g4:journey-of-food:absorb}{sang}
        donné (\cref{ex:g9:unique-individuals:medicine} ---
        le registre tourne grâce aux donneurs) sont des
        ressources partagées, dépensées ou mises en réserve
        par des choix individuels.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Le compte partagé d'un adolescent]\label{ex:g9:our-responsibility:account}
Les écritures ordinaires de la semaine~: resté chez soi avec
de la fièvre (transmission coupée)~; rappels à jour (la
mise en commun tenue)~; une cure d'antibiotiques du
printemps menée à son terme (les biens communs non
dépensés)~; un premier don du \omterm{prop:g4:journey-of-food:absorb}{sang} prévu à dix-huit ans (le
registre approvisionné). Rien d'héroïque --- la leçon de
\cref{ex:g5:healthy-choices:day} à l'échelle d'une ville~:
la santé publique, ce sont des journées ordinaires, tenues.
\end{example}

\section{La responsabilité envers le monde vivant}

\begin{proposition}[La gestion en bon père de famille, réénoncée avec toutes ses raisons]\label{prop:g9:our-responsibility:stewardship}
Les fils «~nature~» du livre, noués avec leurs mécanismes~:
\begin{itemize}
  \item toute \omterm{def:g3:food-chains:chain}{chaîne alimentaire} commence en vert
        (\cref{prop:g6:plants-are-producers:firstlink}), et
        toute table est la production d'un écosystème
        (\cref{prop:g6:producing-our-food:costs})~;
  \item les toiles absorbent les chocs par la variété
        (\cref{prop:g5:biodiversity:web}), et la variété est
        la réserve sur laquelle tourne la sélection
        (\cref{prop:g9:unique-individuals:reserve},
        \cref{exo:g9:how-species-change:12}) --- la
        \omterm{def:g5:biodiversity:biodiversity}{biodiversité} n'est pas un décor, mais un capital de
        travail~;
  \item l'extinction ferme la porte pour toujours
        (\cref{prop:g5:biodiversity:loss}) --- la seule
        exception délibérée, la variole, prouve à quel point
        le mot est absolu
        (\cref{exo:g9:immune-defenses:12})~;
  \item les règles d'un usage durable sont connues
        (\cref{prop:g5:people-and-nature:lasting}), leurs
        violations diagnosticables
        (\cref{ex:g7:oxygen-in-the-water:waste}), leurs
        réparations démontrées
        (\cref{ex:g5:people-and-nature:river},
        \cref{ex:g5:biodiversity:storks}).
\end{itemize}
Prendre soin du monde vivant, c'est le même état d'esprit
d'entretien que la santé, appliqué au corps plus grand dans
lequel nous vivons.
\end{proposition}
\admitted

\begin{example}[À quoi sert la biologie d'un citoyen]\label{ex:g9:our-responsibility:citizen}
Les votes, les achats et les habitudes de la vie adulte
n'arrêtent pas d'arriver déguisés en questions auxquelles ce
livre a préparé~: cette pêcherie prélève-t-elle plus vite
que le renouvellement~? Cette mesure protège-t-elle
l'\omterm{def:g2:where-animals-live:habitat}{habitat} des pollinisateurs, ou le bitume-t-elle~? Ce
régime, ce remède ou ce «~\omterm{def:g9:genes-and-dna:gene}{gène} de X~» présenté comme
miraculeux est-il cohérent
(\cref{met:g9:unique-individuals:claims},
\cref{met:g8:contraception:evaluate})~? Cet antibiotique
a-t-il sa place dans cette auge~? Un citoyen capable de
mener ces examens, voilà à quoi ont servi neuf années de
chapitres.
\end{example}

\begin{method}[La méthode du diplômé]\label{met:g9:our-responsibility:method}
En quittant le Livre 1, emporte la méthode plus que les faits~:
\begin{enumerate}
  \item observe avant de conclure
        (\cref{met:g1:animals-around-us:observe} l'a
        commencé~; tous les relevés et les carnets depuis)~;
  \item teste honnêtement --- une seule différence, un
        témoin (\cref{met:g4:what-plants-need:fairtest},
        employé dans toutes les années suivantes)~;
  \item nomme le mécanisme --- aucune affirmation n'est
        examinée tant que son maillon n'est pas nommé
        (\cref{met:g8:contraception:evaluate})~;
  \item pèse les témoins indépendants
        (\cref{met:g9:evidence-of-evolution:weigh})~;
  \item et respecte la machinerie vivante, dans ton corps
        comme au-dehors~: elle est ancienne, subtile,
        réparable dans certaines limites --- et c'est à toi
        de la garder.
\end{enumerate}
\end{method}

\begin{remark}[Ce qui vient ensuite]\label{rem:g9:our-responsibility:next}
Le volume du lycée reprend ces fils au grossissement
suivant~: la machinerie moléculaire de la \omterm{def:g5:first-look-at-cells:cell}{cellule}, la
génétique avec toutes ses mathématiques, les systèmes
immunitaire et nerveux pour de bon, l'écologie comme science
quantitative, et l'\omterm{prop:g9:how-species-change:selection}{évolution} comme le cadre qu'elle est. Les
questions seront les mêmes, posées de plus près. Apporte la
méthode.
\end{remark}

\section{Problème~: l'examen de fin d'études}

\begin{problem}[{Problème du week-end --- une année dans
une ville, passée au crible du livre entier}]\label{pb:g9:our-responsibility:1}
L'année (inventée) de ta ville en gros titres~: une vague de
grippe en hiver~; un débat au printemps sur le traitement
des platanes~; une mortalité de \omterm{prop:g3:sorting-living-things:groups}{poissons} en été dans la mare
du parc~; une campagne à l'automne pour la nouvelle réserve
du marais~; et, toute l'année, les campagnes de \omterm{prop:g9:immune-defenses:vaccination}{vaccination}
et de don du \omterm{prop:g4:journey-of-food:absorb}{sang} du dispensaire.

\medskip\noindent\textbf{Partie I --- Le dossier santé.}
\begin{enumerate}
  \item La vague de grippe~: nomme les voies de
        transmission que visaient les conseils de la ville,
        et dis pourquoi le dispensaire a vacciné en premier
        le personnel des maisons de retraite.
  \item Les campagnes~: dis ce que le registre du \omterm{prop:g4:journey-of-food:absorb}{sang} et
        le carnet de \omterm{prop:g9:immune-defenses:vaccination}{vaccination} approvisionnent chacun, et
        qui y puise.
  \item Un chroniqueur se moque des «~histoires qu'on fait
        autour du lavage des mains~». Réponds avec le
        service de \cref{ex:g9:microbes-and-infection:history}
        --- un paragraphe.
  \item La pharmacie signale des retours de cures
        d'antibiotiques non terminées. Quel bien commun est
        en train d'être dépensé, et par quel mécanisme~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Le dossier nature.}
\begin{enumerate}[resume]
  \item Le débat sur le traitement~: mène l'examen du
        verger (\cref{exo:g9:our-responsibility:9}) sur le
        «~tuer tous les \omterm{prop:g3:sorting-living-things:groups}{insectes}~» --- qui meurt aussi, et
        quels services gratuits s'arrêtent~?
  \item La mortalité de \omterm{prop:g3:sorting-living-things:groups}{poissons}~: une semaine chaude et
        sans vent, une mare vert petit pois. Diagnostique
        avec \cref{ch:g7:oxygen-in-the-water}, et prescris.
  \item La campagne pour le marais~: défends la réserve
        par trois mécanismes --- l'\omterm{def:g2:where-animals-live:habitat}{habitat}, la toile, la
        réserve de variété --- une phrase pour chacun.
  \item Un propriétaire demande~: «~qu'est-ce que ce marais
        a jamais fait pour la ville~?~» Réponds par le
        registre du travail gratuit
        (\cref{exo:g5:biodiversity:11}, le stockage des
        crues par la rivière de
        \cref{ex:g5:people-and-nature:river}).
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La clôture du
diplômé.}
\begin{enumerate}[resume]
  \item Choisis la meilleure et la pire décision de
        l'année selon les règles de l'usage durable, et
        justifie chaque verdict.
  \item Montre les cinq étapes de la méthode à l'œuvre dans
        la façon dont la mortalité de \omterm{prop:g3:sorting-living-things:groups}{poissons} a été
        résolue --- de la première observation au mécanisme
        accepté.
  \item Rédige l'éditorial de fin d'année du journal de la
        ville en quatre phrases~: la santé partagée, la
        nature mise en réserve, les affirmations examinées,
        la méthode gardée.
  \item La dernière ligne de la dernière page du Livre 1
        est à toi~: une phrase, adressée au lecteur qui a
        commencé avec l'escargot de
        \cref{ch:g1:living-or-not}.
\end{enumerate}
\end{problem}
