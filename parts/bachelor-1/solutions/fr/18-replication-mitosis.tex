\section*{Chapitre \ref{ch:b1:replication-mitosis} --- Réplication de l'ADN et mitose}

\begin{solution}{exo:b1:replication-mitosis:1}
Les polymérases n'ajoutent des \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} qu'à une extrémité $3'$, de
sorte que les brins neufs croissent de $5' \to 3'$~; les deux matrices
sont antiparallèles, de sorte qu'à une fourche un brin neuf peut croître
continûment vers la fourche tandis que l'autre doit croître en s'en
éloignant, en repartant par fragments à mesure que de la matrice est
découverte.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:2}
L'hélicase (déroule), les \omterm{def:b1:proteins:peptide}{protéines} de liaison au simple brin (tiennent
les brins séparés), la topoisomérase (relâche la torsion en avant), la
primase (amorces d'\omterm{def:b1:nucleic-acids:chain}{ARN}), l'\omterm{def:b1:replication-mitosis:fork}{ADN polymérase} (prolonge les amorces, relit),
une seconde polymérase (remplace les amorces), la ligase (soude les
fragments).
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:3}
G$_1$~: $q$, 2n \omterm{def:b1:genomes:genome}{chromosomes} à une \omterm{def:b1:replication-mitosis:mitosis}{chromatide}. G$_2$~: $2q$, 2n
\omterm{def:b1:genomes:genome}{chromosomes} à deux \omterm{def:b1:replication-mitosis:mitosis}{chromatides}. Après la \omterm{def:b1:replication-mitosis:mitosis}{mitose}~: $q$ par \omterm{def:b1:cell-unit-of-life:cell}{cellule}
fille, 2n \omterm{def:b1:genomes:genome}{chromosomes} chacune.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:4}
Prophase (condensation, formation du \omterm{def:b1:replication-mitosis:mitosis}{fuseau}), prométaphase (rupture de
l'enveloppe, attachement des kinétochores), métaphase (alignement à
l'équateur), anaphase (séparation des \omterm{def:b1:replication-mitosis:mitosis}{chromatides}), télophase
(reformation des enveloppes), puis cytodiérèse.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:5}
La moitié du \omterm{def:b1:genomes:genome}{génome} est synthétisée en brin discontinu~: $\num{6.4e9}/150 =
\num{4.3e7}$ fragments~; autant d'amorces, de retraits et de soudures.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:6}
$\num{6.4e9}\times 10^{-5} = \num{64000}$~; $\times 10^{-7}$~: 640~;
$\times 10^{-9}$~: environ 6 mutations par division.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:7}
$5000/80 = 62$ divisions. Une tumeur doit se diviser sans limite~; sans
télomérase, ses \omterm{def:b1:genomes:karyotype}{télomères} s'épuiseraient et ses \omterm{def:b1:cell-unit-of-life:cell}{cellules} s'arrêteraient
ou mourraient, de sorte que presque tous les cancers réactivent
l'\omterm{def:b1:enzymes:enzyme}{enzyme}.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:8}
Cycle $= 1\,\mathrm{h}/0.05 = \qty{20}{h}$~; S $= 0.30\times 20 =
\qty{6}{h}$.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:9}
G$_1$~: 8 \omterm{def:b1:genomes:genome}{chromosomes}, 8 \omterm{def:b1:replication-mitosis:mitosis}{chromatides}, $q$. G$_2$~: 8, 16, $2q$.
Métaphase~: 8, 16, $2q$. \omterm{def:b1:cell-unit-of-life:cell}{Cellule} fille~: 8, 8, $q$.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:10}
De nouveaux cycles démarrent à l'origine toutes les \qty{20}{min} avant
la fin du précédent~: la réplication se chevauche, et une \omterm{def:b1:cell-unit-of-life:cell}{cellule}
nouveau-née a déjà de l'\omterm{def:b1:nucleic-acids:chain}{ADN} partiellement répliqué. À la division, le
\omterm{def:b1:genomes:genome}{chromosome} qui achève son cycle a été commencé \qty{40}{min} plus tôt,
un deuxième cycle commencé \qty{20}{min} plus tôt a ses fourches à
mi-chemin, et un troisième débute à peine~: 4 origines et 6 fourches
dans la \omterm{def:b1:cell-unit-of-life:cell}{cellule} en division.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:11}
Aucun \omterm{def:b1:replication-mitosis:mitosis}{fuseau} ne se forme~: les \omterm{def:b1:genomes:genome}{chromosomes} se condensent mais ne peuvent
ni s'aligner ni se séparer, et la \omterm{def:b1:cell-unit-of-life:cell}{cellule} s'arrête au \omterm{prop:b1:replication-mitosis:checkpoints}{point de contrôle}
du \omterm{def:b1:replication-mitosis:mitosis}{fuseau}, dans un état voisin de la métaphase. Si elle repasse en
\omterm{def:b1:replication-mitosis:cycle}{interphase} sans se diviser, elle a 4n \omterm{def:b1:genomes:genome}{chromosomes} (tétraploïde). Des
\omterm{def:b1:cell-unit-of-life:cell}{cellules} arrêtées avec des \omterm{def:b1:genomes:genome}{chromosomes} condensés et séparés sont
exactement ce qu'exige un \omterm{def:b1:genomes:karyotype}{caryotype}~: on applique le produit pour les
accumuler.
\end{solution}

\begin{solution}{exo:b1:replication-mitosis:12}
La réplication garantit que chaque \omterm{def:b1:genomes:genome}{chromosome} devient deux \omterm{def:b1:replication-mitosis:mitosis}{chromatides}
identiques, à une erreur sur un milliard près~; le \omterm{def:b1:replication-mitosis:mitosis}{fuseau} garantit que
chaque fille reçoit l'une de chaque paire, en attachant chaque
\omterm{def:b1:genomes:genome}{chromosome} des deux pôles et en retenant l'anaphase jusqu'à ce que
chacun soit sous tension. Le \omterm{prop:b1:replication-mitosis:checkpoints}{point de contrôle} de G$_2$ assure que la
copie est achevée avant que le comptage ne commence~; celui du \omterm{def:b1:replication-mitosis:mitosis}{fuseau}
assure que le comptage est en place avant la séparation. Une \omterm{def:b1:cell-unit-of-life:cell}{cellule} qui
répartit mal se retrouve avec un \omterm{def:b1:genomes:genome}{chromosome} en trop ou en moins
(aneuploïdie)~: une copie exacte à la lettre, livrée à la mauvaise
adresse --- ce qui tue la plupart de ces \omterm{def:b1:cell-unit-of-life:cell}{cellules} et caractérise la
plupart des cancers.
\end{solution}

\begin{solution}{pb:b1:replication-mitosis:1}
\textbf{1.} $50\times 28\,800 = \qty{1.44e6}{bp}$.
\textbf{2.} \qty{2.88e6}{bp}.
\textbf{3.} $\num{6.4e9}/\num{2.88e6} = 2220$ origines.
\textbf{4.} \qty{2.9}{Mb}, soit $2.9\times 10^6\times 0.34\,\mathrm{nm} =
\qty{0.98}{mm}$ d'\omterm{def:b1:nucleic-acids:chain}{ADN} entre deux origines.
\textbf{5.} $\num{6.4e9}/\num{40000} = \qty{160}{kb}$ d'espacement~;
chaque fourche parcourt environ \qty{80}{kb}, une demi-heure de travail.
\textbf{6.} \qty{125}{Mb} par fourche à \qty{50}{bp/s}~: $\num{2.5e6}$ s,
soit 29 jours.
\textbf{7.} La vitesse des fourches est limitée par la chimie et par la
chromatine (vingt fois plus lente que chez les bactéries)~; les origines
peuvent être multipliées sans limite, de sorte que la phase S se
raccourcit en ajoutant des points de départ, non en accélérant les
fourches.
\textbf{8.} La synthèse discontinue couvre la moitié du \omterm{def:b1:genomes:genome}{génome}, soit
$\num{3.2e9}$ \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides}~: $\num{2.1e7}$ fragments (les deux fourches
de chaque origine ont chacune un brin discontinu, et les moitiés
totalisent l'équivalent d'un \omterm{def:b1:genomes:genome}{génome}).
\textbf{9.} $\num{2.1e7} + 2\times\num{40000} = \num{2.1e7}$ amorces.
\textbf{10.} Environ $\num{2.1e7}$ soudures (une par fragment) plus une
par rencontre de fourches.
\textbf{11.} $2\times\num{6.4e9} = \num{1.28e10}$ \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides}.
\textbf{12.} $\num{2.56e10}$ \omterm{def:b1:water-small-molecules:atp}{ATP}~; sur \qty{28800}{s}, $\num{8.9e5}$
par seconde.
\textbf{13.} Environ \qty{0.1}{\%} du renouvellement d'\omterm{def:b1:water-small-molecules:atp}{ATP} de la
\omterm{def:b1:cell-unit-of-life:cell}{cellule}~: la polymérisation est bon marché~; ce qui coûte cher, c'est de
fabriquer les \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} et les \omterm{def:b1:genomes:nucleosome}{histones}.
\textbf{14.} \num{64000}, 640 et 6.4 erreurs.
\textbf{15.} $1500\times 10^{-9} = \num{1.5e-6}$~: un \omterm{def:b1:genomes:gene}{gène} sur sept cent
mille par division.
\textbf{16.} $10^{16}\times 6.4 = \num{6.4e16}$ mutations --- chaque
changement ponctuel possible du \omterm{def:b1:genomes:genome}{génome}, bien des fois, dispersé parmi
$10^{13}$ \omterm{def:b1:cell-unit-of-life:cell}{cellules}~: presque toutes tombent dans de l'\omterm{def:b1:nucleic-acids:chain}{ADN} non codant ou
dans des \omterm{def:b1:cell-unit-of-life:cell}{cellules} qui seront desquamées, et une \omterm{def:b1:cell-unit-of-life:cell}{cellule} mutée est une
parmi des milliards~; seules quelques combinaisons dans une même \omterm{def:b1:cell-unit-of-life:cell}{cellule}
comptent.
\textbf{17.} D'un facteur cent ($10^{-7}$)~: la suite de mutations qui
fait un cancer, qui met des décennies à s'accumuler à $10^{-9}$,
s'accumule bien plus tôt --- les déficits héréditaires de la réparation
des mésappariements causent des cancers du côlon dès le début de l'âge
adulte.
\textbf{18.} $\num{2.3e6}$ paires par fourche à \qty{1000}{bp/s}~:
\qty{2300}{s}, soit \qty{38}{min}.
\textbf{19.} Environ deux cycles se chevauchent~; une \omterm{def:b1:cell-unit-of-life:cell}{cellule}
nouveau-née porte deux origines (chacune déjà répliquée une fois) et des
fourches à mi-parcours~: son \omterm{def:b1:genomes:genome}{chromosome} est partiellement répliqué à sa
naissance.
\textbf{20.} $\num{4.6e6}/1500 = 3070$ fragments par cycle, soit environ
1.3 par seconde.
\textbf{21.} $\num{4.6e6}\times 10^{-9} = \num{4.6e-3}$ erreur par
cycle~: environ une \omterm{def:b1:cell-unit-of-life:cell}{cellule} fille sur 200 porte une mutation nouvelle.
\textbf{22.} $10^9\times\num{4.6e-3} = \num{4.6e6}$ mutations nouvelles
par millilitre~; un \omterm{def:b1:genomes:gene}{gène} donné de \qty{1}{kb} est touché $\num{4.6e6}\times
1000/\num{4.6e6} = 1000$ fois.
\textbf{23.} Avec mille mutations indépendantes dans un \omterm{def:b1:genomes:gene}{gène} donné par
millilitre et par division, chaque changement ponctuel possible --- dont
ceux qui confèrent la résistance --- est déjà présent dans une grande
culture avant qu'on n'applique l'antibiotique~; le médicament
sélectionne, il ne crée pas.
\textbf{24.} Humain~: $\num{6.4e9}$ paires, \num{40000} origines,
\qty{50}{bp/s}, \qty{8}{h}. Bactérie~: $\num{4.6e6}$ paires, une
origine, \qty{1000}{bp/s}, \qty{40}{min}. Un \omterm{def:b1:genomes:genome}{génome} mille fois plus
grand, copié par des fourches vingt fois plus lentes, en douze fois le
temps seulement~: la différence tient aux dizaines de milliers
d'origines qui travaillent en parallèle.
\textbf{25.} Environ \num{2200} origines au minimum, si toutes
s'allumaient à la fois~; en réalité quelque \num{40000}, une toutes les
\qty{160}{kb}, s'allumant tour à tour de sorte que chaque fourche
parcourt environ \qty{80}{kb}.
\end{solution}
