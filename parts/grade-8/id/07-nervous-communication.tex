\chapter{Komunikasi Saraf}\label{ch:g8:nervous-communication}

\Cref{ch:g5:brain-in-command} menggambar gelungnya ---
melapor, memutuskan, memerintah --- lalu mengukur seperlima
detiknya. Tahun ini kita membuka kabelnya. Apa sebenarnya
yang berjalan di sepanjang sebuah \omterm{def:g8:nervous-communication:neuron}{saraf}? \omterm{def:g8:nervous-communication:neuron}{Saraf} terbuat dari
apa? Dan apa yang terjadi di celah tempat satu kawat
berakhir dan yang berikutnya bermula --- celah tempat,
nanti akan ternyata, ingatan dan banyak racun sama-sama
bekerja?

\section{Kabelnya dan selnya}

\begin{definition}[Neuron dan pesan saraf]\label{def:g8:nervous-communication:neuron}
\omterm{def:g5:first-look-at-cells:cell}{Sel} kerja \omterm{def:g5:brain-in-command:system}{sistem sarafnya} adalah \emph{neuron}\index{neuron}
(\cref{rem:g6:cell-unit-of-life:twoways} menyebutkannya):
masing-masing punya badan \omterm{def:g5:first-look-at-cells:cell}{sel} dan sebuah serat panjang yang
halus --- sebagian seratnya membentang dari \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}
sampai jari kaki, \omterm{def:g5:first-look-at-cells:cell}{sel} terpanjang di dalam tubuh. Sebuah
\emph{saraf}\index{saraf} adalah kabel yang membundel ribuan
serat semacam itu. Sepanjang tiap seratnya berjalan
\emph{pesan saraf}\index{pesan saraf}: isyarat listrik yang
singkat, melesat sampai \qty{100}{m/s} (angka pada
\cref{def:g5:brain-in-command:system}, kini dengan
pembawanya).
\end{definition}

\begin{proposition}[Penyandian menurut frekuensinya]\label{prop:g8:nervous-communication:code}
Isyaratnya sendiri semuanya serupa --- yang berubah adalah
\emph{lajunya}. Sentuhan yang lembut mengirimkan derap yang
lambat menaiki serat inderanya; sentuhan yang kuat, sebuah
genderang yang cepat. Bahasa \omterm{def:g5:brain-in-command:system}{sistem sarafnya} punya satu kata
yang diulang pada kecepatan berbeda-beda: kekuatannya
disandikan sebagai frekuensi, pada tiap jalur, masuk dan keluar.
\end{proposition}
\admitted

\begin{example}[Tiga jalur, satu bahasa]\label{ex:g8:nervous-communication:lines}
Pemain pada gelungnya (\cref{prop:g5:brain-in-command:loop}),
dirangkai ulang: \emph{\omterm{def:g8:nervous-communication:neuron}{neuron} indera} membawa laporannya ---
\omterm{def:g1:the-five-senses:sense}{kulit} ke \omterm{def:g5:brain-in-command:system}{sumsum tulang belakang}, \omterm{def:g1:the-five-senses:sense}{mata} ke \omterm{def:g5:brain-in-command:system}{otak}; \emph{\omterm{def:g8:nervous-communication:neuron}{neuron}
gerak} membawa perintahnya --- sumsum ke \omterm{def:g3:bones-and-muscles:muscle}{otot}, berakhir di
tempat seratnya bertemu \omterm{def:g3:bones-and-muscles:muscle}{ototnya}, dan genderangnya menetapkan
kekuatan kontraksinya
(\cref{def:g3:bones-and-muscles:muscle}); dan di antara
keduanya, di dalam \omterm{def:g5:brain-in-command:system}{otak} dan sumsumnya, \emph{\omterm{def:g8:nervous-communication:neuron}{neuron}
penghubung} yang tak terhitung banyaknya mengerjakan
pemutusannya. Gerak refleks atau simfoni, lalu lintasnya
adalah isyarat yang sama pada laju berbeda di jalur berbeda.
\end{example}

\section{Celahnya: sinapsis}

\begin{definition}[Sinapsis]\label{def:g8:nervous-communication:synapse}
\omterm{def:g8:nervous-communication:neuron}{Neuron} tidak bersentuhan. Di tempat ujung sebuah serat
bertemu \omterm{def:g5:first-look-at-cells:cell}{sel} berikutnya terdapat sebuah celah renik --- yaitu
\emph{sinapsis}\index{sinapsis}. Isyarat listriknya tidak
dapat melompatinya; sebagai gantinya serat yang tiba
melepaskan sebuah \emph{pengantar pesan kimiawi} ke dalam
celahnya, yang menyeberang, berlabuh di sisi seberangnya,
dan --- bila cukup banyak yang tiba --- memulai isyaratnya
lagi dari awal di dalam \omterm{def:g8:nervous-communication:neuron}{neuron} berikutnya. Listrik di
sepanjang kabelnya, kimia di setiap simpangnya: seluruh
sistemnya menyelang-nyelingkan keduanya.
\end{definition}

\begin{proposition}[Mengapa celah, bukan patrian]\label{prop:g8:nervous-communication:whygaps}
Celahnya tampak seperti cacat dan justru kejeniusan sistemnya:
\begin{itemize}
  \item ia membuat simpangnya \emph{dapat disetel} ---
        \omterm{def:g8:nervous-communication:synapse}{sinapsis} yang sering diseberangi tumbuh menguat,
        yang tak terpakai memudar: itulah belajar
        (``jalur yang lebih baik'' pada
        \cref{rem:g5:brain-in-command:training}, kini
        ditemukan --- ia adalah \omterm{def:g8:nervous-communication:synapse}{sinapsis} yang menguat);
  \item ia membuat isyaratnya \emph{searah} --- pengantar
        pesannya dilepaskan hanya di satu sisi, sehingga
        lalu lintasnya tidak dapat berjalan mundur;
  \item ia membiarkan satu \omterm{def:g8:nervous-communication:neuron}{neuron} \emph{menimbang} ribuan
        masukan --- suara yang memacu dan yang meredam
        dijumlahkan pada tiap \omterm{def:g5:first-look-at-cells:cell}{selnya} sebelum ia menyalak:
        pemutusan pada gelungnya, dilakukan di simpangnya.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Sinapsis sang penyulap bola]\label{ex:g8:nervous-communication:juggler}
Sang penyulap bola pada
\cref{rem:g5:brain-in-command:training}, akhirnya
dijelaskan: sebulan latihan menghantarkan isyarat yang sama
melintasi simpang yang sama sampai \omterm{def:g8:nervous-communication:synapse}{sinapsis} itu menguat ---
perutean yang dahulu ditimbang dengan susah payah kini
berjalan dengan timbangan yang sudah siap. Keterampilan
adalah kimia yang dipadatkan di celahnya; ``latihan membuat
menetap'' adalah sebuah pernyataan tentang \omterm{def:g8:nervous-communication:synapse}{sinapsis}.
\end{example}

\begin{remark}[Celahnya adalah titik rawannya]\label{rem:g8:nervous-communication:vulnerable}
Apa pun yang bekerja pada \omterm{def:g5:brain-in-command:system}{sistem saraf} dari luar bekerja di
\omterm{def:g8:nervous-communication:synapse}{sinapsisnya} --- satu-satunya tempat isyaratnya menjadi
kimia, yang dapat dilabuhi dan diperdaya. Obat bius
menenangkannya; bisa \omterm{def:g1:animals-around-us:animal}{hewan} memacetkannya (sebagian
melumpuhkan dengan menghadang pengantar pesan pada simpang
\omterm{def:g3:bones-and-muscles:muscle}{ototnya}, sebagian dengan membanjirinya); dan alkohol pada
\cref{prop:g5:healthy-choices:substances}, serta setiap obat
yang mengubah suasana hati atau penginderaan, bekerja dengan
menirukan, menghadang atau menguras pengantar pesan
celahnya. \Cref{ch:g8:protecting-the-nervous-system}
dibangun justru di atas titik ini.
\end{remark}

\section{Kabel cepat, pos lambat: dua sistem dibandingkan}

\begin{proposition}[Saraf melawan hormon]\label{prop:g8:nervous-communication:compare}
Tubuhnya kini memperlihatkan dua sistem komunikasi, yang
dibangun untuk pekerjaan yang berlawanan:
\begin{itemize}
  \item \emph{\omterm{def:g8:nervous-communication:neuron}{saraf}}: isyaratnya dalam milidetik, di
        sepanjang jalur khusus, ke alamat yang tepat ---
        satu \omterm{def:g3:bones-and-muscles:muscle}{otot}, satu kelenjar --- dan lenyap ketika
        isyaratnya berhenti: sistem tangkapan, tuturan dan
        gerak refleksnya;
  \item \emph{\omterm{def:g8:puberty:hormones}{hormon}} (\cref{def:g8:puberty:hormones}):
        pengantar pesannya dalam hitungan menit sampai
        bulan, disiarkan oleh \omterm{prop:g4:journey-of-food:absorb}{darah} ke setiap organ yang
        dibangun untuk mendengar, memerintahkan program yang
        lambat dan seluas tubuh --- pertumbuhan, daur,
        pembangunan ulang \omterm{def:g8:puberty:puberty}{pubertasnya}.
\end{itemize}
Telegraf dan jasa pos: tubuhnya menjalankan keduanya, dan
\cref{ch:g8:hormonal-communication} memberi jasa posnya
sebuah bab tersendiri.
\end{proposition}
\admitted

\begin{method}[Mendiagnosis sebuah komunikasi, sistem demi sistem]\label{met:g8:nervous-communication:diagnose}
Diberikan sebuah pemaduan tubuh apa pun, tetapkan sistemnya:
\begin{enumerate}
  \item ukur waktunya: milidetik --- \omterm{def:g8:nervous-communication:neuron}{saraf}; menit sampai
        tahun --- \omterm{def:g8:puberty:hormones}{hormon} (atau keduanya, secara berantai);
  \item petakan jangkauannya: satu alamat yang tepat ---
        \omterm{def:g8:nervous-communication:neuron}{saraf}; seluas tubuh --- \omterm{def:g8:puberty:hormones}{hormon};
  \item temukan rutenya: jalur kabel yang dapat disebutkan
        --- \omterm{def:g8:nervous-communication:neuron}{saraf}; siaran aliran \omterm{prop:g4:journey-of-food:absorb}{darah} --- \omterm{def:g8:puberty:hormones}{hormon};
  \item periksa ketahanannya: lenyap bersama isyaratnya ---
        \omterm{def:g8:nervous-communication:neuron}{saraf}; berjalan selama pengantar pesannya beredar
        --- \omterm{def:g8:puberty:hormones}{hormon}.
\end{enumerate}
\end{method}

\begin{example}[Caranya pada tiga kasus]\label{ex:g8:nervous-communication:cases}
Kedipan terhadap semburan udara: milidetik, satu cincin
\omterm{def:g3:bones-and-muscles:muscle}{otot}, jalur berkabel, lenyap seketika --- \omterm{def:g8:nervous-communication:neuron}{saraf}. Ledakan
pertumbuhannya: bertahun-tahun, seluruh tubuh, aliran \omterm{prop:g4:journey-of-food:absorb}{darah},
berjalan berbagai musim --- \omterm{def:g8:puberty:hormones}{hormon}. \omterm{def:g4:heart-and-blood:heart}{Jantung} yang berpacu
pada lari cepatnya (\cref{prop:g7:organs-at-work:supply}):
keduanya secara berantai --- isyarat \omterm{def:g8:nervous-communication:neuron}{sarafnya} menaikkan
lajunya dalam satu denyut, dan sebuah pengantar pesan aliran
\omterm{prop:g4:journey-of-food:absorb}{darah} (lonjakan termasyhur yang kamu rasakan sebagai
gelombang panas ketakutan) menahan siaga seluruh tubuhnya
selama beberapa menit. Telegraf dahulu, pos menyusul.
\end{example}

\section{Soal: Meja Uji Gerak Refleks}

\begin{problem}[{Soal akhir pekan --- sentakan lutut,
dilengkapi alat}]\label{pb:g8:nervous-communication:1}
Peragaan meja uji yang klasik: sukarelawan yang duduk, palu
kecil di bawah tempurung \omterm{def:g1:my-body:joint}{lututnya}, tendangan kaki yang tak
disengaja --- kini dilengkapi pencatat waktu dan bab ini.

\medskip\noindent\textbf{Bagian I --- Litarnya.}
\begin{enumerate}
  \item Palunya meregangkan sebuah urat; kakinya menendang
        kira-kira seperlima puluh detik kemudian. Bentangkan
        litarnya: jenis \omterm{def:g8:nervous-communication:neuron}{neuron} yang mana, menembus mana ---
        dan mengapa \omterm{def:g5:brain-in-command:system}{otaknya} tidak berada pada gelung ini.
  \item Seperlima puluh detik lebih cepat daripada seperlima
        detik pada tangkapan bolanya. Jelaskan bedanya
        dengan panjang litarnya dan jumlah simpangnya.
  \item Sukarelawannya tidak dapat menahan tendangannya,
        walaupun sudah diperingatkan. Apa yang dikatakannya
        tentang di mana gerak refleksnya diputuskan --- dan
        tentang lalu lintas searah di \omterm{def:g8:nervous-communication:synapse}{sinapsisnya}?
  \item Segera sesudah tendangannya, sukarelawannya
        \emph{merasakan} ketukannya. Jelaskan perjalanan
        kedua yang lebih lambat pada tundaannya.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Ragam meja ujinya.}
\begin{enumerate}[resume]
  \item Ketuk lebih keras: tendangannya menguat. Aturan
        penyandian yang mana pada
        \cref{prop:g8:nervous-communication:code} yang
        memperlihatkan dirinya?
  \item Dokter kelasnya menjelaskan bahwa gerak refleks ini
        diuji pada setiap pemeriksaan. Apa yang disahkan
        oleh sentakan \omterm{def:g1:my-body:joint}{lutut} yang ada dan setangkup, tentang
        sumsum, \omterm{def:g8:nervous-communication:neuron}{saraf} dan simpangnya?
  \item Sukarelawan yang mengatupkan rahangnya menendang
        lebih keras --- sebuah penguatan yang sudah dikenal.
        Apa yang diungkapkannya tentang suara \omterm{def:g8:nervous-communication:neuron}{neuron}
        penghubung yang mencapai bahkan gelung ``tanpa \omterm{def:g5:brain-in-command:system}{otak}''?
  \item Obat bius setempat di \omterm{def:g1:my-body:parts}{tungkainya} menghapus tendangan
        sekaligus rasanya. Sebutkan titik serangannya,
        menurut \cref{rem:g8:nervous-communication:vulnerable}.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Laporannya.}
\begin{enumerate}[resume]
  \item Gambarlah gerak refleksnya sebagai gelung tiga
        langkah pada \cref{prop:g5:brain-in-command:loop},
        tandai di mana yang satu ini memotong jalan.
  \item Bandingkan gerak refleksnya dengan tangkapannya
        dalam tabel dua lajur: jalur, simpang, waktu,
        keterahanan, peran latihan.
  \item Bagian terakhir laporannya bertanya: ``mengapa
        gelung pelindung yang cepat memintas \omterm{def:g5:brain-in-command:system}{otaknya}?''
        Jawablah dengan logika wajan panasnya
        (\cref{exo:g5:brain-in-command:5}).
  \item Satu kalimat untuk menutup: apa yang diperlihatkan
        meja ujinya tentang bahasa, kabel dan simpang \omterm{def:g5:brain-in-command:system}{sistem
        sarafnya} sekaligus.
\end{enumerate}
\end{problem}
