\chapter{Exploring Our Environment}\label{ch:g6:exploring-our-environment}

Cross the schoolyard on a July noon and the wall's south face is
hot under your hand, the air above the asphalt shimmers, and the
only green thing in sight is a weed in a crack. Step behind the
north wall: cool air, damp soil, moss, woodlice under a stone.
Two worlds, three metres apart. This year begins by exploring
\omterm{def:g6:exploring-our-environment:components}{environments} the way scientists do --- measuring them, and
asking why each \omterm{def:g6:exploring-our-environment:components}{living thing} lives exactly where it does.

\section{What an environment is made of}

\begin{definition}[The components of an environment]\label{def:g6:exploring-our-environment:components}
An \emph{environment}\index{environment} --- a yard, a wood, a
pond bank --- contains three kinds of components:
\begin{enumerate}
  \item \emph{living things}\index{living thing}: the \omterm{def:g1:animals-around-us:animal}{animals},
        \omterm{def:g1:plants-around-us:plant}{plants}, mushrooms and microscopic life present;
  \item \emph{mineral matter}\index{mineral matter}, which was
        \omterm{def:g1:living-or-not:nonliving}{never alive}: rock, stone, soil's mineral part, water,
        air;
  \item \emph{traces and remains} of living things: dead
        \omterm{def:g1:plants-around-us:parts}{leaves}, fallen wood, droppings, \omterm{def:g1:animals-around-us:covering}{feathers}, \omterm{def:g1:animals-around-us:covering}{shells} ---
        the ``was living'' box of
        \cref{def:g1:living-or-not:nonliving}, now a clue kit
        for the explorer.
\end{enumerate}
\end{definition}

\begin{example}[Inventory behind the north wall]\label{ex:g6:exploring-our-environment:inventory}
Ten minutes' inventory in the cool corner: living --- moss,
\omterm{prop:g4:plants-without-seeds:damp}{ferns}, woodlice, a snail, a centipede, ivy; mineral --- the
stone of the wall, damp soil, a puddle; traces --- dead
\omterm{def:g1:plants-around-us:parts}{leaves}, a snail \omterm{def:g1:animals-around-us:covering}{shell}, bird droppings on the coping. Three
lists, and the corner is described like a scientist's site.
\end{example}

\begin{omfigure}

\begin{remark}[Traces count as data]\label{rem:g6:exploring-our-environment:traces}
Most \omterm{def:g1:animals-around-us:animal}{animals} hide or flee, so the explorer often reads
\emph{traces} instead: a nibbled hazelnut names the wood
mouse, a spider's web names its owner, tracks in mud name the
heron. An inventory that only lists what sat still enough to
be seen misses half the inhabitants.
\end{remark}

\section{Measuring the conditions}

\begin{definition}[Environmental conditions]\label{def:g6:exploring-our-environment:conditions}
The \emph{conditions}\index{conditions (environmental)} of an
\omterm{def:g6:exploring-our-environment:components}{environment} are its measurable physical characteristics:
\begin{itemize}
  \item \emph{temperature}\index{temperature}, measured with a
        thermometer, in degrees Celsius;
  \item \emph{humidity}\index{humidity} --- how damp the air
        or soil is;
  \item \emph{light}\index{light (condition)} --- from full
        sun to deep shade.
\end{itemize}
Conditions change from spot to spot --- often sharply within
metres --- and through the day and the year.
\end{definition}

\begin{example}[The yard, measured]\label{ex:g6:exploring-our-environment:measured}
The July noon survey, with instruments: south wall foot ---
\qty{38}{\degreeCelsius}, dry, full sun; north corner ---
\qty{22}{\degreeCelsius}, damp soil, deep shade. Three metres
of distance, sixteen degrees of difference. The two worlds of
the opening paragraph are now two \emph{rows in a table} ---
and that is progress, because tables can be compared.
\end{example}

\begin{proposition}[Conditions decide the inhabitants]\label{prop:g6:exploring-our-environment:decide}
The \omterm{def:g6:exploring-our-environment:components}{living things} of a spot match its \omterm{def:g6:exploring-our-environment:conditions}{conditions}: each \omterm{def:g5:biodiversity:species}{species}
thrives only within its own range of \omterm{def:g6:exploring-our-environment:conditions}{temperature}, \omterm{def:g6:exploring-our-environment:conditions}{humidity} and
\omterm{def:g6:exploring-our-environment:conditions}{light}. Moss and woodlice need dampness (the woodlouse's
covering, unlike an insect's, lets water escape --- it dries
out fatally in an hour of sun); the weed in the crack bears
drought and glare. Where \omterm{def:g6:exploring-our-environment:conditions}{conditions} differ, inhabitants
differ --- the rule behind every \omterm{def:g2:where-animals-live:habitat}{habitat} of
\cref{ch:g2:where-animals-live}.
\end{proposition}
\admitted

\begin{example}[The woodlouse chooser]\label{ex:g6:exploring-our-environment:chooser}
The classic experiment: a closed box, half lined with damp
paper, half with dry, and a dozen woodlice released along the
middle. Within minutes, nearly all are on the damp side. Run
again with a shaded half against a lit half: they gather in
the shade. No one herds them --- each \omterm{def:g1:animals-around-us:animal}{animal}, wandering and
turning, ends where its body is comfortable. The map of
woodlice \emph{is} a map of dampness.
\end{example}

\begin{method}[Surveying a site]\label{met:g6:exploring-our-environment:survey}
To survey a small site like a field scientist:
\begin{enumerate}
  \item choose and mark a small area --- a square metre tells
        more, well searched, than a hectare glanced at;
  \item measure the \omterm{def:g6:exploring-our-environment:conditions}{conditions}: \omterm{def:g6:exploring-our-environment:conditions}{temperature}, dampness, \omterm{def:g6:exploring-our-environment:conditions}{light}
        --- same instruments, same way, at each site
        (\cref{met:g4:what-plants-need:fairtest}'s fairness
        rule);
  \item inventory the three component kinds --- \omterm{def:g6:exploring-our-environment:components}{living things},
        \omterm{def:g6:exploring-our-environment:components}{mineral matter}, traces --- searching properly: under
        stones (put them back:
        \cref{met:g2:caring-for-nature:rules}), in crevices,
        on \omterm{def:g1:plants-around-us:parts}{leaf} undersides;
  \item record everything on the spot --- lists, sketches,
        counts; memory is not a recording instrument;
  \item compare sites: pair each difference of inhabitants
        with a difference of \omterm{def:g6:exploring-our-environment:conditions}{conditions}.
\end{enumerate}
\end{method}

\begin{remark}[Why records beat memories]\label{rem:g6:exploring-our-environment:records}
Back indoors, two surveyors who trusted memory will argue
about whether the snails numbered three or five. The notebook
settles it --- or admits honestly that snails were not
counted. Written records, made at the site, are what turn a
walk into data; every science of this book runs on them.
\end{remark}

\section{One environment, many micro-worlds}

\begin{definition}[Microhabitat]\label{def:g6:exploring-our-environment:microhabitat}
Within one \omterm{def:g6:exploring-our-environment:components}{environment}, each small spot with its own
\omterm{def:g6:exploring-our-environment:conditions}{conditions} and its own inhabitants --- the underside of a
stone, a crack in the wall, the shaded foot of the hedge ---
is a \emph{microhabitat}\index{microhabitat}.
\Cref{rem:g2:where-animals-live:layers}'s forest floors were
microhabitats; the surveyor's square metre usually holds
several.
\end{definition}

\begin{example}[The stone, above and below]\label{ex:g6:exploring-our-environment:stone}
One stone in the cool corner: on top, lichen and a sunning
fly --- lit, dry, warm. Beneath, woodlice, a centipede, pale
rootlets --- dark, damp, cool. Two complete \omterm{def:g6:exploring-our-environment:microhabitat}{microhabitats}
separated by five centimetres of rock, each inhabited by
exactly those \omterm{def:g6:exploring-our-environment:components}{living things} whose ranges its \omterm{def:g6:exploring-our-environment:conditions}{conditions}
match.
\end{example}

\section{Problem: The Two Ditches}

\begin{problem}[{Weekend problem --- the naturalist club
surveys two roadside ditches and finds two different
worlds}]\label{pb:g6:exploring-our-environment:1}
The naturalist club surveys two ditches along the same lane,
$50$ metres apart. Ditch N runs along the north side of a
tall hedge; ditch S runs in the open, facing south. The club
works by the book: same instruments, same square metre per
site, same search time.

\medskip\noindent\textbf{Part I --- The measurements.}
\begin{enumerate}
  \item The thermometers read \qty{19}{\degreeCelsius} in
        ditch N and \qty{31}{\degreeCelsius} in ditch S at
        the same hour. Which single feature of the sites
        best explains the difference?
  \item Ditch N's soil sticks to the fingers; ditch S's
        crumbles to dust. Translate both observations into
        the vocabulary of
        \cref{def:g6:exploring-our-environment:conditions}.
  \item The club measures at noon. Give one reason the
        comparison would be less fair if ditch N were
        measured at noon and ditch S at 6 p.m.
  \item Why does the club use the \emph{same} square-metre
        frame and the same search time at both sites?
\end{enumerate}

\medskip\noindent\textbf{Part II --- The inventories.}
\begin{enumerate}[resume]
  \item Ditch N's list: moss, \omterm{prop:g4:plants-without-seeds:damp}{ferns}, woodlice, snails, a
        toad. Ditch S's list: dry grasses, a large weed with
        waxy \omterm{def:g1:plants-around-us:parts}{leaves}, ants, a basking lizard, grasshoppers.
        Sort each list's members into the three component
        kinds --- and note what the lists are missing (check
        \cref{def:g6:exploring-our-environment:components}).
  \item For each of woodlice, toad and lizard, say which
        ditch's \omterm{def:g6:exploring-our-environment:conditions}{conditions} match its needs, and why (use
        \cref{prop:g6:exploring-our-environment:decide} and
        what you know of each \omterm{def:g1:animals-around-us:animal}{animal}).
  \item In ditch S the club finds a snail \omterm{def:g1:animals-around-us:covering}{shell}, but no
        snail, and asks whether snails live there. What are
        the two honest possibilities a trace like this
        \omterm{def:g1:plants-around-us:parts}{leaves} open?
  \item Under a flat stone in ditch N: eight woodlice, a
        centipede, white rootlets. Name the \omterm{def:g6:exploring-our-environment:microhabitat}{microhabitat}'s
        \omterm{def:g6:exploring-our-environment:conditions}{conditions}, and state the rule that put those
        inhabitants there.
\end{enumerate}

\medskip\noindent\textbf{Part III --- The explanation.}
\begin{enumerate}[resume]
  \item Pair each difference of inhabitants with a
        difference of \omterm{def:g6:exploring-our-environment:conditions}{conditions}: complete ``N has moss and
        toads because \dots; S has lizards and waxy-leaved
        weeds because \dots''.
  \item The hedge is scheduled to be grubbed out. Predict
        ditch N's \omterm{def:g6:exploring-our-environment:conditions}{conditions} two summers later --- and its
        inhabitants, using the pairing of question 9.
  \item A member proposes checking the prediction properly:
        survey both ditches again after the hedge is gone.
        Why must the \emph{same} sites, instruments and
        season be used for the re-survey to count?
  \item The club's report ends: ``the inhabitants map the
        \omterm{def:g6:exploring-our-environment:conditions}{conditions}.'' Explain the sentence with the
        woodlouse chooser
        (\cref{ex:g6:exploring-our-environment:chooser}),
        and say why no herding is needed for a map like
        that to draw itself.
\end{enumerate}
\end{problem}
