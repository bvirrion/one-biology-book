\section*{Bab \ref{ch:g8:sexual-reproduction} --- Reproduksi Seksual}

\begin{solution}{exo:g8:sexual-reproduction:1}
\omterm{def:g8:sexual-reproduction:gametes}{Gamet} adalah \omterm{def:g5:first-look-at-cells:cell}{sel} kelamin --- \omterm{def:g4:animal-reproduction:sexual}{jantan} atau \omterm{def:g4:animal-reproduction:sexual}{betina}; \omterm{prop:g8:sexual-reproduction:core}{zigot} adalah satu \omterm{def:g5:first-look-at-cells:cell}{sel}
baru yang dibentuk penggabungannya. \omterm{def:g1:animals-around-us:animal}{Hewan}: \omterm{def:g4:animal-reproduction:sexual}{sel sperma} dan \omterm{def:g4:animal-reproduction:sexual}{sel telur}.
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} berbunga: \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g4:animal-reproduction:sexual}{jantan} butir \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sarinya} dan \omterm{def:g4:animal-reproduction:sexual}{sel telur} \omterm{prop:g5:flower-to-fruit:roles}{bakal
bijinya}.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:2}
Dua induk menghasilkan \omterm{def:g8:sexual-reproduction:gametes}{gamet} \omterm{def:g4:animal-reproduction:sexual}{jantan} dan \omterm{def:g8:sexual-reproduction:gametes}{gamet} \omterm{def:g4:animal-reproduction:sexual}{betina}; \omterm{def:g5:flower-to-fruit:steps}{pembuahan}
menggabungkan satu dari masing-masingnya menjadi sebuah \omterm{prop:g8:sexual-reproduction:core}{zigot};
\omterm{prop:g8:sexual-reproduction:core}{zigotnya} membelah lalu membelah lagi, membangun sebuah individu baru
yang membawa sesuatu dari kedua induknya.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:3}
Tempat \omterm{def:g8:sexual-reproduction:gametes}{gametnya} bertemu; pemondokan anak yang sedang tumbuh; lalu
siasat jumlah-dan-perawatannya. Semuanya adalah setelan di sekeliling inti
yang sama.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:4}
\omterm{def:g3:flowers-fruits-seeds:parts}{Benang sari}: pabrik \omterm{def:g8:sexual-reproduction:gametes}{gamet} \omterm{def:g4:animal-reproduction:sexual}{jantannya}. \omterm{def:g3:flowers-fruits-seeds:parts}{Putik} (beserta \omterm{prop:g5:flower-to-fruit:roles}{bakal bijinya}):
pabrik \omterm{def:g8:sexual-reproduction:gametes}{gamet} \omterm{def:g4:animal-reproduction:sexual}{betinanya}. Buluh \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari}: lorong \omterm{def:g5:flower-to-fruit:steps}{pembuahan} dalamnya.
\omterm{def:g2:from-seed-to-plant:seed}{Biji}: \omterm{def:g5:human-reproduction:pregnancy}{embrio} yang dipondokkan beserta bekalnya --- bungkusan \omterm{def:g4:animal-reproduction:ovivivi}{ovipar}
milik \omterm{def:g1:plants-around-us:plant}{tumbuhannya}.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:5}
Satu induk menghasilkan salinan yang persis --- keseragaman yang tidak
pernah dapat diberikan kocokannya; dua induk menghasilkan paduan yang
baru setiap kali --- keragaman yang tidak pernah dapat diberikan
penyalinannya.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:6}
Tiap saudara kandungnya tumbuh dari sepasang \omterm{def:g8:sexual-reproduction:gametes}{gamet} yang berbeda, dan
tidak ada dua \omterm{def:g8:sexual-reproduction:gametes}{gamet} yang membawa warisan yang benar-benar sama: setiap
\omterm{def:g5:flower-to-fruit:steps}{pembuahan} membagikan sebuah paduan baru dari sumbangan kedua induknya.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:7}
Menyalin (sambungan, \omterm{met:g4:plants-without-seeds:cutting}{stek}) menabung sebuah keunggulan yang sudah
terbukti --- tepat ketika keragaman masa kini persis yang diinginkan;
harganya adalah kelemahan yang dibagi bersama. Mengocok (\omterm{def:g2:from-seed-to-plant:seed}{biji})
menjelajahi paduan yang baru --- tepat ketika perbaikan atau asuransi
yang diinginkan; harganya adalah bahwa kebanyakan bagiannya biasa
saja.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:8}
\omterm{def:g8:sexual-reproduction:gametes}{Gametnya}: \omterm{def:g2:animals-grow-up:born}{telur} dari \omterm{def:g4:animal-reproduction:sexual}{betinanya}, sperma dari \omterm{def:g4:animal-reproduction:sexual}{jantannya}, yang
dikeluarkan di dalam kolamnya. Pertemuannya: \omterm{def:g5:flower-to-fruit:steps}{pembuahan} luar di dalam
air, dengan nyanyian para \omterm{def:g4:animal-reproduction:sexual}{jantannya} yang mengumpulkan kolamnya.
Pemondokannya: \omterm{def:g2:animals-grow-up:born}{telur} di dalam agar-agar, dengan bekal yang kecil.
Jumlah dan perawatannya: ribuan, tanpa dirawat. Musimnya: musim semi
--- paduan suara bulan Mei.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:9}
Catatan pertemuannya runtuh --- sama sekali tidak ada \omterm{def:g5:flower-to-fruit:steps}{pembuahan} ---
namun catatan pemondokan, jumlah dan musimnya terbaca seperti biasa:
\omterm{def:g2:from-seed-to-plant:seed}{biji} berjambul dalam jumlah ribuan, pada musim panas. Keganjilannya
mengajarkan bahwa berkasnya adalah sebuah daftar periksa, bukan sebuah
anggapan: isilah ia dari bukti, dan sebuah ``\omterm{def:g2:from-seed-to-plant:seed}{biji}'' tidak selalu
menjamin adanya kocokan. (Jombang yang mengklon lewat \omterm{def:g2:from-seed-to-plant:seed}{biji} memperoleh
taruhan penyalinan dengan sebuah layanan \omterm{def:g6:how-plants-spread:dispersal}{penyebaran} --- kedua dunia
sekaligus.)
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:10}
Dua abad \omterm{met:g4:plants-without-seeds:cutting}{stek} membuat kebun anggurnya menjadi satu susunan yang dibagi
bersama --- keunggulan yang ditabung, kelemahan yang dibagi bersama.
Hama yang menyukai susunan itu mendapati setiap pohon anggurnya sama
sesuai dengan seleranya: tanpa individu yang beragam, tanpa cadangan,
taruhan penyalinannya ditagih pada meja yang paling buruk. Penyelamatnya,
secara kesejarahan seperti pada latihannya, adalah taruhan yang lain:
menyilangkan masuk galur yang tahan --- yaitu kocokannya.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:11}
Dari intinya: sebuah \omterm{def:g8:sexual-reproduction:gametes}{gamet} yang sendirian mati --- hanya \omterm{def:g5:flower-to-fruit:steps}{pembuahan},
yang menggabungkan dua, yang membuat \omterm{prop:g8:sexual-reproduction:core}{zigot} yang hidup lalu membelah.
Dari kocokannya: tiap \omterm{def:g8:sexual-reproduction:gametes}{gamet} hanya membawa separuh milik satu induknya
dari sebuah paduan yang belum ada sampai penggabungannya membagikannya.
\end{solution}

\begin{solution}{exo:g8:sexual-reproduction:12}
\omterm{ex:g4:plants-without-seeds:runner}{Geragihnya} adalah layanan jarak pendek yang pasti: salinan yang persis
yang menduduki \omterm{def:g6:decomposers-and-soil:soil}{tanah} baik di sebelahnya dengan rapat --- tanaman tahun
ini yang sudah terbukti, yang ditabung. \omterm{def:g2:from-seed-to-plant:seed}{Bijinya} adalah layanan jarak
jauh yang menjelajah: paduan yang terkocok, yang ditebarkan jauh, yang
mengisi cadangan \omterm{def:g5:biodiversity:species}{spesiesnya} bagi \omterm{def:g6:exploring-our-environment:conditions}{kondisi} yang berubah di tempat lain.
Mempertahankan keduanya membeli masa kini dan masa depan sekaligus.
\end{solution}

\begin{solution}{pb:g8:sexual-reproduction:1}
\textbf{1.} Kuasnya berperan sebagai penyerbuk: membawa \omterm{def:g8:sexual-reproduction:gametes}{gamet} \omterm{def:g4:animal-reproduction:sexual}{jantan}
dari A ke \omterm{def:g3:flowers-fruits-seeds:parts}{putik} B --- yaitu langkah pertemuannya. Bungkusnya menolak
lebah sehingga tidak ada \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} nyasar yang menambahkan induk
ketiga: pengendalian atas persilangannya --- semangat uji yang adil di
tingkat \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}.

\textbf{2.} \omterm{def:g2:from-seed-to-plant:seed}{Biji} --- tiap \omterm{def:g2:from-seed-to-plant:seed}{bijinya} sebuah \omterm{def:g5:human-reproduction:pregnancy}{embrio} dari salah satu \omterm{def:g4:animal-reproduction:sexual}{sel
telur} B yang dibuahi salah satu \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} A: warisan A dan B
yang dipadukan.

\textbf{3.} Karena tiap \omterm{def:g2:from-seed-to-plant:seed}{bijinya} adalah pembagian kocokan yang berbeda:
\omterm{def:g8:sexual-reproduction:gametes}{gamet} yang berbeda pada kedua sisinya, sehingga ada tiga ratus paduan
kerenyahan, ketahanan dan segala hal lainnya yang berbeda.

\textbf{4.} Kedua taruhan kebun \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya}
(\cref{ex:g8:sexual-reproduction:bets}): kebanyakan \omterm{prop:g2:from-seed-to-plant:cycle}{kecambahnya} biasa
saja, sebagian lebih buruk, satu yang langka lebih baik --- sebaran
kocokan yang sudah diketahui.

\textbf{5.} Paduannya terkocok pergi lagi: \omterm{def:g2:from-seed-to-plant:seed}{biji} sang pemenang
membagikan campuran \emph{baru} antara warisannya dan warisan induk
\omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sarinya}, dan sifat renyah-dan-tahan itu berhamburan menjadi
kepingan. Menyemai sang pemenang membatalkan kemenangannya.

\textbf{6.} Penyambungan --- \omterm{def:g4:plants-without-seeds:vegetative}{perkembangbiakan vegetatif}: satu induk,
salinan yang persis. Ia menjamin paduan yang berharga itu, pohon demi
pohon.

\textbf{7.} Risiko kelemahan yang dibagi bersama pada taruhan
penyalinannya: satu susunan di seluruh kebun \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya} --- kisah kebun
anggurnya yang menunggu untuk terjadi.

\textbf{8.} Sebuah cadangan keragaman: jenis lain yang ditanam
berselang-seling, dan galur pemuliaan yang dipelihara dari \omterm{def:g2:from-seed-to-plant:seed}{biji} ---
sehingga ketika sebuah penyakit hawar atau iklim baru memanggil,
kocokannya punya persediaan untuk dibagikan. Cadangan setingkat
\omterm{def:g5:biodiversity:species}{spesies}, yang dijaga dengan sengaja pada skala kebun \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}.

\textbf{9.} Tahap pertama (persilangannya dan ketiga ratus
\omterm{prop:g2:from-seed-to-plant:cycle}{kecambahnya}): kocokannya --- ia \emph{menemukan} paduannya. Tahap
kedua (penyambungan sang pemenang): salinannya --- ia
\emph{mempertahankan} paduannya. Penjelajahan, lalu penabungan.

\textbf{10.} Catatan yang berbeda: pertemuannya (luar, di dalam air,
tanpa perlu kuas --- pemulianya sekadar memasangkan induknya),
pemondokannya (\omterm{def:g2:animals-grow-up:born}{telur} di dalam sarang, bukan \omterm{def:g2:from-seed-to-plant:seed}{biji}), perawatannya
(ayahnya mengipasi kelompok \omterm{def:g2:animals-grow-up:born}{telurnya}), musimnya. Yang tak berubah:
\omterm{def:g8:sexual-reproduction:gametes}{gamet} yang dibuat, \omterm{def:g5:flower-to-fruit:steps}{pembuahan} yang menggabungkan dua menjadi sebuah
\omterm{prop:g8:sexual-reproduction:core}{zigot}, \omterm{prop:g8:sexual-reproduction:core}{zigot} yang membelah menjadi sebuah individu yang terkocok ---
intinya tidak mungkin berbeda, sebab kalau berbeda ia bukan lagi
\omterm{def:g4:animal-reproduction:sexual}{reproduksi seksual}.

\textbf{11.} Kocokannya mengisi mejanya --- tiap generasi membagikan
paduan yang tidak dirancang siapa pun; pemilihan oleh pemulianya
hanyalah asas cadangan yang dijalankan dengan sengaja: di tempat
\omterm{def:g6:exploring-our-environment:conditions}{kondisi} alam yang berubah memilih yang cocok, ia memilih yang renyah
dan tahan. Tanpa keragaman yang dibagikan, tidak ada yang dapat
dipilih.

\textbf{12.} \omterm{def:g4:animal-reproduction:sexual}{Reproduksi seksual} dibutuhkan untuk \emph{menciptakan}
paduan yang baru itu; \omterm{def:g4:plants-without-seeds:vegetative}{perkembangbiakan vegetatif} untuk
\emph{melipatgandakannya tanpa berubah}.
\end{solution}
