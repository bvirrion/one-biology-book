\chapter{التكاثر الجنسي}\label{ch:g8:sexual-reproduction}

غناء الضفادع في أيار، وزهر الكرز في نيسان، وخوار الأيّل في الخريف،
وعبّاد شمس يدير وجهه المزدحم نحو النحل: في العالم الحي كله تُنفَق

\cref{ch:g5:flower-to-fruit})؛ وهذه السنة تركّبها نظرية واحدة،
وتسأل السؤال الأحدّ: ولماذا \emph{والدان} أصلًا؟

\section{آلية واحدة في العالم الحي كله}

\begin{definition}[الأمشاج]\label{def:g8:sexual-reproduction:gametes}
\omterm{def:g5:first-look-at-cells:cell}{الخلايا} التكاثرية في \cref{def:g4:animal-reproduction:sexual} تحمل
الاسم العام \emph{الأمشاج}\index{مشيج}: المشيج الذكري (\omterm{def:g4:animal-reproduction:sexual}{النطفة} عند
\omterm{def:g1:animals-around-us:animal}{الحيوانات}، \omterm{def:g5:first-look-at-cells:cell}{وخلية} حبة اللقاح عند \omterm{def:g1:plants-around-us:plant}{النباتات} المزهرة) والمشيج الأنثوي
(\omterm{def:g4:animal-reproduction:sexual}{البويضة}، في المبيض أو في البييضة). والأمشاج \omterm{def:g5:first-look-at-cells:cell}{خلايا} مفردة
(\cref{ch:g5:first-look-at-cells})، وكل واحد منها نصف بداية:
فالمشيج وحده يموت؛ وإذا التحم اثنان صنعا حياة.
\end{definition}

\begin{proposition}[اللبّ الثابت]\label{prop:g8:sexual-reproduction:core}
حيثما جرى \omterm{def:g4:animal-reproduction:sexual}{التكاثر الجنسي} --- في سمكة أو شجرة تنّوب، في حلزون أو
إنسان --- كان اللبّ ثابتًا:
\begin{enumerate}
  \item ينتج والدان أمشاجًا، ذكرية وأنثوية؛
  \item ويقع \emph{\omterm{def:g5:flower-to-fruit:steps}{الإخصاب}}: فيلتحم \omterm{def:g8:sexual-reproduction:gametes}{مشيج} ذكري واحد \omterm{def:g8:sexual-reproduction:gametes}{بمشيج} أنثوي
        واحد، فيتكوّن \omterm{def:g5:first-look-at-cells:cell}{خلية} جديدة واحدة --- هي
        \emph{\omterm{def:g4:animal-reproduction:sexual}{البويضة} المخصَّبة}، أو
        \emph{اللاقحة}\index{لاقحة}؛
  \item وتنقسم \omterm{prop:g8:sexual-reproduction:core}{اللاقحة} وتنقسم
        (\cref{prop:g5:first-look-at-cells:growth})، فتبني فردًا
        جديدًا يحمل شيئًا من \emph{كلا} الوالدين
        (\cref{rem:g4:animal-reproduction:mix}).
\end{enumerate}
وحول هذا اللبّ يكون كل ما عداه تنويعًا: أين تلتقي \omterm{def:g8:sexual-reproduction:gametes}{الأمشاج}، وكم
عددها، ومن يعتني بالصغار.
\end{proposition}
\admitted

\begin{example}[التنويعات مفروزةً]\label{ex:g8:sexual-reproduction:variations}
تُصنَّف المجموعة القديمة تحت اللبّ تصنيفًا أنيقًا. \omterm{def:g5:flower-to-fruit:steps}{فالإخصاب} الخارجي
في مقابل الداخلي
(\cref{prop:g4:animal-reproduction:where}): موضع اللقاء. \omterm{def:g4:animal-reproduction:ovivivi}{والبيوض}
في مقابل \omterm{def:g4:animal-reproduction:ovivivi}{الولود} (\cref{def:g4:animal-reproduction:ovivivi}): مسكن
\omterm{prop:g8:sexual-reproduction:core}{اللاقحة}. وملايين سمكة القدّ في مقابل مفردات الفيلة
(\cref{prop:g4:animal-reproduction:strategies}): دفتر الأعداد.
\omterm{def:g5:flower-to-fruit:steps}{والتأبير} بالنحل أو بالريح
(\cref{def:g5:flower-to-fruit:steps}): خدمة سفر \omterm{def:g8:sexual-reproduction:gametes}{المشيج} الذكري.
آلية واحدة، وفهرس من الإعدادات.
\end{example}

\begin{remark}[النباتات تجري على اللبّ نفسه]\label{rem:g8:sexual-reproduction:plants}
ضع \cref{ch:g5:flower-to-fruit} بجانب
\cref{ch:g4:animal-reproduction} فالمطابقة تامّة: \omterm{def:g3:flowers-fruits-seeds:parts}{فالأسدية} \omterm{def:g3:flowers-fruits-seeds:parts}{والمدقة}
تنتجان \omterm{def:g8:sexual-reproduction:gametes}{الأمشاج}، وأنبوب اللقاح ممرّ \omterm{def:g5:flower-to-fruit:steps}{الإخصاب} الداخلي، \omterm{def:g2:from-seed-to-plant:seed}{والبذرة} تُسكن
\omterm{def:g5:human-reproduction:pregnancy}{مضغة} \omterm{def:g1:plants-around-us:plant}{النبات} مع غدائها المحزوم
(\cref{def:g2:from-seed-to-plant:seed}). ولا يقف على حدة إلا
\omterm{def:g4:plants-without-seeds:vegetative}{التكاثر الخضري} (\cref{def:g4:plants-without-seeds:vegetative}) ---
والد واحد، ولا \omterm{def:g8:sexual-reproduction:gametes}{أمشاج}، ونسخ مطابقة: وهو التقابل الذي يجعل سؤال
القسم التالي حادًّا.
\end{remark}

\section{ولماذا والدان؟}

\begin{proposition}[التكاثر الجنسي يخلط]\label{prop:g8:sexual-reproduction:shuffle}
نسخ الفراولة بالمدّاد
(\cref{rem:g4:plants-without-seeds:copy}) تبين ما يعطيه والد واحد:
التشابه. أما الوالدان فيعطيان النقيض. فلأن كل لاقحة تجمع مشيجًا من
كل والد --- ولأن مشيجين حتى من والد واحد لا يحملان الميراث نفسه
تمامًا، كما سيبين فصل لاحق --- يوزّع \omterm{def:g4:animal-reproduction:sexual}{التكاثر الجنسي} على كل خلَف
\emph{تركيبةً جديدة}. فالإخوة يختلفون؛ ولا بادرتان من قرن واحد
تتطابقان؛ والتنوع داخل كل \omterm{def:g5:biodiversity:species}{نوع}
(\cref{rem:g5:biodiversity:scales}) عمل هذا الخلط في معظمه.
\end{proposition}
\admitted

\begin{example}[رهانا البستان]\label{ex:g8:sexual-reproduction:bets}
بستان \cref{rem:g4:plants-without-seeds:copy} يلعب اللعبتين عن
علم. فالتطعيم \omterm{met:g4:plants-without-seeds:cutting}{والعُقَل}: والد واحد، ونسخ مطابقة للشجرة الممتازة ---
وموطن ضعف مشترك واحد، فالآفة التي تُسقط واحدة تُسقط الجميع. أما
\omterm{def:g2:from-seed-to-plant:seed}{البذور}: فوالدان، وكل بادرة تركيبة جديدة --- أكثرها لا يلفت، وبعضها
أسوأ، وواحدة نادرة أحسن من كلا الوالدين، ولا اثنتان متشابهتان
لتكتسحهما آفة. فالنسخ يودع الحاضر؛ والخلط يستكشف المستقبل.
\end{example}

\begin{remark}[التنوع احتياطي النوع]\label{rem:g8:sexual-reproduction:reserve}
ولماذا يهمّ الخلط خارج البستان: فقد بيّنت
\cref{prop:g5:biodiversity:web} أن الشبكات تمتصّ الصدمات بتنوع
\omterm{def:g5:biodiversity:species}{الأنواع}؛ وداخل \omterm{def:g5:biodiversity:species}{النوع} الواحد يؤدي تنوع الأفراد الدور نفسه. فإذا
تغيرت \omterm{def:g6:exploring-our-environment:conditions}{الشروط} --- مرض جديد، أو شتاء أقسى --- كان \omterm{def:g5:biodiversity:species}{النوع} المتنوع أرجح
أن يضمّ من يناسب \omterm{def:g6:exploring-our-environment:conditions}{الشروط} الجديدة. أما ما يلمّح إليه ذلك في تاريخ
الحياة فستصرّح به \cref{ch:g9:how-species-change}؛ وحسبنا هنا أن
نلاحظ أن \omterm{def:g4:animal-reproduction:sexual}{التكاثر الجنسي} هو الذي يُبقي الاحتياطي ممونًا.
\end{remark}

\section{قراءة تكاثر أي نوع}

\begin{method}[ملف التكاثر]\label{met:g8:sexual-reproduction:dossier}
لأي \omterm{def:g5:biodiversity:species}{نوع}، اجمع:
\begin{enumerate}
  \item \omterm{def:g8:sexual-reproduction:gametes}{الأمشاج}: من ينتج أيها، وأين؛
  \item واللقاء: \omterm{def:g4:animal-reproduction:sexual}{إخصاب} خارجي أم داخلي، وخدمة السفر (ماء، أو
        تزاوج، أو مؤبّر، أو ريح)؛
  \item والمسكن: \omterm{def:g2:animals-grow-up:born}{بيض} يُوضع أم صغار تُحمل؛ وبأي مؤن؛
  \item والأعداد والعناية: الموضع على خط الخطط
        (\cref{prop:g4:animal-reproduction:strategies})؛
  \item والموسم: متى، وبم يُوقَّت --- وهو سؤال الفصل التالي.
\end{enumerate}
\end{method}

\begin{example}[ملفّان جنبًا إلى جنب]\label{ex:g8:sexual-reproduction:dossiers}
السمكة الشوكية (\cref{exo:g4:animal-reproduction:10}): \omterm{def:g8:sexual-reproduction:gametes}{أمشاج}
تُلقى في الماء، \omterm{def:g4:animal-reproduction:sexual}{وإخصاب} خارجي، \omterm{def:g2:animals-grow-up:born}{وبيض} في عش مبني، بالعشرات، يحرسه
الأب، في الربيع. والكرز: \omterm{def:g8:sexual-reproduction:gametes}{أمشاج} في الزهر، \omterm{def:g4:animal-reproduction:sexual}{وإخصاب} داخلي في أنبوب
اللقاح، \omterm{def:g5:human-reproduction:pregnancy}{ومضغ} تسكن بذورًا داخل \omterm{prop:g3:flowers-fruits-seeds:transform}{ثمار}، بالآلاف، تُموَّن ثم تُترك
لخدمة توصيل \omterm{prop:g3:sorting-living-things:groups}{الطيور}
(\cref{exo:g6:how-plants-spread:12})، موقوتةً على نيسان.
الصيغة نفسها لأي \omterm{def:g5:biodiversity:species}{نوع}: فالملف هو النظرية في الاستعمال.
\end{example}

\section{مسألة: برنامج التربية}

\begin{problem}[{مسألة نهاية الأسبوع --- سنة مربّية نباتات،
مُدارة على النظرية}]\label{pb:g8:sexual-reproduction:1}
تريد مربّية تفاحًا جديدًا: مقرمشًا كالصنف أ، ومقاومًا للآفة كالصنف
ب. وأدواتها هي أدوات هذا الفصل بعينها.

\medskip\noindent\textbf{القسم الأول --- التهجين.}
\begin{enumerate}
  \item تأخذ لقاحًا من \omterm{def:g3:flowers-fruits-seeds:parts}{أسدية} أ بفرشاة وتمسّ به مدقات ب، ثم تكيّس
        الأزهار في وجه النحل. سمِّ دور كل خطوة في اللبّ ---
        ولماذا تهمّ الأكياس
        (وقد استعملت \cref{met:g5:flower-to-fruit:diagnose}
        الفرشاة نفسها).
  \item وماذا ينمو في \omterm{prop:g3:flowers-fruits-seeds:transform}{ثمرة} ب بعد زيارتها، وأي ميراثين تجمعهما كل
        \omterm{def:g2:from-seed-to-plant:seed}{بذرة}؟
  \item وتبذر ثلاث مئة \omterm{def:g2:from-seed-to-plant:seed}{بذرة}. فلماذا تتوقع ثلاث مئة شجرة
        \emph{مختلفة}، وفق
        \cref{prop:g8:sexual-reproduction:shuffle}؟
  \item وأكثر البادرات لن يلفت، وقليل منها أسوأ من كلا الوالدين،
        ولعل واحدة تكون المطلوبة تمامًا. فأي مثال في الفصل تنبأ
        بهذا الانتشار بالضبط؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- تثبيت الفائزة.}
\begin{enumerate}[resume]
  \item السنة الثامنة: شجرة واحدة تحمل تفاحًا مقرمشًا مقاومًا
        للآفة. فإن بذرت بذورها، ماذا يحدث للتركيبة الثمينة ---
        ولماذا؟
  \item وبدل ذلك تكثّر الفائزة بالتطعيم --- فأي تكاثر ذاك، وبم
        يضمن؟
  \item وبستانها المستقبلي كله سيكون نسخًا لتلك الشجرة الواحدة.
        سمِّ الخطر الذي قبلته، وعبارة الفصل الدالة عليه.
  \item واقترح التأمين المعتاد: ما الذي ينبغي أن يُبقي عليه
        بستانها أيضًا، ولماذا تردّد الخطة صدى
        \cref{rem:g8:sexual-reproduction:reserve}؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- تدقيق
النظرية.}
\begin{enumerate}[resume]
  \item استعمل برنامجها رهانَي البستان كليهما بالتتابع. اذكر أي
        مرحلة استعملت أيهما، وبم أفادها كل واحد.
  \item ويربّي زميل \omterm{prop:g3:sorting-living-things:groups}{الأسماك} الشوكية بالطريقة نفسها --- باختيار
        الوالدين، وتربية الصغار المخلوطة، والاحتفاظ بالأفضل.
        عدّد بنود الملف
        (\cref{met:g8:sexual-reproduction:dossier}) التي تختلف
        فيها ممارسة السمك عن ممارسة التفاح، وخطوات اللبّ التي
        لا يمكن أن تختلف.
  \item تقول المربّية: ``الجنس يعرض، والانتقاء يقرر --- وأنا لا
        أفعل إلا الاختيار مما يوزّعه الخلط.'' اربط شعارها
        باحتياطي التنوع في
        \cref{rem:g8:sexual-reproduction:reserve}.
  \item جملة واحدة للختام: \emph{لماذا} احتاج البرنامج إلى
        \omterm{def:g4:animal-reproduction:sexual}{التكاثر الجنسي}، ولماذا احتاج إلى \omterm{def:g4:plants-without-seeds:vegetative}{التكاثر الخضري}؟
\end{enumerate}
\end{problem}
