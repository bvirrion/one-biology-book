\chapter{Animals Around Us}\label{ch:g1:animals-around-us}

A blackbird hops across the lawn, a cat watches it from the wall,
an ant carries a crumb bigger than itself along the doorstep.
\omterm{def:g1:animals-around-us:animal}{Animals} are everywhere around us --- in the garden, in the park, in
the house. Let us learn to look at them the way a biologist does:
at their bodies, and at the way they move.

\section{What is an animal?}

\begin{definition}[Animal]\label{def:g1:animals-around-us:animal}
An \emph{animal}\index{animal} is a \omterm{def:g1:living-or-not:living}{living thing} that feeds by
eating other \omterm{def:g1:living-or-not:living}{living things} --- plants, or other animals --- and
that can usually move about on its own. A cat, a blackbird, a fish
and an ant are all animals. We are animals too.
\end{definition}

\begin{example}[Animals of one garden]\label{ex:g1:animals-around-us:garden}
One small garden can be home to many \omterm{def:g1:animals-around-us:animal}{animals}: a blackbird pulling a
worm out of the ground, the worm itself, a snail on a leaf, a
spider in its web, a bee in a flower, and the neighbor's cat.
Big or tiny, they all feed and they all move.
\end{example}

\begin{remark}[Small does not mean unimportant]\label{rem:g1:animals-around-us:small}
An ant is an \omterm{def:g1:animals-around-us:animal}{animal} just as much as an elephant is. Body size does
not decide who is an \omterm{def:g1:animals-around-us:animal}{animal}: eating and moving do. Some of the most
amazing \omterm{def:g1:animals-around-us:animal}{animals} around us are the tiniest ones.
\end{remark}

\section{The parts of an animal's body}

\begin{definition}[Parts of the body]\label{def:g1:animals-around-us:parts}
Most \omterm{def:g1:animals-around-us:animal}{animals} have a \emph{head}\index{head} (with the mouth and
most sense organs), a body, and \emph{limbs}\index{limb} --- legs,
wings or fins --- that they use to move. Many also have a
\emph{tail}\index{tail}.
\end{definition}

\begin{example}[Count the legs]\label{ex:g1:animals-around-us:legs}
Legs come in numbers worth counting. The blackbird walks on $2$
legs, the cat on $4$, the ladybug on $6$, the spider on $8$ --- and
the earthworm and the snail have no legs at all, yet they still get
where they are going.
\end{example}

\begin{definition}[Body covering]\label{def:g1:animals-around-us:covering}
The body of an \omterm{def:g1:animals-around-us:animal}{animal} is protected by a covering:
\emph{fur}\index{fur} (hair) on the cat, \emph{feathers}\index{feather}
on the blackbird, \emph{scales}\index{scale} on the fish, a
\emph{shell}\index{shell} on the snail, and smooth wet
\emph{skin}\index{skin} on the frog.
\end{definition}

\begin{example}[Who wears what]\label{ex:g1:animals-around-us:covering}
Stroke a cat: \omterm{def:g1:animals-around-us:covering}{fur}. Pick up a fallen blackbird \omterm{def:g1:animals-around-us:covering}{feather}: light and
soft. Look closely at a fish at the market: rows of shiny scales,
like tiny roof tiles. Three \omterm{def:g1:animals-around-us:animal}{animals}, three different coats.
\end{example}

\section{How animals move}

\begin{proposition}[Ways of moving]\label{prop:g1:animals-around-us:moving}
\omterm{def:g1:animals-around-us:animal}{Animals} move in ways that match their \omterm{def:g1:animals-around-us:parts}{limbs}:
\begin{itemize}
  \item \omterm{def:g1:animals-around-us:animal}{animals} with legs \emph{walk}, run, hop or crawl --- the
        cat, the blackbird on the lawn, the ant;
  \item \omterm{def:g1:animals-around-us:animal}{animals} with wings \emph{fly}\index{fly} --- the blackbird,
        the bee, the butterfly;
  \item \omterm{def:g1:animals-around-us:animal}{animals} with fins \emph{swim}\index{swim} --- the fish;
  \item \omterm{def:g1:animals-around-us:animal}{animals} with no \omterm{def:g1:animals-around-us:parts}{limbs} slide or wriggle --- the snail on its
        one soft foot, the earthworm stretching and shortening.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[One animal, several ways]\label{rem:g1:animals-around-us:several}
Many \omterm{def:g1:animals-around-us:animal}{animals} have more than one way of moving. The blackbird walks
\emph{and} flies; the frog swims \emph{and} hops; the duck walks,
swims \emph{and} flies. Watch an \omterm{def:g1:animals-around-us:animal}{animal} for a while before deciding
how it moves.
\end{remark}

\begin{method}[Observing an animal]\label{met:g1:animals-around-us:observe}
To describe an \omterm{def:g1:animals-around-us:animal}{animal} like a biologist:
\begin{enumerate}
  \item look at the parts: head, body, \omterm{def:g1:animals-around-us:parts}{limbs}, tail;
  \item count the legs, without guessing;
  \item name the covering: \omterm{def:g1:animals-around-us:covering}{fur}, \omterm{def:g1:animals-around-us:covering}{feathers}, scales, \omterm{def:g1:animals-around-us:covering}{shell}, \omterm{def:g1:animals-around-us:covering}{skin};
  \item watch how it moves: walk, hop, fly, swim, slide;
  \item say where you found it: lawn, tree, pond, under a stone.
\end{enumerate}
Never hurt the \omterm{def:g1:animals-around-us:animal}{animal}: watch, then let it go on its way.
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:1}
Name the four main parts of the cat's body in the figure. Which
part carries the eyes and the mouth?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:2}
How many legs: (a) a hen? (b) a dog? (c) a ladybug? (d) a spider?
(e) an earthworm?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:3}
Give the covering of each \omterm{def:g1:animals-around-us:animal}{animal}: a sheep, a duck, a trout, a
snail.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:4}
Which \omterm{def:g1:animals-around-us:animal}{animals} of this chapter can fly? What do they all have,
instead of front legs or arms?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:5}
The snail and the earthworm have no legs. How does each of them
move?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:6}
A duck can move in three ways. Name them, and say which part of its
body it uses for each.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:7}
Why is an ant an \omterm{def:g1:animals-around-us:animal}{animal}, by \cref{def:g1:animals-around-us:animal}?
Answer with the two ideas of the definition.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:8}
Try it: watch one \omterm{def:g1:animals-around-us:animal}{animal} outside for a little while --- a bird, an
insect, a snail. Describe it with the five steps of
\cref{met:g1:animals-around-us:observe}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:9}
A friend says: ``Fish cannot be \omterm{def:g1:animals-around-us:animal}{animals} --- they do not walk!''
What would you answer? Use \cref{prop:g1:animals-around-us:moving}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:10}
Mystery \omterm{def:g1:animals-around-us:animal}{animal}: it has $2$ legs, $2$ wings, \omterm{def:g1:animals-around-us:covering}{feathers}, and it cannot
fly, but it runs very fast. Is it a bird? Could it be the fastest
runner of your neighborhood? (Think of an ostrich.)
\end{exercise}
