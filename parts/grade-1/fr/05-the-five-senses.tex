\chapter{Les cinq sens}\label{ch:g1:the-five-senses}

Ferme les \omterm{def:g1:the-five-senses:sense}{yeux}~: le monde ne disparaît pas pour autant~; tu entends
encore la classe, tu sens le repas qui cuit, tu sens la chaise sous
toi. Le corps a cinq portes vers le monde --- les sens --- et
chaque porte est un organe du corps, qui travaille pour toi toute la
journée.

\section{Cinq sens, cinq organes}

\begin{definition}[Sens et organe des sens]\label{def:g1:the-five-senses:sense}
Un \emph{sens}\index{sens} est une façon, pour le corps, d'apprendre
quelque chose sur le monde. Chaque sens travaille grâce à un
\emph{organe des sens}\index{organe des sens}, une partie du corps
faite pour ce travail~:
\begin{enumerate}
  \item la \emph{vue}\index{vue} --- les \emph{yeux}\index{œil}
        voient la lumière, les couleurs et les formes~;
  \item l'\emph{ouïe}\index{ouïe} --- les \emph{oreilles}\index{oreille}
        entendent les sons~;
  \item l'\emph{odorat}\index{odorat} --- le \emph{nez}\index{nez}
        sent les odeurs~;
  \item le \emph{goût}\index{goût} --- la \emph{langue}\index{langue}
        goûte les aliments~;
  \item le \emph{toucher}\index{toucher} --- la \emph{peau}\index{peau},
        sur tout le corps, sent le rugueux et le lisse, le chaud et
        le froid.
\end{enumerate}
\end{definition}

\begin{example}[Le petit déjeuner, sens par sens]\label{ex:g1:the-five-senses:breakfast}
Un seul bol de chocolat chaud emploie les cinq~: les \omterm{def:g1:the-five-senses:sense}{yeux} voient la
vapeur, le nez sent le cacao, la \omterm{def:g1:the-five-senses:sense}{peau} des mains sent le bol tiède,
les \omterm{def:g1:the-five-senses:sense}{oreilles} entendent tinter la cuillère, et la langue --- le
meilleur moment --- le goûte.
\end{example}

\begin{remark}[Le toucher est partout]\label{rem:g1:the-five-senses:skin}
Quatre \omterm{def:g1:the-five-senses:sense}{organes des sens} habitent la tête, mais l'organe du toucher
est la \omterm{def:g1:the-five-senses:sense}{peau}, et la \omterm{def:g1:the-five-senses:sense}{peau} te couvre de la tête aux pieds. C'est
pourquoi tu peux sentir un courant d'air sur ta nuque, un caillou
dans ta chaussure et une coccinelle qui se pose sur ton \omterm{def:g1:my-body:parts}{bras} ---
sans regarder.
\end{remark}

\section{Les sens font leur rapport au cerveau}

\begin{proposition}[Des messages vers le cerveau]\label{prop:g1:the-five-senses:brain}
Un \omterm{def:g1:the-five-senses:sense}{organe des sens} ne comprend pas ce qu'il sent --- il \emph{envoie
un message}. Tous les messages voyagent à l'intérieur du corps
jusqu'au \emph{cerveau}\index{cerveau}, dans la tête, qui les
rassemble~: «~c'est la voix de mamie~», «~cette odeur, c'est du pain
grillé~». Sans le cerveau, les \omterm{def:g1:the-five-senses:sense}{yeux} et les \omterm{def:g1:the-five-senses:sense}{oreilles} feraient leur
rapport à personne.
\end{proposition}
\admitted

\begin{example}[Nommer les choses dans le noir]\label{ex:g1:the-five-senses:dark}
Une camarade te met dans les mains un objet familier, les \omterm{def:g1:the-five-senses:sense}{yeux}
fermés~: une clé, une éponge, une cuillère. Tu le nommes presque
tout de suite. La \omterm{def:g1:the-five-senses:sense}{peau} a envoyé son message --- froid, dur, dentelé
--- et le cerveau a répondu~: «~une clé~». Les sens récoltent~; le
cerveau comprend.
\end{example}

\begin{remark}[S'aider les uns les autres]\label{rem:g1:the-five-senses:together}
Les sens travaillent en équipe. Les aliments semblent fades quand un
rhume bouche le nez, parce que l'odorat faisait la moitié du travail
du goût. Et quand un sens ne peut plus faire son travail du tout,
les autres apprennent à en faire davantage~: les personnes qui ne
voient pas lisent du bout des doigts, avec le toucher, et traversent
la rue avec les \omterm{def:g1:the-five-senses:sense}{oreilles}.
\end{remark}

\section{Prendre soin des organes des sens}

\begin{proposition}[Des portes fragiles]\label{prop:g1:the-five-senses:fragile}
Les \omterm{def:g1:the-five-senses:sense}{organes des sens} sont précieux et faciles à abîmer. Un son trop
fort abîme les \omterm{def:g1:the-five-senses:sense}{oreilles}~; regarder le Soleil abîme les \omterm{def:g1:the-five-senses:sense}{yeux}~; une
cuillerée trop chaude brûle la langue. Un \omterm{def:g1:the-five-senses:sense}{organe des sens} abîmé peut
le rester --- alors nous les protégeons.
\end{proposition}
\admitted

\begin{method}[Protéger mes sens]\label{met:g1:the-five-senses:protect}
\begin{enumerate}
  \item Ne regarde jamais le Soleil en face, pas même un instant~;
  \item garde les sons doux~: pas de cris dans les \omterm{def:g1:the-five-senses:sense}{oreilles}, un
        casque à faible volume~;
  \item n'enfonce jamais rien dans ton \omterm{def:g1:the-five-senses:sense}{oreille} ni dans ton nez~;
  \item souffle sur les plats chauds avant d'y goûter~;
  \item lis et dessine avec une bonne lumière.
\end{enumerate}
\end{method}

\begin{example}[Le concert d'à côté]\label{ex:g1:the-five-senses:loud}
Après une musique très forte, les \omterm{def:g1:the-five-senses:sense}{oreilles} sifflent parfois un
moment. Ce sifflement, c'est l'\omterm{def:g1:the-five-senses:sense}{oreille} qui dit «~trop~!~». Les
personnes qui travaillent avec des machines bruyantes portent des
protections pour exactement cette raison --- leurs \omterm{def:g1:the-five-senses:sense}{oreilles} doivent
leur durer toute la vie.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:1}
Nomme les cinq sens et, pour chacun, son \omterm{def:g1:the-five-senses:sense}{organe des sens}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:2}
Quel \omterm{def:g1:the-five-senses:sense}{organe des sens} fait le rapport~: (a) de la couleur d'un
ballon~? (b) de la douceur d'une couverture~? (c) de l'odeur de la
pluie~? (d) du son d'une cloche~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:3}
Quel \omterm{def:g1:the-five-senses:sense}{organe des sens} n'est pas sur la tête~? Jusqu'où va-t-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:4}
Dans \cref{ex:g1:the-five-senses:dark}, les \omterm{def:g1:the-five-senses:sense}{yeux} étaient fermés.
Quel organe a envoyé le message, et quelle partie du corps l'a
compris~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:5}
Avec un rhume et le nez bouché, les aliments semblent perdre leur
saveur. Quels deux sens font équipe d'habitude quand tu manges~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:6}
Donne trois règles de \cref{met:g1:the-five-senses:protect}, et dis
quel \omterm{def:g1:the-five-senses:sense}{organe des sens} chacune protège.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:7}
Pourquoi les personnes qui travaillent avec des machines bruyantes
portent-elles des protections d'\omterm{def:g1:the-five-senses:sense}{oreilles}~? Utilise
\cref{prop:g1:the-five-senses:fragile}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:8}
À toi d'essayer~: les \omterm{def:g1:the-five-senses:sense}{yeux} fermés, demande à quelqu'un de te
présenter trois aliments sous le nez --- banane, fromage, orange par
exemple. Combien ton nez en a-t-il nommé correctement, sans l'aide
des \omterm{def:g1:the-five-senses:sense}{yeux}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:the-five-senses:9}
Un camarade affirme~: «~L'\omterm{def:g1:the-five-senses:sense}{œil} comprend ce qu'il voit.~» Corrige la
phrase avec \cref{prop:g1:the-five-senses:brain}~: que fait
vraiment l'\omterm{def:g1:the-five-senses:sense}{œil}, et qui fait la compréhension~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:the-five-senses:10}
Comment une personne qui ne voit pas peut-elle «~lire~» un livre et
traverser une rue en sécurité~? Dis quels sens reprennent quels
travaux.
\end{exercise}
