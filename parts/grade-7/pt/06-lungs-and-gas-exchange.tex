\chapter{Os pulmões e as trocas gasosas}\label{ch:g7:lungs-and-gas-exchange}

A cada minuto da sua vida, cerca de oito litros de ar passam por dois
órgãos que você nunca viu. O \cref{ch:g4:breathing} esboçou o caminho
e a troca; o \cref{ch:g7:how-animals-breathe} situou a máquina humana
entre as quatro do reino \omterm{def:g1:animals-around-us:animal}{animal}. Este capítulo completa o quadro: a
maquinaria de ventilação, as bolsinhas de ar com o nome próprio delas,
os números da troca --- e o que a fumaça faz com o conjunto todo.

\section{A maquinaria de ventilação}

\begin{proposition}[Como o ar é movido]\label{prop:g7:lungs-and-gas-exchange:ventilation}
Os \omterm{def:g7:how-animals-breathe:four}{pulmões} não têm \omterm{def:g3:bones-and-muscles:muscle}{músculos} próprios; o peito os bombeia como um fole:
\begin{itemize}
  \item inspirar é \emph{trabalho}: os \omterm{def:g3:bones-and-muscles:muscle}{músculos} das \omterm{def:g3:bones-and-muscles:skeleton}{costelas} levantam
        e alargam a caixa, e o \emph{diafragma}\index{diafragma} --- o
        piso muscular largo debaixo dos \omterm{def:g7:how-animals-breathe:four}{pulmões} --- se achata para
        baixo; o peito alargado estica os \omterm{def:g7:how-animals-breathe:four}{pulmões}, e o ar entra para
        enchê-los;
  \item expirar, em repouso, é quase todo \emph{soltura}: os \omterm{def:g3:bones-and-muscles:muscle}{músculos}
        relaxam, o peito esticado volta como uma mola, o ar sai.
        (Apagar velinhas acrescenta \omterm{def:g3:bones-and-muscles:muscle}{músculo} à mola.)
\end{itemize}
Cerca de $15$ vezes por minuto em repouso
(\cref{def:g4:breathing:movements}), sem ninguém pedir
(\cref{rem:g5:brain-in-command:more}) --- e mais fundo e mais rápido
sob demanda (\cref{prop:g7:organs-at-work:supply}).
\end{proposition}
\admitted

\begin{example}[Sentindo o fole]\label{ex:g7:lungs-and-gas-exchange:feel}
Mãos nas \omterm{def:g3:bones-and-muscles:skeleton}{costelas}, inspire fundo: a caixa sobe e se alarga debaixo das
suas palmas (\cref{ex:g4:breathing:watch}). Agora ofegue como um
cachorro: os sopros rápidos e rasos são quase todos \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{diafragma}. Os
soluços, para completar, são as falsas partidas espasmódicas do
\omterm{prop:g7:lungs-and-gas-exchange:ventilation}{diafragma} --- o fole falhando no horário próprio dele.
\end{example}

\section{Os alvéolos e a troca}

\begin{definition}[Alvéolos]\label{def:g7:lungs-and-gas-exchange:alveoli}
As bolsinhas de ar da \cref{def:g4:breathing:organs} são os
\emph{alvéolos}\index{alvéolo}: centenas de milhões de bolhas de
parede fina agrupadas nas pontas mais finas das vias aéreas, cada uma
envolvida por \omterm{def:g4:heart-and-blood:vessels}{vasos sanguíneos} finos. A superfície somada delas é a
meia quadra de tênis da \cref{rem:g4:breathing:sponge} --- a cópia
humana da especificação grande, fina e úmida da
\cref{prop:g7:how-animals-breathe:spec}.
\end{definition}

\begin{proposition}[A troca, em números]\label{prop:g7:lungs-and-gas-exchange:numbers}
Compare o ar que entra com o ar que sai: o ar inspirado tem cerca de
$21$ por cento de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} e quase nada de \omterm{def:g7:what-is-respiration:respiration}{gás carbônico}; o ar
expirado cai para uns $16$ por cento de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} e chega a uns $4$ por
cento de \omterm{def:g7:what-is-respiration:respiration}{gás carbônico} (além de quente e úmido ---
\cref{ex:g4:breathing:mirror}). As diferenças são a troca: através das
paredes alveolares, o \omterm{def:g7:what-is-respiration:respiration}{oxigênio} atravessa para o \omterm{prop:g4:journey-of-food:absorb}{sangue} e o \omterm{def:g7:what-is-respiration:respiration}{gás
carbônico} atravessa de volta
(\cref{prop:g4:breathing:exchange}), todo minuto de todo dia.
\end{proposition}
\admitted

\begin{example}[As contas batem]\label{ex:g7:lungs-and-gas-exchange:accounting}
Oito litros por minuto, com um vigésimo do \omterm{def:g7:what-is-respiration:respiration}{oxigênio} de cada \omterm{def:g7:what-is-respiration:respiration}{respiração}
retido: a aritmética atende mais ou menos a conta de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} do corpo
em repouso --- e, no esforço, com a \omterm{def:g7:what-is-respiration:respiration}{respiração} triplicada e mais funda
e as rodadas do \omterm{def:g4:heart-and-blood:heart}{coração} aceleradas
(\cref{prop:g7:organs-at-work:supply}), a entrega se multiplica para
igualar a conta dos \omterm{def:g3:bones-and-muscles:muscle}{músculos}. Os números da máquina e o balanço do
corpo fecham; são os mesmos livros, mantidos em dois balcões.
\end{example}

\section{A maquinaria maltratada: a fumaça}

\begin{proposition}[O que a fumaça faz com o conjunto]\label{prop:g7:lungs-and-gas-exchange:smoke}
A fumaça do cigarro ataca todos os andares deste capítulo (a
\cref{prop:g5:healthy-choices:substances} deu o aviso; aqui está a
engenharia):
\begin{itemize}
  \item o alcatrão dela cobre e irrita as vias aéreas: as superfícies
        de limpeza vacilam, e a ``tosse do fumante'' é a vassoura
        sobrecarregada das vias aéreas;
  \item as paredes alveolares, irritadas de forma crônica, se
        desfazem: as bolsinhas se fundem em espaços menos numerosos,
        maiores e mais frouxos --- \emph{a superfície é perdida}, e a
        troca com ela (a especificação, desmontada);
  \item um dos gases da fumaça ocupa os assentos de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} do
        \omterm{prop:g4:journey-of-food:absorb}{sangue}, expulsando o \omterm{def:g7:what-is-respiration:respiration}{oxigênio} do próprio transporte dele;
  \item e as vias aéreas se estreitam --- toda \omterm{def:g7:what-is-respiration:respiration}{respiração} pelo peito
        de um fumante é uma \omterm{def:g7:what-is-respiration:respiration}{respiração} por um fole parcialmente
        bloqueado.
\end{itemize}
O dano é lento, cumulativo e só em parte reversível: o \omterm{def:g7:how-animals-breathe:four}{pulmão} que
nunca começou mantém a meia quadra de tênis dele inteira
(\cref{rem:g5:healthy-choices:never}).
\end{proposition}
\admitted

\begin{example}[Dois testes de escada]\label{ex:g7:lungs-and-gas-exchange:stairs}
Os três lances de escada do \cref{ch:g7:organs-at-work}, subidos por
dois adultos de mesma idade e mesmos hábitos de treino, um deles
fumante de longa data: o fumante chega ofegante num \omterm{prop:g4:heart-and-blood:pulse}{pulso} que o outro
só alcança correndo. Menos superfície, assentos ocupados, vias aéreas
estreitadas --- a cadeia de abastecimento paga a conta da
\cref{prop:g7:organs-at-work:bill} por uma entrada estrangulada. Os
testes de condicionamento leem a história dos \omterm{def:g7:how-animals-breathe:four}{pulmões} com a mesma
segurança que qualquer exame de imagem.
\end{example}

\begin{method}[Preservar a maquinaria]\label{met:g7:lungs-and-gas-exchange:keep}
O \cref{met:g4:breathing:care}, atualizado com os motivos deste
capítulo:
\begin{enumerate}
  \item nada de fumaça --- nem a sua, nem a dos outros: superfície
        perdida quase não se recupera;
  \item areje os cômodos (\cref{exo:g4:breathing:10}) e prefira ar
        limpo ao ar livre para o esporte;
  \item exercite o fole: o esforço com \omterm{def:g7:what-is-respiration:respiration}{respiração} funda constrói os
        \omterm{def:g3:bones-and-muscles:muscle}{músculos} respiratórios e mantém toda a superfície ventilada;
  \item respire pelo nariz no frio e na poeira
        (\cref{exo:g4:breathing:7}) --- o filtro protege a vassoura.
\end{enumerate}
\end{method}

\section{Problema: O dossiê da máquina respiratória}

\begin{problem}[{Problema de fim de semana --- a inspeção de um
engenheiro no fole humano}]\label{pb:g7:lungs-and-gas-exchange:1}
Uma turma de engenharia ``inspeciona'' a máquina respiratória humana
no papel: entrada, bomba, superfície de troca, ligação com o
transporte e histórico de manutenção.

\medskip\noindent\textbf{Parte I --- A bomba.}
\begin{enumerate}
  \item Monte o ciclo da bomba: as duas fases, quem move cada uma e
        que fase custa \omterm{prop:g3:balanced-diet:jobs}{energia} em repouso.
  \item A bomba não tem contato com o \omterm{def:g3:bones-and-muscles:muscle}{músculo} do próprio órgão
        bombeado --- os \omterm{def:g7:how-animals-breathe:four}{pulmões} não têm \omterm{def:g3:bones-and-muscles:muscle}{músculo}. Como o alargamento do
        peito consegue mover ar?
  \item Dê o ritmo de repouso da bomba e os dois ajustes de esforço
        dela (\cref{prop:g7:organs-at-work:supply}).
  \item Ofegar, suspirar, apagar velinhas: atribua cada um ao lugar
        dele na mecânica do ciclo.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- A superfície de
troca.}
\begin{enumerate}[resume]
  \item Especifique a superfície: nome, número, parede, envoltório,
        tamanho somado.
  \item Verifique a especificação contra a
        \cref{prop:g7:how-animals-breathe:spec}, item por item.
  \item O ar de entrada e o ar de saída diferem pelas porcentagens do
        capítulo. Calcule a diferença de \omterm{def:g7:what-is-respiration:respiration}{oxigênio} como fração dos $21$
        por cento da entrada --- mais ou menos que parte do \omterm{def:g7:what-is-respiration:respiration}{oxigênio}
        inspirado fica retida?
  \item Por que o ar de saída precisa continuar quente e úmido --- de
        que propriedade da superfície isso é o preço?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O histórico de
manutenção.}
\begin{enumerate}[resume]
  \item Arquive os quatro ataques da fumaça em: entrada, bomba,
        superfície, transporte. Uma linha para cada.
  \item O dossiê anota: ``as perdas de superfície são perdas de
        capital''. Explique, em contraste com as tosses e os
        resfriados, que são reparos correntes.
  \item Escreva o cronograma de manutenção --- o
        \cref{met:g7:lungs-and-gas-exchange:keep} --- como quatro
        diretrizes de engenheiro, com os motivos.
  \item Encerre a inspeção: uma frase sobre por que esta máquina, a
        única entre as bombas e fronteiras do corpo, está metade sob
        o controle voluntário do dono --- e para que serve esse
        controle (a fala, o canto, a \omterm{def:g4:breathing:movements}{expiração} lenta do mergulhador).
\end{enumerate}
\end{problem}
