\section*{Capítulo \ref{ch:g5:flower-to-fruit} --- De la flor al fruto}

\begin{solution}{exo:g5:flower-to-fruit:1}
Los \omterm{def:g3:flowers-fruits-seeds:parts}{estambres} son \omterm{def:g4:animal-reproduction:sexual}{masculinos}: sus granos de \omterm{def:g3:flowers-fruits-seeds:parts}{polen} llevan las \omterm{def:g5:first-look-at-cells:cell}{células}
reproductoras \omterm{def:g4:animal-reproduction:sexual}{masculinas}. El \omterm{def:g3:flowers-fruits-seeds:parts}{pistilo} es \omterm{def:g4:animal-reproduction:sexual}{femenino}: dentro de él están
los \omterm{def:g4:animal-reproduction:sexual}{óvulos}, cada uno con una \omterm{def:g5:first-look-at-cells:cell}{célula} reproductora \omterm{def:g4:animal-reproduction:sexual}{femenina}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:2}
\omterm{def:g5:flower-to-fruit:steps}{Polinización}: un grano de \omterm{def:g3:flowers-fruits-seeds:parts}{polen} viaja de un \omterm{def:g3:flowers-fruits-seeds:parts}{estambre} a lo alto de un
\omterm{def:g3:flowers-fruits-seeds:parts}{pistilo}. \omterm{def:g5:flower-to-fruit:steps}{Fecundación}: el tubo del grano lleva la \omterm{def:g5:first-look-at-cells:cell}{célula} \omterm{def:g4:animal-reproduction:sexual}{masculina}
hasta un \omterm{def:g4:animal-reproduction:sexual}{óvulo}, donde se une a la \omterm{def:g5:first-look-at-cells:cell}{célula} \omterm{def:g4:animal-reproduction:sexual}{femenina}. La \omterm{def:g5:flower-to-fruit:steps}{polinización} va
primero --- el viaje antes del encuentro.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:3}
Se fabrica en un \omterm{def:g3:flowers-fruits-seeds:parts}{estambre}; lo llevan (un insecto o el viento) hasta lo
alto de un \omterm{def:g3:flowers-fruits-seeds:parts}{pistilo} --- \omterm{def:g5:flower-to-fruit:steps}{polinización}; hace crecer un tubo finísimo
hacia abajo por el \omterm{def:g3:flowers-fruits-seeds:parts}{pistilo}; su \omterm{def:g5:first-look-at-cells:cell}{célula} \omterm{def:g4:animal-reproduction:sexual}{masculina} llega a un \omterm{def:g4:animal-reproduction:sexual}{óvulo} y se
une a la \omterm{def:g5:first-look-at-cells:cell}{célula} \omterm{def:g4:animal-reproduction:sexual}{femenina} --- \omterm{def:g5:flower-to-fruit:steps}{fecundación}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:4}
Cada \omterm{def:g4:animal-reproduction:sexual}{óvulo} fecundado se convierte en una \omterm{def:g2:from-seed-to-plant:seed}{semilla}; el \omterm{def:g3:flowers-fruits-seeds:parts}{pistilo} se hincha
hasta ser el \omterm{prop:g3:flowers-fruits-seeds:transform}{fruto}; los \omterm{def:g3:flowers-fruits-seeds:parts}{pétalos} (y los \omterm{def:g3:flowers-fruits-seeds:parts}{estambres}) se marchitan y caen.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:5}
Cada pepita es un \omterm{def:g4:animal-reproduction:sexual}{óvulo} fecundado convertido en \omterm{def:g2:from-seed-to-plant:seed}{semilla}. Las celditas
vacías significan \omterm{def:g4:animal-reproduction:sexual}{óvulos} sin fecundar --- cayó poco \omterm{def:g3:flowers-fruits-seeds:parts}{polen}, y el \omterm{prop:g3:flowers-fruits-seeds:transform}{fruto}
se hinchó de manera desigual alrededor de esa falta.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:6}
Su \omterm{def:g3:flowers-fruits-seeds:parts}{polen} viaja en el viento: fino como el polvo, fabricado en
cantidades enormes, lanzado al aire. El precio es el desperdicio ---
casi ningún grano cae en un sitio útil, así que la cantidad tiene que
sustituir a la precisión.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:7}
Sin visita no hay \omterm{def:g3:flowers-fruits-seeds:parts}{polen} en el \omterm{def:g3:flowers-fruits-seeds:parts}{pistilo} --- la \omterm{def:g5:flower-to-fruit:steps}{polinización}, el primer
paso, no ocurrió nunca, así que no creció ningún tubo y no se fecundó
ningún \omterm{def:g4:animal-reproduction:sexual}{óvulo}; la \omterm{def:g3:flowers-fruits-seeds:parts}{flor} se marchita entera.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:8}
Respuesta abierta. Lo esperable: las \omterm{def:g3:flowers-fruits-seeds:parts}{flores} polinizadas a mano y
marcadas con hilo cuajan \omterm{prop:g3:flowers-fruits-seeds:transform}{fruto} bastante más a menudo --- el pincel
hizo el trabajo de la abeja.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:9}
\omterm{def:g5:first-look-at-cells:cell}{Células} \omterm{def:g4:animal-reproduction:sexual}{masculinas}: en los granos de \omterm{def:g3:flowers-fruits-seeds:parts}{polen} de los \omterm{def:g3:flowers-fruits-seeds:parts}{estambres}. \omterm{def:g5:first-look-at-cells:cell}{Células}
\omterm{def:g4:animal-reproduction:sexual}{femeninas}: en los \omterm{def:g4:animal-reproduction:sexual}{óvulos} de dentro del \omterm{def:g3:flowers-fruits-seeds:parts}{pistilo}. \omterm{def:g5:flower-to-fruit:steps}{Fecundación}: en el
fondo del tubo polínico, dentro del \omterm{def:g4:animal-reproduction:sexual}{óvulo}. Primera \omterm{def:g5:first-look-at-cells:cell}{célula} del
individuo nuevo: la \omterm{def:g5:first-look-at-cells:cell}{célula} \omterm{def:g4:animal-reproduction:sexual}{femenina} fecundada del \omterm{def:g4:animal-reproduction:sexual}{óvulo} --- que se
convierte en la plantita de la \omterm{def:g2:from-seed-to-plant:seed}{semilla}, igual que un \omterm{def:g1:animals-around-us:animal}{animal} sale de su
propia primera \omterm{def:g5:first-look-at-cells:cell}{célula}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:10}
Una cosecha mala --- casi todas las \omterm{def:g3:flowers-fruits-seeds:parts}{flores} cayeron sin \omterm{prop:g3:flowers-fruits-seeds:transform}{fruto}. La
maquinaria estaba bien y las \omterm{def:g3:flowers-fruits-seeds:parts}{flores} no sufrieron daño; lo que falló
fue el paso 1, la \omterm{def:g5:flower-to-fruit:steps}{polinización}: durante dos semanas decisivas los
mensajeros se quedaron en casa, así que casi ningún \omterm{def:g3:flowers-fruits-seeds:parts}{pistilo} recibió
\omterm{def:g3:flowers-fruits-seeds:parts}{polen} y casi ningún \omterm{def:g4:animal-reproduction:sexual}{óvulo} se fecundó.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:11}
Los abejorros venden \omterm{def:g5:flower-to-fruit:steps}{polinización} --- el paso del transporte: llevar
el \omterm{def:g3:flowers-fruits-seeds:parts}{polen} de los \omterm{def:g3:flowers-fruits-seeds:parts}{estambres} a los \omterm{def:g3:flowers-fruits-seeds:parts}{pistilos} dentro de un edificio en el
que no puede entrar ningún polinizador de fuera. El \omterm{def:g3:flowers-fruits-seeds:parts}{polen} del tomate
se suelta con facilidad, así que la vibración de un ventilador y el
aire en movimiento pueden hacer un transporte tosco dentro de cada
\omterm{def:g3:flowers-fruits-seeds:parts}{flor} --- un mensajero mecánico. El \omterm{def:g3:flowers-fruits-seeds:parts}{polen} del cerezo tiene que viajar
de árbol en árbol y de \omterm{def:g3:flowers-fruits-seeds:parts}{flor} en \omterm{def:g3:flowers-fruits-seeds:parts}{flor}, con una precisión que solo da un
cuerpo que visita las \omterm{def:g3:flowers-fruits-seeds:parts}{flores} por turnos; el viento de una nave no
puede apuntar, así que para los cerezos el mensajero vivo no tiene
sustituto barato.
\end{solution}
