\section*{Hoofdstuk \ref{ch:g1:healthy-habits} --- Gezonde gewoonten}

\begin{solution}{exo:g1:healthy-habits:1}
\omterm{def:g1:healthy-habits:germs}{Ziektekiemen} zijn levende wezens die veel te klein zijn om te zien en
die overal zitten; sommige kunnen ons ziek maken als ze in het lichaam
komen. Omdat ze onzichtbaar zijn, kunnen handen er kraakhelder uitzien
en ze toch dragen.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:2}
Voor het eten, na het toilet, en na buiten spelen of het aaien van
\omterm{def:g1:animals-around-us:animal}{dieren}.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:3}
Terwijl je langzaam tot $20$ telt (ongeveer een kort liedje).
Gemakkelijk vergeten: tussen de vingers, rond de duimen en de
handruggen.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:4}
Na het eten blijven er stukjes voedsel op de tanden achter.
\omterm{def:g1:healthy-habits:germs}{Ziektekiemen} in de mond eten van die restjes, vooral van de zoete, en
beetje bij beetje maken ze gaatjes in de tanden, die pijn doen.
Poetsen veegt de restjes weg vóór het smulfeest.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:5}
Na het ontbijt en voor het slapengaan, elke dag --- en het hoort
ongeveer zo lang te duren als een liedje, aan elke kant van elke tand.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:6}
\omterm{prop:g1:healthy-habits:needs}{Gevarieerd eten} (fruit bij het ontbijt, water om te drinken), beweging
(buiten spelen, fietsen) en \omterm{prop:g1:healthy-habits:needs}{slaap} (een lange nacht met vroeg naar bed).
Voorbeelden open.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:7}
Het rust, herstelt zichzelf en groeit. Een kind heeft de langste nacht
nodig --- veel langer dan een volwassene.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:8}
Bijvoorbeeld: gapen, prikkende of zware \omterm{def:g1:the-five-senses:sense}{ogen}, chagrijnig zijn zonder
reden.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:9}
Melktanden moeten nog jaren kauwen voordat ze uitvallen; gaatjes erin
doen net zo veel pijn als in blijvende tanden; en de blijvende tanden
die eronder wachten verdienen een schone, gezonde mond om in aan te
komen. Twee van deze redenen volstaan.
\end{solution}

\begin{solution}{exo:g1:healthy-habits:10}
De glitter staat voor \omterm{def:g1:healthy-habits:germs}{ziektekiemen}. Een hand geven, deuren openen en
potloden delen geeft ze onzichtbaar van hand tot hand door --- en
wassen met zeep gedurende een langzaam telletje tot $20$ verwijdert ze,
glitter en kiemen allebei.
\end{solution}
