\chapter{Apa Itu Respirasi?}\label{ch:g7:what-is-respiration}

Taruhlah seekor kutu kayu di dalam toples tertutup dan, beberapa jam
kemudian, udara di dalam toplesnya sudah berubah: sebagian \omterm{def:g7:what-is-respiration:respiration}{oksigennya}
lenyap, digantikan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}. Ulangi dengan sebutir \omterm{def:g2:from-seed-to-plant:seed}{biji} yang
\omterm{def:g2:from-seed-to-plant:germination}{berkecambah}, sebuah cendawan, seekor \omterm{prop:g3:sorting-living-things:groups}{ikan} mas koki, sesendok \omterm{def:g6:decomposers-and-soil:soil}{tanah} yang
hidup --- hasil yang sama, setiap kali.
\Cref{prop:g4:breathing:exchange} memperlihatkan pertukarannya di
dalam dadamu sendiri; tahun ini membuka gagasannya sampai selebarnya:
pertukaran itu bukanlah kebiasaan manusia melainkan sebuah sifat
kehidupan itu sendiri.

\section{Respirasi, yang didefinisikan dan diukur}

\begin{definition}[Respirasi]\label{def:g7:what-is-respiration:respiration}
Yang disebut \emph{respirasi}\index{respirasi} adalah pertukaran gas
yang membuat sebuah \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} mengambil
\emph{oksigen}\index{oksigen} dari sekelilingnya lalu melepaskan
\emph{karbon dioksida}\index{karbon dioksida}. Ia tidak boleh
dikacaukan dengan \emph{pernapasan}\index{pernapasan} --- gerakan dada
yang terlihat pada \cref{def:g4:breathing:movements}: pernapasan
adalah salah satu cara \emph{melayani} respirasi; respirasi itu
sendiri adalah pertukaran gasnya, dan ia berlangsung di dalam setiap
\omterm{def:g5:first-look-at-cells:cell}{sel} yang hidup, ada gerakan atau tidak.
\end{definition}

\begin{proposition}[Bagaimana pertukarannya diperlihatkan]\label{prop:g7:what-is-respiration:shown}
Dua uji meja baku menyingkapkan \omterm{def:g7:what-is-respiration:respiration}{respirasi}:
\begin{itemize}
  \item sebuah \emph{sensor \omterm{def:g7:what-is-respiration:respiration}{oksigen}} di dalam wadah tertutup
        memperlihatkan taraf \omterm{def:g7:what-is-respiration:respiration}{oksigennya} turun sementara sebuah \omterm{def:g6:exploring-our-environment:components}{makhluk
        hidup} duduk di dalamnya;
  \item \emph{air kapur}\index{air kapur}, cairan bening yang berubah
        keruh seperti susu dalam \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}, menjadi keruh ketika
        udara wadahnya dialirkan menembusnya.
\end{itemize}
\omterm{def:g7:what-is-respiration:respiration}{Oksigen} yang turun ditambah \omterm{prop:g7:what-is-respiration:shown}{air kapur} yang keruh, dengan sebuah wadah
kontrol tanpa nyawa yang tetap tidak berubah
(\cref{met:g4:what-plants-need:fairtest} --- uji yang adil mengikuti
kita ke mana-mana): itulah \omterm{def:g7:what-is-respiration:respiration}{respirasi}, yang terukur.
\end{proposition}
\admitted

\begin{example}[Galeri para pelaku respirasi]\label{ex:g7:what-is-respiration:gallery}
Jalankan kedua ujinya atas sekumpulan makhluk: seekor tikus ---
penurunan yang cepat, kekeruhan yang segera; seekor kutu kayu ---
serupa, lebih kecil; sebutir ercis yang \omterm{def:g2:from-seed-to-plant:germination}{berkecambah} --- tak
terbantahkan (satu toples ercis yang bertunas dapat menghabiskan
\omterm{def:g7:what-is-respiration:respiration}{oksigennya} dalam sehari); sebuah cendawan --- ya; siput kolam --- ya;
sesendok \omterm{def:g6:decomposers-and-soil:soil}{tanah} yang hidup --- ya, dengan miliaran \omterm{def:g6:decomposers-and-soil:decomposers}{pengurainya} yang
berespirasi bersama-sama
(\cref{ex:g6:decomposers-and-soil:handful}). Hanya toples kontrolnya
--- kerikil, pasir kering --- yang tidak mengubah apa pun.
\end{example}

\begin{remark}[Tumbuhan pun berespirasi]\label{rem:g7:what-is-respiration:plants}
\Cref{rem:g6:plants-are-producers:gas} mengatakannya sambil lalu;
toplesnya membuktikannya: sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan} dalam pot \emph{di dalam
gelap} menurunkan \omterm{def:g7:what-is-respiration:respiration}{oksigen} toplesnya lalu mengeruhkan \omterm{prop:g7:what-is-respiration:shown}{air kapurnya}
seperti \omterm{def:g1:animals-around-us:animal}{hewan} mana pun. Pada siang hari, pembuatan oleh \omterm{def:g5:food-webs:roles}{produsennya}
menutupi akibatnya --- karena menyerap \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} jauh lebih
banyak daripada yang dilepaskan \omterm{def:g7:what-is-respiration:respiration}{respirasinya} --- tetapi \omterm{def:g7:what-is-respiration:respiration}{respirasinya}
tidak pernah berhenti. Warna hijau tidak membebaskan sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} dari
bernapas-dalam-arti-luas; tidak ada yang hidup yang dibebaskan.
\end{remark}

\section{Mengapa makhluk hidup berespirasi}

\begin{proposition}[Respirasi menenagai selnya]\label{prop:g7:what-is-respiration:power}
Inti pertukarannya adalah \emph{\omterm{prop:g3:balanced-diet:jobs}{energi}}. Di dalam setiap \omterm{def:g5:first-look-at-cells:cell}{sel} yang
hidup, \omterm{def:g4:journey-of-food:digestion}{zat gizi} (\cref{def:g4:journey-of-food:digestion}) perlahan
\emph{dibakar} dengan \omterm{def:g7:what-is-respiration:respiration}{oksigen} yang dibawa \omterm{def:g7:what-is-respiration:respiration}{respirasinya}: \omterm{prop:g3:balanced-diet:jobs}{energinya}
dilepaskan bagi pekerjaan \omterm{def:g5:first-look-at-cells:cell}{selnya} --- kontraksi, pembangunan,
pemerintahan --- dan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksidanya} adalah sisa yang sudah
terbakar habis, yang diangkut pergi. ``Bagian yang terbakar'' pada
\cref{prop:g6:matter-from-food:losses} persis adalah hal ini, yang
dilihat dari pembukuannya; di sini ia dilihat dari \omterm{def:g5:first-look-at-cells:cell}{selnya}.
\end{proposition}
\admitted

\begin{example}[Perbandingan dengan api, yang dipakai dengan jujur]\label{ex:g7:what-is-respiration:fire}
Nyala sebatang lilin juga melahap \omterm{def:g7:what-is-respiration:respiration}{oksigen} lalu melepaskan \omterm{def:g7:what-is-respiration:respiration}{karbon
dioksida} --- pembakaran dan \omterm{def:g7:what-is-respiration:respiration}{respirasi} sungguh-sungguh merupakan jenis
kimia yang sama. Perbedaannya sama pentingnya: pembakaran oleh \omterm{def:g5:first-look-at-cells:cell}{selnya}
berlangsung lambat, tanpa nyala, terbagi menjadi ribuan langkah kecil,
dan \omterm{prop:g3:balanced-diet:jobs}{energinya} ditangkap untuk pekerjaan alih-alih hilang sebagai
\omterm{def:g6:exploring-our-environment:conditions}{cahaya} dan hangus. \omterm{def:g7:what-is-respiration:respiration}{Respirasi} adalah pembakaran yang dijinakkan --- api
yang bertali kekang, di dalam setiap \omterm{def:g5:first-look-at-cells:cell}{sel} yang kamu miliki.
\end{example}

\begin{example}[Membaca permintaannya]\label{ex:g7:what-is-respiration:demand}
Karena pembakarannya menenagai pekerjaannya, permintaannya mengikuti
usahanya: anak yang berlari cepat pada
\cref{ex:g4:breathing:running} berespirasi jauh lebih cepat daripada
pembaca di kursi malas; ercis yang \omterm{def:g2:from-seed-to-plant:germination}{berkecambah} --- yang sedang
membangun sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan} dengan kecepatan penuh
(\cref{def:g2:from-seed-to-plant:germination}) --- berespirasi jauh
lebih cepat daripada \omterm{def:g2:from-seed-to-plant:seed}{biji} kering yang menunggu di sebelahnya, yang
\omterm{def:g7:what-is-respiration:respiration}{respirasinya} menganggur mendekati nol selama bertahun-tahun. \omterm{def:g7:what-is-respiration:respiration}{Respirasi}
adalah bunyi mesin kehidupan: keras ketika berusaha dan tumbuh, sebuah
bisikan ketika dorman, sunyi hanya ketika mati.
\end{example}

\section{Respirasi di setiap habitat}

\begin{proposition}[Satu kebutuhan, banyak sekeliling]\label{prop:g7:what-is-respiration:habitats}
Setiap \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} harus memperoleh \omterm{def:g7:what-is-respiration:respiration}{oksigennya} \emph{dari
sekelilingnya} --- dan sekelilingnya berbeda-beda. Udara kaya \omterm{def:g7:what-is-respiration:respiration}{oksigen};
air memuat jauh lebih sedikit, dalam keadaan terlarut
(\cref{exo:g4:breathing:11} sudah berjumpa dengan akibatnya); \omterm{def:g6:decomposers-and-soil:soil}{tanah}
dan lumpur memuat lebih sedikit lagi. Karena itu tempat dan cara
sebuah \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} hidup dibentuk oleh cara ia memecahkan masalah
\omterm{def:g7:what-is-respiration:respiration}{oksigennya} --- pokok kedua bab berikutnya, tentang perlengkapan \omterm{def:g1:animals-around-us:animal}{hewan}
(\cref{ch:g7:how-animals-breathe}) dan tentang hemat air
(\cref{ch:g7:oxygen-in-the-water}).
\end{proposition}
\admitted

\begin{method}[Menguji sebuah pertanyaan tentang respirasi]\label{met:g7:what-is-respiration:test}
Untuk menguji pengakuan apa pun tentang \omterm{def:g7:what-is-respiration:respiration}{respirasi} (``\omterm{def:g2:from-seed-to-plant:seed}{biji} berespirasi
lebih cepat kalau hangat'', ``lumpur kolamnya berespirasi''):
\begin{enumerate}
  \item tutuplah subjeknya di dalam sebuah wadah, dengan sebuah sensor
        \omterm{def:g7:what-is-respiration:respiration}{oksigen} atau sebuah perangkap \omterm{prop:g7:what-is-respiration:shown}{air kapur};
  \item bangunlah kontrol kembarnya --- wadah yang sama, dengan
        subjeknya tidak ada atau tanpa nyawa;
  \item ubahlah satu \omterm{def:g6:exploring-our-environment:conditions}{kondisi} dalam satu waktu, kalau pengakuannya
        menyebutkan satu \omterm{def:g6:exploring-our-environment:conditions}{kondisi} (kehangatan, \omterm{def:g6:exploring-our-environment:conditions}{kelembapan}, gelap);
  \item bacalah putusannya pada perbedaannya: \omterm{def:g7:what-is-respiration:respiration}{oksigen} yang turun, \omterm{prop:g7:what-is-respiration:shown}{air
        kapur} yang keruh --- melawan kediaman kontrolnya.
\end{enumerate}
\end{method}

\begin{example}[Metodenya atas sebutir biji yang dingin]\label{ex:g7:what-is-respiration:coldseed}
Pengakuannya: ercis yang \omterm{def:g2:from-seed-to-plant:germination}{berkecambah} berespirasi lebih lambat dalam
keadaan dingin. Dua toples ercis yang bertunas, satu pada kehangatan
ruangan, satu di dalam lemari es; perangkap \omterm{prop:g7:what-is-respiration:shown}{air kapur} pada keduanya;
sebuah toples ketiga tanpa nyawa sebagai kontrolnya. Keesokan harinya:
\omterm{prop:g7:what-is-respiration:shown}{air kapur} toples yang hangat keruh seperti susu, \omterm{prop:g7:what-is-respiration:shown}{air kapur} toples yang
dingin nyaris hanya berembun, \omterm{prop:g7:what-is-respiration:shown}{air kapur} kontrolnya bening. Putusannya:
\omterm{def:g7:what-is-respiration:respiration}{respirasi} berjalan seiring laju kerja ercisnya ---
aturan \omterm{def:g6:exploring-our-environment:conditions}{kondisi} pada \cref{prop:g6:decomposers-and-soil:speed}, yang
diukur dalam gas.
\end{example}

\section{Soal: Toples-Toples Tertutup}

\begin{problem}[{Soal akhir pekan --- enam toples, dua misteri, satu
pertukaran gas}]\label{pb:g7:what-is-respiration:1}
Untuk pasar ilmunya, kelasnya menyiapkan enam toples tertutup,
masing-masing dengan sebuah sensor \omterm{def:g7:what-is-respiration:respiration}{oksigen} dan sebuah perangkap \omterm{prop:g7:what-is-respiration:shown}{air
kapur}: A --- kerikil; B --- ercis kering; C --- ercis yang
\omterm{def:g2:from-seed-to-plant:germination}{berkecambah}; D --- seekor tikus (dengan makanan, air, dan hanya selama
beberapa jam, di bawah pengawasan gurunya); E --- sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan}
hijau di dalam \omterm{def:g6:exploring-our-environment:conditions}{cahaya}; F --- \omterm{def:g1:plants-around-us:plant}{tumbuhan} sejenis di dalam kotak yang
gelap.

\medskip\noindent\textbf{Bagian I --- Ramalannya.}
\begin{enumerate}
  \item Ramalkan bacaan A, lalu nyatakan tugas A dalam percobaannya.
  \item Ramalkan B dan C, lalu jelaskan perbedaannya walaupun kedua
        toplesnya memuat ercis ``yang sama''.
  \item Ramalkan D, lalu sebutkan mengapa gurunya membatasi jamnya.
  \item E dan F memuat \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang serupa. Ramalkan kecenderungan
        \omterm{def:g7:what-is-respiration:respiration}{oksigen} tiap toplesnya, lalu damaikan perbedaannya dengan
        memakai \cref{rem:g7:what-is-respiration:plants}.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Hasil dan bacaannya.}
\begin{enumerate}[resume]
  \item Data paginya cocok dengan ramalannya --- kecuali seorang teman
        sekelas takjub bahwa \omterm{def:g1:plants-around-us:plant}{tumbuhan} F ``bernapas seperti \omterm{def:g1:animals-around-us:animal}{hewan}''.
        Paparkan apa yang sesungguhnya diperlihatkan F, dan apa yang
        disembunyikan E.
  \item \omterm{def:g7:what-is-respiration:respiration}{Oksigen} C turun tiga kali lebih cepat daripada \omterm{def:g7:what-is-respiration:respiration}{oksigen} B per
        gram ercisnya. Terjemahkan faktor tiga itu ke dalam istilah
        \cref{ex:g7:what-is-respiration:demand}.
  \item \omterm{prop:g7:what-is-respiration:shown}{Air kapur} D menjadi paling cepat keruh di antara semuanya.
        Berikan rantainya --- dari tubuh tikus yang hangat dan bergerak
        sampai ke cairan yang keruh seperti susu --- dalam empat
        langkah.
  \item Sebutkan satu kesimpulan tunggal yang didukung B, C, D dan F
        sekaligus, dalam satu kalimat yang bermula dengan ``Setiap
        \omterm{def:g6:exploring-our-environment:components}{makhluk hidup}\dots''.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Mendesak lebih jauh.}
\begin{enumerate}[resume]
  \item Diusulkan toples ketujuh: irisan cendawan. Ramalkan bacaannya,
        lalu sebutkan butir setingkat kerajaan pada
        \cref{ex:g7:what-is-respiration:gallery} yang akan
        ditambahkannya.
  \item Pengunjung pasarnya bertanya mengapa toples C berarti bagi
        petani. Jawablah dengan lumbung pada
        \cref{exo:g7:what-is-respiration:11}.
  \item Toples kedelapan: air kolam beserta lumpurnya. \omterm{def:g7:what-is-respiration:respiration}{Oksigennya} turun
        dalam semalam. Siapa di dalam toplesnya yang berespirasi, dan
        mengapa temuan itu berarti bagi pertanyaan
        \cref{ch:g7:oxygen-in-the-water}?
  \item Tutuplah poster pasarnya dengan pembedaan bab ini: toples mana
        yang memperlihatkan \emph{\omterm{def:g7:what-is-respiration:respiration}{pernapasan}}, dan mana yang
        memperlihatkan \emph{\omterm{def:g7:what-is-respiration:respiration}{respirasi}}? (Hati-hati --- satu kata
        meliputi semuanya.)
\end{enumerate}
\end{problem}
