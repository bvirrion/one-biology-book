\chapter{Human Genetics}\label{ch:g11:human-genetics}

Human traits follow the same Mendelian logic as pea colour, but controlled
crosses are impossible, \omterm{def:g10:biodiversity-classification:hierarchy}{family} sizes are small, and many traits are not
\omterm{def:g11:gene-expression:gene}{single-gene}. This chapter reads \omterm{def:g11:human-genetics:pedigree}{pedigrees}, treats sex linkage, sketches
chromosomal aneuploidies, and introduces polygenic inheritance.

\section{Pedigrees}

\begin{definition}[Pedigree]\label{def:g11:human-genetics:pedigree}
A \emph{pedigree}\index{pedigree} is a \omterm{def:g10:biodiversity-classification:hierarchy}{family} tree that records the presence or
absence of a trait. Standard symbols: squares for males, circles for females;
filled symbols for affected individuals; horizontal lines for partnerships;
vertical lines to children.
\end{definition}

\begin{method}[Reading a pedigree for a rare trait]\label{met:g11:human-genetics:read}
\begin{enumerate}
  \item If two unaffected parents have an affected child, the trait is often
        \omterm{def:g11:mendelian-genetics:dominance}{recessive} (child $aa$, parents $Aa$).
  \item If the trait appears every generation and affected $\times$ unaffected
        often yields affected children, consider \omterm{def:g11:mendelian-genetics:dominance}{dominant} inheritance.
  \item Check sex bias and father-to-son transmission for \omterm{def:g11:human-genetics:xlinked}{X-linked} patterns
        (below).
  \item Rare traits: mates from outside the \omterm{def:g10:biodiversity-classification:hierarchy}{family} are usually \omterm{def:g11:mendelian-genetics:zygous}{homozygous}
        normal unless consanguinity is noted.
\end{enumerate}
\end{method}

\begin{example}[Autosomal recessive sketch]\label{ex:g11:human-genetics:ar}
Cystic fibrosis is autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive}. Unaffected carrier parents $Aa \times Aa$
have a $\tfrac14$ chance of an affected $aa$ child each pregnancy --- matching
many \omterm{def:g11:human-genetics:pedigree}{pedigrees} where the trait ``skips'' generations.
\end{example}

\section{Sex chromosomes and sex linkage}

\begin{definition}[Sex chromosomes]\label{def:g11:human-genetics:sexchrom}
In humans, females are typically $XX$ and males $XY$. The \emph{X
chromosome}\index{X chromosome} carries many \omterm{def:g11:gene-expression:gene}{genes} unrelated to sex; the
\emph{Y chromosome}\index{Y chromosome} is small and includes the
\emph{SRY}\index{SRY} \omterm{def:g11:gene-expression:gene}{gene} that triggers male \omterm{def:g10:scientific-biology:properties}{development}.
\end{definition}

\begin{definition}[X-linked inheritance]\label{def:g11:human-genetics:xlinked}
A \omterm{def:g11:gene-expression:gene}{gene} on the \omterm{def:g11:human-genetics:sexchrom}{X chromosome} is \emph{X-linked}\index{X-linked}. Males are
\emph{hemizygous}\index{hemizygous}: they have one X, so a single \omterm{def:g11:mendelian-genetics:dominance}{recessive}
disease \omterm{def:g11:mendelian-genetics:allele}{allele} on that X is expressed. Females need two copies of a \omterm{def:g11:mendelian-genetics:dominance}{recessive}
\omterm{prop:g11:mendelian-genetics:assortment}{X-linked} \omterm{def:g11:mendelian-genetics:allele}{allele} to express the trait (or one \omterm{def:g11:mendelian-genetics:dominance}{dominant} disease \omterm{def:g11:mendelian-genetics:allele}{allele}).
\end{definition}

\begin{proposition}[Classic X-linked recessive pattern]\label{prop:g11:human-genetics:xr}
For a rare \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive} trait:
\begin{itemize}
  \item more males than females are affected;
  \item affected males are often born to carrier mothers;
  \item there is no father-to-son transmission of the \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:allele}{allele} (a son
        gets his X from his mother);
  \item all daughters of an affected male are at least carriers.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Red--green colour vision]\label{ex:g11:human-genetics:colour}
Certain red--green colour vision deficiencies are \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive}. A carrier
mother $X^{A}X^{a}$ and a normal father $X^{A}Y$ expect half their sons to be
colour-deficient and half their daughters to be carriers --- on average.
\end{example}

\section{Chromosomal aneuploidy}

\begin{definition}[Aneuploidy]\label{def:g11:human-genetics:aneuploidy}
\emph{Aneuploidy}\index{aneuploidy} is an abnormal number of \omterm{def:g11:dna-replication:chromosome}{chromosomes} --- not
an exact multiple of the \omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} set. A common cause is
\emph{nondisjunction}\index{nondisjunction}: failure of \omterm{def:g11:dna-replication:chromosome}{chromosomes} or chromatids
to separate properly in \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis} (or \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}).
\end{definition}

\begin{proposition}[Human examples]\label{prop:g11:human-genetics:trisomy}
\begin{itemize}
  \item \emph{Trisomy 21}\index{trisomy 21} (Down syndrome): three copies of
        \omterm{def:g11:dna-replication:chromosome}{chromosome} 21.
  \item \emph{XXY}\index{Klinefelter syndrome} (\omterm{prop:g11:human-genetics:trisomy}{Klinefelter syndrome}),
        \emph{X0}\index{Turner syndrome} (\omterm{prop:g11:human-genetics:trisomy}{Turner syndrome}): \omterm{def:g11:dna-replication:chromosome}{sex-chromosome}
        aneuploidies compatible with life, with characteristic \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes}.
\end{itemize}
Most other full autosomal aneuploidies are embryonic lethal. Risk of some
\omterm{def:g11:human-genetics:aneuploidy}{nondisjunction} events rises with maternal age.
\end{proposition}
\admitted

\begin{remark}
\omterm{def:g11:human-genetics:aneuploidy}{Aneuploidy} is a change in \omterm{def:g11:dna-replication:chromosome}{chromosome} \emph{number}, not a small mutation within
a \omterm{def:g11:gene-expression:gene}{gene}. Karyotyping and modern genomic tests detect such changes.
\end{remark}

\section{Beyond single genes}

\begin{definition}[Polygenic trait]\label{def:g11:human-genetics:polygenic}
A \emph{polygenic trait}\index{polygenic trait} is influenced by many \omterm{def:g11:gene-expression:gene}{genes},
often plus environment. Such traits are frequently
\emph{quantitative}\index{quantitative trait} (height, skin pigmentation,
blood pressure): they vary continuously rather than in Mendel's discrete classes.
\end{definition}

\begin{proposition}[Many loci approximate a continuum]\label{prop:g11:human-genetics:continuum}
If several unlinked loci add small effects, the number of genotypic classes
grows and the histogram of \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes} approaches a smooth bell-shaped
distribution in the \omterm{def:g10:ecosystems:levels}{population} --- especially when environment adds further
noise.
\end{proposition}
\admitted

\begin{example}[Height]\label{ex:g11:human-genetics:height}
Human height is highly polygenic and also depends on nutrition and health.
Relatives resemble each other more than unrelated people, yet siblings are not
identical --- recombination and environment both matter.
\end{example}

\begin{definition}[Multifactorial]\label{def:g11:human-genetics:multifactorial}
A trait is \emph{multifactorial}\index{multifactorial} when both genetic
variants and environmental factors contribute substantially. Many common adult
diseases (type 2 diabetes, coronary disease) are multifactorial.
\end{definition}

\begin{remark}[Mitochondrial inheritance]
\label{rem:g11:human-genetics:mtDNA}
\omterm{def:g10:the-cell:energy-organelles}{Mitochondria} carry their own small circular \omterm{def:g11:dna-replication:helix}{DNA} (\emph{mtDNA}\index{mtDNA}).
In humans, mitochondria (and thus \omterm{rem:g11:human-genetics:mtDNA}{mtDNA}) are transmitted almost exclusively
through the egg, so mitochondrial traits show
\emph{maternal inheritance}\index{maternal inheritance}: children of an
affected mother may inherit the variant; children of an affected father
typically do not, because sperm contribute negligible cytoplasm to the zygote.
\omterm{rem:g11:human-genetics:mtDNA}{mtDNA} diseases can vary in severity depending on the fraction of mutant
mitochondria in a tissue (\emph{heteroplasmy}\index{heteroplasmy}).
\end{remark}

\begin{method}[Genetic counselling outline]
\label{met:g11:human-genetics:counselling}
When a \omterm{def:g10:biodiversity-classification:hierarchy}{family} seeks information about an inherited condition:
\begin{enumerate}
  \item Take a detailed \omterm{def:g11:human-genetics:pedigree}{pedigree} (who is affected, ages, sexes, consanguinity).
  \item Propose inheritance models consistent with the \omterm{def:g11:human-genetics:pedigree}{pedigree} (autosomal
        \omterm{def:g11:mendelian-genetics:dominance}{recessive}/\omterm{def:g11:mendelian-genetics:dominance}{dominant}, \omterm{def:g11:human-genetics:xlinked}{X-linked}, chromosomal, \omterm{def:g11:human-genetics:multifactorial}{multifactorial}).
  \item Where possible, estimate recurrence risks for future children (e.g.\
        $\tfrac14$ for autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive} when both parents are carriers).
  \item Discuss test options (carrier tests, prenatal tests, karyotype) and
        their limits --- not every risk can be reduced to a single number.
  \item Communicate uncertainty, privacy and non-directive support: \omterm{def:g10:scientific-biology:science}{science}
        informs choices; it does not dictate them.
\end{enumerate}
\end{method}

\begin{example}[Counselling a recessive disease]
\label{ex:g11:human-genetics:counsel}
Unaffected parents have a child with an autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive} disease. Under
standard assumptions both are carriers; each future pregnancy has a
$\tfrac14$ chance of an affected child, independent of previous outcomes.
Counselling states that risk, explains testing if available, and avoids
blaming either parent --- both transmitted one disease \omterm{def:g11:mendelian-genetics:allele}{allele} by chance.
\end{example}

\begin{remark}[Carrier screening and ethics]
Genetic information can guide medical care and reproductive decisions. It also
raises questions of privacy, discrimination, and how uncertainty is communicated
--- scientific literacy is only the first part of responsible use.
\end{remark}

\section{Exercises}

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:human-genetics:4}
A woman whose father was colour-deficient (\omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive}) marries a man

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:human-genetics:5}
Why is father-to-son transmission of an \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:allele}{allele} impossible?

\begin{exercise}[$\star\star$]\label{exo:g11:human-genetics:6}

\begin{exercise}[$\star\star$]\label{exo:g11:human-genetics:7}
Two unaffected parents have a child with an autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive} disease. What

\begin{exercise}[$\star\star$]\label{exo:g11:human-genetics:8}

\begin{exercise}[$\star\star\star$]\label{exo:g11:human-genetics:9}

\begin{exercise}[$\star\star\star$]\label{exo:g11:human-genetics:10}

\begin{exercise}[$\star\star$]\label{exo:g11:human-genetics:11}

\begin{exercise}[$\star\star\star$]\label{exo:g11:human-genetics:12}

\end{exercise}

\begin{problem}[A family with a rare disease]\label{pb:g11:human-genetics:1}
A rare disease appears in a \omterm{def:g10:biodiversity-classification:hierarchy}{family}. Consider two genetic models: autosomal
\omterm{def:g11:mendelian-genetics:dominance}{recessive} ($d$) and \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive} ($d$ on X).
\begin{enumerate}
  \item Unaffected parents have an affected daughter and an unaffected son.
        Which model is more readily compatible, and why?
  \item Under the autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive} model, give the parents' \omterm{def:g11:mendelian-genetics:geno-pheno}{genotypes} and the
        probability that a future son is affected.
  \item An affected man and an unaffected non-carrier woman have children.
        Contrast the expected children under autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive} disease versus
        \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive} disease.
  \item Draw a two-generation \omterm{def:g11:human-genetics:pedigree}{pedigree} (symbols) for an \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive} trait
        with a carrier mother, normal father, one affected son and one carrier
        daughter.
  \item How could a karyotype distinguish \omterm{prop:g11:human-genetics:trisomy}{trisomy 21} from a \omterm{def:g11:gene-expression:gene}{single-gene} disorder
        causing intellectual disability?
  \item Height is polygenic. Why might two short parents still have a child of
        average height?
\end{enumerate}
\end{problem}
