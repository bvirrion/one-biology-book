\chapter{Gene Expression}\label{ch:g11:gene-expression}

\omterm{def:g11:dna-replication:helix}{DNA} stores information; \omterm{def:g10:molecules-of-life:proteins}{proteins} and functional \omterm{def:g10:molecules-of-life:nucleic}{RNAs} do most of the \omterm{def:g10:the-cell:cell}{cell}'s work.
\emph{\omterm{def:g11:gene-expression:gene}{Gene} expression} is the pipeline that turns a stretch of \omterm{def:g11:dna-replication:helix}{DNA} into a product
--- usually a \omterm{def:g10:molecules-of-life:proteins}{protein} --- by two main stages: \omterm{def:g11:gene-expression:transcription}{transcription} (\omterm{def:g11:dna-replication:helix}{DNA}$\to$\omterm{def:g10:molecules-of-life:nucleic}{RNA}) and
\omterm{def:g11:gene-expression:translation}{translation} (\omterm{def:g10:molecules-of-life:nucleic}{RNA}$\to$polypeptide). This chapter states the \omterm{def:g11:gene-expression:dogma}{central dogma}, the
\omterm{def:g11:gene-expression:codon}{genetic code}, and the core machinery of both stages.

\section{Genes and the central dogma}

\begin{definition}[Gene]\label{def:g11:gene-expression:gene}
A \emph{gene}\index{gene} is a segment of \omterm{def:g11:dna-replication:helix}{DNA} that is transcribed into \omterm{def:g10:molecules-of-life:nucleic}{RNA} and
that functions as a unit of hereditary information. Many genes encode \omterm{def:g10:molecules-of-life:proteins}{proteins};
others encode \omterm{def:g10:molecules-of-life:nucleic}{RNA} products that never become \omterm{def:g10:molecules-of-life:proteins}{protein} (for example ribosomal \omterm{def:g10:molecules-of-life:nucleic}{RNA}
and \omterm{def:g11:gene-expression:translation}{transfer RNA}).
\end{definition}

\begin{definition}[Central dogma]\label{def:g11:gene-expression:dogma}
The \emph{central dogma}\index{central dogma} of molecular \omterm{def:g10:scientific-biology:science}{biology} states that
genetic information normally flows from \omterm{def:g11:dna-replication:helix}{DNA} to \omterm{def:g10:molecules-of-life:nucleic}{RNA} to \omterm{def:g10:molecules-of-life:proteins}{protein}. \omterm{def:g11:dna-replication:helix}{DNA} is
replicated to \omterm{def:g11:dna-replication:helix}{DNA}; \omterm{def:g11:dna-replication:helix}{DNA} is transcribed to \omterm{def:g10:molecules-of-life:nucleic}{RNA}; \omterm{prop:g11:gene-expression:tx-stages}{messenger RNA} is translated to a
polypeptide. Information does not flow back from \omterm{def:g10:molecules-of-life:proteins}{protein} to \omterm{def:g10:molecules-of-life:nucleic}{nucleic acid}.
\end{definition}

\begin{remark}
Some viruses reverse-transcribe \omterm{def:g10:molecules-of-life:nucleic}{RNA} into \omterm{def:g11:dna-replication:helix}{DNA} (an exception at the \omterm{def:g11:dna-replication:helix}{DNA}$\leftrightarrow$\omterm{def:g10:molecules-of-life:nucleic}{RNA}
step), but no known pathway writes \omterm{def:g10:molecules-of-life:proteins}{protein} \omterm{def:g10:molecules-of-life:proteins}{sequence} back into \omterm{def:g11:gene-expression:gene}{genes}. The dogma
is a statement about \emph{information flow}, not a claim that every \omterm{def:g11:gene-expression:gene}{gene} is
always on.
\end{remark}

\section{Transcription: DNA to RNA}

\begin{definition}[Transcription]\label{def:g11:gene-expression:transcription}
\emph{Transcription}\index{transcription} is the synthesis of an \omterm{def:g10:molecules-of-life:nucleic}{RNA} strand
\omterm{prop:g11:dna-replication:base-pairs}{complementary} to a \omterm{def:g11:dna-replication:helix}{DNA} \omterm{def:g11:gene-expression:strands}{template strand}. The \omterm{def:g10:membranes-enzymes:enzyme}{enzyme}
\emph{RNA polymerase}\index{RNA polymerase} reads the template $3'\to 5'$ and
builds \omterm{def:g10:molecules-of-life:nucleic}{RNA} $5'\to 3'$. In \omterm{def:g10:molecules-of-life:nucleic}{RNA}, uracil (U) pairs with adenine instead of thymine.
\end{definition}

\begin{definition}[Template and coding strands]\label{def:g11:gene-expression:strands}
For a given \omterm{def:g11:gene-expression:gene}{gene}, the \emph{template strand}\index{template strand} (antisense)
is the \omterm{def:g11:dna-replication:helix}{DNA} strand used by \omterm{def:g11:gene-expression:transcription}{RNA polymerase}. The other \omterm{def:g11:dna-replication:helix}{DNA} strand is the
\emph{coding strand}\index{coding strand} (sense): its \omterm{def:g10:molecules-of-life:proteins}{sequence} matches the \omterm{def:g10:molecules-of-life:nucleic}{RNA}
(with T in \omterm{def:g11:dna-replication:helix}{DNA} corresponding to U in \omterm{def:g10:molecules-of-life:nucleic}{RNA}), written in the same $5'\to 3'$
direction as the transcript.
\end{definition}

\begin{example}[A short transcript]\label{ex:g11:gene-expression:transcript}
Template \omterm{def:g11:dna-replication:helix}{DNA}: \texttt{3'-TAC GGA-5'}. \omterm{def:g10:molecules-of-life:nucleic}{RNA} transcript: \texttt{5'-AUG CCU-3'}.
Coding \omterm{def:g11:dna-replication:helix}{DNA} would read \texttt{5'-ATG CCT-3'} --- same as the \omterm{def:g10:molecules-of-life:nucleic}{RNA} with T for U.
\end{example}

\begin{proposition}[Stages of transcription]\label{prop:g11:gene-expression:tx-stages}
\omterm{def:g11:gene-expression:transcription}{Transcription} has three phases:
\begin{itemize}
  \item \emph{Initiation}\index{initiation (transcription)}: \omterm{def:g11:gene-expression:transcription}{RNA polymerase}
        binds a \emph{promoter}\index{promoter} region and opens the duplex.
  \item \emph{Elongation}\index{elongation (transcription)}: the polymerase
        moves along the \omterm{def:g11:gene-expression:gene}{gene}, adding \omterm{prop:g11:dna-replication:base-pairs}{complementary} ribonucleotides.
  \item \emph{Termination}\index{termination (transcription)}: signals release
        the \omterm{def:g10:molecules-of-life:nucleic}{RNA} and the polymerase from the \omterm{def:g11:dna-replication:helix}{DNA}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Eukaryotic mRNA processing]
In \omterm{def:g10:the-cell:prok-euk}{eukaryotes}, the primary transcript of a protein-coding \omterm{def:g11:gene-expression:gene}{gene} is processed:
a $5'$ cap and a $3'$ poly-A tail are added, and non-coding
\emph{introns}\index{intron} are removed by \emph{splicing}\index{splicing} so
that \emph{exons}\index{exon} form mature \emph{\omterm{prop:g11:gene-expression:tx-stages}{messenger RNA}}
(mRNA)\index{messenger RNA}\index{mRNA}. Prokaryotic mRNA is typically used
without such \omterm{prop:g11:gene-expression:tx-stages}{splicing}.
\end{remark}

\section{The genetic code}

\begin{definition}[Codon]\label{def:g11:gene-expression:codon}
A \emph{codon}\index{codon} is a triplet of \omterm{def:g10:molecules-of-life:nucleic}{nucleotides} in mRNA that specifies
one \omterm{def:g10:molecules-of-life:proteins}{amino acid} (or a stop signal) during \omterm{def:g11:gene-expression:translation}{translation}. The set of all codon
assignments is the \emph{genetic code}\index{genetic code}.
\end{definition}

\begin{proposition}[Properties of the code]\label{prop:g11:gene-expression:code-props}
The standard \omterm{def:g11:gene-expression:codon}{genetic code} is:
\begin{itemize}
  \item \emph{triplet}: three bases per \omterm{def:g10:molecules-of-life:proteins}{amino acid};
  \item \emph{degenerate} (redundant): most \omterm{def:g10:molecules-of-life:proteins}{amino acids} have more than one \omterm{def:g11:gene-expression:codon}{codon};
  \item \emph{nearly universal}: the same \omterm{def:g11:gene-expression:codon}{codon} table is used by nearly all
        \omterm{def:g10:scientific-biology:organism}{organisms} (with minor exceptions in some \omterm{def:g10:the-cell:prok-euk}{organelles} and microbes);
  \item \emph{comma-less and non-overlapping}: \omterm{def:g11:gene-expression:codon}{codons} are read in \omterm{def:g10:biodiversity-classification:hierarchy}{order} from a
        start point, without skipping bases between \omterm{def:g11:gene-expression:codon}{codons} in the standard
        reading frame.
\end{itemize}
The \omterm{def:g11:gene-expression:codon}{codon} \texttt{AUG} both codes for methionine and usually serves as the
\emph{start codon}\index{start codon}. Three \omterm{def:g11:gene-expression:codon}{codons} (\texttt{UAA}, \texttt{UAG},
\texttt{UGA}) are \emph{stop codons}\index{stop codon} and do not code for \omterm{def:g10:molecules-of-life:proteins}{amino
acids}.
\end{proposition}
\admitted

\begin{example}[Reading frame]\label{ex:g11:gene-expression:frame}
mRNA: \texttt{5'-AUG-UUU-GGC-UAA-3'} translates (standard code) as
Met--Phe--Gly, then stop. Shifting the start by one base would scramble every
\omterm{def:g11:gene-expression:codon}{codon} --- a \emph{frameshift} if caused by insertion or deletion in the \omterm{def:g11:gene-expression:gene}{gene}.
\end{example}

\section{Translation: RNA to protein}

\begin{definition}[Translation]\label{def:g11:gene-expression:translation}
\emph{Translation}\index{translation} is the synthesis of a polypeptide on a
\emph{ribosome}\index{ribosome}, directed by mRNA. \omterm{def:g10:molecules-of-life:proteins}{Amino acids} are delivered by
\emph{transfer RNA}\index{transfer RNA}\index{tRNA} (tRNA) molecules.
\end{definition}

\begin{definition}[tRNA and anticodon]\label{def:g11:gene-expression:trna}
Each \omterm{def:g11:gene-expression:translation}{tRNA} carries one \omterm{def:g10:molecules-of-life:proteins}{amino acid} and exposes a three-base
\emph{anticodon}\index{anticodon} that base-pairs with the \omterm{prop:g11:dna-replication:base-pairs}{complementary} mRNA
\omterm{def:g11:gene-expression:codon}{codon}. The \omterm{def:g10:membranes-enzymes:enzyme}{enzyme} that attaches the correct \omterm{def:g10:molecules-of-life:proteins}{amino acid} to each \omterm{def:g11:gene-expression:translation}{tRNA} enforces
the \omterm{def:g11:gene-expression:codon}{genetic code} chemically.
\end{definition}

\begin{proposition}[Ribosome cycle]\label{prop:g11:gene-expression:ribo}
\omterm{def:g11:gene-expression:translation}{Translation} proceeds as:
\begin{itemize}
  \item \emph{\omterm{prop:g11:gene-expression:tx-stages}{Initiation}}: the \omterm{def:g11:gene-expression:translation}{ribosome} assembles at the \omterm{prop:g11:gene-expression:code-props}{start codon} with
        initiator \omterm{def:g11:gene-expression:translation}{tRNA} (Met).
  \item \emph{\omterm{prop:g11:gene-expression:tx-stages}{Elongation}}: successive \omterm{def:g11:gene-expression:translation}{aminoacyl-tRNAs} enter; \omterm{def:g10:molecules-of-life:proteins}{peptide bonds} form;
        the \omterm{def:g11:gene-expression:translation}{ribosome} translocates three bases at a time.
  \item \emph{\omterm{prop:g11:gene-expression:tx-stages}{Termination}}: a \omterm{prop:g11:gene-expression:code-props}{stop codon} is recognized by release factors; the
        completed polypeptide is freed.
\end{itemize}
\end{proposition}
\admitted

\begin{method}[From gene fragment to peptide]\label{met:g11:gene-expression:pipeline}
\begin{enumerate}
  \item Identify the \omterm{def:g11:gene-expression:strands}{template strand}; transcribe to mRNA (U for A on the
        template; A for T; C for G; G for C), writing mRNA $5'\to 3'$.
  \item Or, if given the \omterm{def:g11:gene-expression:strands}{coding strand}, copy it $5'\to 3'$ with T$\to$U.
  \item Partition the mRNA into \omterm{def:g11:gene-expression:codon}{codons} from the start \texttt{AUG}.
  \item Use the \omterm{def:g11:gene-expression:codon}{genetic code} (or a provided table) to name each \omterm{def:g10:molecules-of-life:proteins}{amino acid} until
        a \omterm{prop:g11:gene-expression:code-props}{stop codon}.
\end{enumerate}
\end{method}

\begin{example}[Coding strand to peptide]\label{ex:g11:gene-expression:peptide}
Coding \omterm{def:g11:dna-replication:helix}{DNA}: \texttt{5'-ATG TTT GGC TAA-3'}. mRNA:
\texttt{5'-AUG UUU GGC UAA-3'}. Peptide: Met--Phe--Gly (then stop).
\end{example}

\section{Gene expression is regulated}

\begin{proposition}[Not all genes are equal at all times]\label{prop:g11:gene-expression:regulation}
\omterm{def:g10:the-cell:cell}{Cells} of a multicellular \omterm{def:g10:scientific-biology:organism}{organism} share essentially the same \omterm{prop:g11:dna-replication:genome}{genome}, yet
differ in which \omterm{def:g11:gene-expression:gene}{genes} they express. Regulation can act at \omterm{def:g11:gene-expression:transcription}{transcription},
\omterm{def:g10:molecules-of-life:nucleic}{RNA} processing, mRNA stability, \omterm{def:g11:gene-expression:translation}{translation}, and \omterm{def:g10:molecules-of-life:proteins}{protein} activity. At this
level it is enough to know that \omterm{prop:g11:gene-expression:tx-stages}{promoters}, regulatory \omterm{def:g11:dna-replication:helix}{DNA}, and regulatory
\omterm{def:g10:molecules-of-life:proteins}{proteins} decide whether \omterm{def:g11:gene-expression:transcription}{RNA polymerase} fires often or rarely at a \omterm{def:g11:gene-expression:gene}{gene}.
\end{proposition}
\admitted

\begin{remark}
A mutation that changes a single base in a \omterm{def:g11:gene-expression:gene}{gene} can change one \omterm{def:g11:gene-expression:codon}{codon}, one \omterm{def:g10:molecules-of-life:proteins}{amino
acid}, and sometimes \omterm{def:g10:molecules-of-life:proteins}{protein} function --- or, if it creates or destroys a \omterm{prop:g11:gene-expression:code-props}{stop
codon}, shorten or lengthen the chain. Linkage of genotype to phenotype continues
in \cref{ch:g11:mendelian-genetics,ch:g11:human-genetics}.
\end{remark}

\begin{problem}[From a short gene to a short protein]\label{pb:g11:gene-expression:1}
A bacterial \omterm{def:g11:gene-expression:gene}{gene} fragment has the \omterm{def:g11:gene-expression:strands}{coding strand}
\begin{center}
\texttt{5'-ATG CCT GAA TTC TGG TAG GGA-3'}.
\end{center}
Use the \omterm{def:g11:gene-expression:codon}{codon} hints: \texttt{AUG} Met (start), \texttt{CCU} Pro, \texttt{GAA}
Glu, \texttt{UUC} Phe, \texttt{UGG} Trp, \texttt{UAG} stop, \texttt{GGA} Gly.
\begin{enumerate}
  \item Write the \omterm{def:g11:gene-expression:strands}{template strand} ($3'\to 5'$ under the \omterm{def:g11:gene-expression:strands}{coding strand}).
  \item Write the mRNA ($5'\to 3'$).
  \item Translate until stop; list the \omterm{def:g10:molecules-of-life:proteins}{amino acids}.
  \item A mutation changes the coding-strand \omterm{def:g11:gene-expression:codon}{codon} \texttt{TGG} to
        \texttt{TGA}. What happens to the \omterm{def:g10:molecules-of-life:proteins}{protein}?
  \item A different mutation deletes the first base of the \texttt{GAA} \omterm{def:g11:gene-expression:codon}{codon}
        on the \omterm{def:g11:gene-expression:strands}{coding strand}. Describe the qualitative effect on \omterm{def:g11:gene-expression:translation}{translation}.
  \item Where do \omterm{def:g11:gene-expression:transcription}{transcription} and \omterm{def:g11:gene-expression:translation}{translation} occur in a bacterium versus a
        human \omterm{def:g10:the-cell:cell}{cell}?
  \item Why can a bacterium begin translating an mRNA before \omterm{def:g11:gene-expression:transcription}{transcription} has
        finished, while a human \omterm{def:g10:the-cell:cell}{cell} normally cannot?
\end{enumerate}
\end{problem}
