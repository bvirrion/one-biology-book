\section*{Chapitre \ref{ch:g10:heart-lungs-effort} --- Le cœur et les poumons à l'effort}

\begin{solution}{exo:g10:heart-lungs-effort:1}
$0.6 \times 14 = \qty{8.4}{L/min}$~; $2.2 \times 35 = \qty{77}{L/min}$.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:2}
Le volume de sang pompé par un côté du cœur par minute~:
$Q = f \times V_s$. $65 \times 0.075 \approx \qty{4.9}{L/min}$.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:3}
Oreillette droite, ventricule droit, artère pulmonaire, poumons
(circulation pulmonaire), veines pulmonaires, oreillette gauche,
ventricule gauche, aorte, organes (circulation systémique), veines,
oreillette droite.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:4}
Volume courant \qty{0.5}{L}~; capacité vitale \qty{4.5}{L}~; volume
résiduel \qty{1.2}{L}.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:5}
Environ $220 - 16 = 204$ et $220 - 60 = 160$ battements par minute.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:6}
\qty{100}{W}~: $120 \times 0.105 \approx \qty{12.6}{L/min}$~;
\qty{300}{W}~: $198 \times 0.115 \approx \qty{22.8}{L/min}$~; un facteur
de 1.8.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:7}
Repos~: \qty{2.5}{L/min}~; effort~: $0.04 \times 24 \approx
\qty{1.0}{L/min}$. La part est divisée par douze, le débit absolu
seulement par 2.6, parce que le débit total a lui-même grandi.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:8}
$18 \times 0.14 = \qty{2.5}{L/min}$~: environ les trois quarts des
\qty{3.45}{L/min} d'une élève entraînée --- proche, pour une personne
plus petite.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:9}
Le cerveau ne supporte pas même quelques secondes sans oxygène ni
glucose, et son approvisionnement est donc protégé. Le tube digestif et
les reins peuvent suspendre digestion et filtration une heure sans
dommage, et leur part est prêtée aux muscles.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:10}
$V_s = 5000/42 \approx \qty{120}{mL}$, contre environ \qty{70}{mL}~:
son cœur agrandi délivre le débit de repos en bien moins de battements.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:11}
La commande de mouvement du cerveau, et son anticipation de l'effort,
sont envoyées aux centres cardiaques et aux glandes surrénales avant
que les muscles n'agissent. Un coureur qui ne compterait que sur la
détection du dioxyde de carbone partirait avec un débit de repos,
courrait la première minute sur la \omterm{def:g10:cell-metabolism:fermentation}{fermentation}, et serait contraint de
ralentir par l'acide lactique avant que la livraison ne le rattrape.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:12}
$195 \times 0.110 \times 0.150 \approx \qty{3.2}{L/min}$ et
$195 \times 0.160 \times 0.150 \approx \qty{4.7}{L/min}$. C'est le
volume d'éjection que l'entraînement agrandit~: le second sujet est
celui qui est entraîné en endurance.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:13}
Oxygène inspiré~: $0.21 \times 100 = \qty{21}{L/min}$~; utilisé~:
\qty{3.5}{L/min}, soit 17~\%. Le reste est de nouveau expiré ---
l'air expiré contient encore environ 16~\% d'oxygène, et c'est
pourquoi le bouche-à-bouche fonctionne.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:14}
Débit au plus d'environ $110 \times 0.12 \approx \qty{13}{L/min}$ au
lieu de 24~: la $\dot V\!\mathrm{O_2}$max chute de près de moitié, vers
\qty{2}{L/min}. Le patient peut marcher et pédaler doucement mais ne
peut soutenir aucun effort intense.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:15}
Livraison par litre réduite d'un quart, extraction inchangée en
proportion~: la $\dot V\!\mathrm{O_2}$max baisse de 25~\%. Au cours des
semaines suivantes, le corps fabrique davantage de globules rouges, ce
qui ramène l'oxygène transporté par litre vers \qty{200}{mL}.
\end{solution}

\begin{solution}{pb:g10:heart-lungs-effort:1}
\textbf{1.} $62 \times 0.072 \approx 4.5$~; $105 \times 0.100 = 10.5$~;
$140 \times 0.116 \approx 16.2$~; $175 \times 0.120 = 21.0$~;
$198 \times 0.120 \approx \qty{23.8}{L/min}$.

\textbf{2.} $23.8/4.5 \approx 5.3$~: la fréquence y contribue pour $198/62
\approx 3.2$, le volume d'éjection pour $120/72 \approx 1.7$, et $3.2 \times
1.7 \approx 5.3$.

\textbf{3.} Au-dessus d'environ \qty{225}{W}, le volume d'éjection
reste à \qty{120}{mL}~; seule la fréquence cardiaque élève encore le
débit.

\textbf{4.} $220 - 17 = 203$~: ses 198 tiennent dans la précision de la règle.

\textbf{5.} $5/4.5 \approx \qty{1.1}{min}$, soit environ \qty{67}{s}~; à
\qty{300}{W}, $5/23.8 \approx \qty{0.21}{min}$, environ \qty{13}{s}.

\textbf{6.} $0.5 \times 13 = 6.5$~; $1.2 \times 18 = 21.6$~; $1.8 \times
24 = 43.2$~; $2.3 \times 32 = 73.6$~; $2.6 \times 44 \approx
\qty{114}{L/min}$.

\textbf{7.} $114/6.5 \approx 18$~: le volume courant a été multiplié par
5.2, la fréquence par 3.4~; le volume y contribue davantage.

\textbf{8.} $2.6/5.0 = 52\%$ de la capacité vitale par cycle.

\textbf{9.} Non~: la \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} dispose d'une large réserve et le sang
qui quitte les poumons reste saturé. La limite est le volume d'éjection
du cœur, qui fixe l'oxygène livré à chaque battement.

\textbf{10.} $0.7 \times 6.5 \approx \qty{4.6}{L/min}$ et $0.7 \times
114 \approx \qty{80}{L/min}$.

\textbf{11.} $4.5 \times 0.052 \approx 0.23$~; $10.5 \times 0.090
\approx 0.95$~; $16.2 \times 0.120 \approx 1.94$~; $21.0 \times 0.145
\approx 3.05$~; $23.8 \times 0.155 \approx \qty{3.7}{L/min}$.

\textbf{12.} $3700/68 \approx \qty{54}{mL/min}$ par kilogramme. La
fréquence a atteint son maximum et le gain de 225 à \qty{300}{W} est
faible~: c'est bien sa $\dot V\!\mathrm{O_2}$max, celle d'un élève en
forme et entraîné.

\textbf{13.} $3.7/0.23 \approx 16$~: le facteur 5.3 du cœur multiplié
par celui de l'extraction, $155/52 \approx 3.0$.

\textbf{14.} $3.7 \times 20 = \qty{74}{kJ}$ par minute, soit environ
\qty{1230}{W}~; au repos $0.23 \times 20/60 \approx \qty{77}{W}$~;
au-dessus du repos environ \qty{1150}{W}, d'où un rendement $300/1150 \approx
26\%$.

\textbf{15.} À peu près une droite de \num{0.23} à \qty{3.05}{L/min}
entre 0 et \qty{225}{W}, puis un fléchissement~: les derniers
\qty{75}{W} n'ajoutent que \qty{0.65}{L/min}. Le plafond est d'environ
\qty{3.7}{L/min}.

\textbf{16.} L'anticipation~: l'attente de l'effort est relayée par le
cerveau aux centres cardiaques et aux glandes surrénales, ce qui élève
la fréquence avant tout mouvement.

\textbf{17.} Les \omterm{def:g10:heart-lungs-effort:ventilation}{alvéoles} oxygènent pleinement le sang même à
\omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} et débit maximaux~; les poumons ne sont pas la limite. La
limite est la quantité de sang que le cœur peut envoyer par minute.

\textbf{18.} $20/23.8 \approx 84\%$. Le tube digestif et les reins, qui
prenaient la moitié du débit de repos, reçoivent désormais environ
\qty{1}{L/min}~; leurs artères se sont resserrées et le débit a été
redirigé vers les muscles.

\textbf{19.} $198 \times 0.140 \times 0.155 \approx \qty{4.3}{L/min}$,
soit environ \qty{63}{mL/min} par kilogramme.

\textbf{20.} Le \omterm{def:g10:heart-lungs-effort:output}{débit cardiaque} est multiplié par 5.3,
l'extraction d'oxygène par 3, la \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} par 18~;
l'entraînement a modifié le volume d'éjection du cœur, et avec lui le
plafond.
\end{solution}
