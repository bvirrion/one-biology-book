\section*{Chapter \ref{ch:g5:flower-to-fruit} --- From Flower to Fruit}

\begin{solution}{exo:g5:flower-to-fruit:1}
The \omterm{def:g3:flowers-fruits-seeds:parts}{stamens} are \omterm{def:g4:animal-reproduction:sexual}{male}: their \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} grains carry the \omterm{def:g4:animal-reproduction:sexual}{male}
reproductive \omterm{def:g5:first-look-at-cells:cell}{cells}. The \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} is \omterm{def:g4:animal-reproduction:sexual}{female}: inside it lie the
\omterm{prop:g5:flower-to-fruit:roles}{ovules}, each holding an \omterm{def:g4:animal-reproduction:sexual}{egg cell}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:2}
\omterm{def:g5:flower-to-fruit:steps}{Pollination}: a \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} grain travels from a \omterm{def:g3:flowers-fruits-seeds:parts}{stamen} to the top
of a \omterm{def:g3:flowers-fruits-seeds:parts}{pistil}. \omterm{def:g5:flower-to-fruit:steps}{Fertilization}: the grain's tube carries the \omterm{def:g4:animal-reproduction:sexual}{male}
\omterm{def:g5:first-look-at-cells:cell}{cell} down to an ovule, where it joins the \omterm{def:g4:animal-reproduction:sexual}{egg cell}.
\omterm{def:g5:flower-to-fruit:steps}{Pollination} comes first --- the journey before the meeting.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:3}
Made in a \omterm{def:g3:flowers-fruits-seeds:parts}{stamen}; carried (by insect or wind) to the top of a
\omterm{def:g3:flowers-fruits-seeds:parts}{pistil} --- \omterm{def:g5:flower-to-fruit:steps}{pollination}; grows a fine tube down through the
\omterm{def:g3:flowers-fruits-seeds:parts}{pistil}; its \omterm{def:g4:animal-reproduction:sexual}{male} \omterm{def:g5:first-look-at-cells:cell}{cell} reaches an ovule and joins the \omterm{def:g4:animal-reproduction:sexual}{egg cell}
--- \omterm{def:g5:flower-to-fruit:steps}{fertilization}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:4}
Each fertilized ovule becomes a \omterm{def:g2:from-seed-to-plant:seed}{seed}; the \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} swells into
the \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}; the \omterm{def:g3:flowers-fruits-seeds:parts}{petals} (and \omterm{def:g3:flowers-fruits-seeds:parts}{stamens}) wither and fall.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:5}
Each pip is a fertilized ovule become a \omterm{def:g2:from-seed-to-plant:seed}{seed}. Empty chambers
mean unfertilized \omterm{prop:g5:flower-to-fruit:roles}{ovules} --- too little \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} landed, and
the \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit} swelled unevenly around the shortfall.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:6}
Their \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} rides the wind: dust-fine, made in enormous
quantities, thrown into the air. The cost is waste ---
almost every grain lands nowhere useful, so quantity must
replace precision.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:7}
No visit, no \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} on the \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} --- \omterm{def:g5:flower-to-fruit:steps}{pollination}, the first
step, never happened, so no tube grew and no ovule was
fertilized; the \omterm{def:g3:flowers-fruits-seeds:parts}{flower} withers whole.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:8}
Open answer. Expected: hand-pollinated, thread-marked \omterm{def:g3:flowers-fruits-seeds:parts}{flowers}
set \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit} clearly more often --- the brush did the bee's job.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:9}
\omterm{def:g4:animal-reproduction:sexual}{Male} \omterm{def:g5:first-look-at-cells:cell}{cells}: in the \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} grains of the \omterm{def:g3:flowers-fruits-seeds:parts}{stamens}. \omterm{def:g4:animal-reproduction:sexual}{Egg cells}:
in the \omterm{prop:g5:flower-to-fruit:roles}{ovules} inside the \omterm{def:g3:flowers-fruits-seeds:parts}{pistil}. \omterm{def:g5:flower-to-fruit:steps}{Fertilization}: at the bottom
of the \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} tube, in the ovule. First \omterm{def:g5:first-look-at-cells:cell}{cell} of the new
individual: the fertilized \omterm{def:g4:animal-reproduction:sexual}{egg cell} in the ovule --- which
grows into the tiny \omterm{def:g1:plants-around-us:plant}{plant} of the \omterm{def:g2:from-seed-to-plant:seed}{seed}, as an \omterm{def:g1:animals-around-us:animal}{animal} grows
from its own first \omterm{def:g5:first-look-at-cells:cell}{cell}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:10}
A poor harvest --- most blossoms fell fruitless. The
machinery was sound and the \omterm{def:g3:flowers-fruits-seeds:parts}{flowers} unhurt; what failed was
step 1, \omterm{def:g5:flower-to-fruit:steps}{pollination}: for two crucial weeks the couriers
stayed home, so most \omterm{def:g3:flowers-fruits-seeds:parts}{pistils} never received \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} and most
\omterm{prop:g5:flower-to-fruit:roles}{ovules} were never fertilized.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:11}
The bumblebees sell \omterm{def:g5:flower-to-fruit:steps}{pollination} --- the transport step:
carrying \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} from \omterm{def:g3:flowers-fruits-seeds:parts}{stamens} to \omterm{def:g3:flowers-fruits-seeds:parts}{pistils} inside a building no
outdoor pollinator can enter. Tomato \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} shakes loose
easily, so a fan's vibration and moving air can do crude
transport within each \omterm{def:g3:flowers-fruits-seeds:parts}{flower} --- a mechanical courier. Cherry
\omterm{def:g3:flowers-fruits-seeds:parts}{pollen} must travel from tree to tree, \omterm{def:g3:flowers-fruits-seeds:parts}{flower} to \omterm{def:g3:flowers-fruits-seeds:parts}{flower}, with
a precision only a body that visits blossoms in turn
provides; wind in a hall cannot aim, so for cherries the
living courier has no cheap substitute.
\end{solution}
