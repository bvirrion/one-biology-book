\section{जंतु अपना पदार्थ भोजन से बनाते हैं}

\begin{proposition}[उपभोक्ता कार्बनिक पदार्थ को बदलते हैं]\label{prop:g6:matter-from-food:transform}
\omterm{def:g1:animals-around-us:animal}{जंतु} अपना \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} --- \omterm{def:g3:bones-and-muscles:muscle}{पेशी}, \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी}, चरबी, \omterm{def:g1:animals-around-us:covering}{पंख}, \omterm{def:g1:animals-around-us:covering}{खोल} ---
\emph{उसी \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} से बनाता है जो वह खाता है}। \omterm{def:g4:journey-of-food:digestion}{पाचन}
(\cref{def:g4:journey-of-food:digestion}) भोजन को खोलकर \omterm{def:g4:journey-of-food:digestion}{पोषक तत्व} बना
देता है; \omterm{prop:g4:journey-of-food:absorb}{रक्त} उन्हें पहुँचाता है
(\cref{prop:g4:heart-and-blood:carries}); और शरीर की \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ}
(\cref{prop:g5:first-look-at-cells:growth}) उन्हें फिर से जोड़कर \omterm{def:g1:animals-around-us:animal}{जंतु}
का अपना पदार्थ बना देती हैं। \omterm{def:g1:animals-around-us:animal}{जंतु} पदार्थ के \emph{रूपांतरकार} हैं,
खनिज से बनाने वाले कभी नहीं: वह पेशा अकेले \omterm{def:g5:food-webs:roles}{उत्पादकों} का है।
\end{proposition}
\admitted

\begin{example}[घास से गाय]\label{ex:g6:matter-from-food:cow}
उस आधे टन का पीछा करो: गाय घंटों-घंटों चरती है
(\cref{prop:g2:what-animals-eat:daily} ने उन लंबे घंटों को समझाया था);
उसका भारी \omterm{def:g4:journey-of-food:digestion}{पाचन} (\cref{exo:g4:journey-of-food:11}) घास को खोलकर \omterm{def:g4:journey-of-food:digestion}{पोषक
तत्व} बनाता है; उसका \omterm{prop:g4:journey-of-food:absorb}{रक्त} उन्हें थन, \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} और भीतर बढ़ते बछड़े तक ले
जाता है; और उसकी \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} घास-पदार्थ से गाय-पदार्थ बना देती हैं। वह
चरागाह, सचमुच के अर्थ में, नए रूप में इधर-उधर टहल रहा है।
\end{example}

\begin{remark}[पदार्थ हाथ बदलता है, अपनी प्रकृति नहीं]\label{rem:g6:matter-from-food:hands}
\omterm{def:g3:food-chains:chain}{आहार शृंखला} के हर तीर पर
(\cref{def:g3:food-chains:chain}) \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} मालिक बदलता है और
नए सिरे से बनता है --- घास-पदार्थ से गाय-पदार्थ, और गाय-पदार्थ (\omterm{def:g2:animals-grow-up:born}{दूध},
माँस) से बालक-पदार्थ। \emph{कार्बनिक} कभी शून्य से नहीं आता: \omterm{def:g5:food-webs:roles}{उपभोक्ता}
उसे आगे बढ़ाते और नए साँचे में ढालते हैं। वह नया सिर्फ़ शृंखला की हरी
शुरुआत पर बना था
(\cref{prop:g6:plants-are-producers:firstlink}); और सिर्फ़ अपघटकों वाले
सिरे पर, अगले अध्याय में, वह खनिज में लौटता है।
\end{remark}

\section{वृद्धि पदार्थ का उत्पादन है}

\begin{definition}[जंतु का पदार्थ-उत्पादन]\label{def:g6:matter-from-food:production}
\omterm{def:g1:animals-around-us:animal}{जंतु} का \emph{पदार्थ का निर्माण}\index{पदार्थ का निर्माण} वह नया
\omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} है जो वह बनाता है: नन्हों की वृद्धि (बछड़े के जुड़े हुए
किलो), हर साल की नई-नवेली चीज़ें (झड़कर फिर उगे \omterm{def:g1:animals-around-us:covering}{रोम}, सींग, \omterm{def:g1:animals-around-us:covering}{पंख}), मरम्मत
(\cref{rem:g3:bones-and-muscles:alive} की जुड़ी हुई \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी}), और उत्पाद
(\omterm{def:g2:animals-grow-up:born}{दूध}, \omterm{def:g2:animals-grow-up:born}{अंडे}, ऊन)। यह सब नए सिरे से बनाया गया भोजन ही है।
\end{definition}

\begin{example}[उत्पादन तौलना]\label{ex:g6:matter-from-food:weighing}
किसान रोज़ उत्पादन तौलते हैं: दुधारू गाय दिन में लगभग \qty{25}{kg} \omterm{def:g2:animals-grow-up:born}{दूध}
देती है --- यानी उतना \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} जो उसे दिन दर दिन चारे से फिर से
बनाना पड़ता है; और मुर्ग़ी का \omterm{def:g2:animals-grow-up:born}{अंडा} लगभग \qty{60}{g} बँधा हुआ सामान है
(\cref{rem:g2:animals-grow-up:egg}), जो रात भर में उसके अनाज से जोड़ा
जाता है। चारा नहीं, तो उत्पादन नहीं: बहीखाता बेरहमी से बराबर होता है।
\end{example}

\begin{proposition}[सारा भोजन शरीर नहीं बनता]\label{prop:g6:matter-from-food:losses}
कोई भी \omterm{def:g5:food-webs:roles}{उपभोक्ता} अपने पूरे भोजन को नया पदार्थ नहीं बना पाता:
\begin{itemize}
  \item एक हिस्सा कभी सोखा ही नहीं जाता और मल के रूप में बाहर चला जाता
        है (\cref{prop:g4:journey-of-food:absorb});
  \item एक बड़ा हिस्सा साँस की ऑक्सीजन के साथ मिलकर
        (\cref{prop:g4:breathing:exchange}) \emph{जलाया} जाता है, ताकि
        हरकत और गरमाहट चल सके: यानी ईंधन के रूप में ख़र्च हुआ पदार्थ,
        जो छोड़ी हुई गैस के रूप में निकल जाता है;
  \item और सिर्फ़ बचा-खुचा हिस्सा वृद्धि तथा उत्पादों में बदलता है।
\end{itemize}
गाय जो घास खाती है, उसका बस एक मामूली अंश नई गाय बनकर ख़त्म होता है ---
और इसीलिए \cref{prop:g5:food-webs:pyramid} के पिरामिड की हर मंज़िल अपने
नीचे वाली से इतनी छोटी होती है।
\end{proposition}
\admitted

\begin{example}[पिरामिड, आख़िरकार समझाया हुआ]\label{ex:g6:matter-from-food:pyramid}
मैदान की टनों घास एक झुंड भर ख़रगोशों का पेट भरती है; और ख़रगोशों के
किलो एक-दो लोमड़ियों का। हर तीर पर ज़्यादातर पदार्थ जीने में जल जाता है
या खो जाता है, और सिर्फ़ एक अंश शरीर के रूप में ऊपर बढ़ता है। बड़े
शिकारी दुर्लभ हैं (\cref{prop:g5:food-webs:pyramid}) क्योंकि वे
\emph{बचे-खुचे के बचे-खुचे} पर जीते हैं --- यह गणित है, बदक़िस्मती
नहीं।
\end{example}

\section{मेनू, दाँत और वही एक पेशा}

\begin{proposition}[अलग-अलग मेनू, वही एक रूपांतरण]\label{prop:g6:matter-from-food:menus}
\omterm{def:g2:what-animals-eat:kinds}{शाकाहारी}, \omterm{def:g2:what-animals-eat:kinds}{मांसाहारी}, \omterm{def:g2:what-animals-eat:kinds}{सर्वाहारी}
(\cref{def:g2:what-animals-eat:kinds}) \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} के अलग-अलग
\emph{स्रोतों} के नाम हैं, अलग-अलग पेशों के नहीं। घास को नए सिरे से
बनाती गाय और ख़रगोश को नए सिरे से बनाती लोमड़ी, दोनों
\cref{prop:g6:matter-from-food:transform} वाला रूपांतरण ही चलाती हैं
--- बस साज़-सामान मेनू के अनुरूप होता है
(\cref{prop:g2:what-animals-eat:match},
\cref{exo:g4:journey-of-food:11}): घास पीसने वाले दाँत और विशाल \omterm{def:g4:journey-of-food:digestion}{पाचन}
माँगती है, माँस दबोचने वाले दाँत और \omterm{def:g4:journey-of-food:tract}{छोटी आँत} को इनाम देता है।
\end{proposition}
\admitted

\begin{method}[किसी जंतु के पदार्थ की जाँच]\label{met:g6:matter-from-food:audit}
किसी भी \omterm{def:g1:animals-around-us:animal}{जंतु} के लिए उसका पदार्थ-बहीखाता बनाओ:
\begin{enumerate}
  \item \emph{आमद}: वह कौन-सा \omterm{def:g6:plants-are-producers:matter}{कार्बनिक पदार्थ} खाता है, और किससे?
        (उसके आहार-जाल के भीतर आते तीर,
        \cref{def:g5:food-webs:web});
  \item \emph{बना हुआ}: वह कौन-सी वृद्धि, कौन-सी नई-नवेली चीज़ें और
        कौन-से उत्पाद बनाता है?
  \item \emph{ख़र्च}: मल, और हरकत तथा गरमाहट के लिए जलाया गया पदार्थ;
  \item संतुलन जाँचो: उत्पादन मामूली बचा-खुचा ही होना चाहिए --- अगर
        कोई दावा उसे इससे ज़्यादा बताए (``यह तालाब उतनी मछली-पदार्थ
        देता है जितना भोजन वह उगाता ही नहीं''), तो वह दावा ग़लत है।
\end{enumerate}
\end{method}

\begin{example}[जाँच काम में]\label{ex:g6:matter-from-food:audituse}
कलचिड़ी की जाँच करो (\cref{ex:g2:what-animals-eat:winter}): आमद ---
केंचुए, \omterm{prop:g3:sorting-living-things:groups}{कीट}, जामुन; बना हुआ --- उसकी अपनी वृद्धि, हर साल के नए \omterm{def:g1:animals-around-us:covering}{पंख},
\omterm{def:g2:animals-grow-up:born}{अंडों} का गुच्छा; ख़र्च --- दीवार पर मल, और वह बड़ा जला हुआ हिस्सा जिसने
साल भर की उड़ान और गान चलाए। गाने वाली चिड़िया का रोज़ का भोजन उसके आकार
के हिसाब से इतना भारी ठीक इसीलिए होता है कि उड़ान बहीखाते का बहुत बड़ा
हिस्सा जला डालती है।
\end{example}

\section{समस्या: खेत का बहीखाता}

\begin{problem}[{सप्ताहांत समस्या --- एक छोटे खेत की जाँच, खेत से मेज़
तक}]\label{pb:g6:matter-from-food:1}
दादी की छोटी जोत: एक गाय वाला चरागाह, अनाज खाने वाली दस मुर्ग़ियों का
दड़बा, एक सब्ज़ी की क्यारी, और परिवार की रसोई। इस सप्ताहांत तुम उसका
पदार्थ-बहीखाता रखोगे।

\medskip\noindent\textbf{भाग I --- \omterm{def:g5:food-webs:roles}{उत्पादकों} की मंज़िल।}
\begin{enumerate}
  \item जोत के \omterm{def:g5:food-webs:roles}{उत्पादक} गिनाओ। उनकी खनिज सामग्री क्या है, और उनकी
        \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} का स्रोत क्या?
  \item चरागाह एक मौसम में लगभग \qty{10000}{kg} घास-पदार्थ उगाता है।
        उसका हर ग्राम किसकी \omterm{def:g1:plants-around-us:parts}{पत्तियों} में बना था?
  \item दादी दड़बे की खाद सब्ज़ी की क्यारी पर बिखेरती हैं।
        \cref{prop:g5:people-and-nature:lasting} का कौन-सा नियम वे
        निभा रही हैं, और वह खाद ठीक-ठीक क्या फिर से भरती है?
  \item गाय को सीधे \omterm{prop:g4:what-plants-need:needs}{खनिज लवण} और पानी खिलाकर चरागाह की बचत क्यों नहीं
        की जा सकती?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- \omterm{def:g5:food-webs:roles}{उपभोक्ताओं} की मंज़िल।}
\begin{enumerate}[resume]
  \item गाय दिन में लगभग \qty{70}{kg} घास खाती है और लगभग
        \qty{20}{kg} \omterm{def:g2:animals-grow-up:born}{दूध} देती है। बचे हुए \qty{50}{kg} की तीनों
        मंज़िलें बताओ।
  \item मुर्ग़ियाँ अनाज को \omterm{def:g2:animals-grow-up:born}{अंडों} में बदलती हैं। मुर्ग़ी का बहीखाता
        लिखो --- आमद, बना हुआ, ख़र्च --- और बताओ कि \omterm{def:g1:animals-around-us:covering}{पंख} झाड़ती (और
        सारे \omterm{def:g1:animals-around-us:covering}{पंख} नए बनाती) मुर्ग़ी कुछ समय के लिए कम \omterm{def:g2:animals-grow-up:born}{अंडे} क्यों देती
        है।
  \item एक बाज़ एक चूज़ा उठा ले जाता है। जोत की \omterm{def:g3:food-chains:chain}{आहार शृंखला} उस तक
        बढ़ाओ, और बताओ कि किन कड़ियों पर पदार्थ \emph{बनाया} गया और
        किन पर सिर्फ़ बदला गया।
  \item परिवार \omterm{def:g2:animals-grow-up:born}{अंडे}, \omterm{def:g2:animals-grow-up:born}{दूध} और सब्ज़ियाँ खाता है। एक नाश्ते के \omterm{def:g6:plants-are-producers:matter}{कार्बनिक
        पदार्थ} --- रोटी, उबला \omterm{def:g2:animals-grow-up:born}{अंडा}, \omterm{def:g2:animals-grow-up:born}{दूध} --- में से हर एक का पीछा उस
        \omterm{def:g1:plants-around-us:parts}{पत्ती} तक करो जहाँ वह पहली बार बना था।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- मंज़िलों का गणित।}
\begin{enumerate}[resume]
  \item दादी कहती हैं: ``यह चरागाह एक गाय का पेट भरता है; और अगर उस
        पर घास खाने वाले ख़रगोश खाकर लोमड़ियाँ पलतीं, तो यह लोमड़ियों
        की एक छोटी-सी माँद से ज़्यादा का पेट न भरता।'' हर मंज़िल के
        नुक़सानों से उनके अनुमान को सही ठहराओ।
  \item सब्ज़ी की क्यारी सीधे परिवार का पेट भरती है।
        \cref{exo:g6:matter-from-food:12} के तर्क से समझाओ कि वह
        क्यारी प्रति वर्ग मीटर चरागाह-गाय-दूध वाले रास्ते से ज़्यादा
        लोगों का पेट क्यों भरती है।
  \item सूखे में घास उगना बंद कर देती है। क्रम से भविष्यवाणी करो कि
        गाय के \omterm{def:g2:animals-grow-up:born}{दूध} का, उसके वज़न का और बहीखाते की ``बना हुआ'' पंक्ति
        का क्या होगा --- और यह भी कि बारिश न होने पर कोई अतिरिक्त धूप
        मदद क्यों नहीं कर सकती
        (\cref{pb:g6:plants-are-producers:1} का सातवाँ सवाल इसकी
        चाबी लिए है)।
  \item हर मंज़िल पर एक-एक वाक्य के साथ बहीखाता बंद करो: \omterm{def:g5:food-webs:roles}{उत्पादक},
        \omterm{def:g5:food-webs:roles}{उपभोक्ता} --- और यह भी बताओ कि हिसाब से अभी कौन ग़ायब है
        (यही अगले अध्याय का विषय है)।
\end{enumerate}
\end{problem}
