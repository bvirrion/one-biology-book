\chapter{النباتات منتِجة}\label{ch:g6:plants-are-producers}

زِن أصيصًا مسقيًّا، وازرع فيه \omterm{prop:g3:flowers-fruits-seeds:transform}{ثمرة} بلوط، ثم عُد بعد سنوات لتزن
البلوطة الصغيرة: كيلوغرامات من الخشب واللحاء والورق \omterm{def:g1:plants-around-us:parts}{والجذر} ---
وتربة الأصيص لم تكد تفقد غرامًا (وقد أثارت
\cref{ex:g4:what-plants-need:tree} هذه المفاجأة أول مرة).
كيلوغرامات من شجرة، مصنوعة مِمَّ؟ هذا الفصل يثبت الجواب --- ويسمّي
تجارة \omterm{def:g1:plants-around-us:plant}{النباتات} الفريدة.

\section{المادة العضوية والمادة المعدنية}

\begin{definition}[المادة المعدنية والمادة العضوية]\label{def:g6:plants-are-producers:matter}
المادة عند عالم الأحياء صنفان:
\begin{itemize}
  \item \emph{المادة المعدنية}\index{مادة معدنية}: مادة العالم
        غير الحي --- الماء، وغازات الهواء، والصخر، \omterm{prop:g4:what-plants-need:needs}{والأملاح
        المعدنية} المذابة في \cref{prop:g4:what-plants-need:needs}؛
  \item و\emph{المادة العضوية}\index{مادة عضوية}: مادة تصنعها
        \omterm{def:g6:exploring-our-environment:components}{الكائنات الحية} وتتألف منها أجسامها --- الخشب، والورق،
        والعضل، والشحم، والسكر، والصوف. قابلة للاحتراق، وللتعفن،
        ويمكن دائمًا تتبّعها إلى صانع حي.
\end{itemize}
\end{definition}

\begin{example}[فرز ما في الجيب]\label{ex:g6:plants-are-producers:sorting}
من جيب معطف بعد نزهة: حصاة --- معدنية؛ وبندقة --- عضوية (صنعتها
شجرة البندق)؛ وزرّ بلاستيكي --- عضوي بأصله، على غرابة ذلك
(فاللدائن تُصنع من النفط، وهو بقايا كائنات حية قديمة متحولة)؛
وقطرة مطر --- معدنية؛ وقفاز صوفي --- عضوي (صنع الصوفَ خروف).
\end{example}

\section{تجارة المنتِجين}

\begin{proposition}[النباتات تنتج مادة عضوية من مادة معدنية]\label{prop:g6:plants-are-producers:produce}
\omterm{def:g1:plants-around-us:plant}{النبات} الأخضر، إذا أُعطي مكوّنات \emph{معدنية} فحسب --- ماءً
وأملاحًا مذابة من التربة، وغازًا يؤخذ من \emph{الهواء} --- \omterm{prop:g3:balanced-diet:jobs}{وطاقة}
\emph{\omterm{def:g6:exploring-our-environment:conditions}{الضوء}}، صنع مادة \emph{عضوية}: سكره الخاص، ومنه الخشب
والورق \omterm{def:g1:plants-around-us:parts}{والجذر}. وهذا الصنع هو التجارة التي وعدت
\cref{prop:g4:what-plants-need:make} بشرحها: \omterm{def:g1:plants-around-us:plant}{فالنباتات}
\emph{منتِجة}\index{منتج} --- وهي الصانع الوحيد \omterm{def:g6:plants-are-producers:matter}{لمادة عضوية} جديدة
من موادّ معدنية صرفة. والمادة الخضراء في \omterm{def:g1:plants-around-us:parts}{الأوراق} هي آلة التقاط
\omterm{def:g6:exploring-our-environment:conditions}{الضوء}؛ والصناعة تجري في \omterm{def:g1:plants-around-us:parts}{الورقة}، نهارًا.
\end{proposition}
\admitted

\begin{example}[الأدلة مجموعةً]\label{ex:g6:plants-are-producers:evidence}
ثلاثة أرصاد سابقة تتلاقى الآن. فالأصيص الموزون: كيلوغرامات
البلوطة لم تأتِ من التربة --- بل جاءت في أكثرها من الهواء والماء.
والسجين الشاحب في \cref{ex:g4:what-plants-need:pale}: لا ضوء، فلا
صناعة --- \omterm{def:g1:plants-around-us:plant}{فالنبتة} تجوع في الظلام ومخزنها ملآن بالمعادن. والنعناع
المحروم من المعادن في \cref{exo:g4:what-plants-need:11}: الأملاح
لازمة --- لكن بوصفها موادّ قليلة، لا بوصفها الكتلة. فالكتلة من
الهواء والماء، \omterm{prop:g3:balanced-diet:jobs}{والطاقة} من \omterm{def:g6:exploring-our-environment:conditions}{الضوء}، والتوابل من التربة.
\end{example}

\begin{remark}[تبادل الغازات في الاتجاهين]\label{rem:g6:plants-are-producers:gas}
غاز الهواء الذي تأخذه \omterm{def:g1:plants-around-us:parts}{الورقة} هو \emph{ثاني أكسيد الكربون} ---
وهو الغاز نفسه الذي يزفره كل متنفس
(\cref{prop:g4:breathing:exchange})؛ والصناعة تطلق
\emph{الأكسجين} إلى الهواء --- وهو الغاز نفسه الذي يحتاج إليه كل
متنفس. \omterm{def:g1:plants-around-us:plant}{والنباتات} تتنفس أيضًا، ليلًا ونهارًا، كسائر \omterm{def:g6:exploring-our-environment:components}{الكائنات الحية}؛
لكن صناعة \omterm{def:g1:plants-around-us:parts}{الورقة} العاملة تسبق تنفسها بكثير في النهار. فهواء العالم
الصالح \omterm{def:g4:breathing:movements}{للتنفس} من نواتج \omterm{def:g5:food-webs:roles}{المنتِجين} الجانبية --- دَين آخر يدين به كل
\omterm{def:g1:animals-around-us:animal}{حيوان} للخضرة.
\end{remark}

\section{المنتِجون يُطعمون الجميع --- ويدّخرون لأنفسهم}

\begin{proposition}[الحلقة الأولى مشروحةً]\label{prop:g6:plants-are-producers:firstlink}
قضت \cref{prop:g3:food-chains:start} بأن كل \omterm{def:g3:food-chains:chain}{سلسلة غذائية} تبدأ
\omterm{def:g1:plants-around-us:plant}{بنبات}؛ وللقضاء الآن سببه. \omterm{def:g5:food-webs:roles}{فالمنتِجون} وحدهم يصنعون \omterm{def:g6:plants-are-producers:matter}{مادة عضوية} من
\omterm{def:g6:plants-are-producers:matter}{مادة معدنية}؛ وكل مستهلِك (\cref{def:g5:food-webs:roles}) لا يستطيع
إلا أن \emph{يحوّل} \omterm{def:g6:plants-are-producers:matter}{مادة عضوية} يأكلها. فكل \omterm{def:g6:plants-are-producers:matter}{مادة عضوية} في كل جسم حي
--- أرنب، أو ثعلب، أو صقر، أو أنت --- صُنعت أول مرة في \omterm{def:g1:plants-around-us:parts}{ورقة}، وهرم
الغذاء في \cref{prop:g5:food-webs:pyramid} قائم على الطابق الأخضر
لأن هناك تدخل المادة إلى العالم الحي.
\end{proposition}
\admitted

\begin{example}[مخزن النبات الخاص]\label{ex:g6:plants-are-producers:pantry}
ما يصنعه \omterm{def:g1:plants-around-us:plant}{النبات} فوق حاجة اليوم يدّخره احتياطيًّا عضويًّا: \omterm{ex:g4:plants-without-seeds:bulbtuber}{درنة}
البطاطا \omterm{ex:g4:plants-without-seeds:bulbtuber}{وبصلة} الخزامى
(\cref{ex:g4:plants-without-seeds:bulbtuber})، ونصفا الفولة
السميكان (\cref{def:g2:from-seed-to-plant:seed})، \omterm{def:g1:plants-around-us:parts}{وجذر} الجزرة
السمين (\cref{exo:g1:plants-around-us:10})، وحلاوة الشوندر. وكل
واحد منها ضوء صيف مودَع مادةً --- وكل واحد هو بالضبط ما يحصده
المزارعون والآكلون.
\end{example}

\begin{method}[تتبّع أي غذاء إلى ورقته]\label{met:g6:plants-are-producers:trace}
خذ أي غذاء على المائدة:
\begin{enumerate}
  \item فإن كان مادة نباتية --- كقمح الخبز، والبطاطا، والتفاحة
        --- فهو صناعة منتِج مباشرة: سمِّ \omterm{def:g1:plants-around-us:plant}{النبات} وعضوه المدَّخر؛
  \item وإن كان مادة حيوانية --- كالجبن \omterm{def:g2:animals-grow-up:born}{والبيض} واللحم --- فانزل
        في السلسلة: ماذا أكل \omterm{def:g1:animals-around-us:animal}{الحيوان}؟ ومن صانع ذلك الغذاء؟ وتابع
        حتى تبلغ \omterm{def:g1:plants-around-us:plant}{النبات}؛
  \item واكتب السلسلة بأسهم
        (\cref{def:g3:food-chains:chain})؛
  \item والخلاصة واحدة دائمًا: ينتهي الأثر عند \omterm{def:g1:plants-around-us:parts}{ورقة}، في \omterm{def:g6:exploring-our-environment:conditions}{الضوء}.
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:1}
عرّف \omterm{def:g6:plants-are-producers:matter}{المادة المعدنية} \omterm{def:g6:plants-are-producers:matter}{والمادة العضوية}، مع مثالين على كل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:2}
اذكر تجارة \omterm{def:g5:food-webs:roles}{المنتِجين}: المدخلات، ومصدر \omterm{prop:g3:balanced-diet:jobs}{الطاقة}، والمخرج، والمشغل
الذي تجري فيه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:3}
افرز معدنيًّا أو عضويًّا: الندى، \omterm{def:g1:animals-around-us:covering}{وصدفة} حلزون، وشمع النحل، والغرانيت،
وزيت الطعام، وملح البحر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:4}
من أين جاءت كيلوغرامات البلوطة في أكثرها، إن لم تكن من التربة
تقريبًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:5}
أي غاز تأخذه \omterm{def:g1:plants-around-us:parts}{الورقة} العاملة من الهواء، وأيًّا تطلق؟ ومن يزفر الأول
ويحتاج إلى الثاني؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:plants-are-producers:6}
سمِّ أربعة أعضاء ادّخار نباتية وما يدّخره كل واحد، من
\cref{ex:g6:plants-are-producers:pantry}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:7}
لماذا تجوع \omterm{def:g1:plants-around-us:plant}{نبتة} الخزانة المظلمة ومخزنها ملآن \omterm{prop:g4:what-plants-need:needs}{بالأملاح المعدنية}؟
وأي مدخل لا بديل له؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:8}
لماذا لا يستطيع أي \omterm{def:g1:animals-around-us:animal}{حيوان} أن يكون الحلقة الأولى في \omterm{def:g3:food-chains:chain}{سلسلة غذائية}؟
أجب بالتمييز بين المنتِج والمستهلِك.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:9}
طبّق \cref{met:g6:plants-are-producers:trace} على شطيرة جبن: خبز،
وزبدة، وجبن --- ثلاثة آثار إلى ثلاث \omterm{def:g1:plants-around-us:parts}{أوراق}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:10}
``السماد يُطعم \omterm{def:g1:plants-around-us:plant}{النبات} كما يُطعم الدريس البقرة.'' صحّح الجملة
تصحيحًا دقيقًا مستعملًا \cref{rem:g4:what-plants-need:soil}
وتعريفات هذا الفصل --- ماذا يوفّر السماد حقًّا، وماذا يصنع \omterm{def:g1:plants-around-us:plant}{النبات}
لنفسه؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:plants-are-producers:11}
حوض مائي مغلق مضاء فيه \omterm{def:g1:plants-around-us:plant}{نباتات} مائية وأسماك يمكن أن يعمل شهورًا؛
أما المغلق \emph{المظلم} فيفشل في أيام. استعمل
\cref{rem:g6:plants-are-producers:gas} لتشرح النتيجتين.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:plants-are-producers:12}
تجربة قديمة: غرسة صفصاف نمت خمس سنوات في حوض فازدادت كيلوغرامات
كثيرة؛ ولم تفقد تربة الحوض المجففة إلا غرامات. فاستنتج المجرّب أن
``الشجرة صُنعت من الماء وحده''. فأي جزء من استنتاجه كان صحيحًا،
وأيها ناقصًا --- وأي مدخل معدني ثانٍ، كان يجهله، يضيفه عالم أحياء
معاصر؟
\end{exercise}

\section{مسألة: سنة البستاني}

\begin{problem}[{مسألة نهاية الأسبوع --- الضوء والأوراق ودفتر
حسابات حديقة خضر}]\label{pb:g6:plants-are-producers:1}
يزرع بستاني بطاطا وجزرًا وفاصولياء وخسًّا، ويحتفظ بدفتر حسابات دقيق
لما يدخل وما يخرج.

\medskip\noindent\textbf{القسم الأول --- لغز الدفتر.}
\begin{enumerate}
  \item تغلّ الحديقة طنّين من الخضر في السنة؛ ولا ينثر البستاني
        إلا أكياس سماد. زِن اللغز: من أين يأتي الطنّان إن لم
        يأتيا من الأكياس؟ (أجب بتجارة \omterm{def:g5:food-webs:roles}{المنتِجين}.)
  \item صنّف المحصول: البطاطا، والجزرة، \omterm{def:g2:from-seed-to-plant:seed}{وبذرة} الفاصولياء، \omterm{def:g1:plants-around-us:parts}{وورقة}
        الخس --- سمِّ دور كل واحد عند \omterm{def:g1:plants-around-us:plant}{النبات} (عضو ادّخار، أو عضو
        عامل، أو مخزن \omterm{def:g2:from-seed-to-plant:seed}{بذرة}).
  \item يقول البستاني: ``مورّدي الحقيقي يوصّل مع ضوء النهار.''
        ترجم ذلك إلى مصطلحات
        \cref{prop:g6:plants-are-producers:produce}.
  \item ولماذا يحتاج البستاني إلى السماد أصلًا، إن لم يكن من أجل
        الطنّين؟ (تذكّر النعناع في الرمل.)
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- قراءة المواسم.}
\begin{enumerate}[resume]
  \item خسّ حزيران، وقد ظُلّل تحت قماش تجريبي أسبوعًا، يخرج شاحبًا
        ممتدًّا. سمِّ التجربة التي يعيدها هذا، والمدخل الذي أزاله.
  \item تُكبَّس صفوف البطاطا بالتراب وتُمدح أعاليها الوارفة:
        ``أعالٍ جيدة، \omterm{ex:g4:plants-without-seeds:bulbtuber}{درنات} جيدة.'' اشرح القول --- ما الصلة بين
        \omterm{def:g1:plants-around-us:parts}{الأوراق} العاملة في الأعلى والادّخار المنتفخ في الأسفل؟
  \item جفاف آب يوقف النمو مع أن الشمس متوهجة. فأي مدخل معدني
        أخفق، ولماذا لا يستطيع \omterm{def:g6:exploring-our-environment:conditions}{الضوء} أن يحلّ محله؟
  \item في أيلول يقطع البستاني \omterm{def:g1:plants-around-us:plant}{نبتات} الفاصولياء ويرفعها لكنه
        يقلب جذورها في التربة. فيسمّي جار له ذلك تبذيرًا. دافع عن
        الممارسة بتعريفات المادة --- ما الذي يُردّ إلى التربة،
        ولمن؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- خارج الحديقة.}
\begin{enumerate}[resume]
  \item خنزير البستاني يأكل البطاطا الصغيرة؛ والأسرة تأكل
        الخنزير. اكتب السلسلة، واذكر أين صُنعت مادتها العضوية
        أصلًا.
  \item تزعم زبونة أن سلطتها ``شمس في معظمها''. دقّق في الزعم:
        عدّد مكوّنات \omterm{def:g1:plants-around-us:parts}{ورقة} السلطة الحقيقية ودور الشمس بالضبط.
  \item يقتل الصقيع آخر الخسّ في ليلة؛ فيتعفن حيث هو بحلول تشرين
        الثاني. ومادته العضوية لا تزول. استبق
        \cref{ch:g6:decomposers-and-soil}: من يأخذها، وإلى أي
        حال؟
  \item لخّص الحديقة في جملة واحدة تحوي: \omterm{def:g5:food-webs:roles}{المنتِجين}، \omterm{def:g6:plants-are-producers:matter}{والمادة
        المعدنية}، \omterm{def:g6:plants-are-producers:matter}{والمادة العضوية}، \omterm{def:g6:exploring-our-environment:conditions}{والضوء}.
\end{enumerate}
\end{problem}
