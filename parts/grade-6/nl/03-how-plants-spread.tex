\chapter{Hoe planten zich verspreiden}\label{ch:g6:how-plants-spread}

In maart wordt er voor een nieuwe weg een talud uitgegraven: rauwe, kale
aarde, geen groen spikkeltje. Kom in september terug en het is helemaal
dichtgestikt met \omterm{prop:g2:from-seed-to-plant:cycle}{kiemplantjes} --- klaprozen, paardenbloemen,
wilgenscheuten, grassen. Niemand heeft er gezaaid. En toch hebben
\omterm{def:g1:plants-around-us:plant}{planten}, die geen stap kunnen zetten, de kale grond binnen een paar
maanden gekoloniseerd. Dit hoofdstuk gaat over hun reisafdeling.

\section{Planten reizen als zaad en spore}

\begin{proposition}[Vastzittende planten, reizende beginnetjes]\label{prop:g6:how-plants-spread:travel}
Een volwassen \omterm{def:g1:plants-around-us:plant}{plant} zit haar leven lang vast
(\cref{def:g1:plants-around-us:plant}) --- dus verspreidt een \omterm{def:g5:biodiversity:species}{soort} zich
door haar \emph{beginnetjes} uit te sturen: \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} en \omterm{prop:g3:flowers-fruits-seeds:transform}{vruchten}
(\cref{ch:g5:flower-to-fruit}), \omterm{def:g4:plants-without-seeds:spore}{sporen}
(\cref{def:g4:plants-without-seeds:spore}), of \omterm{ex:g4:plants-without-seeds:runner}{uitlopers} en andere
vegetatieve opmarsen
(\cref{def:g4:plants-without-seeds:vegetative}). Zo nieuw terrein in
bezit nemen heet \emph{kolonisatie}\index{kolonisatie}.
\end{proposition}
\admitted

\begin{definition}[Verspreiding]\label{def:g6:how-plants-spread:dispersal}
\emph{Verspreiding}\index{verspreiding} is het wegdragen van \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden},
\omterm{prop:g3:flowers-fruits-seeds:transform}{vruchten} of \omterm{def:g4:plants-without-seeds:spore}{sporen} bij de moederplant vandaan. De vervoerders zijn weinig
in getal en oeroud: de \emph{wind}, \emph{\omterm{def:g1:animals-around-us:animal}{dieren}}, het \emph{water} ---
en sommige \omterm{prop:g3:flowers-fruits-seeds:transform}{vruchten} die hun \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} zelf wegslingeren.
\end{definition}

\begin{example}[Waarom eigenlijk van huis weg?]\label{ex:g6:how-plants-spread:whyleave}
Een \omterm{def:g2:from-seed-to-plant:seed}{zaad} dat aan de voet van de moederplant valt, \omterm{def:g2:from-seed-to-plant:germination}{ontkiemt} in haar
schaduw, tussen haar \omterm{def:g1:plants-around-us:parts}{wortels} --- en moet om precies het licht en het
water uit \cref{prop:g4:what-plants-need:needs} strijden met een
tegenstander die honderd keer zo groot is. \omterm{def:g6:how-plants-spread:dispersal}{Verspreiding} is de ontsnapping
aan die verloren wedstrijd: afstand is het eerste geschenk dat een \omterm{def:g1:plants-around-us:plant}{plant}
haar jongen geeft.
\end{example}

\section{De vervoerders, en de vruchten die voor hen gebouwd zijn}

\begin{proposition}[Gebouwd naar de vervoerder]\label{prop:g6:how-plants-spread:builds}
De vormen van \omterm{prop:g3:flowers-fruits-seeds:transform}{vruchten} en \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} passen bij hun vervoerder --- het passen
uit \cref{def:g4:adapted-to-their-world:adaptation}, toegepast op bagage:
\begin{itemize}
  \item \emph{wind}: parachutes (de gepluimde \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} van de
        paardenbloem), vleugels (de vruchtjes van esdoorn en es die
        schuin omlaag tollen), of stoffijne \omterm{def:g4:plants-without-seeds:spore}{sporen} die eenvoudig zweven
        (de miljoenen van de varen);
  \item \emph{\omterm{def:g1:animals-around-us:animal}{dieren}, aan de buitenkant}: klitten --- grote klit,
        kleefkruid --- waarvan de haakjes \omterm{def:g1:animals-around-us:covering}{vacht} en sokken grijpen en
        meerijden tot ze eraf geplukt worden;
  \item \emph{\omterm{def:g1:animals-around-us:animal}{dieren}, aan de binnenkant}: zoete vlezige \omterm{prop:g3:flowers-fruits-seeds:transform}{vruchten} ---
        kersen, bramen, rozenbottels --- die in hun geheel opgegeten
        worden; de \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} met hun taaie schil gaan er ongeschonden
        doorheen en worden elders afgezet, met bemesting erbij
        (het antwoord van \cref{exo:g3:flowers-fruits-seeds:10},
        afgemaakt);
  \item \emph{water}: drijvende \omterm{prop:g3:flowers-fruits-seeds:transform}{vruchten} --- de piepkleine vlotjes van
        de els, de zeevarende schil van de kokosnoot;
  \item \emph{zelf wegslingeren}: peulen die opdrogen, onder spanning
        komen en openknappen --- gaspeldoorn en viooltjes schieten hun
        \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} meters van de moederplant weg.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Het talud lezen]\label{ex:g6:how-plants-spread:roadside}
Het talud van september, ontcijferd: paardenbloemen en wilgen ---
gepluimde \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}, met de wind aangevlogen; klaprozen --- hun piepkleine
\omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} lagen al in de verzette grond te wachten zoals alleen \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} dat
kunnen (\cref{exo:g2:from-seed-to-plant:9}); braamzaailingen onder de
hekpalen --- afgezet door zittende \omterm{prop:g3:sorting-living-things:groups}{vogels}, bemesting inbegrepen; grote
klit langs het pad --- binnengebracht op een hond. De aanreisroute van
elke kolonist staat in zijn uitrusting geschreven.
\end{example}

\begin{method}[De vervoerder aan de vrucht herkennen]\label{met:g6:how-plants-spread:name}
Bij een \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht} of een \omterm{def:g2:from-seed-to-plant:seed}{zaad}:
\begin{enumerate}
  \item kijk naar vliegtuig --- pluimen, vleugels, stoffijnheid: de wind;
  \item kijk naar haakjes: dierenvacht, buitenkant;
  \item vlezig, gekleurd, zoet? de darm van een \omterm{def:g1:animals-around-us:animal}{dier}, binnenkant ---
        controleer of de \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} taaie schillen hebben;
  \item drijvend, dicht bij water? het water;
  \item een opgedroogde, gedraaide, gespannen peul? zelf wegslingeren;
  \item toets je oordeel: laat hem vallen, laat hem drijven, druk hem
        tegen een wollen mouw.
\end{enumerate}
\end{method}

\section{Koloniseren zonder zaden}

\begin{example}[De opmars over de grond]\label{ex:g6:how-plants-spread:vegetative}
Naast de aanvliegers lopen sommige kolonisten op hun eigen vegetatieve
voeten (\cref{ch:g4:plants-without-seeds}): kweekgras rukt ondergronds op
met kruipende \omterm{def:g1:plants-around-us:parts}{stengels}, meter na meter; bramen bogen over en \omterm{def:g1:plants-around-us:parts}{wortelen}
waar hun toppen de grond raken; de \omterm{ex:g4:plants-without-seeds:runner}{uitlopers} van de aardbei springen over
open grond. Traag, korte afstand --- en niet te stuiten door een
verwaarloosde tuin, zoals elke tuinier weet.
\end{example}

\begin{remark}[Kolonisatie met twee snelheden]\label{rem:g6:how-plants-spread:twospeeds}
De twee systemen verdelen het werk. \omterm{prop:g3:flowers-fruits-seeds:transform}{Zaden} en \omterm{def:g4:plants-without-seeds:spore}{sporen} zijn de dienst voor
de \emph{lange afstand} --- zeldzame gelukkige landingen ver weg, de
paardenbloem die de stad oversteekt. De vegetatieve opmars is de dienst
voor de \emph{korte afstand} --- dicht en zeker bezit nemen van de grond
ernaast. De bramenpol die zich vanuit de heg uitbreidt gebruikt allebei
tegelijk: bogende \omterm{def:g1:plants-around-us:parts}{stengels} voor de meter, door \omterm{prop:g3:sorting-living-things:groups}{vogels} gedragen \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} voor
de kilometer.
\end{remark}

\section{Probleem: De verlaten akker}

\begin{problem}[{Weekendprobleem --- tien jaar kolonisatie, bekeken vanaf
één hek}]\label{pb:g6:how-plants-spread:1}
Een geploegde akker wordt in de herfst verlaten. Een natuuronderzoeker
fotografeert hem tien jaar lang elke juni vanaf het hek, en houdt
aantekeningen bij.

\medskip\noindent\textbf{Deel I --- Jaar één.}
\begin{enumerate}
  \item Juni van jaar één: de kale aarde is bezaaid met klaprozen,
        paardenbloemen en distels. Noem bij elk de waarschijnlijke bron
        en route --- de klaprozen uit de bodem zelf, de andere door de
        lucht. Welke uitrusting maakte de aanvoer door de lucht mogelijk?
  \item In het midden van de akker verschijnen in jaar één geen bramen,
        maar onder de oude hekpalen staan drie zaailingen. Verklaar dat
        patroon met de dienst binnen in het \omterm{def:g1:animals-around-us:animal}{dier} --- waarom precies
        onder de palen?
  \item De natuuronderzoeker vindt nergens \omterm{prop:g4:plants-without-seeds:damp}{varens}, hoewel er bovenwinds
        een bos vol \omterm{prop:g4:plants-without-seeds:damp}{varens} ligt. De \omterm{def:g4:plants-without-seeds:spore}{sporen} van \omterm{prop:g4:plants-without-seeds:damp}{varens} zijn er zeker
        aangekomen (\cref{ex:g4:plants-without-seeds:fern} telt hun
        miljoenen). Welke omstandigheid van de open akker weigerde ze,
        volgens \cref{prop:g4:plants-without-seeds:damp}?
  \item Kweekgras dringt alleen vanaf de rand van de akker binnen, een
        meter of twee per jaar. Welk kolonisatiesysteem is dat, en
        waarom kan het niet midden in de akker opduiken zoals de
        paardenbloemen deden?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De jaren daarna.}
\begin{enumerate}[resume]
  \item Tegen jaar drie hebben grassen en hoge kruiden de kale grond
        gesloten, en zijn de klaprozen uit de staande begroeiing
        verdwenen. Waar zijn de klaprozen nu, en in welke vorm? (Denk aan
        het talud en aan \cref{prop:g6:life-through-seasons:herbs}.)
  \item Vanaf jaar vier staan er bramenbogen en jonge meidoorns op de
        akker, het dichtst bij de heggen en onder de hekrij. Noem de twee
        diensten die ze daar hebben neergezet, en het bewijs in het
        patroon.
  \item Tegen jaar acht staan wilgen- en berkenopslag hoger dan de
        bramen --- allebei bomen met piepkleine gepluimde of stoflichte
        \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}. Waarom ontbreken de eiken en kastanjes met hun zware \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}
        nog bijna helemaal, en welke dierlijke koerier \omterm{def:g1:plants-around-us:plant}{plant} uiteindelijk
        zelfs die?
  \item De foto's laten de bezetting van de akker jaar na jaar
        veranderen, hoewel geen tuinman haar heeft aangeraakt. Formuleer
        de sturende regel: wat verandert de begroeiing van elk jaar aan
        de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} van de akker zelf --- vooral aan het licht op
        de grond --- en wie bevoordeelt die verandering vervolgens?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De algemene les.}
\begin{enumerate}[resume]
  \item Zet de kolonisten op volgorde van aankomst: de wachtende \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}
        in de bodem, de windrijders, de door \omterm{def:g1:animals-around-us:animal}{dieren} gebrachte struiken,
        de bomen met zware \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}. Koppel elke golf aan de
        verspreidingsdienst die haar timing bepaalde.
  \item Uit de voegen van een stadsplein komt dezelfde bezetting van
        jaar één --- klaprozen, paardenbloemen --- maar de struiken van
        jaar vier komen er nooit. Geef twee redenen waarom de opeenvolging
        daar stilvalt.
  \item De natuuronderzoeker beweert: ``met rust gelaten wordt deze
        akker bos''. Verdedig die bewering met de trend van tien jaar ---
        en noem de weiuitzondering uit
        \cref{prop:g5:people-and-nature:managed} die bewijst dat ``met
        rust gelaten'' de beslissende woorden zijn.
  \item Vat het hele probleem samen in twee zinnen met de woorden
        \emph{\omterm{def:g6:how-plants-spread:dispersal}{verspreiding}}, \emph{\omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}} en \emph{\omterm{prop:g6:how-plants-spread:travel}{kolonisatie}}.
\end{enumerate}
\end{problem}
