\section*{الفصل \ref{ch:b1:species-interactions} --- التفاعلات بين الأنواع}

\begin{solution}{exo:b1:species-interactions:1}
\omterm{def:b1:species-interactions:types}{التنافس} $-/-$ (بلوطتان على الضوء)؛ \omterm{def:b1:species-interactions:types}{والافتراس} $+/-$ (وشق وأرنب)؛
\omterm{def:b1:species-interactions:types}{والتطفل} $+/-$ (قرادة وأيّل)؛ \omterm{def:b1:species-interactions:types}{والتقايض} $+/+$ (نحلة وزهرة)؛
\omterm{def:b1:species-interactions:types}{والمعايشة} $+/0$ (سرخس على قشرة)؛ والتضادّ $-/0$ (عشب تحت ظل
بلوطة).
\end{solution}

\begin{solution}{exo:b1:species-interactions:2}
نوعان يستعملان المورد الحادّ نفسه بالطريقة نفسها لا يستطيعان
التعايش إلى ما لا نهاية في بيئة ثابتة. والهوازج الخمس تستعمل
الغذاء نفسه لكن في أجزاء مختلفة من الشجرة وفي أوقات مختلفة:
فمكاناتها مختلفة، فلا اثنتان منها على «الطريقة نفسها».
\end{solution}

\begin{solution}{exo:b1:species-interactions:3}
\omterm{def:b1:species-interactions:functional}{زمن المعالجة}: الزمن اللازم لصيد فريسة واحدة وأكلها وهضمها، ولا
تُؤخذ خلاله غيرها. ومعدل الهجوم: المساحة الممسوحة في وحدة الزمن
مضروبة في احتمال الأسر. والمعدل الأعظم $1/h = 2$ فريسة في
الساعة.
\end{solution}

\begin{solution}{exo:b1:species-interactions:4}
نوع تغيّر إزالتُه بنيةَ المجتمع خارج
نسبة وفرته، وذلك عادة بتحكمه في منافس سائد. فالوفرة تقيس
الحضور لا الأثر؛ ومفترس نادر يأكل الفائز المحتمل له أثر لا
يُظهره عدده.
\end{solution}

\begin{solution}{exo:b1:species-interactions:5}
\omterm{prop:b1:species-interactions:lotka}{خط التساوي} 1: $N_1 + 0.5 N_2 = 500$ (والتقاطعان 500 على $N_1$ و1000 على
$N_2$)؛ \omterm{prop:b1:species-interactions:lotka}{وخط التساوي} 2: $N_2 + 0.6 N_1 = 400$ (400 على $N_2$ و667 على $N_1$).
وهما يتقاطعان مع كون كل نوع محدودًا بنفسه أكثر ($\alpha = 0.5 <
K_1/K_2 = 1.25$؛ و$\beta = 0.6 < K_2/K_1 = 0.8$): أي تعايش مستقر
عند $N_1 = 429$ و$N_2 = 143$. وعند $\alpha = 2$: يتقاطع \omterm{prop:b1:species-interactions:lotka}{خط التساوي} 1 عند
500 و250، وكلاهما داخل \omterm{prop:b1:species-interactions:lotka}{خط التساوي} 2 (667 و400): فيفوز النوع 2
دائمًا.
\end{solution}

\begin{solution}{exo:b1:species-interactions:6}
$f = aN/(1 + ahN)$: فعند 10، $2/(1 + 0.5) = 1.33$ في الساعة؛ وعند 100،
$20/(1 + 5) = 3.33$ في الساعة (والأعظم $1/h = 4$). ونصف الإشباع عند
$N = 1/(ah) = 20$ في المتر المربع.
\end{solution}

\begin{solution}{exo:b1:species-interactions:7}
مع استجابة من النمط III يرتفع كسر الفريسة المأكول مع
الكثافة عند الكثافة المنخفضة: فتُؤخذ الفريسة النادرة أقل تناسبيًا
وتستطيع التعافي. أما مع النمط II فيكون الكسر المأكول أعلى ما يكون عند
الكثافة المنخفضة (فالمفترس لا يشبع هناك أبدًا)، فتُدفع \omterm{def:b1:populations:population}{جمهرة}
الفريسة الصغيرة إلى أسفل أكثر.
\end{solution}

\begin{solution}{exo:b1:species-interactions:8}
كان \emph{P.~caudatum} يستعمل البكتيريا نفسها في عمود الماء نفسه
الذي يستعمله \emph{P.~aurelia}، ودفع \emph{P.~aurelia}، وهو الأسرع
والأقل استهلاكًا، الغذاءَ دون ما يحتاج إليه. أما \emph{P.~bursaria} فكان
يتغذى في القاع على موارد مختلفة: \omterm{def:b1:ecosystem-organization:niche}{فالمكانة}، لا معدل النمو، هي التي حسمت
--- ذلك أن \emph{P.~bursaria} لا ينمو أسرع من \emph{P.~caudatum}.
\end{solution}

\begin{solution}{exo:b1:species-interactions:9}
تلقى المفترسات النمط أكثر بلا أذى، فتتعلمه أقل
جودة، وتهاجمه أكثر: فتهبط الحماية عن المحاكي والنموذج معًا.
ومن ثم تهبط لياقة المحاكي كلما شاع، وهو تعلق سالب
بالتواتر يبقيه نادرًا نسبة إلى النموذج.
\end{solution}

\begin{solution}{exo:b1:species-interactions:10}
بلا المفترس يقع خط تساوي بلح البحر خارج خط كل
منافس: فيربح الصخر. \omterm{def:b1:species-interactions:types}{والافتراس} يخفض $K$ الفعلية لبلح البحر
(فيُسحب \omterm{prop:b1:species-interactions:lotka}{خط تساويه} إلى الداخل) حتى يقطع
خطوط الآخرين من الداخل: فيصير المنافسون محدودين
بأنفسهم أكثر مما يحدّهم بلح البحر، فيتعايشون معه. فالمفترس
يعمل على أقوى المنافسين ويفسح المكان للباقين.
\end{solution}

\begin{solution}{exo:b1:species-interactions:11}
حين يكون الفوسفات هو الحادّ، تكون \qty{80}{\%} من الفوسفات أثمن
بكثير من \qty{15}{\%} من السكر: فهو \omterm{def:b1:species-interactions:types}{تقايض}. وفي تربة غنية
بالفوسفات يلتقط النبات الفوسفات بنفسه ويكلفه الفطر
\qty{15}{\%} بلا مقابل: فهو \omterm{def:b1:species-interactions:types}{تطفل}. والتجربة: نمِّ نباتات
مع الفطر وبدونه عند مدى من مستويات الفوسفات
وقِس الكتلة الحيوية؛ فالارتباط تقايضي حيث تنمو النباتات
الميكوريزية أكبر، وتطفلي حيث تنمو أصغر.
\end{solution}

\begin{solution}{exo:b1:species-interactions:12}
أقفاص استبعاد: قطع مسيَّجة على ضفاف الأنهار لا تستطيع الأيائل
دخولها، مزدوجة بقطع مفتوحة، مكررة في عدة أودية بذئاب
وبلا ذئاب، مقيسة سنوات قبل إعادة الإدخال
وبعدها؛ وسجّل ارتفاع الصفصاف وتغطيته، وكثافة الأيائل والزمن
الذي تقضيه في كل قطعة (بالآثار والكاميرات)، والصيد وعمق الثلج.
فالشلال يتنبأ بتعافي الصفصاف في القطع المفتوحة حيث توجد الذئاب
وحدها، بالمعدل نفسه الذي داخل أقفاص الاستبعاد؛ فإن تعافى الصفصاف
سواء أوُجدت الذئاب أم لم توجد، أو تتبّع الثلج والصيد بدل ذلك،
فقد دُحض الشلال أو شاركته تلك العوامل.
\end{solution}

\begin{solution}{pb:b1:species-interactions:1}
\textbf{1.} A: $N_A + 1.4 N_B = 200$، والتقاطعان 200 ($N_A$) و143
($N_B$). وB: $N_B + 0.9 N_A = 130$، والتقاطعان 130 ($N_B$) و144
($N_A$).
\textbf{2.} التقاطعات (200 في مقابل 144 على محور $N_A$، و143
في مقابل 130 على محور $N_B$) تضع خط تساوي A خارج خط تساوي B
بتمامه: فحلّ $N_A = 200 - 1.4 N_B$ مع $N_B = 130 - 0.9 N_A$ يعطي
$N_A = -69$، فيلتقي الخطان خارج الربع الموجب. ويفوز A من أي نقطة
بداية: فحيثما استطاع B أن ينمو ينمو A أيضًا ويبقى بعده.
\textbf{3.} $T_A = 0.87$ يوم، و$T_B = 1.26$ يوم: فالنوع A هو الأسرع
نموًا.
\textbf{4.} A: $N_A + 0.3 N_C = 200$ (200 و667)؛ وC: $N_C + 0.3 N_A =
100$ (100 و333). وكل نوع يحدّ نفسه أكثر مما يحدّ الآخر
($0.3 < 200/100$ و$0.3 < 100/200 = 0.5$): فتعايش مستقر.
\textbf{5.} الغابة ليست ثابتة (فالفصول والاضطرابات تعيد ضبط
السباق قبل أن يُحسم) وفيها موارد وأماكن كثيرة، فتتقاسم
الأنواع \omterm{def:b1:ecosystem-organization:niche}{المكانات} بدل أن تتشارك واحدة.
\textbf{6.} $N_A = 200 - 0.3 N_C$، و$N_C = 100 - 0.3 N_A = 100 - 60 +
0.09 N_C$، فيكون $N_C = 44$ و$N_A = 187$.
\textbf{7.} ترتفع ثم تنبسط نحو 10 تقريبًا: أي النمط II.
\textbf{8.} $(1/N, 1/f)$: $(0.2, 0.5)$ و$(0.1, 0.303)$ و$(0.05, 0.2)$
و$(0.025, 0.149)$ و$(0.0125, 0.125)$: أي خط مستقيم.
\textbf{9.} الميل $(0.5 - 0.125)/(0.2 - 0.0125) = 2.0 = 1/a$، فيكون $a =
\qty{0.5}{m^2/d}$؛ والتقاطع $0.5 - 2\times 0.2 = 0.1$ يوم $= h$.
\textbf{10.} الأعظم $1/h = 10$ فريسة في اليوم؛ ونصف الإشباع عند $N =
1/(ah) = 20$ في المتر المربع.
\textbf{11.} كسر المعالجة $= f h = f/10$: فيزيد على النصف حين $f
> 5$، أي عند 40 و80 في المتر المربع (و5.0 عند 20 هي النصف بالضبط).
\textbf{12.} نمو الفريسة $0.1 N (1 - N/200)$ يساوي الاستهلاك
$0.5N/(1 + 0.05N)$: أي $0.1(1 - N/200) = 0.5/(1 + 0.05 N)$، ومنه
$(1 - N/200)(1 + 0.05N) = 5$: أي $1 + 0.05N - N/200 - N^2/4000 = 5$، ومنه
$N^2 - 180 N + 16\,000 = 0$، و$N = 90 \pm \sqrt{8100 - 16\,000}$: ولا
\omterm{def:b1:flowering-plant-organization:organs}{جذر} حقيقي --- فخنفساء واحدة في المتر المربع تأكل أكثر مما تستطيع
الفريسة إنتاجه أبدًا (فالنمو الأعظم $rK/4 = 5$ في اليوم، عند $N = 100$، حيث
تأكل الخنفساء 8.3) وتسوقها إلى الانقراض. فلا توازن؛
ولا تبقى الفريسة إلا إذا قلّت الخنافس أو كان للفريسة ملاجئ.
\textbf{13.} ينتصف \omterm{def:b1:species-interactions:functional}{زمن المعالجة} للفريسة الثانية ($h' = 0.05$)،
فتأكل الخنفساء عند الإشباع حتى 20 منها في اليوم: فيتضاعف مدخولها
عند الكثافة العالية تقريبًا إذا تحولت إلى الفريسة الأسرع.
\textbf{14.} $H_{15} = \ln 15 = 2.71$. وللأنواع الثمانية: $-(0.85\ln 0.85
+ 7\times 0.0214\ln 0.0214) = 0.138 + 0.576 = 0.71$.
\textbf{15.} بمعامل $2.71/0.71 = 3.8$.
\textbf{16.} أبقت إشارة $-$ من نجم البحر على بلح البحر إشارةَ $-$ من بلح البحر
على منافسيه ضعيفةً؛ فأزل الأولى يبلغ بلح البحر
الكثافة التي يقصي عندها تنافسُه الباقين.
\textbf{17.} تقيس الكتلة الحيوية كم من دفق الطاقة يحمله نوع،
لا ما تفعله تفاعلاته؛ وأثر \omterm{def:b1:species-interactions:keystone}{النوع المفتاح} يمر عبر
الأنواع التي يتحكم فيها، وكتلتها الحيوية كبيرة.
\textbf{18.} كانت إزالة نجم البحر ستحرر نوعًا نادرًا
لا يستطيع الاستيلاء على الصخر: فما كان المجتمع الذي يسوده بلح البحر
ليتغير كثيرًا، وما كان نجم البحر
ليكون نوعًا مفتاحًا.
\textbf{19.} أقفاص تستبعد نجم البحر عن رقع صغيرة، مع
أقفاص مفتوحة شواهدَ لأثر القفص نفسه: فإن استولى بلح البحر
في الأقفاص المغلقة وحدها كان \omterm{def:b1:species-interactions:types}{الافتراس} على بلح البحر هو الآلية؛
وأقفاص فيها نجم البحر حاضرًا وبلح البحر مزال باليد
تختبر هل تهم فرائس نجم البحر الأخرى.
\textbf{20.} الكسب $\qty{40}{mg}$، والكلفة $8\times 0.5 = \qty{4}{mg}$: فالصافي
\qty{36}{mg} $=$ \qty{0.58}{kJ}.
\textbf{21.} \qty{40}{\micro g} من \qty{4}{mg}: أي \qty{1}{\%} من
نواتج تركيب الزهرة الضوئية اليومية تشتري نحو ثلاث نقلات لقاح، أي
تكاثر النبتة.
\textbf{22.} كل واحد يأخذ شيئًا من الآخر، لكن كلًّا منهما يكسب أكثر
مما يخسر: فالإشارة إشارة الأثر الصافي، لا
الاستغلال.
\textbf{23.} السارق $+$، والنبتة $-$ (رحيق مأخوذ بلا تلقيح): أي
\omterm{def:b1:species-interactions:types}{تطفل} على \omterm{def:b1:species-interactions:types}{التقايض}؛ والنحلة تجد أزهارًا مفرَّغة فتكسب
أقل، وهو $-$ غير مباشر.
\textbf{24.} النبتة $\leftrightarrow$ النحلة $+/+$؛ والسارق $\to$ النبتة $-$،
والسارق $\to$ النحلة $-$؛ والطائر $\to$ السارق $-$. وإشارتا سالب: فللطائر
أثر $+$ غير مباشر في النبتة وفي النحلة.
\textbf{25.} $a = \qty{0.5}{m^2/d}$، و$h = \qty{0.1}{d}$ (أي \qty{2.4}{h})،
والأعظم \num{10} فرائس في اليوم.
\end{solution}
