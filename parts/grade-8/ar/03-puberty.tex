\chapter{البلوغ}\label{ch:g8:puberty}

في مكان ما بين التاسعة والسادسة عشرة تقريبًا، يعيد كل جسم بشري بناء
نفسه: وهي دفعة النمو الثانية في
\cref{rem:g3:stages-of-life:bursts}، وأكثر بكثير من الطول. وقد
وعدت \cref{rem:g5:human-reproduction:adults} بهذا الفصل منذ
سنوات. وها هو --- الوقائع، وظلم الجدول الزمني المشهور، والآلة، وهي
التي ستسلّمنا اكتشافًا أكبر من \omterm{def:g8:puberty:puberty}{البلوغ} نفسه: أن للجسم نظام تواصل
ثانيًا.

\section{ما الذي يتغير}

\begin{definition}[البلوغ]\label{def:g8:puberty:puberty}
\emph{البلوغ}\index{بلوغ} هو المرحلة التي يصير فيها جسم الطفل جسمَ
بالغ --- وقبل كل شيء بالغًا بمعنى
\cref{ch:g8:sexual-reproduction}: فتنضج أعضاء التكاثر وتبدأ إنتاج
\emph{\omterm{def:g8:sexual-reproduction:gametes}{الأمشاج}}، ويأخذ الجسم صورته البالغة. وهو اللبّ البيولوجي
للمراهقة (\cref{def:g3:stages-of-life:stages}).
\end{definition}

\begin{proposition}[التغيرات المرئية]\label{prop:g8:puberty:changes}
إلى جانب دفعة النمو، على مدى سنوات عدة:
\begin{itemize}
  \item عند الفتيات، من نحو التاسعة إلى الثالثة عشرة في العادة:
        ينمو الثديان، ويتسع الحوض، ويظهر شعر الجسم --- وتعلّم
        \emph{أول الحيضات} بدء الدورات في
        \cref{ch:g8:reproductive-systems}؛
  \item وعند الفتيان، بعد ذلك بسنة أو سنتين في العادة: يغلظ الصوت
        (``ينكسر'')، وتعرض الكتفان، ويظهر شعر الجسم والوجه، وتبدأ
        الخصيتان إنتاج النطف؛
  \item وعند الجنسين: يصير \omterm{def:g1:the-five-senses:sense}{الجلد} أدهن (سنّ البثور)، ويتغير \omterm{rem:g7:removing-wastes:sweat}{العرق}،
        وتركب الأمزجة إعادةَ البناء --- \omterm{def:g5:brain-in-command:system}{فالدماغ} أيضًا يُعاد توصيله
        نحو صورته البالغة.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[الجدول الزمني شخصي]\label{rem:g8:puberty:timetable}
قالتها \cref{rem:g1:my-body:different} في الطول؛ ويكررها \omterm{def:g8:puberty:puberty}{البلوغ}
بصوت أعلى. فزملاء من سنّ واحدة قد يقفون على هذا الطريق متباعدين
سنوات --- والمبكر والمتأخر كلاهما طبيعي، وترتيب التغيرات يختلف،
ويكاد الجميع يبلغ المكان نفسه في النهاية. والمدى المذكور آنفًا
واسع، وأوسع ما فيه قلّةُ ما يقوله عن أي شخص بعينه.
\end{remark}

\section{الآلة: نظام رسائل ثانٍ}

\begin{definition}[الهرمونات، أول لقاء]\label{def:g8:puberty:hormones}
\omterm{def:g8:puberty:puberty}{البلوغ} لا يُقرَّر في الأعضاء التي يغيّرها. بل هو
\emph{مأمور به} --- من رسل كيميائيين يسمَّون
\emph{الهرمونات}\index{هرمون}: وهي موادّ تطلقها في
\emph{\omterm{prop:g4:journey-of-food:absorb}{الدم}} أعضاءٌ متخصصة، فتُحمل إلى كل مكان، وتطيعها الأعضاء
المبنية لسماعها. وعند \omterm{def:g8:puberty:puberty}{البلوغ} يستيقظ مركز تحكم في قاعدة \omterm{def:g5:brain-in-command:system}{الدماغ}،
فتأمر هرموناته أعضاء التكاثر، وتأمر \emph{هرموناتها} هي --- محمولةً
في الجسم كله بخدمة توصيل \cref{prop:g4:heart-and-blood:carries}
--- بإعادة البناء كلها: النمو، والشعر، والصوت، والدورات.
\end{definition}

\begin{example}[إشارة واحدة وورش كثيرة]\label{ex:g8:puberty:sites}
\omterm{prop:g4:journey-of-food:absorb}{الدم} يحمل \omterm{def:g8:puberty:hormones}{الهرمون} إلى كل مكان دفعةً واحدة --- ولهذا كان \omterm{def:g8:puberty:puberty}{البلوغ}
إعادة بناء \emph{منسَّقة}: فالصوت والكتفان والشعر تتبع الرسول نفسه،
على جدول واحد، مع أن المواضع تفصل بينها نصف جسم. ولا سلك عصبي يصل
صندوق الصوت باللحية؛ وإنما يصلهما بثّ مجرى \omterm{prop:g4:journey-of-food:absorb}{الدم}
(و\cref{ch:g8:hormonal-communication} تجعل لهذا النظام فصلًا خاصًّا
به).
\end{example}

\begin{remark}[ولماذا طقس المزاج؟]\label{rem:g8:puberty:moods}
الرسل أنفسهم يبلغون \omterm{def:g5:brain-in-command:system}{الدماغ}، وهو نفسه في وسط الترميم: فتقلبات
المزاج، والحدّة الجديدة، \omterm{prop:g1:healthy-habits:needs}{والنوم} ينجرف متأخرًا --- طقس، لا خطأ.
ومعرفة الآلة تعين على جانبيها: فالعواصف كيمياء مع بناء، وهي طبيعية،
وهي تمضي.
\end{remark}

\section{العيش خلال إعادة البناء}

\begin{proposition}[العناية في أثناء البناء]\label{prop:g8:puberty:care}
الجسم في وسط إعادة البناء له حاجات بنّاء، وقد لُبّيت كلها من قبل في
هذا الكتاب:
\begin{itemize}
  \item المادة والوقود: فالدفعة تُدفع مادةً --- أكلٌ متوازن
        (\cref{met:g3:balanced-diet:check})، ومادة البناء منه
        (\cref{exo:g7:digestion-and-nutrients:11})؛
  \item \omterm{prop:g1:healthy-habits:needs}{والنوم}: فترميم \omterm{def:g5:brain-in-command:system}{الدماغ} ونمو الجسم كلاهما يجري ليلًا
        (\cref{prop:g5:brain-in-command:protect}) --- والساعة
        المنجرفة متأخرةً ما زالت تحتاج إلى ساعاتها؛
  \item والحركة: \omterm{def:g3:bones-and-muscles:skeleton}{فالعظام} \omterm{def:g4:heart-and-blood:heart}{والقلب} في آخر سنوات بنائها وأحسنها
        (\cref{met:g3:bones-and-muscles:care})؛
  \item ولا تخريب: فمواد
        \cref{prop:g5:healthy-choices:substances} تضرب أشدّ ما
        تضرب أجسامًا وأدمغة ما زالت قيد
        البناء.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[للأسئلة عناوين]\label{ex:g8:puberty:questions}
\omterm{def:g8:puberty:puberty}{البلوغ} يثير أسئلة --- عن التوقيت، وعن الأجسام، وعن المشاعر --- ولها
عناوين صحيحة: فممرضات المدارس والأطباء يجيبون عنها كل يوم، في سرّية
وبلا دهشة؛ والكبار الموثوق بهم مرّوا بها جميعًا؛ وفصول هذا الكتاب
مبنية لتُقرأ من جديد. أما ما لا يستحق ثقة: فأساطير الملعب وأسوأ
أركان الشابكة، وهي تتاجر بشكوك هذه السنّ بعينها.
\end{example}

\section{مسألة: مؤتمر منحنيات النمو}

\begin{problem}[{مسألة نهاية الأسبوع --- أربعة منحنيات نمو
(متخيَّلة)، مقروءة بالفصل كله}]\label{pb:g8:puberty:1}
يراجع طبيب مدرسة أربعة منحنيات، الطول في مقابل العمر، مع تواريخ
قياس كل واحد: أ، وقد قفز منحناه قفزًا حادًّا من الحادية عشرة إلى
الثالثة عشرة وهو يستوي الآن عند الخامسة عشرة؛ وب، يصعد صعودًا
مستويًا في الثالثة عشرة، ولم تبدأ دفعته بعد؛ وج، في وسط دفعته عند
الرابعة عشرة؛ ود، جرى قسمه الحادّ من التاسعة إلى الثانية عشرة ثم
استوى.

\medskip\noindent\textbf{القسم الأول --- قراءة المنحنيات.}
\begin{enumerate}
  \item ضع كل منحنى على خريطة
        \cref{rem:g3:stages-of-life:bursts}: أي دفعة تظهر، وأين
        يقف كل تلميذ عليها؟
  \item يقلق ب من أن يكون ``قد تُرك خلفًا''. صُغ جواب الطبيب من
        \cref{rem:g8:puberty:timetable}، مع التنبؤ الذي يؤيده
        صعود منحناه المستوي.
  \item ويسأل د أيعني الاستواء ``انتهاء النمو في الثانية عشرة''.
        صحّح القراءة: ما الذي ينهيه الاستواء، وما الذي يستمر
        بعده؟
  \item ولماذا يقرأ الطبيب \emph{الميل} لا الطول ليحكم أين يقف
        كل واحد؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- خلف المنحنيات.}
\begin{enumerate}[resume]
  \item سمِّ الآلة التي بدأت كل دفعة، والطريق الذي سلكته أوامرها.
  \item ويتابع الطبيب في هذه الزيارات \omterm{prop:g1:healthy-habits:needs}{النوم} والغذاء أيضًا. اربط
        كل واحد بسطره في \cref{prop:g8:puberty:care}.
  \item ويذكر ج عواصف مزاج ونومًا ينجرف متأخرًا. أعطِ تفسير
        \cref{rem:g8:puberty:moods} ذا الشطرين --- والنتيجة
        العملية \omterm{prop:g1:healthy-habits:needs}{لنوم} ج في أسبوع الدراسة.
  \item ويسأل أ الطبيب على انفراد عن تغير لا يستطيع المنحنى
        إظهاره. اذكر، من غير أن تخترع السؤال، لماذا كانت غرفة
        الاستشارة هي العنوان الصحيح
        (\cref{ex:g8:puberty:questions}).
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- مذكرة
المؤتمر.}
\begin{enumerate}[resume]
  \item اكتب مذكرة السطر الواحد لكل منحنى: المرحلة، والطبيعية،
        وأي نقطة عناية.
  \item المنحنيات الأربعة تتباعد سنوات، ومع ذلك يصنّفها الطبيب
        كلها تحت ``طبيعي''. برّر ذلك في جملة واحدة.
  \item ويطلب وليّ أمر ``شيئًا يسرّع ب''. أجب بملاحظة الجدول
        الزمني وبالآلة: ما الذي هناك ليُصلَح؟
  \item اختم بملخص الفصل ذي الجملة الواحدة للوحة إعلانات الصف:
        ما \omterm{def:g8:puberty:puberty}{البلوغ}، ومن يديره، وما الذي يحتاج إليه.
\end{enumerate}
\end{problem}
