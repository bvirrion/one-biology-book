\chapter{पौधे कैसे फैलते हैं}\label{ch:g6:how-plants-spread}

मार्च में सड़क के लिए एक नई कटान बुलडोज़र से बनाई जाती है: कच्ची, नंगी
मिट्टी, हरे का एक कण भी नहीं। सितंबर में लौटो और वह पूरी तरह \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुरों} से
गुँथी मिलती है --- पोस्ते, कुकरौंधे, बेंत के \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर}, घासें। किसी ने उसे
बोया नहीं। फिर भी \omterm{def:g1:plants-around-us:plant}{पौधों} ने, जो एक क़दम भी नहीं चल सकते, महीनों में उस
नंगी ज़मीन पर क़ब्ज़ा कर लिया। यह अध्याय उन्हीं के यात्रा-विभाग के बारे
में है।

\section{पौधे बीजों और बीजाणुओं के रूप में सफ़र करते हैं}

\begin{proposition}[जमे हुए पौधे, सफ़र करती शुरुआतें]\label{prop:g6:how-plants-spread:travel}
वयस्क \omterm{def:g1:plants-around-us:plant}{पौधा} जीवन भर अपनी जगह जमा रहता है
(\cref{def:g1:plants-around-us:plant}) --- इसलिए कोई \omterm{def:g5:biodiversity:species}{प्रजाति} अपनी
\emph{शुरुआतें} भेजकर फैलती है: \omterm{def:g2:from-seed-to-plant:seed}{बीज} और \omterm{prop:g3:flowers-fruits-seeds:transform}{फल}
(\cref{ch:g5:flower-to-fruit}), \omterm{def:g4:plants-without-seeds:spore}{बीजाणु}
(\cref{def:g4:plants-without-seeds:spore}), या \omterm{ex:g4:plants-without-seeds:runner}{भूस्तारी} और कायिक
बढ़त के दूसरे रूप
(\cref{def:g4:plants-without-seeds:vegetative})। इस तरह नई ज़मीन पर
क़ब्ज़ा करने को \emph{उपनिवेशन}\index{उपनिवेशन} कहते हैं।
\end{proposition}
\admitted

\begin{definition}[प्रकीर्णन]\label{def:g6:how-plants-spread:dispersal}
\emph{प्रकीर्णन}\index{प्रकीर्णन} \omterm{def:g2:from-seed-to-plant:seed}{बीजों}, फलों या \omterm{def:g4:plants-without-seeds:spore}{बीजाणुओं} का जनक \omterm{def:g1:plants-around-us:plant}{पौधे} से
दूर पहुँचाया जाना है। ढोने वाले गिनती के और बहुत पुराने हैं:
\emph{हवा}, \emph{\omterm{def:g1:animals-around-us:animal}{जंतु}}, \emph{पानी} --- और कुछ ऐसे \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} जो अपने \omterm{def:g2:from-seed-to-plant:seed}{बीज}
ख़ुद उछाल देते हैं।
\end{definition}

\begin{example}[घर छोड़ना ही क्यों?]\label{ex:g6:how-plants-spread:whyleave}
जनक की \omterm{def:g1:plants-around-us:parts}{जड़} के पास गिरा \omterm{def:g2:from-seed-to-plant:seed}{बीज} जनक की छाया में, उसी की \omterm{def:g1:plants-around-us:parts}{जड़ों} के बीच \omterm{def:g2:from-seed-to-plant:germination}{अंकुरित
होता है} --- और \cref{prop:g4:what-plants-need:needs} के उसी \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} तथा
पानी के लिए ऐसे प्रतिद्वंद्वी से होड़ करता है जो उससे सौ गुना बड़ा है।
\omterm{def:g6:how-plants-spread:dispersal}{प्रकीर्णन} इसी हारी हुई भिड़ंत से बच निकलना है: दूरी वही पहला उपहार है
जो \omterm{def:g1:plants-around-us:plant}{पौधा} अपनी संतान को देता है।
\end{example}

\section{ढोने वाले, और उनके लिए बने फल}

\begin{proposition}[ढोने वाले के अनुरूप बनावट]\label{prop:g6:how-plants-spread:builds}
फलों और \omterm{def:g2:from-seed-to-plant:seed}{बीजों} की बनावट अपने ढोने वालों से मेल खाती है --- यानी
\cref{def:g4:adapted-to-their-world:adaptation} वाला मेल, सामान पर लगाया
हुआ:
\begin{itemize}
  \item \emph{हवा}: पैराशूट (कुकरौंधे के कलगीदार \omterm{def:g2:from-seed-to-plant:seed}{बीज}), डैने (मेपल और
        ऐश के वे \omterm{def:g2:from-seed-to-plant:seed}{बीज} जो तिरछे घूमते हुए उतरते हैं), या धूल जितने महीन
        \omterm{def:g4:plants-without-seeds:spore}{बीजाणु} जो बस तैरते रहते हैं (\omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} के लाखों);
  \item \emph{\omterm{def:g1:animals-around-us:animal}{जंतु}, बाहर से}: चिपटन वाले \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} --- बर्डॉक, क्लीवर्स ---
        जिनके अँकुड़े \omterm{def:g1:animals-around-us:covering}{रोम} और मोज़ों को पकड़ लेते हैं और तब तक सवारी
        करते हैं जब तक साफ़ न कर दिए जाएँ;
  \item \emph{\omterm{def:g1:animals-around-us:animal}{जंतु}, भीतर से}: मीठे गूदेदार \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} --- चेरी, ब्लैकबेरी,
        गुलाब के \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} --- जो पूरे के पूरे खा लिए जाते हैं; सख़्त छिलके
        वाले \omterm{def:g2:from-seed-to-plant:seed}{बीज} बिना नुक़सान पार निकल जाते हैं और कहीं और, खाद के साथ
        छोड़ दिए जाते हैं (\cref{exo:g3:flowers-fruits-seeds:10} के
        उत्तर का बाक़ी हिस्सा);
  \item \emph{पानी}: तैरने वाले \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} --- ऐल्डर के नन्हे बेड़े, नारियल का
        समुद्र पार करने वाला \omterm{def:g1:animals-around-us:covering}{खोल};
  \item \emph{ख़ुद उछालना}: वे फलियाँ जो सूखकर तन जाती हैं और चटककर
        खुलती हैं --- गोर्स और बनफ़्शा अपने \omterm{def:g2:from-seed-to-plant:seed}{बीज} जनक से मीटरों दूर फेंक
        देते हैं।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[सड़क किनारे को पढ़ना]\label{ex:g6:how-plants-spread:roadside}
सितंबर की सड़क-कटान, खोलकर पढ़ी हुई: कुकरौंधे और बेंत --- कलगीदार \omterm{def:g2:from-seed-to-plant:seed}{बीज},
हवा से उड़कर आए; पोस्ते --- उनके नन्हे \omterm{def:g2:from-seed-to-plant:seed}{बीज} हिलाई गई मिट्टी में पहले से
पड़े थे, इंतज़ार करते हुए, जैसा सिर्फ़ \omterm{def:g2:from-seed-to-plant:seed}{बीज} कर सकते हैं
(\cref{exo:g2:from-seed-to-plant:9}); बाड़ के खंभों के नीचे ब्लैकबेरी
के \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} --- बैठने वाले \omterm{prop:g3:sorting-living-things:groups}{पक्षियों} के छोड़े हुए, खाद समेत; और पगडंडी के
पास बर्डॉक --- किसी कुत्ते पर लदकर आया। हर उपनिवेशी के आने का रास्ता
उसके साज़-सामान में लिखा है।
\end{example}

\begin{method}[फल से ढोने वाले का नाम बूझना]\label{met:g6:how-plants-spread:name}
कोई \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} या \omterm{def:g2:from-seed-to-plant:seed}{बीज} सामने हो, तो:
\begin{enumerate}
  \item उड़ान का सामान ढूँढ़ो --- कलगी, डैने, धूल जैसी महीनता: यानी
        हवा;
  \item अँकुड़े ढूँढ़ो: यानी \omterm{def:g1:animals-around-us:animal}{जंतु} के \omterm{def:g1:animals-around-us:covering}{रोम}, बाहर से;
  \item गूदेदार, रंगीन, मीठा? यानी \omterm{def:g1:animals-around-us:animal}{जंतु} की आँत, भीतर से --- और \omterm{def:g2:from-seed-to-plant:seed}{बीजों}
        के सख़्त छिलके जाँचो;
  \item तैरता हुआ, पानी के पास? यानी पानी;
  \item सूखी, ऐंठी हुई, स्प्रिंग जैसी फली? यानी ख़ुद उछालना;
  \item अपने फ़ैसले की जाँच करो: उसे गिराओ, तैराओ, ऊनी आस्तीन से
        दबाओ।
\end{enumerate}
\end{method}

\section{बिना बीजों के उपनिवेशन}

\begin{example}[ज़मीनी बढ़त]\label{ex:g6:how-plants-spread:vegetative}
हवाई रास्ते से आने वालों के साथ-साथ कुछ उपनिवेशी अपने ही कायिक पैरों पर
चलकर आते हैं (\cref{ch:g4:plants-without-seeds}): दूब ज़मीन के नीचे
रेंगते \omterm{def:g1:plants-around-us:parts}{डंठलों} से मीटर दर मीटर बढ़ती है; ब्लैकबेरी की डालियाँ झुककर जहाँ
उनकी नोक ज़मीन छूती है वहाँ \omterm{def:g1:plants-around-us:parts}{जड़ें} जमा लेती हैं; और स्ट्रॉबेरी के
\omterm{ex:g4:plants-without-seeds:runner}{भूस्तारी} खुली ज़मीन पर छलाँग लगाते जाते हैं। धीमे, कम दूरी वाले --- और
लावारिस बग़ीचे में बिलकुल न रुकने वाले, जैसा हर माली जानता है।
\end{example}

\begin{remark}[दो रफ़्तारों का उपनिवेशन]\label{rem:g6:how-plants-spread:twospeeds}
दोनों व्यवस्थाएँ काम बाँट लेती हैं। \omterm{def:g2:from-seed-to-plant:seed}{बीज} और \omterm{def:g4:plants-without-seeds:spore}{बीजाणु} \emph{लंबी दूरी} की
सेवा हैं --- दूर कहीं क़िस्मत से हुई गिनती की लैंडिंग, जैसे पूरे क़स्बे
को पार करता कुकरौंधा। और कायिक बढ़त \emph{कम दूरी} की सेवा है --- यानी
पास वाली ज़मीन पर घना, पक्का क़ब्ज़ा। बाड़ से फैलती ब्लैकबेरी की क्यारी
एक साथ दोनों काम में लेती है: मीटर भर के लिए झुकती डालियाँ, और मील भर
के लिए \omterm{prop:g3:sorting-living-things:groups}{पक्षियों} के ढोए \omterm{def:g2:from-seed-to-plant:seed}{बीज}।
\end{remark}

\end{exercise}

समुद्र से किलोमीटरों दूर एक नंगा द्वीप उभरता है। कौन-सी \omterm{def:g6:how-plants-spread:dispersal}{प्रकीर्णन} सेवाएँ
वहाँ तक पहुँच भी सकती हैं? उसके पहले पौध-उपनिवेशियों के साज़-सामान की
भविष्यवाणी करो।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:how-plants-spread:12}
मीठा गूदा बनाना \omterm{def:g1:plants-around-us:plant}{पौधे} के लिए महँगा है।
\cref{ex:g6:how-plants-spread:whyleave} और \omterm{def:g1:animals-around-us:animal}{जंतु} के भीतर वाली सेवा की
मदद से समझाओ कि चेरी का \omterm{def:g1:plants-around-us:tree}{पेड़} उस सारी चीनी से ख़रीदता क्या है --- और
\omterm{def:g2:from-seed-to-plant:seed}{बीजों} के सख़्त छिलके उसी सौदे का दूसरा आधा हिस्सा क्यों हैं।
\end{exercise}

\section{समस्या: छोड़ा हुआ खेत}

\begin{problem}[{सप्ताहांत समस्या --- उपनिवेशन के दस साल, एक ही फाटक
से देखे हुए}]\label{pb:g6:how-plants-spread:1}
जुते हुए एक खेत को पतझड़ में छोड़ दिया जाता है। एक प्रकृतिविद दस साल तक
हर जून उसकी तस्वीर फाटक से खींचता है और नोट रखता है।

\medskip\noindent\textbf{भाग I --- पहला साल।}
\begin{enumerate}
  \item पहले साल का जून: नंगी मिट्टी पर जगह-जगह पोस्ते, कुकरौंधे और
        ऊँटकटारे। हर एक के लिए संभावित स्रोत और रास्ता बताओ ---
        पोस्ते ख़ुद मिट्टी से, बाक़ी हवा से। हवा से आने वालों को
        कौन-सा साज़-सामान संभव बनाता है?
  \item पहले साल खेत के बीचोंबीच कोई ब्लैकबेरी नहीं उगती, पर बाड़ के
        पुराने खंभों के नीचे तीन \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} खड़े हैं। \omterm{def:g1:animals-around-us:animal}{जंतु} के भीतर वाली सेवा
        से इस बनावट को समझाओ --- ठीक खंभों के नीचे ही क्यों?
  \item प्रकृतिविद को कहीं कोई \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} नहीं मिलता, हालाँकि हवा की दिशा में
        उनसे भरा एक जंगल है। \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} के \omterm{def:g4:plants-without-seeds:spore}{बीजाणु} ज़रूर पहुँचे होंगे
        (\cref{ex:g4:plants-without-seeds:fern} उनके लाखों गिनता है)।
        \cref{prop:g4:plants-without-seeds:damp} के अनुसार खुले खेत की
        कौन-सी परिस्थिति ने उन्हें ठुकरा दिया?
  \item दूब खेत के किनारे से ही भीतर घुसती है, साल में एक-दो मीटर। यह
        कौन-सी \omterm{prop:g6:how-plants-spread:travel}{उपनिवेशन} व्यवस्था है, और यह कुकरौंधों की तरह खेत के
        बीचोंबीच क्यों नहीं उग सकती?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- उसके बाद के साल।}
\begin{enumerate}[resume]
  \item तीसरे साल तक घासों और ऊँची शाकों ने नंगी ज़मीन ढक दी है, और
        खड़ी वनस्पति से पोस्ते ग़ायब हो चुके हैं। पोस्ते अब कहाँ हैं,
        और किस रूप में? (सड़क-कटान और
        \cref{prop:g6:life-through-seasons:herbs} याद करो।)
  \item चौथे साल से खेत में ब्लैकबेरी की डालियाँ और नागफनी के नन्हे
        \omterm{def:g1:plants-around-us:tree}{पेड़} जगह-जगह दिखते हैं, और सबसे घने बाड़ों के पास तथा खंभों की
        क़तार के नीचे। उन्हें वहाँ पहुँचाने वाली दोनों सेवाओं के नाम
        बताओ, और इस बनावट में मौजूद सबूत भी।
  \item आठवें साल तक बेंत और भोजपत्र के नन्हे \omterm{def:g1:plants-around-us:tree}{पेड़} ब्लैकबेरी से ऊँचे
        खड़े हैं --- दोनों ही \omterm{def:g1:plants-around-us:tree}{पेड़ों} के \omterm{def:g2:from-seed-to-plant:seed}{बीज} नन्हे, कलगीदार या धूल जैसे
        हलके होते हैं। भारी \omterm{def:g2:from-seed-to-plant:seed}{बीजों} वाले बलूत और चेस्टनट अब भी लगभग
        ग़ैरहाज़िर क्यों हैं, और कौन-सा जंतु-हरकारा आख़िरकार उन्हें भी
        बो देता है?
  \item तस्वीरें दिखाती हैं कि खेत के बाशिंदे साल दर साल बदलते रहे,
        जबकि किसी माली ने उसे छुआ तक नहीं। चलाने वाला नियम बताओ: हर
        साल की वनस्पति ख़ुद खेत की \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} में --- सबसे बढ़कर
        ज़मीन के स्तर के \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} में --- क्या बदल देती है, और वह बदलाव
        आगे किसके हक़ में जाता है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- आम सीख।}
\begin{enumerate}[resume]
  \item उपनिवेशियों को आने के क्रम में लगाओ: मिट्टी में इंतज़ार करते
        \omterm{def:g2:from-seed-to-plant:seed}{बीज}, हवा के सवार, \omterm{def:g1:animals-around-us:animal}{जंतुओं} के ढोए झाड़-झंखाड़, और भारी \omterm{def:g2:from-seed-to-plant:seed}{बीजों} वाले
        \omterm{def:g1:plants-around-us:tree}{पेड़}। हर लहर को उस \omterm{def:g6:how-plants-spread:dispersal}{प्रकीर्णन} सेवा से जोड़ो जिसने उसका समय तय
        किया।
  \item क़स्बे के चौक की पगडंडी की दरारों में भी वही पहले साल वाले
        बाशिंदे उगते हैं --- पोस्ते, कुकरौंधे --- पर वे कभी चौथे साल
        के झाड़-झंखाड़ तक नहीं पहुँचते। वहाँ यह क्रम रुक क्यों जाता है,
        दो कारण बताओ।
  \item प्रकृतिविद दावा करता है: ``अपने हाल पर छोड़ दिया जाए तो यह
        खेत जंगल बन जाएगा।'' दस साल की प्रवृत्ति से इस दावे का बचाव
        करो --- और \cref{prop:g5:people-and-nature:managed} वाला
        मैदान का अपवाद भी बताओ, जो साबित करता है कि ``अपने हाल पर
        छोड़ दिया जाए'' ही असली शर्त है।
  \item पूरी समस्या का सार दो वाक्यों में लिखो, और उनमें
        \emph{\omterm{def:g6:how-plants-spread:dispersal}{प्रकीर्णन}}, \emph{\omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ}} तथा \emph{\omterm{prop:g6:how-plants-spread:travel}{उपनिवेशन}}
        शब्द इस्तेमाल करो।
\end{enumerate}
\end{problem}
