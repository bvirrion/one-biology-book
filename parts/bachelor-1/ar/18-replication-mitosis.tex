\chapter{تضاعف DNA والتقسم الخيطي}\label{ch:b1:replication-mitosis}

كل ثماني ساعات أو نحوها تنسخ \omterm{def:b1:cell-unit-of-life:cell}{خلية} من بطانة المعي ستة مليارات \omterm{thm:b1:nucleic-acids:helix}{زوج
قواعد} بخطأ واحد في المليار تقريبًا، ثم تفرز النسختين إلى وليدتين بحيث
تنال كل منهما ستة وأربعين صبغيًا بالضبط --- لا خمسة وأربعين ولا سبعة
وأربعين. والنسخ تؤديه آلة تقرأ خصلة وتبني متممتها بخمسين حرفًا في
الثانية من عشرات آلاف نقاط البدء في آن؛ والفرز يؤديه \omterm{def:b1:replication-mitosis:mitosis}{مغزل} من
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{الأنيبيبات الدقيقة} يسحب نسختي كل \omterm{def:b1:genomes:genome}{صبغي} بعيدًا إحداهما عن الأخرى بدقة
خطأ واحد في مئة ألف انقسام. يصف هذا الفصل \omterm{def:b1:replication-mitosis:fork}{شوكة التضاعف} وإنزيماتها،
ومشكلة النهايات، \omterm{def:b1:replication-mitosis:cycle}{ودورة الخلية} التي توقّت التضاعف والانقسام، \omterm{def:b1:replication-mitosis:mitosis}{والتقسم
الخيطي}، أي الانقسام نفسه.

\section{التضاعف نصف المحافظ}

\begin{proposition}[كل خصلة قالب]\label{prop:b1:replication-mitosis:semiconservative}
\index{التضاعف نصف المحافظ}
يتضاعف \omterm{def:b1:nucleic-acids:chain}{DNA} تضاعفًا \emph{نصف محافظ}: فتنفصل خصلتا اللولب، وتعمل كل
منهما قالبًا تُبنى عليه خصلة متممة جديدة، فيكون كل جزيء وليد خصلة
قديمة مقترنة بخصلة جديدة.
\end{proposition}
\begin{proof}[شواهد]
نمّى مزلسون وستال (1958؛ المستعاد من كتاب المرحلة الثانوية)
\emph{E. coli} على نتروجين ثقيل، ثم نقلاها إلى نتروجين خفيف، وفصلا
\omterm{def:b1:nucleic-acids:chain}{DNA} بالكثافة: فبعد جيل واحد كان كل \omterm{def:b1:nucleic-acids:chain}{DNA} بكثافة وسطى، وبعد جيلين نصفه
وسطى ونصفه خفيف --- وهذا بالضبط ما يتنبأ به وجود خصلة قديمة وأخرى
جديدة في كل جزيء، ولا يتنبأ به أي مخطط آخر. ووسم كيرنز (1963) \omterm{def:b1:genomes:genome}{صبغيات}
بكتيرية متضاعفة بالتريتيوم وصورها بالتصوير الإشعاعي الذاتي: فكانت
حلقات فيها «فقاعة» تضاعف تتحرك شوكتاها في اتجاهين متعاكسين من منشأ
واحد. ونقّى كورنبرغ (1956) إنزيمًا يركّب \omterm{def:b1:nucleic-acids:chain}{DNA} في أنبوب اختبار من
النوكليوزيدات الثلاثية الفوسفات الأربعة ومن قالب، بتسلسل القالب.
\end{proof}

\begin{definition}[شوكة التضاعف وإنزيماتها]\label{def:b1:replication-mitosis:fork}
\index{شوكة التضاعف}\index{بوليميراز DNA}\index{قطعة أوكازاكي}
يبدأ التضاعف عند \emph{منشأ} --- واحد على \omterm{def:b1:genomes:genome}{الصبغي} البكتيري، وعشرات
الآلاف على \omterm{def:b1:genomes:genome}{مورثوم} حقيقي \omterm{def:b1:eukaryotic-cell:nucleus}{النواة} --- حيث يُفتح اللولب وتبتعد
\emph{شوكتا تضاعف} في اتجاهين متعاكسين. وعند كل شوكة: \emph{حلّالة}
تفك اللولب (بجزيء \omterm{def:b1:water-small-molecules:atp}{ATP} واحد لكل \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد})، و\emph{\omterm{def:b1:proteins:peptide}{بروتينات} رابطة للخصلة
المفردة} تبقي الخصلتين المنفصلتين متباعدتين،
و\emph{طوبوإيزوميراز} أمام الشوكة يخفف اللي الذي يراكمه الفك.
و\emph{بوليميراز \omterm{def:b1:nucleic-acids:chain}{DNA}} لا يضيف \omterm{def:b1:nucleic-acids:nucleotide}{النوكليوتيدات} إلا إلى هيدروكسيل $3'$
في سلسلة قائمة، فتنمو كل خصلة جديدة $5' \to 3'$ ولا تستطيع أي منها
أن تبدأ من عدم: فتضع \emph{بادئة} أولًا \emph{بادئًا} قصيرًا من \omterm{def:b1:nucleic-acids:chain}{RNA}
يمدّه البوليميراز. وعلى القالب المقروء $3' \to 5'$ تنمو الخصلة الجديدة
نموًا متصلًا نحو الشوكة: وهي \emph{الخصلة الرائدة}. وعلى القالب الآخر
لا بد للخصلة الجديدة أن تنمو مبتعدة عن الشوكة، في قطع من
\qtyrange{1000}{2000}{} \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيد} عند البكتيريا و\qtyrange{100}{200}{}
عند \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقيات النواة}، هي \emph{قطع أوكازاكي}، لكل منها بادئها: وهي
\emph{الخصلة المتأخرة}. ويستبدل بوليميراز ثانٍ بالبادئات \omterm{def:b1:nucleic-acids:chain}{DNA} وتلحم
\emph{رابطة} القطع في خصلة واحدة.
\end{definition}

\begin{proposition}[الأمانة]\label{prop:b1:replication-mitosis:fidelity}
\index{التدقيق}
اقتران القواعد وحده كان ليعطي خطأ واحدًا في $10^5$؛ والبوليميراز
\emph{يدقق} --- إذ ينزع نشاط نوكلياز خارجي $3' \to 5'$ نوكليوتيدًا
غير مقترن قبل إضافة التالي --- فيبلغ بالخطأ واحدًا في $10^7$؛ وبعد
مرور الشوكة تجد جملة \emph{إصلاح عدم التطابق} ما بقي من عدم اقتران،
وتميز الخصلة الجديدة، وتصححها: فيصير خطأ واحدًا في $10^9$ إلى
$10^{10}$ \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} لكل تضاعف. \omterm{def:b1:cell-unit-of-life:cell}{والخلية} البشرية التي تنسخ
$\num{6.4e9}$ زوج تحدث حفنة طفرات في كل انقسام؛ والبكتيريا واحدة في
ألف انقسام.
\end{proposition}

\begin{example}[السرعة ونقاط البدء]\label{ex:b1:replication-mitosis:speed}
تتحرك الشوكة البكتيرية بمقدار \qty{1000}{} \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} في الثانية: فتنسخ شوكتان
\qty{4.6}{Mb} في \emph{E. coli} في \qty{40}{min}، وفي وسط غني تنقسم
فيه \omterm{def:b1:cell-unit-of-life:cell}{الخلية} كل عشرين دقيقة تبدأ جولة جديدة قبل انتهاء السابقة. أما
الشوكة \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقية النواة}، وقد أبطأها الصبغين، فتتحرك بمقدار \qty{50}{} في الثانية:
فنسخ \omterm{def:b1:genomes:genome}{صبغي} بشري من \qty{250}{Mb} من منشأ واحد كان ليستغرق شهرًا،
ولذلك يُنسخ \omterm{def:b1:genomes:genome}{المورثوم} من نحو \num{40000} منشأ في ثماني ساعات، بترتيب
ثابت لكل نمط خلوي.
\end{example}

\section{النهايات}

\begin{proposition}[مشكلة تضاعف النهايات والتيلوميراز]\label{prop:b1:replication-mitosis:telomerase}
\index{تيلوميراز}
على \omterm{def:b1:genomes:genome}{صبغي} خطي لا يمكن إكمال الخصلة المتأخرة إلى النهاية تمامًا: فحين
يُنزع البادئ الأخير لا يوجد هيدروكسيل $3'$ أعلى المجرى يُملأ منه
الفراغ، وكان كل تضاعف ليقصّر \omterm{def:b1:genomes:genome}{الصبغي} بمقدار \qtyrange{50}{100}{}
\omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيد}. ولذلك تنهي \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقيات النواة} صبغياتها \emph{بقسيمات طرفية}،
وهي آلاف النسخ من تكرار قصير (TTAGGG عند الفقاريات) لا تحمل
\omterm{def:b1:genomes:gene}{مورّثات}، وتحمل \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الجنسية والجذعية \emph{تيلوميرازًا}، وهو \omterm{def:b1:enzymes:enzyme}{إنزيم}
له قالب \omterm{def:b1:nucleic-acids:chain}{RNA} خاص به يضيف تكرارات إلى النهاية $3'$، فيعيد ما يفقده
التضاعف. أما معظم \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الجسمية فتفتقر إليه: فتقصر قسيماتها الطرفية
عند كل انقسام، وبعد نحو خمسين انقسامًا في المزرعة تكف عن الانقسام.
أما البكتيريا، وصبغياتها حلقية، فلا نهايات لها.
\end{proposition}

\section{دورة الخلية}

\begin{definition}[دورة الخلية]\label{def:b1:replication-mitosis:cycle}
\index{دورة الخلية}\index{الطور البيني}
\emph{لدورة \omterm{def:b1:cell-unit-of-life:cell}{الخلية}} عند \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقية النواة} أربعة أطوار.
\emph{G$_1$} (\omterm{def:b1:eukaryotic-cell:plantcell}{الفجوة} 1): تنمو \omterm{def:b1:cell-unit-of-life:cell}{الخلية} وتصنع \omterm{def:b1:enzymes:enzyme}{إنزيمات} التضاعف؛
\emph{وS} (التركيب): تضاعف \omterm{def:b1:nucleic-acids:chain}{DNA} لها، فيصير كل \omterm{def:b1:genomes:genome}{صبغي}
\emph{صبيغيتين شقيقتين} متطابقتين موصولتين عند \omterm{def:b1:genomes:karyotype}{المريكز}؛
\emph{وG$_2$}: تنمو أكثر وتتحقق من النسخ؛ \emph{وM}: \emph{\omterm{def:b1:replication-mitosis:mitosis}{التقسم
الخيطي}}، أي انقسام \omterm{def:b1:eukaryotic-cell:nucleus}{النواة}، يتبعه \emph{انقسام \omterm{def:b1:cell-unit-of-life:cell}{السيتوبلازم}}. وG$_1$
وS وG$_2$ معًا هي \emph{الطور البيني}، وفيه تكون \omterm{def:b1:genomes:genome}{الصبغيات} ممدودة
فعالة؛ \omterm{def:b1:cell-unit-of-life:cell}{والخلية} التي تكف عن الانقسام تغادر الدورة من G$_1$ إلى حالة
سكون هي G$_0$. \omterm{def:b1:cell-unit-of-life:cell}{والخلية} البشرية المنقسمة تستغرق نحو \qty{24}{h}:
G$_1$ \qty{10}{h}، وS \qty{8}{h}، وG$_2$ \qty{4}{h}، وM
\qty{1}{h}؛ والخميرة \qty{90}{min}؛ وجنين ضفدع مبكر ثلاثين دقيقة بلا
فجوات البتة.
\end{definition}

\begin{proposition}[نقاط التفتيش]\label{prop:b1:replication-mitosis:checkpoints}
\index{نقطة تفتيش}
تقود الدورةَ إلى الأمام عائلةٌ من الكينازات البروتينية، تنشّطها
بالتناوب \omterm{def:b1:proteins:peptide}{بروتينات} تسمى \emph{السيكلينات} ترتفع مستوياتها وتهبط عبر
الدورة، وتُحبس الدورة عند \emph{نقاط تفتيش} حتى تتحقق الشروط: ففي
نهاية G$_1$ لا تلتزم \omterm{def:b1:cell-unit-of-life:cell}{الخلية} بالتضاعف إلا إذا كانت كبيرة بما يكفي،
ومغذاة، ومأمورة بالانقسام، وكان \omterm{def:b1:nucleic-acids:chain}{DNA} لها سليمًا؛ وفي نهاية G$_2$ لا
تدخل \omterm{def:b1:replication-mitosis:mitosis}{التقسم الخيطي} إلا حين يكتمل التضاعف؛ وفي \omterm{def:b1:replication-mitosis:mitosis}{التقسم الخيطي} لا تنفصل
الصبيغيات إلا حين يرتبط كل \omterm{def:b1:genomes:genome}{صبغي} \omterm{def:b1:replication-mitosis:mitosis}{بالمغزل} من الجهتين. والتلف يوقف
الدورة، وإن تعذر إصلاحه أطلق موت \omterm{def:b1:cell-unit-of-life:cell}{الخلية}. أما آلة هذا التحكم الجزيئية
فتنتمي إلى كتاب السنة الثالثة؛ وفشلها هو السرطان.
\end{proposition}

\section{التقسم الخيطي}

\begin{definition}[التقسم الخيطي]\label{def:b1:replication-mitosis:mitosis}
\index{تقسم خيطي}\index{مغزل}\index{صبيغية}
\emph{التقسم الخيطي} يوزع \omterm{def:b1:genomes:genome}{الصبغيات} المتضاعفة بحيث تنال كل نواة وليدة
صبيغية واحدة من كل \omterm{def:b1:genomes:genome}{صبغي} --- أي العدد نفسه \omterm{def:b1:genomes:gene}{والمورّثات} نفسها التي
للأبوية. وأطواره:
\begin{itemize}
  \item \emph{الطور التمهيدي}: تتكثف \omterm{def:b1:genomes:genome}{الصبغيات} في قضبان مرئية، كل منها
        من صبيغيتين شقيقتين تمسكهما معًا على امتداد طولهما \omterm{def:b1:proteins:peptide}{بروتينات}
        التماسك؛ ويتباعد \emph{الجسيمان المركزيان} وينمو بينهما
        \emph{المغزل الخيطي} من \omterm{def:b1:eukaryotic-cell:cytoskeleton}{الأنيبيبات الدقيقة}؛
  \item \emph{الطور التمهيدي المتأخر}: ينهار \omterm{def:b1:eukaryotic-cell:nucleus}{الغلاف النووي}؛ وترتبط
        أنيبيبات المغزل \emph{بالقسيم المركزي} لكل صبيغية، وهو بنية
        بروتينية عند \omterm{def:b1:genomes:karyotype}{المريكز}، من قطبين متقابلين؛
  \item \emph{الطور الاستوائي}: تصطف \omterm{def:b1:genomes:genome}{الصبغيات} عند خط استواء المغزل،
        وكل منها تحت شد من القطبين؛
  \item \emph{الطور الانفصالي}: يُقص التماسك، فتنفصل الصبيغيات
        الشقيقتان وتُسحبان إلى قطبين متقابلين كلما قصرت الأنيبيبات
        (الانفصال أ) وتباعد القطبان (الانفصال ب)؛
  \item \emph{الطور النهائي}: تفكّ \omterm{def:b1:genomes:genome}{الصبغيات} تكثفها، ويعاد تكوين \omterm{def:b1:eukaryotic-cell:nucleus}{غلاف
        نووي} حول كل مجموعة.
\end{itemize}
ثم يقسم \emph{انقسام \omterm{def:b1:cell-unit-of-life:cell}{السيتوبلازم}} السيتوبلازمَ: ففي \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الحيوانية
تقرص \emph{حلقة متقلصة} من الأكتين والميوزين \omterm{def:b1:cell-unit-of-life:cell}{الخلية} نصفين؛ وفي
\omterm{def:b1:cell-unit-of-life:cell}{الخلايا} النباتية تندمج حويصلات من غولجي عند خط الاستواء في
\emph{صفيحة خلوية} تنمو إلى الخارج فتصير جدارًا جديدًا.
\end{definition}

\begin{proposition}[ما يحفظه التقسم الخيطي]\label{prop:b1:replication-mitosis:conserves}
\omterm{def:b1:replication-mitosis:mitosis}{التقسم الخيطي} انقسام يحفظ عدد \omterm{def:b1:genomes:genome}{الصبغيات}: \omterm{def:b1:cell-unit-of-life:cell}{فخلية} فيها 2n صبغيًا (46 عند
الإنسان) تعطي خليتين فيهما 2n. ومحتوى \omterm{def:b1:nucleic-acids:chain}{DNA} ينتقل من $2q$ (بعد التضاعف)
إلى $q$ في كل وليدة؛ وعدد \omterm{def:b1:genomes:genome}{الصبغيات} لا يتغير أبدًا، لأن \omterm{def:b1:genomes:genome}{الصبغي} يُعدّ
بمريكزه، وكل \omterm{def:b1:replication-mitosis:mitosis}{صبيغية}، ما إن تنفصل، تصير صبغيًا. وهكذا تكون كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} في
جسم مطابقة وراثيًا للبيضة التي انحدرت منها، إلا الطفرات التي أدخلها
النسخ. أما الانقسام الذي ينصّف عدد \omterm{def:b1:genomes:genome}{الصبغيات}، وهو الانقسام المنصّف،
فينتمي إلى كتاب السنة الثانية.
\end{proposition}

\begin{method}[قراءة تجربة على دورة الخلية]\label{met:b1:replication-mitosis:reading}
\begin{enumerate}
  \item قِس \omterm{def:b1:nucleic-acids:chain}{DNA} لكل \omterm{def:b1:cell-unit-of-life:cell}{خلية} (بصباغ تألقه متناسب مع \omterm{def:b1:nucleic-acids:chain}{DNA}، \omterm{def:b1:cell-unit-of-life:cell}{خلية} \omterm{def:b1:cell-unit-of-life:cell}{خلية}):
        \omterm{def:b1:cell-unit-of-life:cell}{فالخلايا} عند $q$ في G$_1$، وعند $2q$ في G$_2$ أو M، وبينهما في
        S. والكسور تعطي المدد: فحصة الطور من الدورة هي حصته من
        \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}.
  \item أعطِ نبضة من \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيد} موسوم (بروموديوكسي يوريدين): فلا
        تأخذه إلا \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} في الطور S؛ والكسر الموسوم هو كسر S، وتتبع
        الوسم إلى \omterm{def:b1:replication-mitosis:mitosis}{التقسم الخيطي} يعطي طول G$_2$.
  \item عُدّ صور التقسم في مقطع مصبوغ: أي \emph{دليل التقسم}، وهو كسر
        \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} في M، هو كسر الدورة الذي يشغله M --- أي \qty{4}{\%}
        لدورة \qty{24}{h} بتقسم يستغرق ساعة.
  \item اسدد خطوة (بدواء يفكك بلمرة \omterm{def:b1:eukaryotic-cell:cytoskeleton}{الأنيبيبات الدقيقة} فيوقف \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}
        في الطور الاستوائي؛ أو بدواء يسد تركيب \omterm{def:b1:nucleic-acids:chain}{DNA} فيوقفها في S) وعُدّ
        ما يتراكم.
\end{enumerate}
\end{method}

\begin{example}[التجدد]\label{ex:b1:replication-mitosis:turnover}
تتجدد \omterm{def:b1:body-plans-tissues:epithelium}{ظهارة} المعي الدقيق كل أربعة أيام، والجلد كل شهر، والكريات
الحمراء كل أربعة أشهر من النقي، الذي ينتج مليونين في الثانية؛ وتنقسم
\omterm{def:b1:cell-unit-of-life:cell}{خلية} الكبد مرة في السنة أو نحوها، \omterm{def:b1:body-plans-tissues:nervous}{والعصبون} لا ينقسم أبدًا. \omterm{def:b1:cell-unit-of-life:cell}{وخلايا}
الورم فقدت نقاط التفتيش: فتنقسم حين لا ينبغي، وتحتمل \omterm{def:b1:genomes:genome}{صبغيات} سيئة
الفرز، وتراكم الطفرات التي تتيح لها أن تنقسم أسرع بعد. والمعالجة
الكيميائية تستغل الفرق: فالأدوية التي تسمم \omterm{def:b1:replication-mitosis:mitosis}{المغزل} أو البوليميراز تقتل
أكثر \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} انقسامًا.
\end{example}

\section{مسألة: نسخ مورثوم بشري}

\begin{problem}[{مسألة نهاية الأسبوع --- طور S من ثماني ساعات: الشوكات
موقَّتة، والمناشئ معدودة، والقطع والبادئات محصاة، والأخطاء محسوبة،
وبكتيريا مقارَنة، وصولًا إلى أدنى عدد من المناشئ}]
\label{pb:b1:replication-mitosis:1}
\omterm{def:b1:cell-unit-of-life:cell}{خلية} بشرية تضاعف $\num{6.4e9}$ \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} في طور S مدته \qty{8}{h}.
والشوكة \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقية النواة} تتحرك بسرعة \qty{50}{bp/s}؛ والشوكة
البكتيرية بسرعة \qty{1000}{bp/s}. وقطع أوكازاكي \qty{150}{} نوكليوتيدًا عند
\omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقيات النواة} و\qty{1500}{} عند البكتيريا. وكل \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيد} يُضاف يكلف
مكافئي \omterm{def:b1:water-small-molecules:atp}{ATP} (أي الطليعة الثلاثية الفوسفات). ومعدلات الخطأ لكل \omterm{def:b1:nucleic-acids:nucleotide}{قاعدة}:
$10^{-5}$ عند البوليميراز وحده، و$10^{-7}$ مع التدقيق،
و$10^{-9}$ مع إصلاح عدم التطابق.

\medskip
\noindent\textbf{الجزء الأول --- الشوكات والمناشئ.}
\begin{enumerate}
  \item كم \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} تنسخ شوكة واحدة في \qty{8}{h}؟
  \item المنشأ يرسل شوكتين. فكم \omterm{thm:b1:nucleic-acids:helix}{زوج قواعد} ينسخ منشأ واحد في
        \qty{8}{h}؟
  \item احسب أدنى عدد من المناشئ يلزم لنسخ \omterm{def:b1:genomes:genome}{المورثوم} في \qty{8}{h} إذا
        انطلقت كلها في البداية.
  \item احسب المباعدة الوسطية بين هذه المناشئ على امتداد \omterm{def:b1:nucleic-acids:chain}{DNA}،
        بالكيلو زوج وبالميكرومتر.
  \item وفي الواقع تنطلق المناشئ طوال الطور S ويُستعمل نحو
        \num{40000} منها. احسب المباعدة الوسطية والمسافة الوسطية التي
        تقطعها شوكة فعلًا.
  \item كم كان أكبر \omterm{def:b1:genomes:genome}{صبغي} (\qty{250}{Mb}) ليستغرق في النسخ من منشأ
        واحد في وسطه؟
  \item فسّر لماذا كان عدد المناشئ، لا سرعة الشوكة، هو ما عدله التطور
        ليقصّر الطور S.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- القطع والبادئات.}
\begin{enumerate}[resume]
  \item احسب عدد قطع أوكازاكي المصنوعة في طور S واحد.
  \item احسب عدد بادئات \omterm{def:b1:nucleic-acids:chain}{RNA} الموضوعة، بما فيها واحد لكل خصلة رائدة
        لكل منشأ.
  \item احسب عدد عمليات اللحم اللازمة.
  \item احسب مجموع \omterm{def:b1:nucleic-acids:nucleotide}{النوكليوتيدات} المبلمرة (على خصلتي \omterm{def:b1:genomes:genome}{المورثوم} كله).
  \item احسب مكافئات \omterm{def:b1:water-small-molecules:atp}{ATP} المنفقة على البلمرة، والمعدل الوسطي بوحدة
        \omterm{def:b1:water-small-molecules:atp}{ATP} في الثانية على مدى الطور S.
  \item قارن مع دوران \omterm{def:b1:water-small-molecules:atp}{ATP} في \omterm{def:b1:cell-unit-of-life:cell}{خلية} مستريحة، وهو نحو $10^9$ في الثانية:
        فأي كسر من طاقة \omterm{def:b1:cell-unit-of-life:cell}{الخلية} يذهب إلى البلمرة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الأخطاء.}
\begin{enumerate}[resume]
  \item احسب عدد الأخطاء في طور S واحد عند كل من معدلات الخطأ
        الثلاثة.
  \item التسلسل المشفِّر \omterm{def:b1:genomes:gene}{لمورّثة} \qty{1.5}{kb}. فمع الآلة الكاملة، ما
        احتمال أن تكتسب \omterm{def:b1:genomes:gene}{مورّثة} معينة طفرة في انقسام واحد؟
  \item تخضع \omterm{def:b1:cell-unit-of-life:cell}{خلايا} الشخص لنحو $10^{16}$ انقسام في العمر. فكم طفرة في
        المجموع، عند $10^{-9}$؟ ولماذا يحتمل الجسم هذا؟
  \item طفرة تعطّل إصلاح عدم التطابق. فبأي عامل يرتفع معدل الطفرات،
        وماذا يفعل هذا بخطر السرطان على مدى العمر؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- البكتيريا.}
\emph{E. coli}: \qty{4.6}{Mb}، ومنشأ واحد، وشوكتان.
\begin{enumerate}[resume]
  \item احسب زمن تضاعف \omterm{def:b1:genomes:genome}{الصبغي}.
  \item في وسط غني تنقسم \omterm{def:b1:cell-unit-of-life:cell}{الخلية} كل \qty{20}{min}. فكم جولة تضاعف
        تجري في آن، وكم منشأً تحمل \omterm{def:b1:cell-unit-of-life:cell}{خلية} وليدة؟
  \item احسب عدد قطع أوكازاكي لكل جولة وفي الثانية.
  \item احسب الأخطاء لكل جولة عند $10^{-9}$ وكسر \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الوليدة
        الحاملة لطفرة جديدة.
  \item مزرعة من $10^9$ \omterm{def:b1:cell-unit-of-life:cell}{خلية} في المليلتر تنقسم مرة. فكم طفرة جديدة
        تظهر في مليلتر، وكم منها يصيب \omterm{def:b1:genomes:gene}{مورّثة} معينة من \qty{1}{kb}؟
  \item فسّر لماذا يمكن أن تُوجد مقاومة المضادات الحيوية في أي مزرعة
        كبيرة تقريبًا من سلالة حساسة.
  \item قارن الكائنين: أزواج القواعد، والمناشئ، وسرعة الشوكة، ومدة
        الطور S، والتصميم الذي يتيح نسخ \omterm{def:b1:genomes:genome}{المورثوم} الأكبر في زمن مقارب.
  \item اذكر النتيجة: أدنى عدد من المناشئ لطور S بشري مدته
        \qty{8}{h}، والعدد الحقيقي والمباعدة.
\end{enumerate}
\end{problem}
