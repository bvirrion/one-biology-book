\section*{Chapter \ref{ch:g3:stages-of-life} --- The Stages of Life}

\begin{solution}{exo:g3:stages-of-life:1}
Baby, child, \omterm{def:g3:stages-of-life:stages}{adolescent}, \omterm{def:g3:stages-of-life:stages}{adult}, \omterm{def:g3:stages-of-life:stages}{old age}.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:2}
(a) In childhood, from about age six. (b) At the \omterm{def:g3:stages-of-life:stages}{adult} stage.
(c) At the baby stage --- the first year is the fastest of all.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:3}
The baby's burst and the \omterm{def:g3:stages-of-life:stages}{adolescent}'s burst, with the calm,
steady growth of childhood between them.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:4}
The line goes flat: age keeps advancing along the bottom, but
the height no longer climbs.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:5}
Between $6$ and $8$: from about $\qty{115}{cm}$ to about
$\qty{127}{cm}$ --- roughly $\qty{12}{cm}$ over the two years.
In the first year alone: from about $\qty{50}{cm}$ to about
$\qty{75}{cm}$ --- roughly $\qty{25}{cm}$, twice as much as
those two childhood years together.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:6}
Because a comparison is only fair if the measurements are made
the same way. Shoes one day and socks the next, or a sloppy
slanted book, would create ``growth'' (or hide it) that never
happened.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:7}
Acorn (\omterm{def:g2:from-seed-to-plant:seed}{seed}), \omterm{prop:g2:from-seed-to-plant:cycle}{seedling}, young oak, \omterm{def:g3:stages-of-life:stages}{adult} flowering-and-acorn-bearing
oak, old oak --- and, in time, death.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:8}
Open answer. A nine-year-old is in the child stage; the
\omterm{def:g3:stages-of-life:stages}{adolescent} burst is still ahead.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:9}
Fly (weeks), cat (about a year), human (nearly twenty years),
oak (several tens of years). In all four, the order of stages is
the same and cannot be skipped or run backward: young, \omterm{def:g3:stages-of-life:stages}{adult},
\omterm{def:g3:stages-of-life:stages}{old age}.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:10}
No. At nine he is in childhood's calm stretch --- about
$\qty{6}{cm}$ a \emph{year}, which is half a centimetre a month:
far too little to see between two measurements a month apart.
The chart climbs surely, but slowly; he need only wait a few
months between points.
\end{solution}
