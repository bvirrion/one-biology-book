\section*{Bab \ref{ch:g12:respiration-fermentation} --- Respirasi Sel dan Fermentasi}

\begin{solution}{exo:g12:respiration-fermentation:1}
Sebuah \omterm{def:g10:universal-dna:nucleotide}{nukleotida} dengan tiga fosfat berderet, yaitu mata uang energi
\omterm{def:g10:cells-common-unit:cell}{sel}. Memakainya berarti melepaskan fosfat terakhirnya, membebaskan
sekitar \qty{30}{kJ} per mol untuk menggerakkan sebuah proses, dan
meninggalkan ADP untuk diisi ulang.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:2}
\omterm{prop:g12:respiration-fermentation:glycolysis}{Glikolisis} di \omterm{def:g10:cells-common-unit:cell}{sitoplasma}: 2 \omterm{def:g12:respiration-fermentation:atp}{ATP} dan 2 NAD tereduksi, serta dua piruvat.
\omterm{prop:g12:respiration-fermentation:mitochondrion}{Siklus Krebs} di matriks \omterm{def:g10:cells-common-unit:organelle}{mitokondria}: karbon dioksida, pengangkut yang
dimuati, 2 \omterm{def:g12:respiration-fermentation:atp}{ATP}. Rantai respirasi di membran dalam: pengangkutnya
dibongkar ke atas oksigen, air, dan sekitar 26 \omterm{def:g12:respiration-fermentation:atp}{ATP}.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:3}
Dua membran, dengan membran dalam yang berlipat menjadi krista, yang
mengurung matriksnya. Matriks: \omterm{prop:g12:respiration-fermentation:mitochondrion}{siklus Krebs}. Membran dalam: rantai
respirasi dan pembuatan \omterm{def:g12:respiration-fermentation:atp}{ATP}.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:4}
Ia membongkar NAD tereduksi hasil \omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisis} ke atas piruvat, sehingga
\omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisisnya} dapat berlanjut tanpa oksigen. Hasilnya: hanya 2 \omterm{def:g12:respiration-fermentation:atp}{ATP}
\omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisis}, ditambah laktat atau etanol dan $\mathrm{CO_2}$.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:5}
Sekitar 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} berbanding 2. Respirasi meninggalkan karbon dioksida dan
air; \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} meninggalkan laktat, atau etanol dan karbon dioksida,
yang masih kaya energi.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:6}
Glukosa: tak berpengaruh --- \omterm{def:g10:cells-common-unit:organelle}{mitokondria} tak dapat memakainya.
Piruvat: pemakaian oksigen dimulai --- itulah \omterm{def:g11:enzymes-and-phenotype:enzyme}{substrat} \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya}.
ADP: pemakaiannya dipercepat --- pemakaian oksigen terkopel dengan
pembuatan \omterm{def:g12:respiration-fermentation:atp}{ATP}. Sianida: pemakaiannya berhenti --- rantai respirasi
adalah tempat oksigen dipakai.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:7}
\omterm{prop:g12:respiration-fermentation:glycolysis}{Glikolisis}, yang mengubah glukosa menjadi piruvat, berlangsung di
dalam \omterm{def:g10:cells-common-unit:cell}{sitoplasma}; \omterm{def:g10:cells-common-unit:organelle}{mitokondria} tak punya \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimnya} dan hanya memasukkan
piruvat, bukan glukosa.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:8}
Setiap glukosa memuati dua NAD; \omterm{def:g10:cells-common-unit:cell}{sel} hanya punya stok NAD yang kecil,
dan begitu semuanya bermuatan, \omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisis} tak punya pengangkut untuk
menyerahkan hidrogennya lalu berhenti. \omterm{def:g10:cell-metabolism:fermentation}{Fermentasi} membongkar NAD itu
ke atas piruvat dan menjaga jalurnya tetap berjalan.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:9}
Respirasi: $60/30 = 2$ glukosa per detik. \omterm{def:g10:cell-metabolism:fermentation}{Fermentasi}: $60/2 = 30$
glukosa per detik, lima belas kali lebih banyak.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:10}
Sianida menghambat rantai respirasi, sehingga tak ada elektron yang
mencapai oksigen dan sekitar 26 dari 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} per glukosa hilang
seketika; \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} tak sanggup menutup kekurangannya. Otak dan
jantung memakai \omterm{def:g12:respiration-fermentation:atp}{ATP} paling cepat dan punya cadangan paling kecil,
sehingga keduanya gagal dalam beberapa menit.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:11}
Karbon dioksidanya dilepaskan di dalam matriks \omterm{def:g10:cells-common-unit:organelle}{mitokondria}, oleh
\omterm{prop:g12:respiration-fermentation:mitochondrion}{siklus Krebs} (dan oleh langkah yang memasukkan piruvat ke dalamnya):
tak ada dalam \omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisis}. Airnya dibuat pada akhir rantai respirasi,
ketika elektron dan proton bergabung dengan oksigen.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:12}
Otot itu memperoleh 2 \omterm{def:g12:respiration-fermentation:atp}{ATP} per glukosa yang difermentasikan; membangun
kembali glukosa itu dari laktat berongkos sekitar 6 \omterm{def:g12:respiration-fermentation:atp}{ATP} bagi hati, yang
dibayar dengan merespirasikan bahan bakar lain. Tubuh telah
membelanjakan 6 untuk memperoleh 2: \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} meminjam energi yang
harus dilunasi berbunga.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:13}
Di udara, respirasi memberi 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} per glukosa, sehingga glukosa yang
diperlukan sedikit dan tak ada yang dibelokkan menjadi etanol; bila
tertutup rapat, \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} memberi 2, sehingga glukosa yang terpakai
lima belas kali lebih banyak dan etanol pun dibuat. Adonan mengembang
oleh karbon dioksida, yang dihasilkan kedua jalur itu (respirasi enam
per glukosa, \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} dua); di dalam adonan yang padat oksigennya
toh cepat habis.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:14}
Bahang: energi rantainya, yang tak lagi ditangkap sebagai \omterm{def:g12:respiration-fermentation:atp}{ATP},
dilepaskan langsung. Bayi baru lahir cepat kehilangan panas dan belum
pandai menggigil; lemak cokelat adalah pemanas bawaannya.
\end{solution}

\begin{solution}{exo:g12:respiration-fermentation:15}
Elektron: \omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} mengambilnya dari air lalu menaruhnya di dalam
gula; respirasi mengambilnya dari gula lalu memberikannya kepada
oksigen. Gas: \omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} memakai $\mathrm{CO_2}$ dan melepaskan
$\mathrm{O_2}$; respirasi sebaliknya. Energi: \omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} memasukkan
cahaya dan menyimpannya di dalam gula; respirasi mengeluarkannya dari
gula sebagai \omterm{def:g12:respiration-fermentation:atp}{ATP} dan bahang. \omterm{def:g10:cells-common-unit:organelle}{Organel}: \omterm{def:g12:photosynthesis:chloroplast}{kloroplas} dan \omterm{def:g10:cells-common-unit:organelle}{mitokondria}.
Masing-masing memakai apa yang dihasilkan yang lain: \omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} akan
menghabiskan $\mathrm{CO_2}$-nya dan respirasi akan menghabiskan
$\mathrm{O_2}$ serta makanannya.
\end{solution}

\begin{solution}{pb:g12:respiration-fermentation:1}
\textbf{1.} Sepertiga dari \qty{1000}{W} adalah \qty{300}{W}: \qty{18}{kJ}
per menit, yaitu \qty{0.6}{mol} \omterm{def:g12:respiration-fermentation:atp}{ATP} per menit.

\textbf{2.} Sekitar \qty{300}{g} per menit --- berbanding beberapa gram
yang dimuat: stoknya didaur ulang terus-menerus.

\textbf{3.} \qty{50}{g} setara \qty{0.1}{mol}: setiap molekulnya diisi
ulang sekitar 6 kali semenit.

\textbf{4.} $0.6/30 = \qty{0.02}{mol}$ glukosa per menit (sekitar
\qty{3.6}{g}).

\textbf{5.} $6 \times 0.02 = \qty{0.12}{mol}$ $\mathrm{O_2}$, sekitar
\qty{2.9}{L} per menit --- di bawah maksimumnya \qty{4}{L/min}, dan
sesuai dengan \qty{20}{kJ} per liter oksigen pada
\cref{ch:g10:exercise-and-energy}: \qty{1000}{W} itu dapat ditahan
oleh respirasi.

\textbf{6.} 2 di \omterm{def:g10:cells-common-unit:cell}{sitoplasma}, 28 di \omterm{def:g10:cells-common-unit:organelle}{mitokondria}: sekitar 93\%
\omterm{def:g12:respiration-fermentation:atp}{ATP-nya}.

\textbf{7.} $6 \times 0.02 = \qty{0.12}{mol}$ $\mathrm{CO_2}$ per
menit, semuanya dari \omterm{prop:g12:respiration-fermentation:mitochondrion}{siklus Krebs} dan masuknya piruvat di matriks.

\textbf{8.} $30 \times 30 = \qty{900}{kJ}$ dari \qty{2870}{kJ}, sekitar
sepertiganya; sisanya bahang, yang dibuangnya lewat keringat dan lewat
aliran darah ke kulitnya.

\textbf{9.} \omterm{def:g11:gene-expression:protein}{Protein} rantainya dan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} pembuat \omterm{def:g12:respiration-fermentation:atp}{ATP} duduk di dalam
membran; laju pembuatan \omterm{def:g12:respiration-fermentation:atp}{ATP} sebanding dengan luas yang memuatnya.

\textbf{10.} \omterm{def:g10:sport-and-health:training}{Latihan} menaikkan laju respirasi dalam membuat \omterm{def:g12:respiration-fermentation:atp}{ATP} (dan
bagian lemak yang dapat dibakarnya), jadi juga daya yang dapat
ditahannya tanpa \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}; latihan tak mengubah hasil per glukosa
atau oksigen per glukosanya.

\textbf{11.} $800 \times 30 = \qty{24}{kJ}$: \qty{0.8}{mol} \omterm{def:g12:respiration-fermentation:atp}{ATP}.

\textbf{12.} Respirasi: $400 \times 30 = \qty{12}{kJ}$, yaitu
\qty{0.4}{mol}; \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} harus memasok \qty{0.4}{mol} sisanya.

\textbf{13.} $0.4/2 = \qty{0.2}{mol}$ glukosa, yang meninggalkan \qty{0.4}{mol}
laktat.

\textbf{14.} Respirasi: $1/30$ glukosa per \omterm{def:g12:respiration-fermentation:atp}{ATP}; \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}: $1/2$
--- lima belas kali lebih banyak glukosa. Ototnya menerima hal itu
karena \omterm{def:g12:respiration-fermentation:atp}{ATP} diperlukan sekarang dan \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya} ada; utangnya dilunasi
kemudian.

\textbf{15.} Laktatnya harus dioksidasi atau dibangun kembali menjadi
glukosa, dan itu berongkos \omterm{def:g12:respiration-fermentation:atp}{ATP} yang dibuat respirasi: pemakaian
oksigennya tetap di atas taraf istirahat sampai utangnya lunas.

\textbf{16.} \qty{500}{W} selama sejam setara \qty{1800}{kJ}: $1800/38
\approx \qty{47}{g}$ lemak.

\textbf{17.} Lemak lebih tereduksi --- lebih kaya hidrogen --- sehingga
tiap gramnya memerlukan lebih banyak oksigen untuk teroksidasi penuh;
per liter oksigen ia memberi energi sedikit lebih kecil daripada
glukosa.

\textbf{18.} \omterm{prop:g12:respiration-fermentation:glycolysis}{Glikolisis} dan \omterm{def:g10:cell-metabolism:fermentation}{fermentasinya}: lemak tak dapat
difermentasikan, sehingga lari cepat, yang berjalan dengan \omterm{def:g10:cell-metabolism:fermentation}{fermentasi},
sama sekali tak dapat memakai lemak --- hanya glukosa dan \omterm{def:g10:exercise-and-energy:glycogen}{glikogen}.

\textbf{19.} Hati melepaskan glukosa dari \omterm{def:g10:exercise-and-energy:glycogen}{glikogennya} dan membangun
kembali sebagian dari laktat, sehingga glukosa darahnya bertahan dekat
\qty{1}{g/L} (\cref{ch:g12:glucose-and-diabetes}); pasokan otaknya
terlindungi.

\textbf{20.} Sekitar \qty{0.6}{mol} \omterm{def:g12:respiration-fermentation:atp}{ATP} per menit; sekitar
\qty{3}{L} oksigen per menit membayarnya; rantai respirasi pada
membran dalam \omterm{def:g10:cells-common-unit:organelle}{mitokondria} membuat sekitar sembilan persepuluhnya.
\end{solution}
