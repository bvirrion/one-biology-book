\chapter{La respiración}\label{ch:g4:breathing}

Aguanta la \omterm{def:g4:breathing:movements}{respiración} y ponte a contar. Mucho antes de un minuto, tu
pecho exige, cada vez con más urgencia, volver a moverse. Te puedes
saltar una comida, el \omterm{prop:g1:healthy-habits:needs}{sueño} puede esperar a esta noche --- pero el
aire hace falta \emph{ahora}, a todas horas, unas quince veces por
minuto. ¿Qué hay tan urgente en una bocanada de aire?

\section{Los movimientos respiratorios}

\begin{definition}[Inspirar y espirar]\label{def:g4:breathing:movements}
La \emph{respiración}\index{respiración} es la alternancia sin fin de
dos movimientos:
\begin{itemize}
  \item \emph{tomar aire} (inspirar)\index{inspirar}: el pecho se
        ensancha y el aire entra por la nariz o por la boca;
  \item \emph{soltar aire} (espirar)\index{espirar}: el pecho se
        relaja y el aire vuelve a salir.
\end{itemize}
En reposo, la pareja se repite unas $15$ veces por minuto --- más
deprisa en los bebés, más deprisa con esfuerzo --- sin que tengas que
pensarlo nunca.
\end{definition}

\begin{example}[Mirar los movimientos]\label{ex:g4:breathing:watch}
Las manos planas sobre las \omterm{def:g3:bones-and-muscles:skeleton}{costillas}: al tomar aire, la caja torácica
sube y se ensancha bajo tus palmas; al soltarlo, vuelve a bajar. El
costado de un gato dormido enseña la misma subida y bajada. Los
movimientos no paran nunca --- despierto, dormido, desde el primer
llanto hasta el último día.
\end{example}

\section{Adónde va el aire}

\begin{definition}[Los órganos de la respiración]\label{def:g4:breathing:organs}
El aire que se inspira pasa por la nariz (o la boca) y baja por la
\emph{tráquea}\index{tráquea} --- el tubo firme y con anillos que
puedes notar delante del cuello ---, que se divide en dos \omterm{def:g1:plants-around-us:tree}{ramas}, una
para cada uno de los \emph{pulmones}\index{pulmón}. Los pulmones, dos
órganos
esponjosos que llenan la caja torácica (detrás de las \omterm{def:g3:bones-and-muscles:skeleton}{costillas} de la
\cref{def:g3:bones-and-muscles:skeleton}), terminan en millones de
bolsitas de aire diminutas, cada una envuelta en vasos sanguíneos
finísimos.
\end{definition}

\begin{remark}[Esponjosos a propósito]\label{rem:g4:breathing:sponge}
Millones de bolsitas en vez de dos bolsas sencillas: es otra vez el
truco del \omterm{def:g4:journey-of-food:tract}{intestino delgado}
(\cref{rem:g4:journey-of-food:surface}). Todas esas paredes de
bolsita suman una superficie de encuentro enorme entre el aire y la
\omterm{prop:g4:journey-of-food:absorb}{sangre} --- del tamaño de media pista de tenis, plegada dentro de tu
pecho.
\end{remark}

\section{Para qué sirve respirar}

\begin{proposition}[Tomar oxígeno, devolver dióxido de carbono]\label{prop:g4:breathing:exchange}
El aire es una mezcla de gases, y el cuerpo quiere uno de ellos: el
\emph{oxígeno}\index{oxígeno}. En las bolsitas de los \omterm{def:g4:breathing:organs}{pulmones}, el
oxígeno del aire fresco pasa a la \omterm{prop:g4:journey-of-food:absorb}{sangre}, que lo reparte a todas las
partes del cuerpo; allí se usa, junto con los \omterm{def:g4:journey-of-food:digestion}{nutrientes}, para liberar
la \omterm{prop:g3:balanced-diet:jobs}{energía} de la \cref{prop:g3:balanced-diet:jobs}. El gas de desecho
del cuerpo, el \emph{dióxido de carbono}\index{dióxido de carbono},
hace el viaje de vuelta y se espira. El aire espirado tiene menos
oxígeno y más dióxido de carbono que el inspirado --- esa diferencia
es \emph{para lo que} sirve respirar.
\end{proposition}
\admitted

\begin{example}[Por qué correr te deja sin aliento]\label{ex:g4:breathing:running}
Al esprintar, tus \omterm{def:g1:my-body:parts}{piernas} queman \omterm{prop:g3:balanced-diet:jobs}{energía} deprisa --- así que necesitan
oxígeno deprisa. La \omterm{def:g4:breathing:movements}{respiración} obedece: de unas $15$ respiraciones
tranquilas por minuto a $40$ o más, y profundas, mientras el corazón
(el \cref{ch:g4:heart-and-blood} cuenta su versión) se acelera para
apurar los repartos. El jadeo de después de la carrera es el cuerpo
saldando sus cuentas de oxígeno.
\end{example}

\begin{example}[El espejo empañado]\label{ex:g4:breathing:mirror}
Echa el aliento sobre un espejo frío: vaho. El aire espirado vuelve
cambiado --- calentado, humedecido y, sin que se vea, más pobre en
oxígeno y más rico en dióxido de carbono. El espejo enseña la
humedad; el cambio de gases que no puede enseñar es lo importante.
\end{example}

\begin{method}[Cuidar la respiración]\label{met:g4:breathing:care}
\begin{enumerate}
  \item ventila las habitaciones a diario --- muchos \omterm{def:g4:breathing:organs}{pulmones} en una
        habitación cerrada dan un aire viciado y rico en dióxido de
        carbono;
  \item respira por la nariz cuando el aire esté frío o polvoriento:
        lo calienta, lo humedece y lo filtra;
  \item corre, nada, juega --- unos \omterm{def:g4:breathing:organs}{pulmones} y unos \omterm{def:g3:bones-and-muscles:muscle}{músculos}
        respiratorios ejercitados se hacen más fuertes y más profundos;
  \item mantén el humo lejos de tus \omterm{def:g4:breathing:organs}{pulmones}: ensucia las delicadas
        bolsitas, y estas se limpian solas muy despacio.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:breathing:1}
Nombra los dos movimientos respiratorios y di qué hace el pecho en cada
uno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:2}
Traza el camino de una \omterm{def:g4:breathing:movements}{respiración} desde la nariz hasta las bolsitas de aire.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:3}
¿Cuántas veces por minuto respira más o menos un niño tranquilo en
reposo? ¿Qué le pasa a ese número durante un esprint y por qué?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:4}
¿Qué gas toma el cuerpo del aire y cuál devuelve? ¿Dónde ocurre
exactamente el intercambio?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:5}
¿En qué se diferencia el aire espirado del inspirado? Nombra tres
diferencias.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:6}
¿Por qué están hechos los \omterm{def:g4:breathing:organs}{pulmones} de millones de bolsitas diminutas y
no de dos bolsas sencillas? ¿Qué otro órgano de este curso usa el
mismo truco?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:7}
¿Por qué es mejor respirar por la nariz un día helado o polvoriento?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:8}
Pruébalo: sentado y tranquilo, cuenta tus respiraciones durante un
minuto. Después haz veinte saltos de tijera y vuelve a contar. Da los
dos números y explica el cambio.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:9}
Las comidas pueden esperar horas; la \omterm{def:g4:breathing:movements}{respiración} no puede esperar un
minuto. ¿Qué dice eso sobre cuánto oxígeno guarda el cuerpo, comparado
con sus reservas de comida?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:10}
Una clase se nota ``cargada'' a última hora de la mañana. ¿Qué ha
cambiado en su aire, según la \cref{prop:g4:breathing:exchange}, y qué
receta el \cref{met:g4:breathing:care}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:breathing:11}
La trucha no tiene \omterm{def:g4:breathing:organs}{pulmones}: el agua pasa sobre sus branquias plumosas
y el oxígeno \emph{disuelto en el agua} pasa allí a su \omterm{prop:g4:journey-of-food:absorb}{sangre}. ¿Qué
tarea comparten las branquias y los \omterm{def:g4:breathing:organs}{pulmones}? ¿Y por qué se asfixia
una trucha en el aire, donde hay oxígeno de sobra? (Pista: fuera del
agua, las superficies plumosas de las branquias se aplastan y se pegan
unas a otras.)
\end{exercise}
