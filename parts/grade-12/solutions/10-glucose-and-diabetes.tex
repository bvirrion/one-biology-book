\section*{Chapter \ref{ch:g12:glucose-and-diabetes} --- Blood Glucose and Diabetes}

\begin{solution}{exo:g12:glucose-and-diabetes:1}
About \qty{1}{g/L}, i.e. \qty{5.5}{mmol/L}. Below \qty{0.5}{g/L} the
brain fails; above \qty{1.8}{g/L} glucose passes into the urine (and
years above \qty{1.3}{g/L} damage the vessels).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:2}
\omterm{prop:g12:glucose-and-diabetes:hormones}{Insulin}, from the $\beta$ \omterm{def:g10:cells-common-unit:cell}{cells}, lowers the \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} (uptake and
storage by muscle, fat and liver); \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon}, from the $\alpha$ \omterm{def:g10:cells-common-unit:cell}{cells},
raises it (release of glucose by the liver).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:3}
The liver, as \omterm{def:g10:exercise-and-energy:glycogen}{glycogen} (and it can also make new glucose from \omterm{def:g10:chemistry-of-life:families}{amino
acids} and lactate).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:4}
Type 1: the $\beta$ \omterm{def:g10:cells-common-unit:cell}{cells} are destroyed and \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} is absent. Type 2:
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} is present but the tissues respond to it poorly, and the
$\beta$ \omterm{def:g10:cells-common-unit:cell}{cells} eventually tire.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:5}
Healthy about \qty{0.95}{g/L}; impaired tolerance about \qty{1.7}{g/L}
(between 1.4 and 2); type 2 about \qty{2.7}{g/L} (above 2).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:6}
The liver releases glucose from its \omterm{def:g10:exercise-and-energy:glycogen}{glycogen} at the rate the brain
removes it, under the control of \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon}, so the pool stays near
\qty{5}{g} although its contents are renewed every hour.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:7}
The duct carries digestive \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymes} to the gut; blocking it leaves
\omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} normal, so the pancreas's effect on glucose is not through
digestion. Removing the whole organ removes the \omterm{prop:g12:glucose-and-diabetes:hormones}{islets}: \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} is
made in the pancreas and reaches its targets through the blood, not
the duct.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:8}
\omterm{prop:g12:glucose-and-diabetes:hormones}{Glucagon} acts by making the liver break down its \omterm{def:g10:exercise-and-energy:glycogen}{glycogen}; after a
night's fast the store is there and the \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} rises; after three
days the \omterm{def:g10:exercise-and-energy:glycogen}{glycogen} is exhausted and \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} has nothing to release.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:9}
Above about \qty{1.8}{g/L} the kidneys cannot reabsorb all the glucose
they filter; it passes into the urine and, by osmosis, draws water
with it. The water lost must be replaced: thirst.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:10}
The \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} drives glucose into the muscles and fat \omterm{def:g10:cells-common-unit:cell}{cells} and stops
the liver's release, but no glucose comes in: the \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} falls
below \qty{0.6}{g/L} within an hour or two, with sweating, trembling
and confusion; sugar taken at once corrects it.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:11}
A working muscle takes up glucose from the blood by a route that does
not need \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}, and trained muscle regains sensitivity to \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}
between sessions; the \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} falls during and after exercise, and
the tissues' resistance, the disease's cause, is reduced.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:12}
No \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}: type 1, the \omterm{prop:g12:glucose-and-diabetes:hormones}{islets}' $\beta$ \omterm{def:g10:cells-common-unit:cell}{cells} are destroyed and secrete
nothing. Three times normal \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin}: type 2, the \omterm{prop:g12:glucose-and-diabetes:hormones}{islets} are secreting
at full stretch against tissues that no longer respond.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:13}
Healthy: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} rises to about \qty{1.3}{g/L} at 30--60 minutes and
returns by 2 hours; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} rises with it and falls with it. Type 1
untreated: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} climbs above \qty{2}{g/L} and stays; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} flat
at zero --- nothing removes the glucose. Type 2: \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} rises high
and falls slowly; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} rises higher than normal and stays high ---
much \omterm{def:g11:hormones-and-reproduction:hormone}{hormone}, little effect.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:14}
$5 + 80 = \qty{85}{g}$ in \qty{5}{L}: \qty{17}{g/L}. It never does
because absorption takes an hour or two, and as glucose arrives
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} sends it into liver \omterm{def:g10:exercise-and-energy:glycogen}{glycogen}, muscle \omterm{def:g10:exercise-and-energy:glycogen}{glycogen} and fat, while
the brain and tissues burn some: the pool never holds more than a few
grams above normal.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:15}
Both: a sensor (islet \omterm{def:g10:cells-common-unit:cell}{cells}; hypothalamus and pituitary), a set point,
effectors, and negative \omterm{def:g11:hormones-and-reproduction:hormone}{feedback} that removes the signal. \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glycaemia}
has two opposed \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} acting on the same organs, so the variable
can be pushed both up and down actively; testosterone's loop has one
\omterm{def:g11:hormones-and-reproduction:hormone}{hormone} and corrects mainly by more or less of it. Two opposed
effectors give a faster, tighter control of a variable that changes
quickly in both directions.
\end{solution}

\begin{solution}{pb:g12:glucose-and-diabetes:1}
\textbf{1.} \qty{5}{g}; $1/180 = \qty{5.55}{mmol/L}$.

\textbf{2.} One hour.

\textbf{3.} $100/5 = 20$ hours. After that the liver makes new glucose
from \omterm{def:g10:chemistry-of-life:families}{amino acids} (and lactate), at the cost of the body's \omterm{def:g11:gene-expression:protein}{proteins}.

\textbf{4.} Muscle \omterm{def:g10:cells-common-unit:cell}{cells} lack the \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzyme} that releases free glucose
from \omterm{def:g10:exercise-and-energy:glycogen}{glycogen} into the blood; their \omterm{def:g10:exercise-and-energy:glycogen}{glycogen} is burned on the spot,
for their own contraction.

\textbf{5.} $5 + 60 = \qty{65}{g}$ in \qty{5}{L}: \qty{13}{g/L}.
\omterm{prop:g12:glucose-and-diabetes:hormones}{Insulin} sends the glucose into liver and muscle \omterm{def:g10:exercise-and-energy:glycogen}{glycogen} and into fat,
and the tissues burn some; the \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} peaks around \qty{1.4}{g/L}.

\textbf{6.} Fasting \qty{1.15}{g/L} (below 1.26) but \qty{1.85}{g/L}
at 120 minutes (between 1.4 and 2): impaired tolerance, at the
threshold of \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes}.

\textbf{7.} No: \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} is at twice a healthy peak. The tissues' response
to it is failing --- \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} resistance.

\textbf{8.} Type 2 (or its warning stage). Expected: adult, overweight,
inactive, often with diabetic relatives.

\textbf{9.} \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glycaemia} starting above \qty{1.5}{g/L}, rising above
\num{2.5} and not falling; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} near zero throughout.

\textbf{10.} Weight loss and regular exercise: they restore the
tissues' sensitivity to \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} and let working muscle take up glucose
without it.

\textbf{11.} $90/10 = 9$ units.

\textbf{12.} The dose disposes of \qty{90}{g} but only \qty{45}{g}
arrive: the surplus \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} drives the \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} down by the
equivalent of \qty{45}{g} in the pool and stores --- well below
\qty{0.6}{g/L}: sweating, trembling, confusion; sugar at once.

\textbf{13.} $(5 + 90)/5 = \qty{19}{g/L}$ if nothing were removed; in
reality several \unit{g/L}. Above \qty{1.8}{g/L} the kidneys let
glucose into the urine, with water: abundant urine and thirst.

\textbf{14.} Between meals the liver would otherwise release glucose
unchecked (\omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} acts, nothing opposes it) and the tissues would
burn fat with acid products: a background of \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} is needed for the
basic balance, not only for meals.

\textbf{15.} The meter measures the \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} as the $\beta$ \omterm{def:g10:cells-common-unit:cell}{cell} did;
the pen delivers the \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} it would have secreted. They cannot
replace the continuous, minute-by-minute adjustment: the patient
measures a few times a day and estimates, where the \omterm{def:g10:cells-common-unit:cell}{cell} measured
constantly and responded exactly.

\textbf{16.} Type 2, whose $\beta$ \omterm{def:g10:cells-common-unit:cell}{cells} exist and can be pushed. In
type 1 there are no $\beta$ \omterm{def:g10:cells-common-unit:cell}{cells} to stimulate.

\textbf{17.} The tissues' response to \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} --- the effector side of
the loop: with less fat and active muscle, a normal \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} signal
again produces normal uptake, and the $\beta$ \omterm{def:g10:cells-common-unit:cell}{cells} are relieved.

\textbf{18.} Each is secreted in response to the opposite deviation of
the same variable: a high \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} calls \omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} and silences
\omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon}, a low one the reverse. At \qty{1}{g/L} the islet secretes
both at a low, balanced rate --- its resting state.

\textbf{19.} A low \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycaemia} kills the brain within minutes, a high one
damages slowly over years: the acute danger has several independent
guards, the slow one a single one --- which is why the failure of
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulin} alone is enough to cause a disease, and why \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes} is
common.

\textbf{20.} About \qty{5}{g} of glucose in the blood; the liver's
\omterm{def:g10:exercise-and-energy:glycogen}{glycogen} covers about 20 hours; the patient of Part II has impaired
tolerance at the edge of type 2 \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes}; the dose of Part III is 9
units.
\end{solution}
