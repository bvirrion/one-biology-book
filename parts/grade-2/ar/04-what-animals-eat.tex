\chapter{ماذا تأكل الحيوانات}\label{ch:g2:what-animals-eat}

الأرنب في المرج يقضم العشب طول النهار. والثعلب الذي يعبر المرج
نفسه عند الغروب لا يعنيه العشب البتة --- بل يعنيه الأرنب.
والشحرور يأخذ الديدان صباحًا والكرز عصرًا. كل \omterm{def:g1:animals-around-us:animal}{حيوان} لا بد له أن
يأكل، لكن ليس كل \omterm{def:g1:animals-around-us:animal}{حيوان} يأكل الشيء نفسه.

\section{ثلاثة أصناف من الآكلين}

\begin{definition}[العاشب واللاحم والقارت]\label{def:g2:what-animals-eat:kinds}
تُصنَّف \omterm{def:g1:animals-around-us:animal}{الحيوانات} بحسب ما تأكل:
\begin{itemize}
  \item \emph{العاشب}\index{عاشب} يأكل \omterm{def:g1:plants-around-us:plant}{النباتات}: العشب، \omterm{def:g1:plants-around-us:parts}{والأوراق}،
        \omterm{def:g2:from-seed-to-plant:seed}{والبذور}، والفواكه --- كالأرنب والبقرة والحلزون؛
  \item و\emph{اللاحم}\index{لاحم} يأكل \omterm{def:g1:animals-around-us:animal}{حيوانات} أخرى: كالثعلب
        والقطة والدعسوقة (التي تصطاد مَنّ \omterm{def:g1:plants-around-us:plant}{النبات} الصغير)؛
  \item و\emph{القارت}\index{قارت} يأكل \omterm{def:g1:plants-around-us:plant}{النباتات} \omterm{def:g1:animals-around-us:animal}{والحيوانات} معًا:
        كالشحرور والخنزير البري --- والبشر.
\end{itemize}
\end{definition}

\begin{example}[مرج واحد وثلاث موائد]\label{ex:g2:what-animals-eat:meadow}
في مرج صيفي واحد: الأرنب يقضم العشب (\omterm{def:g2:what-animals-eat:kinds}{عاشب})؛ والثعلب يصطاد الأرنب
(\omterm{def:g2:what-animals-eat:kinds}{لاحم})؛ والشحرور يأخذ دودة الآن وكرزة بعد حين (\omterm{def:g2:what-animals-eat:kinds}{قارت}). ثلاثة
جيران، وثلاث موائد --- ولاحظ أن عشاء الثعلب أكل العشب أولًا.
\end{example}

\begin{remark}[اللواحم ليست أشرارًا]\label{rem:g2:what-animals-eat:fox}
الثعلب ليس ``لئيمًا'' لأنه يأكل الأرانب: فالصيد ببساطة هو طريقة
تغذي \omterm{def:g2:what-animals-eat:kinds}{اللاحم} --- طبيعية كقضم الأرنب. وفي الطبيعة، الأكل وأن يُؤكل
المرء جزء من طريقة عمل الحياة؛ وسيجمع فصل لاحق هذه الوجبات كلها
في سلاسل غذائية.
\end{remark}

\section{قراءة مائدة الحيوان من جسمه}

\begin{proposition}[الأجساد تناسب الموائد]\label{prop:g2:what-animals-eat:match}
يحمل جسم \omterm{def:g1:animals-around-us:animal}{الحيوان} قرائن على غذائه:
\begin{itemize}
  \item العواشب كالأرنب والبقرة لها أضراس عريضة مسطحة لطحن
        \omterm{def:g1:plants-around-us:plant}{النباتات}، وتقضي ساعات طويلة في الأكل؛
  \item واللواحم كالقطة والثعلب لها أسنان مدببة لإمساك الفريسة،
        ومخالب حادة، \omterm{def:g1:the-five-senses:sense}{وعينان} تنظران إلى الأمام للصيد؛
  \item والطيور تُظهر ذلك في المنقار: قصير غليظ لكسر \omterm{def:g2:from-seed-to-plant:seed}{البذور}
        (كالعصفور)، ومعقوف منحنٍ لتمزيق اللحم (كالنسر)، وطويل رفيع
        لالتقاط الحشرات من اللحاء.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[أسنان القطة]\label{ex:g2:what-animals-eat:cat}
راقب قطة تتثاءب: أسنان طويلة مدببة في زاويتي مقدَّم فمها. تلك
أسنان إمساك --- عتاد \omterm{def:g2:what-animals-eat:kinds}{لاحم}. أما البقرة فلا تُظهر منها شيئًا حين
تتثاءب: ففمها كله طواحن عريضة مسطحة للعشب.
\end{example}

\begin{method}[تخمين مائدة]\label{met:g2:what-animals-eat:guess}
لتخمّن ما يأكله \omterm{def:g1:animals-around-us:animal}{حيوان} مجهول:
\begin{enumerate}
  \item انظر إلى الأسنان أو المنقار: أطواحن مسطحة، أم أسنان إمساك
        مدببة، أم كلاهما؟
  \item وانظر إلى المخالب \omterm{def:g1:the-five-senses:sense}{والعينين}: أهي أدوات صيد أم لا؟
  \item وراقبه وهو يتغذى إن استطعت --- وهذه أوثق القرائن كلها؛
  \item ثم سمّه: عاشبًا، أو لاحمًا، أو قارتًا.
\end{enumerate}
\end{method}

\section{الأكل الكافي يومًا بعد يوم}

\begin{proposition}[التغذي عمل يومي]\label{prop:g2:what-animals-eat:daily}
الغذاء وقود الجسم ومادة بنائه، فعلى \omterm{def:g1:animals-around-us:animal}{الحيوانات} أن تأكل مرة بعد
مرة، يومًا بعد يوم. والوقت الذي يستغرقه ذلك يتوقف على المائدة:
العشب يعطي قليلًا في كل لقمة، فترعى البقرة ساعات؛ واللحم غني،
فربما أكل الثعلب وجبة جيدة واحدة ثم استراح.
\end{proposition}
\admitted

\begin{example}[الشتاء يغيّر المائدة]\label{ex:g2:what-animals-eat:winter}
يتغير الغذاء بتغير فصول السنة. فالشحرور يأكل الديدان والحشرات في
الصيف، أما في الشتاء، حين تصلب الأرض، فينتقل إلى التوت. ومائدة
\omterm{def:g2:what-animals-eat:kinds}{القارت} الواسعة عون كبير حين تشتد الأزمنة.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:1}
أعطِ الكلمات الدالة على: \omterm{def:g1:animals-around-us:animal}{حيوان} يأكل \omterm{def:g1:plants-around-us:plant}{النباتات}؛ \omterm{def:g1:animals-around-us:animal}{وحيوان} يأكل \omterm{def:g1:animals-around-us:animal}{حيوانات}
أخرى؛ \omterm{def:g1:animals-around-us:animal}{وحيوان} يأكل الاثنين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:2}
\omterm{def:g2:what-animals-eat:kinds}{عاشب} أم \omterm{def:g2:what-animals-eat:kinds}{لاحم} أم \omterm{def:g2:what-animals-eat:kinds}{قارت}: (أ) البقرة؟ (ب) الثعلب؟ (ج) الشحرور؟ (د)
الحلزون؟ (ه) الإنسان؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:3}
أي الأسنان يحتاج إليها \omterm{def:g2:what-animals-eat:kinds}{العاشب} أكثر: الطواحن المسطحة أم أسنان
الإمساك المدببة؟ ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:4}
اقرن كل منقار بمائدته: قصير غليظ؛ ومعقوف منحنٍ؛ وطويل رفيع.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:5}
الدعسوقة صغيرة جميلة --- وهي لاحمة. فماذا تصطاد؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:6}
لماذا تقضي البقرة ساعات في الأكل أكثر بكثير مما يقضيه الثعلب؟
استعمل \cref{prop:g2:what-animals-eat:daily}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:7}
ماذا يأكل الشحرور في الصيف، وماذا يأكل في الشتاء؟ ولماذا تكون
المائدة الواسعة نافعة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:8}
جرّب: راقب الطيور في متنزه أو عند مِعلف. وسجّل ما يأخذه كل صنف ---
بذورًا، أو خبزًا، أو حشرات، أو توتًا --- وانظر إلى مناقيرها. أتصدق
\cref{prop:g2:what-animals-eat:match}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:9}
تستقبل حديقة \omterm{def:g1:animals-around-us:animal}{حيوان} حيوانًا غامضًا: أسنان أمامية مدببة، ومخالب
حادة، \omterm{def:g1:the-five-senses:sense}{وعينان} تنظران إلى الأمام. فما مائدته المرجحة؟ وأي خطوات
\cref{met:g2:what-animals-eat:guess} استعملت؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:10}
``لولا العشب لما بقيت ثعالب بعد قليل.'' الثعلب لا يأكل العشب
أبدًا --- ومع ذلك فالجملة صحيحة. اشرح ذلك مستعينًا بمرج
\cref{ex:g2:what-animals-eat:meadow}.
\end{exercise}
