\section{الأسنان: صندوق عدة في الفم}

\begin{definition}[أصناف الأسنان الثلاثة]\label{def:g2:teeth-and-eating-well:kinds}
أسنان الإنسان ثلاثة أصناف، لكل واحد عمله:
\begin{enumerate}
  \item \emph{القواطع}\index{قاطعة}، وهي مسطحة حادة في المقدمة،
        \emph{تقطع} --- فتشقّ التفاحة؛
  \item و\emph{الأنياب}\index{ناب}، وهي مدببة في الزوايا،
        \emph{تمزّق} --- فتمسك الطعام الأقسى وتشدّه؛
  \item و\emph{الأضراس}\index{ضرس}، وهي عريضة متعرجة في الخلف،
        \emph{تطحن} --- فتسحق الطعام عجينًا مهيّأً للبلع.
\end{enumerate}
\end{definition}

\begin{example}[عضّة تفاح واحدة وثلاث أدوات]\label{ex:g2:teeth-and-eating-well:apple}
أكل قطعة تفاح يستعمل صندوق العدة كله بالترتيب: \omterm{def:g2:teeth-and-eating-well:kinds}{فالقواطع} تشقّ
العضّة، \omterm{def:g2:teeth-and-eating-well:kinds}{والأنياب} تعين على انتزاعها، واللسان يمررها إلى الخلف نحو
\omterm{def:g2:teeth-and-eating-well:kinds}{الأضراس} التي تطحنها. وعندئذ فقط تبتلع.
\end{example}

\begin{remark}[تذكّر الحيوانات]\label{rem:g2:teeth-and-eating-well:animals}
صندوق عدتك المختلط يحكي قصة: \omterm{def:g2:teeth-and-eating-well:kinds}{قواطع} مسطحة وطواحن \omterm{def:g2:what-animals-eat:kinds}{كالعاشب}، \omterm{def:g2:teeth-and-eating-well:kinds}{وأنياب}
مدببة \omterm{def:g2:what-animals-eat:kinds}{كاللاحم}. إنه فم \omterm{def:g2:what-animals-eat:kinds}{قارت} --- فالبشر مجهّزون \omterm{def:g1:plants-around-us:plant}{للنبات} \emph{واللحم}
معًا، تمامًا كما صنّفنا \cref{ch:g2:what-animals-eat}.
\end{remark}

\section{الأسنان اللبنية والأسنان الدائمة}

\begin{proposition}[طقمان في حياة واحدة]\label{prop:g2:teeth-and-eating-well:sets}
ينبت للإنسان طقمان من الأسنان. فتصل \emph{الأسنان
اللبنية}\index{أسنان لبنية} العشرون ($20$) في السنوات الأولى من
الحياة. ثم تتخلخل من نحو سن السادسة وتسقط واحدة بعد أخرى، تدفعها
من تحتها \emph{الأسنان الدائمة}\index{أسنان دائمة} --- وهي الطقم
الأخير، ويبلغ $32$ سنًّا، وعليها أن تدوم بقية حياتك.
\end{proposition}
\admitted

\begin{example}[القاطعة المتخلخلة]\label{ex:g2:teeth-and-eating-well:wobbly}
تبدأ سن أمامية بالتخلخل في السن نفسها التي يبدأ فيها الأطفال
المدرسة. وتحتها كانت \omterm{def:g2:teeth-and-eating-well:kinds}{قاطعة} دائمة تنمو في هدوء وتذيب \omterm{def:g1:plants-around-us:parts}{جذر} السن
اللبنية، حتى تُفلت السن الصغيرة. والفجوة ليست حادثًا --- بل هي
تبديل.
\end{example}

\begin{remark}[لا طقم ثالث]\label{rem:g2:teeth-and-eating-well:nothird}
بعد الطقم الدائم لن تأتي أسنان جديدة أبدًا. فالسن اللبنية
المفقودة تُخلَف، أما السن الدائمة المفقودة فمفقودة. وهذا هو السبب
الحقيقي لإلحاح الكبار على التنظيف: فصندوق العدة الثاني عليه أن
يدوم سبعين سنة أو أكثر.
\end{remark}

\section{حسن الأكل}

\begin{proposition}[صحن متوازن]\label{prop:g2:teeth-and-eating-well:balanced}
يحتاج الجسم لينمو ويبقى صحيحًا إلى غذاء من كل أسرة، كل يوم:
\begin{itemize}
  \item الفواكه والخضر --- للفيتامينات، في كل وجبة؛
  \item والخبز والحبوب والبطاطا --- لطاقة تدوم؛
  \item \omterm{def:g2:animals-grow-up:born}{والحليب} واللبن والجبن --- لبناء العظام والأسنان؛
  \item واللحم أو السمك أو \omterm{def:g2:animals-grow-up:born}{البيض} --- لبناء الجسم، ومرة أو مرتان في
        اليوم تكفيان؛
  \item والماء --- وهو الشراب الوحيد الذي يحتاج إليه الجسم حقًّا؛
  \item والحلويات والدسم --- للمتعة، بمقادير صغيرة، لا في كل وجبة.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[كيسا وجبة خفيفة]\label{ex:g2:teeth-and-eating-well:snack}
الكيس الأول: تفاحة، وقطعة خبز، وماء. والكيس الثاني: حلوى، ومشروب
غازي. الكيس الثاني طعمه سكر ولا شيء غيره تقريبًا --- والسكر هو
بالضبط ما تولم عليه \omterm{def:g1:healthy-habits:germs}{جراثيم} \cref{ch:g1:healthy-habits}. أما الكيس
الأول فيغذّي الجسم ويعطي الأسنان عملًا شريفًا. خمّن أيهما تحزم
طبيبة الأسنان لأولادها.
\end{example}

\begin{method}[مراجعة مائدة يوم]\label{met:g2:teeth-and-eating-well:check}
خذ وجبات يوم واحد وراجع:
\begin{enumerate}
  \item أفيها فواكه أو خضر في كل وجبة؟
  \item أفيها شيء من أسر البناء --- الألبان، واللحم أو السمك أو
        \omterm{def:g2:animals-grow-up:born}{البيض}؟
  \item أالماء هو شراب اليوم؟
  \item أالحلويات قليلة ومتفرقة؟
\end{enumerate}
فإن غابت أسرة، كانت الوجبة التالية فرصة لدعوتها من جديد.
\end{method}
