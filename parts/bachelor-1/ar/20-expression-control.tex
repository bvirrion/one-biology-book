\chapter{التحكم في التعبير المورّثي}\label{ch:b1:expression-control}

بكتيريا نامية على الغلوكوز تحمل خمسة جزيئات من \omterm{def:b1:enzymes:enzyme}{الإنزيم} الذي يهضم
اللاكتوز؛ أعطها لاكتوزًا فتحمل، في دقائق، خمسة آلاف. \omterm{def:b1:cell-unit-of-life:cell}{وخلية} كبدية
\omterm{def:b1:body-plans-tissues:nervous}{وعصبون} يحملان \omterm{def:b1:genomes:gene}{المورّثات} العشرين ألفًا نفسها ويصنعان مجموعتين من
\omterm{def:b1:proteins:peptide}{البروتينات} مختلفتين اختلافًا يكاد يكون تامًا، ولن يصنع أحدهما ما
يصنعه الآخر أبدًا. ولا شيء في \cref{ch:b1:gene-expression} يفسّر
هذا: فآلة التعبير واحدة لكل \omterm{def:b1:genomes:gene}{مورّثة}. وإنما يختلف الأمر في: هل
تُقرأ \omterm{def:b1:genomes:gene}{المورّثة}، وكم مرة، وإلى متى --- وهي قرارات تُتخذ عند المروّج
\omterm{def:b1:proteins:peptide}{ببروتينات} ترتبط بجزيء \omterm{def:b1:nucleic-acids:chain}{DNA}، وبعده بمصير الرسول \omterm{def:b1:proteins:peptide}{والبروتين}. ويصف هذا الفصل
كيف تشغّل البكتيريا مورّثاتها وتطفئها استجابة لغذائها، وكيف تتحكم
\omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقيات النواة} في مورّثاتها بالصبغين \omterm{def:b1:expression-control:enhancer}{والمعزِّزات} البعيدة وتوليفات
العوامل، وكيف يصنع التحكم نفسه، مطبَّقًا على \omterm{def:b1:genomes:gene}{مورّثات} مختلفة في \omterm{def:b1:cell-unit-of-life:cell}{خلايا}
مختلفة، جسمًا من \omterm{def:b1:genomes:genome}{مورثوم} واحد.

\section{لماذا يُتحكم في المورّثات وأين}

\begin{proposition}[التعبير منظَّم، على كل مستوى]\label{prop:b1:expression-control:levels}
\index{تنظيم مورّثي}\index{مورّثة بنيوية}
بعض \omterm{def:b1:genomes:gene}{المورّثات} \emph{بنيوية} --- أي معبَّر عنها دائمًا، بمستوى ثابت،
لأن نواتجها مطلوبة دائمًا (\omterm{def:b1:proteins:peptide}{بروتينات} \omterm{def:b1:gene-expression:ribosome}{الريبوزوم}، \omterm{def:b1:enzymes:enzyme}{وإنزيمات} تحلل
السكر). وأكثرها \emph{منظَّم}: لا يُعبَّر عنه إلا في بعض الشروط،
وبعض \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}، وبعض اللحظات. ويعمل التحكم عند كل خطوة من مسار
المعلومة --- وأولها بدء الاستنساخ، وهو أرخص موضع للقرار، ولكن أيضًا
عند معالجة الرسول وتصديره وثباته وترجمته، وعند تعديل \omterm{def:b1:proteins:peptide}{البروتين}
وهضمه --- وتتراوح أزمنة الاستجابة من ثوانٍ (\omterm{def:b1:proteins:peptide}{بروتين} يُعدَّل) إلى
دقائق (\omterm{def:b1:genomes:gene}{مورّثة} بكتيرية تُشغَّل) إلى أيام (حالة صبغين تتغير).
\end{proposition}

\begin{example}[صنع الإنزيمات عند الحاجة وحدها]\label{ex:b1:expression-control:economy}
\omterm{def:b1:enzymes:enzyme}{إنزيم} $\beta$-غالاكتوزيداز، الذي يشطر اللاكتوز، رباعي من أربع
سلاسل من \qty{1024}{} بقية: فخمسة آلاف نسخة منه هي \qty{3}{\%} من
\omterm{def:b1:proteins:peptide}{بروتين} \omterm{def:b1:cell-unit-of-life:cell}{الخلية} وتكلف \qty{8e7}{} \omterm{def:b1:water-small-molecules:atp}{ATP} في الجيل. وبكتيريا تصنعها ولا
لاكتوز لتهضمه تنمو أبطأ بنحو واحد في المئة من منافسة لا تصنعها،
وتصير في بضع مئات من الأجيال أقل عددًا. فالتنظيم منتخَب لأن التعبير
باهظ.
\end{example}

\section{أوبرون اللاكتوز}

\begin{definition}[الأوبرون، والكابح، والمُشغِّل]\label{def:b1:expression-control:operon}
\index{أوبرون}\index{كابح}\index{مُشغِّل}\index{أوبرون اللاكتوز}
\emph{الأوبرون} مجموعة من \omterm{def:b1:genomes:gene}{المورّثات} البكتيرية تُستنسخ معًا من مروّج
واحد في رسول واحد، فيُتحكم فيها معًا. و\emph{أوبرون اللاكتوز} في
\emph{E. coli} يضم ثلاث \omterm{def:b1:genomes:gene}{مورّثات} لاستعمال اللاكتوز --- \emph{lacZ}
($\beta$-غالاكتوزيداز)، و\emph{lacY} (نافذة اللاكتوز التي تدخله)،
و\emph{lacA} --- خلف مروّج و\emph{مُشغِّل}، وهو تسلسل قصير يتراكب مع
المروّج. \omterm{def:b1:genomes:gene}{ومورّثة} منفصلة، هي \emph{lacI}، تصنع \emph{كابح
اللاكتوز}، وهو \omterm{def:b1:proteins:peptide}{بروتين} يرتبط بالمُشغِّل ويحجب البوليميراز. واللاكتوز،
المحوَّل في \omterm{def:b1:cell-unit-of-life:cell}{الخلية} إلى ألولاكتوز، هو \emph{المحرِّض}: يرتبط بالكابح
ويغيّر شكله ويجعله يفلت من المُشغِّل. فالأوبرون إذن مطفأ ما لم يكن
اللاكتوز حاضرًا --- أي \emph{تحكم سالب} يرفعه \emph{التحريض}.
\end{definition}

\begin{proposition}[نموذج الأوبرون]\label{prop:b1:expression-control:jacobmonod}
يُتحكم في \omterm{def:b1:genomes:gene}{مورّثات} اللاكتوز \omterm{def:b1:expression-control:operon}{بكابح} منتشر يعمل على موقع مجاور
\omterm{def:b1:genomes:gene}{للمورّثات}: فالتحكم سالب، \omterm{def:b1:expression-control:operon}{والكابح} يعمل \emph{منتشرًا} (من أي مكان في
\omterm{def:b1:cell-unit-of-life:cell}{الخلية})، والمُشغِّل يعمل \emph{موضعيًا} (على \omterm{def:b1:genomes:gene}{المورّثات} المجاورة له
وحدها).
\end{proposition}
\begin{proof}[شواهد]
استدل جاكوب ومونو (1959--1961) من الطافرات. فالسلالات الطافرة في
\emph{lacI} ($I^-$) تصنع \omterm{def:b1:enzymes:enzyme}{الإنزيم} بلاكتوز أو بغيره
(\emph{بنيويًا})؛ وإدخال \omterm{def:b1:genomes:gene}{مورّثة} \emph{lacI} سليمة على نسخة ثانية من
المنطقة يعيد التنظيم، فناتج \omterm{def:b1:genomes:gene}{المورّثة} إذن \omterm{def:b1:expression-control:operon}{كابح} منتشر يفتقر إليه
الطافر. والسلالات الطافرة في المُشغِّل ($O^c$) بنيوية هي أيضًا، لكن
مُشغِّلًا سليمًا ثانيًا على نسخة أخرى لا يعيد التنظيم --- فالمُشغِّل
لا يتحكم إلا في \omterm{def:b1:genomes:gene}{المورّثات} الملتصقة به فيزيائيًا، فهو موقع لا ناتج.
وصنف ثالث ($I^s$) لا يمكن تحريضه البتة، وهو سائد: أي \omterm{def:b1:expression-control:operon}{كابح} لم يعد
يرتبط بالمحرِّض. وفي تجربة باجامو (1959) أطفأت مورّثةُ \emph{lacI}
سليمةٌ نُقلت بالاقتران إلى \omterm{def:b1:cell-unit-of-life:cell}{خلية} $I^-$ تخليقَ \omterm{def:b1:enzymes:enzyme}{الإنزيم} في دقائق:
\omterm{def:b1:expression-control:operon}{فالكابح} يعمل سريعًا، ويعمل على الاستنساخ --- وقد تبيّن لاحقًا أن عمر
النصف للرسول دقائق.
\end{proof}

\begin{definition}[التحكم الموجب: CAP وAMP الحلقي]\label{def:b1:expression-control:cap}
\index{كبت الهدم}\index{نمو ثنائي الطور}
مروّج اللاكتوز ضعيف: فحتى بلا \omterm{def:b1:expression-control:operon}{كابح} لا يبدأ البوليميراز إلا نادرًا ما
لم يرتبط \omterm{def:b1:proteins:peptide}{بروتين} ثانٍ، هو \emph{CAP} (\omterm{def:b1:proteins:peptide}{بروتين} تنشيط الهدم)، أعلى
المجرى منه مباشرة، فيثني \omterm{def:b1:nucleic-acids:chain}{DNA} ويمسك البوليميراز في مكانه. ولا يرتبط
CAP بجزيء \omterm{def:b1:nucleic-acids:chain}{DNA} إلا وهو حامل \emph{AMP الحلقي}، الذي يهبط مستواه في
\omterm{def:b1:cell-unit-of-life:cell}{الخلية} حين يكثر الغلوكوز. \omterm{def:b1:expression-control:operon}{فالأوبرون} إذن لا يشتغل اشتغالًا تامًا إلا
حين يكون اللاكتوز حاضرًا \emph{و}الغلوكوز غائبًا: أي إشارتان،
سالبة وموجبة، تتكاملان عند مروّج واحد. وإذا أُعطي السكران معًا أكلت
\emph{E. coli} الغلوكوز أولًا، ثم توقفت ريثما يرتفع AMP الحلقي
وتُصنع \omterm{def:b1:enzymes:enzyme}{إنزيمات} اللاكتوز، ثم أكلت اللاكتوز --- وهو منحني النمو ذو
الطورين المسمى \emph{النمو الثنائي الطور}، الذي وصفه مونو سنة 1941
وقاده إلى \omterm{def:b1:expression-control:operon}{الأوبرون}.
\end{definition}

\begin{method}[التنبؤ بنمط ظاهري في أوبرون اللاكتوز]\label{met:b1:expression-control:lacphenotype}
\begin{enumerate}
  \item اكتب النمط الوراثي لكل نسخة من المنطقة: $I$ و$P$ و$O$
        و$Z$ ($^+$ سليم، و$^-$ معطل، و$O^c$ مُشغِّل بنيوي،
        و$I^s$ \omterm{def:b1:expression-control:operon}{كابح} فائق).
  \item \omterm{def:b1:expression-control:operon}{الكابح} يعمل منتشرًا: فوجود $I^+$ واحد في أي مكان يكبح كل
        مُشغِّل سليم؛ و$I^s$ يكبح مهما يكن المحرِّض.
  \item والمُشغِّل والمروّج يعملان موضعيًا: فالطافر $O^c$ لا يحرر إلا
        $Z$ الذي على \omterm{def:b1:nucleic-acids:chain}{DNA} نفسه؛ و$P^-$ لا يُسكت إلا مورّثاته هو.
  \item اسأل، مع المحرِّض وبدونه، أتُستنسخ كل \omterm{def:b1:genomes:gene}{مورّثة} $Z^+$ أم لا؛
        فيكون النمط الظاهري محرَّضًا أو بنيويًا أو غير قابل
        للتحريض تبعًا لذلك. ثم أضف الغلوكوز: فبلا AMP الحلقي مع
        CAP يكون الاستنساخ منخفضًا مهما يفعل \omterm{def:b1:expression-control:operon}{الكابح}.
\end{enumerate}
\end{method}

\begin{example}[ثنائيا صيغة جزئيان]\label{ex:b1:expression-control:diploids}
$I^-\,O^+\,Z^+ / I^+\,O^+\,Z^-$: النسخة $I^+$ تصنع كابحًا يرتبط
بالمُشغِّل $O^+$ المجاور \omterm{def:b1:genomes:gene}{للمورّثة} $Z^+$؛ فالنمط محرَّض --- أي إن الطفرة قد
كُمّلت. و$I^+\,O^c\,Z^+ / I^+\,O^+\,Z^-$: \omterm{def:b1:genomes:gene}{فالمورّثة} $Z$ الوحيدة العاملة تقع
خلف مُشغِّل لا يستطيع \omterm{def:b1:expression-control:operon}{الكابح} أن يرتبط به؛ فالنمط بنيوي --- ولا
يغني المُشغِّل السليم في النسخة الأخرى، إذ لا يتحكم إلا في $Z$
المعطوب.
\end{example}

\section{استراتيجيات بكتيرية أخرى}

\begin{proposition}[الكبت والتوهين وعوامل سيغما]\label{prop:b1:expression-control:trp}
\index{أوبرون التربتوفان}\index{توهين}\index{عامل سيغما}
\omterm{def:b1:expression-control:operon}{أوبرونات} التخليق الحيوي يُتحكم فيها بالمعنى المعاكس: ذلك
أن \emph{\omterm{prop:b1:expression-control:trp}{أوبرون التربتوفان}}، وهو خمس \omterm{def:b1:genomes:gene}{مورّثات} لصنع التربتوفان،
يُستنسخ ما لم يكن التربتوفان حاضرًا --- إذ يرتبط \omterm{def:b1:proteins:aminoacid}{الحمض الأميني}
\omterm{def:b1:expression-control:operon}{بكابح} التربتوفان و\emph{يمكّنه} من الارتباط بالمُشغِّل (فهو \omterm{def:b1:expression-control:operon}{كابح}
مساعد). وثمة تحكم ثانٍ أدق، هو \emph{التوهين}، يستعمل اقتران
الاستنساخ بالترجمة في البكتيريا: فبداية الرسول تشفّر ببتيدًا قصيرًا
فيه تربتوفانان متتاليان؛ فإن كثر RNA الناقل المحمّل بالتربتوفان
ترجمه \omterm{def:b1:gene-expression:ribosome}{ريبوزوم} سريعًا وانطوى \omterm{def:b1:nucleic-acids:chain}{RNA} خلفه في دبوس شعر منهٍ يوقف
البوليميراز قبل \omterm{def:b1:genomes:gene}{المورّثات}؛ وإن ندر التربتوفان تعثّر \omterm{def:b1:gene-expression:ribosome}{الريبوزوم} عند
كودوني التربتوفان وانطوى \omterm{def:b1:nucleic-acids:chain}{RNA} انطواءً مختلفًا يدع الاستنساخ يمضي.
وتبدّل البكتيريا كذلك مجموعات كاملة من \omterm{def:b1:genomes:gene}{المورّثات} بتغيير الوحدة
الفرعية $\sigma$ في بوليميرازها: فعامل $\sigma$ الخاص بالصدمة
الحرارية يوجهه إلى مروّجات \omterm{def:b1:genomes:gene}{مورّثات} \omterm{def:b1:proteins:peptide}{البروتينات} المرافقة؛ وعامل $\sigma$ الخاص
بالتبوغ إلى مروّجات تكوين الأبواغ.
\end{proposition}

\section{التحكم في حقيقيات النواة}

\begin{definition}[عوامل الاستنساخ والمعزِّزات]\label{def:b1:expression-control:enhancer}
\index{عامل استنساخ}\index{معزِّز}
في \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقيات النواة} لا يبدأ البوليميراز وحده أبدًا: فالعوامل العامة
عند المروّج (\cref{ch:b1:gene-expression}) تستقدمه، وتحدد السرعةَ
\emph{عوامل استنساخ} مرتبطة بمواقع هي \emph{تسلسلات منظِّمة} --- بعضها قرب
المروّج، وكثير منها في \emph{معزِّزات} قد تقع على بعد آلاف من أزواج
القواعد، أعلى المجرى أو أسفله أو في \omterm{def:b1:genomes:gene}{إنترون}، بأي اتجاه، وتعمل
عن طريق \emph{انثناء} \omterm{def:b1:nucleic-acids:chain}{DNA} حتى تلامس العواملُ المرتبطة بها معقدَ المروّج
عبر \omterm{def:b1:proteins:peptide}{بروتينات} وسيطة. وعامل الاستنساخ نمطي البنية: فيه \emph{نطاق
ارتباط بجزيء \omterm{def:b1:nucleic-acids:chain}{DNA}} يتعرف تسلسلًا قصيرًا (عائلات لولب-دوران-لولب، وإصبع
الزنك، وسحّاب اللوسين)، و\emph{نطاق تنشيط} أو \emph{نطاق كبت}
يستقدم الآلة أو يحجبها. وتُقرأ \omterm{def:b1:genomes:gene}{المورّثة} حين تحضر \emph{التوليفة}
الصحيحة من العوامل: فبضع مئات من العوامل، في توليفات، تتحكم في
عشرين ألف \omterm{def:b1:genomes:gene}{مورّثة}.
\end{definition}

\begin{proposition}[الصبغين جزء من التحكم]\label{prop:b1:expression-control:chromatin}
\index{أستلة الهستونات}\index{مثيلة DNA}
\omterm{def:b1:genomes:gene}{مورّثة} ملفوفة في \omterm{def:b1:genomes:nucleosome}{نويصمات} مرصوصة في صبغين متغاير لا يمكن قراءتها:
فالعوامل لا تصل إليها. والمنشِّطات تستقدم \omterm{def:b1:enzymes:enzyme}{إنزيمات} \emph{تؤستل}
الهستونات فترخي قبضتها على \omterm{def:b1:nucleic-acids:chain}{DNA} وتسم المنطقة بأنها مفتوحة؛ والكوابح
تستقدم \omterm{def:b1:enzymes:enzyme}{إنزيمات} تنزع زمر الأستيل وتضيف وسومًا تضغط الصبغين. وزمر
الميثيل المضافة إلى سيتوزينات مروّج تُسكته إسكاتًا دائمًا، وتُنسخ
الوسوم عند التضاعف، حتى إن نمط \omterm{def:b1:genomes:gene}{المورّثات} المفتوحة والمغلقة في \omterm{def:b1:cell-unit-of-life:cell}{خلية}
ترثه بناتها --- وهي الذاكرة التي تبقى بها بنات \omterm{def:b1:cell-unit-of-life:cell}{الخلية} الكبدية \omterm{def:b1:cell-unit-of-life:cell}{خلايا}
كبدية (ويتناول كتاب السنة الثالثة هذه الآليات \emph{فوق المورّثية}
بعمق).
\end{proposition}
\begin{proof}[شواهد]
في \omterm{def:b1:genomes:genome}{الصبغيات} عديدة الخيوط العملاقة في يرقات الذباب (ألف نسخة من كل
\omterm{def:b1:genomes:genome}{صبغي} جنبًا إلى جنب، مخططة) تظهر \omterm{def:b1:genomes:gene}{المورّثات} التي تُستنسخ
\emph{انتفاخات} حيث انفرد الصبغين؛ وهرمون الإكديزون، الذي يطلق
الانسلاخ، يجعل مجموعة بعينها من الانتفاخات تظهر في دقائق وأخرى بعد
ساعات، في تتابع ثابت --- أي استنساخ يُرى مباشرة، وتبدّله إشارة.
\omterm{def:b1:genomes:gene}{والمورّثات} التي تنقلها إعادة ترتيب صبغية إلى جوار الصبغين المتغاير
تُسكت في بعض \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} دون بعض، وترث بناتها الحالة نفسها.
\end{proof}

يدخل الكورتيزول \omterm{def:b1:cell-unit-of-life:cell}{خلية} كبدية ويرتبط بمستقبله، وهو \omterm{def:b1:expression-control:enhancer}{عامل استنساخ} يُبقى

\omterm{def:b1:nucleic-acids:nucleotide}{قاعدة} حاضر قرب مئة \omterm{def:b1:genomes:gene}{مورّثة} --- منها \omterm{def:b1:genomes:gene}{مورّثات} \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{استحداث السكر}

واستجابتان: والفرق هو أي \omterm{def:b1:genomes:gene}{المورّثات} كانت في المتناول.
\end{example}

\section{بعد الاستنساخ، وصنع الجسم}

\begin{proposition}[التحكم بعد المروّج]\label{prop:b1:expression-control:posttranscriptional}
\index{ثبات RNA الرسول}
تقرر \omterm{def:b1:cell-unit-of-life:prokeuk}{الخلية حقيقية النواة} أيضًا كيف تُشذَّب النسخة (فالتشذيب البديل
يعطي العضلة والدماغ \omterm{def:b1:proteins:peptide}{بروتينات} مختلفة من \omterm{def:b1:genomes:gene}{مورّثة} واحدة)، وهل تُصدَّر،
وكم تدوم (أعمار نصف من دقائق لرسائل \omterm{def:b1:proteins:peptide}{البروتينات} المنظِّمة إلى أيام
لرسالة الهيموغلوبين، تحددها تسلسلات في المناطق غير المترجمة ترتبط
بها \omterm{def:b1:proteins:peptide}{بروتينات} وأنواع \omterm{def:b1:nucleic-acids:chain}{RNA} منظِّمة صغيرة)، وهل تُترجم (\omterm{def:b1:cell-unit-of-life:cell}{فالخلايا} الجائعة
إلى الحديد تحجب ترجمة رسالة الفيريتين \omterm{def:b1:proteins:peptide}{ببروتين} يرتبط بنهايتها $5'$،
وتحرره حين يحضر الحديد). \omterm{def:b1:proteins:peptide}{والبروتين}، بعد صنعه، يُتحكم فيه بالتعديل
والهضم (\cref{ch:b1:enzymes}). وكل مستوى لاحق أسرع وأشد موضعية من
سابقه، وأبهظ منه: فالاستنساخ يقرر ما تستطيع \omterm{def:b1:cell-unit-of-life:cell}{الخلية} فعله، والمستويات
اللاحقة تقرر ما تفعله الآن.
\end{proposition}

\begin{proposition}[التعبير التفاضلي يصنع جسمًا]\label{prop:b1:expression-control:differentiation}
\index{تمايز}
كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} في كائن عديد \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} تحمل \omterm{def:b1:genomes:genome}{المورثوم} نفسه؛ وإنما تختلف
\omterm{def:b1:cell-unit-of-life:cell}{الخلايا} لأنها تعبّر عن مجموعات جزئية مختلفة منه. \omterm{def:b1:genomes:gene}{فمورّثات}
\emph{الصيانة} --- وهي بضعة آلاف للاستقلاب \omterm{def:b1:gene-expression:ribosome}{والريبوزوم} \omterm{def:b1:eukaryotic-cell:cytoskeleton}{والهيكل
الخلوي} --- مشتغلة في كل \omterm{def:b1:cell-unit-of-life:cell}{خلية}؛ وسائرها يُشغَّل في بعض \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} ويُطفأ
في بعضها بتوليفات عوامل الاستنساخ التي تحملها كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} وحالات
الصبغين التي ورثتها. فسلف الكرية الحمراء يشغّل الغلوبين ويطفئ كل
شيء آخر تقريبًا؛ \omterm{def:b1:body-plans-tissues:nervous}{والعصبون} يشغّل قنواته ولا ينقسم بعدها أبدًا.
و\emph{التمايز} هو اكتساب نمط تعبير ثابت، و\emph{النمو} (كتاب
السنة الثانية) هو المسار الذي تضبط به الإشارات بين \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} تلك
الأنماط في مواضعها الصحيحة.
\end{proposition}
\begin{proof}[شواهد]
نواة مأخوذة من \omterm{def:b1:cell-unit-of-life:cell}{خلية} ضفدع متمايزة وموضوعة في بويضة نُزعت نواتها
تستطيع أن توجه نمو شرغوف كامل (غوردون، 1962؛ المستعاد من كتاب
المرحلة الثانوية): \omterm{def:b1:cell-unit-of-life:cell}{فالخلية} المتمايزة لم تفقد \omterm{def:b1:genomes:gene}{مورّثات}، بل أطفأتها
فحسب، وأعاد \omterm{def:b1:cell-unit-of-life:cell}{سيتوبلازم} البويضة ضبط المفاتيح. \omterm{def:b1:proteins:peptide}{وبروتينات} نمط خلوي،
مفصولة على هلام، تختلف عن \omterm{def:b1:proteins:peptide}{بروتينات} نمط آخر؛ ورسائله، بعد تسلسلها،
مجموعة جزئية من \omterm{def:b1:genomes:genome}{المورثوم} خاصة به؛ وحفنة من عوامل الاستنساخ تُدخل
في \omterm{def:b1:cell-unit-of-life:cell}{خلية} جلدية تستطيع إعادة برمجتها إلى \omterm{def:b1:cell-unit-of-life:cell}{خلية} جذعية أو \omterm{def:b1:body-plans-tissues:nervous}{عصبون}.
\end{proof}

\begin{example}[مورّثة تُقرأ في عضوين]\label{ex:b1:expression-control:twoorgans}
\omterm{def:b1:genomes:gene}{مورّثة} \omterm{def:b1:enzymes:enzyme}{الإنزيم} الذي يصنع الغلوكوز من غلوكوز-6-فوسفات يُعبَّر عنها
في الكبد والكلية ولا في أي \omterm{def:b1:body-plans-tissues:tissue}{نسيج} آخر: فمروّجها يحمل مواقع لعوامل لا
تحضر إلا هناك، وفي \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الأخرى يكون مروّجها مُمثيلًا وصبغينها
مغلقًا. والعضلة، وهي تحمل \omterm{def:b1:genomes:gene}{المورّثة} سليمة، لا تستطيع تحرير الغلوكوز
في الدم (\cref{ch:b1:biosyntheses-integration}) --- لا لغياب
\omterm{def:b1:genomes:gene}{المورّثة} بل لأنها لا تُقرأ. \omterm{def:b1:genomes:genome}{فالمورثوم} واحد؛ والتعبير هو \omterm{def:b1:mammal-organization:organ}{العضو}.
\end{example}

\section{مسألة: النمو الثنائي الطور}

\begin{problem}[{مسألة نهاية الأسبوع --- منحني مونو ذو الطورين
يُعاد بناؤه خلية خلية: نفاد الغلوكوز، وتحريض اللاكتوز، وعدّ
الإنزيمات، والتنبؤ بالطافرات وحساب كلفة التنظيم، وختامًا معامل
تحريض $\beta$-غالاكتوزيداز}]
\label{pb:b1:expression-control:1}
تبدأ مزرعة من \emph{E. coli} عند $10^7$ \omterm{def:b1:cell-unit-of-life:cell}{خلية} لكل ميليلتر في وسط
فيه \qty{0.5}{mmol/L} من الغلوكوز و\qty{1.0}{mmol/L} من اللاكتوز.
وعلى الغلوكوز تتضاعف \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} كل \qty{20}{min}، وعلى اللاكتوز كل
\qty{30}{min}؛ وتحتاج \omterm{def:b1:cell-unit-of-life:cell}{الخلية} إلى \qty{1.5e-12}{g} من السكر لكل
انقسام (\qty{180}{g/mol} للغلوكوز؛ واللاكتوز، \qty{342}{g/mol}،
يُحسب غلوكوزين). \omterm{def:b1:cell-unit-of-life:cell}{والخلايا} غير المحرَّضة تحمل 5 جزيئات من
$\beta$-غالاكتوزيداز؛ والمحرَّضة تمامًا \num{5000}. ويستغرق
التحريض \qty{30}{min}. \omterm{def:b1:enzymes:enzyme}{والإنزيم} رباعي من $4\times 1024$ بقية؛
وتكلف كل بقية 4 \omterm{def:b1:water-small-molecules:atp}{ATP}؛ وميزانية \omterm{def:b1:cell-unit-of-life:cell}{الخلية} من \omterm{def:b1:water-small-molecules:atp}{ATP} هي $10^{10}$ لكل
انقسام وبروتينها $\qty{1.5e-13}{g}$ (\qty{110}{Da} لكل بقية).

\medskip
\noindent\textbf{الجزء الأول --- طور الغلوكوز.}
\begin{enumerate}
  \item احسب الغلوكوز المتاح لكل ميليلتر، بالغرامات.
  \item كم انقسامًا لكل ميليلتر يستطيع أن يعيل؟
  \item انطلاقًا من $10^7$ \omterm{def:b1:cell-unit-of-life:cell}{خلية}، كم \omterm{def:b1:cell-unit-of-life:cell}{خلية} تكون حين ينفد الغلوكوز؟
        (الانقسام يعطي \omterm{def:b1:cell-unit-of-life:cell}{خلية} جديدة واحدة.)
  \item كم تضاعفًا هذا، وكم يدوم طور الغلوكوز؟
  \item في أثناء هذا الطور، ما حالة \omterm{def:b1:expression-control:operon}{أوبرون اللاكتوز}، ولماذا،
        بلغة الجزيئات (\omterm{def:b1:expression-control:operon}{الكابح} وCAP)؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- التبديل.}
\begin{enumerate}[resume]
  \item حين ينفد الغلوكوز، ما الذي يرتفع داخل \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} وبماذا
        يرتبط؟
  \item كان اللاكتوز حاضرًا طوال الوقت. فلماذا كان \omterm{def:b1:expression-control:operon}{الأوبرون} مع
        ذلك مطفأً تقريبًا في طور الغلوكوز؟
  \item في أثناء الركود الذي يدوم \qty{30}{min} تصنع كل \omterm{def:b1:cell-unit-of-life:cell}{خلية}
        إنزيماتها \num{5000}. احسب جزيئات \omterm{def:b1:enzymes:enzyme}{الإنزيم} المصنوعة في
        الثانية لكل \omterm{def:b1:cell-unit-of-life:cell}{خلية}، والبقايا في الثانية.
  \item تحمل \omterm{def:b1:cell-unit-of-life:cell}{الخلية} \num{20000} \omterm{def:b1:gene-expression:ribosome}{ريبوزوم} بسرعة \qty{20}{} بقية في
        الثانية. فأي كسر من ترجمتها يُكرَّس \omterm{def:b1:enzymes:enzyme}{لإنزيم} $\beta$-غالاكتوزيداز
        في أثناء الركود؟
  \item احسب كتلة \num{5000} رباعي والكسر الذي تمثله من \omterm{def:b1:proteins:peptide}{بروتين}
        \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.
  \item احسب معامل التحريض.
  \item لماذا تكون النافذة (\emph{lacY}) ضرورية للنمو على
        اللاكتوز ضرورة \omterm{def:b1:enzymes:enzyme}{الإنزيم} نفسه، وماذا يحدث لطافر
        \emph{lacY}$^-$ يُعطى لاكتوزًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- طور اللاكتوز.}
\begin{enumerate}[resume]
  \item احسب اللاكتوز المتاح لكل ميليلتر بمكافئات الغلوكوز،
        بالغرامات.
  \item كم انقسامًا إضافيًا يعيل، وما عدد \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} النهائي؟
  \item كم يدوم طور اللاكتوز؟
  \item ضع الخط الزمني: طور الغلوكوز، والركود، وطور اللاكتوز،
        والمجموع.
  \item فسّر لماذا يطول زمن التضاعف على اللاكتوز (لسببين: أحدهما
        عن خطوة \omterm{def:b1:enzymes:enzyme}{الإنزيم} والآخر عن النافذة).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الطافرات والكلف.}
\begin{enumerate}[resume]
  \item طافر $I^-$: صف منحني نموه على الوسط نفسه ومستوى إنزيمه
        طوال الوقت.
  \item طافر CAP$^-$: صف منحنيه ومستوى إنزيمه.
  \item طافر $O^c$ في وسط فيه لاكتوز وحده: أثمة فرق عن النمط
        البري؟ وفي وسط فيه غلوكوز وحده؟
  \item احسب كلفة \omterm{def:b1:water-small-molecules:atp}{ATP} في الجيل لصنع \num{5000} رباعي، والكسر الذي
        تمثله من ميزانية \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.
  \item إذا أطال هذا الكسر زمن الجيل بالكسر نفسه، فبكم يتخلف معدل
        نمو الطافر $I^-$ عن معدل النمط البري على الغلوكوز؟
  \item انطلاقًا من عددين متساويين، بعد كم جيل على الغلوكوز يصير
        الطافر عُشر الجمهرة؟ ($r^n =
        0.1/0.9$ حيث $r$ نسبة معدلي النمو في الجيل، أي
        $2^{-\delta}$ لتخلف $\delta$ من التضاعفات.)
  \item فسّر لماذا تشيع مع ذلك الطافرات $I^-$ في السلالات
        المخبرية المرباة على اللاكتوز.
  \item اذكر النتيجة: معامل تحريض $\beta$-غالاكتوزيداز، وكلفة
        صنعه بلا داع ككسر من ميزانية \omterm{def:b1:water-small-molecules:atp}{ATP}، والزمن الكلي لتجربة
        \omterm{def:b1:expression-control:cap}{النمو الثنائي الطور}.
\end{enumerate}
\end{problem}
