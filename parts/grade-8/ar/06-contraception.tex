\chapter{منع الحمل}\label{ch:g8:contraception}

رسمت الفصول الأخيرة خريطة آلة ذات قدرة لافتة: تلتقي
خليتان، فيوجد بعد أربعين أسبوعًا إنسان. والإنسان هو \omterm{def:g5:biodiversity:species}{النوع}
الوحيد الذي يفهم آلته --- والفهم يجلب اختيارًا لا يواجهه
الضفدع قط: أنشغّلها أم لا، ومتى. و\emph{\omterm{prop:g8:contraception:principle}{منع الحمل}} هو
العلم التطبيقي لهذا الاختيار، وهو يعمل، في كل طريقة من
طرقه، بقطع خطوة صرت تعرف اسمها.

\section{المبدأ: سمّ الخطوة، واقطع الخطوة}

\begin{proposition}[منع الحمل ينطبق على الآلة]\label{prop:g8:contraception:principle}
يقتضي \omterm{def:g5:human-reproduction:pregnancy}{الحمل} سلسلة غير مقطوعة
(\cref{prop:g8:sexual-reproduction:core}،
\cref{prop:g8:from-egg-to-birth:firstweek}): إنتاج
\omterm{def:g8:sexual-reproduction:gametes}{الأمشاج} --- إيداع \omterm{def:g8:reproductive-systems:male}{النطف} --- السباحة إلى \omterm{def:g8:reproductive-systems:female}{قناة البيض} ---
إطلاق \omterm{def:g4:animal-reproduction:sexual}{بويضة} ولقاؤها --- \omterm{def:g5:flower-to-fruit:steps}{الإخصاب} --- الانغراس.
و\emph{منع الحمل}\index{منع الحمل} هو أي قطع متعمد لهذه
السلسلة. وكل طريقة على رف الصيدلي هجوم على حلقة مسماة
واحدة، وترتيبها بحسب حلقتها هو النظرية كلها:
\begin{itemize}
  \item الطرق \emph{الحاجزة} تمنع لقاء المشيجين؛
  \item الطرق \emph{الهرمونية} تمنع إطلاق \omterm{def:g4:animal-reproduction:sexual}{البويضة} أصلًا؛
  \item وطرق أخرى تجعل اللقاء أو الانغراس يفشل.
\end{itemize}
\end{proposition}
\admitted

\section{الطرق الرئيسة، بحسب حلقتها}

\begin{definition}[الطرق الحاجزة: الواقي الذكري]\label{def:g8:contraception:condom}
\emph{الواقي الذكري}\index{واقي ذكري} --- غلاف رقيق يلبس
على \omterm{def:g8:reproductive-systems:male}{القضيب} (ولنسخته الأنثوية بطانة داخل المهبل) --- يسد
السلسلة عند خطوة الإيداع: فلا يدخل المني جسم الشريك
أبدًا. متوافر على نطاق واسع، ولا يلزم إلا عند الحاجة ---
وهو وحده على الرف بخدمة ثانية: فهو الطريقة الوحيدة التي
تسد أيضًا طريق \emph{العدوى} التي تنتقل بالجنس، وهي موضوع
عالم \cref{ch:g9:microbes-and-infection}. وهذه المهمة
المزدوجة هي سبب بقاء الواقي في الصورة حتى حين تستعمل
طريقة أخرى.
\end{definition}

\begin{definition}[الطرق الهرمونية: حبة منع الحمل]\label{def:g8:contraception:pill}
تستعمل \emph{حبة منع الحمل}\index{حبة منع الحمل} آلة
\cref{def:g8:puberty:hormones} نفسها: تؤخذ يوميًا، فتبقي
هرموناتها دورة \omterm{def:g8:reproductive-systems:female}{المبيضين} في حالة التوقيف (الحدث 4 في
\cref{prop:g8:reproductive-systems:cycle} من غير أي حمل)
--- \emph{لا إباضة، ولا \omterm{def:g4:animal-reproduction:sexual}{بويضة}، ولا شيء يخصب}. يصفها
الطبيب، وهو يلائم بين الطريقة والشخص؛ وهي عالية الوثوقية
حين تؤخذ بانتظام --- ولا تقدم شيئًا البتة ضد العدوى، ومن
هنا شراكة \omterm{def:g8:contraception:condom}{الواقي} الدائمة.
\end{definition}

\begin{example}[الرف، مرتبًا بحسب الحلقة]\label{ex:g8:contraception:shelf}
والطرق الشائعة الأخرى، مبوبة بحسب الحلقة التي تهاجمها:
الغرسات واللصاقات الهرمونية --- مبدأ الحبة بذاكرة أفضل
(أشهر أو سنوات، بلا أخذ يومي)؛ والأجهزة الصغيرة التي
يضعها الطبيب في \omterm{prop:g5:human-reproduction:plan}{الرحم} --- فيفشل اللقاء والانغراس سنوات
متتالية؛ و\emph{حبة الطوارئ}، وهي جرعة هرمونية تؤخذ بعد
الواقعة في غضون أيام، وتعمل أساسًا بتأخير إباضة لم تحدث
بعد --- كابح أخير، متوافر عند الصيدلي، ولا يغني عن طريقة
تستعمل \emph{قبل}. وما لا تفعله هذه الطرق جميعًا: أي شيء
ضد العدوى.
\end{example}

\begin{remark}[ما لا ينفع]\label{rem:g8:contraception:notwork}
طرق ساحة المدرسة، مقيسة بالسلسلة: ``عد الأيام الآمنة''
ينهار على تغير الدورة في
\cref{rem:g8:reproductive-systems:living} --- فتاريخ
\omterm{prop:g8:reproductive-systems:cycle}{الإباضة} لا يعرف مقدمًا، \omterm{def:g8:reproductive-systems:male}{والنطف} تعيش أيامًا وهي تنتظره؛
و``الانسحاب في الوقت'' ينهار على أن \omterm{def:g8:reproductive-systems:male}{النطف} تسافر قبل
الختام. وكلتاهما تفشل تمامًا حيث تتنبأ نظرية الحلقة
المسماة: فهما لا تسدان حلقة سدًا موثوقًا. والأسطورة ليست
طريقة.
\end{remark}

\section{الاختيار والمسؤولية وخريطة المساعدة}

\begin{proposition}[الاختيار حقيقي ومشترك]\label{prop:g8:contraception:choice}
حقائق تفرضها الآلة:
\begin{itemize}
  \item يكفي \omterm{def:g5:human-reproduction:pregnancy}{الحمل} فعل واحد، لا طول علاقة: فالسلسلة لا
        تعد المرات؛
  \item وهي سلسلة \emph{مشتركة}: \omterm{def:g5:first-look-at-cells:cell}{خلايا} الشريكين معًا،
        ومسؤولية الشريكين معًا --- \omterm{def:g8:contraception:condom}{فالواقي} والحبة ليسا
        شأن أحد الجنسين وحده، بل حديثًا بينهما؛
  \item ولا طريقة كاملة إلا الامتناع، وإن كانت الجيدة
        منها تقارب الكمال حين تستعمل تمامًا كما صممت ---
        و``الاستعمال الصحيح'' هو موضع أكثر الإخفاقات في
        الواقع.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[خريطة المساعدة]\label{ex:g8:contraception:help}
عناوين \Cref{ex:g8:puberty:questions}، ممتدة: ممرضات
المدارس وأطباء الأسرة يجيبون عن أسئلة \omterm{prop:g8:contraception:principle}{منع الحمل} في سرية
--- ولو جاءت من قاصر؛ ومراكز تنظيم الأسرة قائمة في كل
منطقة لهذا الغرض بعينه، مجانًا ومن غير حكم على أحد؛
والصيادلة يتولون ساعة حبة الطوارئ. فالنظام مبني كي لا
يضطر أحد إلى أن يشق طريقه في السلسلة على أسطورة ---
خريطة المساعدة جزء من الطريقة.
\end{example}

\begin{method}[تقييم أي طريقة تسمع بها]\label{met:g8:contraception:evaluate}
في أي طريقة مدعاة، اسأل بالترتيب:
\begin{enumerate}
  \item \emph{أي حلقة} من السلسلة تسدها --- أتستطيع أن
        تسميها على الخريطة؟
  \item \emph{وبأي وثوقية}، حين تستعمل كما يستعملها
        الناس فعلًا؟
  \item وهل تقي من \emph{العدوى} أيضًا، أم يجب أن
        يشاركها \omterm{def:g8:contraception:condom}{الواقي}؟
  \item وماذا تطلب --- ذاكرة يومية، أو \omterm{def:g6:classification-kinship:attribute}{وصفة}، أو تركيبًا
        --- وهل يلائم ذلك مستعملها؟
\end{enumerate}
لا حلقة تسمى، فلا طريقة
(\cref{rem:g8:contraception:notwork} يسقط عند السؤال
الأول).
\end{method}

سمّ ثلاثة عناوين على خريطة المساعدة، وما يقدمه كل واحد

\section{مسألة: استشارة تنظيم الأسرة}

\begin{problem}[{مسألة نهاية الأسبوع --- عصر من الأسئلة
في مركز التنظيم}]\label{pb:g8:contraception:1}
عصر (متخيَّل) في حياة مرشدة لتنظيم الأسرة: أربعة زوار،
وأربعة مواقف، وخريطة سلسلة واحدة على الجدار.

\medskip\noindent\textbf{الجزء الأول --- الخريطة على
الجدار.}
\begin{enumerate}
  \item ارسم خريطة الجدار: السلسلة من \omterm{def:g8:sexual-reproduction:gametes}{الأمشاج} إلى
        الانغراس، وعليها نقاط الهجوم الثلاث في
        \cref{prop:g8:contraception:principle}.
  \item يسأل الزائر الأول ``كيف توقف حبة واحدة في اليوم
        أي شيء؟'' أجب بالآلة: بأي لغة تتكلم الحبة، ومع
        من؟
  \item ويتابع الزائر الأول: ``وإن نسيت بعضها؟'' حدد
        الخطر على الخريطة --- ما الذي يستأنف حين تنقطع
        الإشارة الهرمونية؟
  \item لماذا تبقى توصية المرشدة \omterm{def:g8:contraception:condom}{بالواقي} قائمة مهما كان
        اختيار الطريقة الأخرى؟ (المهمة المزدوجة.)
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- المواقف.}
\begin{enumerate}[resume]
  \item الزائرة الثانية، في السابعة عشرة وقلقة، ظنت أن
        تطبيق الأيام الآمنة يغني عن سائر الطرق. صحّح
        الظن برفق: أي حقيقتين بيولوجيتين تهزمان
        التقويم؟
  \item الزائر الثالث يريد شيئًا ``بلا ذاكرة يومية''.
        اعرض مرشحي الرف وحلقتيهما.
  \item يصل الزائر الرابع في غضون يوم من تمزق واق. سمّ
        الخيار، وساعته، وآليته الرئيسة --- والطريقة
        الدائمة التي ستناقشها المرشدة لما بعد.
  \item يغادر الأربعة جميعًا بجملة ختامية واحدة عن
        العدوى. اكتبها.
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- المراجعة.}
\begin{enumerate}[resume]
  \item تسأل المتدربة لماذا يعلّم المركز السلسلة بدل أن
        يوزع الطرق فحسب. أجب بالسؤال الأول في
        \cref{met:g8:contraception:evaluate}.
  \item رتّب طرق ذلك العصر بحسب ما تطلبه كل واحدة من
        مستعملها (السؤال الرابع في الطريقة).
  \item وتتساءل المتدربة لماذا كانت زيارات القاصرين سرية
        بالتصميم. أجب بمنطق
        \cref{ex:g8:contraception:help}.
  \item اختم العصر بجملة واحدة: ما الذي يصرفه المركز
        حقًا، وراء العلب.
\end{enumerate}
\end{problem}
