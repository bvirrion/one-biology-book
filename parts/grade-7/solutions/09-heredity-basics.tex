\section*{Chapter \ref{ch:g7:heredity} --- Heredity Basics}

\begin{solution}{exo:g7:heredity:1}
\omterm{def:g7:heredity:heredity}{Heredity}: passing genetic information from parents to \omterm{def:g2:growing:offspring}{offspring}. \omterm{def:g7:heredity:trait}{Trait}:
a \omterm{def:g3:classify:char}{characteristic} of an \omterm{def:g4:ecosystems:levels}{organism}.
\end{solution}

\begin{solution}{exo:g7:heredity:2}
A \omterm{def:g7:heredity:gene}{gene} is a DNA segment carrying information for \omterm{def:g7:heredity:trait}{traits}. In human \omterm{def:g1:my-body:body}{body}
\omterm{def:g7:microscope:cell}{cells} \omterm{def:g7:heredity:gene}{genes} are on \omterm{def:g7:heredity:gene}{chromosomes} in the \omterm{def:g7:microscope:parts}{nucleus}.
\end{solution}

\begin{solution}{exo:g7:heredity:3}
\omterm{def:g7:heredity:pheno}{Genotype}: \omterm{def:g7:heredity:allele}{alleles} present (e.g.\ $Rr$). \omterm{def:g7:heredity:pheno}{Phenotype}: observed \omterm{def:g7:heredity:trait}{trait} (e.g.\
round \omterm{def:g6:reproduction:plant}{seeds} or measured height).
\end{solution}

\begin{solution}{exo:g7:heredity:4}
Different versions of the same \omterm{def:g7:heredity:gene}{gene}.
\end{solution}

\begin{solution}{exo:g7:heredity:5}
Continuous: height, mass. Discrete-ish: some \omterm{def:g7:digestion:blood}{blood} groups, attached vs
free earlobes (with caveats), certain Mendelian school examples.
\end{solution}
