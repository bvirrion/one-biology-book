\chapter{Los animales a lo largo de las estaciones}\label{ch:g3:animals-through-seasons}

En octubre las golondrinas se alinean en los cables y, una mañana, ya
no están. El erizo del seto se enrosca en su nido de \omterm{def:g1:plants-around-us:parts}{hojas} y no se le
verá antes de la primavera. El zorro, en cambio, sale con todas las
nevadas. El invierno es una estación dura --- fría y, sobre todo,
escasa de comida --- y los \omterm{def:g1:animals-around-us:animal}{animales} tienen tres grandes maneras de
atravesarlo.

\section{El invierno, la estación de la escasez}

\begin{proposition}[El doble problema del invierno]\label{prop:g3:animals-through-seasons:problem}
El invierno trae a los \omterm{def:g1:animals-around-us:animal}{animales} dos problemas a la vez:
\begin{enumerate}
  \item el \emph{frío}, que se lleva el calor del cuerpo más deprisa
        --- y mantenerse caliente cuesta \omterm{prop:g3:balanced-diet:jobs}{energía}
        (\cref{prop:g3:balanced-diet:jobs});
  \item la \emph{comida escasa}: las \omterm{def:g1:plants-around-us:plant}{plantas} dejan de crecer, los
        \omterm{prop:g3:sorting-living-things:groups}{insectos} desaparecen, los primeros eslabones de las \omterm{def:g3:food-chains:chain}{cadenas
        alimentarias} del \cref{ch:g3:food-chains} adelgazan --- así
        que la escasez sube hasta todo el mundo.
\end{enumerate}
Más calor necesario, menos comida para pagarlo: ese es el problema del
invierno que todo \omterm{def:g1:animals-around-us:animal}{animal} tiene que resolver.
\end{proposition}
\admitted

\begin{example}[¿Adónde se fue la comida de los insectívoros?]\label{ex:g3:animals-through-seasons:insects}
La golondrina se alimenta de \omterm{prop:g3:sorting-living-things:groups}{insectos} voladores, cazados en pleno
vuelo. En noviembre sencillamente no hay: hace demasiado frío. Ningún
cambio de menú arregla eso para un ave hecha para la caza aérea --- la
despensa de la golondrina desaparece por completo.
\end{example}

\section{Tres respuestas al invierno}

\begin{definition}[Migración]\label{def:g3:animals-through-seasons:migration}
\emph{Migrar}\index{migrar} es viajar, cada año, de una región a otra
según cambian las estaciones --- irse antes del invierno a tierras más
cálidas donde queda comida, y volver en primavera. La golondrina, la
cigüeña y la grulla son \omterm{prop:g3:sorting-living-things:groups}{aves} de
\emph{migración}\index{migración}; algunas viajan miles de kilómetros
y encuentran el camino de vuelta a su mismísimo nido.
\end{definition}

\begin{definition}[Hibernación]\label{def:g3:animals-through-seasons:hibernation}
\emph{Hibernar}\index{hibernar} es pasar el invierno en un \omterm{prop:g1:healthy-habits:needs}{sueño}
profundo y frío, escondido en un refugio: el cuerpo se enfría, el
corazón y la respiración se frenan casi hasta pararse, y el \omterm{def:g1:animals-around-us:animal}{animal}
vive despacio de la grasa acumulada en otoño. El erizo, el lirón, la
marmota y el murciélago hibernan.
\end{definition}

\begin{definition}[Seguir activo]\label{def:g3:animals-through-seasons:active}
Los demás \emph{siguen activos}\index{seguir activo (en invierno)} y
cambian su equipo o sus costumbres: un \emph{pelaje de
invierno}\index{pelaje de invierno} más espeso, \emph{despensas de
alimento}\index{despensa de alimento} reunidas en otoño (las nueces
enterradas de la ardilla), un cambio de menú (el mirlo del
\cref{ex:g2:what-animals-eat:winter}) o sencillamente buscar más ---
el zorro, el ciervo, el petirrojo.
\end{definition}

\begin{example}[Tres vecinos, tres estrategias]\label{ex:g3:animals-through-seasons:three}
Un seto cualquiera en enero: las golondrinas que anidaron encima están
en África, cazando \omterm{prop:g3:sorting-living-things:groups}{insectos} bajo un sol cálido; el erizo de debajo
duerme frío y lento en su nido de \omterm{def:g1:plants-around-us:parts}{hojas}, quemando la grasa de agosto;
el petirrojo de arriba, con su plumaje de invierno ahuecado, patrulla
buscando las últimas bayas. Cuando llegue abril, los tres andarán
otra vez atareados en el mismo seto.
\end{example}

\begin{remark}[Dormir no es pereza, viajar no es turismo]\label{rem:g3:animals-through-seasons:cost}
Cada estrategia tiene su precio. El que migra se arriesga a un viaje
agotador; el que hiberna tiene que reunir en otoño cada gramo de su
grasa de invierno --- un erizo demasiado flaco en noviembre no
despertará en primavera; el \omterm{def:g1:animals-around-us:animal}{animal} activo se juega encontrar comida
durante los meses de más hambre. El invierno se paga por adelantado, o
por el camino, o día a día.
\end{remark}

\begin{method}[Adivinar la estrategia de invierno de un animal]\label{met:g3:animals-through-seasons:guess}
\begin{enumerate}
  \item pregunta qué come y si esa comida existe aquí en invierno;
  \item si la comida desaparece del todo (\omterm{prop:g3:sorting-living-things:groups}{insectos} voladores, néctar):
        espera \omterm{def:g3:animals-through-seasons:migration}{migración} o hibernación;
  \item si la comida se puede encontrar (\omterm{prop:g3:flowers-fruits-seeds:transform}{semillas}, bayas, presas):
        espera un invierno activo, quizá con pelaje más espeso o con
        despensas;
  \item comprueba en otoño: engordar y cavar un nido apunta a
        hibernación; juntarse en bandos apunta a marcharse.
\end{enumerate}
\end{method}

\begin{example}[La primavera lo deshace todo]\label{ex:g3:animals-through-seasons:spring}
En marzo y abril el orden se invierte: vuelven los \omterm{prop:g3:sorting-living-things:groups}{insectos}, los que
migraron reaparecen en sus nidos de siempre, los que hibernaron
despiertan flacos y hambrientos, los \omterm{def:g3:animals-through-seasons:active}{pelajes de invierno} se mudan. Las
estaciones giran, y el calendario de los \omterm{def:g1:animals-around-us:animal}{animales} gira con ellas ---
el otoño que viene, las golondrinas volverán a alinearse en los cables.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:1}
¿Qué doble problema le plantea el invierno a todo \omterm{def:g1:animals-around-us:animal}{animal}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:2}
Nombra las tres estrategias de invierno, con dos \omterm{def:g1:animals-around-us:animal}{animales} de cada una.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:3}
¿Qué le pasa al cuerpo de un erizo que hiberna durante su largo \omterm{prop:g1:healthy-habits:needs}{sueño}?
¿De qué vive?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:4}
¿Por qué tiene que irse exactamente la golondrina --- cuál es su
comida y qué pasa con ella aquí en invierno?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:5}
Da tres maneras que tiene un \omterm{def:g1:animals-around-us:animal}{animal} activo de atravesar el invierno,
cada una con su \omterm{def:g1:animals-around-us:animal}{animal}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:6}
¿Cuál es el precio de cada estrategia, según la
\cref{rem:g3:animals-through-seasons:cost}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:7}
En otoño se ve a un \omterm{def:g1:animals-around-us:animal}{animal} comiendo muchísimo y forrando un agujero
con \omterm{def:g1:plants-around-us:parts}{hojas} secas. ¿Qué estrategia está preparando? ¿Qué paso del
\cref{met:g3:animals-through-seasons:guess} te lo dijo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:8}
Pruébalo: haz una lista de invierno con los pájaros que veas de verdad
en la calle durante una semana fría. ¿Están las golondrinas en ella?
¿Cuáles de tus pájaros eligieron la estrategia activa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:animals-through-seasons:9}
Usa el \cref{met:g3:animals-through-seasons:guess} con: (a) la ardilla
(come nueces y \omterm{prop:g3:flowers-fruits-seeds:transform}{semillas}); (b) la grulla (come ranas e \omterm{prop:g3:sorting-living-things:groups}{insectos} de los
prados húmedos). ¿Qué hace probablemente cada una en invierno?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:animals-through-seasons:10}
Un otoño templado mantiene despierto y cazando hasta tarde a un erizo
joven, pero sigue flaco. ¿Por qué es peligroso para él, según la
\cref{rem:g3:animals-through-seasons:cost}? ¿Qué hacen los centros de
recuperación con los erizos demasiado flacos en otoño?
\end{exercise}
