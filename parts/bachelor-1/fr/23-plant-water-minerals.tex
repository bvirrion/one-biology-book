\chapter{Nutrition hydrique et minérale des plantes}\label{ch:b1:plant-water-minerals}

Un pied de maïs, en juillet, tire un litre d'eau par jour d'un sol qui
paraît sec à la main, et avec lui les trente milligrammes d'azote, sous
forme de nitrate, dont il a besoin pour bâtir la croissance d'une
journée. Il n'a aucune \omterm{def:b1:membranes-transport:transporters}{pompe}. L'eau entre parce que la \omterm{def:b1:flowering-plant-organization:organs}{racine} est plus
sèche que le sol, au sens précis du
\cref{ch:b1:membranes-transport}~; le nitrate entre parce que les
\omterm{def:b1:cell-unit-of-life:cell}{cellules} racinaires dépensent de l'\omterm{def:b1:water-small-molecules:atp}{ATP} pour le faire entrer~; et tous
deux doivent franchir une unique assise de \omterm{def:b1:cell-unit-of-life:cell}{cellules} dont les parois
sont scellées par une \omterm{def:b1:lipids:wax}{cire} avant d'atteindre le \omterm{def:b1:flowering-plant-organization:tissues}{xylème}. Ce chapitre
décrit la \omterm{def:b1:flowering-plant-organization:organs}{racine} comme \omterm{def:b1:mammal-organization:organ}{organe} absorbant, les \omterm{def:b1:membranes-transport:osmosis}{potentiels hydriques} qui
conduisent l'eau du sol au \omterm{def:b1:flowering-plant-organization:tissues}{xylème}, les éléments minéraux dont une
plante a besoin et la façon dont elle les prélève, le cas particulier
de l'azote, et les champignons et bactéries que la plupart des \omterm{def:b1:flowering-plant-organization:organs}{racines}
emploient pour faire une part du travail.

\section{La racine, organe absorbant}

\begin{definition}[Apoplasme, symplasme, endoderme]\label{def:b1:plant-water-minerals:pathways}
\index{apoplasme}\index{symplasme}\index{cadre de Caspary}
L'eau et les solutés traversent le cortex racinaire par deux voies~:
l'\emph{apoplasme}, le système continu des \omterm{prop:b1:carbohydrates:wall}{parois cellulaires} et des
espaces intercellulaires, dans lequel ils circulent sans traverser
aucune membrane~; et le \emph{symplasme}, le \omterm{def:b1:cell-unit-of-life:cell}{cytoplasme} continu des
\omterm{def:b1:cell-unit-of-life:cell}{cellules} reliées par des \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesmes} (\cref{ch:b1:eukaryotic-cell}),
où l'on entre en traversant une membrane plasmique. À la limite interne
du cortex, l'\emph{endoderme}
(\cref{ch:b1:flowering-plant-organization}) barre l'apoplasme~: une
bande de subérine et de lignine, le \emph{cadre de Caspary}, imprègne
ses parois radiales, si bien que tout ce qui entre dans le cylindre
central doit passer par la membrane d'une \omterm{def:b1:cell-unit-of-life:cell}{cellule} endodermique. Les
transporteurs de cette membrane décident donc de ce qui atteint le
\omterm{def:b1:flowering-plant-organization:tissues}{xylème}, et empêchent ce qui y a été chargé de refluer.
\end{definition}

\begin{proposition}[La poussée racinaire]\label{prop:b1:plant-water-minerals:rootpressure}
\index{poussee racinaire@poussée racinaire}\index{guttation}
L'endoderme et les \omterm{def:b1:cell-unit-of-life:cell}{cellules} qu'il entoure pompent des ions dans le
\omterm{def:b1:flowering-plant-organization:tissues}{xylème}, ce qui abaisse le \omterm{def:b1:membranes-transport:osmosis}{potentiel hydrique} du \omterm{def:b1:flowering-plant-organization:tissues}{xylème} et y attire
l'eau du cortex par \omterm{def:b1:membranes-transport:osmosis}{osmose}~; quand la partie aérienne transpire peu (la
nuit, en air humide), l'eau s'accumule et la sève brute est poussée
vers le haut sous une \emph{\omterm{prop:b1:plant-water-minerals:rootpressure}{poussée racinaire}} positive de quelques
dixièmes de mégapascal --- assez pour faire sourdre des gouttelettes à
la pointe des \omterm{def:b1:flowering-plant-organization:organs}{feuilles} des petites plantes (\emph{guttation}) et pour
remplir à nouveau les vaisseaux embolisés, mais pas assez pour hisser
la sève jusqu'à la cime d'un arbre. Le jour, la traction de la
transpiration (\cref{ch:b1:plant-transport}) prend le relais et la
\omterm{prop:b1:plant-water-minerals:rootpressure}{poussée racinaire} disparaît.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Une \omterm{def:b1:flowering-plant-organization:organs}{tige} coupée près du sol laisse exsuder de la sève pendant des
heures depuis sa souche, et un manomètre fixé à la souche enregistre
une pression de \qtyrange{0.1}{0.3}{MPa}~; l'exsudat est plus riche en
ions que la solution du sol, et son écoulement cesse quand on
refroidit les \omterm{def:b1:flowering-plant-organization:organs}{racines} ou qu'on les empoisonne avec un \omterm{def:b1:enzymes:inhibitors}{inhibiteur} de la
respiration~: le flux est entraîné par un chargement actif d'ions, non
par un pompage de l'eau.
\end{proof}

\section{L'eau~: du sol au xylème}

\begin{proposition}[L'échelle des potentiels hydriques]\label{prop:b1:plant-water-minerals:ladder}
\index{potentiel hydrique}
L'eau va du sol au \omterm{def:b1:flowering-plant-organization:tissues}{xylème} parce que le \omterm{def:b1:membranes-transport:osmosis}{potentiel hydrique}
(\cref{ch:b1:membranes-transport}) décroît à chaque étape~:
\begin{center}
\small
\begin{tabular}{lr}
\hline
compartiment & $\Psi$ (\unit{MPa}) \\
\hline
sol humide & $-0.03$ à $-0.3$ \\
sol au point de flétrissement & $-1.5$ \\
\omterm{def:b1:cell-unit-of-life:cell}{cellule} du cortex racinaire & $-0.3$ à $-0.6$ \\
\omterm{def:b1:flowering-plant-organization:tissues}{xylème} racinaire & $-0.5$ à $-0.8$ \\
\omterm{def:b1:cell-unit-of-life:cell}{cellules} foliaires & $-1.0$ à $-2.0$ \\
air à \qty{50}{\%} d'humidité relative & $-95$ \\
\hline
\end{tabular}
\end{center}
Le potentiel du sol est fixé surtout par la tension avec laquelle ses
pores retiennent l'eau (capillarité)~; celui des \omterm{def:b1:cell-unit-of-life:cell}{cellules} par leurs
solutés et leur \omterm{def:b1:eukaryotic-cell:plantcell}{turgescence}~; celui du \omterm{def:b1:flowering-plant-organization:tissues}{xylème} par la tension de la
colonne qui transpire~; celui de l'air par son humidité,
$\Psi = (RT/V_w)\ln(\text{HR})$, qui est énorme même en air humide. La
marche de loin la plus haute est la dernière, de la \omterm{def:b1:flowering-plant-organization:organs}{feuille} à l'air~:
c'est là que presque toute la force motrice se dépense, et c'est
pourquoi les plantes la règlent par les \omterm{def:b1:flowering-plant-organization:tissues}{stomates}.
\end{proposition}

\begin{method}[Lire un profil de potentiel hydrique]\label{met:b1:plant-water-minerals:profile}
\begin{enumerate}
  \item Énumérer les compartiments dans l'ordre avec leur $\Psi$~;
        l'eau ne va que du plus élevé vers le plus bas, donc la suite
        doit décroître le long du trajet, sinon le flux s'inverse.
  \item Décomposer le $\Psi$ de chaque \omterm{def:b1:cell-unit-of-life:cell}{cellule} en $\Psi_s$ (solutés,
        négatif) et $\Psi_p$ (\omterm{def:b1:eukaryotic-cell:plantcell}{turgescence}, positif)~; une \omterm{def:b1:cell-unit-of-life:cell}{cellule}
        racinaire à $\Psi = -0.4$ avec $\Psi_s = -0.9$ a
        $\Psi_p = +0.5$. L'eau du \omterm{def:b1:flowering-plant-organization:tissues}{xylème} a $\Psi_s \approx 0$ et un
        $\Psi_p$ négatif~: elle est sous tension.
  \item Le flux à travers une marche vaut la différence de $\Psi$
        multipliée par la conductance hydraulique de la barrière~;
        l'endoderme et les \omterm{def:b1:flowering-plant-organization:tissues}{stomates} sont les deux marches de faible
        conductance où la plante règle.
  \item Séchez le sol et son $\Psi$ chute~; quand il rejoint celui de la
        \omterm{def:b1:flowering-plant-organization:organs}{feuille}, le flux cesse et la plante flétrit~; quand le $\Psi_p$
        des \omterm{def:b1:cell-unit-of-life:cell}{cellules} foliaires atteint zéro, la \omterm{def:b1:flowering-plant-organization:organs}{feuille} pend. La
        récupération exige que le sol change, non la plante.
\end{enumerate}
\end{method}

\begin{example}[Un litre par jour]\label{ex:b1:plant-water-minerals:litre}
Les \omterm{def:b1:flowering-plant-organization:organs}{racines} fines d'un pied de maïs offrent quelque \qty{0.2}{m^2} de
surface, avec une conductance hydraulique d'environ
\qty{2e-7}{m.s^{-1}.MPa^{-1}}~; entre un sol à \qty{-0.2}{MPa} et un
\omterm{def:b1:flowering-plant-organization:tissues}{xylème} racinaire à \qty{-0.6}{MPa}, le flux vaut
$2\times 10^{-7}\times 0.4\times 0.2 = \qty{1.6e-8}{m^3/s}$ --- un litre
et demi par jour, entièrement perdu par les \omterm{def:b1:flowering-plant-organization:organs}{feuilles}. La \omterm{def:b1:flowering-plant-organization:organs}{racine} ne
fournit aucun travail pour le déplacer~: la transpiration de la \omterm{def:b1:flowering-plant-organization:organs}{feuille}
fixe la tension, et le sol fournit l'eau tant que son potentiel reste
au-dessus de celui de la \omterm{def:b1:flowering-plant-organization:organs}{racine}.
\end{example}

\section{La nutrition minérale}

\begin{definition}[Éléments essentiels]\label{def:b1:plant-water-minerals:elements}
\index{element essentiel@élément essentiel}\index{macroelement@macroélément}\index{oligoelement@oligo-élément}
Un élément est \emph{essentiel} à une plante si celle-ci ne peut
accomplir son cycle de vie sans lui et si aucun autre élément ne peut
le remplacer. Il y en a dix-sept~: le carbone, l'hydrogène et l'oxygène
venus de l'air et de l'eau~; les \emph{macroéléments} prélevés dans le
sol en grammes par kilogramme de masse sèche --- azote (nitrate ou
ammonium), phosphore (phosphate), potassium, calcium, magnésium,
soufre (sulfate)~; et les \emph{oligo-éléments} nécessaires en
milligrammes ou moins --- fer, manganèse, zinc, cuivre, bore,
molybdène, chlore, nickel. L'azote bâtit les \omterm{def:b1:proteins:peptide}{protéines} et les acides
nucléiques~; le phosphore, les \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} et les membranes~; le
potassium est le cation principal de la \omterm{def:b1:cell-unit-of-life:cell}{cellule} et l'agent osmotique
des \omterm{def:b1:flowering-plant-organization:tissues}{stomates}~; le magnésium siège dans la \omterm{def:b1:photosynthesis:pigments}{chlorophylle}~; le fer dans les
cytochromes et la ferrédoxine~; le molybdène dans la \omterm{prop:b1:plant-water-minerals:nitrate}{nitrate réductase}
et la nitrogénase.
\end{definition}

\begin{proposition}[Comment la liste a été établie]\label{prop:b1:plant-water-minerals:hydroponics}
Les \omterm{def:b1:plant-water-minerals:elements}{éléments essentiels} ont été identifiés en cultivant des plantes
dans de l'eau ne contenant que des sels connus.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Sachs et Knop (années 1860) menèrent des plantes jusqu'à maturité dans
des solutions de quelques sels minéraux, sans aucun sol, prouvant que
le sol ne fournit rien d'autre que des minéraux et un support~; en
retirant un sel à la fois, ils obtinrent une maladie caractéristique
pour chaque élément, guérie par son retour. Les \omterm{def:b1:plant-water-minerals:elements}{oligo-éléments} furent
découverts plus tard, à mesure que les chimistes apprirent à purifier
les sels~: une plante privée de molybdène, nécessaire à une part sur
dix millions, ne réduit plus le nitrate. La technique, l'hydroponie,
nourrit aujourd'hui des serres entières.
\end{proof}

\begin{method}[Lire une carence]\label{met:b1:plant-water-minerals:deficiency}
\begin{enumerate}
  \item Demander où le symptôme apparaît. Les éléments que la plante
        peut déplacer dans le phloème (N, P, K, Mg) sont retirés des
        vieilles \omterm{def:b1:flowering-plant-organization:organs}{feuilles} pour nourrir les jeunes~: ce sont les
        vieilles \omterm{def:b1:flowering-plant-organization:organs}{feuilles} qui montrent la carence d'abord. Ceux
        qu'elle ne peut pas déplacer (Ca, Fe, B) restent là où ils ont
        été déposés~: ce sont les jeunes \omterm{def:b1:flowering-plant-organization:organs}{feuilles} et les apex qui la
        montrent.
  \item Demander ce que fait l'élément~: sans azote, peu de
        \omterm{def:b1:photosynthesis:pigments}{chlorophylle} et de \omterm{def:b1:proteins:peptide}{protéines} --- plante pâle et rabougrie~; sans
        magnésium ni fer, la \omterm{def:b1:photosynthesis:pigments}{chlorophylle} manque --- \omterm{def:b1:flowering-plant-organization:organs}{feuilles} jaunes à
        nervures vertes~; sans potassium, les marges des \omterm{def:b1:flowering-plant-organization:organs}{feuilles} se
        dessèchent~; sans calcium, les apex meurent.
  \item Confirmer en fournissant seulement l'élément soupçonné et en
        observant les pousses nouvelles.
\end{enumerate}
\end{method}

\begin{proposition}[L'absorption active des ions]\label{prop:b1:plant-water-minerals:uptake}
\index{pompe a protons@pompe à protons}
La solution du sol contient les ions à des concentrations allant du
micromole à quelques millimoles par litre~; les \omterm{def:b1:cell-unit-of-life:cell}{cellules} racinaires
tiennent le potassium à \qty{100}{mmol/L}, le phosphate à \qty{10}{},
le nitrate à plusieurs. L'absorption se fait donc surtout contre le
gradient et se paie en \omterm{def:b1:water-small-molecules:atp}{ATP}, mais indirectement~: une \emph{\omterm{prop:b1:plant-water-minerals:uptake}{pompe à
protons}} de la membrane plasmique (\cref{ch:b1:membranes-transport})
exporte des $\mathrm{H^+}$, ce qui acidifie l'extérieur et rend
l'intérieur négatif de \qtyrange{-120}{-200}{mV}~; les cations comme
$\mathrm{K^+}$ entrent alors par des canaux en suivant le gradient
électrique, et les anions
($\mathrm{NO_3^-}$, $\mathrm{H_2PO_4^-}$, $\mathrm{SO_4^{2-}}$) par des
\omterm{prop:b1:membranes-transport:secondary}{symporteurs} qui ramènent les protons avec eux. Les transporteurs sont
sélectifs~: le potassium est prélevé cent fois plus volontiers que le
sodium, de même charge et de taille presque égale, et la plante tient
le sodium dehors même en sol salé. Les \omterm{def:b1:flowering-plant-organization:organs}{racines} consacrent le tiers de
la respiration de la plante à ce transport.
\end{proposition}

\section{L'azote}

\begin{proposition}[Du nitrate à l'acide aminé]\label{prop:b1:plant-water-minerals:nitrate}
\index{nitrate reductase@nitrate réductase}\index{assimilation de l'azote}
La plupart des plantes prennent l'azote sous forme de nitrate. Dans le
\omterm{def:b1:eukaryotic-cell:organelle}{cytosol}, la \emph{\omterm{prop:b1:plant-water-minerals:nitrate}{nitrate réductase}}, \omterm{def:b1:enzymes:enzyme}{enzyme} à molybdène, le réduit en
nitrite avec les électrons du NADH~; dans le \omterm{def:b1:eukaryotic-cell:plastid}{plaste}, la \emph{nitrite
réductase} réduit le nitrite en ammonium avec les électrons de la
ferrédoxine~; et l'ammonium est aussitôt fixé en glutamine et \omterm{def:b1:biosyntheses-integration:nitrogen}{glutamate}
(\cref{ch:b1:biosyntheses-integration}) --- l'ammonium lui-même est
toxique et ne s'accumule jamais. La réduction complète coûte huit
électrons par azote, un dixième du flux d'électrons photosynthétique
d'une \omterm{def:b1:flowering-plant-organization:organs}{feuille} sur un sol riche en nitrate, et la \omterm{def:b1:flowering-plant-organization:organs}{feuille} en fait une
bonne part à la lumière, directement à partir de la ferrédoxine du
chloroplaste. L'ammonium, là où le sol en offre, est absorbé tel quel
et évite ce coût.
\end{proposition}

\begin{definition}[Fixation de l'azote et nodosités]\label{def:b1:plant-water-minerals:nodules}
\index{fixation de l'azote}\index{nodosite@nodosité}\index{Rhizobium}
Aucun eucaryote ne sait utiliser le $\mathrm{N_2}$ de l'air. Certaines
bactéries le savent~: la \emph{nitrogénase} rompt la triple liaison du
$\mathrm{N_2}$ et le réduit en deux ammoniacs, au prix de seize \omterm{def:b1:water-small-molecules:atp}{ATP} et
huit électrons par molécule, et seulement là où l'oxygène est exclu.
Les légumineuses (pois, haricots, trèfle) hébergent de telles bactéries
(\emph{Rhizobium}) dans des \emph{nodosités} racinaires~: la plante
bâtit une chambre, nourrit les bactéries de sucre, et maintient
l'oxygène bas grâce à une \omterm{def:b1:proteins:peptide}{protéine} rouge, la léghémoglobine, qui le
livre à la respiration des bactéries à une concentration trop faible
pour nuire à l'\omterm{def:b1:enzymes:enzyme}{enzyme}~; les bactéries rendent de l'ammonium, que la
plante transforme en \omterm{def:b1:proteins:aminoacid}{acides aminés}. Un champ de trèfle fixe cent
kilogrammes d'azote par hectare et par an, ce que la plante paie d'un
dixième de sa photosynthèse. Le cycle de l'azote dans son ensemble
relève du volume de deuxième année.
\end{definition}

\begin{example}[Le prix de l'azote]\label{ex:b1:plant-water-minerals:priceN}
Un pied de maïs a besoin d'environ \qty{30}{mg} d'azote par jour~; d'un
nitrate à \qty{2}{mmol/L}, il les obtient dans le litre d'eau qu'il
transpire, et dépense quelques centaines de millimoles d'électrons à
le réduire. Un pied de trèfle qui fixe la même quantité paie
\qty{16}{} \omterm{def:b1:water-small-molecules:atp}{ATP} par $\mathrm{N_2}$ plus le réducteur --- soit à peu près
\qty{6}{g} de sucre par gramme d'azote, quelque \qty{0.2}{g} par jour,
un dixième de sa production --- et peut pousser sur un sol sans le
moindre nitrate. Les paysans alternent légumineuses et céréales depuis
deux mille ans pour porter cet azote d'un champ au suivant.
\end{example}

\section{Les partenaires du sol}

\begin{definition}[Mycorhize]\label{def:b1:plant-water-minerals:mycorrhiza}
\index{mycorhize}
Une \emph{mycorhize} est une association entre une \omterm{def:b1:flowering-plant-organization:organs}{racine} et un
champignon, présente chez neuf plantes sur dix. Dans la forme la plus
répandue, les hyphes du champignon pénètrent dans les \omterm{def:b1:cell-unit-of-life:cell}{cellules} du
cortex et s'y ramifient en \emph{arbuscules}, la \omterm{prop:b1:organism-environment:surfaces}{surface d'échange},
tandis qu'au-dehors un réseau d'hyphes de quelques micromètres explore
le sol sur des mètres. Le champignon livre le phosphate et d'autres
ions peu mobiles, ainsi que de l'eau, que les \omterm{def:b1:flowering-plant-organization:organs}{poils absorbants} de la
\omterm{def:b1:flowering-plant-organization:organs}{racine} n'atteindraient pas~; la plante livre du sucre, un dixième ou
plus de sa photosynthèse. Les arbres des forêts tempérées portent une
seconde forme, où le champignon gaine les extrémités racinaires et se
faufile entre les \omterm{def:b1:cell-unit-of-life:cell}{cellules}~; ses fructifications sont les champignons
du sous-bois.
\end{definition}
\begin{proof}[Faits expérimentaux]
Des plantules cultivées en sol stérilisé poussent mal et absorbent peu
de phosphate~; inoculées avec des spores du champignon, elles
deviennent plusieurs fois plus grandes. Du phosphate radioactif placé
dans le sol hors de portée des \omterm{def:b1:flowering-plant-organization:organs}{racines} n'apparaît dans la plante que
si des hyphes relient les deux, et du carbone marqué donné aux \omterm{def:b1:flowering-plant-organization:organs}{feuilles}
apparaît dans les hyphes en un jour. Sectionner les hyphes par un
tamis fin qui laisse passer les solutés mais non les champignons
supprime le gain.
\end{proof}

\begin{omfigure}
\includegraphics[width=0.52\linewidth]{images/book3/ai/b3-23-mycorrhiza.png}

{\small Une \omterm{def:b1:flowering-plant-organization:organs}{racine} colorée pour montrer son champignon mycorhizien~:
des hyphes courant entre les \omterm{def:b1:cell-unit-of-life:cell}{cellules} et se ramifiant en arbuscules à
l'intérieur --- une seconde surface absorbante, alimentée en sucre par
la plante.}
\end{omfigure}

\begin{example}[Pourquoi le phosphate exige un champignon]\label{ex:b1:plant-water-minerals:phosphate}
Le phosphate se lie aux particules du sol et diffuse d'un millimètre
par jour~; un \omterm{def:b1:flowering-plant-organization:organs}{poil absorbant} prélève en quelques heures tout le
phosphate à sa portée, puis attend. Des hyphes dix fois plus fines
qu'un \omterm{def:b1:flowering-plant-organization:organs}{poil absorbant} atteignent dix fois plus loin par gramme de \omterm{def:b1:body-plans-tissues:tissue}{tissu},
et un mètre d'hyphe coûte à la plante le millième de ce que coûte un
mètre de \omterm{def:b1:flowering-plant-organization:organs}{racine}. Pour le nitrate, qui circule librement avec l'eau du
sol, les poils de la \omterm{def:b1:flowering-plant-organization:organs}{racine} suffisent, et les plantes des sols riches
en nitrate réduisent leur champignon~; pour le phosphate, le champignon
est le prolongement de la \omterm{def:b1:flowering-plant-organization:organs}{racine}.
\end{example}

\section{Problème~: l'échelle des potentiels d'une racine de maïs}

\begin{problem}[{Problème du week-end --- l'eau et le nitrate d'un sol
de juillet dans un pied de maïs~: les potentiels à chaque étape, le flux
à travers les racines, l'azote qu'il porte et le coût de sa réduction,
jusqu'au flux d'eau quotidien entrant dans la racine}]
\label{pb:b1:plant-water-minerals:1}
Un pied de maïs en juillet~: surface des \omterm{def:b1:flowering-plant-organization:organs}{racines} fines \qty{0.2}{m^2},
conductance hydraulique de la \omterm{def:b1:flowering-plant-organization:organs}{racine}
\qty{2e-7}{m.s^{-1}.MPa^{-1}}~; surface foliaire \qty{0.5}{m^2}.
\omterm{def:b1:membranes-transport:osmosis}{Potentiel hydrique} du sol \qty{-0.2}{MPa}~; \omterm{def:b1:cell-unit-of-life:cell}{cellules} du cortex racinaire
avec $\Psi_s = \qty{-0.9}{MPa}$ et $\Psi_p = \qty{0.5}{MPa}$~; \omterm{def:b1:flowering-plant-organization:tissues}{xylème}
racinaire \qty{-0.6}{MPa}~; \omterm{def:b1:cell-unit-of-life:cell}{cellules} foliaires \qty{-1.3}{MPa}~; air à
\qty{50}{\%} d'humidité relative ($RT/V_w = \qty{137}{MPa}$). Nitrate du
sol \qty{2}{mmol/L}~; la plante a besoin de \qty{30}{mg} d'azote par
jour. La réduction du nitrate en ammonium consomme 8 électrons par
azote~; la photosynthèse fait passer 4 électrons par $\mathrm{CO_2}$ et
la \omterm{def:b1:flowering-plant-organization:organs}{feuille} fixe \qty{4}{\micro mol} de $\mathrm{CO_2}$ par mètre carré
et par seconde pendant \qty{12}{h}.

\medskip
\noindent\textbf{Partie I --- L'échelle.}
\begin{enumerate}
  \item Calculer le \omterm{def:b1:membranes-transport:osmosis}{potentiel hydrique} des \omterm{def:b1:cell-unit-of-life:cell}{cellules} du cortex racinaire.
  \item Écrire la suite sol, cortex, \omterm{def:b1:flowering-plant-organization:tissues}{xylème}, \omterm{def:b1:flowering-plant-organization:organs}{feuille}, et vérifier que
        l'eau va vers l'intérieur et vers le haut à chaque étape.
  \item Calculer le \omterm{def:b1:membranes-transport:osmosis}{potentiel hydrique} de l'air.
  \item Calculer la chute du sol à la \omterm{def:b1:flowering-plant-organization:organs}{feuille} et la chute de la \omterm{def:b1:flowering-plant-organization:organs}{feuille}
        à l'air~; quelle fraction du total la dernière marche
        représente-t-elle~?
  \item Le sol s'assèche jusqu'à \qty{-0.9}{MPa}. Qu'advient-il du flux
        vers les \omterm{def:b1:cell-unit-of-life:cell}{cellules} du cortex, et de leur \omterm{def:b1:eukaryotic-cell:plantcell}{turgescence}~?
  \item Le sol s'assèche jusqu'à \qty{-1.5}{MPa}. Qu'advient-il des
        \omterm{def:b1:flowering-plant-organization:organs}{feuilles}, et pourquoi l'arrosage les rétablit-il~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le flux.}
\begin{enumerate}[resume]
  \item Calculer le flux d'eau entrant dans les \omterm{def:b1:flowering-plant-organization:organs}{racines} (conductance
        $\times$ surface $\times$ la différence entre le sol et le
        \omterm{def:b1:flowering-plant-organization:tissues}{xylème} racinaire).
  \item Le convertir en litres par jour.
  \item Tout est perdu par transpiration. Calculer la transpiration par
        mètre carré de \omterm{def:b1:flowering-plant-organization:organs}{feuille} et par heure, en grammes.
  \item Exprimer la transpiration par mètre carré de \omterm{def:b1:flowering-plant-organization:organs}{feuille} et par
        seconde en millimoles d'eau (\qty{18}{g/mol}).
  \item Les \omterm{def:b1:flowering-plant-organization:tissues}{stomates} se ferment à midi et le $\Psi$ de la \omterm{def:b1:flowering-plant-organization:organs}{feuille}
        remonte à \qty{-0.9}{MPa} tandis que celui du \omterm{def:b1:flowering-plant-organization:tissues}{xylème} remonte à
        \qty{-0.4}{MPa}. Recalculer le flux. Qu'a fait la plante, et
        pourquoi~?
  \item Pourquoi la \omterm{def:b1:flowering-plant-organization:organs}{racine} absorbe-t-elle l'eau sans dépenser
        d'énergie, alors qu'il lui en faut pour absorber le nitrate~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- L'azote.}
\begin{enumerate}[resume]
  \item Calculer le nitrate entraîné dans la \omterm{def:b1:flowering-plant-organization:organs}{racine} par le flux d'eau
        de la question 8, en millimoles et en milligrammes d'azote.
  \item Cela couvre-t-il le besoin de la plante~? Qu'advient-il du
        solde~?
  \item La \omterm{def:b1:flowering-plant-organization:organs}{racine} accumule le nitrate à \qty{5}{mmol/L} dans ses
        \omterm{def:b1:cell-unit-of-life:cell}{cellules}. Calculer l'\omterm{prop:b1:water-small-molecules:gibbs}{énergie libre} du transport d'une mole de
        \qty{2}{mmol/L} au dehors à \qty{5}{mmol/L} au dedans, contre
        un \omterm{def:b1:membranes-transport:potential}{potentiel de membrane} de \qty{-150}{mV} ($RT = \qty{2.48}{kJ/mol}$~;
        un anion entrant dans une \omterm{def:b1:cell-unit-of-life:cell}{cellule} négative).
  \item L'absorption est un symport à deux protons. Calculer l'\omterm{prop:b1:water-small-molecules:gibbs}{énergie
        libre} que libèrent deux protons entrant dans une \omterm{def:b1:cell-unit-of-life:cell}{cellule} à
        \qty{-150}{mV} depuis pH 5,5 vers pH 7,5, et vérifier qu'elle
        suffit.
  \item Calculer les électrons nécessaires par jour pour réduire les
        \qty{30}{mg} d'azote en ammonium.
  \item Calculer les électrons que la \omterm{def:b1:flowering-plant-organization:organs}{feuille} fait passer dans ses
        \omterm{def:b1:photosynthesis:zscheme}{photosystèmes} par jour, et la fraction détournée vers le
        nitrate.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Les autres voies.}
\begin{enumerate}[resume]
  \item Si le sol offrait de l'ammonium au lieu du nitrate, que la
        plante économiserait-elle, et quel problème l'absorption
        d'ammonium poserait-elle à son équilibre de pH~?
  \item Un trèfle fixe les mêmes \qty{30}{mg} d'azote par jour. Avec
        \qty{16}{} \omterm{def:b1:water-small-molecules:atp}{ATP} et \qty{8}{} électrons par $\mathrm{N_2}$,
        calculer l'\omterm{def:b1:water-small-molecules:atp}{ATP} dépensé par la nitrogénase et par jour.
  \item À \qty{6}{g} de sucre par gramme d'azote fixé au total,
        calculer le coût quotidien du trèfle en sucre et le comparer à
        une production de \qty{3}{g} de sucre par jour. Le marché
        vaut-il la peine dans un sol à \qty{2}{mmol/L} de nitrate~?
        Dans un sol qui n'en a pas~?
  \item Le phosphate est à \qty{2}{\micro mol/L} dans la solution du
        sol. Calculer le phosphate qu'apporte le flux d'eau de la
        question 8, et le comparer à un besoin de \qty{4}{mg} de
        phosphore par jour (\qty{31}{g/mol}). Sur quoi la plante
        doit-elle compter~?
  \item Expliquer en deux phrases comment le \omterm{def:b1:plant-water-minerals:pathways}{cadre de Caspary} permet à
        la \omterm{def:b1:flowering-plant-organization:organs}{racine} de tenir le nitrate à \qty{5}{mmol/L} dans son \omterm{def:b1:flowering-plant-organization:tissues}{xylème}
        alors que le sol est à \qty{2}{}.
  \item Un mutant est dépourvu de \omterm{def:b1:plant-water-minerals:pathways}{cadre de Caspary}. Prévoir la
        composition de sa sève brute et son comportement en sol salé.
  \item Énoncer le résultat~: le flux d'eau entrant dans la \omterm{def:b1:flowering-plant-organization:organs}{racine} de
        maïs par jour, l'azote qu'il apporte, et la fraction des
        électrons de la \omterm{def:b1:flowering-plant-organization:organs}{feuille} dépensée à le réduire.
\end{enumerate}
\end{problem}
