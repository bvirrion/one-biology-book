\chapter{التنوع الوراثي والمرض}\label{ch:g11:genes-and-disease}

زوجان في تمام الصحة يرزقان طفلًا أول يسعل منذ الأسابيع الأولى، ولا
يزداد وزنه، وعرقه من الملوحة بحيث تذوقه قبلة على الجبين. والتشخيص هو
التليف الكيسي: فكلا الوالدين، من غير أن يدري، كان يحمل نسخة واحدة غير
عاملة من \omterm{def:g10:universal-dna:gene}{المورّثة} نفسها، وتلقى الطفل النسختين. وواحد من كل خمسة وعشرين
شخصًا من أصل أوروبي حامل كهذا؛ وكان في وسع الوالدين أن يعلما، قبل
الحمل، باختبار على قطرة دم. وهذا الفصل عن الأمراض المكتوبة في
\omterm{def:g10:universal-dna:gene}{الأليلات}: كيف تورث، وكيف يحسب الخطر من شجرة عائلة، وكيف يقرأ المختبر
\omterm{def:g10:universal-dna:gene}{المورّثة}، ولماذا كانت معظم الأمراض الشائعة وراثية جزئيًا فقط.

\section{أمراض المورّثة الواحدة}

\begin{definition}[المرض الوراثي، والسائد والمتنحي]\label{def:g11:genes-and-disease:genetic}
\emph{المرض الوراثي}\index{مرض وراثي} مرض تسببه \omterm{def:g10:universal-dna:gene}{أليلات} طافرة، واحد أو
أكثر. وفي مرض \omterm{def:g10:universal-dna:gene}{مورّثة} واحدة، يكون \omterm{def:g10:universal-dna:gene}{الأليل} \emph{متنحيًا}\index{متنح} إذا
لم يظهر المرض إلا عند من يحمل نسختين منه --- ومن يحمل نسخة واحدة
\emph{حامل}\index{حامل} سليم --- ويكون \emph{سائدًا}\index{سائد} إذا
كفت نسخة واحدة. واصطلاحًا تكتب \omterm{def:g10:universal-dna:gene}{أليلات} \omterm{def:g10:universal-dna:gene}{المورّثة} بحرف: $A$ \omterm{def:g10:universal-dna:gene}{للأليل}
العادي، و$a$ \omterm{def:g10:universal-dna:gene}{لأليل} طافر متنح، فيكون الحامل $Aa$ والمصاب $aa$.
\end{definition}

\begin{example}[التليف الكيسي]\label{ex:g11:genes-and-disease:cf}
\omterm{def:g10:universal-dna:gene}{مورّثة} CFTR ترمز \omterm{def:g11:gene-expression:protein}{لبروتين} يتيح لشوارد الكلور عبور غشاء \omterm{def:g10:cells-common-unit:cell}{الخلايا} المبطنة
للطرق الهوائية والمعي والغدد العرقية. \omterm{def:g10:universal-dna:gene}{والأليل} الطافر الأشيع تنقصه
ثلاثة نوكليوتيدات، ومنه \omterm{def:g10:chemistry-of-life:families}{حمض أميني} واحد؛ فيسوء طي \omterm{def:g11:gene-expression:protein}{البروتين} ويهدم. ومن
دونه يصير مخاط الطرق الهوائية غليظًا لزجًا، وتستقر العدوى في الرئتين،
وتنسد قنوات البنكرياس. والمرض \omterm{def:g11:genes-and-disease:genetic}{متنح}: فنحو أوروبي واحد من كل 25 حامل
سليم، وواحد من كل 2500 وليد مصاب. وفقر الدم المنجلي
(\cref{ch:g11:enzymes-and-phenotype}) \omterm{def:g11:genes-and-disease:genetic}{متنح} كذلك؛ أما داء هنتنغتون،
وهو تنكس في الدماغ يبدأ حول الأربعين، \omterm{def:g11:genes-and-disease:genetic}{فسائد} --- إذ يكفي \omterm{def:g10:universal-dna:gene}{أليل} طافر
واحد، وكل مصاب كان له والد مصاب.
\end{example}

\begin{proposition}[انتقال مرض متنح]\label{prop:g11:genes-and-disease:recessive}
ينقل كل والد أحد أليليه، مختارًا عشوائيًا، إلى كل طفل. ومن ثم فحاملان
$Aa \times Aa$ لهما، في كل طفل، حظ $1/4$ في طفل مصاب $aa$، و$1/2$ في
حامل $Aa$، و$1/4$ في غير حامل $AA$. أما حامل وغير حامل
$Aa \times AA$ فلا طفل مصاب لهما وحظ واحد من اثنين في حامل. وأما مصاب
وغير حامل $aa \times AA$ فأطفالهما حاملون كلهم.
\end{proposition}
\admitted

للذكور، ودوائر للإناث، وخط أفقي يصل بين زوجين، وخط شاقولي نازل إلى

ويرقم الأفراد من اليسار.
\end{definition}

\begin{method}[تحليل شجرة نسب]\label{met:g11:genes-and-disease:analyse}
\begin{enumerate}
  \item \emph{\omterm{def:g11:genes-and-disease:genetic}{سائد} أم \omterm{def:g11:genes-and-disease:genetic}{متنح}؟} طفل مصاب لوالدين سليمين يعني أنه \omterm{def:g11:genes-and-disease:genetic}{متنح}
        (فالوالدان حاملان). أما مصاب له والد مصاب في كل جيل فيوحي بأنه
        \omterm{def:g11:genes-and-disease:genetic}{سائد}.
  \item \emph{على \omterm{def:g10:universal-dna:information}{صبغي} جنسي؟} إذا لم يصب إلا الذكور وأصيب أخوال
        أمهاتهم أو آباؤهم كذلك، فاشتبه في \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} X (انظر أدناه). وإلا
        \omterm{def:g10:universal-dna:gene}{فالمورّثة} على \omterm{def:g10:universal-dna:information}{صبغي} عادي.
  \item \emph{أسند الأنماط الوراثية}: فالمصابون في مرض \omterm{def:g11:genes-and-disease:genetic}{متنح} $aa$؛
        ووالداهم وأطفالهم يحملون $a$ واحدًا على الأقل؛ وإخوة المصاب
        السليمون $AA$ أو $Aa$.
  \item \emph{احسب الخطر} لطفل مزمع بجمع نمطي الوالدين الوراثيين في
        مربع بونيت؛ وإذا كان \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي} لأحد الوالدين غير مؤكد،
        فرجّح كل إمكان باحتماله.
\end{enumerate}
\end{method}

\begin{example}[الثلثان]\label{ex:g11:genes-and-disease:twothirds}
لماذا يكون الأخ السليم لطفل مصاب حاملًا باحتمال $2/3$ لا $1/2$؟
الوالدان $Aa \times Aa$؛ والطفل $AA$ أو $Aa$ أو $aA$ أو $aa$ بحظوظ
متساوية. ومعرفة أنه سليم تزيل $aa$: فتبقى ثلاث حالات، اثنتان منها
حاملتان. فإذا تزوج امرأة من عامة الجمهرة (حاملة باحتمال $1/25$)، كان
خطر طفل أول مصاب $2/3 \times 1/25 \times 1/4 = 1/150$.
\end{example}

\section{المورّثات على الصبغي X}

\begin{proposition}[الوراثة المرتبطة بالصبغي X]\label{prop:g11:genes-and-disease:xlinked}
تحمل الإناث صبغيي X، ويحمل الذكور X واحدًا وY واحدًا. \omterm{def:g10:universal-dna:gene}{والأليل} المتنحي
على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} X يقنَّع عند الأنثى \omterm{def:g10:universal-dna:information}{بصبغي} X الثاني، أما الذكر، ولا \omterm{def:g10:universal-dna:information}{صبغي} X
ثانيًا عنده، فيعبّر عنه. ومن ثم يصيب مثل هذا المرض الذكور في معظمه؛
وبنات الذكر المصاب حاملات كلهن؛ والأم الحاملة تنقل \omterm{def:g10:universal-dna:gene}{الأليل} إلى نصف
أبنائها فيصابون، وإلى نصف بناتها فيصرن حاملات. والناعور، ولا يتخثر
فيه الدم، هو المثال الكلاسيكي؛ وعمى الألوان الأحمر والأخضر
(\cref{ch:g11:the-eye}) مثال آخر.
\end{proposition}
\admitted

\section{قراءة المورّثة نفسها}

\begin{proposition}[الاختبار الوراثي]\label{prop:g11:genes-and-disease:testing}
يمكن قراءة \omterm{def:g10:universal-dna:gene}{الأليلات} التي يحملها شخص مباشرةً من \omterm{def:g10:universal-dna:information}{DNA} عنده، مستخلصًا من
بضع \omterm{def:g10:cells-common-unit:cell}{خلايا} (دم أو لعاب، ومن المشيمة أو السائل المحيط بها في الجنين).
فتنسخ منطقة \omterm{def:g10:universal-dna:gene}{المورّثة} ملايين المرات بتفاعل يحاكي التضاعف
(\cref{ch:g11:dna-replication}) في أنبوب اختبار، ثم إما تعيَّن
متتاليتها وإما تقطع \omterm{def:g11:enzymes-and-phenotype:enzyme}{بإنزيمات} تتعرف متتاليات بعينها وتفرز بالحجم في
هلام: \omterm{def:g10:universal-dna:gene}{فالأليل} الطافر الذي اكتسب موضع قطع أو فقده أو غيّره يعطي قطعًا
مختلفة الأطوال، ترى نطاقات. ومثل هذه الاختبارات تعيّن الحاملين قبل
حبل أي طفل، وتشخص الجنين، وتكشف \omterm{def:g11:mutations:mutation}{الطفرة} عند الوليد قبل ظهور الأعراض.
\end{proposition}
\admitted

يقرأ \omterm{def:g10:universal-dna:gene}{الأليلات}.}
\end{omfigure}

\begin{example}[ما يستطيع الاختبار قوله وما لا يستطيع]\label{ex:g11:genes-and-disease:testsays}
اختبار الحمل للتليف الكيسي يبحث عن \omterm{def:g10:universal-dna:gene}{الأليلات} الطافرة الخمسين الأشيع
فيجد نحو 90\% من الحاملين: فالنتيجة السلبية تخفض حظ الشخص في أن يكون
حاملًا من $1/25$ إلى نحو $1/250$، لكنها لا تجعله صفرًا. أما التشخيص قبل
الولادة على جنين لحاملين معروفين فيكاد يكون يقينيًا، ويضع الوالدين
أمام قرار لا تتخذه البيولوجيا عنهما. واختبار داء هنتنغتون يخبر شخصًا
سليمًا في الثلاثين هل سيأتي المرض؛ وثلث من يحق لهم إجراؤه يختارون ألا
يعرفوا.
\end{example}

\section{أمراض المورّثات الكثيرة وأمراض نمط العيش}

\begin{proposition}[الأمراض المتعددة العوامل]\label{prop:g11:genes-and-disease:multifactorial}
معظم الأمراض الشائعة --- السكري من النمط الثاني، وأمراض القلب
والشرايين، ومعظم السرطانات، والربو --- \emph{متعددة
العوامل}\index{مرض متعدد العوامل}: فعشرات أو مئات \omterm{def:g10:universal-dna:gene}{الأليلات} يزيح كل
منها الخطر قليلًا، والمحيط (الغذاء والحركة والتدخين والعدوى) يزيحه
كثيرًا. ومثل هذا المرض يجري في العائلات من دون أن يتبع قواعد \omterm{def:g10:universal-dna:gene}{المورّثة}
الواحدة، والاستعداد الوراثي خطر لا حكم: \omterm{def:g10:universal-dna:gene}{فالأليلات} نفسها تنتج المرض في
نمط عيش ولا تنتجه في آخر.
\end{proposition}
\begin{proof}[شواهد]
التوأمان المتطابقان يتقاسمان أليلاتهما كلها؛ وحين يصاب أحدهما بالسكري
من النمط الثاني، يصاب الآخر في نحو 70\% من الحالات --- وهو أعلى بكثير
من 8\% في الجمهرة، \omterm{def:g10:universal-dna:gene}{فالمورّثات} تهم --- لكنه ليس 100\%، \omterm{def:g10:universal-dna:gene}{فالمورّثات} لا
تقرر. والجمهرات التي انتقلت في جيل واحد من غذاء ريفي إلى غذاء مدني
رأت المرض يرتفع عدة أضعاف \omterm{def:g10:universal-dna:gene}{بالمورّثات} نفسها. ومن بين حاملي كثير من
\omterm{def:g10:universal-dna:gene}{أليلات} الخطر، من يبقون نشطين نحافًا يصابون بالمرض بنسبة جزء من نسبة
غيرهم.
\end{proof}

\begin{remark}[ماذا تعني «وراثي» إذن]\label{rem:g11:genes-and-disease:meaning}
التليف الكيسي وراثي بالمعنى الدقيق: \omterm{def:g11:enzymes-and-phenotype:phenotype}{فالنمط الوراثي} يحدد المرض، أيًّا
كان المحيط، والمحيط لا يغير إلا كيفية عيشه. أما السكري من النمط الثاني
فوراثي بمعنى أضعف: \omterm{def:g11:enzymes-and-phenotype:phenotype}{فالنمط الوراثي} يضع قابلية يحققها المحيط أو لا
يحققها. وبين الاثنين تقع بيلة الفينيل كيتون، حيث يلغي غذاءٌ أثرَ \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط
الوراثي}، ويقع حامل فقر الدم المنجلي، وأليله مرض في مكان وحماية في
آخر. فسؤال «أهو وراثي؟» قلما يكون سؤال نعم أو لا؛ وإنما السؤال الذي
يطرح هو «إلى أي حد، وفي أي شروط؟».
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g11:genes-and-disease:1}
عرّف \omterm{def:g10:universal-dna:gene}{الأليل} المتنحي \omterm{def:g10:universal-dna:gene}{والأليل} \omterm{def:g11:genes-and-disease:genetic}{السائد}، والحامل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:genes-and-disease:2}
حاملان للتليف الكيسي يرزقان طفلًا. أعطِ احتمال أن يكون الطفل مصابًا،
أو حاملًا، أو لا هذا ولا ذاك.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:genes-and-disease:3}
في شكل \omterm{def:g11:genes-and-disease:pedigree}{شجرة النسب}، أعطِ الأنماط الوراثية للأفراد I-1 وI-2 وII-2
وIII-2.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:genes-and-disease:4}
لماذا كانت الأمراض المتنحية المرتبطة \omterm{def:g11:cell-cycle-mitosis:chromosome}{بالصبغي} X أشيع كثيرًا عند الذكور؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:genes-and-disease:5}
ما \omterm{prop:g11:genes-and-disease:multifactorial}{المرض المتعدد العوامل}؟ أعطِ مثالين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:genes-and-disease:6}
مصاب بالتليف الكيسي يتزوج غير حاملة. فما أطفالهما؟ وما أطفال أحدهم مع

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:genes-and-disease:7}
في شكل \omterm{def:g11:genes-and-disease:pedigree}{شجرة النسب}، احسب احتمال أن يكون II-1 حاملًا، واحتمال أن يكون
طفل من II-1 ورجل من عامة الجمهرة مصابًا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:genes-and-disease:8}
رجل مصاب بداء هنتنغتون (\omterm{def:g11:genes-and-disease:genetic}{سائد}، \omterm{def:g10:universal-dna:gene}{بأليل} طافر واحد) وامرأة سليمة لهما أربعة
أطفال. فما احتمال أن يرث طفل معين المرض؟ وأن لا يرثه أي من الأربعة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:genes-and-disease:9}
رجل ناعوري يتزوج امرأة غير حاملة. صف الأنماط الوراثية والظاهرية
لأطفالهما، وأنماط الأحفاد من ابنة متزوجة برجل سليم.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:genes-and-disease:10}
على شكل الهلام، أي نطاقات يبينها شخص يحمل \omterm{def:g10:universal-dna:gene}{أليلين} طافرين مختلفين،
أحدهما يضيف موضع القطع والآخر يحذف 100 زوج قواعد أخرى من قطعة 400؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:genes-and-disease:11}
فسّر لماذا يكون التليف الكيسي أشيع ألف مرة في أوروبا منه في شرق آسيا
مع أن نسبة \omterm{def:g11:mutations:mutation}{الطفرات} واحدة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:genes-and-disease:12}
الطفل الأول لزوجين مصاب بالتليف الكيسي. احسب احتمال أن يكون الثاني
مصابًا، ثم أن يكون واحد على الأقل من التاليين مصابًا. وفسّر لماذا كان
قول «قد نلنا واحدنا من أربعة» مغالطة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:genes-and-disease:13}
اختبار حمل يكشف 90\% من \omterm{def:g10:universal-dna:gene}{الأليلات} الطافرة. امرأة من عامة الجمهرة نتيجتها
سلبية. احسب احتمالها الباقي في أن تكون حاملة، وخطر طفل مصاب مع شريك

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:genes-and-disease:14}

شخص يحمل كثيرًا من \omterm{def:g10:universal-dna:gene}{أليلات} الخطر. ففيم ينبغي أن تستعمل درجة الخطر
الوراثية؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:genes-and-disease:15}

\end{exercise}

\section{مسألة: عائلة واحدة، مورّثة واحدة}

\begin{problem}[{مسألة نهاية الأسبوع --- عائلة فيها التليف الكيسي
تتابع عبر ثلاثة أجيال: أنماط وراثية تسند، وأخطار تحسب لكل طفل مزمع،
واختبار مخبري يقرأ، وحساب جمهرة من الحاملين}]
\label{pb:g11:genes-and-disease:1}
ليا وتوم، وكلاهما سليم، لهما ثلاثة أطفال: ماكس (سليم)، وزوي (مصابة
بالتليف الكيسي)، ولو (سليمة). وأخت توم، آنا، سليمة ومتزوجة من كريم،
ولا تاريخ للمرض في عائلته؛ وهما ينتظران طفلًا. وفي الجمهرة، شخص واحد
من كل 25 حامل.

\medskip
\noindent\textbf{الجزء الأول --- الأنماط الوراثية.}
\begin{enumerate}
  \item أسائد المرض أم \omterm{def:g11:genes-and-disease:genetic}{متنح}؟ علّل من العائلة.
  \item أعطِ الأنماط الوراثية لليا وتوم وزوي.
  \item ما الأنماط الوراثية الممكنة لماكس، وبأي احتمالات؟
  \item ما احتمال أن تكون آنا حاملة؟ (تأمل والديها.)
  \item ما احتمال أن يكون كريم حاملًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الأخطار.}
\begin{enumerate}[resume]
  \item احسب احتمال أن يكون طفل آنا وكريم مصابًا.
  \item تفكر ليا وتوم في طفل رابع. فما احتمال أن يكون مصابًا؟ وأن
        يكون حاملًا؟
  \item يتزوج ماكس، وقد بلغ، امرأةً من عامة الجمهرة. فما احتمال طفل
        أول مصاب؟
  \item تتزوج زوي، وقد بلغت، رجلًا من عامة الجمهرة. فما احتمال طفل
        مصاب؟ وطفل حامل؟
  \item تتزوج لو ابن عمها، وهو طفل لآنا وكريم. احسب احتمال طفل مصاب.
        ولماذا كان أعلى منه عند ماكس؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الاختبار.}
\omterm{def:g10:universal-dna:gene}{الأليل} الطافر المحمول في هذه العائلة يضيف موضع قطع: \omterm{def:g10:universal-dna:gene}{فالأليل} العادي
يعطي قطعة واحدة من 900 زوج قواعد، والطافر يعطي 600 و300.
\begin{enumerate}[resume]
  \item ارسم، أو صف، النطاقات المنتظرة عند زوي، وعند توم، وعند غير
        حامل.
  \item يختبر ماكس: فيظهر نطاق واحد عند 900. فما نمطه الوراثي الآن،
        وما الخطر الجديد على طفله الأول (السؤال 8)؟
  \item تختبر آنا: فتظهر نطاقات عند 900 و600 و300. أعد حساب الخطر على
        طفلها.
  \item لا يجد اختبار كريم أيًّا من \omterm{def:g10:universal-dna:gene}{الأليلات} الطافرة الخمسين الأشيع،
        وهي تغطي 90\% من الحاملين. احسب احتماله الباقي في أن يكون
        حاملًا وأعد حساب الخطر.
  \item يبين اختبار جنين آنا وكريم قبل الولادة نطاقات عند 900 و600
        و300. فما \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي} \omterm{def:g11:enzymes-and-phenotype:phenotype}{والنمط الظاهري} للجنين؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الجمهرة.}
\begin{enumerate}[resume]
  \item إذا كان شخص واحد من كل 25 حاملًا، فما احتمال أن يكون كلا
        عضوي زوجين عشوائيين حاملًا، ومنه ما تواتر الولدان المصابين؟
  \item قارن بواحد من كل 2500 المشاهد. علّق.
  \item كان معظم المصابين يموتون في الطفولة فيما مضى. فسّر لماذا بقي
        \omterm{def:g10:universal-dna:gene}{الأليل} الطافر مع ذلك عند تواتر نسخة واحدة من كل 50، مستعملًا
        عدد النسخ التي يحملها الحاملون السليمون في مقابل التي يحملها
        المصابون.
  \item صارت العلاجات تتيح لمعظم المصابين بلوغ سن الرشد وإنجاب أطفال.
        تنبأ بأثر ذلك في تواتر \omterm{def:g10:universal-dna:gene}{الأليل} عبر الأجيال القادمة، وبحجم ذلك
        الأثر.
  \item اذكر النتيجة: واحد من أربعة عند حاملين، واثنان من ثلاثة عند
        أخ سليم، وما الذي أضافه الهلام إلى حساب هذه العائلة.
\end{enumerate}
\end{problem}
