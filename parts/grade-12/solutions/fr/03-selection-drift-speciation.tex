\section*{Chapitre \ref{ch:g12:selection-drift-speciation} --- Sélection, dérive et spéciation}

\begin{solution}{exo:g12:selection-drift-speciation:1}
La fraction de tous les exemplaires d'un \omterm{def:g10:universal-dna:gene}{gène} d'une \omterm{def:g12:selection-drift-speciation:population}{population} qui sont
un \omterm{def:g10:universal-dna:gene}{allèle} donné. Avec $p = 0.8$, $q = 0.2$~: $AA$ 64~\%, $Aa$ 32~\%,
$aa$ 4~\%.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:2}
La sélection~: une différence de reproduction entre les
\omterm{def:g11:genes-and-disease:genetic}{porteurs} d'\omterm{def:g10:universal-dna:gene}{allèles} différents, fixée par l'environnement, qui change les \omterm{def:g12:selection-drift-speciation:population}{fréquences} dans
une direction. La dérive~: une fluctuation aléatoire des \omterm{def:g12:selection-drift-speciation:population}{fréquences} due
à l'échantillonnage d'une \omterm{def:g12:selection-drift-speciation:population}{population} finie, sans direction. La
sélection dépend de ce que fait l'\omterm{def:g10:universal-dna:gene}{allèle}~; la dérive, non.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:3}
Environ 75~\% en 1880, quand la suie avait noirci les troncs~; environ
30~\% en 1980, après que les lois sur l'air propre eurent laissé les
troncs s'éclaircir de nouveau.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:4}
Les \omterm{def:g10:universal-dna:gene}{allèles} de la génération suivante sont un échantillon de ceux de la
précédente~; un petit échantillon s'écarte davantage de la source qu'un
grand, si bien que les \omterm{def:g12:selection-drift-speciation:population}{fréquences} fluctuent plus, et qu'un \omterm{def:g10:universal-dna:gene}{allèle} peut
être perdu par hasard.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:5}
Une interruption des échanges de \omterm{def:g10:universal-dna:gene}{gènes} entre deux \omterm{def:g12:selection-drift-speciation:population}{populations}, assez
longue pour qu'elles divergent jusqu'à ne plus pouvoir se reproduire
entre elles.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:6}
Exemplaires de $a$~: $160 + 40 = 200$ sur 1000, d'où $q = 0.2$,
$p = 0.8$. Attendus~: $0.64 \times 500 = 320$, $0.32 \times 500 = 160$,
$0.04
\times 500 = 20$ --- exactement le recensement~: la \omterm{def:g12:selection-drift-speciation:population}{population} est à
l'état de référence.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:7}
$q^2 = 0.0001$~: un individu sur dix mille montre le caractère~;
$2pq \approx
0.02$~: un sur cinquante le porte. Quatre-vingt-dix-neuf pour cent des
exemplaires se logent chez des hétérozygotes, sur qui une sélection
contre le caractère n'a aucune prise.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:8}
Toutes deux ont atteint une extrémité à la génération 12, l'une fixée à
100~\%, l'autre perdue. La troisième a erré sans toucher aucune
frontière en 20 générations --- le hasard, là encore~; avec plus de
temps, elle y viendrait.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:9}
Des phalènes claires ont continué de se reproduire dans la campagne non
polluée, où elles étaient favorisées, et les phalènes volent~: des
migrantes rapportaient l'\omterm{def:g10:universal-dna:gene}{allèle} clair dans la région couverte de suie à
chaque génération, si bien qu'il n'a jamais disparu.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:10}
Vingt individus portent au plus quarante exemplaires de chaque \omterm{def:g10:universal-dna:gene}{gène}, un
échantillon minuscule des \omterm{def:g10:universal-dna:gene}{allèles} d'origine~; la plupart ont été perdus
dans le goulot d'étranglement, et la dérive dans la petite \omterm{def:g12:selection-drift-speciation:population}{population}
qui a suivi en a perdu davantage. Les effectifs sont revenus~; les
\omterm{def:g10:universal-dna:gene}{allèles} ne le peuvent pas, sinon par \omterm{def:g11:mutations:mutation}{mutation} nouvelle.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:11}
Dix \omterm{def:g12:selection-drift-speciation:population}{populations} qui bougent dans le même sens, après le même changement
d'environnement, avec un mécanisme plausible~: la sélection. Une seule
\omterm{def:g12:selection-drift-speciation:population}{population} minuscule de trente individus, sans cause rapportée~: la
dérive est l'explication par défaut jusqu'à ce qu'un mécanisme soit
montré.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:12}
$AA$ $0.85^2 \approx 72\%$, $Aa$ $2 \times 0.85 \times 0.15 \approx
26\%$, $aa$ $0.15^2 \approx 2\%$. Les $aa$ meurent, ce qui retire des
exemplaires de $a$~; mais les $Aa$, un quart de la \omterm{def:g12:selection-drift-speciation:population}{population},
survivent mieux au paludisme que les $AA$ et transmettent leurs
exemplaires de $a$. La perte due aux $aa$ est compensée par l'avantage
des $Aa$, et $a$ se maintient à 15~\%.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:13}
Le choix du partenaire d'après la couleur~: une barrière
comportementale. En eau trouble, la barrière tombe et les \omterm{def:g10:biodiversity-scales:species}{espèces}
s'hybrident~: aucune incompatibilité génétique ne s'est donc encore
accumulée~; la séparation est récente et encore réversible.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:14}
Tout autour de l'anneau, chaque \omterm{def:g12:selection-drift-speciation:population}{population} se reproduit avec ses
voisines, avec de faibles différences entre \omterm{def:g12:selection-drift-speciation:population}{populations} contiguës~; les
différences s'additionnent le long du cercle jusqu'à ce que les
extrémités, en se rencontrant, ne se reproduisent plus entre elles. Le
dégradé entier est visible d'un seul coup~: la \omterm{prop:g12:selection-drift-speciation:speciation}{spéciation} est affaire
de petits pas accumulés. Les deux extrémités sont appelées \omterm{def:g10:biodiversity-scales:species}{espèces}
parce que, là où elles se rencontrent, elles se comportent comme des
\omterm{def:g10:biodiversity-scales:species}{espèces}~: aucun échange de \omterm{def:g10:universal-dna:gene}{gènes}.
\end{solution}

\begin{solution}{exo:g12:selection-drift-speciation:15}
La sélection fait que les \omterm{def:g11:genes-and-disease:genetic}{porteurs} de certains \omterm{def:g10:universal-dna:gene}{allèles} se reproduisent
davantage \emph{ici et maintenant}~: meilleurs pour laisser des
descendants dans cet environnement, comparés aux autres membres de la
\omterm{def:g12:selection-drift-speciation:population}{population} --- et non meilleurs en un sens absolu. Quand
l'environnement change, la direction s'inverse (les phalènes)~; la
dérive fait diverger des \omterm{def:g12:selection-drift-speciation:population}{populations} sans aucune raison d'adaptation~;
la sélection sexuelle favorise des ornements qui gênent la survie.
L'évolution n'a pas de direction de progrès, seulement un tri local.
\end{solution}

\begin{solution}{pb:g12:selection-drift-speciation:1}
\textbf{1.} Exemplaires de $D$~: $2 \times 720 + 960 = 2400$~; de $d$~: $960 +
2 \times 320 = 1600$~; sur 4000~: $p = 0.6$, $q = 0.4$.

\textbf{2.} Attendus $0.36 \times 2000 = 720$, $0.48 \times 2000 =
960$, $0.16 \times 2000 = 320$~: exactement le recensement. Oui.

\textbf{3.} $960/1680 \approx 57\%$.

\textbf{4.} L'essentiel des exemplaires de $d$ se trouve chez les
porteuses sombres. Après le retrait~: 1680 souris, 2400 exemplaires de
$D$, 960 de $d$~: $q = 960/3360 \approx
0.29$ --- une baisse depuis 0.40, mais bien loin de l'élimination.

\textbf{5.} La grande taille~: une petite \omterm{def:g12:selection-drift-speciation:population}{population} insulaire dérive,
si bien que les \omterm{def:g12:selection-drift-speciation:population}{fréquences} errent au hasard (et la migration depuis le
continent est absente, la \omterm{def:g11:mutations:mutation}{mutation} négligeable).

\textbf{6.} $DD$ 360, $Dd$ 480, $dd$ 320~: 1160 reproductrices.

\textbf{7.} $D$~: $720 + 480 = 1200$~; $d$~: $480 + 640 = 1120$~; sur
2320~: $p \approx 0.52$, $q \approx 0.48$.

\textbf{8.} Génération suivante (pour 2000)~: $DD$ $0.52^2 \approx 27\%$
(535), $Dd$ $\approx 50\%$ (999), $dd$ $\approx 23\%$ (466). Après les
chouettes~: 267, 500, 466. $D$~: $534 + 500 = 1034$~; $d$~: $500 + 932 =
1432$~; $q \approx 0.58$.

\textbf{9.} Avec $q = 0.58$~: $DD$ 18~\% (353), $Dd$ 49~\% (974), $dd$
34~\% (673)~; survivantes 176, 487, 673~; $d$~: $487 + 1346 = 1833$ sur
2672~: $q \approx 0.69$. Tendance~: 0.40, 0.48, 0.58, 0.69 --- une
montée d'environ 0.1 par génération.

\textbf{10.} Toute porteuse de $D$ est sombre et exposée aux
chouettes, si bien que la sélection agit sur chaque exemplaire de $D$ à
chaque génération~; $q$ monte régulièrement. Un \omterm{def:g10:universal-dna:gene}{allèle} \omterm{def:g11:genes-and-disease:genetic}{récessif} se
cacherait chez les hétérozygotes une fois devenu rare, mais un \omterm{def:g10:universal-dna:gene}{allèle}
\omterm{def:g11:genes-and-disease:genetic}{dominant} ne se cache jamais~: $D$ peut être ramené à zéro.

\textbf{11.} $D$~: $4 + 3 = 7$~; $d$~: $3 + 2 = 5$~; $q = 5/12 \approx
0.42$, proche des 0.40 de l'île par chance du tirage.

\textbf{12.} Chez les reproductrices, $D$ a 5 exemplaires et $d$ 1~:
$q$ à la génération suivante vaut $1/6$ en moyenne, mais l'unique
exemplaire de $d$ peut n'être transmis à aucun jeune ($q = 0$) ou à
plusieurs~; le \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotype} $dd$ a déjà disparu, et avec lui une grande
part de la variation.

\textbf{13.} Avec si peu de reproductrices, les \omterm{def:g10:universal-dna:gene}{allèles} transmis sont
affaire de hasard à chaque génération~; un \omterm{def:g10:universal-dna:gene}{allèle} rare peut être perdu
en une seule portée malchanceuse.

\textbf{14.} $d$ va dériver et, en quelques générations, être fixé ou
perdu --- l'une ou l'autre issue par hasard. Un îlot voisin, parti du
même état, peut finir avec l'\omterm{def:g10:universal-dna:gene}{allèle} opposé~: les deux îles diffèrent de
couleur de pelage sans aucune raison adaptative.

\textbf{15.} La \omterm{prop:g12:selection-drift-speciation:drift}{dérive génétique}~; la taille minuscule de la colonie.

\textbf{16.} Oui~: elles se reproduisent rarement avec les souris de
l'île et les hybrides sont stériles --- il y a isolement reproducteur.

\textbf{17.} La mer, une barrière géographique~; une sélection dans un
environnement différent (pas de chouettes, d'autres conditions) et la
\omterm{prop:g12:selection-drift-speciation:drift}{dérive génétique} dans une petite \omterm{def:g12:selection-drift-speciation:population}{population}.

\textbf{18.} Des souris qui se reproduisent à des saisons différentes
ne se rencontrent jamais comme partenaires~: même sans aucune barrière
géographique, aucun \omterm{def:g10:universal-dna:gene}{gène} ne circule, et les \omterm{def:g12:selection-drift-speciation:population}{populations} continuent de
diverger.

\textbf{19.} Pas maintenant~: elles sont isolées par le comportement et
par la stérilité, et resteraient deux \omterm{def:g10:biodiversity-scales:species}{espèces} côte à côte. Après 100
ans seulement, elles se seraient encore reproduites librement et
auraient fusionné en une seule \omterm{def:g12:selection-drift-speciation:population}{population}.

\textbf{20.} $q$ passe de 0.40 à environ 0.69 après trois générations
de chouettes~; sur l'îlot, $d$ a été fixé ou perdu par \omterm{prop:g12:selection-drift-speciation:drift}{dérive
génétique}~; deux \omterm{def:g10:biodiversity-scales:species}{espèces} à la fin.
\end{solution}
