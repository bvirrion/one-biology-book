\chapter{Fotosintesis dan Autotrofi}\label{ch:b1:photosynthesis}

Tiap gram karbon dalam si kelinci, pohon ek, pembaca dan halaman ini
dahulu adalah karbon dioksida di udara, dan difiksasi menjadi gula oleh
sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} --- atau oleh sebuah alga, atau sebuah sianobakteri ---
dengan memakai energi sinar matahari. Reaksinya, enam $\mathrm{CO_2}$
dan enam air menjadi satu glukosa dan enam oksigen, mendaki sebesar
\qty{2870}{kJ/mol}; sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} menjalankannya dengan membelah air
memakai cahaya, menimbun elektronnya pada NADPH dan energinya pada \omterm{def:b1:water-small-molecules:atp}{ATP},
lalu membelanjakan keduanya dalam sebuah daur yang membangun gula. Bab
ini memerikan \omterm{def:b1:photosynthesis:chloroplast}{kloroplas}, pigmen dan \omterm{prop:b1:photosynthesis:lightreactions}{reaksi terang} yang membuat \omterm{def:b1:water-small-molecules:atp}{ATP} serta
NADPH, \omterm{def:b1:photosynthesis:fixation}{daur Calvin} yang memfiksasi karbon, kebocoran yang disebut
fotorespirasi beserta dua rancangan yang menutupnya, dan apa itu
\omterm{def:b1:photosynthesis:autotrophy}{autotrofi} pada umumnya.

\section{Kloroplas dan pigmennya}

\begin{definition}[Kloroplas]\label{def:b1:photosynthesis:chloroplast}
Yang disebut \emph{kloroplas}\index{kloroplas} adalah organel berbentuk
lensa sepanjang \qtyrange{3}{10}{\micro m} yang dibatasi dua membran
selubung; bagian dalamnya, yaitu \emph{stroma}\index{stroma}, memegang
\omterm{def:b1:enzymes:enzyme}{enzim} fiksasi karbon, butiran \omterm{def:b1:carbohydrates:polysaccharide}{pati}, ribosom dan sebuah \omterm{def:b1:nucleic-acids:chain}{DNA} melingkar; di
dalam stroma sistem membran ketiga, yaitu
\emph{tilakoid}\index{tilakoid}, membentuk kantong pipih yang bertumpuk
menjadi \emph{grana} dan terhubung lewat lamela, yang melingkupi satu
\emph{lumen tilakoid} yang bersinambung. \omterm{prop:b1:photosynthesis:lightreactions}{Reaksi terangnya} berlangsung
dalam membran tilakoid; fiksasi karbonnya dalam stroma. Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{daun}
memegang \qtyrange{20}{100}{} kloroplas; satu milimeter persegi \omterm{def:b1:flowering-plant-organization:organs}{daun},
setengah juta.
\end{definition}

\begin{definition}[Pigmen fotosintesis]\label{def:b1:photosynthesis:pigments}
\emph{Klorofil}\index{klorofil} $a$ dan $b$ adalah cincin porfirin
mengelilingi sebuah atom magnesium dengan ekor hidrofob panjang yang
menambatkannya dalam membran \omterm{def:b1:photosynthesis:chloroplast}{tilakoid}; keduanya menyerap cahaya biru
(\qtyrange{430}{450}{nm}) dan merah (\qtyrange{640}{680}{nm}) serta
memantulkan hijau. \emph{Karotenoid}\index{karotenoid} (\omterm{def:b1:lipids:steroids}{isoprenoid},
\cref{ch:b1:lipids}) menyerap cahaya biru kehijauan lalu meneruskan
energinya ke klorofil, dan melindunginya dengan memadamkan kelebihan
eksitasinya. Beberapa ratus molekul pigmen, yang dipegang \omterm{def:b1:proteins:peptide}{protein}
sebagai sebuah \emph{antena} (kompleks pemanen cahaya), menyalurkan
energi tiap foton yang terserap ke satu pasangan istimewa molekul
klorofil $a$, yaitu \emph{pusat reaksi}, tempat energi itu diubah
menjadi kimia.
\end{definition}

\begin{proposition}[Spektrum aksinya mengikuti pigmennya]\label{prop:b1:photosynthesis:engelmann}
Fotosintesis digerakkan cahaya yang diserap pigmennya: ia tercepat
dalam cahaya merah dan biru, terlemah dalam hijau.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Engelmann (1882) membaringkan sehelai benang alga melintasi sebuah
spektrum yang diproyeksikan di bawah mikroskopnya lalu menambahkan
bakteri pencari oksigen: bakteri itu berkumpul di tempat cahaya merah
dan birunya jatuh, bukan di hijaunya (diingat kembali dari jilid Sekolah
Menengah). Diukur dengan sebuah elektrode oksigen, laju pada tiap
panjang gelombang memberikan spektrum aksi di atas, yang mengikuti
serapan \omterm{def:b1:photosynthesis:pigments}{klorofilnya}; kelebihannya pada biru kehijauan adalah sumbangan
\omterm{def:b1:photosynthesis:pigments}{karotenoidnya}, yang memperlihatkan bahwa mereka meneruskan energinya.
\end{proof}

\section{Reaksi terang}

\begin{proposition}[Apa yang dikerjakan reaksi terang]\label{prop:b1:photosynthesis:lightreactions}
\index{reaksi terang}
Dalam membran \omterm{def:b1:photosynthesis:chloroplast}{tilakoid}, energi cahaya dipakai untuk memindahkan elektron
dari air ke $\mathrm{NADP^+}$ --- mendaki, melawan selisih potensial
redoks \qty{1.1}{V} --- dan untuk memompa proton ke dalam lumennya:
\[
2\,\mathrm{H_2O} + 2\,\mathrm{NADP^+} + 3\,\mathrm{ADP} + 3\,\mathrm{P_i}
\xrightarrow{\ 8\ \text{foton}\ }
\mathrm{O_2} + 2\,\mathrm{NADPH} + 2\,\mathrm{H^+} + 3\,\mathrm{ATP} .
\]
Oksigen yang dilepaskannya adalah oksigen air, bukan oksigen karbon
dioksida; NADPH-nya membawa daya reduksi dan \omterm{def:b1:water-small-molecules:atp}{ATP-nya} membawa energi
yang akan dibelanjakan \omterm{def:b1:photosynthesis:fixation}{daur Calvin}.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Hill (1937) memperlihatkan bahwa \omterm{def:b1:photosynthesis:chloroplast}{kloroplas} terasing dalam cahaya
melepas oksigen tanpa hadirnya $\mathrm{CO_2}$ bila sebuah penerima
elektron buatan hadir: pembelahan airnya terpisahkan dari fiksasi
karbonnya. Ruben dan Kamen (1941) memberi alga air yang ditandai
oksigen berat lalu menemukan tandanya dalam $\mathrm{O_2}$ yang
dilepaskan, dan tidak ketika tandanya ada dalam $\mathrm{CO_2}$.
Emerson (1957) mendapati bahwa cahaya merah \qty{700}{nm}, yang
sendirian nyaris tak berguna, dan cahaya \qty{680}{nm} bersama-sama
memberikan lebih daripada jumlah laju terpisahnya: dua \omterm{def:b1:photosynthesis:zscheme}{fotosistem}
dengan pigmen yang berbeda pasti bekerja berderet.
\end{proof}

\begin{definition}[Fotosistem dan skema Z]\label{def:b1:photosynthesis:zscheme}
\index{fotosistem}\index{skema Z}
Dua \emph{fotosistem} duduk dalam membran \omterm{def:b1:photosynthesis:chloroplast}{tilakoid}. Dalam
\emph{fotosistem II} (PSII, pusat reaksi P680) sebuah foton yang
terserap menaikkan sebuah elektron pasangan istimewanya ke energi yang
tinggi; elektron itu pergi, dan P680 yang teroksidasi, pengoksidasi
biologis terkuat, mengambil sebuah elektron dari air: sebuah gugusan
mangan membelah dua molekul air menjadi $\mathrm{O_2}$, empat proton
(yang dilepas ke dalam lumennya) dan empat elektron, satu foton pada
satu waktu. Elektronnya menuruni sebuah \emph{rantai pengangkut
elektron} --- plastokuinon, \emph{kompleks sitokrom $b_6f$},
plastosianin --- dan energi yang hilang darinya memompa proton ke dalam
lumennya. Dalam \emph{fotosistem I} (PSI, P700) sebuah foton kedua
mengangkatnya lagi, ke potensial yang cukup rendah untuk mereduksi
feredoksin lalu $\mathrm{NADP^+}$ menjadi NADPH. Digambar pada skala
potensial redoks, lintasannya berupa huruf Z: dua pendakian oleh
cahaya, dua penurunan oleh kimia.
\end{definition}

\begin{theorem}[Sintesis ATP secara kemiosmosis]\label{thm:b1:photosynthesis:chemiosmosis}
\index{kemiosmosis}\index{ATP sintase}
Proton yang dilepas pembelahan air dan yang dipompa kompleks sitokrom
$b_6f$ menumpuk dalam lumen \omterm{def:b1:photosynthesis:chloroplast}{tilakoid}, yang dalam cahaya turun ke pH 5
sementara \omterm{def:b1:photosynthesis:chloroplast}{stromanya} naik ke pH 8: selisih tiga satuan, sebuah gaya gerak
proton sekitar \qty{0.18}{V}, setara \qty{17}{kJ} tiap mol proton.
\emph{\omterm{thm:b1:photosynthesis:chemiosmosis}{ATP sintase}}, sebuah motor putar yang menembus membran \omterm{def:b1:photosynthesis:chloroplast}{tilakoid},
membiarkan proton mengalir kembali ke dalam \omterm{def:b1:photosynthesis:chloroplast}{stromanya} lalu memakai
energinya untuk memfosforilasi ADP: sekitar \qty{4}{H^+} tiap \omterm{def:b1:water-small-molecules:atp}{ATP}.
Pengangkutan elektron dan sintesis \omterm{def:b1:water-small-molecules:atp}{ATP} hanya tergandeng lewat gradien
ini (teori kemiosmosis Mitchell, 1961).
\end{theorem}
\begin{proof}[Bukti eksperimental]
Jagendorf (1966) merendam \omterm{def:b1:photosynthesis:chloroplast}{tilakoid} dalam gelap di dalam sebuah rendaman
asam (pH 4) sampai lumennya asam, lalu memindahkannya mendadak ke pH 8
bersama ADP dan fosfat: \omterm{def:b1:water-small-molecules:atp}{ATP} dibuat dalam gelap, dari gradiennya saja.
Zat pemutus gandengan --- asam lemah yang larut \omterm{def:b1:lipids:fattyacid}{lipid} yang membawa
proton melintasi membran --- meniadakan sintesis \omterm{def:b1:water-small-molecules:atp}{ATP-nya} sementara
pengangkutan elektron dan pelepasan oksigennya berlanjut, bahkan
menjadi lebih cepat. \omterm{thm:b1:photosynthesis:chemiosmosis}{ATP sintase} terasing yang dirakit kembali ke dalam
vesikel buatan membuat \omterm{def:b1:water-small-molecules:atp}{ATP} ketika sebuah gradien pH dikenakan, dan
menghidrolisis \omterm{def:b1:water-small-molecules:atp}{ATP} untuk memompa proton ketika tidak. Putaran \omterm{def:b1:enzymes:enzyme}{enzimnya}
sejak itu telah disaksikan langsung, satu molekul pada satu waktu, di
bawah mikroskop.
\end{proof}

\begin{example}[Aliran siklik dan taksiklik]\label{ex:b1:photosynthesis:cyclic}
Lintasan lurus dari air ke NADPH memberikan, tiap $\mathrm{O_2}$, dua
NADPH dan sekitar tiga \omterm{def:b1:water-small-molecules:atp}{ATP}; \omterm{def:b1:photosynthesis:fixation}{daur Calvin} memerlukan rasio tepat tiga \omterm{def:b1:water-small-molecules:atp}{ATP}
terhadap dua NADPH, dan kerja \omterm{def:b1:photosynthesis:chloroplast}{kloroplas} yang lain memerlukan lebih
banyak \omterm{def:b1:water-small-molecules:atp}{ATP}. Ketika \omterm{def:b1:water-small-molecules:atp}{ATP-nya} kurang, elektron dari feredoksin kembali ke
plastokuinon alih-alih pergi ke $\mathrm{NADP^+}$ (\emph{fotofosforilasi
siklik}): PSI sendirian, yang memompa proton dan membuat \omterm{def:b1:water-small-molecules:atp}{ATP} tanpa
oksigen maupun NADPH. Kedua ragamnya diimbangkan menurut permintaannya.
\end{example}

\section{Daur Calvin}

\begin{definition}[Fiksasi karbon]\label{def:b1:photosynthesis:fixation}
\index{daur Calvin}\index{RuBisCO}
\emph{Daur Calvin} dalam \omterm{def:b1:photosynthesis:chloroplast}{stromanya} memfiksasi $\mathrm{CO_2}$ menjadi
gula dalam tiga tahap. \emph{Fiksasi}: \emph{RuBisCO} (ribulosa
bisfosfat karboksilase/oksigenase) menambahkan $\mathrm{CO_2}$ pada
ribulosa-1,5-bisfosfat (RuBP, lima karbon), dan hasil berkarbon enamnya
langsung terbelah menjadi dua molekul 3-fosfogliserat (3-PGA, tiga
karbon). \emph{Reduksi}: tiap 3-PGA difosforilasi \omterm{def:b1:water-small-molecules:atp}{ATP} lalu direduksi
NADPH menjadi gliseraldehida-3-fosfat (G3P), yaitu gula pertamanya.
\emph{Pembentukan kembali}: lima dari tiap enam G3P disusun ulang,
dengan biaya \omterm{def:b1:water-small-molecules:atp}{ATP}, kembali menjadi tiga RuBP, dan satu G3P meninggalkan
daurnya sebagai hasil bersihnya. Stoikiometri bagi satu G3P (tiga
karbon) adalah
\[
3\,\mathrm{CO_2} + 9\,\mathrm{ATP} + 6\,\mathrm{NADPH} + 6\,\mathrm{H^+}
\to \text{G3P} + 9\,\mathrm{ADP} + 8\,\mathrm{P_i} + 6\,\mathrm{NADP^+} ;
\]
dua G3P membuat satu glukosa: \qty{18}{} \omterm{def:b1:water-small-molecules:atp}{ATP} dan \qty{12}{NADPH} tiap glukosa.
\end{definition}

\begin{proposition}[Bukti Calvin]\label{prop:b1:photosynthesis:calvin}
Hasil mantap pertama fiksasi $\mathrm{CO_2}$ adalah 3-fosfogliserat,
dan penerimanya adalah ribulosa bisfosfat.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Calvin dan Benson (1948--1954) menyinari alga dalam sebuah labu pipih
(``permen lolipop''), menyuntikkan $\mathrm{^{14}CO_2}$, lalu mematikan
cuplikannya dalam alkohol mendidih sesudah beberapa sekon; kromatografi
kertas dua dimensi dan autoradiografi memperlihatkan tempat tandanya
berada. Sesudah lima sekon hampir seluruhnya berada dalam 3-PGA;
pemaparan yang lebih lama menyebarkannya melalui gula fosfat lalu ke
dalam sukrosa dan \omterm{def:b1:carbohydrates:polysaccharide}{pati}. Ketika $\mathrm{CO_2}$-nya tiba-tiba
disingkirkan, RuBP menumpuk dan 3-PGA turun; ketika cahayanya
dipadamkan, 3-PGA menumpuk dan RuBP turun --- jadi RuBP adalah
penerima yang dipakai fiksasinya, dan 3-PGA hasil yang dipakai \omterm{def:b1:water-small-molecules:atp}{ATP} serta
NADPH cahayanya.
\end{proof}

\begin{method}[Menyusun neraca fotosintesis]\label{met:b1:photosynthesis:budget}
\begin{enumerate}
  \item Hitunglah karbonnya: tiap $\mathrm{CO_2}$ yang difiksasi
        menghasilkan satu karbon hasil; satu glukosa memerlukan enam
        putaran \omterm{def:b1:photosynthesis:fixation}{RuBisCO}.
  \item Hitunglah pembawanya: tiap putaran memakan biaya 3 \omterm{def:b1:water-small-molecules:atp}{ATP} dan 2
        NADPH (9 dan 6 tiap G3P); satu glukosa memakan biaya 18 \omterm{def:b1:water-small-molecules:atp}{ATP} dan
        12 NADPH.
  \item Hitunglah fotonnya: aliran lurusnya memberikan 2 NADPH dan
        sekitar 3 \omterm{def:b1:water-small-molecules:atp}{ATP} tiap $\mathrm{O_2}$, bagi 8 foton (4 tiap
        \omterm{def:b1:photosynthesis:zscheme}{fotosistem}); 12 NADPH memerlukan 6 $\mathrm{O_2}$ dan 48 foton;
        \omterm{def:b1:water-small-molecules:atp}{ATP} tambahannya datang dari aliran siklik (beberapa foton lagi).
        Sekitar 8 foton tiap $\mathrm{CO_2}$: kebutuhan kuantumnya.
  \item Hitunglah energinya: satu mol foton \qty{680}{nm} adalah
        \qty{176}{kJ}; 48 foton adalah \qty{8450}{kJ} bagi
        \qty{2870}{kJ} yang tersimpan dalam glukosa: \qty{34}{\%}
        paling banyak, sebelum kehilangan apa pun oleh pemantulan,
        respirasi dan bagian spektrum yang tak terserap.
\end{enumerate}
\end{method}

\section{Fotorespirasi, C4 dan CAM}

\begin{proposition}[Cacat RuBisCO]\label{prop:b1:photosynthesis:photorespiration}
\index{fotorespirasi}
\omterm{def:b1:photosynthesis:fixation}{RuBisCO} juga menerima $\mathrm{O_2}$ menggantikan $\mathrm{CO_2}$:
oksigenasi RuBP menghasilkan satu 3-PGA dan satu fosfoglikolat berkarbon
dua, yang harus diselamatkan \omterm{def:b1:cell-unit-of-life:cell}{sel} lewat rute mahal melintasi tiga organel
yang melepas $\mathrm{CO_2}$ dan memakai \omterm{def:b1:water-small-molecules:atp}{ATP} ---
\emph{fotorespirasi}. \omterm{def:b1:enzymes:enzyme}{Enzimnya} memilah dengan buruk (sekitar
\qty{80}{} berbanding 1 memihak $\mathrm{CO_2}$ pada kepekatan yang
sama), dan di udara $\mathrm{O_2}$ lima ratus kali lebih berlimpah
daripada $\mathrm{CO_2}$; pada \qty{25}{\celsius} seperempat fiksasinya
terbuang, lebih banyak lagi pada cuaca panas dan kering ketika
\omterm{def:b1:flowering-plant-organization:tissues}{stomatanya} menutup dan $\mathrm{CO_2}$-nya turun di dalam \omterm{def:b1:flowering-plant-organization:organs}{daunnya}.
\omterm{def:b1:photosynthesis:fixation}{RuBisCO}, yang berkembang ketika udaranya tak beroksigen, juga lambat
(tiga putaran tiap sekon) dan diimbangi kelimpahannya: ia \omterm{def:b1:proteins:peptide}{protein}
terbanyak di Bumi, separuh \omterm{def:b1:proteins:peptide}{protein} terlarut sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun}.
\end{proposition}

\begin{definition}[Tumbuhan C4 dan CAM]\label{def:b1:photosynthesis:c4cam}
\index{tumbuhan C4}\index{tumbuhan CAM}
\emph{Tumbuhan C4} (jagung, tebu, sorgum, banyak rerumputan tropis)
menutup kebocorannya dengan memekatkan $\mathrm{CO_2}$: dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}
mesofilnya sebuah \omterm{def:b1:enzymes:enzyme}{enzim} tanpa afinitas terhadap $\mathrm{O_2}$ (PEP
karboksilase) memfiksasi bikarbonat menjadi asam berkarbon empat, yang
diantarkan ke dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} \emph{seludang berkas} di sekeliling tulang
\omterm{def:b1:flowering-plant-organization:organs}{daunnya} lalu didekarboksilasi di sana, sehingga menaikkan
$\mathrm{CO_2}$ di sekeliling \omterm{def:b1:photosynthesis:fixation}{RuBisCO} sepuluh kali dan menekan
oksigenasinya --- dengan biaya dua \omterm{def:b1:water-small-molecules:atp}{ATP} tambahan tiap $\mathrm{CO_2}$.
Kedua jenis \omterm{def:b1:cell-unit-of-life:cell}{selnya} bersama-sama membentuk anatomi \emph{Kranz}, yang
berarti ``karangan bunga''.
\emph{Tumbuhan CAM} (kaktus, nanas, agave) memisahkan kedua langkah yang
sama dalam waktu alih-alih dalam ruang: mereka membuka \omterm{def:b1:flowering-plant-organization:tissues}{stomatanya} pada
malam hari, menimbun $\mathrm{CO_2}$ sebagai asam malat dalam
\omterm{def:b1:eukaryotic-cell:plantcell}{vakuolanya}, lalu melepasnya ke \omterm{def:b1:photosynthesis:fixation}{RuBisCO} pada siang hari di balik \omterm{def:b1:flowering-plant-organization:tissues}{stomata}
yang tertutup, sehingga kehilangan sepersepuluh air yang hilang dari
tumbuhan C3 tiap karbon yang difiksasi.
\end{definition}

\includegraphics[width=0.6\linewidth]{images/book3/ai/b3-14-maize-kranz.jpg}

dikarangi \omterm{def:b1:cell-unit-of-life:cell}{sel} seludang berkas yang besar dan padat berisi \omterm{def:b1:photosynthesis:chloroplast}{kloroplas},

\begin{example}[Tumbuhan mana di mana]\label{ex:b1:photosynthesis:where}
Pada \qty{20}{\celsius} dalam udara lembap sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} gandum C3 dan
sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} jagung C4 memfiksasi karbon pada laju yang serupa, dan
jagungnya membayar dua \omterm{def:b1:water-small-molecules:atp}{ATP} lebih banyak tiap karbon. Pada
\qty{35}{\celsius} dalam udara kering gandumnya kehilangan sepertiga
fiksasinya bagi fotorespirasi sementara jagungnya tidak kehilangan
apa-apa: rerumputan C4 mendominasi negeri terbuka yang panas, tumbuhan
C3 tempat yang sejuk dan teduh. Sebatang kaktus memfiksasi dengan
lambat menurut kedua ukuran itu tetapi selamat di tempat yang akan
membuat keduanya mati kehausan.
\end{example}

\section{Autotrofi}

\begin{definition}[Autotrofi]\label{def:b1:photosynthesis:autotrophy}
Sebuah \omterm{def:b1:organism-environment:organism}{organisme} bersifat \emph{autotrof}\index{autotrofi} bila ia
membangun materi organiknya dari $\mathrm{CO_2}$
(\cref{ch:b1:organism-environment}). Ia memerlukan sebuah sumber energi
dan sebuah sumber elektron. \emph{\omterm{def:b1:organism-environment:trophy}{Fotoautotrof}} mengambil energinya dari
cahaya: tumbuhan, alga dan sianobakteri mengambil elektronnya dari air
lalu melepas oksigen (\emph{oksigenik}); bakteri ungu dan hijau
mengambilnya dari hidrogen sulfida atau asam organik dan tidak melepas
oksigen. \emph{\omterm{def:b1:organism-environment:trophy}{Kemoautotrof}} mengambil energi maupun elektronnya
dari oksidasi senyawa anorganik --- amonia, nitrit, sulfida, hidrogen,
besi fero --- sama sekali tanpa cahaya: bakteri penitrifikasi tanah, dan
bakteri yang memberi makan ekosistem lubang hidrotermal laut dalam.
Semuanya memakai \omterm{def:b1:photosynthesis:fixation}{daur Calvin}, atau sebuah daur yang berkerabat, untuk
memfiksasi karbonnya; semuanya mereduksinya dengan NADPH dan membayar
dengan \omterm{def:b1:water-small-molecules:atp}{ATP}; mereka hanya berbeda pada dari mana \omterm{def:b1:water-small-molecules:atp}{ATP} dan elektronnya
datang.
\end{definition}

\begin{example}[Neraca global]\label{ex:b1:photosynthesis:global}
Fotosintesis memfiksasi sekitar \qty{120}{Gt} karbon setahun di darat
dan \qty{50}{Gt} di lautan, separuhnya oleh alga bersel tunggal dan
sianobakteri yang tak terlihat mata; ia memutar karbon dioksida
atmosfer tiap tujuh tahun dan oksigennya tiap dua ribu tahun. Oksigennya
datang lewat jalan yang sama: selama dua miliar tahun sianobakteri
melepasnya ke dalam sebuah atmosfer yang tak memilikinya, dan tiap
\omterm{def:b1:organism-environment:organism}{organisme} aerob, dari \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria} ke atas, dibangun atas kebocoran satu
sistem \omterm{def:b1:enzymes:enzyme}{enzim} yang membelah air.
\end{example}

\section{Soal: Harga Sebuah Molekul Glukosa}

\begin{problem}[{Soal akhir pekan --- foton yang dihitung, elektron dan
proton yang diikuti menembus tilakoid, ATP serta NADPH yang
diperhitungkan dan dibelanjakan, panen harian sehelai daun yang
ditimbang, berakhir pada efisiensi energi fotosintesis}]
\label{pb:b1:photosynthesis:1}
Tetapan: $h = \qty{6.63e-34}{J.s}$, $c = \qty{3.00e8}{m/s}$,
$N_A = \num{6.02e23}$, $F = \qty{96500}{C/mol}$, $R = \qty{8.314}{J.mol^{-1}.K^{-1}}$,
$T = \qty{298}{K}$. \omterm{prop:b1:water-small-molecules:gibbs}{Energi bebas} pembakaran glukosa
\qty{2870}{kJ/mol}; sintesis \omterm{def:b1:water-small-molecules:atp}{ATP} dalam \omterm{def:b1:photosynthesis:chloroplast}{kloroplas} memakan biaya
\qty{50}{kJ/mol}; NADPH memegang \qty{220}{kJ/mol} daya reduksi.

\medskip
\noindent\textbf{Bagian I --- Foton dan elektron.}
\begin{enumerate}
  \item Hitunglah energi satu foton \qty{680}{nm} dan satu mol foton
        itu.
  \item Hitunglah energi satu mol foton \qty{700}{nm} dan satu mol
        foton \qty{450}{nm}. Mengapa cahaya biru tidak memberikan
        fotosintesis lebih banyak tiap fotonnya daripada cahaya merah?
  \item Berapa banyak foton \qty{680}{nm} yang membawa satu joule?
  \item Memindahkan satu elektron dari air ($E'_0 = \qty{+0.82}{V}$) ke
        $\mathrm{NADP^+}$ ($E'_0 = \qty{-0.32}{V}$) menuntut berapa
        banyak \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} tiap mol elektron ($\Delta G = -nF\Delta E$)?
  \item Dua foton dipakai tiap elektron (satu pada tiap \omterm{def:b1:photosynthesis:zscheme}{fotosistem}).
        Hitunglah energi yang dipasok tiap mol elektron dan fraksinya
        yang tersimpan dalam perubahan redoksnya.
  \item Berapa banyak elektron, dan dengan itu berapa banyak foton,
        yang dibutuhkan untuk membuat satu $\mathrm{O_2}$ dan dua NADPH?
  \item Berapa banyak foton tiap glukosa, bila 12 NADPH dibutuhkan?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Proton dan \omterm{def:b1:water-small-molecules:atp}{ATP}.}
Bagi tiap $\mathrm{O_2}$ yang dilepas, empat proton dibebaskan dalam
lumennya oleh pembelahan air dan delapan dipompa kompleks sitokrom
$b_6f$; \omterm{thm:b1:photosynthesis:chemiosmosis}{ATP sintase} memakai \qty{4}{H^+} tiap \omterm{def:b1:water-small-molecules:atp}{ATP}. Lumennya berada pada
pH 5 dan \omterm{def:b1:photosynthesis:chloroplast}{stromanya} pada pH 8; \omterm{def:b1:membranes-transport:potential}{potensial membran} melintasi \omterm{def:b1:photosynthesis:chloroplast}{tilakoidnya}
dapat diabaikan.
\begin{enumerate}[resume]
  \item Hitunglah gaya gerak protonnya dari selisih pH-nya, dalam volt
        ($\Delta\mu = 2.303\,RT\,\Delta\mathrm{pH}/F$), dan \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas}
        tiap mol protonnya.
  \item Hitunglah proton yang diantarkan ke lumennya tiap
        $\mathrm{O_2}$ dan \omterm{def:b1:water-small-molecules:atp}{ATP} yang dapat dibuatnya.
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibuat tiap glukosa lewat aliran lurusnya (6
        $\mathrm{O_2}$), lalu bandingkan dengan 18 yang dibutuhkan \omterm{def:b1:photosynthesis:fixation}{daur
        Calvin}. Bagaimana kekurangannya ditutup?
  \item Hitunglah \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} yang tersedia dari protonnya tiap \omterm{def:b1:water-small-molecules:atp}{ATP}
        (\qty{4}{H^+}) dan efisiensi sintasenya.
  \item Lumen sebuah \omterm{def:b1:photosynthesis:chloroplast}{kloroplas} bervolume sekitar \qty{1}{\micro m^3}.
        Berapa banyak proton bebas yang dipegangnya pada pH 5?
        Bandingkan dengan kira-kira $10^5$ proton yang melewati
        sintasenya tiap sekon, lalu simpulkan apa yang menyangga
        lumennya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Gula.}
\begin{enumerate}[resume]
  \item Tuliskan energi yang tersimpan tiap glukosa sebagai 18 \omterm{def:b1:water-small-molecules:atp}{ATP}
        $\times$ \qty{50}{kJ} ditambah 12 NADPH $\times$ \qty{220}{kJ},
        lalu bandingkan dengan \qty{2870}{kJ} glukosanya. Ke mana
        selisihnya pergi?
  \item Hitunglah energi 48 foton pada pertanyaan 6 dan efisiensi
        fotosintesis dari foton sampai glukosa.
  \item Hanya \qty{45}{\%} sinar matahari berada pada panjang gelombang
        yang diserap pigmennya, dan sepersepuluh dari itu dipantulkan
        atau diteruskan. Hitunglah efisiensinya dari sinar matahari
        datang sampai glukosa, sebelum respirasi.
  \item Sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} \qty{50}{cm^2} menerima \qty{400}{W/m^2} sinar
        matahari selama \qty{10}{h}. Hitunglah energi yang diterimanya
        dan, pada efisiensi pertanyaan 14, glukosa yang dibuatnya dalam
        gram
        (\qty{180}{g/mol}).
  \item \omterm{def:b1:flowering-plant-organization:organs}{Daun} itu merespirasikan \qty{40}{\%} glukosa tersebut untuk
        tetap hidup. Berapa untung bersihnya, dalam gram glukosa dan
        dalam gram karbon?
  \item Sebuah tanaman pertanian mencapai efisiensi keseluruhan
        \qty{1}{\%} dalam satu musim. Sebutkan tiga kehilangan, di luar
        yang sudah dihitung, yang memisahkan angka \omterm{def:b1:flowering-plant-organization:organs}{daunnya} dari angka
        tanamannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Kebocorannya.}
\omterm{def:b1:photosynthesis:fixation}{RuBisCO} di udara pada \qty{25}{\celsius} melakukan satu oksigenasi tiap
tiga karboksilasi; tiap oksigenasi memakan biaya setara setengah
$\mathrm{CO_2}$ yang difiksasi dan 3.5 \omterm{def:b1:water-small-molecules:atp}{ATP}.
\begin{enumerate}[resume]
  \item Tiap 100 karboksilasi, hitunglah oksigenasinya, karbon yang
        hilang dan karbon bersih yang diperoleh.
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibelanjakan bagi penyelamatannya tiap 100
        karboksilasi dan total \omterm{def:b1:water-small-molecules:atp}{ATP} tiap karbon bersih yang diperoleh
        (\omterm{def:b1:photosynthesis:fixation}{daur Calvin} 3 \omterm{def:b1:water-small-molecules:atp}{ATP} tiap karboksilasi ditambah penyelamatannya).
  \item Sebuah \omterm{def:b1:photosynthesis:c4cam}{tumbuhan C4} tidak beroksigenasi tetapi membelanjakan 2
        \omterm{def:b1:water-small-molecules:atp}{ATP} tambahan tiap $\mathrm{CO_2}$ yang diantarkan. Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP}
        tiap karbon bersihnya lalu bandingkan.
  \item Pada \qty{35}{\celsius} rasio oksigenasinya naik menjadi satu
        berbanding dua. Hitunglah ulang angka C3-nya lalu sebutkan
        rancangan mana yang menang.
  \item Jelaskan mengapa menaikkan $\mathrm{CO_2}$ sebuah rumah kaca
        menjadi tiga kali nilai atmosfernya menaikkan hasil tanaman C3
        (tomat, gandum) tetapi hampir tidak menaikkan hasil tanaman C4
        (jagung).
  \item Sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} C3 kehilangan sekitar 250 molekul air tiap
        $\mathrm{CO_2}$ yang difiksasi, sebuah \omterm{def:b1:photosynthesis:c4cam}{tumbuhan CAM} sekitar 25.
        Hitunglah air yang hilang dari masing-masing untuk memfiksasi
        \qty{1}{g} karbon.
  \item Nyatakan hasilnya: kebutuhan kuantum tiap $\mathrm{CO_2}$,
        efisiensinya dari foton sampai glukosa, dan efisiensinya dari
        sinar matahari sampai gula bersih \omterm{def:b1:flowering-plant-organization:organs}{daunnya}.
\end{enumerate}
