\chapter{Groups of Animals}\label{ch:g2:animal-groups}

There are so many \omterm{def:g1:plants-animals:animal}{animals}! Sorting them into groups helps us learn.
We start with simple groups you can see: \omterm{def:g1:plants-animals:animal}{animals} with backbones and
some easy \omterm{def:g2:animal-groups:vert}{vertebrate} classes.

\section{Many kinds of animals}

\begin{definition}[Animal groups]\label{def:g2:animal-groups:groups}
Scientists sort \omterm{def:g1:plants-animals:animal}{animals} by shared features: \omterm{def:g1:my-body:body}{body} cover, how they
breathe, how they have young, and whether they have a backbone.
A first big split is \omterm{def:g1:plants-animals:animal}{animals} \emph{with} a backbone and \omterm{def:g1:plants-animals:animal}{animals}
\emph{without}.
\end{definition}

\omimg{animal-groups.jpg}{0.52\linewidth}{\omterm{def:g2:animal-groups:five}{Birds}, \omterm{def:g2:animal-groups:five}{mammals}, \omterm{def:g2:animal-groups:five}{fish},
\omterm{def:g2:animal-groups:five}{reptiles} and \omterm{def:g2:animal-groups:five}{amphibians} --- different covers, same idea: sorted by
traits.}

\begin{definition}[Vertebrate]\label{def:g2:animal-groups:vert}
A \emph{vertebrate}\index{vertebrate} is an \omterm{def:g1:plants-animals:animal}{animal} with a backbone
(spine). \omterm{def:g2:animal-groups:five}{Fish}, \omterm{def:g2:animal-groups:five}{amphibians}, \omterm{def:g2:animal-groups:five}{reptiles}, \omterm{def:g2:animal-groups:five}{birds} and \omterm{def:g2:animal-groups:five}{mammals} are
vertebrates.
\end{definition}

\begin{definition}[Invertebrate]\label{def:g2:animal-groups:invert}
An \emph{invertebrate}\index{invertebrate} is an \omterm{def:g1:plants-animals:animal}{animal} without a
backbone. Insects, spiders, worms and snails are invertebrates.
\end{definition}

\section{Five vertebrate groups}

\begin{definition}[Five groups]\label{def:g2:animal-groups:five}
\begin{itemize}
  \item \emph{fish}\index{fish}: live in \omterm{def:g1:caring:needs}{water}; gills; scales;
  \item \emph{amphibians}\index{amphibian}: e.g.\ frogs; moist
        \omterm{def:g1:senses:five}{skin}; young often in \omterm{def:g1:caring:needs}{water};
  \item \emph{reptiles}\index{reptile}: e.g.\ snakes, lizards;
        dry scales; lay eggs on land;
  \item \emph{birds}\index{bird}: feathers; wings; lay eggs;
  \item \emph{mammals}\index{mammal}: hair or fur; feed young with
        milk; people are mammals.
\end{itemize}
\end{definition}

\begin{example}[Who am I?]\label{ex:g2:animal-groups:who}
I have feathers and a beak. I lay eggs. $\to$ \omterm{def:g2:animal-groups:five}{bird}. I have fur and
feed my baby milk. $\to$ \omterm{def:g2:animal-groups:five}{mammal}. I swim with scales and gills.
$\to$ \omterm{def:g2:animal-groups:five}{fish}.
\end{example}

\begin{method}[Sort an animal]\label{met:g2:animal-groups:sort}
\begin{enumerate}
  \item Does it have a backbone? (If you know it is a worm or
        insect, no.)
  \item Look at the cover: fur, feathers, scales, moist \omterm{def:g1:senses:five}{skin}?
  \item Where does it live? How does it have young?
  \item Choose the best group.
\end{enumerate}
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g2:animal-groups:1}
What is a \omterm{def:g2:animal-groups:vert}{vertebrate}? Name two.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animal-groups:2}
Is a beetle a \omterm{def:g2:animal-groups:vert}{vertebrate} or an \omterm{def:g2:animal-groups:invert}{invertebrate}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animal-groups:3}
Which group has feathers?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animal-groups:4}
True or false? ``People are \omterm{def:g2:animal-groups:five}{mammals}.''
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:animal-groups:5}
Sort: robin, shark, frog, cat, snake --- into the five \omterm{def:g2:animal-groups:vert}{vertebrate}
groups.
\end{exercise}
