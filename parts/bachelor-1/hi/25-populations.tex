\chapter{आबादियाँ और जनांकिकी}\label{ch:b1:populations}

यीस्ट का कोई फ्लास्क दस लाख \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} से शुरू होता है और सोलह घंटे बाद छह
करोड़ रखता है तथा बढ़ना बंद कर चुका होता है; किसी द्वीप पर हिरनों का कोई
झुंड हर साल दसवें भाग से बढ़ता है, उस जाड़े तक जब उनका भोजन चुक जाता है;
उत्तरी वन के लिंक्स हर दस साल पर चढ़ते और गिरते हैं, उन खरगोशों से एक
साल पीछे जिन्हें वे खाते हैं। हर एक एक \emph{\omterm{def:b1:populations:population}{आबादी}} है --- एक ही जगह में
एक ही जाति के प्राणी --- और हर एक ऐसे अंकगणित को मानती है जो लिखा जा
सकता है। यह अध्याय बताता है कि \omterm{def:b1:populations:population}{आबादियाँ} कैसे गिनी जाती हैं, उनके जन्म और
मृत्यु कैसे सारणीबद्ध किए जाते हैं, वृद्धि के दो समीकरण, वे बल जो
संख्याओं पर लगाम रखते हैं, और वही अंकगणित हमारी अपनी जाति के बारे में
क्या कहता है।

\section{किसी आबादी को गिनना}

\begin{definition}[आबादी, घनत्व, परिक्षेपण]\label{def:b1:populations:population}
\index{आबादी}\index{आबादी घनत्व}
\emph{आबादी} किसी दिए गए क्षेत्र में किसी दिए गए समय पर रहते और आपस में
प्रजनन करने योग्य एक ही जाति के प्राणियों का समुच्चय है। उसका
\emph{आकार} $N$ प्राणियों की संख्या है; उसका \emph{घनत्व} प्रति इकाई
क्षेत्रफल या आयतन संख्या; उसका \emph{परिक्षेपण} उनके फैलाव का ढर्रा ---
\emph{गुच्छित} जहाँ संसाधन या सामाजिक जीवन उन्हें बटोरते हों (झुंड,
मछलियों के दल, किसी पौधे के धब्बे), \emph{एकसमान} जहाँ वे एक-दूसरे को
हटाते हों (क्षेत्रधारी पक्षी, जल के लिए स्पर्धा करती मरुस्थली झाड़ियाँ),
\emph{यादृच्छ} जहाँ कोई भी न चलता हो। आबादियों की एक \emph{आयु संरचना},
एक \emph{लिंग अनुपात}, और जन्म, मृत्यु, आप्रवास तथा उत्प्रवास की दरें भी
होती हैं, जिनका संतुलन ही वृद्धि है।
\end{definition}

\begin{method}[किसी आबादी का आकलन]\label{met:b1:populations:counting}
\index{चिह्नन--पुनर्ग्रहण}
\begin{enumerate}
  \item अचल या धीमे \omterm{def:b1:organism-environment:organism}{जीव}: यादृच्छ रखे ज्ञात क्षेत्रफल के
        \emph{कोष्ठकों} में गिनिए, और माध्य घनत्व को कुल क्षेत्रफल से
        गुणा कीजिए; कोष्ठकों के बीच बिखराव परिशुद्धि देता है और
        परिक्षेपण उजागर करता है।
  \item गतिशील जंतु: \emph{चिह्नन--पुनर्ग्रहण}। $M$ प्राणी पकड़िए,
        चिह्नित कीजिए और छोड़ दीजिए; बाद में $C$ पकड़िए, जिनमें से
        $R$ चिह्नित हों। यदि चिह्नित स्वच्छंद घुल-मिल जाएँ और कुछ न
        बदला हो, तो दूसरी पकड़ का चिह्नित अंश \omterm{def:b1:populations:population}{आबादी} के चिह्नित अंश के
        बराबर है, $R/C = M/N$, इसलिए
        $N \approx MC/R$ (लिंकन--पीटरसन आकलन)।
  \item मान्यताएँ जाँचिए: दोनों पकड़ों के बीच कोई जन्म, मृत्यु या
        प्रवास नहीं, ऐसे चिह्न जो न गिरें और न जंतु का व्यवहार बदलें,
        बराबर पकड़-योग्यता। हर विफलता
        $N$ को एक पूर्वानुमेय दिशा में झुका देती है।
  \item जहाँ गिनना असंभव हो वहाँ परोक्ष चिह्न: विष्ठा, पगचिह्न,
        पुकारें, घोंसले, तालाब के एक लीटर जल का \omterm{def:b1:nucleic-acids:chain}{DNA}.
\end{enumerate}
\end{method}

\begin{example}[किसी चरागाह में वोल]\label{ex:b1:populations:voles}
एक हेक्टेयर चरागाह में साठ वोल पकड़े, चिह्नित किए और छोड़े गए; तीन रात
बाद अस्सी पकड़े जाते हैं, जिनमें से बारह चिह्न ढोते हैं:
$N \approx 60\times 80/12 = 400$ वोल, प्रति सौ वर्ग मीटर चार। यदि
चिह्नित वोल, यह सीखकर कि जालों में भोजन रहता है, अधिक सहजता से पकड़े
जाएँ, तो $R$ फूल जाता है और $N$ कम आँका जाता है; यदि चिह्नन ने उन्हें
जालों से डरा दिया हो, तो उलटा।
\end{example}

\section{जनांकिकी: आयु के अनुसार जन्म और मृत्यु}

\begin{definition}[जीवन सारणी]\label{def:b1:populations:lifetable}
\index{जीवन सारणी}\index{उत्तरजीविता}
कोई \emph{जीवन सारणी} साथ जन्मे प्राणियों के किसी समवर्ग को उनके जीवन भर
पीछा करती है। हर आयु $x$ के लिए वह $l_x$ दर्ज करती है, जो
\emph{उत्तरजीविता} है --- समवर्ग का वह अंश जो आयु
$x$ पर अब भी जीवित है --- और $m_x$, जो \emph{प्रजनकता} है --- आयु
$x$ की किसी मादा से जन्मी बेटियों की माध्य संख्या। उनसे: आयुओं के बीच
मर्त्यता, $q_x = 1 - l_{x+1}/l_x$; \emph{शुद्ध जनन
दर} $R_0 = \sum_x l_x m_x$, यानी कोई नवजात मादा अपने जीवनकाल में औसतन
जितनी बेटियाँ पाएगी; और \emph{पीढ़ी-काल}
$T = \sum_x x\,l_x m_x / R_0$, यानी उन बेटियों की माताओं की माध्य आयु।
$R_0 > 1$ वाली \omterm{def:b1:populations:population}{आबादी} बढ़ती है, $R_0 < 1$ वाली घटती है; एक पीढ़ी में वह
$R_0$ से गुणा हो जाती है।
\end{definition}

\begin{proposition}[उत्तरजीविता के तीन आकार]\label{prop:b1:populations:survivorship}
$\log l_x$ को आयु के सापेक्ष खींचने पर तीन तरह के वक्र मिलते हैं।
\emph{प्रकार I}: अधिकांश प्राणी बुढ़ापे तक जीते हैं और एक सँकरे अंतराल
में मरते हैं (बड़े स्तनधारी, धनी देशों के मनुष्य)। \emph{प्रकार II}: हर
आयु पर मृत्यु की अचर प्रायिकता, लघुगणकीय पैमाने पर एक सीधी रेखा (कई
पक्षी, छिपकलियाँ, कृंतक)। \emph{प्रकार III}: बच्चों में भारी मर्त्यता
और थोड़े बचे हुओं के लिए लंबा जीवन (सीप, मछलियाँ, अधिकांश पौधे और कीट)।
आकार \omterm{def:b1:organism-environment:organism}{जीव} के जीने के ढंग से निकलता है: जो जाति अपनी बहुत-सी संततियों में
से हर एक में थोड़ा लगाती है उसका वक्र प्रकार III का होता है; और जो थोड़ी
में से हर एक में बहुत लगाती है, उसका प्रकार I.
\end{proposition}

\begin{example}[किसी हिरन-झुंड के लिए एक जीवन सारणी]\label{ex:b1:populations:deer}
हज़ार शावकों में से छह सौ अपना पहला जन्मदिन देखते हैं, और तब उत्तरजीवी हर
साल अपनी संख्या का दसवाँ भाग खोते हैं जब तक कि थोड़े ही नौ पार करें;
मादाएँ दो की आयु से बच्चे देती हैं, अपने भरपूर वर्षों में साल में लगभग
आठ दसवाँ बेटी। $l_x m_x$ जोड़ने पर $R_0 \approx 1.5$ मिलता है: हर मादा
डेढ़ बेटी छोड़ती है, और झुंड लगभग चार साल की हर पीढ़ी में आधा बढ़ जाता
है --- यानी प्रति वर्ष ग्यारह प्रतिशत की बढ़त, जब तक कुछ उसे रोक न दे।
\end{example}

\section{वृद्धि}

\begin{theorem}[चरघातांकी वृद्धि]\label{thm:b1:populations:exponential}
\index{चरघातांकी वृद्धि}\index{अंतर्निहित वृद्धि दर}
जिस \omterm{def:b1:populations:population}{आबादी} की प्रति-व्यक्ति जन्म और मृत्यु दरें $b$ तथा $d$ अचर हों वह
अपने आकार के अनुपात में एक दर से बढ़ती है,
\[
\frac{\dd N}{\dd t} = rN, \qquad r = b - d, \qquad N(t) = N_0\,e^{rt},
\]
जहाँ $r$ \emph{\omterm{thm:b1:populations:exponential}{अंतर्निहित वृद्धि दर}} है। वह हर
$\ln 2/r$ पर दुगनी होती है; $r = \qty{0.11}{yr^{-1}}$ वाली कोई \omterm{def:b1:populations:population}{आबादी}
\qty{6.3}{yr} में दुगनी होती है, और $r = \qty{0.5}{h^{-1}}$ वाली
\qty{83}{min} में। किसी \omterm{def:b1:populations:lifetable}{जीवन सारणी} से, $r \approx \ln R_0 / T$.
\end{theorem}
\begin{proof}
किसी छोटे अंतराल $\dd t$ में जन्म $bN\,\dd t$ होते हैं और मृत्यु
$dN\,\dd t$, इसलिए $\dd N = (b - d)N\,\dd t$; और समीकरण
$\dd N/\dd t = rN$ का हल $N_0 e^{rt}$ है (क्योंकि
$e^{rt}$ का अवकलज $re^{rt}$ है)। दुगना होना: $e^{rt} = 2$ से $t = \ln 2/r$
मिलता है। एक पीढ़ी $T$ में \omterm{def:b1:populations:population}{आबादी} $R_0 = e^{rT}$ से गुणा होती है,
जिससे $r = \ln R_0/T$.
\end{proof}

\begin{theorem}[संभार-तांत्रिक वृद्धि]\label{thm:b1:populations:logistic}
\index{संभार-तांत्रिक वृद्धि}\index{वहन क्षमता}
यदि प्रति-व्यक्ति वृद्धि दर \omterm{def:b1:populations:population}{आबादी} के किसी \emph{\omterm{thm:b1:populations:logistic}{वहन क्षमता}} $K$ के पास
पहुँचते-पहुँचते रैखिक रूप से गिरे --- वह आकार जिसे पर्यावरण थाम सकता है
---
\[
\frac{\dd N}{\dd t} = rN\left(1 - \frac{N}{K}\right), \qquad
N(t) = \frac{K}{1 + \dfrac{K - N_0}{N_0}\,e^{-rt}} .
\]
जब तक $N \ll K$ रहे वृद्धि लगभग चरघातांकी होती है, $N = K/2$ पर सबसे
तेज़ (जहाँ $\dd N/\dd t = rK/4$), और $N = K$ पर लुप्त हो जाती है, जिसके
पास \omterm{def:b1:populations:population}{आबादी} एक पठार की तरह पहुँचती है: एक S-आकार का वक्र।
\end{theorem}
\begin{proof}
समीकरण को $\dd N/[N(1 - N/K)] = r\,\dd t$ के रूप में लिखिए और बाईं ओर को
$\dd N/N + \dd N/(K - N)$ में बाँटिए; समाकलन करने पर
$\ln[N/(K - N)] = rt + \text{स्थिरांक}$ मिलता है, और $N$ के लिए $N(0) =
N_0$ के साथ हल करने पर सूत्र मिल जाता है। $rN(1 - N/K)$ का
$N$ के सापेक्ष अवकलन करके उसे शून्य रखने पर अधिकतम $N = K/2$ पर मिलता है।
\end{proof}

\begin{proposition}[घनत्व-निर्भरता]\label{prop:b1:populations:densitydependence}
\index{घनत्व-निर्भरता}
घनत्व चढ़ने पर वृद्धि इसलिए धीमी पड़ती है कि भीड़ के साथ प्रति-व्यक्ति
जन्म दर गिरती है या मृत्यु दर चढ़ती है: प्रति सिर भोजन घटता है, क्षेत्र
मिलने बंद हो जाते हैं, अपशिष्ट और रोग जमा होते हैं, परभक्षी सिमट आते
हैं। ये \emph{घनत्व-निर्भर} कारक किसी \omterm{def:b1:populations:population}{आबादी} को $K$ की ओर नियमित करते
हैं। \emph{घनत्व-निरपेक्ष} कारक --- तुषार, बाढ़, आग, सूखा --- संख्या की
परवाह किए बिना एक अंश मार डालते हैं और नियमन नहीं कर सकते, केवल विक्षोभ
कर सकते हैं; कई कीटों की \omterm{def:b1:populations:population}{आबादियाँ} अधिकतर मौसम से शासित होती हैं और कभी
किसी \omterm{thm:b1:populations:logistic}{वहन क्षमता} के पास नहीं पहुँचतीं।
\end{proposition}
\begin{proof}[साक्ष्य]
गॉस (1934) ने \emph{Paramecium} को रोज़ बदले जाते माध्यम की नलियों में
उगाया: गिनतियाँ भोजन से तय किसी पठार तक एक संभार-तांत्रिक वक्र पर चलीं,
और भोजन दुगना करने पर पठार दुगना हो गया। पर्ल (1927) ने बोतलों में फल-मक्खियों
में और कार्लसन (1913) ने यीस्ट में यही पाया। खेत में, किसी द्वीप की
गीत-गौरैयाँ जब प्रजनन-आबादी बड़ी हो तो छोटे अंडे-गुच्छ देती हैं और अधिक
बच्चे खोती हैं, और क्षेत्रों की संख्या प्रजननकर्ताओं की संख्या पर ढक्कन
लगा देती है; पक्षी हटा देने पर उनकी जगह उस अधिशेष से भर जाती है जो अब तक
बाहर रखा गया था।
\end{proof}

\begin{definition}[जीवन-वृत्त की रणनीतियाँ]\label{def:b1:populations:rK}
\index{r-रणनीति}\index{K-रणनीति}
जातियाँ इस बात में भिन्न हैं कि वे दो चरम सिरों के बीच कहाँ बैठती हैं।
कोई \emph{$r$-वरित} जाति --- खरपतवार, एफ़िड, चूहे, अधिकांश मछलियाँ ---
ऊँचे $r$ के लिए गढ़ी होती है: वह जल्दी परिपक्व होती है, बहुत-सी छोटी
संततियाँ पैदा करती है, हर एक में थोड़ा लगाती है, थोड़ा जीती है, और स्पर्धा
आने से पहले नए या विक्षुब्ध आवासों का दोहन कर लेती है, प्रकार III की
\omterm{def:b1:populations:lifetable}{उत्तरजीविता} के साथ। कोई \emph{$K$-वरित} जाति --- बलूत, हाथी, अल्बाट्रॉस
--- $K$ के पास टिके रहने के लिए गढ़ी होती है: वह देर से परिपक्व होती
है, थोड़ी बड़ी संततियाँ पैदा करती है, उनकी देखभाल करती है, लंबा जीती है,
भीड़ भरे स्थिर आवासों में अच्छी स्पर्धा करती है, प्रकार I के वक्र के साथ।
अधिकांश जातियाँ दोनों मिलाती हैं; ये नाम पैमाने के सिरों के हैं।
\end{definition}

\begin{proposition}[परभक्षी और शिकार के चक्र]\label{prop:b1:populations:cycles}
\index{आबादी चक्र}
कुछ \omterm{def:b1:populations:population}{आबादियाँ} जमती नहीं बल्कि दोलन करती हैं। उत्तरी वन का हिम-खरगोश और
उसे खाने वाला लिंक्स लगभग दस साल के आवर्त के साथ साथ-साथ चढ़ते-गिरते
हैं, लिंक्स खरगोश से एक-दो साल पीछे: बहुत खरगोश बहुत लिंक्स पालते हैं,
बहुत लिंक्स खरगोशों को खा डालते हैं, थोड़े खरगोश लिंक्स को भूखा मारते
हैं, थोड़े लिंक्स खरगोशों को उबरने देते हैं। यह चक्र आंशिक रूप से परभक्षी
से और आंशिक रूप से खरगोशों के अपने भोजन से चलता है, जिसे अति-चराई के बाद
उबरने में वर्ष लगते हैं; युग्मित परभक्षी और शिकार के प्रतिरूप
(\cref{ch:b1:species-interactions}) ऐसे दोलन दो समीकरणों से फिर से पैदा
कर देते हैं।
\end{proposition}
\begin{proof}[साक्ष्य]
1840 के दशक से नब्बे साल तक रखे किसी व्यापारिक कंपनी के फ़र-लेखे एक
उपमहाद्वीप भर में हर साल लाए गए लिंक्स और खरगोश की खालों की संख्या दर्ज
करते हैं: दोनों श्रेणियाँ हर नौ-दस साल पर शिखर दिखाती हैं, लिंक्स का शिखर
खरगोश के पीछे, पूरे नौ चक्रों तक। तब से हुए क्षेत्र-अध्ययन, जिनमें
खरगोशों को परभक्षियों से बाड़कर या अतिरिक्त भोजन देकर रखा गया, दिखाते हैं
कि किसी एक कारक को हटाने से चक्र मंद पड़ जाता है और दोनों को हटाने से
मिट जाता है।
\end{proof}

\section{समस्या: फ्लास्क में यीस्ट और द्वीप पर हिरन}

\begin{problem}[{सप्ताहांत समस्या --- एक संवर्धि गिनी गई, एक झुंड
सारणीबद्ध हुआ, एक झील से नमूने लिए गए और एक कटाई तय हुई, और अंत में यीस्ट
की वहन क्षमता तथा अंतर्निहित वृद्धि दर}]
\label{pb:b1:populations:1}
यीस्ट की एक संवर्धि हर दो घंटे पर गिनी जाती है (\omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} प्रति मिलीलीटर,
$\times 10^5$): $10, 29, 71, 175, 351, 513, 595, 641, 656$, क्रमशः $t = 0,
2, \ldots, 16$ h पर। किसी द्वीप पर हिरनों की एक \omterm{def:b1:populations:population}{आबादी} की \omterm{def:b1:populations:lifetable}{जीवन सारणी} यह
है: $l_x = 1.00, 0.60, 0.52, 0.46, 0.40, 0.34, 0.26, 0.16, 0.06, 0$ और
$m_x = 0, 0, 0.6, 0.8, 0.8, 0.8, 0.7, 0.5, 0.3, 0$, जहाँ $x = 0$ से $9$
वर्ष तक है। किसी झील में 120
पर्च चिह्नित करके छोड़े जाते हैं; एक हफ़्ते बाद 150 पकड़े जाते हैं, 18
चिह्नित।

\medskip
\noindent\textbf{भाग I --- यीस्ट।}
\begin{enumerate}
  \item पहले दो घंटे \omterm{thm:b1:populations:exponential}{चरघातांकी वृद्धि} मानते हुए पहली दो गिनतियों से
        $r$ आँकिए।
  \item द्विगुणन-काल निकालिए।
  \item अंतिम गिनतियों से $K$ आँकिए।
  \item $t = 4$, $8$ और \qty{12}{h} पर संभार-तांत्रिक भविष्यवाणी
        निकालिए और गिनतियों से तुलना कीजिए।
  \item किस \omterm{def:b1:populations:population}{आबादी} पर वृद्धि सबसे तेज़ है, और वह कब आती है? अधिकतम
        वृद्धि दर \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} प्रति मिलीलीटर प्रति घंटा में निकालिए।
  \item चरघातांकी प्रतिरूप \qty{12}{h} पर क्या भविष्यवाणी करता? वह
        कितने गुना गलत है?
  \item यीस्ट के किसी बंद फ्लास्क में $K$ तय कर सकने वाली दो चीज़ें
        बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- हिरन।}
\begin{enumerate}[resume]
  \item पहले वर्ष में और पाँचवें में मर्त्यता $q_x$ निकालिए। यह
        \omterm{def:b1:populations:lifetable}{उत्तरजीविता} का कौन-सा प्रकार है?
  \item हर आयु के लिए $l_x m_x$ निकालिए और उन्हें जोड़िए: $R_0$.
  \item $T$ निकालिए।
  \item $r$ और झुंड का द्विगुणन-काल निकालिए।
  \item द्वीप पर आज 200 हिरन हैं। \omterm{thm:b1:populations:exponential}{चरघातांकी वृद्धि} से 10 साल बाद की
        संख्या की भविष्यवाणी कीजिए।
  \item द्वीप 600 हिरन पाल सकता है। \omterm{thm:b1:populations:logistic}{संभार-तांत्रिक वृद्धि} से 10 साल बाद
        की संख्या, और वह वर्ष जिसमें झुंड 500 पार करता है, बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- पर्च और कटाई।}
\begin{enumerate}[resume]
  \item झील की पर्च \omterm{def:b1:populations:population}{आबादी} आँकिए।
  \item यदि चिह्नित पर्चों में से 20 ने दूसरी पकड़ से पहले अपने चिह्न
        खो दिए हों, तो चिह्नित मछलियों की सही संख्या से फिर से निकालिए,
        और पहले आकलन की त्रुटि की दिशा बताइए।
  \item पर्च \omterm{def:b1:populations:population}{आबादी} का $K = 1000$ और $r = \qty{0.4}{yr^{-1}}$ है।
        अधिकतम सतत उपज और वह \omterm{def:b1:populations:population}{आबादी} निकालिए जिस पर वह मिलती है।
  \item $K$ से बहुत नीचे रहने पर पर्च \omterm{def:b1:populations:population}{आबादी} का द्विगुणन-काल निकालिए।
  \item क्लब इसके बजाय हर साल भंडार का एक निश्चित \qty{20}{\%} लेने का
        निर्णय करता है। साम्य \omterm{def:b1:populations:population}{आबादी} और उस साम्य पर वार्षिक पकड़
        निकालिए।
  \item किसी बंसीबाज़ क्लब की साल में 120 पर्च की पकड़ है। $\dd N/\dd t$
        को $N =
        500$ पर और $N = 300$ पर निकालकर दिखाइए कि \omterm{def:b1:populations:population}{आबादी} उसे सह सकती है
        या नहीं।
  \item प्रश्न 14 की मौसम और गिनती-त्रुटि को ध्यान में रखते हुए,
        $K/2$ पर कटाई व्यवहार में जोखिम भरी क्यों है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- नियमन।}
\begin{enumerate}[resume]
  \item झुंड के 400 पार करते ही हिरनों का $m_x$ आधा रह जाता है।
        दिखाइए कि तब $R_0$ 1 से नीचे गिर जाता है --- इससे झुंड का क्या
        होता है, और कैसा कारक काम कर रहा है?
  \item कोई कड़ा जाड़ा हिरनों की संख्या की परवाह किए बिना उनका
        \qty{40}{\%} मार डालता है। क्या यह कारक नियामक है? आगे के
        वर्षों में झुंड का क्या होता है?
  \item भेड़िए आ जाते हैं और साल में निश्चित 30 हिरन ले जाते हैं।
        $K = 600$ और प्रश्न 11 के $r$ के साथ संभार-तांत्रिक प्रतिरूप के
        लिए साम्य झुंड-आकार निकालिए ($rN(1 - N/K) = 30$ हल कीजिए), और
        बताइए कि कौन-सा साम्य स्थायी है।
  \item दो वाक्यों में समझाइए कि लिंक्स का चक्र खरगोश के चक्र से पीछे
        क्यों चलता है।
  \item परिणाम बताइए: यीस्ट की \omterm{thm:b1:populations:logistic}{वहन क्षमता} और \omterm{thm:b1:populations:exponential}{अंतर्निहित वृद्धि दर}, तथा
        हिरनों के $R_0$, $T$ और $r$.
\end{enumerate}
\end{problem}
