\chapter{Les défenses immunitaires}\label{ch:g9:immune-defenses}

Les portes sont tenues
(\cref{prop:g9:microbes-and-infection:doors}) --- et
pourtant, parfois, un envahisseur débarque~: les \omterm{def:g9:microbes-and-infection:sorted}{bactéries}
de l'écharde qui se multiplient dans le tissu chaud, le
\omterm{def:g9:microbes-and-infection:sorted}{virus} qui imprime déjà dans les \omterm{def:g5:first-look-at-cells:cell}{cellules} des voies
aériennes. Ce qui se passe ensuite est le troisième grand
système du corps après les \omterm{def:g5:brain-in-command:system}{nerfs} et les \omterm{def:g8:hormonal-communication:glands}{hormones}~: les
défenses immunitaires --- une armée permanente, un corps de
spécialistes doué de mémoire, et l'astuce de la \omterm{prop:g9:immune-defenses:vaccination}{vaccination}
qui l'entraîne à l'avance.

\section{La réponse rapide~: l'inflammation}

\begin{definition}[L'armée permanente]\label{def:g9:immune-defenses:standing}
Patrouillant dans le \omterm{prop:g4:journey-of-food:absorb}{sang} et les tissus se tiennent les
globules blancs\index{globule blanc} --- les troupes du
système immunitaire, fabriquées dans la moelle des \omterm{def:g3:bones-and-muscles:skeleton}{os}, qui
empruntent le réseau de
\cref{prop:g4:heart-and-blood:carries}. Les premières
arrivées sont les \emph{\omterm{def:g5:first-look-at-cells:cell}{cellules} avaleuses}~: des \omterm{def:g5:first-look-at-cells:cell}{cellules}
qui coulent autour d'un intrus et le digèrent (la goutte
mouvante de \cref{ex:g6:cell-unit-of-life:drop} exécutait
la même manœuvre en chasseuse libre --- le corps emploie
cette vieille astuce comme défense).
\end{definition}

\begin{proposition}[L'inflammation, lue correctement]\label{prop:g9:immune-defenses:inflammation}
Ce qui suit une écharde --- la rougeur, la chaleur, le
gonflement, l'élancement --- n'est pas l'\omterm{def:g9:microbes-and-infection:stages}{infection} qui
gagne, mais la défense qui arrive. Des signaux d'alarme
venus du tissu abîmé élargissent les vaisseaux du coin (le
déroutage de \cref{prop:g7:organs-at-work:supply},
réquisitionné)~: plus de \omterm{prop:g4:journey-of-food:absorb}{sang} chaud, des parois rendues
perméables, des avaleuses qui se faufilent hors des
\omterm{def:g7:heart-and-circulation:vessels}{capillaires} et rampent jusqu'au site, mangeant les \omterm{def:g9:microbes-and-infection:sorted}{bactéries}
en chemin. Le pus est le bilan du champ de bataille ---
avaleuses épuisées, envahisseurs morts, débris. La plupart
des débarquements se terminent là, en quelques jours, sans
qu'on le remarque.
\end{proposition}
\admitted

\begin{example}[Lire une coupure qui guérit]\label{ex:g9:immune-defenses:cut}
Jour un~: un halo rouge, chaud, douloureux --- vaisseaux
élargis, troupes qui arrivent. Jour deux~: un peu de pus ---
le bilan. Jour quatre~: le halo s'efface, la croûte
sèche --- la multiplication a perdu la course contre
l'engloutissement, et la reconstruction de la \omterm{def:g1:the-five-senses:sense}{peau}
(\cref{prop:g5:first-look-at-cells:growth}) referme la porte
derrière la guerre. Toute la campagne, livrée et gagnée
en dessous de l'échelle de l'inquiétude.
\end{example}

\section{Les spécialistes~: reconnaissance et mémoire}

\begin{proposition}[La réponse spécifique]\label{prop:g9:immune-defenses:specific}
Quand la réponse rapide est débordée, le corps de
spécialistes entre en lice --- plus lent, et précis~:
\begin{itemize}
  \item tout envahisseur porte des marques de surface ---
        ses \emph{antigènes}\index{antigène} --- des formes
        moléculaires que les spécialistes lisent comme
        étrangères (les marqueurs du soi de
        \cref{ex:g9:unique-individuals:medicine} sont la
        même lecture, tournée vers l'intérieur)~;
  \item parmi les \omterm{def:g9:immune-defenses:standing}{globules blancs} spécialistes, quelques-uns
        portent des récepteurs qui s'ajustent à \emph{cet}
        antigène-là~; le contact les sélectionne, et ils se
        multiplient en une armée de clones (la logique de la
        sélection, jouée à l'intérieur du corps en quelques
        jours)~;
  \item leurs armes~: les \emph{anticorps}\index{anticorps}
        --- des molécules ajustées, déversées dans le \omterm{prop:g4:journey-of-food:absorb}{sang},
        qui épinglent et agglutinent l'envahisseur pour les
        avaleuses --- et des spécialistes tueurs qui
        détruisent les propres \omterm{def:g5:first-look-at-cells:cell}{cellules} détournées du corps,
        seul moyen d'arrêter les imprimeries d'un \omterm{def:g9:microbes-and-infection:sorted}{virus}
        (\cref{ex:g9:microbes-and-infection:contrast})~;
  \item le coût de la bataille fait partie de la maladie~:
        la fièvre (un corps qui tourne chaud exprès ---
        beaucoup d'\omterm{prop:g9:microbes-and-infection:mostly}{agents pathogènes} se multiplient plus mal
        au chaud), les courbatures, les \emph{ganglions
        lymphatiques} gonflés --- les casernes des
        spécialistes, bondées en temps de guerre.
\end{itemize}
\end{proposition}
\admitted

\begin{definition}[La mémoire immunitaire]\label{def:g9:immune-defenses:memory}
Après la victoire, une partie de l'armée de clones se met en
réserve sous forme de \emph{cellules
mémoire}\index{cellule mémoire}~: des spécialistes de longue
vie qui conservent le signalement de l'antigène. Un second
débarquement du \emph{même} envahisseur rencontre un corps
préparé --- réponse en quelques heures, envahisseur balayé
avant les symptômes~: tu es \emph{immunisé}\index{immunité}.
Une rougeole, jamais deux~; la souche de l'hiver dernier
incapable de rallumer l'épidémie de cette année
(\cref{pb:g9:microbes-and-infection:1}). La mémoire est la
raison pour laquelle le même rhume semble ne jamais revenir
--- et pour laquelle les rhumes reviennent sans cesse~:
leurs \omterm{def:g9:microbes-and-infection:sorted}{virus} mutent vite
(\cref{prop:g9:genes-and-dna:explains}) vers des
signalements que la mémoire ne détient pas.
\end{definition}

\section{Entraîner la défense~: la vaccination}

\begin{proposition}[La vaccination]\label{prop:g9:immune-defenses:vaccination}
La \emph{vaccination}\index{vaccination} est une mémoire
immunitaire créée sans la maladie~: présente aux
spécialistes le \emph{signalement} de l'envahisseur --- des
\omterm{def:g6:producing-our-food:fermentation}{microbes} tués ou affaiblis, des fragments inoffensifs
porteurs de l'antigène --- et tout le programme de sélection
et de mémoire se déroule, sans le danger. Le corps entraîné
rencontre ensuite l'\omterm{prop:g9:microbes-and-infection:mostly}{agent pathogène} réel, si cela arrive,
comme un second débarquement. Deux conséquences portent tout
le poids de la santé publique~:
\begin{enumerate}
  \item les personnes vaccinées sont protégées avant la
        première bataille --- décisif pour les maladies dont
        la première bataille peut tuer ou mutiler~;
  \item assez de voisins vaccinés affament une épidémie en
        hôtes à contaminer (l'extinction de
        \cref{pb:g9:microbes-and-infection:1})~: la
        protection se met en commun --- le bouclier couvre
        ceux qui sont trop jeunes ou trop malades pour être
        eux-mêmes entraînés.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Du pari de la vaccine à l'éradication]\label{ex:g9:immune-defenses:history}
L'histoire va du pari de Jenner en 1796 --- les trayeuses
attrapaient la vaccine, bénigne, et jamais la variole~; une
inoculation de vaccine entraînait contre la tueuse --- en
passant par les vaccins délibérés à \omterm{def:g6:producing-our-food:fermentation}{microbes} affaiblis de
Pasteur (l'auteur de
\cref{ex:g9:microbes-and-infection:history}, qui complétait
sa théorie), jusqu'aux appels du vingtième siècle~: la
variole, vieux fléau de l'humanité, \emph{éradiquée} --- le
\omterm{def:g9:microbes-and-infection:sorted}{virus} disparu à l'état sauvage depuis la fin des années
1970 --- et la poliomyélite repoussée dans les derniers
coins de la carte. La mémoire entraînée, mise en commun, est
la seule arme qui ait jamais exterminé une maladie humaine.
\end{example}

\begin{method}[Lire n'importe quel événement immunitaire]\label{met:g9:immune-defenses:read}
Pour toute \omterm{def:g9:microbes-and-infection:stages}{infection}, guérison, vaccin ou résultat d'analyse~:
\begin{enumerate}
  \item situe la rencontre~: premier ou second
        débarquement~? (la mémoire change tout)~;
  \item nomme ceux qui répondent~: la réponse rapide des
        avaleuses, l'armée de clones des spécialistes, ou la
        mémoire en réserve~;
  \item lis les signes correctement~: l'inflammation et la
        fièvre comme la défense au travail, les ganglions
        gonflés comme des casernes~;
  \item pour un vaccin~: repère le signalement présenté, et
        dis avec qui sa protection est mise en commun.
\end{enumerate}
\end{method}

\section{Problème~: le registre des vaccinations}

\begin{problem}[{Problème du week-end --- une semaine de
centre de santé, lue par la méthode immunitaire}]\label{pb:g9:immune-defenses:1}
La semaine (inventée) d'un centre de santé~: lundi, les
\omterm{prop:g9:immune-defenses:vaccination}{vaccinations} des nourrissons~; mardi, la plaie d'un
agriculteur due à un fil de fer rouillé, et son rappel~;
mercredi, un cas de rougeole chez un enfant non vacciné, et
les contacts vérifiés~; jeudi, des analyses de \omterm{prop:g4:journey-of-food:absorb}{sang} qui
lisent les taux d'\omterm{prop:g9:immune-defenses:specific}{anticorps}~; vendredi, les vaccins annuels
du personnel contre la grippe.

\medskip\noindent\textbf{Partie I --- Les cas de la semaine.}
\begin{enumerate}
  \item Les nourrissons du lundi reçoivent des gouttes à
        \omterm{def:g6:producing-our-food:fermentation}{microbes} affaiblis et des injections de fragments.
        Nomme ce qui est présenté, et dis quel programme se
        déroule dans les semaines suivantes.
  \item L'agriculteur du mardi~: la plaie est nettoyée
        (quelle ligne de défense~?), et le rappel est fait.
        Que fait un \emph{rappel} à une mémoire entraînée
        --- et pourquoi le tétanos, entre toutes les
        maladies, n'autorise-t-il aucun pari sur la première
        bataille~?
  \item L'enfant du mercredi atteint de rougeole~: premier
        débarquement, programme complet. Établis la
        chronologie de la maladie~: réponse rapide débordée,
        armée de clones levée, guérison --- et dis ce qui se
        met en réserve après.
  \item Les contacts vérifiés se répartissent~: vaccinés
        (tranquilles), guéris il y a longtemps (tranquilles),
        et un nourrisson non vacciné, surveillé avec
        inquiétude. Lis chaque cas avec la première question
        de \cref{met:g9:immune-defenses:read}.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les mesures.}
\begin{enumerate}[resume]
  \item Les analyses du jeudi lisent les taux d'\omterm{prop:g9:immune-defenses:specific}{anticorps}
        contre des maladies et des vaccins passés. Que
        certifie un taux élevé, et quel stock de \omterm{def:g5:first-look-at-cells:cell}{cellules}
        implique-t-il~?
  \item Les taux d'un patient se sont estompés des années
        après la \omterm{prop:g9:immune-defenses:vaccination}{vaccination}, et un rappel est prescrit.
        Explique l'estompement et le rappel en termes de
        \omterm{def:g9:immune-defenses:memory}{cellules mémoire}.
  \item Le vaccin contre la grippe du vendredi est
        \emph{annuel}, contrairement aux programmes en une
        ou deux fois du lundi. Quelle propriété du \omterm{def:g9:microbes-and-infection:sorted}{virus} de
        la grippe (\cref{def:g9:immune-defenses:memory})
        impose ce réentraînement chaque année~?
  \item Pourquoi le centre vaccine-t-il son \emph{personnel}
        avec une insistance particulière~? (La protection de
        qui est-elle mise en commun, et où, dans un bâtiment
        plein de malades et de personnes fragiles~?)
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le sens du
registre.}
\begin{enumerate}[resume]
  \item Les parents de l'enfant non vacciné atteint de
        rougeole craignaient les vaccins et faisaient
        confiance aux premières batailles «~naturelles~».
        Rédige l'arithmétique douce du médecin~: compare le
        coût de l'entraînement à celui de la première
        bataille, pour la rougeole.
  \item Le nourrisson surveillé était trop jeune pour le
        vaccin. Dis qui était censé le protéger, et comment
        s'appelle ce devoir dans
        \cref{prop:g9:immune-defenses:vaccination}.
  \item Relie mardi et mercredi aux deux noms historiques de
        \cref{ex:g9:immune-defenses:history}~: l'idée de qui
        tournait dans chaque consultation~?
  \item Ferme le registre par le résumé de la semaine en une
        phrase~: ce que le centre administre réellement, dans
        le vocabulaire de ce chapitre.
\end{enumerate}
\end{problem}
