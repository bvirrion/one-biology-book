\chapter{Kerusakan Genom dan Kanker}\label{ch:g11:cancer}

Di bawah mikroskop, sepotong lapisan usus yang sehat adalah gambaran
ketertiban: \omterm{def:g10:cells-common-unit:cell}{sel} dalam lapisan yang rapi, masing-masing dengan \omterm{def:g10:cells-common-unit:organelle}{inti sel}
yang sederhana, membelah pada dasar lapisannya dan mati pada
puncaknya, seirama. Beberapa milimeter darinya, dalam potongan yang
sama, sebidang \omterm{def:g10:cells-common-unit:cell}{sel} telah melanggar barisan: berjejal, bentuknya ganjil,
dengan \omterm{def:g10:cells-common-unit:organelle}{inti sel} gelap yang besar, membelah di mana saja dan mendesak ke
dalam jaringan di bawahnya. Tiap \omterm{def:g10:cells-common-unit:cell}{sel} bidang itu berasal dari satu \omterm{def:g10:cells-common-unit:cell}{sel}
yang, beberapa tahun sebelumnya, memperoleh \omterm{def:g11:mutations:mutation}{mutasi} yang membiarkannya
membelah ketika ia tidak seharusnya --- lalu, sepanjang beberapa tahun,
beberapa \omterm{def:g11:mutations:mutation}{mutasi} lagi. Bab ini membahas bagaimana kendali \omterm{def:g11:cell-cycle-mitosis:cycle}{daur selnya}
gagal, apa yang merusaknya, dan apa yang dapat diperbuat sebelum dan
sesudahnya.

\section{Membelah ketika seharusnya tidak}

\begin{definition}[Kanker]\label{def:g11:cancer:cancer}
Sebuah \emph{kanker}\index{kanker} adalah populasi \omterm{def:g10:cells-common-unit:cell}{sel} yang membelah
tanpa kendali yang membatasi pembelahan \omterm{def:g10:cells-common-unit:cell}{sel} yang normal, dan yang
menyerbu jaringan di sekelilingnya. Ia bermula dengan satu \omterm{def:g10:cells-common-unit:cell}{sel} yang
\omterm{def:g10:universal-dna:information}{DNA-nya} telah menumpuk \omterm{def:g11:mutations:mutation}{mutasi} pada \omterm{def:g10:universal-dna:gene}{gen} yang mengendalikan \omterm{def:g11:cell-cycle-mitosis:cycle}{daur selnya};
keturunan \omterm{def:g10:cells-common-unit:cell}{sel} itu membentuk sebuah \emph{tumor}\index{tumor}. Tumor
yang tetap terbatas bersifat \emph{jinak}; tumor yang \omterm{def:g10:cells-common-unit:cell}{selnya} menyerbu
jaringan tetangganya dan berjalan lewat darah atau getah bening untuk
mendirikan tumor baru di tempat lain (\emph{metastasis}) bersifat \emph{ganas}.
\end{definition}

\begin{proposition}[Kanker adalah penyakit genom]\label{prop:g11:cancer:genome}
\omterm{def:g10:cells-common-unit:cell}{Sel} sebuah \omterm{def:g11:cancer:cancer}{tumor} membawa \omterm{def:g11:mutations:mutation}{mutasi} yang tidak dibawa \omterm{def:g10:cells-common-unit:cell}{sel} lain pemiliknya:
\omterm{def:g11:mutations:mutation}{mutasi} somatik, yang muncul dalam satu \omterm{def:g10:cells-common-unit:cell}{sel} dan diwarisi keturunannya.
\omterm{def:g11:cancer:cancer}{Tumornya} adalah sebuah \emph{klon}, dan ia berkembang --- \omterm{def:g10:cells-common-unit:cell}{sel} yang
membelah lebih cepat atau bertahan lebih baik mengambil alih, dan
\omterm{def:g11:mutations:mutation}{mutasi} yang belakangan menambahkan kemampuan menyerbu dan menyebar.
\end{proposition}
\begin{proof}[Bukti percobaan]
Membaca urutan \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:cells-common-unit:cell}{sel} \omterm{def:g11:cancer:cancer}{tumor} di samping \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:cells-common-unit:cell}{sel} normal pasiennya
menemukan ribuan beda, yang semuanya hadir dalam \omterm{def:g11:cancer:cancer}{tumornya}, tidak satu
pun dalam jaringan yang sehat; \omterm{def:g10:universal-dna:gene}{gen} yang sama dan sedikit jumlahnya
ditemukan bermutasi pada \omterm{def:g11:cancer:cancer}{tumor} demi \omterm{def:g11:cancer:cancer}{tumor}. Tiap \omterm{def:g10:cells-common-unit:cell}{sel} \omterm{def:g11:cancer:cancer}{tumor} seorang
perempuan membawa satu \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X yang sama dari kedua \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X-nya
dalam keadaan padam --- yaitu tanda keturunan dari satu \omterm{def:g10:cells-common-unit:cell}{sel}. Dan
\omterm{def:g11:mutations:mutagen}{mutagen} bersifat \omterm{def:g11:cancer:carcinogen}{karsinogen}: perantara yang menaikkan laju \omterm{def:g11:mutations:mutation}{mutasi}
(\cref{ch:g11:mutations}) menaikkan laju \omterm{def:g11:cancer:cancer}{kankernya}, dengan tunda
puluhan tahun.
\end{proof}

\section{Pedal gas dan rem}

\begin{proposition}[Dua macam gen mengendalikan daurnya]\label{prop:g11:cancer:genes}
Keputusan sebuah \omterm{def:g10:cells-common-unit:cell}{sel} untuk membelah diambil oleh \omterm{def:g11:gene-expression:protein}{protein} yang bekerja
seperti pedal gas dan rem sebuah kendaraan.
\begin{itemize}
  \item \emph{Proto-onkogen}\index{proto-onkogen} menyandikan \omterm{def:g11:gene-expression:protein}{protein}
        yang mendorong \omterm{def:g10:cells-common-unit:cell}{selnya} masuk ke daurnya ketika isyarat
        pertumbuhan tiba --- reseptor bagi isyarat pertumbuhan, dan
        penerus yang membawa isyaratnya ke \omterm{def:g10:cells-common-unit:organelle}{inti selnya}. \omterm{def:g11:mutations:mutation}{Mutasi} yang
        membuat \omterm{def:g11:gene-expression:protein}{protein} semacam itu aktif terus-menerus mengubah \omterm{def:g10:universal-dna:gene}{gennya}
        menjadi \emph{onkogen}: pedal gasnya terjepit turun. Satu \omterm{def:g10:universal-dna:gene}{alel}
        mutan sudah cukup.
  \item \emph{Gen penekan tumor}\index{gen penekan tumor} menyandikan
        \omterm{def:g11:gene-expression:protein}{protein} yang menghentikan daurnya --- pada pos pemeriksaan
        sebelum S dan sebelum \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} --- ketika \omterm{def:g10:universal-dna:information}{DNA-nya} rusak, dan
        yang memicu perbaikan atau, jika kerusakannya melampaui
        perbaikan, pemusnahan diri \omterm{def:g10:cells-common-unit:cell}{selnya}. \omterm{def:g11:mutations:mutation}{Mutasi} yang menonaktifkan
        \omterm{def:g11:gene-expression:protein}{protein} semacam itu menyingkirkan sebuah rem; karena \omterm{def:g10:cells-common-unit:cell}{selnya}
        punya dua \omterm{def:g10:universal-dna:gene}{alel}, keduanya harus hilang.
\end{itemize}
Penekan yang paling terkenal, yaitu \omterm{def:g11:gene-expression:protein}{protein} p53, dinonaktifkan pada
separuh seluruh \omterm{def:g11:cancer:cancer}{kanker} manusia.
\end{proposition}
\admitted

\begin{example}[Dua gen yang lazim dikenal]\label{ex:g11:cancer:genes}
Sebuah reseptor bagi isyarat pertumbuhan duduk pada membran banyak \omterm{def:g10:cells-common-unit:cell}{sel};
satu substitusi pada \omterm{def:g10:universal-dna:gene}{gennya} membuatnya mengisyaratkan terus-menerus,
tanpa isyarat apa pun dari luar --- yaitu onkogen yang ditemukan pada
sepertiga \omterm{def:g11:cancer:cancer}{kanker} usus besar dan paru-paru. \omterm{def:g10:universal-dna:gene}{Gen} p53, sebaliknya,
dinonaktifkan oleh \omterm{def:g11:mutations:mutation}{mutasi} dengan segala cara yang mungkin: substitusi
yang mengubah bentuknya, delesi, sebuah henti yang terlalu dini. \omterm{def:g10:cells-common-unit:cell}{Sel}
yang telah kehilangan p53 tidak berhenti untuk memperbaiki \omterm{def:g10:universal-dna:information}{DNA-nya}
sebelum menyalinnya, sehingga ia menumpuk \omterm{def:g11:mutations:mutation}{mutasi} berikutnya lebih
cepat: kehilangan rem itu mempercepat seluruh prosesnya.
\end{example}

\section{Beberapa langkah, bertahun-tahun}

\begin{proposition}[Kanker adalah proses banyak langkah]\label{prop:g11:cancer:multistep}
Satu \omterm{def:g11:mutations:mutation}{mutasi} tidak membuat sebuah \omterm{def:g11:cancer:cancer}{kanker}. Sebuah \omterm{def:g10:cells-common-unit:cell}{sel} harus menumpuk
beberapa \omterm{def:g11:mutations:mutation}{mutasi} --- lazimnya sebuah onkogen yang diaktifkan dan dua
penekan atau lebih yang hilang --- dan tiap \omterm{def:g11:mutations:mutation}{mutasi} berikutnya, dengan
menaikkan pembiakan klonnya, menaikkan jumlah \omterm{def:g10:cells-common-unit:cell}{sel} tempat \omterm{def:g11:mutations:mutation}{mutasi}
berikutnya dapat terjadi. Runtunannya memakan waktu bertahun-tahun
sampai puluhan tahun, dan itulah sebabnya kejadian sebagian besar
\omterm{def:g11:cancer:cancer}{kanker} naik tajam seiring umur, dan sebabnya pengaruh sebuah \omterm{def:g11:cancer:carcinogen}{karsinogen}
muncul dua puluh atau tiga puluh tahun setelah paparannya.
\end{proposition}
\begin{proof}[Bukti percobaan]
Pada usus besar, tahapnya dapat terlihat: polip jinak yang kecil yang
membawa satu \omterm{def:g11:mutations:mutation}{mutasi} penekan; polip yang lebih besar dengan tambahan
sebuah onkogen; \omterm{def:g11:cancer:cancer}{tumor} yang menyerbu dengan p53 yang juga sudah hilang
--- yaitu runtunan sekitar empat sampai enam \omterm{def:g11:mutations:mutation}{mutasi} sepanjang sepuluh
sampai dua puluh tahun, dengan tiap tahapnya ditemukan dalam \omterm{def:g10:cells-common-unit:cell}{sel} tahap
berikutnya. Kejadian \omterm{def:g11:cancer:cancer}{kanker} naik kira-kira sebagai pangkat lima
umurnya, yaitu lengkung yang diharapkan jika sekitar lima atau enam
peristiwa langka yang saling bebas harus terjadi semuanya dalam satu
\omterm{def:g10:cells-common-unit:cell}{sel}. Orang yang mewarisi satu \omterm{def:g10:universal-dna:gene}{alel} mutan sebuah \omterm{def:g10:universal-dna:gene}{gen} penekan mengidap
\omterm{def:g11:cancer:cancer}{kanker} yang sepadan lebih awal dan lebih sering, karena satu langkahnya
sudah terambil dalam setiap \omterm{def:g10:cells-common-unit:cell}{selnya}.
\end{proof}

\section{Apa yang merusak genomnya}

\begin{definition}[Karsinogen]\label{def:g11:cancer:carcinogen}
Sebuah \emph{karsinogen}\index{karsinogen} adalah perantara yang
menaikkan risiko \omterm{def:g11:cancer:cancer}{kanker}. Sebagian besarnya bersifat \omterm{def:g11:mutations:mutagen}{mutagen}:
ultraviolet cahaya matahari (\omterm{def:g11:cancer:cancer}{kanker} kulit), tar asap tembakau
(paru-paru, mulut, kandung kemih), radiasi pengion, serat asbes,
alkohol, beberapa kapang. Sebagian bekerja dengan cara lain: virus
tertentu membawa \omterm{def:g10:universal-dna:gene}{gen} yang menjepit rem \omterm{def:g10:cells-common-unit:cell}{selnya} --- papilomavirus manusia
menyebabkan hampir seluruh \omterm{def:g11:cancer:cancer}{kanker} leher rahim, virus hepatitis banyak
\omterm{def:g11:cancer:cancer}{kanker} hati --- dan peradangan menahun atau hormon yang menjaga sebuah
jaringan tetap membelah menaikkan jumlah pembelahan tempat \omterm{def:g11:mutations:mutation}{mutasi} dapat
terjadi.
\end{definition}

\begin{example}[Tembakau]\label{ex:g11:cancer:tobacco}
Asapnya membawa sekitar tujuh puluh \omterm{def:g11:cancer:carcinogen}{karsinogen}; benzopiren, setelah
diaktifkan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} \omterm{def:g10:cells-common-unit:cell}{selnya} sendiri, menempel pada guanin dan menyebabkan
substitusi pada kedudukan itu. \omterm{def:g10:universal-dna:gene}{Gen} p53 sebuah \omterm{def:g11:cancer:cancer}{tumor} paru-paru perokok
lazimnya membawa persis substitusi semacam itu, pada salah satu dari
beberapa kedudukan yang benzopirennya paling mudah mengikat --- \omterm{def:g11:mutations:mutagen}{mutagen}
itu meninggalkan tanda tangannya dalam urutannya. \omterm{def:g11:cancer:cancer}{Kanker} paru-paru
merupakan hal yang langka sebelum ada rokok; merokok menyebabkan
sembilan kasus dari sepuluh, dan berhenti membagi dua risikonya dalam
sepuluh tahun, karena \omterm{def:g10:cells-common-unit:cell}{sel} yang membawa langkah awalnya berangsur digantikan.
\end{example}

\begin{proposition}[Kecenderungan yang diwariskan]\label{prop:g11:cancer:inherited}
\omterm{def:g11:cancer:cancer}{Kankernya} sendiri tidak diwariskan --- ia adalah penyakit \omterm{def:g11:mutations:mutation}{mutasi}
somatik --- tetapi \omterm{def:g10:universal-dna:gene}{alel} mutan sebuah \omterm{def:g10:universal-dna:gene}{gen} penekan dapat diwariskan.
Orang yang mewarisi satu \omterm{def:g10:universal-dna:gene}{alel} \omterm{def:g10:universal-dna:gene}{gen} semacam itu yang tidak bekerja telah
mengambil satu langkah dalam setiap \omterm{def:g10:cells-common-unit:cell}{sel} tubuhnya; satu \omterm{def:g11:mutations:mutation}{mutasi} somatik
pada \omterm{def:g10:universal-dna:gene}{alel} yang lain merampungkan kehilangannya dalam \omterm{def:g10:cells-common-unit:cell}{sel} mana pun,
alih-alih dua \omterm{def:g11:mutations:mutation}{mutasi} yang dituntut pada orang lain. Hasilnya adalah
risiko yang tinggi bagi \omterm{def:g11:cancer:cancer}{kanker} yang sepadan, lebih awal dalam hidupnya:
sekitar 70\% perempuan yang membawa \omterm{def:g10:universal-dna:gene}{alel} mutan sebuah \omterm{def:g10:universal-dna:gene}{gen} \omterm{def:g11:cancer:cancer}{kanker}
payudara tertentu mengidap penyakitnya, berhadapan dengan 12\% pada
umumnya.
\end{proposition}
\admitted

\section{Pencegahan, pengenalan, pengobatan}

\begin{method}[Menurunkan risikonya]\label{met:g11:cancer:prevention}
Karena sebagian besar \omterm{def:g11:mutations:mutation}{mutasi} di balik \omterm{def:g11:cancer:cancer}{kanker} berasal dari paparan yang
dapat dikenali, dan karena prosesnya memakan bertahun-tahun, \omterm{def:g11:cancer:cancer}{kanker}
sebagian besar dapat dicegah dan dikenali.
\begin{enumerate}
  \item \emph{Jauhi \omterm{def:g11:mutations:mutagen}{mutagennya}}: tanpa tembakau (sepertiga seluruh
        kematian \omterm{def:g11:cancer:cancer}{kanker}), sedikit alkohol, perlindungan dari matahari,
        tanpa radiasi yang tidak perlu.
  \item \emph{Vaksinasi} terhadap virus penyebab \omterm{def:g11:cancer:cancer}{kanker}: vaksin
        papilomavirus mencegah hampir seluruh \omterm{def:g11:cancer:cancer}{kanker} leher rahim jika
        diberikan sebelum paparannya.
  \item \emph{Penapisan}: temukan \omterm{def:g11:cancer:cancer}{tumornya} pada tahap polip --- uji
        apus leher rahim, kolonoskopi, mamografi --- ketika
        menyingkirkannya masih sederhana dan menyembuhkan.
  \item \emph{Kenali keluarganya}: kecenderungan yang diwariskan
        menuntut pengawasan yang lebih awal dan lebih ketat.
\end{enumerate}
\end{method}

\begin{example}[Pengobatan dan apa yang disasarnya]\label{ex:g11:cancer:treatments}
Pembedahan menyingkirkan \omterm{def:g11:cancer:cancer}{tumor} yang belum menyebar. Radioterapi
memutus \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:cells-common-unit:cell}{sel} dalam berkasnya melampaui perbaikan, sehingga membunuh
\omterm{def:g10:cells-common-unit:cell}{sel} yang membelah lebih dahulu. Kemoterapi memakai obat yang
menghalangi replikasi atau gelendongnya
(\cref{ch:g11:cell-cycle-mitosis}): \omterm{def:g10:cells-common-unit:cell}{sel} \omterm{def:g11:cancer:cancer}{tumor}, yang membelah
terus-menerus, mati lebih cepat daripada \omterm{def:g10:cells-common-unit:cell}{sel} normal --- tetapi jaringan
normal yang cepat membelah (sumsum tulang, lapisan usus, akar rambut)
ikut menderita, dan dari sanalah efek sampingnya datang. Pengobatan
yang lebih baru menyasar \omterm{def:g11:gene-expression:protein}{protein} mutan tertentu sebuah \omterm{def:g11:cancer:cancer}{tumor}, atau
melepaskan sistem kekebalan kepadanya.
\end{example}

\begin{remark}[Mengapa kita semua membawa awalnya]\label{rem:g11:cancer:everyone}
Tiap orang dewasa membawa klon \omterm{def:g10:cells-common-unit:cell}{sel} dengan \omterm{def:g11:mutations:mutation}{mutasi} yang berkaitan dengan
\omterm{def:g11:cancer:cancer}{kanker} --- pada kulit, usus, darahnya --- yang tidak pernah menjadi
\omterm{def:g11:cancer:cancer}{kanker}: rem yang tersisa masih menahan, sistem kekebalannya
menyingkirkan yang terburuk, dan orangnya meninggal karena hal lain
lebih dahulu. \omterm{def:g11:cancer:cancer}{Kanker} adalah hasil sesekali sebuah proses yang berjalan
pada setiap orang; tujuan pencegahannya adalah menjaga jumlah langkah
yang terambil, dalam satu \omterm{def:g10:cells-common-unit:cell}{sel} mana pun, di bawah jumlah yang diperlukan.
\end{remark}

\section{Soal: Tanda Tangan dalam Gennya}

\begin{problem}[{Soal akhir pekan --- sebuah tumor paru-paru yang
dibaca urutannya: mutasinya yang terhitung, tanda tangan perokoknya
yang terbaca, tahunnya yang diperhitungkan, dan risiko yang dibawa
sebatang rokok}]
\label{pb:g11:cancer:1}
Sebuah \omterm{def:g11:cancer:cancer}{tumor} paru-paru dan jaringan sehat dari pasien yang sama, yaitu
perokok 20 batang sehari selama 35 tahun, dibaca urutannya. \omterm{def:g11:cancer:cancer}{Tumornya}
membawa $\num{5000}$ substitusi somatik yang tidak ditemukan dalam
jaringan yang sehat; 75\% di antaranya menggantikan G dengan T, yaitu
perubahan yang disebabkan benzopiren. Di antaranya: satu pada \omterm{def:g10:universal-dna:gene}{gen} p53,
yang menciptakan henti yang terlalu dini, pada satu \omterm{def:g10:universal-dna:gene}{alel}; \omterm{def:g10:universal-dna:gene}{alel} p53 yang
lain terhapus; dan satu pada \omterm{def:g10:universal-dna:gene}{gen} reseptor isyarat pertumbuhan, yang
membuat \omterm{def:g11:gene-expression:protein}{proteinnya} aktif terus-menerus.

\medskip
\noindent\textbf{Bagian I --- Membaca urutannya.}
\begin{enumerate}
  \item Mengapa ke-$\num{5000}$ substitusinya disebut somatik? Di mana
        substitusinya akan ditemukan seandainya bersifat \omterm{def:g10:cells-common-unit:cell}{sel} kelamin?
  \item Pada \omterm{def:g11:cancer:cancer}{tumor} orang yang tidak merokok, substitusinya berupa
        segala macam dengan perbandingan yang kira-kira sama. Apa yang
        diberitahukan 75\% perubahan G-ke-T itu kepadamu?
  \item Golongkan \omterm{def:g11:mutations:mutation}{mutasi} p53-nya (dengan istilah
        \cref{ch:g11:gene-expression}) dan pengaruhnya pada \omterm{def:g11:gene-expression:protein}{proteinnya}.
  \item Jelaskan mengapa kedua \omterm{def:g10:universal-dna:gene}{alel} p53-nya harus hilang, tetapi satu
        \omterm{def:g10:universal-dna:gene}{alel} mutan \omterm{def:g10:universal-dna:gene}{gen} reseptornya sudah cukup.
  \item Mana dari kedua \omterm{def:g10:universal-dna:gene}{gen} itu yang menjadi pedal gas dan mana yang menjadi rem?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Tahun-tahunnya.}
\begin{enumerate}[resume]
  \item Pasiennya mengisap sekitar $20 \times 365 \times 35$
        batang rokok. Hitung jumlahnya.
  \item Jika tiap batang rokok menghasilkan, rata-rata, satu
        substitusi yang menetap dalam tiap \omterm{def:g10:cells-common-unit:cell}{sel} lapisan paru-parunya,
        berapa substitusi yang akan dibawa sebuah \omterm{def:g10:cells-common-unit:cell}{sel} setelah 35 tahun?
        Bandingkan dengan $\num{5000}$ yang ditemukan.
  \item \omterm{def:g11:mutations:mutation}{Mutasi} \omterm{def:g11:cancer:cancer}{tumornya} tidak seluruhnya dalam urutan \omterm{def:g10:cells-common-unit:cell}{sel} yang sama:
        \omterm{def:g11:mutations:mutation}{mutasi} p53 dan reseptornya ada dalam tiap \omterm{def:g10:cells-common-unit:cell}{sel} \omterm{def:g11:cancer:cancer}{tumornya},
        sedangkan ribuan lainnya hanya dalam sebagiannya. Apa yang
        dinyatakannya tentang kapan kedua \omterm{def:g11:mutations:mutation}{mutasi} pentingnya terjadi?
  \item \omterm{def:g11:cancer:cancer}{Tumornya} berukuran \qty{3}{cm} melintang, sekitar $10^{10}$
        \omterm{def:g10:cells-common-unit:cell}{sel}, dan \omterm{def:g10:cells-common-unit:cell}{sel} \omterm{def:g11:cancer:cancer}{tumornya} berlipat dua tiap 100 hari. Berapa
        pelipatduaan dari satu \omterm{def:g10:cells-common-unit:cell}{sel}, dan berapa tahun, sekurang-kurangnya?
  \item Gabungkan pertanyaan 8 dan 9: kira-kira kapan dalam 35 tahun
        itu \omterm{def:g11:mutations:mutation}{mutasi} penting yang terakhir terjadi?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Risiko sebatang rokok.}
\begin{enumerate}[resume]
  \item Dari gambar tembakaunya, risiko nisbi pasiennya sekitar 17.
        Jika 1\% orang yang tidak merokok mengidap \omterm{def:g11:cancer:cancer}{kanker} paru-paru,
        berapa bagian perokok seperti dia yang mengidapnya?
  \item Di antara 100 perokok semacam itu, berapa \omterm{def:g11:cancer:cancer}{kanker} yang dapat
        ditimpakan pada merokoknya?
  \item Seandainya ia berhenti pada umur 40 (setelah 20 tahun),
        risikonya akan menjadi sekitar sepertiga risiko perokok yang
        terus merokok. Jelaskan, dengan model banyak langkahnya dan
        pembaruan lapisan paru-parunya, mengapa risikonya turun setelah
        berhenti.
  \item Mengapa \omterm{def:g11:cancer:cancer}{kanker} seorang perokok, ketika datang, lazimnya datang
        setelah tiga puluh tahun dan bukan setelah lima tahun?
  \item Seorang murid berpendapat bahwa karena sebatang rokok
        menyebabkan ``hanya satu \omterm{def:g11:mutations:mutation}{mutasi} tiap \omterm{def:g10:cells-common-unit:cell}{sel}'', rokok itu tak
        berbahaya. Jawablah dengan hitungan pertanyaan 7 dan jumlah \omterm{def:g10:cells-common-unit:cell}{selnya}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pencegahan dan populasinya.}
Di sebuah negeri berpenduduk 60 juta, 25\% orang dewasanya merokok;
\omterm{def:g11:cancer:cancer}{kanker} paru-paru membunuh $\num{30000}$ orang tiap tahun, 90\% di
antaranya perokok.
\begin{enumerate}[resume]
  \item Hitung kematian \omterm{def:g11:cancer:cancer}{kanker} paru-paru tiap tahun di antara
        perokoknya dan di antara yang tidak merokok, serta laju
        kematian tiap 100\,000 pada tiap kelompoknya (ambil 45 juta
        orang dewasa).
  \item Dengan faktor berapa lajunya lebih tinggi di antara
        perokoknya? Bandingkan dengan risiko nisbi pada gambarnya lalu
        beri tanggapan.
  \item Jika merokok lenyap hari ini, berapa jumlah kematian \omterm{def:g11:cancer:cancer}{kanker}
        paru-paru tiap tahun dalam jangka panjang? Mengapa tidak seketika?
  \item Penapisan lewat pindaian dosis rendah menemukan \omterm{def:g11:cancer:cancer}{tumor} lebih
        awal dan menurunkan kematian di antara perokok berat sekitar
        20\%. Bandingkan, dalam kematian yang terhindarkan tiap tahun,
        menapis setiap perokok dengan berhenti merokok.
  \item Nyatakan hasilnya: apa yang dibuktikan tanda tangan G-ke-T
        tentang asal \omterm{def:g11:mutations:mutation}{mutasi} \omterm{def:g11:cancer:cancer}{tumornya}, ketiga peristiwa genetik yang
        membuat \omterm{def:g11:cancer:cancer}{kankernya}, dan satu keputusan yang akan mencegahnya.
\end{enumerate}
\end{problem}
