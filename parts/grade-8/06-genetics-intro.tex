\section{Chromosomes carry genes}

\begin{definition}[Chromosome]\label{def:g8:genetics:chromosome}
A \emph{chromosome}\index{chromosome} is a long DNA molecule packaged
with \omterm{def:g6:nutrition:groups}{proteins}. In eukaryotes, \omterm{def:g7:heredity:gene}{chromosomes} reside in the \omterm{def:g8:organelles:nucleus}{nucleus}. \omterm{def:g1:my-body:body}{Body}
\omterm{def:g7:microscope:cell}{cells} of a \omterm{def:g6:classification:species}{species} have a \omterm{def:g3:classify:char}{characteristic} chromosome number (humans: 46,
as 23 pairs).
\end{definition}
\omimg{dna-helix.jpg}{0.50\linewidth}{Genes are carried on DNA, the twisted double helix.}


\begin{definition}[Gene and allele]\label{def:g8:genetics:gene}
A \emph{gene}\index{gene} is a DNA segment that influences one or more
\omterm{def:g7:heredity:trait}{traits}. \emph{Alleles}\index{allele} are alternative versions of a gene
(for example flower-colour \omterm{def:g7:heredity:allele}{alleles}).
\end{definition}

\begin{definition}[Homologous pair]\label{def:g8:genetics:homolog}
\emph{Homologous chromosomes}\index{homologous chromosomes} are a pair
(one from each parent) that carry the same \omterm{def:g8:genetics:gene}{genes} in the same order, though
\omterm{def:g7:heredity:allele}{alleles} may differ.
\end{definition}

\section{Body cells and gametes}

\begin{definition}[Diploid and haploid]\label{def:g8:genetics:ploid}
\emph{Diploid}\index{diploid} \omterm{def:g7:microscope:cell}{cells} ($2n$) have homologous pairs ---
typical \omterm{def:g1:my-body:body}{body} \omterm{def:g7:microscope:cell}{cells}. \emph{Haploid}\index{haploid} \omterm{def:g7:microscope:cell}{cells} ($n$) have one
set --- \omterm{def:g6:reproduction:sexual}{gametes} (sperm and \omterm{def:g3:life-cycles:butterfly}{egg} in \omterm{def:g1:plants-animals:animal}{animals}).
\end{definition}

\begin{proposition}[Fertilisation restores diploidy]\label{prop:g8:genetics:fert}
When \omterm{def:g8:genetics:ploid}{haploid} \omterm{def:g6:reproduction:sexual}{gametes} fuse, the zygote is \omterm{def:g8:genetics:ploid}{diploid} again. \omterm{def:g2:growing:offspring}{Offspring} inherit
half their \omterm{def:g7:heredity:gene}{chromosomes} from each parent --- a source of \omterm{def:g5:evolution:variation}{variation}.
\end{proposition}
\admitted

\section{Genotype and phenotype}

\begin{definition}[Genotype and phenotype]\label{def:g8:genetics:pheno}
\emph{Genotype}\index{genotype}: the \omterm{def:g7:heredity:allele}{alleles} an \omterm{def:g4:ecosystems:levels}{organism} has.
\emph{Phenotype}\index{phenotype}: the \omterm{def:g7:heredity:trait}{trait} observed. \omterm{def:g4:traits:env}{Environment} can
modify phenotype.
\end{definition}

\begin{definition}[Dominant and recessive]\label{def:g8:genetics:dom}
In simple cases, a \emph{dominant}\index{dominant allele} \omterm{def:g8:genetics:gene}{allele} shows
its effect in the \omterm{def:g8:genetics:pheno}{phenotype} when one copy is present; a
\emph{recessive}\index{recessive allele} \omterm{def:g8:genetics:gene}{allele} shows only when two copies
are present. Real genetics is often more complex --- this is a starter
model.
\end{definition}

\begin{example}[Notation]\label{ex:g8:genetics:note}
Use $B$ for a \omterm{def:g8:genetics:dom}{dominant allele} and $b$ for \omterm{def:g8:genetics:dom}{recessive}. \omterm{def:g8:genetics:pheno}{Genotypes} $BB$ and
$Bb$ may share a \omterm{def:g8:genetics:pheno}{phenotype}; $bb$ shows the \omterm{def:g8:genetics:dom}{recessive} \omterm{def:g8:genetics:pheno}{phenotype}.
\end{example}

\begin{method}[Reading a simple trait story]\label{met:g8:genetics:read}
\begin{enumerate}
  \item Identify the \omterm{def:g8:genetics:pheno}{phenotype} in the question.
  \item Ask which \omterm{def:g8:genetics:pheno}{genotypes} could produce it under a \omterm{def:g8:genetics:dom}{dominant}/\omterm{def:g8:genetics:dom}{recessive}
        model.
  \item Remember: two \omterm{def:g8:genetics:dom}{recessive alleles} needed for a \omterm{def:g8:genetics:dom}{recessive} \omterm{def:g8:genetics:pheno}{phenotype}.
\end{enumerate}
\end{method}
