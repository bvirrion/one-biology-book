\section*{Chapter \ref{ch:g12:diversification-of-life} --- Diversification of Living Things}

\begin{solution}{exo:g12:diversification-of-life:1}
\omterm{def:g10:universal-dna:gene}{Gene} duplication and divergence; \omterm{prop:g11:antibiotic-resistance:variation}{horizontal gene transfer};
hybridisation with polyploidy; changes in the regulation of
\omterm{prop:g12:diversification-of-life:development}{developmental genes}; \omterm{def:g12:diversification-of-life:symbiosis}{symbiosis}; transmitted behaviour.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:2}
A set of related \omterm{def:g10:universal-dna:gene}{genes} descended from one ancestral \omterm{def:g10:universal-dna:gene}{gene} by
successive duplications, each copy then diverging by \omterm{def:g11:mutations:mutation}{mutation} to a
distinct function (\omterm{def:g11:the-eye:photoreceptors}{opsins}, globins).
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:3}
Its \omterm{def:g10:universal-dna:information}{chromosomes}, one set from each parent \omterm{def:g10:biodiversity-scales:species}{species}, have no homologues
to pair with at \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis}, so gametes are unbalanced. Doubling gives
every \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} an identical partner: pairing and \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis} become
regular and the plant is fertile.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:4}
Their own circular \omterm{def:g10:universal-dna:information}{DNA}; bacterial-type \omterm{def:g10:cells-common-unit:organelle}{ribosomes} sensitive to
antibacterial \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotics}; multiplication by division; a double
membrane; sequences closest to a group of living bacteria.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:5}
A behaviour learnt from other members of the group and passed on
without \omterm{def:g10:universal-dna:gene}{genes}: nut-cracking with stones in some chimpanzee
communities and not others.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:6}
$14 + 7 = 21$ \omterm{def:g10:universal-dna:information}{chromosomes}: 7 A from einkorn, 7 A and 7 B from emmer.
Seven pairs (A with A) can form; the 7 B \omterm{def:g10:universal-dna:information}{chromosomes} have no partner
and are distributed at random, so the gametes are unbalanced: sterile.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:7}
The $\alpha$--$\beta$ split (50\% identical) is older than the
$\beta$--$\gamma$ split (80\%): $\alpha$ branches first, then $\beta$
and $\gamma$ separate from each other.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:8}
The limb \omterm{def:g10:universal-dna:gene}{genes} are present but the signal that switches them on in
the embryo's flank is not given, or is given and then stopped; the
difference is in the regulation of expression, not in the \omterm{def:g10:universal-dna:gene}{genes}.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:9}
In the amount and timing of a growth signal in the embryonic beak: a
regulatory difference. A change in regulation needs only a small
\omterm{def:g11:mutations:mutation}{mutation} in a control sequence, whereas a new \omterm{def:g11:gene-expression:protein}{protein} would need many;
it can arise and be selected within thousands of generations.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:10}
Its form, its capacity to colonise bare rock, its resistance to
drought and its slow growth belong to neither partner alone; they
emerge from the association. The lichen is a compound organism with
properties of its own.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:11}
Transmitted behaviour: materials are available on both sides, the
populations are closely related, and the difference follows the
river --- a barrier to contact and hence to learning, not to \omterm{def:g10:universal-dna:gene}{genes} or
resources.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:12}
A virus inserted its \omterm{def:g10:universal-dna:gene}{gene} into the genome of a primate ancestor's
germ \omterm{def:g10:cells-common-unit:cell}{cell} some tens of millions of years ago; the \omterm{def:g10:universal-dna:gene}{gene} was inherited
and kept. Test: the same \omterm{def:g10:universal-dna:gene}{gene} should sit at the same chromosomal
position in all primates and be absent from that position in other
mammals; its sequence should be closest to the virus's.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:13}
Mitochondrial \omterm{def:g10:cells-common-unit:organelle}{ribosomes} are of the bacterial type and bind the same
\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotics}: at high doses the drug slows the \omterm{def:g10:cells-common-unit:organelle}{mitochondria}'s \omterm{def:g11:gene-expression:protein}{protein}
synthesis. The sensitivity is inherited from the \omterm{def:g10:cells-common-unit:organelle}{mitochondria}'s
bacterial ancestor.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:14}
The \omterm{prop:g12:diversification-of-life:polyploidy}{polyploid}'s \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} number no longer matches either parent's,
so crosses with them give sterile offspring: isolation is immediate. A
new \omterm{def:g10:universal-dna:gene}{allele} must spread through a population over many generations
before it distinguishes a group. A plant \omterm{def:g10:biodiversity-scales:species}{species} can therefore arise
in a single generation.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:15}
At bottom, every new sequence --- a duplicated copy that diverges, a
regulatory change, the \omterm{def:g10:universal-dna:gene}{genes} of a transferred plasmid --- began as a
\omterm{def:g11:mutations:mutation}{mutation} somewhere. But duplication, transfer and polyploidy move and
multiply whole \omterm{def:g10:universal-dna:gene}{genes} and genomes at once; regulatory change produces
new forms without new \omterm{def:g11:gene-expression:protein}{proteins}; \omterm{def:g12:diversification-of-life:symbiosis}{symbiosis} combines genomes of
different \omterm{def:g10:biodiversity-scales:species}{species}; and culture transmits variation with no \omterm{def:g10:universal-dna:gene}{genes} at
all. \omterm{def:g11:mutations:mutation}{Mutation} supplies the letters; these processes rearrange whole
pages and books.
\end{solution}

\begin{solution}{pb:g12:diversification-of-life:1}
\textbf{1.} Emmer 28, bread wheat 42.

\textbf{2.} None: the A and B \omterm{def:g10:universal-dna:information}{chromosomes} are not \omterm{def:g10:common-ancestry:homology}{homologous}. Gametes
receive random assortments of the 14, almost never a complete set.

\textbf{3.} With 28 \omterm{def:g10:universal-dna:information}{chromosomes}, each A has an identical A partner and
each B a B: 14 pairs, regular \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis}, balanced gametes of 14.

\textbf{4.} None can pair: 7 A, 7 B and 7 D \omterm{def:g10:universal-dna:information}{chromosomes}, each without
a homologue. Sterile.

\textbf{5.} 21 pairs; a pollen grain carries 21.

\textbf{6.} The other two copies still supply the \omterm{def:g11:gene-expression:protein}{protein}. Redundant
copies are free to accumulate \omterm{def:g11:mutations:mutation}{mutations} and diverge --- duplication on
the scale of a whole genome.

\textbf{7.} In fields of the region where emmer was grown, between
10\,000 and 8\,000 years ago, with wild goat grass 2 growing as a weed
among the crop.

\textbf{8.} That the two steps --- hybridisation and doubling --- are
sufficient to produce bread wheat from its parents: the history can
be re-run.

\textbf{9.} Yes: it cannot produce fertile offspring with einkorn, so
by the definition it is a distinct \omterm{def:g10:biodiversity-scales:species}{species}, although it descends from
it.

\textbf{10.} Goat grass 2's D \omterm{def:g10:universal-dna:information}{chromosomes} are \omterm{def:g10:common-ancestry:homology}{homologous} to the wheat's
D set and pair with them, so a cross followed by back-crosses can move
the \omterm{def:g10:universal-dna:gene}{gene} in; an unrelated plant's \omterm{def:g10:universal-dna:information}{chromosomes} would not pair at all.

\textbf{11.} The \omterm{def:g10:universal-dna:gene}{genes} themselves hardly differ; the difference in the
bodies must lie in how the \omterm{def:g10:universal-dna:gene}{genes} are used.

\textbf{12.} A change in the regulation of expression --- the extent
of the region along the axis where the \omterm{def:g10:universal-dna:gene}{gene} is active.

\textbf{13.} The limb \omterm{def:g10:universal-dna:gene}{genes} are present and begin to act; the signal
that sustains limb growth is missing, so the buds stop. Again
regulation, not the loss of the \omterm{def:g10:universal-dna:gene}{genes}.

\textbf{14.} It shows that changing where the \omterm{def:g10:universal-dna:gene}{gene} is expressed is
sufficient to change the body region: the correlation between
expression and form is causal.

\textbf{15.} Bread wheat's change added whole sets to the genome (its
size); the python's changed where shared \omterm{def:g10:universal-dna:gene}{genes} are used (their
regulation).

\textbf{16.} Own circular \omterm{def:g10:universal-dna:information}{DNA} (expected: none, all \omterm{def:g10:universal-dna:gene}{genes} nuclear);
bacterial \omterm{def:g10:cells-common-unit:organelle}{ribosomes} (expected: eukaryotic \omterm{def:g10:cells-common-unit:organelle}{ribosomes}); multiplication
by division (expected: assembly from parts); double membrane
(expected: a single membrane like other \omterm{def:g10:cells-common-unit:organelle}{organelles}). And their \omterm{def:g10:universal-dna:gene}{genes}
resemble bacterial \omterm{def:g10:universal-dna:gene}{genes} (expected: nuclear-type \omterm{def:g10:universal-dna:gene}{genes}).

\textbf{17.} In the \omterm{def:g10:cells-common-unit:organelle}{nucleus}: over time most of the bacterium's \omterm{def:g10:universal-dna:gene}{genes}
were transferred to the nuclear genome, and the \omterm{def:g10:cells-common-unit:organelle}{organelle}'s own genome
shrank to a remnant.

\textbf{18.} Mitochondrial \omterm{def:g10:universal-dna:gene}{genes} pass only from mother to child; a
person's mitochondrial \omterm{def:g10:universal-dna:information}{DNA} traces the unbroken maternal line.

\textbf{19.} The \omterm{def:g10:cells-common-unit:organelle}{mitochondrion} has given up most of its \omterm{def:g10:universal-dna:gene}{genes} to its
host and cannot exist outside the \omterm{def:g10:cells-common-unit:cell}{cell}; the lichen's partners keep
their genomes and can, with difficulty, be separated.

\textbf{20.} Einkorn (14, A), goat grass 1 (14, B), goat grass 2 (14,
D): 42 \omterm{def:g10:universal-dna:information}{chromosomes}; "regulation"; the \omterm{def:g10:cells-common-unit:organelle}{mitochondrion}.
\end{solution}
