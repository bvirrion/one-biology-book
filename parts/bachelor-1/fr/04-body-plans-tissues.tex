\chapter{Plans d'organisation et tissus animaux}\label{ch:b1:body-plans-tissues}

Quatre animaux sur une paillasse~: une hydre d'un centimètre, un ver de
terre, une écrevisse, une grenouille. Sous les différences de taille et
de mœurs, ils sont bâtis sur une poignée de décisions partagées ---
le corps a-t-il un avant et un arrière, de combien de feuillets de
cellules a-t-il grandi, y a-t-il une cavité autour du tube digestif, le
corps est-il fait de segments répétés, où est le squelette --- et
chacun d'eux est bâti des quatre mêmes sortes de \omterm{def:b1:body-plans-tissues:tissue}{tissus}. Ce chapitre
expose les \omterm{def:b1:mammal-organization:bodyplan}{plans d'organisation} des animaux et les \omterm{def:b1:body-plans-tissues:tissue}{tissus} qui les
construisent, et s'achève là où les chapitres précédents s'étaient
arrêtés~: à la paroi d'un tube digestif de mammifère, vue comme des
\omterm{def:b1:body-plans-tissues:tissue}{tissus} assemblés en un \omterm{def:b1:mammal-organization:organ}{organe}.

\section{Les plans d'organisation}

\begin{definition}[Symétrie et céphalisation]\label{def:b1:body-plans-tissues:symmetry}
Un animal a une \emph{symétrie radiaire}\index{symetrie radiaire@symétrie radiaire}
lorsque son corps est organisé autour d'un axe central, sans gauche ni
droite, comme une hydre ou une méduse~: il rencontre son environnement
également de tous côtés, et il est en général fixé ou dérivant. Il a une
\emph{symétrie bilatérale}\index{symetrie bilaterale@symétrie bilatérale} lorsqu'un plan le
divise en deux moitiés en miroir, définissant un avant (antérieur) et un
arrière (postérieur), un dos (dorsal) et un ventre (ventral)~: le plan
d'un animal qui se déplace la tête la première, \omterm{def:b1:mammal-organization:organ}{organes} des sens et
centres nerveux concentrés à l'avant (\omterm{def:b1:mammal-organization:bodyplan}{céphalisation}).
\end{definition}

\begin{definition}[Feuillets embryonnaires, cavités du corps]\label{def:b1:body-plans-tissues:layers}
L'embryon d'un animal forme deux ou trois feuillets de cellules, les
\emph{feuillets embryonnaires}\index{feuillet embryonnaire}, dont
dérivent tous les \omterm{def:b1:body-plans-tissues:tissue}{tissus}~: l'\emph{ectoderme}\index{ectoderme} externe
(peau, système nerveux), l'\emph{endoderme}\index{endoderme} interne
(revêtement du tube digestif et de ses annexes) et, sauf chez les
spongiaires et les cnidaires, un \emph{mésoderme}\index{mesoderme@mésoderme}
médian (muscles, squelette, \omterm{def:b1:body-plans-tissues:connective}{tissu conjonctif}, sang, reins, gonades). Les
cnidaires sont \emph{diploblastiques}, les autres
\emph{triploblastiques}. Chez les triploblastiques, le mésoderme peut
combler l'espace entre le tube digestif et la paroi du corps
(\emph{acœlomates}~: plathelminthes), ou laisser une cavité entre eux~:
un \emph{cœlome}\index{coelome@cœlome} lorsque la cavité est entièrement bordée
de mésoderme (annélides, mollusques, arthropodes, échinodermes,
vertébrés). Le cœlome suspend le tube digestif dans un compartiment
propre, lui permet de bouger indépendamment de la paroi du corps, et
donne aux animaux à corps mou un \omterm{def:b1:body-plans-tissues:segments}{squelette hydrostatique}.
\end{definition}

\begin{definition}[Protostomiens et deutérostomiens]\label{def:b1:body-plans-tissues:proto}
Chez les animaux cœlomates, le destin de la première ouverture du tube
digestif embryonnaire sépare deux grandes lignées~: chez les
\emph{protostomiens}\index{protostomien} (annélides, mollusques,
arthropodes) elle devient la bouche, le \omterm{def:b1:body-plans-tissues:layers}{cœlome} se forme par clivage du
\omterm{def:b1:body-plans-tissues:layers}{mésoderme}, et les premières divisions de l'œuf sont spiralées et leur
destin fixé tôt~; chez les
\emph{deutérostomiens}\index{deuterostomien@deutérostomien} (échinodermes, vertébrés)
elle devient l'anus, le \omterm{def:b1:body-plans-tissues:layers}{cœlome} se forme par évagination du tube
digestif, et le destin des premières cellules reste souple.
\end{definition}

\begin{definition}[Métamérie, squelettes]\label{def:b1:body-plans-tissues:segments}
Un corps est \emph{métamérisé}\index{metamerie@métamérie} lorsqu'il est bâti
d'unités répétées le long de son axe (les anneaux d'un ver de terre, les
segments d'une écrevisse, les vertèbres et les blocs musculaires d'un
poisson), chacune ayant son jeu de muscles, de nerfs et souvent
d'appendices. Les segments peuvent être semblables ou spécialisés en
régions (tête, thorax, abdomen). Le \emph{squelette}\index{squelette},
contre lequel les muscles tirent, est
\emph{hydrostatique}\index{squelette hydrostatique} (un liquide sous
pression dans un \omterm{def:b1:body-plans-tissues:layers}{cœlome}, comme chez le ver de terre), un
\emph{exosquelette}\index{exosquelette} (une \omterm{def:b1:flowering-plant-organization:tissues}{cuticule} rigide à
l'extérieur du corps, articulée, muée pour croître, comme chez les
arthropodes), ou un \emph{endosquelette}\index{endosquelette} (cartilage
ou os à l'intérieur du corps, croissant avec lui, comme chez les
vertébrés).
\end{definition}

\begin{proposition}[Quatre plans comparés]\label{prop:b1:body-plans-tissues:four}
\begin{center}
\small
\begin{tabular}{lllll}
\hline
 & hydre & ver de terre & écrevisse & grenouille \\
\hline
symétrie & radiaire & bilatérale & bilatérale & bilatérale \\
feuillets & 2 & 3 & 3 & 3 \\
\omterm{def:b1:body-plans-tissues:layers}{cœlome} & aucun & oui & réduit & oui \\
lignée & --- & \omterm{def:b1:body-plans-tissues:proto}{protostomien} & \omterm{def:b1:body-plans-tissues:proto}{protostomien} & \omterm{def:b1:body-plans-tissues:proto}{deutérostomien} \\
segments & non & semblables & spécialisés & vertèbres, blocs musculaires \\
squelette & hydrostatique & hydrostatique & \omterm{def:b1:body-plans-tissues:segments}{exosquelette} & \omterm{def:b1:body-plans-tissues:segments}{endosquelette} \\
tube digestif & une ouverture & bouche à anus & bouche à anus & bouche à anus \\
circulation & aucune & close & ouverte & close, cœur \\
système nerveux & réseau & chaîne ventrale & chaîne ventrale & tube dorsal \\
\hline
\end{tabular}
\end{center}
Ces plans ne sont pas les degrés d'une échelle~: chacun est une manière
complète et réussie d'être un animal, et l'écrevisse et la grenouille
sont contemporaines l'une de l'autre, non ancêtres.
\end{proposition}

\begin{example}[Lire un animal à partir de son plan]\label{ex:b1:body-plans-tissues:reading}
Un animal à extrémité céphalique, à tube digestif traversant, au corps
fait d'anneaux semblables et sans parties dures~: un \omterm{def:b1:body-plans-tissues:proto}{protostomien}
cœlomate se déplaçant par un \omterm{def:b1:body-plans-tissues:segments}{squelette hydrostatique}, le liquide
cœlomique de chaque segment étant mis sous pression tour à tour par des
muscles circulaires et longitudinaux --- un ver de terre qui creuse. Un
animal à \omterm{def:b1:flowering-plant-organization:tissues}{cuticule} articulée et à segments spécialisés~: un arthropode,
dont la croissance doit passer par des mues. Des blocs musculaires
bilatéraux et segmentés autour d'un axe dorsal de vertèbres~: un
vertébré, dont l'\omterm{def:b1:body-plans-tissues:segments}{endosquelette} croît continûment.
\end{example}

\section{Les quatre types de tissus}

\begin{definition}[Tissu]\label{def:b1:body-plans-tissues:tissue}
Un \emph{tissu}\index{tissu} est un ensemble de cellules de structure et
d'origine semblables, avec le matériel extracellulaire qu'elles
produisent, organisé pour assurer une fonction commune. Les tissus
animaux se rangent en quatre types~: épithélial, conjonctif, musculaire,
nerveux. Tout \omterm{def:b1:mammal-organization:organ}{organe} est une combinaison d'au moins deux d'entre eux.
\end{definition}

\begin{definition}[Tissu épithélial]\label{def:b1:body-plans-tissues:epithelium}
Un \emph{épithélium}\index{epithelium@épithélium} est une nappe de cellules
jointives couvrant une surface ou tapissant une cavité, reposant sur une
mince couche de \omterm{def:b1:body-plans-tissues:connective}{matrice extracellulaire}, la \emph{lame
basale}\index{lame basale}. Ses cellules sont
\emph{polarisées}\index{polarite cellulaire@polarité cellulaire}~: la face apicale regarde
l'extérieur ou la lumière, la face basale la lame, et les deux portent
des protéines différentes. Elles sont unies par des \emph{jonctions
cellulaires}\index{jonction cellulaire}~: les \emph{jonctions serrées}
près de l'apex, qui scellent l'espace entre les cellules de sorte que
rien ne traverse la nappe autrement qu'à travers les cellules~; les
\emph{jonctions adhérentes} et les \emph{desmosomes}, qui fixent les
cellules mécaniquement~; les \emph{jonctions communicantes}, canaux qui
laissent passer ions et petites molécules d'une cellule à l'autre. Les
épithéliums se classent par le nombre d'assises (simple, stratifié) et
par la forme des cellules (pavimenteuse, cubique, prismatique), et
comprennent les glandes, cellules épithéliales spécialisées dans la
sécrétion. Ils couvrent, protègent, absorbent et sécrètent.
\end{definition}

\begin{definition}[Tissu conjonctif]\label{def:b1:body-plans-tissues:connective}
Le \emph{tissu conjonctif}\index{tissu conjonctif} est fait de cellules
dispersées dans une abondante \emph{matrice
extracellulaire}\index{matrice extracellulaire} qu'elles sécrètent~: des
fibres protéiques --- le \emph{collagène}\index{collagene@collagène} (résistance à
la traction~; le quart de toutes les protéines d'un mammifère),
l'\emph{élastine}\index{elastine@élastine} (élasticité) --- enchâssées dans une
\emph{substance fondamentale} hydratée de polysaccharides. Les
proportions en définissent les variétés~: \omterm{def:b1:body-plans-tissues:tissue}{tissu} conjonctif lâche sous
les \omterm{def:b1:body-plans-tissues:epithelium}{épithéliums} et autour des \omterm{def:b1:mammal-organization:organ}{organes} (fibroblastes, peu de fibres,
beaucoup de substance fondamentale)~; \omterm{def:b1:body-plans-tissues:tissue}{tissu} conjonctif dense des tendons
et des ligaments (collagène parallèle)~; \omterm{def:b1:body-plans-tissues:tissue}{tissu} adipeux (cellules à
réserves de graisse)~; cartilage (gel ferme de collagène et de
polysaccharides, sans vaisseaux)~; os (collagène minéralisé par du
phosphate de calcium, cellules vivantes logées dans des lacunes,
vaisseaux)~; et le sang, dont la matrice est le \omterm{def:b1:mammal-organization:compartments}{plasma}. Il unit,
soutient, stocke, transporte et défend.
\end{definition}

\begin{definition}[Tissu musculaire]\label{def:b1:body-plans-tissues:muscle}
Le \emph{tissu musculaire}\index{tissu musculaire} est fait de cellules
allongées bourrées de filaments protéiques contractiles (actine et
myosine). Trois variétés~: le \emph{muscle squelettique}, fait de
longues fibres plurinucléées à striation transversale, attachées au
squelette et sous contrôle volontaire~; le \emph{muscle cardiaque}, fait
de cellules striées et ramifiées, unies bout à bout par des jonctions
qui transmettent le battement de cellule en cellule, et se contractant
selon son rythme propre~; le \emph{muscle lisse}, fait de cellules
fusiformes non striées, dans la paroi des \omterm{def:b1:mammal-organization:organ}{organes} creux (tube digestif,
vaisseaux, utérus), à contraction lente et involontaire. Le mécanisme de
la contraction est traité dans le volume de deuxième année.
\end{definition}

\begin{definition}[Tissu nerveux]\label{def:b1:body-plans-tissues:nervous}
Le \emph{tissu nerveux}\index{tissu nerveux} est fait de
\emph{neurones}\index{neurone}, cellules munies d'un corps et de longs
prolongements (des dendrites qui reçoivent les signaux, un axone qui les
emporte) qui conduisent des signaux électriques et les transmettent à
d'autres cellules au niveau des synapses, et de \emph{cellules
gliales}\index{cellule gliale}, plusieurs fois plus nombreuses, qui
soutiennent, isolent et nourrissent les neurones. Il perçoit, intègre et
commande.
\end{definition}

\begin{example}[D'où viennent les tissus]\label{ex:b1:body-plans-tissues:origin}
Les \omterm{def:b1:body-plans-tissues:epithelium}{épithéliums} dérivent des trois feuillets (la peau de l'\omterm{def:b1:body-plans-tissues:layers}{ectoderme}, le
revêtement du tube digestif de l'endoderme, celui des vaisseaux et du
\omterm{def:b1:body-plans-tissues:layers}{cœlome} du \omterm{def:b1:body-plans-tissues:layers}{mésoderme})~; le \omterm{def:b1:body-plans-tissues:connective}{tissu conjonctif}, les muscles et le sang du
\omterm{def:b1:body-plans-tissues:layers}{mésoderme}~; le \omterm{def:b1:body-plans-tissues:nervous}{tissu nerveux} de l'\omterm{def:b1:body-plans-tissues:layers}{ectoderme}. La peau de la grenouille
est un \omterm{def:b1:body-plans-tissues:epithelium}{épithélium} ectodermique sur un \omterm{def:b1:body-plans-tissues:connective}{tissu conjonctif} mésodermique~;
son tube digestif, un \omterm{def:b1:body-plans-tissues:epithelium}{épithélium} endodermique enveloppé de muscle
mésodermique~; son encéphale, de l'\omterm{def:b1:body-plans-tissues:layers}{ectoderme} replié vers l'intérieur.
\end{example}

\section{Des tissus à un organe~: la paroi du tube digestif}

\begin{proposition}[Les quatre tuniques de la paroi digestive]\label{prop:b1:body-plans-tissues:gutwall}
\index{paroi digestive}
De la lumière vers l'extérieur, la paroi du tube digestif des vertébrés
est bâtie de quatre tuniques concentriques, chacune combinaison de
\omterm{def:b1:body-plans-tissues:tissue}{tissus}~: la \emph{muqueuse} (un \omterm{def:b1:body-plans-tissues:epithelium}{épithélium} --- absorbant et sécréteur
--- sur un \omterm{def:b1:body-plans-tissues:connective}{tissu conjonctif} lâche portant vaisseaux sanguins et
lymphatiques, avec une mince nappe de muscle lisse au-dessous)~; la
\emph{sous-muqueuse} (\omterm{def:b1:body-plans-tissues:connective}{tissu conjonctif} dense avec de plus gros
vaisseaux et un réseau nerveux)~; la \emph{musculeuse} (deux couches de
muscle lisse, circulaire interne et longitudinale externe, avec un
réseau nerveux entre elles, produisant les mouvements de brassage et de
propulsion)~; et la \emph{séreuse} (une mince couche conjonctive
couverte par l'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium} du \omterm{def:b1:body-plans-tissues:layers}{cœlome}). Les mêmes quatre tuniques vont de
l'œsophage au rectum~; ce qui change, c'est la muqueuse --- plissée en
villosités dans l'intestin grêle, glandulaire dans l'estomac.
\end{proposition}

\begin{method}[Lire une coupe histologique]\label{met:b1:body-plans-tissues:histology}
\begin{enumerate}
  \item Repérer la surface libre ou la lumière~: l'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium} qui la
        borde est la première couche. Compter ses rangées de noyaux
        (simple ou stratifié) et noter la forme des cellules.
  \item Sous l'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium}, chercher la zone pâle et fibreuse à noyaux
        dispersés et à vaisseaux~: le \omterm{def:b1:body-plans-tissues:connective}{tissu conjonctif}.
  \item Le muscle apparaît en faisceaux denses de cellules allongées~:
        strié à noyaux périphériques (squelettique), strié et ramifié
        (cardiaque), non strié à noyau central (lisse). Noter la
        direction des fibres par rapport à la coupe.
  \item Le \omterm{def:b1:body-plans-tissues:nervous}{tissu nerveux}, dans la paroi d'un \omterm{def:b1:mammal-organization:organ}{organe}, apparaît en amas de
        gros corps cellulaires pâles entre les couches musculaires.
  \item Nommer l'\omterm{def:b1:mammal-organization:organ}{organe} d'après la succession des tuniques et les
        spécialisations de la muqueuse (villosités, glandes, kératine).
\end{enumerate}
\end{method}

\begin{example}[La villosité comme surface d'échange]\label{ex:b1:body-plans-tissues:villus}
Une villosité de l'intestin grêle humain est un doigt de \qty{0.6}{mm}
de haut et de \qty{0.15}{mm} de large, couvert d'une assise unique de
cellules prismatiques dont chaque face apicale porte quelque
\qty{1500}{} microvillosités. Les villosités multiplient la surface du
tube par environ sept et les microvillosités par vingt encore~: les
\qty{0.6}{m^2} d'un tube lisse deviennent des dizaines de mètres carrés
d'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium} absorbant, épais d'une cellule, chaque cellule se trouvant
à quelques micromètres d'un capillaire et d'un vaisseau lymphatique dans
l'axe conjonctif de sa villosité
(\cref{ch:b1:digestion-absorption}). L'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium} est renouvelé à partir
des glandes de sa base tous les quatre jours~: la nappe, ses jonctions
et sa polarité sont reconstruites continûment.
\end{example}

\begin{remark}[Les tissus des deux organismes modèles]\label{rem:b1:body-plans-tissues:plants}
Les trois systèmes de \omterm{def:b1:body-plans-tissues:tissue}{tissus} de la plante du
\cref{ch:b1:flowering-plant-organization} n'ont pas ici d'équivalents
exacts. Son épiderme est un \omterm{def:b1:body-plans-tissues:epithelium}{épithélium} par la fonction, mais ses
cellules sont tenues par des parois et non par des jonctions~; son \omterm{def:b1:flowering-plant-organization:tissues}{tissu
fondamental} est à la fois de réserve, de soutien et photosynthétique~;
elle n'a ni muscle ni nerf. Les quatre \omterm{def:b1:body-plans-tissues:tissue}{tissus} de l'animal sont la
trousse à outils d'un corps qui se déplace et qui perçoit~; les trois de
la plante, celle d'un corps qui croît.
\end{remark}

\section{Problème~: la paroi de l'intestin grêle}

\begin{problem}[{Problème du week-end --- une série histologique de
l'intestin grêle humain lue du tube jusqu'à la microvillosité~:
tuniques, villosités, cellules, jonctions et renouvellement, jusqu'à la
surface absorbante d'une seule villosité}]
\label{pb:b1:body-plans-tissues:1}
L'intestin grêle humain est un tube long de \qty{6}{m} et de \qty{3}{cm}
de diamètre intérieur. Les plis circulaires de sa paroi triplent la
surface du tube lisse. Sa muqueuse porte \qty{25}{} villosités par
millimètre carré de surface plissée, chacune étant un cylindre de
\qty{0.6}{mm} de haut et de \qty{0.15}{mm} de diamètre. Les villosités
sont couvertes de cellules prismatiques dont la face apicale est un
carré de \qty{5}{\micro m} de côté~; chaque face apicale porte
\qty{1500}{} microvillosités, cylindres de \qty{1}{\micro m} de long et
de \qty{0.1}{\micro m} de diamètre. L'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium} est renouvelé tous les
\qty{4}{} jours.

\medskip
\noindent\textbf{Partie I --- Les tuniques.}
Une coupe transversale est examinée au microscope.
\begin{enumerate}
  \item Énumérer les quatre tuniques de la paroi, de la lumière vers
        l'extérieur, et les types de \omterm{def:b1:body-plans-tissues:tissue}{tissus} de chacune.
  \item La coupe montre des villosités. De quelle région du tube
        digestif s'agit-il~? Que montrerait à la place une coupe
        d'estomac~?
  \item Entre les deux couches musculaires de la musculeuse se trouvent
        des amas de grosses cellules pâles. Que sont-elles, et que
        commandent-elles~?
  \item Expliquer pourquoi les deux couches musculaires courent dans des
        directions perpendiculaires.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Les surfaces.}
\begin{enumerate}[resume]
  \item Calculer la surface interne du tube lisse.
  \item Calculer la surface après les plis circulaires.
  \item Calculer la surface latérale d'une villosité (cylindre, sommet
        négligé).
  \item Calculer la surface de villosités par millimètre carré de
        muqueuse plissée et le facteur par lequel les villosités
        multiplient la surface.
  \item Calculer la surface après les villosités.
  \item Calculer la surface latérale d'une microvillosité et celle des
        \qty{1500}{} d'une cellule.
  \item Calculer le facteur par lequel les microvillosités multiplient
        la surface apicale d'une cellule.
  \item Calculer la surface absorbante totale de l'intestin grêle.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Les cellules.}
\begin{enumerate}[resume]
  \item Calculer le nombre de cellules épithéliales couvrant une
        villosité.
  \item Calculer le nombre de villosités de l'intestin grêle et le
        nombre de cellules épithéliales qui les couvrent.
  \item L'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium} est renouvelé tous les \qty{4}{} jours. Combien de
        cellules sont desquamées dans la lumière par jour, et par
        seconde~?
  \item Où les cellules de remplacement sont-elles produites, et
        qu'est-ce que cela dit de la position des cellules en division
        par rapport aux cellules absorbantes~?
  \item Une cellule épithéliale est haute de \qty{25}{\micro m}.
        Calculer le volume d'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium} de l'intestin grêle et sa masse
        (masse volumique \qty{1.05}{g/cm^3}).
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Jonctions, matrice et villosité.}
\begin{enumerate}[resume]
  \item Chaque cellule est scellée à ses voisines par une jonction
        serrée qui fait le tour de son apex. Calculer la longueur de
        jonction serrée par cellule et la longueur totale dans
        l'intestin grêle.
  \item Expliquer pourquoi une substance présente dans la lumière ne
        peut gagner le sang qu'en traversant deux membranes cellulaires.
  \item L'axe de la villosité est un \omterm{def:b1:body-plans-tissues:connective}{tissu conjonctif} lâche avec un
        réseau capillaire et un vaisseau lymphatique central. Nommer les
        deux sortes de fibres extracellulaires qu'il contient et la
        cellule qui les sécrète.
  \item Nommer le \omterm{def:b1:body-plans-tissues:layers}{feuillet embryonnaire} d'origine de l'\omterm{def:b1:body-plans-tissues:epithelium}{épithélium} et
        celui de l'axe.
  \item La villosité contient aussi un cordon de muscle lisse qui la
        fait se raccourcir rythmiquement. Proposer sa fonction.
  \item Une maladie aplatit les villosités (la surface revient à celle
        du tube plissé). De quel facteur la surface absorbante
        diminue-t-elle, et quelles en sont les conséquences~?
  \item Calculer la surface absorbante d'une villosité avec ses
        microvillosités, en millimètres carrés.
  \item Énoncer le résultat~: la surface absorbante d'une seule
        villosité, et le nombre de villosités de l'intestin grêle.
\end{enumerate}
\end{problem}
