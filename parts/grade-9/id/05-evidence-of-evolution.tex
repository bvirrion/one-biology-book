\chapter{Bukti Evolusi}\label{ch:g9:evidence-of-evolution}

Dua kali buku ini berhenti di tepi sebuah klaim yang besar:
cabang \omterm{def:g6:classification-kinship:tree}{pohon kekerabatannya} semuanya bertemu
(\cref{rem:g6:classification-kinship:implies}), dan setiap
\omterm{def:g6:exploring-our-environment:components}{makhluk hidup} membaca satu sandi
(\cref{rem:g9:genes-and-dna:universal}). Klaim yang ditunjuk
keduanya punya nama --- \emph{evolusi}: \omterm{def:g5:biodiversity:species}{spesies} berubah
menembus angkatan, dan semuanya berasal dari nenek moyang
yang sama. Sebelum bab berikutnya bertanya
\emph{bagaimana}, bab ini mengerjakan apa yang dituntut
\cref{exo:g6:classification-kinship:12}: ia menggelar buktinya.

\section{Kesaksian bebatuannya}

\begin{definition}[Fosil]\label{def:g9:evidence-of-evolution:fossils}
Sebuah \emph{fosil}\index{fosil} adalah jejak kehidupan masa
lalu yang terawetkan --- \omterm{def:g1:animals-around-us:covering}{cangkang}, \omterm{def:g3:bones-and-muscles:skeleton}{tulang}, cetakan \omterm{def:g1:plants-around-us:parts}{daun},
tapak kaki --- tersegel di dalam batuan endapan. Lapisan
batuannya bertumpuk menurut urutan waktunya, yang tertua
terbawah; membaca ke atas berarti membaca sejarahnya. Fosil
adalah paruh sisa-jasad pada senarai
\cref{def:g6:exploring-our-environment:components}, yang
tersimpan berjuta-juta tahun.
\end{definition}

\begin{proposition}[Apa yang diperlihatkan rekaman berlapisnya]\label{prop:g9:evidence-of-evolution:record}
Dibaca berurutan, rekamannya bersaksi tiga kali:
\begin{enumerate}
  \item \emph{faunanya berubah}: tiap usia lapisan memuat
        pemainnya sendiri --- tidak ada kelinci di antara
        trilobitnya, tidak ada trilobit di lapisan
        kelincinya; sebagian besar \omterm{def:g5:biodiversity:species}{spesies} tiap lapisan yang
        dalam sudah \emph{\omterm{prop:g5:biodiversity:loss}{punah}} (kata pada
        \cref{prop:g5:biodiversity:loss}, kini dengan waktu
        yang dalam di belakangnya);
  \item \emph{urutannya bersarang seperti pohonnya}: \omterm{prop:g3:sorting-living-things:groups}{ikan}
        muncul di lapisan yang dalam, kerabat \omterm{prop:g3:sorting-living-things:groups}{amfibi} berkaki
        empat di atasnya, kerabat \omterm{prop:g4:classifying-animals:classes}{reptil} di atasnya lagi,
        \omterm{prop:g3:sorting-living-things:groups}{mamalia} dan \omterm{prop:g3:sorting-living-things:groups}{burung} lebih tinggi lagi --- kotak
        \omterm{def:g4:classifying-animals:classification}{klasifikasinya} (\cref{pb:g6:classification-kinship:1})
        bertumpuk dalam waktu persis seperti ramalan
        \omterm{prop:g6:classification-kinship:kinship}{kekerabatannya};
  \item \emph{peralihannya ada}: lempeng
        \emph{Archaeopteryx} yang termasyhur --- \omterm{def:g1:animals-around-us:covering}{bulu burung}
        dan sayap pada kerangka \omterm{prop:g4:classifying-animals:classes}{reptil} bergigi dan berekor
        panjang bertulang --- berdiri di tempat cabang \omterm{prop:g3:sorting-living-things:groups}{burung}
        dan \omterm{prop:g4:classifying-animals:classes}{reptilnya} berpisah; \omterm{def:g9:evidence-of-evolution:fossils}{fosil} paus dengan \omterm{def:g1:my-body:parts}{tungkai}
        belakang yang bekerja berdiri di tempat keturunan
        \omterm{prop:g3:sorting-living-things:groups}{mamalia} daratnya pada
        \cref{ex:g6:classification-kinship:whale} menuntutnya.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Deret pausnya]\label{ex:g9:evidence-of-evolution:whales}
Berkas pausnya, diisi oleh bebatuannya: \omterm{def:g9:evidence-of-evolution:fossils}{fosil} yang
berturut-turut, menurut urutan waktunya, berjalan dari
seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia} pantai berkaki empat seukuran serigala,
menembus bentuk dengan \omterm{def:g1:my-body:parts}{tungkai} yang menyusut dan ekor yang
menguat, sampai perenang penuh tanpa \omterm{def:g1:my-body:parts}{tungkai} --- sementara
pausnya hari ini tetap membangun \omterm{def:g3:bones-and-muscles:skeleton}{tulang} pinggul-dan-tungkai
yang mungil dan tersembunyi
(\cref{ex:g6:classification-kinship:placing} mencatat sisa
rudimenternya) dan tetap menyusui, bernapas dengan udara,
serta menumbuhkan \omterm{def:g1:animals-around-us:covering}{bulu} \omterm{def:g5:human-reproduction:pregnancy}{janin}. Ramalan, \omterm{def:g9:evidence-of-evolution:fossils}{fosil}, sisa
rudimenter: tiga kunci, satu gembok.
\end{example}

\section{Kesaksian tubuh yang hidup}

\begin{proposition}[Anatominya mengingat]\label{prop:g9:evidence-of-evolution:anatomy}
Tubuh yang hidup mengangkut riwayatnya:
\begin{itemize}
  \item \emph{rancangan bersamanya}: sayap kelelawar, sirip
        dayung paus, \omterm{def:g1:my-body:parts}{tungkai} kuda dan lenganmu adalah satu
        susunan \omterm{def:g3:bones-and-muscles:skeleton}{tulang} (percabangan berkaki empat pada
        \cref{pb:g6:classification-kinship:1}) yang
        direntangkan, dipendekkan dan dialihtugaskan ---
        pewarisannya menjelaskan apa yang tak akan pernah
        diulang oleh rekayasa yang segar;
  \item \emph{sisa rudimenternya}: pinggul pausnya, sayap
        tersegel kumbang yang tak dapat terbang, \omterm{def:g3:bones-and-muscles:skeleton}{tulang}
        ekormu sendiri dan keadaan umbai cacing yang
        menyusut --- bagian yang dipertahankan karena
        keturunan membangun di atasnya, bukan dari nol;
  \item \emph{\omterm{def:g5:human-reproduction:pregnancy}{embrionya} sepakat}: \omterm{def:g5:human-reproduction:pregnancy}{embrio} \omterm{def:g4:classifying-animals:vertebrate}{vertebrata} \omterm{prop:g3:sorting-living-things:groups}{ikan},
        anak ayam dan manusia berjalan sangat serupa
        sepanjang pekan awalnya (penggelaran rancangan pada
        \cref{prop:g8:from-egg-to-birth:timetable}) ---
        petunjuk pembangunan yang sama, memencar belakangan.
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[Molekulnya memberikan suara]\label{prop:g9:evidence-of-evolution:molecules}
Saksi yang terbaru adalah sandi pada
\cref{rem:g9:genes-and-dna:universal}. Bandingkan ejaan satu
\omterm{def:g9:genes-and-dna:gene}{gen} yang terpelihara baik menembus berbagai \omterm{def:g5:biodiversity:species}{spesies}: manusia
dan simpanse berbeda pada sedikit huruf, manusia dan tikus
pada lebih banyak, manusia dan \omterm{prop:g3:sorting-living-things:groups}{ikan} pada lebih banyak lagi,
manusia dan \omterm{ex:g6:producing-our-food:bread}{ragi} paling banyak --- dan jaraknya, dijumlahkan
\omterm{def:g9:genes-and-dna:gene}{gen} demi \omterm{def:g9:genes-and-dna:gene}{gen}, menggambar ulang \emph{pohon yang sama} yang
digambar para ahli anatomi dari \omterm{def:g3:bones-and-muscles:skeleton}{tulangnya} dan para ahli
paleontologi dari lapisannya. Tiga saksi yang saling bebas,
satu pohon keluarga: kesepakatan itulah sebabnya para ahli
\omterm{def:g1:living-or-not:living}{biologi} berhenti bertengkar tentang \emph{apakah} lalu
berpaling kepada \emph{bagaimana}.
\end{proposition}
\admitted

\begin{example}[Menyaksikannya bergerak]\label{ex:g9:evidence-of-evolution:live}
Perubahannya bahkan dapat disaksikan di tempat angkatannya
berputar cepat. Bentuk gelap ngengat berbintiknya menyebar
menembus hutan industri yang bernoda jelaga lalu surut ketika
udaranya bersih; \omterm{prop:g3:sorting-living-things:groups}{serangga} yang kebal insektisida dan \omterm{def:g6:producing-our-food:fermentation}{mikroba}
yang menentang antibiotik
(\cref{ch:g9:microbes-and-infection} akan menjumpai kasus
kedokterannya) muncul dalam hitungan tahun; sebaran ukuran
paruh \omterm{prop:g3:sorting-living-things:groups}{burung} finch pulau bergeser mengikuti kemarau dan
kelimpahannya, diukur langsung oleh kajian
cincin-dan-jangka-sorong. Gerak yang kecil, waktu yang
pendek --- mekanismenya tertangkap sedang berjalan, apa pun
jangkauan jangka panjangnya.
\end{example}

\begin{method}[Menimbang bukti sejarah]\label{met:g9:evidence-of-evolution:weigh}
Evolusi adalah sejarah, dan klaim sejarah diuji seperti
berkas dingin pada \cref{pb:g9:unique-individuals:1} ---
lewat pertemuan saksi yang saling bebas:
\begin{enumerate}
  \item tanyakan apa yang \emph{diramalkan} klaimnya:
        peralihan yang mana, sisa rudimenter yang mana, jarak
        molekul yang mana;
  \item periksa tiap saksinya secara terpisah: bebatuannya,
        tubuhnya, molekulnya, saksi langsungnya;
  \item timbang kebebasannya: saksi yang tak mungkin
        bersekongkol, bila sepakat, melipatgandakan kelayakan
        percayanya;
  \item biarkan vonisnya terbuka bagi temuan berikutnya ---
        dan catatlah bila temuannya terus mendarat di tempat
        yang disebutkan klaimnya untuk digali (\omterm{def:g1:my-body:parts}{tungkai}
        pausnya dicari, lalu ditemukan).
\end{enumerate}
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:1}
Apa itu \omterm{def:g9:evidence-of-evolution:fossils}{fosil}, dan mengapa lapisan batuannya terbaca sebagai
sebuah garis waktu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:2}
Nyatakan ketiga kesaksian rekaman berlapisnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:3}
Campuran ciri apa yang membuat \emph{Archaeopteryx} sebuah
peralihan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:4}
Berikan tiga sisa rudimenter dan kesaksian masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:5}
Bagaimana perbandingan molekulnya menggambar ulang pohonnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:6}
Sebutkan dua saksi langsung perubahannya dari
\cref{ex:g9:evidence-of-evolution:live}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:7}
Mengapa rancangan \omterm{def:g1:my-body:parts}{anggota gerak} bersamanya menghujahkan
pewarisan alih-alih rancangan bagus yang dipakai ulang? (Apa
yang bebas dilakukan rekayasa yang segar?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:8}
Rakitlah berkas lengkap pausnya: ciri kelasnya, sisa
rudimenternya, deret \omterm{def:g9:evidence-of-evolution:fossils}{fosilnya}, kerabat molekulnya --- satu
baris masing-masing.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:9}
Mengapa \emph{kesepakatan} bebatuan, tubuh dan molekulnya
lebih berbobot daripada saksi mana pun sendirian? Pakailah
langkah kebebasan pada
\cref{met:g9:evidence-of-evolution:weigh}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:10}
``Tidak ada kelinci di antara trilobitnya.'' Mengapa
ketiadaan itu --- yang bertahan menembus setiap penggalian
yang pernah dibuat --- justru merupakan bukti yang kuat?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:11}
Saksi \omterm{def:g5:human-reproduction:pregnancy}{embrionya}: apa yang sepakat, kapan ia memencar, dan
permesinan bab yang mana yang mengatakan mengapa petunjuk
awal bersamanya paling penting?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:evidence-of-evolution:12}
Tuliskan hujah babnya sebagai
\cref{met:g9:evidence-of-evolution:weigh} yang diterapkan
penuh: klaimnya, ramalannya, empat saksinya, vonisnya ---
dan pertanyaan yang diserahkan vonisnya kepada
\cref{ch:g9:how-species-change}.
\end{exercise}

\section{Soal: Sayap Baru Museumnya}

\begin{problem}[{Soal akhir pekan --- mengurasi ruang
buktinya}]\label{pb:g9:evidence-of-evolution:1}
Museum sejarah alamnya membangun sebuah sayap evolusi, dan
kelasnya mengurasi empat ruangan: Bebatuan, Tubuh, Molekul,
Saksi Langsung.

\medskip\noindent\textbf{Bagian I --- Ruangan demi ruangan.}
\begin{enumerate}
  \item Bebatuan: urutkan dindingnya --- lempeng trilobit,
        \omterm{prop:g3:sorting-living-things:groups}{ikan} pertama, tapak kerabat \omterm{prop:g3:sorting-living-things:groups}{amfibi}, kerangka \omterm{prop:g4:classifying-animals:classes}{reptil},
        cetakan \emph{Archaeopteryx}, rahang \omterm{prop:g3:sorting-living-things:groups}{mamalia} awal ---
        lalu tuliskan keterangan satu kalimat ruangannya dari
        \cref{prop:g9:evidence-of-evolution:record}.
  \item Tubuh: pusat perhatiannya menggelar sayap kelelawar,
        sirip dayung paus, \omterm{def:g1:my-body:parts}{tungkai} kuda dan \omterm{def:g1:my-body:parts}{lengan} manusia
        dari satu tiang kerangka. Berikan keterangan bagi
        rancangan bersamanya --- lalu tambahkan tiga pameran
        lemari sisa rudimenternya.
  \item Molekul: bagan dindingnya menjumlahkan perbedaan
        huruf satu \omterm{def:g9:genes-and-dna:gene}{gen}: simpanse 1, tikus 8, \omterm{prop:g3:sorting-living-things:groups}{ikan} 24, \omterm{ex:g6:producing-our-food:bread}{ragi} 41
        (bilangan sekadar gambaran). Pohonnya harus cocok
        dengan apa, supaya ruangannya bekerja?
  \item Saksi Langsung: pilih dua pameran dari
        \cref{ex:g9:evidence-of-evolution:live} lalu katakan
        apa yang diperlihatkan masing-masing sedang bergerak.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pertanyaan para
pengunjungnya.}
\begin{enumerate}[resume]
  \item ``Mengapa masih ada monyet, kalau kita berasal dari
        nenek moyang yang kita bagi bersama mereka?''
        Jawablah dengan percabangan pohonnya --- sepupu,
        bukan tangga (\cref{def:g6:classification-kinship:tree}).
  \item ``Di mana mata rantai yang hilangnya?'' Tunjuklah
        yang sudah ditemukan --- dua dari bab ini --- lalu
        nyatakan masing-masing berdiri di antara apa.
  \item ``Bukankah evolusi hanyalah sebuah teori?'' Jawablah
        dengan cara bukti sejarahnya: apa arti ``teori'' di
        sini, dan apa yang ditegakkan kesepakatan keempat
        ruangannya.
  \item ``Dapatkah penggalian besok menjungkirkannya?''
        Nyatakan temuan jenis apa yang akan menjungkirkannya
        --- kelinci di antara trilobitnya --- dan apa arti
        ketiadaannya, penggalian demi penggalian.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Panel penutup
sayapnya.}
\begin{enumerate}[resume]
  \item Susunlah panel yang menghubungkan sayapnya dengan
        aula \omterm{def:g4:classifying-animals:classification}{klasifikasi} di sebelahnya: mengapa kotak
        bersarang yang lama itu sejak awal memang pohonnya.
  \item Tambahkan alinea molekulnya: apa yang dikatakan sandi
        bersamanya tentang \omterm{def:g1:plants-around-us:parts}{batangnya}.
  \item Tambahkan alinea jujurnya: apa yang diperlihatkan
        sayapnya (perubahan, \omterm{prop:g6:classification-kinship:kinship}{kekerabatan}, peralihan) dan apa
        yang belum dijelaskannya --- sebutkan pertanyaan
        mekanisme yang akan diangkat aula berikutnya.
  \item Tutuplah dengan satu kalimat yang akan diingat
        seorang anak dua belas tahun, yang memuat
        \emph{keluarga} dan \emph{waktu}.
\end{enumerate}
\end{problem}
