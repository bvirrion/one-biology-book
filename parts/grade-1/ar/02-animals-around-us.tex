\chapter{الحيوانات من حولنا}\label{ch:g1:animals-around-us}

شحرور ينطّ على العشب، وقطة ترقبه من فوق الجدار، ونملة تحمل فتاتة
أكبر منها على عتبة الباب. \omterm{def:g1:animals-around-us:animal}{الحيوانات} في كل مكان حولنا --- في
الحديقة، وفي المتنزه، وفي البيت. فلنتعلم أن ننظر إليها كما ينظر
عالم الأحياء: إلى أجسامها، وإلى طريقة تنقلها.

\section{ما الحيوان؟}

\begin{definition}[الحيوان]\label{def:g1:animals-around-us:animal}
\emph{الحيوان}\index{حيوان} \omterm{def:g1:living-or-not:living}{كائن حي} يتغذى بأكل كائنات حية أخرى ---
نباتات أو حيوانات --- ويستطيع في العادة أن يتنقل بنفسه. فالقطة
والشحرور والسمكة والنملة كلها حيوانات. ونحن حيوانات أيضًا.
\end{definition}

\begin{example}[حيوانات حديقة واحدة]\label{ex:g1:animals-around-us:garden}
حديقة صغيرة واحدة قد تكون بيتًا \omterm{def:g1:animals-around-us:animal}{لحيوانات} كثيرة: شحرور يسحب دودة من
الأرض، والدودة نفسها، وحلزون على ورقة، وعنكبوت في شبكته، ونحلة في
زهرة، وقطة الجيران. صغيرة كانت أم كبيرة، فكلها تتغذى وكلها تتنقل.
\end{example}

\begin{remark}[الصغير ليس قليل الشأن]\label{rem:g1:animals-around-us:small}
النملة \omterm{def:g1:animals-around-us:animal}{حيوان} تمامًا كما أن الفيل \omterm{def:g1:animals-around-us:animal}{حيوان}. فحجم الجسم لا يقرر من هو
\omterm{def:g1:animals-around-us:animal}{الحيوان}: التغذي والتنقل هما اللذان يقرران. وبعض أعجب \omterm{def:g1:animals-around-us:animal}{الحيوانات} من
حولنا هي أصغرها.
\end{remark}

\section{أجزاء جسم الحيوان}

\begin{definition}[أجزاء الجسم]\label{def:g1:animals-around-us:parts}
لأكثر \omterm{def:g1:animals-around-us:animal}{الحيوانات} \emph{رأس}\index{رأس} (فيه الفم ومعظم أعضاء الحس)،
وجسم، و\emph{أطراف}\index{طرف} --- أرجل أو أجنحة أو زعانف ---
تستعملها في التنقل. ولكثير منها \emph{ذنب}\index{ذنب} أيضًا.
\end{definition}

\begin{example}[عُدّ الأرجل]\label{ex:g1:animals-around-us:legs}
الأرجل تأتي بأعداد تستحق العدّ. فالشحرور يمشي على $2$ من الأرجل،
والقطة على $4$، والدعسوقة على $6$، والعنكبوت على $8$ --- أما دودة
الأرض والحلزون فلا أرجل لهما البتة، ومع ذلك يبلغان حيث يريدان.
\end{example}

\begin{definition}[كساء الجسم]\label{def:g1:animals-around-us:covering}
يحمي جسمَ \omterm{def:g1:animals-around-us:animal}{الحيوان} كساءٌ يكسوه: \emph{الفرو}\index{فرو} (الشعر) على
القطة، و\emph{الريش}\index{ريش} على الشحرور،
و\emph{الحراشف}\index{حرشفة} على السمكة، و\emph{الصدفة}\index{صدفة}
على الحلزون، و\emph{الجلد}\index{جلد} الأملس الرطب على الضفدع.
\end{definition}

\begin{example}[من يلبس ماذا]\label{ex:g1:animals-around-us:covering}
امسح على قطة: \omterm{def:g1:animals-around-us:covering}{فرو}. والتقط ريشة شحرور ساقطة: خفيفة ناعمة. وانظر عن
قرب إلى سمكة في السوق: صفوف من الحراشف اللامعة، كقرميد صغير على
سطح. ثلاثة \omterm{def:g1:animals-around-us:animal}{حيوانات}، وثلاثة أكسية مختلفة.
\end{example}

\section{كيف تتنقل الحيوانات}

\begin{proposition}[طرق التنقل]\label{prop:g1:animals-around-us:moving}
تتنقل \omterm{def:g1:animals-around-us:animal}{الحيوانات} بطرق تناسب أطرافها:
\begin{itemize}
  \item ذوات الأرجل \emph{تمشي} أو تجري أو تنطّ أو تزحف --- القطة،
        والشحرور على العشب، والنملة؛
  \item ذوات الأجنحة \emph{تطير}\index{طيران} --- الشحرور، والنحلة،
        والفراشة؛
  \item ذوات الزعانف \emph{تسبح}\index{سباحة} --- السمكة؛
  \item وما لا \omterm{def:g1:animals-around-us:parts}{أطراف} له ينزلق أو يتلوى --- الحلزون على قدمه الطرية
        الواحدة، ودودة الأرض تمتدّ ثم تنكمش.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[حيوان واحد وطرق عدة]\label{rem:g1:animals-around-us:several}
لكثير من \omterm{def:g1:animals-around-us:animal}{الحيوانات} أكثر من طريقة للتنقل. فالشحرور يمشي
\emph{ويطير}؛ والضفدع يسبح \emph{وينطّ}؛ والبطة تمشي وتسبح
\emph{وتطير}. فراقب \omterm{def:g1:animals-around-us:animal}{الحيوان} بعض الوقت قبل أن تقرر كيف يتنقل.
\end{remark}

\begin{method}[ملاحظة حيوان]\label{met:g1:animals-around-us:observe}
لتصف حيوانًا كما يصفه عالم الأحياء:
\begin{enumerate}
  \item انظر إلى الأجزاء: الرأس، والجسم، \omterm{def:g1:animals-around-us:parts}{والأطراف}، والذنب؛
  \item عُدّ الأرجل من غير تخمين؛
  \item سمِّ الكساء: \omterm{def:g1:animals-around-us:covering}{فرو}، أو \omterm{def:g1:animals-around-us:covering}{ريش}، أو حراشف، أو \omterm{def:g1:animals-around-us:covering}{صدفة}، أو \omterm{def:g1:animals-around-us:covering}{جلد}؛
  \item راقب كيف يتنقل: مشيًا، أو نطًّا، أو طيرانًا، أو سباحة، أو زحفًا؛
  \item اذكر أين وجدته: على العشب، أو في شجرة، أو في بركة، أو تحت حجر.
\end{enumerate}
ولا تؤذِ \omterm{def:g1:animals-around-us:animal}{الحيوان} أبدًا: راقبه، ثم دعه يمضي في سبيله.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:1}
سمِّ أجزاء جسم القطة الأربعة الرئيسة في الشكل. وأي جزء يحمل
العينين والفم؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:2}
كم رجلًا: (أ) للدجاجة؟ (ب) للكلب؟ (ج) للدعسوقة؟ (د) للعنكبوت؟
(ه) لدودة الأرض؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:3}
اذكر كساء كل \omterm{def:g1:animals-around-us:animal}{حيوان}: خروف، وبطة، وسمكة سلمون مرقط، وحلزون.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:4}
أي \omterm{def:g1:animals-around-us:animal}{حيوانات} هذا الفصل تستطيع الطيران؟ وما الذي لها جميعًا مكان
الأرجل الأمامية أو الذراعين؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:5}
الحلزون ودودة الأرض لا أرجل لهما. فكيف يتنقل كل منهما؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:6}
البطة تتنقل بثلاث طرق. سمِّها، واذكر أي جزء من جسمها تستعمل في كل
طريقة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:7}
لماذا كانت النملة حيوانًا حسب \cref{def:g1:animals-around-us:animal}؟
أجب بفكرتي التعريف.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:8}
جرّب: راقب حيوانًا واحدًا في الخارج بعض الوقت --- طائرًا، أو حشرة،
أو حلزونًا. وصفه بخطوات \cref{met:g1:animals-around-us:observe}
الخمس.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:9}
يقول صديق: ``السمك لا يمكن أن يكون حيوانًا --- فهو لا يمشي!''
بماذا تجيبه؟ استعمل \cref{prop:g1:animals-around-us:moving}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:10}
\omterm{def:g1:animals-around-us:animal}{حيوان} غامض: له $2$ من الأرجل، و$2$ من الأجنحة، \omterm{def:g1:animals-around-us:covering}{وريش}، وهو لا يستطيع
الطيران، لكنه يجري بسرعة كبيرة. أهو طائر؟ أيمكن أن يكون أسرع عدّاء
في حيّك؟ (فكّر في النعامة.)
\end{exercise}
