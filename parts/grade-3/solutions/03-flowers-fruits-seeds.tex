\section*{Chapter \ref{ch:g3:flowers-fruits-seeds} --- Flowers, Fruits and Seeds}

\begin{solution}{exo:g3:flowers-fruits-seeds:1}
\omterm{def:g3:flowers-fruits-seeds:parts}{Petals} (colored and scented, they attract \omterm{prop:g3:sorting-living-things:groups}{insects}); \omterm{def:g3:flowers-fruits-seeds:parts}{stamens}
(thin stalks carrying \omterm{def:g3:flowers-fruits-seeds:parts}{pollen}); the \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} in the center (the
little flask holding the future \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}).
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:2}
\omterm{def:g3:flowers-fruits-seeds:parts}{Pollen} is the fine yellow powder carried by the \omterm{def:g3:flowers-fruits-seeds:parts}{stamens}.
Pollination is \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} reaching the \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} --- mostly delivered
by visiting \omterm{prop:g3:sorting-living-things:groups}{insects}, sometimes by the wind.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:3}
(a) They wither and fall. (b) It swells and becomes the \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}.
(c) They become the \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} inside the \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:4}
April: white blossom, bees visiting --- pollination. May: \omterm{def:g3:flowers-fruits-seeds:parts}{petals}
fall; the \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} of each pollinated \omterm{def:g3:flowers-fruits-seeds:parts}{flower} swells into a tiny
green ball. June: the ball has grown into a red, sweet cherry
with the stone --- the \omterm{def:g2:from-seed-to-plant:seed}{seed} --- inside.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:5}
Because without insect visits most blossoms are never
pollinated, and an unpollinated \omterm{def:g3:flowers-fruits-seeds:parts}{flower} falls without becoming
\omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}. The bees' \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} deliveries are what turn blossom into
cherries.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:6}
Cut it open and look for \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}: \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} inside means it grew from
a \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} --- a \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}; none means another \omterm{def:g1:plants-around-us:plant}{plant} part. The tomato
is full of \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}: a \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}. The carrot has none: not a \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit} but
a \omterm{def:g1:plants-around-us:parts}{root}.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:7}
Cucumber: \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} --- \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}. Lettuce: no \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} --- \omterm{def:g1:plants-around-us:parts}{leaves}, not a
\omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}. Green bean pod: a row of \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} --- \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}. Potato: no
\omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} --- a swollen underground \omterm{def:g1:plants-around-us:parts}{stem}, not a \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:8}
Open answer. The usual surprises: tomato, cucumber and bean pod
turning out to be \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits}; the strawberry carrying its tiny
\omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} on the outside.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:9}
Almost no plums. \omterm{prop:g3:flowers-fruits-seeds:transform}{Fruit} can only come from a pollinated \omterm{def:g3:flowers-fruits-seeds:parts}{flower}'s
\omterm{def:g3:flowers-fruits-seeds:parts}{pistil}; with the blossom dead in April, there is nothing left to
transform, however fine the summer.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:10}
For the tree, the apple is a way of making and protecting
\omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}. At its core lie the pips --- \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} --- and each pip,
planted, could \omterm{def:g2:from-seed-to-plant:germination}{germinate} into a new apple tree. That we find
the packaging delicious is our luck (and, for the tree, a way
of getting its \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} carried far from home).
\end{solution}
