\chapter{श्वासोच्छ्वास}\label{ch:g4:breathing}

साँस रोको और गिनना शुरू करो। एक मिनट पूरा होने से बहुत पहले तुम्हारी
छाती और-और ज़ोर से फिर हिलने की माँग करने लगती है। खाना छोड़ा जा सकता
है, \omterm{prop:g1:healthy-habits:needs}{नींद} रात तक रुक सकती है --- पर हवा \emph{अभी} चाहिए, चौबीसों घंटे,
मिनट में लगभग पंद्रह बार। हवा की एक साँस में इतना ज़रूरी क्या है?

\section{श्वास की गतियाँ}

\begin{definition}[साँस लेना और छोड़ना]\label{def:g4:breathing:movements}
\emph{श्वासोच्छ्वास}\index{श्वासोच्छ्वास} दो गतियों का कभी न रुकने
वाला बदलाव है:
\begin{itemize}
  \item \emph{साँस लेना} (अंतःश्वसन)\index{अंतःश्वसन}: छाती फैलती है
        और हवा नाक या मुँह से भीतर बहती है;
  \item \emph{साँस छोड़ना} (निःश्वसन)\index{निःश्वसन}: छाती ढीली पड़ती
        है और हवा वापस बाहर बहती है।
\end{itemize}
आराम में यह जोड़ी मिनट में लगभग $15$ बार दोहराई जाती है --- शिशुओं में
तेज़, मेहनत में और तेज़ --- और तुम्हें इसके बारे में कभी सोचना तक नहीं
पड़ता।
\end{definition}

\begin{example}[गतियों को देखना]\label{ex:g4:breathing:watch}
हथेलियाँ अपनी \omterm{def:g3:bones-and-muscles:skeleton}{पसलियों} पर चपटी रखो: साँस लेते समय \omterm{def:g3:bones-and-muscles:skeleton}{पसलियों} का पिंजरा
तुम्हारी हथेलियों के नीचे उठता और फैलता है; साँस छोड़ते समय वह वापस बैठ
जाता है। सोती हुई बिल्ली के पहलू में भी वही उठना-गिरना दिखता है। ये
गतियाँ कभी नहीं रुकतीं --- जागते, सोते, पहली किलकारी से आख़िरी दिन तक।
\end{example}

\section{हवा जाती कहाँ है}

\begin{definition}[श्वास के अंग]\label{def:g4:breathing:organs}
भीतर ली गई हवा नाक (या मुँह) से होकर \emph{श्वासनली}\index{श्वासनली}
में उतरती है --- वह मज़बूत, छल्लेदार नली जिसे तुम अपनी गरदन के सामने
महसूस कर सकते हो --- और वह दो शाखाओं में बँट जाती है, हर
\emph{फेफड़े}\index{फेफड़ा} के लिए एक। दोनों फेफड़े स्पंज जैसे अंग हैं
जो \omterm{def:g3:bones-and-muscles:skeleton}{पसलियों} के पिंजरे को भरते हैं (\cref{def:g3:bones-and-muscles:skeleton}
की \omterm{def:g3:bones-and-muscles:skeleton}{पसलियों} के पीछे), और लाखों नन्ही हवा-कोठरियों पर ख़त्म होते हैं,
जिनमें से हर एक महीन \omterm{prop:g4:journey-of-food:absorb}{रक्त} वाहिकाओं में लिपटी है।
\end{definition}

\begin{remark}[स्पंज जैसा, जान-बूझकर]\label{rem:g4:breathing:sponge}
दो सादे थैलों के बजाय लाखों कोठरियाँ: यह फिर वही \omterm{def:g4:journey-of-food:tract}{छोटी आँत} वाली करामात
है (\cref{rem:g4:journey-of-food:surface})। इन सारी कोठरियों की
दीवारें मिलकर हवा और \omterm{prop:g4:journey-of-food:absorb}{रक्त} के बीच एक विशाल मिलन-सतह बना देती हैं ---
आधे टेनिस मैदान जितनी, तुम्हारी छाती में तह करके भरी हुई।
\end{remark}

\section{साँस लेना किसलिए है}

\begin{proposition}[ऑक्सीजन लेना, कार्बन डाइऑक्साइड लौटाना]\label{prop:g4:breathing:exchange}
हवा गैसों का मेल है, और शरीर को उनमें से एक चाहिए:
\emph{ऑक्सीजन}\index{ऑक्सीजन}। फेफड़ों की कोठरियों में ताज़ी हवा की
ऑक्सीजन \omterm{prop:g4:journey-of-food:absorb}{रक्त} में उतर जाती है, और \omterm{prop:g4:journey-of-food:absorb}{रक्त} उसे शरीर के हर हिस्से तक पहुँचाता
है; वहाँ उसका इस्तेमाल पोषक तत्वों के साथ मिलकर
\cref{prop:g3:balanced-diet:jobs} वाली \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} छोड़ने में होता है। शरीर
की कचरा गैस, \emph{कार्बन डाइऑक्साइड}\index{कार्बन डाइऑक्साइड}, वापसी
की यात्रा करती है और साँस के साथ बाहर निकल जाती है। छोड़ी हुई हवा में
ली हुई हवा के मुक़ाबले ऑक्सीजन कम और कार्बन डाइऑक्साइड ज़्यादा होती है
--- यही फ़र्क़ वह है \emph{जिसके लिए} साँस ली जाती है।
\end{proposition}
\admitted

\begin{example}[दौड़ने से साँस क्यों फूलती है]\label{ex:g4:breathing:running}
दौड़ते समय तुम्हारी \omterm{def:g1:my-body:parts}{टाँगें} \omterm{prop:g3:balanced-diet:jobs}{ऊर्जा} तेज़ी से जलाती हैं --- इसलिए उन्हें
ऑक्सीजन भी तेज़ चाहिए। साँस मान जाती है: मिनट में लगभग $15$ शांत
साँसों से बढ़कर $40$ या उससे भी ज़्यादा, और गहरी, जबकि हृदय
(\cref{ch:g4:heart-and-blood} अपनी तरफ़ की कहानी कहता है) पहुँचाने की
रफ़्तार बढ़ाने के लिए दौड़ पड़ता है। दौड़ के बाद की हाँफ शरीर का अपना
ऑक्सीजन-हिसाब चुकाना है।
\end{example}

\begin{example}[धुँधलाया आईना]\label{ex:g4:breathing:mirror}
ठंडे आईने पर साँस छोड़ो: धुंध। छोड़ी हुई हवा बदली हुई लौटती है ---
गरम, नम, और बिना दिखे ऑक्सीजन में ग़रीब तथा कार्बन डाइऑक्साइड में
अमीर। आईना नमी दिखा देता है; गैसों की जो अदला-बदली वह नहीं दिखा सकता,
असली बात वही है।
\end{example}

\includegraphics[width=0.56\linewidth]{images/book1/ai/g4-breathing-mist.jpg}

{\small छोड़ी हुई हवा दिखाई देती हुई: गरम, नम --- और बिना दिखे
ऑक्सीजन में ग़रीब, कार्बन डाइऑक्साइड में अमीर।}
\end{omfigure}

\begin{method}[साँस की देखभाल]\label{met:g4:breathing:care}
\begin{enumerate}
  \item कमरों में रोज़ हवा आने दो --- बंद कमरे में बहुत-से साँस लेने
        वालों का मतलब है बासी, कार्बन डाइऑक्साइड से भरी हवा;
  \item ठंडी या धूल भरी हवा में नाक से साँस लो: वह हवा को गरम, नम और
        साफ़ कर देती है;
  \item दौड़ो, तैरो, खेलो --- क़सरत किए हुए \omterm{def:g4:breathing:organs}{फेफड़े} और श्वास की
        \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} मज़बूत और गहरी होती हैं;
  \item धुएँ को अपने फेफड़ों से दूर रखो: वह नाज़ुक कोठरियों को गंदा
        करता है, और वे अपनी सफ़ाई बहुत धीरे कर पाती हैं।
\end{enumerate}
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:breathing:1}
साँस की दोनों गतियों के नाम बताओ, और हर एक में छाती क्या करती है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:2}
नाक से नन्ही हवा-कोठरियों तक एक साँस का रास्ता बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:3}
आराम में बैठा बच्चा मिनट में लगभग कितनी बार साँस लेता है? दौड़ के दौरान
इस संख्या का क्या होता है, और क्यों?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:4}
शरीर हवा से कौन-सी गैस लेता है, और कौन-सी लौटाता है? यह अदला-बदली
ठीक-ठीक कहाँ होती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:5}
छोड़ी हुई हवा ली हुई हवा से कैसे अलग होती है? तीन फ़र्क़ बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:6}
\omterm{def:g4:breathing:organs}{फेफड़े} दो सादे थैलों के बजाय लाखों नन्ही कोठरियों के रूप में क्यों बने
हैं? इस साल का कौन-सा और अंग यही करामात इस्तेमाल करता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:7}
कड़ाके की ठंड या धूल भरे दिन नाक से साँस लेना बेहतर क्यों है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:8}
करके देखो: शांति से बैठकर एक मिनट अपनी साँसें गिनो। फिर बीस बार
उछल-कूद करके दोबारा गिनो। दोनों संख्याएँ बताओ और बदलाव समझाओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:9}
खाना घंटों रुक सकता है; साँस एक मिनट भी नहीं रुक सकती। इससे यह पता
चलता है कि शरीर भोजन के भंडार के मुक़ाबले ऑक्सीजन का कितना भंडार रखता
है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:10}
सुबह के आख़िर तक कक्षा ``घुटी हुई'' लगती है।
\cref{prop:g4:breathing:exchange} के अनुसार उसकी हवा में क्या बदला है,
और \cref{met:g4:breathing:care} क्या नुस्ख़ा बताता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:breathing:11}
ट्राउट के \omterm{def:g4:breathing:organs}{फेफड़े} नहीं होते: पानी उसके पंखदार क्लोमों पर बहता है, और
\emph{पानी में घुली} ऑक्सीजन वहीं उसके \omterm{prop:g4:journey-of-food:absorb}{रक्त} में उतर जाती है। क्लोम और
\omterm{def:g4:breathing:organs}{फेफड़े} कौन-सा काम साझा करते हैं? और हवा में, जहाँ ऑक्सीजन भरपूर है,
ट्राउट का दम क्यों घुट जाता है? (संकेत: पानी से बाहर पंखदार क्लोम-सतहें
बैठकर आपस में चिपक जाती हैं।)
\end{exercise}
