\chapter{Les poumons et les échanges gazeux}\label{ch:g7:lungs-and-gas-exchange}

Chaque minute de ta vie, environ huit litres d'air traversent deux
organes que tu n'as jamais vus. \Cref{ch:g4:breathing} esquissait le
trajet et l'échange~; \cref{ch:g7:how-animals-breathe} plaçait la
machine humaine parmi les quatre du règne \omterm{def:g1:animals-around-us:animal}{animal}. Ce chapitre
complète le tableau~: la machinerie de la \omterm{def:g7:what-is-respiration:respiration}{ventilation}, les poches
d'air appelées par leur vrai nom, les nombres de l'échange --- et ce
que la fumée fait à l'ensemble.

\section{La machinerie de la ventilation}

\begin{proposition}[Comment l'air est déplacé]\label{prop:g7:lungs-and-gas-exchange:ventilation}
Les \omterm{def:g7:how-animals-breathe:four}{poumons} n'ont pas de \omterm{def:g3:bones-and-muscles:muscle}{muscles} à eux~; c'est la cage thoracique
qui les pompe comme un soufflet~:
\begin{itemize}
  \item inspirer est un \emph{travail}~: les \omterm{def:g3:bones-and-muscles:muscle}{muscles} des \omterm{def:g3:bones-and-muscles:skeleton}{côtes}
        soulèvent et élargissent la cage, et le
        \emph{diaphragme}\index{diaphragme} --- le large plancher
        musclé sous les \omterm{def:g7:how-animals-breathe:four}{poumons} --- s'aplatit vers le bas~; la cage
        élargie étire les \omterm{def:g7:how-animals-breathe:four}{poumons}, et l'air entre les remplir~;
  \item expirer, au repos, est surtout un \emph{relâchement}~: les
        \omterm{def:g3:bones-and-muscles:muscle}{muscles} se détendent, la cage étirée revient d'elle-même,
        l'air ressort. (Souffler des bougies ajoute du \omterm{def:g3:bones-and-muscles:muscle}{muscle} au
        ressort.)
\end{itemize}
Environ $15$ fois par minute au repos
(\cref{def:g4:breathing:movements}), sans qu'on le demande
(\cref{rem:g5:brain-in-command:more}) --- et plus profondément et
plus vite à la demande
(\cref{prop:g7:organs-at-work:supply}).
\end{proposition}
\admitted

\begin{example}[Sentir le soufflet]\label{ex:g7:lungs-and-gas-exchange:feel}
Les mains sur les \omterm{def:g3:bones-and-muscles:skeleton}{côtes}, inspire profondément~: la cage se soulève
et s'élargit sous tes paumes (\cref{ex:g4:breathing:watch}).
Halète maintenant comme un chien~: ces bouffées rapides et
superficielles sont presque tout entières du \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{diaphragme}. Le hoquet,
pour être complet, ce sont les faux départs saccadés du \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{diaphragme}
--- le soufflet qui rate son coup à son propre calendrier.
\end{example}

\section{Les alvéoles et l'échange}

\begin{definition}[Alvéoles]\label{def:g7:lungs-and-gas-exchange:alveoli}
Les minuscules poches d'air de \cref{def:g4:breathing:organs} sont
les \emph{alvéoles}\index{alvéole}~: des centaines de millions de
bulles à paroi mince, groupées à l'extrémité la plus fine des voies
aériennes, chacune enveloppée de fins \omterm{def:g4:heart-and-blood:vessels}{vaisseaux sanguins}. Leur
surface totale est le demi-court de tennis de
\cref{rem:g4:breathing:sponge} --- la copie humaine du cahier des
charges «~grande, mince, humide~» de
\cref{prop:g7:how-animals-breathe:spec}.
\end{definition}

\begin{proposition}[L'échange, en chiffres]\label{prop:g7:lungs-and-gas-exchange:numbers}
Compare l'air qui entre et l'air qui sort~: l'air inspiré contient
environ $21$ pour cent de \omterm{def:g7:what-is-respiration:respiration}{dioxygène} et presque pas de \omterm{def:g7:what-is-respiration:respiration}{dioxyde de
carbone}~; l'air expiré est descendu à environ $16$ pour cent de
\omterm{def:g7:what-is-respiration:respiration}{dioxygène} et monté jusqu'à environ $4$ pour cent de \omterm{def:g7:what-is-respiration:respiration}{dioxyde de
carbone} (chaud et humide de surcroît ---
\cref{ex:g4:breathing:mirror}). Ces différences sont l'échange~: à
travers les parois alvéolaires, le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} passe vers le \omterm{prop:g4:journey-of-food:absorb}{sang} et
le \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} repasse en sens inverse
(\cref{prop:g4:breathing:exchange}), à chaque minute de chaque jour.
\end{proposition}
\admitted

\begin{example}[La comptabilité tombe juste]\label{ex:g7:lungs-and-gas-exchange:accounting}
Huit litres par minute, un vingtième du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} de chaque souffle
retenu~: l'arithmétique fournit à peu près la facture de \omterm{def:g7:what-is-respiration:respiration}{dioxygène}
du corps au repos --- et à l'effort, avec une \omterm{def:g7:what-is-respiration:respiration}{ventilation} triplée et
approfondie et les tournées du \omterm{def:g4:heart-and-blood:heart}{cœur} accélérées
(\cref{prop:g7:organs-at-work:supply}), la livraison se multiplie
pour égaler la facture des \omterm{def:g3:bones-and-muscles:muscle}{muscles}. Les nombres de la machine et le
registre du corps s'accordent~; ce sont les mêmes livres tenus à
deux guichets.
\end{example}

\section{La machinerie maltraitée~: la fumée}

\begin{proposition}[Ce que la fumée fait à l'ensemble]\label{prop:g7:lungs-and-gas-exchange:smoke}
La fumée de tabac attaque chaque étage de ce chapitre
(\cref{prop:g5:healthy-choices:substances} donnait
l'avertissement~; voici l'ingénierie)~:
\begin{itemize}
  \item ses goudrons enduisent et irritent les voies aériennes~:
        les surfaces de nettoyage faiblissent, et la «~toux du
        fumeur~» est le balai débordé des voies aériennes~;
  \item les parois alvéolaires, irritées en permanence, se
        dégradent~: les poches fusionnent en espaces moins
        nombreux, plus grands, plus flasques --- \emph{de la surface
        est perdue}, et l'échange avec elle (le cahier des charges,
        démantelé)~;
  \item l'un des gaz de la fumée occupe les places du \omterm{def:g7:what-is-respiration:respiration}{dioxygène}
        dans le \omterm{prop:g4:journey-of-food:absorb}{sang}, en le chassant de son propre transport~;
  \item et les voies aériennes se rétrécissent --- chaque souffle
        dans une poitrine de fumeur est un souffle dans un soufflet
        en partie bouché.
\end{itemize}
Les dégâts sont lents, cumulatifs et seulement en partie
réversibles~: le \omterm{def:g7:how-animals-breathe:four}{poumon} qui n'a jamais commencé garde entier son
demi-court de tennis (\cref{rem:g5:healthy-choices:never}).
\end{proposition}
\admitted

\begin{example}[Deux épreuves d'escalier]\label{ex:g7:lungs-and-gas-exchange:stairs}
Les trois étages de \cref{ch:g7:organs-at-work}, montés par deux
adultes de même âge et de mêmes habitudes d'entraînement, dont l'un
fume depuis longtemps~: le fumeur arrive essoufflé à un \omterm{prop:g4:heart-and-blood:pulse}{pouls} que
l'autre n'atteint qu'en sprintant. Moins de surface, des places
prises, des voies rétrécies --- la chaîne d'approvisionnement paie
la facture de \cref{prop:g7:organs-at-work:bill} à travers une
admission étranglée. Les tests de condition physique lisent
l'histoire des \omterm{def:g7:how-animals-breathe:four}{poumons} aussi sûrement que n'importe quelle imagerie.
\end{example}

\begin{method}[Entretenir la machinerie]\label{met:g7:lungs-and-gas-exchange:keep}
\Cref{met:g4:breathing:care}, remise à niveau avec les raisons de ce
chapitre~:
\begin{enumerate}
  \item pas de fumée --- ni la tienne ni celle des autres~: la
        surface perdue ne se regagne guère~;
  \item aère les pièces (\cref{exo:g4:breathing:10}) et préfère
        pour le sport un air extérieur propre~;
  \item exerce le soufflet~: un effort qui fait respirer
        profondément \omterm{def:g3:bones-and-muscles:muscle}{muscle} la \omterm{def:g7:what-is-respiration:respiration}{ventilation} et garde toute la
        surface ventilée~;
  \item respire par le nez dans le froid et la poussière
        (\cref{exo:g4:breathing:7}) --- le filtre protège le balai.
\end{enumerate}
\end{method}

\section{Problème~: le dossier de la machine respiratoire}

\begin{problem}[{Problème du week-end --- l'inspection du soufflet
humain par des ingénieurs}]\label{pb:g7:lungs-and-gas-exchange:1}
Une classe d'ingénierie «~inspecte~» sur le papier la machine
respiratoire humaine~: admission, pompe, surface d'échange, liaison
au transport, et carnet d'entretien.

\medskip\noindent\textbf{Partie I --- La pompe.}
\begin{enumerate}
  \item Dresse le cycle de la pompe~: les deux phases, ce qui les
        actionne, et laquelle coûte de l'\omterm{prop:g3:balanced-diet:jobs}{énergie} au repos.
  \item La pompe n'est en contact avec aucun \omterm{def:g3:bones-and-muscles:muscle}{muscle} de l'organe
        pompé --- les \omterm{def:g7:how-animals-breathe:four}{poumons} n'en ont pas. Comment
        l'élargissement de la cage déplace-t-il donc de l'air~?
  \item Donne le rythme de la pompe au repos et ses deux réglages
        d'effort
        (\cref{prop:g7:organs-at-work:supply}).
  \item Haleter, soupirer, souffler des bougies~: attribue à chacun
        sa place dans la mécanique du cycle.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- La surface
d'échange.}
\begin{enumerate}[resume]
  \item Décris la surface~: nom, nombre, paroi, enveloppe, taille
        totale.
  \item Vérifie le cahier des charges de
        \cref{prop:g7:how-animals-breathe:spec}, point par
        point.
  \item L'air d'admission et l'air d'échappement diffèrent des
        pourcentages du chapitre. Calcule la différence de
        \omterm{def:g7:what-is-respiration:respiration}{dioxygène} en fraction des $21$ pour cent de l'admission ---
        quelle part environ du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} inspiré est retenue~?
  \item Pourquoi l'échappement reste-t-il chaud et humide --- de
        quelle propriété de la surface est-ce le prix~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le carnet
d'entretien.}
\begin{enumerate}[resume]
  \item Classe les quatre attaques de la fumée sous~: admission,
        pompe, surface, transport. Une ligne pour chacune.
  \item Le dossier note~: «~les pertes de surface sont des pertes
        en capital~». Explique, par opposition aux toux et aux
        rhumes qui sont des réparations courantes.
  \item Rédige le programme d'entretien ---
        \cref{met:g7:lungs-and-gas-exchange:keep} --- en quatre
        directives d'ingénieur, avec leurs raisons.
  \item Conclus l'inspection~: une phrase sur la raison pour
        laquelle cette machine, seule parmi les pompes et les
        frontières du corps, est à moitié sous le contrôle
        volontaire de son propriétaire --- et sur ce à quoi sert ce
        contrôle (la parole, le chant, l'\omterm{def:g4:breathing:movements}{expiration} lente du
        plongeur).
\end{enumerate}
\end{problem}
