\chapter{خيارات صحية}\label{ch:g5:healthy-choices}

زار هذا الكتاب إلى الآن أكثر خدمات جسمك الكبرى: \omterm{def:g4:journey-of-food:tract}{الأنبوب الهضمي}،
والرئتين، \omterm{def:g4:heart-and-blood:heart}{والقلب} وأنهار دمه، \omterm{def:g3:bones-and-muscles:muscle}{والعضلات} \omterm{def:g3:bones-and-muscles:skeleton}{والعظام}، \omterm{def:g5:brain-in-command:system}{والدماغ} القائد.
ويسأل فصل السنة الختامي هذا سؤالًا عمليًّا: إذا عرفت تلك الآلات
كلها، فكيف تُحسن تشغيلها --- ومن يقرر؟ فمن نحو سنّك فصاعدًا يصير
المزيد من القرارات قراراتك أنت.

\section{ماذا تطلب الآلة}

\begin{proposition}[عقد الجسم اليومي]\label{prop:g5:healthy-choices:contract}
ضع مطالب الفصول السابقة جنبًا إلى جنب يظهر لك عقد يومي:
\begin{itemize}
  \item \emph{\omterm{prop:g1:healthy-habits:needs}{غذاء متنوع} وماء} --- وقود ومادة بناء
        (\cref{prop:g3:balanced-diet:jobs})، يوصّلهما \omterm{prop:g4:journey-of-food:absorb}{الدم} إلى كل
        \omterm{def:g5:first-look-at-cells:cell}{خلية}؛
  \item و\emph{حركة} --- \omterm{def:g3:bones-and-muscles:muscle}{فالعضلات} \omterm{def:g3:bones-and-muscles:skeleton}{والعظام} \omterm{def:g4:heart-and-blood:heart}{والقلب} والرئتان كلها
        تقوى بالاستعمال
        (\cref{met:g3:bones-and-muscles:care}،
        \cref{rem:g4:heart-and-blood:rest})؛
  \item و\emph{\omterm{prop:g1:healthy-habits:needs}{نوم}} --- وهو وقت حفظ ملفات \omterm{def:g5:brain-in-command:system}{الدماغ} وإصلاحات الجسم
        (\cref{prop:g5:brain-in-command:protect})؛ وطفل في هذه
        السن يحتاج إلى نحو عشر ساعات؛
  \item و\emph{هواء نظيف، وأيدٍ نظيفة، وأسنان معتنًى بها} --- وهي
        دفاعات \cref{ch:g1:healthy-habits}
        و\cref{met:g4:breathing:care}، ما زالت في الخدمة.
\end{itemize}
لا شيء جديد --- وهذا هو المقصود. فالصحة هي القواعد القديمة
محفوظةً.
\end{proposition}
\admitted

\begin{example}[يوم واحد، منظورًا إليه من الداخل]\label{ex:g5:healthy-choices:day}
يوم جيد، مقروءًا إدارةً للجسم: الفطور يعيد تموين نفقات الليل
(\cref{ex:g3:balanced-diet:breakfast})؛ والمشي إلى المدرسة يسخّن
\omterm{def:g4:heart-and-blood:heart}{القلب} والرئتين؛ واللعب عند الظهر تمرين متنكر في زيّ متعة؛ وخضر
العشاء تعيد تموين أسرة الطزاجة؛ \omterm{prop:g1:healthy-habits:needs}{والنوم} الباكر يسلّم \omterm{def:g5:brain-in-command:system}{الدماغ} وقت حفظ
ملفاته. ولا بند من هذه بطولي؛ وهي مجتمعةً الصحة.
\end{example}

\section{خيارات جديدة وضغوط جديدة}

\begin{proposition}[الشاشات والنوم]\label{prop:g5:healthy-choices:screens}
الشاشات --- الهواتف، واللوحات، والألعاب --- ليست سُمًّا ولا غذاءً؛
إنها \emph{وقت}. والساعات التي تمتصّها مأخوذة من مكان ما، وضحاياها
المعتادون هم بالضبط بنود العقد: الحركة في الخارج \omterm{prop:g1:healthy-habits:needs}{والنوم}. والأسوأ
أن الشاشات الساطعة في آخر المساء تقول \omterm{def:g5:brain-in-command:system}{للدماغ} ``ما زال النهار''،
فتؤخر \omterm{prop:g1:healthy-habits:needs}{النوم} أبعد من الساعات المتنازَل عنها. فالخيار ليس شاشات أو
لا شاشات، بل أي الساعات تنالها --- وآخر ساعة من المساء أسوأ ساعة
تُعطى لها.
\end{proposition}
\admitted

\begin{example}[تجربة الحادية عشرة]\label{ex:g5:healthy-choices:latenight}
جرّبها اختبارًا منصفًا (\cref{met:g4:what-plants-need:fairtest}) على
نفسك: أسبوع بلا شاشات قبل \omterm{prop:g1:healthy-habits:needs}{النوم} بساعة، في مقابل أسبوع عادي.
وأكثر الناس ينامون أسرع، ويستيقظون أسهل، ويلاحظون صباحات أفضل في
الأسبوع الأول --- وأسبوع الشاهد هو الحجة.
\end{example}

\begin{proposition}[مواد تخدع الجسم]\label{prop:g5:healthy-choices:substances}
بعض المواد تؤذي بالضبط الآلات التي جال فيها هذا الكتاب:
\begin{itemize}
  \item \emph{دخان التبغ} يوسّخ جيوب الرئتين الرقيقة
        (\cref{met:g4:breathing:care}) ويجهد \omterm{def:g4:heart-and-blood:heart}{القلب}؛ وهو يُباع
        بجعل السجائر الأولى تبدو من شأن الكبار --- ويحبس
        مشتريه في \emph{الإدمان}\index{إدمان}: فالجسم، إذا
        اعتاده، احتجّ من غيره؛
  \item و\emph{الكحول} يبلغ \omterm{def:g5:brain-in-command:system}{الدماغ} في دقائق --- وهو القائد نفسه
        (\cref{ch:g5:brain-in-command}) --- فيبطّئ الحلقة،
        ويطمس القرارات؛ \omterm{def:g5:brain-in-command:system}{ودماغ} الصغير، وهو ما زال يمدّ مسالكه،
        أسهل تضررًا من \omterm{def:g5:brain-in-command:system}{دماغ} الكبير؛
  \item وكلاهما سبق أن أُشير إليه خطرًا على \omterm{def:g5:human-reproduction:pregnancy}{الجنين}
        (\cref{prop:g5:human-reproduction:care}) --- المادتان
        نفسهما، والسبب نفسه.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[لماذا يغلب ``ألّا تبدأ'' ``أن تقلع لاحقًا'']\label{rem:g5:healthy-choices:never}
\omterm{prop:g5:healthy-choices:substances}{الإدمان} فخ مدخله سهل ومخرجه عسير: فالبدء خيار واحد، والإقلاع معركة
تتكرر سنوات. وذلك التفاوت أبرد حقيقة في هذا الفصل، وأنفعها: فأرخص
لحظة لكسب المعركة هي ما قبل بدئها. ومن يعرض عليك سيجارة أولى لا
يعرض واحدة --- بل يعرض كل ما يتلوها.
\end{remark}

\section{أن تقرر لنفسك}

\begin{method}[إدارة عقدك بنفسك]\label{met:g5:healthy-choices:run}
من الآن فصاعدًا، وأكثر في كل سنة، تكون أنت حافظ عقد جسمك:
\begin{enumerate}
  \item اعرف بنود العقد ---
        و\cref{prop:g5:healthy-choices:contract} هي القائمة؛
  \item وراجع نفسك بصدق بين حين وآخر، كما في
        \cref{met:g3:balanced-diet:check} --- ما الناقص، وما
        الزائد عن حصته؟
  \item وغيّر شيئًا واحدًا في المرة، وأعطِه أسبوع اختبار منصف؛
  \item وفي المواد، قرر ``لا'' \emph{قبل} اللحظة التي تحتاج فيها
        إليها (\cref{rem:g5:healthy-choices:never}) --- فالقرار
        المتخذ في هدوء سلفًا أسهل حفظًا من قرار يُرتجل تحت ضغط
        ودّي؛
  \item واستعن بالكبار: فالأطباء والممرضون والوالدان والمعلمون في
        صفّك من هذا العقد.
\end{enumerate}
\end{method}

\begin{example}[أجوبة مهيَّأة]\label{ex:g5:healthy-choices:answers}
القرار المسبق في الخطوة الرابعة هو، عمليًّا، جملة جاهزة في الجيب:
``لا شكرًا --- ليس من شأني.'' تُقال بخفة مرة واحدة فتنهي الأمر في
العادة؛ والأصدقاء الجديرون بالاسم يقبلونها من أول مرة. والجملة لا
تكلّف شيئًا في التهيئة، وهي أصعب ما تُؤلَّف حين يُحتاج إليها بالضبط
--- ولذلك تُؤلَّف الآن.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:1}
عدّد بنود عقد الجسم اليومي الأربعة، مع العضو أو الخدمة التي يخدمها
كل بند في المقام الأول.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:2}
كم ساعة \omterm{prop:g1:healthy-habits:needs}{نوم} يحتاج إليها طفل في سنّك تقريبًا، وأي عملين يقوم بهما
الجسم فيها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:3}
بأي معنى تكون ساعات الشاشة ``مأخوذة من مكان ما''؟ سمِّ الضحيتين
المعتادتين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:4}
لماذا كانت آخر ساعة من المساء أسوأ ساعة تُعطى لشاشة ساطعة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:5}
أي الأعضاء يؤذيها دخان التبغ أولًا وفق
\cref{prop:g5:healthy-choices:substances}؟ وأيها يؤذيها الكحول؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:6}
ما \omterm{prop:g5:healthy-choices:substances}{الإدمان}، ولماذا يجعل ``ألّا تبدأ'' أرخص بكثير من ``أن تقلع
لاحقًا''؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:7}
ما فائدة أن تقرر ``لا'' سلفًا، وفق الخطوة الرابعة في
\cref{met:g5:healthy-choices:run}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:8}
جرّب: أجرِ تجربة الحادية عشرة في
\cref{ex:g5:healthy-choices:latenight} أسبوعًا واحدًا، ثم اذكر ---
بصدق --- ما الذي تغيّر.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:healthy-choices:9}
صمّم تجربة الشاشة اختبارًا منصفًا كما ينبغي: ما الذي يبقى واحدًا
بين الأسبوعين، وأي شيء واحد يتغير، وماذا تقيس؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:healthy-choices:10}
``سيجارة واحدة لا يمكن أن تضرّ.'' أجب وفق
\cref{rem:g5:healthy-choices:never}: ما الذي يُعرض حقًّا مع
الأولى؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:healthy-choices:11}
اربط ثلاثة فصول سابقة بهذا الفصل: كيف يشرح كل من
\cref{prop:g4:breathing:exchange}
و\cref{ex:g4:heart-and-blood:effort}
و\cref{prop:g5:brain-in-command:protect} بندًا من بنود العقد أو
تحذيرًا من تحذيرات هذا الفصل؟
\end{exercise}
