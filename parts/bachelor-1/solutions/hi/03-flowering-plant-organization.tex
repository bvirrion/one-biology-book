\section*{अध्याय \ref{ch:b1:flowering-plant-organization} --- किसी पुष्पी पौधे का कार्यात्मक संगठन}

\begin{solution}{exo:b1:flowering-plant-organization:1}
\omterm{def:b1:flowering-plant-organization:organs}{जड़}: लंगर डालना, जल और खनिज आयनों का अवशोषण। \omterm{def:b1:flowering-plant-organization:organs}{तना}: \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} को प्रकाश की
ओर सहारा देना, \omterm{def:b1:flowering-plant-organization:organs}{जड़ों} और \omterm{def:b1:flowering-plant-organization:organs}{पत्तियों} के बीच चालन।
\omterm{def:b1:flowering-plant-organization:organs}{पत्ती}: प्रकाश-संश्लेषण (और गैस-विनिमय, वाष्पोत्सर्जन)।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:2}
कोई मॉड्यूल एक \omterm{def:b1:flowering-plant-organization:organs}{पर्वसंधि} है, उसकी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} और कक्षस्थ कली के साथ, जोड़ नीचे का
पर्व। \omterm{prop:b1:flowering-plant-organization:modular}{अनिश्चित वृद्धि}: कोई वयस्क आकार नहीं; शीर्षस्थ और कक्षस्थ कलियाँ
जब तक पौधा जीता है तब तक मॉड्यूल जोड़ती रहती हैं।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:3}
\omterm{def:b1:flowering-plant-organization:tissues}{अधिचर्म}: जीवित; ढँकना, उपत्वचा, \omterm{def:b1:flowering-plant-organization:tissues}{रंध्र} और \omterm{def:b1:flowering-plant-organization:organs}{मूलरोम}। \omterm{def:b1:flowering-plant-organization:tissues}{मृदूतक}:
जीवित; प्रकाश-संश्लेषण, संचय। \omterm{def:b1:flowering-plant-organization:tissues}{स्थूलकोणोतक}: जीवित; लचीला सहारा।
\omterm{def:b1:flowering-plant-organization:tissues}{दृढ़ोतक}: मृत; कठोर सहारा। \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} की चालक कोशिकाएँ: मृत; जल का
परिवहन। \omterm{def:b1:flowering-plant-organization:tissues}{फ्लोएम} की चालनी नलिकाएँ: जीवित (सहचर कोशिकाओं के साथ); शर्करा का
परिवहन।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:4}
$e^{-0.9} = 0.41$, यानी \qty{41}{\%}; $e^{-3} = 0.05$, यानी
\qty{5}{\%}।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:5}
किसी द्विबीजपत्री की \omterm{def:b1:flowering-plant-organization:organs}{जड़} (एक ही केंद्रीय स्तंभ, \omterm{prop:b1:flowering-plant-organization:stemroot}{अंतस्त्वचा}, किसी त्रिज्या
पर एकांतर \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} तथा \omterm{def:b1:flowering-plant-organization:tissues}{फ्लोएम}; कम जाइलम-भुजाएँ, कोई केंद्रीय मज्जा नहीं)।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:6}
किसी मुड़ते धरन पर प्रतिबल उसके पृष्ठ पर सबसे अधिक और उसके अक्ष के साथ
बिलकुल शून्य होता है, इसलिए किसी परिधीय छल्ले में रखा पदार्थ सबसे कम
द्रव्यमान से मुड़ने का प्रतिरोध करता है (एक नली)। तनाव में कोई रस्सी अपने
काट भर एकसमान प्रतिबल में रहती है और स्वच्छंद मुड़ती है; कोई केंद्रीय
लड़ी खिंचाव ढोती है जबकि वल्कुट लचीला रहता है, और उसमें से पार्श्व \omterm{def:b1:flowering-plant-organization:organs}{जड़ें}
निकल सकती हैं।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:7}
$1 - e^{-2.4} = 0.91$; और $L = 6$ पर, $1 - e^{-3.6} = 0.97$: \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} की दो
और परतें प्रकाश का \qty{6}{\%} जोड़ती हैं जबकि \qty{50}{\%}
अधिक \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} की कीमत पड़ती है।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:8}
$S = \pi d L$: $d = 240/(\pi\times 623\,000) = \qty{1.2e-4}{m}$, यानी
लगभग \qty{0.12}{mm}। $(240 + 400)/5 = 128$।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:9}
प्रति mm \omterm{def:b1:flowering-plant-organization:organs}{जड़} का पृष्ठ: $\pi\times 0.4 = \qty{1.26}{mm^2}$। प्रति mm रोम:
$250\times\pi\times 0.010\times 0.8 = \qty{6.3}{mm^2}$। कुल
$7.5/1.26 = 6$: रोम पृष्ठ को छह गुना कर देते हैं।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:10}
वन में नीचे की कक्षस्थ कलियाँ छाया में रहती हैं; वे जो मॉड्यूल बनातीं वे
अपनी कीमत नहीं चुका सकते, इसलिए वे सुप्त रह जाती हैं या उनकी शाखाएँ मर
जाती हैं, और वृद्धि उस शीर्षस्थ कली को दे दी जाती है जो प्रकाश की ओर ऊपर
दौड़ रही है। खेत में हर कली को उजाला मिलता है और हर शाखा कीमत चुकाती है,
इसलिए शिखर नीचे तक भर जाता है। लागत: कोई ऊँचा पतला \omterm{def:b1:flowering-plant-organization:organs}{तना} अनुत्पादक लकड़ी
में एक बड़ा निवेश है, जो हवा के आगे भंगुर है; लाभ: पौधा अपने रूप को वहाँ
से मिला लेता है जहाँ प्रकाश है, बिना किसी तय योजना के।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:11}
गड्ढा और रोम छिद्र के ऊपर ठहरी, नम हवा की एक परत थामे रखते हैं; वायु-अवकाशों
से चलती हवा तक वाष्प की प्रवणता एक लंबे रास्ते पर फैल जाती है और
वाष्पोत्सर्जन गिर जाता है। कार्बन डाइऑक्साइड को भी उसी लंबे रास्ते से
भीतर विसरित होना पड़ता है, इसलिए ग्रहण भी गिरता है, हालाँकि कम, क्योंकि
$\mathrm{CO_2}$ की प्रवणता वायुमंडल और \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} की खपत से तय होती है। किसी
मरुस्थल में जल ही सीमाकारी संसाधन है और यह सौदा पौधे की जान बचाता है;
किसी वर्षावन में जल मुफ़्त है और होड़ कार्बन की कमाई को ही पुरस्कृत करती है।
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:12}
पौधे के पास ऐसा कोई नियमित बाह्यकोशिकीय तरल नहीं जो उसकी कोशिकाओं और बाहर
के बीच अचर तापमान तथा संघटन पर थामा जाए: उसकी कोशिकाएँ मिट्टी के जल,
हवा और धूप से सीधे मिलती हैं, और उसका तापमान हवा का ही तापमान है। इसलिए
हर कोशिका अपनी झिल्ली और अपनी भित्ति के आरपार अपनी ही सामग्री (आयन,
जल, pH) का नियमन करती है; पौधा अपने विनिमय पृष्ठ पर नियमित करता है (\omterm{def:b1:flowering-plant-organization:tissues}{रंध्र}
खुलते और बंद होते हैं, \omterm{prop:b1:flowering-plant-organization:stemroot}{अंतस्त्वचा} चुनती है कि \omterm{def:b1:flowering-plant-organization:tissues}{जाइलम} में क्या घुसे) और
अपने शरीर को वृद्धि से समायोजित करता है (सूखी मिट्टी में अधिक \omterm{def:b1:flowering-plant-organization:organs}{जड़ें}, छाया
में अधिक \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ})। वह नियमन करता है, पर वह नियमन पृष्ठों पर और रूप में
है, किसी निजी तरल में नहीं।
\end{solution}

\begin{solution}{pb:b1:flowering-plant-organization:1}
\textbf{1.} $5\times 100 = \qty{500}{m^2}$।
\textbf{2.} $e^{-2.5} = 0.082$ ज़मीन तक पहुँचता है; \qty{92}{\%}
रोक लिया गया।
\textbf{3.} $L = 2$: $1 - e^{-1} = \qty{63}{\%}$; $L = 8$: $1 -
e^{-4} = \qty{98}{\%}$।
\textbf{4.} 2 से 5 तक शिखर \qty{300}{m^2} \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} के बदले प्रकाश का
\qty{29}{\%} कमाता है; 5 से 8 तक वह \qty{6}{\%} कमाता, और उसके लिए
\qty{300}{m^2} और चाहिए जिन्हें गढ़ना तथा जल पहुँचाना उतना ही महँगा है:
अतिरिक्त \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} कीमत नहीं चुकातीं।
\textbf{5.} $500/0.005 = \num{100000}$ \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ}।
\textbf{6.} $500\times 10^6\,\mathrm{mm^2}\times 150 = \num{7.5e10}$
\omterm{def:b1:flowering-plant-organization:tissues}{रंध्र}।
\textbf{7.} $\qty{50}{\micro m^2}$; प्रति mm$^2$: $150\times 50\times
10^{-6} = \qty{7.5e-3}{mm^2}$, यानी पृष्ठ का \qty{0.75}{\%}।
\textbf{8.} $500\times 4\times 10^{-6} = \qty{2e-3}{mol/s}$, यानी
\qty{88}{mg/s}।
\textbf{9.} $0.088\times 43\,200 = \qty{3.8}{kg}$ जितनी $\mathrm{CO_2}$,
जिसमें $3.8\times 12/44 = \qty{1.04}{kg}$ कार्बन है।
\textbf{10.} $150\times 1.04 = \qty{156}{kg}$ कार्बन; उसका आधा,
\qty{78}{kg}, \qty{50}{\%} कार्बन वाली लकड़ी में: \qty{156}{kg} लकड़ी।
\textbf{11.} $500\times 10^{-3} = \qty{0.5}{mol/s}$, यानी \qty{9}{g/s}।
\textbf{12.} $9\times 43\,200 = \qty{389}{kg}$, यानी लगभग \qty{390}{L}।
\textbf{13.} $0.5/(2\times 10^{-3}) = 250$ जल-अणु प्रति $\mathrm{CO_2}$।
जो छिद्र $\mathrm{CO_2}$ को भीतर आने देता है वही जल को बाहर जाने देता है;
भीतर की हवा संतृप्त है जबकि $\mathrm{CO_2}$ बाहर की हवा का केवल
\qty{0.04}{\%} है, इसलिए वाष्प की बाहरी प्रवणता $\mathrm{CO_2}$ की भीतरी
प्रवणता से कहीं तीखी है।
\textbf{14.} $\qty{2}{mm}\times\qty{100}{m^2} = \qty{200}{L}$: यानी आधा
दिन।
\textbf{15.} \qty{100}{m^3}; $5\times 100 = \qty{500}{km}$।
\textbf{16.} $\pi\times 0.5\times 10^{-3}\times 5\times 10^5 =
\qty{785}{m^2}$।
\textbf{17.} $200\times 5\times 10^8\,\mathrm{mm} = \num{1e11}$ रोम।
\textbf{18.} हर एक $\pi\times 10^{-5}\times 5\times 10^{-4} =
\qty{1.57e-8}{m^2}$; कुल \qty{1570}{m^2}।
\textbf{19.} $785 + 1570 = \qty{2355}{m^2}$, यानी पर्ण-क्षेत्रफल का $4.7$ गुना।
\textbf{20.} \qty{1}{mm} के भीतर आयतन: $\pi\times(10^{-3})^2\times 5
\times 10^5 = \qty{1.6}{m^3}$, यानी मिट्टी का \qty{1.6}{\%}।
\textbf{21.} राई: \qty{0.05}{m^3} के बक्से में से $\pi\times 10^{-6}\times 6.2\times 10^5 =
\qty{1.95}{m^3}$, यानी \qty{3900}{\%}: मिट्टी का हर
बिंदु चालीस \omterm{def:b1:flowering-plant-organization:organs}{जड़ों} से एक मिलीमीटर के भीतर है। राई अपनी मिट्टी को
दो हज़ार गुने से भी अधिक गहराई से टटोलती है ---
चार महीने के बजट पर कोई फ़सली पौधा, और उसके सामने सौ घन मीटर टटोलता कोई
वृक्ष।
\textbf{22.} $1000 + 2355 = \qty{3355}{m^2}$; ज़मीन के प्रति वर्ग मीटर
$\qty{34}{m^2}$।
\textbf{23.} मनुष्य: $130/2 = 65$; बलूत का अनुपात मनुष्य के मुड़े हुए
पृष्ठों के अनुपात का दुगना है, और वह भी बिना किसी मोड़ के।
\textbf{24.} बलूत हर साल मॉड्यूल, \omterm{def:b1:flowering-plant-organization:organs}{पत्तियाँ} और \omterm{def:b1:flowering-plant-organization:organs}{जड़ें} जोड़ता है, इसलिए उसके
पृष्ठ उसकी उम्र के साथ बढ़ते हैं; स्तनधारी के पृष्ठ नियत \omterm{def:b1:mammal-organization:organ}{अंग} हैं, जो वयस्क
आकार पर पूरे हो जाते हैं, और उसका अनुपात चढ़ाने का एकमात्र उपाय यही था कि
परिवर्धन के दौरान उन्हें मोड़ दिया जाए।
\textbf{25.} ज़मीन के प्रति वर्ग मीटर लगभग \qty{34}{m^2} विनिमय-पृष्ठ:
ऊपर \qty{10}{m^2} \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} (दोनों फलक) और नीचे \qty{24}{m^2}
\omterm{def:b1:flowering-plant-organization:organs}{जड़} तथा \omterm{def:b1:flowering-plant-organization:organs}{मूलरोम}।
\end{solution}
