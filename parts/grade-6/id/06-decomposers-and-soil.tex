\chapter{Pengurai dan Tanah}\label{ch:g6:decomposers-and-soil}

Setiap musim gugur, hutannya menjatuhkan hamparan \omterm{def:g1:plants-around-us:parts}{daun} yang tebal ---
berton-ton per hektare. Namun berjalanlah di hutan yang sama tiap
musim semi dan kamu tidak mengarungi lautan serasah yang meninggi:
hamparannya tipis kembali, tahun demi tahun. Ada yang memakan yang
mati di hutan itu. Bab ini turun ke dalam \omterm{def:g6:decomposers-and-soil:soil}{tanah} untuk menemui para
pemakannya --- dan untuk akhirnya menutup lingkaran materinya.

\section{Tanah itu hidup}

\begin{definition}[Tanah]\label{def:g6:decomposers-and-soil:soil}
Yang disebut \emph{tanah}\index{tanah} adalah \omterm{def:g1:the-five-senses:sense}{kulit} hidup bumi:
campuran butir \emph{mineral} (batuan yang tergerus, dari pasir sampai
lempung), sisa \emph{organik} dalam setiap tahap pelapukan, air, udara
--- dan sebuah populasi \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang amat besar. Lapisan atasnya
yang gelap, yang paling kaya materi lapuk, adalah lapisan
\emph{humus}\index{humus} yang dihargai para tukang kebun.
\end{definition}

\begin{example}[Segenggam, yang didata]\label{ex:g6:decomposers-and-soil:handful}
Segenggam \omterm{def:g6:decomposers-and-soil:soil}{tanah} hutan, yang dikelompokkan di atas nampan putih: butir
mineral; benang \omterm{def:g1:plants-around-us:parts}{akar}; setengah \omterm{def:g1:plants-around-us:parts}{daun}, kerangka \omterm{def:g1:plants-around-us:parts}{daun} dan remah hitam ---
\omterm{def:g6:plants-are-producers:matter}{materi organik} yang sedang dalam perjalanan turun; dan pendataannya
--- seekor cacing \omterm{def:g6:decomposers-and-soil:soil}{tanah}, dua ekor kutu kayu, seekor kaki seribu, ekor
pegas putih yang mungil, anyaman \omterm{def:g6:decomposers-and-soil:decomposers}{jamur} setipis benang. Dan nampannya
hanya memperlihatkan yang terlihat: satu gram \omterm{def:g6:decomposers-and-soil:soil}{tanah} memuat \omterm{def:g6:exploring-our-environment:components}{makhluk
hidup} yang terlalu kecil untuk dilihat lebih banyak daripada jumlah
manusia di sebuah kota besar.
\end{example}

\begin{definition}[Pengurai]\label{def:g6:decomposers-and-soil:decomposers}
Yang disebut \emph{pengurai}\index{pengurai} adalah \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang
makan dari \emph{\omterm{def:g6:plants-are-producers:matter}{materi organik} yang mati} --- \omterm{def:g1:plants-around-us:parts}{daun} yang gugur, kayu
mati, tinja, bangkai:
\begin{itemize}
  \item awak yang terlihat: cacing \omterm{def:g6:decomposers-and-soil:soil}{tanah}, kutu kayu, kaki seribu, ekor
        pegas, larva lalat, kumbang tinja;
  \item \emph{jamur}\index{jamur}: kapang dan cendawan, yang tubuh
        pemakannya berupa anyaman benang putih yang terbentang di dalam
        \omterm{def:g6:decomposers-and-soil:soil}{tanah} dan kayu mati --- cendawan pada
        \cref{exo:g1:living-or-not:10} hanyalah \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya};
  \item dan, yang menyelesaikan setiap pekerjaan, kehidupan \omterm{def:g6:decomposers-and-soil:soil}{tanah} yang
        terlalu kecil untuk dilihat --- para pembongkar yang terakhir.
\end{itemize}
Merekalah para pekerja bayangan pada \cref{def:g5:food-webs:roles},
yang dijanjikan bab tersendiri: inilah bab itu.
\end{definition}

\section{Penguraian: dari organik kembali menjadi mineral}

\begin{proposition}[Apa yang dikerjakan penguraian]\label{prop:g6:decomposers-and-soil:decomposition}
Yang disebut \emph{penguraian}\index{penguraian} adalah kegiatan makan
para \omterm{def:g6:decomposers-and-soil:decomposers}{pengurainya}, dan hasilnya adalah kebalikan dari perdagangan para
\omterm{def:g5:food-webs:roles}{produsen}: materi \emph{organik} yang mati dibongkar, tahap demi tahap
dan pemakan demi pemakan, sampai zatnya dikembalikan ke dunia
\emph{mineral} --- menjadi \omterm{def:g6:decomposers-and-soil:soil}{humus} yang gembur, lalu \omterm{prop:g4:what-plants-need:needs}{garam mineral} yang
terlarut di dalam air \omterm{def:g6:decomposers-and-soil:soil}{tanah}, dan gas yang dikembalikan ke udara.
Hamparan musim gugurnya menjadi, menjelang musim semi, persis apa yang
diminum \omterm{def:g1:plants-around-us:parts}{akar} (\cref{prop:g4:what-plants-need:needs}).
\end{proposition}
\admitted

\begin{example}[Tahun terakhir sehelai daun]\label{ex:g6:decomposers-and-soil:leaf}
Ikutilah sehelai \omterm{def:g1:plants-around-us:parts}{daun} pohon ek turun ke bawah. November: gugur, utuh.
Musim dingin: melunak, ditambang benang \omterm{def:g6:decomposers-and-soil:decomposers}{jamur}, digerogoti menjadi
kerangka renda oleh ekor pegas dan kutu kayu. Musim semi: diseret ke
bawah \omterm{def:g6:decomposers-and-soil:soil}{tanah} oleh seekor cacing \omterm{def:g6:decomposers-and-soil:soil}{tanah}, dimakan, lalu dikeluarkan
sebagai remah olahan cacingnya. Musim panas: kepingan terakhirnya
menjadi \omterm{def:g6:decomposers-and-soil:soil}{humus} hitam, dan mineralnya menumpang air \omterm{def:g6:decomposers-and-soil:soil}{tanah} --- masuk ke
dalam sebuah \omterm{def:g1:plants-around-us:parts}{akar}, barangkali \omterm{def:g1:plants-around-us:parts}{akar} pohon eknya sendiri. Sesudah
dimakan empat kali, \omterm{def:g1:plants-around-us:parts}{daunnya} menjadi makanan bagi pohonnya sendiri.
\end{example}

\begin{remark}[Lingkarannya menutup]\label{rem:g6:decomposers-and-soil:circle}
Dengan para \omterm{def:g6:decomposers-and-soil:decomposers}{pengurainya} pada tempatnya, kisah materi tahun ini menutup
menjadi sebuah roda: \omterm{def:g5:food-webs:roles}{produsen} membangun organik dari mineral
(\cref{ch:g6:plants-are-producers}); \omterm{def:g5:food-webs:roles}{konsumen} meneruskannya
(\cref{ch:g6:matter-from-food}); \omterm{def:g6:decomposers-and-soil:decomposers}{pengurai} mengembalikannya menjadi
mineral, siap bagi para \omterm{def:g5:food-webs:roles}{produsen} lagi. Materi tidak terpakai habis ---
ia berputar. Atom kertas halaman ini sudah pernah menjadi \omterm{def:g1:plants-around-us:parts}{daun}, \omterm{def:g6:decomposers-and-soil:soil}{humus},
\omterm{def:g1:plants-around-us:parts}{akar} dan \omterm{def:g1:plants-around-us:parts}{daun} lagi lebih sering daripada yang dapat dihitung.
\end{remark}

\section{Bekerja bersama para pengurai}

\begin{example}[Kompos, yang akhirnya dijelaskan]\label{ex:g6:decomposers-and-soil:compost}
Tempat kompos pada \cref{ex:g2:caring-for-nature:compost} sekarang
terbaca sebagai sebuah perkakas: sebuah bengkel \omterm{def:g6:decomposers-and-soil:decomposers}{pengurai} yang diisi
kulit sayuran. Cacing, kutu kayu dan awak yang tak terlihat membongkar
sampah organik dapurnya menjadi \omterm{def:g6:decomposers-and-soil:soil}{humus} yang gelap; tukang kebunnya
menebarkannya, dan mineralnya memberi makan petaknya
(\cref{rem:g4:what-plants-need:soil}). Tumpukannya bahkan terasa
hangat --- pembakaran para \omterm{def:g6:decomposers-and-soil:decomposers}{pengurainya} sendiri
(\cref{prop:g6:matter-from-food:losses}) memanaskan bengkelnya.
\end{example}

\begin{proposition}[Apa yang mempercepat dan memperlambat pekerjaannya]\label{prop:g6:decomposers-and-soil:speed}
\omterm{def:g6:decomposers-and-soil:decomposers}{Pengurai} adalah \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang punya \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}
(\cref{prop:g6:exploring-our-environment:decide}): mereka bekerja
paling cepat dalam keadaan \emph{hangat}, \emph{lembap} dan
\emph{berudara}, dan nyaris berhenti ketika dingin, kering, atau
tenggelam tanpa udara. Karena itulah hamparan musim dingin yang lambat
sesudah musim gugur dan pembersihan musim semi yang cepat; karena itu
pula resep kompos sang tukang kebun --- lembap, dibalik agar berudara,
tidak pernah berupa gumpalan tergenang yang tersegel.
\end{proposition}
\admitted

\begin{method}[Membaca kesehatan tanah]\label{met:g6:decomposers-and-soil:read}
Untuk menilai sebuah \omterm{def:g6:decomposers-and-soil:soil}{tanah} dengan cara naturalis:
\begin{enumerate}
  \item galilah sedalam satu telapak tangan lalu amati: \omterm{def:g6:decomposers-and-soil:soil}{humus} gelap
        yang gembur, atau debu pucat yang tak bernyawa?
  \item datalah segenggam di atas nampan putih --- cacing, kutu kayu,
        ekor pegas: awak yang kaya berarti daurnya sedang berputar;
  \item periksalah permukaannya: apakah materi yang mati lenyap
        sepanjang musim, atau menumpuk tanpa melapuk?
  \item perlakukan awaknya sebagai ternak: berilah ia makan dengan sisa
        yang dibenamkan, dengan kompos dan dengan mulsa; jagalah ia
        tetap lembap, dan jauhkanlah ia dari racun.
\end{enumerate}
\end{method}

\begin{example}[Cacing tanah, tukang kebunnya para tukang kebun]\label{ex:g6:decomposers-and-soil:worm}
Cacing \omterm{def:g6:decomposers-and-soil:soil}{tanah} bertugas rangkap: sebagai \omterm{def:g6:decomposers-and-soil:decomposers}{pengurai} --- menyeret \omterm{def:g1:plants-around-us:parts}{daun} ke
bawah, memakan materi yang melapuk --- dan sebagai \emph{pembajak},
liangnya memberi udara pada \omterm{def:g6:decomposers-and-soil:soil}{tanahnya} dan kotorannya mengulen mineral
serta organik menjadi satu. Satu padang rumput dapat menanggung berat
cacing di bawah rumputnya lebih banyak daripada berat sapi di atasnya;
naturalis zaman dahulu menyebut cacing sebagai pembuat lahan pertanian
yang sesungguhnya, dan mereka benar.
\end{example}

\section{Soal: Percobaan Kantong Daun}

\begin{problem}[{Soal akhir pekan --- kantong jaring, daun yang
dikubur, dan sebuah musim dingin penuh ilmu tanah}]\label{pb:g6:decomposers-and-soil:1}
Kelasnya mengubur kantong \omterm{def:g1:plants-around-us:parts}{daun} di tebing pagar tanaman sekolah pada
bulan November: \omterm{def:g1:plants-around-us:parts}{daun} pohon ek segar dengan berat yang setara di dalam
kantong jaring, yang akan digali lalu ditimbang ulang sepanjang tahun.
Ada tiga jenis kantong: jaring kasar (memasukkan cacing, kutu kayu,
segalanya), jaring halus (menolak awak yang terlihat, hanya
memasukkan yang tak terlihat), dan plastik tersegel (menolak segalanya
dan udaranya pula).

\medskip\noindent\textbf{Bagian I --- Menyiapkannya.}
\begin{enumerate}
  \item Mengapa semua kantongnya harus memuat berat yang \emph{sama}
        dari \omterm{def:g1:plants-around-us:parts}{daun} yang \emph{sama}, yang dikubur pada hari yang
        \emph{sama} di tebing yang \emph{sama}? Sebutkan metodenya.
  \item Pertanyaan apa yang diajukan kantong jaring halusnya, yang
        tidak dapat diajukan kantong jaring kasarnya sendirian?
  \item Sebuah regu menginginkan kantong keempat: jaring kasar, tetapi
        dikubur di dalam kerikil kering halaman sekolahnya. Pertanyaan
        apa yang akan diajukannya, menurut
        \cref{prop:g6:decomposers-and-soil:speed}?
  \item Ramalkan, sebelum menggali: urutkan ketiga kantong di tebing
        pagar tanamannya menurut berapa banyak berat \omterm{def:g1:plants-around-us:parts}{daun} yang akan
        hilang menjelang bulan Juni.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Hasilnya.}
\begin{enumerate}[resume]
  \item Penimbangan bulan Juni: jaring kasar --- beratnya paling
        banyak hilang, \omterm{def:g1:plants-around-us:parts}{daunnya} menyusut menjadi kerangka dan remah;
        jaring halus --- jelas meringan, \omterm{def:g1:plants-around-us:parts}{daunnya} melunak dan menghitam
        tetapi lebih lambat; plastik tersegel --- beratnya nyaris tidak
        berubah, isinya lumpur yang masam. Tafsirkan tiap kantongnya.
  \item Di dalam kantong kasarnya kelasnya menemukan kutu kayu, seekor
        cacing, ekor pegas dan benang \omterm{def:g6:decomposers-and-soil:decomposers}{jamur} putih. Berikan peran
        masing-masing dalam pembongkaran \omterm{def:g1:plants-around-us:parts}{daunnya}
        (\cref{ex:g6:decomposers-and-soil:leaf} adalah modelnya).
  \item Kantong halusnya kehilangan berat yang sungguhan walaupun
        setiap \omterm{def:g1:animals-around-us:animal}{hewan} yang terlihat ditolak masuk. Apa yang
        dibuktikannya tentang kehidupan \omterm{def:g6:decomposers-and-soil:soil}{tanah} yang tak terlihat?
  \item Kantong di halaman berkerikilnya (pertanyaan 3) kehilangan
        jauh lebih sedikit daripada kembarannya di tebing pagar
        tanamannya. \omterm{def:g6:exploring-our-environment:conditions}{Kondisi} kerja mana yang ditolak kerikilnya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Pelajaran yang lebih besar.}
\begin{enumerate}[resume]
  \item Ke mana \emph{perginya} berat yang hilang dari kantong
        kasarnya? Jawablah dengan
        \cref{prop:g6:decomposers-and-soil:decomposition}
        --- sedikitnya dua tujuan.
  \item Jelatang musim semi pada pagar tanamannya tumbuh paling hijau
        persis di sepanjang tebing tempat kelasnya mengubur
        kantongnya. Sebuah kebetulan? Berikan alasan dari daur
        materinya.
  \item Pakai kantong tersegelnya untuk menjelaskan mengapa sampah
        organik di tempat pembuangan akhir, yang terbungkus rapat
        tanpa udara di dalam plastik, bertahan berpuluh-puluh tahun
        sementara sampah yang sama menjadi kompos dalam hitungan
        bulan.
  \item Simpulkan dalam dua kalimat: apa yang dibuktikan musim
        dinginnya tentang siapa yang membersihkan hamparan hutannya,
        dan pembersihan itu memberi makan apa?
\end{enumerate}
\end{problem}
