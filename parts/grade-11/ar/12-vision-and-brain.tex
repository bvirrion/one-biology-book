\chapter{الرؤية والدماغ}\label{ch:g11:vision-and-brain}

بعد سكتة دماغية، يرى مريض العالم رؤية تامة --- بالأسود والأبيض
والرمادي. ويرى آخر الألوان والأشكال ولا يرى الحركة: فالسيارة هنا، ثم
هناك، ولا شيء بينهما، وصبّ الشاي مستحيل لأن الخيط لا يبدو متحركًا
أبدًا. وثالثة لا تعرف الوجوه، ومنها وجهها في المرآة، مع أنها تصف كل
ملمح. وعيونهم سليمة. وما فقده كل منهم قطعة من الدماغ تركَّب فيها
الرؤية. ويتابع هذا الفصل العصب البصري إلى داخل الدماغ، ويرسم خريطة ما
تفعله \omterm{prop:g11:vision-and-brain:pathway}{القشرة البصرية} بإشارات الشبكية، ويبين أن الخريطة ليست ثابتة:
فهي تبنى بالتجربة، ويغيرها التعلم، وتضطربها الأدوية.

\section{من الشبكية إلى القشرة}

\begin{proposition}[السبيل البصري]\label{prop:g11:vision-and-brain:pathway}
تسير ألياف كل عصب بصري المليون إلى الخلف حتى \emph{التصالب البصري}،
حيث تعبر ألياف النصف الأنفي من كل شبكية إلى الجانب الآخر بينما تبقى
ألياف النصف الصدغي في جانبها. ووراء التصالب، يحمل كل \emph{مسلك بصري}
إذن إشارات النصفين الأيسرين من الشبكيتين (المسلك الأيمن) أو النصفين
الأيمنين (المسلك الأيسر) --- أي كل ما يرى في نصف واحد من الحقل البصري.
ويتوقف المسلكان مرحليًا في \emph{المهاد} ويبلغان \emph{القشرة البصرية
الأولية}\index{قشرة بصرية}، في مؤخرة كل نصف كرة: فيتلقى نصف الكرة
الأيسر النصف الأيمن من الحقل البصري، ويتلقى الأيمن النصف الأيسر.
\end{proposition}
\begin{proof}[شواهد]
قطع عصب بصري واحد يعمي عينًا واحدة؛ أما تضرر مسلك بصري واحد، أو \omterm{prop:g11:vision-and-brain:pathway}{القشرة
البصرية} في نصف كرة واحد، فيترك العينين تعملان لكنه يعميهما معًا عن
النصف نفسه من العالم --- وهو النصف المقابل للآفة. وتنبيه نقطة من \omterm{prop:g11:vision-and-brain:pathway}{القشرة
البصرية} عند مريض مستيقظ أثناء الجراحة يجعله يرى ومضة في موضع ثابت من
الحقل؛ والنقاط المتجاورة تعطي ومضات متجاورة: فالقشرة تحمل خريطة
للشبكية، ويشغل مركز الحقل أكبر جزء منها.
\end{proof}

\section{مناطق كثيرة، وإدراك واحد}

\begin{proposition}[المناطق البصرية المتخصصة]\label{prop:g11:vision-and-brain:areas}
تمرر \omterm{prop:g11:vision-and-brain:pathway}{القشرة البصرية الأولية} تحليلها إلى نحو اثنتي عشرة \emph{منطقة
بصرية} أخرى، كل منها متخصصة: واحدة تستخرج اللون، وأخرى الحركة، وأخرى
الشكل والعمق والوجوه والكلمات المكتوبة. وتجمَّع نواتجها --- مع الذاكرة،
ومع الحواس الأخرى --- في الإدراك الواحد المتماسك الذي نعيشه. والإدراك
بناء: فالدماغ يملأ البقعة العمياء، ويتم الحدود المستترة، ويبقي الألوان
ثابتة تحت ضوء متغير، ويمكن أن تخدعه أوهام تستغل قواعده.
\end{proposition}
\begin{proof}[شواهد]
الآفات الموضعية تفصل المكونات: فالمريض المتضررة عنده منطقة اللون يرى
عالمًا رماديًا بحدة بصرية طبيعية؛ وتضرر منطقة الحركة يترك اللون والشكل
سليمين لكنه يلغي إدراك الحركة؛ وتضرر منطقة من الفص الصدغي يلغي التعرف
على الوجوه وحده. والتصوير الوظيفي عند متطوعين أصحاء يبين المناطق نفسها
تضيء، واحدةً للأنماط الملونة، وأخرى للنقاط المتحركة، وأخرى للوجوه،
بينما تستجيب القشرة الأولية لها كلها. والأقطاب في الحيوانات تجد، في كل
منطقة، عصبونات تستجيب لسمة تلك المنطقة ولا تكاد تستجيب لغيرها.
\end{proof}

\begin{example}[ثلاثة مرضى، وثلاث مناطق]\label{ex:g11:vision-and-brain:patients}
المريضة التي ترى بالرمادي عندها آفة في منطقة اللون على الجانبين؛
فمخاريط شبكيتها تعمل، وقشرتها الأولية تتلقى إشاراتها، لكن المرحلة التي
تحوّل نسب \omterm{def:g11:the-eye:photoreceptors}{المخاريط} إلى لون قد زالت. والمريضة التي لا ترى الحركة فقدت
منطقة الحركة: فهي ترى لقطات متتالية. والمريضة التي لا تعرف الوجوه
عندها آفة في منطقة الوجوه من الفص الصدغي؛ وهي تعرف الناس بالصوت أو
بقبعة. وفي كل حالة يكون المفقود دقيقًا، ويبين الباقي أن المناطق الأخرى
تعمل وحدها.
\end{example}

\section{قشرة تبنيها التجربة}

\begin{proposition}[اللدونة]\label{prop:g11:vision-and-brain:plasticity}
وصلات \omterm{prop:g11:vision-and-brain:pathway}{القشرة البصرية} ليست ثابتة عند الولادة. ففي \emph{فترة حرجة} من
الحياة الباكرة تشكّلها ما تراه العينان؛ وهي في البالغ تظل تتغير، ببطء
أشد، بالتعلم وبالاستعمال. وهذه \emph{اللدونة
الدماغية}\index{لدونة دماغية} خاصة من خصائص العصبونات: فالوصلات
المستعملة تقوى وتتكاثر، والوصلات غير المستعملة تضعف وتشذَّب.
\end{proposition}
\begin{proof}[شواهد]
سجّل هيوبل وويزل (في الستينيات) من \omterm{prop:g11:vision-and-brain:pathway}{القشرة البصرية} عند القطط والقردة.
ففي الحيوان الطبيعي تستجيب معظم العصبونات للعينين معًا. أما في حيوان
أبقيت إحدى عينيه مغلقة في أشهره الأولى، فكل عصبون تقريبًا لا يستجيب
إلا للعين المفتوحة: فوصلات العين المغلقة قد استولي عليها. وإغلاق عين
بالغ المدة نفسها لا يغير شيئًا. وعند الإنسان، الطفل الذي تكون إحدى
عينيه شديدة الخروج عن البؤرة أو منحرفة ولا تصحح قبل نحو السابعة تبقى
عنده عين ضعيفة دائمًا --- وهو الغمش --- مهما حسنت بصرياتها لاحقًا. وفي
البالغين، تعلّم قراءة برايل يكبّر المساحة القشرية المخصصة للإصبع
القارئة؛ وتعلّم الشعوذة يكبّر المناطق المعالجة للحركة، وتصغر من جديد
إذا توقف التمرين.
\end{proof}

\begin{example}[لماذا يجب علاج الحول باكرًا]\label{ex:g11:vision-and-brain:amblyopia}
طفل في الثالثة عينه مائلة إلى الداخل يرى مزدوجًا، فيكبت الدماغ الصورة
الأضعف. فإذا لم تقوَّم العين، أو لم تغطَّ العين السليمة بضع ساعات في
اليوم لإرغام الضعيفة على العمل، أعادت القشرة تنظيم نفسها حول العين
السليمة أثناء الفترة الحرجة وبقيت العين الضعيفة، وإن كانت تامة
بصريًا، عمياء عن التفاصيل الدقيقة مدى الحياة. فإذا عولج في الثالثة
استعاد رؤية تامة؛ وإذا عولج في الثانية عشرة فلا يكاد يستعيد شيئًا.
\end{example}

\section{الأدوية والدماغ الرائي}

\begin{proposition}[العصبونات والرسل والجزيئات التي تحاكيها]\label{prop:g11:vision-and-brain:drugs}
تتواصل العصبونات عند \emph{المشابك} بإطلاق رسل كيميائية، هي
\emph{النواقل العصبية}، على مستقبلات \omterm{def:g10:cells-common-unit:cell}{الخلية} التالية؛ وتصف السنة
الأخيرة الآلية. وعدة نواقل عصبية، منها \emph{السيروتونين}، تنظم جريان
الإشارات عبر المناطق البصرية. والجزيئات التي يشبه شكلها ناقلًا عصبيًا
تستطيع الارتباط بمستقبلاته وإرباك ذلك الجريان: فثنائي إيثيل أميد حمض
الليسرجيك (LSD) يرتبط بمستقبلات السيروتونين في \omterm{prop:g11:vision-and-brain:pathway}{القشرة البصرية} وينتج
هلوسات --- ألوانًا وأشكالًا وحركات ترى بلا ضوء موافق. ومثل هذه الأدوية
قد تخلف تغيرات باقية: نوبات تتكرر بعد أشهر، واضطرابات بصرية مستمرة عند
بعض الناس.
\end{proposition}
\admitted

\begin{example}[ما الذي تبينه هلوسة]\label{ex:g11:vision-and-brain:hallucination}
تحت LSD تبقى الشبكية على حالها وتظل القشرة الأولية ترسم ما تراه
العينان؛ والاضطراب في المناطق اللاحقة وفي طريقة جمعها. وتظهر أنماط
تشبه لبنات بناء القشرة نفسها --- شبكات ولوالب وأنفاق --- لأن الدواء
يشغّل آلة الإدراك بلا دخلها. فالهلوسة برهان، بالعطب، على أن الرؤية
يصنعها الدماغ وأن جزيئًا يستطيع تغييرها.
\end{example}

\begin{method}[تعيين موضع عجز بصري]\label{met:g11:vision-and-brain:locate}
\begin{enumerate}
  \item عين واحدة عمياء والأخرى طبيعية: فالعين أو عصبها البصري، قبل
        التصالب.
  \item العينان عمياوان عن النصف نفسه من الحقل: فالمسلك البصري المقابل
        أو المهاد أو القشرة الأولية.
  \item العينان عمياوان عن النصفين الخارجيين من الحقل: فالتصالب، حيث
        تقطع الألياف العابرة.
  \item الرؤية سليمة لكن صفة واحدة مفقودة (لون أو حركة أو وجوه):
        فالمنطقة المتخصصة الموافقة.
  \item الإدراك مضطرب مع عينين وقشرة طبيعيتين، واضطراب عابر: فدواء أو
        سبب استقلابي.
\end{enumerate}
\end{method}

\begin{remark}[ما بينته هذه السنة]\label{rem:g11:vision-and-brain:year}
كانت عين \cref{ch:g11:the-eye} مسألة \omterm{def:g10:universal-dna:gene}{مورّثات} وبروتينات: أوبسينات،
ومضاعفاتها، وعمى ألوان يورث على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} X. أما دماغ هذا الفصل فمسألة
وصلات يشكّلها الاستعمال. وبينهما يقع موضوع السنة كله --- من متتالية
\omterm{def:g10:universal-dna:gene}{مورّثة} إلى صفة، ومن الصفة إلى ما يستطيع الشخص فعله --- ودرسُ أن \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط
الوراثي} يقترح في كل خطوة والمحيط، من حرارة أذني قط إلى الضوء الذي
تتلقاه عين هريرة، يتصرف.
\end{remark}

\section{مسألة: أربعة مرضى وهريرة}

\begin{problem}[{مسألة نهاية الأسبوع --- عجوز بصرية تعيَّن مواضعها على
طول السبيل، وقشرة ترسم خريطتها بالتنبيه، وحرمان هريرة يعد، وحول طفل
يعالج في وقته}]
\label{pb:g11:vision-and-brain:1}
يفحص أربعة مرضى. A: أعمى في العين اليسرى، واليمنى طبيعية. B: العينان
عمياوان عن النصف الأيسر من الحقل. C: العينان عمياوان عن النصف الخارجي
من حقلهما (اليسرى إلى اليسار، واليمنى إلى اليمين). D: حقلان طبيعيان،
وحدة بصرية طبيعية، لكنه لا يدرك الحركة.

\medskip
\noindent\textbf{الجزء الأول --- التعيين.}
\begin{enumerate}
  \item عيّن موضع آفة A. أي بنية، وقبل التصالب أم بعده؟
  \item عيّن موضع آفة B، وقل لماذا يجب أن تقع بعد التصالب.
  \item عيّن موضع آفة C بدقة، مفسّرًا أي الألياف قطعت.
  \item عيّن موضع آفة D. وما الذي يخبرك به حفظ الحقلين والحدة البصرية
        عن القشرة الأولية؟
  \item لكل مريض، قل هل الشبكية سليمة، وكيف تتحقق من ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الخريطة.}
أثناء الجراحة، ينتج تنبيه نقاط من \omterm{prop:g11:vision-and-brain:pathway}{القشرة البصرية} اليمنى عند مريض
ومضاتٍ. فتنبيه طرف الفص القذالي يعطي ومضة في مركز الحقل؛ والنقاط
الأبعد إلى الأمام تعطي ومضات أبعد في الحقل الأيسر؛ و\qty{1}{cm} من
القشرة عند الطرف يغطي نحو $2^\circ$ من الحقل، بينما يغطي \qty{1}{cm}
أبعد إلى الأمام $20^\circ$.
\begin{enumerate}[resume]
  \item لماذا تكون الومضات كلها في الحقل الأيسر؟
  \item كم من القشرة يمثل الدرجتين $2^\circ$ المركزيتين قياسًا إلى شريط
        $2^\circ$ على بعد $40^\circ$ من المركز؟ احسب النسبة.
  \item اربط هذه النسبة بنقرة الشبكية وأطرافها.
  \item آفة صغيرة عند طرف الفص القذالي الأيمن: فماذا يفقد المريض؟
        وآفة بالحجم نفسه أبعد إلى الأمام؟
  \item لماذا يستطيع المريض ذو الآفة المركزية الصغيرة أن يقرأ مع ذلك،
        ببطء، بتحريك عينيه؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الهريرة.}
من شكل هيوبل وويزل.
\begin{enumerate}[resume]
  \item في الهريرة الطبيعية، ما النسبة المئوية للعصبونات التي تستجيب
        للعينين معًا بدرجة ما؟
  \item وفي الهريرة المحرومة، ما النسبة التي تستجيب للعين المغلقة
        (اليسرى) أصلًا؟
  \item شبكية العين المغلقة وعصبها البصري سليمان. فأين ذهب دخلهما، وما
        الذي أخذ المكان؟
  \item الإغلاق نفسه عند قط بالغ لا يغير شيئًا. فما الذي يرسيه ذلك، وما
        نظيره عند الإنسان؟
  \item هريرة تغلق عيناها معًا ثلاثة أشهر. تنبأ بالمدرج وبرؤية الحيوان
        بعد ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الطفل.}
طفل في الرابعة عينه اليسرى مائلة إلى الداخل؛ وبصرياتها طبيعية. وإن لم
يعالج استولت القشرة على دخل العين اليمنى.
\begin{enumerate}[resume]
  \item فسّر، بالهريرة، ما الذي سيحدث لرؤية العين اليسرى إن لم يفعل
        شيء، ولماذا لا تمنعه النظارات وحدها.
  \item العلاج أن تغطى العين \emph{اليمنى} عدة ساعات في اليوم أشهرًا.
        فسّر لماذا تغطى العين السليمة.
  \item لماذا يجب فعل ذلك قبل نحو السابعة، ولماذا يخفق العلاج نفسه في
        الخامسة عشرة؟
  \item البالغ الذي فقد عينًا في الثلاثين لا يفقد القشرة المخصصة لها
        على النحو نفسه. وفّق بين ذلك وبين الهريرة.
  \item اذكر النتيجة: الآفات الأربع معيَّنة من مقدمة السبيل إلى مؤخرته،
        وخاصة القشرة الواحدة --- مسمّاةً --- التي تجعل علاج الطفل لا
        ينجح إلا في وقته.
\end{enumerate}
\end{problem}
