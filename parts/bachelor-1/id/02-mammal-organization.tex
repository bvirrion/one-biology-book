\chapter{Organisasi Fungsional Seekor Mamalia}\label{ch:b1:mammal-organization}

Bukalah seekor kelinci di atas meja bedah dan kesan pertamanya adalah
keteraturan. Sehelai lembar berotot, \omterm{def:b1:mammal-organization:bodyplan}{diafragma}, membagi batang tubuh
menjadi rongga dada yang memuat jantung dan dua paru serta rongga perut
yang memuat selebihnya: hati sebesar kepalan, lambung, empat meter usus
yang berakhir pada sekum sebesar lambung, dua ginjal yang menempel pada
dinding belakang. Tiap \omterm{def:b1:mammal-organization:organ}{organ} punya tempat, punya pasokan darah, dan
punya tugas; tak satu pun bekerja sendirian. Bab ini memerikan bagaimana
mamalia dibangun sebagai satu keseluruhan yang bekerja: \omterm{def:b1:mammal-organization:bodyplan}{pola tubuhnya},
sistem tempat \omterm{def:b1:mammal-organization:organ}{organnya} dikelompokkan, ruang cairan tempat selnya hidup,
\omterm{prop:b1:organism-environment:surfaces}{permukaan pertukaran} tempat ia menemui dunia luar, dan \omterm{prop:b1:mammal-organization:heatbudget}{neraca kalor} yang
menahannya pada \qty{39}{\celsius} di padang bulan Januari.

\section{Pola tubuh seekor mamalia}

\begin{definition}[Pola tubuh]\label{def:b1:mammal-organization:bodyplan}
Yang disebut \emph{pola tubuh}\index{pola tubuh} seekor hewan adalah
susunan sumbu, rongga dan \omterm{def:b1:mammal-organization:organ}{organnya} yang dimiliki bersama oleh semua
anggota kelompoknya. Pola tubuh mamalia adalah pola tubuh vertebrata:
simetri bilateral dengan ujung kepala yang membawa otak dan indra utama
(\emph{sefalisasi}\index{sefalisasi}), sebuah sumbu punggung berupa
ruas tulang belakang yang menyelubungi sumsum tulang belakang, empat
tungkai, dan sebuah rongga tubuh di sisi perut --- \emph{selom}
\index{selom} --- yang memuat \omterm{def:b1:mammal-organization:organ}{organ} dalam. Yang khas mamalia: selomnya
dibagi oleh \emph{diafragma}\index{diafragma} berotot menjadi rongga
dada (jantung, paru) dan rongga perut (\omterm{def:b1:mammal-organization:organ}{organ} pencernaan, ginjal,
gonad); tubuhnya berambut; anaknya diberi susu; suhunya diatur.
\end{definition}

\begin{definition}[Organ, sistem organ]\label{def:b1:mammal-organization:organ}
Yang disebut \emph{organ}\index{organ} adalah sebuah struktur yang
tersusun dari beberapa jaringan (\cref{ch:b1:body-plans-tissues}) yang
tertata untuk menjalankan satu fungsi tertentu: jantung memompa, paru
mempertukarkan gas, ginjal menyaring. Yang disebut \emph{sistem
organ}\index{sistem organ} adalah sehimpunan organ yang bekerja sama
dalam satu fungsi luas: sistem pencernaan membentang dari mulut sampai
anus dengan hati dan pankreas yang melekat padanya; sistem peredaran
adalah jantung beserta pembuluhnya; sistem ekskresi, ginjal dan saluran
kemih.
\end{definition}

\section{Sistem organ dan fungsi yang dilayaninya}

\begin{proposition}[Tiga kelompok fungsi]\label{prop:b1:mammal-organization:functions}
\omterm{def:b1:mammal-organization:organ}{Sistem organ} seekor mamalia melayani tiga kelompok fungsi:
\begin{itemize}
  \item \emph{nutrisi}\index{fungsi nutrisi}: memasukkan lalu
        mengedarkan materi dan energi serta membuang sisa --- sistem
        pencernaan, pernapasan, peredaran dan ekskresi;
  \item \emph{relasi}\index{fungsi relasi}: mengindera \omterm{def:b1:organism-environment:organism}{lingkungan} dan
        bertindak atasnya --- indra, sistem saraf, kelenjar endokrin,
        rangka dan otot;
  \item \emph{reproduksi}\index{fungsi reproduksi}: menghasilkan
        keturunan --- gonad, saluran kelamin dan, pada mamalia,
        kelenjar susu.
\end{itemize}
Tak ada sistem yang bekerja sendirian: tiap \omterm{def:b1:mammal-organization:organ}{organ} bergantung pada darah
untuk pasokannya, pada sistem saraf dan endokrin untuk pemaduannya, dan
pada rangka serta kulit untuk perlindungannya.
\end{proposition}

\begin{example}[Sesuap rumput]\label{ex:b1:mammal-organization:mouthful}
Rumput si kelinci dipotong gigi seri dan digiling gigi geraham (rangka
dan otot), dikecap dan dicium (indra), ditelan lalu dicerna di lambung
dan usus halus; gula dan asam aminonya melintasi dinding usus masuk ke
darah dan sampai ke hati, yang menyimpan atau melepasnya di bawah
kendali hormon (endokrin); selulosanya difermentasi bakteri di dalam
sekum; oksigen untuk membakar gulanya datang lewat paru, karbon
dioksidanya pergi lewat jalan yang sama, urea dari asam aminonya pergi
lewat ginjal, dan kalor seluruh proses itu pergi lewat kulit. Tujuh
sistem untuk satu suapan.
\end{example}

\section{Ruang-ruang cairan tubuh}

\begin{definition}[Ruang cairan]\label{def:b1:mammal-organization:compartments}
Air menyusun sekitar \qty{60}{\%} massa seekor mamalia. Dua pertiganya
adalah \emph{cairan intrasel}\index{cairan intrasel}, di dalam sel;
sepertiganya adalah \emph{cairan ekstrasel}\index{cairan ekstrasel},
yakni \omterm{def:b1:organism-environment:milieu}{lingkungan internal} \cref{ch:b1:organism-environment}, yang
sendiri terbagi menjadi \emph{cairan antarsel}\index{cairan antarsel}
yang merendam sel (tiga perempatnya) dan \emph{plasma
darah}\index{plasma} di dalam pembuluh (seperempatnya). \emph{Getah
bening}\index{getah bening} adalah cairan antarsel yang dikumpulkan
pembuluh limfa dan dikembalikan ke darah. Kedua cairan ekstrasel itu
hampir sama susunannya --- plasma berbeda pada proteinnya, yang tidak
melintasi dinding kapiler --- dan keduanya berbeda tajam dari cairan
intrasel.
\end{definition}

\begin{method}[Mengukur sebuah ruang cairan dengan pengenceran]\label{met:b1:mammal-organization:dilution}
\begin{enumerate}
  \item Pilihlah sebuah penanda yang tersebar di dalam ruang itu dan
        tidak di tempat lain: sebuah zat warna yang terikat pada protein
        plasma (biru Evans) untuk plasma; inulin, gula yang keluar dari
        pembuluh tetapi tidak masuk ke sel, untuk \omterm{def:b1:mammal-organization:compartments}{cairan ekstrasel}; air
        berat untuk total air tubuh.
  \item Suntikkan sejumlah $m$ yang diketahui dan tunggulah sampai
        bercampur (beberapa menit untuk plasma, beberapa jam untuk total
        airnya).
  \item Ukurlah kepekatan $c$ pada satu cuplikan plasma, dengan koreksi
        bagi kehilangan apa pun (kemih, metabolisme) selama menunggu.
  \item Volumenya adalah $V = m/c$. Volume intrasel adalah total air
        dikurangi ekstrasel; volume antarsel adalah ekstrasel dikurangi
        plasma. Volume darah adalah volume plasma dibagi fraksi plasma
        dalam darah ($1 - $ hematokrit).
\end{enumerate}
\end{method}

\begin{example}[Sebuah volume plasma]\label{ex:b1:mammal-organization:evansblue}
\qty{50}{mg} biru Evans yang disuntikkan ke seorang pria \qty{70}{kg}
memberikan, sepuluh menit kemudian, kepekatan plasma \qty{16.7}{mg/L}:
$V = 50/16.7 = \qty{3.0}{L}$ plasma. Dengan hematokrit \qty{45}{\%},
volume darahnya adalah $3.0/0.55 = \qty{5.5}{L}$ --- \qty{8}{\%} massa
tubuh, sebuah nisbah yang berlaku dari mencit sampai paus.
\end{example}

\section{Permukaan pertukaran dan peredaran darah}

\begin{proposition}[Tiap permukaan pertukaran terhubung ke darah]\label{prop:b1:mammal-organization:surfaces}
Seekor mamalia bertukar dengan dunia luar lewat empat permukaan, yang
masing-masing terlipat di dalam tubuh dan masing-masing dialiri jaringan
kapiler yang rapat: mukosa usus (\qty{30}{m^2} pada manusia) untuk zat
hara dan air, permukaan alveolus paru (\qty{100}{m^2}) untuk oksigen dan
karbon dioksida, permukaan penyaring ginjal (sejuta glomerulus, yang
menyaring \qty{180}{L} plasma sehari) untuk sisa dan bagi pengaturan
\omterm{def:b1:organism-environment:milieu}{lingkungan internal}, serta kulit (\qty{2}{m^2}) untuk kalor dan sedikit
air. Peredaran darah menghubungkan keempatnya: darah yang meninggalkan
satu permukaan sampai ke permukaan berikutnya dalam satu menit, sehingga
apa yang masuk lewat usus dioksidasi dengan oksigen dari paru dan
sisanya pergi lewat ginjal.
\end{proposition}

\begin{example}[Waktu peredaran]\label{ex:b1:mammal-organization:circtime}
Dengan \qty{5.5}{L} darah dan curah jantung \qty{5}{L/min}, setetes

berbaring, dan dalam seperempatnya selama olahraga berat, ketika

\end{example}

\section{Endotermi dan neraca kalor}

\begin{proposition}[Neraca kalor]\label{prop:b1:mammal-organization:heatbudget}
\index{neraca kalor}
Seekor mamalia yang bersuhu mantap memenuhi
\[
M - W = R + C + K + E ,
\]
dengan $M$ produksi kalor metaboliknya, $W$ kerja mekanis yang
dilakukannya ke luar, dan ruas kanan berupa kehilangan lewat pancaran
$R$, konveksi ke udara $C$, konduksi ke tanah $K$ dan penguapan $E$
(dari paru, dan dari keringat atau napas terengah). Ketiga kehilangan
pertama sebanding dengan $T_b - T_a$ dan diatur oleh penyekatan bulu
serta aliran darah ke kulit; penguapan adalah satu-satunya jalan yang
masih bekerja ketika udara lebih hangat daripada tubuh. Ketika
persamaan itu tidak terpenuhi, suhu tubuh melayang pada laju yang
ditentukan ketimpangannya dan kapasitas kalor tubuh (sekitar
\qty{3.5}{kJ.kg^{-1}.K^{-1}}).
\end{proposition}
\begin{proof}
Kekekalan energi bagi tubuh selama satu selang ketika suhunya tidak
berubah: energi kimia yang dilepaskan ($M$) pergi sebagai kerja atau
sebagai kalor lewat empat saluran fisis itu. Kesebandingan $R$, $C$, $K$
dengan selisih suhu adalah hukum pendinginan Newton
(\cref{ch:b1:organism-environment}); ketidakbergantungan $E$ pada
selisih itu mengikuti kenyataan bahwa penguapan didorong gradien
kelembapan, bukan gradien suhu.
\end{proof}

keempat kehilangan ditambah kerja ke luar; hewan itu menyetel produksi

kulit) dan penguapan untuk menjaga keseimbangannya.}

\begin{proposition}[Sarana pengaturan suhu]\label{prop:b1:mammal-organization:thermoreg}
Melawan dingin, seekor mamalia mengurangi kehilangan --- menegakkan
bulunya, menyempitkan pembuluh kulit dan ujung tubuhnya, meringkuk,
berdesakan --- dan menaikkan produksi: menggigil (kerutan berirama otot)
dan, pada mamalia kecil dan muda, \emph{termogenesis tanpa
menggigil}\index{termogenesis tanpa menggigil} di dalam \emph{jaringan
lemak cokelat}\index{jaringan lemak cokelat}, yang mitokondrianya
melepas energi lemak langsung sebagai kalor. Melawan panas, ia
melebarkan pembuluh kulit (telinga kelinci dapat membuang sepertiga
kalornya), dan menguapkan air dengan berkeringat (kuda, manusia) atau
terengah-engah (anjing, kelinci). Semuanya diperintahkan dari
hipotalamus, yang memegang \omterm{def:b1:organism-environment:homeostasis}{titik acuan} dan membaca suhu darah serta
suhu kulit.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Memanaskan hipotalamus seekor anjing dengan sebuah probe tertanam
membuatnya terengah-engah dan melebarkan pembuluh kulitnya di dalam
ruang dingin; mendinginkan probe itu membuatnya menggigil di dalam ruang
hangat. Konsumsi oksigen seekor tikus yang ditempatkan pada
\qty{5}{\celsius} bertiga lipat dalam satu jam, separuh kenaikannya dari
menggigil (terekam sebagai kegiatan listrik otot) dan separuh dari lemak
cokelat, yang suhunya, diukur dengan termokopel, melebihi suhu jaringan
di sekitarnya. Citra inframerah seekor kelinci pada \qty{30}{\celsius}
memperlihatkan telinganya hampir bersuhu tubuh, dan pada
\qty{5}{\celsius} hampir bersuhu udara.
\end{proof}

\begin{example}[Telinga si kelinci]\label{ex:b1:mammal-organization:ears}
Dua telinga masing-masing sekitar \qty{50}{cm^2}, tipis, hampir
gundul dan kaya pembuluh: dengan pembuluhnya terbuka pada
\qty{30}{\celsius} dalam udara \qty{20}{\celsius}, dengan koefisien
kulit gundul \qty{10}{W.m^{-2}.K^{-1}}, keduanya melepas
$10\times 0.01\times 10 =
\qty{1}{W}$, seperenam produksi kelinci yang berbaring. Pada
\qty{5}{\celsius} pembuluhnya menutup, telinganya mendingin sampai
hampir bersuhu udara, dan kehilangan lewat telinga turun di bawah
\qty{0.2}{W}.
\end{example}

\begin{remark}[Pemaduan]\label{rem:b1:mammal-organization:coordination}
Setiap penyetelan di atas --- pembuluh yang menyempit, otot yang
menggigil, kelenjar yang menyekresi --- adalah sebuah perintah yang
dibawa saraf atau hormon. Sistem saraf bertindak dalam milisekon
sepanjang kawat yang tetap; sistem endokrin bertindak dalam menit sampai
jam lewat darah, mencapai tiap sel yang membawa reseptor yang tepat.
Mekanismenya menjadi pokok bahasan jilid Tahun ke-2; tahun ini
menerimanya sebagai diberikan dan menelaah apa yang dipadukannya.
\end{remark}

dan sebutkan satu \omterm{def:b1:mammal-organization:organ}{organ} tiap sistemnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:mammal-organization:3}
Seorang perempuan \qty{60}{kg}: perkirakan total air tubuhnya, volume

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:mammal-organization:4}
Dari gambar aliran darah, berapa fraksi curah jantung yang menuju
ginjal, dan berapa kali sehari seluruh volume darah (\qty{5.5}{L})

\begin{exercise}[$\star\star$]\label{exo:b1:mammal-organization:5}
\qty{2.0}{g} inulin disuntikkan ke seorang pria \qty{70}{kg}; setelah
bercampur, dan setelah \qty{0.3}{g} hilang dalam kemih, kepekatan
plasmanya \qty{0.12}{g/L}. Hitunglah volume ekstraselnya, lalu, dengan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:mammal-organization:6}
Jelaskan mengapa plasma dan \omterm{def:b1:mammal-organization:compartments}{cairan antarsel} memiliki susunan ion yang

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:mammal-organization:7}
Mengapa \omterm{def:b1:mammal-organization:organ}{organ} peredaran sistemik terletak paralel dan bukan seri? Apa

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:mammal-organization:8}
Seorang manusia yang berbaring ($M = \qty{80}{W}$, $W = 0$) dalam udara
\qty{20}{\celsius} kehilangan \qty{20}{W} lewat penguapan dari paru dan
kulit. Bila pancaran, konveksi dan konduksi bersama-sama sama dengan
$hS(T_b - T_a)$ dengan $S = \qty{1.8}{m^2}$ dan $T_b$ (kulit) $=

ketika udara mencapai \qty{33}{\celsius}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:mammal-organization:9}
Seorang manusia yang berolahraga menghasilkan \qty{800}{W} dan

\begin{exercise}[$\star\star\star$]\label{exo:b1:mammal-organization:10}

\begin{exercise}[$\star\star\star$]\label{exo:b1:mammal-organization:11}

\section{Soal: Sehari Seekor Kelinci}

\begin{problem}[{Soal akhir pekan --- dua puluh empat jam seekor kelinci
dua kilogram pada bulan Januari: energinya, airnya, darahnya dan
kalornya, berakhir pada pengeluaran energi hariannya dan perputaran
airnya}]
\label{pb:b1:mammal-organization:1}
Seekor kelinci liar bermassa \qty{2.0}{kg} dan bersuhu tubuh
\qty{39}{\celsius}. \omterm{def:b1:organism-environment:allometry}{Laju metabolisme basalnya} mengikuti \omterm{thm:b1:organism-environment:kleiber}{hukum Kleiber},
$B = \qty{3.4}{W}\,(M/\qty{1}{kg})^{3/4}$. Harinya: \qty{12}{h}
berbaring di liangnya di dalam \omterm{prop:b1:organism-environment:thermo}{zona termonetralnya}, \qty{6}{h} mencari
makan pada tiga kali laju basalnya, dan \qty{6}{h} berbaring di luar
pada \qty{5}{\celsius}, tempat ia harus menghasilkan dua kali laju
basalnya. Ia memakan rumput segar berkadar bahan kering \qty{25}{\%};
bahan keringnya mengandung \qty{18}{kJ/g}, yang \qty{65}{\%}-nya
dicernanya.

\medskip
\noindent\textbf{Bagian I --- Energi.}
\begin{enumerate}
  \item Hitunglah \omterm{def:b1:organism-environment:allometry}{laju metabolisme basal} si kelinci dalam watt.
  \item Hitunglah energi yang dikeluarkan pada tiap-tiap ketiga selang
        itu, dalam kilojoule.
  \item Hitunglah pengeluaran energi hariannya dan nyatakan sebagai
        kelipatan laju basal selama \qty{24}{h}.
  \item Hitunglah energi tercerna satu gram rumput segar.
  \item Hitunglah massa rumput segar yang dimakan tiap hari, dan massa
        bahan kering yang dikandungnya.
  \item Nyatakan asupan segarnya sebagai fraksi massa tubuh.
  \item Dengan \qty{20}{kJ} tiap liter oksigen yang dipakai, hitunglah
        konsumsi oksigen hariannya dan laju rata-ratanya dalam
        mililiter tiap menit.
  \item Dengan kuosien respirasi $0.9$ (liter $\mathrm{CO_2}$ tiap liter
        $\mathrm{O_2}$), hitunglah keluaran karbon dioksida hariannya
        dalam liter dan dalam gram (\qty{22.4}{L/mol},
        \qty{44}{g/mol}).
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Air.}
Mengoksidasi bahan kering yang tercerna menghasilkan \qty{0.6}{g} air
metabolik tiap gram. Bahan kering yang tak tercerna pergi dalam tinja
yang mengandung \qty{40}{\%} air. Kemihnya \qty{130}{mL} tiap hari. Si
kelinci menghirup \qty{860}{L} udara sehari; ia menghirup udara
bersuhu \qty{10}{\celsius} yang membawa \qty{5}{g/m^3} uap air dan
mengembuskan udara jenuh pada \qty{37}{\celsius}, \qty{44}{g/m^3}.
Penguapan lewat kulit \qty{20}{g} sehari.
\begin{enumerate}[resume]
  \item Hitunglah air yang masuk bersama rumputnya.
  \item Hitunglah air metabolik yang dihasilkan.
  \item Hitunglah massa tinjanya dan air yang dibawanya pergi.
  \item Hitunglah air yang hilang lewat napas.
  \item Susunlah neraca airnya: total pemasukan, total pengeluaran, dan
        selisihnya.
  \item Haruskah si kelinci minum? Apa yang terjadi pada selisihnya?
  \item Pada bulan Agustus rumputnya \qty{50}{\%} bahan kering.
        Hitung ulang air dalam makanannya untuk asupan energi yang
        sama, lalu simpulkan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Darah dan ruang cairan.}
\begin{enumerate}[resume]
  \item Hitunglah total air tubuh si kelinci, volume ekstraselnya dan
        volume plasmanya, dengan memakai fraksi pada bab ini.
  \item \qty{1.5}{mg} biru Evans disuntikkan; kepekatan plasma setelah
        bercampur adalah \qty{15}{mg/L}. Hitunglah volume plasmanya dan
        bandingkan dengan perkiraan tadi.
  \item Dengan hematokrit \qty{40}{\%}, hitunglah volume darahnya.
  \item Curah jantung berskala sebagai $M^{3/4}$ dari nilai manusia
        \qty{5}{L/min} pada \qty{70}{kg}. Hitunglah curah jantung si
        kelinci dan waktu yang dibutuhkan setetes darah untuk
        menyelesaikan peredarannya.
  \item Dengan laju denyut \qty{200}{} kali per menit, hitunglah volume
        sekuncupnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Kalor.}
Permukaan si kelinci adalah $S = \qty{0.1}{m^2}\,(M/\qty{1}{kg})^{2/3}$;
bulunya memberi $h = \qty{2.5}{W.m^{-2}.K^{-1}}$; tiap telinga seluas
\qty{50}{cm^2} kulit yang hampir gundul dengan
$h = \qty{10}{W.m^{-2}.K^{-1}}$.
\begin{enumerate}[resume]
  \item Hitunglah luas permukaan si kelinci.
  \item Hitunglah kalor yang hilang lewat bulunya pada
        \qty{5}{\celsius} dan bandingkan dengan produksi yang
        diandaikan bagi selang di luar liang.
  \item Hitunglah kehilangan lewat kedua telinganya bila pembuluhnya
        terbuka (kulit pada \qty{39}{\celsius}) dan bila menutup (kulit
        pada \qty{8}{\celsius}). Apa yang dilakukan si kelinci pada
        bulan Januari?
  \item Pada bulan Agustus, pada \qty{30}{\celsius}, hitunglah
        kehilangan lewat bulunya dan lewat telinganya yang terbuka,
        lalu bandingkan jumlahnya dengan produksi basalnya. Bagaimana
        si kelinci tetap berada pada
        \qty{39}{\celsius}?
  \item Nyatakan hasilnya: pengeluaran energi harian si kelinci dalam
        megajoule, sebagai kelipatan laju basalnya, dan perputaran air
        hariannya dalam gram serta sebagai fraksi massanya.
\end{enumerate}
\end{problem}
