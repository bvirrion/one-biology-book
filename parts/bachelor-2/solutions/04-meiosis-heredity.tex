\section*{Chapter \ref{ch:b2:meiosis-heredity} --- Meiosis, Genetic Mixing and Heredity}

\begin{solution}{exo:b2:meiosis-heredity:1}
Mitosis: one division, no pairing of homologues, two cells, each
genetically identical to the parent ($2n \to 2n$). \omterm{def:b2:meiosis-heredity:meiosis}{Meiosis}: two
divisions after one replication, homologues pair and cross over,
four cells, \omterm{def:b2:meiosis-heredity:meiosis}{haploid} and all genetically different ($2n \to n$).
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:2}
$2^{23} = 8.4\times 10^{6}$; $2^{4} = 16$; $2^{7} = 128$ --- before
crossing-over multiplies each of these without limit.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:3}
$Tt\times Tt$: \omterm{def:b2:meiosis-heredity:vocabulary}{genotypes} $1\,TT : 2\,Tt : 1\,tt$, \omterm{def:b2:meiosis-heredity:vocabulary}{phenotypes} $3$ tall
: $1$ dwarf. $Tt\times tt$: $1\,Tt : 1\,tt$, $1 : 1$. Test-cross the
tall plant to a dwarf: all tall means $TT$, half dwarf means $Tt$
(or self it: $TT$ breeds true, $Tt$ segregates $3 : 1$).
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:4}
Linked genes lie on the same chromosome and their parental \omterm{def:b2:meiosis-heredity:vocabulary}{allele}
combinations are over-represented among gametes; the \omterm{def:b2:meiosis-heredity:linkage}{recombination
frequency} is the fraction of recombinant gametes (recombinant
offspring of a test cross); one \omterm{def:b2:meiosis-heredity:linkage}{centimorgan} is one percent
recombination. Genes far apart on a long chromosome have at least
one crossover between them in nearly every \omterm{def:b2:meiosis-heredity:meiosis}{meiosis}, and multiple
crossovers randomise the combinations, so $r$ reaches $\tfrac12$ ---
indistinguishable from \omterm{thm:b2:meiosis-heredity:mixing}{independent assortment}.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:5}
Total 556; expected $312.75 : 104.25 : 104.25 : 34.75$; $\chi^2 =
2.25^2/312.75 + 3.75^2/104.25 + 3.25^2/104.25 + 2.75^2/34.75 = 0.016
+ 0.135 + 0.101 + 0.218 = 0.47 < 7.81$: consistent with $9 : 3 : 3 :
1$.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:6}
Against $1 : 1 : 1 : 1$ (250 each): $\chi^2 = (162^2 + 138^2 + 147^2
+ 153^2)/250 = 361 \gg 7.81$: linked. Recombinants $103 + 97 = 200$
of 1000: $r = 0.20$, \qty{20}{cM}. In repulsion, $Ab/aB$ gives 400
$Ab$, 400 $aB$, 100 $AB$, 100 $ab$: the same distance, the parental
and recombinant classes exchanged.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:7}
$d = -\tfrac12\ln(1 - 2r)$: $r = 0.20$ gives $-\tfrac12\ln 0.6 =
0.255$, \qty{25.5}{cM}; $r = 0.45$ gives $-\tfrac12\ln 0.1 = 1.15$,
\qty{115}{cM}. $r = \tfrac12(1 - e^{-2d})$: \qty{30}{cM} gives
$\tfrac12(1 - e^{-0.6}) = 0.226$; \qty{100}{cM} gives $\tfrac12(1 -
e^{-2}) = 0.43$. Below \qty{10}{cM} the correction is negligible;
above \qty{30}{cM} it is essential, and above \qty{50}{cM} $r$
saturates so that a single two-point cross can no longer measure the
distance --- one maps far genes by chaining close ones.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:8}
(a) $X^wX^w \times X^+Y$: all sons white ($X^wY$), all daughters red
($X^+X^w$). (b) $X^+X^w \times X^wY$: sons half red, half white;
daughters half red ($X^+X^w$), half white ($X^wX^w$). (c) daughters
$X^+X^w$ $\times$ brothers $X^wY$: as in (b) --- half of each sex
white, half red.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:9}
Two enzymes in series make the pigment; purple needs a \omterm{def:b2:meiosis-heredity:vocabulary}{dominant}
\omterm{def:b2:meiosis-heredity:vocabulary}{allele} at both loci ($C\_P\_$). The lines are $CCpp$ and $ccPP$
(each white for a different reason); the hybrid $CcPp$ is purple;
selfed it gives $9\,C\_P\_$ purple to $7$ white ($3\,C\_pp + 3\,ccP\_ +
1\,ccpp$). Test cross $CcPp\times ccpp$: $1$ purple ($CcPp$) : $3$
white.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:10}
Parentals $+++$, $abc$; double crossovers $+b+$, $a+c$; $b$ is the
gene that switched: order $a$--$b$--$c$. $r_{ab} = (11 + 9 + 2)/1202
= 0.018$ (\qty{1.8}{cM}); $r_{bc} = (118 + 112 + 2)/1202 = 0.193$
(\qty{19.3}{cM}). Expected doubles $0.018\times 0.193\times 1202 =
4.2$; observed 2: coincidence $0.47$, interference $0.53$.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:11}
Second-division asci $300/1000 = \qty{30}{\%}$; distance $=
\tfrac12\times 30 = \qty{15}{cM}$. A crossover between gene and
centromere involves only two of the four chromatids, so an ascus
showing second-division segregation carries two recombinant and two
parental chromatids: the \omterm{def:b2:meiosis-heredity:linkage}{recombination frequency} is half the
frequency of such asci. For $2 : 4 : 2$: a crossover between the
locus and the centromere exchanges the inner two chromatids, so
that each half-spindle at \omterm{def:b2:meiosis-heredity:meiosis}{meiosis} I carries one $A$ and one $a$
chromatid; \omterm{def:b2:meiosis-heredity:meiosis}{meiosis} II then separates them in the order $A\,a \mid
a\,A$ (or the reverse), and the final mitosis doubles each spore:
$2 : 4 : 2$.
\end{solution}

\begin{solution}{exo:b2:meiosis-heredity:12}
Her mother is an obligate carrier (she received her father's only
X), so the woman received the \omterm{def:b2:meiosis-heredity:vocabulary}{allele} with probability $\tfrac12$.
Two healthy sons: a carrier has probability $(\tfrac12)^2 = \tfrac14$
of two healthy sons, a non-carrier probability 1. Bayes:
$P(\text{carrier}\mid\text{data}) = (\tfrac12\times\tfrac14)/(\tfrac12
\times\tfrac14 + \tfrac12\times 1) = \tfrac{1/8}{5/8} = \tfrac15$.
\end{solution}

\begin{solution}{pb:b2:meiosis-heredity:1}
\textbf{1.} Parents $vg/vg\;+/+$ and $+/+\;e/e$; $F_1$ $+/vg\;+/e$;
gametes $+\,+$, $+\,e$, $vg\,+$, $vg\,e$, each $\tfrac14$.
\textbf{2.} $900 : 300 : 300 : 100$.
\textbf{3.} $\chi^2 = 8^2/900 + 9^2/300 + 4^2/300 + 3^2/100 = 0.07 +
0.27 + 0.05 + 0.09 = 0.48 < 7.81$: \omterm{thm:b2:meiosis-heredity:mixing}{independent assortment} holds.
\textbf{4.} \omterm{def:b2:meiosis-heredity:vocabulary}{Homozygous} wild at both loci is $\tfrac{1}{16}$ of all,
$\tfrac{9}{16}$ are wild: $\tfrac19$.
\textbf{5.} Wild, vestigial, ebony, vestigial ebony, $\tfrac14$ each.
\textbf{6.} The \omterm{def:b2:meiosis-heredity:vocabulary}{phenotype} is the product of \omterm{def:b2:meiosis-heredity:vocabulary}{genotype} and environment
(temperature acts on the pigment enzymes): a \omterm{def:b2:meiosis-heredity:vocabulary}{genotype} specifies a
norm of reaction, not a fixed character.
\textbf{7.} $b\,+/+\,vg$ (repulsion): $b$ came with $+_{vg}$ from one
parent, $vg$ with $+_b$ from the other.
\textbf{8.} Parental: black ($b\,+$) 965 and vestigial ($+\,vg$) 944;
recombinant: black vestigial ($b\,vg$) 206 and wild ($+\,+$) 185.
The wild-type chromosome $+\,+$ did not exist in the $F_1$; it can
only be made by a crossover.
\textbf{9.} $r = 391/2300 = 0.17$: \qty{17}{cM}.
\textbf{10.} $d = -\tfrac12\ln(1 - 0.34) = -\tfrac12\ln 0.66 = 0.21$:
\qty{21}{cM}.
\textbf{11.} There is no crossing-over in male \emph{Drosophila}:
the male's gametes carry only the two parental chromosomes.
\textbf{12.} $F_1$ $b\,vg/+\,+$ (coupling): parental classes wild and
black vestigial, \qty{41.5}{\%} each; recombinant black and
vestigial, \qty{8.5}{\%} each.
\textbf{13.} $X^hX^+$: a daughter receives her father's only X, which
carries $h$, and a normal X from her unaffected mother.
\textbf{14.} $\tfrac12$ (son) $\times$ $\tfrac12$ (receives $X^h$) $=
\tfrac14$.
\textbf{15.} $\tfrac12$: she receives either of the mother's X
chromosomes; the father gives a normal X.
\textbf{16.} Each child unaffected with probability $\tfrac34$:
$(\tfrac34)^3 = \tfrac{27}{64} = 0.42$.
\textbf{17.} A son receives his father's Y, never his X. A woman is
affected only as $X^hX^h$: an affected father and a carrier (or
affected) mother, which is rare.
\textbf{18.} Prior $\tfrac12$. Two healthy sons have probability
$\tfrac14$ if she is a carrier, 1 if not: posterior $(\tfrac12\times
\tfrac14)/(\tfrac12\times\tfrac14 + \tfrac12) = \tfrac15$.
\textbf{19.} A son's only X comes from his mother, and the Y carries
none of these genes, so each son displays one maternal gamete
unmasked: the cross is its own test cross.
\textbf{20.} Parentals $+++$ (853), $ywm$ (833); double crossovers
$+w+$ (2), $y+m$ (2); $w$ switched: order $y$--$w$--$m$.
\textbf{21.} $r_{yw} = (24 + 26 + 2 + 2)/2000 = 0.027$; $r_{wm} = (128
+ 132 + 2 + 2)/2000 = 0.132$. Map: $y$ --- \qty{2.7}{cM} --- $w$ ---
\qty{13.2}{cM} --- $m$.
\textbf{22.} Expected $0.027\times 0.132\times 2000 = 7.1$; observed
4: coincidence $0.56$, interference $0.44$.
\textbf{23.} $y$--$m$ recombinants: $y++$, $+wm$, $yw+$, $++m$: $(24 +
26 + 128 + 132)/2000 = 0.155$, against $0.027 + 0.132 = 0.159$: the
four double crossovers restore the parental $y$--$m$ combination and
are missed, $2\times 4/2000 = 0.004$.
\textbf{24.} Daughters receive the father's $y\,w\,m$ X and one
maternal X: all are \omterm{def:b2:meiosis-heredity:vocabulary}{heterozygous} and wild in \omterm{def:b2:meiosis-heredity:vocabulary}{phenotype}, whatever
recombinant X they carry; the recombination is invisible.
\textbf{25.} $y$ --- \qty{2.7}{cM} --- $w$ --- \qty{13.2}{cM} --- $m$;
coefficient of coincidence $0.56$, interference $0.44$.
\end{solution}
