\chapter{How Living Things Reproduce}\label{ch:g5:reproduction}

Every species must produce new individuals or it disappears. \omterm{def:g5:reproduction:def}{Reproduction}
can be sexual or asexual, and \omterm{def:g5:reproduction:lifecycle}{life cycles} link generations. This chapter
surveys the main patterns without detailed human anatomy.

\section{Why reproduce?}

\begin{definition}[Reproduction]\label{def:g5:reproduction:def}
\emph{Reproduction}\index{reproduction} is the process by which \omterm{def:g1:living-nonliving:living}{living
things} produce \omterm{def:g2:growing:offspring}{offspring}. It continues the species across generations.
\end{definition}

\begin{proposition}[Two broad strategies]\label{prop:g5:reproduction:two}
Some \omterm{def:g4:ecosystems:levels}{organisms} \omterm{def:g2:what-alive:signs}{reproduce} \emph{asexually}\index{asexual reproduction}
(one parent; \omterm{def:g2:growing:offspring}{offspring} often genetically very similar). Others
\omterm{def:g2:what-alive:signs}{reproduce} \emph{sexually}\index{sexual reproduction} (usually two
parents; \omterm{def:g2:growing:offspring}{offspring} mix \omterm{def:g4:traits:trait}{traits}).
\end{proposition}
\admitted

\section{Asexual reproduction}

\begin{definition}[Asexual examples]\label{def:g5:reproduction:asexual}
In asexual \omterm{def:g5:reproduction:def}{reproduction}, a single \omterm{def:g4:ecosystems:levels}{organism} produces new individuals
without \omterm{def:g5:reproduction:sexual}{gametes} combining. Examples: \omterm{def:g4:microbes:def}{bacteria} dividing; strawberry
runners; potato tubers; some \omterm{def:g1:plants-animals:plant}{plants} from cuttings.
\end{definition}

\begin{example}[Potato]\label{ex:g5:reproduction:potato}
An \omterm{def:g5:nerves:senses}{eye} on a potato tuber can \omterm{def:g2:what-alive:signs}{grow} into a new \omterm{def:g1:plants-animals:plant}{plant} genetically like the
parent \omterm{def:g1:plants-animals:plant}{plant} --- useful for farmers who want uniform crops.
\end{example}

\section{Sexual reproduction}

\begin{definition}[Sexual reproduction]\label{def:g5:reproduction:sexual}
In \emph{sexual reproduction}\index{sexual reproduction}, special sex
\omterm{def:g5:cell-structure:cell}{cells} (\emph{gametes}\index{gamete}) join at \emph{fertilisation}
\index{fertilisation} to form a new \omterm{def:g5:cell-structure:cell}{cell} that can develop into an
\omterm{def:g2:growing:offspring}{offspring}. In flowering \omterm{def:g1:plants-animals:plant}{plants}, pollen provides the male gamete; ovules
contain the female gamete. In many \omterm{def:g1:plants-animals:animal}{animals}, sperm and \omterm{def:g3:life-cycles:butterfly}{egg} join.
\end{definition}

\begin{example}[Flowering plants]\label{ex:g5:reproduction:flower}
Pollen reaches a stigma (pollination). \omterm{def:g5:reproduction:sexual}{Fertilisation} leads to \omterm{def:g2:plant-parts:main}{seeds}
inside \omterm{def:g2:plant-parts:main}{fruits}. \omterm{def:g2:plant-parts:main}{Seeds} can \omterm{def:g2:what-alive:signs}{grow} into new \omterm{def:g1:plants-animals:plant}{plants} when conditions are
right.
\end{example}

\section{Life cycles}

\begin{definition}[Life cycle]\label{def:g5:reproduction:lifecycle}
A \emph{life cycle}\index{life cycle} is the series of stages from the
start of one generation to the same stage in the next --- for example
\omterm{def:g3:life-cycles:butterfly}{egg} $\to$ \omterm{def:g3:life-cycles:butterfly}{larva} $\to$ \omterm{def:g3:life-cycles:butterfly}{pupa} $\to$ \omterm{def:g3:life-cycles:butterfly}{adult} in a butterfly, or \omterm{def:g3:seeds:seed}{seed} $\to$
seedling $\to$ \omterm{def:g3:life-cycles:butterfly}{adult} \omterm{def:g1:plants-animals:plant}{plant} $\to$ \omterm{def:g3:seeds:seed}{seed}.
\end{definition}

\begin{method}[Reading a life-cycle diagram]\label{met:g5:reproduction:diagram}
\begin{enumerate}
  \item Find the starting stage marked (\omterm{def:g3:life-cycles:butterfly}{egg}, \omterm{def:g3:seeds:seed}{seed}, birth).
  \item Follow arrows through \omterm{def:g2:growing:growth}{growth} and change.
  \item Note where \omterm{def:g5:reproduction:def}{reproduction} produces the next generation.
  \item Ask whether the young look like the \omterm{def:g3:life-cycles:butterfly}{adult} or transform.
\end{enumerate}
\end{method}

\begin{example}[Frog]\label{ex:g5:reproduction:frog}
\omterm{def:g3:life-cycles:butterfly}{Eggs} in \omterm{def:g4:plants-need:needs}{water} $\to$ tadpoles (gills, tails) $\to$ froglets $\to$ \omterm{def:g3:life-cycles:butterfly}{adult}
frogs. The young do not look like the \omterm{def:g3:life-cycles:butterfly}{adults} at first.
\end{example}
