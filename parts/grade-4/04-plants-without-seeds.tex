\chapter{Plants Without Seeds}\label{ch:g4:plants-without-seeds}

The shady side of the old wall is upholstered in moss; under
the beech trees, \omterm{prop:g4:plants-without-seeds:damp}{ferns} unroll their great green fronds. Search
them all summer: you will find no \omterm{def:g3:flowers-fruits-seeds:parts}{flower}, no \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}, no \omterm{def:g2:from-seed-to-plant:seed}{seed} ---
ever. Yet moss and \omterm{prop:g4:plants-without-seeds:damp}{ferns} spread perfectly well. \omterm{def:g1:plants-around-us:plant}{Plants}, it
turns out, have more than one way of making the next
generation.

\section{Ferns and mosses: reproduction by spores}

\begin{definition}[Spore]\label{def:g4:plants-without-seeds:spore}
A \emph{spore}\index{spore} is a microscopic grain, made in
enormous numbers, from which a new \omterm{def:g1:plants-around-us:plant}{plant} can grow. Unlike a
\omterm{def:g2:from-seed-to-plant:seed}{seed} (\cref{def:g2:from-seed-to-plant:seed}), a spore is a
single speck with \emph{no \omterm{def:g3:animals-through-seasons:active}{food store} and no ready-made tiny
\omterm{def:g1:plants-around-us:plant}{plant} inside} --- it is a beginning at its barest.
\end{definition}

\begin{example}[Under the fern's fronds]\label{ex:g4:plants-without-seeds:fern}
Turn a fern frond over in late summer: neat rows of brown dots
line its underside. Each dot is a packet of thousands of
\omterm{def:g4:plants-without-seeds:spore}{spores}; ripe, the packets split, and the \omterm{def:g4:plants-without-seeds:spore}{spores} drift off on
the air like the finest dust. A \omterm{def:g4:plants-without-seeds:spore}{spore} landing on damp shaded
ground can grow --- by stages --- into a new fern.
\end{example}

\begin{proposition}[Spore plants need dampness]\label{prop:g4:plants-without-seeds:damp}
\omterm{def:g4:plants-without-seeds:spore}{Spore} \omterm{def:g1:plants-around-us:plant}{plants} --- \emph{mosses}\index{moss} and
\emph{ferns}\index{fern} --- can only reproduce where it is
\emph{damp}: their \omterm{def:g4:plants-without-seeds:spore}{spores}, and the delicate stages that grow
from them, dry out and die in dry air. That is why moss
carpets the shaded north wall but never the sunny south one,
and why \omterm{prop:g4:plants-without-seeds:damp}{ferns} crowd damp woods, stream banks and wells.
\end{proposition}
\admitted

\begin{remark}[A spore is not a seed]\label{rem:g4:plants-without-seeds:sporeseed}
Both are beginnings that travel, but the \omterm{def:g2:from-seed-to-plant:seed}{seed} is a survival
kit --- coat, \omterm{def:g3:animals-through-seasons:active}{food store}, ready-made plantlet --- while the
\omterm{def:g4:plants-without-seeds:spore}{spore} is bare. Hence the numbers: the fern's millions of
\omterm{def:g4:plants-without-seeds:spore}{spores} against the bean pod's dozen \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}. Bare beginnings
are cheap to make and easy to lose; the two plans echo the
codfish and the elephant of
\cref{prop:g4:animal-reproduction:strategies}.
\end{remark}

\section{New plants without any sowing}

\begin{definition}[Vegetative reproduction]\label{def:g4:plants-without-seeds:vegetative}
Many \omterm{def:g1:plants-around-us:plant}{plants} can also make new \omterm{def:g1:plants-around-us:plant}{plants} \emph{directly from a
piece of themselves} --- no \omterm{def:g4:plants-without-seeds:spore}{spores}, no \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}, one parent only.
This is \emph{vegetative reproduction}\index{vegetative
reproduction}. The new \omterm{def:g1:plants-around-us:plant}{plant} is a living continuation of the
old one, and resembles it exactly.
\end{definition}

\begin{example}[The strawberry's runners]\label{ex:g4:plants-without-seeds:runner}
A strawberry \omterm{def:g1:plants-around-us:plant}{plant} sends out long thin horizontal \omterm{def:g1:plants-around-us:parts}{stems} ---
\emph{runners}\index{runner} --- across the soil. Where a
runner touches down, it \omterm{def:g1:plants-around-us:parts}{roots}, sprouts \omterm{def:g1:plants-around-us:parts}{leaves}, and becomes a
complete new strawberry \omterm{def:g1:plants-around-us:plant}{plant}; the runner then withers, and
the child stands free. One \omterm{def:g1:plants-around-us:plant}{plant} becomes a patch without a
single \omterm{def:g2:from-seed-to-plant:seed}{seed} sown.
\end{example}

\begin{example}[Bulbs and tubers]\label{ex:g4:plants-without-seeds:bulbtuber}
The tulip \emph{bulb}\index{bulb} and the potato
\emph{tuber}\index{tuber} are underground storerooms --- and
starting kits. Left in the ground, a \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulb} splits into
daughter \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbs}, each a future tulip; every ``\omterm{def:g1:the-five-senses:sense}{eye}'' of a
sprouting potato can grow into a whole new \omterm{def:g1:plants-around-us:plant}{plant}. Gardeners
\omterm{def:g1:plants-around-us:plant}{plant} potatoes, not potato \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}.
\end{example}

\begin{method}[Taking a cutting]\label{met:g4:plants-without-seeds:cutting}
Gardeners copy \omterm{def:g4:plants-without-seeds:vegetative}{vegetative reproduction} with
\emph{cuttings}\index{cutting}:
\begin{enumerate}
  \item cut a healthy young shoot, a hand's length, from the
        parent \omterm{def:g1:plants-around-us:plant}{plant} (mint, willow and geraniums work well);
  \item remove the lower \omterm{def:g1:plants-around-us:parts}{leaves} and stand the shoot in water
        or moist soil;
  \item wait: in days or weeks, \omterm{def:g1:plants-around-us:parts}{roots} grow from the cut \omterm{def:g1:plants-around-us:parts}{stem};
  \item \omterm{def:g1:plants-around-us:plant}{plant} it out --- a complete new \omterm{def:g1:plants-around-us:plant}{plant}, identical to
        its one parent.
\end{enumerate}
\end{method}

\begin{remark}[One parent, exact copies]\label{rem:g4:plants-without-seeds:copy}
\omterm{prop:g3:flowers-fruits-seeds:transform}{Seeds} mix two parents
(\cref{rem:g4:animal-reproduction:mix} holds for \omterm{def:g1:plants-around-us:plant}{plants} too,
via \omterm{def:g3:flowers-fruits-seeds:parts}{pollen}); \omterm{def:g4:plants-without-seeds:vegetative}{vegetative reproduction} copies one. That is why
every \omterm{def:g1:plants-around-us:plant}{plant} of a named apple variety is grown from \omterm{met:g4:plants-without-seeds:cutting}{cuttings}
and grafts, never from pips: the orchard wants exact copies
of one excellent tree. The price of copying is sameness ---
an orchard of identical trees shares identical weaknesses,
as a later year will show.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:1}
Name two \omterm{def:g1:plants-around-us:plant}{plants} that never make \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}. How do they reproduce?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:2}
Give two differences between a \omterm{def:g4:plants-without-seeds:spore}{spore} and a \omterm{def:g2:from-seed-to-plant:seed}{seed}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:3}
Where are the fern's \omterm{def:g4:plants-without-seeds:spore}{spores} made? Describe what you find
under a frond in late summer.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:4}
Why does moss grow on the shaded north wall and not the sunny
south one?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:5}
What is \omterm{def:g4:plants-without-seeds:vegetative}{vegetative reproduction}? How many parents does it
take?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:6}
Tell the story of a strawberry runner, from first \omterm{def:g1:plants-around-us:parts}{stem} to
free-standing child \omterm{def:g1:plants-around-us:plant}{plant}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:7}
Why do gardeners \omterm{def:g1:plants-around-us:plant}{plant} potatoes rather than sow potato \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}?
What can each ``\omterm{def:g1:the-five-senses:sense}{eye}'' do?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:8}
Try it: take a mint cutting by
\cref{met:g4:plants-without-seeds:cutting} and stand it in a
glass of water on the windowsill. What appears, and after
roughly how long?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:9}
The fern makes millions of \omterm{def:g4:plants-without-seeds:spore}{spores}; the bean pod, a dozen
\omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}. Explain the difference in numbers with
\cref{rem:g4:plants-without-seeds:sporeseed} --- and name the
\omterm{def:g1:animals-around-us:animal}{animal} pair whose strategies they echo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:10}
A gardener finds the new strawberry \omterm{def:g1:plants-around-us:plant}{plants} identical to the
parent, while her bean \omterm{def:g1:plants-around-us:plant}{plants} vary a little from their
parents. Explain both observations with
\cref{rem:g4:plants-without-seeds:copy}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:plants-without-seeds:11}
A willow \omterm{def:g1:plants-around-us:tree}{branch} snapped off by a winter flood, carried
downstream and stranded on a mud bank, often grows into a new
willow tree. Which kind of reproduction is this --- and why
are rivers so often lined with willows? What did the flood do
that the gardener of
\cref{met:g4:plants-without-seeds:cutting} does with scissors?
\end{exercise}
