\chapter{De stadia van het leven}\label{ch:g3:stages-of-life}

Zet een familiefoto op een rij: de \omterm{def:g3:stages-of-life:stages}{baby} op oma's schoot, het schoolkind
met twee voortanden weg, de neef in de puberteit die opeens groter is
dan zijn moeder, de ouders, oma zelf. Eén familie, één middag --- en de
hele menselijke \omterm{def:g2:born-grow-die:cycle}{levenscyclus} op één bank. Dit jaar kijken we goed naar
die stadia, en naar hoe je het groeien zelf kunt meten.

\section{De stadia van een mensenleven}

\begin{definition}[De stadia van het mensenleven]\label{def:g3:stages-of-life:stages}
Een mensenleven doorloopt stadia:
\begin{enumerate}
  \item de \emph{baby}\index{baby} (ruwweg de eerste twee jaar): de
        snelste groei van allemaal, eerste tandjes, eerste stapjes,
        eerste woordjes;
  \item het \emph{kind}\index{kind}: gestage groei, \omterm{prop:g2:teeth-and-eating-well:sets}{melktanden} ingeruild
        voor \omterm{prop:g2:teeth-and-eating-well:sets}{blijvende tanden}, leren van alles en nog wat;
  \item de \emph{puber}\index{puber}: een nieuwe stoot van snelle groei,
        en een lichaam dat van dat van een kind in dat van een
        volwassene verandert;
  \item de \emph{volwassene}\index{volwassene}: de groei in lengte is
        voorbij; volwassenen kunnen ouders worden;
  \item de \emph{ouderdom}\index{ouderdom}: het haar wordt wit, de \omterm{def:g1:the-five-senses:sense}{huid}
        rimpelt, de kracht neemt langzaam af --- het natuurlijke laatste
        stadium van de cyclus.
\end{enumerate}
\end{definition}

\begin{example}[De bank lezen]\label{ex:g3:stages-of-life:sofa}
Op de familiebank: de \omterm{def:g3:stages-of-life:stages}{baby} (stadium 1), het schoolkind met de gaten in
zijn gebit (stadium 2 --- de wisseling uit
\cref{ch:g2:teeth-and-eating-well} in volle gang), de neef van zestien
(stadium 3, midden in de stoot), de ouders (stadium 4), oma (stadium 5).
Vijf stadia, één familie.
\end{example}

\begin{remark}[Groeien gaat niet gelijkmatig]\label{rem:g3:stages-of-life:bursts}
Groeien heeft een ritme: heel snel bij de \omterm{def:g3:stages-of-life:stages}{baby} --- de lengte neemt in
het eerste jaar ongeveer met de helft toe --- langzamer en gestaag door
de kindertijd heen, weer snel bij de \omterm{def:g3:stages-of-life:stages}{puber}, en dan voorbij. Twee stoten
met een rustige strook ertussen: dat is de normale vorm van opgroeien.
\end{remark}

\section{Groei meten}

\begin{definition}[Groeicurve]\label{def:g3:stages-of-life:chart}
Een \emph{groeicurve}\index{groeicurve} is een verslag van de lengte
(of het gewicht), keer op keer gemeten in de loop van de tijd --- de
streepjes op de deurpost uit \cref{ch:g1:my-body}, regelmatig gemaakt
en gedateerd. Dokters houden er voor elk kind een bij.
\end{definition}

\begin{example}[De curve lezen]\label{ex:g3:stages-of-life:reading}
Op de curve klimt de lijn in het allereerste jaar ongeveer
$\qty{25}{cm}$ --- en door de kindertijd heen maar zo'n
$\qty{6}{cm}$ per jaar. De twee steile stukken zijn de twee stoten uit
\cref{rem:g3:stages-of-life:bursts}; het vlakke einde zegt dat de
\omterm{def:g3:stages-of-life:stages}{volwassene} niet langer groeit.
\end{example}

\begin{method}[Je eigen groeicurve bijhouden]\label{met:g3:stages-of-life:keep}
\begin{enumerate}
  \item Meet je lengte zorgvuldig (met de muur en het boek uit de
        vorige jaren), altijd zonder schoenen;
  \item schrijf de datum en de lengte in een schrift, of zet een
        streepje op de deurpost en zet er de datum bij;
  \item meet elke paar maanden opnieuw --- regelmaat telt zwaarder dan
        vaak meten;
  \item zet de punten uit om het verhaal te zien: de leeftijd langs de
        onderkant, de lengte langs de zijkant, zoals in de figuur.
\end{enumerate}
\end{method}

\section{Dieren en planten hebben ook stadia}

\begin{proposition}[Stadia door de levende wereld heen]\label{prop:g3:stages-of-life:animals}
Elk \omterm{def:g1:living-or-not:living}{levend wezen} doorloopt de stadia van zijn eigen versie van de
cyclus --- jong, volwassen, oud. Het jonge stadium kan op de \omterm{def:g3:stages-of-life:stages}{volwassene}
lijken (het kitten, het veulen) of er in niets op lijken (de donderpad,
de \omterm{ex:g2:animals-grow-up:butterfly}{rups} uit \cref{ch:g2:animals-grow-up}); bij een \omterm{def:g1:plants-around-us:plant}{plant} lopen de
stadia van \omterm{def:g2:from-seed-to-plant:seed}{zaad} naar \omterm{prop:g2:from-seed-to-plant:cycle}{kiemplantje}, jonge \omterm{def:g1:plants-around-us:plant}{plant} en bloeiende \omterm{def:g3:stages-of-life:stages}{volwassene}.
Wat geen van alle kan, is een stadium overslaan of ze achterstevoren
doorlopen.
\end{proposition}
\admitted

\begin{example}[Hoe snel, hoe langzaam]\label{ex:g3:stages-of-life:speeds}
De snelheden lopen enorm uiteen. Een vlieg is in weken volwassen; een
kat in ongeveer een jaar; een mens in bijna twintig jaar; een eik doet
er tientallen jaren over voor zijn eerste eikels. Langzaam of snel, de
volgorde van de stadia verandert nooit.
\end{example}

\begin{remark}[Ouderdom is een stadium, geen gebrek]\label{rem:g3:stages-of-life:oldage}
Wit haar en rimpels zijn geen ziekte: het is het laatste stadium van
het lichaam, net zo natuurlijk als de eerste tandjes van de \omterm{def:g3:stages-of-life:stages}{baby}. Bij
veel soorten, en in mensenfamilies, dragen de ouderen wat de jongeren
het hardst nodig hebben --- ervaring.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:1}
Noem de vijf stadia van een mensenleven, op volgorde.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:2}
In welk stadium: (a) vallen de \omterm{prop:g2:teeth-and-eating-well:sets}{melktanden} uit? (b) kan iemand ouder
worden? (c) is de groei de snelste van het hele leven?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:3}
Wat zijn de twee groeistoten, en wat ligt ertussen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:4}
Hoe kun je op de \omterm{def:g3:stages-of-life:chart}{groeicurve} alleen aan de lijn zien dat een \omterm{def:g3:stages-of-life:stages}{volwassene}
niet meer groeit?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:5}
Hoeveel centimeter klimt de lijn ongeveer tussen $6$ en $8$ jaar, als
je de curve gebruikt? En in het eerste jaar alleen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:6}
Waarom moet je de lengte altijd zonder schoenen meten, elke keer op
dezelfde manier? Wat zou er anders mis kunnen gaan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:7}
Geef de stadia van het leven van een eik, van eikel tot oude boom.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:8}
Probeer het: begin voor jezelf de \omterm{def:g3:stages-of-life:chart}{groeicurve} uit
\cref{met:g3:stages-of-life:keep}, met het eerste punt van vandaag. In
welk stadium van \cref{def:g3:stages-of-life:stages} zit jij --- en
welke stoot ligt nog voor je?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:9}
Een vlieg, een kat, een mens en een eik doorlopen dezelfde stadia met
heel verschillende snelheden. Zet ze op volgorde van snelst naar
langzaamst volwassen --- en zeg wat alle vier de volgordes van stadia
gemeen hebben.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:10}
Tom van $9$ meet zichzelf twee keer in één maand en vindt ``helemaal
geen groei''. Moet hij zich zorgen maken? Antwoord met
\cref{rem:g3:stages-of-life:bursts} en de vorm van de curve op zijn
leeftijd.
\end{exercise}
