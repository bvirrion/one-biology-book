\section*{Chapter \ref{ch:g12:reproduction-development} --- Reproduction and Development}

\begin{solution}{exo:g12:reproduction-development:1}
A \omterm{def:g12:reproduction-development:sexual}{gamete} is a \omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} reproductive \omterm{def:g10:the-cell:cell}{cell} (sperm or egg). A \omterm{def:g12:reproduction-development:sexual}{zygote} is the
\omterm{def:g11:meiosis-life-cycles:ploidy}{diploid} \omterm{def:g10:the-cell:cell}{cell} formed by fusion of \omterm{def:g12:reproduction-development:sexual}{gametes}. \omterm{def:g12:reproduction-development:fertilisation}{Fertilisation} is that fusion,
restoring diploidy and usually activating \omterm{def:g10:scientific-biology:properties}{development} of the embryo.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:2}
\omterm{def:g11:meiosis-life-cycles:gametogenesis}{Spermatogenesis} typically yields four functional sperm from each \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}.
\omterm{def:g11:meiosis-life-cycles:gametogenesis}{Oogenesis} usually yields one large ovum plus \omterm{def:g11:meiosis-life-cycles:gametogenesis}{polar bodies} that discard extra
\omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} nuclei, so resources concentrate in a single egg.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:3}
\omterm{def:g12:reproduction-development:cleavage}{Cleavage} is the series of rapid early mitoses that partition the \omterm{def:g12:reproduction-development:sexual}{zygote} into
\omterm{def:g12:reproduction-development:cleavage}{blastomeres}. Overall mass grows little because cycles are short and existing
cytoplasm or yolk is subdivided rather than expanded by long \omterm{def:g10:scientific-biology:properties}{growth} phases.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:4}
The three \omterm{def:g12:reproduction-development:gastrulation}{germ layers} of a triploblastic animal are \omterm{prop:g12:reproduction-development:layers}{ectoderm}, \omterm{prop:g12:reproduction-development:layers}{mesoderm} and
\omterm{prop:g12:reproduction-development:layers}{endoderm}. They are established at \omterm{def:g12:reproduction-development:gastrulation}{gastrulation} and \omterm{def:g12:plant-biology:flower}{seed} later organ systems.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:5}
After the first sperm fuses, a fast block (change in membrane potential)
and/or a slow block (cortical granule reaction that hardens the coat) prevent
additional sperm from fertilising the same egg, protecting diploidy.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:6}
Nervous system derives from \omterm{prop:g12:reproduction-development:layers}{ectoderm}; gut lining from \omterm{prop:g12:reproduction-development:layers}{endoderm}; skeletal
muscle from \omterm{prop:g12:reproduction-development:layers}{mesoderm}. These assignments organise \omterm{def:g12:reproduction-development:organo}{organogenesis} even though
details vary among taxa.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:7}
\omterm{def:g12:reproduction-development:gastrulation}{Gastrulation} rearranges the \omterm{def:g12:reproduction-development:cleavage}{blastula} into a multilayered \omterm{def:g12:reproduction-development:gastrulation}{gastrula} with \omterm{def:g12:reproduction-development:gastrulation}{germ
layers} and emerging body axes. It is critical because it sets the spatial
layout on which all later \omterm{def:g12:reproduction-development:organo}{organogenesis} depends.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:8}
Maternal and fetal blood remain in separate circulations. Nutrients, gases
and wastes cross by \omterm{def:g10:membranes-enzymes:diffusion}{diffusion} and transport across closely apposed placental
membranes and tissues without free mixing of the two bloodstreams.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:9}
The \omterm{def:g12:reproduction-development:amniote}{amniotic egg} provides a self-contained aquatic environment (amnion),
routes for \omterm{def:g12:animal-exchange:gas}{gas exchange} and waste storage, and often a protective shell. That
package frees embryonic \omterm{def:g10:scientific-biology:properties}{development} from open water and was a major land
\omterm{def:g11:evolution-mechanisms:fitness}{adaptation} for amniotes.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:10}
Frogs: many \omterm{def:g12:reproduction-development:sexual}{gametes}, external \omterm{def:g12:reproduction-development:fertilisation}{fertilisation}, aquatic eggs and larvae, limited
parental care per offspring. Humans: internal \omterm{def:g12:reproduction-development:fertilisation}{fertilisation}, few offspring,
long gestation with a \omterm{def:g12:reproduction-development:placenta}{placenta}, extensive parental care. The trade-off is
quantity of offspring versus investment per offspring.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:11}
If mitotic spindles fail during \omterm{def:g12:reproduction-development:cleavage}{cleavage}, \omterm{def:g12:reproduction-development:cleavage}{blastomeres} cannot segregate
\omterm{def:g11:dna-replication:chromosome}{chromosomes} properly; \omterm{def:g12:reproduction-development:cleavage}{cleavage} arrests or produces abnormal \omterm{def:g10:the-cell:cell}{cell} numbers. The
\omterm{def:g12:reproduction-development:human}{blastocyst} fails to form correctly and \omterm{def:g12:reproduction-development:human}{implantation} and pregnancy fail.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:12}
\omterm{def:g11:meiosis-life-cycles:meiosis}{Meiosis} halves \omterm{def:g11:dna-replication:chromosome}{chromosome} number so \omterm{def:g12:reproduction-development:fertilisation}{fertilisation} restores $2n$ each
generation rather than doubling ploidy. \omterm{prop:g11:meiosis-life-cycles:prophaseI}{Crossing over} and independent
assortment reshuffle \omterm{def:g11:mendelian-genetics:allele}{alleles} among \omterm{def:g12:reproduction-development:sexual}{gametes}, so siblings (except identical
twins) inherit different genetic combinations.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:13}
\omterm{def:g12:reproduction-development:menstrual}{FSH} promotes early follicle \omterm{def:g10:scientific-biology:properties}{growth}; rising \omterm{def:g12:reproduction-development:menstrual}{oestrogen} from the growing follicle
rebuilds the endometrium and feeds back on the \omterm{def:g12:nerves-hormones:glands}{pituitary}. A mid-cycle \omterm{def:g12:reproduction-development:menstrual}{LH} surge
triggers ovulation. After ovulation the corpus luteum secretes \omterm{def:g12:reproduction-development:menstrual}{progesterone}
(and some \omterm{def:g12:reproduction-development:menstrual}{oestrogen}) to maintain the lining in the luteal phase.
\end{solution}

\begin{solution}{exo:g12:reproduction-development:14}
Early embryonic signals such as hCG maintain the corpus luteum so \omterm{def:g12:reproduction-development:menstrual}{progesterone}
stays high and the endometrium is not shed. Without pregnancy the corpus
luteum fails, \omterm{def:g12:reproduction-development:menstrual}{progesterone} falls, and menstruation resets the uterus.
Successful \omterm{def:g12:reproduction-development:human}{implantation} therefore interrupts the usual menstrual decline.
\end{solution}

\begin{solution}{pb:g12:reproduction-development:1}
\textbf{Part I.} (1)~Sperm in testes, eggs in ovaries; both \omterm{def:g11:meiosis-life-cycles:ploidy}{haploid}.
(2)~Unequal \omterm{def:g11:cell-cycle-mitosis:cytokinesis}{cytokinesis} conserves cytoplasm and resources in one egg.
(3)~\omterm{prop:g11:meiosis-life-cycles:prophaseI}{Crossing over}, independent assortment, and later random \omterm{def:g12:reproduction-development:fertilisation}{fertilisation}.

\textbf{Part II.} (4)--(6)~As in chapter methods; inner \omterm{def:g10:the-cell:cell}{cell} mass $\to$ embryo;
trophoblast $\to$ \omterm{def:g12:reproduction-development:placenta}{placenta}; \omterm{def:g12:reproduction-development:human}{implantation} into the uterus.
(7)~Multiple sperm nuclei risk polyploidy or chaotic \omterm{def:g11:dna-replication:chromosome}{chromosome} number.

\textbf{Part III.} (8)--(10)~\omterm{def:g12:reproduction-development:gastrulation}{Gastrulation} forms layers and axes; adult
derivatives as in the chapter; \omterm{def:g12:reproduction-development:organo}{neurulation} forms the neural tube $\to$ CNS.

\textbf{Part IV.} (11)~$\mathrm{O_2}$, nutrients, wastes, some \omterm{def:g12:immunity:bcell}{antibodies};
large \omterm{def:g12:immunity:pathogen}{pathogens} or \omterm{def:g10:the-cell:cell}{cells} usually blocked (some viruses can cross).
(12)~\omterm{def:g12:reproduction-development:organo}{Organogenesis} has sensitive windows when toxins or poor nutrition
disrupt \omterm{def:g10:scientific-biology:properties}{development}. (13)~The \omterm{def:g12:reproduction-development:amniote}{amniotic egg} is a portable self-contained
package; the \omterm{def:g12:reproduction-development:placenta}{placenta} continuously uses maternal physiology for exchange and
protection.
\end{solution}
