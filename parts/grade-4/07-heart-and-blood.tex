\chapter{The Heart and the Blood}\label{ch:g4:heart-and-blood}

Press two fingers --- not the thumb --- on the inside of your
\omterm{def:g1:my-body:joint}{wrist}, below the thumb's base, and wait. There: a small, steady
knock, again and again, about once a second. That knock is your
\omterm{def:g4:heart-and-blood:heart}{heart} at work, felt from a distance. It has been beating since
before you were born, and it is the engine of the body's
delivery service.

\section{The blood, the body's delivery service}

\begin{proposition}[What the blood carries]\label{prop:g4:heart-and-blood:carries}
The \emph{blood}\index{blood} is a liquid that reaches every
corner of the body, and it is the body's carrier:
\begin{itemize}
  \item from the \omterm{def:g4:journey-of-food:tract}{small intestine}, it collects the
        \emph{\omterm{def:g4:journey-of-food:digestion}{nutrients}} (\cref{prop:g4:journey-of-food:absorb});
  \item from the \omterm{def:g4:breathing:organs}{lungs}, it collects \emph{oxygen}
        (\cref{prop:g4:breathing:exchange});
  \item it delivers both to every part of the body --- \omterm{def:g3:bones-and-muscles:muscle}{muscles},
        brain, \omterm{def:g3:bones-and-muscles:skeleton}{bones}, \omterm{def:g1:the-five-senses:sense}{skin};
  \item on the way back it carries off the wastes, delivering
        the carbon dioxide to the \omterm{def:g4:breathing:organs}{lungs} to be breathed out.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[One delivery round]\label{ex:g4:heart-and-blood:round}
Follow a drop of \omterm{prop:g4:journey-of-food:absorb}{blood}: through the \omterm{def:g4:breathing:organs}{lungs} it turns bright red,
loaded with oxygen; the \omterm{def:g4:heart-and-blood:heart}{heart} sends it out; in a running \omterm{def:g1:my-body:parts}{leg}
\omterm{def:g3:bones-and-muscles:muscle}{muscle} it hands over oxygen and \omterm{def:g4:journey-of-food:digestion}{nutrients}, takes on carbon
dioxide, and darkens; back at the \omterm{def:g4:heart-and-blood:heart}{heart}, it is sent to the
\omterm{def:g4:breathing:organs}{lungs} to unload and reload --- and out again. A full round
takes well under a minute.
\end{example}

\begin{definition}[Blood vessels]\label{def:g4:heart-and-blood:vessels}
\omterm{prop:g4:journey-of-food:absorb}{Blood} travels in tubes called \omterm{prop:g4:journey-of-food:absorb}{blood}
\emph{vessels}\index{blood vessel}: wide ones leaving and
entering the \omterm{def:g4:heart-and-blood:heart}{heart}, branching finer and finer --- down to
hair-thin vessels that thread between the body's parts, where
the deliveries are handed over. Laid end to end, one body's
vessels would stretch for tens of thousands of kilometres.
\end{definition}

\begin{remark}[Seeing the vessels]\label{rem:g4:heart-and-blood:see}
Look at the underside of your \omterm{def:g1:my-body:joint}{wrist}: bluish lines under the
\omterm{def:g1:the-five-senses:sense}{skin} --- \omterm{def:g4:heart-and-blood:vessels}{blood vessels} on their way back to the \omterm{def:g4:heart-and-blood:heart}{heart}. A
scraped \omterm{def:g1:my-body:joint}{knee} shows the other end of the network: even the
thinnest cut finds \omterm{prop:g4:journey-of-food:absorb}{blood}, because the finest \omterm{def:g4:heart-and-blood:vessels}{vessels} are
everywhere.
\end{remark}

\section{The heart, the pump}

\begin{definition}[The heart]\label{def:g4:heart-and-blood:heart}
The \emph{heart}\index{heart} is a hollow \omterm{def:g3:bones-and-muscles:muscle}{muscle}, about the
size of its owner's fist, in the middle of the chest behind
the \omterm{def:g3:bones-and-muscles:skeleton}{rib} cage. By squeezing --- a \emph{beat}\index{heartbeat}
--- it drives the \omterm{prop:g4:journey-of-food:absorb}{blood} through the \omterm{def:g4:heart-and-blood:vessels}{vessels}. It has two sides:
the right side pumps \omterm{prop:g4:journey-of-food:absorb}{blood} to the \emph{\omterm{def:g4:breathing:organs}{lungs}} to load oxygen;
the left side pumps the oxygen-rich \omterm{prop:g4:journey-of-food:absorb}{blood} to the \emph{whole
body}.
\end{definition}

\begin{proposition}[The pulse]\label{prop:g4:heart-and-blood:pulse}
Each \omterm{def:g4:heart-and-blood:heart}{beat} drives a little wave of \omterm{prop:g4:journey-of-food:absorb}{blood} along the \omterm{def:g4:heart-and-blood:vessels}{vessels}; felt
at the \omterm{def:g1:my-body:joint}{wrist} or the neck, that wave is the
\emph{pulse}\index{pulse}. Counting the \omterm{prop:g4:heart-and-blood:pulse}{pulse} counts the
\omterm{def:g4:heart-and-blood:heart}{heartbeats}: about $80$ to $100$ a minute for a child at rest,
$60$ to $80$ for an adult --- and much more during effort.
\end{proposition}
\admitted

\begin{example}[The heart keeps up with the legs]\label{ex:g4:heart-and-blood:effort}
Running, your \omterm{def:g1:my-body:parts}{leg} \omterm{def:g3:bones-and-muscles:muscle}{muscles} burn \omterm{def:g4:journey-of-food:digestion}{nutrients} and oxygen fast; the
deliveries must speed up. So the \omterm{def:g4:heart-and-blood:heart}{heart} \omterm{def:g4:heart-and-blood:heart}{beats} faster \emph{and}
harder --- a child's \omterm{prop:g4:heart-and-blood:pulse}{pulse} can pass $150$ in a sprint --- while
\omterm{def:g4:breathing:movements}{breathing} quickens to reload the \omterm{prop:g4:journey-of-food:absorb}{blood}
(\cref{ex:g4:breathing:running}). \omterm{def:g4:heart-and-blood:heart}{Heart} and \omterm{def:g4:breathing:organs}{lungs} are two
halves of one supply chain.
\end{example}

\begin{method}[Taking a pulse]\label{met:g4:heart-and-blood:pulse}
\begin{enumerate}
  \item Turn a hand palm up; place two fingers of the other
        hand on the \omterm{def:g1:my-body:joint}{wrist}, just below the thumb's base ---
        never the thumb, which has a little \omterm{prop:g4:heart-and-blood:pulse}{pulse} of its own;
  \item press gently until you feel the knocking;
  \item count the knocks for one minute (a clock with seconds
        helps);
  \item write the number down, with what the body was doing:
        resting, walking, after a sprint.
\end{enumerate}
\end{method}

\begin{remark}[A muscle that never rests --- almost]\label{rem:g4:heart-and-blood:rest}
Between two \omterm{def:g4:heart-and-blood:heart}{beats} the \omterm{def:g4:heart-and-blood:heart}{heart} relaxes for a fraction of a second
--- that sliver is all the rest it ever takes. Day and night,
about $100000$ \omterm{def:g4:heart-and-blood:heart}{beats} a day, for a whole life. Like every
\omterm{def:g3:bones-and-muscles:muscle}{muscle} (\cref{met:g3:bones-and-muscles:care}), it grows
stronger with exercise: a trained \omterm{def:g4:heart-and-blood:heart}{heart} moves more \omterm{prop:g4:journey-of-food:absorb}{blood} per
\omterm{def:g4:heart-and-blood:heart}{beat}, and so can \omterm{def:g4:heart-and-blood:heart}{beat} more calmly.
\end{remark}
