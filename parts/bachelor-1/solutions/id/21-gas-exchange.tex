\begin{solution}{exo:b1:gas-exchange:1}
$\dot V = K(A/x)(P_1 - P_2)$: luas yang lebih besar, sawar yang lebih
tipis, \omterm{def:b1:gas-exchange:partial}{tekanan parsial} luar yang lebih tinggi (ventilasi) dan \omterm{def:b1:gas-exchange:partial}{tekanan

\end{solution}

\begin{solution}{exo:b1:gas-exchange:2}
Tiga puluh kali lebih sedikit oksigen tiap liter (\qty{7}{} lawan
\qty{210}{mL/L}); delapan ratus kali lebih rapat dan lima puluh kali
lebih kental, sehingga mahal dipompa; oksigen berdifusi melewatinya
sepuluh ribu kali lebih lambat, sehingga mediumnya harus digerakkan
tepat melintasi permukaannya.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:3}
Arus searah: air dan darah sama-sama pergi pada \qty{12}{kPa};
ekstraksi $(21 - 12)/21 = \qty{43}{\%}$. \omterm{def:b1:gas-exchange:gill}{Arus lawan}: air pergi pada
\qty{4}{kPa}, darah pada \qty{18}{kPa}; ekstraksi $(21 - 4)/21 =
\qty{81}{\%}$.
\end{solution}

\end{solution}

\begin{solution}{exo:b1:gas-exchange:5}
Semenit \qty{6.75}{L/min}; \omterm{def:b1:gas-exchange:lung}{alveolus} $15\times 0.3 = \qty{4.5}{L/min}$;
serapan $4.5\times 0.07 = \qty{0.32}{L/min}$. Dangkal: semenit

bagi kerja bernapas yang sama; ruang matinya dibayar pada tiap napas.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:7}
$A/x$ turun sebesar $2\times 4 = 8$: \qty{31}{mL/min}. Untuk
memulihkan \qty{250}{}, bedanya harus \qty{64}{kPa} --- mustahil, sebab

menaikkannya dengan menghirup oksigen, dan itu pun tidak delapan kali.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:8}
\omterm{prop:b1:gas-exchange:transport}{Anhidrase karbonat} berada di dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya}, sehingga bikarbonatnya
dibuat di sana lalu bertimbun; ia berdifusi keluar menuruni landaiannya
lewat sebuah penukar yang memasukkan sebuah klorida bagi tiap
bikarbonat, sehingga muatannya tetap seimbang. Tanpa pertukaran itu,
bikarbonatnya akan tinggal di dalam bersama pasangan protonnya,
menaikkan osmolaritas \omterm{def:b1:cell-unit-of-life:cell}{selnya}, dan air akan masuk: \omterm{def:b1:cell-unit-of-life:cell}{selnya} akan
menggembung di \omterm{def:b1:body-plans-tissues:tissue}{jaringannya} (ia sedikit menggembung sebagaimana
adanya).
\end{solution}

\begin{solution}{exo:b1:gas-exchange:9}
Dorongan bernapas datang dari naiknya $\mathrm{CO_2}$. Bernapas
berlebihan menurunkan $\mathrm{CO_2}$ jauh di bawah normal tanpa
menambah banyak oksigen (hemoglobinnya sudah \omterm{def:b1:lipids:fattyacid}{jenuh}); selama
selamnya oksigennya turun ke aras pingsan sebelum $\mathrm{CO_2}$-nya
memanjat kembali ke aras yang memaksa sebuah napas. Tanpa bernapas
berlebihan, $\mathrm{CO_2}$ mencapai aras itu selagi oksigennya masih
berlimpah, dan penyelamnya muncul.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:10}
$A = \pi(10^{-5})^2 = \qty{3.1e-10}{m^2}$; $\Delta c = JL/DA =
10^{-9}\times 10^{-3}/(2\times 10^{-5}\times 3.1\times 10^{-10}) =
\qty{160}{mol/m^3}$ --- jauh lebih banyak daripada \qty{8.7}{} yang
tersedia, sehingga tabung sesempit ini tak dapat memasok ototnya lewat
difusi pada \qty{1}{mm}; sebuah tabung yang lebih lebar
(\qty{100}{\micro m}: $\Delta c = \qty{6.4}{mol/m^3}$, baru mungkin)
atau ventilasi aktif dibutuhkan. Pada \qty{10}{mm} tuntutannya sepuluh
kali lagi. Karena panjang tabungnya tumbuh bersama ukuran tubuhnya dan
tuntutannya bersama volumenya, difusi lewat \omterm{def:b1:flowering-plant-organization:tissues}{trakea} membatasi ukuran
serangga pada beberapa sentimeter, lebih daripada itu hanya dengan
pemompaan.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:11}
Ikan: air pada satu arah, darah berlawanan (\omterm{def:b1:gas-exchange:gill}{arus lawan}), ekstraksi
\qty{80}{\%}. Burung: udara pada satu arah lewat parabronkus, darah
bersiku-siku (arus silang), ekstraksinya di tengah tetapi lebih tinggi
daripada milik mamalia, dengan \omterm{prop:b1:organism-environment:surfaces}{permukaan pertukarannya} tak pernah
menerima udara basi. Sebuah \omterm{def:b1:gas-exchange:lung}{paru-paru} bolak-balik mencampur udara
segar dengan gas sisanya, sehingga \omterm{prop:b1:organism-environment:surfaces}{permukaan pertukarannya} paling baik
hanya melihat campuran \omterm{def:b1:gas-exchange:lung}{alveolusnya}, dan darahnya hanya dapat setimbang
dengan itu: berapa pun permukaannya, darah arterinya tak dapat
melampaui tekanan \omterm{def:b1:gas-exchange:lung}{alveolusnya}, dan tekanan \omterm{def:b1:gas-exchange:lung}{alveolusnya} tak dapat
mendekati tekanan terhirupnya selagi sebuah volume sisa masih ada.
\end{solution}

\begin{solution}{exo:b1:gas-exchange:12}
$\mathrm{CO_2}$ menjadi isyarat yang lebih baik sebab aras arterinya
menanggapi ventilasinya secara langsung dan curam (paruhkan
ventilasinya dan ia berlipat dua), ia diukur dengan saksama lewat pH
oleh reseptor pusatnya, dan oksigennya, yang duduk pada puncak datar
kurva hemoglobinnya, sedikit berubah sepanjang jangkau ventilasi yang
normal: sebuah termostat $\mathrm{CO_2}$ menjaga oksigennya tetap benar
sebagai hasil sampingan. Ia gagal ketika oksigennya turun lepas dari
$\mathrm{CO_2}$-nya --- di ketinggian, tempat oksigen terhirupnya
rendah tetapi produksi $\mathrm{CO_2}$-nya tidak, sehingga isyarat
$\mathrm{CO_2}$-nya berkata ``bernapaslah lebih sedikit'' sedangkan
oksigennya berkata ``bernapaslah lebih banyak''; hanya badan
karotisnya, ketika oksigennya sangat rendah, mengatasinya, dan
komprominya (bernapas berlebihan dengan alkalosis) memakan beberapa
hari penyesuaian ginjal untuk mereda.
\end{solution}

\begin{solution}{pb:b1:gas-exchange:1}
\textbf{1.} Udara $210/22.4 = \qty{9.4}{mmol/L}$; air $7/22.4 =
\qty{0.31}{mmol/L}$.
\textbf{2.} 30.
\textbf{3.} Udara: $1.2/9.4 = \qty{0.13}{g}$; air: $1000/0.31 =
\qty{3200}{g}$ --- \num{25000} kali lebih banyak massa.
\textbf{4.} $210/7 = \qty{30}{L}$ air.
\textbf{5.} Air terlalu berat dan terlalu miskin oksigen untuk didorong
masuk lalu keluar; ia harus dilewatkan sekali saja, yang juga membuat
\omterm{def:b1:gas-exchange:gill}{arus lawannya} mungkin; udara cukup murah untuk digerakkan dua kali.
\textbf{6.} Semenit $12\times 0.5 = \qty{6}{L/min}$; \omterm{def:b1:gas-exchange:lung}{alveolus}
$12\times 0.35 = \qty{4.2}{L/min}$.
\textbf{7.} $150/500 = \qty{30}{\%}$ tiap napas, dan ventilasi semenitnya.
\textbf{8.} Yang diantarkan $4.2\times 0.21 = \qty{0.88}{L/min}$;
serapan $4.2\times 0.07 = \qty{0.29}{L/min}$, dekat \qty{250}{mL/min}.
\textbf{9.} $0.25/(6\times 0.21) = \qty{20}{\%}$ (\qty{28}{\%} dari yang
mencapai \omterm{def:b1:gas-exchange:lung}{alveolusnya}).
\textbf{10.} $0.14\times(101 - 6.3) = \qty{13.3}{kPa}$.
\textbf{11.} $250/5 = \qty{50}{mL/L}$; vena \qty{150}{mL/L},
\qty{75}{\%} \omterm{def:b1:lipids:fattyacid}{jenuh}.
\textbf{12.} $3000/20 = \qty{150}{mL/L}$; vena \qty{50}{mL/L},
\qty{25}{\%} \omterm{def:b1:lipids:fattyacid}{jenuh}.
\textbf{13.} $50/60 = \qty{0.83}{mL/min}$.
\textbf{14.} $0.83/(7\times 0.8) = \qty{0.149}{L/min}$, \qty{8.9}{L}
tiap jam.
\textbf{15.} $0.83/(7\times 0.43) = \qty{0.28}{L/min}$: \omterm{def:b1:gas-exchange:gill}{arus lawannya}
memaruhkan air yang harus dipompa, dan kerjanya.
\textbf{16.} \qty{8.9}{kg} air tiap jam; manusianya menggerakkan $6\times
60\times 1.2 = \qty{430}{g}$ udara tiap jam --- seperdua puluh
massanya, bagi tiga ratus kali oksigennya ($15\,000$ lawan
\qty{50}{mL} sejam).
\textbf{17.} $100\times(0.95 - 0.15) = \qty{80}{mL/L}$; curah jantung
$0.83/0.080 = \qty{10.4}{mL/min}$.
\textbf{18.} Air/darah $149/10.4 = 14$; udara/darah $4.2/5 = 0.84$.
Ikannya harus melewatkan empat belas volume air tiap volume darah
sebab masing-masing memuat begitu sedikit oksigen; manusianya, sekitar
satu.
\textbf{19.} Serapan $\times 1.5$ dari air yang memuat $5.5/7 = 0.79$
sebanyak itu: ventilasi $\times 1.9$; biaya bernapasnya, yang kira-kira
sebanding dengan alirnya (atau lebih, sebab ia naik lebih cepat
daripada linear), beranjak dari \qty{10}{\%} menjadi sekurangnya
\qty{13}{\%} sebuah serapan yang lebih besar --- ikannya bekerja lebih
keras untuk bernapas dalam air hangat.
\textbf{20.} Ujung masuk airnya: $21 - 18 = \qty{3}{kPa}$; ujung keluar
airnya: $4 - 3 = \qty{1}{kPa}$: positif pada kedua ujungnya.
\textbf{21.} Ujung masuknya $21 - 3 = \qty{18}{kPa}$; ujung keluarnya $12 - 12 =
0$: kedua fluidanya telah setimbang dan tak ada landaian yang tersisa,
walaupun airnya masih memuat \qty{57}{\%} oksigennya.
\textbf{22.} Arus searahnya: darahnya setimbang dengan sebuah gas
\omterm{def:b1:gas-exchange:lung}{alveolus} tunggal yang tercampur baik lalu pergi pada tekanannya; ia tak
dapat melampaui \qty{13.3}{kPa} sebab tak ada gas yang lebih segar di
hilirnya untuk ditemui, sebagaimana ada dalam \omterm{def:b1:gas-exchange:gill}{arus lawannya}.
\textbf{23.} Trout $2000/(2\times 10^{-4}) = \num{1e7}$ (cm$^2$ tiap cm);
manusia $10^6/(5\times 10^{-5}) = \num{2e10}$: nisbahnya \num{2000}.
Serapannya $250/0.83 = 300$. Troutnya mengambil tujuh kali lebih banyak
oksigen tiap satuan $A/x$: beda tekanan puratanya melintasi lamelanya
tentu lebih besar --- yang disediakan \omterm{def:b1:gas-exchange:gill}{arus lawannya}, yang memelihara
sebuah landaian sepanjang seluruh permukaannya, sedangkan landaian
\omterm{def:b1:gas-exchange:lung}{paru-paru} bolak-balik kecil menjelang akhir kesetimbangannya.
\textbf{24.} $\mathrm{CO_2}$ dua puluh lima kali lebih larut dalam air
daripada oksigen: airnya membawanya pergi secepat ia dihasilkan, dan
$\mathrm{CO_2}$ darah seekor ikan tinggal sangat rendah.
\textbf{25.} \omterm{def:b1:gas-exchange:gill}{Insangnya} sekitar \qty{80}{\%}, \omterm{def:b1:gas-exchange:lung}{paru-parunya} sekitar
\qty{20}{\%} oksigen dalam medium terhirupnya; alir satu arah \omterm{def:b1:gas-exchange:gill}{insangnya}
dan susunan \omterm{def:b1:gas-exchange:gill}{arus lawan} air dan darahnya membuat perbedaannya.
\end{solution}
