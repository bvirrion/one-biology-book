\section*{Bab \ref{ch:b1:cell-unit-of-life} --- Sel: Satuan Kehidupan}

\begin{solution}{exo:b1:cell-unit-of-life:1}
Semua \omterm{def:b1:organism-environment:organism}{organisme} tersusun dari \omterm{def:b1:cell-unit-of-life:cell}{sel} (tiap \omterm{def:b1:body-plans-tissues:tissue}{jaringan} yang diperiksa di bawah
mikroskop, tumbuhan maupun hewan, memperlihatkannya); tiap \omterm{def:b1:cell-unit-of-life:cell}{sel} berasal
dari \omterm{def:b1:cell-unit-of-life:cell}{sel} (pembelahan teramati pada embrio dan alga; labu Pasteur
menyingkirkan generasi spontan); bahan keturunannya diwariskan pada
pembelahan (kromosom diikuti sepanjang mitosis).
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:2}
\omterm{def:b1:cell-unit-of-life:prokeuk}{Inti} (tidak ada / ada), \omterm{def:b1:cell-unit-of-life:prokeuk}{organel} berbatas membran (tidak ada / ada),
kromosom (satu melingkar / beberapa lurus), ukuran
(\qtyrange{0.5}{5}{\micro m} / \qtyrange{10}{100}{\micro m}), dinding
\omterm{def:b1:cell-unit-of-life:cell}{sel} dari peptidoglikan (bakteri) / dari selulosa, kitin atau tidak ada,
rangka \omterm{def:b1:cell-unit-of-life:cell}{sel} (sederhana / rumit).
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:3}
$\qty{1}{mm}/\qty{1}{\micro m} = 1000$ bakteri; $1000/20 = 50$ \omterm{def:b1:cell-unit-of-life:cell}{sel}
hewan.
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:4}
\omterm{def:b1:cell-unit-of-life:resolution}{Daya pisahnya} bergantung pada panjang gelombang dan apertura numerik,
bukan pada \omterm{def:b1:cell-unit-of-life:resolution}{perbesarannya}; melewati sekitar $1000\times$ bayangannya
\omterm{def:b1:cell-unit-of-life:resolution}{memperbesar} kaburnya batas difraksi (\qty{0.2}{\micro m}) tanpa
memisahkan apa pun yang baru.
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:5}
$0.61\times 550/0.65 = \qty{516}{nm}$; dalam cahaya biru, \qty{422}{nm}.
Dua bakteri berjarak \qty{500}{nm}: nyaris, dalam cahaya biru; ribosom
berjarak \qty{30}{nm}: tidak, dengan faktor lima belas.
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:6}
Ungu: dinding peptidoglikan tebal yang menahan kompleks ungu
kristal--iodin, tanpa membran luar (Gram positif). Merah muda:
peptidoglikan tipis di bawah sebuah membran luar, zat warnanya terbilas
keluar, lalu diwarnai lawan (Gram negatif). \emph{E. coli} adalah \omterm{def:b1:flowering-plant-organization:organs}{batang}
Gram negatif: \omterm{def:b1:flowering-plant-organization:organs}{batang} yang merah muda.
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:7}
Mitokondria: $v = \frac{2}{9}\times(5\times 10^{-7})^2\times 100\times
9.81\times 10^4/10^{-3} = \qty{5.4e-4}{m/s}$; \qty{5}{cm} dalam
\qty{92}{s}. \omterm{def:b1:cell-unit-of-life:prokeuk}{Inti} pada $1000\,g$: $v = \frac{2}{9}\times(4\times
10^{-6})^2\times 100\times 9810/10^{-3} = \qty{3.5e-3}{m/s}$; \qty{5}{cm}
dalam \qty{14}{s}.
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:8}
Suksinat dehidrogenase bersifat mitokondria: \qty{80}{\%} mitokondrianya
berada dalam endapan 2; \qty{15}{\%} dalam endapan 1 adalah mitokondria
yang terperangkap dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} utuh dan serpihan atau terseret turun
bersama \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}; \qty{5}{\%} dalam supernatannya berasal dari
mitokondria yang pecah selama penghomogenan. Laktat dehidrogenase
bersifat sitosol dan terlarut: ia tinggal dalam supernatan akhir.
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:9}
(a) Mikroskopi cahaya kontras fase pada \omterm{def:b1:cell-unit-of-life:cell}{sel} hidup (prosesnya berlangsung
beberapa menit dan menuntut \omterm{def:b1:cell-unit-of-life:cell}{sel} yang hidup). (b) Mikroskopi cahaya atas
sayatan terwarnai, atau TEM untuk kepastian (mitokondria berada pada
batas cahaya). (c) TEM (liangnya \qty{100}{nm}). (d) SEM (relief
permukaan).
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:10}
Labu yang jernih menyingkirkan kemungkinan bahwa udara itu sendiri, atau
kaldu yang telah dididihkan itu sendiri, melahirkan kehidupan: udara
masuk dengan bebas lewat lehernya yang terbuka. Labu yang keruh, setelah
bersentuhan dengan debunya, memperlihatkan bahwa apa yang telah
diperangkap lehernya --- partikel dari udara --- adalah sumber
makhluknya. Bersama-sama keduanya membuktikan bahwa mikroba berasal dari
mikroba yang dibawa debu. Labu tersegel yang dididihkan masih membiarkan
bantahan bahwa penyegelannya telah menyingkirkan udara yang dibutuhkan
sebuah ``daya hidup''; leher angsanya meloloskan udara dan hanya
menyingkirkan debu.
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:11}
$10^5 = 2^{n}$ memberi $n = 16.6$ kali lipat dua, yakni \qty{415}{min},
sekitar
\qty{7}{h}. $10^8\times 10^3\,\mathrm{mL}\times\qty{1}{pg} = \qty{0.1}{g}$
tiap liter. Pertumbuhannya berhenti karena oksigen (yang hanya dapat
dipasok mediumnya beberapa miligram tiap liter), sebuah mineral, atau
penumpukan asam dan sisanya menjadi pembatas, bukan glukosanya.
\end{solution}

\begin{solution}{exo:b1:cell-unit-of-life:12}
Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} yang sepuluh kali lebih besar pada tiap arah memiliki
sepersepuluh permukaan membran tiap satuan volume; karena membran
membawa mesin pengubah energi dan pengangkutnya, \omterm{def:b1:cell-unit-of-life:cell}{sel} yang besar itu akan
kelaparan. Eukariot melipatgandakan membrannya dengan melipatnya ke
dalam dirinya (retikulum endoplasma, membran dalam mitokondria,
tilakoid) dan dengan demikian juga menciptakan ruang tertutup tempat
proses yang tak sejalan --- pencernaan asam, kimia oksidatif,
penyimpanan DNA --- berjalan terpisah pada keadaannya sendiri.
Gambarannya tepat sejauh menyangkut membran, dan belum lengkap: eukariot
juga memperoleh rangka \omterm{def:b1:cell-unit-of-life:cell}{sel} dan sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti}, dan mitokondrianya adalah
keturunan bakteri yang tertelan, bukan lipatan.
\end{solution}

\begin{solution}{pb:b1:cell-unit-of-life:1}
\textbf{1.} Endapan 1: \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} (dan \omterm{def:b1:cell-unit-of-life:cell}{sel} utuh, serpihan). Endapan 2:
mitokondria (dengan lisosom dan peroksisom). Endapan 3: mikrosom
(serpihan retikulum endoplasma) dan ribosom.
\textbf{2.} Dingin memperlambat enzimnya (termasuk protease dan
fosfolipase yang terlepas oleh pemecahan) dan mengawetkan \omterm{def:b1:cell-unit-of-life:prokeuk}{organelnya};
isotonik agar \omterm{def:b1:cell-unit-of-life:prokeuk}{organelnya} tidak membengkak dan pecah maupun mengerut.
\textbf{3.} Mitokondria (membran dalam); sitosol.
\textbf{4.} $500 + 200 + 300 + 950 = \qty{1950}{mg}$, \qty{97.5}{\%}:
kehilangan pada dinding tabung dan dalam pemindahan.
\textbf{5.} SDH: $15 + 60 + 5 + 4 = 84$, \qty{84}{\%} (sebagian
aktivitasnya hilang oleh kerusakan mitokondria). LDH:
$20 + 12 + 8 + 350 = 390$,
\qty{97.5}{\%}.
\textbf{6.} $2000/10 = \qty{200}{mg/g}$: \qty{20}{\%} massa segarnya (selebihnya sebagian besar air).
\textbf{7.} $v = \frac{2}{9}\times(5\times 10^{-7})^2\times 100\times
\num{9.81e4}/10^{-3} = \qty{5.4e-4}{m/s}$; \qty{5}{cm} dalam
\qty{92}{s}: \qty{20}{min} lebih dari cukup.
\textbf{8.} Pada $1000\,g$, $v = \qty{5.4e-5}{m/s}$; dalam \qty{600}{s},
\qty{3.3}{cm}. Tersebar merata sepanjang \qty{5}{cm}, mitokondria yang
berada dalam \qty{3.3}{cm} dari dasarnya sampai ke sana: \qty{65}{\%},
seandainya tak ada hal lain yang campur tangan.
\textbf{9.} Hanya \qty{15}{\%} yang benar-benar mengendap: sebuah
endapan yang nyata tidak memerangkap segala yang sampai kepadanya (\omterm{def:b1:cell-unit-of-life:prokeuk}{inti}
yang memadat disuspensikan kembali dengan lembut dan lapisan atasnya
yang longgar tinggal dalam supernatan), dan rumusnya menaksir terlalu
tinggi laju \omterm{def:b1:cell-unit-of-life:prokeuk}{organel} yang tidak bulat dan berair. Mitokondria di dalam
\omterm{def:b1:cell-unit-of-life:cell}{sel} yang belum pecah dan yang melekat pada \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} juga masuk ke endapan 1.
\textbf{10.} $v = \frac{2}{9}\times(4\times 10^{-6})^2\times 100\times
9810/10^{-3} = \qty{3.5e-3}{m/s}$; \qty{14}{s}.
\textbf{11.} $v = \frac{2}{9}\times(1.2\times 10^{-8})^2\times 600\times
\num{9.81e5}/10^{-3} = \qty{1.9e-5}{m/s}$; \qty{5}{cm} dalam
\qty{2650}{s}, \qty{44}{min}. Pada $\num{10000}\,g$ lajunya
\qty{1.9e-6}{m/s}: dalam \qty{20}{min}, \qty{2.3}{mm}, sehingga
ribosomnya tinggal dalam supernatan 2.
\textbf{12.} Lajunya sebanding dengan $r^2$: pada gaya yang rendah hanya
partikel terbesar yang bergerak berarti, sehingga tiap tahap
mengendapkan satu golongan dan menyisakan yang lebih kecil. Terlalu
singkat, golongan itu belum sepenuhnya mengendap; terlalu lama,
golongan berikutnya mulai mencemari endapannya.
\textbf{13.} \omterm{def:b1:cell-unit-of-life:fractionation}{Homogenat} $100/2000 = \qty{0.05}{U/mg}$; endapan 1
$15/500 = 0.03$; endapan 2 $60/200 = 0.30$; endapan 3 $5/300 = 0.017$;
supernatan $4/950 = 0.004$.
\textbf{14.} Pemurnian $0.30/0.05 = 6$; rendemen $60/100 =
\qty{60}{\%}$.
\textbf{15.} $0.017/0.05 = 0.33$: endapan 3 miskin mitokondria; sedikit aktivitas yang dipegangnya berasal dari serpihan membran mitokondria yang dihasilkan penghomogenan, yang mengendap bersama mikrosomnya.
\textbf{16.} LDH: \omterm{def:b1:cell-unit-of-life:fractionation}{homogenat} $400/2000 = \qty{0.2}{U/mg}$; supernatan
$350/950 = \qty{0.37}{U/mg}$; pemurnian $1.8$; rendemen \qty{87.5}{\%}.
\textbf{17.} Sitosol adalah separuh protein \omterm{def:b1:cell-unit-of-life:cell}{selnya}, sehingga sitosol
yang murni sekalipun hanya dapat melipatduakan aktivitas jenisnya;
mitokondrianya seperenam atau kurang, sehingga mengasingkannya
memekatkan enzimnya enam kali atau lebih. Rendemennya tinggi bagi enzim
yang terlarut karena ia tidak ikut hilang bersama \omterm{def:b1:cell-unit-of-life:prokeuk}{organel} yang rusak.
\textbf{18.} Tidak: endapan 2 memegang \qty{200}{mg} protein dan
memerangkap supernatan di antara partikelnya; \qty{12}{U} adalah
\qty{3}{\%} totalnya, yang sesuai dengan sitosol yang terperangkap. Uji:
cucilah endapan itu (suspensikan kembali lalu pusingkan lagi): enzim
yang terperangkap akan pergi, enzim yang terikat akan tinggal.
\textbf{19.} Bila endapan 2 mengandung \qty{60}{\%} mitokondria, maka
mitokondria murni memiliki $0.30/0.6 = \qty{0.5}{U/mg}$; \qty{100}{U}
hati itu lalu sepadan dengan \qty{200}{mg} protein mitokondria, yakni
\qty{10}{\%} dari \qty{2000}{mg}.
\textbf{20.} TEM atas endapan yang terfiksasi, tertanam, tersayat tipis
dan terwarnai logam berat: potongan berbentuk kacang dengan membran
rangkap dan krista, dengan beberapa lisosom (benda padat) dan peroksisom
di antaranya.
\textbf{21.} Lihatlah setetes endapan yang disuspensikan kembali di bawah mikroskop kontras fase dan hitunglah \omterm{def:b1:cell-unit-of-life:cell}{sel} utuh di antara \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} yang bebas; bila banyak, homogenkan lebih saksama (lebih banyak sapuan, alu yang lebih rapat) lalu pusingkan lagi, sebab \omterm{def:b1:cell-unit-of-life:prokeuk}{organel} yang terperangkap menurunkan tiap rendemen.
\textbf{22.} \omterm{def:b1:cell-unit-of-life:fractionation}{Sentrifugasi bertingkat} memisahkan menurut ukuran (laju
$\propto r^2$) dan ketiga \omterm{def:b1:cell-unit-of-life:prokeuk}{organel} itu berukuran serupa; dalam sebuah
gradien massa jenis sebuah partikel berhenti pada massa jenisnya sendiri
tanpa memandang ukurannya, dan ketiga massa jenis itu berbeda.
\textbf{23.} Lisosom yang utuh menahan enzimnya di dalam sebuah membran
yang tak dilewati substratnya (latensi); deterjennya memecah membran itu
dan memaparkannya. Karena itu lisosom dalam endapan itu sebagian besar
utuh.
\textbf{24.} Air bersifat hipotonik: \omterm{def:b1:cell-unit-of-life:prokeuk}{organelnya} membengkak dan pecah.
Mitokondria dan lisosomnya robek, sehingga endapan 2 menyusut; suksinat
dehidrogenase (yang terikat membran) sebagian pergi ke endapan 3 sebagai
serpihan membran; asam fosfatase (yang terlarut di dalam lisosom)
terlepas ke dalam supernatan akhir.
\textbf{25.} Faktor pemurnian $6$, rendemen \qty{60}{\%}. Faktornya
dibatasi pencemaran (lisosom, peroksisom, sitosol yang terperangkap:
endapan 2 mengandung \qty{60}{\%} mitokondria); rendemennya dibatasi
mitokondria yang hilang ke endapan 1 (terperangkap) dan yang pecah
selama penghomogenan.
\end{solution}
