\chapter{Gigi dan Makan dengan Baik}\label{ch:g2:teeth-and-eating-well}

Gigitlah sebuah apel lalu amati bekasnya: sederet rapi jejak kecil yang
melengkung. Gigimulah yang membuatnya --- satu kotak perkakas lengkap
yang memotong, merobek dan menggilas setiap santapan sebelum kamu
menelannya. Dan seperti setiap kotak perkakas yang baik, ia layak
dirawat, dan layak diberi hal-hal yang baik untuk dikerjakan.

\section{Gigi: kotak perkakas di dalam mulut}

\begin{definition}[Tiga macam gigi]\label{def:g2:teeth-and-eating-well:kinds}
Gigi manusia ada tiga macam, masing-masing dengan tugasnya:
\begin{enumerate}
  \item \emph{gigi seri}\index{gigi seri}, pipih dan tajam di bagian
        depan, \emph{memotong} --- ia mengiris apelnya;
  \item \emph{gigi taring}\index{gigi taring}, runcing, di sudutnya,
        \emph{merobek} --- ia mencengkeram dan menarik makanan yang
        lebih liat;
  \item \emph{gigi geraham}\index{gigi geraham}, lebar dan berbenjol di
        bagian belakang, \emph{menggilas} --- ia melumatkan makanan
        menjadi bubur yang siap ditelan.
\end{enumerate}
\end{definition}

\begin{example}[Satu gigitan apel, tiga perkakas]\label{ex:g2:teeth-and-eating-well:apple}
Memakan sepotong apel memakai seluruh kotak perkakasnya secara
berurutan: \omterm{def:g2:teeth-and-eating-well:kinds}{gigi seri} mengiris gigitannya, \omterm{def:g2:teeth-and-eating-well:kinds}{gigi taring} menolong
menariknya lepas, dan lidah mengantarnya ke belakang menuju \omterm{def:g2:teeth-and-eating-well:kinds}{gigi
geraham}, yang menggilasnya. Baru sesudah itu kamu menelan.
\end{example}

\begin{remark}[Ingat hewan-hewannya]\label{rem:g2:teeth-and-eating-well:animals}
Kotak perkakas campuranmu menceritakan sebuah kisah: pemotong dan
penggilas yang pipih seperti seorang \omterm{def:g2:what-animals-eat:kinds}{herbivora}, \omterm{def:g2:teeth-and-eating-well:kinds}{gigi taring} yang
runcing seperti seorang \omterm{def:g2:what-animals-eat:kinds}{karnivora}. Ini mulut seorang \omterm{def:g2:what-animals-eat:kinds}{omnivora} ---
manusia diperlengkapi untuk \omterm{def:g1:plants-around-us:plant}{tumbuhan} \emph{dan} daging, persis seperti
pengelompokan \cref{ch:g2:what-animals-eat} atas kita.
\end{remark}

\section{Gigi susu, gigi tetap}

\begin{proposition}[Dua perangkat dalam satu kehidupan]\label{prop:g2:teeth-and-eating-well:sets}
Manusia menumbuhkan dua perangkat gigi. Ke-$20$ \emph{gigi
susu}\index{gigi susu} datang pada tahun-tahun pertama kehidupan.
Mulai kira-kira umur enam tahun, gigi itu goyah lalu tanggal satu demi
satu, didorong dari bawah oleh \emph{gigi tetap}\index{gigi tetap} ---
perangkat terakhir, sampai $32$ buah banyaknya, yang harus bertahan
sepanjang sisa hidupmu.
\end{proposition}
\admitted

\begin{example}[Gigi seri yang goyah]\label{ex:g2:teeth-and-eating-well:wobbly}
Sebuah gigi depan mulai goyah tepat pada umur anak-anak mulai
bersekolah. Di bawahnya, sebuah \omterm{def:g2:teeth-and-eating-well:kinds}{gigi seri} tetap sudah tumbuh dengan
diam dan melarutkan \omterm{def:g1:plants-around-us:parts}{akar} \omterm{prop:g2:teeth-and-eating-well:sets}{gigi susunya}, sampai gigi kecil itu melepas.
Celahnya bukan kecelakaan --- itu sebuah pergantian.
\end{example}

\begin{remark}[Tidak ada perangkat ketiga]\label{rem:g2:teeth-and-eating-well:nothird}
Sesudah perangkat \omterm{prop:g2:teeth-and-eating-well:sets}{gigi tetap}, tidak akan pernah datang gigi baru. \omterm{prop:g2:teeth-and-eating-well:sets}{Gigi
susu} yang tanggal digantikan; \omterm{prop:g2:teeth-and-eating-well:sets}{gigi tetap} yang hilang, hilang untuk
selamanya. Itulah alasan sebenarnya orang dewasa mendesakkan
penyikatan: kotak perkakas kedua harus bertahan tujuh puluh tahun atau
lebih.
\end{remark}

\section{Makan dengan baik}

\begin{proposition}[Piring yang seimbang]\label{prop:g2:teeth-and-eating-well:balanced}
Untuk tumbuh dan tetap sehat, tubuh membutuhkan makanan dari setiap
keluarga, setiap hari:
\begin{itemize}
  \item buah dan sayur --- untuk vitamin, pada setiap santapan;
  \item roti, serealia, kentang --- untuk tenaga yang bertahan lama;
  \item \omterm{def:g2:animals-grow-up:born}{susu}, yogurt, keju --- untuk membangun tulang dan gigi;
  \item daging, ikan atau \omterm{def:g2:animals-grow-up:born}{telur} --- untuk membangun tubuh, sekali atau
        dua kali sehari sudah cukup;
  \item air --- satu-satunya minuman yang sungguh dibutuhkan tubuh;
  \item kudapan manis dan berlemak --- untuk kesenangan, dalam jumlah
        kecil, bukan pada setiap santapan.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Dua kantong bekal]\label{ex:g2:teeth-and-eating-well:snack}
Bekal A: sebuah apel, sepotong roti, air. Bekal B: permen, sebotol
minuman bersoda. Bekal B terasa gula dan nyaris tidak ada yang lain
--- dan gula justru yang dipestakan \omterm{def:g1:healthy-habits:germs}{kuman}
\cref{ch:g1:healthy-habits}. Bekal A memberi makan tubuhnya dan
memberi gigi pekerjaan yang jujur. Terkalah bekal mana yang dikemas
dokter gigi untuk anak-anaknya sendiri.
\end{example}

\begin{method}[Memeriksa daftar makanan sehari]\label{met:g2:teeth-and-eating-well:check}
Ambil santapan sehari lalu periksa:
\begin{enumerate}
  \item apakah ada buah atau sayur pada setiap santapan?
  \item apakah ada sesuatu dari keluarga pembangun --- produk \omterm{def:g2:animals-grow-up:born}{susu},
        dan daging, ikan atau \omterm{def:g2:animals-grow-up:born}{telur}?
  \item apakah air menjadi minuman sehari-hari?
  \item apakah kudapan manis tetap sedikit dan sesekali saja?
\end{enumerate}
Kalau ada keluarga yang hilang, santapan berikutnya adalah kesempatan
untuk mengundangnya kembali.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:1}
Sebutkan tiga macam gigi dan tugas masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:2}
Gigi mana yang ada di bagian depan mulut, dan mana yang di belakang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:3}
Ikuti sepotong gigitan wortel melintasi mulut: gigi mana mengerjakan
apa, secara berurutan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:4}
Ada berapa banyak \omterm{prop:g2:teeth-and-eating-well:sets}{gigi susu}? Apa yang terjadi padanya mulai kira-kira
umur enam tahun?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:5}
Mengapa \omterm{prop:g2:teeth-and-eating-well:sets}{gigi tetap} harus dirawat lebih baik lagi daripada \omterm{prop:g2:teeth-and-eating-well:sets}{gigi susu}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:6}
Sebutkan keluarga makanan yang seharusnya ada pada hari yang sehat.
Minuman mana yang sungguh dibutuhkan tubuh?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:7}
Bandingkan kedua bekal \cref{ex:g2:teeth-and-eating-well:snack}: apa
yang diberikan bekal A kepada tubuh yang tidak diberikan bekal B?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:8}
Cobalah: amati gigimu sendiri di cermin. Temukan sebuah \omterm{def:g2:teeth-and-eating-well:kinds}{gigi seri},
sebuah \omterm{def:g2:teeth-and-eating-well:kinds}{gigi taring} dan sebuah \omterm{def:g2:teeth-and-eating-well:kinds}{gigi geraham}, lalu cocokkan bentuknya
dengan gambarnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:teeth-and-eating-well:9}
Dengan memakai \cref{rem:g2:teeth-and-eating-well:animals}, jelaskan
apa yang dikatakan bentuk gigi manusia tentang daftar makanan manusia
--- dan \omterm{def:g1:animals-around-us:animal}{hewan} mana pada \cref{ch:g2:what-animals-eat} yang kita
serupai.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:teeth-and-eating-well:10}
Hari Lila: sarapan --- serealia dan \omterm{def:g2:animals-grow-up:born}{susu}; makan siang --- pasta dan
permen; kudapan --- minuman bersoda dan biskuit; makan malam --- pizza.
Jalankan \cref{met:g2:teeth-and-eating-well:check} atasnya: apa yang
hilang, dan apa yang terlalu sering muncul?
\end{exercise}
