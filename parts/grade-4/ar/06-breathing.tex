\chapter{التنفس}\label{ch:g4:breathing}

احبس نفَسك وابدأ العدّ. فقبل الدقيقة بكثير يطالب صدرك، بإلحاح
متزايد، بأن يتحرك من جديد. تستطيع أن تفوّت وجبة، \omterm{prop:g1:healthy-habits:needs}{والنوم} يمكن أن
ينتظر إلى الليل --- أما الهواء فمطلوب \emph{الآن}، على مدار
الساعة، نحو خمس عشرة مرة في الدقيقة. فما هذا الأمر العاجل في نفَس
من الهواء؟

\section{حركتا التنفس}

\begin{definition}[الشهيق والزفير]\label{def:g4:breathing:movements}
\emph{التنفس}\index{تنفس} تناوب لا ينتهي بين حركتين:
\begin{itemize}
  \item \emph{الشهيق} (استنشاق)\index{شهيق}: يتسع الصدر فيدخل
        الهواء من الأنف أو الفم؛
  \item و\emph{الزفير} (إخراج)\index{زفير}: يرتخي الصدر فيخرج
        الهواء راجعًا.
\end{itemize}
وفي الراحة يتكرر الزوجان نحو $15$ مرة في الدقيقة --- وأسرع عند
الرضّع، وأسرع في الجهد --- من غير أن تحتاج قط إلى التفكير في ذلك.
\end{definition}

\begin{example}[مشاهدة الحركتين]\label{ex:g4:breathing:watch}
ضع كفّيك مبسوطتين على أضلاعك: فعند \omterm{def:g4:breathing:movements}{الشهيق} يرتفع القفص الصدري
ويتسع تحت راحتيك؛ وعند \omterm{def:g4:breathing:movements}{الزفير} يهبط راجعًا. وخاصرة قطة نائمة تُظهر
الارتفاع والهبوط نفسهما. والحركتان لا تتوقفان --- في اليقظة
\omterm{prop:g1:healthy-habits:needs}{والنوم}، من أول صرخة إلى آخر يوم.
\end{example}

\section{إلى أين يذهب الهواء}

\begin{definition}[أعضاء التنفس]\label{def:g4:breathing:organs}
يسافر الهواء المستنشَق عبر الأنف (أو الفم)، نازلًا في
\emph{القصبة الهوائية}\index{قصبة هوائية} --- وهي الأنبوب الحلقي
المتين الذي تحسّه في مقدَّم عنقك --- ثم تنقسم إلى فرعين، فرع لكل
\emph{رئة}\index{رئة}. والرئتان عضوان إسفنجيان يملآن القفص الصدري
(خلف \omterm{def:g3:bones-and-muscles:skeleton}{أضلاع} \cref{def:g3:bones-and-muscles:skeleton})، وينتهيان
بملايين الجيوب الهوائية الصغيرة، كل واحد ملفوف بأوعية دموية دقيقة.
\end{definition}

\begin{remark}[إسفنجيتان عن قصد]\label{rem:g4:breathing:sponge}
ملايين الجيوب بدل كيسين بسيطين: إنها حيلة \omterm{def:g4:journey-of-food:tract}{المعي الدقيق} من جديد
(\cref{rem:g4:journey-of-food:surface}). فجدران تلك الجيوب كلها
تتجمع في مساحة لقاء هائلة بين الهواء \omterm{prop:g4:journey-of-food:absorb}{والدم} --- بقدر نصف ملعب
تنس، مطوية في صدرك.
\end{remark}

\section{ما الغاية من التنفس}

\begin{proposition}[أخذ الأكسجين وردّ ثاني أكسيد الكربون]\label{prop:g4:breathing:exchange}
الهواء مزيج من الغازات، والجسم يريد واحدًا منها:
\emph{الأكسجين}\index{أكسجين}. ففي جيوب الرئتين ينتقل أكسجين
الهواء الطازج إلى \omterm{prop:g4:journey-of-food:absorb}{الدم}، فيوصّله إلى كل جزء من الجسم؛ وهناك
يُستعمل، مع \omterm{def:g4:journey-of-food:digestion}{المغذيات}، لتحرير \omterm{prop:g3:balanced-diet:jobs}{الطاقة} في
\cref{prop:g3:balanced-diet:jobs}. أما غاز الجسم الفضلة،
\emph{ثاني أكسيد الكربون}\index{ثاني أكسيد الكربون}، فيقوم برحلة
العودة ويُزفر. وهواء \omterm{def:g4:breathing:movements}{الزفير} أقل أكسجينًا وأكثر ثاني أكسيد كربون من
هواء \omterm{def:g4:breathing:movements}{الشهيق} --- وذلك الفرق هو \emph{غاية} \omterm{def:g4:breathing:movements}{التنفس}.
\end{proposition}
\admitted

\begin{example}[لماذا يقطع الجري أنفاسك]\label{ex:g4:breathing:running}
حين تعدو تحرق رجلاك \omterm{prop:g3:balanced-diet:jobs}{الطاقة} بسرعة --- فتحتاجان إلى الأكسجين بسرعة.
فيلبّي \omterm{def:g4:breathing:movements}{التنفس}: من نحو $15$ نفَسًا هادئًا في الدقيقة إلى $40$ أو
أكثر، عميقة، بينما يسرع القلب (ويحكي
\cref{ch:g4:heart-and-blood} جانبه) ليعجّل التوصيلات. واللهاث بعد
السباق هو الجسم يسوّي حسابات أكسجينه.
\end{example}

\begin{example}[المرآة المغبَّشة]\label{ex:g4:breathing:mirror}
انفخ على مرآة باردة: ضباب. فهواء \omterm{def:g4:breathing:movements}{الزفير} يعود مغيَّرًا --- مدفَّأ،
ومرطَّبًا، وأفقر أكسجينًا وأغنى ثاني أكسيد كربون من غير أن يُرى.
والمرآة تُظهر الرطوبة؛ أما تبادل الغازات الذي لا تستطيع إظهاره فهو
الجزء المهم.
\end{example}

\begin{method}[العناية بالنفَس]\label{met:g4:breathing:care}
\begin{enumerate}
  \item هوّ الغرف كل يوم --- فكثرة المتنفسين في غرفة مغلقة تعني
        هواءً راكدًا غنيًّا بثاني أكسيد الكربون؛
  \item \omterm{def:g4:breathing:movements}{وتنفس} بأنفك في الهواء البارد أو المغبَّر: فهو يدفّئ ويرطّب
        ويصفّي؛
  \item واجرِ واسبح والعب --- فالرئتان \omterm{def:g3:bones-and-muscles:muscle}{وعضلات} \omterm{def:g4:breathing:movements}{التنفس} المتمرنة تقوى
        وتعمق؛
  \item وأبعد الدخان عن رئتيك: فهو يوسّخ الجيوب الرقيقة، وهي لا
        تنظّف نفسها إلا ببطء.
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:breathing:1}
سمِّ حركتي \omterm{def:g4:breathing:movements}{التنفس} وماذا يفعل الصدر في كل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:2}
تتبّع مسار نفَس من الأنف إلى الجيوب الهوائية الصغيرة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:3}
كم مرة يتنفس طفل هادئ مستريح في الدقيقة تقريبًا؟ وماذا يحدث لهذا
العدد في أثناء العدو، ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:4}
أي غاز يأخذه الجسم من الهواء، وأيًّا يردّ؟ وأين يحدث التبادل
بالضبط؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:5}
بم يختلف هواء \omterm{def:g4:breathing:movements}{الزفير} عن هواء \omterm{def:g4:breathing:movements}{الشهيق}؟ سمِّ ثلاثة فروق.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:6}
لماذا بُنيت الرئتان ملايين جيوب صغيرة بدل كيسين بسيطين؟ وأي عضو
آخر من أعضاء هذه السنة يستعمل الحيلة نفسها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:7}
لماذا كان \omterm{def:g4:breathing:movements}{التنفس} بالأنف أفضل في يوم شديد البرد أو مغبَّر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:8}
جرّب: وأنت جالس هادئ، عُدّ أنفاسك دقيقة واحدة. ثم اقفز عشرين قفزة
وعُدّ من جديد. اذكر العددين واشرح التغير.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:9}
الوجبات تستطيع أن تنتظر ساعات؛ أما \omterm{def:g4:breathing:movements}{التنفس} فلا يستطيع أن ينتظر
دقيقة. فعلام يدل هذا في شأن ما يخزنه الجسم من الأكسجين، بالقياس
إلى مخزونه من الغذاء؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:10}
يشعر المرء بأن هواء الصف ``خانق'' في أواخر الصباح. فما الذي تغير
في هوائه وفق \cref{prop:g4:breathing:exchange}، وبم يوصي
\cref{met:g4:breathing:care}؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:breathing:11}
السلمون المرقط لا رئتين له: بل يجري الماء على غلاصمه الريشية،
فيعبر الأكسجين \emph{المذاب في الماء} إلى دمه هناك. فما العمل الذي
تشترك فيه الغلاصم والرئتان؟ ولماذا يختنق السلمون في الهواء والأكسجين
فيه وفير؟ (إرشاد: خارج الماء تنهار سطوح الغلاصم الريشية ويلتصق
بعضها ببعض.)
\end{exercise}
