\section*{अध्याय \ref{ch:g10:heart-lungs-effort} --- मेहनत के दौरान हृदय और फेफड़े}

\begin{solution}{exo:g10:heart-lungs-effort:1}
$0.6 \times 14 = \qty{8.4}{L/min}$; $2.2 \times 35 = \qty{77}{L/min}$।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:2}
हृदय के एक हिस्से से प्रति मिनट पंप किए गए रक्त का आयतन:
$Q = f \times V_s$। $65 \times 0.075 \approx \qty{4.9}{L/min}$।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:3}
दायाँ अलिंद, दायाँ निलय, फुफ्फुसीय धमनी, फेफड़े (फुफ्फुसीय परिसंचरण),
फुफ्फुसीय शिराएँ, बायाँ अलिंद, बायाँ निलय, महाधमनी, अंग (दैहिक
परिसंचरण), शिराएँ, दायाँ अलिंद।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:4}
ज्वारीय आयतन \qty{0.5}{L}; जैव धारिता \qty{4.5}{L}; अवशिष्ट आयतन
\qty{1.2}{L}।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:5}
प्रति मिनट लगभग $220 - 16 = 204$ और $220 - 60 = 160$ धड़कनें।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:6}
\qty{100}{W}: $120 \times 0.105 \approx \qty{12.6}{L/min}$;
\qty{300}{W}: $198 \times 0.115 \approx \qty{22.8}{L/min}$; यानी 1.8 का
गुणक।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:7}
विश्राम: \qty{2.5}{L/min}; व्यायाम: $0.04 \times 24 \approx
\qty{1.0}{L/min}$। हिस्सा बारह गुना गिरता है, पर निरपेक्ष प्रवाह केवल
2.6 गुना, क्योंकि निर्गम ख़ुद बढ़ चुका है।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:8}
$18 \times 0.14 = \qty{2.5}{L/min}$: यानी किसी प्रशिक्षित विद्यार्थी के
\qty{3.45}{L/min} का लगभग तीन-चौथाई --- छोटे क़द के व्यक्ति के लिए उसके
पास ही।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:9}
मस्तिष्क ऑक्सीजन और ग्लूकोज़ के बिना कुछ सेकंड भी नहीं सह सकता, इसलिए
उसकी आपूर्ति बचाकर रखी जाती है। आँत और गुर्दे पाचन तथा छनाई को घंटे भर
बिना नुक़सान के टाल सकते हैं, और उनका हिस्सा पेशियों को उधार दे दिया जाता
है।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:10}
$V_s = 5000/42 \approx \qty{120}{mL}$, जबकि दूसरे में लगभग \qty{70}{mL}:
उसका बड़ा हुआ हृदय विश्राम का निर्गम कहीं कम धड़कनों में पहुँचा देता है।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:11}
हिलने का मस्तिष्क का आदेश, और मेहनत का उसका पहले से अंदाज़ा, पेशियों के
काम शुरू करने से पहले ही हृदय-केंद्रों और अधिवृक्क ग्रंथियों तक भेज दिए
जाते हैं। जो धावक केवल कार्बन डाइऑक्साइड भाँपने के भरोसे होता, वह विश्राम
के निर्गम से शुरू करता, पहला मिनट \omterm{def:g10:cell-metabolism:fermentation}{किण्वन} पर दौड़ता, और आपूर्ति के पकड़ने
से पहले ही लैक्टिक अम्ल उसे धीमा कर देता।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:12}
$195 \times 0.110 \times 0.150 \approx \qty{3.2}{L/min}$ और
$195 \times 0.160 \times 0.150 \approx \qty{4.7}{L/min}$। प्रघात आयतन ही
वह चीज़ है जिसे प्रशिक्षण बड़ा करता है: इसलिए दूसरा व्यक्ति सहनशक्ति का
प्रशिक्षित है।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:13}
साँस में ली गई ऑक्सीजन: $0.21 \times 100 = \qty{21}{L/min}$; इस्तेमाल
हुई: \qty{3.5}{L/min}, यानी 17\%। बाक़ी दोबारा बाहर छोड़ दी जाती है ---
छोड़ी गई हवा में अब भी लगभग 16\% ऑक्सीजन रहती है, और इसीलिए मुँह से मुँह
में साँस देना काम करता है।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:14}
निर्गम ज़्यादा से ज़्यादा 24 की जगह
लगभग $110 \times 0.12 \approx \qty{13}{L/min}$: यानी अधिकतम $\dot V\!\mathrm{O_2}$ लगभग आधी गिरकर
क़रीब \qty{2}{L/min} रह जाती है। रोगी हल्की चाल से चल और साइकिल चला सकता
है, पर कोई कड़ी मेहनत टिका नहीं सकता।
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:15}
प्रति लीटर पहुँचाई गई ऑक्सीजन चौथाई भर घट गई, और निष्कर्षण अनुपात में
अनबदला: इसलिए अधिकतम $\dot V\!\mathrm{O_2}$ 25\% गिरती है। अगले हफ़्तों
में शरीर अधिक लाल रक्त \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} बनाता है और प्रति लीटर ढोई गई ऑक्सीजन को
दोबारा \qty{200}{mL} की ओर ले जाता है।
\end{solution}

\begin{solution}{pb:g10:heart-lungs-effort:1}
\textbf{1.} $62 \times 0.072 \approx 4.5$; $105 \times 0.100 = 10.5$;
$140 \times 0.116 \approx 16.2$; $175 \times 0.120 = 21.0$;
$198 \times 0.120 \approx \qty{23.8}{L/min}$।

\textbf{2.} $23.8/4.5 \approx 5.3$: दर का योगदान $198/62
\approx 3.2$, प्रघात आयतन का $120/72 \approx 1.7$, और $3.2 \times
1.7 \approx 5.3$।

\textbf{3.} लगभग \qty{225}{W} से ऊपर प्रघात आयतन \qty{120}{mL} पर ही
टिका रहता है; उसके आगे निर्गम केवल हृदय दर बढ़ाती है।

\textbf{4.} $220 - 17 = 203$: उसका 198 इस नियम की यथार्थता के भीतर है।

\textbf{5.} $5/4.5 \approx \qty{1.1}{min}$, यानी लगभग \qty{67}{s}; और
\qty{300}{W} पर $5/23.8 \approx \qty{0.21}{min}$, यानी लगभग \qty{13}{s}।

\textbf{6.} $0.5 \times 13 = 6.5$; $1.2 \times 18 = 21.6$; $1.8 \times
24 = 43.2$; $2.3 \times 32 = 73.6$; $2.6 \times 44 \approx
\qty{114}{L/min}$।

\textbf{7.} $114/6.5 \approx 18$: ज्वारीय आयतन 5.2 गुना बढ़ा और दर 3.4
गुना; यानी आयतन का योगदान अधिक है।

\textbf{8.} प्रति साँस जैव धारिता का $2.6/5.0 = 52\%$।

\textbf{9.} नहीं: \omterm{def:g10:heart-lungs-effort:ventilation}{संवातन} के पास बड़ा भंडार है और फेफड़ों से निकलता रक्त
संतृप्त ही बना रहता है। सीमा हृदय का प्रघात आयतन है, जो तय करता है कि हर
धड़कन कितनी ऑक्सीजन पहुँचाती है।

\textbf{10.} $0.7 \times 6.5 \approx \qty{4.6}{L/min}$ और $0.7 \times
114 \approx \qty{80}{L/min}$।

\textbf{11.} $4.5 \times 0.052 \approx 0.23$; $10.5 \times 0.090
\approx 0.95$; $16.2 \times 0.120 \approx 1.94$; $21.0 \times 0.145
\approx 3.05$; $23.8 \times 0.155 \approx \qty{3.7}{L/min}$।

\textbf{12.} प्रति किलोग्राम $3700/68 \approx \qty{54}{mL/min}$। दर
अपनी अधिकतम पर पहुँच चुकी है और 225 से \qty{300}{W} तक की बढ़त छोटी है:
यानी यही उसकी अधिकतम $\dot V\!\mathrm{O_2}$ है, किसी चुस्त, प्रशिक्षित
विद्यार्थी वाली।

\textbf{13.} $3.7/0.23 \approx 16$: यानी हृदय का 5.3 का गुणक, गुणा
निष्कर्षण का $155/52 \approx 3.0$।

\textbf{14.} प्रति मिनट $3.7 \times 20 = \qty{74}{kJ}$, यानी लगभग
\qty{1230}{W}; विश्राम में $0.23 \times 20/60 \approx \qty{77}{W}$;
विश्राम से ऊपर लगभग \qty{1150}{W}, इसलिए दक्षता $300/1150 \approx
26\%$।

\textbf{15.} 0 से \qty{225}{W} के बीच \num{0.23} से \qty{3.05}{L/min}
तक मोटे तौर पर एक सीधी रेखा, और फिर मुड़ती हुई: आख़िरी \qty{75}{W} केवल
\qty{0.65}{L/min} जोड़ते हैं। छत लगभग \qty{3.7}{L/min} है।

\textbf{16.} पहले से अंदाज़ा: मेहनत की मस्तिष्क की अपेक्षा हृदय-केंद्रों
और अधिवृक्क ग्रंथियों तक पहुँचा दी जाती है, जिससे दर किसी हलचल से पहले ही
चढ़ जाती है।

\textbf{17.} \omterm{def:g10:heart-lungs-effort:ventilation}{कूपिकाएँ} अधिकतम \omterm{def:g10:heart-lungs-effort:ventilation}{संवातन} और निर्गम पर भी रक्त को पूरी तरह
ऑक्सीजन दे देती हैं; यानी फेफड़े सीमा नहीं हैं। सीमा वह रक्त है जितना
हृदय प्रति मिनट भेज सकता है।

\textbf{18.} $20/23.8 \approx 84\%$। आँत और गुर्दे, जो विश्राम के निर्गम
का आधा ले जाते थे, अब लगभग \qty{1}{L/min} पाते हैं; उनकी धमनियाँ सँकरी हो
गई हैं और प्रवाह पेशियों की ओर मोड़ दिया गया है।

\textbf{19.} $198 \times 0.140 \times 0.155 \approx \qty{4.3}{L/min}$,
यानी प्रति किलोग्राम लगभग \qty{63}{mL/min}।

\textbf{20.} \omterm{def:g10:heart-lungs-effort:output}{हृदय-निर्गम} 5.3 गुना, ऑक्सीजन-निष्कर्षण 3 गुना और \omterm{def:g10:heart-lungs-effort:ventilation}{संवातन} 18
गुना; और प्रशिक्षण ने हृदय का प्रघात आयतन बदला, तथा उसके साथ वह छत भी।
\end{solution}
