\section*{अध्याय \ref{ch:g6:exploring-our-environment} --- अपने पर्यावरण की खोज}

\begin{solution}{exo:g6:exploring-our-environment:1}
\omterm{def:g6:exploring-our-environment:components}{सजीव} (\omterm{prop:g4:plants-without-seeds:damp}{मॉस}, कठ-जूएँ); \omterm{def:g6:exploring-our-environment:components}{खनिज पदार्थ} (दीवार का पत्थर, पोखर); और सजीवों के
निशान तथा अवशेष (सूखे पत्ते, घोंघे का \omterm{def:g1:animals-around-us:covering}{खोल})।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:2}
\omterm{def:g6:exploring-our-environment:conditions}{ताप}, थर्मामीटर से (डिग्री सेल्सियस में); हवा और मिट्टी की \omterm{def:g6:exploring-our-environment:conditions}{आर्द्रता}; और
\omterm{def:g6:exploring-our-environment:conditions}{प्रकाश}, पूरी धूप से गहरी छाया तक।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:3}
\omterm{def:g6:exploring-our-environment:components}{सजीव}: \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न}, कनखजूरा। खनिज: पोखर, दीवार का पत्थर। निशान: घोंघे का \omterm{def:g1:animals-around-us:covering}{खोल},
\omterm{prop:g3:sorting-living-things:groups}{पक्षियों} का मल।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:4}
सोलह डिग्री (\qty{38}{\degreeCelsius} बनाम
\qty{22}{\degreeCelsius}) और पूरी धूप बनाम गहरी छाया (साथ ही सूखा बनाम
नम)। सूखा और चौंध झेलने वाला खरपतवार दक्षिणी \omterm{def:g1:plants-around-us:parts}{जड़} से मेल खाता है; और
\omterm{prop:g4:plants-without-seeds:damp}{मॉस}, \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न} तथा कठ-जूएँ ठंडे नम कोने से।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:5}
उसका आवरण पानी को निकल जाने देता है, इसलिए सूखी धूप में उसका शरीर अपना
पानी घातक तेज़ी से खो देता है। इसीलिए कठ-जूएँ नम, अँधेरे \omterm{def:g6:exploring-our-environment:microhabitat}{सूक्ष्म आवासों}
में मिलते हैं --- पत्थरों के नीचे, सूखी लकड़ी के नीचे।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:6}
जैसे: कुतरा हुआ हेज़लनट --- जंगली चूहा; मकड़ी का जाला --- उसकी मकड़ी;
कीचड़ में पदचिह्न --- बगुला (मल, \omterm{def:g1:animals-around-us:covering}{पंख} और \omterm{def:g1:animals-around-us:covering}{खोल} भी इसी तरह काम आते हैं)।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:7}
एक बंद डिब्बा, जिसका आधा नम काग़ज़ से और आधा सूखे से ढका हो, और बीच की
रेखा पर छोड़े गए कठ-जूएँ; कुछ मिनट बाद लगभग सारे नम तरफ़ बैठे मिलते हैं
(और वैसे ही छाया बनाम \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} में)। यह दिखाता है कि हर \omterm{def:g1:animals-around-us:animal}{जंतु} वहीं बैठ
जाता है जहाँ \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} उसके शरीर के अनुकूल हों --- और पूरे समूह का
नक़्शा बिना किसी हाँके अपने आप बन जाता है।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:8}
क्योंकि तुलना सिर्फ़ उन्हीं मापों की हो सकती है जो एक ही तरीक़े से लिए
गए हों: फ़र्क़ स्थलों से आना चाहिए, नापने के ढंग से नहीं। यह न्यायी जाँच
का क्षेत्र-रूप है --- एक चीज़ बदलती है (स्थल), और बाक़ी सब बराबर रखा
जाता है।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:9}
पत्थर के ऊपर: उजाला, सूखा, गरम --- लाइकेन (और मेहमान के रूप में धूप
सेंकती मक्खी)। पत्थर के नीचे: अँधेरा, नम, ठंडा --- कठ-जूएँ (या
कनखजूरा)। पाँच सेंटीमीटर की दूरी, और बाशिंदों के दो अलग-अलग मेल।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:10}
चरण 1: एक तय वर्ग मीटर चिह्नित करो। चरण 2: नापो --- थर्मामीटर का पाठ
लिखो, मिट्टी की नमी छूकर दर्ज करो, \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} का वर्ग लिखो। चरण 3: ठीक से
खोजो, पत्थर उठाकर वापस रखो, तीनों तरह के घटक गिनाओ। चरण 4: मौक़े पर ही
दर्ज करो --- गिनतियाँ, ``कुछ'' नहीं। चरण 5: एक ही स्थल से तो तुलना
संभव ही नहीं थी; दूसरा स्थल भी उसी तरह सर्वेक्षण करो।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:11}
व्याख्या: आइवी उत्तरी दीवार की \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} की परास (छाया, नमी) में
फलती-फूलती है और दक्षिणी दीवार की (चौंध, सूखा) में नहीं। जाँच: दोनों
दीवारों की \omterm{def:g1:plants-around-us:parts}{जड़ें} नापो --- \omterm{def:g6:exploring-our-environment:conditions}{ताप}, नमी, \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} --- और आइवी की ज्ञात पसंद
से मिलाओ; या दूसरी दीवारों पर आइवी ढूँढ़ो और लिखो कि उन दीवारों का मुँह
किस दिशा में है।
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:12}
दीवार वाला फ़र्क़ हटा दो: चुनाव को गोल डिब्बे में चलाओ (जहाँ दीवार हर
जगह एक जैसी हो), और नम तथा सूखे आधों को दौड़ों के बीच बदल दो --- पहली
दौड़ में नम पूरब, दूसरी में नम पश्चिम। अगर कठ-जूएँ नमी जहाँ भी रखी जाए
वहीं जाते हैं, तो दीवार का असर ख़ारिज हो जाता है; दौड़ों के बीच सिर्फ़
जाँची जा रही परिस्थिति बदली, बाक़ी सब बराबर रहा।
\end{solution}

\begin{solution}{pb:g6:exploring-our-environment:1}
\textbf{1.} ऊँची बाड़: वह उत्तरी नाली को छाया (और ओट) देती है, जबकि दक्षिणी नाली
दिन भर सूरज की ओर मुँह किए रहती है।

\textbf{2.} \omterm{def:g6:exploring-our-environment:conditions}{ताप}: \qty{19}{\degreeCelsius} बनाम
\qty{31}{\degreeCelsius}। \omterm{def:g6:exploring-our-environment:conditions}{आर्द्रता}: उत्तरी नाली की मिट्टी नम है (चिपकती
है), दक्षिणी नाली की सूखी (धूल)। और \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} तिकड़ी पूरी करता है: छाया बनाम
पूरी धूप।

\textbf{3.} \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} दिन भर बदलती हैं: शाम छह बजे तक कोई भी स्थल
ठंडा हो जाता है और उसका \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} कम, इसलिए नापा गया फ़र्क़ स्थल और घंटे,
दोनों को मिला देता। एक ही घंटा तुलना को न्यायी रखता है।

\textbf{4.} ताकि खोज की मेहनत बराबर रहे: एक स्थल पर ज़्यादा क्षेत्रफल
या ज़्यादा समय वहीं ज़्यादा बाशिंदे दिखा देता --- यानी कौन-सी नाली
ज़्यादा समृद्ध है, इसकी अन्यायी जाँच।

\textbf{5.} दोनों सूचियाँ सिर्फ़ \omterm{def:g6:exploring-our-environment:components}{सजीव} गिनाती हैं। छूट रहे हैं: खनिज
घटक (मिट्टी, पत्थर, पानी) और निशान --- जिन्हें ठीक सूची के हिस्सों में
दर्ज होना ही चाहिए।

\textbf{6.} कठ-जूएँ: उत्तरी नाली --- उन्हें नमी और छाया चाहिए, वरना वे सूख
जाते हैं। भेक: उत्तरी नाली --- नम \omterm{def:g1:the-five-senses:sense}{त्वचा}, और नम ओट की ज़रूरत। छिपकली: दक्षिणी नाली
--- वह धूप में अपना शरीर गरम करती है और वहीं शिकार करती है जहाँ धूप सेंक
सके।

\textbf{7.} या तो घोंघे वहाँ छिपकर रहते हैं (शायद सिर्फ़ नम रातों में
सक्रिय), या वह \omterm{def:g1:animals-around-us:covering}{खोल} कहीं और से या किसी और मौसम का अवशेष है --- निशान
\emph{किसी समय} की मौजूदगी साबित करता है, अभी का बसेरा नहीं।

\textbf{8.} अँधेरा, नम, ठंडा।
\cref{prop:g6:exploring-our-environment:decide} का नियम: हर जगह में वही
\omterm{def:g5:biodiversity:species}{प्रजातियाँ} बसती हैं जिनकी परास उसकी \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} से मेल खाती है।

\textbf{9.} उत्तरी में \omterm{prop:g4:plants-without-seeds:damp}{मॉस} और भेक हैं क्योंकि उसकी छाया उसे ठंडा और नम रखती
है, जो उनके शरीरों को चाहिए; दक्षिणी में छिपकलियाँ और मोम जैसी \omterm{def:g1:plants-around-us:parts}{पत्तियों} वाले
खरपतवार हैं क्योंकि उसकी पूरी धूप उसे गरम और सूखा बना देती है, जो धूप
सेंकते शिकारी और सूखा झेलने वाले \omterm{def:g1:plants-around-us:plant}{पौधे} के अनुकूल है।

\textbf{10.} बाड़ के बिना उत्तरी नाली को पूरी धूप मिलेगी: ज़्यादा गरम,
ज़्यादा सूखी --- यानी \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} आज की दक्षिणी नाली की ओर खिसकती हुईं।
उम्मीद करो कि नमी पसंद करने वाले (\omterm{prop:g4:plants-without-seeds:damp}{मॉस}, \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न}, कठ-जूएँ, भेक) घट जाएँगे
और धूप पसंद करने वाले (सूखी घासें, टिड्डे, शायद छिपकली) आ बसेंगे।

\textbf{11.} क्योंकि दोबारा सर्वेक्षण पहले से सिर्फ़ एक ही बात में अलग
होना चाहिए --- बाड़ का न होना। नए स्थल, नए उपकरण या नया मौसम अपने
फ़र्क़ जोड़ देते, और तब बदलाव को बाड़ के \omterm{def:g1:my-body:parts}{सिर} नहीं मढ़ा जा सकता था:
यही न्यायी जाँच का नियम है, पर पूरे भूदृश्य के पैमाने पर।

\textbf{12.} हर कठ-जूँ तब तक घूमता और मुड़ता रहता है जब तक उसके शरीर को
आराम न मिल जाए, इसलिए पूरा समूह वहीं इकट्ठा हो जाता है जहाँ नमी है ---
यानी \omterm{def:g1:animals-around-us:animal}{जंतुओं} की जगहें \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} को उतनी ही पक्की तरह दर्ज करती हैं
जितनी थर्मामीटर की पढ़तें। कोई उन्हें हाँकता नहीं; नक़्शा हर जीव का अपना
आराम ही खींच देता है, और ठीक इसीलिए बाशिंदों को मापों की तरह पढ़ा जा
सकता है।
\end{solution}
