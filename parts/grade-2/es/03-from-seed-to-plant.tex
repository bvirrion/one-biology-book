\chapter{De la semilla a la planta}\label{ch:g2:from-seed-to-plant}

Agita una vaina seca de judías y salen traqueteando unas judías duras
--- secas, apagadas, aparentemente tan sin vida como la grava. Y sin
embargo, \omterm{def:g1:plants-around-us:plant}{planta} una en la tierra y riégala: en unos pocos días pasa
algo extraordinario bajo tierra. Dentro había una \omterm{def:g1:plants-around-us:plant}{planta} entera
doblada, esperando su señal.

\section{Qué es una semilla}

\begin{definition}[Semilla]\label{def:g2:from-seed-to-plant:seed}
Una \emph{semilla}\index{semilla} es el comienzo de una \omterm{def:g1:plants-around-us:plant}{planta} nueva,
hecha por la \omterm{def:g1:plants-around-us:plant}{planta} madre. Dentro de su cubierta protectora, una
semilla guarda:
\begin{itemize}
  \item una plantita ya empezada --- el \emph{germen}\index{germen (de
        una semilla)} de la \omterm{def:g1:plants-around-us:plant}{planta}, con el principio de una \omterm{def:g1:plants-around-us:parts}{raíz} y de
        un brote;
  \item una despensa de alimento que la mantiene hasta que puede
        alimentarse sola.
\end{itemize}
\end{definition}

\begin{example}[Abrir una judía en remojo]\label{ex:g2:from-seed-to-plant:open}
Deja una judía blanca grande en remojo toda la noche y su cubierta se
suelta con facilidad. La judía se abre en dos mitades gordas --- la
despensa --- y allí, acurrucada en un borde, hay una plantita pálida,
con la punta de la \omterm{def:g1:plants-around-us:parts}{raíz} y las primeras \omterm{def:g1:plants-around-us:parts}{hojas} dobladas, pequeña como un
grano de arroz. La \omterm{def:g1:plants-around-us:plant}{planta} nueva estaba allí todo el tiempo.
\end{example}

\begin{remark}[Las semillas pasan por nuestras cocinas]\label{rem:g2:from-seed-to-plant:kitchen}
Judías, guisantes, lentejas, arroz, pipas de girasol, las pepitas de
la manzana y el hueso del melocotón: la cocina está llena de \omterm{def:g2:from-seed-to-plant:seed}{semillas}.
Algunas son diminutas, como las de amapola; el coco es una gigante.
Cualquiera de ellas, si tuviera ocasión, intentaría hacerse \omterm{def:g1:plants-around-us:plant}{planta}.
\end{remark}

\section{La germinación}

\begin{definition}[Germinación]\label{def:g2:from-seed-to-plant:germination}
La \emph{germinación}\index{germinación} es el momento en que una
\omterm{def:g2:from-seed-to-plant:seed}{semilla} se despierta y empieza a crecer: se hincha con el agua, su
cubierta se abre, la raicita sale la primera y se hunde hacia abajo, y
después sube el brote, que lleva las primeras \omterm{def:g1:plants-around-us:parts}{hojas} verdes hacia la
luz. Decimos que la \omterm{def:g2:from-seed-to-plant:seed}{semilla} \emph{germina}\index{germinar}.
\end{definition}

\begin{proposition}[Lo que necesita una semilla para germinar]\label{prop:g2:from-seed-to-plant:needs}
Para \omterm{def:g2:from-seed-to-plant:germination}{germinar}, una \omterm{def:g2:from-seed-to-plant:seed}{semilla} necesita \emph{agua} y \emph{calor}. Todavía
no necesita luz: la despensa alimenta a la plantita al principio,
incluso bajo tierra, a oscuras. Sin agua, o con frío, una \omterm{def:g2:from-seed-to-plant:seed}{semilla} buena
simplemente espera.
\end{proposition}
\admitted

\begin{example}[Por qué se siembra en primavera]\label{ex:g2:from-seed-to-plant:spring}
Los hortelanos siembran las judías en primavera, no en invierno: la
\omterm{def:g2:from-seed-to-plant:seed}{semilla} en tierra fría esperaría --- o se pudriría --- en vez de
\omterm{def:g2:from-seed-to-plant:germination}{germinar}. Los días cálidos y la tierra regada son exactamente las dos
señales de la \cref{prop:g2:from-seed-to-plant:needs}.
\end{example}

\begin{method}[Sembrar una semilla y verla germinar]\label{met:g2:from-seed-to-plant:sow}
\begin{enumerate}
  \item forra un bote de cristal con algodón o papel húmedo;
  \item desliza una judía entre el cristal y el algodón, para poder verla;
  \item deja el bote en una habitación cálida y el algodón húmedo;
  \item mira cada día y anota lo que ves: cómo se hincha, cómo se abre
        la cubierta, la \omterm{def:g1:plants-around-us:parts}{raíz} hacia abajo, el brote hacia arriba.
\end{enumerate}
El cristal hace visible lo que normalmente pasa bajo tierra.
\end{method}

\begin{remark}[La raíz abajo, el brote arriba --- siempre]\label{rem:g2:from-seed-to-plant:updown}
Da igual cómo esté colocada la judía en el bote --- de lado, boca
abajo ---: la \omterm{def:g1:plants-around-us:parts}{raíz} siempre gira hacia abajo y el brote siempre gira
hacia arriba. La plantita no ve y, aun así, sabe dónde está el arriba.
Las \omterm{def:g1:plants-around-us:plant}{plantas} notan más de lo que enseñan.
\end{remark}

\section{De la plántula a la planta con semillas}

\begin{proposition}[El ciclo de vida de la planta]\label{prop:g2:from-seed-to-plant:cycle}
La \emph{plántula}\index{plántula} se convierte en una \omterm{def:g1:plants-around-us:plant}{planta} con
\omterm{def:g1:plants-around-us:parts}{raíces}, \omterm{def:g1:plants-around-us:parts}{tallo} y \omterm{def:g1:plants-around-us:parts}{hojas}; la \omterm{def:g1:plants-around-us:plant}{planta} florece; las \omterm{def:g1:plants-around-us:parts}{flores} hacen
\emph{\omterm{def:g2:from-seed-to-plant:seed}{semillas}} nuevas; y el ciclo puede empezar otra vez. Una sola
mata de judías puede hacer decenas de judías nuevas en un verano.
\end{proposition}
\admitted

\begin{example}[Una judía, un verano]\label{ex:g2:from-seed-to-plant:summer}
Sembrada en mayo, una judía \omterm{def:g2:from-seed-to-plant:germination}{germina} en una semana; en junio es una
mata frondosa que trepa por su caña; en julio llegan las \omterm{def:g1:plants-around-us:parts}{flores}
blancas y después las vainas verdes; en agosto las vainas cuelgan
pesadas, cada una con una fila de judías nuevas --- las \omterm{def:g2:from-seed-to-plant:seed}{semillas} del
año que viene, hechas por la mata de este año.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:1}
¿Qué dos cosas hay dentro de la cubierta de una \omterm{def:g2:from-seed-to-plant:seed}{semilla}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:2}
¿Qué significa \emph{\omterm{def:g2:from-seed-to-plant:germination}{germinar}}? ¿Qué sale primero de la \omterm{def:g2:from-seed-to-plant:seed}{semilla}, la
\omterm{def:g1:plants-around-us:parts}{raíz} o el brote?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:3}
Nombra las dos cosas que necesita una \omterm{def:g2:from-seed-to-plant:seed}{semilla} para \omterm{def:g2:from-seed-to-plant:germination}{germinar}. ¿Qué
necesidad famosa de las \omterm{def:g1:plants-around-us:plant}{plantas} mayores \emph{no} está en la lista?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:4}
¿Cómo puede crecer a oscuras una \omterm{prop:g2:from-seed-to-plant:cycle}{plántula} enterrada, antes de llegar a
la luz? ¿De dónde saca su alimento al principio?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:5}
Nombra cuatro \omterm{def:g2:from-seed-to-plant:seed}{semillas} que podrías encontrar en una cocina.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:6}
¿Por qué siembran los hortelanos en primavera y no en invierno?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:7}
Ordena: vaina con judías nuevas --- \omterm{def:g2:from-seed-to-plant:seed}{semilla} --- \omterm{def:g1:plants-around-us:plant}{planta} en \omterm{def:g1:plants-around-us:parts}{flor} ---
\omterm{prop:g2:from-seed-to-plant:cycle}{plántula} --- mata frondosa.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:8}
Pruébalo: monta el bote de cristal del
\cref{met:g2:from-seed-to-plant:sow} con dos judías --- una tumbada de
lado y otra boca abajo. Después de la \omterm{def:g2:from-seed-to-plant:germination}{germinación}, compara sus \omterm{def:g1:plants-around-us:parts}{raíces}
y sus brotes. ¿Qué predice la \cref{rem:g2:from-seed-to-plant:updown}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:from-seed-to-plant:9}
Nina guarda un sobre de judías en un cajón seco durante un año entero
y germinan igualmente cuando por fin las siembra. ¿Por qué no
crecieron ni se murieron en el cajón? Usa la
\cref{prop:g2:from-seed-to-plant:needs}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:from-seed-to-plant:10}
El pollito del \cref{ch:g2:animals-grow-up} crece dentro de su \omterm{def:g2:animals-grow-up:born}{huevo},
alimentado por la yema. ¿Qué hace el papel de la yema dentro de una
judía? ¿Qué tienen en común un \omterm{def:g2:animals-grow-up:born}{huevo} y una \omterm{def:g2:from-seed-to-plant:seed}{semilla}?
\end{exercise}
