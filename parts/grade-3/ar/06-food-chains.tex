\chapter{السلاسل الغذائية}\label{ch:g3:food-chains}

ينمو العشب في المرج. ويأكل جندب العشب. وتمسك سحلية الجندب. ويأخذ
صقر يحوم عاليًا السحلية. أربع وجبات --- وإذا بأصيل الصقر يتوقف
على مدى اخضرار العشب. فوجبات الطبيعة تتصل في سلاسل، وتعلّم
قراءتها تعلّم لكيفية تماسك مرج كامل.

\section{قراءة السلسلة الغذائية وكتابتها}

\begin{definition}[السلسلة الغذائية]\label{def:g3:food-chains:chain}
\emph{السلسلة الغذائية}\index{سلسلة غذائية} صف من الكائنات الحية
يأكل كلَّ واحد منها الذي يليه. وتُكتب بأسهم، ويعني كل سهم
\emph{``يأكله''}\index{يأكله}:
\[
\text{عشب} \longleftarrow \text{جندب}
\longleftarrow \text{سحلية} \longleftarrow \text{صقر}.
\]
وكل \omterm{def:g1:living-or-not:living}{كائن حي} \emph{حلقة}\index{حلقة} من السلسلة.
\end{definition}

\begin{remark}[انتبه إلى اتجاه السهم]\label{rem:g3:food-chains:arrow}
السهم يشير من المأكول إلى الآكل --- فهو يتبع الغذاء. و``عشب
$\leftarrow$ جندب'' تُقرأ ``العشب \omterm{def:g3:food-chains:chain}{يأكله} الجندب''. وكتابة الأسهم
معكوسة --- نحو الوجبة --- أشيع خطأ في \omterm{def:g3:food-chains:chain}{السلاسل الغذائية}؛ فتذكّر أن
السهم يبين إلى أين \emph{يذهب} الغذاء.
\end{remark}

\begin{proposition}[كل سلسلة تبدأ بنبات]\label{prop:g3:food-chains:start}
الحلقة الأولى من \omterm{def:g3:food-chains:chain}{السلسلة الغذائية} \omterm{def:g1:plants-around-us:plant}{نبات} دائمًا (أو \omterm{def:g1:living-or-not:living}{كائن حي} آخر يصنع
غذاءه بنفسه). \omterm{def:g1:plants-around-us:plant}{فالنباتات} وحدها تتغذى من غير أن تأكل أحدًا --- من
الضوء والماء والهواء، كما قالت
\cref{def:g1:plants-around-us:plant}. وكل حلقة حيوانية تعيش،
مباشرة أو عبر غيرها، على تلك الحلقة الخضراء الأولى.
\end{proposition}
\admitted

\begin{example}[سلاسل من ثلاثة موائل]\label{ex:g3:food-chains:three}
في البركة: \omterm{def:g1:plants-around-us:plant}{نباتات} مائية $\leftarrow$ شرغوف $\leftarrow$
\omterm{ex:g2:animals-grow-up:butterfly}{يرقة} رعّاشة $\leftarrow$ سمكة شوكية $\leftarrow$ بلشون.
وفي الغابة: \omterm{def:g1:plants-around-us:parts}{أوراق} البلوط $\leftarrow$ \omterm{ex:g2:animals-grow-up:butterfly}{يرقة}
$\leftarrow$ قرقف أزرق $\leftarrow$ باشق. وفي
الحديقة: خس $\leftarrow$ حلزون $\leftarrow$ قنفذ. والحلقة
الأولى: خضراء دائمًا.
\end{example}

\begin{method}[بناء سلسلة غذائية]\label{met:g3:food-chains:build}
\begin{enumerate}
  \item اختر حيوانًا واسأل: ماذا يأكل؟ واكتب الغذاء عن يساره،
        والسهم يشير إلى الآكل؛
  \item واسأل السؤال نفسه عن الغذاء، وتابع نحو اليسار حتى تبلغ
        نباتًا؛
  \item واسأل: من يأكل حيواني الأول؟ ومدّ السلسلة نحو اليمين ما
        دام ثمة آكل؛
  \item وتحقق أن كل سهم يُقرأ \omterm{def:g3:food-chains:chain}{``يأكله''} --- والغذاء يتدفق نحو
        اليمين.
\end{enumerate}
\end{method}

\section{حين تتغير حلقة}

\begin{proposition}[الحلقات يعتمد بعضها على بعض]\label{prop:g3:food-chains:depend}
في \omterm{def:g3:food-chains:chain}{السلسلة الغذائية} تعتمد كل حلقة على الحلقات التي قبلها، وتغذّي
التي بعدها. فإن قلّت حلقة جاعت الحلقات التالية --- والحلقة التي
\emph{قبلها}، إذ لم تعد تُؤكل كثيرًا، قد تتكاثر. والتغير في حلقة
واحدة يسري في السلسلة كلها.
\end{proposition}
\admitted

\begin{example}[الصيف الجاف]\label{ex:g3:food-chains:dry}
يُصفّر جفافٌ عشبَ المرج. فتقلّ الجنادب لقلة غذائها؛ وتجوع السحالي
التي كانت تصطادها، فتغذّي الصقر تغذيةً رديئة بدورها. حلقة أولى
صفراء واحدة، فتنحل السلسلة كلها فوقها --- والصقر لم يأكل عشبًا
قط، ومع ذلك يحسّ بالجفاف.
\end{example}

\begin{example}[إزالة الصائد]\label{ex:g3:food-chains:hunter}
لنفترض أن الصقور اختفت من الوادي. فتتكاثر السحالي لقلة اصطيادها؛
وتُؤكل الجنادب عندئذ بشدة فتتضاءل؛ والعشب، إذ يقلّ قضمه، قد يكثف
أيضًا. فإزالة الحلقة \emph{الأخيرة} تسري في السلسلة كذلك --- في
الاتجاه الآخر.
\end{example}

\begin{remark}[المروج الحقيقية شبكات]\label{rem:g3:food-chains:web}
سلسلتنا خيط واحد، لكن الصقر يأخذ الفئران أيضًا، والسحلية تأكل
الذباب أيضًا، والعشب يغذّي الأرانب كذلك. وسلاسل كثيرة تتقاطع
فتصنع شبكة --- هي \emph{الشبكة الغذائية} التي ستدرسها سنة لاحقة
(وقد ألمح \cref{ch:g2:what-animals-eat} إليها من قبل بالثعلب
والأرنب والعشب). والسلاسل أولًا: فهي الخيوط التي تُنسج منها
الشبكة.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:1}
ماذا يعني سهم \omterm{def:g3:food-chains:chain}{السلسلة الغذائية}؟ وإلى أي جهة يشير؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:2}
أي صنف من الكائنات الحية هو الحلقة الأولى دائمًا؟ ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:3}
اكتب سلسلة المرج في هذا الفصل بحلقاتها الأربع وأسهمها.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:4}
صحّح هذه السلسلة: ثعلب $\leftarrow$ أرنب $\leftarrow$
عشب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:5}
ابنِ سلسلة من هذه الحلقات: قرقف أزرق، \omterm{def:g1:plants-around-us:parts}{وأوراق} بلوط، وباشق،
\omterm{ex:g2:animals-grow-up:butterfly}{ويرقة}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:6}
في سلسلة الحديقة خس $\leftarrow$ حلزون $\leftarrow$
قنفذ، من \omterm{def:g3:food-chains:chain}{يأكله} الحلزون؟ ومن يأكل الحلزون؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:7}
استعمل \cref{met:g3:food-chains:build} لصنع سلسلة من ثلاث حلقات
على الأقل تنتهي ببلشون البركة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:8}
في الصيف الجاف في \cref{ex:g3:food-chains:dry}، لماذا يعاني الصقر
مع أنه لا يأكل العشب قط؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:food-chains:9}
البستانيون الذين يسمّون كل حلزون يفقدون قنفذهم أحيانًا --- ثم
يلقون بزّاقات وحلزونات \emph{أكثر} من قبل. اشرح الشطرين وفق
\cref{prop:g3:food-chains:depend}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:food-chains:10}
``السحلية تعتمد على العشب.'' أصح هذا أم خطأ؟ دافع عن جوابك
بسلسلة المرج --- واذكر في أي اتجاه يسري فشل العشب.
\end{exercise}
