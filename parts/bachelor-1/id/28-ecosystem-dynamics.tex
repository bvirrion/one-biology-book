\chapter{Dinamika Ekosistem}\label{ch:b1:ecosystem-dynamics}

Di tempat sebuah gletser di Pasifik utara telah mundur selama dua
abad, sebuah perjalanan kaki dari moncongnya kembali menuruni fyordnya
adalah sebuah perjalanan menembus waktu. Batu yang tersingkap tahun
lalu gundul; sekilometer kemudian, ia mengenakan lumut dan sebuah
hamparan pemancang nitrogen; lebih jauh, belukar alder; lalu cemara,
lalu sebuah hutan cemara dan hemlock berumur dua ratus tahun dengan
semeter tanah di bawahnya. Tak ada yang tetap di sini: \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistem}
berubah sesudah tiap \omterm{def:b1:ecosystem-dynamics:disturbance}{gangguan}, sepanjang sebuah jalan yang sebagiannya
dapat diramalkan, dan kesanggupannya memegang lalu mendaur ulang
haranya berubah bersamanya. Bab ini mengikuti perubahan itu ---
\omterm{def:b1:ecosystem-dynamics:succession}{suksesi}, \omterm{def:b1:ecosystem-dynamics:disturbance}{gangguan} dan pemulihannya, penahanan haranya, perakitan
\omterm{def:b1:ecosystem-organization:ecosystem}{komunitas} di pulau --- serta percobaan yang mengubahnya dari cerita
menjadi angka.

\section{Suksesi}

\begin{definition}[Suksesi ekologi]\label{def:b1:ecosystem-dynamics:succession}
\index{suksesi}\index{suksesi primer}\index{suksesi sekunder}\index{klimaks}
Yang disebut \emph{suksesi} adalah perubahan berarah sebuah \omterm{def:b1:ecosystem-organization:ecosystem}{komunitas}
sepanjang waktu sesudah sebuah \omterm{def:b1:ecosystem-dynamics:disturbance}{gangguan}, satu perangkat spesies
menggantikan yang lain sampai sebuah \omterm{def:b1:ecosystem-organization:ecosystem}{komunitas} yang cukup mantap,
yaitu \emph{klimaks}nya, bertahan. \emph{Suksesi primer} bermula pada
alas baru tanpa tanah atau \omterm{def:b1:organism-environment:organism}{organisme} hidup: batu yang ditinggalkan
sebuah gletser, sebuah aliran lava, sebuah bukit pasir, sebuah pulau
baru. \emph{Suksesi sekunder} bermula tempat sebuah \omterm{def:b1:ecosystem-organization:ecosystem}{komunitas} telah
dimusnahkan tetapi tanahnya dan bank bijinya tetap ada: sebuah ladang
terbengkalai, sebuah hutan yang terbakar atau tertebang, sebuah padang
rumput yang terbanjiri. Pendatang pertamanya, yaitu \emph{spesies
perintis}, adalah penyebar yang menoleransi tanah gundul; pendatang
akhirnya adalah pesaing yang menoleransi naungan dan memerlukan tanah.
\end{definition}

\begin{proposition}[Mekanisme penggantiannya]\label{prop:b1:ecosystem-dynamics:mechanisms}
\index{fasilitasi}
Tiap tahap sebuah \omterm{def:b1:ecosystem-dynamics:succession}{suksesi} dapat mendatangkan tahap berikutnya dengan
tiga cara. \emph{Fasilitasi}: spesies yang lebih awal membuat tapaknya
pantas bagi yang lebih akhir --- membangun tanah, memancangkan
nitrogen, melemparkan naungan, memegang kelembapan --- lalu tergeser
olehnya. \emph{Inhibisi}: penghuni pertamanya memegang tapaknya lawan
tiap pendatang sampai keduanya mati atau rusak, dan yang
menggantikannya adalah apa pun yang tiba saat itu. \emph{Toleransi}:
spesies yang lebih akhir tiba pada awalnya juga, tumbuh perlahan lalu
sekadar bertahan lebih lama daripada perintisnya, yang kehadirannya
tidak menolong maupun menghalanginya. Sebagian besar \omterm{def:b1:ecosystem-dynamics:succession}{suksesi}
memperlihatkan ketiganya pada tahap yang berbeda; \omterm{def:b1:ecosystem-dynamics:succession}{suksesi primer}
didominasi fasilitasi, sebab tugas pertamanya adalah membuat sebuah
tanah.
\end{proposition}
\begin{proof}[Bukti]
Di Teluk Glacier di Alaska, tempat esnya telah mundur sekitar
\qty{100}{km} sejak 1750, umur tiap tapaknya diketahui dari peta dan
lingkaran tahun pohonnya, dan tahapnya dapat dikunjungi berurutan:
endapan gundul, sebuah kerak sianobakteri dan lumut, hamparan
\emph{Dryas} pemancang nitrogen, belukar alder (pemancang nitrogen,
yang menambahkan \qty{50}{kg} nitrogen tiap hektar tiap tahun), lalu
cemara Sitka, lalu sebuah hutan cemara--hemlock sesudah dua abad.
Nitrogen tanahnya naik dari nyaris nol menjadi \qty{300}{kg/ha} di
bawah aldernya dan cemaranya mapan hanya sesudahnya; pH-nya turun dari
8 ke bawah 5 seiring \omterm{def:b1:ecosystem-dynamics:decomposition}{seresahnya} bertimbun. Semai cemara yang ditanam
pada endapan mudanya tumbuh hanya bila dipupuk nitrogen: aldernya
adalah pemfasilitasi.
\end{proof}

\begin{example}[Sebuah ladang tua]\label{ex:b1:ecosystem-dynamics:oldfield}
Sebuah ladang yang ditinggalkan di Amerika Serikat bagian timur
dikoloni pada tahun pertamanya oleh gulma setahunan (rumput belulang,
ambrosia) dari bank bijinya dan anginnya; pada tahun ketiganya herba
tahunan dan solidago mendominasi; pada tahun kesepuluh, semak dan
semai pinus; pada tahun kedua puluh lima sebuah hutan pinus, yang di
bawah naungannya semai pinusnya sendiri tak dapat tumbuh tetapi pohon
ek dan hickory dapat; dan pada tahun keseratus sebuah hutan
ek--hickory, yang semainya menoleransi naungannya dan yang karena itu
menggantikan dirinya sendiri. Tiap tahapnya tergeser spesies yang
lebih buruk penyebarannya dan lebih baik \omterm{def:b1:species-interactions:types}{persaingannya} daripada
dirinya; tanahnya telah ada sejak awalnya, dan seluruh urutannya
memakan satu masa hidup manusia alih-alih ribuan tahun sebuah \omterm{def:b1:ecosystem-dynamics:succession}{suksesi
primer}.
\end{example}

\section{Gangguan, kemantapan dan keanekaragaman}

\begin{definition}[Gangguan, ketahanan, kelentingan]\label{def:b1:ecosystem-dynamics:disturbance}
\index{gangguan}\index{kelentingan}\index{ketahanan}
Yang disebut \emph{gangguan} adalah sebuah peristiwa --- kebakaran,
badai, banjir, kekeringan, penggembalaan, penebangan, sebuah wabah ---
yang menyingkirkan biomassa lalu membuka ruang. Yang disebut
\emph{ketahanan} sebuah \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistem} adalah kesanggupannya tinggal tak
berubah sepanjang sebuah gangguan; \emph{kelentingan}nya adalah laju
ia kembali ke keadaan terdahulunya sesudahnya. Sebuah hutan tahan
terhadap sebuah musim panas yang kering dan lambat pulih dari sebuah
kebakaran; sebuah padang rumput sebaliknya. \omterm{def:b1:ecosystem-organization:ecosystem}{Komunitas} dibentuk oleh
rezim gangguannya: kekerapan, keparahan dan keluasannya. Banyak
\omterm{def:b1:ecosystem-organization:ecosystem}{komunitas} berupa mosaik petak pada tahap \omterm{def:b1:ecosystem-dynamics:succession}{suksesi} yang berbeda sejak
gangguan terakhirnya, dan keanekaragamannya adalah jumlah pada seluruh
mosaiknya.
\end{definition}

\begin{proposition}[Hipotesis gangguan menengah]\label{prop:b1:ecosystem-dynamics:idh}
\index{hipotesis gangguan menengah}
Keanekaragamannya paling tinggi pada kekerapan dan keparahan \omterm{def:b1:ecosystem-dynamics:disturbance}{gangguan}
yang menengah. Tempat \omterm{def:b1:ecosystem-dynamics:disturbance}{gangguannya} jarang, pesaing yang terbaik
menyisihkan sisanya; tempat \omterm{def:b1:ecosystem-dynamics:disturbance}{gangguannya} kerap, hanya pengoloni yang
tercepat sintas; di antara keduanya, perintis dan pesaingnya
berdampingan, masing-masing dalam petak yang cocok baginya.
\end{proposition}
\begin{proof}[Bukti]
Di pantai berbatu dan terumbu karang, cacah spesies alga atau karang
pada bongkah batu dan bagian terumbunya paling besar pada yang
terbalik atau terpatahkan badai pada laju menengah: bongkah kecil
(yang sering terbalik) membawa beberapa alga perintis, yang besar
(jarang terbalik) sebuah monokultur pesaing terbaiknya, yang sedang
paling banyak spesiesnya. Punuk yang sama muncul pada dasar sungai
terhadap kekerapan banjirnya dan pada padang rumput terhadap keparahan
penggembalaannya.
\end{proof}

\begin{remark}[Adakah keanekaragaman membuat sebuah ekosistem mantap?]\label{rem:b1:ecosystem-dynamics:diversity-stability}
Petak padang rumput yang ditaburi lebih banyak spesies memegang
biomassanya lebih mantap sepanjang kekeringan daripada petak dengan
sedikit spesies, sebab dengan lebih banyak spesies sebagian selalu
cocok bagi cuaca tahunnya --- sebuah pengaruh asuransi. Tetapi
\omterm{def:b1:ecosystem-organization:ecosystem}{komunitas} yang beranekaragam tidak selalu lebih tahan, dan \omterm{def:b1:populations:population}{populasi}
masing-masing di dalamnya naik turun sebanyak di mana pun:
keanekaragaman memantapkan fungsi \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistemnya} (produksi, penahanan
haranya) lebih daripada susunannya.
\end{remark}

\section{Penguraian dan penahanan hara}

\begin{definition}[Penguraian dan anggaran hara]\label{def:b1:ecosystem-dynamics:decomposition}
\index{penguraian}\index{anggaran hara}\index{seresah}
Yang disebut \emph{penguraian} adalah pemecahan bahan organik mati
(\emph{seresah}, kayu mati, bangkai, tinja) oleh detritivor dan
\omterm{def:b1:ecosystem-organization:trophic}{pengurainya} menjadi karbon dioksida, air dan ion mineral ---
\emph{mineralisasi}. Lajunya bergantung pada suhu, kelembapan, oksigen
dan mutu seresahnya (kandungan nitrogen, lignin): sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} lenyap
dalam setahun dalam sebuah hutan hangat yang basah, dalam sepuluh
tahun di bawah konifer pada sebuah lereng dingin, dalam ribuan tahun
dalam sebuah rawa gambut. Ion mineral yang dilepaskannya diserap lagi
oleh \omterm{def:b1:flowering-plant-organization:organs}{akar} dan mikrobanya, dan dalam sebuah \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistem} yang dewasa daur
dalam ini membawa sebagian besar alir haranya: masukan dari pelapukan
dan hujannya serta keluaran dalam air sungainya kecil di sebelahnya.
Sebuah \emph{anggaran hara} membandingkan masukan, keluaran dan lubuk
dalam satu unsurnya bagi sebuah \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistem} tertentu; daur biogeokimia
pada skala planetnya dibahas dalam jilid Tahun ke-2.
\end{definition}

\begin{proposition}[Komunitas hidupnya menahan hara]\label{prop:b1:ecosystem-dynamics:retention}
Hara dipegang dalam sebuah \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistem} oleh serapan tumbuhan dan mikroba
hidupnya; ketika tumbuhannya disingkirkan, mineralisasinya berlanjut
sedangkan serapannya berhenti, dan ion yang terlepas tercuci keluar ke
sungainya --- \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistemnya} bocor sampai sebuah tutupan baru tumbuh
kembali.
\end{proposition}
\begin{proof}[Bukti]
Di Hubbard Brook di New Hampshire, enam lembah berhutan yang kecil
masing-masing dialiri satu sungai yang dipasangi sebuah ambang ukur,
sehingga segala yang pergi dalam airnya dapat diukur terhadap apa yang
tiba dalam hujannya. Pada 1965--66 satu lembahnya ditebang habis lalu
dijaga gundul dengan herbisida selama tiga tahun, tetangganya
ditinggalkan sebagai kendali. Alir sungai dari lembah yang ditebang
naik \qty{40}{\%} (tanpa transpirasi). Nitrat dalam sungainya naik
empat puluh sampai enam puluh kali, kalsiumnya sepuluh kali, kaliumnya
dua puluh kali; kehilangan total zat terlarutnya adalah enam sampai
delapan kali kehilangan kendalinya --- nitrogen yang telah dipegang
hutan utuhnya dalam daurnya selama berabad-abad pergi dalam dua musim
panas. Ketika pertumbuhan kembalinya diizinkan, kehilangannya jatuh
kembali dalam beberapa tahun, secepat tumbuhan barunya menyerap apa
yang dilepaskan tanahnya.
\end{proof}

\begin{definition}[Perekayasa ekosistem]\label{def:b1:ecosystem-dynamics:engineers}
\index{perekayasa ekosistem}
Yang disebut \emph{perekayasa \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistem}} adalah sebuah spesies yang
mengubah \omterm{def:b1:organism-environment:organism}{lingkungan} fisik bagi yang lain: berang-berang membanjiri
lembah, cacing tanah mengaduk dan mengudarakan tanahnya, karang dan
kerang membangun terumbu, pohon melemparkan naungan lalu memegang air,
gajah membuka hutan menjadi padang rumput, penggali liang
mengoksigenkan sedimennya. Perekayasanya bekerja pada \omterm{def:b1:ecosystem-organization:ecosystem}{biotopnya}
alih-alih lewat pautan trofiknya, dan pengaruhnya dapat bertahan lebih
lama daripada dirinya (sebuah padang berang-berang, sebuah terumbu,
sebuah rawa gambut).
\end{definition}

\section{Pulau dan perakitan komunitas}

\begin{theorem}[Teori kesetimbangan biogeografi pulau]\label{thm:b1:ecosystem-dynamics:island}
\index{biogeografi pulau}\index{hubungan spesies--luas}
Cacah spesies di sebuah pulau adalah sebuah kesetimbangan dinamik
antara imigrasi dari sebuah lubuk sumber berisi $P$ spesies dan
kepunahan di pulaunya. Bila laju imigrasi spesies barunya turun secara
linear bersama cacah $S$ yang telah hadir, $I = I_0(1 - S/P)$, dan laju
kepunahannya naik secara linear bersamanya, $E = eS$, kesetimbangannya
adalah
\[
S^* = \frac{I_0 P}{I_0 + eP} ,
\]
yang dicapai ketika $I = E$, dengan sebuah pergantian sinambung
sebesar $eS^*$ spesies tiap satuan waktu. Pulau yang dekat sumbernya
memiliki $I_0$ yang lebih tinggi dan lebih banyak spesies; pulau yang
besar memiliki $e$ yang lebih rendah (\omterm{def:b1:populations:population}{populasi} yang lebih besar) dan
lebih banyak spesies. Pada seperangkat pulau cacah spesiesnya naik
bersama luasnya sebagai $S = cA^z$, sebuah garis lurus berkemiringan
$z$ pada sebuah gambar log--log, dengan $z$ lazimnya antara \num{0.2}
dan \num{0.35}.
\end{theorem}
\begin{proof}[Bukti sebagian]
$S$ naik selagi imigrasinya melampaui kepunahannya lalu turun ketika
kepunahannya melampaui imigrasinya, sehingga $\dd S/\dd t = I_0(1 - S/P) - eS$
adalah sebuah persamaan linear orde pertama yang kesetimbangan
tunggalnya, $I_0 -
(I_0/P + e)S = 0$, memberi $S^*$ dan bersifat mantap, sebab laju
perubahannya positif di bawahnya dan negatif di atasnya. Pada
kesetimbangannya spesiesnya masih berdatangan dan lenyap pada laju
$eS^*$: cacahnya mantap, daftarnya tidak. Hukum pangkatnya bersifat
empiris; eksponennya yang dekat \num{0.25} berarti sebuah pulau sepuluh
kali lebih besar memuat sekitar \num{1.8} kali lebih banyak spesies,
dan bahwa kehilangan \qty{90}{\%} sebuah \omterm{def:b1:ecosystem-organization:niche}{habitat} pada akhirnya akan
kehilangan sekitar separuh spesiesnya.
\end{proof}
\begin{proof}[Bukti]
Simberloff dan Wilson (1969) menghitung spesies artropoda pada pulau
kecil bakau di lepas pantai Florida, mengasapi pulaunya untuk membunuh
tiap hewannya, lalu menghitung lagi pada selang waktu tertentu. Dalam
setahun tiap pulaunya telah pulih mendekati cacah spesies aslinya ---
lebih banyak pada pulau yang dekat dan besar, lebih sedikit pada yang
jauh dan kecil --- tetapi dengan sebuah daftar yang berbeda, dan
daftarnya terus berubah: cacahnya berada dalam kesetimbangan,
susunannya dalam pergantian, sebagaimana dituntut teorinya. Krakatau,
yang disterilkan letusannya pada 1883, memiliki \num{30} spesies
tumbuhan sesudah tiga tahun dan lebih dari \num{270} sesudah lima
puluh, sambil mendekati kekayaan pulau serupa di dekatnya.
\end{proof}

\begin{proposition}[Fragmentasi]\label{prop:b1:ecosystem-dynamics:fragmentation}
\index{fragmentasi habitat}
Sebuah \omterm{def:b1:ecosystem-organization:niche}{habitat} yang dipotong menjadi fragmen berperilaku seperti
sebuah kepulauan: tiap fragmennya kehilangan spesies menuju
kesetimbangan yang diizinkan luas dan keterasingannya, perlahan pada
mulanya (\emph{utang kepunahan}nya), dengan spesies bertubuh besar,
jarang dan mengkhusus lebih dahulu. Fragmennya juga memperoleh tepi
--- yang lebih kering, lebih berangin, lebih terganggu --- atas beban
bagian dalamnya; sebuah fragmen persegi seluas \qty{1}{km^2} dengan
sebuah zona tepi \qty{100}{m} memegang \qty{64}{\%} luasnya sebagai
bagian dalam, sebuah fragmen \qty{0.1}{km^2} hanya \qty{15}{\%}.
Koridor antarfragmennya menaikkan laju imigrasinya, yang menjadi sebab
koridor memulihkan spesies lebih cepat daripada menambahkan luas.
\end{proposition}

\begin{method}[Memakai sebuah hubungan spesies--luas]\label{met:b1:ecosystem-dynamics:sar}
\begin{enumerate}
  \item Gambarkan $\log S$ terhadap $\log A$ bagi seperangkat pulau
        atau fragmen yang dicuplik dengan upaya yang setara; paskan
        sebuah garis dengan mata atau kuadrat terkecil.
  \item Bacalah $z$ sebagai kemiringannya dan $c$ sebagai nilai $S$
        pada $A = 1$.
  \item Ramalkan spesies sebuah luas baru sebagai $cA^z$; ramalkan
        fraksi spesies yang tertahan ketika sebuah luas dikurangi dari
        $A_0$ menjadi $A$ sebagai $(A/A_0)^z$ --- dengan $z = 0.25$,
        memegang sepersepuluh \omterm{def:b1:ecosystem-organization:niche}{habitatnya} memegang \qty{56}{\%}
        spesiesnya pada akhirnya.
  \item Berhati-hatilah: kehilangannya dicapai hanya sesudah utang
        kepunahannya dibayar, kadang berpuluh tahun kemudian; dan
        kepasannya tidak mengatakan apa pun tentang spesies mana yang
        pergi.
\end{enumerate}
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:ecosystem-dynamics:1}
Bedakan \omterm{def:b1:ecosystem-dynamics:succession}{suksesi primer} dari \omterm{def:b1:ecosystem-dynamics:succession}{suksesi sekunder} dengan satu contoh
masing-masing, lalu katakan yang mana yang lebih cepat dan mengapa.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:ecosystem-dynamics:2}
Sebutkan ketiga mekanisme penggantian spesiesnya lalu berikan yang
mendominasi pada sebuah aliran lava yang baru.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:ecosystem-dynamics:3}
Definisikan \omterm{def:b1:ecosystem-dynamics:disturbance}{ketahanan} dan \omterm{def:b1:ecosystem-dynamics:disturbance}{kelentingan}, lalu golongkan sebuah terumbu
karang (lambat tumbuh kembali, mudah memutih) dan sebuah padang
rumput.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:ecosystem-dynamics:4}
Dengan $S = 12 A^{0.25}$, hitunglah spesies yang diharapkan pada pulau
seluas \qty{16}{km^2} dan \qty{625}{km^2}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:ecosystem-dynamics:5}
Mengapa cemara di Teluk Glacier tumbuh pada endapan mudanya hanya bila
dipupuk nitrogen, dan apa yang dibuktikan hal ini tentang aldernya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:ecosystem-dynamics:6}
Sebuah lubuk sumber memuat 200 spesies; sebuah pulau menerima $I_0 = 8$
spesies baru setahun ketika kosong lalu kehilangan $e = 0.05$
spesiesnya setahun. Hitunglah cacah kesetimbangannya dan pergantiannya
pada kesetimbangan. Ulangi bagi sebuah pulau dua kali lebih jauh
($I_0 = 4$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:ecosystem-dynamics:7}
Jelaskan, dalam istilah serapan dan mineralisasinya, mengapa lembah
yang ditebang di Hubbard Brook kehilangan nitrat alih-alih nitrogen
organik, dan mengapa kehilangannya berhenti ketika tumbuhannya tumbuh
kembali.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:ecosystem-dynamics:8}
Sehamparan \omterm{def:b1:ecosystem-dynamics:decomposition}{seresah} \omterm{def:b1:flowering-plant-organization:organs}{daun} kehilangan \qty{50}{\%} massanya pada tahun
pertamanya. Bila peluruhannya berorde pertama, hitunglah tetapan
lajunya dan fraksi yang tersisa sesudah tiga tahun; bandingkanlah
dengan \omterm{def:b1:ecosystem-dynamics:decomposition}{seresah} konifer yang kehilangan \qty{15}{\%} setahun.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:ecosystem-dynamics:9}
Pakailah \omterm{prop:b1:ecosystem-dynamics:idh}{hipotesis gangguan menengahnya} untuk meramalkan bagaimana
keanekaragaman sebuah padang rumput berubah seiring penggembalaannya
naik dari nol menjadi berat, lalu jelaskan kedua ujungnya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:ecosystem-dynamics:10}
Sebuah hutan seluas \qty{1000}{km^2} dikurangi menjadi sepuluh fragmen
seluas \qty{10}{km^2}. Dengan $z = 0.25$, hitunglah spesies yang
diharapkan dalam satu fragmen dan dalam kesepuluhnya seandainya
keduanya terasing penuh satu sama lain, bandingkanlah dengan aslinya,
lalu jelaskan mengapa jumlahnya bukanlah jawabannya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:ecosystem-dynamics:11}
Pulau kecil Simberloff--Wilson memulihkan cacah spesiesnya dalam
setahun tetapi dengan spesies yang berbeda. Jelaskan mengapa hal ini
menyokong teori kesetimbangannya lebih baik daripada sebuah pemulihan
daftar aslinya secara tepat.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:ecosystem-dynamics:12}
``Sebuah hutan \omterm{def:b1:ecosystem-dynamics:succession}{klimaks} adalah sebuah sistem tertutup.'' Bahaslah
dengan mengacu pada anggaran Hubbard Brook: apa yang masuk, apa yang
pergi, apa yang didaur ulang, dan apa yang diperlihatkan penebangannya
tentang dari mana penahanannya datang.
\end{exercise}

\section{Soal: Lembah yang Ditebang dan Kepulauannya}

\begin{problem}[{Soal akhir pekan --- anggaran hara sebuah lembah yang
ditebang habis, dan hukum spesies--luas sebuah kepulauan, berakhir pada
eksponen $z$ dan cacah spesies kesetimbangan sebuah pulau}]
\label{pb:b1:ecosystem-dynamics:1}

\medskip
\noindent\textbf{Bagian I --- Membaca sebuah \omterm{def:b1:ecosystem-dynamics:succession}{suksesi}.}
Sebuah kronosekuens tapak berumur tertentu sejak mundurnya sebuah
gletser memberikan: pada 10 tahun, lumut dan sebuah hamparan pemancang
nitrogen, nitrogen tanah \qty{10}{kg/ha}; pada 40 tahun, belukar alder,
\qty{150}{kg/ha}; pada 80 tahun, hutan cemara, \qty{300}{kg/ha}; pada
200 tahun, cemara--hemlock, \qty{320}{kg/ha}.
\begin{enumerate}
  \item Hitunglah laju purata penimbunan nitrogennya pada tiap
        selangnya lalu katakan kapan ia paling cepat.
  \item Mekanisme penggantian yang mana yang ditunjuk kurva
        nitrogennya, dan bagi peralihan yang mana?
  \item Mengapa penimbunannya nyaris berhenti sesudah 80 tahun
        walaupun hutannya masih hidup?
  \item Aldernya memancangkan \qty{50}{kg/ha} nitrogen setahun. Berapa
        fraksinya yang tertahan dalam tanahnya antara tahun 40 dan 80,
        dan ke mana sisanya pergi?
  \item Usulkan sebuah percobaan yang akan membedakan fasilitasi dari
        toleransi bagi peralihan alder ke cemaranya.
  \item Mengapa sebuah kronosekuens menjadi pengganti bagi, dan bukan
        hal yang sama dengan, mengamati satu tapak selama dua abad?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Lembah yang ditebang.}
Dua lembah bersebelahan seluas \qty{15}{ha} masing-masing menerima
\qty{1300}{mm} hujan setahun dengan \qty{6.5}{kg/ha} nitrogen terlarut
di dalamnya. Lembah kendali yang berhutan mengekspor \qty{820}{mm} air
sungai setahun yang membawa \qty{2}{kg/ha} nitrogen nitrat,
\qty{9}{kg/ha} kalsium dan \qty{2}{kg/ha} kalium. Lembah yang ditebang,
pada tahun kedua sesudah penebangannya, mengekspor \qty{1150}{mm} yang
membawa \qty{120}{kg/ha} nitrogen nitrat, \qty{90}{kg/ha} kalsium dan
\qty{36}{kg/ha} kalium. Tanah dan \omterm{def:b1:ecosystem-dynamics:decomposition}{seresahnya} memegang
\qty{3500}{kg/ha} nitrogen, biomassa yang ditebangnya \qty{350}{kg/ha}
lagi.
\begin{enumerate}[resume]
  \item Seberapa besar hasil airnya naik, dan mengapa?
  \item Hitunglah kepekatan purata nitrogen nitrat tiap sungainya
        dalam miligram tiap liter (\qty{1}{mm} air pada \qty{1}{ha} adalah
        \qty{10}{m^3}).
  \item Hitunglah neraca nitrogennya (masukan dikurangi keluaran) tiap
        lembahnya. Yang mana yang menimbun nitrogen, dan pada laju
        berapa?
  \item Berapa fraksi nitrogen tanah lembah yang ditebangnya yang
        pergi pada tahun itu saja? Pada laju itu, berapa lama lubuknya
        akan bertahan?
  \item Berapa kali lipat ekspor kalsium dan kaliumnya naik? Mengapa
        kaliumnya lebih daripada kalsiumnya (pikirkanlah tempat
        masing-masing dipegang dalam sebuah tanah)?
  \item Dari mana nitrat yang diekspornya datang, mengingat tak ada
        tumbuhan yang tumbuh untuk melepaskannya? Sebutkan kedua
        langkah mikrobanya.
  \item Sesudah pertumbuhan kembalinya diizinkan, ekspornya turun
        menjadi \qty{20}{kg/ha} pada tahun 4 dan \qty{5}{kg/ha} pada
        tahun 6. Berapa lama sampai masukan hujannya kembali melampaui
        keluarannya, dan apa yang telah memulihkan penahanannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Kepulauannya.}
Lima pulau seluas \qtylist{1;10;100;1000;10000}{km^2} membawa \num{12}, \num{21}, \num{38}, \num{68} dan
\num{120} spesies burung darat. Lubuk daratannya berisi \num{300}
spesies.
\begin{enumerate}[resume]
  \item Hitunglah $\log_{10}$ tiap luasnya dan tiap cacah spesiesnya
        lalu gambarkan keduanya.
  \item Perolehlah $z$ dari kedua titik ujungnya dan $c$ dari pulau
        terkecilnya; periksalah pulau tengahnya.
  \item Ramalkan cacah spesies sebuah pulau seluas \qty{250}{km^2}.
  \item Pulau \qty{100}{km^2}-nya menerima $I_0 = 6$ spesies baru
        setahun ketika kosong lalu kehilangan sebuah fraksi $e$
        spesiesnya setahun. Dengan memakai $S^* = 38$, hitunglah $e$
        dan pergantiannya pada kesetimbangan.
  \item Sebuah pulau seluas sama yang dua kali lebih jauh memiliki
        $I_0 = 3$. Hitunglah cacah spesies kesetimbangannya dengan $e$
        yang sama.
  \item Sebuah jembatan menghubungkan pulau \qty{100}{km^2}-nya ke
        daratannya dan $I_0$ naik menjadi \num{30}. Hitunglah
        kesetimbangan barunya lalu jelaskan mengapa ia masih jauh di
        bawah \num{300}.
  \item Bila $z$ bernilai \num{0.35} alih-alih \num{0.25}, akankah
        pulau kecilnya memiliki lebih banyak atau lebih sedikit spesies
        relatif terhadap yang besar? Katakan apa arti sebuah $z$ yang
        tinggi secara biologis.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Fragmentasi dan pemulihan.}
\begin{enumerate}[resume]
  \item Sebuah cagar seluas \qty{1000}{km^2} direncanakan dalam sebuah
        hutan seluas \qty{100000}{km^2} yang selebihnya akan
        dibersihkan. Dengan $z =
        0.25$, berapa fraksi spesies hutannya yang pada akhirnya
        sanggup dipegang cagarnya?
  \item Akankah sepuluh cagar seluas \qty{100}{km^2} memegang lebih
        banyak atau lebih sedikit, bila keduanya terasing? Bila
        keduanya terhubung koridor?
  \item Zona tepi cagarnya selebar \qty{300}{m}. Hitunglah fraksi
        bagian dalam sebuah cagar persegi seluas \qty{1000}{km^2} dan
        sebuah cagar seluas \qty{100}{km^2}.
  \item Sketsalah cacah spesies sebuah fragmen yang baru terasing
        terhadap waktu, lalu jelaskan istilah utang kepunahannya.
  \item Nyatakan hasilnya: eksponen $z$ kepulauannya dan cacah spesies
        kesetimbangan pulau \qty{100}{km^2}-nya sebelum dan sesudah
        jembatannya.
\end{enumerate}
\end{problem}
