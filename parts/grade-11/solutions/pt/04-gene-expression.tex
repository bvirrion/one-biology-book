\section*{Capítulo \ref{ch:g11:gene-expression} --- Do gene à proteína}

\begin{solution}{exo:g11:gene-expression:1}
O \omterm{def:g11:gene-expression:transcription}{RNA} é de fita simples, o açúcar dele é a ribose, e ele usa uracila no
lugar da timina (ele também é de vida curta e deixa o \omterm{def:g10:cells-common-unit:organelle}{núcleo}).
\end{solution}

\begin{solution}{exo:g11:gene-expression:2}
\texttt{5'-AUG GCU UUU UAA-3'}.
\end{solution}

\begin{solution}{exo:g11:gene-expression:3}
Met--Gly--Lys--Cys, depois parada.
\end{solution}

\begin{solution}{exo:g11:gene-expression:4}
O \omterm{def:g10:cells-common-unit:organelle}{ribossomo} lê o mRNA \omterm{def:g11:gene-expression:code}{códon} a \omterm{def:g11:gene-expression:code}{códon} e une os \omterm{def:g10:chemistry-of-life:families}{aminoácidos} numa cadeia;
cada \omterm{def:g11:gene-expression:transcription}{RNA} transportador reconhece um \omterm{def:g11:gene-expression:code}{códon} com o anticódon dele e traz o
\omterm{def:g10:chemistry-of-life:families}{aminoácido} correspondente.
\end{solution}

\begin{solution}{exo:g11:gene-expression:5}
Com 4 letras, palavras de duas dão $4^2 = 16$ combinações, menos que os
20 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}; palavras de três dão 64, o que basta.
\end{solution}

\begin{solution}{exo:g11:gene-expression:6}
$412 \times 3 + 3 = 1239$ \omterm{def:g10:universal-dna:nucleotide}{nucleotídeos}; um \omterm{def:g10:universal-dna:gene}{gene} de pelo menos 1239
pares de bases (mais num eucarionte, com \omterm{prop:g11:gene-expression:splicing}{íntrons}).
\end{solution}

\begin{solution}{exo:g11:gene-expression:7}
Original: Met--Lys--Gly--Trp. Mutante: UGA é parada, logo
Met--Lys--Gly: uma \omterm{def:g11:mutations:mutation}{mutação} sem sentido, que trunca a \omterm{def:g11:gene-expression:protein}{proteína}.
\end{solution}

\begin{solution}{exo:g11:gene-expression:8}
O \texttt{AAG} continua sendo Lys: silenciosa, Met--Lys--Gly--Trp
inalterada. A segunda é uma inserção de G que desloca o quadro:
\texttt{AUG AAA GGG
CUG GUA A}\dots{} Met--Lys--Gly--Leu--Val\dots, um deslocamento do
quadro de leitura sem parada no fragmento.
\end{solution}

\begin{solution}{exo:g11:gene-expression:9}
Na maioria dos blocos da tabela, os quatro \omterm{def:g11:gene-expression:code}{códons} que compartilham as
duas primeiras letras especificam o mesmo \omterm{def:g10:chemistry-of-life:families}{aminoácido} (Pro, Thr, Ala,
Gly, Val, Ser, Leu\dots): mudar a terceira letra não muda nada então. A
degeneração do código está principalmente na terceira posição.
\end{solution}

\begin{solution}{exo:g11:gene-expression:10}
Lido de três em três, \texttt{UCU CUC UCU CUC}\dots{} alterna dois
\omterm{def:g11:gene-expression:code}{códons}, UCU e CUC, de modo que a \omterm{def:g11:gene-expression:protein}{proteína} alterna dois \omterm{def:g10:chemistry-of-life:families}{aminoácidos} ---
como se observou. Com um código de duas letras a mesma repetição daria
um único \omterm{def:g11:gene-expression:code}{códon} e um único \omterm{def:g10:chemistry-of-life:families}{aminoácido}. Como o poli-U dá Phe e, na
tabela, UCU é Ser e CUC é Leu, o experimento atribui UCU a Ser e CUC a
Leu (ou o inverso, resolvido por outros experimentos).
\end{solution}

\begin{solution}{exo:g11:gene-expression:11}
Só 1503 pares de bases são necessários à sequência codificadora; o
resto são \omterm{prop:g11:gene-expression:splicing}{íntrons}, transcritos e depois recortados do pré-mRNA, mais
sequências reguladoras nas extremidades do \omterm{def:g10:universal-dna:gene}{gene}.
\end{solution}

\begin{solution}{exo:g11:gene-expression:12}
Bloquear a \omterm{prop:g11:gene-expression:translation}{tradução} mata a bactéria e deixa as \omterm{def:g10:cells-common-unit:cell}{células} do paciente
funcionando. A seletividade do medicamento implica que os \omterm{def:g10:cells-common-unit:organelle}{ribossomos}
bacterianos e os humanos, embora façam o mesmo trabalho com o mesmo
código, diferem o bastante em estrutura para uma molécula se ligar a um
e não ao outro --- uma diferença acumulada desde a separação deles.
\end{solution}

\begin{solution}{exo:g11:gene-expression:13}
A \omterm{def:g11:gene-expression:transcription}{RNA} polimerase do camundongo transcreveu o \omterm{def:g10:universal-dna:gene}{gene} da água-viva, e os
\omterm{def:g10:cells-common-unit:organelle}{ribossomos} e os tRNAs dele traduziram o mRNA com a mesma tabela de
\omterm{def:g11:gene-expression:code}{códons}, de modo que foi fabricada a mesma sequência de \omterm{def:g10:chemistry-of-life:families}{aminoácidos} e,
portanto, a mesma \omterm{def:g11:gene-expression:protein}{proteína} fluorescente. Com um código diferente, os
\omterm{def:g11:gene-expression:code}{códons} seriam lidos como outros \omterm{def:g10:chemistry-of-life:families}{aminoácidos} e a \omterm{def:g11:gene-expression:protein}{proteína} seria uma
cadeia sem sentido e sem fluorescência.
\end{solution}

\begin{solution}{exo:g11:gene-expression:14}
O UAG deixaria de interromper a \omterm{prop:g11:gene-expression:translation}{tradução}: toda \omterm{def:g11:gene-expression:protein}{proteína} cujo \omterm{def:g10:universal-dna:gene}{gene}
termina em UAG seria estendida além do fim normal dela, muitas vezes
perdendo a função --- um prejuízo generalizado. Mas um \omterm{def:g10:universal-dna:gene}{gene} truncado
por um UAG prematuro passaria a ser lido até o fim, restaurando uma
\omterm{def:g11:gene-expression:protein}{proteína} de comprimento quase completo: a segunda \omterm{def:g11:mutations:mutation}{mutação} suprime a
primeira.
\end{solution}

\begin{solution}{exo:g11:gene-expression:15}
Um deslocamento do quadro embaralha tudo o que vem depois e em geral
encontra uma parada em poucos \omterm{def:g11:gene-expression:code}{códons}: perto do início, praticamente
nenhuma \omterm{def:g11:gene-expression:protein}{proteína} correta é fabricada; perto do fim, a maior parte da
\omterm{def:g11:gene-expression:protein}{proteína} fica intacta e ainda pode se dobrar e funcionar. Uma \omterm{def:g11:mutations:mutation}{mutação}
de sentido trocado muda um \omterm{def:g10:chemistry-of-life:families}{aminoácido}: inofensiva se aquela posição
tolerar a mudança, fatal se ela estiver no sítio ativo ou romper o
dobramento --- o efeito depende de qual \omterm{def:g10:chemistry-of-life:families}{aminoácido}, e onde.
\end{solution}

\begin{solution}{pb:g11:gene-expression:1}
\textbf{1.} \texttt{3'-TAC CGA ACC TTT GGG CTT ATG CCA GTA AAG ACT-5'}.

\textbf{2.} \texttt{AUG GCU UGG AAA CCC GAA UAC GGU CAU UUC UGA}.

\textbf{3.} Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--His--Phe, parada.

\textbf{4.} 10 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}; 33 \omterm{def:g10:universal-dna:nucleotide}{nucleotídeos} (11 \omterm{def:g11:gene-expression:code}{códons}, incluindo o de
parada).

\textbf{5.} O AUG; a posição desse primeiro AUG fixa o quadro, e todo
\omterm{def:g11:gene-expression:code}{códon} seguinte é lido em compasso a partir dele.

\textbf{6.} O CCG continua sendo Pro: silenciosa, \omterm{def:g11:gene-expression:protein}{proteína} inalterada.

\textbf{7.} CGU é Arg no lugar de His: de sentido trocado,
Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--Arg--Phe.

\textbf{8.} UGA é parada: Met--Ala, sem sentido. Uma única substituição
de G por A no terceiro \omterm{def:g11:gene-expression:code}{códon} (TGG para TGA).

\textbf{9.} mRNA \texttt{AUG GCU UUG GAA ACC CGA AUA CGG UCA UUU CUG
A}: Met--Ala--Leu--Glu--Thr--Arg--Ile--Arg--Ser--Phe--Leu\dots, sem
parada no fragmento: deslocamento do quadro, com todo \omterm{def:g10:chemistry-of-life:families}{aminoácido}
depois do segundo trocado.

\textbf{10.} Mutante 1 (silenciosa, sem mudança) $<$ mutante 2 (um
\omterm{def:g10:chemistry-of-life:families}{aminoácido} trocado, função possivelmente conservada) $<$ mutante 4
(deslocamento do quadro, todos os \omterm{def:g10:chemistry-of-life:families}{aminoácidos} errados menos dois)
$\approx$ mutante 3 (sem sentido, um fragmento de dois \omterm{def:g10:chemistry-of-life:families}{aminoácidos}: a
\omterm{def:g11:gene-expression:protein}{proteína} se foi).

\textbf{11.} Leu 6, Ser 6, Trp 1. O Trp, com um único \omterm{def:g11:gene-expression:code}{códon}, é trocado
por qualquer substituição; a Leu ou a Ser, com seis, quase sempre não.

\textbf{12.} CUU, CUC, CUA e CUG são todos Leu: cada uma das 3
substituições possíveis na terceira posição é silenciosa --- 100\%.

\textbf{13.} Duas letras dão 16 \omterm{def:g11:gene-expression:code}{códons}, poucos demais para 20
\omterm{def:g10:chemistry-of-life:families}{aminoácidos} mais uma parada; três dão 64, já mais que suficiente, de
modo que quatro (256) seria desperdício.

\textbf{14.} UUU $=$ Phe e AAA $=$ Lys.

\textbf{15.} Um \omterm{def:g10:universal-dna:gene}{gene} transferido entre \omterm{def:g10:biodiversity-scales:species}{espécies} é lido de forma
idêntica, de modo que a \omterm{prop:g10:universal-dna:universal}{transgênese} funciona; e um código
compartilhado, herdado sem mudança, é evidência de que todos os
organismos descendem de um ancestral comum que já o usava.

\textbf{16.} 10 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}: \qty{1}{s}. 600 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}: \qty{60}{s}.

\textbf{17.} $1800/80 \approx 22$ \omterm{def:g10:cells-common-unit:organelle}{ribossomos} ao mesmo tempo.

\textbf{18.} $7200/8 = 900$ cópias.

\textbf{19.} Um mRNA dá $600/8 = 75$ cópias em 10 minutos; são
necessários cerca de 27 mRNAs.

\textbf{20.} A \omterm{def:g11:gene-expression:protein}{proteína}
Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--His--Phe; o mutante 3, a única
substituição de G por A que cria uma parada no terceiro \omterm{def:g11:gene-expression:code}{códon}, a
aboliu; um \omterm{def:g10:cells-common-unit:organelle}{ribossomo} constrói os dez \omterm{def:g10:chemistry-of-life:families}{aminoácidos} em cerca de um segundo.
\end{solution}
