\chapter{रक्त की ग्लूकोज और मधुमेह}\label{ch:g12:glucose-and-diabetes}

पाँच लीटर रक्त, और हर एक में एक ग्राम ग्लूकोज: यानी कुल पाँच ग्राम, लगभग एक
चीनी की डली। अकेला मस्तिष्क वही मात्रा हर घंटे जला देता है, दिन-रात, और
इसके अलावा कुछ भी इस्तेमाल नहीं कर सकता। फिर भी जो व्यक्ति दिन भर कुछ न
खाए, या एक ही बार में पूरा केक खा ले, वह उस सांद्रता को उसी मान के प्रति
लीटर कुछ दसवें ग्राम के भीतर रखता है --- और जिस व्यक्ति का अग्न्याशय चूक
गया हो वह नहीं रख पाता, जिसके नतीजे प्यास से लेकर कोमा तक हैं। यह अध्याय
शरीर की सबसे क़रीब से देखी जाती संख्या के बारे में है, उन अंगों के बारे में
जो उसे स्थिर रखते हैं, और इस बारे में कि \omterm{def:g12:glucose-and-diabetes:diabetes}{मधुमेह} के दोनों रूपों में, जब वे
नहीं रख पाते, तो क्या होता है।

\section{एक नियंत्रित स्थिरांक}

\begin{definition}[ग्लाइसीमिया]\label{def:g12:glucose-and-diabetes:glycaemia}
\emph{ग्लाइसीमिया}\index{ग्लाइसीमिया} रक्त प्लाज़्मा में ग्लूकोज की
सांद्रता है: किसी स्वस्थ व्यक्ति में भोजन से पहले लगभग \qty{1}{g/L}
(\qty{5.5}{mmol/L}), जो खाने के एक घंटे के भीतर चढ़कर \qty{1.4}{g/L} हो जाती
है और दो के भीतर लौट आती है, तथा एक दिन के उपवास के बाद गिरकर
\qty{0.7}{g/L} रह जाती है। \qty{0.5}{g/L} से नीचे मस्तिष्क चूक जाता है
(भ्रम, फिर कोमा); \qty{1.8}{g/L} से ऊपर ग्लूकोज मूत्र में छलक जाती है; और
\qty{1.3}{g/L} से ऊपर बीते साल वाहिकाओं तथा तंत्रिकाओं को नुक़सान
पहुँचाते हैं। ग्लाइसीमिया एक \emph{नियंत्रित} राशि है: यानी ऐसा मान जिसे
कोई तंत्र किसी निर्धारित बिंदु के पास थामे रखता है, तथा उसे नापता और सुधारता
है।
\end{definition}

\begin{proposition}[ग्लूकोज का कुंड और उसके बहाव]\label{prop:g12:glucose-and-diabetes:pool}
रक्त की वह \qty{5}{g} ग्लूकोज एक कुंड है जिससे होकर बड़े बहाव गुज़रते हैं।
\emph{इनपुट}: भोजन के बाद आँत (एक घंटे में \qty{100}{g} तक), और भोजनों के
बीच \emph{यकृत}, जो अपने जमा \emph{\omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन}} (\qty{100}{g}, यानी आधे दिन की
आपूर्ति) से ग्लूकोज छोड़ता है तथा \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्लों} और लैक्टेट से नई ग्लूकोज
बनाता है। \emph{आउटपुट}: मस्तिष्क (\qty{5}{g/h}, स्थिर), पेशियाँ (काम पर
कुछ ग्रामों से लेकर \qty{100}{g/h} तक), बाक़ी ऊतक, तथा यकृत और पेशियों में
\omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} या वसा की तरह जमाव। \qty{5}{g} का कोई कुंड, जिसमें प्रति घंटा
दसियों ग्राम भरे और निकाले जाते हों, केवल इसलिए स्थिर है क्योंकि वे बहाव
मिनट दर मिनट सधे रहते हैं।
\end{proposition}
\admitted

\begin{example}[ग्लाइसीमिया का एक दिन]\label{ex:g12:glucose-and-diabetes:day}
8 बजे नाश्ता: \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} \num{0.9} से चढ़कर 9 बजे तक \qty{1.3}{g/L} हो
जाती है और 10 बजे तक \num{1.0} पर लौट आती है, क्योंकि वह आधिक्य यकृत तथा
पेशी के \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} में चला गया। 10 से 13 तक यकृत \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} छोड़ता है और वह
स्तर \num{0.9} के पास थमा रहता है। 17 बजे की दौड़ एक घंटे में \qty{60}{g}
पेशियों में खींच लेती है; यकृत क़दम मिलाने के लिए अपना आधा भंडार ख़ाली कर
देता है, और वह स्तर \num{0.8} तक बैठ जाता है। रात भर यकृत \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्लों} से
नई ग्लूकोज बनाता है, और 7 बजे वह स्तर \qty{0.9}{g/L} है। पाँच के किसी कुंड
से तीन सौ ग्राम गुज़र चुके हैं।
\end{example}

\section{अग्न्याशय संतुलन थामे है}

\begin{proposition}[द्वीपिकाओं के दो हॉर्मोन]\label{prop:g12:glucose-and-diabetes:hormones}
अग्न्याशय भर बिखरे, पाचक \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} बनाती \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के बीच, \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के दस लाख
छोटे गुच्छे पड़े हैं, यानी \emph{द्वीपिकाएँ}\index{अग्न्याशय की द्वीपिकाएँ}।
उनकी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} बग़ल से बहते रक्त का \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} नापती हैं और उसी में दो
\omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} स्रावित करती हैं:
\begin{itemize}
  \item \emph{इंसुलिन}\index{इंसुलिन}, जो $\beta$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से आता है, जब
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} चढ़े: वह पेशी तथा वसा \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से ग्लूकोज उठवाता है,
        और यकृत तथा पेशियों से उसे \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} की तरह जमा कराता है; और
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} गिर जाती है;
  \item \emph{ग्लूकैगॉन}\index{ग्लूकैगॉन}, जो $\alpha$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से आता है,
        जब वह गिरे: वह यकृत से \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} तुड़वाता है और ग्लूकोज छुड़वाता
        है; और \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} चढ़ जाती है।
\end{itemize}
ये दोनों उन्हीं अंगों पर उल्टी दिशाओं में काम करते हैं; और उनका संतुलन, जो
नापी गई \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} से हर कुछ मिनट पर फिर बिठाया जाता है, वही नियंत्रण है।
\omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} इकलौता \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} है जो \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} घटाता है; उसे चढ़ाने वाले कई हैं।
\end{proposition}
\begin{proof}[साक्ष्य]
किसी कुत्ते का अग्न्याशय निकाल देना (1889) उसे दिन भर में मधुमेही बना देता
है: उसका \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} तिगुना हो जाता है और उसके मूत्र में ग्लूकोज प्रकट होती
है; जबकि पाचक \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} ढोती नलिका बाँध देने से ऐसा नहीं होता, इसलिए वह असर
पाचन वाला नहीं है। \omterm{prop:g12:glucose-and-diabetes:hormones}{द्वीपिकाओं} का कोई सत्त लगाने (1921) से किसी मधुमेही
कुत्ते का, और किसी मधुमेही बच्चे का, \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} घंटों के भीतर गिर जाता है;
और वह सक्रिय पदार्थ, \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन}, उसी साल शुद्ध किया गया। किसी तश्तरी में अलग की
गई \omterm{prop:g12:glucose-and-diabetes:hormones}{द्वीपिकाएँ} माध्यम की ग्लूकोज चढ़ाने पर \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} स्रावित करती हैं और घटाने
पर \omterm{prop:g12:glucose-and-diabetes:hormones}{ग्लूकैगॉन}। और लगाया गया \omterm{prop:g12:glucose-and-diabetes:hormones}{ग्लूकैगॉन} किसी उपवासी व्यक्ति का \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया}
मिनटों के भीतर चढ़ा देता है, पर तभी जब यकृत \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} थामे हो।
\end{proof}

\begin{example}[काम पर वह फंदा]\label{ex:g12:glucose-and-diabetes:loop}

छोड़ती हैं, पेशियाँ तथा वसा \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} अपने ग्लूकोज परिवाहक खोल देती हैं,
यकृत \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} बनाने पर आ जाता है, और वह स्तर वापस गिर जाता है --- और उसी
वक़्त \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} का स्राव भी गिर जाता है। भोजनों के बीच नीचे की ओर बहाव
\omterm{prop:g12:glucose-and-diabetes:hormones}{ग्लूकैगॉन} से सँभाला जाता है, \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} तोड़ा जाता है, और वह स्तर थमा रहता
है। कोई भी \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} कभी ग़ैरहाज़िर नहीं होता; उनका अनुपात हिलता है, और
\omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} एक सँकरी पट्टी के भीतर उसके पीछे चलती है।
\end{example}

\section{जब वह फंदा चूक जाए: मधुमेह}

\begin{definition}[मधुमेह]\label{def:g12:glucose-and-diabetes:diabetes}
\emph{मधुमेह}\index{मधुमेह} \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} का चिरकालिक आधिक्य है: उपवास में
\qty{1.26}{g/L} से ऊपर, या किसी मानक ग्लूकोज भार के दो घंटे बाद
\qty{2}{g/L} से ऊपर। उसके शुरुआती चिह्न उसी आधिक्य से निकलते हैं: मूत्र में
ग्लूकोज, जो अपने साथ पानी खींचती है (भरपूर मूत्र, प्यास), और वज़न का घटना
जब \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} अपने इर्द-गिर्द की ग्लूकोज इस्तेमाल न कर सकें। और उसका देर से
आता नुक़सान --- \omterm{def:g11:the-eye:parts}{आँख} तथा गुर्दे की छोटी वाहिकाओं को, तंत्रिकाओं को, धमनियों
को --- सालों तक ग्लूकोज के प्रोटीनों से क्रिया करने से आता है। इस नाम को दो
बीमारियाँ साझा करती हैं।
\end{definition}

\begin{proposition}[प्रकार 1 और प्रकार 2]\label{prop:g12:glucose-and-diabetes:types}
\begin{itemize}
  \item \emph{प्रकार 1 \omterm{def:g12:glucose-and-diabetes:diabetes}{मधुमेह}} (लगभग 10\% मामले) रोगी के अपने ही
        प्रतिरक्षा तंत्र द्वारा $\beta$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} का विनाश है, और वह भी आम
        तौर पर बचपन या जवानी में: \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} ग़ैरहाज़िर होता है।
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} बेरोक चढ़ती है, काम आती ग्लूकोज न होने से वसा जलाई
        जाती है और उसके अम्लीय उत्पाद जमा होते हैं; और लगाए गए \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} के
        बिना यह बीमारी महीनों में जानलेवा है। इलाज है \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन}, दिन में कई
        बार, भोजनों तथा नापी गई \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} के हिसाब से ख़ुराक में, जीवन
        भर।
  \item \emph{प्रकार 2 \omterm{def:g12:glucose-and-diabetes:diabetes}{मधुमेह}} (लगभग 90\%) वयस्कों में पनपता है, और वह भी
        अधिकतर मोटे तथा निष्क्रिय लोगों में: वे ऊतक \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} को कम से कम
        जवाब देते हैं (\emph{\omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} प्रतिरोध}), $\beta$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} अधिक
        स्रावित करके उसकी भरपाई करती हैं, और सालों बाद वे थक जाती हैं।
        \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} मौजूद है पर काफ़ी नहीं। वह प्रवृत्ति आंशिक रूप से आनुवंशिक
        है (\cref{ch:g11:genes-and-disease}); उसे छेड़ने वाली चीज़ जीने का
        ढंग है, और पहला इलाज उसी का बदलना है --- यानी वज़न घटाना तथा
        कसरत --- दवाओं से और, देर में, \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} से पहले।
\end{itemize}
\end{proposition}
\begin{proof}[साक्ष्य]
प्रकार 1 के रोगियों में लक्षणों से सालों पहले अपनी ही \omterm{prop:g12:glucose-and-diabetes:hormones}{द्वीपिका} \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के
ख़िलाफ़ प्रतिरक्षी होते हैं, और निदान पर उनकी \omterm{prop:g12:glucose-and-diabetes:hormones}{द्वीपिकाएँ} $\beta$
\omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से लगभग ख़ाली तथा श्वेत \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से भरी होती हैं; उनके रक्त का
\omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} पकड़ में ही नहीं आता और वे लगाए गए \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} को तुरंत जवाब देते हैं।
प्रकार 2 के रोगियों में निदान पर \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} सामान्य या ऊँचा होता है; उनकी पेशी
तथा वसा \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} किसी दी गई \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} ख़ुराक पर किसी स्वस्थ व्यक्ति से कम
ग्लूकोज उठाती हैं; समरूप जुड़वाँ 70\% मामलों में यह बीमारी साझा करते हैं;
और बहुत-से शुरुआती मामलों में अकेला वज़न घटाना या कोई पैदल चलने का कार्यक्रम
सामान्य \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} लौटा देता है।
\end{proof}

\begin{method}[कोई ग्लाइसीमिया अभिलेख पढ़ना]\label{met:g12:glucose-and-diabetes:read}
\begin{enumerate}
  \item निर्धारित बिंदु और वह पट्टी पहचानिए: उपवास में \qtyrange{0.7}{1.1}{g/L},
        भोजनों के बाद \num{1.4} से नीचे; किसी ग्लूकोज भार के दो घंटे बाद
        \num{1.4} से नीचे (स्वस्थ), \qtyrange{1.4}{2}{} (बिगड़ी), और
        \qty{2}{g/L} से ऊपर (\omterm{def:g12:glucose-and-diabetes:diabetes}{मधुमेह})।
  \item हर चढ़ाव के लिए पूछिए कि ग्लूकोज किसने डाली (भोजन, यकृत) और उसे
        कौन निकालना चाहिए (\omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} के निशाने); और हर गिरावट के लिए, कि उसे
        किसने निकाला (मस्तिष्क, पेशी) और उसे वापस कौन डालना चाहिए
        (\omterm{prop:g12:glucose-and-diabetes:hormones}{ग्लूकैगॉन}, यकृत)।
  \item \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोनों} को ग्लूकोज के साथ पढ़िए: \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} के साथ चढ़ना
        चाहिए और \omterm{prop:g12:glucose-and-diabetes:hormones}{ग्लूकैगॉन} गिरना; ग़ैरहाज़िर \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} का मतलब प्रकार 1, और
        ऊँची \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} के साथ ऊँचे \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} का मतलब प्रतिरोध।
  \item ज़रूरत हो तो इकाइयाँ बदलिए: ग्लूकोज की \qty{1}{g/L}
        \qty{5.55}{mmol/L} है।
\end{enumerate}
\end{method}

\begin{remark}[नियंत्रण का एक नमूना]\label{rem:g12:glucose-and-diabetes:model}
संवेदक, निर्धारित बिंदु, दो उल्टे प्रभावक, ऋणात्मक \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि}: \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} का
नियंत्रण \cref{ch:g11:hormones-and-reproduction} में \omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} के
नियंत्रण जैसी ही बनावट है, और शरीर के अधिकांश स्थिरांकों की भी --- ताप,
पानी, रक्तचाप, अम्लता। \omterm{def:g12:glucose-and-diabetes:diabetes}{मधुमेह} वही है जो कोई नियंत्रित स्थिरांक अपना फंदा
टूट जाने पर दिखता है: वह मान इनपुट को अब भी जवाब देता है, पर कोई चीज़ उसे
वापस नहीं लाती। इस अध्याय के वक्र पढ़ना उनमें से किसी को भी पढ़ने का अभ्यास
है।
\end{remark}

और बताइए कि हर रोगी की \omterm{prop:g12:glucose-and-diabetes:hormones}{द्वीपिकाएँ} क्या कर रही हैं।

जाता, तो \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} क्या हो जाती? समझाइए कि ऐसा कभी क्यों नहीं होता, और

\omterm{prop:g11:becoming-male-female:hormones}{टेस्टोस्टेरोन} के नियंत्रण से तुलना कीजिए: संवेदक, निर्धारित बिंदु,
प्रभावक, \omterm{def:g11:hormones-and-reproduction:hormone}{प्रतिपुष्टि} का चिह्न। दो उल्टे \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} होने में ख़ास क्या है?
\end{exercise}

\section{समस्या: चीनी के पाँच ग्राम}

\begin{problem}[{सप्ताहांत समस्या --- रक्त की ग्लूकोज का एक दिन भर का
हिसाब: वह कुंड और उसके बहाव, एक सह्यता परीक्षण पढ़ा गया, इंसुलिन की एक
ख़ुराक निकाली गई, और एक फंदा जिसे हाथ से बदलना पड़ता है}]
\label{pb:g12:glucose-and-diabetes:1}
\qty{5}{L} रक्त आयतन, प्रति घंटा \qty{5}{g} ग्लूकोज की मस्तिष्क खपत,
\qty{100}{g} का यकृत \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} भंडार, और \qty{400}{g} पेशी \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} लीजिए।
तेज़ \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} की एक इकाई कोई \qty{10}{g} \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} ठिकाने लगाती है।

\medskip
\noindent\textbf{भाग I --- वह कुंड।}
\begin{enumerate}
  \item \qty{1}{g/L} पर रक्त की ग्लूकोज का द्रव्यमान निकालिए, और वही
        सांद्रता \unit{mmol/L} में भी (ग्लूकोज
        \qty{180}{g/mol})।
  \item कोई चीज़ उसकी जगह न ले, तो रक्त की ग्लूकोज अकेले मस्तिष्क के लिए
        कितनी देर चलती?
  \item भोजनों के बीच यकृत उसकी जगह लेता है। यकृत का \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} मस्तिष्क के
        लिए कितनी देर चलता है? उसके बाद क्या होता है?
  \item पेशी \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} यकृत वाले से चार गुना है, पर पेशियाँ रक्त में
        ग्लूकोज नहीं छोड़ सकतीं। समझाइए कि पेशी \omterm{def:g10:exercise-and-energy:glycogen}{ग्लाइकोजन} मस्तिष्क की मदद
        क्यों नहीं करता, और वह किसलिए है।
  \item कोई भोजन एक घंटे पर \qty{60}{g} ग्लूकोज पहुँचाता है। उसमें से कुछ भी
        न निकाला जाए, तो \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} कहाँ तक पहुँचती? असल में उसे कौन
        निकालता है, और वह कहाँ जाती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वह परीक्षण।}
कोई रोगी \qty{75}{g} ग्लूकोज पीता है। नापे गए \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} और
\omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन}:

\begin{center}
\begin{tabular}{lcccccc}
मिनट & 0 & 30 & 60 & 90 & 120 & 180 \\ \hline
\omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} (g/L) & 1.15 & 1.8 & 2.05 & 2.0 & 1.85 & 1.6 \\
\omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} (किसी स्वस्थ चोटी का \%) & 90 & 140 & 180 & 200 & 190 & 160 \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item \cref{met:g12:glucose-and-diabetes:read} की दहलीज़ों से उस रोगी का
        वर्गीकरण कीजिए।
  \item क्या \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} की कमी है? तो फिर क्या चूक रहा है?
  \item उस बीमारी का नाम दीजिए, और उस रोगी के इतिहास की दो ऐसी विशेषताएँ
        भी जिनकी आप उम्मीद करेंगे।
  \item बिना इलाज वाले किसी प्रकार 1 रोगी के लिए वही तालिका कैसी दिखती?
  \item पहला कौन-सा इलाज बताया जाता है, और वह किस क्रियाविधि से काम करता
        है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- फंदा हाथ से चलाना।}
प्रकार 1 की किसी रोगी का \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} रात के भोजन से पहले \qty{1.0}{g/L} है;
और उस भोजन में \qty{90}{g} \omterm{def:g10:chemistry-of-life:families}{कार्बोहाइड्रेट} है।
\begin{enumerate}[resume]
  \item उस भोजन के लिए तेज़ \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} की ख़ुराक निकालिए।
  \item वह उसे लगाती है पर भोजन का केवल आधा ही खाती है। आँकिए कि
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} कहाँ तक गिर सकती है, और उसके लक्षण तथा उपाय बताइए।
  \item वह कुछ नहीं लगाती और पूरा भोजन खाती है। वे \qty{90}{g} सब किसी
        \qty{5}{L} कुंड में घुस जाएँ तो \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} आँकिए, और बताइए कि
        \qty{1.8}{g/L} से ऊपर गुर्दे क्या करते हैं।
  \item उसे रात में, जब वह कुछ नहीं खाती, धीमा \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} भी क्यों लगाना पड़ता
        है?
  \item समझाइए कि वह मीटर तथा वह पेन किस अर्थ में $\beta$ \omterm{def:g10:cells-common-unit:cell}{कोशिका} की जगह
        लेते हैं, और वे किसकी जगह नहीं ले सकते।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वह फंदा, तर्क से।}
\begin{enumerate}[resume]
  \item कोई दवा $\beta$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} को अधिक \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} स्रावित करने के लिए
        उकसाती है। वह किस प्रकार के \omterm{def:g12:glucose-and-diabetes:diabetes}{मधुमेह} के लिए काम कर सकती है, और
        दूसरे के लिए क्यों नहीं?
  \item प्रकार 2 वाला कोई व्यक्ति \qty{10}{kg} घटाता है और रोज़ एक घंटा
        पैदल चलता है; छह महीने बाद उसका सह्यता परीक्षण सामान्य है।
        समझाइए कि उस फंदे का कौन-सा हिस्सा ठीक हुआ।
  \item समझाइए कि किसी स्वस्थ व्यक्ति के \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} और \omterm{prop:g12:glucose-and-diabetes:hormones}{ग्लूकैगॉन} दोनों कभी
        ऊँचे क्यों नहीं होते, और किसी तश्तरी की \omterm{prop:g12:glucose-and-diabetes:hormones}{द्वीपिका} माध्यम की ग्लूकोज
        \qty{1}{g/L} पर बिठाने पर क्या करती है।
  \item \omterm{prop:g12:glucose-and-diabetes:hormones}{ग्लूकैगॉन}, एड्रीनलीन, कॉर्टिसोल और वृद्धि \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} सब \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया}
        चढ़ाते हैं; केवल \omterm{prop:g12:glucose-and-diabetes:hormones}{इंसुलिन} उसे घटाता है। प्रस्तावित कीजिए कि शरीर के
        पास नीची \omterm{def:g12:glucose-and-diabetes:glycaemia}{ग्लाइसीमिया} के ख़िलाफ़ कई सुरक्षाएँ और ऊँची के ख़िलाफ़
        एक ही क्यों है।
  \item परिणाम लिखिए: रक्त की ग्लूकोज, वे घंटे जो यकृत ढँक सकता है, भाग II
        का निदान, और भाग III की ख़ुराक।
\end{enumerate}
\end{problem}
