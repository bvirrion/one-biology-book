\chapter{La glycémie et le diabète}\label{ch:g12:glucose-and-diabetes}

Cinq litres de sang, un gramme de glucose dans chacun~: cinq grammes
en tout, à peu près un morceau de sucre. Le cerveau à lui seul en
brûle autant chaque heure, jour et nuit, et ne peut rien utiliser
d'autre. Pourtant, une personne qui ne mange rien de la journée, ou
qui avale un gâteau entier d'un coup, maintient la concentration à
quelques dixièmes de gramme par litre de cette même valeur --- et une
personne dont le pancréas a lâché n'y parvient pas, avec des
conséquences qui vont de la soif au coma. Ce chapitre traite du nombre
le plus surveillé du corps, des organes qui le maintiennent constant,
et de ce qui arrive, dans les deux formes de \omterm{def:g12:glucose-and-diabetes:diabetes}{diabète}, quand ils n'y
parviennent plus.

\section{Une constante régulée}

\begin{definition}[Glycémie]\label{def:g12:glucose-and-diabetes:glycaemia}
La \emph{glycémie}\index{glycémie} est la concentration de glucose
dans le plasma sanguin~: environ \qty{1}{g/L} (\qty{5.5}{mmol/L})
avant un repas chez une personne en bonne santé, montant à
\qty{1.4}{g/L} dans l'heure qui suit un repas et revenue en deux
heures, tombant à \qty{0.7}{g/L} après une journée de jeûne.
Au-dessous de \qty{0.5}{g/L}, le cerveau défaille (confusion, puis
coma)~; au-dessus de \qty{1.8}{g/L}, le glucose passe dans l'urine, et
des années au-dessus de \qty{1.3}{g/L} abîment les vaisseaux et les
nerfs. La glycémie est une grandeur \emph{régulée}~: une valeur
maintenue près d'une valeur de consigne par un système qui la mesure
et la corrige.
\end{definition}

\begin{proposition}[Le compartiment du glucose et ses flux]\label{prop:g12:glucose-and-diabetes:pool}
Les \qty{5}{g} de glucose du sang forment un compartiment que
traversent de grands flux. \emph{Entrées}~: l'intestin après un repas
(jusqu'à \qty{100}{g} en une heure), et entre les repas le
\emph{foie}, qui libère du glucose à partir de son \emph{\omterm{def:g10:exercise-and-energy:glycogen}{glycogène}} de
réserve (\qty{100}{g}, une demi-journée d'approvisionnement) et
fabrique du glucose neuf à partir d'\omterm{def:g10:chemistry-of-life:families}{acides aminés} et de lactate.
\emph{Sorties}~: le cerveau (\qty{5}{g/h}, sans arrêt), les muscles
(de quelques grammes à \qty{100}{g/h} à l'effort), les autres tissus,
et la mise en réserve sous forme de \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} dans le foie et les
muscles ou sous forme de graisse. Un compartiment de \qty{5}{g}
alimenté et vidé à raison de dizaines de grammes par heure n'est
stable que parce que les flux sont ajustés minute par minute.
\end{proposition}
\admitted

\begin{example}[Une journée de glycémie]\label{ex:g12:glucose-and-diabetes:day}
Petit-déjeuner à 8~heures~: la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} monte de \num{0.9} à
\qty{1.3}{g/L} pour 9~heures et est revenue à \num{1.0} à 10~heures,
l'excédent étant parti dans le \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} du foie et des muscles. De 10
à 13~heures, le foie libère du \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} et le niveau se tient près de
\num{0.9}. Une course à 17~heures tire \qty{60}{g} vers les muscles en
une heure~; le foie vide la moitié de sa réserve pour suivre, et le
niveau descend à \num{0.8}. Pendant la nuit, le foie fabrique du
glucose neuf à partir d'\omterm{def:g10:chemistry-of-life:families}{acides aminés}, et à 7~heures le niveau est de
\qty{0.9}{g/L}. Trois cents grammes ont traversé un compartiment de
cinq.
\end{example}

\section{Le pancréas tient la balance}

\begin{proposition}[Deux hormones venues des îlots]\label{prop:g12:glucose-and-diabetes:hormones}
Dispersés dans le pancréas, parmi les \omterm{def:g10:cells-common-unit:cell}{cellules} qui fabriquent les
\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymes} digestives, se trouvent un million de petits amas cellulaires,
les \emph{îlots}\index{îlots du pancréas}. Leurs \omterm{def:g10:cells-common-unit:cell}{cellules} mesurent la
\omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} du sang qui passe et y sécrètent deux \omterm{def:g11:hormones-and-reproduction:hormone}{hormones}~:
\begin{itemize}
  \item l'\emph{insuline}\index{insuline}, produite par les \omterm{def:g10:cells-common-unit:cell}{cellules}
        $\beta$, quand la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} monte~: elle fait entrer le glucose
        dans les \omterm{def:g10:cells-common-unit:cell}{cellules} musculaires et adipeuses, et le fait mettre
        en réserve sous forme de \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} par le foie et les
        muscles~; la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} baisse~;
  \item le \emph{glucagon}\index{glucagon}, produit par les \omterm{def:g10:cells-common-unit:cell}{cellules}
        $\alpha$, quand elle baisse~: il fait dégrader le \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} du
        foie et libérer du glucose~; la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} monte.
\end{itemize}
Les deux agissent sur les mêmes organes en sens contraires~; leur
équilibre, réajusté toutes les quelques minutes d'après la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie}
mesurée, constitue la régulation. L'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} est la seule \omterm{def:g11:hormones-and-reproduction:hormone}{hormone} qui
abaisse la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie}~; plusieurs l'élèvent.
\end{proposition}
\begin{proof}[Preuves]
Retirer le pancréas d'un chien (1889) le rend diabétique en un jour~:
sa \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} triple et du glucose apparaît dans son urine~; ligaturer
le canal qui conduit les \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymes} digestives ne le fait pas~: l'effet
n'est donc pas digestif. Injecter un extrait des \omterm{prop:g12:glucose-and-diabetes:hormones}{îlots} (1921) abaisse
en quelques heures la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} d'un chien diabétique, puis celle d'un
enfant diabétique~; la substance active, l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}, fut purifiée la
même année. Des \omterm{prop:g12:glucose-and-diabetes:hormones}{îlots} isolés dans une boîte sécrètent de l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}
quand on élève le glucose du milieu, et du \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} quand on
l'abaisse. Du \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} injecté élève en quelques minutes la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie}
d'une personne à jeun, et seulement si le foie détient du \omterm{def:g10:exercise-and-energy:glycogen}{glycogène}.
\end{proof}

\begin{example}[La boucle en action]\label{ex:g12:glucose-and-diabetes:loop}
Après un repas, la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} grimpe~; en quelques minutes, les \omterm{def:g10:cells-common-unit:cell}{cellules}
$\beta$ libèrent de l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}, les muscles et les \omterm{def:g10:cells-common-unit:cell}{cellules} adipeuses
ouvrent leurs transporteurs de glucose, le foie se met à fabriquer du
\omterm{def:g10:exercise-and-energy:glycogen}{glycogène}, et le niveau redescend --- moment où la sécrétion
d'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} baisse à son tour. Entre les repas, la dérive vers le bas
est contrée par le \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon}, le \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} est dégradé, et le niveau se
maintient. Aucune des deux \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} n'est jamais absente~; c'est leur
rapport qui bouge, et la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} le suit dans une bande étroite.
\end{example}

\section{Quand la boucle défaille~: le diabète}

\begin{definition}[Diabète]\label{def:g12:glucose-and-diabetes:diabetes}
Le \emph{diabète}\index{diabète} est un excès chronique de \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie}~:
au-dessus de \qty{1.26}{g/L} à jeun, ou au-dessus de \qty{2}{g/L} deux
heures après une charge de glucose standard. Ses premiers signes
découlent de cet excès~: du glucose dans l'urine, qui entraîne de
l'eau avec lui (urines abondantes, soif), et un amaigrissement quand
les \omterm{def:g10:cells-common-unit:cell}{cellules} ne peuvent pas utiliser le glucose qui les entoure. Ses
dommages tardifs --- aux petits vaisseaux de l'\omterm{def:g11:the-eye:parts}{œil} et du rein, aux
nerfs, aux artères --- viennent d'années de glucose réagissant avec
les \omterm{def:g10:chemistry-of-life:families}{protéines}. Deux maladies portent ce nom.
\end{definition}

\begin{proposition}[Le type 1 et le type 2]\label{prop:g12:glucose-and-diabetes:types}
\begin{itemize}
  \item Le \emph{\omterm{def:g12:glucose-and-diabetes:diabetes}{diabète} de type 1} (environ 10\% des cas) est la
        destruction des \omterm{def:g10:cells-common-unit:cell}{cellules} $\beta$ par le système immunitaire du
        patient lui-même, en général dans l'enfance ou la jeunesse~:
        l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} est absente. La \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} monte sans frein, la
        graisse est brûlée faute de glucose utilisable et ses produits
        acides s'accumulent~; sans \omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} injectée, la maladie est
        mortelle en quelques mois. Le traitement est l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline},
        plusieurs fois par jour, dosée d'après les repas et la
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} mesurée, à vie.
  \item Le \emph{\omterm{def:g12:glucose-and-diabetes:diabetes}{diabète} de type 2} (environ 90\%) apparaît chez
        l'adulte, le plus souvent en surpoids et sédentaire~: les
        tissus répondent de moins en moins à l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}
        (\emph{insulinorésistance}), les \omterm{def:g10:cells-common-unit:cell}{cellules} $\beta$ compensent
        en sécrétant davantage, et après des années elles s'épuisent.
        L'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} est présente mais insuffisante. La prédisposition
        est en partie génétique (\cref{ch:g11:genes-and-disease})~; le
        déclencheur est le mode de vie, et le premier traitement en
        est le changement --- perte de poids et exercice --- avant les
        médicaments et, tardivement, l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}.
\end{itemize}
\end{proposition}
\begin{proof}[Preuves]
Les patients de type 1 ont, des années avant les symptômes, des
anticorps dirigés contre leurs propres \omterm{def:g10:cells-common-unit:cell}{cellules} d'\omterm{prop:g12:glucose-and-diabetes:hormones}{îlots}, et au
diagnostic leurs \omterm{prop:g12:glucose-and-diabetes:hormones}{îlots} sont presque vides de \omterm{def:g10:cells-common-unit:cell}{cellules} $\beta$ et
infiltrés de globules blancs~; leur \omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} sanguine est indétectable
et ils répondent aussitôt à l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} injectée. Les patients de
type 2 ont une \omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} normale ou élevée au diagnostic~; leurs
\omterm{def:g10:cells-common-unit:cell}{cellules} musculaires et adipeuses captent moins de glucose, pour une
même dose d'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}, que celles d'une personne en bonne santé~; les
jumeaux vrais partagent la maladie dans 70\% des cas, et une perte de
poids ou un simple programme de marche rétablit une \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} normale
dans bien des cas précoces.
\end{proof}

\begin{example}[Gérer une journée avec un diabète de type 1]\label{ex:g12:glucose-and-diabetes:type1}
Une adolescente atteinte d'un \omterm{def:g12:glucose-and-diabetes:diabetes}{diabète} de type 1 mesure sa \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie}
avant chaque repas avec une goutte de sang, compte les \omterm{def:g10:chemistry-of-life:families}{glucides} de son
assiette et s'injecte une dose d'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} rapide qui leur est
proportionnelle --- environ une unité pour \qty{10}{g} --- plus une
\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} lente une fois par jour pour le fond. Trop d'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}, ou un
repas sauté, et le niveau tombe au-dessous de \qty{0.6}{g/L}~: sueurs,
tremblements, confusion, corrigés par du sucre en quelques minutes.
Trop peu, et le niveau reste au-dessus de \qty{2}{g/L}, avec soif et
fatigue, et les vaisseaux encaissent les dégâts en silence. Elle fait
à la main ce que ses \omterm{def:g10:cells-common-unit:cell}{cellules} $\beta$ faisaient par \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction}.
\end{example}

\begin{method}[Lire un relevé de glycémie]\label{met:g12:glucose-and-diabetes:read}
\begin{enumerate}
  \item Repérer la consigne et la bande~: à jeun
        \qtyrange{0.7}{1.1}{g/L}, après les repas au-dessous de
        \num{1.4}~; deux heures après une charge de glucose,
        au-dessous de \num{1.4} (sain), \qtyrange{1.4}{2}{}
        (tolérance altérée), au-dessus de \qty{2}{g/L} (\omterm{def:g12:glucose-and-diabetes:diabetes}{diabète}).
  \item Pour chaque hausse, se demander ce qui a mis du glucose dans
        le sang (repas, foie) et ce qui devrait l'en retirer (les
        cibles de l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline})~; pour chaque baisse, ce qui l'a retiré
        (cerveau, muscle) et ce qui devrait l'y remettre (\omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon},
        foie).
  \item Lire les \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} avec le glucose~: l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} doit monter
        avec la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} et le \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} baisser~; une \omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} absente
        signe un type 1, une \omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} élevée avec une \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} élevée
        signe une résistance.
  \item Convertir les unités au besoin~: \qty{1}{g/L} de glucose fait
        \qty{5.55}{mmol/L}.
\end{enumerate}
\end{method}

\begin{remark}[Un modèle de régulation]\label{rem:g12:glucose-and-diabetes:model}
Un capteur, une consigne, deux effecteurs opposés, une \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction}
négative~: le contrôle de la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} suit le même plan que le
contrôle de la \omterm{prop:g11:becoming-male-female:hormones}{testostérone} du
\cref{ch:g11:hormones-and-reproduction}, et que la plupart des
constantes du corps --- température, eau, pression artérielle,
acidité. Le \omterm{def:g12:glucose-and-diabetes:diabetes}{diabète} est ce à quoi ressemble une constante régulée
quand sa boucle est rompue~: la valeur répond encore aux entrées, mais
plus rien ne la ramène. Lire les courbes de ce chapitre, c'est
s'exercer à lire n'importe laquelle d'entre elles.
\end{remark}

\section{Problème~: cinq grammes de sucre}

\begin{problem}[{Problème du week-end --- le glucose du sang mis en
comptes sur une journée~: le compartiment et ses flux, un test de
tolérance lu, une dose d'insuline calculée, et une boucle qu'il faut
tenir à la main}]
\label{pb:g12:glucose-and-diabetes:1}
On prendra un volume sanguin de \qty{5}{L}, une consommation cérébrale
de \qty{5}{g/h} de glucose, une réserve de \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} hépatique de
\qty{100}{g}, et un \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} musculaire de \qty{400}{g}. Une unité
d'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} rapide prend en charge environ \qty{10}{g} de \omterm{def:g10:chemistry-of-life:families}{glucides}.

\medskip
\noindent\textbf{Partie I --- Le compartiment.}
\begin{enumerate}
  \item Calculer la masse de glucose du sang à \qty{1}{g/L}, et la
        même concentration en \unit{mmol/L} (glucose
        \qty{180}{g/mol}).
  \item Combien de temps le glucose du sang durerait-il au seul
        cerveau, si rien ne le remplaçait~?
  \item Entre les repas, le foie le remplace. Combien de temps le
        \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} du foie dure-t-il au cerveau~? Que se passe-t-il
        ensuite~?
  \item Le \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} musculaire vaut quatre fois celui du foie, mais
        les muscles ne peuvent pas libérer de glucose dans le sang.
        Expliquer pourquoi le \omterm{def:g10:exercise-and-energy:glycogen}{glycogène} musculaire n'aide pas le
        cerveau, et à quoi il sert.
  \item Un repas délivre \qty{60}{g} de glucose en une heure. Si rien
        n'en était retiré, quelle valeur atteindrait la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie}~?
        Qu'est-ce qui le retire en réalité, et où va-t-il~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le test.}
Un patient boit \qty{75}{g} de glucose. \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glycémie} et \omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}
mesurées~:

\begin{center}
\begin{tabular}{lcccccc}
minutes & 0 & 30 & 60 & 90 & 120 & 180 \\ \hline
\omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} (g/L) & 1.15 & 1.8 & 2.05 & 2.0 & 1.85 & 1.6 \\
\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} (\% d'un pic normal) & 90 & 140 & 180 & 200 & 190 & 160 \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item Classer le patient avec les seuils de la
        \cref{met:g12:glucose-and-diabetes:read}.
  \item L'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} manque-t-elle~? Qu'est-ce qui défaille, alors~?
  \item Nommer la maladie, et donner deux éléments de l'histoire du
        patient auxquels vous vous attendez.
  \item À quoi ressemblerait le même tableau chez un patient de
        type 1 non traité~?
  \item Quel est le premier traitement prescrit, et par quel
        mécanisme agit-il~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Tenir la boucle à la main.}
La \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} d'une patiente de type 1 est de \qty{1.0}{g/L} avant le
dîner~; le repas contient \qty{90}{g} de \omterm{def:g10:chemistry-of-life:families}{glucides}.
\begin{enumerate}[resume]
  \item Calculer la dose d'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} rapide pour ce repas.
  \item Elle l'injecte mais ne mange que la moitié du repas. Estimer
        jusqu'où la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} peut descendre, et nommer les symptômes
        et le remède.
  \item Elle n'injecte rien et mange tout le repas. Estimer la
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} si les \qty{90}{g} entraient tous dans un
        compartiment de \qty{5}{L}, et dire ce que font les reins
        au-dessus de \qty{1.8}{g/L}.
  \item Pourquoi doit-elle aussi injecter une \omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} lente le soir,
        alors qu'elle ne mange rien~?
  \item Expliquer en quel sens le lecteur et le stylo remplacent la
        \omterm{def:g10:cells-common-unit:cell}{cellule} $\beta$, et ce qu'ils ne peuvent pas remplacer.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La boucle, raisonnée.}
\begin{enumerate}[resume]
  \item Un médicament stimule les \omterm{def:g10:cells-common-unit:cell}{cellules} $\beta$ pour qu'elles
        sécrètent plus d'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}. Pour quel type de \omterm{def:g12:glucose-and-diabetes:diabetes}{diabète} peut-il
        marcher, et pourquoi pas pour l'autre~?
  \item Une personne de type 2 perd \qty{10}{kg} et marche une heure
        par jour~; six mois plus tard, le test de tolérance est
        normal. Expliquer quelle partie de la boucle a été réparée.
  \item Expliquer pourquoi l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline} et le \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} d'une personne en
        bonne santé ne sont jamais élevés tous les deux, et ce que
        fait un îlot dans une boîte quand on fixe le glucose du milieu
        à \qty{1}{g/L}.
  \item Le \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon}, l'\omterm{prop:g10:heart-lungs-effort:control}{adrénaline}, le cortisol et l'\omterm{def:g11:hormones-and-reproduction:hormone}{hormone} de
        croissance élèvent tous la \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie}~; seule l'\omterm{prop:g12:glucose-and-diabetes:hormones}{insuline}
        l'abaisse. Proposer une raison pour laquelle le corps dispose
        de plusieurs garde-fous contre une \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} basse et d'un seul
        contre une \omterm{def:g12:glucose-and-diabetes:glycaemia}{glycémie} haute.
  \item Énoncer le résultat~: le glucose contenu dans le sang, les
        heures que le foie peut couvrir, le diagnostic de la partie
        II, et la dose de la partie III.
\end{enumerate}
\end{problem}
