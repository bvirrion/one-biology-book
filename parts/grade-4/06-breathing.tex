\chapter{Breathing}\label{ch:g4:breathing}

Hold your breath and start counting. Long before one minute,
your chest demands, more and more urgently, to move again. You
can skip a meal, \omterm{prop:g1:healthy-habits:needs}{sleep} can wait until tonight --- but air is
needed \emph{now}, around the clock, about fifteen times a
minute. What is so urgent in a breath of air?

\section{The breathing movements}

\begin{definition}[Breathing in and out]\label{def:g4:breathing:movements}
\emph{Breathing}\index{breathing} is the endless alternation of
two movements:
\begin{itemize}
  \item \emph{breathing in} (inhaling)\index{inhaling}: the
        chest expands and air flows in through nose or mouth;
  \item \emph{breathing out} (exhaling)\index{exhaling}: the
        chest relaxes and air flows back out.
\end{itemize}
At rest, the pair repeats about $15$ times a minute --- faster
in babies, faster in effort --- without your ever needing to
think about it.
\end{definition}

\begin{example}[Watching the movements]\label{ex:g4:breathing:watch}
Hands flat on your \omterm{def:g3:bones-and-muscles:skeleton}{ribs}: \omterm{def:g4:breathing:movements}{breathing} in, the \omterm{def:g3:bones-and-muscles:skeleton}{rib} cage rises and
widens under your palms; \omterm{def:g4:breathing:movements}{breathing} out, it sinks back. A
sleeping cat's flank shows the same rise and fall. The
movements never stop --- awake, asleep, from first cry to last
day.
\end{example}

\section{Where the air goes}

\begin{definition}[The breathing organs]\label{def:g4:breathing:organs}
Inhaled air travels through the nose (or mouth), down the
\emph{windpipe}\index{windpipe} --- the firm ringed tube you
can feel at the front of your neck --- which divides into two
\omterm{def:g1:plants-around-us:tree}{branches}, one to each \emph{lung}\index{lung}. The lungs, two
spongy organs filling the \omterm{def:g3:bones-and-muscles:skeleton}{rib} cage
(behind the \omterm{def:g3:bones-and-muscles:skeleton}{ribs} of \cref{def:g3:bones-and-muscles:skeleton}),
end in millions of tiny air pockets, each wrapped in fine
\omterm{prop:g4:journey-of-food:absorb}{blood} vessels.
\end{definition}

\begin{remark}[Spongy on purpose]\label{rem:g4:breathing:sponge}
Millions of pockets instead of two simple bags: it is the
\omterm{def:g4:journey-of-food:tract}{small intestine}'s trick again
(\cref{rem:g4:journey-of-food:surface}). All those pocket
walls add up to an enormous meeting surface between air and
\omterm{prop:g4:journey-of-food:absorb}{blood} --- the size of half a tennis court, folded into your
chest.
\end{remark}

\section{What breathing is for}

\begin{proposition}[Taking oxygen, returning carbon dioxide]\label{prop:g4:breathing:exchange}
Air is a mixture of gases, and the body wants one of them:
\emph{oxygen}\index{oxygen}. In the \omterm{def:g4:breathing:organs}{lungs}' pockets, oxygen
from fresh air passes into the \omterm{prop:g4:journey-of-food:absorb}{blood}, which delivers it to
every part of the body; there it is used, together with
\omterm{def:g4:journey-of-food:digestion}{nutrients}, to release the \omterm{prop:g3:balanced-diet:jobs}{energy} of
\cref{prop:g3:balanced-diet:jobs}. The body's waste gas,
\emph{carbon dioxide}\index{carbon dioxide}, makes the return
trip and is breathed out. Exhaled air has less oxygen and
more carbon dioxide than inhaled air --- the difference is
what \omterm{def:g4:breathing:movements}{breathing} is \emph{for}.
\end{proposition}
\admitted

\begin{example}[Why running takes your breath]\label{ex:g4:breathing:running}
Sprinting, your \omterm{def:g1:my-body:parts}{legs} burn \omterm{prop:g3:balanced-diet:jobs}{energy} fast --- so they need oxygen
fast. \omterm{def:g4:breathing:movements}{Breathing} obliges: from about $15$ calm breaths a
minute to $40$ or more, deep ones, while the heart
(\cref{ch:g4:heart-and-blood} tells its side) races to speed
the deliveries. The panting after the race is the body
settling its oxygen accounts.
\end{example}

\begin{example}[The misted mirror]\label{ex:g4:breathing:mirror}
Breathe on a cold mirror: mist. Exhaled air comes back
changed --- warmed, moistened, and, invisibly, poorer in
oxygen and richer in carbon dioxide. The mirror shows the
moisture; the gas swap it cannot show is the important part.
\end{example}

\begin{method}[Caring for the breath]\label{met:g4:breathing:care}
\begin{enumerate}
  \item Air rooms daily --- many breathers in a closed room
        means stale, carbon-dioxide-rich air;
  \item breathe through the nose in cold or dusty air: it
        warms, moistens and filters;
  \item run, swim, play --- exercised \omterm{def:g4:breathing:organs}{lungs} and \omterm{def:g4:breathing:movements}{breathing}
        \omterm{def:g3:bones-and-muscles:muscle}{muscles} grow stronger and deeper;
  \item keep smoke away from your \omterm{def:g4:breathing:organs}{lungs}: it dirties the
        delicate pockets, and they clean themselves only
        slowly.
\end{enumerate}
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g4:breathing:1}
Name the two \omterm{def:g4:breathing:movements}{breathing} movements and what the chest does in
each.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:2}
Trace a breath's path from the nose to the tiny air pockets.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:3}
About how many times a minute does a calm resting child
breathe? What happens to that number during a sprint, and
why?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:4}
Which gas does the body take from the air, and which does it
return? Where exactly does the swap happen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:5}
How does exhaled air differ from inhaled air? Name three
differences.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:6}
Why are the \omterm{def:g4:breathing:organs}{lungs} built as millions of tiny pockets rather
than two simple bags? Which other organ of this year uses
the same trick?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:7}
Why is it better to breathe through the nose on a freezing
or dusty day?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:8}
Try it: sitting calmly, count your breaths for one minute.
Then do twenty jumping jacks and count again. Report both
numbers and explain the change.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:9}
Meals can wait hours; \omterm{def:g4:breathing:movements}{breathing} cannot wait a minute. What
does this say about how much oxygen the body keeps in store,
compared with its stores of food?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:10}
A classroom feels ``stuffy'' by late morning. What has
changed in its air, by \cref{prop:g4:breathing:exchange},
and what does \cref{met:g4:breathing:care} prescribe?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:breathing:11}
The trout has no \omterm{def:g4:breathing:organs}{lungs}: water flows over its feathery gills,
and the oxygen \emph{dissolved in the water} passes into its
\omterm{prop:g4:journey-of-food:absorb}{blood} there. What job do gills and \omterm{def:g4:breathing:organs}{lungs} share? And why does
a trout suffocate in air, where oxygen is plentiful?
(Hint: out of water, the feathery gill surfaces collapse and
stick together.)
\end{exercise}
