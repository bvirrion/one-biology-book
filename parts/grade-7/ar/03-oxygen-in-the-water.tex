\chapter{الأكسجين في الماء}\label{ch:g7:oxygen-in-the-water}

جدول الجبل وبركة الصيف كلاهما ``ماء'' --- ومع ذلك فالسلمون المرقط
الذي ينطلق في أحدهما يلهث في الآخر، والشبّوط الذي يزدهر في البركة
يجد الجدول خاليًا من غذائه. والفرق بين الماءين عدد واحد غير مرئي في
معظمه: مقدار \omterm{def:g7:oxygen-in-the-water:dissolved}{الأكسجين المذاب} فيهما. وهذا الفصل يتعلم قراءة ذلك
العدد وقياسه والتنبؤ به --- ويتعلم معه خريطة حياة الماء.

\section{الأكسجين المذاب، مخزون شحيح}

\begin{definition}[الأكسجين المذاب]\label{def:g7:oxygen-in-the-water:dissolved}
يحمل الماء \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} كما يحمل الشاي المحلّى السكر --- من غير أن
يُرى، \emph{مذابًا}\index{أكسجين مذاب}. والكمية ضئيلة: فلتر من ماء
حسن التهوية يحمل نحو \qty{10}{mg} من \omterm{def:g7:what-is-respiration:respiration}{الأكسجين}، أي أقل بنحو ثلاثين
مرة من لتر هواء. وكل \omterm{def:g7:how-animals-breathe:four}{غلصمة} وكل متنفس تحت الماء في
\cref{ch:g7:how-animals-breathe} يسحب من هذا المخزون الرقيق الواحد
(\cref{prop:g7:what-is-respiration:habitats}).
\end{definition}

\begin{proposition}[ما يملأ المخزون]\label{prop:g7:oxygen-in-the-water:fills}
مورّدان يعيدان ملء \omterm{def:g7:what-is-respiration:respiration}{أكسجين} الماء:
\begin{enumerate}
  \item \emph{الهواء فوقه}: \omterm{def:g7:what-is-respiration:respiration}{فالأكسجين} يذوب فيه عند السطح ---
        وأسرع حيث يُحرَّك الماء ويُرشّ ويُرغّى (المسايل،
        والشلالات، والأمواج، والمطر)؛
  \item و\emph{\omterm{def:g5:food-webs:roles}{المنتِجون} تحته}: \omterm{def:g1:plants-around-us:plant}{فنباتات} الماء \omterm{def:g5:first-look-at-cells:cell}{والخلايا} المفردة
        الخضراء تطلق \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} في الماء نهارًا وهي تصنع
        (\cref{rem:g6:plants-are-producers:gas}).
\end{enumerate}
\end{proposition}
\admitted

\begin{proposition}[ما يستنزفه، وما يستطيع أن يحمله]\label{prop:g7:oxygen-in-the-water:drains}
يستنزف المخزون \emph{كل متنفس} --- \omterm{def:g1:animals-around-us:animal}{الحيوانات}، \omterm{def:g1:plants-around-us:plant}{والنباتات} ليلًا،
\omterm{def:g6:decomposers-and-soil:decomposers}{والمحلِّلون} قبل كل شيء حين تكثر المادة الميتة التي يأكلونها
(وهو المرطبان الثامن في \cref{pb:g7:what-is-respiration:1}). والسقف
يتحرك: فالماء \emph{البارد} يستطيع أن يحمل من \omterm{def:g7:oxygen-in-the-water:dissolved}{الأكسجين المذاب} أكثر
بوضوح من الدافئ --- فجدول عند \qty{5}{\degreeCelsius} يحمل مرة
ونصفًا ما تحمله بركة عند \qty{25}{\degreeCelsius}. والدفء ضغط
مزدوج: \omterm{def:g7:what-is-respiration:respiration}{أكسجين} أقل محمول، بينما يجري كل صوت محرّك
(\cref{ex:g7:what-is-respiration:demand}) أسرع.
\end{proposition}
\admitted

\begin{example}[الجدول في مقابل البركة]\label{ex:g7:oxygen-in-the-water:streampond}
الآن يُقرأ الماءان دفترَي حساب. جدول الجبل: بارد (سقف عالٍ)، مخضوض
أبيض عند كل صخرة (إعادة ملء سريعة) --- فأكسجينه قرب السقف طول
السنة. وبركة الصيف: دافئة (سقف منخفض)، ساكنة (إعادة ملء بطيئة)،
غنية بالطين والمادة الميتة (محلِّلون جياع) --- فأكسجينها يزحف على
الأرض، وأسوأ ما يكون في الساعات الصغيرة قبل الفجر، حين تكون
\omterm{def:g1:plants-around-us:plant}{النباتات} قد تنفست الليل كله بلا صناعة نهارية تقابلها.
\end{example}

\begin{example}[قراءة منحنى البركة]\label{ex:g7:oxygen-in-the-water:curve}
موجة البركة اليومية هي
\cref{rem:g6:plants-are-producers:gas} مرسومةً معطياتٍ: فالشروق
يبدأ صناعة \omterm{def:g5:food-webs:roles}{المنتِجين} فيصعد المنحنى؛ ويبلغ ذروته في الأصيل؛ ويترك
الليل كل متنفس يسحب من المخزون بلا مورّد إلا السطح الساكن ---
فينزلق المنحنى إلى دُنيا الفجر. وموت \omterm{prop:g3:sorting-living-things:groups}{الأسماك} في البرك المتشابكة
يقع عند الفجر، وقد صرت تعرف السبب.
\end{example}

\section{خريطة السكان}

\begin{proposition}[الأكسجين يرسم الخريطة]\label{prop:g7:oxygen-in-the-water:map}
لما كانت \omterm{def:g5:biodiversity:species}{الأنواع} تختلف في حاجاتها من \omterm{def:g7:what-is-respiration:respiration}{الأكسجين}، رسم العدد المذاب
خريطة السكان
(\cref{prop:g6:exploring-our-environment:decide}، تحت الماء):
\begin{itemize}
  \item \omterm{def:g5:biodiversity:species}{فالأنواع} \emph{الطلوب} --- السلمون المرقط، \omterm{ex:g2:animals-grow-up:butterfly}{ويرقات} ذباب
        مايو بخصلات غلاصمها المتماوجة
        (\cref{pb:g7:how-animals-breathe:1}) --- لا تعيش إلا في
        ماء بارد مخضوض غني \omterm{def:g7:what-is-respiration:respiration}{بالأكسجين}؛
  \item \omterm{def:g5:biodiversity:species}{والأنواع} \emph{المتحمّلة} --- الشبّوط، وكثير من حلزونات
        البركة، \omterm{ex:g2:animals-grow-up:butterfly}{واليرقات} ذوات الذنب الفأري وأنابيبها إلى السطح
        --- تدبّر أمرها في ماء دافئ ساكن أفقر؛
  \item فيُقرأ الإحصاء إذن بالعكس: جد \omterm{ex:g2:animals-grow-up:butterfly}{يرقات} ذباب مايو تحت الحجارة
        فالماء جيد؛ وجد المتنفسين بالأنابيب وبالِعي الهواء وحدهم
        فهو رديء.
\end{itemize}
ويحصي حرّاس الأنهار الحياة الصغيرة لهذا السبب بالضبط: فالسكان سجلّ
لأسوأ أيام الماء الأخيرة، وهو ما قد تفوّته قراءة مِسبار واحدة.
\end{proposition}
\admitted

\begin{example}[حكاية النفايات العضوية]\label{ex:g7:oxygen-in-the-water:waste}
نفايات قرية غير معالَجة --- وهي \omterm{def:g6:plants-are-producers:matter}{مادة عضوية} غنية --- تبلغ جدولًا
صافيًا. فيولم \omterm{def:g6:decomposers-and-soil:decomposers}{المحلِّلون} أسفل الأنبوب ويتكاثرون
(\cref{def:g6:decomposers-and-soil:decomposers})، ويستنزف تنفسهم
المحتشد \omterm{def:g7:what-is-respiration:respiration}{الأكسجين}: فيختفي السلمون المرقط \omterm{ex:g2:animals-grow-up:butterfly}{ويرقات} ذباب مايو، ويستولي
المتحمّلون أصحاب الأنابيب، وأبعد إلى أسفل --- وقد استُهلكت النفايات
وأُعيدت تهوية الماء على الجداول الحصوية --- تعود \omterm{def:g5:biodiversity:species}{الأنواع} الطلوب.
فالتلوث كان \omterm{def:g6:plants-are-producers:matter}{مادة عضوية}؛ والقاتل كان جوع \omterm{def:g7:what-is-respiration:respiration}{الأكسجين}؛ والخريطة سجّلت
ذلك كله.
\end{example}

\begin{method}[الحكم على ماء]\label{met:g7:oxygen-in-the-water:judge}
لتحكم على جدول أو بركة:
\begin{enumerate}
  \item قِس المورّدين أو قدّرهما: الحرارة، والخضّ، والنمو الأخضر؛
  \item وزِن المصارف: الطين والمادة الميتة، والازدحام، وأي وارد
        من نفايات عضوية؛
  \item وأحصِ الحياة الصغيرة تحت الحجارة وبين الأعشاب --- أحاضرة
        \omterm{def:g5:biodiversity:species}{الأنواع} الطلوب أم المتحمّلة وحدها؟
  \item وإن أخذت العينة في ساعة واحدة، فتذكّر منحنى البركة:
        فقراءة أصيل جميلة قد تخفي فجرًا قاتلًا.
\end{enumerate}
\end{method}

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:1}
كم من \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} يحمل لتر من ماء حسن التهوية تقريبًا، وكيف يقارَن

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:2}
سمِّ مورّدَي \omterm{def:g7:what-is-respiration:respiration}{أكسجين} الماء، والشرط الذي يورّد فيه كل واحد أسرع ما

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:3}
سمِّ ثلاثة مصارف للمخزون. وأيها يعظم حين تكثر المادة الميتة؟

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:5}
لماذا يبلغ منحنى \omterm{def:g7:what-is-respiration:respiration}{أكسجين} البركة ذروته في الأصيل ودُنياه عند الفجر؟

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:6}
أي كشفين في الإحصاء يعلّمان ماءً غنيًّا \omterm{def:g7:what-is-respiration:respiration}{بالأكسجين}، وأيهما يعلّم

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:7}

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:8}

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:10}

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:11}

\begin{exercise}[$\star\star\star$]\label{exo:g7:oxygen-in-the-water:12}

\section{مسألة: كتاب الحارس السنوي}

\begin{problem}[{مسألة نهاية الأسبوع --- نهر واحد، وأربع محطات،
وسنة من القراءات}]\label{pb:g7:oxygen-in-the-water:1}
يحتفظ حارس نهر بكتاب سنوي لأربع محطات: م1، مسيل بارد أسفل سدّ
صغير؛ وم2، مقطع بطيء دافئ يمرّ في مراعٍ؛ وم3، أسفل أنبوب نفايات
معمل ألبان مباشرة؛ وم4، على بعد كيلومترين إلى أسفل، بعد جدول حصوي
طويل. ولكل محطة قراءات مِسبار وإحصاء للحياة الصغيرة.

\medskip\noindent\textbf{القسم الأول --- قراءة المحطات.}
\begin{enumerate}
  \item تقرأ م1 \qty{11}{mg/L} عند
        \qty{8}{\degreeCelsius}؛ وتقرأ م2 \qty{7}{mg/L}
        عند \qty{22}{\degreeCelsius}. انسب الفرق إلى المورّدين
        والسقوف الصحيحة.
  \item إحصاء م1: صغار سلمون مرقط، \omterm{ex:g2:animals-grow-up:butterfly}{ويرقات} ذباب مايو. وإحصاء م2:
        شبّوط، وحلزونات بركة، \omterm{ex:g2:animals-grow-up:butterfly}{ويرقات} ذوات ذنب فأري. فسّر
        الاثنين وفق \cref{prop:g7:oxygen-in-the-water:map}.
  \item تقرأ م3 \qty{3}{mg/L} مع أن الماء ليس أدفأ من م2. سمِّ
        المصرف، والعمّال وراءه.
  \item وتقرأ م4 \qty{9}{mg/L} ويُظهر إحصاؤها \omterm{ex:g2:animals-grow-up:butterfly}{يرقات} ذباب مايو من
        جديد. اشرح التعافي بالمورّدين معًا وبمصير النفايات.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- إيقاعات السنة.}
\begin{enumerate}[resume]
  \item تهبط قراءات م2 عند الفجر في آب إلى
        \qty{4}{mg/L} بينما تبلغ قراءات الأصيل \qty{9}{mg/L}.
        ارسم منحناها اليومي أو صفه، وسمِّ العمليتين المتبادلتين
        للمكان.
  \item قراءات كانون الثاني عند م2 أثبت وأعلى من قراءات آب، مع
        أن مقطع المراعي ساكن كما هو. أعطِ سببي الشتاء.
  \item يُتوقَّع أسبوعان حارّان بلا ريح. رتّب المحطات الأربع بحسب
        خطر موت \omterm{prop:g3:sorting-living-things:groups}{الأسماك}، بسبب واحد لكل واحدة.
  \item توقّت الحارسة زيارات مِسبارها في آب قبيل الشروق مباشرة.
        دافع عن هذا الجدول.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- الأحكام
والعلاجات.}
\begin{enumerate}[resume]
  \item يركّب معمل الألبان معالجةً تزيل أكثر \omterm{def:g6:plants-are-producers:matter}{المادة العضوية} من
        تصريفه. تنبأ بمِسبار م3 وإحصائها بعد سنة، بترتيب
        التعافي.
  \item يقترح مالك أرض قطع أشجار ضفة م2 ``ليدخل ضوء أكثر فيصنع
        أكسجينًا أكثر''. زِن الاقتراح: ماذا يفعل الظل أيضًا وفق
        \cref{prop:g7:oxygen-in-the-water:drains}، وبم تجيب؟
  \item يجد الإحصاء \omterm{ex:g2:animals-grow-up:butterfly}{يرقات} ذباب مايو في محطة تبدو قراءة مِسبارها
        الوحيدة في الأصيل متوسطة. فبأي شاهد تصدّق في شأن الموسم
        كله، ولماذا؟
  \item أغلق الكتاب السنوي بقاعدة الحارسة ذات الجملة الواحدة
        للنهر كله، مستعملًا \emph{باردًا} و\emph{مخضوضًا}
        و\emph{غير محمَّل} بالنفايات العضوية.
\end{enumerate}
\end{problem}
