\section*{Bab \ref{ch:g6:plants-are-producers} --- Tumbuhan adalah Produsen}

\begin{solution}{exo:g6:plants-are-producers:1}
\omterm{def:g6:plants-are-producers:matter}{Materi mineral}: materi dunia yang tak hidup --- air, batuan (juga gas
di udara, \omterm{prop:g4:what-plants-need:needs}{garam mineral}). \omterm{def:g6:plants-are-producers:matter}{Materi organik}: materi yang dibuat \omterm{def:g6:exploring-our-environment:components}{makhluk
hidup} dan menyusun tubuhnya --- kayu, \omterm{def:g3:bones-and-muscles:muscle}{otot} (juga gula, wol, lemak).
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:2}
Masukannya: air dan \omterm{prop:g4:what-plants-need:needs}{garam mineral} dari tanah, sebuah gas (karbon
dioksida) dari udara. \omterm{prop:g3:balanced-diet:jobs}{Energinya}: \omterm{def:g6:exploring-our-environment:conditions}{cahaya}, yang ditangkap zat hijaunya.
Keluarannya: \omterm{def:g6:plants-are-producers:matter}{materi organik} --- gula, lalu kayu, \omterm{def:g1:plants-around-us:parts}{daun}, \omterm{def:g1:plants-around-us:parts}{akar}, \omterm{def:g2:from-seed-to-plant:seed}{biji}.
Bengkelnya: \omterm{def:g1:plants-around-us:parts}{daun} hijaunya, pada siang hari.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:3}
Mineral: embun, granit, garam laut. Organik: lilin lebah, minyak
goreng --- dan \omterm{def:g1:animals-around-us:covering}{cangkang} siputnya adalah buatan siputnya, yang dibangun
dari zat mineral yang diendapkan \omterm{def:g1:animals-around-us:animal}{hewannya}: dibuat oleh \omterm{def:g6:exploring-our-environment:components}{makhluk hidup},
dari pasokan mineral. (Kedua jawaban diberi nilai kalau disertai
penalaran itu.)
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:4}
Kebanyakan dari udara dan air --- gas yang diambil dari udara dan air
yang diminum \omterm{def:g1:plants-around-us:parts}{akarnya} --- yang dirangkai di dalam \omterm{def:g1:plants-around-us:parts}{daunnya} dengan \omterm{prop:g3:balanced-diet:jobs}{energi}
\omterm{def:g6:exploring-our-environment:conditions}{cahaya}; tanahnya hanya memberi sejumlah kecil \omterm{prop:g4:what-plants-need:needs}{garam mineral}.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:5}
Ia mengambil karbon dioksida dan melepaskan oksigen. Setiap \omterm{def:g6:exploring-our-environment:components}{makhluk
hidup} yang bernapas mengembuskan gas yang pertama dan membutuhkan gas
yang kedua --- termasuk kita sendiri.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:6}
\omterm{ex:g4:plants-without-seeds:bulbtuber}{Umbi batang} kentang dan \omterm{ex:g4:plants-without-seeds:bulbtuber}{umbi lapis} tulip (gudang bawah tanah), kedua
belahan tebal kacang (\omterm{def:g3:animals-through-seasons:active}{cadangan makanan} \omterm{def:g2:from-seed-to-plant:seed}{bijinya}), \omterm{def:g1:plants-around-us:parts}{akar} wortel yang
gemuk, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} bit yang penuh gula --- semuanya cadangan organik yang
ditabung dari pembuatan musim panas.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:7}
Karena pembuatannya berjalan dengan \omterm{def:g6:exploring-our-environment:conditions}{cahaya}: tanpa \omterm{def:g6:exploring-our-environment:conditions}{cahaya} tidak ada
\omterm{prop:g3:balanced-diet:jobs}{energi} untuk membangun \omterm{def:g6:plants-are-producers:matter}{materi organik}, mineral apa pun yang menunggu
di dalam tanahnya. \omterm{def:g6:exploring-our-environment:conditions}{Cahaya} adalah satu-satunya masukan yang tidak punya
pengganti --- garamnya adalah bumbu, bukan bahan bakar.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:8}
Karena mata rantai pertama sebuah rantai harus memasukkan \omterm{def:g6:plants-are-producers:matter}{materi
organik} yang baru ke dalam dunia yang hidup, sedangkan \omterm{def:g1:animals-around-us:animal}{hewan} adalah
\omterm{def:g5:food-webs:roles}{konsumen}: mereka hanya dapat mengubah \omterm{def:g6:plants-are-producers:matter}{materi organik} yang sudah dibuat
orang lain. Hanya \omterm{def:g5:food-webs:roles}{produsen} yang membuatnya dari pasokan mineral ---
sehingga mata rantai pertamanya \omterm{def:g6:life-through-seasons:trees}{selalu hijau}.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:9}
Roti: gandum --- \omterm{def:g2:from-seed-to-plant:seed}{biji} tanaman gandumnya; jejaknya berakhir di \omterm{def:g1:plants-around-us:parts}{daun}
gandumnya. Mentega: krim --- \omterm{def:g2:animals-grow-up:born}{susu} --- sapinya --- rumput; jejaknya
berakhir di \omterm{def:g1:plants-around-us:parts}{daun} rumputnya. Keju: \omterm{def:g2:animals-grow-up:born}{susu} --- sapinya --- rumput lagi.
Tiga makanan, tiga (atau dua) \omterm{def:g1:plants-around-us:parts}{daun}, semuanya di dalam \omterm{def:g6:exploring-our-environment:conditions}{cahaya}.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:10}
Jerami adalah makanan sapinya --- \omterm{def:g6:plants-are-producers:matter}{materi organik} yang diubahnya
menjadi tubuhnya. Kompos bukanlah makanan \omterm{def:g1:plants-around-us:plant}{tumbuhannya}: ia mengisi
ulang garam \emph{mineral} tanahnya sambil melapuk. \omterm{def:g1:plants-around-us:plant}{Tumbuhannya} lalu
membuat \omterm{def:g6:plants-are-producers:matter}{materi organiknya} sendiri dari mineral itu, udara, air dan
\omterm{def:g6:exploring-our-environment:conditions}{cahaya}. Sapinya menyantap santapan; \omterm{def:g1:plants-around-us:plant}{tumbuhannya} membuat santapannya.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:11}
Kalau tersinari, pembuatan siang hari oleh \omterm{def:g1:plants-around-us:plant}{tumbuhannya} melepaskan
oksigen dan menyerap karbon dioksida, sehingga mengimbangi \omterm{def:g4:breathing:movements}{pernapasan}
\omterm{prop:g3:sorting-living-things:groups}{ikannya} --- pertukaran gasnya berjalan ke dua arah dan toplesnya
berdaur. Kalau gelap, pembuatannya berhenti sementara setiap
penghuninya (termasuk \omterm{def:g1:plants-around-us:plant}{tumbuhannya}) terus bernapas: oksigennya habis,
karbon dioksidanya menumpuk, dan dunia tertutup itu tercekik.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:12}
Benar: materi pohonnya memang tidak datang dari tanahnya ---
penimbangannya membuktikan hal itu untuk selamanya. Belum lengkap: air
hanyalah satu bahan utamanya; ia tidak dapat mengetahui tentang gas
yang diambil dari udara, yang memasok sebagian besar zat kayunya.
Tambahan modernnya: karbon dioksida dari udara (ditambah sejumlah
kecil \omterm{prop:g4:what-plants-need:needs}{garam mineral} tanahnya) --- air dan udara bersama-sama, yang
dirangkai oleh \omterm{def:g6:exploring-our-environment:conditions}{cahaya}.
\end{solution}

\begin{solution}{pb:g6:plants-are-producers:1}
\textbf{1.} Ton-tonnya adalah \omterm{def:g6:plants-are-producers:matter}{materi organik} yang dibuat \omterm{def:g1:plants-around-us:plant}{tumbuhannya}
sendiri dari pasokan mineral --- air, karbon dioksida udaranya, garam
tanahnya --- dengan \omterm{def:g6:exploring-our-environment:conditions}{cahaya} sebagai \omterm{prop:g3:balanced-diet:jobs}{energinya}. Karungnya mengisi ulang
mineralnya; udara dan hujannya mengantarkan bagian terbesarnya;
\omterm{def:g1:plants-around-us:parts}{daunnya} yang mengerjakan pembuatannya.

\textbf{2.} Kentang: organ simpanan bawah tanah (\omterm{ex:g4:plants-without-seeds:bulbtuber}{umbi batang}). Wortel:
\omterm{def:g1:plants-around-us:parts}{akar} simpanan. \omterm{def:g2:from-seed-to-plant:seed}{Biji} kacang: \omterm{def:g3:animals-through-seasons:active}{cadangan makanan} \omterm{def:g2:from-seed-to-plant:seed}{bijinya}, yang dikemas
bagi \omterm{def:g1:plants-around-us:plant}{tumbuhan} berikutnya. \omterm{def:g1:plants-around-us:parts}{Daun} selada: organ yang bekerja itu sendiri
--- bengkel penangkap \omterm{def:g6:exploring-our-environment:conditions}{cahayanya}, yang disantap segar.

\textbf{3.} Pemasok yang sesungguhnya adalah \omterm{def:g6:exploring-our-environment:conditions}{cahaya}: setiap gram
hasilnya dibangun dengan \omterm{prop:g3:balanced-diet:jobs}{energi} \omterm{def:g6:exploring-our-environment:conditions}{cahaya}, pada siang hari, di dalam
\omterm{def:g1:plants-around-us:parts}{daunnya}.

\textbf{4.} Untuk \omterm{prop:g4:what-plants-need:needs}{garam mineralnya}: panen mengangkutnya pergi, dan
tanpa pengisian ulang \omterm{def:g1:plants-around-us:plant}{tumbuhannya} akan kekurangan mineral seperti \omterm{def:g1:plants-around-us:parts}{daun}
mint di dalam pasir --- kecil dalam jumlah, tak tergantikan dalam
fungsi.

\textbf{5.} Percobaan lemari yang gelap (si tawanan pucat): kainnya
menyingkirkan \omterm{def:g6:exploring-our-environment:conditions}{cahaya}, dan seminggu saja sudah terlihat --- pucat,
memanjang, pembuatannya lemah.

\textbf{6.} Umbinya diisi oleh bagian atasnya: \omterm{def:g1:plants-around-us:parts}{daunnya} membuat gula
pada siang hari, dan kelebihannya dipipakan turun lalu ditabung di
dalam umbi yang membengkak. \omterm{def:g1:plants-around-us:parts}{Daun} yang bekerja dengan rimbun berarti
kelebihan yang besar --- bagian atas yang baik, umbi yang baik.

\textbf{7.} Air. Pembuatannya membutuhkan semua masukannya sekaligus:
\omterm{def:g6:exploring-our-environment:conditions}{cahaya} memasok \omterm{prop:g3:balanced-diet:jobs}{energi}, bukan zat --- kalau \omterm{def:g1:plants-around-us:parts}{akarnya} kering, tidak ada
yang dapat dibangun, secerah apa pun langitnya.

\textbf{8.} \omterm{def:g1:plants-around-us:parts}{Akar} yang dibenamkan itu adalah \omterm{def:g6:plants-are-producers:matter}{materi organik} yang
dikembalikan ke dalam tanah: sambil melapuk, ia akan membebaskan
mineralnya bagi tanaman yang akan datang --- aturan pengembalian pada
\cref{prop:g5:people-and-nature:lasting}, yang dibayar dengan materi
\omterm{def:g1:plants-around-us:plant}{tumbuhan}. Pemborosannya justru kalau \omterm{def:g1:plants-around-us:parts}{akarnya} dibakar.

\textbf{9.} Tanaman kentang $\rightarrow$ babi $\rightarrow$
keluarganya. Seluruh \omterm{def:g6:plants-are-producers:matter}{materi organiknya} pertama kali dibuat di dalam
\omterm{def:g1:plants-around-us:parts}{daun} tanaman kentangnya --- babinya dan keluarganya hanya
mengubahnya.

\textbf{10.} Bahannya: air dan \omterm{prop:g4:what-plants-need:needs}{garam mineral} dari tanah, karbon
dioksida dari udara --- yang dirangkai menjadi materi \omterm{def:g1:plants-around-us:parts}{daun}. Peran
mataharinya yang tepat: \omterm{prop:g3:balanced-diet:jobs}{energi}, bukan zat. Saladnya adalah air, udara
dan tanah, yang \emph{dibangun oleh} \omterm{def:g6:exploring-our-environment:conditions}{cahaya} matahari --- pujian
puitisnya layak, daftar bahannya keliru.

\textbf{11.} Para \omterm{def:g5:food-webs:roles}{pengurainya} --- kehidupan tanah mulai dari cacing
dan kutu kayu sampai \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang terlalu kecil untuk dilihat
--- makan dari \omterm{def:g1:plants-around-us:parts}{daun} yang mati lalu memecah \omterm{def:g6:plants-are-producers:matter}{materi organiknya} kembali
menuju \omterm{def:g6:plants-are-producers:matter}{materi mineral}, sehingga mengisi ulang tanah yang akan ditimba
tanaman berikutnya.

\textbf{12.} Misalnya: ``\omterm{def:g5:food-webs:roles}{Produsen} di kebunnya membangun \omterm{def:g6:plants-are-producers:matter}{materi organik}
--- yaitu panennya --- dari \omterm{def:g6:plants-are-producers:matter}{materi mineral} yang berasal dari tanah,
air dan udara, dengan memakai \omterm{def:g6:exploring-our-environment:conditions}{cahaya} sebagai \omterm{prop:g3:balanced-diet:jobs}{energinya}.''
\end{solution}
