\chapter{आहार जाल}\label{ch:g5:food-webs}

\Cref{ch:g3:food-chains} ने हमें चेताया था कि उसकी सुथरी शृंखलाएँ किसी
जाल से खींचे गए अकेले धागे थीं: बाज़ चूहे भी उठाता है, छिपकली मक्खियाँ
भी खाती है, और घास ख़रगोशों का भी पोषण करती है। इस साल हम धागे खींचना
छोड़कर पूरे जाल को देखेंगे --- यानी \omterm{def:g5:food-webs:web}{आहार जाल} --- और उसे भी जो उसका
संतुलन थामे रखता है।

\section{शृंखलाओं से जालों तक}

\begin{definition}[आहार जाल]\label{def:g5:food-webs:web}
किसी \omterm{def:g2:where-animals-live:habitat}{आवास} का \emph{आहार जाल}\index{आहार जाल} उसकी \emph{सारी} \omterm{def:g3:food-chains:chain}{आहार
शृंखलाओं} का एक साथ बना जाल है: हर \omterm{def:g5:biodiversity:species}{प्रजाति} एक बार खींची जाती है, और हर
भोजन-संबंध के लिए एक तीर (जिसका अर्थ अब भी \emph{\omterm{def:g3:food-chains:chain}{द्वारा खाया जाता है}}
ही है, \cref{rem:g3:food-chains:arrow})। ज़्यादातर \omterm{def:g1:animals-around-us:animal}{जंतु} कई \omterm{def:g5:biodiversity:species}{प्रजातियाँ}
खाते हैं और कई के खाए जाते हैं --- इसलिए शृंखलाएँ आपस में कटती हैं, और
चित्र जाल बन जाता है।
\end{definition}

\begin{example}[जाल को पढ़ना]\label{ex:g5:food-webs:reading}
चित्र के दो पाठ। ख़रगोश के तीरों के पीछे चलो: वह घास \emph{और} तिपतिया,
दोनों खाता है, और लोमड़ी \emph{तथा} बाज़, दोनों के खाया जाता है ---
यानी चार संबंध, जहाँ शृंखला ने दो दिखाए थे। और बाज़ के भोजन-तीर गिनो:
छिपकली और ख़रगोश --- एक चला जाए तो दूसरा बचा रहता है। जाल ठीक वे बचे
हुए विकल्प दर्ज करता है जिनकी \cref{prop:g5:biodiversity:web} ने
तारीफ़ की थी।
\end{example}

\section{जाल में भूमिकाएँ}

\begin{definition}[उत्पादक और उपभोक्ता]\label{def:g5:food-webs:roles}
जाल की \omterm{def:g5:biodiversity:species}{प्रजातियाँ} भूमिकाओं में बँट जाती हैं:
\begin{itemize}
  \item \emph{उत्पादक}\index{उत्पादक}: यानी \omterm{def:g1:plants-around-us:plant}{पौधे}, जो अपना पदार्थ ख़ुद
        बनाते हैं (\cref{prop:g4:what-plants-need:make}) --- हर
        तीर-रास्ता किसी उत्पादक से ही शुरू होता है
        (\cref{prop:g3:food-chains:start});
  \item \emph{उपभोक्ता}\index{उपभोक्ता}: यानी \omterm{def:g1:animals-around-us:animal}{जंतु}, जो दूसरों पर जीते
        हैं --- \omterm{def:g2:what-animals-eat:kinds}{शाकाहारी} सीधे उत्पादकों को खाते हैं, \omterm{def:g2:what-animals-eat:kinds}{मांसाहारी} दूसरे
        उपभोक्ताओं को, और \omterm{def:g2:what-animals-eat:kinds}{सर्वाहारी} दोनों को
        (\cref{def:g2:what-animals-eat:kinds});
  \item और पर्दे के पीछे काम करते \emph{अपघटक}\index{अपघटक} ---
        \cref{ex:g2:caring-for-nature:compost} के गलाने वाले, जो मरे
        हुए अवशेषों पर पलते हैं और उनका पदार्थ मिट्टी को लौटा देते
        हैं। आगे का एक साल उन्हें उनका अपना अध्याय देता है।
\end{itemize}
\end{definition}

\begin{example}[चित्र को छाँटना]\label{ex:g5:food-webs:sorting}
जेबी जाल में: \omterm{def:g5:food-webs:roles}{उत्पादक} --- घास, तिपतिया, बलूत; \omterm{def:g2:what-animals-eat:kinds}{शाकाहारी} \omterm{def:g5:food-webs:roles}{उपभोक्ता} ---
टिड्डा, ख़रगोश, \omterm{ex:g2:animals-grow-up:butterfly}{इल्ली}; \omterm{def:g2:what-animals-eat:kinds}{मांसाहारी} \omterm{def:g5:food-webs:roles}{उपभोक्ता} --- छिपकली, नीली चिड़िया
(ज़्यादातर), लोमड़ी (सच में तो \omterm{def:g2:what-animals-eat:kinds}{सर्वाहारी}), बाज़, शिकरा। तीन मंज़िलें:
हरी, \omterm{def:g1:plants-around-us:plant}{पौधे} खाने वाली, माँस खाने वाली --- और हर ऊपरी मंज़िल अपने नीचे
वालों पर खड़ी।
\end{example}

\begin{proposition}[संख्याओं की शक्ल]\label{prop:g5:food-webs:pyramid}
जाल के बाशिंदों को मंज़िल दर मंज़िल गिनने पर एक ही शक्ल बार-बार मिलती
है: \emph{बहुत-से} \omterm{def:g5:food-webs:roles}{उत्पादक}, \emph{कम} \omterm{def:g2:what-animals-eat:kinds}{शाकाहारी}, \emph{बहुत कम}
\omterm{def:g2:what-animals-eat:kinds}{मांसाहारी} --- लाखों घास के \omterm{def:g1:plants-around-us:plant}{पौधों} वाला मैदान हज़ारों टिड्डों का पोषण
करता है, वे मुट्ठी भर छिपकलियों का, और वे बाज़ों की एक जोड़ी का। हर
मंज़िल अपने नीचे वाली पर जीती है और अपने भोजन से ज़्यादा नहीं हो सकती।
इसीलिए बड़े शिकारी हमेशा दुर्लभ होते हैं।
\end{proposition}
\admitted

\section{संतुलन, और उसे बिगाड़ने वाली बातें}

\begin{proposition}[जाल का संतुलन]\label{prop:g5:food-webs:balance}
अपने हाल पर छोड़ दिया जाए तो जाल अपनी संख्याएँ मोटे तौर पर
\emph{संतुलन}\index{संतुलन (आहार-जाल का)} में रखता है: ख़रगोश बढ़ें तो
उनके शिकारी बेहतर खाते हैं और उनके पीछे-पीछे ख़ुद भी बढ़ जाते हैं, और
अतिरिक्त ख़रगोश घुल जाते हैं; ख़रगोश घटें तो भूखे शिकारी दूसरे शिकार
(वही बचे हुए तीर) पर चले जाते हैं और ख़रगोशों की संख्या सँभल जाती है।
\cref{prop:g3:food-chains:depend} वाले सुधार, कई तीरों पर एक साथ
चलते हुए, पूरे जाल को ठहराव देते हैं।
\end{proposition}
\admitted

\begin{example}[वह मशहूर ग़लती]\label{ex:g5:food-webs:mistake}
कभी किसान शिकारी \omterm{prop:g3:sorting-living-things:groups}{पक्षियों} को मिटाने का अभियान चलाते थे --- ``मुर्ग़ी
चुराने वाले''। जहाँ वे कामयाब हुए, वहाँ चूहे और छछूँदर-चूहे अपने मुख्य
शिकारियों से छूटकर इतने बढ़े कि उन्होंने बाज़ों की कभी-कभार की मुर्ग़ी
से कहीं ज़्यादा अनाज खा डाला। यह माली और घोंघों वाली कहानी
(\cref{exo:g3:food-chains:9}) ही है, पर पूरे देहात के पैमाने पर: जाल
की ऊपरी मंज़िल हटाओ और उसके नीचे वाली फट पड़ती है।
\end{example}

\begin{method}[गड़बड़ी की भविष्यवाणी]\label{met:g5:food-webs:predict}
किसी \omterm{def:g5:biodiversity:species}{प्रजाति} के हटने (या जुड़ने) का असर बूझने के लिए:
\begin{enumerate}
  \item जाल में उस \omterm{def:g5:biodiversity:species}{प्रजाति} को ढूँढ़ो और उसके \emph{सारे} तीर गिनाओ,
        भीतर आते और बाहर जाते;
  \item उसका भोजन, कम खाया जाकर, बढ़ने लगता है; और उसके खाने वाले,
        ग़रीब होकर, घटते हैं या शिकार बदल लेते हैं;
  \item हर बदलाव के पीछे जाल में एक क़दम और चलो --- दूसरे क़दम के असर
        अकसर चौंका देते हैं;
  \item जाँचो कि किन पड़ोसियों के पास बचे हुए तीर हैं (वे झटका सोख
        लेते हैं) और कौन सिर्फ़ इसी \omterm{def:g5:biodiversity:species}{प्रजाति} पर निर्भर थे (वे ख़तरे में
        हैं)।
\end{enumerate}
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:1}
\omterm{def:g5:food-webs:web}{आहार जाल} क्या है, और शृंखलाएँ आपस में क्यों कटती हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:2}
\cref{def:g5:food-webs:roles} की तीनों भूमिकाओं के नाम बताओ, और चित्र
से हर एक का एक उदाहरण भी (अपघटकों को छोड़कर)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:3}
चित्र से: नीली चिड़िया जो कुछ खाती है और जो कुछ उसे खाता है, सब गिनाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:4}
चित्र से दो पूरी शृंखलाएँ निकालो जिनमें टिड्डा साझा हो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:5}
संख्याओं की शक्ल मंज़िल दर मंज़िल बताओ। कोई मंज़िल अपने नीचे वाली से
ज़्यादा क्यों नहीं हो सकती?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:6}
एक वाक्य में: बाज़ दुर्लभ और टिड्डे अनगिनत क्यों हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:7}
ख़रगोशों का साल अच्छा रहा और वे बढ़ गए।
\cref{prop:g5:food-webs:balance} के अनुसार आगे क्या होता है --- और
अतिरिक्त ख़रगोश किससे घुलते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:8}
करके देखो: अपने जाने हुए किसी \omterm{def:g2:where-animals-live:habitat}{आवास} --- बग़ीचा, पार्क, तालाब --- का \omterm{def:g5:food-webs:web}{आहार
जाल} बनाओ, कम से कम छह \omterm{def:g5:biodiversity:species}{प्रजातियों} और हर उस तीर के साथ जिसका तुम बचाव कर
सको। \omterm{def:g5:food-webs:roles}{उत्पादकों} को हरे से चिह्नित करो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:food-webs:9}
चित्र से छिपकली हटाने पर \cref{met:g5:food-webs:predict} चलाओ: पहले
क़दम के असर, फिर एक क़दम और।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:food-webs:10}
\cref{ex:g5:food-webs:mistake} को भविष्यवाणियों की शृंखला के रूप में
दोहराओ: बाज़ हटे, फिर क्या, फिर क्या? अभियान चलाने वाले कौन-सा क़दम नहीं
बूझ पाए?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:food-webs:11}
चित्र के जाल में बलूत (एक \omterm{def:g5:food-webs:roles}{उत्पादक}) हटाने की तुलना शिकरे (एक शीर्ष
शिकारी) हटाने से करो। हर नुक़सान पहले किन पड़ोसियों को महसूस होता है,
हर गड़बड़ी किस दिशा में सफ़र करती है, और यहाँ जाल किस नुक़सान को ज़्यादा
आसानी से सोख लेता है?
\cref{ex:g3:food-chains:dry,ex:g3:food-chains:hunter} का इस्तेमाल करो।
\end{exercise}
