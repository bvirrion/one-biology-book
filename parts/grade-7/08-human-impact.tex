\chapter{Humans and the Environment}\label{ch:g7:impact}

Humans reshape \omterm{def:g7:eco-energy:ecosystem}{ecosystems} at local and global scales. Understanding
impacts is the first step toward wiser choices --- \omterm{def:g7:impact:conserve}{conservation},
restoration and sustainable use.

\section{How people change ecosystems}

\begin{definition}[Habitat loss]\label{def:g7:impact:habitat}
\emph{Habitat}\index{habitat} is the \omterm{def:g1:caring:needs}{place} where an \omterm{def:g4:ecosystems:levels}{organism} lives.
\emph{Habitat loss}\index{habitat!loss} and fragmentation (splitting into
isolated patches) reduce space and resources for wild \omterm{def:g6:classification:species}{species} --- a major
cause of \omterm{def:g6:biodiversity:def}{biodiversity} decline.
\end{definition}

\begin{definition}[Pollution]\label{def:g7:impact:pollution}
\emph{Pollution}\index{pollution} is the introduction of harmful
substances or \omterm{def:g6:nutrition:def}{energy} into the \omterm{def:g4:traits:env}{environment} (chemicals, plastics, excess
\omterm{def:g4:plants-need:needs}{nutrients}, noise, \omterm{def:g4:plants-need:needs}{light}). Pollutants can damage \omterm{def:g4:ecosystems:levels}{organisms} directly or
through \omterm{def:g7:eco-energy:chain}{food webs}.
\end{definition}

\begin{definition}[Invasive species]\label{def:g7:impact:invasive}
An \emph{invasive species}\index{invasive species} is a non-native \omterm{def:g6:classification:species}{species}
that spreads and harms \omterm{def:g7:eco-energy:ecosystem}{ecosystems}, economy or \omterm{def:g2:healthy:health}{health} --- often because
natural \omterm{def:g5:food-webs:pred}{predators} or \omterm{def:g6:method:control}{controls} are missing.
\end{definition}

\begin{example}[Overfishing]\label{ex:g7:impact:fish}
Removing \omterm{def:g5:vertebrates:groups}{fish} faster than \omterm{def:g6:ecology:levels}{populations} can recover collapses fisheries and
can rewire marine \omterm{def:g7:eco-energy:chain}{food webs} (for example fewer \omterm{def:g5:food-webs:pred}{predators} or fewer grazers).
\end{example}

\section{Climate and carbon}

\begin{definition}[Greenhouse effect]\label{def:g7:impact:ghg}
Certain gases in the atmosphere (including carbon dioxide and methane)
trap heat --- the \emph{greenhouse effect}\index{greenhouse effect}. Human
burning of \omterm{def:g5:evolution:fossil}{fossil} fuels and some land-use changes raise greenhouse gas
levels, enhancing warming and shifting climates.
\end{definition}

\begin{proposition}[Link to ecosystems]\label{prop:g7:impact:climate}
Changing temperature and rainfall patterns move suitable ranges for
\omterm{def:g6:classification:species}{species}, stress coral reefs, alter fire and ice regimes, and interact with
\omterm{def:g4:protecting:habitat-loss}{habitat loss}. Not every local weather event is ``climate'', but long-term
trends reshape \omterm{def:g6:ecology:levels}{biomes}.
\end{proposition}
\admitted

\section{Responses}

\begin{definition}[Conservation]\label{def:g7:impact:conserve}
\emph{Conservation}\index{conservation} aims to protect \omterm{def:g6:classification:species}{species},
\omterm{def:g7:impact:habitat}{habitats} and ecological processes. Tools include protected areas, laws,
restoration planting, \omterm{def:g7:impact:pollution}{pollution} \omterm{def:g6:method:control}{control}, sustainable harvest rules and
reducing greenhouse gas emissions.
\end{definition}

\begin{method}[Thinking about an impact]\label{met:g7:impact:think}
\begin{enumerate}
  \item Name the human activity.
  \item Identify which part of the \omterm{def:g7:eco-energy:ecosystem}{ecosystem} is hit first (\omterm{def:g7:impact:habitat}{habitat},
        \omterm{def:g4:plants-need:needs}{water}, \omterm{def:g4:plants-need:needs}{air}, \omterm{def:g6:classification:species}{species}).
  \item Trace one food-web or resource consequence.
  \item Propose one mitigation or restoration step and who could do it.
\end{enumerate}
\end{method}

\begin{example}[Local action]\label{ex:g7:impact:local}
Schoolyard native planting can rebuild \omterm{def:g3:insects:insect}{insect} \omterm{def:g7:impact:habitat}{habitat}; reducing plastic
litter protects waterways; walking or public transport cuts some \omterm{def:g5:evolution:fossil}{fossil}
fuel use. Small actions scale when many people share them.
\end{example}
