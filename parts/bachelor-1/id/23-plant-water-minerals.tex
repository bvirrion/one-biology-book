\chapter{Nutrisi Air dan Mineral Tumbuhan}\label{ch:b1:plant-water-minerals}

Sebatang jagung pada bulan Juli menarik seliter air sehari dari tanah
yang terasa kering di tangan, dan bersamanya tiga puluh miligram
nitrogen, sebagai nitrat, yang diperlukannya untuk membangun
pertumbuhan sehari. Ia tak memiliki pompa. Airnya masuk sebab \omterm{def:b1:flowering-plant-organization:organs}{akarnya}
lebih kering daripada tanahnya, dalam arti yang saksama pada
\cref{ch:b1:membranes-transport}; nitratnya masuk sebab \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{akarnya}
membelanjakan \omterm{def:b1:water-small-molecules:atp}{ATP} untuk itu; dan keduanya harus melewati selapis
tunggal \omterm{def:b1:cell-unit-of-life:cell}{sel} yang dindingnya tersegel \omterm{def:b1:lipids:wax}{lilin} sebelum mencapai \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}.
Bab ini memerikan \omterm{def:b1:flowering-plant-organization:organs}{akarnya} sebagai \omterm{def:b1:mammal-organization:organ}{organ} penyerap, \omterm{def:b1:membranes-transport:osmosis}{potensial air} yang
memacu air dari tanah ke \omterm{def:b1:flowering-plant-organization:tissues}{xilem}, unsur mineral yang diperlukan sebatang
tumbuhan dan cara ia mengambilnya, hal khusus nitrogennya, serta jamur
dan bakteri yang dipekerjakan sebagian besar \omterm{def:b1:flowering-plant-organization:organs}{akar} untuk melakukan
sebagian pekerjaannya.

\section{Akar sebagai organ penyerap}

\begin{definition}[Apoplas, simplas, endodermis]\label{def:b1:plant-water-minerals:pathways}
\index{apoplas}\index{simplas}\index{pita Kaspari}
Air dan zat terlarut melintasi korteks \omterm{def:b1:flowering-plant-organization:organs}{akarnya} lewat dua jalur:
\emph{apoplas}, yaitu sistem sinambung \omterm{prop:b1:carbohydrates:wall}{dinding sel} dan ruang antarsel,
yang lewatnya keduanya bergerak tanpa melintasi membran mana pun; dan
\emph{simplas}, yaitu \omterm{def:b1:cell-unit-of-life:cell}{sitoplasma} sinambung \omterm{def:b1:cell-unit-of-life:cell}{sel} yang terhubung
\omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesma} (\cref{ch:b1:eukaryotic-cell}), yang dimasuki dengan
melintasi satu membran plasma. Pada batas dalam korteksnya
\emph{\omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}} (\cref{ch:b1:flowering-plant-organization})
menghalangi apoplasnya: sebuah pita suberin dan lignin, yaitu
\emph{pita Kaspari}, membuluh dinding radialnya, sehingga segala yang
memasuki silinder pembuluhnya harus melewati membran sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}
\omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}. Karena itu pengangkut membrannya memutuskan apa yang
mencapai \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}, lalu mencegah apa yang telah dimuatkan ke dalam
\omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} bocor kembali.
\end{definition}

\begin{proposition}[Tekanan akar]\label{prop:b1:plant-water-minerals:rootpressure}
\index{tekanan akar}\index{gutasi}
\omterm{prop:b1:flowering-plant-organization:stemroot}{Endodermisnya} dan \omterm{def:b1:cell-unit-of-life:cell}{sel} di dalamnya memompa ion ke dalam \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}, yang
menurunkan \omterm{def:b1:membranes-transport:osmosis}{potensial air} \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} lalu menarik air masuk secara
\omterm{def:b1:membranes-transport:osmosis}{osmosis} dari korteksnya; ketika tajuknya sedikit bertranspirasi (pada
malam hari, dalam udara lembap), airnya bertimbun dan getah \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}
terdorong ke atas di bawah sebuah \emph{\omterm{prop:b1:plant-water-minerals:rootpressure}{tekanan akar}} positif sebesar
beberapa persepuluh megapascal --- cukup untuk memaksa tetesan keluar
dari ujung \omterm{def:b1:flowering-plant-organization:organs}{daun} tumbuhan kecil (\emph{gutasi}) lalu mengisi ulang
pembuluh yang tersumbat udara, tetapi tidak cukup untuk mengangkat
getahnya ke puncak sebatang pohon. Pada siang hari tarikan
transpirasinya (\cref{ch:b1:plant-transport}) mengambil alih dan
\omterm{prop:b1:plant-water-minerals:rootpressure}{tekanan akarnya} lenyap.
\end{proposition}
\begin{proof}[Bukti]
Sebuah \omterm{def:b1:flowering-plant-organization:organs}{batang} yang dipotong dekat tanahnya mengeluarkan getah selama
berjam-jam dari tunggulnya, dan sebuah manometer yang dipasang pada
tunggulnya mencatat tekanan sebesar \qtyrange{0.1}{0.3}{MPa};
keluarannya lebih kaya ion daripada larutan tanahnya, dan alirannya
berhenti ketika \omterm{def:b1:flowering-plant-organization:organs}{akarnya} didinginkan atau diracun dengan sebuah
\omterm{def:b1:enzymes:inhibitors}{penghambat} respirasi: alirannya dipacu pemuatan ion secara aktif,
bukan oleh pemompaan air mana pun.
\end{proof}

\section{Air: dari tanah ke xilem}

\begin{proposition}[Tangga potensial airnya]\label{prop:b1:plant-water-minerals:ladder}
\index{potensial air}
Air bergerak dari tanah ke \omterm{def:b1:flowering-plant-organization:tissues}{xilem} sebab \omterm{def:b1:membranes-transport:osmosis}{potensial airnya}
(\cref{ch:b1:membranes-transport}) turun pada tiap langkahnya:
\begin{center}
\small
\begin{tabular}{lr}
\hline
petak & $\Psi$ (\unit{MPa}) \\
\hline
tanah lembap & $-0.03$ sampai $-0.3$ \\
tanah pada titik layu & $-1.5$ \\
\omterm{def:b1:cell-unit-of-life:cell}{sel} korteks \omterm{def:b1:flowering-plant-organization:organs}{akar} & $-0.3$ sampai $-0.6$ \\
\omterm{def:b1:flowering-plant-organization:tissues}{xilem} \omterm{def:b1:flowering-plant-organization:organs}{akar} & $-0.5$ sampai $-0.8$ \\
\omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{daun} & $-1.0$ sampai $-2.0$ \\
udara pada kelembapan nisbi \qty{50}{\%} & $-95$ \\
\hline
\end{tabular}
\end{center}
Potensial tanahnya ditetapkan sebagian besarnya oleh tegangan yang
dengannya pori tanahnya memegang airnya (kapilaritas); potensial
\omterm{def:b1:cell-unit-of-life:cell}{selnya} oleh zat terlarut dan \omterm{def:b1:eukaryotic-cell:plantcell}{turgornya}; potensial \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} oleh
tegangan kolom yang bertranspirasi; potensial udaranya oleh
kelembapannya, $\Psi = (RT/V_w)\ln(\text{RH})$, yang sangat besar
bahkan dalam udara lembap. Langkah yang paling curam sejauh ini adalah
yang terakhir, dari \omterm{def:b1:flowering-plant-organization:organs}{daun} ke udara: di situlah hampir seluruh gaya
pemacunya dibelanjakan, dan itulah sebabnya tumbuhan mengendalikannya
dengan \omterm{def:b1:flowering-plant-organization:tissues}{stomata}.
\end{proposition}

\begin{method}[Membaca sebuah profil potensial air]\label{met:b1:plant-water-minerals:profile}
\begin{enumerate}
  \item Daftarkan petaknya berurutan beserta $\Psi$-nya; air mengalir
        hanya dari yang lebih tinggi ke yang lebih rendah, sehingga
        urutannya harus menurun sepanjang jalurnya atau alirannya
        berbalik.
  \item Belahlah $\Psi$ tiap \omterm{def:b1:cell-unit-of-life:cell}{selnya} menjadi $\Psi_s$ (zat terlarut,
        negatif) dan $\Psi_p$ (\omterm{def:b1:eukaryotic-cell:plantcell}{turgor}, positif); sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{akar} pada
        $\Psi = -0.4$ dengan $\Psi_s = -0.9$ memiliki $\Psi_p = +0.5$.
        Air \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} memiliki $\Psi_s \approx 0$ dan $\Psi_p$ negatif:
        ia berada di bawah tegangan.
  \item Alirannya melintasi sebuah langkah adalah beda $\Psi$ dikali
        konduktans hidraulik sawarnya; \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis} dan \omterm{def:b1:flowering-plant-organization:tissues}{stomatanya} adalah
        kedua langkah berkonduktans rendah tempat tumbuhannya mengatur.
  \item Keringkan tanahnya lalu $\Psi$-nya turun; ketika ia mencapai
        potensial \omterm{def:b1:flowering-plant-organization:organs}{daunnya}, alirannya berhenti dan tumbuhannya layu; ketika
        $\Psi_p$ \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{daunnya} mencapai nol, \omterm{def:b1:flowering-plant-organization:organs}{daunnya} terkulai.
        Pemulihannya menuntut tanahnya, bukan tumbuhannya, yang
        berubah.
\end{enumerate}
\end{method}

\begin{example}[Seliter sehari]\label{ex:b1:plant-water-minerals:litre}
\omterm{def:b1:flowering-plant-organization:organs}{Akar} halus sebatang jagung menyuguhkan sekitar \qty{0.2}{m^2}
permukaan dengan konduktans hidraulik sekitar
\qty{2e-7}{m.s^{-1}.MPa^{-1}}; antara tanah pada \qty{-0.2}{MPa} dan
\omterm{def:b1:flowering-plant-organization:tissues}{xilem} \omterm{def:b1:flowering-plant-organization:organs}{akar} pada \qty{-0.6}{MPa} fluksnya
$2\times 10^{-7}\times 0.4\times 0.2 = \qty{1.6e-8}{m^3/s}$ --- a
seliter setengah sehari, seluruhnya hilang dari \omterm{def:b1:flowering-plant-organization:organs}{daunnya}. \omterm{def:b1:flowering-plant-organization:organs}{Akarnya} tidak
melakukan kerja untuk memindahkannya: transpirasi \omterm{def:b1:flowering-plant-organization:organs}{daunnya} menetapkan
tegangannya, dan tanahnya memasok airnya selama potensialnya tinggal
di atas potensial \omterm{def:b1:flowering-plant-organization:organs}{akarnya}.
\end{example}

\section{Nutrisi mineral}

\begin{definition}[Unsur esensial]\label{def:b1:plant-water-minerals:elements}
\index{unsur esensial}\index{makronutrien}\index{mikronutrien}
Sebuah unsur bersifat \emph{esensial} bagi sebatang tumbuhan bila
tumbuhannya tak dapat merampungkan daur hidupnya tanpanya dan tak ada
unsur lain yang dapat menggantikannya. Ada tujuh belas: karbon,
hidrogen dan oksigen dari udara dan air; \emph{makronutrien} yang
diambil dari tanah dalam gram tiap kilogram massa kering --- nitrogen
(sebagai nitrat atau amonium), fosforus (fosfat), kalium, kalsium,
magnesium, belerang (sulfat); serta \emph{mikronutrien} yang
diperlukan dalam miligram atau kurang --- besi, mangan, seng, tembaga,
boron, molibdenum, klorin, nikel. Nitrogen membangun \omterm{def:b1:proteins:peptide}{protein} dan asam
nukleat; fosforus, \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} dan membran; kalium adalah kation utama
\omterm{def:b1:cell-unit-of-life:cell}{selnya} dan zat osmotik \omterm{def:b1:flowering-plant-organization:tissues}{stomatanya}; magnesium duduk dalam \omterm{def:b1:photosynthesis:pigments}{klorofilnya};
besi dalam sitokrom dan feredoksinnya; molibdenum dalam \omterm{prop:b1:plant-water-minerals:nitrate}{reduktase
nitrat} dan nitrogenasenya.
\end{definition}

\begin{proposition}[Bagaimana daftarnya disusun]\label{prop:b1:plant-water-minerals:hydroponics}
\omterm{def:b1:plant-water-minerals:elements}{Unsur esensialnya} dikenali dengan menumbuhkan tumbuhan dalam air yang
hanya memuat garam yang diketahui.
\end{proposition}
\begin{proof}[Bukti]
Sachs dan Knop (1860-an) menumbuhkan tumbuhan sampai dewasa dalam
larutan beberapa garam mineral, sama sekali tanpa tanah, sehingga
membuktikan bahwa tanahnya tidak memasok apa pun selain mineral dan
sokongan; meniadakan satu garam pada satu waktu memberi sebuah
penyakit yang khas bagi tiap unsurnya, dan menambahkannya kembali
menyembuhkannya. \omterm{def:b1:plant-water-minerals:elements}{Mikronutriennya} ditemukan kemudian, seiring ahli
kimia belajar memurnikan garamnya: sebatang tumbuhan yang dirampas
molibdenumnya, yang diperlukan pada satu bagian dalam sepuluh juta,
gagal mereduksi nitrat. Tekniknya, hidroponik, kini memberi makan
rumah kaca.
\end{proof}

\begin{method}[Membaca sebuah kekurangan]\label{met:b1:plant-water-minerals:deficiency}
\begin{enumerate}
  \item Tanyakan di mana gejalanya muncul. Unsur yang dapat dipindahkan
        tumbuhannya dalam floemnya (N, P, K, Mg) ditarik dari \omterm{def:b1:flowering-plant-organization:organs}{daun} tua
        untuk memberi makan yang muda: \omterm{def:b1:flowering-plant-organization:organs}{daun} tuanya memperlihatkan
        kekurangannya lebih dahulu. Unsur yang tak dapat dipindahkannya
        (Ca, Fe, B) terdampar di tempat ia diletakkan: \omterm{def:b1:flowering-plant-organization:organs}{daun} muda dan
        pucuk tumbuhnya yang memperlihatkannya.
  \item Tanyakan apa yang diperbuat unsurnya: tanpa nitrogen, sedikit
        \omterm{def:b1:photosynthesis:pigments}{klorofil} dan \omterm{def:b1:proteins:peptide}{protein} --- pucat, kerdil; tanpa magnesium atau
        besi, \omterm{def:b1:photosynthesis:pigments}{klorofilnya} gagal --- \omterm{def:b1:flowering-plant-organization:organs}{daun} kuning dengan tulang hijau;
        tanpa kalium, tepi \omterm{def:b1:flowering-plant-organization:organs}{daunnya} hangus; tanpa kalsium, pucuk
        tumbuhnya mati.
  \item Sahkanlah dengan memasok unsur yang dicurigainya saja lalu
        mengamati pertumbuhan barunya.
\end{enumerate}
\end{method}

\begin{proposition}[Serapan ion secara aktif]\label{prop:b1:plant-water-minerals:uptake}
\index{pompa proton}
Larutan tanahnya memegang ion pada mikromol sampai beberapa milimol
tiap liter; \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{akarnya} memegang kalium pada \qty{100}{mmol/L}, fosfat
pada \qty{10}{}, nitrat pada beberapa. Karena itu serapannya sebagian
besarnya menanjak dan dibayar dengan \omterm{def:b1:water-small-molecules:atp}{ATP}, tetapi secara tak langsung:
sebuah \emph{\omterm{prop:b1:plant-water-minerals:uptake}{pompa proton}} dalam membran plasmanya
(\cref{ch:b1:membranes-transport}) mengekspor $\mathrm{H^+}$, sehingga
bagian luarnya asam dan bagian dalamnya negatif sebesar
\qtyrange{-120}{-200}{mV}; kation seperti $\mathrm{K^+}$ lalu masuk
lewat saluran menuruni landaian listriknya, dan anion
($\mathrm{NO_3^-}$, $\mathrm{H_2PO_4^-}$, $\mathrm{SO_4^{2-}}$) lewat
\omterm{prop:b1:membranes-transport:secondary}{simporter} yang menumpang proton yang kembali masuk. Pengangkutnya
bersifat memilih: kalium diserap seratus kali lebih mudah daripada
natrium yang bermuatan sama dan hampir seukuran, dan tumbuhannya
menahan natrium di luar bahkan dalam tanah yang asin. \omterm{def:b1:flowering-plant-organization:organs}{Akar} memakai
sepertiga respirasi tumbuhannya bagi pengangkutan ini.
\end{proposition}

\section{Nitrogen}

\begin{proposition}[Nitrat menjadi asam amino]\label{prop:b1:plant-water-minerals:nitrate}
\index{reduktase nitrat}\index{asimilasi nitrogen}
Sebagian besar tumbuhan mengambil nitrogen sebagai nitrat. Dalam
\omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya} \emph{\omterm{prop:b1:plant-water-minerals:nitrate}{reduktase nitrat}}, sebuah \omterm{def:b1:enzymes:enzyme}{enzim} molibdenum,
mereduksinya menjadi nitrit dengan elektron dari NADH; dalam plastidnya
\emph{reduktase nitrit} mereduksi nitritnya menjadi amonium dengan
elektron dari feredoksin; dan amoniumnya segera dipancangkan menjadi
glutamina dan \omterm{def:b1:biosyntheses-integration:nitrogen}{glutamat} (\cref{ch:b1:biosyntheses-integration}) ---
amoniumnya sendiri beracun dan tak pernah bertimbun. Seluruh
reduksinya memakan biaya delapan elektron tiap nitrogen, sepersepuluh
alir elektron fotosintesis sebuah \omterm{def:b1:flowering-plant-organization:organs}{daun} pada tanah yang kaya nitrat,
dan sebuah \omterm{def:b1:flowering-plant-organization:organs}{daun} melakukan sebagian besarnya dalam terang, langsung
dari feredoksin kloroplasnya. Amonium, tempat tanahnya menyuguhkannya,
diserap langsung lalu melewatkan biayanya.
\end{proposition}

\begin{definition}[Fiksasi nitrogen dan bintil]\label{def:b1:plant-water-minerals:nodules}
\index{fiksasi nitrogen}\index{bintil akar}\index{Rhizobium}
Tak ada eukariot yang dapat memakai $\mathrm{N_2}$ udaranya. Sebagian
bakteri dapat: \emph{nitrogenase} memutuskan ikatan rangkap tiga
$\mathrm{N_2}$ lalu mereduksinya menjadi dua amonia, dengan biaya enam
belas \omterm{def:b1:water-small-molecules:atp}{ATP} dan delapan elektron tiap molekulnya, dan hanya di tempat
oksigennya disingkirkan. Legum (kacang polong, buncis, semanggi)
memondokkan bakteri semacam itu (\emph{Rhizobium}) dalam \emph{bintil
\omterm{def:b1:flowering-plant-organization:organs}{akar}}: tumbuhannya membangun sebuah bilik, memberi bakterinya gula,
lalu menjaga oksigennya rendah dengan sebuah \omterm{def:b1:proteins:peptide}{protein} merah,
leghemoglobin, yang mengantarkannya ke respirasi bakterinya pada
kepekatan yang terlalu rendah untuk merusak \omterm{def:b1:enzymes:enzyme}{enzimnya}; bakterinya
memberikan kembali amonium, yang diubah tumbuhannya menjadi \omterm{def:b1:proteins:aminoacid}{asam
amino}. Sebuah ladang semanggi memancangkan seratus kilogram nitrogen
tiap hektar setahun, yang untuk itu tumbuhannya membayar sepersepuluh
fotosintesisnya. Daur nitrogen sebagai keseluruhan menjadi bagian
jilid Tahun ke-2.
\end{definition}

\includegraphics[width=0.52\linewidth]{images/book3/ai/b3-23-root-nodules.png}

{\small Bintil pada \omterm{def:b1:flowering-plant-organization:organs}{akar} sebatang legum: masing-masing adalah sebuah
bilik \omterm{def:b1:body-plans-tissues:tissue}{jaringan} tumbuhan yang terisi bakteri, merah muda oleh

\end{omfigure}

\begin{example}[Harga nitrogen]\label{ex:b1:plant-water-minerals:priceN}
Sebatang jagung memerlukan sekitar \qty{30}{mg} nitrogen sehari; dari
nitrat pada \qty{2}{mmol/L} ia memperolehnya dalam seliter air yang
ditranspirasikannya, lalu membelanjakan beberapa ratus milimol
elektron untuk mereduksinya. Sebatang semanggi yang memancangkan
jumlah yang sama membayar \qty{16}{} \omterm{def:b1:water-small-molecules:atp}{ATP} tiap $\mathrm{N_2}$ ditambah
pereduksinya --- kira-kira \qty{6}{g} gula tiap gram nitrogen, sekitar
\qty{0.2}{g} sehari, sepersepuluh hasilnya --- dan dapat tumbuh pada
tanah yang sama sekali tanpa nitrat. Petani telah menggilir legum
dengan serealia selama dua ribu tahun untuk memindahkan nitrogen itu
dari satu ladang ke ladang berikutnya.
\end{example}

\section{Sekutu dalam tanah}

\begin{definition}[Mikoriza]\label{def:b1:plant-water-minerals:mycorrhiza}
\index{mikoriza}
Yang disebut \emph{mikoriza} adalah persekutuan antara sebuah \omterm{def:b1:flowering-plant-organization:organs}{akar} dan
sebuah jamur, yang ditemukan pada sembilan dari sepuluh tumbuhan.
Dalam bentuknya yang paling lazim hifa jamurnya tumbuh ke dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}
korteksnya lalu bercabang di dalamnya menjadi \emph{arbuskul}, yaitu
\omterm{prop:b1:organism-environment:surfaces}{permukaan pertukarannya}, sementara di luar \omterm{def:b1:flowering-plant-organization:organs}{akarnya} sebuah \omterm{def:b1:body-plans-tissues:tissue}{jaringan}
hifa yang selebar beberapa mikrometer menjelajahi tanahnya sejauh
beberapa meter. Jamurnya mengantarkan fosfat dan ion lain yang buruk
bergeraknya, serta air, yang tak sanggup dicapai \omterm{def:b1:flowering-plant-organization:organs}{rambut akarnya}
sendiri; tumbuhannya mengantarkan gula, sepersepuluh fotosintesisnya
atau lebih. Pohon hutan iklim sedang membawa sebuah bentuk kedua, yang
di dalamnya jamurnya menyarungi ujung \omterm{def:b1:flowering-plant-organization:organs}{akarnya} lalu menyulam di antara
\omterm{def:b1:cell-unit-of-life:cell}{selnya}; tubuh buahnya adalah cendawan di lantai hutan.
\end{definition}
\begin{proof}[Bukti]
Semai yang ditumbuhkan dalam tanah yang disterilkan tumbuh buruk lalu
menyerap sedikit fosfat; ketika diinokulasi dengan spora jamurnya,
semainya tumbuh beberapa kali lebih besar. Fosfat radioaktif yang
ditaruh dalam tanah di luar jangkauan \omterm{def:b1:flowering-plant-organization:organs}{akarnya} muncul dalam tumbuhannya
hanya bila hifanya menghubungkan keduanya, dan karbon bertanda yang
diberikan kepada \omterm{def:b1:flowering-plant-organization:organs}{daunnya} muncul dalam hifanya dalam sehari. Memotong
hifanya dengan sebuah kasa halus yang membiarkan zat terlarut tetapi
bukan jamurnya lewat meniadakan keuntungannya.
\end{proof}

\begin{omfigure}
\includegraphics[width=0.52\linewidth]{images/book3/ai/b3-23-mycorrhiza.png}

{\small Sebuah \omterm{def:b1:flowering-plant-organization:organs}{akar} yang diwarnai untuk memperlihatkan jamur
\omterm{def:b1:plant-water-minerals:mycorrhiza}{mikorizanya}: hifa yang berjalan di antara \omterm{def:b1:cell-unit-of-life:cell}{selnya} lalu bercabang
menjadi arbuskul di dalamnya --- sebuah permukaan penyerap kedua, yang
dipasok tumbuhannya dengan gula.}
\end{omfigure}

\begin{example}[Mengapa fosfat memerlukan sebuah jamur]\label{ex:b1:plant-water-minerals:phosphate}
Fosfat mengikat pada butiran tanahnya lalu berdifusi semilimeter dalam
sehari; sebuah \omterm{def:b1:flowering-plant-organization:organs}{rambut akar} menyerap seluruh fosfat dalam jangkauannya
dalam beberapa jam lalu menunggu. Hifa yang sepersepuluh setebal
sebuah \omterm{def:b1:flowering-plant-organization:organs}{rambut akar} menjangkau sepuluh kali lebih jauh tiap gram
\omterm{def:b1:body-plans-tissues:tissue}{jaringannya}, dan semeter hifa memakan biaya seperseribu dari apa yang
dimakan semeter \omterm{def:b1:flowering-plant-organization:organs}{akar}. Bagi nitrat, yang bergerak bebas bersama air
tanahnya, \omterm{def:b1:flowering-plant-organization:organs}{rambut akarnya} sendiri mencukupi, dan tumbuhan pada tanah
yang kaya nitrat mengurangi jamurnya; bagi fosfat, jamurnya adalah
perpanjangan \omterm{def:b1:flowering-plant-organization:organs}{akarnya}.
\end{example}

\section{Soal: Tangga Potensial Air Sebuah Akar Jagung}

\begin{problem}[{Soal akhir pekan --- air dan nitrat dari sebuah tanah
bulan Juli ke dalam sebatang jagung: potensial pada tiap langkahnya,
fluks lewat akarnya, nitrogen yang dibawanya dan biaya mereduksinya,
berakhir pada fluks air harian ke dalam akarnya}]
\label{pb:b1:plant-water-minerals:1}
Sebatang jagung pada bulan Juli: permukaan \omterm{def:b1:flowering-plant-organization:organs}{akar} halus \qty{0.2}{m^2},
konduktans hidraulik \omterm{def:b1:flowering-plant-organization:organs}{akarnya} \qty{2e-7}{m.s^{-1}.MPa^{-1}}; luas \omterm{def:b1:flowering-plant-organization:organs}{daun}
\qty{0.5}{m^2}. \omterm{def:b1:membranes-transport:osmosis}{Potensial air} tanahnya \qty{-0.2}{MPa}; \omterm{def:b1:cell-unit-of-life:cell}{sel} korteks
\omterm{def:b1:flowering-plant-organization:organs}{akarnya} dengan $\Psi_s = \qty{-0.9}{MPa}$ dan $\Psi_p = \qty{0.5}{MPa}$;
\omterm{def:b1:flowering-plant-organization:tissues}{xilem} \omterm{def:b1:flowering-plant-organization:organs}{akarnya} \qty{-0.6}{MPa}; \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{daunnya} \qty{-1.3}{MPa}; udara pada
kelembapan nisbi \qty{50}{\%} ($RT/V_w = \qty{137}{MPa}$). Nitrat
tanahnya \qty{2}{mmol/L}; tumbuhannya memerlukan \qty{30}{mg} nitrogen
sehari. Reduksi nitrat menjadi amonium memakai 8 elektron tiap
nitrogen; fotosintesisnya memindahkan 4 elektron tiap $\mathrm{CO_2}$
dan \omterm{def:b1:flowering-plant-organization:organs}{daunnya} memancangkan \qty{4}{\micro mol} $\mathrm{CO_2}$ tiap meter
persegi tiap sekon selama
\qty{12}{h}.

\medskip
\noindent\textbf{Bagian I --- Tangganya.}
\begin{enumerate}
  \item Hitunglah \omterm{def:b1:membranes-transport:osmosis}{potensial air} \omterm{def:b1:cell-unit-of-life:cell}{sel} korteks \omterm{def:b1:flowering-plant-organization:organs}{akarnya}.
  \item Tuliskan urutan tanah, korteks, \omterm{def:b1:flowering-plant-organization:tissues}{xilem}, \omterm{def:b1:flowering-plant-organization:organs}{daun}, lalu sahkan bahwa
        airnya mengalir ke dalam dan ke atas pada tiap langkahnya.
  \item Hitunglah \omterm{def:b1:membranes-transport:osmosis}{potensial air} udaranya.
  \item Hitunglah turunnya dari tanah ke \omterm{def:b1:flowering-plant-organization:organs}{daun} dan turunnya dari \omterm{def:b1:flowering-plant-organization:organs}{daun} ke
        udara; berapa fraksi totalnya yang berupa langkah terakhirnya?
  \item Tanahnya mengering menjadi \qty{-0.9}{MPa}. Apa yang terjadi
        pada aliran ke dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} korteksnya, dan pada \omterm{def:b1:eukaryotic-cell:plantcell}{turgornya}?
  \item Tanahnya mengering menjadi \qty{-1.5}{MPa}. Apa yang terjadi
        pada \omterm{def:b1:flowering-plant-organization:organs}{daunnya}, dan mengapa penyiraman memulihkannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Fluksnya.}
\begin{enumerate}[resume]
  \item Hitunglah fluks air ke dalam \omterm{def:b1:flowering-plant-organization:organs}{akarnya} (konduktans $\times$
        permukaan $\times$ beda antara tanah dan \omterm{def:b1:flowering-plant-organization:tissues}{xilem} \omterm{def:b1:flowering-plant-organization:organs}{akarnya}).
  \item Ubahlah menjadi liter tiap hari.
  \item Seluruhnya hilang lewat transpirasi. Hitunglah transpirasinya
        tiap meter persegi \omterm{def:b1:flowering-plant-organization:organs}{daun} tiap jam, dalam gram.
  \item Nyatakan transpirasinya tiap meter persegi \omterm{def:b1:flowering-plant-organization:organs}{daun} tiap sekon
        dalam milimol air (\qty{18}{g/mol}).
  \item \omterm{def:b1:flowering-plant-organization:tissues}{Stomatanya} menutup pada tengah hari dan $\Psi$ \omterm{def:b1:flowering-plant-organization:organs}{daunnya} naik
        menjadi \qty{-0.9}{MPa} sedangkan potensial \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} naik
        menjadi \qty{-0.4}{MPa}. Hitunglah kembali fluksnya. Apa yang telah
        diperbuat tumbuhannya, dan mengapa?
  \item Mengapa \omterm{def:b1:flowering-plant-organization:organs}{akarnya} menyerap air tanpa membelanjakan energi,
        sedangkan ia harus membelanjakan energi untuk menyerap nitrat?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Nitrogen.}
\begin{enumerate}[resume]
  \item Hitunglah nitrat yang terbawa ke dalam \omterm{def:b1:flowering-plant-organization:organs}{akarnya} oleh fluks air
        pada pertanyaan 8, dalam milimol dan dalam miligram nitrogen.
  \item Apakah itu menutup kebutuhan tumbuhannya? Apa yang terjadi
        pada sisanya?
  \item \omterm{def:b1:flowering-plant-organization:organs}{Akarnya} menimbun nitrat menjadi \qty{5}{mmol/L} dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya}.
        Hitunglah \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} memindahkan satu mol dari
        \qty{2}{mmol/L} di luar ke \qty{5}{mmol/L} di dalam lawan
        sebuah \omterm{def:b1:membranes-transport:potential}{potensial membran} \qty{-150}{mV} ($RT = \qty{2.48}{kJ/mol}$;
        sebuah anion yang memasuki sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} yang negatif).
  \item Serapannya adalah sebuah simpor dengan dua proton. Hitunglah
        \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} yang dilepaskan dua proton yang memasuki sebuah
        \omterm{def:b1:cell-unit-of-life:cell}{sel} pada \qty{-150}{mV} dari pH 5,5 ke pH 7,5, lalu periksalah
        apakah itu mencukupi.
  \item Hitunglah elektron yang dibutuhkan tiap hari untuk mereduksi
        \qty{30}{mg} nitrogennya menjadi amonium.
  \item Hitunglah elektron yang dipindahkan \omterm{def:b1:flowering-plant-organization:organs}{daunnya} lewat \omterm{def:b1:photosynthesis:zscheme}{fotosistemnya}
        tiap hari, dan fraksi yang dialihkan ke nitratnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pilihan lain.}
\begin{enumerate}[resume]
  \item Bila tanahnya menyuguhkan amonium alih-alih nitrat, apa yang
        akan dihemat tumbuhannya, dan masalah apa yang akan ditimbulkan
        serapan amoniumnya bagi neraca pH-nya?
  \item Sebatang semanggi memancangkan \qty{30}{mg} nitrogen sehari
        yang sama. Dengan \qty{16}{} \omterm{def:b1:water-small-molecules:atp}{ATP} dan \qty{8}{} elektron tiap
        $\mathrm{N_2}$, hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibelanjakan nitrogenasenya
        tiap hari.
  \item Pada \qty{6}{g} gula tiap gram nitrogen yang terpancang secara
        keseluruhan, hitunglah biaya harian semangginya dalam gula lalu
        bandingkanlah dengan hasil sebesar \qty{3}{g} gula tiap hari.
        Apakah tawarannya berharga dalam sebuah tanah bernitrat
        \qty{2}{mmol/L}? Dalam sebuah tanah tanpa nitrat?
  \item Fosfat berada pada \qty{2}{\micro mol/L} dalam larutan tanahnya.
        Hitunglah fosfat yang dibawa masuk fluks air pada pertanyaan 8,
        lalu bandingkanlah dengan kebutuhan sebesar \qty{4}{mg}
        fosforus sehari (\qty{31}{g/mol}). Pada apa tumbuhannya harus
        bersandar?
  \item Jelaskan dalam dua kalimat bagaimana \omterm{def:b1:plant-water-minerals:pathways}{pita Kaspari} membiarkan
        \omterm{def:b1:flowering-plant-organization:organs}{akarnya} menjaga nitrat pada \qty{5}{mmol/L} dalam \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}
        sedangkan tanahnya berada pada \qty{2}{}.
  \item Sebuah mutan tak memiliki \omterm{def:b1:plant-water-minerals:pathways}{pita Kaspari}. Ramalkan susunan getah
        \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} dan perilakunya dalam tanah yang asin.
  \item Nyatakan hasilnya: fluks air ke dalam \omterm{def:b1:flowering-plant-organization:organs}{akar} jagungnya tiap hari,
        nitrogen yang diantarkannya, dan fraksi elektron \omterm{def:b1:flowering-plant-organization:organs}{daunnya} yang
        dibelanjakan untuk mereduksinya.
\end{enumerate}
\end{problem}
