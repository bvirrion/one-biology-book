\chapter{Pencernaan dan Penyerapan}\label{ch:b1:digestion-absorption}

Seekor sapi memakan sepuluh kilogram rumput sehari lalu hidup dari
asam asetat; seekor kuda memakan rumput yang sama lalu hidup,
sebagiannya, dari glukosa; seekor kelinci memakan kotoran malamnya
sendiri dan akan sakit tanpanya. Tak satu pun di antaranya dapat
mencerna \omterm{def:b1:carbohydrates:polysaccharide}{selulosa}, dan semuanya melakukannya, dengan memelihara
bakteri yang dapat. Seorang manusia, yang tak dapat, mencerna
sehidangan roti, daging dan mentega menjadi gula, \omterm{def:b1:proteins:aminoacid}{asam amino} dan \omterm{def:b1:lipids:fattyacid}{asam
lemak} dalam beberapa jam lalu menyerap sembilan liter air sehari lewat
sebuah dinding setebal satu \omterm{def:b1:cell-unit-of-life:cell}{sel}. Bab ini memerikan \omterm{prop:b1:digestion-absorption:tract}{saluran pencernaan}
seekor mamalia, \omterm{def:b1:enzymes:enzyme}{enzim} dan getah yang mengurai makanannya, \omterm{def:b1:cell-unit-of-life:cell}{sel} yang
memasukkan kepingannya, isyarat yang mengatur waktunya, serta
persekutuan dengan mikroba yang lewatnya herbivora hidup dari apa yang
tak dapat disentuh \omterm{def:b1:enzymes:enzyme}{enzim} mamalia mana pun.

\section{Nutrisi heterotrof}

\begin{definition}[Nutrisi, pencernaan, penyerapan]\label{def:b1:digestion-absorption:nutrition}
\index{pencernaan}\index{penyerapan}
Sebuah \omterm{def:b1:organism-environment:trophy}{heterotrof} harus mengambil dari makanannya energi dan karbon,
\omterm{def:b1:proteins:aminoacid}{asam amino} dan \omterm{def:b1:lipids:fattyacid}{asam lemak} yang tak dapat dibuatnya
(\cref{ch:b1:biosyntheses-integration}), vitamin, mineral dan air.
Makanannya tiba sebagai makromolekul --- \omterm{def:b1:carbohydrates:polysaccharide}{pati}, \omterm{def:b1:proteins:peptide}{protein}, \omterm{def:b1:lipids:triglyceride}{lemak} --- yang
tak dapat melintasi sebuah membran. \emph{Pencernaan} adalah
\omterm{prop:b1:water-small-molecules:families}{hidrolisisnya}, oleh \omterm{def:b1:enzymes:enzyme}{enzim}, menjadi monomer yang cukup kecil untuk
diserap: glukosa, \omterm{def:b1:proteins:aminoacid}{asam amino}, \omterm{def:b1:lipids:fattyacid}{asam lemak} dan gliserol, \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}.
\emph{Penyerapan} adalah pengangkutan monomer itu melintasi epitelium
ususnya ke dalam darah atau limfanya. Pada hewan pencernaannya bersifat
\emph{ekstrasel}, dalam lumen sebuah tabung yang terbuka pada kedua
ujungnya, yang membiarkan makanannya diolah bertahap seraya bergerak,
dan membiarkan kawasan tabungnya mengkhusus.
\end{definition}

\begin{proposition}[Saluran pencernaan seekor mamalia]\label{prop:b1:digestion-absorption:tract}
\index{saluran pencernaan}
Dari mulut ke anus tabung seekor mamalia terbagi menjadi kawasan, tiap
kawasan dengan epitelium, getah dan ototnya sendiri: \emph{mulut},
tempat gigi menggilas dan liur membasahi lalu mulai mencerna \omterm{def:b1:carbohydrates:polysaccharide}{patinya};
\emph{kerongkongan}, sebuah saluran; \emph{lambung}, sebuah kantung
asam tempat \omterm{def:b1:proteins:peptide}{protein} dibuka lipatannya lalu dipotong, dan makanannya
ditahan lalu dilepaskan sebagai sebuah pasta (kimus); \emph{usus
halus} --- duodenum, jejunum, ileum, enam meter pada seorang manusia
--- tempat getah \emph{hati} (empedu) dan \emph{pankreas} (\omterm{def:b1:enzymes:enzyme}{enzim} dan
bikarbonat) merampungkan \omterm{def:b1:digestion-absorption:nutrition}{pencernaannya} dan tempat hampir seluruh
\omterm{def:b1:digestion-absorption:nutrition}{penyerapannya} berlangsung; \emph{usus besar}, tempat air dipungut
kembali dan sebuah masyarakat mikroba yang padat memfermentasi apa
yang tersisa; serta rektum. Otot dalam dindingnya
(\cref{ch:b1:body-plans-tissues}) mencampur isinya lalu mendorongnya
lewat \emph{peristalsis}; sfingter menahannya pada tiap tahap.
\end{proposition}

\begin{example}[Kimia sebuah hidangan]\label{ex:b1:digestion-absorption:meal}
Sepotong roti dengan mentega dan keju: \omterm{def:b1:carbohydrates:polysaccharide}{patinya} diserang amilase liur
dalam mulutnya, dihentikan asam lambungnya, dilanjutkan amilase
pankreas dalam duodenumnya lalu dirampungkan maltase tepi sikat pada
\omterm{def:b1:digestion-absorption:surface}{enterositnya} sendiri; \omterm{def:b1:proteins:peptide}{protein} kejunya dibuka lipatannya oleh asamnya,
dipotong pepsin, dipotong lebih lanjut oleh tripsin dan kimotripsin,
lalu dirampungkan menjadi \omterm{def:b1:proteins:aminoacid}{asam amino} dan dipeptida pada tepi sikatnya;
menteganya dilelehkan, diemulsikan empedu, dibelah lipase menjadi \omterm{def:b1:lipids:fattyacid}{asam
lemak} dan monogliserida, lalu diseberangkan dalam misel ke membrannya.
Tiga jam dari mulut ke darah.
\end{example}

\begin{proposition}[Kendali getahnya]\label{prop:b1:digestion-absorption:control}
\index{gastrin}\index{sekretin}\index{kolesistokinin}
Getahnya diatur waktunya menurut makanannya. Pemandangan dan bau
sebuah hidangan, lewat saraf vagusnya, memulai aliran liur dan getah
lambungnya sebelum gigitan pertamanya. Makanan dalam lambungnya
meregangkan dindingnya dan peptidanya melepaskan hormon
\emph{gastrin}, yang merangsang getah asamnya; asam yang memasuki
duodenumnya melepaskan \emph{sekretin}, yang membuat pankreasnya
mencurahkan bikarbonat untuk menetralkannya; \omterm{def:b1:lipids:triglyceride}{lemak} dan peptida dalam
duodenumnya melepaskan \emph{kolesistokinin}, yang membuat kantung
empedunya mengerut dan pankreasnya melepaskan \omterm{def:b1:enzymes:enzyme}{enzimnya}, lalu
memperlambat pengosongan lambungnya sehingga ususnya tidak kewalahan.
Tiap getahnya dipanggil oleh \omterm{def:b1:enzymes:enzyme}{substrat} yang dikerjainya.
\end{proposition}
\begin{proof}[Bukti]
Beaumont (1833) mengamati lambung seorang lelaki yang tertinggal
dengan sebuah luka terbuka, lalu melihat getah lambungnya muncul hanya
ketika makanan menyentuh dindingnya, dan mencerna daging dalam sebuah
gelas sebagaimana dalam lambungnya. Pavlov (1890-an) memasangkan pada
anjing kantung lambung dan fistula saluran pankreasnya: pemandangan
makanan membuat lambungnya bergetah (potonglah vagusnya dan ia tidak);
asam yang ditaruh dalam duodenumnya membuat pankreasnya bergetah
bahkan sesudah tiap sarafnya dipotong, dan Bayliss serta Starling
(1902) memperlihatkan bahwa sebuah sari lapisan duodenum yang
disuntikkan ke dalam darahnya berbuat sama --- hormon yang pertama,
sekretin.
\end{proof}

\section{Memasukkan kepingannya}

\begin{definition}[Permukaan penyerapnya]\label{def:b1:digestion-absorption:surface}
\index{vilus}\index{enterosit}
Usus halusnya melipat lapisannya menjadi lipatan melingkar, lalu
menjadi \emph{vili} setinggi satu milimeter, masing-masing tertutup
selapis tunggal \emph{enterosit} yang membran apikalnya membawa
mikrovili --- \emph{tepi sikat} --- sehingga sebuah tabung seluas
\qty{0.6}{m^2} menyuguhkan sekitar \qty{200}{m^2} membran
(\cref{ch:b1:body-plans-tissues}). Di dalam tiap vilusnya berjalan
sebuah jala kapiler, yang membawa pergi gula dan \omterm{def:b1:proteins:aminoacid}{asam amino} ke hatinya
lewat vena porta, dan sebuah \emph{lakteal} pusat, yaitu pembuluh
limfa yang membawa pergi \omterm{def:b1:lipids:triglyceride}{lemaknya}. Enterositnya hidup empat hari lalu
diganti dari kripta di antara vilinya.
\end{definition}

\begin{proposition}[Jalur penyerapannya]\label{prop:b1:digestion-absorption:routes}
\index{kilomikron}
\emph{Gula}: glukosa dan galaktosa memasuki \omterm{def:b1:digestion-absorption:surface}{enterositnya} lewat
\omterm{prop:b1:membranes-transport:secondary}{simporter} natrium--glukosa (\cref{ch:b1:membranes-transport}),
fruktosa lewat sebuah pembawa, dan semuanya pergi lewat sebuah pembawa
pada sisi basalnya ke dalam darahnya. \emph{\omterm{def:b1:proteins:aminoacid}{Asam amino}} masuk lewat
beberapa \omterm{prop:b1:membranes-transport:secondary}{simporter} yang bergandeng natrium, dan peptida kecil lewat
sebuah \omterm{prop:b1:membranes-transport:secondary}{simporter} yang bergandeng proton, untuk dipotong di dalam
\omterm{def:b1:cell-unit-of-life:cell}{selnya}. \emph{\omterm{def:b1:lipids:triglyceride}{Lemak}}: \omterm{def:b1:lipids:fattyacid}{asam lemak} dan monogliserida meninggalkan
miselnya lalu berdifusi menembus membran apikalnya; di dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya}
keduanya diesterkan kembali menjadi \omterm{def:b1:lipids:triglyceride}{trigliserida}, disalut \omterm{def:b1:proteins:peptide}{protein}
menjadi \emph{kilomikron} selebar satu mikrometer, lalu dieksositosis
ke dalam lakteal, yang limfanya mencapai darahnya hanya di lehernya
--- \omterm{def:b1:lipids:triglyceride}{lemaknya} melewati hatinya. \emph{Air} mengikuti zat terlarutnya
secara \omterm{def:b1:membranes-transport:osmosis}{osmosis}: \qty{9}{L} sehari memasuki ususnya (dua diminum, tujuh
digetahkan sebagai liur, getah lambung, empedu, getah pankreas dan
getah usus), dan semuanya kecuali sepersepuluh liter diserap kembali,
sebagian besarnya dalam usus halusnya dan sisanya dalam kolonnya.
\emph{Vitamin}, \emph{besi} dan \emph{kalsium} masing-masing memiliki
pembawanya; vitamin B$_{12}$ memerlukan sebuah \omterm{def:b1:proteins:peptide}{protein} dari lambungnya
untuk dapat diserap sama sekali.
\end{proposition}

\begin{method}[Membaca sebuah percobaan penyerapan]\label{met:b1:digestion-absorption:everted}
\begin{enumerate}
  \item Balikkan sepotong usus sehingga bagian dalamnya di luar
        (sebuah \emph{kantung terbalik}), isilah dengan salin, lalu
        rendamlah dalam sebuah larutan zatnya: apa yang muncul di
        dalamnya telah melintasi epiteliumnya dari bekas lumennya.
  \item Ukurlah pemindahannya terhadap kepekatannya: menjenuh dan
        terhentikan oleh dingin atau oleh racun sintesis \omterm{def:b1:water-small-molecules:atp}{ATP} --- aktif
        atau berperantara pembawa; linear dan tak peka --- difusi.
  \item Singkirkan natrium dari rendamannya: bila pemindahan glukosa
        atau \omterm{def:b1:proteins:aminoacid}{asam aminonya} runtuh, ia bergandeng natrium. Tambahkan
        sebuah gula pesaing: bila pemindahannya turun, keduanya berbagi
        sebuah pembawa.
  \item Bandingkan potongannya: sebagian besar gula dan \omterm{def:b1:proteins:aminoacid}{asam amino}
        melintas dalam jejunumnya, garam empedu dan vitamin B$_{12}$
        hanya dalam ileumnya, air dan garam di mana-mana.
\end{enumerate}
\end{method}

\begin{example}[Anggaran airnya]\label{ex:b1:digestion-absorption:water}
Dari sembilan liter yang memasuki usus seorang manusia tiap hari, usus
halusnya menyerap delapan, kolonnya satu lagi, dan \qty{0.1}{L} pergi
dalam tinjanya. Serapan glukosa usus halusnya yang bergandeng natrium
menyeret air bersamanya --- yang menjadi sebab sebuah larutan glukosa
dan garam, yang diminum lewat mulut, memulihkan cairan seorang anak
yang terkena kolera ketika toksinnya telah mengubah kriptanya menjadi
penggetah: \omterm{prop:b1:membranes-transport:secondary}{simporternya} tak tersentuh lalu menyerap air lebih cepat
daripada toksinnya mengeluarkannya
(\cref{ch:b1:membranes-transport}).
\end{example}

\section{Hidup dari rumput}

\begin{proposition}[Pencernaan selulosa secara bersimbiosis]\label{prop:b1:digestion-absorption:cellulose}
\index{rumen}\index{ruminansia}\index{fermentasi usus belakang}
Tak ada mamalia yang membuat sebuah selulase. Herbivora mencerna
\omterm{def:b1:carbohydrates:polysaccharide}{selulosa} dengan memondokkan bakteri, protista dan jamur yang dapat,
dalam sebuah bilik \omterm{def:b1:respiration-fermentation:fermentation}{fermentasi} tempat mikrobanya memutuskan ikatan
$\beta(1{\to}4)$-nya (\cref{ch:b1:carbohydrates}) lalu memfermentasi
gulanya, tanpa oksigen, menjadi \emph{\omterm{def:b1:lipids:fattyacid}{asam lemak} atsiri} --- asetat,
propionat, butirat --- ditambah karbon dioksida dan metana. Inangnya
menyerap asamnya lalu hidup dari situ. \emph{Ruminansia} (sapi, domba,
rusa) menaruh biliknya \emph{sebelum} lambung sejatinya: \emph{rumen},
sebuah tong seratus liter pada seekor sapi, tempat makanannya
difermentasi sehari atau dua hari, dimuntahkan kembali lalu dikunyah
lagi, dan yang dari situ mikrobanya sendiri berlanjut ke abomasum dan
ususnya untuk dicerna sebagai sumber \omterm{def:b1:proteins:peptide}{protein} utama hewannya.
\emph{Pemfermentasi usus belakang} (kuda, kelinci, gajah) menaruh
biliknya \emph{sesudah} usus halusnya, dalam sebuah sekum dan kolon
yang membesar: keduanya mencerna \omterm{def:b1:carbohydrates:polysaccharide}{pati} dan \omterm{def:b1:proteins:peptide}{proteinnya} sendiri lebih
dahulu, memfermentasi seratnya sesudah itu, lalu kehilangan \omterm{def:b1:proteins:peptide}{protein}
mikrobanya dalam tinjanya --- kecuali, seperti kelincinya, keduanya
memakan kotoran malamnya yang lunak lalu melewatkannya untuk kedua
kalinya (\emph{sekotrofi}).
\end{proposition}

\begin{example}[Dua cara memakan sebuah padang rumput]\label{ex:b1:digestion-absorption:cowhorse}
Seekor sapi memfermentasi dua pertiga seratnya lalu memperoleh enam
puluh persen energinya sebagai asetat dan propionat --- yang dari situ
hatinya harus membuat tiap gram glukosa yang diperlukannya, sebab
hampir tak ada yang mencapai ususnya; rumennya juga mengubah nitrogen
ureanya, yang didaur ulang dari darahnya ke dalam liurnya, menjadi
\omterm{def:b1:proteins:peptide}{protein} mikroba, sehingga ia dapat hidup dari jerami yang lebih miskin
\omterm{def:b1:proteins:peptide}{protein} daripada yang sanggup ditanggung mamalia lain mana pun. Seekor
kuda memfermentasi kurang dari separuh seratnya tetapi mencerna \omterm{def:b1:carbohydrates:polysaccharide}{pati}
dan \omterm{def:b1:proteins:peptide}{protein} rumputnya sendiri, menyerap glukosa, lalu melewatkan
makanannya dalam sehari setengah alih-alih tiga: ia memperoleh lebih
sedikit dari tiap kilogramnya lalu memakan lebih banyak kilogram.
Sapinya menang pada padang yang miskin, kudanya ketika ia harus
berlari.
\end{example}

\begin{remark}[Panjang sebuah usus]\label{rem:b1:digestion-absorption:length}
Usus seekor karnivora empat kali panjang tubuhnya, usus seekor
herbivora sepuluh sampai dua puluh kali: makin miskin makanannya,
makin panjang tabungnya dan makin besar biliknya. Seekor berudu yang
memakan alga memiliki usus panjang yang menggulung yang memendek
mendadak pada metamorfosisnya ketika kataknya beralih ke serangga;
aturan yang sama, tertulis dalam masa hidup satu hewan.
\end{remark}

\end{exercise}

\section{Soal: Sapi dan Kuda}

\begin{problem}[{Soal akhir pekan --- sepuluh kilogram rumput melewati
sebuah rumen dan sebuah sekum: serat yang difermentasi, asam yang
diserap, protein mikroba yang diperoleh atau hilang, metana yang
terbuang, berakhir pada fraksi energi seekor sapi yang datang dari asam
lemak atsiri}]
\label{pb:b1:digestion-absorption:1}
Seekor sapi dan seekor kuda masing-masing memakan \qty{10}{kg} bahan
kering rumput sehari yang memuat, menurut massa, \qty{50}{\%} \omterm{def:b1:carbohydrates:polysaccharide}{selulosa}
dan hemiselulosa (serat), \qty{20}{\%} \omterm{def:b1:carbohydrates:polysaccharide}{pati} dan gula, \qty{15}{\%}
\omterm{def:b1:proteins:peptide}{protein} serta \qty{15}{\%} lignin dan abu yang tak tercerna. Energi:
\omterm{def:b1:carbohydrates:monosaccharide}{karbohidrat} dan \omterm{def:b1:proteins:peptide}{protein} \qty{17.5}{kJ/g}. \omterm{def:b1:respiration-fermentation:fermentation}{Fermentasi} \qty{1}{g}
\omterm{def:b1:carbohydrates:monosaccharide}{karbohidrat} menghasilkan \omterm{def:b1:lipids:fattyacid}{asam lemak} atsiri yang memuat \qty{75}{\%}
energinya, metana yang memuat \qty{8}{\%}, serta panas dan pertumbuhan
mikroba selebihnya; ia juga menghasilkan \qty{0.15}{g} \omterm{def:b1:proteins:peptide}{protein} mikroba.
Rumennya memfermentasi \qty{70}{\%} seratnya dan seluruh \omterm{def:b1:carbohydrates:polysaccharide}{pati} dan
gulanya; usus belakang kudanya memfermentasi \qty{45}{\%} seratnya,
sedangkan usus halusnya mencerna \qty{90}{\%} \omterm{def:b1:carbohydrates:polysaccharide}{patinya} dan \qty{70}{\%}
\omterm{def:b1:proteins:peptide}{proteinnya} sebelum seratnya mencapai usus belakangnya. Kedua hewannya
memerlukan \qty{110}{MJ} energi termetabolis sehari.

\medskip
\noindent\textbf{Bagian I --- Rumen sapinya.}
\begin{enumerate}
  \item Hitunglah massa serat, \omterm{def:b1:carbohydrates:polysaccharide}{pati} dan gula, serta \omterm{def:b1:proteins:peptide}{protein} yang
        dimakan tiap hari.
  \item Hitunglah massa \omterm{def:b1:carbohydrates:monosaccharide}{karbohidrat} yang difermentasi dalam rumennya.
  \item Hitunglah energi \omterm{def:b1:carbohydrates:monosaccharide}{karbohidrat} itu dan energi yang dipungut
        sebagai \omterm{def:b1:lipids:fattyacid}{asam lemak} atsiri.
  \item Hitunglah energi yang hilang sebagai metana, serta massa
        metananya (\qty{55}{kJ/g}) dan volumenya (\qty{1.4}{L/g}).
  \item Hitunglah \omterm{def:b1:proteins:peptide}{protein} mikroba yang dihasilkan.
  \item Jelaskan mengapa pasokan \omterm{def:b1:proteins:peptide}{protein} sapinya sendiri sebagian
        besarnya adalah mikrobanya, dan di mana mereka dicerna.
  \item Hitunglah fraksi seratnya yang meninggalkan rumennya tanpa
        difermentasi, dan nasibnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Neraca sapinya.}
Sapinya juga mencerna \qty{60}{\%} \omterm{def:b1:proteins:peptide}{protein} pakan yang lolos dari
rumennya (\qty{40}{\%} \omterm{def:b1:proteins:peptide}{protein} yang dimakan lolos) dan seluruh \omterm{def:b1:proteins:peptide}{protein}
mikrobanya, pada \qty{17.5}{kJ/g}.
\begin{enumerate}[resume]
  \item Hitunglah energi yang diperoleh sapinya dari \omterm{def:b1:proteins:peptide}{protein} (yang
        lolos dari pakan ditambah yang mikroba).
  \item Hitunglah total energi termetabolis sapinya: \omterm{def:b1:lipids:fattyacid}{asam lemak} atsiri
        ditambah \omterm{def:b1:proteins:peptide}{protein}.
  \item Hitunglah fraksi energi itu yang dipasok sebagai \omterm{def:b1:lipids:fattyacid}{asam lemak}
        atsiri.
  \item Apakah sapinya memenuhi kebutuhannya sebesar \qty{110}{MJ}?
        Apa yang dilakukannya bila tidak?
  \item Hampir tak ada glukosa yang diserap sapinya. Sebutkan asam
        yang dari situ hatinya membuat glukosa, dan prosesnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Usus belakang kudanya.}
\begin{enumerate}[resume]
  \item Hitunglah energi yang diperoleh kudanya dari \omterm{def:b1:carbohydrates:polysaccharide}{pati} yang dicerna
        dalam usus halusnya (diserap sebagai glukosa, \qty{17.5}{kJ/g}).
  \item Hitunglah energi dari \omterm{def:b1:proteins:peptide}{protein} pakan yang dicerna dalam usus
        halusnya.
  \item Hitunglah serat yang difermentasi dalam usus belakangnya dan
        energi yang dipungut sebagai \omterm{def:b1:lipids:fattyacid}{asam lemak} atsiri.
  \item Hitunglah \omterm{def:b1:proteins:peptide}{protein} mikroba yang dihasilkan dalam usus
        belakangnya. Apa yang terjadi padanya?
  \item Hitunglah total energi termetabolis kudanya dan fraksi yang
        dipasok sebagai \omterm{def:b1:lipids:fattyacid}{asam lemak} atsiri.
  \item Apakah kudanya memenuhi \qty{110}{MJ}-nya? Berapa banyak
        rumput yang perlu dimakannya, dan bagaimana ia mengaturnya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Membandingkan rancangannya.}
\begin{enumerate}[resume]
  \item Hitunglah energi yang disarikan tiap hewannya tiap kilogram
        bahan kering rumputnya.
  \item Metana sapinya adalah \qty{8}{\%} energi yang difermentasi.
        Hitunglah bagiannya dari asupan energi kasar sapinya, dan
        massa metana yang dibuang sekawanan seratus sapi dalam
        setahun.
  \item Makanan sapinya tinggal \qty{70}{h} dalam salurannya, makanan
        kudanya \qty{36}{h}. Hitunglah massa bahan kering yang dipegang
        masing-masing dalam ususnya pada tiap saat, pada asupan tetap.
  \item Seekor kelinci memfermentasi serat dalam sekumnya lalu memakan
        kotoran sekumnya. Berapa fraksi \omterm{def:b1:proteins:peptide}{protein} mikroba pada
        pertanyaan 16 yang dapat dipungut seekor kuda bila ia berbuat
        sama? Mengapa tak ada kuda yang melakukannya?
  \item Jelaskan mengapa sapinya dapat hidup dari pakan jerami
        berprotein \qty{4}{\%} ditambah urea, sedangkan kudanya tidak.
  \item Seekor kuda yang berlari cepat membakar glukosa dari
        \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya} sendiri; seekor sapi tak dapat berlari cepat.
        Kaitkanlah hal ini dengan kedua rancangannya.
  \item Nyatakan hasilnya: fraksi energi termetabolis sapinya yang
        datang dari \omterm{def:b1:lipids:fattyacid}{asam lemak} atsiri, fraksi kudanya, dan energi yang
        disarikan masing-masing tiap kilogram rumputnya.
\end{enumerate}
\end{problem}
