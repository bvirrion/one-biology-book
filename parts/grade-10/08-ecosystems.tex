\chapter{Ecosystems}\label{ch:g10:ecosystems}

\omterm{def:g10:scientific-biology:organism}{Organisms} never live alone for long: they form \omterm{def:g10:ecosystems:levels}{populations}, meet in
communities, and exchange energy and matter with the non-living world.
This chapter builds the vocabulary of ecology, traces energy through \omterm{def:g10:ecosystems:food-web}{food
webs}, and introduces the carbon and nitrogen cycles that link life to air,
soil and water.

\section{Ecological organisation}

\begin{definition}[Population, community, ecosystem]\label{def:g10:ecosystems:levels}
A \emph{population}\index{population} is all the individuals of one
species in a defined area. A \emph{community}\index{community} is all the
populations of different species living and interacting there. An
\emph{ecosystem}\index{ecosystem} is the community plus the abiotic
environment (energy, climate, water, nutrients, soil or water chemistry).
\end{definition}

\begin{definition}[Habitat and niche]\label{def:g10:ecosystems:habitat-niche}
A \emph{habitat}\index{habitat} is the place where an \omterm{def:g10:scientific-biology:organism}{organism} lives. A
\emph{niche}\index{niche} is its ecological role: resources used,
interactions and conditions tolerated --- ``address'' versus
``profession''.
\end{definition}

\begin{example}[Pond]\label{ex:g10:ecosystems:pond}
All water lilies in a pond form a \omterm{def:g10:ecosystems:levels}{population}; lilies, fish, snails and
bacteria together form the \omterm{def:g10:ecosystems:levels}{community}; adding light, temperature, dissolved
oxygen and nutrients makes the pond \omterm{def:g10:ecosystems:levels}{ecosystem}.
\end{example}

\section{Energy flow}

\begin{definition}[Producers and consumers]\label{def:g10:ecosystems:trophic-roles}
\emph{Producers}\index{producer} (autotrophs) fix energy into organic
matter --- mainly by \omterm{def:g10:photosynthesis:def}{photosynthesis}. \emph{Consumers}\index{consumer}
(heterotrophs) eat living or recently dead organic matter:
\emph{primary consumers}\index{primary consumer} (herbivores),
\emph{secondary}\index{secondary consumer} and higher-level consumers
(carnivores, omnivores). \emph{Decomposers}\index{decomposer} (many
fungi and bacteria) break down dead material, recycling nutrients.
\end{definition}

\begin{definition}[Food chain and food web]\label{def:g10:ecosystems:food-web}
A \emph{food chain}\index{food chain} is a single pathway of who-eats-whom.
A \emph{food web}\index{food web} is the network of interconnected chains
in a \omterm{def:g10:ecosystems:levels}{community} --- the more realistic picture.
\end{definition}

\begin{definition}[Trophic level]\label{def:g10:ecosystems:trophic}
A \emph{trophic level}\index{trophic level} is a feeding step:
\omterm{def:g10:ecosystems:trophic-roles}{producers} at level 1, \omterm{def:g10:ecosystems:trophic-roles}{primary consumers} at level 2, and so on. Energy
enters mainly as sunlight and leaves as heat; matter cycles.
\end{definition}

\begin{proposition}[Energy pyramids]\label{prop:g10:ecosystems:pyramid}
Only a fraction of energy at one \omterm{def:g10:ecosystems:trophic}{trophic level} becomes biomass at the
next (often illustrated as $\sim 10\%$, varying widely). Higher levels
therefore support less biomass: pyramids of energy are upright. This
limits food-chain length.
\end{proposition}
\admitted

\begin{definition}[Ecological pyramids]\label{def:g10:ecosystems:eco-pyramids}
An \emph{ecological pyramid}\index{ecological pyramid} displays quantities
by \omterm{def:g10:ecosystems:trophic}{trophic level}. A \emph{pyramid of energy}\index{pyramid of energy} is
always upright (energy is lost as heat). A \emph{pyramid of
biomass}\index{pyramid of biomass} is usually upright on land but can invert
temporarily in some aquatic systems (small phytoplankton supporting larger
\omterm{def:g10:ecosystems:trophic-roles}{consumer} biomass at an instant). A \emph{pyramid of
numbers}\index{pyramid of numbers} counts individuals and may invert when
one large \omterm{def:g10:ecosystems:trophic-roles}{producer} supports many small herbivores.
\end{definition}

\begin{method}[Analysing a food web]\label{met:g10:ecosystems:web}
\begin{enumerate}
  \item Identify \omterm{def:g10:ecosystems:trophic-roles}{producers} (usually start from the sun's entry).
  \item Trace arrows from prey to predator (energy direction).
  \item Assign \omterm{def:g10:ecosystems:trophic}{trophic levels} carefully: omnivores occupy more than one.
  \item Predict knock-on effects if one species is removed (who loses a
        food source? who is released from predation?).
\end{enumerate}
\end{method}

\section{Matter cycles: carbon}

\begin{definition}[Carbon cycle (overview)]\label{def:g10:ecosystems:carbon}
Carbon moves among atmosphere, \omterm{def:g10:scientific-biology:organism}{organisms}, soils, oceans and rocks.
Key biological fluxes: \omterm{def:g10:photosynthesis:def}{photosynthesis} removes atmospheric
$\mathrm{CO_2}$; respiration and decomposition return it; combustion of
biomass and fossil fuels adds $\mathrm{CO_2}$ rapidly.
\end{definition}

\section{Matter cycles: nitrogen}

\begin{definition}[Nitrogen cycle (overview)]\label{def:g10:ecosystems:nitrogen}
Most \omterm{def:g10:scientific-biology:organism}{organisms} cannot use atmospheric $\mathrm{N_2}$ gas directly.
\emph{Nitrogen-fixing}\index{nitrogen fixation} microbes convert
$\mathrm{N_2}$ to ammonia compounds plants can assimilate (some in
root nodules of legumes). \omterm{def:g10:ecosystems:trophic-roles}{Consumers} obtain nitrogen in organic form.
\emph{Nitrifying}\index{nitrification} bacteria convert ammonia to
nitrites/nitrates; \emph{denitrifying}\index{denitrification} bacteria
return $\mathrm{N_2}$ to the air. Decomposition releases nitrogen from
dead matter.
\end{definition}

\begin{proposition}[Energy flows, matter cycles]\label{prop:g10:ecosystems:flow-cycle}
\omterm{def:g10:ecosystems:levels}{Ecosystems} are open to energy (sunlight in, heat out) but recycle
chemical elements. Removing or overloading a nutrient (fertiliser runoff,
excess $\mathrm{CO_2}$) rewires both cycles and communities.
\end{proposition}
\admitted

\begin{example}[Eutrophication sketch]\label{ex:g10:ecosystems:eutro}
Extra nitrogen and phosphorus from fertilisers can trigger algal blooms;
when algae die, \omterm{def:g10:ecosystems:trophic-roles}{decomposers} consume oxygen and fish may suffocate --- a
nutrient cycle problem with a food-web ending.
\end{example}

\section{Keystone species and succession}

\begin{definition}[Keystone species]\label{def:g10:ecosystems:keystone}
A \emph{keystone species}\index{keystone species} has a disproportionately
large effect on \omterm{def:g10:ecosystems:levels}{community} structure relative to its abundance. Removing it
can collapse diversity or rewrite the \omterm{def:g10:ecosystems:food-web}{food web}.
\end{definition}

\begin{example}[Sea otters and kelp]
\label{ex:g10:ecosystems:otter}
In some Pacific kelp forests, sea otters prey on sea urchins that graze kelp.
When otters decline, urchins can overgraze kelp forests into ``urchin
barrens,'' reducing \omterm{def:g10:ecosystems:habitat-niche}{habitat} for many species. The otter is not the most
numerous animal, but its trophic role is keystone-like.
\end{example}

\begin{definition}[Ecological succession]
\label{def:g10:ecosystems:succession}
\emph{Ecological succession}\index{ecological succession} is the directional
change in \omterm{def:g10:ecosystems:levels}{community} composition over time after a disturbance or on newly
exposed \omterm{def:g10:membranes-enzymes:enzyme}{substrate}. \emph{Primary succession}\index{primary succession} begins
on bare rock or new land (no soil); \emph{secondary succession}\index{secondary
succession} follows disturbance that leaves soil (e.g.\ abandoned field, fire).
\end{definition}

\begin{proposition}[Disturbance resets the clock]
\label{prop:g10:ecosystems:disturbance}
Fire, flood, logging or disease can return a \omterm{def:g10:ecosystems:levels}{community} to an earlier
successional stage. Many \omterm{def:g10:ecosystems:levels}{ecosystems} are mosaics of patches at different
stages; diversity often depends on that mosaic, not on a single climax
snapshot.
\end{proposition}
\admitted

\section{Human impact (brief)}

\begin{remark}[Pressures]\label{rem:g10:ecosystems:humans}
\omterm{def:g10:ecosystems:habitat-niche}{Habitat} loss, overharvesting, invasive species, pollution and climate
change driven by greenhouse gases are major pressures on \omterm{def:g10:ecosystems:levels}{ecosystems}.
Understanding energy flow and nutrient cycles is the starting point for
conservation and sustainable use --- themes returned to in later years.
\end{remark}

\section{Problem: Mapping a local ecosystem}

\begin{problem}[{Weekend problem --- field sketch to systems thinking:
build a food web, an energy argument and a nutrient story for one real
site you can visit or recall}]
\label{pb:g10:ecosystems:1}
Choose a real site: schoolyard, park pond, stretch of beach, farm margin
or balcony pots.

\medskip
\noindent\textbf{Part I --- Inventory.}
\begin{enumerate}
  \item List at least eight species or identifiable groups (include a
        \omterm{def:g10:ecosystems:trophic-roles}{decomposer} category if species are unknown).
  \item Classify each as \omterm{def:g10:ecosystems:trophic-roles}{producer}, \omterm{def:g10:ecosystems:trophic-roles}{consumer} (rough level) or \omterm{def:g10:ecosystems:trophic-roles}{decomposer}.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Web and energy.}
\begin{enumerate}
  \item Draw a \omterm{def:g10:ecosystems:food-web}{food web} with at least six arrows.
  \item Mark the most likely longest chain; estimate how many trophic
        transfers it contains.
  \item Explain why the total mass of predators should be less than that
        of their prey over the long term (energy argument).
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Cycles and change.}
\begin{enumerate}
  \item Describe one human influence on your site's carbon or nitrogen
        budget (e.g.\ mowing, fertiliser, traffic $\mathrm{CO_2}$,
        feeding birds).
  \item Predict one \omterm{def:g10:ecosystems:levels}{community} change if \omterm{def:g10:ecosystems:trophic-roles}{decomposers} were wiped out for a
        season.
  \item Propose one practical action that would protect \omterm{def:g10:biodiversity-classification:biodiversity}{biodiversity} at
        the site and justify it ecologically.
\end{enumerate}
\end{problem}
