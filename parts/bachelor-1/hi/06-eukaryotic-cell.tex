\chapter{सुकेंद्रकी कोशिका का कार्यात्मक संगठन}\label{ch:b1:eukaryotic-cell}

अग्न्याशय की एक \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} हर दिन लगभग अपने ही द्रव्यमान जितने पाचक एंजाइम
बनाती है और उन्हें किसी वाहिनी में उँड़ेल देती है। ये एंजाइम ऐसे प्रोटीन
हैं जो उसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को पचा डालें जिसने उन्हें बनाया; इसलिए \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} उन्हें
झिल्ली की एक सील की हुई चादर के भीतर बनाती है, उन्हें एक दूसरे कक्ष से
होकर भेजती है जहाँ वे छाँटे और गाढ़े किए जाते हैं, उन्हें कणिकाओं में
संचित करती है, और उन्हें केवल किसी भोजन के संकेत पर छोड़ती है। हर चरण
किसी अलग झिल्ली-बद्ध स्थान में होता है, और स्थानों के बीच का आवागमन
पुटिकाएँ ढोती हैं। यह अध्याय \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकी कोशिका} के कक्षों, उस \omterm{def:b1:eukaryotic-cell:endomembrane}{अंतःझिल्ली
तंत्र} का जिसमें से होकर प्रोटीन बहते हैं, ऊर्जा रूपांतरित करते
\omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांगों}, उस कंकाल का जो \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की आकृति थामता है और उसके भागों को
चलाता है, तथा उन लक्षणों का वर्णन करता है जो किसी पादप \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को किसी
जंतु \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} से अलग करते हैं।

\section{कक्ष}

\begin{definition}[कोशिकांग, कोशिकाद्रव]\label{def:b1:eukaryotic-cell:organelle}
\emph{कोशिकांग}\index{कोशिकांग} किसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के भीतर की वह संरचना है
जिसका संघटन और कार्य निश्चित हो; झिल्ली-बद्ध \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग} ऐसे कक्ष हैं
जिनका भीतरी भाग संघटन में उस \emph{कोशिकाद्रव}\index{कोशिकाद्रव} से
भिन्न है जो उनके बाहर \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाद्रव्य} का जलीय विलयन है। झिल्ली-बद्ध कक्ष:
\omterm{def:b1:eukaryotic-cell:nucleus}{केंद्रक}, \omterm{def:b1:eukaryotic-cell:endomembrane}{अंतर्द्रव्यी जालिका}, \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी उपकरण}, \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकाय}, \omterm{def:b1:eukaryotic-cell:peroxisome}{परऑक्सीकाय},
\omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाएँ}, और पौधों में \omterm{def:b1:eukaryotic-cell:plastid}{लवक} तथा \omterm{def:b1:eukaryotic-cell:plantcell}{रसधानी}। बिना झिल्ली के \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग}:
राइबोसोम, \omterm{def:b1:eukaryotic-cell:cytoskeleton}{तारककाय}, \omterm{def:b1:eukaryotic-cell:cytoskeleton}{कोशिका-कंकाल}।
\end{definition}

\begin{definition}[केंद्रक]\label{def:b1:eukaryotic-cell:nucleus}
\emph{केंद्रक}\index{केंद्रक} गुणसूत्रों को
\emph{क्रोमैटिन}\index{क्रोमैटिन} के रूप में रखता है, यानी हिस्टोन
प्रोटीनों पर लिपटे DNA के रूप में (\cref{ch:b1:genomes})। उसकी सीमा
\emph{केंद्रक आवरण}\index{केंद्रक आवरण} बनाता है, यानी दो झिल्लियाँ,
जिनमें से बाहरी \omterm{def:b1:eukaryotic-cell:endomembrane}{अंतर्द्रव्यी जालिका} से जुड़ी है, और जो \emph{केंद्रक
छिद्रों}\index{केंद्रक छिद्र} से बिंधी हैं, जिनसे होकर RNA और प्रोटीन
नियंत्रण में दोनों दिशाओं में जाते हैं। \emph{केंद्रिका}\index{केंद्रिका}
वह क्षेत्र है जहाँ राइबोसोमी RNA बनता है और राइबोसोम जुड़ते हैं। केंद्रक
अनुलेखन (भीतर) को अनुवादन (बाहर) से अलग करता है, और यह अलगाव
\omterm{def:b1:cell-unit-of-life:prokeuk}{प्राक्केंद्रकियों} के पास नहीं है।
\end{definition}

\begin{proposition}[किसी यकृत कोशिका के कक्ष]\label{prop:b1:eukaryotic-cell:budget}
\qty{5000}{\micro m^3} के किसी यकृत-कोशिका में \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} लगभग आधा आयतन
लेता है, \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाएँ} पाँचवाँ भाग, \omterm{def:b1:eukaryotic-cell:endomembrane}{अंतर्द्रव्यी जालिका} छठा भाग, \omterm{def:b1:eukaryotic-cell:nucleus}{केंद्रक}
सोलहवाँ भाग, और \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी}, \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकाय} तथा \omterm{def:b1:eukaryotic-cell:peroxisome}{परऑक्सीकाय} मिलकर कुछ प्रतिशत।
झिल्लियाँ कुछ और ही कहती हैं: \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की \qty{110000}{\micro m^2}
झिल्ली में से आधी \omterm{def:b1:eukaryotic-cell:endomembrane}{अंतर्द्रव्यी जालिका} है, एक-तिहाई \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाओं} की भीतरी
झिल्ली, और केवल \qty{2}{\%} प्लाज़्मा झिल्ली। \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की अधिकांश झिल्ली
उसके भीतर होती है।
\end{proposition}

\section{अंतःझिल्ली तंत्र}

\begin{definition}[अंतर्द्रव्यी जालिका, गॉल्जी उपकरण, लयनकाय]\label{def:b1:eukaryotic-cell:endomembrane}
\emph{अंतर्द्रव्यी जालिका}\index{अंतर्द्रव्यी जालिका} (ER) झिल्ली की
चादरों और नलियों का वह जाल है जो एक ही सतत स्थान, यानी अवकाशिका, को
घेरता है और जो \omterm{def:b1:eukaryotic-cell:nucleus}{केंद्रक आवरण} से जुड़ा है। उसका
\emph{खुरदरा}\index{खुरदरी अंतर्द्रव्यी जालिका} भाग उन राइबोसोमों से
जड़ा है जो अपने बनाए प्रोटीन अवकाशिका में या झिल्ली में डाल देते हैं;
उसका \emph{चिकना}\index{चिकनी अंतर्द्रव्यी जालिका} भाग लिपिड बनाता है,
कैल्सियम संचित करता है और विषहरण करता है। \emph{गॉल्जी
उपकरण}\index{गॉल्जी उपकरण} चपटी थैलियों (कुंडों) की एक चट्टी है, जिसका
ग्राही \emph{सिस} फलक ER के पास है और प्रेषक \emph{ट्रांस} फलक दूर, और
जिसमें ER से आए प्रोटीन रूपांतरित (शर्कराएँ छाँटी और जोड़ी) होते हैं,
छाँटे जाते हैं, और अपने गंतव्यों के लिए पुटिकाओं में बाँधे जाते हैं।
\emph{लयनकाय}\index{लयनकाय} अम्लीय पुटिकाएँ हैं (pH $\approx 5$) जो
जल-अपघटनी एंजाइमों से भरी हैं और बाहर से लाई गई सामग्री या पुराने पड़
चुके \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांगों} को पचाती हैं। इनके बीच और प्लाज़्मा झिल्ली के बीच चलती
पुटिकाओं के साथ मिलकर ये \emph{अंतःझिल्ली तंत्र}\index{अंतःझिल्ली तंत्र}
बनाते हैं: कक्षों का एक जुड़ा हुआ समुच्चय, जिनकी अवकाशिकाएँ सांस्थितिक
रूप से \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} का बाहर हैं।
\end{definition}

\begin{proposition}[स्रावी पथ]\label{prop:b1:eukaryotic-cell:secretory}
\index{स्रावी पथ}
स्राव के लिए, प्लाज़्मा झिल्ली के लिए, या किसी \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकाय} के लिए नियत कोई
प्रोटीन खुरदरी ER से बँधे राइबोसोमों द्वारा संश्लेषित होता है, बनते-बनते
ER की अवकाशिका में घुसता है, वहीं वलित होता है, पुटिकाओं में \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} के
सिस फलक तक जाता है, रूपांतरित होते हुए चट्टी पार करता है, और ट्रांस फलक
से अपने गंतव्य के पते वाली किसी पुटिका में निकलता है: कोई स्रावी कणिका
जो संकेत मिलने पर प्लाज़्मा झिल्ली से जुड़ जाती है
(\emph{नियमित बहिर्क्षेपण}\index{बहिर्क्षेपण}), कोई पुटिका जो लगातार
जुड़ती रहती है (\emph{सतत} बहिर्क्षेपण, जो नई झिल्ली भी पहुँचाता है), या
कोई \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकाय}। कोशिकाद्रवी और केंद्रकी प्रोटीन मुक्त राइबोसोमों पर बनते
हैं और इस तंत्र में कभी नहीं घुसते। तय करने वाली चीज़ प्रोटीन के आरंभ का
\emph{संकेत अनुक्रम}\index{संकेत अनुक्रम} है
(\cref{ch:b1:gene-expression})।
\end{proposition}
\begin{proof}[साक्ष्य]
पलाडे (1960 का दशक) ने गिनी-पिग के अग्न्याशय की फाँकों को तीन मिनट तक
रेडियोसक्रिय ऐमीनो अम्ल खिलाए (\emph{स्पंद}), फिर बिना चिह्नित वाले
(\emph{पीछा}), और अंतरालों पर इलेक्ट्रॉन सूक्ष्मचित्रों की
स्वप्रकाशचित्रिकी से रेडियोसक्रियता का स्थान निकाला। \qty{3}{min} पर
चाँदी के कण खुरदरी ER पर पड़े थे; \qty{20}{min} पर \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} पर;
\qty{40}{min} पर गाढ़ी होती रसधानियों और नई कणिकाओं पर; और \qty{2}{h} पर
वाहिनी की अवकाशिका में। \omterm{def:b1:cell-unit-of-life:fractionation}{कोशिका प्रभाजन} ने वही क्रम रासायनिक रूप से दिया।
ब्लोबेल (1970 का दशक) ने दिखाया कि झिल्लियों के बिना परखनली में बने
स्रावी प्रोटीन स्रावित रूप से बीस अवशेष लंबे होते हैं और मिलाई गई
प्रोटिएज़ से रक्षित नहीं होते, जबकि ER झिल्लियों की उपस्थिति में बनने पर
वे काटे भी जाते हैं और रक्षित भी होते हैं: वे अतिरिक्त अवशेष ही वह संकेत
हैं जो बढ़ती शृंखला को ER में भेजता है।
\end{proof}

\begin{method}[किसी अणु का कोशिका में पीछा करना]\label{met:b1:eukaryotic-cell:pulsechase}
\begin{enumerate}
  \item एक छोटा दल चिह्नित कीजिए: \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} को कुछ मिनट के लिए किसी
        रेडियोसक्रिय या प्रतिदीप्त पूर्ववर्ती के सामने रखिए (स्पंद), फिर
        उसे बिना चिह्नित पूर्ववर्ती की अधिकता से बदल दीजिए (पीछा), ताकि
        चिह्न केवल स्पंद के दौरान बने अणुओं पर रहे।
  \item अंतरालों पर नमूने लीजिए, स्थिरीकृत कीजिए, और चिह्न का स्थान
        निकालिए: काटों की स्वप्रकाशचित्रिकी से, प्रतिदीप्ति
        सूक्ष्मदर्शिकी से, या प्रभाजन और गणना से।
  \item जिस कक्ष का चिह्न सबसे पहले शिखर पर पहुँचता है वह ऊपर-प्रवाह में
        है; शिखरों का क्रम ही मार्ग है; और अंतराल ही पारगमन-काल हैं।
  \item किसी चरण को रोकिए (ठंड, ऊर्जा का कोई संदमक, कोई उत्परिवर्ती) और
        देखिए कि चिह्न कहाँ जमा होता है: वही कक्ष रोक से पहले का है।
\end{enumerate}
\end{method}

\begin{definition}[अंतर्ग्रहण]\label{def:b1:eukaryotic-cell:endocytosis}
\emph{अंतर्ग्रहण}\index{अंतर्ग्रहण} सामग्री भीतर लाता है: प्लाज़्मा
झिल्ली उसके चारों ओर भीतर की ओर धँसती है और एक पुटिका चुटककर अलग कर देती
है, जो प्रायः किसी \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकाय} से जुड़ जाती है। \emph{भक्षकाणुक्रिया} कणों को
निगलती है (कोई जीवाणु, कोई मरी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका}); \emph{पिनोसाइटोसिस} द्रव की
बूँदें भीतर लेती है; और \emph{ग्राही-माध्यमित अंतर्ग्रहण} किसी विशेष
अणु (कोलेस्टेरॉल ढोते कण, लोहा ढोता ट्रांसफ़ेरिन) को भीतर लेने से पहले
लिपे गड्ढों के ग्राहियों पर गाढ़ा करता है। बहिर्क्षेपण \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की सतह में
झिल्ली जोड़ता है और अंतर्ग्रहण उसे हटाता है; किसी स्रावी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में ये
दोनों संतुलित रहते हैं, इसलिए सतह अचर रहती है जबकि झिल्ली के पूरे-पूरे
क्षेत्र हर घंटे उसमें से चक्कर लगाते रहते हैं।
\end{definition}

\section{सूत्रकणिकाएँ और लवक}

\begin{definition}[सूत्रकणिका]\label{def:b1:eukaryotic-cell:mitochondrion}
\emph{सूत्रकणिका}\index{सूत्रकणिका} वह \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग} है जो आरपार
\qtyrange{0.5}{1}{\micro m} और लंबाई में कुछ माइक्रोमीटर का होता है, और
जिसकी सीमा दो झिल्लियाँ बनाती हैं: एक चिकनी बाहरी झिल्ली जो छोटे अणुओं
के लिए पारगम्य है, और एक भीतरी झिल्ली जो \emph{क्रिस्टा}\index{क्रिस्टा}
में मुड़ी है, आयनों के लिए अपारगम्य है और श्वसन शृंखला तथा ATP सिंथेज़
धारण करती है (\cref{ch:b1:respiration-fermentation})। भीतर
\emph{आधात्री}\index{सूत्रकणिका आधात्री} है, जिसमें क्रेब्स चक्र के
एंजाइम, जीवाणु-प्रकार के राइबोसोम और किसी छोटे वर्तुल DNA की कई प्रतियाँ
हैं। सूत्रकणिकाएँ विखंडन से बँटती हैं, \omterm{def:b1:eukaryotic-cell:cytoskeleton}{कोशिका-कंकाल} के सहारे चलती हैं,
और प्रति \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} कुछ सौ से कुछ हज़ार तक होती हैं।
\end{definition}

\begin{definition}[लवक]\label{def:b1:eukaryotic-cell:plastid}
\emph{लवक}\index{लवक} पादप \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} के दोहरी-झिल्ली वाले \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग} हैं:
\emph{हरितलवक}\index{हरितलवक}, जिनमें झिल्लियों का एक तीसरा तंत्र, यानी
\emph{थाइलेकॉइड}\index{थाइलेकॉइड}, ग्रैना में चिना हुआ और
\emph{पीठिका}\index{पीठिका} में डूबा हुआ, प्रकाश-संश्लेषण करता है
(\cref{ch:b1:photosynthesis}); मंडलवक, जो मंड संचित करते हैं; और
वर्णिलवक, जो फलों तथा पंखुड़ियों के वर्णक रखते हैं। सभी आदिलवकों से
निकलते हैं और, \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाओं} की तरह, अपना ही वर्तुल DNA तथा
जीवाणु-प्रकार के राइबोसोम रखते हैं और विखंडन से बँटते हैं।
\end{definition}

\begin{proposition}[अंतःसहजीवी उद्गम]\label{prop:b1:eukaryotic-cell:endosymbiosis}
\index{अंतःसहजीविता}
\omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाएँ} और \omterm{def:b1:eukaryotic-cell:plastid}{लवक} उन स्वतंत्र जीवाणुओं से उतरे हैं जिन्हें किसी
पूर्वज \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकी कोशिका} ने निगला और सहजीवी के रूप में रख लिया:
\omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाएँ} किसी वायुजीवी जीवाणु से, \omterm{def:b1:eukaryotic-cell:plastid}{लवक} किसी सायनोबैक्टीरियम से।
\end{proposition}
\begin{proof}[साक्ष्य]
दोनों \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांगों} की दो झिल्लियाँ हैं, जिनमें भीतरी का संघटन जीवाणु जैसा
है और बाहरी पोषी जैसी; बिना हिस्टोन का एक वर्तुल गुणसूत्र; जीवाणु के
आकार के और प्रतिजैविकों के प्रति संवेदनशील राइबोसोम; जीवाणुवत विचलनों
वाला एक आनुवंशिक कूट; \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के विभाजन से स्वतंत्र द्विअंगी विखंडन से
विभाजन; और उनके जीन, जब अनुक्रमित किए जाते हैं, उस \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के केंद्रकी
जीनों के साथ नहीं बल्कि विशेष जीवाणु वंशरेखाओं (ऐल्फ़ाप्रोटिओबैक्टीरिया,
सायनोबैक्टीरिया) के जीनों के साथ समूह बनाते हैं। उनके मूल जीनों में से
अधिकांश तब से \omterm{def:b1:eukaryotic-cell:nucleus}{केंद्रक} में चले गए हैं, और ये \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग} अब अकेले नहीं जी
सकते।
\end{proof}

\begin{definition}[परऑक्सीकाय]\label{def:b1:eukaryotic-cell:peroxisome}
\emph{परऑक्सीकाय}\index{परऑक्सीकाय} एक छोटा इकहरी-झिल्ली वाला \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग}
है जिसमें ऐसे ऑक्सीकारक एंजाइम हैं जो क्रियाधारों (वसा अम्ल, ऐल्कोहॉल)
से हाइड्रोजन ऑक्सीजन पर पहुँचाते हैं और हाइड्रोजन परॉक्साइड बनाते हैं,
तथा कैटेलेज़, जो उस परॉक्साइड को वहीं के वहीं नष्ट कर देता है। वह एक
ख़तरनाक रसायन को घेरकर रखता है।
\end{definition}

\section{कोशिका-कंकाल}

\begin{definition}[कोशिका-कंकाल]\label{def:b1:eukaryotic-cell:cytoskeleton}
\emph{कोशिका-कंकाल}\index{कोशिका-कंकाल} प्रोटीन तंतुओं का वह जाल है जो
\omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को उसकी आकृति देता है, उसके \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांगों} को थामता है, और उन्हें तथा
स्वयं \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को चलाता है। तीन प्रकार: ऐक्टिन के \emph{सूक्ष्म
तंतु}\index{सूक्ष्म तंतु} (\qty{7}{nm}, गोलिकाकार उपइकाइयों की दो ऐंठी
लड़ियाँ), जो प्लाज़्मा झिल्ली के नीचे जमा रहते हैं और \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की आकृति,
रेंगने, विभाजन की संकुचनशील वलय तथा मायोसिन के साथ पेशी-संकुचन के लिए
ज़िम्मेदार हैं; \emph{सूक्ष्म नलिकाएँ}\index{सूक्ष्म नलिका}
(ट्यूबुलिन की \qty{25}{nm} खोखली नलियाँ), जो
\emph{तारककाय}\index{तारककाय} से फैलती हैं और वे पटरियाँ हैं जिन पर
प्रेरक प्रोटीन (बाहर की ओर काइनेसिन, भीतर की ओर डाइनीन) पुटिकाएँ तथा
\omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग} ढोते हैं, और जो विभाजन का तर्कु तथा पक्ष्माभों और कशाभों का
अंतर्भाग भी हैं; और \emph{मध्यवर्ती तंतु}\index{मध्यवर्ती तंतु}
(\qty{10}{nm}, रस्सी जैसे, केराटिनों और उनसे जुड़े प्रोटीनों के), जो वे
यांत्रिक रूप से प्रतिरोधी रेशे हैं जो \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} को कड़ा करते हैं और
डेस्मोसोमों पर टिकते हैं। सूक्ष्म तंतु और सूक्ष्म नलिकाएँ अपनी उपइकाइयों
के भंडार से लगातार जुड़ती और खुलती रहती हैं।
\end{definition}

\begin{example}[किसी पुटिका को चलाना]\label{ex:b1:eukaryotic-cell:transport}
\omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} से निकलती किसी स्रावी पुटिका को काइनेसिन उठा लेता है, यानी वह
प्रेरक जो किसी \omterm{def:b1:eukaryotic-cell:cytoskeleton}{सूक्ष्म नलिका} के सहारे \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की परिधि की ओर लगभग
\qty{1}{\micro m/s} से चलता है और हर \qty{8}{nm} डग पर एक ATP ख़र्च करता
है; किसी \omterm{def:b1:body-plans-tissues:nervous}{न्यूरॉन} में वही प्रेरक पुटिकाओं को एक मीटर लंबे अक्षतंतु की पूरी
लंबाई तक ले जाता है --- यानी दस दिन की यात्रा। सतह के पास पुटिका ऐक्टिन
वल्कुट को और अंतिम माइक्रोमीटर के लिए मायोसिन को सौंप दी जाती है। विसरण
से \qty{100}{nm} की किसी पुटिका को \qty{20}{\micro m} की कोई \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} पार
करने में लगभग एक मिनट लगता, और एक मीटर चलने में वर्षों: \omterm{def:b1:cell-unit-of-life:prokeuk}{कोशिकांग} ढोए
जाते हैं, भटकने के लिए छोड़े नहीं जाते।
\end{example}

\section{पादप और जंतु कोशिकाएँ}

\begin{definition}[कोशिका भित्ति, रसधानी, जीवद्रव्यतंतु]\label{def:b1:eukaryotic-cell:plantcell}
पादप \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} साझा योजना में तीन संरचनाएँ जोड़ती है। \emph{कोशिका
भित्ति}\index{कोशिका भित्ति}, जो प्लाज़्मा झिल्ली के बाहर है, दूसरी
बहुशर्कराओं की किसी आधात्री में सेलुलोस सूक्ष्मतंतुकों की एक मिश्रित
रचना है (\cref{ch:b1:carbohydrates}); वह \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की आकृति तय करती है,
भीतर के दाब का प्रतिरोध करती है, और एक साझा मध्य पटलिका के माध्यम से
पड़ोसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} को आपस में जोड़ती है। \emph{रसधानी}\index{रसधानी}, जो
\omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के आयतन का \qty{90}{\%} तक घेरता एक इकहरी-झिल्ली वाला कक्ष है, जल,
आयन, शर्कराएँ, वर्णक और अपशिष्ट संचित करती है; उसका परासरणी दाब झिल्ली को
भित्ति से दबाता है, और यही \emph{स्फीति}\index{स्फीति} किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} को
\omterm{def:b1:flowering-plant-organization:organs}{तना} हुआ रखती है। \emph{जीवद्रव्यतंतु}\index{जीवद्रव्यतंतु} भित्तियों में
से होकर जाते वे चैनल हैं जो प्लाज़्मा झिल्ली से अस्तरित हैं और जिनसे
पड़ोसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} के \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} सतत जुड़े रहते हैं, इसलिए कोई पादप \omterm{def:b1:body-plans-tissues:tissue}{ऊतक}
अंशतः एक ही जुड़ा हुआ \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाद्रव्य} है (जीवद्रव्य पथ)।
\end{definition}

\begin{proposition}[पादप और जंतु कोशिका की तुलना]\label{prop:b1:eukaryotic-cell:compare}
दोनों में \omterm{def:b1:eukaryotic-cell:nucleus}{केंद्रक}, ER, \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी}, \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाएँ}, \omterm{def:b1:eukaryotic-cell:peroxisome}{परऑक्सीकाय}, राइबोसोम और एक
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{कोशिका-कंकाल} हैं। पादप \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में सेलुलोस की भित्ति, एक बड़ी \omterm{def:b1:eukaryotic-cell:plantcell}{रसधानी}, \omterm{def:b1:eukaryotic-cell:plastid}{लवक}
और \omterm{def:b1:eukaryotic-cell:plantcell}{जीवद्रव्यतंतु} हैं, और तारकके तथा \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकाय} नहीं (पाचन \omterm{def:b1:eukaryotic-cell:plantcell}{रसधानी} करती है);
वह न रेंगती है न निगलती है, और बीच में एक नई भित्ति बनाकर विभाजित होती
है। जंतु \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में भित्ति नहीं है (इसलिए वह आकृति बदल सकती है, रेंग
सकती है और भक्षण कर सकती है), \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकाय} हैं, तारककों सहित एक \omterm{def:b1:eukaryotic-cell:cytoskeleton}{तारककाय} है, और
वह जीवद्रव्यतंतुओं के बजाय संधियों से संपर्क करती है; वह चुटककर दो भागों
में बँटती है।
\end{proposition}

\begin{example}[स्फीति]\label{ex:b1:eukaryotic-cell:turgor}
जिस पर्ण-कोशिका की \omterm{def:b1:eukaryotic-cell:plantcell}{रसधानी} में \qty{0.3}{mol/L} विलेय हैं वह तब तक जल
खींचती है जब तक भित्ति लगभग \qty{0.7}{MPa} के दाब से, यानी सात वायुमंडल
से, वापस न दबाने लगे; तब \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} स्फीत होती है और \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} तनी हुई। जल से
वंचित होने पर \omterm{def:b1:eukaryotic-cell:plantcell}{रसधानी} आयतन खो देती है, दाब शून्य पर गिर जाता है, और
\omterm{def:b1:flowering-plant-organization:organs}{पत्ती} मुरझा जाती है --- भित्ति वही रहती है, दाब चला जाता है
(\cref{ch:b1:membranes-transport})। उसी विलयन में रखी कोई जंतु \omterm{def:b1:cell-unit-of-life:cell}{कोशिका},
जिसके पास भित्ति नहीं है, बस फट जाती।
\end{example}

\section{अभ्यास}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:eukaryotic-cell:3}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:eukaryotic-cell:4}
पादप \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की तीन ऐसी संरचनाएँ दीजिए जो जंतु \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में नहीं हैं, और दो
ऐसी जो जंतु \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में हैं और पादप \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में नहीं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:5}
स्पंद-पीछा वाले आलेख से बताइए कि चिह्न ER में, \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} में और कणिकाओं में
किस समय शिखर पर पहुँचता है। ER से कणिका तक का पारगमन-काल और वह समय
आँकिए जब तक आधा चिह्न स्रावित हो चुका हो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:6}
किसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} पर ऐसी दवा लगाई जाती है जो \omterm{def:b1:eukaryotic-cell:cytoskeleton}{सूक्ष्म नलिकाओं} को विबहुलकित कर
देती है। \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} की स्थिति पर, पुटिका-आवागमन पर, \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} विभाजन पर और
पक्ष्माभों की धड़कन पर पड़ते प्रभावों की भविष्यवाणी कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:7}
\omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाओं} के जीवाणु उद्गम के चार साक्ष्य दीजिए, और समझाइए कि फिर भी
कोई \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका} किसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के बाहर क्यों नहीं जी सकती।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:8}
ER की अवकाशिका ``सांस्थितिक रूप से \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के बाहर'' है। इस वाक्यांश को
किसी ऐसे झिल्ली प्रोटीन का ER से प्लाज़्मा झिल्ली तक पीछा करके समझाइए
जिसकी शर्करा-शृंखलाएँ अवकाशिका की ओर हैं: अंत में वे शर्कराएँ किस ओर
होती हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:eukaryotic-cell:9}
कोई स्रावी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} \qty{40}{\micro m^2} की किसी शीर्ष सतह से होकर

प्रतिदिन जुड़ी झिल्ली निकालिए और यह भी कि शीर्ष सतह कितनी बार पुनःचक्रित

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:eukaryotic-cell:10}
किसी परखनली में राइबोसोम, किसी स्रावी प्रोटीन का mRNA और ऐमीनो अम्ल हैं;
उत्पाद \qty{320}{} अवशेषों की एक शृंखला है; ER पुटिकाएँ मिलाने पर
\qty{300}{} अवशेषों की एक शृंखला पुटिकाओं के भीतर दिखती है और बाद में
मिलाई गई प्रोटिएज़ से पचती नहीं। हर प्रेक्षण की व्याख्या कीजिए, और
बताइए कि पहले बीस अवशेषों से रहित कोई उत्परिवर्ती क्या करता।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:eukaryotic-cell:11}
किसी रोगी की \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में वह लयनकायी एंजाइम नहीं है जो किसी लिपिड को
तोड़ता है; वह लिपिड फूले हुए \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकायों} में जमा होता जाता है और \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ}
मर जाती हैं। \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} जीवविज्ञान समझाइए, बताइए कि रोग उन \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में सबसे
भयंकर क्यों है जो अपनी झिल्लियाँ सबसे तेज़ी से नई करती हैं, और सुझाइए कि
रक्त में डाला गया कोई एंजाइम \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकायों} तक कैसे पहुँच सकता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:eukaryotic-cell:12}
``\omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकी कोशिका} पूर्व जीवाणुओं का वह संघ है जिसे कोई झिल्ली तंत्र
जोड़े रखता है।'' एक अनुच्छेद में चर्चा कीजिए, और तौलिए कि अंतःसहजीविता
सिद्धांत क्या समझाता है और क्या नहीं (\omterm{def:b1:eukaryotic-cell:nucleus}{केंद्रक}, ER, \omterm{def:b1:eukaryotic-cell:cytoskeleton}{कोशिका-कंकाल})।
\end{exercise}

\section{समस्या: अग्न्याशय की कोशिकागुच्छ कोशिका}

\begin{problem}[{सप्ताहांत समस्या --- वह कोशिका जो हर दिन अपने ही
द्रव्यमान जितना प्रोटीन स्रावित करती है: पलाडे की घड़ी, वह झिल्ली जो
उसे पुनःचक्रित करनी पड़ती है, वह संकेत जो प्रोटीन का पता तय करता है, और
अंत में किसी पाचक एंजाइम का पारगमन-काल}]
\label{pb:b1:eukaryotic-cell:1}
अग्न्याशय की कोई कोशिकागुच्छ \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} \qty{12}{\micro m} भुजा का एक घन है,
जिसका शीर्ष फलक (\qty{144}{\micro m^2}) वाहिनी पर है और आधार फलक रक्त
पर। \qty{80}{g} के किसी मानव अग्न्याशय में लगभग $\num{8e10}$ ऐसी
\omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} हैं और वह प्रतिदिन \qty{10}{g} एंजाइम प्रोटीन स्रावित करता है।
कोई स्रावी कणिका \qty{1}{\micro m} व्यास का एक गोला है जिसमें प्रोटीन
\qty{200}{mg/mL} पर है। राइबोसोम प्रति सेकंड \qty{5}{} ऐमीनो अम्ल जोड़ते
हैं; किसी प्रारूपिक एंजाइम में \qty{250}{} अवशेष हैं जिनका माध्य
द्रव्यमान \qty{110}{g/mol} है।

\medskip
\noindent\textbf{भाग I --- पलाडे की घड़ी।}
अध्याय का स्पंद-पीछा आलेख समय के साथ चिह्न का स्थान देता है।
\begin{enumerate}
  \item स्पंद छोटा और पीछे में बिना चिह्नित ऐमीनो अम्ल की बड़ी अधिकता
        क्यों होनी चाहिए?
  \item वह समय पढ़िए जिस पर चिह्न ER में, \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} में और कणिकाओं में
        शिखर पर पहुँचता है।
  \item ER से \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} तक और \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} से कणिका तक का पारगमन-काल आँकिए।
  \item \qty{60}{min} पर चिह्न कहाँ है, और उसका कुछ भाग \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} से पहले
        ही क्यों निकल चुका है जबकि अधिकांश अब भी कणिकाओं में है?
  \item दूसरा प्रयोग \qty{15}{\celsius} पर किया जाता है: चिह्न घंटे भर
        से अधिक ER में ही रहता है। व्याख्या कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- प्रोटीन का द्रव्यमान।}
\begin{enumerate}[resume]
  \item प्रति \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} प्रतिदिन स्रावित प्रोटीन पिकोग्राम में निकालिए।
  \item एक कणिका का आयतन और उसमें रखे प्रोटीन का द्रव्यमान निकालिए।
  \item प्रति \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} प्रतिदिन और प्रति मिनट छोड़ी गई कणिकाओं की संख्या
        निकालिए।
  \item \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} का आयतन और, \qty{1.05}{g/cm^3} घनत्व के साथ, उसका
        द्रव्यमान निकालिए। वह हर दिन अपने ही द्रव्यमान का कितना भिन्न
        स्रावित करती है?
  \item एक कणिका में एंजाइम अणुओं की संख्या निकालिए (आवोगाद्रो संख्या
        $\num{6.0e23}$)।
  \item किसी राइबोसोम को एक एंजाइम अणु बनाने में लगने वाला समय
        निकालिए।
  \item दैनिक उत्पादन बनाने के लिए कितने राइबोसोमों को लगातार काम करना
        पड़ेगा? इसकी तुलना उन $\num{4e6}$ राइबोसोमों से कीजिए जो ऐसी
        \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में होते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- झिल्ली।}
\begin{enumerate}[resume]
  \item एक कणिका का झिल्ली-क्षेत्रफल निकालिए।
  \item प्रश्न 8 की कणिकाओं के बहिर्क्षेपण से प्रतिदिन शीर्ष फलक तक
        पहुँचाई गई झिल्ली निकालिए।
  \item शीर्ष फलक का क्षेत्रफल प्रतिदिन कितनी बार जुड़ता है? \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को
        उसका क्या करना पड़ता है, और किस प्रक्रिया से?
  \item कणिका की झिल्ली \omterm{def:b1:eukaryotic-cell:endomembrane}{गॉल्जी} से आती है, जो उसे ER से पाता है। यदि
        \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की ER में \qty{20000}{\micro m^2} झिल्ली हो, तो वह कितने
        दिनों की कणिका-झिल्ली के बराबर है? इससे \omterm{def:b1:eukaryotic-cell:endomembrane}{अंतःझिल्ली तंत्र} के भीतर
        पुनःचक्रण के बारे में क्या पता चलता है?
  \item शीर्ष फलक \qty{0.1}{\micro m} व्यास और \qty{1}{\micro m} लंबाई
        के \qty{2000}{} सूक्ष्मांकुर धारण करता है। वे
        \qty{144}{\micro m^2} के फलक को कितने गुणक से बड़ा करते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- प्रोटीन पर लिखा पता।}
\begin{enumerate}[resume]
  \item किसी एंजाइम का mRNA झिल्लियों के बिना परखनली में अनूदित करने पर
        ऐसी शृंखला देता है जो कणिकाओं में मिले एंजाइम से \qty{20}{}
        अवशेष लंबी है। वे अतिरिक्त अवशेष कहाँ हैं और क्या हैं?
  \item ER पुटिकाओं की उपस्थिति में उत्पाद सही लंबाई का है, पुटिकाओं के
        भीतर है, और किसी प्रोटिएज़ से रक्षित है। व्याख्या कीजिए।
  \item वही प्रयोग किसी कोशिकाद्रवी एंजाइम के mRNA के साथ ऐसा उत्पाद
        देता है जो पुटिकाओं के बाहर और अरक्षित है। निष्कर्ष निकालिए।
  \item कोई उत्परिवर्तन उन बीस अवशेषों को हटा देता है। भविष्यवाणी
        कीजिए कि एंजाइम कहाँ पहुँचेगा और \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} का क्या होगा।
  \item कोई दूसरा एंजाइम कणिकाओं के बजाय \omterm{def:b1:eukaryotic-cell:endomembrane}{लयनकायों} में मिलता है। वह भी
        उसी प्रकार का \omterm{prop:b1:eukaryotic-cell:secretory}{संकेत अनुक्रम} धारण करता है। मार्ग के किस स्थान पर
        दूसरा निर्णय लिया जाना चाहिए, और मोटे तौर पर किस क्रियाविधि से?
  \item किसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} पर ऐसी दवा लगाई जाती है जो ATP संश्लेषण रोक देती
        है। भविष्यवाणी कीजिए कि किसी स्पंद-पीछा का चिह्न कहाँ जमा होगा
        और क्यों (दो कारण)।
  \item प्रश्न 3 और 8 मिलाइए: स्थायी अवस्था में किसी भी क्षण ER और
        कणिका के बीच कितनी कणिकाएँ ``रास्ते में'' होती हैं?
  \item परिणाम का सार दीजिए: संश्लेषण से कणिका तक का पारगमन-काल, स्राव
        तक का समय, प्रति मिनट छोड़ी गई कणिकाएँ, और \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} प्रतिदिन अपने
        ही द्रव्यमान का जो भिन्न स्रावित करती है।
\end{enumerate}
\end{problem}
