\section*{अध्याय \ref{ch:g11:vision-and-brain} --- दृष्टि और मस्तिष्क}

\begin{solution}{exo:g11:vision-and-brain:1}
बाएँ क्षेत्र से आता प्रकाश हर दृष्टिपटल के दाएँ आधे हिस्से पर पड़ता है;
बाईं \omterm{def:g11:the-eye:parts}{आँख} के दाएँ आधे वाले तंतु दृक्-संधि पर पार करते हैं, और दाईं \omterm{def:g11:the-eye:parts}{आँख} के दाईं ओर
ही रह जाते हैं; मिलकर वे दायाँ दृक् पथ बनाते हैं, चेतक में बदलते हैं और
दाएँ \omterm{prop:g11:vision-and-brain:pathway}{प्राथमिक दृष्टि वल्कुट} तक पहुँचते हैं।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:2}
दोनों दृक् तंत्रिकाएँ मिलती हैं; हर दृष्टिपटल के नासिका-पक्ष वाले आधे हिस्से
के तंतु दूसरी ओर पार करते हैं, और कनपटी-पक्ष वाले नहीं। उसके आगे हर पथ
दृष्टि-क्षेत्र का एक आधा हिस्सा ढोता है।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:3}
रंग वाला क्षेत्र (दुनिया धूसर दिखती है), हलचल वाला क्षेत्र (गति का कोई बोध
नहीं), चेहरे वाला क्षेत्र (चेहरे नहीं पहचाने जाते)।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:4}
सुनम्यता: तंत्रिका-जोड़ों का वह गुण जिससे वे इस्तेमाल के अनुसार मज़बूत या
कमज़ोर होते, बनते या छँट जाते हैं। क्रांतिक अवधि: जीवन का वह शुरुआती दौर
जिसमें अनुभव वल्कुट को निर्णायक और अपरिवर्तनीय ढंग से गढ़ता है।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:5}
वह दृष्टि क्षेत्रों के न्यूरॉनों के सेरोटोनिन ग्राहियों से बँध जाती है,
संकेतों का बहाव बिगाड़ देती है और बिना किसी संगत प्रकाश के विभ्रम पैदा
करती है।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:6}
दृक्-संधि के बाद, बाईं ओर: बायाँ दृक् पथ, बायाँ चेतक-केंद्र या बायाँ \omterm{prop:g11:vision-and-brain:pathway}{प्राथमिक
दृष्टि वल्कुट}।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:7}
पार करते तंतु दोनों दृष्टिपटलों के नासिका-पक्ष वाले आधे हिस्सों से आते हैं,
जो क्षेत्र के बाहरी (कनपटी-पक्ष वाले) आधे देखते हैं: वह रोगी बाईं \omterm{def:g11:the-eye:parts}{आँख} के
क्षेत्र का बायाँ आधा और दाईं \omterm{def:g11:the-eye:parts}{आँख} के क्षेत्र का दायाँ आधा खो देता है ---
यानी दोनों ओर सुरंग जैसी दृष्टि।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:8}
सामान्य: $12 + 20 + 36 + 20 = 88\%$। वंचित: $2 + 3 + 5 + 30 = 40\%$,
जिनमें से केवल 10\% मुख्यतः।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:9}
पट्टी बाँधना वल्कुट को कमज़ोर \omterm{def:g11:the-eye:parts}{आँख} का इनपुट इस्तेमाल करने पर मजबूर करता है,
इसलिए क्रांतिक अवधि के दौरान उसके जोड़ छँटने के बजाय बचे और मज़बूत रहते
हैं। किसी वयस्क में वह अवधि बीत चुकी है: जोड़ उस पैमाने पर फिर से नहीं सजते,
इसलिए कमज़ोर \omterm{def:g11:the-eye:parts}{आँख} अपना इलाक़ा वापस नहीं पा सकती।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:10}
उसके \omterm{def:g11:the-eye:photoreceptors}{शंकु} और उनके \omterm{def:g11:the-eye:photoreceptors}{ऑप्सिन} सामान्य हैं; जो खोया है वह वल्कुट का वह चरण है जो
उनके अनुपातों की व्याख्या करता है। रंगीन प्रकाशों पर दृष्टिपटल के जवाब की
विद्युत रिकॉर्डिंग उसमें सामान्य होगी और किसी वर्णांध व्यक्ति में असामान्य;
और उसकी कमी पूरी तथा अर्जित है, उनकी आंशिक तथा जन्मजात।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:11}
सालों के अभ्यास ने उसकी उँगलियों की सेवा करते जोड़ मज़बूत और कई गुना कर दिए
हैं, जिससे उनका वल्कुटीय इलाक़ा बड़ा हो गया है। अभ्यास के बिना वह क्षेत्र
सालों में फिर सिकुड़ जाएगा, जैसा गेंदें उछालने वाले अध्ययन दिखाते हैं।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:12}
शुरू में उसे प्रकाश, रंग और चलते धब्बे दिखेंगे पर वह आकार, चेहरे या गहराई
नहीं पहचान पाएगी: दृष्टिपटल काम करता है, पर वल्कुट ने क्रांतिक अवधि के
दौरान वे जोड़ बनाए ही नहीं जो उसके संकेतों की व्याख्या करते हैं। वह दृष्टि
का इस्तेमाल आंशिक रूप से, धीरे-धीरे सीख लेगी, पर कभी उस सहजता के साथ नहीं
जो जन्म से देखने वाले के पास होती है।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:13}
गर्तिका पर हर \omterm{def:g11:the-eye:photoreceptors}{शंकु} की अपनी गुच्छिका \omterm{def:g10:cells-common-unit:cell}{कोशिका} होती है, इसलिए क्षेत्र का केंद्र
प्रति अंश कहीं अधिक तंतु भेजता है, जबकि परिधि पर सौ \omterm{def:g11:the-eye:photoreceptors}{शलाकाएँ} एक ही साझा करती
हैं; वल्कुट क्षेत्रफल तंतुओं के अनुपात में बाँटता है, यानी ब्योरे के अनुपात
में, न कि क्षेत्र के अंशों के।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:14}
उस क्षेत्र की सक्रियता ही रंग का बोध है, चाहे संकेत \omterm{def:g11:the-eye:parts}{आँख} से आए या स्मृति से;
बोध वही है जो ये क्षेत्र करते हैं, न कि वह जो \omterm{def:g11:the-eye:parts}{आँख} पहुँचाती है। और विभ्रम
वही सक्रियता है, जो इरादे या प्रकाश के बजाय किसी दवा से शुरू हुई हो।
\end{solution}

\begin{solution}{exo:g11:vision-and-brain:15}
\omterm{def:g11:the-eye:parts}{आँख} प्रकाश को संकेतों में बदलती है; देखने का काम मस्तिष्क करता है।
सही-सलामत \omterm{def:g11:the-eye:parts}{आँखों} वाला रोगी वल्कुट की किसी क्षति से रंग, गति या चेहरे खो सकता
है; सही-सलामत \omterm{def:g11:the-eye:parts}{आँखों} वाला बिल्ली का बच्चा उनमें से एक से व्यावहारिक रूप से
अंधा हो जाता है यदि वल्कुट को उसे जोड़ने न दिया जाए; और जो दवा न \omterm{def:g11:the-eye:parts}{आँख} को
छूती है न वल्कुट की रचना को, वह किसी व्यक्ति को वह दिखा देती है जो वहाँ है
ही नहीं। \omterm{def:g11:the-eye:parts}{आँख} पहुँचाती है; मस्तिष्क देखता है।
\end{solution}

\begin{solution}{pb:g11:vision-and-brain:1}
\textbf{1.} बाईं \omterm{def:g11:the-eye:parts}{आँख} या उसकी दृक् तंत्रिका, दृक्-संधि से पहले: केवल एक ही \omterm{def:g11:the-eye:parts}{आँख} के
तंतु प्रभावित हैं।

\textbf{2.} दायाँ दृक् पथ, चेतक या \omterm{prop:g11:vision-and-brain:pathway}{दृष्टि वल्कुट}: दोनों \omterm{def:g11:the-eye:parts}{आँखों} में उसी एक
आधे क्षेत्र की साझा कमी केवल वहीं हो सकती है जहाँ उस आधे के लिए दोनों \omterm{def:g11:the-eye:parts}{आँखों}
के तंतु एक साथ इकट्ठे कर दिए गए हों, यानी दृक्-संधि के बाद।

\textbf{3.} दृक्-संधि ही: पार करते तंतु, जो दोनों दृष्टिपटलों के नासिका-पक्ष वाले
आधे हिस्सों से आते हैं और क्षेत्र के कनपटी-पक्ष वाले आधे देखते हैं, कटे हैं।

\textbf{4.} हलचल वाला क्षेत्र, प्राथमिक वल्कुट के आगे। सही-सलामत क्षेत्र और
तीक्ष्णता दिखाते हैं कि प्राथमिक वल्कुट और उससे पहले का सब कुछ काम कर रहा है।

\textbf{5.} B, C और D के दृष्टिपटल सही-सलामत हैं; A का हो भी सकता है और नहीं
भी। जाँच के लिए नेत्रदर्शी से दृष्टिपटल देखिए और कौंधों पर उसका विद्युत
जवाब रिकॉर्ड कीजिए।

\textbf{6.} दायाँ वल्कुट क्षेत्र का बायाँ आधा हिस्सा पाता है।

\textbf{7.} केंद्रीय $2^\circ$: \qty{1}{cm}; $2^\circ$ चौड़ी वह पट्टी जो
$40^\circ$ पर हो: \qty{0.1}{cm}। अनुपात 10।

\textbf{8.} गर्तिका अपने \omterm{def:g11:the-eye:photoreceptors}{शंकु} सघन भरती है, और हर एक की अपनी गुच्छिका \omterm{def:g10:cells-common-unit:cell}{कोशिका}
होती है, इसलिए केंद्रीय अंश परिधि से कहीं अधिक तंतु भेजते हैं; वल्कुट उन्हें
उसी अनुपात में अधिक क्षेत्रफल देता है।

\textbf{9.} बाएँ क्षेत्र के केंद्र में एक छोटा अंध बिंदु, जो पढ़ने के लिए
अपंग कर देने वाला है; और आगे की ओर हो तो बाएँ क्षेत्र की परिधि में एक बड़ा
अंधा हिस्सा, जो कम खटकता है।

\textbf{10.} \omterm{def:g11:the-eye:parts}{आँखें} हिलाकर वह रोगी हर शब्द दृष्टिपटल के उस हिस्से पर ले आता
है जिसका वल्कुट सही-सलामत है, यानी क्षति के किनारे का इस्तेमाल करके; धीरे,
पर संभव।

\textbf{11.} $20 + 36 + 20 = 76\%$।

\textbf{12.} $2 + 3 + 5 + 30 = 40\%$, और केवल 10\% के लिए बाईं \omterm{def:g11:the-eye:parts}{आँख} मुख्य या
बराबर का इनपुट है।

\textbf{13.} उसका इनपुट वल्कुट तक पहुँचता है पर इस्तेमाल न होने से उसके जोड़
कमज़ोर पड़कर छँट गए; और दाईं \omterm{def:g11:the-eye:parts}{आँख} के जोड़ उस जगह में फैल गए।

\textbf{14.} यह कि वह पुनर्संयोजन जीवन के शुरुआती हिस्से की एक क्रांतिक अवधि
तक सीमित है; मनुष्यों में उसका समकक्ष मंददृष्टि है, जो केवल बचपन में पनपती
है और उसके बाद लाइलाज है।

\textbf{15.} किसी \omterm{def:g11:the-eye:parts}{आँख} को तरजीह न मिलने पर कोई क़ब्ज़ा नहीं होता: आयतचित्र
लगभग सममित रहता है पर कुल मिलाकर जवाब देते न्यूरॉन कम हो जाते हैं, और उस
बच्चे की दृष्टि दोनों \omterm{def:g11:the-eye:parts}{आँखों} में ख़राब रहती है --- यानी प्रतिस्पर्धा के बिना
वंचन दोनों को कमज़ोर करता है।

\textbf{16.} दोहरी दृष्टि से बचने के लिए मस्तिष्क मुड़ी \omterm{def:g11:the-eye:parts}{आँख} का प्रतिबिंब दबा
देता है; बिना इस्तेमाल के उसके जोड़ क्रांतिक अवधि के दौरान छँट जाते हैं और
वह \omterm{def:g11:the-eye:parts}{आँख} हमेशा के लिए कमज़ोर हो जाती है। चश्मा प्रकाशिकी सुधारता है, पर उस
\omterm{def:g11:the-eye:parts}{आँख} का इस्तेमाल नहीं होता, इसलिए छँटाई चलती रहती है।

\textbf{17.} अच्छी \omterm{def:g11:the-eye:parts}{आँख} ढँकने से वल्कुट कमज़ोर \omterm{def:g11:the-eye:parts}{आँख} के संकेत इस्तेमाल करने पर
मजबूर होता है, जिससे उसके जोड़ बचे और मज़बूत होते हैं।

\textbf{18.} क्रांतिक अवधि लगभग सात पर ख़त्म होती है; उसके बाद वल्कुट उस
पैमाने पर इलाक़ा दोबारा नहीं बाँटता, इसलिए कमज़ोर \omterm{def:g11:the-eye:parts}{आँख} के खोए जोड़ फिर से
नहीं बनाए जा सकते।

\textbf{19.} वयस्क में जोड़ पहले ही जम और ठहर चुके हैं; दूसरी \omterm{def:g11:the-eye:parts}{आँख} का क़ब्ज़ा
विकसित होते वल्कुट की परिघटना है, क्रांतिक अवधि के दौरान की, न कि परिपक्व
वल्कुट की।

\textbf{20.} A: \omterm{def:g11:the-eye:parts}{आँख} या दृक् तंत्रिका; C: दृक्-संधि; B: दायाँ पथ, चेतक या प्राथमिक
वल्कुट; D: उसके आगे का हलचल वाला क्षेत्र। वल्कुट की सुनम्यता, जो एक क्रांतिक
अवधि तक सीमित है, उस बच्चे की पट्टी को चार पर चलने देती है और पंद्रह पर चूका
देती है।
\end{solution}
