\chapter{O oxigênio na água}\label{ch:g7:oxygen-in-the-water}

O riacho de montanha e a lagoa de verão são os dois ``água'' --- e,
mesmo assim, a truta que dispara num deles ficaria ofegante no outro,
e a carpa que prospera na lagoa acharia o riacho vazio de alimento. A
diferença entre as duas águas é, em boa parte, um número invisível:
quanto \omterm{def:g7:what-is-respiration:respiration}{oxigênio} está \omterm{def:g7:oxygen-in-the-water:dissolved}{dissolvido} nelas. Este capítulo aprende a ler,
medir e prever esse número --- e, com ele, o mapa da vida aquática.

\section{O oxigênio dissolvido, um estoque escasso}

\begin{definition}[Oxigênio dissolvido]\label{def:g7:oxygen-in-the-water:dissolved}
A água guarda \omterm{def:g7:what-is-respiration:respiration}{oxigênio} do jeito que o chá doce guarda açúcar ---
invisivelmente, \emph{dissolvido}\index{oxigênio dissolvido}. A
quantidade é minúscula: um litro de água bem arejada carrega uns
\qty{10}{mg} de \omterm{def:g7:what-is-respiration:respiration}{oxigênio}, umas trinta vezes menos que um litro de ar.
Cada \omterm{def:g7:how-animals-breathe:four}{brânquia} e cada respirador subaquático do
\cref{ch:g7:how-animals-breathe} depende desse único estoque magro
(\cref{prop:g7:what-is-respiration:habitats}).
\end{definition}

\begin{proposition}[O que enche o estoque]\label{prop:g7:oxygen-in-the-water:fills}
Dois fornecedores reabastecem o \omterm{def:g7:what-is-respiration:respiration}{oxigênio} de uma água:
\begin{enumerate}
  \item o \emph{ar de cima}: o \omterm{def:g7:what-is-respiration:respiration}{oxigênio} se dissolve na superfície ---
        mais rápido onde a água é agitada, respingada e espumada
        (corredeiras, cachoeiras, ondas, chuva);
  \item as \emph{produtoras de baixo}: de dia, as \omterm{def:g1:plants-around-us:plant}{plantas} aquáticas e
        as \omterm{def:g5:first-look-at-cells:cell}{células} verdes sozinhas liberam \omterm{def:g7:what-is-respiration:respiration}{oxigênio} na água enquanto
        fabricam (\cref{rem:g6:plants-are-producers:gas}).
\end{enumerate}
\end{proposition}
\admitted

\begin{proposition}[O que esvazia o estoque, e quanto ele comporta]\label{prop:g7:oxygen-in-the-water:drains}
O estoque é esvaziado por \emph{todo respirador} --- os \omterm{def:g1:animals-around-us:animal}{animais}, as
\omterm{def:g1:plants-around-us:plant}{plantas} à noite e, acima de tudo, os \omterm{def:g6:decomposers-and-soil:decomposers}{decompositores} quando há muita
matéria morta para comer (o oitavo pote do
\cref{pb:g7:what-is-respiration:1}). E o teto se move: a água
\emph{fria} comporta bem mais \omterm{def:g7:oxygen-in-the-water:dissolved}{oxigênio dissolvido} que a quente --- um
riacho a \qty{5}{\degreeCelsius} carrega metade a mais que uma lagoa a
\qty{25}{\degreeCelsius}. O calor é um aperto duplo: menos \omterm{def:g7:what-is-respiration:respiration}{oxigênio}
comportado, enquanto todo ronco de motor
(\cref{ex:g7:what-is-respiration:demand}) acelera.
\end{proposition}
\admitted

\begin{example}[Riacho contra lagoa]\label{ex:g7:oxygen-in-the-water:streampond}
Agora as duas águas se leem como balanços. O riacho de montanha: frio
(teto alto), batido em espuma branca em cada pedra (reabastecimento
rápido) --- \omterm{def:g7:what-is-respiration:respiration}{oxigênio} perto do teto o ano inteiro. A lagoa de verão:
quente (teto baixo), parada (reabastecimento lento), rica em lama e
matéria morta (\omterm{def:g6:decomposers-and-soil:decomposers}{decompositores} famintos) --- \omterm{def:g7:what-is-respiration:respiration}{oxigênio} raspando o chão,
pior nas primeiras horas antes do amanhecer, quando as \omterm{def:g1:plants-around-us:plant}{plantas}
passaram a noite inteira respirando sem nenhuma fabricação diurna para
compensar.
\end{example}

\begin{example}[Lendo a curva da lagoa]\label{ex:g7:oxygen-in-the-water:curve}
A onda diária da lagoa é a \cref{rem:g6:plants-are-producers:gas}
desenhada como dados: o nascer do sol inicia a fabricação das
produtoras e a curva sobe; o fim da tarde a leva ao pico; a noite
deixa todo respirador tirando do estoque sem outro fornecedor além da
superfície parada --- a curva escorrega até o mínimo do amanhecer. As
mortandades de \omterm{prop:g3:sorting-living-things:groups}{peixes} em lagoas tomadas de vegetação acontecem ao
amanhecer, e agora você sabe por quê.
\end{example}

\section{O mapa dos moradores}

\begin{proposition}[O oxigênio desenha o mapa]\label{prop:g7:oxygen-in-the-water:map}
Como as \omterm{def:g5:biodiversity:species}{espécies} diferem nas necessidades de \omterm{def:g7:what-is-respiration:respiration}{oxigênio}, o número do
\omterm{def:g7:oxygen-in-the-water:dissolved}{oxigênio dissolvido} mapeia os moradores
(\cref{prop:g6:exploring-our-environment:decide}, debaixo d'água):
\begin{itemize}
  \item as \omterm{def:g5:biodiversity:species}{espécies} \emph{exigentes} --- truta, larvas de efêmera com
        os tufos branquiais agitados
        (\cref{pb:g7:how-animals-breathe:1}) --- só vivem em água
        fria, batida e rica em \omterm{def:g7:what-is-respiration:respiration}{oxigênio};
  \item as \omterm{def:g5:biodiversity:species}{espécies} \emph{tolerantes} --- carpa, muitos caramujos de
        lagoa, larvas de cauda de rato com tubinhos até a superfície
        --- se viram em água quente, parada e mais pobre;
  \item então o censo se lê de trás para a frente: encontre larvas de
        efêmera debaixo das pedras e a água é boa; encontre só
        respiradores de tubinho e engolidores de ar, e ela é ruim.
\end{itemize}
Quem fiscaliza rios recenseia a vida pequena exatamente por isso: os
moradores são um registro dos piores dias recentes da água, que uma
única leitura de sonda pode não pegar.
\end{proposition}
\admitted

\begin{example}[A história do resíduo orgânico]\label{ex:g7:oxygen-in-the-water:waste}
O esgoto sem tratamento de um vilarejo --- rico em \omterm{def:g6:plants-are-producers:matter}{matéria orgânica} ---
chega a um riacho limpo. Abaixo do cano, os \omterm{def:g6:decomposers-and-soil:decomposers}{decompositores} se
banqueteiam e se multiplicam
(\cref{def:g6:decomposers-and-soil:decomposers}), e a \omterm{def:g7:what-is-respiration:respiration}{respiração} em
massa deles esvazia o \omterm{def:g7:what-is-respiration:respiration}{oxigênio}: a truta e as larvas de efêmera somem,
os tolerantes respiradores de tubinho tomam conta e, mais para baixo
ainda --- com o resíduo consumido e a água reoxigenada nas corredeiras
--- as \omterm{def:g5:biodiversity:species}{espécies} exigentes voltam. A poluição foi \omterm{def:g6:plants-are-producers:matter}{matéria orgânica}; o
assassino foi a fome de \omterm{def:g7:what-is-respiration:respiration}{oxigênio}; o mapa registrou tudo.
\end{example}

\begin{method}[Julgar uma água]\label{met:g7:oxygen-in-the-water:judge}
Para julgar um riacho ou uma lagoa:
\begin{enumerate}
  \item meça ou estime os fornecedores: \omterm{def:g6:exploring-our-environment:conditions}{temperatura}, agitação,
        crescimento verde;
  \item pese os ralos: lama e matéria morta, superpopulação, qualquer
        entrada de resíduo orgânico;
  \item recenseie a vida pequena debaixo das pedras e no meio das
        \omterm{def:g1:plants-around-us:plant}{plantas} --- há \omterm{def:g5:biodiversity:species}{espécies} exigentes presentes, ou só tolerantes?
  \item se amostrar numa única hora, lembre a curva da lagoa: uma boa
        leitura de tarde pode esconder um amanhecer letal.
\end{enumerate}
\end{method}

\section{Problema: O anuário da fiscal}

\begin{problem}[{Problema de fim de semana --- um rio, quatro
estações, um ano de leituras}]\label{pb:g7:oxygen-in-the-water:1}
Uma fiscal de rio mantém um anuário para quatro estações: K1, uma
corredeira fria abaixo de um açude; K2, um trecho lento e quente que
atravessa pastagens; K3, logo abaixo do cano de resíduos de um
laticínio; K4, dois quilômetros mais abaixo, depois de uma longa
corredeira de pedras. Cada estação recebe leituras de sonda e um censo
da vida pequena.

\medskip\noindent\textbf{Parte I --- Lendo as estações.}
\begin{enumerate}
  \item K1 marca \qty{11}{mg/L} a \qty{8}{\degreeCelsius}; K2 marca
        \qty{7}{mg/L} a \qty{22}{\degreeCelsius}. Atribua a diferença
        aos fornecedores e aos tetos certos.
  \item Censo de K1: alevinos de truta, larvas de efêmera. Censo de
        K2: carpas, caramujos de lagoa, larvas de cauda de rato.
        Interprete os dois com a
        \cref{prop:g7:oxygen-in-the-water:map}.
  \item K3 marca \qty{3}{mg/L}, embora a água não esteja mais quente
        que a de K2. Diga o ralo e os trabalhadores por trás dele.
  \item K4 marca \qty{9}{mg/L} e o censo dela mostra larvas de efêmera
        de novo. Explique a recuperação com os dois fornecedores e com
        o destino do resíduo.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Os ritmos do ano.}
\begin{enumerate}[resume]
  \item As leituras de amanhecer de K2 em agosto caem a
        \qty{4}{mg/L}, enquanto as leituras de tarde chegam a
        \qty{9}{mg/L}. Desenhe ou descreva a curva diária dela e diga
        os dois processos que trocam de lugar.
  \item As leituras de janeiro em K2 são mais estáveis e mais altas
        que as de agosto, embora o trecho de pastagem continue tão
        parado quanto sempre. Dê os dois motivos de inverno.
  \item Uma quinzena quente e sem vento está prevista. Ponha as quatro
        estações em ordem de risco de mortandade de \omterm{prop:g3:sorting-living-things:groups}{peixes}, com um
        motivo para cada.
  \item A fiscal marca as visitas de sonda de agosto para logo antes
        do nascer do sol. Defenda esse horário.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- Vereditos e
remédios.}
\begin{enumerate}[resume]
  \item O laticínio instala um tratamento que remove quase toda a
        \omterm{def:g6:plants-are-producers:matter}{matéria orgânica} do efluente dele. Preveja a sonda e o censo
        de K3 um ano depois, em ordem de recuperação.
  \item Um proprietário propõe derrubar as árvores da margem de K2
        ``para deixar mais \omterm{def:g6:exploring-our-environment:conditions}{luz} fabricar mais \omterm{def:g7:what-is-respiration:respiration}{oxigênio}''. Pese a
        proposta: o que a sombra também faz, segundo a
        \cref{prop:g7:oxygen-in-the-water:drains}, e o que você
        responderia?
  \item O censo encontra larvas de efêmera numa estação cuja única
        leitura de sonda da tarde parece medíocre. Em que testemunha
        você acredita sobre a estação do ano como um todo, e por quê?
  \item Feche o anuário com a regra de uma frase da fiscal para o rio
        inteiro, usando \emph{frio}, \emph{agitado} e \emph{sem carga}
        (de resíduo orgânico).
\end{enumerate}
\end{problem}
