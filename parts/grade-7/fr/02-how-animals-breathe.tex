\chapter{Comment les animaux respirent}\label{ch:g7:how-animals-breathe}

La truite sous le pont, le ver de terre sous la pelouse, la
libellule au-dessus et toi qui te tiens là faites tous la même chose
en cette minute~: prendre du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} dans votre milieu
(\cref{def:g7:what-is-respiration:respiration}). Mais vos milieux
diffèrent --- air, \omterm{def:g6:decomposers-and-soil:soil}{sol}, eau --- et l'équipement aussi. Ce chapitre
est une visite des machines respiratoires du règne \omterm{def:g1:animals-around-us:animal}{animal}.

\section{Un seul problème, quatre machines}

\begin{proposition}[Le cahier des charges commun]\label{prop:g7:how-animals-breathe:spec}
Tout organe respiratoire, quel qu'en soit le propriétaire, est bâti
selon un seul cahier des charges, déjà rencontré dans
\cref{rem:g4:breathing:sponge}~: une surface \emph{grande},
\emph{mince} et \emph{humide} où le \omterm{prop:g4:journey-of-food:absorb}{sang} de l'\omterm{def:g1:animals-around-us:animal}{animal} (ou son liquide
interne) s'approche assez de son milieu pour que le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} entre
et que le \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} sorte. Grande pour un passage
suffisant~; mince pour un passage facile~; humide parce que les gaz
traversent dissous. Les machines diffèrent par la façon dont elles
présentent cette surface au milieu.
\end{proposition}
\admitted

\begin{definition}[Les quatre machines]\label{def:g7:how-animals-breathe:four}
Les \omterm{def:g1:animals-around-us:animal}{animaux} présentent leur surface d'échange de quatre grandes
façons~:
\begin{enumerate}
  \item les \emph{poumons}\index{poumon}~: des poches remplies d'air
        repliées \emph{à l'intérieur} du corps, ventilées par des
        mouvements respiratoires --- \omterm{prop:g3:sorting-living-things:groups}{mammifères}, \omterm{prop:g3:sorting-living-things:groups}{oiseaux}, \omterm{prop:g4:classifying-animals:classes}{reptiles},
        \omterm{prop:g3:sorting-living-things:groups}{amphibiens} adultes~;
  \item les \emph{branchies}\index{branchie}~: des surfaces
        plumeuses tenues \emph{à l'extérieur ou au bord du corps},
        lavées par un courant d'eau --- \omterm{prop:g3:sorting-living-things:groups}{poissons}
        (\cref{exo:g4:breathing:11}), \omterm{ex:g2:animals-grow-up:frog}{têtards}, nombreuses larves
        aquatiques, moules~;
  \item les \emph{trachées}\index{trachée (insecte)}~: un réseau de
        fins tubes à air conduits depuis des ouvertures réparties
        sur le corps jusqu'à chaque organe --- les \omterm{prop:g3:sorting-living-things:groups}{insectes}~; l'air
        est livré à domicile, en grande partie sans l'aide du
        \omterm{prop:g4:journey-of-food:absorb}{sang}~;
  \item la \emph{respiration cutanée}\index{respiration cutanée}~:
        toute la \omterm{def:g1:the-five-senses:sense}{peau} humide comme surface d'échange --- le ver de
        terre entièrement~; les \omterm{prop:g3:sorting-living-things:groups}{amphibiens} en gros complément de
        leurs poumons rudimentaires.
\end{enumerate}
\end{definition}

\begin{example}[Voir tourner les machines]\label{ex:g7:how-animals-breathe:watch}
Chaque machine montre son rythme. La bouche et les opercules de la
truite pompent en alternance régulière --- l'eau entre par la
bouche, passe sur les \omterm{def:g7:how-animals-breathe:four}{branchies}, ressort sous les opercules.
L'abdomen de la sauterelle au repos bat --- il pompe ses tubes à
air. La gorge de la grenouille palpite --- elle pousse l'air dans
ses \omterm{def:g7:how-animals-breathe:four}{poumons} (aucune cage thoracique pour le faire). Seul le ver de
terre ne montre rien~: la \omterm{def:g7:how-animals-breathe:four}{respiration cutanée} n'a aucun mouvement,
et c'est justement pourquoi
\cref{def:g7:what-is-respiration:respiration} distinguait l'échange
du pompage.
\end{example}

\section{La machine va avec le milieu --- et avec la vie}

\begin{proposition}[Pourquoi chaque machine, et où]\label{prop:g7:how-animals-breathe:fits}
Les machines se répartissent selon les milieux et les modes de vie
(\cref{def:g4:adapted-to-their-world:adaptation} à l'œuvre sur la
\omterm{def:g7:what-is-respiration:respiration}{respiration})~:
\begin{itemize}
  \item les \omterm{def:g7:how-animals-breathe:four}{branchies} conviennent à l'eau~: leurs surfaces
        plumeuses s'y déploient, sans cesse lavées --- et
        s'affaissent inutilement à l'air
        (la truite qui étouffe de \cref{exo:g4:breathing:11})~;
  \item les \omterm{def:g7:how-animals-breathe:four}{poumons} conviennent à l'air~: repliés à l'intérieur,
        ils restent humides là où de l'eau à découvert serait
        perdue --- mais ils ne peuvent pas récolter le maigre
        \omterm{def:g7:what-is-respiration:respiration}{dioxygène} dissous de l'eau~;
  \item les \omterm{def:g7:how-animals-breathe:four}{trachées} conviennent aux petits corps~: excellentes sur
        des distances de l'ordre du millimètre, elles livrent trop
        lentement sur de longues distances --- une des raisons pour
        lesquelles les \omterm{prop:g3:sorting-living-things:groups}{insectes} restent petits~;
  \item la \omterm{def:g7:how-animals-breathe:four}{respiration cutanée} convient aux vies petites, à \omterm{def:g1:the-five-senses:sense}{peau}
        humide, dans les endroits mouillés~: toute la surface du
        ver suffit à peine à son corps mince, et l'air sec le tue
        --- souviens-toi des vers de terre échoués après la pluie.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Deux vies, deux machines chez un même animal]\label{ex:g7:how-animals-breathe:frog}
La grenouille parcourt tout le chapitre en une seule vie~: des
\omterm{def:g7:how-animals-breathe:four}{branchies} quand elle est têtard (une machine d'habitante de l'eau,
\cref{ex:g2:animals-grow-up:frog}), des \omterm{def:g7:how-animals-breathe:four}{poumons} plus une \omterm{def:g1:the-five-senses:sense}{peau} humide
à l'âge adulte --- les \omterm{def:g7:how-animals-breathe:four}{poumons} rudimentaires, la \omterm{def:g1:the-five-senses:sense}{peau} assurant une
bonne part du travail, surtout pendant l'hiver froid passé au fond
de la mare (\cref{exo:g6:life-through-seasons:11}~: cette grenouille
engourdie respire par la seule \omterm{def:g1:the-five-senses:sense}{peau}). La \omterm{def:g2:animals-grow-up:metamorphosis}{métamorphose} reconstruit
jusqu'à la machinerie respiratoire.
\end{example}

\begin{example}[Respirer de l'air en vivant dans l'eau]\label{ex:g7:how-animals-breathe:airinwater}
Les habitants de l'eau munis de \omterm{def:g7:how-animals-breathe:four}{poumons} doivent faire la navette. Le
dauphin et la baleine remontent souffler et inspirer
(\cref{ex:g6:classification-kinship:whale}~: des \omterm{def:g7:how-animals-breathe:four}{poumons} de
mammifère, hérités d'ancêtres terrestres)~; la limnée grimpe à la
surface pour refaire le plein~; le dytique emporte sous ses élytres
une bulle d'air argentée --- une bouteille de plongée, renouvelée en
surface. Le \omterm{def:g2:where-animals-live:habitat}{milieu de vie} se négocie~; celui de la machine, non.
\end{example}

\begin{method}[Lire la respiration d'un animal]\label{met:g7:how-animals-breathe:read}
Devant un \omterm{def:g1:animals-around-us:animal}{animal} que tu ne connais pas~:
\begin{enumerate}
  \item où vit-il --- et peut-il en sortir~? (eau, air, \omterm{def:g6:decomposers-and-soil:soil}{sol}~;
        navetteur ou résident)~;
  \item cherche les signes de la machine~: des opercules ou des
        touffes plumeuses~; des narines et une poitrine qui bouge~;
        des rangées de minuscules ouvertures le long d'un corps
        d'insecte~; ou rien de visible, sinon une \omterm{def:g1:the-five-senses:sense}{peau} humide~;
  \item observe le rythme~: opercules qui pompent, abdomen qui bat,
        poitrine qui se soulève, remontées en surface~;
  \item nomme la machine --- et confronte-la au milieu avec
        \cref{prop:g7:how-animals-breathe:fits}.
\end{enumerate}
\end{method}

\begin{remark}[Le cahier des charges ne change jamais]\label{rem:g7:how-animals-breathe:same}
Sous la variété, examine de près n'importe laquelle des quatre
machines et \cref{prop:g7:how-animals-breathe:spec} reparaît~: les
lamelles de la \omterm{def:g7:how-animals-breathe:four}{branchie}, les poches du \omterm{def:g7:how-animals-breathe:four}{poumon}, les extrémités les
plus fines des \omterm{def:g7:how-animals-breathe:four}{trachées} et la \omterm{def:g1:the-five-senses:sense}{peau} du ver sont toutes des lieux de
rencontre grands, minces et humides entre le milieu et le corps.
Quatre machines, un seul cahier des charges --- l'\omterm{prop:g5:first-look-at-cells:principle}{unité du vivant}
(\cref{prop:g5:first-look-at-cells:principle}), écrite en
équipement respiratoire.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g7:how-animals-breathe:1}
Énonce le cahier des charges commun que respecte tout organe
respiratoire, et pourquoi chacune de ses trois propriétés compte.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:how-animals-breathe:2}
Nomme les quatre machines, chacune avec deux propriétaires.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:how-animals-breathe:3}
Décris le pompage de la truite~: le trajet de l'eau à l'entrée, à
la traversée et à la sortie.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:how-animals-breathe:4}
Comment les \omterm{prop:g3:sorting-living-things:groups}{insectes} livrent-ils le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} sans \omterm{def:g7:how-animals-breathe:four}{poumons} --- et
en grande partie sans l'aide du \omterm{prop:g4:journey-of-food:absorb}{sang}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:how-animals-breathe:5}
Pourquoi une truite échouée étouffe-t-elle dans un air riche en
\omterm{def:g7:what-is-respiration:respiration}{dioxygène}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:how-animals-breathe:6}
Pourquoi un soleil sec tue-t-il un ver de terre sur le trottoir~?

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:how-animals-breathe:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:how-animals-breathe:8}

\begin{exercise}[$\star\star$]\label{exo:g7:how-animals-breathe:9}

\begin{exercise}[$\star\star$]\label{exo:g7:how-animals-breathe:10}

\begin{exercise}[$\star\star$]\label{exo:g7:how-animals-breathe:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:how-animals-breathe:12}
«~Quatre machines, un seul cahier des charges.~» Défends la phrase~:
nomme le cahier des charges commun, et montre comment la \omterm{def:g7:how-animals-breathe:four}{branchie},

exigences.
\end{exercise}

\section{Problème~: le recensement des respirants de la mare}

\begin{problem}[{Problème du week-end --- une mare, toutes les
machines au travail}]\label{pb:g7:how-animals-breathe:1}
Un après-midi à la mare avec épuisette, boîte-loupe et carnet. Le
recensement~: des alevins de truite et des épinoches~; des \omterm{ex:g2:animals-grow-up:frog}{têtards}
(certains avec des pattes arrière)~; des grenouilles adultes~; des
limnées qui croisent sous la pellicule de surface~; un dytique qui
lance des éclats d'argent~; des larves d'éphémère à touffes
plumeuses sous les pierres~; et, dans la pelouse humide du bord, des
vers de terre.

\medskip\noindent\textbf{Partie I --- Nommer les machines.}
\begin{enumerate}
  \item Attribue à chaque \omterm{def:g1:my-body:parts}{membre} du recensement sa machine
        respiratoire.
  \item Les touffes plumeuses de la larve d'éphémère s'agitent sans
        cesse, même en eau immobile. Que sont-elles, et à quoi sert
        cette agitation~?
  \item Quels \omterm{def:g1:my-body:parts}{membres} du recensement doivent monter en surface, et
        pourquoi ceux-là exactement~?
  \item Les \omterm{ex:g2:animals-grow-up:frog}{têtards} sans pattes et ceux qui en ont peuvent différer
        autant à l'intérieur qu'à l'extérieur. Quelle
        reconstruction est en cours, d'après
        \cref{ex:g7:how-animals-breathe:frog}~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les machines à
l'épreuve.}
\begin{enumerate}[resume]
  \item Une semaine chaude et sans vent fait baisser le \omterm{def:g7:what-is-respiration:respiration}{dioxygène}
        dissous de la mare (\cref{ch:g7:oxygen-in-the-water}
        expliquera pourquoi). Classe le recensement~: qui souffre
        en premier --- les utilisateurs de \omterm{def:g7:how-animals-breathe:four}{branchies}, ou les
        navetteurs de surface~? Pourquoi~?
  \item Cette semaine-là, les épinoches se tiennent juste sous la
        surface, la bouche s'agitant à la pellicule. Interprète ce
        comportement avec le cahier des charges~: qu'a de
        particulier le millimètre supérieur de l'eau~?
  \item Un promeneur maladroit soulève des nuages de vase fine.
        Quelle machine s'encrasse, et que t'attendrais-tu à voir
        faire aux \omterm{def:g1:animals-around-us:animal}{animaux} concernés~?
  \item Après une nuit de pluie, les vers de terre de la pelouse
        sont étalés sur le chemin. Explique ce spectacle
        classique~: quelle machine, et quelle condition, les ont
        fait monter~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La carte d'ensemble.}
\begin{enumerate}[resume]
  \item Dresse le recensement en tableau~: \omterm{def:g1:animals-around-us:animal}{animal}, machine, milieu,
        navetteur ou résident. Quel cahier des charges unique
        chapeaute chaque colonne de machine, d'après
        \cref{rem:g7:how-animals-breathe:same}~?
  \item La mare gèle en janvier et sa surface se ferme. Prédis les
        sorts de l'hiver~: les grenouilles
        (\cref{exo:g6:life-through-seasons:11}), les limnées, le
        dytique, les \omterm{prop:g3:sorting-living-things:groups}{poissons} --- qui est en difficulté, qui est
        pourvu~?
  \item Une nouvelle venue arrive~: une argyronète qui tisse une
        cloche de soie et la garnit d'air de surface. Place son
        astuce à côté de celle du dytique, et nomme la machine
        qu'elle continue de servir.
  \item Conclus~: pourquoi une seule petite mare a-t-elle besoin de
        quatre machines respiratoires différentes pour être
        entièrement habitée~? Une phrase, en employant
        \emph{milieu} et \emph{\omterm{def:g7:what-is-respiration:respiration}{dioxygène}}.
\end{enumerate}
\end{problem}
