\chapter{Adaptados ao mundo deles}\label{ch:g4:adapted-to-their-world}

Ponha um urso-polar no Saara e um camelo sobre o gelo do mar, e
nenhum dos dois duraria uma estação. E, no entanto, cada um deles,
em casa, é um campeão: o urso nada em mares gelados com todo o
conforto, o camelo atravessa desertos que acabariam com um cavalo em
poucos dias. Os \omterm{def:g1:living-or-not:living}{seres vivos} se encaixam nos mundos deles com uma
exatidão que compensa estudar de perto --- característica por
característica.

\section{O que quer dizer adaptação}

\begin{definition}[Adaptação]\label{def:g4:adapted-to-their-world:adaptation}
Uma \emph{adaptação}\index{adaptação} é uma característica do corpo
ou do comportamento de um \omterm{def:g1:living-or-not:living}{ser vivo} que o encaixa nas condições do
\omterm{def:g2:caring-for-nature:environment}{ambiente} dele --- ao frio ou à seca, ao alimento dele, aos caçadores
dele. A \cref{prop:g2:where-animals-live:fit} mostrou \omterm{def:g1:animals-around-us:animal}{animais} se
encaixando nos \omterm{def:g2:where-animals-live:habitat}{hábitats} deles; a adaptação é esse encaixe, examinado
peça por peça.
\end{definition}

\begin{example}[O camelo, peça por peça]\label{ex:g4:adapted-to-their-world:camel}
Os problemas do deserto: calor, falta de água, areia fofa. As
respostas do camelo: corcovas de gordura --- uma \omterm{def:g3:animals-through-seasons:active}{reserva de alimento}
para travessias magras; a capacidade de beber cem litros em minutos
e depois passar dias sem beber; pés largos de dois dedos que não
afundam na areia; narinas que se fecham contra a poeira levantada
pelo vento e cílios compridos e duplos para combinar. Cada problema
do deserto, respondido por uma característica.
\end{example}

\begin{example}[O urso-polar, peça por peça]\label{ex:g4:adapted-to-their-world:bear}
Os problemas do gelo do mar: frio, água gelada, presas difíceis de
abordar sobre o gelo branco. As respostas do urso: pelo grosso sobre
uma camada funda de gordura --- isolamento duplo; patas largas e um
pouco palmadas --- raquetes de neve e remos ao mesmo tempo; pelagem
branca --- invisível contra a neve em que ele caça. A mesma
exatidão, ajustada ao mundo oposto.
\end{example}

\begin{proposition}[As adaptações respondem a problemas]\label{prop:g4:adapted-to-their-world:answers}
Onde quer que o \omterm{def:g2:caring-for-nature:environment}{ambiente} imponha um problema difícil, os \omterm{def:g1:living-or-not:living}{seres vivos}
que moram ali carregam características que respondem a esse problema:
\begin{itemize}
  \item frio: pelo grosso, camadas de gordura, orelhas pequenas que
        perdem pouco calor;
  \item seca: armazenamento de água, \omterm{def:g1:the-five-senses:sense}{peles} enceradas, \omterm{def:g1:plants-around-us:parts}{raízes} fundas;
  \item vida na água: corpos hidrodinâmicos, nadadeiras e pés
        palmados, pelagens que escorrem a água;
  \item caçadores por toda parte: cores de camuflagem, armaduras,
        espinhos, velocidade e \omterm{def:g1:the-five-senses:sense}{olhos} bem afastados para vigiar em
        volta;
  \item caçar para viver: sentidos aguçados, \omterm{def:g1:the-five-senses:sense}{olhos} voltados para a
        frente, garras e velocidade
        (\cref{prop:g2:what-animals-eat:match}).
\end{itemize}
\end{proposition}
\admitted

\section{As plantas também são adaptadas}

\begin{example}[O cacto]\label{ex:g4:adapted-to-their-world:cactus}
O deserto é o que mais põe as \omterm{def:g1:plants-around-us:plant}{plantas} à prova --- elas não podem
caminhar até a água. O cacto responde: um \omterm{def:g1:plants-around-us:parts}{caule} grosso em forma de
barril que armazena água e faz o trabalho de captar luz das \omterm{def:g1:plants-around-us:parts}{folhas};
\omterm{def:g1:plants-around-us:parts}{folhas} encolhidas até virar espinhos, que quase não perdem água ---
e que defendem o barril contra os \omterm{def:g1:animals-around-us:animal}{animais} sedentos; \omterm{def:g1:plants-around-us:parts}{raízes} espalhadas
largas e rasas para apanhar de uma vez uma chuva rara inteira.
\end{example}

\begin{example}[Almofadas de montanha]\label{ex:g4:adapted-to-their-world:alpine}
As \omterm{def:g1:plants-around-us:plant}{plantas} de alta montanha enfrentam um vento gelado. Muitas
crescem como almofadas apertadas, agarradas ao chão: dentro da
almofada, o vento é quebrado e o ar fica mais ameno --- uma forma que
já é abrigo. As primas de terras baixas dessas mesmas \omterm{def:g1:plants-around-us:plant}{plantas}
crescem altas e soltas; as formas da montanha abraçam a rocha.
\end{example}

\begin{method}[Ler o mundo de um organismo pelo corpo dele]\label{met:g4:adapted-to-their-world:read}
Diante de um \omterm{def:g1:animals-around-us:animal}{animal} ou de uma \omterm{def:g1:plants-around-us:plant}{planta} desconhecidos:
\begin{enumerate}
  \item liste as características marcantes dele --- pelagem, pés,
        \omterm{def:g1:the-five-senses:sense}{olhos}, formato do corpo, \omterm{def:g1:plants-around-us:parts}{folhas}, \omterm{def:g1:plants-around-us:parts}{raízes};
  \item para cada uma, pergunte: \emph{que problema isso resolveria?}
  \item junte as respostas num retrato da casa dele --- fria ou
        quente, úmida ou seca, caçador ou caçado;
  \item confira o palpite com o lugar onde o organismo vive de
        verdade.
\end{enumerate}
Este é o \cref{met:g2:where-animals-live:matching} crescido: de
encaixar um \omterm{def:g2:where-animals-live:habitat}{hábitat} a reconstruir um mundo.
\end{method}

\begin{example}[O método numa pele misteriosa]\label{ex:g4:adapted-to-their-world:mystery}
Uma gaveta de museu guarda uma \omterm{def:g1:the-five-senses:sense}{pele} sem etiqueta: pelo denso e
macio, \omterm{def:g1:the-five-senses:sense}{olhos} minúsculos, garras dianteiras enormes em forma de pá. O
pelo diz mamífero; as pás dizem cavar; os \omterm{def:g1:the-five-senses:sense}{olhos} inúteis dizem vida no
escuro. Veredito: um cavador subterrâneo --- e é mesmo uma toupeira
(\cref{ex:g2:where-animals-live:mole}), lida corretamente a partir
de três características só.
\end{example}

\begin{remark}[Como surge esse encaixe?]\label{rem:g4:adapted-to-their-world:how}
Um camelo nunca \emph{decide} criar pés largos, e urso-polar nenhum
escolhe a pelagem branca dele. Como, então, os \omterm{def:g1:living-or-not:living}{seres vivos} passam a
se encaixar tão bem nos mundos deles? Essa é uma das perguntas mais
profundas da \omterm{def:g1:living-or-not:living}{biologia}, e a resposta completa --- como o encaixe se
constrói, geração após geração --- é uma história que este livro
guarda para um ano posterior. Por enquanto, aprenda a \emph{enxergar}
o encaixe: foi o enxergar que tornou a pergunta possível.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:1}
O que é uma \omterm{def:g4:adapted-to-their-world:adaptation}{adaptação}? Dê um exemplo em \omterm{def:g1:animals-around-us:animal}{animal} e um exemplo em
\omterm{def:g1:plants-around-us:plant}{planta}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:2}
Diga três problemas do deserto e a resposta do camelo a cada um.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:3}
Diga três problemas do gelo do mar e a resposta do urso-polar a cada
um.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:4}
Que trabalhos os espinhos do cacto fazem? O que o \omterm{def:g1:plants-around-us:parts}{caule} em barril
armazena, e que trabalho esse \omterm{def:g1:plants-around-us:parts}{caule} assumiu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:5}
Por que muitas \omterm{def:g1:plants-around-us:plant}{plantas} de alta montanha crescem como almofadas
apertadas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:6}
Dê duas \omterm{def:g4:adapted-to-their-world:adaptation}{adaptações} de \omterm{def:g1:animals-around-us:animal}{animais} caçados e duas de caçadores, tiradas
da \cref{prop:g4:adapted-to-their-world:answers}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:7}
Reconte o \cref{ex:g4:adapted-to-their-world:mystery}: que três
características entregaram a toupeira, e o que cada uma disse?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:8}
Faça a experiência: escolha um \omterm{def:g1:animals-around-us:animal}{animal} qualquer que você possa
observar --- um pato, um gato, uma aranha --- e rode o
\cref{met:g4:adapted-to-their-world:read} nele: três características,
três problemas resolvidos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:9}
O raposão-do-deserto tem orelhas enormes; a raposa-do-ártico, orelhas
minúsculas. Usando a linha do frio da
\cref{prop:g4:adapted-to-their-world:answers}, explique os dois casos
--- o que orelhas grandes fazem com o calor?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:10}
Um \omterm{def:g1:animals-around-us:animal}{animal} misterioso: corpo hidrodinâmico, pés traseiros palmados,
pelo denso que escorre a água, bigodes, e ele segura bem a
\omterm{def:g4:breathing:movements}{respiração}. Reconstrua o mundo dele com o
\cref{met:g4:adapted-to-their-world:read}. (Ele constrói represas.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:adapted-to-their-world:11}
A baleia (\cref{ex:g3:sorting-living-things:surprises}) é um
mamífero com formato de peixe: hidrodinâmica, com nadadeiras, lisa. A
truta é um peixe com o mesmo formato. O que essa semelhança entre
dois nadadores \emph{sem parentesco} mostra sobre as \omterm{def:g4:adapted-to-their-world:adaptation}{adaptações} ---
quem determinou o formato, a família ou a água?
\end{exercise}
