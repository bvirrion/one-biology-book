\chapter{Ecology Basics}\label{ch:g6:ecology}

\omterm{def:g6:ecology:def}{Ecology} studies how \omterm{def:g4:ecosystems:levels}{organisms} interact with one another and with their
non-living \omterm{def:g4:traits:env}{environment}. This chapter builds core vocabulary and the
rules of \omterm{def:g2:food-energy:energy}{energy} flow and interdependence.

\section{Levels of ecology}

\begin{definition}[Ecology]\label{def:g6:ecology:def}
\emph{Ecology}\index{ecology} is the study of relationships among
\omterm{def:g4:ecosystems:levels}{organisms} and between \omterm{def:g4:ecosystems:levels}{organisms} and their \omterm{def:g4:traits:env}{environment}.
\end{definition}

\begin{definition}[Population to biosphere]\label{def:g6:ecology:levels}
A \emph{population}\index{population} is members of one \omterm{def:g6:classification:species}{species} in an
area. A \emph{community}\index{community} is interacting populations.
An \emph{ecosystem}\index{ecosystem} adds \omterm{def:g6:ecology:factors}{abiotic factors}. A
\emph{biome}\index{biome} is a large region with \omterm{def:g3:classify:char}{characteristic} climate
and life (for example tropical rainforest). The \emph{biosphere}
\index{biosphere} is all life-supporting areas of Earth.
\end{definition}

\begin{example}[Lake]\label{ex:g6:ecology:lake}
All perch in a lake: \omterm{def:g6:ecology:levels}{population}. Perch, \omterm{def:g1:plants-animals:plant}{plants}, \omterm{def:g3:insects:insect}{insects}: \omterm{def:g6:ecology:levels}{community}.
Plus \omterm{def:g4:plants-need:needs}{water} chemistry and \omterm{def:g4:plants-need:needs}{light}: \omterm{def:g6:ecology:levels}{ecosystem}.
\end{example}

\section{Abiotic and biotic factors}

\begin{definition}[Abiotic and biotic]\label{def:g6:ecology:factors}
\emph{Abiotic factors}\index{abiotic factor} are non-living (temperature,
\omterm{def:g4:plants-need:needs}{light}, \omterm{def:g4:plants-need:needs}{water}, soil, pH). \emph{Biotic factors}\index{biotic factor} are
living influences (\omterm{def:g5:food-webs:pred}{predators}, \omterm{def:g5:food-webs:pred}{prey}, competitors, \omterm{def:g4:microbes:pathogen}{pathogens}, partners).
\end{definition}

\section{Energy and feeding}

\begin{definition}[Producers to decomposers]\label{def:g6:ecology:trophic}
\emph{Producers}\index{producer} capture \omterm{def:g2:food-energy:energy}{energy} (mainly sunlight).
\emph{Consumers}\index{consumer} feed on other \omterm{def:g4:ecosystems:levels}{organisms}.
\emph{Decomposers}\index{decomposer} recycle dead matter.
\end{definition}

\omimg{food-web.jpg}{0.52\linewidth}{\omterm{def:g5:food-webs:chain-web}{Food webs} map feeding links in a
\omterm{def:g6:ecology:levels}{community}.}

\begin{proposition}[Energy flows, matter cycles]\label{prop:g6:ecology:flow}
\omterm{def:g2:food-energy:energy}{Energy} enters \omterm{def:g6:ecology:levels}{ecosystems} (usually as \omterm{def:g4:plants-need:needs}{light}) and \omterm{def:g2:plant-parts:main}{leaves} as heat after
work is done. Chemical elements such as carbon and nitrogen are reused
in cycles.
\end{proposition}
\admitted

\section{Relationships}

\begin{definition}[Competition and predation]\label{def:g6:ecology:rel}
\emph{Competition}\index{competition} occurs when \omterm{def:g4:ecosystems:levels}{organisms} need the
same limited resource. \emph{Predation}\index{predation} is one \omterm{def:g1:plants-animals:animal}{animal}
killing and eating another. \emph{Mutualism}\index{mutualism} is a
partnership that benefits both (for example bee and \omterm{def:g2:plant-parts:main}{flower} nectar).
\end{definition}

\begin{method}[Analysing an ecosystem sketch]\label{met:g6:ecology:analyse}
\begin{enumerate}
  \item List \omterm{def:g4:ecosystems:bio-abio}{biotic} and \omterm{def:g6:ecology:factors}{abiotic factors} shown.
  \item Identify \omterm{def:g6:ecology:trophic}{producers} and main \omterm{def:g6:ecology:trophic}{consumers}.
  \item Trace one \omterm{def:g5:food-webs:chain-web}{food chain} inside the web.
  \item Predict effects of removing a \omterm{def:g6:classification:species}{species} or changing an \omterm{def:g6:ecology:factors}{abiotic
        factor}.
\end{enumerate}
\end{method}

\begin{example}[Drought]\label{ex:g6:ecology:drought}
Less \omterm{def:g4:plants-need:needs}{water} (\omterm{def:g4:ecosystems:bio-abio}{abiotic}) stresses \omterm{def:g1:plants-animals:plant}{plants} (\omterm{def:g6:ecology:trophic}{producers}), then \omterm{def:g5:food-webs:roles}{herbivores} and
\omterm{def:g5:food-webs:pred}{predators} --- a cascade through the \omterm{def:g6:ecology:levels}{community}.
\end{example}
