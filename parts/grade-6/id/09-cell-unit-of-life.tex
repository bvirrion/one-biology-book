\chapter{Sel, Satuan Kehidupan}\label{ch:g6:cell-unit-of-life}

\Cref{ch:g5:first-look-at-cells} berakhir dengan sebuah janji:
kamar-kamar kecil itu tidak kosong. Tahun ini \omterm{def:g5:first-look-at-cells:microscope}{mikroskop} sekolahnya
dikeluarkan lagi --- dengan kaca preparat yang lebih baik dan
pertanyaan yang lebih tajam. Apa tepatnya yang ada di dalam sebuah
\omterm{def:g5:first-look-at-cells:cell}{sel}? Apakah \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g1:plants-around-us:plant}{tumbuhan} dan \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g1:animals-around-us:animal}{hewan} berbeda? Dan apakah sebuah \omterm{def:g5:first-look-at-cells:cell}{sel}
merupakan \emph{bagian} dari kehidupan, atau dapatkah satu \omterm{def:g5:first-look-at-cells:cell}{sel} menjadi
satu \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang utuh? Satuan terkecil \omterm{def:g1:living-or-not:living}{biologi} layak mendapat
satu bab yang saksama.

\section{Di dalam sel}

\begin{definition}[Bagian-bagian sel]\label{def:g6:cell-unit-of-life:parts}
Kalau diamati dengan benar, setiap \omterm{def:g5:first-look-at-cells:cell}{sel} memperlihatkan perlengkapan
dasar yang sama:
\begin{enumerate}
  \item \emph{membran}\index{membran}, selubung luar \omterm{def:g5:first-look-at-cells:cell}{sel} yang tipis,
        yang menahannya tetap menyatu dan mengendalikan apa yang masuk
        serta keluar;
  \item \emph{sitoplasma}\index{sitoplasma}, isian mirip agar-agar
        tempat pekerjaan \omterm{def:g5:first-look-at-cells:cell}{selnya} berlangsung;
  \item \emph{inti sel}\index{inti sel}
        (bintik gelap pada
        \cref{rem:g5:first-look-at-cells:nucleus}), pusat komando yang
        mengarahkan kehidupan \omterm{def:g5:first-look-at-cells:cell}{selnya} dan pembelahannya.
\end{enumerate}
Membran, sitoplasma, inti \omterm{def:g5:first-look-at-cells:cell}{sel}: perlengkapan tiga bagian milik hampir
setiap \omterm{def:g5:first-look-at-cells:cell}{sel} pada setiap \omterm{def:g1:plants-around-us:plant}{tumbuhan}, \omterm{def:g1:animals-around-us:animal}{hewan} dan \omterm{def:g6:decomposers-and-soil:decomposers}{jamur}.
\end{definition}

\begin{proposition}[Sel tumbuhan membawa tambahan]\label{prop:g6:cell-unit-of-life:plantextras}
\omterm{def:g5:first-look-at-cells:cell}{Sel} \omterm{def:g1:plants-around-us:plant}{tumbuhan} menambahkan perlengkapannya sendiri:
\begin{itemize}
  \item sebuah \emph{dinding sel}\index{dinding sel} yang kaku di luar
        \omterm{def:g6:cell-unit-of-life:parts}{membrannya} --- dindingnya itulah yang membuat susunan bata
        pada \omterm{def:g1:the-five-senses:sense}{kulit} bawang bombai
        (\cref{ex:g5:first-look-at-cells:classics}) dan memberi
        \omterm{def:g1:plants-around-us:plant}{tumbuhan} kekakuan berdirinya tanpa \omterm{def:g3:bones-and-muscles:skeleton}{tulang};
  \item butir hijau berisi zat penangkap \omterm{def:g6:exploring-our-environment:conditions}{cahaya} di dalam \omterm{def:g6:cell-unit-of-life:parts}{sitoplasmanya}
        --- bengkel para \omterm{def:g5:food-webs:roles}{produsen}
        (\cref{prop:g6:plants-are-producers:produce}), yang tidak ada
        pada setiap \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g1:animals-around-us:animal}{hewan};
  \item sering pula sebuah kantong besar berisi air yang tekanannya
        menjaga \omterm{def:g5:first-look-at-cells:cell}{selnya} tetap gemuk --- layu
        (\cref{ex:g1:plants-around-us:wilt}) adalah kantong ini yang
        mengendur.
\end{itemize}
\end{proposition}
\admitted

\section{Seluruh kehidupan dalam satu sel}

\begin{definition}[Makhluk hidup bersel tunggal]\label{def:g6:cell-unit-of-life:unicellular}
Suatu \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} \emph{bersel tunggal}\index{bersel tunggal} adalah
sebuah organisme lengkap yang terdiri atas persis satu \omterm{def:g5:first-look-at-cells:cell}{sel}
(\cref{rem:g5:first-look-at-cells:onecelled}). Di dalam satu \omterm{def:g6:cell-unit-of-life:parts}{membran}
ia makan, bergerak, mengindera dan berkembang biak --- dengan membelah
menjadi dua
(pembelahan pada \cref{prop:g5:first-look-at-cells:growth}, sebagai
sebuah kisah hidup yang utuh). \omterm{ex:g6:producing-our-food:bread}{Ragi} pada
\cref{ex:g6:producing-our-food:bread} adalah salah satunya; begitu
pula para perenang di setetes air kolam mana pun, dan kebanyakan
\omterm{def:g6:producing-our-food:fermentation}{mikroba} di \omterm{def:g6:decomposers-and-soil:soil}{tanah}, yogurt dan tangan.
\end{definition}

\begin{example}[Setetes air kolam, yang dikelilingi]\label{ex:g6:cell-unit-of-life:drop}
Setetes dari dinding hijau akuariumnya, di bawah lensa: sebuah \omterm{def:g5:first-look-at-cells:cell}{sel}
berbentuk sandal yang mendayung lewat dengan rambut yang berdetak,
lebih cepat daripada apa pun yang lain dalam pandangan; \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal
yang hijau --- \omterm{def:g5:food-webs:roles}{produsen} yang berenang, yang menangkap \omterm{def:g6:exploring-our-environment:conditions}{cahaya} seperti
\omterm{def:g1:plants-around-us:parts}{daun} mana pun; sebuah gumpalan yang mengalir mengelilingi makanannya
lalu menelannya bulat-bulat. Masing-masingnya sedang makan, bergerak,
menghindar --- sebuah kehidupan hewan-atau-tumbuhan yang lengkap, yang
dianggarkan ke dalam satu \omterm{def:g5:first-look-at-cells:cell}{sel}.
\end{example}

\begin{remark}[Satu sel atau banyak: dua cara untuk hidup]\label{rem:g6:cell-unit-of-life:twoways}
Organisme \omterm{def:g6:cell-unit-of-life:unicellular}{bersel tunggal} mengerjakan segalanya dengan satu \omterm{def:g5:first-look-at-cells:cell}{sel};
tubuhmu mengerjakan segalanya dengan puluhan ribu miliar \omterm{def:g5:first-look-at-cells:cell}{sel}, yang
\emph{terkhususkan} --- \omterm{def:g5:first-look-at-cells:cell}{sel} pipi untuk melapisi, \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g3:bones-and-muscles:muscle}{otot} yang
\omterm{def:g3:bones-and-muscles:muscle}{berkontraksi} (\cref{def:g3:bones-and-muscles:muscle}), \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g5:brain-in-command:system}{saraf} yang
membawa pesan pada \cref{prop:g5:brain-in-command:loop}. Banyak \omterm{def:g5:first-look-at-cells:cell}{sel}
memungkinkan pembagian kerja; satu \omterm{def:g5:first-look-at-cells:cell}{sel} berarti setiap perdagangan di
dalam satu kamar. Kedua rencananya berhasil: makhluk \omterm{def:g6:cell-unit-of-life:unicellular}{bersel tunggal}
adalah \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang paling banyak jumlahnya di Bumi.
\end{remark}

\section{Teori sel}

\begin{proposition}[Teori sel]\label{prop:g6:cell-unit-of-life:theory}
Dua kalimat, yang dibangun berabad-abad, merangkum bab ini dan
\cref{ch:g5:first-look-at-cells}:
\begin{enumerate}
  \item setiap \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} tersusun atas satu \omterm{def:g5:first-look-at-cells:cell}{sel} atau lebih, yaitu
        satuan kehidupan yang terkecil;
  \item setiap \omterm{def:g5:first-look-at-cells:cell}{sel} datang dari pembelahan sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} yang sudah ada.
\end{enumerate}
Belum pernah ditemukan satu perkecualian pun --- tidak di laut yang
terdalam, tidak di kubangan batu yang tertua. Kalimat yang kedua punya
akibat yang menyapu luas: setiap \omterm{def:g5:first-look-at-cells:cell}{sel} yang hidup hari ini meneruskan
sebuah rantai pembelahan yang tak terputus yang menjangkau kembali
sampai kehidupan yang paling awal --- \omterm{prop:g6:classification-kinship:kinship}{kekerabatan}
(\cref{prop:g6:classification-kinship:kinship}) yang tertulis pada
skala \omterm{def:g5:first-look-at-cells:cell}{selnya}.
\end{proposition}
\admitted

\begin{method}[Membaca organisme apa pun pada skala sel]\label{met:g6:cell-unit-of-life:read}
Ketika berhadapan dengan \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} apa pun:
\begin{enumerate}
  \item tanyakan lebih dahulu: satu \omterm{def:g5:first-look-at-cells:cell}{sel}, atau banyak?
  \item kalau banyak: harapkanlah jenis yang terkhususkan --- yang
        melapisi, yang mengangkut, yang \omterm{def:g3:bones-and-muscles:muscle}{berkontraksi}, yang memerintah;
  \item \omterm{def:g1:plants-around-us:plant}{tumbuhan} atau \omterm{def:g1:animals-around-us:animal}{hewan}? periksalah adanya dinding dan butir hijau
        di \omterm{def:g5:first-look-at-cells:microscope}{mikroskop};
  \item setiap pertumbuhan, perbaikan atau perkembangbiakan yang kamu
        amati: terjemahkanlah menjadi pembelahan \omterm{def:g5:first-look-at-cells:cell}{sel} --- kalimat 2
        menjamin bahwa terjemahannya ada.
\end{enumerate}
\end{method}

\begin{example}[Teorinya yang bekerja di dalam buku ini]\label{ex:g6:cell-unit-of-life:atwork}
Kalimat 1 menjamin \omterm{prop:g5:first-look-at-cells:principle}{kesatuan kehidupan}
(\cref{prop:g5:first-look-at-cells:principle}); kalimat 2 menjamin
pertumbuhan-lewat-pembelahan
(\cref{ex:g5:first-look-at-cells:one}), penggandaan \omterm{ex:g6:producing-our-food:bread}{ragi} di dalam
adonannya, \omterm{def:g6:decomposers-and-soil:decomposers}{pengurai} yang berlipat ganda di dalam tumpukan yang hangat,
dan angkatan kerja \omterm{def:g6:producing-our-food:fermentation}{mikroba} pada
\cref{ch:g6:producing-our-food}. Satu teori kecil, yang membawa
sebagian besar \omterm{def:g1:living-or-not:living}{biologi} tahun ini.
\end{example}

\section{Soal: Sore Bermikroskop}

\begin{problem}[{Soal akhir pekan --- empat kaca preparat, satu
teori}]\label{pb:g6:cell-unit-of-life:1}
Sore bermikroskop di kelasnya: empat stasiun, empat kaca preparat ---
\omterm{def:g1:the-five-senses:sense}{kulit} bawang bombai (A), kerokan pipi (B), setetes air kolam (C), dan
olesan yogurt (D). Tiap regu berkeliling ke keempatnya lalu menyimpan
gambarnya.

\medskip\noindent\textbf{Bagian I --- Stasiun A dan B.}
\begin{enumerate}
  \item Di stasiun A regunya menggambar ``bata'' berdinding lurus,
        masing-masing dengan sebuah bintik gelap yang terdesak ke satu
        sisi. Sebutkan dindingnya, bintiknya, dan pelaku pendesakannya.
  \item Di stasiun B: \omterm{def:g5:first-look-at-cells:cell}{sel} yang membulat, tanpa dinding, bintiknya di
        tengah. Daftarkan perlengkapan yang dimiliki \omterm{def:g5:first-look-at-cells:cell}{sel} A dan tidak
        dimiliki \omterm{def:g5:first-look-at-cells:cell}{sel} B --- serta perlengkapan yang dibagi bersama
        keduanya.
  \item Seorang anggota regu bertanya mengapa \omterm{def:g5:first-look-at-cells:cell}{sel} bawang bombainya,
        yang berasal dari sebuah umbi yang ditumbuhkan di bawah \omterm{def:g6:decomposers-and-soil:soil}{tanah},
        tidak memperlihatkan butir hijau. Jawablah dengan tugas
        butirnya dan alamat umbinya.
  \item \omterm{def:g5:first-look-at-cells:cell}{Sel} siapa yang ada di kaca preparat B? Nyatakan apa yang
        disiratkannya menurut kalimat 2 teori \omterm{def:g5:first-look-at-cells:cell}{selnya}: dari mana tiap
        \omterm{def:g5:first-look-at-cells:cell}{sel} itu datang?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Stasiun C dan D.}
\begin{enumerate}[resume]
  \item Stasiun C: si pendayung berbentuk sandal, para perenang hijau,
        gumpalan yang mengalir. Untuk masing-masing, sebutkan kegiatan
        hidup yang terlihat dalam pengamatan semenit --- lalu
        simpulkan masing-masingnya itu apa.
  \item Perenang hijaunya adalah \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal namun berperan sebagai
        \omterm{def:g5:food-webs:roles}{produsen}. Perlengkapan mana yang harus diperlihatkan gambarnya
        untuk membuktikannya?
  \item Olesan yogurt stasiun D memperlihatkan \omterm{def:g5:first-look-at-cells:cell}{sel} mungil yang
        berpasangan dan berantai dalam jumlah amat besar. Hubungkanlah
        mereka dengan \cref{ex:g6:producing-our-food:yogurt}: apa yang
        dilakukan \omterm{def:g5:first-look-at-cells:cell}{sel} ini terhadap \omterm{def:g2:animals-grow-up:born}{susunya}, dan mereka itu apa dalam
        perbendaharaan kata bab ini?
  \item Sebuah regu mengaku bahwa kaca preparat D memperlihatkan
        ``bukan \omterm{def:g6:exploring-our-environment:components}{makhluk hidup}, hanya bintik''. Satu pengamatan tunggal
        apa --- kalau ada kesabaran atau kaca preparat yang hangat ---
        yang akan menyelesaikan apakah bintiknya hidup, menurut
        kalimat 2 teori \omterm{def:g5:first-look-at-cells:cell}{selnya}?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Menuliskan laporannya.}
\begin{enumerate}[resume]
  \item Laporannya harus mengelompokkan keempat kaca preparatnya
        menurut: satu \omterm{def:g5:first-look-at-cells:cell}{sel} atau banyak? \omterm{def:g1:plants-around-us:plant}{tumbuhan}, \omterm{def:g1:animals-around-us:animal}{hewan} atau bukan
        keduanya? Kerjakanlah.
  \item Setiap gambar sore itu memperlihatkan perlengkapan tiga bagian
        yang sama. Nyatakan perlengkapannya, dan kesemestaannya itu
        merupakan bukti bagi apa
        (\cref{prop:g5:first-look-at-cells:principle}).
  \item Gurunya menutup: ``segala yang kalian gambar hari ini dahulu
        pernah menjadi satu \omterm{def:g5:first-look-at-cells:cell}{sel}.'' Belalah kalimat itu bagi bawang
        bombainya, bagi pemilik \omterm{def:g5:first-look-at-cells:cell}{sel} pipinya, dan bagi rantai yogurtnya.
  \item Rumuskan temuan sore itu sebagai teori \omterm{def:g5:first-look-at-cells:cell}{selnya} --- dua kalimat
        --- lalu tandailah, bagi tiap kaca preparatnya, kalimat mana
        yang paling baik digambarkannya.
\end{enumerate}
\end{problem}
