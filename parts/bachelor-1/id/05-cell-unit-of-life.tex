\chapter{Sel: Satuan Kehidupan}\label{ch:b1:cell-unit-of-life}

Pada 1665 Robert Hooke meletakkan sekeping tipis gabus di bawah sebuah
lensa dan melihatnya terbagi menjadi kotak-kotak kecil yang disebutnya
\omterm{def:b1:cell-unit-of-life:cell}{sel}, menurut bilik sebuah biara. Sepuluh tahun kemudian seorang pedagang
kain Belanda, Antoni van Leeuwenhoek, yang mengasah lensanya sendiri
sampai kekuatan yang tak tercapai orang lain, melihat makhluk hidup
berenang dalam setetes air kolam, dalam kerokan giginya, dalam air
maninya sendiri. Butuh satu setengah abad lagi untuk merumuskan apa yang
telah dilihat keduanya: bahwa setiap \omterm{def:b1:organism-environment:organism}{organisme} tersusun dari \omterm{def:b1:cell-unit-of-life:cell}{sel}, dan
bahwa setiap \omterm{def:b1:cell-unit-of-life:cell}{sel} berasal dari \omterm{def:b1:cell-unit-of-life:cell}{sel}. Bab ini merumuskan \omterm{prop:b1:cell-unit-of-life:theory}{teori sel} beserta
buktinya, mempertentangkan kedua macam \omterm{def:b1:cell-unit-of-life:cell}{sel}, dan memerikan peralatan ---
mikroskop, sentrifus, labu biakan --- yang memungkinkan kita melihat
\omterm{def:b1:cell-unit-of-life:cell}{sel}, membongkarnya, dan menjaganya tetap hidup di luar \omterm{def:b1:organism-environment:organism}{organismenya}.

\section{Teori sel}

\begin{definition}[Sel]\label{def:b1:cell-unit-of-life:cell}
Yang disebut \emph{sel}\index{sel} adalah satuan terkecil materi hidup:
sebuah volume \emph{sitoplasma}\index{sitoplasma} berair yang dibatasi
sebuah \emph{membran plasma}\index{membran plasma}, yang memuat bahan
genetiknya (DNA) beserta mesin yang mengungkapkannya, yang sanggup
memasukkan zat hara, mengubah energi, memelihara susunannya sendiri
dan, bila keadaan mengizinkan, membelah menjadi dua sel.
\end{definition}

\begin{proposition}[Teori sel]\label{prop:b1:cell-unit-of-life:theory}
\index{teori sel}
\begin{enumerate}
  \item Setiap \omterm{def:b1:organism-environment:organism}{organisme} tersusun dari satu \omterm{def:b1:cell-unit-of-life:cell}{sel} atau lebih, dan \omterm{def:b1:cell-unit-of-life:cell}{sel}
        adalah satuan struktur dan fungsi makhluk hidup (Schleiden dan
        Schwann, 1838--1839).
  \item Setiap \omterm{def:b1:cell-unit-of-life:cell}{sel} muncul dari \omterm{def:b1:cell-unit-of-life:cell}{sel} yang sudah ada lewat pembelahan
        (Virchow, 1855: \emph{omnis cellula e cellula}); tidak ada
        generasi spontan.
  \item Bahan keturunannya diwariskan dari \omterm{def:b1:cell-unit-of-life:cell}{sel} ke \omterm{def:b1:cell-unit-of-life:cell}{sel} pada pembelahan,
        sehingga kesinambungan kehidupan adalah kesinambungan \omterm{def:b1:cell-unit-of-life:cell}{sel}.
\end{enumerate}
\end{proposition}
\begin{proof}[Bukti eksperimental]
Gabus Hooke (1665) dan sayatan tumbuhan Grew serta Malpighi
memperlihatkan \omterm{def:b1:body-plans-tissues:tissue}{jaringan} tumbuhan sebagai hamparan bilik; lensa
Leeuwenhoek (tahun 1670-an) memperlihatkan \omterm{def:b1:organism-environment:organism}{organisme} bersel tunggal, \omterm{def:b1:cell-unit-of-life:cell}{sel}
darah dan sperma. Dua abad pengasahan lensa yang makin baik menemukan
\omterm{def:b1:cell-unit-of-life:cell}{sel} pada tiap \omterm{def:b1:body-plans-tissues:tissue}{jaringan} yang diperiksa, hewan maupun tumbuhan, sesudah
Schwann mengenali \omterm{def:b1:cell-unit-of-life:cell}{sel} berinti pada tulang rawan dan embrio. Pembelahan
terlihat langsung pada alga dan pada embrio, dan Virchow tidak menemukan
\omterm{def:b1:body-plans-tissues:tissue}{jaringan} mana pun, sehat maupun sakit, yang \omterm{def:b1:cell-unit-of-life:cell}{selnya} tidak berasal dari
\omterm{def:b1:cell-unit-of-life:cell}{sel} lain. Labu berleher angsa Pasteur (1861) --- kaldu yang tetap steril
berbulan-bulan ketika debu tak dapat mencapainya, dan penuh makhluk
dalam beberapa hari ketika debu dapat --- menutup pertanyaan tentang
generasi spontan. Pernyataan ketiga adalah karya akhir abad kesembilan
belas, yang mengikuti kromosom sepanjang pembelahan
(\cref{ch:b1:replication-mitosis}).
\end{proof}

\begin{example}[Satu sel, banyak sel]\label{ex:b1:cell-unit-of-life:onemany}
Sebuah bakteri, sebuah khamir, sebuah amoeba adalah \omterm{def:b1:organism-environment:organism}{organisme} utuh
bersel satu, yang mengerjakan segalanya --- makan, bergerak, membelah
--- dalam beberapa mikrometer. Seorang manusia adalah sekitar
$\num{3e13}$ \omterm{def:b1:cell-unit-of-life:cell}{sel} dari dua ratus macam, tak satu pun di antaranya sanggup
hidup sendirian dalam sebuah kolam, masing-masing menurun lewat
pembelahan yang tak terputus dari satu \omterm{def:b1:cell-unit-of-life:cell}{sel} telur yang dibuahi. Kelinci
\cref{ch:b1:mammal-organization} dan pohon ek
\cref{ch:b1:flowering-plant-organization} adalah teori yang sama,
diterapkan dua kali.
\end{example}

\section{Dua macam sel}

\begin{definition}[Sel prokariot dan sel eukariot]\label{def:b1:cell-unit-of-life:prokeuk}
Sebuah \emph{sel prokariot}\index{sel prokariot} (bakteri, arkea) tidak
berinti: DNA-nya, biasanya satu kromosom melingkar, terletak dalam
kawasan \omterm{def:b1:cell-unit-of-life:cell}{sitoplasma} yang disebut \emph{nukleoid}\index{nukleoid}, dan ia
tidak memiliki ruang berbatas membran. Ia kecil
(\qtyrange{0.5}{5}{\micro m}) dan dibatasi sebuah \omterm{def:b1:cell-unit-of-life:cell}{membran plasma} serta,
di luarnya, sebuah \emph{dinding sel}\index{dinding sel}. Sebuah
\emph{sel eukariot}\index{sel eukariot} (protista, jamur, tumbuhan,
hewan) menyimpan DNA-nya, berupa beberapa kromosom lurus, di dalam
sebuah \emph{inti}\index{inti sel} yang dibatasi membran rangkap, dan
membagi \omterm{def:b1:cell-unit-of-life:cell}{sitoplasmanya} menjadi \emph{organel}\index{organel} berbatas
membran (\cref{ch:b1:eukaryotic-cell}). Ia lebih besar
(\qtyrange{10}{100}{\micro m}) dan memiliki \omterm{def:b1:body-plans-tissues:segments}{rangka dalam} berupa filamen
protein.
\end{definition}

\begin{proposition}[Sel bakteri]\label{prop:b1:cell-unit-of-life:bacterium}
\emph{E. coli}, bakteri model, adalah \omterm{def:b1:flowering-plant-organization:organs}{batang} sepanjang \qty{2}{\micro m}
dan selebar \qty{0.8}{\micro m}: sebuah volume sekitar
\qty{1}{\micro m^3} yang memuat satu kromosom berisi $\num{4.6e6}$
pasangan basa (\qty{1.5}{mm} DNA yang terlipat sepuluh ribu kali),
sekitar \num{20000} ribosom, beberapa juta molekul protein dari
\num{2000} macam, dan sering satu atau beberapa
\emph{plasmid}\index{plasmid} --- molekul DNA melingkar kecil yang
membawa gen tambahan. Selubungnya, dari dalam ke luar, adalah \omterm{def:b1:cell-unit-of-life:cell}{membran
plasma}, sebuah dinding tipis \emph{peptidoglikan}\index{peptidoglikan}
(sebuah jala rantai gula yang bersilang ikat lewat peptida pendek, yang
menahan tekanan dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya}), dan sebuah membran luar. Bakteri dengan
dinding peptidoglikan tebal dan tanpa membran luar menahan zat warna
ungu kristal dari \emph{pewarnaan Gram}\index{pewarnaan Gram}, yaitu
bakteri Gram positif; bakteri berdinding tipis di bawah sebuah membran
luar kehilangannya, yaitu Gram negatif, seperti \emph{E. coli}. Flagel berputar seperti baling-baling;
pili melekatkan \omterm{def:b1:cell-unit-of-life:cell}{sel} pada permukaan dan pada \omterm{def:b1:cell-unit-of-life:cell}{sel} lain.
\end{proposition}

\begin{method}[Pewarnaan Gram]\label{met:b1:cell-unit-of-life:gram}
\begin{enumerate}
  \item Lekatkan sapuan tipis bakteri pada sebuah kaca objek dengan
        pemanasan.
  \item Warnai dengan ungu kristal (semua \omterm{def:b1:cell-unit-of-life:cell}{sel} menjadi ungu), lalu dengan
        iodin, yang membentuk kompleks besar dengan zat warna di dalam
        \omterm{def:b1:cell-unit-of-life:cell}{sel}.
  \item Bilas dengan alkohol: peptidoglikan tebal \omterm{def:b1:cell-unit-of-life:cell}{sel} Gram positif
        menyusut dan memerangkap kompleks itu; dinding tipis \omterm{def:b1:cell-unit-of-life:cell}{sel} Gram
        negatif, begitu alkohol melarutkan membran luarnya, membiarkannya
        terbilas keluar.
  \item Warnai lawan dengan zat warna merah muda (safranin): \omterm{def:b1:cell-unit-of-life:cell}{sel} Gram
        positif tampak ungu, sedangkan \omterm{def:b1:cell-unit-of-life:cell}{sel} Gram negatif tampak merah
        muda. Hasilnya
        meramalkan struktur dindingnya dan, dengan itu, kepekaannya
        terhadap banyak antibiotik.
\end{enumerate}
\end{method}

\begin{example}[Mengapa sel eukariot terbagi menjadi ruang]\label{ex:b1:cell-unit-of-life:whycompart}
Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} \qty{20}{\micro m} bervolume seribu kali volume sebuah
bakteri tetapi hanya berpermukaan membran seratus kali
(\cref{prop:b1:organism-environment:sv}). Membran adalah tempat sebuah
\omterm{def:b1:cell-unit-of-life:cell}{sel} membawa rantai pengubah energi dan sistem pengangkutnya; eukariot
mendapatkan kembali permukaannya dengan melipat membran di dalam dirinya
--- membran dalam sebuah mitokondria, lembaran retikulum endoplasma ---
dan memperoleh, di samping itu, bilik terpisah tempat kimia yang tak
sejalan (pencernaan di dalam lisosom, sintesis di dalam sitosol) dapat
berjalan berdampingan.
\end{example}

\section{Melihat sel: mikroskopi}

\begin{definition}[Perbesaran dan daya pisah]\label{def:b1:cell-unit-of-life:resolution}
Sebuah mikroskop \emph{memperbesar}\index{perbesaran} (memperbesar
bayangan) dan \emph{memisahkan}\index{daya pisah} (memisahkan titik
yang bertetangga). Hanya yang kedua membatasi apa yang dapat dilihat:
\emph{daya pisah}\index{daya pisah mikroskop} adalah jarak terkecil $d$
antara dua titik yang masih tampak berbeda,
\[
d = \frac{0.61\,\lambda}{\mathrm{NA}},
\]
dengan $\lambda$ adalah panjang gelombang penyinarannya dan
$\mathrm{NA}$ apertura numerik lensa objektifnya (sinus setengah sudut
tangkapan cahayanya dikali indeks bias mediumnya, paling besar sekitar
$1.4$ dengan minyak). Memperbesar melewati daya pisahnya hanya
memperbesar kaburnya, bukan rinciannya.
\end{definition}

\begin{proposition}[Mikroskop cahaya dan mikroskop elektron]\label{prop:b1:cell-unit-of-life:microscopes}
Dengan cahaya tampak ($\lambda \approx \qty{550}{nm}$) dan
$\mathrm{NA} = 1.4$, \emph{mikroskop cahaya}\index{mikroskop cahaya}
memisahkan \qty{0.24}{\micro m}: \omterm{def:b1:cell-unit-of-life:cell}{sel}, \omterm{def:b1:cell-unit-of-life:prokeuk}{inti}, kloroplas, mitokondria
sebagai titik, bakteri sebagai \omterm{def:b1:flowering-plant-organization:organs}{batang} --- dan cuplikan yang hidup serta
bergerak. Kontras dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} yang bening diperoleh lewat pewarnaan (yang
biasanya mematikan), lewat \emph{kontras fase} (yang mengubah selisih indeks
bias menjadi selisih kecerahan), atau lewat \emph{fluoresensi} (yang
menandai satu molekul dengan zat warna atau protein berpendar lalu
mencitrakan cahayanya saja). \emph{Mikroskop elektron}\index{mikroskop elektron}
memakai elektron berpanjang gelombang beberapa pikometer; \omterm{def:b1:cell-unit-of-life:resolution}{daya pisah}
praktisnya, yang dibatasi lensanya, sekitar \qty{0.2}{nm} pada mikroskop
transmisi (TEM: elektron menembus sayatan setebal \qty{50}{nm} yang
diwarnai logam berat, yang memberikan struktur dalam \omterm{def:b1:cell-unit-of-life:prokeuk}{organel}) dan
beberapa nanometer pada mikroskop payar (SEM: sebuah berkas disapukan
pada permukaan berlapis logam, yang memberikan reliefnya). Mikroskopi
elektron menuntut ruang hampa serta cuplikan yang mati, terfiksasi dan
kering.
\end{proposition}

\begin{method}[Memilih peralatannya]\label{met:b1:cell-unit-of-life:choose}
\begin{enumerate}
  \item Untuk mengikuti sebuah proses yang hidup (pembelahan \omterm{def:b1:cell-unit-of-life:cell}{sel}, gerak,
        sebuah protein berpendar yang berpindah antarruang): mikroskopi
        cahaya, kontras fase atau fluoresensi, pada \omterm{def:b1:cell-unit-of-life:cell}{sel} yang tak
        terfiksasi.
  \item Untuk menghitung dan mengenali \omterm{def:b1:cell-unit-of-life:cell}{sel} dalam sebuah \omterm{def:b1:body-plans-tissues:tissue}{jaringan}:
        sediaan yang terfiksasi, tersayat dan terwarnai di bawah
        \omterm{prop:b1:cell-unit-of-life:microscopes}{mikroskop cahaya}.
  \item Untuk melihat struktur dalam sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{organel}, sebuah membran,
        sebuah virus: mikroskopi elektron transmisi atas sayatan tipis,
        atau atas partikel yang diwarnai negatif.
  \item Untuk melihat relief permukaan sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} atau \omterm{def:b1:body-plans-tissues:tissue}{jaringan}:
        mikroskopi elektron payar.
  \item Untuk menemukan letak satu protein: imunofluoresensi (cahaya)
        atau penandaan imunoemas (elektron), dengan antibodi terhadapnya.
\end{enumerate}
\end{method}

\begin{definition}[Fraksinasi sel]\label{def:b1:cell-unit-of-life:fractionation}
Yang disebut \emph{fraksinasi sel}\index{fraksinasi sel} memisahkan

aktivitas yang diketahui hanya berada dalam satu ruang.
\end{definition}

\begin{proposition}[Pengendapan]\label{prop:b1:cell-unit-of-life:stokes}
Sebuah partikel bulat berjari-jari $r$ dan bermassa jenis $\rho_p$ dalam
fluida bermassa jenis $\rho_f$ dan berkekentalan $\eta$, dalam medan
sentrifugal $g_c$ (kelipatan $g$), mengendap pada laju mantap
\[
v = \frac{2}{9}\,\frac{r^2\,(\rho_p - \rho_f)\,g_c}{\eta}
\]
(hukum Stokes). Lajunya tumbuh sebagai kuadrat jari-jarinya: sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti}
\qty{5}{\micro m} jatuh seratus kali lebih cepat daripada sebuah
mitokondria \qty{0.5}{\micro m}, yang jatuh seribu kali lebih cepat
daripada sebuah ribosom \qty{15}{nm}. Karena itulah tabung yang sama
memisahkannya pada laju yang meningkat.
\end{proposition}
\begin{proof}
Pada laju yang mantap seretan Stokes, $6\pi\eta r v$, mengimbangi berat
bersih partikel itu dalam fluidanya,
$\frac{4}{3}\pi r^3(\rho_p - \rho_f)g_c$; menyelesaikannya untuk $v$
memberikan rumus tadi.
\end{proof}

\begin{definition}[Biakan sel]\label{def:b1:cell-unit-of-life:culture}
Yang disebut \emph{biakan sel}\index{biakan sel} menjaga \omterm{def:b1:cell-unit-of-life:cell}{sel} tetap hidup
dan membelah di luar \omterm{def:b1:organism-environment:organism}{organismenya}, dalam medium steril yang memasok zat
hara, garam, faktor tumbuh dan, bagi \omterm{def:b1:cell-unit-of-life:cell}{sel} hewan, sebuah permukaan untuk
melekat, pada suhu \omterm{def:b1:organism-environment:organism}{organismenya}. Bakteri dan khamir tumbuh dalam medium
sederhana dan berlipat dua dalam beberapa menit sampai jam; \omterm{def:b1:cell-unit-of-life:cell}{sel} hewan
yang diambil dari sebuah \omterm{def:b1:body-plans-tissues:tissue}{jaringan} (\emph{biakan primer}) membelah
sejumlah kali yang terbatas, sedangkan \emph{galur \omterm{def:b1:cell-unit-of-life:cell}{sel}} yang diturunkan
dari tumor atau \omterm{def:b1:cell-unit-of-life:cell}{sel} yang tertransformasi membelah tanpa batas. Biakan
menjadikan \omterm{def:b1:cell-unit-of-life:cell}{sel} sebuah objek percobaan: satu jenis, satu medium, satu
variabel pada satu waktu.
\end{definition}

\begin{example}[Sebuah kurva pertumbuhan]\label{ex:b1:cell-unit-of-life:growthcurve}
Sebuah labu \emph{E. coli} dalam medium kaya pada \qty{37}{\celsius}:
sebuah jeda setengah jam, lalu pertumbuhan eksponensial --- cacahnya
berlipat dua tiap \qty{20}{min}, sehingga $N = N_0\,2^{t/20}$ --- sampai
sebuah zat hara habis atau sisanya menumpuk dan biakan itu mencapai
dataran dekat $10^9$ \omterm{def:b1:cell-unit-of-life:cell}{sel} tiap mililiter. Dari satu \omterm{def:b1:cell-unit-of-life:cell}{sel},
$2^{30} \approx 10^9$ \omterm{def:b1:cell-unit-of-life:cell}{sel} dalam sepuluh jam. Kurva yang sama, dengan waktu lipat dua
satu hari, memerikan \omterm{def:b1:cell-unit-of-life:culture}{biakan sel} manusia.
\end{example}

\section{Soal: Memfraksinasi Sebuah Hati}

\begin{problem}[{Soal akhir pekan --- sepuluh gram hati tikus yang
dibongkar lewat sentrifugasi: endapan, penanda, hukum Stokes dan neraca
sebuah enzim, berakhir pada faktor pemurnian fraksi mitokondrianya}]
\label{pb:b1:cell-unit-of-life:1}
\qty{10}{g} hati tikus \omterm{def:b1:cell-unit-of-life:fractionation}{dihomogenkan} dalam \qty{50}{mL} sukrosa isotonik
yang dingin lalu dipusingkan dalam tiga tahap: $1000\,g$ selama
\qty{10}{min} (endapan 1), $\num{10000}\,g$ selama \qty{20}{min}
(endapan 2), $\num{100000}\,g$ selama \qty{60}{min} (endapan 3), yang
menyisakan supernatan akhir. Kadar protein dan dua aktivitas enzim
diukur dalam \omterm{def:b1:cell-unit-of-life:fractionation}{homogenat} dan dalam tiap fraksinya:

\begin{center}
\small
\begin{tabular}{lccc}
\hline
fraksi & protein (mg) & suksinat dehidrogenase (U) & laktat dehidrogenase (U) \\
\hline
\omterm{def:b1:cell-unit-of-life:fractionation}{homogenat} & 2000 & 100 & 400 \\
endapan 1 & 500 & 15 & 20 \\
endapan 2 & 200 & 60 & 12 \\
endapan 3 & 300 & 5 & 8 \\
supernatan akhir & 950 & 4 & 350 \\
\hline
\end{tabular}
\end{center}

Ambil $\eta = \qty{1e-3}{Pa.s}$ bagi mediumnya, lintasan pengendapan
\qty{5}{cm}, dan $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{Bagian I --- Apa berada di mana.}
\begin{enumerate}
  \item Namailah \omterm{def:b1:cell-unit-of-life:prokeuk}{organel} utama dalam tiap endapan.
  \item Mengapa medium penghomogenannya harus dingin dan isotonik?
  \item Suksinat dehidrogenase adalah penanda ruang yang mana? Laktat
        dehidrogenase penanda ruang yang mana?
  \item Hitunglah perolehan kembali proteinnya: jumlah atas semua
        fraksinya sebagai persentase \omterm{def:b1:cell-unit-of-life:fractionation}{homogenatnya}. Beri tanggapan.
  \item Hitunglah perolehan kembali tiap aktivitas enzim dengan cara
        yang sama.
  \item Hitunglah kadar protein tiap gram hati dan fraksi massa segar
        hati itu yang berupa protein.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Hukum Stokes.}
Mitokondria: jari-jari \qty{0.5}{\micro m}, kelebihan massa jenis
\qty{100}{kg/m^3}. \omterm{def:b1:cell-unit-of-life:prokeuk}{Inti}: jari-jari \qty{4}{\micro m}, kelebihan yang
sama. Ribosom: jari-jari \qty{12}{nm}, kelebihan massa jenis
\qty{600}{kg/m^3}.
\begin{enumerate}[resume]
  \item Hitunglah laju pengendapan sebuah mitokondria pada
        $\num{10000}\,g$ dan waktu tempuhnya sejauh \qty{5}{cm}.
        Cukupkah \qty{20}{min}?
  \item Hitunglah jarak tempuh sebuah mitokondria selama pemusingan
        pertama ($1000\,g$, \qty{10}{min}). Berapa fraksi mitokondria,
        yang tersebar merata pada awalnya, yang akan mengendap pada
        tahap pertama itu?
  \item Bandingkan dengan \qty{15}{\%} suksinat dehidrogenase yang
        ditemukan dalam endapan 1. Apa lagi yang dapat menyeret
        mitokondria ke dalam endapan 1?
  \item Hitunglah waktu sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} menempuh \qty{5}{cm} pada
        $1000\,g$.
  \item Hitunglah waktu sebuah ribosom menempuh \qty{5}{cm} pada
        $\num{100000}\,g$. Akankah sebuah ribosom mengendap pada
        $\num{10000}\,g$ dalam \qty{20}{min}?
  \item Jelaskan mengapa urutan pemusingannya dengan gaya yang
        meningkat dan mengapa tiap tahapnya harus cukup lama tetapi
        tidak terlalu lama.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Aktivitas jenis dan pemurnian.}
Aktivitas jenis sebuah enzim dalam sebuah fraksi adalah aktivitasnya
tiap miligram protein. Faktor pemurnian sebuah fraksi bagi sebuah enzim
adalah aktivitas jenisnya dibagi aktivitas jenis dalam \omterm{def:b1:cell-unit-of-life:fractionation}{homogenatnya};
rendemennya adalah aktivitas fraksi itu dibagi aktivitas \omterm{def:b1:cell-unit-of-life:fractionation}{homogenatnya}.
\begin{enumerate}[resume]
  \item Hitunglah aktivitas jenis suksinat dehidrogenase dalam
        \omterm{def:b1:cell-unit-of-life:fractionation}{homogenat} dan dalam tiap fraksinya.
  \item Hitunglah faktor pemurnian dan rendemen suksinat dehidrogenase
        dalam endapan 2.
  \item Hitunglah faktor pemurnian suksinat dehidrogenase dalam
        endapan 3 dan tafsirkan nilainya.
  \item Hitunglah aktivitas jenis laktat dehidrogenase dalam \omterm{def:b1:cell-unit-of-life:fractionation}{homogenat}
        dan dalam supernatan akhir, serta faktor pemurnian dan
        rendemen yang bersesuaian.
  \item Mengapa faktor pemurnian enzim sitosol lebih rendah daripada
        faktor enzim mitokondria meskipun rendemennya lebih tinggi?
  \item Endapan 2 memuat \qty{12}{U} laktat dehidrogenase. Apakah enzim
        itu sebagian bersifat mitokondria? Ajukan penjelasan yang lebih
        baik dan sebuah cara mengujinya.
  \item Berapa fraksi protein hati itu yang bersifat mitokondria,
        seandainya seluruh suksinat dehidrogenasenya bersifat
        mitokondria dan endapan 2 murni \qty{60}{\%}? (Petunjuk: pakai
        aktivitas jenis mitokondria murni, yang disimpulkan dari
        endapan 2, dan aktivitas totalnya.)
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Memeriksa fraksinya.}
\begin{enumerate}[resume]
  \item Endapan 2 diperiksa dengan mikroskopi elektron: peralatan dan
        sediaan yang mana, dan apa yang seharusnya terlihat?
  \item Bagaimana cara memeriksa endapan 1 terhadap \omterm{def:b1:cell-unit-of-life:cell}{sel} yang belum
        pecah, dan apa yang akan dilakukan bila ditemukan banyak?
  \item Endapan 2 disuspensikan kembali dan dilapiskan pada sebuah
        gradien massa jenis sukrosa, lalu dipusingkan sampai tiap
        partikel berhenti di tempat massa jenisnya sama dengan massa
        jenis mediumnya. Mitokondria membentuk pita pada
        \qty{1.18}{g/mL}, lisosom pada \qty{1.22}{g/mL}, peroksisom
        pada \qty{1.24}{g/mL}. Jelaskan mengapa pemisahan ini berhasil
        sedangkan \omterm{def:b1:cell-unit-of-life:fractionation}{sentrifugasi bertingkat} tidak sanggup mencapainya.
  \item Aktivitas asam fosfatase dalam endapan 2 naik tajam ketika
        sebuah deterjen ditambahkan pada pengujiannya. Jelaskan, dan
        sebutkan apa yang dikatakannya tentang keadaan lisosom dalam
        endapan itu.
