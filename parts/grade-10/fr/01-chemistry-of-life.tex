\chapter{La composition chimique des êtres vivants}\label{ch:g10:chemistry-of-life}

Laissez une tranche de pain trop longtemps sur le gril~: elle noircit, et
ce qui subsiste est presque uniquement du carbone. Pesez une tranche de
pomme fraîche, séchez-la une journée dans un four tiède, pesez-la de
nouveau~: les cinq sixièmes de sa masse sont partis en vapeur d'eau.
Le pain, la pomme, le chêne, la méduse, vous --- tout être vivant est
construit, au fond, à partir de la même courte liste d'ingrédients
chimiques, et cette liste n'est pas celle dont les roches sont faites.
Ce chapitre démonte la matière vivante, ingrédient par ingrédient, et
demande ce qui la rend singulière.

\section{Les éléments~: les mêmes atomes que les roches, en d'autres proportions}

\begin{definition}[Composition chimique de la matière vivante]\label{def:g10:chemistry-of-life:composition}
La \emph{composition chimique}\index{composition chimique} d'un
échantillon de matière vivante est la liste des éléments chimiques
qu'il contient, avec la fraction de la masse de l'échantillon que
chacun représente. Elle se mesure en démontant chimiquement
l'échantillon --- en le brûlant, en pesant ce qui s'évapore et ce qui
reste, en analysant les gaz dégagés --- et non en le regardant.
\end{definition}

\begin{proposition}[Quatre éléments font 96~\% d'un corps vivant]\label{prop:g10:chemistry-of-life:four}
En masse, la matière vivante est faite presque entièrement de quatre
éléments~: oxygène (O), carbone (C), hydrogène (H) et azote (N). Dans
le corps humain, ils comptent pour environ 65~\%, 18.5~\%, 9.5~\% et
3.2~\% de la masse --- 96~\% ensemble. Le calcium, le phosphore, le
potassium, le soufre, le sodium, le chlore et le magnésium se
partagent l'essentiel des 4~\% restants, et une douzaine
d'\emph{oligo-éléments} (fer, zinc, iode, cuivre\dots) sont présents
très en dessous de 0.1~\% tout en étant indispensables. Les mêmes
quatre éléments dominent chez une bactérie, un blé ou une baleine,
dans des proportions voisines.
\end{proposition}
\admitted

\begin{example}[Matière vivante et roche]\label{ex:g10:chemistry-of-life:crust}
La croûte terrestre est elle aussi à 46~\% d'oxygène, mais ses éléments
suivants sont le silicium (28~\%), l'aluminium (8~\%) et le fer
(5.6~\%). Le carbone ne fait que 0.02~\% de la croûte et l'azote
0.002~\%, contre 18.5~\% et 3.2~\% d'un corps. Un être vivant n'est
donc pas un échantillon de son environnement~: il \emph{concentre}
quelques éléments d'un facteur mille ou davantage. Cette sélection est
la première signature chimique de la vie.
\end{example}

\begin{remark}[D'où viennent les atomes]\label{rem:g10:chemistry-of-life:origin}
Chacun de ces atomes est ordinaire~: le carbone de vos muscles était
du dioxyde de carbone de l'air avant qu'une plante ne le fixe, l'azote
de vos \omterm{def:g10:chemistry-of-life:families}{protéines} était de l'azote de l'air avant que des bactéries du
sol ne le captent. Ce qui est propre à la vie, ce ne sont pas les
atomes, mais la façon dont ils sont assemblés et les proportions dans
lesquelles ils sont rassemblés.
\end{remark}

\section{L'eau et les sels minéraux}

\begin{definition}[Matière minérale et matière organique]\label{def:g10:chemistry-of-life:mineral-organic}
Les molécules d'un être vivant se rangent en deux groupes. La
\emph{matière minérale}\index{matière minérale} est faite de molécules
qui existent aussi hors des êtres vivants~: l'eau et les
\emph{sels minéraux}\index{sels minéraux} dissous (ions tels que
$\mathrm{Na^+}$, $\mathrm{K^+}$, $\mathrm{Ca^{2+}}$, $\mathrm{Cl^-}$,
phosphate et bicarbonate). La \emph{matière organique}\index{matière organique}
est faite de molécules bâties sur un squelette d'atomes de carbone liés
à de l'hydrogène, et le plus souvent à de l'oxygène et de l'azote ---
des molécules que, sur Terre, seuls les êtres vivants ou leurs restes
produisent.
\end{definition}

\begin{proposition}[L'eau est le constituant principal]\label{prop:g10:chemistry-of-life:water}
L'eau est la molécule la plus abondante de tout être vivant~: environ
60 à 65~\% de la masse d'un adulte, 75~\% de celle d'un nouveau-né,
85~\% d'une pomme, 95~\% d'une méduse, mais seulement 13~\% d'un grain
de blé dormant. La teneur en eau d'un échantillon se mesure en le
pesant, en le séchant vers \qty{105}{\celsius} jusqu'à ce que sa masse
cesse de diminuer, puis en le pesant de nouveau~: la masse perdue est
l'eau.
\end{proposition}
\admitted

\begin{example}[Sécher une tranche de pomme]\label{ex:g10:chemistry-of-life:apple}
Une tranche de pomme fraîche de masse \qty{24.0}{g} est séchée jusqu'à
ce que sa masse se stabilise à \qty{3.6}{g}. Eau perdue~:
$24.0 - 3.6 = \qty{20.4}{g}$, soit
$20.4/24.0 = 0.85$, c'est-à-dire 85~\% de la masse fraîche. Les
\qty{3.6}{g} qui restent sont la \emph{matière sèche}~: molécules
organiques et \omterm{def:g10:chemistry-of-life:mineral-organic}{sels minéraux}. Brûlez cette matière sèche~: environ
\qty{0.1}{g} de cendre pâle --- les \omterm{def:g10:chemistry-of-life:mineral-organic}{sels minéraux} --- subsiste~; le
reste, la part organique, est parti en dioxyde de carbone et en vapeur
d'eau.
\end{example}

\begin{proposition}[Ce que fait l'eau]\label{prop:g10:chemistry-of-life:water-roles}
L'eau est le lieu où se fait la chimie d'une cellule~: elle dissout les
sels, les sucres et la plupart des petites molécules organiques, de
sorte qu'ils peuvent se rencontrer et réagir~; elle les transporte d'un
endroit à un autre (le plasma sanguin est à 90~\% d'eau, la sève bien
davantage)~; et comme il faut beaucoup d'énergie pour la chauffer, elle
amortit les changements brusques de température de l'organisme. Une
perte de 10~\% de l'eau du corps est une urgence médicale~; une graine
dormante, à 13~\%, a simplement éteint sa chimie jusqu'au retour de
l'eau.
\end{proposition}
\admitted

\begin{example}[Des sels minéraux que l'on sait nommer]\label{ex:g10:chemistry-of-life:salts}
Le phosphate de calcium rigidifie l'os et la dent (le calcium fait
1.5~\% de la masse du corps, presque tout dans le squelette). Les ions
sodium et chlorure fixent la salinité du sang, proche de
\qty{9}{g/L}~; les ions potassium sont concentrés dans les cellules~;
le fer siège au cœur de l'hémoglobine~; l'iode est nécessaire à la
fabrication de l'hormone de la thyroïde. Un \omterm{def:g10:chemistry-of-life:mineral-organic}{sel minéral} n'est jamais
une source d'énergie, mais un corps privé de l'un d'eux échoue à une
tâche précise.
\end{example}

\section{Les molécules organiques~: quatre familles}

\begin{definition}[Glucides, lipides, protéines, acides nucléiques]\label{def:g10:chemistry-of-life:families}
Les molécules organiques appartiennent à quatre familles.
\begin{itemize}
  \item Les \emph{glucides}\index{glucide} (les sucres) ne contiennent
        que du carbone, de l'hydrogène et de l'oxygène. Les sucres
        simples sont des unités isolées, comme le glucose
        $\mathrm{C_6H_{12}O_6}$~; les glucides complexes sont des
        chaînes de centaines ou de milliers d'unités de glucose ---
        amidon et cellulose chez les végétaux, glycogène chez les
        animaux.
  \item Les \emph{lipides}\index{lipide} (graisses et huiles) ne
        contiennent eux aussi que C, H et O, mais avec bien moins
        d'oxygène, et ils ne se dissolvent pas dans l'eau. Les plus
        répandus sont les triglycérides, faits chacun d'une molécule de
        glycérol liée à trois longues chaînes d'acides gras.
  \item Les \emph{protéines}\index{protéine} sont des chaînes
        d'\emph{acides aminés}\index{acide aminé}~: vingt sortes
        d'acides aminés, tous azotés, liés dans un ordre précis en
        chaînes de cinquante à plusieurs milliers d'unités qui se
        replient en une forme définie.
  \item Les \emph{acides nucléiques}\index{acide nucléique} (ADN et
        ARN) sont des chaînes de nucléotides, contenant du phosphore en
        plus de C, H, O et N~; ils stockent et transmettent
        l'information génétique, et font l'objet du
        \cref{ch:g10:universal-dna}.
\end{itemize}
\end{definition}

\begin{example}[Lire une étiquette alimentaire]\label{ex:g10:chemistry-of-life:label}
L'étiquette d'un pain complet indique, pour \qty{100}{g}~: \omterm{def:g10:chemistry-of-life:families}{glucides}
\qty{41}{g} (dont sucres \qty{3}{g}), matières grasses \qty{3}{g},
\omterm{def:g10:chemistry-of-life:families}{protéines} \qty{9}{g}, fibres \qty{7}{g}, sel \qty{1.1}{g}. Le reste ---
environ \qty{39}{g} --- est de l'eau. Les «~\omterm{def:g10:chemistry-of-life:families}{glucides}~» sont ici surtout
de l'amidon, les «~sucres~» les sucres simples, les «~fibres~» de la
cellulose (un \omterm{def:g10:chemistry-of-life:families}{glucide} que nous ne savons pas digérer), les «~matières
grasses~» les \omterm{def:g10:chemistry-of-life:families}{lipides}, le «~sel~» un \omterm{def:g10:chemistry-of-life:mineral-organic}{sel minéral}. Une étiquette est une
analyse chimique, arrondie.
\end{example}

\begin{proposition}[À quoi sert chaque famille]\label{prop:g10:chemistry-of-life:roles}
Les familles se partagent le travail de la cellule, avec des
recouvrements~:
\begin{itemize}
  \item les \omterm{def:g10:chemistry-of-life:families}{glucides} sont le carburant de tous les jours (le glucose)
        et sa réserve à court terme (amidon, glycogène), et la
        cellulose est le matériau de construction des parois végétales~;
  \item les \omterm{def:g10:chemistry-of-life:families}{lipides} sont la réserve d'énergie compacte à long terme
        (\qty{38}{kJ} par gramme, contre \qty{17}{kJ} par gramme pour
        les \omterm{def:g10:chemistry-of-life:families}{glucides} et les \omterm{def:g10:chemistry-of-life:families}{protéines}), et les \omterm{def:g10:chemistry-of-life:families}{lipides} membranaires
        forment l'enveloppe de chaque cellule~;
  \item les \omterm{def:g10:chemistry-of-life:families}{protéines} sont les machines~: les enzymes qui font tourner
        chaque réaction, les fibres qui contractent le muscle, les
        transporteurs, les anticorps, les récepteurs~;
  \item les \omterm{def:g10:chemistry-of-life:families}{acides nucléiques} détiennent les instructions de
        fabrication des \omterm{def:g10:chemistry-of-life:families}{protéines}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Les grandes molécules et la forme de la vie]\label{rem:g10:chemistry-of-life:macromolecules}
Les molécules de la roche sont petites et répétitives (un cristal de
quartz, c'est $\mathrm{SiO_2}$ répété sans fin). Les molécules
caractéristiques de la vie sont énormes --- une seule \omterm{def:g10:chemistry-of-life:families}{protéine} peut
contenir dix mille atomes, une molécule d'ADN des milliards --- et
\emph{informatives}~: l'ordre de leurs briques porte un message. Rien
dans le monde minéral ne ressemble à une molécule dont la séquence
signifie quelque chose.
\end{remark}

\section{Identifier les constituants}

\begin{method}[Les tests d'identification des familles]\label{met:g10:chemistry-of-life:tests}
Chaque famille possède un test simple qui donne un résultat visible.
\begin{enumerate}
  \item \emph{Eau}~: une bandelette de papier au chlorure de cobalt
        bleu vire au rose~; ou bien sécher l'échantillon et mesurer la
        masse perdue.
  \item \emph{Amidon}~: une goutte d'eau iodée passe du jaune-brun au
        bleu-noir.
  \item \emph{Sucres simples} (glucose, fructose\dots)~: chauffer avec
        la liqueur de Fehling ou le réactif de Benedict~; la solution
        bleue donne un précipité rouge brique.
  \item \emph{\omterm{def:g10:chemistry-of-life:families}{Protéines}}~: le test du biuret --- quelques gouttes de
        sulfate de cuivre en milieu alcalin virent au violet.
  \item \emph{\omterm{def:g10:chemistry-of-life:families}{Lipides}}~: l'échantillon laisse sur le papier une tache
        translucide qui ne sèche pas~; le colorant rouge Soudan III la
        teinte.
  \item \emph{\omterm{def:g10:chemistry-of-life:mineral-organic}{Sels minéraux}}~: brûler complètement la matière sèche~;
        la cendre qui subsiste est minérale. Les ions particuliers
        s'identifient par précipitation (le nitrate d'argent pour le
        chlorure, par exemple).
\end{enumerate}
Faites toujours un témoin à l'eau pure, et un témoin positif connu,
pour être sûr que le réactif fonctionne et qu'un changement de couleur
signifie bien ce que vous croyez.
\end{method}

\begin{example}[Analyser une pomme de terre]\label{ex:g10:chemistry-of-life:potato}
Une tranche de pomme de terre crue~: le papier au chlorure de cobalt
vire au rose (eau)~; l'eau iodée vire au bleu-noir aussitôt (amidon,
abondant)~; la liqueur de Fehling ne donne qu'un soupçon d'orangé (peu
de sucre libre)~; le biuret donne un violet pâle (un peu de \omterm{def:g10:chemistry-of-life:families}{protéines},
environ 2~\% de la masse fraîche)~; aucune tache translucide (presque
pas de \omterm{def:g10:chemistry-of-life:families}{lipides}). Le séchage laisse 22~\% de la masse~; la combustion de
ce résidu en laisse 1~\% de cendre. La pomme de terre est une réserve
d'amidon emballée dans de l'eau --- exactement ce qu'il faut à une
plante pour passer l'hiver sous terre.
\end{example}

\begin{proposition}[Unité de composition]\label{prop:g10:chemistry-of-life:unity}
Quel que soit l'organisme --- bactérie, champignon, plante, animal ---
les mêmes tests trouvent les mêmes quatre familles, faites des mêmes
briques~: les mêmes vingt \omterm{def:g10:chemistry-of-life:families}{acides aminés}, le même glucose, les mêmes
quatre nucléotides. Un être vivant ne se distingue pas en possédant des
atomes ou des briques particuliers, mais par les proportions qu'il
maintient, les très grandes molécules qu'il assemble et l'information
que ces molécules portent. Cette unité chimique est l'un des arguments
les plus forts en faveur d'une origine commune à tous les êtres
vivants, argument que le \cref{ch:g10:common-ancestry} reprendra.
\end{proposition}
\admitted

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:1}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:2}

chlorure de sodium, un triglycéride, phosphate de calcium, un \omterm{def:g10:chemistry-of-life:families}{acide
aminé}, dioxyde de carbone.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:3}

cent~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:4}

(a) de l'amidon, (b) des \omterm{def:g10:chemistry-of-life:families}{protéines}, (c) des \omterm{def:g10:chemistry-of-life:families}{lipides}~? Donner le
résultat attendu dans chaque cas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:5}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:chemistry-of-life:6}

\end{exercise}

\section{Problème~: le carbone d'un être humain}

\begin{problem}[{Problème du week-end --- d'un champ de blé à un corps~:
un bilan de matière, de l'eau d'un grain au nombre de kilogrammes de
carbone qu'une personne porte}]
\label{pb:g10:chemistry-of-life:1}
Une classe suit la matière d'un champ de blé jusqu'au corps de l'un de
ses élèves. Données~: un grain de blé est à 13~\% d'eau, et sa matière
sèche est faite d'environ 70~\% d'amidon, 12~\% de \omterm{def:g10:chemistry-of-life:families}{protéines}, 2~\% de
\omterm{def:g10:chemistry-of-life:families}{lipides}, 2~\% de minéraux, le reste étant de la cellulose. La farine à
pain garde en gros la composition de la partie interne du grain.
L'élève, Léa, a une masse de \qty{60}{kg}~; on prendra la composition
adulte de la deuxième figure (eau 60~\%, \omterm{def:g10:chemistry-of-life:families}{protéines} 17~\%, \omterm{def:g10:chemistry-of-life:families}{lipides}
16~\%, minéraux 6~\%, \omterm{def:g10:chemistry-of-life:families}{glucides} 1~\%). Fractions massiques en éléments
dans chaque famille~: prendre les \omterm{def:g10:chemistry-of-life:families}{glucides} à 40~\% de carbone, les
\omterm{def:g10:chemistry-of-life:families}{lipides} à 77~\% de carbone, les \omterm{def:g10:chemistry-of-life:families}{protéines} à 53~\% de carbone et 16~\%
d'azote.

\medskip
\noindent\textbf{Partie I --- Le grain.}
\begin{enumerate}
  \item Un grain de masse \qty{45}{mg} est séché. Quelle masse d'eau
        perd-il, et quelle est sa masse sèche~?
  \item Calculer les masses d'amidon, de \omterm{def:g10:chemistry-of-life:families}{protéines} et de \omterm{def:g10:chemistry-of-life:families}{lipides} de la
        matière sèche d'un grain.
  \item Quels tests de la \cref{met:g10:chemistry-of-life:tests}
        confirmeraient chacun de ces trois constituants sur un grain
        écrasé~?
  \item Le grain est semé et arrosé~; trois jours plus tard, la
        plantule est à 88~\% d'eau. Expliquer pourquoi la germination
        ne peut pas démarrer à 13~\%.
  \item Un hectare produit \qty{8}{t} de grain. Quelle masse d'eau
        est récoltée avec lui, et quelle masse de matière sèche~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le pain.}
\begin{enumerate}[resume]
  \item Un pain est cuit à partir de \qty{500}{g} de farine,
        \qty{320}{g} d'eau et \qty{10}{g} de sel~; il perd
        \qty{130}{g} d'eau au four. Quelle est la masse du pain cuit,
        et sa teneur en eau en pour cent~?
  \item En prenant pour la farine la composition sèche du grain,
        estimer les masses d'amidon et de \omterm{def:g10:chemistry-of-life:families}{protéines} du pain.
  \item En prenant \qty{17}{kJ/g} pour les \omterm{def:g10:chemistry-of-life:families}{glucides} et les \omterm{def:g10:chemistry-of-life:families}{protéines}
        et \qty{38}{kJ/g} pour les \omterm{def:g10:chemistry-of-life:families}{lipides}, estimer le contenu
        énergétique du pain.
  \item Léa mange \qty{120}{g} de ce pain. Quelle fraction d'un besoin
        quotidien d'environ \qty{10000}{kJ} cela couvre-t-il~?
  \item Pourquoi le sel du pain n'est-il pas une source d'énergie, et
        à quoi sert-il dans le corps~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le corps~: l'eau et les familles.}
\begin{enumerate}[resume]
  \item Calculer la masse d'eau du corps de Léa.
  \item Calculer les masses de \omterm{def:g10:chemistry-of-life:families}{protéines}, de \omterm{def:g10:chemistry-of-life:families}{lipides}, de minéraux et
        de \omterm{def:g10:chemistry-of-life:families}{glucides}.
  \item Léa perd \qty{1.2}{kg} par sudation au cours d'une longue
        course. Quelle fraction de l'eau de son corps cela
        représente-t-il~? Pourquoi faut-il la remplacer vite~?
  \item Son sang, environ \qty{4.5}{L}, est à 55~\% de plasma, et le
        plasma est à 90~\% d'eau. Estimer la masse d'eau de son sang
        (on prendra \qty{1}{kg} par litre). Quelle fraction de l'eau
        de son corps cela fait-il~? Où est le reste~?
  \item Un nouveau-né est à 75~\% d'eau. Expliquer en une phrase
        pourquoi un nourrisson est plus vulnérable à la déshydratation
        qu'un adulte.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Le corps~: les éléments.}
\begin{enumerate}[resume]
  \item À l'aide des fractions de carbone de chaque famille, calculer
        la masse de carbone des \omterm{def:g10:chemistry-of-life:families}{glucides}, des \omterm{def:g10:chemistry-of-life:families}{lipides} et des \omterm{def:g10:chemistry-of-life:families}{protéines}
        de Léa, puis la masse totale de carbone.
  \item Quelle fraction de sa masse corporelle est du carbone~?
        Comparer aux 18.5~\% de la
        \cref{prop:g10:chemistry-of-life:four} et commenter l'accord.
  \item Calculer la masse d'azote de ses \omterm{def:g10:chemistry-of-life:families}{protéines}. D'où venait
        finalement cet azote~?
  \item L'eau est à 89~\% d'oxygène en masse. Calculer la masse
        d'oxygène portée par la seule eau de son corps, et comparer
        aux 65~\% de la \cref{prop:g10:chemistry-of-life:four}. Quelle
        molécule explique l'essentiel de l'oxygène du corps~?
  \item Énoncer le résultat en une phrase~: combien de kilogrammes de
        carbone une personne de \qty{60}{kg} porte-t-elle, et par
        quels deux processus biologiques ce carbone a-t-il quitté
        l'air pour entrer dans le corps~?
\end{enumerate}
\end{problem}
