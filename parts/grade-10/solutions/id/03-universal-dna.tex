\begin{solution}{exo:g10:universal-dna:5}

\begin{solution}{exo:g10:universal-dna:8}
Pecahan yang ``murni'' masih mungkin memuat jejak molekul lain,
sehingga kegiatannya sendiri tidak membuktikan bahwa DNA-lah yang
bertanggung jawab. Merusak \omterm{def:g10:universal-dna:information}{DNA} secara khusus, dalam sari yang lengkap,
lalu melihat kegiatannya lenyap sedangkan enzim perusak \omterm{def:g10:chemistry-of-life:families}{protein} dan RNA
membiarkannya utuh, menyematkan akibatnya pada \omterm{def:g10:universal-dna:information}{DNA} dan bukan pada yang lain.
\end{solution}

\begin{solution}{exo:g10:universal-dna:9}
Pengulangan yang tetap akan memberi komposisi yang sama (25\% tiap
\omterm{def:g10:universal-dna:nucleotide}{basa}) pada setiap spesies. Chargaff menemukan A+T yang merentang luas
antarspesies: urutan \omterm{def:g10:universal-dna:nucleotide}{basanya} bukan pola yang tetap, sehingga urutannya
bebas berubah --- dan itulah yang membuatnya sanggup membawa keterangan.
\end{solution}

\begin{solution}{exo:g10:universal-dna:10}
Urutannya dibaca berturut-turut untuk membangun \omterm{def:g10:chemistry-of-life:families}{proteinnya}; \omterm{def:g10:universal-dna:nucleotide}{basa} yang
berubah pada kedudukan yang menetapkan sebuah \omterm{def:g10:chemistry-of-life:families}{asam amino} dapat
menggantikan satu \omterm{def:g10:chemistry-of-life:families}{asam amino} dengan yang lain dan mengubah bentuk atau
kegiatan \omterm{def:g10:chemistry-of-life:families}{proteinnya}. Jika \omterm{def:g10:universal-dna:nucleotide}{basa} yang berubah itu terletak pada bagian
\omterm{def:g10:universal-dna:gene}{gen} yang tidak dibaca, atau menetapkan \omterm{def:g10:chemistry-of-life:families}{asam amino} yang sama, \omterm{def:g10:chemistry-of-life:families}{proteinnya} tidak berubah.
\end{solution}

\begin{solution}{exo:g10:universal-dna:11}
Seluruh \omterm{def:g10:cells-common-unit:cell}{sel} mencit itu berasal lewat pembelahan dari \omterm{def:g10:cells-common-unit:cell}{sel} telur yang
telah dibuahi; pada tiap pembelahan \omterm{def:g10:universal-dna:information}{DNA-nya}, termasuk transgennya,
disalin (\omterm{prop:g10:universal-dna:helix}{komplementaritas}) dan dibagi. Karena itu \omterm{def:g10:cells-common-unit:cell}{sel} kulit, \omterm{def:g10:cells-common-unit:cell}{sel} hati
dan \omterm{def:g10:cells-common-unit:cell}{sel} spermanya semuanya membawa \omterm{def:g10:universal-dna:gene}{gen} itu --- dan spermanya
mewariskannya kepada keturunan berikutnya.
\end{solution}

\begin{solution}{exo:g10:universal-dna:12}
$\num{4.6e6} \times \qty{0.34}{nm} \approx \qty{1.6}{mm}$, sekitar 800
kali panjang \omterm{def:g10:cells-common-unit:cell}{selnya}. Tetapi molekulnya hanya selebar \qty{2}{nm}:
volumenya, sekitar $\pi \times 1^2 \times \num{1.6e6}\,
\unit{nm^3} \approx \num{5e-3}\,\unit{\micro m^3}$, sekitar 1\% volume
\omterm{def:g10:cells-common-unit:cell}{selnya}; setelah terlipat dan tergulung, ia muat dengan ruang bersisa.
\end{solution}

\begin{solution}{exo:g10:universal-dna:13}
\omterm{def:g10:universal-dna:information}{DNA} manusia 700 kali lebih panjang tetapi hanya membawa sekitar 5 kali
lebih banyak \omterm{def:g10:universal-dna:gene}{gen}: sebagian besarnya bukan \omterm{def:g10:universal-dna:gene}{gen}. Sebagian \omterm{def:g10:universal-dna:information}{DNA} berlebih
itu mengatur kapan \omterm{def:g10:universal-dna:gene}{gennya} dipakai, sebagian lagi berupa urutan berulang
tanpa fungsi yang dikenal. Cacah \omterm{def:g10:universal-dna:nucleotide}{basa} mengukur besarnya teks, bukan
kerumitan organismenya; jumlah \omterm{def:g10:universal-dna:gene}{gen} dan, terutama, cara \omterm{def:g10:universal-dna:gene}{gen} itu
digabungkan dan diatur lebih menentukan.
\end{solution}

\begin{solution}{exo:g10:universal-dna:14}
Ia akan membantah kesemestaan \omterm{def:g10:universal-dna:information}{DNA}
(\cref{prop:g10:universal-dna:universal}). Karena molekul dan sandi
yang sama diwariskan dari nenek moyang bersama, spesies semacam itu
tentu berasal dari asal usul kehidupan yang terpisah: \omterm{def:g10:universal-dna:gene}{gennya} tidak akan
dapat dibaca \omterm{def:g10:cells-common-unit:cell}{sel} kita dan \omterm{prop:g10:universal-dna:universal}{transgenesis} antara kedua garis keturunan itu
tidak akan mungkin.
\end{solution}

\begin{solution}{exo:g10:universal-dna:15}

Lembaga pengatur tetap akan menanyakan apakah \omterm{def:g10:chemistry-of-life:families}{protein} yang telah

\end{solution}

\begin{solution}{pb:g10:universal-dna:1}
\textbf{1.} $720 \times 0.34 \approx \qty{245}{nm}$; $720/10 = 72$
putaran.

\textbf{2.} \texttt{TACTCATTTCCTCTTCTTG}.

\textbf{3.} T $= 27\%$; G $=$ C $= (100 - 54)/2 = 23\%$.

\textbf{4.} $720/238 \approx 3.0$: tiga \omterm{def:g10:universal-dna:nucleotide}{basa} tiap \omterm{def:g10:chemistry-of-life:families}{asam amino} (tiga yang
terakhir menandai akhir rantainya).

\textbf{5.} $4^{720}$ urutan, sekitar $10^{433}$. Kimianya membolehkan
urutan apa pun, sehingga urutan tertentu yang terpilih di antara semua
kemungkinan itu adalah sebuah pesan --- yaitu keterangan.

\textbf{6.} Kesemestaannya: setiap makhluk hidup memakai empat
\omterm{def:g10:universal-dna:nucleotide}{nukleotida} yang sama dan membaca urutannya menurut aturan yang sama.

\textbf{7.} Seluruh \omterm{def:g10:cells-common-unit:cell}{sel} mencit itu berasal dari \omterm{def:g10:cells-common-unit:cell}{sel} telurnya lewat
pembelahan, sehingga \omterm{def:g10:universal-dna:gene}{gen} yang disisipkan pada \omterm{def:g10:cells-common-unit:cell}{sel} telur tersalin ke
dalam setiap \omterm{def:g10:cells-common-unit:cell}{sel}. Jika disisipkan ke dalam kulit dewasa, \omterm{def:g10:universal-dna:gene}{gen} itu hanya
akan hadir pada \omterm{def:g10:cells-common-unit:cell}{sel} kulit yang diperlakukan dan keturunannya.

\textbf{8.} Setiap jenis \omterm{def:g10:cells-common-unit:cell}{sel} memuat seluruh \omterm{def:g10:universal-dna:information}{DNA-nya}, termasuk
transgennya: spesialisasi tidak membuang \omterm{def:g10:universal-dna:gene}{gen}.

\textbf{9.} Ya: \omterm{def:g10:cells-common-unit:cell}{sel} kelaminnya membawa \omterm{def:g10:universal-dna:gene}{gen} itu. Jika mencitnya punya
satu salinan pada satu \omterm{def:g10:universal-dna:information}{kromosom} dari sepasang, sekitar separuh
keturunannya mewarisinya dan berpendar.

\textbf{10.} \omterm{def:g10:chemistry-of-life:families}{Proteinnya} bukan keterangan: ia terpakai habis dan tidak
pernah diperbarui, sehingga pendarnya memudar. \omterm{def:g10:universal-dna:gene}{Gennya} disalin pada tiap
pembelahan dan dibaca terus-menerus, sehingga pendarnya tetap dan diwariskan.

\textbf{11.} Pecahan \omterm{def:g10:universal-dna:information}{DNA-nya}; perlakuan dengan enzim yang merusak \omterm{def:g10:universal-dna:information}{DNA}.

\textbf{12.} Bakterinya menyerap \omterm{def:g10:universal-dna:information}{DNA} dari galur lain, memadukannya
serta mengungkapkan dan mewariskan sifatnya: sebuah \omterm{def:g10:universal-dna:gene}{gen} berpindah dari
satu organisme ke organisme lain, dan itulah yang dilakukan
\omterm{prop:g10:universal-dna:universal}{transgenesis} dengan sengaja.

\textbf{13.} A$\,=\,$T dan G$\,=\,$C menyokong pemasangannya; bagian
A+T yang berubah-ubah menunjukkan bahwa urutan pasangannya bebas.

\textbf{14.} A dan G adalah \omterm{def:g10:universal-dna:nucleotide}{basa} besar bercincin dua, T dan C \omterm{def:g10:universal-dna:nucleotide}{basa}
kecil bercincin satu: \omterm{def:g10:universal-dna:nucleotide}{basa} besar yang berpasangan dengan \omterm{def:g10:universal-dna:nucleotide}{basa} kecil
selalu merentang jarak yang sama, sehingga kedua tulang punggungnya
tetap berjarak \qty{2}{nm} di sepanjang molekulnya. Dua \omterm{def:g10:universal-dna:nucleotide}{basa} besar akan
menggembung, dua \omterm{def:g10:universal-dna:nucleotide}{basa} kecil akan menyisakan celah.

\textbf{15.} Karena tiap untai adalah komplemen untai yang lain:
pisahkan keduanya, bangun satu pasangan pada masing-masing menurut
aturan pemasangannya, dan terhasillah dua molekul yang serupa.

\textbf{16.} $2 \times \num{2.7e9} \times \qty{0.34}{nm} \approx
\qty{1.8}{m}$.

\textbf{17.} $\qty{1.8}{m} / \qty{6}{\micro m} = \num{3e5}$:
pemampatan tiga ratus ribu kali lipat.

\textbf{18.} $720 / \num{5.4e9} \approx \num{1.3e-7}$: sekitar
sepersepuluh juta \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:cells-common-unit:cell}{selnya}.

\textbf{19.} $\num{2e4} \times \num{3e4} = \num{6e8}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa},
sekitar 22\% genomnya; sisanya terletak di antara \omterm{def:g10:universal-dna:gene}{gennya}.

\textbf{20.} \omterm{def:g10:cells-common-unit:cell}{Sel} mencit membawa sekitar \qty{1.8}{m} \omterm{def:g10:universal-dna:information}{DNA}; karena \omterm{def:g10:universal-dna:information}{DNA}
itu dibangun dan dibaca persis seperti \omterm{def:g10:universal-dna:information}{DNA} ubur-ubur, satu \omterm{def:g10:universal-dna:gene}{gen} asing di
antara dua puluh ribu \omterm{def:g10:universal-dna:gene}{gennya} dibaca seperti \omterm{def:g10:universal-dna:gene}{gen} yang lain.
\end{solution}
