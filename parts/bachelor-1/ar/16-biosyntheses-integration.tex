\chapter{التراكيب الحيوية والخلية المتكاملة}\label{ch:b1:biosyntheses-integration}

في الثامنة صباحًا يحوّل كبد سكر الفطور إلى \omterm{def:b1:carbohydrates:polysaccharide}{غليكوجين} \omterm{def:b1:lipids:triglyceride}{وشحم}؛ وفي الثالثة
من صباح اليوم التالي يصنع الكبد نفسه سكرًا من الحموض الأمينية للعضل
ويصبه في الدم لدماغ لم يأكل شيئًا منذ تسع ساعات. ولم يتغير شيء في
\omterm{def:b1:enzymes:enzyme}{الإنزيمات}؛ وإنما تغير أيها مشغَّل. لقد فككت الفصول السابقة جزيئات
\omterm{def:b1:cell-unit-of-life:cell}{الخلية}؛ وهذا الفصل يركّبها --- تراكيب السكريات \omterm{def:b1:carbohydrates:polysaccharide}{والغليكوجين} والشحوم
والحموض الأمينية --- ثم يسأل كيف تقرر \omterm{def:b1:cell-unit-of-life:cell}{خلية}، \omterm{def:b1:organism-environment:organism}{وكائن حي}، في كل لحظة أي
السبل يجري. والجواب مجموعة صغيرة من العملات المشتركة، وبضعة \omterm{def:b1:enzymes:enzyme}{إنزيمات}
عند مفترقات الطرق، وهرمونات تخبر كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} بحال الكل.

\section{ما الذي يحتاج إليه التركيب}

\begin{proposition}[متطلبات البناء الثلاثة]\label{prop:b1:biosyntheses-integration:needs}
\index{البناء الحيوي}
كل تركيب حيوي يحتاج إلى \emph{هياكل كربونية}، تُسحب من حفنة من وسائط
\omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري} \omterm{def:b1:respiration-fermentation:krebs}{ودورة كريبس}؛ وإلى \emph{قدرة مرجعة}، تُزوَّد في هيئة
\omterm{def:b1:biosyntheses-integration:ppp}{NADPH}؛ وإلى \emph{طاقة}، تُزوَّد في هيئة \omterm{def:b1:water-small-molecules:atp}{ATP}. ولذلك يلتقي الهدم
والبناء عند مفترقات الطرق نفسها --- غلوكوز-6-فوسفات، وفوسفات السكر
الثلاثي، والبيروفات، \omterm{def:b1:respiration-fermentation:pdh}{وأسيتيل-CoA}، و$\alpha$-كيتوغلوتارات،
والأوكسالوأسيتات --- لكنهما يجريان على \omterm{def:b1:enzymes:enzyme}{إنزيمات} منفصلة عند الخطوات
اللاعكوسة، حتى يمكن تشغيل كل اتجاه على حدة، وهما يستعملان مرافقين
مختلفين لإلكتروناتهما: NADH، مُبقًى مؤكسدًا، لأخذ الإلكترونات في
الهدم؛ \omterm{def:b1:biosyntheses-integration:ppp}{وNADPH}، مُبقًى مرجعًا، لتسليمها في التركيب.
\end{proposition}

\begin{definition}[سبيل فوسفات الخماسيات]\label{def:b1:biosyntheses-integration:ppp}
\index{سبيل فوسفات الخماسيات}\index{NADPH}
\emph{سبيل فوسفات الخماسيات} في \omterm{def:b1:eukaryotic-cell:organelle}{العصارة الخلوية} يؤكسد
غلوكوز-6-فوسفات إلى ريبولوز-5-فوسفات و$\mathrm{CO_2}$، فيرجع
$\mathrm{NADP^+}$ اثنين إلى NADPH؛ ثم يحوّل فرعه غير المؤكسد فوسفات
السكريات الخماسية والرباعية والسداسية والسباعية بعضها إلى بعض، فتستطيع
\omterm{def:b1:cell-unit-of-life:cell}{الخلية} أن تصنع ريبوز-5-فوسفات \omterm{def:b1:nucleic-acids:nucleotide}{للنوكليوتيدات} حين تحتاج إلى خماسيات،
وNADPH حين تحتاج إلى قدرة مرجعة (تركيب \omterm{def:b1:lipids:triglyceride}{الشحم}، والدفاع عن المؤكسدات)،
أو الأمرين معًا، وأن تعيد الباقي إلى \omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري}. وهو المصدر الرئيس
لمركب NADPH في \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الحيوانية؛ أما في النبات فتزود \omterm{prop:b1:photosynthesis:lightreactions}{التفاعلات الضوئية} في
\omterm{def:b1:eukaryotic-cell:plastid}{البلاستيدة} NADPH نهارًا.
\end{definition}

\section{صنع الغلوكوز: استحداث السكر}

\begin{definition}[استحداث السكر]\label{def:b1:biosyntheses-integration:gluconeogenesis}
\index{استحداث السكر}
\emph{استحداث السكر} يصنع الغلوكوز من طلائع غير كربوهيدراتية ---
اللاكتات، والبيروفات، والغليسرول، والهياكل الكربونية لمعظم الحموض
الأمينية --- في الكبد (وقليلًا في الكلية). وهو يجري \omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري}
بالمقلوب عبر خطواته العكوسة السبع ويتجاوز اللاعكوسة الثلاث \omterm{def:b1:enzymes:enzyme}{بإنزيمات}
مختلفة: فيُكربكس البيروفات إلى أوكسالوأسيتات (في \omterm{def:b1:eukaryotic-cell:mitochondrion}{المتقدرة}، بجزيء \omterm{def:b1:water-small-molecules:atp}{ATP}
واحد) ويُنزع كربوكسيله إلى فوسفوإينول بيروفات (بجزيء GTP واحد)؛
ويُحلمه فركتوز-1،6-ثنائي الفوسفات إلى فركتوز-6-فوسفات؛ ويُحلمه
غلوكوز-6-فوسفات إلى غلوكوز حر يغادر \omterm{def:b1:cell-unit-of-life:cell}{الخلية}. ولكل غلوكوز: 6 مكافئات
\omterm{def:b1:water-small-molecules:atp}{ATP} و2 NADH، مقابل \omterm{def:b1:water-small-molecules:atp}{ATP} اثنين يعطيهما \omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري} --- وهو ثمن إجراء
سبيل هبوطي صعودًا.
\end{definition}

\begin{proposition}[ما يستطيع أن يصير غلوكوزًا وما لا يستطيع]\label{prop:b1:biosyntheses-integration:cannot}
اللاكتات والغليسرول والحموض الأمينية التي تعطي بيروفاتًا أو وسائط
\omterm{def:b1:respiration-fermentation:krebs}{لدورة كريبس} (وهي كلها إلا اللوسين واللايسين) يمكن تحويلها إلى غلوكوز.
أما \omterm{def:b1:lipids:fattyacid}{الحموض الدسمة} فلا، عند الحيوان: فأكسدتها $\beta$ تعطي
\omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA}، وكربوناه يدخلان الدورة ويغادران في هيئة $\mathrm{CO_2}$
اثنين قبل أن يُكتسب أي أوكسالوأسيتات، فلا طريق صافيًا من \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA}
إلى السكر. أما النباتات والفطريات والبكتيريا فتملك \emph{دورة
الغليوكسيلات}، وهي طريق مختصر يتخطى نزعي الكربوكسيل ويتيح لبذرة زيتية
منبتة أن تحول شحمها إلى السكر الذي تحتاجه بادرتها. أما الثديي الجائع
فلا بد له من صنع غلوكوزه من \omterm{def:b1:proteins:peptide}{البروتين}.
\end{proposition}

\begin{example}[دورة كوري]\label{ex:b1:biosyntheses-integration:cori}
عضلة تعدو تخمّر \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} إلى لاكتات تدخل الدم؛ فيأخذ الكبد
اللاكتات، ويؤكسدها إلى بيروفات، ويصنع منها غلوكوزًا بكلفة 6 \omterm{def:b1:water-small-molecules:atp}{ATP} لكل
غلوكوز، ويعيد الغلوكوز إلى الدم لتستعمله العضلة من جديد. فتكون العضلة
قد كسبت \omterm{def:b1:water-small-molecules:atp}{ATP} اثنين لكل غلوكوز بلا أكسجين ومررت فاتورة قدرها 6 إلى
الكبد، الذي يدفعها بالأكسجين على مهل: و«دين الأكسجين» في العدو
السريع يُسدَّد في الكبد أساسًا.
\end{example}

\section{الادخار: الغليكوجين والشحم}

\begin{definition}[تركيب الغليكوجين]\label{def:b1:biosyntheses-integration:glycogensynthesis}
يُدخَّر الغلوكوز غليكوجينًا بتنشيط غلوكوز-6-فوسفات إلى
\emph{غلوكوز-UDP} (بجزيء UTP واحد)، يضيفه \emph{سنتاز \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين}} إلى
الأطراف غير المرجعة في الشجرة القائمة؛ و\emph{\omterm{def:b1:enzymes:enzyme}{إنزيم} التفريع} يقص
سلاسل قصيرة ويعيد وصلها ليصنع تفرعات $\alpha(1{\to}6)$
(\cref{ch:b1:carbohydrates}). والسنتاز والفسفوريلاز ينظمهما الإشارات
نفسها في اتجاهين متعاكسين: فالفسفرة التي تشغّل الفسفوريلاز تطفئ
السنتاز، فلا تُبنى الشجرة وتُفكك في آن.
\end{definition}

\begin{definition}[تركيب الحموض الدسمة]\label{def:b1:biosyntheses-integration:fasynthesis}
\index{تركيب الحموض الدسمة}
تُصنع \omterm{def:b1:lipids:fattyacid}{الحموض الدسمة} في \omterm{def:b1:eukaryotic-cell:organelle}{العصارة الخلوية} من \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA} المصدَّر من
\omterm{def:b1:eukaryotic-cell:mitochondrion}{المتقدرة}. \emph{وكربوكسيلاز \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA}}، وهو الخطوة المنظَّمة،
يكربكسه (بجزيء \omterm{def:b1:water-small-molecules:atp}{ATP} واحد) إلى مالونيل-CoA؛ ثم يضيف \emph{سنتاز \omterm{def:b1:lipids:fattyacid}{الحموض
الدسمة}} كربونين كربونين من مالونيل-CoA إلى سلسلة نامية، ويرجع كل
إضافة بجزيئي \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}، حتى يتحرر البالميتات (C16): أي 8 \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA}، و7
\omterm{def:b1:water-small-molecules:atp}{ATP}، و14 \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} لكل بالميتات. والاستطالة ونزع الإشباع في \omterm{def:b1:eukaryotic-cell:endomembrane}{الشبكة
الإندوبلازمية} يصنعان الحموض الأخرى؛ والأسترة بغليسرول-3-فوسفات (من
فوسفات السكر الثلاثي) تصنع ثلاثيات الغليسيريد. وهذا السبيل ليس أكسدة
$\beta$ بالمقلوب: \omterm{def:b1:enzymes:enzyme}{فالإنزيمات} مختلفة، والحجيرة مختلفة، ومالونيل-CoA
بدل \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA}، \omterm{def:b1:biosyntheses-integration:ppp}{وNADPH} بدل $\mathrm{FADH_2}$ وNADH --- ومالونيل-CoA
يسد نقل \omterm{def:b1:lipids:fattyacid}{الحموض الدسمة} إلى \omterm{def:b1:eukaryotic-cell:mitochondrion}{المتقدرة}، فلا تصنع \omterm{def:b1:cell-unit-of-life:cell}{الخلية} \omterm{def:b1:lipids:triglyceride}{الشحم} وتحرقه في
آن.
\end{definition}

\begin{example}[شحم من سكر]\label{ex:b1:biosyntheses-integration:fatfromsugar}
كبد يُعطى غلوكوزًا أكثر مما يستطيع ادخاره غليكوجينًا يصنع شحمًا:
غلوكوز إلى بيروفات إلى \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA} (فيفقد ثلث الكربون في هيئة
$\mathrm{CO_2}$ ويكسب \omterm{def:b1:water-small-molecules:atp}{ATP})، ثم \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA} إلى بالميتات بكلفة ذلك
\omterm{def:b1:water-small-molecules:atp}{ATP} \omterm{def:b1:biosyntheses-integration:ppp}{وNADPH} من \omterm{def:b1:biosyntheses-integration:ppp}{سبيل فوسفات الخماسيات}، ثم بالميتات إلى ثلاثي غليسيريد
يُشحن إلى \omterm{def:b1:body-plans-tissues:tissue}{النسيج} الشحمي. وتضيع نحو \qty{15}{\%} من طاقة السكر في
التحويل؛ ويُدخَّر الباقي أكثف بتسع مرات من \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين}
(\cref{ch:b1:lipids}). أما العكس --- \omterm{def:b1:lipids:triglyceride}{شحم} إلى سكر --- فمستحيل.
\end{example}

\section{النتروجين: الحموض الأمينية والنوكليوتيدات}

\begin{definition}[استيعاب النتروجين ونقل الأمين]\label{def:b1:biosyntheses-integration:nitrogen}
\index{نقل الأمين}\index{غلوتامات}
يدخل الأمونيوم المادة العضوية عبر \emph{الغلوتامات} كله تقريبًا:
فنازعة هيدروجين الغلوتامات تضيفه إلى $\alpha$-كيتوغلوتارات، وسنتيتاز
الغلوتامين يضيف ثانيًا إلى \omterm{def:b1:proteins:aminoacid}{السلسلة الجانبية} للغلوتامات (بجزيء \omterm{def:b1:water-small-molecules:atp}{ATP} واحد)،
فيصنع الغلوتامين. ومن هذين المانحين تنقل \emph{ناقلات الأمين}
(\omterm{def:b1:enzymes:enzyme}{بمرافق إنزيمي} هو فيتامين B$_6$) الزمرة الأمينية إلى أي \omterm{def:b1:water-small-molecules:ph}{حمض}
$\alpha$-كيتوني، فتبني الحموض الأمينية من الهياكل الكربونية في السبل
المركزية: فعائلة الغلوتامات من $\alpha$-كيتوغلوتارات، وعائلة
الأسبارتات من الأوكسالوأسيتات، والألانين والسيرين من البيروفات
و3-فوسفوغليسيرات. والنباتات والبكتيريا تصنع العشرين كلها؛ أما
الحيوانات فقد فقدت السبل الطويلة إلى تسعة منها، وهي الحموض الأمينية
\emph{الأساسية}، وتأخذها من الغذاء. وناقلات الأمين نفسها، مجراةً
بالمقلوب، تنزع النتروجين من فائض الحموض الأمينية لطرحه نشادرًا
(الأسماك) أو بولةً (الثدييات) أو \omterm{def:b1:water-small-molecules:ph}{حمض} بول (الطيور والحشرات).
\end{definition}

\begin{example}[النوكليوتيدات]\label{ex:b1:biosyntheses-integration:nucleotides}
تُركَّب حلقة \omterm{def:b1:nucleic-acids:nucleotide}{البورين} على ريبوز-5-فوسفات من غليسين، وأسبارتات،
وغلوتامينين، ووحدتين أحاديتي الكربون، و$\mathrm{CO_2}$، بكلفة ستة
\omterm{def:b1:water-small-molecules:atp}{ATP}؛ ويُركَّب \omterm{def:b1:nucleic-acids:nucleotide}{البيريميدين} من أسبارتات وفوسفات الكرباميل. وتُصنع
\omterm{def:b1:nucleic-acids:nucleotide}{النوكليوتيدات} منقوصة الأكسجين من \omterm{def:b1:nucleic-acids:nucleotide}{النوكليوتيدات} الريبية بإرجاع السكر،
باستعمال \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}. \omterm{def:b1:cell-unit-of-life:cell}{والخلية} المقبلة على الانقسام تصنع نحو $10^{10}$

\end{example}

\begin{proposition}[التغذي والصيام]\label{prop:b1:biosyntheses-integration:fedfasting}
\index{أنسولين}\index{غلوكاغون}
تتقاسم \omterm{def:b1:mammal-organization:organ}{الأعضاء} العمل. فبعد وجبة يخبر \emph{الأنسولين} الآتي من
البنكرياس الكبدَ أن يدخر الغلوكوز غليكوجينًا ويحول الفائض شحمًا،
والعضلاتِ أن تأخذ الغلوكوز وتدخر \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين}، \omterm{def:b1:body-plans-tissues:tissue}{والنسيج} الشحمي أن يأخذ
\omterm{def:b1:lipids:triglyceride}{الشحم}؛ فيعود غلوكوز الدم، وقد ارتفع إلى \qty{8}{mmol/L}، إلى
\qty{5}{} في ساعتين. وبين الوجبات يخبر \emph{الغلوكاغون} الكبدَ أن
يفكك غليكوجينه، ثم، كلما قلّ، أن يصنع الغلوكوز من اللاكتات والغليسرول
والحموض الأمينية، والنسيجَ الشحمي أن يطلق \omterm{def:b1:lipids:fattyacid}{الحموض الدسمة}، فتحرقها
العضلات والكبد بدل الغلوكوز؛ أما الدماغ، الذي لا يستطيع حرق \omterm{def:b1:lipids:triglyceride}{الشحم}،
فيحتفظ بمقدار \qty{120}{g} من الغلوكوز في اليوم. وفي صيام أيام يحول الكبد
\omterm{def:b1:lipids:fattyacid}{الحموض الدسمة} إلى \emph{أجسام كيتونية}، وهي حموض صغيرة يستطيع الدماغ
استعمالها، فيهبط الطلب على الغلوكوز --- وعلى \omterm{def:b1:proteins:peptide}{البروتين} الذي يصنعه ---
بمقدار الثلثين. أما الآليات الهرمونية فتنتمي إلى كتاب السنة الثانية؛
والمنطق الاستقلابي منطق هذا الفصل.
\end{proposition}

\begin{method}[قراءة حالة استقلابية]\label{met:b1:biosyntheses-integration:state}
\begin{enumerate}
  \item اسأل أي وقود وفير: فالغلوكوز المرتفع يعني الادخار
        (غليكوجينًا، ثم شحمًا)؛ والغلوكوز المنخفض يعني الحشد.
  \item تتبع الإشارات: فالأنسولين ينشّط السنتازات ويعطّل الفسفوريلاز
        والليباز؛ والغلوكاغون يفعل العكس --- وأساسًا بفسفرة \omterm{def:b1:enzymes:enzyme}{الإنزيمات}
        نفسها (\cref{ch:b1:enzymes}).
  \item تحقق من مفترقات الطرق: فمركب \omterm{def:b1:water-small-molecules:atp}{ATP} والسترات يسدان PFK ويفتحان
        \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{استحداث السكر}؛ وAMP يفعل العكس؛ ومالونيل-CoA يسد حرق \omterm{def:b1:lipids:triglyceride}{الشحم} ما
        دام \omterm{def:b1:lipids:triglyceride}{الشحم} يُصنع؛ \omterm{def:b1:respiration-fermentation:pdh}{وأسيتيل-CoA} ينشّط الكربوكسيلاز الذي يبدأ
        \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{استحداث السكر}.
  \item انسب \omterm{def:b1:mammal-organization:organ}{الأعضاء}: الكبد (يدخر الغلوكوز ويصنعه ويصدّره؛ ويصنع
        الكيتونات والبولة)، والعضلة (تدخر \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} لنفسها، وتصدّر
        اللاكتات والألانين)، \omterm{def:b1:body-plans-tissues:tissue}{والنسيج} الشحمي (يدخر \omterm{def:b1:lipids:triglyceride}{الشحم} ويطلقه)،
        والدماغ (يحرق الغلوكوز، ثم الكيتونات).
\end{enumerate}
\end{method}

\section{النبات المتكامل}

\begin{proposition}[المنبع والمصب]\label{prop:b1:biosyntheses-integration:sourcesink}
نهارًا تصدّر بلاستيدات \omterm{def:b1:flowering-plant-organization:organs}{الورقة} الخضراء فوسفات السكر الثلاثي إلى
\omterm{def:b1:eukaryotic-cell:organelle}{العصارة الخلوية}، حيث يُصنع منه \emph{سكروز} يُصدَّر في اللحاء
(\cref{ch:b1:plant-transport}) إلى \omterm{def:b1:flowering-plant-organization:organs}{الجذور} والثمار والقمم النامية؛ وما
لا يستطيع اللحاء أخذه يُحفظ في \omterm{def:b1:eukaryotic-cell:plastid}{البلاستيدة} \emph{\omterm{def:b1:carbohydrates:polysaccharide}{نشا}}، وهو مخزن عابر،
ويُحشد ليلًا لإبقاء السكروز جاريًا. وحيث يقرر كبد الثديي بين
\omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين} والتصدير بالهرمونات، تقرر \omterm{def:b1:flowering-plant-organization:organs}{الورقة} بين \omterm{def:b1:carbohydrates:polysaccharide}{النشا} والسكروز بمستوى
الفوسفات في العصارة، وهو يتحكم في مصدّر غلاف \omterm{def:b1:eukaryotic-cell:plastid}{البلاستيدة}. فالنبات ينظم
بتراكيز مستقلباته هو، عضوًا عضوًا؛ والثديي يضيف أمرًا عصبيًا وهرمونيًا
على الكل.
\end{proposition}

\section{مسألة: أربع وعشرون ساعة من حياة إنسان}

\begin{problem}[{مسألة نهاية الأسبوع --- يوم بلا فطور: المخازن موزونة،
والدماغ مغذى، والبروتين معدود، والشحم محشود، وصولًا إلى ساعات الصيام
التي يغطيها غليكوجين الكبد}]
\label{pb:b1:biosyntheses-integration:1}
بالغ كتلته \qty{70}{kg} ينفق \qty{8.4}{MJ} في اليوم (أي \qty{100}{W}
بالمتوسط) ويحمل: \omterm{def:b1:carbohydrates:polysaccharide}{غليكوجين} كبد \qty{100}{g}، \omterm{def:b1:carbohydrates:polysaccharide}{وغليكوجين} عضل

\noindent\textbf{الجزء الأول --- المخازن.}

  \item بعد وجبة يكون \omterm{def:b1:carbohydrates:polysaccharide}{غليكوجين} الكبد ممتلئًا وغلوكوز الدم ما زال
        يرتفع. سمِّ المصير التالي للغلوكوز والهرمون الذي يوجهه.
\end{enumerate}
