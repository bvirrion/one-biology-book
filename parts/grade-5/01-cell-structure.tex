\chapter{Cell Structure}\label{ch:g5:cell-structure}

Grade~4 introduced \omterm{def:g5:cell-structure:cell}{cells} as life's building blocks. This chapter looks
more carefully at \omterm{def:g5:cell-structure:cell}{cell} parts, compares \omterm{def:g1:plants-animals:plant}{plant} and \omterm{def:g1:plants-animals:animal}{animal} \omterm{def:g5:cell-structure:cell}{cells}, and
links structure to function --- why a part's shape and position match
its job.

\section{Review: the cell idea}

\begin{definition}[Cell]\label{def:g5:cell-structure:cell}
A \emph{cell}\index{cell} is a membrane-bounded unit of life with
genetic material and the machinery to maintain itself. \omterm{def:g4:ecosystems:levels}{Organisms} may be
one cell or many \omterm{def:g4:cells:special}{specialised cells}.
\end{definition}

\section{Parts of animal cells}

\begin{definition}[Main animal-cell parts]\label{def:g5:cell-structure:animal}
The \emph{cell membrane}\index{cell membrane} controls entry and exit.
The \emph{nucleus}\index{nucleus (cell)} holds genetic instructions.
\emph{Cytoplasm}\index{cytoplasm} is the interior material where many
reactions occur. \emph{Mitochondria}\index{mitochondrion} release usable
\omterm{def:g2:food-energy:energy}{energy} from \omterm{def:g2:food-energy:food}{food} molecules (studied more later).
\end{definition}

\section{Parts of plant cells}

\begin{definition}[Plant extras]\label{def:g5:cell-structure:plant}
\omterm{def:g1:plants-animals:plant}{Plant} \omterm{def:g5:cell-structure:cell}{cells} usually have a rigid \emph{cell wall}\index{cell wall}
outside the membrane, \emph{chloroplasts}\index{chloroplast} for
\omterm{def:g4:plants-need:food}{photosynthesis}, and a large central \emph{vacuole}\index{vacuole} that
stores \omterm{def:g4:plants-need:needs}{water} and helps keep the \omterm{def:g5:cell-structure:cell}{cell} firm.
\end{definition}

\begin{example}[Why plants stand]\label{ex:g5:cell-structure:turgor}
Water-filled \omterm{def:g5:cell-structure:plant}{vacuoles} press the \omterm{def:g5:cell-structure:animal}{cytoplasm} against the wall. That
pressure (turgor) helps soft \omterm{def:g1:plants-animals:plant}{plant} parts stay upright. Wilted \omterm{def:g1:plants-animals:plant}{plants}
have lost too much \omterm{def:g4:plants-need:needs}{water} from \omterm{def:g5:cell-structure:cell}{cells}.
\end{example}

\begin{proposition}[Shared toolkit]\label{prop:g5:cell-structure:shared}
\omterm{def:g1:plants-animals:plant}{Plant} and \omterm{def:g1:plants-animals:animal}{animal} \omterm{def:g5:cell-structure:cell}{cells} both have membrane, \omterm{def:g5:cell-structure:animal}{cytoplasm}, \omterm{def:g5:cell-structure:animal}{nucleus} and
mitochondria. Plant-only features match \omterm{def:g1:plants-animals:plant}{plant} life: support, \omterm{def:g4:plants-need:needs}{water}
storage and light-based \omterm{def:g2:food-energy:food}{food} making.
\end{proposition}
\admitted

\section{Structure matches function}

\begin{definition}[Specialised cells]\label{def:g5:cell-structure:special}
\emph{Specialised cells}\index{cell!specialised} have shapes and parts
suited to particular jobs: \omterm{def:g4:plants-need:parts}{root} hair \omterm{def:g5:cell-structure:cell}{cells} increase surface for \omterm{def:g4:plants-need:needs}{water}
uptake; \omterm{def:g4:plants-need:parts}{leaf} palisade \omterm{def:g5:cell-structure:cell}{cells} pack \omterm{def:g5:cell-structure:plant}{chloroplasts}; nerve \omterm{def:g5:cell-structure:cell}{cells} are long for
signal travel; red blood \omterm{def:g5:cell-structure:cell}{cells} carry oxygen.
\end{definition}

\begin{method}[Linking form to job]\label{met:g5:cell-structure:link}
\begin{enumerate}
  \item Name the \omterm{def:g5:cell-structure:cell}{cell} type and where it lives in the \omterm{def:g4:ecosystems:levels}{organism}.
  \item List unusual features (shape, extra organelles, missing parts).
  \item Explain how each feature helps the \omterm{def:g5:cell-structure:cell}{cell}'s job.
\end{enumerate}
\end{method}

\begin{example}[Root hair]\label{ex:g5:cell-structure:root}
A long thin extension increases surface area so more \omterm{def:g4:plants-need:needs}{water} and minerals
can enter from soil --- structure serving absorption.
\end{example}
