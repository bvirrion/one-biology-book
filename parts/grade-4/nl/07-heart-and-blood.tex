\chapter{Het hart en het bloed}\label{ch:g4:heart-and-blood}

Druk twee vingers --- niet je duim --- op de binnenkant van je \omterm{def:g1:my-body:joint}{pols},
onder de duimmuis, en wacht. Daar: een klein, gestaag klopje, keer op
keer, ongeveer één keer per seconde. Dat klopje is je \omterm{def:g4:heart-and-blood:heart}{hart} aan het werk,
op afstand gevoeld. Het klopt al van voor je \omterm{def:g2:born-grow-die:cycle}{geboorte}, en het is de
motor van de bezorgdienst van het lichaam.

\section{Het bloed, de bezorgdienst van het lichaam}

\begin{proposition}[Wat het bloed vervoert]\label{prop:g4:heart-and-blood:carries}
Het \emph{bloed}\index{bloed} is een vloeistof die elke hoek van het
lichaam bereikt, en het is de vervoerder van het lichaam:
\begin{itemize}
  \item bij de \omterm{def:g4:journey-of-food:tract}{dunne darm} haalt het de \emph{\omterm{def:g4:journey-of-food:digestion}{voedingsstoffen}} op
        (\cref{prop:g4:journey-of-food:absorb});
  \item bij de \omterm{def:g4:breathing:organs}{longen} haalt het \emph{zuurstof} op
        (\cref{prop:g4:breathing:exchange});
  \item het levert allebei aan elk deel van het lichaam --- \omterm{def:g3:bones-and-muscles:muscle}{spieren},
        hersenen, \omterm{def:g3:bones-and-muscles:skeleton}{botten}, \omterm{def:g1:the-five-senses:sense}{huid};
  \item op de terugweg voert het het afval af en levert het de
        koolstofdioxide bij de \omterm{def:g4:breathing:organs}{longen} af om uitgeademd te worden.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Eén bezorgronde]\label{ex:g4:heart-and-blood:round}
Volg een druppel \omterm{prop:g4:journey-of-food:absorb}{bloed}: in de \omterm{def:g4:breathing:organs}{longen} wordt hij helderrood, geladen met
zuurstof; het \omterm{def:g4:heart-and-blood:heart}{hart} stuurt hem eropuit; in een rennende beenspier geeft
hij zuurstof en \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} af, neemt hij koolstofdioxide op en
wordt hij donkerder; terug bij het \omterm{def:g4:heart-and-blood:heart}{hart} wordt hij naar de \omterm{def:g4:breathing:organs}{longen}
gestuurd om te lossen en te laden --- en er weer op uit. Een hele ronde
duurt ruim minder dan een minuut.
\end{example}

\begin{definition}[Bloedvaten]\label{def:g4:heart-and-blood:vessels}
\omterm{prop:g4:journey-of-food:absorb}{Bloed} reist door buizen die \emph{bloedvaten}\index{bloedvat} heten:
brede die het \omterm{def:g4:heart-and-blood:heart}{hart} verlaten en binnengaan, en die zich steeds fijner
vertakken --- tot haardunne vaatjes die tussen de delen van het lichaam
door lopen en waar de leveringen worden afgegeven. Achter elkaar gelegd
zouden de vaten van één lichaam tienduizenden kilometers lang zijn.
\end{definition}

\begin{remark}[De vaten zien]\label{rem:g4:heart-and-blood:see}
Kijk naar de binnenkant van je \omterm{def:g1:my-body:joint}{pols}: blauwige lijnen onder de \omterm{def:g1:the-five-senses:sense}{huid} ---
\omterm{def:g4:heart-and-blood:vessels}{bloedvaten} op weg terug naar het \omterm{def:g4:heart-and-blood:heart}{hart}. Een geschaafde \omterm{def:g1:my-body:joint}{knie} laat het
andere uiteinde van het netwerk zien: zelfs het dunste sneetje vindt
\omterm{prop:g4:journey-of-food:absorb}{bloed}, want de fijnste vaatjes zitten overal.
\end{remark}

\section{Het hart, de pomp}

\begin{definition}[Het hart]\label{def:g4:heart-and-blood:heart}
Het \emph{hart}\index{hart} is een holle \omterm{def:g3:bones-and-muscles:muscle}{spier}, ongeveer zo groot als de
vuist van zijn eigenaar, midden in de borst achter de ribbenkast. Door
samen te knijpen --- een \emph{hartslag}\index{hartslag} --- drijft het
het \omterm{prop:g4:journey-of-food:absorb}{bloed} door de vaten. Het heeft twee helften: de rechterhelft pompt
\omterm{prop:g4:journey-of-food:absorb}{bloed} naar de \emph{\omterm{def:g4:breathing:organs}{longen}} om zuurstof te laden; de linkerhelft pompt
het zuurstofrijke \omterm{prop:g4:journey-of-food:absorb}{bloed} naar het \emph{hele lichaam}.
\end{definition}

\begin{proposition}[De pols]\label{prop:g4:heart-and-blood:pulse}
Elke \omterm{def:g4:heart-and-blood:heart}{hartslag} drijft een golfje \omterm{prop:g4:journey-of-food:absorb}{bloed} door de vaten; gevoeld aan de \omterm{def:g1:my-body:joint}{pols}
of in de \omterm{def:g1:my-body:parts}{hals} is dat golfje de \emph{polsslag}\index{polsslag}. De
\omterm{prop:g4:heart-and-blood:pulse}{polsslag} tellen is de \omterm{def:g4:heart-and-blood:heart}{hartslagen} tellen: ongeveer $80$ tot $100$ per
minuut voor een kind in rust, $60$ tot $80$ voor een volwassene --- en
veel meer bij inspanning.
\end{proposition}
\admitted

\begin{example}[Het hart houdt de benen bij]\label{ex:g4:heart-and-blood:effort}
Als je rent, verbranden je beenspieren snel \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} en
zuurstof; de leveringen moeten versnellen. Dus klopt het \omterm{def:g4:heart-and-blood:heart}{hart} sneller
\emph{en} harder --- de \omterm{prop:g4:heart-and-blood:pulse}{polsslag} van een kind kan bij een sprint de
$150$ voorbijgaan --- terwijl de ademhaling versnelt om het \omterm{prop:g4:journey-of-food:absorb}{bloed} weer
te laden (\cref{ex:g4:breathing:running}). \omterm{def:g4:heart-and-blood:heart}{Hart} en \omterm{def:g4:breathing:organs}{longen} zijn twee
helften van één bevoorradingsketen.
\end{example}

\begin{method}[Een polsslag opnemen]\label{met:g4:heart-and-blood:pulse}
\begin{enumerate}
  \item Draai een hand met de palm omhoog; leg twee vingers van de
        andere hand op de \omterm{def:g1:my-body:joint}{pols}, vlak onder de duimmuis --- nooit je
        duim, die zijn eigen kleine \omterm{prop:g4:heart-and-blood:pulse}{polsslag} heeft;
  \item druk zachtjes tot je het kloppen voelt;
  \item tel de klopjes gedurende één minuut (een klok met seconden helpt);
  \item schrijf het getal op, met wat het lichaam aan het doen was:
        rusten, wandelen, net na een sprint.
\end{enumerate}
\end{method}

\begin{remark}[Een spier die nooit rust --- bijna]\label{rem:g4:heart-and-blood:rest}
Tussen twee slagen ontspant het \omterm{def:g4:heart-and-blood:heart}{hart} een fractie van een seconde --- dat
splintertje is alle rust die het ooit neemt. Dag en nacht, ongeveer
$100000$ slagen per dag, een heel leven lang. Zoals elke \omterm{def:g3:bones-and-muscles:muscle}{spier}
(\cref{met:g3:bones-and-muscles:care}) wordt het sterker van oefening:
een getraind \omterm{def:g4:heart-and-blood:heart}{hart} verplaatst meer \omterm{prop:g4:journey-of-food:absorb}{bloed} per slag, en kan dus rustiger
kloppen.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:1}
Noem twee dingen die het \omterm{prop:g4:journey-of-food:absorb}{bloed} aflevert, en waar het elk ervan ophaalt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:2}
Welk afvalgas voert het \omterm{prop:g4:journey-of-food:absorb}{bloed} af, en waar levert het dat af?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:3}
Waarvan is het \omterm{def:g4:heart-and-blood:heart}{hart} gemaakt, hoe groot is het en waar zit het?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:4}
Wat doet elke helft van het \omterm{def:g4:heart-and-blood:heart}{hart}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:5}
Wat is de \omterm{prop:g4:heart-and-blood:pulse}{polsslag}, en wat vertelt het tellen ervan je?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:6}
Waarom moet je de \omterm{prop:g4:heart-and-blood:pulse}{polsslag} met je vingers opnemen en niet met je duim?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:7}
Waarom raast de \omterm{prop:g4:heart-and-blood:pulse}{polsslag} tijdens een sprint? Vertel het verhaal vanuit
het oogpunt van de beenspieren.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:8}
Probeer het: neem je \omterm{prop:g4:heart-and-blood:pulse}{polsslag} in rust op met
\cref{met:g4:heart-and-blood:pulse}, doe dertig seconden sprongen met
gespreide \omterm{def:g1:my-body:parts}{armen} en \omterm{def:g1:my-body:parts}{benen}, en neem hem opnieuw op. Geef beide getallen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:heart-and-blood:9}
In \cref{ex:g4:heart-and-blood:round} is de druppel \omterm{prop:g4:journey-of-food:absorb}{bloed} helderrood als
hij de \omterm{def:g4:breathing:organs}{longen} verlaat en donkerder als hij de beenspier verlaat. Wat is
er elke keer in hem veranderd?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:heart-and-blood:10}
Het \omterm{prop:g4:journey-of-food:absorb}{bloed} verbindt drie organen van jouw leerjaar: \omterm{def:g4:journey-of-food:tract}{dunne darm}, \omterm{def:g4:breathing:organs}{longen},
\omterm{def:g4:heart-and-blood:heart}{hart}. Teken of beschrijf het netwerk: wat draagt elk bij aan de
bezorgdienst?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:heart-and-blood:11}
De \omterm{prop:g4:heart-and-blood:pulse}{polsslag} van een getrainde wielrenner kan in rust $50$ zijn, die van
een ongetrainde volwassene $75$. Leg met
\cref{rem:g4:heart-and-blood:rest} uit hoe \emph{langzamer}
\emph{sterker} kan betekenen --- wat moet elke slag van de wielrenner doen?
\end{exercise}
