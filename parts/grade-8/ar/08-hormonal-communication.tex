عرّف بها \omterm{def:g8:puberty:puberty}{البلوغ}

\section{غدد وهرمونات وأهداف}

\begin{definition}[الغدد الصماء]\label{def:g8:hormonal-communication:glands}
كل \emph{غدة}\index{غدة} يطلق منتجها \emph{في \omterm{prop:g4:journey-of-food:absorb}{الدم}} (لا
في قناة، كقناة \omterm{def:g4:journey-of-food:tract}{اللعاب}) هي غدة \emph{صماء}، ومنتجاتها هي
\emph{الهرمونات}\index{هرمون}. وأشهر الكتّاب:
\emph{الغدة القائدة} في قاعدة \omterm{def:g5:brain-in-command:system}{الدماغ} --- مشغّلة \omterm{def:g8:puberty:puberty}{البلوغ}،
وقائدة أكثر الأخريات؛ و\emph{الدرقية} في العنق ---
هرمونها يضبط سرعة عمل الجسم كله؛ و\emph{الكظريتان} فوق
\omterm{def:g7:removing-wastes:kidneys}{الكليتين} --- هرمون الإنذار عند الفزع والعدو؛
و\emph{البنكرياس} --- هرمونه يدير غلوكوز \omterm{prop:g4:journey-of-food:absorb}{الدم}؛
و\emph{\omterm{def:g8:reproductive-systems:female}{المبيضان} \omterm{def:g8:reproductive-systems:male}{والخصيتان}} --- هرمونات الدورة وإعادة
البناء في \cref{ch:g8:reproductive-systems}.
\end{definition}

\begin{proposition}[العنونة بالمستقبل]\label{prop:g8:hormonal-communication:address}
يوصل \omterm{prop:g4:journey-of-food:absorb}{الدم} كل \omterm{def:g8:hormonal-communication:glands}{هرمون} إلى كل مكان
(\cref{ex:g8:puberty:sites}) --- ومع ذلك لا يؤثر كل
واحد إلا في \emph{أعضائه الهدف}\index{عضو هدف}: \omterm{def:g5:first-look-at-cells:cell}{الخلايا}
المبنية بمواضع رسو، هي \emph{المستقبلات}، لذلك الرسول.
حمل بالإذاعة، وعنونة عند المتلقي: رسالة يتلقاها كل بيت،
ولا تقرأ إلا حيث يوافق المفتاح. (ورسو \omterm{def:g8:nervous-communication:synapse}{المشبك} في
\cref{def:g8:nervous-communication:synapse} هو الحيلة
نفسها على مدى النانومتر.)
\end{proposition}
\admitted

\begin{example}[جولة رسول واحد: هرمون الإنذار]\label{ex:g8:hormonal-communication:alarm}
نجاة بفارق شعرة على الدراجة: تصب الغدتان الكظريتان
\omterm{def:g8:hormonal-communication:glands}{هرمون} الإنذار في \omterm{prop:g4:journey-of-food:absorb}{الدم}، وفي ثوان --- دورة واحدة في
المجرى (\cref{ex:g4:heart-and-blood:round}) --- يستجيب
كل متلق مجهز في الحال: فيقفز إيقاع \omterm{def:g4:heart-and-blood:heart}{القلب}، وتتسع المسالك
الهوائية، ويطلق \omterm{prop:g7:digestion-and-nutrients:absorption}{الكبد} \omterm{prop:g7:organs-at-work:bill}{الغلوكوز} المخزون (وقد نودي مصرف
\cref{ex:g7:digestion-and-nutrients:breadtrace})، وتتسع
\omterm{def:g4:heart-and-blood:vessels}{أوعية} \omterm{def:g3:bones-and-muscles:muscle}{العضلات} وتضيق \omterm{def:g4:heart-and-blood:vessels}{أوعية} \omterm{def:g4:journey-of-food:tract}{الأنبوب الهضمي} (إعادة التوجيه
في \cref{prop:g7:organs-at-work:supply}، ممسوكة
\omterm{def:g8:hormonal-communication:glands}{بالهرمون})، ويشحب \omterm{def:g1:the-five-senses:sense}{الجلد} ويقشعر. رسالة واحدة، وستة أجوبة،
وحال منسقة واحدة: الاستعداد.
\end{example}

\section{التوازن بالتغذية الراجعة}

\begin{proposition}[مبدأ التغذية الراجعة]\label{prop:g8:hormonal-communication:feedback}
تضبط أجهزة \omterm{def:g8:hormonal-communication:glands}{الهرمونات} نفسها بنفسها عن طريق \emph{التغذية
الراجعة}\index{تغذية راجعة}: \omterm{def:g8:hormonal-communication:glands}{فالغدة} الكاتبة \emph{ترصد
نتيجة} رسائلها، وتبطئ متى بلغت النتيجة --- منظم حرارة،
لا حنفية. والمثال البارز غلوكوز \omterm{prop:g4:journey-of-food:absorb}{الدم}: فبعد الوجبة، يثير
ارتفاع \omterm{prop:g7:organs-at-work:bill}{الغلوكوز} نفسه \omterm{def:g8:hormonal-communication:glands}{هرمون} التخزين في البنكرياس؛ ومتى
خزن \omterm{prop:g7:organs-at-work:bill}{الغلوكوز} هبط المستوى، وخمد الإفراز معه؛ وبين الوجبات
يفتح \omterm{def:g8:hormonal-communication:glands}{هرمون} شريك المصرف في الاتجاه الآخر. فيسير المستوى
في ممر ضيق، ليلًا ونهارًا، بلا سائق واع.
\end{proposition}
\admitted

\begin{example}[حين تفشل التغذية الراجعة: السكري]\label{ex:g8:hormonal-communication:diabetes}
في \emph{السكري}\index{سكري}، تفشل إشارة \omterm{def:g8:hormonal-communication:glands}{هرمون} التخزين
--- فإما أن يكف البنكرياس عن كتابتها، وإما أن تكف
المستقبلات عن الجواب. فيتراكم \omterm{prop:g7:organs-at-work:bill}{الغلوكوز} في \omterm{prop:g4:journey-of-food:absorb}{الدم} بعد كل
وجبة؛ ويغلب استرداد \omterm{def:g7:removing-wastes:kidneys}{الكلية}، فيظهر السكر في \omterm{def:g7:removing-wastes:kidneys}{البول} ---
وهي الإشارة الكلاسيكية في
\cref{ex:g7:removing-wastes:thrift}، مفسرة أخيرًا.
والعلاج فصل تطبيقي: جرعات مقيسة من \omterm{def:g8:hormonal-communication:glands}{الهرمون} الناقص،
موقوتة بالوجبات --- الرسالة تكتب بالمحقنة. وملايين
يعيشون عيشًا حسنًا على هذا الترتيب بعينه.
\end{example}

\begin{remark}[الجهازان في شراكة]\label{rem:g8:hormonal-communication:partnership}
البرق والبريد في
\Cref{prop:g8:nervous-communication:compare} إدارة
واحدة. \omterm{def:g8:hormonal-communication:glands}{والغدة} القائدة في قاعدة \omterm{def:g5:brain-in-command:system}{الدماغ} هي الواجهة:
\omterm{def:g5:brain-in-command:system}{الجهاز العصبي} فوقها، \omterm{def:g8:hormonal-communication:glands}{والهرمونات} تحتها --- ففزع راكب
الدراجة جرى \omterm{def:g8:nervous-communication:neuron}{بالعصب} أولًا، وأدامه \omterm{def:g8:hormonal-communication:glands}{الهرمون}؛ \omterm{def:g8:puberty:puberty}{والبلوغ} يجري
ببطء \omterm{def:g8:hormonal-communication:glands}{الهرمون} تحت قائد في \omterm{def:g5:brain-in-command:system}{الدماغ}؛ وخراف
\cref{ex:g8:reproduction-and-environment:lambs} تحول طول
النهار (وهو إدراك عصبي) إلى هرمونات تناسل. أسلاك سريعة
حيث تهم السرعة، وبريد بطيء حيث يجب أن يجري البرنامج
أشهرًا، ومكاتب ترجمة بينهما.
\end{remark}

\section{قراءة جهاز هرموني}

\begin{method}[الأسئلة الأربعة في أي هرمون]\label{met:g8:hormonal-communication:read}
في أي \omterm{def:g8:hormonal-communication:glands}{هرمون} تلقاه، أثبت:
\begin{enumerate}
  \item \emph{الكاتب}: أي \omterm{def:g8:hormonal-communication:glands}{غدة}، وفي \omterm{prop:g4:journey-of-food:absorb}{الدم}، وبأي مثير؟
  \item \emph{المتلقون}: أي \omterm{prop:g8:hormonal-communication:address}{أعضاء هدف} تحمل مستقبلاته،
        وماذا يفعل كل واحد منها عند التسليم؟
  \item \emph{\omterm{prop:g8:hormonal-communication:feedback}{التغذية الراجعة}}: أي نتيجة تبلغ راجعة،
        وكيف تمنع الكتابة الزائدة؟
  \item \emph{الفشل والاستعمال}: ماذا يحدث حين تفشل
        الرسالة --- وهل يحاكيها الطب (الحبة، وجرعات
        المصاب \omterm{ex:g8:hormonal-communication:diabetes}{بالسكري})؟
\end{enumerate}
\end{method}

\begin{example}[الطريقة على ضابط الإيقاع الدرقي]\label{ex:g8:hormonal-communication:thyroid}
الكاتب: الدرقية، تقودها \omterm{def:g8:hormonal-communication:glands}{الغدة} القائدة. والمتلقون: كل
\omterm{def:g5:first-look-at-cells:cell}{خلية} تقريبًا --- \omterm{def:g8:hormonal-communication:glands}{فالهرمون} يضبط معدل النار الهادئة في
\cref{prop:g7:what-is-respiration:power}. \omterm{prop:g8:hormonal-communication:feedback}{والتغذية
الراجعة}: \omterm{def:g8:hormonal-communication:glands}{الغدة} القائدة تقرأ المستوى وتعدل قيادتها.
والفشل: قليل منه، فيعمل المحرك على الحد الأدنى --- تعب،
وبرد، وبطء في كل شيء؛ وكثير منه، فيتسارع. والاستعمال:
\omterm{def:g8:hormonal-communication:glands}{الهرمون} الناقص قرص يومي --- من أقدم رسائل الطب بالبريد.
\end{example}

\section{مسألة: دفتر الغلوكوز}

\begin{problem}[{مسألة نهاية الأسبوع --- يوم من سكر الدم،
مقروءًا مراسلةً}]\label{pb:g8:hormonal-communication:1}
سجل يوم (متخيَّل) من مرقاب غلوكوز مستمر: ثبات طوال
الليل؛ وفطور في الساعة 8 --- ارتفاع، ثم هبوط هادئ عند
الساعة 10؛ ورياضة في الساعة 12 --- انخفاض وجيز، يتدارك
سريعًا؛ وغداء فائت --- ثبات طوال العصر مع ذلك؛ وفزع في
الساعة 17 (نجاة بفارق شعرة على الدراجة) --- ارتفاع حاد
وجيز؛ وعشاء في الساعة 19 --- ارتفاع وهبوط من جديد.

\medskip\noindent\textbf{الجزء الأول --- الصباح.}
\begin{enumerate}
  \item ثبات الليل: أي مراسلة برسالتين أمسكت الممر
        بينما صاحبها نائم؟
  \item ارتفاع الفطور وهبوطه: سمّ المثير، والكاتب،
        والرسالة، والمتلقين الذين خزنوا الفائض.
  \item لماذا كان \emph{توقف} الهبوط --- من غير أن يغوص
        تحت الممر --- هو نفسه توقيع \omterm{prop:g8:hormonal-communication:feedback}{التغذية الراجعة}؟
  \item مستوى الساعة 10 يوافق مستوى الساعة 7. بم تتنبأ
        صورة منظم الحرارة عن مسار الإفراز بينهما؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- العصر.}
\begin{enumerate}[resume]
  \item انخفاض الرياضة وتداركه: أي رسالة نادت مصرف
        \omterm{prop:g7:digestion-and-nutrients:absorption}{الكبد}، وأي فصل كان حسابه ينفق؟
  \item الغداء الفائت لم يغير شيئًا ظاهرًا. أي \omterm{def:g8:hormonal-communication:glands}{هرمون}
        شريك جسر العصر بعمله الهادئ؟
  \item ارتفاع الفزع الحاد: سمّ الكاتب، والغاية من
        السكر المفاجئ --- استعدادًا لماذا، وفق
        \cref{ex:g8:hormonal-communication:alarm}؟
  \item لماذا خمد ارتفاع الفزع في غضون ساعة، خلافًا
        لتراكم \omterm{ex:g8:hormonal-communication:diabetes}{السكري} الدائم؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- الاستشارة.}
\begin{enumerate}[resume]
  \item سجل مصاب \omterm{ex:g8:hormonal-communication:diabetes}{بالسكري} لليوم نفسه يختلف عند كل وجبة.
        صف منحنى الفطور من غير علاج، وأين تظهر إشارة
        \omterm{def:g7:removing-wastes:kidneys}{الكلية}.
  \item والنسخة المعالجة: ماذا تحاكي الجرعة المقيسة قبل
        كل وجبة، ولماذا يجب أن يوافق مقدارها الصحن؟
  \item يسمي الطبيب السجل السليم ``حديثًا لا تسمعه
        قط''. اذكر متحدثي اليوم --- \omterm{def:g8:hormonal-communication:glands}{الغدد}، والرسائل،
        والمتلقين --- في جدول موجز واحد أو فقرة واحدة.
  \item اختم بعبرة الفصل في جملة: أي نوع من الأجهزة
        يبقي قيمة في ممر ضيق عمرًا كاملًا بلا سائق؟
\end{enumerate}
\end{problem}
