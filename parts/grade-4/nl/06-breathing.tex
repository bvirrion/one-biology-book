\chapter{Ademhalen}\label{ch:g4:breathing}

Houd je adem in en begin te tellen. Ver voor de minuut om is, eist je
borst steeds dringender om weer te mogen bewegen. Een maaltijd kun je
overslaan, slapen kan tot vanavond wachten --- maar lucht is \emph{nu}
nodig, dag en nacht, ongeveer vijftien keer per minuut. Wat is er zo
dringend aan een teug lucht?

\section{De ademhalingsbewegingen}

\begin{definition}[In- en uitademen]\label{def:g4:breathing:movements}
\emph{Ademhalen}\index{ademhalen} is het eindeloze afwisselen van twee
bewegingen:
\begin{itemize}
  \item \emph{inademen} (inhaleren)\index{inademen}: de borst zet uit en er stroomt
        lucht naar binnen door \omterm{def:g1:the-five-senses:sense}{neus} of \omterm{def:g4:journey-of-food:tract}{mond};
  \item \emph{uitademen} (exhaleren)\index{uitademen}: de borst ontspant en de lucht
        stroomt weer naar buiten.
\end{itemize}
In rust herhaalt dat paar zich ongeveer $15$ keer per minuut --- sneller
bij \omterm{def:g3:stages-of-life:stages}{baby}'s, sneller bij inspanning --- zonder dat je er ooit bij hoeft
na te denken.
\end{definition}

\begin{example}[De bewegingen bekijken]\label{ex:g4:breathing:watch}
Handen plat op je \omterm{def:g3:bones-and-muscles:skeleton}{ribben}: bij het \omterm{def:g4:breathing:movements}{inademen} komt de ribbenkast onder je
handpalmen omhoog en wordt hij breder; bij het \omterm{def:g4:breathing:movements}{uitademen} zakt hij weer
terug. De flank van een slapende kat laat hetzelfde stijgen en dalen
zien. De bewegingen stoppen nooit --- wakker, slapend, van de eerste
kreet tot de laatste dag.
\end{example}

\section{Waar de lucht heen gaat}

\begin{definition}[De ademhalingsorganen]\label{def:g4:breathing:organs}
Ingeademde lucht reist door de \omterm{def:g1:the-five-senses:sense}{neus} (of de \omterm{def:g4:journey-of-food:tract}{mond}) omlaag door de
\emph{luchtpijp}\index{luchtpijp} --- de stevige geringde buis die je
voor in je \omterm{def:g1:my-body:parts}{hals} kunt voelen --- die zich in twee takken splitst, één
naar elke \emph{long}\index{long}. De longen, twee sponsachtige organen
die de ribbenkast vullen (achter de \omterm{def:g3:bones-and-muscles:skeleton}{ribben} uit
\cref{def:g3:bones-and-muscles:skeleton}), eindigen in miljoenen
piepkleine luchtzakjes, elk omwikkeld met fijne bloedvaten.
\end{definition}

\begin{remark}[Met opzet sponsachtig]\label{rem:g4:breathing:sponge}
Miljoenen zakjes in plaats van twee eenvoudige zakken: het is opnieuw de
truc van de \omterm{def:g4:journey-of-food:tract}{dunne darm}
(\cref{rem:g4:journey-of-food:surface}). Al die zakjeswanden tellen op
tot een enorm ontmoetingsoppervlak tussen lucht en \omterm{prop:g4:journey-of-food:absorb}{bloed} --- zo groot als
een halve tennisbaan, opgevouwen in je borst.
\end{remark}

\section{Waar ademhalen voor dient}

\begin{proposition}[Zuurstof opnemen, koolstofdioxide teruggeven]\label{prop:g4:breathing:exchange}
Lucht is een mengsel van gassen, en het lichaam wil er één van:
\emph{zuurstof}\index{zuurstof}. In de zakjes van de \omterm{def:g4:breathing:organs}{longen} gaat de
zuurstof uit verse lucht het \omterm{prop:g4:journey-of-food:absorb}{bloed} in, dat haar naar elk deel van het
lichaam brengt; daar wordt ze samen met \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} gebruikt om de
\omterm{prop:g3:balanced-diet:jobs}{energie} van \cref{prop:g3:balanced-diet:jobs} vrij te maken. Het
afvalgas van het lichaam, \emph{koolstofdioxide}\index{koolstofdioxide},
maakt de terugreis en wordt uitgeademd. Uitgeademde lucht bevat minder
zuurstof en meer koolstofdioxide dan ingeademde lucht --- en dat
verschil is precies waar \omterm{def:g4:breathing:movements}{ademhalen} \emph{voor} dient.
\end{proposition}
\admitted

\begin{example}[Waarom je van rennen buiten adem raakt]\label{ex:g4:breathing:running}
Als je sprint, verbranden je \omterm{def:g1:my-body:parts}{benen} snel \omterm{prop:g3:balanced-diet:jobs}{energie} --- dus hebben ze snel
zuurstof nodig. De ademhaling doet mee: van ongeveer $15$ rustige teugen
per minuut naar $40$ of meer, en diepe, terwijl het hart
(\cref{ch:g4:heart-and-blood} vertelt zijn kant van het verhaal) raast
om de leveringen te versnellen. Het hijgen na de wedloop is het lichaam
dat zijn zuurstofrekening vereffent.
\end{example}

\begin{example}[De beslagen spiegel]\label{ex:g4:breathing:mirror}
Adem op een koude spiegel: damp. Uitgeademde lucht komt veranderd terug
--- opgewarmd, bevochtigd, en, onzichtbaar, armer aan zuurstof en rijker
aan koolstofdioxide. De spiegel laat het vocht zien; de gaswisseling die
hij niet kan tonen is het belangrijke deel.
\end{example}

\begin{method}[Voor je adem zorgen]\label{met:g4:breathing:care}
\begin{enumerate}
  \item Lucht kamers dagelijks --- veel ademhalers in een gesloten kamer
        betekent muffe lucht vol koolstofdioxide;
  \item adem in koude of stoffige lucht door je \omterm{def:g1:the-five-senses:sense}{neus}: die verwarmt,
        bevochtigt en filtert;
  \item ren, zwem, speel --- geoefende \omterm{def:g4:breathing:organs}{longen} en ademhalingsspieren
        worden sterker en dieper;
  \item houd rook bij je \omterm{def:g4:breathing:organs}{longen} vandaan: die maakt de tere zakjes vuil,
        en ze maken zichzelf maar langzaam weer schoon.
\end{enumerate}
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:breathing:1}
Noem de twee ademhalingsbewegingen en wat de borst bij elk doet.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:2}
Volg de weg van een ademteug van de \omterm{def:g1:the-five-senses:sense}{neus} tot de piepkleine luchtzakjes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:3}
Ongeveer hoe vaak per minuut ademt een rustig kind in rust? Wat gebeurt
er met dat getal tijdens een sprint, en waarom?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:4}
Welk gas haalt het lichaam uit de lucht, en welk gas geeft het terug?
Waar precies vindt de wisseling plaats?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:5}
Hoe verschilt uitgeademde lucht van ingeademde lucht? Noem drie verschillen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:6}
Waarom zijn de \omterm{def:g4:breathing:organs}{longen} gebouwd als miljoenen piepkleine zakjes in plaats
van als twee eenvoudige zakken? Welk ander orgaan van dit jaar gebruikt
dezelfde truc?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:7}
Waarom kun je op een ijskoude of stoffige dag beter door je \omterm{def:g1:the-five-senses:sense}{neus} ademen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:breathing:8}
Probeer het: tel rustig zittend een minuut lang je ademteugen. Doe
daarna twintig jumping jacks en tel opnieuw. Geef beide getallen en
verklaar het verschil.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:9}
Maaltijden kunnen uren wachten; \omterm{def:g4:breathing:movements}{ademhalen} kan geen minuut wachten. Wat
zegt dat over hoeveel zuurstof het lichaam in voorraad houdt,
vergeleken met zijn voedselvoorraden?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:breathing:10}
Een klaslokaal voelt tegen het eind van de ochtend ``benauwd''. Wat is
er met de lucht veranderd, volgens \cref{prop:g4:breathing:exchange}, en
wat schrijft \cref{met:g4:breathing:care} voor?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:breathing:11}
De forel heeft geen \omterm{def:g4:breathing:organs}{longen}: er stroomt water over haar geveerde kieuwen,
en daar gaat de zuurstof die \emph{in het water is opgelost} haar \omterm{prop:g4:journey-of-food:absorb}{bloed}
in. Welke taak delen kieuwen en \omterm{def:g4:breathing:organs}{longen}? En waarom stikt een forel in de
lucht, waar zuurstof volop aanwezig is? (Tip: buiten het water zakken de
geveerde kieuwoppervlakken in elkaar en plakken ze aan elkaar.)
\end{exercise}
