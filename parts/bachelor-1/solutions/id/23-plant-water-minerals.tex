\section*{Bab \ref{ch:b1:plant-water-minerals} --- Nutrisi Air dan Mineral Tumbuhan}

\begin{solution}{exo:b1:plant-water-minerals:1}
\omterm{def:b1:plant-water-minerals:pathways}{Apoplas}: dinding dan ruang yang sinambung, yang dilintasi tanpa
melewati sebuah membran. \omterm{def:b1:plant-water-minerals:pathways}{Simplas}: \omterm{def:b1:cell-unit-of-life:cell}{sitoplasma} \omterm{def:b1:cell-unit-of-life:cell}{sel} yang terhubung, yang
dimasuki lewat satu membran lalu berlanjut lewat \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesma}. \omterm{def:b1:plant-water-minerals:pathways}{Pita
Kaspari} menghalangi \omterm{def:b1:plant-water-minerals:pathways}{apoplasnya} pada \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermisnya}, sehingga memaksa
segalanya lewat sebuah membran lalu mencegah kebocoran balik dari
\omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:2}
N (\omterm{def:b1:proteins:peptide}{protein}, asam nukleat), P (\omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}, membran), K (zat osmotik,
\omterm{def:b1:flowering-plant-organization:tissues}{stomata}), Ca (dinding, pengisyaratan), Mg (\omterm{def:b1:photosynthesis:pigments}{klorofil}), dan S (\omterm{def:b1:proteins:aminoacid}{asam
amino} sisteina dan metionina).
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:3}
Tanah \qty{-0.2}{}, \omterm{def:b1:flowering-plant-organization:tissues}{xilem} \omterm{def:b1:flowering-plant-organization:organs}{akar} \qty{-0.6}{}, \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{daun} \qty{-1.3}{MPa}:
dari tanah ke dalam \omterm{def:b1:flowering-plant-organization:organs}{akarnya}, dari \omterm{def:b1:flowering-plant-organization:tissues}{xilem} \omterm{def:b1:flowering-plant-organization:organs}{akarnya} naik ke \omterm{def:b1:flowering-plant-organization:organs}{daunnya} ---
selalu menuju nilai yang lebih negatif.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:4}
Nitrogen bergerak dalam floemnya: tumbuhannya menariknya dari \omterm{def:b1:flowering-plant-organization:organs}{daun} tua
untuk memberi makan yang muda, sehingga \omterm{def:b1:flowering-plant-organization:organs}{daun} tuanya menguning lebih
dahulu. Besi tidak digerakkan kembali: \omterm{def:b1:flowering-plant-organization:organs}{daun} mudanya, yang dibangun
tanpanya, itulah yang menguning.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:5}
\omterm{def:b1:cell-unit-of-life:cell}{Sel} $\Psi = -0.9 + 0.5 = \qty{-0.4}{MPa}$: tanah ($-0.2$) $>$ \omterm{def:b1:cell-unit-of-life:cell}{sel}
($-0.4$) $>$ \omterm{def:b1:flowering-plant-organization:tissues}{xilem} ($-0.6$), sehingga air mengalir tanah $\to$ \omterm{def:b1:cell-unit-of-life:cell}{sel}
$\to$ \omterm{def:b1:flowering-plant-organization:tissues}{xilem}. Serapannya berhenti ketika \omterm{def:b1:cell-unit-of-life:cell}{selnya} mencapai
\qty{-0.2}{MPa}, yakni
$\Psi_p = -0.2 + 0.9 = \qty{0.7}{MPa}$.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:6}
$\Psi = 137\ln(\text{RH})$: \qty{-14}{MPa}, \qty{-95}{MPa},
\qty{-315}{MPa}. Bahkan udara lembap sepuluh kali lebih kering, dalam
arti ini, daripada tanah pada titik layunya: sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} selalu
kehilangan air ke udaranya; satu-satunya pertanyaannya adalah seberapa
cepat.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:7}
$E_K = 25.7\ln(0.1/100) = \qty{-178}{mV}$: pada \qty{-150}{mV} \omterm{def:b1:cell-unit-of-life:cell}{selnya}
sedikit kurang negatif daripada yang dibutuhkan, sehingga penimbunan
seribu kali memerlukan sedikit lebih daripada saluran --- tetapi sebuah
nisbah 100 akan berada pada kesetimbangan: salurannya melakukan hampir
seluruhnya. Nitrat: $E = -25.7\ln(0.5/5)
= \qty{+59}{mV}$; \omterm{def:b1:cell-unit-of-life:cell}{selnya} berada pada \qty{-150}{}, sejauh \qty{209}{mV}
dan pada arah yang salah: sebuah anion harus dipompa masuk, lewat
simpor proton.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:8}
Pada malam hari, tanpa transpirasi, pemuatan ion ke dalam \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}
menarik air masuk dan sebuah \omterm{prop:b1:plant-water-minerals:rootpressure}{tekanan akar} sebesar beberapa persepuluh
megapascal mendorong getahnya naik beberapa meter paling banyak ---
cukup untuk mencapai ujung helai rumputnya lalu memaksa tetesan
keluar, bukan untuk mencapai puncak sebatang pohon. Ketika \omterm{def:b1:flowering-plant-organization:tissues}{stomatanya}
membuka, transpirasinya menaruh \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} di bawah tegangan dan \omterm{prop:b1:plant-water-minerals:rootpressure}{tekanan
akarnya} lenyap.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:9}
Tanpa molibdenum \omterm{prop:b1:plant-water-minerals:nitrate}{reduktase nitratnya} tidak bekerja: pada nitrat
tumbuhannya tak dapat membuat \omterm{def:b1:proteins:aminoacid}{asam amino} lalu memperlihatkan
kekurangan nitrogen walaupun nitratnya berlimpah. Pada amonium ia
tumbuh biasa, sebab amoniumnya memasuki \omterm{def:b1:proteins:aminoacid}{asam aminonya} tanpa reduksi.
Perbedaannya menempatkan unsurnya dalam satu \omterm{def:b1:enzymes:enzyme}{enzim}.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:10}
\qty{30}{mg} N $= \qty{2.14}{mmol}$; $\times 8 = \qty{17}{mmol}$
elektron, \qty{8.6}{mmol} setara \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}. \omterm{def:b1:flowering-plant-organization:organs}{Daunnya}: $0.05\times
4\times 10^{-6}\times 43\,200 = \qty{8.6}{mmol}$ $\mathrm{CO_2}$,
\qty{35}{mmol} elektron: reduksi nitratnya memakan separuh elektron
fiksasi karbonnya menurut perhitungan ini --- sebuah fraksi yang besar,
yang menjadi sebab tumbuhan melakukannya dalam terang dan sebab \omterm{def:b1:flowering-plant-organization:organs}{daun}
yang tumbuh cepat menjadi pemakan pereduksi yang rakus.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:11}
\qty{2.14}{mmol} N $= \qty{1.07}{mmol}$ $\mathrm{N_2}$: \omterm{def:b1:water-small-molecules:atp}{ATP}
$16\times 1.07 = \qty{17}{mmol}$, yakni \qty{0.57}{mmol} glukosa,
\qty{0.10}{g}; ditambah 8 elektron tiap $\mathrm{N_2}$ (\qty{0.05}{g}
glukosa lagi). Aturan \qty{6}{g/g}-nya memberi \qty{0.18}{g} sehari:
nitrogenasenya sendiri adalah sebagian besarnya, dan sisanya adalah
membangun dan memelihara bintilnya, leghemoglobinnya, serta respirasi
bakterinya sendiri untuk menjaga oksigennya rendah.
\end{solution}

\begin{solution}{exo:b1:plant-water-minerals:12}
\omterm{def:b1:flowering-plant-organization:organs}{Akarnya} sendiri mengambil air dan ion yang bergerak (nitrat, kalium)
yang dibawa air tanahnya ke rambutnya, sambil membelanjakan \omterm{def:b1:water-small-molecules:atp}{ATP} hanya
pada \omterm{prop:b1:plant-water-minerals:uptake}{pompa protonnya}. Bagi fosfat, yang tak akan datang kepadanya, ia
membayar sebuah jamur dengan gula untuk memperpanjang jangkauannya;
bagi nitrogen dalam sebuah tanah tanpa nitrat ia membayar bakteri
dengan gula untuk memancangkannya. Upahnya sepersepuluh fotosintesisnya
atau lebih, dan tumbuhannya berhenti membayar ketika jasanya tidak
diperlukan: pada tanah yang kaya fosfat \omterm{def:b1:plant-water-minerals:mycorrhiza}{mikorizanya} dikurangi, pada
tanah yang kaya nitrat legum membuat lebih sedikit bintil --- pemborong
dipelihara hanya selama mereka lebih murah daripada mengerjakannya
sendiri.
\end{solution}

\begin{solution}{pb:b1:plant-water-minerals:1}
\textbf{1.} $-0.9 + 0.5 = \qty{-0.4}{MPa}$.
\textbf{2.} $-0.2 > -0.4 > -0.6 > -1.3$: tiap langkahnya lebih rendah,
airnya bergerak tanah $\to$ korteks $\to$ \omterm{def:b1:flowering-plant-organization:tissues}{xilem} $\to$ \omterm{def:b1:flowering-plant-organization:organs}{daun}.
\textbf{3.} $137\ln 0.5 = \qty{-95}{MPa}$.
\textbf{4.} Tanah ke \omterm{def:b1:flowering-plant-organization:organs}{daun} \qty{1.1}{MPa}; \omterm{def:b1:flowering-plant-organization:organs}{daun} ke udara \qty{94}{MPa}:
\qty{99}{\%} totalnya adalah langkah terakhirnya.
\textbf{5.} Tanah pada $-0.9$ berada di bawah $-0.4$ \omterm{def:b1:cell-unit-of-life:cell}{sel} korteksnya:
airnya meninggalkan \omterm{def:b1:cell-unit-of-life:cell}{selnya} sampai \omterm{def:b1:cell-unit-of-life:cell}{selnya} turun ke $-0.9$, yakni
\omterm{def:b1:eukaryotic-cell:plantcell}{turgornya} jatuh ke nol; serapannya berhenti sampai \omterm{def:b1:cell-unit-of-life:cell}{selnya} menurunkan
$\Psi_s$-nya dengan menimbun zat terlarut.
\textbf{6.} Tanahnya kini berada di bawah $-1.3$ \omterm{def:b1:flowering-plant-organization:organs}{daunnya}: airnya
mengalir keluar dari tumbuhannya; \omterm{def:b1:flowering-plant-organization:organs}{daunnya} kehilangan \omterm{def:b1:eukaryotic-cell:plantcell}{turgor} lalu layu.
Penyiraman menaikkan $\Psi$ tanahnya di atas potensial \omterm{def:b1:flowering-plant-organization:organs}{akarnya} dan
alirannya berlanjut; \omterm{def:b1:flowering-plant-organization:organs}{daunnya}, yang dindingnya utuh, terisi kembali.
\textbf{7.} $2\times 10^{-7}\times 0.2\times(-0.2 - (-0.6)) =
\qty{1.6e-8}{m^3/s}$.
\textbf{8.} $1.6\times 10^{-8}\times 86\,400 = \qty{1.4e-3}{m^3} =
\qty{1.4}{L}$ tiap hari.
\textbf{9.} \qty{1400}{g} pada \qty{0.5}{m^2} dan \qty{12}{h} siang hari
(sebagian besarnya): sekitar \qty{230}{g} tiap meter persegi tiap jam.
\textbf{10.} $230/18 = \qty{12.8}{mol}$ tiap meter persegi tiap jam, \qty{3.5}{mmol} tiap meter persegi tiap sekon --- seribu molekul air bagi tiap $\mathrm{CO_2}$ yang dipancangkan pada \qty{4}{\micro mol}.
\textbf{11.} Bedanya $-0.2 - (-0.4) = 0.2$: fluksnya terparuh,
\qty{0.7}{L} sehari. Menutup \omterm{def:b1:flowering-plant-organization:tissues}{stomatanya} telah memotong kehilangannya,
tegangannya telah mengendur, dan seluruh tangganya telah naik:
tumbuhannya menukar perolehan karbon dengan air ketika udaranya paling
kering.
\textbf{12.} Air bergerak menuruni landaian potensialnya sendiri, yang
dibuat penguapan \omterm{def:b1:flowering-plant-organization:organs}{daunnya}, sehingga \omterm{def:b1:flowering-plant-organization:organs}{akarnya} bersifat pasif; nitratnya
harus dipekatkan lawan landaian kepekatan maupun landaian listrik,
yang hanya dapat dibayar dengan \omterm{def:b1:water-small-molecules:atp}{menggandengkannya} pada \omterm{prop:b1:plant-water-minerals:uptake}{pompa protonnya}.
\textbf{13.} $\qty{1.4}{L}\times\qty{2}{mmol/L} = \qty{2.8}{mmol}$,
\qty{39}{mg} nitrogen.
\textbf{14.} Ya, dengan sepertiga berlebih; kelebihannya disimpan
dalam \omterm{def:b1:eukaryotic-cell:plantcell}{vakuolanya} sebagai nitrat atau ditolak pada \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermisnya}.
\textbf{15.} $\Delta G = RT\ln(5/2) + zFV = 2.48\times 0.92 + (-1)(96.5)(-0.15)
= 2.3 + 14.5 = \qty{16.8}{kJ/mol}$.
\textbf{16.} Tiap proton: $RT\ln(10^{-7.5}/10^{-5.5}) + F(-0.15) =
-11.4 - 14.5 = \qty{-25.9}{kJ}$; dua proton: \qty{-52}{kJ}, tiga kali
apa yang dibutuhkan.
\textbf{17.} $\qty{2.14}{mmol}\times 8 = \qty{17}{mmol}$ elektron.
\textbf{18.} $0.5\times 4\times 10^{-6}\times 43\,200 = \qty{86}{mmol}$
$\mathrm{CO_2}$, \qty{346}{mmol} elektron: \qty{5}{\%} ke nitratnya.
\textbf{19.} Ia akan menghemat 8 elektron tiap nitrogennya; tetapi
menyerap sebuah kation lalu melepaskan sebuah proton bagi
masing-masing mengasamkan tanah di sekeliling \omterm{def:b1:flowering-plant-organization:organs}{akarnya} dan \omterm{def:b1:cell-unit-of-life:cell}{selnya} harus
mengekspor asamnya, dan amonium beracun bila ia bertimbun, sehingga ia
harus diasimilasi segera dalam \omterm{def:b1:flowering-plant-organization:organs}{akarnya}.
\textbf{20.} \qty{1.07}{mmol} $\mathrm{N_2}$ $\times 16 =
\qty{17}{mmol}$ \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{21.} $6\times 0.03 = \qty{0.18}{g}$ gula, \qty{6}{\%} dari
\qty{3}{g}. Dengan \qty{2}{mmol/L} nitrat yang tersedia seharga biaya
reduksinya (\qty{5}{\%} elektronnya), fiksasinya menjadi tawaran yang
buruk dan sebatang legum membuat sedikit bintil; tanpa nitrat ia
satu-satunya jalan, dan \qty{6}{\%} hasilnya murah bagi seluruh \omterm{def:b1:proteins:peptide}{protein}
tumbuhannya.
\textbf{22.} $1.4\times 2\times 10^{-6} = \qty{2.8}{\micro mol}$,
\qty{0.09}{mg} fosforus --- seperempat puluh kebutuhannya; tumbuhannya
harus bersandar pada difusi ke \omterm{def:b1:flowering-plant-organization:organs}{rambut akarnya}, yang segera menghabiskan
sekelilingnya, dan terutama pada jamur \omterm{def:b1:plant-water-minerals:mycorrhiza}{mikorizanya}.
\textbf{23.} Pitanya menghalangi \omterm{def:b1:plant-water-minerals:pathways}{apoplasnya}, sehingga nitratnya dapat
memasuki \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} hanya lewat membran \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis} yang memompanya ke
dalam; dan ia mencegah getah \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} yang pekat bocor keluar kembali
sepanjang dindingnya, sehingga landaiannya tertahan.
\textbf{24.} Getah \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} akan menyerupai larutan tanahnya ---
encer, tak terpilih, dengan natrium dan ion lain yang tak diinginkan
--- lalu tak dapat dijaga tetap pekat; dalam tanah yang asin natriumnya
akan mencapai \omterm{def:b1:flowering-plant-organization:organs}{daunnya} dengan bebas dan tumbuhannya akan teracuni,
sedangkan sebatang tumbuhan yang normal menolaknya pada \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermisnya}.
\textbf{25.} Sekitar \qty{1.4}{L} air sehari lewat \qty{0.2}{m^2}
\omterm{def:b1:flowering-plant-organization:organs}{akarnya}, yang membawa \qty{39}{mg} nitrogen lawan sebuah kebutuhan
\qty{30}{}, yang reduksinya memakan sekitar \qty{5}{\%} elektron
fotosintesis \omterm{def:b1:flowering-plant-organization:organs}{daunnya}.
\end{solution}
