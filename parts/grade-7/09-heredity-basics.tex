\chapter{Heredity Basics}\label{ch:g7:heredity}

Children resemble parents, yet siblings are not identical. \omterm{def:g7:heredity:heredity}{Heredity} is the
passing of information for \omterm{def:g7:heredity:trait}{traits} from one generation to the next. This
chapter introduces \omterm{def:g7:heredity:gene}{genes}, \omterm{def:g5:evolution:variation}{variation} and simple inheritance ideas --- a
bridge to later genetics.

\section{Traits and variation}

\begin{definition}[Trait]\label{def:g7:heredity:trait}
A \emph{trait}\index{trait} is a \omterm{def:g3:classify:char}{characteristic} of an \omterm{def:g4:ecosystems:levels}{organism} (\omterm{def:g5:nerves:senses}{eye}
colour, \omterm{def:g7:digestion:blood}{blood} group, \omterm{def:g2:plant-parts:main}{flower} colour, height). Some traits are strongly
influenced by \omterm{def:g7:heredity:gene}{genes}; many also depend on \omterm{def:g4:traits:env}{environment}.
\end{definition}

\begin{definition}[Heredity]\label{def:g7:heredity:heredity}
\emph{Heredity}\index{heredity} (inheritance) is the transmission of
genetic information from parents to \omterm{def:g2:growing:offspring}{offspring}.
\end{definition}

\begin{proposition}[Variation matters]\label{prop:g7:heredity:variation}
Individuals in a \omterm{def:g6:classification:species}{species} show \emph{variation}\index{variation
(biological)}. \omterm{def:g5:evolution:variation}{Variation} can be continuous (height) or more discrete
(some \omterm{def:g7:digestion:blood}{blood} groups). \omterm{def:g5:evolution:variation}{Variation} is the raw material for \omterm{def:g5:evolution:ns}{natural selection}
(later grades).
\end{proposition}
\admitted

\section{Genes and chromosomes (first look)}

\begin{definition}[Gene and chromosome]\label{def:g7:heredity:gene}
A \emph{gene}\index{gene} is a segment of DNA that carries information
affecting one or more \omterm{def:g7:heredity:trait}{traits}. Genes are arranged on
\emph{chromosomes}\index{chromosome} inside the \omterm{def:g7:microscope:parts}{nucleus} of eukaryotic
\omterm{def:g7:microscope:cell}{cells}. Humans have 23 pairs of chromosomes in \omterm{def:g1:my-body:body}{body} \omterm{def:g7:microscope:cell}{cells}.
\end{definition}

\begin{definition}[Allele idea]\label{def:g7:heredity:allele}
Different versions of the same \omterm{def:g7:heredity:gene}{gene} are called \emph{alleles}\index{allele}.
For some \omterm{def:g7:heredity:trait}{traits}, one allele may be \emph{dominant}\index{dominant allele}
(shows its effect when present) and another \emph{recessive}\index{recessive
allele} (shows only when two copies are present). Full Mendelian detail
comes in later years; the idea is enough here.
\end{definition}

\section{Sexual reproduction and unique mixes}

\begin{proposition}[Why offspring are unique]\label{prop:g7:heredity:unique}
In \omterm{prop:g5:reproduction:two}{sexual reproduction}, \omterm{def:g2:growing:offspring}{offspring} receive genetic material from two
parents. Sorting and combining of \omterm{def:g7:heredity:gene}{chromosomes} (and \omterm{def:g7:heredity:gene}{genes}) produce new
combinations --- siblings share parents but not identical genomes
(except identical twins).
\end{proposition}
\admitted

\begin{example}[Family resemblance]\label{ex:g7:heredity:family}
A child may inherit a father's hair texture \omterm{def:g7:heredity:allele}{allele} pattern and a mother's
freckle tendency --- plus environmental effects such as sun exposure on
\omterm{def:g5:nerves:senses}{skin}.
\end{example}

\begin{definition}[Genotype and phenotype]\label{def:g7:heredity:pheno}
\emph{Genotype}\index{genotype} is the genetic makeup (\omterm{def:g7:heredity:allele}{alleles} present).
\emph{Phenotype}\index{phenotype} is the observable \omterm{def:g7:heredity:trait}{trait}. Phenotype $=$
genotype interpreted in an \omterm{def:g4:traits:env}{environment}.
\end{definition}

\begin{method}[Talking carefully about inheritance]\label{met:g7:heredity:talk}
\begin{enumerate}
  \item Name the \omterm{def:g7:heredity:trait}{trait} observed (\omterm{def:g7:heredity:pheno}{phenotype}).
  \item Ask whether \omterm{def:g4:traits:env}{environment} could affect it.
  \item Avoid \omterm{def:g7:heredity:gene}{one-gene} slogans for complex \omterm{def:g7:heredity:trait}{traits} (intelligence, athletic
        skill).
  \item Use \omterm{def:g7:heredity:allele}{dominant}/\omterm{def:g7:heredity:allele}{recessive} language only when the chapter context
        supports a simple model.
\end{enumerate}
\end{method}
