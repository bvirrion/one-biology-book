\chapter{قراءة القرابة: الأشجار التطورية}\label{ch:g12:phylogenetic-trees}

في سنة 1837، على صفحة من دفتر خاص، خطّ داروين خطًا متفرعًا بأغصان قليلة
وكتب فوقه كلمتين: "أظن". وقد اقترح الرسم أن \omterm{def:g10:biodiversity-scales:species}{الأنواع} تتقارب كما تتقارب
أغصان شجرة --- بالانحدار من فروع مشتركة --- وأن قرابتها يمكن أن يعاد
بناؤها. والرسم اليوم أداة قياسية: \omterm{def:g12:phylogenetic-trees:tree}{شجرة تطورية}، تبنى من صفات مشتركة ومن
\omterm{def:g10:universal-dna:information}{DNA}، وتقرأ وفق قواعد. ويعلّم هذا الفصل تلك القواعد: كيف تبنى شجرة من
جدول صفات، وكيف تقرأ من غير أن يساء قراءتها، وكيف تجعلها الجزيئات
ساعة.

\section{ماذا تقول الشجرة}

\begin{definition}[الشجرة التطورية]\label{def:g12:phylogenetic-trees:tree}
\emph{الشجرة التطورية}\index{شجرة تطورية} مخطط لقرابة طقم من \omterm{def:g10:biodiversity-scales:species}{الأنواع}
(أو من مجموعات أخرى). فأطرافها \omterm{def:g10:biodiversity-scales:species}{الأنواع} المقارنة؛ وكل \emph{عقدة}
يلتقي عندها فرعان تمثل أحدث سلف مشترك لهما --- وهو \omterm{def:g12:selection-drift-speciation:population}{جمهرة} مفترضة، لا
أحفورة معروفة؛ و\emph{الجذر} هو السلف المشترك للجميع. ويكون نوعان
أقرب قرابة من ثالث حين يكون سلفهما المشترك (عقدتهما) أحدث من العقدة
التي يشتركان فيها مع الثالث. والقرابة تقرأ من العقد، لا من موضع
الأطراف من اليسار إلى اليمين البتة.
\end{definition}

\begin{proposition}[قواعد القراءة]\label{prop:g12:phylogenetic-trees:rules}
\begin{itemize}
  \item يمكن أن تدار الشجرة عند أي عقدة من غير أن يتغير معناها: فترتيب
        الأطراف اعتباطي.
  \item العقدة لا اسم لها وليست طرفًا من الأطراف: فما من \omterm{def:g10:biodiversity-scales:species}{نوع} سلف \omterm{def:g10:biodiversity-scales:species}{لنوع}
        آخر على الشجرة؛ وإنما الطرفان \emph{مجموعتان شقيقتان}، منحدرتان
        من سلف مشترك.
  \item المجموعة المؤلفة من سلف و\emph{كل} نسله --- أي كل ما وراء عقدة
        واحدة --- هي \emph{فرع حيوي}\index{فرع حيوي}، وهو الصنف الوحيد
        من المجموعات الذي يسميه التصنيف التطوري.
  \item أطوال الأفرع لا تحمل معلومة إلا إذا قالت الشجرة ذلك (زمنًا، أو
        عدد تغيرات)؛ وإلا فترتيب التفرع وحده هو ما يعد.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[أخطاء قراءة تتجنب]\label{ex:g12:phylogenetic-trees:misreadings}
"الإنسان منحدر من الشمبانزي": لا --- فعلى الشجرة، الإنسان والشمبانزي
طرفان شقيقان، منحدران من سلف مشترك لم يكن أحدهما. و"الضفدع أكثر
بدائية": لا --- فكل طرف \omterm{def:g10:biodiversity-scales:species}{نوع} حاضر له طول التاريخ نفسه منذ الجذر؛ وسلالة
الضفدع تطورت مثل ما تطورت سلالتنا. و"السحلية أقرب إلى الضفدع منها إلى
الفأر" (من الشجرة A): لا --- فالسحلية والفأر يشتركان في عقدة أحدث.
\end{example}

\section{بناء شجرة من الصفات}

\begin{definition}[الحالة السلفية والحالة المشتقة]\label{def:g12:phylogenetic-trees:states}
\emph{الصفة} سمة تتغير بين \omterm{def:g10:biodiversity-scales:species}{الأنواع} المقارنة (كوجود عمود فقري، أو أربعة
أطراف، أو بيضة سلوية، أو شعر). ولكل صفة حالة \emph{سلفية} --- وهي التي
كانت عند السلف المشترك للمجموعة --- وحالة \emph{مشتقة} أو أكثر ظهرت
لاحقًا في سلالة ما. و\emph{الحالات المشتقة المشتركة} وحدها تدل على
القرابة: فنوعان يشتركان في حالة مشتقة قد ورثاها من السلف الذي نشأت
عنده. أما الحالة السلفية المشتركة فلا تقول شيئًا، إذ كانت عند سلف
الجميع.
\end{definition}

\begin{method}[من مصفوفة صفات إلى شجرة]\label{met:g12:phylogenetic-trees:build}
\begin{enumerate}
  \item اصنع جدولًا: \omterm{def:g10:biodiversity-scales:species}{الأنواع} في السطور، والصفات في الأعمدة، و1 للحالة
        المشتقة و0 للحالة السلفية. وقرر أي الحالتين سلفية بالمقارنة
        \emph{بمجموعة خارجية}، وهي \omterm{def:g10:biodiversity-scales:species}{نوع} يعلم أنه خارج المجموعة
        المدروسة.
  \item جد الحالة المشتقة التي يشترك فيها أكبر عدد من \omterm{def:g10:biodiversity-scales:species}{الأنواع}: فهي
        تحدد \omterm{prop:g12:phylogenetic-trees:rules}{الفرع الحيوي} الأول الأكبر. وداخله، جد الحالة المشتقة
        التالية الأوسع اشتراكًا، وهكذا.
  \item ارسم الأفرع الحيوية المتداخلة شجرةً: فكل حالة مشتقة مشتركة تعلّم
        عقدة واحدة، وتكتب الصفة على الفرع الذي ظهرت عنده.
  \item تفقّد التعارضات: فصفة تشترك فيها \omterm{def:g10:biodiversity-scales:species}{أنواع} تفرقها صفات أخرى إما أن
        تكون قد نشأت مرتين (تقارب) وإما أن تكون قد فقدت مرة. واختر
        الشجرة التي تقتضي أقل عدد من التغيرات جملةً --- أي الشجرة
        \emph{الأكثر اقتصادًا}.
\end{enumerate}
\end{method}

\begin{example}[تعارض يفصل فيه الاقتصاد]\label{ex:g12:phylogenetic-trees:parsimony}
أضف "الأجنحة" إلى مصفوفة فيها الخفاش والحمامة والفأر. فالأجنحة تجمع
الخفاش مع الحمامة؛ والشعر واللبن يجمعان الخفاش مع الفأر. والشجرة التي
تضم الخفاش والفأر معًا تقتضي أن تكون الأجنحة قد نشأت مرتين (تغيران)؛
أما الشجرة التي تضم الخفاش والحمامة معًا فتقتضي أن يكون الشعر واللبن
والفك الثديي وعشرات الصفات الأخرى قد نشأت مرتين كل منها. فالشجرة
الأولى أكثر اقتصادًا بكثير: فالأجنحة تقارب، والصفات الثديية إرث مشترك.
والاقتصاد لا يبرهن الشجرة؛ بل يختار أقلها استبعادًا.
\end{example}

\section{أشجار من الجزيئات}

\begin{proposition}[التطور الجزيئي]\label{prop:g12:phylogenetic-trees:molecular}
\omterm{def:g10:universal-dna:gene}{المورّثة} أو \omterm{def:g11:gene-expression:protein}{البروتين} الذي تشترك فيه \omterm{def:g10:biodiversity-scales:species}{الأنواع} المقارنة كلها مصفوفة صفات
في ذاته: فكل موضع من التسلسل صفة، وكل \omterm{def:g10:universal-dna:nucleotide}{نوكليوتيد} أو \omterm{def:g10:chemistry-of-life:families}{حمض أميني} حالة.
ومحاذاة التسلسلات وإحصاء الفروق بين كل زوج يعطيان مصفوفة مسافات، تبنى
منها شجرة بضم أقرب الأزواج أولًا. والشجرة الجزيئية، محسوبة من تسلسلات
لا تدين للتشريح بشيء، توافق الشجرة التشريحية في الأغلبية العظمى من
الحالات --- وهو أقوى اختبار لكلتيهما.
\end{proposition}
\begin{proof}[شواهد]
أعطى \cref{ch:g10:common-ancestry} مسافات السيتوكروم c عند \omterm{def:g10:common-ancestry:bodyplan}{الفقاريات}،
وهي تعيد إنتاج الأطر المرسومة من هياكلها. وبين الرئيسيات، تتفق تسلسلات
عشرات \omterm{def:g10:universal-dna:gene}{المورّثات}: فالإنسان والشمبانزي يختلفان بنحو 1.2\% من \omterm{def:g10:universal-dna:information}{DNA}،
ويختلف كل منهما عن الغوريلا بنسبة 1.6\%، وتختلف الثلاثة عن
الأورانغوتان بنسبة 3\%، وتختلف القردة العليا عن المكاك بنسبة 7\%.
وللشجرة المبنية من أي واحدة من هذه \omterm{def:g10:universal-dna:gene}{المورّثات} الشكل نفسه؛ والاستثناءات
القليلة تخص أحدث العقد، حيث تكون الفروق أصغر ما تكون.
\end{proof}

\begin{proposition}[الساعة الجزيئية]\label{prop:g12:phylogenetic-trees:clock}
عند \omterm{def:g10:universal-dna:gene}{مورّثة} بعينها، تتراكم الفروق بين سلالتين بمعدل ثابت تقريبًا على
مدد طويلة، إذ إن معظم الاستبدالات حيادية وتثبَّت بالانسياق بالمعدل الذي
تنشأ به. ومن ثم يتناسب عدد الفروق بين نوعين مع الزمن المنقضي منذ سلفهما
المشترك. والساعة، معايرةً على عقدة واحدة مؤرخة بالأحافير، تؤرخ سائر
العقد --- مع التحفظ بأن المعدلات تختلف بين \omterm{def:g10:universal-dna:gene}{المورّثات} وبين السلالات،
وبأن المعايرة تحمل عدم يقينها.
\end{proposition}
\admitted

\begin{example}[تأريخ عقدة]\label{ex:g12:phylogenetic-trees:dating}
نوعان يختلفان بنسبة 4.5\% من \omterm{def:g10:universal-dna:information}{DNA}؛ وعند 0.27\% لكل مليون سنة، يكون
سلفهما المشترك قد عاش قبل نحو 17 مليون سنة. والتقدير يفترض أن معدل
الرئيسيات ينطبق؛ فلمورّثة تتطور بضعف تلك السرعة، أو في سلالة أقصر
أجيالًا، كانت النسبة 4.5\% نفسها تعني نصف ذلك الزمن. فالساعة لا تجود
إلا بقدر معايرتها.
\end{example}

\section{التصنيف، من الأشجار}

\begin{proposition}[التصنيف بالانحدار]\label{prop:g12:phylogenetic-trees:classification}
التصنيف التطوري لا يسمي إلا الأفرع الحيوية: أي المجموعات المحددة بعقدة
والحاوية كل نسلها. أما المجموعات المألوفة التي تسقط في هذا الاختبار
فلا تستعمل: أما "الأسماك" فتستثني رباعيات الأطراف المنحدرة من العقدة
نفسها؛ و"الزواحف" تستثني الطيور المنحدرة من داخلها؛ و"اللافقاريات" كل
شيء إلا فرعًا حيويًا. والنتائج تستحق أن تذكر صريحة: فالطيور ديناصورات،
والسمكة الرئوية أقرب إلينا منها إلى التروتة، والإنسان فرع من القردة
العليا، وهي فرع من الرئيسيات، ومن الثدييات، ومن السلويات، ومن رباعيات
الأطراف، ومن \omterm{def:g10:common-ancestry:bodyplan}{الفقاريات}.
\end{proposition}
\admitted

\begin{remark}[ما ليست الشجرة]\label{rem:g12:phylogenetic-trees:not}
الشجرة فرضية عن التاريخ، تبنى من \omterm{def:g10:biodiversity-scales:species}{الأنواع} الحاضرة وتختبرها كل صفة جديدة
وكل تسلسل جديد؛ وتعاد رسمًا حين تتعارض، وإعادة الرسم هي عمل هذا الحقل.
ولا قمة لها ولا اتجاه تقدم: فالإنسان يقع عند طرف واحد بين ملايين، لا
أعلى ولا أتم من الجلكى عند طرف آخر. ولا سلف لها عند أي طرف: فالأسلاف
هي العقد، \omterm{def:g12:selection-drift-speciation:population}{جمهرات} زالت، تستدل الشجرة على وجودها، وتقع أحافيرها، متى
وجدت، إلى جانب الأفرع لا عليها.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:phylogenetic-trees:1}
ماذا تمثل أطراف \omterm{def:g12:phylogenetic-trees:tree}{الشجرة التطورية} وعقدها وجذرها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:phylogenetic-trees:2}
في الشجرة A من الشكل الأول، أي \omterm{def:g10:biodiversity-scales:species}{الأنواع} أقرب أقرباء السحلية؟ وأيها أقرب
أقرباء الضفدع؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:phylogenetic-trees:3}
عرّف الحالة المشتقة المشتركة وفسّر لماذا لا تدل الحالة السلفية المشتركة
على القرابة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:phylogenetic-trees:4}
ما \omterm{prop:g12:phylogenetic-trees:rules}{الفرع الحيوي}؟ أ"الطيور" \omterm{prop:g12:phylogenetic-trees:rules}{فرع حيوي}؟ أ"الزواحف" (بالمعنى الدارج) كذلك؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:phylogenetic-trees:5}
من شجرة الرئيسيات، من أقرب أقرباء الإنسان، وأي العقد أقدم؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:phylogenetic-trees:6}
أعد رسم الشجرة A من الشكل الأول والإنسان في الأعلى والضفدع في الأسفل،
وتحقق أن كل قرابة لم تتغير.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:phylogenetic-trees:7}
أضف الحمامة (فكوك 1، أطراف 1، سلى 1، شعر 0، ريش 1) إلى مصفوفة الشكل
الثاني وضعها على الشجرة. ماذا تضيف صفة "الريش"؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:phylogenetic-trees:8}
للدلفين جسم انسيابي وزعانف كزعانف القرش، وله شعر ولبن وفك ثديي. أي
الشجرتين أكثر اقتصادًا، وما الزعانف؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:phylogenetic-trees:9}
مستعينًا بشكل الساعة، أرّخ السلف المشترك لنوعين يختلف \omterm{def:g10:universal-dna:information}{DNA} عندهما بنسبة
2.2\%.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:phylogenetic-trees:10}
فسّر لماذا تسيء جملة "الشمبانزي سلفنا" قراءة الشجرة، واكتب الجملة
الصحيحة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:phylogenetic-trees:11}
لماذا استعمل الجلكى مجموعةً خارجية في الشكل الثاني، وماذا كان سيخطئ لو
استعمل الفأر مكانه؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:phylogenetic-trees:12}
مورّثتان تعطيان شجرتين مختلفتين لثلاثة \omterm{def:g10:biodiversity-scales:species}{أنواع} وثيقة القرابة انفصلت في
غضون مليون سنة بعضها عن بعض. فسّر كيف يقع ذلك من غير أن تكون أي من
الشجرتين "خاطئة"، مستعينًا بخلط \omterm{def:g10:universal-dna:gene}{الأليلات} في \omterm{def:g12:selection-drift-speciation:population}{الجمهرة} السلفية.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:phylogenetic-trees:13}
الساعة الجزيئية لفيروس سريع التطور تمشي بنسبة 1\% في السنة؛ وساعة
\omterm{def:g10:universal-dna:gene}{مورّثة} ثديية بنسبة 0.3\% لكل مليون سنة. فسّر لماذا تستعمل الأولى لتتبع
وباء والثانية لتأريخ أصل الثدييات، ولماذا لا تستطيع أي منهما عمل
الأخرى.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:phylogenetic-trees:14}
وجد طائر أحفوري ذو أسنان وذنب عظمي طويل في صخور عمرها 150 مليون سنة.
فسّر لماذا يوضع إلى جانب فرع من الشجرة لا عند عقدة، وماذا يخبر عن
العقدة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:phylogenetic-trees:15}
"شجرة الحياة على قمتها الإنسان." ناقش في فقرة مستعينًا بقواعد القراءة،
وبتساوي أعمار الأطراف، وبعدد الأطراف.
\end{exercise}

\section{مسألة: ستة أنواع وساعة}

\begin{problem}[{مسألة نهاية الأسبوع --- شجرة تبنى مرتين، من جدول صفات
ومن مسافات DNA، وتقارن الشجرتان، وتؤرخ العقد}]
\label{pb:g12:phylogenetic-trees:1}
ستة \omterm{def:g10:biodiversity-scales:species}{أنواع}: القرش والسلمون والسمندل والتمساح والحمامة والقط. والمجموعة
الخارجية هي الجلكى. وجدول الصفات (1 = مشتقة):

\begin{center}
\small
\begin{tabular}{lcccccc}
 & هيكل عظمي & أربعة أطراف & بيضة سلوية & ريش & شعر & فتحة جمجمية \\ \hline
قرش & 0 & 0 & 0 & 0 & 0 & 0 \\
سلمون & 1 & 0 & 0 & 0 & 0 & 0 \\
سمندل & 1 & 1 & 0 & 0 & 0 & 0 \\
تمساح & 1 & 1 & 1 & 0 & 0 & 0 \\
حمامة & 1 & 1 & 1 & 1 & 0 & 0 \\
قط & 1 & 1 & 1 & 0 & 1 & 1 \\
\end{tabular}
\end{center}

فروق \omterm{def:g10:universal-dna:information}{DNA} (\%) عند \omterm{def:g10:universal-dna:gene}{مورّثة} واحدة:

\begin{center}
\begin{tabular}{lccccc}
 & سلمون & سمندل & تمساح & حمامة & قط \\ \hline
قرش & 30 & 31 & 30 & 31 & 30 \\
سلمون & & 24 & 25 & 25 & 24 \\
سمندل & & & 19 & 20 & 19 \\
تمساح & & & & 8 & 15 \\
حمامة & & & & & 15 \\
\end{tabular}
\end{center}

\medskip
\noindent\textbf{القسم الأول --- من الصفات.}
\begin{enumerate}
  \item أي حالة مشتقة يشترك فيها أكبر عدد من \omterm{def:g10:biodiversity-scales:species}{الأنواع}؟ سمّ \omterm{prop:g12:phylogenetic-trees:rules}{الفرع الحيوي}
        الذي تحدده \omterm{def:g10:biodiversity-scales:species}{والأنواع} الباقية خارجه. (والصفة الأخيرة هي الفتحة
        الوحيدة في جانب الجمجمة التي عند الثدييات.)
  \item تابع بالحالات التالية الأوسع اشتراكًا واذكر الأفرع الحيوية
        المتداخلة.
  \item الريش والشعر والفتحة الجمجمية توجد كل منها عند \omterm{def:g10:biodiversity-scales:species}{نوع} واحد. أتعين
        على بناء هذه الشجرة؟ وما الذي كان سيجعلها مفيدة؟
  \item ارسم الشجرة، كاتبًا كل صفة على الفرع الذي ظهرت عنده.
  \item أي \omterm{def:g10:biodiversity-scales:species}{الأنواع} أقرب أقرباء التمساح بحسب هذا الجدول؟ وأي زوج لا
        يمكن الفصل فيه بالجدول وحده؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثاني --- من \omterm{def:g10:universal-dna:information}{DNA}.}
\begin{enumerate}[resume]
  \item أي زوج من \omterm{def:g10:biodiversity-scales:species}{الأنواع} أصغر مسافة؟ اضممهما أولًا.
  \item أي \omterm{def:g10:biodiversity-scales:species}{نوع} ينضم إلى ذلك الزوج بعده؟ تابع حتى تتم الشجرة.
  \item قارن شجرة \omterm{def:g10:universal-dna:information}{DNA} بشجرة الصفات: أين تتفقان، وما الذي يفصل فيه \omterm{def:g10:universal-dna:information}{DNA}
        مما لم تستطع الصفات الفصل فيه؟
  \item مسافات القرش إلى سائر \omterm{def:g10:biodiversity-scales:species}{الأنواع} متساوية تقريبًا. فسّر لماذا، من
        شكل الشجرة.
  \item مسافة التمساح--الحمامة (8) أصغر من مسافة التمساح--القط (15).
        فماذا يقول ذلك عن المجموعة الدارجة "الزواحف"؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثالث --- الساعة.}
تؤرخ الأحافير انفصال سلالة السمندل عن السلويات عند 340 مليون سنة.
\begin{enumerate}[resume]
  \item مستعينًا بمسافات السمندل--السلويات (نحو 19\%)، احسب معدل
        الساعة بالنسبة المئوية لكل مليون سنة. (فالسلالتان كلتاهما ما
        زالت تتغير: والمسافة تحصي التغيرات على الفرعين معًا.)
  \item أرّخ عقدة التمساح--الحمامة وعقدة السلويات (التمساح والحمامة في
        مقابل القط).
  \item أرّخ عقدة السلمون وعقدة القرش.
  \item تعطي الأحافير نحو 250 مليون سنة لانفصال التمساح والطير و310
        لانفصال السلويات. قارن بتواريخك وعلّق على دقة ساعة مبنية على
        \omterm{def:g10:universal-dna:gene}{مورّثة} واحدة.
  \item مسافات القرش (30--31\%) لا تكاد تتجاوز مسافات السلمون
        (24--25\%) مع أن عقدة القرش أقدم بكثير. اقترح تفسيرًا، وقل ماذا
        يستلزم ذلك في تأريخ العقد القديمة جدًا بهذه \omterm{def:g10:universal-dna:gene}{المورّثة}.
\end{enumerate}

\medskip
\noindent\textbf{القسم الرابع --- القراءة والتسمية.}
\begin{enumerate}[resume]
  \item اذكر الأفرع الحيوية في شجرتك التي تحوي القط، من الأصغر إلى
        الأكبر.
  \item أ"الأسماك" (القرش والسلمون) \omterm{prop:g12:phylogenetic-trees:rules}{فرع حيوي}؟ وأ"السلويات" كذلك؟ برهن.
  \item يقول طالب إن السمندل "وسط بين الأسماك والزواحف". صحّح العبارة
        بلغة الشجرة.
  \item وجدت أحفورة ذات هيكل عظمي وأربعة أطراف وبلا بيضة سلوية، مؤرخة
        عند 360 مليون سنة. أين تقع من الشجرة، وماذا يؤكد تاريخها؟
  \item اذكر النتيجة: شجرة \omterm{def:g10:biodiversity-scales:species}{الأنواع} الستة (بأقواس متداخلة)، والقرابة
        الوحيدة التي فصل فيها \omterm{def:g10:universal-dna:information}{DNA}، والتاريخ المقدَّر لانفصال التمساح
        والحمامة.
\end{enumerate}
\end{problem}
