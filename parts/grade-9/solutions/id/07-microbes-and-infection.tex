\section*{Bab \ref{ch:g9:microbes-and-infection} --- Mikroba dan Infeksi}

\begin{solution}{exo:g9:microbes-and-infection:1}
\omterm{def:g9:microbes-and-infection:sorted}{Bakteri} (\omterm{def:g5:first-look-at-cells:cell}{sel} tunggal tanpa \omterm{def:g6:cell-unit-of-life:parts}{inti sel}), fungi \omterm{def:g6:cell-unit-of-life:unicellular}{bersel tunggal}
seperti \omterm{ex:g6:producing-our-food:bread}{ragi} dan kehidupan \omterm{def:g6:cell-unit-of-life:unicellular}{bersel tunggal} lainnya di
kolamnya --- dan \omterm{def:g9:microbes-and-infection:sorted}{virus}, yang berbeda jenisnya: tanpa \omterm{def:g5:first-look-at-cells:cell}{sel},
tanpa kehidupannya sendiri, hanya teks genetik yang dikemas
lalu meminjam permesinan sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} yang hidup.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:2}
Triliunan \omterm{def:g9:microbes-and-infection:sorted}{bakteri} tak merugikan yang menetap di \omterm{def:g1:the-five-senses:sense}{kulit} dan di
dalam ususnya. Jasanya: membantu \omterm{def:g4:journey-of-food:digestion}{pencernaannya}, dan
menduduki lahannya --- mendesak \omterm{prop:g9:microbes-and-infection:mostly}{patogennya} turun dari titik
pendaratannya.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:3}
Penularan (tetesan sebuah batuk); masuk (sebuah sayatan,
saluran napasnya); perbanyakan (pembelahan yang berlipat dua
tiap jam, atau \omterm{def:g5:first-look-at-cells:cell}{sel} yang dibajak menyalin); penyakit (denyut
nyeri jari yang bernanah, tenggorokan perih pileknya).
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:4}
\omterm{def:g9:microbes-and-infection:sorted}{Bakteri} luka sayatnya memperbanyak diri di antara \omterm{def:g5:first-look-at-cells:cell}{sel} di
dalam jaringan yang hangat dan basah, sambil meracuni
tetangganya. \omterm{def:g9:microbes-and-infection:sorted}{Virus} pileknya memperbanyak diri di dalam \omterm{def:g5:first-look-at-cells:cell}{sel}
lapisan saluran napasnya, dan tiap \omterm{def:g5:first-look-at-cells:cell}{sel} yang dibajak
melepaskan segerombol salinan.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:5}
\omterm{def:g1:the-five-senses:sense}{Kulit}: dindingnya, pada tahap masuknya. Sapu dan rendaman
lapisan dalamnya (pembersihan saluran napas, air \omterm{def:g1:the-five-senses:sense}{mata}, asam
\omterm{def:g4:journey-of-food:tract}{lambung}): masuk dan perbanyakan awalnya. \omterm{prop:g9:microbes-and-infection:mostly}{Flora penetapnya}:
masuk --- lahannya sudah diduduki. Sabun, memasak dan dingin
pada \omterm{def:g1:healthy-habits:hygiene}{kebersihan dirinya}: penularan dan perbanyakannya.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:6}
Penularannya --- pengembara demamnya yang terbawa di tangan
para dokter dari ruang bedah mayat ke bangsal bersalinnya.
Pasteur memperlihatkan para pengembaranya sendiri: \omterm{def:g6:producing-our-food:fermentation}{mikroba}
sebagai pelaku pengasaman, penyakit dan \omterm{def:g9:microbes-and-infection:stages}{infeksinya} --- dan
panas sebagai penghentinya.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:7}
Antibiotik melaparkan lalu mematahkan \omterm{def:g9:microbes-and-infection:sorted}{bakteri} --- \omterm{def:g5:first-look-at-cells:cell}{sel} yang
mandiri, dengan permesinannya sendiri yang dapat diracuni.
\omterm{def:g9:microbes-and-infection:sorted}{Virus} sebuah pilek tidak punya permesinannya sendiri: ia
berjalan di atas permesinan \emph{milikmu}, dan sebuah obat
yang dibidikkan pada kerjanya berarti dibidikkan pada \omterm{def:g5:first-look-at-cells:cell}{sel}
milikmu --- itulah sebabnya obat \omterm{def:g9:microbes-and-infection:sorted}{bakterinya} membiarkan
virusnya tak tersentuh.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:8}
Cuci tangan sebelum makan: menghadang penularan
tangan-ke-mulut. Cuci tangan sesudah dari kakus: menghadang
gelung keluar-ke-masuk jalur ususnya. Membuang ingus dan
menutup batuk (serta \omterm{def:g1:my-body:joint}{siku} untuk bersinnya): menghadang
penularan tetesan pada sumbernya. (Aturan gigi dan makanannya
menghadang tempat perbanyakan di mulutnya.)
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:9}
\omterm{prop:g9:microbes-and-infection:mostly}{Patogennya}: \omterm{def:g9:microbes-and-infection:sorted}{bakteri} keracunan makanan, pada ayamnya.
Penularannya: hidangannya sendiri. Masuknya: ususnya.
Diperbanyak oleh: tiga jam hangat di dalam mobilnya ---
aturan \omterm{def:g6:exploring-our-environment:conditions}{kondisi} yang menyuapi pelipatduaan tiap setengah jam
atau lebih. Penghadang termurah: rantai dinginnya --- sebuah
kotak pendingin (atau memakannya segar); yang termurah
berikutnya, memasaknya sampai matang lebih dahulu.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:10}
Karena tangan adalah kurir antarpetak: dokter Semmelweis
sudah bersih menurut ukuran sehari-hari mana pun, namun tetap
membawa sang pembunuh bangsalnya dari mayat ke ibunya.
Cucian yang penting adalah cucian \emph{di batasnya} ---
ayam mentah ke salad, kakus ke meja makan --- yaitu tempat
jalurnya akan tersambung bila tidak.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:11}
Antibiotiknya menyerang perbanyakannya --- mematahkan
\omterm{def:g9:microbes-and-infection:sorted}{bakteri} yang sedang membelah. Dipakai compang-camping, ia
menjadi penahan seleksi yang dijalankan setengah jalan: yang
rentan mati, minoritas yang tahan mewarisi bangsalnya ---
membiakkan \omterm{prop:g9:microbes-and-infection:mostly}{patogen} yang akan ditemui pasien berikutnya.
\end{solution}

\begin{solution}{exo:g9:microbes-and-infection:12}
Teksnya: sepotong huruf genetik, dengan abjad yang sama
seperti abjad tiap \omterm{def:g5:first-look-at-cells:cell}{selnya}
(\cref{rem:g9:genes-and-dna:universal}). Mesin cetaknya:
permesinan pembacaan-dan-pembangunan sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} yang hidup,
yang dipalingkan oleh teks yang masuk itu untuk mencetak
salinan dirinya. Tepi kehidupannya: sendirian, virusnya
tidak melakukan apa pun pada daftar
\cref{prop:g1:living-or-not:signs} --- tanpa makan, tanpa
tumbuh, tanpa membelah; setelah meminjam, ia menjalankan
sebuah parodi reproduksi. Sukar diobati: tidak ada permesinan
\omterm{def:g9:microbes-and-infection:sorted}{virus} untuk diracuni --- hanya mesin cetaknya, dan mesin
cetaknya adalah kamu.
\end{solution}

\begin{solution}{pb:g9:microbes-and-infection:1}
\textbf{1.} Penularan lewat tetesan --- terbatukkan dan
terucapkan --- dibantu permukaan bersama pada bet tenis
mejanya; masuknya lewat saluran napas.

\textbf{2.} Perbanyakannya: \omterm{def:g5:first-look-at-cells:cell}{sel} lapisan yang dibajak pada
tiap kasusnya mencetak salinan yang menyemai kasus
berikutnya --- pelipatduaannya adalah permesinan pinjaman
virusnya, yang berbunga.

\textbf{3.} Penyakitnya. Dari kerusakan \omterm{def:g5:first-look-at-cells:cell}{sel} lapisan yang
dibajak lalu tumbang --- dan, sebagian, dari pertempuran
pertahanannya sendiri, seperti dirinci bab berikutnya.

\textbf{4.} Karena rapatnya menyambungkan jangkauan tetesan
tiap kelasnya selama sejam: penularannya berskala dengan
kepadatannya, dan dua pencampuran terpadat sekolahnya adalah
jalur antarpetaknya.

\textbf{5.} Murid yang sakit dipulangkan: penularannya
disingkirkan pada sumbernya. Jendelanya: takaran tetesannya
diencerkan --- penularan. Cuci tangannya: jalur permukaannya
diputus --- penularan. Melap perlengkapannya: jalur benda
bersamanya diputus --- penularan. (Semuanya dibidikkan pada
tahap termurah yang sama.)

\textbf{6.} ``Antibiotik membunuh \omterm{def:g9:microbes-and-infection:sorted}{bakteri}; agen ini adalah
\omterm{def:g9:microbes-and-infection:sorted}{virus} yang berjalan di atas \omterm{def:g5:first-look-at-cells:cell}{sel} anak-anak itu sendiri ---
obatnya tak dapat menyentuhnya, dan menaburkannya tetap saja
hanya akan menjalankan ayakan seleksi atas \omterm{def:g9:microbes-and-infection:sorted}{bakteri} penetap
semua orang, membiakkan ketahanan untuk hari ketika
antibiotiknya sungguh dibutuhkan.''

\textbf{7.} Agennya: \omterm{def:g9:microbes-and-infection:sorted}{bakteri} keracunan makanan.
Penularannya: pencuci mulutnya. Masuknya: ususnya.
Perbanyakannya: berjam-jam tanpa pendingin --- kehangatan,
\omterm{def:g6:exploring-our-environment:conditions}{kelembapan}, makanan, waktu, semuanya diberikan. Penghadangnya:
lemari esnya --- \omterm{def:g6:exploring-our-environment:conditions}{kondisinya} ditolak; kedua wabahnya berbagi
sebuah sekolah dan tidak ada yang lain.

\textbf{8.} Sebuah peta wabah adalah penularan yang dibuat
terlihat: siapa-duduk-di-mana menggambar jalurnya --- baris,
pasangan, rapatnya --- dan jalurnya, begitu tergambar,
memperlihatkan di mana harus dipotong. Pengobatan acak
melawan kasusnya; petanya melawan \omterm{def:g6:how-plants-spread:dispersal}{penyebarannya}.

\textbf{9.} Penularannya: tetesan kasus penunjuknya, baris
dan pasangannya lebih dahulu, lalu rapatnya ke seluruh
sekolah. Masuknya: saluran napas, kelas demi kelas.
Perbanyakannya: pelipatduaan dua harian sepanjang pekan
pertamanya. Penyakit dan padamnya: demamnya memuncak,
tindakannya memotong jalurnya, kolam yang rentan menipis ---
kurvanya berbalik turun.

\textbf{10.} Penghadang tindakannya melaparkan jalurnya ---
dan kolamnya sendiri lebih kecil daripada daftar hadirnya:
murid yang mengangkut kekebalan galur sepupu musim dingin
lalu tak dapat dinyalakan, dan sebuah wabah mati ketika
terlalu sedikit yang masih dapat tertular: ingatan bab
berikutnya, ditayangkan lebih dahulu.

\textbf{11.} Pelajaran Semmelweis: potong jalur kurirnya ---
pengasingan di rumah, tangan yang dicuci, bet yang dilap.
Pelajaran Pasteur: kenali pengembaranya --- sebutkan agennya,
tolak \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}, lalu bidikkan obatnya pada jenisnya dan
bukan pada kebiasaan.

\textbf{12.} ``Wabah paling murah dihentikan pada
penularannya --- sabun, udara, jarak dan sebuah kotak
pendingin yang tertutup mengalahkan obat apa pun yang
menunggu sampai perbanyakannya.''
\end{solution}
