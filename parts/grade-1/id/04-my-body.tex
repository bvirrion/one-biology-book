\chapter{Tubuhku}\label{ch:g1:my-body}

Berdirilah di depan cermin lalu melambailah: satu tubuh hidup yang
utuh melambai balik. Tubuh itu sudah tumbuh sejak sebelum kamu lahir,
ia menekuk di beberapa tempat dan tidak di tempat lain, dan ia milikmu
seumur hidup. Saatnya memandangnya seperti kita memandang kucing dan
\omterm{def:g1:plants-around-us:parts}{bunga} aster --- bagian demi bagian.

\section{Bagian-bagian tubuhku}

\begin{definition}[Bagian-bagian tubuh manusia]\label{def:g1:my-body:parts}
Tubuh manusia mempunyai tiga bagian utama:
\begin{enumerate}
  \item \emph{kepala}\index{kepala}, yang membawa wajah, mata,
        telinga, hidung dan mulut;
  \item \emph{batang tubuh}\index{batang tubuh}, bagian tengah tubuh
        --- dada, perut dan punggung;
  \item \emph{anggota gerak}\index{anggota gerak}: dua
        \emph{lengan}\index{lengan} yang berujung pada tangan, dua
        \emph{tungkai}\index{tungkai} yang berujung pada kaki.
\end{enumerate}
Kepalanya bertumpu pada \emph{leher}\index{leher}, yang
menyambungkannya ke batang tubuh.
\end{definition}

\begin{example}[Rancangan yang sama dengan kucing?]\label{ex:g1:my-body:cat}
Bandingkan dirimu dengan kucing: kepala, tubuh, empat \omterm{def:g1:my-body:parts}{anggota gerak}
--- rancangannya tampak mirip. Tetapi kucing berjalan dengan keempatnya;
kamu berdiri di atas dua \omterm{def:g1:my-body:parts}{tungkai} dan membiarkan dua \omterm{def:g1:my-body:parts}{lengan} bebas untuk
membawa, menggambar dan melambai. Dan di tempat kucing punya ekor, kamu
tidak punya.
\end{example}

\section{Menekuk: sendi}

\begin{definition}[Sendi]\label{def:g1:my-body:joint}
Suatu \emph{sendi}\index{sendi} adalah tempat tubuh dapat menekuk atau
berputar. \emph{Siku}\index{siku} menekukkan \omterm{def:g1:my-body:parts}{lengan},
\emph{lutut}\index{lutut} menekukkan \omterm{def:g1:my-body:parts}{tungkai},
\emph{bahu}\index{bahu} memutar seluruh \omterm{def:g1:my-body:parts}{lengan},
\emph{pergelangan tangan}\index{pergelangan tangan} dan
\emph{pergelangan kaki}\index{pergelangan kaki} menekukkan tangan dan
kaki. Di antara dua sendi, tubuh tetap lurus.
\end{definition}

\begin{example}[Cobalah: berjalan berlutut kaku]\label{ex:g1:my-body:stiff}
Cobalah berjalan tanpa menekuk lututmu. Bisa --- meski nyaris tidak
--- dengan langkah kecil kaku seperti bebek. Sekarang cobalah memungut
pensil dari lantai tanpa menekuk \omterm{def:g1:my-body:joint}{lutut}, pinggul atau punggung. Sendilah
yang membuat gerakan tubuh menjadi mulus, cepat dan mudah.
\end{example}

\begin{remark}[Kanan dan kiri]\label{rem:g1:my-body:sides}
Tubuh mempunyai dua sisi, kanan dan kiri, hampir seperti cermin: satu
\omterm{def:g1:my-body:parts}{lengan} dan satu \omterm{def:g1:my-body:parts}{tungkai} di tiap sisi, satu mata dan satu telinga di
tiap sisi kepala. Mengetahui mana tangan kananmu dan mana kirimu juga
pengetahuan tentang tubuh --- ia memberi tahu orang arah yang kamu
maksud.
\end{remark}

\section{Tubuhku tumbuh}

\begin{proposition}[Bertambah besar]\label{prop:g1:my-body:growth}
Tubuh seorang anak \emph{tumbuh}\index{tumbuh}: ia bertambah tinggi
dan bertambah berat, tahun demi tahun, sampai seukuran orang dewasa.
Tumbuh berlangsung pelan dan tetap --- terlalu pelan untuk dirasakan,
mudah untuk diukur.
\end{proposition}
\admitted

\begin{example}[Coretan pada kusen pintu]\label{ex:g1:my-body:doorframe}
Berdirilah merapat ke kusen pintu; seorang dewasa menandai tinggimu
dengan pensil. Lakukan lagi beberapa bulan kemudian: tanda barunya
lebih tinggi! Kamu tidak pernah merasa dirimu sedang tumbuh, tetapi
kedua tanda itu, yang satu di atas yang lain, memperlihatkannya dengan
jelas.
\end{example}

\begin{method}[Mengukur tinggi badanku]\label{met:g1:my-body:measure}
\begin{enumerate}
  \item Lepaskan sepatumu;
  \item berdirilah tegak, punggung dan tumit merapat ke dinding,
        pandangan lurus ke depan;
  \item seorang penolong meletakkan buku datar di atas kepalamu,
        mendatar, menyentuh dinding;
  \item tandai titik di bawah bukunya, dan tuliskan tanggalnya di
        sebelahnya.
\end{enumerate}
Diukur dengan cara yang sama setiap kali, tanda-tandanya dapat
dipercaya --- dan dibandingkan.
\end{method}

\begin{remark}[Setiap tubuh berbeda]\label{rem:g1:my-body:different}
Dalam satu kelas, ada anak yang lebih tinggi, ada yang lebih pendek;
ada yang tumbuh lebih awal, ada yang lebih lambat. Semua itu wajar.
Tubuh itu seperti wajah: dibangun atas rancangan yang sama, dan
masing-masing berbeda. Tanda-tanda di kusen pintumu adalah milikmu ---
bukan untuk berlomba dengan tetangga.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g1:my-body:1}
Sebutkan tiga bagian utama tubuh manusia. Apa yang menyambungkan
kepala ke batang tubuh?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:2}
Di mana tiap \omterm{def:g1:my-body:parts}{anggota gerak} berakhir --- apa yang ada di ujung \omterm{def:g1:my-body:parts}{lengan},
dan di ujung \omterm{def:g1:my-body:parts}{tungkai}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:3}
\omterm{def:g1:my-body:joint}{Sendi} mana yang menekuk: (a) \omterm{def:g1:my-body:parts}{lengan} di bagian tengahnya? (b) \omterm{def:g1:my-body:parts}{tungkai}
di bagian tengahnya? (c) tangan di ujung \omterm{def:g1:my-body:parts}{lengan}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:4}
Sentuhlah telinga kananmu dengan tangan kirimu. \omterm{def:g1:my-body:joint}{Sendi} mana saja yang
baru saja dipakai lenganmu? Sebutkan sedikitnya dua.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:5}
Pada \cref{ex:g1:my-body:cat}, temukan satu hal yang membuat tubuhmu
mirip tubuh kucing dan dua hal yang membuatnya berbeda.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:6}
Mengapa penolong pada \cref{met:g1:my-body:measure} meletakkan buku
datar di atas kepalamu dan tidak sekadar melihat saja?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:7}
Dua tanda pada kusen pintu: tanda tahun lalu, dan tanda hari ini yang
sedikit lebih tinggi. Apa yang diperlihatkan kedua tanda itu? Apakah
kamu merasakannya terjadi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:8}
Cobalah: lakukan jalan berlutut kaku pada \cref{ex:g1:my-body:stiff},
lalu cobalah bertepuk tangan tanpa menekuk \omterm{def:g1:my-body:joint}{siku}. Gerakan sehari-hari
mana saja yang ternyata membutuhkan \omterm{def:g1:my-body:joint}{sendi}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:9}
Mia paling kecil di kelasnya dan mencemaskan hal itu. Apa jawaban
\cref{rem:g1:my-body:different} untuknya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:10}
Sebuah boneka yang terbuat dari satu batang kayu tidak dapat duduk di
kursi. Di mana kamu akan memotong tongkatnya dan menambahkan titik
putar agar ia dapat duduk? Sebutkan \omterm{def:g1:my-body:joint}{sendi} tubuh yang ditiru oleh
potonganmu.
\end{exercise}
