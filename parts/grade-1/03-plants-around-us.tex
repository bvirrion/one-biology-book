\chapter{Plants Around Us}\label{ch:g1:plants-around-us}

A daisy in the lawn, moss on the wall, the big chestnut \omterm{def:g1:plants-around-us:tree}{tree} in the
schoolyard: none of them will ever run away, yet they are all
alive. \omterm{def:g1:plants-around-us:plant}{Plants} are the quiet \omterm{def:g1:living-or-not:living}{living things} --- they are born, they
feed, they grow, sometimes for hundreds of years, all without
moving from their spot.

\section{Plants are living things}

\begin{definition}[Plant]\label{def:g1:plants-around-us:plant}
A \emph{plant}\index{plant} is a \omterm{def:g1:living-or-not:living}{living thing} that stays fixed in
the ground and makes its own food, using light. Grass, daisies,
tomato plants and \omterm{def:g1:plants-around-us:tree}{trees} are all plants. Most plants are green.
\end{definition}

\begin{example}[The daisy is alive]\label{ex:g1:plants-around-us:daisy}
Check the signs of life on a daisy: it was born from a seed; it
feeds --- with water from the soil and light from the sky; it
grows; it makes seeds that become new daisies; it dies in the
frost. All the signs are there. The daisy is as alive as the cat
--- it is just much quieter about it.
\end{example}

\begin{remark}[Alive without running]\label{rem:g1:plants-around-us:still}
\omterm{def:g1:plants-around-us:plant}{Plants} do not walk, but they are not completely still. A sunflower
turns to follow the light during the day, and a \omterm{def:g1:plants-around-us:plant}{plant} on a
windowsill slowly leans toward the window. \omterm{def:g1:plants-around-us:plant}{Plants} move by
\emph{growing} --- too slowly for our eyes, fast enough for a
week's watching.
\end{remark}

\section{The parts of a plant}

\begin{definition}[Parts of a plant]\label{def:g1:plants-around-us:parts}
A flowering \omterm{def:g1:plants-around-us:plant}{plant} has four main parts:
\begin{enumerate}
  \item the \emph{roots}\index{root}, hidden in the soil, which
        hold the \omterm{def:g1:plants-around-us:plant}{plant} in place and drink water;
  \item the \emph{stem}\index{stem}, which stands the \omterm{def:g1:plants-around-us:plant}{plant} up and
        carries water to the leaves;
  \item the \emph{leaves}\index{leaf}, flat and green, which catch
        the light;
  \item the \emph{flower}\index{flower}, which will make the seeds.
\end{enumerate}
\end{definition}

\begin{example}[Pulling up a weed]\label{ex:g1:plants-around-us:weed}
Pull a weed gently out of loose, wet earth and you bring up the
hidden part: a tuft of pale \omterm{def:g1:plants-around-us:parts}{roots}, sometimes longer than the whole
green \omterm{def:g1:plants-around-us:plant}{plant} above. Now the \omterm{def:g1:plants-around-us:plant}{plant} can be read from bottom to top:
\omterm{def:g1:plants-around-us:parts}{roots}, \omterm{def:g1:plants-around-us:parts}{stem}, \omterm{def:g1:plants-around-us:parts}{leaves} --- and, if you are lucky, a \omterm{def:g1:plants-around-us:parts}{flower}.
\end{example}

\begin{definition}[Tree]\label{def:g1:plants-around-us:tree}
A \emph{tree}\index{tree} is a big \omterm{def:g1:plants-around-us:plant}{plant} whose \omterm{def:g1:plants-around-us:parts}{stem} is made of
wood. The tree's woody \omterm{def:g1:plants-around-us:parts}{stem} is called the \emph{trunk}\index{trunk};
it divides into \emph{branches}\index{branch}, which carry the
\omterm{def:g1:plants-around-us:parts}{leaves}.
\end{definition}

\begin{example}[The chestnut tree]\label{ex:g1:plants-around-us:tree}
The chestnut \omterm{def:g1:plants-around-us:tree}{tree} in the yard has the same plan as the daisy, but
bigger and harder: \omterm{def:g1:plants-around-us:parts}{roots} under the ground, a trunk instead of a
soft \omterm{def:g1:plants-around-us:parts}{stem}, \omterm{def:g1:plants-around-us:tree}{branches} with thousands of \omterm{def:g1:plants-around-us:parts}{leaves}, and in spring,
\omterm{def:g1:plants-around-us:parts}{flowers}. The daisy lives one year; the chestnut can live longer
than a hundred.
\end{example}

\section{What a plant needs}

\begin{proposition}[The needs of a plant]\label{prop:g1:plants-around-us:needs}
To live and grow, a \omterm{def:g1:plants-around-us:plant}{plant} needs \emph{water}, \emph{light}, and a
place for its \omterm{def:g1:plants-around-us:parts}{roots}, usually \emph{soil}\index{soil}. A \omterm{def:g1:plants-around-us:plant}{plant} kept
in the dark, or never watered, wilts and dies.
\end{proposition}
\admitted

\begin{example}[The forgotten plant]\label{ex:g1:plants-around-us:wilt}
Forget to water the classroom \omterm{def:g1:plants-around-us:plant}{plant} before the holidays: its \omterm{def:g1:plants-around-us:parts}{leaves}
hang down, soft and sad --- it wilts. Water it, and a few hours
later the \omterm{def:g1:plants-around-us:parts}{leaves} stand up again. Water was the missing need.
\end{example}

\begin{method}[Looking after a plant]\label{met:g1:plants-around-us:care}
\begin{enumerate}
  \item Put the pot where daylight reaches it;
  \item water the soil when it feels dry --- not every hour, the
        \omterm{def:g1:plants-around-us:parts}{roots} also need air;
  \item do not pull off the \omterm{def:g1:plants-around-us:parts}{leaves}: they are the \omterm{def:g1:plants-around-us:plant}{plant}'s light
        catchers;
  \item be patient: a \omterm{def:g1:plants-around-us:plant}{plant} answers in days, not in minutes.
\end{enumerate}
\end{method}

\begin{remark}[Plants feed everyone]\label{rem:g1:plants-around-us:food}
\omterm{def:g1:plants-around-us:plant}{Plants} matter to every \omterm{def:g1:animals-around-us:animal}{animal}. The snail eats the salad, the
blackbird eats berries, we eat bread made from wheat --- a \omterm{def:g1:plants-around-us:plant}{plant}.
Even the cat, who eats no salad, eats \omterm{def:g1:animals-around-us:animal}{animals} that ate \omterm{def:g1:plants-around-us:plant}{plants}.
Without \omterm{def:g1:plants-around-us:plant}{plants}, no one else would have anything to eat; a later
chapter follows these food chains.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:1}
Name the four parts of a flowering \omterm{def:g1:plants-around-us:plant}{plant}, from under the soil to
the top.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:2}
Which part of the \omterm{def:g1:plants-around-us:plant}{plant}: (a) drinks water from the soil? (b)
catches the light? (c) will make the seeds? (d) holds the \omterm{def:g1:plants-around-us:plant}{plant}
upright?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:3}
The daisy never walks. Go through the signs of life and say why it
is a \omterm{def:g1:living-or-not:living}{living thing} anyway.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:4}
What are the special names of the \omterm{def:g1:plants-around-us:tree}{tree}'s \omterm{def:g1:plants-around-us:parts}{stem} and of the parts that
carry its \omterm{def:g1:plants-around-us:parts}{leaves}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:5}
Give the three things a \omterm{def:g1:plants-around-us:plant}{plant} needs to live, from
\cref{prop:g1:plants-around-us:needs}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:6}
The classroom \omterm{def:g1:plants-around-us:plant}{plant} has soft, hanging \omterm{def:g1:plants-around-us:parts}{leaves}. What has probably
been forgotten? What will happen after it is given?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:7}
In what ways are the daisy and the chestnut \omterm{def:g1:plants-around-us:tree}{tree} the same? In what
two ways are they different?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:8}
Try it: put one bean \omterm{def:g1:plants-around-us:plant}{plant} (or cress pot) near the window and one
in a dark cupboard, and water both. Look after one week. What
difference do you expect, and which need explains it?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-around-us:9}
Leo says: ``\omterm{def:g1:plants-around-us:plant}{Plants} are not really alive --- they never do
anything.'' Give him two things a \omterm{def:g1:plants-around-us:plant}{plant} does, slowly but surely,
from \cref{rem:g1:plants-around-us:still} and
\cref{ex:g1:plants-around-us:daisy}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-around-us:10}
A carrot is orange and grows under the ground. Looking at the
figure of the \omterm{def:g1:plants-around-us:plant}{plant}'s parts, which part of the carrot \omterm{def:g1:plants-around-us:plant}{plant} do you
think we eat? What clue does ``under the ground'' give?
\end{exercise}
