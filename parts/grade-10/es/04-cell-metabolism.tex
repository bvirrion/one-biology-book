\chapter{El metabolismo celular}\label{ch:g10:cell-metabolism}

Estira un globo sobre el cuello de un matraz con agua azucarada tibia y
una cucharada de levadura de panadería. En una hora el líquido se
enturbia y burbujea, el globo se levanta y el matraz huele levemente a
pan y a cerveza. Las \omterm{def:g10:cells-common-unit:cell}{células} de levadura se están comiendo el azúcar y
lo convierten en otra cosa, y el gas que desprenden es el mismo dióxido
de carbono que tú espiras. Una hoja al sol hace lo contrario: toma
dióxido de carbono y desprende oxígeno. Ambos procesos son química,
ejecutada dentro de \omterm{def:g10:cells-common-unit:cell}{células}, a la temperatura del cuerpo, sin llama y
sin ácido --- y este capítulo trata de esa química.

\section{El metabolismo: la química de una célula}

\begin{definition}[Metabolismo]\label{def:g10:cell-metabolism:metabolism}
El \emph{metabolismo}\index{metabolismo} de una \omterm{def:g10:cells-common-unit:cell}{célula} es el conjunto
de todas las reacciones químicas que tienen lugar dentro de ella: las
que degradan moléculas para liberar energía y las que construyen las
moléculas propias de la \omterm{def:g10:cells-common-unit:cell}{célula} a partir de otras más sencillas. Cada
reacción del metabolismo la lleva una \emph{enzima}\index{enzima}, una
\omterm{def:g10:chemistry-of-life:families}{proteína} que acelera una reacción concreta sin consumirse en ella.
\end{definition}

\begin{proposition}[Las enzimas hacen posible la química celular]\label{prop:g10:cell-metabolism:enzymes}
La glucosa dejada al aire a \qty{37}{\celsius} no arde; en una \omterm{def:g10:cells-common-unit:cell}{célula}
se oxida por completo en unos minutos. La diferencia son las \omterm{def:g10:cell-metabolism:metabolism}{enzimas}:
cada una se une a una molécula concreta (su \emph{sustrato}), la
transforma en un producto, lo libera y vuelve a empezar, miles de veces
por segundo. El \omterm{def:g10:cell-metabolism:metabolism}{metabolismo} de una \omterm{def:g10:cells-common-unit:cell}{célula} queda fijado, por tanto, por
el conjunto de \omterm{def:g10:cell-metabolism:metabolism}{enzimas} que posee --- y ese conjunto lo fijan sus \omterm{def:g10:universal-dna:gene}{genes}.
\end{proposition}
\admitted

\begin{example}[Una enzima en acción]\label{ex:g10:cell-metabolism:catalase}
Vierte agua oxigenada sobre un trozo de hígado crudo o de patata:
espuma con violencia mientras se desprende oxígeno. La \omterm{def:g10:cell-metabolism:metabolism}{enzima} catalasa,
presente en casi todas las \omterm{def:g10:cells-common-unit:cell}{células}, rompe el peróxido tóxico en agua y
oxígeno a un ritmo de millones de moléculas por segundo y por molécula
de \omterm{def:g10:cell-metabolism:metabolism}{enzima}. Hierve antes el hígado y no pasa nada: la \omterm{def:g10:cell-metabolism:metabolism}{enzima}, que es una
\omterm{def:g10:chemistry-of-life:families}{proteína}, ha quedado destruida por el calor. El agua oxigenada sobre
una piedra, sin una sola burbuja, muestra que la reacción no ocurre
sola.
\end{example}

\section{Dos maneras de alimentar una célula}

\begin{definition}[Autotrofia y heterotrofia]\label{def:g10:cell-metabolism:autotrophy}
Una \omterm{def:g10:cells-common-unit:cell}{célula} es \emph{autótrofa}\index{autotrofia} si construye sus
propias moléculas orgánicas solo a partir de \omterm{def:g10:chemistry-of-life:mineral-organic}{materia mineral} ---
dióxido de carbono, agua, \omterm{def:g10:chemistry-of-life:mineral-organic}{sales minerales} --- con una fuente externa de
energía. Las \omterm{def:g10:cells-common-unit:cell}{células} vegetales con \omterm{def:g10:cells-common-unit:organelle}{cloroplastos}, las algas y algunas
bacterias son autótrofas, y su fuente de energía es la luz: eso es la
\emph{fotosíntesis}\index{fotosíntesis}. Una \omterm{def:g10:cells-common-unit:cell}{célula} es
\emph{heterótrofa}\index{heterotrofia} si necesita tomar moléculas
orgánicas fabricadas por otros organismos, tanto de material de
construcción como de fuente de energía: las \omterm{def:g10:cells-common-unit:cell}{células} animales, los
hongos, la mayoría de las bacterias y las \omterm{def:g10:cells-common-unit:cell}{células} no verdes de las
plantas (las de la raíz, por ejemplo) son heterótrofas.
\end{definition}

\begin{proposition}[La fotosíntesis]\label{prop:g10:cell-metabolism:photosynthesis}
A la luz, una \omterm{def:g10:cells-common-unit:cell}{célula} con \omterm{def:g10:cells-common-unit:organelle}{cloroplastos} absorbe dióxido de carbono y
agua, desprende oxígeno y construye glucosa, que después convierte en
almidón, celulosa, \omterm{def:g10:chemistry-of-life:families}{lípidos} y, con nitrógeno mineral, \omterm{def:g10:chemistry-of-life:families}{aminoácidos}. En
resumen:
\[
6\,\mathrm{CO_2} + 6\,\mathrm{H_2O} \xrightarrow{\ \text{luz, clorofila}\ }
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2}.
\]
La energía de la luz queda almacenada en los enlaces químicos de la
glucosa.
\end{proposition}
\begin{proof}[Pruebas experimentales]
Una suspensión del alga verde \emph{Chlorella} en un recipiente cerrado
con una sonda de oxígeno muestra que el oxígeno disuelto sube de forma
sostenida con la lámpara encendida y baja cuando se apaga; la subida se
detiene si se retira el dióxido de carbono del agua. Las hojas dejadas
a oscuras dos días ya no se tiñen de azul muy oscuro con yodo, y una
hoja tapada a medias con papel de aluminio solo se tiñe en la mitad
expuesta: el almidón se fabrica donde llega la luz. El dióxido de
carbono radiactivo suministrado a una planta acaba en sus azúcares en
unos minutos.
\end{proof}

\begin{example}[La euglena, las dos cosas a la vez]\label{ex:g10:cell-metabolism:euglena}
La unicelular \emph{Euglena} tiene \omterm{def:g10:cells-common-unit:organelle}{cloroplastos} y nada con un flagelo.
A la luz, en agua mineral, crece: \omterm{def:g10:cell-metabolism:autotrophy}{autótrofa}. A oscuras solo sobrevive
si se añaden al agua moléculas orgánicas, que absorbe: \omterm{def:g10:cell-metabolism:autotrophy}{heterótrofa}.
Mantenida mucho tiempo a oscuras pierde los \omterm{def:g10:cells-common-unit:organelle}{cloroplastos}; devuelta a la
luz con unos días de paciencia vuelve a fabricarlos. Su \omterm{def:g10:cell-metabolism:metabolism}{metabolismo}
depende de sus \omterm{def:g10:universal-dna:gene}{genes}, que permiten ambas opciones, y de su entorno, que
selecciona una.
\end{example}

\section{Obtener energía de las moléculas orgánicas}

\begin{proposition}[La respiración celular]\label{prop:g10:cell-metabolism:respiration}
Toda \omterm{def:g10:cells-common-unit:cell}{célula}, \omterm{def:g10:cell-metabolism:autotrophy}{autótrofa} o \omterm{def:g10:cell-metabolism:autotrophy}{heterótrofa}, obtiene energía utilizable
degradando moléculas orgánicas. Cuando hay oxígeno disponible la
degradación es completa --- es la \emph{respiración
celular}\index{respiración celular}, en las \omterm{def:g10:cells-common-unit:organelle}{mitocondrias}:
\[
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \longrightarrow
6\,\mathrm{CO_2} + 6\,\mathrm{H_2O} + \text{energía},
\]
unos \qty{2870}{kJ} por mol de glucosa (\qty{180}{g}), de los que
aproximadamente el 40\% se recoge en una molécula pequeña, el
\emph{ATP}\index{ATP}, del que echa mano todo proceso de la \omterm{def:g10:cells-common-unit:cell}{célula} que
requiera energía --- movimiento, transporte, síntesis ---; el resto se
libera como calor.
\end{proposition}
\begin{proof}[Pruebas experimentales]
Una levadura en un recipiente cerrado con una sonda de oxígeno no
consume oxígeno mientras está en ayunas; se le añade glucosa y el
oxígeno disuelto baja a velocidad constante mientras aparece dióxido de
carbono, hasta agotar el oxígeno. Unas semillas en germinación dentro
de un termo calientan el termo varios grados en un día --- el calor de
la respiración ---, mientras que las semillas hervidas no. Las \omterm{def:g10:cells-common-unit:cell}{células}
musculares vistas al microscopio electrónico están abarrotadas de
\omterm{def:g10:cells-common-unit:organelle}{mitocondrias}, y un veneno de las \omterm{def:g10:cells-common-unit:organelle}{mitocondrias} detiene el músculo.
\end{proof}

\begin{definition}[Fermentación]\label{def:g10:cell-metabolism:fermentation}
Sin oxígeno, algunas \omterm{def:g10:cells-common-unit:cell}{células} degradan la glucosa solo parcialmente, en
el \omterm{def:g10:cells-common-unit:cell}{citoplasma}, hasta moléculas orgánicas menores que conservan casi
toda la energía: eso es la \emph{fermentación}\index{fermentación}. La
levadura realiza la fermentación alcohólica,
\[
\mathrm{C_6H_{12}O_6} \longrightarrow 2\,\mathrm{C_2H_5OH} +
2\,\mathrm{CO_2} + \text{energía},
\]
y las \omterm{def:g10:cells-common-unit:cell}{células} musculares faltas de oxígeno, así como las bacterias del
yogur, realizan la fermentación láctica, que convierte la glucosa en
ácido láctico. Una fermentación rinde unas quince veces menos ATP por
glucosa que la respiración.
\end{definition}

\begin{example}[La levadura con aire y sin él]\label{ex:g10:cell-metabolism:pasteur}
Una levadura aireada con burbujas consume glucosa con moderación y
crece deprisa: respira. La misma levadura en una cuba cerrada consume
glucosa con avidez, crece poco y produce etanol: fermenta. Pasteur, que
lo describió en 1861, llamó a la \omterm{def:g10:cell-metabolism:fermentation}{fermentación} ``la vida sin aire''; las
\omterm{def:g10:cells-common-unit:cell}{células} cambian su \omterm{def:g10:cell-metabolism:metabolism}{metabolismo} a la vía que su entorno permite y, como
la \omterm{def:g10:cell-metabolism:fermentation}{fermentación} da tan poca energía por glucosa, tienen que quemar
mucho más azúcar para vivir.
\end{example}

\section{Qué decide el metabolismo de una célula}

\begin{proposition}[Genes y ambiente]\label{prop:g10:cell-metabolism:decides}
El \omterm{def:g10:cell-metabolism:metabolism}{metabolismo} de una \omterm{def:g10:cells-common-unit:cell}{célula} depende de dos cosas: de su
\emph{equipamiento} --- las \omterm{def:g10:cell-metabolism:metabolism}{enzimas} y los \omterm{def:g10:cells-common-unit:organelle}{orgánulos} que sus \omterm{def:g10:universal-dna:gene}{genes} le
permiten construir, de modo que una \omterm{def:g10:cells-common-unit:cell}{célula} sin clorofila nunca podrá
hacer la \omterm{def:g10:cell-metabolism:autotrophy}{fotosíntesis} --- y de su \emph{ambiente} --- luz, oxígeno,
nutrientes disponibles ---, que decide cuál de las vías posibles
funciona de hecho. Una levadura respira con aire y fermenta sin él; una
\emph{Euglena} es \omterm{def:g10:cell-metabolism:autotrophy}{autótrofa} a la luz y \omterm{def:g10:cell-metabolism:autotrophy}{heterótrofa} a oscuras; una
\omterm{def:g10:cells-common-unit:cell}{célula} de la raíz de una planta verde es \omterm{def:g10:cell-metabolism:autotrophy}{heterótrofa} toda su vida,
alimentada por las hojas.
\end{proposition}
\admitted

\begin{method}[Leer un registro metabólico]\label{met:g10:cell-metabolism:trace}
Los experimentos con sonda de oxígeno dan una concentración frente al
tiempo.
\begin{enumerate}
  \item Identifica qué cambió en cada instante marcado (se enciende la
        luz, se añade sustrato, se agota el oxígeno).
  \item Lee la pendiente de cada tramo recto: subida partido por
        avance, en \unit{mg/L} por minuto. Una pendiente positiva
        significa producción neta de oxígeno, y una negativa consumo
        neto.
  \item Recuerda que una \omterm{def:g10:cells-common-unit:cell}{célula} que hace la \omterm{def:g10:cell-metabolism:autotrophy}{fotosíntesis} también
        respira: la pendiente a la luz es la producción menos la
        respiración, así que la velocidad verdadera de la \omterm{def:g10:cell-metabolism:autotrophy}{fotosíntesis}
        es la pendiente a la luz más el valor absoluto de la pendiente
        a oscuras.
  \item Convierte si te lo piden: a estas concentraciones,
        \qty{1}{mg/L} de oxígeno disuelto son $1/32$ milimoles por
        litro.
\end{enumerate}
\end{method}

\begin{example}[La velocidad verdadera del alga]\label{ex:g10:cell-metabolism:truerate}
En la figura de \emph{Chlorella} el oxígeno sube \qty{2.5}{mg/L} en 5
minutos de luz (pendiente $+0.5$) y baja \qty{0.5}{mg/L} en 5 minutos
de oscuridad (pendiente $-0.1$). La respiración consume también
\qty{0.1}{mg/L} por minuto durante la luz, así que la \omterm{def:g10:cell-metabolism:autotrophy}{fotosíntesis}
produce $0.5 + 0.1 = \qty{0.6}{mg/L}$ por minuto: seis veces lo que
gasta la respiración. El alga es una productora neta de oxígeno y de
\omterm{def:g10:chemistry-of-life:mineral-organic}{materia orgánica}, y por eso crece.
\end{example}

\section{Problema: La cuba del bodeguero}

\begin{problem}[{Problema de fin de semana --- una cuba de mosto de uva
dejada a la levadura: el alcohol que dará, el gas que desprenderá, el
calor que producirá y por qué una bodega puede ser un lugar peligroso}]
\label{pb:g10:cell-metabolism:1}
El mosto de uva contiene unos \qty{200}{g} de azúcar por litro, que
tomaremos como glucosa, y nada de oxígeno disuelto una vez cerrada la
cuba. Masas molares: glucosa \qty{180}{g/mol}, etanol
$\mathrm{C_2H_5OH}$ \qty{46}{g/mol}, dióxido de carbono
\qty{44}{g/mol}, oxígeno \qty{32}{g/mol}. La densidad del etanol es
\qty{0.79}{g/mL}. Un mol de gas ocupa unos \qty{24}{L} a la temperatura
de una bodega. La cuba contiene \qty{1000}{L}.

\medskip
\noindent\textbf{Parte I --- Lo que fabrica la levadura.}
\begin{enumerate}
  \item ¿Qué vía usa la levadura en la cuba cerrada, y por qué?
        Escribe su ecuación.
  \item ¿Cuántos moles de glucosa contiene un litro de mosto?
  \item Calcula la masa de etanol producida por litro, después su
        volumen y después el grado alcohólico del vino en tanto por
        ciento de volumen.
  \item Calcula la masa de dióxido de carbono producida por litro de
        mosto, y su volumen en forma de gas.
  \item Para la cuba entera, ¿qué volumen de dióxido de carbono se
        desprende? Compáralo con el volumen de una bodega de \qty{4}{m}
        por \qty{6}{m} por \qty{2.5}{m}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Si la levadura tuviera aire.}
\begin{enumerate}[resume]
  \item Escribe la ecuación que seguiría la levadura si el mosto se
        aireara de forma continua.
  \item Calcula la masa de oxígeno necesaria para respirar el azúcar de
        un litro de mosto, y la masa de dióxido de carbono desprendida.
  \item Compara el dióxido de carbono de la pregunta 7 con el de la
        pregunta 4 y explica la diferencia en función de hasta dónde se
        degrada la glucosa.
  \item La respiración rinde unos 30 ATP por glucosa y la \omterm{def:g10:cell-metabolism:fermentation}{fermentación}
        2. Si la levadura necesita el mismo ATP en ambos casos, ¿en
        cuál dura más el azúcar y por qué factor?
  \item Explica por qué, aun así, un bodeguero deja el aire fuera.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El calor de la cuba.}
La \omterm{def:g10:cell-metabolism:fermentation}{fermentación} libera unos \qty{70}{kJ} de calor por mol de glucosa;
calentar \qty{1}{kg} de agua \qty{1}{\celsius} cuesta \qty{4.2}{kJ}.
\begin{enumerate}[resume]
  \item Calcula el calor liberado por litro de mosto, y después el de
        la cuba entera.
  \item Si no se escapara nada de ese calor, ¿cuánto subiría la
        temperatura de la cuba? (Toma el mosto como agua.)
  \item Las \omterm{def:g10:cell-metabolism:metabolism}{enzimas} de la levadura se destruyen por encima de unos
        \qty{40}{\celsius}. Partiendo de \qty{20}{\celsius}, ¿sobreviviría
        una cuba sin refrigerar? ¿Qué te dice esto sobre por qué se
        refrigeran las cubas grandes?
  \item Explica, con la \cref{prop:g10:cell-metabolism:enzymes}, por
        qué el calor detiene la \omterm{def:g10:cell-metabolism:fermentation}{fermentación} aunque siga habiendo
        azúcar.
  \item La respiración habría liberado unos \qty{2870}{kJ} por mol.
        ¿Dónde se queda la energía que la \omterm{def:g10:cell-metabolism:fermentation}{fermentación} \emph{no}
        libera?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La bodega.}
\begin{enumerate}[resume]
  \item El dióxido de carbono es más denso que el aire. ¿Dónde se
        acumula en la bodega, y por qué llevar una vela encendida a
        ras de suelo es una prueba de seguridad tradicional?
  \item Las \omterm{def:g10:cells-common-unit:cell}{células} de un trabajador, privadas de oxígeno, pasarían a
        cuál de las vías --- y por qué un cuerpo humano no puede vivir
        de ella más que unos minutos?
  \item La \omterm{def:g10:cell-metabolism:fermentation}{fermentación} se detiene sola cuando el alcohol llega a un
        15\% de volumen aproximadamente. Propón una explicación en
        términos de las \omterm{def:g10:cell-metabolism:metabolism}{enzimas} de la levadura.
  \item La masa del pan usa la misma levadura durante una hora. ¿Qué
        contienen las burbujas de la miga y dónde ha ido a parar el
        etanol después de la cocción?
  \item Enuncia el resultado: para un litro de mosto, el grado
        alcohólico y el volumen de gas producidos --- y el único hecho
        metabólico que explica ambos.
\end{enumerate}
\end{problem}
