\chapter{حياة النبات: ازدهار وهو متجذر}\label{ch:g12:plant-rooted-life}

لا تستطيع البلوطة أن تمشي إلى الماء، ولا أن تفر من يرقة، ولا أن تذهب
باحثة عن قرين. فهي تقف حيث سقطت بلوطتها ثلاثة قرون، وترفع في تلك المدة
مئات الأطنان من الماء من التربة، وتدفع عن نفسها ألوف \omterm{def:g10:biodiversity-scales:species}{الأنواع} التي كانت
ستأكلها، وتدير أوراقها لتتبع الضوء، وترسل غبار طلعها وبلوطها عبر الريف
بالريح وبالقيق. فالحياة المتجذرة ليست حياة سلبية؛ بل هي طقم مختلف من
الحلول للمسائل نفسها --- التغذي والدفاع والحركة والتكاثر --- ويستعرضها
هذا الفصل.

\section{جسم مبني للتبادل}

\begin{proposition}[جهازان وسطحان]\label{prop:g12:plant-rooted-life:surfaces}
النبات الزهري منظَّم في جهازين: \emph{الجهاز الجذري} في التربة، وهو
يمتص الماء والشوارد المعدنية ويثبّت النبات، و\emph{الجهاز الخضري} في
الهواء --- السوق والأوراق --- وهو يلتقط الضوء وثنائي أكسيد الكربون.
وكل منهما سطح تبادل هائل: فأوراق شجرة كبيرة تنشر مئات الأمتار المربعة
من السطح إلى الضوء، وجذورها، ممدودةً بمليارات \emph{الشعيرات
الجذرية}\index{شعيرة جذرية} المجهرية، تنشر مئات الأمتار المربعة إلى
التربة. فالنبات، قبل كل شيء، طريقة في نشر السطح في اتجاهين من نقطة
ثابتة.
\end{proposition}
\begin{proof}[شواهد]
نبتة جاودار واحدة نميت أربعة أشهر في صندوق تراب وجد لها
$\num{1.4e10}$ \omterm{prop:g12:plant-rooted-life:surfaces}{شعيرة جذرية} طولها الكلي يزيد على
\qty{10000}{km} وسطحها نحو \qty{400}{m^2}، أي عشرة أمثال سطح أوراقها.
وشجرة بلوط ناضجة تحمل نحو 250\,000 ورقة؛ وهكتار من الغابة يقدم
\qtyrange{5}{8}{ha} من سطح الأوراق إلى السماء. وقص \omterm{prop:g12:plant-rooted-life:surfaces}{الشعيرات الجذرية}
عن بادرة، أو طلاء الأوراق، يوقف نموها في غضون أيام.
\end{proof}

\begin{definition}[الثغور]\label{def:g12:plant-rooted-life:stomata}
\emph{الثغر}\index{ثغر} مسام في جلد الورقة، تحده \emph{خليتان
حارستان} تفتحانه بالانتفاخ وتغلقانه بالانكماش. ومن الثغور المفتوحة
يدخل ثنائي أكسيد الكربون \omterm{def:g10:cell-metabolism:autotrophy}{للتركيب الضوئي} ويخرج بخار الماء --- وهو
\emph{النتح} الذي لا يستطيع النبات تجنبه ما دام يتغذى. وتحمل الورقة
من مئة إلى ثلاثمئة ثغر في الميليمتر المربع، ومعظمها على سطحها السفلي؛
وهي تنفتح في الضوء وتنغلق في الظلام وفي الجفاف.
\end{definition}

\section{نقل الماء والغذاء بلا قلب}

\begin{proposition}[نسغان]\label{prop:g12:plant-rooted-life:saps}
للنبات جهازا نقل، كلاهما مؤلف من \omterm{def:g10:cells-common-unit:cell}{خلايا} طويلة ملتحمة أنابيب في عروق كل
عضو.
\begin{itemize}
  \item \emph{الخشب}\index{خشب} يحمل \emph{النسغ الناقص} --- الماء
        والشوارد المعدنية التي تمتصها الجذور --- صعدًا إلى الأوراق.
        والقوة الدافعة هي النتح: فالماء المتبخر من \omterm{def:g10:cells-common-unit:cell}{خلايا} الورقة يشد
        عمود الماء المتصل في \omterm{prop:g12:plant-rooted-life:saps}{الخشب} صعدًا من الجذور، كما تشد الفتيلة
        الزيت. فلا مضخة، ولا طاقة ينفقها النبات: الشمس هي التي ترفع.
  \item \emph{اللحاء}\index{لحاء} يحمل \emph{النسغ الكامل} --- الماء
        مع السكريات التي صنعتها الأوراق --- من الأوراق إلى كل عضو
        يستهلكها أو يخزنها: القمم النامية، والجذور، والثمار، والبذور،
        والدرنات. وهو يجري من المنبع إلى المصب، في أي اتجاه كانت
        المصبات.
\end{itemize}
\end{proposition}
\begin{proof}[شواهد]
ساق مقطوعة توضع في ماء مصبوغ تبدي الصبغ صاعدًا في \omterm{prop:g12:plant-rooted-life:saps}{الخشب}، أسرع في فرع
ينتح منه في فرع محبوس في كيس؛ والفرع المنزوعة أوراقه لا يرفع شيئًا.
وحلقة من القشرة (وفيها \omterm{prop:g12:plant-rooted-life:saps}{اللحاء}) تنزع من جذع توقف وصول السكر إلى
الجذور، فتجوع في غضون أشهر، بينما تبقى الأوراق فوق الحلقة موردة
بالماء؛ ويتراكم السكر فوق الحلقة وتنتفخ القشرة. وحشرات المنّ التي
تتغذى من \omterm{prop:g12:plant-rooted-life:saps}{اللحاء} بأفواهها الإبرية تعطي، إذا فصلت عن الإبرة، قطرة من
نسغ سكري تحت ضغط.
\end{proof}

\begin{example}[أرقام شجرة]\label{ex:g12:plant-rooted-life:numbers}
شجرة بتولا ناضجة تنتح في يوم صيفي \qtyrange{300}{400}{L}
من الماء، مرفوعًا \qty{20}{m} عبر أنابيب خشب عرضها عشر
الميليمتر بسرعة \qtyrange{1}{5}{m} في الساعة. ويحتفظ بأقل من 1\% من
ذلك الماء \omterm{def:g10:cell-metabolism:autotrophy}{للتركيب الضوئي}؛ والباقي ثمن \omterm{def:g12:plant-rooted-life:stomata}{الثغور} المفتوحة. وفي المقابل

ناضجة.
\end{example}

\section{الثبات في الموضع}

\begin{proposition}[دفاعات كائن متجذر]\label{prop:g12:plant-rooted-life:defences}
النباتات، إذ لا تستطيع الفرار، تدافع عن نفسها في مكانها.
\begin{itemize}
  \item \emph{بنى}: طبقة شمعية وقشرة ضد الجفاف والعدوى؛ وأشواك وشويكات
        وشعيرات لاسعة وسيليكا ضد آكلات العشب؛ وجدار سميك من السيللوز
        في كل \omterm{def:g10:cells-common-unit:cell}{خلية}.
  \item \emph{كيمياء}: تانينات تجعل الأوراق عسيرة الهضم، وقلويدات مرة
        (الكافيين والنيكوتين والمورفين)، وسموم، وراتنجات ولبن يسدان
        الجروح ويسممان الحشرات؛ وكثير منها لا يصنع إلا بعد هجوم، وبعضه
        يطلق إشارات طيارة تنذر الأوراق المجاورة وتجذب مفترسات الحشرة
        المهاجمة.
  \item \emph{الاحتمال والتجدد}: نقاط نمو عند كل عقدة، فيعود الفرع
        المرعي نموًا؛ وسكون عبر الشتاء أو الجفاف؛ وبذور تنتظر في
        التربة سنين.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[وجبة يرقة، ويجاب عنها]\label{ex:g12:plant-rooted-life:caterpillar}
في غضون ساعات من قضمات اليرقة الأولى، تصنع نبتة البندورة، في أوراقها
كلها، بروتينات تسد \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيمات} الحشرة الهاضمة؛ وفي غضون يوم، تكون مادة
كيميائية تطلقها في الهواء قد جذبت زنبورًا طفيليًا إلى اليرقة. ونبتات
البندورة المجاورة، إذ تتلقى الإشارة الهوائية نفسها، تشرع في صنع
سادّات الهضم قبل أن تبلغها أي يرقة. فالنبات لا يتحرك، لكنه ليس أعزل،
وليس وحده.
\end{example}

\section{نمو نحو الشيء، ونمو بعيدًا عنه}

\begin{proposition}[النمو حركةً]\label{prop:g12:plant-rooted-life:tropisms}
النبات يتحرك بالنمو. فأفرعه تنمو نحو الضوء وضد الجاذبية، وجذوره نحو
الجاذبية والماء: وهي \emph{الانتحاءات}\index{انتحاء}، أي استجابات
نمو موجَّهة. وتضبطها \omterm{def:g11:hormones-and-reproduction:hormone}{الهرمونات}، وأولها المكتشف \emph{الأوكسين}، يصنع في
القمة النامية للفرع وينقل نزولًا في الساق؛ \omterm{def:g10:cells-common-unit:cell}{والخلايا} التي تتلقى أوكسينًا
أكثر تستطيل أكثر. والضوء الساقط على جانب واحد من فرع ينقل الأوكسين
إلى الجانب المظلل، فينمو أسرع فيثني الفرع نحو الضوء.
\end{proposition}
\begin{proof}[شواهد]
بادرات الأعشاب المضاءة من جانب واحد تنثني نحو الضوء؛ فإن غطيت عند
القمة لم تنثن، مع أن منطقة الانثناء أسفل منها؛ وإن غطيت عند القاعدة
وحدها انثنت انثناءها المعتاد: فالقمة تدرك، والقاعدة تستجيب، وشيء ينتقل
بينهما. وقمة تقص وتوضع على قالب هلام، ثم يوضع الهلام على جانب واحد من
بادرة مقطوعة الرأس، تجعلها تنثني بعيدًا عن القالب في الظلام --- فالمادة
التي في الهلام، وهي الأوكسين، تكفي. وبالقياس المباشر، يحمل الجانب
المظلل من بادرة مضاءة نحو ضعف الأوكسين الذي يحمله الجانب المضاء.
\end{proof}

\section{التكاثر بلا حركة}

\begin{proposition}[الأزهار وزوارها]\label{prop:g12:plant-rooted-life:flowers}
\emph{الزهرة} عضو التكاثر عند معظم النباتات البرية: \emph{أسدية} تصنع
غبار الطلع، وهو مشيج النبات الذكري، و\emph{مدقة} تحمل بييضاتها المشيج
الأنثوي. ولما كان الوالدان لا يتحركان، وجب أن يحمل غبار الطلع من زهرة إلى
أخرى --- بالريح، كما في الأعشاب وكثير من الأشجار، وأزهارها صغيرة باهتة
تنثر غبار الطلع بالملايين، أو بالحيوان، كما في معظم النباتات الزهرية،
وبتلاتها وروائحها ورحيقها هي الأجر المدفوع للحشرات والطيور والخفافيش
التي تحمل غبار الطلع وهي تتغذى. وكثيرًا ما تطورت الزهرة وملقّحها معًا، كل
منهما متكيف مع شكل الآخر وعاداته.
\end{proposition}
\admitted

\begin{example}[أشكال يوافق بعضها بعضًا]\label{ex:g12:plant-rooted-life:match}
زهرة لها أنبوب رحيق طوله \qty{30}{cm} تزورها عثة لسانها
\qty{30}{cm}، ولا يزورها سواها؛ والزهرة الحمراء عديمة الرائحة
الأنبوبية المفتوحة نهارًا زهرة طير؛ والزهرة الفاتحة التي تنفتح ليلًا
برائحة ثقيلة زهرة عثة؛ وطاس من بتلات صفر بمدرج هبوط زهرة نحلة. فالنبات
يدفع سكرًا لخدمة لا يستطيع أداءها، وتخصص الملقّح يضمن أن يذهب غبار الطلع
إلى زهرة أخرى من \omterm{def:g10:biodiversity-scales:species}{النوع} نفسه.
\end{example}

\begin{proposition}[الثمار والانتشار]\label{prop:g12:plant-rooted-life:dispersal}
بعد الإخصاب تصير البييضة \emph{بذرة} --- جنينًا مع مخزن غذاء، ساكنًا،
في غلاف واق --- وتصير المدقة \emph{ثمرة} تحمل البذور بعيدًا عن الوالد:
على الريح بأجنحة وزغب؛ وبالماء؛ وعلى فراء الحيوان بخطاطيف؛ وداخل
الحيوان، حين تؤكل ثمرة لحمية وتسقط بذورها في مكان آخر، سليمةً، مع جرعة
سماد. فالبذرة طور النبات المتحرك الوحيد ووسيلته الوحيدة إلى الانتظار:
تستطيع أن تسافر كيلومترات وأن تنام سنين.
\end{proposition}
\admitted

\begin{method}[قراءة تكيّف مع الحياة المتجذرة]\label{met:g12:plant-rooted-life:read}
عند أي سمة من سمات نبات، اسأل:
\begin{enumerate}
  \item أي حاجة من حاجات كائن ثابت تخدم --- التقاط الضوء أو ثنائي
        أكسيد الكربون، أم امتصاص الماء، أم النقل، أم الدفاع، أم
        التوجه، أم نشر غبار الطلع أو البذور؟
  \item أي سطح أو بنية أو مادة تستعمل، وبأي كلفة (ماء يفقد عبر
        \omterm{def:g12:plant-rooted-life:stomata}{الثغور}، وطاقة تنفق على الرحيق أو السموم)؟
  \item أتشرك كائنًا آخر (ملقّحًا، أو ناشرًا، أو فطر جذور، أو آكل عشب)،
        وماذا يربح كل شريك؟
  \item أتبنى مرة واحدة (القشرة، الأشواك) أم تنتج عند الطلب (سم بعد
        هجوم، وانثناء نحو الضوء)؟
\end{enumerate}
\end{method}

\begin{remark}[نصف الحياة، واقفًا في مكانه]\label{rem:g12:plant-rooted-life:half}
النباتات تؤلف معظم الكتلة الحية على الأرض وتغذي كل ما سواها تقريبًا.
وفصلا \omterm{def:g10:cell-metabolism:autotrophy}{التركيب الضوئي} والتنفس التاليان يصفان كيمياءها؛ وهذا الفصل قد
وصف الجسم الذي يشغّلها: نقطة ثابتة، تنتشر منها سطوح في التربة وفي
الهواء، وتجري فيها الأنساغ بلا مضخة، وتحل الكيمياء محل الطيران، ويحل
النمو محل الحركة، وتستأجر الحيوانات لتقوم عنها بالسفر.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:plant-rooted-life:1}
سمّ جهازي النبات الزهري وسطح التبادل عند كل منهما.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:plant-rooted-life:2}
ما \omterm{def:g12:plant-rooted-life:stomata}{الثغر}، وماذا يمر عبره في كل اتجاه، ومتى يكون مفتوحًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:plant-rooted-life:3}
ميّز النسغ الناقص من النسغ الكامل: المحتوى، والأنابيب، والاتجاه،
والقوة الدافعة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:plant-rooted-life:4}
أعط دفاعين بنيويين ودفاعين كيميائيين عند النبات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:plant-rooted-life:5}
ما الانتحاء؟ وأي \omterm{def:g11:hormones-and-reproduction:hormone}{هرمون} يضبط انثناء الفرع نحو الضوء، وأين يصنع؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:plant-rooted-life:6}
ورقة مساحتها \qty{40}{cm^2} تحمل 200 \omterm{def:g12:plant-rooted-life:stomata}{ثغر} في الميليمتر المربع على
سطحها السفلي. كم ثغرًا لها؟ وكم عند شجرة لها 200\,000 ورقة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:plant-rooted-life:7}
من شكل اليوم الصيفي، في أي ساعة يكون النتح أعظم ما يكون، ولماذا ينخفض
في منتصف العصر مع أن الضوء لا ينخفض؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:plant-rooted-life:8}
تنزع حلقة من القشرة حول جذع في الربيع. تنبأ بما يحل بالأوراق
وبالجذور وبالقشرة فوق الحلقة مباشرة في الأشهر التالية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:plant-rooted-life:9}
في شكل البادرات، فسّر ماذا تثبت كل تجربة من التجارب الأربع وماذا تنفي،
ولماذا لزمت الأربع كلها.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:plant-rooted-life:10}
نبات يرفع \qty{300}{L} من الماء في اليوم إلى أوراقه من غير أن ينفق
طاقة في ذلك. فسّر من أين تأتي الطاقة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:plant-rooted-life:11}
قارن زهرة تلقحها الريح بزهرة تلقحها الحشرات في أربع نقاط، وفسّر كل فرق
بالحامل.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:plant-rooted-life:12}
نبات يغلق ثغوره أثناء جفاف. فسّر ماذا يربح وماذا يخسر، ولماذا يقتله
جفاف طويل مع أنه كف عن فقد الماء.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:plant-rooted-life:13}
الكافيين سام للحشرات بالجرعات التي في ورقة البن، لكنه موجود بعشر تلك
الجرعة في الرحيق، حيث يحسّن ذاكرة النحل للزهرة. اقرأ ذلك بالطريقة في
\cref{met:g12:plant-rooted-life:read}: ماذا يفعل الجزيء نفسه في
الموضعين؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:plant-rooted-life:14}
عثة لسانها \qty{30}{cm} هي الملقّح الوحيد لزهرة أنبوب رحيقها
\qty{30}{cm}. فسّر كيف يستطيع توافق كهذا أن يتطور خطوة خطوة، وماذا
يخاطر به كل شريك إذا اختفى الآخر.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:plant-rooted-life:15}
"النبات كائن سلبي." أعد كتابة الجملة على الصواب في فقرة، مع مثال واحد
لكل من الإحساس والاستجابة والدفاع والاستقدام.
\end{exercise}

\section{مسألة: يوم صيفي لبلوطة}

\begin{problem}[{مسألة نهاية الأسبوع --- تبادلات شجرة تحسب: الأوراق
والثغور تعد، والماء يرفع والسكر يصنع، وتوصيلات اللحاء، والبذور ترسل}]
\label{pb:g12:plant-rooted-life:1}
شجرة بلوط ناضجة تحمل 250\,000 ورقة مساحة كل منها \qty{25}{cm^2}.
وسطوح أوراقها السفلى تحمل 250 ثغرًا في الميليمتر المربع. وفي يوم صيفي
ينتح كل متر مربع من الورق \qty{0.6}{L} من الماء ويثبّت
\qty{8}{g} من ثنائي أكسيد الكربون في السكر. والأوراق 90\% ماء؛
ويستعمل \omterm{def:g10:cell-metabolism:autotrophy}{التركيب الضوئي} \qty{1}{g} من الماء لكل \qty{1.6}{g} من السكر
المصنوع.

\medskip
\noindent\textbf{القسم الأول --- السطوح.}
\begin{enumerate}
  \item احسب مساحة أوراق البلوطة الكلية بالأمتار المربعة.
  \item احسب عدد \omterm{def:g12:plant-rooted-life:stomata}{الثغور} على الشجرة.
  \item يغطي تاج الشجرة \qty{150}{m^2} من الأرض. فكم مترًا مربعًا من
        الورق يتراكم فوق كل متر مربع من الأرض؟
  \item شعيرات جذور شجرة كهذه تقدم نحو \qty{1000}{m^2} إلى التربة.
        قارن ذلك بالأوراق وفسّر لماذا كان السطحان من الرتبة نفسها.
  \item لماذا وجب أن ينشر سطح الورق رقيقًا متراكمًا، لا أن يركّز في
        عضو واحد سميك؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثاني --- الماء.}
\begin{enumerate}[resume]
  \item احسب حجم الماء الذي تنتحه البلوطة في اليوم.
  \item احسب كتلة السكر المصنوع، وكتلة الماء المستعملة في صنعه. وأي
        كسر من الماء المنتوح ذلك؟
  \item فسّر لماذا لا تستطيع الشجرة أن تحتفظ بالباقي: ماذا كان سيحدث
        لو أغلقت ثغورها لتوفره؟
  \item يصعد الماء \qty{20}{m} عبر \omterm{prop:g12:plant-rooted-life:saps}{الخشب}. فما الذي يمد بالطاقة، وكم
        كانت مضخة خاصة بالشجرة ستكلفها؟
  \item في عصر حار جاف يهبط النتح إلى الثلث بينما يكون الضوء في أعظمه.
        فماذا فعلت الشجرة، وبكم يكلفها ذلك من السكر؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثالث --- السكر.}
جذور البلوطة وجذعها وأفرعها النامية تستهلك \qty{2}{kg} من السكر في
اليوم في التنفس والنمو؛ ويأخذ البلوط \qty{0.5}{kg} أخرى في موسمه.
\begin{enumerate}[resume]
  \item احسب السكر المصنوع في اليوم والكسر الموصل إلى الجذور والجذع
        والأفرع، والكسر الموصل إلى البلوط.
  \item أي نسيج يحمل السكر، ومن أين إلى أين؟
  \item ماذا يحل بالسكر الفائض، وأين يوجد في الربيع التالي؟
  \item تقطع حلقة من القشرة في حزيران. أي أرقام السؤال 11 تتغير، وكيف؟
  \item فسّر لماذا تستطيع شجرة فقدت نصف أوراقها لليرقات في أيار أن
        تنتج بلوطًا مع ذلك، إذا كان الموسم جيدًا.
\end{enumerate}

\medskip
\noindent\textbf{القسم الرابع --- إرسال الجيل التالي.}
البلوطة تلقحها الريح وتنتج، في سنة جيدة، 10\,000 بلوطة؛ ويحمل القيق
ويدفن نحو 3000 منها، يأكل معظمها لاحقًا؛ وتأكل السناجب والفئران كل ما
تبقى تقريبًا.
\begin{enumerate}[resume]
  \item اذكر ثلاث سمات تتوقعها في أزهار البلوط، من تلقيحها بالريح.
  \item لماذا "تدفع" الشجرة للقيق بلوطًا ستخسره، بدل أن تسقط البلوط
        تحتها؟
  \item إذا نسي القيق 5\% مما يدفن وصارت بلوطة مدفونة واحدة من كل 50
        شتلة، فكم شتلة يعطي محصول السنة؟
  \item تعيش البلوطة 300 سنة وتحتاج أن تعوَّض بشجرة بالغة واحدة. قارن
        ذلك بجوابك السابق وفسّر لماذا كانت الأرقام معقولة.
  \item اذكر النتيجة: مساحة أوراق البلوطة وعدد ثغورها، واللترات التي
        ترفعها في يوم مقابل الكيلوغرامات التي تصنعها من السكر،
        والعاملان اللذان يقومان عنها بالحركة.
\end{enumerate}
\end{problem}
