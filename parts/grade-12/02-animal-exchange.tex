\chapter{Animal Exchange Systems}\label{ch:g12:animal-exchange}

Every animal \omterm{def:g10:the-cell:cell}{cell} needs nutrients and oxygen and must dump carbon dioxide
and nitrogenous wastes. Unicellular \omterm{def:g10:scientific-biology:organism}{organisms} exchange these substances
directly with the water around them. Large multicellular animals cannot:
distance and \omterm{def:g10:the-cell:sa-v}{volume} conspire against pure \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}. This chapter shows
how specialised surfaces, digestive tracts, circulatory pumps and gas
exchangers solve the same problem at animal scale.

\section{Exchange surfaces and size}

\begin{definition}[Exchange surface]\label{def:g12:animal-exchange:surface}
An \emph{exchange surface}\index{exchange surface} is a specialised
interface across which substances move between an \omterm{def:g10:scientific-biology:organism}{organism} and its
environment (or between internal compartments) by \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}, facilitated
transport or \omterm{def:g12:animal-exchange:bulk}{bulk flow}. Effective surfaces are typically thin, moist and
highly folded to raise area relative to the \omterm{def:g10:the-cell:sa-v}{volume} served.
\end{definition}

\begin{proposition}[Surface-to-volume ratio]\label{prop:g12:animal-exchange:sv}
As a body scales up geometrically, \omterm{def:g10:the-cell:sa-v}{volume} grows as length cubed while
\omterm{def:g10:the-cell:sa-v}{surface area} grows only as length squared. The \emph{surface-to-volume
ratio}\index{surface-to-volume ratio} therefore falls with size: large
animals need internal exchange organs, not a plain outer skin alone.
\end{proposition}
\admitted

\begin{definition}[Bulk flow]\label{def:g12:animal-exchange:bulk}
\emph{Bulk flow}\index{bulk flow} is the movement of a fluid (blood,
\omterm{def:g12:animal-exchange:open}{haemolymph}, air, water) driven by pressure differences, carrying many
solutes together. It complements \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}, which is efficient only over
short distances (micrometres).
\end{definition}

\section{Digestion and absorption}

\begin{definition}[Digestion]\label{def:g12:animal-exchange:digestion}
\emph{Digestion}\index{digestion} breaks food into molecules small
enough to absorb: mechanical (chewing, churning) and chemical (\omterm{def:g10:membranes-enzymes:enzyme}{enzymes}
hydrolysing \omterm{def:g10:molecules-of-life:polymer}{polymers}). \emph{Absorption}\index{absorption} transfers
those products across the gut epithelium into blood or lymph.
\end{definition}

\begin{definition}[Alimentary canal]\label{def:g12:animal-exchange:canal}
An \emph{alimentary canal}\index{alimentary canal} (complete gut) is a
one-way tube from mouth to anus, allowing regional specialisation
(stomach, small intestine, large intestine). Incomplete guts (e.g.\ a
gastrovascular cavity with one opening) mix intake and waste.
\end{definition}

\begin{proposition}[Intestinal surface amplification]
\label{prop:g12:animal-exchange:villi}
In vertebrates the small intestine multiplies area by circular folds,
\emph{villi}\index{villus} and microvilli on epithelial \omterm{def:g10:the-cell:cell}{cells} --- often
by two \omterm{def:g10:biodiversity-classification:hierarchy}{orders} of magnitude relative to a smooth tube of the same length.
\end{proposition}
\admitted

\begin{example}[Human path of a starch meal]
\label{ex:g12:animal-exchange:starch}
Salivary and pancreatic amylases cleave starch to maltose; brush-border
\omterm{def:g10:membranes-enzymes:enzyme}{enzymes} yield glucose. Glucose is absorbed by epithelial \omterm{def:g10:the-cell:cell}{cells} (often
with Na$^{+}$ cotransport) and exits to capillaries bound for the liver
via the hepatic portal vein.
\end{example}

\begin{remark}[Intracellular vs extracellular digestion]
\label{rem:g12:animal-exchange:extra}
Sponges and some cnidarians rely heavily on intracellular \omterm{def:g12:animal-exchange:digestion}{digestion} after
phagocytosis. Most bilaterians digest extracellularly in a gut lumen,
then absorb products --- allowing larger prey and continuous processing.
\end{remark}

\section{Circulation: open and closed systems}

\begin{definition}[Circulatory system]\label{def:g12:animal-exchange:circ}
A \emph{circulatory system}\index{circulatory system} transports fluid
that distributes gases, nutrients, hormones and wastes. It includes a
pump (heart or contractile vessels), a fluid and vessels or sinuses.
\end{definition}

\begin{definition}[Open circulation]\label{def:g12:animal-exchange:open}
In an \emph{open circulatory system}\index{open circulatory system}
(many arthropods, most molluscs), a heart pumps \emph{haemolymph}
\index{haemolymph} into sinuses bathing the organs; fluid is not always
confined to vessels.
\end{definition}

\begin{definition}[Closed circulation]\label{def:g12:animal-exchange:closed}
In a \emph{closed circulatory system}\index{closed circulatory system}
(annelids, cephalopods, vertebrates), blood remains inside vessels;
exchange with tissues occurs across thin capillary walls. Pressure and
flow can be regulated regionally.
\end{definition}

\begin{definition}[Single and double circulation]
\label{def:g12:animal-exchange:double}
Fish have \emph{single circulation}\index{single circulation}: heart
$\to$ \omterm{def:g12:animal-exchange:organs}{gills} $\to$ body $\to$ heart. Mammals and birds have \emph{double
circulation}\index{double circulation}: a pulmonary circuit (heart
$\leftrightarrow$ \omterm{def:g12:animal-exchange:organs}{lungs}) and a systemic circuit (heart $\leftrightarrow$
body), so oxygenated blood is repressurised after the \omterm{def:g12:animal-exchange:organs}{lungs}.
\end{definition}

\begin{proposition}[Advantage of double circulation]
\label{prop:g12:animal-exchange:double}
Separating pulmonary and systemic circuits allows high systemic pressure
for large active bodies without over-pressuring delicate \omterm{def:g12:animal-exchange:organs}{lung}
capillaries.
\end{proposition}
\admitted

\begin{example}[Mammalian path of a red blood cell]
\label{ex:g12:animal-exchange:rbc}
Right ventricle $\to$ pulmonary arteries $\to$ \omterm{def:g12:animal-exchange:organs}{lung} capillaries (load
O$_2$, unload CO$_2$) $\to$ pulmonary veins $\to$ left atrium $\to$ left
ventricle $\to$ aorta $\to$ systemic capillaries $\to$ venae cavae
$\to$ right atrium.
\end{example}

\section{Gas exchange}

\begin{definition}[Gas exchange]\label{def:g12:animal-exchange:gas}
\emph{Gas exchange}\index{gas exchange} is the uptake of O$_2$ and
release of CO$_2$ across a respiratory surface by \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}, driven by
partial-pressure gradients.
\end{definition}

\begin{definition}[Gills, tracheae, lungs]
\label{def:g12:animal-exchange:organs}
\emph{Gills}\index{gill} are external or chambered outgrowths for
water-breathing. \emph{Tracheal systems}\index{tracheal system} in
insects pipe air directly to tissues via spiracles. \emph{Lungs}
\index{lung} are invaginated air sacs; in mammals, branching airways end
in \emph{alveoli}\index{alveolus}.
\end{definition}

\begin{proposition}[Fick's logic for gas exchange]
\label{prop:g12:animal-exchange:fick}
Diffusive flux rises with \omterm{def:g10:the-cell:sa-v}{surface area} and partial-pressure difference,
and falls with membrane thickness. Respiratory organs maximise area and
gradient and minimise distance --- often with ventilation and perfusion
keeping gradients steep.
\end{proposition}
\admitted

\begin{definition}[Countercurrent exchange]
\label{def:g12:animal-exchange:counter}
In fish \omterm{def:g12:animal-exchange:organs}{gills}, blood flows opposite to water across the lamellae
(\emph{countercurrent exchange}\index{countercurrent exchange}), so a
favourable O$_2$ gradient is maintained along the whole \omterm{def:g12:animal-exchange:surface}{exchange
surface} and extraction efficiency is high.
\end{definition}

\begin{example}[Insect tracheae]\label{ex:g12:animal-exchange:trachea}
Insect blood does not carry bulk respiratory O$_2$; air-filled tracheoles
approach mitochondria. That design supports high metabolic rates in
small bodies but scales poorly to very large sizes in dense air.
\end{example}

\begin{remark}[Link to cellular respiration]
\label{rem:g12:animal-exchange:link}
Exchange systems feed mitochondria: absorbed glucose and inhaled O$_2$
meet in \omterm{def:g10:cellular-respiration:def}{cellular respiration}; CO$_2$ and H$_2$O are reverse traffic for
the \omterm{def:g12:animal-exchange:organs}{lungs} and kidneys (osmoregulation is treated more fully in later
physiology).
\end{remark}

\section{Integrated design}

\begin{method}[Auditing an animal's exchange strategy]
\label{met:g12:animal-exchange:audit}
For any animal, ask:
\begin{enumerate}
  \item What is its size and metabolic rate? (Sets demand.)
  \item Is the respiratory medium air or water? (O$_2$ content and
        viscosity differ sharply.)
  \item Open or closed circulation? Single or double circuit?
  \item How is \omterm{def:g10:the-cell:sa-v}{surface area} amplified (\omterm{def:g12:animal-exchange:organs}{gills}, \omterm{def:g12:animal-exchange:organs}{alveoli}, \omterm{prop:g12:animal-exchange:villi}{villi})?
  \item How is \omterm{def:g12:animal-exchange:bulk}{bulk flow} used (ventilation, heart, gut motility)?
\end{enumerate}
\end{method}

\begin{proposition}[Matching of systems]\label{prop:g12:animal-exchange:match}
Digestive, circulatory and respiratory capacities are co-adapted:
endurance athletes enlarge cardiac output and capillary density;
high-energy fish couple large \omterm{def:g12:animal-exchange:organs}{gill} area with efficient countercurrent
flow.
\end{proposition}
\admitted

\section{Problem: Building an exchange animal}

\begin{problem}[{Weekend problem --- from diffusion limits to a
coherent design for digestion, circulation and breathing}]
\label{pb:g12:animal-exchange:1}
You are asked to reason about two animals: a small flatworm living in
pond water, and a large endothermic mammal on land.

\medskip
\noindent\textbf{Part I --- Scale and \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}.}
\begin{enumerate}
  \item State why \omterm{def:g10:membranes-enzymes:diffusion}{diffusion} alone can serve the flatworm's \omterm{def:g10:the-cell:cell}{cells} but
        not those deep in the mammal.
  \item Using $A/V = 6/L$ for a cube, compare $A/V$ for $L=\qty{1}{mm}$
        and $L=\qty{100}{mm}$ (ratios only).
  \item Name three internal structures in the mammal that restore
        effective exchange area.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- \omterm{def:g12:animal-exchange:digestion}{Digestion}.}
\begin{enumerate}[resume]
  \item Why is a one-way \omterm{def:g12:animal-exchange:canal}{alimentary canal} advantageous over a single
        opening gastrovascular cavity for a large active feeder?
  \item Describe how \omterm{prop:g12:animal-exchange:villi}{villi} and microvilli cooperate with capillaries
        (and lymphatics) after a meal.
  \item Predict how gut length and \omterm{def:g10:the-cell:sa-v}{surface area} might differ between a
        herbivore and a carnivore of similar body mass; justify.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Circulation and \omterm{def:g12:animal-exchange:gas}{gas exchange}.}
\begin{enumerate}[resume]
  \item Choose open or closed circulation for the mammal and justify.
  \item Sketch in words the double circuit and the advantage over a
        fish-like single circuit for a high metabolic rate.
  \item Compare the respiratory organs of a fish and a mammal: medium,
        surface geometry, and how the medium is moved.
  \item Explain \omterm{def:g12:animal-exchange:counter}{countercurrent exchange} in \omterm{def:g12:animal-exchange:organs}{gills} and why \omterm{def:g12:animal-exchange:organs}{lungs} use a
        different strategy (tidal ventilation and large alveolar area).
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Integration.}
\begin{enumerate}[resume]
  \item During hard exercise, how must heart rate, ventilation and
        local blood flow to muscle change together?
  \item In one short paragraph, argue that ``exchange systems are
        matched designs'', using one \omterm{def:g11:human-genetics:polygenic}{quantitative} idea (area, pressure
        or gradient) and one comparative example.
\end{enumerate}
\end{problem}
