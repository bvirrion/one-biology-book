\chapter{تلف المجين والسرطان}\label{ch:g11:cancer}

تحت المجهر، تبدو شريحة من بطانة معي سليمة صورةً للنظام: \omterm{def:g10:cells-common-unit:cell}{خلايا} في طبقات
مرتبة، لكل منها \omterm{def:g10:cells-common-unit:organelle}{نواة} متواضعة، تنقسم في أسفل الطبقة وتموت في أعلاها،
على وتيرة واحدة. وعلى بعد بضعة مليمترات، في الشريحة نفسها، رقعة من
\omterm{def:g10:cells-common-unit:cell}{الخلايا} قد خرجت عن الصف: مزدحمة، غريبة الشكل، بأنوية ضخمة داكنة،
تنقسم في أي مكان وتدفع في النسيج الذي تحتها. وكل \omterm{def:g10:cells-common-unit:cell}{خلية} من الرقعة تنحدر
من \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة اكتسبت، قبل سنين، \omterm{def:g11:mutations:mutation}{طفرة} تتيح لها الانقسام حين ما كان
ينبغي لها --- ثم اكتسبت، على مدى السنين، بضع \omterm{def:g11:mutations:mutation}{طفرات} أخرى. وهذا الفصل عن
كيف تخفق ضوابط \omterm{def:g11:cell-cycle-mitosis:cycle}{الدورة الخلوية}، وما الذي يتلفها، وما الذي يمكن فعله
قبل وبعد.

\section{الانقسام حين لا ينبغي}

\begin{definition}[السرطان]\label{def:g11:cancer:cancer}
\emph{السرطان}\index{سرطان} جمهرة من \omterm{def:g10:cells-common-unit:cell}{الخلايا} تنقسم من دون الضوابط التي
تحد انقسام \omterm{def:g10:cells-common-unit:cell}{الخلايا} الطبيعية، وتغزو النسج المحيطة. وهو يبدأ \omterm{def:g10:cells-common-unit:cell}{بخلية} واحدة
تراكمت في \omterm{def:g10:universal-dna:information}{DNA} فيها \omterm{def:g11:mutations:mutation}{طفرات} في \omterm{def:g10:universal-dna:gene}{المورّثات} الضابطة \omterm{def:g11:cell-cycle-mitosis:cycle}{للدورة الخلوية}؛ وتكوّن
ذرية تلك \omterm{def:g10:cells-common-unit:cell}{الخلية} \emph{ورمًا}\index{ورم}. والورم الذي يبقى محصورًا
\emph{حميد}؛ أما الذي تغزو خلاياه النسيج المجاور وتسافر عبر الدم أو
اللمف لتؤسس أورامًا جديدة في مواضع أخرى (\emph{النقائل}) فهو
\emph{خبيث}.
\end{definition}

\begin{proposition}[السرطان مرض في المجين]\label{prop:g11:cancer:genome}
تحمل \omterm{def:g10:cells-common-unit:cell}{خلايا} \omterm{def:g11:cancer:cancer}{الورم} \omterm{def:g11:mutations:mutation}{طفرات} لا تحملها سائر \omterm{def:g10:cells-common-unit:cell}{خلايا} صاحبه: أي \omterm{def:g11:mutations:mutation}{طفرات} جسدية،
نشأت في \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة وورثتها ذريتها. \omterm{def:g11:cancer:cancer}{فالورم} \emph{نسيلة}، وهو يتطور ---
إذ تسود \omterm{def:g10:cells-common-unit:cell}{الخلايا} التي تنقسم أسرع أو تبقى أفضل، وتضيف \omterm{def:g11:mutations:mutation}{طفرات} لاحقة القدرة
على الغزو والانتشار.
\end{proposition}
\begin{proof}[شواهد]
تعيين متتالية \omterm{def:g10:universal-dna:information}{DNA} في \omterm{def:g10:cells-common-unit:cell}{خلايا} \omterm{def:g11:cancer:cancer}{الورم} إلى جانب متتالية \omterm{def:g10:cells-common-unit:cell}{خلايا} المريض
الطبيعية يجد آلاف الفروق، كلها موجودة في \omterm{def:g11:cancer:cancer}{الورم}، ولا شيء منها في
النسيج السليم؛ \omterm{def:g10:universal-dna:gene}{والمورّثات} القليلة نفسها توجد طافرة في \omterm{def:g11:cancer:cancer}{ورم} بعد \omterm{def:g11:cancer:cancer}{ورم}. وكل
\omterm{def:g10:cells-common-unit:cell}{خلية} \omterm{def:g11:cancer:cancer}{ورم} عند امرأة تحمل \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} نفسه من صبغيي X عندها مطفأً --- وهي
بصمة الانحدار من \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة. ثم إن المطفِّرات مسرطنات: فالعوامل التي
ترفع نسبة \omterm{def:g11:mutations:mutation}{الطفرات} (\cref{ch:g11:mutations}) ترفع نسبة \omterm{def:g11:cancer:cancer}{السرطان}، بتأخير
عقود.
\end{proof}

\section{مسرّعات وكوابح}

\begin{proposition}[صنفان من المورّثات يضبطان الدورة]\label{prop:g11:cancer:genes}
قرارُ \omterm{def:g10:cells-common-unit:cell}{الخلية} أن تنقسم تتخذه بروتينات تعمل عمل دواسة الوقود والمكابح في
مركبة.
\begin{itemize}
  \item \emph{المورّثات الورمية الأولية}\index{مورّثة ورمية أولية}
        ترمز لبروتينات تدفع \omterm{def:g10:cells-common-unit:cell}{الخلية} إلى الدورة حين تصل إشارة نمو ---
        وهي مستقبلات إشارات النمو، والمرحّلات التي تحمل الإشارة إلى
        \omterm{def:g10:cells-common-unit:organelle}{النواة}. \omterm{def:g11:mutations:mutation}{والطفرة} التي تترك مثل ذلك \omterm{def:g11:gene-expression:protein}{البروتين} فعالًا على الدوام
        تحوّل \omterm{def:g10:universal-dna:gene}{المورّثة} إلى \emph{\omterm{def:g10:universal-dna:gene}{مورّثة} ورمية}: أي دواسة وقود عالقة
        مضغوطة. ويكفي \omterm{def:g10:universal-dna:gene}{أليل} طافر واحد.
  \item \emph{المورّثات الكابحة للورم}\index{مورّثة كابحة للورم}
        ترمز لبروتينات توقف الدورة --- عند نقطتي التحقق قبل S وقبل
        \omterm{def:g11:cell-cycle-mitosis:cycle}{الانقسام المتساوي} --- حين يتضرر \omterm{def:g10:universal-dna:information}{DNA}، وتطلق الإصلاح أو، إن جاوز
        الضرر الإصلاح، هدمَ \omterm{def:g10:cells-common-unit:cell}{الخلية} لنفسها. \omterm{def:g11:mutations:mutation}{والطفرة} التي تعطل مثل ذلك
        \omterm{def:g11:gene-expression:protein}{البروتين} تزيل كابحًا؛ ولما كان \omterm{def:g10:cells-common-unit:cell}{للخلية} أليلان، وجب فقدهما معًا.
\end{itemize}
وأشهر الكوابح، وهو \omterm{def:g11:gene-expression:protein}{البروتين} p53، معطَّل في نصف السرطانات البشرية كلها.
\end{proposition}
\admitted

\begin{example}[مورّثتان مألوفتان]\label{ex:g11:cancer:genes}
مستقبل لإشارة نمو يقع في غشاء \omterm{def:g10:cells-common-unit:cell}{خلايا} كثيرة؛ واستبدال واحد في مورّثته
يجعله يشير باستمرار، من دون أي إشارة خارجية --- وهي \omterm{def:g10:universal-dna:gene}{مورّثة} ورمية توجد
في ثلث سرطانات القولون والرئة. أما \omterm{def:g10:universal-dna:gene}{مورّثة} p53 فتعطَّل \omterm{def:g11:mutations:mutation}{بالطفرات} بكل
طريقة ممكنة: باستبدالات تبدل شكلها، وبحذوف، وبتوقف سابق لأوانه.
\omterm{def:g10:cells-common-unit:cell}{والخلية} التي فقدت p53 لا تقف لتصلح \omterm{def:g10:universal-dna:information}{DNA} فيها قبل نسخه، فتراكم \omterm{def:g11:mutations:mutation}{طفرات}
أخرى أسرع: ففقد ذلك الكابح يسرّع العملية كلها.
\end{example}

\section{خطوات عدة، وسنون كثيرة}

\begin{proposition}[السرطان عملية متعددة الخطوات]\label{prop:g11:cancer:multistep}
\omterm{def:g11:mutations:mutation}{الطفرة} الواحدة لا تصنع سرطانًا. فعلى \omterm{def:g10:cells-common-unit:cell}{الخلية} أن تراكم عدة \omterm{def:g11:mutations:mutation}{طفرات} ---
عادةً \omterm{def:g10:universal-dna:gene}{مورّثة} ورمية تنشَّط وكابحان أو أكثر يفقدان --- وكل \omterm{def:g11:mutations:mutation}{طفرة} تالية،
بزيادتها تكاثر النسيلة، تزيد عدد \omterm{def:g10:cells-common-unit:cell}{الخلايا} التي يمكن أن تقع فيها التي
تليها. ويستغرق التتابع سنين إلى عقود، ولذلك يرتفع وقوع معظم السرطانات
ارتفاعًا حادًا مع العمر، ولذلك يظهر أثر \omterm{def:g11:cancer:carcinogen}{المسرطن} بعد التعرض بعشرين أو
ثلاثين سنة.
\end{proposition}
\begin{proof}[شواهد]
في القولون ترى المراحل: سليلة حميدة صغيرة تحمل \omterm{def:g11:mutations:mutation}{طفرة} كابح واحدة؛ ثم
سليلة أكبر أضيفت إليها \omterm{def:g10:universal-dna:gene}{مورّثة} ورمية؛ ثم \omterm{def:g11:cancer:cancer}{ورم} غاز فقد p53 كذلك --- أي
تتابع من أربع إلى ست \omterm{def:g11:mutations:mutation}{طفرات} على مدى عشر إلى عشرين سنة، وكل مرحلة توجد
في \omterm{def:g10:cells-common-unit:cell}{خلايا} التي تليها. ووقوع \omterm{def:g11:cancer:cancer}{السرطان} يرتفع تقريبًا كالقوة الخامسة للعمر،
وهو المنحني المنتظر إذا وجب وقوع نحو خمسة أو ستة أحداث نادرة مستقلة
كلها في \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة. والذين يرثون أليلًا طافرًا واحدًا \omterm{def:g10:universal-dna:gene}{لمورّثة} كابحة
يصابون \omterm{def:g11:cancer:cancer}{بالسرطان} الموافق أبكر وأكثر، إذ خطوة واحدة قد قطعت في كل \omterm{def:g10:cells-common-unit:cell}{خلية}
عندهم.
\end{proof}

\section{ما الذي يتلف المجين}

\begin{definition}[المسرطن]\label{def:g11:cancer:carcinogen}
\emph{المسرطن}\index{مسرطن} عامل يرفع خطر \omterm{def:g11:cancer:cancer}{السرطان}. ومعظم المسرطنات
مطفِّرات: كفوق البنفسجي في ضوء الشمس (سرطانات الجلد)، وقطران دخان
التبغ (الرئة والفم والمثانة)، والإشعاع المؤين، وألياف الأسبستوس،
والكحول، وبعض العفن. وبعضها يعمل على نحو آخر: ففيروسات بعينها تحمل
\omterm{def:g10:universal-dna:gene}{مورّثات} تعطل كوابح \omterm{def:g10:cells-common-unit:cell}{الخلية} --- فالفيروسات الحليمية البشرية تسبب كل
سرطانات عنق الرحم تقريبًا، وفيروسات التهاب الكبد كثيرًا من سرطانات
الكبد --- والالتهاب المزمن أو الهرمونات التي تبقي نسيجًا منقسمًا ترفع
عدد الانقسامات التي يمكن أن تقع فيها \omterm{def:g11:mutations:mutation}{الطفرات}.
\end{definition}

\begin{example}[التبغ]\label{ex:g11:cancer:tobacco}
يحمل الدخان نحو سبعين مسرطنًا؛ ويلتصق البنزوبيرين، حالما تنشّطه
\omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيمات} \omterm{def:g10:cells-common-unit:cell}{الخلايا} نفسها، بالغوانينات ويسبب استبدالات في تلك المواضع.
\omterm{def:g10:universal-dna:gene}{ومورّثة} p53 في \omterm{def:g11:cancer:cancer}{ورم} رئة مدخن تحمل عادةً استبدالًا من هذا القبيل بالضبط،
في أحد مواضع قليلة يرتبط بها البنزوبيرين أسهل ما يرتبط --- \omterm{def:g11:mutations:mutagen}{فالمطفِّر}
يترك بصمته في المتتالية. وكان \omterm{def:g11:cancer:cancer}{سرطان} الرئة نادرًا قبل السجائر؛ والتدخين
يسبب تسع حالات من عشر، والإقلاع ينصّف الخطر في غضون عشر سنين، لأن
\omterm{def:g10:cells-common-unit:cell}{الخلايا} التي حملت الخطوات الباكرة تستبدل بالتدريج.
\end{example}

\begin{proposition}[الاستعداد الموروث]\label{prop:g11:cancer:inherited}
\omterm{def:g11:cancer:cancer}{السرطان} نفسه لا يورث --- فهو مرض \omterm{def:g11:mutations:mutation}{طفرات} جسدية --- لكن أليلًا طافرًا
\omterm{def:g10:universal-dna:gene}{لمورّثة} كابحة يمكن أن يورث. فمن يرث أليلًا واحدًا غير عامل لمثل تلك
\omterm{def:g10:universal-dna:gene}{المورّثة} يكون قد قطع خطوة واحدة في كل \omterm{def:g10:cells-common-unit:cell}{خلية} من جسمه؛ \omterm{def:g11:mutations:mutation}{وطفرة} جسدية واحدة
في \omterm{def:g10:universal-dna:gene}{الأليل} الآخر تتم الفقد في أي \omterm{def:g10:cells-common-unit:cell}{خلية}، بدل الطفرتين اللتين تلزمان
غيره. والنتيجة خطر مرتفع للسرطانات الموافقة، وأبكر في العمر: فنحو
70\% من النساء الحاملات \omterm{def:g10:universal-dna:gene}{لأليل} طافر \omterm{def:g10:universal-dna:gene}{لمورّثة} واحدة بعينها من \omterm{def:g10:universal-dna:gene}{مورّثات}
\omterm{def:g11:cancer:cancer}{سرطان} الثدي يصبن بالمرض، مقابل 12\% في العموم.
\end{proposition}
\admitted

\section{الوقاية والكشف والعلاج}

\begin{method}[خفض الخطر]\label{met:g11:cancer:prevention}
لما كانت معظم \omterm{def:g11:mutations:mutation}{الطفرات} وراء السرطانات آتية من تعرضات يمكن تعيينها،
ولما كانت العملية تستغرق سنين، كان \omterm{def:g11:cancer:cancer}{السرطان} قابلًا للوقاية والكشف إلى
حد بعيد.
\begin{enumerate}
  \item \emph{تجنّب المطفِّرات}: لا تبغ (وهو ثلث وفيات \omterm{def:g11:cancer:cancer}{السرطان} كلها)،
        وكحول قليل، ووقاية من الشمس، ولا إشعاع بلا ضرورة.
  \item \emph{لقّح} ضد الفيروسات المسببة \omterm{def:g11:cancer:cancer}{للسرطان}: فلقاح الفيروس
        الحليمي يمنع سرطانات عنق الرحم كلها تقريبًا إذا أعطي قبل
        التعرض.
  \item \emph{افحص}: اعثر على \omterm{def:g11:cancer:cancer}{الورم} في مرحلة السليلة --- بلطاخات عنق
        الرحم، وتنظير القولون، والتصوير الشعاعي للثدي --- حين يكون
        استئصاله بسيطًا وشافيًا.
  \item \emph{اعرف العائلة}: فالاستعداد الموروث يقتضي مراقبة أبكر
        وأدق.
\end{enumerate}
\end{method}

\begin{example}[العلاجات وما تستهدفه]\label{ex:g11:cancer:treatments}
الجراحة تستأصل ورمًا لم ينتشر. والمعالجة الشعاعية تكسر \omterm{def:g10:universal-dna:information}{DNA} في \omterm{def:g10:cells-common-unit:cell}{الخلايا}
الواقعة في الحزمة كسرًا لا يصلح، فتقتل \omterm{def:g10:cells-common-unit:cell}{الخلايا} المنقسمة تفضيلًا.
والمعالجة الكيميائية تستعمل أدوية تحصر التضاعف أو المغزل
(\cref{ch:g11:cell-cycle-mitosis}): \omterm{def:g10:cells-common-unit:cell}{فخلايا} \omterm{def:g11:cancer:cancer}{الورم}، وهي تنقسم باستمرار،
تموت أسرع من \omterm{def:g10:cells-common-unit:cell}{الخلايا} الطبيعية --- لكن النسج الطبيعية السريعة الانقسام
(نقي العظم، وبطانة المعي، وجذور الشعر) تصاب كذلك، ومن هناك تأتي
الآثار الجانبية. أما العلاجات الأحدث فتستهدف \omterm{def:g11:gene-expression:protein}{البروتين} الطافر الخاص
\omterm{def:g11:cancer:cancer}{بورم} ما، أو تطلق الجهاز المناعي عليه.
\end{example}

\begin{remark}[لماذا نحمل كلنا البدايات]\label{rem:g11:cancer:everyone}
كل بالغ يحمل نسائل من \omterm{def:g10:cells-common-unit:cell}{خلايا} فيها \omterm{def:g11:mutations:mutation}{طفرات} ذات صلة \omterm{def:g11:cancer:cancer}{بالسرطان} --- في الجلد
والمعي والدم --- لا تصير سرطانات أبدًا: فالكوابح الباقية تصمد، والجهاز
المناعي يزيل أسوأها، ويموت الشخص بشيء آخر قبل ذلك. \omterm{def:g11:cancer:cancer}{فالسرطان} حصيلة
عارضة لعملية تجري عند الجميع؛ وغاية الوقاية أن يبقى عدد الخطوات
المقطوعة، في أي \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة، دون العدد اللازم.
\end{remark}

\section{مسألة: البصمة في المورّثة}

\begin{problem}[{مسألة نهاية الأسبوع --- ورم رئة تعيَّن متتاليته:
طفرات تعد، وبصمة المدخن تقرأ، وسنون تحسب، والخطر الذي تحمله سيجارة}]
\label{pb:g11:cancer:1}
\omterm{def:g11:cancer:cancer}{ورم} رئة ونسيج سليم من المريض نفسه، وهو مدخن عشرين سيجارة في اليوم
خمسًا وثلاثين سنة، تعيَّن متتاليتاهما. ويحمل \omterm{def:g11:cancer:cancer}{الورم} $\num{5000}$
استبدال جسدي لا توجد في النسيج السليم؛ و75\% منها تستبدل G بالقاعدة T، وهو
التغير الذي يسببه البنزوبيرين. ومن بينها: واحد في \omterm{def:g10:universal-dna:gene}{مورّثة} p53، ينشئ
توقفًا سابقًا لأوانه، على \omterm{def:g10:universal-dna:gene}{أليل} واحد؛ \omterm{def:g10:universal-dna:gene}{والأليل} الآخر من p53 محذوف؛ وواحد
في \omterm{def:g10:universal-dna:gene}{مورّثة} مستقبل لإشارة نمو، تجعل \omterm{def:g11:gene-expression:protein}{البروتين} فعالًا على الدوام.

\medskip
\noindent\textbf{الجزء الأول --- قراءة المتتالية.}
\begin{enumerate}
  \item لماذا تسمى الاستبدالات $\num{5000}$ جسدية؟ وأين كانت لتوجد لو
        كانت سلالية؟
  \item في \omterm{def:g11:cancer:cancer}{ورم} غير مدخن، تكون الاستبدالات من كل الأصناف بنسب متقاربة.
        فبم تخبرك نسبة 75\% من التغيرات من G إلى T؟
  \item صنّف \omterm{def:g11:mutations:mutation}{طفرة} p53 (بمفردات \cref{ch:g11:gene-expression}) وأثرها
        في \omterm{def:g11:gene-expression:protein}{البروتين}.
  \item فسّر لماذا وجب فقد أليلي p53 معًا، بينما كفى \omterm{def:g10:universal-dna:gene}{أليل} طافر واحد
        \omterm{def:g10:universal-dna:gene}{لمورّثة} المستقبل.
  \item أي المورّثتين مسرّع وأيهما كابح؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- السنون.}
\begin{enumerate}[resume]
  \item دخّن المريض نحو $20 \times 365 \times 35$ سيجارة. احسب
        العدد.
  \item إذا أنتجت كل سيجارة، وسطيًا، استبدالًا باقيًا واحدًا في كل
        \omterm{def:g10:cells-common-unit:cell}{خلية} من \omterm{def:g10:cells-common-unit:cell}{خلايا} بطانة الرئة، فكم استبدالًا تحمل \omterm{def:g10:cells-common-unit:cell}{الخلية} بعد 35
        سنة؟ قارن بالعدد $\num{5000}$ الموجود.
  \item \omterm{def:g11:mutations:mutation}{طفرات} \omterm{def:g11:cancer:cancer}{الورم} ليست كلها في \omterm{def:g10:cells-common-unit:cell}{خلايا} بالترتيب نفسه: فطفرتا p53
        والمستقبل في كل \omterm{def:g10:cells-common-unit:cell}{خلية} \omterm{def:g11:cancer:cancer}{ورم}، أما الآلاف الأخرى ففي مجموعات جزئية
        فقط. فما الذي يقوله ذلك عن زمن وقوع الطفرتين المفتاحيتين؟
  \item عرض \omterm{def:g11:cancer:cancer}{الورم} \qty{3}{cm}، أي نحو $10^{10}$ \omterm{def:g10:cells-common-unit:cell}{خلية}، \omterm{def:g10:cells-common-unit:cell}{وخلايا} \omterm{def:g11:cancer:cancer}{الورم}
        تتضاعف كل 100 يوم. فكم مضاعفة من \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة، وكم سنة على
        الأقل؟
  \item اجمع السؤالين 8 و9: متى تقريبًا في السنين الخمس والثلاثين
        وقعت آخر \omterm{def:g11:mutations:mutation}{الطفرات} المفتاحية؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- خطر سيجارة.}
\begin{enumerate}[resume]
  \item من شكل التبغ، كان الخطر النسبي عند المريض نحو 17. فإذا أصيب
        1\% من غير المدخنين \omterm{def:g11:cancer:cancer}{بسرطان} الرئة، فأي نسبة من المدخنين مثله
        تصاب؟
  \item من بين 100 مدخن كهذا، كم سرطانًا يعزى إلى التدخين؟
  \item لو أنه أقلع في الأربعين (بعد 20 سنة)، لكان خطره نحو ثلث خطر
        المدخن المستمر. فسّر، بنموذج الخطوات المتعددة وتجدد بطانة
        الرئة، لماذا يهبط الخطر بعد الإقلاع.
  \item لماذا يأتي \omterm{def:g11:cancer:cancer}{سرطان} المدخن، حين يأتي، بعد ثلاثين سنة عادةً لا بعد
        خمس؟
  \item يحتج تلميذ بأن السيجارة، ما دامت تسبب «\omterm{def:g11:mutations:mutation}{طفرة} واحدة فقط في كل
        \omterm{def:g10:cells-common-unit:cell}{خلية}»، غير مؤذية. أجب بحساب السؤال 7 وبعدد \omterm{def:g10:cells-common-unit:cell}{الخلايا}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الوقاية والجمهرة.}
في بلد فيه 60 مليون نسمة، يدخن 25\% من البالغين؛ ويقتل \omterm{def:g11:cancer:cancer}{سرطان} الرئة
$\num{30000}$ شخص في السنة، 90\% منهم مدخنون.
\begin{enumerate}[resume]
  \item احسب وفيات \omterm{def:g11:cancer:cancer}{سرطان} الرئة السنوية بين المدخنين وبين غير
        المدخنين، ونسبة الوفيات لكل 100\,000 في كل مجموعة (خذ 45
        مليون بالغ).
  \item بأي عامل تكون النسبة أعلى عند المدخنين؟ قارنها بالخطر النسبي
        في الشكل وعلّق.
  \item لو اختفى التدخين اليوم، فماذا يصير عدد وفيات \omterm{def:g11:cancer:cancer}{سرطان} الرئة
        السنوي على المدى البعيد؟ ولماذا لا يصير كذلك في الحال؟
  \item الفحص بمسح منخفض الجرعة يجد الأورام أبكر ويخفض الوفيات بين
        المدخنين المفرطين نحو 20\%. قارن، بعدد الوفيات المتفاداة في
        السنة، بين فحص كل مدخن وبين الإقلاع عن التدخين.
  \item اذكر النتيجة: ما الذي أثبتته بصمة الاستبدال من G إلى T عن أصل
        \omterm{def:g11:mutations:mutation}{طفرات} \omterm{def:g11:cancer:cancer}{الورم}، والأحداث الوراثية الثلاثة التي صنعت \omterm{def:g11:cancer:cancer}{السرطان}،
        والقرار الواحد الذي كان ليمنعه.
\end{enumerate}
\end{problem}
