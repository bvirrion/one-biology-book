\chapter{Bunga, Buah dan Biji}\label{ch:g3:flowers-fruits-seeds}

Pada bulan April pohon ceri memutih oleh \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}; pada bulan Juni ia
sarat oleh \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} ceri. Ke mana perginya \omterm{def:g3:flowers-fruits-seeds:parts}{bunga-bunga} itu --- dan dari
mana datangnya \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya}? Jawabannya adalah salah satu perubahan
terindah di dunia yang hidup: \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya} tidak lenyap, ia \emph{menjadi}
\omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya}.

\section{Mengamati bagian dalam sekuntum bunga}

\begin{definition}[Bagian-bagian bunga]\label{def:g3:flowers-fruits-seeds:parts}
Suatu \emph{bunga}\index{bunga} adalah bagian \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang kelak
membuat \omterm{def:g2:from-seed-to-plant:seed}{biji}. Dari luar ke dalam:
\begin{enumerate}
  \item \emph{mahkota bunga}\index{mahkota bunga}, yang berwarna dan
        berbau harum, yang memikat \omterm{prop:g3:sorting-living-things:groups}{serangga} yang berkunjung;
  \item \emph{benang sari}\index{benang sari}, tangkai tipis yang
        berdebu \emph{serbuk sari}\index{serbuk sari}, sebuah bubuk
        kuning yang halus;
  \item tepat di tengahnya, \emph{putik}\index{putik}, sebuah labu
        kecil; jauh di dalamnya terbaring calon \omterm{def:g2:from-seed-to-plant:seed}{biji} bunganya.
\end{enumerate}
\end{definition}

\begin{example}[Membedah sekuntum tulip]\label{ex:g3:flowers-fruits-seeds:tulip}
Bukalah sekuntum tulip yang layu pelan-pelan, mahkota demi mahkota. Di
dalamnya berdiri \omterm{def:g3:flowers-fruits-seeds:parts}{benang sarinya} --- sapukan satu ke jarimu dan ia
meninggalkan debu \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} yang kuning --- dan di tengahnya \omterm{def:g3:flowers-fruits-seeds:parts}{putik}
hijau yang gemuk. Iris \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya} melintang (pekerjaan pisau untuk orang
dewasa) dan di situlah semuanya: deretan calon \omterm{def:g2:from-seed-to-plant:seed}{biji} yang mungil dan
pucat.
\end{example}

\section{Dari bunga menjadi buah}

\begin{proposition}[Penyerbukan, lalu perubahan]\label{prop:g3:flowers-fruits-seeds:transform}
Agar perubahannya bermula, \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} harus mencapai \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya} ---
langkah itu disebut \emph{penyerbukan}\index{penyerbukan}. Lebah dan
\omterm{prop:g3:sorting-living-things:groups}{serangga} lain mengerjakan sebagian besar pengangkutannya, berdebu
\omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} ketika mereka minum dari \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} demi \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}; angin
mengerjakan sebagian pula. Sesudah penyerbukan:
\begin{itemize}
  \item \omterm{def:g3:flowers-fruits-seeds:parts}{mahkota bunga} dan \omterm{def:g3:flowers-fruits-seeds:parts}{benang sarinya}, yang tugasnya selesai, layu
        lalu gugur;
  \item \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya} membengkak lalu menjadi \emph{buah}\index{buah};
  \item calon \omterm{def:g2:from-seed-to-plant:seed}{biji} di dalamnya menjadi \emph{biji}\index{biji}.
\end{itemize}
Sebuah \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} adalah \omterm{def:g3:flowers-fruits-seeds:parts}{putik} yang berubah, dan setiap \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} sejati membawa
\omterm{def:g2:from-seed-to-plant:seed}{biji}.
\end{proposition}
\admitted

\begin{example}[Menyaksikan satu buah ceri terjadi]\label{ex:g3:flowers-fruits-seeds:cherry}
Ikuti sekuntum \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} ceri. April: mahkota putih, lebah berdengung.
Mei: mahkota berguguran seperti salju; di tempat tiap \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} berdiri,
sebuah bola hijau mungil --- \omterm{def:g3:flowers-fruits-seeds:parts}{putik} yang membengkak. Juni: bolanya
tumbuh merah dan manis --- sebuah \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} ceri, dengan \omterm{def:g2:from-seed-to-plant:seed}{biji} besarnya
(\omterm{def:g2:from-seed-to-plant:seed}{bijinya}) di dalam. \omterm{def:g3:flowers-fruits-seeds:parts}{Bunganya} tidak gugur; hanya mahkotanya.
\end{example}

\begin{remark}[Tanpa lebah, tanpa buah ceri]\label{rem:g3:flowers-fruits-seeds:bees}
Petani ceri menyambut sarang para peternak lebah masuk ke kebunnya ---
tanpa kunjungan \omterm{prop:g3:sorting-living-things:groups}{serangga}, kebanyakan \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} akan tetap tidak diserbuki
lalu gugur tanpa pernah menjadi \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}. Lebahnya tidak sedang mencuri
dari \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya}; \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} dan lebah saling memberi makan: nektar bagi
lebahnya, pengantaran \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} bagi \omterm{def:g1:plants-around-us:plant}{tumbuhannya}.
\end{remark}

\section{Buah atau bukan buah?}

\begin{method}[Uji biji]\label{met:g3:flowers-fruits-seeds:test}
Untuk memutuskan apakah sesuatu dari penjual sayur termasuk \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} dalam
pengertian ahli \omterm{def:g1:living-or-not:living}{biologi}:
\begin{enumerate}
  \item belah ia;
  \item carilah \omterm{def:g2:from-seed-to-plant:seed}{biji} (atau sebuah \omterm{def:g2:from-seed-to-plant:seed}{biji} besar, yaitu \omterm{def:g2:from-seed-to-plant:seed}{biji} di dalam
        \omterm{def:g1:animals-around-us:covering}{cangkang} yang keras);
  \item ada \omterm{def:g2:from-seed-to-plant:seed}{biji} di dalamnya --- ia tumbuh dari \omterm{def:g3:flowers-fruits-seeds:parts}{putik} sekuntum \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}:
        sebuah \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}. Tidak ada \omterm{def:g2:from-seed-to-plant:seed}{biji} di mana pun --- ia bagian lain
        \omterm{def:g1:plants-around-us:plant}{tumbuhannya}: \omterm{def:g1:plants-around-us:parts}{akar}, \omterm{def:g1:plants-around-us:parts}{batang} atau \omterm{def:g1:plants-around-us:parts}{daun}.
\end{enumerate}
\end{method}

\begin{example}[Kejutan di kios penjual sayur]\label{ex:g3:flowers-fruits-seeds:surprise}
Uji \omterm{def:g2:from-seed-to-plant:seed}{biji} melahirkan kejutan. Tomat: penuh \omterm{def:g2:from-seed-to-plant:seed}{biji} --- sebuah \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}, apa
pun kata mangkuk saladnya. Mentimun dan zukini: berbiji --- \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}.
Polong buncis hijau: \omterm{def:g2:from-seed-to-plant:seed}{biji} berderet --- sebuah \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}. Wortel: tanpa \omterm{def:g2:from-seed-to-plant:seed}{biji}
--- sebuah \omterm{def:g1:plants-around-us:parts}{akar} (\cref{exo:g1:plants-around-us:10} sudah menerkanya).
Selada: tanpa \omterm{def:g2:from-seed-to-plant:seed}{biji} --- \omterm{def:g1:plants-around-us:parts}{daun}. Kentang: tanpa \omterm{def:g2:from-seed-to-plant:seed}{biji} --- \omterm{def:g1:plants-around-us:parts}{batang} bawah
tanah yang membengkak.
\end{example}

\begin{remark}[Kata dapur, kata ilmu]\label{rem:g3:flowers-fruits-seeds:words}
Di dapur, ``\omterm{prop:g3:flowers-fruits-seeds:transform}{buah}'' berarti manis dan ``sayur'' berarti gurih ---
berguna untuk merencanakan hidangan penutup, tidak berguna bagi
\omterm{def:g1:living-or-not:living}{biologi}. Kata ahli \omterm{def:g1:living-or-not:living}{biologi} mengikuti uji \omterm{def:g2:from-seed-to-plant:seed}{biji}. Kedua perbendaharaan
kata itu baik-baik saja, asalkan kamu tahu yang mana yang sedang kamu
pakai.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g3:flowers-fruits-seeds:1}
Sebutkan bagian-bagian sekuntum \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} dari luar ke dalam, dengan tugas
masing-masing bagian.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:flowers-fruits-seeds:2}
Apa itu \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari}, dan di mana ia ditemukan? Apa itu penyerbukan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:flowers-fruits-seeds:3}
Sesudah penyerbukan, apa yang terjadi pada: (a) \omterm{def:g3:flowers-fruits-seeds:parts}{mahkota bunganya}? (b)
\omterm{def:g3:flowers-fruits-seeds:parts}{putiknya}? (c) calon \omterm{def:g2:from-seed-to-plant:seed}{biji} di dalamnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:flowers-fruits-seeds:4}
Ceritakan kembali kisah satu \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} ceri, dari \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} putih sampai \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}
merah, dalam tiga langkah bertanggal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:flowers-fruits-seeds:5}
Mengapa petani ceri menyambut sarang lebah di kebunnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:flowers-fruits-seeds:6}
Nyatakan uji \omterm{def:g2:from-seed-to-plant:seed}{bijinya}, lalu terapkan pada tomat dan wortel.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:flowers-fruits-seeds:7}
\omterm{prop:g3:flowers-fruits-seeds:transform}{Buah} atau bukan, menurut uji \omterm{def:g2:from-seed-to-plant:seed}{biji}: mentimun, selada, polong buncis
hijau, kentang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:flowers-fruits-seeds:8}
Cobalah: di rumah, belahlah tiga benda dari dapur lalu jalankan uji
\omterm{def:g2:from-seed-to-plant:seed}{biji} atas masing-masing. Mana yang mengejutkanmu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:flowers-fruits-seeds:9}
Embun beku yang datang terlambat mematikan seluruh \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} pohon prem
pada bulan April. Bagaimana panen bulan Juninya, dan mengapa? Pakai
\cref{prop:g3:flowers-fruits-seeds:transform}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:flowers-fruits-seeds:10}
``Apel tumbuh supaya kita punya sesuatu yang enak untuk dimakan.''
Sebenarnya untuk apa apelnya, dari sudut pandang pohon apelnya? Apa
yang terletak di intinya, dan masing-masing dapat menjadi apa?
(Ingatlah \cref{ch:g2:from-seed-to-plant}.)
\end{exercise}
