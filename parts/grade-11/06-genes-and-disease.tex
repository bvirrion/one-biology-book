\chapter{Genetic Variation and Disease}\label{ch:g11:genes-and-disease}

A couple in perfect health have a first child who, from the first
weeks, coughs, fails to gain weight, and has sweat so salty that a
kiss on the forehead tastes of it. The diagnosis is cystic fibrosis:
both parents, without knowing it, carried one non-working copy of the
same \omterm{def:g10:universal-dna:gene}{gene}, and the child received both. One in twenty-five people of
European ancestry is such a carrier; the parents could have been
told, before the pregnancy, by a test on a drop of blood. This chapter
is about diseases written in the \omterm{def:g10:universal-dna:gene}{alleles}: how they are inherited, how
the risk is computed from a family tree, how a laboratory reads the
\omterm{def:g10:universal-dna:gene}{gene}, and why most common diseases are only partly genetic.

\section{Diseases of a single gene}

\begin{definition}[Genetic disease, dominant and recessive]\label{def:g11:genes-and-disease:genetic}
A \emph{genetic disease}\index{genetic disease} is a disease caused by
one or more mutant \omterm{def:g10:universal-dna:gene}{alleles}. For a disease of a single \omterm{def:g10:universal-dna:gene}{gene}, an \omterm{def:g10:universal-dna:gene}{allele}
is \emph{recessive}\index{recessive} if the disease appears only in
people carrying two copies of it --- a person with one copy is a
healthy \emph{carrier}\index{carrier} --- and \emph{dominant}\index{dominant}
if one copy is enough. By convention the \omterm{def:g10:universal-dna:gene}{alleles} of a \omterm{def:g10:universal-dna:gene}{gene} are written
with a letter: $A$ for the ordinary \omterm{def:g10:universal-dna:gene}{allele}, $a$ for a recessive mutant
one, so that a carrier is $Aa$ and an affected person $aa$.
\end{definition}

\begin{example}[Cystic fibrosis]\label{ex:g11:genes-and-disease:cf}
The CFTR \omterm{def:g10:universal-dna:gene}{gene} encodes a \omterm{def:g11:gene-expression:protein}{protein} that lets chloride ions cross the
membrane of the \omterm{def:g10:cells-common-unit:cell}{cells} lining the airways, the gut and the sweat
glands. The commonest mutant \omterm{def:g10:universal-dna:gene}{allele} lacks three \omterm{def:g10:universal-dna:nucleotide}{nucleotides}, hence one
\omterm{def:g10:chemistry-of-life:families}{amino acid}; the \omterm{def:g11:gene-expression:protein}{protein} misfolds and is destroyed. Without it the
mucus of the airways is thick and sticky, infections settle in the
lungs, and the pancreas's ducts clog. The disease is \omterm{def:g11:genes-and-disease:genetic}{recessive}: about
one European in 25 is a healthy carrier, and one newborn in 2500 is
affected. \omterm{def:g10:cells-common-unit:cell}{Sickle-cell} disease (\cref{ch:g11:enzymes-and-phenotype})
is likewise \omterm{def:g11:genes-and-disease:genetic}{recessive}; Huntington's disease, a degeneration of the
brain beginning around forty, is \omterm{def:g11:genes-and-disease:genetic}{dominant} --- one mutant \omterm{def:g10:universal-dna:gene}{allele}
suffices, and every affected person had an affected parent.
\end{example}

\begin{proposition}[Transmission of a recessive disease]\label{prop:g11:genes-and-disease:recessive}
Each parent passes one of its two \omterm{def:g10:universal-dna:gene}{alleles}, chosen at random, to each
child. Two carriers $Aa \times Aa$ therefore have, for each child, a
chance of $1/4$ of an affected child $aa$, $1/2$ of a carrier $Aa$,
and $1/4$ of a non-carrier $AA$. A carrier and a non-carrier
$Aa \times AA$ have no affected child and one chance in two of a
carrier. An affected person and a non-carrier $aa \times AA$ have only
carrier children.
\end{proposition}
\admitted

\section{Reading a family tree}

\begin{definition}[Pedigree]\label{def:g11:genes-and-disease:pedigree}
A \emph{pedigree}\index{pedigree} is a diagram of a family over
several generations: squares for males, circles for females, a
horizontal line joining a couple, a vertical line down to their
children, filled symbols for affected people. Generations are numbered
I, II, III from the top; individuals are numbered from the left.
\end{definition}

\begin{method}[Analysing a pedigree]\label{met:g11:genes-and-disease:analyse}
\begin{enumerate}
  \item \emph{\omterm{def:g11:genes-and-disease:genetic}{Dominant} or \omterm{def:g11:genes-and-disease:genetic}{recessive}?} An affected child of two healthy
        parents means \omterm{def:g11:genes-and-disease:genetic}{recessive} (the parents are carriers). An
        affected person with an affected parent in every generation
        suggests \omterm{def:g11:genes-and-disease:genetic}{dominant}.
  \item \emph{On a sex \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}?} If only males are affected and
        their mothers' brothers or fathers are too, suspect the X
        \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} (below). Otherwise the \omterm{def:g10:universal-dna:gene}{gene} is on an ordinary
        \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}.
  \item \emph{Assign \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypes}}: affected people in a \omterm{def:g11:genes-and-disease:genetic}{recessive}
        disease are $aa$; their parents and children carry at least
        one $a$; healthy siblings of an affected person are $AA$ or
        $Aa$.
  \item \emph{Compute the risk} for a planned child by combining the
        parents' \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypes} in a Punnett square; when a parent's
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} is uncertain, weight each possibility by its
        probability.
\end{enumerate}
\end{method}

\begin{example}[The two-thirds]\label{ex:g11:genes-and-disease:twothirds}
Why is a healthy brother of an affected child a carrier with
probability $2/3$ and not $1/2$? The parents are $Aa \times Aa$; a
child is $AA$, $Aa$, $aA$ or $aa$ with equal chances. Knowing he is
healthy removes $aa$: three cases remain, of which two are carriers.
If he marries a woman of the general population (carrier with
probability $1/25$), the risk of an affected first child is
$2/3 \times 1/25 \times 1/4 = 1/150$.
\end{example}

\section{Genes on the X chromosome}

\begin{proposition}[X-linked inheritance]\label{prop:g11:genes-and-disease:xlinked}
Females carry two X \omterm{def:g10:universal-dna:information}{chromosomes}, males one X and one Y. A \omterm{def:g11:genes-and-disease:genetic}{recessive}
\omterm{def:g10:universal-dna:gene}{allele} on the X \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} is masked in a female by her second X, but
a male, who has no second X, expresses it. Such a disease therefore
affects mostly males; an affected male's daughters are all carriers;
a carrier mother passes the \omterm{def:g10:universal-dna:gene}{allele} to half her sons, who are affected,
and half her daughters, who are carriers. Haemophilia, in which the
blood fails to clot, is the classic example; red-green colour
blindness (\cref{ch:g11:the-eye}) is another.
\end{proposition}
\admitted

\section{Reading the gene itself}

\begin{proposition}[Genetic testing]\label{prop:g11:genes-and-disease:testing}
The \omterm{def:g10:universal-dna:gene}{alleles} a person carries can be read directly from their \omterm{def:g10:universal-dna:information}{DNA},
extracted from a few \omterm{def:g10:cells-common-unit:cell}{cells} (blood, saliva, and for a foetus a sample
of the placenta or of the fluid around it). The region of the \omterm{def:g10:universal-dna:gene}{gene} is
copied millions of times by a reaction that mimics replication
(\cref{ch:g11:dna-replication}) in a test tube, then either sequenced
or cut by \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymes} that recognise specific sequences and sorted by
size in a gel: a mutant \omterm{def:g10:universal-dna:gene}{allele} that has gained, lost or changed a cut
site gives fragments of different lengths, visible as bands. Such
tests identify carriers before any child is conceived, diagnose a
foetus, and detect the \omterm{def:g11:mutations:mutation}{mutation} in a newborn before symptoms appear.
\end{proposition}
\admitted

\begin{example}[What a test can and cannot say]\label{ex:g11:genes-and-disease:testsays}
A carrier test for cystic fibrosis looks for the fifty commonest
mutant \omterm{def:g10:universal-dna:gene}{alleles} and finds about 90\% of carriers: a negative result
lowers a person's chance of being a carrier from $1/25$ to about
$1/250$, but does not make it zero. A prenatal diagnosis on a foetus
of two known carriers is close to certain, and confronts the parents
with a decision that the biology does not make for them. A test for
Huntington's disease tells a healthy thirty-year-old whether the
disease will come; a third of those entitled to it choose not to know.
\end{example}

\section{Diseases of many genes and of a way of life}

\begin{proposition}[Multifactorial diseases]\label{prop:g11:genes-and-disease:multifactorial}
Most common diseases --- type 2 diabetes, heart and artery disease,
most cancers, asthma --- are \emph{multifactorial}\index{multifactorial disease}:
dozens or hundreds of \omterm{def:g10:universal-dna:gene}{alleles} each shift the risk a little, and the
environment (diet, activity, smoking, infections) shifts it a lot.
Such a disease runs in families without following the rules of a
single \omterm{def:g10:universal-dna:gene}{gene}, and a genetic predisposition is a risk, not a sentence:
the same \omterm{def:g10:universal-dna:gene}{alleles} produce disease in one way of life and not in
another.
\end{proposition}
\begin{proof}[Evidence]
Identical twins share all their \omterm{def:g10:universal-dna:gene}{alleles}; when one has type 2
diabetes, the other has it in about 70\% of cases --- far above the
population's 8\%, so \omterm{def:g10:universal-dna:gene}{genes} matter --- but not 100\%, so \omterm{def:g10:universal-dna:gene}{genes} do not
decide. Populations that moved in a generation from a rural to an
urban diet saw the disease rise several-fold with the same \omterm{def:g10:universal-dna:gene}{genes}.
Among people carrying many risk \omterm{def:g10:universal-dna:gene}{alleles}, those who keep active and
lean develop the disease at a fraction of the rate of those who do
not.
\end{proof}

\begin{remark}[What "genetic" means, then]\label{rem:g11:genes-and-disease:meaning}
Cystic fibrosis is genetic in the strict sense: the \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype}
determines the disease, whatever the environment, and the environment
only changes how it is lived. Type 2 diabetes is genetic in a weaker
sense: the \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} sets a susceptibility that the environment
realises or not. Between the two lie phenylketonuria, where a diet
cancels the \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype}'s effect, and the \omterm{def:g10:cells-common-unit:cell}{sickle-cell} carrier, whose
\omterm{def:g10:universal-dna:gene}{allele} is a disease in one place and a protection in another. "Is it
genetic?" is rarely a yes-or-no question; "how much, and under what
conditions?" is the one to ask.
\end{remark}

\section{Problem: One Family, One Gene}

\begin{problem}[{Weekend problem --- a family with cystic fibrosis
followed over three generations: genotypes assigned, risks computed
for each planned child, a laboratory test read, and the arithmetic of
a carrier population}]
\label{pb:g11:genes-and-disease:1}
Léa and Tom, both healthy, have three children: Max (healthy), Zoé
(cystic fibrosis) and Lou (healthy). Tom's sister Ana is healthy and
married to Karim, whose family has no history of the disease; they are
expecting a child. In the population, one person in 25 is a carrier.

\medskip
\noindent\textbf{Part I --- \omterm{def:g11:enzymes-and-phenotype:phenotype}{Genotypes}.}
\begin{enumerate}
  \item Is the disease \omterm{def:g11:genes-and-disease:genetic}{dominant} or \omterm{def:g11:genes-and-disease:genetic}{recessive}? Justify from the family.
  \item Give the \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypes} of Léa, Tom and Zoé.
  \item What are the possible \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypes} of Max, with what
        probabilities?
  \item What is the probability that Ana is a carrier? (Consider her
        parents.)
  \item What is the probability that Karim is a carrier?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Risks.}
\begin{enumerate}[resume]
  \item Compute the probability that Ana and Karim's child is
        affected.
  \item Léa and Tom consider a fourth child. Probability that it is
        affected? That it is a carrier?
  \item Max, adult, marries a woman from the general population.
        Probability of an affected first child?
  \item Zoé, adult, marries a man from the general population.
        Probability of an affected child? Of a carrier child?
  \item Lou marries a first cousin, a child of Ana and Karim.
        Compute the probability of an affected child. Why is it higher
        than for Max?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The test.}
The mutant \omterm{def:g10:universal-dna:gene}{allele} carried in this family adds a cut site: the ordinary
\omterm{def:g10:universal-dna:gene}{allele} gives one fragment of 900 base pairs, the mutant one gives 600
and 300.
\begin{enumerate}[resume]
  \item Draw, or describe, the bands expected for Zoé, for Tom, and
        for a non-carrier.
  \item Max is tested: one band at 900. What is his \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} now, and
        the new risk for his first child (question 8)?
  \item Ana is tested: bands at 900, 600 and 300. Recompute the risk
        for her child.
  \item Karim's test finds none of the fifty commonest mutant \omterm{def:g10:universal-dna:gene}{alleles},
        which cover 90\% of carriers. Compute his remaining
        probability of being a carrier and recompute the risk.
  \item A prenatal test of Ana and Karim's foetus shows bands at 900,
        600 and 300. What is the foetus's \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} and \omterm{def:g11:enzymes-and-phenotype:phenotype}{phenotype}?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The population.}
\begin{enumerate}[resume]
  \item If one person in 25 is a carrier, what is the probability
        that both members of a random couple are carriers, and hence
        the frequency of affected newborns?
  \item Compare with the observed one in 2500. Comment.
  \item Almost all affected people used to die in childhood. Explain
        why the mutant \omterm{def:g10:universal-dna:gene}{allele} nevertheless stayed at a frequency of
        one in 50 copies, using the number of copies carried by
        healthy carriers versus by affected people.
  \item Treatments now let most affected people reach adulthood and
        have children. Predict the effect on the \omterm{def:g10:universal-dna:gene}{allele}'s frequency
        over the coming generations, and its size.
  \item State the result: the one-in-four of two carriers, the
        two-in-three of a healthy sibling, and what the gel added to
        the arithmetic for this family.
\end{enumerate}
\end{problem}
