\section*{Hoofdstuk \ref{ch:g12:meiosis-genetic-shuffling} --- Meiose en genetisch schudden}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:1}
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{Diploïde}: twee kopieën van elk \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom} ($2n$); \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{haploïde}: één ($n$). De
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} halveert het aantal \omterm{def:g10:universal-dna:information}{chromosomen}, van $2n$ naar $n$, en brengt het
\omterm{def:g10:universal-dna:information}{DNA} terug van $2q$ (na de verdubbeling) naar $q/2$ per gameet.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:2}
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{Meiose} I scheidt de homologe \omterm{def:g10:universal-dna:information}{chromosomen} van elk paar; \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} II scheidt
de twee \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} van elk \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}. De eerste is de reductiedeling.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:3}
Na \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} I: 4 \omterm{def:g10:universal-dna:information}{chromosomen} (elk twee \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden}), \omterm{def:g10:universal-dna:information}{DNA} $q$. Na \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} II:
4 \omterm{def:g10:universal-dna:information}{chromosomen} (elk één \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatide}), \omterm{def:g10:universal-dna:information}{DNA} $q/2$.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:4}
$2^4 = 16$.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:5}
Een kruising met een partner die alleen recessieve \omterm{def:g10:universal-dna:gene}{allelen} draagt, zodat
elke nakomeling de gameet van de geteste ouder laat zien. Het aandeel
recombinanten onthult of twee \omterm{def:g10:universal-dna:gene}{genen} op hetzelfde \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom} liggen en, zo
ja, hoe ver zij uit elkaar liggen.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:6}
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{Meiose} I: $2q$; tussen de delingen: $q$; gameet: $q/2$. Zonder S-fase
ertussen halveert de tweede deling het \omterm{def:g10:universal-dna:information}{DNA} opnieuw, en dat is wat de
gameet op de helft van de inhoud van een \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{diploïde} \omterm{def:g10:cells-common-unit:cell}{cel} brengt.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:7}
Vier even grote klassen: niet gekoppeld, op verschillende \omterm{def:g10:universal-dna:information}{chromosomen}.
Gameten $AB$, $Ab$, $aB$, $ab$, elk een kwart.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:8}
Gekoppeld: twee grote klassen ($AB$, $ab$, ouderlijk) en twee kleine
($Ab$, $aB$, recombinant). Recombinanten $85/1000 = 8.5\%$.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:9}
Alleen $AB$ en $ab$: een uitwisseling boven beide \omterm{def:g10:universal-dna:gene}{genen} verwisselt stukken
die geen van beide dragen, dus blijven de combinaties van $A/a$ en $B/b$
onveranderd. Recombinatie tussen twee \omterm{def:g10:universal-dna:gene}{genen} vergt een uitwisseling
tussen hen.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:10}
Ongeveer 0.8, 2.9 en 16 per 1000: een factor 20 tussen 25 en 42.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:11}
In \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} I gaan beide homologen naar dezelfde \omterm{def:g10:cells-common-unit:cell}{cel}: twee gameten met twee
kopieën en twee zonder. In \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} II gaan beide \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} van één
\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom} naar dezelfde \omterm{def:g10:cells-common-unit:cell}{cel}: één gameet met twee kopieën, één zonder, en
twee normale.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:12}
De recombinatie is evenredig met de afstand: 2\% betekent dat de \omterm{def:g10:universal-dna:gene}{genen}
heel dicht bij elkaar liggen, zodat er zelden een uitwisseling tussen hen
valt; 48\% betekent ver uit elkaar, met bij vrijwel elke \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} een
uitwisseling ertussen. Omdat één uitwisseling 50\% recombinante gameten
geeft, recombineren ver uiteen liggende gekoppelde \omterm{def:g10:universal-dna:gene}{genen} even vrij als
onafhankelijke.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:13}
$AB$, $Ab$, $aB$, $ab$, elk een kwart, door de onafhankelijke verdeling
van de twee paren. \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{Crossing-over} verandert voor deze \omterm{def:g10:universal-dna:gene}{genen} niets: zij
recombineert \omterm{def:g10:universal-dna:gene}{allelen} langs één \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}, en deze liggen op
verschillende \omterm{def:g10:universal-dna:information}{chromosomen}.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:14}
Elke ouder geeft elk kind bij elk \omterm{def:g10:universal-dna:gene}{gen} een van twee \omterm{def:g10:universal-dna:gene}{allelen} door,
willekeurig gekozen; bij een gegeven \omterm{def:g10:universal-dna:gene}{gen} kregen twee broers of zussen met
kans $1/2$ hetzelfde \omterm{def:g10:universal-dna:gene}{allel} van een ouder. Over vele \omterm{def:g10:universal-dna:gene}{genen} komt het
gedeelde aandeel gemiddeld op $1/2$ uit, maar voor één bepaald paar
broers of zussen schommelt het daaromheen.
\end{solution}

\begin{solution}{exo:g12:meiosis-genetic-shuffling:15}
Klonen van de moederplant, genetisch gelijk aan haar en aan elkaar. De
plant wint een snelle, zekere vermenigvuldiging van een \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} dat het
hier en nu doet; zij verliest de verscheidenheid die enkele nakomelingen
een verandering van omgeving of een nieuwe parasiet zou laten overleven.
\end{solution}

\begin{solution}{pb:g12:meiosis-genetic-shuffling:1}
\textbf{1.} Vrouwtje $b^+b$, $vg^+vg$, met $b^+vg^+$ op het ene \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}
en $b\,vg$ op het andere (van haar twee zuivere ouders). Mannetje $bb$,
$vg\,vg$.

\textbf{2.} Alleen gameten $b\,vg$: het mannetje draagt uitsluitend
recessieve \omterm{def:g10:universal-dna:gene}{allelen} bij, zodat het \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} van elke nakomeling de gameet
van het vrouwtje verraadt.

\textbf{3.} $b^+vg^+$ (grijs-normaal), $b\,vg$ (zwart-rudimentair),
$b^+vg$ (grijs-rudimentair), $b\,vg^+$ (zwart-normaal).

\textbf{4.} Grijs-normaal 42\%, zwart-rudimentair 41\%, grijs-rudimentair
9\%, zwart-normaal 8\%.

\textbf{5.} Op hetzelfde \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}: twee grote en twee kleine klassen in
plaats van $1:1:1:1$.

\textbf{6.} Grijs-normaal en zwart-rudimentair: de combinaties die haar
twee homologen dragen, ongeschonden van haar zuivere ouders geërfd.

\textbf{7.} Grijs-rudimentair en zwart-normaal, door een \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{crossing-over}
tussen de twee \omterm{def:g10:universal-dna:gene}{genen} in de profase I.

\textbf{8.} $(206 + 185)/2300 \approx 17\%$. Een \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{crossing-over} tussen de
\omterm{def:g10:universal-dna:gene}{genen} maakt twee van de vier \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} recombinant, dus betekent 17\%
recombinante gameten een uitwisseling in ongeveer 34\% van de \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosen}.

\textbf{9.} Eén uitwisseling brengt de twee recombinante \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden}
samen voort, één van elk type; en de twee homologen gaan gelijk over de
gameten, zodat ook de twee ouderlijke klassen aan elkaar gelijk zijn.

\textbf{10.} De ouderlijke klassen zijn nu grijs-rudimentair en
zwart-normaal, elk ongeveer 41.5\%; de recombinanten grijs-normaal en
zwart-rudimentair, elk ongeveer 8.5\%.

\textbf{11.} Ja: bij allebei ligt de recombinatie ruim onder 50\%.

\textbf{12.} $b$ --- $cn$ --- $vg$: 9 en 8 tellen op tot 17.

\textbf{13.} De kans op een uitwisseling in een interval is evenredig met
de lengte ervan, dus tellen de frequenties van aangrenzende intervallen
op: de recombinatiefrequentie meet de afstand langs het \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}.

\textbf{14.} Eén uitwisseling geeft twee recombinante \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} op vier:
hoogstens 50\%; verdere uitwisselingen schudden opnieuw maar brengen het
recombinante aandeel nooit boven de helft.

\textbf{15.} Het ligt op een ander \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom} of heel ver weg op hetzelfde;
de kruisingen kunnen niet zeggen welk van beide.

\textbf{16.} $2^4 = 16$ gameten; $16 \times 16 = 256$ combinaties.

\textbf{17.} Elke uitwisseling brengt \omterm{def:g10:universal-dna:information}{chromosomen} voort die nooit eerder
hebben bestaan, op een plaats die van \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} tot \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiose} verschilt; het
aantal verschillende \omterm{def:g10:universal-dna:information}{chromosomen}, en dus gameten, kent in de praktijk geen
grens.

\textbf{18.} $2^{23} \times 2^{23} \approx \num{7e13}$: zevenhonderd keer
meer combinaties dan het aantal mensen dat ooit heeft geleefd, en dat nog
vóór de \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{crossing-over}.

\textbf{19.} Broers en zussen trekken hun \omterm{def:g10:universal-dna:gene}{allelen} uit dezelfde vier sets
(twee per ouder) en delen er gemiddeld de helft van; vreemden trekken uit
verschillende sets.

\textbf{20.} 17\%, het deel van de gameten van het vrouwtje dat tussen $b$
en $vg$ is gerecombineerd, meet de afstand tussen de \omterm{def:g10:universal-dna:gene}{genen}; de
onafhankelijke verdeling van de paren en de \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{crossing-over} binnen elk paar
maken elke gameet anders.
\end{solution}
