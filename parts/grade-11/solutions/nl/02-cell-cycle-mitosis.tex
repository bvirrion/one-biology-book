\section*{Hoofdstuk \ref{ch:g11:cell-cycle-mitosis} --- De celcyclus en de mitose}

\begin{solution}{exo:g11:cell-cycle-mitosis:1}
$\mathrm{G_1}$: groei; S: \omterm{def:g11:dna-replication:replication}{DNA-replicatie}; $\mathrm{G_2}$: voorbereiding
op de deling; M: \omterm{def:g11:cell-cycle-mitosis:cycle}{mitose} (deling van de kern) en cytokinese (deling van
het \omterm{def:g10:cells-common-unit:cell}{cytoplasma}).
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:2}
Een \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatide} is een van de twee gelijke kopieën van een verdubbeld
\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}; het \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromeer} is het gebied waar de twee bijeen worden
gehouden. Een \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom} heeft twee \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} vanaf het eind van S tot
aan de anafase.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:3}
Profase: de \omterm{def:g10:universal-dna:information}{chromosomen} verdichten zich, het kernmembraan valt uiteen.
Metafase: de \omterm{def:g10:universal-dna:information}{chromosomen} gaan op de evenaar in het gelid. Anafase: de
\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} gaan uiteen en bewegen naar de polen. Telofase: de
kernmembranen vormen zich opnieuw, gevolgd door de cytokinese.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:4}
In de S-fase, halverwege de replicatie.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:5}
46 \omterm{def:g10:universal-dna:information}{chromosomen}, 92 \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden}.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:6}
$\mathrm{G_1}$ \qty{11}{h}, S \qty{8}{h}, $\mathrm{G_2}$ \qty{4}{h},
M \qty{1}{h}. Een \omterm{def:g10:cells-common-unit:cell}{cel} met $2q$ is in $\mathrm{G_2}$ of in de \omterm{def:g11:cell-cycle-mitosis:cycle}{mitose} vóór
het eind van de anafase.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:7}
Metafase: 40 \omterm{def:g10:universal-dna:information}{chromosomen}, 80 \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden}. Anafase: 80 \omterm{def:g10:universal-dna:information}{chromosomen} (met
één \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatide}), 40 naar elke pool. Dochter: 40 \omterm{def:g10:universal-dna:information}{chromosomen}, 40
\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden}.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:8}
Mitose-index $80/800 = 10\%$; \omterm{def:g11:cell-cycle-mitosis:cycle}{mitose} \qty{2.4}{h}. Profase
$48/800 \times 24 = \qty{1.44}{h}$; metafase \qty{0.48}{h} (29 min);
anafase \qty{0.18}{h} (11 min); telofase \qty{0.3}{h} (18 min).
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:9}
Zonder spoel kunnen de \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} niet uit elkaar worden getrokken: de
\omterm{def:g10:cells-common-unit:cell}{cel} blijft in de metafase steken met haar \omterm{def:g10:universal-dna:information}{chromosomen} verdicht en goed
zichtbaar, elk met twee \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden}. Dat is precies de toestand waarin
\omterm{def:g10:universal-dna:information}{chromosomen} voor een \omterm{def:g11:cell-cycle-mitosis:chromosome}{karyotype} worden gefotografeerd.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:10}
Twee meter verwarde draad valt niet te sorteren; verdicht tot staafjes
van enkele micrometers kunnen de 46 \omterm{def:g10:universal-dna:information}{chromosomen} in het gelid worden gezet
en uit elkaar worden getrokken zonder te breken of in de knoop te raken.
Zij rollen zich daarna weer af omdat \omterm{def:g10:universal-dna:gene}{genen} alleen in de losse vorm
gelezen kunnen worden.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:11}
Tien delingen: $2^{10} = 1024$ \omterm{def:g10:cells-common-unit:cell}{cellen}. Nee: het jonge embryo groeit
tussen twee delingen niet, en zijn cycli bestaan vrijwel alleen uit S en
M, met nauwelijks $\mathrm{G_1}$.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:12}
De ene dochter heeft 47 \omterm{def:g10:universal-dna:information}{chromosomen} (één te veel), de andere 45 (één te
weinig); hun \omterm{def:g10:universal-dna:information}{DNA} is $q$ plus of min de inhoud van één \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom}. In een
lichaamscel treft de fout één \omterm{def:g10:cells-common-unit:cell}{cel} op biljoenen, die meestal sterft of
wordt opgeruimd; maar als de \omterm{def:g10:cells-common-unit:cell}{cel} overleeft en zich deelt, en het verloren
of gewonnen \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosoom} haar groei ontregelt, kan er een tumor beginnen.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:13}
Ongeveer $\num{2e11}$ delingen per dag, dus $\num{2e11}$ \omterm{def:g10:cells-common-unit:cell}{cellen} die op
enig moment in de cyclus zitten (elk levert één nieuwe \omterm{def:g10:cells-common-unit:cell}{cel} per dag). De
\omterm{def:g11:cell-cycle-mitosis:cycle}{mitose} blokkeren legt de aanvoer stil; rode \omterm{def:g10:cells-common-unit:cell}{cellen} leven 120 dagen maar
sommige witte slechts dagen, dus stort binnen een week het aantal witte
\omterm{def:g10:cells-common-unit:cell}{cellen} in en volgen infecties --- de klassieke bijwerking van middelen
tegen kanker.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:14}
In de interfase zijn de \omterm{def:g10:universal-dna:information}{chromosomen} afgerold en onzichtbaar; in de
anafase bewegen zij en lopen zij door elkaar. In de metafase zijn zij
volledig verdicht, gescheiden, nog uit twee \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} opgebouwd en in
één vlak gelegen: elk is aan zijn grootte en vorm te herkennen.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:15}
Levercellen die in $\mathrm{G_1}$ rustten, gingen de cyclus weer in,
doorliepen S, $\mathrm{G_2}$ en de \omterm{def:g11:cell-cycle-mitosis:cycle}{mitose} herhaaldelijk tot de massa was
hersteld, en keerden daarna terug naar de rust. Op dag 0 bevatten vrijwel
alle \omterm{def:g10:cells-common-unit:cell}{cellen} $q$; op dag 3 bevatten er veel iets tussen $q$ en $2q$ (in S)
of $2q$ (in $\mathrm{G_2}$ of in de \omterm{def:g11:cell-cycle-mitosis:cycle}{mitose}).
\end{solution}

\begin{solution}{pb:g11:cell-cycle-mitosis:1}
\textbf{1.} $100/1000 = 10\%$.

\textbf{2.} $0.10 \times 20 = \qty{2}{h}$.

\textbf{3.} Profase $60/1000 \times 20 = \qty{1.2}{h}$ (72 min);
metafase $0.36$ h (22 min); anafase $0.16$ h (ongeveer 10 min);
telofase $0.28$ h (17 min).

\textbf{4.} De anafase. De \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} zijn al verdicht en gehecht; de
beweging is een trek van enkele micrometers door vezels die in enkele
minuten korter worden.

\textbf{5.} Zeldzame stadia worden in weinig \omterm{def:g10:cells-common-unit:cell}{cellen} gezien, en een kleine
telling geeft een grote fout; en alleen de groeizone doorloopt de cyclus
--- elders zou de index nul zijn en de cijfers verdunnen.

\textbf{6.} 16 \omterm{def:g10:universal-dna:information}{chromosomen}, 32 \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden}.

\textbf{7.} 32 bewegende \omterm{def:g10:universal-dna:information}{chromosomen}, 16 naar elke pool.

\textbf{8.} Profase $2q$; anafase $2q$ in de hele \omterm{def:g10:cells-common-unit:cell}{cel}; telofase $q$ per
kern.

\textbf{9.} S is $7/19$ van de interfase: ongeveer $900 \times 7/19 \approx
330$ \omterm{def:g10:cells-common-unit:cell}{cellen}.

\textbf{10.} De anafase: de \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromeren} zijn zojuist gesplitst en de 32
\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatiden} van de 16 \omterm{def:g10:universal-dna:information}{chromosomen} zijn nu 32 \omterm{def:g10:universal-dna:information}{chromosomen} op weg naar de
polen.

\textbf{11.} $\num{20000}/20 = 1000$ nieuwe \omterm{def:g10:cells-common-unit:cell}{cellen} per uur.

\textbf{12.} 1000 \omterm{def:g10:cells-common-unit:cell}{cellen} per uur over de hele zone; in één rij is de zone
in elke richting misschien honderd \omterm{def:g10:cells-common-unit:cell}{cellen} breed, dus ruwweg
$\num{24000}/(100 \times 100) \approx 2$--3 \omterm{def:g10:cells-common-unit:cell}{cellen} per rij per dag, elk
\qty{100}{\micro m}: enkele tienden van een millimeter. De waargenomen
\qty{1}{mm} betekent dat de zone smaller is dan aangenomen of dat de
\omterm{def:g10:cells-common-unit:cell}{cellen} zich verder uitrekken; de orde van grootte klopt.

\textbf{13.} Eén startpunt kopieert $2 \times 50 \times 7 \times 3600 =
\num{2.5e6}$ basenparen; $\num{1.6e10}/\num{2.5e6} \approx 6400$
startpunten op zijn minst.

\textbf{14.} Elk stadium duurt langer, dus voltooien er per dag minder
\omterm{def:g10:cells-common-unit:cell}{cellen} de cyclus; vertraagt de kou alle fasen even sterk, dan veranderen
de verhoudingen, en dus de mitose-index, niet.

\textbf{15.} \omterm{def:g10:cells-common-unit:cell}{Cellen} die S al voorbij zijn, voltooien in de eerste uren
$\mathrm{G_2}$ en de \omterm{def:g11:cell-cycle-mitosis:cycle}{mitose}; na een dag kan geen \omterm{def:g10:cells-common-unit:cell}{cel} de \omterm{def:g11:cell-cycle-mitosis:cycle}{mitose} meer
bereiken, dus zijn profase tot en met telofase verdwenen en blijven
alleen \omterm{def:g10:cells-common-unit:cell}{cellen} in interfase over, vastgelopen in $\mathrm{G_1}$ of aan het
begin van S.

\textbf{16.} $4/1000 \times 20 = \qty{0.08}{h}$, ongeveer 5 minuten,
tegen 10. Niet verrassend: bij 4 of 8 \omterm{def:g10:cells-common-unit:cell}{cellen} wordt de telling door het
toeval beheerst.

\textbf{17.} $12/2000 \times 20 = \qty{0.12}{h}$, ongeveer 7 minuten.

\textbf{18.} Mens: $0.04 \times 24 \approx \qty{1}{h}$; knoflook
\qty{2}{h}. Dezelfde orde, plantencellen wat trager.

\textbf{19.} Tumorcellen doorlopen de cyclus onafgebroken terwijl de
meeste normale \omterm{def:g10:cells-common-unit:cell}{cellen} in $\mathrm{G_1}$ rusten, dus is er op elk moment
een groter deel in \omterm{def:g11:cell-cycle-mitosis:cycle}{mitose}. Een middel dat de spoel blokkeert, treft
tumorcellen dus vaker dan normale --- maar ook de normale weefsels die
zich snel delen.

\textbf{20.} Cyclus 20 uur: $\mathrm{G_1}$ 8, S 7, $\mathrm{G_2}$ 4,
\omterm{def:g11:cell-cycle-mitosis:cycle}{mitose} ongeveer 2 (profase 72 min, metafase 22, anafase 7--10, telofase
17). De anafase, de kortste, is het moment waarop de \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromeren}
splitsen en elke pool van elk van de zestien \omterm{def:g10:universal-dna:information}{chromosomen} één \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatide}
ontvangt.
\end{solution}
