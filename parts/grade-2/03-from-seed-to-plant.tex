\chapter{From Seed to Plant}\label{ch:g2:from-seed-to-plant}

Shake a dried bean pod and little hard beans rattle out --- dry,
dull, apparently as lifeless as gravel. Yet \omterm{def:g1:plants-around-us:plant}{plant} one in the earth
and water it, and within days something extraordinary happens
underground. A whole \omterm{def:g1:plants-around-us:plant}{plant} was folded up inside, waiting for its
signal.

\section{What a seed is}

\begin{definition}[Seed]\label{def:g2:from-seed-to-plant:seed}
A \emph{seed}\index{seed} is the beginning of a new \omterm{def:g1:plants-around-us:plant}{plant}, made by
the parent \omterm{def:g1:plants-around-us:plant}{plant}. Inside its protective coat, a seed holds:
\begin{itemize}
  \item a tiny \omterm{def:g1:plants-around-us:plant}{plant} already started --- the
        \emph{germ}\index{germ (of a seed)} of the \omterm{def:g1:plants-around-us:plant}{plant}, with the
        beginnings of a \omterm{def:g1:plants-around-us:parts}{root} and a shoot;
  \item a store of food to feed it until it can feed itself.
\end{itemize}
\end{definition}

\begin{example}[Opening a soaked bean]\label{ex:g2:from-seed-to-plant:open}
Soak a big white bean overnight, and its coat slips off easily.
The bean opens into two thick halves --- the food store --- and
there, nestled at one edge, lies a tiny pale \omterm{def:g1:plants-around-us:plant}{plant}, \omterm{def:g1:plants-around-us:parts}{root} tip and
folded first \omterm{def:g1:plants-around-us:parts}{leaves}, small as a grain of rice. The new \omterm{def:g1:plants-around-us:plant}{plant} was
there all along.
\end{example}

\begin{remark}[Seeds travel through our kitchens]\label{rem:g2:from-seed-to-plant:kitchen}
Beans, peas, lentils, rice, sunflower \omterm{def:g2:from-seed-to-plant:seed}{seeds}, the pips in the apple
and the stone in the peach: the kitchen is full of \omterm{def:g2:from-seed-to-plant:seed}{seeds}. Some are
tiny, like poppy \omterm{def:g2:from-seed-to-plant:seed}{seeds}; the coconut is a giant one. Each of them,
given the chance, would try to become a \omterm{def:g1:plants-around-us:plant}{plant}.
\end{remark}

\section{Germination}

\begin{definition}[Germination]\label{def:g2:from-seed-to-plant:germination}
\emph{Germination}\index{germination} is the moment a \omterm{def:g2:from-seed-to-plant:seed}{seed} wakes
and starts to grow: it swells with water, its coat splits, the
little \omterm{def:g1:plants-around-us:parts}{root} comes out first and dives downward, then the shoot
rises, carrying the first green \omterm{def:g1:plants-around-us:parts}{leaves} toward the light. We say
the \omterm{def:g2:from-seed-to-plant:seed}{seed} \emph{germinates}\index{germinate}.
\end{definition}

\begin{proposition}[What a seed needs to germinate]\label{prop:g2:from-seed-to-plant:needs}
To \omterm{def:g2:from-seed-to-plant:germination}{germinate}, a \omterm{def:g2:from-seed-to-plant:seed}{seed} needs \emph{water} and \emph{warmth}. It does
not yet need light: the food store feeds the \omterm{prop:g2:from-seed-to-plant:cycle}{seedling} at first,
even under the soil, in the dark. Without water, or in the cold, a
good \omterm{def:g2:from-seed-to-plant:seed}{seed} simply waits.
\end{proposition}
\admitted

\begin{example}[Why sowing waits for spring]\label{ex:g2:from-seed-to-plant:spring}
Gardeners sow beans in spring, not in winter: the \omterm{def:g2:from-seed-to-plant:seed}{seed} in cold
earth would wait --- or rot --- instead of germinating. Warm days
and watered soil are exactly the two signals of
\cref{prop:g2:from-seed-to-plant:needs}.
\end{example}

\begin{method}[Sowing a seed and watching it germinate]\label{met:g2:from-seed-to-plant:sow}
\begin{enumerate}
  \item Line a glass jar with damp cotton or paper;
  \item slide a bean between glass and cotton, so you can see it;
  \item keep the jar in a warm room and the cotton damp;
  \item look every day and note what you see: swelling, the coat
        splitting, \omterm{def:g1:plants-around-us:parts}{root} down, shoot up.
\end{enumerate}
The glass makes visible what usually happens under the soil.
\end{method}

\begin{remark}[Root down, shoot up --- always]\label{rem:g2:from-seed-to-plant:updown}
However the bean lies in the jar --- on its side, upside down ---
the \omterm{def:g1:plants-around-us:parts}{root} always turns downward and the shoot always turns upward.
The \omterm{prop:g2:from-seed-to-plant:cycle}{seedling} cannot see, yet it knows which way is up. \omterm{def:g1:plants-around-us:plant}{Plants}
sense more than they show.
\end{remark}

\section{From seedling to plant with seeds}

\begin{proposition}[The plant's life cycle]\label{prop:g2:from-seed-to-plant:cycle}
The \emph{seedling}\index{seedling} grows into a \omterm{def:g1:plants-around-us:plant}{plant} with \omterm{def:g1:plants-around-us:parts}{roots},
\omterm{def:g1:plants-around-us:parts}{stem} and \omterm{def:g1:plants-around-us:parts}{leaves}; the \omterm{def:g1:plants-around-us:plant}{plant} \omterm{def:g1:plants-around-us:parts}{flowers}; the \omterm{def:g1:plants-around-us:parts}{flowers} make new
\emph{\omterm{def:g2:from-seed-to-plant:seed}{seeds}}; and the cycle can start again. A single bean \omterm{def:g1:plants-around-us:plant}{plant}
can make dozens of new beans in one summer.
\end{proposition}
\admitted

\begin{example}[One bean, one summer]\label{ex:g2:from-seed-to-plant:summer}
Sown in May, a bean \omterm{def:g2:from-seed-to-plant:germination}{germinates} in a week; by June it is a leafy
\omterm{def:g1:plants-around-us:plant}{plant} climbing its pole; in July come white \omterm{def:g1:plants-around-us:parts}{flowers}, then green
pods; in August the pods hang heavy, each holding a row of new
beans --- next year's \omterm{def:g2:from-seed-to-plant:seed}{seeds}, made by this year's \omterm{def:g1:plants-around-us:plant}{plant}.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:1}

What does \emph{\omterm{def:g2:from-seed-to-plant:germination}{germinate}} mean? What comes out of the \omterm{def:g2:from-seed-to-plant:seed}{seed}
first: the \omterm{def:g1:plants-around-us:parts}{root}, or the shoot?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:3}
Name the two things a \omterm{def:g2:from-seed-to-plant:seed}{seed} needs to \omterm{def:g2:from-seed-to-plant:germination}{germinate}. Which famous need
of grown \omterm{def:g1:plants-around-us:plant}{plants} is \emph{not} on the list?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:4}
How can a buried \omterm{prop:g2:from-seed-to-plant:cycle}{seedling} grow in the dark, before reaching the

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:5}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:6}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:7}

\omterm{prop:g2:from-seed-to-plant:cycle}{seedling} --- leafy \omterm{def:g1:plants-around-us:plant}{plant}.

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:8}
Try it: set up the glass jar of \cref{met:g2:from-seed-to-plant:sow}

\begin{exercise}[$\star\star$]\label{exo:g2:from-seed-to-plant:9}

\cref{prop:g2:from-seed-to-plant:needs}.

\begin{exercise}[$\star\star$]\label{exo:g2:from-seed-to-plant:10}

\end{exercise}
