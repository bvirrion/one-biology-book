\section*{Bab \ref{ch:g4:heart-and-blood} --- Jantung dan Darah}

\begin{solution}{exo:g4:heart-and-blood:1}
\omterm{def:g4:journey-of-food:digestion}{Zat gizi}, yang dipungut di \omterm{def:g4:journey-of-food:tract}{usus halus}; dan oksigen, yang dipungut di
\omterm{def:g4:breathing:organs}{paru-paru}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:2}
Karbon dioksida --- yang diturunkan di \omterm{def:g4:breathing:organs}{paru-paru}, tempat ia
diembuskan keluar.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:3}
Ia sebuah \omterm{def:g3:bones-and-muscles:muscle}{otot} berongga, kira-kira sebesar kepalan tangan pemiliknya,
yang duduk di tengah dada di balik sangkar rusuk.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:4}
Sisi kanannya memompa \omterm{prop:g4:journey-of-food:absorb}{darah} ke \omterm{def:g4:breathing:organs}{paru-paru} untuk memuat oksigen; sisi
kirinya memompa \omterm{prop:g4:journey-of-food:absorb}{darah} yang kaya oksigen keluar ke seluruh tubuh.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:5}
Gelombang kecil yang didorong tiap \omterm{def:g4:heart-and-blood:heart}{denyut jantung} menyusuri
\omterm{def:g4:heart-and-blood:vessels}{pembuluhnya}, yang terasa di \omterm{def:g1:my-body:joint}{pergelangan tangan} atau leher. Menghitungnya
berarti menghitung \omterm{def:g4:heart-and-blood:heart}{denyut jantung} per menit.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:6}
Ibu jari punya denyut kecilnya sendiri yang cukup terasa, sehingga
kamu bisa saja menghitung ibu jarimu alih-alih \omterm{def:g1:my-body:joint}{pergelangan tangan} yang
sedang kamu uji.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:7}
\omterm{def:g1:my-body:parts}{Tungkai} yang berlari cepat membakar \omterm{def:g4:journey-of-food:digestion}{zat gizi} dan oksigen dengan cepat
dan harus mendapat gantinya dengan cepat; \omterm{def:g4:heart-and-blood:heart}{jantungnya} mempercepat dan
memperkuat denyutnya sehingga \omterm{prop:g4:journey-of-food:absorb}{darahnya} menjalankan ronde pengantarannya
lebih sering.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:8}
Jawaban terbuka. Biasanya kira-kira $80$--$100$ ketika beristirahat,
$120$--$150$ sesudah lompatan bintangnya --- rantai pasokan yang
sedang bekerja.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:9}
Ketika meninggalkan \omterm{def:g4:breathing:organs}{paru-paru} ia baru saja memuat oksigen --- merah
cerah. Di dalam \omterm{def:g3:bones-and-muscles:muscle}{ototnya} ia menyerahkan oksigennya lalu menerima karbon
dioksida --- lebih gelap. Warnanya mengikuti muatannya.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:10}
\omterm{def:g4:journey-of-food:tract}{Usus halus} memasukkan \omterm{def:g4:journey-of-food:digestion}{zat gizi} ke dalam \omterm{prop:g4:journey-of-food:absorb}{darah}; \omterm{def:g4:breathing:organs}{paru-paru} memasukkan
oksigen dan mengeluarkan karbon dioksida; \omterm{def:g4:heart-and-blood:heart}{jantung} adalah pompa yang
menjaga \omterm{prop:g4:journey-of-food:absorb}{darahnya} tetap bergerak di antara keduanya lalu keluar ke
setiap pemakainya --- \omterm{def:g3:bones-and-muscles:muscle}{otot}, otak, \omterm{def:g3:bones-and-muscles:skeleton}{tulang}, \omterm{def:g1:the-five-senses:sense}{kulit}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:11}
\omterm{def:g4:heart-and-blood:heart}{Jantung} yang terlatih adalah \omterm{def:g3:bones-and-muscles:muscle}{otot} yang lebih kuat: tiap denyutnya
memindahkan lebih banyak \omterm{prop:g4:journey-of-food:absorb}{darah}. Mengantarkan pasokan yang sama lalu
membutuhkan lebih sedikit denyut per menit --- sehingga denyut $50$
yang tenang pada sang pengendara sepeda mengerjakan pekerjaan yang
sama dengan denyut $75$ pada orang yang tidak terlatih, dengan tenaga
cadangan yang tersimpan untuk saat berusaha.
\end{solution}
