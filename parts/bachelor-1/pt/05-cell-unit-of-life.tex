\chapter{A célula: unidade da vida}\label{ch:b1:cell-unit-of-life}

Em 1665 Robert Hooke pôs uma lasca de cortiça sob uma lente e a viu
dividida em caixinhas que ele chamou de \omterm{def:b1:cell-unit-of-life:cell}{células}, em referência aos
aposentos de um mosteiro. Dez anos depois um comerciante de \omterm{def:b1:body-plans-tissues:tissue}{tecidos}
holandês, Antoni van Leeuwenhoek, polindo as próprias lentes até um
\omterm{def:b1:cell-unit-of-life:resolution}{aumento} que ninguém mais alcançava, viu seres vivos nadando numa gota de
água de lagoa, no raspado dos dentes dele, no próprio sêmen. Foi preciso
mais um século e meio para enunciar o que os dois tinham visto: que todo
\omterm{def:b1:organism-environment:organism}{organismo} é feito de \omterm{def:b1:cell-unit-of-life:cell}{células}, e que toda \omterm{def:b1:cell-unit-of-life:cell}{célula} vem de uma \omterm{def:b1:cell-unit-of-life:cell}{célula}. Este
capítulo enuncia a \omterm{prop:b1:cell-unit-of-life:theory}{teoria celular} e as evidências dela, contrasta os dois
tipos de \omterm{def:b1:cell-unit-of-life:cell}{célula}, e descreve os instrumentos --- os microscópios, a
centrífuga, o frasco de cultura --- que nos deixam ver as \omterm{def:b1:cell-unit-of-life:cell}{células},
desmontá-las e mantê-las vivas fora de um \omterm{def:b1:organism-environment:organism}{organismo}.

\section{A teoria celular}

\begin{definition}[Célula]\label{def:b1:cell-unit-of-life:cell}
Uma \emph{célula}\index{célula} é a menor unidade de matéria viva: um
volume de \emph{citoplasma}\index{citoplasma} aquoso delimitado por uma
\emph{membrana plasmática}\index{membrana plasmática}, contendo o
material genético (DNA) e a maquinaria que o expressa, capaz de captar
nutrientes, transformar energia, manter a própria organização e, quando
as condições permitem, dividir-se em duas células.
\end{definition}

\begin{proposition}[A teoria celular]\label{prop:b1:cell-unit-of-life:theory}
\index{teoria celular}
\begin{enumerate}
  \item Todo \omterm{def:b1:organism-environment:organism}{organismo} é feito de uma ou de várias \omterm{def:b1:cell-unit-of-life:cell}{células}, e a \omterm{def:b1:cell-unit-of-life:cell}{célula} é
        a unidade de estrutura e de função dos seres vivos (Schleiden e
        Schwann, 1838--1839).
  \item Toda \omterm{def:b1:cell-unit-of-life:cell}{célula} surge de uma \omterm{def:b1:cell-unit-of-life:cell}{célula} preexistente por divisão
        (Virchow, 1855: \emph{omnis cellula e cellula}); não há geração
        espontânea.
  \item O material hereditário é transmitido de \omterm{def:b1:cell-unit-of-life:cell}{célula} a \omterm{def:b1:cell-unit-of-life:cell}{célula} na
        divisão, de modo que a continuidade da vida é a continuidade das
        \omterm{def:b1:cell-unit-of-life:cell}{células}.
\end{enumerate}
\end{proposition}
\begin{proof}[Evidências]
A cortiça de Hooke (1665) e os cortes de plantas de Grew e Malpighi
mostraram os \omterm{def:b1:body-plans-tissues:tissue}{tecidos} vegetais como conjuntos de câmaras; as lentes de
Leeuwenhoek (década de 1670) mostraram \omterm{def:b1:organism-environment:organism}{organismos} unicelulares, \omterm{def:b1:cell-unit-of-life:cell}{células}
do sangue e espermatozoides. Dois séculos de lentes cada vez melhores
encontraram \omterm{def:b1:cell-unit-of-life:cell}{células} em todo \omterm{def:b1:body-plans-tissues:tissue}{tecido} examinado, animal tanto quanto
vegetal, depois que Schwann reconheceu \omterm{def:b1:cell-unit-of-life:cell}{células} nucleadas na cartilagem e
em embriões. A divisão foi vista diretamente em algas e em embriões, e
Virchow não encontrou nenhum \omterm{def:b1:body-plans-tissues:tissue}{tecido}, são ou doente, cujas \omterm{def:b1:cell-unit-of-life:cell}{células} não
viessem de outras \omterm{def:b1:cell-unit-of-life:cell}{células}. Os frascos de pescoço de cisne de Pasteur
(1861) --- caldos que ficavam estéreis por meses quando a poeira não os
alcançava, e que fervilhavam em poucos dias quando ela alcançava ---
encerraram a questão da geração espontânea. O terceiro enunciado é obra
do fim do século XIX, que acompanhou os cromossomos pela divisão
(\cref{ch:b1:replication-mitosis}).
\end{proof}

\begin{example}[Uma célula, muitas células]\label{ex:b1:cell-unit-of-life:onemany}
Uma bactéria, uma levedura, uma ameba são \omterm{def:b1:organism-environment:organism}{organismos} inteiros de uma só
\omterm{def:b1:cell-unit-of-life:cell}{célula}, que fazem tudo --- alimentar-se, mover-se, dividir-se --- em
poucos micrômetros. Um ser humano tem cerca de $\num{3e13}$ \omterm{def:b1:cell-unit-of-life:cell}{células} de
duzentos tipos, nenhuma das quais poderia viver sozinha numa lagoa, e
cada uma delas descende por divisões ininterruptas de um único óvulo
fecundado. O coelho do \cref{ch:b1:mammal-organization} e o carvalho do
\cref{ch:b1:flowering-plant-organization} são a mesma teoria aplicada
duas vezes.
\end{example}

\section{Dois tipos de célula}

\begin{definition}[Células procarióticas e eucarióticas]\label{def:b1:cell-unit-of-life:prokeuk}
Uma \emph{célula procariótica}\index{célula procariótica} (bactérias,
arqueias) não tem núcleo: o DNA dela, em geral um único cromossomo
circular, fica numa região do \omterm{def:b1:cell-unit-of-life:cell}{citoplasma} chamada
\emph{nucleoide}\index{nucleoide}, e ela não tem compartimentos
delimitados por membrana. Ela é pequena
(\qtyrange{0.5}{5}{\micro m}) e delimitada por uma \omterm{def:b1:cell-unit-of-life:cell}{membrana plasmática}
e, por fora dela, uma \emph{parede celular}\index{parede celular}. Uma
\emph{célula eucariótica}\index{célula eucariótica} (protistas, fungos,
plantas, animais) guarda o DNA dela, na forma de vários cromossomos
lineares, dentro de um \emph{núcleo}\index{núcleo} delimitado por uma
membrana dupla, e divide o \omterm{def:b1:cell-unit-of-life:cell}{citoplasma} em \emph{organelas}\index{organela}
delimitadas por membrana (\cref{ch:b1:eukaryotic-cell}). Ela é maior
(\qtyrange{10}{100}{\micro m}) e tem um esqueleto interno de filamentos
proteicos.
\end{definition}

\begin{proposition}[A célula bacteriana]\label{prop:b1:cell-unit-of-life:bacterium}
A \emph{E. coli}, a bactéria-modelo, é um bastonete de \qty{2}{\micro m}
de comprimento e \qty{0.8}{\micro m} de largura: um volume de cerca de
\qty{1}{\micro m^3} que abriga um cromossomo de $\num{4.6e6}$ pares de
bases (\qty{1.5}{mm} de DNA dobrado dez mil vezes), uns \num{20000}
ribossomos, alguns milhões de moléculas de proteína de \num{2000} tipos,
e muitas vezes um ou mais \emph{plasmídeos}\index{plasmídeo} ---
pequenas moléculas circulares de DNA que carregam genes adicionais. O
envoltório dela é, de dentro para fora, a \omterm{def:b1:cell-unit-of-life:cell}{membrana plasmática}, uma
parede fina de \emph{peptidoglicano}\index{peptidoglicano} (uma malha de
cadeias de açúcares ligadas transversalmente por peptídeos curtos, que
resiste à pressão interna da \omterm{def:b1:cell-unit-of-life:cell}{célula}) e uma membrana externa. As
bactérias com uma parede espessa de peptidoglicano e sem membrana
externa retêm o cristal violeta da
\emph{coloração de Gram}\index{coloração de Gram} (gram-positivas);
aquelas com parede fina sob uma membrana externa o perdem
(gram-negativas, como a \emph{E. coli}). Os flagelos giram como hélices;
os pili prendem a \omterm{def:b1:cell-unit-of-life:cell}{célula} a superfícies e a outras \omterm{def:b1:cell-unit-of-life:cell}{células}.
\end{proposition}

\begin{method}[A coloração de Gram]\label{met:b1:cell-unit-of-life:gram}
\begin{enumerate}
  \item Fixe ao calor um esfregaço fino de bactérias sobre uma lâmina.
  \item Core com cristal violeta (todas as \omterm{def:b1:cell-unit-of-life:cell}{células} roxas) e depois com
        iodo, que forma um complexo grande com o corante dentro das
        \omterm{def:b1:cell-unit-of-life:cell}{células}.
  \item Enxágue com álcool: o peptidoglicano espesso das \omterm{def:b1:cell-unit-of-life:cell}{células}
        gram-positivas se contrai e retém o complexo; a parede fina das
        gram-negativas, uma vez que o álcool dissolveu a membrana
        externa, deixa que ele seja lavado.
  \item Contraste com um corante rosa (safranina): as \omterm{def:b1:cell-unit-of-life:cell}{células}
        gram-positivas aparecem roxas e as gram-negativas, rosas. O
        resultado indica a estrutura da parede e, com ela, a
        sensibilidade a muitos antibióticos.
\end{enumerate}
\end{method}

\begin{example}[Por que a célula eucariótica é compartimentada]\label{ex:b1:cell-unit-of-life:whycompart}
Uma \omterm{def:b1:cell-unit-of-life:cell}{célula} de \qty{20}{\micro m} tem mil vezes o volume de uma bactéria,
mas apenas cem vezes a superfície de membrana dela
(\cref{prop:b1:organism-environment:sv}). As membranas são onde uma
\omterm{def:b1:cell-unit-of-life:cell}{célula} carrega as cadeias transdutoras de energia e os sistemas de
transporte dela; o eucarioto recupera superfície dobrando membranas
dentro de si --- a membrana interna de uma mitocôndria, as lâminas do
retículo endoplasmático --- e ganha, além disso, câmaras separadas em
que químicas incompatíveis (a digestão no lisossomo, a síntese no
citosol) podem funcionar lado a lado.
\end{example}

\section{Ver as células: a microscopia}

\begin{definition}[Aumento e resolução]\label{def:b1:cell-unit-of-life:resolution}
Um microscópio \emph{amplia}\index{aumento} (aumenta a imagem) e
\emph{resolve}\index{resolução} (separa pontos vizinhos). Só o segundo
limita o que se pode ver: o
\emph{poder de resolução}\index{poder de resolução} é a menor distância
$d$ entre dois pontos que ainda aparecem distintos,
\[
d = \frac{0.61\,\lambda}{\mathrm{NA}},
\]
em que $\lambda$ é o comprimento de onda da iluminação e $\mathrm{NA}$ a
abertura numérica da objetiva (o seno do semiângulo de coleta de luz
dela vezes o índice de refração do meio, no máximo cerca de $1.4$ com
óleo). Ampliar além do poder de resolução aumenta o borrão, e não o
detalhe.
\end{definition}

\begin{proposition}[Microscópios óptico e eletrônico]\label{prop:b1:cell-unit-of-life:microscopes}
Com luz visível ($\lambda \approx \qty{550}{nm}$) e $\mathrm{NA} = 1.4$,
o \emph{microscópio óptico}\index{microscópio óptico} resolve
\qty{0.24}{\micro m}: \omterm{def:b1:cell-unit-of-life:cell}{células}, \omterm{def:b1:cell-unit-of-life:prokeuk}{núcleos}, cloroplastos, mitocôndrias como
pontos, bactérias como bastonetes --- e espécimes vivos e em movimento.
O contraste numa \omterm{def:b1:cell-unit-of-life:cell}{célula} transparente é obtido por coloração (que em
geral mata), por \emph{contraste de fase} (que converte diferenças de
índice de refração em diferenças de brilho) ou por
\emph{fluorescência} (marcar uma molécula com um corante ou uma proteína
fluorescente e captar apenas a luz dela). O
\emph{microscópio eletrônico}\index{microscópio eletrônico} usa elétrons
de comprimento de onda de alguns picômetros; a \omterm{def:b1:cell-unit-of-life:resolution}{resolução} prática dele,
limitada pelas lentes, é de cerca de \qty{0.2}{nm} na transmissão (MET:
os elétrons atravessam um corte de \qty{50}{nm} de espessura corado com
metais pesados, o que dá a estrutura interna das \omterm{def:b1:cell-unit-of-life:prokeuk}{organelas}) e de alguns
nanômetros na varredura (MEV: um feixe varre uma superfície recoberta de
metal, o que dá o relevo dela em profundidade). A microscopia eletrônica
exige vácuo e espécimes mortos, fixados e desidratados.
\end{proposition}

\begin{method}[Escolher o instrumento]\label{met:b1:cell-unit-of-life:choose}
\begin{enumerate}
  \item Para acompanhar um processo vivo (divisão celular, movimento,
        uma proteína fluorescente que passa de um compartimento a
        outro): microscopia óptica, com contraste de fase ou
        fluorescência, em \omterm{def:b1:cell-unit-of-life:cell}{células} não fixadas.
  \item Para contar e identificar \omterm{def:b1:cell-unit-of-life:cell}{células} num \omterm{def:b1:body-plans-tissues:tissue}{tecido}: uma preparação
        fixada, cortada e corada, ao \omterm{prop:b1:cell-unit-of-life:microscopes}{microscópio óptico}.
  \item Para ver a estrutura interna de uma \omterm{def:b1:cell-unit-of-life:prokeuk}{organela}, de uma membrana,
        de um vírus: microscopia eletrônica de transmissão de cortes
        finos, ou de partículas com coloração negativa.
  \item Para ver o relevo da superfície de uma \omterm{def:b1:cell-unit-of-life:cell}{célula} ou de um \omterm{def:b1:body-plans-tissues:tissue}{tecido}:
        microscopia eletrônica de varredura.
  \item Para localizar uma proteína: imunofluorescência (óptica) ou
        marcação com imunogold (eletrônica), com um anticorpo contra ela.
\end{enumerate}
\end{method}

\begin{example}[Uma bactéria e uma mitocôndria]\label{ex:b1:cell-unit-of-life:seeing}
Ao \omterm{prop:b1:cell-unit-of-life:microscopes}{microscópio óptico} com $1000\times$, uma \omterm{def:b1:cell-unit-of-life:cell}{célula} de \emph{E. coli} é

\end{example}

\section{Desmontar as células e mantê-las vivas}

\begin{definition}[Fracionamento celular]\label{def:b1:cell-unit-of-life:fractionation}
O \emph{fracionamento celular}\index{fracionamento celular} separa os
componentes de uma \omterm{def:b1:cell-unit-of-life:cell}{célula} para estudo. O \omterm{def:b1:body-plans-tissues:tissue}{tecido} é
\emph{homogeneizado}\index{homogeneizado} --- as \omterm{def:b1:cell-unit-of-life:cell}{células} são rompidas
num tampão frio e isotônico, por trituração ou cisalhamento, deixando as
\omterm{def:b1:cell-unit-of-life:prokeuk}{organelas} intactas --- e o homogeneizado é submetido à
\emph{centrifugação diferencial}\index{centrifugação diferencial}: uma
série de centrifugações de força e duração crescentes, cada uma das
quais sedimenta as partículas maiores e mais densas que ainda restam em
suspensão. Cada fração é identificada pela aparência ao \omterm{prop:b1:cell-unit-of-life:microscopes}{microscópio
eletrônico} e pelas \emph{enzimas marcadoras}\index{enzima marcadora},
atividades que se sabe residirem em um único compartimento.
\end{definition}

\begin{proposition}[Sedimentação]\label{prop:b1:cell-unit-of-life:stokes}
Uma partícula esférica de raio $r$ e densidade $\rho_p$ num fluido de
densidade $\rho_f$ e viscosidade $\eta$, num campo centrífugo $g_c$ (um
múltiplo de $g$), sedimenta à velocidade estacionária
\[
v = \frac{2}{9}\,\frac{r^2\,(\rho_p - \rho_f)\,g_c}{\eta}
\]
(lei de Stokes). A velocidade cresce com o quadrado do raio: um \omterm{def:b1:cell-unit-of-life:prokeuk}{núcleo}
de \qty{5}{\micro m} cai cem vezes mais rápido que uma mitocôndria de
\qty{0.5}{\micro m}, que cai mil vezes mais rápido que um ribossomo de
\qty{15}{nm}. É por isso que o mesmo tubo os separa a velocidades
crescentes.
\end{proposition}
\begin{proof}
Na velocidade estacionária o arrasto de Stokes, $6\pi\eta r v$,
equilibra o peso aparente da partícula no fluido,
$\frac{4}{3}\pi r^3(\rho_p - \rho_f)g_c$; resolver para $v$ dá a
fórmula.
\end{proof}

\begin{definition}[Cultura de células]\label{def:b1:cell-unit-of-life:culture}
A \emph{cultura de células}\index{cultura de células} mantém as \omterm{def:b1:cell-unit-of-life:cell}{células}
vivas e em divisão fora do \omterm{def:b1:organism-environment:organism}{organismo}, num meio estéril que fornece
nutrientes, sais, fatores de crescimento e, para as \omterm{def:b1:cell-unit-of-life:cell}{células} animais, uma
superfície de adesão, à temperatura do \omterm{def:b1:organism-environment:organism}{organismo}. Bactérias e leveduras
crescem em meios simples e duplicam em minutos a horas; as \omterm{def:b1:cell-unit-of-life:cell}{células}
animais retiradas de um \omterm{def:b1:body-plans-tissues:tissue}{tecido} (as \emph{culturas primárias}) se dividem
um número limitado de vezes, ao passo que as
\emph{linhagens celulares} derivadas de tumores ou de \omterm{def:b1:cell-unit-of-life:cell}{células}
transformadas se dividem indefinidamente. A cultura faz da \omterm{def:b1:cell-unit-of-life:cell}{célula} um
objeto experimental: um tipo, um meio, uma variável de cada vez.
\end{definition}

\begin{example}[Uma curva de crescimento]\label{ex:b1:cell-unit-of-life:growthcurve}
Um frasco de \emph{E. coli} num meio rico a \qty{37}{\celsius}: uma fase
de latência de meia hora, depois crescimento exponencial --- a contagem
dobra a cada \qty{20}{min}, de modo que $N = N_0\,2^{t/20}$ --- até que
um nutriente se esgote ou os resíduos se acumulem e a cultura atinja um
patamar perto de $10^9$ \omterm{def:b1:cell-unit-of-life:cell}{células} por mililitro. A partir de uma única
\omterm{def:b1:cell-unit-of-life:cell}{célula}, $2^{30} \approx 10^9$ \omterm{def:b1:cell-unit-of-life:cell}{células} em dez horas. A mesma curva, com
um tempo de duplicação de um dia, descreve uma \omterm{def:b1:cell-unit-of-life:culture}{cultura de células}
humanas.
\end{example}

\section{Problema: Fracionar um fígado}

\begin{problem}[{Problema de fim de semana --- dez gramas de fígado de
rato desmontados por centrifugação: sedimentos, marcadores, lei de
Stokes e o balanço de uma enzima, terminando no fator de purificação da
fração mitocondrial}]
\label{pb:b1:cell-unit-of-life:1}
\qty{10}{g} de fígado de rato são \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizados} em \qty{50}{mL} de
sacarose isotônica fria e centrifugados em três etapas: $1000\,g$ por
\qty{10}{min} (sedimento 1), $\num{10000}\,g$ por \qty{20}{min}
(sedimento 2), $\num{100000}\,g$ por \qty{60}{min} (sedimento 3),
restando o sobrenadante final. O teor de proteína e duas atividades
enzimáticas são medidos no \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado} e em cada fração:

\begin{center}
\small
\begin{tabular}{lccc}
\hline
fração & proteína (mg) & succinato-desidrogenase (U) & lactato-desidrogenase (U) \\
\hline
\omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado} & 2000 & 100 & 400 \\
sedimento 1 & 500 & 15 & 20 \\
sedimento 2 & 200 & 60 & 12 \\
sedimento 3 & 300 & 5 & 8 \\
sobrenadante final & 950 & 4 & 350 \\
\hline
\end{tabular}
\end{center}

Tome $\eta = \qty{1e-3}{Pa.s}$ para o meio, um percurso de sedimentação
de \qty{5}{cm} e $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{Parte I --- O que está onde.}
\begin{enumerate}
  \item Cite a \omterm{def:b1:cell-unit-of-life:prokeuk}{organela} principal de cada sedimento.
  \item Por que o meio de homogeneização precisa ser frio e isotônico?
  \item A succinato-desidrogenase é marcadora de qual compartimento? E a
        lactato-desidrogenase?
  \item Calcule a recuperação de proteína: a soma sobre as frações como
        porcentagem do \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado}. Comente.
  \item Calcule da mesma maneira a recuperação de cada atividade
        enzimática.
  \item Calcule o teor de proteína por grama de fígado e a fração da
        massa fresca do fígado que é proteína.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Lei de Stokes.}
Mitocôndrias: raio \qty{0.5}{\micro m}, excesso de densidade
\qty{100}{kg/m^3}. \omterm{def:b1:cell-unit-of-life:prokeuk}{Núcleos}: raio \qty{4}{\micro m}, mesmo excesso.
Ribossomos: raio \qty{12}{nm}, excesso de densidade \qty{600}{kg/m^3}.
\begin{enumerate}[resume]
  \item Calcule a velocidade de sedimentação de uma mitocôndria a
        $\num{10000}\,g$ e o tempo para percorrer \qty{5}{cm}.
        \qty{20}{min} bastam?
  \item Calcule a distância que uma mitocôndria percorre durante a
        primeira centrifugação ($1000\,g$, \qty{10}{min}). Que fração
        das mitocôndrias, uniformemente distribuídas no início, seria
        sedimentada na primeira etapa?
  \item Compare com os \qty{15}{\%} de succinato-desidrogenase
        encontrados no sedimento 1. O que mais pode arrastar
        mitocôndrias para o sedimento 1?
  \item Calcule o tempo para um \omterm{def:b1:cell-unit-of-life:prokeuk}{núcleo} percorrer \qty{5}{cm} a
        $1000\,g$.
  \item Calcule o tempo para um ribossomo percorrer \qty{5}{cm} a
        $\num{100000}\,g$. Um ribossomo seria sedimentado a
        $\num{10000}\,g$ em \qty{20}{min}?
  \item Explique por que a sequência de centrifugações é de força
        crescente e por que cada etapa precisa ser longa o bastante, mas
        não longa demais.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Atividade específica e purificação.}
A atividade específica de uma enzima numa fração é a atividade dela por
miligrama de proteína. O fator de purificação de uma fração para uma
enzima é a atividade específica dela dividida pela do \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado}; o
rendimento é a atividade da fração dividida pela do \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado}.
\begin{enumerate}[resume]
  \item Calcule a atividade específica da succinato-desidrogenase no
        \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado} e em cada fração.
  \item Calcule o fator de purificação e o rendimento da
        succinato-desidrogenase no sedimento 2.
  \item Calcule o fator de purificação da succinato-desidrogenase no
        sedimento 3 e interprete o valor dele.
  \item Calcule a atividade específica da lactato-desidrogenase no
        \omterm{def:b1:cell-unit-of-life:fractionation}{homogeneizado} e no sobrenadante final, e o fator de purificação
        e o rendimento correspondentes.
  \item Por que o fator de purificação da enzima citosólica é menor que
        o da enzima mitocondrial, embora o rendimento dela seja maior?
  \item O sedimento 2 contém \qty{12}{U} de lactato-desidrogenase. A
        enzima é em parte mitocondrial? Proponha uma explicação melhor e
        uma maneira de testá-la.
  \item Que fração da proteína do fígado é mitocondrial, se toda a
        succinato-desidrogenase for mitocondrial e o sedimento 2 tiver
        \qty{60}{\%} de pureza? (Dica: use a atividade específica das
        mitocôndrias puras, deduzida do sedimento 2, e a atividade
        total.)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Conferir as frações.}
\begin{enumerate}[resume]
  \item O sedimento 2 é examinado por microscopia eletrônica: qual
        instrumento e qual preparação, e o que deve ser visto?
  \item Como você verificaria se há \omterm{def:b1:cell-unit-of-life:cell}{células} intactas no sedimento 1, e o
        que faria se encontrasse muitas?
  \item O sedimento 2 é ressuspenso e depositado sobre um gradiente de
        densidade de sacarose, e depois centrifugado até que cada
        partícula pare onde a densidade dela iguala a do meio. As
        mitocôndrias formam banda a \qty{1.18}{g/mL}, os lisossomos a
        \qty{1.22}{g/mL} e os peroxissomos a \qty{1.24}{g/mL}. Explique
        por que essa separação funciona quando a \omterm{def:b1:cell-unit-of-life:fractionation}{centrifugação
        diferencial} não conseguia obtê-la.
  \item A atividade da fosfatase ácida no sedimento 2 sobe bruscamente
        quando um detergente é acrescentado ao ensaio. Explique, e diga
        o que isso informa sobre o estado dos lisossomos no sedimento.
  \item Um estudante homogeneíza em água pura em vez de sacarose
        isotônica. Preveja o efeito sobre o sedimento 2 e sobre a
        distribuição da succinato-desidrogenase e da fosfatase ácida.
  \item Resuma o resultado: o fator de purificação e o rendimento da
        fração mitocondrial, e o que limita cada um.
\end{enumerate}
\end{problem}
