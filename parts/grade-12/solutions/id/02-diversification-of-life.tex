\section*{Bab \ref{ch:g12:diversification-of-life} --- Penganekaragaman Makhluk Hidup}

\begin{solution}{exo:g12:diversification-of-life:1}
Penggandaan \omterm{def:g10:universal-dna:gene}{gen} dan pemencarannya; \omterm{prop:g11:antibiotic-resistance:variation}{pemindahan gen mendatar};
hibridisasi bersama poliploidi; perubahan pengaturan \omterm{prop:g12:diversification-of-life:development}{gen perkembangan};
\omterm{def:g12:diversification-of-life:symbiosis}{simbiosis}; dan perilaku yang diwariskan.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:2}
Seperangkat \omterm{def:g10:universal-dna:gene}{gen} berkerabat yang berasal dari satu \omterm{def:g10:universal-dna:gene}{gen} moyang lewat
penggandaan yang berturut-turut, dengan tiap salinannya lalu memencar
oleh \omterm{def:g11:mutations:mutation}{mutasi} menuju fungsi yang berbeda (\omterm{def:g11:the-eye:photoreceptors}{opsin}, globin).
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:3}
\omterm{def:g11:cell-cycle-mitosis:chromosome}{Kromosomnya}, satu perangkat dari tiap \omterm{def:g10:biodiversity-scales:species}{spesies} induknya, tidak punya
\omterm{def:g10:common-ancestry:homology}{homolog} untuk berpasangan pada \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosisnya}, sehingga gametnya tidak
seimbang. Pelipatduaannya memberi tiap \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya} pasangan yang
serupa: pemasangan dan \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosisnya} menjadi teratur dan tumbuhannya subur.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:4}
\omterm{def:g10:universal-dna:information}{DNA} melingkarnya sendiri; \omterm{def:g10:cells-common-unit:organelle}{ribosom} jenis bakteri yang peka terhadap
\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotik} antibakteri; berlipat ganda lewat pembelahan; membran ganda;
urutan yang paling dekat dengan sekelompok bakteri yang masih hidup.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:5}
Perilaku yang dipelajari dari anggota kelompok lain dan diteruskan
tanpa \omterm{def:g10:universal-dna:gene}{gen}: memecah kacang dengan batu pada sebagian masyarakat simpanse
dan tidak pada yang lain.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:6}
$14 + 7 = 21$ \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}: 7 A dari einkorn, 7 A dan 7 B dari emmer. Tujuh
pasangan (A dengan A) dapat terbentuk; ke-7 \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} B-nya tidak punya
pasangan dan terbagi secara acak, sehingga gametnya tidak seimbang:
mandul.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:7}
Perpisahan $\alpha$--$\beta$ (50\% serupa) lebih tua daripada
perpisahan $\beta$--$\gamma$ (80\%): $\alpha$ bercabang lebih dahulu,
lalu $\beta$ dan $\gamma$ berpisah satu sama lain.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:8}
\omterm{def:g10:universal-dna:gene}{Gen} tungkainya hadir tetapi isyarat yang menyalakannya pada sisi
embrionya tidak diberikan, atau diberikan lalu dihentikan; bedanya
terletak pada pengaturan pengungkapannya, bukan pada \omterm{def:g10:universal-dna:gene}{gennya}.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:9}
Pada jumlah dan saat sebuah isyarat pertumbuhan dalam paruh embrionya:
yaitu beda pengaturan. Perubahan pengaturan hanya menuntut \omterm{def:g11:mutations:mutation}{mutasi} kecil
pada sebuah urutan kendali, sedangkan \omterm{def:g11:gene-expression:protein}{protein} baru akan menuntut banyak
\omterm{def:g11:mutations:mutation}{mutasi}; perubahan itu dapat muncul dan terpilih dalam ribuan keturunan.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:10}
Bentuknya, kemampuannya menghuni batu telanjang, ketahanannya terhadap
kekeringan dan pertumbuhannya yang lambat tidak dimiliki satu pun
mitranya sendirian; semuanya muncul dari pergaulannya. Lumut keraknya
adalah organisme majemuk dengan sifatnya sendiri.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:11}
Perilaku yang diwariskan: bahannya tersedia di kedua sisinya,
populasinya berkerabat dekat, dan bedanya mengikuti sungainya --- yaitu
penghalang bagi sentuhan dan karena itu bagi pembelajarannya, bukan
bagi \omterm{def:g10:universal-dna:gene}{gen} atau sumber dayanya.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:12}
Sebuah virus menyisipkan \omterm{def:g10:universal-dna:gene}{gennya} ke dalam genom \omterm{def:g10:cells-common-unit:cell}{sel} kelamin seorang
moyang primata puluhan juta tahun lalu; \omterm{def:g10:universal-dna:gene}{gennya} lalu diwariskan dan
tersimpan. Ujinya: \omterm{def:g10:universal-dna:gene}{gen} yang sama seharusnya duduk pada kedudukan
\omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} yang sama pada seluruh primata dan tidak ada pada kedudukan
itu pada mamalia lain; urutannya seharusnya paling dekat dengan urutan
virusnya.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:13}
\omterm{def:g10:cells-common-unit:organelle}{Ribosom} \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya} berjenis bakteri dan mengikat \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotik} yang
sama: pada dosis yang tinggi obatnya memperlambat penyusunan \omterm{def:g11:gene-expression:protein}{protein}
\omterm{def:g10:cells-common-unit:organelle}{mitokondrianya}. Kepekaan itu diwarisi dari moyang bakteri \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya}.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:14}
Bilangan \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} \omterm{prop:g12:diversification-of-life:polyploidy}{poliploidnya} tidak lagi cocok dengan bilangan
\omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} kedua induknya, sehingga persilangan dengannya memberi
keturunan yang mandul: pemencilannya seketika. \omterm{def:g10:universal-dna:gene}{Alel} baru harus menyebar
dalam sebuah populasi sepanjang banyak keturunan sebelum ia membedakan
sebuah kelompok. Karena itu \omterm{def:g10:biodiversity-scales:species}{spesies} tumbuhan dapat muncul dalam satu keturunan.
\end{solution}

\begin{solution}{exo:g12:diversification-of-life:15}
Pada dasarnya, tiap urutan baru --- salinan tergandakan yang memencar,
perubahan pengaturan, \omterm{def:g10:universal-dna:gene}{gen} sebuah plasmid yang dipindahkan --- bermula
sebagai sebuah \omterm{def:g11:mutations:mutation}{mutasi} di suatu tempat. Tetapi penggandaan, pemindahan
dan poliploidi memindahkan serta melipatgandakan \omterm{def:g10:universal-dna:gene}{gen} dan genom yang
utuh sekaligus; perubahan pengaturan menghasilkan bentuk baru tanpa
\omterm{def:g11:gene-expression:protein}{protein} baru; \omterm{def:g12:diversification-of-life:symbiosis}{simbiosis} menggabungkan genom \omterm{def:g10:biodiversity-scales:species}{spesies} yang berbeda; dan
budaya mewariskan keragaman sama sekali tanpa \omterm{def:g10:universal-dna:gene}{gen}. \omterm{def:g11:mutations:mutation}{Mutasi} memasok
hurufnya; proses ini menyusun ulang halaman dan buku yang utuh.
\end{solution}

\begin{solution}{pb:g12:diversification-of-life:1}
\textbf{1.} Emmer 28, gandum roti 42.

\textbf{2.} Tidak ada: \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} A dan B-nya tidak \omterm{def:g10:common-ancestry:homology}{homolog}. Gametnya
menerima pilahan acak dari ke-14 \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya}, dan nyaris tidak pernah
menerima perangkat yang lengkap.

\textbf{3.} Dengan 28 \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}, tiap A punya pasangan A yang serupa dan
tiap B punya B: 14 pasangan, \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosis} yang teratur, gamet seimbang
berisi 14.

\textbf{4.} Tidak ada yang dapat berpasangan: 7 \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} A, 7 B dan 7
D, yang masing-masing tanpa \omterm{def:g10:common-ancestry:homology}{homolog}. Mandul.

\textbf{5.} 21 pasangan; sebutir serbuk sarinya membawa 21.

\textbf{6.} Kedua salinan lainnya tetap memasok \omterm{def:g11:gene-expression:protein}{proteinnya}. Salinan
yang berlebih bebas menumpuk \omterm{def:g11:mutations:mutation}{mutasi} dan memencar --- yaitu penggandaan
pada skala genom yang utuh.

\textbf{7.} Di ladang daerah tempat emmer ditanam, antara 10\,000 dan
8\,000 tahun lalu, dengan rumput kambing liar 2 yang tumbuh sebagai
gulma di antara tanamannya.

\textbf{8.} Bahwa kedua langkahnya --- hibridisasi dan pelipatduaannya
--- sudah cukup untuk menghasilkan gandum roti dari induknya:
sejarahnya dapat dijalankan ulang.

\textbf{9.} Ya: ia tidak dapat menghasilkan keturunan yang subur
dengan einkorn, sehingga menurut rumusannya ia \omterm{def:g10:biodiversity-scales:species}{spesies} yang berbeda,
walaupun ia berasal darinya.

\textbf{10.} \omterm{def:g11:cell-cycle-mitosis:chromosome}{Kromosom} D rumput kambing 2 \omterm{def:g10:common-ancestry:homology}{homolog} dengan perangkat D
gandumnya dan berpasangan dengannya, sehingga sebuah persilangan yang
disusul persilangan balik dapat memindahkan \omterm{def:g10:universal-dna:gene}{gennya}; \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} tumbuhan
yang tak berkerabat sama sekali tidak akan berpasangan.

\textbf{11.} \omterm{def:g10:universal-dna:gene}{Gennya} sendiri nyaris tidak berbeda; beda pada tubuhnya
pasti terletak pada cara \omterm{def:g10:universal-dna:gene}{gennya} dipakai.

\textbf{12.} Perubahan pengaturan pengungkapannya --- yaitu bentang
daerah di sepanjang sumbunya tempat \omterm{def:g10:universal-dna:gene}{gennya} aktif.

\textbf{13.} \omterm{def:g10:universal-dna:gene}{Gen} tungkainya hadir dan mulai bekerja; isyarat yang
menopang pertumbuhan tungkainya tidak ada, sehingga tunasnya berhenti.
Lagi-lagi pengaturannya, bukan hilangnya \omterm{def:g10:universal-dna:gene}{gennya}.

\textbf{14.} Percobaan itu menunjukkan bahwa mengubah tempat \omterm{def:g10:universal-dna:gene}{gennya}
diungkapkan sudah cukup untuk mengubah daerah tubuhnya: kaitan antara
pengungkapan dan bentuknya bersifat sebab akibat.

\textbf{15.} Perubahan gandum rotinya menambahkan perangkat yang utuh
ke dalam genomnya (besarnya); perubahan ular sancanya mengubah tempat
\omterm{def:g10:universal-dna:gene}{gen} bersamanya dipakai (pengaturannya).

\textbf{16.} \omterm{def:g10:universal-dna:information}{DNA} melingkarnya sendiri (yang diharapkan: tidak ada,
seluruh \omterm{def:g10:universal-dna:gene}{gennya} dalam \omterm{def:g10:cells-common-unit:organelle}{inti sel}); \omterm{def:g10:cells-common-unit:organelle}{ribosom} bakteri (yang diharapkan:
\omterm{def:g10:cells-common-unit:organelle}{ribosom} eukariot); berlipat ganda lewat pembelahan (yang diharapkan:
dirakit dari bagiannya); membran ganda (yang diharapkan: membran
tunggal seperti \omterm{def:g10:cells-common-unit:organelle}{organel} lain). Dan \omterm{def:g10:universal-dna:gene}{gennya} menyerupai \omterm{def:g10:universal-dna:gene}{gen} bakteri (yang
diharapkan: \omterm{def:g10:universal-dna:gene}{gen} berjenis \omterm{def:g10:cells-common-unit:organelle}{inti sel}).

\textbf{17.} Dalam \omterm{def:g10:cells-common-unit:organelle}{inti selnya}: seiring waktu sebagian besar \omterm{def:g10:universal-dna:gene}{gen}
bakterinya dipindahkan ke genom \omterm{def:g10:cells-common-unit:organelle}{inti selnya}, dan genom \omterm{def:g10:cells-common-unit:organelle}{organelnya}
sendiri menyusut menjadi sisa.

\textbf{18.} \omterm{def:g10:universal-dna:gene}{Gen} \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya} hanya berpindah dari ibu kepada
anaknya; \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:cells-common-unit:organelle}{mitokondria} seseorang melacak garis ibunya yang tak terputus.

\textbf{19.} \omterm{def:g10:cells-common-unit:organelle}{Mitokondrianya} telah menyerahkan sebagian besar \omterm{def:g10:universal-dna:gene}{gennya}
kepada inangnya dan tidak dapat ada di luar \omterm{def:g10:cells-common-unit:cell}{selnya}; mitra lumut
keraknya menjaga genomnya sendiri dan dapat, dengan susah payah, dipisahkan.

\textbf{20.} Einkorn (14, A), rumput kambing 1 (14, B), rumput kambing
2 (14, D): 42 \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}; ``pengaturan''; dan \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya}.
\end{solution}
