\chapter{श्वसनी गैस विनिमय}\label{ch:b1:gas-exchange}

एक लीटर हवा में एक लीटर पानी से तीस गुना अधिक ऑक्सीजन है, उसका भार आठ सौ
गुना कम है, और उसमें ऑक्सीजन दस हज़ार गुना तेज़ विसरित होती है। किसी
ट्राउट को अपनी ली हुई हर मिलीग्राम ऑक्सीजन के लिए डेढ़ सौ ग्राम पानी अपने
\omterm{def:b1:gas-exchange:gill}{क्लोमों} पर से गुज़ारना पड़ता है, और वह पानी की ढोई ऑक्सीजन का चार-पाँचवाँ
भाग निकाल लेता है; जबकि मनुष्य प्रति मिलीग्राम कुछ ही ग्राम हवा चलाता है
और उसका तीन-चौथाई भाग बरबाद कर देता है। दोनों \omterm{def:b1:mammal-organization:organ}{अंग} एक ही समीकरण के उत्तर
हैं --- यानी किसी सतह के आरपार किसी गैस का फ्लक्स --- पर दो माध्यमों में।
यह अध्याय वह भौतिकी बताता है, यह निकालता है कि किसी \omterm{prop:b1:organism-environment:surfaces}{विनिमय सतह} को कैसा
होना चाहिए, \omterm{def:b1:gas-exchange:gill}{क्लोमों}, फेफड़ों तथा श्वासनलिकाओं का वर्णन करता है, ऑक्सीजन
और कार्बन डाइऑक्साइड का रक्त में पीछा करता है, और देखता है कि संवातन कैसे
नियंत्रित होता है और ऊँचाई पर तथा पानी के नीचे क्या बदलता है।

\section{दो माध्यमों में किसी गैस की भौतिकी}

\begin{definition}[आंशिक दाब, विलेयता]\label{def:b1:gas-exchange:partial}
\index{आंशिक दाब}
गैसों के किसी मिश्रण में हर गैस अपने भिन्न के समानुपाती \emph{आंशिक दाब}
लगाती है: \qty{101}{kPa} की हवा में ऑक्सीजन
(\qty{20.9}{\%}) का $P_{\mathrm{O_2}} = \qty{21.2}{kPa}$ है, और कार्बन
डाइऑक्साइड (\qty{0.04}{\%}) का \qty{0.04}{kPa}। कोई गैस किसी द्रव में
अपने आंशिक दाब के समानुपात में घुलती है (हेनरी का नियम):
\qty{15}{\celsius} पर हवा के साथ साम्य में जल \qty{0.32}{mmol/L}
ऑक्सीजन रखता है (\qty{10}{mg/L}, यानी \qty{7}{mL/L} गैस), जबकि स्वयं
हवा में \qty{8.7}{mmol/L} (\qty{210}{mL/L})। आंशिक दाब --- जो साम्य पर
जल और हवा में एक ही होता है --- वही है जो माध्यमों के बीच विसरण चलाता है;
और मात्रा वह है जिसे कोई साँस लेने वाला निकाल सकता है। कार्बन
डाइऑक्साइड ऑक्सीजन से पच्चीस गुना अधिक घुलनशील है, इसलिए वह कभी विनिमय
के लिए कठिन गैस नहीं होती।
\end{definition}

\begin{theorem}[किसी विनिमय सतह के लिए फ़िक का नियम]\label{thm:b1:gas-exchange:fick}
\index{फ़िक का नियम}
$A$ क्षेत्रफल तथा $x$ मोटाई की कोई झिल्ली $P_1$ और $P_2$ आंशिक दाबों के
दो माध्यमों को अलग करती है; कोई गैस उसे जिस दर से पार करती है वह यह है
\[
\dot V = K\,\frac{A}{x}\,(P_1 - P_2),
\]
जिसमें $K$, यानी झिल्ली का विसरण (क्रो) स्थिरांक, \omterm{def:b1:body-plans-tissues:tissue}{ऊतक} में उस गैस की
विसरणशीलता तथा विलेयता को समेट लेता है। फ्लक्स बड़े क्षेत्रफल से, पतले
अवरोध से और \omterm{def:b1:gas-exchange:partial}{आंशिक दाब} के बड़े अंतर से बढ़ता है; और किसी चीज़ से नहीं।
\end{theorem}
\begin{proof}
झिल्ली में घुली गैस के लिए फ़िक का पहला नियम
(\cref{ch:b1:membranes-transport}) प्रति एकांक क्षेत्रफल ऐसा फ्लक्स देता
है जो सांद्रता की प्रवणता के समानुपाती है; और हेनरी के नियम से हर फलक पर
सांद्रता वहाँ के माध्यम के \omterm{def:b1:gas-exchange:partial}{आंशिक दाब} के समानुपाती है, इसलिए प्रवणता
विलेयता गुणा $(P_1 - P_2)/x$ है, और स्थिरांक विलेयता तथा विसरणशीलता को
सोख लेता है।
\end{proof}

\begin{proposition}[किसी विनिमय सतह को कैसा होना चाहिए]\label{prop:b1:gas-exchange:requirements}
\index{विनिमय सतह}
एक मिलीमीटर से बड़े किसी \omterm{def:b1:organism-environment:organism}{जीव} को उसकी बाहरी सतह से होते विसरण से आपूर्ति
नहीं मिल सकती (\cref{ch:b1:organism-environment})।
उसे ऐसी \emph{श्वसनी सतह} चाहिए जो बड़ी हो ($A$ बढ़ाने को),
पतली ($x$ घटाने को), नम (गैसें झिल्लियाँ केवल घोल में ही पार करती हैं),
\emph{संवातित} --- यानी बाहरी माध्यम नवीकृत होता रहे, ताकि $P_1$ ऊँचा
बना रहे --- और \emph{पोषित} --- यानी उसके पीछे का रक्त नवीकृत होता रहे,
ताकि $P_2$ नीचा बना रहे; और उसे वह परिसंचरण चाहिए जो गैस सतह से \omterm{def:b1:body-plans-tissues:tissue}{ऊतकों} तक
ले जाए। \omterm{def:b1:gas-exchange:gill}{क्लोम}, फेफड़े और श्वासनलिकाएँ ऐसी सतह को किसी शरीर में मोड़ने और
उसके आरपार दोनों तरल चलाने के तीन ढंग हैं।
\end{proposition}

\section{क्लोम और प्रतिधारा}

\begin{definition}[क्लोम]\label{def:b1:gas-exchange:gill}
\index{क्लोम}
\emph{क्लोम} देह-सतह का जल में निकला हुआ उभार है, जो तंतुओं और कुछ
माइक्रोमीटर मोटी \emph{पटलिकाओं} में मुड़ा होता है और जिनमें रक्त
केशिकाओं में बहता है। किसी मछली के क्लोम हर ओर चार अस्थिल चापों से लटकते
हैं; मुँह से भीतर खींचा गया और दो-पंप तंत्र से क्लोम-ढकनों के नीचे से
बाहर धकेला गया जल पटलिकाओं के बीच एक दिशा में बहता है, और पटलिकाओं में
रक्त दूसरी दिशा में: यही \emph{प्रतिधारा}\index{प्रतिधारा विनिमय} है।
एक किलोग्राम की किसी ट्राउट की पटलिका-सतह कोई \qty{2000}{cm^2} है, और
\qty{2}{\micro m} मोटी।
\end{definition}

\begin{theorem}[प्रतिधारा विनिमय]\label{thm:b1:gas-exchange:countercurrent}
जब जल और रक्त किसी \omterm{prop:b1:organism-environment:surfaces}{विनिमय सतह} के सहारे विपरीत दिशाओं में बहते हैं, तब
\omterm{def:b1:gas-exchange:gill}{क्लोम} से निकलता रक्त उस जल के \omterm{def:b1:gas-exchange:partial}{आंशिक दाब} के पास पहुँच सकता है जो उसमें
\emph{घुस} रहा है, और जल की लगभग सारी ऑक्सीजन निकाली जा सकती है: यानी
\qty{80}{\%} या उससे अधिक निष्कर्षण। और जब वे एक ही दिशा में बहते हैं
(सहधारा), तब दोनों तरल किसी साझा मध्यवर्ती मान पर आ मिलते हैं और
निष्कर्षण \qty{50}{\%} से अधिक नहीं हो सकता।
\end{theorem}
\begin{proof}
किसी सहधारा विनिमयक के सहारे अंतर $P_{\text{जल}} -
P_{\text{रक्त}}$ अपने प्रवेश-मान से घटकर शून्य हो जाता है क्योंकि दोनों
एक ही दाब के पास आ जाते हैं; और बराबर होते ही और कोई गैस नहीं चलती,
और बराबर प्रवाहों के लिए मिलन-बिंदु आधे रास्ते पर होता है। \omterm{def:b1:gas-exchange:gill}{प्रतिधारा}
प्रवाह में रक्त, अपने निकास की ओर बढ़ते हुए, हर बार और ताज़ा जल से मिलता
है, इसलिए पूरी लंबाई भर एक धनात्मक अंतर बना रहता है; अपने निकास पर रक्त
\qty{21}{kPa} पर भीतर आते जल के सामने होता है, और अपने निकास पर जल कुछ
किलोपास्कल के भीतर आते शिरा-रक्त के सामने। सीमा सतह तय करती है, कोई
साम्य नहीं।
\end{proof}

\begin{example}[जल की क़ीमत]\label{ex:b1:gas-exchange:priceofwater}
\qty{1}{kg} की कोई विश्राम कर रही ट्राउट घंटे भर में लगभग \qty{50}{mL}
ऑक्सीजन लेती है। \qty{7}{mL/L} के जल से, \qty{80}{\%} निष्कर्षण पर, उसे
घंटे भर में \qty{9}{L} जल अपने \omterm{def:b1:gas-exchange:gill}{क्लोमों} पर से पंप करना पड़ता है --- यानी
हवा से पचास गुना अधिक श्यान माध्यम के नौ किलोग्राम, और वह भी मिलीमीटर के
किसी भिन्न भर चौड़ी नलियों से। उसके उपापचय का दसवाँ भाग साँस लेने में
जाता है; और विश्राम कर रहे किसी मनुष्य का पचासवाँ। तालाब को
\qty{30}{\celsius} तक गरम कीजिए और वही मछली, जिसे कम ऑक्सीजन रखते जल से
अधिक ऑक्सीजन चाहिए, तीन गुना जल पंप करेगी।
\end{example}

\section{हवा में साँस लेना: फेफड़े और श्वासनलिकाएँ}

\begin{definition}[स्तनधारी का फेफड़ा]\label{def:b1:gas-exchange:lung}
\index{फेफड़ा}\index{कूपिका}
\emph{फेफड़ा} देह-सतह का किसी नम भीतरी कक्ष में धँसा हुआ भाग है, जो सूखी
हवा में भी उस सतह को गीली रखता है। स्तनधारी का फेफड़ा वायुमार्गों का वह
वृक्ष है जो कोई तीस करोड़ \emph{कूपिकाओं} पर समाप्त होता है, यानी
\qty{0.2}{mm} आरपार की उन थैलियों पर जिनकी भित्तियाँ --- एक \omterm{def:b1:body-plans-tissues:epithelium}{उपकला} \omterm{def:b1:cell-unit-of-life:cell}{कोशिका}
और एक केशिका अंतःकला, कुल मिलाकर \qty{0.5}{\micro m} --- मनुष्य में
\qty{100}{m^2} बनाती हैं, और जो ऐसे केशिका जाल में लिपटी हैं जिससे पूरा
हृदय-निर्गम गुज़रता है। वह \emph{ज्वारीय} रूप से संवातित होता है:
\omterm{def:b1:mammal-organization:bodyplan}{मध्यपट} और पसली की पेशियाँ वक्ष बड़ा करती हैं, हवा भीतर बहती है; वे ढीली
पड़ती हैं, हवा उसी रास्ते बाहर बहती है। एक साँस \qty{500}{mL} चलाती है,
जिसमें से \qty{150}{mL} केवल वायुमार्ग भरती है (\emph{मृत स्थान}) और
किसी कूपिका तक पहुँचती ही नहीं; और कूपिका की गैस, जो हर साँस पर सातवें
भाग तक नवीकृत होती है, $P_{\mathrm{O_2}} =
\qty{13.3}{kPa}$ और $P_{\mathrm{CO_2}} = \qty{5.3}{kPa}$ पर बनी --- वे दाब
जिनसे रक्त साम्य में आता है।
\end{definition}

\begin{proposition}[हवा के लिए तीन अभिकल्पनाएँ]\label{prop:b1:gas-exchange:designs}
\index{श्वासनलिकाएँ}
\emph{स्तनधारी} ज्वारीय रूप से संवातन करते हैं: सरल है, पर ताज़ी हवा
बासी से मिल जाती है, \omterm{def:b1:gas-exchange:lung}{कूपिका} की ऑक्सीजन वायुमंडल की दो-तिहाई रहती है,
और निष्कर्षण एक-चौथाई। \emph{पक्षी} हवा को दो वायु-कोश समुच्चयों के बीच
की महीन नलियों (परानलिकाओं) से एक ही दिशा में गुज़ारते हैं, और वे कोश
धौंकनी का काम करते हैं, इसलिए ताज़ी हवा अंतःश्वसन तथा निःश्वसन दोनों में
\omterm{prop:b1:organism-environment:surfaces}{विनिमय सतह} से बहती है, और वह भी रक्त के साथ अनुप्रस्थ धारा में: उनका
निष्कर्षण ऊँचा है, और पट्टसिर हंस हिमालय के ऊपर उड़ता है जहाँ कोई
स्तनधारी बेहोश हो जाता। \emph{कीटों} में श्वसनी रक्त होता ही नहीं:
हवा वाल्वों (\emph{श्वासरंध्रों}) से \emph{श्वासनलिकाओं} में घुसती है,
यानी उन नलियों में जो सर्पिल मोटाइयों से कड़ी रखी जाती हैं और जो एक
माइक्रोमीटर तक शाखित होकर हर \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} तक ऑक्सीजन सीधे विसरण से पहुँचाती
हैं, और बड़े या सक्रिय कीटों में उदर के पंपन से भी। नलियों के सहारे विसरण
इस अभिकल्पना को छोटे शरीरों तक सीमित कर देता है --- और यही एक कारण है कि
कोई कीट चूहे से बड़ा नहीं।
\end{proposition}

\begin{method}[किसी संवातन बजट को पढ़ना]\label{met:b1:gas-exchange:budget}
\begin{enumerate}
  \item मिनट संवातन $=$ ज्वारीय आयतन $\times$ प्रति मिनट साँसें;
        \omterm{def:b1:gas-exchange:lung}{कूपिका} संवातन $=$ (ज्वारीय आयतन $-$ मृत स्थान)
        $\times$ प्रति मिनट साँसें। केवल दूसरा गैस विनिमय करता है।
  \item ऑक्सीजन ग्रहण $=$ \omterm{def:b1:gas-exchange:lung}{कूपिका} संवातन $\times$ (भीतर ली गई
        भिन्न $-$ \omterm{def:b1:gas-exchange:lung}{कूपिका} भिन्न); विश्राम में
        $\qty{4.2}{L/min}\times(0.21 - 0.14) \approx \qty{0.3}{L/min}$।
  \item निष्कर्षण $=$ ग्रहण $/$ (संवातन $\times$ भीतर ली गई
        भिन्न): किसी ज्वारीय फेफड़े में एक-चौथाई, और किसी \omterm{def:b1:gas-exchange:gill}{प्रतिधारा}
        \omterm{def:b1:gas-exchange:gill}{क्लोम} में चार-पाँचवाँ।
  \item रक्त की ओर: ग्रहण $=$ हृदय-निर्गम $\times$
        (धमनी $-$ शिरा मात्रा); विश्राम में \qty{5}{L/min}$\times$
        \qty{50}{mL/L}, यानी वही \qty{250}{mL/min}। दोनों
        बहीखाते मेल खाने चाहिए।
\end{enumerate}
\end{method}

\section{रक्त में गैसें}

\begin{proposition}[ऑक्सीजन और कार्बन डाइऑक्साइड का अभिगमन]\label{prop:b1:gas-exchange:transport}
\index{कार्बोनिक ऐनहाइड्रेज़}\index{बोर प्रभाव}\index{हैल्डेन प्रभाव}
प्लाज़्मा धमनी दाब पर प्रति लीटर केवल \qty{3}{mL} ऑक्सीजन घोलता है;
हीमोग्लोबिन (\cref{ch:b1:proteins}) \qty{200}{mL} ढोता है, जो फेफड़े में
\qty{13.3}{kPa} पर लदता है और \omterm{def:b1:body-plans-tissues:tissue}{ऊतकों} में पाँचवें से आधे तक उतार देता है।
कार्बन डाइऑक्साइड तीन तरह से चलती है: \qty{7}{\%}
घुली हुई, \qty{23}{\%} हीमोग्लोबिन के ऐमीनो समूहों से बँधी, और
\qty{70}{\%} बाइकार्बोनेट के रूप में: लाल \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में \emph{\omterm{prop:b1:gas-exchange:transport}{कार्बोनिक
ऐनहाइड्रेज़}} $\mathrm{CO_2}$ और जल को मिलीसेकंडों में कार्बोनिक अम्ल बना
देता है, वह अम्ल वियोजित होता है, बाइकार्बोनेट क्लोराइड के बदले \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} से
बाहर जाता है (\emph{क्लोराइड खिसकाव}), और प्रोटॉन हीमोग्लोबिन ले लेता
है। दोनों गैसें एक-दूसरे की मदद करती हैं: अम्ल और
$\mathrm{CO_2}$ हीमोग्लोबिन की ऑक्सीजन-बंधुता घटा देते हैं (\emph{\omterm{prop:b1:gas-exchange:transport}{बोर
प्रभाव}}), इसलिए ऑक्सीजन वहीं छूटती है जहाँ $\mathrm{CO_2}$ बनती है; और
ऑक्सीजन-रहित हीमोग्लोबिन प्रोटॉन तथा $\mathrm{CO_2}$ बेहतर बाँधता है
(\emph{\omterm{prop:b1:gas-exchange:transport}{हैल्डेन प्रभाव}}), इसलिए $\mathrm{CO_2}$ वहीं ली जाती है जहाँ
ऑक्सीजन छूटती है, और फेफड़े में ऑक्सीजन लदते ही छोड़ दी जाती है।
\end{proposition}

\section{नियंत्रण, ऊँचाई और गोता}

\begin{proposition}[संवातन कार्बन डाइऑक्साइड से नियंत्रित होता है]\label{prop:b1:gas-exchange:control}
\index{रसग्राही}
साँस की लय मस्तिष्क-स्तंभ में बनती है और \emph{रसग्राहियों} से
समायोजित होती है: केंद्रीय रसग्राही, जो मस्तिष्क के चारों ओर के द्रव में
डूबे हैं, उस pH परिवर्तन पर अनुक्रिया करते हैं जो $\mathrm{CO_2}$ पैदा
करती है; और ग्रीवा धमनियों के परिधीय रसग्राही धमनी ऑक्सीजन के लगभग
\qty{8}{kPa} से नीचे गिरने पर। साधारण जीवन में राज
$\mathrm{CO_2}$ का चलता है: धमनी $P_{\mathrm{CO_2}}$ में \qty{1}{kPa}
की वृद्धि संवातन मोटे तौर पर दुगुना कर देती है, जबकि ऑक्सीजन को
एक-तिहाई गिरना पड़ता है तब कहीं संवातन अनुक्रिया करता है। साँस रोकना तब
समाप्त होता है जब $\mathrm{CO_2}$ सीमा पर पहुँचती है, ऑक्सीजन नहीं; और
पहले अतिश्वसन करने से $\mathrm{CO_2}$ गिर जाती है और कोई गोताख़ोर तब तक
नीचे रह जाता है जब तक ऑक्सीजन न चुक जाए --- यानी साँस लेने की तड़प
लौटने से पहले ही बेहोश।
\end{proposition}

\qty{7}{}: ग्रीवा काय संवातन बढ़ा देते हैं, जिससे
$\mathrm{CO_2}$ उड़ जाती है और रक्त क्षारीय हो जाता है --- और यह ब्रेक
गुर्दे दिनों में बाइकार्बोनेट उत्सर्जित करके छोड़ते हैं; लाल \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ}
दिनों में अपना BPG बढ़ा लेती हैं और मज्जा हफ़्तों में हीमोग्लोबिन
(\cref{ch:b1:proteins})। बीस मिनट गोता लगाती कोई सील इसका
उल्टा करती है: वह गोता लगाने से पहले साँस छोड़ देती है, अपना हृदय दसवें
भाग तक धीमा कर लेती है, मस्तिष्क तथा हृदय को छोड़कर बाक़ी सबका रक्त बंद
कर देती है, और अपनी पेशियाँ उनके अपने मायोग्लोबिन से बँधी ऑक्सीजन पर
चलाती है, जो हमारे मायोग्लोबिन से दस गुना अधिक सांद्र है, और फिर \omterm{def:b1:respiration-fermentation:fermentation}{किण्वन}
पर, और उस लैक्टेट को सह लेती है जिसे शरीर सतह पर आकर साफ़ करता है।
उसके फेफड़े \qty{50}{m} से नीचे बैठ जाते हैं, जिससे नाइट्रोजन उसके रक्त
से बाहर रहती है।
\end{example}

डूबने का जोखिम क्यों उठाता है, जबकि जो नहीं करता वह असहज पर होश में,

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:gas-exchange:10}

लेती है। हवा में $D = \qty{2e-5}{m^2/s}$ और श्वासरंध्र पर
\qty{8.7}{mol/m^3} ऑक्सीजन के साथ पेशी वाले सिरे पर सांद्रता निकालिए (फ़िक:
$J = DA\,\Delta c/L$)। \qty{10}{mm} लंबी श्वासनलिका के लिए दोहराइए।
कीटों का आकार क्या सीमित करता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:gas-exchange:11}
पक्षी और मछलियाँ दोनों ऐसे प्रवाह काम में लेते हैं जो ज्वारीय नहीं।
उनके विनिमयकों की तुलना कीजिए (माध्यम की दिशा, रक्त-प्रवाह से संबंध,
निष्कर्षण) और समझाइए कि कोई ज्वारीय \omterm{def:b1:gas-exchange:lung}{फेफड़ा}, उसकी सतह चाहे जितनी हो, उनके
निष्कर्षण तक क्यों नहीं पहुँच सकता।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:gas-exchange:12}
``\omterm{def:b1:gas-exchange:lung}{फेफड़ा} उस गैस से नियमित होता है जिसे वह बाहर करता है, उस गैस से नहीं
जिसे वह भीतर लेता है।'' एक अनुच्छेद में चर्चा कीजिए: समुद्र-तल पर
$\mathrm{CO_2}$ बेहतर संकेत क्यों है, यह व्यवस्था कब चूकती है, और ऊँचाई
उसे कैसे उघाड़ देती है।
\end{exercise}

\section{समस्या: एक ट्राउट और एक मनुष्य}

\begin{problem}[{सप्ताहांत समस्या --- एक क्लोम और एक फेफड़ा आमने-सामने:
प्रति लीटर माध्यम ऑक्सीजन, संवातन और निष्कर्षण, प्रतिधारा तथा ज्वारीय
प्रवाह, और हर एक के पीछे का रक्त, और अंत में प्रतिधारा क्लोम की
निष्कर्षण दक्षता}]
\label{pb:b1:gas-exchange:1}
विश्राम कर रहा कोई मनुष्य प्रति मिनट \qty{250}{mL} ऑक्सीजन लेता है, और
मिनट में \qty{12}{} बार साँस लेता है जिसका ज्वारीय आयतन \qty{500}{mL} और
मृत स्थान \qty{150}{mL} है; भीतर ली गई हवा \qty{21}{\%} ऑक्सीजन है,
\omterm{def:b1:gas-exchange:lung}{कूपिका} गैस \qty{14}{\%}; हृदय-निर्गम \qty{5}{L/min}, धमनी
ऑक्सीजन-मात्रा \qty{200}{mL/L}। \qty{1}{kg} की कोई विश्राम कर रही ट्राउट
\qty{15}{\celsius} के उस जल में, जो \qty{7}{mL/L} रखता है, प्रति घंटा
\qty{50}{mL} ऑक्सीजन लेती है; उसका हीमोग्लोबिन संतृप्त होने पर
\qty{100}{mL/L} ढोता है; उसका रक्त \omterm{def:b1:gas-exchange:gill}{क्लोमों} से \qty{95}{\%} संतृप्त निकलता
है और \qty{15}{\%} संतृप्त लौटता है। जल हवा से \qty{800}{} गुना सघन है।

\medskip
\noindent\textbf{भाग I --- दो माध्यम।}
\begin{enumerate}
  \item हवा की और ट्राउट के जल की ऑक्सीजन-मात्रा मिलीमोल प्रति लीटर में
        लिखिए (\qty{22.4}{L/mol})।
  \item दोनों मात्राओं का अनुपात निकालिए।
  \item हर स्थिति में उस माध्यम का द्रव्यमान निकालिए (हवा \qty{1.2}{g/L};
        जल \qty{1000}{g/L}) जो \qty{1}{mmol} ऑक्सीजन रखता है।
  \item ट्राउट के जल का वह आयतन निकालिए जो एक लीटर हवा जितनी ऑक्सीजन
        रखता है।
  \item एक वाक्य में समझाइए कि मछली के \omterm{def:b1:gas-exchange:gill}{क्लोमों} का संवातन एक ही दिशा में
        क्यों होना चाहिए और फेफड़े ज्वारीय क्यों हो सकते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{def:b1:gas-exchange:lung}{फेफड़ा}।}
\begin{enumerate}[resume]
  \item मिनट संवातन और \omterm{def:b1:gas-exchange:lung}{कूपिका} संवातन निकालिए।
  \item हर साँस का कितना भिन्न मृत स्थान है, और इसलिए मिनट संवातन का
        कितना भिन्न बरबाद जाता है?
  \item प्रति मिनट \omterm{def:b1:gas-exchange:lung}{कूपिकाओं} तक पहुँचाई गई ऑक्सीजन और ली गई ऑक्सीजन
        निकालिए (\omterm{def:b1:gas-exchange:lung}{कूपिका} संवातन $\times$ भिन्नों का
        अंतर)। \qty{250}{mL/min} से जाँचिए।
  \item निष्कर्षण दक्षता निकालिए: ग्रहण को भीतर ली गई मिनट संवातन की
        ऑक्सीजन से भाग देकर।
  \item \omterm{def:b1:gas-exchange:lung}{कूपिका} भिन्न से \omterm{def:b1:gas-exchange:lung}{कूपिका} $P_{\mathrm{O_2}}$ निकालिए (कुल दाब
        \qty{101}{kPa}, और पहले जलवाष्प के \qty{6.3}{kPa} घटाइए)।
  \item रक्त की ओर, \qty{5}{L/min} पर \qty{250}{mL/min} के लिए चाहिए
        धमनी-शिरा अंतर, तथा शिरा-मात्रा और संतृप्ति निकालिए।
  \item व्यायाम में ग्रहण बढ़कर \qty{3}{L/min} हो जाता है और
        हृदय-निर्गम \qty{20}{L/min}। धमनी-शिरा अंतर और शिरा-संतृप्ति
        निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:b1:gas-exchange:gill}{क्लोम}।}
\begin{enumerate}[resume]
  \item ट्राउट का ऑक्सीजन ग्रहण मिलीलीटर प्रति मिनट में निकालिए।
  \item \qty{80}{\%} निष्कर्षण (\omterm{def:b1:gas-exchange:gill}{प्रतिधारा}) के साथ प्रति मिनट और प्रति
        घंटा संवातित जल निकालिए।
  \item उतनी ही सतह का कोई सहधारा विनिमयक जो निष्कर्षण देता
        (\qty{43}{\%}) उसके साथ चाहिए जल निकालिए। \omterm{def:b1:gas-exchange:gill}{प्रतिधारा} क्या बचा
        देती है?
  \item प्रति घंटा पंप किए गए जल का द्रव्यमान निकालिए, और उसकी तुलना
        प्रति घंटा किसी मनुष्य की चलाई हवा के द्रव्यमान से कीजिए।
  \item ट्राउट का ऑक्सीजन-मात्रा में धमनी-शिरा अंतर और उसके ग्रहण के
        लिए चाहिए हृदय-निर्गम निकालिए।
  \item \omterm{def:b1:gas-exchange:gill}{क्लोम} पर जल-प्रवाह का रक्त-प्रवाह से अनुपात, और मानव फेफड़े पर
        वायु-प्रवाह का रक्त-प्रवाह से अनुपात (\omterm{def:b1:gas-exchange:lung}{कूपिका} संवातन बटा
        हृदय-निर्गम) निकालिए। टिप्पणी कीजिए।
  \item जल पंप करने में ट्राउट को उसके ऑक्सीजन ग्रहण का लगभग
        \qty{10}{\%} लगता है। यदि जल गरम होकर \qty{25}{\celsius}
        (\qty{5.5}{mL/L}) हो जाए और उसका ग्रहण आधा और बढ़ जाए, तो
        संवातन कितने गुणक से बढ़ना चाहिए, और साँस लेने पर ख़र्च भिन्न का
        क्या होता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- स्वयं विनिमयक।}
\begin{enumerate}[resume]
  \item \omterm{def:b1:gas-exchange:gill}{प्रतिधारा} \omterm{def:b1:gas-exchange:gill}{क्लोम} में रक्त \qty{18}{kPa} पर निकलता है और
        \qty{21}{kPa} पर भीतर आते जल के सामने होता है, और जल
        \qty{4}{kPa} पर निकलता है तथा \qty{3}{kPa} पर भीतर आते रक्त के
        सामने। हर सिरे पर दाबांतर निकालिए।
  \item किसी सहधारा विनिमयक में दोनों तरल \qty{12}{kPa} पर निकलते हैं।
        हर सिरे पर अंतर निकालिए और समझाइए कि जल के चुकने से पहले ही
        विनिमय क्यों रुक जाता है।
  \item मानव \omterm{def:b1:gas-exchange:lung}{फेफड़ा}: \omterm{def:b1:gas-exchange:lung}{कूपिका} \qty{13.3}{kPa}, आता शिरा-रक्त
        \qty{5.3}{kPa}, निकलता धमनी-रक्त \qty{13.3}{kPa}।
        कोई ज्वारीय \omterm{def:b1:gas-exchange:lung}{फेफड़ा} किस व्यवस्था जैसा है, और उसका धमनी-रक्त
        \omterm{def:b1:gas-exchange:lung}{कूपिका} के मान से आगे क्यों नहीं जा सकता?
  \item किसी ट्राउट की पटलिकाएँ मिलकर \qty{2000}{cm^2} हैं और
        \qty{2}{\micro m} मोटी; और किसी मनुष्य की \omterm{def:b1:gas-exchange:lung}{कूपिकाएँ}
        \qty{100}{m^2} तथा \qty{0.5}{\micro m}।
        दोनों का $A/x$ अनुपात, और उनके ऑक्सीजन ग्रहणों का अनुपात
        निकालिए। यह तुलना किसी \omterm{def:b1:gas-exchange:lung}{कूपिका} भित्ति के सापेक्ष किसी पटलिका के
        आरपार माध्य दाबांतर के बारे में क्या सुझाती है?
  \item समझाइए कि जल इतनी कम ऑक्सीजन रखते हुए भी किसी मछली के लिए
        कार्बन डाइऑक्साइड कभी सीमक क्यों नहीं होती।
  \item परिणाम बताइए: ट्राउट के \omterm{def:b1:gas-exchange:gill}{क्लोम} की और मनुष्य के फेफड़े की
        निष्कर्षण दक्षता, और \omterm{def:b1:gas-exchange:gill}{क्लोम} के वे दो लक्षण जो यह अंतर बनाते हैं।
