\chapter{Air dan Biomolekul Kecil}\label{ch:b1:water-small-molecules}

Sebuah bakteri bervolume satu mikrometer kubik memuat sekitar dua puluh
miliar molekul air dan beberapa ratus juta molekul selebihnya. Air
bukanlah latar belakang \omterm{def:b1:cell-unit-of-life:cell}{sel} melainkan mediumnya: ia melarutkan ion dan
gula, ia menyembunyikan bagian berminyak protein dan membran, ia
menetapkan keasaman yang menjadi sandaran tiap enzim, dan ia adalah
pereaksi atau hasil pada separuh reaksi metabolisme. Bab ini memerikan
air dan ikatan lemah yang dibentuk serta diputuskannya, aritmetika asam,
\omterm{def:b1:water-small-molecules:ph}{basa} dan \omterm{def:b1:water-small-molecules:buffer}{penyangga}, molekul organik kecil yang membangun makromolekul
empat bab berikutnya, dan \omterm{def:b1:water-small-molecules:atp}{mata uang energi} --- \omterm{def:b1:water-small-molecules:atp}{ATP} --- yang dipakai \omterm{def:b1:cell-unit-of-life:cell}{sel}
untuk membayar pembangunannya.

\section{Air}

\begin{definition}[Molekul air]\label{def:b1:water-small-molecules:water}
\index{air}
Air, $\mathrm{H_2O}$, adalah molekul bengkok (sudut H--O--H-nya
$104.5^\circ$) yang oksigennya menarik elektron bersama ke arah
dirinya: ia bersifat \emph{polar}\index{molekul polar}, dengan muatan
negatif parsial pada oksigennya dan muatan positif parsial pada
hidrogennya. Tiap molekul dapat membentuk sampai empat \emph{ikatan
hidrogen}\index{ikatan hidrogen} --- tarikan elektrostatik antara sebuah
hidrogen yang terikat pada atom elektronegatif (O, N) dan sepasang
elektron bebas atom elektronegatif yang lain --- dua lewat hidrogennya,
dua lewat oksigennya. Air cair adalah jala ikatan semacam itu, yang
masing-masing bertahan beberapa pikosekon.
\end{definition}

\begin{proposition}[Sifat air yang penting bagi kehidupan]\label{prop:b1:water-small-molecules:properties}
Jala \omterm{def:b1:water-small-molecules:water}{ikatan hidrogennya} menerangkan sifat air yang luar biasa:
\begin{itemize}
  \item \emph{kelekatan} dan tegangan permukaan yang tinggi
        (\qty{72}{mN/m}): kolom air dapat ditarik naik sebatang pohon
        tanpa putus (\cref{ch:b1:plant-transport}); serangga berjalan
        di atas kolam;
  \item \emph{kapasitas kalor} yang tinggi
        (\qty{4.18}{kJ.kg^{-1}.K^{-1}}) dan kalor penguapan yang tinggi
        (\qty{2.4}{kJ/g}): suhu \omterm{def:b1:organism-environment:organism}{organisme} dan badan air berubah
        perlahan, dan menguapkan segram keringat membuang \qty{2.4}{kJ};
  \item \emph{wujud padat yang massa jenisnya lebih kecil daripada
        cairnya}: es mengapung, dan danau membeku dari atas ke bawah,
        sehingga air cair tertinggal di bawahnya;
  \item \emph{pelarut} yang sangat baik bagi ion dan \omterm{def:b1:water-small-molecules:water}{molekul polar},
        yang dikelilinginya dengan selubung terarah (\emph{hidrasi}),
        dan pelarut yang buruk bagi molekul nonpolar, yang justru
        dipaksanya berdempet (\emph{efek hidrofob}\index{efek hidrofob}).
\end{itemize}
\end{proposition}
\begin{proof}[Bukti sebagian]
Kelekatan dan tegangan permukaan mengukur kerja memisahkan molekul yang
masing-masing dipegang empat ikatan sekitar \qty{20}{kJ/mol}. Memanaskan
air harus memutus ikatan sekaligus mengguncang molekulnya, karena itulah
kapasitas kalornya besar; menguapkannya harus memutus semuanya. Es
memegang tiap molekul dalam kisi tetrahedral terbuka berisi empat
ikatan, yang menempati volume lebih besar daripada cairan yang tak
teratur, tempat molekulnya berkemas lebih rapat. Ion dan gugus polar
dimantapkan oleh \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen} ke air; sebuah molekul nonpolar tak
sanggup membentuknya, dan air di sekelilingnya harus menata diri menjadi
sangkar, dengan biaya entropi yang berkurang ketika molekul nonpolar
bergerombol dan berbagi satu sangkar.
\end{proof}

\begin{example}[Efek hidrofob bekerja]\label{ex:b1:water-small-molecules:hydrophobic}
Kocoklah minyak ke dalam air dan dalam beberapa menit tetesnya telah
menyatu: air memeras keluar apa yang tak sanggup diikatnya. Efek yang
sama melipat sebuah protein (residu berminyaknya bersembunyi di dalam,
\cref{ch:b1:proteins}), merakit sebuah membran (ekor lipidnya berkumpul
menjauhi air, \cref{ch:b1:lipids}), dan mendorong dua permukaan molekul
yang berpasangan agar bertemu. Tak ada yang menarik minyak ke minyak;
airlah yang mendorong.
\end{example}

\section{Ikatan lemah}

\begin{definition}[Ikatan kovalen dan nonkovalen]\label{def:b1:water-small-molecules:bonds}
Sebuah \emph{ikatan kovalen}\index{ikatan kovalen} berbagi sepasang
elektron antara dua atom dan menuntut \qtyrange{300}{500}{kJ/mol} untuk
diputus: kerangka molekul bersifat kovalen, dan hanya enzim yang
membentuk atau memutusnya di dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}. \emph{Ikatan
nonkovalen}\index{ikatan nonkovalen} adalah tarikan yang lemah dan
berbalikan yang memegang molekul \emph{terhadap satu sama lain} dan
membentuknya: \emph{\omterm{def:b1:water-small-molecules:water}{ikatan hidrogen}} (\qtyrange{4}{20}{kJ/mol} dalam
air); \emph{ikatan ion}\index{ikatan ion} antara muatan yang berlawanan
(sekitar \qty{12}{kJ/mol} dalam air, yang tetapan dielektriknya
menyekat muatan delapan puluh kali lipat); tarikan \emph{van der
Waals}\index{gaya van der Waals} antara dua atom mana pun yang
bersentuhan rapat (\qtyrange{1}{4}{kJ/mol} masing-masing, tetapi
terjumlahkan atas seluruh permukaan); dan \emph{\omterm{prop:b1:water-small-molecules:properties}{efek hidrofob}}, yang
bukan sebuah ikatan melainkan pengucilan permukaan nonpolar oleh air.
\end{definition}

\begin{proposition}[Mengapa ikatan lemah adalah ikatan biologi]\label{prop:b1:water-small-molecules:weak}
Energi termal pada \qty{37}{\celsius} adalah $RT = \qty{2.6}{kJ/mol}$.
Sebuah ikatan lemah tunggal hanya beberapa kali nilai itu dan putus
dalam mikrosekon; sepuluh ikatan bersama-sama, pada dua permukaan yang
berpasangan, bertahan berjam-jam. \emph{Pengenalan}\index{pengenalan molekul}
molekul --- sebuah enzim bagi substratnya, sebuah antibodi bagi
antigennya, sebuah untai DNA bagi pasangannya --- bersandar pada banyak
ikatan lemah yang terbentuk sekaligus antara permukaan yang saling
melengkapi, dan karena itulah ia sekaligus khas (hanya permukaan yang
cocok membentuk semuanya) dan berbalikan (kalor, atau seorang pesaing,
membatalkannya). \omterm{def:b1:water-small-molecules:bonds}{Ikatan kovalen}, yang seratus kali lebih kuat, akan
bersifat tetap.
\end{proposition}

\begin{example}[Melelehkan sebuah heliks ganda]\label{ex:b1:water-small-molecules:melting}
Kedua untai DNA dipegang dua atau tiga \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen} tiap pasangan
\omterm{def:b1:water-small-molecules:ph}{basa} ditambah penumpukan (\omterm{def:b1:water-small-molecules:bonds}{van der Waals}) dari pasangan yang
bertetangga dengannya;
sepasang tunggal akan terpisah seketika, tetapi seribu pasangan
berderet bertahan sampai suhunya mencapai sekitar \qty{90}{\celsius},
lalu bersatu kembali, tepat pada susunan yang sama, ketika suhunya
diturunkan (\cref{ch:b1:nucleic-acids}). Khas, kuat karena jumlahnya,
berbalikan.
\end{example}

\section{Asam, basa dan penyangga}

\begin{definition}[pH, asam, basa, p$K_a$]\label{def:b1:water-small-molecules:ph}
Air terurai sedikit, $\mathrm{H_2O} \rightleftharpoons
\mathrm{H^+} + \mathrm{OH^-}$, dengan $[\mathrm{H^+}][\mathrm{OH^-}] =
K_w = \qty{e-14}{mol^2/L^2}$ pada \qty{25}{\celsius}. Yang disebut
\emph{pH}\index{pH} adalah $-\log_{10}[\mathrm{H^+}]$: 7 bagi air murni,
lebih rendah bagi asam, lebih tinggi bagi basa. Sebuah
\emph{asam}\index{asam} $\mathrm{HA}$ melepas sebuah proton,
$\mathrm{HA} \rightleftharpoons \mathrm{H^+} +
\mathrm{A^-}$, dengan tetapan penguraian $K_a =
[\mathrm{H^+}][\mathrm{A^-}]/[\mathrm{HA}]$ dan
\emph{p$K_a$}\index{pKa@p$K_a$} $= -\log_{10}K_a$; \emph{basa}\index{basa}
konjugatnya $\mathrm{A^-}$ menerima satu proton. Asam kuat (HCl) terurai
seluruhnya; asam di dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} (asam karboksilat, fosfat, amonium)
bersifat lemah, dengan p$K_a$ antara 2 dan 10.
\end{definition}

\begin{theorem}[Henderson--Hasselbalch]\label{thm:b1:water-small-molecules:hh}
\index{persamaan Henderson--Hasselbalch}
Bagi sebuah asam lemah dalam larutan,
\[
\mathrm{pH} = \mathrm{p}K_a + \log_{10}\frac{[\mathrm{A^-}]}{[\mathrm{HA}]} .
\]
Pada $\mathrm{pH} = \mathrm{p}K_a$ asamnya setengah terurai; satu satuan
\omterm{def:b1:water-small-molecules:ph}{pH} di atasnya, ia \qty{91}{\%} terurai; satu satuan di bawahnya, \qty{9}{\%}.
\end{theorem}
\begin{proof}
Ambil logaritma $K_a = [\mathrm{H^+}][\mathrm{A^-}]/[\mathrm{HA}]$:
$\log K_a = \log[\mathrm{H^+}] + \log([\mathrm{A^-}]/[\mathrm{HA}])$,
yakni $-\mathrm{p}K_a = -\mathrm{pH} + \log([\mathrm{A^-}]/[\mathrm{HA}])$.
Rasionya adalah $1$, $10$ dan $0.1$ pada \omterm{def:b1:water-small-molecules:ph}{pH} $= \mathrm{p}K_a$, $\mathrm{p}K_a + 1$
dan $\mathrm{p}K_a - 1$.
\end{proof}

\begin{definition}[Penyangga]\label{def:b1:water-small-molecules:buffer}
Yang disebut \emph{penyangga}\index{penyangga} adalah larutan yang memuat
sebuah asam lemah dan \omterm{def:b1:water-small-molecules:ph}{basa} konjugatnya dalam jumlah yang sebanding; ia
menahan perubahan \omterm{def:b1:water-small-molecules:ph}{pH}, sebab $\mathrm{H^+}$ yang ditambahkan diserap
$\mathrm{A^-}$ dan $\mathrm{OH^-}$ yang ditambahkan diserap
$\mathrm{HA}$. Ia bekerja paling baik dalam satu satuan dari p$K_a$-nya,
dan \emph{kapasitasnya} tumbuh bersama kepekatan total pasangan itu.
Penyangga \omterm{def:b1:cell-unit-of-life:cell}{sel}: fosfat ($\mathrm{H_2PO_4^-}/\mathrm{HPO_4^{2-}}$, p$K_a$
6.8) dan rantai samping protein di dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}; bikarbonat
($\mathrm{CO_2}/\mathrm{HCO_3^-}$, p$K_a$ efektif 6.1) dalam darah, yang
pasangan asamnya adalah gas yang dapat disingkirkan paru.
\end{definition}

\begin{method}[Aritmetika penyangga]\label{met:b1:water-small-molecules:bufferarith}
\begin{enumerate}
  \item Kenali pasangan asam--basanya dan p$K_a$-nya pada suhu kerja.
  \item Terapkan Henderson--Hasselbalch untuk memperoleh rasio \omterm{def:b1:water-small-molecules:ph}{basa}
        terhadap asam dari \omterm{def:b1:water-small-molecules:ph}{pH-nya}, atau \omterm{def:b1:water-small-molecules:ph}{pH-nya} dari kepekatannya.
  \item Untuk menambahkan asam kuat: kurangkan jumlahnya dari \omterm{def:b1:water-small-molecules:ph}{basanya}
        dan tambahkan pada asamnya, lalu hitung ulang \omterm{def:b1:water-small-molecules:ph}{pH-nya}. Untuk
        menambahkan \omterm{def:b1:water-small-molecules:ph}{basa} kuat: kebalikannya.
  \item Bila salah satu pasangannya dapat meninggalkan sistemnya
        (karbon dioksida yang diembuskan, amonia yang diekskresikan),
        sistemnya bersifat \emph{terbuka}: tahanlah pasangan itu pada
        nilai yang dikenakan alih-alih membiarkannya menumpuk.
        \omterm{def:b1:water-small-molecules:buffer}{Penyangga} terbuka menahan \omterm{def:b1:water-small-molecules:ph}{pH} jauh lebih baik daripada yang
        tertutup.
\end{enumerate}
\end{method}

\begin{example}[Darah]\label{ex:b1:water-small-molecules:blood}
Darah arteri memegang \qty{24}{mmol/L} bikarbonat dan $\mathrm{CO_2}$
terlarut pada \qty{1.2}{mmol/L} (yang ditetapkan paru): $\mathrm{pH} =
6.1 + \log(24/1.2) = 6.1 + 1.30 = 7.4$. Menambahkan \qty{5}{mmol/L}
asam mengubah \qty{5}{mmol/L} bikarbonat menjadi $\mathrm{CO_2}$; bila
paru mengembuskan $\mathrm{CO_2}$ itu dan menahan nilai terlarutnya,
\omterm{def:b1:water-small-molecules:ph}{pH-nya} menjadi $6.1 + \log(19/1.2) = 7.30$; dalam labu tertutup
$\mathrm{CO_2}$-nya akan naik ke \qty{6.2}{mmol/L} dan \omterm{def:b1:water-small-molecules:ph}{pH-nya} turun ke
$6.1 + \log(19/6.2) =
6.59$. Parulah yang membuat perbedaannya.
\end{example}

\section{Biomolekul kecil}

\begin{definition}[Gugus fungsi]\label{def:b1:water-small-molecules:groups}
Molekul organik \omterm{def:b1:cell-unit-of-life:cell}{sel} adalah kerangka karbon yang membawa sehimpunan kecil
\emph{gugus fungsi}\index{gugus fungsi} yang menentukan kimianya:
\begin{center}
\small
\begin{tabular}{llll}
\hline
gugus & rumus & sifat & terdapat pada \\
\hline
hidroksil & $-\mathrm{OH}$ & polar, berikatan H & gula, alkohol \\
karbonil & $>\!\mathrm{C{=}O}$ & polar, reaktif & aldosa, ketosa \\
karboksil & $-\mathrm{COOH}$ & asam (p$K_a \approx 4$) & asam lemak, asam amino \\
amino & $-\mathrm{NH_2}$ & \omterm{def:b1:water-small-molecules:ph}{basa} (p$K_a \approx 9$) & asam amino \\
fosfat & $-\mathrm{OPO_3^{2-}}$ & bermuatan, tautan kaya energi & nukleotida, \omterm{def:b1:water-small-molecules:atp}{ATP} \\
sulfhidril & $-\mathrm{SH}$ & membentuk jembatan S--S & sisteina \\
metil & $-\mathrm{CH_3}$ & nonpolar & lipid, tanda DNA \\
\hline
\end{tabular}
\end{center}
Pada \omterm{def:b1:water-small-molecules:ph}{pH} \omterm{def:b1:cell-unit-of-life:cell}{sel} gugus karboksil berada dalam bentuk ion ($-\mathrm{COO^-}$)
dan gugus amino terprotonasi ($-\mathrm{NH_3^+}$).
\end{definition}

\begin{proposition}[Empat keluarga satuan pembangun]\label{prop:b1:water-small-molecules:families}
Hampir seluruh materi organik sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} dibangun dari empat macam
molekul kecil, masing-masing beberapa ratus dalton: \emph{gula}
(\cref{ch:b1:carbohydrates}), \emph{asam lemak} (\cref{ch:b1:lipids}),
\emph{asam amino} (\cref{ch:b1:proteins}) dan \emph{nukleotida}
(\cref{ch:b1:nucleic-acids}). Tiga di antaranya disambung menjadi
\emph{polimer} --- polisakarida, protein, asam nukleat --- lewat
\emph{kondensasi}\index{kondensasi} (sebuah ikatan yang terbentuk
dengan lepasnya sebuah molekul air), dan dibongkar lewat
\emph{hidrolisis}\index{hidrolisis} (ikatannya diputus dengan
menambahkan air). \omterm{prop:b1:water-small-molecules:families}{Kondensasi} bersifat mendaki dan dibayar \omterm{def:b1:water-small-molecules:atp}{ATP};
\omterm{prop:b1:water-small-molecules:families}{hidrolisis} bersifat menurun dan hanya memerlukan sebuah enzim.
\end{proposition}

\section{Energi: energi bebas dan ATP}

\begin{proposition}[Energi bebas sebuah reaksi]\label{prop:b1:water-small-molecules:gibbs}
\index{energi bebas}
Sebuah reaksi pada suhu dan tekanan tetap berlangsung spontan ke arah
yang menurunkan \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} Gibbs $G$. Bagi
$\mathrm{A} + \mathrm{B} \rightleftharpoons \mathrm{C} + \mathrm{D}$,
\[
\Delta G = \Delta G^{\circ\prime} + RT\ln\frac{[\mathrm{C}][\mathrm{D}]}{[\mathrm{A}][\mathrm{B}]},
\]
dengan $\Delta G^{\circ\prime}$ adalah perubahan \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} baku
(semua kepekatan \qty{1}{mol/L}, \omterm{def:b1:water-small-molecules:ph}{pH} 7) dan suku logaritmanya mengoreksi
kepekatan yang sesungguhnya. Pada kesetimbangan $\Delta G = 0$,
sehingga $\Delta G^{\circ\prime} = -RT\ln K'_{\text{eq}}$: tiap
\qty{5.7}{kJ/mol} $\Delta G^{\circ\prime}$ menggeser tetapan
kesetimbangannya sepuluh kali lipat. $\Delta G$ menyatakan ke arah mana
sebuah reaksi dapat berjalan, bukan secepat apa: itu urusan enzimnya
(\cref{ch:b1:enzymes}).
\end{proposition}
\begin{proof}
Potensial kimia tiap spesies adalah $\mu = \mu^\circ + RT\ln c$ (hasil
larutan ideal pada mata kuliah fisika); $\Delta G$ adalah jumlah
potensial hasil reaksinya dikurangi jumlah potensial pereaksinya, yang
memberikan rumus tadi; menyamakannya dengan nol pada kesetimbangan
memberikan hubungannya dengan
$K'_{\text{eq}}$.
\end{proof}

\begin{definition}[ATP, penggandengan]\label{def:b1:water-small-molecules:atp}
\emph{ATP}\index{ATP} (adenosina trifosfat) adalah sebuah nukleotida
(\cref{ch:b1:nucleic-acids}) yang membawa tiga fosfat berderet; kedua
ikatan terluarnya adalah ikatan
\emph{fosfoanhidrida}\index{ikatan fosfoanhidrida} yang \omterm{prop:b1:water-small-molecules:families}{hidrolisisnya},
$\mathrm{ATP} + \mathrm{H_2O} \to \mathrm{ADP} +
\mathrm{P_i}$, memiliki $\Delta G^{\circ\prime} = \qty{-30.5}{kJ/mol}$
dan, pada kepekatan sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}, $\Delta G \approx \qty{-50}{kJ/mol}$.
ATP adalah \emph{mata uang energi}\index{mata uang energi} \omterm{def:b1:cell-unit-of-life:cell}{sel}:
katabolisme membuatnya (\cref{ch:b1:respiration-fermentation}), dan
biosintesis, pengangkutan serta gerak membelanjakannya. Sebuah reaksi
yang mendaki dibuat berjalan dengan
\emph{menggandengkannya}\index{penggandengan energi} pada \omterm{prop:b1:water-small-molecules:families}{hidrolisis}
ATP lewat sebuah zat antara bersama, sehingga jumlah kedua $\Delta G$
menjadi negatif.
\end{definition}

\begin{example}[Mengapa hidrolisis ATP melepas begitu banyak]\label{ex:b1:water-small-molecules:whyatp}
Ada tiga sebab: keempat muatan negatif fosfat \omterm{def:b1:water-small-molecules:atp}{ATP} saling menolak dan
kelegaannya diperoleh lewat pembelahannya; hasilnya, $\mathrm{ADP}$ dan
$\mathrm{P_i}$, lebih dimantapkan resonansi dan hidrasi daripada
anhidridanya; dan \omterm{def:b1:cell-unit-of-life:cell}{sel} menjaga \omterm{def:b1:water-small-molecules:atp}{ATP} seribu kali di atas kesetimbangannya
dengan ADP dan fosfat. Yang terakhir adalah suku terbesarnya: dengan
\omterm{def:b1:water-small-molecules:atp}{ATP} pada \qty{5}{mmol/L}, ADP \qty{1}{mmol/L}, $\mathrm{P_i}$
\qty{10}{mmol/L},
$\Delta G = -30.5 + 2.58\ln\frac{10^{-3}\times 10^{-2}}{5\times 10^{-3}}
= -30.5 - 16 = \qty{-46.5}{kJ/mol}$.
\end{example}

\begin{example}[Penggandengan]\label{ex:b1:water-small-molecules:coupling}
Glukosa $+\ \mathrm{P_i} \to$ glukosa-6-fosfat $+\ \mathrm{H_2O}$
memiliki $\Delta G^{\circ\prime} = \qty{+13.8}{kJ/mol}$: pada
kesetimbangan hampir tak ada glukosa yang terfosforilasi. Heksokinase
justru memindahkan fosfatnya langsung dari \omterm{def:b1:water-small-molecules:atp}{ATP}: glukosa $+$ \omterm{def:b1:water-small-molecules:atp}{ATP} $\to$
glukosa-6-fosfat $+$ ADP, $\Delta G^{\circ\prime} = 13.8 - 30.5 =
\qty{-16.7}{kJ/mol}$, tetapan kesetimbangannya $800$, reaksinya pada
dasarnya lengkap. Fosfatnya tak pernah muncul bebas: kedua reaksi itu
satu.
\end{example}

\section{Soal: Darah sebagai Penyangga}

\begin{problem}[{Soal akhir pekan --- lima liter darah pada pH 7.40
sepanjang sebuah lari cepat: bikarbonat, karbon dioksida, paru dan
ginjal, berakhir pada geseran pH bagi sebuah beban laktat tertentu}]
\label{pb:b1:water-small-molecules:1}
Darah arteri: \omterm{def:b1:water-small-molecules:ph}{pH} 7.40, bikarbonat $[\mathrm{HCO_3^-}] =
\qty{24}{mmol/L}$, karbon dioksida terlarut $[\mathrm{CO_2}] =
\qty{1.2}{mmol/L}$ (sebanding dengan tekanan parsialnya: tekanan
parsial \qty{40}{mmHg} memberikan \qty{1.2}{mmol/L}). Sistem
bikarbonatnya memiliki p$K_a$ efektif 6.1: $\mathrm{CO_2} + \mathrm{H_2O}
\rightleftharpoons \mathrm{H^+} + \mathrm{HCO_3^-}$. Volume darahnya
\qty{5}{L}. Protein plasma dan hemoglobinnya menambahkan \omterm{def:b1:water-small-molecules:buffer}{penyangga}
kedua berkapasitas \qty{25}{mmol} $\mathrm{H^+}$ tiap liter tiap satuan
\omterm{def:b1:water-small-molecules:ph}{pH}. Sebuah lari cepat melepaskan \qty{60}{mmol} asam laktat (asam kuat
pada \omterm{def:b1:water-small-molecules:ph}{pH} darah) ke dalam darah dalam waktu semenit.

\medskip
\noindent\textbf{Bagian I --- Keadaan istirahatnya.}
\begin{enumerate}
  \item Periksalah \omterm{def:b1:water-small-molecules:ph}{pH} darah arteri dari nilai bikarbonat dan karbon
        dioksidanya.
  \item Hitunglah $[\mathrm{H^+}]$ pada \omterm{def:b1:water-small-molecules:ph}{pH} 7.40, dalam nanomol tiap
        liter.
  \item Berapa banyak proton bebas yang ada dalam seluruh volume
        darahnya? Bandingkan dengan \qty{120}{mmol} bikarbonatnya.
  \item Hitunglah rasio bikarbonat terhadap karbon dioksida terlarut
        dan fraksi pasangan \omterm{def:b1:water-small-molecules:buffer}{penyangganya} yang hadir sebagai \omterm{def:b1:water-small-molecules:ph}{basa}.
  \item Darah vena memiliki tekanan parsial $\mathrm{CO_2}$ sebesar
        \qty{46}{mmHg} dan \qty{25}{mmol/L} bikarbonat. Hitunglah
        \omterm{def:b1:water-small-molecules:ph}{pH}.
  \item Mengapa p$K_a$ 6.1, yang lebih dari satu satuan di bawah \omterm{def:b1:water-small-molecules:ph}{pH}
        darah, bukan sebuah kelemahan bagi \omterm{def:b1:water-small-molecules:buffer}{penyangga} ini di dalam
        tubuh, meskipun ia akan menjadi kelemahan di dalam sebuah labu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Lari cepatnya, sistem tertutup.}
Andaikan mula-mula tak ada karbon dioksida yang boleh keluar: tiap
proton yang diserap bikarbonat menjadi $\mathrm{CO_2}$ terlarut.
\begin{enumerate}[resume]
  \item Hitunglah kepekatan asam laktat yang ditambahkan ke darahnya.
  \item Hitunglah bikarbonat baru dan $\mathrm{CO_2}$ terlarut yang baru
        bila seluruh asamnya disangga bikarbonat.
  \item Hitunglah \omterm{def:b1:water-small-molecules:ph}{pH} yang dihasilkannya.
  \item Hitunglah \omterm{def:b1:water-small-molecules:ph}{pH} yang akan diberikan asam yang sama dalam
        \qty{5}{L} air murni.
  \item Apakah sistem bikarbonat yang tertutup itu \omterm{def:b1:water-small-molecules:buffer}{penyangga} yang baik
        bagi beban ini? Nyatakan dengan angka lewat geseran \omterm{def:b1:water-small-molecules:ph}{pH-nya}.
  \item Hitunglah perubahan $[\mathrm{H^+}]$ antara \omterm{def:b1:water-small-molecules:ph}{pH} 7.40 dan \omterm{def:b1:water-small-molecules:ph}{pH} pada
        pertanyaan 9, serta fraksi proton tambahan yang tetap bebas
        dalam larutannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Lari cepatnya, \omterm{def:b1:organism-environment:organism}{sistem terbuka}.}
Sekarang paru menahan $\mathrm{CO_2}$ terlarut pada \qty{1.2}{mmol/L}
dengan mengembuskan kelebihannya.
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:b1:water-small-molecules:ph}{pH-nya} dengan bikarbonat pada nilai barunya dan
        $\mathrm{CO_2}$ ditahan pada \qty{1.2}{mmol/L}.
  \item Berapa milimol $\mathrm{CO_2}$ yang harus disingkirkan paru, di
        atas keluaran istirahatnya, untuk menahan nilai itu?
  \item Keluaran istirahatnya adalah \qty{10}{mmol} $\mathrm{CO_2}$
        tiap menit. Berapa kali lipat ventilasinya harus naik selama
        semenit untuk membersihkan kelebihan itu?
  \item Pelari itu bernapas berlebih lebih jauh lagi, sehingga tekanan
        parsial $\mathrm{CO_2}$-nya turun ke \qty{30}{mmHg}. Hitunglah
        $\mathrm{CO_2}$ terlarut yang baru dan \omterm{def:b1:water-small-molecules:ph}{pH-nya}.
  \item Sekarang masukkan \omterm{def:b1:water-small-molecules:buffer}{penyangga} proteinnya: dari \qty{60}{mmol}
        proton itu, sebagian diambil protein sebanding dengan kapasitas
        kedua \omterm{def:b1:water-small-molecules:buffer}{penyangganya}. Taksirlah kapasitas sistem bikarbonat
        terbuka dekat \omterm{def:b1:water-small-molecules:ph}{pH} 7.4 (milimol asam tiap liter yang menggeser
        \omterm{def:b1:water-small-molecules:ph}{pH-nya} satu satuan, dari Henderson--Hasselbalch dengan
        $\mathrm{CO_2}$ tetap) dan bagian proton yang diambil tiap
        \omterm{def:b1:water-small-molecules:buffer}{penyangganya}.
  \item Hitunglah ulang \omterm{def:b1:water-small-molecules:ph}{pH} sesudah lari cepat itu dengan kedua
        \omterm{def:b1:water-small-molecules:buffer}{penyangganya}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pemulihan, dan ginjalnya.}
\begin{enumerate}[resume]
  \item Sepanjang jam berikutnya hati mengoksidasi laktatnya
        (menyingkirkan asamnya). Apa yang terjadi pada bikarbonat dan
        \omterm{def:b1:water-small-molecules:ph}{pH-nya}, dengan paru menahan $\mathrm{CO_2}$ tetap?
  \item Seorang pasien dengan gagal ginjal menahan \qty{60}{mmol} asam
        sehari dan tidak sanggup mengekskresikannya. Hitunglah
        turunnya bikarbonat tiap hari (\omterm{def:b1:organism-environment:organism}{sistem terbuka}) dan \omterm{def:b1:water-small-molecules:ph}{pH} sesudah
        tiga hari bila bikarbonatnya tidak diganti.
  \item Seorang pasien dengan penyakit paru menahan $\mathrm{CO_2}$
        pada tekanan parsial \qty{60}{mmHg} dengan bikarbonat pada
        \qty{24}{mmol/L}. Hitunglah \omterm{def:b1:water-small-molecules:ph}{pH-nya}. Ginjalnya menanggapi
        sepanjang beberapa hari dengan menaikkan bikarbonatnya ke
        \qty{33}{mmol/L}; hitunglah \omterm{def:b1:water-small-molecules:ph}{pH-nya} pada saat itu.
  \item Jelaskan dalam dua kalimat mengapa paru mengoreksi \omterm{def:b1:water-small-molecules:ph}{pH} dalam
        beberapa menit dan ginjal dalam beberapa hari, dan apa yang
        dikendalikan masing-masing dalam rasio
        Henderson--Hasselbalch.
  \item Ginjal mengekskresikan \qty{60}{mmol} asam sehari itu dalam
        \qty{1.5}{L} kemih pada \omterm{def:b1:water-small-molecules:ph}{pH} 5. Hitunglah proton bebas dalam
        kemih itu, lalu simpulkan bagaimana asamnya diangkut.
  \item Muntah menghilangkan \qty{100}{mmol} HCl. Ramalkan arah
        perubahan \omterm{def:b1:water-small-molecules:ph}{pH-nya} lalu hitunglah bagi \omterm{def:b1:organism-environment:organism}{sistem terbukanya}.
  \item Nyatakan hasilnya: geseran \omterm{def:b1:water-small-molecules:ph}{pH} darah bagi beban laktat
        \qty{60}{mmol} dalam sistem tertutup, dalam \omterm{def:b1:organism-environment:organism}{sistem terbuka}
        dengan bikarbonat saja, dan dengan \omterm{def:b1:water-small-molecules:buffer}{penyangga} proteinnya
        ditambahkan.
\end{enumerate}
\end{problem}
