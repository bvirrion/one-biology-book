\section*{Chapter \ref{ch:g6:uni-multi} --- One Cell or Many}

\begin{solution}{exo:g6:uni-multi:1}
An \omterm{def:g4:ecosystems:levels}{organism} made of a single \omterm{def:g6:cell-theory:cell}{cell} that carries out all life functions.
\end{solution}

\begin{solution}{exo:g6:uni-multi:2}
An \omterm{def:g4:ecosystems:levels}{organism} made of many \omterm{def:g6:cell-theory:cell}{cells} that usually specialise and cooperate.
\end{solution}

\begin{solution}{exo:g6:uni-multi:3}
A group of similar \omterm{def:g6:cell-theory:cell}{cells} working together on a shared job.
\end{solution}

\begin{solution}{exo:g6:uni-multi:4}
\omterm{def:g1:plants-animals:animal}{Animal}: \omterm{def:g5:nerves:system}{nerve}, \omterm{def:g3:bones:muscle}{muscle}, red \omterm{def:g5:circulation:blood}{blood} \omterm{def:g6:cell-theory:cell}{cell}, etc. \omterm{def:g1:plants-animals:plant}{Plant}: \omterm{def:g4:plants-need:parts}{root} hair, palisade,
xylem --- any valid pair.
\end{solution}

\begin{solution}{exo:g6:uni-multi:5}
It is specialised and depends on other \omterm{def:g6:cell-theory:cell}{cells}/systems for \omterm{def:g2:food-energy:food}{food}, oxygen
and waste removal; it is not a complete free-living \omterm{def:g4:ecosystems:levels}{organism}.
\end{solution}
