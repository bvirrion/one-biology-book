\section*{Chapitre \ref{ch:g5:human-reproduction} --- La reproduction humaine}

\begin{solution}{exo:g5:human-reproduction:1}
La \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} vient de la mère, la \omterm{def:g4:animal-reproduction:sexual}{cellule mâle} du père~; leur
rencontre est la \omterm{def:g5:flower-to-fruit:steps}{fécondation}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:2}
Dans l'\omterm{prop:g5:human-reproduction:plan}{utérus} de sa mère, pendant environ neuf mois --- la
\omterm{def:g5:human-reproduction:pregnancy}{grossesse}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:3}
D'abord \omterm{def:g5:human-reproduction:pregnancy}{embryon}, puis \omterm{def:g5:human-reproduction:pregnancy}{fœtus} --- à partir de deux mois environ, une
fois le plan d'organisation en place.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:4}
Par le \omterm{prop:g5:human-reproduction:cord}{cordon ombilical}~: les \omterm{def:g4:journey-of-food:digestion}{nutriments} et le dioxygène passent du
\omterm{prop:g4:journey-of-food:absorb}{sang} de la mère à celui du bébé, et les déchets du bébé repassent en
sens inverse pour que le corps de la mère s'en occupe.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:5}
L'endroit où ton propre \omterm{prop:g5:human-reproduction:cord}{cordon ombilical} était attaché, avant d'être
coupé à la \omterm{def:g2:born-grow-die:cycle}{naissance}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:6}
Le premier cri remplit les \omterm{def:g4:breathing:organs}{poumons} d'air pour la première fois~:
jusqu'à cet instant ils n'avaient jamais respiré --- le dioxygène
était toujours arrivé par le cordon.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:7}
La \omterm{def:g5:flower-to-fruit:steps}{fécondation} dans le corps de la mère~; une \omterm{def:g2:born-grow-die:cycle}{naissance} \omterm{def:g4:animal-reproduction:ovivivi}{vivipare} (le
bébé grandit dans l'\omterm{prop:g5:human-reproduction:plan}{utérus} et naît vivant)~; l'allaitement ensuite
--- exactement les caractères de la classe des \omterm{prop:g3:sorting-living-things:groups}{mammifères}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:8}
Parce que tout ce qui se trouve dans le \omterm{prop:g4:journey-of-food:absorb}{sang} de la mère parvient au
bébé par le cordon --- les substances nocives de la fumée et de
l'alcool comprises --- et qu'un \omterm{def:g5:human-reproduction:pregnancy}{fœtus} est bien plus fragile qu'un
adulte.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:9}
Vrais jumeaux~: une seule \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée s'est partagée en
deux \omterm{def:g5:human-reproduction:pregnancy}{embryons}. Faux jumeaux~: deux \omterm{def:g4:animal-reproduction:sexual}{cellules œufs} fécondées par deux
\omterm{def:g4:animal-reproduction:sexual}{cellules mâles}, qui n'ont partagé que l'\omterm{prop:g5:human-reproduction:plan}{utérus}. Les vrais jumeaux se
ressemblent comme des copies parce qu'un corps se construit à partir
de sa première \omterm{def:g5:first-look-at-cells:cell}{cellule} --- et qu'ils sont partis de la même.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:10}
Du côté de l'éléphante, et même au-delà~: d'ordinaire un seul bébé,
des années sans défense, et des soins qui comprennent non seulement
la nourriture et la protection mais l'enseignement --- la langue,
les savoir-faire, l'école. Les millions sans soins de la morue sont
le pôle opposé.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:11}
Le travail de charger du dioxygène et de décharger du dioxyde de
carbone~: jusqu'à la \omterm{def:g2:born-grow-die:cycle}{naissance}, le \omterm{prop:g4:journey-of-food:absorb}{sang} de la mère le faisait par le
cordon~; dès la première \omterm{def:g4:breathing:movements}{inspiration}, les poches d'air des \omterm{def:g4:breathing:organs}{poumons}
du bébé le reprennent. Le cri tombe à point nommé parce qu'il arrive
à la minute exacte où l'ancienne ligne d'approvisionnement est
coupée --- les \omterm{def:g4:breathing:organs}{poumons} doivent démarrer précisément quand le cordon
s'arrête.
\end{solution}
