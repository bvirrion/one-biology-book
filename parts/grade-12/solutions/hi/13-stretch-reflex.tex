\section*{अध्याय \ref{ch:g12:stretch-reflex} --- तनन प्रतिवर्त और तंत्रिका संदेश}

\begin{solution}{exo:g12:stretch-reflex:1}
ग्राही (पेशी तर्कु), संवेदी \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन}, एकीकरण केंद्र (मेरुरज्जु का धूसर
द्रव्य, एक \omterm{def:g12:stretch-reflex:synapse}{तंत्रिका-संधि}), प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन}, प्रभावक (वह पेशी)।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:2}
\omterm{def:g10:cells-common-unit:organelle}{केंद्रक} वाली एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} काया, संकेत लेती द्रुमिकाएँ, और एक तंत्रिकाक्ष
(अक्सर मायलिन वाला) जो उस \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन} का संदेश उसके सिरों तक ढोता है। वह संदेश
द्रुमिकाओं तथा \omterm{def:g10:cells-common-unit:cell}{कोशिका} काया से उस तंत्रिकाक्ष के सहारे उन सिरों तक चलता है।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:3}
झिल्ली वोल्टता का एक छोटा-सा पलटाव, जो \qty{-70}{mV} से लगभग
\qty{+30}{mV} तक और एक मिलीसेकंड के भीतर वापस जाता है, तथा उस तंत्रिकाक्ष पर
अनबदला चलता है। सब या कुछ नहीं: देहली से नीचे कुछ नहीं, और उससे ऊपर हमेशा वही
पूरा संकेत। तीव्रता \omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभवों} की बारंबारता में कूटबद्ध होती है।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:4}
\omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभव} उस सिरे तक पहुँचते हैं; पुटिकाएँ वह \omterm{def:g12:stretch-reflex:synapse}{तंत्रिकाप्रेषी} उस दरार में
छोड़ती हैं; वह पाती \omterm{def:g10:cells-common-unit:cell}{कोशिका} पर बैठे ग्राहियों से बँधता है और उसकी वोल्टता
बदल देता है (देहली की ओर या उससे दूर); और फिर वह नष्ट कर दिया जाता या वापस
ले लिया जाता है।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:5}
ऐसिटिलकोलीन। क्यूरारे उस पेशी पर उसके ग्राही बिना उन्हें सक्रिय किए घेर
लेता है; और बोटुलिनम विष पुटिकाओं से उसका छूटना रोक देता है। दोनों लकवा
मारते हैं।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:6}
\qty{100}{ms} में 6, 12 और 24 शिखर: यानी प्रति सेकंड 60, 120 और 240। उन
शिखरों का आकार तथा रूप नहीं बदलता।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:7}
\qty{1.5}{m} के लिए चालन समय $30 - 12 = \qty{18}{ms}$: यानी लगभग
\qty{83}{m/s}।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:8}
पृष्ठ मूल संवेदी तंतु ढोता है: वह कटे तो तनाव का संदेश उस रज्जु तक कभी पहुँचता
ही नहीं, पर वह प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन} तथा उसका तंतु सही-सलामत हैं और उन्हें उद्दीप्त
करके उस पेशी को अब भी सिकोड़ा जा सकता है। और अधर मूल प्रेरक तंतु ढोता है: वह
कटे तो उस रज्जु से उस पेशी तक कुछ भी पहुँच ही नहीं सकता।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:9}
प्रतिपक्षी भी सिकुड़ता तो वह उस हरकत का विरोध करता और ख़ुद तन जाता, जिससे
उसका अपना \omterm{def:g12:stretch-reflex:reflex}{प्रतिवर्त} छिड़ जाता --- यानी एक रस्साकशी। वह संदमन उस रज्जु का
अंतरन्यूरॉन करता है, और वह भी प्रतिपक्षी के प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉनों} पर किसी
संदमनकारी \omterm{def:g12:stretch-reflex:synapse}{तंत्रिका-संधि} से होकर।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:10}
1 इकाई: कुछ नहीं (देहली से नीचे)। 4 और 8 इकाइयाँ: उन्हीं \qty{100}{mV} के
\omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभव}, पर ऊँची दरों पर, और 4 के मुक़ाबले 8 के लिए ऊँची।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:11}
प्रेषी की पुटिकाएँ केवल उस सिरे में हैं और ग्राही केवल उस पाती झिल्ली पर,
इसलिए वह रसायन एक ही ओर काम कर सकता है। और उस प्रेषी का छूटना, उसका उस दरार
के पार फैलना तथा उन ग्राहियों की प्रतिक्रिया लगभग एक मिलीसेकंड लेते हैं।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:12}
ऐसिटिलकोलीन उन जोड़ों में जमा हो जाता है और उन तंतुओं को चलाता रहता है:
यानी फड़कन, फिर टिका हुआ संकुचन तथा पेशियों की थकावट, जिनमें साँस लेने वाली
भी हैं --- यानी दम घुटने से मौत। कोई ग्राही रोकती दवा (ऐट्रोपीन जैसी) उस
अतिरिक्त उत्तेजन को घटाती है; बहुत कम हो तो वह विषाक्तता बनी रहती है, और
बहुत अधिक हो तो वह क्यूरारे की तरह लकवा मार देती है, इसलिए ख़ुराक ठीक बैठनी
चाहिए।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:13}
हर संवेदी इनपुट अपने प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉनों} को उत्तेजित करता है और, सामान्य रूप
से, अंतरन्यूरॉनों से होकर प्रतिपक्षियों को संदमित। संदमन रुक जाए तो कोई
स्पर्श हर जोड़ के दोनों पक्ष उत्तेजित कर देता है और वह उत्तेजन उस रज्जु भर
बेरोक फैलता है: सारी पेशियाँ एक साथ सिकुड़ती हैं, और वह ऐंठन दूसरी पेशियाँ
तान देती है जिनके \omterm{def:g12:stretch-reflex:reflex}{प्रतिवर्त} उसी में जुड़ जाते हैं।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:14}
\qty{1.5}{m} के लिए चालन समय $60 - 12 = \qty{48}{ms}$: यानी लगभग
\qty{31}{m/s}, जो सामान्य का एक तिहाई है। मायलिन \omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभव} को लगातार चलने
के बजाय उस ग़िलाफ़ की ख़ाली जगहों के बीच छलाँग लगाने देता है; और उसके बिना
वह तंतु किसी पतले बिना मायलिन वाले तंतु जैसा चालन करता है।
\end{solution}

\begin{solution}{exo:g12:stretch-reflex:15}
वह रज्जु मस्तिष्क तथा शरीर के बीच संदेश ढोती तो है, पर उन्हें संसाधित भी
करती है: उसका धूसर द्रव्य \omterm{def:g12:stretch-reflex:reflex}{प्रतिवर्त} चापों की तंत्रिका-संधियाँ थामे है, जहाँ
कोई तनाव किसी संकुचन में और, अंतरन्यूरॉनों से होकर, प्रतिपक्षी के संदमन में
बदल दिया जाता है, वह भी बिना किसी संदेश के मस्तिष्क तक पहुँचे। जिस जानवर की
रज्जु उसके मस्तिष्क से काट दी गई हो वह आज भी किसी चिकोटी से अपना पंजा खींच
लेता है और आज भी घुटने की उछाल दिखाता है: यानी वह रज्जु एक केंद्र है, केवल
कोई तार नहीं।
\end{solution}

\begin{solution}{pb:g12:stretch-reflex:1}
\textbf{1.} $0.75 + 0.75 = \qty{1.5}{m}$।

\textbf{2.} चालन के $30 - 4 - 4 - 1 = \qty{21}{ms}$: यानी $1.5/0.021
\approx \qty{70}{m/s}$।

\textbf{3.} हर ओर लगभग \qty{10}{cm} और, यानी कुल \qty{0.2}{m}:
$0.2/70 \approx \qty{3}{ms}$ और, यानी कोई \qty{33}{ms}।

\textbf{4.} वह विद्युत संकेत उन तंतुओं की सक्रियता है; जबकि ख़ुद वह संकुचन
--- यानी तंतुकों का फिसलना और उस \omterm{def:g10:muscles-and-joints:muscle}{कंडरा} पर खिंचाव --- उस टाँग को हिलाने भर
बल बनाने में दसियों मिलीसेकंड लेता है।

\textbf{5.} रज्जु से मस्तिष्क तक का संदेश और वापस आता हुक्म लगभग
\qty{70}{m/s} पर \qty{1.2}{m} चलते हैं --- यानी \qty{17}{ms} --- और उसमें
कई तंत्रिका-संधियाँ तथा ख़ुद मस्तिष्क का संसाधन जुड़ता है: यानी कम से कम
\qty{50}{ms}, और कोई फ़ैसला तो कहीं अधिक। वह उछाल किसी भी मस्तिष्क संकेत के
प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉनों} तक पहुँचने से पहले ही ख़त्म हो चुकी होती है।

\textbf{6.} प्रति सेकंड 200 पर, \qty{10}{ms} में 2 \omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभव}
(विश्राम में 0.2)। वह दर बदलती है; हर शिखर का आकार तथा रूप नहीं बदलता।

\textbf{7.} 10 के गुणक से।

\textbf{8.} प्रति सेकंड 20 पर प्रेषी की ख़ुराकें \qty{50}{ms} के अंतर पर
होती हैं और हर एक अगली से पहले मिट जाती है; इसलिए वह प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन} कभी
देहली तक पहुँचता ही नहीं। जबकि प्रति सेकंड 200 पर वे \qty{5}{ms} के अंतर पर
पहुँचती हैं और जुड़ जाती हैं: देहली पार होती है और वह प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन} संकेत
छोड़ देता है।

\textbf{9.} कम प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन} देहली तक पहुँचते हैं और हर एक कम \omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभव}
छोड़ता है: यानी कम प्रेरक इकाइयाँ, और वह भी नीची दर पर, सिकुड़ती हैं, तथा वह
उछाल कमज़ोर होती है।

\textbf{10.} \omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभव} सब या कुछ नहीं वाले हैं: उनका आकार उस \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन} से तय
है, इसलिए वह उस उद्दीपन के बारे में कोई जानकारी नहीं ढोता। केवल वह दर बदल
सकती है।

\textbf{11.} ऐसिटिलकोलीन; और ग्राहियों से बँधने के बाद वह उस दरार के किसी
\omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} से मिलीसेकंडों के भीतर नष्ट कर दिया जाता है।

\textbf{12.} उस \omterm{def:g10:muscles-and-joints:muscle}{पेशी तंतु} के ग्राही: ऐसिटिलकोलीन छूटता तो है पर काम नहीं कर
सकता।

\textbf{13.} पुटिकाओं से ऐसिटिलकोलीन का छूटना। प्रयोगशाला में सीधे उस पेशी
पर लगाया ऐसिटिलकोलीन बोटुलिनम विष के नीचे उसे सिकोड़ देता (ग्राही मुक्त हैं)
पर क्यूरारे के नीचे नहीं (ग्राही रुके हैं)।

\textbf{14.} उस संवेदी तंतु के \omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभव} और, चूँकि उस रज्जु तक कुछ पहुँचता
ही नहीं, उस प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉन} के तथा वह विद्युत-पेशीलेख भी; पर उस रुकावट से
नीचे उसकी तंत्रिका उद्दीप्त की जाए तो वह पेशी अब भी सिकुड़ती है।

\textbf{15.} क्यूरारे ग्रहण रोकता है, बोटुलिनम विष छूटना रोकता है, और वह
निश्चेतक चालन: यानी तीन क़दम, और एक चुप पेशी।

\textbf{16.} वह संवेदी तंतु ऐसे अंतरन्यूरॉन को भी उत्तेजित करता है जो पिछली
जाँघ की पेशियों के प्रेरक \omterm{def:g12:stretch-reflex:neuron}{न्यूरॉनों} को संदमित करता है; इसलिए उनकी विश्राम
सक्रियता दब जाती है, और वह प्रतिपक्षी अपने विश्राम तनाव से भी नीचे ढीला पड़
जाता है।

\textbf{17.} उस \omterm{def:g12:stretch-reflex:reflex}{प्रतिवर्त} को केवल वह रज्जु चाहिए, जो उस स्तर पर सही-सलामत
है। और उसका बढ़ जाना दिखाता है कि मस्तिष्क सामान्य रूप से ऐसे संकेत भेजता है
जो उस \omterm{def:g12:stretch-reflex:reflex}{प्रतिवर्त} को दबाते हैं; उनके बिना वह चाप पूरी संवेदनशीलता पर चलती है।

\textbf{18.} वह तर्कु, वह संवेदी तंत्रिका या पृष्ठ मूल, उस स्तर पर रज्जु का
धूसर द्रव्य, वह प्रेरक तंत्रिका या अधर मूल, और ख़ुद वह पेशी। प्रेरक
तंत्रिका सीधे उद्दीप्त करके उस पेशी को रिकॉर्ड करना बता देता है कि प्रेरक
पक्ष काम करता है या नहीं; और किसी तनाव के दौरान संवेदी तंत्रिका रिकॉर्ड करना
बता देता है कि संवेदी पक्ष करता है या नहीं।

\textbf{19.} वह एक ही ठोंक में पूरी चाप --- संवेदी तंत्रिका, रज्जु, प्रेरक
तंत्रिका, पेशी --- और उस पर मस्तिष्क का दबाव, दोनों जाँच लेता है: कोई
ग़ैरहाज़िर, कमज़ोर या बढ़ा हुआ \omterm{def:g12:stretch-reflex:reflex}{प्रतिवर्त} किसी और चिह्न से पहले ही तंत्रिका
तंत्र के किसी स्तर की ओर इशारा कर देता है।

\textbf{20.} लगभग \qty{70}{m/s}; उस तर्कु के \omterm{prop:g12:stretch-reflex:actionpotential}{क्रिया विभवों} की बारंबारता; और
उस पेशी के ग्राहियों से ऐसिटिलकोलीन का बँधना।
\end{solution}
