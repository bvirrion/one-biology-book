\chapter{Komunikasi Hormon}\label{ch:g8:hormonal-communication}

\omterm{def:g8:puberty:puberty}{Pubertas} memperkenalkannya
(\cref{def:g8:puberty:hormones}); pilnya memanfaatkannya
(\cref{def:g8:contraception:pill}); gelombang panas
ketakutannya mengantarkan satu di tengah lari cepat
(\cref{ex:g8:nervous-communication:cases}). Jasa posnya
layak mendapat lawatannya sendiri: \omterm{def:g8:hormonal-communication:glands}{kelenjar} yang menulis
suratnya, \omterm{prop:g4:journey-of-food:absorb}{darah} yang mengangkutnya, kiat pengalamatan yang
membuat sebuah siaran menjadi tepat --- dan gelung \omterm{prop:g8:hormonal-communication:feedback}{umpan
balik} yang menjaga seluruh surat-menyuratnya tetap seimbang.

\section{Kelenjar, hormon, sasaran}

\begin{definition}[Kelenjar endokrin]\label{def:g8:hormonal-communication:glands}
Sebuah \emph{kelenjar}\index{kelenjar} yang hasilnya
dilepaskan \emph{ke dalam \omterm{prop:g4:journey-of-food:absorb}{darah}} (alih-alih menuruni sebuah
saluran, seperti \omterm{def:g4:journey-of-food:tract}{air liurnya}) adalah kelenjar
\emph{endokrin}, dan hasilnya adalah
\emph{hormon}\index{hormon}. Penulis utamanya:
\emph{kelenjar kendali} di dasar \omterm{def:g5:brain-in-command:system}{otak} --- pemula
\omterm{def:g8:puberty:puberty}{pubertasnya}, dan pemimpin sebagian besar yang lain;
\emph{kelenjar gondok} di lehernya --- hormonnya menetapkan
laju jalan seluruh tubuhnya; \emph{tudung adrenal} di atas
\omterm{def:g7:removing-wastes:kidneys}{ginjalnya} --- hormon siaga ketakutan dan lari cepatnya;
\emph{pankreas} --- hormonnya mengelola \omterm{prop:g7:organs-at-work:bill}{glukosa} \omterm{prop:g4:journey-of-food:absorb}{darahnya};
dan \emph{\omterm{def:g8:reproductive-systems:female}{indung telur} serta buah zakar} --- hormon daur dan
pembangunan ulang pada \cref{ch:g8:reproductive-systems}.
\end{definition}

\begin{proposition}[Pengalamatan menurut reseptornya]\label{prop:g8:hormonal-communication:address}
\omterm{prop:g4:journey-of-food:absorb}{Darahnya} mengantarkan setiap \omterm{def:g8:hormonal-communication:glands}{hormon} ke mana-mana
(\cref{ex:g8:puberty:sites}) --- namun masing-masing hanya
bekerja pada \emph{organ sasarannya}\index{organ sasaran}:
\omterm{def:g5:first-look-at-cells:cell}{sel} yang dibangun dengan tempat berlabuh, yaitu
\emph{reseptor}, bagi pengantar pesan itu. Pengangkutan
secara siaran, pengalamatan di sisi penerimanya: sebuah
surat yang diterima setiap rumah, hanya terbaca di tempat
kuncinya cocok. (Pelabuhan pada \omterm{def:g8:nervous-communication:synapse}{sinapsis}
\cref{def:g8:nervous-communication:synapse} adalah kiat yang
sama pada jarak nanometer.)
\end{proposition}
\admitted

\begin{example}[Lawatan satu pengantar pesan: hormon siaganya]\label{ex:g8:hormonal-communication:alarm}
Nyaris tertabrak ketika bersepeda: \omterm{def:g8:hormonal-communication:glands}{kelenjar} adrenalnya
mencurahkan \omterm{def:g8:hormonal-communication:glands}{hormon} siaganya ke dalam \omterm{prop:g4:journey-of-food:absorb}{darah}, dan dalam
hitungan detik --- satu putaran peredarannya
(\cref{ex:g4:heart-and-blood:round}) --- setiap penerima
yang berlengkap menanggapinya seketika: laju \omterm{def:g4:heart-and-blood:heart}{jantungnya}
melonjak, saluran napasnya melebar, \omterm{prop:g7:digestion-and-nutrients:absorption}{hatinya} melepaskan
\omterm{prop:g7:organs-at-work:bill}{glukosa} simpanannya (bank pada
\cref{ex:g7:digestion-and-nutrients:breadtrace}, dipanggil),
\omterm{def:g4:heart-and-blood:vessels}{pembuluh} \omterm{def:g3:bones-and-muscles:muscle}{ototnya} melebar dan \omterm{def:g4:heart-and-blood:vessels}{pembuluh} \omterm{def:g4:journey-of-food:tract}{saluran pencernaannya}
menyempit (pengalihan rute pada
\cref{prop:g7:organs-at-work:supply}, kini ditahan \omterm{def:g8:hormonal-communication:glands}{hormon}),
dan \omterm{def:g1:the-five-senses:sense}{kulitnya} memucat serta meremang. Satu surat, enam
jawaban, satu keadaan yang terpadu: siap.
\end{example}

\section{Keseimbangan lewat umpan balik}

\begin{proposition}[Asas umpan baliknya]\label{prop:g8:hormonal-communication:feedback}
Sistem \omterm{def:g8:hormonal-communication:glands}{hormon} mengatur dirinya sendiri lewat \emph{umpan
balik}\index{umpan balik}: \omterm{def:g8:hormonal-communication:glands}{kelenjar} penulisnya
\emph{mengawasi hasil} suratnya sendiri, lalu memperlambat
ketika hasilnya tercapai --- sebuah termostat, bukan sebuah
keran. Peragaan terbaiknya adalah \omterm{prop:g7:organs-at-work:bill}{glukosa} \omterm{prop:g4:journey-of-food:absorb}{darah}: sesudah
sebuah santapan, \omterm{prop:g7:organs-at-work:bill}{glukosa} yang naik itu sendiri memicu \omterm{def:g8:hormonal-communication:glands}{hormon}
penyimpanan pankreasnya; \omterm{prop:g7:organs-at-work:bill}{glukosanya} disimpan, kadarnya
turun, dan pengeluarannya mereda bersamanya; di antara
santapannya sebuah \omterm{def:g8:hormonal-communication:glands}{hormon} mitra melepaskan simpanannya ke
arah sebaliknya. Kadarnya melaju di sebuah lajur yang
sempit, siang dan malam, tanpa pengemudi yang sadar.
\end{proposition}
\admitted

\begin{example}[Ketika umpan baliknya gagal: diabetes]\label{ex:g8:hormonal-communication:diabetes}
Pada \emph{diabetes}\index{diabetes}, isyarat \omterm{def:g8:hormonal-communication:glands}{hormon}
penyimpanannya gagal --- pankreasnya berhenti menulisnya,
atau penerimanya berhenti menjawab. \omterm{prop:g7:organs-at-work:bill}{Glukosanya} menumpuk di
dalam \omterm{prop:g4:journey-of-food:absorb}{darah} sesudah setiap santapan; penarikan kembali oleh
\omterm{def:g7:removing-wastes:kidneys}{ginjalnya} kewalahan dan gula muncul di dalam \omterm{def:g7:removing-wastes:kidneys}{urine} ---
isyarat klasik pada \cref{ex:g7:removing-wastes:thrift},
akhirnya dijelaskan. Pengobatannya adalah bab yang
diterapkan: takaran terukur \omterm{def:g8:hormonal-communication:glands}{hormon} yang hilang itu, diatur
waktunya menurut santapannya --- surat yang ditulis dengan
jarum suntik. Jutaan orang hidup baik dengan pengaturan
persis seperti ini.
\end{example}

\begin{remark}[Kedua sistemnya dalam kemitraan]\label{rem:g8:hormonal-communication:partnership}
Telegraf dan pos pada
\cref{prop:g8:nervous-communication:compare} adalah satu
tata usaha. \omterm{def:g8:hormonal-communication:glands}{Kelenjar} kendali di dasar \omterm{def:g5:brain-in-command:system}{otaknya} adalah
antarmukanya: \omterm{def:g5:brain-in-command:system}{sistem saraf} di atasnya, \omterm{def:g8:hormonal-communication:glands}{hormon} di bawahnya
--- ketakutan sang pengendara sepeda berjalan \omterm{def:g8:nervous-communication:neuron}{saraf} lebih
dahulu lalu ditahan \omterm{def:g8:hormonal-communication:glands}{hormon}; \omterm{def:g8:puberty:puberty}{pubertas} berjalan lambat secara
hormonal di bawah seorang pemimpin yang berpangkal di \omterm{def:g5:brain-in-command:system}{otak};
dombanya pada \cref{ex:g8:reproduction-and-environment:lambs}
mengubah panjang hari (sebuah penginderaan \omterm{def:g8:nervous-communication:neuron}{saraf}) menjadi
\omterm{def:g8:hormonal-communication:glands}{hormon} pembiakan. Kabel cepat di tempat kecepatannya
penting, pos lambat di tempat programnya harus berjalan
berbulan-bulan, dan kantor penerjemahan di antaranya.
\end{remark}

\section{Membaca sebuah sistem hormon}

\begin{method}[Empat pertanyaan bagi hormon apa pun]\label{met:g8:hormonal-communication:read}
Untuk \omterm{def:g8:hormonal-communication:glands}{hormon} apa pun yang kamu jumpai, tetapkan:
\begin{enumerate}
  \item \emph{penulisnya}: \omterm{def:g8:hormonal-communication:glands}{kelenjar} yang mana, ke dalam
        \omterm{prop:g4:journey-of-food:absorb}{darah}, atas pemicu apa?
  \item \emph{penerimanya}: \omterm{prop:g8:hormonal-communication:address}{organ sasaran} mana yang membawa
        reseptornya, dan apa yang dilakukan masing-masing?
  \item \emph{\omterm{prop:g8:hormonal-communication:feedback}{umpan baliknya}}: hasil apa yang melapor
        kembali, dan bagaimana penulisan berlebih dicegah?
  \item \emph{kegagalan dan pemakaiannya}: apa yang terjadi
        ketika suratnya gagal --- dan apakah kedokteran
        menirunya (pilnya, takaran penderita \omterm{ex:g8:hormonal-communication:diabetes}{diabetesnya})?
\end{enumerate}
\end{method}

\begin{example}[Caranya pada penetap laju kelenjar gondoknya]\label{ex:g8:hormonal-communication:thyroid}
Penulisnya: \omterm{def:g8:hormonal-communication:glands}{kelenjar} gondoknya, dipimpin oleh \omterm{def:g8:hormonal-communication:glands}{kelenjar}
kendalinya. Penerimanya: nyaris setiap \omterm{def:g5:first-look-at-cells:cell}{sel} --- \omterm{def:g8:hormonal-communication:glands}{hormonnya}
menetapkan laju api lambat pada
\cref{prop:g7:what-is-respiration:power}. \omterm{prop:g8:hormonal-communication:feedback}{Umpan baliknya}:
\omterm{def:g8:hormonal-communication:glands}{kelenjar} kendalinya membaca kadarnya lalu memangkas
pimpinannya. Kegagalannya: terlalu sedikit, dan seluruh
mesinnya melangsam --- lelah, kedinginan, segalanya
melambat; terlalu banyak, dan ia berpacu. Pemakaiannya:
\omterm{def:g8:hormonal-communication:glands}{hormon} yang hilang itu berupa sebutir tablet harian ---
salah satu surat-lewat-pos tertua dalam kedokteran.
\end{example}

\section{Soal: Buku Catatan Glukosanya}

\begin{problem}[{Soal akhir pekan --- sehari gula darah,
dibaca sebagai surat-menyurat}]\label{pb:g8:hormonal-communication:1}
Catatan harian (rekaan) sebuah pemantau \omterm{prop:g7:organs-at-work:bill}{glukosa} sinambung:
mantap sepanjang malam; sarapan pukul 8 --- sebuah kenaikan,
lalu penurunan yang tenang menjelang pukul 10; olahraga
pukul 12 --- sebuah cekungan singkat, cepat tertangkap;
makan siang yang terlewat --- mantap sepanjang petang tanpa
terpengaruh; sebuah ketakutan pukul 17 (nyaris tertabrak
ketika bersepeda) --- kenaikan tajam yang singkat; makan
malam pukul 19 --- naik lalu turun lagi.

\medskip\noindent\textbf{Bagian I --- Paginya.}
\begin{enumerate}
  \item Kemantapan sepanjang malamnya: surat-menyurat dua
        pucuk surat yang mana yang menahan lajurnya selagi
        pemiliknya \omterm{prop:g1:healthy-habits:needs}{tidur}?
  \item Naik dan turunnya sarapannya: sebutkan pemicunya,
        penulisnya, suratnya, dan penerima yang menyimpan
        kelebihannya.
  \item Mengapa \emph{berhentinya} penurunan itu --- tanpa
        terjun di bawah lajurnya --- justru merupakan tanda
        tangan \omterm{prop:g8:hormonal-communication:feedback}{umpan baliknya}?
  \item Kadar pukul 10 menyamai kadar pukul 7. Apa yang
        diramalkan gambaran termostatnya tentang perjalanan
        pengeluarannya di antaranya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Petangnya.}
\begin{enumerate}[resume]
  \item Cekungan dan tangkapan olahraganya: surat yang mana
        yang memanggil bank \omterm{prop:g7:digestion-and-nutrients:absorption}{hatinya}, dan rekening bab yang
        mana yang sedang dibelanjakan?
  \item Makan siang yang terlewat tidak mengubah apa pun
        yang terlihat. Kerja diam \omterm{def:g8:hormonal-communication:glands}{hormon} mitra yang mana
        yang menjembatani petangnya?
  \item Kenaikan tajam ketakutannya: sebutkan penulisnya dan
        maksud gula yang mendadak itu --- siap untuk apa,
        menurut \cref{ex:g8:hormonal-communication:alarm}?
  \item Mengapa kenaikan ketakutannya mereda dalam sejam,
        tidak seperti tumpukan menetap pada \omterm{ex:g8:hormonal-communication:diabetes}{diabetesnya}?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Konsultasinya.}
\begin{enumerate}[resume]
  \item Catatan seorang penderita \omterm{ex:g8:hormonal-communication:diabetes}{diabetes} pada hari yang
        sama akan berbeda pada setiap santapannya. Uraikan
        kurva sarapannya tanpa pengobatan, dan di mana
        isyarat \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} akan muncul.
  \item Versi yang diobati: apa yang ditiru oleh takaran
        terukur sebelum tiap santapannya, dan mengapa
        besarnya harus sepadan dengan piringnya?
  \item Dokternya menyebut catatan yang sehat itu ``sebuah
        percakapan yang tak pernah kamu dengar''. Sebutkan
        para penuturnya sepanjang hari itu --- \omterm{def:g8:hormonal-communication:glands}{kelenjar},
        surat, penerima --- dalam satu tabel atau alinea
        ringkasan.
  \item Tutuplah dengan pesan babnya dalam satu kalimat:
        sistem semacam apa yang menjaga sebuah nilai tetap
        di lajurnya seumur hidup tanpa seorang pengemudi?
\end{enumerate}
\end{problem}
