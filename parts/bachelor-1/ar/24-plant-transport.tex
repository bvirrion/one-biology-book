\chapter{التبادل الغازي ونقل العصارة في النبات}\label{ch:b1:plant-transport}

في قمة سكوية على ارتفاع مئة متر فوق الأرض، يُسحب الماء من التربة
بالتبخر من \omterm{def:b1:flowering-plant-organization:organs}{الأوراق}، صاعدًا في عمود من \omterm{def:b1:cell-unit-of-life:cell}{خلايا} ميتة لا يزيد عرضه على
عرض الشعرة، تحت توتر يقصف سلكًا فولاذيًا بالمقطع نفسه. ولا \omterm{def:b1:membranes-transport:transporters}{مضخة}
هناك؛ وإنما تجرّ الماءَ روابطُ الهيدروجين نفسها التي تجعله يتشبث
بذاته. وفي الوقت نفسه، في جملة أنابيب ثانية إلى جانب الأولى، يجري
السكر الذي صنعته تلك \omterm{def:b1:flowering-plant-organization:organs}{الأوراق} في الاتجاه الآخر، مدفوعًا بضغط تولّده
\omterm{def:b1:flowering-plant-organization:organs}{الأوراق} نفسها. فجملتا السباكة في النبات، \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} \omterm{def:b1:plant-transport:phloem}{واللحاء}، تسوقهما
فيزياء يهيّئها النبات ثم يدعها تعمل. ويصف هذا الفصل \omterm{def:b1:plant-transport:stoma}{الثغور} التي
تقايض بها \omterm{def:b1:flowering-plant-organization:organs}{الورقة} الماء بثاني أكسيد الكربون، وآلية التماسك والتوتر
التي ترفع العصارة، والأوعية التي تحملها، \omterm{thm:b1:plant-transport:pressureflow}{وآلية جريان الضغط} التي
تحرك السكر، والمساومة بين الشرب والأكل التي تحكم ذلك كله.

\section{الثغور: البوابة}

\begin{definition}[الثغر وخليتا الحرس]\label{def:b1:plant-transport:stoma}
\index{ثغر}\index{خلية حارسة}
\emph{الثغر} مسام في بشرة \omterm{def:b1:flowering-plant-organization:organs}{الورقة} (أو ساق خضراء) تحدّه
\emph{خليتان حارستان}، وهما خليتان كلويتا الشكل جدرهما مثخنة في
الجانب المواجه للمسام ولييفات السللوز الدقيقة فيهما تلتف حولهما
كالأطواق. فإذا التقطت الخليتان الحارستان الماء وانتفختا منعتهما
الأطواق من التوسع وأجبرتهما على التقوس متباعدتين: فينفتح المسام.
وإذا فقدتا الماء انغلق. وتحمل \omterm{def:b1:flowering-plant-organization:organs}{الورقة} بضع مئات من الثغور في
الميليمتر المربع، أكثرها على وجهها السفلي؛ ومساميها مفتوحة تساوي
واحدًا أو اثنين في المئة من مساحة \omterm{def:b1:flowering-plant-organization:organs}{الورقة} ويمر بها تبادلها الغازي
كله تقريبًا.
\end{definition}

\begin{proposition}[كيف ينفتح الثغر ومتى]\label{prop:b1:plant-transport:opening}
\index{حمض الأبسيسيك}
تفتح الخليتان الحارستان المسام بضخ البروتونات إلى الخارج
(\cref{ch:b1:plant-water-minerals})، فيدخل البوتاسيوم والكلوريد
ويُصنع المالات في الداخل؛ فتخفض المنحلات كمونهما المائي، ويتبع
الماء من \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} المجاورة، ويرتفع امتلاؤهما بميغاباسكال. وعكس دفوق
الأيونات يغلق المسام في دقائق. أما الإشارات: فأولها \emph{الضوء} (الضوء
الأزرق على الخليتين الحارستين مباشرة، وهبوط $\mathrm{CO_2}$ الداخلي
مع بدء التركيب الضوئي) يفتح؛ و\emph{ارتفاع $\mathrm{CO_2}$
الداخلي} يغلق؛ و\emph{الإجهاد المائي} يغلق، عبر هرمون \emph{\omterm{prop:b1:plant-transport:opening}{حمض
الأبسيسيك}}، المصنوع في \omterm{def:b1:flowering-plant-organization:organs}{الجذور} \omterm{def:b1:flowering-plant-organization:organs}{والأوراق} الذابلة، الذي يطلق فقد
الأيونات من الخليتين الحارستين في دقائق؛ والهواء الجاف يغلق. \omterm{def:b1:flowering-plant-organization:organs}{فالورقة}
تفتح بوابتها حين يستحق الكربون الماء وتغلقها حين يشحّ الماء ---
وأكثر النباتات يفعل ذلك بنظم يومي، يفتح عند الفجر ويغلق عند
الغسق، ويستمر أيامًا في ضوء ثابت.
\end{proposition}

\section{رفع الماء: التماسك والتوتر}

\begin{theorem}[آلية التماسك والتوتر]\label{thm:b1:plant-transport:cohesiontension}
\index{نظرية التماسك والتوتر}\index{نتح}
يصعد الماء في \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} لأنه \emph{يُسحب} من فوق، لا لأنه يُدفع من
تحت. فالتبخر من الجدر المبتلة \omterm{def:b1:cell-unit-of-life:cell}{لخلايا} \omterm{def:b1:body-plans-tissues:tissue}{النسيج} الوسطي في \omterm{def:b1:flowering-plant-organization:organs}{الورقة} إلى
الحيزات الهوائية (\emph{النتح}) يسحب الماء في مسام الجدار الدقيقة
إلى هلالات مقعّرة، يضع توتر سطحها الماء الذي خلفها تحت توتر؛
ويُنقل التوتر، عبر أعمدة الماء المتصلة في \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} المتماسكة بالتماسك
(روابط الهيدروجين في
\cref{ch:b1:water-small-molecules})، نازلًا في الساق وإلى \omterm{def:b1:flowering-plant-organization:organs}{الجذور}،
فيرفع الماء من التربة. وحمل عمود ارتفاعه $h$ ضد الجاذبية يقتضي
توترًا مقداره $\rho g h$ --- أي \qty{1}{MPa} لكل مئة متر --- زائد ما
يلزم للتغلب على احتكاك الجريان، وهو بقدره تقريبًا؛ وأعمدة شجرة
عالية عند \qtyrange{-2}{-3}{MPa}، وهو ضمن مقاومة الماء للشد في
أنبوب دقيق.
\end{theorem}
\begin{proof}[شواهد]
بيّن ديكسون وجولي (1894) أن الماء في أنبوب زجاجي محكم يحتمل توترات
من عدة ميغاباسكالات دون أن ينقطع، وأن غصنًا مورقًا متنتحًا، محكمًا
على أنبوب ماء غاطس في زئبق، يرفع الزئبق أعلى مما يستطيعه الضغط
الجوي. وتقيس \emph{حجرة الضغط} عند شولاندر (1965) التوتر مباشرة:
إذ تُحكم \omterm{def:b1:flowering-plant-organization:organs}{ورقة} مقطوعة في حجرة وعنقها بارز، ويكون ضغط الغاز الذي يدفع
العصارة بالكاد إلى سطح القطع مساويًا للتوتر الذي كانت تحته ---
\qtyrange{0.5}{1}{MPa} في محصول جيد السقاية عند الظهيرة،
و\qtyrange{2}{4}{MPa} في جنبة صحراوية أو في تاج شجرة عالية. ويُظهر
مقياس القطر جذع الشجرة ينكمش نهارًا، إذ يعصر التوتر الأوعية،
وينتفخ ليلًا؛ ويُظهر مقياس حرارة في \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} العصارة تجري أسرع ما
تجري حين يبلغ النتح ذروته، وتتوقف ليلًا. ولا يلزم شيء من \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}
الحية: فساق قُتلت بسم أو حرارة ما تزال تنقل.
\end{proof}

\begin{definition}[قنوات الخشب والتجوّف]\label{def:b1:plant-transport:xylem}
\index{وعاء خشبي}\index{تجوّف}
يسافر الماء في \emph{\omterm{def:b1:flowering-plant-organization:tissues}{القصيبات}} و\emph{الأوعية} في \omterm{def:b1:flowering-plant-organization:tissues}{الخشب}
(\cref{ch:b1:flowering-plant-organization}): وهي \omterm{def:b1:cell-unit-of-life:cell}{خلايا} ميتة أُفرغت
من محتواها، وجدرها المتخشبة مثخنة بحلقات وحلزونات تمنعها من
الانطباق تحت التوتر الذي في داخلها، موصولة طرفًا بطرف عبر ثقوب
(الأوعية) أو عبر نقر في جدرها الجانبية (\omterm{def:b1:flowering-plant-organization:tissues}{القصيبات}). وقد يكون عرض
الوعاء \qtyrange{20}{300}{\micro m} وطوله سنتيمترات إلى أمتار:
فالأنابيب الواسعة تحمل أكثر بكثير (فالجريان في أنبوب يرتفع كالقوة
الرابعة لنصف قطره) لكنها أشد عرضة \emph{للتجوّف} --- وهو التكوّن
المفاجئ لفقاعة غاز في ماء تحت توتر، يفرّغ \omterm{def:b1:membranes-transport:transporters}{القناة} ويخرجها من
الخدمة. والنقر بين \omterm{def:b1:membranes-transport:transporters}{القنوات} توقف انتشار الفقاعات؛ والتجمد والجفاف
كلاهما يسبب التجوّف، وأوعية الشجرة يعاد ملء بعضها \omterm{prop:b1:plant-water-minerals:rootpressure}{بالضغط الجذري} في
الربيع أو تُستبدل \omterm{def:b1:flowering-plant-organization:tissues}{بخشب} جديد كل سنة.
\end{definition}

\begin{example}[قمة الشجرة]\label{ex:b1:plant-transport:top}
عند \qty{100}{m} يقتضي وزن العمود وحده \qty{1}{MPa} من التوتر؛
ويضيف الاحتكاك في الأوعية بقدره نهارًا، فيكون \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} عند التاج قرب
\qty{-2}{MPa} \omterm{def:b1:cell-unit-of-life:cell}{وخلايا} \omterm{def:b1:flowering-plant-organization:organs}{الورقة}، لتسحب منه الماء، أخفض من ذلك: فوجب أن
تنغلق ثغورها أبكر في النهار وأن يكون تركيبها الضوئي أبطأ من تركيب
غصن منخفض، \omterm{def:b1:flowering-plant-organization:organs}{والأوراق} في القمة صغيرة سميكة. وارتفاع أطول الأشجار،
وهو نحو \qty{120}{m}، تحدده على الأرجح هذه الحال: فما فوقه لا
تستطيع \omterm{def:b1:flowering-plant-organization:organs}{الأوراق} حفظ التوتر دون \omterm{def:b1:plant-transport:xylem}{تجوّف} في عصر جاف.
\end{example}

\section{تحريك السكر: جريان الضغط}

\begin{definition}[اللحاء والأنابيب الغربالية والمصدر والمصرف]\label{def:b1:plant-transport:phloem}
\index{لحاء}\index{أنبوب غربالي}\index{مصدر}\index{مصرف}
يسافر السكر في \emph{اللحاء}، في \emph{\omterm{def:b1:flowering-plant-organization:tissues}{أنابيب غربالية}}: وهي صفوف من
\omterm{def:b1:cell-unit-of-life:cell}{خلايا} حية فقدت نواها ومعظم عضياتها لكنها احتفظت بأغشيتها، موصولة
طرفًا بطرف عبر \emph{صفائح غربالية} مثقوبة، ترعى كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} منها
\emph{\omterm{def:b1:cell-unit-of-life:cell}{خلية} مرافقة} تمدها ببروتيناتها \omterm{def:b1:water-small-molecules:atp}{وATP}. والعصارة
\qtyrange{10}{30}{\%} سكروزًا، مع \omterm{def:b1:proteins:aminoacid}{أحماض أمينية} وأيونات وجزيئات
إشارة، وتجري من \emph{المصادر} --- \omterm{def:b1:flowering-plant-organization:organs}{الأوراق} المركّبة ضوئيًا، أو \omterm{def:b1:mammal-organization:organ}{أعضاء}
الادخار وهي تُفرَّغ --- إلى \emph{المصارف} --- القمم النامية،
\omterm{def:b1:flowering-plant-organization:organs}{والجذور}، والثمار، والبذور، \omterm{def:b1:mammal-organization:organ}{وأعضاء} الادخار وهي تُملأ --- بسرعة
\qtyrange{0.5}{1}{m/h}، في أي اتجاه، وفي اتجاهات مختلفة في أنابيب
مختلفة.
\end{definition}

\begin{theorem}[جريان الضغط]\label{thm:b1:plant-transport:pressureflow}
\index{آلية جريان الضغط}
تنتقل عصارة \omterm{def:b1:plant-transport:phloem}{اللحاء} بجريان كتلي يسوقه فرق في الضغط الهيدروستاتيكي
بين \omterm{def:b1:plant-transport:phloem}{المصدر} \omterm{def:b1:plant-transport:phloem}{والمصرف} (مونش، 1930). فعند \omterm{def:b1:plant-transport:phloem}{المصدر} \emph{تحمّل} \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}
المرافقة السكروز في \omterm{def:b1:flowering-plant-organization:tissues}{الأنابيب الغربالية} ضد تدرجه (بنقل مشترك مع
البروتون، من \omterm{def:b1:plant-water-minerals:pathways}{الأبوبلاست}، أو عبر \omterm{def:b1:eukaryotic-cell:plantcell}{الروابط السيتوبلازمية})؛ فيهبط \omterm{def:b1:membranes-transport:osmosis}{الكمون
المائي} للأنبوب، ويدخل الماء من \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} المجاور، ويرتفع \omterm{def:b1:eukaryotic-cell:plantcell}{الامتلاء} إلى
\qty{1}{MPa} أو أكثر. وعند \omterm{def:b1:plant-transport:phloem}{المصرف} \emph{يُفرَّغ} السكروز ويُستهلك أو
يُدَّخر، فيرتفع \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي}، ويغادر الماء، وينخفض \omterm{def:b1:eukaryotic-cell:plantcell}{الامتلاء}.
وبينهما تجري العصارة نازلة مع تدرج الضغط عبر الصفائح الغربالية،
حاملة ما ذاب فيها. فالنبات لا ينفق طاقة إلا عند الطرفين؛ أما
الجريان نفسه فسلبي.
\end{theorem}
\begin{proof}[شواهد]
حشرة منّ تتغذى على ساق تغرز مخرزها في \omterm{def:b1:plant-transport:phloem}{أنبوب غربالي} واحد؛ فإذا
قُطعت الحشرة وتُرك المخرز في مكانه نضح عصارة ساعات تحت ضغط الأنبوب
نفسه --- أي عصارة \omterm{def:b1:plant-transport:phloem}{لحاء} صافية، يمكن قياس تركيز السكروز فيها وسرعة
جريانها، وضغطها، مقيسًا بمانومتر على المخرز، نحو \qty{1}{MPa} قرب
\omterm{def:b1:plant-transport:phloem}{المصدر} وأخفض نحو \omterm{def:b1:plant-transport:phloem}{المصرف}. و$\mathrm{^{14}CO_2}$ الموسوم المطعَم
\omterm{def:b1:flowering-plant-organization:organs}{لورقة} يظهر في \omterm{def:b1:plant-transport:phloem}{لحاء} الساق في دقائق وينتقل بالسرعة التي يتنبأ بها
جريان المخرز؛ وتبريد قطعة من الساق، وهو يبطئ \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الحية ولا
يكاد يمسّ جريانًا يسوقه الضغط، لا يبطئ النقل إلا قليلًا؛ وتسميم
تحميل \omterm{def:b1:flowering-plant-organization:organs}{الورقة} \omterm{def:b1:plant-transport:phloem}{المصدر} يوقفه.
\end{proof}

\begin{method}[قراءة تجربة نقل]\label{met:b1:plant-transport:translocation}
\begin{enumerate}
  \item أطعم ورقةً $\mathrm{CO_2}$ موسومًا؛ وخذ عينات من الساق فوقها
        وتحتها على فترات: فموضع الوسم في مقابل الزمن يعطي اتجاه
        الجريان وسرعته.
  \item حلّق الساق (انزع حلقة من \omterm{def:b1:plant-transport:phloem}{اللحاء} الحامل \omterm{def:b1:plant-transport:phloem}{للحاء}، مبقيًا
        \omterm{def:b1:flowering-plant-organization:tissues}{الخشب}): فيتراكم السكر فوق الحلقة وتجوع \omterm{def:b1:flowering-plant-organization:organs}{الجذور} تحتها ---
        \omterm{def:b1:plant-transport:phloem}{فاللحاء} لا \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} هو الذي يحمله؛ وتموت الشجرة في موسم.
  \item اجمع العصارة من مخارز منّ مقطوعة على ارتفاعين: فهبوط
        السكروز والضغط بينهما هو التدرج الذي يسوق الجريان.
  \item غيّر المصارف: انزع الثمار فيذهب الوسم إلى \omterm{def:b1:flowering-plant-organization:organs}{الجذور}؛ وظلّل
        \omterm{def:b1:flowering-plant-organization:organs}{الورقة} فتصير هي نفسها مصرفًا. فنمط الجريان هو نمط الطلب.
\end{enumerate}
\end{method}

\begin{example}[بطاطا في فصلين]\label{ex:b1:plant-transport:potato}
في الصيف تكون \omterm{def:b1:flowering-plant-organization:organs}{الأوراق} مصادر والدرنات تحت الأرض مصارف: فيجري
السكروز نازلًا، ويُفرَّغ، ويُحوَّل نشاءً، وتنتفخ الدرنات. وفي
الربيع تنبت الدرنة: فيُعاد نشاؤها سكروزًا، ويُحمَّل في \omterm{def:b1:plant-transport:phloem}{اللحاء}،
ويجري صاعدًا إلى المجموع الخضري النامي --- الأنبوب نفسه، والاتجاه
المعاكس، لأن \omterm{def:b1:plant-transport:phloem}{المصدر} \omterm{def:b1:plant-transport:phloem}{والمصرف} تبادلا المكان.
\end{example}

\section{المساومة}

\begin{proposition}[كفاءة استعمال الماء]\label{prop:b1:plant-transport:wue}
\index{كفاءة استعمال الماء}
كل \omterm{def:b1:plant-transport:stoma}{ثغر} يُدخل ثاني أكسيد الكربون يفقد ماءً، والمقايضة غير متكافئة:
فداخل \omterm{def:b1:flowering-plant-organization:organs}{الورقة} مشبع بالبخار والهواء خارجها جاف، بينما يكون
$\mathrm{CO_2}$ عند \qty{0.04}{\%} خارجًا وأخفض داخلًا، فيكون تدرج
الماء إلى الخارج مئات الأمثال من تدرج ثاني أكسيد الكربون إلى
الداخل. \omterm{def:b1:flowering-plant-organization:organs}{فورقة} C3 تفقد \qtyrange{200}{500}{} جزيء ماء لكل
$\mathrm{CO_2}$ مثبَّت؛ \omterm{def:b1:flowering-plant-organization:organs}{وورقة} C4، وهي تركّز $\mathrm{CO_2}$ وتبقي
ثغورها أضيق، تفقد \qtyrange{100}{200}{}؛ \omterm{def:b1:photosynthesis:c4cam}{ونبات CAM}، وهو لا يفتح
إلا ليلًا، \qtyrange{20}{50}{} (\cref{ch:b1:photosynthesis}).
و\emph{\omterm{prop:b1:plant-transport:wue}{كفاءة استعمال الماء}} هي الكربون المكسوب لكل ماء مفقود،
وتحسينها --- بالإغلاق عند الظهيرة، وبالثغور الغائرة والأديمات
السميكة، وبمسلكي C4 وCAM --- هو جانب النبات في الصفقة مع جو جاف.
\end{proposition}

\begin{example}[الصفقة بالأرقام]\label{ex:b1:plant-transport:bargain}
نبتة ذرة تنتح لترًا ونصفًا في اليوم تثبّت نحو \qty{10}{g} من ثاني
أكسيد الكربون: أي \qty{80}{mol} من الماء مقابل \qty{0.23}{mol} من
$\mathrm{CO_2}$، أي ثلاث مئة وخمسين إلى واحد. وحقل قمح في موسم
ينتح خمس مئة طن من الماء في الهكتار ليصنع ثمانية أطنان من الحَب؛
فالماء ثمن الكربون، وهو في أكثر حقول العالم الثمن الذي يحدّ
المحصول.
\end{example}

\section{مسألة: السكوية}

\begin{problem}[{مسألة نهاية الأسبوع --- شجرة ارتفاعها مئة متر في
يوم صيفي: التوتر عند تاجها، وسرعة عصارتها، والماء الذي تحركه،
والسكر الذي ترسله نازلًا والكربون الذي تشتريه، وختامًا توتر الخشب
عند التاج}]
\label{pb:b1:plant-transport:1}
سكوية ارتفاعها \qty{100}{m} وتاجها \qty{2000}{m^2} من الورق. وفي
يوم صيفي تنتح \qty{600}{L} من الماء في \qty{12}{h} عبر \omterm{def:b1:flowering-plant-organization:tissues}{خشب} مقطعه
\qty{0.5}{m^2}، منه \qty{20}{\%} لمعة ناقلة. والماء:
$\rho = \qty{1000}{kg/m^3}$ و$g =
\qty{9.8}{m/s^2}$ و\qty{18}{g/mol}.
والاحتكاك على طول الجذع يكلف من التوتر قدر ما تكلفه الجاذبية.
\omterm{def:b1:membranes-transport:osmosis}{والكمون المائي} للتربة \qty{-0.1}{MPa}. ويثبّت التاج
\qty{4}{\micro mol} من $\mathrm{CO_2}$ في المتر المربع من الورق في
الثانية مدة \qty{12}{h}؛ والسكروز \qty{342}{g/mol} ويحمل اثنتي عشرة
ذرة كربون؛ وعصارة \omterm{def:b1:plant-transport:phloem}{اللحاء} \qty{15}{\%} سكروزًا بالكتلة (كثافتها
\qty{1.06}{g/mL}) وتجري بسرعة \qty{0.6}{m/h}.

\medskip
\noindent\textbf{الجزء الأول --- التوتر.}
\begin{enumerate}
  \item احسب التوتر اللازم لحمل العمود ضد الجاذبية.
  \item أضف الاحتكاك: فما توتر \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} عند التاج؟
  \item احسب \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي} \omterm{def:b1:flowering-plant-organization:tissues}{للخشب} عند التاج، بأخذ $\Psi_s = 0$
        للعصارة.
  \item ما الذي يجب أن يكون عليه \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي} \omterm{def:b1:cell-unit-of-life:cell}{لخلايا} \omterm{def:b1:flowering-plant-organization:organs}{الورقة} عند
        التاج ليدخلها الماء؟ وإذا كان $\Psi_s = \qty{-2.5}{MPa}$
        لها، فأي \omterm{def:b1:eukaryotic-cell:plantcell}{امتلاء} لها؟
  \item قارن \omterm{def:b1:flowering-plant-organization:organs}{بورقة} على ارتفاع \qty{10}{m} في الشجرة نفسها
        ($\Psi_s =
        \qty{-1.5}{MPa}$).
  \item يحتمل الماء في الأنابيب الدقيقة نحو \qty{-30}{MPa}؛ فلماذا
        لا تنمو الشجرة ببساطة إلى \qty{1000}{m}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الجريان.}
\begin{enumerate}[resume]
  \item احسب معدل النتح باللترات في الساعة وبالأمتار المكعبة في
        الثانية.
  \item احسب المقطع الناقل من \omterm{def:b1:flowering-plant-organization:tissues}{الخشب}.
  \item احسب سرعة العصارة الوسطية.
  \item احسب كتلة الماء المحمولة في اللمعة الناقلة في الجذع
        (\qty{100}{m} منه).
  \item كم يستغرق جزيء ماء ليسافر من \omterm{def:b1:flowering-plant-organization:organs}{الجذر} إلى التاج بتلك السرعة؟
  \item احسب النتح لكل متر مربع من الورق في الساعة، بالغرامات، وفي
        الثانية بالمليمولات.
  \item تجري العصارة نازلة مع تدرج كمون من \qty{-0.1}{} عند التربة
        إلى قيمة التاج في السؤال 3 على \qty{100}{m}. فما التدرج
        بالميغاباسكال لكل متر؟ وكيف يقارَن بالجزء الجاذبي وحده؟
\end{enumerate}

\noindent\textbf{الجزء الثالث --- السكر.}
\begin{enumerate}[resume]
  \item احسب $\mathrm{CO_2}$ الذي يثبّته التاج في اليوم،

  \item احسب السكروز الموافق له، بالمولات وبالكيلوغرامات.
  \item يُتنفس نصفه في التاج؛ ويُرسل الباقي نازلًا في \omterm{def:b1:plant-transport:phloem}{اللحاء}.
        احسب كتلة السكروز المنقول في اليوم وكتلة العصارة التي
        تحمله.
  \item احسب حجم العصارة، وعند \qty{0.6}{m/h} احسب مقطع \omterm{def:b1:flowering-plant-organization:tissues}{الأنابيب
        الغربالية} اللازم لحمله في \qty{24}{h}.
  \item احسب زمن سفر السكروز من التاج إلى \omterm{def:b1:flowering-plant-organization:organs}{الجذر}.
  \item احسب نسبة استعمال الماء: مولات الماء المنتوحة لكل مول من
        $\mathrm{CO_2}$ مثبَّت.
\end{enumerate}

\noindent\textbf{الجزء الرابع --- المساومات.}
\begin{enumerate}[resume]
  \item \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي} لعصارة \omterm{def:b1:plant-transport:phloem}{اللحاء} عند التاج، مع

        وقارنه بكمون \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} هناك. فبأي اتجاه ينتقل الماء بينهما، وهل

  \item تنغلق \omterm{def:b1:plant-transport:stoma}{الثغور} عند الظهيرة فينتصف النتح. أعد حساب توتر

  \item فسّر لماذا تكون أعلى \omterm{def:b1:flowering-plant-organization:organs}{الأوراق} أصغرها وأسمكها.
  \item اذكر النتيجة: توتر \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} عند تاج شجرة ارتفاعها
