\chapter{Las redes alimentarias}\label{ch:g5:food-webs}

El \cref{ch:g3:food-chains} nos avisó de que sus cadenas tan ordenadas
eran hilos sueltos sacados de una red: el ratonero también coge
ratones, la lagartija también come moscas y la hierba también alimenta
a los conejos. Este curso dejamos de tirar de los hilos y miramos la
red entera --- la \omterm{def:g5:food-webs:web}{red alimentaria} --- y lo que la mantiene en
equilibrio.

\section{De las cadenas a las redes}

\begin{definition}[Red alimentaria]\label{def:g5:food-webs:web}
La \emph{red alimentaria}\index{red alimentaria} de un \omterm{def:g2:where-animals-live:habitat}{hábitat} es la
red de \emph{todas} sus \omterm{def:g3:food-chains:chain}{cadenas alimentarias} a la vez: cada \omterm{def:g5:biodiversity:species}{especie}
dibujada una vez, con una flecha (que sigue significando \emph{\omterm{def:g3:food-chains:chain}{es
comido por}}, \cref{rem:g3:food-chains:arrow}) por cada relación de
comida. Casi todos los \omterm{def:g1:animals-around-us:animal}{animales} comen varias \omterm{def:g5:biodiversity:species}{especies} y se los comen
varias --- así que las cadenas se cruzan y el dibujo se convierte en
una red.
\end{definition}

\begin{example}[Leer la red]\label{ex:g5:food-webs:reading}
Dos lecturas de la figura. Sigue las flechas del conejo: come hierba
\emph{y} trébol, y se lo comen el zorro \emph{y} el ratonero --- cuatro
relaciones, donde una cadena enseñaba dos. Y cuenta las flechas de
comida del ratonero: lagartija y conejo --- si pierde una, le queda la
otra. La red registra exactamente esas opciones de repuesto que
elogiaba la \cref{prop:g5:biodiversity:web}.
\end{example}

\section{Los papeles de la red}

\begin{definition}[Productores y consumidores]\label{def:g5:food-webs:roles}
Las \omterm{def:g5:biodiversity:species}{especies} de la red se reparten en papeles:
\begin{itemize}
  \item los \emph{productores}\index{productor}: las \omterm{def:g1:plants-around-us:plant}{plantas}, que
        fabrican su propia materia
        (\cref{prop:g4:what-plants-need:make}) --- todos los caminos
        de flechas arrancan en uno
        (\cref{prop:g3:food-chains:start});
  \item los \emph{consumidores}\index{consumidor}: los \omterm{def:g1:animals-around-us:animal}{animales}, que
        viven de otros --- los \omterm{def:g2:what-animals-eat:kinds}{herbívoros} que comen productores
        directamente, los \omterm{def:g2:what-animals-eat:kinds}{carnívoros} que comen otros consumidores y
        los \omterm{def:g2:what-animals-eat:kinds}{omnívoros} que comen de todo
        (\cref{def:g2:what-animals-eat:kinds});
  \item y, trabajando en la sombra, los
        \emph{descomponedores}\index{descomponedor} --- los que hacen
        pudrirse las cosas en el \cref{ex:g2:caring-for-nature:compost},
        que se alimentan de restos muertos y devuelven su sustancia a
        la tierra. Un curso posterior les dedica un capítulo propio.
\end{itemize}
\end{definition}

\begin{example}[Clasificar la figura]\label{ex:g5:food-webs:sorting}
En la red de bolsillo: \omterm{def:g5:food-webs:roles}{productores} --- hierba, trébol, roble;
\omterm{def:g5:food-webs:roles}{consumidores} \omterm{def:g2:what-animals-eat:kinds}{herbívoros} --- saltamontes, conejo, \omterm{ex:g2:animals-grow-up:butterfly}{oruga}; \omterm{def:g5:food-webs:roles}{consumidores}
\omterm{def:g2:what-animals-eat:kinds}{carnívoros} --- lagartija, herrerillo (casi siempre), zorro (en
realidad \omterm{def:g2:what-animals-eat:kinds}{omnívoro}), ratonero, gavilán. Tres pisos: verde, comedores de
\omterm{def:g1:plants-around-us:plant}{plantas}, comedores de carne --- y cada piso de arriba se apoya en los
de abajo.
\end{example}

\begin{proposition}[La forma de los números]\label{prop:g5:food-webs:pyramid}
Contar los habitantes de la red piso a piso da una forma constante:
\emph{muchos} \omterm{def:g5:food-webs:roles}{productores}, \emph{menos} \omterm{def:g2:what-animals-eat:kinds}{herbívoros}, \emph{pocos}
\omterm{def:g2:what-animals-eat:kinds}{carnívoros} --- un prado de millones de matas de hierba alimenta a
miles de saltamontes, que alimentan a un puñado de lagartijas, que
alimentan a una pareja de ratoneros. Cada piso vive del de abajo y no
puede ser más numeroso que su comida. Por eso los grandes cazadores
son siempre escasos.
\end{proposition}
\admitted

\section{El equilibrio y lo que lo altera}

\begin{proposition}[El equilibrio de la red]\label{prop:g5:food-webs:balance}
Si se la deja en paz, una red mantiene sus números en un
\emph{equilibrio}\index{equilibrio (de una red alimentaria)}
aproximado: si los conejos se multiplican, sus cazadores comen mejor y
se multiplican detrás de ellos, y el excedente se deshace; si los
conejos menguan, los cazadores hambrientos se pasan a otras presas
(las flechas de repuesto) y los conejos se recuperan. Las correcciones
de la \cref{prop:g3:food-chains:depend}, corriendo por muchas flechas
a la vez, estabilizan el conjunto.
\end{proposition}
\admitted

\begin{example}[El error clásico]\label{ex:g5:food-webs:mistake}
Los agricultores hicieron campaña en su día para acabar con las \omterm{prop:g3:sorting-living-things:groups}{aves}
rapaces --- ``ladronas de gallinas''. Donde lo consiguieron, los
ratones y los topillos, libres de sus principales cazadores, se
multiplicaron hasta formar plagas que comieron mucho más grano de lo
que valía la gallina ocasional de los gavilanes. La historia del
caracol del hortelano (\cref{exo:g3:food-chains:9}), a escala de un
país entero: quita el piso de arriba de una red y estalla el de abajo.
\end{example}

\begin{method}[Predecir una alteración]\label{met:g5:food-webs:predict}
Para predecir qué hace quitar (o añadir) una \omterm{def:g5:biodiversity:species}{especie}:
\begin{enumerate}
  \item busca la \omterm{def:g5:biodiversity:species}{especie} en la red y enumera \emph{todas} sus flechas,
        las de entrada y las de salida;
  \item sus alimentos, menos comidos, tienden a aumentar; los que se
        la comían, más pobres, tienden a disminuir o a cambiar de presa;
  \item sigue cada cambio un paso más por la red --- los efectos de
        segundo paso sorprenden a menudo;
  \item comprueba qué vecinos tienen flechas de repuesto (esos absorben
        el golpe) y cuáles dependían solo de esta \omterm{def:g5:biodiversity:species}{especie} (esos están
        en peligro).
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:1}
¿Qué es una \omterm{def:g5:food-webs:web}{red alimentaria} y por qué se cruzan las cadenas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:2}
Nombra los tres papeles de la \cref{def:g5:food-webs:roles}, con un
ejemplo de cada uno sacado de la figura (menos los \omterm{def:g5:food-webs:roles}{descomponedores}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:3}
A partir de la figura: enumera todo lo que come el herrerillo y todo
lo que se lo come.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:4}
Saca de la figura dos cadenas completas que compartan el saltamontes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:5}
Enuncia la forma de los números, piso a piso. ¿Por qué no puede ser
ningún piso más numeroso que el de debajo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:6}
¿Por qué son escasos los ratoneros e incontables los saltamontes, en
una sola frase?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:7}
Los conejos tienen un buen año y se multiplican. Según la
\cref{prop:g5:food-webs:balance}, ¿qué pasa después --- y qué deshace
el excedente?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:food-webs:8}
Pruébalo: dibuja la \omterm{def:g5:food-webs:web}{red alimentaria} de un \omterm{def:g2:where-animals-live:habitat}{hábitat} que conozcas ---
jardín, parque, charca --- con al menos seis \omterm{def:g5:biodiversity:species}{especies} y todas las
flechas que puedas defender. Marca los \omterm{def:g5:food-webs:roles}{productores} en verde.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:food-webs:9}
Pásale el \cref{met:g5:food-webs:predict} a quitar la lagartija de la
figura: efectos de primer paso y después un paso más.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:food-webs:10}
Vuelve a contar el \cref{ex:g5:food-webs:mistake} como una cadena de
predicciones: se quitan los gavilanes, ¿y luego?, ¿y luego? ¿Qué paso
no supieron predecir los que hicieron la campaña?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:food-webs:11}
En la red de la figura, compara quitar el roble (un \omterm{def:g5:food-webs:roles}{productor}) con
quitar el gavilán (un cazador de arriba). ¿Qué vecinos notan primero
cada pérdida, en qué dirección viaja cada alteración y cuál de las dos
pérdidas absorbe la red aquí con más facilidad? Usa los
\cref{ex:g3:food-chains:dry,ex:g3:food-chains:hunter}.
\end{exercise}
