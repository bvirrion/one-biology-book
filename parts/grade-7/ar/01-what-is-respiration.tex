\chapter{ما التنفس؟}\label{ch:g7:what-is-respiration}

ضع حشرة قمل خشبي في مرطبان مغلق، فيتغير هواء المرطبان بعد ساعات:
إذ يذهب بعض أكسجينه ويحلّ محله \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون}. وأعد ذلك \omterm{def:g2:from-seed-to-plant:seed}{ببذرة}
منبتة، وبفطر، وبسمكة ذهبية، وبملعقة \omterm{def:g6:decomposers-and-soil:soil}{تربة} حية --- فالنتيجة نفسها
في كل مرة. وقد أظهرت \cref{prop:g4:breathing:exchange} التبادل في
صدرك أنت؛ وهذه السنة تفتح الفكرة على سعتها كلها: فالتبادل ليس عادة
بشرية بل خاصية للحياة نفسها.

\section{التنفس، معرَّفًا ومقيسًا}

\begin{definition}[التنفس]\label{def:g7:what-is-respiration:respiration}
\emph{التنفس}\index{تنفس} هو التبادل الغازي الذي يأخذ به \omterm{def:g1:living-or-not:living}{الكائن
الحي} \emph{الأكسجين}\index{أكسجين} من محيطه ويطلق
\emph{ثاني أكسيد الكربون}\index{ثاني أكسيد الكربون}. ولا يجوز
الخلط بينه وبين \emph{الحركات التنفسية}\index{حركات تنفسية} ---
أي حركات الصدر المرئية في \cref{def:g4:breathing:movements}:
فتلك الحركات إحدى طرق \emph{خدمة} التنفس؛ أما التنفس نفسه فهو
تبادل الغازين، وهو يجري في كل \omterm{def:g5:first-look-at-cells:cell}{خلية} حية، بحركات أو بغير حركات.
\end{definition}

\begin{proposition}[كيف يُبرهن على التبادل]\label{prop:g7:what-is-respiration:shown}
اختباران قياسيان على المنضدة يكشفان \omterm{def:g7:what-is-respiration:respiration}{التنفس}:
\begin{itemize}
  \item \emph{مِسبار \omterm{def:g7:what-is-respiration:respiration}{أكسجين}} في وعاء مغلق يُظهر مستوى \omterm{def:g7:what-is-respiration:respiration}{الأكسجين}
        هابطًا ما دام \omterm{def:g6:exploring-our-environment:components}{كائن حي} جالسًا في داخله؛
  \item و\emph{ماء الجير}\index{ماء الجير}، وهو سائل صافٍ يعكرّ
        لبنيًّا في \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون}، يتعكّر حين يُمرَّر فيه هواء
        الوعاء.
\end{itemize}
هبوط \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} مع تعكّر \omterm{prop:g7:what-is-respiration:shown}{ماء الجير}، ووعاء شاهد بلا حياة يبقى على
حاله (\cref{met:g4:what-plants-need:fairtest} --- فالاختبار المنصف
يتبعنا في كل مكان): ذلك هو \omterm{def:g7:what-is-respiration:respiration}{التنفس}، مقيسًا.
\end{proposition}
\admitted

\begin{example}[معرض المتنفسين]\label{ex:g7:what-is-respiration:gallery}
أجرِ الاختبارين على مجموعة: فأرة --- هبوط سريع وتعكّر عاجل؛ وحشرة
قمل خشبي --- الشيء نفسه أصغر؛ وبازلاء منبتة --- لا لبس فيه
(فمرطبان من البازلاء الناتشة قد يستنفد أكسجينه في يوم)؛ \omterm{def:g6:decomposers-and-soil:decomposers}{وفطر} ---
نعم؛ وحلزونات بركة --- نعم؛ وملعقة \omterm{def:g6:decomposers-and-soil:soil}{تربة} حية --- نعم، بملاييرها من
المحلِّلين يتنفسون معًا
(\cref{ex:g6:decomposers-and-soil:handful}). ولا يتغير شيء إلا في
مرطبان الشاهد --- حصى ورمل جاف.
\end{example}

\begin{remark}[والنباتات تتنفس أيضًا]\label{rem:g7:what-is-respiration:plants}
قالت \cref{rem:g6:plants-are-producers:gas} ذلك عرضًا؛ والمرطبان
يبرهن عليه: \omterm{def:g1:plants-around-us:plant}{فنبتة} في أصيص \emph{في الظلام} تخفض \omterm{def:g7:what-is-respiration:respiration}{أكسجين} مرطبانها
وتعكّر \omterm{prop:g7:what-is-respiration:shown}{ماء الجير} كأي \omterm{def:g1:animals-around-us:animal}{حيوان}. وفي ضوء النهار تحجب صناعة \omterm{def:g5:food-webs:roles}{المنتِجين}
الأثر --- إذ تأخذ من \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون} أكثر بكثير مما يطلقه
\omterm{def:g7:what-is-respiration:respiration}{التنفس} --- لكن \omterm{def:g7:what-is-respiration:respiration}{التنفس} لا يتوقف أبدًا. فالخضرة لا تعفي \omterm{def:g5:first-look-at-cells:cell}{خلية} من
\omterm{def:g7:what-is-respiration:respiration}{التنفس} بمعناه الواسع؛ ولا شيء حي معفًى.
\end{remark}

\section{لماذا تتنفس الكائنات الحية}

\begin{proposition}[التنفس يشغّل الخلايا]\label{prop:g7:what-is-respiration:power}
مقصود التبادل هو \emph{\omterm{prop:g3:balanced-diet:jobs}{الطاقة}}. ففي كل \omterm{def:g5:first-look-at-cells:cell}{خلية} حية تُحرَق \omterm{def:g4:journey-of-food:digestion}{المغذيات}
(\cref{def:g4:journey-of-food:digestion}) \emph{حرقًا} بطيئًا مع
\omterm{def:g7:what-is-respiration:respiration}{الأكسجين} الذي يجلبه \omterm{def:g7:what-is-respiration:respiration}{التنفس}: فتُحرَّر طاقتها لعمل \omterm{def:g5:first-look-at-cells:cell}{الخلية} ---
انقباضًا، وبناءً، وقيادةً --- ويكون \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون} هو البقية
المحترقة، تُحمل بعيدًا. وقد كانت ``الحصة المحروقة'' في
\cref{prop:g6:matter-from-food:losses} هي هذه بعينها، منظورًا
إليها من الدفتر؛ وهي هنا منظور إليها من \omterm{def:g5:first-look-at-cells:cell}{الخلية}.
\end{proposition}
\admitted

\begin{example}[مقارنة النار، مستعملةً بصدق]\label{ex:g7:what-is-respiration:fire}
لهب الشمعة يستهلك أيضًا \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} ويطلق \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد الكربون} ---
فالاحتراق \omterm{def:g7:what-is-respiration:respiration}{والتنفس} كيمياء واحدة الصنف حقًّا. والفروق تهمّ بالقدر
نفسه: فاحتراق \omterm{def:g5:first-look-at-cells:cell}{الخلية} بطيء بلا لهب، موزّع على آلاف الخطوات الصغيرة،
وطاقته تُلتقط للعمل لا تُفقد ضوءًا ولفحًا. \omterm{def:g7:what-is-respiration:respiration}{فالتنفس} احتراق مستأنَس
--- نار في مقود، في كل \omterm{def:g5:first-look-at-cells:cell}{خلية} تملكها.
\end{example}

\begin{example}[قراءة الطلب]\label{ex:g7:what-is-respiration:demand}
لما كان الاحتراق يشغّل العمل، تبع الطلبُ الجهدَ: فالطفل العادي في
\cref{ex:g4:breathing:running} يتنفس أسرع بكثير من القارئ في
مقعده؛ والبازلاء المنبتة --- وهي تبني نباتًا بأقصى سرعة
(\cref{def:g2:from-seed-to-plant:germination}) --- تتنفس أسرع
بكثير من \omterm{def:g2:from-seed-to-plant:seed}{البذرة} الجافة المنتظرة بجانبها، وتنفسها يكاد يكون صفرًا
سنوات. \omterm{def:g7:what-is-respiration:respiration}{فالتنفس} صوت محرّك الحياة: عالٍ في الجهد والنمو، همسٌ في
الخمود، صامتٌ في الموت وحده.
\end{example}

\section{التنفس في كل موئل}

\begin{proposition}[حاجة واحدة ومحيطات كثيرة]\label{prop:g7:what-is-respiration:habitats}
على كل \omterm{def:g6:exploring-our-environment:components}{كائن حي} أن ينال أكسجينه \emph{من محيطه} --- والمحيطات
تختلف. فالهواء غني \omterm{def:g7:what-is-respiration:respiration}{بالأكسجين}؛ والماء يحمل منه أقل بكثير، مذابًا
(وقد لقيت \cref{exo:g4:breathing:11} النتيجة)؛ \omterm{def:g6:decomposers-and-soil:soil}{والتربة} والطين
تحملان أقل من ذلك. ولذلك يتشكّل مكان عيش \omterm{def:g1:living-or-not:living}{الكائن الحي} وكيفيته بكيفية
حلّه مشكلةَ أكسجينه --- وهو موضوع الفصلين التاليين، في عتاد
\omterm{def:g1:animals-around-us:animal}{الحيوانات} (\cref{ch:g7:how-animals-breathe}) وفي اقتصاد الماء
(\cref{ch:g7:oxygen-in-the-water}).
\end{proposition}
\admitted

\begin{method}[اختبار سؤال في التنفس]\label{met:g7:what-is-respiration:test}
لتختبر أي زعم في شأن \omterm{def:g7:what-is-respiration:respiration}{التنفس} (``\omterm{def:g2:from-seed-to-plant:seed}{البذور} تتنفس أسرع في الدفء''،
``طين البركة يتنفس''):
\begin{enumerate}
  \item أغلق على الموضوع في وعاء، مع مِسبار \omterm{def:g7:what-is-respiration:respiration}{أكسجين} أو مصيدة ماء
        جير؛
  \item وابنِ الشاهد التوأم --- الوعاء نفسه، والموضوع غائب أو بلا
        حياة؛
  \item وغيّر شرطًا واحدًا في المرة، إن كان الزعم يسمّي شرطًا
        (الدفء، أو الندى، أو الظلام)؛
  \item واقرأ الحكم في الفرق: \omterm{def:g7:what-is-respiration:respiration}{أكسجين} هابط، وماء جير متعكّر ---
        في مقابل سكون الشاهد.
\end{enumerate}
\end{method}

\begin{example}[الطريقة على بذرة باردة]\label{ex:g7:what-is-respiration:coldseed}
الزعم: البازلاء المنبتة تتنفس أبطأ في البرد. مرطبانان من البازلاء
الناتشة، أحدهما في دفء الغرفة والآخر في الثلاجة؛ ومصيدتا ماء جير
عليهما؛ وثالث بلا حياة شاهدًا. وفي اليوم التالي: ماء جير المرطبان
الدافئ لبني، وماء جير البارد لم يكد يتغبّش، وماء جير الشاهد صافٍ.
والحكم: \omterm{def:g7:what-is-respiration:respiration}{التنفس} يجري مع وتيرة عمل البازلاء --- وهي قاعدة \omterm{def:g6:exploring-our-environment:conditions}{الشروط} في
\cref{prop:g6:decomposers-and-soil:speed}، مقيسةً بالغاز.
\end{example}

\section{مسألة: المرطبانات المختومة}

\begin{problem}[{مسألة نهاية الأسبوع --- ستة مرطبانات، ولغزان،
وتبادل غازي واحد}]\label{pb:g7:what-is-respiration:1}
لمعرض العلوم يهيّئ الصف ستة مرطبانات مختومة، في كل واحد مِسبار
\omterm{def:g7:what-is-respiration:respiration}{أكسجين} ومصيدة ماء جير: أ --- حصى؛ ب --- بازلاء جافة؛ ج --- بازلاء
منبتة؛ د --- فأرة (مع غذاء وماء، ولساعات قليلة فحسب، تحت مراقبة
المعلم)؛ ه --- \omterm{def:g1:plants-around-us:plant}{نبتة} خضراء في \omterm{def:g6:exploring-our-environment:conditions}{الضوء}؛ و --- \omterm{def:g1:plants-around-us:plant}{النبتة} نفسها في صندوق
مظلم.

\medskip\noindent\textbf{القسم الأول --- التنبؤات.}
\begin{enumerate}
  \item تنبأ بقراءات أ، واذكر عمل أ في التجربة.
  \item وتنبأ بالمرطبانين ب وج، واشرح الفرق مع أن المرطبانين يحملان
        البازلاء ``نفسها''.
  \item وتنبأ بالمرطبان د، واذكر لماذا يحدّ المعلم الساعات.
  \item ويحمل ه وو نبتتين متطابقتين. تنبأ باتجاه \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} في كل
        مرطبان، ووفّق بين الفرقين مستعملًا
        \cref{rem:g7:what-is-respiration:plants}.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- النتائج وقراءتها.}
\begin{enumerate}[resume]
  \item توافق معطيات الصباح التنبؤات --- إلا أن زميلًا يعجب من أن
        \omterm{def:g1:plants-around-us:plant}{نبتة} و ``تتنفس \omterm{def:g1:animals-around-us:animal}{كالحيوان}''. بيّن ما يُظهره و حقًّا، وما
        يخفيه ه.
  \item هبط \omterm{def:g7:what-is-respiration:respiration}{أكسجين} ج ثلاث مرات أسرع من \omterm{def:g7:what-is-respiration:respiration}{أكسجين} ب في الغرام الواحد
        من البازلاء. ترجم عامل الثلاثة إلى مصطلحات
        \cref{ex:g7:what-is-respiration:demand}.
  \item تعكّر ماء جير د أسرع من الجميع. أعطِ السلسلة --- من جسم
        الفأرة الدافئ المتحرك إلى السائل اللبني --- في أربع
        خطوات.
  \item سمِّ الخلاصة الواحدة التي تؤيدها ب وج ود وو جميعًا، في
        جملة تبدأ بعبارة ``كل \omterm{def:g6:exploring-our-environment:components}{كائن حي}\dots''.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- المضي أبعد.}
\begin{enumerate}[resume]
  \item يُقترح مرطبان سابع: شرائح \omterm{def:g6:decomposers-and-soil:decomposers}{فطر}. تنبأ بقراءاته، واذكر أي
        نقطة على مستوى المملكة في
        \cref{ex:g7:what-is-respiration:gallery} يضيفها.
  \item يسأل زوار المعرض لماذا يهمّ المرطبان ج المزارعين. أجب
        بمخزن الحبوب في \cref{exo:g7:what-is-respiration:11}.
  \item مرطبان ثامن: ماء بركة بطينه. فيهبط \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} في الليل.
        فمن في المرطبان يتنفس، ولماذا يهمّ هذا الاكتشاف في سؤال
        \cref{ch:g7:oxygen-in-the-water}؟
  \item أغلق ملصق المعرض بتمييز الفصل: أي المرطبانات أظهرت
        \emph{\omterm{def:g7:what-is-respiration:respiration}{الحركات التنفسية}}، وأيها أظهرت \emph{\omterm{def:g7:what-is-respiration:respiration}{التنفس}}؟
        (وانتبه --- فكلمة واحدة تشملها كلها.)
\end{enumerate}
\end{problem}
