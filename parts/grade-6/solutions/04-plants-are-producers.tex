\section*{Chapter \ref{ch:g6:plants-are-producers} --- Plants Are Producers}

\begin{solution}{exo:g6:plants-are-producers:1}
\omterm{def:g6:plants-are-producers:matter}{Mineral matter}: matter of the non-living world --- water,
rock (also air's gases, \omterm{prop:g4:what-plants-need:needs}{mineral salts}). \omterm{def:g6:plants-are-producers:matter}{Organic matter}:
matter made by \omterm{def:g6:exploring-our-environment:components}{living things} and composing their bodies ---
wood, \omterm{def:g3:bones-and-muscles:muscle}{muscle} (also sugar, wool, fat).
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:2}
Inputs: water and \omterm{prop:g4:what-plants-need:needs}{mineral salts} from the soil, a gas (carbon
dioxide) from the air. \omterm{prop:g3:balanced-diet:jobs}{Energy}: \omterm{def:g6:exploring-our-environment:conditions}{light}, caught by the green
substance. Output: \omterm{def:g6:plants-are-producers:matter}{organic matter} --- sugar, then wood, \omterm{def:g1:plants-around-us:parts}{leaf},
\omterm{def:g1:plants-around-us:parts}{root}, \omterm{def:g2:from-seed-to-plant:seed}{seed}. Workshop: the green \omterm{def:g1:plants-around-us:parts}{leaf}, by day.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:6}
The potato \omterm{ex:g4:plants-without-seeds:bulbtuber}{tuber} and the tulip \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulb} (underground stores), the

\end{solution}

\begin{solution}{exo:g6:plants-are-producers:7}
Because the manufacture runs on \omterm{def:g6:exploring-our-environment:conditions}{light}: with no \omterm{def:g6:exploring-our-environment:conditions}{light} there is
no \omterm{prop:g3:balanced-diet:jobs}{energy} to build \omterm{def:g6:plants-are-producers:matter}{organic matter}, whatever minerals wait in
the soil. \omterm{def:g6:exploring-our-environment:conditions}{Light} is the one input with no substitute --- the
salts are seasoning, not fuel.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:8}
Because a chain's first link must put new \omterm{def:g6:plants-are-producers:matter}{organic matter} into
the living world, and \omterm{def:g1:animals-around-us:animal}{animals} are \omterm{def:g5:food-webs:roles}{consumers}: they can only
transform \omterm{def:g6:plants-are-producers:matter}{organic matter} already made by someone else. Only
\omterm{def:g5:food-webs:roles}{producers} manufacture it from mineral supplies --- so the
first link is always green.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:9}
Bread: wheat --- the wheat \omterm{def:g1:plants-around-us:plant}{plant}'s \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}; trail ends in the
wheat \omterm{def:g1:plants-around-us:parts}{leaf}. Butter: cream --- \omterm{def:g2:animals-grow-up:born}{milk} --- the cow --- grass;
trail ends in the grass \omterm{def:g1:plants-around-us:parts}{leaf}. Cheese: \omterm{def:g2:animals-grow-up:born}{milk} --- the cow ---
grass again. Three foods, three (or two) \omterm{def:g1:plants-around-us:parts}{leaves}, all in the
\omterm{def:g6:exploring-our-environment:conditions}{light}.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:10}
Hay is the cow's food --- \omterm{def:g6:plants-are-producers:matter}{organic matter} she transforms into
her body. Compost is not the \omterm{def:g1:plants-around-us:plant}{plant}'s food: it restocks the
soil's \emph{mineral} salts as it rots. The \omterm{def:g1:plants-around-us:plant}{plant} then makes
its own \omterm{def:g6:plants-are-producers:matter}{organic matter} from those minerals, air, water and
\omterm{def:g6:exploring-our-environment:conditions}{light}. The cow eats meals; the \omterm{def:g1:plants-around-us:plant}{plant} manufactures them.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:11}
Lit, the \omterm{def:g1:plants-around-us:plant}{plants}' daytime manufacture releases oxygen and
takes up carbon dioxide, balancing the \omterm{prop:g3:sorting-living-things:groups}{fish}'s \omterm{def:g4:breathing:movements}{breathing} ---
the gas swap runs both ways and the jar cycles. Dark, the
manufacture stops while every inhabitant (\omterm{def:g1:plants-around-us:plant}{plants} included)
keeps \omterm{def:g4:breathing:movements}{breathing}: oxygen runs out, carbon dioxide piles up,
and the sealed world suffocates.
\end{solution}

\begin{solution}{exo:g6:plants-are-producers:12}
Right: the tree's matter did not come from the soil --- his
weighing proved that forever. Incomplete: water was only one
bulk ingredient; he could not know about the gas taken from
the air, which supplies much of the wood's substance. The
modern addition: carbon dioxide from the air (plus the
soil's small \omterm{prop:g4:what-plants-need:needs}{mineral salts}) --- water and air together,
assembled by \omterm{def:g6:exploring-our-environment:conditions}{light}.
\end{solution}

\begin{solution}{pb:g6:plants-are-producers:1}
\textbf{1.} The tonnes are \omterm{def:g6:plants-are-producers:matter}{organic matter} manufactured by
the \omterm{def:g1:plants-around-us:plant}{plants} themselves from mineral supplies --- water, the
air's carbon dioxide, the soil's salts --- with \omterm{def:g6:exploring-our-environment:conditions}{light} as the
\omterm{prop:g3:balanced-diet:jobs}{energy}. The sacks restock minerals; the air and rain deliver
the bulk; the \omterm{def:g1:plants-around-us:parts}{leaves} do the making.

\textbf{2.} Potato: underground storage organ (\omterm{ex:g4:plants-without-seeds:bulbtuber}{tuber}).
Carrot: storage \omterm{def:g1:plants-around-us:parts}{root}. Bean \omterm{def:g2:from-seed-to-plant:seed}{seed}: the \omterm{def:g2:from-seed-to-plant:seed}{seed}'s \omterm{def:g3:animals-through-seasons:active}{food store},
packed for the next \omterm{def:g1:plants-around-us:plant}{plant}. Lettuce \omterm{def:g1:plants-around-us:parts}{leaf}: the working organ
itself --- the light-catching workshop, eaten fresh.

\textbf{3.} The true supplier is \omterm{def:g6:exploring-our-environment:conditions}{light}: every gram of yield
was built with \omterm{def:g6:exploring-our-environment:conditions}{light}'s \omterm{prop:g3:balanced-diet:jobs}{energy}, by day, in the \omterm{def:g1:plants-around-us:parts}{leaves}.

\textbf{4.} For the \omterm{prop:g4:what-plants-need:needs}{mineral salts}: harvests carry them off,
and without restocking the \omterm{def:g1:plants-around-us:plant}{plants} would run mineral-short
like the mint in sand --- small in bulk, indispensable in
function.

\textbf{5.} The dark-cupboard experiment (the pale
prisoner): the cloth removed \omterm{def:g6:exploring-our-environment:conditions}{light}, and a week of it already
shows --- pale, stretched, weak manufacture.

\textbf{6.} The \omterm{ex:g4:plants-without-seeds:bulbtuber}{tubers} are filled by the tops: \omterm{def:g1:plants-around-us:parts}{leaves}
manufacture sugar by day, and the surplus is piped down and
banked in the swelling \omterm{ex:g4:plants-without-seeds:bulbtuber}{tubers}. Lush working \omterm{def:g1:plants-around-us:parts}{leaves} mean a
big surplus --- good tops, good \omterm{ex:g4:plants-without-seeds:bulbtuber}{tubers}.

\textbf{7.} Water. Manufacture needs all its inputs at once:
\omterm{def:g6:exploring-our-environment:conditions}{light} supplies \omterm{prop:g3:balanced-diet:jobs}{energy}, not substance --- with the \omterm{def:g1:plants-around-us:parts}{roots} dry,
there is nothing to build with, however bright the sky.

\textbf{8.} The dug-in \omterm{def:g1:plants-around-us:parts}{roots} are \omterm{def:g6:plants-are-producers:matter}{organic matter} returned to
the soil: rotting, they will free their minerals for coming
crops --- the return rule of
\cref{prop:g5:people-and-nature:lasting}, paid in \omterm{def:g1:plants-around-us:plant}{plant}
matter. The waste would be burning them.

\textbf{9.} Potato \omterm{def:g1:plants-around-us:plant}{plant} $\rightarrow$ pig $\rightarrow$ the
family. All its \omterm{def:g6:plants-are-producers:matter}{organic matter} was first manufactured in the
potato \omterm{def:g1:plants-around-us:plant}{plant}'s \omterm{def:g1:plants-around-us:parts}{leaves} --- the pig and the family only
transformed it.

\textbf{10.} Ingredients: water and \omterm{prop:g4:what-plants-need:needs}{mineral salts} from the
soil, carbon dioxide from the air --- assembled into \omterm{def:g1:plants-around-us:parts}{leaf}
matter. The sun's exact role: \omterm{prop:g3:balanced-diet:jobs}{energy}, not substance. Her
salad is water, air and earth, \emph{built by} sunlight ---
poetic credit deserved, ingredient list wrong.

\textbf{11.} The \omterm{def:g5:food-webs:roles}{decomposers} --- soil life from worms and
woodlice down to \omterm{def:g6:exploring-our-environment:components}{living things} too small to see --- feed on
the dead \omterm{def:g1:plants-around-us:parts}{leaves} and break their \omterm{def:g6:plants-are-producers:matter}{organic matter} down toward
\omterm{def:g6:plants-are-producers:matter}{mineral matter} again, restocking the soil the next crops
will draw on.

\textbf{12.} For example: ``The garden's \omterm{def:g5:food-webs:roles}{producers} build
\omterm{def:g6:plants-are-producers:matter}{organic matter} --- the harvest --- out of \omterm{def:g6:plants-are-producers:matter}{mineral matter}
from soil, water and air, using \omterm{def:g6:exploring-our-environment:conditions}{light} as their \omterm{prop:g3:balanced-diet:jobs}{energy}.''
\end{solution}
