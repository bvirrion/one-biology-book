\section*{Chapter \ref{ch:g11:mendelian-genetics} --- Mendelian Genetics}

\begin{solution}{exo:g11:mendelian-genetics:1}
An \omterm{def:g11:mendelian-genetics:allele}{allele} is one alternative version of a \omterm{def:g11:gene-expression:gene}{gene}. The \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} is the allelic
composition of an individual (for one \omterm{def:g11:gene-expression:gene}{gene} or many). The \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype} is the
observable trait that results from \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} interacting with the environment.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:2}
Under complete dominance there are three \omterm{def:g11:mendelian-genetics:geno-pheno}{genotypes}: $AA$ and $Aa$ both show the
\omterm{def:g11:mendelian-genetics:dominance}{dominant} \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}, while $aa$ shows the \omterm{def:g11:mendelian-genetics:dominance}{recessive} \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}. \omterm{def:g11:mendelian-genetics:geno-pheno}{Phenotype} alone
cannot distinguish $AA$ from $Aa$; a \omterm{def:g11:mendelian-genetics:testcross}{test cross} or progeny test is needed.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:3}
The two \omterm{def:g11:mendelian-genetics:allele}{alleles} of a \omterm{def:g11:gene-expression:gene}{gene} separate during gamete formation so that each gamete
carries only one \omterm{def:g11:mendelian-genetics:allele}{allele} of that \omterm{def:g11:gene-expression:gene}{gene}. Random fertilisation then pairs one \omterm{def:g11:mendelian-genetics:allele}{allele}
from each parent in the zygote. Segregation is the genetic reading of \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}
separating homologous \omterm{def:g11:dna-replication:chromosome}{chromosomes}.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:4}
The tall parent $Tt$ produces gametes $T$ and $t$; the dwarf parent $tt$
produces only $t$. Offspring \omterm{def:g11:mendelian-genetics:geno-pheno}{genotypes} are $\tfrac12\,Tt$ and $\tfrac12\,tt$.
\omterm{def:g11:mendelian-genetics:geno-pheno}{Phenotypes} are $\tfrac12$ tall and $\tfrac12$ dwarf --- a $1:1$ test-cross
style ratio.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:5}
The tall parent must have been \omterm{def:g11:mendelian-genetics:zygous}{heterozygous} $Tt$. Only $Tt \times tt$ yields
about half dwarf ($tt$) offspring. A \omterm{def:g11:mendelian-genetics:zygous}{homozygous} $TT$ parent would have produced
only tall $Tt$ offspring.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:6}
Among offspring of $Aa \times Aa$, \omterm{def:g11:mendelian-genetics:geno-pheno}{genotypes} occur in the ratio
$1\,AA:2\,Aa:1\,aa$. If $A$ is completely \omterm{def:g11:mendelian-genetics:dominance}{dominant}, $AA$ and $Aa$ share one
\omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}, so the four genotypic parts collapse into a $3:1$ phenotypic ratio.
That pattern is classic evidence for a single \omterm{def:g11:gene-expression:gene}{gene} with \omterm{def:g11:mendelian-genetics:dominance}{dominant} and \omterm{def:g11:mendelian-genetics:dominance}{recessive
alleles}.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:7}
For unlinked loci, $AaBb$ produces four gamete types --- $AB$, $Ab$, $aB$ and
$ab$ --- each with frequency $\tfrac14$, because each locus segregates $1:1$
and assortment is independent.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:8}
Green wrinkled seeds have \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} $yyrr$. From $YyRr \times YyRr$ with
independent assortment, $P(yy)=\tfrac14$ and $P(rr)=\tfrac14$, so
$P(yyrr)=\tfrac14\times\tfrac14=\tfrac1{16}$.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:9}
Both black parents must be \omterm{def:g11:mendelian-genetics:zygous}{heterozygous} $Bb$ to produce white $bb$ offspring.
Among black offspring (\omterm{def:g11:mendelian-genetics:geno-pheno}{genotypes} $BB$ and $Bb$ in the ratio $1:2$ within the
$\tfrac34$ that are black), the fraction that are \omterm{def:g11:mendelian-genetics:zygous}{heterozygous} is $\tfrac23$.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:10}
The double heterozygote $AaBb$ makes four equally likely gametes; the tester
$aabb$ contributes only $ab$. Offspring $AaBb$ arises only when the $AB$ gamete
is used, fraction $\tfrac14$. Both \omterm{def:g11:mendelian-genetics:dominance}{dominant} \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes} require $A\_B\_$, which
in this cross is exactly the $AaBb$ \omterm{def:g10:biodiversity-classification:hierarchy}{class}, also $\tfrac14$.
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:11}
Pink $C^{R}C^{W}$ gametes are $\tfrac12\,C^{R}$ and $\tfrac12\,C^{W}$; white
$C^{W}C^{W}$ gametes are all $C^{W}$. Offspring: $\tfrac12\,C^{R}C^{W}$ (pink)
and $\tfrac12\,C^{W}C^{W}$ (white). Under complete dominance a monohybrid
heterozygote $\times$ \omterm{def:g11:mendelian-genetics:dominance}{recessive} would give only two \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes} in a $1:1$ ratio
as here, but selfing two incomplete-dominance heterozygotes yields three
\omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes} ($1:2:1$), whereas complete dominance yields two ($3:1$).
\end{solution}

\begin{solution}{exo:g11:mendelian-genetics:12}
(1)~The man is type A with type O mother, so he must be $I^{A}i$ (he received
$i$ from his mother). The woman is type AB, \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} $I^{A}I^{B}$.
(2)~Gametes: man $I^{A}$ or $i$; woman $I^{A}$ or $I^{B}$. Children:
$\tfrac14\,I^{A}I^{A}$ (A), $\tfrac14\,I^{A}I^{B}$ (AB), $\tfrac14\,I^{A}i$ (A),
$\tfrac14\,I^{B}i$ (B). Phenotypic probabilities: type A $\tfrac12$, type AB
$\tfrac14$, type B $\tfrac14$.
(3)~No type O ($ii$) child is possible: the AB parent never contributes $i$, so
the child cannot be $ii$.
\end{solution}

\begin{solution}{pb:g11:mendelian-genetics:1}
\textbf{1.} Cross the red smooth plant to true-breeding white serrated
$rrss$. If the unknown plant is $RrSs$, expect four \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes} in roughly equal
numbers: red smooth, red serrated, white smooth and white serrated --- the
dihybrid test-cross ratio $1:1:1:1$.

\textbf{2.} If the plant is $RRSS \times rrss$, all offspring are $RrSs$ and
therefore all red smooth.

\textbf{3.} From $RrSs \times rrss$, a red smooth offspring must be $RrSs$. It
is still \omterm{def:g11:mendelian-genetics:zygous}{heterozygous} at both loci and is not true-breeding.

\textbf{4.} Intercross or self red smooth plants; select red smooth progeny;
progeny-test (or test-cross) candidates until individuals never produce white
or serrated offspring --- those lines are $RRSS$. Repeated selfing with selection
and testing also converges on true-breeding stock.

\textbf{5.} White smooth is $rrS\_$: probability $rr$ is $\tfrac14$ and
probability $S\_$ is $\tfrac34$, so $\tfrac14\times\tfrac34=\tfrac3{16}$.

\textbf{6.} Tight linkage makes parental gamete types far more common than
recombinants, so the four phenotypic classes are not $9:3:3:1$; ratios depend
on recombination frequency between the loci.
\end{solution}
