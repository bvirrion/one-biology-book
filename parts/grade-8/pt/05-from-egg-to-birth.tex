\chapter{Da célula reprodutora ao nascimento}\label{ch:g8:from-egg-to-birth}

O \cref{ch:g5:human-reproduction} contou os nove meses como uma
história; agora você tem todos os instrumentos de que a história
precisa para ser tocada direito: as \omterm{def:g5:first-look-at-cells:cell}{células} e as divisões delas, os
dois \omterm{def:g8:sexual-reproduction:gametes}{gametas} e o lugar do encontro deles, os \omterm{def:g8:puberty:hormones}{hormônios}, as entregas do
\omterm{prop:g4:journey-of-food:absorb}{sangue}. Este capítulo repassa o desenvolvimento humano com a
maquinaria à mostra --- de uma \omterm{def:g5:first-look-at-cells:cell}{célula} a três mil bilhões, em quarenta
semanas.

\section{A primeira semana: do zigoto ao hóspede}

\begin{proposition}[A viagem pela tuba uterina]\label{prop:g8:from-egg-to-birth:firstweek}
A \omterm{def:g5:flower-to-fruit:steps}{fecundação} na \omterm{def:g8:reproductive-systems:female}{tuba uterina}
(\cref{def:g8:reproductive-systems:female}) dispara dois relógios ao
mesmo tempo:
\begin{enumerate}
  \item a \emph{divisão}: o \omterm{prop:g8:sexual-reproduction:core}{zigoto} se divide em menos de um dia e
        depois de novo e de novo
        (\cref{ex:g5:first-look-at-cells:one}) --- duas, quatro, oito
        \omterm{def:g5:first-look-at-cells:cell}{células}, uma bolinha parecida com uma amora, de dezenas de
        \omterm{def:g5:first-look-at-cells:cell}{células}, no quarto dia, tudo isso enquanto desce à deriva pela
        \omterm{def:g8:reproductive-systems:female}{tuba uterina} rumo ao \omterm{prop:g5:human-reproduction:plan}{útero};
  \item a \emph{implantação}: por volta do sétimo dia a bolinha chega
        ao \omterm{prop:g5:human-reproduction:plan}{útero} e se encaixa no revestimento preparado (o
        acontecimento 4 da \cref{prop:g8:reproductive-systems:cycle})
        --- o hóspede chega, o ciclo para e os \omterm{def:g8:puberty:hormones}{hormônios} avisam o
        sistema inteiro: \omterm{def:g5:human-reproduction:pregnancy}{gravidez}.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Do que vivem as primeiras divisões]\label{ex:g8:from-egg-to-birth:provisions}
Durante a primeira semana, a viajante não come refeição nenhuma nem
recebe entrega nenhuma: ela vive das próprias provisões da \omterm{def:g4:animal-reproduction:sexual}{célula
reprodutora feminina} --- exatamente para isso servia o abastecimento do
\cref{ex:g8:reproductive-systems:egg}. O \omterm{def:g8:reproductive-systems:male}{espermatozoide} trouxe o
\omterm{def:g6:cell-unit-of-life:parts}{núcleo} do pai e pouco mais; a \omterm{def:g5:first-look-at-cells:cell}{célula} feminina trouxe o outro \omterm{def:g6:cell-unit-of-life:parts}{núcleo}
\emph{e} a despensa. A implantação é, entre outras coisas, a despensa
acabando e o hóspede se ligando ao serviço de quarto.
\end{example}

\section{Embrião, placenta, feto}

\begin{definition}[A placenta]\label{def:g8:from-egg-to-birth:placenta}
Da bolinha implantada crescem duas construções ao mesmo tempo: o
próprio \emph{\omterm{def:g5:human-reproduction:pregnancy}{embrião}} e a interface de suporte de vida dele --- a
\emph{placenta}\index{placenta}, um disco que cresce dentro da parede
do \omterm{prop:g5:human-reproduction:plan}{útero}, onde o \omterm{prop:g4:journey-of-food:absorb}{sangue} da mãe e o \omterm{prop:g4:journey-of-food:absorb}{sangue} do \omterm{def:g5:human-reproduction:pregnancy}{embrião} chegam
intimamente perto \emph{sem se misturar}. Pelas superfícies vastas e
finas dela --- a especificação da
\cref{prop:g7:how-animals-breathe:spec}, construída mais uma vez ---
\omterm{def:g7:digestion-and-nutrients:families}{nutrientes} e \omterm{def:g7:what-is-respiration:respiration}{oxigênio} atravessam para o \omterm{def:g5:human-reproduction:pregnancy}{embrião}, e resíduos e \omterm{def:g7:what-is-respiration:respiration}{gás
carbônico} atravessam de volta (o cordão da
\cref{prop:g5:human-reproduction:cord} corre entre a placenta e o
\omterm{def:g5:human-reproduction:pregnancy}{embrião}). A placenta é \omterm{def:g7:how-animals-breathe:four}{pulmão}, intestino e \omterm{def:g7:removing-wastes:kidneys}{rim} para o passageiro ---
as três fronteiras num órgão só.
\end{definition}

\begin{proposition}[O calendário, revisitado com instrumentos]\label{prop:g8:from-egg-to-birth:timetable}
Os marcos do \cref{ex:g5:human-reproduction:timetable}, agora
explicados:
\begin{itemize}
  \item semanas 1 a 8, o \emph{\omterm{def:g5:human-reproduction:pregnancy}{embrião}}: o plano do corpo é traçado
        --- as \omterm{def:g5:first-look-at-cells:cell}{células} se dividem, migram e se \emph{especializam}
        (\cref{rem:g6:cell-unit-of-life:twoways}) em \omterm{def:g4:heart-and-blood:heart}{coração}, \omterm{def:g5:brain-in-command:system}{nervo},
        \omterm{def:g3:bones-and-muscles:skeleton}{osso} e todo o resto; o \omterm{def:g4:heart-and-blood:heart}{coração} bate desde a quarta semana,
        porque a economia da despensa e da \omterm{def:g8:from-egg-to-birth:placenta}{placenta} precisa cedo da
        bomba dela;
  \item semanas 9 a 40, o \emph{\omterm{def:g5:human-reproduction:pregnancy}{feto}}: com o plano construído, vêm o
        crescimento e o amadurecimento --- o \omterm{def:g5:brain-in-command:system}{cérebro} se religando, os
        \omterm{def:g7:how-animals-breathe:four}{pulmões} ensaiando vazios
        (o \cref{exo:g5:human-reproduction:11} explicou a estreia
        deles no primeiro choro), os chutes do quarto mês anunciando
        \omterm{def:g3:bones-and-muscles:muscle}{músculos} em funcionamento;
  \item o tempo todo: os \omterm{def:g8:puberty:hormones}{hormônios} mantêm o \omterm{prop:g5:human-reproduction:plan}{útero} quieto e o ciclo
        suspenso --- a química da
        \cref{def:g8:puberty:hormones}, cuidando da administração da
        \omterm{def:g5:human-reproduction:pregnancy}{gravidez}.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Por que as primeiras semanas pedem mais cuidado]\label{ex:g8:from-egg-to-birth:earlycare}
A \cref{prop:g5:human-reproduction:care} avisou que o tabaco e o
álcool atravessam até o bebê; o calendário diz quando isso importa
mais: nas semanas de \omterm{def:g5:human-reproduction:pregnancy}{embrião}, em que o plano está sendo traçado e cada
órgão é uma pequena multidão de \omterm{def:g5:first-look-at-cells:cell}{células} se especializando, uma
instrução perturbada pode ecoar pela construção inteira --- e essas são
justamente as semanas em que uma \omterm{def:g5:human-reproduction:pregnancy}{gravidez} pode passar despercebida.
Daí a aritmética direta dos médicos: o cuidado começa não no primeiro
exame de imagem, mas na primeira possibilidade.
\end{example}

\section{O nascimento e a troca de fronteiras}

\begin{proposition}[O nascimento como uma troca ordenada por hormônios]\label{prop:g8:from-egg-to-birth:birth}
Por volta da semana quarenta, sinais hormonais --- os do bebê e os da
mãe em diálogo --- encerram o longo silêncio do \omterm{prop:g5:human-reproduction:plan}{útero}:
\begin{enumerate}
  \item o \emph{trabalho de parto}: o \omterm{prop:g5:human-reproduction:plan}{útero}, o \omterm{def:g3:bones-and-muscles:muscle}{músculo} mais forte do
        corpo (a \cref{rem:g8:reproductive-systems:living} encontrou o
        trabalho mais brando dele), se contrai em ondas cada vez mais
        fortes que abrem o caminho e empurram o bebê pelo canal de
        parto;
  \item o \emph{\omterm{def:g2:born-grow-die:cycle}{nascimento}}: em geral com a cabeça primeiro
        (\cref{ex:g5:human-reproduction:birth});
  \item a \emph{troca}: o primeiro choro enche os \omterm{def:g7:how-animals-breathe:four}{pulmões} nunca
        usados; o tráfego do cordão termina e o cordão é cortado; em
        poucos minutos o recém-nascido funciona com os próprios
        \omterm{def:g7:how-animals-breathe:four}{pulmões}, com o intestino que vem em seguida e com os \omterm{def:g7:removing-wastes:kidneys}{rins} ---
        todas as fronteiras que a \omterm{def:g8:from-egg-to-birth:placenta}{placenta} cobria
        (\cref{exo:g5:human-reproduction:11}, agora com o mapa
        completo);
  \item a \emph{dequitação}: a \omterm{def:g8:from-egg-to-birth:placenta}{placenta}, com o serviço feito, se
        solta e sai atrás --- o único órgão que o corpo constrói para
        um uso só e descarta.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[A primeira auditoria do recém-nascido]\label{ex:g8:from-egg-to-birth:audit}
As parteiras dão nota ao recém-nascido poucos minutos depois do parto
--- \omterm{def:g7:what-is-respiration:respiration}{respiração}, batimento, cor, movimento, choro. Leia essa lista com
os \omterm{def:g1:the-five-senses:sense}{olhos} deste livro: ela é o ciclo de abastecimento e eliminação da
\cref{prop:g7:removing-wastes:complete}, auditado na primeira volta a
\omterm{def:g6:decomposers-and-soil:soil}{solo} dele. A primeira prova que todo ser humano faz é um exame de
fisiologia.
\end{example}

\begin{method}[Explicar qualquer etapa da gravidez]\label{met:g8:from-egg-to-birth:explain}
Diante de uma pergunta sobre qualquer momento dos nove meses, responda
com quatro instrumentos:
\begin{enumerate}
  \item as \emph{\omterm{def:g5:first-look-at-cells:cell}{células}}: o que as divisões e as especializações
        estão fazendo?
  \item o \emph{abastecimento}: despensa ou \omterm{def:g8:from-egg-to-birth:placenta}{placenta} --- o que
        atravessa e o que não pode atravessar?
  \item os \emph{\omterm{def:g8:puberty:hormones}{hormônios}}: o que está sendo mantido, ou ordenado?
  \item o \emph{cuidado}: o que essa etapa exige dos hábitos da mãe e
        do calendário dos médicos?
\end{enumerate}
\end{method}

\section{Problema: A linha do tempo da parteira}

\begin{problem}[{Problema de fim de semana --- uma gravidez
acompanhada consulta a consulta}]\label{pb:g8:from-egg-to-birth:1}
A ficha de uma parteira para uma \omterm{def:g5:human-reproduction:pregnancy}{gravidez} (inventada): consulta de
abertura na semana 8, exames de imagem nas semanas 12 e 20, controles
nas semanas 28 e 36, parto na semana 40. Leia a ficha com os
instrumentos do capítulo.

\medskip\noindent\textbf{Parte I --- As primeiras páginas
da ficha.}
\begin{enumerate}
  \item A \omterm{def:g5:human-reproduction:pregnancy}{gravidez} foi confirmada por um teste hormonal semanas antes
        da primeira consulta. Que sinal um teste desses lê, e a
        partir de que acontecimento da
        \cref{prop:g8:from-egg-to-birth:firstweek} esse sinal existe?
  \item Na semana 8 a parteira registra ``\omterm{def:g5:human-reproduction:pregnancy}{embrião} completo em plano,
        do tamanho de um feijão''. Traduza para os termos do
        calendário --- que fase se encerra aqui?
  \item A orientação da consulta de abertura começa com nada de
        álcool, nada de fumaça --- e ácido fólico, uma vitamina de
        obra, ``de preferência desde antes do começo''. Justifique as
        três coisas a partir do
        \cref{ex:g8:from-egg-to-birth:earlycare}.
  \item O exame da semana 12 mostra o batimento que começou na semana
        4. Por que tão cedo, em termos de abastecimento?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Os meses do meio.}
\begin{enumerate}[resume]
  \item O exame da semana 20 mede cada órgão e a posição da \omterm{def:g8:from-egg-to-birth:placenta}{placenta}.
        Diga o assunto da fase de \omterm{def:g5:human-reproduction:pregnancy}{feto} e os três trabalhos da \omterm{def:g8:from-egg-to-birth:placenta}{placenta}
        que estão sendo conferidos.
  \item A mãe relata os primeiros chutes perto da semana 18. Que
        maquinaria está se anunciando (a
        \cref{prop:g5:brain-in-command:loop} dá o nome da cadeia)?
  \item Na semana 28 a parteira confere o açúcar e a pressão do
        \omterm{prop:g4:journey-of-food:absorb}{sangue} da mãe. Por que a circulação da mãe está com carga
        dupla --- as fronteiras de quem ela está operando?
  \item Semanas 30 a 36: o \omterm{def:g5:human-reproduction:pregnancy}{feto} basicamente cresce e os \omterm{def:g7:how-animals-breathe:four}{pulmões}
        ensaiam. Ensaiam o quê --- e para que momento único?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- A última página.}
\begin{enumerate}[resume]
  \item Semana 40: trabalho de parto. Diga o \omterm{def:g3:bones-and-muscles:muscle}{músculo}, as duas tarefas
        dele e quem ordenou o trabalho
        (\cref{prop:g8:from-egg-to-birth:birth}).
  \item A anotação do parto na ficha: ``choro forte na hora; cordão
        pinçado e cortado; \omterm{def:g8:from-egg-to-birth:placenta}{placenta} completa''. Audite as três frases
        com a troca de fronteiras.
  \item O recém-nascido tira nota alta na lista de conferência da
        escala de minutos. Diga o que essa lista atesta, nos termos do
        \cref{ex:g8:from-egg-to-birth:audit}.
  \item Encerre a ficha com uma frase para os pais: o que as quarenta
        semanas construíram, a partir de que \omterm{def:g5:first-look-at-cells:cell}{célula} única, e por meio
        de que maquinaria.
\end{enumerate}
\end{problem}
