\chapter{Los cromosomas}\label{ch:g9:chromosomes}

La \cref{rem:g9:heredity-and-traits:question} acorraló a los
determinantes en el \omterm{def:g6:cell-unit-of-life:parts}{núcleo}. Sorprende a una \omterm{def:g5:first-look-at-cells:cell}{célula} en el momento justo
y el \omterm{def:g6:cell-unit-of-life:parts}{núcleo} enseña sus cartas: su contenido se enrolla en cuerpos
contables y fotografiables --- los \omterm{def:g9:chromosomes:chromosomes}{cromosomas}. Contarlos, en las
\omterm{def:g5:first-look-at-cells:cell}{células} del cuerpo y en los \omterm{def:g8:sexual-reproduction:gametes}{gametos}, resulta explicar casi todo lo que
exigía el álbum de familia: las mitades, las parejas, el barajado y la
moneda al aire que decide si un bebé es niño o niña.

\section{El núcleo, pillado en el acto}

\begin{definition}[Los cromosomas]\label{def:g9:chromosomes:chromosomes}
Un \emph{cromosoma}\index{cromosoma} es un cuerpo con forma de hilo del
\omterm{def:g6:cell-unit-of-life:parts}{núcleo}, visible (bien teñido, en el momento de la división) cuando el
contenido del \omterm{def:g6:cell-unit-of-life:parts}{núcleo} se enrolla para el transporte. Entre división y
división esos mismos hilos corren sueltos e invisibles --- estado de
trabajo; al dividirse se condensan --- estado de viaje. Fotografiados y
ordenados, los cromosomas de una \omterm{def:g5:first-look-at-cells:cell}{célula} forman su
\emph{cariotipo}\index{cariotipo}: la baraja entera, extendida.
\end{definition}

\begin{proposition}[Las cuentas humanas]\label{prop:g9:chromosomes:counts}
Los \omterm{def:g9:chromosomes:chromosomes}{cariotipos} entregan tres números que sostienen el capítulo entero:
\begin{enumerate}
  \item toda \omterm{def:g5:first-look-at-cells:cell}{célula} del cuerpo humano tiene \emph{46} \omterm{def:g9:chromosomes:chromosomes}{cromosomas} ---
        \omterm{def:g1:the-five-senses:sense}{piel}, \omterm{prop:g7:digestion-and-nutrients:absorption}{hígado} y \omterm{def:g8:nervous-communication:neuron}{neurona} por igual (las divisiones de la
        \cref{prop:g5:first-look-at-cells:growth} copian la baraja
        fielmente a todas las \omterm{def:g5:first-look-at-cells:cell}{células});
  \item extendidos, los 46 se ordenan en \emph{23 parejas}, y los dos
        miembros de una pareja coinciden en tamaño y en bandas;
  \item los \emph{\omterm{def:g8:sexual-reproduction:gametes}{gametos}} rompen la regla: el \omterm{def:g4:animal-reproduction:sexual}{óvulo} y el
        \omterm{def:g8:reproductive-systems:male}{espermatozoide} llevan \emph{23} --- un \omterm{def:g9:chromosomes:chromosomes}{cromosoma} de cada
        pareja, ninguna pareja completa.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[La aritmética de la fecundación]\label{ex:g9:chromosomes:arithmetic}
Las cuentas encajan como una demostración. \omterm{def:g8:sexual-reproduction:gametes}{Gametos} con 23 cada uno; la
\omterm{def:g5:flower-to-fruit:steps}{fecundación} (\cref{prop:g8:sexual-reproduction:core}) los suma: el
\omterm{prop:g8:sexual-reproduction:core}{cigoto} tiene $23 + 23 = 46$ --- las parejas restauradas, un miembro de
cada pareja de la madre y otro del padre. Todas las \omterm{def:g5:first-look-at-cells:cell}{células} del cuerpo
del hijo llevan entonces la aportación de los dos progenitores --- que
es exactamente lo que observaba desde fuera la
\cref{rem:g4:animal-reproduction:mix}, el potro y su lucero, hace ya
una generación de capítulos.
\end{example}

\begin{remark}[Por qué tienen que reducirse a la mitad los gametos]\label{rem:g9:chromosomes:whyhalve}
Supón que no lo hicieran: unos \omterm{def:g8:sexual-reproduction:gametes}{gametos} de 46 \omterm{def:g9:chromosomes:chromosomes}{cromosomas} darían hijos de
92 y nietos de 184 --- la baraja doblándose en cada generación. La
división especial de reducción a la mitad de los \omterm{def:g8:sexual-reproduction:gametes}{gametos} --- cada uno
recibe \emph{un miembro de cada pareja} --- es lo que mantiene fija en
46 la cuenta de la \omterm{def:g5:biodiversity:species}{especie}, generación tras generación. Y fíjate en la
propina: \emph{cuál} de los dos miembros de cada pareja le toca a un
\omterm{def:g8:sexual-reproduction:gametes}{gameto} se decide de manera independiente, pareja por pareja ---
$2^{23}$ barajas posibles de un solo progenitor, el barajado de la
\cref{prop:g8:sexual-reproduction:shuffle} contado por fin.
\end{remark}

\section{Los cromosomas llevan los determinantes}

\begin{proposition}[Pruebas de que la baraja importa]\label{prop:g9:chromosomes:evidence}
Que los \omterm{def:g9:chromosomes:chromosomes}{cromosomas} \emph{sean} los vehículos de los determinantes no es
una suposición, sino una conclusión con pruebas:
\begin{itemize}
  \item se comportan exactamente como exige la
        \cref{prop:g9:heredity-and-traits:conclusions}: presentes en
        todas las \omterm{def:g5:first-look-at-cells:cell}{células}, reducidos a la mitad en los \omterm{def:g8:sexual-reproduction:gametes}{gametos},
        restaurados en la \omterm{def:g5:flower-to-fruit:steps}{fecundación} y seleccionables miembro a
        miembro;
  \item los accidentes de \omterm{def:g9:chromosomes:chromosomes}{cariotipo} demuestran cuál es la carga: una
        baraja con un \omterm{def:g9:chromosomes:chromosomes}{cromosoma} de más o de menos cambia el desarrollo
        del cuerpo entero --- una copia de más de la pequeña pareja 21,
        por ejemplo, produce el conjunto reconocible de \omterm{def:g9:heredity-and-traits:traits}{rasgos} que se
        llama síndrome de Down. Un hilo mal contado, muchos \omterm{def:g9:heredity-and-traits:traits}{rasgos}
        cambiados: los hilos llevan instrucciones de \omterm{def:g9:heredity-and-traits:traits}{rasgos};
  \item y una pareja anuncia su carga en todos los \omterm{def:g9:chromosomes:chromosomes}{cariotipos}: los
        \emph{\omterm{def:g9:chromosomes:chromosomes}{cromosomas} sexuales}.
\end{itemize}
\end{proposition}
\admitted

\begin{definition}[X, Y y la moneda al aire]\label{def:g9:chromosomes:xy}
La pareja 23 viene en dos versiones: dos \omterm{def:g9:chromosomes:chromosomes}{cromosomas} \emph{X} iguales
--- el \omterm{def:g9:chromosomes:chromosomes}{cariotipo} de una niña; o una X y una \emph{Y} pequeña --- un
niño. La reducción a la mitad tira entonces la moneda: todos los \omterm{def:g4:animal-reproduction:sexual}{óvulos}
llevan una X; un \omterm{def:g8:reproductive-systems:male}{espermatozoide} lleva X o Y, mitad y mitad. La
\omterm{def:g5:flower-to-fruit:steps}{fecundación} por un \omterm{def:g8:reproductive-systems:male}{espermatozoide} X hace XX; por uno Y, XY --- una
posibilidad entre dos (el lenguaje del azar del
\cref{ch:g9:heredity-and-traits}, por fin con su mecanismo), decidida
por cuál de los nadadores llega, en el instante de la \omterm{def:g5:flower-to-fruit:steps}{fecundación}.
\end{definition}

\begin{example}[Preguntas viejas, respuestas cortas]\label{ex:g9:chromosomes:answers}
La baraja salda cuentas históricas. Que un hijo sea niño o niña no debe
nada a los deseos, las dietas ni las estaciones de ninguno de los dos
padres --- es la moneda al aire, tirada con el mecanismo de la
\cref{def:g9:chromosomes:xy} (y, para que conste en el registro de las
injusticias viejas: la diferencia que decide viaja en el
\emph{\omterm{def:g8:reproductive-systems:male}{espermatozoide}}). Las mitades iguales responden también al álbum
de familia: la madre y el padre aportan una media baraja completa cada
uno --- 23 y 23 --- y la aritmética de la \omterm{def:g9:heredity-and-traits:heredity}{herencia} honra por igual las
dos \omterm{def:g1:plants-around-us:tree}{ramas}.
\end{example}

\begin{method}[Leer un cariotipo]\label{met:g9:chromosomes:read}
Con una fotografía de \omterm{def:g9:chromosomes:chromosomes}{cariotipo} en la mano:
\begin{enumerate}
  \item cuenta: se esperan 46 en una \omterm{def:g5:first-look-at-cells:cell}{célula} del cuerpo y 23 en un
        \omterm{def:g8:sexual-reproduction:gametes}{gameto};
  \item empareja: los miembros se juntan por tamaño y por bandas, 22
        parejas corrientes más la 23;
  \item lee la 23: XX o XY;
  \item revisa las cuentas: cualquier pareja con uno o con tres
        miembros explica, por la
        \cref{prop:g9:chromosomes:evidence}, un desarrollo cambiado.
\end{enumerate}
\end{method}

\section{Problema: La mesa de citogenética}

\begin{problem}[{Problema de fin de semana --- cuatro cariotipos en
una sola mesa de luz}]\label{pb:g9:chromosomes:1}
La mesa de enseñanza de un laboratorio de citogenética de hospital:
\omterm{def:g9:chromosomes:chromosomes}{cariotipo} A --- 46 \omterm{def:g9:chromosomes:chromosomes}{cromosomas}, pareja 23 XY; B --- 46, pareja 23 XX;
C --- 23 \omterm{def:g9:chromosomes:chromosomes}{cromosomas}, una X entre ellos; D --- 47, con tres copias en la
pareja 21, pareja 23 XX.

\medskip\noindent\textbf{Parte I --- Las lecturas.}
\begin{enumerate}
  \item Lee A, B y C con el \cref{met:g9:chromosomes:read}: ¿de quién
        podrían ser las \omterm{def:g5:first-look-at-cells:cell}{células} de cada uno?
  \item La cuenta de C no es un error. ¿Qué clase de \omterm{def:g5:first-look-at-cells:cell}{célula} es, y qué
        historia de divisiones registran sus 23?
  \item Lee D: cuenta, emparejamiento y el desarrollo que anuncia.
  \item ¿El \omterm{def:g8:sexual-reproduction:gametes}{gameto} de qué progenitor decidió la pareja 23 de A, y
        cuáles eran las probabilidades en aquella \omterm{def:g5:flower-to-fruit:steps}{fecundación}?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Las preguntas de
aritmética.}
\begin{enumerate}[resume]
  \item Una alumna en prácticas pregunta por qué todas las \omterm{def:g5:first-look-at-cells:cell}{células} del
        cuerpo de una persona coinciden con el \omterm{def:g9:chromosomes:chromosomes}{cariotipo} A. Responde
        con las divisiones de copia de la
        \cref{prop:g5:first-look-at-cells:growth}.
  \item Otra pregunta cómo hicieron dos progenitores de \omterm{def:g5:first-look-at-cells:cell}{células} de 46
        los 23 de C. Nombra la división especial y su regla.
  \item Combina C con un \omterm{def:g4:animal-reproduction:sexual}{óvulo} portador de X y con su vecino del propio
        laboratorio portador de Y --- espera: C \emph{está} libre de Y.
        Enuncia qué garantiza la X de C sobre cualquier hijo que ayude
        a hacer.
  \item Calcula las barajas posibles que puede repartir un progenitor:
        enuncia la independencia pareja a pareja y la conclusión
        $2^{23}$ --- y añade después el emparejamiento de dos repartos
        así que hace la \omterm{def:g5:flower-to-fruit:steps}{fecundación}.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La sala de consejo
genético.}
\begin{enumerate}[resume]
  \item Los padres de un niño con el \omterm{def:g9:chromosomes:chromosomes}{cariotipo} D preguntan ``¿qué
        hicimos mal?''. Da la respuesta honrada del mecanismo sobre los
        accidentes de la reducción a la mitad --- y qué cambian y qué
        no cambian los 47 hilos del niño sobre el cariño que hay en esa
        sala. (Responde como quien aconseja, con la \omterm{def:g1:living-or-not:living}{biología} derecha.)
  \item Una pareja con tres hijas pregunta por las probabilidades de un
        hijo la próxima vez. Responde con la moneda al aire y corrige
        el instinto de las rachas de suerte.
  \item Un visitante afirma que una dieta puede elegir el sexo de un
        bebé. Audita la afirmación con la
        \cref{def:g9:chromosomes:xy}: ¿dónde se toma la decisión y
        hasta dónde podría llegar una dieta?
  \item Cierra la mesa con una frase: qué demuestran juntas las cuatro
        fotografías sobre en qué viaja la \omterm{def:g9:heredity-and-traits:heredity}{herencia}.
\end{enumerate}
\end{problem}
