\chapter{المحلِّلون والتربة}\label{ch:g6:decomposers-and-soil}

في كل خريف يُسقط الحرش فرشًا كثيفًا من \omterm{def:g1:plants-around-us:parts}{الأوراق} --- أطنانًا منها في
الهكتار. ومع ذلك امشِ في الحرش نفسه كل ربيع فلن تخوض في بحر صاعد
من فرش \omterm{def:g1:plants-around-us:parts}{الأوراق}: بل الفرش رقيق من جديد، سنة بعد سنة. فثمة من يأكل
موتى الغابة. وهذا الفصل ينزل إلى \omterm{def:g6:decomposers-and-soil:soil}{التربة} ليلقى الآكلين --- وليغلق
أخيرًا دائرة المادة.

\section{التربة حية}

\begin{definition}[التربة]\label{def:g6:decomposers-and-soil:soil}
\emph{التربة}\index{تربة} \omterm{def:g1:the-five-senses:sense}{جلد} الأرض الحي: خليط من حبيبات
\emph{معدنية} (صخر مطحون، من الرمل إلى الطين)، وبقايا
\emph{عضوية} في كل أطوار التعفن، وماء، وهواء --- وجمهرة هائلة من
\omterm{def:g6:exploring-our-environment:components}{الكائنات الحية}. وطبقتها العليا الداكنة، وهي أغناها بالمادة
المتعفنة، هي طبقة \emph{الدُّبال}\index{دبال} التي يثمّنها
البستانيون.
\end{definition}

\begin{example}[حفنة مُحصاة]\label{ex:g6:decomposers-and-soil:handful}
حفنة واحدة من \omterm{def:g6:decomposers-and-soil:soil}{تربة} حرش، مفروزة على صينية بيضاء: حبيبات معدنية؛
وخيوط \omterm{def:g1:plants-around-us:parts}{جذور}؛ وأنصاف \omterm{def:g1:plants-around-us:parts}{أوراق}، وهياكل \omterm{def:g1:plants-around-us:parts}{أوراق}، وفتات أسود --- \omterm{def:g6:plants-are-producers:matter}{مادة عضوية}
في طريقها إلى أسفل؛ والإحصاء --- دودة أرض، وحشرتا قمل خشبي، وأم
أربعة وأربعين، وذنبيات قافزة بيضاء صغيرة، ولبّادات \omterm{def:g6:decomposers-and-soil:decomposers}{فطر} في رقّة
الخيط. والصينية لا تُظهر إلا المرئي: فغرام واحد من \omterm{def:g6:decomposers-and-soil:soil}{التربة} يحوي من
\omterm{def:g6:exploring-our-environment:components}{الكائنات الحية} الأصغر من أن تُرى أكثر مما في مدينة كبيرة من الناس.
\end{example}

\begin{definition}[المحلِّلون]\label{def:g6:decomposers-and-soil:decomposers}
\emph{المحلِّلون}\index{محلل} هم \omterm{def:g6:exploring-our-environment:components}{الكائنات الحية} التي تتغذى على
\emph{\omterm{def:g6:plants-are-producers:matter}{المادة العضوية} الميتة} --- \omterm{def:g1:plants-around-us:parts}{الأوراق} الساقطة، والخشب الميت،
والفضلات، والجثث:
\begin{itemize}
  \item الطاقم المرئي: ديدان الأرض، وحشرات القمل الخشبي، وأمّات
        أربعة وأربعين، والذنبيات القافزة، \omterm{ex:g2:animals-grow-up:butterfly}{ويرقات} الذباب، وخنافس
        الروث؛
  \item و\emph{الفطور}\index{فطر}: العفن والفطر، وجسمها المتغذي
        لبّادة من خيوط بيضاء منتشرة في \omterm{def:g6:decomposers-and-soil:soil}{التربة} والخشب الميت ---
        وفطر \cref{exo:g1:living-or-not:10} ليس إلا ثمرتها؛
  \item وحياة \omterm{def:g6:decomposers-and-soil:soil}{التربة} الأصغر من أن تُرى، وهي التي تنهي كل عمل ---
        المفكّكون الأخيرون.
\end{itemize}
وهم عمّال الظل في \cref{def:g5:food-webs:roles}، وقد وُعدوا بفصل
خاص بهم: وهذا هو.
\end{definition}

\section{التحلل: من العضوي إلى المعدني}

\begin{proposition}[ماذا يفعل التحلل]\label{prop:g6:decomposers-and-soil:decomposition}
\emph{التحلل}\index{تحلل} هو تغذّي المحلِّلين، ونتيجته عكس تجارة
\omterm{def:g5:food-webs:roles}{المنتِجين}: فالمادة \emph{العضوية} الميتة تُفكَّك، طورًا بعد طور
وآكلًا بعد آكل، حتى تُردّ مادتها إلى العالم \emph{المعدني} ---
دُبالًا هشًّا، ثم أملاحًا معدنية مذابة في ماء \omterm{def:g6:decomposers-and-soil:soil}{التربة}، وغازًا يعود
إلى الهواء. ففرش الخريف يصير بحلول الربيع بالضبط ما تشربه \omterm{def:g1:plants-around-us:parts}{الجذور}
(\cref{prop:g4:what-plants-need:needs}).
\end{proposition}
\admitted

\begin{example}[سنة الورقة الأخيرة]\label{ex:g6:decomposers-and-soil:leaf}
تتبّع \omterm{def:g1:plants-around-us:parts}{ورقة} بلوط واحدة نازلة. تشرين الثاني: ساقطة، كاملة. والشتاء:
مليّنة، منقوبة بخيوط الفطور، منهوشة حتى صارت هيكلًا مخرّمًا
بالذنبيات القافزة وحشرات القمل الخشبي. والربيع: مجرورة تحت الأرض
بدودة أرض، ومأكولة، ومقذوفة فتاتًا معالَجًا. والصيف: آخر شظاياها
دُبال أسود، ومعادنها تركب ماء \omterm{def:g6:decomposers-and-soil:soil}{التربة} --- إلى \omterm{def:g1:plants-around-us:parts}{جذر}، ولعله \omterm{def:g1:plants-around-us:parts}{جذر}
البلوطة نفسها. أُكلت أربع مرات، فصارت \omterm{def:g1:plants-around-us:parts}{الورقة} غذاءً لشجرتها هي.
\end{example}

\begin{remark}[الدائرة تُغلق]\label{rem:g6:decomposers-and-soil:circle}
بحضور المحلِّلين تنغلق حكاية المادة في هذه السنة عجلةً: \omterm{def:g5:food-webs:roles}{فالمنتِجون}
يبنون العضوي من المعدني
(\cref{ch:g6:plants-are-producers})؛ \omterm{def:g5:food-webs:roles}{والمستهلِكون} يمرّرونه
(\cref{ch:g6:matter-from-food})؛ \omterm{def:g6:decomposers-and-soil:decomposers}{والمحلِّلون} يردّونه معدنيًّا، جاهزًا
\omterm{def:g5:food-webs:roles}{للمنتِجين} من جديد. فالمادة لا تُستهلك --- بل تدور. وذرّات ورق هذه
الصفحة كانت ورقةً ودُبالًا وجذرًا ثم ورقةً من جديد مرات لا تُحصى.
\end{remark}

\section{العمل مع المحلِّلين}

\begin{example}[السماد مشروحًا أخيرًا]\label{ex:g6:decomposers-and-soil:compost}
صندوق السماد في \cref{ex:g2:caring-for-nature:compost} يُقرأ الآن
جهازًا: مشغل محلِّلين مزوَّد بالقشور. فالديدان وحشرات القمل الخشبي
والأطقم غير المرئية تفكّك نفايات المطبخ العضوية إلى دُبال داكن؛
وينثره البستاني، فتغذّي معادنه الأحواض
(\cref{rem:g4:what-plants-need:soil}). بل إن الكومة تعمل دافئة ---
فاحتراق المحلِّلين أنفسهم
(\cref{prop:g6:matter-from-food:losses}) يدفّئ مشغلهم.
\end{example}

\begin{proposition}[ما يسرّع العمل وما يبطّئه]\label{prop:g6:decomposers-and-soil:speed}
\omterm{def:g6:decomposers-and-soil:decomposers}{المحلِّلون} كائنات حية لها شروطها
(\cref{prop:g6:exploring-our-environment:decide}): فهم يعملون أسرع
ما يعملون في \emph{الدفء} و\emph{الندى} و\emph{التهوية}، ويكادون
يتوقفون في البرد أو الجفاف أو الغرق بلا هواء. ومن هنا بطء فرش
الشتاء وسرعة تنظيف الربيع؛ ومن هنا أيضًا وصفة سماد البستاني ---
رطبة، مقلَّبة للهواء، لا كتلة مبتلة مختومة أبدًا.
\end{proposition}
\admitted

\begin{method}[قراءة صحة التربة]\label{met:g6:decomposers-and-soil:read}
لتحكم على \omterm{def:g6:decomposers-and-soil:soil}{تربة} حكم عالم الطبيعة:
\begin{enumerate}
  \item احفر بعمق شبر وانظر: أدُبال داكن هشّ، أم غبار شاحب بلا
        حياة؟
  \item وأحصِ حفنة على صينية بيضاء --- ديدان، وحشرات قمل خشبي،
        وذنبيات قافزة: فالطاقم الغني يعني أن الدورة تدور؛
  \item وراجع السطح: أتختفي المادة الميتة عبر المواسم أم تتكوّم
        بلا تعفن؟
  \item وعامل الطاقم معاملة الماشية: أطعمه (بقايا مقلوبة، وسمادًا،
        وغطاء نباتيًّا)، وأبقِه نديًّا، وجنّبه السموم.
\end{enumerate}
\end{method}

\begin{example}[دودة الأرض، بستانية البستانيين]\label{ex:g6:decomposers-and-soil:worm}
تؤدي دودة الأرض عملين: فهي محلِّلة --- تجرّ \omterm{def:g1:plants-around-us:parts}{الأوراق} إلى أسفل وتأكل
المادة المتعفنة --- وهي \emph{حرّاثة}، فأنفاقها تهوّي \omterm{def:g6:decomposers-and-soil:soil}{التربة}
وقذفاتها تعجن المعدني والعضوي معًا. ومرج واحد قد يحمل من وزن
الديدان تحت العشب أكثر من وزن البقر فوقه؛ وقد سمّى علماء الطبيعة
القدماء الدودة صانعةَ الأرض الزراعية الحقيقية، وكانوا على صواب.
\end{example}

\section{مسألة: تجارب أكياس الأوراق}

\begin{problem}[{مسألة نهاية الأسبوع --- أكياس شبكية، وأوراق
مدفونة، وشتاء من علم التربة}]\label{pb:g6:decomposers-and-soil:1}
يدفن الصف أكياس \omterm{def:g1:plants-around-us:parts}{أوراق} في ضفة سياج المدرسة في تشرين الثاني: أوزان
متساوية من \omterm{def:g1:plants-around-us:parts}{أوراق} بلوط طازجة في أكياس شبكية، تُخرج وتُوزن عبر
السنة. وثلاثة أصناف من الأكياس: شبكة خشنة (تُدخل الديدان وحشرات
القمل الخشبي وكل شيء)، وشبكة ناعمة (تستبعد الطاقم المرئي ولا تُدخل
إلا غير المرئي)، وبلاستيك مختوم (يستبعد كل شيء والهواء معه).

\medskip\noindent\textbf{القسم الأول --- التهيئة.}
\begin{enumerate}
  \item لماذا يجب أن تحمل الأكياس كلها الوزن \emph{نفسه} من
        \omterm{def:g1:plants-around-us:parts}{الأوراق} \emph{نفسها}، مدفونة في اليوم \emph{نفسه} في
        الضفة \emph{نفسها}؟ سمِّ الطريقة.
  \item وأي سؤال يطرحه كيس الشبكة الناعمة مما لا يستطيع الخشن أن
        يطرحه وحده؟
  \item يريد فريق كيسًا رابعًا: شبكة خشنة، لكن مدفونًا في حصى ساحة
        المدرسة الجاف. فأي سؤال يطرحه وفق
        \cref{prop:g6:decomposers-and-soil:speed}؟
  \item تنبأ قبل الحفر: رتّب أكياس ضفة السياج الثلاثة بحسب مقدار
        وزن \omterm{def:g1:plants-around-us:parts}{الأوراق} الذي ستفقده بحلول حزيران.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- النتائج.}
\begin{enumerate}[resume]
  \item أوزان حزيران: الشبكة الخشنة --- أكثر الوزن ذهب، \omterm{def:g1:plants-around-us:parts}{والأوراق}
        صارت هياكل وفتاتًا؛ والشبكة الناعمة --- خفّت بوضوح،
        \omterm{def:g1:plants-around-us:parts}{والأوراق} ليّنت واسودّت لكن أبطأ؛ والبلاستيك المختوم ---
        الوزن لم يكد يتغير، والمحتوى حمأة حامضة. فسّر كل كيس.
  \item في الكيس الخشن يجد الصف حشرات قمل خشبي، ودودة، وذنبيات
        قافزة، وخيوط \omterm{def:g6:decomposers-and-soil:decomposers}{فطر} بيضاء. أعطِ كل واحد دوره في تفكيك \omterm{def:g1:plants-around-us:parts}{الورقة}
        (و\cref{ex:g6:decomposers-and-soil:leaf} هي النموذج).
  \item فقد الكيس الناعم وزنًا حقيقيًّا مع أن كل \omterm{def:g1:animals-around-us:animal}{حيوان} مرئي كان
        مستبعَدًا. فعلام يبرهن ذلك في شأن حياة \omterm{def:g6:decomposers-and-soil:soil}{التربة} غير المرئية؟
  \item كيس ساحة الحصى (كيس السؤال الثالث) فقد أقل بكثير من
        توأمه في ضفة السياج. فأي \omterm{def:g6:exploring-our-environment:conditions}{شروط} عمل يمنعها الحصى؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- الدرس الأكبر.}
\begin{enumerate}[resume]
  \item إلى أين \emph{ذهب} الوزن المفقود من الكيس الخشن؟ أجب وفق
        \cref{prop:g6:decomposers-and-soil:decomposition} ---
        بوجهتين على الأقل.
  \item قرّاص الربيع في السياج يخضرّ أشدّ الاخضرار بالضبط على
        امتداد الضفة حيث دفن الصف أكياسه. أهي مصادفة؟ حاجج من
        دورة المادة.
  \item استعمل الكيس المختوم لتشرح لماذا تبقى النفايات العضوية
        المطمورة، ملفوفة بإحكام في بلاستيك، عقودًا بينما تتسمّد
        النفايات نفسها في شهور.
  \item اختم بجملتين: علام برهن الشتاء في شأن من ينظّف فرش
        الغابة، وماذا يغذّي ذلك التنظيف؟
\end{enumerate}
\end{problem}
