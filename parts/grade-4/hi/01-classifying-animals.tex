\chapter{जंतुओं का वर्गीकरण}\label{ch:g4:classifying-animals}

प्राकृतिक इतिहास के किसी संग्रहालय में लाखों \omterm{def:g1:animals-around-us:animal}{जंतु} रखे रहते हैं ---
भृंगों की दराज़ें, \omterm{def:g3:bones-and-muscles:skeleton}{कंकालों} के कक्ष, \omterm{prop:g3:sorting-living-things:groups}{मछलियों} के मरतबान। फिर भी वहाँ का
अध्यक्ष उनमें से किसी एक तक मिनटों में पहुँच जाता है। राज़ याददाश्त
नहीं, \emph{क्रम} है: हर \omterm{def:g1:animals-around-us:animal}{जंतु} की जगह समूहों के भीतर समूहों की उस बड़ी
व्यवस्था में तय है जो \cref{ch:g3:sorting-living-things} के छाँटने के
नियमों से बनी है --- और इस साल हम सीखेंगे कि यह व्यवस्था काम कैसे करती
है।

\section{समूहों के भीतर समूह}

\begin{definition}[वर्गीकरण]\label{def:g4:classifying-animals:classification}
सजीवों को \emph{वर्गीकृत करना}\index{वर्गीकृत करना} यानी उन्हें शरीर की
साझा ख़ासियतों से समूहों में छाँटना, और फिर उन समूहों को बड़े समूहों
में --- यानी \emph{समूहों के भीतर बैठे
समूह}\index{अंतर्निहित समूह}, डिब्बों के भीतर डिब्बों की तरह। यह पूरी
व्यवस्था ही \emph{वर्गीकरण}\index{वर्गीकरण} है।
\end{definition}

\begin{example}[बिल्ली के पते]\label{ex:g4:classifying-animals:cat}
एक बिल्ली, और कई नत्थी पते: बिल्ली एक \emph{\omterm{prop:g3:sorting-living-things:groups}{स्तनधारी}} है (\omterm{def:g1:animals-around-us:covering}{रोम}, बच्चों
के लिए \omterm{def:g2:animals-grow-up:born}{दूध}); हर \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} एक \emph{\omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी}} है (\omterm{def:g4:classifying-animals:vertebrate}{रीढ़} के चारों ओर बना
भीतरी \omterm{def:g3:bones-and-muscles:skeleton}{कंकाल}); और हर \omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी} एक \emph{\omterm{def:g1:animals-around-us:animal}{जंतु}} है। बिल्ली \omterm{prop:g3:sorting-living-things:groups}{स्तनधारियों} के
भीतर, \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} \omterm{def:g4:classifying-animals:vertebrate}{कशेरुकियों} के भीतर, \omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी} \omterm{def:g1:animals-around-us:animal}{जंतुओं} के भीतर: हर डिब्बा
किसी बड़े डिब्बे के भीतर बैठा है।
\end{example}

\begin{definition}[कशेरुकी और अकशेरुकी]\label{def:g4:classifying-animals:vertebrate}
\emph{कशेरुकी}\index{कशेरुकी} वह \omterm{def:g1:animals-around-us:animal}{जंतु} है जिसका भीतरी \omterm{def:g3:bones-and-muscles:skeleton}{कंकाल}
\emph{रीढ़}\index{रीढ़} के चारों ओर बना होता है --- वही रीढ़ जिसे तुमने
\cref{ex:g3:bones-and-muscles:feel} में महसूस किया था। बिना रीढ़ वाला
\omterm{def:g1:animals-around-us:animal}{जंतु} \emph{अकशेरुकी}\index{अकशेरुकी} है: \omterm{prop:g3:sorting-living-things:groups}{कीट}, मकड़ियाँ, घोंघे, केंचुए,
जेलीफ़िश --- यानी सारे \omterm{def:g1:animals-around-us:animal}{जंतुओं} का बहुत बड़ा हिस्सा।
\end{definition}

\section{कशेरुकियों के पाँच वर्ग}

\begin{proposition}[कशेरुकियों के पाँच वर्ग]\label{prop:g4:classifying-animals:classes}
\omterm{def:g4:classifying-animals:vertebrate}{कशेरुकियों} को पाँच बड़े समूहों में वर्गीकृत किया जाता है, जिन्हें
\emph{वर्ग}\index{वर्ग (कशेरुकियों का)} कहते हैं:
\begin{enumerate}
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{मछलियाँ}}: \omterm{def:g1:animals-around-us:covering}{शल्क}, मीन-पंख, पानी में साँस --- ट्राउट,
        शार्क, सारडीन;
  \item \emph{उभयचर}\index{उभयचर}: नम खुली \omterm{def:g1:the-five-senses:sense}{त्वचा}, पानी में \omterm{def:g2:animals-grow-up:born}{अंडे},
        \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} वाला बचपन --- मेंढक, भेक, न्यूट;
  \item \emph{सरीसृप}\index{सरीसृप}: सूखी शल्कदार \omterm{def:g1:the-five-senses:sense}{त्वचा}, ज़्यादातर
        ज़मीन पर खोलदार \omterm{def:g2:animals-grow-up:born}{अंडे} देते हैं --- छिपकली, साँप, कछुआ,
        मगरमच्छ;
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{पक्षी}}: \omterm{def:g1:animals-around-us:covering}{पंख}, चोंच, खोलदार \omterm{def:g2:animals-grow-up:born}{अंडे} --- कलचिड़ी,
        शुतुरमुर्ग़, पेंग्विन;
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{स्तनधारी}}: \omterm{def:g1:animals-around-us:covering}{रोम} या बाल, और बच्चे \omterm{def:g2:animals-grow-up:born}{दूध} पर पलते हैं ---
        बिल्ली, ह्वेल, चमगादड़, मनुष्य।
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[चिड़ियाघर की खेप छाँटना]\label{ex:g4:classifying-animals:zoo}
छह पिंजरे आते हैं: एक अजगर (सूखी शल्कदार \omterm{def:g1:the-five-senses:sense}{त्वचा}, खोलदार \omterm{def:g2:animals-grow-up:born}{अंडे} ---
\omterm{prop:g4:classifying-animals:classes}{सरीसृप}), एक भेक (नम खुली \omterm{def:g1:the-five-senses:sense}{त्वचा} --- \omterm{prop:g3:sorting-living-things:groups}{उभयचर}), एक पेंग्विन (\omterm{def:g1:animals-around-us:covering}{पंख} ---
\omterm{prop:g3:sorting-living-things:groups}{पक्षी}), एक शार्क (मीन-पंख और क्लोम --- मछली), एक डॉल्फ़िन (अपने बच्चे
के लिए \omterm{def:g2:animals-grow-up:born}{दूध} --- \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी}), और विशाल घोंघों का एक पिंजरा (\omterm{def:g4:classifying-animals:vertebrate}{रीढ़} बिलकुल
नहीं --- \omterm{def:g4:classifying-animals:vertebrate}{अकशेरुकी}, किसी भी \omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी} वर्ग में नहीं)।
\end{example}

\begin{remark}[सरीसृप या उभयचर?]\label{rem:g4:classifying-animals:confusion}
पहली नज़र में छिपकली और न्यूट एक जैसे लगते हैं --- चार \omterm{def:g1:my-body:parts}{टाँगें}, लंबी
पूँछ। फ़ैसला \omterm{def:g1:the-five-senses:sense}{त्वचा} करती है: छिपकली की सूखी और शल्कदार है, न्यूट की नम
और खुली; छिपकली के \omterm{def:g2:animals-grow-up:born}{अंडों} पर \omterm{def:g1:animals-around-us:covering}{खोल} होता है और वे ज़मीन पर दिए जाते हैं,
न्यूट के खुले \omterm{def:g2:animals-grow-up:born}{अंडे} पानी में। एक जैसी छाया, अलग-अलग वर्ग।
\end{remark}

\section{वर्गीकरण का इस्तेमाल}

\begin{method}[किसी कशेरुकी को उसके वर्ग में रखना]\label{met:g4:classifying-animals:place}
सवाल क्रम से पूछो, और पहले ``हाँ'' पर रुक जाओ:
\begin{enumerate}
  \item \omterm{def:g1:animals-around-us:covering}{पंख}? --- \omterm{prop:g3:sorting-living-things:groups}{पक्षी}।
  \item \omterm{def:g1:animals-around-us:covering}{रोम} या बाल (या बच्चों के लिए \omterm{def:g2:animals-grow-up:born}{दूध})? --- \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी}।
  \item सूखी शल्कदार \omterm{def:g1:the-five-senses:sense}{त्वचा}? --- \omterm{prop:g4:classifying-animals:classes}{सरीसृप}।
  \item नम खुली \omterm{def:g1:the-five-senses:sense}{त्वचा} और \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} वाला बचपन? --- \omterm{prop:g3:sorting-living-things:groups}{उभयचर}।
  \item मीन-पंख और क्लोम, पानी में जीवन? --- मछली।
\end{enumerate}
\omterm{def:g4:classifying-animals:vertebrate}{रीढ़} बिलकुल नहीं? तो वह \omterm{def:g4:classifying-animals:vertebrate}{अकशेरुकी} था, और ये सवाल उस पर लागू ही नहीं
होते।
\end{method}

\begin{proposition}[वर्ग जो मुफ़्त में बता देता है]\label{prop:g4:classifying-animals:free}
किसी \omterm{def:g1:animals-around-us:animal}{जंतु} का वर्ग जान लेना देखे बिना ही तथ्यों का पूरा गट्ठर जान लेना
है: सिर्फ़ इतना बताया जाए कि ``यह \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} है'', और तुम पहले से \omterm{def:g1:animals-around-us:covering}{रोम},
बच्चों के लिए \omterm{def:g2:animals-grow-up:born}{दूध}, और --- लगभग सभी \omterm{prop:g3:sorting-living-things:groups}{स्तनधारियों} की तरह --- \omterm{def:g2:animals-grow-up:born}{जीवित जन्म}
लेते बच्चों की उम्मीद करने लगते हो। यही बचत \omterm{def:g4:classifying-animals:classification}{वर्गीकरण} का पूरा मक़सद है:
समूह एक बार सीख लो, और उसका हर सदस्य आधा जाना-पहचाना आता है।
\end{proposition}
\admitted

\begin{example}[नई प्रजाति आती है]\label{ex:g4:classifying-animals:new}
प्रकृतिविद आज भी ऐसे \omterm{def:g1:animals-around-us:animal}{जंतु} खोजते हैं जो पहले कभी नहीं देखे गए। जब \omterm{def:g1:animals-around-us:covering}{रोम}
और \omterm{def:g2:animals-grow-up:born}{दूध} वाला कोई नया चमगादड़-जैसा \omterm{def:g1:animals-around-us:animal}{जंतु} मिलता है, वह सीधे \omterm{prop:g3:sorting-living-things:groups}{स्तनधारियों} में
चला जाता है --- और जीवविज्ञानी उसी पल \omterm{def:g4:classifying-animals:vertebrate}{रीढ़}, चार कक्षों वाले हृदय और
गरम रक्त की उम्मीद करने लगते हैं, इनमें से किसी की जाँच से पहले ही।
\omterm{def:g4:classifying-animals:classification}{वर्गीकरण} भविष्यवाणियाँ चुकाता है।
\end{example}

\begin{remark}[कुल-वृक्ष की ओर]\label{rem:g4:classifying-animals:tree}
\omterm{def:g1:animals-around-us:covering}{रोम}, \omterm{def:g2:animals-grow-up:born}{दूध} और \omterm{def:g4:classifying-animals:vertebrate}{रीढ़} इतनी ईमानदारी से साथ-साथ क्यों चलते हैं? पूरा उत्तर
--- कि किसी समूह के सदस्य अपनी साझा ख़ासियतें साझे पूर्वजों से पाते हैं
--- नातेदारी और जैव विकास की कहानी का हिस्सा है, जिसे आगे के सालों में
उठाया जाएगा। फ़िलहाल ये नत्थी डिब्बे बस काम करते हैं, और कमाल का काम
करते हैं।
\end{remark}

ह्वेल के लगभग कोई बाल नहीं होते और वह मछली की तरह रहती है।

\cref{ex:g3:sorting-living-things:surprises} से समझाओ कि फिर भी वह
\omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} क्यों है --- और ``समुद्र में रहती है'' कभी \omterm{def:g4:classifying-animals:classification}{वर्गीकरण} की
ख़ासियत क्यों नहीं रही।
\end{exercise}
