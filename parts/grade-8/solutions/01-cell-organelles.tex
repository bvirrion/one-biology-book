\section*{Chapter \ref{ch:g8:organelles} --- Cell Organelles}

\begin{solution}{exo:g8:organelles:1}
Specialised internal structure with a job. Compartments concentrate
reactions, protect the rest of the \omterm{def:g7:microscope:cell}{cell}, and increase membrane surface.
\end{solution}

\begin{solution}{exo:g8:organelles:2}
\omterm{def:g8:organelles:nucleus}{Nucleus}: stores/protects DNA and \omterm{def:g6:method:control}{controls} \omterm{def:g7:microscope:cell}{cell} activities via \omterm{def:g7:heredity:gene}{gene}
information. \omterm{def:g8:organelles:ribosome}{Ribosomes}: \omterm{def:g6:nutrition:groups}{protein} synthesis.
\end{solution}

\begin{solution}{exo:g8:organelles:3}
Rough ER has \omterm{def:g8:organelles:ribosome}{ribosomes} and handles many \omterm{def:g6:nutrition:groups}{proteins} for export/membranes;
smooth ER lacks \omterm{def:g8:organelles:ribosome}{ribosomes} and is linked to \omterm{def:g6:nutrition:groups}{lipids}/detox.
\end{solution}

\begin{solution}{exo:g8:organelles:4}
\omterm{def:g8:organelles:energy}{Mitochondria}: aerobic respiration/ATP. \omterm{def:g8:organelles:energy}{Chloroplasts}: \omterm{def:g7:photo-resp:photo}{photosynthesis}.
\end{solution}

\begin{solution}{exo:g8:organelles:5}
\omterm{def:g8:organelles:wall}{Cell wall}, \omterm{def:g7:microscope:parts}{chloroplasts}, large central \omterm{def:g8:organelles:lyso}{vacuole} (typical).
\end{solution}
