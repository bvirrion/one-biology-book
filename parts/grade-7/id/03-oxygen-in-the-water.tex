\chapter{Oksigen di dalam Air}\label{ch:g7:oxygen-in-the-water}

Sungai pegunungan dan kolam musim panas keduanya berupa ``air'' ---
namun \omterm{prop:g3:sorting-living-things:groups}{ikan} trout yang meluncur di yang satu akan megap-megap di yang
lain, dan \omterm{prop:g3:sorting-living-things:groups}{ikan} mas yang tumbuh subur di kolamnya akan mendapati
sungainya kosong dari makanannya. Perbedaan antara kedua airnya
sebagian besar berupa satu bilangan yang tak terlihat: berapa banyak
\omterm{def:g7:what-is-respiration:respiration}{oksigen} yang \omterm{def:g7:oxygen-in-the-water:dissolved}{terlarut} di dalamnya. Bab ini belajar membaca, mengukur
dan meramalkan bilangan itu --- dan bersamanya, peta kehidupan air.

\section{Oksigen terlarut, sebuah persediaan yang langka}

\begin{definition}[Oksigen terlarut]\label{def:g7:oxygen-in-the-water:dissolved}
Air memuat \omterm{def:g7:what-is-respiration:respiration}{oksigen} sebagaimana teh manis memuat gula --- tanpa
terlihat, dalam keadaan \emph{terlarut}\index{oksigen terlarut}.
Jumlahnya mungil: seliter air yang teraerasi dengan baik membawa
kira-kira \qty{10}{mg} \omterm{def:g7:what-is-respiration:respiration}{oksigen}, sekitar tiga puluh kali lebih sedikit
daripada seliter udara. Setiap \omterm{def:g7:how-animals-breathe:four}{insang} dan setiap pelaku \omterm{def:g7:what-is-respiration:respiration}{respirasi}
bawah air pada \cref{ch:g7:how-animals-breathe} menimba satu
persediaan yang tipis ini
(\cref{prop:g7:what-is-respiration:habitats}).
\end{definition}

\begin{proposition}[Apa yang mengisi persediaannya]\label{prop:g7:oxygen-in-the-water:fills}
Dua pemasok mengisi ulang \omterm{def:g7:what-is-respiration:respiration}{oksigen} sebuah perairan:
\begin{enumerate}
  \item \emph{udara di atasnya}: \omterm{def:g7:what-is-respiration:respiration}{oksigen} melarut masuk di permukaannya
        --- lebih cepat di tempat airnya teraduk, terpercik dan
        berbuih (jeram, air terjun, ombak, hujan);
  \item \emph{para \omterm{def:g5:food-webs:roles}{produsen} di bawahnya}: pada siang hari, \omterm{def:g1:plants-around-us:plant}{tumbuhan}
        air dan \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal hijau melepaskan \omterm{def:g7:what-is-respiration:respiration}{oksigen} ke dalam airnya
        sambil membuat
        (\cref{rem:g6:plants-are-producers:gas}).
\end{enumerate}
\end{proposition}
\admitted

\begin{proposition}[Apa yang menguras dan berapa banyak yang dapat dimuatnya]\label{prop:g7:oxygen-in-the-water:drains}
Persediaannya dikuras oleh \emph{setiap pelaku \omterm{def:g7:what-is-respiration:respiration}{respirasi}} --- \omterm{def:g1:animals-around-us:animal}{hewan},
\omterm{def:g1:plants-around-us:plant}{tumbuhan} pada malam hari, dan terutama para \omterm{def:g6:decomposers-and-soil:decomposers}{pengurai} ketika ada banyak
materi mati untuk dimakan
(toples kedelapan pada \cref{pb:g7:what-is-respiration:1}). Dan
langit-langitnya bergerak: air yang \emph{dingin} dapat memuat \omterm{def:g7:oxygen-in-the-water:dissolved}{oksigen
terlarut} jauh lebih banyak daripada yang hangat --- sebuah sungai pada
\qty{5}{\degreeCelsius} membawa satu setengah kali lebih banyak
daripada sebuah kolam pada \qty{25}{\degreeCelsius}. Kehangatan adalah
perasan ganda: \omterm{def:g7:what-is-respiration:respiration}{oksigen} yang dimuat lebih sedikit, sementara setiap
bunyi mesin (\cref{ex:g7:what-is-respiration:demand}) berjalan lebih
cepat.
\end{proposition}
\admitted

\begin{example}[Sungai melawan kolam]\label{ex:g7:oxygen-in-the-water:streampond}
Sekarang kedua perairannya terbaca seperti pembukuan. Sungai
pegunungannya: dingin (langit-langit yang tinggi), teraduk memutih di
setiap bongkah batu (pengisian ulang yang cepat) --- \omterm{def:g7:what-is-respiration:respiration}{oksigennya} dekat
langit-langitnya sepanjang tahun. Kolam musim panasnya: hangat
(langit-langit yang rendah), tenang (pengisian ulang yang lambat),
kaya lumpur dan materi mati (\omterm{def:g6:decomposers-and-soil:decomposers}{pengurai} yang lapar) --- \omterm{def:g7:what-is-respiration:respiration}{oksigennya}
menggesek lantainya, paling buruk pada jam-jam kecil sebelum fajar,
ketika \omterm{def:g1:plants-around-us:plant}{tumbuhannya} sudah berespirasi semalaman tanpa pembuatan siang
hari untuk mengimbanginya.
\end{example}

\begin{example}[Membaca lengkung kolamnya]\label{ex:g7:oxygen-in-the-water:curve}
Gelombang harian kolamnya adalah
\cref{rem:g6:plants-are-producers:gas} yang digambar sebagai data:
terbitnya matahari memulai pembuatan oleh \omterm{def:g5:food-webs:roles}{produsennya} dan lengkungnya
menanjak; sore hari memuncakkannya; malam meninggalkan setiap pelaku
\omterm{def:g7:what-is-respiration:respiration}{respirasi} menimba persediaannya tanpa pemasok kecuali permukaannya
yang tenang --- lengkungnya meluncur ke minimum fajarnya. Kematian
\omterm{prop:g3:sorting-living-things:groups}{ikan} di kolam yang terlalu rimbun terjadi pada waktu fajar, dan
sekarang kamu tahu sebabnya.
\end{example}

\section{Peta para penghuninya}

\begin{proposition}[Oksigen menggambar petanya]\label{prop:g7:oxygen-in-the-water:map}
Karena \omterm{def:g5:biodiversity:species}{spesies} berbeda dalam kebutuhan \omterm{def:g7:what-is-respiration:respiration}{oksigennya}, bilangan \omterm{def:g7:oxygen-in-the-water:dissolved}{terlarut}
itu memetakan penghuninya
(\cref{prop:g6:exploring-our-environment:decide}, di bawah air):
\begin{itemize}
  \item \omterm{def:g5:biodiversity:species}{spesies} yang \emph{menuntut} --- \omterm{prop:g3:sorting-living-things:groups}{ikan} trout, larva lalat
        sehari dengan berkas \omterm{def:g7:how-animals-breathe:four}{insangnya} yang melambai
        (\cref{pb:g7:how-animals-breathe:1}) --- hanya hidup di air
        yang dingin, teraduk dan kaya \omterm{def:g7:what-is-respiration:respiration}{oksigen};
  \item \omterm{def:g5:biodiversity:species}{spesies} yang \emph{tahan} --- \omterm{prop:g3:sorting-living-things:groups}{ikan} mas, banyak siput kolam,
        larva berekor tikus dengan snorkel ke permukaannya ---
        bertahan di air yang hangat, tenang dan lebih miskin;
  \item sehingga pendataannya terbaca terbalik: temukan larva lalat
        sehari di bawah batunya dan airnya baik; temukan hanya para
        pemakai snorkel dan penelan udara, dan airnya buruk.
\end{itemize}
Penjaga sungai mendata kehidupan kecilnya justru karena alasan ini:
penghuninya adalah catatan tentang hari-hari terburuk perairannya
belakangan ini, yang dapat dilewatkan satu bacaan sensor tunggal.
\end{proposition}
\admitted

\begin{example}[Kisah limbah organiknya]\label{ex:g7:oxygen-in-the-water:waste}
Limbah sebuah desa yang tidak diolah --- kaya \omterm{def:g6:plants-are-producers:matter}{materi organik} ---
mencapai sebuah sungai yang jernih. Di hilir pipanya, para \omterm{def:g6:decomposers-and-soil:decomposers}{pengurainya}
berpesta lalu berlipat ganda
(\cref{def:g6:decomposers-and-soil:decomposers}), dan \omterm{def:g7:what-is-respiration:respiration}{respirasi}
massalnya menguras \omterm{def:g7:what-is-respiration:respiration}{oksigennya}: \omterm{prop:g3:sorting-living-things:groups}{ikan} trout dan larva lalat seharinya
lenyap, para pemakai snorkel yang tahan mengambil alih, dan lebih jauh
lagi ke hilir --- limbahnya sudah dilahap, airnya teraerasi ulang di
atas riamnya --- \omterm{def:g5:biodiversity:species}{spesies} yang menuntut itu kembali. Pencemarnya adalah
\omterm{def:g6:plants-are-producers:matter}{materi organik}; pembunuhnya adalah kelaparan \omterm{def:g7:what-is-respiration:respiration}{oksigen}; petanya mencatat
semuanya.
\end{example}

\begin{method}[Menilai sebuah perairan]\label{met:g7:oxygen-in-the-water:judge}
Untuk menilai sebuah sungai atau kolam:
\begin{enumerate}
  \item ukur atau perkirakan pemasoknya: \omterm{def:g6:exploring-our-environment:conditions}{suhu}, pengadukan, pertumbuhan
        hijaunya;
  \item timbanglah penguranya: lumpur dan materi mati, kepadatan,
        masuknya limbah organik apa pun;
  \item datalah kehidupan kecil di bawah batu dan di antara
        rerumputannya --- adakah \omterm{def:g5:biodiversity:species}{spesies} yang menuntut, atau hanya
        yang tahan?
  \item kalau mengambil contoh pada satu jam saja, ingatlah lengkung
        kolamnya: sebuah bacaan sore yang bagus dapat menyembunyikan
        fajar yang mematikan.
\end{enumerate}
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:1}
Kira-kira berapa banyak \omterm{def:g7:what-is-respiration:respiration}{oksigen} yang dimuat seliter air yang teraerasi
dengan baik, dan bagaimana perbandingannya dengan udara?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:2}
Sebutkan kedua pemasok \omterm{def:g7:what-is-respiration:respiration}{oksigen} sebuah perairan, dan \omterm{def:g6:exploring-our-environment:conditions}{kondisi} ketika
masing-masing memasok paling cepat.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:3}
Sebutkan tiga penguras persediaannya. Yang mana yang menjadi amat
besar ketika materi mati berlimpah?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:4}
Nyatakan perasan ganda kehangatan atas para pernapas air.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:5}
Mengapa lengkung \omterm{def:g7:what-is-respiration:respiration}{oksigen} kolamnya memuncak pada sore hari dan mencapai

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:oxygen-in-the-water:6}
Dua temuan pendataan yang mana yang menandai perairan yang kaya
\omterm{def:g7:what-is-respiration:respiration}{oksigen}, dan yang mana yang menandai perairan yang miskin?

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:7}

\cref{ex:g7:oxygen-in-the-water:curve} kepada seorang pemilik kolam

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:8}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:9}
Mengapa penjaga sungai lebih memercayai pendataan kehidupan kecilnya

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:10}
Sebuah kolam kebun berubah hijau seperti sup kacang, lalu mengalami
seminggu yang tenang dan panas. Telusuri bahayanya dengan kedua

mancurnya paling keras.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:oxygen-in-the-water:11}
Peternakan \omterm{prop:g3:sorting-living-things:groups}{ikan} trout dibangun di sungai yang dingin dengan bendung
dan saluran pacu yang memercik, tidak pernah di danau yang hangat dan
tenang. Belalah setiap anak kalimatnya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:oxygen-in-the-water:12}
``Di dalam air, \omterm{def:g7:what-is-respiration:respiration}{oksigen} adalah mata uangnya; \omterm{def:g6:exploring-our-environment:conditions}{suhu} menetapkan jumlah

dengan mengutip satu proposisi atau contoh untuk masing-masingnya.
\end{exercise}

\section{Soal: Buku Tahunan Sang Penjaga Sungai}

\begin{problem}[{Soal akhir pekan --- satu sungai, empat stasiun,
setahun bacaan}]\label{pb:g7:oxygen-in-the-water:1}
Seorang penjaga sungai menyimpan sebuah buku tahunan untuk empat
stasiun: K1, sebuah jeram dingin di bawah sebuah bendung; K2, sebuah
ruas lambat yang hangat melewati padang penggembalaan; K3, tepat di
bawah pipa limbah sebuah pabrik olahan \omterm{def:g2:animals-grow-up:born}{susu}; K4, dua kilometer lebih
jauh ke hilir, melewati sebuah riam berbatu yang panjang. Tiap stasiun
mendapat bacaan sensor dan sebuah pendataan kehidupan kecil.

\medskip\noindent\textbf{Bagian I --- Membaca stasiunnya.}
\begin{enumerate}
  \item K1 menunjukkan \qty{11}{mg/L} pada
        \qty{8}{\degreeCelsius}; K2 menunjukkan \qty{7}{mg/L}
        pada \qty{22}{\degreeCelsius}. Timpakan perbedaannya kepada
        pemasok dan langit-langit yang tepat.
  \item Pendataan K1: benih \omterm{prop:g3:sorting-living-things:groups}{ikan} trout, larva lalat sehari. Pendataan
        K2: \omterm{prop:g3:sorting-living-things:groups}{ikan} mas, siput kolam, larva berekor tikus. Tafsirkan
        keduanya dengan \cref{prop:g7:oxygen-in-the-water:map}.
  \item K3 menunjukkan \qty{3}{mg/L} walaupun airnya tidak lebih
        hangat daripada K2. Sebutkan pengurasnya, dan para pekerja di
        baliknya.
  \item K4 menunjukkan \qty{9}{mg/L} dan pendataannya kembali
        memperlihatkan larva lalat sehari. Jelaskan pemulihannya
        dengan kedua pemasoknya dan nasib limbahnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Irama tahunannya.}
\begin{enumerate}[resume]
  \item Bacaan fajar K2 pada bulan Agustus melorot sampai
        \qty{4}{mg/L} sementara bacaan sorenya mencapai
        \qty{9}{mg/L}. Gambarkan atau lukiskan lengkung hariannya,
        lalu sebutkan kedua proses yang bertukar tempat.
  \item Bacaan bulan Januari di K2 lebih mantap dan lebih tinggi
        daripada bacaan bulan Agustus, walaupun ruas padang
        penggembalaannya setenang biasanya. Berikan kedua alasan musim
        dinginnya.
  \item Diramalkan akan datang dua minggu yang panas dan tanpa angin.
        Urutkan keempat stasiunnya menurut risiko kematian \omterm{prop:g3:sorting-living-things:groups}{ikan},
        dengan satu alasan masing-masing.
  \item Penjaga sungainya mengatur kunjungan sensor bulan Agustusnya
        tepat sebelum matahari terbit. Belalah jadwalnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Putusan dan
penawarnya.}
\begin{enumerate}[resume]
  \item Pabrik olahan \omterm{def:g2:animals-grow-up:born}{susunya} memasang pengolahan yang menyingkirkan
        sebagian besar \omterm{def:g6:plants-are-producers:matter}{materi organik} dari air buangannya. Ramalkan
        sensor dan pendataan K3 setahun kemudian, menurut urutan
        pemulihannya.
  \item Seorang pemilik \omterm{def:g6:decomposers-and-soil:soil}{tanah} mengusulkan menebang pohon tepian K2
        ``supaya lebih banyak \omterm{def:g6:exploring-our-environment:conditions}{cahaya} membuat lebih banyak \omterm{def:g7:what-is-respiration:respiration}{oksigen}''.
        Timbanglah usulnya: apa lagi yang dikerjakan naungannya,
        menurut \cref{prop:g7:oxygen-in-the-water:drains}, dan apa
        jawabanmu?
  \item Pendataannya menemukan larva lalat sehari di sebuah stasiun
        yang satu bacaan sensor sorenya tampak biasa-biasa saja. Saksi
        mana yang kamu percayai tentang musimnya secara keseluruhan,
        dan mengapa?
  \item Tutuplah buku tahunannya dengan aturan satu kalimat sang
        penjaga sungai bagi seluruh sungainya, dengan memakai
        \emph{dingin}, \emph{teraduk} dan \emph{tidak terbebani}
        (limbah organik).
\end{enumerate}
\end{problem}
