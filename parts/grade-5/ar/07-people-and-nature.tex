\chapter{الناس والطبيعة}\label{ch:g5:people-and-nature}

من قمة تلّ، انظر إلى الوادي في الأسفل: حقول في مربعات أنيقة، \omterm{prop:g5:people-and-nature:managed}{وغابة
مُدارة}، وضفاف نهر مقوَّمة، وطرق، وسقوف. ولا يكاد شيء مما تراه قد
نما هكذا من تلقاء نفسه. فالبشر هم \omterm{def:g5:biodiversity:species}{النوع} الوحيد الذي يعيد تشكيل
مناظر طبيعية بأسرها على حاجاته --- وهذا الفصل ينظر بصدق إلى تلك
القدرة: ماذا تعطينا، وكم تكلّف، وكيف تُستعمل استعمالًا حسنًا.

\section{مناظر طبيعية صنعها الناس}

\begin{proposition}[أكثر المناظر الطبيعية مُدارة]\label{prop:g5:people-and-nature:managed}
يكاد كل منظر طبيعي حولنا يكون مشكّلًا بأيدي الناس:
\begin{itemize}
  \item \emph{الحقول}\index{حقل} أرض مُدارة يُزرع فيها \omterm{def:g1:plants-around-us:plant}{نبات} واحد
        مختار --- قمح، أو ذرة، أو بطاطا --- مكان الخليط البري؛
  \item وأكثر \emph{الغابات}\index{غابة مُدارة} مغروسة تُخفَّف
        وتُحصد كمحاصيل بطيئة؛
  \item و\emph{المروج} موجودة لأنها تُجزّ أو تُرعى --- فإن تُركت
        عادت أدغالًا ثم حرشًا؛
  \item والمدن والطرق والسدود تعيد رسم الأرض نفسها.
\end{itemize}
والأركان ``البكر'' استثناء، وأكثرها محمي عن قصد.
\end{proposition}
\admitted

\begin{example}[قراءة الوادي]\label{ex:g5:people-and-nature:valley}
منظر قمة التلّ ذاك، بندًا بندًا: حقل القمح يُطعم المخبز؛ وصفوف
الشربين محصول أخشاب أقدم من حارسه؛ والمرج المزهر يُجزّ كل تموز ---
ففوّت صيفين يستولِ عليه العُلَّيق؛ والنهر قُوّم منذ زمن بعيد لحماية
الحقول من الفيضانات. فالوادي مشغل --- مشغل يعمل منذ قرون.
\end{example}

\begin{remark}[لا شرير ولا قزم حديقة]\label{rem:g5:people-and-nature:honest}
يكون من الباطل أن نحكي هذه القصة تدميرًا خالصًا --- فالزراعة تُطعم
البشرية، والأرض الزراعية المسيَّجة قد تكون غنية \omterm{def:g5:biodiversity:species}{بالأنواع}
(\cref{ex:g5:biodiversity:orchard}) --- ومن الباطل مثله أن ندّعي
أن إعادة التشكيل لا تكلّف شيئًا
(وقد عدّدت \cref{prop:g5:biodiversity:loss} الفواتير). والسؤال
الصادق ليس ``أنمسّ الطبيعة أم لا؟'' بل ``كيف ندير ما نمسّه ليبقى
عاملًا --- لنا ولبقية الشبكة؟''
\end{remark}

\section{الإدارة الحسنة والإدارة السيئة}

\begin{example}[طريقتان لزراعة سفح]\label{ex:g5:people-and-nature:twofarms}
السفح الأول: حقل عملاق واحد، والأسيجة مقتلعة، والتربة عارية الشتاء
كله. فيغسل المطر التربة المفكّكة إلى أسفل؛ ولا بيت لمعيني
\cref{ex:g5:biodiversity:orchard}؛ وآفة واحدة تكتسح المحصول
الأوحد بلا مقاوم. والسفح الثاني: حقول أصغر وأسيجة محفوظة، وتربة
مغطاة طول الشتاء، ومحاصيل متناوبة سنة بعد سنة. فالأسيجة تمسك
التربة وتؤوي آكلي الآفات؛ والتناوب يريح الأرض. وكلا السفحين يُطعم
الناس --- لكن أحدهما سيظل يفعل ذلك بعد مئة سنة.
\end{example}

\begin{proposition}[قواعد الاستعمال الدائم]\label{prop:g5:people-and-nature:lasting}
يدوم استعمال الطبيعة حين يحترم ثلاث قواعد:
\begin{enumerate}
  \item \emph{لا تأخذ أسرع مما يتجدد}: فاحصد الخشب على وتيرة نمو
        الغابة، واصطد السمك على وتيرة تكاثر السرب --- وإفراط
        الصيد في \cref{prop:g5:biodiversity:loss} خرق لهذه
        القاعدة؛
  \item و\emph{أبقِ العمّال}: فحياة التربة، والمؤبّرون، وآكلو
        الآفات --- وهي الخدمات المجانية في
        \cref{exo:g5:biodiversity:11} --- تحتاج إلى إبقاء
        بيوتها؛
  \item و\emph{ردّ ما تستعير}: أي حلقة السماد في
        \cref{rem:g4:what-plants-need:soil}، على كل مقياس.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[حساب الحرّاج]\label{ex:g5:people-and-nature:forester}
الغابة الحسنة الإدارة أقدم برهان على نجاح القاعدة الأولى: فالحرّاج
يحصون ما تنميه الغابة كل سنة، ولا يقطعون أكثر منه. فتغلّ الغابة
عندئذ خشبًا كل سنة إلى الأبد --- وتبقى غابة، بأبوامها وأيائلها
وخنافسها. وبعض \omterm{prop:g5:people-and-nature:managed}{الغابات} المُدارة في العالم حُصدت هكذا مئات السنين
من غير أن تتقلص.
\end{example}

\section{المدن، وإفساح المكان من جديد}

\begin{example}[الطبيعة في المدينة]\label{ex:g5:people-and-nature:city}
تبدو المدينة نقيض الطبيعة، ومع ذلك فهي \omterm{def:g2:where-animals-live:habitat}{موئل} أيضًا --- \omterm{def:g2:where-animals-live:habitat}{موئل} صخري
غريب. فالسمامة تعشّش على ``جروف'' من الخرسانة، والثعالب تجوب
الحاويات، \omterm{def:g1:plants-around-us:plant}{والنباتات} البرية تشقّ شقوق الإسفلت. ويستطيع سكان المدن
أن يوسّعوا الترحيب: بالمتنزهات، والسقوف الخضراء، وأزهار الشرفات
للنحل، وبركة المدرسة في \cref{ex:g2:where-animals-live:schoolpond}.
فطرائق \cref{met:g5:biodiversity:help} تصلح بين المباني أيضًا.
\end{example}

\begin{example}[إعادة انحناءات النهر إليه]\label{ex:g5:people-and-nature:river}
نهر الوادي المقوَّم جرى سريعًا وفاض فيضانًا سيئًا في الأسفل؛ وذهبت
أسماكه ورعّاشاته مع انحناءاته. وصار المهندسون أحيانًا يعيدون
\emph{ثني} هذه الأنهار: فالانحناءات المستعادة والمروج الرطبة تبطّئ
الماء وتخزن الفيضانات، وتجد لقالق \cref{ex:g5:biodiversity:storks}
مروجها من جديد. فالإدارة يمكن أن تجري إلى الوراء --- والإصلاح إدارة
أيضًا.
\end{example}

\begin{method}[الحكم على استعمال للطبيعة]\label{met:g5:people-and-nature:judge}
إذا أُعطيت أي استعمال لأرض أو ماء --- مزرعة، أو مصيدة سمك، أو طريقًا
مخططًا، أو حديقة:
\begin{enumerate}
  \item اسأل ما الذي \emph{يُؤخذ}، وأهو أسرع مما يتجدد (القاعدة
        الأولى)؛
  \item واسأل أي \emph{بيوت وعمّال} تُحفظ أو تُفقد (القاعدة
        الثانية)؛
  \item واسأل ما الذي \emph{يُردّ} --- وأي فضلات تُترك (القاعدة
        الثالثة)؛
  \item ثم احكم: أيمكن أن يستمر هذا مئة سنة؟ وأي تغيير واحد يحسّن
        الجواب أكثر من غيره؟
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:1}
أعطِ أربعة أصناف من المناظر الطبيعية التي شكّلها الإنسان في
\cref{prop:g5:people-and-nature:managed}، ولكل واحد ما يُدار من
أجله.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:2}
ماذا يحدث لمرج يكفّ عن أن يُجزّ أو يُرعى؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:3}
اذكر قواعد الاستعمال الدائم الثلاث.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:4}
في \cref{ex:g5:people-and-nature:twofarms}، عدّد ثلاثة فروق بين
السفحين، وما يغيّره كل فرق.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:5}
اشرح حساب الحرّاج. وأي قاعدة هو، وماذا تغلّ الغابة في المقابل ---
وإلى متى؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:6}
سمِّ ثلاثة سكان بريين للمدن، وأي صفة من صفات المدينة يستعملها كل
واحد موئلًا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:7}
ماذا استعاد إعادة ثني النهر --- للناس، وللشبكة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:people-and-nature:8}
جرّب: اختر منظرًا واحدًا قرب البيت --- شارعًا، أو حقلًا، أو متنزهًا
--- وعدّد كل ما يديره الناس فيه، كما في
\cref{ex:g5:people-and-nature:valley}. أبقي شيء غير مُدار؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:people-and-nature:9}
``حماية الطبيعة تعني ألّا نمسّها أبدًا.'' ناقش ذلك مستعملًا المرج
المجزوز في \cref{prop:g5:people-and-nature:managed} --- وهو \omterm{def:g2:where-animals-live:habitat}{موئل}
\emph{يختفي} من غير مسّ الإنسان --- ومستعملًا
\cref{rem:g5:people-and-nature:honest}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:people-and-nature:10}
طبّق \cref{met:g5:people-and-nature:judge} على السفح الأول في
\cref{ex:g5:people-and-nature:twofarms}: أي القواعد يخرقها، وأي
تغيير واحد يحسّنه أكثر من غيره؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:people-and-nature:11}
يضاعف أسطول صيد صيده خمس سنوات باهرة؛ وفي السادسة تختفي الأسراب
ويقف الأسطول عاطلًا. اشرح السنوات الست بالقاعدة الأولى --- ولماذا
\emph{بدا} خرقها نجاحًا في البداية؟ قارن ذلك بحساب الحرّاج المقابل.
\end{exercise}
