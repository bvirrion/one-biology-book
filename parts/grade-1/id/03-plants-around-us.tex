\chapter{Tumbuhan di Sekitar Kita}\label{ch:g1:plants-around-us}

Setangkai \omterm{def:g1:plants-around-us:parts}{bunga} aster di halaman rumput, lumut di tembok, \omterm{def:g1:plants-around-us:tree}{pohon}
berangan besar di halaman sekolah: tidak satu pun akan pernah lari,
namun semuanya hidup. \omterm{def:g1:plants-around-us:plant}{Tumbuhan} adalah \omterm{def:g1:living-or-not:living}{makhluk hidup} yang diam ---
mereka lahir, makan, dan tumbuh, kadang selama ratusan tahun, tanpa
sekali pun berpindah dari tempatnya.

\section{Tumbuhan adalah makhluk hidup}

\begin{definition}[Tumbuhan]\label{def:g1:plants-around-us:plant}
Suatu \emph{tumbuhan}\index{tumbuhan} adalah \omterm{def:g1:living-or-not:living}{makhluk hidup} yang
menetap di tanah dan membuat makanannya sendiri dengan memakai cahaya.
Rumput, \omterm{def:g1:plants-around-us:parts}{bunga} aster, tanaman tomat dan \omterm{def:g1:plants-around-us:tree}{pohon} semuanya tumbuhan.
Kebanyakan tumbuhan berwarna hijau.
\end{definition}

\begin{example}[Bunga asternya hidup]\label{ex:g1:plants-around-us:daisy}
Periksa tanda-tanda kehidupan pada setangkai \omterm{def:g1:plants-around-us:parts}{bunga} aster: ia lahir
dari biji; ia makan --- dengan air dari tanah dan cahaya dari langit;
ia tumbuh; ia membuat biji yang menjadi \omterm{def:g1:plants-around-us:parts}{bunga} aster baru; ia mati
ketika beku. Semua tandanya ada. \omterm{def:g1:plants-around-us:parts}{Bunga} aster sama hidupnya dengan
kucing --- ia hanya jauh lebih diam tentang hal itu.
\end{example}

{\small \omterm{def:g1:plants-around-us:parts}{Bunga} aster di halaman rumput: lahir dari biji, makan dengan

\begin{remark}[Hidup tanpa berlari]\label{rem:g1:plants-around-us:still}
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} tidak berjalan, tetapi mereka tidak sepenuhnya diam. \omterm{def:g1:plants-around-us:parts}{Bunga}
matahari berputar mengikuti cahaya sepanjang hari, dan \omterm{def:g1:plants-around-us:plant}{tumbuhan} di
ambang jendela perlahan condong ke arah jendela. \omterm{def:g1:plants-around-us:plant}{Tumbuhan} bergerak
dengan cara \emph{tumbuh} --- terlalu lambat bagi mata kita, cukup
cepat bagi pengamatan seminggu.
\end{remark}

\section{Bagian-bagian tumbuhan}

\begin{definition}[Bagian-bagian tumbuhan]\label{def:g1:plants-around-us:parts}
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} berbunga mempunyai empat bagian utama:
\begin{enumerate}
  \item \emph{akar}\index{akar}, tersembunyi di dalam tanah, yang
        menahan \omterm{def:g1:plants-around-us:plant}{tumbuhannya} pada tempatnya dan minum air;
  \item \emph{batang}\index{batang}, yang menegakkan \omterm{def:g1:plants-around-us:plant}{tumbuhannya} dan
        mengantar air ke daun;
  \item \emph{daun}\index{daun}, pipih dan hijau, yang menangkap
        cahaya;
  \item \emph{bunga}\index{bunga}, yang kelak membuat biji.
\end{enumerate}
\end{definition}

\begin{example}[Mencabut rumput liar]\label{ex:g1:plants-around-us:weed}
Cabut sebatang rumput liar pelan-pelan dari tanah gembur yang basah,
dan bagian yang tersembunyi ikut terangkat: seberkas \omterm{def:g1:plants-around-us:parts}{akar} pucat,
kadang lebih panjang daripada seluruh bagian hijau di atasnya.
Sekarang \omterm{def:g1:plants-around-us:plant}{tumbuhannya} dapat dibaca dari bawah ke atas: \omterm{def:g1:plants-around-us:parts}{akar}, \omterm{def:g1:plants-around-us:parts}{batang},
\omterm{def:g1:plants-around-us:parts}{daun} --- dan, kalau kamu beruntung, sekuntum \omterm{def:g1:plants-around-us:parts}{bunga}.
\end{example}

\begin{definition}[Pohon]\label{def:g1:plants-around-us:tree}
Suatu \emph{pohon}\index{pohon} adalah \omterm{def:g1:plants-around-us:plant}{tumbuhan} besar yang \omterm{def:g1:plants-around-us:parts}{batangnya}
terbuat dari kayu. \omterm{def:g1:plants-around-us:parts}{Batang} berkayu sebuah pohon disebut
\emph{batang pokok}\index{batang pokok}; ia bercabang menjadi
\emph{ranting}\index{ranting}, yang membawa \omterm{def:g1:plants-around-us:parts}{daun}.
\end{definition}

\begin{example}[Pohon berangan]\label{ex:g1:plants-around-us:tree}
\omterm{def:g1:plants-around-us:tree}{Pohon} berangan di halaman mempunyai rancangan yang sama dengan \omterm{def:g1:plants-around-us:parts}{bunga}
aster, tetapi lebih besar dan lebih keras: \omterm{def:g1:plants-around-us:parts}{akar} di dalam tanah, \omterm{def:g1:plants-around-us:tree}{batang
pokok} sebagai ganti \omterm{def:g1:plants-around-us:parts}{batang} yang lunak, cabang-cabang dengan ribuan
\omterm{def:g1:plants-around-us:parts}{daun}, dan pada musim semi, \omterm{def:g1:plants-around-us:parts}{bunga}. \omterm{def:g1:plants-around-us:parts}{Bunga} aster hidup satu tahun; \omterm{def:g1:plants-around-us:tree}{pohon}
berangan dapat hidup lebih dari seratus tahun.
\end{example}

\section{Apa yang dibutuhkan tumbuhan}

\begin{proposition}[Kebutuhan tumbuhan]\label{prop:g1:plants-around-us:needs}
Untuk hidup dan tumbuh, sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan} membutuhkan \emph{air},
\emph{cahaya}, dan tempat bagi \omterm{def:g1:plants-around-us:parts}{akarnya}, biasanya
\emph{tanah}\index{tanah}. \omterm{def:g1:plants-around-us:plant}{Tumbuhan} yang disimpan dalam gelap, atau
tidak pernah disiram, akan layu lalu mati.
\end{proposition}
\admitted

\begin{example}[Tumbuhan yang terlupakan]\label{ex:g1:plants-around-us:wilt}
Lupa menyiram \omterm{def:g1:plants-around-us:plant}{tumbuhan} di kelas sebelum liburan: \omterm{def:g1:plants-around-us:parts}{daunnya} terkulai,
lemas dan sedih --- ia layu. Siram ia, dan beberapa jam kemudian
\omterm{def:g1:plants-around-us:parts}{daunnya} tegak kembali. Air adalah kebutuhan yang hilang.
\end{example}

\begin{method}[Merawat sebuah tumbuhan]\label{met:g1:plants-around-us:care}
\begin{enumerate}
  \item Letakkan potnya di tempat yang dijangkau cahaya siang;
  \item siram tanahnya ketika terasa kering --- bukan tiap jam, sebab
        \omterm{def:g1:plants-around-us:parts}{akarnya} juga butuh udara;
  \item jangan menarik \omterm{def:g1:plants-around-us:parts}{daunnya} sampai lepas: \omterm{def:g1:plants-around-us:parts}{daun} adalah penangkap
        cahaya bagi \omterm{def:g1:plants-around-us:plant}{tumbuhannya};
  \item bersabarlah: \omterm{def:g1:plants-around-us:plant}{tumbuhan} menjawab dalam hitungan hari, bukan
        menit.
\end{enumerate}
\end{method}

\begin{remark}[Tumbuhan memberi makan semua]\label{rem:g1:plants-around-us:food}
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} penting bagi setiap \omterm{def:g1:animals-around-us:animal}{hewan}. Siput makan selada, burung hitam
makan buah beri, kita makan roti yang dibuat dari gandum --- sebuah
\omterm{def:g1:plants-around-us:plant}{tumbuhan}. Bahkan kucing, yang tidak makan selada, memakan \omterm{def:g1:animals-around-us:animal}{hewan} yang
telah memakan \omterm{def:g1:plants-around-us:plant}{tumbuhan}. Tanpa \omterm{def:g1:plants-around-us:plant}{tumbuhan}, tidak ada lagi yang punya
sesuatu untuk dimakan; sebuah bab berikutnya menelusuri rantai makanan
ini.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:1}
Sebutkan keempat bagian \omterm{def:g1:plants-around-us:plant}{tumbuhan} berbunga, dari bawah tanah sampai ke
pucuknya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:2}
Bagian \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang mana: (a) minum air dari tanah? (b) menangkap
cahaya? (c) kelak membuat biji? (d) menahan \omterm{def:g1:plants-around-us:plant}{tumbuhannya} tetap tegak?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:3}
\omterm{def:g1:plants-around-us:parts}{Bunga} aster tidak pernah berjalan. Telusuri tanda-tanda kehidupan dan
katakan mengapa ia tetap \omterm{def:g1:living-or-not:living}{makhluk hidup}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:4}
Apa nama khusus bagi \omterm{def:g1:plants-around-us:parts}{batang} sebuah \omterm{def:g1:plants-around-us:tree}{pohon} dan bagi bagian-bagian yang
membawa \omterm{def:g1:plants-around-us:parts}{daunnya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:5}
Sebutkan tiga hal yang dibutuhkan \omterm{def:g1:plants-around-us:plant}{tumbuhan} untuk hidup, menurut
\cref{prop:g1:plants-around-us:needs}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:6}
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} di kelas berdaun lemas dan terkulai. Apa yang mungkin
terlupakan? Apa yang akan terjadi sesudah hal itu diberikan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:7}
Dalam hal apa saja \omterm{def:g1:plants-around-us:parts}{bunga} aster dan \omterm{def:g1:plants-around-us:tree}{pohon} berangan sama? Dalam dua hal
apa mereka berbeda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:8}
Cobalah: letakkan satu tanaman kacang (atau pot selada air) dekat
jendela dan satu lagi di dalam lemari yang gelap, lalu siram keduanya.
Amati sesudah seminggu. Perbedaan apa yang kamu duga, dan kebutuhan
mana yang menjelaskannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-around-us:9}
Leo berkata: ``\omterm{def:g1:plants-around-us:plant}{Tumbuhan} sebenarnya tidak hidup --- mereka tidak pernah
melakukan apa pun.'' Berikan kepadanya dua hal yang dilakukan
\omterm{def:g1:plants-around-us:plant}{tumbuhan}, pelan tetapi pasti, dari
\cref{rem:g1:plants-around-us:still} dan
\cref{ex:g1:plants-around-us:daisy}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-around-us:10}
Wortel berwarna jingga dan tumbuh di dalam tanah. Dengan melihat
gambar bagian-bagian \omterm{def:g1:plants-around-us:plant}{tumbuhan}, menurutmu bagian mana dari tanaman
wortel yang kita makan? Petunjuk apa yang diberikan kata ``di dalam
tanah''?
\end{exercise}
