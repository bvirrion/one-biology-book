\section*{अध्याय \ref{ch:g1:plants-around-us} --- हमारे आस-पास के पौधे}

\begin{solution}{exo:g1:plants-around-us:1}
\omterm{def:g1:plants-around-us:parts}{जड़ें} (मिट्टी के नीचे), \omterm{def:g1:plants-around-us:parts}{डंठल}, \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}, \omterm{def:g1:plants-around-us:parts}{फूल}।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:2}
(क) \omterm{def:g1:plants-around-us:parts}{जड़ें}। (ख) \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}। (ग) \omterm{def:g1:plants-around-us:parts}{फूल}। (घ) \omterm{def:g1:plants-around-us:parts}{डंठल}।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:3}
वह एक बीज से जन्मी, वह पोषण लेती है (मिट्टी से पानी, आसमान से
प्रकाश), वह बढ़ती है, वह ऐसे बीज बनाती है जिनसे नई गुलबहार बनती हैं,
और एक दिन वह मर जाती है। जीवन के सारे लक्षण मौजूद हैं; चलना उनमें से
एक नहीं है।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:4}
\omterm{def:g1:plants-around-us:tree}{पेड़} के \omterm{def:g1:plants-around-us:parts}{डंठल} को तना कहते हैं; \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} शाखाएँ लिए रहती हैं।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:5}
पानी, प्रकाश, और \omterm{def:g1:plants-around-us:parts}{जड़ों} के लिए एक जगह --- आम तौर पर मिट्टी।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:6}
पानी देना। पानी न मिलने से \omterm{def:g1:plants-around-us:plant}{पौधा} मुरझा गया है; पानी मिलने के कुछ घंटों
बाद उसकी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} फिर तन जाएँगी।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:7}
एक जैसे: दोनों \omterm{def:g1:plants-around-us:plant}{पौधे} हैं, दोनों में \omterm{def:g1:plants-around-us:parts}{जड़ें}, \omterm{def:g1:plants-around-us:parts}{डंठल}, \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} और \omterm{def:g1:plants-around-us:parts}{फूल} होते
हैं, और दोनों को पानी तथा प्रकाश चाहिए। अलग: \omterm{def:g1:plants-around-us:tree}{पेड़} का \omterm{def:g1:plants-around-us:parts}{डंठल} लकड़ी का
तना है जबकि गुलबहार का \omterm{def:g1:plants-around-us:parts}{डंठल} मुलायम और हरा है, और \omterm{def:g1:plants-around-us:tree}{पेड़} कहीं बड़ा है
तथा कहीं ज़्यादा जीता है।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:8}
खिड़की वाला \omterm{def:g1:plants-around-us:plant}{पौधा} हरा और सीधा रहना चाहिए; अलमारी वाला \omterm{def:g1:plants-around-us:plant}{पौधा} पीला, खिंचा
हुआ और कमज़ोर, हालाँकि उसे पानी मिला था। गुम ज़रूरत प्रकाश है ---
अकेला पानी काफ़ी नहीं।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:9}
जैसे: \omterm{def:g1:plants-around-us:plant}{पौधा} बढ़ता है --- हफ़्ते-दर-हफ़्ते ऊँचा होता है --- और वह
प्रकाश की ओर मुड़ता या झुकता है, जैसे सूरजमुखी दिन का पीछा करता है।
धीमा होना कुछ न करने के बराबर नहीं है।
\end{solution}

\begin{solution}{exo:g1:plants-around-us:10}
\omterm{def:g1:plants-around-us:parts}{जड़}। ``ज़मीन के नीचे'' चित्र में \omterm{def:g1:plants-around-us:parts}{जड़ों} की जगह है --- हम जो गाजर खाते
हैं वह एक बड़ी, मोटी नारंगी \omterm{def:g1:plants-around-us:parts}{जड़} है जिसे \omterm{def:g1:plants-around-us:plant}{पौधे} ने भोजन से भर दिया है।
\end{solution}
