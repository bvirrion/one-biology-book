\chapter{Adapted to Their World}\label{ch:g4:adapted-to-their-world}

Put a polar bear in the Sahara and a camel on the pack ice, and
neither would last a season. Yet each, at home, is a champion:
the bear swims iced seas in comfort, the camel crosses deserts
that would finish a horse in days. \omterm{def:g1:living-or-not:living}{Living things} fit their
worlds with an exactness that repays close study --- feature by
feature.

\section{What adaptation means}

\begin{definition}[Adaptation]\label{def:g4:adapted-to-their-world:adaptation}
An \emph{adaptation}\index{adaptation} is a feature of a \omterm{def:g1:living-or-not:living}{living
thing}'s body or behavior that fits it to the conditions of its
\omterm{def:g2:caring-for-nature:environment}{environment} --- to the cold or the drought, to its food, to
its hunters. \Cref{prop:g2:where-animals-live:fit} showed
\omterm{def:g1:animals-around-us:animal}{animals} fitting their \omterm{def:g2:where-animals-live:habitat}{habitats}; adaptation is that fit,
examined piece by piece.
\end{definition}

\begin{example}[The camel, piece by piece]\label{ex:g4:adapted-to-their-world:camel}
The desert's problems: heat, no water, soft sand. The camel's
answers: humps of fat --- a \omterm{def:g3:animals-through-seasons:active}{food store} for lean crossings;
the ability to drink a hundred litres in minutes and then go
days without; wide two-toed feet that do not sink in sand;
nostrils that close against blowing dust and long double
eyelashes to match. Every desert problem, met by a feature.
\end{example}

\includegraphics[width=0.6\linewidth]{images/book1/ai/g4-adapted-camel.png}

{\small The desert specialist at work: wide feet on soft sand,
a hump of reserves, and eyelashes built for dust.}
\end{omfigure}

\begin{example}[The polar bear, piece by piece]\label{ex:g4:adapted-to-their-world:bear}
The pack ice's problems: cold, icy water, prey that is hard to
approach on white ice. The bear's answers: thick \omterm{def:g1:animals-around-us:covering}{fur} over a
deep layer of fat --- double insulation; broad, slightly
webbed paws --- snowshoes and paddles in one; white coat ---
invisible against the snow it hunts on. The same exactness,
tuned to the opposite world.
\end{example}

\begin{proposition}[Adaptations answer problems]\label{prop:g4:adapted-to-their-world:answers}
Wherever the \omterm{def:g2:caring-for-nature:environment}{environment} sets a hard problem, the \omterm{def:g1:living-or-not:living}{living
things} at home there carry features answering it:
\begin{itemize}
  \item cold: thick \omterm{def:g1:animals-around-us:covering}{fur}, fat layers, small \omterm{def:g1:the-five-senses:sense}{ears} that lose
        little warmth;
  \item drought: water storage, waxy \omterm{def:g1:the-five-senses:sense}{skins}, deep \omterm{def:g1:plants-around-us:parts}{roots};
  \item life in water: streamlined bodies, fins and webbed
        feet, water-shedding coats;
  \item hunters all around: camouflage colors, armor, \omterm{def:g3:bones-and-muscles:skeleton}{spines},
        speed, and \omterm{def:g1:the-five-senses:sense}{eyes} set wide to watch all round;
  \item hunting for a living: sharp senses, forward \omterm{def:g1:the-five-senses:sense}{eyes},
        claws and speed
        (\cref{prop:g2:what-animals-eat:match}).
\end{itemize}
\end{proposition}
\admitted

\section{Plants are adapted too}

\begin{example}[The cactus]\label{ex:g4:adapted-to-their-world:cactus}
The desert tests \omterm{def:g1:plants-around-us:plant}{plants} hardest --- they cannot walk to
water. The cactus answers: a thick, barrel-shaped \omterm{def:g1:plants-around-us:parts}{stem} that
stores water and does the \omterm{def:g1:plants-around-us:parts}{leaves}' light-catching work; \omterm{def:g1:plants-around-us:parts}{leaves}
shrunk to \omterm{def:g3:bones-and-muscles:skeleton}{spines}, losing almost no water --- and defending
the barrel against thirsty \omterm{def:g1:animals-around-us:animal}{animals}; \omterm{def:g1:plants-around-us:parts}{roots} spread wide and
shallow to catch a whole rare shower at once.
\end{example}

\begin{example}[Alpine cushions]\label{ex:g4:adapted-to-their-world:alpine}
High mountain \omterm{def:g1:plants-around-us:plant}{plants} face freezing wind. Many grow as tight,
ground-hugging cushions: inside the cushion, the wind is
broken and the air stays milder — a shape that is shelter.
Lowland cousins of the same \omterm{def:g1:plants-around-us:plant}{plants} grow tall and loose; the
mountain forms hug the rock.
\end{example}

\begin{method}[Reading an organism's world from its body]\label{met:g4:adapted-to-their-world:read}
Given an unfamiliar \omterm{def:g1:animals-around-us:animal}{animal} or \omterm{def:g1:plants-around-us:plant}{plant}:
\begin{enumerate}
  \item list its striking features --- coat, feet, \omterm{def:g1:the-five-senses:sense}{eyes}, body
        shape, \omterm{def:g1:plants-around-us:parts}{leaves}, \omterm{def:g1:plants-around-us:parts}{roots};
  \item for each, ask: \emph{what problem would this solve?}
  \item assemble the answers into a picture of its home ---
        cold or hot, wet or dry, hunter or hunted;
  \item check the guess against where the organism really
        lives.
\end{enumerate}
This is \cref{met:g2:where-animals-live:matching} grown up:
from matching a \omterm{def:g2:where-animals-live:habitat}{habitat} to reconstructing a world.
\end{method}

\begin{example}[The method on a mystery skin]\label{ex:g4:adapted-to-their-world:mystery}
A museum drawer holds an unlabelled \omterm{def:g1:the-five-senses:sense}{skin}: dense soft \omterm{def:g1:animals-around-us:covering}{fur},
tiny \omterm{def:g1:the-five-senses:sense}{eyes}, huge shovel-shaped front claws. \omterm{def:g1:animals-around-us:covering}{Fur} says mammal;
the shovels say digging; the useless \omterm{def:g1:the-five-senses:sense}{eyes} say life in the
dark. Verdict: an underground digger --- and indeed it is a
mole (\cref{ex:g2:where-animals-live:mole}), read correctly
from three features alone.
\end{example}

\begin{remark}[How does the fit come about?]\label{rem:g4:adapted-to-their-world:how}
A camel never \emph{decides} to grow wide feet, and no polar
bear chooses its white coat. How, then, do \omterm{def:g1:living-or-not:living}{living things} come
to fit their worlds so well? It is one of \omterm{def:g1:living-or-not:living}{biology}'s deepest
questions, and its full answer --- how the fit builds up,
generation after generation --- is a story this book keeps
for a later year. For now, learn to \emph{see} the fit: the
seeing is what made the question askable.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:1}
What is an \omterm{def:g4:adapted-to-their-world:adaptation}{adaptation}? Give one \omterm{def:g1:animals-around-us:animal}{animal} example and one \omterm{def:g1:plants-around-us:plant}{plant}
example.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:2}
Name three desert problems and the camel's answer to each.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:3}
Name three pack-ice problems and the polar bear's answer to
each.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:4}
What jobs do the cactus's \omterm{def:g3:bones-and-muscles:skeleton}{spines} do? What does its barrel
\omterm{def:g1:plants-around-us:parts}{stem} store, and what work has the \omterm{def:g1:plants-around-us:parts}{stem} taken over?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:5}
Why do many high-mountain \omterm{def:g1:plants-around-us:plant}{plants} grow as tight cushions?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:6}
Give two \omterm{def:g4:adapted-to-their-world:adaptation}{adaptations} of hunted \omterm{def:g1:animals-around-us:animal}{animals} and two of hunters,
from \cref{prop:g4:adapted-to-their-world:answers}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:7}
Retell \cref{ex:g4:adapted-to-their-world:mystery}: which
three features gave the mole away, and what did each one
say?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:8}
Try it: pick any \omterm{def:g1:animals-around-us:animal}{animal} you can watch --- a duck, a cat, a
spider --- and run
\cref{met:g4:adapted-to-their-world:read} on it: three
features, three problems solved.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:9}
The desert fox has enormous \omterm{def:g1:the-five-senses:sense}{ears}; the arctic fox, tiny ones.
Using the cold line of
\cref{prop:g4:adapted-to-their-world:answers}, explain both
--- what must big \omterm{def:g1:the-five-senses:sense}{ears} do with warmth?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:10}
A mystery \omterm{def:g1:animals-around-us:animal}{animal}: streamlined body, webbed hind feet, dense
water-shedding \omterm{def:g1:animals-around-us:covering}{fur}, whiskers, and it holds its breath well.
Reconstruct its world with
\cref{met:g4:adapted-to-their-world:read}. (It builds dams.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:adapted-to-their-world:11}
The whale (\cref{ex:g3:sorting-living-things:surprises}) is a
mammal shaped like a \omterm{prop:g3:sorting-living-things:groups}{fish}: streamlined, finned, smooth. The
trout is a \omterm{prop:g3:sorting-living-things:groups}{fish} shaped the same way. What does this
resemblance between two \emph{unrelated} swimmers show about
\omterm{def:g4:adapted-to-their-world:adaptation}{adaptations} --- who set the shape, the family or the water?
\end{exercise}
