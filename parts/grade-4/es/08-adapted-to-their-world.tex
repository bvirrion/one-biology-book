\chapter{Adaptados a su mundo}\label{ch:g4:adapted-to-their-world}

Pon un oso polar en el Sáhara y un camello sobre el hielo marino y
ninguno de los dos aguantaría una temporada. Y sin embargo, cada uno
en su casa es un campeón: el oso nada tan a gusto en mares helados y
el camello cruza desiertos que acabarían con un caballo en unos días.
Los \omterm{def:g1:living-or-not:living}{seres vivos} encajan en su mundo con una exactitud que compensa
estudiar de cerca --- rasgo a rasgo.

\section{Qué significa adaptación}

\begin{definition}[Adaptación]\label{def:g4:adapted-to-their-world:adaptation}
Una \emph{adaptación}\index{adaptación} es un rasgo del cuerpo o del
comportamiento de un \omterm{def:g1:living-or-not:living}{ser vivo} que lo ajusta a las condiciones de su
\omterm{def:g2:caring-for-nature:environment}{entorno} --- al frío o a la sequía, a su comida, a los que lo cazan.
La \cref{prop:g2:where-animals-live:fit} enseñaba a los \omterm{def:g1:animals-around-us:animal}{animales}
encajando en su \omterm{def:g2:where-animals-live:habitat}{hábitat}; la adaptación es ese encaje, examinado pieza
a pieza.
\end{definition}

\begin{example}[El camello, pieza a pieza]\label{ex:g4:adapted-to-their-world:camel}
Los problemas del desierto: calor, nada de agua, arena blanda. Las
respuestas del camello: jorobas de grasa --- una despensa para las
travesías flacas; la capacidad de beber cien litros en unos minutos y
aguantar después días sin beber; pies anchos de dos dedos que no se
hunden en la arena; unos ollares que se cierran contra el polvo que
levanta el viento y unas pestañas dobles y largas a juego. Cada
problema del desierto, resuelto por un rasgo.
\end{example}

\begin{example}[El oso polar, pieza a pieza]\label{ex:g4:adapted-to-their-world:bear}
Los problemas del hielo marino: frío, agua helada y presas difíciles
de abordar sobre hielo blanco. Las respuestas del oso: \omterm{def:g1:animals-around-us:covering}{pelo} espeso
sobre una capa profunda de grasa --- doble aislamiento; zarpas anchas
y algo palmeadas --- raquetas de nieve y remos a la vez; pelaje
blanco --- invisible contra la nieve sobre la que caza. La misma
exactitud, afinada para el mundo contrario.
\end{example}

\begin{proposition}[Las adaptaciones responden a problemas]\label{prop:g4:adapted-to-their-world:answers}
Allí donde el \omterm{def:g2:caring-for-nature:environment}{entorno} plantea un problema duro, los \omterm{def:g1:living-or-not:living}{seres vivos} que
viven allí llevan rasgos que lo responden:
\begin{itemize}
  \item frío: \omterm{def:g1:animals-around-us:covering}{pelo} espeso, capas de grasa, \omterm{def:g1:the-five-senses:sense}{orejas} pequeñas que pierden
        poco calor;
  \item sequía: almacenes de agua, \omterm{def:g1:the-five-senses:sense}{pieles} cerosas, \omterm{def:g1:plants-around-us:parts}{raíces} profundas;
  \item vida en el agua: cuerpos afinados, aletas y pies palmeados,
        pelajes que repelen el agua;
  \item cazadores por todas partes: colores de camuflaje, armaduras,
        púas, velocidad y \omterm{def:g1:the-five-senses:sense}{ojos} muy separados para vigilar alrededor;
  \item vivir de la caza: sentidos finos, \omterm{def:g1:the-five-senses:sense}{ojos} hacia delante, garras y
        velocidad (\cref{prop:g2:what-animals-eat:match}).
\end{itemize}
\end{proposition}
\admitted

\section{Las plantas también están adaptadas}

\begin{example}[El cactus]\label{ex:g4:adapted-to-their-world:cactus}
El desierto es donde más se pone a prueba a las \omterm{def:g1:plants-around-us:plant}{plantas} --- no pueden
andar hasta el agua. El cactus responde: un \omterm{def:g1:plants-around-us:parts}{tallo} grueso en forma de
barril que almacena agua y hace el trabajo de atrapar la luz que
harían las \omterm{def:g1:plants-around-us:parts}{hojas}; \omterm{def:g1:plants-around-us:parts}{hojas} reducidas a púas, que casi no pierden agua ---
y defienden el barril de los \omterm{def:g1:animals-around-us:animal}{animales} sedientos; \omterm{def:g1:plants-around-us:parts}{raíces} extendidas a
lo ancho y poco profundas para recoger de golpe un chaparrón entero de
los que caen pocas veces.
\end{example}

\begin{example}[Cojines de montaña]\label{ex:g4:adapted-to-their-world:alpine}
Las \omterm{def:g1:plants-around-us:plant}{plantas} de alta montaña se enfrentan a un viento helado. Muchas
crecen como cojines apretados pegados al suelo: dentro del cojín el
viento se rompe y el aire se queda más templado --- una forma que es
un refugio. Sus primas de tierras bajas crecen altas y sueltas; las
formas de montaña se abrazan a la roca.
\end{example}

\begin{method}[Leer en el cuerpo de un organismo el mundo en el que vive]\label{met:g4:adapted-to-their-world:read}
Ante un \omterm{def:g1:animals-around-us:animal}{animal} o una \omterm{def:g1:plants-around-us:plant}{planta} que no conoces:
\begin{enumerate}
  \item enumera sus rasgos llamativos --- pelaje, pies, \omterm{def:g1:the-five-senses:sense}{ojos}, forma del
        cuerpo, \omterm{def:g1:plants-around-us:parts}{hojas}, \omterm{def:g1:plants-around-us:parts}{raíces};
  \item pregunta de cada uno: \emph{¿qué problema resolvería esto?}
  \item junta las respuestas en un retrato de su casa --- fría o
        caliente, húmeda o seca, cazador o cazado;
  \item comprueba tu conjetura con el sitio donde vive de verdad el
        organismo.
\end{enumerate}
Este es el \cref{met:g2:where-animals-live:matching} hecho mayor: de
emparejar un \omterm{def:g2:where-animals-live:habitat}{hábitat} a reconstruir un mundo.
\end{method}

\begin{example}[El método sobre una piel misteriosa]\label{ex:g4:adapted-to-their-world:mystery}
Un cajón de museo guarda una \omterm{def:g1:the-five-senses:sense}{piel} sin etiqueta: \omterm{def:g1:animals-around-us:covering}{pelo} denso y suave,
\omterm{def:g1:the-five-senses:sense}{ojos} diminutos, enormes garras delanteras en forma de pala. El \omterm{def:g1:animals-around-us:covering}{pelo}
dice mamífero; las palas dicen cavar; los \omterm{def:g1:the-five-senses:sense}{ojos} inservibles dicen vida
en la oscuridad. Veredicto: un excavador subterráneo --- y en efecto
es un topo (\cref{ex:g2:where-animals-live:mole}), leído correctamente
a partir de tres rasgos nada más.
\end{example}

\begin{remark}[¿Cómo se llega a ese encaje?]\label{rem:g4:adapted-to-their-world:how}
Un camello no \emph{decide} nunca criar pies anchos, y ningún oso
polar elige su pelaje blanco. Entonces, ¿cómo llegan los \omterm{def:g1:living-or-not:living}{seres vivos} a
encajar tan bien en su mundo? Es una de las preguntas más profundas de
la \omterm{def:g1:living-or-not:living}{biología}, y su respuesta completa --- cómo se va construyendo el
encaje, generación tras generación --- es una historia que este libro
guarda para un curso posterior. Por ahora, aprende a \emph{ver} el
encaje: verlo es lo que hizo posible la pregunta.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:1}
¿Qué es una \omterm{def:g4:adapted-to-their-world:adaptation}{adaptación}? Da un ejemplo de un \omterm{def:g1:animals-around-us:animal}{animal} y un ejemplo de una
\omterm{def:g1:plants-around-us:plant}{planta}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:2}
Nombra tres problemas del desierto y la respuesta del camello a cada uno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:3}
Nombra tres problemas del hielo marino y la respuesta del oso polar a
cada uno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:4}
¿Qué tareas hacen las púas del cactus? ¿Qué almacena su \omterm{def:g1:plants-around-us:parts}{tallo} en forma
de barril y qué trabajo ha asumido ese \omterm{def:g1:plants-around-us:parts}{tallo}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:5}
¿Por qué crecen muchas \omterm{def:g1:plants-around-us:plant}{plantas} de alta montaña como cojines apretados?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:6}
Da dos adaptaciones de los \omterm{def:g1:animals-around-us:animal}{animales} cazados y dos de los cazadores,
sacadas de la \cref{prop:g4:adapted-to-their-world:answers}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:7}
Vuelve a contar el \cref{ex:g4:adapted-to-their-world:mystery}: ¿qué
tres rasgos delataron al topo y qué decía cada uno?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:8}
Pruébalo: elige cualquier \omterm{def:g1:animals-around-us:animal}{animal} que puedas observar --- un pato, un
gato, una araña --- y pásale el
\cref{met:g4:adapted-to-their-world:read}: tres rasgos, tres problemas
resueltos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:9}
El zorro del desierto tiene unas \omterm{def:g1:the-five-senses:sense}{orejas} enormes; el zorro ártico, unas
diminutas. Con la línea del frío de la
\cref{prop:g4:adapted-to-their-world:answers}, explica las dos cosas
--- ¿qué tienen que hacer unas \omterm{def:g1:the-five-senses:sense}{orejas} grandes con el calor?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:10}
Un \omterm{def:g1:animals-around-us:animal}{animal} misterioso: cuerpo afinado, patas traseras palmeadas, \omterm{def:g1:animals-around-us:covering}{pelo}
denso que repele el agua, bigotes, y aguanta bien la \omterm{def:g4:breathing:movements}{respiración}.
Reconstruye su mundo con el
\cref{met:g4:adapted-to-their-world:read}. (Construye presas.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:adapted-to-their-world:11}
La ballena (\cref{ex:g3:sorting-living-things:surprises}) es un
mamífero con forma de \omterm{prop:g3:sorting-living-things:groups}{pez}: afinada, con aletas, lisa. La trucha es un
\omterm{prop:g3:sorting-living-things:groups}{pez} con la misma forma. ¿Qué enseña este parecido entre dos nadadores
\emph{sin parentesco} sobre las adaptaciones --- quién impuso la
forma, la familia o el agua?
\end{exercise}
