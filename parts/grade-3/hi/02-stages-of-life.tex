\chapter{जीवन की अवस्थाएँ}\label{ch:g3:stages-of-life}

परिवार की एक तस्वीर सामने रखो: दादी की गोद में शिशु, आगे के दो दाँत
गँवा चुका स्कूल जाता बालक, अचानक अपनी माँ से लंबा हो गया \omterm{def:g3:stages-of-life:stages}{किशोर} भाई,
माता-पिता, और ख़ुद दादी। एक परिवार, एक दोपहर --- और पूरा मानव \omterm{def:g2:born-grow-die:cycle}{जीवन चक्र}
एक ही सोफ़े पर बैठा हुआ। इस साल हम इन अवस्थाओं को ग़ौर से देखेंगे, और
यह भी कि वृद्धि को नापा कैसे जाता है।

\section{मानव जीवन की अवस्थाएँ}

\begin{definition}[मानव जीवन की अवस्थाएँ]\label{def:g3:stages-of-life:stages}
मानव जीवन कई अवस्थाओं से गुज़रता है:
\begin{enumerate}
  \item \emph{शिशु}\index{शिशु} (मोटे तौर पर पहले दो साल): सबसे तेज़
        वृद्धि, पहले दाँत, पहले क़दम, पहले शब्द;
  \item \emph{बालक}\index{बालक}: लगातार वृद्धि, \omterm{prop:g2:teeth-and-eating-well:sets}{दूध के दाँतों} की जगह
        \omterm{prop:g2:teeth-and-eating-well:sets}{स्थायी दाँत}, और हर तरह की सीख;
  \item \emph{किशोर}\index{किशोर}: तेज़ वृद्धि का नया दौर, और शरीर का
        बालक के शरीर से वयस्क के शरीर में बदलना;
  \item \emph{वयस्क}\index{वयस्क}: क़द बढ़ना ख़त्म; वयस्क माता-पिता
        बन सकते हैं;
  \item \emph{बुढ़ापा}\index{बुढ़ापा}: बाल सफ़ेद होते हैं, \omterm{def:g1:the-five-senses:sense}{त्वचा} पर
        झुर्रियाँ पड़ती हैं, ताक़त धीरे-धीरे घटती है --- चक्र की
        स्वाभाविक आख़िरी अवस्था।
\end{enumerate}
\end{definition}

\begin{example}[सोफ़े को पढ़ना]\label{ex:g3:stages-of-life:sofa}
परिवार के सोफ़े पर: शिशु (अवस्था 1), दाँत गँवाया स्कूल जाता बालक
(अवस्था 2 --- \cref{ch:g2:teeth-and-eating-well} की अदला-बदली पूरे
ज़ोर पर), सोलह साल का चचेरा भाई (अवस्था 3, दौर के बीच में), माता-पिता
(अवस्था 4), दादी (अवस्था 5)। पाँच अवस्थाएँ, एक परिवार।
\end{example}

\begin{omfigure}
\includegraphics[width=0.62\linewidth]{images/book1/ai/g3-stages-family.jpg}

{\small पूरा चक्र एक ही सोफ़े पर: शिशु, बालक, \omterm{def:g3:stages-of-life:stages}{किशोर}, \omterm{def:g3:stages-of-life:stages}{वयस्क}, दादी।}
\end{omfigure}

\begin{remark}[वृद्धि एक-सी नहीं होती]\label{rem:g3:stages-of-life:bursts}
वृद्धि की एक लय होती है: शिशु के लिए बहुत तेज़ --- पहले ही साल में क़द
मोटे तौर पर आधा और बढ़ जाता है --- बचपन भर धीमी और लगातार, \omterm{def:g3:stages-of-life:stages}{किशोर} के लिए
फिर तेज़, और उसके बाद ख़त्म। बीच में एक शांत हिस्सा लिए दो तेज़ दौर:
बड़े होने की सामान्य शक्ल यही है।
\end{remark}

\section{वृद्धि नापना}

\begin{definition}[वृद्धि-आलेख]\label{def:g3:stages-of-life:chart}
\emph{वृद्धि-आलेख}\index{वृद्धि-आलेख} समय के साथ बार-बार नापे गए क़द
(या भार) का लेखा है --- \cref{ch:g1:my-body} वाले चौखट के निशान, पर
नियमित और तारीख़ के साथ। डॉक्टर हर बच्चे के लिए एक रखते हैं।
\end{definition}

लिए लगातार, \omterm{def:g3:stages-of-life:stages}{किशोर} के लिए फिर तीखा, और फिर सपाट --- वृद्धि ख़त्म।}
\end{omfigure}

\begin{example}[आलेख पढ़ना]\label{ex:g3:stages-of-life:reading}
आलेख में रेखा पहले ही साल में लगभग $\qty{25}{cm}$ चढ़ती है --- और बचपन
भर साल में सिर्फ़ लगभग $\qty{6}{cm}$। दोनों तीखे हिस्से
\cref{rem:g3:stages-of-life:bursts} वाले दो दौर हैं; सपाट सिरा कहता है
कि \omterm{def:g3:stages-of-life:stages}{वयस्क} का क़द बढ़ना रुक चुका है।
\end{example}

\begin{method}[अपना वृद्धि-आलेख रखना]\label{met:g3:stages-of-life:keep}
\begin{enumerate}
  \item क़द सावधानी से नापो (पिछले सालों वाला दीवार-और-किताब का
        तरीक़ा), हमेशा बिना जूतों के;
  \item तारीख़ और क़द किसी कॉपी में लिखो, या चौखट पर निशान लगाकर उस पर
        तारीख़ डालो;
  \item हर कुछ महीनों में फिर नापो --- नियमितता बार-बार नापने से
        ज़्यादा मायने रखती है;
  \item कहानी देखने के लिए बिंदु आरेखित करो: नीचे आयु, बग़ल में क़द,
        चित्र की तरह।
\end{enumerate}
\end{method}

\section{जंतुओं और पौधों की भी अवस्थाएँ होती हैं}

\begin{proposition}[जीवित दुनिया भर की अवस्थाएँ]\label{prop:g3:stages-of-life:animals}
हर \omterm{def:g1:living-or-not:living}{सजीव} अपने चक्र के अपने ही रूप की अवस्थाओं से गुज़रता है --- नन्हा,
\omterm{def:g3:stages-of-life:stages}{वयस्क}, \omterm{def:g3:stages-of-life:stages}{बुढ़ापा}। नन्ही अवस्था \omterm{def:g3:stages-of-life:stages}{वयस्क} जैसी दिख सकती है (बिल्ली का बच्चा,
बछेड़ा) या बिलकुल भी नहीं (\cref{ch:g2:animals-grow-up} का \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} और
\omterm{ex:g2:animals-grow-up:butterfly}{इल्ली}); \omterm{def:g1:plants-around-us:plant}{पौधे} की अवस्थाएँ चलती हैं \omterm{def:g2:from-seed-to-plant:seed}{बीज}, \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर}, नन्हा \omterm{def:g1:plants-around-us:plant}{पौधा}, फूलों वाला
\omterm{def:g3:stages-of-life:stages}{वयस्क}। इनमें से कोई भी न तो कोई अवस्था छोड़ सकता है, न उन्हें उलटा चला
सकता है।
\end{proposition}
\admitted

\begin{example}[कितना तेज़, कितना धीमा]\label{ex:g3:stages-of-life:speeds}
रफ़्तारें बहुत अलग-अलग हैं। मक्खी हफ़्तों में \omterm{def:g3:stages-of-life:stages}{वयस्क} होती है; बिल्ली
लगभग एक साल में; मनुष्य क़रीब बीस साल में; और बलूत को पहले फल तक
पहुँचने में कई दशक लगते हैं। धीमी हो या तेज़, अवस्थाओं का क्रम कभी नहीं
बदलता।
\end{example}

\begin{remark}[बुढ़ापा एक अवस्था है, कोई ख़राबी नहीं]\label{rem:g3:stages-of-life:oldage}
सफ़ेद बाल और झुर्रियाँ कोई बीमारी नहीं हैं: वे शरीर की आख़िरी अवस्था
हैं, उतनी ही स्वाभाविक जितने शिशु के पहले दाँत। बहुत-सी प्रजातियों में,
और मानव परिवारों में, बूढ़े वही लिए रहते हैं जिसकी नन्हों को सबसे
ज़्यादा ज़रूरत होती है --- अनुभव।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:1}
मानव जीवन की पाँच अवस्थाओं के नाम क्रम से बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:2}
किस अवस्था में: (क) \omterm{prop:g2:teeth-and-eating-well:sets}{दूध के दाँत} गिरते हैं? (ख) कोई व्यक्ति माता-पिता
बन सकता है? (ग) पूरे जीवन की सबसे तेज़ वृद्धि होती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:3}
वृद्धि के दोनों तेज़ दौर कौन-से हैं, और उनके बीच क्या है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:4}
\omterm{def:g3:stages-of-life:chart}{वृद्धि-आलेख} में सिर्फ़ रेखा देखकर तुम कैसे बता सकते हो कि \omterm{def:g3:stages-of-life:stages}{वयस्क} का
बढ़ना रुक चुका है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:5}
आलेख की मदद से बताओ कि $6$ और $8$ साल के बीच रेखा लगभग कितने सेंटीमीटर
चढ़ती है? और अकेले पहले साल में?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:6}
क़द हमेशा बिना जूतों के, हर बार एक ही तरीक़े से क्यों नापना चाहिए? ऐसा
न करने पर क्या ग़लत हो सकता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:7}
बलूत के \omterm{def:g1:plants-around-us:tree}{पेड़} के जीवन की अवस्थाएँ बताओ, फल से लेकर बूढ़े \omterm{def:g1:plants-around-us:tree}{पेड़} तक।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:8}
करके देखो: \cref{met:g3:stages-of-life:keep} वाला \omterm{def:g3:stages-of-life:chart}{वृद्धि-आलेख} अपने लिए
शुरू करो, आज के पहले बिंदु के साथ।
\cref{def:g3:stages-of-life:stages} की कौन-सी अवस्था में तुम हो --- और
कौन-सा दौर अभी आगे बाक़ी है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:9}
मक्खी, बिल्ली, मनुष्य और बलूत, चारों वही अवस्थाएँ बहुत अलग-अलग
रफ़्तारों से चलते हैं। वयस्कता तक पहुँचने में सबसे तेज़ से सबसे धीमे
के क्रम में लगाओ --- और बताओ कि चारों के अवस्था-क्रमों में क्या बात
एक जैसी है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:10}
$9$ साल का टॉम एक ही महीने में अपने को दो बार नापता है और पाता है कि
``बढ़ना बिलकुल नहीं हुआ''। क्या उसे चिंता करनी चाहिए?
\cref{rem:g3:stages-of-life:bursts} और उसकी उम्र पर आलेख की शक्ल से
उत्तर दो।
\end{exercise}
