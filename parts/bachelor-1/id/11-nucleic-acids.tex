\chapter{Nukleotida dan Asam Nukleat}\label{ch:b1:nucleic-acids}

Tuangkan etanol dingin ke atas larutan \omterm{def:b1:cell-unit-of-life:cell}{sel} yang telah dipecah dan
muncullah awan putih berupa serat yang dapat digulung pada sebatang
kaca: \omterm{def:b1:nucleic-acids:chain}{DNA}, molekul yang membawa tiap petunjuk \omterm{def:b1:organism-environment:organism}{organismenya}, tampak oleh
mata telanjang. Dua meter \omterm{def:b1:nucleic-acids:chain}{DNA} terlipat dalam tiap \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} \omterm{def:b1:cell-unit-of-life:cell}{sel} manusia;
seluruhnya tertulis dalam abjad empat huruf, dibaca pada satu arah, dan
disalin dengan memisahkan dua untai yang saling cocok seperti tangan dan
sarungnya. Bab ini memerikan \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} --- huruf, \omterm{def:b1:water-small-molecules:atp}{mata uang energi} dan
koenzim sekaligus --- rantai yang dibentuknya, \omterm{thm:b1:nucleic-acids:helix}{heliks ganda} beserta
buktinya, apa yang terjadi ketika heliks itu dilelehkan lalu dipulihkan,
dan \omterm{def:b1:nucleic-acids:chain}{RNA} yang membawa pesannya ke dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}.

\section{Nukleotida}

\begin{definition}[Nukleotida]\label{def:b1:nucleic-acids:nucleotide}
Yang disebut \emph{nukleotida}\index{nukleotida} tersusun dari tiga
bagian: sebuah \emph{basa}\index{basa nitrogen} bernitrogen --- sebuah
\emph{purin}\index{purin} (adenina A, guanina G: dua cincin bergabung)
atau sebuah \emph{pirimidin}\index{pirimidin} (sitosina C, timina T,
urasil U: satu cincin); sebuah gula berkarbon lima, \emph{ribosa} dalam
\omterm{def:b1:nucleic-acids:chain}{RNA} atau \emph{2$'$-deoksiribosa} dalam \omterm{def:b1:nucleic-acids:chain}{DNA} (yang kehilangan hidroksil
pada karbon $2'$); dan satu sampai tiga gugus \emph{fosfat} pada karbon
$5'$ gulanya. Basanya terpaut pada karbon $1'$; karbon gulanya
dinomori dengan aksen agar berbeda dari karbon basanya. Basa ditambah
gula adalah sebuah \emph{nukleosida} (adenosina, guanosina, sitidina,
timidina, uridina); dengan fosfat ia menjadi sebuah nukleotida (AMP,
ADP, \omterm{def:b1:water-small-molecules:atp}{ATP} dan kerabatnya).
\end{definition}

\begin{proposition}[Nukleotida mengerjakan tiga tugas]\label{prop:b1:nucleic-acids:threejobs}
Selain menjadi monomer \omterm{def:b1:nucleic-acids:chain}{DNA} dan \omterm{def:b1:nucleic-acids:chain}{RNA}, \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} adalah \emph{\omterm{def:b1:water-small-molecules:atp}{mata uang
energi}} \omterm{def:b1:cell-unit-of-life:cell}{sel} --- \omterm{def:b1:water-small-molecules:atp}{ATP} dan GTP, yang \omterm{def:b1:water-small-molecules:atp}{ikatan fosfoanhidridanya}
(\cref{ch:b1:water-small-molecules}) membayar sintesis, pengangkutan dan
gerak; \emph{koenzim}nya --- $\mathrm{NAD^+}$, $\mathrm{NADP^+}$ dan
FAD, yang membawa elektron dalam respirasi dan fotosintesis
(\cref{ch:b1:respiration-fermentation}), serta koenzim A, yang membawa
gugus asil; dan \emph{pembawa pesan}nya --- AMP siklik, yang meneruskan
isyarat hormon di dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}. Semuanya adalah \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} adenina atau
dibangun atasnya: molekul yang sama yang mengeja pesannya juga menenagai
pelaksanaannya.
\end{proposition}

\section{Rantai polinukleotida}

\begin{definition}[Ikatan fosfodiester, kekutuban]\label{def:b1:nucleic-acids:chain}
\omterm{def:b1:nucleic-acids:nucleotide}{Nukleotida} disambung menjadi sebuah
\emph{polinukleotida}\index{polinukleotida} lewat \emph{ikatan
fosfodiester}\index{ikatan fosfodiester}: fosfat pada karbon $5'$
sebuah \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} bertaut pada hidroksil $3'$ \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} berikutnya.
Karena itu rantainya memiliki \emph{tulang punggung} gula--fosfat, yang
seragam dan bermuatan negatif (satu muatan tiap fosfat), yang \omterm{def:b1:nucleic-acids:nucleotide}{basanya}
menjulur sebagai sebuah urutan; dan ia berarah: sebuah \emph{ujung
$5'$}\index{5' end@ujung $5'$} yang membawa fosfat bebas dan sebuah
\emph{ujung $3'$}\index{3' end@ujung $3'$} yang membawa hidroksil bebas.
Urutannya ditulis dan dibaca $5' \to 3'$, yaitu arah rantainya
disintesis. \emph{DNA}\index{DNA} (asam deoksiribonukleat) memakai
deoksiribosa dan \omterm{def:b1:nucleic-acids:nucleotide}{basa} A, G, C, T; \emph{RNA}\index{RNA} (asam
ribonukleat) memakai ribosa dan A, G, C, U.
\end{definition}

\begin{example}[RNA rapuh, DNA awet]\label{ex:b1:nucleic-acids:fragile}
Hidroksil $2'$ ribosa dapat menyerang \omterm{def:b1:nucleic-acids:chain}{ikatan fosfodiester} di sebelahnya:
dalam \omterm{def:b1:nucleic-acids:nucleotide}{basa} lemah \omterm{def:b1:nucleic-acids:chain}{RNA} terhidrolisis menjadi \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} dalam beberapa
jam, sedangkan \omterm{def:b1:nucleic-acids:chain}{DNA}, yang tidak memiliki hidroksil itu, selamat dari
pendidihan dalam \omterm{def:b1:nucleic-acids:nucleotide}{basa} dan telah diperoleh kembali secara utuh dari
tulang berumur puluhan ribu tahun. Urasil dalam \omterm{def:b1:nucleic-acids:chain}{RNA} adalah timina tanpa
gugus metil; \omterm{def:b1:nucleic-acids:chain}{DNA} memakai timina agar sitosina, yang perlahan kehilangan
gugus aminonya menjadi urasil, dapat dikenali sebagai rusak lalu
diperbaiki --- sebuah urasil dalam \omterm{def:b1:nucleic-acids:chain}{DNA} selalu sebuah kekeliruan.
\end{example}

\section{Heliks ganda}

\begin{proposition}[Kaidah Chargaff]\label{prop:b1:nucleic-acids:chargaff}
\index{kaidah Chargaff}
Dalam \omterm{def:b1:nucleic-acids:chain}{DNA} \omterm{def:b1:organism-environment:organism}{organisme} mana pun jumlah adenina sama dengan jumlah timina
dan jumlah guanina sama dengan jumlah sitosina ($A = T$, $G = C$,
sehingga \omterm{def:b1:nucleic-acids:nucleotide}{purin} $=$ \omterm{def:b1:nucleic-acids:nucleotide}{pirimidin}), sedangkan rasio $(G + C)/(A + T)$
beragam dari spesies ke spesies (dari \qty{25}{\%} sampai \qty{75}{\%}
$G + C$).
\end{proposition}
\begin{proof}[Bukti eksperimental]
Chargaff (1950) menghidrolisis \omterm{def:b1:nucleic-acids:chain}{DNA} dari banyak \omterm{def:b1:organism-environment:organism}{organisme} dan memisahkan
\omterm{def:b1:nucleic-acids:nucleotide}{basanya} lewat kromatografi, lalu mengukur masing-masing lewat serapan
ultraungunya. Kesamaannya berlaku dalam batas galat percobaan bagi tiap
\omterm{def:b1:nucleic-acids:chain}{DNA}; kadar $G + C$-nya tidak, dan sama pada tiap \omterm{def:b1:body-plans-tissues:tissue}{jaringan} satu
\omterm{def:b1:organism-environment:organism}{organisme}.
\end{proof}

\begin{theorem}[Heliks ganda Watson--Crick]\label{thm:b1:nucleic-acids:helix}
\index{heliks ganda}\index{pasangan basa}\index{antiparalel}
\omterm{def:b1:nucleic-acids:chain}{DNA} adalah dua rantai \omterm{def:b1:nucleic-acids:chain}{polinukleotida} yang terpilin mengelilingi sebuah
sumbu bersama menjadi \emph{\omterm{thm:b1:nucleic-acids:helix}{heliks ganda}} tangan kanan: tulang punggung
gula--fosfatnya di luar, \omterm{def:b1:nucleic-acids:nucleotide}{basanya} di dalam, pipih dan bertumpuk tegak
lurus terhadap sumbunya. Kedua rantainya bersifat \emph{antiparalel}
(yang satu berjalan $5' \to 3'$ ke atas, yang lain ke bawah) dan
\emph{saling melengkapi}: berhadapan dengan tiap A ada sebuah T, yang
dipegang dua \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen}, dan berhadapan dengan tiap G sebuah C,
yang dipegang tiga; tiap \emph{\omterm{thm:b1:nucleic-acids:helix}{pasangan basa}} semacam itu sama lebarnya,
sehingga heliksnya seragam apa pun urutannya. Dalam bentuk B yang lazim
heliksnya selebar \qty{2}{nm}, naik \qty{0.34}{nm} tiap \omterm{thm:b1:nucleic-acids:helix}{pasangan basa}
dan \qty{3.4}{nm} tiap putaran berisi sepuluh pasangan, serta
memperlihatkan sebuah \emph{lekuk besar} yang lebar dan sebuah
\emph{lekuk kecil} yang sempit tempat tepi \omterm{def:b1:nucleic-acids:nucleotide}{basanya} terpapar pada
protein.
\end{theorem}
\begin{proof}[Bukti eksperimental]
Gambar difraksi sinar-X serat \omterm{def:b1:nucleic-acids:chain}{DNA} karya Franklin (1952) memperlihatkan
tanda silang sebuah heliks dengan pengulangan \qty{3.4}{nm} sepanjang
sumbunya, sebuah pantulan kuat pada \qty{0.34}{nm} (satuan pipih yang
bertumpuk), dan garis tengah \qty{2}{nm}; fosfatnya, sebagai atom
terberat, harus berada di luar. Watson dan Crick (1953) mendapati bahwa
sebuah \omterm{def:b1:nucleic-acids:nucleotide}{purin} yang berpasangan dengan sebuah \omterm{def:b1:nucleic-acids:nucleotide}{pirimidin} --- dan hanya A
dengan T serta G dengan C --- memberikan pasangan yang sama lebarnya
yang cocok dengan garis tengah \qty{2}{nm}, sehingga menerangkan \omterm{prop:b1:nucleic-acids:chargaff}{kaidah
Chargaff}, dan bahwa dua rantai antiparalel yang berpasangan demikian
membangun heliks teratur berukuran seperti seratnya. Modelnya
meramalkan bahwa tiap rantai menjadi cetakan bagi rantai yang lain;
peragaan replikasi semikonservatif oleh Meselson dan Stahl (yang
diingat kembali dari jilid Sekolah Menengah, dan pada
\cref{ch:b1:replication-mitosis}) membenarkannya.
\end{proof}

\begin{example}[Membaca sebuah untai]\label{ex:b1:nucleic-acids:complement}
Bila salah satu untainya berbunyi $5'$-ATGGCATTC-$3'$, pasangannya,
yang ditulis $5' \to 3'$, adalah $5'$-GAATGCCAT-$3'$: lengkapi tiap
\omterm{def:b1:nucleic-acids:nucleotide}{basanya}, lalu balikkan urutannya. Sembilan pasangan, \qty{3.1}{nm}
heliks, dua puluh dua \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen} (lima G--C, empat A--T). Sebuah
kromosom manusia berisi \num{2.5e8} pasangan sepanjang \qty{8.5}{cm}
dalam bentuk ini dan harus dilipat seratus ribu kali agar muat dalam
sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} (\cref{ch:b1:genomes}).
\end{example}

\begin{proposition}[Apa yang memegang heliksnya]\label{prop:b1:nucleic-acids:stability}
\omterm{def:b1:water-small-molecules:water}{Ikatan hidrogen} antara \omterm{def:b1:nucleic-acids:nucleotide}{basa} yang berpasangan memberikan kekhasannya;
\emph{penumpukan}\index{penumpukan basa} \omterm{def:b1:nucleic-acids:nucleotide}{basa} yang pipih satu di atas
yang lain --- sentuhan \omterm{def:b1:water-small-molecules:bonds}{van der Waals} dan \omterm{prop:b1:water-small-molecules:properties}{efek hidrofob} yang menjauhkan
\omterm{def:b1:nucleic-acids:nucleotide}{basanya} dari air --- memberikan sebagian besar kemantapannya; tulang
punggungnya yang bermuatan negatif saling menolak, dan heliksnya hanya
mantap dengan hadirnya ion positif ($\mathrm{Mg^{2+}}$,
$\mathrm{Na^+}$ yang fisiologis, atau protein \omterm{def:b1:nucleic-acids:nucleotide}{basa} kromatinnya) yang
menyekat muatan itu. Sebuah pasangan G--C, dengan tiga \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen}
dan penumpukan yang lebih kuat, memegang lebih baik daripada sebuah
pasangan A--T.
\end{proposition}

\section{Peleburan dan hibridisasi}

\begin{definition}[Denaturasi, suhu leleh]\label{def:b1:nucleic-acids:melting}
Pemanasan, atau pH yang ekstrem, memisahkan kedua untai \omterm{def:b1:nucleic-acids:chain}{DNA}:
\emph{denaturasi}\index{denaturasi DNA} (peleburan), sebuah transisi
bekerja sama sepanjang beberapa derajat, yang titik tengahnya adalah
\emph{suhu leleh}\index{suhu leleh} $T_m$. Ia diikuti \emph{efek
hiperkromik}: untai tunggal menyerap sekitar \qty{40}{\%} lebih banyak
cahaya ultraungu pada \qty{260}{nm} daripada \omterm{def:b1:nucleic-acids:nucleotide}{basa} yang sama yang
bertumpuk dalam sebuah heliks. $T_m$ naik bersama kadar $G + C$
(\qty{0.4}{\celsius} tiap persen), bersama kepekatan garamnya, dan
bersama panjangnya; dalam garam \qty{0.15}{mol/L} sebuah \omterm{def:b1:nucleic-acids:chain}{DNA} panjang
berkadar $G + C$ \qty{50}{\%} meleleh dekat \qty{90}{\celsius}. Bila
didinginkan perlahan, untainya menemukan pasangannya lalu membentuk
kembali heliksnya: \emph{renaturasi}, atau
\emph{hibridisasi}\index{hibridisasi} bila untainya berasal dari sumber
yang berbeda.
\end{definition}

\begin{method}[Mengukur asam nukleat lewat serapan]\label{met:b1:nucleic-acids:absorbance}
\begin{enumerate}
  \item \omterm{def:b1:nucleic-acids:nucleotide}{Basa} menyerap cahaya ultraungu dengan puncak pada
        \qty{260}{nm}. Dalam sebuah kuvet \qty{1}{cm}, serapan $1.0$
        sepadan dengan \qty{50}{\micro g/mL} \omterm{def:b1:nucleic-acids:chain}{DNA} untai ganda
        (\qty{40}{\micro g/mL} \omterm{def:b1:nucleic-acids:chain}{RNA}, \qty{33}{\micro g/mL} \omterm{def:b1:nucleic-acids:chain}{DNA} untai
        tunggal).
  \item Rasio $A_{260}/A_{280}$ adalah $1.8$ bagi \omterm{def:b1:nucleic-acids:chain}{DNA} murni dan $2.0$
        bagi \omterm{def:b1:nucleic-acids:chain}{RNA} murni; protein menyerap pada \qty{280}{nm} dan
        menurunkannya.
  \item Untuk mengikuti peleburannya, rekamlah $A_{260}$ sambil
        memanaskan perlahan; titik tengah kurvanya adalah $T_m$ dan
        curamnya adalah ukuran keseragaman cuplikannya.
  \item Untuk mendeteksi sebuah urutan, denaturasikan cuplikannya lalu
        tambahkan sebuah untai tunggal bertanda yang melengkapinya
        (sebuah \emph{pelacak}); ia berhibridisasi hanya di tempat
        pasangannya berada, dan tandanya menandai tempat itu --- pada
        sebuah gel, pada sebuah kromosom, di dalam sebuah \omterm{def:b1:body-plans-tissues:tissue}{jaringan}.
\end{enumerate}
\end{method}

\begin{example}[Sebuah pelacak menemukan sebuah gen]\label{ex:b1:nucleic-acids:probe}
Sebuah pelacak dua puluh \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} yang melengkapi satu gen, pada
\qty{5}{\celsius} di bawah $T_m$-nya sendiri, mengikat pasangan
persisnya dan tak ada yang lain dalam tiga miliar \omterm{thm:b1:nucleic-acids:helix}{pasangan basa}: satu
ketakcocokan dalam dua puluh menurunkan $T_m$ beberapa derajat dan
hibrid yang tak cocok itu meleleh. Kekhasan pemasangan \omterm{def:b1:nucleic-acids:nucleotide}{basa}, yang
dilipatgandakan atas dua puluh kedudukan, adalah sandaran tiap uji \omterm{def:b1:nucleic-acids:chain}{DNA},
dari uji kebapakan sampai pendeteksian patogen.
\end{example}

\section{RNA}

\begin{definition}[Berbagai RNA]\label{def:b1:nucleic-acids:rnas}
\omterm{def:b1:nucleic-acids:chain}{RNA} biasanya berutas tunggal, dan melipat balik pada dirinya sendiri di
mana pun ada bentangan pendek yang saling melengkapi, menjadi jepit
rambut, gelung dan tonjolan --- sebuah \emph{struktur sekunder} --- yang
lalu berkemas menjadi sebuah bentuk. Tiga golongan membawa aliran
informasi genetik (\cref{ch:b1:gene-expression}): \emph{RNA pembawa
pesan}\index{RNA pembawa pesan} (mRNA), salinan sebuah gen yang dibaca
ribosom; \emph{RNA pemindah}\index{RNA pemindah} (tRNA), sekitar
\qty{75}{} \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} yang terlipat menjadi \omterm{def:b1:flowering-plant-organization:organs}{daun} semanggi, yang membawa
sebuah asam amino pada ujung $3'$-nya dan sebuah \emph{antikodon} yang
berpasangan dengan pesannya; \emph{RNA ribosom}\index{RNA ribosom}
(rRNA), yaitu teras struktural sekaligus katalitik ribosomnya, empat
perlima \omterm{def:b1:nucleic-acids:chain}{RNA} sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}. Banyak \omterm{def:b1:nucleic-acids:chain}{RNA} lain mengatur, menyambung dan
menuntun (jilid Tahun ke-3).
\end{definition}

\begin{remark}[RNA dapat menjadi katalis]\label{rem:b1:nucleic-acids:ribozyme}
Sebagian \omterm{def:b1:nucleic-acids:chain}{RNA} adalah enzim (\emph{ribozim}): ikatan peptida tiap protein
dibentuk \omterm{def:b1:nucleic-acids:rnas}{RNA ribosom}, bukan sebuah protein, dan sebagian \omterm{def:b1:nucleic-acids:chain}{RNA} memotong
serta menyambung dirinya sendiri. Sebuah molekul yang sekaligus membawa
informasi dan mengatalisis reaksi adalah bahan yang mungkin menyusun
sistem hidup pertama; dunia DNA--protein lalu menjadi pembagian kerja
yang belakangan, dengan \omterm{def:b1:nucleic-acids:chain}{RNA} tetap di pusatnya.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:1}
Sebutkan ketiga bagian sebuah \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} dan kedua perbedaan antara
sebuah ribonukleotida dan sebuah deoksiribonukleotida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:2}
Tuliskan untai pelengkap $5'$-GGATCCTA-$3'$ dalam arah $5' \to 3'$, dan
hitunglah \omterm{def:b1:water-small-molecules:water}{ikatan hidrogennya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:3}
Sebuah \omterm{def:b1:nucleic-acids:chain}{DNA} memuat \qty{22}{\%} adenina. Berikan persentase T, G dan C,
serta kadar $G + C$-nya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:4}
Dari gambar peleburannya, berapa $T_m$ sebuah \omterm{def:b1:nucleic-acids:chain}{DNA} berkadar $G + C$
\qty{50}{\%} pada keadaan yang sama, dan seberapa besar serapannya naik
saat meleleh?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:5}
Hitunglah panjang kromosom \emph{E. coli} ($\num{4.6e6}$ \omterm{thm:b1:nucleic-acids:helix}{pasangan basa})
dalam milimeter, banyaknya putarannya, dan faktor pemampatan yang
dituntut agar ia muat dalam sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} \qty{2}{\micro m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:6}
Jelaskan mengapa kedua untai \omterm{def:b1:nucleic-acids:chain}{DNA} harus antiparalel, dengan memakai
geometri tulang punggung gula--fosfatnya dan tuntutan bahwa tiap
\omterm{thm:b1:nucleic-acids:helix}{pasangan basa} harus sama lebarnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:7}
Sebuah larutan \omterm{def:b1:nucleic-acids:chain}{DNA} memiliki $A_{260} = 0.35$ dalam kuvet \qty{1}{cm}
dan $A_{280} = 0.20$. Hitunglah kepekatannya lalu beri tanggapan atas
kemurniannya. Berapa banyak \omterm{thm:b1:nucleic-acids:helix}{pasangan basa} dalam \qty{1}{mL}, bila massa
rata-rata sebuah pasangan adalah
\qty{650}{Da}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:8}
Jelaskan efek hiperkromiknya dan mengapa kurva peleburan sebuah \omterm{def:b1:nucleic-acids:chain}{DNA}
murni curam sedangkan kurva sebuah campuran \omterm{def:b1:nucleic-acids:chain}{DNA} dari banyak spesies
melebar.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:9}
Sebuah pelacak dua puluh \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} memiliki $T_m = \qty{62}{\celsius}$
dengan pasangan persisnya; satu ketakcocokan menurunkan $T_m$ sekitar
\qty{5}{\celsius}. Pada suhu berapa \omterm{def:b1:nucleic-acids:melting}{hibridisasinya} sebaiknya dilakukan
agar urutan persisnya terdeteksi dan ketakcocokan tunggalnya ditolak?
Apa yang terjadi pada
\qty{45}{\celsius}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:10}
\omterm{prop:b1:nucleic-acids:chargaff}{Kaidah Chargaff} berlaku bagi \omterm{def:b1:nucleic-acids:chain}{DNA} untai ganda tetapi tidak bagi \omterm{def:b1:nucleic-acids:chain}{RNA}
maupun bagi \omterm{def:b1:nucleic-acids:chain}{DNA} virus kecil tertentu. Jelaskan apa yang dikatakan tiap
kekecualian itu tentang struktur asam nukleat tersebut.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:11}
Dua \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen} memegang sebuah pasangan A--T, namun sebuah heliks
berisi seribu pasangan A--T mantap pada \qty{60}{\celsius} sedangkan
sepasang tunggal sama sekali tidak terbentuk. Jelaskan dengan gagasan
\cref{ch:b1:water-small-molecules} (ikatan lemah, kerja sama,
penumpukan), lalu sebutkan apa siratannya bagi kesetiaan sebuah
pelacak.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:12}
``Struktur \omterm{def:b1:nucleic-acids:chain}{DNA} adalah penjelasan pewarisan sifat.'' Bahaslah dalam satu
paragraf: apa yang langsung dijelaskan \omterm{thm:b1:nucleic-acids:helix}{heliks ganda} (penyalinan, mutasi
sebagai perubahan urutan), apa yang tidak dijelaskannya sendirian
(bagaimana urutannya menjadi sebuah protein), dan mengapa model Watson
dan Crick diterima sebelum ada uji biokimia apa pun atasnya.
\end{exercise}

\section{Soal: Genom Sebuah Virus}

\begin{problem}[{Soal akhir pekan --- DNA bakteriofag $\lambda$ yang
diukur, ditimbang, dihitung, dilelehkan dan dikemas ke dalam kapsidnya,
berakhir pada suhu lelehnya dan panjang konturnya}]
\label{pb:b1:nucleic-acids:1}
Genom bakteriofag $\lambda$ adalah satu \omterm{def:b1:nucleic-acids:chain}{DNA} untai ganda yang lurus
berisi \num{48502} \omterm{thm:b1:nucleic-acids:helix}{pasangan basa} dengan kadar $G + C$ \qty{49.9}{\%}.
Ia dikemas ke dalam sebuah kapsid ikosahedral bergaris tengah dalam
\qty{55}{nm}. Ambil \qty{0.34}{nm} tiap \omterm{thm:b1:nucleic-acids:helix}{pasangan basa}, \qty{2.0}{nm}
bagi garis tengah heliksnya, \qty{650}{Da} tiap \omterm{thm:b1:nucleic-acids:helix}{pasangan basa}, dan
$T_m = 81.5 + 16.6\log_{10}[\mathrm{Na^+}]
+ 0.41\,(\%\,G{+}C) - 500/L$ dalam derajat Celsius, dengan
$[\mathrm{Na^+}]$
dalam \unit{mol/L} dan $L$ panjangnya dalam \omterm{thm:b1:nucleic-acids:helix}{pasangan basa}.

\medskip
\noindent\textbf{Bagian I --- Diukur dan ditimbang.}
\begin{enumerate}
  \item Hitunglah panjang kontur molekulnya dalam mikrometer.
  \item Hitunglah banyaknya putaran heliksnya.
  \item Hitunglah massa molarnya dalam dalton.
  \item Hitunglah massa satu molekulnya dalam gram.
  \item Hitunglah banyaknya gugus fosfatnya dan dengan itu muatan bersih
        molekulnya.
  \item Hitunglah banyaknya \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} tiap \omterm{def:b1:nucleic-acids:nucleotide}{basanya} (A, T, G, C).
  \item Hitunglah total banyaknya \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen} yang memegang kedua
        untainya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Dilelehkan.}
\begin{enumerate}[resume]
  \item Hitunglah $T_m$ dalam natrium \qty{0.1}{mol/L}.
  \item Hitunglah $T_m$ dalam natrium \qty{0.01}{mol/L}, lalu jelaskan
        mengapa garam memantapkan heliksnya.
  \item Genom $\lambda$ memuat sebuah kawasan berisi \qty{1000}{}
        pasangan berkadar $G + C$ \qty{35}{\%} dan satu berkadar
        \qty{62}{\%}. Hitunglah $T_m$ tiap kawasan itu dalam natrium
        \qty{0.1}{mol/L} lalu perikan bentuk kurva peleburan seluruh
        molekulnya.
  \item Sebuah larutan \omterm{def:b1:nucleic-acids:chain}{DNA} $\lambda$ memiliki $A_{260} = 0.50$ sebelum
        peleburannya. Berapa kepekatannya, dan berapa $A_{260}$-nya
        sesudah peleburannya lengkap?
  \item Berapa banyak molekul $\lambda$ dalam \qty{1}{mL} larutan itu?
  \item Kedua ujung genom $\lambda$ adalah juluran untai tunggal berisi
        12 \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} yang saling melengkapi, yang membuat molekulnya
        melingkar di dalam inangnya. Hitunglah $T_m$-nya (pakailah
        $L = 12$) lalu sebutkan mengapa lingkarannya tetap bertahan pada
        \qty{37}{\celsius} di dalam \omterm{def:b1:cell-unit-of-life:cell}{selnya}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Dikemas.}
\begin{enumerate}[resume]
  \item Hitunglah volume \omterm{def:b1:nucleic-acids:chain}{DNA-nya} sebagai sebuah tabung.
  \item Hitunglah volume dalam kapsidnya.
  \item Berapa fraksi kapsidnya yang diisi \omterm{def:b1:nucleic-acids:chain}{DNA} itu? Bandingkan dengan
        kemasan terapat tabung sejajar (\qty{91}{\%}).
  \item \omterm{def:b1:nucleic-acids:chain}{DNA} itu membawa \num{97000} muatan negatif dalam sebuah bola
        \qty{55}{nm}. Jelaskan apa yang harus menyertai \omterm{def:b1:nucleic-acids:chain}{DNA} itu masuk ke
        dalam kapsidnya, dan mengapa \omterm{def:b1:nucleic-acids:chain}{DNA} yang terkemas mengerahkan
        tekanan pada cangkangnya (puluhan atmosfer) yang membantu
        menyuntikkannya ke dalam sebuah bakteri.
  \item Molekulnya harus dibengkokkan menjadi gelung berjari-jari
        sebanding dengan kapsidnya. \omterm{def:b1:nucleic-acids:chain}{DNA} menahan pembengkokan di bawah
        jari-jari sekitar \qty{50}{nm}. Beri tanggapan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Dibaca.}
\begin{enumerate}[resume]
  \item Satu untai genomnya diawali $5'$-GGGCGGCGACCT-$3'$. Tuliskan
        untai pelengkapnya $5' \to 3'$.
  \item Bila sebuah gen berisi \num{1200} \omterm{thm:b1:nucleic-acids:helix}{pasangan basa} ditranskripsikan
        menjadi sebuah mRNA, berapa panjang mRNA itu dalam \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}
        dan, pada \qty{330}{Da} tiap \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}, berapa massanya?
  \item Genomnya memiliki \qty{75}{} gen dalam \num{48502} pasangan.
        Berapa fraksi genomnya yang menyandi bila gen rata-ratanya
        \num{600} pasangan? Bandingkan dengan genom manusia
        (\qty{1.5}{\%}).
  \item Genomnya direplikasi dalam \qty{20}{min} pada
        \qty{37}{\celsius} oleh enzim inangnya. Hitunglah lajunya dalam
        \omterm{thm:b1:nucleic-acids:helix}{pasangan basa} tiap sekon bila replikasinya mulai pada satu titik
        awal lalu berjalan ke dua arah.
  \item Berapa banyak \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen} yang diputus tiap sekon pada
        kedua garpunya selama replikasi (pakailah rata-rata tiap
        pasangan dari pertanyaan 7)?
  \item Sebuah mutasi mengubah satu pasangan G--C menjadi A--T.
        Seberapa besar $T_m$ seluruh molekulnya berubah? Dapatkah
        peleburan mendeteksinya?
  \item Nyatakan hasilnya: \omterm{def:b1:nucleic-acids:melting}{suhu leleh} \omterm{def:b1:nucleic-acids:chain}{DNA} $\lambda$ dalam natrium
        \qty{0.1}{mol/L} dan panjang konturnya, serta pemampatan yang
        memuatkannya ke dalam kapsidnya.
\end{enumerate}
\end{problem}
