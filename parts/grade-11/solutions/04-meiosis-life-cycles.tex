\section*{Chapter \ref{ch:g11:meiosis-life-cycles} --- Meiosis and Life Cycles}

\begin{solution}{exo:g11:meiosis-life-cycles:1}
\omterm{def:g11:meiosis-life-cycles:meiosis}{Meiosis} halves the \omterm{def:g11:dna-replication:chromosome}{chromosome} number in gametes so that fusion of two gametes
restores the \omterm{def:g11:meiosis-life-cycles:ploidy}{diploid} number instead of doubling it every generation. Without
\omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}, fertilisation would ratchet ploidy upward indefinitely.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:2}
\omterm{def:g11:meiosis-life-cycles:meiosis}{Meiosis} involves one round of \omterm{def:g11:dna-replication:semi}{DNA replication} followed by two nuclear
divisions (\omterm{def:g11:meiosis-life-cycles:MI}{meiosis I} and \omterm{def:g11:meiosis-life-cycles:MII}{meiosis II}). There is no intervening \omterm{def:g11:cell-cycle-mitosis:interphase}{S phase} between
the two divisions.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:3}
At anaphase I, homologous \omterm{def:g11:dna-replication:chromosome}{chromosomes} separate to opposite poles (\omterm{def:g11:cell-cycle-mitosis:chromatid}{sister
chromatids} stay together). At anaphase II, \omterm{def:g11:cell-cycle-mitosis:chromatid}{sister chromatids} separate ---
similar to mitotic anaphase --- producing \omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} nuclei with unreplicated
\omterm{def:g11:dna-replication:chromosome}{chromosomes}.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:4}
\omterm{prop:g11:meiosis-life-cycles:prophaseI}{Crossing over} is reciprocal exchange of corresponding \omterm{def:g11:dna-replication:helix}{DNA} segments between
homologous \omterm{def:g11:dna-replication:chromosome}{chromosomes} during \omterm{prop:g11:meiosis-life-cycles:prophaseI}{prophase I}. It occurs between non-sister
chromatids of a \omterm{prop:g11:meiosis-life-cycles:prophaseI}{bivalent}, creating recombinant chromatids that mosaic maternal
and paternal alleles.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:5}
If $2n = 8$, then $n = 4$: each meiotic product has $4$ \omterm{def:g11:dna-replication:chromosome}{chromosomes}. Just after
\omterm{def:g11:meiosis-life-cycles:MI}{meiosis I}, each \omterm{def:g11:dna-replication:chromosome}{chromosome} is still a dyad of $2$ chromatids until \omterm{def:g11:meiosis-life-cycles:MII}{meiosis II}
separates them.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:6}
Any three valid contrasts: one nuclear division versus two; \omterm{def:g11:meiosis-life-cycles:ploidy}{diploid} versus
\omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} products; two versus four daughters; homologous pairing only in
\omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}; routine \omterm{prop:g11:meiosis-life-cycles:prophaseI}{crossing over} only in \omterm{def:g11:meiosis-life-cycles:MI}{meiosis I}; \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis} for \omterm{def:g10:scientific-biology:properties}{growth}/repair
versus \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis} for gametes or spores.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:7}
Independent assortment alone yields $2^{n}$ combinations. For $n = 3$,
$2^{3} = 8$ different gamete \omterm{def:g11:dna-replication:chromosome}{chromosome} combinations (ignoring \omterm{prop:g11:meiosis-life-cycles:prophaseI}{crossing over}).
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:8}
The \omterm{def:g11:meiosis-life-cycles:alt}{sporophyte} generation is \omterm{def:g11:meiosis-life-cycles:ploidy}{diploid}. Spores are produced by \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}; they
grow into the \omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} \omterm{def:g11:meiosis-life-cycles:alt}{gametophyte}, which makes gametes by \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:9}
Independent assortment randomly orients whole maternal and paternal
\omterm{def:g11:dna-replication:chromosome}{chromosomes} at metaphase I, producing different combinations of intact
\omterm{def:g11:dna-replication:chromosome}{chromosomes} in gametes. \omterm{prop:g11:meiosis-life-cycles:prophaseI}{Crossing over} physically exchanges segments so that
individual chromatids are not purely maternal or paternal --- new allele
combinations along a \omterm{def:g11:dna-replication:chromosome}{chromosome}.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:10}
(1)~\omterm{def:g11:meiosis-life-cycles:MI}{Meiosis I} separates homologues after pairing as \omterm{prop:g11:meiosis-life-cycles:prophaseI}{bivalents}; \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis} never
routinely pairs homologues that way. (2)~One \omterm{def:g11:dna-replication:semi}{replication} and two divisions
produce four \omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} nuclei; two successive mitoses would keep diploidy and
would include \omterm{def:g11:dna-replication:semi}{DNA replication} before each \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}. Also, there is no \omterm{def:g11:cell-cycle-mitosis:interphase}{S phase}
between \omterm{def:g11:meiosis-life-cycles:MI}{meiosis I} and II.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:11}
\omterm{def:g11:meiosis-life-cycles:gametogenesis}{Spermatogenesis} typically yields four functional sperm of similar small size.
\omterm{def:g11:meiosis-life-cycles:gametogenesis}{Oogenesis} yields one large egg and \omterm{def:g11:meiosis-life-cycles:gametogenesis}{polar bodies} that discard extra nuclei,
conserving cytoplasm in the ovum. In humans, sperm production is continuous
after puberty, whereas primary oocytes begin \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis} before birth and arrest
for years before maturation.
\end{solution}

\begin{solution}{exo:g11:meiosis-life-cycles:12}
If both homologues of one pair enter the same daughter at \omterm{def:g11:meiosis-life-cycles:MI}{meiosis I}, one
gamete is $n+1$ and another $n-1$. Fertilisation of an $n+1$ gamete by a
normal $n$ gamete produces a $2n+1$ (trisomic) zygote. \omterm{prop:g11:cell-cycle-mitosis:stages}{Anaphase} I is critical
because that is when homologues must disjoin; failure there is a major route
to \omterm{def:g11:meiosis-life-cycles:nondisjunction}{aneuploidy}.
\end{solution}

\begin{solution}{pb:g11:meiosis-life-cycles:1}
\textbf{1--4.} Diagrams: G$_2$ shows long and short dyads (M and P). \omterm{prop:g11:cell-cycle-mitosis:stages}{Metaphase}
I: two \omterm{prop:g11:meiosis-life-cycles:prophaseI}{bivalents} at the plate. After MI: two \omterm{def:g10:the-cell:cell}{cells} each with one long and one
short dyad. After MII: four \omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} \omterm{def:g10:the-cell:cell}{cells} with single \omterm{def:g11:dna-replication:chromosome}{chromosomes}.

\textbf{5.} $2^{2} = 4$ orientations. Gamete types (long, short origin):
(M,M), (M,P), (P,M), (P,P) for maternal/paternal content of each \omterm{def:g11:dna-replication:chromosome}{chromosome}.

\textbf{6.} Mark a chiasma between one maternal and one paternal (non-sister)
chromatid of the same pair --- not between \omterm{def:g11:cell-cycle-mitosis:chromatid}{sister chromatids}.

\textbf{7.} Binary fission typically yields genetically nearly identical
offspring (mutation aside). \omterm{def:g11:meiosis-life-cycles:meiosis}{Meiosis} plus fertilisation recombine and mix two
parents' alleles, so offspring differ from parents and from each other.
\end{solution}
