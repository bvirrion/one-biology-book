\section*{अध्याय \ref{ch:g11:genes-and-disease} --- आनुवंशिक विविधता और रोग}

\begin{solution}{exo:g11:genes-and-disease:1}
\omterm{def:g11:genes-and-disease:genetic}{अप्रभावी}: वह रोग केवल उस \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} की दो प्रतियों के साथ उभरता है; \omterm{def:g11:genes-and-disease:genetic}{प्रभावी}:
एक ही प्रति काफ़ी है। वाहक एक \omterm{def:g11:genes-and-disease:genetic}{अप्रभावी} उत्परिवर्ती \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोता है और
स्वस्थ रहता है।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:2}
रोगग्रस्त $1/4$, वाहक $1/2$, इनमें से कुछ नहीं $1/4$।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:3}
I-1 $Aa$, I-2 $Aa$ (किसी रोगग्रस्त बच्चे के माता-पिता), II-2 $aa$, III-2
$aa$।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:4}
किसी पुरुष के पास एक ही X \omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} होता है: इसलिए उस पर का कोई \omterm{def:g11:genes-and-disease:genetic}{अप्रभावी}
\omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} व्यक्त हो जाता है, क्योंकि उसे ढँकने वाली दूसरी प्रति है ही नहीं।
किसी स्त्री को वह \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} अपने दोनों X \omterm{def:g10:universal-dna:information}{गुणसूत्रों} पर चाहिए।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:5}
वह रोग जिसका जोखिम छोटे-छोटे असर वाले बहुत-से विकल्पियों पर और पर्यावरण पर
निर्भर करता है: टाइप 2 मधुमेह, हृदय रोग (और साथ ही दमा, अधिकांश कैंसर)।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:6}
$aa \times AA$: सारे बच्चे $Aa$, यानी स्वस्थ वाहक। किसी वाहक के साथ कोई
वाहक बच्चा: $1/4$ रोगग्रस्त, $1/2$ वाहक, $1/4$ बिना वाहक।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:7}
II-1 दो वाहकों का स्वस्थ बच्चा है: इसलिए $2/3$ प्रायिकता से वाहक। आबादी
के किसी पुरुष के साथ जोखिम: $2/3 \times 1/25 \times 1/4
= 1/150$।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:8}
हर बच्चे के लिए $1/2$। चारों में से किसी को भी नहीं: $(1/2)^4 = 1/16$।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:9}
$X^hY \times X^HX^H$: सारी बेटियाँ $X^HX^h$, यानी स्वस्थ वाहक; और सारे
बेटे $X^HY$, यानी स्वस्थ। किसी स्वस्थ पुरुष के साथ कोई वाहक बेटी: उसके आधे
बेटे हीमोफ़ीलिया वाले, और आधी बेटियाँ वाहक।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:10}
एक \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} 400 और 200 देता है; और दूसरा, उस अतिरिक्त विलोपन के साथ, 300
और 200। इसलिए 400, 300 तथा 200 पर पट्टियाँ --- और 600 पर कोई नहीं।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:11}
किसी आबादी में किसी \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} की बारंबारता उसके इतिहास पर निर्भर है, केवल
\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} की दर पर नहीं: यूरोपीय \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} बहुत पहले उठा और फैल गया, शायद
इसलिए कि वाहकों को कोई लाभ था (किसी दस्त वाले रोग के प्रतिरोध का शक है);
और पूर्वी एशिया में ऐसा नहीं हुआ।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:12}
हर बच्चा एक स्वतंत्र चिट्ठी है: इसलिए दूसरे के लिए $1/4$। अगले दो में से
कम से कम एक के लिए: $1 - (3/4)^2 = 7/16$। हर गर्भाधान पर \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} नए सिरे
से बाँटे जाते हैं; और पहले हुआ कोई रोगग्रस्त बच्चा अगले का जोखिम घटाता
नहीं।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:13}
जाँच से पहले वह $1/25$ प्रायिकता से वाहक है; जाँच 10\% वाहक चूक जाती है,
इसलिए 1000 स्त्रियों में 40 वाहक हैं और उनमें से 4 की जाँच ऋणात्मक आती है,
जबकि 960 बिना वाहक वालियों की भी ऋणात्मक आती है: यानी वाहक होने की
प्रायिकता $4/964 \approx 1/240$। किसी ज्ञात वाहक के साथ जोखिम:
$1/240 \times 1/4
\approx 1/960$।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:14}
बहुत से थोड़े विकल्पियों तक, गतिहीन रहते हुए: 34\% से 9\%; और बहुत
विकल्पियों के साथ गतिहीन से सक्रिय होते हुए: 34\% से 12\%। यानी जो बदलाव
कोई कर सकता है वह लगभग उतनी ही कमी देता है जितनी वह जो कोई नहीं कर सकता।
जोखिम-अंक यह पहचानने के काम का है कि जीने का ढंग बदलने से किसे सबसे अधिक
मिलेगा, किसी फ़ैसले के रूप में नहीं।
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:15}
वह जाँच निश्चितता देती है --- ऋणात्मक हो तो पचास प्रतिशत की उस छाया से
छुटकारा, और धनात्मक हो तो जीवन तथा परिवार की योजना बनाने की क्षमता --- और
धनात्मक होने पर दशकों पहले किसी लाइलाज रोग का ज्ञान ले लेती है, जिसका असर
मनोदशा, बीमा और उस \omterm{def:g10:universal-dna:gene}{जीन} को साझा करते नातेदारों पर पड़ता है। नतीजे से कोई
इलाज निकलता ही नहीं, इसलिए जानने की अकेली वजह उस व्यक्ति की अपनी है; और
इसीलिए वह चुनाव, तथा उसका समय, उसी का है।
\end{solution}

\begin{solution}{pb:g11:genes-and-disease:1}
\textbf{1.} \omterm{def:g11:genes-and-disease:genetic}{अप्रभावी}: ज़ोए रोगग्रस्त है और उसके दोनों माता-पिता स्वस्थ,
इसलिए हर एक एक छिपा हुआ उत्परिवर्ती \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोता है।

\textbf{2.} लिया $Aa$, टॉम $Aa$, ज़ोए $aa$।

\textbf{3.} $AA$ की प्रायिकता $1/3$, और $Aa$ की $2/3$ (वह स्वस्थ है,
इसलिए $aa$ बाहर है)।

\textbf{4.} टॉम वाहक है, इसलिए उसके माता-पिता में से कम से कम एक भी; और उस
पीढ़ी में कोई रोगग्रस्त न होने से आना को वह \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} उस जनक से $1/2$
प्रायिकता से मिला: यानी $1/2$ (दूसरे जनक को बिना वाहक वाला मानते हुए)।

\textbf{5.} $1/25$, यानी आबादी वाला आँकड़ा।

\textbf{6.} $1/2 \times 1/25 \times 1/4 = 1/200$।

\textbf{7.} रोगग्रस्त $1/4$; वाहक $1/2$।

\textbf{8.} $2/3 \times 1/25 \times 1/4 = 1/150$।

\textbf{9.} ज़ोए $aa$ है; और उसका साथी $Aa$ हो, इसकी प्रायिकता $1/25$ है।
रोगग्रस्त: $1/25 \times 1/2 = 1/50$। वाहक: हर बच्चे को ज़ोए से $a$ मिलता
ही है, इसलिए वाहक होने की प्रायिकता $24/25 \times 1 + 1/25 \times
1/2 \approx 0.98$।

\textbf{10.} लू $2/3$ प्रायिकता से वाहक है; और उस चचेरे भाई-बहन को $a$
आना से (जो $1/2$ से वाहक है) $1/2$ प्रायिकता से मिला, यानी वह
$1/4$ प्रायिकता से वाहक है (करीम को छोड़ते हुए): यानी $2/3 \times 1/4
\times 1/4 = 1/24$, जो मैक्स से चार गुना अधिक है, क्योंकि वह चचेरा
भाई-बहन उसी परिवार से आता है और किसी अजनबी से कहीं अधिक संभावना से वही
\omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोता है।

\textbf{11.} ज़ोए: केवल 600 और 300. टॉम: 900, 600 और 300. बिना वाहक वाला:
केवल 900.

\textbf{12.} $AA$: वह वाहक है ही नहीं; इसलिए उसके बच्चे का जोखिम
$1/150$ से गिरकर शून्य हो जाता है (किसी नए \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} को छोड़कर)।

\textbf{13.} आना $Aa$ है: जोखिम $1 \times 1/25 \times 1/4 = 1/100$।

\textbf{14.} करीम जैसे 1000 लोगों में 40 वाहक हैं, जिनमें से 4 की जाँच
ऋणात्मक आती है, और 960 बिना वाहक वालों की भी ऋणात्मक: यानी $4/964 \approx
1/240$। जोखिम: $1 \times 1/240 \times 1/4 \approx 1/960$।

\textbf{15.} $Aa$: यानी अपनी माता जैसा एक स्वस्थ वाहक।

\textbf{16.} दंपतियों में से $(1/25)^2 = 1/625$ दो वाहकों वाले हैं; और
उनके एक-चौथाई बच्चे रोगग्रस्त होते हैं: यानी $1/2500$।

\textbf{17.} ठीक वही बारंबारता जो देखी गई: वाहकों की बारंबारता और \omterm{def:g11:genes-and-disease:genetic}{अप्रभावी}
वाला नियम मिलकर उसका हिसाब दे देते हैं।

\textbf{18.} $1/25$ की वाहक-बारंबारता और $1/2500$ की रोगग्रस्त-बारंबारता
के साथ स्वस्थ वाहकों के ढोए उत्परिवर्ती \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} रोगग्रस्तों वालों से
$2 \times 1/25$ बनाम $2 \times
1/2500$, यानी सौ के मुक़ाबले एक, की तरह अधिक हैं। रोगग्रस्तों की मृत्यु हर
पीढ़ी में उन प्रतियों का केवल 1\% हटाती है; और बाक़ी वाहक चुपचाप आगे
पहुँचाते रहते हैं।

\textbf{19.} उस \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} की बारंबारता बढ़ेगी, पर बहुत धीरे: जो 1\% प्रतियाँ
पहले हर पीढ़ी खोती थीं वे अब बची रहती हैं, इसलिए बदलाव प्रति पीढ़ी उस
बारंबारता के 1\% की कोटि का है।

\textbf{20.} दो वाहकों पर चार में एक रोगग्रस्त बच्चे का जोखिम है; किसी
रोगग्रस्त बच्चे का स्वस्थ सहोदर तीन में दो बार वाहक होता है; और जेल ने
प्रायिकताओं को \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूपों} में बदल दिया --- मैक्स को बरी करके, आना की
पुष्टि करके, और करीम को तसल्ली देकर पर बरी किए बिना --- तथा गर्भस्थ शिशु
को एक स्वस्थ वाहक के रूप में पहचान लिया।
\end{solution}
