\section*{Capítulo \ref{ch:g4:heart-and-blood} --- El corazón y la sangre}

\begin{solution}{exo:g4:heart-and-blood:1}
Los \omterm{def:g4:journey-of-food:digestion}{nutrientes}, recogidos en el \omterm{def:g4:journey-of-food:tract}{intestino delgado}; y el oxígeno,
recogido en los \omterm{def:g4:breathing:organs}{pulmones}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:2}
El dióxido de carbono --- lo deja en los \omterm{def:g4:breathing:organs}{pulmones}, donde se espira.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:3}
Es un \omterm{def:g3:bones-and-muscles:muscle}{músculo} hueco, más o menos del tamaño del puño de su dueño,
colocado en el centro del pecho detrás de la caja torácica.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:4}
El lado derecho bombea \omterm{prop:g4:journey-of-food:absorb}{sangre} a los \omterm{def:g4:breathing:organs}{pulmones} para cargar oxígeno; el
lado izquierdo bombea la \omterm{prop:g4:journey-of-food:absorb}{sangre} rica en oxígeno a todo el cuerpo.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:5}
La pequeña onda que cada \omterm{def:g4:heart-and-blood:heart}{latido} empuja por los \omterm{def:g4:heart-and-blood:vessels}{vasos}, notada en la
\omterm{def:g1:my-body:joint}{muñeca} o en el cuello. Contarla es contar los \omterm{def:g4:heart-and-blood:heart}{latidos} por minuto.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:6}
El pulgar tiene un pulsito propio bastante notable, así que podrías
acabar contando tu pulgar en vez de la \omterm{def:g1:my-body:joint}{muñeca} que estás midiendo.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:7}
Las \omterm{def:g1:my-body:parts}{piernas} que esprintan queman \omterm{def:g4:journey-of-food:digestion}{nutrientes} y oxígeno deprisa y hay
que reponérselos deprisa; el \omterm{def:g4:heart-and-blood:heart}{corazón} acelera y refuerza sus \omterm{def:g4:heart-and-blood:heart}{latidos}
para que la \omterm{prop:g4:journey-of-food:absorb}{sangre} haga sus rondas de reparto más a menudo.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:8}
Respuesta abierta. Lo típico: unos $80$--$100$ en reposo y
$120$--$150$ después de los saltos de estrella --- la cadena de
suministro en marcha.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:9}
Al salir de los \omterm{def:g4:breathing:organs}{pulmones} acababa de cargar oxígeno --- roja brillante.
En el \omterm{def:g3:bones-and-muscles:muscle}{músculo} entregó el oxígeno y cargó dióxido de carbono --- más
oscura. El color sigue a la carga.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:10}
El \omterm{def:g4:journey-of-food:tract}{intestino delgado} pone \omterm{def:g4:journey-of-food:digestion}{nutrientes} en la \omterm{prop:g4:journey-of-food:absorb}{sangre}; los \omterm{def:g4:breathing:organs}{pulmones} ponen
oxígeno y sacan dióxido de carbono; el \omterm{def:g4:heart-and-blood:heart}{corazón} es la bomba que
mantiene la \omterm{prop:g4:journey-of-food:absorb}{sangre} en movimiento entre ellos y hasta todos los
consumidores --- \omterm{def:g3:bones-and-muscles:muscle}{músculos}, cerebro, \omterm{def:g3:bones-and-muscles:skeleton}{huesos}, \omterm{def:g1:the-five-senses:sense}{piel}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:11}
El \omterm{def:g4:heart-and-blood:heart}{corazón} entrenado es un \omterm{def:g3:bones-and-muscles:muscle}{músculo} más fuerte: cada \omterm{def:g4:heart-and-blood:heart}{latido} mueve más
\omterm{prop:g4:journey-of-food:absorb}{sangre}. Entregar el mismo suministro necesita entonces menos \omterm{def:g4:heart-and-blood:heart}{latidos}
por minuto --- así que los tranquilos $50$ del ciclista hacen el
trabajo de los $75$ sin entrenar, con potencia de sobra guardada para
el esfuerzo.
\end{solution}
