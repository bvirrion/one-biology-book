\chapter{Genom Sel dan Virus}\label{ch:b1:genomes}

Dua meter \omterm{def:b1:nucleic-acids:chain}{DNA} terlipat ke dalam sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} berlebar enam mikrometer,
dan terlipat sedemikian sehingga mana pun dari dua puluh ribu \omterm{def:b1:genomes:gene}{gennya}
dapat ditemukan lalu dibaca dalam beberapa menit. Sebuah bakteri
melipat satu setengah milimeter ke dalam sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} yang seribu kali
lebih pendek, dan menyalinnya tiap dua puluh menit. Sebuah \omterm{def:b1:genomes:virus}{virus}
membawa beberapa ribu huruf dalam sebuah cangkang \omterm{def:b1:proteins:peptide}{protein} lalu
membacanya dengan sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} yang bukan miliknya. Bab ini memerikan
\emph{\omterm{def:b1:genomes:genome}{genom}} --- seluruh \omterm{def:b1:nucleic-acids:chain}{DNA} sebuah \omterm{def:b1:organism-environment:organism}{organisme} --- sebagai sebuah benda
fisik: bagaimana \omterm{def:b1:nucleic-acids:chain}{DNA} bakteri, \omterm{def:b1:cell-unit-of-life:prokeuk}{sel eukariot}, organel dan \omterm{def:b1:genomes:virus}{virus} tersusun,
apa yang dikandungnya selain \omterm{def:b1:genomes:gene}{gen}, dan mengapa banyaknya begitu sedikit
mengatakan tentang \omterm{def:b1:organism-environment:organism}{organisme} yang membawanya.

\section{Genom bakteri}

\begin{definition}[Genom, kromosom, nukleoid]\label{def:b1:genomes:genome}
Yang disebut \emph{genom}\index{genom} sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} adalah keseluruhan
\omterm{def:b1:nucleic-acids:chain}{DNA-nya}: informasi bagi tiap \omterm{def:b1:proteins:peptide}{protein} dan \omterm{def:b1:nucleic-acids:chain}{RNA} yang dapat dibuatnya, serta
urutan yang mengendalikan kapan keduanya dibuat. Sebuah
\emph{kromosom}\index{kromosom} adalah satu molekul \omterm{def:b1:nucleic-acids:chain}{DNA} beserta \omterm{def:b1:proteins:peptide}{protein}
yang menatanya. Sebuah bakteri biasanya memiliki satu kromosom,
melingkar, sepanjang satu sampai sepuluh juta \omterm{thm:b1:nucleic-acids:helix}{pasangan basa}
(\emph{E. coli}: $\num{4.6e6}$, sepanjang \qty{1.5}{mm}), yang terlipat
ke dalam sebuah kawasan \omterm{def:b1:cell-unit-of-life:cell}{sitoplasma}, yaitu \emph{nukleoid}\index{nukleoid},
tanpa membran di sekelilingnya. Selain kromosomnya banyak bakteri
membawa \emph{plasmid}\index{plasmid}: lingkaran kecil berisi beberapa
ribu sampai beberapa ratus ribu pasangan, dalam satu sampai ratusan
salinan, yang bereplikasi sendiri dan membawa \omterm{def:b1:genomes:gene}{gen} pilihan --- ketahanan
antibiotik, racun, mesin untuk memindahkan dirinya ke \omterm{def:b1:cell-unit-of-life:cell}{sel} lain.
\end{definition}

\begin{proposition}[Bagaimana sebuah bakteri melipat kromosomnya]\label{prop:b1:genomes:supercoiling}
\index{pilinan lebih}
Sebuah \omterm{def:b1:nucleic-acids:chain}{DNA} melingkar yang ujungnya tak dapat berputar dapat dipilin
seperti sebuah karet gelang: \emph{\omterm{prop:b1:genomes:supercoiling}{pilinan lebih}}. \omterm{def:b1:enzymes:enzyme}{Enzim} yang disebut
\emph{topoisomerase} memotong, melewatkan lalu menyegel kembali
untainya untuk menambah atau menyingkirkan putaran; \emph{girase}
memakai \omterm{def:b1:water-small-molecules:atp}{ATP} untuk memasukkan \omterm{prop:b1:genomes:supercoiling}{pilinan lebih} negatif, yang mengurangi
lilitan heliksnya dan membuatnya sekaligus lebih padat serta lebih
mudah dibuka. \omterm{def:b1:genomes:genome}{Kromosom} \emph{E. coli} tertata menjadi sekitar lima
puluh gelung, yang masing-masing berpilin lebih sendiri-sendiri dan
ditambatkan pada sebuah teras \omterm{def:b1:proteins:peptide}{protein}, sehingga \qty{1.5}{mm}-nya
menempati sebuah \omterm{def:b1:genomes:genome}{nukleoid} selebar sekitar satu mikrometer ---
pemampatan seribu kali --- sementara gelung mana pun dapat diurai untuk
dibaca atau disalin tanpa mengganggu selebihnya. \omterm{def:b1:proteins:peptide}{Protein} basa yang
kecil (serupa \omterm{def:b1:genomes:nucleosome}{histon}, meskipun tak berkerabat dengan \omterm{def:b1:genomes:nucleosome}{histon}) menekuk
dan menjembatani \omterm{def:b1:nucleic-acids:chain}{DNA-nya} di seluruh bagian.
\end{proposition}

\begin{example}[Sebuah genom yang padat]\label{ex:b1:genomes:dense}
Dari \qty{4.6}{Mb} \omterm{def:b1:genomes:genome}{kromosom} \emph{E. coli}, \qty{88}{\%} menyandi
\omterm{def:b1:proteins:peptide}{protein}: sekitar \num{4300} \omterm{def:b1:genomes:gene}{gen} yang masing-masing sekitar \num{1000}
pasangan, terkemas kepala ke ekor dengan seratus pasangan di antaranya,
banyak di antaranya berkelompok menjadi \emph{operon} yang
ditranskripsikan bersama (\cref{ch:b1:expression-control}). Sebuah
\omterm{def:b1:genomes:genome}{plasmid} \qty{5}{kb} dapat membawa tiga \omterm{def:b1:genomes:gene}{gen} dan hadir dalam lima puluh
salinan; \omterm{def:b1:genomes:gene}{gen} ketahanan yang dibawanya dapat menyebar ke seluruh bakteri
sebuah bangsal rumah sakit dalam beberapa minggu, lewat konjugasi, lebih
cepat daripada yang dapat dimunculkan mutasi mana pun.
\end{example}

\section{Genom eukariot: kromatin}

\begin{definition}[Nukleosom, kromatin]\label{def:b1:genomes:nucleosome}
\index{nukleosom}\index{histon}\index{kromatin}
Dalam sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} eukariot \omterm{def:b1:nucleic-acids:chain}{DNA-nya} digulung pada \omterm{def:b1:proteins:peptide}{protein}: \qty{147}{}
\omterm{thm:b1:nucleic-acids:helix}{pasangan basa} membuat 1.7 putaran mengelilingi sebuah oktamer berisi
delapan \emph{histon} (masing-masing dua H2A, H2B, H3, H4, yaitu
\omterm{def:b1:proteins:peptide}{protein} kecil yang kaya lisina dan arginina yang menetralkan fosfatnya),
sehingga membentuk sebuah \emph{nukleosom}\index{nukleosom} selebar
\qty{11}{nm}; nukleosomnya menyusul satu sama lain tiap sekitar
\qty{200}{} pasangan, seperti manik pada seutas tali, dengan histon
kelima, H1, yang menyegel tiap maniknya. \emph{Kromatin} ini melipat
lebih jauh: menjadi sebuah serat \qty{30}{nm}, menjadi gelung berisi
puluhan ribu pasangan yang ditambatkan pada sebuah perancah \omterm{def:b1:proteins:peptide}{protein},
dan --- pada pembelahannya --- menjadi \omterm{def:b1:flowering-plant-organization:organs}{batang} padat \omterm{def:b1:genomes:genome}{kromosom} metafase,
sepuluh ribu kali lebih pendek daripada \omterm{def:b1:nucleic-acids:chain}{DNA} telanjangnya.
\emph{Eukromatin} adalah bentuk interfase yang lebih longgar, kaya \omterm{def:b1:genomes:gene}{gen}
dan ditranskripsikan; \emph{heterokromatin} bentuk yang memadat dan
miskin \omterm{def:b1:genomes:gene}{gen} yang tetap padat (di sekeliling \omterm{def:b1:genomes:karyotype}{sentromer}, pada ujungnya,
dan pada sebuah \omterm{def:b1:genomes:genome}{kromosom} X yang dipadamkan).
\end{definition}

\begin{definition}[Kariotipe, sentromer, telomer]\label{def:b1:genomes:karyotype}
\index{kariotipe}\index{sentromer}\index{telomer}
Yang disebut \emph{kariotipe} adalah perangkat \omterm{def:b1:genomes:genome}{kromosom} sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel},
yang dipilah menurut ukuran dan pola pitanya: manusia memiliki 23
pasang (22 autosom dan XY atau XX), $\num{6.4e9}$ \omterm{thm:b1:nucleic-acids:helix}{pasangan basa}
seluruhnya, sepanjang \qty{2.2}{m}; \omterm{def:b1:genomes:genome}{kromosom} terbesarnya memegang
\qty{250}{Mb}, terkecilnya \qty{50}{}. Tiap \omterm{def:b1:genomes:genome}{kromosom} lurus membawa
sebuah \emph{sentromer}, yaitu kawasan tersempit tempat kedua salinannya
tetap bergabung sesudah replikasi dan tempat gelendongnya melekat
(\cref{ch:b1:replication-mitosis}), serta dua \emph{telomer}, yaitu
urutan berulang pada ujungnya yang melindunginya agar tidak dikira \omterm{def:b1:nucleic-acids:chain}{DNA}
yang patah dan agar tidak memendek pada tiap replikasinya.
\end{definition}

\begin{proposition}[Ukuran genom tidak mengukur kerumitan]\label{prop:b1:genomes:cvalue}
\index{paradoks nilai C}

Ukuran \omterm{def:b1:genomes:genome}{genom} beragam seribu kali di antara eukariot tanpa hubungan

antaranya.
\end{proposition}

\section{Apa yang dikandung sebuah genom}

\begin{definition}[Anatomi sebuah gen]\label{def:b1:genomes:gene}
\index{gen}\index{ekson}\index{intron}\index{promotor}
Yang disebut \emph{gen} adalah sebentang \omterm{def:b1:nucleic-acids:chain}{DNA} yang ditranskripsikan
menjadi sebuah \omterm{def:b1:nucleic-acids:chain}{RNA} yang berfungsi --- biasanya sebuah pesan bagi satu
\omterm{def:b1:proteins:peptide}{protein}. Sebuah gen eukariot penyandi \omterm{def:b1:proteins:peptide}{protein} terdiri atas sebuah
\emph{promotor}, yaitu urutan di hulunya tempat transkripsinya dimulai
dan dikendalikan; \emph{ekson}, yaitu bagian yang disimpan dalam pesan
matangnya; \emph{intron}, yaitu bagian yang ditranskripsikan lalu
dipotong keluar (\cref{ch:b1:gene-expression}); kawasan tak tertranslasi
pada tiap ujung pesannya; dan sebuah pengakhir. Sebuah gen manusia
rata-rata \qty{27}{kb} dengan delapan ekson yang seluruhnya
\qty{1.5}{kb}: sembilan belas per dua puluh sebuah gen yang lazim
adalah intron. Gen bakteri tidak berintron.
\end{definition}

\begin{proposition}[Cacah isi genom manusia]\label{prop:b1:genomes:census}
\index{transposon}\index{DNA berulang}
Dari \qty{3.2}{Gb}-nya: sekitar \qty{1.5}{\%} menyandi \omterm{def:b1:proteins:peptide}{protein};
\qty{25}{\%} berupa \omterm{def:b1:genomes:gene}{intron}; \qty{45}{\%} adalah sisa \emph{unsur yang
dapat berpindah} --- yaitu urutan yang menyalin dirinya ke tempat baru,
sebagian besarnya sudah lama mati (unsur Alu saja, \qty{300}{bp}, hadir
dalam sejuta salinan, sepersepuluh \omterm{def:b1:genomes:genome}{genomnya}); \qty{8}{\%} berupa ulangan
pendek yang berderet (\omterm{def:b1:nucleic-acids:chain}{DNA} satelit, pada \omterm{def:b1:genomes:karyotype}{sentromer} dan \omterm{def:b1:genomes:karyotype}{telomernya});
selebihnya adalah urutan bukan penyandi yang khas yang mencakup
\omterm{def:b1:genomes:gene}{promotor}, penguat dan \omterm{def:b1:genomes:gene}{gen} \omterm{def:b1:nucleic-acids:chain}{RNA} yang mengendalikan pengungkapannya. Banyak
\omterm{def:b1:genomes:gene}{gen} termasuk dalam \emph{keluarga} yang muncul lewat penggandaan
(globin, reseptor penciuman --- seribu jumlahnya), dan \omterm{def:b1:genomes:genome}{genomnya} memegang
ribuan \emph{pseudogen}, yaitu salinan mati yang tak lagi bekerja.
Sebuah \omterm{def:b1:genomes:genome}{genom} bukanlah sebuah rancangan melainkan sebuah catatan.
\end{proposition}

\section{Genom organel dan virus}

\begin{definition}[Genom organel]\label{def:b1:genomes:organelle}
\omterm{def:b1:eukaryotic-cell:mitochondrion}{Mitokondria} dan kloroplas menyimpan sisa \omterm{def:b1:genomes:genome}{kromosom} leluhur bakterinya
(\cref{ch:b1:eukaryotic-cell}): sebuah lingkaran kecil ---
\qty{16.6}{kb} dan 37 \omterm{def:b1:genomes:gene}{gen} dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria} manusia,
\qtyrange{120}{160}{kb} dan sekitar 100 \omterm{def:b1:genomes:gene}{gen} dalam kloroplas --- dalam
banyak salinan tiap organelnya, yang menyandi sebagian \omterm{def:b1:proteins:peptide}{proteinnya}
sendiri dan \omterm{def:b1:nucleic-acids:rnas}{RNA ribosomnya}. Selebihnya \omterm{def:b1:proteins:peptide}{proteinnya} datang dari \omterm{def:b1:genomes:gene}{gen}
\omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}. Pada sebagian besar hewan \omterm{def:b1:genomes:genome}{genom} \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondrianya} diwarisi hanya
dari ibunya, bersama \omterm{def:b1:cell-unit-of-life:cell}{sitoplasma} \omterm{def:b1:cell-unit-of-life:cell}{sel} telurnya.
\end{definition}

\begin{definition}[Virus]\label{def:b1:genomes:virus}
\index{virus}\index{kapsid}
Yang disebut \emph{virus} adalah sebuah \omterm{def:b1:genomes:genome}{genom} dalam sebuah cangkang
\omterm{def:b1:proteins:peptide}{protein}, yaitu \emph{kapsid} --- kadang terbungkus sebuah membran, yaitu
\emph{selubung}, yang diambil dari sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} inang --- yang berbiak
hanya di dalam sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}, dengan memakai ribosom, energi dan prazat
\omterm{def:b1:cell-unit-of-life:cell}{sel} itu. \omterm{def:b1:genomes:genome}{Genomnya} dapat berupa \omterm{def:b1:nucleic-acids:chain}{DNA} atau \omterm{def:b1:nucleic-acids:chain}{RNA}, berutas ganda atau
tunggal, lurus atau melingkar, satu molekul atau beberapa: dari
\qty{5}{kb} (beberapa \omterm{def:b1:genomes:gene}{gen}) sampai \qty{1.2}{Mb} (seribu \omterm{def:b1:genomes:gene}{gen}, pada virus
raksasa). Di luar sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} sebuah virus tidak berbuat apa-apa: ia
bukan \omterm{def:b1:cell-unit-of-life:cell}{sel}, tidak bermetabolisme, dan hidup hanya dalam arti bahwa ia
membawa informasi dan berevolusi. \emph{Bakteriofag} adalah virus
bakteri.
\end{definition}

{\small Bakteriofag yang melekat pada sebuah bakteri: kepala bersegi
banyak yang memegang \omterm{def:b1:nucleic-acids:chain}{DNA-nya}, ekor yang akan menjadi jalan
penyuntikannya. Satu \omterm{def:b1:cell-unit-of-life:cell}{sel} akan melepas seratus fag baru dalam setengah

\end{omfigure}

\begin{proposition}[Kedua daur fag $\lambda$]\label{prop:b1:genomes:lambda}
\index{daur litik}\index{daur lisogenik}
Fag $\lambda$ menyuntikkan \qty{48.5}{kb} \omterm{def:b1:nucleic-acids:chain}{DNA-nya}
(\cref{ch:b1:nucleic-acids}) ke dalam \emph{E. coli}, tempat \omterm{def:b1:nucleic-acids:chain}{DNA} itu
melingkar. Dua nasib lalu mengikuti. Dalam \emph{\omterm{prop:b1:genomes:lambda}{daur litik}} \omterm{def:b1:genomes:gene}{gen} fagnya
ditranskripsikan polimerase inangnya, \omterm{def:b1:nucleic-acids:chain}{DNA-nya} direplikasi seratus kali,
\omterm{def:b1:proteins:peptide}{protein} \omterm{def:b1:genomes:virus}{kapsidnya} dibuat lalu dirakit, \omterm{def:b1:nucleic-acids:chain}{DNA-nya} dikemas ke dalamnya, dan
sebuah \omterm{def:b1:enzymes:enzyme}{enzim} melarutkan dindingnya: \omterm{def:b1:cell-unit-of-life:cell}{selnya} pecah, sehingga melepas
seratus fag, empat puluh menit sesudah infeksinya. Dalam \emph{\omterm{prop:b1:genomes:lambda}{daur
lisogenik}} \omterm{def:b1:nucleic-acids:chain}{DNA} fagnya disisipkan ke dalam \omterm{def:b1:genomes:genome}{kromosom} inangnya lalu
dibungkam sebuah penekan buatannya sendiri: \emph{profag} itu disalin
bersama \omterm{def:b1:genomes:genome}{kromosomnya} selama beberapa generasi, tak terlihat, sampai
kerusakan \omterm{def:b1:nucleic-acids:chain}{DNA} inangnya mengangkat penekanannya dan \omterm{prop:b1:genomes:lambda}{daur litiknya}
berlanjut. Pilihannya adalah sebuah sakelar molekul, salah satu yang
pertama dipahami (\cref{ch:b1:expression-control}).
\end{proposition}

\begin{example}[Virus bergenom RNA]\label{ex:b1:genomes:rna}
Influenza membawa delapan \omterm{def:b1:flowering-plant-organization:organs}{ruas} \omterm{def:b1:nucleic-acids:chain}{RNA} berutas tunggal beserta
polimerasenya sendiri untuk menyalinnya, sebab \omterm{def:b1:cell-unit-of-life:cell}{sel} tidak memiliki \omterm{def:b1:enzymes:enzyme}{enzim}
yang menyalin \omterm{def:b1:nucleic-acids:chain}{RNA}; polimerasenya membuat satu kekeliruan tiap
\num{10000} basa dan \omterm{def:b1:genomes:virus}{virusnya} berubah tiap musim. Sebuah retrovirus
(HIV) membawa \omterm{def:b1:nucleic-acids:chain}{RNA} dan sebuah \emph{transkriptase balik} yang menyalinnya
menjadi \omterm{def:b1:nucleic-acids:chain}{DNA}, yang disisipkan ke dalam \omterm{def:b1:genomes:genome}{kromosom} inangnya seperti sebuah
profag --- sebuah infeksi yang menetap. Mekanismenya, dan tanggapan
kekebalan terhadapnya, menjadi bagian jilid Tahun ke-3; di sini keduanya
menandai rentang apa yang dapat menjadi sebuah \omterm{def:b1:genomes:genome}{genom}.
\end{example}

\section{Soal: Dua Meter dalam Enam Mikrometer}

\begin{problem}[{Soal akhir pekan --- sebuah inti manusia yang diukur
dari heliksnya sampai kromosomnya: panjang, volume, nukleosom, histon
dan nisbah pengemasannya pada tiap tingkat, berakhir pada pemampatan
total metafasenya}]
\label{pb:b1:genomes:1}
Sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} diploid manusia adalah bola bergaris tengah
\qty{6}{\micro m} yang memegang $\num{6.4e9}$ \omterm{thm:b1:nucleic-acids:helix}{pasangan basa} pada 46
\omterm{def:b1:genomes:genome}{kromosom}; \omterm{def:b1:genomes:genome}{kromosom} terbesarnya membawa \qty{250}{Mb}, terkecilnya
\qty{50}{}. Ambil \qty{0.34}{nm} tiap pasangan, \qty{2}{nm} bagi garis
tengah heliksnya, \qty{650}{Da} tiap pasangan, \num{200} pasangan tiap
\omterm{def:b1:genomes:nucleosome}{nukleosom}, sebuah oktamer \omterm{def:b1:genomes:nucleosome}{histon} \qty{108}{kDa}, dan
$\qty{1.66e-24}{g}$ tiap dalton.

\medskip
\noindent\textbf{Bagian I --- Panjang dan volume.}
\begin{enumerate}
  \item Hitunglah total panjang \omterm{def:b1:nucleic-acids:chain}{DNA} dalam \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}.
  \item Hitunglah banyaknya putaran \omterm{thm:b1:nucleic-acids:helix}{heliks ganda} yang dikandungnya.
  \item Hitunglah panjang \omterm{def:b1:nucleic-acids:chain}{DNA} \omterm{def:b1:genomes:genome}{kromosom} terbesar dan terkecilnya.
  \item Hitunglah volume \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}.
  \item Hitunglah volume \omterm{def:b1:nucleic-acids:chain}{DNA-nya} sebagai sebuah tabung, dan fraksi
        \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya} yang ditempatinya.
  \item Hitunglah massa \omterm{def:b1:nucleic-acids:chain}{DNA} dalam \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya} dalam pikogram.
  \item \omterm{def:b1:cell-unit-of-life:prokeuk}{Intinya} memuat sekitar \qty{25}{pg} \omterm{def:b1:proteins:peptide}{protein}. Berapa fraksi
        massa \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya} yang berupa \omterm{def:b1:nucleic-acids:chain}{DNA}, bila airnya \qty{80}{\%} \omterm{def:b1:cell-unit-of-life:prokeuk}{inti}
        itu (massa jenis \qty{1.1}{g/mL})?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- \omterm{def:b1:genomes:nucleosome}{Nukleosom}.}
\begin{enumerate}[resume]
  \item Hitunglah banyaknya \omterm{def:b1:genomes:nucleosome}{nukleosom} dalam \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}.
  \item Hitunglah massa \omterm{def:b1:genomes:nucleosome}{histonnya} lalu bandingkan dengan massa \omterm{def:b1:nucleic-acids:chain}{DNA-nya}.
  \item \qty{200}{} pasangan satu \omterm{def:b1:genomes:nucleosome}{nukleosom} menjangkau \qty{68}{nm}
        heliks tetapi menempati sebuah manik sepanjang \qty{11}{nm}.
        Hitunglah faktor pemampatan ``manik pada seutas tali''-nya.
  \item Serat \qty{30}{nm} memegang enam \omterm{def:b1:genomes:nucleosome}{nukleosom} tiap \qty{11}{nm}
        panjangnya. Hitunglah faktor pemampatannya terhadap \omterm{def:b1:nucleic-acids:chain}{DNA}
        telanjang.
  \item Hitunglah panjang serat \qty{30}{nm} bagi seluruh \omterm{def:b1:genomes:genome}{genomnya},
        lalu bandingkan dengan garis tengah \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}.
  \item Tiap \omterm{def:b1:genomes:nucleosome}{nukleosom} harus dibongkar lalu dirakit kembali ketika
        \omterm{def:b1:nucleic-acids:chain}{DNA-nya} direplikasi atau ditranskripsikan. Berapa banyak
        \omterm{def:b1:genomes:nucleosome}{nukleosom} yang dirakit kembali dalam fase S (\qty{8}{h})? Tiap
        sekon?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Gelung dan \omterm{def:b1:genomes:genome}{kromosom}.}
\begin{enumerate}[resume]
  \item Gelung berisi \qty{75}{kb} serat \qty{30}{nm} menggantung dari
        sebuah perancah. Hitunglah panjang serat dalam satu gelung dan
        banyaknya gelung dalam \omterm{def:b1:genomes:genome}{genomnya}.
  \item Sebuah kromatid metafase \omterm{def:b1:genomes:genome}{kromosom} terbesarnya sepanjang
        \qty{10}{\micro m}. Hitunglah faktor pemampatan dari \omterm{def:b1:nucleic-acids:chain}{DNA}
        telanjang sampai \omterm{def:b1:genomes:genome}{kromosom} metafasenya.
  \item Hitunglah volume kromatid itu sebagai sebuah tabung bergaris
        tengah \qty{0.7}{\micro m}, dan fraksinya yang berupa \omterm{def:b1:nucleic-acids:chain}{DNA}.
  \item Ke-46 \omterm{def:b1:genomes:genome}{kromosom} pada metafase, yang masing-masing sebuah \omterm{def:b1:flowering-plant-organization:organs}{batang}
        berkerapatan sama: hitunglah total volumenya lalu bandingkan
        dengan \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} interfasenya.
  \item Jelaskan mengapa \omterm{def:b1:genomes:nucleosome}{kromatin} interfase tidak dapat dikemas
        serapat \omterm{def:b1:genomes:nucleosome}{kromatin} metafase.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Menemukan sebuah \omterm{def:b1:genomes:gene}{gen}.}
\begin{enumerate}[resume]
  \item Sebuah \omterm{def:b1:proteins:peptide}{protein} harus menemukan satu sisi \qty{20}{bp} dalam
        \omterm{def:b1:genomes:genome}{genomnya}. Bila ia menguji satu sisi acak tiap milisekon, berapa
        lama pencariannya? Apa yang dikatakan hal ini tentang bagaimana
        sisinya ditemukan?
  \item \qty{20000}{} \omterm{def:b1:genomes:gene}{gen} \omterm{def:b1:genomes:genome}{genomnya} rata-rata \qty{27}{kb}. Berapa
        fraksi \omterm{def:b1:genomes:genome}{genomnya} yang berada di dalam \omterm{def:b1:genomes:gene}{gen}? Berapa fraksinya yang
        menyandi, pada \qty{1.5}{kb} \omterm{def:b1:genomes:gene}{ekson} tiap \omterm{def:b1:genomes:gene}{gen}?
  \item Tiap \omterm{def:b1:genomes:genome}{kromosom} menempati wilayahnya sendiri dalam \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}.
        Hitunglah volume wilayah \omterm{def:b1:genomes:genome}{kromosom} terbesarnya bila wilayahnya
        sebanding dengan kandungan \omterm{def:b1:nucleic-acids:chain}{DNA-nya}, dan kepekatan \omterm{def:b1:nucleic-acids:chain}{DNA} di
        dalamnya dalam \omterm{thm:b1:nucleic-acids:helix}{pasangan basa} tiap mikrometer kubik.
  \item Sebuah \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} ikan paru dengan kerapatan \omterm{def:b1:nucleic-acids:chain}{DNA} yang sama akan
        bervolume berapa bagi \qty{260}{Gb} (diploid)? Bergaris tengah
        berapa?
  \item Hitunglah banyaknya oktamer \omterm{def:b1:genomes:nucleosome}{histon} yang harus dibuat sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}
        dalam fase S dan \omterm{def:b1:proteins:aminoacid}{asam amino} yang dikandungnya (\qty{1000}{}
        residu tiap oktamer). Bandingkan dengan \qty{3e9}{} residu
        \omterm{def:b1:proteins:peptide}{protein} yang dibuat sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} dalam sehari.
  \item Jelaskan dalam dua kalimat mengapa bakteri dapat tanpa
        \omterm{def:b1:genomes:nucleosome}{nukleosom} dan eukariot tidak.
  \item Nyatakan hasilnya: faktor pemampatan total dari heliksnya

\end{enumerate}
