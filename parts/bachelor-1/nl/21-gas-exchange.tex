\chapter{Gasuitwisseling bij de ademhaling}\label{ch:b1:gas-exchange}

Een liter lucht bevat dertig keer meer zuurstof dan een liter water, weegt
achthonderd keer minder, en laat zuurstof er tienduizend keer sneller doorheen
diffunderen. Een forel moet voor elke milligram zuurstof die zij opneemt
honderdvijftig gram water over haar \omterm{def:b1:gas-exchange:gill}{kieuwen} laten stromen, en zij haalt er vier
vijfde uit van wat het water draagt; een mens verplaatst per milligram enkele
grammen lucht en verspilt er drie kwart van. De twee \omterm{def:b1:mammal-organization:organ}{organen} zijn antwoorden op
dezelfde vergelijking --- de flux van een gas door een oppervlak --- in twee media.
Dit hoofdstuk zet de fysica uiteen, leidt af wat een \omterm{prop:b1:organism-environment:surfaces}{uitwisselingsoppervlak} moet
zijn, beschrijft \omterm{def:b1:gas-exchange:gill}{kieuwen}, \omterm{def:b1:gas-exchange:lung}{longen} en tracheeën, volgt zuurstof en koolstofdioxide
door het bloed, en kijkt naar de regeling van de ventilatie en naar wat er op
hoogte en onder water verandert.

\section{De fysica van een gas in twee media}

\begin{definition}[Partiële druk, oplosbaarheid]\label{def:b1:gas-exchange:partial}
\index{partiële druk}
In een gasmengsel oefent elk gas een \emph{partiële druk} uit die evenredig is met
zijn aandeel: in lucht bij \qty{101}{kPa} heeft zuurstof (\qty{20.9}{\%})
$P_{\mathrm{O_2}} = \qty{21.2}{kPa}$, koolstofdioxide (\qty{0.04}{\%})
\qty{0.04}{kPa}. Een gas lost in een vloeistof op naar rato van zijn partiële druk
(de wet van Henry): water dat bij \qty{15}{\celsius} met lucht in evenwicht is
bevat \qty{0.32}{mmol/L} zuurstof (\qty{10}{mg/L}, \qty{7}{mL/L} gas), tegen
\qty{8.7}{mmol/L} (\qty{210}{mL/L}) in de lucht zelf. De partiële druk --- in
evenwicht in het water en de lucht dezelfde --- is wat de diffusie tussen de media
aandrijft; het gehalte is wat een ademhaler eruit kan halen. Koolstofdioxide is
vijfentwintig keer beter oplosbaar dan zuurstof, en is dus nooit het gas dat het
moeilijkst is uit te wisselen.
\end{definition}

\begin{theorem}[De wet van Fick voor een uitwisselingsoppervlak]\label{thm:b1:gas-exchange:fick}
\index{wet van Fick}
De snelheid waarmee een gas een membraan met oppervlak $A$ en dikte $x$ passeert
die twee media met partiële drukken $P_1$ en $P_2$ scheidt, is
\[
\dot V = K\,\frac{A}{x}\,(P_1 - P_2),
\]
waarin $K$, de diffusieconstante (van Krogh) van de membraan, de diffusiviteit en
de oplosbaarheid van het gas in het \omterm{def:b1:body-plans-tissues:tissue}{weefsel} samenvat. De flux wordt verhoogd door
een groter oppervlak, een dunnere barrière en een groter verschil in \omterm{def:b1:gas-exchange:partial}{partiële druk};
door niets anders.
\end{theorem}
\begin{proof}
De eerste \omterm{thm:b1:membranes-transport:fick}{wet van Fick} voor het opgeloste gas in de membraan
(\cref{ch:b1:membranes-transport}) geeft een flux per eenheid oppervlak die
evenredig is met de concentratiegradiënt; volgens de wet van Henry is de
concentratie aan elke zijde evenredig met de \omterm{def:b1:gas-exchange:partial}{partiële druk} in het medium daar,
zodat de gradiënt $(P_1 - P_2)/x$ maal de oplosbaarheid is, en de constante de
oplosbaarheid en de diffusiviteit opneemt.
\end{proof}

\begin{proposition}[Wat een uitwisselingsoppervlak moet zijn]\label{prop:b1:gas-exchange:requirements}
\index{uitwisselingsoppervlak}
Een \omterm{def:b1:organism-environment:organism}{organisme} van meer dan een millimeter kan niet door diffusie via zijn
buitenoppervlak worden bevoorraad (\cref{ch:b1:organism-environment}). Het heeft
een \emph{ademhalingsoppervlak} nodig dat groot is (om $A$ te verhogen), dun (om
$x$ te verlagen), vochtig (gassen passeren membranen alleen in oplossing),
\emph{geventileerd} --- het buitenmidden wordt ververst zodat $P_1$ hoog blijft ---
en \emph{doorbloed} --- het bloed erachter wordt ververst zodat $P_2$ laag blijft;
en een bloedsomloop om het gas van het oppervlak naar de \omterm{def:b1:body-plans-tissues:tissue}{weefsels} te brengen.
\omterm{def:b1:gas-exchange:gill}{Kieuwen}, \omterm{def:b1:gas-exchange:lung}{longen} en tracheeën zijn drie manieren om zo'n oppervlak in een lichaam te
vouwen en de twee vloeistoffen eroverheen te bewegen.
\end{proposition}

\section{Kieuwen en de tegenstroom}

\begin{definition}[Kieuw]\label{def:b1:gas-exchange:gill}
\index{kieuw}
Een \emph{kieuw} is een uitstulping van het lichaamsoppervlak het water in,
gevouwen tot filamenten en \emph{lamellen} van enkele micrometers dik waar bloed in
haarvaten doorheen stroomt. De kieuwen van een vis hangen aan vier benige bogen aan
elke kant; water dat door de bek wordt binnengehaald en door een stelsel van twee
pompen onder de kieuwdeksels naar buiten wordt geduwd, stroomt in de ene richting
tussen de lamellen door, en het bloed in de lamellen stroomt de andere kant op: een
\emph{tegenstroom}\index{tegenstroomuitwisseling}. Een forel van een kilogram heeft
zo'n \qty{2000}{cm^2} lamellair oppervlak van \qty{2}{\micro m} dik.
\end{definition}

\begin{theorem}[Tegenstroomuitwisseling]\label{thm:b1:gas-exchange:countercurrent}
Wanneer water en bloed in tegengestelde richting langs een \omterm{prop:b1:organism-environment:surfaces}{uitwisselingsoppervlak}
stromen, kan het bloed dat de \omterm{def:b1:gas-exchange:gill}{kieuw} verlaat de \omterm{def:b1:gas-exchange:partial}{partiële druk} benaderen van het
water dat er \emph{binnenkomt}, en kan het water van vrijwel al zijn zuurstof
worden ontdaan: onttrekkingen van \qty{80}{\%} of meer. Stromen zij in dezelfde
richting (meestroom), dan naderen beide vloeistoffen een gemeenschappelijke
tussenwaarde en kan de onttrekking de \qty{50}{\%} niet overschrijden.
\end{theorem}
\begin{proof}
Langs een meestroomwisselaar daalt het verschil $P_{\text{water}} -
P_{\text{bloed}}$ van zijn waarde bij de ingang tot nul naarmate de twee dezelfde
druk naderen; zodra zij gelijk zijn beweegt er geen gas meer, en bij gelijke
stromen ligt het ontmoetingspunt halverwege. Bij \omterm{def:b1:gas-exchange:gill}{tegenstroom} komt het bloed,
terwijl het naar zijn uitgang beweegt, telkens vers water tegen, zodat er over de
hele lengte een positief verschil in stand blijft; het bloed aan zijn uitgang
staat tegenover het binnenkomende water van \qty{21}{kPa}, en het water aan zijn
uitgang tegenover het binnenkomende aderlijke bloed van enkele kilopascal. De grens
wordt door het oppervlak gesteld, niet door een evenwicht.
\end{proof}

\begin{example}[De prijs van water]\label{ex:b1:gas-exchange:priceofwater}
Een rustende forel van \qty{1}{kg} neemt ongeveer \qty{50}{mL} zuurstof per uur op.
Uit water met \qty{7}{mL/L}, bij \qty{80}{\%} onttrekking, moet zij per uur
\qty{9}{L} water over haar \omterm{def:b1:gas-exchange:gill}{kieuwen} pompen --- negen kilogram van een midden dat
vijftig keer stroperiger is dan lucht, door kanalen van een fractie van een
millimeter breed. Een tiende van haar stofwisseling gaat naar het ademen; een mens
in rust geeft er een vijftigste aan uit. Verwarm de vijver tot \qty{30}{\celsius} en
dezelfde vis, die meer zuurstof nodig heeft uit water dat er minder bevat, moet drie
keer zoveel pompen.
\end{example}

\section{Lucht ademen: longen en tracheeën}

\begin{definition}[De long van een zoogdier]\label{def:b1:gas-exchange:lung}
\index{long}\index{longblaasje}
Een \emph{long} is een instulping van het lichaamsoppervlak tot een vochtige
inwendige kamer, waardoor het oppervlak in droge lucht nat blijft. De long van een
zoogdier is een boom van luchtwegen die eindigt in zo'n driehonderd miljoen
\emph{longblaasjes}, zakjes van \qty{0.2}{mm} breed waarvan de wanden --- één
epitheelcel en één haarvatendotheel, samen \qty{0.5}{\micro m} --- bij een mens
\qty{100}{m^2} beslaan, gehuld in een haarvatennet waar het hele hartminuutvolume
doorheen gaat. Zij wordt \emph{getijdengewijs} geventileerd: het \omterm{def:b1:mammal-organization:bodyplan}{middenrif} en de
ribspieren maken de borstkas groter, er stroomt lucht in; zij ontspannen zich, en
de lucht stroomt langs dezelfde weg weer naar buiten. Eén ademteug verplaatst
\qty{500}{mL}, waarvan \qty{150}{mL} alleen de luchtwegen vult (de \emph{dode
ruimte}) en nooit een longblaasje bereikt; het gas in de longblaasjes, dat bij elke
ademteug voor een zevende wordt ververst, blijft op $P_{\mathrm{O_2}} =
\qty{13.3}{kPa}$ en $P_{\mathrm{CO_2}} = \qty{5.3}{kPa}$ --- de drukken waarmee het
bloed in evenwicht komt.
\end{definition}

\begin{proposition}[Drie ontwerpen voor lucht]\label{prop:b1:gas-exchange:designs}
\index{tracheeën}
\emph{Zoogdieren} ventileren getijdengewijs: eenvoudig, maar de verse lucht mengt
zich met de oude, de zuurstof in de \omterm{def:b1:gas-exchange:lung}{longblaasjes} is twee derde van die in de
atmosfeer, en de onttrekking is een kwart. \emph{Vogels} laten lucht in één
richting door fijne buisjes (parabronchi) stromen tussen twee stellen luchtzakken
die als blaasbalgen werken, zodat er zowel bij het inademen als bij het uitademen
verse lucht door het \omterm{prop:b1:organism-environment:surfaces}{uitwisselingsoppervlak} gaat, in dwarsstroom met het bloed: hun
onttrekking ligt hoger, en een indische gans vliegt over de Himalaya waar een
zoogdier zou bezwijmen. \emph{Insecten} hebben in het geheel geen ademhalend bloed:
de lucht komt door kleppen (\emph{stigma's}) de \emph{tracheeën} binnen, buizen die
door spiraalvormige verdikkingen worden opengehouden, die zich tot op een
micrometer vertakken en de zuurstof rechtstreeks door diffusie bij elke \omterm{def:b1:cell-unit-of-life:cell}{cel}
afleveren, bij grote of actieve insecten geholpen door het pompen van het
achterlijf. Diffusie door buizen beperkt dit ontwerp tot kleine lichamen --- wat
een van de redenen is dat geen insect groter is dan een muis.
\end{proposition}

\begin{method}[Een ventilatiebalans lezen]\label{met:b1:gas-exchange:budget}
\begin{enumerate}
  \item Minuutventilatie $=$ ademvolume $\times$ ademteugen per minuut;
        alveolaire ventilatie $=$ (ademvolume $-$ dode ruimte) $\times$
        ademteugen per minuut. Alleen de tweede wisselt gas uit.
  \item Zuurstofopname $=$ alveolaire ventilatie $\times$ (ingeademd
        aandeel $-$ alveolair aandeel); in rust
        $\qty{4.2}{L/min}\times(0.21 - 0.14) \approx \qty{0.3}{L/min}$.
  \item Onttrekking $=$ opname $/$ (ventilatie $\times$ ingeademd
        aandeel): een kwart bij een getijdenlong, vier vijfde bij een
        tegenstroomkieuw.
  \item Aan de bloedzijde: opname $=$ hartminuutvolume $\times$
        (slagaderlijk $-$ aderlijk gehalte); \qty{5}{L/min}$\times$
        \qty{50}{mL/L} in rust, dezelfde \qty{250}{mL/min}. De twee
        balansen moeten kloppen.
\end{enumerate}
\end{method}

\section{Gassen in het bloed}

\begin{proposition}[Het vervoer van zuurstof en koolstofdioxide]\label{prop:b1:gas-exchange:transport}
\index{koolzuuranhydrase}\index{bohreffect}\index{haldane-effect}
\omterm{def:b1:mammal-organization:compartments}{Plasma} lost bij slagaderlijke druk maar \qty{3}{mL} zuurstof per liter op;
hemoglobine (\cref{ch:b1:proteins}) draagt er \qty{200}{mL} van, laadt in de \omterm{def:b1:gas-exchange:lung}{long}
bij \qty{13.3}{kPa} en lost er in de \omterm{def:b1:body-plans-tissues:tissue}{weefsels} een vijfde tot de helft van.
Koolstofdioxide reist op drie manieren: \qty{7}{\%} opgelost, \qty{23}{\%} gebonden
aan de aminogroepen van hemoglobine, en \qty{70}{\%} als bicarbonaat: in de rode \omterm{def:b1:cell-unit-of-life:cell}{cel}
zet \emph{\omterm{prop:b1:gas-exchange:transport}{koolzuuranhydrase}} $\mathrm{CO_2}$ en water binnen milliseconden om in
koolzuur, dissocieert het \omterm{def:b1:water-small-molecules:ph}{zuur}, verlaat het bicarbonaat de \omterm{def:b1:cell-unit-of-life:cell}{cel} in ruil voor chloride
(de \emph{chlorideverschuiving}), en wordt het proton door hemoglobine opgenomen. De
twee gassen helpen elkaar: \omterm{def:b1:water-small-molecules:ph}{zuur} en $\mathrm{CO_2}$ verlagen de affiniteit van
hemoglobine voor zuurstof (het \emph{\omterm{prop:b1:gas-exchange:transport}{bohreffect}}), zodat er zuurstof vrijkomt waar
$\mathrm{CO_2}$ wordt gemaakt; en gedeoxygeneerd hemoglobine bindt protonen en
$\mathrm{CO_2}$ beter (het \emph{\omterm{prop:b1:gas-exchange:transport}{haldane-effect}}), zodat $\mathrm{CO_2}$ wordt
opgenomen waar zuurstof vrijkomt, en in de \omterm{def:b1:gas-exchange:lung}{long} wordt afgegeven zodra er zuurstof
laadt.
\end{proposition}

\begin{example}[Een liter bloed door een werkende spier]\label{ex:b1:gas-exchange:litre}
Slagaderlijk bloed brengt per liter \qty{200}{mL} zuurstof en \qty{480}{mL}
koolstofdioxide (grotendeels als bicarbonaat). In een rustende spier laat het
\qty{50}{mL} zuurstof achter en neemt het \qty{40}{mL} $\mathrm{CO_2}$ op; de
aderlijke pH daalt met slechts \qty{0.04}{}, omdat het hemoglobine de protonen
heeft opgenomen. In een sprintende spier, bij pH 7,2 en een
$P_{\mathrm{O_2}}$ van \qty{2.7}{kPa}, laat diezelfde liter \qty{150}{mL} zuurstof
achter (\cref{ch:b1:proteins}) en voert hij ook het lactaat af.
\end{example}

\section{Regeling, hoogte en duiken}

\begin{proposition}[De ventilatie wordt door koolstofdioxide geregeld]\label{prop:b1:gas-exchange:control}
\index{chemoreceptor}
Het ritme van de ademhaling ontstaat in de hersenstam en wordt bijgesteld door
\emph{chemoreceptoren}: centrale, die in het vocht rond de hersenen liggen,
reageren op de pH-verandering die $\mathrm{CO_2}$ teweegbrengt; perifere in de
halsslagaders reageren op een daling van de slagaderlijke zuurstof onder ongeveer
\qty{8}{kPa}. In het dagelijks leven heerst het $\mathrm{CO_2}$: een stijging van
\qty{1}{kPa} in de slagaderlijke $P_{\mathrm{CO_2}}$ verdubbelt de ventilatie
ruwweg, terwijl de zuurstof met een derde moet dalen voordat de ventilatie
reageert. Je adem inhouden houdt op wanneer het $\mathrm{CO_2}$, en niet de
zuurstof, de grens bereikt; eerst hyperventileren verlaagt het $\mathrm{CO_2}$ en
laat een duiker onder blijven tot de zuurstof op is --- bewusteloos voordat de
ademprikkel terugkomt.
\end{proposition}

\begin{example}[Hoogte en diepte]\label{ex:b1:gas-exchange:altitudediving}
Op \qty{4000}{m} bevat de lucht \qty{12}{kPa} zuurstof en de \omterm{def:b1:gas-exchange:lung}{longblaasjes}
\qty{7}{}: de halsslagaderlichaampjes drijven de ventilatie op, wat $\mathrm{CO_2}$
wegblaast en het bloed alkalisch maakt --- een rem die de nieren over dagen
opheffen door bicarbonaat uit te scheiden; de rode \omterm{def:b1:cell-unit-of-life:cell}{cellen} verhogen hun BPG binnen
dagen en het beenmerg het hemoglobine binnen weken (\cref{ch:b1:proteins}). Een
zeehond die twintig minuten duikt doet het omgekeerde: hij ademt vóór het duiken
uit, vertraagt zijn hart tot een tiende, sluit het bloed af van alles behalve
hersenen en hart, en laat zijn spieren lopen op de zuurstof die aan hun eigen
myoglobine is gebonden, dat tien keer zo geconcentreerd is als het onze, en daarna
op \omterm{def:b1:respiration-fermentation:fermentation}{gisting}, terwijl hij een lactaat verdraagt dat het lichaam aan de oppervlakte
opruimt. Zijn \omterm{def:b1:gas-exchange:lung}{longen} klappen onder \qty{50}{m} in, wat stikstof uit zijn bloed
houdt.
\end{example}

\section{Vraagstuk: een forel en een mens}

\begin{problem}[{Weekendvraagstuk --- een kieuw en een long naast elkaar gezet:
zuurstof per liter medium, ventilatie en onttrekking, tegenstroom en getijdenstroom,
en het bloed achter elk, eindigend op het onttrekkingsrendement van de
tegenstroomkieuw}]
\label{pb:b1:gas-exchange:1}
Een rustende mens neemt \qty{250}{mL} zuurstof per minuut op en ademt \qty{12}{}
keer per minuut met een ademvolume van \qty{500}{mL} en een dode ruimte van
\qty{150}{mL}; ingeademde lucht is voor \qty{21}{\%} zuurstof, het alveolaire gas
voor \qty{14}{\%}; het hartminuutvolume is \qty{5}{L/min} en het slagaderlijke
zuurstofgehalte \qty{200}{mL/L}. Een rustende forel van \qty{1}{kg} neemt
\qty{50}{mL} zuurstof per uur op in water van \qty{15}{\celsius} met \qty{7}{mL/L};
haar hemoglobine draagt bij verzadiging \qty{100}{mL/L}; haar bloed verlaat de
\omterm{def:b1:gas-exchange:gill}{kieuwen} voor \qty{95}{\%} verzadigd en keert voor \qty{15}{\%} verzadigd terug.
Water is \qty{800}{} keer dichter dan lucht.

\medskip
\noindent\textbf{Deel I --- Twee media.}
\begin{enumerate}
  \item Druk het zuurstofgehalte van lucht en van het water van de forel
        uit in millimol per liter (\qty{22.4}{L/mol}).
  \item Bereken de verhouding van de twee gehalten.
  \item Bereken de massa medium (lucht \qty{1.2}{g/L}; water
        \qty{1000}{g/L}) die in elk geval \qty{1}{mmol} zuurstof bevat.
  \item Bereken het volume water van de forel dat evenveel zuurstof bevat
        als één liter lucht.
  \item Leg in één zin uit waarom \omterm{def:b1:gas-exchange:gill}{kieuwen} van vissen in één richting
        moeten worden geventileerd en \omterm{def:b1:gas-exchange:lung}{longen} getijdengewijs mogen zijn.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De \omterm{def:b1:gas-exchange:lung}{long}.}
\begin{enumerate}[resume]
  \item Bereken de minuutventilatie en de alveolaire ventilatie.
  \item Welk deel van elke ademteug is dode ruimte, en welk deel van de
        minuutventilatie gaat daardoor verloren?
  \item Bereken de zuurstof die per minuut in de \omterm{def:b1:gas-exchange:lung}{longblaasjes} wordt
        afgeleverd en de zuurstof die wordt opgenomen (de alveolaire
        ventilatie $\times$ het verschil van de aandelen). Controleer dat
        tegen \qty{250}{mL/min}.
  \item Bereken het onttrekkingsrendement: de opname gedeeld door de
        zuurstof in de ingeademde minuutventilatie.
  \item Bereken de alveolaire $P_{\mathrm{O_2}}$ uit het alveolaire
        aandeel (totale druk \qty{101}{kPa}, waarvan eerst \qty{6.3}{kPa}
        waterdamp moet worden afgetrokken).
  \item Bereken aan de bloedzijde het arterioveneuze verschil dat voor
        \qty{250}{mL/min} bij \qty{5}{L/min} nodig is, en het aderlijke
        gehalte en de aderlijke verzadiging.
  \item Bij inspanning stijgt de opname tot \qty{3}{L/min} bij een
        hartminuutvolume van \qty{20}{L/min}. Bereken het arterioveneuze
        verschil en de aderlijke verzadiging.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De \omterm{def:b1:gas-exchange:gill}{kieuw}.}
\begin{enumerate}[resume]
  \item Bereken de zuurstofopname van de forel in milliliter per minuut.
  \item Bereken bij \qty{80}{\%} onttrekking (\omterm{def:b1:gas-exchange:gill}{tegenstroom}) het water dat
        per minuut en per uur wordt geventileerd.
  \item Bereken het benodigde water bij de onttrekking die een
        meestroomwisselaar met hetzelfde oppervlak zou geven
        (\qty{43}{\%}). Wat bespaart de \omterm{def:b1:gas-exchange:gill}{tegenstroom}?
  \item Bereken de massa water die per uur wordt gepompt, en vergelijk die
        met de massa lucht die een mens per uur verplaatst.
  \item Bereken het arterioveneuze verschil in zuurstofgehalte van de forel
        en het hartminuutvolume dat haar opname vereist.
  \item Bereken de verhouding van waterstroom tot bloedstroom bij de
        \omterm{def:b1:gas-exchange:gill}{kieuw}, en de verhouding van luchtstroom tot bloedstroom bij de
        menselijke \omterm{def:b1:gas-exchange:lung}{long} (alveolaire ventilatie gedeeld door
        hartminuutvolume). Becommentarieer dat.
  \item Het pompen van water kost de forel ongeveer \qty{10}{\%} van haar
        zuurstofopname. Als het water opwarmt tot \qty{25}{\celsius}
        (\qty{5.5}{mL/L}) en haar opname met de helft stijgt, met welke
        factor moet de ventilatie dan stijgen, en wat gebeurt er met het
        deel dat aan het ademen wordt besteed?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De wisselaar zelf.}
\begin{enumerate}[resume]
  \item In de tegenstroomkieuw verlaat het bloed de \omterm{def:b1:gas-exchange:gill}{kieuw} bij
        \qty{18}{kPa} tegenover binnenkomend water van \qty{21}{kPa}, en
        verlaat het water haar bij \qty{4}{kPa} tegenover binnenkomend
        bloed van \qty{3}{kPa}. Bereken het drukverschil aan elk uiteinde.
  \item In een meestroomwisselaar vertrekken beide vloeistoffen bij
        \qty{12}{kPa}. Bereken het verschil aan elk uiteinde en leg uit
        waarom de uitwisseling stopt voordat het water is opgebruikt.
  \item De menselijke \omterm{def:b1:gas-exchange:lung}{long}: alveolair \qty{13.3}{kPa}, aderlijk bloed dat
        bij \qty{5.3}{kPa} aankomt, slagaderlijk bloed dat bij
        \qty{13.3}{kPa} vertrekt. Op welke opstelling lijkt een
        getijdenlong, en waarom kan haar slagaderlijke bloed de alveolaire
        waarde niet overtreffen?
  \item De lamellen van een forel beslaan samen \qty{2000}{cm^2} bij
        \qty{2}{\micro m} dikte; de \omterm{def:b1:gas-exchange:lung}{longblaasjes} van een mens
        \qty{100}{m^2} bij \qty{0.5}{\micro m}. Bereken de verhouding
        $A/x$ van beide, en de verhouding van hun zuurstofopnamen. Wat
        suggereert die vergelijking over het gemiddelde drukverschil over
        een lamel ten opzichte van een longblaasjeswand?
  \item Leg uit waarom koolstofdioxide voor een vis nooit beperkend is,
        hoewel water zo weinig zuurstof bevat.
  \item Geef het resultaat: het onttrekkingsrendement van de \omterm{def:b1:gas-exchange:gill}{kieuw} van de
        forel en van de \omterm{def:b1:gas-exchange:lung}{long} van de mens, en de twee kenmerken van de
        \omterm{def:b1:gas-exchange:gill}{kieuw} die het verschil maken.
\end{enumerate}
\end{problem}
