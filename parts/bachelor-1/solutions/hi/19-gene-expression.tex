\section*{अध्याय \ref{ch:b1:gene-expression} --- जीन अभिव्यक्ति: अनुलेखन और अनुवादन}

\omterm{def:b1:nucleic-acids:chain}{RNA} $5'$-AUGCCUAAG-$3'$; कूटलेखी लड़ी $5'$-ATGCCTAAG-$3'$।
\end{solution}

\begin{solution}{exo:b1:gene-expression:2}
एक $5'$ टोपी (रूपांतरित ग्वानीन), एक $3'$ पॉली-A पूँछ, और
अंतर्वेशियों का विभाजन द्वारा हटाया जाना।
\end{solution}

\begin{solution}{exo:b1:gene-expression:3}
AUG CCG AAA GUU UGA: मेट-प्रो-लिस-वैल, फिर विराम।
\end{solution}

\begin{solution}{exo:b1:gene-expression:4}
A: आती हुई आवेशित tRNA भीतर ली और जाँची जाती है। P: बढ़ती शृंखला ढोती
tRNA; बंध उसकी शृंखला और A-स्थल के \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} के बीच बनता है। E: खाली हुई
tRNA अपने निकलने के रास्ते पर।
\end{solution}

\begin{solution}{exo:b1:gene-expression:5}
कूटलेखी $2400 - 600 = 1800$ \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड}: यानी 599 अवशेष (600 कूटशब्द,
जिनमें एक विराम)। समय $599/4 = \qty{150}{s}$। एक साथ \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} $1800/90 = 20$।
\end{solution}

\begin{solution}{exo:b1:gene-expression:6}
संदेश किसी नियत आरंभ से तीन-तीन में पढ़ा जाता है; एक या दो अतिरिक्त
क्षार आगे के हर कूटशब्द को खिसका देते हैं और \omterm{def:b1:proteins:peptide}{प्रोटीन} का बाकी हिस्सा
बकवास हो जाता है, जो आम तौर पर किसी समय-पूर्व विराम पर खत्म होता है।
तीन अतिरिक्त क्षारक एक कूटशब्द जोड़ते हैं और फ्रेम बहाल कर देते हैं:
\omterm{def:b1:proteins:peptide}{प्रोटीन} में एक अतिरिक्त अवशेष और प्रवेशनों के बीच कुछ गलत अवशेष रहते हैं,
और वह अक्सर तब भी काम करता है।
\end{solution}

\begin{solution}{exo:b1:gene-expression:7}
CAG (ग्लन) से UAG (विराम): शृंखला अवशेष 150 पर खत्म हो जाती है, यानी एक
कटा-छँटा, लगभग निश्चित रूप से निष्क्रिय \omterm{def:b1:proteins:peptide}{प्रोटीन} (निरर्थक उत्परिवर्तन)।
CAG से CAA: तब भी ग्लन, मौन। CAG से CGG: ग्लन की जगह आर्ज, यानी एक
प्रतिस्थापन (कुअर्थक) जिसका असर स्थिति पर निर्भर करता है।
\end{solution}

\begin{solution}{exo:b1:gene-expression:8}
\omterm{def:b1:gene-expression:ribosome}{राइबोसोम} केवल कूटशब्द--प्रतिकूटशब्द का युग्मन जाँचता है; वह \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल}
देख ही नहीं सकता। यदि अपनी tRNA पर बैठी सिस्टीन रासायनिक ढंग से
ऐलानीन में बदल दी जाए, तो \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} वहीं ऐलानीन डाल देता है जहाँ संदेश
सिस्टीन कहता है: यानी tRNA पढ़ी जाती है, \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} नहीं। इसलिए सिंथेटेज़,
जो हर \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} को सही tRNA के साथ जोड़ती है, ही असली
अनुवादक है।
\end{solution}

\begin{solution}{exo:b1:gene-expression:9}
$400\times 4 = 1600$ बंध; इनमें \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} के प्रति अवशेष दो GTP आधे हैं,
और सिंथेटेज़ों के दो बाकी आधे।
\end{solution}

\begin{solution}{exo:b1:gene-expression:10}
\omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकियों} में अनुलेख केंद्रक में बनता है और
\omterm{def:b1:gene-expression:ribosome}{राइबोसोम} \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाद्रव्य} में होते हैं, जिन्हें आवरण अलग करता है; और
अनुलेख तब तक संदेशवाहक नहीं होता जब तक उसका विभाजन तथा टोपी न हो जाए।
यह अलगाव उस प्रसंस्करण को ही संभव बनाता है --- वैकल्पिक
विभाजन, निर्यात का नियंत्रण, और यह जाँच कि पढ़े जाने से पहले संदेश
पूरा है।
\end{solution}

\begin{solution}{exo:b1:gene-expression:11}
चार संदेशवाहक: दोनों बहिर्वेशियों के साथ (190 \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} जुड़े), केवल
3 के साथ (90), केवल 4 के साथ (100), किसी के बिना (0)। फ्रेम तब बचता है
जब जुड़ी लंबाई 3 का गुणज हो: 0 और 90 उसे बचाते हैं; 100 और 190 उसे
खिसका देते हैं, इसलिए अकेले \omterm{def:b1:genomes:gene}{एक्सॉन} 4 वाले या दोनों वाले संदेशवाहक
\omterm{def:b1:genomes:gene}{एक्सॉन} 5 को फ्रेम से बाहर पढ़ते हैं और \omterm{def:b1:proteins:peptide}{प्रोटीन} काट देते हैं। वैकल्पिक
विभाजन को फ्रेम का आदर करना ही पड़ता है, और इसी कारण \omterm{def:b1:genomes:gene}{एक्सॉन} अक्सर तीन
के गुणज होते हैं।
\end{solution}

\begin{solution}{exo:b1:gene-expression:12}
मनमाना: रसायन में ऐसा कुछ नहीं जो कहे कि UUU का अर्थ फ़ेन ही हो;
ये नियुक्तियाँ सिंथेटेज़ों की थामी हुई रूढ़ियाँ हैं। अनुकूलित: तीसरी
स्थिति सबसे अधिक अपह्रासी है, इसलिए जो गलतियाँ और उत्परिवर्तन वहाँ पड़ते
हैं वे अक्सर मौन होते हैं, और जो कूटशब्द एक क्षारक से भिन्न होते हैं वे
प्रायः मिलते-जुलते \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} कूटबद्ध करते हैं, इसलिए कोई प्रतिस्थापन
अक्सर हलका रहता है --- कूट गलती की हानि न्यूनतम कर देता है। जमा हुआ: जो
सिंथेटेज़ अपनी विशिष्टता बदले वह \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के हर \omterm{def:b1:proteins:peptide}{प्रोटीन} को, हज़ारों को,
उसी क्षण बदल देती; कोई \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} इसे शायद ही सह पाती है, इसलिए कूट बह नहीं
सकता और सारे जीवधारियों के साझे पूर्वज से नियत बना हुआ है।
\end{solution}

\begin{solution}{pb:b1:gene-expression:1}
\textbf{1.} $30\,000/30 = \qty{1000}{s}$, यानी लगभग \qty{17}{min}।
\textbf{2.} $28\,000/30\,000 = \qty{93}{\%}$।
\textbf{3.} $30\,000/2000 = 15$।
\textbf{4.} $2\times 30\,000 = \num{60000}$ \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{5.} \omterm{def:b1:genomes:gene}{जीन} पर $1000/10 = 100$ पॉलिमरेज़; प्रति घंटा 360 संदेशवाहक।
\textbf{6.} सात \omterm{def:b1:genomes:gene}{इंट्रॉन}, माध्य $28\,000/7 = \qty{4000}{}$ \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड}।
\textbf{7.} अनुलेखन (\qty{17}{min}); \omterm{def:b1:genomes:gene}{इंट्रॉन} बनते-बनते ही निकाल दिए
जाते हैं, और आखिरी केवल एक मिनट जोड़ता है।
\textbf{8.} \omterm{def:b1:genomes:gene}{जीन} $30\,000\times 0.34\,\mathrm{nm} = \qty{10}{\micro m}$; संदेशवाहक \qty{0.68}{\micro m}।
\textbf{9.} $500/4 = \qty{125}{s}$।
\textbf{10.} $2000/80 = 25$।
\textbf{11.} हर $80/4 = \qty{20}{s}$ में एक शृंखला पूरी होती है: यानी
प्रति घंटा 180.
\textbf{12.} पूरे उत्पादन के $2/\ln 2 = \qty{2.9}{h}$: यानी लगभग 520
\omterm{def:b1:proteins:peptide}{प्रोटीन}।
\textbf{13.} $4\times 500 = 2000$ \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{14.} प्रति \omterm{def:b1:proteins:peptide}{प्रोटीन} $60\,000/520 = 115$ \omterm{def:b1:water-small-molecules:atp}{ATP}.
\textbf{15.} लगभग 2100 \omterm{def:b1:water-small-molecules:atp}{ATP}, जिसका \qty{95}{\%} अनुवादन में।
\textbf{16.} $1 - (1 - 10^{-5})^{1503} \approx 1503\times 10^{-5} =
\qty{1.5}{\%}$।
\textbf{17.} $1 - (1 - 10^{-4})^{500} \approx \qty{5}{\%}$।
\textbf{18.} अनुलेखन: $1.5\%\times 0.75\times 0.67 \approx
0.75\%$ संदेशवाहक दोषपूर्ण, और इसलिए उतने ही \omterm{def:b1:proteins:peptide}{प्रोटीन}; अनुवादन:
$5\%\times 0.67 = 3.3\%$; यानी अणुओं का लगभग \qty{4}{\%}।
\textbf{19.} \omterm{def:b1:proteins:peptide}{प्रोटीन} की कोई गलती एक ही अणु तक सीमित रहती है, जो
जल्दी ही नष्ट हो जाता है; संदेशवाहक की गलती सैकड़ों \omterm{def:b1:proteins:peptide}{प्रोटीनों} में नकल हो
जाती है; और प्रतिकृतिकरण की गलती हर वंशज को सदा के लिए वंशागत होती है।
किसी गलती की लागत, और इसलिए वह शुद्धता जिसकी कीमत चुकाना बनता है, हर
पिछले चरण पर चढ़ती जाती है।
\textbf{20.} उपस्थित संदेशवाहक $3000/180 = 17$।
\textbf{21.} $17\times 25 = 420$ \omterm{def:b1:gene-expression:ribosome}{राइबोसोम}।
\textbf{22.} प्रति दिन $\num{3e9}\times 4 = \num{1.2e10}$ \omterm{def:b1:water-small-molecules:atp}{ATP}
$= \qty{2e-14}{mol}$, यानी प्रति दिन \qty{1e-9}{J}, यानी \qty{1.2e-14}{W}
--- यानी प्रति घन माइक्रोमीटर \qty{6}{fW}।
\textbf{23.} प्रति सेकंड $10^9$ \omterm{def:b1:water-small-molecules:atp}{ATP} यानी प्रति दिन $\num{8.6e13}$:
इस धीरे नवीकृत होती \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में प्रोटीन-संश्लेषण उसका सौवाँ भाग है (किसी
विभाजित होते बैक्टीरियम में वह आधा है)।
\textbf{24.} कोई परिपक्व संदेशवाहक नहीं: प्राथमिक अनुलेख केंद्रक में
जमा हो जाते हैं और \omterm{def:b1:proteins:peptide}{प्रोटीन} अपने संदेशवाहकों के क्षय होते ही घंटों के
भीतर लुप्त हो जाता है। बैक्टीरिया का \omterm{def:b1:proteins:peptide}{प्रोटीन}, जिसके \omterm{def:b1:genomes:gene}{जीन} में कोई \omterm{def:b1:genomes:gene}{इंट्रॉन}
नहीं, अछूता रहता है।
\textbf{25.} प्रति अणु लगभग 2100 \omterm{def:b1:water-small-molecules:atp}{ATP} --- अनुवादन के लिए 2000, और
अनुलेख में उसके हिस्से के लिए सौ; पहला \omterm{def:b1:proteins:peptide}{प्रोटीन} \qty{1000}{s} के अनुलेखन,
एक मिनट के प्रसंस्करण तथा निर्यात, और \qty{125}{s} के अनुवादन के बाद:
यानी लगभग बीस मिनट।
\end{solution}
