\section*{Chapitre \ref{ch:g6:exploring-our-environment} --- Explorer notre environnement}

\begin{solution}{exo:g6:exploring-our-environment:1}
Des \omterm{def:g6:exploring-our-environment:components}{êtres vivants} (mousse, cloportes)~; de la \omterm{def:g6:exploring-our-environment:components}{matière minérale} (la
pierre du mur, la flaque)~; des traces et des restes d'\omterm{def:g6:exploring-our-environment:components}{êtres vivants}
(\omterm{def:g1:plants-around-us:parts}{feuilles} mortes, \omterm{def:g1:animals-around-us:covering}{coquille} d'escargot).
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:2}
La \omterm{def:g6:exploring-our-environment:conditions}{température}, au thermomètre (en degrés Celsius)~; l'\omterm{def:g6:exploring-our-environment:conditions}{humidité} de
l'air et du sol~; la \omterm{def:g6:exploring-our-environment:conditions}{lumière}, du plein soleil à l'ombre profonde.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:3}
Vivants~: la fougère, le mille-pattes. Minéral~: la flaque, la
pierre du mur. Traces~: la \omterm{def:g1:animals-around-us:covering}{coquille} d'escargot, les fientes
d'oiseau.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:4}
Seize degrés (\qty{38}{\degreeCelsius} contre
\qty{22}{\degreeCelsius}) et plein soleil contre ombre profonde (sec
contre humide, également). L'herbe folle de la sécheresse et de
l'éblouissement correspond au pied sud~; mousse, \omterm{prop:g4:plants-without-seeds:damp}{fougères} et
cloportes correspondent au coin frais et humide.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:5}
Son revêtement laisse l'eau s'échapper~: en plein soleil sec, son
corps perd son eau à une vitesse mortelle. Voilà pourquoi on trouve
les cloportes dans des \omterm{def:g6:exploring-our-environment:microhabitat}{micro-habitats} humides et sombres --- sous
les pierres, sous le bois mort.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:6}
Par exemple~: une noisette grignotée --- le mulot~; une toile
d'araignée --- son araignée~; des empreintes dans la vase --- le
héron (crottes, \omterm{def:g1:animals-around-us:covering}{plumes} et \omterm{def:g1:animals-around-us:covering}{coquilles} servent de la même façon).
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:7}
Une boîte fermée, moitié papier humide et moitié papier sec, des
cloportes lâchés le long de la ligne médiane~; quelques minutes plus
tard, presque tous sont du côté humide (de même pour l'ombre contre
la \omterm{def:g6:exploring-our-environment:conditions}{lumière}). Cela montre que chaque \omterm{def:g1:animals-around-us:animal}{animal} s'installe là où les
\omterm{def:g6:exploring-our-environment:conditions}{conditions} conviennent à son corps --- la carte du groupe se dessine
sans que personne ne le rassemble.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:8}
Parce que seules des mesures faites de la même façon se comparent~:
une différence doit venir des sites, non de la mesure. C'est la
version de terrain de l'expérience honnête --- une seule chose
varie (le site), tout le reste est tenu égal.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:9}
Dessus de la pierre~: éclairé, sec, chaud --- du lichen (avec la
mouche au soleil comme visiteuse). Dessous~: sombre, humide, frais
--- des cloportes (ou le mille-pattes). À cinq centimètres l'un de
l'autre, deux jeux d'habitants assortis.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:10}
Étape 1~: délimiter un mètre carré précis. Étape 2~: mesurer --- la
lecture du thermomètre notée, l'\omterm{def:g6:exploring-our-environment:conditions}{humidité} de la terre tâtée et notée,
la classe de \omterm{def:g6:exploring-our-environment:conditions}{lumière} notée. Étape 3~: chercher correctement, pierres
soulevées et remises, les trois sortes de composantes listées. Étape
4~: noter sur place --- des comptages, pas «~quelques~». Étape 5~:
aucune comparaison n'était même possible avec un seul site~; relever
un second site de la même façon.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:11}
Explication~: le lierre prospère dans l'intervalle de \omterm{def:g6:exploring-our-environment:conditions}{conditions} du
mur nord (ombre, \omterm{def:g6:exploring-our-environment:conditions}{humidité}) et non dans celui du mur sud
(éblouissement, sécheresse). Test~: mesurer les deux pieds de mur
--- \omterm{def:g6:exploring-our-environment:conditions}{température}, \omterm{def:g6:exploring-our-environment:conditions}{humidité}, \omterm{def:g6:exploring-our-environment:conditions}{lumière} --- et confronter aux préférences
connues du lierre~; ou chercher du lierre sur d'autres murs et noter
l'orientation qu'ils ont en commun.
\end{solution}

\begin{solution}{exo:g6:exploring-our-environment:12}
Supprimer la différence de paroi~: mener le choix dans une boîte
ronde (la même paroi partout), en échangeant les moitiés humide et
sèche entre les essais --- humide à l'est au premier essai, humide à
l'ouest au second. Si les cloportes suivent l'\omterm{def:g6:exploring-our-environment:conditions}{humidité} où qu'on la
mette, les parois sont écartées~; seule la condition testée a changé
entre les essais, tout le reste étant tenu égal.
\end{solution}

\begin{solution}{pb:g6:exploring-our-environment:1}
\textbf{1.} La haie haute~: elle ombrage le fossé N (et l'abrite),
alors que le fossé S est exposé au soleil toute la journée.

\textbf{2.} \omterm{def:g6:exploring-our-environment:conditions}{Température}~: \qty{19}{\degreeCelsius} contre
\qty{31}{\degreeCelsius}. \omterm{def:g6:exploring-our-environment:conditions}{Humidité}~: la terre du fossé N est humide
(elle colle), celle du fossé S est sèche (poussière). La \omterm{def:g6:exploring-our-environment:conditions}{lumière}
complète le trio~: ombre contre plein soleil.

\textbf{3.} Les \omterm{def:g6:exploring-our-environment:conditions}{conditions} changent au fil de la journée~: à 18
heures, tout site est plus frais et moins éclairé, si bien que
l'écart mesuré mêlerait le site et l'heure. La même heure garde la
comparaison honnête.

\textbf{4.} Pour que l'effort de recherche soit égal~: plus de
surface ou plus de temps sur un site y ferait trouver davantage
d'habitants --- un test malhonnête de la richesse des fossés.

\textbf{5.} Les deux listes ne nomment que des \omterm{def:g6:exploring-our-environment:components}{êtres vivants}. Il
manque~: les composantes minérales (terre, pierres, eau) et les
traces --- que la partie inventaire d'un relevé correct doit aussi
consigner.

\textbf{6.} Cloportes~: fossé N --- il leur faut de l'\omterm{def:g6:exploring-our-environment:conditions}{humidité} et de
l'ombre, sans quoi ils se dessèchent. Le crapaud~: fossé N --- une
\omterm{def:g1:the-five-senses:sense}{peau} humide, qui réclame un abri humide. Le lézard~: fossé S --- il
réchauffe son corps au soleil et chasse là où il peut se chauffer.

\textbf{7.} Ou bien des escargots vivent là, cachés (peut-être
actifs seulement pendant les heures humides de la nuit), ou bien la
\omterm{def:g1:animals-around-us:covering}{coquille} est un reste venu d'ailleurs ou d'une autre saison --- une
trace prouve une présence \emph{à un moment}, pas une résidence
actuelle.

\textbf{8.} Sombre, humide, frais. La règle de
\cref{prop:g6:exploring-our-environment:decide}~: chaque endroit est
habité par les \omterm{def:g5:biodiversity:species}{espèces} dont les intervalles correspondent à ses
\omterm{def:g6:exploring-our-environment:conditions}{conditions}.

\textbf{9.} N a de la mousse et des crapauds parce que son ombre le
garde frais et humide, ce que leur corps exige~; S a des lézards et
des \omterm{def:g1:plants-around-us:plant}{plantes} à \omterm{def:g1:plants-around-us:parts}{feuilles} cireuses parce que son plein soleil le rend
chaud et sec, ce qui convient à un chasseur qui se chauffe et à une
\omterm{def:g1:plants-around-us:plant}{plante} à l'épreuve de la sécheresse.

\textbf{10.} Sans la haie, le fossé N reçoit le plein soleil~: plus
chaud, plus sec --- des \omterm{def:g6:exploring-our-environment:conditions}{conditions} qui dérivent vers celles du fossé
S d'aujourd'hui. Attends-toi à voir s'amenuiser les amateurs
d'\omterm{def:g6:exploring-our-environment:conditions}{humidité} (mousse, \omterm{prop:g4:plants-without-seeds:damp}{fougères}, cloportes, le crapaud) et s'installer
les amateurs de soleil (herbes sèches, sauterelles, peut-être le
lézard).

\textbf{11.} Parce que le nouveau relevé ne doit différer du premier
que par une seule chose --- l'absence de la haie. De nouveaux sites,
instruments ou saisons ajouteraient leurs propres différences, et le
changement ne pourrait plus être attribué à la haie~: la règle de
l'expérience honnête à l'échelle d'un paysage.

\textbf{12.} Chaque cloporte erre et tourne jusqu'à ce que son corps
soit à l'aise, si bien que le groupe finit rassemblé là où est
l'\omterm{def:g6:exploring-our-environment:conditions}{humidité} --- les positions des \omterm{def:g1:animals-around-us:animal}{animaux} enregistrent les
\omterm{def:g6:exploring-our-environment:conditions}{conditions}, aussi sûrement que les lectures d'un thermomètre.
Personne ne les rassemble~; c'est le confort de chaque individu qui
dessine la carte, et c'est exactement pourquoi les habitants
peuvent se lire comme des mesures.
\end{solution}
