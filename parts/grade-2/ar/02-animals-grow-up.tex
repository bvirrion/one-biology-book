\chapter{كيف تكبر الحيوانات}\label{ch:g2:animals-grow-up}

الهر الصغير يشبه قطة صغيرة منذ يومه الأول. أما الضفدع في البركة
فقد بدأ شرغوفًا على شكل سمكة بلا \omterm{def:g1:my-body:parts}{أرجل}، وأما الفراشة على شجيرة
الفراشات فقد قضت أسابيع يرقةً زاحفة. صغار \omterm{def:g1:animals-around-us:animal}{الحيوانات} لا تشبه
والديها كلها --- فبعضها عليه أن يتحول تحولًا كاملًا في طريقه إلى
البلوغ.

\section{من البيضة أو ولادةً حية}

\begin{definition}[طريقتان للولادة]\label{def:g2:animals-grow-up:born}
تأتي صغار \omterm{def:g1:animals-around-us:animal}{الحيوانات} إلى العالم بطريقتين:
\begin{itemize}
  \item أكثرها يفقس من \emph{بيضة}\index{بيضة} تضعها الأم: الطيور،
        والأسماك، والضفادع، والحلزونات، والحشرات؛
  \item وغيرها \emph{يولد حيًّا}\index{ولادة حية}، فيخرج من جسم
        الأم مكتمل التكوين، ويشرب \emph{حليبها}\index{حليب}:
        القطط، والكلاب، والأبقار، والأرانب --- والبشر.
\end{itemize}
\end{definition}

\begin{example}[في قن الدجاج وفي السلة]\label{ex:g2:animals-grow-up:henhouse}
تضع الدجاجة بيضها وترقد عليه لتبقيه دافئًا؛ وبعد ثلاثة أسابيع
تكسر الكتاكيت القشرة وتفقس. وفي السلة قرب المدفأة، وُلدت هريرات
القطة حية في الليلة نفسها، فوجدت حليبها في غضون ساعة.
\end{example}

\begin{remark}[البيضة ليست حجرًا]\label{rem:g2:animals-grow-up:egg}
داخل \omterm{def:g2:animals-grow-up:born}{البيضة} المرقود عليها يُبنى كتكوت في هدوء، ويغذّيه المحّ
الأصفر. \omterm{def:g2:animals-grow-up:born}{فالبيضة} لا تنتظر --- بل تعمل. ولهذا تبقي الدجاجة بيضها
دافئًا، ولهذا لا يُلام البرد على \omterm{def:g2:animals-grow-up:born}{بيضة} من الثلاجة لم تكن ستفقس على
كل حال.
\end{remark}

\section{النمو من غير تحول}

\begin{definition}[النمو على هيئة الوالدين]\label{def:g2:animals-grow-up:direct}
كثير من صغار \omterm{def:g1:animals-around-us:animal}{الحيوانات} \emph{نسخ مصغرة من
والديها}\index{نسخة مصغرة من الوالدين}: فالهر الصغير قطة صغيرة،
والمهر حصان صغير، والكتكوت دجاجة صغيرة. والكبر عندها يعني ازدياد
الحجم والقوة --- أما شكل الجسم فيبقى كما هو.
\end{definition}

\begin{example}[من هر صغير إلى قطة]\label{ex:g2:animals-grow-up:kitten}
الهر المولود حديثًا يشرب \omterm{def:g2:animals-grow-up:born}{الحليب} وعيناه مغمضتان. وفي الشهر الأول
يلعب وينقضّ؛ وفي الشهر السادس يصطاد كقطة كبيرة؛ وفي السنة يصير
بالغًا، وربما صارت له هريرات. قطة صغيرة، فقطة أكبر، فقطة: الشكل
نفسه طول الطريق.
\end{example}

\section{النمو بالتحول}

\begin{definition}[التحول]\label{def:g2:animals-grow-up:metamorphosis}
بعض صغار \omterm{def:g1:animals-around-us:animal}{الحيوانات} لا يشبه والديه في شيء، وعليه أن يغير شكل جسمه
كله ليصير بالغًا. ويسمى هذا التغير الكبير
\emph{التحول}\index{تحول}. فالشرغوف يصير ضفدعًا؛ \omterm{ex:g2:animals-grow-up:butterfly}{واليرقة} تصير
فراشة.
\end{definition}

\begin{example}[طريق الضفدع]\label{ex:g2:animals-grow-up:frog}
من \omterm{def:g2:animals-grow-up:born}{بيض} الضفدع في البركة تفقس \emph{شراغيف}\index{شرغوف}: مستديرة
الرؤوس، طويلة الأذناب، بلا \omterm{def:g1:my-body:parts}{أرجل}، تتنفس في الماء كالأسماك. ثم،
أسبوعًا بعد أسبوع: تظهر \omterm{def:g1:my-body:parts}{الرجلان} الخلفيتان، ثم الأماميتان؛ ويضمر
الذنب حتى يزول؛ وفي يوم من الأيام يتسلق ضفدع صغير إلى الضفة.
\omterm{def:g1:animals-around-us:animal}{الحيوان} نفسه --- وقد أُعيد بناؤه.
\end{example}

\begin{example}[طريق الفراشة]\label{ex:g2:animals-grow-up:butterfly}
من \omterm{def:g2:animals-grow-up:born}{بيضة} الفراشة تفقس \emph{يرقة}\index{يرقة} لا تفعل شيئًا يُذكر
إلا أن تأكل \omterm{def:g1:plants-around-us:parts}{الأوراق} وتنمو. وفي يوم من الأيام تلفّ نفسها في غلاف هو
\emph{الشرنقة}\index{شرنقة}، وتكفّ عن الحركة. وفي الداخل يُعاد بناء
الجسم؛ وبعد أسابيع ينشقّ الغلاف، فتخرج منه فراشة تنشر جناحين
رطبين جديدين. \omterm{def:g2:animals-grow-up:born}{بيضة}، \omterm{ex:g2:animals-grow-up:butterfly}{فيرقة}، فشرنقة، ففراشة: أربع مراحل \omterm{def:g1:animals-around-us:animal}{لحيوان} واحد
بعينه.
\end{example}

\begin{method}[قراءة دورة حياة حيوان]\label{met:g2:animals-grow-up:read}
لكل \omterm{def:g1:animals-around-us:animal}{حيوان} تلقاه:
\begin{enumerate}
  \item اسأل كيف يولد: من \omterm{def:g2:animals-grow-up:born}{بيضة}، أم ولادةً حية؟
  \item واسأل كيف يبدو الصغير: أنسخة مصغرة من والديه، أم شكل
        مختلف؟
  \item فإن كان شكله مختلفًا فسيتحول: فابحث عن مراحل تحوله؛
  \item وارسم الدورة عجلةً، وفيها سهم من البالغ عائدًا إلى \omterm{def:g2:animals-grow-up:born}{البيض}.
\end{enumerate}
\end{method}

\begin{remark}[حليب، أو عمل من أول يوم]\label{rem:g2:animals-grow-up:care}
الهريرات تُطعمها أمها وتغسلها أسابيع؛ أما الشرغوف الحديث الفقس فلا
يجد عونًا البتة وعليه أن يطعم نفسه من اليوم الأول. فبعض صغار
\omterm{def:g1:animals-around-us:animal}{الحيوانات} يُعتنى به، وبعضها يبدأ حياته وحيدًا --- وكلا الطريقين
يؤدي إلى البلوغ.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:1}
اذكر الطريقتين اللتين تأتي بهما صغار \omterm{def:g1:animals-around-us:animal}{الحيوانات} إلى العالم، مع
مثال على كل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:2}
من \omterm{def:g2:animals-grow-up:born}{بيضة} أم ولادةً حية: (أ) كتكوت؟ (ب) هر صغير؟ (ج) ضفدع؟ (د)
أرنب؟ (ه) حلزون؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:3}
ماذا يشرب الهر الصغير أول ما يشرب؟ وأي \omterm{def:g1:animals-around-us:animal}{حيوانات} هذا الفصل تشرب
مثله عند \omterm{def:g2:born-grow-die:cycle}{الولادة}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:4}
ما \omterm{def:g2:animals-grow-up:metamorphosis}{التحول}؟ وسمِّ المثالين المشهورين في هذا الفصل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:5}
رتّب مراحل الضفدع: ضفدع --- \omterm{def:g2:animals-grow-up:born}{بيض} --- شرغوف \omterm{def:g1:my-body:parts}{بأرجل} --- شرغوف.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:6}
رتّب مراحل الفراشة: فراشة --- \omterm{ex:g2:animals-grow-up:butterfly}{يرقة} --- \omterm{def:g2:animals-grow-up:born}{بيضة} --- شرنقة. وفي أي
مرحلة يأكل \omterm{def:g1:animals-around-us:animal}{الحيوان} \omterm{def:g1:plants-around-us:parts}{الأوراق} وينمو أكثر ما ينمو؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:7}
أي الصغيرين نسخة مصغرة من والديه: الشرغوف أم المهر؟ وما معنى
الكبر عنده؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:8}
لماذا ترقد الدجاجة على بيضها؟ وماذا يجري داخله وفق
\cref{rem:g2:animals-grow-up:egg}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:animals-grow-up:9}
في البركة تجد زوي حيوانًا صغيرًا مستدير الرأس، طويل الذنب، له
\omterm{def:g1:my-body:parts}{رجلان} خلفيتان ولا \omterm{def:g1:my-body:parts}{رجلين} أماميتين. مستعينًا بشكل دورة الضفدع، اذكر
أين هو بالضبط من طريقه، وما كان قبل ذلك، وما يأتي بعده.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:animals-grow-up:10}
``ماتت \omterm{ex:g2:animals-grow-up:butterfly}{اليرقة} في غلافها، وحلّت فراشة محلها.'' ماذا حدث حقًّا داخل
\omterm{ex:g2:animals-grow-up:butterfly}{الشرنقة}؟ أجب وفق \cref{def:g2:animals-grow-up:metamorphosis}
و\cref{ex:g2:animals-grow-up:butterfly}.
\end{exercise}
