\chapter{Vertebrates and Invertebrates}\label{ch:g5:vertebrates}

\omterm{def:g1:plants-animals:animal}{Animals} can be sorted in many ways. One useful split is whether they
have a backbone. This chapter introduces \omterm{def:g5:vertebrates:split}{vertebrates} and \omterm{def:g5:vertebrates:split}{invertebrates}
and tours the main \omterm{def:g5:vertebrates:split}{vertebrate} groups.

\section{Backbone or no backbone?}

\begin{definition}[Vertebrate and invertebrate]\label{def:g5:vertebrates:split}
A \emph{vertebrate}\index{vertebrate} is an \omterm{def:g1:plants-animals:animal}{animal} with a backbone
(spine) made of vertebrae. An \emph{invertebrate}\index{invertebrate}
is an \omterm{def:g1:plants-animals:animal}{animal} without a backbone. Most \omterm{def:g1:plants-animals:animal}{animal} species on Earth are
invertebrates.
\end{definition}

\omimg{animal-groups.jpg}{0.52\linewidth}{\omterm{def:g1:plants-animals:animal}{Animals} form many groups.
Looking for a backbone is a first sorting step, not the whole story of
\omterm{def:g3:classify:class}{classification}.}

\begin{example}[Sort these]\label{ex:g5:vertebrates:sort}
Dog, human, sparrow, frog and goldfish are \omterm{def:g5:vertebrates:split}{vertebrates}. Earthworm,
\omterm{def:g3:insects:compare}{snail}, \omterm{def:g3:insects:compare}{spider}, butterfly and jellyfish are \omterm{def:g5:vertebrates:split}{invertebrates}.
\end{example}

\section{Vertebrate groups}

\begin{definition}[Five familiar groups]\label{def:g5:vertebrates:groups}
Common \omterm{def:g5:vertebrates:split}{vertebrate} groups include:
\begin{itemize}
  \item \emph{Fish}\index{fish} --- live in \omterm{def:g4:plants-need:needs}{water}, gills, usually scales
        and fins.
  \item \emph{Amphibians}\index{amphibian} --- often live in \omterm{def:g4:plants-need:needs}{water} when
        young and on land as \omterm{def:g3:life-cycles:butterfly}{adults} (frogs, salamanders); moist \omterm{def:g1:senses:five}{skin}.
  \item \emph{Reptiles}\index{reptile} --- dry scaly \omterm{def:g1:senses:five}{skin}; most lay
        shelled \omterm{def:g3:life-cycles:butterfly}{eggs} on land (snakes, lizards, turtles, crocodiles).
  \item \emph{Birds}\index{bird} --- feathers, wings, lay hard-shelled
        \omterm{def:g3:life-cycles:butterfly}{eggs}; most fly.
  \item \emph{Mammals}\index{mammal} --- hair or fur; mothers produce
        milk for young.
\end{itemize}
\end{definition}

\begin{example}[Body coverings]\label{ex:g5:vertebrates:cover}
Scales (many \omterm{def:g5:vertebrates:groups}{fish} and \omterm{def:g2:animal-groups:five}{reptiles}), moist \omterm{def:g1:senses:five}{skin} (\omterm{def:g2:animal-groups:five}{amphibians}), feathers
(\omterm{def:g2:animal-groups:five}{birds}), fur (\omterm{def:g2:animal-groups:five}{mammals}) --- coverings match lifestyle and \omterm{def:g4:ecosystems:habitat}{habitat}.
\end{example}

\section{Invertebrate diversity}

\begin{proposition}[Many body plans]\label{prop:g5:vertebrates:invert}
\omterm{def:g5:vertebrates:split}{Invertebrates} include \omterm{def:g3:insects:insect}{insects} (six \omterm{def:g1:my-body:parts}{legs}, often wings), \omterm{def:g3:insects:compare}{spiders} (eight
\omterm{def:g1:my-body:parts}{legs}), crustaceans (crabs, many with shells), molluscs (\omterm{def:g3:insects:compare}{snails},
octopuses), \omterm{def:g3:insects:compare}{worms} and more. They have no backbone but may have shells
or hard outer \omterm{def:g3:bones:bone}{skeletons}.
\end{proposition}
\admitted

\begin{example}[Insects rule by number]\label{ex:g5:vertebrates:insects}
\omterm{def:g3:insects:insect}{Insects} are the most species-rich \omterm{def:g1:plants-animals:animal}{animal} group. Beetles alone outnumber
many \omterm{def:g5:vertebrates:split}{vertebrate} groups combined.
\end{example}

\section{Using features to classify}

\begin{method}[Simple animal key steps]\label{met:g5:vertebrates:key}
\begin{enumerate}
  \item Does it have a backbone? (If unknown, use \omterm{def:g4:ecosystems:habitat}{habitat} and \omterm{def:g1:my-body:body}{body}
        covering clues carefully.)
  \item If \omterm{def:g5:vertebrates:split}{vertebrate}: check \omterm{def:g1:senses:five}{skin} covering, limbs, and how it breathes.
  \item If \omterm{def:g5:vertebrates:split}{invertebrate}: count \omterm{def:g1:my-body:parts}{legs}, look for shell, segments, or soft
        \omterm{def:g1:my-body:body}{body}.
  \item Compare with reliable pictures or field guides.
\end{enumerate}
\end{method}
