\chapter{Os animais ao nosso redor}\label{ch:g1:animals-around-us}

Um sabiá pula pelo gramado, um gato o observa do muro, uma formiga
carrega pela soleira da porta uma migalha maior do que ela. Os
\omterm{def:g1:animals-around-us:animal}{animais} estão por toda parte ao nosso redor --- no quintal, no
parque, dentro de casa. Vamos aprender a olhar para eles como um
biólogo faz: para o corpo deles e para o jeito como se movem.

\section{O que é um animal?}

\begin{definition}[Animal]\label{def:g1:animals-around-us:animal}
Um \emph{animal}\index{animal} é um \omterm{def:g1:living-or-not:living}{ser vivo} que se alimenta comendo
outros \omterm{def:g1:living-or-not:living}{seres vivos} --- plantas ou outros animais --- e que quase
sempre consegue se deslocar sozinho. Um gato, um sabiá, um peixe e
uma formiga são animais. Nós também somos animais.
\end{definition}

\begin{example}[Os animais de um quintal]\label{ex:g1:animals-around-us:garden}
Um quintal pequeno pode ser a casa de muitos \omterm{def:g1:animals-around-us:animal}{animais}: um sabiá
puxando uma minhoca da terra, a própria minhoca, um caracol numa
folha, uma aranha na teia, uma abelha numa flor e o gato do vizinho.
Grandes ou pequenininhos, todos se alimentam e todos se movem.
\end{example}

\begin{remark}[Pequeno não quer dizer sem importância]\label{rem:g1:animals-around-us:small}
Uma formiga é um \omterm{def:g1:animals-around-us:animal}{animal} tanto quanto um elefante. O tamanho do corpo
não decide quem é \omterm{def:g1:animals-around-us:animal}{animal}: comer e se mover é que decidem. Alguns dos
\omterm{def:g1:animals-around-us:animal}{animais} mais incríveis ao nosso redor são os menorzinhos.
\end{remark}

\section{As partes do corpo de um animal}

\begin{definition}[Partes do corpo]\label{def:g1:animals-around-us:parts}
Quase todos os \omterm{def:g1:animals-around-us:animal}{animais} têm uma \emph{cabeça}\index{cabeça} (com a
boca e a maior parte dos órgãos dos sentidos), um corpo e
\emph{membros}\index{membro} --- patas, asas ou nadadeiras --- que
usam para se mover. Muitos também têm uma \emph{cauda}\index{cauda}.
\end{definition}

\begin{example}[Conte as patas]\label{ex:g1:animals-around-us:legs}
As patas vêm em números que vale a \omterm{def:g1:animals-around-us:covering}{pena} contar. O sabiá anda com $2$
patas, o gato com $4$, a joaninha com $6$, a aranha com $8$ --- e a
minhoca e o caracol não têm pata nenhuma, mas mesmo assim chegam
aonde querem ir.
\end{example}

\begin{definition}[Revestimento do corpo]\label{def:g1:animals-around-us:covering}
O corpo de um \omterm{def:g1:animals-around-us:animal}{animal} é protegido por um revestimento:
\emph{pelo}\index{pelo} no gato, \emph{penas}\index{pena} no sabiá,
\emph{escamas}\index{escama} no peixe, uma \emph{concha}\index{concha}
no caracol e uma \emph{pele}\index{pele} lisa e úmida no sapo.
\end{definition}

\begin{example}[Quem veste o quê]\label{ex:g1:animals-around-us:covering}
Faça carinho num gato: pelo. Pegue uma \omterm{def:g1:animals-around-us:covering}{pena} caída de sabiá: leve e
macia. Olhe de pertinho um peixe na feira: fileiras de escamas
brilhantes, como telhinhas de telhado. Três \omterm{def:g1:animals-around-us:animal}{animais}, três roupas
diferentes.
\end{example}

\section{Como os animais se movem}

\begin{proposition}[Jeitos de se mover]\label{prop:g1:animals-around-us:moving}
Os \omterm{def:g1:animals-around-us:animal}{animais} se movem de um jeito que combina com os \omterm{def:g1:animals-around-us:parts}{membros} deles:
\begin{itemize}
  \item os \omterm{def:g1:animals-around-us:animal}{animais} com patas \emph{andam}, correm, pulam ou rastejam
        --- o gato, o sabiá no gramado, a formiga;
  \item os \omterm{def:g1:animals-around-us:animal}{animais} com asas \emph{voam}\index{voar} --- o sabiá, a
        abelha, a borboleta;
  \item os \omterm{def:g1:animals-around-us:animal}{animais} com nadadeiras \emph{nadam}\index{nadar} --- o peixe;
  \item os \omterm{def:g1:animals-around-us:animal}{animais} sem \omterm{def:g1:animals-around-us:parts}{membros} deslizam ou se contorcem --- o caracol
        no seu único pé macio, a minhoca esticando e encolhendo.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Um animal, vários jeitos]\label{rem:g1:animals-around-us:several}
Muitos \omterm{def:g1:animals-around-us:animal}{animais} têm mais de um jeito de se mover. O sabiá anda
\emph{e} voa; o sapo nada \emph{e} pula; o pato anda, nada \emph{e}
voa. Observe um \omterm{def:g1:animals-around-us:animal}{animal} um tempinho antes de decidir como ele se
move.
\end{remark}

\begin{method}[Observar um animal]\label{met:g1:animals-around-us:observe}
Para descrever um \omterm{def:g1:animals-around-us:animal}{animal} como um biólogo:
\begin{enumerate}
  \item olhe as partes: cabeça, corpo, \omterm{def:g1:animals-around-us:parts}{membros}, cauda;
  \item conte as patas, sem chutar;
  \item diga o revestimento: pelo, \omterm{def:g1:animals-around-us:covering}{penas}, escamas, \omterm{def:g1:animals-around-us:covering}{concha}, \omterm{def:g1:animals-around-us:covering}{pele};
  \item veja como ele se move: andar, pular, voar, nadar, deslizar;
  \item diga onde você o encontrou: gramado, árvore, lagoa, embaixo
        de uma pedra.
\end{enumerate}
Nunca machuque o \omterm{def:g1:animals-around-us:animal}{animal}: observe e depois deixe que ele siga o caminho dele.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:1}
Diga as quatro partes principais do corpo do gato da figura. Qual
delas carrega os olhos e a boca?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:2}
Quantas patas: (a) uma galinha? (b) um cachorro? (c) uma joaninha?
(d) uma aranha? (e) uma minhoca?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:3}
Diga o revestimento de cada \omterm{def:g1:animals-around-us:animal}{animal}: uma ovelha, um pato, uma truta,
um caracol.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:4}
Quais \omterm{def:g1:animals-around-us:animal}{animais} deste capítulo sabem voar? O que todos eles têm no
lugar das patas da frente ou dos braços?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:5}
O caracol e a minhoca não têm patas. Como cada um deles se move?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:6}
Um pato consegue se mover de três jeitos. Diga quais são e qual
parte do corpo ele usa em cada um.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:7}
Por que uma formiga é um \omterm{def:g1:animals-around-us:animal}{animal}, segundo a
\cref{def:g1:animals-around-us:animal}? Responda com as duas ideias
da definição.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:8}
Faça a experiência: observe um \omterm{def:g1:animals-around-us:animal}{animal} lá fora por um tempinho ---
um pássaro, um inseto, um caracol. Descreva ele com os cinco passos
do \cref{met:g1:animals-around-us:observe}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:9}
Uma amiga diz: ``Peixe não pode ser \omterm{def:g1:animals-around-us:animal}{animal} --- eles não andam!'' O
que você responderia? Use a \cref{prop:g1:animals-around-us:moving}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:10}
\omterm{def:g1:animals-around-us:animal}{Animal} misterioso: ele tem $2$ patas, $2$ asas, \omterm{def:g1:animals-around-us:covering}{penas}, e não
consegue voar, mas corre muito rápido. Ele é uma ave? Será que ele
seria o corredor mais rápido do seu bairro? (Pense na ema.)
\end{exercise}
