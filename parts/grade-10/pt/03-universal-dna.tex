\chapter{O DNA: uma molécula genética universal}\label{ch:g10:universal-dna}

Num laboratório às escuras, sob uma lâmpada azul, um de dois
camundongos brancos brilha em verde do focinho à cauda. Ele carrega, em
cada uma das \omterm{def:g10:cells-common-unit:cell}{células} dele, um único \omterm{def:g10:universal-dna:gene}{gene} retirado de uma água-viva que
se acende no Pacífico --- e as \omterm{def:g10:cells-common-unit:cell}{células} do camundongo leem esse \omterm{def:g10:universal-dna:gene}{gene} de
água-viva exatamente como as \omterm{def:g10:cells-common-unit:cell}{células} da água-viva o leem. O experimento
funciona porque todo ser vivo escreve os \omterm{def:g10:universal-dna:gene}{genes} dele na mesma molécula,
com as mesmas quatro letras, e os lê pelas mesmas regras. Este capítulo
trata dessa molécula, o \omterm{def:g10:universal-dna:information}{DNA}: como ela é construída, o que a estrutura
dela explica, e o que o camundongo brilhante prova.

\section{O material genético}

\begin{definition}[Informação genética]\label{def:g10:universal-dna:information}
A \emph{informação genética}\index{informação genética} de uma \omterm{def:g10:cells-common-unit:cell}{célula} é
o conjunto de instruções, transmitidas de uma \omterm{def:g10:cells-common-unit:cell}{célula} às células-filhas
dela e dos pais aos descendentes, que determina as características do
organismo e as \omterm{def:g10:chemistry-of-life:families}{proteínas} que as \omterm{def:g10:cells-common-unit:cell}{células} dele conseguem fabricar. Em
todo ser vivo ela é carregada por moléculas de \emph{DNA}\index{DNA}
(ácido desoxirribonucleico), empacotadas com \omterm{def:g10:chemistry-of-life:families}{proteínas} nos
\emph{cromossomos}\index{cromossomo} --- no \omterm{def:g10:cells-common-unit:organelle}{núcleo} de uma \omterm{def:g10:cells-common-unit:prokaryote}{célula
eucariótica}, livres no \omterm{def:g10:cells-common-unit:cell}{citoplasma} de uma bactéria.
\end{definition}

\begin{proposition}[O DNA é a molécula da hereditariedade]\label{prop:g10:universal-dna:material}
O material hereditário de uma \omterm{def:g10:cells-common-unit:cell}{célula} é o \omterm{def:g10:universal-dna:information}{DNA} dela, e não as \omterm{def:g10:chemistry-of-life:families}{proteínas}
nem qualquer outro constituinte: transferir \omterm{def:g10:universal-dna:information}{DNA} purificado de uma
linhagem de bactérias para outra transfere as características da
primeira linhagem.
\end{proposition}
\begin{proof}[Evidências]
Griffith (1928) descobriu que bactérias inofensivas da pneumonia,
misturadas com \omterm{def:g10:cells-common-unit:cell}{células} mortas de uma linhagem letal, tornavam-se letais
e assim permaneciam ao longo das gerações: alguma coisa vinda das
\omterm{def:g10:cells-common-unit:cell}{células} mortas as tinha ``transformado''. Avery, MacLeod e McCarty
(1944) separaram o extrato das \omterm{def:g10:cells-common-unit:cell}{células} mortas nas famílias de moléculas
dele e testaram cada uma: só a fração de \omterm{def:g10:universal-dna:information}{DNA} transformava; tratar o
extrato com uma enzima que destrói o \omterm{def:g10:universal-dna:information}{DNA} abolia o efeito, enquanto
enzimas que destroem \omterm{def:g10:chemistry-of-life:families}{proteínas} ou RNA não. A substância transformadora,
e portanto a portadora da característica hereditária, era o \omterm{def:g10:universal-dna:information}{DNA}.
\end{proof}

\begin{example}[Onde está o DNA]\label{ex:g10:universal-dna:where}
Um corante específico para o \omterm{def:g10:universal-dna:information}{DNA} tinge o \omterm{def:g10:cells-common-unit:organelle}{núcleo} de uma \omterm{def:g10:cells-common-unit:cell}{célula} de
bochecha e nada mais no \omterm{def:g10:cells-common-unit:cell}{citoplasma}; numa \omterm{def:g10:cells-common-unit:cell}{célula} em divisão ele tinge os
\omterm{def:g10:universal-dna:information}{cromossomos} compactos. Uma bactéria capta o corante numa região central
sem envelope. Nos dois casos, a quantidade de \omterm{def:g10:universal-dna:information}{DNA} dobra antes de cada
divisão e é dividida pela metade entre as duas células-filhas --- o
comportamento esperado de uma informação que precisa ser copiada e
repartida.
\end{example}

\section{A estrutura do DNA}

\begin{definition}[Nucleotídeo]\label{def:g10:universal-dna:nucleotide}
O \omterm{def:g10:universal-dna:information}{DNA} é uma cadeia de \emph{nucleotídeos}\index{nucleotídeo}. Cada
nucleotídeo é feito de três partes: um grupo fosfato, um açúcar (a
desoxirribose) e uma de quatro \emph{bases}\index{base (DNA)}
nitrogenadas: adenina (A), timina (T), guanina (G) e citosina (C).
Açúcares e fosfatos se alternam para formar o esqueleto da cadeia; as
bases se projetam dele, uma por nucleotídeo. Os quatro nucleotídeos
diferem só pela base, de modo que uma fita de \omterm{def:g10:universal-dna:information}{DNA} é descrita pela
sequência das bases dela, escrita como uma sequência de letras:
\texttt{ATGGCTTAC\dots}
\end{definition}

\begin{proposition}[A dupla-hélice]\label{prop:g10:universal-dna:helix}
Uma molécula de \omterm{def:g10:universal-dna:information}{DNA} é feita de duas fitas enroladas uma na outra numa
\emph{dupla-hélice}\index{dupla-hélice} destrógira de cerca de
\qty{2}{nm} de largura, com uma volta a cada dez pares de bases
(\qty{3.4}{nm}). As fitas correm em sentidos opostos e se voltam uma
para a outra pelas bases, que se pareiam segundo uma regra estrita de
\emph{complementaridade}\index{complementaridade}: A se pareia só com
T, G só com C. A sequência de uma fita fixa, portanto, a sequência da
outra.
\end{proposition}
\begin{proof}[Evidências]
Chargaff (1950) mediu a composição em bases do \omterm{def:g10:universal-dna:information}{DNA} de muitas espécies e
encontrou, em todas elas, tanto A quanto T e tanto G quanto C,
enquanto a proporção de A+T para G+C variava de espécie para espécie
--- uma regra de pareamento, e não um acaso químico. As fotografias de
raios X de fibras de \omterm{def:g10:universal-dna:information}{DNA} feitas por Franklin (1952) mostraram a
assinatura de uma hélice de \qty{2}{nm} de diâmetro e \qty{3.4}{nm} de
passo. Watson e Crick (1953) construíram o único modelo que se ajusta
aos dois: dois esqueletos por fora, bases pareadas por dentro, com os
pares A--T e G--C tendo a mesma largura para que a hélice seja regular.
\end{proof}

\begin{example}[A regra de Chargaff em números]\label{ex:g10:universal-dna:chargaff}
\omterm{def:g10:universal-dna:information}{DNA} humano: A 30.9\%, T 29.4\%, G 19.9\%, C 19.8\% --- A$\,\approx\,$T
e G$\,\approx\,$C dentro da precisão da medida, e A+T $= 60\%$.
\emph{E.~coli}: A 24.7\%, T 23.6\%, G 26.0\%, C 25.7\%, A+T $= 48\%$.
Levedura: A+T $= 64\%$. Se uma amostra de \omterm{def:g10:universal-dna:information}{DNA} contém 32\% de A, a
\omterm{prop:g10:universal-dna:helix}{complementaridade} prevê 32\% de T e $(100 - 64)/2 = 18\%$ de G e de C.
\end{example}

\begin{method}[Escrever a fita complementar]\label{met:g10:universal-dna:complement}
Dada uma fita, escreva sob cada base a parceira dela
(A$\leftrightarrow$T, G$\leftrightarrow$C), depois leia o resultado no
sentido oposto se a convenção do exercício exigir. Para
\texttt{5'-ATGGCTTAC-3'} a fita parceira, lida no sentido próprio dela,
é \texttt{5'-GTAAGCCAT-3'}. Contagem: uma fita com $n_\mathrm{A}$
adeninas fica diante de uma fita com $n_\mathrm{A}$ timinas, de modo
que na molécula inteira $n_\mathrm{A} = n_\mathrm{T}$ e
$n_\mathrm{G} = n_\mathrm{C}$.
\end{method}

\begin{proposition}[O que a estrutura explica]\label{prop:g10:universal-dna:explains}
Duas propriedades do material genético decorrem da estrutura.
\begin{itemize}
  \item \emph{Informação}: a ordem das bases ao longo de uma fita não é
        restringida pela química --- qualquer sequência é possível ---,
        de modo que a sequência pode carregar uma mensagem, como a
        ordem das letras carrega um texto. Uma \omterm{def:g10:cells-common-unit:cell}{célula} humana guarda
        $\num{3.2e9}$ pares de bases em cada conjunto de \omterm{def:g10:universal-dna:information}{cromossomos}.
  \item \emph{Cópia}: como cada fita determina a outra, a molécula pode
        ser duplicada separando as fitas e construindo uma nova
        parceira sobre cada uma. O mecanismo é assunto do
        \cref{ch:g11:dna-replication}.
\end{itemize}
\end{proposition}
\admitted

\section{Genes, alelos, cromossomos}

\begin{definition}[Gene e alelo]\label{def:g10:universal-dna:gene}
Um \emph{gene}\index{gene} é um segmento de uma molécula de \omterm{def:g10:universal-dna:information}{DNA} ---
tipicamente alguns milhares de pares de bases --- cuja sequência
carrega as instruções para fabricar uma \omterm{def:g10:chemistry-of-life:families}{proteína} (ou, às vezes, uma
molécula funcional de RNA). Ele ocupa uma posição fixa num dado
\omterm{def:g10:universal-dna:information}{cromossomo}. As versões de um gene que diferem por algumas bases são os
\emph{alelos}\index{alelo} dele: mesma posição, mesma função, sequência
ligeiramente diferente e, portanto, muitas vezes uma \omterm{def:g10:chemistry-of-life:families}{proteína}
ligeiramente diferente.
\end{definition}

\begin{example}[Números]\label{ex:g10:universal-dna:numbers}
O genoma humano --- um conjunto completo de \omterm{def:g10:universal-dna:information}{cromossomos} --- carrega
$\num{3.2e9}$ pares de bases e cerca de $\num{20000}$ \omterm{def:g10:universal-dna:gene}{genes}; os \omterm{def:g10:universal-dna:gene}{genes}
ocupam apenas alguns por cento da sequência. Uma \omterm{def:g10:cells-common-unit:cell}{célula} de
\emph{E.~coli} carrega uma única molécula circular de \omterm{def:g10:universal-dna:information}{DNA} de
$\num{4.6e6}$ pares de bases e cerca de 4300 \omterm{def:g10:universal-dna:gene}{genes}. Dois seres humanos
sem parentesco diferem em aproximadamente uma base em mil: uns três
milhões de posições, a maioria fora dos \omterm{def:g10:universal-dna:gene}{genes} e sem consequência,
algumas delas \omterm{def:g10:universal-dna:gene}{alelos} que mudam uma característica.
\end{example}

\begin{proposition}[Os cromossomos]\label{prop:g10:universal-dna:chromosome}
Um \omterm{def:g10:universal-dna:information}{cromossomo} é uma molécula de \omterm{def:g10:universal-dna:information}{DNA}, enrolada em \omterm{def:g10:chemistry-of-life:families}{proteínas} de
empacotamento e espiralada sobre si mesma. Esticado, o \omterm{def:g10:universal-dna:information}{DNA} de um
\omterm{def:g10:universal-dna:information}{cromossomo} humano teria de \qty{1.5}{cm} a \qty{8.5}{cm} de
comprimento; os 46 \omterm{def:g10:universal-dna:information}{cromossomos} de uma \omterm{def:g10:cells-common-unit:cell}{célula} guardam juntos cerca de
\qty{2}{m} de \omterm{def:g10:universal-dna:information}{DNA} num \omterm{def:g10:cells-common-unit:organelle}{núcleo} de \qty{6}{\micro m} de diâmetro. Os
\omterm{def:g10:universal-dna:information}{cromossomos} só são visíveis como corpos separados durante a divisão
celular, quando estão espiralados o mais apertado possível; entre duas
divisões eles ficam desenrolados e preenchem o \omterm{def:g10:cells-common-unit:organelle}{núcleo} como um emaranhado.
\end{proposition}
\admitted

\section{Uma molécula para todos os seres vivos}

\begin{proposition}[Universalidade do DNA]\label{prop:g10:universal-dna:universal}
Todos os seres vivos --- bactérias, fungos, vegetais, animais ---
carregam a \omterm{def:g10:universal-dna:information}{informação genética} deles como \omterm{def:g10:universal-dna:information}{DNA} construído com os mesmos
quatro \omterm{def:g10:universal-dna:nucleotide}{nucleotídeos}, e leem a sequência de um \omterm{def:g10:universal-dna:gene}{gene} pelas mesmas regras.
Em consequência, um \omterm{def:g10:universal-dna:gene}{gene} retirado de uma espécie e inserido no \omterm{def:g10:universal-dna:information}{DNA} de
outra é lido pelas \omterm{def:g10:cells-common-unit:cell}{células} da receptora, que fabricam a \omterm{def:g10:chemistry-of-life:families}{proteína} que
ele codifica: a \emph{transgênese}\index{transgênese} é possível, e o
organismo transgênico passa o \omterm{def:g10:universal-dna:gene}{gene} estrangeiro aos descendentes dele
como qualquer outro.
\end{proposition}
\begin{proof}[Evidências]
O \omterm{def:g10:universal-dna:gene}{gene} da \omterm{def:g10:chemistry-of-life:families}{proteína} verde fluorescente (GFP) da água-viva
\emph{Aequorea victoria} já foi inserido em bactérias, leveduras,
vegetais, peixes, camundongos e coelhos: cada um deles fabrica a
\omterm{def:g10:chemistry-of-life:families}{proteína} e brilha em verde sob luz azul, e a característica é herdada.
O \omterm{def:g10:universal-dna:gene}{gene} humano da insulina, inserido em \emph{E.~coli} em 1978, faz a
bactéria produzir insulina humana, que desde então é a fonte do hormônio
para os diabéticos. Não se conhece nenhum caso de espécie cujos \omterm{def:g10:universal-dna:gene}{genes}
estejam escritos em outra química.
\end{proof}

\begin{example}[Usos da transgênese]\label{ex:g10:universal-dna:uses}
Insulina e hormônio de crescimento humanos fabricados por bactérias; um
\omterm{def:g10:universal-dna:gene}{gene} de resistência a uma praga de insetos inserido no milho ou no
algodão; arroz que carrega \omterm{def:g10:universal-dna:gene}{genes} para fabricar um precursor da vitamina
A; o \omterm{def:g10:universal-dna:gene}{gene} da GFP acoplado a outro \omterm{def:g10:universal-dna:gene}{gene} para que as \omterm{def:g10:cells-common-unit:cell}{células} em que esse
\omterm{def:g10:universal-dna:gene}{gene} está ativo se acendam ao microscópio --- este último uso, uma
ferramenta de pesquisa, é o mais comum de todos. Cada uso levanta
questões próprias de segurança e de consequências para o ambiente; cada
um se apoia no mesmo fato, a universalidade da molécula genética.
\end{example}

\begin{remark}[Unidade e diversidade no nível molecular]\label{rem:g10:universal-dna:unity}
A universalidade do \omterm{def:g10:universal-dna:information}{DNA} é a face molecular da unidade da vida
encontrada no \cref{ch:g10:chemistry-of-life}: uma química, uma
molécula de informação, um sistema de leitura para todo organismo da
Terra, que é o que uma origem comum prevê
(\cref{ch:g10:common-ancestry}). A diversidade da vida também está
escrita na mesma molécula, como diferenças de sequência: entre dois
\omterm{def:g10:universal-dna:gene}{alelos}, algumas bases; entre duas espécies, alguns por cento do genoma;
entre uma bactéria e um ser humano, toda a organização do texto.
\end{remark}

\section{Exercícios}

\section{Problema: o gene da água-viva}

\begin{problem}[{Problema de fim de semana --- um gene de água-viva
seguido até um camundongo: o comprimento dele, as letras dele, por que o
camundongo consegue lê-lo, e os dois metros de DNA que o carregam em cada
célula}]
\label{pb:g10:universal-dna:1}
A \omterm{def:g10:chemistry-of-life:families}{proteína} verde fluorescente da água-viva \emph{Aequorea victoria} é
uma cadeia de 238 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}. O \omterm{def:g10:universal-dna:gene}{gene} dela, tal como usado no
laboratório, é um segmento de \omterm{def:g10:universal-dna:information}{DNA} de 720 pares de bases. Tome
\qty{0.34}{nm} por par de bases, dez pares de bases por volta, e um
genoma de camundongo de $\num{2.7e9}$ pares de bases por conjunto de
\omterm{def:g10:universal-dna:information}{cromossomos}, dois conjuntos por \omterm{def:g10:cells-common-unit:cell}{célula}.

\medskip
\noindent\textbf{Parte I --- A molécula.}
\begin{enumerate}
  \item Calcule o comprimento do \omterm{def:g10:universal-dna:gene}{gene} da GFP em nanômetros, e o número
        de voltas de hélice que ele faz.
  \item Uma fita do \omterm{def:g10:universal-dna:gene}{gene} começa por \texttt{ATGAGTAAAGGAGAAGAAC}.
        Escreva a fita complementar.
  \item No \omterm{def:g10:universal-dna:gene}{gene} inteiro, a adenina representa 27\% das bases. Dê as
        porcentagens das outras três bases.
  \item O \omterm{def:g10:universal-dna:gene}{gene} tem 720 pares de bases e codifica 238 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}.
        Calcule a razão, e proponha o que ela sugere sobre quantas
        bases especificam um \omterm{def:g10:chemistry-of-life:families}{aminoácido} (uma hipótese que o
        \cref{ch:g11:gene-expression} vai confirmar).
  \item Quantas sequências diferentes de 720 bases são possíveis em
        princípio? Escreva a resposta como uma potência de 4, depois
        explique numa frase por que a sequência pode ser chamada de
        informação.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Por que o camundongo consegue lê-lo.}
\begin{enumerate}[resume]
  \item Enuncie a propriedade do \omterm{def:g10:universal-dna:information}{DNA} que torna possível a uma \omterm{def:g10:cells-common-unit:cell}{célula} de
        camundongo usar um \omterm{def:g10:universal-dna:gene}{gene} de água-viva.
  \item O \omterm{def:g10:universal-dna:gene}{gene} é inserido num óvulo fecundado, e não na pele de um
        camundongo adulto. Usando a teoria celular, explique que
        diferença isso faz para o lugar onde o \omterm{def:g10:universal-dna:gene}{gene} vai parar.
  \item O camundongo adulto brilha em todos os tecidos. O que isso
        mostra sobre o conteúdo em \omterm{def:g10:universal-dna:information}{DNA} dos diferentes tipos de \omterm{def:g10:cells-common-unit:cell}{células}
        dele?
  \item O camundongo brilhante é cruzado com um camundongo comum.
        Preveja, com uma justificativa, se parte dos descendentes
        brilha.
  \item Um camundongo de controle recebe uma injeção de \omterm{def:g10:chemistry-of-life:families}{proteína} GFP
        purificada em vez do \omterm{def:g10:universal-dna:gene}{gene}: ele brilha fracamente por alguns
        dias, depois para. Explique a diferença entre os dois
        experimentos.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- As evidências por trás da história.}
\begin{enumerate}[resume]
  \item No experimento de Avery, cite a fração que transformou as
        bactérias e o tratamento que aboliu a transformação.
  \item Explique por que o experimento de transformação é uma versão
        natural da \omterm{prop:g10:universal-dna:universal}{transgênese}.
  \item Chargaff encontrou A$\,=\,$T e G$\,=\,$C em todas as espécies,
        mas A+T variando de 25\% a 75\%. Qual das duas observações
        apoia o pareamento de bases, e qual mostra que a sequência é
        livre?
  \item As fotografias de Franklin deram o diâmetro da hélice como
        \qty{2}{nm}. Explique por que parear uma base grande com uma
        pequena (A com T, G com C), e não duas grandes, torna a hélice
        regular.
  \item Por que a descoberta da estrutura sugeriu imediatamente como o
        \omterm{def:g10:universal-dna:information}{DNA} é copiado?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Dois metros num \omterm{def:g10:cells-common-unit:organelle}{núcleo}.}
\begin{enumerate}[resume]
  \item Calcule o comprimento total de \omterm{def:g10:universal-dna:information}{DNA} de uma \omterm{def:g10:cells-common-unit:cell}{célula} de camundongo.
  \item O \omterm{def:g10:cells-common-unit:organelle}{núcleo} tem \qty{6}{\micro m} de diâmetro. Por que fator o \omterm{def:g10:universal-dna:information}{DNA}
        precisa ser encurtado pelo enrolamento para caber?
  \item Que fração do \omterm{def:g10:universal-dna:information}{DNA} da \omterm{def:g10:cells-common-unit:cell}{célula} é o \omterm{def:g10:universal-dna:gene}{gene} da GFP?
  \item Um camundongo tem cerca de $\num{2e4}$ \omterm{def:g10:universal-dna:gene}{genes} com média de
        $\num{3e4}$ pares de bases, incluindo as partes não
        codificantes deles. Que fração do genoma os \omterm{def:g10:universal-dna:gene}{genes} ocupam?
  \item Enuncie o resultado numa frase: quanto \omterm{def:g10:universal-dna:information}{DNA} uma \omterm{def:g10:cells-common-unit:cell}{célula} de
        camundongo carrega, e que fato único a respeito desse \omterm{def:g10:universal-dna:information}{DNA}
        permite que um único \omterm{def:g10:universal-dna:gene}{gene} de água-viva entre os milhares dele
        seja lido?
\end{enumerate}
\end{problem}
