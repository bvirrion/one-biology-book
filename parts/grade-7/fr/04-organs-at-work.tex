\chapter{Les organes au travail}\label{ch:g7:organs-at-work}

Monte trois étages en courant et fais le point~: les \omterm{def:g1:my-body:parts}{jambes} qui
brûlent, le \omterm{def:g4:heart-and-blood:heart}{cœur} qui cogne, le souffle qui s'emballe --- et,
curieusement, la \omterm{def:g1:the-five-senses:sense}{peau} rouge et chaude. Personne n'a commandé ces
changements~; ils se sont commandés eux-mêmes, et dans des
proportions exactes. Ce chapitre mesure ce qu'un organe au travail
consomme vraiment, et comment le corps réorganise ses livraisons dès
que le travail commence.

\section{Ce que consomme un muscle au travail}

\begin{proposition}[La facture du muscle]\label{prop:g7:organs-at-work:bill}
Un \omterm{def:g3:bones-and-muscles:muscle}{muscle} au travail dépense, et sa dépense se mesure. Comparé au
même \omterm{def:g3:bones-and-muscles:muscle}{muscle} au repos, un \omterm{def:g3:bones-and-muscles:muscle}{muscle} qui travaille dur~:
\begin{itemize}
  \item prélève bien plus de \emph{\omterm{def:g7:what-is-respiration:respiration}{dioxygène}} dans le \omterm{prop:g4:journey-of-food:absorb}{sang} qui le
        traverse --- plusieurs fois son prélèvement de repos~;
  \item prélève bien plus de \emph{glucose}\index{glucose} --- le
        \omterm{def:g4:journey-of-food:digestion}{nutriment} sucré tout prêt du \omterm{prop:g4:journey-of-food:absorb}{sang}, son carburant de tous
        les jours~;
  \item rejette d'autant plus de \emph{\omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone}} --- et
        de \emph{chaleur}.
\end{itemize}
C'est la combustion cellulaire de
\cref{prop:g7:what-is-respiration:power}, lue sur les compteurs d'un
organe~: du carburant et du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} qui entrent, du travail, un
gaz de rebut et de la chaleur qui sortent.
\end{proposition}
\admitted

\begin{example}[Mesurer de part et d'autre d'un muscle]\label{ex:g7:organs-at-work:measure}
Les physiologistes comparent le \omterm{prop:g4:journey-of-food:absorb}{sang} qui \emph{entre} dans un \omterm{def:g3:bones-and-muscles:muscle}{muscle}
et le \omterm{prop:g4:journey-of-food:absorb}{sang} qui en \emph{sort}. Au repos, le \omterm{prop:g4:journey-of-food:absorb}{sang} sortant n'a cédé
qu'une part modeste de son \omterm{def:g7:what-is-respiration:respiration}{dioxygène} et de son \omterm{prop:g7:organs-at-work:bill}{glucose}. Lors d'un
travail intense, la même comparaison montre le \omterm{prop:g4:journey-of-food:absorb}{sang} sortant dépouillé
--- l'essentiel de son \omterm{def:g7:what-is-respiration:respiration}{dioxygène} livrable prélevé --- tandis que le
\omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} a bondi. Même organe, mêmes compteurs~: seul le
travail a changé, et la facture a suivi.
\end{example}

\begin{remark}[La chaleur est le reçu]\label{rem:g7:organs-at-work:warmth}
La chaleur d'un corps au travail n'est pas un accident~: brûler du
carburant dégage toujours de la chaleur, et le \omterm{def:g3:bones-and-muscles:muscle}{muscle} est la
chaufferie du corps. Frissonner, c'est ce mécanisme déclenché
exprès --- des \omterm{def:g3:bones-and-muscles:muscle}{muscles} mis au travail pour rien d'autre que la
chaleur (l'abonnement de \cref{ex:g3:balanced-diet:rest}, brièvement
doublé). La \omterm{def:g1:the-five-senses:sense}{peau} rouge du sprint est le système de refroidissement
qui évacue la chaleur de la chaufferie --- l'affaire de la section
suivante.
\end{remark}

\section{Des livraisons à la demande}

\begin{proposition}[Le corps réorganise ses approvisionnements]\label{prop:g7:organs-at-work:supply}
Dès que le travail commence, les lignes d'approvisionnement se
réorganisent pour payer la facture
(\cref{ex:g4:heart-and-blood:effort} l'annonçait~; voici l'ordre de
marche complet)~:
\begin{enumerate}
  \item la \emph{\omterm{def:g7:what-is-respiration:respiration}{ventilation}} s'approfondit et s'accélère --- elle
        recharge le \omterm{prop:g4:journey-of-food:absorb}{sang} plus vite aux \omterm{def:g7:how-animals-breathe:four}{poumons}
        (\cref{ex:g4:breathing:running})~;
  \item le \emph{\omterm{def:g4:heart-and-blood:heart}{cœur}} bat plus vite et plus fort --- davantage de
        tournées par minute
        (\cref{prop:g4:heart-and-blood:pulse})~;
  \item le \omterm{prop:g4:journey-of-food:absorb}{sang} est \emph{redirigé}~: les vaisseaux qui desservent
        les \omterm{def:g3:bones-and-muscles:muscle}{muscles} au travail (et la \omterm{def:g1:the-five-senses:sense}{peau} qui refroidit)
        s'élargissent, tandis que ceux du \omterm{def:g4:journey-of-food:tract}{tube digestif} au repos se
        rétrécissent --- le même \omterm{prop:g4:journey-of-food:absorb}{sang}, réacheminé vers l'étage qui
        dépense.
\end{enumerate}
Les trois suivent le travail automatiquement, pilotés par les
services que le \omterm{def:g5:brain-in-command:system}{cerveau} assure sans qu'on les lui demande
(\cref{rem:g5:brain-in-command:more}).
\end{proposition}
\admitted

\begin{example}[L'ordre de marche, ressenti de l'intérieur]\label{ex:g7:organs-at-work:felt}
Le point fait après le sprint dans l'escalier, ligne par ligne~: le
souffle emballé --- le rechargement~; le \omterm{def:g4:heart-and-blood:heart}{cœur} qui cogne --- des
tournées plus rapides~; les \omterm{def:g1:my-body:parts}{jambes} qui brûlent --- la facture des
\omterm{def:g3:bones-and-muscles:muscle}{muscles} payée à la limite~; la \omterm{def:g1:the-five-senses:sense}{peau} chaude et rouge --- la chaleur
de la chaufferie portée jusqu'à la surface de refroidissement. Même
le classique point de côté et la lourdeur après un gros repas y
trouvent leur place~: la \omterm{def:g4:journey-of-food:digestion}{digestion} et le sprint se disputent le \omterm{prop:g4:journey-of-food:absorb}{sang}
réacheminé, et c'est pourquoi
\cref{met:g4:journey-of-food:habits} déconseillait de courir le
ventre plein.
\end{example}

\begin{definition}[Récupération]\label{def:g7:organs-at-work:recovery}
Quand l'effort s'arrête, le \omterm{prop:g4:heart-and-blood:pulse}{pouls} et la \omterm{def:g7:what-is-respiration:respiration}{ventilation} ne s'arrêtent
pas~: ils ne redescendent que peu à peu --- c'est la phase de
\emph{récupération}\index{récupération}. Le corps solde ses dettes~:
il recharge ses réserves de \omterm{def:g7:what-is-respiration:respiration}{dioxygène}, évacue les déchets de
l'effort, se débarrasse de la chaleur accumulée. Plus un corps est
entraîné, plus sa récupération est rapide --- et c'est pourquoi le
temps de récupération est lui-même une mesure de la condition
physique.
\end{definition}

\section{L'entraînement, et les limites de la machine}

\begin{proposition}[L'entraînement relève le plafond]\label{prop:g7:organs-at-work:training}
Exercée régulièrement, toute la chaîne d'approvisionnement se
renforce (\cref{rem:g4:heart-and-blood:rest},
\cref{met:g4:breathing:care})~: le \omterm{def:g4:heart-and-blood:heart}{cœur} déplace plus de \omterm{prop:g4:journey-of-food:absorb}{sang} à
chaque battement, les \omterm{def:g3:bones-and-muscles:muscle}{muscles} respiratoires gagnent en ampleur, les
\omterm{def:g3:bones-and-muscles:muscle}{muscles} eux-mêmes grossissent et stockent plus de carburant, et
leurs fins vaisseaux se multiplient. Résultat~: le même effort coûte
à un corps entraîné un \omterm{prop:g4:heart-and-blood:pulse}{pouls} plus bas, et son plafond --- l'effort
le plus dur qu'il puisse payer --- s'élève. La machine est rebâtie
par son propre usage.
\end{proposition}
\admitted

\begin{example}[Deux coureurs, une même côte]\label{ex:g7:organs-at-work:runners}
Un coureur non entraîné et un coureur entraîné montent la même \omterm{def:g3:bones-and-muscles:skeleton}{côte}.
Non entraîné~: \omterm{prop:g4:heart-and-blood:pulse}{pouls} à $180$ à mi-pente, essoufflé, obligé de
marcher --- les lignes d'approvisionnement ne peuvent pas payer la
facture, et la dette impose l'arrêt. Entraîné~: \omterm{prop:g4:heart-and-blood:pulse}{pouls} à $150$ au
sommet, la parole possible, \omterm{def:g7:organs-at-work:recovery}{récupération} en trois minutes. Même
\omterm{def:g3:bones-and-muscles:skeleton}{côte}, même facture --- des plafonds différents, bâtis par des mois
de \omterm{def:g3:bones-and-muscles:skeleton}{côtes} exactement comme celle-là, prises tranquillement.
\end{example}

\begin{method}[Lire sa propre machine]\label{met:g7:organs-at-work:read}
Une expérience sur soi, avec une montre
(\cref{met:g4:heart-and-blood:pulse} fournit la technique de prise
du \omterm{prop:g4:heart-and-blood:pulse}{pouls})~:
\begin{enumerate}
  \item note le \omterm{prop:g4:heart-and-blood:pulse}{pouls} au repos, assis et calme~;
  \item fais trois minutes d'effort régulier --- montées sur une
        marche, corde à sauter~;
  \item note le \omterm{prop:g4:heart-and-blood:pulse}{pouls} aussitôt après, puis chaque minute jusqu'à ce
        qu'il revienne au repos~: c'est le temps de \omterm{def:g7:organs-at-work:recovery}{récupération}~;
  \item recommence chaque semaine pendant un trimestre d'exercice
        régulier~: regarde le \omterm{prop:g4:heart-and-blood:pulse}{pouls} de repos, le \omterm{prop:g4:heart-and-blood:pulse}{pouls} d'effort et
        le temps de \omterm{def:g7:organs-at-work:recovery}{récupération} descendre tous les trois --- la
        reconstruction de
        \cref{prop:g7:organs-at-work:training}, dans tes propres
        données.
\end{enumerate}
\end{method}

\section{Problème~: l'étude de condition physique de l'école}

\begin{problem}[{Problème du week-end --- un trimestre de données de
pouls, lues en physiologiste}]\label{pb:g7:organs-at-work:1}
La classe applique \cref{met:g7:organs-at-work:read} chaque lundi
pendant un trimestre~: \omterm{prop:g4:heart-and-blood:pulse}{pouls} de repos, \omterm{prop:g4:heart-and-blood:pulse}{pouls} après trois minutes de
montées sur une marche, et temps de \omterm{def:g7:organs-at-work:recovery}{récupération}. Deux relevés
anonymisés~: élève A --- semaine 1~: repos $84$, effort $168$,
\omterm{def:g7:organs-at-work:recovery}{récupération} $9$ minutes~; semaine 12~: repos $72$, effort $150$,
\omterm{def:g7:organs-at-work:recovery}{récupération} $5$ minutes. Élève B --- semaine 1~: repos $70$, effort
$150$, \omterm{def:g7:organs-at-work:recovery}{récupération} $4$ minutes~; semaine 12~: repos $69$, effort
$148$, \omterm{def:g7:organs-at-work:recovery}{récupération} $4$ minutes.

\medskip\noindent\textbf{Partie I --- À l'intérieur d'une séance.}
\begin{enumerate}
  \item Pendant les montées, énumère ce qui arrive au prélèvement
        de \omterm{def:g7:what-is-respiration:respiration}{dioxygène} et de \omterm{prop:g7:organs-at-work:bill}{glucose} d'un \omterm{def:g3:bones-and-muscles:muscle}{muscle} de cuisse, et à ses
        rejets de \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} et de chaleur.
  \item Explique pourquoi la \omterm{def:g7:what-is-respiration:respiration}{ventilation} et le \omterm{prop:g4:heart-and-blood:pulse}{pouls} montent
        \emph{ensemble} pendant les minutes d'effort.
  \item Un élève se sent chaud et rouge à la troisième minute.
        Donne le récit de la chaufferie et du refroidissement.
  \item Pourquoi le \omterm{prop:g4:heart-and-blood:pulse}{pouls} ne retombe-t-il pas au repos dès l'arrêt
        des montées~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Lire le trimestre.}
\begin{enumerate}[resume]
  \item Décris les changements de A au fil du trimestre, et nomme
        les reconstructions de
        \cref{prop:g7:organs-at-work:training} derrière chaque
        nombre.
  \item Les nombres de B n'ont presque pas bougé. Donne les deux
        lectures honnêtes --- et dis quel fait supplémentaire sur
        la vie de B hors de la classe trancherait entre elles.
  \item Semaine 5, un élève se mesure juste après un déjeuner
        avalé en vitesse et obtient des nombres bizarres. Quelle
        règle de \cref{prop:g7:organs-at-work:supply} a interféré,
        et que devrait dire le protocole sur les horaires de
        mesure~?
  \item Pourquoi chaque élève doit-il toujours mesurer de la même
        façon --- même effort, mêmes moments~? (Une vieille
        connaissance parmi les méthodes~; nomme-la.)
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La lettre du
physiologiste.}
\begin{enumerate}[resume]
  \item Rédige en deux phrases le résumé du trimestre de A pour le
        panneau d'affichage de la classe, en employant
        \emph{plafond} et \emph{\omterm{def:g7:organs-at-work:recovery}{récupération}}.
  \item Un parent demande si un \omterm{prop:g4:heart-and-blood:pulse}{pouls} de repos bas est inquiétant.
        Réponds avec le cycliste de
        \cref{exo:g4:heart-and-blood:11}.
  \item La classe veut un seul nombre à suivre le trimestre
        prochain. Plaide pour le temps de \omterm{def:g7:organs-at-work:recovery}{récupération} plutôt que
        pour le \omterm{prop:g4:heart-and-blood:pulse}{pouls} d'effort, en te servant de
        \cref{def:g7:organs-at-work:recovery}.
  \item Termine par la morale de l'étude en une phrase~: qu'est-ce
        qui a rebâti la machine de A~?
\end{enumerate}
\end{problem}
