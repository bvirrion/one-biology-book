\section*{अध्याय \ref{ch:g2:animals-grow-up} --- जंतु कैसे बड़े होते हैं}

\begin{solution}{exo:g2:animals-grow-up:1}
माँ के दिए \omterm{def:g2:animals-grow-up:born}{अंडे} से (चूज़ा, मेंढक, घोंघा), या माँ के शरीर से \omterm{def:g2:animals-grow-up:born}{जीवित जन्म}
लेकर और उसका \omterm{def:g2:animals-grow-up:born}{दूध} पीकर (बिल्ली का बच्चा, बछड़ा, मानव शिशु)।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:2}
(क) \omterm{def:g2:animals-grow-up:born}{अंडे} से। (ख) \omterm{def:g2:animals-grow-up:born}{जीवित जन्म}। (ग) \omterm{def:g2:animals-grow-up:born}{अंडे} से। (घ) \omterm{def:g2:animals-grow-up:born}{जीवित जन्म}। (ङ) \omterm{def:g2:animals-grow-up:born}{अंडे}
से।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:3}
अपनी माँ का \omterm{def:g2:animals-grow-up:born}{दूध}। इस अध्याय के सभी \omterm{def:g2:animals-grow-up:born}{जीवित जन्म} लेने वाले \omterm{def:g1:animals-around-us:animal}{जंतु} ---
पिल्ले, बछड़े, ख़रगोश के बच्चे, मानव शिशु --- भी \omterm{def:g2:animals-grow-up:born}{दूध} से ही शुरू करते
हैं।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:4}
\omterm{def:g2:animals-grow-up:metamorphosis}{कायांतरण} शरीर की बनावट का वह बड़ा बदलाव है जिससे कुछ नन्हे \omterm{def:g1:animals-around-us:animal}{जंतु} वयस्क
बनने के लिए गुज़रते हैं। \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} का मेंढक बनना, और \omterm{ex:g2:animals-grow-up:butterfly}{इल्ली} का तितली
बनना।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:5}
\omterm{def:g2:animals-grow-up:born}{अंडे}, \omterm{ex:g2:animals-grow-up:frog}{टैडपोल}, \omterm{def:g1:my-body:parts}{टाँगों} वाला \omterm{ex:g2:animals-grow-up:frog}{टैडपोल}, मेंढक।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:6}
\omterm{def:g2:animals-grow-up:born}{अंडा}, \omterm{ex:g2:animals-grow-up:butterfly}{इल्ली}, \omterm{ex:g2:animals-grow-up:butterfly}{प्यूपा}, तितली। \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} \omterm{ex:g2:animals-grow-up:butterfly}{इल्ली} ही खाती है और लगभग सारी
वृद्धि भी वही करती है।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:7}
बछेड़ा --- पहले ही दिन से छोटा घोड़ा। उसके लिए बड़ा होने का मतलब है
वही बनावट रखते हुए बड़ा और मज़बूत होते जाना।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:8}
उन्हें गरम रखने के लिए। हर गरम \omterm{def:g2:animals-grow-up:born}{अंडे} के भीतर एक चूज़ा बन रहा है, जिसे
ज़र्दी खिलाती है, जब तक वह \omterm{def:g1:animals-around-us:covering}{खोल} तोड़कर बाहर आने लायक़ न हो जाए।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:9}
वह \omterm{def:g2:animals-grow-up:metamorphosis}{कायांतरण} के बीच का \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} है: पिछली \omterm{def:g1:my-body:parts}{टाँगें} उग चुकी हैं, अगली
\omterm{def:g1:my-body:parts}{टाँगें} अभी नहीं आईं। इससे पहले वह बिना \omterm{def:g1:my-body:parts}{टाँगों} वाला \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} था (और उससे
पहले एक \omterm{def:g2:animals-grow-up:born}{अंडा}); आगे अगली \omterm{def:g1:my-body:parts}{टाँगें} आएँगी, पूँछ सिमटेगी, और वह छोटा मेंढक
बन जाएगा।
\end{solution}

\begin{solution}{exo:g2:animals-grow-up:10}
कोई आकर बसा नहीं --- \omterm{ex:g2:animals-grow-up:butterfly}{इल्ली} और तितली एक ही \omterm{def:g1:animals-around-us:animal}{जंतु} हैं। \omterm{ex:g2:animals-grow-up:butterfly}{प्यूपा} के भीतर
\omterm{ex:g2:animals-grow-up:butterfly}{इल्ली} का शरीर तितली के शरीर में नए सिरे से बना दिया गया था: वही
नए सिरे से बनना \omterm{def:g2:animals-grow-up:metamorphosis}{कायांतरण} कहलाता है।
\end{solution}
