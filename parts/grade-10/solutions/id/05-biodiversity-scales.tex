\section*{Bab \ref{ch:g10:biodiversity-scales} --- Keanekaragaman Hayati pada Setiap Skala}

\begin{solution}{exo:g10:biodiversity-scales:1}
\omterm{def:g10:biodiversity-scales:biodiversity}{Ekosistem} (sebuah kolam, hutan beech); \omterm{def:g10:biodiversity-scales:species}{spesies} (katak, salamander air
dan buluh di kolam itu); \omterm{def:g10:universal-dna:gene}{gen} (\omterm{def:g10:universal-dna:gene}{alel} \omterm{def:g10:universal-dna:gene}{gen} ABO pada sebuah populasi
manusia).
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:2}
Kelompok organisme serupa yang berbiak bersama dan memberi keturunan
yang subur. Singa dan harimau memberi hibrid yang mandul, sehingga
keduanya dua \omterm{def:g10:biodiversity-scales:species}{spesies}.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:3}
$\num{1000000}/6400 \approx 156$: sekitar 150 kali lebih banyak.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:4}
Bagian seluruh salinan sebuah \omterm{def:g10:universal-dna:gene}{gen} dalam sebuah populasi yang berupa
\omterm{def:g10:universal-dna:gene}{alel} tertentu. B $= 1 - 0.66 - 0.27 = 0.07$.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:5}
Lima; yang terbesar adalah kepunahan akhir Perem, 252 juta tahun lalu,
yang menyingkirkan sekitar 95\% \omterm{def:g10:biodiversity-scales:species}{spesies} laut.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:6}
Yang kedua: kekayaannya sama, tetapi kelimpahan nisbinya jauh lebih
merata. Kekayaan sendirian tidak membedakan keduanya; kelimpahan
nisbinyalah yang membedakan.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:7}
Ladangnya sama; temannya mencuplik sepuluh kali lebih banyak. \omterm{def:g10:biodiversity-scales:species}{Spesies}
yang ditemukan bertambah seiring usaha sampai lengkungnya mendatar,
sehingga hitungan dengan usaha yang berbeda tidak sebanding --- angka 8
murid itu adalah cuplikan yang kurang, bukan ladang yang lebih miskin.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:8}
Segelintir yang selamat hanya membawa sebagian \omterm{def:g10:universal-dna:gene}{alel} \omterm{def:g10:biodiversity-scales:species}{spesiesnya};
keturunannya, sebanyak apa pun, berbagi himpunan yang telah menyusut
itu. \omterm{def:g10:biodiversity-scales:biodiversity}{Keanekaragaman genetiknya} hilang dalam leher botol itu dan tidak
dipulihkan oleh jumlah: \omterm{def:g10:biodiversity-scales:species}{spesiesnya} tidak punya cadangan apa pun
terhadap penyakit baru atau iklim yang berubah (hal citah).
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:9}
Sekitar 560 famili sebelumnya, 300 sesudahnya: $260/560 \approx 45\%$
famili hilang. Sebuah famili lenyap hanya ketika seluruh \omterm{def:g10:biodiversity-scales:species}{spesiesnya}
lenyap, sehingga banyak famili bertahan lewat satu atau dua \omterm{def:g10:biodiversity-scales:species}{spesies}
sambil kehilangan sisanya; bagian \omterm{def:g10:biodiversity-scales:species}{spesies} yang hilang (sekitar
95\%) jauh lebih besar daripada bagian familinya.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:10}
Setiap populasi membawa ketiga \omterm{def:g10:universal-dna:gene}{alelnya}, dan anggotanya berbeda
antarsesamanya dalam \omterm{def:g10:universal-dna:gene}{alel} mana yang dibawanya; beda antarpopulasinya
hanya berupa pergeseran frekuensi. Karena itu sebagian besar
\omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman genetik} \omterm{def:g10:biodiversity-scales:species}{spesiesnya} ada di dalam tiap populasi: dua
tetangga lazimnya berbeda pada sebanyak kedudukan seperti dua orang
dari tempat yang berjauhan.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:11}
\omterm{def:g10:biodiversity-scales:biodiversity}{Ekosistem}: kolamnya sendiri, beserta masyarakatnya. \omterm{def:g10:biodiversity-scales:species}{Spesies}: salamander
air atau capung yang setempat hanya hidup di sana. \omterm{def:g10:universal-dna:gene}{Gen}: \omterm{def:g10:universal-dna:gene}{alel} yang
dibawa populasi katak kolam itu, yang mungkin berbeda dari \omterm{def:g10:universal-dna:gene}{alel} kolam
yang lain.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:12}
Dengan umur sekitar $\num{2e6}$ tahun, $\num{2e6}$ \omterm{def:g10:biodiversity-scales:species}{spesies} memberi
sekitar satu kepunahan tiap tahun. Seribu kali lebih banyak berarti
sekitar 1000 \omterm{def:g10:biodiversity-scales:species}{spesies} tiap tahun, $\num{100000}$ tiap abad --- 5\%
\omterm{def:g10:biodiversity-scales:species}{spesies} yang dikenal dalam seratus tahun.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:13}
Kepunahan itu mengosongkan cara hidup --- pemakan tumbuhan yang besar,
pemangsa yang besar, hewan terbang, hewan perenang --- yang tadinya
ditempati dinosaurus. Garis keturunan mamalia yang selamat, yang bebas
dari pesaing itu dan punya \omterm{def:g10:biodiversity-scales:biodiversity}{keanekaragaman genetiknya} sendiri untuk
disandari, menyesuaikan diri dengan peran yang kosong itu dan terpecah
menjadi \omterm{def:g10:biodiversity-scales:species}{spesies} baru: kehilangannya membuka peluang yang lalu diisi
keragaman baru.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:14}
Gandum budi daya telah kehilangan sebagian besar \omterm{def:g10:universal-dna:gene}{alel} moyangnya.
Kerabat liarnya menyimpannya: \omterm{def:g10:universal-dna:gene}{alel} ketahanan terhadap sebuah penyakit,
terhadap kekeringan, terhadap garam, yang tidak dibawa varietas mana
pun saat ini. Menyilangkannya ke dalam gandum memulihkan pilihan itu;
begitu satu kerabat liar punah, \omterm{def:g10:universal-dna:gene}{alelnya} hilang untuk selamanya.
\end{solution}

\begin{solution}{exo:g10:biodiversity-scales:15}
Kepunahan itu alami, tetapi dengan sekitar satu \omterm{def:g10:biodiversity-scales:species}{spesies} tiap tahun;
laju masa kini ratusan sampai seribu kali lebih cepat, setara dengan
\omterm{prop:g10:biodiversity-scales:changes}{kepunahan massal}, dan pemulihan dari kepunahan semacam itu menuntut
jutaan tahun --- lebih lama daripada usia \omterm{def:g10:biodiversity-scales:species}{spesies} kita. Sementara itu
\omterm{def:g10:biodiversity-scales:biodiversity}{ekosistem} yang sedang hilang memasok pangan, penyerbukan, tanah,
penjernihan air dan cadangan genetik tanaman budi daya kita. Contoh
masa lalunya menunjukkan bahwa kehilangannya nyata dan pemulihannya
tidak tersedia pada skala waktu manusia mana pun.
\end{solution}

\begin{solution}{pb:g10:biodiversity-scales:1}
\textbf{1.} 31 dan 9 \omterm{def:g10:biodiversity-scales:species}{spesies}: nisbahnya $31/9 \approx 3.4$.

\textbf{2.} Ladang A: tidak ada \omterm{def:g10:biodiversity-scales:species}{spesies} di atas 22\% berhadapan dengan
satu \omterm{def:g10:biodiversity-scales:species}{spesies} pada 84\% di Ladang B. Kemerataan menyatakan bagaimana
individunya terbagi di antara \omterm{def:g10:biodiversity-scales:species}{spesies} yang ditemukan; masyarakat yang
kaya tetapi tidak merata dikuasai satu \omterm{def:g10:biodiversity-scales:species}{spesies} dan sisanya langka.

\textbf{3.} Tambahannya menyusut (7, 4, 2): lengkungnya sedang
mendatar, sehingga cuplikannya mendekati lengkap; beberapa petak lagi
paling banyak akan menambah satu atau dua \omterm{def:g10:biodiversity-scales:species}{spesies}.

\textbf{4.} Lengkung Ladang B mencapai 9 pada petak ke-15 dan tidak
menambah apa pun setelahnya: lengkung pencuplikan yang lengkap mendatar
menjadi dataran, yang telah dicapai Ladang B dan hampir dicapai Ladang A.

\textbf{5.} Banyak tumbuhan hanya terlihat atau dapat dikenali ketika
berbunga, dan serangga serta burung berubah menurut musim; bulan yang
berbeda akan mengubah hitungannya karena alasan yang tidak berhubungan
dengan pagarnya.

\textbf{6.} Serangga $46/14 \approx 3.3$; burung $17/5 = 3.4$. Ketiga
kelompoknya memberi faktor sekitar 3: cerita yang sama.

\textbf{7.} Pagarnya memasok tumbuhan (daun, bunga, kayu mati,
naungan) yang menjadi makanan dan tempat berbiak serangganya; \omterm{def:g10:biodiversity-scales:species}{spesies}
tumbuhan yang lebih sedikit berarti \omterm{def:g10:biodiversity-scales:species}{spesies} serangga yang lebih
sedikit; serangga serta buah dan tempat bersarang pagarnya memberi
makan dan rumah bagi burungnya, sehingga keragaman burungnya ikut turun.

\textbf{8.} Tidak: panennya hanya mengukur hasil taninya. Sebuah pagar
dapat menaungi ladangnya dari angin, menahan tanahnya, merumahkan
penyerbuk dan pemangsa hama taninya, dan pengaruhnya muncul pada tahun
yang buruk atau selama puluhan tahun, bukan pada angka satu musim.

\textbf{9.} Burung yang bersarang di pagar seperti burung hitam atau
burung pipit pagar (atau siput pagar, kepik perisai hawthorn): ia
memerlukan semak untuk bersarang, makan atau bernaung, yang tidak lagi
ditawarkan Ladang B.

\textbf{10.} \omterm{def:g10:biodiversity-scales:biodiversity}{Ekosistem}: pagarnya sendiri, yaitu sebuah habitat, telah
disingkirkan, dan \omterm{def:g10:biodiversity-scales:species}{spesies} serta \omterm{def:g10:universal-dna:gene}{gennya} ikut lenyap bersamanya;
kehilangan pada skala lainnya adalah akibatnya.

\textbf{11.} Ladang A: salinan y $2 \times 100 + 200 = 400$, salinan b
$200 + 2 \times 100 = 400$, dari 800: frekuensi y 0.5. Ladang B: y
$8 + 32 = 40$, b $32 + 128 = 160$, dari 200: frekuensi y 0.2.

\textbf{12.} Ladang A: kedua \omterm{def:g10:universal-dna:gene}{alelnya} sama sering dan 50\% siputnya
membawa keduanya; Ladang B: 32\% membawa keduanya dan b berkuasa.
Ladang A lebih beragam.

\textbf{13.} Pada tanggul yang gundul, siput kuning lebih sering
dimakan; dari keturunan ke keturunan, salinan y yang diwariskan makin
sedikit, dan frekuensi y turun dari sekitar 0.5 menjadi 0.2.

\textbf{14.} Keuntungannya akan berbalik: siput kuning akan lebih baik
bertahan dalam pagar yang teduh dan frekuensi y akan naik lagi
sepanjang beberapa keturunan --- asalkan \omterm{def:g10:universal-dna:gene}{alelnya} masih ada.

\textbf{15.} Siput Ladang B: dalam populasi beberapa ratus, satu tahun
yang buruk atau beberapa kematian kebetulan dapat menyingkirkan setiap
salinan \omterm{def:g10:universal-dna:gene}{alel} yang lebih langka; di antara puluhan ribu, \omterm{def:g10:universal-dna:gene}{alelnya} dibawa
ribuan siput dan kebetulan tidak dapat menghapusnya.

\textbf{16.} Ladang A: $31 + 46 + 17 = 94$; Ladang B: $9 + 14 + 5 = 28$;
nisbahnya $94/28 \approx 3.4$.

\textbf{17.} Tiga persen lahannya memuat habitat sekitar dua pertiga
\omterm{def:g10:biodiversity-scales:species}{spesies} ladangnya.

\textbf{18.} Tumbuhan $24/31 \approx 77\%$; serangga $35/46 \approx 76\%$;
burung $12/17 \approx 71\%$.

\textbf{19.} \omterm{def:g10:biodiversity-scales:biodiversity}{Keanekaragaman genetiknya}: \omterm{def:g10:biodiversity-scales:species}{spesiesnya} kembali dari
populasi tetangga, tetapi \omterm{def:g10:universal-dna:gene}{alel} yang hilang dari populasi setempat
(siput Ladang B, misalnya) kembali hanya jika pendatangnya kebetulan
membawanya; \omterm{def:g10:universal-dna:gene}{alel} yang hilang tidak diciptakan ulang oleh penanaman kembali.

\textbf{20.} Pagar tanaman melipatgandakan kekayaan \omterm{def:g10:biodiversity-scales:species}{spesies} ladangnya
sekitar 3.4 kali pada setiap kelompok yang terhitung; pagar yang
ditanam kembali mengembalikan \omterm{def:g10:biodiversity-scales:biodiversity}{ekosistemnya} dan, seiring waktu,
sebagian besar \omterm{def:g10:biodiversity-scales:species}{spesiesnya}, tetapi tidak dengan sendirinya \omterm{def:g10:universal-dna:gene}{alel} yang
telah hilang dari populasi setempat.
\end{solution}
