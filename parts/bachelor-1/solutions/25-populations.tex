\section*{Chapter \ref{ch:b1:populations} --- Populations and Demography}

\begin{solution}{exo:b1:populations:1}
\omterm{def:b1:populations:population}{Population}: the individuals of one species in one place and time,
able to interbreed. Density: individuals per unit area or volume.
Dispersion: their spatial pattern --- clumped (a herd, a patch of
nettles), uniform (nesting gannets a peck apart, creosote bushes),
random (dandelions on a lawn).
\end{solution}

\begin{solution}{exo:b1:populations:2}
$l_x$: fraction of a cohort alive at age $x$. $m_x$: daughters per
female of age $x$. $R_0 = \sum l_x m_x$: daughters per newborn female
over her life. $T = \sum x l_x m_x / R_0$: mean age of mothers.
$R_0 = 1$: each female replaces herself; the \omterm{def:b1:populations:population}{population} is stable.
\end{solution}

\begin{solution}{exo:b1:populations:3}
Type III: about 15 of 1000, \qty{1.5}{\%}. Type I: about 700 of
1000, \qty{70}{\%}.
\end{solution}

\begin{solution}{exo:b1:populations:4}
$\dd N/\dd t = rN$ and $\dd N/\dd t = rN(1 - N/K)$. $r$ is the
per-capita growth rate when resources are unlimited (births minus
deaths per individual per unit time); $K$ the \omterm{thm:b1:populations:logistic}{carrying capacity}, the
size at which growth stops.
\end{solution}

\begin{solution}{exo:b1:populations:5}
$r = \ln(1000)/5 = \qty{1.38}{h^{-1}}$; doubling $\ln 2/1.38 =
\qty{0.5}{h}$.
\end{solution}

\begin{solution}{exo:b1:populations:6}
$l_x m_x = 0, 0.5, 0.6, 0.2, 0$: $R_0 = 1.3$. $T = (1\times 0.5 +
2\times 0.6 + 3\times 0.2)/1.3 = 2.3/1.3 = 1.77$. $r = \ln 1.3/1.77 =
\qty{0.15}{}$ per unit time: it grows.
\end{solution}

\begin{solution}{exo:b1:populations:7}
$N = 50\times 70/7 = 500$. Only 40 marked fish remained: $N = 40\times
70/7 = 400$; the first estimate was too high by a quarter (the marked
fraction of the \omterm{def:b1:populations:population}{population} was smaller than assumed).
\end{solution}

\begin{solution}{exo:b1:populations:8}
$0.2\times 100\times 0.9 = 18$; $0.2\times 500\times 0.5 = 50$;
$0.2\times 900\times 0.1 = 18$ per year. Maximum $rK/4 = 50$ a year, at
$N = 500$.
\end{solution}

\begin{solution}{exo:b1:populations:9}
Dandelion: $r$ (thousands of seeds, annual). Elephant: $K$ (one calf
in four years, long life, care). Cod: $r$ (millions of eggs, no care).
Gorilla: $K$ (one infant in four years, decades of life). Housefly: $r$
(hundreds of eggs, matures in ten days).
\end{solution}

\begin{solution}{exo:b1:populations:10}
$r = \ln(29/10)/2 = \qty{0.53}{h^{-1}}$; $K \approx 660$. At \qty{8}{h}:
$660/(1 + 65e^{-4.24}) = 660/(1 + 0.94) = 340$, against 351 counted.
Fastest at $K/2 = 330$, reached at $t = \ln 65/0.53 = \qty{7.9}{h}$,
where $\dd N/\dd t = rK/4 = 87\times 10^5$ \omterm{def:b1:cell-unit-of-life:cell}{cells} per millilitre per
hour.
\end{solution}

\begin{solution}{exo:b1:populations:11}
Regulation requires the per-capita rate to fall as $N$ rises, so that
$N$ returns toward an equilibrium after a disturbance; a factor that
kills a fixed fraction changes $N$ but not the dependence of the rate
on $N$, so after it the \omterm{def:b1:populations:population}{population} resumes the same growth. Weather
limits insects by striking often enough, and at random, that the
\omterm{def:b1:populations:population}{population} is set back before it can approach any ceiling: an
unregulated \omterm{def:b1:populations:population}{population} held down by repeated disturbance, whose
average size depends on the frequency of bad seasons rather than on a
\omterm{thm:b1:populations:logistic}{carrying capacity}.
\end{solution}

\begin{solution}{exo:b1:populations:12}
Two \omterm{def:b1:populations:population}{populations} of the same size, one with a broad young base and one
with even age classes, have different futures: the first will grow
for decades even if its $m_x$ falls to replacement now, because its
many young have yet to reproduce (demographic momentum); the second
will not. $N$ records the past; the age distribution and the
schedule of $m_x$ over it are what the equations act on, and the
pyramid predicts the next generation's size while $N$ says nothing
about it.
\end{solution}

\begin{solution}{pb:b1:populations:1}
\textbf{1.} $r = \ln(29/10)/2 = \qty{0.53}{h^{-1}}$.
\textbf{2.} $\ln 2/0.53 = \qty{1.3}{h}$.
\textbf{3.} $K \approx 660\times 10^5 = \qty{6.6e7}{}$ \omterm{def:b1:cell-unit-of-life:cell}{cells} per
millilitre.
\textbf{4.} $N(t) = 660/(1 + 65e^{-0.53t})$: at 4 h, $660/(1 + 7.8) =
75$ (71 counted); at 8 h, 340 (351); at 12 h, $660/(1 + 0.112) = 594$
(595).
\textbf{5.} At $K/2 = 330$, reached at $t = \ln 65/0.53 = \qty{7.9}{h}$;
maximum rate $rK/4 = 0.53\times 660/4 = 87\times 10^5$ \omterm{def:b1:cell-unit-of-life:cell}{cells} per
millilitre per hour.
\textbf{6.} $10e^{0.53\times 12} = 10\times 578 = 5780$: ten times the
observed 595.
\textbf{7.} Exhaustion of the sugar (or oxygen); accumulation of
ethanol and acids that poison the \omterm{def:b1:cell-unit-of-life:cell}{cells}.
\textbf{8.} $q_0 = 0.40$; $q_4 = 1 - 0.34/0.40 = 0.15$: heavy early
mortality then a steadier rate --- between type II and III, as for
most large mammals in the wild.
\textbf{9.} $l_x m_x = 0, 0, 0.312, 0.368, 0.320, 0.272, 0.182, 0.080,
0.018, 0$: $R_0 = 1.55$.
\textbf{10.} $\sum x l_x m_x = 0.624 + 1.104 + 1.280 + 1.360 + 1.092 +
0.560 + 0.144 = 6.16$; $T = 6.16/1.55 = \qty{4.0}{years}$.
\textbf{11.} $r = \ln 1.55/4.0 = \qty{0.11}{year^{-1}}$; doubling in
\qty{6.3}{years}.
\textbf{12.} $200e^{1.1} = 600$ deer.
\textbf{13.} $N(10) = 600/(1 + 2e^{-1.1}) = 600/1.67 = 360$; passes 500
when $(K - N_0)/N_0\,e^{-rt} = 0.2$, i.e.\ $e^{-rt} = 0.1$, $t = \ln
10/0.11 = 21$ years.
\textbf{14.} $120\times 150/18 = 1000$ perch.
\textbf{15.} 100 marked: $100\times 150/18 = 833$; the first estimate
was too high (fewer marked fish were at large than assumed).
\textbf{16.} $T_2 = \ln 2/0.4 = \qty{1.7}{years}$.
\textbf{17.} Catch $hN$ with $h = 0.2$: $rN(1 - N/K) = hN$ gives $N^* =
K(1 - h/r) = 1000\times 0.5 = 500$ and a catch of \num{100} a year ---
below the maximum sustainable yield, but a fraction adjusts itself to
the stock: if the stock halves, so does the catch, and the \omterm{def:b1:populations:population}{population}
returns.
\textbf{18.} $rK/4 = 100$ a year, at $N = 500$.
\textbf{19.} At 500: $0.4\times 500\times 0.5 = 100 < 120$: the
\omterm{def:b1:populations:population}{population} declines. At 300: $0.4\times 300\times 0.7 = 84 < 120$:
declines faster --- once below $K/2$ a fixed harvest above the maximum
yield drives it to extinction.
\textbf{20.} At $K/2$ the surplus is at its maximum but any error ---
a \omterm{def:b1:populations:population}{population} overestimated by a fifth, a bad year that cuts $r$ ---
turns a sustainable take into a decline, and below $K/2$ the surplus
shrinks as the \omterm{def:b1:populations:population}{population} does: the equilibrium is unstable to
overharvest. Harvesting somewhat above $K/2$ \omterm{def:b1:flowering-plant-organization:organs}{leaves} a margin.
\textbf{21.} $R_0$ halves to $0.78 < 1$: the herd shrinks until it
falls below 400 and $m_x$ recovers --- a density-dependent factor,
regulating the herd near 400.
\textbf{22.} No: it removes \qty{40}{\%} whatever the size and \omterm{def:b1:flowering-plant-organization:organs}{leaves}
the per-capita rate unchanged; in the following years the herd grows
back at the same $r$ from a lower start --- a disturbance, not a
regulation.
\textbf{23.} $0.11N(1 - N/600) = 30$: $N^2 - 600N + 163\,600 = 0$,
$N = 300\pm\sqrt{90\,000 - 163\,600}$ --- no real solution: 30 a year
exceeds the maximum surplus $rK/4 = 16.5$, and the wolves drive the
herd to extinction. With 15 a year: $N = 300\pm\sqrt{90\,000 -
81\,800} = 300\pm 91$, i.e.\ 209 and 391; the upper one is stable
(above it the surplus is less than 15 and the herd falls back, below
it more and it rises), the lower unstable.
\textbf{24.} Lynx numbers respond to food with a delay: a rise of hares
raises lynx births and survival over the following year or two, and a
crash of hares starves lynx only after their \omterm{def:b1:lipids:triglyceride}{fat} and the season's
kits are gone.
\textbf{25.} Yeast: $K = \qty{6.6e7}{}$ \omterm{def:b1:cell-unit-of-life:cell}{cells} per millilitre, $r =
\qty{0.53}{h^{-1}}$. Deer: $R_0 = 1.55$, $T = \qty{4.0}{years}$, $r =
\qty{0.11}{year^{-1}}$.
\end{solution}
