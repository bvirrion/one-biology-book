\section*{Hoofdstuk \ref{ch:g1:plants-around-us} --- Planten om ons heen}

\begin{solution}{exo:g1:plants-around-us:1}
\omterm{def:g1:plants-around-us:parts}{Wortels} (onder de grond), \omterm{def:g1:plants-around-us:parts}{stengel}, \omterm{def:g1:plants-around-us:parts}{bladeren}, \omterm{def:g1:plants-around-us:parts}{bloem}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:2}
(a) De \omterm{def:g1:plants-around-us:parts}{wortels}. (b) De \omterm{def:g1:plants-around-us:parts}{bladeren}. (c) De \omterm{def:g1:plants-around-us:parts}{bloem}. (d) De \omterm{def:g1:plants-around-us:parts}{stengel}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:3}
Het is uit een zaadje geboren, het eet (water uit de grond, licht uit
de lucht), het groeit, het maakt zaadjes waaruit nieuwe madeliefjes
komen, en het gaat op een dag dood. Alle kenmerken van leven zijn er;
lopen hoort daar niet bij.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:4}
De \omterm{def:g1:plants-around-us:parts}{stengel} van de \omterm{def:g1:plants-around-us:tree}{boom} is de stam; de \omterm{def:g1:plants-around-us:parts}{bladeren} worden door de takken
gedragen.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:5}
Water, licht en een plaats voor haar \omterm{def:g1:plants-around-us:parts}{wortels} --- meestal aarde.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:6}
Het water geven. De \omterm{def:g1:plants-around-us:plant}{plant} is verwelkt door gebrek aan water; een paar
uur nadat ze water heeft gekregen, staan haar \omterm{def:g1:plants-around-us:parts}{bladeren} weer rechtop.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:7}
Hetzelfde: allebei zijn het \omterm{def:g1:plants-around-us:plant}{planten}, met \omterm{def:g1:plants-around-us:parts}{wortels}, een \omterm{def:g1:plants-around-us:parts}{stengel}, \omterm{def:g1:plants-around-us:parts}{bladeren}
en \omterm{def:g1:plants-around-us:parts}{bloemen}, en allebei hebben ze water en licht nodig. Verschillend: de
\omterm{def:g1:plants-around-us:parts}{stengel} van de \omterm{def:g1:plants-around-us:tree}{boom} is een houtige stam terwijl de \omterm{def:g1:plants-around-us:parts}{stengel} van het
madeliefje zacht en groen is, en de \omterm{def:g1:plants-around-us:tree}{boom} is veel groter en leeft veel langer.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:8}
De \omterm{def:g1:plants-around-us:plant}{plant} bij het raam hoort groen en rechtop te staan; die uit de kast
bleek, uitgerekt en slap, ook al kreeg ze water. Wat ontbrak is licht
--- water alleen is niet genoeg.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:9}
Bijvoorbeeld: de \omterm{def:g1:plants-around-us:plant}{plant} groeit --- week na week hoger --- en ze draait
of buigt naar het licht toe, zoals de zonnebloem de dag volgt. Langzaam
is niet hetzelfde als niets doen.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:10}
De \omterm{def:g1:plants-around-us:parts}{wortel}. ``Onder de grond'' is in de figuur de plaats van de \omterm{def:g1:plants-around-us:parts}{wortels}
--- de \omterm{def:g1:plants-around-us:parts}{wortel} die wij eten is een grote, dikke oranje \omterm{def:g1:plants-around-us:parts}{wortel} die de
\omterm{def:g1:plants-around-us:plant}{plant} met voedsel heeft gevuld.
\end{solution}
