\chapter{Circulation and Breathing}\label{ch:g5:circulation}

\omterm{def:g5:cell-structure:cell}{Cells} need a constant supply of oxygen and \omterm{def:g4:plants-need:needs}{nutrients}, and they need
wastes removed. \omterm{def:g5:circulation:breathing}{Breathing} refreshes the \omterm{def:g5:circulation:blood}{blood} with oxygen; circulation
delivers that \omterm{def:g5:circulation:blood}{blood} around the \omterm{def:g1:my-body:body}{body}. This chapter links the two systems.

\section{Why transport matters}

\begin{proposition}[Delivery service]\label{prop:g5:circulation:need}
In large \omterm{def:g1:plants-animals:animal}{animals}, diffusion alone is too slow to supply every \omterm{def:g5:cell-structure:cell}{cell}.
\omterm{def:g5:circulation:blood}{Blood}, pumped by the \omterm{def:g5:circulation:heart}{heart}, carries materials where they are needed.
\end{proposition}
\admitted

\begin{definition}[Blood]\label{def:g5:circulation:blood}
\emph{Blood}\index{blood} is a liquid tissue that transports oxygen,
carbon dioxide, \omterm{def:g4:plants-need:needs}{nutrients}, hormones and wastes. It also helps fight
infection and distribute heat.
\end{definition}

\section{The heart and vessels}

\begin{definition}[Heart and vessels]\label{def:g5:circulation:heart}
The \emph{heart}\index{heart} is a muscular pump. \emph{Arteries}
\index{artery} carry \omterm{def:g5:circulation:blood}{blood} away from the heart; \emph{veins}\index{vein}
return \omterm{def:g5:circulation:blood}{blood} toward the heart; tiny \emph{capillaries}\index{capillary}
exchange materials with \omterm{def:g1:my-body:body}{body} \omterm{def:g5:cell-structure:cell}{cells}.
\end{definition}

\begin{example}[Pulse]\label{ex:g5:circulation:pulse}
You can feel a pulse where an \omterm{def:g5:circulation:heart}{artery} lies near the \omterm{def:g1:senses:five}{skin}. Pulse rate
rises with exercise because \omterm{def:g3:bones:muscle}{muscles} need more oxygen more quickly.
\end{example}

\section{Breathing and gas exchange}

\begin{definition}[Breathing]\label{def:g5:circulation:breathing}
\emph{Breathing}\index{breathing} moves \omterm{def:g4:plants-need:needs}{air} into and out of the
\emph{lungs}\index{lung}. In the lungs, oxygen enters the \omterm{def:g5:circulation:blood}{blood} and
carbon dioxide \omterm{def:g2:plant-parts:main}{leaves} it --- \emph{gas exchange}\index{gas exchange}.
\end{definition}

\begin{definition}[Respiratory path]\label{def:g5:circulation:path}
\omterm{def:g4:plants-need:needs}{Air} typically flows: \omterm{def:g1:senses:five}{nose}/\omterm{def:g3:teeth:path}{mouth} $\to$ windpipe $\to$ bronchi $\to$
smaller tubes $\to$ \omterm{def:g4:plants-need:needs}{air} sacs in the \omterm{def:g5:circulation:breathing}{lungs}, surrounded by \omterm{def:g5:circulation:heart}{capillaries}.
\end{definition}

\begin{proposition}[Systems linked]\label{prop:g5:circulation:link}
\omterm{def:g5:circulation:breathing}{Breathing} loads \omterm{def:g5:circulation:blood}{blood} with oxygen and unloads carbon dioxide.
Circulation carries oxygen-rich \omterm{def:g5:circulation:blood}{blood} to \omterm{def:g5:cell-structure:cell}{cells} and returns carbon
dioxide to the \omterm{def:g5:circulation:breathing}{lungs}.
\end{proposition}
\admitted

\section{Exercise and health}

\begin{method}[Looking after heart and lungs]\label{met:g5:circulation:health}
\begin{enumerate}
  \item Stay physically active to strengthen the \omterm{def:g5:circulation:heart}{heart} and improve
        fitness.
  \item Avoid tobacco smoke and polluted \omterm{def:g4:plants-need:needs}{air} when possible.
  \item Eat a balanced diet that supports healthy \omterm{def:g5:circulation:blood}{blood} vessels.
  \item Never block airways; learn basic first-aid rules for choking
        from a qualified \omterm{def:g3:life-cycles:butterfly}{adult}.
\end{enumerate}
\end{method}

\begin{example}[Climbing stairs]\label{ex:g5:circulation:stairs}
\omterm{def:g5:circulation:breathing}{Breathing} rate and \omterm{def:g5:circulation:heart}{heart} rate both rise. More oxygen is delivered to
\omterm{def:g1:my-body:parts}{leg} \omterm{def:g3:bones:muscle}{muscles}; more carbon dioxide is removed.
\end{example}
