\chapter{متكيفة مع عالمها}\label{ch:g4:adapted-to-their-world}

ضع دبًّا قطبيًّا في الصحراء الكبرى وجملًا على الجليد البحري، فلا
يدوم أي منهما موسمًا واحدًا. ومع ذلك فكل واحد في موطنه بطل: فالدب
يسبح البحار المتجمدة في راحة، والجمل يعبر صحارى تقضي على حصان في
أيام. والكائنات الحية تناسب عوالمها مناسبة دقيقة تستحق الدرس عن
قرب --- صفةً صفة.

\section{ما معنى التكيف}

\begin{definition}[التكيف]\label{def:g4:adapted-to-their-world:adaptation}
\emph{التكيف}\index{تكيف} صفة في جسم \omterm{def:g1:living-or-not:living}{الكائن الحي} أو سلوكه تلائمه
لظروف بيئته --- للبرد أو للجفاف، ولغذائه، ولصائديه. وقد بينت
\cref{prop:g2:where-animals-live:fit} أن \omterm{def:g1:animals-around-us:animal}{الحيوانات} تناسب موائلها؛
والتكيف هو تلك الملاءمة، منظورًا إليها قطعة قطعة.
\end{definition}

\begin{example}[الجمل، قطعة قطعة]\label{ex:g4:adapted-to-their-world:camel}
مشكلات الصحراء: الحرّ، وانعدام الماء، والرمل الرخو. وأجوبة الجمل:
سنام من الشحم --- \omterm{def:g3:animals-through-seasons:active}{مخزن غذاء} للعبور الشحيح؛ والقدرة على شرب مئة
لتر في دقائق ثم البقاء أيامًا بلا ماء؛ وقدمان عريضتان ذواتا
إصبعين لا تغوصان في الرمل؛ ومنخران ينغلقان أمام الغبار الطائر
وأهداب طويلة مزدوجة تناسبهما. فكل مشكلة صحراوية قوبلت بصفة.
\end{example}

\includegraphics[width=0.6\linewidth]{images/book1/ai/g4-adapted-camel.png}

\begin{example}[الدب القطبي، قطعة قطعة]\label{ex:g4:adapted-to-their-world:bear}
مشكلات الجليد البحري: البرد، والماء المتجمد، وفريسة يصعب الاقتراب
منها على جليد أبيض. وأجوبة الدب: \omterm{def:g1:animals-around-us:covering}{فرو} كثيف فوق طبقة عميقة من الشحم
--- عزل مزدوج؛ وكفوف عريضة مكفوفة قليلًا --- حذاء ثلج ومجداف
معًا؛ وفروة بيضاء --- غير مرئية على الثلج الذي يصطاد عليه. الدقة
نفسها، مضبوطة على العالم المقابل.
\end{example}

\begin{proposition}[التكيفات أجوبة عن مشكلات]\label{prop:g4:adapted-to-their-world:answers}
حيثما وضعت \omterm{def:g2:caring-for-nature:environment}{البيئة} مشكلة صعبة، حملت الكائنات الحية المقيمة هناك
صفات تجيب عنها:
\begin{itemize}
  \item البرد: \omterm{def:g1:animals-around-us:covering}{فرو} كثيف، وطبقات شحم، \omterm{def:g1:the-five-senses:sense}{وآذان} صغيرة تفقد دفئًا قليلًا؛
  \item والجفاف: تخزين للماء، وجلود شمعية، \omterm{def:g1:plants-around-us:parts}{وجذور} عميقة؛
  \item والحياة في الماء: أجسام انسيابية، وزعانف وأقدام مكفوفة،
        وأكسية تدفع الماء؛
  \item والصائدون في كل مكان: ألوان تمويه، ودروع، وأشواك، وسرعة،
        وعيون متباعدة تراقب ما حولها كله؛
  \item والصيد للعيش: حواسّ حادة، وعيون إلى الأمام، ومخالب وسرعة
        (\cref{prop:g2:what-animals-eat:match}).
\end{itemize}
\end{proposition}
\admitted

\section{والنباتات متكيفة أيضًا}

\begin{example}[الصبّار]\label{ex:g4:adapted-to-their-world:cactus}
الصحراء تختبر \omterm{def:g1:plants-around-us:plant}{النباتات} أشدّ اختبار --- فهي لا تستطيع أن تمشي إلى
الماء. وجواب الصبّار: \omterm{def:g1:plants-around-us:parts}{ساق} سميكة على هيئة برميل تخزن الماء وتقوم
بعمل \omterm{def:g1:plants-around-us:parts}{الأوراق} في التقاط الضوء؛ \omterm{def:g1:plants-around-us:parts}{وأوراق} تقلّصت أشواكًا، فلا تكاد تفقد
ماءً --- وهي تدافع عن البرميل أمام \omterm{def:g1:animals-around-us:animal}{الحيوانات} العطشى؛ \omterm{def:g1:plants-around-us:parts}{وجذور} تنتشر
عريضة قريبة من السطح لتلتقط زخّة نادرة كاملة دفعةً واحدة.
\end{example}

\begin{example}[وسائد الجبال]\label{ex:g4:adapted-to-their-world:alpine}
\omterm{def:g1:plants-around-us:plant}{نباتات} الجبال العالية تواجه ريحًا متجمدة. وكثير منها ينمو وسائد
متراصّة ملتصقة بالأرض: فداخل الوسادة تنكسر الريح ويبقى الهواء
ألطف --- شكل هو مأوًى في ذاته. أما بنات عمومتها في السهول فتنمو
طويلة متباعدة؛ وصور الجبال تعانق الصخر.
\end{example}

\begin{method}[قراءة عالم الكائن من جسمه]\label{met:g4:adapted-to-their-world:read}
إذا أُعطيت حيوانًا أو نباتًا غير مألوف:
\begin{enumerate}
  \item عدّد صفاته اللافتة --- الكساء، والأقدام، \omterm{def:g1:the-five-senses:sense}{والعينين}، وشكل
        الجسم، \omterm{def:g1:plants-around-us:parts}{والأوراق}، \omterm{def:g1:plants-around-us:parts}{والجذور}؛
  \item واسأل عن كل واحدة: \emph{أي مشكلة تحلّها هذه؟}
  \item واجمع الأجوبة في صورة لموطنه --- بارد أم حارّ، رطب أم جاف،
        صائد أم مَصيد؛
  \item وراجع تخمينك بمكان عيش الكائن حقًّا.
\end{enumerate}
وهذه هي \cref{met:g2:where-animals-live:matching} وقد كبرت: من
مطابقة \omterm{def:g2:where-animals-live:habitat}{موئل} إلى إعادة بناء عالم.
\end{method}

\begin{example}[الطريقة على جلد غامض]\label{ex:g4:adapted-to-their-world:mystery}
في درج متحف \omterm{def:g1:the-five-senses:sense}{جلد} بلا \omterm{prop:g3:balanced-diet:jobs}{بطاقة}: \omterm{def:g1:animals-around-us:covering}{فرو} ناعم كثيف، \omterm{def:g1:the-five-senses:sense}{وعينان} صغيرتان جدًّا،
ومخالب أمامية ضخمة على هيئة مجرفة. \omterm{def:g1:animals-around-us:covering}{فالفرو} يقول ثديي؛ والمجارف
تقول حفرًا؛ \omterm{def:g1:the-five-senses:sense}{والعينان} عديمتا النفع تقولان حياة في الظلام. والحكم:
حفّار تحت الأرض --- وهو فعلًا خُلد
(\cref{ex:g2:where-animals-live:mole})، قُرئ قراءة صحيحة من ثلاث
صفات فحسب.
\end{example}

\begin{remark}[كيف تحدث هذه الملاءمة؟]\label{rem:g4:adapted-to-their-world:how}
لا الجمل \emph{يقرر} أن \omterm{def:g2:from-seed-to-plant:germination}{تنبت} له قدمان عريضتان، ولا الدب القطبي
يختار فروته البيضاء. فكيف إذن صارت الكائنات الحية تناسب عوالمها
هذه المناسبة الحسنة؟ إنه من أعمق أسئلة \omterm{def:g1:living-or-not:living}{علم الأحياء}، وجوابه الكامل
--- كيف تتراكم الملاءمة جيلًا بعد جيل --- حكاية يدّخرها هذا الكتاب
لسنة لاحقة. أما الآن فتعلّم أن \emph{ترى} الملاءمة: فالرؤية هي
التي جعلت السؤال ممكنًا.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:1}
ما \omterm{def:g4:adapted-to-their-world:adaptation}{التكيف}؟ أعطِ مثالًا حيوانيًّا ومثالًا نباتيًّا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:2}
سمِّ ثلاث مشكلات صحراوية وجواب الجمل عن كل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:3}
سمِّ ثلاث مشكلات في الجليد البحري وجواب الدب القطبي عن كل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:4}
ما أعمال أشواك الصبّار؟ وماذا تخزن ساقه البرميلية، وأي عمل تولّته
\omterm{def:g1:plants-around-us:parts}{الساق}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:5}
لماذا تنمو كثير من \omterm{def:g1:plants-around-us:plant}{نباتات} الجبال العالية وسائد متراصّة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:6}
أعطِ تكيفين \omterm{def:g1:animals-around-us:animal}{للحيوانات} المصيدة وتكيفين للصائدة، من
\cref{prop:g4:adapted-to-their-world:answers}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:7}
أعد \cref{ex:g4:adapted-to-their-world:mystery}: أي ثلاث صفات
كشفت الخُلد، وعلام دلّت كل واحدة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:8}
جرّب: اختر أي \omterm{def:g1:animals-around-us:animal}{حيوان} تستطيع مراقبته --- بطة، أو قطة، أو عنكبوتًا ---
وطبّق عليه \cref{met:g4:adapted-to-their-world:read}: ثلاث صفات،
وثلاث مشكلات محلولة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:9}
لثعلب الصحراء \omterm{def:g1:the-five-senses:sense}{أذنان} هائلتان؛ ولثعلب القطب \omterm{def:g1:the-five-senses:sense}{أذنان} صغيرتان جدًّا.
استعمل سطر البرد في \cref{prop:g4:adapted-to-their-world:answers}
لتشرح الأمرين --- وما الذي يجب أن تفعله \omterm{def:g1:the-five-senses:sense}{الأذنان} الكبيرتان بالدفء؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:10}
\omterm{def:g1:animals-around-us:animal}{حيوان} غامض: جسم انسيابي، وقدمان خلفيتان مكفوفتان، \omterm{def:g1:animals-around-us:covering}{وفرو} كثيف يدفع
الماء، وشوارب، وهو يحبس نفَسه جيدًا. أعد بناء عالمه وفق
\cref{met:g4:adapted-to-their-world:read}. (وهو يبني السدود.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:adapted-to-their-world:11}
الحوت (\cref{ex:g3:sorting-living-things:surprises}) ثديي على شكل
سمكة: انسيابي، ذو زعانف، أملس. والسلمون المرقط سمكة على الشكل
نفسه. فعلام يدل هذا التشابه بين سابحين \emph{لا قرابة بينهما} في
شأن التكيفات --- من وضع الشكل، الأسرة أم الماء؟
\end{exercise}
