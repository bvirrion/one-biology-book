\section*{Capítulo \ref{ch:g2:what-animals-eat} --- O que os animais comem}

\begin{solution}{exo:g2:what-animals-eat:1}
\omterm{def:g2:what-animals-eat:kinds}{Herbívoro}; \omterm{def:g2:what-animals-eat:kinds}{carnívoro}; \omterm{def:g2:what-animals-eat:kinds}{onívoro}.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:2}
(a) \omterm{def:g2:what-animals-eat:kinds}{Herbívoro}. (b) \omterm{def:g2:what-animals-eat:kinds}{Carnívoro}. (c) \omterm{def:g2:what-animals-eat:kinds}{Onívoro}. (d) \omterm{def:g2:what-animals-eat:kinds}{Herbívoro}. (e)
\omterm{def:g2:what-animals-eat:kinds}{Onívoro}.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:3}
Trituradores chatos --- \omterm{def:g1:plants-around-us:plant}{plantas} como o capim são duras e precisam
ser trituradas por muito tempo antes de engolir; não há presa para
agarrar.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:4}
Curto e grosso: quebrar \omterm{def:g2:from-seed-to-plant:seed}{sementes}. Curvo e em gancho: rasgar carne.
Comprido e fino: pescar insetos na casca das árvores e nas frestas.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:5}
Pulgõezinhos das \omterm{def:g1:plants-around-us:plant}{plantas} (os pulgões) --- a joaninha come um monte
deles, e é por isso que quem cuida de horta a adora.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:6}
O capim alimenta pouco em cada bocado, então a vaca precisa pastar
durante horas para se satisfazer; a carne é rica, então uma boa
refeição pode durar bastante para a raposa.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:7}
Minhocas e insetos no verão; frutinhas no inverno, quando o chão
está duro demais para minhoca. Um cardápio largo deixa o \omterm{def:g2:what-animals-eat:kinds}{onívoro}
trocar de alimento quando um deles acaba --- uma grande ajuda nas
estações difíceis.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:8}
Resposta aberta. Em geral: pardais e tico-ticos (bicos curtos e
grossos) pegam \omterm{def:g2:from-seed-to-plant:seed}{sementes}; os pequenos insetívoros pegam insetos e
\omterm{def:g2:from-seed-to-plant:seed}{sementes}; os sabiás fuçam atrás de minhocas e comem frutinhas. A
combinação bico--cardápio da proposição se confirma bem.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:9}
O cardápio de um \omterm{def:g2:what-animals-eat:kinds}{carnívoro}: outros \omterm{def:g1:animals-around-us:animal}{animais}. Os passos 1 e 2 --- os
dentes (agarradores pontudos) e as ferramentas de caça (garras,
\omterm{def:g1:the-five-senses:sense}{olhos} para a frente); o passo 4 deu o nome.
\end{solution}

\begin{solution}{exo:g2:what-animals-eat:10}
A raposa come coelhos, e os coelhos comem capim. Sem capim, sem
coelhos; sem coelhos, raposas com fome. A raposa depende do capim
\emph{através} da presa dela --- as refeições da natureza formam
cadeias.
\end{solution}
