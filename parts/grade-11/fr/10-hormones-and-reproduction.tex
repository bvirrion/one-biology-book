\chapter{Le contrôle hormonal de la reproduction}\label{ch:g11:hormones-and-reproduction}

Une femme tient un relevé~: sa température chaque matin, avec une
petite montée vers le milieu de chaque mois~; un saignement tous les
vingt-huit jours environ. Un laboratoire qui doserait chaque jour
quatre \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} dans son sang tracerait quatre courbes qui montent et
descendent dans un ordre fixe, mois après mois, chaque pic causant le
suivant. Le relevé d'un homme montrerait les mêmes \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} à des
niveaux presque constants, jour après jour. Les organes diffèrent~; le
système de commande --- un centre du cerveau, une glande située en
dessous, la gonade, et des messages portés par le sang dans les deux
sens --- est le même. Ce chapitre lit ces relevés, et montre comment
les comprendre a permis d'empêcher une grossesse, ou d'en obtenir une.

\section{Les hormones et leurs boucles}

\begin{definition}[Hormone, rétroaction]\label{def:g11:hormones-and-reproduction:hormone}
Une \emph{hormone}\index{hormone} est une molécule sécrétée dans le
sang par une glande et qui agit, à distance, sur les \omterm{def:g10:cells-common-unit:cell}{cellules} porteuses
d'un \emph{récepteur} pour elle. Un système hormonal est réglé par une
\emph{rétroaction}\index{rétroaction}~: l'hormone produite au bout
d'une chaîne agit en retour sur les glandes qui la commandent. Une
\emph{rétroaction négative} --- le produit diminue la commande ---
maintient un niveau constant~; une \emph{rétroaction positive} --- le
produit augmente la commande --- produit une bouffée.
\end{definition}

\begin{proposition}[La chaîne commune]\label{prop:g11:hormones-and-reproduction:chain}
Dans les deux sexes, la même chaîne à trois étages commande les
gonades. Un centre du cerveau, l'\emph{hypothalamus}, sécrète par
impulsions une \omterm{def:g11:hormones-and-reproduction:hormone}{hormone} de libération, la GnRH, dans les petits
vaisseaux qui mènent à l'\emph{hypophyse} juste en dessous~;
l'hypophyse répond en sécrétant dans le sang général deux \omterm{def:g11:hormones-and-reproduction:hormone}{hormones}, la
\emph{LH} et la \emph{FSH}~; celles-ci agissent sur les gonades, qui
produisent à la fois des gamètes et des \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} sexuelles --- la
\omterm{prop:g11:becoming-male-female:hormones}{testostérone} dans le testicule, les œstrogènes et la progestérone dans
l'ovaire. Les \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} sexuelles agissent sur le corps, et agissent en
retour sur l'hypothalamus et l'hypophyse.
\end{proposition}
\admitted

\section{Chez l'homme~: un niveau constant}

\begin{proposition}[Le contrôle du testicule]\label{prop:g11:hormones-and-reproduction:male}
Le testicule a deux compartiments. Les \emph{tubes séminifères},
stimulés par la FSH et par la \omterm{prop:g11:becoming-male-female:hormones}{testostérone}, produisent des
spermatozoïdes en continu depuis la \omterm{prop:g11:becoming-male-female:puberty}{puberté}, une centaine de millions
par jour. Les \omterm{def:g10:cells-common-unit:cell}{cellules} situées entre les tubes, stimulées par la LH,
sécrètent la \emph{\omterm{prop:g11:becoming-male-female:hormones}{testostérone}}, qui entretient le tractus
reproducteur, les caractères secondaires et la production de
spermatozoïdes. La \omterm{prop:g11:becoming-male-female:hormones}{testostérone} inhibe l'hypothalamus et l'hypophyse~:
quand elle monte, la GnRH, la LH et la FSH baissent et le testicule
sécrète moins~; quand elle baisse, elles montent. Cette \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction}
négative maintient le niveau presque constant, à de faibles
fluctuations quotidiennes près, de la \omterm{prop:g11:becoming-male-female:puberty}{puberté} au grand âge.
\end{proposition}
\begin{proof}[Faits expérimentaux]
La castration d'un animal multiplie sa LH par plusieurs en quelques
jours~; une injection de \omterm{prop:g11:becoming-male-female:hormones}{testostérone} ramène la LH à son niveau. Une
lésion de l'hypothalamus abolit ensemble la LH, la FSH et la
\omterm{prop:g11:becoming-male-female:hormones}{testostérone}~; injecter la GnRH par impulsions rétablit les trois,
alors qu'une perfusion continue de GnRH n'y parvient pas --- c'est la
pulsatilité qui compte. Les hommes traités par \omterm{prop:g11:becoming-male-female:hormones}{testostérone} pour
d'autres raisons voient leur production de spermatozoïdes chuter~:
l'\omterm{def:g11:hormones-and-reproduction:hormone}{hormone} extérieure éteint leur propre hypophyse.
\end{proof}

\section{Chez la femme~: des cycles}

\begin{proposition}[Le cycle ovarien]\label{prop:g11:hormones-and-reproduction:ovary}
De la \omterm{prop:g11:becoming-male-female:puberty}{puberté} à la ménopause, l'ovaire déroule un cycle d'environ 28
jours.
\begin{itemize}
  \item \emph{Phase folliculaire} (jours 1 à 13)~: sous l'effet de la
        FSH, un follicule --- un ovocyte entouré de ses \omterm{def:g10:cells-common-unit:cell}{cellules}
        nourricières --- grandit et sécrète des quantités croissantes
        d'\emph{œstrogènes}.
  \item \emph{Ovulation} (jour 14)~: une bouffée de LH rompt le
        follicule mûr, qui libère son ovocyte dans la trompe.
  \item \emph{Phase lutéale} (jours 15 à 28)~: le follicule vidé
        devient le \emph{corps jaune}, qui sécrète de la
        \emph{progestérone} et des œstrogènes~; si aucune grossesse ne
        commence, il dégénère au bout d'une douzaine de jours, les deux
        \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} chutent, et un nouveau cycle démarre.
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[Le cycle utérin et les rétroactions]\label{prop:g11:hormones-and-reproduction:uterus}
La muqueuse de l'utérus suit les \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} de l'ovaire~: les œstrogènes
l'épaississent pendant la phase folliculaire~; la progestérone la rend
sécrétrice et prête à recevoir un embryon pendant la phase lutéale~; la
chute des deux à la fin du cycle la fait se rompre et saigner --- ce
sont les \emph{règles}, jours 1 à 5 du cycle suivant. Les \omterm{def:g11:hormones-and-reproduction:hormone}{rétroactions}
changent de signe au cours du cycle~: à niveaux faibles et modérés, les
œstrogènes et la progestérone inhibent l'hypothalamus et l'hypophyse
(\omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} négative)~; mais le fort taux d'œstrogènes du follicule
mûr, maintenu deux jours, les \emph{stimule} (\omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} positive), ce
qui produit la bouffée de LH qui déclenche l'ovulation. Après
l'ovulation, la progestérone du corps jaune rétablit la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction}
négative, et aucune seconde bouffée ne survient.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Des dosages sanguins quotidiens donnent les courbes de la figure
suivante~: la bouffée de LH suit d'un jour le pic d'œstrogènes et
précède l'ovulation d'un jour~; la progestérone n'apparaît qu'après
l'ovulation. Retirer les ovaires fait monter la LH et la FSH~; de
faibles doses d'œstrogènes les font baisser~; une dose forte et
prolongée administrée en milieu de cycle provoque une bouffée de LH ---
le signe de la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} s'inverse avec la dose. Une femme dont
l'hypophyse ne peut pas répondre à la montée des œstrogènes n'a pas de
bouffée et n'ovule pas, bien que ses follicules grandissent.
\end{proof}

\begin{method}[Lire un relevé de cycle]\label{met:g11:hormones-and-reproduction:read}
\begin{enumerate}
  \item Repérer le pic de LH~: l'ovulation a lieu le lendemain~; la
        phase folliculaire est tout ce qui précède, la phase lutéale
        tout ce qui suit.
  \item Vérifier l'ordre~: pic d'œstrogènes, puis bouffée de LH, puis
        ovulation, puis progestérone. Une courbe hors de l'ordre est le
        signe d'un trouble.
  \item Compter~: la phase lutéale dure environ 14 jours chez toutes
        les femmes~; c'est la phase folliculaire qui fait varier la
        durée du cycle. L'ovulation a donc lieu 14 jours avant les
        règles suivantes, et non 14 jours après les précédentes.
  \item Interpréter un médicament ou une lésion en se demandant quelle
        \omterm{def:g11:hormones-and-reproduction:hormone}{hormone} il ajoute ou retire, et ce que la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} fait
        alors des autres.
\end{enumerate}
\end{method}

\section{Choisir~: la contraception}

\begin{proposition}[La contraception hormonale]\label{prop:g11:hormones-and-reproduction:contraception}
La pilule combinée apporte, chaque jour, de faibles doses d'un
œstrogène et d'une molécule proche de la progestérone. À ces niveaux,
la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} reste négative tout le temps~: la FSH reste basse, aucun
follicule ne mûrit, aucun pic d'œstrogènes ne survient, aucune bouffée
de LH, aucune ovulation. La muqueuse de l'utérus, maintenue mince, et
la glaire du col, maintenue épaisse, ajoutent deux barrières.
Interrompre les comprimés une semaine laisse les \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} chuter et la
muqueuse saigner. D'autres méthodes agissent ailleurs~: un dispositif
au cuivre dans l'utérus empêche la fécondation et l'implantation~; la
pilule d'urgence, prise dans les jours qui suivent un rapport, retarde
l'ovulation~; les préservatifs empêchent les gamètes de se rencontrer
--- et sont la seule méthode qui arrête aussi les infections
sexuellement transmissibles.
\end{proposition}
\admitted

\begin{example}[Lire l'effet de la pilule]\label{ex:g11:hormones-and-reproduction:pill}
Sur le relevé d'une femme qui prend la pilule, les quatre courbes sont
plates~: œstrogènes et progestérone au niveau bas et régulier que les
comprimés apportent, LH et FSH sous leur minimum de cycle, aucun pic
nulle part. Oubliez les comprimés quelques jours au début de la
plaquette et la FSH s'échappe~: un follicule peut démarrer, et le reste
peut suivre. La pilule fonctionne parce que la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} fonctionne,
et un comprimé oublié est un trou dans la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction}.
\end{example}

\section{Obtenir~: l'assistance médicale à la procréation}

\begin{proposition}[Quand une grossesse ne vient pas]\label{prop:g11:hormones-and-reproduction:art}
Environ un couple sur sept consulte après un an d'essais. Les causes
sont partagées~: ovulation absente ou irrégulière, trompes bouchées,
spermatozoïdes trop peu nombreux ou immobiles, et dans un tiers des cas
aucune cause identifiée. Les traitements suivent la biologie~:
\begin{itemize}
  \item l'\emph{induction de l'ovulation}~: des injections de FSH, ou
        un médicament qui bloque la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} négative des
        œstrogènes, font mûrir des follicules~; une injection
        semblable à la LH déclenche ensuite l'ovulation à une heure
        choisie~;
  \item l'\emph{insémination}~: des spermatozoïdes préparés sont
        déposés dans l'utérus au moment de l'ovulation~;
  \item la \emph{fécondation in vitro}~: après stimulation, des
        ovocytes sont prélevés dans l'ovaire à l'aiguille, fécondés
        dans une boîte --- au besoin en injectant un spermatozoïde dans
        chaque ovocyte --- et un ou deux embryons sont déposés dans
        l'utérus trois à cinq jours plus tard. Environ une tentative
        sur trois ou quatre aboutit à une naissance~; les autres
        peuvent être recommencées avec des embryons congelés.
\end{itemize}
Chaque étape de la chaîne naturelle qui échoue est remplacée par
l'intervention correspondante.
\end{proposition}
\admitted

\begin{example}[Les chiffres d'une tentative]\label{ex:g11:hormones-and-reproduction:ivf}
La stimulation donne 10 ovocytes~; 8 sont mûrs~; 5 sont fécondés~; 3
atteignent le cinquième jour~; 1 est transféré et 2 congelés. Le
transfert réussit dans un cas sur trois environ~; avec les deux
embryons congelés, la chance d'une naissance issue d'un seul
prélèvement monte à environ 60~\%. La technique a donné plusieurs
millions d'enfants depuis 1978~; elle a aussi soulevé des questions ---
sur les embryons non utilisés, sur l'âge des parents, sur ce qui peut
être choisi --- auxquelles la biologie ne répond pas.
\end{example}

\begin{remark}[Deux systèmes qui se rencontrent]\label{rem:g11:hormones-and-reproduction:brain}
Les \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} de ce chapitre commandent les gonades et le corps~; elles
ne commandent pas, chez l'humain, le comportement sexuel comme elles le
font chez beaucoup d'animaux, où l'accouplement est lié au cycle
ovarien. L'activité sexuelle humaine est largement indépendante du
cycle et dépend des systèmes cérébraux du désir, du plaisir et de
l'attachement --- des circuits de la récompense partagés avec d'autres
motivations --- ainsi que de l'histoire, de la culture et des choix de
la personne. Reproduction et sexualité se recouvrent~; ce ne sont pas
la même chose, et les \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} ne gouvernent que la première.
\end{remark}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:hormones-and-reproduction:13}
Le relevé d'une femme montre un pic d'œstrogènes mais aucune bouffée de
LH et aucune progestérone. Localiser le défaut dans la chaîne et
proposer un traitement.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:hormones-and-reproduction:14}
Pendant la grossesse, les tissus de l'embryon sécrètent une \omterm{def:g11:hormones-and-reproduction:hormone}{hormone}
semblable à la LH qui maintient le corps jaune en vie. Expliquer
pourquoi cela empêche les règles et toute nouvelle ovulation, et
pourquoi les tests de grossesse détectent cette \omterm{def:g11:hormones-and-reproduction:hormone}{hormone}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:hormones-and-reproduction:15}
Un cycle de fécondation in vitro prélève 12 ovocytes, dont 60~\% sont
fécondés et dont 40~\% de ceux-là atteignent le cinquième jour~; chaque
embryon transféré donne une naissance dans 35~\% des cas. Calculer le
nombre attendu d'embryons, et la probabilité d'au moins une naissance
s'ils sont transférés un à un jusqu'au succès ou à l'épuisement.
\end{exercise}

\section{Problème~: le relevé}

\begin{problem}[{Problème du week-end --- le cycle d'une femme relevé
hormone par hormone, lu pour ses rétroactions, perturbé par une pilule,
et assisté quand il échoue}]
\label{pb:g11:hormones-and-reproduction:1}
Des prélèvements sanguins sont faits chaque jour au cours d'un cycle.
Valeurs arrondies, en pourcentage du maximum de chaque \omterm{def:g11:hormones-and-reproduction:hormone}{hormone}~:

\begin{center}
\begin{tabular}{lccccccccc}
jour & 2 & 6 & 10 & 12 & 13 & 14 & 18 & 22 & 27 \\ \hline
œstrogènes & 10 & 20 & 55 & 100 & 80 & 45 & 45 & 60 & 15 \\
LH & 15 & 15 & 20 & 35 & 100 & 60 & 15 & 15 & 12 \\
FSH & 45 & 35 & 30 & 35 & 70 & 45 & 22 & 25 & 25 \\
progestérone & 3 & 3 & 3 & 3 & 4 & 5 & 55 & 100 & 15 \\
\end{tabular}
\end{center}

\medskip
\noindent\textbf{Partie I --- Lire le relevé.}
\begin{enumerate}
  \item Quel jour l'ovulation a-t-elle lieu~? Justifier à partir de
        deux \omterm{def:g11:hormones-and-reproduction:hormone}{hormones}.
  \item Donner les phases folliculaire et lutéale, avec leurs durées.
  \item Quelle structure de l'ovaire sécrète les œstrogènes du jour
        10~? Ceux du jour 22~? Laquelle sécrète la progestérone du jour
        22~?
  \item Décrire l'état de la muqueuse utérine aux jours 6, 22 et 28.
  \item Pourquoi la FSH est-elle la plus élevée au début du cycle et la
        plus basse en phase lutéale~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Les \omterm{def:g11:hormones-and-reproduction:hormone}{rétroactions}.}
\begin{enumerate}[resume]
  \item Entre les jours 2 et 10, les œstrogènes montent alors que la
        FSH baisse. Quelle \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} est à l'œuvre~? Expliquer.
  \item Entre les jours 12 et 13, les œstrogènes sont à leur pic et la
        LH est multipliée par sept. Quelle \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction}~? Quelle
        condition sur le niveau d'œstrogènes la rend possible~?
  \item Entre les jours 18 et 27, les œstrogènes sont modérés, la
        progestérone élevée et la LH basse. Quelle \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction}, et
        pourquoi n'y a-t-il pas de seconde bouffée~?
  \item Le corps jaune dégénère au jour 26. Prévoir les \omterm{def:g11:hormones-and-reproduction:hormone}{hormones} du
        jour 28 et ce qui suit.
  \item Expliquer pourquoi le cycle se relance de lui-même~: quelle
        montée du jour 1 lance le follicule suivant~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La pilule.}
La même femme prend la pilule combinée pendant un mois~; celle-ci
apporte chaque jour des œstrogènes à 25~\% et une \omterm{def:g11:hormones-and-reproduction:hormone}{hormone} proche de la
progestérone à 40~\% de leurs maxima.
\begin{enumerate}[resume]
  \item Prévoir les niveaux de FSH et de LH tout au long du mois, et le
        sort des follicules.
  \item Y aura-t-il un pic d'œstrogènes~? Une bouffée de LH~? Une
        ovulation~?
  \item Pendant les sept jours sans comprimé, qu'arrive-t-il à la
        muqueuse utérine, et pourquoi ne s'agit-il pas de véritables
        règles~?
  \item Elle oublie les comprimés des jours 5 à 8. Expliquer le risque,
        étape par étape.
  \item Expliquer pourquoi une pilule à la progestérone seule, à un
        niveau qui maintient la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} négative, peut aussi
        empêcher l'ovulation.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Quand la chaîne est rompue.}
\begin{enumerate}[resume]
  \item Le relevé d'une autre femme montre une FSH et une LH toujours
        sous 10~\% et des œstrogènes sous 10~\%. Où est le défaut~?
        Quel traitement rétablirait l'ovulation, et par quelle voie~?
  \item Le relevé d'une troisième femme est normal mais ses trompes
        sont bouchées. Quelles étapes de la
        \cref{prop:g11:hormones-and-reproduction:art} remplacent
        l'étape bloquée~?
  \item En fécondation in vitro, une injection unique d'une \omterm{def:g11:hormones-and-reproduction:hormone}{hormone}
        semblable à la LH est faite 36 heures avant le prélèvement des
        ovocytes. Expliquer ce qu'elle imite et pourquoi le moment doit
        être exact.
  \item Après stimulation, 10 follicules mûrissent au lieu d'un.
        Expliquer, avec la FSH, pourquoi le cycle naturel n'en mûrit
        d'ordinaire qu'un seul.
  \item Énoncer le résultat~: le jour de l'ovulation sur le relevé, les
        deux signes de la \omterm{def:g11:hormones-and-reproduction:hormone}{rétroaction} qui l'ont produite, et le
        mécanisme par lequel la pilule l'a supprimée.
\end{enumerate}
\end{problem}
