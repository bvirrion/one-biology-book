\chapter{यौवनारंभ}\label{ch:g8:puberty}

लगभग नौ और सोलह के बीच कहीं, हर इंसानी शरीर ख़ुद को दोबारा बनाता है:
यानी \cref{rem:g3:stages-of-life:bursts} वाली वृद्धि की दूसरी उछाल, और
सिर्फ़ लंबाई से कहीं ज़्यादा।
\Cref{rem:g5:human-reproduction:adults} ने बरसों पहले इस अध्याय का वादा
किया था। तो लीजिए --- तथ्य, समय-सारणी की वह मशहूर बेइंसाफ़ी, और वह
मशीनरी, जो हमें ख़ुद \omterm{def:g8:puberty:puberty}{यौवनारंभ} से भी बड़ी एक खोज थमा देगी: शरीर के पास
संवाद का एक दूसरा तंत्र भी है।

\section{क्या-क्या बदलता है}

\begin{definition}[यौवनारंभ]\label{def:g8:puberty:puberty}
\emph{यौवनारंभ}\index{यौवनारंभ} वह अवस्था है जिसमें बच्चे का शरीर
वयस्क का शरीर बन जाता है --- और सबसे बढ़कर,
\cref{ch:g8:sexual-reproduction} के अर्थ में वयस्क का: जनन-अंग परिपक्व
होते हैं और \emph{\omterm{def:g8:sexual-reproduction:gametes}{युग्मक}} बनाने लगते हैं, तथा शरीर अपना वयस्क रूप ले
लेता है। यह \omterm{def:g3:stages-of-life:stages}{किशोर} अवस्था (\cref{def:g3:stages-of-life:stages}) का
जैविक केंद्र है।
\end{definition}

\begin{proposition}[दिखने वाले बदलाव]\label{prop:g8:puberty:changes}
वृद्धि की उछाल के साथ-साथ, कई सालों में:
\begin{itemize}
  \item लड़कियों में, आम तौर पर लगभग नौ से तेरह की उम्र में: स्तन
        विकसित होते हैं, कूल्हे चौड़े होते हैं, शरीर पर बाल आते हैं
        --- और \emph{पहला मासिक धर्म} बताता है कि
        \cref{ch:g8:reproductive-systems} वाले चक्र चालू हो गए हैं;
  \item लड़कों में, आम तौर पर साल-दो साल बाद: आवाज़ भारी होती है
        (``फटती'' है), \omterm{def:g1:my-body:joint}{कंधे} चौड़े होते हैं, शरीर तथा चेहरे पर बाल आते
        हैं, और वृषण \omterm{def:g4:animal-reproduction:sexual}{शुक्राणु} बनाने लगते हैं;
  \item और दोनों में: \omterm{def:g1:the-five-senses:sense}{त्वचा} ज़्यादा तैलीय हो जाती है (यानी मुँहासों की
        उम्र), \omterm{rem:g7:removing-wastes:sweat}{पसीना} बदलता है, और मिज़ाज इस दोबारा बनावट पर सवार रहते
        हैं --- क्योंकि \omterm{def:g5:brain-in-command:system}{मस्तिष्क} भी अपने वयस्क रूप की ओर नए सिरे से
        जुड़ रहा होता है।
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[समय-सारणी निजी होती है]\label{rem:g8:puberty:timetable}
\Cref{rem:g1:my-body:different} ने यह लंबाई के बारे में कहा था;
\omterm{def:g8:puberty:puberty}{यौवनारंभ} इसे और ज़ोर से दोहराता है। एक ही उम्र के सहपाठी इस रास्ते पर
सालों की दूरी पर खड़े हो सकते हैं --- जल्दी और देर, दोनों सामान्य हैं,
बदलावों का क्रम भी बदलता रहता है, और आख़िर में क़रीब-क़रीब सब वहीं
पहुँचते हैं। ऊपर दी गई सीमाएँ चौड़ी हैं, और उनके बारे में सबसे चौड़ी बात
यह है कि वे किसी एक व्यक्ति के बारे में कितना कम कहती हैं।
\end{remark}

\section{मशीनरी: संदेशवाहकों का एक दूसरा तंत्र}

\begin{definition}[हॉर्मोन, पहली मुलाक़ात]\label{def:g8:puberty:hormones}
\omterm{def:g8:puberty:puberty}{यौवनारंभ} का फ़ैसला उन अंगों में नहीं होता जिन्हें वह बदलता है। उसका
\emph{हुक्म} रासायनिक संदेशवाहक देते हैं, जिन्हें
\emph{हॉर्मोन}\index{हॉर्मोन} कहते हैं: वे पदार्थ, जिन्हें ख़ास अंग
\emph{\omterm{prop:g4:journey-of-food:absorb}{रक्त}} में छोड़ते हैं, जो हर जगह पहुँचाए जाते हैं, और जिनका कहना
वही अंग मानते हैं जो उन्हें सुनने के लिए बने हैं। \omterm{def:g8:puberty:puberty}{यौवनारंभ} पर \omterm{def:g5:brain-in-command:system}{मस्तिष्क}
के तल का एक कमान-केंद्र जागता है, उसके हॉर्मोन जनन-अंगों को हुक्म देते
हैं, और \emph{उनके} हॉर्मोन ---
\cref{prop:g4:heart-and-blood:carries} वाली पहुँचाने की सेवा से पूरे
शरीर में ढोए हुए --- पूरी दोबारा बनावट का हुक्म देते हैं: वृद्धि, बाल,
आवाज़, चक्र।
\end{definition}

\begin{example}[एक इशारा, कई उसारियाँ]\label{ex:g8:puberty:sites}
\omterm{prop:g4:journey-of-food:absorb}{रक्त} किसी \omterm{def:g8:puberty:hormones}{हॉर्मोन} को एक ही साथ हर जगह ले जाता है --- और इसीलिए
\omterm{def:g8:puberty:puberty}{यौवनारंभ} एक \emph{तालमेल वाली} दोबारा बनावट है: आवाज़, \omterm{def:g1:my-body:joint}{कंधे} और बाल, सब
एक ही संदेशवाहक के पीछे, एक ही समय-सारणी पर चलते हैं, हालाँकि उनकी जगहें
आधे शरीर की दूरी पर हैं। किसी स्वर-यंत्र को दाढ़ी से कोई तंत्रिका-तार
नहीं जोड़ता; जोड़ता है तो रक्त-धारा का वह प्रसारण
(\cref{ch:g8:hormonal-communication} इस तंत्र को अपना पूरा अध्याय दे
देता है)।
\end{example}

\begin{remark}[मिज़ाज का मौसम क्यों?]\label{rem:g8:puberty:moods}
वही संदेशवाहक \omterm{def:g5:brain-in-command:system}{मस्तिष्क} तक भी पहुँचते हैं, और वह ख़ुद बीच मरम्मत में है:
मिज़ाज का झूलना, नई तीव्रताएँ, \omterm{prop:g1:healthy-habits:needs}{नींद} का देर की ओर खिसकना --- यह मौसम है,
कोई ख़ता नहीं। मशीनरी जानना उसके दोनों तरफ़ काम आता है: ये तूफ़ान रसायन
और उसारी का जोड़ हैं, ये सामान्य हैं, और ये गुज़र जाते हैं।
\end{remark}

\section{दोबारा बनावट के बीच जीना}

\begin{proposition}[उसारी के दौरान देखभाल]\label{prop:g8:puberty:care}
बीच बनावट में खड़े शरीर की ज़रूरतें किसी उसारी वाले की ज़रूरतें हैं, और
ये सब इस किताब में पहले पूरी की जा चुकी हैं:
\begin{itemize}
  \item सामान और ईंधन: इस उछाल का दाम पदार्थ में चुकता है --- यानी
        संतुलित खाना (\cref{met:g3:balanced-diet:check}), और उसमें
        शरीर बनाने वाला पदार्थ भी
        (\cref{exo:g7:digestion-and-nutrients:11});
  \item \omterm{prop:g1:healthy-habits:needs}{नींद}: \omterm{def:g5:brain-in-command:system}{मस्तिष्क} की मरम्मत और शरीर की वृद्धि, दोनों रात को चलती
        हैं (\cref{prop:g5:brain-in-command:protect}) --- और देर की
        ओर खिसकी घड़ी को भी अपने घंटे चाहिए ही;
  \item हरकत: \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ} और \omterm{def:g4:heart-and-blood:heart}{दिल} अपने आख़िरी, सबसे अच्छे बनने वाले सालों
        में हैं (\cref{met:g3:bones-and-muscles:care});
  \item और कोई तोड़-फोड़ नहीं:
        \cref{prop:g5:healthy-choices:substances} वाले पदार्थ उन्हीं
        शरीरों और \omterm{def:g5:brain-in-command:system}{मस्तिष्कों} पर सबसे कड़ी चोट करते हैं जो अब भी बन रहे
        हैं।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[सवालों के पते होते हैं]\label{ex:g8:puberty:questions}
\omterm{def:g8:puberty:puberty}{यौवनारंभ} सवाल खड़े करता है --- समय के बारे में, शरीरों के बारे में,
भावनाओं के बारे में --- और उनके ठीक-ठीक पते भी हैं: स्कूल की नर्सें और
डॉक्टर इनके जवाब रोज़ देते हैं, गोपनीयता के साथ और बिना चौंके; भरोसे के
बड़े ख़ुद इससे गुज़र चुके हैं; और इस किताब के अध्याय दोबारा पढ़े जाने के
लिए ही बने हैं। और भरोसे के लायक़ बिलकुल नहीं: मैदान की गप और इंटरनेट के
सबसे बुरे कोने, जो ठीक इसी उम्र की अनिश्चितताओं का सौदा करते हैं।
\end{example}

\section{समस्या: वृद्धि-चार्टों की बैठक}

\begin{problem}[{सप्ताहांत समस्या --- चार (कल्पित) वृद्धि-चार्ट, पूरे
अध्याय के साथ पढ़े हुए}]\label{pb:g8:puberty:1}
स्कूल की डॉक्टर उम्र के सामने लंबाई वाले चार चार्ट देखती हैं, और हर एक
पर नापने की तारीख़ें भी हैं: क, जिसका वक्र ग्यारह से तेरह के बीच तेज़ी
से उछला और अब पंद्रह पर चपटा पड़ रहा है; ख, जो तेरह पर सपाट-सी चढ़ाई पर
है और जिसकी उछाल अभी शुरू ही नहीं हुई; ग, जो चौदह पर उछाल के बीच में है;
और घ, जिसका खड़ा हिस्सा नौ से बारह तक चला और अब पठार पर है।

\medskip\noindent\textbf{भाग I --- वक्र पढ़ना।}
\begin{enumerate}
  \item हर चार्ट को \cref{rem:g3:stages-of-life:bursts} के नक़्शे पर
        रखो: कौन-सी उछाल दिख रही है, और हर विद्यार्थी उस पर कहाँ
        खड़ा है?
  \item ख को ``पीछे छूट जाने'' की फ़िक्र है।
        \cref{rem:g8:puberty:timetable} से डॉक्टर का जवाब लिखो, और
        साथ में वह भविष्यवाणी भी जिसे उसके चार्ट की सपाट चढ़ाई
        सहारा देती है।
  \item घ पूछता है कि क्या इस पठार का मतलब ``बारह पर ही बढ़ना ख़त्म''
        है। इस पढ़त को सुधारो: यह पठार क्या ख़त्म करता है, और उसके
        बाद क्या जारी रहता है?
  \item डॉक्टर हर एक की जगह आँकने के लिए लंबाई नहीं, बल्कि \emph{ढाल}
        क्यों पढ़ती हैं?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- वक्रों के पीछे।}
\begin{enumerate}[resume]
  \item हर उछाल को शुरू करने वाली मशीनरी का नाम बताओ, और वह रास्ता भी
        जिससे उसके हुक्म चले।
  \item डॉक्टर इन मुलाक़ातों में \omterm{prop:g1:healthy-habits:needs}{नींद} और खाने का भी लेखा रखती हैं।
        हर एक को \cref{prop:g8:puberty:care} की उसकी पंक्ति से
        जोड़ो।
  \item ग मिज़ाज के तूफ़ान और देर की ओर खिसकते सोने के वक़्त की बात
        बताता है। \cref{rem:g8:puberty:moods} वाली दो हिस्सों की
        व्याख्या दो --- और ग की स्कूल वाले हफ़्ते की \omterm{prop:g1:healthy-habits:needs}{नींद} के लिए
        उसका व्यावहारिक नतीजा भी।
  \item क डॉक्टर से अकेले में एक ऐसे बदलाव के बारे में पूछता है
        जिसे चार्ट दिखा ही नहीं सकता। सवाल गढ़े बिना बताओ कि
        डॉक्टर का कमरा ही सही पता क्यों है
        (\cref{ex:g8:puberty:questions})।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- बैठक की टिप्पणी।}
\begin{enumerate}[resume]
  \item हर चार्ट के लिए एक पंक्ति की टिप्पणी लिखो: अवस्था,
        सामान्यता, और देखभाल की कोई बात, अगर हो।
  \item चारों चार्ट सालों के फ़ासले पर हैं, फिर भी डॉक्टर चारों को
        ``सामान्य'' में दर्ज करती हैं। एक वाक्य में इसका औचित्य दो।
  \item एक अभिभावक ``ख को कुछ देकर तेज़ करने'' की माँग करते हैं।
        समय-सारणी वाली टिप्पणी और मशीनरी से जवाब दो: यहाँ ठीक करने
        को है ही क्या?
  \item कक्षा के सूचना-पट के लिए अध्याय का एक वाक्य वाला सार लिखकर
        बात ख़त्म करो: \omterm{def:g8:puberty:puberty}{यौवनारंभ} है क्या, उसे चलाता कौन है, और उसे
        चाहिए क्या।
\end{enumerate}
\end{problem}
