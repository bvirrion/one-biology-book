\chapter{La contraception}\label{ch:g8:contraception}

Les derniers chapitres ont cartographié un mécanisme d'une puissance
remarquable~: deux \omterm{def:g5:first-look-at-cells:cell}{cellules} se rencontrent, et quarante semaines plus
tard il y a une personne. Les humains sont la seule \omterm{def:g5:biodiversity:species}{espèce} qui
comprenne son propre mécanisme --- et cette compréhension amène un
choix que la grenouille n'affronte jamais~: le mettre en route ou
non, et quand. La \emph{\omterm{prop:g8:contraception:principle}{contraception}} est la science appliquée de ce
choix, et elle fonctionne, dans chacune de ses méthodes, en
interrompant une étape que tu sais maintenant nommer.

\section{Le principe~: nommer le maillon, bloquer le maillon}

\begin{proposition}[La contraception se superpose au mécanisme]\label{prop:g8:contraception:principle}
Une \omterm{def:g5:human-reproduction:pregnancy}{grossesse} exige une chaîne ininterrompue
(\cref{prop:g8:sexual-reproduction:core},
\cref{prop:g8:from-egg-to-birth:firstweek})~: \omterm{def:g8:sexual-reproduction:gametes}{gamètes} produits ---
sperme déposé --- la nage jusqu'à la \omterm{def:g8:reproductive-systems:female}{trompe} --- une \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}
libérée et rencontrée --- \omterm{def:g5:flower-to-fruit:steps}{fécondation} --- nidation. La
\emph{contraception}\index{contraception} est toute interruption
volontaire de cette chaîne. Chaque méthode de l'étagère du
pharmacien est une attaque contre un maillon nommé, et les classer
par maillon, c'est toute la théorie~:
\begin{itemize}
  \item les méthodes \emph{barrières} empêchent les \omterm{def:g8:sexual-reproduction:gametes}{gamètes} de se
        rencontrer~;
  \item les méthodes \emph{hormonales} empêchent la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}
        d'être libérée~;
  \item d'autres méthodes font échouer la rencontre ou la nidation.
\end{itemize}
\end{proposition}
\admitted

\section{Les principales méthodes, classées par maillon}

\begin{definition}[Méthodes barrières~: le préservatif]\label{def:g8:contraception:condom}
Le \emph{préservatif}\index{préservatif} --- une fine gaine portée
sur le \omterm{def:g8:reproductive-systems:male}{pénis} (une version féminine tapisse le vagin) --- bloque la
chaîne à l'étape du dépôt~: le sperme n'entre jamais dans le corps
du partenaire. Largement disponible, nécessaire seulement au moment
voulu --- et unique sur l'étagère par un second service~: c'est la
seule méthode qui bloque aussi le passage des \emph{infections}
transmises par les rapports sexuels, le monde dont traitera
\cref{ch:g9:microbes-and-infection}. Cette double fonction explique
que le préservatif reste dans le tableau même quand une autre
méthode est utilisée.
\end{definition}

\begin{definition}[Méthodes hormonales~: la pilule]\label{def:g8:contraception:pill}
La \emph{pilule contraceptive}\index{pilule contraceptive} se sert
de la machinerie même de \cref{def:g8:puberty:hormones}~: prise
chaque jour, ses \omterm{def:g8:puberty:hormones}{hormones} maintiennent le cycle des \omterm{def:g8:reproductive-systems:female}{ovaires} dans son
état suspendu (l'événement 4 de
\cref{prop:g8:reproductive-systems:cycle}, sans aucune \omterm{def:g5:human-reproduction:pregnancy}{grossesse})
--- \emph{pas d'\omterm{prop:g8:reproductive-systems:cycle}{ovulation}, pas de \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}, rien à féconder}.
Prescrite par un médecin, qui ajuste la méthode à la personne~; très
fiable quand elle est prise régulièrement --- et n'offrant
absolument rien contre les infections, d'où le partenariat permanent
du \omterm{def:g8:contraception:condom}{préservatif}.
\end{definition}

\begin{example}[L'étagère, classée par maillon]\label{ex:g8:contraception:shelf}
Les autres méthodes courantes, rangées d'après le maillon qu'elles
attaquent~: implants et patchs hormonaux --- le principe de la
pilule avec une meilleure mémoire (des mois ou des années, sans
prise quotidienne)~; les petits dispositifs qu'un médecin place dans
l'\omterm{prop:g5:human-reproduction:plan}{utérus} --- la rencontre et la nidation rendues impossibles pendant
des années~; et la \emph{pilule du lendemain}, une dose d'\omterm{def:g8:puberty:hormones}{hormones}
prise après coup en quelques jours, qui agit surtout en retardant une
\omterm{prop:g8:reproductive-systems:cycle}{ovulation} qui n'a pas encore eu lieu --- un frein de dernier
recours, disponible en pharmacie, et qui ne remplace aucune méthode
employée \emph{avant}. Ce qu'aucune d'elles ne fait~: quoi que ce
soit contre les infections.
\end{example}

\begin{remark}[Ce qui ne marche pas]\label{rem:g8:contraception:notwork}
Les méthodes de la cour de récréation, mesurées à la chaîne~:
«~compter les jours sans risque~» échoue sur la variabilité de
\cref{rem:g8:reproductive-systems:living} --- la date de l'\omterm{prop:g8:reproductive-systems:cycle}{ovulation}
n'est pas connaissable d'avance, et les \omterm{def:g8:reproductive-systems:male}{cellules mâles} survivent
plusieurs jours en l'attendant~; «~se retirer à temps~» échoue sur
le fait que des \omterm{def:g8:reproductive-systems:male}{cellules mâles} voyagent avant le final. Les deux
échouent exactement là où la théorie du maillon nommé le prédit~:
elles ne bloquent aucun maillon de façon fiable. Une légende n'est
pas une méthode.
\end{remark}

\section{Le choix, la responsabilité, et la carte des aides}

\begin{proposition}[Le choix est réel et partagé]\label{prop:g8:contraception:choice}
Des faits que le mécanisme impose~:
\begin{itemize}
  \item une \omterm{def:g5:human-reproduction:pregnancy}{grossesse} ne demande qu'un seul rapport, pas la durée
        d'une relation~: la chaîne ne compte pas les occasions~;
  \item c'est une chaîne \emph{partagée}~: les \omterm{def:g5:first-look-at-cells:cell}{cellules} des deux
        partenaires, la responsabilité des deux --- le \omterm{def:g8:contraception:condom}{préservatif}
        et la pilule ne sont pas l'affaire d'un sexe chacun, mais
        une conversation~;
  \item et aucune méthode sauf l'abstinence n'est parfaite, même si
        les bonnes en approchent quand elles sont employées
        exactement comme prévu --- «~employée correctement~» est
        l'endroit où vivent la plupart des échecs réels.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[La carte des aides]\label{ex:g8:contraception:help}
Les adresses de \cref{ex:g8:puberty:questions}, élargies~: les
infirmières scolaires et les médecins de famille répondent aux
questions de \omterm{prop:g8:contraception:principle}{contraception} en confidence --- y compris à des
mineurs~; des centres de planification existent dans chaque région
exactement pour cela, gratuits et sans jugement~; les pharmaciens
gèrent l'horloge de la pilule du lendemain. Le système est bâti pour
que personne n'ait à naviguer dans la chaîne à coups de légendes ---
la carte des aides fait partie de la méthode.
\end{example}

\begin{method}[Évaluer n'importe quelle méthode dont on te parle]\label{met:g8:contraception:evaluate}
Devant toute méthode annoncée, demande, dans l'ordre~:
\begin{enumerate}
  \item \emph{quel maillon} de la chaîne bloque-t-elle --- peux-tu
        le nommer sur la carte~?
  \item \emph{avec quelle fiabilité}, employée comme les gens
        l'emploient vraiment~?
  \item protège-t-elle aussi contre les \emph{infections}, ou le
        \omterm{def:g8:contraception:condom}{préservatif} doit-il l'accompagner~?
  \item que demande-t-elle --- une mémoire quotidienne, une
        ordonnance, une pose --- et cela convient-il à
        l'utilisateur~?
\end{enumerate}
Pas de maillon nommable, pas de méthode
(\cref{rem:g8:contraception:notwork} échoue dès la question
1).
\end{method}

\end{exercise}

\section{Problème~: la consultation au centre de planification}

\begin{problem}[{Problème du week-end --- un après-midi de questions
au centre de planification}]\label{pb:g8:contraception:1}
L'après-midi (imaginaire) d'une conseillère en planification
familiale~: quatre visiteurs, quatre situations, une carte de la
chaîne au mur.

\medskip\noindent\textbf{Partie I --- La carte au
mur.}
\begin{enumerate}
  \item Dessine la carte du mur~: la chaîne des \omterm{def:g8:sexual-reproduction:gametes}{gamètes} jusqu'à la
        nidation, avec les trois points d'attaque de
        \cref{prop:g8:contraception:principle} marqués.
  \item Le visiteur 1 demande~: «~comment une pilule par jour
        peut-elle arrêter quoi que ce soit~?~» Réponds avec la
        machinerie~: quelle langue la pilule parle-t-elle, et à qui~?
  \item Le visiteur 1 enchaîne~: «~et si j'en oublie~?~» Situe le
        danger sur la carte --- qu'est-ce qui reprend quand le
        signal hormonal se relâche~?
  \item Pourquoi la recommandation du \omterm{def:g8:contraception:condom}{préservatif} par la conseillère
        survit-elle à tout autre choix de méthode~? (La double
        fonction.)
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les situations.}
\begin{enumerate}[resume]
  \item La visiteuse 2, dix-sept ans et inquiète, croyait qu'une
        application de jours sans risque rendait toute autre méthode
        inutile. Corrige gentiment cette croyance~: quels deux faits
        biologiques mettent en échec le calendrier~?
  \item La visiteuse 3 veut quelque chose «~sans mémoire
        quotidienne~». Propose les deux candidats de l'étagère et
        leurs maillons.
  \item Le visiteur 4 arrive moins d'un jour après la rupture d'un
        \omterm{def:g8:contraception:condom}{préservatif}. Nomme l'option, son horloge, son mécanisme
        principal --- et la méthode permanente dont la conseillère
        parlera pour la suite.
  \item Tous les quatre repartent avec la même phrase finale sur les
        infections. Rédige-la.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le débriefing.}
\begin{enumerate}[resume]
  \item La stagiaire demande pourquoi le centre enseigne la chaîne
        plutôt que de simplement distribuer des méthodes. Réponds
        avec la première question de
        \cref{met:g8:contraception:evaluate}.
  \item Classe les méthodes de l'après-midi selon ce que chacune
        demande à son utilisateur (question 4 de la méthode).
  \item La stagiaire se demande pourquoi les visites de mineurs sont
        confidentielles par principe. Réponds avec la logique de
        \cref{ex:g8:contraception:help}.
  \item Termine l'après-midi en une phrase~: ce que le centre
        distribue réellement, au-delà des boîtes.
\end{enumerate}
\end{problem}
