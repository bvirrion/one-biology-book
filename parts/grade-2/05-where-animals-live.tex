\chapter{Where Animals Live}\label{ch:g2:where-animals-live}

You will not meet a trout in the schoolyard, nor a squirrel at the
bottom of the pond. Each \omterm{def:g1:animals-around-us:animal}{animal} lives in the kind of place that
suits it --- where it finds its food, its shelter and its
neighbors. Biologists have a name for an \omterm{def:g1:animals-around-us:animal}{animal}'s home ground.

\section{Habitats}

\begin{definition}[Habitat]\label{def:g2:where-animals-live:habitat}
The \emph{habitat}\index{habitat} of an \omterm{def:g1:animals-around-us:animal}{animal} is the kind of
place where it naturally lives and finds everything it needs:
food, water, shelter, and a safe spot for its young. The pond,
the forest, the meadow, the hedge and the seashore are habitats.
\end{definition}

\begin{example}[Four habitats, four neighborhoods]\label{ex:g2:where-animals-live:four}
In the \emph{pond}: frogs, dragonflies, sticklebacks, pond
snails. In the \emph{forest}: squirrels, deer, woodpeckers, wood
ants. In the \emph{meadow}: rabbits, grasshoppers, butterflies,
field mice. Under the stones of an old \emph{wall}: woodlice,
spiders, sometimes a lizard sunning itself on top.
\end{example}

\begin{remark}[One habitat, many homes]\label{rem:g2:where-animals-live:layers}
A \omterm{def:g2:where-animals-live:habitat}{habitat} has many corners. In the one forest, the woodpecker
lives high on the trunks, the squirrel in the \omterm{def:g1:plants-around-us:tree}{branches}, the deer
among the bushes, and the wood ants on the ground in their heaped
nest. Same forest, different floors --- so many \omterm{def:g1:animals-around-us:animal}{animals} can share
one \omterm{def:g2:where-animals-live:habitat}{habitat} without crowding.
\end{remark}

\section{Fitted to the habitat}

\begin{proposition}[Animals fit their habitat]\label{prop:g2:where-animals-live:fit}
An \omterm{def:g1:animals-around-us:animal}{animal}'s body and habits fit the place it lives:
\begin{itemize}
  \item the frog's webbed feet push water --- made for the pond;
  \item the squirrel's gripping claws and big balancing tail are
        made for trees;
  \item the rabbit's digging paws and long warning \omterm{def:g1:the-five-senses:sense}{ears} are made
        for open meadow with burrows below;
  \item the duck's \omterm{def:g1:animals-around-us:covering}{feathers} shed water, and its webbed feet
        paddle.
\end{itemize}
Moved to the wrong \omterm{def:g2:where-animals-live:habitat}{habitat}, an \omterm{def:g1:animals-around-us:animal}{animal} loses these advantages.
\end{proposition}
\admitted

\begin{example}[The mole's equipment]\label{ex:g2:where-animals-live:mole}
The mole lives underground, in tunnels it digs beneath the
meadow. Its hands are wide digging shovels, its \omterm{def:g1:animals-around-us:covering}{fur} lies flat in
both directions so soil never ruffles it, and its weak \omterm{def:g1:the-five-senses:sense}{eyes}
hardly matter in the dark. Above ground it is clumsy; in its own
\omterm{def:g2:where-animals-live:habitat}{habitat} it is a champion.
\end{example}

\begin{method}[Matching an animal to its habitat]\label{met:g2:where-animals-live:matching}
Given an unfamiliar \omterm{def:g1:animals-around-us:animal}{animal}:
\begin{enumerate}
  \item look at its equipment: webbed feet? claws? wings? digging
        paws?
  \item ask what it eats, and where that food is found;
  \item ask where it can shelter and raise young;
  \item the \omterm{def:g2:where-animals-live:habitat}{habitat} that answers all three is its likely home.
\end{enumerate}
\end{method}

\section{Habitats can be lost}

\begin{proposition}[No habitat, no inhabitants]\label{prop:g2:where-animals-live:loss}
An \omterm{def:g1:animals-around-us:animal}{animal} cannot live without its \omterm{def:g2:where-animals-live:habitat}{habitat}. If a pond is filled
in, its frogs and dragonflies disappear with it; if a hedge is
torn out, the birds that nested there lose their shelter.
Protecting \omterm{def:g1:animals-around-us:animal}{animals} starts with protecting the places they live.
\end{proposition}
\admitted

\begin{example}[The school pond]\label{ex:g2:where-animals-live:schoolpond}
A school digs a small pond in a corner of the yard. Within two
years, dragonflies patrol it, pond skaters walk on it, and frogs
have moved in --- nobody brought them. Make the \omterm{def:g2:where-animals-live:habitat}{habitat}, and the
inhabitants find it.
\end{example}

\begin{remark}[Looking without disturbing]\label{rem:g2:where-animals-live:respect}
Visiting a \omterm{def:g2:where-animals-live:habitat}{habitat}, we are guests. Watch quietly, turn stones
back the way you found them, leave \omterm{def:g2:animals-grow-up:born}{eggs} and nests alone, and
carry your litter home --- the \omterm{def:g2:where-animals-live:habitat}{habitat} should forget you were
there. The next chapters give more ways to care for nature.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:1}
What is a \omterm{def:g2:where-animals-live:habitat}{habitat}? Name the four things an \omterm{def:g1:animals-around-us:animal}{animal} finds in it.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:2}
Give two \omterm{def:g1:animals-around-us:animal}{animals} of the pond, two of the forest, and two of the
meadow.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:3}
Which \omterm{def:g2:where-animals-live:habitat}{habitat} fits each \omterm{def:g1:animals-around-us:animal}{animal}: the stickleback, the woodpecker,
the grasshopper, the woodlouse?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:4}
What is the frog's swimming equipment? And the squirrel's
tree-climbing equipment?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:5}
List the mole's three pieces of underground equipment from
\cref{ex:g2:where-animals-live:mole}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:6}
In \cref{rem:g2:where-animals-live:layers}, which ``floor'' of
the forest does each \omterm{def:g1:animals-around-us:animal}{animal} use: woodpecker, squirrel, deer, wood
ant?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:7}
What happens to the frogs and dragonflies when a pond is filled
in? What does that teach about protecting \omterm{def:g1:animals-around-us:animal}{animals}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:8}
Try it: choose one small \omterm{def:g2:where-animals-live:habitat}{habitat} near you --- under a stone, a
bush, a flowerbed. Watch quietly and list every \omterm{def:g1:animals-around-us:animal}{animal} you see in
a fixed time. Put each stone back. How many inhabitants did your
little \omterm{def:g2:where-animals-live:habitat}{habitat} have?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:where-animals-live:9}
A mystery \omterm{def:g1:animals-around-us:animal}{animal} has webbed feet, dense \omterm{def:g1:animals-around-us:covering}{feathers} that shed
water, and a flat beak for sifting. Use
\cref{met:g2:where-animals-live:matching} to name its \omterm{def:g2:where-animals-live:habitat}{habitat} ---
and perhaps the \omterm{def:g1:animals-around-us:animal}{animal}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:where-animals-live:10}
Tom wants to ``help'' a frog by taking it home to a dry terrarium
with plenty of food. Using
\cref{prop:g2:where-animals-live:fit} and
\cref{prop:g2:where-animals-live:loss}, explain why this would
not help the frog at all.
\end{exercise}
