\chapter{Como as plantas se espalham}\label{ch:g6:how-plants-spread}

Um corte novo de estrada é aberto na máquina em março: terra crua,
nua, sem um pontinho verde. Volte em setembro e ele está todo
costurado de plantinhas --- papoulas, dentes-de-leão, \omterm{def:g6:life-through-seasons:buds}{brotos} de
salgueiro, capins. Ninguém plantou nada. E mesmo assim as \omterm{def:g1:plants-around-us:plant}{plantas}, que
não conseguem dar um único passo, colonizaram o terreno nu em poucos
meses. Este capítulo é sobre o departamento de viagens delas.

\section{As plantas viajam como sementes e esporos}

\begin{proposition}[Plantas paradas, começos que viajam]\label{prop:g6:how-plants-spread:travel}
Uma \omterm{def:g1:plants-around-us:plant}{planta} adulta fica presa no lugar a vida inteira
(\cref{def:g1:plants-around-us:plant}) --- então a \omterm{def:g5:biodiversity:species}{espécie} se espalha
mandando os \emph{começos} dela: \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} e \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos}
(\cref{ch:g5:flower-to-fruit}), \omterm{def:g4:plants-without-seeds:spore}{esporos}
(\cref{def:g4:plants-without-seeds:spore}) ou \omterm{ex:g4:plants-without-seeds:runner}{estolhos} e outros
avanços vegetativos
(\cref{def:g4:plants-without-seeds:vegetative}). Ocupar terreno novo
desse jeito se chama \emph{colonização}\index{colonização}.
\end{proposition}
\admitted

\begin{definition}[Dispersão]\label{def:g6:how-plants-spread:dispersal}
A \emph{dispersão}\index{dispersão} é o transporte de \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes},
\omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} ou \omterm{def:g4:plants-without-seeds:spore}{esporos} para longe da planta-mãe. Os transportadores são
poucos e antigos: o \emph{vento}, os \emph{\omterm{def:g1:animals-around-us:animal}{animais}}, a \emph{água}
--- e alguns \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} que arremessam as próprias \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}.
\end{definition}

\begin{example}[Por que sair de casa?]\label{ex:g6:how-plants-spread:whyleave}
Uma \omterm{def:g2:from-seed-to-plant:seed}{semente} que cai no pé da planta-mãe \omterm{def:g2:from-seed-to-plant:germination}{germina} na sombra da mãe,
entre as \omterm{def:g1:plants-around-us:parts}{raízes} da mãe --- disputando justamente a \omterm{def:g6:exploring-our-environment:conditions}{luz} e a água da
\cref{prop:g4:what-plants-need:needs} com uma rival cem vezes maior
que ela. A \omterm{def:g6:how-plants-spread:dispersal}{dispersão} é a fuga dessa disputa perdida: a distância é o
primeiro presente que uma \omterm{def:g1:plants-around-us:plant}{planta} dá aos filhos dela.
\end{example}

\section{Os transportadores e os frutos feitos para eles}

\begin{proposition}[Construídos para combinar com o transportador]\label{prop:g6:how-plants-spread:builds}
Os formatos dos \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} e das \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} combinam com os
transportadores deles --- é o encaixe da
\cref{def:g4:adapted-to-their-world:adaptation}, aplicado à bagagem:
\begin{itemize}
  \item \emph{vento}: paraquedas (as \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} emplumadas do
        dente-de-leão), asas (as sâmaras do bordo e do freixo, que
        descem girando de lado) ou \omterm{def:g4:plants-without-seeds:spore}{esporos} finos como poeira que
        simplesmente flutuam (os milhões da samambaia);
  \item \emph{\omterm{def:g1:animals-around-us:animal}{animais}, por fora}: carrapichos --- bardana, amor-de-hortelão
        --- cujos ganchos agarram pelo e meia e pegam carona até
        alguém tirá-los;
  \item \emph{\omterm{def:g1:animals-around-us:animal}{animais}, por dentro}: \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} doces e carnudos ---
        cerejas, amoras, \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} da roseira --- comidos inteiros; as
        \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} de casca dura passam ilesas e são depositadas em
        outro lugar, com adubo junto (a resposta do
        \cref{exo:g3:flowers-fruits-seeds:10}, completada);
  \item \emph{água}: \omterm{prop:g3:flowers-fruits-seeds:transform}{frutos} que boiam --- as jangadinhas do amieiro,
        a casca oceânica do coco;
  \item \emph{arremesso próprio}: vagens que secam, ficam tensas e
        estalam --- o tojo e as violetas atiram \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} a metros da
        planta-mãe.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Lendo a beira da estrada]\label{ex:g6:how-plants-spread:roadside}
O corte de estrada de setembro, decifrado: dentes-de-leão e
salgueiros --- \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} emplumadas, trazidas pelo vento; papoulas ---
as sementinhas delas já estavam na terra revolvida, esperando como só
\omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} sabem esperar (\cref{exo:g2:from-seed-to-plant:9});
plantinhas de amoreira silvestre debaixo dos mourões da cerca ---
depositadas por \omterm{prop:g3:sorting-living-things:groups}{aves} pousadas, com adubo incluso; bardana à beira do
caminho --- trazida grudada num cachorro. A rota de chegada de cada
colonizador está escrita no equipamento dele.
\end{example}

\begin{method}[Descobrir o transportador pelo fruto]\label{met:g6:how-plants-spread:name}
Diante de um \omterm{prop:g3:flowers-fruits-seeds:transform}{fruto} ou de uma \omterm{def:g2:from-seed-to-plant:seed}{semente}:
\begin{enumerate}
  \item procure equipamento de voo --- plumas, asas, finura de
        poeira: o vento;
  \item procure ganchos: pelo de \omterm{def:g1:animals-around-us:animal}{animal}, por fora;
  \item carnudo, colorido, doce? intestino de \omterm{def:g1:animals-around-us:animal}{animal}, por dentro ---
        confira se as \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} têm casca dura;
  \item boia, e fica perto da água? a água;
  \item uma vagem seca, torcida, com mola? arremesso próprio;
  \item teste o seu veredito: deixe cair, ponha para boiar, encoste
        numa manga de lã.
\end{enumerate}
\end{method}

\section{Colonizar sem sementes}

\begin{example}[O avanço rasteiro]\label{ex:g6:how-plants-spread:vegetative}
Ao lado das chegadas pelo ar, alguns colonizadores andam com os
próprios pés vegetativos (\cref{ch:g4:plants-without-seeds}): a
grama-seda avança debaixo da terra com \omterm{def:g1:plants-around-us:parts}{caules} rastejantes, metro a
metro; as amoreiras silvestres se arqueiam e criam \omterm{def:g1:plants-around-us:parts}{raiz} onde as
pontas encostam; os \omterm{ex:g4:plants-without-seeds:runner}{estolhos} do morangueiro saltitam pelo terreno
aberto. Devagar, de curto alcance --- e impossíveis de parar num
quintal abandonado, como sabe qualquer pessoa que cuide de horta.
\end{example}

\begin{remark}[Colonização em duas velocidades]\label{rem:g6:how-plants-spread:twospeeds}
Os dois sistemas dividem o trabalho. As \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} e os \omterm{def:g4:plants-without-seeds:spore}{esporos} são o
serviço de \emph{longo alcance} --- pousos raros e de sorte lá longe,
o dente-de-leão atravessando a cidade. O avanço vegetativo é o
serviço de \emph{curto alcance} --- ocupação densa e garantida do
terreno vizinho. A moita de amoreira silvestre que se espalha a partir
da cerca viva usa os dois ao mesmo tempo: \omterm{def:g1:plants-around-us:parts}{caules} arqueados para o
metro, \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} carregadas por \omterm{prop:g3:sorting-living-things:groups}{aves} para o quilômetro.
\end{remark}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:1}
O que é \omterm{prop:g6:how-plants-spread:travel}{colonização}? O que é \omterm{def:g6:how-plants-spread:dispersal}{dispersão}, e quais são os principais
transportadores?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:2}
Por que uma \omterm{def:g2:from-seed-to-plant:seed}{semente} se dá melhor germinando longe da planta-mãe?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:3}
Diga o transportador de cada um: a pluma do dente-de-leão, o

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:4}
Como as \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} de uma amora comida por uma ave sobrevivem à viagem,

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:5}
Que serviço de viagem o tojo e as violetas usam? Descreva o
mecanismo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:6}
Diga dois colonizadores vegetativos e como cada um avança.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:7}
Decifre o corte de estrada: para o dente-de-leão, a papoula, a

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:8}
Rode o \cref{met:g6:how-plants-spread:name} em: (a) uma \omterm{def:g2:from-seed-to-plant:seed}{semente}
emplumada de cardo; (b) um \omterm{prop:g3:flowers-fruits-seeds:transform}{fruto} de roseira; (c) uma bolinha de

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:9}
Explique a ideia das ``duas velocidades'' da
\cref{rem:g6:how-plants-spread:twospeeds} com a moita de amoreira

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:10}
A samambaia colonizou a margem úmida e sombreada do corte novo, mas
não a margem seca e ensolarada. Combine a
\cref{prop:g4:plants-without-seeds:damp} e a
\cref{prop:g6:exploring-our-environment:decide} para montar a
explicação.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:11}
Uma ilha nova emerge nua do mar, a quilômetros da costa. Que serviços
de \omterm{def:g6:how-plants-spread:dispersal}{dispersão} conseguem chegar até ela? Preveja o equipamento das
primeiras \omterm{def:g1:plants-around-us:plant}{plantas} colonizadoras dela.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:how-plants-spread:12}
Construir polpa doce é caro para uma \omterm{def:g1:plants-around-us:plant}{planta}. Usando o
\cref{ex:g6:how-plants-spread:whyleave} e o serviço por dentro dos
\omterm{def:g1:animals-around-us:animal}{animais}, explique o que a cerejeira compra com todo aquele açúcar ---
e por que as cascas duras das \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} são a outra metade da compra.
\end{exercise}

\section{Problema: O campo abandonado}

\begin{problem}[{Problema de fim de semana --- dez anos de
colonização, observados de um portão}]\label{pb:g6:how-plants-spread:1}
Uma lavoura arada é abandonada no outono. Um naturalista fotografa
essa lavoura do portão todo mês de junho durante dez anos e mantém
anotações.

\medskip\noindent\textbf{Parte I --- O primeiro ano.}
\begin{enumerate}
  \item Junho do primeiro ano: a terra nua está salpicada de
        papoulas, dentes-de-leão e cardos. Para cada um, diga a
        origem e a rota prováveis --- as papoulas vindas da própria
        terra, os outros pelo ar. Que equipamento tornou possíveis
        as chegadas pelo ar?
  \item Nenhuma amoreira silvestre aparece no meio do campo no
        primeiro ano, mas três plantinhas estão de pé debaixo dos
        mourões velhos da cerca. Explique o padrão com o serviço por
        dentro dos \omterm{def:g1:animals-around-us:animal}{animais} --- por que exatamente debaixo dos
        mourões?
  \item O naturalista não encontra samambaia nenhuma, embora um
        bosque cheio delas fique do lado de onde vem o vento. Os
        \omterm{def:g4:plants-without-seeds:spore}{esporos} das \omterm{prop:g4:plants-without-seeds:damp}{samambaias} com certeza chegaram (o
        \cref{ex:g4:plants-without-seeds:fern} conta os milhões
        deles). Que condição do campo aberto, segundo a
        \cref{prop:g4:plants-without-seeds:damp}, recusou esses
        \omterm{def:g4:plants-without-seeds:spore}{esporos}?
  \item A grama-seda invade só a partir da borda do campo, um metro
        ou dois por ano. Que sistema de \omterm{prop:g6:how-plants-spread:travel}{colonização} é esse, e por
        que ele não pode aparecer no meio do campo como os
        dentes-de-leão apareceram?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Os anos seguintes.}
\begin{enumerate}[resume]
  \item No terceiro ano, capins e ervas altas fecharam o terreno nu,
        e as papoulas sumiram da vegetação em pé. Onde estão as
        papoulas agora, e em que forma? (Lembre o corte de estrada e
        a \cref{prop:g6:life-through-seasons:herbs}.)
  \item A partir do quarto ano, arcos de amoreira silvestre e
        pequenos espinheiros salpicam o campo, mais densos perto das
        cercas vivas e ao longo da linha da cerca. Diga os dois
        serviços que puseram essas \omterm{def:g1:plants-around-us:plant}{plantas} ali e a prova disso no
        padrão.
  \item No oitavo ano, mudas de salgueiro e de bétula estão mais
        altas que as amoreiras --- as duas árvores com \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}
        minúsculas, emplumadas ou leves como poeira. Por que os
        carvalhos e os castanheiros, de \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} pesadas, ainda
        estão quase ausentes, e que transportador \omterm{def:g1:animals-around-us:animal}{animal} acaba
        plantando até esses?
  \item As fotografias mostram o elenco do campo mudando ano a ano,
        embora nenhum jardineiro tenha encostado nele. Enuncie a
        regra que comanda isso: o que a vegetação de cada ano muda
        nas próprias \omterm{def:g6:exploring-our-environment:conditions}{condições} do campo --- a \omterm{def:g6:exploring-our-environment:conditions}{luz} ao nível do chão
        acima de tudo --- e a quem essa mudança favorece em seguida?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- A lição geral.}
\begin{enumerate}[resume]
  \item Ponha os colonizadores em ordem de chegada: as \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} que
        esperavam na terra, os que vêm com o vento, os arbustos
        trazidos por \omterm{def:g1:animals-around-us:animal}{animais}, as árvores de \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} pesadas.
        Emparelhe cada onda com o serviço de \omterm{def:g6:how-plants-spread:dispersal}{dispersão} que definiu o
        tempo dela.
  \item As rachaduras do calçamento de uma praça brotam o mesmo
        elenco do primeiro ano --- papoulas, dentes-de-leão --- mas
        nunca chegam aos arbustos do quarto ano. Dê dois motivos
        pelos quais a sucessão empaca ali.
  \item O naturalista afirma: ``deixado em paz, este campo vai virar
        mata''. Defenda a afirmação com a tendência dos dez anos ---
        e diga a exceção do campo ceifado da
        \cref{prop:g5:people-and-nature:managed}, que prova que
        ``deixado em paz'' é a expressão decisiva.
  \item Resuma o problema inteiro em duas frases, usando as palavras
        \emph{\omterm{def:g6:how-plants-spread:dispersal}{dispersão}}, \emph{\omterm{def:g6:exploring-our-environment:conditions}{condições}} e \emph{\omterm{prop:g6:how-plants-spread:travel}{colonização}}.
\end{enumerate}
\end{problem}
