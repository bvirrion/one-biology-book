\section*{Bab \ref{ch:g3:sorting-living-things} --- Mengelompokkan Makhluk Hidup}

\begin{solution}{exo:g3:sorting-living-things:1}
\omterm{def:g3:sorting-living-things:criterion}{Kriteria} adalah aturan yang dipakai untuk mengelompokkan. Yang baik:
ciri tubuh yang dapat kamu amati --- ``berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}'', ``berkaki
enam''. Yang dihindari: tempat ia hidup atau apa yang kebetulan sedang
dilakukannya --- ``hidup dekat air'', ``terbang''.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:2}
\omterm{prop:g3:sorting-living-things:groups}{Serangga}: berkaki enam dan bertubuh tiga bagian. \omterm{prop:g3:sorting-living-things:groups}{Ikan}: bersirip dan
bersisik, hidup di dalam air. \omterm{prop:g3:sorting-living-things:groups}{Burung}: berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}, bersayap dan
berparuh. \omterm{prop:g3:sorting-living-things:groups}{Mamalia}: berbulu atau berambut, dan induknya memberi anaknya
\omterm{def:g2:animals-grow-up:born}{susu}.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:3}
Lebah: \omterm{prop:g3:sorting-living-things:groups}{serangga}. \omterm{prop:g3:sorting-living-things:groups}{Ikan} mas koki: \omterm{prop:g3:sorting-living-things:groups}{ikan}. \omterm{prop:g3:sorting-living-things:groups}{Burung} unta: \omterm{prop:g3:sorting-living-things:groups}{burung}. Sapi:
\omterm{prop:g3:sorting-living-things:groups}{mamalia}. Katak: \omterm{prop:g3:sorting-living-things:groups}{amfibi}.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:4}
Ia berbulu dan memberi anaknya \omterm{def:g2:animals-grow-up:born}{susu}, serta tidak berbulu \omterm{prop:g3:sorting-living-things:groups}{burung} dan
tidak berparuh. Terbang adalah sesuatu yang dilakukannya, bukan
sesuatu yang menentukan seekor \omterm{prop:g3:sorting-living-things:groups}{burung} --- ciri-cirinya berkata:
\omterm{prop:g3:sorting-living-things:groups}{mamalia}.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:5}
Ia tidak bersisik dan tidak bertubuh insang-dan-sirip seperti \omterm{prop:g3:sorting-living-things:groups}{ikan} ---
dan ia memberi anaknya \omterm{def:g2:animals-grow-up:born}{susu}. Berenang adalah kebiasaannya;
ciri-cirinya berkata: \omterm{prop:g3:sorting-living-things:groups}{mamalia}.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:6}
Tidak. \omterm{def:g3:sorting-living-things:criterion}{Kriteria} \omterm{prop:g3:sorting-living-things:groups}{serangga} adalah berkaki enam (dan bertubuh tiga
bagian); kedelapan kaki laba-laba tidak memenuhinya. Laba-labanya
termasuk kelompok lain, seperti apa pun rupanya pada pandangan
pertama.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:7}
\omterm{def:g1:plants-around-us:parts}{Bunga} aster: berbunga dan berbiji, \omterm{def:g1:plants-around-us:parts}{batang} hijau yang lunak. Lumut:
tidak pernah berbunga, tidak berbatang sejati. Pohon pinus: sebuah
pohon (\omterm{def:g1:plants-around-us:tree}{batang pokok} berkayu), \omterm{def:g1:plants-around-us:parts}{daun} berupa jarum, tidak berbunga tetapi
berbuah runjung yang berbiji. Semak mawar: berbunga dan berbiji,
\omterm{def:g1:plants-around-us:parts}{batang} berkayu.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:8}
Jawaban terbuka. Di kebanyakan jalan, seranggalah yang menang jauh ---
semut, lalat, lebah, kumbang, kupu-kupu --- dengan \omterm{prop:g3:sorting-living-things:groups}{burung} di tempat
kedua.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:9}
\omterm{def:g3:sorting-living-things:criterion}{Pengelompokan} temannya. ``Peliharaan lawan liar'' mengelompokkan
menurut cara \omterm{def:g1:animals-around-us:animal}{hewan} hidup bersama manusia, yang dapat berubah ---
seekor kucing liar adalah \omterm{def:g1:animals-around-us:animal}{hewan} yang sama di dalam maupun di luar
rumah. ``\omterm{prop:g3:sorting-living-things:groups}{Mamalia} lawan \omterm{prop:g3:sorting-living-things:groups}{burung}'' mengelompokkan menurut ciri tubuh yang
menetap seumur hidup: jenis \omterm{def:g3:sorting-living-things:criterion}{kriteria} milik ahli \omterm{def:g1:living-or-not:living}{biologi}.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:10}
Langkah 1: penutupnya --- \omterm{def:g1:animals-around-us:covering}{bulu burung} (tersusun rapat), sebuah paruh,
dua kaki, sayap (berupa sirip pendayung). Langkah 2: anaknya menetas
dari \omterm{def:g2:animals-grow-up:born}{telur}. Langkah 3: ciri-cirinya cocok dengan \omterm{prop:g3:sorting-living-things:groups}{burung}. Langkah 4:
kesan ``berenang seperti \omterm{prop:g3:sorting-living-things:groups}{ikan}, tidak dapat terbang'' bertentangan ---
dan kalah. Penguinnya seekor \omterm{prop:g3:sorting-living-things:groups}{burung}.
\end{solution}
