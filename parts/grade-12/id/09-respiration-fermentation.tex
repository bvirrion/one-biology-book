\chapter{Respirasi Sel dan Fermentasi}\label{ch:g12:respiration-fermentation}

Otot seorang pelari maraton membelanjakan, sepanjang tiga setengah
jam, sekitar setengah mol \omterm{def:g12:respiration-fermentation:atp}{ATP} setiap menit --- seperempat kilogram
molekul itu, yang dibuat, dipakai dan dibuat ulang ribuan kali dari
stok yang hanya cukup untuk lima detik. Setiap molekul \omterm{def:g12:respiration-fermentation:atp}{ATP} itu dibayar
oleh sebutir glukosa atau asam lemak yang dibongkar, atom demi atom,
dan oleh sebutir molekul oksigen yang dihirup pada awalnya dan
diembuskan sebagai air pada akhirnya.
\cref{ch:g10:cell-metabolism} memberi neracanya; bab ini membuka
\omterm{def:g10:cells-common-unit:organelle}{mitokondria} dan mengikuti glukosa melewati tiga tahap yang mengubah
energinya menjadi \omterm{def:g12:respiration-fermentation:atp}{ATP} --- serta menunjukkan apa yang dilakukan sebuah
\omterm{def:g10:cells-common-unit:cell}{sel} ketika oksigennya habis.

\section{ATP, mata uangnya}

\begin{definition}[ATP]\label{def:g12:respiration-fermentation:atp}
\emph{ATP}\index{ATP} (adenosina trifosfat) adalah sebuah \omterm{def:g10:universal-dna:nucleotide}{nukleotida}
yang membawa tiga gugus fosfat berderet. Melepaskan yang terakhir
membebaskan sekitar \qty{30}{kJ} per mol pada kondisi \omterm{def:g10:cells-common-unit:cell}{sel}, dan setiap
proses \omterm{def:g10:cells-common-unit:cell}{sel} yang menuntut energi --- kontraksi, pengangkutan,
pembuatan, cahaya seekor kunang-kunang --- digerakkan oleh pelepasan
itu. Produknya, ADP, diisi ulang menjadi ATP oleh reaksi dalam bab
ini. Sebuah \omterm{def:g10:cells-common-unit:cell}{sel} memuat ATP sebanyak beberapa detik pemakaian dan
memutarnya terus-menerus; manusia yang sedang beristirahat mendaur
ulang ATP kira-kira sebanyak massa tubuhnya sendiri dalam sehari.
\end{definition}

\begin{proposition}[Di mana energinya berada]\label{prop:g12:respiration-fermentation:oxidation}
Glukosa menyimpan energi dalam ikatan karbon--hidrogennya;
membebaskannya berarti memindahkan hidrogen itu, beserta elektronnya,
kepada oksigen --- mengoksidasi glukosanya menjadi karbon dioksida dan
mereduksi oksigennya menjadi air. Bila dikerjakan dalam satu langkah,
seperti dalam nyala api, energinya akan pergi sebagai bahang. \omterm{def:g10:cells-common-unit:cell}{Sel}
mengerjakannya dalam puluhan langkah kecil, masing-masing dijalankan
sebuah \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}, dan menangkap sebagian energinya pada beberapa langkah:
hidrogennya mula-mula dimuatkan ke molekul pengangkut --- \emph{NAD
tereduksi} --- dan baru pada akhirnya diserahkan kepada oksigen.
\end{proposition}
\admitted

\section{Tiga tahap}

\begin{proposition}[Glikolisis]\label{prop:g12:respiration-fermentation:glycolysis}
Di dalam \omterm{def:g10:cells-common-unit:cell}{sitoplasma}, tanpa oksigen, sebutir glukosa (enam karbon)
dibelah menjadi dua molekul \emph{piruvat} (masing-masing tiga karbon)
oleh sepuluh langkah \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim}: \emph{glikolisis}\index{glikolisis}.
Neracanya: dua \omterm{def:g12:respiration-fermentation:atp}{ATP} terpakai untuk memulai, empat dihasilkan, jadi
keuntungan bersihnya 2 \omterm{def:g12:respiration-fermentation:atp}{ATP}, dan dua NAD tereduksi yang bermuatan
hidrogen. Piruvatnya masih memuat sebagian besar energi glukosanya.
\end{proposition}
\admitted

\begin{proposition}[Mitokondria merampungkan pengoksidasiannya]\label{prop:g12:respiration-fermentation:mitochondrion}
Ketika oksigen tersedia, piruvat masuk ke \emph{mitokondria}\index{mitokondria},
sebuah \omterm{def:g10:cells-common-unit:organelle}{organel} yang dibatasi dua membran, dan membran dalamnya berlipat
menjadi \emph{krista} yang mengurung cairan \emph{matriks}.
\begin{itemize}
  \item Di dalam matriks, setiap piruvat dibongkar dalam sebuah siklus
        reaksi, yaitu \emph{siklus Krebs}\index{siklus Krebs}: ketiga
        karbonnya pergi sebagai tiga $\mathrm{CO_2}$, hidrogennya
        dimuatkan ke pengangkut (NAD tereduksi dan sebuah pengangkut
        kedua), dan sedikit \omterm{def:g12:respiration-fermentation:atp}{ATP} dibuat langsung --- satu per piruvat.
  \item Di dalam membran dalam, pengangkut yang bermuatan itu
        menyerahkan elektronnya kepada sederet \omterm{def:g11:gene-expression:protein}{protein}, yaitu
        \emph{rantai respirasi}, yang mengoperkannya turun sampai ke
        oksigen, yang lalu bergabung dengan proton membentuk air.
        Energi yang dibebaskan sepanjang rantai itu memompa proton
        menyeberangi membrannya, dan kembalinya proton lewat sebuah
        \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} berputar menggerakkan pembuatan \omterm{def:g12:respiration-fermentation:atp}{ATP} --- sekitar 26 per
        glukosa.
\end{itemize}
Seluruhnya untuk satu glukosa: sekitar 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} (2 dari \omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisis}, 2
dari siklusnya, sekitar 26 dari rantainya), enam $\mathrm{CO_2}$
dilepaskan, enam $\mathrm{O_2}$ terpakai, dan sisa \qty{2870}{kJ}-nya
menjadi bahang.
\end{proposition}
\begin{proof}[Bukti percobaan]
\omterm{def:g10:cells-common-unit:organelle}{Mitokondria} terasing di dalam bejana tertutup dengan sensor oksigen
tidak memakai oksigen ketika diberi glukosa, tetapi memakainya dengan
cepat ketika diberi piruvat --- \omterm{def:g10:cells-common-unit:organelle}{mitokondria} tak dapat mulai dari
glukosa; sitoplasmalah yang harus. Menambahkan racun rantai respirasi
(sianida) menghentikan pemakaiannya seketika, apa pun \omterm{def:g11:enzymes-and-phenotype:enzyme}{substratnya}.
Menambahkan ADP mempercepat pemakaiannya dan habisnya ADP
memperlambatnya: pemakaian oksigen terkopel dengan pembuatan \omterm{def:g12:respiration-fermentation:atp}{ATP}. Dan
\omterm{def:g10:cells-common-unit:organelle}{mitokondria} otot terbang seekor burung kolibri atau paha seorang pelari
maraton punya krista yang berjejal beberapa kali lebih rapat daripada
\omterm{def:g10:cells-common-unit:organelle}{mitokondria} jaringan yang beristirahat.
\end{proof}

\section{Tanpa oksigen: fermentasi}

\begin{proposition}[Fermentasi membentuk kembali pengangkutnya]\label{prop:g12:respiration-fermentation:fermentation}
\omterm{prop:g12:respiration-fermentation:glycolysis}{Glikolisis} memuati dua pengangkut NAD per glukosa; dengan oksigen,
\omterm{def:g10:cells-common-unit:organelle}{mitokondria} membongkar muatannya. Tanpa oksigen semua pengangkutnya
akan bermuatan dalam beberapa detik dan \omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisisnya} akan berhenti.
\emph{Fermentasi}\index{fermentasi} adalah cara \omterm{def:g10:cells-common-unit:cell}{sitoplasma}
membongkarnya: piruvat itu sendiri menerima hidrogennya lalu menjadi
\emph{laktat} (\omterm{def:g10:cells-common-unit:cell}{sel} otot, bakteri susu) atau, setelah kehilangan sebuah
$\mathrm{CO_2}$, menjadi \emph{etanol} (ragi). Tak ada \omterm{def:g12:respiration-fermentation:atp}{ATP} tambahan
yang dibuat; hasilnya adalah 2 \omterm{def:g12:respiration-fermentation:atp}{ATP} \omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisis}, sekitar lima belas kali
lebih sedikit daripada hasil respirasi, dan produknya --- laktat atau
etanol --- masih memuat sebagian besar energi glukosanya.
\end{proposition}
\admitted

\begin{example}[Pilihan sang otot]\label{ex:g12:respiration-fermentation:muscle}
\omterm{def:g10:muscles-and-joints:muscle}{Serat otot} seorang pelari cepat memerlukan \omterm{def:g12:respiration-fermentation:atp}{ATP} lebih cepat daripada
kemampuan \omterm{def:g10:cells-common-unit:organelle}{mitokondria} dan pasokan oksigennya membuatnya; \omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisis}
dengan \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} laktat mengantarkan \omterm{def:g12:respiration-fermentation:atp}{ATP} tiga kali lebih cepat,
dengan ongkos glukosa lima belas kali lipat, dan laktatnya menumpuk
sampai nyerinya menghentikan usahanya. Serat seorang pelari maraton
berjalan dengan respirasi, pelan dan nyaris tanpa batas, membakar lemak
selain glukosa; \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya} lebih besar dan lebih banyak, kapilernya
lebih rapat, dan laktatnya tetap rendah. Jenis serat pada
\cref{ch:g10:muscles-and-joints} adalah kedua siasat itu yang terpasang
di dalam \omterm{def:g10:cells-common-unit:cell}{sel}.
\end{example}

\begin{method}[Menyeimbangkan sebuah anggaran energi]\label{met:g12:respiration-fermentation:budget}
\begin{enumerate}
  \item Ubahlah daya yang diperlukan menjadi \omterm{def:g12:respiration-fermentation:atp}{ATP}: pada sekitar
        \qty{30}{kJ} per mol \omterm{def:g12:respiration-fermentation:atp}{ATP} yang dapat dipakai, \qty{1}{kW} daya
        metabolik setara \qty{2}{mol} \omterm{def:g12:respiration-fermentation:atp}{ATP} per menit.
  \item Ubahlah \omterm{def:g12:respiration-fermentation:atp}{ATP} menjadi bahan bakar: 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} per glukosa pada
        respirasi, 2 pada \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}.
  \item Ubahlah bahan bakar menjadi oksigen: 6 $\mathrm{O_2}$ per
        glukosa yang direspirasikan, \qty{24}{L} per mol gas.
  \item Bandingkan oksigen yang diperlukan dengan yang diantarkan
        darah (\cref{ch:g10:heart-lungs-effort}): kekurangannya ditutup
        oleh \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}, beserta laktatnya.
\end{enumerate}
\end{method}

\begin{remark}[Cermin fotosintesis]\label{rem:g12:respiration-fermentation:mirror}
\omterm{def:g10:cell-metabolism:autotrophy}{Fotosintesis} memuatkan elektron dari air ke pengangkut dengan cahaya
lalu memakainya untuk mereduksi karbon dioksida menjadi gula, sambil
melepaskan oksigen; respirasi membongkar elektron dari gula ke
pengangkut lalu mengoperkannya kepada oksigen, membuat air dan
melepaskan karbon dioksida, serta memakai energinya untuk membuat \omterm{def:g12:respiration-fermentation:atp}{ATP}.
Keduanya dijalankan \omterm{def:g10:cells-common-unit:organelle}{organel} yang dahulu sama-sama bakteri bebas
(\cref{ch:g12:diversification-of-life}), dengan \omterm{def:g12:photosynthesis:chloroplast}{kloroplas} memasok apa
yang dipakai \omterm{def:g10:cells-common-unit:organelle}{mitokondria}; bersama-sama keduanya mengubah sinar matahari
menjadi mata uang setiap \omterm{def:g10:cells-common-unit:cell}{sel}, dan oksigen atmosfer adalah saldo
pembukuan keduanya.
\end{remark}

\section{Soal: Setengah Mol ATP Semenit}

\begin{problem}[{Soal akhir pekan --- energi seorang pelari diikuti
dari watt sampai molekul: ATP-nya dicacah, glukosa dan oksigennya
dihitung, bagian mitokondrianya diukur, dan lari cepat yang tak sanggup
dibayar rantainya}]
\label{pb:g12:respiration-fermentation:1}
Seorang pelari menahan daya metabolik \qty{1000}{W}, dan sekitar
sepertiganya ditangkap sebagai \omterm{def:g12:respiration-fermentation:atp}{ATP}, sisanya pergi sebagai bahang.
Ambillah \qty{30}{kJ} energi yang dapat dipakai per mol \omterm{def:g12:respiration-fermentation:atp}{ATP}, 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} per
glukosa pada respirasi dan 2 pada \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}, \qty{2870}{kJ} per mol
glukosa, \qty{24}{L} per mol gas, dan glukosa \qty{180}{g/mol}.

\medskip
\noindent\textbf{Bagian I --- \omterm{def:g12:respiration-fermentation:atp}{ATP-nya}.}
\begin{enumerate}
  \item Berapa mol \omterm{def:g12:respiration-fermentation:atp}{ATP} yang dipakai pelari itu tiap menit?
        (Hanya sepertiga yang tertangkap sebagai \omterm{def:g12:respiration-fermentation:atp}{ATP} yang dihitung.)
  \item Berapa massa \omterm{def:g12:respiration-fermentation:atp}{ATP} itu, pada \qty{507}{g/mol}? Bandingkan dengan
        beberapa gram yang dimuat ototnya pada suatu saat.
  \item Berapa kali tiap menit setiap molekul \omterm{def:g12:respiration-fermentation:atp}{ATP} diisi ulang, jika
        stok tubuhnya \qty{50}{g}?
  \item Jika \omterm{def:g12:respiration-fermentation:atp}{ATP} itu dibuat oleh respirasi saja, berapa mol glukosa
        per menit yang akan terpakai?
  \item Berapa liter oksigen per menit yang diperlukannya? Bandingkan
        dengan $\dot V\!\mathrm{O_2}$maks sebesar \qty{4}{L/min}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Bagian \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya}.}
\begin{enumerate}[resume]
  \item Dari 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} per glukosa, berapa yang dibuat di \omterm{def:g10:cells-common-unit:cell}{sitoplasma} dan
        berapa di \omterm{def:g10:cells-common-unit:organelle}{mitokondria}? Berapa bagian \omterm{def:g12:respiration-fermentation:atp}{ATP} pelari itu yang
        berasal dari \omterm{def:g10:cells-common-unit:organelle}{mitokondria}?
  \item Berapa mol $\mathrm{CO_2}$ yang diembuskannya tiap menit, dan
        dari tahap respirasi yang mana asalnya?
  \item Berapa banyak dari \qty{2870}{kJ} sebutir glukosa yang berakhir
        di dalam \omterm{def:g12:respiration-fermentation:atp}{ATP}? Ke mana sisanya pergi, dan apa yang dilakukan
        pelari itu terhadapnya?
  \item \omterm{def:g10:cells-common-unit:organelle}{Mitokondria} ototnya punya luas membran dalam seluruhnya
        sekitar \qty{1000}{m^2}. Mengapa rantainya memerlukan permukaan
        seluas itu?
  \item \omterm{def:g10:sport-and-health:training}{Latihan} melipatduakan \omterm{def:g10:cells-common-unit:organelle}{mitokondria} seratnya. Besaran mana di
        atas yang berubah karenanya, dan mana yang tidak?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Lari cepatnya.}
Pada lari cepat terakhir ototnya memerlukan daya \qty{800}{W} sebagai
\omterm{def:g12:respiration-fermentation:atp}{ATP} selama 30 detik, sementara respirasi, yang dibatasi oksigen yang
diantarkan darah, paling banyak dapat memasok \qty{400}{W} sebagai \omterm{def:g12:respiration-fermentation:atp}{ATP}.
\begin{enumerate}[resume]
  \item Berapa mol \omterm{def:g12:respiration-fermentation:atp}{ATP} seluruhnya yang diperlukan lari cepat itu?
  \item Berapa banyak yang dapat dipasok respirasi dalam 30 detik, dan
        berapa yang harus datang dari \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}?
  \item Berapa mol glukosa yang dipakai bagian yang difermentasikan
        itu, dan berapa mol laktat yang ditinggalkannya?
  \item Bandingkan glukosa yang terpakai per \omterm{def:g12:respiration-fermentation:atp}{ATP} pada kedua jalur itu
        selama lari cepatnya. Mengapa ototnya menerima pemborosan itu?
  \item Setelah lomba, laktatnya dioksidasi atau diubah kembali
        menjadi glukosa. Jelaskan mengapa pemakaian oksigennya tetap
        tinggi selama beberapa menit setelah ia berhenti.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Bahan bakar yang lain.}
Lemak memasok sekitar \qty{38}{kJ} per gram dan, per gram, memerlukan
sekitar 20\% lebih banyak oksigen daripada glukosa untuk energi yang sama.
\begin{enumerate}[resume]
  \item Jika separuh dari \qty{1000}{W}-nya datang dari lemak, berapa
        massa lemak yang akan dibakarnya tiap jam?
  \item Mengapa lemak, meskipun lebih kaya energi per gram, memberi
        daya lebih kecil per liter oksigen?
  \item Asam lemak masuk ke respirasi pada \omterm{prop:g12:respiration-fermentation:mitochondrion}{siklus Krebs}, melewatkan
        \omterm{prop:g12:respiration-fermentation:glycolysis}{glikolisis}. Jadi, tahap mana di antara ketiganya yang tak dapat
        dipakai lemak, dan apa akibatnya bagi lari cepat?
  \item Mengapa otak, yang berjalan dengan glukosa saja, tetap bekerja
        selama lomba walaupun ototnya mengambil sebagian besar glukosa
        darahnya?
  \item Nyatakan hasilnya: berapa mol \omterm{def:g12:respiration-fermentation:atp}{ATP} yang dipakai pelari itu tiap
        menit, berapa liter oksigen yang membayarnya, dan tahap
        respirasi mana yang membuat sebagian besarnya.
\end{enumerate}
\end{problem}
