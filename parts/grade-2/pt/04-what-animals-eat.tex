\chapter{O que os animais comem}\label{ch:g2:what-animals-eat}

O coelho do campo fica o dia inteiro mordiscando capim. A raposa que
atravessa o mesmo campo ao anoitecer não está nem um pouco
interessada no capim --- ela está interessada no coelho. E o sabiá
come minhoca de manhã e cereja de tarde. Todo \omterm{def:g1:animals-around-us:animal}{animal} precisa comer,
mas nem todo \omterm{def:g1:animals-around-us:animal}{animal} come as mesmas coisas.

\section{Três tipos de comedores}

\begin{definition}[Herbívoro, carnívoro, onívoro]\label{def:g2:what-animals-eat:kinds}
Os \omterm{def:g1:animals-around-us:animal}{animais} são separados pelo que comem:
\begin{itemize}
  \item um \emph{herbívoro}\index{herbívoro} come \omterm{def:g1:plants-around-us:plant}{plantas}: capim,
        \omterm{def:g1:plants-around-us:parts}{folhas}, \omterm{def:g2:from-seed-to-plant:seed}{sementes}, frutas --- o coelho, a vaca, o caracol;
  \item um \emph{carnívoro}\index{carnívoro} come outros \omterm{def:g1:animals-around-us:animal}{animais}: a
        raposa, o gato, a joaninha (que caça pulgõezinhos das
        \omterm{def:g1:plants-around-us:plant}{plantas});
  \item um \emph{onívoro}\index{onívoro} come \omterm{def:g1:plants-around-us:plant}{plantas} e \omterm{def:g1:animals-around-us:animal}{animais}: o
        sabiá, o javali --- e os seres humanos.
\end{itemize}
\end{definition}

\begin{example}[Um campo, três cardápios]\label{ex:g2:what-animals-eat:meadow}
Num campo de verão: o coelho apara o capim (\omterm{def:g2:what-animals-eat:kinds}{herbívoro}); a raposa
caça o coelho (\omterm{def:g2:what-animals-eat:kinds}{carnívoro}); o sabiá pega uma minhoca agora e uma
cereja mais tarde (\omterm{def:g2:what-animals-eat:kinds}{onívoro}). Três vizinhos, três cardápios --- e
repare que o jantar da raposa comeu capim antes.
\end{example}

\begin{remark}[Carnívoros não são vilões]\label{rem:g2:what-animals-eat:fox}
A raposa não é ``malvada'' por comer coelhos: caçar é simplesmente o
jeito de um \omterm{def:g2:what-animals-eat:kinds}{carnívoro} se alimentar --- tão natural quanto o coelho
mordiscar. Na natureza, comer e ser comido fazem parte do modo como
a vida funciona; um capítulo mais adiante vai ligar todas essas
refeições em cadeias alimentares.
\end{remark}

\section{Ler o cardápio de um animal pelo corpo dele}

\begin{proposition}[Os corpos combinam com os cardápios]\label{prop:g2:what-animals-eat:match}
O corpo de um \omterm{def:g1:animals-around-us:animal}{animal} traz pistas sobre o alimento dele:
\begin{itemize}
  \item os \omterm{def:g2:what-animals-eat:kinds}{herbívoros}, como o coelho e a vaca, têm dentes de trás
        grandes e chatos para triturar \omterm{def:g1:plants-around-us:plant}{plantas}, e passam longas
        horas comendo;
  \item os \omterm{def:g2:what-animals-eat:kinds}{carnívoros}, como o gato e a raposa, têm dentes pontudos
        para agarrar a presa, garras afiadas e \omterm{def:g1:the-five-senses:sense}{olhos} voltados para
        a frente, bons para caçar;
  \item nas aves isso aparece no bico: curto e grosso para quebrar
        \omterm{def:g2:from-seed-to-plant:seed}{sementes} (o pardal), curvo e em forma de gancho para
        rasgar carne (a águia), comprido e fino para pescar insetos
        na casca das árvores.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Os dentes do gato]\label{ex:g2:what-animals-eat:cat}
Olhe um gato bocejar: dentes compridos e pontudos nos cantos da
frente da boca. Esses são dentes de agarrar --- o equipamento de um
\omterm{def:g2:what-animals-eat:kinds}{carnívoro}. A vaca, bocejando, não mostraria nenhum: a boca dela é
toda de trituradores largos e chatos, feitos para o capim.
\end{example}

\begin{method}[Adivinhar um cardápio]\label{met:g2:what-animals-eat:guess}
Para adivinhar o que um \omterm{def:g1:animals-around-us:animal}{animal} desconhecido come:
\begin{enumerate}
  \item olhe os dentes ou o bico: trituradores chatos, agarradores
        pontudos, ou os dois?
  \item olhe as garras e os \omterm{def:g1:the-five-senses:sense}{olhos}: são ferramentas de caça ou não?
  \item observe o \omterm{def:g1:animals-around-us:animal}{animal} se alimentando, se der --- a pista mais
        certeira de todas;
  \item depois dê o nome: \omterm{def:g2:what-animals-eat:kinds}{herbívoro}, \omterm{def:g2:what-animals-eat:kinds}{carnívoro} ou \omterm{def:g2:what-animals-eat:kinds}{onívoro}.
\end{enumerate}
\end{method}

\section{Comer o bastante, dia após dia}

\begin{proposition}[Alimentar-se é tarefa de todo dia]\label{prop:g2:what-animals-eat:daily}
O alimento é o combustível e o material de construção do corpo, então
os \omterm{def:g1:animals-around-us:animal}{animais} precisam comer de novo e de novo, dia após dia. Quanto
tempo isso leva depende do cardápio: o capim dá pouco em cada
bocado, então a vaca pasta durante horas; a carne é rica, então a
raposa pode fazer uma boa refeição e descansar.
\end{proposition}
\admitted

\begin{example}[O inverno muda o cardápio]\label{ex:g2:what-animals-eat:winter}
O alimento muda com as estações. O sabiá come minhocas e insetos no
verão, mas no inverno, quando o chão está duro, ele passa para as
frutinhas. O cardápio largo de um \omterm{def:g2:what-animals-eat:kinds}{onívoro} é uma grande ajuda quando
os tempos ficam difíceis.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:1}
Diga as palavras para: um \omterm{def:g1:animals-around-us:animal}{animal} que come \omterm{def:g1:plants-around-us:plant}{plantas}; um \omterm{def:g1:animals-around-us:animal}{animal} que
come outros \omterm{def:g1:animals-around-us:animal}{animais}; um \omterm{def:g1:animals-around-us:animal}{animal} que come os dois.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:2}
\omterm{def:g2:what-animals-eat:kinds}{Herbívoro}, \omterm{def:g2:what-animals-eat:kinds}{carnívoro} ou \omterm{def:g2:what-animals-eat:kinds}{onívoro}: (a) a vaca; (b) a raposa; (c) o
sabiá; (d) o caracol; (e) o ser humano?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:3}
De que dentes um \omterm{def:g2:what-animals-eat:kinds}{herbívoro} precisa mais: trituradores chatos ou
agarradores pontudos? Por quê?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:4}
Ligue cada bico a um cardápio: curto e grosso; curvo e em gancho;
comprido e fino.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:5}
A joaninha é pequena, bonitinha --- e carnívora. O que ela caça?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:6}
Por que a vaca passa muito mais horas comendo do que a raposa? Use a
\cref{prop:g2:what-animals-eat:daily}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:7}
O que o sabiá come no verão e no inverno? Por que um cardápio largo
é útil?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:8}
Faça a experiência: observe as aves num parque ou num comedouro.
Anote o que cada espécie pega --- \omterm{def:g2:from-seed-to-plant:seed}{sementes}, pão, insetos, frutinhas
--- e olhe o bico delas. A \cref{prop:g2:what-animals-eat:match} se
confirma?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:9}
Um zoológico recebe um \omterm{def:g1:animals-around-us:animal}{animal} misterioso: dentes da frente pontudos,
garras afiadas, \omterm{def:g1:the-five-senses:sense}{olhos} voltados para a frente. Qual é o cardápio
provável dele? Que passos do \cref{met:g2:what-animals-eat:guess}
você usou?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:10}
``Sem capim logo não haveria raposas.'' A raposa nunca come capim
--- e mesmo assim a frase está certa. Explique, usando o campo do
\cref{ex:g2:what-animals-eat:meadow}.
\end{exercise}
