\chapter{Les preuves de l'évolution}\label{ch:g9:evidence-of-evolution}

Deux fois, ce livre s'est arrêté au bord d'une grande
affirmation~: les \omterm{def:g1:plants-around-us:tree}{branches} de l'\omterm{def:g6:classification-kinship:tree}{arbre de parenté} se
rejoignent toutes
(\cref{rem:g6:classification-kinship:implies}), et tout \omterm{def:g6:exploring-our-environment:components}{être
vivant} lit un seul code
(\cref{rem:g9:genes-and-dna:universal}). L'affirmation vers
laquelle elles pointent a un nom --- l'\emph{évolution}~:
les \omterm{def:g5:biodiversity:species}{espèces} changent au fil des générations, et toutes
descendent d'ancêtres communs. Avant que le prochain
chapitre ne demande \emph{comment}, celui-ci fait ce que
réclamait \cref{exo:g6:classification-kinship:12}~: il
expose les preuves.

\section{Le témoignage des roches}

\begin{definition}[Les fossiles]\label{def:g9:evidence-of-evolution:fossils}
Un \emph{fossile}\index{fossile} est une trace conservée de
la vie passée --- \omterm{def:g1:animals-around-us:covering}{coquille}, \omterm{def:g3:bones-and-muscles:skeleton}{os}, empreinte de \omterm{def:g1:plants-around-us:parts}{feuille},
empreinte de pas --- scellée dans une roche sédimentaire.
Les couches de roche s'empilent dans l'ordre du temps, la
plus ancienne en bas~; lire vers le haut, c'est lire
l'histoire. Les fossiles sont la moitié «~restes~» de
l'inventaire de
\cref{def:g6:exploring-our-environment:components},
conservée pendant des millions d'années.
\end{definition}

\begin{proposition}[Ce que montre l'archive en couches]\label{prop:g9:evidence-of-evolution:record}
Lue dans l'ordre, l'archive témoigne trois fois~:
\begin{enumerate}
  \item \emph{les faunes changent}~: chaque âge de couche a
        sa propre distribution --- pas de lapins parmi les
        trilobites, pas de trilobites dans les couches à
        lapins~; la plupart des \omterm{def:g5:biodiversity:species}{espèces} de chaque couche
        profonde sont \emph{\omterm{prop:g5:biodiversity:loss}{éteintes}} (le mot de
        \cref{prop:g5:biodiversity:loss}, avec le temps
        profond derrière lui)~;
  \item \emph{l'ordre est emboîté comme l'arbre}~: les
        \omterm{prop:g3:sorting-living-things:groups}{poissons} apparaissent dans les couches profondes,
        les parents des \omterm{prop:g3:sorting-living-things:groups}{amphibiens} à quatre \omterm{def:g1:my-body:parts}{membres}
        au-dessus, les parents des \omterm{prop:g4:classifying-animals:classes}{reptiles} au-dessus
        d'eux, les \omterm{prop:g3:sorting-living-things:groups}{mammifères} et les \omterm{prop:g3:sorting-living-things:groups}{oiseaux} plus haut
        encore --- les boîtes de la \omterm{def:g4:classifying-animals:classification}{classification}
        (\cref{pb:g6:classification-kinship:1}) s'empilent
        dans le temps exactement comme la \omterm{prop:g6:classification-kinship:kinship}{parenté} le
        prédit~;
  \item \emph{des intermédiaires existent}~: la fameuse
        dalle d'\emph{Archaeopteryx} --- des \omterm{def:g1:animals-around-us:covering}{plumes} et des
        ailes sur une charpente de reptile à dents et à
        longue queue osseuse --- se tient là où les
        \omterm{def:g1:plants-around-us:tree}{branches} des \omterm{prop:g3:sorting-living-things:groups}{oiseaux} et des \omterm{prop:g4:classifying-animals:classes}{reptiles} se séparent~;
        des baleines \omterm{def:g9:evidence-of-evolution:fossils}{fossiles} à pattes arrière
        fonctionnelles se tiennent là où l'ascendance de
        mammifère terrestre de
        \cref{ex:g6:classification-kinship:whale} les
        exigeait.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[La série des baleines]\label{ex:g9:evidence-of-evolution:whales}
Le dossier de la baleine, rempli par les roches~: des
\omterm{def:g9:evidence-of-evolution:fossils}{fossiles} successifs, dans l'ordre du temps, vont d'un
mammifère de rivage à quatre pattes de la taille d'un loup,
en passant par des formes aux pattes qui rétrécissent et à
la queue qui se renforce, jusqu'à des nageurs complets sans
pattes --- tandis que la baleine d'aujourd'hui construit
encore de minuscules \omterm{def:g3:bones-and-muscles:skeleton}{os} de hanche et de patte cachés
(\cref{ex:g6:classification-kinship:placing} notait ces
vestiges), allaite encore, respire de l'air et fait pousser
des poils au \omterm{def:g5:human-reproduction:pregnancy}{fœtus}. Prédiction, \omterm{def:g9:evidence-of-evolution:fossils}{fossiles}, vestiges~: trois
clés, une seule serrure.
\end{example}

\section{Le témoignage des corps vivants}

\begin{proposition}[L'anatomie se souvient]\label{prop:g9:evidence-of-evolution:anatomy}
Les corps vivants portent leur histoire~:
\begin{itemize}
  \item \emph{le plan partagé}~: l'aile de la chauve-souris,
        la nageoire de la baleine, la patte du cheval et
        ton \omterm{def:g1:my-body:parts}{bras} sont un seul arrangement d'\omterm{def:g3:bones-and-muscles:skeleton}{os} (la
        fourche à quatre \omterm{def:g1:my-body:parts}{membres} de
        \cref{pb:g6:classification-kinship:1}) étiré,
        raccourci et rechargé d'un autre métier ---
        l'héritage explique ce qu'une ingénierie repartant
        de zéro ne répéterait jamais~;
  \item les \emph{vestiges}~: les hanches de la baleine,
        les ailes soudées du coléoptère incapable de voler,
        ton propre coccyx et l'état rabougri de
        l'appendice --- des pièces gardées parce que
        l'ascendance construit par-dessus, et non à partir
        de rien~;
  \item \emph{les \omterm{def:g5:human-reproduction:pregnancy}{embryons} s'accordent}~: les \omterm{def:g5:human-reproduction:pregnancy}{embryons} de
        \omterm{def:g4:classifying-animals:vertebrate}{vertébrés} du poisson, du poussin et de l'humain se
        déroulent de façon frappante de la même manière
        pendant leurs premières semaines (la pose du plan
        de \cref{prop:g8:from-egg-to-birth:timetable}) ---
        des consignes de construction partagées, qui
        divergent tard.
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[Les molécules votent]\label{prop:g9:evidence-of-evolution:molecules}
Le témoin le plus récent est le code de
\cref{rem:g9:genes-and-dna:universal}. Compare l'orthographe
d'un même \omterm{def:g9:genes-and-dna:gene}{gène} bien conservé d'une \omterm{def:g5:biodiversity:species}{espèce} à l'autre~:
l'humain et le chimpanzé diffèrent de peu de lettres,
l'humain et la souris de davantage, l'humain et le poisson
de plus encore, l'humain et la \omterm{ex:g6:producing-our-food:bread}{levure} le plus --- et les
distances, comptées \omterm{def:g9:genes-and-dna:gene}{gène} après \omterm{def:g9:genes-and-dna:gene}{gène}, redessinent le
\emph{même arbre} que les anatomistes tiraient des \omterm{def:g3:bones-and-muscles:skeleton}{os} et les
paléontologues des couches. Trois témoins indépendants, un
seul arbre généalogique~: c'est cet accord qui a fait cesser
aux biologistes de discuter du \emph{si} pour se tourner
vers le \emph{comment}.
\end{proposition}
\admitted

\begin{example}[La regarder bouger]\label{ex:g9:evidence-of-evolution:live}
Le changement est même observable là où les générations
tournent vite. La forme sombre de la phalène du bouleau
s'est répandue dans les bois industriels noircis de suie et
a reculé quand l'air s'est éclairci~; des \omterm{prop:g3:sorting-living-things:groups}{insectes}
insensibles aux insecticides et des \omterm{def:g6:producing-our-food:fermentation}{microbes} qui défient les
antibiotiques (\cref{ch:g9:microbes-and-infection}
rencontrera le cas médical) apparaissent en quelques
années~; la répartition de la taille des becs chez les
pinsons des îles se déplace avec la sécheresse et
l'abondance, mesurée en direct au pied à coulisse et à la
bague. De petits déplacements, en peu de temps --- le
mécanisme pris en flagrant délit, quelle que soit sa portée
sur la longue durée.
\end{example}

\begin{method}[Peser une preuve historique]\label{met:g9:evidence-of-evolution:weigh}
L'évolution est de l'histoire, et les affirmations
historiques se testent comme l'affaire non élucidée de
\cref{pb:g9:unique-individuals:1} --- par la convergence de
témoins indépendants~:
\begin{enumerate}
  \item demande ce que l'affirmation \emph{prédit}~: quels
        intermédiaires, quels vestiges, quelles distances
        moléculaires~;
  \item vérifie chaque témoin séparément~: les roches, les
        corps, les molécules, les observations en direct~;
  \item pèse leur indépendance~: des témoins qui n'ont pas
        pu se concerter, et qui s'accordent, multiplient la
        crédibilité~;
  \item garde le verdict ouvert à la prochaine
        découverte --- et note quand les découvertes
        continuent de tomber là où l'affirmation disait de
        creuser (les pattes des baleines ont été cherchées,
        et trouvées).
\end{enumerate}
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:1}
Qu'est-ce qu'un \omterm{def:g9:evidence-of-evolution:fossils}{fossile}, et pourquoi les couches de roche
se lisent-elles comme une frise du temps~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:2}
Énonce les trois témoignages de l'archive en couches.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:3}
Quel mélange de \omterm{def:g9:heredity-and-traits:traits}{caractères} fait d'\emph{Archaeopteryx} un
intermédiaire~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:4}
Donne trois vestiges et dis de quoi chacun témoigne.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:5}
Comment les comparaisons moléculaires redessinent-elles
l'arbre~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:6}
Nomme deux observations en direct du changement, tirées de
\cref{ex:g9:evidence-of-evolution:live}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:7}
Pourquoi le plan des \omterm{def:g1:my-body:parts}{membres} partagé plaide-t-il pour
l'héritage plutôt que pour une bonne conception
réemployée~? (Que serait libre de faire une ingénierie
repartant de zéro~?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:8}
Assemble le dossier complet de la baleine~: les \omterm{def:g9:heredity-and-traits:traits}{caractères}
de sa classe, les vestiges, la série \omterm{def:g9:evidence-of-evolution:fossils}{fossile}, la \omterm{prop:g6:classification-kinship:kinship}{parenté}
moléculaire --- une ligne pour chacun.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:9}
Pourquoi l'\emph{accord} des roches, des corps et des
molécules compte-t-il plus que n'importe quel témoin pris
seul~? Sers-toi de l'étape de l'indépendance de
\cref{met:g9:evidence-of-evolution:weigh}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:10}
«~Pas de lapins parmi les trilobites.~» Pourquoi cette
absence --- maintenue dans toutes les fouilles jamais
faites --- est-elle à elle seule une preuve forte~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:11}
Le témoin embryonnaire~: qu'est-ce qui s'accorde, quand
cela diverge-t-il, et la machinerie de quel chapitre dit
pourquoi les consignes partagées du début comptent le plus~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:evidence-of-evolution:12}
Écris l'argument du chapitre comme
\cref{met:g9:evidence-of-evolution:weigh} appliquée en
entier~: l'affirmation, les prédictions, les quatre témoins,
le verdict --- et la question que ce verdict passe à
\cref{ch:g9:how-species-change}.
\end{exercise}

\section{Problème~: la nouvelle aile du musée}

\begin{problem}[{Problème du week-end --- concevoir la
salle des preuves}]\label{pb:g9:evidence-of-evolution:1}
Le muséum d'histoire naturelle construit une aile
consacrée à l'évolution, et la classe en conçoit quatre
salles~: Roches, Corps, Molécules, Observations en direct.

\medskip\noindent\textbf{Partie I --- Salle par salle.}
\begin{enumerate}
  \item Roches~: ordonne le mur --- dalle à trilobites,
        premiers \omterm{prop:g3:sorting-living-things:groups}{poissons}, traces de parents des
        \omterm{prop:g3:sorting-living-things:groups}{amphibiens}, \omterm{def:g3:bones-and-muscles:skeleton}{squelettes} de \omterm{prop:g4:classifying-animals:classes}{reptiles}, moulage
        d'\emph{Archaeopteryx}, mâchoire de mammifère
        ancien --- et écris le cartel d'une phrase de la
        salle à partir de
        \cref{prop:g9:evidence-of-evolution:record}.
  \item Corps~: la pièce maîtresse déploie une aile de
        chauve-souris, une nageoire de baleine, une patte
        de cheval et un \omterm{def:g1:my-body:parts}{bras} humain sur un même support de
        \omterm{def:g3:bones-and-muscles:skeleton}{squelette}. Rédige le cartel du plan partagé --- et
        ajoute les trois pièces de la vitrine des vestiges.
  \item Molécules~: le tableau mural compte les différences
        de lettres d'un même \omterm{def:g9:genes-and-dna:gene}{gène}~: chimpanzé 1, souris 8,
        poisson 24, \omterm{ex:g6:producing-our-food:bread}{levure} 41 (chiffres illustratifs). Avec
        quoi l'arbre du tableau doit-il concorder pour que
        la salle fonctionne~?
  \item Observations en direct~: choisis deux pièces dans
        \cref{ex:g9:evidence-of-evolution:live} et dis ce
        que chacune montre en train de bouger.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les questions des
visiteurs.}
\begin{enumerate}[resume]
  \item «~Pourquoi y a-t-il encore des singes, si nous
        descendons d'ancêtres partagés avec eux~?~»
        Réponds par la ramification de l'arbre --- des
        cousins, non des échelles
        (\cref{def:g6:classification-kinship:tree}).
  \item «~Où sont les chaînons manquants~?~» Montre ceux
        que l'on a trouvés --- deux dans ce chapitre ---
        et dis entre quoi et quoi chacun se tenait.
  \item «~L'évolution n'est-elle pas qu'une théorie~?~»
        Réponds par la méthode des preuves historiques~: ce
        que «~théorie~» veut dire ici, et ce qu'établit
        l'accord des quatre salles.
  \item «~La fouille de demain pourrait-elle la renverser~?~»
        Dis quel genre de découverte le ferait --- des
        lapins parmi les trilobites --- et ce qu'a
        signifié son absence, fouille après fouille.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le panneau de
clôture de l'aile.}
\begin{enumerate}[resume]
  \item Rédige le panneau qui relie cette aile à la salle
        de \omterm{def:g4:classifying-animals:classification}{classification} d'à côté~: pourquoi les vieilles
        boîtes emboîtées étaient l'arbre depuis toujours.
  \item Ajoute le paragraphe des molécules~: ce que le code
        partagé dit du tronc.
  \item Ajoute le paragraphe honnête~: ce que l'aile montre
        (le changement, la \omterm{prop:g6:classification-kinship:kinship}{parenté}, les intermédiaires) et
        ce qu'elle n'explique pas encore --- en nommant la
        question du mécanisme que la salle suivante
        reprendra.
  \item Termine par une phrase dont un enfant de douze ans
        se souviendra, contenant les mots \emph{famille}
        et \emph{temps}.
\end{enumerate}
\end{problem}
