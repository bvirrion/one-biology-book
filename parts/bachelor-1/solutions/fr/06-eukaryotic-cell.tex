\section*{Chapitre \ref{ch:b1:eukaryotic-cell} --- Organisation fonctionnelle de la cellule eucaryote}

\begin{solution}{exo:b1:eukaryotic-cell:1}
\omterm{def:b1:eukaryotic-cell:nucleus}{Noyau} (chromosomes, transcription)~; RE granuleux (protéines de
sécrétion et membranaires)~; RE \omterm{def:b1:eukaryotic-cell:endomembrane}{lisse} (lipides, calcium,
détoxication)~; \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} (tri, modification, emballage)~; \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes}
(digestion)~; \omterm{def:b1:eukaryotic-cell:peroxisome}{peroxysomes} (oxydations, catalase)~; \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondries}
(respiration, ATP).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:2}
Synthétisée par des ribosomes liés au RE granuleux et engagée dans sa
lumière (\omterm{prop:b1:eukaryotic-cell:secretory}{séquence signal})~; des vésicules bourgeonnent du RE et
fusionnent avec le \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} cis~; la protéine traverse l'empilement, y est
modifiée et triée~; elle quitte la face trans dans une \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole} de
condensation qui mûrit en granule de sécrétion~; le granule fusionne
avec la membrane apicale sur un signal et se vide dans le canal.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:3}
$7 + 32 = \qty{39}{\%}$. La membrane interne est repliée en \omterm{def:b1:eukaryotic-cell:mitochondrion}{crêtes}
parce qu'elle porte la chaîne respiratoire et l'ATP synthase, dont le
débit est proportionnel à sa surface~; l'externe n'est qu'une enveloppe
simple.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:4}
Végétale~: \omterm{def:b1:eukaryotic-cell:plantcell}{paroi} de cellulose, grande \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole}, \omterm{def:b1:eukaryotic-cell:plastid}{plastes} (et
\omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesmes}). Animale~: \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes}, centrioles (et la capacité de ramper
et de phagocyter).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:5}
RE à \qty{3}{min} (déjà en baisse), \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} vers \qty{15}{min}, granules
vers \qty{60}{min}. Transit du RE au granule environ \qty{45}{min}~;
moitié sécrétée vers \qty{110}{min}.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:6}
Le \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}, maintenu près du \omterm{def:b1:eukaryotic-cell:cytoskeleton}{centrosome} par la dynéine sur les
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtubules}, se fragmente et se disperse~; le trafic vésiculaire se
réduit à la diffusion et le transport vers la périphérie s'arrête~; le
fuseau ne peut pas se former et les \omterm{def:b1:cell-unit-of-life:cell}{cellules} se bloquent en mitose~; les
cils, dont l'axe est fait de \omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtubules}, s'arrêtent (les cils déjà
formés sont stables, mais ils ne sont plus renouvelés).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:7}
Deux membranes dont l'interne est de type bactérien~; ADN circulaire
sans histones~; ribosomes de taille bactérienne, sensibles aux
antibiotiques~; division par fission~; séquences géniques se groupant
avec les alphaprotéobactéries. Elle ne peut pas vivre seule parce que
la plupart de ses gènes ont migré vers le \omterm{def:b1:eukaryotic-cell:nucleus}{noyau}~: elle importe
l'essentiel de ses protéines depuis le \omterm{def:b1:eukaryotic-cell:organelle}{cytosol}.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:8}
Les sucres sont ajoutés du côté luminal dans le RE et le \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}. Chaque
fusion de vésicule conserve l'orientation~: la face luminale d'une
vésicule devient la face luminale du compartiment suivant, et quand la
vésicule fusionne avec la membrane plasmique sa lumière s'ouvre vers
l'extérieur, de sorte que les sucres sont tournés vers le dehors. Ce
qui est luminal est, topologiquement, à l'extérieur.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:9}
Chaque granule $\pi d^2 = \qty{3.14}{\micro m^2}$~; par jour
\qty{6280}{\micro m^2}, soit \qty{157}{} fois la face apicale. La
\omterm{def:b1:cell-unit-of-life:cell}{cellule} doit récupérer de la membrane par \omterm{def:b1:eukaryotic-cell:endocytosis}{endocytose} au même rythme, de
sorte que la membrane apicale se renouvelle toutes les neuf minutes et
que la membrane récupérée est rendue au \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} pour être réutilisée.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:10}
La chaîne de \qty{320}{} résidus est le précurseur avec sa \omterm{prop:b1:eukaryotic-cell:secretory}{séquence
signal} de \qty{20}{} résidus, normalement clivée à l'entrée dans le RE.
En présence de vésicules, la chaîne entre dans la lumière pendant sa
synthèse, le signal est clivé, et la membrane protège le produit de la
protéase. Le mutant dépourvu du signal n'est pas reconnu par la
machinerie du RE~: il est fabriqué sur des ribosomes libres et reste
dans le \omterm{def:b1:eukaryotic-cell:organelle}{cytosol}, de pleine longueur, jamais sécrété.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:11}
Les membranes internalisées par \omterm{def:b1:eukaryotic-cell:endocytosis}{endocytose} et les \omterm{def:b1:cell-unit-of-life:prokeuk}{organites} usés sont
livrés aux \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes} pour y être digérés~; sans l'enzyme, le lipide
qu'ils contiennent ne peut pas être dégradé et remplit les \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes},
qui gonflent jusqu'à ce que la \omterm{def:b1:cell-unit-of-life:cell}{cellule} cède. Les \omterm{def:b1:cell-unit-of-life:cell}{cellules} dont le
renouvellement membranaire est le plus rapide (\omterm{def:b1:body-plans-tissues:nervous}{neurones}, qui recyclent
continûment leurs vésicules synaptiques~; macrophages) l'accumulent le
plus vite. Une enzyme injectée portant le signal glucidique que portent
les enzymes lysosomales peut se lier à des récepteurs de la surface
cellulaire et être internalisée par \omterm{def:b1:eukaryotic-cell:endocytosis}{endocytose} vers les \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes} ---
c'est le principe de l'enzymothérapie substitutive.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:12}
L'endosymbiose rend bien compte des \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondries} et des \omterm{def:b1:eukaryotic-cell:plastid}{plastes} (deux
membranes, génome et ribosomes propres, fission, parents bactériens), et
donc du métabolisme énergétique de l'eucaryote. Elle n'explique ni le
\omterm{def:b1:eukaryotic-cell:nucleus}{noyau}, ni le RE et le \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}, ni le \omterm{def:b1:eukaryotic-cell:cytoskeleton}{cytosquelette}, qui n'ont pas
d'équivalents bactériens et paraissent être des inventions de la lignée
hôte --- des invaginations de sa membrane et des protéines nouvelles.
«~Fédération~» convient pour les \omterm{def:b1:cell-unit-of-life:prokeuk}{organites}~; «~tenue par un système de
membranes~» désigne la part la plus grande et inexpliquée~: une \omterm{def:b1:cell-unit-of-life:cell}{cellule}
hôte déjà eucaryote par son architecture avant qu'elle n'ait accueilli
les bactéries.
\end{solution}

\begin{solution}{pb:b1:eukaryotic-cell:1}
\textbf{1.} Un marquage bref étiquette une cohorte étroite de molécules,
de sorte que leur position marque un seul instant~; l'excès d'acide
aminé froid arrête toute incorporation ultérieure, de sorte que la
cohorte n'est pas étalée par des synthèses plus tardives.
\textbf{2.} RE~: \qty{3}{min} (point le plus précoce)~; \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} et
\omterm{def:b1:eukaryotic-cell:plantcell}{vacuoles} de condensation~: vers \qty{15}{min}~; granules~: vers
\qty{60}{min}.
\textbf{3.} Du RE au \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} environ \qty{10}{min}~; du \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} au granule
environ \qty{45}{min}.
\textbf{4.} Surtout dans les granules (\qty{70}{\%}), un dixième dans le
\omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}, un dixième déjà sécrété~: la voie constitutive et les premiers
granules formés en libèrent une part, tandis que l'essentiel attend un
stimulus.
\textbf{5.} Le bourgeonnement et le transport des vésicules exigent de
l'énergie et des membranes fluides~; au froid, l'étape \omterm{def:b1:eukaryotic-cell:endomembrane}{RE-Golgi}
s'arrête et la protéine s'accumule en amont du blocage.
\textbf{6.} $10\,\mathrm{g}/\num{8e10} = \qty{1.25e-10}{g} =
\qty{125}{pg}$ par \omterm{def:b1:cell-unit-of-life:cell}{cellule} et par jour.
\textbf{7.} $V = \frac{\pi}{6}(1)^3 = \qty{0.52}{\micro m^3} =
\qty{5.2e-13}{mL}$~; protéines $\qty{1.05e-13}{g} = \qty{0.1}{pg}$.
\textbf{8.} $125/0.1 = 1250$ granules par jour, $0.87$ par minute.
\textbf{9.} $V = 12^3 = \qty{1728}{\micro m^3} = \qty{1.73e-9}{cm^3}$~;
masse \qty{1.8e-9}{g} = \qty{1800}{pg}. Elle sécrète $125/1800 = 7\%$ de
sa masse par jour (la \omterm{def:b1:cell-unit-of-life:cell}{cellule} entière n'est qu'à \qty{20}{\%} de
protéines, soit le tiers de son contenu protéique).
\textbf{10.} Masse molaire $250\times 110 = \qty{27500}{g/mol}$~;
$\qty{1.05e-13}{g}/27500\times\num{6e23} = \num{2.3e6}$ molécules.
\textbf{11.} $250/5 = \qty{50}{s}$.
\textbf{12.} Molécules par jour $1250\times\num{2.3e6} = \num{2.9e9}$~;
un ribosome en fait $86\,400/50 = 1728$ par jour~; il en faut
$\num{1.7e6}$, soit environ \qty{40}{\%} de ceux de la \omterm{def:b1:cell-unit-of-life:cell}{cellule}.
\textbf{13.} $\pi d^2 = \qty{3.14}{\micro m^2}$.
\textbf{14.} $1250\times 3.14 = \qty{3900}{\micro m^2}$.
\textbf{15.} $3900/144 = 27$ fois~: la \omterm{def:b1:cell-unit-of-life:cell}{cellule} récupère la même surface
par \omterm{def:b1:eukaryotic-cell:endocytosis}{endocytose} et la rend au \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi}.
\textbf{16.} $20\,000/3900 = 5$ jours~: le RE ne pourrait fournir que
cinq jours de membrane de granules si rien ne revenait~; la membrane
doit être recyclée continûment.
\textbf{17.} Chaque microvillosité $\pi\times 0.1\times 1 =
\qty{0.31}{\micro m^2}$~; \qty{630}{\micro m^2} pour 2000~; facteur
$5.4$.
\textbf{18.} En tête de la chaîne~: la \omterm{prop:b1:eukaryotic-cell:secretory}{séquence signal}, vingt résidus
pour la plupart hydrophobes.
\textbf{19.} Le signal a dirigé la chaîne naissante dans les vésicules,
où une peptidase du signal l'a ôté~; la membrane met la protéine à
l'abri de la protéase.
\textbf{20.} Seules les protéines munies d'un signal entrent dans le
RE~; une protéine cytosolique n'en a pas et reste dehors.
\textbf{21.} L'enzyme est fabriquée sur des ribosomes libres et reste
dans le \omterm{def:b1:eukaryotic-cell:organelle}{cytosol}~; étant une enzyme digestive (sous forme de précurseur
inactif elle fait peu de dégâts, mais si elle est activée) elle
digérerait la \omterm{def:b1:cell-unit-of-life:cell}{cellule} de l'intérieur.
\textbf{22.} Dans le \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi} trans~: l'enzyme lysosomale porte une marque
distinctive (un sucre phosphorylé) reconnue par un récepteur qui
l'emballe dans des vésicules à destination des \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomes} plutôt que dans
des granules de sécrétion.
\textbf{23.} Dans le RE (et le \omterm{def:b1:eukaryotic-cell:endomembrane}{Golgi})~: la formation des vésicules, le
transport par moteurs le long des \omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtubules} et la fusion des
membranes consomment tous de l'ATP~; et la synthèse protéique elle-même
s'arrête, de sorte que le marquage n'est même pas mené à son terme.
\textbf{24.} \qty{45}{min} de transit à $0.87$ granule par minute~:
l'équivalent d'environ \qty{40}{} granules de protéines en transit.
\textbf{25.} Transit de la synthèse au granule environ \qty{45}{min}
(\qty{55}{min} en comptant le RE), sécrétion après environ deux heures
dans l'expérience (quatre heures ou plus dans la glande vivante,
puisque les granules attendent un repas)~; \qty{0.9}{} granule libéré
par minute~; \qty{7}{\%} de la masse de la \omterm{def:b1:cell-unit-of-life:cell}{cellule} par jour.
\end{solution}
