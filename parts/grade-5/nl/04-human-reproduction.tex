\chapter{Voortplanting bij de mens}\label{ch:g5:human-reproduction}

Iedereen die je kent --- vrienden, leraren, de oudste van je familie ---
is op dezelfde manier aangekomen: als \omterm{def:g3:stages-of-life:stages}{baby}, \omterm{def:g2:born-grow-die:cycle}{geboren} na maanden groeien
in de buik van zijn moeder. Mensen zijn \omterm{prop:g3:sorting-living-things:groups}{zoogdieren}
(\cref{prop:g4:classifying-animals:classes}), en de voortplanting bij de
mens verloopt volgens het zoogdierplan van
\cref{ch:g4:animal-reproduction} --- met de getallen, data en
bijzonderheden van onze eigen \omterm{def:g5:biodiversity:species}{soort}, die dit hoofdstuk helder op een rij zet.

\section{De menselijke versie van het plan}

\begin{proposition}[Voortplanting bij de mens is voortplanting van een zoogdier]\label{prop:g5:human-reproduction:plan}
Zoals alle \omterm{prop:g3:sorting-living-things:groups}{zoogdieren} \omterm{def:g1:plants-around-us:plant}{planten} mensen zich geslachtelijk voort
(\cref{def:g4:animal-reproduction:sexual}):
\begin{itemize}
  \item het lichaam van de moeder maakt \emph{\omterm{def:g4:animal-reproduction:sexual}{eicellen}}; dat van de
        vader \emph{\omterm{def:g4:animal-reproduction:sexual}{zaadcellen}};
  \item de \omterm{def:g5:flower-to-fruit:steps}{bevruchting} --- één \omterm{def:g4:animal-reproduction:sexual}{zaadcel} die zich met één \omterm{def:g4:animal-reproduction:sexual}{eicel} verenigt
        --- gebeurt \emph{in het lichaam van de moeder}
        (\cref{prop:g4:animal-reproduction:where});
  \item de bevruchte \omterm{def:g4:animal-reproduction:sexual}{eicel} --- één enkele \omterm{def:g5:first-look-at-cells:cell}{cel}
        (\cref{ex:g5:first-look-at-cells:one}) --- nestelt zich in de
        \emph{baarmoeder}\index{baarmoeder} van de moeder en groeit daar;
  \item mensen zijn \omterm{def:g4:animal-reproduction:ovivivi}{levendbarend}
        (\cref{def:g4:animal-reproduction:ovivivi}): de \omterm{def:g3:stages-of-life:stages}{baby} wordt
        \omterm{def:g2:animals-grow-up:born}{levend geboren} en krijgt eerst \omterm{def:g2:animals-grow-up:born}{melk}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Voortplanting vraagt volwassen lichamen]\label{rem:g5:human-reproduction:adults}
Het lichaam van een kind kan geen \omterm{def:g3:stages-of-life:stages}{baby}'s maken: de
voortplantingsorganen gaan pas werken in de \emph{puberteit}, de
verandering van het pubertijdstadium uit
\cref{def:g3:stages-of-life:stages}, waarin het lichaam zich van dat van
een kind ombouwt tot dat van een volwassene. Wat de puberteit verandert,
en hoe, is het onderwerp van een heel hoofdstuk in de onderbouwjaren van
dit boek.
\end{remark}

\section{Negen maanden in de baarmoeder}

\begin{definition}[Zwangerschap]\label{def:g5:human-reproduction:pregnancy}
\emph{Zwangerschap}\index{zwangerschap} is de tijd die de toekomstige
\omterm{def:g3:stages-of-life:stages}{baby} groeiend in de \omterm{prop:g5:human-reproduction:plan}{baarmoeder} van zijn moeder doorbrengt: bij mensen
ongeveer \emph{negen maanden}. De groeiende \omterm{def:g3:stages-of-life:stages}{baby} heet eerst een
\emph{embryo}\index{embryo}, en daarna --- zodra zijn bouwplan is
uitgezet, na ongeveer twee maanden --- een \emph{foetus}\index{foetus}.
\end{definition}

\begin{proposition}[Gevoed door de navelstreng]\label{prop:g5:human-reproduction:cord}
In de \omterm{prop:g5:human-reproduction:plan}{baarmoeder} drijft de \omterm{def:g3:stages-of-life:stages}{baby} in een zak vol vloeistof en kan hij niet
eten of ademen. Alles komt kant-en-klaar uit het \omterm{prop:g4:journey-of-food:absorb}{bloed} van de moeder:
\omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} en zuurstof
(\cref{prop:g4:heart-and-blood:carries}) steken over naar het \omterm{prop:g4:journey-of-food:absorb}{bloed} van
de \omterm{def:g3:stages-of-life:stages}{baby} via de \emph{navelstreng}\index{navelstreng} aan zijn buik ---
en zijn afvalstoffen steken langs dezelfde weg terug, zodat het lichaam
van de moeder ze afhandelt. Jouw navel is de plek waar jouw eigen streng
vastzat.
\end{proposition}
\admitted

\begin{example}[Het tijdschema van negen maanden]\label{ex:g5:human-reproduction:timetable}
Maand 1: een klompje delende cellen, kleiner dan een rijstkorrel, maar
al snel groeiend. Maand 2: een \omterm{def:g5:human-reproduction:pregnancy}{embryo} met de schets van een \omterm{def:g1:my-body:parts}{hoofd}, een
kloppend \omterm{def:g4:heart-and-blood:heart}{hart}, knopjes voor \omterm{def:g1:my-body:parts}{armen} en \omterm{def:g1:my-body:parts}{benen} --- nog steeds kleiner dan een
duim. Maand 3--4: een \omterm{def:g5:human-reproduction:pregnancy}{foetus}, compleet van plan, die begint te bewegen;
de moeder voelt al snel de schopjes. Maand 5--8: groeien en rijpen ---
het gehoor komt erbij, en een eerste laagje haar. Maand 9: de \omterm{def:g3:stages-of-life:stages}{baby},
meestal met het \omterm{def:g1:my-body:parts}{hoofd} omlaag en zo'n \qty{3}{kg}, is er klaar voor.
\end{example}

\begin{example}[De geboorte]\label{ex:g5:human-reproduction:birth}
Bij de \omterm{def:g2:born-grow-die:cycle}{geboorte} duwen krachtige \omterm{def:g3:bones-and-muscles:muscle}{spieren} van de \omterm{prop:g5:human-reproduction:plan}{baarmoeder} de \omterm{def:g3:stages-of-life:stages}{baby} de
wereld in, meestal met het \omterm{def:g1:my-body:parts}{hoofd} eerst. De eerste kreet vult de \omterm{def:g4:breathing:organs}{longen}
met lucht --- ze hadden nog nooit geademd
(\cref{prop:g5:human-reproduction:cord}) --- de streng, die niet meer
nodig is, wordt pijnloos doorgeknipt, en het zoogdierplan gaat buiten
verder: \omterm{def:g2:animals-grow-up:born}{melk}, warmte en zorg
(\cref{ex:g4:animal-reproduction:whalecalf} zou het goedkeuren).
\end{example}

\begin{remark}[Tweelingen]\label{rem:g5:human-reproduction:twins}
Soms groeien er twee \omterm{def:g3:stages-of-life:stages}{baby}'s samen in de \omterm{prop:g5:human-reproduction:plan}{baarmoeder}. Als één bevruchte
\omterm{def:g4:animal-reproduction:sexual}{eicel} zich in twee \omterm{def:g5:human-reproduction:pregnancy}{embryo}'s heeft gesplitst, is de tweeling
\emph{eeneiig} --- gebouwd uit dezelfde begincel, gelijk als twee
kopieën. Als er twee \omterm{def:g4:animal-reproduction:sexual}{eicellen} door twee \omterm{def:g4:animal-reproduction:sexual}{zaadcellen} bevrucht zijn, lijkt
de tweeling niet meer op elkaar dan gewone broers en zussen --- ze
deelden alleen de \omterm{prop:g5:human-reproduction:plan}{baarmoeder}. Twee \omterm{def:g5:biodiversity:species}{soorten} tweelingen, uit elkaar te
houden aan hun begin.
\end{remark}

\section{Zorg ervoor en erna}

\begin{proposition}[De baby deelt in de voorraden van zijn moeder]\label{prop:g5:human-reproduction:care}
Alles wat de \omterm{def:g3:stages-of-life:stages}{baby} bereikt, reist door het \omterm{prop:g4:journey-of-food:absorb}{bloed} van de moeder --- en
daarom eet een zwangere vrouw goed (\cref{ch:g3:balanced-diet}), rust ze,
wordt ze door artsen of verloskundigen gevolgd, en mijdt ze wat de \omterm{def:g3:stages-of-life:stages}{baby}
zou kunnen schaden: tabaksrook en alcohol gaan ook door de streng, en
een \omterm{def:g5:human-reproduction:pregnancy}{foetus} is veel kwetsbaarder dan een volwassene. Voor de moeder
zorgen \emph{is} voor de \omterm{def:g3:stages-of-life:stages}{baby} zorgen.
\end{proposition}
\admitted

\begin{example}[Een uiterst goed verzorgde soort]\label{ex:g5:human-reproduction:longcare}
Zelfs onder de \omterm{prop:g3:sorting-living-things:groups}{zoogdieren} drijven mensen de strategie van
weinig-en-goed-verzorgd uit
\cref{prop:g4:animal-reproduction:strategies} tot het uiterste: meestal
één \omterm{def:g3:stages-of-life:stages}{baby} tegelijk, veel langer hulpeloos dan een veulen of een kitten,
en verzorgd --- gevoed, onderwezen, bemind --- gedurende vele
\emph{jaren}. De menselijke kindertijd is een van de langste in de
dierenwereld; dit boek, een cursus van tien jaar, is er zelf een stukje
van die lange zorg.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:1}
Welke \omterm{def:g5:first-look-at-cells:cell}{cel} komt van de moeder, welke van de vader, en hoe heet hun
samengaan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:2}
Waar groeit de menselijke \omterm{def:g3:stages-of-life:stages}{baby}, en ongeveer hoe lang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:3}
Wat zijn de twee namen van de groeiende \omterm{def:g3:stages-of-life:stages}{baby}, op volgorde, en wanneer
wordt de eerste ongeveer de tweede?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:4}
De \omterm{def:g3:stages-of-life:stages}{baby} in de \omterm{prop:g5:human-reproduction:plan}{baarmoeder} eet niet en ademt niet. Hoe komt hij aan
\omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} en zuurstof, en waar gaan zijn afvalstoffen heen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:5}
Waarvan is je navel het teken?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:6}
Wat gebeurt er bij de eerste kreet, en waarom is dat zo'n verandering
voor het lichaam van de \omterm{def:g3:stages-of-life:stages}{baby}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:7}
Welke kenmerken maken de voortplanting bij de mens typisch die van een
zoogdier? Noem er drie, met een verwijzing naar
\cref{prop:g4:classifying-animals:classes} of
\cref{ch:g4:animal-reproduction}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:8}
Waarom vragen artsen zwangere vrouwen tabaksrook en alcohol te mijden?
Gebruik \cref{prop:g5:human-reproduction:care}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:9}
Verklaar de twee \omterm{def:g5:biodiversity:species}{soorten} tweelingen aan de hand van hun begin. Welke
\omterm{def:g5:biodiversity:species}{soort} lijkt op elkaar ``als twee kopieën'', en waarom volgt dat uit
\cref{ex:g5:first-look-at-cells:one}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:10}
Plaats de mens op de strategielijn van
\cref{prop:g4:animal-reproduction:strategies}, in vergelijking met de
kabeljauw en de olifant: aantal jongen, duur van de zorg, en wat die
lange zorg bij mensen inhoudt.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:human-reproduction:11}
Bij de \omterm{def:g2:born-grow-die:cycle}{geboorte} beëindigt de doorgeknipte streng de bevoorradingslijn
van de \omterm{def:g3:stages-of-life:stages}{baby} op precies het moment dat de \omterm{def:g4:breathing:organs}{longen} hun eerste teug nemen.
Leg met \cref{prop:g5:human-reproduction:cord} en
\cref{prop:g4:breathing:exchange} uit welke taak er in die minuut van
het lichaam van de moeder naar de eigen \omterm{def:g4:breathing:organs}{longen} van de \omterm{def:g3:stages-of-life:stages}{baby} overgaat ---
en waarom de eerste kreet zo goed getimed is.
\end{exercise}
