\chapter{Food Chains}\label{ch:g3:food-chains}

Who eats whom? \omterm{def:g2:food-energy:energy}{Energy} moves from the Sun to \omterm{def:g1:plants-animals:plant}{plants}, then to \omterm{def:g1:plants-animals:animal}{animals}
that eat \omterm{def:g1:plants-animals:plant}{plants}, then to \omterm{def:g1:plants-animals:animal}{animals} that eat \omterm{def:g1:plants-animals:animal}{animals}. A simple line of
``eaten by'' is a \omterm{def:g3:food-chains:chain}{food chain}.

\section{Links in the chain}

\begin{definition}[Food chain]\label{def:g3:food-chains:chain}
A \emph{food chain}\index{food chain} shows how \omterm{def:g2:food-energy:energy}{energy} and \omterm{def:g2:food-energy:food}{food} pass
from one \omterm{def:g2:what-alive:organism}{organism} to the next. Arrows mean ``is eaten by'' or ``gives
\omterm{def:g2:food-energy:energy}{energy} to''.
\end{definition}

\omimg{food-chain.jpg}{0.52\linewidth}{\omterm{def:g1:plants-animals:plant}{Plant} $\to$ plant-eater $\to$
meat-eater --- \omterm{def:g2:food-energy:energy}{energy} steps along the chain.}

\begin{definition}[Producers and consumers]\label{def:g3:food-chains:roles}
\begin{itemize}
  \item A \emph{producer}\index{producer} makes \omterm{def:g2:food-energy:food}{food} from light ---
        usually a green \omterm{def:g1:plants-animals:plant}{plant}.
  \item A \emph{consumer}\index{consumer} eats other \omterm{def:g2:what-alive:organism}{organisms}.
  \item A \emph{herbivore}\index{herbivore} eats \omterm{def:g1:plants-animals:plant}{plants}.
  \item A \emph{carnivore}\index{carnivore} eats \omterm{def:g1:plants-animals:animal}{animals}.
  \item An \emph{omnivore}\index{omnivore} eats both.
\end{itemize}
\end{definition}

\begin{example}[Garden chain]\label{ex:g3:food-chains:garden}
Lettuce $\to$ \omterm{def:g3:insects:compare}{snail} $\to$ \omterm{def:g2:animal-groups:five}{bird}. Sunlight fed the lettuce; the \omterm{def:g3:insects:compare}{snail}
ate \omterm{def:g2:plant-parts:main}{leaves}; the \omterm{def:g2:animal-groups:five}{bird} ate the \omterm{def:g3:insects:compare}{snail}.
\end{example}

\begin{example}[Pond chain]\label{ex:g3:food-chains:pond}
\omterm{def:g1:caring:needs}{Water} \omterm{def:g1:plants-animals:plant}{plant} $\to$ tiny \omterm{def:g1:plants-animals:animal}{animal} $\to$ \omterm{def:g2:animal-groups:five}{fish} $\to$ heron.
\end{example}

\section{Reading and building chains}

\begin{method}[Build a food chain]\label{met:g3:food-chains:build}
\begin{enumerate}
  \item Start with a \omterm{def:g1:plants-animals:plant}{plant} (\omterm{def:g3:food-chains:roles}{producer}).
  \item Add a plant-eater.
  \item Add an \omterm{def:g1:plants-animals:animal}{animal} that eats that plant-eater.
  \item Draw arrows from \omterm{def:g2:food-energy:food}{food} toward the eater.
\end{enumerate}
\end{method}

\begin{proposition}[Sun first]\label{prop:g3:food-chains:sun}
Almost all \omterm{def:g3:food-chains:chain}{food chains} begin with \omterm{def:g2:food-energy:energy}{energy} from the Sun captured by
\omterm{def:g3:food-chains:roles}{producers}.
\end{proposition}
\admitted
