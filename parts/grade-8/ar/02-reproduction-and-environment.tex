\chapter{التكاثر والبيئة}\label{ch:g8:reproduction-and-environment}

لماذا تبدأ جوقة الضفادع في أيار لا في تشرين الثاني؟ ولماذا تملأ
سنة زان جيدة الحرش بالبادرات، ويُفرغ حزيران بارد ممطر بركة الربيع
التالي من الضفيدعات؟ لقد بنى الفصل السابق آلية الجيل التالي؛ وهذا
الفصل يوصلها بالعالم: فالتكاثر موقوت، ومموَّن --- ومحدود ---
\omterm{def:g6:exploring-our-environment:components}{بالبيئة}.

\section{موقوت بفصول السنة}

\begin{proposition}[التوالد يتبع تقويم الوفرة]\label{prop:g8:reproduction-and-environment:timing}
توقّت \omterm{def:g5:biodiversity:species}{الأنواع} تكاثرها ليصل \emph{الصغار} حين تخدمهم \omterm{def:g6:exploring-our-environment:conditions}{الشروط} والغذاء
(\cref{prop:g6:life-through-seasons:conditions}):
\begin{itemize}
  \item فالقرقف الأزرق يبيض ليفقس فراخه عند ذروة \omterm{ex:g2:animals-grow-up:butterfly}{يرقات} الربيع
        (وهي سلسلة بلوط \cref{ex:g3:food-chains:three}، موقوتةً
        إلى اليوم)؛
  \item والضفدع يبيض في البركة وهي تدفأ، فيمنح \omterm{ex:g2:animals-grow-up:frog}{الشراغيف} صيفًا من
        النمو قبل الشتاء؛
  \item \omterm{def:g1:plants-around-us:plant}{والنباتات} تزهر على موسم مؤبّريها وتُنضج \omterm{def:g2:from-seed-to-plant:seed}{البذور} على تقويم
        \omterm{def:g6:how-plants-spread:dispersal}{الانتثار} (\cref{pb:g6:life-through-seasons:1})؛
  \item وخوار الأيّل في الخريف يضع الولادات في \emph{ربيع} السنة
        التالية --- ومدة \omterm{def:g5:human-reproduction:pregnancy}{الحمل} محسوبة.
\end{itemize}
والإشارات التي تشغّل الآلة هي إشارات \omterm{def:g6:exploring-our-environment:components}{البيئة} نفسها: طول النهار قبل
كل شيء، والدفء، والمطر.
\end{proposition}
\admitted

\begin{example}[حملان في الداخل]\label{ex:g8:reproduction-and-environment:lambs}
مربّو الأغنام يستغلون الإشارة مباشرة: فالنعاج تدخل في حال التوالد
مع \emph{قِصَر} الأيام؛ وإذا حُفظ القطيع تحت ضوء اصطناعي يطول ثم
يقصر، أمكن أن يُحمل على \omterm{def:g2:born-grow-die:cycle}{الولادة} في السوق المبكرة المربحة. فالآلة
تقرأ \omterm{def:g6:exploring-our-environment:conditions}{الضوء} لا التقاويم --- فغيّر \omterm{def:g6:exploring-our-environment:conditions}{الضوء} يتبعك التقويم. والإشارة إذا
عُرفت صارت رافعة.
\end{example}

\section{مموَّن بالموئل}

\begin{proposition}[التكاثر يُدفع ثمنه مادةً]\label{prop:g8:reproduction-and-environment:cost}
\omterm{def:g2:animals-grow-up:born}{البيض} \omterm{def:g2:animals-grow-up:born}{والحليب} والرحيق \omterm{def:g2:from-seed-to-plant:seed}{والبذور} والقرون كلها \emph{إنتاج} بمعنى
\cref{def:g6:matter-from-food:production} --- أي \omterm{def:g6:plants-are-producers:matter}{مادة عضوية} على
الوالدين أن يكتسباها أولًا. ومن ثم تضع غنى \omterm{def:g2:where-animals-live:habitat}{الموئل} الميزانية:
\begin{itemize}
  \item فالدجاجة حسنة التغذية تبيض كل يوم؛ والجائعة تتوقف ---
        ودفتر \cref{pb:g6:matter-from-food:1} يوازن عند سطر
        \omterm{def:g2:animals-grow-up:born}{البيض}؛
  \item والزان لا يثمر بغزارة إلا في ``سنوات المسْط'' بعد مواسم
        نمو جيدة --- فمحاصيل \omterm{def:g2:from-seed-to-plant:seed}{البذور} صيوف مودَعة
        (\cref{ex:g6:plants-are-producers:pantry})؛
  \item والسنوات الصعبة تخفّف الحضنة: فكثير من \omterm{prop:g3:sorting-living-things:groups}{الطيور} يبيض بيضًا
        أقل، أو لا يبيض، حين تقلّ الفريسة.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[عدّ حضنة القرقف]\label{ex:g8:reproduction-and-environment:clutch}
تبين دراسات صناديق الأعشاش أن حضنات القرقف الأزرق تتعقب عرض
\omterm{ex:g2:animals-grow-up:butterfly}{اليرقات}: ففي غابات البلوط الغنية عشر بيضات أو أكثر؛ وفي الحدائق
الفقيرة ستّ أو سبع. والطائر لا يقرر ذلك بدفتر --- بل حال جسمها،
المبني من غذاء \omterm{def:g2:where-animals-live:habitat}{الموئل}، هي التي تضع ما تستطيع تموينه. فحجم الحضنة
جودةُ \omterm{def:g2:where-animals-live:habitat}{موئل} مكتوبةً \omterm{def:g2:animals-grow-up:born}{بالبيض}.
\end{example}

\section{محدود بالعالم: الجمهرات}

\begin{definition}[الجمهرة]\label{def:g8:reproduction-and-environment:population}
\emph{الجمهرة}\index{جمهرة} هي كل أفراد \omterm{def:g5:biodiversity:species}{نوع} واحد يعيشون في مكان
واحد --- ضفادع البركة، وزان الحرش. والتكاثر يضيف إليها؛ والوفيات
والرحيل يطرحان منها؛ \omterm{def:g6:exploring-our-environment:components}{والبيئة} تضع السقف.
\end{definition}

\begin{proposition}[لماذا لا تنفجر الجمهرات]\label{prop:g8:reproduction-and-environment:limits}
كل \omterm{def:g5:biodiversity:species}{نوع} يفرط في الإنتاج --- آلاف \omterm{def:g4:animal-reproduction:ovivivi}{البيوض} لكل زوج ضفادع، وآلاف \omterm{def:g2:from-seed-to-plant:seed}{البذور}
لكل زانة (\cref{prop:g4:animal-reproduction:strategies}) --- ومع
ذلك فالبرك ليست ملأى بالضفادع حتى الصلابة. فالفائض يُنفَق في مقابل
كوابح \omterm{def:g6:exploring-our-environment:components}{البيئة}:
\begin{itemize}
  \item \emph{الافتراس} --- بلاشين \omterm{ex:g2:animals-grow-up:butterfly}{ويرقات} رعّاشات
        \cref{ex:g4:animal-reproduction:pondcount}؛
  \item و\emph{الغذاء والمكان} --- فليس هناك إلا هذا القدر من
        \omterm{ex:g2:animals-grow-up:butterfly}{اليرقات}، وجحور الأعشاش، وبقع \omterm{def:g6:exploring-our-environment:conditions}{الضوء} على أرض الغابة؛
  \item و\emph{\omterm{def:g6:exploring-our-environment:conditions}{الشروط}} --- صقيع متأخر على الزهر
        (\cref{exo:g3:flowers-fruits-seeds:9})، وجفاف على بركة
        \omterm{ex:g2:animals-grow-up:frog}{الشراغيف} المتقلصة.
\end{itemize}
وفي المتوسط، في \omterm{def:g2:where-animals-live:habitat}{موئل} مستقر، يحلّ كل زوج محل نفسه --- وهو توازن
\cref{prop:g5:food-webs:balance}، منظورًا إليه من جانب
الوالدين.
\end{proposition}
\admitted

\begin{example}[ربيعان عند البركة]\label{ex:g8:reproduction-and-environment:twosprings}
الربيع الأول، معتدل ممطر: ماء عالٍ، وحشرات مبكرة --- فآلاف \omterm{def:g4:animal-reproduction:ovivivi}{البيوض}،
ومئات الضفيدعات، وترتفع
\omterm{def:g8:reproduction-and-environment:population}{الجمهرة}. أما الربيع الثاني، وهو ربيع جفاف،
فتنكمش فيه البركة
حتى تصير بركة صغيرة، وتزدحم \omterm{ex:g2:animals-grow-up:frog}{الشراغيف}، ويهبط \omterm{def:g7:what-is-respiration:respiration}{الأكسجين}
(\cref{prop:g7:oxygen-in-the-water:drains})، وتولم البلاشين على
السرب المركَّز --- فحفنة من الضفيدعات. الضفادع نفسها، والآلية
نفسها، وحكمان: \omterm{def:g6:exploring-our-environment:components}{فالبيئة} توقّع على كل جيل.
\end{example}

\begin{remark}[البشر يحرّكون الكوابح]\label{rem:g8:reproduction-and-environment:humans}
أكثر آثار البشر في الحياة البرية يجري عبر هذه الكوابح لا عبر
الآلية اللبّية: فالبرك المجففة والأسيجة المقتلعة تزيل \omterm{def:g2:where-animals-live:habitat}{موئل} التوالد
(\cref{prop:g2:where-animals-live:loss})؛ والمبيدات تُفرغ ذروة
\omterm{ex:g2:animals-grow-up:butterfly}{اليرقات} التي وُقّت لها القرقف؛ وصناديق أعشاش الحدائق والمروج
الرطبة المستعادة في \cref{ex:g5:biodiversity:storks} تعيد السقف.
فحماية \omterm{def:g5:biodiversity:species}{نوع} عمليًّا تعني حماية \omterm{def:g6:exploring-our-environment:components}{البيئة} التي وُصل بها تكاثره ---
الإشارة والميزانية والكوابح معًا.
\end{remark}

\begin{method}[تدقيق تغيّر جمهرة]\label{met:g8:reproduction-and-environment:audit}
حين ترتفع \omterm{def:g8:reproduction-and-environment:population}{جمهرة} أو تنخفض، دقّق بالترتيب:
\begin{enumerate}
  \item \emph{التوقيت}: أما زالت الإشارة وذروة المورد
        تتزامنان؟ (فربيع مبكر أكثر من اللازم قد يترك الفراخ بعد
        \omterm{ex:g2:animals-grow-up:butterfly}{اليرقات})؛
  \item و\emph{الميزانية}: ما الذي أتاح \omterm{def:g2:where-animals-live:habitat}{موئل} الموسم للوالدين أن
        يموّناه؟
  \item و\emph{الكوابح}: أيها تغيّر --- المفترسون، أم الغذاء، أم
        المكان، أم \omterm{def:g6:exploring-our-environment:conditions}{الشروط}؟
  \item و\emph{\omterm{def:g2:where-animals-live:habitat}{الموئل}}: أكُسبت أرض التوالد نفسها أم فُقدت؟
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g8:reproduction-and-environment:1}
لماذا يبيض القرقف الأزرق حين يبيض؟ سمِّ القاعدة والإشارة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:reproduction-and-environment:2}
ماذا يحقق توقيت الأيّل الخريفي، وما الذي يجب حسابه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:reproduction-and-environment:3}
لماذا كان التكاثر ``يُدفع ثمنه مادةً''؟ أعطِ ثلاث دفعات من
\cref{prop:g8:reproduction-and-environment:cost}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:reproduction-and-environment:4}
ما \omterm{def:g8:reproduction-and-environment:population}{الجمهرة}؟ وما الذي يضيف، وما الذي يطرح، وما الذي يضع السقف؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:reproduction-and-environment:5}
سمِّ الكوابح الثلاثة التي تنفق فائض الضفادع.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:reproduction-and-environment:6}
علام تدل حضنة قرقف من عشر بيضات في مقابل حضنة من ستّ، وعن ماذا؟

\begin{exercise}[$\star\star$]\label{exo:g8:reproduction-and-environment:7}
اشرح حيلة \omterm{def:g6:exploring-our-environment:conditions}{الضوء} عند مربّي الأغنام، وما تبرهن عليه في شأن

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:reproduction-and-environment:8}
أعد \cref{ex:g8:reproduction-and-environment:twosprings} أسئلةَ

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:reproduction-and-environment:9}
مناخ يدفأ يُخرج \omterm{ex:g2:animals-grow-up:butterfly}{يرقات} البلوط قبل موعدها بأسبوعين، وإشارة طول
النهار عند القرقف على حالها. تنبأ بعدم التطابق وبسطره في التدقيق.

\begin{exercise}[$\star\star$]\label{exo:g8:reproduction-and-environment:10}

\begin{exercise}[$\star\star$]\label{exo:g8:reproduction-and-environment:11}
أي كابح يحرّكه كل فعل بشري: تجفيف مستنقع؛ وتعليق صناديق أعشاش؛

\begin{exercise}[$\star\star\star$]\label{exo:g8:reproduction-and-environment:12}

\end{exercise}

\section{مسألة: دفتر صناديق الأعشاش}

\begin{problem}[{مسألة نهاية الأسبوع --- عشر سنوات من القرقف
الأزرق، مدقَّقة صندوقًا صندوقًا}]\label{pb:g8:reproduction-and-environment:1}
ترصد مدرسة عشرين صندوق عشّ عشر سنوات: تواريخ \omterm{def:g2:animals-grow-up:born}{البيض}، وأحجام
الحضنات، وعدد الفراخ الطائرة في كل صندوق --- إلى جانب سجلّ الطقس
ومفكرة حارس الأرض.

\medskip\noindent\textbf{القسم الأول --- قراءة السنوات
العادية.}
\begin{enumerate}
  \item السنوات من الأولى إلى الرابعة: \omterm{def:g2:animals-grow-up:born}{بيض} في أواخر نيسان،
        وحضنات من ثمانٍ إلى عشر، وأكثر \omterm{def:g2:animals-grow-up:born}{البيض} يطير. سمِّ الإشارة
        التي تضع أواخر نيسان، والذروة التي يقصدها التوقيت.
  \item حضنات السنة الثالثة كانت أكبر \omterm{def:g2:animals-grow-up:born}{ببيضة} أو بيضتين بعد أيار
        دافئ غني \omterm{ex:g2:animals-grow-up:butterfly}{باليرقات} في السنة الثانية --- إذ شتّى الوالدان
        شتاءً حسنًا. فأي قضية يوضّحها هذا الاقتران؟
  \item وحتى في السنوات الجيدة لا يطير نحو خُمس \omterm{def:g2:animals-grow-up:born}{البيض}. عدّد
        الكوابح المرجّحة، من
        \cref{prop:g8:reproduction-and-environment:limits}.
  \item ولماذا تسجّل المدرسة الطقس وأعمال الأرض إلى جانب \omterm{def:g2:animals-grow-up:born}{البيض}؟
        (أجب بأسئلة التدقيق الأربعة.)
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- الشذوذات.}
\begin{enumerate}[resume]
  \item السنة الخامسة: آذار دافئ شاذّ أخرج \omterm{ex:g2:animals-grow-up:butterfly}{اليرقات} قبل موعدها
        بأسبوعين؛ ولم تكد تواريخ \omterm{def:g2:animals-grow-up:born}{البيض} تتحرك. تنبأ بطيران السنة
        الخامسة، وسمِّ سطر الإخفاق في التدقيق.
  \item السنة السادسة: تسجّل مفكرة حارس الأرض إزالة \omterm{def:g1:plants-around-us:tree}{الأغصان}
        الميتة من البلوطات العتيقة وجزّ شريط المرج كل شهر.
        فينخفض الطيران مع أن الطقس كان لطيفًا. فأي سطور التدقيق
        تتهمها المفكرة؟
  \item السنة السابعة: يستقر زوج باشق. فتبقى الحضنات على حالها،
        لكن أعداد الفراخ الطائرة تهبط. فأي كابح تحرّك، ولماذا لم
        يتحرك سطر \emph{الحضنة} معه؟
  \item السنة الثامنة: تتضاعف الصناديق إلى أربعين؛ فيرتفع مجموع
        الفراخ الطائرة، لكن النجاح \emph{في الصندوق الواحد} يبقى
        ثابتًا. فأي سقف رفعته الصناديق الجديدة، وأيها تركته على
        حاله؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- التقرير.}
\begin{enumerate}[resume]
  \item اكتب جملة ملخص السنوات العشر التي تربط حجم الحضنة بجودة
        \omterm{def:g2:where-animals-live:habitat}{الموئل}، مستشهدًا بما في
        \cref{ex:g8:reproduction-and-environment:clutch}.
  \item تقترح البلدية قطع صف البلوط لموقف سيارات، ``تعويضًا''
        بخمسين صندوق عشّ إضافيًّا. دقّق في العرض: أي بنود الإشارة
        والميزانية والكوابح \omterm{def:g2:where-animals-live:habitat}{والموئل} تحلّ الصناديق محلها، وأيها
        لا يوفّره إلا البلوط؟
  \item أوصِ بإجراءين في الأرض للسنة الحادية عشرة، كل واحد
        مربوط بسطره في التدقيق.
  \item اختم بعبرة الفصل في جملة واحدة: ما الذي يحتاج إليه كل
        فرخ طائر غير والدين اثنين؟
\end{enumerate}
\end{problem}
