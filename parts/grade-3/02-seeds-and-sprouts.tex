\chapter{Seeds and Sprouts}\label{ch:g3:seeds}

Inside a small \omterm{def:g3:seeds:seed}{seed} waits a new \omterm{def:g1:plants-animals:plant}{plant}. Give it \omterm{def:g1:caring:needs}{water}, warmth and
\omterm{def:g1:caring:needs}{air}, and it may \emph{germinate} --- wake up and sprout. This
chapter follows \omterm{def:g2:plant-parts:main}{seeds} from rest to seedling.

\section{What is a seed?}

\begin{definition}[Seed]\label{def:g3:seeds:seed}
A \emph{seed}\index{seed} is a \omterm{def:g1:plants-animals:plant}{plant}'s package for a new \omterm{def:g1:plants-animals:plant}{plant}. It
holds a tiny \omterm{def:g1:plants-animals:plant}{plant} embryo and a store of \omterm{def:g2:food-energy:food}{food}, often protected by a
seed coat.
\end{definition}

\omimg{seed-sprouting.jpg}{0.52\linewidth}{From dry \omterm{def:g3:seeds:seed}{seed} to \omterm{def:g2:plant-parts:main}{root} and
shoot --- \omterm{def:g3:seeds:germ}{germination} in stages.}

\begin{example}[Seeds you know]\label{ex:g3:seeds:know}
Beans, peas, sunflower \omterm{def:g2:plant-parts:main}{seeds}, apple pips and acorns are \omterm{def:g2:plant-parts:main}{seeds}. Some
are eaten as \omterm{def:g2:food-energy:food}{food}; all can \omterm{def:g2:what-alive:signs}{grow} if conditions are right and the \omterm{def:g3:seeds:seed}{seed}
is alive.
\end{example}

\section{Germination}

\begin{definition}[Germination]\label{def:g3:seeds:germ}
\emph{Germination}\index{germination} is the start of \omterm{def:g2:growing:growth}{growth} from a
\omterm{def:g3:seeds:seed}{seed} into a seedling. The \omterm{def:g2:plant-parts:main}{root} usually appears first, then the
shoot \omterm{def:g2:what-alive:signs}{grows} up toward light.
\end{definition}

\begin{method}[Help a seed sprout]\label{met:g3:seeds:grow}
\begin{enumerate}
  \item \omterm{def:g1:caring:needs}{Place} \omterm{def:g2:plant-parts:main}{seeds} on moist paper or in damp soil.
  \item Keep them warm --- not scorching.
  \item Give \omterm{def:g1:caring:needs}{air} (do not seal in a full \omterm{def:g1:caring:needs}{water} jar with no \omterm{def:g1:caring:needs}{air}).
  \item Wait and check moisture every day.
  \item When green \omterm{def:g2:plant-parts:main}{leaves} open, give light.
\end{enumerate}
\end{method}

\begin{proposition}[Needs for germination]\label{prop:g3:seeds:needs}
Most \omterm{def:g2:plant-parts:main}{seeds} need \omterm{def:g1:caring:needs}{water}, a suitable temperature and \omterm{def:g1:caring:needs}{air} to germinate.
Light is needed later for green \omterm{def:g2:growing:growth}{growth}; some \omterm{def:g2:plant-parts:main}{seeds} also need light
to start, others prefer dark.
\end{proposition}
\admitted

\begin{example}[Bean jar]\label{ex:g3:seeds:jar}
Line a jar with paper, add a little \omterm{def:g1:caring:needs}{water}, tuck in bean \omterm{def:g2:plant-parts:main}{seeds}.
In days you may see white \omterm{def:g2:plant-parts:main}{roots} and green shoots --- a window into
\omterm{def:g3:seeds:germ}{germination}.
\end{example}
