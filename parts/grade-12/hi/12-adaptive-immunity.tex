\chapter{अनुकूली प्रतिरक्षा और टीकाकरण}\label{ch:g12:adaptive-immunity}

1796 में किसी देहाती डॉक्टर ने दूध दुहने वाली एक स्त्री के गोशीतला फफोले से
पदार्थ खुरचकर किसी लड़के की बाँह में डाला, छह हफ़्ते इंतज़ार किया, और फिर
उसे चेचक का \omterm{def:g12:adaptive-immunity:vaccine}{टीका} लगाया। वह लड़का बीमार नहीं पड़ा। उस डॉक्टर की दुनिया में
कुछ भी इसकी व्याख्या नहीं करता था; लगता था कि उस शरीर को सिखा दिया गया है।
दो सदियों बाद चेचक लुप्त है, और वह सिखाना अणुओं तक समझा जा चुका है: यानी
\omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} का एक समुच्चय जो आपस में मिलकर लगभग हर उस अणु के लिए एक ग्राही ढोता
है जो कभी हो सकता था, जो अपना अणु प्रकट होने पर कई गुना होता है, और जो याद
रखता है। यह अध्याय उसी तंत्र का, उसकी दो बाँहों का, उसकी स्मृति का, उस पर
बने \omterm{def:g12:adaptive-immunity:vaccine}{टीकों} का --- और उस विषाणु का वर्णन करता है जो उसे नष्ट कर देता है।

\section{प्रतिजन और लसीकाणु}

\begin{definition}[प्रतिजन, लसीकाणु]\label{def:g12:adaptive-immunity:antigen}
\emph{प्रतिजन}\index{प्रतिजन} कोई भी ऐसा अणु है जिसे \omterm{def:g12:innate-immunity:immunity}{अनुकूली प्रतिरक्षा}
तंत्र पहचान सकता है: आम तौर पर किसी रोगाणु का कोई \omterm{def:g11:gene-expression:protein}{प्रोटीन} या शर्करा, कोई
विष, किसी पराई \omterm{def:g10:cells-common-unit:cell}{कोशिका} की सतह --- या, ग़लती से, शरीर का अपना ही कोई। पहचानती
\omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} \emph{लसीकाणु}\index{लसीकाणु} हैं, यानी अस्थि मज्जा में बनी और
लसीका ग्रंथियों, प्लीहा तथा आँत के लसीकाभ ऊतक में जमा श्वेत \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}। हर
लसीकाणु किसी एक \emph{ग्राही} की हज़ारों प्रतियाँ ढोता है, और हर ग्राही केवल
एक ही प्रतिजन से बँधता है। \emph{B लसीकाणु} मज्जा में परिपक्व होते हैं और
\omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} बनाएँगे; और \emph{T लसीकाणु} थाइमस में परिपक्व होते हैं तथा
संपर्क से और संकेतों से काम करते हैं।
\end{definition}

\begin{proposition}[पहले से बनाया गया भंडार]\label{prop:g12:adaptive-immunity:repertoire}
हर \omterm{def:g12:adaptive-immunity:antigen}{लसीकाणु} का ग्राही, उस \omterm{def:g10:cells-common-unit:cell}{कोशिका} के परिपक्व होते समय, \omterm{def:g10:universal-dna:gene}{जीन} खंडों के किसी
यादृच्छिक पुनर्विन्यास से जोड़ा जाता है, जो हर \omterm{def:g10:cells-common-unit:cell}{कोशिका} में अलग होता है।
इसलिए वह शरीर किसी \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} से मिलने से पहले ही कोई $10^{8}$ अलग-अलग ग्राही
थामे रहता है --- यानी इतने कि लगभग हर अणु को उससे बँधती कुछ \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} मिल
जाएँ। जिन \omterm{def:g12:adaptive-immunity:antigen}{लसीकाणुओं} का ग्राही शरीर के अपने अणुओं से बँधता हो वे परिपक्वन के
दौरान मिटा दिए जाते हैं; और बाक़ी इंतज़ार करते हैं, हर संभव \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} के लिए
कुछ \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}, जब तक उनका \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} न आ जाए।
\end{proposition}
\admitted

\section{क्लोनी वरण}

\begin{proposition}[वरण और विस्तार]\label{prop:g12:adaptive-immunity:selection}
कोई \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} घुसे तो केवल वही \omterm{def:g12:adaptive-immunity:antigen}{लसीकाणु} जवाब देते हैं जिनका ग्राही उस पर बैठता
हो: वे \emph{चुने} जाते हैं, और कई दिनों पर \omterm{def:g11:cell-cycle-mitosis:cycle}{समसूत्री विभाजन} से हज़ारों समरूप
\omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के किसी क्लोन में कई गुना हो जाते हैं --- यानी
\emph{क्लोनी वरण}\index{क्लोनी वरण}। फिर वह क्लोन विभेदित होता है: अधिकांश
\omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} \emph{प्रभावक \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}} बनती हैं जो एक-दो हफ़्ते उस \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} से
लड़ती हैं और मर जाती हैं; और एक अल्पसंख्या दीर्घजीवी
\emph{स्मृति कोशिकाएँ}\index{स्मृति कोशिका} बनती है, जो सालों टिकी रहती हैं
और, वह \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} लौटे तो, तेज़ तथा कहीं बड़ी संख्या में जवाब देती हैं। वह
प्रतिक्रिया इसलिए विशिष्ट है कि केवल बैठते क्लोन ही फैलते हैं; वह पहली बार
इसलिए धीमी है कि वे मुट्ठी भर \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से शुरू होते हैं; और वह दूसरी बार
इसलिए तेज़ है कि \omterm{prop:g12:adaptive-immunity:selection}{स्मृति कोशिकाएँ} पहले ही बहुत हैं।
\end{proposition}
\begin{proof}[साक्ष्य]
किसी दिए गए \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} के ख़िलाफ़ \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} पहले इंजेक्शन के लगभग एक हफ़्ते
बाद रक्त में प्रकट होते हैं, दो हफ़्ते चढ़ते हैं और फिर घटते हैं; जबकि महीनों
बाद का दूसरा इंजेक्शन दो-तीन दिनों के भीतर पहली चोटी से दस से सौ गुना
\omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} बना देता है, जो सालों टिके रहते हैं। उस दूसरी ख़ुराक के साथ ही
कोई दूसरा, असंबंधित \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} लगाया जाए तो उस \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} के लिए पहले प्रकार की
धीमी प्रतिक्रिया मिलती है: यानी वह स्मृति विशिष्ट है। किसी प्रतिरक्षित जानवर
से लिए \omterm{def:g12:adaptive-immunity:antigen}{लसीकाणु} किसी अनजान जानवर में डालने पर वह स्मृति भी चली जाती है; जबकि
किसी बिना प्रतिरक्षित जानवर के \omterm{def:g12:adaptive-immunity:antigen}{लसीकाणु} नहीं ले जाते।
\end{proof}

\section{दो बाँहें: प्रतिरक्षी और मारक कोशिकाएँ}

\begin{proposition}[B लसीकाणु और प्रतिरक्षी]\label{prop:g12:adaptive-immunity:antibodies}
कोई चुना गया B क्लोन \emph{प्लाज़्मा \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं}} में विभेदित होता है, यानी
ऐसे कारख़ानों में जो प्रति सेकंड हज़ारों \emph{प्रतिरक्षी}\index{प्रतिरक्षी}
स्रावित करते हैं: यानी उस क्लोन के ग्राही की घुलनशील प्रतियाँ, जो Y आकार के
\omterm{def:g11:gene-expression:protein}{प्रोटीन} हैं और जिनकी दोनों बाँहें उस \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} से बँधती हैं। \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} मारते
नहीं; वे \emph{चिह्नित} करते हैं। किसी विषाणु या किसी विष से बँधकर वे उसे
\omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में घुसने या काम करने से रोक देते हैं; और किसी जीवाणु से बँधकर वे
उस पर एक परत चढ़ा देते हैं, तथा वही परत है जिसे
\cref{ch:g12:innate-immunity} की \omterm{def:g12:innate-immunity:phagocytosis}{भक्षककोशिकाएँ} सबसे अच्छी तरह पकड़ती हैं।
बहुत-से \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षियों} से आड़े-तिरछे जुड़े \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} \emph{प्रतिरक्षा संकुलों}
में गुँथ जाते हैं, जो \omterm{def:g12:innate-immunity:phagocytosis}{भक्षकाणुक्रिया} से साफ़ किए जाते हैं। यह रक्त तथा तरलों
में बैठे घुसपैठियों के ख़िलाफ़ वाली बाँह है।
\end{proposition}
\admitted

\begin{proposition}[T लसीकाणु]\label{prop:g12:adaptive-immunity:tcells}
T \omterm{def:g12:adaptive-immunity:antigen}{लसीकाणु} \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} को केवल किसी दूसरी \omterm{def:g10:cells-common-unit:cell}{कोशिका} की सतह पर दिखाए गए टुकड़ों की
तरह पहचानते हैं (\cref{ch:g12:innate-immunity})। दो क़िस्में:
\begin{itemize}
  \item \emph{सहायक T कोशिकाएँ}\index{सहायक T कोशिका} उन \omterm{prop:g12:innate-immunity:handover}{वृक्षाभ कोशिकाओं}
        से सक्रिय होती हैं जो वह \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} लसीका ग्रंथि तक लाती हैं; और चुने
        तथा कई गुना हो जाने के बाद वे ऐसे साइटोकाइन स्रावित करती हैं जो B
        \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की तथा बाक़ी T \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की प्रतिक्रिया को इजाज़त देते और
        बढ़ाते हैं। वे समन्वयक हैं; उनके बिना शायद ही कोई अनुकूली
        प्रतिक्रिया होती है।
  \item \emph{कोशिकाविषी T \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}} किसी संक्रमित \omterm{def:g10:cells-common-unit:cell}{कोशिका} के दिखाए किसी
        विषाणु (या किसी \omterm{def:g11:cancer:cancer}{अर्बुद} \omterm{def:g11:gene-expression:protein}{प्रोटीन}) के टुकड़े पहचानती हैं, और उस \omterm{def:g10:cells-common-unit:cell}{कोशिका}
        को संपर्क से मार देती हैं, इससे पहले कि उसके भीतर का विषाणु कई गुना
        हो सके। यह \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के भीतर छिपे घुसपैठियों के ख़िलाफ़ वाली बाँह
        है, जहाँ \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} पहुँच ही नहीं सकते।
\end{itemize}
\end{proposition}
\admitted

दिन कई गुना होता है; दिन 7 तक कोशिकाविषी T \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} वायुमार्गों में
संक्रमित \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} मार रही होती हैं और \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} मुक्त विषाणु उदासीन कर रहे
होते हैं; और दिन 10 तक वह व्यक्ति उबर रहा होता है, दोनों क़िस्म की \omterm{prop:g12:adaptive-immunity:selection}{स्मृति
कोशिकाओं} के साथ। साल भर बाद वही प्रभेद: रक्त में पहले से मौजूद \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी}
अधिकांश विषाणु प्रवेश पर ही रोक देते हैं, \omterm{prop:g12:adaptive-immunity:selection}{स्मृति कोशिकाएँ} दो दिनों के भीतर
फैल जाती हैं, और वह संक्रमण ध्यान में आने से पहले ही ख़त्म हो जाता है। पर
कोई ऐसा अलग प्रभेद जिसके सतह \omterm{def:g11:gene-expression:protein}{प्रोटीन} उत्परिवर्तित हो चुके हों उन \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षियों}
से बच निकलता है --- इसीलिए वह \omterm{def:g12:adaptive-immunity:vaccine}{टीका} हर साल दोबारा बनाया जाता है।
\end{example}

\section{टीकाकरण}

\begin{definition}[टीका]\label{def:g12:adaptive-immunity:vaccine}
\emph{टीका}\index{टीका} ऐसी तैयारी है जो किसी रोगजनक के ख़िलाफ़ कोई
प्राथमिक अनुकूली प्रतिक्रिया, और इसलिए स्मृति, छेड़ती है, वह भी उसकी बीमारी
पैदा किए बिना: यानी वह रोगजनक कमज़ोर किया हुआ (ख़सरा, तपेदिक), मारा हुआ
(इंजेक्शन वाला पोलियो), उसके किसी एक \omterm{def:g11:gene-expression:protein}{प्रोटीन} या किसी निष्क्रिय विष तक घटाया
हुआ (टिटनेस, हेपेटाइटिस B), या, सबसे हाल में, उस \omterm{def:g10:cells-common-unit:cell}{कोशिका} के लिए वह \omterm{def:g11:gene-expression:protein}{प्रोटीन}
बनाने के निर्देशों तक। उस तैयारी में जोड़ा गया कोई \emph{सहायक द्रव्य} वह
छोटा शोथ छेड़ता है जिसके बिना \omterm{prop:g12:innate-immunity:handover}{वृक्षाभ कोशिकाएँ} वह \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} लसीका ग्रंथि तक न
ले जातीं। और बूस्टर ख़ुराकें उस प्राथमिक प्रतिक्रिया को द्वितीयक में बदल
देती हैं तथा उस स्मृति को दशकों तक बढ़ा देती हैं।
\end{definition}

\begin{proposition}[समूह की रक्षा]\label{prop:g12:adaptive-immunity:herd}
कोई \omterm{def:g12:adaptive-immunity:vaccine}{टीका} लगा व्यक्ति सुरक्षित है; और \omterm{def:g12:adaptive-immunity:vaccine}{टीका} लगी कोई \omterm{def:g12:selection-drift-speciation:population}{आबादी} अपने बिना \omterm{def:g12:adaptive-immunity:vaccine}{टीके} वाले
सदस्यों की भी रक्षा करती है। कोई संक्रमित व्यक्ति बिना किसी प्रतिरक्षा वाली
\omterm{def:g12:selection-drift-speciation:population}{आबादी} में औसतन $R_0$ और लोगों को कोई बीमारी दे देता है (ख़सरे के लिए लगभग
15, इन्फ़्लुएंज़ा के लिए 2 से 3); और कोई अंश $v$ प्रतिरक्षित हो तो उन
संपर्कों में से केवल $R_0(1 - v)$ ही संक्रमित हो सकते हैं, तथा वह बीमारी तब
पीछे हटती है जब वह संख्या 1 से नीचे गिरे --- यानी जब
\[
v > 1 - \frac{1}{R_0}.
\]
ख़सरे के लिए उन शिशुओं की भी रक्षा करने को, जो \omterm{def:g12:adaptive-immunity:vaccine}{टीके} के लिए बहुत छोटे हैं, और
उन लोगों की जो उसे नहीं ले सकते, \omterm{def:g12:selection-drift-speciation:population}{आबादी} के 93\% को प्रतिरक्षित होना चाहिए; और
इन्फ़्लुएंज़ा के लिए लगभग 60\%। यही \emph{सामूहिक प्रतिरक्षा}\index{सामूहिक प्रतिरक्षा}
वह वजह है कि टीकाकरण एक सामूहिक काम है।
\end{proposition}
\admitted

\section{जब वह समन्वयक ही नष्ट हो जाए}

\begin{proposition}[HIV और एड्स]\label{prop:g12:adaptive-immunity:hiv}
मानव प्रतिरक्षा-न्यूनता विषाणु, यानी HIV, \omterm{prop:g12:adaptive-immunity:tcells}{सहायक T कोशिकाओं} को संक्रमित करता
है, और वह भी ठीक उसी ग्राही से घुसकर जो उन्हें चिह्नित करता है, तथा उन्हीं
में कई गुना होता है। सालों तक वह \omterm{def:g12:innate-immunity:immunity}{प्रतिरक्षा तंत्र} संक्रमित \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} मारता
और \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} बनाता रहता है --- वह व्यक्ति \emph{सीरम-धनात्मक} है, और
संक्रामक भी --- जबकि वह विषाणु, लगातार उत्परिवर्तित होकर, हर प्रतिक्रिया से
बच निकलता है और धीरे-धीरे सहायक \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} चुका देता है। उनकी गिनती सामान्य के
लगभग पाँचवें हिस्से से नीचे गिरे तो अनुकूली प्रतिक्रिया का समन्वय ही नहीं हो
पाता: यानी \emph{एड्स}\index{एड्स}, जिसमें वे संक्रमण जो कोई स्वस्थ शरीर
बिना ध्यान दिए साफ़ कर देता है --- मुँह का कोई कवक, फेफड़ों का कोई हल्का
परजीवी --- जानलेवा हो जाते हैं, और दुर्लभ \omterm{def:g11:cancer:cancer}{कैंसर} प्रकट होते हैं। विषाणुरोधी
दवाएँ उस विषाणु के \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} रोकती हैं, सहायक \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} लौटा देती हैं और, जीवन
भर ली जाएँ तो, उस बीमारी को दूर रखती हैं तथा संचरण रोक देती हैं; अभी तक न
कोई \omterm{def:g12:adaptive-immunity:vaccine}{टीका} है, न कोई इलाज।
\end{proposition}
\admitted

\begin{method}[किसी प्रतिरक्षा प्रतिक्रिया को पढ़ना]\label{met:g12:adaptive-immunity:read}
\begin{enumerate}
  \item उस \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} को और यह पहचानिए कि वह कहाँ है: तरलों में मुक्त
        (\omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} काम करते हैं) या \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के भीतर (कोशिकाविषी T
        \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} काम करती हैं)।
  \item समय पढ़िए: चोटी तक एक हफ़्ते का मतलब कोई प्राथमिक प्रतिक्रिया, और
        कुछ ही दिनों का मतलब स्मृति।
  \item विशिष्टता पढ़िए: किसी एक \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} पर प्रतिक्रिया दूसरे पर
        प्रतिक्रिया को अछूता छोड़ देती है।
  \item \omterm{prop:g12:adaptive-immunity:tcells}{सहायक T कोशिकाएँ} ढूँढ़िए: उनके बिना कुछ नहीं होता, और \omterm{prop:g12:adaptive-immunity:hiv}{एड्स} यही
        दिखाता है।
  \item किसी \omterm{def:g12:adaptive-immunity:vaccine}{टीके} के लिए उस \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} का रूप, सहायक द्रव्य, ख़ुराकों की
        संख्या, और \omterm{def:g12:selection-drift-speciation:population}{आबादी} का ज़रूरी अंश बताइए।
\end{enumerate}
\end{method}

\begin{remark}[वरण, एक बार फिर]\label{rem:g12:adaptive-immunity:selection}
\omterm{def:g12:innate-immunity:immunity}{अनुकूली प्रतिरक्षा} तंत्र डार्विनी है: यादृच्छिक ग्राहियों वाली \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की
एक \omterm{def:g12:selection-drift-speciation:population}{आबादी}, जिसमें से पर्यावरण --- यानी वह \omterm{def:g12:adaptive-immunity:antigen}{प्रतिजन} --- कुछ को कई गुना होने के
लिए चुनता है, और जिनके वंशजों को वही जीतने वाला ग्राही विरासत में मिलता है।
वह \cref{ch:g12:selection-drift-speciation} की प्रक्रिया एक ही शरीर के भीतर
एक हफ़्ते में चला देता है, और स्मृति के रास्ते उसके नतीजे जीवन भर रख लेता
है। यही वह वजह भी है कि \omterm{def:g12:innate-immunity:immunity}{प्रतिरक्षा तंत्र} को सिखाया जा सकता है, जो कोई \omterm{def:g12:adaptive-immunity:vaccine}{टीका}
करता है, और यह भी कि जो विषाणु उस तंत्र के चुनने से तेज़ उत्परिवर्तित हो
--- HIV, इन्फ़्लुएंज़ा --- उसे हराना इतना मुश्किल क्यों है।
\end{remark}

\section{समस्या: तिरानबे प्रतिशत}

\begin{problem}[{सप्ताहांत समस्या --- किसी टीकाकरण अभियान का हिसाब: दो
ख़ुराकों पर प्रतिक्रियाएँ, सामूहिक प्रतिरक्षा का अंकगणित, ख़सरे के किसी
प्रकोप का नमूना, और वह विषाणु जो इस तंत्र को ही उलट देता है}]
\label{pb:g12:adaptive-immunity:1}
ख़सरे का $R_0 = 15$ है। 50\,000 लोगों के किसी क़स्बे की 90\% \omterm{def:g12:selection-drift-speciation:population}{आबादी} \omterm{def:g12:adaptive-immunity:vaccine}{टीके} से
या बीते संक्रमण से प्रतिरक्षित है; और 1500 एक साल से छोटे शिशु हैं, जो \omterm{def:g12:adaptive-immunity:vaccine}{टीके}
के लिए बहुत छोटे हैं।

\medskip
\noindent\textbf{भाग I --- एक व्यक्ति की प्रतिक्रिया।}
कोई बच्ची 12 महीने पर पहली ख़ुराक पाती है और 18 पर दूसरी।
\begin{enumerate}
  \item पहली ख़ुराक के बाद \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} वक्र का वर्णन कीजिए: देरी, चोटी,
        गिरावट।
  \item दूसरी ख़ुराक के बाद उसका वर्णन कीजिए और \omterm{prop:g12:adaptive-immunity:selection}{क्लोनी वरण} से वह अंतर
        समझाइए।
  \item \omterm{def:g12:adaptive-immunity:vaccine}{टीके} का विषाणु कमज़ोर है पर ज़िंदा है। कोई जीवित विषाणु क्यों
        इस्तेमाल होता है, और वह प्रतिक्रिया की कौन-सी बाँह सिखाता है जो
        कोई मारा हुआ न सिखाता?
  \item दस साल बाद वह बच्ची ख़सरे से मिलती है। पहले तीन दिनों में क्या होता
        है, और वह बीमार क्यों नहीं पड़ती?
  \item उसका बिना \omterm{def:g12:adaptive-immunity:vaccine}{टीके} वाला चचेरा भाई उसी विषाणु से मिलता है। उसके पहले दस
        दिनों का, और उसके बाद उसके पास जो बचता है उसका वर्णन कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वह दहलीज़।}
\begin{enumerate}[resume]
  \item ख़सरे का फैलना रुकने के लिए \omterm{def:g12:selection-drift-speciation:population}{आबादी} का जितना अंश प्रतिरक्षित होना
        चाहिए वह निकालिए।
  \item 90\% प्रतिरक्षित के साथ कोई एक संक्रमित व्यक्ति औसतन कितने लोगों को
        संक्रमित कर सकता है? क्या वह क़स्बा सुरक्षित है?
  \item उस क़स्बे में कितने लोग प्रतिरक्षित नहीं हैं, और उनमें से कितने वे
        शिशु हैं?
  \item कवरेज चढ़कर 95\% हो जाए, तो कितने ग्राह्य लोग बचते, और कोई एक मामला
        औसतन क्या पैदा करता?
  \item एक वाक्य में समझाइए कि उन शिशुओं की सलामती वयस्कों के फ़ैसलों पर
        क्यों निर्भर है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- एक प्रकोप।}
कोई यात्री 90\% प्रतिरक्षा वाले उस क़स्बे में ख़सरा ले आता है।
\begin{enumerate}[resume]
  \item हर मामला 12 दिनों की एक के बाद एक पीढ़ियों में
        $15 \times 0.10 = 1.5$ और लोगों को संक्रमित करता है, यह लीजिए।
        तीसरी पीढ़ी में कितने मामले? और छठी में?
  \item छह पीढ़ियों के बाद कुल कितने मामले?
  \item 5000 ग्राह्य लोगों में से 30\% शिशु हैं। यदि मामले ग्राह्य लोगों पर
        बराबरी से पड़ें, तो छह पीढ़ियों बाद मामलों में शिशुओं की संख्या
        आँकिए।
  \item मामलों की हर पीढ़ी पीछे प्रतिरक्षित बचे लोग छोड़ जाती है: समझाइए कि
        वह प्रकोप अपने आप क्यों धीमा पड़ता है, और मोटे तौर पर कब।
  \item 95\% प्रतिरक्षा वाले किसी क़स्बे के लिए सवाल 11 दोहराइए। दोनों
        क़स्बों की तुलना कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वह तंत्र, उलटा हुआ।}
HIV से संक्रमित कोई रोगी, बिना इलाज के, 8 सालों में प्रति माइक्रोलीटर 1000
से 200 तक \omterm{prop:g12:adaptive-immunity:tcells}{सहायक T कोशिकाएँ} खो देता है।
\begin{enumerate}[resume]
  \item प्रति साल, और प्रति महीने, औसत नुक़सान निकालिए।
  \item प्रति माइक्रोलीटर 200 पर \omterm{prop:g12:adaptive-immunity:hiv}{एड्स} शुरू होता है। संगठन वाले चित्र से
        समझाइए कि \omterm{prop:g12:adaptive-immunity:antibodies}{प्रतिरक्षी} तथा मारक \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} दोनों क्यों प्रभावित होती
        हैं जबकि संक्रमित केवल सहायक \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} हैं।
