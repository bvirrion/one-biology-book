\chapter{Dari Sel Telur sampai Kelahiran}\label{ch:g8:from-egg-to-birth}

\Cref{ch:g5:human-reproduction} menceritakan sembilan
bulannya sebagai sebuah kisah; kini kamu memiliki setiap alat
yang dibutuhkan agar kisah itu dimainkan dengan benar: \omterm{def:g5:first-look-at-cells:cell}{sel}
dan pembelahannya, kedua \omterm{def:g8:sexual-reproduction:gametes}{gamet} dan tempat pertemuannya,
\omterm{def:g8:puberty:hormones}{hormon}, pengantaran \omterm{prop:g4:journey-of-food:absorb}{darahnya}. Bab ini memutar ulang
perkembangan manusia dengan permesinannya terlihat --- dari
satu \omterm{def:g5:first-look-at-cells:cell}{sel} menjadi tiga ribu miliar, dalam empat puluh pekan.

\section{Pekan pertama: dari zigot menjadi penghuni}

\begin{proposition}[Perjalanan menuruni saluran telur]\label{prop:g8:from-egg-to-birth:firstweek}
\omterm{def:g5:flower-to-fruit:steps}{Pembuahan} di dalam \omterm{def:g8:reproductive-systems:female}{saluran telur}
(\cref{def:g8:reproductive-systems:female}) menjalankan dua
jam sekaligus:
\begin{enumerate}
  \item \emph{pembelahan}: \omterm{prop:g8:sexual-reproduction:core}{zigotnya} membelah dalam sehari,
        lalu membelah lagi dan lagi
        (\cref{ex:g5:first-look-at-cells:one}) --- dua,
        empat, delapan \omterm{def:g5:first-look-at-cells:cell}{sel}, sebuah bola murbei berisi
        puluhan \omterm{def:g5:first-look-at-cells:cell}{sel} pada hari keempat, semuanya sambil
        hanyut menuruni \omterm{def:g8:reproductive-systems:female}{saluran telur} ke arah \omterm{prop:g5:human-reproduction:plan}{rahimnya};
  \item \emph{penanaman}: sekitar hari ketujuh bola kecil
        itu mencapai \omterm{prop:g5:human-reproduction:plan}{rahimnya} lalu membenamkan diri ke dalam
        lapisan yang sudah disiapkan (peristiwa 4
        \cref{prop:g8:reproductive-systems:cycle}) ---
        penghuninya tiba, daurnya berhenti, dan \omterm{def:g8:puberty:hormones}{hormon}
        mengisyaratkan kepada seluruh sistemnya: \omterm{def:g5:human-reproduction:pregnancy}{kehamilan}.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Apa yang menghidupi pembelahan pertamanya]\label{ex:g8:from-egg-to-birth:provisions}
Sepanjang pekan pertamanya sang pengembara tidak menyantap
apa pun dan tidak menerima pengantaran apa pun: ia berjalan
dengan bekal \omterm{def:g4:animal-reproduction:sexual}{sel telurnya} sendiri --- persis itulah gunanya
pembekalan pada \cref{ex:g8:reproductive-systems:egg}.
Spermanya membawa \omterm{def:g6:cell-unit-of-life:parts}{inti sel} ayahnya dan nyaris tidak ada yang
lain; \omterm{def:g2:animals-grow-up:born}{telurnya} membawa \omterm{def:g6:cell-unit-of-life:parts}{inti sel} yang satu lagi
\emph{beserta} lumbungnya. Penanaman adalah, antara lain,
lumbung yang menipis dan penghuninya menyambungkan diri ke
layanan kamar.
\end{example}

\section{Embrio, plasenta, janin}

\begin{definition}[Plasenta]\label{def:g8:from-egg-to-birth:placenta}
Dari bola yang tertanam itu tumbuh dua bangunan sekaligus:
\emph{\omterm{def:g5:human-reproduction:pregnancy}{embrio}} itu sendiri, dan antarmuka penyangga hidupnya
--- \emph{plasenta}\index{plasenta}, sebuah cakram yang
tumbuh ke dalam dinding \omterm{prop:g5:human-reproduction:plan}{rahim}, tempat \omterm{prop:g4:journey-of-food:absorb}{darah} ibunya dan \omterm{prop:g4:journey-of-food:absorb}{darah}
\omterm{def:g5:human-reproduction:pregnancy}{embrionya} mendekat dengan akrab \emph{tanpa bercampur}.
Melintasi permukaannya yang teramat luas dan tipis ---
spesifikasi pada \cref{prop:g7:how-animals-breathe:spec},
dibangun sekali lagi --- \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} dan \omterm{def:g7:what-is-respiration:respiration}{oksigen} menyeberang ke
\omterm{def:g5:human-reproduction:pregnancy}{embrionya}, buangan dan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} menyeberang kembali
(\omterm{prop:g5:human-reproduction:cord}{tali pusar} pada \cref{prop:g5:human-reproduction:cord}
membentang antara plasenta dan \omterm{def:g5:human-reproduction:pregnancy}{embrionya}). Plasenta adalah
\omterm{def:g7:how-animals-breathe:four}{paru-paru}, usus dan \omterm{def:g7:removing-wastes:kidneys}{ginjal} bagi penumpangnya --- ketiga
perbatasan di dalam satu organ.
\end{definition}

\begin{proposition}[Jadwalnya, ditinjau ulang dengan alatnya]\label{prop:g8:from-egg-to-birth:timetable}
Tonggak pada \cref{ex:g5:human-reproduction:timetable}, kini
dijelaskan:
\begin{itemize}
  \item pekan 1--8, tahap \emph{\omterm{def:g5:human-reproduction:pregnancy}{embrio}}: rancangan tubuhnya
        digelar --- \omterm{def:g5:first-look-at-cells:cell}{selnya} membelah, berpindah dan
        \emph{mengkhusus}
        (\cref{rem:g6:cell-unit-of-life:twoways}) menjadi
        \omterm{def:g4:heart-and-blood:heart}{jantung}, \omterm{def:g5:brain-in-command:system}{saraf}, \omterm{def:g3:bones-and-muscles:skeleton}{tulang} dan yang lain; \omterm{def:g4:heart-and-blood:heart}{jantungnya}
        berdenyut sejak pekan keempat, karena tata rumah
        tangga lumbung-dan-plasentanya butuh pompanya awal;
  \item pekan 9--40, tahap \emph{\omterm{def:g5:human-reproduction:pregnancy}{janin}}: rancangannya sudah
        terbangun, kini tumbuh dan matang --- \omterm{def:g5:brain-in-command:system}{otaknya}
        merangkai dirinya, \omterm{def:g7:how-animals-breathe:four}{paru-parunya} berlatih dalam
        keadaan kosong (\cref{exo:g5:human-reproduction:11}
        menjelaskan pementasan perdana tangis pertamanya),
        tendangan bulan keempat mengumumkan \omterm{def:g3:bones-and-muscles:muscle}{otot} yang bekerja;
  \item sepanjang itu: \omterm{def:g8:puberty:hormones}{hormon} menjaga \omterm{prop:g5:human-reproduction:plan}{rahimnya} tetap tenang
        dan daurnya tertunda --- kimia pada
        \cref{def:g8:puberty:hormones}, yang menjalankan
        tata usaha \omterm{def:g5:human-reproduction:pregnancy}{kehamilannya}.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Mengapa pekan awalnya paling menuntut perawatan]\label{ex:g8:from-egg-to-birth:earlycare}
\Cref{prop:g5:human-reproduction:care} memperingatkan bahwa
tembakau dan alkohol menyeberang ke bayinya; jadwalnya
berkata kapan hal itu paling penting: pada pekan \omterm{def:g5:human-reproduction:pregnancy}{embrio}
ketika rancangannya sedang digelar, ketika tiap organ masih
berupa segerombol kecil \omterm{def:g5:first-look-at-cells:cell}{sel} yang sedang mengkhusus, satu
petunjuk yang terganggu dapat bergaung menembus seluruh
pembangunannya --- dan justru itulah pekan ketika sebuah
\omterm{def:g5:human-reproduction:pregnancy}{kehamilan} dapat berlalu tanpa disadari. Karena itulah
hitungan tumpul para dokternya: perawatannya dimulai bukan
pada pemindaian pertama melainkan pada kemungkinan pertama.
\end{example}

\section{Kelahiran, dan pergantian tugasnya}

\begin{proposition}[Kelahiran sebagai pergantian tugas atas perintah hormon]\label{prop:g8:from-egg-to-birth:birth}
Sekitar pekan keempat puluh, isyarat \omterm{def:g8:puberty:hormones}{hormon} --- milik bayinya
dan milik ibunya dalam percakapan --- mengakhiri ketenangan
panjang \omterm{prop:g5:human-reproduction:plan}{rahimnya}:
\begin{enumerate}
  \item \emph{persalinan}: \omterm{prop:g5:human-reproduction:plan}{rahimnya}, \omterm{def:g3:bones-and-muscles:muscle}{otot} terkuat di dalam
        tubuh (\cref{rem:g8:reproductive-systems:living}
        menjumpai kerjanya yang lebih lembut), \omterm{def:g3:bones-and-muscles:muscle}{berkontraksi}
        dalam gelombang yang menguat, yang membuka jalannya
        lalu mendorong bayinya menuruni jalan lahirnya;
  \item \emph{\omterm{def:g2:born-grow-die:cycle}{kelahiran}}: biasanya kepala lebih dahulu
        (\cref{ex:g5:human-reproduction:birth});
  \item \emph{pergantian tugasnya}: tangis pertamanya
        mengisi \omterm{def:g7:how-animals-breathe:four}{paru-paru} yang belum pernah dipakai; lalu
        lintas \omterm{prop:g5:human-reproduction:cord}{tali pusarnya} berakhir lalu ia dipotong;
        dalam beberapa menit sang bayi berjalan dengan
        \omterm{def:g7:how-animals-breathe:four}{paru-paru}, usus yang akan datang dan \omterm{def:g7:removing-wastes:kidneys}{ginjalnya}
        sendiri --- setiap perbatasan yang tadinya ditanggung
        \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya} (\cref{exo:g5:human-reproduction:11},
        kini dengan peta lengkapnya);
  \item \emph{ari-ari}: \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya}, tugasnya selesai,
        terlepas lalu menyusul keluar --- satu-satunya organ
        yang dibangun tubuh untuk sekali pakai lalu dibuang.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Audit pertama sang bayi baru lahir]\label{ex:g8:from-egg-to-birth:audit}
Bidan memberi nilai kepada bayi baru lahir beberapa menit
sesudah \omterm{def:g2:born-grow-die:cycle}{kelahirannya} --- \omterm{def:g7:what-is-respiration:respiration}{pernapasan}, \omterm{def:g4:heart-and-blood:heart}{denyut jantung}, warna,
gerak, tangis. Bacalah daftar periksanya dengan mata buku
ini: itulah gelung pasokan-dan-pembuangan pada
\cref{prop:g7:removing-wastes:complete}, yang diaudit pada
jalan mandirinya yang pertama. Ujian yang pertama-tama
dilalui setiap manusia adalah ujian faal.
\end{example}

\begin{method}[Menjelaskan tahap mana pun pada kehamilan]\label{met:g8:from-egg-to-birth:explain}
Bila ditanya tentang saat mana pun dari sembilan bulannya,
jawablah dengan empat alat:
\begin{enumerate}
  \item \emph{\omterm{def:g5:first-look-at-cells:cell}{sel}}: apa yang sedang dikerjakan pembelahan
        dan pengkhususannya?
  \item \emph{pasokan}: lumbung, atau \omterm{def:g8:from-egg-to-birth:placenta}{plasenta} --- apa yang
        menyeberang, apa yang tidak boleh?
  \item \emph{\omterm{def:g8:puberty:hormones}{hormon}}: apa yang ditahan, atau diperintahkan?
  \item \emph{perawatan}: apa yang dituntut tahapnya dari
        kebiasaan ibunya dan penanggalan dokternya?
\end{enumerate}
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g8:from-egg-to-birth:1}
Tentukan waktu pekan pertamanya: mulainya pembelahan, bola
murbeinya, penanamannya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:from-egg-to-birth:2}
Apa yang menghidupi sang pengembara pada pekan pertamanya,
dan apa yang mengakhiri pengaturan itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:from-egg-to-birth:3}
Apa itu \omterm{def:g8:from-egg-to-birth:placenta}{plasenta} --- apa yang menyeberanginya ke tiap arah,

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:from-egg-to-birth:4}
Bagilah keempat puluh pekannya menjadi \omterm{def:g5:human-reproduction:pregnancy}{embrio} dan \omterm{def:g5:human-reproduction:pregnancy}{janin},

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:from-egg-to-birth:5}
Mengapa \omterm{def:g4:heart-and-blood:heart}{jantungnya} berdenyut sejak pekan keempat,

\begin{exercise}[$\star$]\label{exo:g8:from-egg-to-birth:6}

\begin{exercise}[$\star\star$]\label{exo:g8:from-egg-to-birth:7}

\begin{exercise}[$\star\star$]\label{exo:g8:from-egg-to-birth:8}

\begin{exercise}[$\star\star$]\label{exo:g8:from-egg-to-birth:9}

\begin{exercise}[$\star\star$]\label{exo:g8:from-egg-to-birth:10}

\begin{exercise}[$\star\star\star$]\label{exo:g8:from-egg-to-birth:12}

membuat pergantian itu menakjubkan.
\end{exercise}

\section{Soal: Garis Waktu Sang Bidan}

\begin{problem}[{Soal akhir pekan --- sebuah kehamilan
diikuti kunjungan demi kunjungan}]\label{pb:g8:from-egg-to-birth:1}
Berkas seorang bidan untuk satu \omterm{def:g5:human-reproduction:pregnancy}{kehamilan} (rekaan):
kunjungan pendaftaran pada pekan 8, pemindaian pada pekan 12
dan 20, pemeriksaan pada pekan 28 dan 36, \omterm{def:g2:born-grow-die:cycle}{kelahiran} pada
pekan 40. Bacalah berkasnya dengan alat babnya.

\medskip\noindent\textbf{Bagian I --- Halaman awal
berkasnya.}
\begin{enumerate}
  \item \omterm{def:g5:human-reproduction:pregnancy}{Kehamilannya} dipastikan oleh sebuah uji \omterm{def:g8:puberty:hormones}{hormon}
        berpekan-pekan sebelum kunjungan pertamanya. Isyarat
        mana yang dibaca uji semacam itu, dan sejak
        peristiwa yang mana pada
        \cref{prop:g8:from-egg-to-birth:firstweek} ia ada?
  \item Pada pekan 8 bidannya mencatat ``\omterm{def:g5:human-reproduction:pregnancy}{embrio} lengkap
        dalam rancangannya, kira-kira sebesar sebutir
        kacang''. Terjemahkan ke dalam istilah jadwalnya ---
        tahap apa yang ditutup di sini?
  \item Nasihat kunjungan pendaftarannya dibuka dengan tanpa
        alkohol, tanpa asap --- dan asam folat, vitamin sang
        pembangun, ``sebaiknya sejak sebelum permulaannya''.
        Berikan alasan ketiganya dari
        \cref{ex:g8:from-egg-to-birth:earlycare}.
  \item Pemindaian pekan 12 memperlihatkan \omterm{def:g4:heart-and-blood:heart}{denyut jantung}
        yang mulai pada pekan 4. Mengapa sedini itu, dalam
        istilah pasokannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Bulan tengahnya.}
\begin{enumerate}[resume]
  \item Pemindaian pekan 20 mengukur setiap organ dan
        kedudukan \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya}. Sebutkan urusan tahap
        \omterm{def:g5:human-reproduction:pregnancy}{janinnya}, dan ketiga tugas \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya} yang sedang
        diperiksa.
  \item Ibunya melaporkan tendangan pertama menjelang pekan
        18. Permesinan apa yang sedang mengumumkan dirinya
        (\cref{prop:g5:brain-in-command:loop} menyebutkan
        rantainya)?
  \item Pada pekan 28 bidannya memeriksa gula dan tekanan
        \omterm{prop:g4:journey-of-food:absorb}{darah} ibunya. Mengapa peredaran \omterm{prop:g4:journey-of-food:absorb}{darah} ibunya
        terbebani ganda --- perbatasan siapa saja yang
        sedang dijalankannya?
  \item Pekan 30--36: \omterm{def:g5:human-reproduction:pregnancy}{janinnya} terutama tumbuh dan
        \omterm{def:g7:how-animals-breathe:four}{paru-parunya} berlatih. Berlatih apa --- dan untuk
        satu saat yang mana?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Halaman terakhirnya.}
\begin{enumerate}[resume]
  \item Pekan 40: persalinan. Sebutkan \omterm{def:g3:bones-and-muscles:muscle}{ototnya}, kedua
        tugasnya, dan siapa yang memerintahkan kerjanya
        (\cref{prop:g8:from-egg-to-birth:birth}).
  \item Catatan \omterm{def:g2:born-grow-die:cycle}{kelahiran} pada berkasnya: ``tangis kuat
        seketika; \omterm{prop:g5:human-reproduction:cord}{tali pusar} dijepit lalu dipotong; \omterm{def:g8:from-egg-to-birth:placenta}{plasenta}
        lengkap.'' Auditlah ketiga anak kalimatnya dengan
        pergantian tugasnya.
  \item Sang bayi baru lahir mendapat nilai tinggi pada
        daftar periksa berskala menit itu. Nyatakan apa yang
        disahkan daftar periksanya, dalam istilah
        \cref{ex:g8:from-egg-to-birth:audit}.
  \item Tutuplah berkasnya dengan satu kalimat untuk orang
        tuanya: apa yang dibangun keempat puluh pekannya,
        dari \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal yang mana, oleh permesinan yang mana.
\end{enumerate}
\end{problem}
