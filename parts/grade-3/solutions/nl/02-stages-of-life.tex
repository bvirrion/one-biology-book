\section*{Hoofdstuk \ref{ch:g3:stages-of-life} --- De stadia van het leven}

\begin{solution}{exo:g3:stages-of-life:1}
\omterm{def:g3:stages-of-life:stages}{Baby}, kind, \omterm{def:g3:stages-of-life:stages}{puber}, \omterm{def:g3:stages-of-life:stages}{volwassene}, \omterm{def:g3:stages-of-life:stages}{ouderdom}.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:2}
(a) In de kindertijd, vanaf ongeveer zes jaar. (b) In het volwassen
stadium. (c) In het babystadium --- het eerste jaar is het snelste van allemaal.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:3}
De stoot van de \omterm{def:g3:stages-of-life:stages}{baby} en de stoot van de \omterm{def:g3:stages-of-life:stages}{puber}, met de rustige, gestage
groei van de kindertijd ertussen.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:4}
De lijn wordt vlak: de leeftijd schuift langs de onderkant door, maar
de lengte klimt niet meer.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:5}
Tussen $6$ en $8$: van ongeveer $\qty{115}{cm}$ tot ongeveer
$\qty{127}{cm}$ --- ruwweg $\qty{12}{cm}$ over die twee jaar. In het
eerste jaar alleen: van ongeveer $\qty{50}{cm}$ tot ongeveer
$\qty{75}{cm}$ --- ruwweg $\qty{25}{cm}$, twee keer zo veel als die
twee kinderjaren samen.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:6}
Omdat een vergelijking alleen eerlijk is als de metingen op dezelfde
manier gedaan zijn. De ene dag schoenen en de volgende sokken, of een
slordig schuin boek, zouden ``groei'' maken (of verbergen) die er
nooit geweest is.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:7}
Eikel (\omterm{def:g2:from-seed-to-plant:seed}{zaad}), \omterm{prop:g2:from-seed-to-plant:cycle}{kiemplantje}, jonge eik, volwassen eik die bloeit en
eikels draagt, oude eik --- en met de tijd de dood.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:8}
Open antwoord. Een negenjarige zit in het kindstadium; de stoot van de
puberteit ligt nog voor hem.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:9}
Vlieg (weken), kat (ongeveer een jaar), mens (bijna twintig jaar), eik
(tientallen jaren). Bij alle vier is de volgorde van de stadia dezelfde
en kan ze niet worden overgeslagen of achterstevoren doorlopen: jong,
volwassen, oud.
\end{solution}

\begin{solution}{exo:g3:stages-of-life:10}
Nee. Op zijn negende zit hij in de rustige strook van de kindertijd ---
ongeveer $\qty{6}{cm}$ per \emph{jaar}, dus een halve centimeter per
maand: veel te weinig om te zien tussen twee metingen die een maand uit
elkaar liggen. De curve klimt zeker, maar langzaam; hij hoeft alleen
een paar maanden tussen zijn punten te laten.
\end{solution}
