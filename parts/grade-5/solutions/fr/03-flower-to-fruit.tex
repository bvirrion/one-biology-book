\section*{Chapitre \ref{ch:g5:flower-to-fruit} --- De la fleur au fruit}

\begin{solution}{exo:g5:flower-to-fruit:1}
Les \omterm{def:g3:flowers-fruits-seeds:parts}{étamines} sont \omterm{def:g4:animal-reproduction:sexual}{mâles}~: leurs grains de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} portent les
\omterm{def:g5:first-look-at-cells:cell}{cellules} reproductrices \omterm{def:g4:animal-reproduction:sexual}{mâles}. Le \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} est \omterm{def:g4:animal-reproduction:sexual}{femelle}~: à
l'intérieur reposent les \omterm{prop:g5:flower-to-fruit:roles}{ovules}, chacun contenant une \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:2}
\omterm{def:g5:flower-to-fruit:steps}{Pollinisation}~: un grain de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} voyage d'une \omterm{def:g3:flowers-fruits-seeds:parts}{étamine} jusqu'au
sommet d'un \omterm{def:g3:flowers-fruits-seeds:parts}{pistil}. \omterm{def:g5:flower-to-fruit:steps}{Fécondation}~: le tube du grain porte la \omterm{def:g4:animal-reproduction:sexual}{cellule
mâle} jusqu'à un ovule, où elle rejoint la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}. La
\omterm{def:g5:flower-to-fruit:steps}{pollinisation} vient en premier --- le voyage avant la rencontre.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:3}
Fabriqué dans une \omterm{def:g3:flowers-fruits-seeds:parts}{étamine}~; porté (par un insecte ou par le vent)
jusqu'au sommet d'un \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} --- la \omterm{def:g5:flower-to-fruit:steps}{pollinisation}~; il fait pousser
un fin tube à travers le \omterm{def:g3:flowers-fruits-seeds:parts}{pistil}~; sa \omterm{def:g4:animal-reproduction:sexual}{cellule mâle} atteint un ovule
et rejoint la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} --- la \omterm{def:g5:flower-to-fruit:steps}{fécondation}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:4}
Chaque ovule fécondé devient une \omterm{def:g2:from-seed-to-plant:seed}{graine}~; le \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} gonfle en
\omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}~; les \omterm{def:g3:flowers-fruits-seeds:parts}{pétales} (et les \omterm{def:g3:flowers-fruits-seeds:parts}{étamines}) se fanent et tombent.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:5}
Chaque pépin est un ovule fécondé devenu \omterm{def:g2:from-seed-to-plant:seed}{graine}. Des loges vides
signifient des \omterm{prop:g5:flower-to-fruit:roles}{ovules} non fécondés --- trop peu de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} s'est
posé, et le \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit} a gonflé de façon inégale autour du manque.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:6}
Leur \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} voyage par le vent~: fin comme de la poussière,
fabriqué en quantités énormes, jeté en l'air. Le prix est le
gaspillage --- presque aucun grain ne se pose à un endroit utile,
si bien que la quantité doit remplacer la précision.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:7}
Pas de visite, pas de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} sur le \omterm{def:g3:flowers-fruits-seeds:parts}{pistil} --- la \omterm{def:g5:flower-to-fruit:steps}{pollinisation}, la
première étape, n'a jamais eu lieu, donc aucun tube n'a poussé et
aucun ovule n'a été fécondé~; la \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} se fane tout entière.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:8}
Réponse libre. Attendu~: les \omterm{def:g3:flowers-fruits-seeds:parts}{fleurs} pollinisées à la main et
marquées d'un fil donnent nettement plus souvent des \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} --- le
pinceau a fait le travail de l'abeille.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:9}
\omterm{def:g4:animal-reproduction:sexual}{Cellules mâles}~: dans les grains de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} des \omterm{def:g3:flowers-fruits-seeds:parts}{étamines}. \omterm{def:g4:animal-reproduction:sexual}{Cellules
œufs}~: dans les \omterm{prop:g5:flower-to-fruit:roles}{ovules} à l'intérieur du \omterm{def:g3:flowers-fruits-seeds:parts}{pistil}. \omterm{def:g5:flower-to-fruit:steps}{Fécondation}~: au
fond du tube pollinique, dans l'ovule. Première \omterm{def:g5:first-look-at-cells:cell}{cellule} du nouvel
individu~: la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée dans l'ovule --- qui devient la
petite \omterm{def:g1:plants-around-us:plant}{plante} de la \omterm{def:g2:from-seed-to-plant:seed}{graine}, comme un \omterm{def:g1:animals-around-us:animal}{animal} se développe à partir de
sa propre première \omterm{def:g5:first-look-at-cells:cell}{cellule}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:10}
Une mauvaise récolte --- la plupart des \omterm{def:g3:flowers-fruits-seeds:parts}{fleurs} sont tombées sans
\omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}. La machinerie était saine et les \omterm{def:g3:flowers-fruits-seeds:parts}{fleurs} intactes~; ce qui a
échoué, c'est l'étape 1, la \omterm{def:g5:flower-to-fruit:steps}{pollinisation}~: pendant deux semaines
décisives les livreurs sont restés chez eux, si bien que la plupart
des \omterm{def:g3:flowers-fruits-seeds:parts}{pistils} n'ont jamais reçu de \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} et que la plupart des \omterm{prop:g5:flower-to-fruit:roles}{ovules}
n'ont jamais été fécondés.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:11}
Les bourdons vendent la \omterm{def:g5:flower-to-fruit:steps}{pollinisation} --- l'étape du transport~:
porter le \omterm{def:g3:flowers-fruits-seeds:parts}{pollen} des \omterm{def:g3:flowers-fruits-seeds:parts}{étamines} aux \omterm{def:g3:flowers-fruits-seeds:parts}{pistils} à l'intérieur d'un
bâtiment où aucun pollinisateur du dehors ne peut entrer. Le \omterm{def:g3:flowers-fruits-seeds:parts}{pollen}
de tomate se décroche facilement, si bien que la vibration d'un
ventilateur et l'air en mouvement peuvent assurer un transport
grossier à l'intérieur de chaque \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} --- un livreur mécanique. Le
\omterm{def:g3:flowers-fruits-seeds:parts}{pollen} de cerisier, lui, doit voyager d'arbre en arbre, de \omterm{def:g3:flowers-fruits-seeds:parts}{fleur} en
\omterm{def:g3:flowers-fruits-seeds:parts}{fleur}, avec une précision que seul un corps visitant les \omterm{def:g3:flowers-fruits-seeds:parts}{fleurs}
l'une après l'autre apporte~; le vent dans un hangar ne sait pas
viser, et pour les cerises le livreur vivant n'a pas de substitut
bon marché.
\end{solution}
