\section*{Bab \ref{ch:g4:plants-without-seeds} --- Tumbuhan Tanpa Biji}

\begin{solution}{exo:g4:plants-without-seeds:1}
\omterm{prop:g4:plants-without-seeds:damp}{Lumut} dan \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku}. Keduanya berkembang biak dengan \omterm{def:g4:plants-without-seeds:spore}{spora} ---
butir renik yang dilepaskan dalam jumlah amat besar.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:2}
Sebutir \omterm{def:g4:plants-without-seeds:spore}{spora} itu renik dan telanjang --- tanpa \omterm{def:g3:animals-through-seasons:active}{cadangan makanan},
tanpa \omterm{def:g1:plants-around-us:plant}{tumbuhan} mungil yang sudah jadi; sebutir \omterm{def:g2:from-seed-to-plant:seed}{biji} membawa keduanya,
di dalam kulit pelindung. Dan \omterm{def:g4:plants-without-seeds:spore}{spora} dibuat dalam jumlah jutaan, \omterm{def:g2:from-seed-to-plant:seed}{biji}
dalam jumlah lusinan.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:3}
Di sisi bawah \omterm{def:g1:plants-around-us:parts}{daunnya}: deretan rapi bintik cokelat, masing-masing
sebuah bungkusan berisi ribuan \omterm{def:g4:plants-without-seeds:spore}{spora} yang terbelah ketika masak lalu
membiarkannya melayang pergi seperti debu.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:4}
\omterm{prop:g4:plants-without-seeds:damp}{Lumut} berkembang biak dengan \omterm{def:g4:plants-without-seeds:spore}{spora}, dan \omterm{def:g4:plants-without-seeds:spore}{spora} beserta tahap mudanya
yang halus membutuhkan kelembapan --- keduanya mengering lalu mati di
udara yang kering. Tembok utara yang teduh tetap lembap; tembok
selatan yang tersinari terpanggang.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:5}
Membuat \omterm{def:g1:plants-around-us:plant}{tumbuhan} baru langsung dari sepotong \omterm{def:g1:plants-around-us:plant}{tumbuhan} induknya ---
\omterm{ex:g4:plants-without-seeds:runner}{geragih}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{umbi lapis}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{umbi batang}, \omterm{met:g4:plants-without-seeds:cutting}{stek}. Hanya satu induk.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:6}
Tanamannya menjulurkan \omterm{def:g1:plants-around-us:parts}{batang} panjang yang tipis --- \omterm{ex:g4:plants-without-seeds:runner}{geragihnya} ---
melintasi tanah; di tempat ia menyentuh tanah, ia berakar lalu
menumbuhkan \omterm{def:g1:plants-around-us:parts}{daun}, menjadi tanaman kecil yang lengkap; \omterm{ex:g4:plants-without-seeds:runner}{geragihnya}
kemudian layu dan anaknya berdiri bebas, serupa dengan induknya.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:7}
Sebutir kentang yang ditanam (sebuah \omterm{ex:g4:plants-without-seeds:bulbtuber}{umbi batang}) bertunas dengan
andal dari ``\omterm{def:g1:the-five-senses:sense}{mata}''-nya, dan tiap \omterm{def:g1:the-five-senses:sense}{mata} dapat tumbuh menjadi satu
tanaman utuh yang serupa dengan jenis unggul yang ditanam --- lebih
cepat dan lebih pasti daripada \omterm{def:g2:from-seed-to-plant:seed}{biji}.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:8}
Jawaban terbuka. \omterm{def:g1:plants-around-us:parts}{Akar} putih muncul dari \omterm{def:g1:plants-around-us:parts}{batang} yang terendam,
biasanya sesudah satu sampai dua minggu; \omterm{met:g4:plants-without-seeds:cutting}{steknya} lalu siap ditanam ---
sebuah tanaman mint baru yang lengkap, dari satu induk.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:9}
\omterm{def:g4:plants-without-seeds:spore}{Sporanya} telanjang dan murah: tanpa \omterm{def:g3:animals-through-seasons:active}{cadangan makanan} atau \omterm{def:g1:plants-around-us:plant}{tumbuhan}
mungil, hampir setiap butirnya hilang, sehingga jutaan harus dibuat.
\omterm{def:g2:from-seed-to-plant:seed}{Bijinya} adalah perlengkapan bertahan hidup yang mahal, sehingga
selusin yang berperlengkapan lengkap sudah cukup. Keduanya
menggemakan \omterm{prop:g3:sorting-living-things:groups}{ikan} kod (jutaan, tanpa perawatan) dan gajah (sedikit,
berbekal baik).
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:10}
Stroberinya datang lewat \omterm{ex:g4:plants-without-seeds:runner}{geragih} --- \omterm{def:g4:plants-without-seeds:vegetative}{perkembangbiakan vegetatif}, satu
induk, salinan yang persis. Kacangnya datang dari \omterm{def:g2:from-seed-to-plant:seed}{biji} yang dibuat
oleh \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}, yang mencampurkan sumbangan dua induk (\omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} dari
satu \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}, \omterm{def:g4:animal-reproduction:sexual}{sel telur} di \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} yang lain), sehingga tiap tanaman
kacangnya sedikit berbeda.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:11}
\omterm{def:g4:plants-without-seeds:vegetative}{Perkembangbiakan vegetatif} --- dahannya adalah \omterm{met:g4:plants-without-seeds:cutting}{stek} alami. Banjirnya
mengerjakan pekerjaan gunting sang tukang kebun: ia memotong tunasnya,
membawanya, lalu menanamnya di lumpur basah, ranjang perakaran yang
ideal bagi dedalu. Sungai memagari dirinya dengan pohon dedalu dengan
mematahkan lalu menanamnya dalam satu tarikan yang sama.
\end{solution}
