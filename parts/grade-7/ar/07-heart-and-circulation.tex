\section{غرف القلب الأربع}

\begin{definition}[الأذينان والبطينان]\label{def:g7:heart-and-circulation:rooms}
لكل من جانبي \omterm{def:g4:heart-and-blood:heart}{القلب} (\cref{def:g4:heart-and-blood:heart}) غرفتان:
\emph{أذين}\index{أذين} في الأعلى يستقبل \omterm{prop:g4:journey-of-food:absorb}{الدم} الوارد،
و\emph{بطين}\index{بطين} في الأسفل --- وهو غرفة الضخّ الغليظة
الجدار --- يدفعه إلى الخارج. وبينهما وبعدهما
\emph{صمامات}\index{صمام القلب} --- أبواب ذات اتجاه واحد ---
تنغلق بشدة خلف كل دفقة؛ وانغلاقاتها المزدوجة هي بالضبط ``لَبْ
دَبْ'' التي تسمعها السمّاعة (أو \omterm{def:g1:the-five-senses:sense}{أذن} على صدر).
\end{definition}

\begin{proposition}[دقّة واحدة، بالترتيب]\label{prop:g7:heart-and-circulation:beat}
كل دقّة تجري التسلسل نفسه في الجانبين معًا: ينقبض \omterm{def:g7:heart-and-circulation:rooms}{الأذينان} فيملآن
البطينين؛ ثم ينقبض \omterm{def:g7:heart-and-circulation:rooms}{البطينان} فيدفعان \omterm{prop:g4:journey-of-food:absorb}{الدم} إلى الخارج --- الجانب
الأيمن نحو \omterm{def:g7:how-animals-breathe:four}{الرئتين}، والأيسر نحو الجسم --- بينما ينغلق صمّاما
الدخول بشدة (لَبْ)؛ ثم يرتخي \omterm{def:g7:heart-and-circulation:rooms}{البطينان}، وينغلق صمّاما الخروج بشدة
(دَبْ)، وتعود الغرف فتمتلئ. ثم من جديد، $100000$ مرة في اليوم
(\cref{rem:g4:heart-and-blood:rest}).
\end{proposition}
\admitted

\section{دارتان وثلاثة أوعية}

\begin{definition}[الدوران المزدوج]\label{def:g7:heart-and-circulation:double}
شبكة طرق \omterm{prop:g4:journey-of-food:absorb}{الدم} حلقتان عبر مضخة واحدة
(وهما جانبا \cref{def:g4:heart-and-blood:heart}، الآن
دارتين):
\begin{itemize}
  \item \emph{الدارة الرئوية}: \omterm{def:g7:heart-and-circulation:rooms}{البطين} الأيمن
        $\leftarrow$ \omterm{def:g7:how-animals-breathe:four}{الرئتان} (تحميل \omterm{def:g7:what-is-respiration:respiration}{الأكسجين} وتفريغ \omterm{def:g7:what-is-respiration:respiration}{ثاني أكسيد
        الكربون}) $\leftarrow$ \omterm{def:g7:heart-and-circulation:rooms}{الأذين} الأيسر؛
  \item و\emph{الدارة الجسمية}: \omterm{def:g7:heart-and-circulation:rooms}{البطين} الأيسر
        $\leftarrow$ الجسم كله (تسليمًا وجمعًا)
        $\leftarrow$ \omterm{def:g7:heart-and-circulation:rooms}{الأذين} الأيمن.
\end{itemize}
وكل قطرة تتناوب الدارتين: رئتان، فجسم، فرئتان، فجسم --- ولهذا كان
\omterm{def:g7:heart-and-circulation:rooms}{البطين} الأيسر، وهو يدفع في حلقة الجسم الأكبر بكثير، أغلظ غرف \omterm{def:g4:heart-and-blood:heart}{القلب}
جدارًا.
\end{definition}

\begin{definition}[الشرايين والشعيرات والأوردة]\label{def:g7:heart-and-circulation:vessels}
ثلاثة أصناف من \omterm{def:g4:heart-and-blood:vessels}{الأوعية} تصنع كل حلقة
(\cref{def:g4:heart-and-blood:vessels}، وقد سُمّيت الآن):
\begin{itemize}
  \item \emph{الشرايين}\index{شريان}: غليظة الجدران، مرنة، تحمل
        \omterm{prop:g4:journey-of-food:absorb}{الدم} \emph{بعيدًا} عن \omterm{def:g4:heart-and-blood:heart}{القلب} تحت ضغط --- \omterm{prop:g4:heart-and-blood:pulse}{والنبض}
        (\cref{prop:g4:heart-and-blood:pulse}) هو تمدّدها عند كل
        دفقة؛
  \item و\emph{الشعيرات الدموية}\index{شعيرة دموية}: \omterm{def:g4:heart-and-blood:vessels}{الأوعية}
        الرفيعة كالشعر التي تتخلل كل عضو --- وجدرانها بسمك \omterm{def:g5:first-look-at-cells:cell}{خلية}
        واحدة، وعندها تحدث \emph{كل} التبادلات: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{الأسناخ}،
        \omterm{prop:g7:digestion-and-nutrients:absorption}{والزغابات}، \omterm{def:g3:bones-and-muscles:muscle}{والعضلات}، \omterm{def:g1:the-five-senses:sense}{والجلد}؛
  \item و\emph{الأوردة}\index{وريد}: واسعة، أرقّ جدارًا، تعيد
        \omterm{prop:g4:journey-of-food:absorb}{الدم} \emph{إلى} \omterm{def:g4:heart-and-blood:heart}{القلب} تحت ضغط منخفض، ولها \omterm{def:g7:heart-and-circulation:rooms}{صمامات} ذات
        اتجاه واحد خاصة بها --- وهي الخطوط المزرقّة في
        \cref{rem:g4:heart-and-blood:see}.
\end{itemize}
\end{definition}

\begin{remark}[لماذا كانت الشعيرات هي المقصود]\label{rem:g7:heart-and-circulation:capillaries}
\omterm{def:g7:heart-and-circulation:vessels}{الشرايين} \omterm{def:g7:heart-and-circulation:vessels}{والأوردة} طرق سريعة لا غير؛ وكل تسليم فعلي --- كل تبادل
درسته هذه السنة --- يحدث عبر جدران الشعيرات. ومواصفة
\cref{prop:g7:how-animals-breathe:spec} تعيش هنا أيضًا: جدران \omterm{def:g5:first-look-at-cells:cell}{بخلية}
واحدة (رقيقة)، تتخلل كل نسيج بعشرات آلاف الكيلومترات
(\cref{def:g4:heart-and-blood:vessels} --- كبيرة)، مغمورة \omterm{def:g6:exploring-our-environment:conditions}{برطوبة}
الجسم. \omterm{def:g4:heart-and-blood:heart}{والقلب} موجود ليُبقي \omterm{prop:g4:journey-of-food:absorb}{الدم} متحركًا \emph{عبر الشعيرات}؛ وكل
ما عداه سباكة إليها ومنها.
\end{remark}

\section{قراءة الشبكة وصيانتها}

\begin{example}[قياسات على قارعة الطريق]\label{ex:g7:heart-and-circulation:measures}
ثلاث قراءات يومية للشبكة: \emph{\omterm{prop:g4:heart-and-blood:pulse}{النبض}} --- تمدّد \omterm{def:g7:heart-and-circulation:vessels}{الشريان}، وعدّ
الدفقات (\cref{met:g4:heart-and-blood:pulse})؛ و\emph{ضغط \omterm{prop:g4:journey-of-food:absorb}{الدم}}،
وهو عددا الكمّ --- ضغط \omterm{def:g7:heart-and-circulation:vessels}{الشريان} عند الدفقة وبين الدفقات؛
و\emph{لَبْ دَبْ} في السمّاعة --- أبواب \omterm{def:g7:heart-and-circulation:rooms}{الصمامات} تُسمع وهي تنغلق. ثلاث نوافذ،
بلا شقّ.
\end{example}

\begin{proposition}[صيانة الشبكة]\label{prop:g7:heart-and-circulation:keep}
أعداء الشبكة وأصدقاؤها على المدى الطويل معروفون:
\begin{itemize}
  \item \emph{التمرين} يقوّي المضخة
        (\cref{prop:g7:organs-at-work:training}) ويُبقي الطرق
        السريعة لدنة؛
  \item و\emph{الأكل المتوازن}
        (\cref{met:g3:balanced-diet:check}) يحمي جدران
        \omterm{def:g7:heart-and-circulation:vessels}{الشرايين}، وهي التي يكسوها \omterm{prop:g4:journey-of-food:absorb}{الدم} المفرط الغنى ببطء
        فيتصلّب؛
  \item و\emph{الدخان} يهاجم \omterm{def:g4:heart-and-blood:vessels}{الأوعية} كما يهاجم \omterm{def:g7:how-animals-breathe:four}{الرئتين}
        (\cref{prop:g7:lungs-and-gas-exchange:smoke}) ---
        تضييقًا وتصليبًا وسرقةً للمقاعد؛
  \item وتموين \omterm{def:g4:heart-and-blood:heart}{القلب} نفسه --- فهو يتغذى بشرايينه الصغيرة الخاصة،
        لا من \omterm{prop:g4:journey-of-food:absorb}{الدم} الذي في غرفه --- يتوقف على الثلاثة كلها.
\end{itemize}
وعادات \cref{prop:g5:healthy-choices:contract} هي، عمليًّا، صيانة
\omterm{def:g4:heart-and-blood:vessels}{أوعية}.
\end{proposition}
\admitted

\begin{method}[إدخال أي عضو في الشبكة]\label{met:g7:heart-and-circulation:map}
لأي عضو يلقاه هذا الكتاب:
\begin{enumerate}
  \item ضعه على الدارة الجسمية: \omterm{def:g7:heart-and-circulation:vessels}{شريان} داخل، \omterm{def:g7:heart-and-circulation:vessels}{ووريد} خارج، وشعيرات
        في داخله؛
  \item وعدّد ما يعبر جدران شعيراته، داخلًا وخارجًا (\omterm{prop:g7:digestion-and-nutrients:absorption}{الزغابات}:
        \omterm{def:g7:digestion-and-nutrients:families}{مغذيات} إلى الداخل؛ \omterm{def:g3:bones-and-muscles:muscle}{والعضلة}: وقود \omterm{def:g7:what-is-respiration:respiration}{وأكسجين} إلى الداخل،
        وثاني أكسيد كربون إلى الخارج؛ \omterm{def:g1:the-five-senses:sense}{والجلد}: حرارة إلى الخارج)؛
  \item ولاحظ وصلته الخاصة إن وُجدت (فدم المعي يمرّ \omterm{prop:g7:digestion-and-nutrients:absorption}{بالكبد}
        أولًا؛ \omterm{def:g7:how-animals-breathe:four}{والرئتان} تقعان على دارتهما الخاصة)؛
  \item وراجع التدفقات في مقابل عمل العضو --- فرسم الشبكة ينبغي
        أن يعيد حكاية فصل العضو.
\end{enumerate}
\end{method}

\section{مسألة: صباح طبيب القلب}

\begin{problem}[{مسألة نهاية الأسبوع --- أربع استشارات، وخريطة
واحدة للشبكة}]\label{pb:g7:heart-and-circulation:1}
رافق طبيب \omterm{def:g4:heart-and-blood:heart}{قلب} صباحًا في أربع استشارات (متخيَّلة)، والخريطة في يدك.

\medskip\noindent\textbf{القسم الأول --- خطّ الأساس
السليم.}
\begin{enumerate}
  \item المريضة الأولى مراهقة لائقة، تنال قراءات قارعة الطريق
        الثلاث في
        \cref{ex:g7:heart-and-circulation:measures}. سمِّ كل
        واحدة، وأي جزء من الآلة تقرؤه.
  \item ونبضها في الراحة $58$. فيبتسم الطبيب. ولماذا
        (\cref{exo:g4:heart-and-blood:11})؟
  \item و``لَبْ دَبْ'' عندها حادّة مزدوجة. فما الذي انغلق بشدة،
        مرتين، في كل دقّة؟
  \item ارسم مسار دمها من \omterm{def:g7:heart-and-circulation:rooms}{الأذين} الأيمن عبر الدارتين وعودةً، في
        ثماني محطات مسمّاة.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- ثلاثة أعطال.}
\begin{enumerate}[resume]
  \item المريض الثاني عنده صمام راشح بين \omterm{def:g7:heart-and-circulation:rooms}{الأذين} \omterm{def:g7:heart-and-circulation:rooms}{والبطين}
        الأيسرين: فيندفع بعض \omterm{prop:g4:journey-of-food:absorb}{الدم} إلى الوراء عند كل دقّة، مع
        لغط أزيزيّ. تنبأ بعواقب ذلك على توصيلات الدارة الجسمية
        وعلى قدرة المريض على الاحتمال.
  \item والمريض الثالث مدخّن شره ذو نظام غذائي دسم، وشرايينه
        مكسوّة متصلّبة. فأي قراءات الاستشارة الأولى ستكون قد
        ساءت، ولماذا يعمل \omterm{def:g7:heart-and-circulation:rooms}{البطين} الأيسر ساعات إضافية؟
  \item وكسوة المريض الثالث تشمل \omterm{def:g7:heart-and-circulation:vessels}{شرايين} \omterm{def:g4:heart-and-blood:heart}{القلب} المغذية نفسها.
        اشرح قلق الطبيب الخاص في جملة واحدة.
  \item والمريض الرابع تتورم كاحلاه بعد أيام وقوف طويلة: فعودة
        \omterm{def:g7:heart-and-circulation:vessels}{الأوردة} الصاعدة تكافح. فأي البنى تخفق، ولماذا يخفّف
        المشي --- \omterm{def:g3:bones-and-muscles:muscle}{وعضلات} \omterm{def:g1:plants-around-us:parts}{الساق} تعصر \omterm{def:g7:heart-and-circulation:vessels}{الأوردة} --- من ذلك؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- لوحة الجدار.}
\begin{enumerate}[resume]
  \item لوحة العيادة الجدارية تعرض الحلقة المزدوجة. اشرح لمريض
        ينتظر لماذا يجب أن يزور \omterm{prop:g4:journey-of-food:absorb}{الدم} \omterm{def:g7:how-animals-breathe:four}{الرئتين} بين كل جولتين
        جسميتين.
  \item وأضف مفتاح \omterm{def:g4:heart-and-blood:vessels}{الأوعية}: \omterm{def:g7:heart-and-circulation:vessels}{شريان}، وشعيرة، \omterm{def:g7:heart-and-circulation:vessels}{ووريد} --- سطر واحد
        لكل واحد عن البنية والضغط والعمل.
  \item وذيل اللوحة يقول: ``كل علاج يخدم الشعيرات.'' برّر ذلك
        وفق \cref{rem:g7:heart-and-circulation:capillaries}.
  \item اختم الصباح بوصفة الوقاية التي يسلّمها الطبيب لكل مريض
        --- ثلاث عادات، كل واحدة مربوطة بآليتها في
        \cref{prop:g7:heart-and-circulation:keep}.
\end{enumerate}
\end{problem}
