\section*{Bab \ref{ch:b1:eukaryotic-cell} --- Organisasi Fungsional Sel Eukariot}

\begin{solution}{exo:b1:eukaryotic-cell:1}
\omterm{def:b1:cell-unit-of-life:prokeuk}{Inti} (kromosom, transkripsi); RE kasar (protein sekresi dan protein
membran); RE halus (lipid, kalsium, penawaran racun); Golgi (pemilahan,
pengubahan, pengemasan); \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosom} (pencernaan); \omterm{def:b1:eukaryotic-cell:peroxisome}{peroksisom} (oksidasi,
katalase); \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria} (respirasi, ATP).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:2}
Disintesis oleh ribosom yang terikat pada RE kasar lalu dirangkai masuk
ke lumennya (\omterm{prop:b1:eukaryotic-cell:secretory}{urutan isyarat}); vesikel bertunas dari RE dan melebur
dengan Golgi cis; proteinnya melintasi tumpukan itu, diubah dan
dipilah; ia meninggalkan sisi trans dalam sebuah \omterm{def:b1:eukaryotic-cell:plantcell}{vakuola} pemekat yang
mematang menjadi butiran sekresi; butiran itu melebur dengan membran
apikal pada sebuah isyarat lalu mengosongkan diri ke dalam salurannya.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:3}
$7 + 32 = \qty{39}{\%}$. Membran dalamnya terlipat menjadi \omterm{def:b1:eukaryotic-cell:mitochondrion}{krista}
karena ia membawa rantai respirasi dan ATP sintase, yang keluarannya
sebanding dengan luasnya; membran luarnya hanyalah sebuah selubung.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:4}
Tumbuhan: dinding selulosa, \omterm{def:b1:eukaryotic-cell:plantcell}{vakuola} besar, \omterm{def:b1:eukaryotic-cell:plastid}{plastida} (plasmodesmata
juga). Hewan: \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosom}, sentriol (serta kesanggupan merayap dan
memfagosit).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:5}
RE pada \qty{3}{min} (sudah menurun), Golgi pada sekitar \qty{15}{min},
butiran pada sekitar \qty{60}{min}. Singgah dari RE ke butiran sekitar
\qty{45}{min}; separuhnya tersekresikan pada sekitar \qty{110}{min}.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:6}
Golgi, yang ditahan dekat \omterm{def:b1:eukaryotic-cell:cytoskeleton}{sentrosom} oleh dinein pada \omterm{def:b1:eukaryotic-cell:cytoskeleton}{mikrotubulnya},
terpecah dan menyebar; lalu lintas vesikelnya melambat menjadi difusi
dan pengangkutan ke tepinya berhenti; gelendongnya tidak dapat
terbentuk dan \omterm{def:b1:cell-unit-of-life:cell}{selnya} terhenti pada mitosis; silia, yang terasnya
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{mikrotubul}, berhenti (silia yang ada mantap, tetapi tidak diperbarui).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:7}
Dua membran dengan membran dalam bertipe bakteri; DNA melingkar tanpa
histon; ribosom seukuran ribosom bakteri yang peka terhadap antibiotik;
pembelahan biner; urutan gen yang mengelompok bersama alfaproteobakteri.
Ia tidak sanggup hidup sendirian karena sebagian besar gennya telah
berpindah ke \omterm{def:b1:cell-unit-of-life:prokeuk}{inti}: ia mengimpor sebagian besar proteinnya dari \omterm{def:b1:eukaryotic-cell:organelle}{sitosol}.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:8}
Gulanya ditambahkan pada sisi lumen di dalam RE dan Golgi. Tiap
peleburan vesikel mempertahankan sisinya: sisi lumen sebuah vesikel
menjadi sisi lumen ruang berikutnya, dan ketika vesikel itu melebur
dengan membran plasma lumennya terbuka ke luar, sehingga gulanya
menghadap sisi luar. Apa yang berada di lumen, secara topologis, berada
di luar.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:9}
Tiap butiran $\pi d^2 = \qty{3.14}{\micro m^2}$; tiap hari
\qty{6280}{\micro m^2}, yakni \qty{157}{} kali sisi apikalnya. \omterm{def:b1:cell-unit-of-life:cell}{Sel} itu
harus menarik kembali membrannya lewat \omterm{def:b1:eukaryotic-cell:endocytosis}{endositosis} pada laju yang sama,
sehingga membran apikalnya berputar tiap sembilan menit dan membran
yang ditarik kembali itu dikembalikan ke Golgi untuk dipakai lagi.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:10}
Rantai \qty{320}{} residu itu adalah prazatnya beserta \omterm{prop:b1:eukaryotic-cell:secretory}{urutan isyarat}
\qty{20}{} residunya, yang biasanya terpotong pada saat masuk ke RE.
Dengan vesikelnya, rantai itu masuk ke lumen selama sintesisnya,
isyaratnya terpotong, dan membrannya melindungi hasilnya dari
proteasenya. Mutan yang kehilangan isyaratnya tidak dikenali mesin RE:
ia dibuat pada ribosom bebas dan tinggal dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosol}, dengan panjang
penuh, tak pernah disekresikan.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:11}
Membran yang dimasukkan lewat \omterm{def:b1:eukaryotic-cell:endocytosis}{endositosis} dan \omterm{def:b1:eukaryotic-cell:organelle}{organel} yang aus
diantarkan ke \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosom} untuk dicerna; tanpa enzim itu, lipid yang
dikandungnya tidak dapat diuraikan dan memenuhi \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosomnya}, yang
membengkak sampai \omterm{def:b1:cell-unit-of-life:cell}{selnya} gagal. \omterm{def:b1:cell-unit-of-life:cell}{Sel} yang perputaran membrannya tercepat
(\omterm{def:b1:body-plans-tissues:nervous}{neuron}, yang mendaur ulang vesikel sinapsisnya terus-menerus;
makrofag) menimbunnya paling cepat. Sebuah enzim suntikan yang membawa
isyarat gula yang dibawa enzim \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosom} dapat mengikat reseptor pada
permukaan \omterm{def:b1:cell-unit-of-life:cell}{sel} lalu dimasukkan lewat \omterm{def:b1:eukaryotic-cell:endocytosis}{endositosis} berperantara reseptor
ke dalam \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosomnya} --- asas terapi penggantian enzim.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:12}
Endosimbiosis menerangkan dengan baik \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria} dan \omterm{def:b1:eukaryotic-cell:plastid}{plastida} (dua
membran, genom dan ribosom sendiri, pembelahan biner, kerabat bakteri),
dan karenanya metabolisme energi eukariot. Ia tidak menerangkan \omterm{def:b1:cell-unit-of-life:prokeuk}{inti},
RE dan Golgi, maupun \omterm{def:b1:eukaryotic-cell:cytoskeleton}{rangka sel}, yang tidak memiliki padanan bakteri
dan tampaknya merupakan penemuan garis keturunan inangnya --- lipatan
membrannya ke dalam dan protein baru. ``Federasi'' tepat bagi
\omterm{def:b1:eukaryotic-cell:organelle}{organelnya}; ``disatukan sebuah sistem membran'' menunjuk pada bagian
yang lebih besar dan belum terjelaskan: sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} inang yang sudah
bersifat eukariot dalam arsitekturnya sebelum ia menerima bakteri itu
masuk.
\end{solution}

\begin{solution}{pb:b1:eukaryotic-cell:1}
\textbf{1.} Denyut yang singkat menandai satu rombongan molekul yang
sempit sehingga letaknya menandai satu waktu; asam amino dingin yang
berlebih menghentikan penyisipan lebih lanjut sehingga rombongan itu
tidak dikaburkan sintesis yang belakangan.
\textbf{2.} RE: \qty{3}{min} (titik paling awal); Golgi dan \omterm{def:b1:eukaryotic-cell:plantcell}{vakuola}
pemekat: sekitar \qty{15}{min}; butiran: sekitar \qty{60}{min}.
\textbf{3.} RE ke Golgi sekitar \qty{10}{min}; Golgi ke butiran sekitar
\qty{45}{min}.
\textbf{4.} Sebagian besar dalam butirannya (\qty{70}{\%}),
sepersepuluh dalam Golgi, sepersepuluh sudah tersekresikan: lintasan
konstitutif dan butiran pertama yang terbentuk melepas sebagian,
sedangkan sebagian besarnya menunggu sebuah rangsangan.
\textbf{5.} Pertunasan dan pengangkutan vesikel menuntut energi dan
membran yang cair; dalam dingin langkah dari RE ke Golgi berhenti dan
proteinnya menumpuk di hulu hambatannya.
\textbf{6.} $10\,\mathrm{g}/\num{8e10} = \qty{1.25e-10}{g} =
\qty{125}{pg}$ tiap \omterm{def:b1:cell-unit-of-life:cell}{sel} tiap hari.
\textbf{7.} $V = \frac{\pi}{6}(1)^3 = \qty{0.52}{\micro m^3} =
\qty{5.2e-13}{mL}$; proteinnya $\qty{1.05e-13}{g} = \qty{0.1}{pg}$.
\textbf{8.} $125/0.1 = 1250$ butiran tiap hari, $0.87$ tiap menit.
\textbf{9.} $V = 12^3 = \qty{1728}{\micro m^3} = \qty{1.73e-9}{cm^3}$;
massanya \qty{1.8e-9}{g} = \qty{1800}{pg}. Ia menyekresi
$125/1800 = 7\%$ massanya tiap hari (seluruh \omterm{def:b1:cell-unit-of-life:cell}{selnya} hanya \qty{20}{\%}
protein, jadi sepertiga kandungan proteinnya).
\textbf{10.} Massa molarnya $250\times 110 = \qty{27500}{g/mol}$;
$\qty{1.05e-13}{g}/27500\times\num{6e23} = \num{2.3e6}$ molekul.
\textbf{11.} $250/5 = \qty{50}{s}$.
\textbf{12.} Molekul tiap hari $1250\times\num{2.3e6} = \num{2.9e9}$;
satu ribosom membuat $86\,400/50 = 1728$ tiap hari; yang dibutuhkan:
$\num{1.7e6}$ ribosom, sekitar \qty{40}{\%} milik \omterm{def:b1:cell-unit-of-life:cell}{sel} itu.
\textbf{13.} $\pi d^2 = \qty{3.14}{\micro m^2}$.
\textbf{14.} $1250\times 3.14 = \qty{3900}{\micro m^2}$.
\textbf{15.} $3900/144 = 27$ kali: \omterm{def:b1:cell-unit-of-life:cell}{sel} itu menarik kembali luas yang
sama lewat \omterm{def:b1:eukaryotic-cell:endocytosis}{endositosis} dan mengembalikannya ke Golgi.
\textbf{16.} $20\,000/3900 = 5$ hari: RE-nya hanya sanggup memasok
membran butiran lima hari seandainya tak ada yang kembali; membrannya
harus didaur ulang terus-menerus.
\textbf{17.} Tiap mikrovilus $\pi\times 0.1\times 1 =
\qty{0.31}{\micro m^2}$; \qty{630}{\micro m^2} bagi 2000; faktor $5.4$.
\textbf{18.} Pada awal rantainya: \omterm{prop:b1:eukaryotic-cell:secretory}{urutan isyaratnya}, dua puluh residu
yang sebagian besar hidrofobik.
\textbf{19.} Isyarat itu mengarahkan rantai yang sedang tumbuh masuk ke
vesikelnya, tempat sebuah peptidase isyarat menyingkirkannya;
membrannya melindungi proteinnya dari proteasenya.
\textbf{20.} Hanya protein yang berisyarat yang masuk ke RE; sebuah
protein \omterm{def:b1:eukaryotic-cell:organelle}{sitosol} tidak memilikinya dan tinggal di luar.
\textbf{21.} Enzim itu dibuat pada ribosom bebas dan tinggal dalam
\omterm{def:b1:eukaryotic-cell:organelle}{sitosol}; karena ia enzim pencernaan (sebagai prazat tak aktif ia sedikit
merugikan, tetapi bila teraktifkan) ia akan mencerna \omterm{def:b1:cell-unit-of-life:cell}{selnya} dari dalam.
\textbf{22.} Di dalam Golgi trans: enzim \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosom} itu membawa tanda
pembeda (sebuah gula terfosforilasi) yang dikenali sebuah reseptor yang
mengemasnya ke dalam vesikel menuju \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosom} alih-alih ke dalam butiran
sekresi.
\textbf{23.} Di dalam RE (dan Golgi): pembentukan vesikel, pengangkutan
motor sepanjang \omterm{def:b1:eukaryotic-cell:cytoskeleton}{mikrotubul} dan peleburan membran semuanya memakai ATP;
lagi pula sintesis proteinnya sendiri berhenti, sehingga denyutnya
bahkan tidak selesai.
\textbf{24.} Singgah \qty{45}{min} pada $0.87$ butiran tiap menit:
sekitar \qty{40}{} butiran senilai protein sedang dalam perjalanan.
\textbf{25.} Singgah dari sintesis sampai butiran sekitar \qty{45}{min}
(\qty{55}{min} bila RE-nya dihitung), sekresi sesudah sekitar dua jam
dalam percobaan itu (empat jam atau lebih dalam kelenjar yang hidup,
sebab butirannya menunggu sebuah santapan); \qty{0.9}{} butiran dilepas
tiap menit; \qty{7}{\%} massa \omterm{def:b1:cell-unit-of-life:cell}{selnya} tiap hari.
\end{solution}
