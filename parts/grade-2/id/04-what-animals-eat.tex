\chapter{Apa yang Dimakan Hewan}\label{ch:g2:what-animals-eat}

Kelinci di padang rumput mengunyah rumput sepanjang hari. Rubah yang
melintasi padang rumput yang sama pada waktu senja sama sekali tidak
tertarik pada rumput --- ia tertarik pada kelincinya. Dan burung hitam
mengambil cacing pada pagi hari dan buah ceri pada sore hari. Setiap
\omterm{def:g1:animals-around-us:animal}{hewan} harus makan, tetapi tidak setiap \omterm{def:g1:animals-around-us:animal}{hewan} memakan hal yang sama.

\section{Tiga macam pemakan}

\begin{definition}[Herbivora, karnivora, omnivora]\label{def:g2:what-animals-eat:kinds}
\omterm{def:g1:animals-around-us:animal}{Hewan} dikelompokkan menurut apa yang dimakannya:
\begin{itemize}
  \item seekor \emph{herbivora}\index{herbivora} memakan \omterm{def:g1:plants-around-us:plant}{tumbuhan}:
        rumput, \omterm{def:g1:plants-around-us:parts}{daun}, \omterm{def:g2:from-seed-to-plant:seed}{biji}, buah --- kelinci, sapi, siput;
  \item seekor \emph{karnivora}\index{karnivora} memakan \omterm{def:g1:animals-around-us:animal}{hewan} lain:
        rubah, kucing, kepik (yang memburu kutu \omterm{def:g1:plants-around-us:parts}{daun} yang mungil);
  \item seekor \emph{omnivora}\index{omnivora} memakan \omterm{def:g1:plants-around-us:plant}{tumbuhan}
        sekaligus \omterm{def:g1:animals-around-us:animal}{hewan}: burung hitam, babi hutan --- dan manusia.
\end{itemize}
\end{definition}

\begin{example}[Satu padang rumput, tiga daftar makanan]\label{ex:g2:what-animals-eat:meadow}
Di satu padang rumput musim panas: kelinci memangkas rumputnya
(\omterm{def:g2:what-animals-eat:kinds}{herbivora}); rubah memburu kelincinya (\omterm{def:g2:what-animals-eat:kinds}{karnivora}); burung hitam
mengambil seekor cacing sekarang, sebutir ceri kemudian (\omterm{def:g2:what-animals-eat:kinds}{omnivora}).
Tiga tetangga, tiga daftar makanan --- dan perhatikan bahwa santapan
rubahnya lebih dahulu memakan rumput.
\end{example}

\begin{remark}[Karnivora bukan penjahat]\label{rem:g2:what-animals-eat:fox}
Rubah tidak ``jahat'' karena memakan kelinci: berburu memang cara
seekor \omterm{def:g2:what-animals-eat:kinds}{karnivora} makan --- sewajar kunyahan kelinci. Di alam, memakan
dan dimakan adalah bagian dari cara kehidupan bekerja; sebuah bab
berikutnya akan merangkai semua santapan ini menjadi rantai makanan.
\end{remark}

\section{Membaca daftar makanan hewan dari tubuhnya}

\begin{proposition}[Tubuh sepadan dengan makanannya]\label{prop:g2:what-animals-eat:match}
Tubuh seekor \omterm{def:g1:animals-around-us:animal}{hewan} membawa petunjuk tentang makanannya:
\begin{itemize}
  \item \omterm{def:g2:what-animals-eat:kinds}{herbivora} seperti kelinci dan sapi mempunyai gigi geraham yang
        besar dan rata untuk menggilas \omterm{def:g1:plants-around-us:plant}{tumbuhan}, dan menghabiskan
        berjam-jam untuk makan;
  \item \omterm{def:g2:what-animals-eat:kinds}{karnivora} seperti kucing dan rubah mempunyai gigi runcing
        untuk menerkam mangsa, cakar yang tajam, dan \omterm{def:g1:the-five-senses:sense}{mata} yang
        menghadap ke depan untuk berburu;
  \item burung memperlihatkannya pada paruhnya: pendek dan tebal untuk
        memecah \omterm{def:g2:from-seed-to-plant:seed}{biji} (burung pipit), melengkung dan berkait untuk
        merobek daging (elang), panjang dan tipis untuk mencungkil
        serangga dari kulit kayu.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Gigi kucing]\label{ex:g2:what-animals-eat:cat}
Amatilah seekor kucing menguap: gigi runcing yang panjang di sudut
depan mulutnya. Itulah gigi penerkam --- perlengkapan seorang
\omterm{def:g2:what-animals-eat:kinds}{karnivora}. Sapi, kalau menguap, tidak akan memperlihatkan gigi
semacam itu: mulutnya penuh gigi penggilas rumput yang lebar dan rata.
\end{example}

\begin{method}[Menerka daftar makanan]\label{met:g2:what-animals-eat:guess}
Untuk menerka apa yang dimakan \omterm{def:g1:animals-around-us:animal}{hewan} yang belum dikenal:
\begin{enumerate}
  \item amati gigi atau paruhnya: penggilas yang rata, penerkam yang
        runcing, atau keduanya?
  \item amati cakar dan \omterm{def:g1:the-five-senses:sense}{matanya}: perkakas berburu, atau bukan?
  \item amati ia makan kalau kamu bisa --- petunjuk yang paling pasti;
  \item lalu sebutkan namanya: \omterm{def:g2:what-animals-eat:kinds}{herbivora}, \omterm{def:g2:what-animals-eat:kinds}{karnivora} atau \omterm{def:g2:what-animals-eat:kinds}{omnivora}.
\end{enumerate}
\end{method}

\section{Makan cukup, hari demi hari}

\begin{proposition}[Makan adalah tugas harian]\label{prop:g2:what-animals-eat:daily}
Makanan adalah bahan bakar sekaligus bahan bangunan bagi tubuh, jadi
\omterm{def:g1:animals-around-us:animal}{hewan} harus makan lagi dan lagi, hari demi hari. Berapa lama waktunya
bergantung pada daftar makanannya: rumput memberi sedikit pada tiap
suapan, jadi sapi merumput berjam-jam; daging itu kaya, jadi rubah
mungkin makan sekali dengan baik lalu beristirahat.
\end{proposition}
\admitted

\begin{example}[Musim dingin mengubah daftar makanan]\label{ex:g2:what-animals-eat:winter}
Makanan berubah bersama musimnya. Burung hitam memakan cacing dan
serangga pada musim panas, tetapi pada musim dingin, ketika tanahnya
keras, ia beralih ke buah beri. Daftar makanan seorang \omterm{def:g2:what-animals-eat:kinds}{omnivora} yang
lebar sangat menolong ketika keadaan menjadi sulit.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:1}
Sebutkan kata untuk: \omterm{def:g1:animals-around-us:animal}{hewan} yang memakan \omterm{def:g1:plants-around-us:plant}{tumbuhan}; \omterm{def:g1:animals-around-us:animal}{hewan} yang memakan
\omterm{def:g1:animals-around-us:animal}{hewan} lain; \omterm{def:g1:animals-around-us:animal}{hewan} yang memakan keduanya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:2}
\omterm{def:g2:what-animals-eat:kinds}{Herbivora}, \omterm{def:g2:what-animals-eat:kinds}{karnivora} atau \omterm{def:g2:what-animals-eat:kinds}{omnivora}: (a) sapi; (b) rubah; (c) burung
hitam; (d) siput; (e) manusia?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:3}
Gigi mana yang paling dibutuhkan seekor \omterm{def:g2:what-animals-eat:kinds}{herbivora}: penggilas yang
rata, atau penerkam yang runcing? Mengapa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:4}
Pasangkan tiap paruh dengan daftar makanannya: pendek dan tebal;
melengkung dan berkait; panjang dan tipis.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:5}
Kepik itu kecil, elok --- dan seorang \omterm{def:g2:what-animals-eat:kinds}{karnivora}. Apa yang diburunya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:6}
Mengapa sapi menghabiskan waktu makan jauh lebih lama daripada rubah?
Pakai \cref{prop:g2:what-animals-eat:daily}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:7}
Apa yang dimakan burung hitam pada musim panas, dan pada musim dingin?
Mengapa daftar makanan yang lebar berguna?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:8}
Cobalah: amati burung di taman atau di tempat pakan. Catat apa yang
diambil tiap jenisnya --- \omterm{def:g2:from-seed-to-plant:seed}{biji}, roti, serangga, buah beri --- dan
amati paruhnya. Apakah \cref{prop:g2:what-animals-eat:match} berlaku?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:9}
Sebuah kebun binatang menerima seekor \omterm{def:g1:animals-around-us:animal}{hewan} misteri: gigi depan yang
runcing, cakar yang tajam, \omterm{def:g1:the-five-senses:sense}{mata} yang menghadap ke depan. Apa daftar
makanan yang mungkin baginya? Langkah mana saja dari
\cref{met:g2:what-animals-eat:guess} yang kamu pakai?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:10}
``Tanpa rumput, tak lama lagi tidak akan ada rubah.'' Rubah tidak
pernah memakan rumput --- namun kalimat itu benar. Jelaskan, dengan
memakai padang rumput \cref{ex:g2:what-animals-eat:meadow}.
\end{exercise}
