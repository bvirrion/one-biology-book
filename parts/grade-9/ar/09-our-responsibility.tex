\chapter{مسؤوليتنا}\label{ch:g9:our-responsibility}

تسع سنوات من هذا الكتاب تنتهي هنا، وخيوطه تلتقي في حقيقة
بسيطة: \omterm{def:g5:biodiversity:species}{النوع} الذي يستطيع قراءة هذه الصفحة هو \omterm{def:g5:biodiversity:species}{النوع}
الوحيد الذي يفهم الآلة التي ينتمي إليها --- والفهم، كما
قال \cref{ch:g8:contraception} أول مرة، يجلب الاختيار.
وهذا الفصل الختامي يجمع المسؤوليات التي ينشئها الفهم:
تجاه صحة المرء نفسه، وتجاه صحة الآخرين، وتجاه شبكة
\omterm{def:g6:exploring-our-environment:components}{الكائنات الحية} التي تحملنا جميعًا.

\section{المسؤولية عن آلة المرء نفسه}

\begin{proposition}[الصحة إدارة ذاتية مستنيرة]\label{prop:g9:our-responsibility:self}
خيوط الصحة في الكتاب، معقودة:
\begin{itemize}
  \item العقد اليومي
        (\cref{prop:g5:healthy-choices:contract}) ---
        الغذاء، والحركة، \omterm{prop:g1:healthy-habits:needs}{والنوم}، \omterm{def:g1:healthy-habits:hygiene}{والنظافة} --- صيانة
        أعضاء تستطيع الآن تبريرها جهازًا جهازًا
        (\cref{prop:g7:heart-and-circulation:keep}،
        \cref{met:g7:lungs-and-gas-exchange:keep}،
        \cref{met:g8:protecting-the-nervous-system:manual})؛
  \item وآليات المواد معروفة
        (\cref{prop:g8:protecting-the-nervous-system:substances})،
        وتباين الفخ مسعَّر
        (\cref{rem:g5:healthy-choices:never})؛
  \item ولاختيارات التكاثر خريطتها وعناوينها
        (\cref{ch:g8:contraception})؛
  \item وتاريخ الجسم موزَّع في بعضه
        (\cref{ch:g9:heredity-and-traits}): فمعرفة نزعات
        العائلة --- الشيب المبكر، والقلوب المتكلسة ---
        ليست قدرًا بل جدول صيانة
        (\cref{met:g9:unique-individuals:claims}: مديات،
        يعاش فيها).
\end{itemize}
\end{proposition}
\admitted

\begin{example}[الفحص الدوري، مقروءًا بالبيولوجيا]\label{ex:g9:our-responsibility:checkup}
الفحص الدوري هو هذا الكتاب في موعد واحد: \omterm{prop:g4:heart-and-blood:pulse}{النبض} والضغط
(الشبكة مقروءة على قارعة الطريق،
\cref{ex:g7:heart-and-circulation:measures})، ومطرقة
المنعكس (\cref{pb:g8:nervous-communication:1})، وسجل
\omterm{prop:g9:immune-defenses:vaccination}{التلقيح} (\cref{pb:g9:immune-defenses:1})، وتاريخ العائلة
(النزعات الموزَّعة)، والنصيحة القائمة --- العقد، من
جديد. فالطب في أكثره صيانة تطبيقية لآلات صرت تملك
أدلتها.
\end{example}

\section{المسؤولية تجاه الآخرين}

\begin{proposition}[الصحة آلة مشتركة]\label{prop:g9:our-responsibility:others}
ثلاث آليات تجعل الصحة حسابًا مشتركًا:
\begin{itemize}
  \item \emph{الانتقال}
        (\cref{def:g9:microbes-and-infection:stages}):
        فنظافتك، وتغطيتك لسعلتك، ويوم مرضك في البيت هي
        حماية للآخرين؛
  \item \emph{\omterm{def:g9:immune-defenses:memory}{المناعة} المتجمعة}
        (\cref{prop:g9:immune-defenses:vaccination}):
        \omterm{prop:g9:immune-defenses:vaccination}{فالتلقيح} يحرس، من خلالك، من لا يمكن تدريبهم؛
  \item \emph{مشاع الطب}: فالمضادات الحيوية وقد أبقيت
        عاملة (\cref{exo:g9:how-species-change:10})،
        \omterm{prop:g4:journey-of-food:absorb}{والدم} وقد تبرع به
        (\cref{ex:g9:unique-individuals:medicine} ---
        فالسجل يجري على المتبرعين) موارد مشتركة تنفقها
        اختيارات الأفراد أو تخزنها.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[حساب مراهق واحد المشترك]\label{ex:g9:our-responsibility:account}
قيود الأسبوع العادية: بقي في البيت محمومًا (الانتقال
مقطوع)؛ وجرعاته المعزّزة سارية (الحوض ممسوك)؛ وجرعة مضاد
حيوي أتمّها في الربيع (المشاع غير منفق)؛ وأول تبرع \omterm{prop:g4:journey-of-food:absorb}{بالدم}
مقرر عند الثامنة عشرة (السجل مخزون). ولا شيء من هذا
بطولي --- درس \cref{ex:g5:healthy-choices:day} على مقياس
بلدة: فالصحة العامة أيام عادية، محفوظة.
\end{example}

\section{المسؤولية تجاه العالم الحي}

\begin{proposition}[الرعاية، معادة الصياغة بكل أسبابها]\label{prop:g9:our-responsibility:stewardship}
خيوط الطبيعة، معقودة بآلياتها:
\begin{itemize}
  \item كل \omterm{def:g3:food-chains:chain}{سلسلة غذائية} تبدأ خضراء
        (\cref{prop:g6:plants-are-producers:firstlink})،
        وكل مائدة ناتج نظام بيئي
        (\cref{prop:g6:producing-our-food:costs})؛
  \item والشبكات تمتص الصدمات بالتنوع
        (\cref{prop:g5:biodiversity:web})، والتنوع هو
        الاحتياطي الذي يجري عليه الانتقاء
        (\cref{prop:g9:unique-individuals:reserve}،
        \cref{exo:g9:how-species-change:12}) ---
        \omterm{def:g5:biodiversity:biodiversity}{فالتنوع الحيوي} ليس منظرًا بل رأس مال عامل؛
  \item والانقراض يغلق الباب إلى الأبد
        (\cref{prop:g5:biodiversity:loss}) ---
        والاستثناء المتعمد الوحيد، الجدري، يثبت كم
        الكلمة مطلقة
        (\cref{exo:g9:immune-defenses:12})؛
  \item وقواعد الاستعمال الدائم معروفة
        (\cref{prop:g5:people-and-nature:lasting})،
        ومخالفاتها قابلة للتشخيص
        (\cref{ex:g7:oxygen-in-the-water:waste})،
        وإصلاحاتها مبرهنة
        (\cref{ex:g5:people-and-nature:river}،
        \cref{ex:g5:biodiversity:storks}).
\end{itemize}
والرعاية هي عقلية الصيانة نفسها التي في الصحة، مطبقة على
الجسم الأكبر الذي نعيش داخله.
\end{proposition}
\admitted

\begin{example}[ما فائدة بيولوجيا المواطن]\label{ex:g9:our-responsibility:citizen}
أصوات حياة الكبار ومشترياتها وعاداتها تظل تصل ملبوسة
أسئلة جهّز هذا الكتاب لها: أهذا المصيد يأخذ أسرع من
التجدد؟ أتحمي هذه السياسة \omterm{def:g2:where-animals-live:habitat}{موئل} المؤبِّرات أم ترصفه؟ أهذه
الحمية أو الشفاء أو ``مورّثة كذا'' المعجزة المدَّعاة
متماسكة (\cref{met:g9:unique-individuals:claims}،
\cref{met:g8:contraception:evaluate})؟ أينبغي أن يكون
هذا المضاد الحيوي في هذا المعلف؟ فالمواطن الذي يستطيع
إجراء هذه التدقيقات هو غاية تسع سنوات من الفصول.
\end{example}

\begin{method}[منهج المتخرج]\label{met:g9:our-responsibility:method}
وأنت تغادر الكتاب الأول، احمل المنهج أكثر من الوقائع:
\begin{enumerate}
  \item ارصد قبل أن تستنتج
        (\cref{met:g1:animals-around-us:observe} بدأها؛
        وكل جرد ودفتر منذ ذلك الحين)؛
  \item واختبر اختبارًا عادلًا --- فرق واحد، وشاهد
        (\cref{met:g4:what-plants-need:fairtest}،
        مستعمل في كل سنة بعده)؛
  \item وسمّ الآلية --- فلا ادعاء يدقَّق حتى تسمى حلقته
        (\cref{met:g8:contraception:evaluate})؛
  \item وزن الشهود المستقلين
        (\cref{met:g9:evidence-of-evolution:weigh})؛
  \item واحترم الآلة الحية، في جسدك وخارجه: فهي قديمة،
        ولطيفة، وقابلة للإصلاح في حدود --- وهي أمانتك.
\end{enumerate}
\end{method}

\begin{remark}[ما يأتي بعد]\label{rem:g9:our-responsibility:next}
يتناول مجلد الثانوية هذه الخيوط عند التكبير التالي: آلة
\omterm{def:g5:first-look-at-cells:cell}{الخلية} الجزيئية، \omterm{def:g9:heredity-and-traits:heredity}{والوراثة} برياضياتها الكاملة، والجهازان
المناعي والعصبي على الجد، والإيكولوجيا علمًا كميًا،
والتطور إطارًا كما هو. والأسئلة هي هي، مسؤولة من أقرب.
فاحمل المنهج.
\end{remark}

\section{مسألة: تدقيق التخرج}

\begin{problem}[{مسألة نهاية الأسبوع --- سنة بلدة واحدة،
مدققة بالكتاب كله}]\label{pb:g9:our-responsibility:1}
سنة بلدتك (المتخيَّلة) في العناوين: موجة أنفلونزا في
الشتاء؛ وجدل في الربيع حول رش أشجار الدلب؛ ونفوق أسماك
في بركة الحديقة في الصيف؛ وحملة في الخريف من أجل محمية
المستنقع الجديدة؛ وعلى مدار السنة، حملات العيادة للتلقيح
والتبرع.

\medskip\noindent\textbf{الجزء الأول --- ملف الصحة.}
\begin{enumerate}
  \item موجة الأنفلونزا: سمّ طرق الانتقال التي صوّبت
        نصيحة البلدة إليها، ولماذا لقّحت العيادة عاملي
        دور الرعاية أولًا.
  \item الحملات: اذكر ما الذي يخزنه سجل \omterm{prop:g4:journey-of-food:absorb}{الدم} وسجل
        \omterm{prop:g9:immune-defenses:vaccination}{التلقيح}، ومن يسحب منهما.
  \item يسخر كاتب عمود من ``الضجة حول غسل الأيدي''. أجب
        بجناح
        \cref{ex:g9:microbes-and-infection:history} ---
        في فقرة واحدة.
  \item تبلّغ الصيدلية عن جرعات مضادات حيوية غير متمَّمة
        تعاد إليها. أي مشاع ينفق، وبأي آلية؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- ملف الطبيعة.}
\begin{enumerate}[resume]
  \item جدل الرش: أجر تدقيق البستان
        (\cref{exo:g9:our-responsibility:9}) على ``اقتل
        كل حشرة'' --- من يموت أيضًا، وأي خدمات مجانية
        تتوقف؟
  \item نفوق \omterm{prop:g3:sorting-living-things:groups}{الأسماك}: أسبوع حار ساكن، وبركة خضراء
        كالبازلاء. شخّص بمساعدة
        \cref{ch:g7:oxygen-in-the-water}، وصف العلاج.
  \item حملة المستنقع: دافع عن المحمية بثلاث آليات ---
        \omterm{def:g2:where-animals-live:habitat}{الموئل}، والشبكة، واحتياطي التنوع --- بجملة لكل
        واحدة.
  \item يسأل مالك أرض: ``وماذا فعل المستنقع للبلدة
        يومًا؟'' أجب بسجل العمل المجاني
        (\cref{exo:g5:biodiversity:11}، وتخزين الفيضان
        في نهر \cref{ex:g5:people-and-nature:river}).
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- ختام
المتخرج.}
\begin{enumerate}[resume]
  \item اختر أفضل قرارات السنة وأسوأها بقواعد الاستعمال
        الدائم، وبرر كل حكم.
  \item بيّن خطوات المنهج الخمس في عملها في كيفية حل
        مسألة نفوق \omterm{prop:g3:sorting-living-things:groups}{الأسماك} --- من أول رصد إلى آلية
        مقبولة.
  \item اكتب مسودة افتتاحية جريدة البلدة لآخر السنة في
        أربع جمل: صحة مشتركة، وطبيعة مودعة، وادعاءات
        مدققة، ومنهج محفوظ.
  \item والسطر الأخير من الصفحة الأخيرة في الكتاب الأول
        لك أن تكتبه: جملة واحدة، موجهة إلى القارئ الذي
        بدأ عند حلزون \cref{ch:g1:living-or-not}.
\end{enumerate}
\end{problem}
