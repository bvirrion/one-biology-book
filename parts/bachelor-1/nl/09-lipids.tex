\chapter{Lipiden}\label{ch:b1:lipids}

Een kameel die een woestijn doorkruist draagt dertig kilogram \omterm{def:b1:lipids:triglyceride}{vet} in zijn bult en
daar helemaal geen water; een forel in een bergbeek van \qty{4}{\celsius} heeft
membranen die even vloeibaar zijn als die van een karper in een warme vijver; en
een druppel olijfolie die door water wordt geschud valt uiteen in een wolk
druppeltjes die, met rust gelaten, weer tot één geheel samenvloeien. Alle drie
zijn \omterm{def:b1:lipids:fattyacid}{lipiden} aan het werk: de dichtste voorraad chemische energie die een
\omterm{def:b1:organism-environment:organism}{organisme} kan meedragen, de film die elke \omterm{def:b1:cell-unit-of-life:cell}{cel} en elk organel begrenst, en de
klasse moleculen die water weigert op te lossen. Dit hoofdstuk beschrijft
\omterm{def:b1:lipids:fattyacid}{vetzuren} en de \omterm{def:b1:lipids:triglyceride}{vetten} die eruit worden gebouwd, de amfifiele \omterm{def:b1:lipids:fattyacid}{lipiden} die zich tot
membranen samenvoegen, wat een membraan meer of minder vloeibaar maakt, en de
overige \omterm{def:b1:lipids:fattyacid}{lipiden} --- \omterm{def:b1:lipids:amphiphilic}{sterolen}, kleurstoffen, hormonen --- die enkele koolstoffen
in ringen en ketens kunnen zijn.

\section{Vetzuren}

\begin{definition}[Lipide, vetzuur]\label{def:b1:lipids:fattyacid}
\emph{Lipiden}\index{lipide} zijn de biologische moleculen die niet door een
gemeenschappelijke bouw maar door een gemeenschappelijke eigenschap worden
omschreven: zij zijn onoplosbaar in water en oplosbaar in apolaire
oplosmiddelen. De meeste zijn op \emph{vetzuren}\index{vetzuur} gebouwd:
carbonzuren met een onvertakte koolwaterstofketen van \qtyrange{12}{24}{}
koolstoffen, vrijwel altijd een even aantal (zij worden telkens twee koolstoffen
tegelijk opgebouwd, \cref{ch:b1:biosyntheses-integration}). Een
\emph{verzadigd}\index{verzadigd vetzuur} vetzuur heeft alleen enkelvoudige
bindingen en een rechte, buigzame keten; een \emph{onverzadigd}\index{onverzadigd vetzuur} vetzuur heeft een of meer dubbele bindingen, in de \emph{cis}-stand, die
elk een stijve knik van ongeveer $30^\circ$ in de keten leggen. Notatie: C18:1
$\Delta^9$ is een \omterm{def:b1:water-small-molecules:ph}{zuur} met 18 koolstoffen en één dubbele binding na koolstof 9
(oliezuur); een \emph{omega-3}-zuur heeft zijn laatste dubbele binding drie
koolstoffen van het methyluiteinde.
\end{definition}

\begin{proposition}[Smeltpunten]\label{prop:b1:lipids:melting}
Het smeltpunt van een \omterm{def:b1:lipids:fattyacid}{vetzuur}, en de \omterm{def:b1:lipids:fluidity}{vloeibaarheid} van elk lipidegeheel waarvan
het deel uitmaakt, wordt bepaald door hoe dicht de ketens zich kunnen pakken: het
stijgt met de ketenlengte (\qty{44}{\celsius} voor C12:0, \qty{63}{\celsius} voor
C16:0, \qty{70}{\celsius} voor C18:0) en daalt scherp met elke
\emph{cis}-dubbele binding (C18:0 \qty{70}{\celsius}, C18:1 \qty{13}{\celsius},
C18:2 \qty{-5}{\celsius}, C18:3 \qty{-11}{\celsius}). Dierlijke \omterm{def:b1:lipids:triglyceride}{vetten}, rijk aan
verzadigde zuren, zijn bij kamertemperatuur vast; plantaardige en visoliën, rijk
aan onverzadigde, zijn vloeibaar.
\end{proposition}
\begin{proof}
Rechte ketens liggen zij aan zij en elke $\mathrm{CH_2}$ maakt
vanderwaalscontacten met haar buren, die elk enkele kilojoules per mol waard zijn,
opgeteld langs de keten: meer koolstoffen, meer contacten, meer warmte om ze te
scheiden. Een knik verhindert nauw contact over verscheidene koolstoffen aan
weerszijden en neemt in één keer vele contacten weg.
\end{proof}

\section{Vetten: de energievoorraad}

\begin{definition}[Triglyceride]\label{def:b1:lipids:triglyceride}
Een \emph{triglyceride}\index{triglyceride} (triacylglycerol, een
\emph{vet}\index{vet} wanneer het vast is, een \emph{olie}\index{olie} wanneer het
vloeibaar is) is glycerol --- een alcohol met drie koolstoffen --- veresterd op
zijn drie hydroxylgroepen met drie \omterm{def:b1:lipids:fattyacid}{vetzuren}. Het heeft geen lading en geen polaire
groep meer over: het is volledig hydrofoob, en het wordt als watervrije druppels
opgeslagen in \emph{adipocyten}\index{adipocyt}, \omterm{def:b1:cell-unit-of-life:cell}{cellen} van het vetweefsel die
weinig meer zijn dan een druppel vet met een kern die opzij is gedrukt.
\end{definition}

\begin{proposition}[Vet is de dichtste energievoorraad]\label{prop:b1:lipids:energystore}
Het oxideren van \qty{1}{g} \omterm{def:b1:lipids:triglyceride}{vet} levert \qty{38}{kJ}, tegen \qty{17}{kJ} voor
\qty{1}{g} koolhydraat of eiwit: de koolstoffen van een \omterm{def:b1:lipids:fattyacid}{vetzuur} zijn sterker
gereduceerd (meer C--H-bindingen, minder C--O) en staan dus meer elektronen aan
zuurstof af. Bovendien wordt \omterm{def:b1:lipids:triglyceride}{vet} droog opgeslagen, terwijl glycogeen ongeveer
\qty{3}{g} water per gram bindt: \qty{1}{g} opgeslagen glycogeen levert
\qty{4}{kJ} per gram opgeslagen massa, \omterm{def:b1:lipids:triglyceride}{vet} negen keer meer. Een slanke mens draagt
\qty{0.5}{kg} glycogeen mee (de energie voor een dag) en \qty{12}{kg} \omterm{def:b1:lipids:triglyceride}{vet} (voor
meer dan een maand).
\end{proposition}

\begin{example}[Waarom niet alles als vet opslaan]\label{ex:b1:lipids:whyglycogen}
\omterm{def:b1:lipids:triglyceride}{Vet} kan alleen met zuurstof worden verbrand, langzaam, en kan bij dieren niet
worden teruggezet in glucose (\cref{ch:b1:biosyntheses-integration}); de hersenen
en de rode \omterm{def:b1:cell-unit-of-life:cell}{cellen} hebben glucose nodig, en een sprintende spier heeft haar sneller
nodig dan \omterm{def:b1:lipids:triglyceride}{vet} haar kan leveren. Glycogeen is de snelle, zuurstofvrije
glucosevoorraad voor uren; \omterm{def:b1:lipids:triglyceride}{vet} de dichte voorraad voor weken. Een trekvogel, een
winterslapende zevenslaper en een kameel zijn \omterm{def:b1:lipids:triglyceride}{vet}; de benen van een sprinter zijn
glycogeen.
\end{example}

\begin{definition}[Wassen]\label{def:b1:lipids:wax}
\emph{Wassen}\index{was} zijn esters van een \omterm{def:b1:lipids:fattyacid}{vetzuur} met een alcohol met een lange
keten: vast, volledig hydrofoob, en gebruikt als bekleding --- de \omterm{def:b1:flowering-plant-organization:tissues}{cuticula} van
\omterm{def:b1:flowering-plant-organization:organs}{bladeren} (\cref{ch:b1:flowering-plant-organization}), het oppervlak van insecten,
de waterafstotende laag van veren en vacht, de raat van bijen.
\end{definition}

\section{Membraanlipiden en zelfassemblage}

\begin{definition}[Amfifiele lipiden]\label{def:b1:lipids:amphiphilic}
De \omterm{def:b1:lipids:fattyacid}{lipiden} van membranen zijn \emph{amfifiel}\index{amfifiel molecuul}: een
polaire \emph{kop} die water solvateert en een of twee apolaire \emph{staarten}
die het buitensluit. \emph{Glycerofosfolipiden}\index{fosfolipide} zijn glycerol
met twee \omterm{def:b1:lipids:fattyacid}{vetzuren} en, op de derde koolstof, een fosfaat dat aan een kleine
polaire alcohol is gekoppeld (choline, ethanolamine, serine, inositol):
fosfatidylcholine is het meest voorkomende \omterm{def:b1:lipids:fattyacid}{lipide} van dierlijke membranen.
\emph{Sfingolipiden}\index{sfingolipide} zijn op de aminoalcohol sfingosine
gebouwd in plaats van op glycerol, met één \omterm{def:b1:lipids:fattyacid}{vetzuur} en een kop van fosfocholine
(sfingomyeline) of van suikers (glycolipiden), en zijn verrijkt in het
buitenste \omterm{def:b1:flowering-plant-organization:organs}{blad} en in zenuwscheden. \emph{Sterolen}\index{sterol} ---
\emph{cholesterol}\index{cholesterol} bij dieren, verwante sterolen bij planten en
schimmels --- zijn stijve moleculen met vier ringen en één hydroxylgroep als kop,
die tussen de staarten van de andere \omterm{def:b1:lipids:fattyacid}{lipiden} liggen.
\end{definition}

\begin{proposition}[Zelfassemblage]\label{prop:b1:lipids:selfassembly}
\index{zelfassemblage}\index{micel}\index{liposoom}
In water voegen amfifiele moleculen zich vanzelf zo samen dat hun staarten
verborgen zijn en hun koppen blootliggen. De vorm van het molecuul bepaalt de
structuur: \omterm{def:b1:lipids:fattyacid}{lipiden} met één staart (\omterm{def:b1:lipids:fattyacid}{vetzuren}, detergentia), met de vorm van een
kegel, vormen \emph{micellen} --- bolletjes van enkele nanometers met de staarten
binnenin; \omterm{def:b1:lipids:amphiphilic}{fosfolipiden} met twee staarten, met de vorm van een cilinder, vormen
\emph{dubbellagen}, die zich tot \emph{liposomen} (blaasjes) sluiten zodat er geen
rand blootligt. De assemblage wordt gedreven door het \omterm{prop:b1:water-small-molecules:properties}{hydrofobe effect}
(\cref{ch:b1:water-small-molecules}): zij kost geen energie en vraagt geen enzym,
en een gescheurde dubbellaag sluit zich vanzelf. Membranen zijn zelfherstellende
bladen die groeien door inbouw van nieuwe \omterm{def:b1:lipids:fattyacid}{lipiden} en nooit uit het niets ontstaan:
elke membraan komt uit een membraan voort.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
Gedroogd \omterm{def:b1:lipids:amphiphilic}{fosfolipide} dat in water wordt gedispergeerd vormt gesloten blaasjes met
wanden van een dubbellaag, zichtbaar in de \omterm{prop:b1:cell-unit-of-life:microscopes}{elektronenmicroscoop} (Bangham, 1965);
hun doorlaatbaarheid voor ionen en water komt overeen met die van celmembranen, en
zij kunnen met geneesmiddelen worden geladen en met \omterm{def:b1:cell-unit-of-life:cell}{cellen} versmolten. Het
toevoegen van een detergens (met één staart, kegelvormig) lost de dubbellaag op
tot gemengde micellen; het verwijderen ervan door dialyse laat de dubbellaag zich
opnieuw vormen.
\end{proof}

\begin{example}[Galzouten]\label{ex:b1:lipids:bile}
Voedingsvet komt als \omterm{def:b1:lipids:triglyceride}{olie} in de darm aan; het lipase dat het verteert werkt alleen
op het grensvlak van \omterm{def:b1:lipids:triglyceride}{olie} en water. Galzouten, amfifiele afgeleiden van
\omterm{def:b1:lipids:amphiphilic}{cholesterol}, bekleden het \omterm{def:b1:lipids:triglyceride}{vet} tot druppels van een micrometer, waarmee zij het
grensvlak duizendvoudig vergroten, en vervoeren daarna de vrijgekomen \omterm{def:b1:lipids:fattyacid}{vetzuren}, in
micellen, naar de opnemende \omterm{def:b1:cell-unit-of-life:cell}{cellen} (\cref{ch:b1:digestion-absorption}). Zeep
gebruikt dezelfde truc.
\end{example}

\section{Membraanvloeibaarheid}

\begin{definition}[Vloeibaarheid, faseovergang]\label{def:b1:lipids:fluidity}
Een \omterm{def:b1:membranes-transport:membrane}{lipidedubbellaag} heeft een \emph{faseovergangstemperatuur}\index{faseovergangstemperatuur}
$T_m$: eronder zijn de ketens in een geordende, gelachtige toestand gepakt waarin
de \omterm{def:b1:lipids:fattyacid}{lipiden} nauwelijks bewegen; erboven zijn zij wanordelijk en is de dubbellaag
een tweedimensionale vloeistof waarin een \omterm{def:b1:lipids:fattyacid}{lipide} een miljoen keer per seconde van
plaats wisselt met haar buur en het vlak van de membraan met \qty{1}{\micro m/s}
doorkruist. \omterm{def:b1:cell-unit-of-life:cell}{Cellen} houden hun membranen boven $T_m$: de
\emph{vloeibaarheid}\index{membraanvloeibaarheid} laat eiwitten diffunderen en
draaien, blaasjes afknoppen en versmelten, en de dubbellaag zich sluiten.
\end{definition}

\begin{proposition}[Wat de vloeibaarheid bepaalt]\label{prop:b1:lipids:fluidityfactors}
$T_m$ stijgt met de lengte van de ketens en daalt met hun onverzadigdheid, net als
bij vrije \omterm{def:b1:lipids:fattyacid}{vetzuren}; een dubbellaag van C18:0-ketens smelt bij \qty{55}{\celsius},
een van C18:1-ketens bij \qty{-20}{\celsius}. \omterm{def:b1:lipids:amphiphilic}{Cholesterol}, tussen de staarten
ingebracht, heeft een dubbel effect: boven $T_m$ beperken zijn stijve ringen de
beweging van de ketens en verstijven zij de membraan; onder $T_m$ verhinderen zij
dat de ketens zich pakken en houden zij haar van bevriezing af. Het verbreedt de
overgang tot een geleidelijke verandering en houdt de \omterm{def:b1:lipids:fluidity}{vloeibaarheid} over een breed
temperatuurbereik vrijwel constant --- een \omterm{def:b1:water-small-molecules:buffer}{buffer} van \omterm{def:b1:lipids:fluidity}{vloeibaarheid}. Dierlijke
plasmamembranen bevatten tot één \omterm{def:b1:lipids:amphiphilic}{cholesterol} per \omterm{def:b1:lipids:amphiphilic}{fosfolipide}.
\end{proposition}

\begin{proposition}[Homeoviskeuze aanpassing]\label{prop:b1:lipids:homeoviscous}
\index{homeoviskeuze aanpassing}
\omterm{def:b1:organism-environment:organism}{Organismen} die hun temperatuur niet kunnen regelen stellen de samenstelling van
hun membranen bij om de \omterm{def:b1:lipids:fluidity}{vloeibaarheid} constant te houden: naarmate de temperatuur
daalt vervangen zij verzadigde door onverzadigde \omterm{def:b1:lipids:fattyacid}{vetzuren} en lange ketens door
kortere, en omgekeerd wanneer zij stijgt.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
\emph{E. coli} gekweekt bij \qty{10}{\celsius} heeft twee keer zoveel onverzadigde
\omterm{def:b1:lipids:fattyacid}{vetzuren} als dezelfde stam gekweekt bij \qty{40}{\celsius}, en de membranen van de
twee kweken zijn, gemeten aan de beweeglijkheid van een probe, bij hun eigen
kweektemperatuur even vloeibaar. Forellen die aan \qty{5}{\celsius} zijn gewend
dragen meer meervoudig onverzadigde zuren in hun membraanlipiden dan forellen bij
\qty{20}{\celsius}; het \omterm{def:b1:lipids:triglyceride}{vet} in de poten van een rendier, vlak bij de sneeuw, is
onverzadigder dan het \omterm{def:b1:lipids:triglyceride}{vet} van de romp. Planten die vorst overleven verrijken hun
membranen in de herfst met onverzadigde \omterm{def:b1:lipids:fattyacid}{lipiden}, en mutanten die dat niet kunnen
sterven in de kou.
\end{proof}

\begin{example}[Boter, olijfolie, visolie]\label{ex:b1:lipids:kitchen}
Boter (\qty{65}{\%} verzadigd) is vast in de koelkast en zacht bij
kamertemperatuur; olijfolie (\qty{75}{\%} oliezuur) is vloeibaar bij
kamertemperatuur en wordt troebel in de koelkast; visolie, rijk aan
omega-3-zuren met vijf en zes dubbele bindingen, blijft vloeibaar bij
\qty{-20}{\celsius}, en dat is wat de membranen van een kabeljauw in de
Noord-Atlantische Oceaan nodig hebben.
\end{example}

\section{Sterolen, kleurstoffen en signalen}

\begin{definition}[Steroïden, isoprenoïden]\label{def:b1:lipids:steroids}
\emph{Steroïden}\index{steroïde} delen de vier gefuseerde ringen van \omterm{def:b1:lipids:amphiphilic}{cholesterol}:
het \omterm{def:b1:lipids:amphiphilic}{cholesterol} zelf (membranen, en de voorloper van de rest), de galzouten,
vitamine D, en de steroïdhormonen --- cortisol, aldosteron, de
geslachtshormonen --- kleine, hydrofobe moleculen die membranen vrij passeren en
op receptoren binnen de \omterm{def:b1:cell-unit-of-life:cell}{cel} inwerken. \emph{Isoprenoïden}\index{isoprenoïde}
(terpenen) zijn ketens en ringen van isopreeneenheden met vijf koolstoffen: de
\emph{carotenoïden} die \omterm{def:b1:flowering-plant-organization:organs}{wortels} kleuren en \omterm{def:b1:eukaryotic-cell:plastid}{chloroplasten} beschermen
(\cref{ch:b1:photosynthesis}), de zijketen van chlorofyl, de chinonen van de
elektronentransportketens, rubber, en de vetoplosbare vitaminen A, E en K.
\end{definition}

\begin{proposition}[Wat lipiden doen]\label{prop:b1:lipids:functions}
Energieopslag (\omterm{def:b1:lipids:triglyceride}{triglyceriden}); membranen (\omterm{def:b1:lipids:amphiphilic}{fosfolipiden}, \omterm{def:b1:lipids:amphiphilic}{sfingolipiden}, \omterm{def:b1:lipids:amphiphilic}{sterolen});
isolatie en waterafstoting (\omterm{def:b1:lipids:triglyceride}{vet}, \omterm{def:b1:lipids:wax}{wassen}); lichtopname en bescherming
(carotenoïden); elektronenvervoer (chinonen); signalering tussen \omterm{def:b1:cell-unit-of-life:cell}{cellen}
(steroïdhormonen, prostaglandinen) en binnen \omterm{def:b1:cell-unit-of-life:cell}{cellen} (inositolfosfolipiden,
diacylglycerol); vitaminen. Eén eigenschap --- onoplosbaarheid in water --- ligt
aan al deze rollen ten grondslag: een \omterm{def:b1:lipids:fattyacid}{lipide} blijft waar het is neergelegd, in een
druppel, een film of een membraan, of passeert een membraan zonder \omterm{def:b1:membranes-transport:transporters}{carrier}.
\end{proposition}

\begin{example}[Een hormoon dat geen receptor aan het oppervlak nodig heeft]\label{ex:b1:lipids:steroidhormone}
Cortisol, dat door de bijnier wordt afgescheiden, reist door het bloed gebonden aan
een draagereiwit, laat het bij een doelcel los, lost binnen seconden door de
plasmamembraan heen op, en bindt een receptor in het \omterm{def:b1:eukaryotic-cell:organelle}{cytosol} die vervolgens de
kern binnengaat en genen aanzet (\cref{ch:b1:expression-control}). Een
eiwithormoon zoals insuline, wateroplosbaar en te groot om te passeren, moet een
receptor aan de buitenkant binden en zijn boodschap naar binnen laten doorgeven.
De chemie van de boodschapper bepaalt de route van de boodschap.
\end{example}

\section{Opgaven}

\begin{exercise}[$\star$]\label{exo:b1:lipids:1}
Definieer een \omterm{def:b1:lipids:fattyacid}{lipide} en leg uit waarom de definitie op een eigenschap berust en

Schrijf de notatie van linolzuur (18 koolstoffen, dubbele bindingen na de

\end{exercise}

Waarom vormt een \omterm{def:b1:lipids:triglyceride}{triglyceride} druppels en een \omterm{def:b1:lipids:amphiphilic}{fosfolipide} dubbellagen? Welk

\end{exercise}

Bij welke temperatuur zit volgens de vloeibaarheidsfiguur elk van de drie
dubbellagen halverwege haar overgang? Welke zou je verwachten bij een bacterie die
bij \qty{10}{\celsius} leeft?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:lipids:5}

massa gehydrateerd glycogeen die dezelfde energie zou bevatten.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:lipids:6}

\begin{exercise}[$\star\star$]\label{exo:b1:lipids:7}
Een rode bloedcel heeft een oppervlak van \qty{140}{\micro m^2}; een fosfolipidekop
neemt \qty{0.6}{nm^2} in. Bereken het aantal fosfolipidemoleculen in de membraan
(twee bladen) en, bij \qty{750}{g/mol}, hun totale massa in picogram.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:lipids:8}
\qty{1}{g} olijfolie (dichtheid \qty{0.9}{g/mL}) wordt geëmulgeerd tot druppels van
\qty{1}{\micro m} doorsnede. Bereken het totale grensvlak. Herhaal dat voor één

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:lipids:9}
Leg de twee tegengestelde effecten van \omterm{def:b1:lipids:amphiphilic}{cholesterol} op de \omterm{def:b1:lipids:fluidity}{membraanvloeibaarheid} uit
en waarom een membraan met \omterm{def:b1:lipids:amphiphilic}{cholesterol} geen scherpe overgang heeft.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:lipids:10}
Een mutante plant kan geen onverzadigde \omterm{def:b1:lipids:fattyacid}{vetzuren} maken. Voorspel de toestand van
haar membranen bij \qty{25}{\celsius} en bij \qty{5}{\celsius}, wat er in een
koude nacht met haar \omterm{def:b1:cell-unit-of-life:cell}{cellen} gebeurt, en waarom het wildtype het overleeft.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:lipids:11}
\omterm{def:b1:lipids:triglyceride}{Vet} geeft \qty{1.07}{g} water per gram dat wordt geoxideerd (glycogeen
\qty{0.56}{g}, eiwit \qty{0.4}{g}). Bereken het water dat een kameel maakt door
\qty{1}{kg} \omterm{def:b1:lipids:triglyceride}{vet} te oxideren. De benodigde zuurstof is \qty{2}{L} per gram \omterm{def:b1:lipids:triglyceride}{vet}; die
inademen in droge woestijnlucht kost ongeveer \qty{0.03}{g} water per liter
geventileerde lucht, bij \qty{5}{\%} zuurstofonttrekking. Bereken het water dat
met de ademhaling verloren gaat en zeg of de bult een watervoorraad is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:lipids:12}
``Een membraan bouwt zichzelf op, maar een \omterm{def:b1:cell-unit-of-life:cell}{cel} kan geen membraan uit het niets
maken.'' Breng de twee helften van die zin in een alinea met elkaar in
overeenstemming, aan de hand van het \omterm{prop:b1:water-small-molecules:properties}{hydrofobe effect}, de groei van membranen door
inbouw, en de continuïteit van membranen door de celdeling heen.
\end{exercise}

\section{Vraagstuk: koud water en woestijnvet}

\begin{problem}[{Weekendvraagstuk --- de membranen van een forel in een bergbeek
en de bult van een kameel in de woestijn: onverzadigdheid,
overgangstemperaturen, energiedichtheid en stofwisselingswater, eindigend op de
massa die vet opslaan in plaats van glycogeen bespaart}]
\label{pb:b1:lipids:1}
Deel I gaat over een forel waarvan de membraanfosfolipiden ketens van C16:0,
C18:1 en C22:6 (een omega-3-zuur met zes dubbele bindingen) dragen. Deel II gaat
over een dromedaris van \qty{500}{kg} met een bult van \qty{30}{kg} \omterm{def:b1:lipids:triglyceride}{triglyceride},
die een woestijn doorkruist met een \omterm{def:b1:organism-environment:allometry}{stofwisselingssnelheid} van \qty{60}{MJ/d}.
Energie: \omterm{def:b1:lipids:triglyceride}{vet} \qty{38}{kJ/g}, glycogeen \qty{17}{kJ/g} droog, opgeslagen met
\qty{3}{g} water per gram. Stofwisselingswater: \qty{1.07}{g/g} \omterm{def:b1:lipids:triglyceride}{vet}. Zuurstof:
\qty{2.0}{L} per gram \omterm{def:b1:lipids:triglyceride}{vet}; lucht is \qty{21}{\%} zuurstof; de kameel onttrekt
\qty{5}{\%} van de zuurstof die hij inademt; uitgeademde lucht draagt
\qty{30}{g/m^3} meer water dan de ingeademde woestijnlucht.

\medskip
\noindent\textbf{Deel I --- De forel.}
Forellen die aan \qty{20}{\celsius} gewend zijn hebben membraanketens die voor
\qty{40}{\%} C16:0, voor \qty{45}{\%} C18:1 en voor \qty{15}{\%} C22:6 zijn;
forellen bij \qty{5}{\celsius} voor \qty{25}{\%}, \qty{40}{\%} en \qty{35}{\%}.
\begin{enumerate}
  \item Schrijf de notatie van de drie zuren en tel per keten het aantal
        dubbele bindingen in elk.
  \item Bereken bij elke temperatuur het gemiddelde aantal dubbele
        bindingen per keten.
  \item Leg vanuit de ketenpakking uit waarom de verandering de
        overgangstemperatuur van de membraan verlaagt.
  \item De $T_m$ van een dubbellaag daalt met ongeveer \qty{20}{\celsius}
        per extra \qty{0.5}{} dubbele binding per keten gemiddeld. Schat de
        verschuiving van $T_m$ tussen de twee forellen.
  \item De membraan van de warme forel is vloeibaar bij \qty{20}{\celsius}
        maar zou bij \qty{5}{\celsius} geleren. Wat zou er in een plotselinge
        koude nacht met haar transporteiwitten, haar pompen en haar
        zenuwgeleiding gebeuren?
  \item De aanpassing duurt dagen. Noem de enzymen waarvan de forel er meer
        moet maken, en waar zij in de \omterm{def:b1:cell-unit-of-life:cell}{cel} werken (\cref{ch:b1:eukaryotic-cell}).
  \item \omterm{def:b1:lipids:amphiphilic}{Cholesterol} zit bij beide forellen op één molecuul per twee
        \omterm{def:b1:lipids:amphiphilic}{fosfolipiden}. Leg uit wat het in elk geval bijdraagt.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De energie van de kameel.}
\begin{enumerate}[resume]
  \item Bereken de energie die in de bult is opgeslagen.
  \item Voor hoeveel dagen van de tocht volstaat die?
  \item Bereken de massa droog glycogeen die dezelfde energie bevat, en de
        massa gehydrateerd glycogeen.
  \item Bereken de massa die wordt bespaard door \omterm{def:b1:lipids:triglyceride}{vet} in plaats van
        glycogeen op te slaan, en druk die uit als een fractie van de
        lichaamsmassa van de kameel.
  \item Het \omterm{def:b1:lipids:triglyceride}{vet} wordt in een bult opgeslagen en niet onder de huid
        uitgespreid. Geef daarvoor een reden, uit de \omterm{prop:b1:mammal-organization:heatbudget}{warmtebalans} van
        \cref{ch:b1:mammal-organization}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Is de bult een watervoorraad?}
\begin{enumerate}[resume]
  \item Bereken het stofwisselingswater dat het oxideren van de hele bult
        oplevert.
  \item Bereken de benodigde zuurstof, in liters.
  \item Bereken het luchtvolume dat de kameel moet ventileren om die te
        krijgen.
  \item Bereken het water dat met het uitademen van die lucht verloren
        gaat.
  \item Vergelijk het gewonnen water met het verloren water en trek een
        conclusie.
  \item In plaats van te zweten laat de kameel zijn lichaamstemperatuur
        overdag met \qty{6}{\celsius} stijgen en 's nachts weer dalen.
        Bereken de warmte die hij zo opslaat (warmtecapaciteit
        \qty{3.5}{kJ.kg^{-1}.K^{-1}}) en het water dat het verdampen van
        diezelfde warmte zou kosten (\qty{2.4}{kJ/g}).
  \item Een kameel kan \qty{25}{\%} van zijn lichaamswater verliezen en
        overleven, en zijn rode \omterm{def:b1:cell-unit-of-life:cell}{cellen} zwellen tot twee keer hun volume
        zonder te barsten wanneer hij in tien minuten \qty{100}{L} drinkt.
        Breng de tweede eigenschap in verband met de membranen van
        \cref{ch:b1:membranes-transport}.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het rekenwerk van de dubbellaag.}
Een fosfolipidekop neemt \qty{0.65}{nm^2} in; een dubbellaag is \qty{5}{nm} dik;
de rode \omterm{def:b1:cell-unit-of-life:cell}{cellen} van de kameel hebben een oppervlak van \qty{80}{\micro m^2} en
komen met $\num{8e12}$ per liter bloed voor.
\begin{enumerate}[resume]
  \item Bereken het aantal \omterm{def:b1:lipids:amphiphilic}{fosfolipiden} in de membraan van één rode \omterm{def:b1:cell-unit-of-life:cell}{cel}.
  \item Bereken het aantal in alle rode \omterm{def:b1:cell-unit-of-life:cell}{cellen} van \qty{40}{L} bloed.
  \item Bereken bij \qty{750}{g/mol} de massa van die \omterm{def:b1:lipids:fattyacid}{lipide} in gram.
  \item Een \omterm{def:b1:lipids:amphiphilic}{fosfolipide} klapt ongeveer eens per week vanzelf van het ene
        \omterm{def:b1:flowering-plant-organization:organs}{blad} naar het andere, maar diffundeert zijdelings \qty{1}{\micro m}
        per seconde. Verklaar beide getallen uit het \omterm{prop:b1:water-small-molecules:properties}{hydrofobe effect}.
  \item Leg uit waarom de twee bladen van een membraan, gegeven die
        omklapsnelheid, het hele leven van een \omterm{def:b1:cell-unit-of-life:cell}{cel} in samenstelling kunnen
        verschillen.
  \item Geef het resultaat: de massa die de kameel bespaart door
        \qty{30}{kg} \omterm{def:b1:lipids:triglyceride}{vet} mee te dragen in plaats van het glycogeen met
        dezelfde energie, en het oordeel over de bult als watervoorraad.
\end{enumerate}
\end{problem}
