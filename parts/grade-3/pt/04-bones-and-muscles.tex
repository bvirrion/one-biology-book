\chapter{Ossos e músculos}\label{ch:g3:bones-and-muscles}

Pule e caia: alguma coisa dentro de você acabou de funcionar como
uma máquina perfeita. Hastes rígidas seguraram a sua forma,
dobradiças deixaram as suas \omterm{def:g1:my-body:parts}{pernas} se recolherem e dispararem, e
centenas de motorzinhos puxaram exatamente na hora certa. As hastes
são os seus \omterm{def:g3:bones-and-muscles:skeleton}{ossos}, as dobradiças são as suas \omterm{prop:g3:bones-and-muscles:joints}{articulações}, os
motores são os seus \omterm{def:g3:bones-and-muscles:muscle}{músculos} --- o equipamento do corpo para todos
os movimentos que ele vai fazer na vida.

\section{O esqueleto}

\begin{definition}[Esqueleto e ossos]\label{def:g3:bones-and-muscles:skeleton}
O \emph{esqueleto}\index{esqueleto} é a armação de cerca de $200$
\emph{ossos}\index{osso} que existe dentro do corpo. As regiões
principais dele:
\begin{itemize}
  \item o \emph{crânio}\index{crânio}, um capacete sólido de osso em
        volta do cérebro;
  \item a \emph{coluna}\index{coluna vertebral}, uma torre flexível
        de ossinhos empilhados do pescoço até os quadris, que segura
        o tronco em pé;
  \item as \emph{costelas}\index{costela}, ossos curvos que formam
        uma caixa em volta do coração e dos pulmões;
  \item os \emph{ossos dos \omterm{def:g1:my-body:parts}{membros}}: ossos compridos nos \omterm{def:g1:my-body:parts}{braços} e
        nas \omterm{def:g1:my-body:parts}{pernas} --- o osso da coxa é o mais comprido do corpo.
\end{itemize}
\end{definition}

\begin{example}[Sinta a sua armação]\label{ex:g3:bones-and-muscles:feel}
Não precisa de raio X: aperte de leve e o \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto} responde. Os nós
dos dedos; a crista da canela; a fileira de calombinhos que desce
pelo meio das suas costas --- a \omterm{def:g3:bones-and-muscles:skeleton}{coluna}, um ossinho depois do outro;
as \omterm{def:g3:bones-and-muscles:skeleton}{costelas} que você consegue contar com os dedos. A armação está
logo debaixo da \omterm{def:g1:the-five-senses:sense}{pele}.
\end{example}

\begin{remark}[Duro por fora, vivo por dentro]\label{rem:g3:bones-and-muscles:alive}
Um \omterm{def:g3:bones-and-muscles:skeleton}{osso} não é material morto como um giz: ele é vivo, alimentado
pelo sangue, crescendo enquanto você cresce --- e capaz de se
consertar sozinho. Um \omterm{def:g1:my-body:parts}{braço} quebrado dentro do gesso se recupera
porque o próprio \omterm{def:g3:bones-and-muscles:skeleton}{osso} reconstrói devagar a fratura. Giz nenhum faz
isso.
\end{remark}

\section{As articulações, onde a armação dobra}

\begin{proposition}[Hastes rígidas, lugares que dobram]\label{prop:g3:bones-and-muscles:joints}
Os \omterm{def:g3:bones-and-muscles:skeleton}{ossos}, por si só, nunca dobram --- um \omterm{def:g3:bones-and-muscles:skeleton}{osso} dobrado é um \omterm{def:g3:bones-and-muscles:skeleton}{osso}
quebrado. Todo o dobrar do corpo acontece nas
\emph{articulações}\index{articulação}, onde dois \omterm{def:g3:bones-and-muscles:skeleton}{ossos} se
encontram: \omterm{def:g1:my-body:joint}{ombro}, \omterm{def:g1:my-body:joint}{cotovelo}, \omterm{def:g1:my-body:joint}{punho}, quadril, \omterm{def:g1:my-body:joint}{joelho}, \omterm{def:g1:my-body:joint}{tornozelo} e as
muitas articulaçõezinhas da \omterm{def:g3:bones-and-muscles:skeleton}{coluna}, cujas dobrinhas somadas dão umas
costas flexíveis.
\end{proposition}
\admitted

\begin{example}[O truque da coluna]\label{ex:g3:bones-and-muscles:spinetrick}
Nenhum \omterm{def:g3:bones-and-muscles:skeleton}{osso} das suas costas consegue se enrolar --- e mesmo assim
você consegue se dobrar para a frente quase virando uma bolinha. A
\omterm{def:g3:bones-and-muscles:skeleton}{coluna} é uma torre de mais de $20$ ossinhos, cada um girando alguns
graus sobre os vizinhos: muitas dobras pequenas fazendo uma grande.
É também por isso que você deve dobrar os \omterm{def:g1:my-body:joint}{joelhos}, e não as costas,
para levantar algo pesado --- os \omterm{def:g1:my-body:joint}{joelhos} são feitos para dobras
grandes, a \omterm{def:g3:bones-and-muscles:skeleton}{coluna} para dobras suaves.
\end{example}

\section{Os músculos, os que movem}

\begin{definition}[Músculo]\label{def:g3:bones-and-muscles:muscle}
Um \emph{músculo}\index{músculo} é um motor de carne preso aos
\omterm{def:g3:bones-and-muscles:skeleton}{ossos}. Um músculo consegue \emph{se contrair}\index{contrair} ---
encurtar e engrossar --- e, ao se contrair, ele \emph{puxa} o \omterm{def:g3:bones-and-muscles:skeleton}{osso}
em que está preso. Os músculos só puxam; nenhum músculo consegue
empurrar.
\end{definition}

\begin{proposition}[Trabalhando em pares]\label{prop:g3:bones-and-muscles:pairs}
Como os \omterm{def:g3:bones-and-muscles:muscle}{músculos} só puxam, mover uma \omterm{def:g1:my-body:joint}{articulação} nos dois sentidos
exige um \emph{par}: um \omterm{def:g3:bones-and-muscles:muscle}{músculo} na frente do \omterm{def:g1:my-body:parts}{braço} dobra o \omterm{def:g1:my-body:joint}{cotovelo},
o parceiro dele atrás estica o \omterm{def:g1:my-body:joint}{cotovelo}; cada um relaxa enquanto o
outro puxa. Quase toda \omterm{def:g1:my-body:joint}{articulação} do corpo é movida por pares
assim.
\end{proposition}
\admitted

\begin{example}[Sinta o par trabalhando]\label{ex:g3:bones-and-muscles:pair}
Segure o seu \omterm{def:g1:my-body:parts}{braço} com a outra mão e dobre o \omterm{def:g1:my-body:joint}{cotovelo} devagar: o
\omterm{def:g3:bones-and-muscles:muscle}{músculo} da frente incha e endurece --- ele está se contraindo. Agora
estique o \omterm{def:g1:my-body:parts}{braço}: a frente amolece e, atrás do \omterm{def:g1:my-body:parts}{braço}, o parceiro dele
se aperta no lugar. Você acabou de sentir a
\cref{prop:g3:bones-and-muscles:pairs} acontecer.
\end{example}

\begin{method}[Cuidar dos ossos e dos músculos]\label{met:g3:bones-and-muscles:care}
\begin{enumerate}
  \item Mexa-se todo dia --- os \omterm{def:g3:bones-and-muscles:skeleton}{ossos} ficam mais fortes com o uso, e
        os \omterm{def:g3:bones-and-muscles:muscle}{músculos} também;
  \item coma os alimentos que constroem: laticínios para os \omterm{def:g3:bones-and-muscles:skeleton}{ossos}, e
        refeições variadas (\cref{ch:g2:teeth-and-eating-well});
  \item aqueça antes do exercício pesado, para os \omterm{def:g3:bones-and-muscles:muscle}{músculos} esticarem
        em vez de rasgar;
  \item levante coisas pesadas com os \omterm{def:g1:my-body:joint}{joelhos} dobrados e as costas
        retas;
  \item capacete e proteções onde as quedas são prováveis: os \omterm{def:g3:bones-and-muscles:skeleton}{ossos}
        se consertam, mas devagar.
\end{enumerate}
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:1}
Diga as quatro regiões do \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto} da
\cref{def:g3:bones-and-muscles:skeleton} e o que cada uma protege ou
sustenta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:2}
Mais ou menos quantos \omterm{def:g3:bones-and-muscles:skeleton}{ossos} tem o \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto}? Qual é o mais comprido?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:3}
Um \omterm{def:g3:bones-and-muscles:skeleton}{osso} consegue dobrar? Onde acontece todo o dobrar do corpo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:4}
A \omterm{def:g3:bones-and-muscles:skeleton}{coluna} consegue se enrolar, embora nenhum \omterm{def:g3:bones-and-muscles:skeleton}{osso} dela consiga.
Explique o truque dela.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:5}
O que um \omterm{def:g3:bones-and-muscles:muscle}{músculo} faz quando se contrai? O que nenhum \omterm{def:g3:bones-and-muscles:muscle}{músculo}
consegue fazer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:6}
Por que dobrar e esticar o \omterm{def:g1:my-body:joint}{cotovelo} precisa de dois \omterm{def:g3:bones-and-muscles:muscle}{músculos}? Diga o
que cada um faz em cada momento.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:7}
Como um \omterm{def:g1:my-body:parts}{braço} quebrado dentro do gesso se recupera? O que isso
mostra sobre o \omterm{def:g3:bones-and-muscles:skeleton}{osso}, segundo a \cref{rem:g3:bones-and-muscles:alive}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:8}
Faça a experiência: faça o teste de segurar o \omterm{def:g1:my-body:parts}{braço} do
\cref{ex:g3:bones-and-muscles:pair} no seu próprio \omterm{def:g1:my-body:parts}{braço}. Descreva o
que a frente do \omterm{def:g1:my-body:parts}{braço} faz quando você dobra e quando você estica.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:9}
Por que os adultos dizem ``levante com os \omterm{def:g1:my-body:joint}{joelhos}, não com as
costas''? Responda com o \cref{ex:g3:bones-and-muscles:spinetrick}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:10}
Um boneco de marionete pode ser feito acenar: cada fio só consegue
\emph{puxar} o \omterm{def:g1:my-body:parts}{braço} dele para um lado. Como isso é exatamente a
situação dos \omterm{def:g3:bones-and-muscles:muscle}{músculos}, e qual é a resposta de quem fabrica
marionetes? Compare com a \cref{prop:g3:bones-and-muscles:pairs}.
\end{exercise}
