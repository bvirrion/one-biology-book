\section*{Chapter \ref{ch:g10:membranes-enzymes} --- Membranes and Enzymes}

\begin{solution}{exo:g10:membranes-enzymes:1}
The \omterm{def:g10:membranes-enzymes:fluid-mosaic}{plasma membrane} is a \omterm{def:g10:membranes-enzymes:fluid-mosaic}{phospholipid bilayer} with embedded \omterm{def:g10:molecules-of-life:proteins}{proteins}
forming a mosaic; components can move laterally. ``Fluid'' refers to that
lateral mobility of \omterm{def:g10:molecules-of-life:lipids}{lipids} and many \omterm{def:g10:molecules-of-life:proteins}{proteins}, not to the membrane being
a liquid drop.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:2}
$\mathrm{O_2}$: simple \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}. Glucose down gradient via carrier:
\omterm{def:g10:membranes-enzymes:facilitated}{facilitated diffusion}. $\mathrm{Na^+}$ out against gradient: \omterm{def:g10:membranes-enzymes:active}{active
transport}.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:3}
\omterm{def:g10:membranes-enzymes:osmosis}{Osmosis}: net water movement across a \omterm{def:g10:membranes-enzymes:perm}{selectively permeable} membrane down
its water-potential gradient. Distilled water (\omterm{def:g10:membranes-enzymes:tonicity}{hypotonic}): water enters,
\omterm{def:g10:the-cell:cell}{cell} swells and may lyse. Concentrated salt (\omterm{def:g10:membranes-enzymes:tonicity}{hypertonic}): water leaves,
\omterm{def:g10:the-cell:cell}{cell} shrinks (crenates).
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:4}
The rigid \omterm{def:g10:the-cell:support}{cell wall} resists expansion; the \omterm{def:g10:the-cell:cell}{cell} becomes turgid rather
than bursting. Animal \omterm{def:g10:the-cell:cell}{cells} lack a wall.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:5}
\omterm{def:g10:membranes-enzymes:enzyme}{Enzyme}: biological catalyst (usually \omterm{def:g10:molecules-of-life:proteins}{protein}). \omterm{def:g10:membranes-enzymes:enzyme}{Substrate}: reactant bound
and converted. \omterm{def:g10:membranes-enzymes:enzyme}{Active site}: binding/catalysis pocket. \omterm{def:g10:membranes-enzymes:ES}{Specificity} arises
from complementary shape and chemistry of the \omterm{def:g10:membranes-enzymes:enzyme}{active site}.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:6}
Curve rises to an \omterm{prop:g10:membranes-enzymes:temp-pH}{optimum} near body temperature, then falls. Rising:
faster collisions. Falling: denaturation of the \omterm{def:g10:membranes-enzymes:enzyme}{enzyme}.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:7}
Competitive: binds \omterm{def:g10:membranes-enzymes:enzyme}{active site}, blocks \omterm{def:g10:membranes-enzymes:enzyme}{substrate}; often relieved by more
\omterm{def:g10:membranes-enzymes:enzyme}{substrate}. Non-competitive: binds elsewhere, alters shape; not overcome
simply by more \omterm{def:g10:membranes-enzymes:enzyme}{substrate}.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:8}
Plot $\%$ mass change versus sucrose concentration; the concentration
where mass change is zero (interpolation) is \omterm{def:g10:membranes-enzymes:tonicity}{isotonic} to the tissue's
average \omterm{def:g10:the-cell:cell}{cell} contents.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:9}
\omterm{def:g10:membranes-enzymes:facilitated}{Facilitated diffusion} is passive: net flux requires a concentration
difference. At equilibrium, inward and outward movements balance; no net
transport, though carriers still flicker.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:10}
Same \omterm{def:g10:membranes-enzymes:enzyme}{enzyme} \omterm{def:g10:the-cell:sa-v}{volume}, same peroxide concentration and \omterm{def:g10:the-cell:sa-v}{volume}, same pH and
disk type; vary temperature in water baths. Measure time for disk to
float (or $\mathrm{O_2}$ \omterm{def:g10:the-cell:sa-v}{volume}). Controlled: $[\mathrm{H_2O_2}]$, pH,
\omterm{def:g10:membranes-enzymes:enzyme}{enzyme} amount, disk size, mixing.
\end{solution}

\begin{solution}{exo:g10:membranes-enzymes:11}
(a) Simple \omterm{def:g10:membranes-enzymes:diffusion}{diffusion} of $\mathrm{O_2}$ continues (no ATP needed).
(b) Pump slows/stops without ATP. (c) Gradients run down by leak pathways;
secondary transports fail --- \omterm{def:g10:the-cell:cell}{cells} lose ion \omterm{def:g10:scientific-biology:properties}{homeostasis}.
\end{solution}

\begin{solution}{pb:g10:membranes-enzymes:1}
\textbf{Part I.} Salty water is \omterm{def:g10:membranes-enzymes:tonicity}{hypertonic}: water leaves \omterm{def:g10:the-cell:cell}{cells}, \omterm{def:g10:membranes-enzymes:fluid-mosaic}{plasma
membrane} pulls from wall (plasmolysis), tissue loses turgor and goes limp.
Recovery: rinse and place in pure or lightly \omterm{def:g10:membranes-enzymes:tonicity}{hypotonic} water to restore
turgor. Animal \omterm{def:g10:the-cell:cell}{cell} in same medium would crenate/shrink without a wall.

\textbf{Part II.} Fatty acids lower pH. Rate--temperature: rise to
\omterm{prop:g10:membranes-enzymes:temp-pH}{optimum} then fall (denaturation). \omterm{def:g10:membranes-enzymes:inhibitors}{Competitive inhibitor}: apparent need
for more \omterm{def:g10:membranes-enzymes:enzyme}{substrate} to reach same rate; $V_{\max}$ still approachable at
high [S] (classic competitive pattern).

\textbf{Part III.} Channel (e.g.\ aquaporin, ion channel): passive.
Pump (e.g.\ $\mathrm{Na^+/K^+}$ ATPase): active. ATP keeps gradients and
composition far from equilibrium, enabling work, signalling and \omterm{def:g10:the-cell:sa-v}{volume}
control; equilibrium would mean death of the \omterm{def:g10:the-cell:cell}{cell} as an open system.
\end{solution}
