\chapter{सेहतमंद चुनाव}\label{ch:g5:healthy-choices}

अब तक यह पुस्तक तुम्हारे शरीर की ज़्यादातर बड़ी सेवाओं का चक्कर लगा
चुकी है: \omterm{def:g4:journey-of-food:tract}{पाचन नली}, \omterm{def:g4:breathing:organs}{फेफड़े}, \omterm{def:g4:heart-and-blood:heart}{हृदय} और उसकी \omterm{prop:g4:journey-of-food:absorb}{रक्त} की नदियाँ, \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} और
\omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ}, और कमान संभालता \omterm{def:g5:brain-in-command:system}{मस्तिष्क}। साल का यह आख़िरी अध्याय एक काम का
सवाल पूछता है: यह सारी मशीनरी जानने के बाद तुम उसे अच्छी तरह चलाते कैसे
हो --- और फ़ैसला कौन करता है? क्योंकि लगभग तुम्हारी उम्र से आगे
ज़्यादा से ज़्यादा फ़ैसले तुम्हारे अपने होते जाते हैं।

\section{मशीनरी क्या माँगती है}

\begin{proposition}[शरीर का रोज़ का क़रार]\label{prop:g5:healthy-choices:contract}
पिछले अध्यायों की माँगों को अगल-बगल रखो और एक रोज़ का क़रार सामने आ जाता
है:
\begin{itemize}
  \item \emph{\omterm{prop:g1:healthy-habits:needs}{विविध भोजन} और पानी} --- ईंधन और बनाने वाला सामान
        (\cref{prop:g3:balanced-diet:jobs}), जो \omterm{prop:g4:journey-of-food:absorb}{रक्त} हर \omterm{def:g5:first-look-at-cells:cell}{कोशिका} तक
        पहुँचाता है;
  \item \emph{हलचल} --- \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ}, \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ}, \omterm{def:g4:heart-and-blood:heart}{हृदय} और \omterm{def:g4:breathing:organs}{फेफड़े}, सब
        इस्तेमाल से मज़बूत होते हैं
        (\cref{met:g3:bones-and-muscles:care},
        \cref{rem:g4:heart-and-blood:rest});
  \item \emph{\omterm{prop:g1:healthy-habits:needs}{नींद}} --- \omterm{def:g5:brain-in-command:system}{मस्तिष्क} की फ़ाइलिंग और शरीर की मरम्मत
        (\cref{prop:g5:brain-in-command:protect}); इस उम्र के बच्चे को
        लगभग दस घंटे चाहिए;
  \item \emph{साफ़ हवा, साफ़ हाथ, सँभाले हुए दाँत} ---
        \cref{ch:g1:healthy-habits} और \cref{met:g4:breathing:care}
        के बचाव, जो आज भी ड्यूटी पर हैं।
\end{itemize}
कुछ भी नया नहीं --- बात यही है। सेहत पुराने नियमों का निभाया जाना है।
\end{proposition}
\admitted

\begin{example}[एक दिन, भीतर से देखा हुआ]\label{ex:g5:healthy-choices:day}
एक अच्छा दिन, शरीर-प्रबंधन की तरह पढ़ा हुआ: नाश्ता रात के ख़र्च की भरपाई
करता है (\cref{ex:g3:balanced-diet:breakfast}); स्कूल तक की पैदल चाल
\omterm{def:g4:heart-and-blood:heart}{हृदय} और फेफड़ों को गरमाती है; दोपहर का खेल मज़े के भेस में की गई क़सरत
है; रात के खाने की सब्ज़ियाँ ताज़गी वाले वर्ग की भरपाई करती हैं; और
जल्दी सोना \omterm{def:g5:brain-in-command:system}{मस्तिष्क} को उसका फ़ाइलिंग का समय सौंप देता है। कोई भी चीज़
वीरता वाली नहीं; पर सब मिलकर सेहत हैं।
\end{example}

\section{नए चुनाव, नए दबाव}

\begin{proposition}[परदे और नींद]\label{prop:g5:healthy-choices:screens}
परदे --- फ़ोन, टैबलेट, खेल --- न ज़हर हैं न भोजन; वे \emph{समय} हैं।
जो घंटे वे सोख लेते हैं वे कहीं न कहीं से आते हैं, और आम तौर पर उनका
शिकार ठीक क़रार की शर्तें बनती हैं: बाहर की हलचल और \omterm{prop:g1:healthy-habits:needs}{नींद}। इससे भी बुरा,
शाम को देर तक जलते चमकीले परदे \omterm{def:g5:brain-in-command:system}{मस्तिष्क} से कहते हैं ``अभी दिन ही है'',
और \omterm{prop:g1:healthy-habits:needs}{नींद} उन घंटों से भी आगे खिसक जाती है जो पहले ही दे दिए गए थे। सवाल
परदे हों या न हों का नहीं है, बल्कि यह है कि उन्हें कौन-से घंटे मिलें
--- और शाम का आख़िरी घंटा उन्हें देने के लिए सबसे बुरा घंटा है।
\end{proposition}
\admitted

\begin{example}[ग्यारह बजे वाला प्रयोग]\label{ex:g5:healthy-choices:latenight}
इसे अपने ही ऊपर एक न्यायी जाँच
(\cref{met:g4:what-plants-need:fairtest}) की तरह आज़माओ: एक हफ़्ता
सोने से एक घंटा पहले परदे बंद, और उसके मुक़ाबले एक आम हफ़्ता। ज़्यादातर
लोग पहले हफ़्ते में जल्दी सोते हैं, आसानी से जागते हैं और सुबहें बेहतर
पाते हैं --- और \omterm{met:g4:what-plants-need:fairtest}{नियंत्रण} वाला हफ़्ता ही वह दलील है।
\end{example}

\begin{proposition}[शरीर को धोखा देने वाले पदार्थ]\label{prop:g5:healthy-choices:substances}
कुछ पदार्थ ठीक उसी मशीनरी को नुक़सान पहुँचाते हैं जिसका चक्कर इस पुस्तक
ने लगाया है:
\begin{itemize}
  \item \emph{तंबाकू का धुआँ} फेफड़ों की नाज़ुक कोठरियों को गंदा करता
        है (\cref{met:g4:breathing:care}) और \omterm{def:g4:heart-and-blood:heart}{हृदय} पर बोझ डालता है; वह
        पहली सिगरेटों को बड़प्पन का रूप देकर बेचा जाता है --- और अपने
        ख़रीदारों को \emph{लत}\index{लत} से थामे रखता है: शरीर, एक बार
        उसका आदी हो जाए, तो उसके बिना विरोध करने लगता है;
  \item \emph{शराब} मिनटों में \omterm{def:g5:brain-in-command:system}{मस्तिष्क} तक पहुँच जाती है --- यानी
        कमान संभालने वाले तक ही
        (\cref{ch:g5:brain-in-command}) --- और घेरे को धीमा तथा फ़ैसलों
        को धुँधला कर देती है; और जवान \omterm{def:g5:brain-in-command:system}{मस्तिष्क}, जो अभी अपने रास्ते बना
        ही रहा होता है, बड़े \omterm{def:g5:brain-in-command:system}{मस्तिष्क} से ज़्यादा आसानी से नुक़सान खाता
        है;
  \item दोनों को \omterm{def:g5:human-reproduction:pregnancy}{गर्भस्थ शिशु} के लिए ख़तरा पहले ही बताया जा चुका है
        (\cref{prop:g5:human-reproduction:care}) --- वही पदार्थ, वही
        वजह।
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[कभी-शुरू-न-करना बाद-में-छोड़ने से बेहतर क्यों है]\label{rem:g5:healthy-choices:never}
\omterm{prop:g5:healthy-choices:substances}{लत} एक ऐसा जाल है जिसका दरवाज़ा आसान और निकास कठिन है: शुरू करना एक
फ़ैसला है, छोड़ना सालों तक दोहराई जाने वाली लड़ाई। यह असमानता इस अध्याय
का सबसे ठंडा तथ्य है, और सबसे काम का भी: यह लड़ाई जीतने का सबसे सस्ता
पल वह है जब वह शुरू ही न हुई हो। पहली सिगरेट देने वाले लोग एक सिगरेट
नहीं दे रहे --- वे उसके बाद की सारी सिगरेटें दे रहे हैं।
\end{remark}

\section{अपने लिए ख़ुद तय करना}

\begin{method}[अपना क़रार ख़ुद चलाना]\label{met:g5:healthy-choices:run}
अब से, हर साल कुछ और ज़्यादा, तुम्हारे शरीर के क़रार के रखवाले तुम ख़ुद
हो:
\begin{enumerate}
  \item क़रार की शर्तें जानो ---
        \cref{prop:g5:healthy-choices:contract} वही सूची है;
  \item बीच-बीच में ईमानदारी से अपनी जाँच करो, जैसे
        \cref{met:g3:balanced-diet:check} में --- क्या छूट रहा है,
        और क्या अपने हिस्से से ज़्यादा है?
  \item एक बार में एक ही चीज़ बदलो, और उसे न्यायी जाँच जितना एक हफ़्ता
        दो;
  \item पदार्थों के बारे में अपना ``ना'' उस पल के आने से
        \emph{पहले} तय कर लो
        (\cref{rem:g5:healthy-choices:never}) --- इत्मीनान से पहले
        लिया गया फ़ैसला दोस्ताना दबाव में उसी वक़्त गढ़े गए फ़ैसले से
        कहीं आसानी से निभता है;
  \item और बड़ों का सहारा लो: डॉक्टर, नर्स, माता-पिता और शिक्षक इस
        क़रार में तुम्हारी ही तरफ़ हैं।
\end{enumerate}
\end{method}

\begin{example}[तैयार उत्तर]\label{ex:g5:healthy-choices:answers}
चौथे चरण का पहले से लिया गया फ़ैसला असल में जेब में रखा एक वाक्य है:
``नहीं, शुक्रिया --- यह मेरे मतलब की चीज़ नहीं।'' हलके से, एक ही बार
कहा गया, वह आम तौर पर बात वहीं ख़त्म कर देता है; और सच्चे दोस्त उसे
पहली ही बार में मान लेते हैं। यह वाक्य तैयार करने में कुछ ख़र्च नहीं
होता, और ठीक जब उसकी ज़रूरत हो तभी उसे गढ़ना सबसे कठिन होता है ---
इसीलिए वह अभी गढ़ा जाता है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:1}
शरीर के रोज़ के क़रार की चारों शर्तें गिनाओ, और हर एक के साथ वह अंग या
सेवा भी जिसे वह मुख्य रूप से सँभालती है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:2}
तुम्हारी उम्र के बच्चे को लगभग कितनी \omterm{prop:g1:healthy-habits:needs}{नींद} चाहिए, और शरीर उससे कौन-से दो
काम करता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:3}
परदों के घंटे किस अर्थ में ``कहीं न कहीं से लिए'' जाते हैं? उनके दोनों
आम शिकार बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:4}
शाम का आख़िरी घंटा चमकीले परदे को देने के लिए सबसे बुरा क्यों है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:5}
\cref{prop:g5:healthy-choices:substances} के अनुसार तंबाकू का धुआँ पहले
किन अंगों को नुक़सान पहुँचाता है? और शराब?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:6}
\omterm{prop:g5:healthy-choices:substances}{लत} क्या है, और वह ``कभी शुरू न करना'' को ``बाद में छोड़ना'' से इतना
सस्ता क्यों बना देती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:7}
\cref{met:g5:healthy-choices:run} के चौथे चरण के अनुसार ``ना'' पहले से
तय कर लेने का क्या फ़ायदा है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:healthy-choices:8}
करके देखो: \cref{ex:g5:healthy-choices:latenight} वाला ग्यारह बजे का
प्रयोग एक हफ़्ते चलाओ और --- ईमानदारी से --- बताओ कि क्या बदला।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:healthy-choices:9}
परदे वाले प्रयोग को ठीक न्यायी जाँच की तरह बनाओ: दोनों हफ़्तों में क्या
एक जैसा रहेगा, कौन-सी एक चीज़ बदलेगी, और तुम नापोगे क्या?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:healthy-choices:10}
``एक सिगरेट से क्या बिगड़ेगा।''
\cref{rem:g5:healthy-choices:never} से उत्तर दो: पहली सिगरेट के साथ
असल में दिया क्या जा रहा है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:healthy-choices:11}
पिछले तीन अध्यायों को इस अध्याय से जोड़ो:
\cref{prop:g4:breathing:exchange},
\cref{ex:g4:heart-and-blood:effort} और
\cref{prop:g5:brain-in-command:protect} में से हर एक इस अध्याय की
कौन-सी एक शर्त या कौन-सी एक चेतावनी समझाता है?
\end{exercise}
