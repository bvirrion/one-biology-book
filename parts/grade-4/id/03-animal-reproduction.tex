\chapter{Bagaimana Hewan Berkembang Biak}\label{ch:g4:animal-reproduction}

Pada bulan Mei kolamnya riuh oleh nyanyian katak; menjelang Juni ia
menghitam oleh \omterm{ex:g2:animals-grow-up:frog}{berudu}. Nyanyian itu, ternyata, adalah permulaan
kisahnya: reproduksi --- cara \omterm{def:g1:animals-around-us:animal}{hewan} membuat generasi berikutnya ---
mengikuti aturan yang berlaku mulai dari katak sampai gajah, dengan
ragam yang menakjubkan pada perinciannya.

\section{Diperlukan dua}

\begin{definition}[Reproduksi seksual]\label{def:g4:animal-reproduction:sexual}
Pada \omterm{def:g1:animals-around-us:animal}{hewan}, membuat anak biasanya membutuhkan dua induk: seekor
\emph{betina}\index{betina} dan seekor \emph{jantan}\index{jantan}.
Betinanya menghasilkan \emph{sel telur}\index{sel telur}; jantannya
menghasilkan \emph{sel sperma}\index{sel sperma}. Seekor \omterm{def:g1:animals-around-us:animal}{hewan} muda
bermula ketika satu sel sperma bergabung dengan satu sel \omterm{def:g2:animals-grow-up:born}{telur} ---
penggabungan itu disebut \emph{pembuahan}\index{pembuahan}. Reproduksi
dengan dua induk disebut \emph{reproduksi
seksual}\index{reproduksi seksual}.
\end{definition}

\begin{example}[Mengapa katak bernyanyi]\label{ex:g4:animal-reproduction:song}
Paduan suara bulan Mei itu adalah para \omterm{def:g4:animal-reproduction:sexual}{jantan} yang memanggil para
\omterm{def:g4:animal-reproduction:sexual}{betina} ke kolamnya. Tiap pasangan lalu melepaskan selnya ke dalam air
--- \omterm{def:g2:animals-grow-up:born}{telur} si \omterm{def:g4:animal-reproduction:sexual}{betina}, sperma si \omterm{def:g4:animal-reproduction:sexual}{jantan} --- dan di situlah \omterm{def:g4:animal-reproduction:sexual}{pembuahan}
terjadi. Titik-titik hitam di dalam agar-agarnya beberapa hari
kemudian adalah anak-anaknya yang sudah tumbuh: nyanyian itu adalah
langkah pertama daur katak \cref{ch:g2:animals-grow-up}.
\end{example}

{\small Beberapa hari sesudah paduan suaranya: \omterm{def:g2:animals-grow-up:born}{telur} katak di antara

\begin{remark}[Dua induk, keberuntungan yang berbaur]\label{rem:g4:animal-reproduction:mix}
Seekor \omterm{def:g1:animals-around-us:animal}{hewan} muda memperoleh sesuatu dari tiap induknya --- itulah
sebabnya seekor anak kuda dapat berwarna seperti ibunya dan berbelang
putih seperti ayahnya. Apa tepatnya yang diserahkan tiap induk, dan
bagaimana caranya, adalah salah satu pertanyaan besar yang menunggu di
buku ini; untuk sekarang, catatlah sekadar bahwa setiap \omterm{def:g1:animals-around-us:animal}{hewan} yang
dihasilkan secara seksual punya dua induk dan menyerupai keduanya.
\end{remark}

\section{Telur dikeluarkan, atau anak lahir hidup}

\begin{definition}[Ovipar dan vivipar]\label{def:g4:animal-reproduction:ovivivi}
\Cref{def:g2:animals-grow-up:born} memberikan dua cara dilahirkan;
keduanya sekarang mendapat nama ilmiahnya:
\begin{itemize}
  \item seekor \omterm{def:g1:animals-around-us:animal}{hewan} disebut \emph{ovipar}\index{ovipar} kalau
        \omterm{def:g4:animal-reproduction:sexual}{betinanya} mengeluarkan \emph{\omterm{def:g2:animals-grow-up:born}{telur}}, dan anaknya
        menyelesaikan pertumbuhannya di dalam \omterm{def:g2:animals-grow-up:born}{telur}, di luar tubuh
        induknya: \omterm{prop:g3:sorting-living-things:groups}{burung}, kebanyakan \omterm{prop:g3:sorting-living-things:groups}{ikan}, kebanyakan \omterm{prop:g4:classifying-animals:classes}{reptil}, \omterm{prop:g3:sorting-living-things:groups}{amfibi},
        \omterm{prop:g3:sorting-living-things:groups}{serangga}, siput;
  \item seekor \omterm{def:g1:animals-around-us:animal}{hewan} disebut \emph{vivipar}\index{vivipar} kalau
        anaknya tumbuh \emph{di dalam tubuh induknya}, diberi makan
        olehnya, dan \emph{\omterm{def:g2:animals-grow-up:born}{lahir hidup}}: hampir semua \omterm{prop:g3:sorting-living-things:groups}{mamalia}.
\end{itemize}
\end{definition}

\begin{example}[Dua ruang bayi]\label{ex:g4:animal-reproduction:nurseries}
Anak ayam sang induk melewatkan tiga minggu yang hangat di dalam \omterm{def:g2:animals-grow-up:born}{telur}
bercangkang, hidup dari kuning telurnya
(\cref{rem:g2:animals-grow-up:egg}); anak kuda melewatkan sebelas
bulan di dalam kuda \omterm{def:g4:animal-reproduction:sexual}{betinanya}, diberi makan terus-menerus oleh
tubuhnya, lalu lahir dalam keadaan siap berdiri dalam waktu satu jam.
Tujuan yang sama, dua ruang bayi: yang satu bekal terbungkus \omterm{def:g1:animals-around-us:covering}{cangkang},
yang lain layanan kamar di dalam.
\end{example}

\begin{proposition}[Tempat pembuahan terjadi]\label{prop:g4:animal-reproduction:where}
Bagi penghuni air seperti \omterm{prop:g3:sorting-living-things:groups}{ikan} dan katak, \omterm{def:g4:animal-reproduction:sexual}{pembuahan} lazimnya terjadi
\emph{di dalam air}, di luar tubuh \omterm{def:g4:animal-reproduction:sexual}{betinanya} --- \omterm{def:g2:animals-grow-up:born}{telur} dan sperma
dilepaskan bersama-sama lalu bertemu dalam jumlah ribuan. Bagi \omterm{def:g1:animals-around-us:animal}{hewan}
darat --- \omterm{prop:g3:sorting-living-things:groups}{burung}, \omterm{prop:g4:classifying-animals:classes}{reptil}, \omterm{prop:g3:sorting-living-things:groups}{mamalia}, \omterm{prop:g3:sorting-living-things:groups}{serangga} --- selnya akan mengering
di udara terbuka, sehingga \omterm{def:g4:animal-reproduction:sexual}{jantannya} menaruh spermanya \emph{di dalam
tubuh \omterm{def:g4:animal-reproduction:sexual}{betinanya}} ketika \emph{kawin}\index{perkawinan}, dan \omterm{def:g4:animal-reproduction:sexual}{pembuahan}
terjadi di situ.
\end{proposition}
\admitted

\section{Sedikit anak dengan banyak perawatan, atau banyak tanpa perawatan}

\begin{proposition}[Dua siasat bagi generasi berikutnya]\label{prop:g4:animal-reproduction:strategies}
Bandingkan jumlahnya: seekor \omterm{prop:g3:sorting-living-things:groups}{ikan} kod melepaskan \emph{jutaan} \omterm{def:g2:animals-grow-up:born}{telur}
setahun dan tidak merawat satu pun; seekor gajah punya \emph{satu}
anak setiap beberapa tahun dan menjaganya selama sepuluh tahun. Di
antara kedua ujung itu, sebuah aturan berlaku di seluruh dunia \omterm{def:g1:animals-around-us:animal}{hewan}:
makin banyak anak yang dihasilkan sebuah spesies, makin sedikit
perawatan yang diterima tiap anak --- dan makin sedikit yang
dihasilkannya, makin terlindungi dan terberi makan tiap anaknya. Kedua
siasat itu berhasil: rata-rata, cukup banyak anak yang bertahan untuk
menggantikan induknya.
\end{proposition}
\admitted

\begin{example}[Menghitung kolamnya]\label{ex:g4:animal-reproduction:pondcount}
Sekali bertelur, satu pasang katak menghasilkan ribuan butir. Kuntul,
\omterm{prop:g3:sorting-living-things:groups}{ikan} duri dan larva capung (lihat \cref{ex:g3:food-chains:three})
memakan \omterm{ex:g2:animals-grow-up:frog}{berudu} sepanjang musim semi --- dan dari ribuan itu, segelintir
katak kecil memanjat tepian pada bulan Juli. Kelompok \omterm{def:g2:animals-grow-up:born}{telur} yang amat
besar itu bukan pemborosan: itulah jawaban katak atas bahaya kolamnya,
sama sekali tanpa penjagaan induk.
\end{example}

\begin{example}[Ujung yang satunya]\label{ex:g4:animal-reproduction:whalecalf}
Anak paus lahir sendirian, seketika berenang di samping induknya,
meminum \omterm{def:g2:animals-grow-up:born}{susunya} yang kaya selama berbulan-bulan dan dibela dari setiap
bahaya. Satu anak dalam satu waktu --- tetapi yang satu itu memperoleh
segalanya. \omterm{prop:g3:sorting-living-things:groups}{Mamalia}, dengan \omterm{def:g2:animals-grow-up:born}{susunya}
(\cref{def:g2:animals-grow-up:born}), adalah ahli besar siasat
sedikit-dan-terawat.
\end{example}

\begin{method}[Membaca siasat sebuah spesies]\label{met:g4:animal-reproduction:read}
Diberikan sebuah spesies, tanyakan:
\begin{enumerate}
  \item Berapa banyak anak dalam satu waktu --- satu, selusin, ribuan?
  \item Siapa yang menjaga dan memberinya makan --- kedua induk,
        induk \omterm{def:g4:animal-reproduction:sexual}{betinanya}, tidak seorang pun?
  \item Di mana mereka bermula --- \omterm{def:g2:animals-grow-up:born}{telur} di luar, atau tumbuh di
        dalam?
  \item Periksa keseimbangan pada
        \cref{prop:g4:animal-reproduction:strategies}: banyak anak
        dengan sedikit perawatan, atau sedikit dengan banyak.
\end{enumerate}
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:1}
Apa itu \omterm{def:g4:animal-reproduction:sexual}{pembuahan}? Sel mana yang datang dari tiap induknya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:2}
Definisikan \omterm{def:g4:animal-reproduction:ovivivi}{ovipar} dan \omterm{def:g4:animal-reproduction:ovivivi}{vivipar}, dengan dua \omterm{def:g1:animals-around-us:animal}{hewan} untuk tiap jenisnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:3}
Mengapa katak bernyanyi pada bulan Mei? Ceritakan kembali kisah
kolamnya dalam tiga langkah.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:4}
Di mana \omterm{def:g4:animal-reproduction:sexual}{pembuahan} terjadi pada \omterm{prop:g3:sorting-living-things:groups}{ikan} trout, dan di mana pada ayam
\omterm{def:g4:animal-reproduction:sexual}{betina}? Mengapa berbeda, menurut
\cref{prop:g4:animal-reproduction:where}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:5}
Bandingkan ruang bayi anak ayam dan anak kuda: di mana tiap anak
tumbuh, dan apa yang memberinya makan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:6}
Nyatakan aturan yang menghubungkan jumlah anak dengan perawatan yang
diterimanya. Berikan kedua contoh ujung dalam bab ini.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:7}
Jutaan \omterm{def:g2:animals-grow-up:born}{telur} \omterm{prop:g3:sorting-living-things:groups}{ikan} kod kebanyakan tidak pernah menjadi \omterm{prop:g3:sorting-living-things:groups}{ikan} kod dewasa.
Mengapa siasatnya tetap berhasil?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:8}
Jalankan \cref{met:g4:animal-reproduction:read} atas kucing: sekali
melahirkan biasanya kira-kira lima anak, disusui dan dijaga induknya
selama berminggu-minggu, tumbuh di dalam tubuhnya. Ia lebih dekat ke
sisi siasat yang mana?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:animal-reproduction:9}
Seekor anak kuda sering menyerupai kedua induknya sekaligus. Apa yang
ditunjukkannya tentang apa yang diterima si anak, menurut
\cref{rem:g4:animal-reproduction:mix}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:animal-reproduction:10}
\omterm{prop:g3:sorting-living-things:groups}{Ikan} duri \omterm{def:g4:animal-reproduction:sexual}{jantan} membangun sarang, mengipasi \omterm{def:g2:animals-grow-up:born}{telurnya} dan mengusir
penyusup --- pengabdian yang jarang bagi seekor \omterm{prop:g3:sorting-living-things:groups}{ikan}. Dengan memakai
\cref{prop:g4:animal-reproduction:strategies}, apa yang akan kamu
ramalkan tentang besarnya kelompok \omterm{def:g2:animals-grow-up:born}{telurnya}, dibandingkan dengan
jutaan \omterm{def:g2:animals-grow-up:born}{telur} \omterm{prop:g3:sorting-living-things:groups}{ikan} kod?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:animal-reproduction:11}
Penyu merangkak ke pantai untuk mengubur \omterm{def:g2:animals-grow-up:born}{telurnya} di pasir yang
hangat, lalu kembali ke laut. \omterm{def:g4:animal-reproduction:ovivivi}{Ovipar} atau \omterm{def:g4:animal-reproduction:ovivivi}{vivipar}? Dirawat atau tidak?
Dan mengapa \omterm{prop:g4:classifying-animals:classes}{reptil} ini, yang menjadi perenang seumur hidupnya, tetap
harus bertelur di \emph{darat}? (Pakai ciri \omterm{prop:g4:classifying-animals:classes}{reptil} pada
\cref{prop:g4:classifying-animals:classes}.)
\end{exercise}
