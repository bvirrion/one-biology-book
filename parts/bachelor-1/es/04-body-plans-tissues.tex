\chapter{Planes de organización animal y tejidos}\label{ch:b1:body-plans-tissues}

Cuatro animales sobre la mesa del laboratorio: una hidra de un centímetro,
una lombriz de tierra, un cangrejo de río y una rana. Bajo las diferencias
de talla y de costumbres, todos están construidos sobre un puñado de
decisiones compartidas --- si el cuerpo tiene una parte anterior y otra
posterior, cuántas capas de células lo originaron, si hay una cavidad
alrededor del tubo digestivo, si el cuerpo está hecho de segmentos
repetidos, dónde está el esqueleto --- y todos ellos están construidos con
las mismas cuatro clases de \omterm{def:b1:body-plans-tissues:tissue}{tejido}. Este capítulo expone los \omterm{def:b1:mammal-organization:bodyplan}{planes de
organización} de los animales y los \omterm{def:b1:body-plans-tissues:tissue}{tejidos} que los construyen, y termina
donde lo dejaron los capítulos anteriores: en la pared del tubo digestivo de
un mamífero, vista como \omterm{def:b1:body-plans-tissues:tissue}{tejidos} ensamblados en un \omterm{def:b1:mammal-organization:organ}{órgano}.

\section{Planes de organización}

\begin{definition}[Simetría y cefalización]\label{def:b1:body-plans-tissues:symmetry}
Un animal tiene \emph{simetría radial}\index{simetría radial} cuando su
cuerpo se organiza en torno a un eje central sin izquierda ni derecha, como
una hidra o una medusa: se enfrenta a su medio por igual desde todos los
lados y suele estar fijo o a la deriva. Tiene \emph{simetría
bilateral}\index{simetría bilateral} cuando un plano lo divide en dos
mitades especulares, definiendo una parte anterior y otra posterior, un
dorso y un vientre: el plan de un animal que se desplaza con la cabeza por
delante, con los \omterm{def:b1:mammal-organization:organ}{órganos} de los sentidos y los centros nerviosos
concentrados en el extremo anterior (\omterm{def:b1:mammal-organization:bodyplan}{cefalización}).
\end{definition}

\begin{definition}[Hojas embrionarias, cavidades corporales]\label{def:b1:body-plans-tissues:layers}
El embrión de un animal forma dos o tres capas de células, las \emph{hojas
embrionarias}\index{hoja embrionaria}, de las que derivan todos los \omterm{def:b1:body-plans-tissues:tissue}{tejidos}:
el \emph{ectodermo}\index{ectodermo} externo (piel, sistema nervioso), el
\emph{endodermo}\index{endodermo} interno (revestimiento del tubo digestivo
y de sus derivados) y, en todos salvo las esponjas y los cnidarios, un
\emph{mesodermo}\index{mesodermo} intermedio (músculo, esqueleto, \omterm{def:b1:body-plans-tissues:connective}{tejido
conjuntivo}, sangre, riñones, gónadas). Los cnidarios son
\emph{diblásticos}; el resto, \emph{triblásticos}. Entre los animales
triblásticos, el mesodermo puede rellenar el espacio entre el tubo digestivo
y la pared del cuerpo (\emph{acelomados}: platelmintos) o dejar una cavidad
entre ambos: un \emph{celoma}\index{celoma} cuando la cavidad está revestida
enteramente de mesodermo (anélidos, moluscos, artrópodos, equinodermos,
vertebrados). El celoma suspende el tubo digestivo en un compartimento
propio, le permite moverse con independencia de la pared del cuerpo y da a
los animales de cuerpo blando un \omterm{def:b1:body-plans-tissues:segments}{esqueleto hidrostático}.
\end{definition}

\begin{definition}[Protóstomos y deuteróstomos]\label{def:b1:body-plans-tissues:proto}
Entre los animales celomados, el destino de la primera abertura del tubo
digestivo embrionario separa dos grandes linajes: en los
\emph{protóstomos}\index{protóstomo} (anélidos, moluscos, artrópodos) esa
abertura se convierte en la boca, el \omterm{def:b1:body-plans-tissues:layers}{celoma} se forma por escisión del
\omterm{def:b1:body-plans-tissues:layers}{mesodermo} y las primeras divisiones del huevo son espirales y de destino
fijado pronto; en los \emph{deuteróstomos}\index{deuteróstomo}
(equinodermos, vertebrados) se convierte en el ano, el \omterm{def:b1:body-plans-tissues:layers}{celoma} se forma por
evaginación del tubo digestivo y el destino de las primeras células sigue
siendo flexible.
\end{definition}

\begin{definition}[Segmentación, esqueletos]\label{def:b1:body-plans-tissues:segments}
Un cuerpo está \emph{segmentado}\index{segmentación} cuando está construido
con unidades repetidas a lo largo de su eje (los anillos de una lombriz de
tierra, los segmentos de un cangrejo de río, las vértebras y los bloques
musculares de un pez), cada una con su propio juego de músculos, nervios y
a menudo apéndices. Los segmentos pueden ser semejantes o especializarse en
regiones (cabeza, tórax, abdomen). El \emph{esqueleto}\index{esqueleto},
contra el que tiran los músculos, es
\emph{hidrostático}\index{esqueleto hidrostático} (un líquido a presión en
un \omterm{def:b1:body-plans-tissues:layers}{celoma}, como en la lombriz), un \emph{exoesqueleto}\index{exoesqueleto}
(una \omterm{def:b1:flowering-plant-organization:tissues}{cutícula} rígida por fuera del cuerpo, articulada, que se muda para
crecer, como en los artrópodos) o un
\emph{endoesqueleto}\index{endoesqueleto} (cartílago o hueso dentro del
cuerpo, que crece con él, como en los vertebrados).
\end{definition}

\begin{proposition}[Cuatro planes comparados]\label{prop:b1:body-plans-tissues:four}
\begin{center}
\footnotesize
\begin{tabular}{lllll}
\hline
 & hidra & lombriz & cangrejo & rana \\
\hline
simetría & radial & bilateral & bilateral & bilateral \\
\omterm{def:b1:body-plans-tissues:layers}{hojas embrionarias} & 2 & 3 & 3 & 3 \\
\omterm{def:b1:body-plans-tissues:layers}{celoma} & ninguno & sí & reducido & sí \\
linaje & --- & \omterm{def:b1:body-plans-tissues:proto}{protóstomo} & \omterm{def:b1:body-plans-tissues:proto}{protóstomo} & \omterm{def:b1:body-plans-tissues:proto}{deuteróstomo} \\
segmentos & no & semejantes & especializados & vértebras, bloques musculares \\
esqueleto & \omterm{def:b1:body-plans-tissues:segments}{hidrostático} & \omterm{def:b1:body-plans-tissues:segments}{hidrostático} & \omterm{def:b1:body-plans-tissues:segments}{exoesqueleto} & \omterm{def:b1:body-plans-tissues:segments}{endoesqueleto} \\
tubo digestivo & una abertura & de boca a ano & de boca a ano & de boca a ano \\
circulación & ninguna & cerrada & abierta & cerrada, con corazón \\
sistema nervioso & difuso & cordón ventral & cordón ventral & cordón dorsal \\
\hline
\end{tabular}
\end{center}
Los planes no son peldaños de una escalera: cada uno es una manera completa
y lograda de ser animal, y el cangrejo de río y la rana son contemporáneos
el uno del otro, no antepasados.
\end{proposition}

\begin{example}[Leer un animal a partir de su plan]\label{ex:b1:body-plans-tissues:reading}
Un animal con un extremo cefálico, un tubo digestivo completo, un cuerpo de
anillos semejantes y sin partes duras: un \omterm{def:b1:body-plans-tissues:proto}{protóstomo} celomado que se
desplaza con un \omterm{def:b1:body-plans-tissues:segments}{esqueleto hidrostático}, en el que el líquido celomático de
cada segmento lo presurizan por turnos los músculos circulares y los
longitudinales; una lombriz de tierra excavando. Un animal con una \omterm{def:b1:flowering-plant-organization:tissues}{cutícula}
articulada y segmentos especializados: un artrópodo, cuyo crecimiento debe
pasar por mudas. Bloques musculares bilaterales y \omterm{def:b1:body-plans-tissues:segments}{segmentados} en torno a un
eje dorsal de vértebras: un vertebrado, cuyo \omterm{def:b1:body-plans-tissues:segments}{endoesqueleto} crece de forma
continua.
\end{example}

\section{Los cuatro tipos de tejido}

\begin{definition}[Tejido]\label{def:b1:body-plans-tissues:tissue}
Un \emph{tejido}\index{tejido} es un conjunto de células de estructura y
origen semejantes, junto con el material extracelular que producen,
organizado para desempeñar una función común. Los tejidos animales se
reparten en cuatro tipos: epitelial, conjuntivo, muscular y nervioso. Todo
\omterm{def:b1:mammal-organization:organ}{órgano} es una combinación de al menos dos de ellos.
\end{definition}

\begin{definition}[Tejido epitelial]\label{def:b1:body-plans-tissues:epithelium}
Un \emph{epitelio}\index{epitelio} es una lámina de células muy juntas que
recubre una superficie o tapiza una cavidad, apoyada sobre una capa fina de
\omterm{def:b1:body-plans-tissues:connective}{matriz extracelular}, la \emph{lámina basal}\index{lámina basal}. Sus células
están \emph{polarizadas}\index{polaridad celular}: la cara apical mira al
exterior o a la luz, la cara basal a la lámina, y ambas llevan proteínas
distintas. Las une un conjunto de \emph{uniones
celulares}\index{unión celular}: las \emph{uniones estrechas}, cerca del
ápice, que sellan el espacio entre células de modo que nada atraviesa la
lámina si no es por dentro de las células; las \emph{uniones adherentes} y
los \emph{desmosomas}, que sujetan mecánicamente las células; y las
\emph{uniones comunicantes}, canales que dejan pasar iones y moléculas
pequeñas de una célula a otra. Los epitelios se clasifican por el número de
capas (simples, estratificados) y por la forma de las células (planas,
cúbicas, cilíndricas), e incluyen las glándulas, que son células
epiteliales especializadas en la secreción. Recubren, protegen, absorben y
secretan.
\end{definition}

\begin{definition}[Tejido conjuntivo]\label{def:b1:body-plans-tissues:connective}
El \emph{tejido conjuntivo}\index{tejido conjuntivo} está formado por
células dispersas en una abundante \emph{matriz extracelular}\index{matriz extracelular} que ellas mismas secretan: fibras proteicas ---
\emph{colágeno}\index{colágeno} (resistencia a la tracción; la cuarta parte
de toda la proteína de un mamífero) y \emph{elastina}\index{elastina}
(recuperación elástica) --- embebidas en una \emph{sustancia fundamental}
hidratada de polisacáridos. Las proporciones definen sus clases: \omterm{def:b1:body-plans-tissues:tissue}{tejido}
conjuntivo laxo bajo los \omterm{def:b1:body-plans-tissues:epithelium}{epitelios} y alrededor de los \omterm{def:b1:mammal-organization:organ}{órganos} (fibroblastos,
pocas fibras, mucha sustancia fundamental); \omterm{def:b1:body-plans-tissues:tissue}{tejido} conjuntivo denso de
tendones y ligamentos (colágeno paralelo); \omterm{def:b1:body-plans-tissues:tissue}{tejido} adiposo (células que
almacenan grasa); cartílago (un gel firme de colágeno y polisacáridos, sin
vasos); hueso (colágeno mineralizado con fosfato de calcio, células vivas en
cavidades, vasos); y la sangre, cuya matriz es el \omterm{def:b1:mammal-organization:compartments}{plasma}. Une, sostiene,
almacena, transporta y defiende.
\end{definition}

\begin{definition}[Tejido muscular]\label{def:b1:body-plans-tissues:muscle}
El \emph{tejido muscular}\index{tejido muscular} está formado por células
alargadas repletas de filamentos proteicos contráctiles (actina y miosina).
Hay tres clases: el \emph{músculo esquelético}, de fibras largas y
plurinucleadas con estriación transversal, unido al esqueleto y de control
voluntario; el \emph{músculo cardíaco}, de células estriadas ramificadas
unidas por sus extremos mediante uniones que transmiten el latido de célula
a célula, y que se contrae a su propio ritmo; y el \emph{músculo liso}, de
células fusiformes sin estrías, en las paredes de los \omterm{def:b1:mammal-organization:organ}{órganos} huecos (tubo
digestivo, vasos, útero), que se contrae despacio y de forma involuntaria.
El mecanismo de la contracción se trata en el volumen del segundo año.
\end{definition}

\begin{definition}[Tejido nervioso]\label{def:b1:body-plans-tissues:nervous}
El \emph{tejido nervioso}\index{tejido nervioso} está formado por
\emph{neuronas}\index{neurona}, células con un cuerpo y prolongaciones
largas (las dendritas, que reciben señales, y un axón, que las conduce hacia
fuera) que conducen señales eléctricas y las transmiten a otras células en
las sinapsis, y por \emph{células gliales}\index{célula glial}, varias veces
más numerosas, que sostienen, aíslan y nutren a las neuronas. Percibe,
integra y ordena.
\end{definition}

\begin{example}[De dónde vienen los tejidos]\label{ex:b1:body-plans-tissues:origin}
Los \omterm{def:b1:body-plans-tissues:epithelium}{epitelios} derivan de las tres \omterm{def:b1:body-plans-tissues:layers}{hojas embrionarias} (la piel del
\omterm{def:b1:body-plans-tissues:layers}{ectodermo}, el revestimiento digestivo del \omterm{def:b1:body-plans-tissues:layers}{endodermo}, el revestimiento de los
vasos y del \omterm{def:b1:body-plans-tissues:layers}{celoma} del \omterm{def:b1:body-plans-tissues:layers}{mesodermo}); el \omterm{def:b1:body-plans-tissues:connective}{tejido conjuntivo}, el músculo y la
sangre, del \omterm{def:b1:body-plans-tissues:layers}{mesodermo}; el \omterm{def:b1:body-plans-tissues:nervous}{tejido nervioso}, del \omterm{def:b1:body-plans-tissues:layers}{ectodermo}. La piel de la rana
es un \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} ectodérmico sobre \omterm{def:b1:body-plans-tissues:connective}{tejido conjuntivo} mesodérmico; su tubo
digestivo, un \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} endodérmico envuelto en músculo mesodérmico; su
encéfalo, \omterm{def:b1:body-plans-tissues:layers}{ectodermo} plegado hacia dentro.
\end{example}

\section{De los tejidos a un órgano: la pared del tubo digestivo}

\begin{proposition}[Las cuatro capas de la pared digestiva]\label{prop:b1:body-plans-tissues:gutwall}
\index{pared digestiva}
Desde la luz hacia fuera, la pared del tubo digestivo de los vertebrados
está construida con cuatro capas concéntricas, cada una una combinación de
\omterm{def:b1:body-plans-tissues:tissue}{tejidos}: la \emph{mucosa} (un \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} --- absorbente y secretor --- sobre
un \omterm{def:b1:body-plans-tissues:connective}{tejido conjuntivo} laxo que lleva vasos sanguíneos y linfáticos, con una
lámina delgada de músculo liso por debajo); la \emph{submucosa} (\omterm{def:b1:body-plans-tissues:connective}{tejido
conjuntivo} denso con vasos mayores y una red nerviosa); la \emph{capa
muscular} (dos estratos de músculo liso, circular el interno y longitudinal
el externo, con una red nerviosa entre ellos, que producen los movimientos
de mezcla y de propulsión); y la \emph{serosa} (una capa conjuntiva delgada
cubierta por el \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} del \omterm{def:b1:body-plans-tissues:layers}{celoma}). Las mismas cuatro capas van del
esófago al recto; lo que cambia es la mucosa: plegada en vellosidades en el
intestino delgado, glandular en el estómago.
\end{proposition}

\begin{method}[Lectura de una sección histológica]\label{met:b1:body-plans-tissues:histology}
\begin{enumerate}
  \item Búsquese la superficie libre o la luz: el \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} que la tapiza es
        la primera capa. Cuéntense sus filas de núcleos (simple o
        estratificado) y obsérvese la forma de las células.
  \item Bajo el \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} búsquese la zona pálida y fibrosa con núcleos
        dispersos y vasos: es el \omterm{def:b1:body-plans-tissues:connective}{tejido conjuntivo}.
  \item El músculo aparece como haces densos de células alargadas: estriado
        con núcleos periféricos (esquelético), estriado y ramificado
        (cardíaco), sin estrías y con núcleo central (liso). Obsérvese la
        dirección de las fibras respecto de la sección.
  \item El \omterm{def:b1:body-plans-tissues:nervous}{tejido nervioso} en la pared de un \omterm{def:b1:mammal-organization:organ}{órgano} aparece como grupos de
        cuerpos neuronales grandes y pálidos entre los estratos musculares.
  \item Nómbrese el \omterm{def:b1:mammal-organization:organ}{órgano} a partir de la secuencia de capas y de las
        especializaciones de la mucosa (vellosidades, glándulas, queratina).
\end{enumerate}
\end{method}

\begin{example}[La vellosidad como superficie de intercambio]\label{ex:b1:body-plans-tissues:villus}
Una vellosidad del intestino delgado humano es un dedo de \qty{0.6}{mm} de
alto y \qty{0.15}{mm} de ancho, cubierto por una sola capa de células
cilíndricas cuyas caras apicales llevan cada una unas \qty{1500}{}
microvellosidades. Las vellosidades multiplican la superficie del tubo unas
siete veces y las microvellosidades otras veinte: los \qty{0.6}{m^2} de un
tubo liso se convierten en decenas de metros cuadrados de \omterm{def:b1:body-plans-tissues:epithelium}{epitelio}
absorbente, de una célula de espesor, cada célula a unos pocos micrómetros
de un capilar y de un vaso linfático en el eje conjuntivo de su vellosidad
(\cref{ch:b1:digestion-absorption}). El \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} se renueva desde las
glándulas de su base cada cuatro días: la lámina, sus uniones y su polaridad
se reconstruyen sin cesar.
\end{example}

\begin{remark}[Los tejidos de los dos organismos modelo]\label{rem:b1:body-plans-tissues:plants}
Los tres sistemas de \omterm{def:b1:body-plans-tissues:tissue}{tejidos} de la planta del
\cref{ch:b1:flowering-plant-organization} no tienen aquí equivalentes
exactos. Su epidermis es un \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} por su función, pero sus células se
sujetan por paredes, no por uniones; su \omterm{def:b1:flowering-plant-organization:tissues}{tejido fundamental} es a la vez de
reserva, de sostén y fotosintético; y no tiene ni músculo ni nervio. Los
cuatro \omterm{def:b1:body-plans-tissues:tissue}{tejidos} del animal son la caja de herramientas de un cuerpo que se
mueve y percibe; los tres de la planta, la de un cuerpo que crece.
\end{remark}

\section{Problema: la pared del intestino delgado}

\begin{problem}[{Problema de fin de semana --- una serie histológica del
intestino delgado humano leída desde el tubo hasta la microvellosidad:
capas, vellosidades, células, uniones y renovación, hasta llegar a la
superficie absorbente de una sola vellosidad}]
\label{pb:b1:body-plans-tissues:1}
El intestino delgado humano es un tubo de \qty{6}{m} de largo y \qty{3}{cm}
de diámetro interno. Los pliegues circulares de su pared triplican la
superficie del tubo liso. Su mucosa lleva \qty{25}{} vellosidades por
milímetro cuadrado de superficie plegada, cada una un cilindro de
\qty{0.6}{mm} de alto y \qty{0.15}{mm} de diámetro. Las vellosidades están
cubiertas por células cilíndricas cuya cara apical es un cuadrado de
\qty{5}{\micro m} de lado; cada cara apical lleva \qty{1500}{}
microvellosidades, cilindros de \qty{1}{\micro m} de largo y
\qty{0.1}{\micro m} de diámetro. El \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} se renueva cada \qty{4}{}
días.

\medskip
\noindent\textbf{Parte I --- Las capas.}
Se examina al microscopio una sección transversal.
\begin{enumerate}
  \item Enumérense las cuatro capas de la pared desde la luz hacia fuera y
        los tipos de \omterm{def:b1:body-plans-tissues:tissue}{tejido} de cada una.
  \item La sección muestra vellosidades. ¿De qué región del tubo digestivo
        se trata? ¿Qué mostraría en su lugar una sección de estómago?
  \item Entre los dos estratos musculares de la capa muscular hay grupos de
        células grandes y pálidas. ¿Qué son y qué controlan?
  \item Explíquese por qué los dos estratos musculares corren en
        direcciones perpendiculares.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Superficies.}
\begin{enumerate}[resume]
  \item Calcúlese la superficie interna del tubo liso.
  \item Calcúlese la superficie después de los pliegues circulares.
  \item Calcúlese la superficie lateral de una vellosidad (cilindro, sin
        contar el extremo).
  \item Calcúlense la superficie de vellosidades por milímetro cuadrado de
        mucosa plegada y el factor por el que las vellosidades multiplican
        la superficie.
  \item Calcúlese la superficie después de las vellosidades.
  \item Calcúlense la superficie lateral de una microvellosidad y la de
        las \qty{1500}{} de una célula.
  \item Calcúlese el factor por el que las microvellosidades multiplican la
        superficie apical de una célula.
  \item Calcúlese la superficie absorbente total del intestino delgado.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Células.}
\begin{enumerate}[resume]
  \item Calcúlese el número de células epiteliales que cubren una
        vellosidad.
  \item Calcúlense el número de vellosidades del intestino delgado y el
        número de células epiteliales que las cubren.
  \item El \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} se renueva cada \qty{4}{} días. ¿Cuántas células se
        desprenden hacia la luz al día, y por segundo?
  \item ¿Dónde se producen las células de recambio, y qué dice esto sobre
        la posición de las células que se dividen respecto de las que
        absorben?
  \item Una célula epitelial mide \qty{25}{\micro m} de alto. Calcúlense
        el volumen de \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} del intestino delgado y su masa (densidad
        \qty{1.05}{g/cm^3}).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Uniones, matriz y vellosidad.}
\begin{enumerate}[resume]
  \item Cada célula está sellada a sus vecinas por una unión estrecha que
        rodea su ápice. Calcúlense la longitud de unión estrecha por
        célula y la longitud total en el intestino delgado.
  \item Explíquese por qué una sustancia de la luz solo puede llegar a la
        sangre atravesando dos membranas celulares.
  \item El eje de la vellosidad es \omterm{def:b1:body-plans-tissues:connective}{tejido conjuntivo} laxo con una red
        capilar y un vaso linfático central. Nómbrense las dos clases de
        fibra extracelular que contiene y la célula que las secreta.
  \item Nómbrese la \omterm{def:b1:body-plans-tissues:layers}{hoja embrionaria} de origen del \omterm{def:b1:body-plans-tissues:epithelium}{epitelio} y la del eje.
  \item La vellosidad contiene además un cordón de músculo liso que la hace
        acortarse rítmicamente. Propóngase su función.
  \item Una enfermedad aplana las vellosidades (la superficie vuelve a ser
        la del tubo plegado). ¿En qué factor cae la superficie absorbente,
        y cuáles son las consecuencias?
  \item Calcúlese la superficie absorbente de una vellosidad con sus
        microvellosidades, en milímetros cuadrados.
  \item Enúnciese el resultado: la superficie absorbente de una sola
        vellosidad y el número de vellosidades del intestino delgado.
\end{enumerate}
\end{problem}
