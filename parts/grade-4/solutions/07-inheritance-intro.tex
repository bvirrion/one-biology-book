\section*{Chapter \ref{ch:g4:traits} --- Traits Run in Families}

\begin{solution}{exo:g4:traits:1}
A \omterm{def:g4:traits:trait}{trait} is a feature of a \omterm{def:g1:living-nonliving:living}{living thing}. Examples: \omterm{def:g1:senses:five}{eye} colour, height,
\omterm{def:g2:plant-parts:main}{flower} colour, \omterm{def:g4:plants-need:parts}{leaf} shape.
\end{solution}

\begin{solution}{exo:g4:traits:2}
\omterm{def:g4:traits:inherited}{Inherited traits} are passed from parents to \omterm{def:g2:growing:offspring}{offspring}. \omterm{def:g4:traits:acquired}{Acquired traits}
develop during life through experience or \omterm{def:g4:traits:env}{environment} and are not
passed on biologically in the same way.
\end{solution}

\begin{solution}{exo:g4:traits:3}
No. A scar is acquired from injury during life, not inherited.
\end{solution}

\begin{solution}{exo:g4:traits:4}
They inherit from the same parents but not identical combinations of
information, and they may have different experiences and \omterm{def:g4:traits:env}{environments}.
\end{solution}

\begin{solution}{exo:g4:traits:5}
Inherited (mainly): \omterm{def:g1:senses:five}{eye} colour, freckles, natural hair texture.
Acquired: speaking Spanish, scar from a break. (Freckles also interact
with sun exposure.)
\end{solution}
