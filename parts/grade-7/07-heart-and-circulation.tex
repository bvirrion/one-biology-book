\chapter{The Heart and Circulation}\label{ch:g7:heart-and-circulation}

\Cref{ch:g4:heart-and-blood} introduced the pump and its
delivery rounds; since then the \omterm{prop:g4:journey-of-food:absorb}{blood} has acquired
responsibilities --- \omterm{def:g7:digestion-and-nutrients:families}{nutrients} at the \omterm{prop:g7:digestion-and-nutrients:absorption}{villi}
(\cref{prop:g7:digestion-and-nutrients:absorption}), \omterm{def:g7:what-is-respiration:respiration}{oxygen}
at the \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveoli}
(\cref{prop:g7:lungs-and-gas-exchange:numbers}), bills at
every working \omterm{def:g3:bones-and-muscles:muscle}{muscle} (\cref{prop:g7:organs-at-work:bill}).
Time to map the road network properly: the \omterm{def:g4:heart-and-blood:heart}{heart}'s four
rooms, the two circuits, and the three kinds of vessel.

\section{The heart's four rooms}

\begin{definition}[Atria and ventricles]\label{def:g7:heart-and-circulation:rooms}
The \omterm{def:g4:heart-and-blood:heart}{heart}'s two sides (\cref{def:g4:heart-and-blood:heart})
each hold two rooms: an upstairs
\emph{atrium}\index{atrium} that receives incoming \omterm{prop:g4:journey-of-food:absorb}{blood},
and a downstairs \emph{ventricle}\index{ventricle} --- the
thick-walled pumping room --- that drives it out. Between
and beyond them, \emph{valves}\index{valve (heart)} ---
one-way doors --- snap shut behind each surge; their
paired snaps are exactly the ``lub-dub'' a stethoscope (or
an \omterm{def:g1:the-five-senses:sense}{ear} on a chest) hears.
\end{definition}

\begin{proposition}[One beat, in order]\label{prop:g7:heart-and-circulation:beat}
Each beat runs the same sequence on both sides at once:
the atria contract, topping up the \omterm{def:g7:heart-and-circulation:rooms}{ventricles}; the
\omterm{def:g7:heart-and-circulation:rooms}{ventricles} contract, driving \omterm{prop:g4:journey-of-food:absorb}{blood} out --- right side
toward the \omterm{def:g7:how-animals-breathe:four}{lungs}, left side toward the body --- while the
entry \omterm{def:g7:heart-and-circulation:rooms}{valves} slam (lub); the \omterm{def:g7:heart-and-circulation:rooms}{ventricles} relax, the exit
\omterm{def:g7:heart-and-circulation:rooms}{valves} slam (dub), and the rooms refill. Then again,
$100000$ times a day
(\cref{rem:g4:heart-and-blood:rest}).
\end{proposition}
\admitted

\section{Two circuits, three vessels}

\begin{definition}[The double circulation]\label{def:g7:heart-and-circulation:double}
The \omterm{prop:g4:journey-of-food:absorb}{blood}'s road network is two loops through one pump
(\cref{def:g4:heart-and-blood:heart}'s two sides, now as
circuits):
\begin{itemize}
  \item the \emph{\omterm{def:g7:how-animals-breathe:four}{lung} circuit}: right \omterm{def:g7:heart-and-circulation:rooms}{ventricle}
        $\rightarrow$ \omterm{def:g7:how-animals-breathe:four}{lungs} (loading \omterm{def:g7:what-is-respiration:respiration}{oxygen}, unloading
        \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide}) $\rightarrow$ left \omterm{def:g7:heart-and-circulation:rooms}{atrium};
  \item the \emph{body circuit}: left \omterm{def:g7:heart-and-circulation:rooms}{ventricle}
        $\rightarrow$ the whole body (delivering,
        collecting) $\rightarrow$ right \omterm{def:g7:heart-and-circulation:rooms}{atrium}.
\end{itemize}
Every drop alternates circuits: \omterm{def:g7:how-animals-breathe:four}{lungs}, body, \omterm{def:g7:how-animals-breathe:four}{lungs}, body
--- which is why the left \omterm{def:g7:heart-and-circulation:rooms}{ventricle}, pushing round the
far larger body loop, is the \omterm{def:g4:heart-and-blood:heart}{heart}'s thickest-walled
room.
\end{definition}

\begin{definition}[Arteries, capillaries, veins]\label{def:g7:heart-and-circulation:vessels}
Three kinds of vessel make each loop
(\cref{def:g4:heart-and-blood:vessels}, now named):
\begin{itemize}
  \item \emph{arteries}\index{artery}: thick-walled,
        elastic, carrying \omterm{prop:g4:journey-of-food:absorb}{blood} \emph{away} from the
        \omterm{def:g4:heart-and-blood:heart}{heart} under pressure --- the \omterm{prop:g4:heart-and-blood:pulse}{pulse}
        (\cref{prop:g4:heart-and-blood:pulse}) is their
        stretch at each surge;
  \item \emph{capillaries}\index{capillary}: the
        hair-fine \omterm{def:g4:heart-and-blood:vessels}{vessels} threading every organ ---
        walls one \omterm{def:g5:first-look-at-cells:cell}{cell} thin, where \emph{all} the
        exchanges happen: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveoli}, \omterm{prop:g7:digestion-and-nutrients:absorption}{villi}, \omterm{def:g3:bones-and-muscles:muscle}{muscles}, \omterm{def:g1:the-five-senses:sense}{skin};
  \item \emph{veins}\index{vein}: wide, thinner-walled,
        returning \omterm{prop:g4:journey-of-food:absorb}{blood} \emph{to} the \omterm{def:g4:heart-and-blood:heart}{heart} at low
        pressure, with one-way \omterm{def:g7:heart-and-circulation:rooms}{valves} of their own ---
        the bluish lines of
        \cref{rem:g4:heart-and-blood:see}.
\end{itemize}
\end{definition}

\begin{remark}[Why capillaries are the point]\label{rem:g7:heart-and-circulation:capillaries}
\omterm{def:g7:heart-and-circulation:vessels}{Arteries} and \omterm{def:g7:heart-and-circulation:vessels}{veins} are only highways; every actual
delivery --- every exchange this year has studied ---
happens across \omterm{def:g7:heart-and-circulation:vessels}{capillary} walls. The specification of
\cref{prop:g7:how-animals-breathe:spec} lives here too:
\omterm{def:g5:first-look-at-cells:cell}{one-cell} walls (thin), threading every tissue by the
tens of thousands of kilometres
(\cref{def:g4:heart-and-blood:vessels} --- large), bathed
in the body's moisture. The \omterm{def:g4:heart-and-blood:heart}{heart} exists to keep \omterm{prop:g4:journey-of-food:absorb}{blood}
moving \emph{through \omterm{def:g7:heart-and-circulation:vessels}{capillaries}}; all the rest is
plumbing to and from.
\end{remark}

\section{Reading and keeping the network}

\begin{example}[Measurements at the roadside]\label{ex:g7:heart-and-circulation:measures}
Three everyday readings of the network: the \emph{\omterm{prop:g4:heart-and-blood:pulse}{pulse}}
--- \omterm{def:g7:heart-and-circulation:vessels}{artery} stretch, counting the surges
(\cref{met:g4:heart-and-blood:pulse}); \emph{\omterm{prop:g4:journey-of-food:absorb}{blood}
pressure}, the cuff's two numbers --- the \omterm{def:g7:heart-and-circulation:vessels}{artery} pressure
at surge and between surges; and the stethoscope's
\emph{lub-dub} --- the \omterm{def:g7:heart-and-circulation:rooms}{valves}' doors heard closing. Three
windows, no incision.
\end{example}

\begin{proposition}[Keeping the network]\label{prop:g7:heart-and-circulation:keep}
The network's long-term enemies and friends are known:
\begin{itemize}
  \item \emph{exercise} strengthens the pump
        (\cref{prop:g7:organs-at-work:training}) and keeps
        the highways supple;
  \item \emph{balanced eating}
        (\cref{met:g3:balanced-diet:check}) protects the
        \omterm{def:g7:heart-and-circulation:vessels}{artery} walls, which too-rich \omterm{prop:g4:journey-of-food:absorb}{blood} slowly
        \omterm{def:g1:animals-around-us:covering}{furs} up and stiffens;
  \item \emph{smoke} attacks \omterm{def:g4:heart-and-blood:vessels}{vessels} as well as \omterm{def:g7:how-animals-breathe:four}{lungs}
        (\cref{prop:g7:lungs-and-gas-exchange:smoke}) ---
        narrowing, stiffening, seat-stealing;
  \item the \omterm{def:g4:heart-and-blood:heart}{heart}'s own supply --- it feeds by its own
        small \omterm{def:g7:heart-and-circulation:vessels}{arteries}, not from the \omterm{prop:g4:journey-of-food:absorb}{blood} in its rooms
        --- depends on all three.
\end{itemize}
The habits of \cref{prop:g5:healthy-choices:contract} are,
concretely, vessel maintenance.
\end{proposition}
\admitted

\begin{method}[Mapping any organ into the network]\label{met:g7:heart-and-circulation:map}
For any organ this book meets:
\begin{enumerate}
  \item place it on the body circuit: \omterm{def:g7:heart-and-circulation:vessels}{artery} in, \omterm{def:g7:heart-and-circulation:vessels}{vein}
        out, \omterm{def:g7:heart-and-circulation:vessels}{capillaries} within;
  \item list what crosses its \omterm{def:g7:heart-and-circulation:vessels}{capillary} walls, in and
        out (\omterm{prop:g7:digestion-and-nutrients:absorption}{villi}: \omterm{def:g7:digestion-and-nutrients:families}{nutrients} in; \omterm{def:g3:bones-and-muscles:muscle}{muscle}: fuel and
        \omterm{def:g7:what-is-respiration:respiration}{oxygen} in, \omterm{def:g7:what-is-respiration:respiration}{carbon dioxide} out; \omterm{def:g1:the-five-senses:sense}{skin}: heat out);
  \item note its special connection, if any (intestine's
        \omterm{prop:g4:journey-of-food:absorb}{blood} calls at the \omterm{prop:g7:digestion-and-nutrients:absorption}{liver} first; the \omterm{def:g7:how-animals-breathe:four}{lungs} sit on
        their own circuit);
  \item check the flows against the organ's job --- the
        network diagram should retell the organ's
        chapter.
\end{enumerate}
\end{method}

\section{Problem: The Cardiologist's Morning}

\begin{problem}[{Weekend problem --- four consultations,
one map of the network}]\label{pb:g7:heart-and-circulation:1}
Shadow a cardiologist for a morning of four (invented)
consultations, map in hand.

\medskip\noindent\textbf{Part I --- The healthy
baseline.}
\begin{enumerate}
  \item Patient 1, a fit teenager, gets the three
        roadside readings of
        \cref{ex:g7:heart-and-circulation:measures}.
        Name each, and what part of the machinery each
        reads.
  \item Her resting \omterm{prop:g4:heart-and-blood:pulse}{pulse} is $58$. The cardiologist
        smiles. Why
        (\cref{exo:g4:heart-and-blood:11})?
  \item Her lub-dub is crisp and double. What exactly
        snapped, twice, in every beat?
  \item Sketch her \omterm{prop:g4:journey-of-food:absorb}{blood}'s path from right \omterm{def:g7:heart-and-circulation:rooms}{atrium} round
        both circuits and back, in eight named stops.
\end{enumerate}

\medskip\noindent\textbf{Part II --- Three troubles.}
\begin{enumerate}[resume]
  \item Patient 2 has a leaky valve between left \omterm{def:g7:heart-and-circulation:rooms}{atrium}
        and \omterm{def:g7:heart-and-circulation:rooms}{ventricle}: some \omterm{prop:g4:journey-of-food:absorb}{blood} surges backward at
        each beat, with a whooshing murmur. Predict the
        consequences for the body circuit's deliveries
        and the patient's stamina.
  \item Patient 3, a heavy smoker with a rich diet, has
        furred, stiffened \omterm{def:g7:heart-and-circulation:vessels}{arteries}. Which readings of
        consultation 1 will have worsened, and why does
        the left \omterm{def:g7:heart-and-circulation:rooms}{ventricle} work overtime?
  \item Patient 3's furring includes the \omterm{def:g4:heart-and-blood:heart}{heart}'s own
        feeding \omterm{def:g7:heart-and-circulation:vessels}{arteries}. Explain the cardiologist's
        special concern in one sentence.
  \item Patient 4 has swollen \omterm{def:g1:my-body:joint}{ankles} after long
        standing days: the \omterm{def:g7:heart-and-circulation:vessels}{veins}' uphill return is
        struggling. Which structures are failing, and
        why does walking --- calf \omterm{def:g3:bones-and-muscles:muscle}{muscles} squeezing the
        \omterm{def:g7:heart-and-circulation:vessels}{veins} --- relieve it?
\end{enumerate}

\medskip\noindent\textbf{Part III --- The wall chart.}
\begin{enumerate}[resume]
  \item The clinic's wall chart shows the double loop.
        Explain to a waiting patient why \omterm{prop:g4:journey-of-food:absorb}{blood} must
        visit the \omterm{def:g7:how-animals-breathe:four}{lungs} between every two body tours.
  \item Add the vessel legend: \omterm{def:g7:heart-and-circulation:vessels}{artery}, \omterm{def:g7:heart-and-circulation:vessels}{capillary}, \omterm{def:g7:heart-and-circulation:vessels}{vein}
        --- one line each on build, pressure and job.
  \item The chart's footer reads: ``all treatment
        serves the \omterm{def:g7:heart-and-circulation:vessels}{capillaries}.'' Justify it with
        \cref{rem:g7:heart-and-circulation:capillaries}.
  \item Close the morning with the prevention
        prescription the cardiologist hands every
        patient --- three habits, each tied to its
        mechanism in
        \cref{prop:g7:heart-and-circulation:keep}.
\end{enumerate}
\end{problem}
