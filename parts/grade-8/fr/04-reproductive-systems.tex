\chapter{Les appareils reproducteurs}\label{ch:g8:reproductive-systems}

\Cref{ch:g8:sexual-reproduction} avait besoin de \omterm{def:g8:sexual-reproduction:gametes}{gamètes}~;
\cref{ch:g8:puberty} a mis leurs usines en marche. Ce chapitre
cartographie les usines elles-mêmes~: les appareils reproducteurs
masculin et féminin --- les organes, le \omterm{def:g8:sexual-reproduction:gametes}{gamète} que chacun produit,
et le remarquable rythme mensuel de l'appareil féminin. De
l'anatomie simple, nommée simplement~: ce sont des organes comme le
\omterm{def:g4:heart-and-blood:heart}{cœur} et les \omterm{def:g7:how-animals-breathe:four}{poumons}, et voici leur chapitre.

\section{L'appareil masculin}

\begin{definition}[L'appareil reproducteur masculin]\label{def:g8:reproductive-systems:male}
Les organes de l'appareil masculin~:
\begin{itemize}
  \item les deux \emph{testicules}\index{testicule}, tenus hors de
        la chaleur du corps dans les \emph{bourses} --- les usines à
        \omterm{def:g8:sexual-reproduction:gametes}{gamètes}~: à partir de la \omterm{def:g8:puberty:puberty}{puberté}, ils produisent en continu
        des \emph{cellules mâles}\index{cellule mâle} --- les
        spermatozoïdes --- par centaines de millions~;
  \item des conduits qui stockent et transportent les \omterm{def:g5:first-look-at-cells:cell}{cellules}
        \omterm{def:g4:animal-reproduction:sexual}{mâles}, rejoints en chemin par des glandes dont les liquides
        les nourrissent et les portent --- l'ensemble formant le
        \emph{sperme}~;
  \item le \emph{pénis}\index{pénis}, par lequel le sperme quitte le
        corps --- le même organe qui, dans son autre fonction, évacue
        l'\omterm{def:g7:removing-wastes:kidneys}{urine} (\cref{def:g7:removing-wastes:kidneys})~; un jeu de
        \omterm{def:g7:heart-and-circulation:rooms}{valves} tient les deux fonctions strictement séparées.
\end{itemize}
\end{definition}

\begin{example}[La cellule mâle, de près]\label{ex:g8:reproductive-systems:sperm}
Au \omterm{def:g5:first-look-at-cells:microscope}{microscope}
(\cref{def:g5:first-look-at-cells:microscope}), une \omterm{def:g8:reproductive-systems:male}{cellule mâle} est
une \omterm{def:g5:first-look-at-cells:cell}{cellule} allégée pour le voyage~: une tête presque entièrement
faite du \omterm{def:g6:cell-unit-of-life:parts}{noyau} --- l'apport du père, bien tassé
(\cref{rem:g5:first-look-at-cells:nucleus}) --- et une queue qui
fouette. C'est la plus petite \omterm{def:g5:first-look-at-cells:cell}{cellule} du corps, bâtie pour nager,
pourvue pour quelques jours. La production court de la \omterm{def:g8:puberty:puberty}{puberté} à la
\omterm{def:g3:stages-of-life:stages}{vieillesse}~: l'usine masculine est continue.
\end{example}

\section{L'appareil féminin}

\begin{definition}[L'appareil reproducteur féminin]\label{def:g8:reproductive-systems:female}
Les organes de l'appareil féminin, tous dans le bas-ventre~:
\begin{itemize}
  \item les deux \emph{ovaires}\index{ovaire} --- les réserves et
        les usines à \omterm{def:g8:sexual-reproduction:gametes}{gamètes}~: contrairement aux \omterm{def:g8:reproductive-systems:male}{testicules}, ils
        détiennent dès la \omterm{def:g2:born-grow-die:cycle}{naissance} leur stock de futures
        \emph{\omterm{def:g4:animal-reproduction:sexual}{cellules œufs}} --- les \omterm{prop:g5:flower-to-fruit:roles}{ovules} --- et, à partir de la
        \omterm{def:g8:puberty:puberty}{puberté}, en libèrent une à la fois~;
  \item les deux \emph{trompes}\index{trompe}, des conduits frangés
        qui recueillent la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} libérée --- le lieu habituel
        de la rencontre de l'étape 2 de
        \cref{prop:g8:sexual-reproduction:core}~;
  \item l'\emph{\omterm{prop:g5:human-reproduction:plan}{utérus}} (la matrice de
        \cref{prop:g5:human-reproduction:plan}), dont l'épaisse
        paroi musclée et la muqueuse renouvelable sont le logement
        du \omterm{prop:g8:sexual-reproduction:core}{zygote}~;
  \item le \emph{vagin}, qui relie l'\omterm{prop:g5:human-reproduction:plan}{utérus} à l'extérieur --- où le
        sperme est reçu, et la voie de la \omterm{def:g2:born-grow-die:cycle}{naissance} de
        \cref{ex:g5:human-reproduction:birth}.
\end{itemize}
\end{definition}

\begin{example}[La cellule œuf, de près]\label{ex:g8:reproductive-systems:egg}
La \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} est le contraire de la \omterm{def:g8:reproductive-systems:male}{cellule mâle} en tout, sauf en
rôle~: c'est la \emph{plus grande} \omterm{def:g5:first-look-at-cells:cell}{cellule} du corps --- un dixième
de millimètre, à la limite de la vue --- ronde, immobile, garnie de
provisions pour les premiers jours du voyage (le principe du jaune
de \cref{rem:g2:animals-grow-up:egg}, à l'échelle de la \omterm{def:g5:first-look-at-cells:cell}{cellule}).
Une usine sort des millions en continu~; l'autre libère une seule
\omterm{def:g5:first-look-at-cells:cell}{cellule} bien pourvue par mois~: le «~nombre contre soin~» de
\cref{prop:g4:animal-reproduction:strategies}, inscrit dans les
\omterm{def:g8:sexual-reproduction:gametes}{gamètes} eux-mêmes.
\end{example}

\section{Le cycle}

\begin{proposition}[Le cycle menstruel]\label{prop:g8:reproductive-systems:cycle}
De la \omterm{def:g8:puberty:puberty}{puberté} à cinquante ans environ, l'appareil féminin déroule un
programme qui se répète tous les \emph{28 jours} en gros --- le
\emph{cycle menstruel}\index{cycle menstruel}, commandé par la
chaîne d'\omterm{def:g8:puberty:hormones}{hormones} de \cref{def:g8:puberty:hormones}~:
\begin{enumerate}
  \item la muqueuse de l'\omterm{prop:g5:human-reproduction:plan}{utérus} se reconstruit, épaisse et riche en
        \omterm{prop:g4:journey-of-food:absorb}{sang} --- le logement préparé~;
  \item au milieu du cycle, vers le jour 14, un \omterm{def:g8:reproductive-systems:female}{ovaire} libère une
        \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} --- c'est l'\emph{ovulation}\index{ovulation}~;
        la \omterm{def:g8:reproductive-systems:female}{trompe} la recueille, et pendant environ une journée elle
        peut être fécondée~;
  \item pas de \omterm{def:g5:flower-to-fruit:steps}{fécondation}~: la muqueuse ne sert à rien~; elle se
        détruit et s'évacue par le vagin en quelques jours --- ce
        sont les \emph{règles}\index{règles} --- et le programme
        recommence~;
  \item \omterm{def:g5:flower-to-fruit:steps}{fécondation}~: le \omterm{prop:g8:sexual-reproduction:core}{zygote} s'installe dans la muqueuse
        préparée
        (\cref{def:g5:human-reproduction:pregnancy}), le cycle
        s'interrompt, et la \omterm{def:g5:human-reproduction:pregnancy}{grossesse} commence.
\end{enumerate}
\end{proposition}
\admitted

\begin{remark}[Vivre avec le cycle]\label{rem:g8:reproductive-systems:living}
Des aspects pratiques, dits simplement~: les cycles sont irréguliers
les premières années --- la machinerie se règle~; leur durée varie
d'une personne à l'autre et d'un mois à l'autre~; les règles peuvent
s'accompagner de douleurs (l'\omterm{prop:g5:human-reproduction:plan}{utérus} est un \omterm{def:g3:bones-and-muscles:muscle}{muscle},
\cref{def:g3:bones-and-muscles:muscle}, et il travaille pendant
l'évacuation) --- la chaleur, le mouvement et, si besoin, les
conseils de l'infirmière scolaire aident tous~; et serviettes,
tampons et coupes ne sont que l'équipement pratique des règles, un
rayon ordinaire de toutes les pharmacies. Rien de tout cela n'est
une maladie~: le cycle, c'est l'appareil qui fonctionne exactement
comme il est bâti.
\end{remark}

\begin{method}[Lire les deux appareils côte à côte]\label{met:g8:reproductive-systems:sides}
Pour garder la carte au clair, compare par fonction~:
\begin{enumerate}
  \item usine à \omterm{def:g8:sexual-reproduction:gametes}{gamètes}~: \omterm{def:g8:reproductive-systems:male}{testicules} --- \omterm{def:g8:reproductive-systems:female}{ovaires}~;
  \item \omterm{def:g8:sexual-reproduction:gametes}{gamète}~: des millions de nageuses en continu --- une seule
        \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} bien pourvue par mois~;
  \item lieu de rencontre~: la \omterm{def:g8:reproductive-systems:female}{trompe}, atteinte par la nage des
        \omterm{def:g8:reproductive-systems:male}{cellules mâles} depuis le vagin à travers l'\omterm{prop:g5:human-reproduction:plan}{utérus}~;
  \item logement~: aucun sur la carte masculine --- l'\omterm{prop:g5:human-reproduction:plan}{utérus},
        préparé chaque mois, sur la féminine~;
  \item calendrier~: continu depuis la \omterm{def:g8:puberty:puberty}{puberté} --- cyclique, de la
        \omterm{def:g8:puberty:puberty}{puberté} à cinquante ans environ.
\end{enumerate}
\end{method}

\section{Problème~: les modèles pédagogiques du centre de santé}

\begin{problem}[{Problème du week-end --- expliquer les appareils
avec deux modèles et un calendrier}]\label{pb:g8:reproductive-systems:1}
La journée portes ouvertes d'un centre de santé utilise trois
supports pédagogiques~: un modèle de l'appareil masculin, un modèle
de l'appareil féminin, et un grand calendrier de cycle sur 28 jours.
C'est toi qui expliques, en bénévole.

\medskip\noindent\textbf{Partie I --- Les deux modèles.}
\begin{enumerate}
  \item Une visiteuse suit le modèle masculin des \omterm{def:g8:reproductive-systems:male}{testicules} à la
        sortie. Raconte le trajet, en nommant ce qui s'ajoute en
        chemin et le dispositif de l'organe à double fonction.
  \item Sur le modèle féminin, montre où attend le stock de
        \omterm{def:g8:sexual-reproduction:gametes}{gamètes}, où se produit habituellement la \omterm{def:g5:flower-to-fruit:steps}{fécondation}, et où
        se loge une \omterm{def:g5:human-reproduction:pregnancy}{grossesse}.
  \item Un visiteur demande pourquoi les usines à \omterm{def:g8:sexual-reproduction:gametes}{gamètes} des deux
        modèles diffèrent si nettement --- des millions par jour
        contre une par mois. Réponds avec la phrase de conclusion de
        \cref{ex:g8:reproductive-systems:egg}.
  \item Un autre demande ce que faisaient les appareils avant la
        \omterm{def:g8:puberty:puberty}{puberté}. Réponds avec la machinerie de
        \cref{ch:g8:puberty}, en une ligne.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Le calendrier.}
\begin{enumerate}[resume]
  \item Parcours le calendrier~: place les règles, la
        reconstruction, l'\omterm{prop:g8:reproductive-systems:cycle}{ovulation} et le jour fécond, avec leurs
        dates approximatives.
  \item Une visiteuse suppose que le jour 28 est le même pour tout
        le monde. Corrige cette supposition avec
        \cref{rem:g8:reproductive-systems:living} --- deux sources
        de variation.
  \item Montre sur le calendrier ce qui change si la \omterm{def:g5:flower-to-fruit:steps}{fécondation} a
        lieu vers le jour 14 --- quels événements sont annulés,
        lesquels commencent
        (\cref{def:g5:human-reproduction:pregnancy} poursuit
        l'histoire).
  \item Une adolescente demande, discrètement, si des douleurs de
        règles signifient que quelque chose ne va pas. Donne la
        réponse du \omterm{def:g3:bones-and-muscles:muscle}{muscle} et les deux aides --- et l'adresse pour
        tout ce qui les dépasse.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le métier de celui qui
explique.}
\begin{enumerate}[resume]
  \item Des visiteurs ricanent devant les modèles. Ta phrase
        d'ouverture traite les appareils comme le fait le premier
        paragraphe de
        \cref{ch:g8:reproductive-systems}. Rédige-la.
  \item Résume les deux appareils sous la forme de la comparaison
        en cinq lignes de
        \cref{met:g8:reproductive-systems:sides}, de mémoire.
  \item Un visiteur demande où se situe, sur les supports, le début
        des bébés. Enchaîne les trois supports avec les trois étapes
        de \cref{prop:g8:sexual-reproduction:core}.
  \item Termine la journée par une phrase sur la raison pour
        laquelle les centres de santé enseignent cette anatomie ---
        le savoir contre la légende, comme le plaidait
        \cref{ex:g8:puberty:questions}.
\end{enumerate}
\end{problem}
