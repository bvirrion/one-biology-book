\chapter{Van bloem tot vrucht}\label{ch:g5:flower-to-fruit}

\Cref{ch:g3:flowers-fruits-seeds} keek toe hoe de kersenbloesem een kers
werd en liet één vraag open: \emph{wat gebeurt er eigenlijk} als een
korreltje \omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel} op de \omterm{def:g3:flowers-fruits-seeds:parts}{stamper} landt? Dit jaar kunnen we, gewapend
met het idee van \omterm{def:g5:flower-to-fruit:steps}{bevruchting} uit \cref{ch:g4:animal-reproduction} en de
cellen uit \cref{ch:g5:first-look-at-cells}, de machinerie van de \omterm{def:g3:flowers-fruits-seeds:parts}{bloem}
echt openen.

\section{Bloemen hebben mannelijke en vrouwelijke delen}

\begin{proposition}[De twee rollen van de bloem in één]\label{prop:g5:flower-to-fruit:roles}
De voortplanting bij bloeiende \omterm{def:g1:plants-around-us:plant}{planten} is \emph{geslachtelijke}
voortplanting, precies in de zin van
\cref{def:g4:animal-reproduction:sexual} --- en de meeste \omterm{def:g3:flowers-fruits-seeds:parts}{bloemen} dragen
beide rollen tegelijk:
\begin{itemize}
  \item de \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{meeldraden}} zijn het mannelijke deel: elk korreltje
        \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel}} draagt de mannelijke voortplantingscellen van
        de \omterm{def:g1:plants-around-us:plant}{plant};
  \item de \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{stamper}} is het vrouwelijke deel: diep daarin liggen de
        \emph{zaadknoppen}\index{zaadknop}, elk met een \omterm{def:g4:animal-reproduction:sexual}{eicel} erin.
\end{itemize}
De ``toekomstige \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden}'' uit
\cref{def:g3:flowers-fruits-seeds:parts} hebben nu hun echte naam:
\omterm{prop:g5:flower-to-fruit:roles}{zaadknoppen}, wachtend om bevrucht te worden.
\end{proposition}
\admitted

\begin{remark}[Hetzelfde plan als bij de dieren]\label{rem:g5:flower-to-fruit:sameplan}
Mannelijke cellen, \omterm{def:g4:animal-reproduction:sexual}{eicellen}, \omterm{def:g5:flower-to-fruit:steps}{bevruchting}: de woordenschat van
\cref{ch:g4:animal-reproduction} laat zich woord voor woord vertalen.
Opnieuw de \omterm{prop:g5:first-look-at-cells:principle}{eenheid van het leven}
(\cref{prop:g5:first-look-at-cells:principle}) --- madeliefjes en
dolfijnen \omterm{def:g1:plants-around-us:plant}{planten} zich voort volgens één onderliggend plan, hoe
verschillend de machinerie er ook uitziet.
\end{remark}

\section{Bestuiving, dan bevruchting}

\begin{definition}[De twee stappen]\label{def:g5:flower-to-fruit:steps}
Een \omterm{def:g2:from-seed-to-plant:seed}{zaad} maken kost twee duidelijk verschillende stappen:
\begin{enumerate}
  \item \emph{bestuiving}\index{bestuiving}: een stuifmeelkorrel reist
        van een \omterm{def:g3:flowers-fruits-seeds:parts}{meeldraad} naar de top van een \omterm{def:g3:flowers-fruits-seeds:parts}{stamper} --- gedragen door
        een insect
        (\cref{prop:g3:flowers-fruits-seeds:transform}) of door de wind;
  \item \emph{bevruchting}\index{bevruchting (planten)}: de gelande
        korrel laat een fijne buis \emph{omlaag door de \omterm{def:g3:flowers-fruits-seeds:parts}{stamper}}
        groeien; daardoorheen bereikt de mannelijke \omterm{def:g5:first-look-at-cells:cell}{cel} een zaadknop en
        verenigt ze zich met de \omterm{def:g4:animal-reproduction:sexual}{eicel}.
\end{enumerate}
Bestuiving is de reis; bevruchting is de ontmoeting.
\end{definition}

\begin{example}[Waarom onbezochte bloemen afvallen]\label{ex:g5:flower-to-fruit:unvisited}
De boomgaardregel uit \cref{rem:g3:flowers-fruits-seeds:bees} heeft nu
haar mechanisme: geen bezoek, geen \omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel} op de \omterm{def:g3:flowers-fruits-seeds:parts}{stamper}; geen
\omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel}, geen buis, geen \omterm{def:g5:flower-to-fruit:steps}{bevruchting}; en een onbevruchte \omterm{def:g3:flowers-fruits-seeds:parts}{bloem}
verwelkt in haar geheel --- geen opzwellende \omterm{def:g3:flowers-fruits-seeds:parts}{stamper}, geen kers. De
ontbrekende stap was onzichtbaar, de reis van een \omterm{def:g5:first-look-at-cells:cell}{cel} door een buis ---
maar de lege tak in juli is zichtbaar genoeg.
\end{example}

\section{Na de ontmoeting: zaad en vrucht}

\begin{proposition}[Wat elk deel wordt]\label{prop:g5:flower-to-fruit:becomes}
Na de \omterm{def:g5:flower-to-fruit:steps}{bevruchting} verloopt de verandering uit
\cref{prop:g3:flowers-fruits-seeds:transform} met nieuwe precisie:
\begin{itemize}
  \item elke \emph{bevruchte zaadknop} wordt een \emph{\omterm{def:g2:from-seed-to-plant:seed}{zaad}} --- het
        piepkleine plantje erin
        (\cref{def:g2:from-seed-to-plant:seed}) groeit uit de bevruchte
        \omterm{def:g4:animal-reproduction:sexual}{eicel}, precies zoals een \omterm{def:g1:animals-around-us:animal}{dier} uit de zijne groeit
        (\cref{ex:g5:first-look-at-cells:one});
  \item de \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{stamper}} eromheen zwelt op tot de \emph{\omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht}};
  \item \omterm{def:g3:flowers-fruits-seeds:parts}{kroonbladen} en \omterm{def:g3:flowers-fruits-seeds:parts}{meeldraden} verwelken en vallen af, hun werk
        gedaan.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Pitjes tellen]\label{ex:g5:flower-to-fruit:pips}
Snijd een appel dwars door: vijf kamertjes, met in elk een paar pitjes.
Elk pitje is één zaadknop die door één stuifmeelbuis bereikt is. Een
scheve appel met lege kamertjes aan één kant vertelt over een slecht
bestoven \omterm{def:g3:flowers-fruits-seeds:parts}{bloem} --- er landden te weinig korrels om alle \omterm{prop:g5:flower-to-fruit:roles}{zaadknoppen} te
bevruchten, en de \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht} zwol ongelijkmatig op.
\end{example}

\begin{example}[Stuifmeel op de wind]\label{ex:g5:flower-to-fruit:wind}
Grassen, tarwe en hazelaars hebben geen \omterm{def:g3:flowers-fruits-seeds:parts}{kroonbladen}, geen geur, geen
nectar --- niets om bezoekers aan te bieden. Hun \omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel} rijdt mee op
de \emph{wind}: in enorme hoeveelheden gemaakt, zo fijn als stof, de
lucht in geworpen om te landen waar het geluk het brengt. Die strategie
ruilt precisie in voor hoeveelheid --- de logica van de sporenwolk uit
\cref{rem:g4:plants-without-seeds:sporeseed}, gespeeld met \omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel}.
(Het is dat rondvliegende \omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel} dat en passant in juni de neuzen van
hooikoortspatiënten kietelt.)
\end{example}

\begin{method}[Een vruchteloze plant onderzoeken]\label{met:g5:flower-to-fruit:diagnose}
Een \omterm{def:g1:plants-around-us:plant}{plant} heeft goed gebloeid maar draagt geen \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht}. Vraag op volgorde:
\begin{enumerate}
  \item Hebben bestuivers haar bereikt? (Binnen, onder een net, geen
        \omterm{prop:g3:sorting-living-things:groups}{insecten} in de buurt, koude regenachtige bloeiweken?)
  \item Zo niet, bestuif dan met de hand: raak met een zacht penseel de
        \omterm{def:g3:flowers-fruits-seeds:parts}{meeldraden} van de ene \omterm{def:g3:flowers-fruits-seeds:parts}{bloem} aan, en daarna de \omterm{def:g3:flowers-fruits-seeds:parts}{stamper} van een
        andere;
  \item heeft vorst of schade de \omterm{def:g3:flowers-fruits-seeds:parts}{bloemen} gedood voordat de \omterm{prop:g5:flower-to-fruit:roles}{zaadknoppen}
        bevrucht konden worden
        (\cref{exo:g3:flowers-fruits-seeds:9})?
  \item Als er nu wel \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht} komt, was de machinerie van de \omterm{def:g1:plants-around-us:plant}{plant} in
        orde --- alleen de reis van stap 1 was mislukt.
\end{enumerate}
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:1}
Welk deel van de \omterm{def:g3:flowers-fruits-seeds:parts}{bloem} is mannelijk, en wat maakt het? Welk deel is
vrouwelijk, en wat ligt daarin?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:2}
Omschrijf \omterm{def:g5:flower-to-fruit:steps}{bestuiving} en \omterm{def:g5:flower-to-fruit:steps}{bevruchting}, en zeg welke van de twee eerst komt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:3}
Vertel de reis van één stuifmeelkorrel, van \omterm{def:g3:flowers-fruits-seeds:parts}{meeldraad} tot \omterm{def:g4:animal-reproduction:sexual}{eicel}, in vier
stappen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:4}
Wat wordt er na de \omterm{def:g5:flower-to-fruit:steps}{bevruchting} van elke bevruchte zaadknop? Wat wordt de
\omterm{def:g3:flowers-fruits-seeds:parts}{stamper}? Wat gebeurt er met de \omterm{def:g3:flowers-fruits-seeds:parts}{kroonbladen}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:5}
Wat is elk appelpitje, in de woordenschat van dit hoofdstuk? Wat
vertelt een scheve appel met lege kamertjes?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:6}
Hoe bestuiven grassen en hazelaars zonder \omterm{def:g3:flowers-fruits-seeds:parts}{kroonbladen}, geur of nectar?
Wat is de prijs van hun methode?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:7}
Waarom maakt een \omterm{def:g3:flowers-fruits-seeds:parts}{bloem} die geen enkel insect bezocht heeft meestal geen
\omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht}? Noem de ontbrekende stap.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:flower-to-fruit:8}
Probeer het (met een bloeiende balkon- of vensterbankplant): bestuif
twee \omterm{def:g3:flowers-fruits-seeds:parts}{bloemen} met de hand met een zacht penseel, zoals in
\cref{met:g5:flower-to-fruit:diagnose}, en markeer ze met een draadje.
Zetten de gemarkeerde \omterm{def:g3:flowers-fruits-seeds:parts}{bloemen} vaker \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht} dan de andere?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:flower-to-fruit:9}
Noem de overeenkomsten tussen de voortplanting van \omterm{def:g1:plants-around-us:plant}{planten} en die van
\omterm{def:g1:animals-around-us:animal}{dieren}: mannelijke cellen, \omterm{def:g4:animal-reproduction:sexual}{eicellen}, \omterm{def:g5:flower-to-fruit:steps}{bevruchting}, de eerste \omterm{def:g5:first-look-at-cells:cell}{cel} van het
nieuwe individu. Waar zit elk daarvan bij de \omterm{def:g3:flowers-fruits-seeds:parts}{bloem}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:flower-to-fruit:10}
Het voorjaar van een kersenteler was koud en regenachtig: bijen vlogen
twee weken lang nauwelijks, hoewel er geen vorst kwam en de bomen
prachtig bloeiden. Voorspel de oogst van juli en verklaar hem met
\cref{def:g5:flower-to-fruit:steps}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:flower-to-fruit:11}
Een tomatenteler zet in zijn kas een volk hommels neer, hoewel er vanuit
de afgesloten kas geen enkele \omterm{def:g3:flowers-fruits-seeds:parts}{bloem} buiten te bereiken is. Leg met dit
hoofdstuk precies uit welke dienst de hommels verkopen --- en waarom
ventilatoren die lucht over de \omterm{def:g1:plants-around-us:plant}{planten} blazen ze voor tomaten (waarvan
het \omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel} gemakkelijk loslaat) deels kunnen vervangen, terwijl dat
bij kersenbomen niet zou kunnen.
\end{exercise}
