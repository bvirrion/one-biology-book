\chapter{De la cellule œuf à la naissance}\label{ch:g8:from-egg-to-birth}

\Cref{ch:g5:human-reproduction} racontait les neuf mois comme une
histoire~; tu possèdes maintenant tous les instruments qu'il faut
pour la jouer correctement~: les \omterm{def:g5:first-look-at-cells:cell}{cellules} et leurs divisions, les
deux \omterm{def:g8:sexual-reproduction:gametes}{gamètes} et leur lieu de rencontre, les \omterm{def:g8:puberty:hormones}{hormones}, les livraisons
du \omterm{prop:g4:journey-of-food:absorb}{sang}. Ce chapitre rejoue le développement humain, machinerie
apparente --- d'une \omterm{def:g5:first-look-at-cells:cell}{cellule} à trois mille milliards, en quarante
semaines.

\section{La première semaine~: du zygote au locataire}

\begin{proposition}[Le voyage le long de la trompe]\label{prop:g8:from-egg-to-birth:firstweek}
La \omterm{def:g5:flower-to-fruit:steps}{fécondation} dans la \omterm{def:g8:reproductive-systems:female}{trompe}
(\cref{def:g8:reproductive-systems:female}) déclenche deux horloges
à la fois~:
\begin{enumerate}
  \item la \emph{division}~: le \omterm{prop:g8:sexual-reproduction:core}{zygote} se divise en moins d'un jour,
        puis encore et encore
        (\cref{ex:g5:first-look-at-cells:one}) --- deux, quatre,
        huit \omterm{def:g5:first-look-at-cells:cell}{cellules}, une boule en forme de mûre de plusieurs
        dizaines de \omterm{def:g5:first-look-at-cells:cell}{cellules} au quatrième jour, le tout en dérivant
        le long de la \omterm{def:g8:reproductive-systems:female}{trompe} vers l'\omterm{prop:g5:human-reproduction:plan}{utérus}~;
  \item la \emph{nidation}~: vers le septième jour, la petite boule
        atteint l'\omterm{prop:g5:human-reproduction:plan}{utérus} et s'enfonce dans la muqueuse préparée
        (l'événement 4 de
        \cref{prop:g8:reproductive-systems:cycle}) --- le locataire
        arrive, le cycle s'arrête, et des \omterm{def:g8:puberty:hormones}{hormones} signalent à tout
        l'organisme~: \omterm{def:g5:human-reproduction:pregnancy}{grossesse}.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[De quoi vivent les premières divisions]\label{ex:g8:from-egg-to-birth:provisions}
Pendant toute la première semaine, la voyageuse ne prend aucun repas
et ne reçoit aucune livraison~: elle vit sur les provisions propres
de la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} --- exactement ce à quoi servait le garnissage de
\cref{ex:g8:reproductive-systems:egg}. La \omterm{def:g8:reproductive-systems:male}{cellule mâle} a apporté le
\omterm{def:g6:cell-unit-of-life:parts}{noyau} du père et guère plus~; la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} a apporté l'autre \omterm{def:g6:cell-unit-of-life:parts}{noyau}
\emph{et} le garde-manger. La nidation est, entre autres choses, le
garde-manger qui s'épuise et le locataire qui se branche sur le
service en chambre.
\end{example}

\section{Embryon, placenta, fœtus}

\begin{definition}[Le placenta]\label{def:g8:from-egg-to-birth:placenta}
De la boule nidée poussent deux constructions à la fois~:
l'\emph{\omterm{def:g5:human-reproduction:pregnancy}{embryon}} lui-même, et son interface de survie --- le
\emph{placenta}\index{placenta}, un disque enfoncé dans la paroi de
l'\omterm{prop:g5:human-reproduction:plan}{utérus}, où le \omterm{prop:g4:journey-of-food:absorb}{sang} de la mère et le \omterm{prop:g4:journey-of-food:absorb}{sang} de l'\omterm{def:g5:human-reproduction:pregnancy}{embryon} s'approchent
intimement \emph{sans se mélanger}. À travers ses vastes surfaces
minces --- le cahier des charges de
\cref{prop:g7:how-animals-breathe:spec}, bâti une fois de plus ---
les \omterm{def:g7:digestion-and-nutrients:families}{nutriments} et le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} passent vers l'\omterm{def:g5:human-reproduction:pregnancy}{embryon}, les déchets
et le \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} repassent en sens inverse (le cordon de
\cref{prop:g5:human-reproduction:cord} court entre le placenta et
l'\omterm{def:g5:human-reproduction:pregnancy}{embryon}). Le placenta est pour le passager à la fois \omterm{def:g7:how-animals-breathe:four}{poumon},
intestin et \omterm{def:g7:removing-wastes:kidneys}{rein} --- les trois frontières en un seul organe.
\end{definition}

\begin{proposition}[Le calendrier, revu avec les instruments]\label{prop:g8:from-egg-to-birth:timetable}
Les étapes de \cref{ex:g5:human-reproduction:timetable}, maintenant
expliquées~:
\begin{itemize}
  \item semaines 1 à 8, l'\emph{\omterm{def:g5:human-reproduction:pregnancy}{embryon}}~: le plan d'organisation
        est posé --- les \omterm{def:g5:first-look-at-cells:cell}{cellules} se divisent, migrent et se
        \emph{spécialisent}
        (\cref{rem:g6:cell-unit-of-life:twoways}) en \omterm{def:g4:heart-and-blood:heart}{cœur}, en \omterm{def:g5:brain-in-command:system}{nerf},
        en \omterm{def:g3:bones-and-muscles:skeleton}{os} et en tout le reste~; le \omterm{def:g4:heart-and-blood:heart}{cœur} bat dès la quatrième
        semaine, parce que l'économie garde-manger-et-placenta a
        besoin de sa pompe très tôt~;
  \item semaines 9 à 40, le \emph{\omterm{def:g5:human-reproduction:pregnancy}{fœtus}}~: le plan étant bâti,
        viennent la croissance et la maturation --- le \omterm{def:g5:brain-in-command:system}{cerveau} qui
        se câble, les \omterm{def:g7:how-animals-breathe:four}{poumons} qui répètent à vide
        (\cref{exo:g5:human-reproduction:11} expliquait leurs
        débuts au premier cri), les coups de pied du quatrième mois
        qui annoncent des \omterm{def:g3:bones-and-muscles:muscle}{muscles} en état de marche~;
  \item tout du long~: des \omterm{def:g8:puberty:hormones}{hormones} maintiennent l'\omterm{prop:g5:human-reproduction:plan}{utérus} au calme
        et le cycle suspendu --- la chimie de
        \cref{def:g8:puberty:hormones}, qui tient
        l'administration de la \omterm{def:g5:human-reproduction:pregnancy}{grossesse}.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Pourquoi les premières semaines exigent le plus de soin]\label{ex:g8:from-egg-to-birth:earlycare}
\Cref{prop:g5:human-reproduction:care} avertissait que le tabac et
l'alcool passent jusqu'au bébé~; le calendrier dit quand cela compte
le plus~: pendant les semaines embryonnaires où le plan se pose,
quand chaque organe est une petite foule de \omterm{def:g5:first-look-at-cells:cell}{cellules} en train de se
spécialiser, une seule instruction perturbée peut se répercuter sur
toute la construction --- et ce sont précisément les semaines où une
\omterm{def:g5:human-reproduction:pregnancy}{grossesse} peut passer inaperçue. D'où l'arithmétique sans détour des
médecins~: le soin commence non à la première échographie, mais à la
première possibilité.
\end{example}

\section{La naissance, et le passage de relais}

\begin{proposition}[La naissance comme relais commandé par les hormones]\label{prop:g8:from-egg-to-birth:birth}
Vers la quarantième semaine, des signaux hormonaux --- ceux du bébé
et ceux de la mère en dialogue --- mettent fin au long calme de
l'\omterm{prop:g5:human-reproduction:plan}{utérus}~:
\begin{enumerate}
  \item le \emph{travail}~: l'\omterm{prop:g5:human-reproduction:plan}{utérus}, le \omterm{def:g3:bones-and-muscles:muscle}{muscle} le plus puissant du
        corps (\cref{rem:g8:reproductive-systems:living} en avait
        rencontré le travail plus doux), se contracte en vagues de
        plus en plus fortes qui ouvrent le passage et poussent le
        bébé le long de la voie de \omterm{def:g2:born-grow-die:cycle}{naissance}~;
  \item la \emph{\omterm{def:g2:born-grow-die:cycle}{naissance}}~: la tête la première, le plus souvent
        (\cref{ex:g5:human-reproduction:birth})~;
  \item le \emph{relais}~: le premier cri remplit des \omterm{def:g7:how-animals-breathe:four}{poumons}
        jamais utilisés~; le trafic du cordon cesse et on le coupe~;
        en quelques minutes, le nouveau-né tourne sur ses propres
        \omterm{def:g7:how-animals-breathe:four}{poumons}, sur son intestin à venir et sur ses \omterm{def:g7:removing-wastes:kidneys}{reins} ---
        chacune des frontières que le \omterm{def:g8:from-egg-to-birth:placenta}{placenta} avait assurées
        (\cref{exo:g5:human-reproduction:11}, maintenant avec la
        carte complète)~;
  \item la \emph{délivrance}~: le \omterm{def:g8:from-egg-to-birth:placenta}{placenta}, sa tâche accomplie, se
        détache et suit --- le seul organe que le corps bâtit pour
        un seul usage et qu'il jette ensuite.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Le premier bilan du nouveau-né]\label{ex:g8:from-egg-to-birth:audit}
Les sages-femmes notent un nouveau-né quelques minutes après la
\omterm{def:g2:born-grow-die:cycle}{naissance} --- \omterm{def:g7:what-is-respiration:respiration}{respiration}, battements du \omterm{def:g4:heart-and-blood:heart}{cœur}, coloration,
mouvements, cri. Lis cette grille avec les \omterm{def:g1:the-five-senses:sense}{yeux} de ce livre~: c'est
la boucle d'approvisionnement et d'élimination de
\cref{prop:g7:removing-wastes:complete}, vérifiée à son premier tour
en solo. Le premier examen que tout humain passe est une épreuve de
physiologie.
\end{example}

\begin{method}[Expliquer n'importe quelle étape de la grossesse]\label{met:g8:from-egg-to-birth:explain}
Interrogé sur n'importe quel moment des neuf mois, réponds avec
quatre instruments~:
\begin{enumerate}
  \item les \emph{\omterm{def:g5:first-look-at-cells:cell}{cellules}}~: que font les divisions et les
        spécialisations~?
  \item l'\emph{approvisionnement}~: garde-manger, ou \omterm{def:g8:from-egg-to-birth:placenta}{placenta} ---
        qu'est-ce qui traverse, qu'est-ce qui ne doit pas~?
  \item les \emph{\omterm{def:g8:puberty:hormones}{hormones}}~: qu'est-ce qui est maintenu, ou
        commandé~?
  \item le \emph{soin}~: qu'est-ce que cette étape demande aux
        habitudes de la mère et au calendrier des médecins~?
\end{enumerate}
\end{method}

\section{Problème~: la chronologie de la sage-femme}

\begin{problem}[{Problème du week-end --- une grossesse suivie
consultation par consultation}]\label{pb:g8:from-egg-to-birth:1}
Le dossier d'une sage-femme pour une \omterm{def:g5:human-reproduction:pregnancy}{grossesse} (imaginaire)~:
consultation d'ouverture à la semaine 8, échographies aux semaines
12 et 20, contrôles à 28 et 36, \omterm{def:g2:born-grow-die:cycle}{naissance} à la semaine 40. Lis ce
dossier avec les instruments du chapitre.

\medskip\noindent\textbf{Partie I --- Les premières pages du
dossier.}
\begin{enumerate}
  \item La \omterm{def:g5:human-reproduction:pregnancy}{grossesse} a été confirmée par un test hormonal des
        semaines avant la première consultation. Quel signal un tel
        test lit-il, et à partir de quel événement de
        \cref{prop:g8:from-egg-to-birth:firstweek} ce signal
        existe-t-il~?
  \item À la semaine 8, la sage-femme note «~\omterm{def:g5:human-reproduction:pregnancy}{embryon} complet dans
        son plan, de la taille d'un haricot~». Traduis dans les
        termes du calendrier --- quelle phase se referme ici~?
  \item Le conseil de la consultation d'ouverture commence par
        aucun alcool, aucune fumée --- et de l'acide folique, une
        vitamine de bâtisseur, «~idéalement dès avant le début~».
        Justifie les trois à partir de
        \cref{ex:g8:from-egg-to-birth:earlycare}.
  \item L'échographie de la semaine 12 montre les battements du
        \omterm{def:g4:heart-and-blood:heart}{cœur} commencés à la semaine 4. Pourquoi si tôt, du point de
        vue de l'approvisionnement~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les mois du milieu.}
\begin{enumerate}[resume]
  \item L'échographie de la semaine 20 mesure chaque organe et la
        position du \omterm{def:g8:from-egg-to-birth:placenta}{placenta}. Donne l'activité de la phase fœtale,
        et les trois métiers du \omterm{def:g8:from-egg-to-birth:placenta}{placenta} que l'on vérifie.
  \item La mère signale les premiers coups de pied vers la semaine
        18. Quelle machinerie s'annonce
        (\cref{prop:g5:brain-in-command:loop} en nomme la chaîne)~?
  \item À la semaine 28, la sage-femme contrôle le sucre et la
        pression du \omterm{prop:g4:journey-of-food:absorb}{sang} de la mère. Pourquoi sa circulation
        est-elle doublement chargée --- les frontières de qui
        fait-elle fonctionner~?
  \item Semaines 30 à 36~: le \omterm{def:g5:human-reproduction:pregnancy}{fœtus} grandit surtout, et les \omterm{def:g7:how-animals-breathe:four}{poumons}
        répètent. Répètent quoi --- et pour quel unique moment~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La dernière page.}
\begin{enumerate}[resume]
  \item Semaine 40~: le travail. Nomme le \omterm{def:g3:bones-and-muscles:muscle}{muscle}, ses deux tâches,
        et qui a commandé ce travail
        (\cref{prop:g8:from-egg-to-birth:birth}).
  \item L'entrée «~\omterm{def:g2:born-grow-die:cycle}{naissance}~» du dossier~: «~cri vigoureux
        d'emblée~; cordon clampé et coupé~; \omterm{def:g8:from-egg-to-birth:placenta}{placenta} complet.~»
        Analyse les trois segments de phrase avec le relais.
  \item Le nouveau-né obtient un bon score à la grille des
        premières minutes. Dis ce que cette grille certifie, dans
        les termes de \cref{ex:g8:from-egg-to-birth:audit}.
  \item Referme le dossier par une phrase pour les parents~: ce que
        les quarante semaines ont bâti, à partir de quelle unique
        \omterm{def:g5:first-look-at-cells:cell}{cellule}, et par quelle machinerie.
\end{enumerate}
\end{problem}
