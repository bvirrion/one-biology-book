\chapter{Human Reproduction}\label{ch:g5:human-reproduction}

Every person you know --- friends, teachers, the oldest person
in your family --- arrived the same way: as a baby, born after
months of growing inside their mother. Humans are \omterm{prop:g3:sorting-living-things:groups}{mammals}
(\cref{prop:g4:classifying-animals:classes}), and human
reproduction runs on the mammal plan of
\cref{ch:g4:animal-reproduction} --- with the numbers, dates
and details of our own \omterm{def:g5:biodiversity:species}{species}, which this chapter sets out
plainly.

\section{The human version of the plan}

\begin{proposition}[Human reproduction is mammal reproduction]\label{prop:g5:human-reproduction:plan}
Like all \omterm{prop:g3:sorting-living-things:groups}{mammals}, humans reproduce sexually
(\cref{def:g4:animal-reproduction:sexual}):
\begin{itemize}
  \item the mother's body produces \emph{\omterm{def:g4:animal-reproduction:sexual}{egg cells}}; the
        father's produces \emph{\omterm{def:g4:animal-reproduction:sexual}{sperm cells}};
  \item \omterm{def:g5:flower-to-fruit:steps}{fertilization} --- one \omterm{def:g4:animal-reproduction:sexual}{sperm cell} joining one \omterm{def:g4:animal-reproduction:sexual}{egg cell}
        --- happens \emph{inside the mother's body}
        (\cref{prop:g4:animal-reproduction:where});
  \item the fertilized \omterm{def:g4:animal-reproduction:sexual}{egg cell} --- one single \omterm{def:g5:first-look-at-cells:cell}{cell}
        (\cref{ex:g5:first-look-at-cells:one}) --- settles in
        the mother's \emph{womb}\index{womb} and grows there;
  \item humans are \omterm{def:g4:animal-reproduction:ovivivi}{viviparous}
        (\cref{def:g4:animal-reproduction:ovivivi}): the baby
        is \omterm{def:g2:animals-grow-up:born}{born alive}, and is fed at first on \omterm{def:g2:animals-grow-up:born}{milk}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Reproduction needs grown-up bodies]\label{rem:g5:human-reproduction:adults}
A child's body cannot make babies: the reproductive organs
only start working at \emph{puberty}, the change of
\cref{def:g3:stages-of-life:stages}'s \omterm{def:g3:stages-of-life:stages}{adolescent} stage, when
the body rebuilds itself from a child's into an adult's. What
puberty changes, and how, is the subject of a full chapter in
the middle-school years of this book.
\end{remark}

\section{Nine months in the womb}

\begin{definition}[Pregnancy]\label{def:g5:human-reproduction:pregnancy}
\emph{Pregnancy}\index{pregnancy} is the time the future baby
spends growing in its mother's \omterm{prop:g5:human-reproduction:plan}{womb}: about \emph{nine months}
in humans. The growing baby is first called an
\emph{embryo}\index{embryo}, then --- once its body plan is
laid out, after about two months --- a
\emph{fetus}\index{fetus}.
\end{definition}

\begin{proposition}[Fed through the cord]\label{prop:g5:human-reproduction:cord}
Inside the \omterm{prop:g5:human-reproduction:plan}{womb}, the baby floats in a pocket of liquid and
cannot eat or breathe. Everything comes ready-made from the
mother's \omterm{prop:g4:journey-of-food:absorb}{blood}: \omterm{def:g4:journey-of-food:digestion}{nutrients} and oxygen
(\cref{prop:g4:heart-and-blood:carries}) cross into the
baby's \omterm{prop:g4:journey-of-food:absorb}{blood} through the
\emph{umbilical cord}\index{umbilical cord} at its belly ---
and its wastes cross back the same way, for the mother's body
to deal with. Your navel is where your own cord was attached.
\end{proposition}
\admitted

\begin{example}[The nine-month timetable]\label{ex:g5:human-reproduction:timetable}
Month 1: a cluster of dividing \omterm{def:g5:first-look-at-cells:cell}{cells}, smaller than a grain of
rice, but already growing fast. Month 2: an \omterm{def:g5:human-reproduction:pregnancy}{embryo} with the
sketch of a head, a beating \omterm{def:g4:heart-and-blood:heart}{heart}, buds for \omterm{def:g1:my-body:parts}{arms} and \omterm{def:g1:my-body:parts}{legs} ---
still smaller than a thumb. Months 3--4: a \omterm{def:g5:human-reproduction:pregnancy}{fetus}, complete in
plan, that begins to move; the mother soon feels the kicks.
Months 5--8: growing and ripening --- hearing arrives, and a
first coat of hair. Month 9: the baby, usually head-down and
around \qty{3}{kg}, is ready.
\end{example}

\begin{example}[Birth]\label{ex:g5:human-reproduction:birth}
At \omterm{def:g2:born-grow-die:cycle}{birth}, powerful \omterm{def:g3:bones-and-muscles:muscle}{muscles} of the \omterm{prop:g5:human-reproduction:plan}{womb} push the baby out into
the world, usually head first. The first cry fills the \omterm{def:g4:breathing:organs}{lungs}
with air --- they had never breathed before
(\cref{prop:g5:human-reproduction:cord}) --- the cord, no
longer needed, is cut painlessly, and the mammal plan
continues outside: \omterm{def:g2:animals-grow-up:born}{milk}, warmth and care
(\cref{ex:g4:animal-reproduction:whalecalf} would approve).
\end{example}

\begin{remark}[Twins]\label{rem:g5:human-reproduction:twins}
Sometimes two babies grow in the \omterm{prop:g5:human-reproduction:plan}{womb} together. If one
fertilized \omterm{def:g4:animal-reproduction:sexual}{egg cell} split into two \omterm{def:g5:human-reproduction:pregnancy}{embryos}, the twins are
\emph{identical} --- built from the same starting \omterm{def:g5:first-look-at-cells:cell}{cell}, alike
as two copies. If two \omterm{def:g4:animal-reproduction:sexual}{egg cells} were fertilized by two \omterm{def:g4:animal-reproduction:sexual}{sperm
cells}, the twins are no more alike than ordinary siblings ---
they only shared the \omterm{prop:g5:human-reproduction:plan}{womb}. Two kinds of twins, told apart by
their beginning.
\end{remark}

\section{Care before and after}

\begin{proposition}[The baby shares its mother's supplies]\label{prop:g5:human-reproduction:care}
Everything reaching the baby travels through the mother's
\omterm{prop:g4:journey-of-food:absorb}{blood} --- which is why an expecting mother eats well
(\cref{ch:g3:balanced-diet}), rests, is followed by doctors
or midwives, and avoids what could harm the baby: tobacco
smoke and alcohol pass through the cord too, and a \omterm{def:g5:human-reproduction:pregnancy}{fetus} is
far more fragile than an adult. Caring for the mother
\emph{is} caring for the baby.
\end{proposition}
\admitted

\begin{example}[A very cared-for species]\label{ex:g5:human-reproduction:longcare}
Even among \omterm{prop:g3:sorting-living-things:groups}{mammals}, humans push the few-and-cared-for
strategy of \cref{prop:g4:animal-reproduction:strategies} to
the extreme: usually one baby at a time, helpless far longer
than a foal or a kitten, and cared for --- fed, taught,
loved --- for many \emph{years}. Human childhood is one of
the longest in the \omterm{def:g1:animals-around-us:animal}{animal} world; this book, a decade-long
course, is itself a piece of that long care.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:1}
Which \omterm{def:g5:first-look-at-cells:cell}{cell} comes from the mother, which from the father, and
what is their joining called?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:2}
Where does the human baby grow, and for about how long?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:3}
What are the growing baby's two names, in order, and roughly
when does the first become the second?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:4}
The baby in the \omterm{prop:g5:human-reproduction:plan}{womb} neither eats nor breathes. How does it
get \omterm{def:g4:journey-of-food:digestion}{nutrients} and oxygen, and where do its wastes go?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:5}
What is your navel the mark of?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:6}
What happens at the first cry, and why is it such a change
for the baby's body?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:7}
Which features make human reproduction typically
mammalian? Name three, citing
\cref{prop:g4:classifying-animals:classes} or
\cref{ch:g4:animal-reproduction}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:8}
Why do doctors ask expecting mothers to avoid tobacco smoke
and alcohol? Use \cref{prop:g5:human-reproduction:care}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:9}
Explain the two kinds of twins by their beginnings. Which
kind is alike ``as two copies'', and why does that follow
from \cref{ex:g5:first-look-at-cells:one}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:10}
Place the human on the strategy line of
\cref{prop:g4:animal-reproduction:strategies}, comparing
with the codfish and the elephant: number of young, length
of care, and what that long care contains for humans.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:human-reproduction:11}
At \omterm{def:g2:born-grow-die:cycle}{birth}, the cut cord ends the baby's supply line at the
very moment the \omterm{def:g4:breathing:organs}{lungs} take their first breath. Using
\cref{prop:g5:human-reproduction:cord} and
\cref{prop:g4:breathing:exchange}, explain what job
switches, in that minute, from the mother's body to the
baby's own \omterm{def:g4:breathing:organs}{lungs} --- and why the first cry is well timed.
\end{exercise}
