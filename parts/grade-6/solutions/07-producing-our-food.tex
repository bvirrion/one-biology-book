\section*{Chapter \ref{ch:g6:producing-our-food} --- Producing Our Food}

\begin{solution}{exo:g6:producing-our-food:1}
\omterm{def:g6:producing-our-food:farming}{Crop growing} manages \omterm{def:g5:food-webs:roles}{producers} --- \omterm{def:g1:plants-around-us:plant}{plants} grown for their
\omterm{def:g6:plants-are-producers:matter}{organic matter}. \omterm{def:g6:producing-our-food:farming}{Livestock raising} manages \omterm{def:g5:food-webs:roles}{consumers} ---
\omterm{def:g1:animals-around-us:animal}{animals} raised for their production: \omterm{def:g2:animals-grow-up:born}{milk}, meat, \omterm{def:g2:animals-grow-up:born}{eggs}, wool.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:2}
Grain: the wheat \omterm{def:g1:plants-around-us:plant}{plant}'s \omterm{def:g2:from-seed-to-plant:seed}{seed} (its packed store). Potato:
the potato \omterm{def:g1:plants-around-us:plant}{plant}'s \omterm{ex:g4:plants-without-seeds:bulbtuber}{tuber}. Apple: the apple tree's \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}.
\omterm{def:g2:animals-grow-up:born}{Milk}: the cow's mother-care product. \omterm{def:g2:animals-grow-up:born}{Egg}: the hen's packed
nursery.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:3}
The transformation of a raw material by feeding \omterm{def:g6:producing-our-food:fermentation}{microbes}.
Bread (raw material: flour dough, worker: \omterm{ex:g6:producing-our-food:bread}{yeast}) and yogurt
(raw material: \omterm{def:g2:animals-grow-up:born}{milk}, workers: souring \omterm{def:g6:producing-our-food:fermentation}{microbes}); cheese and
vinegar serve too.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:4}
The \omterm{ex:g6:producing-our-food:bread}{yeast} --- a single-celled \omterm{def:g6:decomposers-and-soil:decomposers}{fungus} --- feeds on the
flour's sugars and releases a gas; the stretchy dough traps
the gas as bubbles and puffs up. The oven's heat ends the
\omterm{ex:g6:producing-our-food:bread}{yeast}'s work and sets the risen crumb, holes and all.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:5}
The acid released by the \omterm{def:g6:producing-our-food:fermentation}{microbes} feeding on the \omterm{def:g2:animals-grow-up:born}{milk}'s
sugar: the acid thickens and sets the \omterm{def:g2:animals-grow-up:born}{milk}, and the fresh
tang is the acid itself.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:6}
Salting, cooling (fridge), drying --- also cooking: each
denies unwanted \omterm{def:g6:producing-our-food:fermentation}{microbes} a working condition (moisture,
warmth), exactly as cold slows a compost heap.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:7}
Water: the farmer relies on (and sometimes brings) rain.
\omterm{def:g6:exploring-our-environment:conditions}{Light}: sowing in open fields, spaced so every \omterm{def:g1:plants-around-us:plant}{plant} gets
its share. Warmth: sowing timed to the warm season. Air:
open ground, loosened \omterm{def:g6:decomposers-and-soil:soil}{soil}. \omterm{prop:g4:what-plants-need:needs}{Mineral salts}: manure and
restocking after harvest.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:8}
(1) Organism and stage: cow's \omterm{def:g2:animals-grow-up:born}{milk} --- a \omterm{def:g5:food-webs:roles}{consumer}'s
product. (2) Cycle place: \omterm{def:g5:food-webs:roles}{consumer}'s product,
microbe-finished. (3) Raw material \omterm{def:g2:animals-grow-up:born}{milk}; the workers sour
and set it into curds, then ripen the pressed cheese for
months --- holes, rind and smell are their log. (4) Trail:
\omterm{def:g2:animals-grow-up:born}{milk} from the cow, cow from grass, grass from \omterm{def:g6:exploring-our-environment:conditions}{light}.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:9}
\omterm{def:g6:producing-our-food:fermentation}{Microbes} are living workers with \omterm{def:g6:exploring-our-environment:conditions}{conditions}: cold nearly
stops their feeding. In the fridge that protects the food
(the spoilers idle); in the winter heap it delays the
compost (the \omterm{def:g6:decomposers-and-soil:decomposers}{decomposers} idle). The principle:
decomposition and spoilage run at the pace \omterm{def:g6:exploring-our-environment:conditions}{conditions}
allow.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:10}
``Microbe'' names a vast crowd of different tiny \omterm{def:g6:exploring-our-environment:components}{living
things}. Some, in the wrong place, cause illness --- those
the hand-washing targets; others, chosen and managed in
controlled \omterm{def:g6:exploring-our-environment:conditions}{conditions}, make bread rise and \omterm{def:g2:animals-grow-up:born}{milk} set. Enemy
or employee is not the microbe's nature but the match of
\omterm{def:g5:biodiversity:species}{species}, place and management.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:11}
The ham routed through a pig: its weight cost many times
its weight in feed, plus the pig's burned share, land and
water --- the \omterm{def:g5:food-webs:roles}{consumer} floor's toll. The bread's wheat came
one floor from the \omterm{def:g6:exploring-our-environment:conditions}{light}. Wasting the sandwich wastes both
ledgers; the ham's is the longer one.
\end{solution}

\begin{solution}{exo:g6:producing-our-food:12}
They managed the workers' \emph{\omterm{def:g6:exploring-our-environment:conditions}{conditions}}: warmth for
the dough's corner, the starter kept and fed, salt and
timing tuned by trial across generations. The recipes
worked because keeping \omterm{def:g6:exploring-our-environment:conditions}{conditions} right keeps the right
\omterm{def:g6:exploring-our-environment:components}{living things} thriving --- whether or not anyone knows the
\omterm{def:g6:exploring-our-environment:components}{living things} are there. They farmed \omterm{def:g6:producing-our-food:fermentation}{microbes} blind, and
well.
\end{solution}

\begin{solution}{pb:g6:producing-our-food:1}
\textbf{1.} Each grain is a \omterm{def:g2:from-seed-to-plant:seed}{seed}: a tiny wheat \omterm{def:g1:plants-around-us:plant}{plant} packed
with a \omterm{def:g3:animals-through-seasons:active}{food store}, made by the parent for that plantlet's
first days --- not for us, though we divert it.

\textbf{2.} In the wheat \omterm{def:g1:plants-around-us:plant}{plants}' own green \omterm{def:g1:plants-around-us:parts}{leaves}, from
water and \omterm{prop:g4:what-plants-need:needs}{mineral salts} of the \omterm{def:g6:decomposers-and-soil:soil}{soil} and carbon dioxide of
the air, with \omterm{def:g6:exploring-our-environment:conditions}{light} as the \omterm{prop:g3:balanced-diet:jobs}{energy}.

\textbf{3.} Sown, a grain runs its \omterm{def:g2:born-grow-die:cycle}{life cycle}: \omterm{def:g2:from-seed-to-plant:germination}{germination}
to \omterm{def:g1:plants-around-us:plant}{plant} to new grain. Sold and milled, its \omterm{def:g2:born-grow-die:cycle}{life cycle} ends
and its \omterm{def:g3:animals-through-seasons:active}{food store} becomes ours --- lunch packed for a
plantlet, eaten by a person.

\textbf{4.} Return what you borrow: the manure's rotting
restocks the \omterm{prop:g4:what-plants-need:needs}{mineral salts} that the harvest carried off ---
serving the need for \omterm{prop:g4:what-plants-need:needs}{mineral salts}.

\textbf{5.} Because \omterm{ex:g6:producing-our-food:bread}{yeast} is a living worker: warm, it
feeds and works fast; cold, it idles --- the compost crew's
rule, at bakery scale.

\textbf{6.} It fed on the flour's sugars and released gas;
the stretchy dough trapped the gas as bubbles, and the
bubbles doubled the dough.

\textbf{7.} The cold window idled the \omterm{ex:g6:producing-our-food:bread}{yeast} --- barely fed,
barely any gas. The two teams ran an accidental fair test
of warmth: same dough, same day, one condition different,
two risings to compare.

\textbf{8.} The heat has killed the \omterm{ex:g6:producing-our-food:bread}{yeast}, ending the
shift; the set crumb keeps the bubbles' holes --- the
workforce's signature, baked in place.

\textbf{9.} Slice from loaf, loaf from dough, dough from
flour, flour from grain, grain from the wheat \omterm{def:g1:plants-around-us:plant}{plant}'s
\omterm{def:g1:plants-around-us:parts}{leaves} --- manufactured there from mineral inputs with
\omterm{def:g6:exploring-our-environment:conditions}{light}: the trail ends in the July field's sunshine.

\textbf{10.} Slice: one floor (\omterm{def:g5:food-webs:roles}{producer} to table, with a
microbe finish). Butter: two floors (grass to cow to
cream). Ham: two floors with the biggest toll --- the pig
burned and dropped most of its feed's matter before any
became ham. The slice is cheapest in matter; the ham
dearest.

\textbf{11.} The compost's \omterm{def:g6:decomposers-and-soil:decomposers}{decomposers} --- worms, woodlice,
\omterm{def:g6:decomposers-and-soil:decomposers}{fungi}, unseen crews --- dismantle the dough's \omterm{def:g6:plants-are-producers:matter}{organic
matter} to \omterm{def:g6:decomposers-and-soil:soil}{humus} and \omterm{prop:g4:what-plants-need:needs}{mineral salts}: one turn on from food
to next year's \omterm{def:g6:decomposers-and-soil:soil}{soil} fertility.

\textbf{12.} For example: ``Our bread was built by a
\omterm{def:g5:food-webs:roles}{producer} from \omterm{def:g6:plants-are-producers:matter}{mineral matter} and \omterm{def:g6:exploring-our-environment:conditions}{light}, raised by a microbe
workforce, and needed no \omterm{def:g5:food-webs:roles}{consumer} floor at all --- though
the ham sandwiches next door needed one.''
\end{solution}
