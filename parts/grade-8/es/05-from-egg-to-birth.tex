\chapter{De la célula femenina al parto}\label{ch:g8:from-egg-to-birth}

El \cref{ch:g5:human-reproduction} contó los nueve meses como una
historia; ahora tienes todos los instrumentos que hacen falta para
tocarla como es debido: las \omterm{def:g5:first-look-at-cells:cell}{células} y sus divisiones, los dos \omterm{def:g8:sexual-reproduction:gametes}{gametos} y
su sitio de encuentro, las \omterm{def:g8:puberty:hormones}{hormonas} y los repartos de la \omterm{prop:g4:journey-of-food:absorb}{sangre}. Este
capítulo vuelve a poner el desarrollo humano con la maquinaria a la
vista --- de una \omterm{def:g5:first-look-at-cells:cell}{célula} a tres billones, en cuarenta semanas.

\section{La primera semana: del cigoto al inquilino}

\begin{proposition}[El viaje por el oviducto]\label{prop:g8:from-egg-to-birth:firstweek}
La \omterm{def:g5:flower-to-fruit:steps}{fecundación} en el \omterm{def:g8:reproductive-systems:female}{oviducto}
(\cref{def:g8:reproductive-systems:female}) pone en marcha dos relojes
a la vez:
\begin{enumerate}
  \item la \emph{división}: el \omterm{prop:g8:sexual-reproduction:core}{cigoto} se divide en menos de un día y
        después una y otra vez
        (\cref{ex:g5:first-look-at-cells:one}) --- dos, cuatro, ocho
        \omterm{def:g5:first-look-at-cells:cell}{células} y una bolita de mora de docenas de ellas para el cuarto
        día, todo mientras baja a la deriva por el \omterm{def:g8:reproductive-systems:female}{oviducto} hacia el
        \omterm{prop:g5:human-reproduction:plan}{útero};
  \item la \emph{implantación}: alrededor del séptimo día la bolita
        llega al \omterm{prop:g5:human-reproduction:plan}{útero} y se incrusta en el revestimiento preparado (el
        acontecimiento 4 de la
        \cref{prop:g8:reproductive-systems:cycle}) --- llega el
        inquilino, el ciclo se para y unas \omterm{def:g8:puberty:hormones}{hormonas} le avisan al
        aparato entero: \omterm{def:g5:human-reproduction:pregnancy}{embarazo}.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[De qué viven las primeras divisiones]\label{ex:g8:from-egg-to-birth:provisions}
Durante toda la primera semana, la viajera no come ninguna comida ni
recibe ninguna entrega: funciona con las provisiones de la propia
\omterm{def:g5:first-look-at-cells:cell}{célula} \omterm{def:g4:animal-reproduction:sexual}{femenina} --- exactamente para lo que se cargó en el
\cref{ex:g8:reproductive-systems:egg}. El \omterm{def:g8:reproductive-systems:male}{espermatozoide} trajo el
\omterm{def:g6:cell-unit-of-life:parts}{núcleo} del padre y poco más; la \omterm{def:g5:first-look-at-cells:cell}{célula} \omterm{def:g4:animal-reproduction:sexual}{femenina} trajo el otro \omterm{def:g6:cell-unit-of-life:parts}{núcleo}
\emph{y} la despensa. La implantación es, entre otras cosas, la
despensa quedándose corta y el inquilino enchufándose al servicio de
habitaciones.
\end{example}

\section{Embrión, placenta, feto}

\begin{definition}[La placenta]\label{def:g8:from-egg-to-birth:placenta}
De la bolita implantada crecen dos construcciones a la vez: el propio
\emph{\omterm{def:g5:human-reproduction:pregnancy}{embrión}} y su interfaz de soporte vital --- la
\emph{placenta}\index{placenta}, un disco crecido dentro de la pared
del \omterm{prop:g5:human-reproduction:plan}{útero} donde la \omterm{prop:g4:journey-of-food:absorb}{sangre} de la madre y la \omterm{prop:g4:journey-of-food:absorb}{sangre} del \omterm{def:g5:human-reproduction:pregnancy}{embrión} se
acercan íntimamente \emph{sin mezclarse}. A través de sus superficies
enormes y finas --- el pliego de \omterm{def:g6:exploring-our-environment:conditions}{condiciones} de la
\cref{prop:g7:how-animals-breathe:spec}, construido una vez más ---
pasan los \omterm{def:g7:digestion-and-nutrients:families}{nutrientes} y el \omterm{def:g7:what-is-respiration:respiration}{oxígeno} hacia el \omterm{def:g5:human-reproduction:pregnancy}{embrión}, y los desechos y el
\omterm{def:g7:what-is-respiration:respiration}{dióxido de carbono} de vuelta (el cordón de la
\cref{prop:g5:human-reproduction:cord} va entre la placenta y el
\omterm{def:g5:human-reproduction:pregnancy}{embrión}). La placenta es \omterm{def:g7:how-animals-breathe:four}{pulmón}, intestino y \omterm{def:g7:removing-wastes:kidneys}{riñón} para la pasajera ---
las tres fronteras en un solo órgano.
\end{definition}

\begin{proposition}[El calendario, revisitado con instrumentos]\label{prop:g8:from-egg-to-birth:timetable}
Los hitos del \cref{ex:g5:human-reproduction:timetable}, ahora
explicados:
\begin{itemize}
  \item semanas 1--8, el \emph{\omterm{def:g5:human-reproduction:pregnancy}{embrión}}: se traza el plano corporal ---
        las \omterm{def:g5:first-look-at-cells:cell}{células} se dividen, migran y se
        \emph{especializan}
        (\cref{rem:g6:cell-unit-of-life:twoways}) en \omterm{def:g4:heart-and-blood:heart}{corazón}, \omterm{def:g5:brain-in-command:system}{nervio},
        \omterm{def:g3:bones-and-muscles:skeleton}{hueso} y todo lo demás; el \omterm{def:g4:heart-and-blood:heart}{corazón} late desde la cuarta semana,
        porque la economía de despensa y \omterm{def:g8:from-egg-to-birth:placenta}{placenta} necesita pronto su
        bomba;
  \item semanas 9--40, el \emph{\omterm{def:g5:human-reproduction:pregnancy}{feto}}: construido el plano, crecimiento
        y maduración --- el \omterm{def:g5:brain-in-command:system}{cerebro} cableándose, los \omterm{def:g7:how-animals-breathe:four}{pulmones} ensayando
        en vacío (el \cref{exo:g5:human-reproduction:11} explicó su
        debut en el primer llanto) y las pataditas del cuarto mes
        anunciando unos \omterm{def:g3:bones-and-muscles:muscle}{músculos} que funcionan;
  \item durante todo el proceso: unas \omterm{def:g8:puberty:hormones}{hormonas} mantienen el \omterm{prop:g5:human-reproduction:plan}{útero}
        tranquilo y el ciclo suspendido --- la química de la
        \cref{def:g8:puberty:hormones}, llevando la administración del
        \omterm{def:g5:human-reproduction:pregnancy}{embarazo}.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Por qué las primeras semanas son las que más cuidado piden]\label{ex:g8:from-egg-to-birth:earlycare}
La \cref{prop:g5:human-reproduction:care} avisaba de que el tabaco y el
alcohol pasan al bebé; el calendario dice cuándo importa más: en las
semanas del \omterm{def:g5:human-reproduction:pregnancy}{embrión}, cuando se traza el plano y cada órgano es un
grupito de \omterm{def:g5:first-look-at-cells:cell}{células} que se especializan, y una sola instrucción alterada
puede resonar por toda la construcción --- y esas son justamente las
semanas en las que un \omterm{def:g5:human-reproduction:pregnancy}{embarazo} puede pasar desapercibido. De ahí la
aritmética contundente de los médicos: los cuidados no empiezan en la
primera ecografía, sino en la primera posibilidad.
\end{example}

\section{El parto y el relevo}

\begin{proposition}[El parto como un relevo ordenado por hormonas]\label{prop:g8:from-egg-to-birth:birth}
Alrededor de la semana cuarenta, unas señales hormonales --- las del
bebé y las de la madre en diálogo --- terminan con la larga tranquilidad
del \omterm{prop:g5:human-reproduction:plan}{útero}:
\begin{enumerate}
  \item el \emph{parto}: el \omterm{prop:g5:human-reproduction:plan}{útero}, el \omterm{def:g3:bones-and-muscles:muscle}{músculo} más fuerte del cuerpo (la
        \cref{rem:g8:reproductive-systems:living} se encontró con su
        trabajo más suave), se contrae en oleadas cada vez más fuertes
        que abren el paso y empujan al bebé por el canal del parto;
  \item el \emph{\omterm{def:g2:born-grow-die:cycle}{nacimiento}}: normalmente de cabeza
        (\cref{ex:g5:human-reproduction:birth});
  \item el \emph{relevo}: el primer llanto llena unos \omterm{def:g7:how-animals-breathe:four}{pulmones} que
        nunca se habían usado; el tráfico del cordón termina y el
        cordón se corta; y en unos minutos el recién nacido funciona
        con sus propios \omterm{def:g7:how-animals-breathe:four}{pulmones}, con su intestino por estrenar y con
        sus \omterm{def:g7:removing-wastes:kidneys}{riñones} --- todas las fronteras que había cubierto la
        \omterm{def:g8:from-egg-to-birth:placenta}{placenta} (\cref{exo:g5:human-reproduction:11}, ahora con el
        mapa completo);
  \item el \emph{alumbramiento}: la \omterm{def:g8:from-egg-to-birth:placenta}{placenta}, con su trabajo hecho, se
        desprende y sale detrás --- el único órgano que el cuerpo
        construye para un solo uso y desecha.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[La primera auditoría del recién nacido]\label{ex:g8:from-egg-to-birth:audit}
Las matronas puntúan a un recién nacido a los pocos minutos del parto
--- \omterm{def:g7:what-is-respiration:respiration}{respiración}, \omterm{def:g4:heart-and-blood:heart}{latido}, color, movimiento, llanto. Lee la lista con
los \omterm{def:g1:the-five-senses:sense}{ojos} de este libro: es el bucle de suministro y eliminación de la
\cref{prop:g7:removing-wastes:complete}, auditado en su primera vuelta
en solitario. El primer examen que aprueba toda persona es una prueba
de fisiología.
\end{example}

\begin{method}[Explicar cualquier etapa del embarazo]\label{met:g8:from-egg-to-birth:explain}
Ante una pregunta sobre cualquier momento de los nueve meses, responde
con cuatro instrumentos:
\begin{enumerate}
  \item las \emph{\omterm{def:g5:first-look-at-cells:cell}{células}}: ¿qué están haciendo las divisiones y las
        especializaciones?
  \item el \emph{suministro}: ¿despensa o \omterm{def:g8:from-egg-to-birth:placenta}{placenta} --- qué pasa y qué
        no puede pasar?
  \item las \emph{\omterm{def:g8:puberty:hormones}{hormonas}}: ¿qué se está manteniendo o qué se está
        ordenando?
  \item los \emph{cuidados}: ¿qué le pide esa etapa a los hábitos de la
        madre y al calendario de los médicos?
\end{enumerate}
\end{method}

\section{Problema: La línea de tiempo de la matrona}

\begin{problem}[{Problema de fin de semana --- un embarazo seguido
visita a visita}]\label{pb:g8:from-egg-to-birth:1}
El expediente de una matrona para un \omterm{def:g5:human-reproduction:pregnancy}{embarazo} (inventado): primera cita
en la semana 8, ecografías en las semanas 12 y 20, revisiones en la 28
y en la 36, y parto en la semana 40. Lee el expediente con los
instrumentos del capítulo.

\medskip\noindent\textbf{Parte I --- Las primeras páginas del expediente.}
\begin{enumerate}
  \item El \omterm{def:g5:human-reproduction:pregnancy}{embarazo} se confirmó con una prueba hormonal semanas antes
        de la primera visita. ¿Qué señal lee una prueba así y a partir
        de qué acontecimiento de la
        \cref{prop:g8:from-egg-to-birth:firstweek} existe?
  \item En la semana 8 la matrona anota ``\omterm{def:g5:human-reproduction:pregnancy}{embrión} con el plano
        completo, del tamaño de una alubia''. Tradúcelo a los términos
        del calendario --- ¿qué fase se cierra aquí?
  \item El consejo de la primera cita empieza por nada de alcohol, nada
        de humo --- y ácido fólico, una vitamina de obra, ``idealmente
        desde antes del principio''. Justifica las tres cosas con el
        \cref{ex:g8:from-egg-to-birth:earlycare}.
  \item La ecografía de la semana 12 enseña el \omterm{def:g4:heart-and-blood:heart}{latido} que empezó en la
        semana 4. ¿Por qué tan pronto, en términos de suministro?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Los meses centrales.}
\begin{enumerate}[resume]
  \item La ecografía de la semana 20 mide todos los órganos y la
        posición de la \omterm{def:g8:from-egg-to-birth:placenta}{placenta}. Di cuál es el asunto de la fase de
        \omterm{def:g5:human-reproduction:pregnancy}{feto} y las tres tareas de la \omterm{def:g8:from-egg-to-birth:placenta}{placenta} que se están comprobando.
  \item La madre nota las primeras pataditas cerca de la semana 18.
        ¿Qué maquinaria se está anunciando (la
        \cref{prop:g5:brain-in-command:loop} nombra la cadena)?
  \item En la semana 28 la matrona le comprueba a la madre el azúcar y
        la tensión de la \omterm{prop:g4:journey-of-food:absorb}{sangre}. ¿Por qué está doblemente cargada su
        circulación --- las fronteras de quién está llevando?
  \item Semanas 30--36: el \omterm{def:g5:human-reproduction:pregnancy}{feto} sobre todo crece y los \omterm{def:g7:how-animals-breathe:four}{pulmones}
        ensayan. ¿Ensayan qué --- y para qué único momento?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La última página.}
\begin{enumerate}[resume]
  \item Semana 40: el parto. Nombra el \omterm{def:g3:bones-and-muscles:muscle}{músculo}, sus dos tareas y quién
        ordenó el trabajo (\cref{prop:g8:from-egg-to-birth:birth}).
  \item La anotación del parto en el expediente: ``llanto fuerte
        enseguida; cordón pinzado y cortado; \omterm{def:g8:from-egg-to-birth:placenta}{placenta} completa''.
        Audita las tres frases con el relevo.
  \item El recién nacido puntúa alto en la lista de comprobación de los
        primeros minutos. Di qué certifica esa lista, en los términos
        del \cref{ex:g8:from-egg-to-birth:audit}.
  \item Cierra el expediente con una frase para los padres: qué
        construyeron las cuarenta semanas, a partir de qué única \omterm{def:g5:first-look-at-cells:cell}{célula}
        y con qué maquinaria.
\end{enumerate}
\end{problem}
