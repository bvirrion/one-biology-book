\chapter{Pandangan Pertama pada Sel}\label{ch:g5:first-look-at-cells}

Amati punggung tanganmu: \omterm{def:g1:the-five-senses:sense}{kulit} yang halus, satu permukaan tanpa
sambungan. Sekarang bayangkan sebuah kaca pembesar yang cukup kuat
untuk memperbesar seribu kali. Permukaan tanpa sambungan itu akan
larut menjadi hamparan petak-petak mungil, yang dipasang bersama
seperti ubin batu --- dan setiap \omterm{def:g1:living-or-not:living}{makhluk hidup} yang sudah kamu jumpai
dalam buku ini, dari \omterm{prop:g4:plants-without-seeds:damp}{lumut} sampai paus, dihampari dengan cara yang
sama. Rahasia terbesar \omterm{def:g1:living-or-not:living}{biologi} berukuran kecil.

\section{Makhluk hidup tersusun atas sel}

\begin{definition}[Sel]\label{def:g5:first-look-at-cells:cell}
Suatu \emph{sel}\index{sel} adalah balok pembangun hidup yang terkecil
pada sebuah \omterm{def:g1:living-or-not:living}{makhluk hidup}. Hampir semua sel jauh terlalu kecil untuk
dilihat dengan \omterm{def:g1:the-five-senses:sense}{mata} telanjang; sebuah tubuh dibangun darinya
sebagaimana sebuah tembok dibangun dari bata. Tubuhmu sendiri tersusun
atas puluhan ribu miliar sel.
\end{definition}

\begin{proposition}[Asas sel]\label{prop:g5:first-look-at-cells:principle}
Setiap \omterm{def:g1:living-or-not:living}{makhluk hidup} tersusun atas \omterm{def:g5:first-look-at-cells:cell}{sel} --- setiap \omterm{def:g1:animals-around-us:animal}{hewan}, setiap
\omterm{def:g1:plants-around-us:plant}{tumbuhan}, setiap jamur, sampai \omterm{def:g1:living-or-not:living}{makhluk hidup} yang hanya terdiri atas
\emph{satu} \omterm{def:g5:first-look-at-cells:cell}{sel}. Pembangunan bersama ini adalah salah satu tanda besar
\emph{kesatuan kehidupan}\index{kesatuan kehidupan}: sebeda apa pun
\omterm{def:g3:flowers-fruits-seeds:parts}{bunga} aster dan paus, kalau dilihat dari dekat keduanya dibangun dari
jenis satuan hidup kecil yang sama.
\end{proposition}
\admitted

\begin{example}[Sekecil apa itu kecil?]\label{ex:g5:first-look-at-cells:small}
Sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} tubuhmu yang lazim berukuran kira-kira seperlima puluh
milimeter. Jajarkan $50$ \omterm{def:g5:first-look-at-cells:cell}{sel} dan mereka membentang sepanjang satu
milimeter --- satu titik pensil dapat menutupi seluruh barisannya.
Itulah sebabnya tidak seorang pun mencurigai \omterm{def:g5:first-look-at-cells:cell}{sel} sepanjang sebagian
besar sejarah manusia: balok pembangun kehidupan lebih kecil daripada
bintik terkecil yang dapat dilihat \omterm{def:g1:the-five-senses:sense}{mata}.
\end{example}

\section{Alat yang menyingkapkannya}

\begin{definition}[Mikroskop]\label{def:g5:first-look-at-cells:microscope}
Suatu \emph{mikroskop}\index{mikroskop} adalah alat yang lensanya
memperbesar benda yang sangat kecil ratusan kali. Ketika diarahkan
pada irisan tipis bahan hidup, ia menyingkapkan apa yang belum pernah
dilihat \omterm{def:g1:the-five-senses:sense}{mata} mana pun: \omterm{def:g5:first-look-at-cells:cell}{sel}. Namanya sendiri diciptakan seorang
pengamat awal yang mengira petak-petak kecil di dalam sepotong gabus
tampak seperti kamar-kamar kecil para biarawan --- \emph{\omterm{def:g5:first-look-at-cells:cell}{sel}}.
\end{definition}

\begin{remark}[Bintik gelapnya]\label{rem:g5:first-look-at-cells:nucleus}
Pada kedua pemandangan itu, kebanyakan \omterm{def:g5:first-look-at-cells:cell}{selnya} memperlihatkan sebuah
bintik yang lebih gelap di dalamnya: pusat komando \omterm{def:g5:first-look-at-cells:cell}{selnya}, yang
disebut \emph{inti sel}\index{inti sel}. Dengan apa inti \omterm{def:g5:first-look-at-cells:cell}{sel} itu
memerintah --- benang petunjuk yang termasyhur, yang tergulung di
dalamnya --- adalah kisah untuk tahun-tahun akhir buku ini; untuk
sekarang, catatlah saja bahwa kamar-kamar kecil itu tidak kosong.
\end{remark}

\section{Dari satu sel menjadi sebuah tubuh}

\begin{proposition}[Tubuh tumbuh lewat selnya]\label{prop:g5:first-look-at-cells:growth}
Sebuah tubuh tumbuh, dan memperbaiki dirinya, karena \omterm{def:g5:first-look-at-cells:cell}{selnya} dapat
\emph{membelah}: satu \omterm{def:g5:first-look-at-cells:cell}{sel} terbelah menjadi dua \omterm{def:g5:first-look-at-cells:cell}{sel} hidup, dua menjadi
empat, dan seterusnya. Bertambah tinggi
(\cref{prop:g3:stages-of-life:animals}) sesungguhnya adalah \omterm{def:g5:first-look-at-cells:cell}{sel} yang
berlipat ganda; lecet yang sembuh
(\cref{rem:g3:bones-and-muscles:alive}) adalah \omterm{def:g5:first-look-at-cells:cell}{sel} yang membangun
ulang. Dan setiap tubuh, bahkan yang terbesar, bermula sebagai
\emph{satu \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal} --- \omterm{def:g4:animal-reproduction:sexual}{sel telur} yang terbuahi pada
\cref{def:g4:animal-reproduction:sexual}.
\end{proposition}
\admitted

\begin{example}[Dari satu menjadi triliunan]\label{ex:g5:first-look-at-cells:one}
\omterm{def:g4:animal-reproduction:sexual}{Sel telur} yang terbuahi membelah menjadi $2$ \omterm{def:g5:first-look-at-cells:cell}{sel}, lalu $4$, $8$,
$16$ \dots{} Pembelahan yang berulang, sepanjang bulan dan tahun,
membangun puluhan ribu miliar \omterm{def:g5:first-look-at-cells:cell}{sel} yang sedang membaca kalimat ini.
Setiap \omterm{def:g5:first-look-at-cells:cell}{sel} dirimu berketurunan dari \omterm{def:g5:first-look-at-cells:cell}{sel} yang pertama itu.
\end{example}

\begin{remark}[Seluruh kehidupan dalam satu sel]\label{rem:g5:first-look-at-cells:onecelled}
Sebagian \omterm{def:g1:living-or-not:living}{makhluk hidup} tidak pernah melampaui satu \omterm{def:g5:first-look-at-cells:cell}{sel} --- dan tidak
membutuhkan apa pun lagi. Setetes air kolam berlimpah oleh mereka: \omterm{def:g5:first-look-at-cells:cell}{sel}
tunggal yang berenang, makan, dan membelah, seluruh kehidupan pada
skala \omterm{def:g5:first-look-at-cells:microscope}{mikroskop}. \omterm{def:g1:healthy-habits:germs}{Kuman} pada \cref{def:g1:healthy-habits:germs}
kebanyakan adalah \omterm{def:g1:living-or-not:living}{makhluk hidup} bersel tunggal semacam itu --- kenalan
lama, yang sekarang punya penjelasan.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:1}
Apa itu \omterm{def:g5:first-look-at-cells:cell}{sel}? Kira-kira berapa banyak yang menyusun tubuhmu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:2}
Mengapa tidak seorang pun mengetahui tentang \omterm{def:g5:first-look-at-cells:cell}{sel} sepanjang sebagian
besar sejarah? Alat mana yang mengubahnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:3}
Lukiskan apa yang diperlihatkan \omterm{def:g5:first-look-at-cells:microscope}{mikroskop} pada \omterm{def:g1:the-five-senses:sense}{kulit} bawang bombai,
dan pada kerokan pipi. Apa persamaan kedua pemandangan itu?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:4}
Bintik gelap kecil yang terlihat pada kebanyakan \omterm{def:g5:first-look-at-cells:cell}{sel} disebut apa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:5}
Kira-kira berapa banyak \omterm{def:g5:first-look-at-cells:cell}{sel} tubuh, kalau dijajarkan, membentang
sepanjang satu milimeter --- dan berapa panjang satu \omterm{def:g5:first-look-at-cells:cell}{sel}, dalam
milimeter?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:6}
Apa yang sesungguhnya sedang dikerjakan \omterm{def:g5:first-look-at-cells:cell}{sel} ketika seorang anak
bertambah tinggi? Dan ketika \omterm{def:g1:my-body:joint}{lutut} yang lecet sembuh?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:7}
Setiap tubuh manusia bermula sebagai berapa banyak \omterm{def:g5:first-look-at-cells:cell}{sel}? \omterm{def:g5:first-look-at-cells:cell}{Sel} yang
pertama itu adalah apa, menurut
\cref{def:g4:animal-reproduction:sexual}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:8}
Sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} membelah menjadi $2$; masing-masing membelah lagi, dan
seterusnya. Lanjutkan penggandaannya dari $1$: ada berapa banyak \omterm{def:g5:first-look-at-cells:cell}{sel}
sesudah $5$ pembelahan? Sesudah $10$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:first-look-at-cells:9}
Jelaskan kalimat: ``\omterm{def:g3:flowers-fruits-seeds:parts}{bunga} aster dan paus memperlihatkan \omterm{prop:g5:first-look-at-cells:principle}{kesatuan
kehidupan}.'' Apa yang dibagi bersama keduanya, menurut
\cref{prop:g5:first-look-at-cells:principle}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:first-look-at-cells:10}
Setetes air kolam memuat \omterm{def:g1:living-or-not:living}{makhluk hidup} yang masing-masing terdiri atas
satu \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal. Periksalah mereka terhadap tanda-tanda kehidupan
pada \cref{prop:g1:living-or-not:signs}: tanda mana saja yang dapat
diperlihatkan sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} tunggal?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:first-look-at-cells:11}
\omterm{def:g5:first-look-at-cells:cell}{Sel} seekor gajah kira-kira sebesar \omterm{def:g5:first-look-at-cells:cell}{sel} seekor tikus. Lalu apa yang
membuat gajahnya lebih besar --- \omterm{def:g5:first-look-at-cells:cell}{sel} yang lebih besar, atau \omterm{def:g5:first-look-at-cells:cell}{sel} yang
lebih banyak? Apa yang dikatakan jawabanmu tentang cara kerja
pertumbuhan, menurut \cref{prop:g5:first-look-at-cells:growth}?
\end{exercise}
