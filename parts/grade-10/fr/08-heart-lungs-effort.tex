\chapter{Le cœur et les poumons à l'effort}\label{ch:g10:heart-lungs-effort}

Montez quatre étages en courant et arrêtez-vous en haut. Votre poitrine
se soulève --- quarante respirations par minute au lieu de douze,
chacune trois fois plus profonde --- et votre cœur cogne à cent
soixante-dix. Rien dans ces escaliers ne vous demandait d'en décider~;
le corps l'a organisé, en quelques secondes, parce que les muscles du
\cref{ch:g10:exercise-and-energy} réclamaient dix fois plus d'oxygène
qu'une minute plus tôt. Ce chapitre porte sur le système de livraison
--- poumons, cœur, vaisseaux sanguins --- et sur la façon dont son
débit est multiplié quand l'effort commence.

\section{La ventilation}

\begin{definition}[Ventilation]\label{def:g10:heart-lungs-effort:ventilation}
La \emph{ventilation pulmonaire}\index{ventilation} est le volume d'air
qui entre et sort des poumons par minute~: le volume d'une respiration
(le \emph{volume courant}) multiplié par le nombre de respirations par
minute (la \emph{fréquence respiratoire}). Au repos, environ
\qty{0.5}{L} douze à quinze fois par minute~: \qtyrange{6}{8}{L/min}.
Dans les poumons, l'air atteint quelque 300 millions
d'\emph{alvéoles}\index{alvéole}, sacs à paroi mince enveloppés de
capillaires, où l'oxygène passe dans le sang et le dioxyde de carbone
en sort.
\end{definition}

\begin{proposition}[La ventilation à l'effort]\label{prop:g10:heart-lungs-effort:vent-exercise}
À l'effort, le volume courant et la fréquence montent tous deux~: un
adulte en forme peut respirer \qty{2.5}{L} quarante fois par minute,
soit \qty{100}{L/min} ou plus, quinze fois la valeur de repos. La
montée commence dès les premières secondes de l'effort, avant tout
changement dans le sang, et se stabilise à un niveau proportionnel à
l'oxygène consommé.
\end{proposition}
\admitted

\section{Le cœur et les deux circulations}

\begin{definition}[Débit cardiaque]\label{def:g10:heart-lungs-effort:output}
Le cœur est une double pompe à quatre cavités. Le côté droit reçoit le
sang qui revient des organes et l'envoie aux poumons (la
\emph{circulation pulmonaire})~; le côté gauche le reçoit en retour des
poumons et l'envoie aux organes (la \emph{circulation systémique}). Des
valves entre les cavités et à leurs sorties rendent l'écoulement à sens
unique. Le \emph{débit cardiaque}\index{débit cardiaque} $Q$ est le
volume de sang que chaque côté pompe par minute~: le volume éjecté à
chaque battement, le \emph{volume d'éjection systolique} $V_s$,
multiplié par la \emph{fréquence cardiaque} $f$,
\[
Q = f \times V_s .
\]
Au repos, $70$ battements par minute de \qty{70}{mL}~: environ
\qty{5}{L/min} --- tout le volume sanguin une fois par minute.
\end{definition}

\begin{proposition}[Le débit cardiaque à l'effort]\label{prop:g10:heart-lungs-effort:output-exercise}
À l'effort, la fréquence cardiaque monte avec la puissance fournie,
jusqu'à un maximum d'à peu près $220$ moins l'âge en années, et le
volume d'éjection systolique monte aussi, de moitié environ chez une
personne non entraînée et davantage chez une personne entraînée. Le
\omterm{def:g10:heart-lungs-effort:output}{débit cardiaque} d'un jeune adulte passe de \qty{5}{L/min} au repos à
\qtyrange{20}{25}{L/min} à l'effort maximal, et à \qty{35}{L/min} ou
plus chez un champion d'endurance, dont le volume d'éjection peut
dépasser \qty{180}{mL}.
\end{proposition}
\admitted

\section{Amener le sang là où il faut}

\begin{proposition}[Redistribution du débit sanguin]\label{prop:g10:heart-lungs-effort:redistribution}
Au repos, les muscles reçoivent environ un cinquième du \omterm{def:g10:heart-lungs-effort:output}{débit
cardiaque}~; le tube digestif, le foie et les reins en prennent la
moitié. À l'effort intense, les petites artères des muscles au travail
s'élargissent et celles du tube digestif et des reins se resserrent~:
les muscles reçoivent alors plus des quatre cinquièmes d'un débit qui a
lui-même quadruplé --- quelque \qty{20}{L/min} au lieu de
\qty{1}{L/min}, vingt fois plus. La part du cerveau est maintenue
constante en valeur absolue, et celle de la peau augmente pour évacuer
la chaleur.
\end{proposition}
\admitted

\begin{proposition}[Livraison et extraction de l'oxygène]\label{prop:g10:heart-lungs-effort:extraction}
Chaque litre de sang artériel transporte environ \qty{200}{mL}
d'oxygène, presque tout fixé sur l'hémoglobine des globules rouges. Au
repos, les organes en retirent environ \qty{50}{mL} par litre~; à
l'effort, les muscles au travail en retirent jusqu'à \qty{150}{mL},
trois fois plus. L'oxygène consommé par minute est le \omterm{def:g10:heart-lungs-effort:output}{débit cardiaque}
multiplié par la quantité retirée par litre --- de sorte que la
multiplication par cinq du débit et celle par trois de l'extraction
donnent ensemble la multiplication par quinze de la consommation
d'oxygène du \cref{ch:g10:exercise-and-energy}.
\end{proposition}
\admitted

\begin{example}[Le bilan de l'oxygène vérifié]\label{ex:g10:heart-lungs-effort:fick}
Repos~: \qty{5}{L/min} $\times$ \qty{50}{mL/L} $= \qty{250}{mL/min}$,
la consommation de repos. Effort maximal d'un élève entraîné~:
\qty{23}{L/min} $\times$ \qty{150}{mL/L} $= \qty{3.45}{L/min}$ --- une
$\dot V\!\mathrm{O_2}$max de \qty{50}{mL/min} par kilogramme pour
\qty{69}{kg}, exactement le chiffre du chapitre précédent.
\end{example}

\section{Comment la réponse est commandée}

\begin{proposition}[Commande nerveuse et hormonale]\label{prop:g10:heart-lungs-effort:control}
Le cœur et les muscles respiratoires sont commandés par des centres
nerveux situés à la base du cerveau. Deux jeux de nerfs atteignent le
cœur~: l'un le ralentit, actif au repos~; l'autre l'accélère et
renforce chaque battement. Au début de l'effort, la commande que le
cerveau envoie aux muscles est copiée vers ces centres --- le cœur
accélère en un ou deux battements --- et des capteurs situés dans les
muscles et les articulations signalent le mouvement. Puis, l'effort se
poursuivant, des capteurs des artères mesurent le dioxyde de carbone et
l'acidité du sang et ajustent la \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} et la fréquence cardiaque
pour les maintenir près de leurs valeurs de repos. Les glandes
surrénales y ajoutent l'hormone \emph{adrénaline}\index{adrénaline},
qui renforce l'accélération et élargit les artères des muscles.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Sectionner le nerf ralentisseur d'un animal fait passer sa fréquence
cardiaque de repos de 70 à environ 100~; le stimuler arrête le cœur.
Stimuler le nerf accélérateur double la fréquence. Une personne
prévenue qu'un effort l'attend voit sa fréquence cardiaque monter avant
tout mouvement~; un sujet respirant un air enrichi en dioxyde de
carbone augmente sa \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} en moins d'une minute, et un sujet dont
on maintient artificiellement constant le dioxyde de carbone sanguin
pendant l'effort l'augmente quand même dès le début du mouvement ---
les deux mécanismes, anticipation et correction, sont bien réels.
\end{proof}

\begin{method}[Calculer la livraison d'oxygène]\label{met:g10:heart-lungs-effort:compute}
\begin{enumerate}
  \item \omterm{def:g10:heart-lungs-effort:ventilation}{Ventilation}~: volume courant $\times$ fréquence. Seuls 70~\%
        environ de chaque cycle atteignent les \omterm{def:g10:heart-lungs-effort:ventilation}{alvéoles}~; le reste
        emplit les voies aériennes.
  \item \omterm{def:g10:heart-lungs-effort:output}{Débit cardiaque}~: $Q = f \times V_s$, en litres par minute si
        $V_s$ est en litres.
  \item Oxygène consommé par minute~: $Q \times$ (oxygène retiré par
        litre de sang).
  \item Débit vers un organe~: sa part de $Q$. Comparer les débits
        absolus, non les parts --- une part plus petite d'un débit plus
        grand peut représenter davantage de sang.
  \item Vérifier~: la consommation ne peut pas dépasser la
        $\dot V\!\mathrm{O_2}$max, et $f$ ne peut pas dépasser environ
        $220 - $ l'âge.
\end{enumerate}
\end{method}

\begin{remark}[Où se situe la limite]\label{rem:g10:heart-lungs-effort:limit}
La \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} peut dépasser \qty{150}{L/min} et le sang qui quitte les
poumons reste presque saturé à l'effort maximal~: les poumons ne sont
pas le goulot d'étranglement. Le volume d'éjection systolique l'est~: il
fixe la quantité de sang, donc d'oxygène, que chaque battement livre.
L'entraînement en endurance agrandit le cœur et son volume d'éjection
--- ce qui explique qu'un cœur entraîné batte au repos à 45 au lieu de
70 pour les mêmes \qty{5}{L/min}, et que la
$\dot V\!\mathrm{O_2}$max monte avec l'entraînement.
\end{remark}

fraction de l'oxygène qu'elle inspire utilise-t-elle réellement~? Que
devient le reste~?
\end{exercise}

\section{Problème~: l'épreuve d'effort}

\begin{problem}[{Problème du week-end --- une épreuve d'effort chez le
cardiologue~: fréquence cardiaque, volume d'éjection, ventilation et
extraction d'oxygène mesurés à puissance croissante, et la
multiplication qu'ils réalisent ensemble}]
\label{pb:g10:heart-lungs-effort:1}
Tom, 17 ans, \qty{68}{kg}, pédale sur un vélo de laboratoire pendant
qu'on enregistre sa fréquence cardiaque, sa \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} et son air
expiré. Résultats~:

\begin{center}
\begin{tabular}{lccccc}
puissance (W) & 0 & 75 & 150 & 225 & 300 \\ \hline
fréquence cardiaque (par min) & 62 & 105 & 140 & 175 & 198 \\
volume d'éjection (mL) & 72 & 100 & 116 & 120 & 120 \\
volume courant (L) & 0.5 & 1.2 & 1.8 & 2.3 & 2.6 \\
respirations par min & 13 & 18 & 24 & 32 & 44 \\
$\mathrm{O_2}$ retiré par litre de sang (mL) & 52 & 90 & 120 & 145 & 155 \\
\end{tabular}
\end{center}

\medskip
\noindent\textbf{Partie I --- Le cœur.}
\begin{enumerate}
  \item Calculer le \omterm{def:g10:heart-lungs-effort:output}{débit cardiaque} de Tom à chaque puissance.
  \item De quel facteur le débit a-t-il monté à \qty{300}{W}~? Quelle
        part de ce facteur vient de la fréquence et quelle part du
        volume d'éjection~?
  \item Au-dessus de quelle puissance le volume d'éjection cesse-t-il
        de monter~? Qu'est-ce qui augmente seul le débit au-delà~?
  \item Comparer la fréquence cardiaque maximale de Tom à la règle
        empirique.
  \item Son débit de repos vaut environ \qty{4.5}{L/min}. Si son volume
        sanguin est de \qty{5}{L}, combien de temps dure un tour
        complet au repos, et à \qty{300}{W}~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Les poumons.}
\begin{enumerate}[resume]
  \item Calculer la \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} à chaque puissance.
  \item De quel facteur a-t-elle monté à \qty{300}{W}~? Lequel des deux
        facteurs, volume ou fréquence, y contribue le plus~?
  \item La capacité vitale de Tom vaut \qty{5.0}{L}. Quelle fraction en
        utilise chaque cycle à \qty{300}{W}~?
  \item La \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} a monté d'un facteur plus grand que le \omterm{def:g10:heart-lungs-effort:output}{débit
        cardiaque}. Cela signifie-t-il que les poumons limitent la
        performance~? Utiliser la
        \cref{rem:g10:heart-lungs-effort:limit}.
  \item Seuls 70~\% de chaque cycle atteignent les \omterm{def:g10:heart-lungs-effort:ventilation}{alvéoles}. Calculer
        la \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation} alvéolaire au repos et à \qty{300}{W}.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- L'oxygène.}
\begin{enumerate}[resume]
  \item Calculer la consommation d'oxygène de Tom à chaque puissance,
        à partir du \omterm{def:g10:heart-lungs-effort:output}{débit cardiaque} et de l'oxygène retiré par litre.
  \item Exprimer la valeur à \qty{300}{W} en millilitres par minute et
        par kilogramme. Est-ce une $\dot V\!\mathrm{O_2}$max, et de
        quel genre de sujet~?
  \item De quel facteur la consommation a-t-elle monté du repos à
        \qty{300}{W}~? La décomposer en facteur dû au cœur et facteur
        dû à l'extraction.
  \item En prenant \qty{20}{kJ} par litre d'oxygène, calculer la
        puissance métabolique à \qty{300}{W} et le rendement des
        muscles (puissance mécanique divisée par la puissance
        métabolique au-dessus du repos).
  \item Tracer, ou décrire, la consommation d'oxygène en fonction de
        la puissance mécanique. Est-elle linéaire~? Où est le plafond~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La commande et le verdict.}
\begin{enumerate}[resume]
  \item La fréquence cardiaque de Tom atteignait 75 alors qu'il
        attendait assis le début de l'épreuve. Expliquer.
  \item À \qty{300}{W}, son sang artériel est encore saturé à 95~\% en
        oxygène. Que cela vous apprend-il sur les échanges alvéolaires,
        et sur le lieu où se situe la limite de sa
        $\dot V\!\mathrm{O_2}$max~?
  \item Le cardiologue note que les muscles reçoivent désormais environ
        \qty{20}{L/min} de sang. Quelle fraction du débit cela
        représente-t-il, et où est passé le reste de la répartition de
        repos~?
  \item Après six mois d'entraînement, le volume d'éjection maximal de
        Tom vaut \qty{140}{mL} à la même fréquence maximale et à la
        même extraction. Calculer sa nouvelle
        $\dot V\!\mathrm{O_2}$max.
  \item Énoncer le résultat~: les trois nombres (cœur, extraction,
        \omterm{def:g10:heart-lungs-effort:ventilation}{ventilation}) par lesquels le corps de Tom a multiplié son
        apport d'oxygène de repos, et celui des trois que
        l'entraînement a modifié.
\end{enumerate}
\end{problem}
