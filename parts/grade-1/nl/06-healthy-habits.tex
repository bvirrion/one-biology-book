\chapter{Gezonde gewoonten}\label{ch:g1:healthy-habits}

Waarom zegt iedereen toch steeds ``was je handen'', ``poets je tanden'',
``naar bed''? Niet om de pret te bederven. Je lichaam is het levende
wezen dat je het langst zult houden --- je hele leven --- en een paar
kleine gewoonten, elke dag gedaan, houden het sterk. Hier zijn ze, en
waarom ze werken.

\section{Schoon blijven}

\begin{definition}[Ziektekiemen]\label{def:g1:healthy-habits:germs}
\emph{Ziektekiemen}\index{ziektekiem} zijn levende wezens die veel te
klein zijn om te zien en die overal zitten --- op handen, op
deurklinken, in de aarde. De meeste zijn onschadelijk, maar sommige
kunnen ons ziek maken als ze in het lichaam komen. Omdat ze onzichtbaar
zijn, kunnen handen die schoon lijken ze toch dragen.
\end{definition}

\begin{definition}[Hygiëne]\label{def:g1:healthy-habits:hygiene}
\emph{Hygiëne}\index{hygiëne} zijn de gewoonten die het lichaam schoon
houden en de \omterm{def:g1:healthy-habits:germs}{ziektekiemen} buiten houden: handen wassen, het lichaam
wassen, tanden poetsen, het eten schoon houden.
\end{definition}

\begin{method}[Goed je handen wassen]\label{met:g1:healthy-habits:hands}
\begin{enumerate}
  \item Maak je handen nat en neem zeep;
  \item wrijf overal --- handpalmen, handruggen, tussen de vingers,
        rond de duimen --- terwijl je langzaam tot $20$ telt of een
        kort liedje zingt;
  \item spoel af onder stromend water;
  \item droog af met een schone handdoek.
\end{enumerate}
Was je handen voor het eten, na het toilet, en na buiten spelen of het
aaien van \omterm{def:g1:animals-around-us:animal}{dieren}.
\end{method}

\begin{example}[De glitterproef]\label{ex:g1:healthy-habits:glitter}
Wrijf een snufje glitter over je handpalmen, geef dan een vriend een
hand, doe een deur open, pak een potlood: overal glitter.
\omterm{def:g1:healthy-habits:germs}{Ziektekiemen} reizen precies zo, alleen onzichtbaar. Was nu met zeep,
tellend tot $20$ --- de glitter verdwijnt, en de \omterm{def:g1:healthy-habits:germs}{ziektekiemen} ook.
\end{example}

\section{Tanden om te houden}

\begin{proposition}[Waarom tanden gepoetst moeten worden]\label{prop:g1:healthy-habits:teeth}
Na het eten blijven er stukjes voedsel op de tanden achter.
\omterm{def:g1:healthy-habits:germs}{Ziektekiemen} in de mond eten van die restjes --- vooral van de zoete
--- en maken langzaam gaatjes in de tanden, die pijn doen. Poetsen
veegt de restjes weg voordat de kiemen kunnen smullen.
\end{proposition}
\admitted

\begin{method}[Mijn tanden poetsen]\label{met:g1:healthy-habits:brush}
\begin{enumerate}
  \item Poets na het ontbijt en voor het slapengaan --- elke dag;
  \item poets elke kant van elke tand: de voorkant, de achterkant en de
        bovenkant waarmee je kauwt;
  \item neem de tijd --- ongeveer zo lang als een liedje;
  \item neem een nieuwe tandenborstel als de haren uitwaaieren.
\end{enumerate}
\end{method}

\begin{remark}[Melktanden tellen ook mee]\label{rem:g1:healthy-habits:babyteeth}
``Deze tanden vallen toch uit --- waarom zou ik ze poetsen?'' Omdat ze
nog jarenlang voor je moeten kauwen, omdat gaatjes erin net zo veel
pijn doen, en omdat de blijvende tanden er al onder wachten en een
schone, gezonde mond verdienen om in aan te komen.
\end{remark}

\section{Eten, bewegen, slapen}

\begin{proposition}[De drie dagelijkse behoeften]\label{prop:g1:healthy-habits:needs}
Een groeiend lichaam heeft elke dag nodig:
\begin{enumerate}
  \item \emph{gevarieerd eten}\index{gevarieerd eten} --- van alles een
        beetje: groente en fruit, brood en granen, melk en kaas, vlees
        of vis of eieren, en water om te drinken; snoep alleen af en toe;
  \item \emph{beweging}\index{beweging} --- rennen, klimmen, fietsen,
        buiten spelen: bewegen maakt spieren, botten en hart sterker;
  \item \emph{slaap}\index{slaap} --- 's nachts rust het lichaam,
        herstelt het zichzelf en groeit het; een kind heeft een lange
        nacht nodig, veel langer dan een volwassene.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Een goede dag voor het lichaam]\label{ex:g1:healthy-habits:day}
Ontbijt met fruit; handen gewassen voor de lunch; een middag rennen in
het park; tanden gepoetst na het avondeten; vroeg het licht uit met een
verhaaltje. Niets bijzonders --- en dat is precies het punt: gezondheid
bestaat uit gewone dagen als deze.
\end{example}

\begin{remark}[Moe zijn is een boodschap]\label{rem:g1:healthy-habits:tired}
Gapen, prikkende \omterm{def:g1:the-five-senses:sense}{ogen}, chagrijnig zijn zonder reden: dat is het lichaam
dat zegt ``ik heb \omterm{prop:g1:healthy-habits:needs}{slaap} nodig''. Daarnaar luisteren is ook een gezonde
gewoonte. Uitgeslapen hersenen leren 's ochtends beter, spelen beter en
lachen gemakkelijker.
\end{remark}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:3}
Hoe lang moet je in \cref{met:g1:healthy-habits:hands} met zeep
wrijven? Noem twee plekjes die je gemakkelijk vergeet.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:4}
Waarom krijgen tanden gaatjes als ze niet gepoetst worden? Vertel het
verhaal met de \omterm{def:g1:healthy-habits:germs}{ziektekiemen} en de restjes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:5}
Wanneer moeten tanden gepoetst worden, en hoe lang moet dat duren?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:6}
Noem de drie dagelijkse behoeften uit \cref{prop:g1:healthy-habits:needs},
en van elk een voorbeeld uit je eigen dag.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:7}
Wat doet het lichaam tijdens de \omterm{prop:g1:healthy-habits:needs}{slaap}? Wie heeft de langste nacht
nodig, een kind of een volwassene?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:8}
Geef twee tekenen dat een lichaam om \omterm{prop:g1:healthy-habits:needs}{slaap} vraagt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:healthy-habits:9}
Sam zegt: ``Mijn melktanden vallen toch uit, dus ze poetsen heeft geen
zin.'' Antwoord hem met \cref{rem:g1:healthy-habits:babyteeth} ---
minstens twee redenen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:healthy-habits:10}

\cref{ex:g1:healthy-habits:glitter}? Wat laat de proef zien over een
hand geven --- en over zeep?
\end{exercise}
