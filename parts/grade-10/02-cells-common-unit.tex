\chapter{Cells: The Common Unit of Life}\label{ch:g10:cells-common-unit}

Scrape the inside of your cheek with a cotton bud, smear it on a slide,
add a drop of blue dye, and look through a microscope at four hundred
times: a scatter of flat, irregular tiles, each with a darker spot near
its centre. Peel the thin skin from inside an onion scale and look
again: neat rectangular bricks, each with its own spot. Pond water shows
oval creatures rowing past with a fringe of hairs. The three samples have
nothing in common to the naked eye. Under the lens they are built from
the same object --- the \omterm{def:g10:cells-common-unit:cell}{cell} --- and this chapter is about what that
object is, how it is seen, and why every living thing is made of it.

\section{The cell theory}

\begin{definition}[Cell]\label{def:g10:cells-common-unit:cell}
A \emph{cell}\index{cell} is the smallest unit of a living thing that is
itself alive: a volume of watery content, the
\emph{cytoplasm}\index{cytoplasm}, enclosed by a thin
\emph{plasma membrane}\index{plasma membrane} that separates it from
its surroundings while letting selected substances cross. A cell takes
in matter and energy, transforms them, grows, and divides into two
cells.
\end{definition}

\begin{proposition}[The cell theory]\label{prop:g10:cells-common-unit:theory}
Three statements, established by microscopy in the nineteenth century,
form the \omterm{def:g10:cells-common-unit:cell}{cell} theory:
\begin{enumerate}
  \item every living thing is made of one or more \omterm{def:g10:cells-common-unit:cell}{cells};
  \item the \omterm{def:g10:cells-common-unit:cell}{cell} is the basic unit of structure and function of living
        things --- nothing smaller is alive on its own;
  \item every \omterm{def:g10:cells-common-unit:cell}{cell} comes from a pre-existing \omterm{def:g10:cells-common-unit:cell}{cell}, by division.
\end{enumerate}
\end{proposition}
\begin{proof}[Evidence]
Statement 1 rests on the systematic examination of plants (Schleiden,
1838) and animals (Schwann, 1839) with improved microscopes: every
tissue examined, however different in appearance, resolved into \omterm{def:g10:cells-common-unit:cell}{cells}.
Statement 3 was established by Virchow (1855) and by the observation of
dividing \omterm{def:g10:cells-common-unit:cell}{cells} in growing tissues; the alternative --- \omterm{def:g10:cells-common-unit:cell}{cells} forming
spontaneously from fluid --- was never observed, and Pasteur's
experiments on microbes (\cref{ch:g11:antibiotic-resistance} returns to
them) closed the question. Statement 2 follows from the first and from
the fact that isolated \omterm{def:g10:cells-common-unit:cell}{cells}, kept in a suitable liquid, go on living
while any fragment of a \omterm{def:g10:cells-common-unit:cell}{cell} does not.
\end{proof}

\begin{example}[One cell or many]\label{ex:g10:cells-common-unit:onemany}
A yeast, a paramecium or a bacterium is a single \omterm{def:g10:cells-common-unit:cell}{cell} that does
everything --- feeds, moves, divides. An oak or a human is a
cooperative of \omterm{def:g10:cells-common-unit:cell}{cells} of many kinds, each doing a share of the work: a
human body contains some $\num{3e13}$ \omterm{def:g10:cells-common-unit:cell}{cells}, of roughly two hundred
kinds. Between the two lie colonies and simple multicellular
organisms, but there is no living thing built of anything other than
\omterm{def:g10:cells-common-unit:cell}{cells}.
\end{example}

\section{Seeing cells: scales and microscopes}

\begin{definition}[Magnification and resolution]\label{def:g10:cells-common-unit:magnification}
The \emph{magnification}\index{magnification} of an instrument is the
ratio of the size of the image to the size of the object. Its
\emph{resolution}\index{resolution} is the smallest distance between
two points that it still shows as separate. Magnifying an image beyond
what the resolution justifies enlarges the blur, not the detail.
\end{definition}

\begin{proposition}[Two instruments, two scales]\label{prop:g10:cells-common-unit:microscopes}
The light microscope magnifies up to about $\times 1500$ with a
\omterm{def:g10:cells-common-unit:magnification}{resolution} of about \qty{0.2}{\micro m}: it shows \omterm{def:g10:cells-common-unit:cell}{cells}, nuclei,
\omterm{def:g10:cells-common-unit:organelle}{chloroplasts} and bacteria, in colour and alive. The electron microscope
uses a beam of electrons instead of light: \omterm{def:g10:cells-common-unit:magnification}{resolution} below
\qty{1}{nm}, \omterm{def:g10:cells-common-unit:magnification}{magnification} up to $\times 10^6$; it reveals the inner
structure of the \omterm{def:g10:cells-common-unit:cell}{cell} --- membranes, \omterm{def:g10:cells-common-unit:organelle}{mitochondria}, \omterm{def:g10:cells-common-unit:organelle}{ribosomes}, even
large molecules --- but on a sample that is dead, dried and cut into
slices thinner than a micrometre.
\end{proposition}
\admitted

\begin{method}[Measuring an object on a micrograph]\label{met:g10:cells-common-unit:measure}
A micrograph carries either a \omterm{def:g10:cells-common-unit:magnification}{magnification} or a scale bar.
\begin{enumerate}
  \item With a scale bar: measure the bar on the page (say \qty{12}{mm}
        for a bar labelled \qty{10}{\micro m}) and the object
        (\qty{30}{mm}); the real size is $30 \times 10/12 =
        \qty{25}{\micro m}$.
  \item With a \omterm{def:g10:cells-common-unit:magnification}{magnification}: real size $=$ measured size divided by
        the \omterm{def:g10:cells-common-unit:magnification}{magnification}; a \omterm{def:g10:cells-common-unit:cell}{cell} drawn \qty{20}{mm} long at
        $\times 400$ is $20/400 = \qty{0.05}{mm} = \qty{50}{\micro m}$.
  \item Always state the unit, and convert: $\qty{1}{mm} =
        \qty{1000}{\micro m}$, $\qty{1}{\micro m} = \qty{1000}{nm}$.
\end{enumerate}
\end{method}

\begin{example}[Cheek cells]\label{ex:g10:cells-common-unit:cheek}
A cheek \omterm{def:g10:cells-common-unit:cell}{cell} drawn \qty{24}{mm} wide from a $\times 400$ image is
$24/400 = \qty{0.06}{mm} = \qty{60}{\micro m}$ across, and its \omterm{def:g10:cells-common-unit:organelle}{nucleus},
drawn \qty{3}{mm}, is about \qty{8}{\micro m}. A red blood \omterm{def:g10:cells-common-unit:cell}{cell} is
\qty{7}{\micro m}; the bacteria of the mouth, \qty{1}{\micro m} long,
appear as barely resolved dots at the same \omterm{def:g10:cells-common-unit:magnification}{magnification}.
\end{example}

\section{Inside the cell}

\begin{definition}[Organelle]\label{def:g10:cells-common-unit:organelle}
An \emph{organelle}\index{organelle} is a structure inside a \omterm{def:g10:cells-common-unit:cell}{cell},
enclosed by its own membrane, that performs a specific task. The
principal ones are:
\begin{itemize}
  \item the \emph{nucleus}\index{nucleus}, which contains the
        chromosomes --- the DNA of \cref{ch:g10:universal-dna} --- and
        controls the \omterm{def:g10:cells-common-unit:cell}{cell}'s activity;
  \item the \emph{mitochondria}\index{mitochondrion}, some
        \qty{1}{\micro m} long, where organic molecules are broken down
        with oxygen to release usable energy (cellular respiration,
        \cref{ch:g10:cell-metabolism});
  \item in plant \omterm{def:g10:cells-common-unit:cell}{cells}, the \emph{chloroplasts}\index{chloroplast},
        green with chlorophyll, where photosynthesis takes place, and a
        large water-filled \emph{vacuole}\index{vacuole} that keeps the
        \omterm{def:g10:cells-common-unit:cell}{cell} taut.
\end{itemize}
Outside the membrane, a plant \omterm{def:g10:cells-common-unit:cell}{cell} is boxed in a rigid
\emph{cell wall}\index{cell wall} of cellulose; animal \omterm{def:g10:cells-common-unit:cell}{cells} have none.
The \omterm{def:g10:cells-common-unit:cell}{cytoplasm} of every \omterm{def:g10:cells-common-unit:cell}{cell} also contains millions of
\emph{ribosomes}\index{ribosome}, particles of about \qty{25}{nm} that
assemble \omterm{def:g10:chemistry-of-life:families}{proteins}, visible only with the electron microscope.
\end{definition}

\begin{definition}[Prokaryotes and eukaryotes]\label{def:g10:cells-common-unit:prokaryote}
A \omterm{def:g10:cells-common-unit:cell}{cell} whose DNA is enclosed in a \omterm{def:g10:cells-common-unit:organelle}{nucleus} is a
\emph{eukaryotic cell}\index{eukaryotic cell}: animals, plants, fungi
and many single-celled organisms are eukaryotes. A
\emph{prokaryotic cell}\index{prokaryotic cell} has no \omterm{def:g10:cells-common-unit:organelle}{nucleus} and no
membrane-bound \omterm{def:g10:cells-common-unit:organelle}{organelles}: its DNA lies free in the \omterm{def:g10:cells-common-unit:cell}{cytoplasm}, its
size is around \qty{1}{\micro m}, and it is surrounded by a wall of its
own kind. Bacteria are prokaryotes.
\end{definition}

\begin{example}[Reading a bacterium]\label{ex:g10:cells-common-unit:bacterium}
An electron micrograph of the gut bacterium \emph{Escherichia coli}
shows a rod \qty{2}{\micro m} long and \qty{0.5}{\micro m} wide: a
wall, a membrane just inside it, a granular \omterm{def:g10:cells-common-unit:cell}{cytoplasm} packed with
\omterm{def:g10:cells-common-unit:organelle}{ribosomes}, and a paler central region where the single circular DNA
molecule lies. No \omterm{def:g10:cells-common-unit:organelle}{nucleus}, no \omterm{def:g10:cells-common-unit:organelle}{mitochondria}. Yet the \omterm{def:g10:cells-common-unit:cell}{cell} feeds, grows
and divides every twenty minutes in a rich broth: everything the \omterm{def:g10:cells-common-unit:cell}{cell}
theory demands, in a thousandth of the volume of a cheek \omterm{def:g10:cells-common-unit:cell}{cell}.
\end{example}

\section{Many cells, many shapes: specialisation}

\begin{proposition}[Specialised cells]\label{prop:g10:cells-common-unit:specialised}
In a multicellular organism the \omterm{def:g10:cells-common-unit:cell}{cells} share the same basic plan and the
same DNA, but each kind has a shape and an equipment matched to its
task: a neuron extends a fibre up to a metre long to carry signals; a
red blood \omterm{def:g10:cells-common-unit:cell}{cell} has lost its \omterm{def:g10:cells-common-unit:organelle}{nucleus} and is a flattened disc packed with
haemoglobin; a muscle fibre is a giant \omterm{def:g10:cells-common-unit:cell}{cell} full of contractile
\omterm{def:g10:chemistry-of-life:families}{proteins}; a root hair is a thin tube that maximises the surface for
absorbing water; a guard \omterm{def:g10:cells-common-unit:cell}{cell} of a leaf changes shape to open or close
a pore. Specialisation is a matter of which parts of the shared plan
are developed.
\end{proposition}
\admitted

\begin{example}[Surface and volume]\label{ex:g10:cells-common-unit:surface}
Why are \omterm{def:g10:cells-common-unit:cell}{cells} so small? Everything a \omterm{def:g10:cells-common-unit:cell}{cell} takes in or gives out crosses
its membrane, whose area grows with the square of the size, while the
needs grow with the volume, the cube. A cubic \omterm{def:g10:cells-common-unit:cell}{cell} of side
\qty{10}{\micro m} has a surface of \qty{600}{\micro m^2} for a volume
of \qty{1000}{\micro m^3}: \qty{0.6}{\micro m^2} of membrane per cubic
micrometre. At side \qty{100}{\micro m} the ratio drops to
\qty{0.06}{\micro m^2}: ten times less membrane per unit of content. A
large \omterm{def:g10:cells-common-unit:cell}{cell} would starve at its centre; the \omterm{def:g10:cells-common-unit:cell}{cells} that are large
(a frog egg, a muscle fibre) are either mostly inert reserve or
elongated so that no point is far from the surface.
\end{example}

\begin{remark}[The cell as an open system]\label{rem:g10:cells-common-unit:open}
A \omterm{def:g10:cells-common-unit:cell}{cell} is not a sealed bag. Water, oxygen, glucose and ions cross its
membrane continuously, and its wastes leave the same way; a cheek \omterm{def:g10:cells-common-unit:cell}{cell}
placed in pure water swells and bursts within minutes, and one placed
in strong salt water shrivels. The membrane's control of what enters
and leaves is the subject of the Year 1 volume; for now, remember that
a \omterm{def:g10:cells-common-unit:cell}{cell} lives by exchanging with its surroundings, and dies when the
exchanges stop.
\end{remark}

\section{Problem: Counting the Cells of a Body}

\begin{problem}[{Weekend problem --- a microscopy session, from the
onion skin on the slide to an estimate of the number of cells in a human
body}]
\label{pb:g10:cells-common-unit:1}
A class spends an afternoon at the microscope. The objectives available
give total \omterm{def:g10:cells-common-unit:magnification}{magnifications} of $\times 40$, $\times 100$ and
$\times 400$; the field of view at $\times 100$ is \qty{1.8}{mm} across.

\medskip
\noindent\textbf{Part I --- The onion.}
\begin{enumerate}
  \item At $\times 100$, about 9 onion \omterm{def:g10:cells-common-unit:cell}{cells} fit side by side across
        the field of view. Estimate the width of one \omterm{def:g10:cells-common-unit:cell}{cell}, in
        micrometres.
  \item At $\times 400$ the field of view is four times narrower. How
        wide is it, and how many of these \omterm{def:g10:cells-common-unit:cell}{cells} fit across it?
  \item A student draws one \omterm{def:g10:cells-common-unit:cell}{cell} \qty{28}{mm} wide from the
        $\times 400$ image. What is the \omterm{def:g10:cells-common-unit:magnification}{magnification} of the drawing
        relative to the real \omterm{def:g10:cells-common-unit:cell}{cell}?
  \item Iodine stains the \omterm{def:g10:cells-common-unit:organelle}{nucleus} yellow-brown; the student measures it
        at \qty{4}{mm} on the drawing. Real diameter?
  \item Name the structures the student should label on the drawing,
        and one structure present that cannot be seen at this
        \omterm{def:g10:cells-common-unit:magnification}{magnification}.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The cheek and the pond.}
\begin{enumerate}[resume]
  \item Cheek \omterm{def:g10:cells-common-unit:cell}{cells} stained with methylene blue appear as flat
        polygons about \qty{60}{\micro m} across but only
        \qty{5}{\micro m} thick. Why are they flat, given where they
        come from?
  \item A drop of pond water shows a paramecium \qty{240}{\micro m}
        long. Can it be seen with the naked eye? At which
        \omterm{def:g10:cells-common-unit:magnification}{magnification} does it fill a quarter of the field?
  \item Tiny moving dots about \qty{1}{\micro m} long are visible near
        the paramecium at $\times 400$. What are they, most probably,
        and what would an electron microscope add?
  \item The paramecium is a single \omterm{def:g10:cells-common-unit:cell}{cell}; the cheek \omterm{def:g10:cells-common-unit:cell}{cell} is one of
        trillions. In one sentence each, say what "specialised" means
        for the cheek \omterm{def:g10:cells-common-unit:cell}{cell} and why it cannot apply to the paramecium.
  \item A student adds a drop of salt water to the cheek \omterm{def:g10:cells-common-unit:cell}{cells} and sees
        them shrivel. Explain with
        \cref{rem:g10:cells-common-unit:open}.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The size of a \omterm{def:g10:cells-common-unit:cell}{cell}.}
\begin{enumerate}[resume]
  \item Take a typical human \omterm{def:g10:cells-common-unit:cell}{cell} as a cube of side \qty{20}{\micro m}.
        Compute its volume in cubic micrometres, then in cubic
        millimetres ($\qty{1}{mm^3} = 10^9\,\unit{\micro m^3}$).
  \item Compute its surface area and the ratio surface/volume.
  \item Repeat for a bacterium taken as a cube of side
        \qty{1}{\micro m}. By what factor is its ratio larger?
  \item \omterm{def:g10:cells-common-unit:cell}{Cells} take up a mass roughly equal to that of an equal volume
        of water ($\qty{1}{g}$ per \unit{cm^3}). Estimate the mass of
        the \qty{20}{\micro m} \omterm{def:g10:cells-common-unit:cell}{cell} in nanograms
        ($\qty{1}{g} = 10^9\,\unit{ng}$).
  \item A red blood \omterm{def:g10:cells-common-unit:cell}{cell} is a disc \qty{7}{\micro m} across and
        \qty{2}{\micro m} thick. Estimate its volume (treat it as a
        cylinder, $V = \pi r^2 h$) and compare with the cube of
        question 11.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The number of \omterm{def:g10:cells-common-unit:cell}{cells} in a body.}
\begin{enumerate}[resume]
  \item A student of mass \qty{60}{kg} has a volume of about
        \qty{60}{L}. If the body were made only of \qty{20}{\micro m}
        cubic \omterm{def:g10:cells-common-unit:cell}{cells}, how many would it contain?
  \item In fact about a third of the body's volume is fluid between the
        \omterm{def:g10:cells-common-unit:cell}{cells} and other non-cellular material. Correct the estimate.
  \item Red blood \omterm{def:g10:cells-common-unit:cell}{cells} alone number about $\num{2.5e13}$. Using the
        volume of question 15, what volume do they occupy? Is that
        consistent with a blood volume of about \qty{4.5}{L} of which
        45\% is \omterm{def:g10:cells-common-unit:cell}{cells}?
  \item Combine questions 17 and 18: give an order of magnitude for the
        total number of \omterm{def:g10:cells-common-unit:cell}{cells} in the body, and say why a precise count
        is impossible.
  \item State the result: how many \omterm{def:g10:cells-common-unit:cell}{cells}, roughly, make a human, and
        by what process --- from which starting number --- were they all
        produced?
\end{enumerate}
\end{problem}
