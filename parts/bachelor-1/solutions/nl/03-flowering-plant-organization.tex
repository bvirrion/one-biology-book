\section*{Hoofdstuk \ref{ch:b1:flowering-plant-organization} --- Functionele organisatie van een bloemplant}

\begin{solution}{exo:b1:flowering-plant-organization:1}
\omterm{def:b1:flowering-plant-organization:organs}{Wortel}: verankering, opname van water en minerale ionen. \omterm{def:b1:flowering-plant-organization:organs}{Stengel}: het dragen
van de \omterm{def:b1:flowering-plant-organization:organs}{bladeren} naar het licht, geleiding tussen \omterm{def:b1:flowering-plant-organization:organs}{wortels} en \omterm{def:b1:flowering-plant-organization:organs}{bladeren}. \omterm{def:b1:flowering-plant-organization:organs}{Blad}:
fotosynthese (en gasuitwisseling, transpiratie).
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:2}
Een module is één knoop met zijn \omterm{def:b1:flowering-plant-organization:organs}{blad} en okselknop, plus het \omterm{def:b1:flowering-plant-organization:organs}{internodium}
eronder. \omterm{prop:b1:flowering-plant-organization:modular}{Onbepaalde groei}: geen volwassen maat; de eind- en okselknoppen blijven
modules toevoegen zolang de plant leeft.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:3}
Epidermis: levend; bedekking, \omterm{def:b1:flowering-plant-organization:tissues}{cuticula}, \omterm{def:b1:flowering-plant-organization:tissues}{huidmondjes}, \omterm{def:b1:flowering-plant-organization:organs}{wortelharen}. \omterm{def:b1:flowering-plant-organization:tissues}{Parenchym}:
levend; fotosynthese, opslag. \omterm{def:b1:flowering-plant-organization:tissues}{Collenchym}: levend; buigzame steun. \omterm{def:b1:flowering-plant-organization:tissues}{Sclerenchym}:
dood; stijve steun. Geleidende cellen van het \omterm{def:b1:flowering-plant-organization:tissues}{xyleem}: dood; watertransport.
\omterm{def:b1:flowering-plant-organization:tissues}{Zeefvaten} van het \omterm{def:b1:flowering-plant-organization:tissues}{floëem}: levend (met begeleidende cellen); suikertransport.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:4}
$e^{-0.9} = 0.41$, oftewel \qty{41}{\%}; $e^{-3} = 0.05$, oftewel
\qty{5}{\%}.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:5}
Een \omterm{def:b1:flowering-plant-organization:organs}{wortel} (één centrale cilinder, \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}, \omterm{def:b1:flowering-plant-organization:tissues}{xyleem} en \omterm{def:b1:flowering-plant-organization:tissues}{floëem} die elkaar op
een straal afwisselen) van een tweezaadlobbige (weinig xyleemarmen, geen merg).
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:6}
Een gebogen balk wordt het sterkst belast aan zijn oppervlak en helemaal niet
langs zijn as, zodat materiaal in een randkring buiging met de kleinste massa
weerstaat (een buis). Een kabel onder trek wordt over zijn hele doorsnede
gelijkmatig belast en buigt vrij; een centrale streng draagt de trek terwijl de
schors buigzaam blijft, en zijwortels kunnen erdoorheen naar buiten breken.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:7}
$1 - e^{-2.4} = 0.91$; bij $L = 6$ is $1 - e^{-3.6} = 0.97$: twee extra
bladlagen leveren \qty{6}{\%} van het licht op en kosten \qty{50}{\%}
meer \omterm{def:b1:flowering-plant-organization:organs}{blad}.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:8}
$S = \pi d L$: $d = 240/(\pi\times 623\,000) = \qty{1.2e-4}{m}$, ongeveer
\qty{0.12}{mm}. $(240 + 400)/5 = 128$.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:9}
Worteloppervlak per mm: $\pi\times 0.4 = \qty{1.26}{mm^2}$. Haren per mm:
$250\times\pi\times 0.010\times 0.8 = \qty{6.3}{mm^2}$. Samen
$7.5/1.26 = 6$: de haren vermenigvuldigen het oppervlak met zes.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:10}
In het bos liggen de onderste okselknoppen in de schaduw; de modules die zij
zouden maken kunnen zichzelf niet terugverdienen, zodat zij in rust blijven of
hun takken afsterven, en de groei naar de eindknop gaat die omhoog naar het licht
racet. In het veld is elke knop belicht en verdient elke tak zichzelf terug,
zodat de kroon zich laag vult. De kosten: een hoge dunne stam is een grote
investering in onproductief hout, kwetsbaar voor wind; de winst: de plant stemt
haar vorm af op waar het licht is, zonder enig vast plan.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:11}
Het kuiltje en de haren houden een laag stilstaande, vochtige lucht boven de
porie vast; de dampgradiënt van de luchtholten naar de bewegende lucht wordt over
een langere weg uitgesmeerd en de transpiratie daalt. Koolstofdioxide moet langs
dezelfde langere weg naar binnen diffunderen, zodat de opname eveneens daalt, zij
het minder, omdat de $\mathrm{CO_2}$-gradiënt wordt bepaald door de atmosfeer en
door het verbruik van het \omterm{def:b1:flowering-plant-organization:organs}{blad}. In een woestijn is water de beperkende
hulpbron en redt de ruil het leven van de plant; in een regenwoud is water gratis
en is koolstofwinst wat de concurrentie beloont.
\end{solution}

\begin{solution}{exo:b1:flowering-plant-organization:12}
De plant heeft geen geregelde \omterm{def:b1:mammal-organization:compartments}{extracellulaire vloeistof} die op constante
temperatuur en samenstelling tussen haar cellen en de buitenwereld in staat:
haar cellen ontmoeten bodemwater, lucht en zonlicht rechtstreeks, en haar
temperatuur is die van de lucht. Elke cel regelt daarom haar eigen inhoud
(ionen, water, pH) over haar membraan en wand; de plant regelt haar
uitwisselingen aan het oppervlak (\omterm{def:b1:flowering-plant-organization:tissues}{huidmondjes} gaan open en dicht, de \omterm{prop:b1:flowering-plant-organization:stemroot}{endodermis}
selecteert wat het \omterm{def:b1:flowering-plant-organization:tissues}{xyleem} in mag) en stelt haar lichaam bij door groei (meer
\omterm{def:b1:flowering-plant-organization:organs}{wortels} in droge grond, meer \omterm{def:b1:flowering-plant-organization:organs}{bladeren} in de schaduw). Zij regelt wel degelijk,
maar de regeling zit aan de oppervlakken en in de vorm, niet in een eigen
vloeistof.
\end{solution}

\begin{solution}{pb:b1:flowering-plant-organization:1}
\textbf{1.} $5\times 100 = \qty{500}{m^2}$.
\textbf{2.} $e^{-2.5} = 0.082$ bereikt de bodem; \qty{92}{\%}
onderschept.
\textbf{3.} $L = 2$: $1 - e^{-1} = \qty{63}{\%}$; $L = 8$: $1 -
e^{-4} = \qty{98}{\%}$.
\textbf{4.} Van 2 naar 5 wint de kroon \qty{29}{\%} van het licht voor
\qty{300}{m^2} \omterm{def:b1:flowering-plant-organization:organs}{blad}; van 5 naar 8 zou zij \qty{6}{\%} winnen voor nog
eens \qty{300}{m^2} die evenveel kosten om te bouwen en van water te
voorzien: de extra \omterm{def:b1:flowering-plant-organization:organs}{bladeren} betalen zichzelf niet terug.
\textbf{5.} $500/0.005 = \num{100000}$ \omterm{def:b1:flowering-plant-organization:organs}{bladeren}.
\textbf{6.} $500\times 10^6\,\mathrm{mm^2}\times 150 = \num{7.5e10}$
\omterm{def:b1:flowering-plant-organization:tissues}{huidmondjes}.
\textbf{7.} $\qty{50}{\micro m^2}$; per mm$^2$: $150\times 50\times
10^{-6} = \qty{7.5e-3}{mm^2}$, oftewel \qty{0.75}{\%} van het oppervlak.
\textbf{8.} $500\times 4\times 10^{-6} = \qty{2e-3}{mol/s}$, oftewel
\qty{88}{mg/s}.
\textbf{9.} $0.088\times 43\,200 = \qty{3.8}{kg}$ $\mathrm{CO_2}$,
met daarin $3.8\times 12/44 = \qty{1.04}{kg}$ koolstof.
\textbf{10.} $150\times 1.04 = \qty{156}{kg}$ koolstof; de helft,
\qty{78}{kg}, in hout met \qty{50}{\%} koolstof: \qty{156}{kg} hout.
\textbf{11.} $500\times 10^{-3} = \qty{0.5}{mol/s}$, oftewel \qty{9}{g/s}.
\textbf{12.} $9\times 43\,200 = \qty{389}{kg}$, ongeveer \qty{390}{L}.
\textbf{13.} $0.5/(2\times 10^{-3}) = 250$ watermoleculen per
$\mathrm{CO_2}$. De porie die $\mathrm{CO_2}$ binnenlaat, laat water naar
buiten; de lucht binnen is verzadigd, terwijl $\mathrm{CO_2}$ slechts
\qty{0.04}{\%} van de buitenlucht is, zodat de gradiënt van damp naar buiten
veel steiler is dan de gradiënt van $\mathrm{CO_2}$ naar binnen.
\textbf{14.} $\qty{2}{mm}\times\qty{100}{m^2} = \qty{200}{L}$: een halve
dag.
\textbf{15.} \qty{100}{m^3}; $5\times 100 = \qty{500}{km}$.
\textbf{16.} $\pi\times 0.5\times 10^{-3}\times 5\times 10^5 =
\qty{785}{m^2}$.
\textbf{17.} $200\times 5\times 10^8\,\mathrm{mm} = \num{1e11}$ haren.
\textbf{18.} Elk $\pi\times 10^{-5}\times 5\times 10^{-4} =
\qty{1.57e-8}{m^2}$; samen \qty{1570}{m^2}.
\textbf{19.} $785 + 1570 = \qty{2355}{m^2}$, $4.7$ keer het bladoppervlak.
\textbf{20.} Volume binnen \qty{1}{mm}: $\pi\times(10^{-3})^2\times 5
\times 10^5 = \qty{1.6}{m^3}$, oftewel \qty{1.6}{\%} van de bodem.
\textbf{21.} Rogge: $\pi\times 10^{-6}\times 6.2\times 10^5 =
\qty{1.95}{m^3}$ van de bak van \qty{0.05}{m^3}, oftewel \qty{3900}{\%}: elk
punt van de bodem ligt op minder dan een millimeter van veertig \omterm{def:b1:flowering-plant-organization:organs}{wortels}. De
rogge doorzoekt haar bodem meer dan tweeduizend keer grondiger --- een
akkergewas met een budget van vier maanden tegenover een boom die honderd
kubieke meter doorzoekt.
\textbf{22.} $1000 + 2355 = \qty{3355}{m^2}$; $\qty{34}{m^2}$ per
vierkante meter grond.
\textbf{23.} Mens: $130/2 = 65$; de verhouding van de eik is de helft van die
van de gevouwen oppervlakken van de mens, en dat zonder enige vouwing.
\textbf{24.} De eik voegt elk jaar modules, \omterm{def:b1:flowering-plant-organization:organs}{bladeren} en \omterm{def:b1:flowering-plant-organization:organs}{wortels} toe, zodat zijn
oppervlakken met zijn leeftijd meegroeien; de oppervlakken van het zoogdier zijn
vaste \omterm{def:b1:mammal-organization:organ}{organen}, af bij volwassen maat, en de enige manier om zijn verhouding op te
voeren was ze tijdens de ontwikkeling te vouwen.
\textbf{25.} Ongeveer \qty{34}{m^2} \omterm{prop:b1:organism-environment:surfaces}{uitwisselingsoppervlak} per vierkante meter
grond: \qty{10}{m^2} \omterm{def:b1:flowering-plant-organization:organs}{blad} (beide zijden) erboven en \qty{24}{m^2}
\omterm{def:b1:flowering-plant-organization:organs}{wortel} en \omterm{def:b1:flowering-plant-organization:organs}{wortelhaar} eronder.
\end{solution}
