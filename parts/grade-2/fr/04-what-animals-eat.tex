\chapter{Ce que mangent les animaux}\label{ch:g2:what-animals-eat}

Le lapin du pré grignote de l'herbe toute la journée. Le renard qui
traverse le même pré au crépuscule ne s'intéresse pas du tout à
l'herbe --- il s'intéresse au lapin. Et le merle prend des vers le
matin et des cerises l'après-midi. Tout \omterm{def:g1:animals-around-us:animal}{animal} doit manger, mais
tous ne mangent pas la même chose.

\section{Trois sortes de mangeurs}

\begin{definition}[Herbivore, carnivore et omnivore]\label{def:g2:what-animals-eat:kinds}
On range les \omterm{def:g1:animals-around-us:animal}{animaux} selon ce qu'ils mangent~:
\begin{itemize}
  \item un \emph{herbivore}\index{herbivore} mange des \omterm{def:g1:plants-around-us:plant}{plantes}~:
        herbe, \omterm{def:g1:plants-around-us:parts}{feuilles}, \omterm{def:g2:from-seed-to-plant:seed}{graines}, fruits --- le lapin, la vache,
        l'escargot~;
  \item un \emph{carnivore}\index{carnivore} mange d'autres
        \omterm{def:g1:animals-around-us:animal}{animaux}~: le renard, le chat, la coccinelle (qui chasse de
        minuscules pucerons)~;
  \item un \emph{omnivore}\index{omnivore} mange à la fois des
        \omterm{def:g1:plants-around-us:plant}{plantes} et des \omterm{def:g1:animals-around-us:animal}{animaux}~: le merle, le sanglier --- et les
        humains.
\end{itemize}
\end{definition}

\begin{example}[Un pré, trois menus]\label{ex:g2:what-animals-eat:meadow}
Dans un pré d'été~: le lapin tond l'herbe (\omterm{def:g2:what-animals-eat:kinds}{herbivore})~; le renard
chasse le lapin (\omterm{def:g2:what-animals-eat:kinds}{carnivore})~; le merle prend un ver maintenant, une
cerise plus tard (\omterm{def:g2:what-animals-eat:kinds}{omnivore}). Trois voisins, trois menus --- et
remarque que le dîner du renard a d'abord mangé de l'herbe.
\end{example}

\begin{remark}[Les carnivores ne sont pas des méchants]\label{rem:g2:what-animals-eat:fox}
Le renard n'est pas «~méchant~» parce qu'il mange des lapins~:
chasser est simplement la façon dont un \omterm{def:g2:what-animals-eat:kinds}{carnivore} se nourrit ---
aussi naturelle que le grignotage du lapin. Dans la nature, manger
et être mangé font partie du fonctionnement de la vie~; un chapitre
plus tard reliera tous ces repas en chaînes alimentaires.
\end{remark}

\section{Lire le menu d'un animal sur son corps}

\begin{proposition}[Le corps va avec le menu]\label{prop:g2:what-animals-eat:match}
Le corps d'un \omterm{def:g1:animals-around-us:animal}{animal} porte des indices sur sa nourriture~:
\begin{itemize}
  \item les \omterm{def:g2:what-animals-eat:kinds}{herbivores} comme le lapin et la vache ont de grosses
        dents plates au fond de la bouche pour broyer les \omterm{def:g1:plants-around-us:plant}{plantes},
        et passent de longues heures à manger~;
  \item les \omterm{def:g2:what-animals-eat:kinds}{carnivores} comme le chat et le renard ont des dents
        pointues pour saisir leurs proies, des griffes acérées et
        des \omterm{def:g1:the-five-senses:sense}{yeux} tournés vers l'avant pour chasser~;
  \item chez les oiseaux, cela se voit au bec~: court et épais pour
        casser les \omterm{def:g2:from-seed-to-plant:seed}{graines} (le moineau), courbé et crochu pour
        déchirer la viande (l'aigle), long et fin pour prendre les
        insectes dans l'écorce.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Les dents du chat]\label{ex:g2:what-animals-eat:cat}
Regarde un chat bâiller~: de longues dents pointues aux coins avant
de la bouche. Ce sont des dents à saisir --- l'équipement d'un
\omterm{def:g2:what-animals-eat:kinds}{carnivore}. La vache, en bâillant, n'en montrerait aucune~: sa bouche
n'est faite que de larges broyeuses plates pour l'herbe.
\end{example}

\begin{method}[Deviner un menu]\label{met:g2:what-animals-eat:guess}
Pour deviner ce que mange un \omterm{def:g1:animals-around-us:animal}{animal} inconnu~:
\begin{enumerate}
  \item regarde les dents ou le bec~: des broyeuses plates, des
        pointes qui saisissent, ou les deux~?
  \item regarde les griffes et les \omterm{def:g1:the-five-senses:sense}{yeux}~: des outils de chasse, ou
        non~?
  \item observe-le manger si tu le peux --- l'indice le plus sûr de
        tous~;
  \item puis nomme-le~: \omterm{def:g2:what-animals-eat:kinds}{herbivore}, \omterm{def:g2:what-animals-eat:kinds}{carnivore} ou \omterm{def:g2:what-animals-eat:kinds}{omnivore}.
\end{enumerate}
\end{method}

\section{Manger assez, jour après jour}

\begin{proposition}[Se nourrir est un travail quotidien]\label{prop:g2:what-animals-eat:daily}
La nourriture est le carburant et le matériau de construction du
corps~; les \omterm{def:g1:animals-around-us:animal}{animaux} doivent donc manger encore et encore, jour après
jour. Le temps que cela prend dépend du menu~: l'herbe donne peu à
chaque bouchée, alors la vache broute pendant des heures~; la viande
est riche, alors le renard peut faire un bon repas et se reposer.
\end{proposition}
\admitted

\begin{example}[L'hiver change le menu]\label{ex:g2:what-animals-eat:winter}
La nourriture change avec les saisons. Le merle mange des vers et
des insectes en été, mais en hiver, quand le sol est dur, il passe
aux baies. Le large menu d'un \omterm{def:g2:what-animals-eat:kinds}{omnivore} est une grande aide quand les
temps deviennent difficiles.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:1}
Donne les mots qui désignent~: un \omterm{def:g1:animals-around-us:animal}{animal} qui mange des \omterm{def:g1:plants-around-us:plant}{plantes}~; un
\omterm{def:g1:animals-around-us:animal}{animal} qui mange d'autres \omterm{def:g1:animals-around-us:animal}{animaux}~; un \omterm{def:g1:animals-around-us:animal}{animal} qui mange les deux.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:2}
\omterm{def:g2:what-animals-eat:kinds}{Herbivore}, \omterm{def:g2:what-animals-eat:kinds}{carnivore} ou \omterm{def:g2:what-animals-eat:kinds}{omnivore}~: (a) la vache~? (b) le renard~?
(c) le merle~? (d) l'escargot~? (e) l'humain~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:3}
Quelles dents sont les plus utiles à un \omterm{def:g2:what-animals-eat:kinds}{herbivore}~: des broyeuses
plates ou des pointes qui saisissent~? Pourquoi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:4}
Associe chaque bec à un menu~: court et épais~; courbé et crochu~;
long et fin.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:5}
La coccinelle est petite, jolie --- et \omterm{def:g2:what-animals-eat:kinds}{carnivore}. Que chasse-t-elle~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:6}
Pourquoi la vache passe-t-elle bien plus d'heures à manger que le
renard~? Utilise \cref{prop:g2:what-animals-eat:daily}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:7}
Que mange le merle en été, et en hiver~? Pourquoi un large menu
est-il utile~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:what-animals-eat:8}
À toi d'essayer~: observe les oiseaux dans un parc ou à une
mangeoire. Note ce que chaque espèce prend --- \omterm{def:g2:from-seed-to-plant:seed}{graines}, pain,
insectes, baies --- et regarde leur bec.
\cref{prop:g2:what-animals-eat:match} se vérifie-t-elle~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:9}
Un zoo reçoit un \omterm{def:g1:animals-around-us:animal}{animal} mystère~: dents pointues à l'avant, griffes
acérées, \omterm{def:g1:the-five-senses:sense}{yeux} tournés vers l'avant. Quel est son menu probable~?
Quelles étapes de \cref{met:g2:what-animals-eat:guess} as-tu
utilisées~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:what-animals-eat:10}
«~Sans herbe, il n'y aurait bientôt plus de renards.~» Le renard ne
mange jamais d'herbe --- et pourtant la phrase est juste. Explique,
en te servant du pré de \cref{ex:g2:what-animals-eat:meadow}.
\end{exercise}
