\section*{Chapter \ref{ch:b1:biosyntheses-integration} --- Biosyntheses and the Integrated Cell}

\begin{solution}{exo:b1:biosyntheses-integration:1}
Carbon skeletons (from the central pathways), reducing power (\omterm{def:b1:biosyntheses-integration:ppp}{NADPH}),
energy (\omterm{def:b1:water-small-molecules:atp}{ATP}). The \omterm{def:b1:biosyntheses-integration:ppp}{pentose phosphate pathway}.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:2}
Hexokinase, bypassed by glucose-6-phosphatase; phosphofructokinase,
by fructose-1,6-bisphosphatase; pyruvate kinase, by pyruvate
carboxylase plus PEP carboxykinase.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:3}
Peak about \qty{7.5}{mmol/L} an hour after the meal; back to
\qty{5}{} after about three hours; lowest about \qty{4.5}{mmol/L}
before breakfast.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:4}
The seed has the glyoxylate cycle, which turns two \omterm{def:b1:respiration-fermentation:pdh}{acetyl-CoA} into one
four-carbon acid without losing carbon as $\mathrm{CO_2}$, and from
that acid makes sugar. Animals lack the two \omterm{def:b1:enzymes:enzyme}{enzymes} of the shortcut:
their \omterm{def:b1:respiration-fermentation:pdh}{acetyl-CoA} enters the \omterm{def:b1:respiration-fermentation:krebs}{Krebs cycle} and its two carbons leave as
$\mathrm{CO_2}$ before any net oxaloacetate appears.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:5}
Six \omterm{def:b1:water-small-molecules:atp}{ATP} equivalents (two carboxylations, two GTP, two \omterm{def:b1:water-small-molecules:atp}{ATP} at the
phosphoglycerate step) per glucose. Net per cycle: $2 - 6 = -4$ \omterm{def:b1:water-small-molecules:atp}{ATP}.
The liver pays, with oxygen, after the sprint --- the muscle borrowed
\omterm{def:b1:water-small-molecules:atp}{ATP} it could not make aerobically in time.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:6}
Each glucose gives 2 \omterm{def:b1:respiration-fermentation:pdh}{acetyl-CoA}: 4 glucose. Each glucose through the
oxidative branch gives 2 \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}: 7 glucose. Eleven glucoses for one
palmitate --- of which four provide carbon and seven provide electrons.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:7}
Different \omterm{def:b1:enzymes:enzyme}{enzymes} let each direction be switched on or off
independently; different compartments (\omterm{def:b1:eukaryotic-cell:organelle}{cytosol} for synthesis, matrix
for oxidation) separate the two pools of intermediates; \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} is kept
reduced for synthesis while \omterm{def:b1:respiration-fermentation:carriers}{NAD} is kept oxidised for degradation, so
each direction is thermodynamically favoured in its own place.
Malonyl-CoA, the first committed intermediate of synthesis, inhibits
the carnitine shuttle that carries \omterm{def:b1:lipids:fattyacid}{fatty acids} into the \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondrion}:
while \omterm{def:b1:lipids:triglyceride}{fat} is being made, none is burned.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:8}
Alanine $+ \alpha$-ketoglutarate $\rightleftharpoons$ pyruvate $+$
\omterm{def:b1:biosyntheses-integration:nitrogen}{glutamate}. After a \omterm{def:b1:proteins:peptide}{protein} meal, surplus alanine gives its nitrogen to
\omterm{def:b1:biosyntheses-integration:nitrogen}{glutamate} (rightward), for disposal as urea and its carbon for fuel or
glucose. In a fast, muscle \omterm{def:b1:proteins:peptide}{protein} is broken down and its amino groups
collected onto pyruvate as alanine (leftward in muscle), shipped to
the liver, and there converted back to pyruvate and glucose.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:9}
The liver cannot release free glucose: blood glucose falls
dangerously between meals (hypoglycaemia). Glucose-6-phosphate
accumulates and is diverted into \omterm{def:b1:carbohydrates:polysaccharide}{glycogen}, which the liver cannot
mobilise to the blood: the liver enlarges with \omterm{def:b1:carbohydrates:polysaccharide}{glycogen}. The excess
glucose-6-phosphate also runs down \omterm{def:b1:respiration-fermentation:glycolysis}{glycolysis} to lactate, which rises
in the blood.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:10}
$120/0.55 = \qty{218}{g}$ of \omterm{def:b1:proteins:peptide}{protein} per day, about \qty{1.1}{kg} of
muscle. With ketone bodies, $40/0.55 = \qty{73}{g}$ of \omterm{def:b1:proteins:peptide}{protein},
\qty{360}{g} of muscle: the loss falls by two thirds, and survival on
the \omterm{def:b1:lipids:triglyceride}{fat} store is extended from weeks to months.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:11}
The same phosphorylation cascade activates one and inactivates the
other, so the signal that opens one valve shuts the other. Were both
active, each glucose added (one UTP, equivalent to one \omterm{def:b1:water-small-molecules:atp}{ATP}) and removed
(free) would cost one \omterm{def:b1:water-small-molecules:atp}{ATP} per turn and produce nothing but heat. The
bumblebee runs exactly such a cycle (on fructose-6-phosphate) in its
flight muscles before take-off on a cold morning, burning \omterm{def:b1:water-small-molecules:atp}{ATP} as a
heater until the muscle is warm enough to fly.
\end{solution}

\begin{solution}{exo:b1:biosyntheses-integration:12}
Each crossroads \omterm{def:b1:enzymes:enzyme}{enzyme} responds only to the molecules around it ---
\omterm{def:b1:water-small-molecules:atp}{ATP}, AMP, citrate, \omterm{def:b1:respiration-fermentation:pdh}{acetyl-CoA}, a phosphate put on it by a kinase ---
and opens or shuts accordingly; separate \omterm{def:b1:enzymes:enzyme}{enzymes} for the two
directions mean that opening one direction can shut the other. No
\omterm{def:b1:enzymes:enzyme}{enzyme} knows about breakfast or the brain; yet the sum of these local
responses is that the liver stores after a meal, releases at night,
spares \omterm{def:b1:proteins:peptide}{protein} in a fast: a coherent strategy, with no strategist. The
``plan'' is the wiring of the valves, chosen by selection, and the
hormones that set many valves at once are what makes the \omterm{def:b1:organism-environment:organism}{organism} act
as one.
\end{solution}

\begin{solution}{pb:b1:biosyntheses-integration:1}
\textbf{1.} Liver \omterm{def:b1:carbohydrates:polysaccharide}{glycogen} $100\times 17 = \qty{1.7}{MJ}$, 0.2 day;
muscle \omterm{def:b1:carbohydrates:polysaccharide}{glycogen} \qty{6.8}{MJ}, 0.8 day; \omterm{def:b1:lipids:triglyceride}{fat} $12\,000\times 38 =
\qty{456}{MJ}$, 54 days; \omterm{def:b1:proteins:peptide}{protein} \qty{170}{MJ}, 20 days (never fully
usable).
\textbf{2.} Liver \omterm{def:b1:carbohydrates:polysaccharide}{glycogen}: the liver has glucose-6-phosphatase.
Muscle lacks it, so its \omterm{def:b1:carbohydrates:polysaccharide}{glycogen} yields glucose-6-phosphate for its
own \omterm{def:b1:respiration-fermentation:glycolysis}{glycolysis} only.
\textbf{3.} $120 + 40 = \qty{160}{g}$ per day.
\textbf{4.} $0.005\times 5\times 180 = \qty{4.5}{g}$: at \qty{5}{g/h}, about \qty{54}{min}.
\textbf{5.} $120\times 17 = \qty{2.0}{MJ}$, a quarter of the \qty{8.4}{MJ}.
\textbf{6.} Conversion to \omterm{def:b1:lipids:fattyacid}{fatty acids} and \omterm{def:b1:lipids:triglyceride}{triglyceride} in the liver,
exported to adipose \omterm{def:b1:body-plans-tissues:tissue}{tissue}; insulin.
\textbf{7.} $160/24 = \qty{6.7}{g/h}$.
\textbf{8.} $100/6.7 = \qty{15}{h}$.
\textbf{9.} \omterm{def:b1:carbohydrates:polysaccharide}{Glycogen} supplies two thirds, \qty{4.4}{g/h}: $100/4.4 =
\qty{22}{h}$.
\textbf{10.} 11:00 the next morning by the first estimate, 18:00 by the
second --- through the night in either case, with the margin
depending on how early \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{gluconeogenesis} takes over.
\textbf{11.} Resting expenditure over \qty{12}{h}: \qty{4.2}{MJ};
\qty{60}{\%} is \qty{2.5}{MJ}; $2500/38 = \qty{66}{g}$ of \omterm{def:b1:lipids:triglyceride}{fat}.
\textbf{12.} $160/0.55 = \qty{290}{g}$ of \omterm{def:b1:proteins:peptide}{protein}, i.e.\ \qty{1.45}{kg}
of muscle a day.
\textbf{13.} $160/180 = \qty{0.89}{mol}$ of glucose; at 6 \omterm{def:b1:water-small-molecules:atp}{ATP}
each that is \qty{5.3}{mol} of \omterm{def:b1:water-small-molecules:atp}{ATP}, and at \qty{50}{kJ/mol}
about \qty{265}{kJ} --- some \qty{3}{\%} of the day's \qty{8.4}{MJ}: making the
glucose is cheap, and it is the carbon skeletons, not the energy, that
the body is short of.
\textbf{14.} Non-glucose expenditure $8.4 - 160\times 0.017 =
\qty{5.7}{MJ}$: $5700/38 = \qty{150}{g}$ of \omterm{def:b1:lipids:triglyceride}{fat}, whose glycerol gives
\qty{15}{g} of glucose.
\textbf{15.} $(160 - 15)/0.55 = \qty{264}{g}$ of \omterm{def:b1:proteins:peptide}{protein}, \qty{1.3}{kg}
of muscle a day.
\textbf{16.} \qty{3}{kg} of muscle \omterm{def:b1:proteins:peptide}{protein} at \qty{264}{g} a day:
about \qty{11}{d}.
\textbf{17.} $(80 - 15)/0.55 = \qty{118}{g}$ of \omterm{def:b1:proteins:peptide}{protein}, \qty{0.6}{kg}
of muscle a day; half the muscle in \qty{25}{d} --- and in practice
the loss falls further, to \qtyrange{20}{30}{g} a day, as the body
economises.
\textbf{18.} $456/8.4 = 54$ days; at \qty{6.5}{MJ}, 70 days.
\textbf{19.} \omterm{def:b1:proteins:peptide}{Protein} is not a store but the machinery: losing a third
of it (respiratory muscles, heart, immune \omterm{def:b1:proteins:peptide}{proteins}) is fatal, and at
\qty{100}{g} a day or more that happens within weeks, while the \omterm{def:b1:lipids:triglyceride}{fat}
would last two months. Ketone bodies let the brain run on \omterm{def:b1:lipids:triglyceride}{fat}, cut the
glucose demand and hence the \omterm{def:b1:proteins:peptide}{protein} loss to a third, and stretch
survival toward the limit set by the \omterm{def:b1:lipids:triglyceride}{fat}.
\textbf{20.} \omterm{def:b1:carbohydrates:polysaccharide}{Glycogen} phosphorylase active at 03:00, \omterm{def:b1:carbohydrates:polysaccharide}{glycogen} synthase
at 09:00.
\textbf{21.} Glucagon (03:00) and insulin (09:00); phosphorylation of
both \omterm{def:b1:enzymes:enzyme}{enzymes} by a kinase cascade, reversed by a phosphatase.
\textbf{22.} \omterm{def:b1:respiration-fermentation:pdh}{Acetyl-CoA} activates pyruvate carboxylase; \omterm{def:b1:water-small-molecules:atp}{ATP} (and
citrate) inhibits phosphofructokinase.
\textbf{23.} Glucose to pyruvate yields 2 \omterm{def:b1:water-small-molecules:atp}{ATP}; pyruvate back to
glucose costs 6: each round trip burns 4 \omterm{def:b1:water-small-molecules:atp}{ATP} for no product. With
distinct \omterm{def:b1:enzymes:enzyme}{enzymes} at the three bypasses, the same signals (\omterm{def:b1:water-small-molecules:atp}{ATP},
\omterm{def:b1:respiration-fermentation:pdh}{acetyl-CoA}, phosphorylation) activate one set and inhibit the other,
so only one direction is open.
\textbf{24.} Without insulin the valves are set as in fasting whatever
the glucose: phosphorylase, \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{gluconeogenesis} and lipolysis are on,
synthesis is off, and the liver, ``believing'' the body starved, pours
out glucose and ketones into blood already full of glucose that the
insulin-dependent \omterm{def:b1:body-plans-tissues:tissue}{tissues} cannot take up.
\textbf{25.} About \qty{15}{h} without \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{gluconeogenesis}, \qty{22}{h}
with it --- a night and a morning; \omterm{def:b1:proteins:peptide}{protein} loss about \qty{260}{g} a
day at first and \qty{120}{g} (falling further) once ketone bodies feed
the brain; the \omterm{def:b1:lipids:triglyceride}{fat} store lasts some \qtyrange{55}{70}{d}.
\end{solution}
