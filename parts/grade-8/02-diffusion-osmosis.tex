\chapter{Diffusion and Osmosis}\label{ch:g8:diffusion}

\omterm{def:g7:microscope:cell}{Cells} live in a watery world of dissolved substances. Materials move in
and out without the \omterm{def:g7:microscope:cell}{cell} always spending \omterm{def:g6:nutrition:def}{energy} --- and sometimes with
\omterm{def:g6:nutrition:def}{energy}. This chapter focuses on \omterm{def:g8:diffusion:diff}{diffusion}, \omterm{def:g8:diffusion:osmosis}{osmosis} and why \omterm{def:g8:diffusion:diff}{concentration
gradients} matter.

\section{Diffusion}

\begin{definition}[Diffusion]\label{def:g8:diffusion:diff}
\emph{Diffusion}\index{diffusion} is the net movement of particles from a
region of higher concentration to a region of lower concentration, down a
\emph{concentration gradient}\index{concentration gradient}, due to random
motion. It does not require the \omterm{def:g7:microscope:cell}{cell} to supply metabolic \omterm{def:g6:nutrition:def}{energy}.
\end{definition}

\begin{proposition}[What speeds diffusion]\label{prop:g8:diffusion:rate}
\omterm{def:g8:diffusion:diff}{Diffusion} is faster when the gradient is steeper, temperature is higher
(particles move faster), the distance is shorter, and the surface area for
exchange is larger. Small molecules generally diffuse faster than large
ones in the same medium.
\end{proposition}
\admitted

\begin{example}[Smell of perfume]\label{ex:g8:diffusion:perfume}
Perfume molecules diffuse through \omterm{def:g4:plants-need:needs}{air} from high concentration near the
bottle toward lower concentration across the room.
\end{example}

\section{Membranes and osmosis}

\begin{definition}[Partially permeable membrane]\label{def:g8:diffusion:membrane}
A \emph{partially permeable membrane}\index{partially permeable membrane}
(selectively permeable) lets some substances pass more easily than others.
The \omterm{def:g7:microscope:cell}{cell}'s plasma membrane is partially permeable.
\end{definition}

\begin{definition}[Osmosis]\label{def:g8:diffusion:osmosis}
\emph{Osmosis}\index{osmosis} is the net \omterm{def:g8:diffusion:diff}{diffusion} of \emph{\omterm{def:g4:plants-need:needs}{water}} across
a \omterm{def:g8:diffusion:membrane}{partially permeable membrane} from a region of higher \omterm{def:g4:plants-need:needs}{water} concentration
(more dilute solution) to a region of lower \omterm{def:g4:plants-need:needs}{water} concentration (more
concentrated solution).
\end{definition}

\begin{definition}[Hypotonic, hypertonic, isotonic]\label{def:g8:diffusion:tonic}
Compared with the \omterm{def:g7:microscope:cell}{cell} contents: a \emph{hypotonic}\index{hypotonic}
solution is more dilute (\omterm{def:g4:plants-need:needs}{water} tends to enter the \omterm{def:g7:microscope:cell}{cell}); a
\emph{hypertonic}\index{hypertonic} solution is more concentrated (\omterm{def:g4:plants-need:needs}{water}
tends to leave); an \emph{isotonic}\index{isotonic} solution has equal
effective concentration (no net \omterm{def:g4:plants-need:needs}{water} movement).
\end{definition}

\begin{proposition}[Animal vs plant cells in water]\label{prop:g8:diffusion:cells}
\omterm{def:g1:plants-animals:animal}{Animal} \omterm{def:g7:microscope:cell}{cells} in pure \omterm{def:g4:plants-need:needs}{water} may swell and burst (no wall). \omterm{def:g1:plants-animals:plant}{Plant} \omterm{def:g7:microscope:cell}{cells}
become \emph{turgid}\index{turgid} as the wall resists; in strong sugar
or salt solutions they may plasmolyse (membrane pulls from wall) as \omterm{def:g4:plants-need:needs}{water}
\omterm{def:g7:tissues:plant-organs}{leaves}.
\end{proposition}
\admitted

\begin{example}[Raisins in water]\label{ex:g8:diffusion:raisin}
Dried \omterm{def:g6:reproduction:plant}{fruit} placed in \omterm{def:g4:plants-need:needs}{water} swells as \omterm{def:g4:plants-need:needs}{water} enters by \omterm{def:g8:diffusion:osmosis}{osmosis}.
\end{example}

\section{Active transport (brief)}

\begin{definition}[Active transport]\label{def:g8:diffusion:active}
\emph{Active transport}\index{active transport} moves substances against
a \omterm{def:g8:diffusion:diff}{concentration gradient} using \omterm{def:g6:nutrition:def}{energy} from the \omterm{def:g7:microscope:cell}{cell} (ATP) and membrane
\omterm{def:g6:nutrition:groups}{proteins}. \omterm{def:g7:plant-structure:organs}{Root} \omterm{def:g7:microscope:cell}{cells} use active transport to absorb some \omterm{def:g6:nutrition:groups}{minerals}.
\end{definition}

\begin{method}[Deciding which process]\label{met:g8:diffusion:which}
\begin{enumerate}
  \item Is \omterm{def:g4:plants-need:needs}{water} the main mover across a membrane? $\to$ consider \omterm{def:g8:diffusion:osmosis}{osmosis}.
  \item Is a solute moving down its gradient without \omterm{def:g6:nutrition:def}{energy}? $\to$
        \omterm{def:g8:diffusion:diff}{diffusion} (or facilitated \omterm{def:g8:diffusion:diff}{diffusion} if via \omterm{def:g6:nutrition:groups}{proteins}).
  \item Is movement against the gradient needing \omterm{def:g6:nutrition:def}{energy}? $\to$ \omterm{def:g8:diffusion:active}{active
        transport}.
\end{enumerate}
\end{method}
