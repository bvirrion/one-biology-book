\chapter{What Is Alive?}\label{ch:g2:what-alive}

In Grade~1 we sorted living and \omterm{def:g1:living-nonliving:nonliving}{nonliving things}. Now we look more
carefully: what do all \omterm{def:g1:living-nonliving:living}{living things} \emph{do}? Biologists use a
short list of life signs.

\section{Signs of life}

\begin{definition}[Organism]\label{def:g2:what-alive:organism}
An \emph{organism}\index{organism} is a \omterm{def:g1:living-nonliving:living}{living thing} --- a \omterm{def:g1:plants-animals:plant}{plant}, an
\omterm{def:g1:plants-animals:animal}{animal}, or another form of life. You are an organism. A tree is an
organism.
\end{definition}
\omimg{living-nonliving.jpg}{0.50\linewidth}{Living things share needs and ways of changing.}


\begin{definition}[Signs of life]\label{def:g2:what-alive:signs}
Living \omterm{def:g2:what-alive:organism}{organisms} typically:
\begin{itemize}
  \item \emph{grow}\index{growth} and change over time;
  \item need \emph{energy}\index{energy (food)} from \omterm{def:g1:caring:needs}{food} or light;
  \item respond to the world (light, sound, \omterm{def:g1:senses:five}{touch});
  \item get rid of waste;
  \item can \emph{reproduce}\index{reproduction} --- make more of
        their own kind.
\end{itemize}
\end{definition}

\begin{example}[Puppy and toy puppy]\label{ex:g2:what-alive:puppy}
A real puppy \omterm{def:g2:what-alive:signs}{grows}, eats, barks when surprised, and can later have
puppies. A stuffed toy puppy does none of these by itself. Only the
real puppy is an \omterm{def:g2:what-alive:organism}{organism}.
\end{example}

\section{Testing carefully}

\begin{method}[Is it an organism?]\label{met:g2:what-alive:test}
\begin{enumerate}
  \item Watch over time: does it \omterm{def:g2:what-alive:signs}{grow}?
  \item Does it need \omterm{def:g1:caring:needs}{food}, \omterm{def:g1:caring:needs}{water} or light?
  \item Does it react when the world changes?
  \item Can its kind make offspring (babies, eggs, seeds)?
  \item If most answers are yes, it is living.
\end{enumerate}
\end{method}

\begin{example}[Sort again]\label{ex:g2:what-alive:sort}
Living: moss, spider, child. Nonliving: cloud, bicycle, plastic
flower. Once living, now not: wooden chair, dry leaf, wool jumper
(from sheep's hair, no longer growing).
\end{example}

\begin{proposition}[Shared life]\label{prop:g2:what-alive:shared}
All \omterm{def:g2:what-alive:organism}{organisms} share the main signs of life, even when they look very
different --- from tiny insects to tall trees.
\end{proposition}
\admitted

\section{Exercises}

\end{exercise}

\end{exercise}
