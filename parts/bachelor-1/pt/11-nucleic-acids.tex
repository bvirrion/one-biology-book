\chapter{Nucleotídeos e ácidos nucleicos}\label{ch:b1:nucleic-acids}

Despeje etanol gelado sobre uma solução de \omterm{def:b1:cell-unit-of-life:cell}{células} rompidas e aparece
uma nuvem branca de fibras que pode ser enrolada num bastão de vidro: o
\omterm{def:b1:nucleic-acids:chain}{DNA}, a molécula que carrega cada instrução do \omterm{def:b1:organism-environment:organism}{organismo}, visível a olho
nu. Dois metros dele estão dobrados dentro de cada núcleo humano; ele
inteiro é escrito num alfabeto de quatro letras, lido num único sentido,
e copiado pela separação de duas fitas que se encaixam como uma mão e a
luva dela. Este capítulo descreve os \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} --- letras, \omterm{def:b1:water-small-molecules:atp}{moeda
energética} e coenzimas ao mesmo tempo ---, as cadeias que eles formam, a
dupla-hélice e as evidências dela, o que acontece quando ela é fundida e
reassociada, e os \omterm{def:b1:nucleic-acids:chain}{RNAs} que levam a mensagem dela para dentro da \omterm{def:b1:cell-unit-of-life:cell}{célula}.

\section{Nucleotídeos}

\begin{definition}[Nucleotídeo]\label{def:b1:nucleic-acids:nucleotide}
Um \emph{nucleotídeo}\index{nucleotídeo} é feito de três partes: uma
\emph{base}\index{base nitrogenada} nitrogenada --- uma
\emph{purina}\index{purina} (adenina A, guanina G: dois anéis fundidos)
ou uma \emph{pirimidina}\index{pirimidina} (citosina C, timina T,
uracila U: um anel); um açúcar de cinco carbonos, a \emph{ribose} no \omterm{def:b1:nucleic-acids:chain}{RNA}
ou a \emph{2$'$-desoxirribose} no \omterm{def:b1:nucleic-acids:chain}{DNA} (sem a hidroxila do carbono
$2'$); e de um a três grupos \emph{fosfato} no carbono $5'$ do açúcar. A
base é ligada ao carbono $1'$; os carbonos do açúcar são numerados com
linhas para distingui-los dos da base. Base mais açúcar formam um
\emph{nucleosídeo} (adenosina, guanosina, citidina, timidina, uridina);
com fosfatos, um nucleotídeo (AMP, ADP, \omterm{def:b1:water-small-molecules:atp}{ATP} e os parentes deles).
\end{definition}

\begin{proposition}[Os nucleotídeos fazem três trabalhos]\label{prop:b1:nucleic-acids:threejobs}
Além de serem os monômeros do \omterm{def:b1:nucleic-acids:chain}{DNA} e do \omterm{def:b1:nucleic-acids:chain}{RNA}, os \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} são a
\emph{\omterm{def:b1:water-small-molecules:atp}{moeda energética}} da \omterm{def:b1:cell-unit-of-life:cell}{célula} --- o \omterm{def:b1:water-small-molecules:atp}{ATP} e o GTP, cujas \omterm{def:b1:water-small-molecules:atp}{ligações
fosfoanidrido} (\cref{ch:b1:water-small-molecules}) pagam pela síntese,
pelo transporte e pelo movimento; as \emph{coenzimas} dela ---
$\mathrm{NAD^+}$, $\mathrm{NADP^+}$ e FAD, que carregam elétrons na
respiração e na fotossíntese
(\cref{ch:b1:respiration-fermentation}), e a coenzima A, que carrega
grupos acila; e os \emph{mensageiros} dela --- o AMP cíclico, que
retransmite sinais hormonais dentro das \omterm{def:b1:cell-unit-of-life:cell}{células}. Todos eles são
\omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} de adenina ou são construídos sobre eles: a mesma molécula
que escreve a mensagem alimenta a execução dela.
\end{proposition}

\section{A cadeia polinucleotídica}

\begin{definition}[Ligação fosfodiéster, polaridade]\label{def:b1:nucleic-acids:chain}
Os \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} são unidos num
\emph{polinucleotídeo}\index{polinucleotídeo} por
\emph{ligações fosfodiéster}\index{ligação fosfodiéster}: o fosfato do
carbono $5'$ de um \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeo} é ligado à hidroxila $3'$ do seguinte. A
cadeia tem, portanto, um \emph{esqueleto} de açúcar e fosfato, uniforme
e carregado negativamente (uma carga por fosfato), do qual as \omterm{def:b1:nucleic-acids:nucleotide}{bases} se
projetam como uma sequência; e ela tem um sentido: uma
\emph{extremidade $5'$}\index{extremidade 5'@extremidade $5'$} com um
fosfato livre e uma
\emph{extremidade $3'$}\index{extremidade 3'@extremidade $3'$} com uma
hidroxila livre. As sequências são escritas e lidas de $5' \to 3'$, o
sentido em que as cadeias são sintetizadas. O \emph{DNA}\index{DNA}
(ácido desoxirribonucleico) usa desoxirribose e as \omterm{def:b1:nucleic-acids:nucleotide}{bases} A, G, C, T; o
\emph{RNA}\index{RNA} (ácido ribonucleico) usa ribose e A, G, C, U.
\end{definition}

\begin{example}[O RNA é frágil, o DNA é durável]\label{ex:b1:nucleic-acids:fragile}
A hidroxila $2'$ da ribose pode atacar a \omterm{def:b1:nucleic-acids:chain}{ligação fosfodiéster} vizinha:
em álcali brando o \omterm{def:b1:nucleic-acids:chain}{RNA} é hidrolisado a \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} em poucas horas, ao
passo que o \omterm{def:b1:nucleic-acids:chain}{DNA}, que não tem essa hidroxila, sobrevive à fervura em
álcali e já foi recuperado intacto de ossos com dezenas de milhares de
anos. A uracila do \omterm{def:b1:nucleic-acids:chain}{RNA} é a timina sem o grupo metila; o \omterm{def:b1:nucleic-acids:chain}{DNA} usa timina
para que a citosina, que lentamente perde o grupo amino e vira uracila,
possa ser reconhecida como danificada e reparada --- uma uracila no \omterm{def:b1:nucleic-acids:chain}{DNA}
é sempre um erro.
\end{example}

\section{A dupla-hélice}

\begin{proposition}[As regras de Chargaff]\label{prop:b1:nucleic-acids:chargaff}
\index{regras de Chargaff}
No \omterm{def:b1:nucleic-acids:chain}{DNA} de qualquer \omterm{def:b1:organism-environment:organism}{organismo} a quantidade de adenina é igual à de timina
e a de guanina é igual à de citosina ($A = T$, $G = C$, e portanto
\omterm{def:b1:nucleic-acids:nucleotide}{purinas} $=$ \omterm{def:b1:nucleic-acids:nucleotide}{pirimidinas}), ao passo que a razão $(G + C)/(A + T)$ varia
de espécie para espécie (de \qty{25}{\%} a \qty{75}{\%} de $G + C$).
\end{proposition}
\begin{proof}[Evidências]
Chargaff (1950) hidrolisou \omterm{def:b1:nucleic-acids:chain}{DNA} de muitos \omterm{def:b1:organism-environment:organism}{organismos} e separou as \omterm{def:b1:nucleic-acids:nucleotide}{bases}
por cromatografia, medindo cada uma pela absorção no ultravioleta. As
igualdades valeram dentro do erro experimental para todo \omterm{def:b1:nucleic-acids:chain}{DNA}; o teor de
$G + C$ não, e ele era o mesmo em todos os \omterm{def:b1:body-plans-tissues:tissue}{tecidos} de um mesmo
\omterm{def:b1:organism-environment:organism}{organismo}.
\end{proof}

\begin{theorem}[A dupla-hélice de Watson e Crick]\label{thm:b1:nucleic-acids:helix}
\index{dupla-hélice}\index{par de bases}\index{antiparalelo}
O \omterm{def:b1:nucleic-acids:chain}{DNA} são duas cadeias polinucleotídicas enroladas em torno de um eixo
comum numa \emph{dupla-hélice} destra: os esqueletos de açúcar e fosfato
por fora, as \omterm{def:b1:nucleic-acids:nucleotide}{bases} por dentro, planas e empilhadas perpendicularmente ao
eixo. As cadeias são \emph{antiparalelas} (uma corre de $5' \to 3'$ para
cima, a outra para baixo) e \emph{complementares}: em frente a cada A
há um T, seguro por duas \omterm{def:b1:water-small-molecules:water}{ligações de hidrogênio}, e em frente a cada G um
C, seguro por três; cada um desses \emph{\omterm{thm:b1:nucleic-acids:helix}{pares de bases}} tem a mesma
largura, de modo que a hélice é uniforme seja qual for a sequência. Na
forma B, a comum, a hélice tem \qty{2}{nm} de diâmetro, sobe
\qty{0.34}{nm} por \omterm{thm:b1:nucleic-acids:helix}{par de bases} e \qty{3.4}{nm} por volta de dez pares,
e mostra um \emph{sulco maior} largo e um \emph{sulco menor} estreito,
em que as bordas das \omterm{def:b1:nucleic-acids:nucleotide}{bases} ficam expostas às proteínas.
\end{theorem}
\begin{proof}[Evidências]
As imagens de difração de raios X de fibras de \omterm{def:b1:nucleic-acids:chain}{DNA} obtidas por Franklin
(1952) mostravam a cruz de uma hélice com repetição de \qty{3.4}{nm} ao
longo do eixo, uma reflexão forte a \qty{0.34}{nm} (unidades planas
empilhadas) e um diâmetro de \qty{2}{nm}; os fosfatos, os átomos mais
pesados, tinham de estar por fora. Watson e Crick (1953) descobriram
que uma \omterm{def:b1:nucleic-acids:nucleotide}{purina} pareada com uma \omterm{def:b1:nucleic-acids:nucleotide}{pirimidina} --- e apenas A com T e G com C
--- dá pares de largura igual, que cabem no diâmetro de \qty{2}{nm},
explicando as \omterm{prop:b1:nucleic-acids:chargaff}{regras de Chargaff}, e que duas cadeias antiparalelas assim
pareadas constroem uma hélice regular com as dimensões da fibra. O
modelo previa que cada cadeia é molde da outra; a demonstração da
replicação semiconservativa por Meselson e Stahl (retomada do volume do
Ensino Médio, e no \cref{ch:b1:replication-mitosis}) o confirmou.
\end{proof}

\begin{example}[Ler uma fita]\label{ex:b1:nucleic-acids:complement}
Se uma fita lê $5'$-ATGGCATTC-$3'$, a parceira dela, escrita de
$5' \to 3'$, é $5'$-GAATGCCAT-$3'$: complemente cada \omterm{def:b1:nucleic-acids:nucleotide}{base} e depois
inverta a ordem. Nove pares, \qty{3.1}{nm} de hélice, vinte e duas
\omterm{def:b1:water-small-molecules:water}{ligações de hidrogênio} (cinco G--C, quatro A--T). Um cromossomo humano
de \num{2.5e8} pares tem \qty{8.5}{cm} de comprimento nessa forma e
precisa ser dobrado cem mil vezes para caber num núcleo
(\cref{ch:b1:genomes}).
\end{example}

\begin{proposition}[O que mantém a hélice unida]\label{prop:b1:nucleic-acids:stability}
As \omterm{def:b1:water-small-molecules:water}{ligações de hidrogênio} entre as \omterm{def:b1:nucleic-acids:nucleotide}{bases} pareadas dão a especificidade;
o \emph{empilhamento}\index{empilhamento de bases} das \omterm{def:b1:nucleic-acids:nucleotide}{bases} planas umas
sobre as outras --- contatos \omterm{def:b1:water-small-molecules:bonds}{de van der Waals} e o \omterm{prop:b1:water-small-molecules:properties}{efeito hidrofóbico},
que mantém as \omterm{def:b1:nucleic-acids:nucleotide}{bases} fora da água --- dá a maior parte da estabilidade;
os esqueletos carregados negativamente se repelem, e a hélice só é
estável na presença de cátions (o $\mathrm{Mg^{2+}}$ e o
$\mathrm{Na^+}$ fisiológicos, ou as proteínas básicas da cromatina) que
blindem a carga. Um par G--C, com três \omterm{def:b1:water-small-molecules:water}{ligações de hidrogênio} e
empilhamento mais forte, segura melhor que um par A--T.
\end{proposition}

\section{Fusão e hibridização}

\begin{definition}[Desnaturação, temperatura de fusão]\label{def:b1:nucleic-acids:melting}
O aquecimento, ou um pH extremo, separa as duas fitas do \omterm{def:b1:nucleic-acids:chain}{DNA}: é a
\emph{desnaturação}\index{desnaturação (DNA)} (fusão), uma transição
cooperativa ao longo de poucos graus, cujo ponto médio é a
\emph{temperatura de fusão}\index{temperatura de fusão} $T_m$. Ela é
acompanhada do \emph{efeito hipercrômico}: as fitas simples absorvem
cerca de \qty{40}{\%} mais luz ultravioleta a \qty{260}{nm} que as
mesmas \omterm{def:b1:nucleic-acids:nucleotide}{bases} empilhadas numa hélice. O $T_m$ sobe com o teor de $G + C$
(\qty{0.4}{\celsius} por ponto percentual), com a concentração de sal e
com o comprimento; em sal a \qty{0.15}{mol/L}, um \omterm{def:b1:nucleic-acids:chain}{DNA} longo com
\qty{50}{\%} de $G + C$ funde perto de \qty{90}{\celsius}. Resfriadas
devagar, as fitas encontram as complementares e refazem a hélice: é a
\emph{renaturação}, ou \emph{hibridização}\index{hibridização} quando as
fitas vêm de fontes diferentes.
\end{definition}

\begin{method}[Medir ácidos nucleicos pela absorbância]\label{met:b1:nucleic-acids:absorbance}
\begin{enumerate}
  \item As \omterm{def:b1:nucleic-acids:nucleotide}{bases} absorvem luz ultravioleta com máximo em \qty{260}{nm}.
        Numa cubeta de \qty{1}{cm}, uma absorbância de $1.0$ corresponde
        a \qty{50}{\micro g/mL} de \omterm{def:b1:nucleic-acids:chain}{DNA} de fita dupla
        (\qty{40}{\micro g/mL} de \omterm{def:b1:nucleic-acids:chain}{RNA}, \qty{33}{\micro g/mL} de \omterm{def:b1:nucleic-acids:chain}{DNA} de
        fita simples).
  \item A razão $A_{260}/A_{280}$ é $1.8$ para \omterm{def:b1:nucleic-acids:chain}{DNA} puro e $2.0$ para \omterm{def:b1:nucleic-acids:chain}{RNA}
        puro; as proteínas absorvem a \qty{280}{nm} e a diminuem.
  \item Para acompanhar a fusão, registre $A_{260}$ enquanto aquece
        devagar; o ponto médio da curva é o $T_m$ e a inclinação dela é
        uma medida da homogeneidade da amostra.
  \item Para detectar uma sequência, desnature a amostra e acrescente
        uma fita simples marcada, complementar a ela (uma
        \emph{sonda}); ela hibridiza só onde está a complementar dela, e
        a marcação assinala o lugar --- num gel, num cromossomo, num
        \omterm{def:b1:body-plans-tissues:tissue}{tecido}.
\end{enumerate}
\end{method}

\begin{example}[Uma sonda encontra um gene]\label{ex:b1:nucleic-acids:probe}
Uma sonda de vinte \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} complementar a um gene, a
\qty{5}{\celsius} abaixo do $T_m$ dela, liga-se à complementar exata
dela e a mais nada em três bilhões de \omterm{thm:b1:nucleic-acids:helix}{pares de bases}: um único
descasamento em vinte baixa o $T_m$ em vários graus e o híbrido
descasado funde. A especificidade do pareamento de \omterm{def:b1:nucleic-acids:nucleotide}{bases}, multiplicada
por vinte posições, é aquilo em que se apoia todo teste de \omterm{def:b1:nucleic-acids:chain}{DNA}, do exame
de paternidade à detecção de patógenos.
\end{example}

\section{RNA}

\begin{definition}[Os RNAs]\label{def:b1:nucleic-acids:rnas}
O \omterm{def:b1:nucleic-acids:chain}{RNA} em geral tem fita simples e se dobra sobre si mesmo onde quer que
trechos complementares curtos permitam, formando grampos, alças e
protuberâncias --- uma \emph{estrutura secundária} --- que depois se
compacta numa forma. Três classes carregam o fluxo da informação
genética (\cref{ch:b1:gene-expression}): o
\emph{RNA mensageiro}\index{RNA mensageiro} (mRNA), uma cópia de um gene
que o ribossomo lê; o
\emph{RNA transportador}\index{RNA transportador} (tRNA), com cerca de
\qty{75}{} \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} dobrados numa \omterm{def:b1:flowering-plant-organization:organs}{folha} de trevo, que carrega um
aminoácido na extremidade $3'$ dele e um \emph{anticódon} que pareia com
a mensagem; e o \emph{RNA ribossômico}\index{RNA ribossômico} (rRNA), o
núcleo estrutural e catalítico do ribossomo, quatro quintos do \omterm{def:b1:nucleic-acids:chain}{RNA} de
uma \omterm{def:b1:cell-unit-of-life:cell}{célula}. Muitos outros \omterm{def:b1:nucleic-acids:chain}{RNAs} regulam, fazem splicing e guiam (o volume
do Ano 3).
\end{definition}

\begin{remark}[O RNA pode ser catalisador]\label{rem:b1:nucleic-acids:ribozyme}
Alguns \omterm{def:b1:nucleic-acids:chain}{RNAs} são enzimas (as \emph{ribozimas}): a ligação peptídica de
toda proteína é formada pelo \omterm{def:b1:nucleic-acids:rnas}{RNA ribossômico}, e não por uma proteína, e
alguns \omterm{def:b1:nucleic-acids:chain}{RNAs} cortam e emendam a si mesmos. Uma molécula que ao mesmo
tempo carrega informação e catalisa reações é do que um primeiro sistema
vivo pode ter sido feito; o mundo do \omterm{def:b1:nucleic-acids:chain}{DNA} e das proteínas seria então uma
divisão de trabalho posterior, com o \omterm{def:b1:nucleic-acids:chain}{RNA} mantido no centro dela.
\end{remark}

$5' \to 3'$, e conte as \omterm{def:b1:water-small-molecules:water}{ligações de hidrogênio} dela.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:3}
Um \omterm{def:b1:nucleic-acids:chain}{DNA} contém \qty{22}{\%} de adenina. Dê as porcentagens de T, G e C, e
o teor de $G + C$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:4}
A partir da figura da fusão, qual é o $T_m$ de um \omterm{def:b1:nucleic-acids:chain}{DNA} com \qty{50}{\%}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:5}

\omterm{thm:b1:nucleic-acids:helix}{pares de bases}) em milímetros, o número de voltas dele, e o fator pelo

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:6}

que todo \omterm{thm:b1:nucleic-acids:helix}{par de bases} tenha a mesma largura.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:7}
Uma solução de \omterm{def:b1:nucleic-acids:chain}{DNA} tem $A_{260} = 0.35$ numa cubeta de \qty{1}{cm} e
$A_{280} = 0.20$. Calcule a concentração e comente a pureza. Quantos
\omterm{thm:b1:nucleic-acids:helix}{pares de bases} há em \qty{1}{mL}, se a massa média de um par for

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:8}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:9}

\qty{5}{\celsius}. Em que temperatura a \omterm{def:b1:nucleic-acids:melting}{hibridização} deve ser feita para

acontece a \qty{45}{\celsius}?

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:10}
As \omterm{prop:b1:nucleic-acids:chargaff}{regras de Chargaff} valem para o \omterm{def:b1:nucleic-acids:chain}{DNA} de fita dupla, mas não para o \omterm{def:b1:nucleic-acids:chain}{RNA}
nem para o \omterm{def:b1:nucleic-acids:chain}{DNA} de certos vírus pequenos. Explique o que cada exceção diz
sobre a estrutura desses ácidos nucleicos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:11}
Duas \omterm{def:b1:water-small-molecules:water}{ligações de hidrogênio} seguram um par A--T, e no entanto uma hélice
de mil pares A--T é estável a \qty{60}{\celsius}, ao passo que um par
isolado nem chega a se formar. Explique com as ideias do
\cref{ch:b1:water-small-molecules} (ligações fracas, cooperatividade,
empilhamento), e diga o que isso implica para a fidelidade de uma sonda.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:12}
``A estrutura do \omterm{def:b1:nucleic-acids:chain}{DNA} é a explicação da hereditariedade.'' Discuta em um
parágrafo: o que a dupla-hélice explica de imediato (a cópia, a mutação
como mudança de sequência), o que ela não explica sozinha (como a
sequência vira uma proteína), e por que o modelo de Watson e Crick foi
aceito antes de qualquer teste bioquímico dele.
\end{exercise}

\section{Problema: O genoma de um vírus}

\begin{problem}[{Problema de fim de semana --- o DNA do fago $\lambda$
medido, pesado, contado, fundido e empacotado no capsídeo dele,
terminando na temperatura de fusão e no comprimento de contorno dele}]
\label{pb:b1:nucleic-acids:1}
O genoma do bacteriófago $\lambda$ é um único \omterm{def:b1:nucleic-acids:chain}{DNA} linear de fita dupla
com \num{48502} \omterm{thm:b1:nucleic-acids:helix}{pares de bases} e teor de $G + C$ de \qty{49.9}{\%}. Ele
é empacotado num capsídeo icosaédrico de \qty{55}{nm} de diâmetro
interno. Tome \qty{0.34}{nm} por \omterm{thm:b1:nucleic-acids:helix}{par de bases}, \qty{2.0}{nm} para o
diâmetro da hélice, \qty{650}{Da} por \omterm{thm:b1:nucleic-acids:helix}{par de bases}, e
$T_m = 81.5 + 16.6\log_{10}[\mathrm{Na^+}]
+ 0.41\,(\%\,G{+}C) - 500/L$ em graus Celsius, com $[\mathrm{Na^+}]$ em
\unit{mol/L} e $L$ o comprimento em \omterm{thm:b1:nucleic-acids:helix}{pares de bases}.

\medskip
\noindent\textbf{Parte I --- Medido e pesado.}
\begin{enumerate}
  \item Calcule o comprimento de contorno da molécula em micrômetros.
  \item Calcule o número de voltas da hélice.
  \item Calcule a massa molar dela em dáltons.
  \item Calcule a massa de uma molécula em gramas.
  \item Calcule o número de grupos fosfato e, portanto, a carga líquida
        da molécula.
  \item Calcule o número de \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} de cada \omterm{def:b1:nucleic-acids:nucleotide}{base} (A, T, G, C).
  \item Calcule o número total de \omterm{def:b1:water-small-molecules:water}{ligações de hidrogênio} que seguram as
        duas fitas.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Fundido.}
\begin{enumerate}[resume]
  \item Calcule o $T_m$ em sódio a \qty{0.1}{mol/L}.
  \item Calcule o $T_m$ em sódio a \qty{0.01}{mol/L} e explique por que
        o sal estabiliza a hélice.
  \item O genoma de $\lambda$ contém uma região de \qty{1000}{} pares
        com \qty{35}{\%} de $G + C$ e outra com \qty{62}{\%}. Calcule o
        $T_m$ de cada região em sódio a \qty{0.1}{mol/L} e descreva a
        forma da curva de fusão da molécula inteira.
  \item Uma solução de \omterm{def:b1:nucleic-acids:chain}{DNA} de $\lambda$ tem $A_{260} = 0.50$ antes da
        fusão. Qual é a concentração dela, e quanto valerá $A_{260}$
        depois da fusão completa?
  \item Quantas moléculas de $\lambda$ há em \qty{1}{mL} dessa solução?
  \item As duas extremidades do genoma de $\lambda$ são saliências de
        fita simples de 12 \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos}, complementares entre si, pelas
        quais a molécula se circulariza no hospedeiro. Calcule o $T_m$
        delas (use $L = 12$) e diga por que, apesar disso, o círculo se
        mantém a \qty{37}{\celsius} dentro da \omterm{def:b1:cell-unit-of-life:cell}{célula}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Empacotado.}
\begin{enumerate}[resume]
  \item Calcule o volume do \omterm{def:b1:nucleic-acids:chain}{DNA} como um cilindro.
  \item Calcule o volume interno do capsídeo.
  \item Que fração do capsídeo o \omterm{def:b1:nucleic-acids:chain}{DNA} preenche? Compare com o
        empacotamento mais compacto de cilindros paralelos
        (\qty{91}{\%}).
  \item O \omterm{def:b1:nucleic-acids:chain}{DNA} carrega \num{97000} cargas negativas numa esfera de
        \qty{55}{nm}. Explique o que precisa acompanhar o \omterm{def:b1:nucleic-acids:chain}{DNA} para
        dentro do capsídeo, e por que o \omterm{def:b1:nucleic-acids:chain}{DNA} empacotado exerce sobre a
        casca uma pressão (dezenas de atmosferas) que ajuda a injetá-lo
        numa bactéria.
  \item A molécula precisa ser dobrada em alças de raio comparável ao do
        capsídeo. O \omterm{def:b1:nucleic-acids:chain}{DNA} resiste ao dobramento abaixo de um raio de cerca
        de \qty{50}{nm}. Comente.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Lido.}
\begin{enumerate}[resume]
  \item Uma fita do genoma começa por $5'$-GGGCGGCGACCT-$3'$. Escreva a
        fita complementar de $5' \to 3'$.
  \item Se um gene de \num{1200} \omterm{thm:b1:nucleic-acids:helix}{pares de bases} for transcrito num mRNA,
        qual é o comprimento do mRNA em \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeos} e, a \qty{330}{Da}
        por \omterm{def:b1:nucleic-acids:nucleotide}{nucleotídeo}, a massa dele?
  \item O genoma tem \qty{75}{} genes em \num{48502} pares. Que fração
        do genoma é codificante, se o gene médio tiver \num{600} pares?
        Compare com um genoma humano (\qty{1.5}{\%}).
  \item O genoma é replicado em \qty{20}{min} a \qty{37}{\celsius} pelas
        enzimas do hospedeiro. Calcule a velocidade em \omterm{thm:b1:nucleic-acids:helix}{pares de bases}
        por segundo, se a replicação começar numa única origem e
        prosseguir nos dois sentidos.
  \item Quantas \omterm{def:b1:water-small-molecules:water}{ligações de hidrogênio} são rompidas por segundo nas duas
        forquilhas durante a replicação (use a média por par da
        questão 7)?
  \item Uma mutação muda um par G--C em A--T. Quanto muda o $T_m$ da
        molécula inteira? A fusão conseguiria detectá-la?
  \item Enuncie o resultado: a \omterm{def:b1:nucleic-acids:melting}{temperatura de fusão} do \omterm{def:b1:nucleic-acids:chain}{DNA} de $\lambda$
        em sódio a \qty{0.1}{mol/L} e o comprimento de contorno dele, e
        a compactação que o faz caber no capsídeo.
\end{enumerate}
\end{problem}
