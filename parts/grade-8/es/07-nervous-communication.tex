\chapter{La comunicación nerviosa}\label{ch:g8:nervous-communication}

El \cref{ch:g5:brain-in-command} dibujó el bucle --- informar, decidir,
ordenar --- y midió su quinto de segundo. Este curso abrimos los
cables. ¿Qué viaja en realidad por un \omterm{def:g8:nervous-communication:neuron}{nervio}? ¿De qué están hechos los
\omterm{def:g5:brain-in-command:system}{nervios}? ¿Y qué pasa en los huecos donde acaba un cable y empieza el
siguiente --- los huecos en los que, como se verá, hacen su trabajo
tanto la memoria como muchos venenos?

\section{El cableado y sus células}

\begin{definition}[Neuronas y mensajes nerviosos]\label{def:g8:nervous-communication:neuron}
Las \omterm{def:g5:first-look-at-cells:cell}{células} que trabajan en el \omterm{def:g5:brain-in-command:system}{sistema nervioso} son las
\emph{neuronas}\index{neurona} (ya las enumeraba la
\cref{rem:g6:cell-unit-of-life:twoways}): cada una tiene un cuerpo
celular y una fibra larga y fina --- algunas fibras van de la columna
al dedo del pie, las \omterm{def:g5:first-look-at-cells:cell}{células} más largas del cuerpo. Un
\emph{nervio}\index{nervio} es un cable que agrupa miles de esas
fibras. Por cada fibra viajan \emph{mensajes nerviosos}\index{mensaje
nervioso}: señales eléctricas breves, que corren hasta a
\qty{100}{m/s} (la cifra de la
\cref{def:g5:brain-in-command:system}, ahora con su portador).
\end{definition}

\begin{proposition}[Codificar por frecuencia]\label{prop:g8:nervous-communication:code}
Las señales en sí son todas iguales --- lo que varía es su
\emph{ritmo}. Un roce suave manda un repiqueteo lento por la fibra
sensitiva; uno fuerte, un redoble rápido. El idioma del \omterm{def:g5:brain-in-command:system}{sistema
nervioso} tiene una sola palabra repetida a distintas velocidades: la
intensidad se codifica como frecuencia, en todas las líneas, de entrada
y de salida.
\end{proposition}
\admitted

\begin{example}[Tres líneas, un solo idioma]\label{ex:g8:nervous-communication:lines}
El reparto del bucle (\cref{prop:g5:brain-in-command:loop}), vuelto a
cablear: las \emph{\omterm{def:g8:nervous-communication:neuron}{neuronas} sensitivas} llevan los informes --- de la
\omterm{def:g1:the-five-senses:sense}{piel} a la \omterm{def:g5:brain-in-command:system}{médula espinal}, del \omterm{def:g1:the-five-senses:sense}{ojo} al \omterm{def:g5:brain-in-command:system}{cerebro}; las \emph{\omterm{def:g8:nervous-communication:neuron}{neuronas}
motoras} llevan las órdenes --- de la médula al \omterm{def:g3:bones-and-muscles:muscle}{músculo}, y terminan
donde la fibra se encuentra con el \omterm{def:g3:bones-and-muscles:muscle}{músculo} y su redoble fija la fuerza
de la contracción (\cref{def:g3:bones-and-muscles:muscle}); y entre
unas y otras, en el \omterm{def:g5:brain-in-command:system}{cerebro} y en la médula, un número incontable de
\emph{\omterm{def:g8:nervous-communication:neuron}{neuronas} de conexión} hacen el trabajo de decidir. Reflejo o
sinfonía, el tráfico son las mismas señales a distintos ritmos por
distintas líneas.
\end{example}

\section{Los huecos: las sinapsis}

\begin{definition}[La sinapsis]\label{def:g8:nervous-communication:synapse}
Las \omterm{def:g8:nervous-communication:neuron}{neuronas} no se tocan. Donde el extremo de una fibra se encuentra
con la \omterm{def:g5:first-look-at-cells:cell}{célula} siguiente hay un hueco microscópico --- la
\emph{sinapsis}\index{sinapsis}. La señal eléctrica no lo puede saltar;
en su lugar, la fibra que llega suelta un \emph{mensajero químico} al
hueco, que lo cruza, se acopla al otro lado y --- si llega bastante ---
arranca la señal de nuevo en la \omterm{def:g8:nervous-communication:neuron}{neurona} siguiente. Eléctrico por el
cable y químico en cada empalme: el sistema entero va alternando las
dos cosas.
\end{definition}

\begin{proposition}[Por qué huecos y no soldadura]\label{prop:g8:nervous-communication:whygaps}
Los huecos parecen un defecto y son la genialidad del sistema:
\begin{itemize}
  \item hacen que los empalmes sean \emph{ajustables} --- una \omterm{def:g8:nervous-communication:synapse}{sinapsis}
        que se cruza a menudo se hace más fuerte y una que no se usa se
        apaga: el aprendizaje (los ``mejores caminos'' de la
        \cref{rem:g5:brain-in-command:training}, encontrados --- son
        \omterm{def:g8:nervous-communication:synapse}{sinapsis} reforzadas);
  \item hacen que las señales vayan en \emph{un solo sentido} --- los
        mensajeros se sueltan solo por un lado, así que el tráfico no
        puede ir hacia atrás;
  \item permiten que una \omterm{def:g8:nervous-communication:neuron}{neurona} \emph{sopese} miles de entradas ---
        voces que excitan y voces que frenan sumadas en cada \omterm{def:g5:first-look-at-cells:cell}{célula}
        antes de que dispare: el decidir del bucle, hecho en los
        empalmes.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Las sinapsis de la malabarista]\label{ex:g8:nervous-communication:juggler}
La malabarista de la \cref{rem:g5:brain-in-command:training},
explicada por fin: un mes de práctica condujo las mismas señales por
los mismos empalmes hasta que esas \omterm{def:g8:nervous-communication:synapse}{sinapsis} se reforzaron --- el
enrutado que antes se sopesaba a duras penas viene ahora ya sopesado.
La destreza es química consolidada en los huecos; ``la práctica hace
permanente'' es una afirmación sobre \omterm{def:g8:nervous-communication:synapse}{sinapsis}.
\end{example}

\begin{remark}[Los huecos son los puntos vulnerables]\label{rem:g8:nervous-communication:vulnerable}
Todo lo que actúa sobre el \omterm{def:g5:brain-in-command:system}{sistema nervioso} desde fuera actúa en las
\omterm{def:g8:nervous-communication:synapse}{sinapsis} --- el único sitio en el que la señal se convierte en química,
acoplable y engañable. Los anestésicos las acallan; los venenos las
atascan (unos paralizan bloqueando el mensajero del empalme con el
\omterm{def:g3:bones-and-muscles:muscle}{músculo} y otros inundándolo); y el alcohol de la
\cref{prop:g5:healthy-choices:substances}, y todas las drogas que
cambian el ánimo o la percepción, funcionan imitando, bloqueando o
agotando a los mensajeros de los huecos. El
\cref{ch:g8:protecting-the-nervous-system} se construye justo sobre
este punto.
\end{remark}

\section{Cables rápidos, correo lento: dos sistemas comparados}

\begin{proposition}[Nervios frente a hormonas]\label{prop:g8:nervous-communication:compare}
El cuerpo enseña ya dos sistemas de comunicación, construidos para
tareas contrarias:
\begin{itemize}
  \item el \emph{nervioso}: señales en milisegundos, por líneas
        dedicadas y hacia direcciones precisas --- un \omterm{def:g3:bones-and-muscles:muscle}{músculo}, una
        glándula --- y que se acaban cuando se acaba la señal: el
        sistema de las paradas, del habla y de los reflejos;
  \item el \emph{hormonal} (\cref{def:g8:puberty:hormones}):
        mensajeros en minutos o en meses, emitidos por la \omterm{prop:g4:journey-of-food:absorb}{sangre} a
        todos los órganos construidos para oírlos, que ordenan
        programas lentos y de todo el cuerpo --- crecimiento, ciclos,
        la reconstrucción de la \omterm{def:g8:puberty:puberty}{pubertad}.
\end{itemize}
Telégrafo y servicio de correos: el cuerpo lleva los dos, y el
\cref{ch:g8:hormonal-communication} le dedica al de correos un capítulo
propio.
\end{proposition}
\admitted

\begin{method}[Diagnosticar una comunicación, sistema a sistema]\label{met:g8:nervous-communication:diagnose}
Ante cualquier coordinación del cuerpo, asígnale su sistema:
\begin{enumerate}
  \item cronométrala: milisegundos --- nervioso; de minutos a años ---
        hormonal (o los dos, en relevo);
  \item cartografía su alcance: una dirección precisa --- nervioso;
        todo el cuerpo --- hormonal;
  \item busca la ruta: un camino de cable que se pueda nombrar ---
        nervioso; una emisión por el torrente sanguíneo --- hormonal;
  \item comprueba su persistencia: se acaba con la señal --- nervioso;
        sigue mientras circule el mensajero --- hormonal.
\end{enumerate}
\end{method}

\begin{example}[El método en tres casos]\label{ex:g8:nervous-communication:cases}
El parpadeo ante un soplo de aire: milisegundos, un anillo de \omterm{def:g3:bones-and-muscles:muscle}{músculo},
camino cableado, se acaba en el acto --- nervioso. El estirón de
crecimiento: años, todo el cuerpo, torrente sanguíneo, en marcha
durante temporadas --- hormonal. El \omterm{def:g4:heart-and-blood:heart}{corazón} acelerado del esprint
(\cref{prop:g7:organs-at-work:supply}): los dos en relevo --- unas
señales nerviosas suben el ritmo en un \omterm{def:g4:heart-and-blood:heart}{latido}, y un mensajero del
torrente sanguíneo (la famosa oleada que se nota como el calor de un
susto) mantiene la alerta del cuerpo entero durante minutos. Primero el
telégrafo y después el correo.
\end{example}

\section{Problema: El banco de los reflejos}

\begin{problem}[{Problema de fin de semana --- el reflejo rotuliano,
instrumentado}]\label{pb:g8:nervous-communication:1}
La demostración clásica de laboratorio: el voluntario sentado, el
martillito debajo de la rótula y la patada involuntaria del pie ---
ahora instrumentada con cronómetros y con este capítulo.

\medskip\noindent\textbf{Parte I --- El circuito.}
\begin{enumerate}
  \item El martillo estira un tendón; el pie da la patada una
        cincuentava parte de segundo después. Expón el circuito: qué
        clases de \omterm{def:g8:nervous-communication:neuron}{neuronas}, por dónde --- y por qué el \omterm{def:g5:brain-in-command:system}{cerebro} no está
        en este bucle.
  \item Una cincuentava parte de segundo es más rápida que el quinto de
        la parada de pelota. Explica la diferencia con la longitud del
        circuito y con el número de empalmes.
  \item El voluntario no puede reprimir la patada, aunque se le avise.
        ¿Qué dice esto sobre dónde se decide el reflejo --- y sobre el
        tráfico de un solo sentido en las \omterm{def:g8:nervous-communication:synapse}{sinapsis}?
  \item Justo después de la patada, el voluntario \emph{nota} el
        golpecito. Explica el segundo viaje, más lento, de ese retraso.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Variaciones de laboratorio.}
\begin{enumerate}[resume]
  \item Golpea más fuerte: la patada se hace más fuerte. ¿Qué regla de
        codificación de la \cref{prop:g8:nervous-communication:code} se
        acaba de enseñar?
  \item El médico de la clase explica que este reflejo se prueba en
        todas las revisiones. ¿Qué certifica sobre la médula, los
        \omterm{def:g5:brain-in-command:system}{nervios} y los empalmes un reflejo rotuliano presente y
        simétrico?
  \item Un voluntario que aprieta la mandíbula da una patada más fuerte
        --- una amplificación conocida. ¿Qué revela esto sobre las
        voces de las \omterm{def:g8:nervous-communication:neuron}{neuronas} de conexión llegando incluso a bucles
        ``sin \omterm{def:g5:brain-in-command:system}{cerebro}''?
  \item Un anestésico local en la \omterm{def:g1:my-body:parts}{pierna} suprime la patada y la
        sensación. Nombra el punto de ataque, según la
        \cref{rem:g8:nervous-communication:vulnerable}.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La redacción.}
\begin{enumerate}[resume]
  \item Dibuja el reflejo como el bucle de tres pasos de la
        \cref{prop:g5:brain-in-command:loop}, señalando dónde se lo
        salta este.
  \item Compara el reflejo y la parada en una tabla de dos columnas:
        camino, empalmes, tiempo, si se puede reprimir y papel de la
        práctica.
  \item La última sección de la redacción pregunta: ``¿por qué se
        saltan el \omterm{def:g5:brain-in-command:system}{cerebro} los bucles rápidos de protección?''. Responde
        con la lógica de la sartén caliente
        (\cref{exo:g5:brain-in-command:5}).
  \item Una frase para cerrar: qué enseñó el banco sobre el idioma, el
        cableado y los empalmes del \omterm{def:g5:brain-in-command:system}{sistema nervioso}, todo a la vez.
\end{enumerate}
\end{problem}
