\chapter{Dieren om ons heen}\label{ch:g1:animals-around-us}

Een merel huppelt over het gazon, een kat kijkt vanaf de muur toe, een
mier sleept een kruimel groter dan zichzelf over de drempel. \omterm{def:g1:animals-around-us:animal}{Dieren}
zijn overal om ons heen --- in de tuin, in het park, in huis. Laten we
leren naar ze te kijken zoals een bioloog dat doet: naar hun lichaam,
en naar de manier waarop ze bewegen.

\section{Wat is een dier?}

\begin{definition}[Dier]\label{def:g1:animals-around-us:animal}
Een \emph{dier}\index{dier} is een \omterm{def:g1:living-or-not:living}{levend wezen} dat zich voedt door
andere levende wezens te eten --- planten of andere dieren --- en dat
zich meestal zelf kan verplaatsen. Een kat, een merel, een vis en een
mier zijn allemaal dieren. Wij zijn ook dieren.
\end{definition}

\begin{example}[De dieren van één tuin]\label{ex:g1:animals-around-us:garden}
In één kleine tuin kunnen heel wat \omterm{def:g1:animals-around-us:animal}{dieren} wonen: een merel die een
worm uit de grond trekt, de worm zelf, een slak op een blad, een spin
in haar web, een bij in een bloem, en de kat van de buren. Groot of
piepklein, ze eten allemaal en ze bewegen allemaal.
\end{example}

\begin{remark}[Klein betekent niet onbelangrijk]\label{rem:g1:animals-around-us:small}
Een mier is net zo goed een \omterm{def:g1:animals-around-us:animal}{dier} als een olifant. De grootte van het
lichaam beslist niet wie een \omterm{def:g1:animals-around-us:animal}{dier} is: eten en bewegen doen dat. Een
paar van de wonderlijkste \omterm{def:g1:animals-around-us:animal}{dieren} om ons heen zijn juist de allerkleinste.
\end{remark}

\section{De delen van het lichaam van een dier}

\begin{definition}[Lichaamsdelen]\label{def:g1:animals-around-us:parts}
De meeste \omterm{def:g1:animals-around-us:animal}{dieren} hebben een \emph{kop}\index{kop} (met de bek en de
meeste zintuigen), een romp, en \emph{ledematen}\index{ledemaat} ---
poten, vleugels of vinnen --- waarmee ze bewegen. Veel \omterm{def:g1:animals-around-us:animal}{dieren} hebben
ook een \emph{staart}\index{staart}.
\end{definition}

\begin{example}[Tel de poten]\label{ex:g1:animals-around-us:legs}
Poten komen in aantallen die het tellen waard zijn. De merel loopt op
$2$ poten, de kat op $4$, het lieveheersbeestje op $6$, de spin op $8$
--- en de regenworm en de slak hebben helemaal geen poten, en toch
komen ze waar ze wezen willen.
\end{example}

\begin{definition}[Lichaamsbedekking]\label{def:g1:animals-around-us:covering}
Het lichaam van een \omterm{def:g1:animals-around-us:animal}{dier} wordt beschermd door een bedekking:
\emph{vacht}\index{vacht} (haren) bij de kat, \emph{veren}\index{veer}
bij de merel, \emph{schubben}\index{schub} bij de vis, een
\emph{schelp}\index{schelp} bij de slak, en een gladde natte
\emph{huid}\index{huid} bij de kikker.
\end{definition}

\begin{example}[Wie draagt wat]\label{ex:g1:animals-around-us:covering}
Aai een kat: \omterm{def:g1:animals-around-us:covering}{vacht}. Raap een gevallen merelveer op: licht en zacht.
Kijk goed naar een vis op de markt: rijen glanzende \omterm{def:g1:animals-around-us:covering}{schubben}, als
kleine dakpannen. Drie \omterm{def:g1:animals-around-us:animal}{dieren}, drie verschillende jassen.
\end{example}

\section{Hoe dieren bewegen}

\begin{proposition}[Manieren van bewegen]\label{prop:g1:animals-around-us:moving}
\omterm{def:g1:animals-around-us:animal}{Dieren} bewegen op manieren die bij hun \omterm{def:g1:animals-around-us:parts}{ledematen} passen:
\begin{itemize}
  \item \omterm{def:g1:animals-around-us:animal}{dieren} met poten \emph{lopen}, rennen, huppelen of kruipen ---
        de kat, de merel op het gazon, de mier;
  \item \omterm{def:g1:animals-around-us:animal}{dieren} met vleugels \emph{vliegen}\index{vliegen} --- de merel,
        de bij, de vlinder;
  \item \omterm{def:g1:animals-around-us:animal}{dieren} met vinnen \emph{zwemmen}\index{zwemmen} --- de vis;
  \item \omterm{def:g1:animals-around-us:animal}{dieren} zonder \omterm{def:g1:animals-around-us:parts}{ledematen} glijden of kronkelen --- de slak op
        zijn ene zachte voet, de regenworm die zich rekt en inkort.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Eén dier, meerdere manieren]\label{rem:g1:animals-around-us:several}
Veel \omterm{def:g1:animals-around-us:animal}{dieren} hebben meer dan één manier van bewegen. De merel loopt
\emph{en} vliegt; de kikker zwemt \emph{en} springt; de eend loopt,
zwemt \emph{en} vliegt. Kijk een tijdje naar een \omterm{def:g1:animals-around-us:animal}{dier} voordat je
beslist hoe het beweegt.
\end{remark}

\begin{method}[Een dier waarnemen]\label{met:g1:animals-around-us:observe}
Om een \omterm{def:g1:animals-around-us:animal}{dier} te beschrijven zoals een bioloog:
\begin{enumerate}
  \item kijk naar de delen: kop, romp, \omterm{def:g1:animals-around-us:parts}{ledematen}, \omterm{def:g1:animals-around-us:parts}{staart};
  \item tel de poten, zonder te gokken;
  \item noem de bedekking: \omterm{def:g1:animals-around-us:covering}{vacht}, \omterm{def:g1:animals-around-us:covering}{veren}, \omterm{def:g1:animals-around-us:covering}{schubben}, \omterm{def:g1:animals-around-us:covering}{schelp}, \omterm{def:g1:animals-around-us:covering}{huid};
  \item kijk hoe het beweegt: lopen, huppelen, vliegen, zwemmen, glijden;
  \item zeg waar je het gevonden hebt: gazon, boom, vijver, onder een steen.
\end{enumerate}
Doe het \omterm{def:g1:animals-around-us:animal}{dier} nooit pijn: kijk, en laat het daarna weer zijn gang gaan.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:1}
Noem de vier hoofddelen van het lichaam van de kat in de figuur. Welk
deel draagt de ogen en de bek?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:2}
Hoeveel poten: (a) een kip? (b) een hond? (c) een lieveheersbeestje?
(d) een spin? (e) een regenworm?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:3}
Geef de bedekking van elk \omterm{def:g1:animals-around-us:animal}{dier}: een schaap, een eend, een forel, een
slak.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:4}
Welke \omterm{def:g1:animals-around-us:animal}{dieren} uit dit hoofdstuk kunnen vliegen? Wat hebben ze allemaal,
in plaats van voorpoten of armen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:5}
De slak en de regenworm hebben geen poten. Hoe beweegt elk van beide
zich?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:6}
Een eend kan op drie manieren bewegen. Noem ze, en zeg welk deel van
haar lichaam ze voor elke manier gebruikt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:7}
Waarom is een mier een \omterm{def:g1:animals-around-us:animal}{dier}, volgens \cref{def:g1:animals-around-us:animal}?
Antwoord met de twee ideeën van de definitie.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:8}
Probeer het: kijk buiten een tijdje naar één \omterm{def:g1:animals-around-us:animal}{dier} --- een vogel, een
insect, een slak. Beschrijf het met de vijf stappen van
\cref{met:g1:animals-around-us:observe}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:9}
Een vriend zegt: ``Vissen kunnen geen \omterm{def:g1:animals-around-us:animal}{dieren} zijn --- ze lopen niet!''
Wat zou jij antwoorden? Gebruik \cref{prop:g1:animals-around-us:moving}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:10}
Raadseldier: het heeft $2$ poten, $2$ vleugels en \omterm{def:g1:animals-around-us:covering}{veren}, en het kan
niet vliegen, maar het rent heel hard. Is het een vogel? Zou het de
snelste renner van jouw buurt kunnen zijn? (Denk aan een struisvogel.)
\end{exercise}
