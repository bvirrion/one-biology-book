\begin{solution}{exo:b1:expression-control:1}

\end{solution}

\begin{solution}{exo:b1:expression-control:2}

से उससे जुड़े हैं: संचालक।
\end{solution}

\begin{solution}{exo:b1:expression-control:4}
प्रबलक: कोई नियामक अनुक्रम, अक्सर दूर का, जो कारकों के बँधने पर किसी \omterm{def:b1:genomes:gene}{जीन}
का अनुलेखन चढ़ा देता है। \omterm{def:b1:expression-control:enhancer}{अनुलेखन कारक}: कोई ऐसा \omterm{def:b1:proteins:peptide}{प्रोटीन} जिसमें एक
DNA-बंधनकारी प्रांत और एक सक्रियण या दमन प्रांत हो और जो
अनुलेखन का नियमन करे। गृहस्थी \omterm{def:b1:genomes:gene}{जीन}: वह जो हर
\omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में एक ठहरे स्तर पर अभिव्यक्त हो।
\end{solution}

\begin{solution}{exo:b1:expression-control:5}
$I^-$: संघटक। $O^c$: संघटक। $I^s$: अप्रेरणीय।
$I^-/I^+$: प्रेरणीय ($I^+$ \omterm{def:b1:expression-control:operon}{दमनकारी} ट्रांस में काम करता है)। $I^s/I^+$:
अप्रेरणीय ($I^s$ प्रभावी)। $I^+\,O^c\,Z^- / I^+\,O^+\,Z^+$: प्रेरणीय
--- एकमात्र काम करता $Z$ किसी सामान्य संचालक के पीछे है।
\end{solution}

\begin{solution}{exo:b1:expression-control:6}
लैक: शर्करा (प्रेरक) \omterm{def:b1:expression-control:operon}{दमनकारी} से बँधती है और उसे संचालक से
\emph{छुड़ा} देती है --- यानी \omterm{def:b1:enzymes:enzyme}{एंजाइम} तब बनते हैं जब उनका \omterm{def:b1:enzymes:enzyme}{क्रियाधार} मौजूद
हो। ट्रिप: \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} (सहदमनकारी) \omterm{def:b1:expression-control:operon}{दमनकारी} से बँधता है और उसे बँधने में
\emph{समर्थ} करता है --- यानी \omterm{def:b1:enzymes:enzyme}{एंजाइम} तब बनते हैं जब उनका उत्पाद
न हो। अपचयी पथ अपने निवेश से चालू होते हैं,
और उपचयी अपने निर्गम से बंद।
\end{solution}

\begin{solution}{exo:b1:expression-control:7}
ग्लूकोज पर: सामान्य वृद्धि (ग्लूकोज को कोई CAP नहीं चाहिए)। अकेले
लैक्टोस पर: लगभग कोई वृद्धि नहीं --- चक्रीय AMP के बिना CAP लैक
प्रवर्तक को सक्रिय नहीं कर सकता, इसलिए \omterm{def:b1:expression-control:operon}{दमनकारी} के हट जाने पर भी \omterm{def:b1:enzymes:enzyme}{एंजाइम}
प्रेरित स्तरों के कुछ ही प्रतिशत पर रह जाता है। दोनों पर: ग्लूकोज पर
वृद्धि, फिर ठहराव; कोई दूसरा चरण नहीं। हर प्रसंग में \omterm{def:b1:enzymes:enzyme}{एंजाइम} नीचा।
\end{solution}

\begin{solution}{exo:b1:expression-control:8}
अवमंदन को यह चाहिए कि \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} संदेशवाहक का अनुवादन तब कर रहा हो जब
पॉलिमरेज़ अभी उसका अनुलेखन कर ही रही हो, ताकि \omterm{def:b1:gene-expression:ribosome}{राइबोसोम} की
स्थिति पॉलिमरेज़ के पीछे के \omterm{def:b1:nucleic-acids:chain}{RNA} को आकार दे। \omterm{def:b1:cell-unit-of-life:prokeuk}{सुकेंद्रकियों} में
अनुलेखन केंद्रक में होता है और अनुवादन \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाद्रव्य} में, और
संदेश का अनुवादन तब तक नहीं होता जब तक उसका प्रसंस्करण तथा निर्यात न हो जाए।
\end{solution}

\begin{solution}{exo:b1:expression-control:9}
प्रबलक दूर से, किसी भी दिशा में और किसी भी ओर काम करते हैं, इसलिए वे
पढ़े जाकर या पॉलिमरेज़ को सीधे बिठाकर काम नहीं कर सकते; वे ऐसे कारक
बाँधकर काम करते हैं जो \omterm{def:b1:nucleic-acids:chain}{DNA} के किसी फंदे से होकर प्रवर्तक को छूते हैं,
और उसके लिए दूरी तथा दिशा कम ही मायने रखती हैं।
\end{solution}

\begin{solution}{exo:b1:expression-control:10}
$5000\times 4096\times 4 = \num{8.2e7}$ \omterm{def:b1:water-small-molecules:atp}{ATP}: यानी बजट का \qty{0.8}{\%}।
वृद्धि दर \qty{0.8}{\%} नीची: प्रति पीढ़ी उत्परिवर्ती का
हिस्सा गुणक $2^{-0.008} = 0.9945$ से गिरता है। $1:1$ से
$1:99$ तक: $0.9945^n = 1/99$, $n = \ln 99/0.00555 = 830$ पीढ़ियाँ ---
यानी सतत संवर्धन के कुछ हफ़्ते।
\end{solution}

\begin{solution}{exo:b1:expression-control:11}
हर कोशिका-प्ररूप \omterm{def:b1:expression-control:enhancer}{अनुलेखन कारकों} का एक विशिष्ट समुच्चय ढोता है, जो
अपने \omterm{def:b1:genomes:gene}{जीनों} के प्रबलक बाँधते हैं और ऐसी ऐसिटिलेज़ बुलाते हैं जो
उनका क्रोमैटिन खोल देती हैं, जबकि दूसरे प्ररूपों के \omterm{def:b1:genomes:gene}{जीन} मेथिलित होकर
ऐसे विषमक्रोमैटिन में बँध जाते हैं जहाँ कोई कारक पहुँच नहीं सकता। कारक
एक-दूसरे की अभिव्यक्ति बनाए रखते हैं (स्थायी अवस्थाओं वाला एक जाल), और
क्रोमैटिन के चिह्न प्रतिकृतिकरण पर नकल हो जाते हैं, इसलिए संतति दोनों
कारक और खुले तथा बंद क्षेत्र दोनों वंशागत पाती है। \omterm{def:b1:genomes:genome}{जीनोम} वही है;
पहुँच में आने वाला \omterm{def:b1:genomes:genome}{जीनोम} नहीं, और वह अपहुँच
वंशागत होती है।
\end{solution}

\begin{solution}{exo:b1:expression-control:12}
किसी बैक्टीरियम के नियंत्रण माध्यम के हिसाब से बैठाए गए हैं: कोई शर्करा,
कोई \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल}, कोई तापमान किसी \omterm{def:b1:expression-control:operon}{दमनकारी} या किसी \omterm{prop:b1:expression-control:trp}{सिग्मा कारक} से बँधता है
और अनुक्रिया मिनटों में पूरी हो जाती है तथा संकेत के हटते ही उलट जाती है
--- अभिव्यक्ति पर्यावरण का पीछा करती है। किसी शरीर की \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} भी संकेतों
का जवाब देती है, पर उसके मुख्य निर्णय परिवर्धन के दौरान लिए गए और
क्रोमैटिन में जकड़ दिए गए: वह क्या अभिव्यक्त कर सकती है यह उन कारकों ने
तय किया जो उसे वंशागत मिले और उसके \omterm{def:b1:nucleic-acids:chain}{DNA} पर पड़े चिह्नों ने, और वह केवल
विभाजन से बदलता है तथा अधिकतर अनुत्क्रमणीय है। बैक्टीरियम वर्तमान का
पाठक है; और विभेदित \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} की एक स्मृति है जो उसकी पहचान है।
\end{solution}

\begin{solution}{pb:b1:expression-control:1}
\textbf{1.} $0.5\times 10^{-3}\times 180 = \qty{0.09}{g/L} =
\qty{9e-5}{g/mL}$।
\textbf{2.} $9\times 10^{-5}/1.5\times 10^{-12} = \num{6e7}$ विभाजन।
\textbf{3.} प्रति mL $10^7 + \num{6e7} = \num{7e7}$ \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ}।
\textbf{4.} $7 = 2^n$: $n = 2.8$ द्विगुणन, यानी \qty{56}{min}।
\textbf{5.} बंद: लैक्टोस (ऐलोलैक्टोस) ने \omterm{def:b1:expression-control:operon}{दमनकारी} छुड़ा दिया है,
पर ग्लूकोज चक्रीय AMP नीचा रखता है, इसलिए CAP बँधा नहीं है और प्रवर्तक
लगभग मौन है; और पारगमेज़ भी कम है, इसलिए लैक्टोस कम ही भीतर घुसता है।
\textbf{6.} चक्रीय AMP चढ़ता है और CAP से बँधता है, जो प्रवर्तक से बँध
जाता है।
\textbf{7.} ऋणात्मक नियंत्रण हट गया था पर धनात्मक नियंत्रण
अनुपस्थित था: CAP के बिना कोई कमज़ोर प्रवर्तक कम ही शुरू करता है
(\omterm{def:b1:expression-control:cap}{अपचयज दमन})।
\textbf{8.} प्रति सेकंड $5000/1800 = 2.8$ चतुष्क; $2.8\times 4096 =
\num{11400}$ अवशेष प्रति सेकंड।
\textbf{9.} क्षमता प्रति सेकंड $20\,000\times 20 = \num{4e5}$ अवशेष:
यानी \omterm{def:b1:gene-expression:ribosome}{राइबोसोमों} का \qty{3}{\%}।
\textbf{10.} $5000\times 4096\times 110\times 1.66\times 10^{-24} =
\qty{3.7e-15}{g}$: यानी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के \omterm{def:b1:proteins:peptide}{प्रोटीन} का \qty{2.5}{\%}।
\textbf{11.} $5000/5 = 1000$।
\textbf{12.} पारगमेज़ के बिना लैक्टोस झिल्ली पार नहीं कर सकता;
किसी \emph{lacY}$^-$ उत्परिवर्ती के पास \omterm{def:b1:enzymes:enzyme}{एंजाइम} तो है पर भीतर कोई
\omterm{def:b1:enzymes:enzyme}{क्रियाधार} नहीं और वह लैक्टोस पर नहीं बढ़ता (न ठीक से प्रेरित होता है,
क्योंकि प्रेरक \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के भीतर लैक्टोस से बनता है)।
\textbf{13.} $1.0\times 10^{-3}\times 342 = \qty{0.342}{g/L}$, जो प्रति
मोल $2\times 180 = \qty{360}{g}$ ग्लूकोज के बराबर है, यानी \qty{0.36}{g/L}
$= \qty{3.6e-4}{g/mL}$।
\textbf{14.} $3.6\times 10^{-4}/1.5\times 10^{-12} = \num{2.4e8}$
विभाजन; अंतिम प्रति mL $\num{7e7} + \num{2.4e8} = \num{3.1e8}$।
\textbf{15.} $3.1/0.7 = 4.4 = 2^n$: $n = 2.1$ द्विगुणन, यानी \qty{64}{min}।
\textbf{16.} ग्लूकोज \qty{56}{min}, पिछड़ाव \qty{30}{min}, लैक्टोस
\qty{64}{min}: कुल मिलाकर \qty{150}{min}।
\textbf{17.} लैक्टोस को पहले जल-अपघटित होना पड़ता है, जो एक अतिरिक्त
एंजाइमी चरण है और जिसकी दर ग्लूकोज की आपूर्ति सीमित करती है; और
पारगमेज़ उसे भीतर लाने के लिए प्रोटॉन-प्रवणता बरतती है, जो कीमत
ग्लूकोज का परिवहन नहीं चुकाता।
\textbf{18.} $I^-$: \omterm{def:b1:enzymes:enzyme}{एंजाइम} भर \num{5000} पर (संघटक, एक बार
ग्लूकोज चुक जाने पर; ग्लूकोज पर CAP उसे तब भी सीमित करता है, इसलिए स्तर
मध्यवर्ती रहता है); उसका पिछड़ाव छोटा या अनुपस्थित रहता है, इसलिए वक्र
छोटा ठहराव दिखाता है; और लागत उसे ग्लूकोज पर ज़रा धीमा कर देती है।
\textbf{19.} CAP$^-$: ग्लूकोज पर $\num{7e7}$ तक बढ़ता है, फिर रुक जाता
है: लैक \omterm{def:b1:expression-control:operon}{ऑपेरॉन} सक्रिय ही नहीं हो सकता, \omterm{def:b1:enzymes:enzyme}{एंजाइम} 5 अणुओं के पास रह जाता है,
और लैक्टोस कभी बरता नहीं जाता।
\textbf{20.} केवल लैक्टोस पर: एक बार प्रेरित हो जाने पर एंजाइम-स्तर में
कोई अंतर नहीं (दोनों पूरी तरह चालू) और वृद्धि में भी लगभग कोई नहीं;
$O^c$ में बस प्रेरण का विलंब नहीं होता। केवल ग्लूकोज पर: $O^c$ \omterm{def:b1:enzymes:enzyme}{एंजाइम}
बेकार बनाता है (जितना CAP होने दे), वन्य प्ररूप नहीं बनाता।
\textbf{21.} $5000\times 4096\times 4 = \num{8.2e7}$ \omterm{def:b1:water-small-molecules:atp}{ATP}: यानी
$10^{10}$ का \qty{0.8}{\%}।
\textbf{22.} पीढ़ी-काल \qty{0.8}{\%} लंबा: यानी वृद्धि दर
\qty{0.8}{\%} नीची, प्रति पीढ़ी $\delta = 0.008$ द्विगुणन।
\textbf{23.} $r = 2^{-0.008} = 0.99447$; $r^n = 0.111$: $n =
\ln 0.111/\ln 0.99447 = 2.20/0.00555 = 400$ पीढ़ियाँ।
\textbf{24.} लैक्टोस पर \omterm{def:b1:enzymes:enzyme}{एंजाइम} वैसे भी चाहिए, इसलिए उत्परिवर्ती कुछ
नहीं चुकाता और पिछड़ाव की बचत कमा लेता है: उसके विरुद्ध वरण नहीं होता और
कभी-कभी उसके पक्ष में होता है; और प्रयोगशाला के प्रभेद सुविधा के लिए
उगाए जाते हैं, होड़ के लिए नहीं।
\textbf{25.} प्रेरण गुणक 1000 (5 से \num{5000} अणु);
अनावश्यक संश्लेषण ATP-बजट का \qty{0.8}{\%} लेता है; और यह
प्रयोग लगभग ढाई घंटे चलता है।
\end{solution}
