\section*{Bab \ref{ch:g8:from-egg-to-birth} --- Dari Sel Telur sampai Kelahiran}

\begin{solution}{exo:g8:from-egg-to-birth:1}
Pembelahannya dalam sehari sesudah \omterm{def:g5:flower-to-fruit:steps}{pembuahan}; sebuah bola
murbei berisi puluhan \omterm{def:g5:first-look-at-cells:cell}{sel} pada hari keempat, hanyut menuruni
\omterm{def:g8:reproductive-systems:female}{saluran telurnya}; penanamannya di dalam lapisan dalam \omterm{prop:g5:human-reproduction:plan}{rahim}
sekitar hari ketujuh.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:2}
Dengan bekal \omterm{def:g4:animal-reproduction:sexual}{sel telurnya} sendiri --- lumbung yang menjadi
isi \omterm{def:g8:sexual-reproduction:gametes}{gamet} besarnya. Penanaman mengakhiri pengaturan itu:
penghuninya menyambungkan diri ke pasokan lapisan dalamnya,
yang segera ditata sebagai \omterm{def:g8:from-egg-to-birth:placenta}{plasenta}.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:3}
Cakram penyangga hidup yang tumbuh ke dalam dinding \omterm{prop:g5:human-reproduction:plan}{rahim},
tempat kedua \omterm{prop:g4:journey-of-food:absorb}{darahnya} mendekat tanpa bercampur. Ke \omterm{def:g5:human-reproduction:pregnancy}{embrionya}:
\omterm{def:g7:digestion-and-nutrients:families}{zat gizi} dan \omterm{def:g7:what-is-respiration:respiration}{oksigen}. Kembali: buangan dan \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}.
Yang tidak pernah menyeberang: kedua \omterm{prop:g4:journey-of-food:absorb}{darahnya} sendiri.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:4}
Pekan 1--8, \omterm{def:g5:human-reproduction:pregnancy}{embrionya}: rancangan tubuhnya digelar oleh \omterm{def:g5:first-look-at-cells:cell}{sel}
yang membelah, berpindah dan mengkhusus. Pekan 9--40,
\omterm{def:g5:human-reproduction:pregnancy}{janinnya}: rancangannya sudah terbangun --- tumbuh dan matang,
\omterm{def:g5:brain-in-command:system}{otaknya} merangkai diri, \omterm{def:g7:how-animals-breathe:four}{paru-parunya} berlatih.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:5}
Karena seluruh pembangunannya berjalan dengan pengantaran:
begitu tata rumah tangga \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya} mulai, sebuah pompa
harus menggerakkan \omterm{prop:g4:journey-of-food:absorb}{darah} \omterm{def:g5:human-reproduction:pregnancy}{embrionya} menembus \omterm{prop:g5:human-reproduction:cord}{tali pusar} dan
tubuhnya --- jalur pasokannya tak dapat menunggu sisa
rancangannya.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:6}
Kontraksi persalinannya yang menguat; \omterm{def:g2:born-grow-die:cycle}{kelahirannya}, biasanya
kepala lebih dahulu; pergantian tugasnya --- tangis pertama,
\omterm{prop:g5:human-reproduction:cord}{tali pusar} dipotong, perbatasan sang bayi sendiri mengambil
alih; ari-arinya --- \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya} terlepas lalu menyusul.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:7}
Tugas \omterm{def:g7:how-animals-breathe:four}{paru-parunya} (\omterm{def:g7:what-is-respiration:respiration}{oksigen} masuk, \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} keluar
--- \cref{ch:g7:lungs-and-gas-exchange}), tugas ususnya (\omterm{def:g7:digestion-and-nutrients:families}{zat
gizi} masuk --- \cref{ch:g7:digestion-and-nutrients}), dan
tugas \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} (buangan keluar ---
\cref{ch:g7:removing-wastes}). Tiga perbatasan, satu cakram.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:8}
Karena pekan penggelaran rancangannyalah saat satu petunjuk
yang terganggu bergaung menembus seluruh pembangunannya ---
tiap organ masih segerombol kecil \omterm{def:g5:first-look-at-cells:cell}{sel} yang sedang mengkhusus;
dan karena pekan itu dapat berlalu bahkan sebelum
\omterm{def:g5:human-reproduction:pregnancy}{kehamilannya} dicurigai --- sehingga perawatannya harus mulai
pada kemungkinan pertama, bukan pada pemindaian pertama.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:9}
\omterm{def:g7:what-is-respiration:respiration}{Pernapasan} dan tangisnya: perbatasan \omterm{def:g7:how-animals-breathe:four}{paru-parunya}, baru
berumur beberapa menit. \omterm{def:g4:heart-and-blood:heart}{Denyut jantungnya}: pompa yang
memutar kedua litarnya. Warnanya: jasa pengantarannya
sungguh mengantar --- \omterm{prop:g4:journey-of-food:absorb}{darah} kaya \omterm{def:g7:what-is-respiration:respiration}{oksigen} sampai ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler}
\omterm{def:g1:the-five-senses:sense}{kulitnya}. Geraknya melengkapi pemeriksaan sang panglima.
Nilainya adalah gelung itu, diaudit secara langsung.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:10}
Pada pembelahan pertamanya --- bola \omterm{prop:g8:sexual-reproduction:core}{zigotnya} terbelah
menjadi dua bola dalam beberapa hari pertamanya. Kedua
kembarnya lalu membangun dari warisan \omterm{def:g5:first-look-at-cells:cell}{sel} awal yang sama
(\cref{ex:g5:first-look-at-cells:one}), sehingga kedua
pembangunannya menjalankan rancangan yang sama: salinan,
karena percabangannya datang sesudah kesepakatannya.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:11}
\omterm{def:g5:first-look-at-cells:cell}{Sel}: tahap \omterm{def:g5:human-reproduction:pregnancy}{janinnya} --- organnya sudah terbangun, kini
tumbuh; \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g3:bones-and-muscles:muscle}{otot} dan \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g5:brain-in-command:system}{sarafnya} saling merangkai. Pasokan:
tata rumah tangga \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya} berjalan penuh --- \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} dan
\omterm{def:g7:what-is-respiration:respiration}{oksigen} masuk, buangan keluar, tidak ada yang berbahaya boleh
ada di dalam \omterm{prop:g4:journey-of-food:absorb}{darah} ibunya. \omterm{def:g8:puberty:hormones}{Hormon}: menjaga \omterm{prop:g5:human-reproduction:plan}{rahimnya} tetap
tenang, daurnya tertunda. Perawatan: piring dan istirahat
ibunya membiayai lalu lintas senilai dua perbatasan;
tendangannya sendiri adalah tanda yang menenangkan bahwa
gelungnya bekerja.
\end{solution}

\begin{solution}{exo:g8:from-egg-to-birth:12}
\omterm{def:g8:from-egg-to-birth:placenta}{Plasentanya} menangani pemuatan \omterm{def:g7:what-is-respiration:respiration}{oksigen}, pembuangan \omterm{def:g7:what-is-respiration:respiration}{karbon
dioksida}, pasokan \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} dan pembersihan buangannya. Pada
tangis pertamanya \omterm{def:g7:how-animals-breathe:four}{paru-parunya} mengambil kedua gasnya;
ususnya (sejak suapan pertamanya) mengambil pasokannya;
\omterm{def:g7:removing-wastes:kidneys}{ginjalnya} mengambil pembersihannya --- dan semuanya dalam
beberapa menit sesudah \omterm{prop:g5:human-reproduction:cord}{tali pusarnya} dipotong, sebuah serah
terima perbatasan yang harus dilalui setiap organ pada
percobaan pertamanya, dan justru itulah yang diawasi daftar
periksa ruang bersalinnya.
\end{solution}

\begin{solution}{pb:g8:from-egg-to-birth:1}
\textbf{1.} \omterm{def:g8:puberty:hormones}{Hormon} \omterm{def:g5:human-reproduction:pregnancy}{kehamilan} yang mulai bersama penanamannya
--- isyarat kimiawi sang penghuni yang sudah menetap; sejak
kira-kira hari ketujuh, ia ada di sana untuk dibaca.

\textbf{2.} Tahap \omterm{def:g5:human-reproduction:pregnancy}{embrionya} ditutup: pekan 1--8 menggelar
rancangan yang lengkap --- setiap organ sudah dimulai,
sebesar sebutir kacang. Tahap \omterm{def:g5:human-reproduction:pregnancy}{janinnya} --- membangun dan
mematangkan --- dimulai.

\textbf{3.} Alkohol dan asapnya menyeberang ke penumpangnya,
dan pekan penggelaran rancangannya paling rawan terhadap
petunjuk yang terganggu; asam folat adalah bahan yang
ditarik oleh pembangunan paling awalnya --- dan karena pekan
itu dapat berlalu tanpa disadari, ``sejak sebelum
permulaannya'' pada vitaminnya adalah hitungan yang sama
dengan pantangannya.

\textbf{4.} Tata rumah tangga \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya} membutuhkan
pompanya sejak awal: pengantarannya tak dapat beredar tanpa
\omterm{def:g4:heart-and-blood:heart}{jantung} yang berdenyut, sehingga itulah mesin pertama yang
dinyalakan --- dan kerlip pada pemindaian pekan 12 adalah
mesin itu, delapan pekan sesudah ia mulai berjalan.

\textbf{5.} Tahap \omterm{def:g5:human-reproduction:pregnancy}{janinnya}: tumbuh dan matangnya rancangan
yang sudah digelar. \omterm{def:g8:from-egg-to-birth:placenta}{Plasentanya} diperiksa sebagai \omterm{def:g7:how-animals-breathe:four}{paru-paru}
(pertukaran gas), usus (pasokan) dan \omterm{def:g7:removing-wastes:kidneys}{ginjal} (pembersihan) ---
ketiga perbatasan penumpangnya di dalam satu cakram, beserta
kedudukannya yang bebas dari pintu keluarnya.

\textbf{6.} Gelung melapor-memutuskan-memerintah: \omterm{def:g3:bones-and-muscles:muscle}{ototnya}
\omterm{def:g3:bones-and-muscles:muscle}{berkontraksi} atas perintah \omterm{def:g5:brain-in-command:system}{sarafnya} --- kabel sang panglima
dan motornya, yang mengumumkan dirinya menembus dinding
\omterm{prop:g5:human-reproduction:plan}{rahimnya}.

\textbf{7.} Karena ia melayani perbatasan dua pelanggan:
perbatasan tubuhnya sendiri, dan --- lewat lalu lintas
\omterm{def:g8:from-egg-to-birth:placenta}{plasentanya} --- perbatasan penumpangnya. Gula dan tekanannya
membaca bagaimana beban ganda itu sedang dipikul.

\textbf{8.} Gerakan bernapas tanpa udara --- puputannya
berlatih dalam keadaan kosong, \omterm{def:g3:bones-and-muscles:muscle}{otot} dan gerak refleksnya
bersiap untuk satu saat tangis pertamanya, ketika organnya
harus bekerja pada percobaan pertamanya.

\textbf{9.} \omterm{prop:g5:human-reproduction:plan}{Rahimnya} --- \omterm{def:g3:bones-and-muscles:muscle}{otot} terkuat di dalam tubuh;
tugasnya: membuka jalannya, dan mendorong bayinya menuruni
jalan lahirnya; perintahnya datang sebagai \omterm{def:g8:puberty:hormones}{hormon}, kimia
bayinya dan ibunya dalam percakapan.

\textbf{10.} Tangisnya: \omterm{def:g7:how-animals-breathe:four}{paru-parunya} terisi dan bekerja ---
perbatasan pertamanya dialihkan. \omterm{prop:g5:human-reproduction:cord}{Tali pusarnya} dijepit lalu
dipotong: lalu lintas \omterm{def:g8:from-egg-to-birth:placenta}{plasentanya} berakhir. \omterm{def:g8:from-egg-to-birth:placenta}{Plasentanya}
lengkap: organ sekali pakai itu terlepas seutuhnya ---
pergantian tugasnya ditutup tanpa ada yang tertinggal.

\textbf{11.} Bahwa gelung pasokan-dan-pembuangannya ---
\omterm{def:g7:how-animals-breathe:four}{paru-paru}, pompa, pengantaran, panglima --- berjalan mandiri:
audit pertama atas mesin yang dipetakan buku ini selama dua
tahun ajaran.

\textbf{12.} ``Empat puluh pekan membangun tiga ribu miliar
\omterm{def:g5:first-look-at-cells:cell}{sel} yang mengkhusus dari satu \omterm{def:g4:animal-reproduction:sexual}{sel telur} yang dibuahi --- oleh
pembelahan, pengkhususan dan tata usaha yang dijalankan
\omterm{def:g8:puberty:hormones}{hormon}, dipasok melintasi satu cakram yang menakjubkan.''
\end{solution}
