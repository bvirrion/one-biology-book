\chapter{कोशिकाओं पर पहली नज़र}\label{ch:g5:first-look-at-cells}

अपने हाथ की पीठ देखो: चिकनी \omterm{def:g1:the-five-senses:sense}{त्वचा}, एक ही अटूट सतह। अब ऐसा आवर्धक शीशा
सोचो जो हज़ार गुना पास ले आए। वह अटूट सतह घुलकर नन्हे-नन्हे कक्षों की
एक फ़र्शबंदी बन जाएगी, फ़र्श के पत्थरों की तरह आपस में जुड़े हुए ---
और इस पुस्तक में तुम्हें मिला हर \omterm{def:g1:living-or-not:living}{सजीव}, \omterm{prop:g4:plants-without-seeds:damp}{मॉस} से लेकर ह्वेल तक, इसी तरह
जड़ा हुआ है। \omterm{def:g1:living-or-not:living}{जीवविज्ञान} का सबसे बड़ा राज़ छोटा है।

\section{सजीव कोशिकाओं से बने हैं}

\begin{definition}[कोशिका]\label{def:g5:first-look-at-cells:cell}
\emph{कोशिका}\index{कोशिका} किसी \omterm{def:g1:living-or-not:living}{सजीव} की सबसे छोटी जीवित निर्माण-ईंट
है। लगभग सारी कोशिकाएँ नंगी \omterm{def:g1:the-five-senses:sense}{आँख} से देखने के लिए बहुत ही छोटी हैं; शरीर
उनसे वैसे ही बना है जैसे दीवार ईंटों से। तुम्हारा अपना शरीर दसियों
हज़ार अरब कोशिकाओं से बना है।
\end{definition}

\begin{proposition}[कोशिका का सिद्धांत]\label{prop:g5:first-look-at-cells:principle}
हर \omterm{def:g1:living-or-not:living}{सजीव} \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} से बना है --- हर \omterm{def:g1:animals-around-us:animal}{जंतु}, हर \omterm{def:g1:plants-around-us:plant}{पौधा}, हर कुकुरमुत्ता, और
वे \omterm{def:g1:living-or-not:living}{सजीव} भी जो सिर्फ़ \emph{एक} \omterm{def:g5:first-look-at-cells:cell}{कोशिका} के हैं। यह साझी बनावट
\emph{जीवन की एकता}\index{जीवन की एकता} के बड़े लक्षणों में से एक है:
गुलबहार और ह्वेल चाहे कितने ही अलग हों, पास से देखो तो दोनों उसी तरह की
छोटी जीवित इकाइयों से बने हैं।
\end{proposition}
\admitted

\begin{example}[छोटा कितना छोटा?]\label{ex:g5:first-look-at-cells:small}
तुम्हारे शरीर की एक आम \omterm{def:g5:first-look-at-cells:cell}{कोशिका} मिलीमीटर के लगभग पचासवें हिस्से जितनी
होती है। उनमें से $50$ को क़तार में लगाओ तो वे एक मिलीमीटर घेरती हैं
--- पूरी क़तार पेंसिल की एक बिंदी के नीचे छिप जाए। इसीलिए मानव इतिहास
के ज़्यादातर हिस्से में किसी को \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} का शक तक नहीं हुआ: जीवन की
निर्माण-ईंटें \omterm{def:g1:the-five-senses:sense}{आँख} के सबसे छोटे दिखने वाले कण से भी छोटी हैं।
\end{example}

\section{वह उपकरण जिसने उन्हें दिखाया}

\begin{definition}[सूक्ष्मदर्शी]\label{def:g5:first-look-at-cells:microscope}
\emph{सूक्ष्मदर्शी}\index{सूक्ष्मदर्शी} वह उपकरण है जिसके लेंस बहुत
छोटी चीज़ों को सैकड़ों गुना बड़ा कर देते हैं। जीवित पदार्थ की पतली
क़तरनों पर लगाया गया, उसने वह दिखाया जो किसी \omterm{def:g1:the-five-senses:sense}{आँख} ने कभी नहीं देखा था:
\omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ}। यह नाम ख़ुद एक शुरुआती प्रेक्षक ने गढ़ा था, जिसे कॉर्क की
क़तरन के नन्हे कक्ष भिक्षुओं की छोटी कोठरियों जैसे लगे थे ---
\emph{\omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ}}।
\end{definition}

पर रखी एक पतली क़तरन।}

\begin{example}[लेंस के नीचे दो मशहूर नज़ारे]\label{ex:g5:first-look-at-cells:classics}
दो प्रेक्षण, जिन्हें स्कूल का कोई भी \omterm{def:g5:first-look-at-cells:microscope}{सूक्ष्मदर्शी} दोहरा सकता है: प्याज़
के भीतर की पतली झिल्ली का एक टुकड़ा ईंट जैसी \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} की सुथरी क़तारें
दिखाता है, दीवार से सटी दीवार; और गाल के भीतर से हलके से खुरचा गया नमूना
बिखरी हुई, ज़्यादा गोल \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} दिखाता है --- तुम्हारी अपनी। \omterm{def:g1:plants-around-us:plant}{पौधा} और
\omterm{def:g1:animals-around-us:animal}{जंतु}, अगल-बगल, दोनों \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} के बने।
\end{example}

हर कक्ष --- अपने छोटे गहरे धब्बे के साथ --- एक \omterm{def:g5:first-look-at-cells:cell}{कोशिका} है।}

\begin{remark}[वह गहरा धब्बा]\label{rem:g5:first-look-at-cells:nucleus}
दोनों नज़ारों में ज़्यादातर \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} के भीतर एक छोटा गहरा धब्बा दिखता
है: \omterm{def:g5:first-look-at-cells:cell}{कोशिका} का कमान-केंद्र, जिसे उसका \emph{केंद्रक}\index{केंद्रक}
कहते हैं। केंद्रक जिससे कमान चलाता है --- उसके भीतर लिपटा निर्देशों का
वह मशहूर धागा --- वह इस पुस्तक के आगे के सालों की कहानी है; फ़िलहाल बस
इतना ध्यान रखो कि ये छोटी कोठरियाँ ख़ाली नहीं हैं।
\end{remark}

\section{एक कोशिका से पूरे शरीर तक}

\begin{proposition}[शरीर अपनी कोशिकाओं से बढ़ते हैं]\label{prop:g5:first-look-at-cells:growth}
शरीर बढ़ता है और अपनी मरम्मत करता है, क्योंकि उसकी \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ}
\emph{विभाजित} हो सकती हैं: एक \omterm{def:g5:first-look-at-cells:cell}{कोशिका} दो जीवित \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} में बँट जाती
है, दो चार में, और आगे इसी तरह। लंबा होना
(\cref{prop:g3:stages-of-life:animals}) भीतर से देखो तो \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} का
बढ़ना है; और छिलन का भरना
(\cref{rem:g3:bones-and-muscles:alive}) \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} का फिर से बनाना। और
हर शरीर, सबसे बड़ा भी, एक \emph{अकेली \omterm{def:g5:first-look-at-cells:cell}{कोशिका}} से शुरू हुआ था ---
\cref{def:g4:animal-reproduction:sexual} का निषेचित \omterm{def:g4:animal-reproduction:sexual}{अंडाणु}।
\end{proposition}
\admitted

\begin{example}[एक से खरबों तक]\label{ex:g5:first-look-at-cells:one}
निषेचित \omterm{def:g2:animals-grow-up:born}{अंडा} बँटकर $2$ \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} बनता है, फिर $4$, $8$,
$16$ \dots{} बार-बार होते इस विभाजन ने महीनों और सालों में वे दसियों
हज़ार अरब \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} बना दीं जो यह वाक्य पढ़ रही हैं। तुम्हारी हर \omterm{def:g5:first-look-at-cells:cell}{कोशिका}
उसी पहली \omterm{def:g5:first-look-at-cells:cell}{कोशिका} की वंशज है।
\end{example}

\begin{remark}[एक ही कोशिका में पूरा जीवन]\label{rem:g5:first-look-at-cells:onecelled}
कुछ \omterm{def:g1:living-or-not:living}{सजीव} कभी एक \omterm{def:g5:first-look-at-cells:cell}{कोशिका} से आगे नहीं जाते --- और उन्हें इससे ज़्यादा कुछ
चाहिए भी नहीं। तालाब के पानी की एक बूँद उनसे भरी रहती है: अकेली
\omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ}, जो तैरती हैं, पोषण लेती हैं और बँटती हैं, यानी \omterm{def:g5:first-look-at-cells:microscope}{सूक्ष्मदर्शी}
के पैमाने पर पूरे-पूरे जीवन। \cref{def:g1:healthy-habits:germs} के
\omterm{def:g1:healthy-habits:germs}{रोगाणु} ज़्यादातर ऐसे ही एककोशिकीय \omterm{def:g1:living-or-not:living}{सजीव} हैं --- पुराने परिचित, जिनकी
अब व्याख्या भी है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:1}
\omterm{def:g5:first-look-at-cells:cell}{कोशिका} क्या है? तुम्हारा शरीर मोटे तौर पर कितनी \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} से बना है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:2}
इतिहास के ज़्यादातर हिस्से में किसी को \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} का पता क्यों नहीं था?
किस उपकरण ने यह बदला?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:3}
बताओ कि \omterm{def:g5:first-look-at-cells:microscope}{सूक्ष्मदर्शी} प्याज़ की झिल्ली में क्या दिखाता है, और गाल की
खुरचन में क्या। दोनों नज़ारों में क्या बात एक जैसी है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:4}
ज़्यादातर \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} में दिखने वाले छोटे गहरे धब्बे को क्या कहते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:5}
शरीर की लगभग कितनी \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} क़तार में लगकर एक मिलीमीटर घेरती हैं ---
और एक \omterm{def:g5:first-look-at-cells:cell}{कोशिका} मिलीमीटर में कितनी लंबी है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:6}
जब कोई बच्चा लंबा होता है, तब भीतर \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} क्या कर रही होती हैं? और
जब छिला हुआ \omterm{def:g1:my-body:joint}{घुटना} भरता है, तब?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:7}

\cref{def:g4:animal-reproduction:sexual} के अनुसार वह पहली \omterm{def:g5:first-look-at-cells:cell}{कोशिका} क्या
थी?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:first-look-at-cells:8}
