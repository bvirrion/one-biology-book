\chapter{Mengklasifikasikan Hewan}\label{ch:g4:classifying-animals}

Sebuah museum sejarah alam menyimpan jutaan \omterm{def:g1:animals-around-us:animal}{hewan} --- laci berisi
kumbang, aula berisi \omterm{def:g3:bones-and-muscles:skeleton}{rangka}, toples berisi \omterm{prop:g3:sorting-living-things:groups}{ikan}. Namun seorang kurator
dapat berjalan menuju salah satunya dalam hitungan menit. Rahasianya
bukan ingatan melainkan \emph{keteraturan}: setiap \omterm{def:g1:animals-around-us:animal}{hewan} punya
tempatnya dalam sebuah sistem besar berupa kelompok di dalam kelompok,
yang dibangun dari aturan \omterm{def:g3:sorting-living-things:criterion}{pengelompokan}
\cref{ch:g3:sorting-living-things} --- dan tahun ini kita mempelajari
bagaimana sistem itu bekerja.

\section{Kelompok di dalam kelompok}

\begin{definition}[Klasifikasi]\label{def:g4:classifying-animals:classification}
\emph{Mengklasifikasikan}\index{mengklasifikasikan} \omterm{def:g1:living-or-not:living}{makhluk hidup}
berarti mengelompokkannya menurut ciri tubuh yang sama ke dalam
kelompok, lalu mengelompokkan kelompok itu ke dalam kelompok yang
lebih besar --- \emph{kelompok yang bersarang di dalam
kelompok}\index{kelompok bersarang}, seperti kotak di dalam kotak.
Seluruh susunan itu disebut sebuah
\emph{klasifikasi}\index{klasifikasi}.
\end{definition}

\begin{example}[Alamat-alamat sang kucing]\label{ex:g4:classifying-animals:cat}
Satu kucing, beberapa alamat yang bersarang: kucingnya adalah seekor
\emph{\omterm{prop:g3:sorting-living-things:groups}{mamalia}} (berbulu, memberi \omterm{def:g2:animals-grow-up:born}{susu} kepada anaknya); setiap \omterm{prop:g3:sorting-living-things:groups}{mamalia}
adalah seekor \emph{\omterm{def:g4:classifying-animals:vertebrate}{vertebrata}} (\omterm{def:g3:bones-and-muscles:skeleton}{rangka} dengan \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} di
dalamnya); setiap \omterm{def:g4:classifying-animals:vertebrate}{vertebrata} adalah seekor \emph{\omterm{def:g1:animals-around-us:animal}{hewan}}. Kucing di
dalam \omterm{prop:g3:sorting-living-things:groups}{mamalia}, \omterm{prop:g3:sorting-living-things:groups}{mamalia} di dalam \omterm{def:g4:classifying-animals:vertebrate}{vertebrata}, \omterm{def:g4:classifying-animals:vertebrate}{vertebrata} di dalam \omterm{def:g1:animals-around-us:animal}{hewan}:
tiap kotak duduk di dalam kotak yang lebih besar.
\end{example}

\begin{definition}[Vertebrata dan invertebrata]\label{def:g4:classifying-animals:vertebrate}
Suatu \emph{vertebrata}\index{vertebrata} adalah \omterm{def:g1:animals-around-us:animal}{hewan} dengan \omterm{def:g3:bones-and-muscles:skeleton}{rangka}
dalam yang dibangun mengelilingi sebuah \emph{tulang
belakang}\index{tulang belakang} --- seperti \omterm{def:g3:bones-and-muscles:skeleton}{tulang} belakang yang kamu
rasakan pada \cref{ex:g3:bones-and-muscles:feel}. \omterm{def:g1:animals-around-us:animal}{Hewan} tanpa \omterm{def:g3:bones-and-muscles:skeleton}{tulang}
belakang adalah seekor \emph{invertebrata}\index{invertebrata}:
\omterm{prop:g3:sorting-living-things:groups}{serangga}, laba-laba, siput, cacing, ubur-ubur --- sebagian besar dari
seluruh \omterm{def:g1:animals-around-us:animal}{hewan}.
\end{definition}

\section{Kelima kelas vertebrata}

\begin{proposition}[Kelima kelas vertebrata]\label{prop:g4:classifying-animals:classes}
\omterm{def:g4:classifying-animals:vertebrate}{Vertebrata} diklasifikasikan ke dalam lima kelompok besar, yang disebut
\emph{kelas}\index{kelas vertebrata}:
\begin{enumerate}
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{ikan}}: bersisik, bersirip, bernapas di dalam air ---
        \omterm{prop:g3:sorting-living-things:groups}{ikan} trout, hiu, sarden;
  \item \emph{amfibi}\index{amfibi}: berkulit telanjang yang lembap,
        \omterm{def:g2:animals-grow-up:born}{telurnya} di air, masa kecilnya sebagai \omterm{ex:g2:animals-grow-up:frog}{berudu} --- katak,
        kodok, salamander;
  \item \emph{reptil}\index{reptil}: berkulit bersisik yang kering,
        kebanyakan mengeluarkan \omterm{def:g2:animals-grow-up:born}{telur} bercangkang di darat --- kadal,
        ular, kura-kura, buaya;
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{burung}}: berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}, berparuh, bertelur bercangkang
        --- \omterm{prop:g3:sorting-living-things:groups}{burung} hitam, \omterm{prop:g3:sorting-living-things:groups}{burung} unta, penguin;
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{mamalia}}: berbulu atau berambut, anaknya diberi \omterm{def:g2:animals-grow-up:born}{susu} ---
        kucing, paus, kelelawar, manusia.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Mengelompokkan kiriman kebun binatang]\label{ex:g4:classifying-animals:zoo}
Enam peti datang: seekor ular sanca (\omterm{def:g1:the-five-senses:sense}{kulit} bersisik yang kering, \omterm{def:g2:animals-grow-up:born}{telur}
bercangkang --- \omterm{prop:g4:classifying-animals:classes}{reptil}), seekor kodok (\omterm{def:g1:the-five-senses:sense}{kulit} telanjang yang lembap ---
\omterm{prop:g3:sorting-living-things:groups}{amfibi}), seekor penguin (berbulu \omterm{prop:g3:sorting-living-things:groups}{burung} --- \omterm{prop:g3:sorting-living-things:groups}{burung}), seekor hiu
(bersirip dan berinsang --- \omterm{prop:g3:sorting-living-things:groups}{ikan}), seekor lumba-lumba (\omterm{def:g2:animals-grow-up:born}{susu} bagi
anaknya --- \omterm{prop:g3:sorting-living-things:groups}{mamalia}), dan sepeti siput raksasa (sama sekali tanpa
\omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} --- \omterm{def:g4:classifying-animals:vertebrate}{invertebrata}, tidak masuk \omterm{prop:g4:classifying-animals:classes}{kelas vertebrata} mana
pun).
\end{example}

\begin{remark}[Reptil atau amfibi?]\label{rem:g4:classifying-animals:confusion}
Kadal dan salamander tampak mirip pada pandangan sekilas --- berkaki
empat, berekor panjang. Kulitnyalah yang memutuskan: \omterm{def:g1:the-five-senses:sense}{kulit} kadal
kering dan bersisik, \omterm{def:g1:the-five-senses:sense}{kulit} salamander lembap dan telanjang; \omterm{def:g2:animals-grow-up:born}{telur}
kadal bercangkang dan diletakkan di darat, \omterm{def:g2:animals-grow-up:born}{telur} salamander diletakkan
telanjang di dalam air. Bayangan tubuh yang sama, kelas yang berbeda.
\end{remark}

\section{Memakai sebuah klasifikasi}

\begin{method}[Menempatkan seekor vertebrata dalam kelasnya]\label{met:g4:classifying-animals:place}
Ajukan pertanyaannya berurutan, dan berhentilah pada jawaban ya yang
pertama:
\begin{enumerate}
  \item Berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}? --- seekor \omterm{prop:g3:sorting-living-things:groups}{burung}.
  \item Berbulu atau berambut (atau memberi \omterm{def:g2:animals-grow-up:born}{susu} kepada anaknya)? ---
        seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia}.
  \item Berkulit bersisik yang kering? --- seekor \omterm{prop:g4:classifying-animals:classes}{reptil}.
  \item Berkulit telanjang yang lembap dan bermasa kecil sebagai
        \omterm{ex:g2:animals-grow-up:frog}{berudu}? --- seekor \omterm{prop:g3:sorting-living-things:groups}{amfibi}.
  \item Bersirip dan berinsang, hidupnya di air? --- seekor \omterm{prop:g3:sorting-living-things:groups}{ikan}.
\end{enumerate}
Sama sekali tidak bertulang belakang? Berarti ia seekor \omterm{def:g4:classifying-animals:vertebrate}{invertebrata},
dan pertanyaannya tidak berlaku.
\end{method}

\begin{proposition}[Apa yang diberitahukan sebuah kelas secara cuma-cuma]\label{prop:g4:classifying-animals:free}
Mengetahui kelas seekor \omterm{def:g1:animals-around-us:animal}{hewan} berarti mengetahui seberkas fakta
sebelum melihatnya: kalau hanya diberi tahu ``ia seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia}'',
kamu sudah mengharapkan \omterm{def:g1:animals-around-us:covering}{bulu}, \omterm{def:g2:animals-grow-up:born}{susu} bagi anaknya, dan --- seperti
hampir semua \omterm{prop:g3:sorting-living-things:groups}{mamalia} --- anak yang \omterm{def:g2:animals-grow-up:born}{lahir hidup}. Penghematan itulah
seluruh inti \omterm{def:g4:classifying-animals:classification}{klasifikasi}: pelajari kelompoknya sekali, dan setiap
anggotanya datang dalam keadaan setengah dikenal.
\end{proposition}
\admitted

\begin{example}[Sebuah spesies baru datang]\label{ex:g4:classifying-animals:new}
Naturalis masih menemukan \omterm{def:g1:animals-around-us:animal}{hewan} yang belum pernah terlihat
sebelumnya. Ketika seekor \omterm{def:g1:animals-around-us:animal}{hewan} mirip kelelawar yang baru ditemukan
dengan \omterm{def:g1:animals-around-us:covering}{bulu} dan \omterm{def:g2:animals-grow-up:born}{susu}, ia langsung masuk ke \omterm{prop:g3:sorting-living-things:groups}{mamalia} --- dan seketika
ahli \omterm{def:g1:living-or-not:living}{biologi} mengharapkan \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}, jantung berbilik empat dan
darah yang hangat, sebelum satu pun diperiksa. \omterm{def:g4:classifying-animals:classification}{Klasifikasinya}
membayarkan ramalan.
\end{example}

\begin{remark}[Menuju pohon keluarga]\label{rem:g4:classifying-animals:tree}
Mengapa \omterm{def:g1:animals-around-us:covering}{bulu}, \omterm{def:g2:animals-grow-up:born}{susu} dan \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} mengembara bersama dengan
begitu setia? Jawaban lengkapnya --- bahwa anggota sebuah kelompok
mewarisi ciri bersamanya dari nenek moyang bersama --- termasuk kisah
kekerabatan dan evolusi, yang diambil pada tahun-tahun berikutnya.
Untuk sekarang, kotak yang bersarang itu memang bekerja, dan bekerja
dengan menakjubkan.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:1}
Apa itu \omterm{def:g4:classifying-animals:vertebrate}{vertebrata}? Apa itu \omterm{def:g4:classifying-animals:vertebrate}{invertebrata}? Berikan dua contoh
masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:2}
Sebutkan kelima \omterm{prop:g4:classifying-animals:classes}{kelas vertebrata} beserta ciri yang menentukannya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:3}
Tempatkan tiap \omterm{def:g1:animals-around-us:animal}{hewan}: sarden, buaya, salamander, \omterm{prop:g3:sorting-living-things:groups}{burung} unta, paus,
cacing tanah.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:4}
Tuliskan alamat bersarang sang kucing, dari kotak terkecil ke
terbesar, seperti pada \cref{ex:g4:classifying-animals:cat}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:5}
Seekor kadal dan seekor salamander tampak mirip. Dua ciri mana yang
memisahkan kelasnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:6}
Kelompok mana yang memuat sebagian besar dari seluruh \omterm{def:g1:animals-around-us:animal}{hewan}? Sebutkan
empat anggotanya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:7}
Jalankan \cref{met:g4:classifying-animals:place} dengan suara keras
atas penguin dan atas lumba-lumba, pertanyaan demi pertanyaan.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:8}
Kamu hanya diberi tahu: ``\omterm{def:g1:animals-around-us:animal}{hewan} ini seekor \omterm{prop:g3:sorting-living-things:groups}{burung}.'' Sebutkan tiga hal
yang sekarang kamu harapkan tanpa melihatnya, menurut
\cref{prop:g4:classifying-animals:free}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:9}
Paus hampir tidak punya rambut dan hidup seperti \omterm{prop:g3:sorting-living-things:groups}{ikan}. Dengan memakai
\cref{met:g4:classifying-animals:place} dan
\cref{ex:g3:sorting-living-things:surprises}, jelaskan mengapa ia
tetap seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia} --- dan mengapa ``hidup di laut'' tidak pernah
menjadi ciri \omterm{def:g4:classifying-animals:classification}{klasifikasi}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:10}
Ular tidak berkaki, dan cacing tanah pun tidak. Mengapa yang satu
\omterm{prop:g4:classifying-animals:classes}{reptil} \omterm{def:g4:classifying-animals:vertebrate}{vertebrata} dan yang lain \omterm{def:g4:classifying-animals:vertebrate}{invertebrata}? Ciri tunggal mana yang
mengalahkan ketiadaan kakinya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:classifying-animals:11}
Platipus membingungkan para naturalis: berbulu dan bersusu, namun ia
mengeluarkan \omterm{def:g2:animals-grow-up:born}{telur} bercangkang. Kelas mana yang menang, dan mengapa?
Apa yang ditunjukkan teka-teki itu tentang \omterm{def:g4:classifying-animals:classification}{mengklasifikasikan} menurut
\emph{berkas} ciri alih-alih menurut satu ciri saja?
\end{exercise}
