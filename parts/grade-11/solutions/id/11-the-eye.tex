\section*{Bab \ref{ch:g11:the-eye} --- Mata dan Fotoreseptornya}

\begin{solution}{exo:g11:the-eye:1}
Kornea, pupil (lewat irisnya), lensa, badan kaca, lalu lapisan retina
yang bening (\omterm{def:g10:cells-common-unit:cell}{sel} ganglion dan \omterm{def:g10:cells-common-unit:cell}{sel} bipolar) sampai ke \omterm{def:g11:the-eye:photoreceptors}{fotoreseptor} di
belakangnya.
\end{solution}

\begin{solution}{exo:g11:the-eye:2}
\omterm{def:g11:the-eye:photoreceptors}{Sel batang}: sekitar 120 juta, sangat peka, satu pigmen, penglihatan
malam dalam kelabu. \omterm{def:g11:the-eye:photoreceptors}{Sel kerucut}: sekitar 6 juta, menuntut cahaya
terang, tiga pigmen, penglihatan siang dan warna, paling tajam pada foveanya.
\end{solution}

\begin{solution}{exo:g11:the-eye:3}
Fovea adalah lekuk tengah retinanya, yang hanya terbuat dari \omterm{def:g11:the-eye:photoreceptors}{sel
kerucut} yang berjejal rapat, tempat penglihatannya paling tajam.
Bintik buta adalah tempat saraf optiknya keluar, tanpa \omterm{def:g11:the-eye:photoreceptors}{fotoreseptor}.
\end{solution}

\begin{solution}{exo:g11:the-eye:4}
Kerucut S sekitar 45\%, \omterm{def:g11:the-eye:photoreceptors}{sel batang} sekitar 60\%, kerucut M sekitar
20\%, kerucut L sekitar 10\%.
\end{solution}

\begin{solution}{exo:g11:the-eye:5}
Pada \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X; keadaannya \omterm{def:g11:genes-and-disease:genetic}{resesif} terpaut X: terungkapkan pada
setiap lelaki yang membawa \omterm{def:g10:universal-dna:gene}{alelnya}, pada perempuan hanya dengan dua
salinan, sehingga ia jauh lebih lazim pada lelaki.
\end{solution}

\begin{solution}{exo:g11:the-eye:6}
Pada malam hari hanya \omterm{def:g11:the-eye:photoreceptors}{sel batangnya} yang menanggapi, dan \omterm{def:g10:cells-common-unit:cell}{sel} itu punya
satu pigmen: \omterm{def:g10:cells-common-unit:cell}{sel} itu melaporkan terangnya, bukan panjang gelombangnya,
sehingga tidak ada warna. Foveanya hanya punya \omterm{def:g11:the-eye:photoreceptors}{sel kerucut}, yang
terlalu tidak peka bagi bintangnya; jika ditengok dari samping,
bayangan bintangnya jatuh pada \omterm{def:g11:the-eye:photoreceptors}{sel batang}, yang mengenalinya.
\end{solution}

\begin{solution}{exo:g11:the-eye:7}
Warnanya sama: nisbah S:M:L-nya 0:0.6:1 pada keduanya. Terangnya
berbeda: cahaya 2 memancing tiap kerucutnya separuh sebanyak itu.
Warnanya adalah nisbahnya; terangnya adalah jumlahnya.
\end{solution}

\begin{solution}{exo:g11:the-eye:8}
Anak lelakinya menerima \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X-nya dari ibunya: semuanya normal.
Anak perempuannya menerima \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X ayahnya: semuanya \omterm{def:g11:genes-and-disease:genetic}{pembawa}, dengan
penglihatan yang normal. Anak lelaki dari tiap anak perempuan \omterm{def:g11:genes-and-disease:genetic}{pembawa}
itu buta warna dengan peluang setengah.
\end{solution}

\begin{solution}{exo:g11:the-eye:9}
Seorang lelaki memerlukan satu X mutan, seorang perempuan dua. Jika
frekuensi \omterm{def:g10:universal-dna:gene}{alelnya} 0.08, seorang perempuan membawa dua dengan peluang
$0.08^2 = 0.0064$, yaitu sekitar 0.6\% --- enam belas kali lebih
sedikit daripada 8\%.
\end{solution}

\begin{solution}{exo:g11:the-eye:10}
Bedanya menumpuk seiring waktu sejak sebuah penggandaan: 4\% antara L
dan M berarti salinan yang belakangan; 57\% antara salah satunya dan S
berarti salinan yang jauh lebih tua. Penggandaan L/M-nya adalah
peristiwa yang terakhir, sedangkan perpisahan S-nya jauh lebih awal.
\end{solution}

\begin{solution}{exo:g11:the-eye:11}
Moyang primata Dunia Lama menggandakan \omterm{def:g10:universal-dna:gene}{gen} \omterm{def:g11:the-eye:photoreceptors}{opsin} terpaut X-nya yang
satu-satunya itu; satu salinannya menggeser puncaknya ke panjang gelombang yang
lebih panjang, sehingga memberi jenis kerucut ketiga dan sebuah dimensi
warna yang baru. Penggandaan itu terjaga, barangkali karena ia menolong
menemukan buah yang masak dan daun muda di antara dedaunannya.
\end{solution}

\begin{solution}{exo:g11:the-eye:12}
Biru dan kuning (dan biru kehijauan terhadap hijau). \omterm{def:g10:universal-dna:gene}{Gen} S-nya berada
pada \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} biasa, sehingga kedua \omterm{def:g10:universal-dna:gene}{alelnya} harus mutan pada kelamin
mana pun --- langka dan tidak bergantung pada kelaminnya.
\end{solution}

\begin{solution}{exo:g11:the-eye:13}
$\num{2.5e-6}/\num{17e-3} \approx \num{1.5e-4}\,\unit{rad}$, yaitu
sekitar setengah menit busur. Pada \qty{10}{m}: $\num{1.5e-4} \times 10 =
\qty{1.5}{mm}$.
\end{solution}

\begin{solution}{exo:g11:the-eye:14}
Seekor jantan punya satu X, jadi satu \omterm{def:g10:universal-dna:gene}{alel}, jadi satu kerucut jenis
M/L: dua jenis kerucut bersama S-nya. Seekor betina dengan dua \omterm{def:g10:universal-dna:gene}{alel}
yang berbeda mengungkapkan yang satu pada sebagian kerucutnya dan yang
lain pada kerucut lainnya: tiga jenis, sehingga penglihatan tiga warna;
betina dengan dua \omterm{def:g10:universal-dna:gene}{alel} yang serupa punya dua jenis seperti jantannya.
\end{solution}

\begin{solution}{exo:g11:the-eye:15}
\omterm{def:g10:cells-common-unit:cell}{Sel} ganglion dan saraf optiknya utuh, sehingga denyut listrik yang
diantarkan kepadanya mencapai otak penglihatannya sebagai isyarat.
Mutunya dibatasi jumlah elektrodenya (ratusan, berhadapan dengan
sejuta serabut dan jutaan \omterm{def:g10:cells-common-unit:cell}{sel} penerima) dan hilangnya penyaringan
bayangan yang dikerjakan retinanya sendiri.
\end{solution}

\begin{solution}{pb:g11:the-eye:1}
\textbf{1.} \qty{470}{nm}: S 40, M 35, L 18. \qty{530}{nm}: S 0, M 100,
L 80. \qty{590}{nm}: S 0, M 40, L 90. \qty{650}{nm}: S 0, M 2, L 12.

\textbf{2.} \qty{470}{nm} memancing ketiganya; \qty{650}{nm} pada
dasarnya hanya L.

\textbf{3.} Dengan S dan M saja: \qty{590}{nm} memberi (0, 40) dan
\qty{650}{nm} memberi (0, 2); \qty{530}{nm} memberi (0, 100). Cahaya
\qty{590}{nm} (jingga) dan \qty{530}{nm} (hijau) hanya berbeda pada
kuatnya M, seperti hijau yang redup dan hijau yang terang, dan itulah
pasangan yang sulit; \qty{650}{nm} tampak seperti cahaya yang sangat redup.

\textbf{4.} Dengan S dan L saja: \qty{530}{nm} memberi (0, 80) dan
\qty{590}{nm} memberi (0, 90) --- nyaris serupa: hijau dan jingga terkacaukan.

\textbf{5.} Keduanya tetap mengenali cahaya merah, dengan jenis kerucut
yang masih dimilikinya; yang hilang adalah perbandingan antara
tanggapan M dan L-nya, dan itulah yang memisahkan merah dari hijau.

\textbf{6.} $X^NX^c$: ayahnya memberinya satu-satunya X yang dimiliki
ayahnya, yang membawa \omterm{def:g10:universal-dna:gene}{alel} mutannya; penglihatannya yang normal
menunjukkan bahwa X-nya yang lain normal.

\textbf{7.} Anak lelakinya: $X^NY$ normal atau $X^cY$ buta warna,
masing-masing setengah. Anak perempuannya: $X^NX^N$ atau $X^NX^c$,
keduanya berpenglihatan normal, masing-masing setengah (yang kedua
menjadi \omterm{def:g11:genes-and-disease:genetic}{pembawa}).

\textbf{8.} Ines adalah \omterm{def:g11:genes-and-disease:genetic}{pembawa} dengan peluang $1/2$. Seorang anak
perempuan buta warna jika ia menerima $X^c$ dari kedua orang tuanya:
dari ayahnya sudah pasti, dari Ines dengan peluang $1/2 \times 1/2$:
jadi $1/4$.

\textbf{9.} Anak perempuan yang buta warna punya dua \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X mutan,
satu dari tiap orang tuanya; seorang ayah mewariskan satu-satunya X-nya
kepada tiap anak perempuannya, sehingga X-nya mutan dan ia sendiri buta warna.

\textbf{10.} Ya: anak lelakinya menerima X-nya darinya, dan X itu
membawa \omterm{def:g10:universal-dna:gene}{alel} mutannya; dengan penglihatan yang normal, ia adalah
$X^NX^c$.

\textbf{11.} Rodopsinnya bercabang lebih dahulu; lalu S berpisah dari
garis L/M-nya; L dan M berpisah paling akhir: (rodopsin, (S, (L, M))).

\textbf{12.} Beda 4\% bagi 35 juta tahun; 57\% akan memakan sekitar
$35 \times 57/4 \approx 500$ juta tahun --- taat asas dengan asal
\omterm{def:g10:common-ancestry:bodyplan}{vertebrata} yang awal (hubungannya tidak persis sebanding pada beda yang
besar, sehingga ini hanyalah orde besarnya).

\textbf{13.} Kedua \omterm{def:g10:universal-dna:gene}{gen} yang berdampingan dan nyaris serupa itu dapat
salah berpasangan selama kedua \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X-nya berpasangan; sebuah
pertukaran lalu memberi satu X dengan tiga salinan dan satu X dengan
satu salinan. Seorang lelaki yang menerima X bersalinan tunggal itu
hanya punya satu kerucut jenis M/L: yaitu buta warna merah--hijau.

\textbf{14.} Pemasangannya bersandar pada keserupaan urutannya: dua \omterm{def:g10:universal-dna:gene}{gen}
yang berdampingan yang 96\% serupa dapat berpasangan tanpa sejajar, \omterm{def:g10:universal-dna:gene}{gen}
1 dengan \omterm{def:g10:universal-dna:gene}{gen} 2, lalu bertukar secara tak seimbang.

\textbf{15.} Penggandaan itu terjadi sekali, pada moyang bersama
primata Dunia Lama, setelah perpisahannya dari mamalia lain dan monyet
Dunia Baru, sekitar 35 juta tahun lalu, dan diwarisi seluruh keturunannya.

\textbf{16.} $\num{2.5e-6}/\num{17e-3} \approx \num{1.5e-4}$ rad, yaitu
sekitar $0.5'$.

\textbf{17.} $\num{1.5e-4} \times 6 \approx \qty{0.9}{mm}$: lebih halus
daripada goresan \qty{1.7}{mm}-nya, dan itulah sebabnya penglihatan
yang normal membaca baris itu dengan nyaman (dalam praktiknya kaburnya
optik membawa batasnya mendekati $1'$).

\textbf{18.} Sedikitnya sepuluh kali lebih kasar (100 \omterm{def:g11:the-eye:photoreceptors}{sel batang}
berbagi satu isyarat, sekitar 10 kali 10): beberapa menit busur;
sebagai gantinya, seratus \omterm{def:g10:cells-common-unit:cell}{sel} penerima menyatukan cahayanya dan
mengenali sumber yang jauh lebih redup.

\textbf{19.} Kerapatan dua kali berarti jarak $\sqrt{2}$ lebih rapat,
dan \omterm{def:g11:the-eye:parts}{mata} yang lebih besar menambahkan jarak fokus yang lebih panjang:
sekitar dua kali lebih halus.

\textbf{20.} Kerucut S, M dan L, yang masing-masing nyaris buta
terhadap ujung spektrum yang jauh dari puncaknya; satu penggandaan
bersambung \omterm{def:g10:universal-dna:gene}{gen} \omterm{def:g11:the-eye:photoreceptors}{opsin} terpaut X-nya, 35 juta tahun lalu, memberi primata
kerucut ketiganya; foveanya memisahkan sekitar setengah menit busur.
\end{solution}
