\chapter{Digestion et nutriments}\label{ch:g7:digestion-and-nutrients}

\Cref{ch:g4:journey-of-food} suivait la bouchée de pain le long du
tube et promettait que «~les \omterm{def:g4:journey-of-food:digestion}{sucs digestifs} la démontent~». Cette
année, nous ouvrons les portes de l'atelier~: que font exactement
les sucs, quels sont les \omterm{def:g7:digestion-and-nutrients:families}{nutriments} qu'ils produisent, et comment la
paroi de l'\omterm{def:g4:journey-of-food:tract}{intestin grêle} gère-t-elle le plus grand passage de
frontière du corps~? La bouchée de pain est de retour --- sous de
meilleurs instruments.

\section{Les sucs au travail}

\begin{proposition}[La digestion est un démontage chimique]\label{prop:g7:digestion-and-nutrients:chemical}
Mâcher et brasser ne font que casser les aliments en morceaux
\emph{plus petits}~; le vrai démontage est chimique. Les \emph{sucs
digestifs}\index{suc digestif} --- la \omterm{def:g4:journey-of-food:tract}{salive}, le suc de l'\omterm{def:g4:journey-of-food:tract}{estomac},
les sucs déversés dans l'\omterm{def:g4:journey-of-food:tract}{intestin grêle} --- contiennent des
substances de travail qui coupent les gros constituants des aliments
en petits~: l'amidon en sucres, la matière de construction de la
viande et des \omterm{def:g2:animals-grow-up:born}{œufs} en ses petites unités, les graisses en les leurs.
Chaque suc a ses spécialités, et chacun travaille au mieux dans les
\omterm{def:g6:exploring-our-environment:conditions}{conditions} de son propre poste.
\end{proposition}
\admitted

\begin{example}[La version de paillasse]\label{ex:g7:digestion-and-nutrients:bench}
L'expérience classique fait tourner la \omterm{def:g4:journey-of-food:digestion}{digestion} dans du verre. Deux
tubes de pâte d'amidon cuite à la \omterm{def:g6:exploring-our-environment:conditions}{température} du corps~: l'un reçoit
de la \omterm{def:g4:journey-of-food:tract}{salive} fraîche, l'autre de la \omterm{def:g4:journey-of-food:tract}{salive} bouillie (donc gâtée). Une
heure plus tard, le test de l'amidon trouve le tube 2 inchangé ---
toujours de l'amidon --- alors que la pâte du tube 1 s'est changée
en sucres, exactement le goût sucré de
\cref{ex:g4:journey-of-food:sweet}, produit sans la moindre bouche.
Le suc seul a fait le démontage --- et le témoin bouilli montre que
sa substance de travail est détruite par la chaleur, comme le serait
une machinerie délicate.
\end{example}

\begin{remark}[Pourquoi démonter, au fond~?]\label{rem:g7:digestion-and-nutrients:why}
Les gros constituants des aliments ne peuvent pas plus traverser la
paroi de l'intestin qu'une armoire ne passe par une boîte aux
lettres. Et le corps n'en veut pas entiers~: il veut des
\emph{pièces} --- de petites unités qu'il peut rebâtir en sa propre
substance (\cref{prop:g6:matter-from-food:transform}). La \omterm{def:g4:journey-of-food:digestion}{digestion}
est un démontage pour le transport \emph{et} pour le remontage~: du
meuble en kit, à chaque repas.
\end{remark}

\section{Les nutriments}

\begin{definition}[Les familles de nutriments]\label{def:g7:digestion-and-nutrients:families}
Le démontage donne les \emph{nutriments}\index{nutriment} --- les
petits constituants prêts pour le \omterm{prop:g4:journey-of-food:absorb}{sang}~:
\begin{itemize}
  \item les \emph{sucres} (le \omterm{prop:g7:organs-at-work:bill}{glucose} avant tout, le carburant de
        \cref{prop:g7:organs-at-work:bill}), issus des aliments
        amylacés et sucrés~;
  \item les petites unités de \emph{matière de construction},
        issues de la viande, du poisson, des \omterm{def:g2:animals-grow-up:born}{œufs}, des produits
        laitiers et des légumineuses --- les briques de la
        construction de \cref{prop:g3:balanced-diet:jobs}~;
  \item les unités des \emph{graisses} --- à la fois carburant
        dense et matériau de construction~;
  \item plus des passagers qui n'ont besoin d'aucun démontage~: les
        \emph{vitamines}, les \emph{\omterm{prop:g4:what-plants-need:needs}{sels minéraux}}, et l'eau
        elle-même.
\end{itemize}
Les \omterm{def:g3:balanced-diet:families}{familles d'aliments} de \cref{def:g3:balanced-diet:families}
sont, au fond, des fournisseurs de ces familles de nutriments ---
la liste de courses derrière la liste de courses.
\end{definition}

\begin{example}[Un déjeuner, détaillé]\label{ex:g7:digestion-and-nutrients:lunch}
Le déjeuner équilibré de \cref{ex:g3:balanced-diet:lunch}, relu en
\omterm{def:g7:digestion-and-nutrients:families}{nutriments}~: les pâtes --- de l'amidon, démonté en \omterm{prop:g7:organs-at-work:bill}{glucose} par
litres~; le poisson --- de la matière de construction, coupée en ses
unités~; l'huile sur les haricots --- des unités de graisse~; les
haricots et la pomme --- des vitamines, des \omterm{prop:g4:what-plants-need:needs}{sels minéraux}, de l'eau,
plus des fibres. Les fibres, remarque-le, sont l'exception qui
éclaire le système~: aucun de nos sucs ne sait les couper, si bien
qu'elles ne traversent rien et poursuivent leur route --- le volume
des restes (\cref{prop:g4:journey-of-food:absorb}), qui fait son
propre balayage utile.
\end{example}

\section{Le grand passage}

\begin{proposition}[L'absorption au niveau des villosités]\label{prop:g7:digestion-and-nutrients:absorption}
Le passage promis dans
\cref{prop:g4:journey-of-food:absorb} se fait à travers la paroi
interne de l'\omterm{def:g4:journey-of-food:tract}{intestin grêle}, et cette paroi est bâtie pour cela~:
repliée, et tapissée de millions de
\emph{villosités}\index{villosité} --- des bosses en forme de doigt,
hautes d'un millimètre, chacune enveloppant ses propres fins
\omterm{def:g4:heart-and-blood:vessels}{vaisseaux sanguins}. Les \omterm{prop:g7:digestion-and-nutrients:absorption}{villosités} sont la raison pour laquelle la
surface de passage atteint la taille d'un tapis de pièce
(\cref{rem:g4:journey-of-food:surface})~: des replis sur des replis
sur des doigts. Les \omterm{def:g7:digestion-and-nutrients:families}{nutriments} traversent leurs parois minces et
rejoignent le \omterm{prop:g4:journey-of-food:absorb}{sang}, qui les porte d'abord à un grand organe de tri
et de stockage --- le \emph{foie}\index{foie} --- avant la livraison
générale.
\end{proposition}
\admitted

\begin{example}[Encore le cahier des charges]\label{ex:g7:digestion-and-nutrients:spec}
Grandes, minces, humides, tapissées de \omterm{prop:g4:journey-of-food:absorb}{sang}~: les \omterm{prop:g7:digestion-and-nutrients:absorption}{villosités}
satisfont exactement le cahier des charges de
\cref{prop:g7:how-animals-breathe:spec} --- pour des \omterm{def:g7:digestion-and-nutrients:families}{nutriments} au
lieu du \omterm{def:g7:what-is-respiration:respiration}{dioxygène}. Poches des \omterm{def:g7:how-animals-breathe:four}{poumons}, lamelles des \omterm{def:g7:how-animals-breathe:four}{branchies},
\omterm{prop:g7:digestion-and-nutrients:absorption}{villosités} de l'intestin~: partout où le corps doit faire passer
beaucoup de choses par une frontière, il bâtit la même machine.
Repère le cahier des charges une fois, et tu le reconnaîtras
partout.
\end{example}

\begin{method}[Suivre un nutriment d'un bout à l'autre]\label{met:g7:digestion-and-nutrients:trace}
Pour n'importe quelle bouchée, suis un \omterm{def:g7:digestion-and-nutrients:families}{nutriment}~:
\begin{enumerate}
  \item nomme la famille de l'aliment et ses principaux
        constituants~;
  \item poste par poste, note où son démontage commence et
        s'achève, et quels sucs ont travaillé dessus~;
  \item traverse avec lui au niveau des \omterm{prop:g7:digestion-and-nutrients:absorption}{villosités}, jusque dans le
        \omterm{prop:g4:journey-of-food:absorb}{sang}, en passant par le \omterm{prop:g7:digestion-and-nutrients:absorption}{foie}~;
  \item livre-le~: à une \omterm{def:g5:first-look-at-cells:cell}{cellule} au travail comme carburant
        (\cref{prop:g7:organs-at-work:bill}), ou à un chantier
        comme matériau
        (\cref{def:g6:matter-from-food:production}).
\end{enumerate}
\end{method}

\begin{example}[Le suivi, appliqué au pain]\label{ex:g7:digestion-and-nutrients:breadtrace}
Le pain~: famille des féculents, essentiellement de l'amidon. Le
démontage commence dans la bouche (la \omterm{def:g4:journey-of-food:tract}{salive} --- le goût sucré de la
longue mastication), fait une pause dans l'acide de l'\omterm{def:g4:journey-of-food:tract}{estomac},
s'achève dans les sucs de l'\omterm{def:g4:journey-of-food:tract}{intestin grêle}~: du \omterm{prop:g7:organs-at-work:bill}{glucose}. Passage~:
\omterm{prop:g7:digestion-and-nutrients:absorption}{villosités}, \omterm{prop:g4:journey-of-food:absorb}{sang}, \omterm{prop:g7:digestion-and-nutrients:absorption}{foie} --- qui met une partie du \omterm{prop:g7:organs-at-work:bill}{glucose} en banque
pour plus tard, l'un de ses nombreux services. Livraison~: un \omterm{def:g3:bones-and-muscles:muscle}{muscle}
de cuisse en plein sprint le brûle dans l'heure. La bouchée de
l'année 4, entièrement comptabilisée.
\end{example}

\section{Problème~: l'intestin de verre}

\begin{problem}[{Problème du week-end --- bâtir la digestion sur la
paillasse, tube par tube}]\label{pb:g7:digestion-and-nutrients:1}
La classe bâtit un «~intestin de verre~»~: un portoir de tubes à la
\omterm{def:g6:exploring-our-environment:conditions}{température} du corps, chacun simulant un poste du \omterm{def:g4:journey-of-food:tract}{tube digestif}.
Tube M (bouche)~: pâte d'amidon plus \omterm{def:g4:journey-of-food:tract}{salive} fraîche. Tube M$'$~:
pâte d'amidon plus \omterm{def:g4:journey-of-food:tract}{salive} bouillie. Tube S (\omterm{def:g4:journey-of-food:tract}{estomac})~: cubes de
blanc d'\omterm{def:g2:animals-grow-up:born}{œuf} plus suc gastrique acheté en pharmacie. Tube S$'$~: les
mêmes cubes dans de l'eau pure. Tous tournent deux heures~; puis
viennent les tests --- de l'amidon, des sucres, et un coup d'\omterm{def:g1:the-five-senses:sense}{œil} aux
cubes.

\medskip\noindent\textbf{Partie I --- Prédictions et
résultats.}
\begin{enumerate}
  \item Prédis les résultats des quatre tubes au bout de deux
        heures.
  \item M devient sucré, M$'$ non. Énonce la conclusion, et le rôle
        exact de M$'$.
  \item Les cubes de S rétrécissent, ramollissent et s'éclaircissent~;
        ceux de S$'$ restent inchangés. Qu'a fait le suc gastrique
        --- et quelle famille de \omterm{def:g7:digestion-and-nutrients:families}{nutriments} est en train d'être
        produite~?
  \item Aucun tube n'a été mâché ni brassé. Que prouve le portoir
        sur l'endroit où se trouve le vrai travail de la \omterm{def:g4:journey-of-food:digestion}{digestion}~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Améliorer le modèle.}
\begin{enumerate}[resume]
  \item Un élève propose de faire tourner le tube M à la
        \omterm{def:g6:exploring-our-environment:conditions}{température} du réfrigérateur «~pour la propreté~». Objecte,
        avec le modèle de
        \cref{prop:g6:decomposers-and-soil:speed} appliqué aux
        sucs.
  \item Un autre propose un seul gros tube avec les deux sucs
        mélangés. Quel fait poste par poste de
        \cref{prop:g7:digestion-and-nutrients:chemical} le mélange
        brouillerait-il~?
  \item L'intestin de verre n'a pas de paroi~: rien n'y est absorbé.
        Quelle structure un modèle complet devrait-il ajouter, et
        quelles deux propriétés son matériau doit-il avoir~?
  \item Ajoute des fibres --- des flocons de son --- dans un
        cinquième tube où tous les sucs sont disponibles. Prédis son
        état au bout de deux heures, et ce que cela signifie pour le
        modèle.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Du verre à l'intestin.}
\begin{enumerate}[resume]
  \item Dans le corps, les sucres de M rencontreraient ensuite les
        \omterm{prop:g7:digestion-and-nutrients:absorption}{villosités}. Poursuis leur voyage en quatre étapes.
  \item Le suc de l'\omterm{def:g4:journey-of-food:tract}{estomac} travaille en milieu très acide~; les
        sucs de l'intestin ont besoin que cette acidité soit
        neutralisée. Que dit cela sur «~chaque suc dans les
        \omterm{def:g6:exploring-our-environment:conditions}{conditions} de son propre poste~» --- et sur le fait
        d'avaler en traversant les postes dans l'ordre~?
  \item Un laboratoire pharmaceutique vend des gélules de sucs pour
        les personnes dont les postes fonctionnent mal. Explique,
        avec le portoir, pourquoi une telle thérapie est seulement
        possible.
  \item Résume le travail pratique en une phrase contenant
        \emph{sucs}, \emph{\omterm{def:g7:digestion-and-nutrients:families}{nutriments}} et \emph{tubes témoins}.
\end{enumerate}
\end{problem}
