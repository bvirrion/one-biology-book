\section*{Capítulo \ref{ch:g4:plants-without-seeds} --- Plantas sem sementes}

\begin{solution}{exo:g4:plants-without-seeds:1}
\omterm{prop:g4:plants-without-seeds:damp}{Musgos} e \omterm{prop:g4:plants-without-seeds:damp}{samambaias}. Eles se reproduzem por \omterm{def:g4:plants-without-seeds:spore}{esporos} --- grãos
microscópicos liberados em quantidades enormes.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:2}
Um \omterm{def:g4:plants-without-seeds:spore}{esporo} é microscópico e nu --- sem \omterm{def:g3:animals-through-seasons:active}{reserva de alimento}, sem
plantinha pronta; uma \omterm{def:g2:from-seed-to-plant:seed}{semente} carrega as duas coisas, dentro de uma
casca de proteção. E os \omterm{def:g4:plants-without-seeds:spore}{esporos} são feitos aos milhões, as \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}
às dúzias.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:3}
Na parte de baixo das \omterm{def:g1:plants-around-us:parts}{folhas}: fileiras caprichadas de pontinhos
marrons, cada um um pacote com milhares de \omterm{def:g4:plants-without-seeds:spore}{esporos} que se abrem
quando amadurecem e deixam os \omterm{def:g4:plants-without-seeds:spore}{esporos} partirem como poeira.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:4}
O musgo se reproduz por \omterm{def:g4:plants-without-seeds:spore}{esporos}, e os \omterm{def:g4:plants-without-seeds:spore}{esporos} e as etapas jovens e
delicadas deles precisam de umidade --- eles ressecam e morrem no ar
seco. O lado sombreado do muro continua úmido; o lado ensolarado torra.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:5}
Fazer uma \omterm{def:g1:plants-around-us:plant}{planta} nova diretamente a partir de um pedaço da
planta-mãe --- estolho, \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbo}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{tubérculo}, estaca. Com um único pai.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:6}
A \omterm{def:g1:plants-around-us:plant}{planta} lança um \omterm{def:g1:plants-around-us:parts}{caule} comprido e fino --- o estolho --- por cima
da terra; onde ele encosta no chão, cria \omterm{def:g1:plants-around-us:parts}{raízes} e brota \omterm{def:g1:plants-around-us:parts}{folhas},
virando uma plantinha completa; o estolho depois seca e a filha fica
livre, idêntica ao pai dela.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:7}
Uma batata plantada (um \omterm{ex:g4:plants-without-seeds:bulbtuber}{tubérculo}) brota com segurança dos
``\omterm{def:g1:the-five-senses:sense}{olhos}'' dela, e cada \omterm{def:g1:the-five-senses:sense}{olho} pode virar uma \omterm{def:g1:plants-around-us:plant}{planta} inteira idêntica à
boa variedade plantada --- mais rápido e mais garantido que
\omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:8}
Resposta aberta. \omterm{def:g1:plants-around-us:parts}{Raízes} brancas aparecem no \omterm{def:g1:plants-around-us:parts}{caule} submerso, em geral
depois de uma a duas semanas; a estaca então está pronta para ser
plantada --- uma hortelã nova e completa, vinda de um único pai.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:9}
O \omterm{def:g4:plants-without-seeds:spore}{esporo} é nu e barato: sem \omterm{def:g3:animals-through-seasons:active}{reserva de alimento} nem plantinha, quase
todos se perdem, então é preciso fazer milhões. A \omterm{def:g2:from-seed-to-plant:seed}{semente} é um kit
de sobrevivência caro, então uma dúzia bem equipada basta. Eles
ecoam o bacalhau (milhões, sem cuidado) e a elefanta (poucos, bem
providos).
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:10}
Os morangueiros vieram de estolho --- \omterm{def:g4:plants-without-seeds:vegetative}{reprodução vegetativa}, um
único pai, cópias exatas. Os feijões vieram de \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} feitas por
\omterm{def:g3:flowers-fruits-seeds:parts}{flores}, misturando as contribuições de dois pais (o \omterm{def:g3:flowers-fruits-seeds:parts}{pólen} de uma
\omterm{def:g3:flowers-fruits-seeds:parts}{flor}, as \omterm{def:g4:animal-reproduction:sexual}{células reprodutoras femininas} de outra), então cada pé de
feijão varia um pouco.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:11}
\omterm{def:g4:plants-without-seeds:vegetative}{Reprodução vegetativa} --- o \omterm{def:g1:plants-around-us:tree}{galho} é uma estaca natural. A enchente
fez o trabalho de tesoura de quem cuida de horta: cortou o broto,
carregou ele e plantou ele na lama molhada, o leito ideal para o
salgueiro criar \omterm{def:g1:plants-around-us:parts}{raiz}. Os rios se margeiam de salgueiros quebrando e
plantando os salgueiros no mesmo gesto.
\end{solution}
