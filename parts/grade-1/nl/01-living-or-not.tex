\chapter{Levend of niet-levend?}\label{ch:g1:living-or-not}

Een slak glijdt langzaam over het natte pad. Naast hem ligt een grijze
kiezelsteen, even klein, even rond. De slak en de kiezel zitten in
dezelfde regen --- en toch leeft de een en de ander niet. Hoe weten we
dat? Dat is de allereerste vraag van de \omterm{def:g1:living-or-not:living}{biologie}, de wetenschap van de
levende wezens.

\section{Wat levende wezens doen}

\begin{definition}[Levend wezen]\label{def:g1:living-or-not:living}
Een \emph{levend wezen}\index{levend wezen} is iets dat
\emph{leeft}\index{leven}: het wordt geboren, het eet, het groeit, het
kan jongen krijgen, en op een dag gaat het dood. Dieren en planten zijn
levende wezens. De biologie\index{biologie} is de wetenschap die ze bestudeert.
\end{definition}

\begin{example}[De slak en de kiezel]\label{ex:g1:living-or-not:snail}
De slak is geboren uit een piepklein eitje. Hij eet slablaadjes. Hij
groeit --- zijn huisje groeit met hem mee. Er kunnen babyslakjes komen.
Dus is de slak een \omterm{def:g1:living-or-not:living}{levend wezen}. De kiezel eet niets, groeit helemaal
niet en krijgt nooit babykiezels: de kiezel leeft niet.
\end{example}

\begin{proposition}[De kenmerken van leven]\label{prop:g1:living-or-not:signs}
Een \omterm{def:g1:living-or-not:living}{levend wezen} vertoont deze kenmerken:
\begin{enumerate}
  \item het wordt \emph{geboren}\index{geboren worden};
  \item het \emph{eet}\index{eten} --- het neemt voedsel op of maakt
        zijn eigen voedsel, zoals planten doen met licht;
  \item het \emph{groeit}\index{groeien};
  \item het kan jongen krijgen --- nieuwe levende wezens zoals het zelf;
  \item het \emph{gaat dood}\index{doodgaan} op een dag.
\end{enumerate}
Iets dat geen van deze kenmerken vertoont, leeft niet.
\end{proposition}
\admitted

\begin{remark}[Elk kenmerk telt]\label{rem:g1:living-or-not:allsigns}
Eén kenmerk alleen is niet genoeg. Een auto ``drinkt'' brandstof, en
een ijspegel groeit langer in de kou --- maar de auto is gebouwd, niet
geboren, en de ijspegel krijgt geen babyijspegels. Om te leven moet
een ding de kenmerken van leven \emph{samen} vertonen: geboren worden,
eten, groeien, nieuwe levende wezens maken, doodgaan.
\end{remark}

\section{Levend, niet-levend, nooit geleefd}

\begin{definition}[Niet-levend ding]\label{def:g1:living-or-not:nonliving}
Een \emph{niet-levend ding}\index{niet-levend ding} is iets dat niet
leeft. Er zijn twee soorten:
\begin{itemize}
  \item dingen die \emph{nooit geleefd hebben}\index{nooit geleefd}: een
        kiezel, water, de wind, een fiets;
  \item dingen die \emph{niet meer leven}\index{niet meer
        leven} of die \emph{afkomstig zijn van} levende wezens: een dood
        blad op de grond, een houten tafel, een wollen trui.
\end{itemize}
\end{definition}

\begin{example}[De houten tafel]\label{ex:g1:living-or-not:table}
Een houten tafel eet niet en groeit niet: hij leeft niet. Maar zijn
hout heeft ooit wel geleefd: het komt van een boom die uit een zaadje
geboren is, jarenlang gegroeid is en omgehakt werd. De tafel is
niet-levend --- en hij komt van een \omterm{def:g1:living-or-not:living}{levend wezen}.
\end{example}

\begin{example}[En een robot dan?]\label{ex:g1:living-or-not:robot}
Een speelgoedrobot beweegt, maakt geluid, en zijn lampje knippert als
een klein oog. Leeft hij? Loop de kenmerken na: hij is in een fabriek
gebouwd, niet geboren; zijn batterij is geen voedsel waarvan hij
groeit; hij zal nooit babyrobots krijgen. De robot beweegt, maar leeft niet.
\end{example}

\begin{method}[Hoe beslis je: levend of niet?]\label{met:g1:living-or-not:decide}
Stel de vragen van het leven, een voor een:
\begin{enumerate}
  \item Is het geboren? Eet het? Groeit het?
  \item Kunnen er jongen van komen? Gaat het op een dag dood?
  \item Als de antwoorden ja zijn: het leeft.
  \item Als de antwoorden nee zijn, stel dan nog één vraag: komt het
        \emph{van} iets levends, zoals hout van een boom?
\end{enumerate}
\end{method}

\begin{remark}[Zaden lijken te slapen]\label{rem:g1:living-or-not:seed}
Een droge boon in een zakje lijkt net zo stil als een kiezel. Maar leg
hem op natte watten, en na een paar dagen komt er een worteltje uit:
het zaad leefde al die tijd, wachtend. Trage, stille levende wezens
worden gemakkelijk voor niet-levende aangezien --- de vragen van het
leven helpen ons beter te kijken.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:1}
Noem de vijf kenmerken van leven. Welke daarvan kun je een katje
vandaag zien doen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:2}
Levend of niet-levend? (a) een vlinder; (b) een fiets; (c) een
madeliefje; (d) een wolk.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:3}
Een dood blad ligt op het pad. Leeft het nu? Heeft het ooit geleefd?
Waar komt het vandaan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:4}
Tom zegt: ``De rivier leeft --- kijk, ze beweegt en ze maakt
lawaai!'' Gebruik de vragen van \cref{met:g1:living-or-not:decide} om
hem vriendelijk te antwoorden.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:5}
Sorteer in de drie bakken van de figuur --- leeft, heeft geleefd,
\omterm{def:g1:living-or-not:nonliving}{nooit geleefd}: een spin, een wollen sok, een regendruppel, een appel
aan de boom, een gevallen dennenappel.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:6}
Een kaarsvlam ``eet'' de was, wordt groter en gaat dood als je hem
uitblaast. Is de vlam een \omterm{def:g1:living-or-not:living}{levend wezen}? Welke vraag van het leven zegt
nee?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:7}
Noem twee dingen in je slaapkamer die van levende wezens komen, zoals
de houten tafel van \cref{ex:g1:living-or-not:table}. Zeg van welk
\omterm{def:g1:living-or-not:living}{levend wezen} elk ding komt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:8}
Probeer het: leg een droge boon op natte watten bij een raam, en een
kiezel op een ander stukje natte watten ernaast. Kijk elke dag, een
week lang. Welke van de twee verandert? Wat laat dat zien?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:9}
Een speelgoedrobot en een slak schuiven allebei over de vloer. Geef
twee vragen van het leven waarop hun antwoorden verschillen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:10}
Paddenstoelen rennen niet zoals dieren en zijn niet groen zoals de
meeste planten. Leg met de kenmerken van
\cref{prop:g1:living-or-not:signs} uit waarom een paddenstoel toch een
\omterm{def:g1:living-or-not:living}{levend wezen} is.
\end{exercise}
