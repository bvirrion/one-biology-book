\begin{solution}{exo:g8:biomes:1}
Major regional living zone with \omterm{def:g3:classify:char}{characteristic} climate and vegetation.
A local \omterm{def:g7:eco-energy:ecosystem}{ecosystem} is a smaller interacting system within (or across)
\omterm{def:g8:biomes:biome}{biome} types.
\end{solution}

\begin{solution}{exo:g8:biomes:2}

\end{solution}

\begin{solution}{exo:g8:biomes:3}

\omterm{def:g8:biomes:terra}{tundra} --- cold, low \omterm{def:g1:plants-animals:plant}{plants}; \omterm{def:g8:biomes:terra}{taiga} --- conifers (any three with features).
\end{solution}
