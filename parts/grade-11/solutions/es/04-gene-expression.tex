\begin{solution}{exo:g11:gene-expression:4}

\end{solution}

\begin{solution}{exo:g11:gene-expression:5}

\end{solution}

\begin{solution}{exo:g11:gene-expression:7}
Original: Met--Lys--Gly--Trp. Mutante: UGA es un stop, así que
Met--Lys--Gly: una \omterm{def:g11:mutations:mutation}{mutación} sin sentido, que trunca la \omterm{def:g11:gene-expression:protein}{proteína}.
\end{solution}

\begin{solution}{exo:g11:gene-expression:8}
\texttt{AAG} sigue siendo Lys: silenciosa, Met--Lys--Gly--Trp sin
cambios. La segunda es una inserción de G que desplaza el marco:
\texttt{AUG AAA GGG CUG GUA A}\dots{} Met--Lys--Gly--Leu--Val\dots, un
desplazamiento del marco sin stop en el fragmento.
\end{solution}

\begin{solution}{exo:g11:gene-expression:9}
En casi todos los bloques de la tabla, los cuatro \omterm{def:g11:gene-expression:code}{codones} que comparten
las dos primeras letras especifican el mismo \omterm{def:g10:chemistry-of-life:families}{aminoácido} (Pro, Thr, Ala,
Gly, Val, Ser, Leu\dots): cambiar la tercera letra no cambia entonces
nada. La degeneración del código está sobre todo en la tercera
posición.
\end{solution}

\begin{solution}{exo:g11:gene-expression:10}
Leído de tres en tres, \texttt{UCU CUC UCU CUC}\dots{} alterna dos
\omterm{def:g11:gene-expression:code}{codones}, UCU y CUC, así que la \omterm{def:g11:gene-expression:protein}{proteína} alterna dos \omterm{def:g10:chemistry-of-life:families}{aminoácidos} ---
como se observa. Con un código de dos letras, la misma repetición daría
un solo \omterm{def:g11:gene-expression:code}{codón} y un solo \omterm{def:g10:chemistry-of-life:families}{aminoácido}. Como el poli-U da Phe, y en la
tabla UCU es Ser y CUC es Leu, el experimento asigna UCU a Ser y CUC a
Leu (o al revés, cosa que zanjan otros experimentos).
\end{solution}

\begin{solution}{exo:g11:gene-expression:11}
Solo hacen falta 1503 pares de \omterm{def:g10:universal-dna:nucleotide}{bases} para la secuencia codificante; el
resto son \omterm{prop:g11:gene-expression:splicing}{intrones}, transcritos y luego recortados del pre-ARNm, más
secuencias reguladoras en los extremos del \omterm{def:g10:universal-dna:gene}{gen}.
\end{solution}

\begin{solution}{exo:g11:gene-expression:12}
Bloquear la \omterm{prop:g11:gene-expression:translation}{traducción} mata a la bacteria y deja funcionando a las
\omterm{def:g10:cells-common-unit:cell}{células} del paciente. La selectividad del fármaco implica que los
\omterm{def:g10:cells-common-unit:organelle}{ribosomas} bacterianos y los humanos, aunque hacen el mismo trabajo con
el mismo código, difieren lo bastante en su estructura como para que
una molécula se una a unos y no a otros --- una diferencia acumulada
desde que se separaron.
\end{solution}

\begin{solution}{exo:g11:gene-expression:13}
La \omterm{def:g11:gene-expression:transcription}{ARN} polimerasa del ratón transcribió el \omterm{def:g10:universal-dna:gene}{gen} de medusa, y sus
\omterm{def:g10:cells-common-unit:organelle}{ribosomas} y sus ARNt tradujeron el ARNm con la misma tabla de \omterm{def:g11:gene-expression:code}{codones},
así que se fabricó la misma secuencia de \omterm{def:g10:chemistry-of-life:families}{aminoácidos} y, por tanto, la
misma \omterm{def:g11:gene-expression:protein}{proteína} fluorescente. Con otro código, los \omterm{def:g11:gene-expression:code}{codones} se leerían
como otros \omterm{def:g10:chemistry-of-life:families}{aminoácidos} y la \omterm{def:g11:gene-expression:protein}{proteína} sería una cadena sin sentido y sin
fluorescencia.
\end{solution}

\begin{solution}{exo:g11:gene-expression:14}
UAG dejaría de detener la \omterm{prop:g11:gene-expression:translation}{traducción}: toda \omterm{def:g11:gene-expression:protein}{proteína} cuyo \omterm{def:g10:universal-dna:gene}{gen} termine en
UAG se alargaría más allá de su final normal y a menudo perdería su
función --- un daño generalizado. Pero un \omterm{def:g10:universal-dna:gene}{gen} truncado por un UAG
prematuro se leería ahora de largo y recuperaría una \omterm{def:g11:gene-expression:protein}{proteína} casi
completa: la segunda \omterm{def:g11:mutations:mutation}{mutación} suprime a la primera.
\end{solution}

\begin{solution}{exo:g11:gene-expression:15}
Un desplazamiento del marco revuelve todo lo que viene detrás y suele
topar con un stop en unos pocos \omterm{def:g11:gene-expression:code}{codones}: cerca del principio no se
fabrica prácticamente ninguna \omterm{def:g11:gene-expression:protein}{proteína} correcta; cerca del final, casi
toda la \omterm{def:g11:gene-expression:protein}{proteína} está intacta y aún puede plegarse y funcionar. Una
\omterm{def:g11:mutations:mutation}{mutación} de cambio de sentido cambia un \omterm{def:g10:chemistry-of-life:families}{aminoácido}: inofensiva si esa
posición tolera el cambio, mortal si está en el centro activo o rompe
el plegamiento --- el efecto depende de qué \omterm{def:g10:chemistry-of-life:families}{aminoácido} y de dónde.
\end{solution}

\begin{solution}{pb:g11:gene-expression:1}
\textbf{1.} \texttt{3'-TAC CGA ACC TTT GGG CTT ATG CCA GTA AAG ACT-5'}.

\textbf{2.} \texttt{AUG GCU UGG AAA CCC GAA UAC GGU CAU UUC UGA}.

\textbf{3.} Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--His--Phe, y stop.

\textbf{4.} 10 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}; 33 \omterm{def:g10:universal-dna:nucleotide}{nucleótidos} (11 \omterm{def:g11:gene-expression:code}{codones}, incluido el de
stop).

\textbf{5.} AUG; la posición de ese primer AUG fija el marco, y todos
los \omterm{def:g11:gene-expression:code}{codones} siguientes se leen acompasados a él.

\textbf{6.} CCG sigue siendo Pro: silenciosa, \omterm{def:g11:gene-expression:protein}{proteína} sin cambios.

\textbf{7.} CGU es Arg en lugar de His: de cambio de sentido,
Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--Arg--Phe.

\textbf{8.} UGA es stop: Met--Ala, sin sentido. Una sola sustitución de
G por A en el tercer \omterm{def:g11:gene-expression:code}{codón} (TGG a TGA).

\textbf{9.} ARNm \texttt{AUG GCU UUG GAA ACC CGA AUA CGG UCA UUU CUG
A}: Met--Ala--Leu--Glu--Thr--Arg--Ile--Arg--Ser--Phe--Leu\dots, sin
stop en el fragmento: desplazamiento del marco, con todos los
\omterm{def:g10:chemistry-of-life:families}{aminoácidos} cambiados a partir del segundo.

\textbf{10.} Mutante 1 (silenciosa, sin cambio) $<$ mutante 2 (un
\omterm{def:g10:chemistry-of-life:families}{aminoácido} cambiado, función quizá conservada) $<$ mutante 4
(desplazamiento del marco, todos los \omterm{def:g10:chemistry-of-life:families}{aminoácidos} mal salvo dos)
$\approx$ mutante 3 (sin sentido, un fragmento de dos \omterm{def:g10:chemistry-of-life:families}{aminoácidos}: la
\omterm{def:g11:gene-expression:protein}{proteína} ha desaparecido).

\textbf{11.} Leu 6, Ser 6, Trp 1. El Trp, con un solo \omterm{def:g11:gene-expression:code}{codón}, cambia con
cualquier sustitución; la Leu o la Ser, con seis, quedan sin cambiar
casi siempre.

\textbf{12.} CUU, CUC, CUA y CUG son todos Leu: cada una de las 3
sustituciones posibles en la tercera posición es silenciosa --- el
100\%.

\textbf{13.} Dos letras dan 16 \omterm{def:g11:gene-expression:code}{codones}, muy pocos para 20 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}
más un stop; tres dan 64, ya de sobra, así que cuatro (256) sería
derroche.

\textbf{14.} UUU $=$ Phe y AAA $=$ Lys.

\textbf{15.} Un \omterm{def:g10:universal-dna:gene}{gen} transferido entre \omterm{def:g10:biodiversity-scales:species}{especies} se lee igual, así que la
\omterm{prop:g10:universal-dna:universal}{transgénesis} funciona; y un código compartido, heredado sin cambios, es
una prueba de que todos los organismos descienden de un antepasado
común que ya lo usaba.

\textbf{16.} 10 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}: \qty{1}{s}. 600 \omterm{def:g10:chemistry-of-life:families}{aminoácidos}: \qty{60}{s}.

\textbf{17.} $1800/80 \approx 22$ \omterm{def:g10:cells-common-unit:organelle}{ribosomas} a la vez.

\textbf{18.} $7200/8 = 900$ copias.

\textbf{19.} Un ARNm da $600/8 = 75$ copias en 10 minutos; hacen falta
unos 27 ARNm.

\textbf{20.} La \omterm{def:g11:gene-expression:protein}{proteína}
Met--Ala--Trp--Lys--Pro--Glu--Tyr--Gly--His--Phe; el mutante 3, la
única sustitución de G por A que crea un stop en el tercer \omterm{def:g11:gene-expression:code}{codón}, la
abolió; un \omterm{def:g10:cells-common-unit:organelle}{ribosoma} construye los diez \omterm{def:g10:chemistry-of-life:families}{aminoácidos} en alrededor de un
segundo.
\end{solution}
