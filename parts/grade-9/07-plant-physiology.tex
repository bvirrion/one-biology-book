\omterm{def:g1:plants-animals:plant}{Plants} make \omterm{def:g2:food-energy:food}{food}, move \omterm{def:g4:plants-need:needs}{water} and sugars, \omterm{def:g2:what-alive:signs}{grow} toward resources and respond

\section{Photosynthesis and gas exchange}

\begin{definition}[Photosynthesis reminder]\label{def:g9:plant-phys:photo}
\omterm{def:g4:plants-need:parts}{Leaves} use \omterm{def:g4:plants-need:needs}{light}, $\mathrm{CO_2}$ and \omterm{def:g4:plants-need:needs}{water} to make sugars and release
$\mathrm{O_2}$ in \omterm{def:g7:microscope:parts}{chloroplasts}. \omterm{def:g7:plant-structure:leaf}{Stomata} \omterm{def:g6:method:control}{control} \omterm{def:g5:circulation:breathing}{gas exchange} and \omterm{def:g4:plants-need:needs}{water}
loss.
\end{definition}

\begin{proposition}[Trade-off]\label{prop:g9:plant-phys:trade}
Open \omterm{def:g7:plant-structure:leaf}{stomata} admit $\mathrm{CO_2}$ but allow \omterm{def:g9:plant-phys:trans}{transpiration}. \omterm{def:g1:plants-animals:plant}{Plants} balance
carbon gain against \omterm{def:g4:plants-need:needs}{water} loss using guard \omterm{def:g7:microscope:cell}{cells} and often close \omterm{def:g7:plant-structure:leaf}{stomata}
in drought.
\end{proposition}
\admitted

\section{Transport}

\begin{definition}[Transpiration stream]\label{def:g9:plant-phys:trans}
\emph{Transpiration}\index{transpiration} is \omterm{def:g4:plants-need:needs}{water} evaporation from \omterm{def:g7:tissues:plant-organs}{leaves}.
It helps pull the continuous \omterm{def:g4:plants-need:needs}{water} column up \emph{xylem}\index{xylem}
from \omterm{def:g7:tissues:plant-organs}{roots}, carrying \omterm{def:g6:nutrition:groups}{minerals}.
\end{definition}

\begin{definition}[Translocation]\label{def:g9:plant-phys:transloc}
\emph{Translocation}\index{translocation} in \emph{phloem}\index{phloem}
moves sugars from sources (often \omterm{def:g7:tissues:plant-organs}{leaves}) to sinks (\omterm{def:g7:tissues:plant-organs}{roots}, \omterm{def:g6:reproduction:plant}{fruits}, growing
tips).
\end{definition}

\section{Growth and responses}

\begin{definition}[Tropisms]\label{def:g9:plant-phys:tropism}
A \emph{tropism}\index{tropism} is a \omterm{def:g2:growing:growth}{growth} \omterm{def:g7:nervous:stim}{response} toward or away from a
\omterm{def:g7:nervous:stim}{stimulus}. \emph{Phototropism}\index{phototropism}: \omterm{def:g2:what-alive:signs}{grow} toward \omterm{def:g4:plants-need:needs}{light}.
\emph{Gravitropism}\index{gravitropism}: \omterm{def:g7:tissues:plant-organs}{roots} down, shoots up relative
to gravity.
\end{definition}

\begin{example}[Auxin idea]\label{ex:g9:plant-phys:auxin}
\omterm{def:g1:plants-animals:plant}{Plant} hormones such as auxins redistribute and change \omterm{def:g7:microscope:cell}{cell} elongation,
bending shoots toward \omterm{def:g4:plants-need:needs}{light} --- a hormone-mediated \omterm{def:g9:plant-phys:tropism}{tropism}.
\end{example}

\begin{definition}[Germination needs]\label{def:g9:plant-phys:germ}
\omterm{def:g6:reproduction:plant}{Seeds} usually need \omterm{def:g4:plants-need:needs}{water}, a suitable temperature and oxygen to germinate;
some also need \omterm{def:g4:plants-need:needs}{light} or a cold period. Stored \omterm{def:g2:food-energy:food}{food} fuels early \omterm{def:g2:growing:growth}{growth}
until \omterm{def:g7:photo-resp:photo}{photosynthesis} begins.
\end{definition}

\begin{method}[Investigating transpiration]\label{met:g9:plant-phys:potom}
\begin{enumerate}
  \item Use a potometer or mass-loss of a leafy shoot.
  \item Change one factor (fan, \omterm{def:g4:plants-need:needs}{light}, humidity).
  \item Keep \omterm{def:g7:plant-structure:organs}{leaf} area and temperature notes; compare rates.
\end{enumerate}
\end{method}
