\section*{Bab \ref{ch:g11:cancer} --- Kerusakan Genom dan Kanker}

\begin{solution}{exo:g11:cancer:1}
\omterm{def:g11:cancer:cancer}{Kanker}: \omterm{def:g10:cells-common-unit:cell}{sel} yang membelah tanpa kendali yang normal dan menyerbu
jaringan tetangganya. \omterm{def:g11:cancer:cancer}{Tumor}: gumpalan yang dibentuk klon semacam itu.
Metastasis: \omterm{def:g11:cancer:cancer}{tumor} baru yang didirikan di tempat lain oleh \omterm{def:g10:cells-common-unit:cell}{sel} yang
berjalan lewat darah atau getah bening.
\end{solution}

\begin{solution}{exo:g11:cancer:2}
\omterm{def:g10:universal-dna:gene}{Gen} yang \omterm{def:g11:gene-expression:protein}{proteinnya} meneruskan isyarat pertumbuhan dan mendorong \omterm{def:g10:cells-common-unit:cell}{selnya}
masuk ke daurnya; setelah bermutasi sehingga \omterm{def:g11:gene-expression:protein}{proteinnya} aktif
terus-menerus, ia menjadi onkogen. Satu pedal gas yang terjepit
menjalankan \omterm{def:g10:cells-common-unit:cell}{selnya} apa pun yang diperbuat \omterm{def:g10:universal-dna:gene}{alel} yang lain.
\end{solution}

\begin{solution}{exo:g11:cancer:3}
Ia menghentikan daurnya ketika \omterm{def:g10:universal-dna:information}{DNA-nya} rusak dan memicu perbaikan atau
pemusnahan diri. Satu \omterm{def:g10:universal-dna:gene}{alel} yang bekerja tetap membuat cukup \omterm{def:g11:gene-expression:protein}{protein}
untuk mengerem; hanya hilangnya keduanya yang menyingkirkan remnya.
\end{solution}

\begin{solution}{exo:g11:cancer:4}
Cahaya ultraviolet (\omterm{def:g11:cancer:cancer}{kanker} kulit), tar tembakau (\omterm{def:g11:cancer:cancer}{kanker} paru-paru),
radiasi pengion, asbes, alkohol, dan papilomavirus (\omterm{def:g11:cancer:cancer}{kanker} leher rahim).
\end{solution}

\begin{solution}{exo:g11:cancer:5}
Seluruh \omterm{def:g10:cells-common-unit:cell}{selnya} berasal dari satu \omterm{def:g10:cells-common-unit:cell}{sel} tempat \omterm{def:g11:mutations:mutation}{mutasi} pertamanya terjadi;
\omterm{def:g10:cells-common-unit:cell}{selnya} berbagi \omterm{def:g11:mutations:mutation}{mutasi} itu dan \omterm{def:g11:mutations:mutation}{mutasi} yang menyusulnya, serta penanda
seperti \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X tak aktif yang sama.
\end{solution}

\begin{solution}{exo:g11:cancer:6}
Sekitar 200 pada umur 40 dan 2700 pada umur 80: faktor 13 bagi faktor 2
pada umurnya --- yaitu beberapa peristiwa yang saling mengalikan, bukan
satu sebab yang sebanding.
\end{solution}

\begin{solution}{exo:g11:cancer:7}
Sekitar 10 dan 17. Perokok 20 batang sehari: $17 \times 1\% = 17\%$,
yaitu satu dari enam.
\end{solution}

\begin{solution}{exo:g11:cancer:8}
Tanpa p53, \omterm{def:g10:cells-common-unit:cell}{selnya} tidak lagi berhenti untuk memperbaiki \omterm{def:g10:universal-dna:information}{DNA} yang rusak
sebelum replikasi, dan tidak pula memusnahkan dirinya ketika
kerusakannya parah: lesinya disalin menjadi \omterm{def:g11:mutations:mutation}{mutasi} pada tiap
pembelahannya, dan \omterm{def:g10:cells-common-unit:cell}{sel} dengan genom yang kacau tetap bertahan.
\end{solution}

\begin{solution}{exo:g11:cancer:9}
Setiap \omterm{def:g10:cells-common-unit:cell}{sel} tubuhnya sudah kehilangan satu \omterm{def:g10:universal-dna:gene}{alel}; satu \omterm{def:g11:mutations:mutation}{mutasi} somatik
pada \omterm{def:g10:universal-dna:gene}{alel} yang lain merampungkan kehilangannya, alih-alih dua \omterm{def:g11:mutations:mutation}{mutasi}
yang saling bebas. Dengan satu langkah yang sudah terambil sejak lahir,
langkah selebihnya tercapai lebih cepat dan pada lebih banyak \omterm{def:g10:cells-common-unit:cell}{sel}.
Saudara perempuannya mewarisi \omterm{def:g10:universal-dna:gene}{alel} itu dari orang tua yang sama dengan
peluang $1/2$.
\end{solution}

\begin{solution}{exo:g11:cancer:10}
Obatnya membunuh \omterm{def:g10:cells-common-unit:cell}{sel} yang membelah tanpa pandang bulu: \omterm{def:g10:cells-common-unit:cell}{sel} akar rambut,
lapisan usus dan sumsum tulangnya cepat membelah dan ikut terkena
bersama \omterm{def:g11:cancer:cancer}{tumornya} --- karena itulah rambutnya rontok, mualnya datang,
dan \omterm{def:g10:cells-common-unit:cell}{sel} putihnya berkurang.
\end{solution}

\begin{solution}{exo:g11:cancer:11}
Berhenti mengakhiri pasokan \omterm{def:g11:mutations:mutation}{mutasi} yang baru, dan \omterm{def:g10:cells-common-unit:cell}{sel} lapisannya
berangsur digantikan, sehingga banyak klon yang membawa langkah awalnya
tersingkir. Tetapi sebagian klon dengan beberapa langkah tetap bertahan,
dan \omterm{def:g11:mutations:mutation}{mutasi} yang sudah ada tidak terhapus; risikonya tetap di atas
risiko orang yang tidak merokok.
\end{solution}

\begin{solution}{exo:g11:cancer:12}
$10^{12} \times 10^{-7} = 10^5$ \omterm{def:g10:cells-common-unit:cell}{sel} memperoleh \omterm{def:g11:mutations:mutation}{mutasi} pertamanya.
Masing-masing mendirikan klon yang dapat membelah ribuan kali, sehingga
\omterm{def:g11:mutations:mutation}{mutasi} keduanya dicari bukan dalam satu \omterm{def:g10:cells-common-unit:cell}{sel} melainkan dalam seluruh
pembelahan klonnya, dan begitu pula yang ketiga: langkahnya
melipatgandakan sasarannya, dan peluang yang sebenarnya berorde jauh di
atas $10^{-21}$.
\end{solution}

\begin{solution}{exo:g11:cancer:13}
\omterm{def:g11:gene-expression:protein}{Protein} virusnya menyingkirkan remnya tanpa menyentuh \omterm{def:g10:universal-dna:gene}{gennya}: \omterm{def:g10:cells-common-unit:cell}{sel} yang
terinfeksi berperilaku seolah-olah p53-nya hilang dan menumpuk \omterm{def:g11:mutations:mutation}{mutasi}
yang merampungkan prosesnya. Vaksinnya membuat sistem kekebalannya
memusnahkan virusnya sebelum virus itu menginfeksi leher rahimnya;
begitu \omterm{def:g10:cells-common-unit:cell}{selnya} terinfeksi dan berubah, vaksinnya tidak lagi menolong ---
karena itulah vaksinasinya diberikan pada awal masa remaja.
\end{solution}

\begin{solution}{exo:g11:cancer:14}
Sebuah polip adalah klon yang telah mengambil satu atau dua langkah;
menyingkirkannya menyingkirkan \omterm{def:g10:cells-common-unit:cell}{sel} tempat langkah selebihnya dapat
terjadi, sehingga \omterm{def:g11:cancer:cancer}{kankernya} tidak pernah terbentuk. Karena runtunan
polip ke \omterm{def:g11:cancer:cancer}{kanker} memakan sepuluh sampai dua puluh tahun, selang sepuluh
tahun menangkap sebagian besar polipnya sebelum runtunannya rampung.
\end{solution}

\begin{solution}{exo:g11:cancer:15}
Kejadiannya naik seiring umur karena langkahnya memakan puluhan tahun,
tetapi langkahnya terambil selama hidupnya, sebagian besar di bawah
paparan --- tembakau sendirian menjelaskan sepertiga kematian
\omterm{def:g11:cancer:cancer}{kankernya}, dan infeksi, alkohol, matahari serta pola makan menjelaskan
sebagian besar sisanya. Menyingkirkan sebuah paparan pada umur berapa
pun mengurangi \omterm{def:g11:mutations:mutation}{mutasi} yang masih akan datang; penapisan menyingkirkan
klonnya sebelum langkah terakhirnya. Usia lanjut adalah saat tagihannya
disodorkan, bukan saat tagihannya ditulis.
\end{solution}

\begin{solution}{pb:g11:cancer:1}
\textbf{1.} Substitusinya hadir dalam \omterm{def:g11:cancer:cancer}{tumornya} dan tidak ada pada \omterm{def:g10:cells-common-unit:cell}{sel}
lain pasiennya, sehingga substitusinya muncul dalam sebuah \omterm{def:g10:cells-common-unit:cell}{sel} tubuh.
\omterm{def:g11:mutations:mutation}{Mutasi} \omterm{def:g10:cells-common-unit:cell}{sel} kelamin akan ada dalam setiap \omterm{def:g10:cells-common-unit:cell}{selnya}, termasuk jaringan yang sehat.

\textbf{2.} Bahwa sebagian besar \omterm{def:g11:mutations:mutation}{mutasi} \omterm{def:g11:cancer:cancer}{tumornya} disebabkan benzopiren
--- \omterm{def:g11:mutations:mutagen}{mutagen} asap tembakau meninggalkan tanda tangannya.

\textbf{3.} \omterm{def:g11:mutations:mutation}{Mutasi} tanpa makna: \omterm{def:g11:gene-expression:code}{kodon} henti yang terlalu dini
memenggal p53, yang lalu tidak dapat bekerja.

\textbf{4.} p53 adalah sebuah rem: satu \omterm{def:g10:universal-dna:gene}{alel} yang bekerja tetap
mengerem, sehingga keduanya harus lenyap (satu bermutasi, satu
terhapus). Reseptornya adalah pedal gas: satu \omterm{def:g11:gene-expression:protein}{protein} yang aktif
terus-menerus menjalankan \omterm{def:g10:cells-common-unit:cell}{selnya} apa pun yang diperbuat \omterm{def:g10:universal-dna:gene}{alel} yang lain.

\textbf{5.} \omterm{def:g10:universal-dna:gene}{Gen} reseptornya adalah pedal gasnya (proto-onkogen yang
berubah menjadi onkogen); p53 adalah remnya (penekan \omterm{def:g11:cancer:cancer}{tumor}).

\textbf{6.} $20 \times 365 \times 35 \approx \num{255000}$ batang rokok.

\textbf{7.} Sekitar $\num{255000}$ substitusi tiap \omterm{def:g10:cells-common-unit:cell}{sel} --- lima puluh
kali 5000 yang ditemukan. Sebagian besar lesinya diperbaiki, dan \omterm{def:g10:cells-common-unit:cell}{sel}
dengan \omterm{def:g11:mutations:mutation}{mutasi} yang terlalu banyak mati; 5000 adalah bagian yang selamat.

\textbf{8.} Keduanya terjadi lebih awal, pada \omterm{def:g10:cells-common-unit:cell}{sel} pendirinya atau
keturunan pertamanya, sehingga tiap \omterm{def:g10:cells-common-unit:cell}{sel} \omterm{def:g11:cancer:cancer}{tumornya} mewarisinya; yang
lainnya muncul kemudian pada anak klonnya.

\textbf{9.} $10^{10} \approx 2^{33}$: 33 pelipatduaan, dengan 100 hari
tiap satunya, yaitu sekitar 9 tahun.

\textbf{10.} Jika \omterm{def:g11:cancer:cancer}{tumornya} memakan 9 tahun atau lebih untuk tumbuh dari
\omterm{def:g10:cells-common-unit:cell}{sel} pendirinya, \omterm{def:g11:mutations:mutation}{mutasi} penting yang terakhir terjadi sekitar 25 tahun
setelah awal 35 tahun merokoknya, atau lebih awal.

\textbf{11.} $17 \times 1\% = 17\%$.

\textbf{12.} 17 \omterm{def:g11:cancer:cancer}{kanker}, yang 1 di antaranya akan terjadi bagaimanapun:
16 dari 100 dapat ditimpakan pada merokoknya.

\textbf{13.} Tidak ada \omterm{def:g11:mutations:mutation}{mutasi} baru yang ditambahkan; lapisannya
memperbarui dirinya dan banyak klon yang membawa langkah awalnya
terlepas dan tergantikan sepanjang beberapa tahun; klon yang tersisa
punya langkah yang lebih sedikit daripada yang akan diperolehnya jika
merokoknya berlanjut.

\textbf{14.} Beberapa \omterm{def:g11:mutations:mutation}{mutasi} langka harus menumpuk dalam satu garis
keturunan, dan garis keturunan itu lalu harus tumbuh sampai berukuran
terdeteksi: tiap tahapnya memakan bertahun-tahun, dan jumlahnya
berpuluh tahun.

\textbf{15.} Satu \omterm{def:g11:mutations:mutation}{mutasi} tiap \omterm{def:g10:cells-common-unit:cell}{sel} tiap batang rokok, pada sekitar
$10^9$ \omterm{def:g10:cells-common-unit:cell}{sel} lapisannya, sepanjang $\num{255000}$ batang rokok, berarti
$10^{14}$ \omterm{def:g11:mutations:mutation}{mutasi} yang tersebar pada paru-parunya; beberapa di antaranya
akan mengenai p53 atau proto-onkogen dalam \omterm{def:g10:cells-common-unit:cell}{sel} yang sudah membawa
langkah yang lain. Jumlah percobaannyalah yang membuat yang tak mungkin
menjadi pasti.

\textbf{16.} Perokoknya: $\num{27000}$ kematian di antara
$\num{11.25e6}$, sekitar 240 tiap 100\,000; yang tidak merokok: 3000 di
antara $\num{33.75e6}$, sekitar 9 tiap 100\,000.

\textbf{17.} Sekitar 27 kali --- seorde dengan 17 sampai 25 pada
gambarnya bagi perokok berat; angka populasinya mencampur perokok
ringan, perokok berat dan bekas perokok.

\textbf{18.} Sekitar 3000 tiap tahun, yaitu jatah yang tidak merokok.
Tidak seketika: \omterm{def:g11:mutations:mutation}{mutasi} yang sudah ada pada perokok saat ini dan bekas
perokok akan menghasilkan \omterm{def:g11:cancer:cancer}{kanker} selama dua puluh atau tiga puluh tahun lagi.

\textbf{19.} Menapis setiap perokok menyelamatkan sekitar 20\% dari
$\num{27000}$, yaitu sekitar 5000 kematian tiap tahun; berhenti merokok
akhirnya akan mencegah sekitar $\num{24000}$ --- lima kali lebih banyak.

\textbf{20.} Tanda tangan G-ke-T membuktikan bahwa \omterm{def:g11:mutations:mutation}{mutasi} \omterm{def:g11:cancer:cancer}{tumornya}
berasal dari asap tembakau; \omterm{def:g11:cancer:cancer}{kankernya} dibuat oleh sebuah onkogen yang
diaktifkan dan kedua \omterm{def:g10:universal-dna:gene}{alel} p53 yang hilang; tidak merokok akan mencegahnya.
\end{solution}
