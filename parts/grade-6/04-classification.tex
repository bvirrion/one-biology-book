\chapter{Classification of Life}\label{ch:g6:classification}

With millions of \omterm{def:g6:classification:species}{species}, biologists need orderly names and groups.
\omterm{def:g6:classification:def}{Classification} sorts \omterm{def:g1:living-nonliving:living}{living things} by shared features and evolutionary
relationships, and scientific names avoid language confusion.

\section{Why classify?}

\begin{definition}[Classification]\label{def:g6:classification:def}
\emph{Classification}\index{classification} (taxonomy) is the grouping
of \omterm{def:g4:ecosystems:levels}{organisms} based on similarities and relationships so they can be
named, compared and studied systematically.
\end{definition}

\begin{proposition}[Purpose]\label{prop:g6:classification:why}
\omterm{def:g6:classification:def}{Classification} makes \omterm{def:g4:protecting:biodiversity}{biodiversity} manageable, reveals patterns of
relatedness, and lets scientists worldwide use the same names for the
same \omterm{def:g6:classification:species}{species}.
\end{proposition}
\admitted

modern \omterm{def:g6:classification:def}{classification} also uses genetics and evolutionary evidence.}

\section{Species and scientific names}

\begin{definition}[Species]\label{def:g6:classification:species}
A \emph{species}\index{species} is a group of \omterm{def:g4:ecosystems:levels}{organisms} that can
typically interbreed to produce fertile \omterm{def:g2:growing:offspring}{offspring} (a useful working
definition for many \omterm{def:g1:plants-animals:animal}{animals} and \omterm{def:g1:plants-animals:plant}{plants}, with known exceptions).
\end{definition}

\begin{definition}[Binomial name]\label{def:g6:classification:binomial}
A \emph{binomial name}\index{binomial name} has two parts: genus and
\omterm{def:g6:classification:species}{species}, usually printed in italics, for example \emph{Homo sapiens}
for humans. The genus starts with a capital letter.
\end{definition}

\begin{example}[Same animal, many common names]\label{ex:g6:classification:names}
``Mountain lion,'' ``puma'' and ``cougar'' can mean the same \omterm{def:g6:classification:species}{species}.
The scientific name \emph{Puma concolor} points to one agreed group.
\end{example}

\section{Levels of groups}

\begin{definition}[Hierarchy]\label{def:g6:classification:hierarchy}
\omterm{def:g4:ecosystems:levels}{Organisms} are nested in ranks such as domain/kingdom, phylum, class,
order, family, genus and \omterm{def:g6:classification:species}{species}. Each step down is a smaller, more
closely related group.
\end{definition}

\section{Keys and sorting}

\begin{method}[Using a simple dichotomous key]\label{met:g6:classification:key}
\begin{enumerate}
  \item Start at couplet 1: two contrasting choices.
  \item Observe the specimen carefully.
  \item Follow the direction given by the true statement.
  \item Continue until a name or group is reached.
  \item Check with a second feature or image when possible.
\end{enumerate}
\end{method}

\begin{example}[Backbone key start]\label{ex:g6:classification:key}
1a Has a backbone $\to$ \omterm{def:g5:vertebrates:split}{vertebrates}. 1b No backbone $\to$
\omterm{def:g5:vertebrates:split}{invertebrates}. Further couplets split \omterm{def:g5:vertebrates:split}{vertebrates} into \omterm{def:g5:vertebrates:groups}{fish}, \omterm{def:g2:animal-groups:five}{amphibians},
and so on.
\end{example}
