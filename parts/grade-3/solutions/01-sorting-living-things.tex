\section*{Chapter \ref{ch:g3:sorting-living-things} --- Sorting Living Things}

\begin{solution}{exo:g3:sorting-living-things:1}
A \omterm{def:g3:sorting-living-things:criterion}{criterion} is the rule used for \omterm{def:g3:sorting-living-things:criterion}{sorting}. Good: a body feature
you can observe --- ``has \omterm{def:g1:animals-around-us:covering}{feathers}'', ``has six \omterm{def:g1:my-body:parts}{legs}''. Avoided:
where it lives or what it happens to be doing --- ``lives near
water'', ``flies''.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:2}
\omterm{prop:g3:sorting-living-things:groups}{Insects}: six \omterm{def:g1:my-body:parts}{legs} and a body in three parts. \omterm{prop:g3:sorting-living-things:groups}{Fish}: fins and
scales, living in water. \omterm{prop:g3:sorting-living-things:groups}{Birds}: \omterm{def:g1:animals-around-us:covering}{feathers}, wings and a beak.
\omterm{prop:g3:sorting-living-things:groups}{Mammals}: \omterm{def:g1:animals-around-us:covering}{fur} or hair, and mothers that feed their young \omterm{def:g2:animals-grow-up:born}{milk}.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:3}
Bee: insect. Goldfish: \omterm{prop:g3:sorting-living-things:groups}{fish}. Ostrich: bird. Cow: mammal. Frog:
amphibian.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:4}
It has \omterm{def:g1:animals-around-us:covering}{fur} and feeds its young \omterm{def:g2:animals-grow-up:born}{milk}, and it has no \omterm{def:g1:animals-around-us:covering}{feathers} and
no beak. Flying is something it does, not something that defines
a bird --- the features say mammal.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:5}
It has no scales and no gills-and-fins body plan of a \omterm{prop:g3:sorting-living-things:groups}{fish} ---
and it feeds its calf \omterm{def:g2:animals-grow-up:born}{milk}. Swimming is its habit; its features
say mammal.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:6}
No. The insect \omterm{def:g3:sorting-living-things:criterion}{criterion} is six \omterm{def:g1:my-body:parts}{legs} (and a three-part body); the
spider's eight \omterm{def:g1:my-body:parts}{legs} fail it. The spider belongs to another group,
whatever it may look like at first glance.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:7}
Daisy: \omterm{def:g1:plants-around-us:parts}{flowers} and \omterm{def:g2:from-seed-to-plant:seed}{seeds}, soft green \omterm{def:g1:plants-around-us:parts}{stem}. Moss: no \omterm{def:g1:plants-around-us:parts}{flowers} ever,
no real \omterm{def:g1:plants-around-us:parts}{stem}. Pine: a tree (woody trunk), \omterm{def:g1:plants-around-us:parts}{leaves} as needles, no
\omterm{def:g1:plants-around-us:parts}{flowers} but cones with \omterm{def:g2:from-seed-to-plant:seed}{seeds}. Rose bush: \omterm{def:g1:plants-around-us:parts}{flowers} and \omterm{def:g2:from-seed-to-plant:seed}{seeds}, woody
\omterm{def:g1:plants-around-us:parts}{stems}.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:8}
Open answer. In most streets the \omterm{prop:g3:sorting-living-things:groups}{insects} win by far --- ants,
flies, bees, beetles, butterflies --- with \omterm{prop:g3:sorting-living-things:groups}{birds} second.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:9}
The friend's. ``Pets versus wild'' sorts by how \omterm{def:g1:animals-around-us:animal}{animals} live with
people, which can change --- a stray cat is the same \omterm{def:g1:animals-around-us:animal}{animal}
indoors and out. ``\omterm{prop:g3:sorting-living-things:groups}{Mammals} versus \omterm{prop:g3:sorting-living-things:groups}{birds}'' sorts by body features
that stay for life: the biologist's kind of \omterm{def:g3:sorting-living-things:criterion}{criterion}.
\end{solution}

\begin{solution}{exo:g3:sorting-living-things:10}
Step 1: covering --- \omterm{def:g1:animals-around-us:covering}{feathers} (tightly packed), a beak, two \omterm{def:g1:my-body:parts}{legs},
wings (as flippers). Step 2: young hatch from \omterm{def:g2:animals-grow-up:born}{eggs}. Step 3: the
features match the \omterm{prop:g3:sorting-living-things:groups}{birds}. Step 4: the impressions ``swims like a
\omterm{prop:g3:sorting-living-things:groups}{fish}, cannot fly'' disagree --- and lose. The penguin is a bird.
\end{solution}
