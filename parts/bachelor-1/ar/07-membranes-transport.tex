\chapter{الأغشية والنقل الغشائي}\label{ch:b1:membranes-transport}

ضع كرية حمراء في ماء نقي فتنتفخ وتنفجر في ثانية واحدة؛ وضعها في ملح
قوي فتنكمش. ثم أعدها إلى البلازما فتحفظ شكلها المقعر الوجهين أربعة
أشهر، وهي تضخ الصوديوم خارجًا والبوتاسيوم داخلًا في كل ثانية من ذلك
الزمن. والغشاء الذي يفعل هذا سماكته خمسة نانومترات --- غلالة من
الدهن بعمق جزيئين، مخترقة ببروتينات --- وهو الحد الذي يجري عبره
أخيرًا كل تبادل من تبادلات \cref{ch:b1:organism-environment}. يصف هذا
الفصل بنية الغشاء، وفيزياء ما يعبره بلا عون، والبروتينات التي تنقل
الباقي وتضخه وتقنّته، والكمون الكهربائي الذي تنتجه هذه الحركة،
والحويصلات التي تنقل ما لا يستطيع بروتين نقله.

\section{الفسيفساء المائعة}

\begin{definition}[الغشاء البلازمي]\label{def:b1:membranes-transport:membrane}
\emph{الغشاء البلازمي}\index{غشاء بلازمي} --- وكل غشاء في \omterm{def:b1:cell-unit-of-life:cell}{الخلية} ---
\emph{طبقة دهنية مضاعفة}\index{طبقة دهنية مضاعفة} سماكتها نحو
\qty{5}{nm}: صفيحتان من الفوسفوليبيدات (\cref{ch:b1:lipids}) ذيولهما
الكارهة للماء متقابلة ورؤوسهما القطبية مواجهة للجهتين المائيتين، مع
كولستيرول بين الذيول، و\emph{بروتينات غشائية}\index{بروتين غشائي}
مغروسة فيها أو ملتصقة بها. وفي نموذج \emph{الفسيفساء
المائعة}\index{نموذج الفسيفساء المائعة} (سينغر ونيكولسون، 1972) تكون
الطبقة المضاعفة مائعًا ثنائي البعد تنتشر فيه الدهون والبروتينات
انتشارًا جانبيًا، وتكون البروتينات فسيفساء من وحدات مستقلة لا غلافًا
متصلًا. والوجهان مختلفان: فالسكريات المرتبطة بالدهون والبروتينات لا
تواجه إلا الخارج (\emph{الغلاف السكري})، وبعض الفوسفوليبيدات محصور في
صفيحة واحدة.
\end{definition}

\begin{proposition}[الأغشية طبقات مضاعفة ومائعة]\label{prop:b1:membranes-transport:evidence}
سماكة \omterm{def:b1:membranes-transport:membrane}{الغشاء البلازمي} جزيئان دهنيان بالضبط، ومكوناته تتحرك في
مستواه.
\end{proposition}
\begin{proof}[شواهد]
استخلص غورتر وغرينديل (1925) دهون عدد معلوم من الكريات الحمراء
ونشراها طبقة واحدة على الماء: فكانت المساحة ضعف مجموع سطح \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}،
فالغشاء إذن طبقة مضاعفة. والمجهرية الإلكترونية تظهر كل غشاء خطين
داكنين (الرؤوس المصبوغة) حول لب شاحب (الذيول)، \qty{5}{nm} في
المجموع. وأدمج فراي وإيديدين (1970) \omterm{def:b1:cell-unit-of-life:cell}{خلية} فأر \omterm{def:b1:cell-unit-of-life:cell}{بخلية} إنسان وُسمت
بروتينات سطحها بصبغين مختلفي اللون: فبعد أربعين دقيقة عند
\qty{37}{\celsius} كان اللونان قد امتزجا امتزاجًا تامًا على \omterm{def:b1:cell-unit-of-life:cell}{الخلية}
الهجينة، ولم يمتزجا عند \qty{4}{\celsius} حيث يكون الدهن شبه صلب.
وقصر بقعة من \omterm{def:b1:membranes-transport:membrane}{بروتين غشائي} متألق بالليزر ومراقبة عودة التألق بانتشار
الجزيئات غير المقصورة إليها (تقنية استرداد التألق) يقيس الانتشار
الجانبي: $\qty{1}{\micro m^2}$ في ثوان معدودة عند الدهون، وأبطأ عند
البروتينات، وصفر عند البروتينات المرساة على \omterm{def:b1:eukaryotic-cell:cytoskeleton}{الهيكل الخلوي}.
\end{proof}

\begin{definition}[البروتينات الغشائية]\label{def:b1:membranes-transport:proteins}
\omterm{def:b1:membranes-transport:membrane}{البروتينات الغشائية} \emph{المندمجة}\index{بروتين مندمج} تعبر الطبقة
المضاعفة بلولب كاره للماء أو أكثر ولا تُستخلص إلا بالمنظفات؛
والبروتينات \emph{المحيطية} مرتبطة بوجه واحد. وبحسب الوظيفة:
\emph{ناقلات} (قنوات، ونواقل، ومضخات)، و\emph{مستقبِلات} ترتبط بإشارة
في الخارج وتعمل في الداخل، و\emph{إنزيمات}، و\emph{مراسٍ} تصل \omterm{def:b1:eukaryotic-cell:cytoskeleton}{الهيكل
الخلوي} \omterm{def:b1:body-plans-tissues:connective}{بالمادة خارج الخلوية}، وبروتينات \emph{تعرّف} تحمل السكريات
التي تتعرف بها \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} بعضها على بعض. والبروتينات نصف كتلة الغشاء
النموذجي وكل وظيفته تقريبًا.
\end{definition}

\section{عبور الغشاء بلا عون}

\begin{proposition}[نفوذية الطبقة المضاعفة]\label{prop:b1:membranes-transport:permeability}
\index{نفوذية}
\omterm{def:b1:membranes-transport:membrane}{الطبقة الدهنية المضاعفة} النقية نفوذة بحرية للجزيئات الصغيرة غير
القطبية ($\mathrm{O_2}$، و$\mathrm{CO_2}$، و$\mathrm{N_2}$،
والستيروئيدات)، ونفوذة إلى حد ما للجزيئات القطبية الصغيرة غير
المشحونة (الماء، والبولة، والإيثانول)، وضعيفة النفوذية للجزيئات
القطبية الأكبر (الغلوكوز، والحموض الأمينية)، وغير نفوذة عمليًا
للشوارد ($\mathrm{Na^+}$، و$\mathrm{K^+}$، و$\mathrm{Cl^-}$،
و$\mathrm{H^+}$) ولا للجزيئات الكبيرة. وتمتد النفوذية على اثنتي عشرة
مرتبة قدر، من $\qty{e-2}{cm/s}$ للماء إلى $\qty{e-14}{cm/s}$
لشاردة $\mathrm{Na^+}$: فلبٌّ كاره للماء سماكته \qty{3}{nm} يمرر ما يذوب
في الزيت ويوقف ما يحمل شحنة.
\end{proposition}

\begin{theorem}[قانون فيك في الانتشار]\label{thm:b1:membranes-transport:fick}
\index{قانون فيك}\index{انتشار}
التدفق الصافي $J$ لمذاب عبر غشاء مساحته $S$ (بالمول في الثانية)
يتناسب مع فرق التركيز عبره:
\[
J = P\,S\,(c_{\text{خارج}} - c_{\text{داخل}}),
\]
حيث \emph{معامل النفوذية} $P$ (بوحدة \unit{m/s}) يجمع معًا معامل
انتشار المذاب في الغشاء، وانحلاليته في الدهن، وسماكة الغشاء
($P = DK/\ell$). والانتشار لا يحتاج إلى طاقة ويجري دائمًا في اتجاه
هبوط التدرج؛ فهو \emph{نقل منفعل}\index{نقل منفعل}.
\end{theorem}
\begin{proof}
داخل الغشاء ينتشر المذاب هابطًا على مقطع تركيز خطي من
$Kc_{\text{خارج}}$ إلى $Kc_{\text{داخل}}$ (حيث $K$ معامل التوزع بين
الدهن والماء)؛ \omterm{thm:b1:membranes-transport:fick}{وقانون فيك} الأول في الكتلة،
$J/S = -D\,\dd c/\dd x$، يعطي
$J/S = DK(c_{\text{خارج}} -
c_{\text{داخل}})/\ell$.
\end{proof}

\begin{definition}[التناضح والكمون المائي]\label{def:b1:membranes-transport:osmosis}
\emph{التناضح}\index{تناضح} هو الانتشار الصافي للماء عبر غشاء يمرر
الماء ولا يمرر المذابات، من المحلول الذي يكون فيه الماء أكثر تركيزًا
(مذابات أقل) إلى الذي يكون فيه أقل. ويوصف \emph{بالكمون
المائي}\index{كمون مائي} $\Psi$، وهو الكمون الكيميائي للماء معبرًا
عنه بضغط، وهو صفر عند الماء النقي تحت الضغط الجوي:
\[
\Psi = \Psi_s + \Psi_p, \qquad \Psi_s = -RTc_s ,
\]
حيث $\Psi_s$، وهو \emph{كمون المذاب}\index{كمون المذاب}، يهبط مع
مجموع تركيز المذابات $c_s$ (بالأوسمول في اللتر، وهو قانون فان
ت هوف)، و$\Psi_p$، وهو \emph{كمون الضغط}\index{كمون الضغط}، هو الضغط
الهيدروستاتيكي فوق الجوي. والماء ينتقل من $\Psi$ الأعلى إلى الأدنى.
وعند \qty{25}{\celsius} يكون $RT = \qty{2.48}{MPa.L/mol}$: فمحلول
\qty{0.3}{osmol/L} له $\Psi_s = \qty{-0.74}{MPa}$. ويكون محلول
\emph{متساوي التوتر}\index{متساوي التوتر} مع \omterm{def:b1:cell-unit-of-life:cell}{خلية} حين لا ينتقل ماء
صافٍ، و\emph{ناقص التوتر} حين يدخل الماء، و\emph{زائد التوتر} حين
يخرج.
\end{definition}

\begin{example}[كرية حمراء في ثلاثة محاليل]\label{ex:b1:membranes-transport:redcell}
البلازما \qty{0.3}{osmol/L}: فيكون $\Psi_s = \qty{-0.74}{MPa}$ على
الجهتين، ولا جريان صافٍ. وفي الماء النقي ($\Psi = 0$) تأخذ \omterm{def:b1:cell-unit-of-life:cell}{الخلية}،
وهي عند \qty{-0.74}{MPa} في داخلها، ماءً حتى يتمزق غشاؤها، الذي لا
يستطيع أن يتمدد إلا بضعة في المئة. وفي ملح \qty{0.6}{osmol/L} تفقد
ماءً حتى يصير داخلها بالتركيز نفسه، عند نصف حجمها. أما \omterm{def:b1:cell-unit-of-life:cell}{خلية} نباتية في
ماء نقي فلا تنفجر: إذ يُشدّ الجدار كلما دخل الماء ويرتفع $\Psi_p$
حتى يصير $\Psi_p = -\Psi_s$، فيكون \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي} في الداخل صفرًا،
ويقف الجريان \omterm{def:b1:cell-unit-of-life:cell}{والخلية} ممتلئة عند \qty{0.74}{MPa}.
\end{example}

\begin{method}[محاسبة الكمون المائي]\label{met:b1:membranes-transport:psi}
\begin{enumerate}
  \item حوّل كل تركيز مذاب إلى أوسمولات (فالملح الذي ينفصل إلى
        شاردتين يُعدّ مرتين) واحسب $\Psi_s =
        -RTc_s$.
  \item أضف حد الضغط: $\Psi_p = 0$ في محلول في وعاء مفتوح أو في \omterm{def:b1:cell-unit-of-life:cell}{خلية}
        حيوانية، وموجب في \omterm{def:b1:cell-unit-of-life:cell}{خلية} نباتية ممتلئة، وسالب في ماء تحت شد في
        وعاء خشبي.
  \item ينساب الماء نحو $\Psi$ الأدنى. ويُبلغ التوازن حين يتساوى
        مقدارا $\Psi$: بتمديد محتوى \omterm{def:b1:cell-unit-of-life:cell}{الخلية} (\omterm{def:b1:cell-unit-of-life:cell}{خلية} حيوانية)، أو
        بارتفاع الضغط (\omterm{def:b1:cell-unit-of-life:cell}{خلية} نباتية).
  \item اقرأ إشارة النتيجة اتجاهًا للجريان، وقيمتها قوةً دافعة؛
        وتتوقف سرعة الجريان أيضًا على نفوذية الغشاء للماء، التي
        ترفعها قنوات \omterm{def:b1:membranes-transport:transporters}{الأكوابورين} مئة ضعف.
\end{enumerate}
\end{method}

\section{بروتينات النقل}

\begin{definition}[القنوات والنواقل والمضخات]\label{def:b1:membranes-transport:transporters}
\emph{القنوات}\index{قناة} بروتينات مندمجة لها مسام مملوء بالماء
تنتشر عبره شاردة أو جزيء صغير بعينه هابطًا مع تدرجه بمعدل يبلغ
$10^8$ في الثانية؛ وكثير منها \emph{مبوَّب}، ينفتح استجابة لتوتر أو
لربيطة أو لقوة ميكانيكية. و\emph{الأكوابورينات}\index{أكوابورين}
قنوات للماء. و\emph{النواقل}\index{ناقل} ترتبط بمذابها على وجه، وتغير
تشكلها، وتطلقه على الوجه الآخر، ألف مرة في الثانية على الأكثر؛ وناقل
الغلوكوز في الكرية الحمراء واحد منها. والقنوات والنواقل العاملة مع
التدرج تؤدي \emph{انتشارًا ميسرًا}\index{انتشار ميسر}، وهو منفعل
لكنه انتقائي وقابل للإشباع. و\emph{المضخات}\index{مضخة} نواقل تقرن
نقل مذاب ضد تدرجه بحلمهة ATP: وهو \emph{النقل الفعال
الأولي}\index{نقل فعال}. و\emph{مضخة الصوديوم
والبوتاسيوم}\index{مضخة الصوديوم والبوتاسيوم} في \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الحيوانية
تصدّر ثلاث شوارد $\mathrm{Na^+}$ وتستورد شاردتي $\mathrm{K^+}$ لكل
ATP؛ و\emph{مضخة البروتون} في \omterm{def:b1:membranes-transport:membrane}{أغشية} النبات والفطر والبكتيريا تصدّر
$\mathrm{H^+}$.
\end{definition}

\begin{proposition}[النقل الفعال الثانوي]\label{prop:b1:membranes-transport:secondary}
\index{النقل الفعال الثانوي}
التدرج الذي تبنيه \omterm{def:b1:membranes-transport:transporters}{مضخة} مخزن طاقة تنفقه نواقل أخرى: \emph{فالناقل
المتشارك}\index{ناقل متشارك} ينقل مذابًا صعودًا بأن يقرنه بشاردة
تهبط في الاتجاه نفسه (\omterm{def:b1:membranes-transport:transporters}{ناقل} الصوديوم والغلوكوز في المعي والكلية؛
\omterm{def:b1:membranes-transport:transporters}{وناقل} البروتون والسكروز في اللحاء)، و\emph{الناقل
المتعاكس}\index{ناقل متعاكس} يقرنه بشاردة تتحرك في الاتجاه المعاكس
(مبادل الصوديوم والكالسيوم). وفي \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الحيوانية تكون العملة تدرج
$\mathrm{Na^+}$؛ وفي النبات والفطر والبكتيريا تدرج $\mathrm{H^+}$.
\end{proposition}

\begin{example}[الحركية تكشف الآلية]\label{ex:b1:membranes-transport:kinetics}

عند حد أقصى، كما يفعل إنزيم (\cref{ch:b1:enzymes}): فهو \omterm{def:b1:membranes-transport:transporters}{ناقل}، قابل
للإشباع، وثابته $K_m$ يناهز \qty{1.5}{mmol/L}؛ ويزاحمه عليه سكر مشابه هو
المانوز. أما تدفق البولة فيرتفع تناسبًا مع تركيزها بلا حد: فهو انتشار
بسيط. وتدفق الغلوكوز إلى \omterm{def:b1:cell-unit-of-life:cell}{خلية} معوية يتوقف حين يُنزع الصوديوم من
اللمعة، أو حين تُسمم \omterm{def:b1:membranes-transport:transporters}{المضخة} بالأوابائين: فهو \omterm{def:b1:membranes-transport:transporters}{نقل فعال} ثانوي.
\end{example}

\section{كمون الغشاء}

\begin{definition}[كمون الغشاء]\label{def:b1:membranes-transport:potential}
تحمل كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} حية فرق كمون كهربائي عبر غشائها البلازمي، هو \emph{كمون
الغشاء}\index{كمون الغشاء} $V_m = V_{\text{داخل}} - V_{\text{خارج}}$،
وهو سالب في الداخل: \qty{-70}{mV} في \omterm{def:b1:body-plans-tissues:nervous}{عصبون}، و\qty{-90}{mV} في ليف
عضلي، و\qty{-120}{mV} أو أكثر في \omterm{def:b1:cell-unit-of-life:cell}{خلية} نباتية. وهو ينشأ لأن الغشاء
نفوذ انتقائيًا لشوارد تختلف تراكيزها عبره.
\end{definition}

\begin{theorem}[معادلة نرنست]\label{thm:b1:membranes-transport:nernst}
\index{معادلة نرنست}\index{كمون التوازن}
تكون شاردة شحنتها $z$ عند تركيزين $c_{\text{خارج}}$ و$c_{\text{داخل}}$
في توازن عبر الغشاء --- أي بلا تدفق صافٍ، وإن كان الغشاء نفوذًا لها
--- حين يساوي الكمون \emph{كمون توازنها}
\[
E_{\text{شاردة}} = \frac{RT}{zF}\,\ln\frac{c_{\text{خارج}}}{c_{\text{داخل}}}
= \frac{\qty{61.5}{mV}}{z}\,\log_{10}\frac{c_{\text{خارج}}}{c_{\text{داخل}}}
\quad (\qty{37}{\celsius}).
\]
فمن أجل $\mathrm{K^+}$ عند \qty{5}{mmol/L} في الخارج
و\qty{140}{mmol/L} في الداخل يكون
$E_K = 61.5\log(5/140) = \qty{-89}{mV}$؛ ومن أجل $\mathrm{Na^+}$ عند
\qty{145}{} خارجًا و\qty{12}{} داخلًا يكون $E_{Na} = \qty{+67}{mV}$.
\end{theorem}
\begin{proof}
نقل مول واحد من الشاردة من الخارج إلى الداخل يغير الطاقة الحرة بمجموع
حد تركيزي وحد كهربائي:
\[
\Delta G = RT\ln\frac{c_{\text{داخل}}}{c_{\text{خارج}}} + zFV_m .
\]
وعند التوازن يكون $\Delta G = 0$، فينتج $V_m = (RT/zF)\ln(c_{\text{خارج}}/c_{\text{داخل}})$.
والتحويل إلى لوغاريتمات الأساس عشرة وإدخال $R = \qty{8.314}{J.mol^{-1}.K^{-1}}$
و$T = \qty{310}{K}$ و$F = \qty{96485}{C/mol}$ يعطي الصورة العددية.
\end{proof}

\begin{proposition}[منشأ كمون الراحة]\label{prop:b1:membranes-transport:resting}
الغشاء في الراحة أنفذ بكثير لشاردة $\mathrm{K^+}$ (عبر قنوات بوتاسيوم
مفتوحة) منه لشاردة $\mathrm{Na^+}$؛ فيرشح $\mathrm{K^+}$ إلى الخارج هابطًا
مع تدرجه التركيزي، فيترك الداخل سالبًا، إلى أن يوازن الجذب الكهربائي
الراجع الرشحَ تقريبًا. ولذلك يقع كمون الراحة قريبًا من $E_K$، مشدودًا
قليلًا نحو $E_{Na}$ برشح الصوديوم الضئيل (ومعادلة غولدمان ترجّح حد
نرنست لكل شاردة بنفوذيتها). \omterm{def:b1:membranes-transport:transporters}{والمضخة} تصون التدرجات التي كان الرشح
ليبددها؛ وهي تسهم أيضًا ببضعة ميليفولطات مباشرة، إذ تُخرج ثلاث شحنات
مقابل شحنتين تدخلان.
\end{proposition}
\begin{proof}[شواهد]
في محور حبار يتبع كمون الراحة $E_K$ حين يُغيَّر البوتاسيوم الخارجي في
مدى واسع (مستقيم ميله \qty{58}{mV} لكل مرتبة قدر عند التراكيز
المرتفعة)، ويحيد عنه عند التراكيز المنخفضة تمامًا كما تتنبأ نفوذية
صوديوم ضئيلة. وسدّ \omterm{def:b1:membranes-transport:transporters}{المضخة} بالأوابائين يترك الكمون شبه ثابت دقائق، ثم
يدعه يضمحل على مدى ساعات كلما نفدت التدرجات: فالكمون كمون انتشار،
\omterm{def:b1:membranes-transport:transporters}{والمضخة} سنده على المدى الطويل.
\end{proof}

\begin{example}[أين تذهب طاقة المضخة]\label{ex:b1:membranes-transport:pumpcost}

\qty{13.2}{kJ/mol}$؛
وثلاث منها \qty{40}{kJ}، تضاف إليها شاردتا $\mathrm{K^+}$ داخلًا عند

الحيوان المستريح، وثلثيه في الدماغ.
\end{example}

\section{النقل بالكتلة}

\begin{proposition}[النقل الحويصلي]\label{prop:b1:membranes-transport:vesicles}
تعبر الجزيئات الكبيرة والجسيمات الغشاءَ البلازمي دون أن تمر عبره:
فعن طريق \emph{\omterm{prop:b1:eukaryotic-cell:secretory}{الإخراج الخلوي}} تندمج حويصلة بالغشاء وتفرغ إلى الخارج؛
وعن طريق \emph{\omterm{def:b1:eukaryotic-cell:endocytosis}{الإدخال الخلوي}} يبتلع الغشاء المادة في حويصلة
(\cref{ch:b1:eukaryotic-cell}).
وكلاهما يقتضي ATP، وكلاهما ينقل الغشاء كما ينقل المحتوى، وكلاهما يحفظ
جهتي الغشاء. والطرق الثلاث لعبور غشاء --- عبر الدهن، وعبر بروتين،
وداخل حويصلة --- انتقائية بثلاث طرق: بالانحلالية، وبالتعرف الجزيئي،
وباختيار ما يُبتلع.
\end{proposition}

\begin{example}[إيصال الكولستيرول]\label{ex:b1:membranes-transport:ldl}
يسافر الكولستيرول في الدم داخل جسيمات مغلفة ببروتين أكبر من أن
يحملها أي \omterm{def:b1:membranes-transport:transporters}{ناقل}. \omterm{def:b1:cell-unit-of-life:cell}{والخلية} المحتاجة إلى الكولستيرول تعرض مستقبِلات
للجسيم؛ فيتجمع الجسيم ومستقبِله في حفرة مغلفة تُقرص، وتوصل الحويصلة
الجسيم إلى \omterm{def:b1:eukaryotic-cell:endomembrane}{جسيم حال} يُحرَّر فيه الكولستيرول. أما من كانت مستقبِلاته
معيبة فلا يستطيع تصفية الجسيمات: فيتضاعف كولستيرول دمه وتنسد شرايينه
في مطلع الرشد. \omterm{def:b1:membranes-transport:membrane}{بروتين غشائي} واحد، ومرض واحد.
\end{example}

\begin{exercise}[$\star\star\star$]\label{exo:b1:membranes-transport:11}
أظهرت \omterm{def:b1:cell-unit-of-life:cell}{خلايا} فراي وإيديدين المدمجة امتزاجًا تامًا لبروتينات السطح عند

\qty{4}{\celsius}، ولا امتزاج عند \qty{37}{\celsius} حين يُسد تصنيع
ATP --- وقد تبين لاحقًا أن النتيجة الأخيرة خاطئة. قل ماذا كانت كل
نتيجة لتعنيه عن آلية الامتزاج، ولماذا كانت الصورة الصحيحة (أن
الامتزاج لا يحتاج إلى ATP) مهمة \omterm{def:b1:membranes-transport:membrane}{لنموذج الفسيفساء المائعة}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:membranes-transport:12}
«\omterm{def:b1:membranes-transport:potential}{كمون الغشاء} ناتج عرضي لتدرجات الشوارد، لا شيء تبنيه \omterm{def:b1:cell-unit-of-life:cell}{الخلية}
مباشرة.» ناقش في فقرة، مستعينًا \omterm{thm:b1:membranes-transport:nernst}{بمعادلة نرنست}، وبالمضخة، وبعدد
الشوارد اللازم فعلًا لشحن الغشاء (بسعة \qty{1}{\micro F/cm^2}).
\end{exercise}

\section{مسألة: الخلية المعوية}

\begin{problem}[{مسألة نهاية الأسبوع --- الخلية التي تسحب الغلوكوز من
المعي: المضخة، والناقل المتشارك، والناقل، ومعادلة نرنست، وصولًا إلى
أعلى نسبة غلوكوز تستطيع الخلية أن تبنيها}]
\label{pb:b1:membranes-transport:1}
\omterm{def:b1:cell-unit-of-life:cell}{خلية} ظهارية معوية ارتفاعها \qty{25}{\micro m} وعرضها
\qty{5}{\micro m}؛ غشاؤها القمي يواجه اللمعة، وغشاؤها القاعدي
الجانبي يواجه الدم. وعصارتها تحمل \qty{12}{mmol/L} من
$\mathrm{Na^+}$ و\qty{140}{mmol/L} من $\mathrm{K^+}$؛ واللمعة والدم
تحملان \qty{145}{mmol/L} من $\mathrm{Na^+}$ و\qty{5}{mmol/L} من
$\mathrm{K^+}$. \omterm{def:b1:membranes-transport:potential}{وكمون الغشاء} \qty{-60}{mV}. خذ $T = \qty{310}{K}$
و$R = \qty{8.314}{J.mol^{-1}.K^{-1}}$ و$F = \qty{96500}{C/mol}$
و$RT/F = \qty{26.7}{mV}$ و$\Delta G_{\text{ATP}} = \qty{-50}{kJ/mol}$
في \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.

\medskip
\noindent\textbf{الجزء الأول --- التدرجات.}
\begin{enumerate}
  \item احسب كموني نرنست لشاردتي $\mathrm{Na^+}$ و$\mathrm{K^+}$.
  \item أي الشاردتين أقرب إلى التوازن عند \qty{-60}{mV}؟ وماذا يقول
        هذا عن نفوذيتي الراحة؟
  \item احسب تغير الطاقة الحرة لمول واحد من $\mathrm{Na^+}$ يدخل
        \omterm{def:b1:cell-unit-of-life:cell}{الخلية}، ولمول واحد من $\mathrm{K^+}$ يخرج.
  \item احسب كلفة الطاقة الحرة لدورة \omterm{def:b1:membranes-transport:transporters}{مضخة} واحدة ($3\,\mathrm{Na^+}$
        خارجًا و$2\,\mathrm{K^+}$ داخلًا) وقارنها بطاقة ATP واحد. فما
        المردود؟
  \item \omterm{def:b1:membranes-transport:transporters}{المضخة} على الغشاء القاعدي الجانبي وحده. فسّر لماذا كان هذا
        الموضع لازمًا لتنقل \omterm{def:b1:cell-unit-of-life:cell}{الخلية} الغلوكوز من اللمعة إلى الدم.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الغلوكوز صعودًا.}
يحمل الغشاء القمي ناقلًا متشاركًا يُدخل غلوكوزًا واحدًا مع شاردتي
$\mathrm{Na^+}$؛ ويحمل الغشاء القاعدي الجانبي \omterm{def:b1:membranes-transport:transporters}{ناقل} غلوكوز (\omterm{def:b1:membranes-transport:transporters}{انتشار
ميسر}).
\begin{enumerate}[resume]
  \item احسب الطاقة الحرة المحررة بدخول شاردتي $\mathrm{Na^+}$ إلى
        \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.
  \item اكتب الطاقة الحرة اللازمة لنقل غلوكوز واحد من اللمعة (تركيزها
        $c_L$) إلى العصارة ($c_C$).
  \item عند حد \omterm{prop:b1:membranes-transport:secondary}{الناقل المتشارك} تتساوى الطاقتان. احسب النسبة القصوى
        $c_C/c_L$.
  \item إذا هبطت اللمعة إلى \qty{0.1}{mmol/L} من الغلوكوز في أواخر
        الامتصاص، فأي تركيز عصاري يستطيع \omterm{prop:b1:membranes-transport:secondary}{الناقل المتشارك} أن يبقي
        عليه؟
  \item غلوكوز الدم \qty{5}{mmol/L}. فسّر كيف يغادر الغلوكوز \omterm{def:b1:cell-unit-of-life:cell}{الخلية}
        إلى الدم دون أي طاقة إضافية.
  \item لماذا وُضع \omterm{prop:b1:membranes-transport:secondary}{الناقل المتشارك} على الغشاء القمي \omterm{def:b1:membranes-transport:transporters}{والناقل} على
        القاعدي الجانبي، لا العكس؟
  \item تحوي محاليل الإماهة الفموية في الكوليرا غلوكوزًا وملحًا. فسّر،
        انطلاقًا من \omterm{prop:b1:membranes-transport:secondary}{الناقل المتشارك}، لماذا يجعل الغلوكوز الملح ---
        والماء --- يُمتصان.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- عدّ الحركة.}
تمتص \omterm{def:b1:cell-unit-of-life:cell}{الخلية} \qty{2e-15}{mol} من الغلوكوز في الثانية.
\begin{enumerate}[resume]
  \item احسب $\mathrm{Na^+}$ الداخل في الثانية عبر \omterm{def:b1:membranes-transport:transporters}{النواقل} المتشاركة،
        وعدد دورات \omterm{def:b1:membranes-transport:transporters}{المضخة} في الثانية اللازم لتصديره.
  \item احسب ATP المنفق في الثانية على هذا التصدير، والكسر من
        الغلوكوز الممتص الذي كان يجب أن يُتنفس لدفع ثمنه (نحو 30 ATP
        لكل غلوكوز).
  \item كل \omterm{prop:b1:membranes-transport:secondary}{ناقل متشارك} يدور \qty{50}{} مرة في الثانية. فكم ناقلًا
        متشاركًا يحتاج الغشاء القمي؟ والغشاء القمي \qty{25}{\micro m^2}
        تضاعفه الزغيبات \qty{20}{} مرة: احسب كثافة \omterm{def:b1:membranes-transport:transporters}{النواقل} المتشاركة
        في الميكرومتر المربع.
  \item يجلب $\mathrm{Na^+}$ الداخل ماءً \omterm{def:b1:membranes-transport:osmosis}{بالتناضح}. فشاردتا
        $\mathrm{Na^+}$ وغلوكوز واحد، مع $\mathrm{Cl^-}$ المرافق،
        أربعة أوسمولات؛ \omterm{def:b1:cell-unit-of-life:cell}{والخلية} تبقي أسموليتها عند
        \qty{0.3}{osmol/L}. احسب حجم الماء الذي يجب أن يتبع المذابات
        في الثانية، والزمن الذي كان سيلزم لمضاعفة حجم \omterm{def:b1:cell-unit-of-life:cell}{الخلية} لو لم
        يغادر الماء عبر الغشاء القاعدي الجانبي.
  \item احسب عدد جزيئات الماء الممتصة لكل جزيء غلوكوز
        (\qty{18}{g/mol}، وكثافة \qty{1}{g/mL}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- شحن الغشاء.}
الغشاء مكثفة سعتها \qty{1}{\micro F/cm^2}.
\begin{enumerate}[resume]
  \item احسب مساحة غشاء \omterm{def:b1:cell-unit-of-life:cell}{الخلية} (خذها علبة \qty{25}{\micro m} في
        \qty{5}{\micro m} في \qty{5}{\micro m} بلا زغيبات) وسعتها.
  \item احسب الشحنة اللازمة لحفظ \qty{-60}{mV} وعدد الشوارد أحادية
        التكافؤ التي تمثلها.
  \item احسب عدد الشوارد الفائضة في الميكرومتر المربع من الغشاء الذي
        يمثله ذلك.
  \item احسب عدد شوارد $\mathrm{K^+}$ في \omterm{def:b1:cell-unit-of-life:cell}{الخلية}، والكسر الذي يجب أن
        يغادر لشحن الغشاء. وعلّق.
  \item تنقل \omterm{def:b1:membranes-transport:transporters}{المضخة} شحنة صافية إلى الخارج في كل دورة. استعمل السؤال
        13 لحساب التيار الذي تحمله والتغير في الكمون الذي كانت
        لتحدثه في ثانية واحدة لو لم يتحرك شيء آخر. ولماذا لا يجمح
        الكمون فعلًا؟
  \item دواء يسد \omterm{def:b1:membranes-transport:transporters}{المضخة}. تنبأ، بالترتيب، بالتغيرات في تدرج الصوديوم،
        وفي امتصاص الغلوكوز، وفي \omterm{def:b1:membranes-transport:potential}{كمون الغشاء}، وفي حجم \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.
  \item دواء يسد قنوات البوتاسيوم بدلًا من ذلك. تنبأ بالتغير في كمون
        الراحة.
  \item اذكر النتيجة: أقصى نسبة تركيز غلوكوز تستطيع \omterm{def:b1:cell-unit-of-life:cell}{الخلية} المعوية أن
        تبنيها، والبروتينان الغشائيان اللذان يتيحان ذلك.
\end{enumerate}
\end{problem}
