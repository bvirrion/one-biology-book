\section*{Chapitre \ref{ch:g12:domesticated-plants} --- La plante domestiquée}

\begin{solution}{exo:g12:domesticated-plants:1}
\omterm{def:g12:domesticated-plants:domestication}{Domestication}~: la transformation d'une \omterm{def:g10:biodiversity-scales:species}{espèce} sauvage en une \omterm{def:g10:biodiversity-scales:species}{espèce}
qui dépend des humains et les sert. \omterm{def:g12:domesticated-plants:domestication}{Sélection artificielle}~: les
humains conservent et sèment les individus qui portent les caractères
qu'ils veulent, ce qui élève la fréquence des \omterm{def:g10:universal-dna:gene}{allèles} responsables.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:2}
Épis non égrenants (aucune dissémination)~; germination immédiate (pas
d'attente d'une mauvaise année)~; perte des toxines (mangée par tout
le monde)~; un seul gros épi sur une plante non ramifiée (tous les
œufs dans le même panier)~; maturation simultanée (une mauvaise
semaine détruit toute la récolte).
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:3}
La descendance uniforme et vigoureuse de deux lignées pures
homozygotes. Sa propre graine disjoint en un mélange de plantes moins
vigoureuses, si bien que le cultivateur rachète de la semence hybride
chaque année.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:4}
Le \omterm{def:g10:universal-dna:gene}{gène} est inséré dans l'\omterm{def:g10:universal-dna:information}{ADN} qu'une bactérie du sol transfère
naturellement dans les \omterm{def:g10:cells-common-unit:cell}{cellules} végétales~; les \omterm{def:g10:cells-common-unit:cell}{cellules} infectées qui
prennent le \omterm{def:g10:universal-dna:gene}{gène} sont régénérées en plantes entières.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:5}
D'environ \num{1.9} à \qty{7.2}{t/ha}~: presque quatre fois plus.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:6}
Les plantes dont le grain s'égrenait le perdaient avant ou pendant la
moisson~; les mutants au rachis solide gardaient le leur et étaient
ceux qu'on récoltait, donc ceux qu'on semait. La moisson elle-même a
sélectionné l'\omterm{def:g10:universal-dna:gene}{allèle} du rachis solide, génération après génération.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:7}
Une seule \omterm{def:g10:biodiversity-scales:species}{espèce}~: ils se croisent et donnent une descendance féconde.
Les cultivateurs ont sélectionné dans chaque lignée l'exagération d'un
organe différent --- feuilles, bourgeon terminal, boutons
floraux ---, si bien qu'ils diffèrent par la régulation de la
croissance, non par leur aptitude à se croiser.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:8}
F1 toutes $Aa$. F2~: $AA$ $1/4$, $Aa$ $1/2$, $aa$ $1/4$. La F1 est
uniforme parce que chaque plante reçoit $A$ d'un parent et $a$ de
l'autre~; en F2, la \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{méiose} et la fécondation redistribuent les \omterm{def:g10:universal-dna:gene}{allèles}
au hasard.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:9}
Parmi des millions de pyrales, quelques-unes portaient une \omterm{def:g11:mutations:mutation}{mutation}
qui leur faisait tolérer la toxine~; dans un champ où chaque plante en
fabrique, elles seules ont survécu et se sont reproduites --- c'est la
sélection. Un refuge maintient en vie de nombreuses pyrales
sensibles~; leurs accouplements avec les rares résistantes donnent des
descendants \omterm{def:g11:genes-and-disease:genetic}{porteurs} d'un seul \omterm{def:g10:universal-dna:gene}{allèle} de résistance qui, cet \omterm{def:g10:universal-dna:gene}{allèle}
étant \omterm{def:g11:genes-and-disease:genetic}{récessif}, meurent sur le maïs à toxine~: l'\omterm{def:g10:universal-dna:gene}{allèle} est donc dilué
à chaque génération.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:10}
Les \omterm{def:g10:universal-dna:gene}{gènes} en cause commandent le développement --- la ramification de
la plante, la croissance de la coque du grain, le nombre de rangées~;
un changement dans la régulation d'un tel \omterm{def:g10:universal-dna:gene}{gène} modifie la forme d'un
organe entier. Peu de changements régulateurs, de grands effets
visibles.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:11}
Ils détiennent les \omterm{def:g10:universal-dna:gene}{allèles} que la sélection a retirés de la plante
cultivée~: résistance aux maladies et aux ravageurs, tolérance à la
sécheresse, au sel ou au froid. Quand une nouvelle maladie ou un
nouveau climat survient, les sélectionneurs réintroduisent ces \omterm{def:g10:universal-dna:gene}{allèles}
dans la plante cultivée par croisement~; une fois un parent sauvage
éteint, ses \omterm{def:g10:universal-dna:gene}{allèles} sont perdus.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:12}
Le pollen du colza féconde la moutarde~; les hybrides portent le \omterm{def:g10:universal-dna:gene}{gène}
de tolérance~; recroisée avec la moutarde pendant quelques générations
et sélectionnée par l'herbicide pulvérisé sur les champs, une moutarde
sauvage tolérante se répand. Processus~: hybridation et passage d'un
\omterm{def:g10:universal-dna:gene}{gène} d'une \omterm{def:g12:selection-drift-speciation:population}{population} à l'autre, puis \omterm{def:g12:selection-drift-speciation:selection}{sélection naturelle} par
l'herbicide.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:13}
Biologiques~: la \omterm{def:g10:cell-metabolism:autotrophy}{photosynthèse} de la plante et sa capacité à remplir
le grain sont proches de leurs limites, et la \omterm{def:g10:biodiversity-scales:biodiversity}{diversité génétique} des
variétés d'élite, d'où doivent venir les gains suivants, est étroite.
Non biologique~: l'usage des engrais et des pesticides n'augmente plus,
ou se réduit, et les stress climatiques se sont accrus. Un
sélectionneur pourrait apporter des \omterm{def:g10:universal-dna:gene}{allèles} des parents sauvages, ou
éditer des \omterm{def:g10:universal-dna:gene}{gènes} de la \omterm{def:g10:cell-metabolism:autotrophy}{photosynthèse} ou de la tolérance à la
sécheresse.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:14}
Croisement~: l'\omterm{def:g10:universal-dna:gene}{allèle} doit exister chez une plante croisable avec la
plante cultivée~; des milliers de \omterm{def:g10:universal-dna:gene}{gènes} sont rebrassés et doivent être
triés par la sélection sur une dizaine de générations. Transgénèse~:
le \omterm{def:g10:universal-dna:gene}{gène} peut venir de n'importe quel organisme~; un seul \omterm{def:g10:universal-dna:gene}{gène} est
ajouté et le reste du génome n'est pas touché~; quelques années.
\end{solution}

\begin{solution}{exo:g12:domesticated-plants:15}
L'arithmétique est identique~: les individus \omterm{def:g11:genes-and-disease:genetic}{porteurs} de certains
\omterm{def:g10:universal-dna:gene}{allèles} se reproduisent davantage, et la fréquence de ces \omterm{def:g10:universal-dna:gene}{allèles}
s'élève. Ce qui diffère, c'est l'agent --- un choix humain à la place
du milieu --- et l'issue pour la plante~: la \omterm{def:g12:selection-drift-speciation:selection}{sélection naturelle}
conserve à une \omterm{def:g10:biodiversity-scales:species}{espèce} la capacité de vivre seule, tandis que la
\omterm{def:g12:domesticated-plants:domestication}{domestication} sélectionne des caractères qui servent le sélectionneur
et laisse la plante dépendante de lui. La \omterm{def:g12:domesticated-plants:domestication}{domestication} est une
\omterm{def:g12:selection-drift-speciation:selection}{sélection naturelle} à sélectionneur différent, non son contraire.
\end{solution}

\begin{solution}{pb:g12:domesticated-plants:1}
\textbf{1.} Téosinte $12 \times 0.08 \approx \qty{1}{g}$~; maïs $600
\times 0.3 = \qty{180}{g}$~: près de deux cents fois plus.

\textbf{2.} De nombreux grains sur une rafle rigide (aucune
dissémination)~; des grains nus (sans protection contre les animaux et
la pourriture)~; un seul gros épi sur une tige unique (aucune
répartition du risque)~; des grains retenus dans des spathes (ils ne
peuvent même pas tomber au sol).

\textbf{3.} L'\omterm{def:g10:universal-dna:gene}{allèle} «~grain nu~» est \omterm{def:g11:genes-and-disease:genetic}{récessif}~: une plante ne montre
des grains nus que si elle en porte deux copies, ce qui, pour un
\omterm{def:g10:universal-dna:gene}{allèle} rare, n'arrive que chez une infime minorité de plantes~; les
\omterm{def:g11:genes-and-disease:genetic}{porteurs} d'une seule copie ont l'air ordinaire.

\textbf{4.} $q^2 = 0.0001$~: une plante sur dix mille.

\textbf{5.} Toute la semence vient de plantes $nn$, dont les grains
ont reçu $n$ de la mère~; le pollen venait surtout du champ ordinaire
($N$ à la fréquence 0.99). La semence est presque entièrement $Nn$, de
sorte que la fréquence de $n$ bondit à environ 0.5 --- une seule
génération de sélection sévère.

\textbf{6.} $600/12 = 50$ en 9000 générations~: un facteur de $50^{1/9000}
\approx 1.0004$ par génération, un gain de 0.04\%. De minuscules pas,
composés sur des milliers de générations, font un grand
changement --- l'arithmétique de la sélection.

\textbf{7.} Avec de nombreux \omterm{def:g10:universal-dna:gene}{gènes} apportant chacun peu, la sélection
élève la fréquence des \omterm{def:g10:universal-dna:gene}{allèles} favorables à tous et continue longtemps
de trouver de nouvelles combinaisons~; un caractère à un seul \omterm{def:g10:universal-dna:gene}{gène}
bascule dès que l'\omterm{def:g10:universal-dna:gene}{allèle} \omterm{def:g11:genes-and-disease:genetic}{récessif} est rendu homozygote.

\textbf{8.} Une seule \omterm{def:g10:biodiversity-scales:species}{espèce}~: les hybrides sont féconds. La
\omterm{def:g12:domesticated-plants:domestication}{domestication} a changé les fréquences des \omterm{def:g10:universal-dna:gene}{allèles} à quelques \omterm{def:g10:universal-dna:gene}{gènes} et
la forme de la plante~; elle n'a pas créé de barrière reproductive.

\textbf{9.} Une maladie ou un ravageur qui vient à bout d'une plante
en vient à bout de toutes~; la réponse est la diversité conservée dans
les variétés anciennes et chez les parents sauvages, en banques de
graines et dans les champs.

\textbf{10.} Le cultivateur est la dissémination de la plante, son
milieu sélectif et sa seule voie vers la génération suivante~: le maïs
est soumis à la \omterm{def:g12:domesticated-plants:domestication}{sélection artificielle} à chaque génération et ne peut
pas s'y soustraire.

\textbf{11.} $10/5.5 - 1 \approx 82\%$ au-dessus du meilleur parent~;
$1 - 8.2/10 = 18\%$ au-dessous de la F1.

\textbf{12.} Chaque lignée est homozygote pour quelques \omterm{def:g10:universal-dna:gene}{allèles}
\omterm{def:g11:genes-and-disease:genetic}{récessifs} défavorables~; en F1, chacun de ces \omterm{def:g10:universal-dna:gene}{allèles} est masqué par
l'\omterm{def:g10:universal-dna:gene}{allèle} fonctionnel de l'autre lignée. En F2, un quart des plantes
est homozygote pour l'\omterm{def:g10:universal-dna:gene}{allèle} défavorable à chacun de ces \omterm{def:g10:universal-dna:gene}{gènes}, et la
moyenne baisse.

\textbf{13.} Acheter~: $10 - 0.4 = \qty{9.6}{t/ha}$~; conserver~:
\qty{8.2}{t/ha}. L'achat l'emporte de \qty{1.4}{t/ha} chaque année, et
c'est pourquoi une entreprise qui détient les lignées pures peut
vendre la semence hybride à chaque saison.

\textbf{14.} L'entreprise garde les lignées~; avec elles, un
cultivateur pourrait refaire l'hybride indéfiniment et l'entreprise ne
vendrait qu'une fois. Les lignées sont son capital~; seul leur produit
est vendu.

\textbf{15.} Chaque champ d'une variété hybride est génétiquement
identique, et les régions ne cultivent que quelques hybrides~: la
diversité se rétrécit. Le cultivateur ne produit plus de semence et
dépend chaque année de son fournisseur.

\textbf{16.} $q^2 = 10^{-6}$~: une pyrale sur un million survit~;
parmi les survivantes, toutes $rr$, la fréquence de $r$ vaut 1.

\textbf{17.} Les survivantes se reproduisent entre elles~; leurs
descendants sont tous $rr$~; en quelques générations, les pyrales du
champ sont résistantes --- comme les bactéries chez un patient qui
suit une cure incomplète.

\textbf{18.} Une survivante résistante ($rr$) accouplée à une pyrale
du refuge ($RR$, l'immense majorité) donne des descendants $Rr$, qui
meurent sur le maïs à toxine. Tant que les pyrales sensibles sont bien
plus nombreuses que les résistantes, presque tous les enfants d'une
pyrale résistante sont hétérozygotes et sont éliminés, et $r$ reste
rare.

\textbf{19.} En commun~: les deux visent à empêcher la multiplication
des rares individus résistants. Différence~: la cure complète les
élimine en tuant chaque individu avant que la résistance ne puisse
être sélectionnée~; le refuge accepte des survivants et les dilue par
des partenaires sensibles.

\textbf{20.} Près de deux cents fois plus de grain par épi~; environ
cinq \omterm{def:g10:universal-dna:gene}{gènes} majeurs~; une plante qui dépend du cultivateur pour être
disséminée et semée.
\end{solution}
