\chapter{Karbohidrat}\label{ch:b1:carbohydrates}

Sebutir kentang dan selembar kertas terbuat dari molekul yang sama,
glukosa, yang dirangkai menjadi rantai. Yang satu makanan dan yang lain
bukan, sebab rantainya disambung dua ikatan yang berbeda: tautan
$\alpha$ \omterm{def:b1:carbohydrates:polysaccharide}{pati} menggulung rantainya menjadi heliks yang dapat dibuka
enzim kita, tautan $\beta$ \omterm{def:b1:carbohydrates:polysaccharide}{selulosa} membentangkannya menjadi pita yang
bertumpuk menjadi serat yang tak sanggup dicerna mamalia mana pun.
Sebuah hati menimbun seratus gram glukosa sebagai \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}, sebatang
pohon menegakkan dirinya dengan \omterm{def:b1:carbohydrates:polysaccharide}{selulosa}, seekor kumbang berbaju \omterm{def:b1:carbohydrates:polysaccharide}{kitin},
dan tiap \omterm{def:b1:cell-unit-of-life:cell}{sel} memajang gula pada permukaannya sebagai jati dirinya. Bab
ini memerikan gula sederhana, ikatan yang menyambungnya, \omterm{def:b1:carbohydrates:polysaccharide}{polisakarida}
yang dibangunnya, serta molekul berhias gula pada permukaan \omterm{def:b1:cell-unit-of-life:cell}{sel} dan
dinding tumbuhan.

\section{Monosakarida}

\begin{definition}[Monosakarida]\label{def:b1:carbohydrates:monosaccharide}
Yang disebut \emph{karbohidrat}\index{karbohidrat} adalah molekul
bersusunan $(\mathrm{CH_2O})_n$ beserta turunannya. Sebuah
\emph{monosakarida}\index{monosakarida} (gula sederhana) adalah rantai
tiga sampai tujuh karbon yang salah satunya membawa gugus karbonil dan
selebihnya hidroksil: sebuah \emph{aldosa}\index{aldosa} bila
karbonilnya di ujung (sebuah aldehida: glukosa, galaktosa, ribosa),
sebuah \emph{ketosa}\index{ketosa} bila ia di tengah (sebuah keton:
fruktosa). Menurut panjangnya: triosa ($n = 3$: gliseraldehida),
pentosa ($n = 5$: ribosa, deoksiribosa), heksosa ($n = 6$: glukosa,
fruktosa, galaktosa). Tiap karbon yang membawa empat gugus berbeda
adalah pusat \emph{kiral}, sehingga tiap rumusnya mewakili beberapa
\emph{stereoisomer}: glukosa memiliki empat karbon kiral dan enam belas
isomer, yang hanya satu dipakai makhluk hidup, yaitu D-glukosa.
\emph{Epimer} berbeda hanya pada satu karbon (glukosa dan galaktosa
pada C4).
\end{definition}

\begin{proposition}[Bentuk cincin dan anomer]\label{prop:b1:carbohydrates:rings}
\index{anomer}\index{proyeksi Haworth}
Dalam air sebuah pentosa atau heksosa hampir seluruhnya berada dalam
bentuk cincin: karbon karbonilnya bereaksi dengan sebuah hidroksil
molekul yang sama (C5 pada glukosa) sehingga menutup cincin
\emph{piranosa} beranggota enam atau cincin \emph{furanosa} beranggota
lima (fruktosa, ribosa). Karbon karbonilnya menjadi pusat kiral baru,
yaitu \emph{karbon anomerik}, yang hidroksilnya dapat terletak di bawah
bidang cincin (anomer $\alpha$) atau di atasnya (anomer $\beta$) pada
gambar Haworth; kedua bentuk itu saling berubah lewat rantai terbuka
dalam beberapa menit. Dalam larutan, glukosa adalah \qty{36}{\%}
$\alpha$, \qty{64}{\%} $\beta$ dan kurang dari \qty{0.1}{\%} rantai
terbuka. Karbon anomerik yang bebas itulah yang membuat sebuah gula
bersifat \emph{pereduksi}: ia dapat membuka lalu mereduksi sebuah
pereaksi tembaga atau perak.
\end{proposition}

\begin{method}[Membaca sebuah proyeksi Haworth]\label{met:b1:carbohydrates:haworth}
\begin{enumerate}
  \item Carilah oksigen cincinnya; karbon anomeriknya (C1 pada sebuah
        \omterm{def:b1:carbohydrates:monosaccharide}{aldosa}, C2 pada sebuah \omterm{def:b1:carbohydrates:monosaccharide}{ketosa}) adalah karbon cincin di
        sebelah kanannya, yang membawa sebuah hidroksil sekaligus
        oksigen cincinnya.
  \item Karbon yang membawa gugus $\mathrm{CH_2OH}$ di luar cincinnya
        (C5 pada glukosa) berada di sebelah kiri oksigen cincinnya;
        $\mathrm{CH_2OH}$-nya menunjuk ke atas pada gula D.
  \item Hidroksil anomerik ke bawah: $\alpha$; ke atas (pada sisi yang
        sama dengan
        $\mathrm{CH_2OH}$): $\beta$.
  \item Bandingkan hidroksil yang lain dengan hidroksil glukosa (bawah,
        atas, bawah bagi C2, C3, C4): satu perbedaan saja sudah mengenali
        sebuah epimer (galaktosa: C4 ke atas).
\end{enumerate}
\end{method}

\begin{example}[Gula yang bukan $(\mathrm{CH_2O})_n$]\label{ex:b1:carbohydrates:derivatives}
Deoksiribosa kehilangan hidroksil pada C2 (\cref{ch:b1:nucleic-acids});
glukosamina dan N-asetilglukosamina membawa sebuah amina alih-alih
hidroksil C2 (\omterm{def:b1:carbohydrates:polysaccharide}{kitin}, \omterm{def:b1:carbohydrates:polysaccharide}{peptidoglikan}); asam glukuronat membawa sebuah
karboksil pada C6 (\omterm{def:b1:carbohydrates:polysaccharide}{polisakarida} \omterm{def:b1:eukaryotic-cell:mitochondrion}{matriks}); gula fosfat
(glukosa-6-fosfat, ribosa-5-fosfat) adalah bentuk gula ketika masuk ke
metabolisme; alkohol gula (gliserol, sorbitol) tidak berkarbonil.
Terasnya sama: rantai karbon kecil berhidroksil banyak yang dicintai air.
\end{example}

\section{Ikatan glikosidik}

\begin{definition}[Ikatan glikosidik, disakarida]\label{def:b1:carbohydrates:glycosidic}
Sebuah \emph{ikatan glikosidik}\index{ikatan glikosidik} menyambungkan
hidroksil anomerik sebuah gula pada sebuah hidroksil gula yang lain
(atau sebuah alkohol, sebuah amina, sebuah basa) lewat \omterm{prop:b1:water-small-molecules:families}{kondensasi},
sehingga mengunci karbon anomeriknya dalam susunan $\alpha$ atau
$\beta$-nya; ia dinamai menurut susunannya dan karbon yang disambungnya:
$\alpha(1{\to}4)$, $\beta(1{\to}4)$, $\alpha(1{\to}6)$. Dua gula yang
tersambung demikian membentuk sebuah
\emph{disakarida}\index{disakarida}: \emph{maltosa} (glukosa
$\alpha(1{\to}4)$ glukosa, dari pencernaan \omterm{def:b1:carbohydrates:polysaccharide}{pati}), \emph{laktosa}
(galaktosa $\beta(1{\to}4)$ glukosa, gula susu), \emph{sukrosa}
(glukosa $\alpha(1{\to}2)\beta$ fruktosa, gula getah tumbuhan dan gula
dapur). Pada sukrosa kedua karbon anomeriknya terpakai: ia tidak
mereduksi dan tidak membuka, dan itulah sebabnya tumbuhan
mengangkutnya (\cref{ch:b1:plant-transport}).
\end{definition}

\begin{method}[Menguji sebuah gula pereduksi]\label{met:b1:carbohydrates:reducing}
\begin{enumerate}
  \item Panaskan larutannya dengan pereaksi tembaga(II) basa (Benedict
        atau Fehling).
  \item Sebuah karbon anomerik yang bebas membuka menjadi aldehida,
        mereduksi $\mathrm{Cu^{2+}}$ yang biru menjadi endapan
        $\mathrm{Cu_2O}$ merah bata; pekatnya warna itu sebanding dengan
        banyaknya gula.
  \item Glukosa, fruktosa, maltosa, laktosa: positif. Sukrosa: negatif
        --- kecuali bila lebih dahulu dihidrolisis asam atau enzim
        invertase, dan sesudahnya glukosa serta fruktosanya bereaksi.
  \item \omterm{def:b1:carbohydrates:polysaccharide}{Pati} tidak bereaksi (satu ujung pereduksinya adalah satu di
        antara ribuan); ia dideteksi dengan iodin, yang masuk ke dalam
        heliks amilosa lalu berubah biru kehitaman.
\end{enumerate}
\end{method}

\section{Polisakarida}

\begin{definition}[Polisakarida]\label{def:b1:carbohydrates:polysaccharide}
Yang disebut \emph{polisakarida}\index{polisakarida} adalah rantai
ratusan sampai puluhan ribu \omterm{def:b1:carbohydrates:monosaccharide}{monosakarida} yang tersambung \omterm{def:b1:carbohydrates:glycosidic}{ikatan
glikosidik}, lurus atau bercabang. Polisakarida simpanan:
\emph{pati}\index{pati} pada tumbuhan --- \emph{amilosa}, glukosa
$\alpha(1{\to}4)$ tak bercabang, \qty{20}{\%}, dan
\emph{amilopektin}, $\alpha(1{\to}4)$ dengan cabang $\alpha(1{\to}6)$
tiap \qtyrange{24}{30}{} residu, \qty{80}{\%} --- yang berkemas dalam
butiran; \emph{glikogen}\index{glikogen} pada hewan dan jamur, seperti
amilopektin tetapi bercabang tiap \qtyrange{8}{12}{} residu, dalam
butiran \omterm{def:b1:eukaryotic-cell:organelle}{sitosol} hati dan otot. Polisakarida struktural:
\emph{selulosa}\index{selulosa}, glukosa $\beta(1{\to}4)$ tak
bercabang, \qtyrange{2000}{15000}{} residu, pada dinding tumbuhan;
\emph{kitin}\index{kitin}, N-asetilglukosamina $\beta(1{\to}4)$, pada
dinding jamur dan \omterm{def:b1:flowering-plant-organization:tissues}{kutikula} artropoda;
\emph{peptidoglikan}\index{peptidoglikan}, N-asetilglukosamina dan asam
N-asetilmuramat berselang-seling yang bersilang ikat lewat peptida
pendek, yakni jala satu molekul yang menjadi dinding bakteri.
\end{definition}

\begin{proposition}[Ikatannya menentukan bentuknya, dan bentuknya fungsinya]\label{prop:b1:carbohydrates:alphabeta}
Dalam sebuah rantai $\alpha(1{\to}4)$ tiap glukosa menghadap arah yang
sama dengan pendahulunya dan rantainya melingkar menjadi heliks berisi
enam residu tiap putaran, yaitu gulungan terbuka dan berair yang mudah
dimasuki enzim: bentuk sebuah \emph{simpanan}. Dalam sebuah rantai
$\beta(1{\to}4)$ tiap glukosa terbalik $180^\circ$ terhadap tetangganya
dan rantainya menjadi pita lurus yang pipih; pitanya berbaring
berdampingan dan berikatan hidrogen menjadi
\emph{mikrofibril} berisi tiga puluh sampai seratus rantai, yang
berhablur, tak larut, dan berkekuatan tarik sekuat baja tiap satuan
massanya: bentuk sebuah \emph{serat}. Enzim yang menghidrolisis ikatan
$\alpha$ (amilase) tidak sanggup menyentuh ikatan $\beta$;
\emph{selulase} ada pada bakteri, jamur dan beberapa hewan, dan mamalia
yang hidup dari rumput melakukannya lewat simbion
(\cref{ch:b1:digestion-absorption}).
\end{proposition}

\begin{example}[Mengapa glikogen bercabang]\label{ex:b1:carbohydrates:branching}
\omterm{def:b1:carbohydrates:polysaccharide}{Glikogen} diuraikan dari ujung tak mereduksinya, satu glukosa pada satu
waktu, oleh \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} fosforilase. Sebuah rantai tak bercabang berisi
\num{50000} glukosa akan memiliki satu ujung semacam itu dan melepas
satu glukosa tiap putaran enzimnya; sebuah partikel \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} seukuran
itu, yang bercabang tiap dua belas residu, memiliki sekitar \num{2000}
ujung dan melepas dua ribu sekaligus. Percabangannya juga menjaga
partikelnya tetap padat dan larut. Seratus gram \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} sebuah hati
dapat digerakkan sepuluh gram tiap jam; glukosa yang sama yang ditimbun
sebagai satu rantai panjang akan menuntut bertahun-tahun untuk terurai.
\end{example}

\begin{example}[Mengapa tidak menimbun glukosanya sendiri]\label{ex:b1:carbohydrates:osmotic}
Seratus gram glukosa (\qty{0.55}{mol}) yang terlarut dalam \qty{1}{L}
air \omterm{def:b1:cell-unit-of-life:cell}{sel} sebuah hati akan menambahkan \qty{0.55}{osmol/L} pada \omterm{def:b1:eukaryotic-cell:organelle}{sitosol}
yang \qty{0.3}{osmol/L}: \omterm{def:b1:cell-unit-of-life:cell}{selnya} akan membengkak sampai tiga kali
volumenya lalu pecah. Sebagai \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} glukosa yang sama menjadi
$\num{7e18}$ partikel, \qty{1e-5}{mol/L}, tak tampak secara osmotik.
Sebuah polimer adalah cara menimbun amat banyak molekul sebagai satu.
\end{example}

\section{Glikokonjugat dan dinding tumbuhan}

\begin{definition}[Glikoprotein, proteoglikan, glikolipid]\label{def:b1:carbohydrates:glycoconjugate}
Gula dipautkan secara kovalen pada protein dan \omterm{def:b1:lipids:fattyacid}{lipid} sehingga membentuk
\emph{glikokonjugat}\index{glikokonjugat}. Sebuah
\emph{glikoprotein}\index{glikoprotein} membawa rantai pendek bercabang
(selusin gula) pada sebagian residu asparagina, serina atau treoninanya,
yang ditambahkan di RE dan Golgi (\cref{ch:b1:eukaryotic-cell}): hampir
tiap protein sekresi dan \omterm{def:b1:membranes-transport:membrane}{protein membran} adalah glikoprotein. Sebuah
\emph{proteoglikan}\index{proteoglikan} adalah teras protein yang
membawa rantai panjang tak bercabang berupa \omterm{def:b1:carbohydrates:glycosidic}{disakarida} asam berulang,
yaitu \emph{glikosaminoglikan}\index{glikosaminoglikan} (hialuronan,
kondroitin sulfat, heparin), yang mengikat air dalam jumlah besar dan
memberi tulang rawan, badan kaca dan \omterm{def:b1:body-plans-tissues:connective}{matriks ekstrasel} kekenyalannya.
\emph{Glikolipid}\index{glikolipid} membawa gula pada ekor \omterm{def:b1:lipids:fattyacid}{lipid} di
lembar luar membran plasma. Bersama gula glikoproteinnya mereka
membentuk \emph{glikokaliks}\index{glikokaliks}, yaitu selimut gula
yang menjadi cara \omterm{def:b1:cell-unit-of-life:cell}{sel} dikenali.
\end{definition}

\begin{example}[Golongan darah]\label{ex:b1:carbohydrates:abo}
Golongan darah ABO adalah tiga versi satu rantai gula pada \omterm{def:b1:carbohydrates:glycoconjugate}{glikolipid}
dan \omterm{def:b1:carbohydrates:glycoconjugate}{glikoprotein} permukaan \omterm{def:b1:cell-unit-of-life:cell}{sel} merah: rantai O berakhir pada fukosa; A
menambahkan sebuah N-asetilgalaktosamina padanya, B menambahkan sebuah
galaktosa, lewat dua versi satu enzim; \omterm{def:b1:cell-unit-of-life:cell}{sel} AB membawa keduanya. Satu
residu gula sudah cukup bagi sistem kekebalan untuk membedakan diri
sendiri dari yang asing, dan bagi sebuah transfusi untuk berhasil atau
membunuh.
\end{example}

\begin{proposition}[Dinding sel tumbuhan]\label{prop:b1:carbohydrates:wall}
\index{dinding sel}
Dinding primer sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} tumbuhan adalah bahan gabungan yang diperkuat
serat: mikrofibril \omterm{def:b1:carbohydrates:polysaccharide}{selulosa} (seperempat massanya, kekuatan tariknya)
yang ditaruh berlapis, yang diikat menjadi satu oleh
\emph{hemiselulosa} (\omterm{def:b1:carbohydrates:polysaccharide}{polisakarida} bercabang yang berikatan hidrogen
pada mikrofibrilnya dan menjembataninya) dan terbenam dalam gel
\emph{pektin} (\omterm{def:b1:carbohydrates:polysaccharide}{polisakarida} asam yang memegang air dan kalsium serta
merekatkan \omterm{def:b1:cell-unit-of-life:cell}{sel} pada lamela tengahnya), dengan beberapa persen protein.
Dinding itu tipis (\qtyrange{0.1}{1}{\micro m}), tertembus air dan zat
terlarut yang kecil, dan cukup kuat menahan \omterm{def:b1:eukaryotic-cell:plantcell}{turgor} \qty{1}{MPa}. \omterm{def:b1:cell-unit-of-life:cell}{Sel}
yang berhenti tumbuh dapat menambahkan \emph{dinding sekunder} yang
tebal dengan lebih banyak \omterm{def:b1:carbohydrates:polysaccharide}{selulosa} dan \emph{lignin}, yaitu polimer
aromatik yang mengedapkan air dan mengakukannya: kayu adalah dinding
sekunder.
\end{proposition}

\section{Gula sebagai bahan bakar dan karbon}

\begin{proposition}[Kedudukan glukosa]\label{prop:b1:carbohydrates:fuel}
\omterm{def:b1:carbohydrates:monosaccharide}{Karbohidrat} menghasilkan \qty{17}{kJ/g} pada oksidasinya dan menjadi
bahan bakar yang dapat dipakai tiap \omterm{def:b1:cell-unit-of-life:cell}{sel}, dengan atau tanpa oksigen
(\cref{ch:b1:respiration-fermentation}); glukosa adalah gula darah
(\qty{5}{mmol/L}), sukrosa gula getah tumbuhan, sedangkan \omterm{def:b1:carbohydrates:polysaccharide}{pati} dan
\omterm{def:b1:carbohydrates:polysaccharide}{glikogen} simpanan keduanya. Gula juga menjadi karbon yang dipakai \omterm{def:b1:cell-unit-of-life:cell}{sel}
untuk membuat segala yang lain: pentosa nukleotidanya, gliserol
\omterm{def:b1:lipids:fattyacid}{lipidnya}, kerangka karbon asam aminonya semuanya meninggalkan lintasan
gulanya (\cref{ch:b1:biosyntheses-integration}). Fotosintesis membuat
gula lebih dahulu; selebihnya menyusul.
\end{proposition}

\begin{example}[Gula sehari]\label{ex:b1:carbohydrates:day}
Seorang manusia memakan sekitar \qty{300}{g} \omterm{def:b1:carbohydrates:monosaccharide}{karbohidrat} sehari, hampir
semuanya sebagai \omterm{def:b1:carbohydrates:polysaccharide}{pati} dan sukrosa; ususnya menghidrolisisnya menjadi
glukosa, fruktosa dan galaktosa, hatinya mengubah kedua yang terakhir
menjadi glukosa, dan glukosanya beredar sebanyak \qty{5}{g} dalam
seluruh darahnya --- pasokan dua puluh menit bagi otak, yang mengambil
\qty{120}{g} sehari, yang diisi ulang terus-menerus dari \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}
hatinya. Lensa mata, \omterm{def:b1:cell-unit-of-life:cell}{sel} darah merah dan medula ginjal hidup dari
glukosa saja; tak ada lagi dalam makanannya yang sanggup menggantinya,
dan ketika glukosanya habis hati membuatnya dari protein.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:carbohydrates:1}
Golongkan glukosa, fruktosa, ribosa dan gliseraldehida menurut cacah
karbonnya dan menurut \omterm{def:b1:carbohydrates:monosaccharide}{aldosa} atau \omterm{def:b1:carbohydrates:monosaccharide}{ketosa}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:carbohydrates:2}
Apa itu karbon anomerik, dan mengapa sebuah \omterm{def:b1:carbohydrates:glycosidic}{disakarida} yang kedua karbon
anomeriknya terpakai dalam ikatannya tidak bereaksi dengan pereaksi
Benedict?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:carbohydrates:3}
Sebutkan monomer dan ikatan \omterm{def:b1:carbohydrates:polysaccharide}{pati}, \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}, \omterm{def:b1:carbohydrates:polysaccharide}{selulosa} dan \omterm{def:b1:carbohydrates:polysaccharide}{kitin}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:carbohydrates:4}
Dari gambar Haworth, sebutkan letak (atas atau bawah) hidroksil pada
karbon 1 sampai 4 $\alpha$-D-glukosa, lalu gambar atau perikan
$\alpha$-D-galaktosa.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:5}
Jelaskan, dari geometri ikatan $\alpha$ dan $\beta$, mengapa amilosa
membentuk heliks dan \omterm{def:b1:carbohydrates:polysaccharide}{selulosa} membentuk pita, dan mengapa uji iodin
berhasil pada yang pertama dan tidak pada yang kedua.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:6}
Tiga tabung memuat glukosa, sukrosa dan \omterm{def:b1:carbohydrates:polysaccharide}{pati}. Rancanglah urutan uji
(Benedict, iodin, \omterm{prop:b1:water-small-molecules:families}{hidrolisis} asam) yang mengenali masing-masing, dengan
menyebutkan hasil yang diharapkan pada tiap ujinya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:7}
Sebuah partikel \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} memegang \num{50000} residu glukosa yang
bercabang tiap 12. Taksirlah banyaknya ujung tak mereduksinya lalu
bandingkan dengan sebuah molekul amilosa seukuran itu. Bila fosforilase
melepas satu glukosa tiap sekon dari tiap ujung, berapa lama
masing-masing melepas \qty{10}{\%} glukosanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:8}
Mikrofibril \omterm{def:b1:carbohydrates:polysaccharide}{selulosa} memiliki kekuatan tarik sekitar \qty{1}{GPa}.
Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} bergaris tengah \qty{50}{\micro m} dengan dinding setebal
\qty{0.5}{\micro m} memegang \omterm{def:b1:eukaryotic-cell:plantcell}{turgor} \qty{0.8}{MPa}. Hitunglah tegangan
dindingnya ($\sigma = Pr/2t$ bagi sebuah bola) lalu bandingkan dengan
kekuatannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:carbohydrates:9}
Ketakcocokan laktosa: orang dewasa yang kekurangan laktase usus
mengalami kram dan diare sesudah minum susu. Jelaskan, dengan memakai
ikatan yang diputus laktase, apa yang terjadi pada laktosa yang tak
tercerna di dalam usus besarnya (bakteri, \omterm{def:b1:membranes-transport:osmosis}{osmosis}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:carbohydrates:10}
Seekor sapi memakan \qty{10}{kg} bahan kering rumput sehari, yang
\qty{40}{\%}-nya \omterm{def:b1:carbohydrates:polysaccharide}{selulosa}. Ia tidak memiliki selulase. Jelaskan
bagaimana ia tetap memperoleh energi dari \omterm{def:b1:carbohydrates:polysaccharide}{selulosanya}, mengapa
prosesnya harus berlangsung sebelum usus halusnya, lalu taksirlah
energi yang dipertaruhkan bila \qty{60}{\%} \omterm{def:b1:carbohydrates:polysaccharide}{selulosanya} difermentasi
dan menghasilkan \qty{12}{kJ} hasil yang berguna tiap
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:carbohydrates:11}
Antigen golongan darah A dan B berbeda pada satu gula yang ditambahkan
dua versi sebuah transferase; golongan O tidak memiliki keduanya.
Jelaskan mengapa seorang bergolongan O dapat menyumbangkan \omterm{def:b1:cell-unit-of-life:cell}{sel} merahnya
kepada siapa pun tetapi hanya dapat menerima dari O, dan apa yang harus
dimuat plasma seorang bergolongan A.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:carbohydrates:12}
``Glukosa adalah bahan bakar semesta, \omterm{def:b1:carbohydrates:polysaccharide}{selulosa} bahan bangunan semesta,
dan keduanya molekul yang sama.'' Bahaslah dalam satu paragraf: apa
yang benar dalam kalimat itu, apa yang diubah satu ikatan, dan apa yang
dikatakannya tentang hubungan antara kimia dan biologi.
\end{exercise}

\section{Soal: Pohon Glikogen}

\begin{problem}[{Soal akhir pekan --- sebuah partikel glikogen yang
dihitung cabang demi cabang, simpanan hati yang ditimbang dan
pelepasannya yang diukur waktunya, harga osmotik yang dihindari
pemolimeran, berakhir pada laju pelepasan glukosa oleh hati}]
\label{pb:b1:carbohydrates:1}
Sebuah partikel \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} adalah sebuah pohon rantai: sebuah rantai dalam
berisi \qty{13}{} residu glukosa membawa dua cabang, yang masing-masing
berisi \qty{13}{} residu dan masing-masing membawa dua cabang lagi,
begitu seterusnya selama \qty{12}{} tingkat; rantai tingkat terluarnya
tak bercabang. Tiap residu glukosa dalam polimernya bermassa
\qty{162}{g/mol}. Sebuah hati manusia yang beristirahat bermassa
\qty{1.5}{kg} memegang \qty{100}{g} \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} dalam \qty{1.0}{L} air
\omterm{def:b1:cell-unit-of-life:cell}{selnya}, dan melepas glukosa ke dalam darah sebanyak \qty{10}{g/h} di
antara waktu makan. \omterm{def:b1:carbohydrates:polysaccharide}{Glikogen} fosforilase menyingkirkan satu glukosa
dari sebuah ujung tak mereduksi tiap putarannya, pada \qty{20}{}
putaran tiap sekon tiap molekul enzimnya.

\medskip
\noindent\textbf{Bagian I --- Menghitung pohonnya.}
\begin{enumerate}
  \item Berapa banyak rantai yang dimuat tingkat $t$ ($t = 1$ bagi
        rantai dalamnya)? Berikan bilangannya bagi tingkat 1, 2, 3 dan
        12.
  \item Hitunglah total banyaknya rantai dalam partikelnya.
  \item Hitunglah total banyaknya residu glukosanya.
  \item Hitunglah massa partikelnya dalam dalton dan dalam gram.
  \item Berapa fraksi seluruh rantainya yang berada pada tingkat
        terluar? Berapa banyak ujung tak mereduksi yang disediakan
        partikel itu?
  \item Sebuah molekul amilosa dengan cacah residu yang sama memiliki
        berapa banyak ujung tak mereduksi?
  \item Bila tiap tingkat menambahkan \qty{1.9}{nm} pada jari-jarinya,
        hitunglah garis tengah partikelnya lalu bandingkan dengan
        sebuah ribosom
        (\qty{25}{nm}).
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Simpanan hatinya.}
\begin{enumerate}[resume]
  \item Hitunglah banyaknya partikel dalam hati itu.
  \item Hitunglah glukosa yang diwakili simpanannya dalam mol dan dalam
        gram glukosa bebas (\qty{180}{g/mol}).
  \item \omterm{def:b1:carbohydrates:polysaccharide}{Glikogen} mengikat \qty{3}{g} air tiap gram. Berapa massa hati
        itu yang berupa \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} ditambah airnya?
  \item Hitunglah laju pelepasan \qty{10}{g/h} dalam molekul glukosa
        tiap sekon.
  \item Berapa jam simpanannya bertahan pada laju itu? Permintaan \omterm{def:b1:mammal-organization:organ}{organ}
        mana yang terutama dilayani pelepasan itu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Ujung dan enzimnya.}
\begin{enumerate}[resume]
  \item Hitunglah total banyaknya ujung tak mereduksi dalam hati itu.
  \item Seandainya tiap ujung diserang sekaligus pada \qty{20}{} glukosa
        tiap sekon, berapa laju pelepasannya, dalam gram tiap sekon?
        Bandingkan dengan laju yang sesungguhnya.
  \item Berapa banyak molekul fosforilase, yang masing-masing bekerja
        pada \qty{20}{} tiap sekon, yang dibutuhkan bagi laju yang
        sesungguhnya?
  \item Andaikan \qty{100}{g} yang sama ditimbun sebagai rantai amilosa
        seukuran partikelnya. Hitunglah banyaknya ujung dan laju
        pelepasan maksimumnya.
  \item Jelaskan mengapa titik cabangnya adalah kunci tanggapan hati
        terhadap turunnya glukosa darah dalam beberapa menit.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Harga osmotiknya.}
\begin{enumerate}[resume]
  \item Seandainya \qty{100}{g} itu terlarut sebagai glukosa bebas
        dalam \qty{1.0}{L} air \omterm{def:b1:cell-unit-of-life:cell}{selnya}, osmolaritas berapa yang akan
        ditambahkannya?
  \item Bandingkan dengan \qty{0.3}{osmol/L} \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya} lalu ramalkan
        apa yang akan terjadi pada \omterm{def:b1:cell-unit-of-life:cell}{selnya}.
  \item Hitunglah osmolaritas yang disumbangkan partikel \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya}
        sendiri.
  \item Hitunglah \omterm{def:b1:membranes-transport:osmosis}{potensial air} yang akan diciptakan glukosa bebas itu
        ($\Psi_s = -RTc$, $RT = \qty{2.58}{MPa.L/mol}$) dan tekanan yang
        harus ditahan sebuah dinding bertipe tumbuhan.
  \item Otot menimbun \qty{400}{g} \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} dalam \qty{30}{kg} otot
        tetapi tidak sanggup melepas glukosa ke dalam darah. Ajukan
        alasannya, mengingat untuk apa \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} berada dalam sebuah otot.
  \item Sebutir kentang menimbun \omterm{def:b1:carbohydrates:polysaccharide}{pati} dalam butiran \qty{50}{\micro m},
        bukan dalam partikel \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} \qty{40}{nm}. Kaitkan
        perbedaannya dengan skala waktu penggerakan yang dibutuhkan tiap
        \omterm{def:b1:organism-environment:organism}{organismenya}.
  \item Sesudah sebuah santapan hati membangun kembali \qty{100}{g}
        \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} dalam empat jam. Hitunglah lajunya dalam residu
        glukosa tiap sekon dan biaya \omterm{def:b1:water-small-molecules:atp}{ATP-nya} bila menambahkan satu
        residu memakan biaya dua setara \omterm{def:b1:water-small-molecules:atp}{ATP}.
  \item Nyatakan hasilnya: laju pelepasan glukosa oleh hati dalam
        molekul tiap sekon, banyaknya ujung tak mereduksi yang
        memungkinkannya, dan osmolaritas yang dihindarkan polimernya
        dari \omterm{def:b1:cell-unit-of-life:cell}{selnya}.
\end{enumerate}
\end{problem}
