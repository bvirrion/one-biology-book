\chapter{Teeth and Eating Well}\label{ch:g2:teeth-and-eating-well}

Bite into an apple and look at the mark: a neat curved row of
little prints. Your teeth made that --- a full toolbox that cuts,
tears and grinds every meal before you swallow it. And like every
good toolbox, it deserves care, and good things to work on.

\section{The teeth: a toolbox in the mouth}

\begin{definition}[The three kinds of teeth]\label{def:g2:teeth-and-eating-well:kinds}
Human teeth come in three kinds, each with its job:
\begin{enumerate}
  \item the \emph{incisors}\index{incisor}, flat and sharp at the
        front, \emph{cut} --- they slice into the apple;
  \item the \emph{canines}\index{canine}, pointed, at the corners,
        \emph{tear} --- they grip and pull tougher food;
  \item the \emph{molars}\index{molar}, broad and bumpy at the
        back, \emph{grind} --- they crush food into a paste ready
        to swallow.
\end{enumerate}
\end{definition}

\begin{example}[One bite of apple, three tools]\label{ex:g2:teeth-and-eating-well:apple}
Eating a piece of apple uses the whole toolbox in order: the
\omterm{def:g2:teeth-and-eating-well:kinds}{incisors} slice off the bite, the \omterm{def:g2:teeth-and-eating-well:kinds}{canines} help pull it free, and
the tongue passes it back to the \omterm{def:g2:teeth-and-eating-well:kinds}{molars}, which grind it up. Only
then do you swallow.
\end{example}

\begin{remark}[Remember the animals]\label{rem:g2:teeth-and-eating-well:animals}
Your mixed toolbox tells a story: flat cutters and grinders like a
\omterm{def:g2:what-animals-eat:kinds}{herbivore}, pointed \omterm{def:g2:teeth-and-eating-well:kinds}{canines} like a \omterm{def:g2:what-animals-eat:kinds}{carnivore}. It is an \omterm{def:g2:what-animals-eat:kinds}{omnivore}'s
mouth --- humans are equipped for \omterm{def:g1:plants-around-us:plant}{plants} \emph{and} meat, just as
\cref{ch:g2:what-animals-eat} sorted us.
\end{remark}

\section{Baby teeth, adult teeth}

\begin{proposition}[Two sets in one life]\label{prop:g2:teeth-and-eating-well:sets}
Humans grow two sets of teeth. The $20$ \emph{baby
teeth}\index{baby teeth} arrive in the first years of life. From
about age six they loosen and fall out one by one, pushed from
below by the \emph{adult teeth}\index{adult teeth} --- the final
set, up to $32$ of them, which must last the rest of your life.
\end{proposition}
\admitted

\begin{example}[The wobbly incisor]\label{ex:g2:teeth-and-eating-well:wobbly}
A front tooth starts to wobble at the very age children start
school. Under it, an adult \omterm{def:g2:teeth-and-eating-well:kinds}{incisor} has been quietly growing and
dissolving the baby tooth's \omterm{def:g1:plants-around-us:parts}{root}, until the little tooth lets go.
The gap is not an accident --- it is a changeover.
\end{example}

\begin{remark}[No third set]\label{rem:g2:teeth-and-eating-well:nothird}
After the adult set, no new teeth will ever come. A lost baby
tooth is replaced; a lost adult tooth is lost. That is the real
reason grown-ups insist on brushing: the second toolbox must last
seventy years or more.
\end{remark}

\section{Eating well}

\begin{proposition}[A balanced plate]\label{prop:g2:teeth-and-eating-well:balanced}
To grow and stay healthy, the body needs food from each family,
every day:
\begin{itemize}
  \item fruit and vegetables --- for vitamins, at every meal;
  \item bread, cereals, potatoes --- for lasting energy;
  \item \omterm{def:g2:animals-grow-up:born}{milk}, yogurt, cheese --- for building bones and teeth;
  \item meat, fish or \omterm{def:g2:animals-grow-up:born}{eggs} --- for building the body, once or
        twice a day is enough;
  \item water --- the only drink the body truly needs;
  \item sugary and fatty treats --- for pleasure, in small
        amounts, not at every meal.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Two snack bags]\label{ex:g2:teeth-and-eating-well:snack}
Snack A: an apple, a piece of bread, water. Snack B: sweets, a
soda. Snack B tastes of sugar and little else --- and sugar is
just what the \omterm{def:g1:healthy-habits:germs}{germs} of \cref{ch:g1:healthy-habits} feast on.
Snack A feeds the body and gives the teeth honest work. Guess
which one the dentist packs for her own children.
\end{example}

\begin{method}[Checking a day's menu]\label{met:g2:teeth-and-eating-well:check}
Take a day's meals and check:
\begin{enumerate}
  \item fruit or vegetables at each meal?
  \item something from the building families --- dairy, and meat,
        fish or \omterm{def:g2:animals-grow-up:born}{eggs}?
  \item water as the everyday drink?
  \item treats kept small and occasional?
\end{enumerate}
If a family is missing, the next meal is the chance to invite it
back.
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:1}
Name the three kinds of teeth and the job of each.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:2}
Which teeth are at the front of the mouth, and which at the back?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:3}
Follow a bite of carrot through the mouth: which tooth does what,
in order?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:4}
How many \omterm{prop:g2:teeth-and-eating-well:sets}{baby teeth} are there? What happens to them from about
age six?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:5}
Why must \omterm{prop:g2:teeth-and-eating-well:sets}{adult teeth} be cared for even more than \omterm{prop:g2:teeth-and-eating-well:sets}{baby teeth}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:6}
Name the food families a healthy day should include. Which drink
does the body truly need?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:7}
Compare the two snacks of \cref{ex:g2:teeth-and-eating-well:snack}:
what does snack A give the body that snack B does not?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:teeth-and-eating-well:8}
Try it: look at your own teeth in a mirror. Find an \omterm{def:g2:teeth-and-eating-well:kinds}{incisor}, a
\omterm{def:g2:teeth-and-eating-well:kinds}{canine} and a \omterm{def:g2:teeth-and-eating-well:kinds}{molar}, and check their shapes against the figure.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:teeth-and-eating-well:9}
Using \cref{rem:g2:teeth-and-eating-well:animals}, explain what
the shapes of human teeth say about the human menu --- and which
\omterm{def:g1:animals-around-us:animal}{animals} of \cref{ch:g2:what-animals-eat} we resemble.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:teeth-and-eating-well:10}
Lila's day: breakfast --- cereal and \omterm{def:g2:animals-grow-up:born}{milk}; lunch --- pasta and
sweets; snack --- soda and biscuits; supper --- pizza. Run
\cref{met:g2:teeth-and-eating-well:check} on it: what is missing,
and what appears too often?
\end{exercise}
