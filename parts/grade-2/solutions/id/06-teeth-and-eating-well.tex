\section*{Bab \ref{ch:g2:teeth-and-eating-well} --- Gigi dan Makan dengan Baik}

\begin{solution}{exo:g2:teeth-and-eating-well:1}
\omterm{def:g2:teeth-and-eating-well:kinds}{Gigi seri} memotong; \omterm{def:g2:teeth-and-eating-well:kinds}{gigi taring} merobek; \omterm{def:g2:teeth-and-eating-well:kinds}{gigi geraham} menggilas.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:2}
\omterm{def:g2:teeth-and-eating-well:kinds}{Gigi serinya} ada di depan (dengan \omterm{def:g2:teeth-and-eating-well:kinds}{gigi taring} di sudut-sudut di
sebelahnya); \omterm{def:g2:teeth-and-eating-well:kinds}{gigi gerahamnya} ada di belakang.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:3}
\omterm{def:g2:teeth-and-eating-well:kinds}{Gigi seri} mengiris gigitannya; \omterm{def:g2:teeth-and-eating-well:kinds}{gigi taring} menolong mencengkeram dan
menariknya lepas; lidah mengantarnya ke belakang; \omterm{def:g2:teeth-and-eating-well:kinds}{gigi geraham}
menggilasnya menjadi bubur; lalu ia ditelan.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:4}
Ada $20$ \omterm{prop:g2:teeth-and-eating-well:sets}{gigi susu}. Mulai kira-kira umur enam tahun, gigi itu goyah
lalu tanggal satu demi satu, didorong dari bawah oleh \omterm{prop:g2:teeth-and-eating-well:sets}{gigi tetap} yang
sedang tumbuh.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:5}
Karena \omterm{prop:g2:teeth-and-eating-well:sets}{gigi tetap} adalah perangkat terakhir: \omterm{prop:g2:teeth-and-eating-well:sets}{gigi susu} yang tanggal
digantikan oleh \omterm{prop:g2:teeth-and-eating-well:sets}{gigi tetap}, tetapi \omterm{prop:g2:teeth-and-eating-well:sets}{gigi tetap} yang hilang tidak pernah
digantikan. Perangkat tetapnya harus bertahan seumur hidup.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:6}
Buah dan sayur; roti dan serealia; produk \omterm{def:g2:animals-grow-up:born}{susu} (\omterm{def:g2:animals-grow-up:born}{susu}, yogurt, keju);
daging, ikan atau \omterm{def:g2:animals-grow-up:born}{telur}; dengan kudapan manis yang hanya sedikit dan
sesekali. Satu-satunya minuman yang sungguh dibutuhkan tubuh adalah
air.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:7}
Vitamin dari apelnya, tenaga yang bertahan lama dari rotinya, dan air
--- zat makanan yang sesungguhnya, serta pekerjaan mengunyah yang
jujur bagi giginya. Bekal B kebanyakan memberi gula, yang justru
memberi makan \omterm{def:g1:healthy-habits:germs}{kuman} perusak gigi.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:8}
Jawaban terbuka --- di cermin: \omterm{def:g2:teeth-and-eating-well:kinds}{gigi seri} yang pipih dan bertepi tajam
di depan, sebuah \omterm{def:g2:teeth-and-eating-well:kinds}{gigi taring} yang runcing di tiap sudut, \omterm{def:g2:teeth-and-eating-well:kinds}{gigi geraham}
yang lebar dan berbenjol di belakang, sesuai dengan gambarnya.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:9}
\omterm{def:g2:teeth-and-eating-well:kinds}{Gigi seri} pemotong yang pipih dan \omterm{def:g2:teeth-and-eating-well:kinds}{gigi geraham} penggilas adalah
perkakas bergaya \omterm{def:g2:what-animals-eat:kinds}{herbivora}; \omterm{def:g2:teeth-and-eating-well:kinds}{gigi taring} yang runcing bergaya
\omterm{def:g2:what-animals-eat:kinds}{karnivora}. Memiliki keduanya berarti daftar makanan yang memuat
\omterm{def:g1:plants-around-us:plant}{tumbuhan} sekaligus daging: mulut seorang \omterm{def:g2:what-animals-eat:kinds}{omnivora}, seperti daftar
makanan burung hitam dan babi hutan pada bab itu.
\end{solution}

\begin{solution}{exo:g2:teeth-and-eating-well:10}
Yang hilang: buah dan sayur (tidak ada seharian!) serta air sebagai
minuman sehari-hari; daging, ikan atau \omterm{def:g2:animals-grow-up:born}{telur} juga tidak ada. Yang
terlalu sering: hal-hal yang manis --- permen, minuman bersoda,
biskuit. Satu perbaikan: buah saat sarapan dan kudapan, sayur serta
sedikit ikan atau \omterm{def:g2:animals-grow-up:born}{telur} pada makan siang atau makan malam, air sebagai
ganti minuman bersoda, kudapan manis dibatasi satu saja.
\end{solution}
