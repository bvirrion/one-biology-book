\chapter{Las pruebas de la evolución}\label{ch:g9:evidence-of-evolution}

Dos veces se ha parado este libro al borde de una afirmación enorme:
las \omterm{def:g1:plants-around-us:tree}{ramas} del \omterm{def:g6:classification-kinship:tree}{árbol del parentesco} se juntan todas
(\cref{rem:g6:classification-kinship:implies}) y todos los \omterm{def:g6:exploring-our-environment:components}{seres vivos}
leen un mismo código (\cref{rem:g9:genes-and-dna:universal}). La
afirmación a la que apuntan tiene nombre --- la \emph{evolución}: las
\omterm{def:g5:biodiversity:species}{especies} cambian a lo largo de las generaciones y todas descienden de
antepasados comunes. Antes de que el capítulo siguiente pregunte
\emph{cómo}, este hace lo que exigía el
\cref{exo:g6:classification-kinship:12}: expone las pruebas.

\section{El testimonio de las rocas}

\begin{definition}[Los fósiles]\label{def:g9:evidence-of-evolution:fossils}
Un \emph{fósil}\index{fósil} es un rastro conservado de vida pasada ---
una \omterm{def:g1:animals-around-us:covering}{concha}, un \omterm{def:g3:bones-and-muscles:skeleton}{hueso}, la huella de una \omterm{def:g1:plants-around-us:parts}{hoja}, una pisada --- sellado en
roca sedimentaria. Las capas de roca se apilan en orden de tiempo, la
más vieja abajo; leer hacia arriba es leer la historia. Los fósiles son
la mitad de restos del inventario de la
\cref{def:g6:exploring-our-environment:components}, guardada durante
millones de años.
\end{definition}

\begin{proposition}[Lo que enseña el registro en capas]\label{prop:g9:evidence-of-evolution:record}
Leído en orden, el registro testifica tres veces:
\begin{enumerate}
  \item \emph{las faunas cambian}: cada edad de capa tiene su propio
        reparto --- ni conejos entre los trilobites ni trilobites en
        las capas de los conejos; la mayoría de las \omterm{def:g5:biodiversity:species}{especies} de
        cualquier capa honda están \emph{\omterm{prop:g5:biodiversity:loss}{extinguidas}} (la palabra de la
        \cref{prop:g5:biodiversity:loss}, ahora con el tiempo profundo
        detrás);
  \item \emph{el orden encaja como el árbol}: los \omterm{prop:g3:sorting-living-things:groups}{peces} aparecen en las
        capas hondas, los parientes \omterm{prop:g3:sorting-living-things:groups}{anfibios} de cuatro patas por encima
        de ellos, los parientes \omterm{prop:g4:classifying-animals:classes}{reptiles} por encima de esos, y los
        \omterm{prop:g3:sorting-living-things:groups}{mamíferos} y las \omterm{prop:g3:sorting-living-things:groups}{aves} más arriba todavía --- las cajas de la
        \omterm{def:g4:classifying-animals:classification}{clasificación} (\cref{pb:g6:classification-kinship:1}) se apilan
        en el tiempo exactamente como predice el \omterm{prop:g6:classification-kinship:kinship}{parentesco};
  \item \emph{existen los intermedios}: la famosa losa del
        \emph{Archaeopteryx} --- \omterm{def:g1:animals-around-us:covering}{plumas} y alas sobre un armazón de
        reptil con dientes y con la cola larga y ósea --- está justo
        donde se separan la \omterm{def:g1:plants-around-us:tree}{rama} de las \omterm{prop:g3:sorting-living-things:groups}{aves} y la de los \omterm{prop:g4:classifying-animals:classes}{reptiles}; unas
        ballenas \omterm{def:g9:evidence-of-evolution:fossils}{fósiles} con patas traseras que funcionaban están donde
        las exigía la ascendencia de mamífero terrestre del
        \cref{ex:g6:classification-kinship:whale}.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[La serie de las ballenas]\label{ex:g9:evidence-of-evolution:whales}
El expediente de la ballena, rellenado por las rocas: unos \omterm{def:g9:evidence-of-evolution:fossils}{fósiles}
sucesivos, en orden de tiempo, van desde un mamífero costero de cuatro
patas del tamaño de un lobo, pasando por formas con las patas cada vez
más pequeñas y la cola cada vez más fuerte, hasta unos nadadores
completos sin patas --- mientras que la ballena de hoy todavía
construye unos huesecillos escondidos de cadera y de pata
(el \cref{ex:g6:classification-kinship:placing} señaló los vestigios) y
todavía amamanta, respira aire y cría \omterm{def:g1:animals-around-us:covering}{pelo} fetal. Predicción, \omterm{def:g9:evidence-of-evolution:fossils}{fósiles} y
vestigios: tres llaves, una cerradura.
\end{example}

\section{El testimonio de los cuerpos vivos}

\begin{proposition}[La anatomía se acuerda]\label{prop:g9:evidence-of-evolution:anatomy}
Los cuerpos vivos llevan encima su historia:
\begin{itemize}
  \item \emph{el plan compartido}: el ala del murciélago, la aleta de
        la ballena, la pata del caballo y tu \omterm{def:g1:my-body:parts}{brazo} son una misma
        disposición de \omterm{def:g3:bones-and-muscles:skeleton}{huesos} (la horquilla de las cuatro \omterm{def:g1:my-body:parts}{extremidades}
        del \cref{pb:g6:classification-kinship:1}) estirada, acortada y
        recolocada en otro oficio --- la \omterm{def:g9:heredity-and-traits:heredity}{herencia} explica lo que una
        ingeniería nueva no repetiría jamás;
  \item \emph{los vestigios}: las caderas de la ballena, las alas
        selladas del escarabajo que no vuela, tu propio cóccix y el
        estado encogido del apéndice --- piezas conservadas porque la
        ascendencia construye encima, no desde cero;
  \item \emph{los embriones coinciden}: los embriones de \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} de
        un \omterm{prop:g3:sorting-living-things:groups}{pez}, de un pollo y de un ser humano corren asombrosamente
        parecidos por sus primeras semanas (el trazado del plan de la
        \cref{prop:g8:from-egg-to-birth:timetable}) --- instrucciones
        de obra compartidas, que divergen tarde.
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[Las moléculas votan]\label{prop:g9:evidence-of-evolution:molecules}
El testigo más nuevo es el código de la
\cref{rem:g9:genes-and-dna:universal}. Compara la ortografía de un \omterm{def:g9:genes-and-dna:gene}{gen}
bien conservado entre \omterm{def:g5:biodiversity:species}{especies}: el ser humano y el chimpancé difieren
en pocas letras, el ser humano y el ratón en más, el ser humano y un
\omterm{prop:g3:sorting-living-things:groups}{pez} en más todavía, y el ser humano y la \omterm{ex:g6:producing-our-food:bread}{levadura} en muchísimas más ---
y las distancias, sumadas \omterm{def:g9:genes-and-dna:gene}{gen} tras \omterm{def:g9:genes-and-dna:gene}{gen}, vuelven a dibujar el
\emph{mismo árbol} que dibujaron los anatomistas con los \omterm{def:g3:bones-and-muscles:skeleton}{huesos} y los
paleontólogos con las capas. Tres testigos independientes, un solo
árbol de familia: ese acuerdo es la razón de que los biólogos dejaran
de discutir el \emph{si} y se volvieran hacia el \emph{cómo}.
\end{proposition}
\admitted

\begin{example}[Verlo moverse]\label{ex:g9:evidence-of-evolution:live}
El cambio hasta se puede mirar allí donde las generaciones pasan
deprisa. La forma oscura de la mariposa del abedul se extendió por los
\omterm{prop:g5:people-and-nature:managed}{bosques} industriales manchados de hollín y retrocedió cuando el aire se
limpió; los \omterm{prop:g3:sorting-living-things:groups}{insectos} a prueba de insecticidas y los \omterm{def:g6:producing-our-food:fermentation}{microbios} que
desafían a los antibióticos (el
\cref{ch:g9:microbes-and-infection} se topará con el caso médico)
aparecen en cuestión de años; y en las islas, la distribución de
tamaños del pico de los pinzones se desplaza con las sequías y las
épocas de abundancia, medida en vivo con anillos y calibres.
Movimientos pequeños, tiempos cortos --- el mecanismo pillado
funcionando, sea cual sea su alcance a la larga.
\end{example}

\begin{method}[Pesar unas pruebas históricas]\label{met:g9:evidence-of-evolution:weigh}
La evolución es historia, y las afirmaciones históricas se ponen a
prueba como el caso sin resolver del
\cref{pb:g9:unique-individuals:1} --- por la convergencia de testigos
independientes:
\begin{enumerate}
  \item pregunta qué \emph{predice} la afirmación: qué intermedios, qué
        vestigios, qué distancias moleculares;
  \item comprueba cada testigo por separado: las rocas, los cuerpos,
        las moléculas, lo que se mira en vivo;
  \item pesa su independencia: unos testigos que no han podido
        confabularse, si coinciden, multiplican la credibilidad;
  \item deja el veredicto abierto al próximo hallazgo --- y fíjate
        cuando los hallazgos siguen cayendo donde la afirmación decía
        que había que cavar (las patas de las ballenas se buscaron, y
        se encontraron).
\end{enumerate}
\end{method}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:7}
¿Por qué argumenta \omterm{def:g9:heredity-and-traits:heredity}{herencia}, y no un buen diseño reutilizado, el plan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:8}
Monta el expediente completo de la ballena: \omterm{def:g9:heredity-and-traits:traits}{rasgos} de clase,

uno.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:9}
¿Por qué cuenta más el \emph{acuerdo} de las rocas, los cuerpos y las
moléculas que cualquiera de los testigos por separado? Usa el paso de

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:10}
``Ni un conejo entre los trilobites''. ¿Por qué es esa ausencia ---
sostenida en todas las excavaciones que se han hecho jamás --- una

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:11}

de qué capítulo dice por qué importan más las instrucciones tempranas
compartidas?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:evidence-of-evolution:12}
Escribe el argumento del capítulo como el
\cref{met:g9:evidence-of-evolution:weigh} aplicado del todo:
afirmación, predicciones, cuatro testigos, veredicto --- y la pregunta
que le entrega el veredicto al \cref{ch:g9:how-species-change}.
\end{exercise}

\section{Problema: El ala nueva del museo}

\begin{problem}[{Problema de fin de semana --- comisariar la
sala de las pruebas}]\label{pb:g9:evidence-of-evolution:1}
El museo de historia natural construye un ala sobre la evolución, y la
clase comisaría cuatro salas: Rocas, Cuerpos, Moléculas y Cambios en
vivo.

\medskip\noindent\textbf{Parte I --- Sala por sala.}
\begin{enumerate}
  \item Rocas: ordena la pared --- losa de trilobites, primeros \omterm{prop:g3:sorting-living-things:groups}{peces},
        rastros de parientes \omterm{prop:g3:sorting-living-things:groups}{anfibios}, \omterm{def:g3:bones-and-muscles:skeleton}{esqueletos} de reptil, molde de
        \emph{Archaeopteryx} y mandíbula de mamífero temprano --- y
        escribe el cartel de una frase de la sala a partir de la
        \cref{prop:g9:evidence-of-evolution:record}.
  \item Cuerpos: la pieza central despliega el ala del murciélago, la
        aleta de la ballena, la pata del caballo y el \omterm{def:g1:my-body:parts}{brazo} humano
        desde un mismo soporte de \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto}. Ponle cartel al plan
        compartido --- y añade las tres piezas de la vitrina de los
        vestigios.
  \item Moléculas: el cuadro de la pared suma las diferencias de letras
        de un \omterm{def:g9:genes-and-dna:gene}{gen}: chimpancé 1, ratón 8, \omterm{prop:g3:sorting-living-things:groups}{pez} 24, \omterm{ex:g6:producing-our-food:bread}{levadura} 41 (cuentas a
        modo de ejemplo). ¿Con qué tiene que coincidir el árbol del
        cuadro para que la sala funcione?
  \item Cambios en vivo: elige dos piezas del
        \cref{ex:g9:evidence-of-evolution:live} y di qué enseña
        moviéndose cada una.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Las preguntas de los
visitantes.}
\begin{enumerate}[resume]
  \item ``¿Por qué sigue habiendo monos, si descendemos de antepasados
        compartidos con ellos?''. Responde con la ramificación del
        árbol --- primos, no escaleras
        (\cref{def:g6:classification-kinship:tree}).
  \item ``¿Dónde están los eslabones perdidos?''. Señala los
        encontrados --- dos de este capítulo --- y di entre qué estaba
        cada uno.
  \item ``¿No es la evolución solo una teoría?''. Responde con el
        método de las pruebas históricas: qué quiere decir aquí
        ``teoría'' y qué establece el acuerdo de las cuatro salas.
  \item ``¿Podría echarlo abajo la excavación de mañana?''. Di qué
        clase de hallazgo lo haría --- conejos entre los trilobites ---
        y qué ha significado su ausencia, excavación tras excavación.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- El panel de cierre
del ala.}
\begin{enumerate}[resume]
  \item Redacta el panel que une el ala con la sala de la \omterm{def:g4:classifying-animals:classification}{clasificación}
        de al lado: por qué las cajas encajadas de siempre eran el
        árbol desde el principio.
  \item Añade el párrafo de las moléculas: qué dice sobre el tronco el
        código compartido.
  \item Añade el párrafo honrado: qué enseña el ala (cambio, \omterm{prop:g6:classification-kinship:kinship}{parentesco}
        e intermedios) y qué no explica todavía --- nombrando la
        pregunta del mecanismo que recogerá la sala siguiente.
  \item Cierra con una frase que se vaya a acordar alguien de doce
        años, y que contenga \emph{familia} y \emph{tiempo}.
\end{enumerate}
\end{problem}
