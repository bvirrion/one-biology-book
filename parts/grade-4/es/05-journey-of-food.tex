\chapter{El viaje de los alimentos}\label{ch:g4:journey-of-food}

Un mordisco de pan: masticado, tragado --- desaparecido. ¿Adónde? Esta
noche, una parte de ese pan estará ayudando a correr a tus \omterm{def:g1:my-body:parts}{piernas} y a
crecer a tus \omterm{def:g3:bones-and-muscles:skeleton}{huesos}, lo que significa que de alguna manera tiene que
\emph{salir} del tubo por el que se tragó y \emph{entrar} en el resto
del cuerpo. Sigue el mordisco: su viaje es una de las mejores
historias del cuerpo.

\section{El tubo digestivo}

\begin{definition}[El tubo digestivo]\label{def:g4:journey-of-food:tract}
Los alimentos viajan por el \emph{tubo digestivo}\index{tubo
digestivo}, un tubo largo que atraviesa el cuerpo entero:
\begin{enumerate}
  \item la \emph{boca}\index{boca}, donde los dientes trituran
        (\cref{ch:g2:teeth-and-eating-well}) y la
        \emph{saliva}\index{saliva} humedece;
  \item el \emph{esófago}\index{esófago}, el tobogán que va de la
        garganta al estómago;
  \item el \emph{estómago}\index{estómago}, una bolsa musculosa donde
        el alimento se revuelve y se mezcla durante horas con los \omterm{def:g4:journey-of-food:digestion}{jugos
        digestivos};
  \item el \emph{intestino delgado}\index{intestino delgado}, un tubo
        plegado de varios metros donde termina la \omterm{def:g4:journey-of-food:digestion}{digestión} y donde lo
        bueno del alimento pasa al cuerpo;
  \item el \emph{intestino grueso}\index{intestino grueso}, donde se
        recupera el agua de lo que queda, antes de que las sobras
        salgan del cuerpo.
\end{enumerate}
\end{definition}

\begin{example}[Cronometrar el viaje]\label{ex:g4:journey-of-food:timing}
El mordisco de pan pasa unos segundos en la boca, un par de segundos
bajando por el \omterm{def:g4:journey-of-food:tract}{esófago}, unas horas revolviéndose en el \omterm{def:g4:journey-of-food:tract}{estómago}, varias
más abriéndose paso por el \omterm{def:g4:journey-of-food:tract}{intestino delgado} y hasta un día o más
terminando en el \omterm{def:g4:journey-of-food:tract}{intestino grueso}. El viaje entero suele durar uno o
dos días.
\end{example}

\section{La digestión: de los alimentos a los nutrientes}

\begin{definition}[Digestión y nutrientes]\label{def:g4:journey-of-food:digestion}
La \emph{digestión}\index{digestión} es la transformación de los
alimentos, a lo largo del tubo, en componentes lo bastante pequeños y
sencillos para entrar en el cuerpo: los
\emph{nutrientes}\index{nutriente}. Los dientes y el revuelto del
\omterm{def:g4:journey-of-food:tract}{estómago} rompen el alimento; los \emph{jugos digestivos}\index{jugo
digestivo} --- entre ellos la \omterm{def:g4:journey-of-food:tract}{saliva} --- lo desmontan aún más, pieza a
pieza, hasta que lo que era pan es una mezcla líquida de nutrientes.
\end{definition}

\begin{example}[El pan se vuelve dulce]\label{ex:g4:journey-of-food:sweet}
Mastica un trozo de pan solo durante mucho rato --- una cuenta lenta
hasta sesenta --- y aparece un dulzor leve. No se le añadió nada: la
\omterm{def:g4:journey-of-food:tract}{saliva} ya está desmontando el pan en trozos más sencillos y más
dulces. Acabas de saborear el principio de la \omterm{def:g4:journey-of-food:digestion}{digestión}.
\end{example}

\begin{proposition}[Pasar al interior del cuerpo]\label{prop:g4:journey-of-food:absorb}
El interior del tubo está todavía, en cierto sentido, fuera del cuerpo
--- como el agujero de un dónut. El paso ocurre en el \emph{\omterm{def:g4:journey-of-food:tract}{intestino
delgado}}: a través de sus paredes finas y muy plegadas, los \omterm{def:g4:journey-of-food:digestion}{nutrientes}
pasan a la \emph{sangre}\index{sangre}, que los lleva a todas las
partes del cuerpo (el servicio de reparto que explora el
\cref{ch:g4:heart-and-blood}). Lo que no puede pasar --- fibras,
cáscaras --- sigue su camino, pierde el agua en el \omterm{def:g4:journey-of-food:tract}{intestino grueso} y
sale en forma de excrementos.
\end{proposition}
\admitted

\begin{remark}[Por qué los pliegues y los metros]\label{rem:g4:journey-of-food:surface}
Pasar exige superficie de pared, y el \omterm{def:g4:journey-of-food:tract}{intestino delgado} está hecho
para eso: metros de longitud y paredes plegadas y replegadas por
dentro. Estirada, la superficie de paso alfombraría una habitación
pequeña --- y va metida en tu barriga. En el cuerpo, los trabajos
grandes tienen superficies grandes plegadas en poco sitio.
\end{remark}

\section{Ayudar a la digestión a hacer su trabajo}

\begin{method}[Costumbres que ayudan a la digestión]\label{met:g4:journey-of-food:habits}
\begin{enumerate}
  \item mastica bien: el \omterm{def:g4:journey-of-food:tract}{estómago} no tiene dientes --- lo que se salta
        la boca, el \omterm{def:g4:journey-of-food:tract}{estómago} tiene que pelearlo;
  \item come con calma, a horas regulares;
  \item bebe agua en las comidas;
  \item come alimentos ricos en fibra --- verdura, fruta, pan integral:
        la fibra mantiene en marcha las sobras;
  \item deja el juego fuerte o la piscina para cuando la comida se haya
        asentado: un \omterm{def:g4:journey-of-food:tract}{estómago} que trabaja duro y unas \omterm{def:g1:my-body:parts}{piernas} que
        trabajan duro compiten.
\end{enumerate}
\end{method}

\begin{example}[El detective del dolor de tripa]\label{ex:g4:journey-of-food:ache}
Comida engullida y después un esprint: el clásico dolor de tripa de la
tarde. La comida se tragó a medio masticar (regla 1 rota) y al
\omterm{def:g4:journey-of-food:tract}{estómago}, que estaba digiriendo, lo metieron con prisas en una carrera
(regla 5). El arreglo no es una medicina, sino buenos modales --- con
el propio \omterm{def:g4:journey-of-food:tract}{tubo digestivo}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:1}
Nombra las partes del \omterm{def:g4:journey-of-food:tract}{tubo digestivo} en orden, de la boca a la salida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:2}
¿Qué les pasa a los alimentos en el \omterm{def:g4:journey-of-food:tract}{estómago} y cuánto tiempo se quedan
allí, más o menos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:3}
¿Qué es la \omterm{def:g4:journey-of-food:digestion}{digestión}? ¿Qué son los \omterm{def:g4:journey-of-food:digestion}{nutrientes}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:4}
¿En qué parte pasan los \omterm{def:g4:journey-of-food:digestion}{nutrientes} a la \omterm{prop:g4:journey-of-food:absorb}{sangre}? ¿Qué pasa con lo que
no puede pasar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:5}
¿Por qué se vuelve levemente dulce el pan muy masticado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:6}
¿Por qué es el \omterm{def:g4:journey-of-food:tract}{intestino delgado} tan largo y tan plegado por dentro?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:7}
Da tres costumbres que ayudan a la \omterm{def:g4:journey-of-food:digestion}{digestión} y la razón de cada una.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:journey-of-food:8}
Pruébalo (en la mesa): cronometra tu próxima comida. ¿Cuánto tiempo
pasa tu comida en tu boca --- y cómo se compara eso con los días que
tiene por delante?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:9}
Diagnostica el dolor de tripa del \cref{ex:g4:journey-of-food:ache}:
¿qué dos costumbres se rompieron y qué cambiarías mañana?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:journey-of-food:10}
``La comida entra en el cuerpo en el momento en que te la tragas''.
Corrígelo con la \cref{prop:g4:journey-of-food:absorb}: ¿dónde entra
de verdad la comida en el cuerpo y qué le tiene que pasar antes?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:journey-of-food:11}
La hierba de la vaca es un alimento mucho más duro que tu pan --- y la
vaca tiene cuatro cavidades en el \omterm{def:g4:journey-of-food:tract}{estómago}, mastica su comida dos
veces y aloja en sus tripas \omterm{def:g1:living-or-not:living}{seres vivos} que la ayudan a desmontar la
hierba. Con la \cref{def:g4:journey-of-food:digestion} y la
\cref{prop:g2:what-animals-eat:match}, explica por qué un comedor de
hierba necesita esta maquinaria más pesada.
\end{exercise}
