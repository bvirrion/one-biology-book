\chapter{التركيب الكيميائي للكائنات الحية}\label{ch:g10:chemistry-of-life}

اترك شريحة خبز على الشواية أكثر مما ينبغي فتسودّ: وما يبقى كربون في
معظمه. زِن شريحة تفاح طازجة، ثم جففها يومًا في فرن دافئ، وزِنها من
جديد: خمسة أسداس كتلتها غادرت بخار ماء. الخبز والتفاحة والبلوطة
وقنديل البحر وأنت --- كل كائن حي مبني، في أساسه، من القائمة القصيرة
نفسها من المكونات الكيميائية، وليست تلك هي قائمة ما تتكون منه الصخور.
يفكك هذا الفصل المادة الحية، مكونًا مكونًا، ويسأل ما الذي يجعلها
متميزة.

\section{العناصر: ذرات الصخور نفسها، بنسب مختلفة}

\begin{definition}[التركيب الكيميائي للمادة الحية]\label{def:g10:chemistry-of-life:composition}
\emph{التركيب الكيميائي}\index{تركيب كيميائي} لعينة من المادة الحية
هو قائمة العناصر الكيميائية التي تحتويها، مع نسبة كتلة العينة التي
يمثلها كل عنصر منها. ويقاس بتفكيك العينة كيميائيًا --- بحرقها، ووزن
ما يتبخر وما يبقى، وتحليل الغازات المنطلقة --- لا بالنظر إليها.
\end{definition}

\begin{proposition}[أربعة عناصر تؤلف 96\% من الجسم الحي]\label{prop:g10:chemistry-of-life:four}
المادة الحية مؤلفة، من حيث الكتلة، من أربعة عناصر تكاد تكون وحدها:
الأكسجين (O) والكربون (C) والهيدروجين (H) والنيتروجين (N). وهي تمثل
في جسم الإنسان نحو 65\% و18.5\% و9.5\% و3.2\% من الكتلة --- أي 96\%
مجتمعة. ويتقاسم الكالسيوم والفوسفور والبوتاسيوم والكبريت والصوديوم
والكلور والمغنيزيوم معظم النسبة الباقية البالغة 4\%، وهناك نحو اثني
عشر \emph{عنصرًا نزرًا} (حديد وزنك ويود ونحاس\dots) بنسب تقل كثيرًا
عن 0.1\% ولا غنى عنها مع ذلك. والعناصر الأربعة نفسها تسود في بكتيريا
أو نبتة قمح أو حوت، بنسب متقاربة.
\end{proposition}
\admitted

\begin{example}[المادة الحية والصخر]\label{ex:g10:chemistry-of-life:crust}
قشرة الأرض هي أيضًا 46\% أكسجين، لكن عناصرها التالية هي السيليسيوم
(28\%) والألمنيوم (8\%) والحديد (5.6\%). ويؤلف الكربون 0.02\% من
القشرة والنيتروجين 0.002\%، مقابل 18.5\% و3.2\% من الجسم. فالكائن
الحي إذن ليس عينة من محيطه: إنه \emph{يركز} عناصر قليلة بعامل ألف أو
أكثر. وهذا الانتقاء هو أول توقيع كيميائي للحياة.
\end{example}

\begin{remark}[من أين تأتي الذرات]\label{rem:g10:chemistry-of-life:origin}
كل ذرة من تلك الذرات ذرة عادية: فكربون عضلاتك كان ثاني أكسيد كربون في
الهواء قبل أن يثبته نبات، ونيتروجين بروتيناتك كان نيتروجين الهواء قبل
أن تلتقطه بكتيريا التربة. وليس الخاص في الحياة هو الذرات، بل طريقة
تركيبها والنسب التي تجمع بها.
\end{remark}

\section{الماء والأملاح المعدنية}

\begin{definition}[المادة المعدنية والمادة العضوية]\label{def:g10:chemistry-of-life:mineral-organic}
تنقسم جزيئات الكائن الحي إلى زمرتين. \emph{المادة
المعدنية}\index{مادة معدنية} مؤلفة من جزيئات توجد كذلك خارج الكائنات
الحية: الماء و\emph{الأملاح المعدنية}\index{أملاح معدنية} المنحلة
(شوارد مثل $\mathrm{Na^+}$ و$\mathrm{K^+}$ و$\mathrm{Ca^{2+}}$
و$\mathrm{Cl^-}$، والفوسفات وثنائي الكربونات). أما \emph{المادة
العضوية}\index{مادة عضوية} فمؤلفة من جزيئات مبنية على هيكل من ذرات
كربون مرتبطة بالهيدروجين، وبالأكسجين والنيتروجين عادة --- وهي جزيئات
لا تصنعها على الأرض إلا الكائنات الحية أو ما خلفته.
\end{definition}

\begin{proposition}[الماء هو المكون الرئيسي]\label{prop:g10:chemistry-of-life:water}
الماء هو أوفر جزيء في كل كائن حي: نحو 60\% إلى 65\% من كتلة الإنسان
البالغ، و75\% من كتلة الوليد، و85\% من كتلة التفاحة، و95\% من كتلة
قنديل البحر، لكنه 13\% فقط من حبة قمح ساكنة. ويقاس محتوى عينة من
الماء بوزنها، ثم تجفيفها عند \qty{105}{\celsius} حتى تكف كتلتها عن
التناقص، ثم وزنها من جديد: والكتلة المفقودة هي الماء.
\end{proposition}
\admitted

\begin{example}[تجفيف شريحة تفاح]\label{ex:g10:chemistry-of-life:apple}
شريحة تفاح طازجة كتلتها \qty{24.0}{g} تجفف حتى تستقر كتلتها عند
\qty{3.6}{g}. الماء المفقود: $24.0 - 3.6 = \qty{20.4}{g}$، أي
$20.4/24.0 = 0.85$، أو 85\% من الكتلة الطازجة. أما \qty{3.6}{g}
الباقية فهي \emph{المادة الجافة}: جزيئات عضوية \omterm{def:g10:chemistry-of-life:mineral-organic}{وأملاح معدنية}. أحرق
تلك المادة الجافة فيبقى نحو \qty{0.1}{g} من رماد شاحب هو \omterm{def:g10:chemistry-of-life:mineral-organic}{الأملاح
المعدنية}؛ أما الباقي، وهو الجزء العضوي، فقد غادر ثاني أكسيد كربون
وبخار ماء.
\end{example}

\begin{proposition}[ما الذي يفعله الماء]\label{prop:g10:chemistry-of-life:water-roles}
الماء هو حيث تجري كيمياء الخلية: فهو يحل الأملاح والسكريات ومعظم
الجزيئات العضوية الصغيرة، فتلتقي وتتفاعل؛ وهو ينقلها من موضع إلى آخر
(بلازما الدم 90\% ماء، والنسغ أكثر من ذلك بكثير)؛ ولأن تسخينه يستهلك
طاقة كبيرة، فإنه يخفف تغيرات درجة حرارة الكائن المفاجئة. وفقدان 10\%
من ماء الجسم حالة إسعافية؛ أما البذرة الساكنة، عند 13\%، فقد أطفأت
كيمياءها ببساطة حتى يعود الماء.
\end{proposition}
\admitted

\begin{example}[أملاح معدنية يمكن تسميتها]\label{ex:g10:chemistry-of-life:salts}
فوسفات الكالسيوم تصلب العظم والأسنان (والكالسيوم 1.5\% من كتلة الجسم،
وكله تقريبًا في الهيكل العظمي). وشوارد الصوديوم والكلور تضبط ملوحة
الدم، قرب \qty{9}{g/L}؛ وشوارد البوتاسيوم مركزة داخل الخلايا؛ والحديد
يقع في قلب الهيموغلوبين؛ واليود لازم لصنع هرمون الغدة الدرقية. والملح
المعدني ليس مصدر طاقة قط، لكن جسمًا حُرم واحدًا منها يخفق في مهمة
محددة.
\end{example}

\section{الجزيئات العضوية: أربع عائلات}

\begin{definition}[الكربوهيدرات والدهون والبروتينات والحموض النووية]\label{def:g10:chemistry-of-life:families}
تنتمي الجزيئات العضوية إلى أربع عائلات.
\begin{itemize}
  \item \emph{الكربوهيدرات}\index{كربوهيدرات} (السكريات) لا تحتوي
        إلا الكربون والهيدروجين والأكسجين. والسكريات البسيطة وحدات
        مفردة، مثل الغلوكوز $\mathrm{C_6H_{12}O_6}$؛ أما
        الكربوهيدرات المعقدة فسلاسل من مئات أو آلاف وحدات الغلوكوز
        --- النشاء والسللوز في النبات، والغليكوجين في الحيوان.
  \item \emph{الدهون}\index{دهن} (الشحوم والزيوت) لا تحتوي هي أيضًا
        إلا C وH وO، لكن بأكسجين أقل بكثير، وهي لا تنحل في الماء.
        وأشيعها ثلاثيات الغليسيريد، كل منها من جزيء غليسرول واحد
        متصل بثلاث سلاسل طويلة من الحموض الدهنية.
  \item \emph{البروتينات}\index{بروتين} سلاسل من \emph{الحموض
        الأمينية}\index{حمض أميني}: عشرون نوعًا من الحمض الأميني،
        كلها تحوي النيتروجين، مرتبطة بترتيب دقيق في سلاسل من خمسين
        إلى عدة آلاف وحدة تنطوي على شكل محدد.
  \item \emph{الحموض النووية}\index{حمض نووي} (DNA وRNA) سلاسل من
        النوكليوتيدات، تحوي الفوسفور إلى جانب C وH وO وN؛ وهي تخزن
        المعلومة الوراثية وتنقلها، وهي موضوع
        \cref{ch:g10:universal-dna}.
\end{itemize}
\end{definition}

\begin{example}[قراءة بطاقة غذائية]\label{ex:g10:chemistry-of-life:label}
تقرأ بطاقة على خبز أسمر، لكل \qty{100}{g}: \omterm{def:g10:chemistry-of-life:families}{كربوهيدرات} \qty{41}{g}
(منها سكريات \qty{3}{g})، وشحوم \qty{3}{g}، \omterm{def:g10:chemistry-of-life:families}{وبروتين} \qty{9}{g}،
وألياف \qty{7}{g}، وملح \qty{1.1}{g}. والباقي --- نحو \qty{39}{g} ---
ماء. و«\omterm{def:g10:chemistry-of-life:families}{الكربوهيدرات}» هنا نشاء في معظمها، و«السكريات» هي البسيطة منها،
و«الألياف» سللوز (\omterm{def:g10:chemistry-of-life:families}{كربوهيدرات} لا نقدر على هضمه)، و«الشحوم» هي \omterm{def:g10:chemistry-of-life:families}{الدهون}،
و«الملح» ملح معدني. فالبطاقة تحليل كيميائي، مقرَّب.
\end{example}

\begin{proposition}[وظيفة كل عائلة]\label{prop:g10:chemistry-of-life:roles}
تتقاسم العائلات عمل الخلية، مع تداخل بينها:
\begin{itemize}
  \item \omterm{def:g10:chemistry-of-life:families}{الكربوهيدرات} هي الوقود اليومي (الغلوكوز) ومخزنه القصير الأمد
        (النشاء والغليكوجين)، والسللوز مادة بناء جدران الخلايا
        النباتية؛
  \item \omterm{def:g10:chemistry-of-life:families}{الدهون} هي المخزن الطاقي الطويل الأمد المكتنز (\qty{38}{kJ}
        للغرام، مقابل \qty{17}{kJ} للغرام في \omterm{def:g10:chemistry-of-life:families}{الكربوهيدرات}
        \omterm{def:g10:chemistry-of-life:families}{والبروتينات})، ودهون الأغشية تبني غلاف كل خلية؛
  \item \omterm{def:g10:chemistry-of-life:families}{البروتينات} هي الآلات: الإنزيمات التي تدير كل تفاعل، والألياف
        التي تقبض العضلة، والنواقل، والأجسام المضادة، والمستقبلات؛
  \item \omterm{def:g10:chemistry-of-life:families}{الحموض النووية} تحمل تعليمات بناء \omterm{def:g10:chemistry-of-life:families}{البروتينات}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[الجزيئات الكبيرة وشكل الحياة]\label{rem:g10:chemistry-of-life:macromolecules}
جزيئات الصخر صغيرة متكررة (فبلورة الكوارتز هي $\mathrm{SiO_2}$ مكررًا
بلا نهاية). أما جزيئات الحياة المميزة فضخمة --- \omterm{def:g10:chemistry-of-life:families}{فالبروتين} الواحد قد
يحوي عشرة آلاف ذرة، وجزيء DNA مليارات --- وهي \emph{حاملة للمعلومة}:
ترتيب وحدات بنائها يحمل رسالة. وليس في عالم المعادن ما يشبه جزيئًا
تعني متتاليته شيئًا.
\end{remark}

\section{التعرف على المكونات}

\begin{method}[الكواشف المعيارية للعائلات]\label{met:g10:chemistry-of-life:tests}
لكل عائلة كاشف بسيط يعطي نتيجة مرئية.
\begin{enumerate}
  \item \emph{الماء}: شريط من ورق كلور الكوبالت الأزرق يتحول إلى
        الوردي؛ أو جفف العينة وقس الكتلة المفقودة.
  \item \emph{النشاء}: قطرة من محلول اليود تنتقل من الأصفر البني إلى
        الأزرق الأسود.
  \item \emph{السكريات البسيطة} (غلوكوز، فركتوز\dots): سخنها مع كاشف
        فهلنغ أو بندكت؛ فيعطي المحلول الأزرق راسبًا أحمر آجريًا.
  \item \emph{\omterm{def:g10:chemistry-of-life:families}{البروتينات}}: كاشف البيوريه --- قطرات من كبريتات النحاس
        في وسط قلوي تتحول إلى البنفسجي.
  \item \emph{\omterm{def:g10:chemistry-of-life:families}{الدهون}}: تترك العينة بقعة شفافة لا تجف على الورق؛
        وصباغ السودان III الأحمر يلونها.
  \item \emph{\omterm{def:g10:chemistry-of-life:mineral-organic}{الأملاح المعدنية}}: أحرق المادة الجافة حرقًا تامًا؛
        والرماد الباقي معدني. وتُعرَّف شوارد بعينها بالترسيب (نترات
        الفضة للكلور مثلًا).
\end{enumerate}
وأجرِ دائمًا شاهدًا بماء نقي، وآخر بعينة موجبة معروفة، لتتأكد من أن
الكاشف يعمل ومن أن تغير اللون يعني ما تظنه.
\end{method}

\begin{example}[تحليل حبة بطاطا]\label{ex:g10:chemistry-of-life:potato}
شريحة بطاطا نيئة: ورق كلور الكوبالت يتحول إلى الوردي (ماء)؛ واليود
يعطي الأزرق الأسود في الحال (نشاء، وفير)؛ وفهلنغ لا يعطي إلا مسحة
برتقالية باهتة (سكر حر قليل)؛ والبيوريه يعطي بنفسجيًا شاحبًا (بعض
\omterm{def:g10:chemistry-of-life:families}{البروتين}، نحو 2\% من الكتلة الطازجة)؛ ولا بقعة شفافة (لا دهن تقريبًا).
ويترك التجفيف 22\% من الكتلة؛ ويترك حرق ذلك 1\% رمادًا. فالبطاطا مخزن
نشاء ملفوف في ماء --- وهو بالضبط ما يحتاج إليه النبات ليعبر الشتاء
تحت الأرض.
\end{example}

\begin{proposition}[وحدة التركيب]\label{prop:g10:chemistry-of-life:unity}
أيًّا كان الكائن --- بكتيريا أو فطرًا أو نباتًا أو حيوانًا --- تجد
الكواشف نفسها العائلات الأربع نفسها، مبنية من وحدات البناء نفسها:
\omterm{def:g10:chemistry-of-life:families}{الحموض الأمينية} العشرين نفسها، والغلوكوز نفسه، والنوكليوتيدات الأربعة
نفسها. فالكائن الحي لا يتميز بامتلاك ذرات خاصة ولا وحدات بناء خاصة،
بل بالنسب التي يحافظ عليها، وبالجزيئات البالغة الكبر التي يركبها،
وبالمعلومة التي تحملها تلك الجزيئات. وهذه الوحدة الكيميائية من أقوى
الحجج على أن الكائنات الحية جميعًا تتقاسم أصلًا مشتركًا، وهي حجة
سيتناولها \cref{ch:g10:common-ancestry}.
\end{proposition}
\admitted

\section{مسألة: الكربون في جسم إنسان}

\begin{problem}[{مسألة نهاية الأسبوع --- من حقل قمح إلى جسم: جرد
للمادة، من ماء الحبة إلى عدد الكيلوغرامات من الكربون التي يحملها
إنسان}]
\label{pb:g10:chemistry-of-life:1}
يتابع صف مسار المادة من حقل قمح إلى جسم واحدة من تلميذاته. المعطيات:
حبة القمح 13\% ماء، ومادتها الجافة نحو 70\% نشاء و12\% \omterm{def:g10:chemistry-of-life:families}{بروتين} و2\%
دهن و2\% معادن، والباقي سللوز. ويحتفظ دقيق الخبز تقريبًا بتركيب الجزء
الداخلي من الحبة. وكتلة التلميذة، ليلى، \qty{60}{kg}؛ استعمل تركيب
البالغ في الشكل الثاني (ماء 60\%، بروتينات 17\%، دهون 16\%، معادن
6\%، \omterm{def:g10:chemistry-of-life:families}{كربوهيدرات} 1\%). أما النسب الكتلية للعناصر داخل كل عائلة: فخذ
\omterm{def:g10:chemistry-of-life:families}{الكربوهيدرات} 40\% كربونًا، \omterm{def:g10:chemistry-of-life:families}{والدهون} 77\% كربونًا، \omterm{def:g10:chemistry-of-life:families}{والبروتينات} 53\%
كربونًا و16\% نيتروجينًا.

\medskip
\noindent\textbf{الجزء الأول --- الحبة.}
\begin{enumerate}
  \item حبة كتلتها \qty{45}{mg} تجفف. ما كتلة الماء التي تفقدها، وما
        كتلتها الجافة؟
  \item احسب كتل النشاء \omterm{def:g10:chemistry-of-life:families}{والبروتين} والدهن في المادة الجافة لحبة واحدة.
  \item أي كواشف \cref{met:g10:chemistry-of-life:tests} تؤكد كلًا من
        هذه المكونات الثلاثة على حبة مسحوقة؟
  \item تزرع الحبة وتسقى؛ وبعد ثلاثة أيام تصبح البادرة 88\% ماء. فسّر
        لماذا لا يمكن أن يبدأ الإنبات عند 13\%.
  \item يعطي هكتار واحد \qty{8}{t} من الحبوب. ما كتلة الماء التي تحصد
        معها، وما كتلة المادة الجافة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الخبز.}
\begin{enumerate}[resume]
  \item يخبز رغيف من \qty{500}{g} دقيق و\qty{320}{g} ماء و\qty{10}{g}
        ملح؛ ويفقد \qty{130}{g} من الماء في الفرن. ما كتلة الرغيف
        المخبوز، وما محتواه المائي بالمئة؟
  \item باستعمال التركيب الجاف للحبة بدل تركيب الدقيق، قدّر كتلتي
        النشاء \omterm{def:g10:chemistry-of-life:families}{والبروتين} في الرغيف.
  \item باستعمال \qty{17}{kJ/g} \omterm{def:g10:chemistry-of-life:families}{للكربوهيدرات} \omterm{def:g10:chemistry-of-life:families}{والبروتينات}
        و\qty{38}{kJ/g} للدهون، قدّر المحتوى الطاقي للرغيف.
  \item تأكل ليلى \qty{120}{g} من هذا الخبز. ما النسبة التي يغطيها
        ذلك من حاجة يومية تبلغ نحو \qty{10000}{kJ}؟
  \item لماذا لا يكون ملح الرغيف مصدر طاقة، وما فائدته في الجسم؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الجسم: الماء والعائلات.}
\begin{enumerate}[resume]
  \item احسب كتلة الماء في جسم ليلى.
  \item احسب كتل \omterm{def:g10:chemistry-of-life:families}{البروتين} والدهن والمعادن \omterm{def:g10:chemistry-of-life:families}{والكربوهيدرات}.
  \item تفقد ليلى \qty{1.2}{kg} بالتعرق أثناء جري طويل. ما نسبة ذلك
        من ماء جسمها؟ ولماذا يجب تعويضه بسرعة؟
  \item دمها، ونحو \qty{4.5}{L}، 55\% منه بلازما، والبلازما 90\% ماء.
        قدّر كتلة الماء في دمها (خذ \qty{1}{kg} لكل لتر). ما نسبة ذلك
        من ماء جسمها؟ وأين الباقي؟
  \item الوليد 75\% ماء. فسّر في جملة واحدة لماذا يكون الرضيع أشد
        عرضة للتجفاف من البالغ.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الجسم: العناصر.}
\begin{enumerate}[resume]
  \item باستعمال نسب الكربون في كل عائلة، احسب كتلة الكربون في
        \omterm{def:g10:chemistry-of-life:families}{كربوهيدرات} ليلى ودهونها وبروتيناتها، ثم كتلة الكربون
        الإجمالية.
  \item ما نسبة الكربون من كتلة جسمها؟ قارن بنسبة 18.5\% في
        \cref{prop:g10:chemistry-of-life:four} وعلّق على التوافق.
  \item احسب كتلة النيتروجين في بروتيناتها. من أين جاء ذلك النيتروجين
        في نهاية المطاف؟
  \item الماء 89\% أكسجين من حيث الكتلة. احسب كتلة الأكسجين التي
        يحملها ماء جسمها وحده، وقارنها بنسبة 65\% في
        \cref{prop:g10:chemistry-of-life:four}. أي جزيء يفسر معظم
        أكسجين الجسم؟
  \item اذكر النتيجة في جملة واحدة: كم كيلوغرامًا من الكربون يحمل شخص
        كتلته \qty{60}{kg}، وعبر أي عمليتين بيولوجيتين غادر ذلك
        الكربون الهواء ودخل الجسم؟
\end{enumerate}
\end{problem}
