\chapter{Comment les plantes se répandent}\label{ch:g6:how-plants-spread}

Un talus de route neuve est décapé au bulldozer en mars~: de la
terre nue, brute, pas un brin de vert. Reviens en septembre~: il est
brodé de plantules d'un bout à l'autre --- coquelicots, pissenlits,
pousses de saule, graminées. Personne ne l'a semé. Et pourtant les
\omterm{def:g1:plants-around-us:plant}{plantes}, qui ne peuvent pas faire un pas, ont colonisé le sol nu en
quelques mois. Ce chapitre parle de leur service de voyages.

\section{Les plantes voyagent en graines et en spores}

\begin{proposition}[Des plantes fixées, des commencements voyageurs]\label{prop:g6:how-plants-spread:travel}
Une \omterm{def:g1:plants-around-us:plant}{plante} adulte est fixée pour la vie
(\cref{def:g1:plants-around-us:plant}) --- une \omterm{def:g5:biodiversity:species}{espèce} se répand donc
en envoyant au loin ses \emph{commencements}~: \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} et \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits}
(\cref{ch:g5:flower-to-fruit}), \omterm{def:g4:plants-without-seeds:spore}{spores}
(\cref{def:g4:plants-without-seeds:spore}), ou \omterm{ex:g4:plants-without-seeds:runner}{stolons} et autres
avancées végétatives
(\cref{def:g4:plants-without-seeds:vegetative}). Occuper ainsi un
sol nouveau s'appelle la
\emph{colonisation}\index{colonisation}.
\end{proposition}
\admitted

\begin{definition}[Dissémination]\label{def:g6:how-plants-spread:dispersal}
La \emph{dissémination}\index{dissémination} est le transport des
\omterm{prop:g3:flowers-fruits-seeds:transform}{graines}, des \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} ou des \omterm{def:g4:plants-without-seeds:spore}{spores} loin de la \omterm{def:g1:plants-around-us:plant}{plante} mère. Les
transporteurs sont peu nombreux et anciens~: le \emph{vent}, les
\emph{\omterm{def:g1:animals-around-us:animal}{animaux}}, l'\emph{eau} --- et quelques \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} qui projettent
eux-mêmes leurs \omterm{prop:g3:flowers-fruits-seeds:transform}{graines}.
\end{definition}

\begin{example}[Pourquoi quitter la maison~?]\label{ex:g6:how-plants-spread:whyleave}
Une \omterm{def:g2:from-seed-to-plant:seed}{graine} tombée au pied de sa mère germe à l'ombre de sa mère,
parmi les \omterm{def:g1:plants-around-us:parts}{racines} de sa mère --- en se disputant la \omterm{def:g6:exploring-our-environment:conditions}{lumière} et l'eau
mêmes de \cref{prop:g4:what-plants-need:needs} avec une rivale cent
fois plus grande qu'elle. La \omterm{def:g6:how-plants-spread:dispersal}{dissémination} est l'échappatoire à ce
match perdu d'avance~: la distance est le premier cadeau qu'une
\omterm{def:g1:plants-around-us:plant}{plante} fait à ses jeunes.
\end{example}

\section{Les transporteurs, et les fruits bâtis pour eux}

\begin{proposition}[Bâtis pour aller avec le transporteur]\label{prop:g6:how-plants-spread:builds}
La forme des \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} et des \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} va avec leur transporteur ---
l'ajustement de
\cref{def:g4:adapted-to-their-world:adaptation}, appliqué aux
bagages~:
\begin{itemize}
  \item le \emph{vent}~: des parachutes (les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} plumeuses du
        pissenlit), des ailes (les samares d'érable et de frêne qui
        tournoient en descendant de biais), ou des \omterm{def:g4:plants-without-seeds:spore}{spores} fines
        comme de la poussière qui flottent tout simplement (les
        millions de la fougère)~;
  \item les \emph{\omterm{def:g1:animals-around-us:animal}{animaux}, au-dehors}~: des \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} accrocheurs ---
        bardane, gaillet --- dont les crochets s'agrippent aux
        poils et aux chaussettes et voyagent jusqu'au toilettage~;
  \item les \emph{\omterm{def:g1:animals-around-us:animal}{animaux}, au-dedans}~: des \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} charnus et
        sucrés --- cerises, mûres, cynorhodons --- avalés entiers~;
        les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} à enveloppe coriace passent sans dommage et
        sont déposées ailleurs, avec de l'engrais
        (la réponse de \cref{exo:g3:flowers-fruits-seeds:10},
        complétée)~;
  \item l'\emph{eau}~: des \omterm{prop:g3:flowers-fruits-seeds:transform}{fruits} flottants --- les minuscules
        radeaux de l'aulne, la coque océanique de la noix de
        coco~;
  \item la \emph{projection}~: des gousses qui sèchent, se tendent
        et claquent --- l'ajonc et les violettes projettent leurs
        \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} à plusieurs mètres de la \omterm{def:g1:plants-around-us:plant}{plante} mère.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Lire le bord de la route]\label{ex:g6:how-plants-spread:roadside}
Le talus de septembre, décodé~: pissenlits et saules --- \omterm{prop:g3:flowers-fruits-seeds:transform}{graines}
plumeuses, arrivées par le vent~; coquelicots --- leurs minuscules
\omterm{prop:g3:flowers-fruits-seeds:transform}{graines} étaient déjà dans la terre remuée, en attente comme seules
des \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} savent l'être (\cref{exo:g2:from-seed-to-plant:9})~;
plantules de mûrier sous les piquets de clôture --- déposées par les
\omterm{prop:g3:sorting-living-things:groups}{oiseaux} perchés, engrais compris~; bardane le long du sentier ---
apportée par un chien. La route d'arrivée de chaque colon est écrite
dans son équipement.
\end{example}

\begin{method}[Nommer le transporteur d'après le fruit]\label{met:g6:how-plants-spread:name}
Devant un \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit} ou une \omterm{def:g2:from-seed-to-plant:seed}{graine}~:
\begin{enumerate}
  \item cherche un équipement de vol --- aigrettes, ailes, finesse
        de poussière~: le vent~;
  \item cherche des crochets~: les poils des \omterm{def:g1:animals-around-us:animal}{animaux}, au-dehors~;
  \item charnu, coloré, sucré~? le \omterm{def:g4:journey-of-food:tract}{tube digestif} d'un \omterm{def:g1:animals-around-us:animal}{animal}, au-dedans
        --- vérifie que les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} ont une enveloppe coriace~;
  \item flottant, près de l'eau~? l'eau~;
  \item une gousse sèche, tordue, sous tension~? la projection~;
  \item vérifie ton verdict~: laisse-le tomber, fais-le flotter,
        presse-le sur une manche en laine.
\end{enumerate}
\end{method}

\section{Coloniser sans graines}

\begin{example}[L'avancée au sol]\label{ex:g6:how-plants-spread:vegetative}
À côté des arrivées aériennes, certains colons marchent sur leurs
propres pieds végétatifs
(\cref{ch:g4:plants-without-seeds})~: le chiendent avance sous terre
par des \omterm{def:g1:plants-around-us:parts}{tiges} rampantes, mètre après mètre~; les ronces s'arquent et
s'enracinent là où leur pointe touche le sol~; les \omterm{ex:g4:plants-without-seeds:runner}{stolons} du
fraisier sautent en travers du terrain ouvert. Lent, à courte
portée --- et impossible à arrêter dans un jardin négligé, comme le
sait tout jardinier.
\end{example}

\begin{remark}[Une colonisation à deux vitesses]\label{rem:g6:how-plants-spread:twospeeds}
Les deux systèmes se partagent le travail. Les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} et les \omterm{def:g4:plants-without-seeds:spore}{spores}
sont le service à \emph{longue portée} --- de rares atterrissages
chanceux au loin, le pissenlit qui traverse la ville. L'avancée
végétative est le service à \emph{courte portée} --- l'occupation
dense et sûre du terrain d'à côté. Le buisson de ronces qui s'étend
depuis la haie emploie les deux à la fois~: des \omterm{def:g1:plants-around-us:parts}{tiges} arquées pour
le mètre, des \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} portées par les \omterm{prop:g3:sorting-living-things:groups}{oiseaux} pour le kilomètre.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:1}
Qu'est-ce que la \omterm{prop:g6:how-plants-spread:travel}{colonisation}~? Qu'est-ce que la \omterm{def:g6:how-plants-spread:dispersal}{dissémination}, et
quels en sont les principaux transporteurs~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:2}
Pourquoi une \omterm{def:g2:from-seed-to-plant:seed}{graine} a-t-elle intérêt à \omterm{def:g2:from-seed-to-plant:germination}{germer} loin de sa mère~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:3}
Nomme le transporteur de chacun~: l'aigrette du pissenlit, le \omterm{prop:g3:flowers-fruits-seeds:transform}{fruit}
de la bardane, la cerise, la samare d'érable, la noix de coco.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:4}
Comment les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} d'une mûre mangée par un oiseau survivent-elles
au voyage, et avec quoi arrivent-elles~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:5}
Quel service de voyage emploient l'ajonc et les violettes~? Décris
le mécanisme.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:how-plants-spread:6}
Nomme deux colons végétatifs et la façon dont chacun avance.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:7}
Décode le talus de la route~: pour le pissenlit, le coquelicot, la
mûre et la bardane, donne la route d'arrivée que nomme le chapitre.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:8}
Applique \cref{met:g6:how-plants-spread:name} à~: (a) une \omterm{def:g2:from-seed-to-plant:seed}{graine}
plumeuse de chardon~; (b) un cynorhodon~; (c) une boule de gaillet
qui s'accroche à ton pull.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:9}
Explique l'idée des «~deux vitesses~» de
\cref{rem:g6:how-plants-spread:twospeeds} avec le buisson de
ronces~: quel système fait le mètre, lequel fait le kilomètre~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:10}
La fougère a colonisé le talus nord humide de la nouvelle route mais
pas son talus sud sec. Combine
\cref{prop:g4:plants-without-seeds:damp} et
\cref{prop:g6:exploring-our-environment:decide} en une explication.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:how-plants-spread:11}
Une île surgit nue de la mer, à des kilomètres du rivage. Quels
services de \omterm{def:g6:how-plants-spread:dispersal}{dissémination} peuvent seulement l'atteindre~? Prédis
l'équipement de ses premières \omterm{def:g1:plants-around-us:plant}{plantes} colonisatrices.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:how-plants-spread:12}
La chair sucrée coûte cher à fabriquer pour une \omterm{def:g1:plants-around-us:plant}{plante}. À l'aide de
\cref{ex:g6:how-plants-spread:whyleave} et du service au-dedans des
\omterm{def:g1:animals-around-us:animal}{animaux}, explique ce que le cerisier achète avec tout ce sucre ---
et pourquoi l'enveloppe coriace des \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} est l'autre moitié de
l'achat.
\end{exercise}

\section{Problème~: le champ abandonné}

\begin{problem}[{Problème du week-end --- dix ans de colonisation,
observés depuis une barrière}]\label{pb:g6:how-plants-spread:1}
Un champ labouré est abandonné en automne. Une naturaliste le
photographie depuis la barrière chaque mois de juin pendant dix ans,
et tient un carnet.

\medskip\noindent\textbf{Partie I --- La première année.}
\begin{enumerate}
  \item Juin de la première année~: la terre nue est parsemée de
        coquelicots, de pissenlits et de chardons. Pour chacun,
        nomme la source et la route probables --- les coquelicots
        venus du sol lui-même, les autres par les airs. Quel
        équipement a rendu possibles les arrivées aériennes~?
  \item Aucun mûrier n'apparaît au milieu du champ la première
        année, mais trois plantules se dressent sous les vieux
        piquets de clôture. Explique cette répartition avec le
        service au-dedans des \omterm{def:g1:animals-around-us:animal}{animaux} --- pourquoi exactement sous
        les piquets~?
  \item La naturaliste ne trouve de \omterm{prop:g4:plants-without-seeds:damp}{fougères} nulle part, alors
        qu'un bois qui en regorge se trouve au vent.
        Des \omterm{def:g4:plants-without-seeds:spore}{spores} de fougère sont sûrement arrivées
        (\cref{ex:g4:plants-without-seeds:fern} en compte des
        millions). Quelle condition du champ découvert, d'après
        \cref{prop:g4:plants-without-seeds:damp}, les a refusées~?
  \item Le chiendent envahit le champ depuis la bordure seulement,
        d'un mètre ou deux par an. De quel système de \omterm{prop:g6:how-plants-spread:travel}{colonisation}
        s'agit-il, et pourquoi ne peut-il pas apparaître au milieu
        du champ comme l'ont fait les pissenlits~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les années suivantes.}
\begin{enumerate}[resume]
  \item À la troisième année, graminées et hautes herbes ont
        refermé le sol nu, et les coquelicots ont disparu de la
        végétation sur pied. Où sont les coquelicots maintenant, et
        sous quelle forme~? (Souviens-toi du talus de la route et
        de \cref{prop:g6:life-through-seasons:herbs}.)
  \item À partir de la quatrième année, des arceaux de ronces et de
        jeunes aubépines parsèment le champ, plus denses près des
        haies et sous la ligne de clôture. Nomme les deux services
        qui les ont mis là, et l'indice contenu dans la
        répartition.
  \item À la huitième année, des jeunes saules et bouleaux
        dépassent les ronces --- deux arbres à \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} minuscules,
        plumeuses ou légères comme la poussière. Pourquoi les
        chênes et les marronniers, à \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} lourdes, sont-ils
        encore presque absents, et quel messager \omterm{def:g1:animals-around-us:animal}{animal} finit par
        les planter eux aussi~?
  \item Les photographies montrent la distribution du champ changer
        d'année en année alors qu'aucun jardinier n'y a touché.
        Énonce la règle qui gouverne cela~: qu'est-ce que la
        végétation de chaque année change aux \omterm{def:g6:exploring-our-environment:conditions}{conditions} du champ
        lui-même --- la \omterm{def:g6:exploring-our-environment:conditions}{lumière} au ras du sol avant tout --- et qui
        ce changement favorise-t-il ensuite~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La leçon générale.}
\begin{enumerate}[resume]
  \item Range les colons par ordre d'arrivée~: \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} en attente
        dans le sol, voyageurs du vent, arbustes portés par les
        \omterm{def:g1:animals-around-us:animal}{animaux}, arbres à \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} lourdes. Associe chaque vague au
        service de \omterm{def:g6:how-plants-spread:dispersal}{dissémination} qui en a fixé le moment.
  \item Les fissures du pavage d'une place de ville donnent la même
        distribution que la première année --- coquelicots,
        pissenlits --- mais n'atteignent jamais les arbustes de la
        quatrième. Donne deux raisons pour lesquelles la succession
        s'y arrête.
  \item La naturaliste affirme~: «~laissé tranquille, ce champ
        deviendra un bois~». Défends cette affirmation avec la
        tendance des dix ans --- et nomme l'exception du pré de
        \cref{prop:g5:people-and-nature:managed} qui prouve que
        «~laissé tranquille~» est l'expression décisive.
  \item Résume tout le problème en deux phrases, en employant les
        mots \emph{\omterm{def:g6:how-plants-spread:dispersal}{dissémination}}, \emph{\omterm{def:g6:exploring-our-environment:conditions}{conditions}} et
        \emph{\omterm{prop:g6:how-plants-spread:travel}{colonisation}}.
\end{enumerate}
\end{problem}
