\chapter{Classificar os animais}\label{ch:g4:classifying-animals}

Um museu de história natural guarda milhões de \omterm{def:g1:animals-around-us:animal}{animais} --- gavetas
de besouros, salões de \omterm{def:g3:bones-and-muscles:skeleton}{esqueletos}, potes de \omterm{prop:g3:sorting-living-things:groups}{peixes}. E mesmo assim um
curador chega a qualquer um deles em minutos. O segredo não é a
memória, é a \emph{ordem}: cada \omterm{def:g1:animals-around-us:animal}{animal} tem o seu lugar num grande
sistema de grupos dentro de grupos, construído com as regras de
\omterm{def:g4:classifying-animals:classification}{classificação} do \cref{ch:g3:sorting-living-things} --- e neste ano
vamos aprender como esse sistema funciona.

\section{Grupos dentro de grupos}

\begin{definition}[Classificação]\label{def:g4:classifying-animals:classification}
\emph{Classificar}\index{classificar} os \omterm{def:g1:living-or-not:living}{seres vivos} é separá-los
por características do corpo em comum, formando grupos, e separar
esses grupos em grupos maiores --- \emph{grupos encaixados dentro de
grupos}\index{grupos encaixados}, como caixas dentro de caixas. Todo
esse arranjo é uma \emph{classificação}\index{classificação}.
\end{definition}

\begin{example}[Os endereços do gato]\label{ex:g4:classifying-animals:cat}
Um gato, vários endereços encaixados: o gato é um \emph{mamífero}
(pelo, \omterm{def:g2:animals-grow-up:born}{leite} para os filhotes); todo mamífero é um
\emph{\omterm{def:g4:classifying-animals:vertebrate}{vertebrado}} (um \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto} com \omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral} por dentro);
todo \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} é um \emph{\omterm{def:g1:animals-around-us:animal}{animal}}. Gato dentro dos \omterm{prop:g3:sorting-living-things:groups}{mamíferos},
\omterm{prop:g3:sorting-living-things:groups}{mamíferos} dentro dos \omterm{def:g4:classifying-animals:vertebrate}{vertebrados}, \omterm{def:g4:classifying-animals:vertebrate}{vertebrados} dentro dos \omterm{def:g1:animals-around-us:animal}{animais}:
cada caixa fica dentro de uma maior.
\end{example}

\begin{definition}[Vertebrados e invertebrados]\label{def:g4:classifying-animals:vertebrate}
Um \emph{vertebrado}\index{vertebrado} é um \omterm{def:g1:animals-around-us:animal}{animal} com um \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto}
interno construído em volta de uma \emph{coluna
vertebral}\index{coluna vertebral} --- como a \omterm{def:g3:bones-and-muscles:skeleton}{coluna} que você sentiu
no \cref{ex:g3:bones-and-muscles:feel}. Um \omterm{def:g1:animals-around-us:animal}{animal} sem \omterm{def:g3:bones-and-muscles:skeleton}{coluna}
vertebral é um \emph{invertebrado}\index{invertebrado}: \omterm{prop:g3:sorting-living-things:groups}{insetos},
aranhas, caracóis, minhocas, águas-vivas --- a imensa maioria de
todos os \omterm{def:g1:animals-around-us:animal}{animais}.
\end{definition}

\section{As cinco classes de vertebrados}

\begin{proposition}[As cinco classes de vertebrados]\label{prop:g4:classifying-animals:classes}
Os \omterm{def:g4:classifying-animals:vertebrate}{vertebrados} são classificados em cinco grandes grupos, chamados
\emph{classes}\index{classe (de vertebrados)}:
\begin{enumerate}
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{peixes}}: escamas, nadadeiras, respiram na água ---
        truta, tubarão, sardinha;
  \item \emph{anfíbios}\index{anfíbio}: \omterm{def:g1:the-five-senses:sense}{pele} nua e úmida, \omterm{def:g2:animals-grow-up:born}{ovos} na
        água, infância de girino --- sapo, rã, salamandra;
  \item \emph{répteis}\index{réptil}: \omterm{def:g1:the-five-senses:sense}{pele} seca e escamosa, quase
        todos põem \omterm{def:g2:animals-grow-up:born}{ovos} com casca em terra --- lagarto, cobra,
        tartaruga, crocodilo;
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{aves}}: \omterm{def:g1:animals-around-us:covering}{penas}, bico, \omterm{def:g2:animals-grow-up:born}{ovos} com casca --- sabiá, ema,
        pinguim;
  \item \emph{\omterm{prop:g3:sorting-living-things:groups}{mamíferos}}: pelo, filhotes alimentados com \omterm{def:g2:animals-grow-up:born}{leite} ---
        gato, baleia, morcego, ser humano.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Classificando uma entrega do zoológico]\label{ex:g4:classifying-animals:zoo}
Chegam seis caixotes: uma píton (\omterm{def:g1:the-five-senses:sense}{pele} seca e escamosa, \omterm{def:g2:animals-grow-up:born}{ovos} com
casca --- réptil), uma rã (\omterm{def:g1:the-five-senses:sense}{pele} nua e úmida --- anfíbio), um pinguim
(\omterm{def:g1:animals-around-us:covering}{penas} --- ave), um tubarão (nadadeiras e brânquias --- peixe), um
golfinho (\omterm{def:g2:animals-grow-up:born}{leite} para o filhote --- mamífero) e um caixote de
caracóis gigantes (nenhuma \omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral} --- \omterm{def:g4:classifying-animals:vertebrate}{invertebrados}, em
classe de \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} nenhuma).
\end{example}

\begin{remark}[Réptil ou anfíbio?]\label{rem:g4:classifying-animals:confusion}
Lagartos e salamandras parecem iguais à primeira vista --- quatro
patas, rabo comprido. Quem decide é a \omterm{def:g1:the-five-senses:sense}{pele}: a do lagarto é seca e
escamosa, a da salamandra é úmida e nua; os \omterm{def:g2:animals-grow-up:born}{ovos} do lagarto têm
casca e são postos em terra, os da salamandra são postos nus na
água. A mesma silhueta, classes diferentes.
\end{remark}

\section{Usar uma classificação}

\begin{method}[Pôr um vertebrado na classe dele]\label{met:g4:classifying-animals:place}
Faça as perguntas em ordem, parando no primeiro sim:
\begin{enumerate}
  \item Tem \omterm{def:g1:animals-around-us:covering}{penas}? --- é uma ave.
  \item Tem pelo (ou \omterm{def:g2:animals-grow-up:born}{leite} para os filhotes)? --- é um mamífero.
  \item Tem \omterm{def:g1:the-five-senses:sense}{pele} seca e escamosa? --- é um réptil.
  \item Tem \omterm{def:g1:the-five-senses:sense}{pele} nua e úmida e uma infância de girino? --- é um
        anfíbio.
  \item Tem nadadeiras e brânquias, vida na água? --- é um peixe.
\end{enumerate}
Nenhuma \omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral}? Então era um \omterm{def:g4:classifying-animals:vertebrate}{invertebrado}, e as perguntas
não se aplicam.
\end{method}

\begin{proposition}[O que uma classe conta de graça]\label{prop:g4:classifying-animals:free}
Saber a classe de um \omterm{def:g1:animals-around-us:animal}{animal} é saber um pacote de fatos antes mesmo
de olhar: se disserem só ``é um mamífero'', você já espera pelo,
\omterm{def:g2:animals-grow-up:born}{leite} para os filhotes e --- como em quase todos os \omterm{prop:g3:sorting-living-things:groups}{mamíferos} ---
filhotes nascidos de barriga. Essa economia é justamente o objetivo
da \omterm{def:g4:classifying-animals:classification}{classificação}: aprenda o grupo uma vez, e cada \omterm{def:g1:my-body:parts}{membro} chega já
meio conhecido.
\end{proposition}
\admitted

\begin{example}[Uma espécie nova aparece]\label{ex:g4:classifying-animals:new}
Os naturalistas ainda descobrem \omterm{def:g1:animals-around-us:animal}{animais} nunca vistos antes. Quando
um \omterm{def:g1:animals-around-us:animal}{animal} parecido com um morcego é encontrado com pelo e \omterm{def:g2:animals-grow-up:born}{leite}, ele
vai direto para os \omterm{prop:g3:sorting-living-things:groups}{mamíferos} --- e na hora os biólogos esperam uma
\omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral}, um coração de quatro cavidades e sangue quente,
antes de conferir qualquer uma dessas coisas. A \omterm{def:g4:classifying-animals:classification}{classificação} paga
em previsões.
\end{example}

\begin{remark}[Rumo à árvore de parentesco]\label{rem:g4:classifying-animals:tree}
Por que o pelo, o \omterm{def:g2:animals-grow-up:born}{leite} e a \omterm{def:g4:classifying-animals:vertebrate}{coluna vertebral} andam juntos com tanta
fidelidade? A resposta completa --- que os \omterm{def:g1:my-body:parts}{membros} de um grupo
herdam as características em comum de ancestrais comuns --- pertence
à história do parentesco e da evolução, retomada em anos posteriores.
Por enquanto, as caixas encaixadas simplesmente funcionam, e
funcionam maravilhosamente.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:1}
O que é um \omterm{def:g4:classifying-animals:vertebrate}{vertebrado}? O que é um \omterm{def:g4:classifying-animals:vertebrate}{invertebrado}? Dê dois exemplos de
cada.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:2}
Diga as cinco classes de \omterm{def:g4:classifying-animals:vertebrate}{vertebrados} com as características que as
definem.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:3}
Ponha cada \omterm{def:g1:animals-around-us:animal}{animal} no lugar: a sardinha, o crocodilo, a salamandra, a
ema, a baleia, a minhoca.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:4}
Escreva os endereços encaixados do gato, da caixa menor para a
maior, como no \cref{ex:g4:classifying-animals:cat}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:5}
Um lagarto e uma salamandra parecem iguais. Que duas características
separam as classes deles?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:6}
Que grupo reúne a imensa maioria de todos os \omterm{def:g1:animals-around-us:animal}{animais}? Diga quatro
\omterm{def:g1:my-body:parts}{membros} dele.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:7}
Rode o \cref{met:g4:classifying-animals:place} em voz alta no
pinguim e no golfinho, pergunta por pergunta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:8}
Dizem a você apenas: ``este \omterm{def:g1:animals-around-us:animal}{animal} é uma ave''. Liste três coisas
que você já espera sem olhar, segundo a
\cref{prop:g4:classifying-animals:free}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:9}
A baleia quase não tem pelo e vive como um peixe. Usando o
\cref{met:g4:classifying-animals:place} e o
\cref{ex:g3:sorting-living-things:surprises}, explique por que ela é
mesmo assim um mamífero --- e por que ``vive no mar'' nunca foi uma
característica de \omterm{def:g4:classifying-animals:classification}{classificação}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:10}
A cobra não tem patas, e a minhoca também não. Por que uma é um
réptil \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} e a outra é um \omterm{def:g4:classifying-animals:vertebrate}{invertebrado}? Que característica
sozinha vale mais que a falta de patas?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:classifying-animals:11}
O ornitorrinco deixou os naturalistas confusos: pelo e \omterm{def:g2:animals-grow-up:born}{leite}, e
ainda assim ele põe \omterm{def:g2:animals-grow-up:born}{ovos} com casca. Que classe ganhou, e por quê? O
que essa confusão mostra sobre \omterm{def:g4:classifying-animals:classification}{classificar} por \emph{pacotes} de
características em vez de por uma só?
\end{exercise}
