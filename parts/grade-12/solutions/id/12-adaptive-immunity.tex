\section*{Bab \ref{ch:g12:adaptive-immunity} --- Imunitas Adaptif dan Vaksinasi}

\begin{solution}{exo:g12:adaptive-immunity:1}
\omterm{def:g12:adaptive-immunity:antigen}{Antigen} adalah molekul apa pun yang dapat dikenali sistem adaptif;
\omterm{def:g12:adaptive-immunity:antigen}{limfosit} adalah \omterm{def:g10:cells-common-unit:cell}{sel} darah putih yang membawa reseptor bagi satu
\omterm{def:g12:adaptive-immunity:antigen}{antigen}. Kekhasannya datang dari reseptornya, yang dirakit secara acak
selama pematangan dan identik pada semua salinannya.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:2}
\omterm{def:g12:adaptive-immunity:antigen}{Antigennya} mengikat segelintir \omterm{def:g12:adaptive-immunity:antigen}{limfosit} yang reseptornya cocok; \omterm{def:g10:cells-common-unit:cell}{sel} itu
berlipat menjadi sebuah klon dalam beberapa hari; klonnya
berdiferensiasi menjadi \omterm{def:g10:cells-common-unit:cell}{sel} efektor yang bertempur; sebagian kecil
menjadi \omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} yang bertahan.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:3}
Sebuah \omterm{def:g11:gene-expression:protein}{protein} berbentuk Y yang disekresikan \omterm{def:g10:cells-common-unit:cell}{sel} plasma, yaitu salinan
larut reseptor \omterm{def:g10:cells-common-unit:cell}{sel} B; tiap lengannya mengikat \omterm{def:g12:adaptive-immunity:antigen}{antigennya}. Pada sebutir
bakteri, \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} membentuk selubung yang dicengkeram \omterm{def:g12:innate-immunity:phagocytosis}{fagosit}, dan
menyilangikat bakterinya menjadi kompleks yang lalu ditelan.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:4}
\omterm{prop:g12:adaptive-immunity:tcells}{Sel T penolong}, yang diaktifkan \omterm{prop:g12:innate-immunity:handover}{sel dendritik}, menyekresikan sitokina
yang mengizinkan dan memperkuat \omterm{def:g10:cells-common-unit:cell}{sel} B serta \omterm{def:g10:cells-common-unit:cell}{sel} T lainnya. \omterm{def:g10:cells-common-unit:cell}{Sel} T
sitotoksik membunuh \omterm{def:g10:cells-common-unit:cell}{sel} yang memajang serpihan sebuah virus atau
\omterm{def:g11:gene-expression:protein}{protein} sebuah \omterm{def:g11:cancer:cancer}{tumor}.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:5}
Bentuk \omterm{def:g12:adaptive-immunity:antigen}{antigen} patogen yang tak berbahaya (dilemahkan, dimatikan,
sebuah \omterm{def:g11:gene-expression:protein}{protein}, racun yang dinonaktifkan, atau petunjuknya) beserta
adjuvan. \omterm{def:g12:adaptive-immunity:vaccine}{Vaksin} memancing tanggapan primer dan meninggalkan \omterm{prop:g12:adaptive-immunity:selection}{sel
ingatan}, sehingga patogen yang sungguhan disambut tanggapan sekunder.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:6}
Primer: tundaan sepekan, puncak yang rendah (sekitar sepersepuluh
puncak sekundernya) pada pekan kedua sampai ketiga, lalu menurun dalam
sebulan. Sekunder: tundaan dua atau tiga hari, puncak sepuluh kali
lebih tinggi dalam sepuluh hari, dan bertahan berbulan-bulan.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:7}
Ingatannya bersifat khas: \omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} yang ditinggalkan dosis pertama A
hanya menanggapi A; B, yang ditemui untuk pertama kali, mendapat
tanggapan primer.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:8}
\omterm{prop:g12:adaptive-immunity:antibodies}{Antibodi} adalah \omterm{def:g11:gene-expression:protein}{protein} di dalam cairan dan tak dapat masuk ke \omterm{def:g10:cells-common-unit:cell}{sel}. \omterm{def:g10:cells-common-unit:cell}{Sel}
yang terinfeksi memajang serpihan virusnya di permukaannya, dan \omterm{def:g10:cells-common-unit:cell}{sel} T
sitotoksik mengenalinya lalu membunuh \omterm{def:g10:cells-common-unit:cell}{sel} itu.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:9}
$1 - 1/4 = 75\%$; $1 - 1/18 \approx 94\%$.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:10}
Di atas ambangnya tiap kasus menjangkiti kurang dari satu orang lain,
sehingga virusnya tak dapat beredar dan jarang mencapai seorang bayi;
orang dewasa yang kebal itulah tembok di sekeliling mereka.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:11}
Sekitar 750, 520 dan 180 per mikroliter; \omterm{prop:g12:adaptive-immunity:hiv}{AIDS} mulai sekitar tahun ke-9,
ketika cacahnya melewati 200.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:12}
Sitokina \omterm{prop:g12:adaptive-immunity:tcells}{sel T penolong} itulah yang mengizinkan \omterm{def:g10:cells-common-unit:cell}{sel} B menjadi \omterm{def:g10:cells-common-unit:cell}{sel}
plasma dan \omterm{def:g10:cells-common-unit:cell}{sel} T sitotoksik berlipat; kedua lengannya bergantung pada
pengoordinasi yang sama, sehingga menyingkirkannya membungkam keduanya.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:13}
\omterm{prop:g12:adaptive-immunity:antibodies}{Antibodinya} terseleksi terhadap virus dalam keadaannya yang dahulu;
virusnya memutasikan \omterm{def:g11:gene-expression:protein}{protein} permukaannya lebih cepat daripada
kesanggupan klon baru terseleksi, sehingga tiap tanggapannya
tertinggal. Dan virusnya bersembunyi lalu berlipat justru di dalam \omterm{def:g10:cells-common-unit:cell}{sel}
penolong yang seharusnya mengoordinasikan tanggapannya.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:14}
\omterm{prop:g12:adaptive-immunity:selection}{Sel ingatan} mengenali \omterm{def:g11:gene-expression:protein}{protein} permukaan yang disajikan \omterm{def:g12:adaptive-immunity:vaccine}{vaksinnya}.
\omterm{def:g11:gene-expression:protein}{Protein} influenza bermutasi setiap tahun menjadi bentuk yang tak
diikat ingatan lama; \omterm{def:g11:gene-expression:protein}{protein} permukaan campak nyaris tak berubah,
sehingga ingatan tahun 1970 masih cocok dengan virus hari ini.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:15}
Yang beragam: reseptor \omterm{def:g12:adaptive-immunity:antigen}{limfositnya}, yang dibuat secara acak. Yang
menyeleksi: \omterm{def:g12:adaptive-immunity:antigen}{antigennya}, yang hanya membiarkan klon yang cocok berlipat.
Yang diwariskan: reseptornya, kepada keturunan klon itu, termasuk \omterm{prop:g12:adaptive-immunity:selection}{sel
ingatannya}. Skala waktunya: beberapa hari untuk sebuah tanggapan,
seumur hidup untuk ingatannya, berbanding bergenerasi-generasi bagi
sebuah \omterm{def:g12:selection-drift-speciation:population}{populasi}. Kesejajarannya berhenti pada penurunan: klon yang
terseleksi mati bersama tubuhnya dan tidak diteruskan kepada generasi
berikutnya.
\end{solution}

\begin{solution}{pb:g12:adaptive-immunity:1}
\textbf{1.} Tak ada apa-apa selama sekitar sepekan, puncak yang
sederhana pada pekan kedua sampai ketiga, lalu menurun pada pekan-pekan
berikutnya.

\textbf{2.} \omterm{prop:g12:adaptive-immunity:antibodies}{Antibodi} dalam dua atau tiga hari, puncak sepuluh kali
lebih tinggi, dan bertahan berbulan-bulan: \omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} yang
ditinggalkan dosis pertama sudah merupakan klon yang besar, sehingga
pelipatannya mulai dari ribuan \omterm{def:g10:cells-common-unit:cell}{sel}, bukan dari segelintir.

\textbf{3.} Virus yang hidup berlipat sebentar di dalam \omterm{def:g10:cells-common-unit:cell}{sel}, sehingga
\omterm{def:g10:cells-common-unit:cell}{sel} yang terinfeksi memajang serpihannya dan \omterm{def:g10:cells-common-unit:cell}{sel} T sitotoksik ikut
terseleksi selain \omterm{prop:g12:adaptive-immunity:antibodies}{antibodinya}; virus yang mati terutama melatih lengan
\omterm{prop:g12:adaptive-immunity:antibodies}{antibodinya} saja.

\textbf{4.} \omterm{prop:g12:adaptive-immunity:antibodies}{Antibodi} yang sudah ada menghadang sebagian besar virusnya;
\omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} B dan T berlipat dalam dua hari; \omterm{def:g10:cells-common-unit:cell}{sel} yang terinfeksi
dibunuh sebelum virusnya menyebar. Jangkitannya dihentikan sebelum ada
gejala.

\textbf{5.} Tiga hari pelipatan yang diam-diam, lalu demam, ruam dan
sakit sementara tanggapan primernya terbangun sepanjang sepekan;
pulih sekitar hari ke-10, dengan \omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} dan imunitas seumur hidup
--- bila ia selamat dari penyulitnya.

\textbf{6.} $1 - 1/15 \approx 93\%$.

\textbf{7.} $15 \times 0.10 = 1.5$: tiap kasus menjangkiti lebih dari
satu orang lain, sehingga penyakitnya dapat menyebar. Tidak terlindungi.

\textbf{8.} $10\% \times \num{50000} = 5000$ orang tidak kebal, dan
1500 di antaranya bayi.

\textbf{9.} 2500 orang rentan; $15 \times 0.05 = 0.75$: kurang dari
satu, sehingga penyakitnya padam.

\textbf{10.} Bayi tak dapat divaksinasi, sehingga satu-satunya
perlindungannya adalah orang dewasa di sekelilingnya yang, dengan
menjadi kebal, tak menyisakan jalan bagi virusnya untuk mencapai mereka.

\textbf{11.} Generasi 3: $1.5^3 \approx 3.4$ kasus; generasi 6:
$1.5^6 \approx 11$.

\textbf{12.} $1 + 1.5 + 2.25 + 3.4 + 5.1 + 7.6 + 11.4 \approx 32$ kasus.

\textbf{13.} Sekitar 30\% dari 32: kira-kira 10 bayi.

\textbf{14.} Tiap kasus menyingkirkan satu orang rentan dan menambah
satu orang kebal, sehingga bagian yang rentan menurun dan begitu pula
cacah yang dijangkiti tiap kasus; dengan 5000 orang rentan, wabahnya
baru akan melambat setelah ratusan kasus, sepanjang berbulan-bulan ---
kecuali vaksinasi turun tangan.

\textbf{15.} Pada 95\%: $0.75^3 \approx 0.4$ dan $0.75^6 \approx 0.18$
kasus; seluruhnya sekitar $1 + 0.75 + 0.56 + \dots \approx 4$ kasus.
Sepuluh kali lebih sedikit kasusnya, dan wabahnya padam sendiri.

\textbf{16.} 800 dalam 8 tahun: 100 per mikroliter tiap tahun, sekitar
8 tiap bulan.

\textbf{17.} \omterm{def:g10:cells-common-unit:cell}{Sel} B memerlukan sitokina \omterm{def:g10:cells-common-unit:cell}{sel} penolong untuk menjadi \omterm{def:g10:cells-common-unit:cell}{sel}
plasma, dan \omterm{def:g10:cells-common-unit:cell}{sel} T sitotoksik memerlukannya untuk berlipat: dengan \omterm{def:g10:cells-common-unit:cell}{sel}
penolong yang terlalu sedikit, tak satu lengan pun diizinkan, walaupun
\omterm{def:g10:cells-common-unit:cell}{selnya} masih ada.

\textbf{18.} Ya: \omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} tetap memerlukan sitokina penolong untuk
berkembang menjadi tanggapan sekunder; tanpa koordinasinya ingatan itu
tak dapat dipakai, dan campak, yang tak berbahaya bagi orang dewasa
yang sehat, dapat mematikan.

\textbf{19.} \omterm{def:g12:selection-drift-speciation:population}{Populasi} \omterm{def:g10:cells-common-unit:cell}{sel} penolongnya tumbuh kembali dari \omterm{def:g10:cells-common-unit:cell}{sel} yang
selamat dan dari sumsumnya, dan \omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} lengan lainnya tak pernah
rusak --- khazanahnya utuh begitu koordinasinya kembali. Virusnya
bertahan di dalam \omterm{def:g10:cells-common-unit:cell}{sel} yang tersembunyi, sehingga menghentikan obatnya
membiarkannya bangkit lagi.

\textbf{20.} Cakupan 93\%; sekitar 32 kasus dari satu musafir pada
90\%; dan \omterm{prop:g12:adaptive-immunity:tcells}{sel T penolong}.
\end{solution}
