\chapter{منعكس الشد والرسالة العصبية}\label{ch:g12:stretch-reflex}

يطرق طبيب \omterm{def:g10:muscles-and-joints:muscle}{الوتر} تحت رضفتك مباشرة بمطرقة مطاطية صغيرة، فتركل قدمك إلى
الأمام قبل أن تحس شيئًا. وثلاثون ميلي ثانية تفصل الطرقة عن الركلة: وهي
أقل من أن تكفي الدماغ، بل أقل من أن تكفي قرارًا. فالإشارة ذهبت من
العضلة إلى النخاع الشوكي وعادت --- دارة من عصبونين ووصلة واحدة بينهما
--- والدارة نفسها هي ما يبقيك منتصبًا، في صمت، كل ثانية تقفها. ويستعمل
هذا الفصل تلك الدارة ليقدّم \omterm{def:g12:stretch-reflex:neuron}{العصبون}، والإشارة الكهربائية التي يحملها،
والملتقى الكيميائي حيث يكلّم \omterm{def:g12:stretch-reflex:neuron}{عصبون} \omterm{def:g12:stretch-reflex:neuron}{العصبون} التالي.

\section{القوس المنعكسة}

\begin{definition}[المنعكس]\label{def:g12:stretch-reflex:reflex}
\emph{المنعكس}\index{منعكس} استجابة لاإرادية سريعة نمطية من منفِّذ
(عضلة أو غدة) لمنبه، تنتجها دارة ثابتة من \omterm{def:g12:stretch-reflex:neuron}{العصبونات}، هي \emph{القوس
المنعكسة}\index{قوس منعكسة}، تمر عبر النخاع الشوكي أو جذع الدماغ من
غير أن تحتاج الدماغ. و\emph{منعكس الشد}\index{منعكس شد} أبسطها: فشدّ
عضلة شدًا مفاجئًا يجعل تلك العضلة نفسها تتقلص.
\end{definition}

\begin{proposition}[دارة منعكس الشد]\label{prop:g12:stretch-reflex:arc}
خمسة عناصر، بالترتيب:
\begin{enumerate}
  \item \emph{المستقبل}: \emph{المغازل العضلية}، وهي أعضاء حسية صغيرة
        تقع بين ألياف العضلة، تشدّ حين تشدّ العضلة وتستجيب بإطلاق
        إشارات؛
  \item و\emph{\omterm{def:g12:stretch-reflex:neuron}{العصبون} الحسي}، الذي يمتد ليفه من المغزل إلى النخاع
        الشوكي، ويدخل بالجذر الظهري؛ وجسم خليته يقع في عقدة عصبية إلى
        جانب النخاع؛
  \item و\emph{مركز التكامل}: وهو المادة الرمادية في النخاع الشوكي،
        حيث يتصل الليف الحسي مباشرة --- عبر \omterm{def:g12:stretch-reflex:synapse}{مشبك} واحد --- \omterm{def:g12:stretch-reflex:neuron}{بالعصبون}
        الحركي للعضلة نفسها، ويثبّط، عبر \omterm{def:g12:stretch-reflex:neuron}{عصبون} بيني، \omterm{def:g12:stretch-reflex:neuron}{العصبونات} الحركية
        للعضلة المضادة؛
  \item و\emph{\omterm{def:g12:stretch-reflex:neuron}{العصبون} الحركي}، الذي يغادر ليفه بالجذر البطني ويمتد
        إلى العضلة؛
  \item و\emph{المنفِّذ}: وهو ألياف العضلة، فتتقلص.
\end{enumerate}
والطرقة على \omterm{def:g10:muscles-and-joints:muscle}{وتر} الرضفة تشدّ عضلة الفخذ؛ فتطلق مغازلها؛ فتطلق \omterm{def:g12:stretch-reflex:neuron}{العصبونات}
الحركية لتلك العضلة؛ فتتقلص وتنبسط الساق، بينما تؤمر العضلات المأبضية
بأن تسترخي.
\end{proposition}
\begin{proof}[شواهد]
قطع الجذر الظهري يبطل \omterm{def:g12:stretch-reflex:reflex}{المنعكس} بينما يمكن أن تجعل العضلة تتقلص بتنبيه
الجذر البطني؛ وقطع الجذر البطني يبطله بينما تظل الألياف الحسية تحمل
إشارات حين تشدّ العضلة: فللقوس مدخل حسي ومخرج حركي. \omterm{def:g12:stretch-reflex:reflex}{والمنعكس} يبقى عند
حيوان قطع نخاعه الشوكي عن دماغه: فالدماغ ليس لازمًا. والتسجيل من ألياف
مفردة يبين إشارات المغزل تبدأ في غضون ميلي ثانية من الشدّ، \omterm{def:g12:stretch-reflex:neuron}{والعصبون}
الحركي يطلق بعد نحو ميلي ثانية من وصول الإشارة الحسية --- وهو تأخر
\omterm{def:g12:stretch-reflex:synapse}{مشبك} واحد.
\end{proof}

\begin{example}[لماذا وجد المنعكس]\label{ex:g12:stretch-reflex:posture}
أنت واقف تتمايل؛ وكل تمايلة تشدّ عضلات جانب من الكاحل، فتطلق مغازلها
ويشدّك تقلصها \omterm{def:g12:stretch-reflex:reflex}{المنعكس} إلى الوراء قبل أن تلاحظ. وأنت تحمل صينية، فيشدّ
ثقل زائد مفاجئ عضلات الذراع ويقسّيها \omterm{def:g12:stretch-reflex:reflex}{المنعكس} في غضون \qty{30}{ms}، قبل
أن تميل الصينية. \omterm{def:g12:stretch-reflex:reflex}{فمنعكس الشد} آلية إتباع: يعارض أي تغير في طول عضلة،
ويعدّل الدماغ حساسيته ليضبط توتر الجسم --- أعلى حين تتأهب، وأدنى حين
تنام.
\end{example}

\section{العصبون ورسالته}

\begin{definition}[العصبون]\label{def:g12:stretch-reflex:neuron}
\emph{العصبون}\index{عصبون} \omterm{def:g10:cells-common-unit:cell}{خلية} متخصصة في الإشارة: جسم \omterm{def:g10:cells-common-unit:cell}{خلية} بنواته،
و\emph{تغصنات} متفرعة تتلقى الإشارات، وليف واحد طويل هو \emph{المحوار}
يحمل إشارة العصبون نفسه إلى نهاياته --- وهي، عند عصبون حركي في الساق،
على بعد متر. وكثير من المحاوير ملفوف في غمد دهني هو \emph{النخاعين}،
تضعه \omterm{def:g10:cells-common-unit:cell}{خلايا} رفيقة، وهو يسرّع النقل. والعصب حزمة من مئات المحاوير أو
آلافها.
\end{definition}

\begin{proposition}[كمون العمل]\label{prop:g12:stretch-reflex:actionpotential}
غشاء \omterm{def:g12:stretch-reflex:neuron}{العصبون} يحمل فرق كمون كهربائي: فالداخل نحو \qty{-70}{mV} قياسًا
بالخارج في الراحة. و\emph{الرسالة العصبية}\index{رسالة عصبية} قطار من
\emph{كمونات العمل}\index{كمون عمل}: أي انقلابات وجيزة لذلك الكمون،
يرتفع كل منها إلى نحو \qty{+30}{mV} ويعود في غضون ميلي ثانية، وتسافر
على طول المحوار من غير أن تضعف. \omterm{prop:g12:stretch-reflex:actionpotential}{وكمون العمل} \emph{الكل أو لا شيء}: فمنبه
دون العتبة لا ينتج شيئًا، وأي منبه فوقها ينتج الإشارة الكاملة نفسها.
ومن ثم فشدة المنبه لا ترمَّز في حجم الإشارات بل في \emph{تواترها}: فالشدّ
الأقوى يجعل المغزل يطلق كمونات عمل أكثر في الثانية.
\end{proposition}
\begin{proof}[شواهد]
المسارٍّ الدقيقة المدخلة في محوار تسجل كمون الراحة، وتسجل، حين تطلق
\omterm{def:g10:cells-common-unit:cell}{الخلية}، شوكات متطابقة لا يتغير شكلها بتغير المنبه ولا بتغير المسافة
المقطوعة. والتسجيلات من ليف مغزل واحد أثناء شدّات متزايدة تبدي شوكات
ثابتة الحجم بمعدلات ترتفع من بضع إلى بضع مئات في الثانية. والمنبه الذي
يقع دون العتبة بقليل لا يعطي شوكة البتة؛ ومضاعفة منبه فوق العتبة أصلًا
تغير المعدل، لا الارتفاع أبدًا.
\end{proof}

\begin{example}[السرعة]\label{ex:g12:stretch-reflex:speed}
\omterm{prop:g12:stretch-reflex:actionpotential}{كمونات العمل} تسافر بسرعة \qty{1}{m/s} في الألياف الرقيقة غير المنخّعة،
وحتى \qty{100}{m/s} في الألياف الغليظة المنخّعة في \omterm{def:g12:stretch-reflex:reflex}{منعكس الشد}. والمسافة
من الركبة إلى النخاع الشوكي والعودة نحو \qty{1.5}{m}: أي نحو
\qty{15}{ms} من السفر، تضيف إليها استجابة المغزل \omterm{def:g12:stretch-reflex:synapse}{ومشبك} واحد وتنشيط
العضلة \qty{15}{ms} أخرى --- وهي \qty{30}{ms} الركلة الركبية. ورسالة
من إصبع قدم إلى الدماغ، على بعد \qty{1.8}{m}، تأخذ نحو \qty{20}{ms}؛
ويأخذ قرار تحريكه وتنفيذه نحو \qty{150}{ms} أخرى.
\end{example}

\section{المشبك}

\begin{definition}[المشبك والناقل العصبي]\label{def:g12:stretch-reflex:synapse}
\emph{المشبك}\index{مشبك} الملتقى الذي تلاقي عنده نهاية محوار \omterm{def:g12:stretch-reflex:neuron}{عصبون}
\omterm{def:g10:cells-common-unit:cell}{الخلية} التالية --- عصبونًا أو ليف عضلة --- عبر \omterm{def:g10:cells-common-unit:organelle}{فجوة} قدرها نحو
\qty{20}{nm}، هي \emph{الشق المشبكي}. والإشارة لا تقفز \omterm{def:g10:cells-common-unit:organelle}{الفجوة}
كهربائيًا: بل \omterm{prop:g12:stretch-reflex:actionpotential}{كمونات العمل} الواصلة تجعل النهاية تحرر، من حويصلات
صغيرة، رسولًا كيميائيًا هو \emph{الناقل العصبي}\index{ناقل عصبي}، في
الشق؛ فيرتبط \emph{بمستقبلات} على غشاء \omterm{def:g10:cells-common-unit:cell}{الخلية} المتلقية، فتغير كمون تلك
\omterm{def:g10:cells-common-unit:cell}{الخلية}؛ ثم يهدَّم أو يستعاد في غضون ميلي ثوان. فالمشبك يحوّل رسالة
كهربائية إلى كيميائية ثم يعيدها، وكمية الناقل المحررة --- ويضبطها
تواتر \omterm{prop:g12:stretch-reflex:actionpotential}{كمونات العمل} الواصلة --- ترمّز شدة الرسالة.
\end{definition}

\begin{proposition}[التنبيه والتثبيط وملتقى العضلة]\label{prop:g12:stretch-reflex:junction}
بحسب الناقل ومستقبله، \emph{ينبّه} \omterm{def:g12:stretch-reflex:synapse}{المشبك} \omterm{def:g12:stretch-reflex:neuron}{العصبون} المتلقي (فيقربه من
عتبته) أو \emph{يثبّطه} (فيبعده عنها). \omterm{def:g12:stretch-reflex:reflex}{ومنعكس الشد} يستعمل الاثنين:
تنبيه \omterm{def:g12:stretch-reflex:neuron}{العصبون} الحركي للعضلة نفسها، وتثبيط \omterm{def:g12:stretch-reflex:neuron}{عصبون} العضلة المضادة عبر
\omterm{def:g12:stretch-reflex:neuron}{العصبون} البيني. وعند \emph{الملتقى العصبي العضلي}، وهو \omterm{def:g12:stretch-reflex:synapse}{المشبك} بين \omterm{def:g12:stretch-reflex:neuron}{عصبون}
حركي وليف عضلة، يكون الناقل \emph{الأستيل كولين}: فكل \omterm{prop:g12:stretch-reflex:actionpotential}{كمون عمل} عند
\omterm{def:g12:stretch-reflex:neuron}{العصبون} الحركي يحرر ما يكفي لإطلاق الليف، فيتقلص. \omterm{def:g12:stretch-reflex:neuron}{والعصبون} الحركي
والألياف التي يأمرها يؤلفان \emph{وحدة حركية}؛ وقوة العضلة يضبطها كم
وحدة تطلق وبأي سرعة.
\end{proposition}
\begin{proof}[شواهد]
الأستيل كولين المطبَّق على ليف عضلة يجعله يتقلص؛ و\emph{الكورار}، وهو
سم السهام، يرتبط بمستقبلات الأستيل كولين عند الليف من غير أن ينشّطها
فيشل كل عضلة بينما تظل الأعصاب تطلق؛ وذيفان الوشيقية يسد تحرر
الحويصلات فيشل كذلك؛ ومبيد حشرات يسد \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيم} الذي يهدّم الأستيل كولين
يجعل العضلات تتقلص بلا ضابط. والإستركنين يسد \omterm{def:g12:stretch-reflex:synapse}{مشابك} النخاع الشوكي
المثبِّطة، فيثير كل شدّ تقلص العضلة والعضلة المضادة معًا: أي اختلاجات.
فكل سم يسمي خطوة من خطوات الملتقى بأن ينزعها.
\end{proof}

\begin{method}[تحليل منعكس أو سم]\label{met:g12:stretch-reflex:analyse}
\begin{enumerate}
  \item تتبع القوس: المستقبل، \omterm{def:g12:stretch-reflex:neuron}{والعصبون} الحسي، والمركز، \omterm{def:g12:stretch-reflex:neuron}{والعصبون}
        الحركي، والمنفِّذ؛ وعيّن أي عنصر تصيبه آفة أو دواء.
  \item وعند كل \omterm{def:g12:stretch-reflex:neuron}{عصبون}، اسأل ما الذي يرمَّز --- تواتر \omterm{prop:g12:stretch-reflex:actionpotential}{كمونات العمل} ---
        وعند كل \omterm{def:g12:stretch-reflex:synapse}{مشبك}، ما الذي يحمل الرسالة --- الناقل وكميته.
  \item وعند السم، جد خطوته: تحرر الناقل، أم المستقبل، أم تهديم
        الناقل، أم \omterm{def:g12:stretch-reflex:synapse}{مشبك} مثبِّط؛ وتنبأ بشلل (فلا شيء يمر) أو باختلاج
        (فلا شيء يوقَف).
  \item وقدّر الزمن: المسافة مقسومة على سرعة النقل، مضافًا إليها نحو
        ميلي ثانية لكل \omterm{def:g12:stretch-reflex:synapse}{مشبك}.
\end{enumerate}
\end{method}

\begin{remark}[رسالة من الشوكات نفسها]\label{rem:g12:stretch-reflex:spikes}
كل رسالة في الجهاز العصبي --- شدّ المغزل، وضوء الشبكية، وأمر الدماغ
بالحركة --- مؤلفة من \omterm{prop:g12:stretch-reflex:actionpotential}{كمونات العمل} نفسها، من نمط الكل أو لا شيء، ولا
تختلف إلا في المعدل وفي الأسلاك التي تحملها؛ والمعنى في أي \omterm{def:g12:stretch-reflex:neuron}{عصبون} يطلق،
لا في ما يطلقه. وعند كل \omterm{def:g12:stretch-reflex:synapse}{مشبك} تصير الرسالة جرعة من مادة كيميائية، ولهذا
استطاع جزيء --- ناقل، أو سم، أو دواء --- أن يدخل المحادثة عند أي نقطة.
والفصل الأخير يتابع الشوكات \omterm{def:g12:stretch-reflex:synapse}{والمشابك} نفسها صعودًا إلى الدماغ، ويتابع
الأمر النازل منه.
\end{remark}

ما \omterm{prop:g12:stretch-reflex:actionpotential}{كمون العمل}؟ وماذا يعني "الكل أو لا شيء"، وكيف ترمَّز الشدة؟

\begin{exercise}[$\star$]\label{exo:g12:stretch-reflex:4}
صف ما يقع عند \omterm{def:g12:stretch-reflex:synapse}{مشبك} من وصول \omterm{prop:g12:stretch-reflex:actionpotential}{كمونات العمل} إلى استجابة \omterm{def:g10:cells-common-unit:cell}{الخلية} المتلقية.

\begin{exercise}[$\star\star$]\label{exo:g12:stretch-reflex:6}

\begin{exercise}[$\star\star$]\label{exo:g12:stretch-reflex:10}

\begin{exercise}[$\star\star\star$]\label{exo:g12:stretch-reflex:13}

\section{مسألة: ثلاثون ميلي ثانية}

\begin{problem}[{مسألة نهاية الأسبوع --- الركلة الركبية توقَّت وتفكك:
ترميز المغزل، وسرعة الألياف، وتأخر المشبك، وكيمياء الملتقى، والسموم
التي تسمي كل خطوة}]
\label{pb:g12:stretch-reflex:1}
تسجل أقطاب النشاط الكهربائي لعضلة فخذ شخص (تخطيط عضل كهربائي) بينما
يطرق \omterm{def:g10:muscles-and-joints:muscle}{وتر} الرضفة. وإشارة العضلة تبدأ بعد \qty{30}{ms} من الطرقة؛ وتتحرك
الساق بعدها \qty{20}{ms}. والمسافة من \omterm{def:g10:muscles-and-joints:muscle}{الوتر} إلى النخاع الشوكي على طول
العصب \qty{75}{cm}، ومثلها في العودة.

\medskip
\noindent\textbf{القسم الأول --- التوقيت.}
\begin{enumerate}
  \item احسب الطول الكلي للمسار العصبي في \omterm{def:g12:stretch-reflex:reflex}{المنعكس}.
  \item خذ استجابة المغزل وتنشيط العضلة يحتاج كل منهما \qty{4}{ms}،
        \omterm{def:g12:stretch-reflex:synapse}{والمشبك} \qty{1}{ms}. فكم من \qty{30}{ms} هو نقل، وما سرعة
        النقل؟
  \item وعند شخص يزيد طوله \qty{20}{cm}، بالألياف نفسها، تنبأ بالكمون.
  \item ولماذا تتحرك الساق بعد \qty{20}{ms} من بدء الإشارة الكهربائية
        عند العضلة؟
  \item ويقال لشخص أن يتوقع الطرقة وأن "يفكر في" إيقاف الركلة؛ فلا
        تتغير الركلة. فسّر بتوقيت رسالة إلى الدماغ ومنه (\qty{1.2}{m}
        زائدة من الألياف، \omterm{def:g12:stretch-reflex:synapse}{ومشابك} عدة).
\end{enumerate}

\medskip
\noindent\textbf{القسم الثاني --- الترميز.}
ليف المغزل يطلق 20 \omterm{prop:g12:stretch-reflex:actionpotential}{كمون عمل} في الثانية في الراحة، و200 أثناء الطرقة.
\begin{enumerate}[resume]
  \item كم \omterm{prop:g12:stretch-reflex:actionpotential}{كمون عمل} يرسل المغزل في \qty{10}{ms} التالية للطرقة؟ وما
        الذي يتغير بين الراحة والطرقة، وما الذي لا يتغير؟
  \item كل \omterm{prop:g12:stretch-reflex:actionpotential}{كمون عمل} يصل \omterm{def:g12:stretch-reflex:synapse}{المشبك} يحرر كمية ثابتة من الناقل. فبأي معامل
        ترتفع كمية الناقل المحررة في الثانية أثناء الطرقة؟
  \item \omterm{def:g12:stretch-reflex:neuron}{والعصبون} الحركي لا يطلق إلا إذا وصل ناقل كاف في غضون ميلي ثوان
        قليلة. فسّر لماذا لا يجعل معدل الراحة العضلة تتقلص، ويجعلها
        معدل الطرقة تتقلص.
  \item وطرقة أرفق تعطي 100 في الثانية. تنبأ باستجابة \omterm{def:g12:stretch-reflex:neuron}{العصبون} الحركي
        وبقوة الركلة، مستعينًا بالوحدات الحركية.
  \item فسّر لماذا ما كان يمكن أن ترمَّز رسالة المغزل في حجم كمونات
        عمله.
\end{enumerate}

\medskip
\noindent\textbf{القسم الثالث --- الملتقى.}
\begin{enumerate}[resume]
  \item سمّ الناقل عند الملتقى العصبي العضلي وصف مصيره بعد أن يعمل.
  \item ويحقن الكورار: فلا يبدي تخطيط العضل إشارة بعد الطرقة، لكن
        العصب ما زال يحمل كمونات عمل. فأي خطوة سدّت؟
  \item وذيفان الوشيقية بدلًا منه: التخطيط نفسه، وإشارة العصب نفسها.
        فأي خطوة، وكيف تميز بين الاثنين في المختبر؟
  \item ومخدر يسد \omterm{prop:g12:stretch-reflex:actionpotential}{كمونات العمل} في الألياف الحسية. فأي التسجيلات تختفي،
        وأيها يبقى؟
  \item فسّر لماذا تنتج الثلاثة كلها ساقًا رخوة لكن بثلاث آليات
        مختلفة.
\end{enumerate}

\medskip
\noindent\textbf{القسم الرابع --- المركز.}
\begin{enumerate}[resume]
  \item أثناء الركلة يسكت تخطيط العضلات المأبضية، ويهبط دون مستوى
        راحته. فسّر \omterm{def:g12:stretch-reflex:neuron}{بالعصبون} البيني.
  \item ومريض قطع نخاعه الشوكي فوق مستوى \omterm{def:g12:stretch-reflex:reflex}{المنعكس} ما زالت عنده ركلة
        ركبية --- مبالغ فيها. فماذا يبين وجود \omterm{def:g12:stretch-reflex:reflex}{المنعكس}، وماذا توحي به
        المبالغة عن دور الدماغ الطبيعي؟
  \item وعند مريض غاب منعكسه في جانب واحد، اذكر عناصر القوس التي قد
        تكون متضررة، واختبارًا واحدًا لتحديد موضع الضرر.
  \item ولماذا يفحص الأطباء هذا \omterm{def:g12:stretch-reflex:reflex}{المنعكس} روتينيًا، في ثوان قليلة، عند كل
        مريض؟
  \item اذكر النتيجة: سرعة النقل في ألياف \omterm{def:g12:stretch-reflex:reflex}{المنعكس}، والمقدار الذي يرمّز
        شدة الشدّ، والخطوة الكيميائية الواحدة التي يعمل عندها الكورار.
\end{enumerate}
\end{problem}
