\chapter{Geslachtelijke voortplanting}\label{ch:g8:sexual-reproduction}

herfst, een zonnebloem die haar volgepakte gezicht naar de bijen draait:
overal in de levende wereld wordt er enorm veel \omterm{prop:g3:balanced-diet:jobs}{energie} aan één project
besteed --- de volgende generatie. Eerdere jaren verzamelden de stukken
(\cref{ch:g4:animal-reproduction},
\cref{ch:g5:flower-to-fruit}); dit jaar zet ze in elkaar tot één theorie,
en stelt de scherpere vraag: waarom eigenlijk \emph{twee} ouders?

\section{Eén mechanisme door de hele levende wereld}

\begin{definition}[Gameten]\label{def:g8:sexual-reproduction:gametes}
De voortplantingscellen uit
\cref{def:g4:animal-reproduction:sexual} dragen de algemene naam
\emph{gameten}\index{gameet}: de mannelijke gameet (de \omterm{def:g4:animal-reproduction:sexual}{zaadcel} bij \omterm{def:g1:animals-around-us:animal}{dieren},
de \omterm{def:g5:first-look-at-cells:cell}{cel} van de stuifmeelkorrel bij bloeiende \omterm{def:g1:plants-around-us:plant}{planten}) en de vrouwelijke
gameet (de \omterm{def:g4:animal-reproduction:sexual}{eicel}, in eierstok of zaadknop). Gameten zijn losse cellen
(\cref{ch:g5:first-look-at-cells}), en elk is een half begin: alleen gaat
een gameet dood; samengevoegd maken er twee een leven.
\end{definition}

\begin{proposition}[De onveranderlijke kern]\label{prop:g8:sexual-reproduction:core}
Waar de \omterm{def:g4:animal-reproduction:sexual}{geslachtelijke voortplanting} ook draait --- vis of spar, slak of
mens --- de kern is onveranderlijk:
\begin{enumerate}
  \item twee ouders maken \omterm{def:g8:sexual-reproduction:gametes}{gameten}, mannelijke en vrouwelijke;
  \item de \emph{\omterm{def:g5:flower-to-fruit:steps}{bevruchting}}: één mannelijke \omterm{def:g8:sexual-reproduction:gametes}{gameet} verenigt zich met
        één vrouwelijke \omterm{def:g8:sexual-reproduction:gametes}{gameet}, en zo ontstaat één nieuwe \omterm{def:g5:first-look-at-cells:cell}{cel} --- de
        \emph{bevruchte \omterm{def:g4:animal-reproduction:sexual}{eicel}}, of \emph{zygote}\index{zygote};
  \item de \omterm{prop:g8:sexual-reproduction:core}{zygote} deelt zich keer op keer
        (\cref{prop:g5:first-look-at-cells:growth}) en bouwt zo een nieuw
        individu dat van \emph{allebei} de ouders iets meedraagt
        (\cref{rem:g4:animal-reproduction:mix}).
\end{enumerate}
Rond die kern is al het andere variatie: waar de \omterm{def:g8:sexual-reproduction:gametes}{gameten} elkaar treffen,
hoeveel het er zijn, wie voor de jongen zorgt.
\end{proposition}
\admitted

\begin{example}[De variaties, ingedeeld]\label{ex:g8:sexual-reproduction:variations}
De oude verzameling laat zich netjes onder de kern archiveren. Uitwendige
tegenover inwendige \omterm{def:g5:flower-to-fruit:steps}{bevruchting}
(\cref{prop:g4:animal-reproduction:where}): de ontmoetingsplaats.
\omterm{def:g4:animal-reproduction:ovivivi}{Eierleggend} tegenover \omterm{def:g4:animal-reproduction:ovivivi}{levendbarend}
(\cref{def:g4:animal-reproduction:ovivivi}): het onderdak van de \omterm{prop:g8:sexual-reproduction:core}{zygote}.
De miljoenen van de kabeljauw tegenover het enkele kalf van de olifant
(\cref{prop:g4:animal-reproduction:strategies}): de aantallenboekhouding.
\omterm{def:g5:flower-to-fruit:steps}{Bestuiving} door bij of wind
(\cref{def:g5:flower-to-fruit:steps}): de reisdienst van de mannelijke
\omterm{def:g8:sexual-reproduction:gametes}{gameet}. Eén mechanisme, een catalogus vol instellingen.
\end{example}

\begin{remark}[Planten draaien dezelfde kern]\label{rem:g8:sexual-reproduction:plants}
Zet \cref{ch:g5:flower-to-fruit} naast
\cref{ch:g4:animal-reproduction} en de overeenkomst is exact: \omterm{def:g3:flowers-fruits-seeds:parts}{meeldraden}
en \omterm{def:g3:flowers-fruits-seeds:parts}{stamper} maken de \omterm{def:g8:sexual-reproduction:gametes}{gameten}, de stuifmeelbuis is de gang van de inwendige
\omterm{def:g5:flower-to-fruit:steps}{bevruchting}, en het \omterm{def:g2:from-seed-to-plant:seed}{zaad} huisvest het \omterm{def:g5:human-reproduction:pregnancy}{embryo} van de \omterm{def:g1:plants-around-us:plant}{plant} met zijn
lunchpakket (\cref{def:g2:from-seed-to-plant:seed}). Alleen de \omterm{def:g4:plants-without-seeds:vegetative}{vegetatieve
voortplanting} (\cref{def:g4:plants-without-seeds:vegetative}) staat apart
--- één ouder, geen \omterm{def:g8:sexual-reproduction:gametes}{gameten}, exacte kopieën: het contrast dat de vraag van
de volgende paragraaf scherp maakt.
\end{remark}

\section{Waarom twee ouders?}

\begin{proposition}[Geslachtelijke voortplanting schudt]\label{prop:g8:sexual-reproduction:shuffle}
De aardbeienkopieën van de \omterm{ex:g4:plants-without-seeds:runner}{uitlopers}
(\cref{rem:g4:plants-without-seeds:copy}) laten zien wat één ouder
oplevert: eenvormigheid. Twee ouders leveren het tegendeel. Omdat elke
\omterm{prop:g8:sexual-reproduction:core}{zygote} van elke ouder één \omterm{def:g8:sexual-reproduction:gametes}{gameet} combineert --- en omdat, zoals een later
hoofdstuk zal laten zien, zelfs geen twee \omterm{def:g8:sexual-reproduction:gametes}{gameten} van één ouder precies
dezelfde erfelijke inhoud dragen --- deelt de \omterm{def:g4:animal-reproduction:sexual}{geslachtelijke voortplanting}
elk nakomelingetje een \emph{nieuwe combinatie} uit. Broers en zussen
verschillen; geen twee \omterm{prop:g2:from-seed-to-plant:cycle}{kiemplantjes} uit één peul zijn gelijk; en de
verscheidenheid binnen elke \omterm{def:g5:biodiversity:species}{soort}
(\cref{rem:g5:biodiversity:scales}) is grotendeels het werk van dat
schudden.
\end{proposition}
\admitted

\begin{example}[De twee weddenschappen van de boomgaard]\label{ex:g8:sexual-reproduction:bets}
De boomgaard uit \cref{rem:g4:plants-without-seeds:copy} speelt beide
spellen met opzet. Entingen en \omterm{met:g4:plants-without-seeds:cutting}{stekken}: één ouder, exacte kopieën van de
uitstekende boom --- en één gedeelde zwakke plek, zodat een ziekte die er
één velt ze allemaal kan vellen. Pitjes: twee ouders, en elke zaailing een
verse combinatie --- de meeste onopvallend, sommige slechter, een enkele
beter dan allebei de ouders, en geen twee gelijk voor een ziekte om weg te
vagen. Kopiëren zet het heden op de bank; schudden verkent de toekomst.
\end{example}

\begin{remark}[Verscheidenheid is de reserve van de soort]\label{rem:g8:sexual-reproduction:reserve}
Waarom het schudden ook buiten de boomgaard telt:
\cref{prop:g5:biodiversity:web} liet zien hoe webben klappen opvangen door
verscheidenheid aan \omterm{def:g5:biodiversity:species}{soorten}; binnen één \omterm{def:g5:biodiversity:species}{soort} speelt de verscheidenheid
aan individuen dezelfde rol. Als de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} omslaan --- een nieuwe
ziekte, een strengere winter --- heeft een gevarieerde \omterm{def:g5:biodiversity:species}{soort} meer kans dat
er iemand bij zit die op de nieuwe voorwaarden past. Wat dat betekent voor
de geschiedenis van het leven, zal \cref{ch:g9:how-species-change}
uitspreken; hou hier alleen vast dat de \omterm{def:g4:animal-reproduction:sexual}{geslachtelijke voortplanting} de
reserve gevuld houdt.
\end{remark}

\section{De voortplanting van elke soort lezen}

\begin{method}[Het voortplantingsdossier]\label{met:g8:sexual-reproduction:dossier}
Stel voor elke \omterm{def:g5:biodiversity:species}{soort} samen:
\begin{enumerate}
  \item \omterm{def:g8:sexual-reproduction:gametes}{gameten}: wie maakt welke, en waar;
  \item ontmoeting: uitwendige of inwendige \omterm{def:g5:flower-to-fruit:steps}{bevruchting}, en de reisdienst
        (water, \omterm{prop:g4:animal-reproduction:where}{paring}, bestuiver, wind);
  \item onderdak: eieren gelegd of jongen gedragen; met welke proviand;
  \item aantallen en zorg: de positie op de strategielijn
        (\cref{prop:g4:animal-reproduction:strategies});
  \item seizoen: wanneer, en waardoor getimed --- de vraag van het
        volgende hoofdstuk.
\end{enumerate}
\end{method}

\begin{example}[Twee dossiers naast elkaar]\label{ex:g8:sexual-reproduction:dossiers}
De stekelbaars (\cref{exo:g4:animal-reproduction:10}): \omterm{def:g8:sexual-reproduction:gametes}{gameten} in het
water losgelaten, uitwendige \omterm{def:g5:flower-to-fruit:steps}{bevruchting}, eitjes in een gebouwd nest,
enkele tientallen, bewaakt door de vader, in het voorjaar. De kers:
\omterm{def:g8:sexual-reproduction:gametes}{gameten} in de bloesem, inwendige \omterm{def:g5:flower-to-fruit:steps}{bevruchting} door de stuifmeelbuis
omlaag, \omterm{def:g5:human-reproduction:pregnancy}{embryo}'s ondergebracht in \omterm{prop:g3:flowers-fruits-seeds:transform}{zaden} binnen een \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht}, duizenden,
van proviand voorzien en daarna aan de bezorgdienst van de \omterm{prop:g3:sorting-living-things:groups}{vogels}
overgelaten (\cref{exo:g6:how-plants-spread:12}), getimed op april.
Dezelfde vorm, voor elke \omterm{def:g5:biodiversity:species}{soort}: het dossier is de theorie in gebruik.
\end{example}

\section{Probleem: Het veredelingsprogramma}

\begin{problem}[{Weekendprobleem --- een jaar van een plantenveredelaar,
gedraaid op de theorie}]\label{pb:g8:sexual-reproduction:1}
Een veredelaarster wil een nieuwe appel: knapperig als ras A, bestand
tegen ziekte als ras B. Haar gereedschap is precies dat van dit hoofdstuk.

\medskip\noindent\textbf{Deel I --- De kruising.}
\begin{enumerate}
  \item Ze neemt met een penseel \omterm{def:g3:flowers-fruits-seeds:parts}{stuifmeel} van de \omterm{def:g3:flowers-fruits-seeds:parts}{meeldraden} van A en
        raakt daarmee de \omterm{def:g3:flowers-fruits-seeds:parts}{stampers} van B aan, en verpakt de \omterm{def:g3:flowers-fruits-seeds:parts}{bloemen} daarna
        in zakjes tegen de bijen. Noem de rol van elke stap in de kern ---
        en waarom de zakjes ertoe doen
        (\cref{met:g5:flower-to-fruit:diagnose} gebruikte hetzelfde
        penseel).
  \item Wat groeit er na haar bezoek in de \omterm{prop:g3:flowers-fruits-seeds:transform}{vrucht} van B, en welke twee
        erfenissen combineert elk pitje?
  \item Ze zaait driehonderd pitjes. Waarom verwacht ze driehonderd
        \emph{verschillende} bomen, volgens
        \cref{prop:g8:sexual-reproduction:shuffle}?
  \item De meeste zaailingen zullen onopvallend zijn, een paar slechter
        dan allebei de ouders, en misschien één precies goed. Welk
        voorbeeld uit het hoofdstuk voorspelde precies die spreiding?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De winnaar vastleggen.}
\begin{enumerate}[resume]
  \item Jaar acht: één boom draagt knapperige appels die tegen ziekte
        bestand zijn. Wat gebeurt er met de gewilde combinatie als ze
        zijn pitjes zaait --- en waarom?
  \item In plaats daarvan vermeerdert ze de winnaar door te enten ---
        welke voortplanting is dat, en wat garandeert die?
  \item Haar hele toekomstige boomgaard zal uit kopieën van die ene boom
        bestaan. Noem het risico dat ze daarmee aanvaardt, en de
        uitdrukking van het hoofdstuk ervoor.
  \item Stel de gebruikelijke verzekering voor: wat zou haar boomgaard
        daarnaast moeten aanhouden, en waarom echoot die strategie
        \cref{rem:g8:sexual-reproduction:reserve}?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De theorie,
doorgelicht.}
\begin{enumerate}[resume]
  \item Haar programma gebruikte allebei de weddenschappen van de
        boomgaard achter elkaar. Zeg welke fase welke gebruikte, en wat
        elk haar opleverde.
  \item Een collega kweekt op dezelfde manier stekelbaarzen --- ouders
        kiezen, de geschudde jongen grootbrengen, de beste houden. Noem de
        rubrieken van het dossier (\cref{met:g8:sexual-reproduction:dossier})
        waar de visserijpraktijk van de appelpraktijk verschilt, en de
        stappen van de kern waar dat niet kan.
  \item De veredelaarster zegt: ``de seks stelt voor, de selectie
        beschikt --- ik kies alleen uit wat het schudden uitdeelt''.
        Verbind haar slogan met de verscheidenheidsreserve uit
        \cref{rem:g8:sexual-reproduction:reserve}.
  \item Eén zin ter afsluiting: waar had het programma de \omterm{def:g4:animal-reproduction:sexual}{geslachtelijke
        voortplanting} \emph{voor} nodig, en waar de vegetatieve?
\end{enumerate}
\end{problem}
