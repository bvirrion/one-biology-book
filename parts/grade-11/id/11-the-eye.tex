\chapter{Mata dan Fotoreseptornya}\label{ch:g11:the-eye}

Pada malam tanpa bulan, jauh dari lampu mana pun, kamu dapat melihat
jalannya tetapi tidak warnanya; jaket merah seorang teman hanyalah
bentuk kelabu. Tataplah lurus sebuah bintang yang samar dan bintang itu
lenyap; tengoklah sedikit ke samping dan ia muncul lagi. Kedua keganjilan
itu berasal dari retina, yaitu lembar tipis di belakang \omterm{def:g11:the-eye:parts}{mata} tempat
cahaya menjadi isyarat saraf: dua macam \omterm{def:g10:cells-common-unit:cell}{sel} penerima, yang tersebar tak
merata, satu buta warna dan yang lain buta dalam gelap. Bab ini
membongkar \omterm{def:g11:the-eye:parts}{mata} dari kornea sampai saraf optiknya, dan membaca, dalam
\omterm{def:g10:universal-dna:gene}{gen} pigmen retinanya, sepotong sejarah evolusi kita.

\section{Mata sebagai alat optik}

\begin{definition}[Bagian mata]\label{def:g11:the-eye:parts}
\emph{Mata}\index{mata} adalah bola bergaris tengah sekitar
\qty{24}{mm}. Cahaya masuk lewat \emph{kornea} yang bening, melewati
\emph{pupil} --- yaitu bukaan pada \emph{iris} yang berwarna, yang
melebar dalam gelap dan menyempit dalam cahaya terang --- dan
\emph{lensa}, menyeberangi jeli bening \emph{badan kaca}, lalu mencapai
\emph{retina}, yaitu lapisan peka cahaya yang melapisi bagian belakang
bolanya. Kornea dan lensanya bersama-sama membentuk, pada retinanya,
bayangan kecil yang terbalik dari pemandangannya; lensanya, yang
kelengkungannya dapat diubah otot, menyetel fokusnya dari benda jauh ke
benda dekat (\emph{akomodasi}). \emph{Saraf optik} meninggalkan bagian
belakang matanya sambil membawa isyarat retinanya ke otak.
\end{definition}

\begin{example}[Fokus, dan cacatnya]\label{ex:g11:the-eye:focus}
\omterm{def:g11:the-eye:parts}{Mata} yang mengendur memfokuskan benda jauh pada retinanya; untuk
membaca pada \qty{25}{cm} lensanya harus menggembung. \omterm{def:g11:the-eye:parts}{Mata} yang sedikit
terlalu panjang memfokuskan benda jauh di depan retinanya --- yaitu
rabun jauh, yang dibetulkan lensa kacamata cekung; \omterm{def:g11:the-eye:parts}{mata} yang terlalu
pendek, atau lensa yang mengaku seiring umur, tidak dapat memfokuskan
benda dekat --- yaitu rabun dekat, yang dibetulkan lensa cembung. Pada
tiap halnya retinanya utuh: cacatnya bersifat optik, dan kacamatanya
menaruh kembali bayangannya di tempat penerimanya berada.
\end{example}

\section{Retinanya}

\begin{definition}[Fotoreseptor]\label{def:g11:the-eye:photoreceptors}
Retinanya memuat dua macam \omterm{def:g10:cells-common-unit:cell}{sel}
\emph{fotoreseptor}\index{fotoreseptor}, yang dinamai menurut
bentuknya. \emph{Sel batang}\index{sel batang}, yang berjumlah sekitar
120 juta, menanggapi cahaya yang sangat redup tetapi semuanya dengan
pigmen yang sama: \omterm{def:g10:cells-common-unit:cell}{sel} itu memberi penglihatan dalam bayangan kelabu
pada malam hari. \emph{Sel kerucut}\index{sel kerucut}, yang berjumlah
sekitar 6 juta, menuntut cahaya yang lebih terang dan hadir dalam tiga
jenis, yang masing-masing berpigmen paling peka terhadap bagian
spektrum yang berbeda: \omterm{def:g10:cells-common-unit:cell}{sel} itu memberi penglihatan siang hari dan
warna. Tiap fotoreseptornya memuat sebuah \emph{fotopigmen} --- yaitu
\omterm{def:g11:gene-expression:protein}{protein}, sebuah \emph{opsin}\index{opsin}, yang memegang molekul kecil
penyerap cahaya yang berasal dari vitamin A --- yang perubahan
bentuknya saat menyerap sebuah foton memulai tanggapan listrik \omterm{def:g10:cells-common-unit:cell}{selnya}.
\end{definition}

\begin{proposition}[Penataan retinanya]\label{prop:g11:the-eye:organisation}
\omterm{def:g11:the-eye:photoreceptors}{Fotoreseptornya} terletak di belakang retinanya, menempel pada sebuah
lapisan gelap; cahayanya harus menyeberangi lapisan lainnya untuk
mencapainya. Isyaratnya berpindah ke \emph{\omterm{def:g10:cells-common-unit:cell}{sel} bipolar}, lalu ke
\emph{\omterm{def:g10:cells-common-unit:cell}{sel} ganglion}, yang serabut panjangnya --- sekitar sejuta ---
berkumpul menjadi saraf optiknya. Sebarannya tidak merata: pada
\emph{fovea}, yaitu lekuk tengah kecil yang ditujukan \omterm{def:g11:the-eye:parts}{matanya}, hanya
ada \omterm{def:g11:the-eye:photoreceptors}{sel kerucut}, yang berjejal rapat dan masing-masing terhubung ke \omterm{def:g10:cells-common-unit:cell}{sel}
ganglionnya sendiri, sehingga memberi penglihatan yang paling tajam;
menjauhi pusatnya, \omterm{def:g11:the-eye:photoreceptors}{sel batang} berkuasa dan banyak di antaranya berbagi
satu \omterm{def:g10:cells-common-unit:cell}{sel} ganglion, sehingga memberi kepekaan dengan harga rinciannya.
Di tempat saraf optiknya keluar, sama sekali tidak ada \omterm{def:g10:cells-common-unit:cell}{sel} penerima:
itulah \emph{bintik buta}.
\end{proposition}
\admitted

\begin{example}[Kedua keganjilannya dijelaskan]\label{ex:g11:the-eye:oddities}
Bintang yang samar lenyap ketika ditatap langsung karena foveanya hanya
punya \omterm{def:g11:the-eye:photoreceptors}{sel kerucut}, yang menuntut cahaya lebih banyak daripada yang
disediakan bintangnya; jika ditengok dari samping, cahayanya jatuh pada
\omterm{def:g11:the-eye:photoreceptors}{sel batang}, yang mengenalinya. Jaketnya kelabu pada malam hari karena
\omterm{def:g11:the-eye:photoreceptors}{sel batang} yang melihatnya punya satu pigmen saja: \omterm{def:g10:cells-common-unit:cell}{sel} itu dapat
melaporkan berapa banyak cahayanya, bukan panjang gelombang mana.
Warnanya adalah perbandingan antara ketiga jenis \omterm{def:g11:the-eye:photoreceptors}{sel kerucutnya}, dan
\omterm{def:g11:the-eye:photoreceptors}{sel kerucutnya} bisu dalam gelap.
\end{example}

\section{Warna: tiga pigmen}

\begin{proposition}[Penglihatan warna]\label{prop:g11:the-eye:colour}
Tiap jenis \omterm{def:g11:the-eye:photoreceptors}{sel kerucutnya} memuat satu dari tiga \omterm{def:g11:the-eye:photoreceptors}{opsin}, yang puncak
serapannya mendekati \qty{420}{nm} (kerucut S, biru), \qty{530}{nm}
(kerucut M, hijau) dan \qty{560}{nm} (kerucut L, merah). Sebuah cahaya
tertentu memancing ketiga jenisnya dalam perbandingan yang bergantung
pada panjang gelombangnya; otaknya membaca warnanya dari perbandingan
itu. Orang yang kekurangan satu jenis kerucut mengacaukan warna yang
dibedakan jenis itu: tanpa kerucut L atau M yang bekerja, merah dan
hijau sulit dibedakan --- yaitu \emph{buta warna merah--hijau}, yang
mengenai sekitar 8\% lelaki dan 0.5\% perempuan.
\end{proposition}
\admitted

\begin{example}[Membaca lengkungnya]\label{ex:g11:the-eye:curves}
Cahaya \qty{500}{nm}: kerucut S pada 8\%, M pada 75\%, L pada 40\% ---
terlihat biru kehijauan. Cahaya \qty{620}{nm}: S pada 0, M pada 10\%, L
pada 42\% --- merah. Orang tanpa kerucut L hanya menerima, bagi
keduanya, sebuah nilai S dan sebuah nilai M, dan cahaya yang kedua
memberi S~0, M~10: nyaris tak terbedakan dari hijau yang redup. Warna
yang akan dibedakan jenis yang hilang itu meruntuh menjadi satu.
\end{example}

\section{Gen opsinnya: sebuah keluarga dengan sejarah}

\begin{proposition}[Gen opsinnya]\label{prop:g11:the-eye:genes}
Tiap \omterm{def:g11:the-eye:photoreceptors}{opsinnya} disandikan \omterm{def:g10:universal-dna:gene}{gennya} sendiri. \omterm{def:g10:universal-dna:gene}{Gen} \omterm{def:g11:the-eye:photoreceptors}{opsin} S terletak pada
\omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} biasa; \omterm{def:g10:universal-dna:gene}{gen} \omterm{def:g11:the-eye:photoreceptors}{opsin} M dan L terletak berdampingan pada \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}
X. \omterm{def:g11:gene-expression:protein}{Protein} L dan M hanya berbeda pada 15 dari 364 \omterm{def:g10:chemistry-of-life:families}{asam aminonya} (96\%
serupa), sedangkan \omterm{def:g11:gene-expression:protein}{protein} S berbeda pada lebih dari separuh
kedudukannya terhadap keduanya. Karena berada pada X, \omterm{def:g10:universal-dna:gene}{alel} M atau L
yang tidak bekerja terungkapkan pada setiap lelaki yang membawanya dan
hanya pada perempuan yang membawa dua --- yaitu pola
\cref{ch:g11:genes-and-disease}, dan alasan mengapa buta warna
merah--hijau enam belas kali lebih lazim pada lelaki.
\end{proposition}
\admitted

\begin{proposition}[Sebuah keluarga gen dari penggandaan]\label{prop:g11:the-eye:family}
Ketiga \omterm{def:g10:universal-dna:gene}{gen} \omterm{def:g11:the-eye:photoreceptors}{opsinnya}, dan \omterm{def:g10:universal-dna:gene}{gen} pigmen \omterm{def:g11:the-eye:photoreceptors}{sel batangnya}, adalah versi satu
\omterm{def:g10:universal-dna:gene}{gen} moyang yang disalin lewat \emph{penggandaan} lalu memencar oleh
\omterm{def:g11:mutations:mutation}{mutasi}. Derajat keserupaannya menarikhkan salinannya: garis S dan L/M
berpisah sangat awal dalam sejarah \omterm{def:g10:common-ancestry:bodyplan}{vertebrata}, sekitar 500 juta tahun
lalu; \omterm{def:g10:universal-dna:gene}{gen} L dan M muncul dari satu penggandaan sekitar 30 sampai 40
juta tahun lalu pada moyang primata Dunia Lama, yang keturunannya ---
monyet, kera, manusia --- punya tiga jenis kerucut sedangkan sebagian
besar mamalia lain punya dua.
\end{proposition}
\begin{proof}[Bukti percobaan]
Perbandingan urutannya: \omterm{def:g10:universal-dna:gene}{gen} L dan M 96\% serupa dan berdampingan pada
\omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X --- yaitu tanda khas penggandaan bersambung yang belakangan;
masing-masing sekitar 43\% serupa dengan \omterm{def:g10:universal-dna:gene}{gen} S, dan ketiganya sekitar
40\% serupa dengan \omterm{def:g10:universal-dna:gene}{gen} \omterm{def:g11:the-eye:photoreceptors}{opsin} \omterm{def:g11:the-eye:photoreceptors}{sel batangnya}, yaitu salinan yang lebih
jauh. Anjing, sapi dan sebagian besar mamalia hanya punya satu \omterm{def:g10:universal-dna:gene}{gen}
\omterm{def:g11:the-eye:photoreceptors}{opsin} terpaut X, pada kedudukan tempat primata punya dua; monyet Dunia
Baru punya satu, dengan beberapa \omterm{def:g10:universal-dna:gene}{alel}, dan hanya betina yang membawa
dua \omterm{def:g10:universal-dna:gene}{alel} berbeda yang melihat tiga warna. Cacah \omterm{def:g10:universal-dna:gene}{gennya} cocok dengan
cacah kerucutnya pada tiap garis keturunannya, dan pohon \omterm{def:g10:universal-dna:gene}{gennya} cocok
dengan pohon \omterm{def:g10:biodiversity-scales:species}{spesiesnya}.
\end{proof}

\begin{example}[Apa yang diubah satu penggandaan]\label{ex:g11:the-eye:duplication}
Monyet dengan dua jenis kerucut melihat hutannya dalam dua dimensi
warna, dan buah yang masak sulit dipilah dari dedaunannya. Setelah
penggandaan L/M-nya, satu salinannya dapat bermutasi ke arah panjang
gelombang yang lebih panjang sementara salinan yang lain menjaga yang
asli --- sehingga muncul kerucut ketiga, dan merah yang masak
berhadapan dengan hijau. \omterm{def:g11:mutations:mutation}{Mutasi} yang kemudian merusak satu salinannya
pada 8\% lelaki sekadar mengembalikan \omterm{def:g11:the-eye:parts}{matanya} ke keadaan dua kerucut moyangnya.
\end{example}

\begin{method}[Menalar tentang sebuah kasus penglihatan warna]\label{met:g11:the-eye:case}
\begin{enumerate}
  \item Jenis kerucut mana yang hilang atau berubah? Bacalah dari warna
        yang dikacaukannya (merah--hijau: L atau M; biru--kuning, yang
        langka: S).
  \item \omterm{def:g10:universal-dna:gene}{Gen} yang mana, dan pada \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} yang mana? L dan M pada X, S
        di tempat lain.
  \item Terapkan pewarisannya: \omterm{def:g11:genes-and-disease:genetic}{resesif} terpaut X bagi merah--hijau ---
        anak lelaki ibu \omterm{def:g11:genes-and-disease:genetic}{pembawa} terkena satu kali dari dua, anak
        perempuan ayah yang terkena semuanya menjadi \omterm{def:g11:genes-and-disease:genetic}{pembawa}.
  \item Hubungkan dengan keluarga \omterm{def:g10:universal-dna:gene}{gennya}: sebuah salinan yang hilang,
        atau salinan L dan M yang dibuat terlalu serupa oleh sebuah
        pertukaran antara \omterm{def:g10:universal-dna:gene}{gen} yang berdampingan itu.
\end{enumerate}
\end{method}

\begin{remark}[Dari mata ke otak]\label{rem:g11:the-eye:tobrain}
Retinanya tidak melihat; ia mengubah dan mulai memilah. Sejuta serabut
saraf optiknya membawa lukisan yang tersandi --- kontras, tepi, nisbah
tanggapan kerucutnya --- ke otak, tempat bab berikutnya mengambilnya.
Apa yang kita sebut melihat berlangsung di sana, yang dibangun di atas
isyarat \omterm{def:g10:cells-common-unit:cell}{sel} yang dilukiskan di sini dan tidak di atas yang lain: orang
yang retinanya bekerja tetapi otak penglihatannya rusak akan buta, dan
orang yang otak penglihatannya bekerja tetapi \omterm{def:g11:the-eye:photoreceptors}{fotoreseptornya} telah
mati juga buta.
\end{remark}

\section{Soal: Tiga Pigmen dan Satu Penggandaan}

\begin{problem}[{Soal akhir pekan --- penglihatan warna yang
dibongkar: tanggapan kerucutnya terhadap sebuah spektrum, sebuah
keluarga dengan buta warna, gen opsin yang dibandingkan, dan ketajaman
yang dapat dicapai sebuah fovea}]
\label{pb:g11:the-eye:1}
Pakailah gambar serapannya. Sebuah uji menyajikan cahaya \qty{470}{nm},
\qty{530}{nm}, \qty{590}{nm} dan \qty{650}{nm}.

\medskip
\noindent\textbf{Bagian I --- Kerucutnya menanggapi.}
\begin{enumerate}
  \item Untuk tiap cahayanya, bacalah tanggapan hampiran kerucut S, M
        dan L-nya.
  \item Cahaya mana yang memancing ketiga jenisnya? Mana yang memancing
        satu jenis saja?
  \item Orang yang kekurangan kerucut L hanya menerima nilai S dan
        M-nya. Dua dari keempat cahayanya yang mana yang menjadi sulit
        dibedakan baginya? Berikan alasannya dengan angkanya.
  \item Orang yang kekurangan kerucut M: pasangan mana yang
        dikacaukannya?
  \item Jelaskan mengapa tidak seorang pun dari keduanya ``buta
        terhadap merah'' tetapi keduanya buruk dalam membedakan merah
        dari hijau.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Sebuah keluarga.}
Nora melihat warna secara normal; ayahnya buta warna merah--hijau. Ia
menikahi Sam, yang penglihatannya normal.
\begin{enumerate}[resume]
  \item Berikan \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotipe} Nora bagi \omterm{def:g10:universal-dna:gene}{gen} \omterm{def:g11:the-eye:photoreceptors}{opsin} terpaut X-nya, dan berikan
        alasannya.
  \item Berikan \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotipe} dan penglihatan warna anak lelaki dan anak
        perempuan yang mungkin dimilikinya, beserta peluangnya.
  \item Anak perempuannya, Ines, yang penglihatan warnanya normal,
        menikahi lelaki yang buta warna. Hitung peluang Ines menjadi
        \omterm{def:g11:genes-and-disease:genetic}{pembawa}, lalu peluang anak perempuan mereka buta warna.
  \item Jelaskan mengapa anak perempuan yang buta warna selalu punya
        ayah yang buta warna.
  \item Ibu dari ayah Nora melihat warna secara normal. Apakah ia pasti
        seorang \omterm{def:g11:genes-and-disease:genetic}{pembawa}? Jelaskan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- \omterm{def:g10:universal-dna:gene}{Gennya}.}
Keserupaan antara urutan \omterm{def:g11:gene-expression:protein}{protein} \omterm{def:g11:the-eye:photoreceptors}{opsinnya}: L--M 96\%, L--S 43\%, M--S
43\%, S--rodopsin 41\%.
\begin{enumerate}[resume]
  \item Gambarkan, atau lukiskan, pohon keempat \omterm{def:g10:universal-dna:gene}{gennya} yang tersirat
        dari angka ini.
  \item Jika bedanya menumpuk dengan laju yang mantap dan penggandaan
        L/M-nya berumur 35 juta tahun, perkirakan umur perpisahan S
        terhadap L/M-nya.
  \item \omterm{def:g10:universal-dna:gene}{Gen} L dan M terletak berdampingan pada \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X. Jelaskan
        bagaimana pertukaran yang tak seimbang di antara keduanya
        selama meiosis dapat menghasilkan sebuah X yang hanya membawa
        satu dari keduanya, dan penglihatan warna apa yang dihasilkannya.
  \item Pertukaran semacam itu adalah sebab buta warna merah--hijau
        yang terlazim. Mengapa dua \omterm{def:g10:universal-dna:gene}{gen} yang berdampingan dan nyaris
        serupa sangat mudah mengalaminya?
  \item Penggandaan L/M-nya ditemukan pada seluruh primata Dunia Lama
        dan tidak pada mamalia lain. Apa yang dinyatakannya tentang
        kapan dan pada siapa penggandaan itu terjadi?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Ketajaman foveanya.}
Kerucut foveanya berjarak \qty{2.5}{\micro m}; jarak fokus \omterm{def:g11:the-eye:parts}{matanya}
\qty{17}{mm}. Pakai sudut (radian) $=$ ukuran $/$ jarak.
\begin{enumerate}[resume]
  \item Hitung sudut yang dibentangkan dua kerucut yang bertetangga,
        dalam radian dan dalam menit busur ($1' = \num{2.9e-4}$ rad).
  \item Berapa rincian huruf terkecil yang dapat dipisahkan pada
        \qty{6}{m}? Bandingkan dengan goresan huruf pada kartu uji
        \omterm{def:g11:the-eye:parts}{mata}, yaitu sekitar \qty{1.7}{mm} pada jarak itu bagi
        penglihatan yang ``normal''.
  \item Menjauhi foveanya, 100 \omterm{def:g11:the-eye:photoreceptors}{sel batang} berbagi satu \omterm{def:g10:cells-common-unit:cell}{sel} ganglion.
        Perkirakan sudut yang dapat dipisahkan di sana, dan jelaskan
        apa yang diperoleh sebagai gantinya.
  \item Kerucut fovea seekor elang dua kali lebih rapat dan \omterm{def:g11:the-eye:parts}{matanya}
        sedikit lebih besar. Dengan faktor berapa \omterm{def:g10:cells-common-unit:magnification}{daya pisahnya} lebih halus?
  \item Nyatakan hasilnya: ketiga jenis kerucutnya dan cahaya yang
        tidak terlihat masing-masing, satu peristiwa genetik yang
        memberi primata kerucut ketiganya, dan sudut yang dapat
        dipisahkan foveanya.
\end{enumerate}
\end{problem}
