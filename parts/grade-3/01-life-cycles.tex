\chapter{Life Cycles}\label{ch:g3:life-cycles}

\omterm{def:g1:living-nonliving:living}{Living things} are born (or hatch, or sprout), \omterm{def:g2:what-alive:signs}{grow}, make young, and
eventually die. The pattern of stages from start to \omterm{def:g3:life-cycles:butterfly}{adult} --- and
often back to young again --- is a \emph{\omterm{def:g3:life-cycles:cycle}{life cycle}}.

\section{What is a life cycle?}

\begin{definition}[Life cycle]\label{def:g3:life-cycles:cycle}
A \emph{life cycle}\index{life cycle} is the series of stages an
\omterm{def:g2:what-alive:organism}{organism} passes through during its life, including how it produces
the next generation.
\end{definition}

\omimg{butterfly-lifecycle.jpg}{0.52\linewidth}{\omterm{def:g3:life-cycles:butterfly}{Egg}, caterpillar,
\omterm{def:g3:life-cycles:butterfly}{pupa} and \omterm{def:g3:life-cycles:butterfly}{adult} butterfly --- a classic four-stage \omterm{def:g3:life-cycles:cycle}{life cycle}.}

\begin{example}[People]\label{ex:g3:life-cycles:people}
Baby $\to$ child $\to$ \omterm{def:g3:life-cycles:butterfly}{adult}. \omterm{def:g3:life-cycles:butterfly}{Adults} can have babies, and the cycle
continues in the family line. People do not change into a completely
different shape the way some insects do.
\end{example}

\section{Butterfly life cycle}

\begin{definition}[Butterfly stages]\label{def:g3:life-cycles:butterfly}
Many butterflies have four stages:
\begin{enumerate}
  \item \emph{egg}\index{egg};
  \item \emph{larva}\index{larva} (caterpillar) --- eats and \omterm{def:g2:what-alive:signs}{grows};
  \item \emph{pupa}\index{pupa} (chrysalis) --- \omterm{def:g1:my-body:body}{body} rebuilds;
  \item \emph{adult}\index{adult} butterfly --- can lay eggs.
\end{enumerate}
\end{definition}

\section{Other cycles}

\begin{example}[Frog]\label{ex:g3:life-cycles:frog}
\omterm{def:g3:life-cycles:butterfly}{Egg} in \omterm{def:g1:caring:needs}{water} $\to$ tadpole $\to$ tadpole with \omterm{def:g1:my-body:parts}{legs} $\to$ \omterm{def:g3:life-cycles:butterfly}{adult} frog.
The young live in \omterm{def:g1:caring:needs}{water}; \omterm{def:g3:life-cycles:butterfly}{adults} can live on land and in \omterm{def:g1:caring:needs}{water}.
\end{example}

\begin{example}[Flowering plant]\label{ex:g3:life-cycles:plant}
\omterm{def:g2:plant-parts:main}{Seed} $\to$ seedling $\to$ \omterm{def:g3:life-cycles:butterfly}{adult} \omterm{def:g1:plants-animals:plant}{plant} $\to$ \omterm{def:g2:plant-parts:main}{flower} $\to$ \omterm{def:g2:plant-parts:main}{fruit} with
\omterm{def:g2:plant-parts:main}{seeds}. New \omterm{def:g2:plant-parts:main}{seeds} start the cycle again.
\end{example}

\begin{method}[Read a life-cycle diagram]\label{met:g3:life-cycles:read}
\begin{enumerate}
  \item Find the starting stage (often \omterm{def:g3:life-cycles:butterfly}{egg} or \omterm{def:g2:plant-parts:main}{seed}).
  \item Follow the arrows in order.
  \item Notice which stage can produce the next generation.
  \item Count how many main stages are shown.
\end{enumerate}
\end{method}
