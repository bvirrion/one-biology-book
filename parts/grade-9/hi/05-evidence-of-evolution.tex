\chapter{जैव विकास के सबूत}\label{ch:g9:evidence-of-evolution}

यह किताब दो बार एक बड़े दावे के किनारे पर आकर रुकी है:
\omterm{def:g6:classification-kinship:tree}{नातेदारी-वृक्ष} की सारी शाखाएँ आपस में मिलती हैं
(\cref{rem:g6:classification-kinship:implies}), और हर \omterm{def:g6:exploring-our-environment:components}{सजीव} एक ही कूट
पढ़ता है (\cref{rem:g9:genes-and-dna:universal})। जिस दावे की ओर वे
इशारा करती हैं उसका एक नाम है --- \emph{जैव विकास}: \omterm{def:g5:biodiversity:species}{प्रजातियाँ}
पीढ़ी-दर-पीढ़ी बदलती हैं, और सब साझे पूर्वजों से उतरी हैं। अगला अध्याय
\emph{कैसे} पूछे, उससे पहले यह अध्याय वही करता है जो
\cref{exo:g6:classification-kinship:12} ने माँगा था: वह सबूत बिछा देता
है।

\section{चट्टानों की गवाही}

\begin{definition}[जीवाश्म]\label{def:g9:evidence-of-evolution:fossils}
\emph{जीवाश्म}\index{जीवाश्म} बीते हुए जीवन का सँभला हुआ कोई निशान है
--- \omterm{def:g1:animals-around-us:covering}{खोल}, \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी}, \omterm{def:g1:plants-around-us:parts}{पत्ती} की छाप, पैरों के निशान --- जो अवसादी चट्टान में बंद
हो गया। चट्टान की परतें समय के क्रम में जमती हैं, सबसे पुरानी सबसे नीचे;
इसलिए ऊपर की ओर पढ़ना इतिहास पढ़ना है। जीवाश्म
\cref{def:g6:exploring-our-environment:components} वाली सूची का
अवशेषों वाला आधा हिस्सा हैं, जो लाखों साल तक सँभला रहा।
\end{definition}

\begin{proposition}[परतों का लेखा क्या दिखाता है]\label{prop:g9:evidence-of-evolution:record}
क्रम से पढ़ा जाए तो यह लेखा तीन बार गवाही देता है:
\begin{enumerate}
  \item \emph{जीव-मंडलियाँ बदलती हैं}: हर परत-युग का अपना कलाकार-दल
        होता है --- ट्राइलोबाइटों के बीच कोई ख़रगोश नहीं, और ख़रगोश
        वाली परतों में कोई ट्राइलोबाइट नहीं; और हर गहरी परत की
        ज़्यादातर \omterm{def:g5:biodiversity:species}{प्रजातियाँ} \emph{\omterm{prop:g5:biodiversity:loss}{विलुप्त}} हैं
        (\cref{prop:g5:biodiversity:loss} वाला शब्द, अब अपने पीछे गहरे
        समय के साथ);
  \item \emph{क्रम \omterm{def:g6:classification-kinship:tree}{वृक्ष} की तरह घोंसलों में बैठा है}: \omterm{prop:g3:sorting-living-things:groups}{मछलियाँ} गहरी
        परतों में आती हैं, उनके ऊपर चार \omterm{def:g1:my-body:parts}{टाँगों} वाले \omterm{prop:g3:sorting-living-things:groups}{उभयचरों} के नातेदार,
        उनके ऊपर \omterm{prop:g4:classifying-animals:classes}{सरीसृपों} के नातेदार, और उससे भी ऊपर \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} तथा \omterm{prop:g3:sorting-living-things:groups}{पक्षी}
        --- यानी \omterm{def:g4:classifying-animals:classification}{वर्गीकरण} के डिब्बे
        (\cref{pb:g6:classification-kinship:1}) समय में ठीक वैसे ही
        चढ़ते हैं जैसा \omterm{prop:g6:classification-kinship:kinship}{नातेदारी} कहती है;
  \item \emph{बीच की कड़ियाँ मौजूद हैं}: \emph{आर्कियोप्टेरिक्स} की वह
        मशहूर शिला --- दाँतों और लंबी हड्डीदार पूँछ वाले \omterm{prop:g4:classifying-animals:classes}{सरीसृप} ढाँचे पर
        \omterm{def:g1:animals-around-us:covering}{पंख} और डैने --- ठीक वहाँ खड़ी है जहाँ \omterm{prop:g3:sorting-living-things:groups}{पक्षी} और \omterm{prop:g4:classifying-animals:classes}{सरीसृप} की शाखाएँ
        अलग होती हैं; और काम करती पिछली \omterm{def:g1:my-body:parts}{टाँगों} वाली \omterm{def:g9:evidence-of-evolution:fossils}{जीवाश्म} ह्वेलें
        ठीक वहाँ, जहाँ \cref{ex:g6:classification-kinship:whale} वाली
        \omterm{prop:g3:sorting-living-things:groups}{थलचर-स्तनधारी} वंशावली ने उनकी माँग की थी।
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[ह्वेल की शृंखला]\label{ex:g9:evidence-of-evolution:whales}
ह्वेल की मिसल, चट्टानों की भरी हुई: एक के बाद एक \omterm{def:g9:evidence-of-evolution:fossils}{जीवाश्म}, समय के क्रम
में, भेड़िये जितने बड़े चार \omterm{def:g1:my-body:parts}{टाँगों} वाले किनारे के \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} से चलकर,
सिकुड़ती \omterm{def:g1:my-body:parts}{टाँगों} तथा मज़बूत होती पूँछ वाले रूपों से होते हुए, बिना \omterm{def:g1:my-body:parts}{टाँगों}
वाले पूरे तैराकों तक जाते हैं --- जबकि आज की ह्वेल भी कूल्हे तथा \omterm{def:g1:my-body:parts}{टाँग} की
नन्ही छिपी \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियाँ} बनाती है
(\cref{ex:g6:classification-kinship:placing} ने उन अवशेषी हिस्सों का
ज़िक्र किया था), और आज भी \omterm{def:g2:animals-grow-up:born}{दूध} पिलाती है, हवा में साँस लेती है तथा गर्भ
में बाल उगाती है। भविष्यवाणी, \omterm{def:g9:evidence-of-evolution:fossils}{जीवाश्म}, अवशेषी हिस्से: तीन चाबियाँ, और
ताला एक।
\end{example}

\section{जीते-जागते शरीरों की गवाही}

\begin{proposition}[शरीर-रचना को याद रहता है]\label{prop:g9:evidence-of-evolution:anatomy}
जीते-जागते शरीर अपना इतिहास साथ ढोते हैं:
\begin{itemize}
  \item \emph{साझा नक़्शा}: चमगादड़ का डैना, ह्वेल का चप्पू, घोड़े की
        \omterm{def:g1:my-body:parts}{टाँग} और तुम्हारी \omterm{def:g1:my-body:parts}{बाँह}, सब \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियों} की एक ही तरतीब हैं
        (\cref{pb:g6:classification-kinship:1} वाला चार उपांगों का
        मोड़), बस खींची, छोटी की और नई ड्यूटी पर लगाई हुई --- और जो कोई
        नई इंजीनियरी कभी दोहराती ही नहीं, उसे विरासत समझा देती है;
  \item \emph{अवशेषी हिस्से}: ह्वेल के कूल्हे, न उड़ पाने वाले भृंग के
        बंद डैने, और ख़ुद तुम्हारी पूँछ की \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} तथा सिकुड़ी हुई हालत
        में अपेंडिक्स --- यानी वे हिस्से, जो इसलिए रह जाते हैं कि
        वंशावली आगे जोड़ती है, शुरू से नहीं बनाती;
  \item \emph{\omterm{def:g5:human-reproduction:pregnancy}{भ्रूण} आपस में मिलते हैं}: मछली, चूज़े और इंसान के \omterm{def:g4:classifying-animals:vertebrate}{कशेरुकी}
        \omterm{def:g5:human-reproduction:pregnancy}{भ्रूण} अपने शुरुआती हफ़्तों में हैरतअंगेज़ हद तक एक जैसे चलते हैं
        (\cref{prop:g8:from-egg-to-birth:timetable} वाला नक़्शा
        बिछाना) --- यानी साझे बनावट-निर्देश, जो देर में अलग होते हैं।
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[अणु वोट देते हैं]\label{prop:g9:evidence-of-evolution:molecules}
सबसे नई गवाही \cref{rem:g9:genes-and-dna:universal} वाला कूट है।
अलग-अलग \omterm{def:g5:biodiversity:species}{प्रजातियों} में किसी एक अच्छी तरह सँभले \omterm{def:g9:genes-and-dna:gene}{जीन} के हिज्जे मिलाओ: इंसान
और चिंपैंज़ी कुछ ही अक्षरों में अलग हैं, इंसान और चूहा उससे ज़्यादा में,
इंसान और मछली और भी ज़्यादा में, और इंसान तथा ख़मीर सबसे ज़्यादा में ---
और ये दूरियाँ, जीन-दर-जीन जोड़ी जाएँ, तो \emph{वही \omterm{def:g6:classification-kinship:tree}{वृक्ष}} दोबारा खींच
देती हैं जो शरीर-रचना वालों ने \omterm{def:g3:bones-and-muscles:skeleton}{हड्डियों} से और \omterm{def:g9:evidence-of-evolution:fossils}{जीवाश्म} वालों ने परतों से
खींचा था। तीन आज़ाद गवाह, और वंश-वृक्ष एक: इसी मेल की वजह से जीव
वैज्ञानिकों ने \emph{क्या} पर बहस छोड़ दी और \emph{कैसे} की ओर मुड़ गए।
\end{proposition}
\admitted

\begin{example}[उसे चलते हुए देखना]\label{ex:g9:evidence-of-evolution:live}
जहाँ पीढ़ियाँ तेज़ी से बदलती हैं वहाँ यह बदलाव देखा भी जा सकता है।
चितकबरे पतंगे का काला रूप कालिख से सने औद्योगिक जंगलों में फैल गया और
हवा साफ़ होने पर घट गया; कीटनाशकों से बचे \omterm{prop:g3:sorting-living-things:groups}{कीट} और प्रतिजैविकों को हराते
सूक्ष्मजीव (\cref{ch:g9:microbes-and-infection} उस चिकित्सा वाले मामले
से मिलवाएगा) सालों के भीतर पैदा हो जाते हैं; और द्वीपों की फ़िंच
चिड़ियों की चोंच के आकार का बँटवारा सूखे तथा बहुतायत के साथ खिसकता है,
जिसे छल्ला-और-नापने वाले अध्ययनों ने जीते-जी नापा है। छोटे क़दम, छोटे
समय --- यानी वह कार्य-प्रणाली, चलते हुए पकड़ी गई, उसकी लंबी दौड़ वाली
पहुँच चाहे जो हो।
\end{example}

\begin{method}[ऐतिहासिक सबूत तौलना]\label{met:g9:evidence-of-evolution:weigh}
जैव विकास इतिहास है, और ऐतिहासिक दावे वैसे ही परखे जाते हैं जैसे
\cref{pb:g9:unique-individuals:1} वाला पुराना मुक़दमा --- यानी आज़ाद
गवाहों के आपस में मिल जाने से:
\begin{enumerate}
  \item पूछो कि वह दावा क्या \emph{भविष्यवाणी} करता है: कौन-सी बीच की
        कड़ियाँ, कौन-से अवशेषी हिस्से, कौन-सी आणविक दूरियाँ;
  \item हर गवाह को अलग-अलग जाँचो: चट्टानें, शरीर, अणु, और चलती हुई
        निगरानियाँ;
  \item उनकी आज़ादी तौलो: जो गवाह आपस में साज़िश कर ही नहीं सकते थे,
        उनका मिल जाना भरोसा कई गुना कर देता है;
  \item और फ़ैसला अगली खोज के लिए खुला रखो --- और यह भी ध्यान रखो कि
        खोजें बार-बार वहीं निकलती रही हैं जहाँ दावे ने खोदने को कहा था
        (ह्वेल की \omterm{def:g1:my-body:parts}{टाँगें} ढूँढ़ी गई थीं, और मिलीं भी)।
\end{enumerate}
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:1}
\omterm{def:g9:evidence-of-evolution:fossils}{जीवाश्म} क्या है, और चट्टान की परतें समय-रेखा की तरह क्यों पढ़ी जाती हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:2}
परतों के लेखे की तीनों गवाहियाँ बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:3}
\omterm{def:g9:heredity-and-traits:traits}{लक्षणों} का कौन-सा मेल \emph{आर्कियोप्टेरिक्स} को बीच की कड़ी बनाता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:4}
तीन अवशेषी हिस्से बताओ, और हर एक किस बात की गवाही देता है यह भी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:5}
आणविक तुलनाएँ उस \omterm{def:g6:classification-kinship:tree}{वृक्ष} को दोबारा कैसे खींच देती हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:evidence-of-evolution:6}
\cref{ex:g9:evidence-of-evolution:live} में से बदलाव की दो चलती हुई
निगरानियों के नाम बताओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:7}
उपांगों का साझा नक़्शा दोबारा इस्तेमाल किए गए अच्छे नक़्शे के बजाय विरासत
की दलील क्यों देता है? (कोई नई इंजीनियरी क्या-क्या करने को आज़ाद होती?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:8}
ह्वेल की पूरी मिसल जोड़ो: वर्ग वाले \omterm{def:g9:heredity-and-traits:traits}{लक्षण}, अवशेषी हिस्से, \omterm{def:g9:evidence-of-evolution:fossils}{जीवाश्मों} की
शृंखला, और आणविक नातेदार --- हर एक पर एक पंक्ति।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:9}
चट्टानों, शरीरों और अणुओं का \emph{मिल जाना} किसी भी अकेले गवाह से
ज़्यादा क्यों गिना जाता है?
\cref{met:g9:evidence-of-evolution:weigh} के आज़ादी वाले क़दम का
इस्तेमाल करो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:10}
``ट्राइलोबाइटों के बीच कोई ख़रगोश नहीं।'' अब तक हुई हर खुदाई में बनी रही
यह ग़ैर-हाज़िरी ख़ुद एक मज़बूत सबूत क्यों है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:evidence-of-evolution:11}
\omterm{def:g5:human-reproduction:pregnancy}{भ्रूण} वाला गवाह: क्या मिलता है, वह अलग कब होता है, और किस अध्याय की
मशीनरी बताती है कि साझे शुरुआती निर्देश सबसे ज़्यादा क्यों मायने रखते
हैं?
\end{exercise}

\section{समस्या: संग्रहालय का नया खंड}

\begin{problem}[{सप्ताहांत समस्या --- सबूतों का कक्ष
सजाना}]\label{pb:g9:evidence-of-evolution:1}
प्राकृतिक इतिहास का संग्रहालय जैव विकास वाला एक खंड बनाता है, और कक्षा
उसके चार कमरे सजाती है: चट्टानें, शरीर, अणु, और चलती हुई निगरानियाँ।

\medskip\noindent\textbf{भाग I --- कमरा-दर-कमरा।}
\begin{enumerate}
  \item चट्टानें: दीवार को क्रम में लगाओ --- ट्राइलोबाइट की शिला, सबसे
        पहली \omterm{prop:g3:sorting-living-things:groups}{मछलियाँ}, \omterm{prop:g3:sorting-living-things:groups}{उभयचरों} के नातेदारों के पैरों के निशान, \omterm{prop:g4:classifying-animals:classes}{सरीसृपों}
        के \omterm{def:g3:bones-and-muscles:skeleton}{कंकाल}, \emph{आर्कियोप्टेरिक्स} का ढाँचा, और शुरुआती
        \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} का जबड़ा --- और इस कमरे का एक वाक्य वाला शीर्षक
        \cref{prop:g9:evidence-of-evolution:record} से लिखो।
  \item शरीर: बीच वाली चीज़ एक ही कंकाल-स्टैंड पर चमगादड़ का डैना,
        ह्वेल का चप्पू, घोड़े की \omterm{def:g1:my-body:parts}{टाँग} और इंसान की \omterm{def:g1:my-body:parts}{बाँह} फैलाए है। उस
        साझे नक़्शे का शीर्षक लिखो --- और अवशेषी हिस्सों वाली अलमारी
        की तीनों चीज़ें भी जोड़ो।
  \item अणु: दीवार का चार्ट किसी एक \omterm{def:g9:genes-and-dna:gene}{जीन} के अक्षरों के फ़र्क़ जोड़ता है:
        चिंपैंज़ी 1, चूहा 8, मछली 24, ख़मीर 41 (मिसाल भर की गिनतियाँ)।
        इस कमरे के काम करने के लिए उस चार्ट का \omterm{def:g6:classification-kinship:tree}{वृक्ष} किससे मिलना ही
        चाहिए?
  \item चलती हुई निगरानियाँ: \cref{ex:g9:evidence-of-evolution:live}
        में से दो चीज़ें चुनो और बताओ कि हर एक क्या चलते हुए दिखाती
        है।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दर्शकों के सवाल।}
\begin{enumerate}[resume]
  \item ``अगर हम उन पूर्वजों से उतरे हैं जो बंदरों के साथ साझे हैं, तो
        बंदर अब भी क्यों हैं?'' जवाब \omterm{def:g6:classification-kinship:tree}{वृक्ष} की शाखाओं से दो --- यानी
        चचेरे भाई-बहन, सीढ़ियाँ नहीं
        (\cref{def:g6:classification-kinship:tree})।
  \item ``वे ग़ायब कड़ियाँ कहाँ हैं?'' जो मिल चुकी हैं उन्हीं की ओर
        इशारा करो --- इस अध्याय से दो --- और बताओ कि हर एक किन दो के
        बीच खड़ी थी।
  \item ``जैव विकास तो बस एक सिद्धांत ही है न?'' जवाब ऐतिहासिक सबूत
        वाले तरीक़े से दो: यहाँ ``सिद्धांत'' का मतलब क्या है, और उन
        चारों कमरों का मिल जाना क्या साबित करता है।
  \item ``क्या कल की कोई खुदाई इसे उलट सकती है?'' बताओ कि कैसी खोज
        उलट देती --- ट्राइलोबाइटों के बीच ख़रगोश --- और खुदाई-दर-खुदाई
        उसकी ग़ैर-हाज़िरी का मतलब क्या रहा है।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- खंड का समापन-पट्ट।}
\begin{enumerate}[resume]
  \item वह पट्ट लिखो जो इस खंड को बग़ल वाले \omterm{def:g4:classifying-animals:classification}{वर्गीकरण} कक्ष से जोड़े:
        वे पुराने घोंसलेदार डिब्बे असल में शुरू से ही वह \omterm{def:g6:classification-kinship:tree}{वृक्ष} क्यों
        थे।
  \item अणु वाला अनुच्छेद जोड़ो: वह साझा कूट तने के बारे में क्या कहता
        है।
  \item ईमानदार अनुच्छेद भी जोड़ो: यह खंड क्या दिखाता है (बदलाव,
        \omterm{prop:g6:classification-kinship:kinship}{नातेदारी}, बीच की कड़ियाँ) और क्या अभी नहीं समझाता --- और उस
        कार्य-प्रणाली वाले सवाल का नाम भी लो जिसे अगला कक्ष उठाएगा।
  \item समापन एक ऐसे वाक्य से करो जो कोई बारह साल का बच्चा याद रखे, और
        जिसमें \emph{परिवार} तथा \emph{समय} शब्द हों।
\end{enumerate}
\end{problem}
