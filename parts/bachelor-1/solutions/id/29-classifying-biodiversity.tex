\section*{Bab \ref{ch:b1:classifying-biodiversity} --- Menggolongkan Dunia Hidup}

\begin{solution}{exo:b1:classifying-biodiversity:1}
\emph{Felis catus}: \omterm{def:b1:classifying-biodiversity:nomenclature}{genus} \emph{Felis}, julukan \emph{catus}; famili
Felidae, ordo Carnivora, kelas Mammalia, filum Chordata.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:2}
\omterm{def:b1:classifying-biodiversity:characters}{Homologi}: kemiripan lewat keturunan dari seorang moyang bersama
(tulang tungkai depan seekor kelelawar dan seorang manusia).
\omterm{def:b1:classifying-biodiversity:characters}{Homoplasi}: kemiripan lewat pemusatan atau pembalikan (sayap seekor
kelelawar dan seekor burung). Hanya \omterm{def:b1:classifying-biodiversity:characters}{homologi} turunan yang dikongsi
yang menjadi bukti bagi sebuah \omterm{def:b1:classifying-biodiversity:systematics}{klade}.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:3}
``Ikan'': parafiletik (mengecualikan tetrapoda yang menurun dari
ikannya). Burung: monofiletik (seluruh keturunan seorang moyang
berbulu). ``Hewan berdarah panas'': polifiletik (burung dan mamalianya
memperoleh endoterminya secara terpisah).
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:4}
Bacteria, Archaea, Eukarya. Archaea memiliki \omterm{def:b1:lipids:fattyacid}{lipid} membran berpaut
eter dan tanpa \omterm{def:b1:carbohydrates:polysaccharide}{peptidoglikan}; \omterm{def:b1:gene-expression:transcription}{RNA polimerase} dan \omterm{def:b1:proteins:peptide}{protein} \omterm{def:b1:gene-expression:ribosome}{ribosomnya}
menyerupai milik eukariotnya.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:5}
Bakteri tidak berbiak silang --- keduanya membelah, lalu menukar \omterm{def:b1:genomes:gene}{gen}
melintasi apa yang akan menjadi batas \omterm{def:b1:classifying-biodiversity:species}{spesies} --- dan fosil tak dapat
diuji perbiakan silangnya. Bakteri dibatasi lewat kemiripan urutannya
(sebuah ambang pada \omterm{def:b1:nucleic-acids:rnas}{RNA ribosom} atau \omterm{def:b1:genomes:genome}{genomnya}); fosil lewat
morfologinya.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:6}
Pohon (X,Y),Z: ciri 1, 3, 4 satu perubahan masing-masing, ciri 2 dua
perubahan: panjangnya 5. Pohon (Y,Z),X: ciri 2 satu perubahan, ciri
1, 3, 4 dua masing-masing: panjangnya 7. (X,Y) lebih disukai; ciri 2
adalah \omterm{def:b1:classifying-biodiversity:characters}{homoplasinya}.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:7}
\omterm{def:b1:classifying-biodiversity:characters}{Kelompok luarnya} menetapkan polaritas tiap cirinya: keadaannya diambil
sebagai keadaan moyang, sehingga hanya keadaan turunannya yang
terhitung sebagai bukti. Sebuah \omterm{def:b1:classifying-biodiversity:characters}{kelompok luar} yang sebenarnya berada
di dalam kelompoknya membuat sebagian keadaan turunannya tampak
sebagai keadaan moyang lalu mengakarkan pohonnya di tempat yang salah,
sehingga mengubah \omterm{def:b1:classifying-biodiversity:systematics}{klade} menjadi kelompok parafiletik.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:8}
Urutan \omterm{def:b1:gene-expression:ribosome}{ribosom} dan \omterm{def:b1:proteins:peptide}{proteinnya} menempatkan hewan dan jamur sebagai
garis keturunan saudara, keduanya di dalam opistokonta, yang \omterm{def:b1:cell-unit-of-life:cell}{sel}
berenangnya (sperma, zoospora khitrid) memiliki sebuah flagelum
belakang tunggal; tumbuhannya terletak dalam sebuah garis keturunan
terpisah yang ditetapkan kloroplas primernya.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:9}
Sebuah kerajaan di dalam bakterinya akan menyiratkan metanogennya lebih
dekat pada bakteri lain daripada pada eukariotnya; urutannya
memperlihatkan tiga garis keturunan yang sama jauhnya satu sama lain,
sehingga prokariotnya bukanlah satu kelompok dan pembagian kehidupan
yang terdalam bersifat rangkap tiga.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:10}
Kumbang khas inang tiap \omterm{def:b1:classifying-biodiversity:species}{spesies} pohonnya: $0.2\times 1200 = 240$;
serangga tajuk yang khas bagi pohonnya: $240/0.4 = 600$; seluruh
pohonnya: $600/(2/3) = 900$; dikali \num{50000} pohon: \num{45} juta.
Anggapannya: pohon yang diasapinya khas, kekhasannya nyata dan bukan
kekeliruan cuplikan, kumbangnya adalah \qty{40}{\%} serangga pada tiap
pohonnya, dan \omterm{def:b1:classifying-biodiversity:species}{spesies} yang khas bagi satu pohon tidak terhitung dua
kali --- masing-masing tak pasti sebesar faktor dua atau lebih, dan
taksiran kemudiannya telah jatuh menuju \num{5} juta.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:11}
\omterm{def:b1:classifying-biodiversity:nomenclature}{Spesimen tipenya} adalah benda yang diacu namanya: \omterm{def:b1:populations:population}{populasi} hidupnya
adalah \omterm{def:b1:classifying-biodiversity:species}{spesies} yang sama bila keduanya dinilai sekonspesifik dengan
tipenya, dan sebuah \omterm{def:b1:classifying-biodiversity:species}{spesies} baru bila tidak; pada hal yang kedua nama
lamanya tinggal bersama tipenya dan \omterm{def:b1:populations:population}{populasi} barunya memerlukan sebuah
nama baru. Keutamaannya menjamin bahwa bila dua nama ternyata mengacu
pada satu \omterm{def:b1:classifying-biodiversity:species}{spesies}, yang lebih tua dipegang, sehingga namanya tidak
berlipat ganda atau hanyut bersama tiap pendapat baru.
\end{solution}

\begin{solution}{exo:b1:classifying-biodiversity:12}
\omterm{def:b1:classifying-biodiversity:parsimony}{Parsimoni} hanya menganggap bahwa pemusatan dan pembalikan lebih jarang
daripada keturunan yang dikongsi, sehingga sebuah pohon tidak boleh
dibuat menuntut lebih banyak daripada yang dipaksakan datanya. Ia
gagal ketika pemusatannya lazim (\omterm{def:b1:ecosystem-organization:niche}{habitat} yang mirip), ketika cirinya
berkorelasi (satu penyesuaian terhitung sebagai banyak), dan ketika
garis keturunan yang berevolusi cepat menimbun kemiripan kebetulan
lalu tertarik bersama (``cabang panjang saling menarik''). Urutan
molekulnya memasok banyak ciri bebas dan model penggantian untuk
menguji pohon yang sama; fosilnya memasok urutan kemunculan keadaan
turunannya dan tarikh bagi belahannya; kesesuaian di antara data yang
bebas itulah ujinya.
\end{solution}

\begin{solution}{pb:b1:classifying-biodiversity:1}
\textbf{1.} Ciri 1, rahang: dikongsi semuanya kecuali lampreynya,
sehingga ia menetapkan kelompok dalamnya lalu mengesahkan lampreynya
sebagai \omterm{def:b1:classifying-biodiversity:characters}{kelompok luar}.
\textbf{2.} 1 (7 \omterm{def:b1:classifying-biodiversity:nomenclature}{takson}: vertebrata berahang), 2 (6: vertebrata
bertulang), 3 (5: ikan paru dan tetrapodanya), 4 (4: tetrapoda), 5 (3:
amniota), 8, 9 dan 10 (2 masing-masing: kadal+merpati, tikus+merpati,
kadal+merpati), 6 dan 7 (1 masing-masing).
\textbf{3.} Rambut (tikus) dan bulu (merpati): keadaan yang unik,
autapomorfi, yang mendiagnosis \omterm{def:b1:classifying-biodiversity:nomenclature}{taksonnya} tetapi tak mengatakan apa pun
tentang kerabatnya.
\textbf{4.} Ciri 9 (endotermi: tikus+merpati) berselisih dengan 8 dan
10 (kadal+merpati); perselisihannya berada di antara ketiga amniotanya.
\textbf{5.} Ketiadaan \omterm{def:b1:gas-exchange:lung}{paru-paru} adalah keadaan moyangnya (lamprey, hiu
dan troutnya tak memilikinya), kehadirannya keadaan turunannya;
keadaan \omterm{def:b1:classifying-biodiversity:characters}{kelompok luarnya} diambil sebagai keadaan moyangnya.
\textbf{6.} Tiga: salah satu dari ketiganya dapat menjadi yang
terpisah --- (kadal,merpati) dengan tikusnya di luar,
(tikus,merpati) dengan kadalnya di luar, (kadal,tikus) dengan
merpatinya di luar. Hanya ciri 8, 9 dan 10 yang berbeda di antara
ketiganya, sehingga hanya ketiganya yang berperan dalam pilihannya.
\textbf{7.} (lamprey,(hiu,(trout,(ikan paru,(katak,(kadal, tikus,
merpati)))))), tiap simpulnya ditandai satu ciri.
\textbf{8.} Pohon A: (tikus,(kadal,merpati)); pohon B: (kadal,(tikus,
merpati)).
\textbf{9.} Pohon A: ciri 1--8 dan 10 satu perubahan masing-masing (9),
ciri 9 dua perubahan: panjangnya \num{11}.
\textbf{10.} Pohon B: ciri 1--7 dan 9 satu masing-masing (8), ciri 8 dan
10 dua masing-masing: panjangnya \num{12}.
\textbf{11.} Pohon A, sebesar satu langkah.
\textbf{12.} Endotermi: ia muncul dua kali, sekali pada garis
mamalianya dan sekali pada garis burungnya --- sebuah pemusatan, yang
sesuai dengan mekanismenya yang berbeda (bulu tubuh dan keringat lawan
bulu dan kantung udara).
\textbf{13.} Tanpa 8 dan 10: pohon A panjangnya 9 (endotermi dua kali),
pohon B panjangnya 8 (semuanya sesuai): B akan lebih disukai.
Simpulannya bergantung pada ciri yang dipilihnya; dengan sedikit ciri
satu \omterm{def:b1:classifying-biodiversity:characters}{homoplasi} dapat membalikkan hasilnya, yang menjadi sebab banyak
ciri, dan ciri molekul, dibutuhkan.
\textbf{14.} Tidak: moyang bersamanya juga moyang tetrapodanya,
sehingga keempat tetrapodanya harus ditambahkan.
\textbf{15.} Ya: telur amniotiknya.
\textbf{16.} Tidak: merpatinya menurun dari dalam reptilianya (ia
saudara kadalnya di sini, dan saudara buayanya dalam pohon yang lebih
penuh), sehingga reptilia tanpa burung bersifat parafiletik.
\textbf{17.} Tidak: moyang bersama terdekatnya adalah moyang seluruh
amniotanya, yang tidak endotermik; polifiletik, yang dikelompokkan
lewat sebuah ciri yang memusat.
\textbf{18.} Lebih dekat dengan kataknya: ia mengongsi \omterm{def:b1:gas-exchange:lung}{paru-paru} (ciri
3) dengan tetrapodanya, yang tak dimiliki troutnya; ``ikan'' tanpa
tetrapoda bersifat parafiletik.
\textbf{19.} Kelas Reptilia, kelas Mammalia, kelas Aves: tiga kelas
berperingkat sama, walaupun belahan kadal--merpatinya jauh lebih baru
daripada belahan masing-masing dari mamalianya; peringkatnya adalah
kelaziman, dan peringkat yang sama tidak berarti umur atau kekhasan
yang sama.
\textbf{20.} Mamalia: \num{20} baru setahun pada \num{6500}
(\qty{0.3}{\%}); ikan: \num{300} pada \num{35000} (\qty{0.9}{\%}) ---
inventaris mamalianya lebih dekat pada selesai, inventaris ikannya
masih tumbuh cepat.
\textbf{21.} Sembilan persepuluh urutannya milik \omterm{def:b1:organism-environment:organism}{organisme} yang tak
pernah dibiakkan atau dinamai: prokariot yang diperikan adalah sebuah
fraksi kecil dari yang ada bahkan dalam satu liter air, dan
inventarisnya nyaris belum dimulai.
\textbf{22.} Latarnya: \num{9} setahun. Seribu kali: \num{9000}
setahun --- sebagian besarnya \omterm{def:b1:classifying-biodiversity:species}{spesies} yang tak pernah diperikan.
\textbf{23.} Cirinya, urutannya dan tempatnya pada pohonnya hilang
bersamanya; sebuah cabang pohon kehidupan terhapus sebelum ia
digambar, dan tiap \omterm{def:b1:classifying-biodiversity:systematics}{klade} yang hanya ia sendiri akan menetapkannya
pergi bersamanya.
\textbf{24.} Famili itu besar, mencolok dan nyaris seluruhnya telah
diperikan (famili baru kini jarang), sehingga cacahnya dekat pada
lengkap; nisbah teratur \omterm{def:b1:classifying-biodiversity:species}{spesies} terhadap \omterm{def:b1:classifying-biodiversity:nomenclature}{genus} terhadap famili pada
kelompok yang dikenal baik lalu membiarkan cacah familinya
diekstrapolasikan ke \omterm{def:b1:classifying-biodiversity:species}{spesiesnya} tempat cacah \omterm{def:b1:classifying-biodiversity:species}{spesiesnya} tidak lengkap.
\textbf{25.} (lamprey,(hiu,(trout,(ikan paru,(katak,(tikus,(kadal,
merpati))))))), panjangnya \num{11}.
\end{solution}
