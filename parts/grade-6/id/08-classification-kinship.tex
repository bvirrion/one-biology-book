\chapter{Klasifikasi dan Kekerabatan}\label{ch:g6:classification-kinship}

Mengapa \omterm{def:g4:classifying-animals:classification}{klasifikasi} pada \cref{ch:g4:classifying-animals} berhasil
dengan begitu baik? Kelompokkan \omterm{def:g1:animals-around-us:animal}{hewan} menurut \omterm{def:g1:animals-around-us:covering}{bulu burung} dan \omterm{def:g1:animals-around-us:covering}{bulu},
dan seribu ciri lain berbaris dengan patuh --- pemberi \omterm{def:g2:animals-grow-up:born}{susu} bertulang
belakang, pemakai \omterm{def:g1:animals-around-us:covering}{bulu burung} bertelur bercangkang, dan tidak seorang
pun perlu memeriksanya. \Cref{rem:g4:classifying-animals:tree}
membiarkan pertanyaannya menggantung. Bab ini menjawabnya dengan satu
kata --- \omterm{prop:g6:classification-kinship:kinship}{kekerabatan} --- lalu membangun ulang kotaknya sebagai sebuah
\omterm{def:g6:classification-kinship:tree}{pohon} keluarga.

\section{Atribut, yang dibagi bersama dan bersarang}

\begin{definition}[Atribut]\label{def:g6:classification-kinship:attribute}
Suatu \emph{atribut}\index{atribut} adalah ciri yang \emph{dimiliki}
sebuah organisme, yang dinyatakan sedemikian rupa sehingga dapat
diperiksa: punya \omterm{def:g3:bones-and-muscles:skeleton}{rangka} dengan \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}; punya empat \omterm{def:g1:my-body:parts}{anggota
gerak}; punya \omterm{def:g1:animals-around-us:covering}{bulu} dan memberi \omterm{def:g2:animals-grow-up:born}{susu}; punya \omterm{def:g1:animals-around-us:covering}{bulu burung}. \omterm{def:g4:classifying-animals:classification}{Klasifikasi}
dibangun di atas atribut --- \omterm{def:g3:sorting-living-things:criterion}{kriteria} yang baik pada
\cref{def:g3:sorting-living-things:criterion}, yang dibakukan.
\end{definition}

\begin{proposition}[Atribut bersarang]\label{prop:g6:classification-kinship:nest}
Kalau diperiksa pada banyak \omterm{def:g5:biodiversity:species}{spesies}, \omterm{def:g6:classification-kinship:attribute}{atribut} jatuh ke dalam himpunan
yang \emph{bersarang} --- setiap \omterm{def:g1:animals-around-us:animal}{hewan} berbulu juga punya empat
\omterm{def:g1:my-body:parts}{anggota gerak}; setiap \omterm{def:g1:animals-around-us:animal}{hewan} berkaki empat juga punya \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang};
tidak pernah pola yang sebaliknya (berbulu tanpa \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}) dan
tidak pernah himpunan yang bersilangan (berbulu dan berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}
sekaligus). Persarangan itulah sebabnya kelompok-di-dalam-kelompok
(\cref{def:g4:classifying-animals:classification}) sama sekali cocok
dengan dunia yang hidup.
\end{proposition}
\admitted

\begin{example}[Menempatkan tiga hewan]\label{ex:g6:classification-kinship:placing}
Periksalah kotaknya bagi paus, \omterm{prop:g3:sorting-living-things:groups}{ikan} trout dan \omterm{prop:g3:sorting-living-things:groups}{burung} hitam. Paus:
\omterm{def:g4:classifying-animals:vertebrate}{tulang belakang} ya, empat \omterm{def:g1:my-body:parts}{anggota gerak} ya (sirip dayung depan, dan
sisa rudimenter yang tersembunyi di belakang), bulu-dan-susu ya ---
kotak yang terdalam, bersama kucingnya. \omterm{prop:g3:sorting-living-things:groups}{Ikan} trout: \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}
ya, berhenti --- hanya kotak yang terluar. \omterm{prop:g3:sorting-living-things:groups}{Burung} hitam: \omterm{def:g4:classifying-animals:vertebrate}{tulang
belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak} (dua kaki, dua sayap), \omterm{def:g1:animals-around-us:covering}{bulu burung} ---
kotak berkaki empatnya, di dalam bilik \omterm{def:g1:animals-around-us:covering}{bulu burungnya} sendiri. Tidak
ada \omterm{def:g1:animals-around-us:animal}{hewan} yang pernah menuntut dua kotak yang bersilangan sekaligus.
\end{example}

\section{Kekerabatan: alasan di balik persarangannya}

\begin{proposition}[Atribut bersama berarti nenek moyang bersama]\label{prop:g6:classification-kinship:kinship}
Penjelasan persarangannya adalah pewarisan
(\cref{rem:g4:animal-reproduction:mix}: anak menerima dari induknya).
\omterm{def:g5:biodiversity:species}{Spesies} yang berbagi sebuah \omterm{def:g6:classification-kinship:attribute}{atribut} membaginya karena mereka
\emph{mewarisinya dari seorang nenek moyang bersama} --- sebuah
\omterm{def:g5:biodiversity:species}{spesies} kakek buyut yang jauh. Makin banyak \omterm{def:g6:classification-kinship:attribute}{atribut} yang dibagi dua
\omterm{def:g5:biodiversity:species}{spesies}, makin dekat \emph{kekerabatan}\index{kekerabatan} mereka:
kucing dan paus, yang sama-sama berbulu dan memberi \omterm{def:g2:animals-grow-up:born}{susu}, lebih dekat
sepupunya satu sama lain daripada masing-masingnya dengan \omterm{prop:g3:sorting-living-things:groups}{ikan} trout.
Sebuah kelompok \omterm{def:g4:classifying-animals:classification}{klasifikasi}, pada dasarnya, adalah sebuah
\emph{keluarga}: seorang nenek moyang beserta semua keturunannya.
\end{proposition}
\admitted

\begin{example}[Akhirnya membaca pausnya dengan benar]\label{ex:g6:classification-kinship:whale}
Inilah sebabnya teka-teki paus
(\cref{exo:g4:classifying-animals:9}) harus terpecahkan sebagaimana ia
terpecahkan: bulu-dan-susu menandai garis keluarga pausnya --- nenek
moyangnya adalah \omterm{prop:g3:sorting-living-things:groups}{mamalia} darat yang memberi \omterm{def:g2:animals-grow-up:born}{susu} --- sedangkan
bentuknya yang mirip \omterm{prop:g3:sorting-living-things:groups}{ikan} adalah pekerjaan airnya
(\cref{exo:g4:adapted-to-their-world:11}), yang dicapkan pada sebuah
keluarga yang tidak berkerabat oleh cara hidup yang sama. \omterm{def:g6:classification-kinship:attribute}{Atribut}
keturunan mengelompokkan keluarga; kemiripan \omterm{def:g2:where-animals-live:habitat}{habitat} dapat dipinjam
oleh orang asing. \omterm{def:g4:classifying-animals:classification}{Klasifikasi} memercayai yang pertama dan mencurigai
yang kedua.
\end{example}

\begin{definition}[Pohon kekerabatan]\label{def:g6:classification-kinship:tree}
\omterm{prop:g6:classification-kinship:kinship}{Kekerabatan} digambarkan sebagai sebuah \emph{pohon}\index{pohon
kekerabatan}: titik percabangannya melambangkan nenek moyang bersama,
dan tiap \omterm{def:g6:classification-kinship:attribute}{atribut} muncul sekali, pada percabangan tempat ia timbul ---
setiap \omterm{def:g5:biodiversity:species}{spesies} di seberang percabangan itu mewarisinya. Kotak yang
bersarang dan pohonnya membawa keterangan yang sama: sebuah kotak
adalah sebuah cabang beserta semua \omterm{def:g1:plants-around-us:tree}{rantingnya}.
\end{definition}

\begin{method}[Membangun sebuah pohon kekerabatan kecil]\label{met:g6:classification-kinship:build}
Diberikan segenggam \omterm{def:g5:biodiversity:species}{spesies}:
\begin{enumerate}
  \item tabelkan \omterm{def:g6:classification-kinship:attribute}{atributnya} --- \omterm{def:g5:biodiversity:species}{spesies} pada baris, \omterm{def:g6:classification-kinship:attribute}{atribut} yang dapat
        diperiksa pada lajur, centang di tempat ia dimiliki;
  \item kelompokkan menurut centang yang dibagi bersama, \omterm{def:g6:classification-kinship:attribute}{atribut} yang
        terluas lebih dahulu: persarangannya muncul di dalam tabelnya;
  \item gambarlah \omterm{def:g6:classification-kinship:tree}{pohonnya}: satu percabangan per \omterm{def:g6:classification-kinship:attribute}{atribut}, yang terluas
        paling dekat dengan \omterm{def:g1:plants-around-us:parts}{batangnya}; tempatkan tiap \omterm{def:g5:biodiversity:species}{spesies} melewati
        persis percabangan yang \omterm{def:g6:classification-kinship:attribute}{atributnya} dimilikinya;
  \item bacalah \omterm{prop:g6:classification-kinship:kinship}{kekerabatannya}: kedekatan dua \omterm{def:g5:biodiversity:species}{spesies} adalah
        percabangan terakhir yang dibaginya bersama.
\end{enumerate}
\end{method}

\begin{remark}[Apa yang diam-diam disiratkan pohonnya]\label{rem:g6:classification-kinship:implies}
Anggaplah \omterm{def:g6:classification-kinship:tree}{pohonnya} sungguh-sungguh dan ia mengatakan sesuatu yang amat
besar: ikutilah dua cabang mana pun ke belakang dan keduanya
\emph{bertemu}. Kucing dan \omterm{prop:g3:sorting-living-things:groups}{ikan} trout berbagi sebuah \omterm{def:g5:biodiversity:species}{spesies} kakek
buyut; jadi, lebih jauh ke belakang, demikian pula kucing dan siput,
\omterm{def:g3:flowers-fruits-seeds:parts}{bunga} aster dan anak manusia. Semua \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} kalau begitu
merupakan satu keluarga, dan kesatuannya
(\cref{prop:g5:first-look-at-cells:principle}) merupakan sebuah
kemiripan keluarga --- dan \omterm{def:g5:biodiversity:species}{spesies} pastilah dapat \emph{berubah}
lintas generasi, sebab kalau tidak, percabangannya tidak akan pernah
dapat terjadi. Apakah hal itu benar, dan bagaimana ia dapat bekerja,
adalah pertanyaan besar pada tahun
\cref{ch:g9:evidence-of-evolution} --- \omterm{def:g6:classification-kinship:tree}{pohon} inilah bagan yang
memaksakannya.
\end{remark}

\section{Soal: Laci Museum}

\begin{problem}[{Soal akhir pekan --- enam spesimen, satu tabel
atribut, satu pohon}]\label{pb:g6:classification-kinship:1}
Sebuah museum meminjamkan kepada kelasnya sebuah laci berisi enam
spesimen: seekor \omterm{prop:g3:sorting-living-things:groups}{ikan} trout, seekor katak, seekor kadal, seekor
merpati, seekor tikus, dan sebuah \omterm{def:g3:bones-and-muscles:skeleton}{tengkorak} lumba-lumba beserta cetakan
sirip dayungnya. Tugasnya: klasifikasikan mereka menurut \omterm{def:g6:classification-kinship:attribute}{atribut} lalu
gambarkan \omterm{def:g6:classification-kinship:tree}{pohon kekerabatannya}.

\medskip\noindent\textbf{Bagian I --- Tabelnya.}
\begin{enumerate}
  \item Bangunlah tabel bagi keenamnya, dengan \omterm{def:g6:classification-kinship:attribute}{atribut}: \omterm{def:g4:classifying-animals:vertebrate}{tulang
        belakang}; empat \omterm{def:g1:my-body:parts}{anggota gerak}; \omterm{def:g1:the-five-senses:sense}{kulit} bersisik yang kering;
        \omterm{def:g1:animals-around-us:covering}{bulu burung}; \omterm{def:g1:animals-around-us:covering}{bulu} dan \omterm{def:g2:animals-grow-up:born}{susu}. (Untuk lumba-lumbanya, yang hampir
        tak berambut, label museumnya menjamin: anaknya diberi \omterm{def:g2:animals-grow-up:born}{susu}
        --- \cref{ex:g3:sorting-living-things:surprises} menjelaskan
        mengapa kesaksian itu menyelesaikannya.)
  \item \omterm{def:g6:classification-kinship:attribute}{Atribut} mana yang dibagi bersama keenamnya? Apa arti perkongsian
        yang terluas itu, dalam istilah \omterm{prop:g6:classification-kinship:kinship}{kekerabatan}?
  \item Spesimen mana yang mencentang ``empat \omterm{def:g1:my-body:parts}{anggota gerak}''?
        Jelaskan centang merpatinya dan centang lumba-lumbanya ---
        sayap dan sirip dayung itu apa, menurut logika \omterm{def:g6:classification-kinship:tree}{pohonnya}?
  \item Mengapa ``hidup di air'' tidak muncul di mana pun dalam
        tabelnya, walaupun \omterm{prop:g3:sorting-living-things:groups}{ikan} trout dan lumba-lumba sama-sama
        melakukannya? Kutiplah aturan bab ini.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- \omterm{def:g6:classification-kinship:tree}{Pohonnya}.}
\begin{enumerate}[resume]
  \item Gambarlah \omterm{def:g6:classification-kinship:tree}{pohonnya}: urutkan percabangannya --- \omterm{def:g4:classifying-animals:vertebrate}{tulang
        belakang}, empat \omterm{def:g1:my-body:parts}{anggota gerak}, lalu penutupnya (\omterm{def:g1:animals-around-us:covering}{sisik}, \omterm{def:g1:animals-around-us:covering}{bulu
        burung}, \omterm{def:g1:animals-around-us:covering}{bulu}) pada cabangnya masing-masing --- lalu tempatkan
        keenamnya.
  \item Bacalah dari pohonmu: sepupu terdekat tikusnya di dalam
        lacinya, dan sepupu terdekat \omterm{prop:g3:sorting-living-things:groups}{ikan} troutnya.
  \item Kataknya berdiri melewati ``empat \omterm{def:g1:my-body:parts}{anggota gerak}'' tetapi
        sebelum setiap percabangan penutupnya. \omterm{def:g1:the-five-senses:sense}{Kulitnya} yang telanjang
        dan lembap (\cref{rem:g4:classifying-animals:confusion})
        menandainya sebagai apa, dan di mana cabangnya berpisah?
  \item Sebuah label terlepas dari spesimen ketujuh: \omterm{def:g1:the-five-senses:sense}{kulit} bersisik
        yang kering, empat \omterm{def:g1:my-body:parts}{anggota gerak}, \omterm{def:g4:classifying-animals:vertebrate}{tulang belakang}. Tempatkan ia
        pada \omterm{def:g6:classification-kinship:tree}{pohonnya} lalu sebutkan kelasnya --- tanpa melihat
        labelnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Maknanya.}
\begin{enumerate}[resume]
  \item Lumba-lumba dan \omterm{prop:g3:sorting-living-things:groups}{ikan} trout berbagi bentuk yang ramping;
        lumba-lumba dan tikus berbagi keturunan bulu-dan-susu. Pasangan
        mana yang duduk lebih dekat pada \omterm{def:g6:classification-kinship:tree}{pohonnya}, dan mengapa
        \omterm{def:g6:classification-kinship:tree}{pohonnya} harus mengabaikan kerampingannya?
  \item Nyatakan apa yang diakui percabangan ``empat \omterm{def:g1:my-body:parts}{anggota gerak}''
        secara kesejarahan: satu hal tunggal apa, sejak lama sekali,
        yang menjadi asal keturunan katak, kadal, merpati, tikus dan
        lumba-lumba?
  \item \omterm{def:g6:classification-kinship:tree}{Pohon} kelasnya membuat setiap cabang bertemu, pada akhirnya,
        di \omterm{def:g1:plants-around-us:parts}{batangnya}. Sebutkan dalam satu kalimat apa yang diam-diam
        diakui gambar itu tentang keenam spesimennya --- dan tentang
        anggota kelas yang menggambarnya.
  \item Seorang peragu berkata: ``pohonmu hanyalah sebuah sistem
        pengarsipan.'' Berikan jawaban dua bagiannya: persarangan
        \omterm{def:g6:classification-kinship:attribute}{atribut} itu sebuah \emph{kenyataan} tentang apa, dan \omterm{def:g6:classification-kinship:tree}{pohonnya}
        \emph{menjelaskan} apa yang akan dibiarkan tak terjelaskan
        oleh pengarsipan belaka.
\end{enumerate}
