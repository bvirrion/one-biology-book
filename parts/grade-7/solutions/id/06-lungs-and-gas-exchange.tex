\section*{Bab \ref{ch:g7:lungs-and-gas-exchange} --- Paru-Paru dan Pertukaran Gas}

\begin{solution}{exo:g7:lungs-and-gas-exchange:1}
\omterm{def:g4:breathing:movements}{Menarik napas}: \omterm{def:g3:bones-and-muscles:muscle}{otot} rusuknya mengangkat lalu melebarkan sangkarnya
sementara \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{diafragmanya} mendatar ke bawah; dada yang terentang menarik
\omterm{def:g7:how-animals-breathe:four}{paru-parunya} melebar dan udara mengalir masuk --- pekerjaan \omterm{def:g3:bones-and-muscles:muscle}{otot}.
\omterm{def:g4:breathing:movements}{Mengembuskan napas} ketika beristirahat: \omterm{def:g3:bones-and-muscles:muscle}{ototnya} mengendur dan dada
yang terentang melenting kembali, sehingga mendorong udaranya keluar
--- pelepasan, yang nyaris gratis.
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:2}
Lantai \omterm{def:g3:bones-and-muscles:muscle}{otot} yang lebar di bawah \omterm{def:g7:how-animals-breathe:four}{paru-parunya}, yang menutup dadanya dari
bawah; pendatarannya adalah paruh yang dalam dari setiap tarikan napas.
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:3}
Kantong udara \omterm{def:g7:how-animals-breathe:four}{paru-parunya}: ratusan juta gelembung berdinding tipis
pada ujung terhalus saluran udaranya, masing-masing terbungkus
\omterm{def:g4:heart-and-blood:vessels}{pembuluh darah} halus.
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:4}
Yang ditarik masuk: kira-kira $21$ persen \omterm{def:g7:what-is-respiration:respiration}{oksigen}, nyaris tanpa \omterm{def:g7:what-is-respiration:respiration}{karbon
dioksida}. Yang diembuskan: kira-kira $16$ persen \omterm{def:g7:what-is-respiration:respiration}{oksigen} dan kira-kira
$4$ persen \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}. Perbedaannya --- lima angka \omterm{def:g7:what-is-respiration:respiration}{oksigen} yang
diambil, empat angka \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} yang ditambahkan --- adalah
pertukarannya sendiri.
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:5}
Permukaan penyeberangan yang luas-tipis-lembap dan berlapis \omterm{prop:g4:journey-of-food:absorb}{darah}.
Dibagi bersama umbai \omterm{def:g7:how-animals-breathe:four}{insangnya}
(\cref{prop:g7:how-animals-breathe:spec}) dan jonjot ususnya
(\cref{ex:g7:digestion-and-nutrients:spec}).
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:6}
Tar yang melapisi dan mengiritasi saluran udaranya (sapu yang
kelebihan beban dan batuknya); dinding \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolus} yang rusak ---
permukaannya hilang; sebuah gas asap yang menumpangi kursi \omterm{def:g7:what-is-respiration:respiration}{oksigen} di
dalam \omterm{prop:g4:journey-of-food:absorb}{darahnya}; saluran udara yang menyempit sehingga mencekik
puputannya.
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:7}
\omterm{def:g4:breathing:movements}{Menarik napas} harus merentangkan dada dan \omterm{def:g7:how-animals-breathe:four}{paru-parunya} melawan
lentingannya --- perentangan itulah pekerjaannya. \omterm{def:g4:breathing:movements}{Mengembuskan napas}
ketika beristirahat adalah lentingannya sendiri yang kembali: dada
yang melebar dan \omterm{def:g7:how-animals-breathe:four}{paru-paru} yang terentang melenting balik, dan
lentingannya mendorong udaranya.
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:8}
Pertukarannya membutuhkan permukaan sebanyak ia membutuhkan udara
segar: penyeberangan hanya terjadi di tempat udara berjumpa dinding
yang berlapis \omterm{prop:g4:journey-of-food:absorb}{darah}. Puputan yang lebih cepat melewati kantong yang
lebih sedikit dan lebih kendur mengudarai permukaan yang tidak lagi
ada --- faktor yang hilang adalah luas penyeberangannya sendiri.
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:9}
Si perokok mendaki dengan saluran udara yang menyempit (pemasukan yang
tercekik), permukaan \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolus} yang berkurang (pemuatan yang lebih
sedikit tiap tarikan napas), dan kursi \omterm{def:g7:what-is-respiration:respiration}{oksigen} yang diduduki gas
asapnya (yang terbawa lebih sedikit tiap ronde). Untuk membayar
tagihan \omterm{def:g3:bones-and-muscles:muscle}{otot} yang sama, \omterm{def:g4:heart-and-blood:heart}{jantungnya} harus menjalankan lebih banyak
ronde --- karena itulah ia tiba terengah-engah pada \omterm{prop:g4:heart-and-blood:pulse}{denyut nadi}
setingkat lari cepat.
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:10}
Keduanya adalah kendali terlatih atas tahap \emph{keluarnya}:
menambahkan \omterm{def:g3:bones-and-muscles:muscle}{otot} yang terukur pada sesuatu yang biasanya berupa
lentingan yang pasif --- si penyelam untuk membuang udara yang
mengembang dengan mantap, si pemain untuk menakar satu aliran keluar
yang panjang dan terkendali. (Tarikan dalam mereka melatih pekerjaan
masuknya juga.)
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:11}
\omterm{def:g7:what-is-respiration:respiration}{Pernapasan} yang lebih dalam dan lebih cepat melipatgandakan pemuatan
pada loket alveolusnya; \omterm{def:g4:heart-and-blood:heart}{jantung} yang lebih cepat dan lebih keras
melipatgandakan rondenya pada loket pengangkutannya; penjaluran
ulangnya melebarkan \omterm{def:g4:heart-and-blood:vessels}{pembuluh} setempat \omterm{def:g3:bones-and-muscles:muscle}{ototnya} pada loket
pengantarannya. Tiga penyetelan, satu pengantaran yang berlipat ganda
(buku yang seimbang pada
\cref{ex:g7:lungs-and-gas-exchange:accounting}).
\end{solution}

\begin{solution}{exo:g7:lungs-and-gas-exchange:12}
Perbatasannya: ratusan juta \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolus} berjumlah kira-kira separuh
lapangan tenis permukaan pertukaran, yang terlipat di dalam sebuah
dada --- dan menyeberanginya lima angka persentase \omterm{def:g7:what-is-respiration:respiration}{oksigen} tiap
tarikan napas menyeberang masuk dan empat angka \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}
menyeberang keluar. Api yang lambat: tar dan iritasi asapnya melahap
perbatasan itu tahun demi tahun --- melapisi jalan pendekatnya,
meruntuhkan kantongnya, merebut kursi pengangkutannya --- menumpuk,
diam-diam, dan hanya sebagian dapat dipulihkan, seperti sebuah
perbatasan yang terbakar lebih cepat daripada yang dapat dibangun
ulang.
\end{solution}

\begin{solution}{pb:g7:lungs-and-gas-exchange:1}
\textbf{1.} Tahap masuk: \omterm{def:g3:bones-and-muscles:muscle}{otot} rusuknya mengangkat lalu melebarkan
sangkarnya, \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{diafragmanya} mendatar --- berharga \omterm{prop:g3:balanced-diet:jobs}{energi}. Tahap keluar
(ketika beristirahat): \omterm{def:g3:bones-and-muscles:muscle}{ototnya} melepas, dada dan \omterm{def:g7:how-animals-breathe:four}{paru-parunya}
melenting kembali --- nyaris gratis. Daurnya kira-kira $15$ kali
semenit.

\textbf{2.} \omterm{def:g7:how-animals-breathe:four}{Paru-parunya} melekat pada dinding dalam dadanya:
lebarkanlah dadanya dan \omterm{def:g7:how-animals-breathe:four}{paru-parunya} ikut terentang lebar bersamanya,
lalu udara luarnya mengalir masuk untuk mengisi ruang yang terentang
itu. Pompanya bekerja dengan membentuk ulang ruangannya, bukan dengan
memeras organnya.

\textbf{3.} Kira-kira $15$ daur semenit ketika beristirahat; ketika
berusaha daurnya mencepat \emph{dan} mendalam --- lebih banyak tarikan
napas, dan lebih banyak udara tiap tarikan.

\textbf{4.} Terengah-engah: daur pendek yang cepat, nyaris seluruhnya
\omterm{prop:g7:lungs-and-gas-exchange:ventilation}{diafragma}. Mendesah: satu perentangan yang luar biasa dalam dan sebuah
pelepasan yang panjang --- penyetelan ulang lentingannya. Lilin: tahap
keluarnya dengan \omterm{def:g3:bones-and-muscles:muscle}{otot} yang ditambahkan pada lentingannya.

\textbf{5.} Alveolusnya: ratusan juta gelembung berdinding tipis,
masing-masing terbungkus \omterm{def:g4:heart-and-blood:vessels}{pembuluh} halus, yang berjumlah kira-kira
separuh lapangan tenis.

\textbf{6.} Luas: lapangan yang berjumlah itu. Tipis: dinding
gelembung setebal sebuah \omterm{def:g5:first-look-at-cells:cell}{sel}. Lembap: lapisannya. Berlapis \omterm{prop:g4:journey-of-food:absorb}{darah}:
\omterm{def:g4:heart-and-blood:vessels}{pembuluh} yang membungkusnya. Spesifikasinya terpenuhi pada keempat
hitungannya.

\textbf{7.} Lima angka yang disimpan dari $21$ --- kasarnya seperempat
\omterm{def:g7:what-is-respiration:respiration}{oksigen} yang ditarik masuk diambil tiap tarikan napas.

\textbf{8.} Harga \omterm{def:g6:exploring-our-environment:conditions}{kelembapan} permukaannya: udara yang pergi sudah
menyapu melintasi dinding yang basah (dan hangat), sehingga buangannya
mengangkut pergi air dan kehangatan --- cermin yang berembun adalah
biaya pemeliharaan permukaannya, yang dibayar tiap tarikan napas.

\textbf{9.} Pemasukan: tar melapisi dan mengiritasi saluran udaranya.
Pompa: saluran udara yang menyempit mencekik setiap daurnya.
Permukaan: dinding alveolusnya rusak lalu bergabung --- luasnya
hilang. Pengangkutan: gas asapnya menduduki kursi \omterm{def:g7:what-is-respiration:respiration}{oksigen} di dalam
\omterm{prop:g4:journey-of-food:absorb}{darahnya}.

\textbf{10.} Batuk dan pilek meradang lalu membersihkan ---
permesinannya memperbaiki diri lalu kembali bertugas. Dinding \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolus}
yang rusak tidak membangun ulang: tiap kantong yang bergabung adalah
luas pertukaran yang dicoret dari pembukuannya untuk selamanya ---
modal, bukan biaya berjalan.

\textbf{11.} (1) Jangan masukkan asap: kehilangan modal itu selamanya.
(2) Berjalanlah dengan udara yang bersih: penyaring dan sapunya punya
daya tampung yang terbatas. (3) Latihlah pompanya di bawah beban:
usaha yang dalam memelihara \omterm{def:g3:bones-and-muscles:muscle}{ototnya} dan mengudarai seluruh
permukaannya. (4) Masukkan udara lewat hidung di udara dingin dan
berdebu: penyiapan awalnya melindungi perangkatnya.

\textbf{12.} Sendirian di antara pompa tubuhnya, puputannya menjawab
baik layanan otomatisnya maupun kehendaknya --- karena aliran keluar
yang sama yang membuang \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} adalah angin bagi ucapan dan
nyanyian, dan sebuah mesin yang dipinjamkan kepada bahasa harus
menerima tangan sang pembicara pada kendalinya.
\end{solution}
