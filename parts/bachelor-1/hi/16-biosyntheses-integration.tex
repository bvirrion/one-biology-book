\chapter{जैवसंश्लेषण और समाकलित कोशिका}\label{ch:b1:biosyntheses-integration}

सुबह आठ बजे कोई यकृत नाश्ते की शर्करा को \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} और \omterm{def:b1:lipids:triglyceride}{वसा} में बदल रहा
होता है; और अगली सुबह तीन बजे वही यकृत पेशी के \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} से शर्करा बना
रहा होता है और उसे उस मस्तिष्क के लिए रक्त में उँड़ेल रहा होता है जिसने
नौ घंटे से कुछ नहीं खाया। \omterm{def:b1:enzymes:enzyme}{एंजाइमों} में कुछ नहीं बदला; जो बदला वह यह है कि
उनमें से कौन चालू हैं। पिछले अध्यायों ने \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के अणु अलग-अलग किए; यह
अध्याय उन्हें जोड़ता है --- शर्कराओं, \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन}, \omterm{def:b1:lipids:triglyceride}{वसाओं} और \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} के
संश्लेषण --- और फिर पूछता है कि कोई \omterm{def:b1:cell-unit-of-life:cell}{कोशिका}, और कोई \omterm{def:b1:organism-environment:organism}{जीव}, हर क्षण यह कैसे
तय करते हैं कि कौन-से पथ चलें। उत्तर है: साझा मुद्राओं का एक छोटा
समुच्चय, चौराहों पर कुछ \omterm{def:b1:enzymes:enzyme}{एंजाइम}, और वे हॉर्मोन जो हर \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को समग्र की
अवस्था बता देते हैं।

\section{किसी संश्लेषण को क्या चाहिए}

\begin{proposition}[उपचय की तीन आवश्यकताएँ]\label{prop:b1:biosyntheses-integration:needs}
\index{उपचय}
हर जैवसंश्लेषण को \emph{कार्बन कंकाल} चाहिए, जो \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} और \omterm{def:b1:respiration-fermentation:krebs}{क्रेब्स
चक्र} के मुट्ठी भर मध्यवर्तियों से लिए जाते हैं; \emph{अपचायक
शक्ति}, जो \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} के रूप में मिलती है; और \emph{ऊर्जा}, जो \omterm{def:b1:water-small-molecules:atp}{ATP} के रूप में।
इसलिए अपचय और उपचय उन्हीं चौराहों पर मिलते हैं ---
ग्लूकोज़-6-फ़ॉस्फ़ेट, ट्रायोस फ़ॉस्फ़ेट, पाइरुवेट, \omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA},
$\alpha$-कीटोग्लूटारेट, ऑक्सैलोऐसीटेट --- पर अनुत्क्रमणीय चरणों पर
अलग-अलग \omterm{def:b1:enzymes:enzyme}{एंजाइमों} से चलते हैं, ताकि हर दिशा स्वतंत्र रूप से चालू-बंद की जा
सके, और वे अपने इलेक्ट्रॉनों के लिए अलग-अलग \omterm{def:b1:enzymes:enzyme}{सहएंजाइम} काम में लेते हैं:
अपचय में इलेक्ट्रॉन हटाने के लिए NADH, जिसे ऑक्सीकृत रखा जाता है; और
संश्लेषण में उन्हें सौंपने के लिए \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}, जिसे अपचयित रखा जाता है।
\end{proposition}

\begin{definition}[पेंटोस फ़ॉस्फ़ेट पथ]\label{def:b1:biosyntheses-integration:ppp}
\index{पेंटोस फ़ॉस्फ़ेट पथ}\index{NADPH}
\omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} का \emph{पेंटोस फ़ॉस्फ़ेट पथ}
ग्लूकोज़-6-फ़ॉस्फ़ेट का ऑक्सीकरण करके राइबुलोस-5-फ़ॉस्फ़ेट और
$\mathrm{CO_2}$ बनाता है, तथा दो $\mathrm{NADP^+}$ को NADPH में अपचयित
करता है; और उसकी अनॉक्सीकारक शाखा फिर पाँच-, चार-, छह- और सात-कार्बन वाले
शर्करा फ़ॉस्फ़ेटों को आपस में बदलती रहती है, ताकि \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} पेंटोस चाहिए
होने पर \omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइडों} के लिए राइबोस-5-फ़ॉस्फ़ेट बना सके, अपचायक शक्ति
चाहिए होने पर NADPH (वसा-संश्लेषण, ऑक्सीकारकों से बचाव), या दोनों, और
बाक़ी \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} को लौटा दे। जंतु \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} में NADPH का मुख्य स्रोत
यही है; और पौधों में दिन के समय \omterm{def:b1:eukaryotic-cell:plastid}{हरितलवक} की \omterm{prop:b1:photosynthesis:lightreactions}{प्रकाश अभिक्रियाएँ} NADPH देती
हैं।
\end{definition}

\section{ग्लूकोज़ बनाना: ग्लूकोजनन}

\begin{definition}[ग्लूकोजनन]\label{def:b1:biosyntheses-integration:gluconeogenesis}
\index{ग्लूकोजनन}
\emph{ग्लूकोजनन} अकार्बोहाइड्रेट पूर्ववर्तियों से ग्लूकोज़ बनाता है
--- लैक्टेट, पाइरुवेट, ग्लिसरॉल और अधिकांश \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} के कार्बन कंकाल
--- और यह यकृत में होता है (और थोड़ा गुर्दे में)। वह \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} को
उसके सात उत्क्रमणीय चरणों से उल्टा चलाता है और तीनों अनुत्क्रमणीय चरणों को
अलग \omterm{def:b1:enzymes:enzyme}{एंजाइमों} से घुमा देता है: पाइरुवेट का कार्बोक्सिलन होकर
ऑक्सैलोऐसीटेट बनता है (\omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका} में, एक \omterm{def:b1:water-small-molecules:atp}{ATP}) और उसका विकार्बोक्सिलन होकर
फ़ॉस्फ़ोइनॉलपाइरुवेट (एक GTP); फ्रक्टोस-1,6-बिसफ़ॉस्फ़ेट का \omterm{prop:b1:water-small-molecules:families}{जल-अपघटन} होकर
फ्रक्टोस-6-फ़ॉस्फ़ेट बनता है; और ग्लूकोज़-6-फ़ॉस्फ़ेट का \omterm{prop:b1:water-small-molecules:families}{जल-अपघटन} होकर
मुक्त ग्लूकोज़, जो \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} से निकल जाता है। प्रति ग्लूकोज़: 6 \omterm{def:b1:water-small-molecules:atp}{ATP}
तुल्यांक और 2 NADH, जबकि \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} 2 \omterm{def:b1:water-small-molecules:atp}{ATP} देता है --- यही किसी ढलान
वाले पथ को चढ़ाई पर चलाने की क़ीमत है।
\end{definition}

\begin{proposition}[क्या ग्लूकोज़ बन सकता है और क्या नहीं]\label{prop:b1:biosyntheses-integration:cannot}
लैक्टेट, ग्लिसरॉल, और वे \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} जो पाइरुवेट या क्रेब्स-चक्र के
मध्यवर्ती देते हैं (ल्यूसीन और लाइसीन को छोड़कर सब) ग्लूकोज़ में बदले जा
सकते हैं। जंतुओं में \omterm{def:b1:lipids:fattyacid}{वसा अम्ल} नहीं बदल सकते: उनका $\beta$-ऑक्सीकरण
\omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} देता है, जिसके दोनों कार्बन चक्र में घुसते हैं और कोई
ऑक्सैलोऐसीटेट कमाए जाने से पहले ही दो $\mathrm{CO_2}$ के रूप में निकल
जाते हैं, इसलिए \omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} से शर्करा तक कोई शुद्ध रास्ता नहीं है। पौधों,
कवकों और जीवाणुओं के पास \emph{ग्लाइऑक्सिलेट चक्र} है, यानी वह छोटा
रास्ता जो दोनों विकार्बोक्सिलन छोड़ देता है और किसी अंकुरित होते तैलीय
बीज को अपनी \omterm{def:b1:lipids:triglyceride}{वसा} उस शर्करा में बदलने देता है जो उसके अंकुर को चाहिए।
इसके विपरीत किसी भूखे स्तनधारी को अपना ग्लूकोज़ \omterm{def:b1:proteins:peptide}{प्रोटीन} से बनाना पड़ता है।
\end{proposition}

\begin{example}[कोरी चक्र]\label{ex:b1:biosyntheses-integration:cori}
दौड़ती कोई पेशी \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} को किण्वित करके लैक्टेट बनाती है, जो रक्त में
घुस जाता है; यकृत उस लैक्टेट को ले लेता है, उसका ऑक्सीकरण करके पाइरुवेट
बनाता है, उससे प्रति ग्लूकोज़ 6 \omterm{def:b1:water-small-molecules:atp}{ATP} की क़ीमत पर ग्लूकोज़ बनाता है, और वह
ग्लूकोज़ पेशी के फिर काम आने के लिए रक्त में लौटा देता है। पेशी ने बिना
ऑक्सीजन के प्रति ग्लूकोज़ 2 \omterm{def:b1:water-small-molecules:atp}{ATP} कमाए और 6 का बिल यकृत को थमा दिया, जो उसे
फ़ुरसत से ऑक्सीजन के साथ चुकाता है: किसी दौड़ का ``ऑक्सीजन ऋण'' बड़े
हिस्से में यकृत ही चुकाता है।
\end{example}

\section{संचय: ग्लाइकोजन और वसा}

\begin{definition}[ग्लाइकोजन संश्लेषण]\label{def:b1:biosyntheses-integration:glycogensynthesis}
ग्लूकोज़ को \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} के रूप में संचित करने के लिए
ग्लूकोज़-6-फ़ॉस्फ़ेट को \emph{UDP-ग्लूकोज़} में सक्रिय किया जाता है (एक
UTP), जिसे \emph{\omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} सिंथेज़} पहले से मौजूद वृक्ष के अनपचायक सिरों
पर जोड़ती है; और कोई \emph{शाखन \omterm{def:b1:enzymes:enzyme}{एंजाइम}} छोटी शृंखलाएँ काटकर फिर जोड़
देता है ताकि $\alpha(1{\to}6)$ शाखाएँ बनें
(\cref{ch:b1:carbohydrates})। सिंथेज़ और फ़ॉस्फ़ोरिलेज़ को वही संकेत
विपरीत दिशाओं में नियंत्रित करते हैं: जो फ़ॉस्फ़ोरिलीकरण
फ़ॉस्फ़ोरिलेज़ चालू करता है वही सिंथेज़ बंद कर देता है, इसलिए वृक्ष कभी
एक ही साथ बनता और गिरता नहीं।
\end{definition}

\begin{definition}[वसा अम्ल संश्लेषण]\label{def:b1:biosyntheses-integration:fasynthesis}
\index{वसा अम्ल संश्लेषण}
\omterm{def:b1:lipids:fattyacid}{वसा अम्ल} \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} में \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका} से निर्यात हुए \omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} से बनते
हैं। \emph{\omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} कार्बोक्सिलेज़}, जो नियमित चरण है,
उसका कार्बोक्सिलन करके (एक \omterm{def:b1:water-small-molecules:atp}{ATP}) मैलोनिल-CoA बनाता है; फिर \emph{\omterm{def:b1:lipids:fattyacid}{वसा अम्ल}
सिंथेज़} मैलोनिल-CoA से एक बार में दो कार्बन किसी बढ़ती शृंखला पर जोड़ता
है और हर जोड़ को दो \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} से अपचयित करता है, जब तक पामिटेट (C16)
न छूट जाए: प्रति पामिटेट 8 \omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA}, 7 \omterm{def:b1:water-small-molecules:atp}{ATP} और 14 \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}। बाक़ी अम्ल
\omterm{def:b1:eukaryotic-cell:endomembrane}{अंतर्द्रव्यी जालिका} में दीर्घीकरण तथा असंतृप्तीकरण से बनते हैं; और
ग्लिसरॉल-3-फ़ॉस्फ़ेट (जो ट्रायोस फ़ॉस्फ़ेट से आता है) से एस्टरीकरण
\omterm{def:b1:lipids:triglyceride}{ट्राइग्लिसराइड} बनाता है। यह पथ उल्टा चलाया गया $\beta$-ऑक्सीकरण नहीं है:
अलग \omterm{def:b1:enzymes:enzyme}{एंजाइम}, अलग कक्ष, \omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} के बजाय मैलोनिल-CoA, और
$\mathrm{FADH_2}$ तथा NADH के बजाय \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} --- और
मैलोनिल-CoA \omterm{def:b1:lipids:fattyacid}{वसा अम्लों} का \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका} में अभिगमन रोक देता है,
इसलिए कोई \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} एक ही साथ \omterm{def:b1:lipids:triglyceride}{वसा} बनाती और जलाती नहीं।
\end{definition}

\begin{example}[शर्करा से वसा]\label{ex:b1:biosyntheses-integration:fatfromsugar}
जिस यकृत को \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} के रूप में संचित कर सकने से अधिक ग्लूकोज़ मिले वह
\omterm{def:b1:lipids:triglyceride}{वसा} बनाता है: ग्लूकोज़ से पाइरुवेट, फिर \omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} (जिसमें कार्बन का
एक-तिहाई $\mathrm{CO_2}$ के रूप में खोता है और \omterm{def:b1:water-small-molecules:atp}{ATP} कमाया जाता है), फिर
\omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} से पामिटेट, उसी \omterm{def:b1:water-small-molecules:atp}{ATP} की और \omterm{def:b1:biosyntheses-integration:ppp}{पेंटोस फ़ॉस्फ़ेट पथ} के \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} की
क़ीमत पर, और फिर पामिटेट से \omterm{def:b1:lipids:triglyceride}{ट्राइग्लिसराइड}, जो \omterm{def:b1:lipids:triglyceride}{वसा} \omterm{def:b1:body-plans-tissues:tissue}{ऊतक} को भेज दिया जाता
है। शर्करा की ऊर्जा का कोई \qty{15}{\%} इस बदलाव में खो जाता है; और बाक़ी
\omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} से नौ गुना अधिक संहत रूप में संचित होता है
(\cref{ch:b1:lipids})। इसका उल्टा --- \omterm{def:b1:lipids:triglyceride}{वसा} से शर्करा --- असंभव है।
\end{example}

\section{नाइट्रोजन: ऐमीनो अम्ल और न्यूक्लियोटाइड}

\begin{definition}[नाइट्रोजन स्वांगीकरण, ऐमीन-अंतरण]\label{def:b1:biosyntheses-integration:nitrogen}
\index{ऐमीन-अंतरण}\index{ग्लूटामेट}
अमोनियम कार्बनिक पदार्थ में लगभग पूरा
\emph{ग्लूटामेट} के रास्ते घुसता है: ग्लूटामेट डिहाइड्रोजनेज़ उसे
$\alpha$-कीटोग्लूटारेट से जोड़ती है, और ग्लूटामीन सिंथेटेज़ एक दूसरा
अमोनियम ग्लूटामेट की \omterm{def:b1:proteins:aminoacid}{पार्श्व शृंखला} पर जोड़कर (एक \omterm{def:b1:water-small-molecules:atp}{ATP}) ग्लूटामीन बनाती है।
इन दोनों दाताओं से \emph{ऐमीन-अंतरेज़} (ऐमीनोहस्तांतरेज़, जिनका \omterm{def:b1:enzymes:enzyme}{सहएंजाइम}
विटामिन B$_6$ है) ऐमीनो समूह किसी भी $\alpha$-कीटो अम्ल पर पहुँचा देते
हैं, और इस तरह केंद्रीय पथों के कार्बन कंकालों से \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} बनते हैं:
$\alpha$-कीटोग्लूटारेट से ग्लूटामेट कुल, ऑक्सैलोऐसीटेट से ऐस्पार्टेट
कुल, और पाइरुवेट तथा 3-फ़ॉस्फ़ोग्लिसरेट से ऐलानीन और सेरीन। पौधे और
जीवाणु बीसों बनाते हैं; जंतु उनमें से नौ तक जाने वाले लंबे पथ खो चुके हैं,
यानी \emph{आवश्यक} \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल}, और उन्हें भोजन से लेते हैं। वही
ऐमीन-अंतरेज़, उल्टे चलाकर, अतिरिक्त \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} से नाइट्रोजन उतारकर
अमोनिया (मछलियाँ), यूरिया (स्तनधारी) या यूरिक अम्ल (पक्षी, कीट) के रूप
में उत्सर्जन के लिए भेज देते हैं।
\end{definition}

\begin{example}[न्यूक्लियोटाइड]\label{ex:b1:biosyntheses-integration:nucleotides}
कोई \omterm{def:b1:nucleic-acids:nucleotide}{प्यूरीन} वलय राइबोस-5-फ़ॉस्फ़ेट पर ग्लाइसीन,
ऐस्पार्टेट, दो ग्लूटामीन, दो एक-कार्बन इकाइयों और $\mathrm{CO_2}$ से जोड़ा
जाता है, और इसकी क़ीमत छह \omterm{def:b1:water-small-molecules:atp}{ATP} है; और कोई \omterm{def:b1:nucleic-acids:nucleotide}{पिरिमिडीन} ऐस्पार्टेट तथा
कार्बामॉइल फ़ॉस्फ़ेट से। डिऑक्सीन्यूक्लियोटाइड राइबोन्यूक्लियोटाइडों से
शर्करा का अपचयन करके बनाए जाते हैं, और इसमें \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} काम आता है। विभाजित
होने को तैयार कोई \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} घंटे भर में कोई $10^{10}$
\omterm{def:b1:nucleic-acids:nucleotide}{न्यूक्लियोटाइड} बनाती है; और जो दवाएँ इन संश्लेषणों को रोकती हैं ---
मेथोट्रेक्सेट, 5-फ़्लोरोयूरैसिल --- सबसे पहले विभाजित होती \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाओं} को
रोकती हैं, और इसीलिए वे कैंसर का इलाज करती हैं।
\end{example}

\section{समाकलित स्तनधारी}

\begin{proposition}[भरा हुआ और उपवास में]\label{prop:b1:biosyntheses-integration:fedfasting}
\index{इंसुलिन}\index{ग्लूकैगॉन}
\omterm{def:b1:mammal-organization:organ}{अंग} काम बाँट लेते हैं। भोजन के बाद अग्न्याशय का \emph{इंसुलिन}
यकृत से कहता है कि ग्लूकोज़ को \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} के रूप में संचित करे और
अतिरिक्त को \omterm{def:b1:lipids:triglyceride}{वसा} में बदले, पेशियों से कि ग्लूकोज़ लें और \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} संचित
करें, और \omterm{def:b1:lipids:triglyceride}{वसा} \omterm{def:b1:body-plans-tissues:tissue}{ऊतक} से कि \omterm{def:b1:lipids:triglyceride}{वसा} ले ले; और रक्त का ग्लूकोज़, जो चढ़कर
\qty{8}{mmol/L} हो गया था, दो घंटे में \qty{5}{} पर लौट आता है। भोजनों के
बीच \emph{ग्लूकैगॉन} यकृत से कहता है कि अपना \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} तोड़े और, उसके
चुकते जाने पर, लैक्टेट, ग्लिसरॉल तथा \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्लों} से ग्लूकोज़ बनाए, और
\omterm{def:b1:lipids:triglyceride}{वसा} \omterm{def:b1:body-plans-tissues:tissue}{ऊतक} से कि \omterm{def:b1:lipids:fattyacid}{वसा अम्ल} छोड़े, जिन्हें पेशियाँ तथा यकृत ग्लूकोज़ की जगह
जलाते हैं; और मस्तिष्क, जो \omterm{def:b1:lipids:triglyceride}{वसा} नहीं जला सकता, अपने \qty{120}{g} ग्लूकोज़
प्रतिदिन बनाए रखता है। दिनों के उपवास में यकृत \omterm{def:b1:lipids:fattyacid}{वसा अम्लों} को
\emph{कीटोन काय} में बदल देता है, जो छोटे अम्ल हैं और जिन्हें मस्तिष्क
काम में ले सकता है, और तब ग्लूकोज़ की --- तथा उसे बनाने वाले \omterm{def:b1:proteins:peptide}{प्रोटीन} की
--- माँग दो-तिहाई घट जाती है। हॉर्मोनी क्रियाविधियाँ वर्ष 2 के खंड की
हैं; उपापचयी तर्क इस अध्याय का।
\end{proposition}

\begin{method}[किसी उपापचयी अवस्था को पढ़ना]\label{met:b1:biosyntheses-integration:state}
\begin{enumerate}
  \item पूछिए कि कौन-सा ईंधन प्रचुर है: ऊँचे ग्लूकोज़ का अर्थ है संचय
        (\omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन}, फिर \omterm{def:b1:lipids:triglyceride}{वसा}); और नीचे ग्लूकोज़ का अर्थ है जुटाव।
  \item संकेतों का पीछा कीजिए: इंसुलिन सिंथेज़ों को सक्रिय करता है और
        फ़ॉस्फ़ोरिलेज़ तथा लाइपेज़ को निष्क्रिय; ग्लूकैगॉन उल्टा करता है
        --- और अधिकतर उन्हीं \omterm{def:b1:enzymes:enzyme}{एंजाइमों} के फ़ॉस्फ़ोरिलीकरण से
        (\cref{ch:b1:enzymes})।
  \item चौराहे जाँचिए: \omterm{def:b1:water-small-molecules:atp}{ATP} और साइट्रेट PFK रोकते हैं तथा \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{ग्लूकोजनन}
        खोलते हैं; AMP उल्टा करता है; मैलोनिल-CoA \omterm{def:b1:lipids:triglyceride}{वसा} बनते समय \omterm{def:b1:lipids:triglyceride}{वसा} का
        जलना रोकता है; और \omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} उस कार्बोक्सिलेज़ को सक्रिय करता
        है जो \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{ग्लूकोजनन} शुरू करती है।
  \item \omterm{def:b1:mammal-organization:organ}{अंग} बाँटिए: यकृत (ग्लूकोज़ संचित करता, बनाता और निर्यात करता है;
        कीटोन तथा यूरिया बनाता है), पेशी (अपने लिए \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} संचित करती
        है, लैक्टेट तथा ऐलानीन निर्यात करती है), \omterm{def:b1:lipids:triglyceride}{वसा} \omterm{def:b1:body-plans-tissues:tissue}{ऊतक} (\omterm{def:b1:lipids:triglyceride}{वसा} संचित
        करता और छोड़ता है), मस्तिष्क (ग्लूकोज़ जलाता है, फिर कीटोन)।
\end{enumerate}
\end{method}

\section{समाकलित पौधा}

\begin{proposition}[स्रोत और ग्राही]\label{prop:b1:biosyntheses-integration:sourcesink}
दिन में किसी \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} के \omterm{def:b1:eukaryotic-cell:plastid}{हरितलवक} ट्रायोस फ़ॉस्फ़ेट \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} को निर्यात
करते हैं, जहाँ उसे \emph{सुक्रोस} बना दिया जाता है ताकि वह फ्लोएम में
(\cref{ch:b1:plant-transport}) \omterm{def:b1:flowering-plant-organization:organs}{जड़ों}, फलों और बढ़ते सिरों तक जाए;
और जो फ्लोएम नहीं ले जा सकता वह \omterm{def:b1:eukaryotic-cell:plastid}{हरितलवक} में \emph{\omterm{def:b1:carbohydrates:polysaccharide}{मंड}} के रूप में रख लिया
जाता है, जो एक क्षणिक भंडार है और जिसे रात में जुटाकर सुक्रोस का प्रवाह
बनाए रखा जाता है। जहाँ किसी स्तनधारी का यकृत \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} और निर्यात के बीच
का निर्णय हॉर्मोनों से करता है, वहाँ कोई \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} \omterm{def:b1:carbohydrates:polysaccharide}{मंड} और सुक्रोस के बीच का
निर्णय \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} में फ़ॉस्फ़ेट के स्तर से करती है, जो \omterm{def:b1:eukaryotic-cell:plastid}{हरितलवक} के आवरण के
निर्यातक को नियंत्रित करता है। कोई पौधा अपने ही उपापचयजों की सांद्रताओं
से, अंग-दर-अंग नियमन करता है; और स्तनधारी उस पर पूरे शरीर की एक
तंत्रिकीय तथा हॉर्मोनी कमान जोड़ देता है।
\end{proposition}

\begin{example}[एक रात का मंड]\label{ex:b1:biosyntheses-integration:nightstarch}
कोई \omterm{def:b1:flowering-plant-organization:organs}{पत्ती} दिन में जो स्थिर करती है उसका लगभग आधा \omterm{def:b1:carbohydrates:polysaccharide}{मंड} के रूप में अलग रख
देती है और रात भर उसे लगभग अचर दर से तोड़ती है, ताकि भंडार ठीक भोर से
पहले चुक जाए: रात को कृत्रिम रूप से छोटा कीजिए तो \omterm{def:b1:carbohydrates:polysaccharide}{मंड} बच जाता है; और लंबा
कीजिए तो पौधा अंतिम घंटों में भूखा रह जाता है। यह दर अँधेरे के पहले घंटे
के भीतर भंडार के आकार और रात की अपेक्षित लंबाई के अनुसार समायोजित हो जाती
है --- यानी \omterm{def:b1:enzymes:enzyme}{एंजाइमों} और एक घड़ी से की गई एक गणना, उस \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में जिसका कोई
मस्तिष्क नहीं है।
\end{example}

  \item असल में \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{ग्लूकोजनन} पहले घंटों से ही लगभग एक-तिहाई ग्लूकोज़ देता
        है। \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} की अवधि फिर से निकालिए।
  \item हर आकलन के अनुसार सुबह किस समय यकृत का \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} चुक जाता है?
  \item इस बीच पेशियाँ \omterm{def:b1:lipids:fattyacid}{वसा अम्ल} जलाती हैं। यदि वे विश्राम-व्यय का
        \qty{60}{\%} लें और केवल \omterm{def:b1:lipids:triglyceride}{वसा} जलाएँ, तो रात भर (\qty{12}{h}) में
        ऑक्सीकृत \omterm{def:b1:lipids:triglyceride}{वसा} निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- अगला दिन, बिना खाए।}
\begin{enumerate}[resume]
  \item \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} चुक जाने पर वे \qty{160}{g} ग्लूकोज़ \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{ग्लूकोजनन} से आने
        चाहिए। प्रतिदिन चाहिए \omterm{def:b1:proteins:peptide}{प्रोटीन}, और वह कितनी पेशी के बराबर है, यह
        निकालिए।
  \item पाइरुवेट से \qty{160}{g} ग्लूकोज़ बनाने में यकृत जितना \omterm{def:b1:water-small-molecules:atp}{ATP}
        ख़र्च करता है (प्रति ग्लूकोज़ 6 \omterm{def:b1:water-small-molecules:atp}{ATP}) वह और
        \qty{50}{kJ/mol} पर उसकी ऊर्जा निकालिए।
  \item \omterm{def:b1:lipids:triglyceride}{वसा} के ग्लिसरॉल का योगदान निकालिए: यदि \omterm{def:b1:lipids:triglyceride}{वसा} सारा
        अ-ग्लूकोज़ व्यय ढके तो प्रतिदिन कितनी \omterm{def:b1:lipids:triglyceride}{वसा} ऑक्सीकृत होती है,
        और उसका ग्लिसरॉल कितना ग्लूकोज़ देता है?
  \item ग्लिसरॉल का योगदान घटाकर चाहिए \omterm{def:b1:proteins:peptide}{प्रोटीन} फिर से निकालिए।
  \item उस दर पर आधी पेशी चुकने में कितने दिन लगेंगे?
  \item तीसरे दिन से यकृत कीटोन काय बनाता है और मस्तिष्क अपनी ऊर्जा का
        दो-तिहाई उन्हीं से लेता है; तब ग्लूकोज़ की माँग गिरकर
        मस्तिष्क के लिए \qty{40}{g} और बाक़ी के लिए \qty{40}{g} रह जाती
        है। प्रतिदिन प्रोटीन-हानि और आधी पेशी चुकने के दिन फिर से
        निकालिए।
  \item प्रतिदिन \qty{8.4}{MJ} पर \omterm{def:b1:lipids:triglyceride}{वसा} भंडार कितने दिन चलता है, यह
        निकालिए (लंबे उपवास में यह गिरकर \qty{6.5}{MJ} हो जाता है: फिर
        से निकालिए)।
  \item समझाइए कि उपवास कर रहा कोई व्यक्ति \omterm{def:b1:lipids:triglyceride}{वसा} चुकने से पहले
        प्रोटीन-हानि से क्यों मरता है, और कीटोन काय किसे टाल देते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वाल्व।}
\begin{enumerate}[resume]
  \item यकृत का वह \omterm{def:b1:enzymes:enzyme}{एंजाइम} बताइए जो 03:00 बजे सक्रिय और 09:00 बजे
        निष्क्रिय होना चाहिए, और वह भी जिसे \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} के लिए इसका उल्टा
        होना चाहिए।
  \item वह संकेत (हॉर्मोन) बताइए जो उन्हें हर दिशा में बिठाता है, और वह
        रासायनिक रूपांतरण भी जिससे वह काम करता है।
  \item 03:00 बजे किसी यकृत \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में वसा-ऑक्सीकरण से \omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} ऊँचा
        है। \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{ग्लूकोजनन} का वह \omterm{def:b1:enzymes:enzyme}{एंजाइम} बताइए जिसे यह सक्रिय करता है और
        \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} का वह \omterm{def:b1:enzymes:enzyme}{एंजाइम} भी जिसे उसी समय \omterm{def:b1:water-small-molecules:atp}{ATP} संदमित करता है।
  \item समझाइए कि एक ही \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} में एक साथ चलते \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{ग्लूकोजनन} और
        \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} केवल \omterm{def:b1:water-small-molecules:atp}{ATP} को ऊष्मा में क्यों बदलते, और तीनों
        वैकल्पिक मार्गों के अलग \omterm{def:b1:enzymes:enzyme}{एंजाइम} इसे कैसे रोकते हैं।
  \item इंसुलिन-रहित किसी मधुमेही में रक्त ग्लूकोज़ ऊँचा है और फिर भी
        यकृत ग्लूकोज़ तथा कीटोन काय बनाता रहता है। वाल्वों से यह
        विरोधाभास समझाइए।
  \item परिणाम बताइए: उपवास के वे घंटे जिन्हें यकृत का \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} ढकता
        है (\omterm{def:b1:biosyntheses-integration:gluconeogenesis}{ग्लूकोजनन} के साथ और उसके बिना), कीटोन काय पर जाने से पहले
        और बाद की दैनिक प्रोटीन-हानि, और \omterm{def:b1:lipids:triglyceride}{वसा} भंडार कितने दिन चलता है।
\end{enumerate}
\end{problem}
