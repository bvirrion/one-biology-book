\section*{الفصل \ref{ch:g12:adaptive-immunity} --- المناعة التكيّفية والتلقيح}

\begin{solution}{exo:g12:adaptive-immunity:1}
\omterm{def:g12:adaptive-immunity:antigen}{المستضد} أي جزيء يستطيع الجهاز التكيّفي أن يتعرفه؛ \omterm{def:g12:adaptive-immunity:antigen}{واللمفاوية} \omterm{def:g10:cells-common-unit:cell}{خلية} بيضاء
تحمل مستقبلات \omterm{def:g12:adaptive-immunity:antigen}{لمستضد} واحد. ونوعيتها آتية من مستقبلها، المركَّب عشوائيًا
أثناء النضج والمتطابق على نسخها كلها.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:2}
\omterm{def:g12:adaptive-immunity:antigen}{المستضد} يرتبط \omterm{def:g12:adaptive-immunity:antigen}{باللمفاويات} القليلة التي يوافقه مستقبلها؛ فتتكاثر تلك
\omterm{def:g10:cells-common-unit:cell}{الخلايا} إلى نسيلة على مدى أيام؛ وتتمايز النسيلة إلى \omterm{def:g10:cells-common-unit:cell}{خلايا} منفِّذة
تقاتل؛ وتصير أقلية \omterm{prop:g12:adaptive-immunity:selection}{خلايا ذاكرة} تبقى.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:3}
\omterm{def:g11:gene-expression:protein}{بروتين} على شكل حرف Y تفرزه \omterm{def:g10:cells-common-unit:cell}{الخلايا} البلازمية، وهو نسخة منحلة من مستقبل
\omterm{def:g10:cells-common-unit:cell}{الخلية} البائية؛ وكل ذراع منه ترتبط \omterm{def:g12:adaptive-immunity:antigen}{بالمستضد}. وهو على الجرثومة يكوّن
كسوة تمسكها \omterm{def:g12:innate-immunity:phagocytosis}{البالعات}، ويشابك الجراثيم في معقدات تبتلع.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:4}
\omterm{prop:g12:adaptive-immunity:tcells}{الخلايا التائية المساعدة}، تنشّطها \omterm{def:g10:cells-common-unit:cell}{الخلايا} التغصنية، تفرز السيتوكينات
التي تأذن \omterm{def:g10:cells-common-unit:cell}{للخلايا} البائية ولسائر \omterm{def:g10:cells-common-unit:cell}{الخلايا} التائية وتضخّمها. \omterm{def:g10:cells-common-unit:cell}{والخلايا}
التائية السامة \omterm{def:g10:cells-common-unit:cell}{للخلايا} تقتل \omterm{def:g10:cells-common-unit:cell}{الخلايا} التي تعرض قطعًا من فيروس أو من
\omterm{def:g11:gene-expression:protein}{بروتين} ورمي.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:5}
صورة غير ضارة من \omterm{def:g12:adaptive-immunity:antigen}{مستضد} الممرض (موهَّنة، أو مقتولة، أو بروتينًا، أو
ذيفانًا معطَّلًا، أو تعليماته) مع مساعد. وهو يثير استجابة أولية ويترك
\omterm{prop:g12:adaptive-immunity:selection}{خلايا ذاكرة}، فيلقى الممرض الحقيقي استجابة ثانوية.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:6}
الأولية: تأخر أسبوع، وذروة منخفضة (نحو عشر ذروة الثانوية) عند أسبوعين
إلى ثلاثة، وانحدار في غضون شهر. والثانوية: تأخر يومين أو ثلاثة، وذروة
أعلى بعشرة أمثال في غضون عشرة أيام، تبقى أشهرًا.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:7}
الذاكرة نوعية: \omterm{prop:g12:adaptive-immunity:selection}{فخلايا الذاكرة} التي تركتها الجرعة الأولى من \omterm{def:g12:adaptive-immunity:antigen}{المستضد} A
لا تستجيب إلا له؛ أما B، وقد لقي أول مرة، فينال استجابة أولية.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:8}
\omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} بروتينات في السوائل ولا تستطيع دخول \omterm{def:g10:cells-common-unit:cell}{الخلايا}. \omterm{def:g10:cells-common-unit:cell}{والخلية}
المصابة تعرض قطعًا من الفيروس على سطحها، \omterm{def:g10:cells-common-unit:cell}{والخلايا} التائية السامة
تتعرفها وتقتل \omterm{def:g10:cells-common-unit:cell}{الخلية}.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:10}

\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:12}
سيتوكينات \omterm{prop:g12:adaptive-immunity:tcells}{الخلايا التائية المساعدة} هي التي تأذن \omterm{def:g10:cells-common-unit:cell}{للخلايا} البائية بأن
تصير \omterm{def:g10:cells-common-unit:cell}{خلايا} بلازمية وللخلايا التائية السامة بأن تتكاثر؛ فالذراعان معًا
تتوقفان على المنسّق نفسه، ونزعه يسكتهما معًا.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:13}
\omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} انتقيت في وجه الفيروس كما كان؛ والفيروس يطفر بروتينات
سطحه أسرع مما يمكن أن تنتقى نسائل جديدة، فتسبق كل استجابة. وهو يختبئ
ويتكاثر داخل \omterm{def:g10:cells-common-unit:cell}{الخلايا} المساعدة عينها التي كانت لتنسق الاستجابة.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:14}
\omterm{prop:g12:adaptive-immunity:selection}{خلايا الذاكرة} تتعرف بروتينات السطح التي عرضها \omterm{def:g12:adaptive-immunity:vaccine}{اللقاح}. وبروتينات
الأنفلونزا تطفر كل سنة إلى صور لا ترتبط بها الذاكرة القديمة؛ أما
بروتينات سطح الحصبة فلا تكاد تتغير، فذاكرة سنة 1970 ما زالت توافق
فيروس اليوم.
\end{solution}

\begin{solution}{exo:g12:adaptive-immunity:15}
ما يتنوع: مستقبلات \omterm{def:g12:adaptive-immunity:antigen}{اللمفاويات}، مصنوعةً عشوائيًا. وما ينتقي: \omterm{def:g12:adaptive-immunity:antigen}{المستضد}،
وهو لا يدع إلا النسائل الموافقة تتكاثر. وما يورَّث: المستقبل، يرثه نسل
النسيلة، ومنه \omterm{prop:g12:adaptive-immunity:selection}{خلايا الذاكرة}. والمقياس الزمني: أيام للاستجابة، وعمر
كامل للذاكرة، في مقابل أجيال عند \omterm{def:g12:selection-drift-speciation:population}{جمهرة}. ويقف التشبيه عند النقل:
فالنسائل المنتقاة تموت مع الجسد ولا تنقل إلى الجيل التالي.
\end{solution}

\begin{solution}{pb:g12:adaptive-immunity:1}
\textbf{1.} لا شيء نحو أسبوع، ثم ذروة متواضعة عند أسبوعين إلى ثلاثة،
ثم انحدار على مدى الأسابيع التالية.

\textbf{2.} \omterm{prop:g12:adaptive-immunity:antibodies}{أجسام مضادة} في غضون يومين أو ثلاثة، وذروة أعلى بعشرة
أمثال، تدوم أشهرًا: \omterm{prop:g12:adaptive-immunity:selection}{فخلايا الذاكرة} التي تركتها الجرعة الأولى نسيلة
كبيرة أصلًا، فيبدأ التكاثر من آلاف \omterm{def:g10:cells-common-unit:cell}{الخلايا} بدل قليل منها.

\textbf{3.} الفيروس الحي يتكاثر مدة قصيرة داخل \omterm{def:g10:cells-common-unit:cell}{الخلايا}، فتعرض \omterm{def:g10:cells-common-unit:cell}{الخلايا}
المصابة قطعه وتنتقى \omterm{def:g10:cells-common-unit:cell}{الخلايا} التائية السامة كما تنتقى \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة}؛
أما الفيروس المقتول فيدرّب ذراع \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} غالبًا.

\textbf{4.} \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} الموجودة أصلًا تسد معظم الفيروس؛ \omterm{prop:g12:adaptive-immunity:selection}{وخلايا
الذاكرة} البائية والتائية تتكاثر في غضون يومين؛ وتقتل \omterm{def:g10:cells-common-unit:cell}{الخلايا} المصابة
قبل أن ينتشر الفيروس. فتوقف العدوى قبل الأعراض.

\textbf{5.} ثلاثة أيام من تكاثر صامت، ثم حمى وطفح ومرض بينما تبنى
الاستجابة الأولية على مدى أسبوع؛ وتعاف حول اليوم العاشر، مع \omterm{prop:g12:adaptive-immunity:selection}{خلايا
ذاكرة} ومناعة مدى العمر --- إن نجا من المضاعفات.

\textbf{6.} $1 - 1/15 \approx 93\%$.

\textbf{7.} $15 \times 0.10 = 1.5$: أي إن كل حالة تعدي أكثر من واحد
غيرها، فيستطيع المرض أن ينتشر. فالبلدة غير محمية.

\textbf{8.} $10\% \times \num{50000} = 5000$ غير ممنَّعين، منهم 1500
من الرضّع.

\textbf{9.} 2500 قابل للإصابة؛ و$15 \times 0.05 = 0.75$: أي أقل من
واحد، فينقرض المرض.

\textbf{10.} الرضّع لا يمكن تلقيحهم، فحمايتهم الوحيدة أن البالغين
حولهم، إذ كانوا ممنَّعين، لا يتركون للفيروس سبيلًا يبلغهم به.

\textbf{11.} الجيل الثالث: $1.5^3 \approx 3.4$ حالة؛ والجيل السادس:
$1.5^6 \approx 11$.

\textbf{12.} $1 + 1.5 + 2.25 + 3.4 + 5.1 + 7.6 + 11.4 \approx 32$ حالة.

\textbf{13.} نحو 30\% من 32: أي نحو 10 من الرضّع.

\textbf{14.} كل حالة تنزع شخصًا قابلًا للإصابة وتضيف شخصًا ممنَّعًا،
فيهبط كسر القابلين للإصابة ويهبط معه عدد من تعديه كل حالة؛ ومع 5000
قابل للإصابة لا تبطئ الفاشية إلا بعد مئات الحالات، على مدى أشهر ---
ما لم يتدخل التلقيح.

\textbf{15.} عند 95\%: $0.75^3 \approx 0.4$ و$0.75^6 \approx 0.18$
حالة؛ والمجموع نحو $1 + 0.75 + 0.56 + \dots \approx 4$ حالات. أي عشر
الحالات، والفاشية تنطفئ من تلقاء نفسها.

\textbf{16.} 800 في ثماني سنين: أي 100 في الميكرولتر في السنة، ونحو 8
في الشهر.

\textbf{17.} \omterm{def:g10:cells-common-unit:cell}{الخلايا} البائية تحتاج سيتوكينات \omterm{def:g10:cells-common-unit:cell}{الخلايا} المساعدة لتصير
\omterm{def:g10:cells-common-unit:cell}{خلايا} بلازمية، \omterm{def:g10:cells-common-unit:cell}{والخلايا} التائية السامة تحتاجها لتتكاثر: فمع قلة
المساعدات، لا يؤذن لأي من الذراعين، وإن كانت خلاياهما موجودة.

\textbf{18.} نعم: \omterm{prop:g12:adaptive-immunity:selection}{فخلايا الذاكرة} ما زالت تحتاج سيتوكينات المساعدات
لتتكاثر إلى استجابة ثانوية؛ ومن غير تنسيق لا يمكن استعمال الذاكرة،
والحصبة، وهي غير ضارة لبالغ سليم، قد تكون مميتة.

\textbf{19.} \omterm{def:g12:selection-drift-speciation:population}{جمهرة} المساعدات تنمو من جديد من \omterm{def:g10:cells-common-unit:cell}{الخلايا} الناجية ومن
النقي، \omterm{prop:g12:adaptive-immunity:selection}{وخلايا ذاكرة} الذراعين الأخريين لم تخرَّب قط --- فالذخيرة سليمة
متى عاد التنسيق. والفيروس يبقى في \omterm{def:g10:cells-common-unit:cell}{خلايا} مختبئة، فإيقاف الأدوية يدعه
يستأنف.

\textbf{20.} تغطية 93\%؛ ونحو 32 حالة من مسافر واحد عند 90\%؛ \omterm{prop:g12:adaptive-immunity:tcells}{والخلية
التائية المساعدة}.
\end{solution}
