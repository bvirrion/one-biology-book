\chapter{सहज प्रतिरक्षा}\label{ch:g12:innate-immunity}

कोई किरच किसी उँगली की नोक में घुसती है और मिनट भर में खींच निकाली जाती है।
शाम तक वह जगह लाल, गरम, सूजी और टीसती हुई है; अगले दिन तक त्वचा के नीचे पीले
पीब की एक बूँद बन चुकी है; और तीसरे दिन तक वह जा चुकी है। कुछ तय नहीं किया
गया, कुछ सीखा नहीं गया: वही चार चिह्न हर घाव में, हर व्यक्ति में, हर
स्तनधारी में प्रकट होते हैं, और तब भी होते थे जब किसी को मालूम नहीं था कि
कोई किरच जीवाणु ढोती है। वे शरीर की बचाव की पहली क़तार की दिखती सतह हैं ---
यानी \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} तथा अणुओं का एक समुच्चय जो किसी घुसपैठिए को मिनटों के भीतर
पहचान लेता है और उससे कभी मिले बिना ही उस पर हमला कर देता है। यह अध्याय उसी
क़तार का वर्णन करता है; और अगला उस धीमे, सीखते बचाव का जिसे वह बुलाती है।

\section{दो प्रतिरक्षाएँ}

\begin{definition}[सहज और अनुकूली प्रतिरक्षा]\label{def:g12:innate-immunity:immunity}
\emph{प्रतिरक्षा तंत्र}\index{प्रतिरक्षा तंत्र} उन अंगों, \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} तथा
अणुओं का समुच्चय है जो शरीर का संक्रमण और नुक़सान से बचाव करते हैं। उसके दो
हिस्से हैं। \emph{सहज प्रतिरक्षा}\index{सहज प्रतिरक्षा} जन्म से मौजूद है,
मिनटों से घंटों के भीतर काम करती है, घुसपैठियों के चौड़े वर्ग --- जीवाणु,
विषाणु, कवक, क्षतिग्रस्त \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} --- उन लक्षणों से पहचानती है जो वे साझा
करते हैं, और हर बार वही जवाब देती है।
\emph{अनुकूली प्रतिरक्षा}\index{अनुकूली प्रतिरक्षा} को काम करने में दिन
लगते हैं, वह हर घुसपैठिए को अलग-अलग पहचानती है, और उसे याद रखती है
(\cref{ch:g12:adaptive-immunity})। सहज तंत्र सारे जंतुओं में साझा है; और
अनुकूली केवल \omterm{def:g10:common-ancestry:bodyplan}{कशेरुकियों} में है तथा छिड़ने के लिए सहज पर निर्भर है।
\end{definition}

\begin{proposition}[पहले रुकावटें]\label{prop:g12:innate-immunity:barriers}
अधिकांश घुसपैठिए भीतर घुसते ही नहीं। त्वचा, जो सूखी तथा अम्लीय है और अपनी
सतह लगातार झाड़ती रहती है, लगभग सब कुछ रोक देती है; वायुमार्गों, आँत तथा
जनन मार्ग की परतें गीली हैं पर ऐसे श्लेष्म से लिपटी हैं जो रोगाणु फँसा लेता
है और बहा दिया जाता या नया कर दिया जाता है; आँसू, लार तथा श्लेष्म ऐसा
\omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} ढोते हैं जो जीवाणु दीवारें चीर देता है; और आमाशय का अम्ल तथा आँत के
अपने जीवाणु, जो वह जगह घेरे बैठे हैं, नए आने वालों के लिए कम ही जगह छोड़ते
हैं। इस अध्याय की प्रतिरक्षा प्रतिक्रिया तब शुरू होती है जब कोई रुकावट टूटे।
\end{proposition}
\admitted

\section{शोथ प्रतिक्रिया}

\begin{proposition}[वे चार चिह्न और उन्हें बनाने वाली चीज़ें]\label{prop:g12:innate-immunity:inflammation}
कोई सेंध --- कोई घाव, कोई संक्रमण, कोई जलन --- मिनटों के भीतर
\emph{शोथ प्रतिक्रिया}\index{शोथ} छेड़ती है: लाली, गरमी, सूजन तथा दर्द। हर
एक का एक कारण है:
\begin{itemize}
  \item उस ऊतक में बसी \emph{प्रहरी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}} --- महाभक्षक, \omterm{prop:g12:innate-immunity:handover}{वृक्षाभ
        कोशिकाएँ}, मास्ट \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} --- ऐसे ग्राही ढोती हैं जो रोगाणुओं के
        पूरे-पूरे वर्गों में आम अणु (जीवाणु दीवारों के टुकड़े, विषाणु
        न्यूक्लीक अम्ल) और क्षतिग्रस्त \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के छोड़े अणु पहचान लेते
        हैं। पहचान पर वे \emph{मध्यस्थ}\index{शोथ मध्यस्थ} छोड़ती हैं:
        हिस्टामिन, प्रोस्टाग्लैंडिन, और साइटोकाइन कहलाते संकेत \omterm{def:g11:gene-expression:protein}{प्रोटीन}।
  \item वे \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थ} स्थानीय रक्त वाहिकाएँ चौड़ी करते हैं और उनकी दीवारें
        रिसने लायक़ बना देते हैं: यानी अधिक रक्त (लाली, गरमी), और ऊतक में
        रिसता प्लाज़्मा (सूजन)। वे तंत्रिका सिरों को भी संवेदनशील बनाते
        हैं (दर्द), और वाहिका दीवारों को गुज़रती श्वेत \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के लिए
        चिपचिपा।
  \item \emph{\omterm{def:g12:innate-immunity:phagocytosis}{भक्षककोशिकाएँ}} --- घंटों के भीतर रक्त से आते न्यूट्रोफिल, और
        दिन भर में महाभक्षक बन जाते मोनोसाइट --- वाहिका दीवारों से निचुड़कर
        निकलती हैं, \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थों} की प्रवणता पर चढ़कर उस जगह तक रेंगती हैं, और
        रोगाणुओं तथा मलबे को निगलकर पचा देती हैं। मरी \omterm{def:g12:innate-immunity:phagocytosis}{भक्षककोशिकाएँ}, जीवाणु
        और तरल ही वह पीब हैं।
\end{itemize}
यह प्रतिक्रिया स्थानीय, तेज़, और किसी किरच के लिए वैसी ही है जैसी किसी जीवाणु
के लिए: इसे उस घुसपैठिए से पहले किसी संपर्क की ज़रूरत नहीं।
\end{proposition}
\begin{proof}[साक्ष्य]
बिना किसी जीवित जीवाणु के, अकेला जीवाणु दीवार का कोई टुकड़ा त्वचा में लगाने
पर घंटे भर के भीतर पूरी प्रतिक्रिया बन जाती है: यानी वह संकेत कोई आणविक
नमूना है, वह रोगाणु नहीं। हिस्टामिन रोकने पर (परागज ज्वर की
प्रतिहिस्टामिन दवाओं से) शुरुआती लाली तथा सूजन मिट जाती है; और
प्रोस्टाग्लैंडिन बनाता \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} रोकने पर (ऐस्पिरिन या आइबुप्रोफ़ेन से) सूजन,
दर्द तथा बुख़ार घट जाते हैं। किसी घाव को सूक्ष्मदर्शी के नीचे देखने पर
न्यूट्रोफिल वाहिका दीवार पर लुढ़कते, रुकते, उसे पार करते और दो घंटों के भीतर
उस जगह पर झुंड बनाते दिखते हैं; और जिन चूहों के न्यूट्रोफिल घटा दिए गए हों
वे उन संक्रमणों से मर जाते हैं जिन्हें कोई सामान्य चूहा रात भर में साफ़ कर
देता है।
\end{proof}

\begin{definition}[भक्षकाणुक्रिया]\label{def:g12:innate-immunity:phagocytosis}
\emph{भक्षकाणुक्रिया}\index{भक्षकाणुक्रिया} किसी कण --- किसी जीवाणु, किसी
मरी \omterm{def:g10:cells-common-unit:cell}{कोशिका}, धूल के किसी कण --- का ऐसी \omterm{def:g10:cells-common-unit:cell}{कोशिका} द्वारा निगला जाना है जो अपनी
झिल्ली उसके चारों ओर लपेट लेती है, उसे किसी पुटिका में भीतर खींच लेती है,
और एंजाइमों तथा क्रियाशील रसायनों से पचा देती है। सहज प्रतिक्रिया की
\emph{भक्षककोशिकाएँ}\index{भक्षककोशिका} न्यूट्रोफिल हैं, जो अल्पजीवी तथा
भरपूर हैं, और महाभक्षक, जो दीर्घजीवी तथा वहीं बसे होते हैं। जिस महाभक्षक ने
किसी रोगाणु को पचाया हो वह उसके प्रोटीनों के टुकड़े रख लेता है और उन्हें अपनी
सतह पर दिखाता है --- यानी वह क़दम जो इस प्रतिक्रिया को अगले अध्याय वाली से
जोड़ेगा।
\end{definition}

\begin{example}[किसी किरच की समय-सारणी]\label{ex:g12:innate-immunity:timetable}
मिनट 0: उस किरच के जीवाणु ऊतक में। मिनट 1--10: मास्ट \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} तथा महाभक्षक
उनकी दीवारों के टुकड़े पहचानते हैं और हिस्टामिन तथा साइटोकाइन छोड़ते हैं;
वाहिकाएँ चौड़ी होती हैं। घंटा 1: लाली, गरमाहट, और वाहिका दीवारें पार करते
पहले न्यूट्रोफिल। घंटे 2--12: न्यूट्रोफिल झुंड बनाते, जीवाणु निगलते, मरते
हैं; सूजन तथा दर्द अपनी चोटी पर। दिन 1: मोनोसाइट आते हैं और महाभक्षक बन
जाते हैं, तथा मलबा साफ़ करते हैं; पीब दिखने लगती है। दिन 2--3: जीवाणु जा
चुके, \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थ} अब नहीं छूटते, वाहिकाएँ सामान्य पर लौटती हैं, और मरम्मत शुरू
होती है। यदि जीवाणु उतनी तेज़ी से बढ़ें कि \omterm{def:g12:innate-immunity:phagocytosis}{भक्षककोशिकाएँ} उन्हें साफ़ न कर
पाएँ, तो वह प्रतिक्रिया चौड़ी हो जाती है, बुख़ार प्रकट होता है, और अनुकूली
प्रतिक्रिया --- जो लसीका ग्रंथि में पहले ही शुरू हो चुकी है --- सँभाल लेती है।
\end{example}

\section{प्रतिक्रिया को शांत करना}

\begin{proposition}[शोथरोधी दवाएँ]\label{prop:g12:innate-immunity:drugs}
\omterm{prop:g12:innate-immunity:inflammation}{शोथ प्रतिक्रिया} उपयोगी भी है और महँगी भी: उसकी सूजन तथा दर्द उस घुसपैठिए को
साफ़ करने की क़ीमत हैं, और वह हद से ज़्यादा हो, ग़लत दिशा में मुड़ी हो
(किसी हानिरहित पराग के, या शरीर के अपने ही जोड़ों के ख़िलाफ़) या लंबी खिंच
जाए, तो वह अपना ही नुक़सान करती है। \emph{शोथरोधी
दवाएँ}\index{शोथरोधी दवा} उसके \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थों} पर काम करती हैं: ऐस्पिरिन,
आइबुप्रोफ़ेन और उनके नातेदार प्रोस्टाग्लैंडिन बनाता \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} रोक देते हैं,
जिससे दर्द, सूजन तथा बुख़ार घटते हैं; और कॉर्टिकोस्टेरॉइड, जो अधिवृक्क
ग्रंथि के किसी \omterm{def:g11:hormones-and-reproduction:hormone}{हॉर्मोन} से निकले हैं, अधिकांश \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थों} का उत्पादन बंद कर
देते हैं तथा गंभीर या चिरकालिक शोथ के लिए इस्तेमाल होते हैं। ये सब उन
चिह्नों को घटाते हैं; इनमें से कोई भी उस कारण को नहीं हटाता, और उस
प्रतिक्रिया को दबाकर वे किसी असली संक्रमण की सफ़ाई धीमी कर सकते हैं।
\end{proposition}
\admitted

\end{example}

\section{सहज से अनुकूली तक}

\begin{proposition}[वह सौंपाई]\label{prop:g12:innate-immunity:handover}
प्रहरी \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में \emph{वृक्षाभ कोशिकाओं}\index{वृक्षाभ कोशिका}
का एक दूसरा काम भी है। उस घुसपैठिए को निगल लेने के बाद वे लसीका वाहिकाओं से
होकर वह ऊतक छोड़ती हैं, और उसके \omterm{def:g11:gene-expression:protein}{प्रोटीन} टुकड़े अपनी सतह पर दिखाती हुई सबसे
पास की \emph{लसीका ग्रंथि} तक जाती हैं। वहाँ वे उन टुकड़ों को अनुकूली तंत्र
के लसीकाणुओं के सामने पेश करती हैं, और उनके ढोए साइटोकाइन उन लसीकाणुओं को
बताते हैं कि वह किस क़िस्म का घुसपैठिया था। इस तरह सहज प्रतिक्रिया केवल पहले
दिन ही नहीं लड़ती; वह उस दुश्मन को पहचानती भी है और उसका ब्योरा पहुँचाती
है। उस सौंपाई के बिना कोई अनुकूली प्रतिक्रिया शुरू ही नहीं होती --- और
इसीलिए किसी टीके में उस प्रतिजन के अलावा ऐसे पदार्थ भी होते हैं जो थोड़ा
शोथ छेड़ें।
\end{proposition}

\begin{method}[किसी शोथ को पढ़ना]\label{met:g12:innate-immunity:read}
\begin{enumerate}
  \item उस सेंध और संभावित घुसपैठिए का नाम लीजिए (या बिना किसी घुसपैठिए के
        वह नुक़सान: किसी मोच में भी शोथ होता है)।
  \item हर चिह्न को उसके \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थ} तथा वाहिका बदलाव से जोड़िए: लाली और गरमी
        चौड़ी वाहिकाओं से, सूजन रिसती वाहिकाओं से, और दर्द संवेदनशील
        तंत्रिकाओं से।
  \item हर चरण पर मौजूद \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} पहचानिए: पहले प्रहरी, घंटों तक
        न्यूट्रोफिल, दिन भर में महाभक्षक; और पीब उनके अवशेष की तरह।
  \item परखिए कि वह प्रतिक्रिया संतुलित है या नहीं: स्थानीय और दिनों में
        सुलझती (यानी इरादे के मुताबिक़ काम करती), बुख़ार के साथ फैलती
        (यानी संक्रमण जीत रहा है, अनुकूली प्रतिक्रिया चाहिए), या किसी
        हानिरहित चीज़ पर टिकी अथवा चिरकालिक (यानी शोथरोधी इलाज का मामला)।
\end{enumerate}
\end{method}

\begin{remark}[पुरानी, तेज़ और भोथरी]\label{rem:g12:innate-immunity:old}
\omterm{def:g12:innate-immunity:immunity}{सहज प्रतिरक्षा} जितनी किसी मनुष्य की है उतनी ही किसी स्पंज, किसी कीट और किसी
बलूत की भी: नमूना ग्राही तथा \omterm{def:g12:innate-immunity:phagocytosis}{भक्षककोशिकाएँ} \omterm{def:g10:common-ancestry:bodyplan}{कशेरुकियों} से करोड़ों साल पुराने
हैं। वह इसलिए तेज़ है कि वह दुश्मन के वर्गों के लिए पहले से तैयार है, और उसी
वजह से भोथरी भी: वह एक जीवाणु को दूसरे से नहीं बता सकती, और वह सुधर नहीं
सकती। अगले अध्याय का तंत्र दोनों कर सकता है --- पर तभी जब यह वाला ख़तरे की
घंटी बजा चुका हो।
\end{remark}

दो-प्रतिक्रिया वाले चित्र से बताइए कि सहज प्रतिक्रिया की चोटी कब आती है,

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:7}
अकेला लगाया गया जीवाणु दीवार का कोई टुकड़ा पूरी \omterm{prop:g12:innate-immunity:inflammation}{शोथ प्रतिक्रिया} पैदा करता
है। यह क्या दिखाता है, और बाँझ नमकीन पानी के किसी इंजेक्शन पर क्या होता?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:8}
पीब किससे बनी होती है? वह उसी वक़्त के बजाय घाव के एक दिन बाद क्यों प्रकट
होती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:9}
समझाइए कि कोई प्रतिहिस्टामिन परागज ज्वर में राहत क्यों देता है पर किसी
जीवाणु संक्रमण के बाद के चरणों पर कुछ नहीं करता।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:10}
सूजन वाले चित्र से 12 और 36 घंटे पर सूजन की, दवा के साथ और उसके बिना, तुलना
कीजिए। क्या वह दवा उस प्रतिक्रिया को छोटा करती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:innate-immunity:11}
मुड़ा हुआ कोई टख़ना, जहाँ न कोई घाव है न कोई रोगाणु, लाल, गरम, सूजा और
दर्दभरा हो जाता है। समझाइए कि प्रहरी \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} को किसने छेड़ा।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:innate-immunity:12}
जिन चूहों में न्यूट्रोफिल न हों वे उन संक्रमणों से मर जाते हैं जिन्हें
सामान्य चूहे रात भर में साफ़ कर देते हैं; और जिनमें \omterm{def:g12:innate-immunity:immunity}{अनुकूली प्रतिरक्षा} न हो
वे उनसे बच जाते हैं पर कुछ हफ़्ते बाद संक्रमणों से मर जाते हैं। समझाइए कि
हर नतीजा दोनों तंत्रों की भूमिका तथा समय के बारे में क्या कहता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:innate-immunity:13}
समझाइए कि कोई डॉक्टर किसी फोड़े वाले रोगी को \omterm{prop:g12:innate-immunity:drugs}{शोथरोधी दवा} देने से क्यों मना

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:innate-immunity:14}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:innate-immunity:15}

\end{exercise}

\section{समस्या: किसी किरच के तीन दिन}

\begin{problem}[{सप्ताहांत समस्या --- कोई घाव घंटा दर घंटा पीछा किया गया:
कोशिकाएँ गिनी गईं, मध्यस्थों का पीछा हुआ, दवाएँ आज़माई गईं, और लसीका
ग्रंथि तक की सौंपाई का समय नापा गया}]
\label{pb:g12:innate-immunity:1}
कोई किरच किसी उँगली की नोक में लगभग $\num{10000}$ जीवाणु ले जाती है; वे हर
40 मिनट पर विभाजित होते हैं। उस घाव के एक मिलीमीटर के भीतर वह ऊतक 200 बसे
हुए महाभक्षक थामे है; और न्यूट्रोफिल 1 घंटे बाद रक्त से प्रति घंटा
$\num{50000}$ की दर से आने लगते हैं, जिनमें से हर एक मरने से पहले लगभग 20
जीवाणु निगल सकता है।

\medskip
\noindent\textbf{भाग I --- वह दौड़।}
\begin{enumerate}
  \item उन्हें कोई न मारे, तो 2 घंटे बाद, फिर 4 बाद, कितने जीवाणु होते?
  \item वे 200 महाभक्षक हर एक घंटे में लगभग 5 जीवाणु निगलते हैं। वे पहले
        घंटे में कितने हटाते हैं, और क्या इससे वह बढ़त रुकती है?
  \item घंटा 1 से न्यूट्रोफिल आते हैं। पहले घंटे के न्यूट्रोफिल कुल कितने
        जीवाणु हटा सकते हैं?
  \item घंटा 2 पर जीवाणु \omterm{def:g12:selection-drift-speciation:population}{आबादी} से तुलना कीजिए। कौन जीत रहा है?
  \item मोटे तौर पर किस घंटे तक वह घाव साफ़ हो जाता है? उस जगह पर क्या बचता
        है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वे चिह्न।}
\begin{enumerate}[resume]
  \item लाली, गरमी, सूजन और दर्द में से हर एक को किसी \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थ} से और
        वाहिकाओं अथवा तंत्रिकाओं के किसी बदलाव से जोड़िए।
  \item समझाइए कि वह सूजन न्यूट्रोफिलों की मदद क्यों करती है, और वह दर्द
        क्यों देती है।
  \item वह उँगली की नोक हर धड़कन के साथ टीसती है। चौड़ी वाहिकाओं और सूजे ऊतक
        के दाब से समझाइए।
  \item एक दिन बाद पीब की एक बूँद बन चुकी है। उसकी सामग्री गिनाइए।
  \item दिन 3 पर वे चिह्न जा चुके हैं। \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थों} का छूटना किसने रोका?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- दवाएँ।}
\begin{enumerate}[resume]
  \item घंटा 1 पर कोई प्रतिहिस्टामिन लिया जाता है। वह कौन-से चिह्न घटाता
        है, और कौन-से नहीं?
  \item घंटा 6 पर आइबुप्रोफ़ेन लिया जाता है। वह कौन-सा \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थ} रोकता है,
        और कौन-से चिह्न घटाता है?
  \item कोई कॉर्टिकोस्टेरॉइड मरहम लगाया जाता है। न्यूट्रोफिलों की भर्ती का
        क्या होता है, और उस संक्रमण के लिए इससे कौन-सा जोखिम आता है?
  \item समझाइए कि इन तीनों में से कोई दवा उस संक्रमण को छोटा क्यों नहीं
        करती, और कौन-सी उसे लंबा कर सकती है।
  \item उस रोगी को \qty{38.5}{\celsius} का बुख़ार भी है। वह किन अणुओं ने
        पैदा किया, और वह किसलिए है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वह रिपोर्ट।}
\begin{enumerate}[resume]
  \item घंटा 2 पर जीवाणु निगलती \omterm{prop:g12:innate-immunity:handover}{वृक्षाभ कोशिकाएँ} घंटा 20 पर कोहनी की लसीका
        ग्रंथि तक पहुँचती हैं, और जीवाणु टुकड़े दिखाती हैं। वे उन टुकड़ों
        के अलावा क्या ढोती हैं, और किसे रिपोर्ट करती हैं?
  \item दो-प्रतिक्रिया वाले चित्र से बताइए कि इन जीवाणुओं के ख़िलाफ़
        प्रतिरक्षी भरपूर होने से पहले कितने दिन बीतेंगे? क्या वे इस किरच के
        लिए चाहिए होंगे?
  \item वही जीवाणु एक महीने बाद किसी दूसरी किरच से घुसते हैं। दूसरी बार
        दोनों प्रतिक्रियाओं में से कौन-सी अलग है, और कौन-सी वही है?
  \item काम करते न्यूट्रोफिलों के बिना जन्मा कोई व्यक्ति रोज़ाना
        \omterm{def:g11:antibiotic-resistance:antibiotic}{प्रतिजैविकों} से बचा रहता है। समझाइए कि वे \omterm{def:g11:antibiotic-resistance:antibiotic}{प्रतिजैविक} उस बचाव के
        किस हिस्से की जगह लेते हैं और किसकी नहीं ले सकते।
  \item परिणाम लिखिए: वह घंटा जिस पर \omterm{def:g12:innate-immunity:phagocytosis}{भक्षककोशिकाएँ} जीवाणुओं से आगे निकलती
        हैं, चारों चिह्नों में से हर एक के पीछे का \omterm{prop:g12:innate-immunity:inflammation}{मध्यस्थ}, और वह \omterm{def:g10:cells-common-unit:cell}{कोशिका} जो
        लसीका ग्रंथि तक रिपोर्ट ले जाती है।
\end{enumerate}
\end{problem}
