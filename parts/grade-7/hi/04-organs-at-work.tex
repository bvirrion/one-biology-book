\chapter{काम पर लगे अंग}\label{ch:g7:organs-at-work}

सीढ़ियों की तीन मंज़िलें दौड़कर चढ़ो और हिसाब लगाओ: \omterm{def:g1:my-body:parts}{टाँगें} जलती हुईं,
\omterm{def:g4:heart-and-blood:heart}{दिल} हथौड़े की तरह पीटता हुआ, साँस दौड़ती हुई --- और अजीब बात, \omterm{def:g1:the-five-senses:sense}{त्वचा} लाल
और गरम। ये बदलाव किसी ने आदेश देकर नहीं कराए; इन्होंने ख़ुद अपना हुक्म
दिया, और वह भी बिलकुल ठीक अनुपात में। यह अध्याय नापता है कि काम करता
हुआ कोई अंग असल में क्या-क्या ख़र्च करता है, और काम शुरू होते ही शरीर
अपनी आपूर्ति को कैसे नए सिरे से बाँट देता है।

\section{काम करती पेशी क्या ख़र्च करती है}

\begin{proposition}[पेशी का बिल]\label{prop:g7:organs-at-work:bill}
काम करती \omterm{def:g3:bones-and-muscles:muscle}{पेशी} ख़र्च करती है, और उसका ख़र्च नापा जा सकता है। आराम करती
उसी \omterm{def:g3:bones-and-muscles:muscle}{पेशी} के मुक़ाबले, कड़ी मेहनत करती \omterm{def:g3:bones-and-muscles:muscle}{पेशी}:
\begin{itemize}
  \item अपने ऊपर से गुज़रते \omterm{prop:g4:journey-of-food:absorb}{रक्त} से कहीं ज़्यादा \emph{\omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन}} लेती है
        --- आराम वाली खींच से कई गुना;
  \item कहीं ज़्यादा \emph{ग्लूकोज़}\index{ग्लूकोज़} लेती है --- यानी
        \omterm{prop:g4:journey-of-food:absorb}{रक्त} का तैयार शर्करा-पोषक, उसका रोज़मर्रा का ईंधन;
  \item और उसी हिसाब से ज़्यादा \emph{\omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड}} --- तथा
        \emph{ऊष्मा} --- छोड़ती है।
\end{itemize}
यह \cref{prop:g7:what-is-respiration:power} वाला \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} का जलना ही
है, जो किसी अंग के मीटरों पर पढ़ा गया है: ईंधन और \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} भीतर; काम,
कचरा-गैस और गरमाहट बाहर।
\end{proposition}
\admitted

\begin{example}[पेशी के दोनों सिरों पर नापना]\label{ex:g7:organs-at-work:measure}
शरीर-क्रिया वैज्ञानिक \omterm{def:g3:bones-and-muscles:muscle}{पेशी} में \emph{घुसते} \omterm{prop:g4:journey-of-food:absorb}{रक्त} की तुलना उसमें से
\emph{निकलते} \omterm{prop:g4:journey-of-food:absorb}{रक्त} से करते हैं। आराम में निकलता \omterm{prop:g4:journey-of-food:absorb}{रक्त} अपनी \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} और
\omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़} का एक मामूली हिस्सा ही छोड़ आया होता है। कड़ी मेहनत में वही
तुलना दिखाती है कि निकलता \omterm{prop:g4:journey-of-food:absorb}{रक्त} छिन चुका है --- उसकी पहुँचाने लायक़
ज़्यादातर \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} ले ली गई है --- जबकि \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड} उछल पड़ी है।
वही अंग, वही मीटर: बदला सिर्फ़ काम, और बिल उसके पीछे-पीछे चला।
\end{example}

\begin{remark}[गरमाहट ही रसीद है]\label{rem:g7:organs-at-work:warmth}
काम करते शरीर की ऊष्मा कोई इत्तिफ़ाक़ नहीं: ईंधन जलने से हमेशा गरमाहट
निकलती है, और \omterm{def:g3:bones-and-muscles:muscle}{पेशी} शरीर की भट्ठी-कोठरी है। कँपकँपी वही तरकीब है, जो
जान-बूझकर चलाई जाती है --- \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} सिर्फ़ और सिर्फ़ गरमाहट के लिए काम पर
लगा दी जाती हैं (\cref{ex:g3:balanced-diet:rest} वाला ठहरा हुआ ख़र्च,
थोड़ी देर के लिए दोगुना)। दौड़ के बाद की लाल \omterm{def:g1:the-five-senses:sense}{त्वचा} वह ठंडक-तंत्र है जो
भट्ठी की ऊष्मा बाहर ले जाता है --- और यही अगले खंड का काम है।
\end{remark}

\section{माँग पर आपूर्ति}

\begin{proposition}[शरीर अपनी आपूर्ति नए सिरे से बाँटता है]\label{prop:g7:organs-at-work:supply}
काम शुरू होते ही आपूर्ति की लकीरें बिल चुकाने के लिए नए सिरे से बँट
जाती हैं (\cref{ex:g4:heart-and-blood:effort} ने इसकी घोषणा कर दी थी;
यहाँ पूरा फ़रमान है):
\begin{enumerate}
  \item \emph{साँस} गहरी और तेज़ होती है --- फेफड़ों पर \omterm{prop:g4:journey-of-food:absorb}{रक्त} को और तेज़ी
        से फिर से लादती हुई (\cref{ex:g4:breathing:running});
  \item \emph{\omterm{def:g4:heart-and-blood:heart}{दिल}} तेज़ और ज़ोर से धड़कता है --- यानी हर मिनट ज़्यादा
        चक्कर (\cref{prop:g4:heart-and-blood:pulse});
  \item और \omterm{prop:g4:journey-of-food:absorb}{रक्त} का रुख़ \emph{मोड़ा} जाता है: काम करती \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} (और
        ठंडक करती \omterm{def:g1:the-five-senses:sense}{त्वचा}) की सेवा में लगी \omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ} चौड़ी हो जाती हैं,
        जबकि आराम करती पाचन-नली की सिकुड़ जाती हैं --- वही \omterm{prop:g4:journey-of-food:absorb}{रक्त}, उस
        मंज़िल की ओर मोड़ दिया गया जो ख़र्च कर रही है।
\end{enumerate}
तीनों ही काम के पीछे अपने आप चलते हैं, और उन्हें \omterm{def:g5:brain-in-command:system}{मस्तिष्क} की बिन-माँगी
सेवाएँ हाँकती हैं (\cref{rem:g5:brain-in-command:more})।
\end{proposition}
\admitted

\begin{example}[फ़रमान, भीतर से महसूस किया हुआ]\label{ex:g7:organs-at-work:felt}
सीढ़ी वाली दौड़ का हिसाब, एक-एक लकीर करके समझाया हुआ: दौड़ती साँस ---
फिर से लदाई; हथौड़े जैसा \omterm{def:g4:heart-and-blood:heart}{दिल} --- तेज़ चक्कर; जलती \omterm{def:g1:my-body:parts}{टाँगें} --- \omterm{def:g3:bones-and-muscles:muscle}{पेशियों} का
बिल हद पर चुकाया जाता हुआ; लाल गरम \omterm{def:g1:the-five-senses:sense}{त्वचा} --- भट्ठी की ऊष्मा ठंडक वाली
सतह तक ले जाई हुई। \omterm{def:g3:bones-and-muscles:skeleton}{पसली} के नीचे उठती वह जानी-पहचानी चुभन और भारी खाने के
बाद की सुस्ती भी इसी में बैठ जाती हैं: \omterm{def:g4:journey-of-food:digestion}{पाचन} और दौड़, दोनों उसी मोड़े गए
\omterm{prop:g4:journey-of-food:absorb}{रक्त} के लिए होड़ करते हैं --- इसीलिए
\cref{met:g4:journey-of-food:habits} ने भरे पेट दौड़ लगाने से मना किया
था।
\end{example}

\begin{definition}[बहाली]\label{def:g7:organs-at-work:recovery}
मेहनत रुक जाए तो भी \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} और साँस नहीं रुकते: वे धीरे-धीरे ही नीचे
उतरते हैं --- यही \emph{बहाली}\index{बहाली} का दौर है। शरीर उधार चुका
रहा होता है: अपने \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} के भंडार फिर से भरता हुआ, मेहनत का कचरा साफ़
करता हुआ, और जमा हुई ऊष्मा उतारता हुआ। शरीर जितना चुस्त, उसकी बहाली उतनी
तेज़ --- और इसीलिए बहाली का समय ख़ुद एक चुस्ती-माप है।
\end{definition}

\section{प्रशिक्षण, और मशीन की हदें}

\begin{proposition}[प्रशिक्षण छत ऊपर उठाता है]\label{prop:g7:organs-at-work:training}
नियमित कसरत हो तो आपूर्ति की पूरी शृंखला मज़बूत होती है
(\cref{rem:g4:heart-and-blood:rest}, \cref{met:g4:breathing:care}):
\omterm{def:g4:heart-and-blood:heart}{दिल} हर धड़कन पर ज़्यादा \omterm{prop:g4:journey-of-food:absorb}{रक्त} चलाता है, साँस की \omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} गहरी होती हैं,
\omterm{def:g3:bones-and-muscles:muscle}{पेशियाँ} ख़ुद बढ़ती हैं और ज़्यादा ईंधन जमा करती हैं, और उनकी महीन
\omterm{def:g4:heart-and-blood:vessels}{वाहिकाएँ} बढ़ जाती हैं। नतीजा: वही मेहनत प्रशिक्षित शरीर को कम \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} पर
पड़ती है, और उसकी छत --- यानी सबसे कड़ी मेहनत जिसका बिल वह चुका सके ---
ऊपर उठ जाती है। मशीन अपने ही इस्तेमाल से नए सिरे से बनती है।
\end{proposition}
\admitted

\begin{example}[दो दौड़ने वाले, एक ही चढ़ाई]\label{ex:g7:organs-at-work:runners}
बिना प्रशिक्षण वाला और प्रशिक्षित, दोनों वही चढ़ाई दौड़ते हैं। बिना
प्रशिक्षण वाला: आधी चढ़ाई पर \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} $180$, हाँफता हुआ, चलने पर मजबूर ---
आपूर्ति की लकीरें बिल नहीं चुका पातीं, और उधार रुकवा देता है।
प्रशिक्षित: चोटी पर \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} $150$, बोलना मुमकिन, और तीन मिनट में \omterm{def:g7:organs-at-work:recovery}{बहाली}।
वही चढ़ाई, वही बिल --- छतें अलग, जो महीनों तक ऐसी ही चढ़ाइयाँ आराम से
लेकर बनाई गई हैं।
\end{example}

\begin{method}[अपनी ही मशीन पढ़ना]\label{met:g7:organs-at-work:read}
घड़ी लेकर अपने ऊपर एक प्रयोग
(\cref{met:g4:heart-and-blood:pulse} \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} वाली तरकीब देता है):
\begin{enumerate}
  \item बैठे-बैठे और शांत हालत में आराम वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} लिखो;
  \item तीन मिनट की एक-सी मेहनत करो --- सीढ़ी चढ़ना-उतरना, रस्सी कूदना;
  \item फ़ौरन \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} लिखो, और फिर हर मिनट पर, जब तक वह आराम वाली पर न
        लौट आए: यही \omterm{def:g7:organs-at-work:recovery}{बहाली} का समय है;
  \item नियमित कसरत के एक पूरे सत्र भर हर हफ़्ते यही दोहराओ: देखो कि
        आराम वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी}, मेहनत वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} और \omterm{def:g7:organs-at-work:recovery}{बहाली} का समय, तीनों नीचे
        सरकते जाते हैं --- यानी
        \cref{prop:g7:organs-at-work:training} वाली दोबारा बनावट,
        तुम्हारे अपने आँकड़ों में।
\end{enumerate}
\end{method}

\section{समस्या: स्कूल का चुस्ती-अध्ययन}

\begin{problem}[{सप्ताहांत समस्या --- एक सत्र भर के नाड़ी-आँकड़े, शरीर-क्रिया
वैज्ञानिक की तरह पढ़े हुए}]\label{pb:g7:organs-at-work:1}
कक्षा एक सत्र भर हर सोमवार \cref{met:g7:organs-at-work:read} चलाती है:
आराम वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी}, तीन मिनट सीढ़ी चढ़ने-उतरने के बाद वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी}, और \omterm{def:g7:organs-at-work:recovery}{बहाली}
का समय। दो बिना नाम वाले लेखे: छात्र क --- हफ़्ता 1: आराम $84$, मेहनत
$168$, \omterm{def:g7:organs-at-work:recovery}{बहाली} $9$ मिनट; हफ़्ता 12: आराम $72$, मेहनत $150$, \omterm{def:g7:organs-at-work:recovery}{बहाली} $5$
मिनट। छात्र ख --- हफ़्ता 1: आराम $70$, मेहनत $150$, \omterm{def:g7:organs-at-work:recovery}{बहाली} $4$ मिनट;
हफ़्ता 12: आराम $69$, मेहनत $148$, \omterm{def:g7:organs-at-work:recovery}{बहाली} $4$ मिनट।

\medskip\noindent\textbf{भाग I --- एक ही बैठक के भीतर।}
\begin{enumerate}
  \item सीढ़ी चढ़ने-उतरने के दौरान गिनाओ कि जाँघ की \omterm{def:g3:bones-and-muscles:muscle}{पेशी} की \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} और
        \omterm{prop:g7:organs-at-work:bill}{ग्लूकोज़} की खींच का क्या हो रहा है, और उसकी \omterm{def:g7:what-is-respiration:respiration}{कार्बन डाइऑक्साइड}
        तथा ऊष्मा की पैदावार का क्या।
  \item समझाओ कि मेहनत के उन मिनटों में साँस और \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} \emph{साथ-साथ}
        क्यों चढ़ती हैं।
  \item तीसरे मिनट तक एक विद्यार्थी को गरमी लगती है और चेहरा लाल पड़
        जाता है। भट्ठी और ठंडक वाला हिसाब दो।
  \item सीढ़ी चढ़ना-उतरना रुकते ही \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} आराम वाली पर क्यों नहीं गिर
        जाती?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- पूरा सत्र पढ़ना।}
\begin{enumerate}[resume]
  \item सत्र भर छात्र क के बदलाव बताओ, और हर संख्या के पीछे
        \cref{prop:g7:organs-at-work:training} वाली दोबारा बनावटों के
        नाम भी।
  \item छात्र ख की संख्याएँ मुश्किल से हिलीं। दोनों ईमानदार पढ़तें दो
        --- और यह भी कि कक्षा के बाहर उसके जीवन का कौन-सा एक और तथ्य
        इनमें फ़ैसला कर देगा।
  \item पाँचवें हफ़्ते एक विद्यार्थी जल्दबाज़ी में खाए दोपहर के खाने के
        ठीक बाद नापता है और अजीब संख्याएँ पाता है।
        \cref{prop:g7:organs-at-work:supply} के किस नियम ने दख़ल दिया,
        और नापने के समय के बारे में कार्य-विधि में क्या लिखा होना
        चाहिए?
  \item हर विद्यार्थी को हमेशा एक ही तरह से क्यों नापना चाहिए --- वही
        मेहनत, वही समय? (तरीक़ों में से एक पुराना दोस्त; उसका नाम बताओ।)
\end{enumerate}

\medskip\noindent\textbf{भाग III --- शरीर-क्रिया वैज्ञानिक का ख़त।}
\begin{enumerate}[resume]
  \item कक्षा के सूचना-पट के लिए छात्र क के सत्र का दो वाक्यों वाला
        सार लिखो, और उसमें \emph{छत} तथा \emph{\omterm{def:g7:organs-at-work:recovery}{बहाली}} शब्द इस्तेमाल
        करो।
  \item एक अभिभावक पूछते हैं कि क्या कम आराम वाली \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} चिंता की बात
        है। \cref{exo:g4:heart-and-blood:11} वाले साइकिल-सवार से जवाब
        दो।
  \item कक्षा अगले सत्र में एक ही संख्या पर नज़र रखना चाहती है।
        \cref{def:g7:organs-at-work:recovery} की मदद से मेहनत वाली
        \omterm{prop:g4:heart-and-blood:pulse}{नाड़ी} के मुक़ाबले \omterm{def:g7:organs-at-work:recovery}{बहाली} के समय की वकालत करो।
  \item अध्ययन की सीख एक वाक्य में देकर बात ख़त्म करो: छात्र क की मशीन
        को किसने दोबारा बनाया?
\end{enumerate}
\end{problem}
