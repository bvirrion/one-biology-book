\chapter{Nucleotides and Nucleic Acids}\label{ch:b1:nucleic-acids}

Pour cold ethanol onto a solution of broken \omterm{def:b1:cell-unit-of-life:cell}{cells} and a white cloud of
fibres appears that can be wound onto a glass rod: \omterm{def:b1:nucleic-acids:chain}{DNA}, the molecule
that carries every instruction of the \omterm{def:b1:organism-environment:organism}{organism}, visible to the naked
eye. Two metres of it are folded into every human nucleus; the whole of
it is written in an alphabet of four letters, read in one direction,
and copied by separating two strands that fit each other like a hand
and its glove. This chapter describes the \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides} --- letters,
\omterm{def:b1:water-small-molecules:atp}{energy currency} and coenzymes at once --- the chains they form, the
\omterm{thm:b1:nucleic-acids:helix}{double helix} and the evidence for it, what happens when it is melted
and re-annealed, and the \omterm{def:b1:nucleic-acids:chain}{RNAs} that carry its message into the \omterm{def:b1:cell-unit-of-life:cell}{cell}.

\section{Nucleotides}

\begin{definition}[Nucleotide]\label{def:b1:nucleic-acids:nucleotide}
A \emph{nucleotide}\index{nucleotide} is made of three parts: a
nitrogenous \emph{base}\index{nitrogenous base} --- a
\emph{purine}\index{purine} (adenine A, guanine G: two fused rings) or a
\emph{pyrimidine}\index{pyrimidine} (cytosine C, thymine T, uracil U:
one ring); a five-carbon sugar, \emph{ribose} in \omterm{def:b1:nucleic-acids:chain}{RNA} or
\emph{2$'$-deoxyribose} in \omterm{def:b1:nucleic-acids:chain}{DNA} (lacking the hydroxyl on carbon
$2'$); and one to three \emph{phosphate} groups on the sugar's carbon
$5'$. The base is attached to carbon $1'$; the sugar carbons are
numbered with primes to distinguish them from the base's. Base plus
sugar is a \emph{nucleoside} (adenosine, guanosine, cytidine,
thymidine, uridine); with phosphates it is a nucleotide (AMP, ADP, \omterm{def:b1:water-small-molecules:atp}{ATP}
and their kin).
\end{definition}

\begin{proposition}[Nucleotides do three jobs]\label{prop:b1:nucleic-acids:threejobs}
Beyond being the monomers of \omterm{def:b1:nucleic-acids:chain}{DNA} and \omterm{def:b1:nucleic-acids:chain}{RNA}, \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides} are the \omterm{def:b1:cell-unit-of-life:cell}{cell}'s
\emph{\omterm{def:b1:water-small-molecules:atp}{energy currency}} --- \omterm{def:b1:water-small-molecules:atp}{ATP} and GTP, whose \omterm{def:b1:water-small-molecules:atp}{phosphoanhydride bonds}
(\cref{ch:b1:water-small-molecules}) pay for synthesis, transport and
movement; its \emph{coenzymes} --- $\mathrm{NAD^+}$, $\mathrm{NADP^+}$
and FAD, which carry electrons in respiration and photosynthesis
(\cref{ch:b1:respiration-fermentation}), and coenzyme A, which carries
acyl groups; and its \emph{messengers} --- cyclic AMP, which relays
hormone signals inside \omterm{def:b1:cell-unit-of-life:cell}{cells}. All of them are adenine \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides} or
built on them: the same molecule that spells the message powers its
execution.
\end{proposition}

\section{The polynucleotide chain}

\begin{definition}[Phosphodiester bond, polarity]\label{def:b1:nucleic-acids:chain}
\omterm{def:b1:nucleic-acids:nucleotide}{Nucleotides} are joined into a \emph{polynucleotide}\index{polynucleotide}
by \emph{phosphodiester bonds}\index{phosphodiester bond}: the
phosphate on carbon $5'$ of one \omterm{def:b1:nucleic-acids:nucleotide}{nucleotide} is linked to the $3'$
hydroxyl of the next. The chain therefore has a sugar--phosphate
\emph{backbone}, uniform and negatively charged (one charge per
phosphate), from which the \omterm{def:b1:nucleic-acids:nucleotide}{bases} project as a sequence; and it has a
direction: a \emph{$5'$ end}\index{5' end@$5'$ end} bearing a free
phosphate and a \emph{$3'$ end}\index{3' end@$3'$ end} bearing a free
hydroxyl. Sequences are written and read $5' \to 3'$, the direction in
which chains are synthesised. \emph{DNA}\index{DNA} (deoxyribonucleic
acid) uses deoxyribose and the \omterm{def:b1:nucleic-acids:nucleotide}{bases} A, G, C, T; \emph{RNA}\index{RNA}
(ribonucleic acid) uses ribose and A, G, C, U.
\end{definition}

\begin{example}[RNA is fragile, DNA is durable]\label{ex:b1:nucleic-acids:fragile}
The $2'$ hydroxyl of ribose can attack the neighbouring \omterm{def:b1:nucleic-acids:chain}{phosphodiester
bond}: in mild alkali \omterm{def:b1:nucleic-acids:chain}{RNA} is hydrolysed to \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides} within hours,
while \omterm{def:b1:nucleic-acids:chain}{DNA}, lacking that hydroxyl, survives boiling in alkali and has
been recovered intact from bones tens of thousands of years old. Uracil
in \omterm{def:b1:nucleic-acids:chain}{RNA} is thymine without a methyl group; \omterm{def:b1:nucleic-acids:chain}{DNA} uses thymine so that
cytosine, which slowly loses its amino group to become uracil, can be
recognised as damaged and repaired --- a uracil in \omterm{def:b1:nucleic-acids:chain}{DNA} is always a
mistake.
\end{example}

\section{The double helix}

\begin{proposition}[Chargaff's rules]\label{prop:b1:nucleic-acids:chargaff}
\index{Chargaff's rules}
In the \omterm{def:b1:nucleic-acids:chain}{DNA} of any \omterm{def:b1:organism-environment:organism}{organism} the amount of adenine equals that of
thymine and the amount of guanine that of cytosine ($A = T$,
$G = C$, hence \omterm{def:b1:nucleic-acids:nucleotide}{purines} $=$ \omterm{def:b1:nucleic-acids:nucleotide}{pyrimidines}), while the ratio $(G + C)/(A + T)$
varies from species to species (from \qty{25}{\%} to \qty{75}{\%}
$G + C$).
\end{proposition}
\begin{proof}[Evidence]
\omterm{prop:b1:nucleic-acids:chargaff}{Chargaff} (1950) hydrolysed \omterm{def:b1:nucleic-acids:chain}{DNA} from many \omterm{def:b1:organism-environment:organism}{organisms} and separated the
\omterm{def:b1:nucleic-acids:nucleotide}{bases} by chromatography, measuring each by its ultraviolet absorption.
The equalities held within experimental error for every \omterm{def:b1:nucleic-acids:chain}{DNA}; the
$G + C$ content did not, and was the same in every \omterm{def:b1:body-plans-tissues:tissue}{tissue} of one
\omterm{def:b1:organism-environment:organism}{organism}.
\end{proof}

\begin{theorem}[The Watson--Crick double helix]\label{thm:b1:nucleic-acids:helix}
\index{double helix}\index{base pair}\index{antiparallel}
\omterm{def:b1:nucleic-acids:chain}{DNA} is two \omterm{def:b1:nucleic-acids:chain}{polynucleotide} chains wound around a common axis into a
right-handed \emph{\omterm{thm:b1:nucleic-acids:helix}{double helix}}: the sugar--phosphate backbones
outside, the \omterm{def:b1:nucleic-acids:nucleotide}{bases} inside, flat and stacked perpendicular to the axis.
The chains are \emph{antiparallel} (one runs $5' \to 3'$ upward, the
other downward) and \emph{complementary}: opposite every A is a T,
held by two \omterm{def:b1:water-small-molecules:water}{hydrogen bonds}, and opposite every G a C, held by three;
each such \emph{\omterm{thm:b1:nucleic-acids:helix}{base pair}} is the same width, so the helix is uniform
whatever the sequence. In the common B form the helix is \qty{2}{nm}
across, rises \qty{0.34}{nm} per \omterm{thm:b1:nucleic-acids:helix}{base pair} and \qty{3.4}{nm} per turn
of ten pairs, and shows a wide \emph{major groove} and a narrow
\emph{minor groove} in which the edges of the \omterm{def:b1:nucleic-acids:nucleotide}{bases} are exposed to
proteins.
\end{theorem}
\begin{proof}[Evidence]
Franklin's X-ray diffraction pictures of \omterm{def:b1:nucleic-acids:chain}{DNA} fibres (1952) showed the
cross of a helix with a repeat of \qty{3.4}{nm} along the axis, a
strong reflection at \qty{0.34}{nm} (stacked flat units), and a
diameter of \qty{2}{nm}; the phosphates, the heaviest atoms, had to be
outside. Watson and Crick (1953) found that a \omterm{def:b1:nucleic-acids:nucleotide}{purine} paired with a
\omterm{def:b1:nucleic-acids:nucleotide}{pyrimidine} --- and only A with T and G with C --- gives pairs of equal
width fitting the \qty{2}{nm} diameter, explaining \omterm{prop:b1:nucleic-acids:chargaff}{Chargaff}'s rules,
and that two antiparallel chains so paired build a regular helix with
the fibre's dimensions. The model predicted that each chain is a
template for the other; Meselson and Stahl's demonstration of
semi-conservative replication (recalled from the High School volume,
and in \cref{ch:b1:replication-mitosis}) confirmed it.
\end{proof}

\begin{example}[Reading a strand]\label{ex:b1:nucleic-acids:complement}
If one strand reads $5'$-ATGGCATTC-$3'$, its partner, written
$5' \to 3'$, is $5'$-GAATGCCAT-$3'$: complement each \omterm{def:b1:nucleic-acids:nucleotide}{base}, then reverse
the order. Nine pairs, \qty{3.1}{nm} of helix, twenty-two \omterm{def:b1:water-small-molecules:water}{hydrogen
bonds} (five G--C, four A--T). A human chromosome of \num{2.5e8} pairs
is \qty{8.5}{cm} long in this form and must be folded a hundred
thousand times to fit a nucleus (\cref{ch:b1:genomes}).
\end{example}

\begin{proposition}[What holds the helix together]\label{prop:b1:nucleic-acids:stability}
The \omterm{def:b1:water-small-molecules:water}{hydrogen bonds} between paired \omterm{def:b1:nucleic-acids:nucleotide}{bases} give specificity; the
\emph{stacking}\index{base stacking} of the flat \omterm{def:b1:nucleic-acids:nucleotide}{bases} on one another
--- \omterm{def:b1:water-small-molecules:bonds}{van der Waals} contacts and the \omterm{prop:b1:water-small-molecules:properties}{hydrophobic effect} that keeps the
\omterm{def:b1:nucleic-acids:nucleotide}{bases} out of water --- gives most of the stability; the negatively
charged backbones repel each other, and the helix is stable only in the
presence of cations (the physiological $\mathrm{Mg^{2+}}$,
$\mathrm{Na^+}$, or the basic proteins of chromatin) that screen the
charge. A G--C pair, with three \omterm{def:b1:water-small-molecules:water}{hydrogen bonds} and stronger stacking,
holds better than an A--T pair.
\end{proposition}

\section{Melting and hybridisation}

\begin{definition}[Denaturation, melting temperature]\label{def:b1:nucleic-acids:melting}
Heating, or extreme pH, separates the two strands of \omterm{def:b1:nucleic-acids:chain}{DNA}:
\emph{denaturation}\index{denaturation (DNA)} (melting), a cooperative
transition over a few degrees, whose midpoint is the \emph{melting
temperature}\index{melting temperature} $T_m$. It is followed by the
\emph{hyperchromic effect}: single strands absorb about \qty{40}{\%}
more ultraviolet light at \qty{260}{nm} than the same \omterm{def:b1:nucleic-acids:nucleotide}{bases} stacked in
a helix. $T_m$ rises with the $G + C$ content (\qty{0.4}{\celsius} per
percent), with the salt concentration, and with length; in
\qty{0.15}{mol/L} salt a long \omterm{def:b1:nucleic-acids:chain}{DNA} of \qty{50}{\%} $G + C$ melts near
\qty{90}{\celsius}. Cooled slowly, the strands find their complements
and re-form the helix: \emph{renaturation}, or \emph{hybridisation}\index{hybridisation}
when the strands come from different sources.
\end{definition}

\begin{method}[Measuring nucleic acids by absorbance]\label{met:b1:nucleic-acids:absorbance}
\begin{enumerate}
  \item \omterm{def:b1:nucleic-acids:nucleotide}{Bases} absorb ultraviolet light with a maximum at
        \qty{260}{nm}. In a \qty{1}{cm} \omterm{def:b1:cell-unit-of-life:cell}{cell}, an absorbance of $1.0$
        corresponds to \qty{50}{\micro g/mL} of double-stranded \omterm{def:b1:nucleic-acids:chain}{DNA}
        (\qty{40}{\micro g/mL} of \omterm{def:b1:nucleic-acids:chain}{RNA}, \qty{33}{\micro g/mL} of
        single-stranded \omterm{def:b1:nucleic-acids:chain}{DNA}).
  \item The ratio $A_{260}/A_{280}$ is $1.8$ for pure \omterm{def:b1:nucleic-acids:chain}{DNA} and $2.0$ for
        pure \omterm{def:b1:nucleic-acids:chain}{RNA}; proteins absorb at \qty{280}{nm} and lower it.
  \item To follow melting, record $A_{260}$ while heating slowly; the
        curve's midpoint is $T_m$ and its steepness a measure of the
        homogeneity of the sample.
  \item To detect a sequence, denature the sample and add a labelled
        single strand complementary to it (a \emph{probe}); it
        hybridises only where its complement is, and the label marks
        the place --- on a gel, on a chromosome, in a \omterm{def:b1:body-plans-tissues:tissue}{tissue}.
\end{enumerate}
\end{method}

\begin{example}[A probe finds a gene]\label{ex:b1:nucleic-acids:probe}
A \omterm{def:b1:nucleic-acids:nucleotide}{twenty-nucleotide} probe complementary to one gene, at
\qty{5}{\celsius} below its own $T_m$, binds its exact complement and
nothing else in three billion \omterm{thm:b1:nucleic-acids:helix}{base pairs}: a single mismatch in twenty
lowers $T_m$ by several degrees and the mismatched hybrid melts. The
specificity of \omterm{def:b1:nucleic-acids:nucleotide}{base} pairing, multiplied over twenty positions, is what
every \omterm{def:b1:nucleic-acids:chain}{DNA} test, from paternity to pathogen detection, relies on.
\end{example}

\section{RNA}

\begin{definition}[The RNAs]\label{def:b1:nucleic-acids:rnas}
\omterm{def:b1:nucleic-acids:chain}{RNA} is usually single-stranded, and folds back on itself wherever
short complementary stretches allow, into hairpins, loops and bulges
--- a \emph{secondary structure} --- that then packs into a shape. Three
classes carry the flow of genetic information
(\cref{ch:b1:gene-expression}): \emph{messenger RNA}\index{messenger RNA} (mRNA), a copy of a gene that the ribosome reads;
\emph{transfer RNA}\index{transfer RNA} (tRNA), some \qty{75}{}
\omterm{def:b1:nucleic-acids:nucleotide}{nucleotides} folded into a cloverleaf, carrying an amino acid at its
$3'$ end and an \emph{anticodon} that pairs with the message;
\emph{ribosomal RNA}\index{ribosomal RNA} (rRNA), the structural and
catalytic core of the ribosome, four fifths of the \omterm{def:b1:nucleic-acids:chain}{RNA} of a \omterm{def:b1:cell-unit-of-life:cell}{cell}. Many
other \omterm{def:b1:nucleic-acids:chain}{RNAs} regulate, splice, and guide (the Year 3 volume).
\end{definition}

\begin{remark}[RNA can be a catalyst]\label{rem:b1:nucleic-acids:ribozyme}
Some \omterm{def:b1:nucleic-acids:chain}{RNAs} are enzymes (\emph{ribozymes}): the peptide bond of every
protein is formed by the \omterm{def:b1:nucleic-acids:rnas}{ribosomal RNA}, not by a protein, and some
\omterm{def:b1:nucleic-acids:chain}{RNAs} cut and splice themselves. A molecule that both carries
information and catalyses reactions is what a first living system may
have been made of; the DNA--protein world would then be a later
division of labour, with \omterm{def:b1:nucleic-acids:chain}{RNA} kept at its centre.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:1}
Name the three parts of a \omterm{def:b1:nucleic-acids:nucleotide}{nucleotide} and the two differences between
a ribonucleotide and a deoxyribonucleotide.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:2}
Write the complementary strand of $5'$-GGATCCTA-$3'$ in the $5' \to 3'$
direction, and count its \omterm{def:b1:water-small-molecules:water}{hydrogen bonds}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:3}
A \omterm{def:b1:nucleic-acids:chain}{DNA} contains \qty{22}{\%} adenine. Give the percentages of T, G and
C, and the $G + C$ content.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:4}
From the melting figure, what is the $T_m$ of a \omterm{def:b1:nucleic-acids:chain}{DNA} of \qty{50}{\%}
$G + C$ in the same conditions, and by how much does the absorbance
rise on melting?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:5}
Compute the length of the \emph{E. coli} chromosome ($\num{4.6e6}$
\omterm{thm:b1:nucleic-acids:helix}{base pairs}) in millimetres, its number of turns, and the factor by
which it must be compacted to fit in a \omterm{def:b1:cell-unit-of-life:cell}{cell} of \qty{2}{\micro m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:6}
Explain why the two strands of \omterm{def:b1:nucleic-acids:chain}{DNA} must be antiparallel, using the
geometry of the sugar--phosphate backbone and the requirement that
every \omterm{thm:b1:nucleic-acids:helix}{base pair} have the same width.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:7}
A solution of \omterm{def:b1:nucleic-acids:chain}{DNA} has $A_{260} = 0.35$ in a \qty{1}{cm} \omterm{def:b1:cell-unit-of-life:cell}{cell} and
$A_{280} = 0.20$. Compute the concentration and comment on the purity.
How many \omterm{thm:b1:nucleic-acids:helix}{base pairs} are in \qty{1}{mL}, if the mean mass of a pair is
\qty{650}{Da}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:8}
Explain the hyperchromic effect and why the melting curve of a pure
\omterm{def:b1:nucleic-acids:chain}{DNA} is steep while that of a mixture of \omterm{def:b1:nucleic-acids:chain}{DNAs} from many species is
spread out.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:9}
A \omterm{def:b1:nucleic-acids:nucleotide}{twenty-nucleotide} probe has $T_m = \qty{62}{\celsius}$ with its exact
complement; one mismatch lowers $T_m$ by about \qty{5}{\celsius}. At
which temperature should \omterm{def:b1:nucleic-acids:melting}{hybridisation} be done to detect the exact
sequence and reject single mismatches? What happens at
\qty{45}{\celsius}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:10}
\omterm{prop:b1:nucleic-acids:chargaff}{Chargaff}'s rules hold for double-stranded \omterm{def:b1:nucleic-acids:chain}{DNA} but not for \omterm{def:b1:nucleic-acids:chain}{RNA} or for
the \omterm{def:b1:nucleic-acids:chain}{DNA} of certain small viruses. Explain what each exception tells
about the structure of those nucleic acids.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:11}
Two \omterm{def:b1:water-small-molecules:water}{hydrogen bonds} hold an A--T pair, yet a helix of a thousand A--T
pairs is stable at \qty{60}{\celsius} while a single pair does not
form at all. Explain with the ideas of \cref{ch:b1:water-small-molecules}
(weak bonds, cooperativity, stacking), and say what this implies for
the fidelity of a probe.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:12}
``The structure of \omterm{def:b1:nucleic-acids:chain}{DNA} is the explanation of heredity.'' Discuss in a
paragraph: what the \omterm{thm:b1:nucleic-acids:helix}{double helix} explains immediately (copying,
mutation as a change of sequence), what it does not explain by itself
(how the sequence becomes a protein), and why Watson and Crick's model
was accepted before any biochemical test of it.
\end{exercise}

\section{Problem: The Genome of a Virus}

\begin{problem}[{Weekend problem --- the DNA of phage $\lambda$
measured, weighed, counted, melted and packed into its capsid, ending
on its melting temperature and its contour length}]
\label{pb:b1:nucleic-acids:1}
The genome of bacteriophage $\lambda$ is one linear double-stranded \omterm{def:b1:nucleic-acids:chain}{DNA}
of \num{48502} \omterm{thm:b1:nucleic-acids:helix}{base pairs} with a $G + C$ content of \qty{49.9}{\%}. It
is packed into an icosahedral capsid of \qty{55}{nm} internal diameter.
Take \qty{0.34}{nm} per \omterm{thm:b1:nucleic-acids:helix}{base pair}, \qty{2.0}{nm} for the helix
diameter, \qty{650}{Da} per \omterm{thm:b1:nucleic-acids:helix}{base pair}, and $T_m = 81.5 + 16.6\log_{10}[\mathrm{Na^+}]
+ 0.41\,(\%\,G{+}C) - 500/L$ in degrees Celsius, with $[\mathrm{Na^+}]$
in \unit{mol/L} and $L$ the length in \omterm{thm:b1:nucleic-acids:helix}{base pairs}.

\medskip
\noindent\textbf{Part I --- Measured and weighed.}
\begin{enumerate}
  \item Compute the contour length of the molecule in micrometres.
  \item Compute the number of turns of the helix.
  \item Compute its molar mass in daltons.
  \item Compute the mass of one molecule in grams.
  \item Compute the number of phosphate groups and hence the net charge
        of the molecule.
  \item Compute the number of \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides} of each \omterm{def:b1:nucleic-acids:nucleotide}{base} (A, T, G, C).
  \item Compute the total number of \omterm{def:b1:water-small-molecules:water}{hydrogen bonds} holding the two
        strands.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Melted.}
\begin{enumerate}[resume]
  \item Compute $T_m$ in \qty{0.1}{mol/L} sodium.
  \item Compute $T_m$ in \qty{0.01}{mol/L} sodium, and explain why
        salt stabilises the helix.
  \item The $\lambda$ genome contains a region of \qty{1000}{} pairs at
        \qty{35}{\%} $G + C$ and one at \qty{62}{\%}. Compute the $T_m$
        of each region in \qty{0.1}{mol/L} sodium and describe the
        shape of the melting curve of the whole molecule.
  \item A solution of $\lambda$ \omterm{def:b1:nucleic-acids:chain}{DNA} has $A_{260} = 0.50$ before
        melting. What is its concentration, and what will $A_{260}$
        be after complete melting?
  \item How many $\lambda$ molecules are in \qty{1}{mL} of that
        solution?
  \item The two ends of the $\lambda$ genome are single-stranded
        overhangs of 12 \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides}, complementary to each other, by
        which the molecule circularises in the host. Compute their
        $T_m$ (use $L = 12$) and say why the circle nevertheless holds
        at \qty{37}{\celsius} inside the \omterm{def:b1:cell-unit-of-life:cell}{cell}.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Packed.}
\begin{enumerate}[resume]
  \item Compute the volume of the \omterm{def:b1:nucleic-acids:chain}{DNA} as a cylinder.
  \item Compute the internal volume of the capsid.
  \item What fraction of the capsid does the \omterm{def:b1:nucleic-acids:chain}{DNA} fill? Compare with
        the closest packing of parallel cylinders (\qty{91}{\%}).
  \item The \omterm{def:b1:nucleic-acids:chain}{DNA} carries \num{97000} negative charges in a sphere of
        \qty{55}{nm}. Explain what must accompany the \omterm{def:b1:nucleic-acids:chain}{DNA} into the
        capsid, and why the packed \omterm{def:b1:nucleic-acids:chain}{DNA} exerts a pressure on the shell
        (tens of atmospheres) that helps inject it into a bacterium.
  \item The molecule must be bent into loops of radius comparable to
        the capsid's. \omterm{def:b1:nucleic-acids:chain}{DNA} resists bending below a radius of about
        \qty{50}{nm}. Comment.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Read.}
\begin{enumerate}[resume]
  \item One strand of the genome starts $5'$-GGGCGGCGACCT-$3'$. Write the
        complementary strand $5' \to 3'$.
  \item If a gene of \num{1200} \omterm{thm:b1:nucleic-acids:helix}{base pairs} is transcribed into an mRNA,
        what is the mRNA's length in \omterm{def:b1:nucleic-acids:nucleotide}{nucleotides} and, at \qty{330}{Da}
        per \omterm{def:b1:nucleic-acids:nucleotide}{nucleotide}, its mass?
  \item The genome has \qty{75}{} genes in \num{48502} pairs. What
        fraction of the genome is coding if the average gene is
        \num{600} pairs? Compare with a human genome (\qty{1.5}{\%}).
  \item The genome is replicated in \qty{20}{min} at \qty{37}{\celsius}
        by the host's enzymes. Compute the rate in \omterm{thm:b1:nucleic-acids:helix}{base pairs} per second
        if replication starts at one origin and proceeds in both
        directions.
  \item How many \omterm{def:b1:water-small-molecules:water}{hydrogen bonds} are broken per second at the two
        forks during replication (use the mean per pair from
        question 7)?
  \item A mutation changes one G--C pair to A--T. By how much does the
        $T_m$ of the whole molecule change? Could melting detect it?
  \item State the result: the \omterm{def:b1:nucleic-acids:melting}{melting temperature} of $\lambda$ \omterm{def:b1:nucleic-acids:chain}{DNA} in
        \qty{0.1}{mol/L} sodium and its contour length, and the
        compaction that fits it into the capsid.
\end{enumerate}
\end{problem}
