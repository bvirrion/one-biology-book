\chapter{الكائن الحي: نظام في تفاعل مع محيطه}\label{ch:b1:organism-environment}

أرنب رابض على حافة مرج في صباح من صباحات يناير. في الساعة التالية
سيأكل مئة غرام من العشب، ويتنفس نحو \qty{400}{L} من الهواء، ويفقد
حرارة عبر فروه إلى هواء عند \qty{2}{\celsius}، ويشرب من بركة، ويفرّ
حين يعبر ظل حوّام أرض المرج. ولا شيء مما يفعله ممكن دون أن يعبر شيء
ما حدّه: مادة تدخل وتخرج، وطاقة تدخل وتخرج، ومعلومة تدخل. وعلى بعد
مئتي متر تفعل بلوطة الأشياء نفسها دون أن تتحرك --- تأخذ ثاني أكسيد
الكربون والضوء، وتسحب الماء عبر مجموع جذري أوسع من تاجها، وتفقد ذلك
الماء إلى الهواء البارد نفسه. يبين هذا الفصل ما \omterm{def:b1:organism-environment:organism}{الكائن الحي} بوصفه
نظامًا، ولماذا يملي حجمه طريقة بنائه، وكيف يثبّت داخله بينما يتغير
كل ما حوله، وكيف تتحجم حاجته من الطاقة مع كتلته. والكائنان الحيان في
هذه الافتتاحية، ثديي ونبات زهري، هما النموذجان اللذان تعود إليهما
السنة كلها.

\section{نظام مفتوح}

\begin{definition}[الكائن الحي، النظام المفتوح]\label{def:b1:organism-environment:organism}
\emph{الكائن الحي}\index{كائن حي} فرد حي: نظام محدود، يصون نفسه
ويتكاثر، مكوّن من خلية واحدة أو من خلايا كثيرة تشترك في أصل واحد
وعمل واحد. وهو \emph{نظام مفتوح}\index{نظام مفتوح}: تعبر المادة
والطاقة حدّه باستمرار، ولا يبقى حيًا إلا ما دامتا تعبرانه.
و\emph{محيط}\index{محيط} الكائن الحي هو كل ما يقع خارج ذلك الحد
ويتبادل معه --- الوسط الفيزيائي (الهواء والماء والتربة والضوء ودرجة
الحرارة) والكائنات الحية الأخرى.
\end{definition}

\begin{proposition}[التبادلات الثلاثة]\label{prop:b1:organism-environment:exchanges}
يتبادل كل \omterm{def:b1:organism-environment:organism}{كائن حي} ثلاثة أشياء مع محيطه:
\begin{itemize}
  \item \emph{المادة}\index{تبادل المادة}: يأخذ المغذيات والماء
        والغازات، ويطرح الفضلات والغازات والماء؛
  \item \emph{الطاقة}\index{تبادل الطاقة}: يأخذ الطاقة ضوءًا (ذاتيات
        التغذي) أو طاقة كيميائية في الغذاء (غير ذاتيات التغذي)،
        ويطرحها حرارةً في جلها؛
  \item \emph{المعلومة}\index{تبادل المعلومة}: يكشف الإشارات الآتية
        من \omterm{def:b1:organism-environment:organism}{المحيط} (ضوء، ومواد كيميائية، ودرجة حرارة، ولمس، وصوت)
        ويستجيب لها.
\end{itemize}
وعلى مدى فترة لا ينمو فيها \omterm{def:b1:organism-environment:organism}{الكائن الحي} ولا يضمر، تكون حصيلته مغلقة:
المادة الداخلة تساوي المادة الخارجة، والطاقة الداخلة تساوي الحرارة
الخارجة زائد الشغل المبذول على الخارج.
\end{proposition}
\begin{proof}
الأولان يتبعان من انحفاظ الكتلة والطاقة مطبَّقًا على الحجم الواقع
داخل الحد: فما يتراكم هو الفرق بين الوارد والصادر، \omterm{def:b1:organism-environment:organism}{والكائن الحي}
المستقر لا يراكم شيئًا. والطاقة الكيميائية التي لا تُخزَّن كتلة حية
جديدة تنتهي حرارةً، لأن كل تفاعل من تفاعلات \omterm{def:b1:organism-environment:organism}{الكائن الحي} يبدّد طاقته
الحرة حرارةً، والشغل الميكانيكي المبذول على الخارج (تنقّل، رفع) هو
المخرج الوحيد الآخر. أما الثالث فملاحظة مفادها أن ما من \omterm{def:b1:organism-environment:organism}{كائن حي}
جامد إزاء ما يحيط به.
\end{proof}

\begin{definition}[ذاتي التغذي وغير ذاتي التغذي]\label{def:b1:organism-environment:trophy}
يكون \omterm{def:b1:organism-environment:organism}{الكائن الحي} \emph{ذاتي التغذي}\index{التغذي الذاتي} حين يبني
مادته العضوية من الكربون المعدني (ثاني أكسيد الكربون) بمنبع طاقة
خارجي --- هو الضوء عند \emph{ذاتيات التغذي الضوئية}\index{ذاتي التغذي الضوئي}
(النباتات والطحالب والبكتيريا الزرقاء)، وأكسدة المركبات المعدنية عند
\emph{ذاتيات التغذي الكيميائية}\index{ذاتي التغذي الكيميائي}. ويكون
\emph{غير ذاتي التغذي}\index{التغذي غير الذاتي} حين يلزمه أن يأخذ
مادة عضوية صنعتها \omterm{def:b1:organism-environment:organism}{كائنات} أخرى، فتمده بكربونه وبطاقته معًا (الحيوانات
والفطريات ومعظم البكتيريا).
\end{definition}

\begin{example}[الأرنب والبلوطة]\label{ex:b1:organism-environment:rabbitoak}
مئة غرام العشب عند الأرنب تمده، بعد الهضم، بنحو \qty{500}{kJ} من
الطاقة الكيميائية وبكربون كل جزيء سيبنيه؛ والأكسجين الذي يتنفسه
يتيح له أكسدة ذلك الغذاء، وثاني أكسيد الكربون الذي يزفره هو ذلك
الكربون نفسه وهو يخرج. أما مدخول البلوطة فمعكوس: ثاني أكسيد الكربون
داخلًا، والأكسجين خارجًا، والطاقة آتية ضوءًا --- نحو \qty{5}{kW}
تسقط على تاجها عند الظهيرة، تخزن منها أقل من واحد في المئة سكرًا.
وكلاهما يطرح جل ما يعالجه من طاقة حرارةً: ففرو الأرنب دافئ، وأوراق
البلوطة يبردها التبخر الذي يقوده امتصاصها للضوء نفسه.
\end{example}

\section{مستويات التنظيم ومشكلة الحجم}

\begin{definition}[مستويات التنظيم]\label{def:b1:organism-environment:levels}
تنتظم المادة الحية في \emph{مستويات تنظيم}\index{مستويات التنظيم}
متداخلة، يُبنى كل منها من وحدات المستوى الذي دونه، ولكل منها خواص لا
يملكها المستوى الأدنى: الجزيئات ($\qty{1}{nm}$)، والخلايا
($\qtyrange{1}{100}{\micro m}$)، والأنسجة، والأعضاء
($\qtyrange{1}{100}{mm}$)، والأجهزة، \omterm{def:b1:organism-environment:organism}{والكائن الحي}
($\qty{1}{\micro m}$ إلى $\qty{100}{m}$)، ثم فوقه الجماعة، والمجتمع،
والنظام البيئي، والمحيط الحيوي. فالخلية تتنفس، والجزيء لا يتنفس؛
والجماعة تتطور، \omterm{def:b1:organism-environment:organism}{والكائن الحي} لا يتطور.
\end{definition}

\begin{proposition}[السطح والحجم]\label{prop:b1:organism-environment:sv}
في الأجسام المتشابهة الشكل ذات القياس المميز $L$، ينمو السطح مثل
$L^2$ وينمو الحجم مثل $L^3$، فتهبط نسبة السطح إلى الحجم مثل $1/L$:
\[
\frac{S}{V} \propto \frac{1}{L}, \qquad
\frac{S}{V}\Big|_{\text{كرة}} = \frac{4\pi r^2}{\tfrac{4}{3}\pi r^3} = \frac{3}{r}.
\]
وكل ما يتبادله \omterm{def:b1:organism-environment:organism}{الكائن الحي} يعبر سطحًا، وكل ما يستهلكه يتناسب مع
حجمه: \omterm{def:b1:organism-environment:organism}{فالكائن الحي} الكبير لا يستطيع أن يغذي حجمه عبر سطحه الخارجي.
\end{proposition}
\begin{proof}
تكبير شكل بعامل $k$ يضرب كل مساحة في $k^2$ وكل حجم في $k^3$، فتُضرب
النسبة في $1/k$. وفي حالة الكرة تكون الصيغة تامة الدقة.
\end{proof}

\begin{example}[بكتيريا وإنسان]\label{ex:b1:organism-environment:svnumbers}
بكتيريا نصف قطرها $\qty{0.5}{\micro m}$ لها
$S/V = 3/r = \qty{6e6}{m^{-1}}$: فكل ميكرومتر مكعب من سيتوبلازمها لا
يبعد عن الخارج أكثر من نصف ميكرومتر، والانتشار عبر غشائها يكفي
لتغذيتها كلها. أما إنسان يُنمذج بكرة حجمها \qty{70}{L} فله
$r = \qty{0.26}{m}$ و$S/V = \qty{12}{m^{-1}}$، أي أقل بنصف مليون
مرة. والجلد، وهو نحو \qty{2}{m^2}، عاجز عن تأمين الأكسجين الذي
تستهلكه \qty{70}{kg} من الأنسجة؛ أما الرئة، مطوية داخل القفص
الصدري، فتقدم \qty{100}{m^2}، والمعي، مطويًا ثلاث مرات،
\qty{30}{m^2}.
\end{example}

\begin{proposition}[سطوح التبادل مطوية ورقيقة ومهوّاة]\label{prop:b1:organism-environment:surfaces}
فوق قياس يناهز المليمتر، لا يمكن تغذية \omterm{def:b1:organism-environment:organism}{كائن حي} بالانتشار عبر سطحه
الخارجي. لذلك تحمل \omterm{def:b1:organism-environment:organism}{الكائنات} المتعددة الخلايا، مهما كان قياسها،
\emph{سطوح تبادل متخصصة}\index{سطح التبادل}، يحل تصميمها المشترك
المشكلة نفسها بأربع طرق: مساحة كبيرة (بالطي: الزغابات، والأسناخ،
وصفائح الغلاصم، وأشعار الجذور، ونصال الأوراق)، وسماكة صغيرة (طبقة
خلية واحدة أو أقل بين الوسطين)، ووسط متجدد في الخارج (تهوية، تيار
مائي، تيار نتح)، وجملة نقل في الداخل (دم، عصارة) تحمل ما عبر السطح
إلى بقية الحجم.
\end{proposition}
\begin{proof}
يعطي قانون فيك (\cref{ch:b1:membranes-transport}) التدفق عبر سطح
متناسبًا مع المساحة، ومع فرق التركيز، ومع مقلوب السماكة؛ وتجديد
الوسط الخارجي يبقي فرق التركيز كبيرًا؛ وجملة النقل تزيح ما عبر
فتبقي التركيز الداخلي منخفضًا. وكل واحدة من الخصائص الأربع تزيد
التدفق؛ \omterm{def:b1:organism-environment:organism}{والكائن الحي} الكبير يحتاج إليها جميعًا.
\end{proof}

\section{الوسط الداخلي والاستتباب}

\begin{definition}[الوسط الداخلي]\label{def:b1:organism-environment:milieu}
في حيوان متعدد الخلايا، معظم الخلايا لا تلامس العالم الخارجي بل
تغمرها \emph{وسط داخلي}\index{الوسط الداخلي}: هو السائل خارج الخلوي
(بلازما الدم، واللمف، والسائل الخلالي بين الخلايا) الذي ينظم \omterm{def:b1:organism-environment:organism}{الكائن
الحي} تركيبه ودرجة حرارته وحجمه. فالخلايا تتبادل مع هذا السائل،
والسائل يتبادل مع الخارج عبر السطوح المتخصصة. والفكرة لكلود برنار
(1865): ثبات الوسط الداخلي شرط الحياة الحرة المستقلة.
\end{definition}

\begin{definition}[الاستتباب]\label{def:b1:organism-environment:homeostasis}
\emph{الاستتباب}\index{استتباب} هو إبقاء متغير منظَّم من متغيرات
\omterm{def:b1:organism-environment:milieu}{الوسط الداخلي} (درجة الحرارة، وتركيز الغلوكوز، وpH، والتناضحية، وضغط
الدم) قريبًا من \emph{قيمة مرجعية}\index{القيمة المرجعية} رغم
التغيرات الخارجية. ويتحقق ذلك \emph{بتغذية راجعة سالبة}\index{التغذية الراجعة السالبة}:
\emph{مستقبِل} يقيس المتغير، ومركز مُدمِج يقارنه بالقيمة المرجعية،
\emph{ومنفِّذ} يعمل في الاتجاه الذي يلغي الانحراف. والمتغير ليس
ثابتًا بل يتذبذب في نطاق ضيق حول القيمة المرجعية.
\end{definition}

\begin{proposition}[داخليات الحرارة وخارجيات الحرارة]\label{prop:b1:organism-environment:thermo}
\emph{خارجي الحرارة}\index{خارجي الحرارة} (معظم اللافقاريات،
والأسماك، والبرمائيات، والزواحف) درجة حرارة جسمه تتبع درجة حرارة
\omterm{def:b1:organism-environment:organism}{المحيط}؛ ويرتفع معدل استقلابه مع الحرارة، فيتضاعف تقريبًا كل
\qty{10}{\celsius}. أما \emph{داخلي الحرارة}\index{داخلي الحرارة}
(الثدييات والطيور) فيبقي حرارة جسمه عند \omterm{def:b1:organism-environment:homeostasis}{قيمة مرجعية} بإنتاج الحرارة
استقلابيًا. وضمن \emph{منطقة حرارية محايدة}\index{المنطقة الحرارية المحايدة}
من درجات حرارة الجو يفعل ذلك بمعدله القاعدي، فلا يعدل إلا العزل
وجريان الدم إلى الجلد؛ ودون تلك المنطقة يرتفع معدل استقلابه خطيًا
كلما برد الهواء، لأن فقد الحرارة يتناسب مع فرق الحرارة ولا بد أن
يقابله إنتاج مساو؛ وفوقها يكلف التبريد بالتبخر ماءً وطاقة.
\end{proposition}
\begin{proof}[برهان جزئي]
من أجل جسم عند $T_b$ في هواء عند $T_a$، تكون الحرارة المفقودة في
واحدة الزمن عبر سطحه $P = hS\,(T_b - T_a)$ (قانون نيوتن في التبريد)،
حيث $h$ معامل يخفضه الفرو أو الريش أو الشحم. واستقرار درجة الحرارة
يقتضي إنتاجًا حراريًا يساوي $P$، فيكون خطيًا في $T_a$ دون \omterm{prop:b1:organism-environment:thermo}{المنطقة
الحرارية المحايدة}، بميل $-hS$. أما وجود المنطقة وعرضها، حيث يُعدَّل
$h$ نفسه (نفش الفرو، وتضييق أوعية الجلد)، فحقائق فيزيولوجية لا نتائج
للمعادلة.
\end{proof}

\section{معدل الاستقلاب وكتلة الجسم}

\begin{definition}[معدل الاستقلاب، التباين القياسي]\label{def:b1:organism-environment:allometry}
\emph{معدل الاستقلاب}\index{معدل الاستقلاب} $B$ عند \omterm{def:b1:organism-environment:organism}{كائن حي} هو
المعدل الذي يحول به الطاقة الكيميائية، مقيسًا بالواط إنتاجًا للحرارة
أو استهلاكًا للأكسجين ($\qty{1}{L}$ من $\mathrm{O_2}$ مستهلكًا
$\approx \qty{20}{kJ}$). و\emph{معدل الاستقلاب القاعدي}\index{معدل الاستقلاب القاعدي}
هو معدل داخلي حرارة مستريح صائم في منطقته الحرارية المحايدة. ويقال
إن مقدارًا $Y$ يتحجم \emph{تباينًا قياسيًا}\index{التباين القياسي}
مع كتلة الجسم $M$ حين يكون $Y = a\,M^{b}$ بأس $b \neq 1$؛ وعلى مخطط
لوغاريتمي مزدوج تكون العلاقة مستقيمًا ميله $b$.
\end{definition}

\begin{theorem}[قانون كلايبر]\label{thm:b1:organism-environment:kleiber}
\index{قانون كلايبر}
عبر الثدييات والطيور، من الزبابة إلى الحوت، يتحجم \omterm{def:b1:organism-environment:allometry}{معدل الاستقلاب
القاعدي} مثل القوة ثلاثة أرباع لكتلة الجسم:
\[
B \approx \qty{3.4}{W}\times\left(\frac{M}{\qty{1}{kg}}\right)^{3/4}.
\]
ومن ثم يهبط المعدل \emph{النوعي} $B/M$ مثل $M^{-1/4}$: فكل غرام من
فأر يحرق الطاقة أسرع بنحو عشرين مرة من كل غرام من فيل.
\end{theorem}
\begin{proof}[شواهد]
رسم كلايبر (1932) إنتاج الحرارة المقيس عند حيوانات من \qty{20}{g}
إلى \qty{600}{kg} بدلالة كتلها على محورين لوغاريتميين، فوجد مستقيمًا
ميله $0.74$، لا $0.67$ الذي كان فقد حرارة متناسب مع السطح ليتنبأ به.
وقد مُدَّ المستقيم منذ ذلك الحين على ثماني مراتب قدر في الكتلة
بالميل نفسه في حدود أجزاء من المئة. أما لماذا كان الأس $3/4$ لا
$2/3$ فما زال موضع جدل: والتفسير الأرجح يربطه بهندسة الشبكات
المتفرعة (الأوعية، والمسالك الهوائية) التي توصل الأكسجين إلى كل
خلية.
\end{proof}

\begin{method}[قراءة مخطط تباين قياسي]\label{met:b1:organism-environment:loglog}
\begin{enumerate}
  \item ارسم $\log Y$ بدلالة $\log M$ (أو استعمل محورين
        لوغاريتميين). فقانون القوة $Y = aM^b$ يصير المستقيم
        $\log Y = \log a + b\log M$.
  \item الأس $b$ هو الميل: خذ نقطتين تفصل بينهما مرتبة قدر في $M$؛
        فيكون $b$ عدد المراتب التي صعدها $Y$. وأس يساوي $1$ يعني
        التناسب (تحجمًا متساويًا)؛ و$b < 1$ يعني أن المقدار ينمو
        أبطأ من الكتلة؛ و$b > 1$ أسرع منها.
  \item المعامل $a$ هو قيمة $Y$ عند $M = 1$ (حيث $\log M = 0$).
  \item للمقارنة بين كائنين، استعمل النسبة: $Y_2/Y_1 =
        (M_2/M_1)^b$. وللواحدة من الكتلة يصير الأس $b-1$.
\end{enumerate}
\end{method}

\begin{example}[الفأر والفيل]\label{ex:b1:organism-environment:mouseelephant}
فأر كتلته \qty{20}{g}: $B = 3.4\times 0.02^{0.75} = \qty{0.18}{W}$،
أي \qty{9}{W/kg}. وفيل كتلته \qty{4}{t}: $B = 3.4\times 4000^{0.75} =
\qty{1700}{W}$، أي \qty{0.43}{W/kg}.
فالفيل يحرق أكثر بعشرة آلاف مرة من الفأر، لكن كل غرام من غراماته
يحرق أقل بإحدى وعشرين مرة: $(4000/0.02)^{-1/4} = 1/21$. ولا بد
للفأر أن يأكل عُشر كتلته كل يوم، وللفيل جزءًا من خمسين. وقلب الفأر
ينبض \qty{600}{} مرة في الدقيقة، وقلب الفيل \qty{30}{}: فالتواتر
القلبي، مثل $B/M$، يتحجم مثل $M^{-1/4}$، وقلبا الحيوانين ينبضان نحو
مليار نبضة في العمر.
\end{example}

\begin{remark}[أسّا تحجم، وما يتبعهما]\label{rem:b1:organism-environment:consequences}
كثير من معدلات \omterm{def:b1:organism-environment:organism}{الكائن الحي} تتحجم مثل $M^{-1/4}$ (التواتر القلبي،
وتواتر التنفس، والاستقلاب لواحدة الكتلة)، وكثير من أزمنته مثل
$M^{+1/4}$ (طول العمر، والحمل، وزمن بلوغ النضج)؛ فتكون جداءاتها،
مثل عدد النبضات في العمر، مستقلة عن الكتلة. أما قياسات الأعضاء
فتتحجم قريبًا من $M^{1}$ (القلب والرئة وحجم الدم كسور ثابتة من
الجسم)، بينما تتحجم السطوح التي تتبادل عبرها مثل $M^{3/4}$ --- ولهذا
كانت رئة حيوان كبير أدق طيًا من رئة حيوان صغير، لا أكبر منها فحسب.
\end{remark}

\section{كائنان نموذجان}

\begin{proposition}[ثديي ونبات زهري]\label{prop:b1:organism-environment:models}
الكائنان المتعددا الخلايا اللذان تتخذهما هذه السنة نموذجين يحلان
المشكلات السابقة بطريقتين متعاكستين. فالثديي \omterm{def:b1:organism-environment:trophy}{غير ذاتي التغذي} متنقل
ذو \omterm{def:b1:organism-environment:milieu}{وسط داخلي} منظَّم: يحصل على المادة والطاقة بأن يجد الغذاء
ويأكله، ويتبادل عبر سطوح داخلية (المعي، والرئة، والكلية) تمدها
دورة دموية، ويبقي حرارته وتركيب دمه ثابتين، ويبلغ شكلًا بالغًا
محددًا. أما النبات الزهري \omterm{def:b1:organism-environment:trophy}{فذاتي التغذي} ضوئيًا ثابت في مكانه ذو جسم
مفتوح مركّب من وحدات: يحصل على الطاقة والكربون من الضوء والهواء عند
أوراقه، وعلى الماء والأملاح من التربة عند جذوره، ويتبادل عبر سطوح
خارجية لا يكف عن توسيعها بالنمو طول حياته، ولا ينظم حرارته، ويكيّف
شكله مع موقعه. والفصلان التاليان يصفان كلًا منهما على حدة.
\end{proposition}

\begin{example}[الساعة نفسها، ميزانيتان]\label{ex:b1:organism-environment:budgets}
في ساعة الافتتاحية، حول الأرنب \qty{500}{kJ} من العشب إلى حرارة
وحركة وقليل من النمو، مبقيًا دمه عند \qty{39}{\celsius} وغلوكوزه
قرب \qty{6}{mmol/L}؛ وفعل ذلك بأن انتقل إلى الغذاء. وحولت البلوطة
بضع مئات من الكيلوجولات من ضوء الشمس الساقط على تاجها إلى سكر،
وفقدت \qty{5}{L} من الماء عبر أوراقها لتفعل ذلك، ونمت بضعة
ميكرومترات من الخشب الجديد؛ وفعلت ذلك بأن كانت حيث الضوء والماء.
وكلاهما \omterm{def:b1:organism-environment:organism}{نظام مفتوح}؛ وكلاهما يبقي نفسه بالتبادل؛ وأشكال تبادلاتهما
هي أشكال حياتهما.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:organism-environment:1}
اذكر أنواع التبادل الثلاثة بين \omterm{def:b1:organism-environment:organism}{كائن حي} ومحيطه، مع مثال على كل نوع
عند ثديي ومثال عند نبات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:organism-environment:2}
بيّن ماذا يحدث لنسبة السطح إلى الحجم حين يضاعف نصف قطر خلية كروية،
ولماذا يحد هذا من قياس الخلية.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:organism-environment:3}
عرّف \omterm{def:b1:organism-environment:homeostasis}{الاستتباب}، واذكر مكونات حلقة التغذية الراجعة السالبة الثلاثة،
وما يقابلها في تنظيم حرارة الجسم.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:organism-environment:4}
من شكل \omterm{def:b1:organism-environment:allometry}{معدل الاستقلاب}، بأي عامل يهبط معدل استقلاب السحلية بين
\qty{30}{\celsius} و\qty{10}{\celsius}، وبأي عامل يرتفع معدل الفأر
بين \qty{20}{\celsius} و\qty{0}{\celsius}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:organism-environment:5}
\omterm{def:b1:organism-environment:organism}{كائن حي} على شكل مكعب طول ضلعه \qty{1}{mm} يحتاج \qty{1}{nmol} من
الأكسجين في الثانية لكل مليمتر مكعب، ولا يستطيع سطحه أن يأخذ أكثر
من \qty{2}{nmol/s} لكل مليمتر مربع. فهل يستطيع العيش بالانتشار عبر
سطحه؟ أعد الحساب من أجل ضلع \qty{10}{mm}. ثم أوجد أكبر ضلع ممكن.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:organism-environment:6}
باستعمال \omterm{thm:b1:organism-environment:kleiber}{قانون كلايبر}، احسب \omterm{def:b1:organism-environment:allometry}{معدل الاستقلاب القاعدي} عند حصان كتلته
\qty{500}{kg} وعند زبابة كتلتها \qty{5}{g}، والمعدل النوعي عند كل
منهما. وكم مرة من كتلته يجب أن تأكل الزبابة من الغذاء يوميًا، عند
\qty{5}{kJ/g}، إذا كان إنفاقها اليومي ثلاثة أمثال معدلها القاعدي؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:organism-environment:7}
إنسان مستريح ينتج \qty{80}{W}. حوّل هذا إلى لترات أكسجين مستهلكة في
الدقيقة وإلى كيلوجولات في اليوم. وحمية من \qty{2000}{kcal} تقابل كم
واطًا؟ (\qty{1}{kcal} = \qty{4.18}{kJ}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:organism-environment:8}
رئة إنسان كتلته \qty{70}{kg} لها سطح تبادل يناهز \qty{100}{m^2}.
فلو تحجم سطح الرئة مع كتلة الجسم مرفوعة إلى القوة $3/4$، فما السطح
الذي ستكون عليه رئة فأر كتلته \qty{20}{g}، ورئة فيل كتلته
\qty{4000}{kg}؟ قارن مع سطح كرة حجمها حجم الحيوان (بكثافة
\qty{1000}{kg/m^3}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:organism-environment:9}
لا تستطيع حلقة تغذية راجعة سالبة أن تبقي متغيرًا عند قيمته المرجعية
تمامًا. فسّر لماذا، مستعينًا بحلقة الحرارة، وقل ما الذي يحدد عرض
النطاق الذي يتذبذب فيه المتغير.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:organism-environment:10}
جسم يفقد الحرارة بالمعدل $P = hS(T_b - T_a)$ حيث
$h = \qty{5}{W.m^{-2}.K^{-1}}$ عند ثديي ذي فرو. من أجل فأر كتلته
\qty{20}{g} يُنمذج بكرة كثافتها \qty{1000}{kg/m^3}، احسب السطح،
وفقد الحرارة عند $T_a = \qty{0}{\celsius}$ مع $T_b =
\qty{37}{\celsius}$، وقارنه بمعدله القاعدي من \omterm{thm:b1:organism-environment:kleiber}{قانون كلايبر}.
فما الذي يجب على الفأر أن يفعله؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:organism-environment:11}
لو تحجم فقد الحرارة مع السطح ($M^{2/3}$) وتحجم إنتاجها مع \omterm{thm:b1:organism-environment:kleiber}{قانون
كلايبر} ($M^{3/4}$)، فبيّن أن أصغر داخليات الحرارة هي أشدها معاناة
في البرد، وأن الحيوان فوق كتلة ما يصعب عليه طرح الحرارة. ثم قدّر،
مع $h = \qty{5}{W.m^{-2}.K^{-1}}$ و$T_b - T_a = \qty{20}{K}$،
الكتلة التي يتساوى عندها الإنتاج القاعدي والفقد من أجل جسم كروي.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:organism-environment:12}
«لأن النبات لا ينظم حرارته ولا سوائله الداخلية، فهو نظام أبسط من
الحيوان.» ناقش هذا في فقرة، مستعملًا استراتيجيتي التبادل في هذا
الفصل، والسطوح التي يستعملها كل منهما، ومعنى \omterm{def:b1:organism-environment:homeostasis}{الاستتباب}.
\end{exercise}

\section{مسألة: الفأر والفيل}

\begin{problem}[{مسألة نهاية الأسبوع --- ثدييان يفصل بينهما خمس مراتب
قدر: سطوحهما، وحرارتهما، وغذاؤهما، وقلباهما، وسنوهما، وصولًا إلى
النسبة التي يتنبأ بها قانون كلايبر}]
\label{pb:b1:organism-environment:1}
فأر منزلي كتلته \qty{20}{g}، وفيل أفريقي كتلته \qty{4000}{kg}. وكلاهما
يبقي حرارة جسمه قرب \qty{37}{\celsius}، وكلاهما يُنمذج، حين يلزم
شكل، بكرة كثافتها \qty{1000}{kg/m^3}. \omterm{thm:b1:organism-environment:kleiber}{وقانون كلايبر} هو
$B = \qty{3.4}{W}\,(M/\qty{1}{kg})^{3/4}$.

\medskip
\noindent\textbf{الجزء الأول --- الهندسة.}
\begin{enumerate}
  \item احسب حجم كل حيوان.
  \item احسب نصف قطر الكرة المكافئة لكل منهما.
  \item احسب سطح كل كرة.
  \item احسب نسبة السطح إلى الحجم عند كل منهما، ونسبتهما.
  \item بيّن أن هذه النسبة تساوي الجذر التكعيبي لنسبة الكتلتين،
        وتحقق من ذلك عدديًا.
  \item الفئران والفيلة الحقيقية ليست كرات. قل، معللًا، هل نسبة
        السطح إلى الحجم الحقيقية عند كل منهما أكبر أم أصغر من نسبة
        الكرة، وأي الحيوانين أبعد عنها.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الحرارة.}
تُفقد الحرارة عبر السطح بالمعدل $P = hS\,(T_b - T_a)$. خذ
$h = \qty{10}{W.m^{-2}.K^{-1}}$ من أجل جلد عار و$T_a =
\qty{17}{\celsius}$.
\begin{enumerate}[resume]
  \item احسب فقد الحرارة عند كل كرة عارية.
  \item احسب \omterm{def:b1:organism-environment:allometry}{معدل الاستقلاب القاعدي} عند كل منهما من \omterm{thm:b1:organism-environment:kleiber}{قانون كلايبر}.
  \item قارن الفقد والإنتاج عند الفيل. فهل الجلد العاري مشكلة له؟
  \item قارن الفقد والإنتاج عند الفأر. وما قيمة $h$ التي يجب أن
        يوفرها فروه ليغطي الإنتاج القاعدي الفقد؟
  \item أذنا الفيل تضيفان \qty{4}{m^2} من سطح رقيق غزير التروية،
        وهو يخفق بهما في الحر. فسّر ما فائدتهما، مستعينًا بالجزء
        الأول.
  \item فسّر لماذا لا يوجد داخلي حرارة أصغر من بضعة غرامات، ولماذا
        تترك أصغرها (الزبابات، والطنانات) حرارتها تهبط ليلًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الغذاء.}
حيوان يعيش حرًا ينفق نحو مثلي معدله القاعدي في اليوم. وبذور الفأر
تعطي \qty{18}{kJ} لكل غرام، وعشب الفيل الطري \qty{5}{kJ} من الطاقة
القابلة للهضم لكل غرام.
\begin{enumerate}[resume]
  \item احسب الإنفاق الطاقي اليومي عند كل منهما بالكيلوجول.
  \item احسب المدخول الغذائي اليومي عند كل منهما بالغرام.
  \item عبّر عن كل مدخول بكسر من كتلة الجسم، وقارن.
  \item معي الفيل طوله نحو \qty{35}{m} ومعي الفأر \qty{40}{cm}.
        احسب نسبة طولي المعيين وقارنها بنسبة طولي الجسمين (نصفا
        القطر في الجزء الأول). فماذا توحي المقارنة عن تحجم المعي؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- القلوب والسنون.}
يتحجم التواتر القلبي مثل $f = f_0\,(M/\qty{1}{kg})^{-1/4}$، وطول
العمر مثل $\tau = \tau_0\,(M/\qty{1}{kg})^{1/4}$. وقلب الفأر ينبض
\qty{600}{} مرة في الدقيقة وهو يعيش سنتين.
\begin{enumerate}[resume]
  \item احسب $f_0$ و$\tau_0$ من الفأر.
  \item تنبأ بالتواتر القلبي وطول العمر عند الفيل.
  \item احسب عدد النبضات القلبية في عمر كل حيوان.
  \item فسّر، انطلاقًا من الأسّين، لماذا كان هذا العدد واحدًا.
  \item تنبأ بالتواتر القلبي وطول العمر عند إنسان كتلته
        \qty{70}{kg} من الصيغتين نفسيهما، وقارن بالقيمتين
        الحقيقيتين (\qty{70}{} نبضة في الدقيقة، ونحو \qty{80}{}
        سنة). فأي التنبؤين يفشل، وما السبب المحتمل؟
  \item يتحجم تواتر التنفس مثل التواتر القلبي. والفأر يتنفس
        \qty{160}{} مرة في الدقيقة؛ تنبأ بتواتر تنفس الفيل، واحسب
        عدد النبضات القلبية لكل نفَس عند كل منهما.
  \item اكتب \omterm{def:b1:organism-environment:allometry}{معدل الاستقلاب} لواحدة الكتلة $B/M$ قوةً من $M$ وأعطِ
        أسه.
  \item احسب $B/M$ عند الفأر وعند الفيل.
  \item اذكر النتيجة: نسبة معدل استقلاب الفأر لواحدة الكتلة إلى
        نظيره عند الفيل، وقوة نسبة الكتلتين التي تساويها.
\end{enumerate}
\end{problem}
