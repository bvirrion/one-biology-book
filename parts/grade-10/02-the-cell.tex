\chapter{The Cell}\label{ch:g10:the-cell}

Every free-living \omterm{def:g10:scientific-biology:organism}{organism} is built from \omterm{def:g10:the-cell:cell}{cells} --- the smallest units
that still show the full set of life's properties. This chapter maps
\omterm{prop:g10:scientific-biology:cell-unit}{cell theory}, contrasts \omterm{def:g10:the-cell:prok-euk}{prokaryotes} with \omterm{def:g10:the-cell:prok-euk}{eukaryotes}, tours the main
\omterm{def:g10:the-cell:prok-euk}{organelles}, and links shape to the surface-to-volume constraint that
keeps \omterm{def:g10:the-cell:cell}{cells} small.

\section{Cell theory}

\begin{definition}[Cell]\label{def:g10:the-cell:cell}
A \emph{cell}\index{cell} is a membrane-bounded unit of cytoplasm
containing genetic material and the machinery to maintain and (usually)
reproduce itself. The plasma membrane separates the cell's interior
from the surroundings while controlling what enters and leaves.
\end{definition}

\begin{proposition}[Cell theory]\label{prop:g10:the-cell:cell-theory}
\begin{enumerate}
  \item All free-living \omterm{def:g10:scientific-biology:organism}{organisms} are composed of one or more \omterm{def:g10:the-cell:cell}{cells}.
  \item The \omterm{def:g10:the-cell:cell}{cell} is the basic unit of structure and function in
        \omterm{def:g10:scientific-biology:organism}{organisms}.
  \item All \omterm{def:g10:the-cell:cell}{cells} arise from pre-existing \omterm{def:g10:the-cell:cell}{cells} by division.
\end{enumerate}
\end{proposition}
\admitted

\begin{remark}[Historical note]\label{rem:g10:the-cell:history}
Hooke's cork ``\omterm{def:g10:the-cell:cell}{cells}'' (1665) were empty walls; later microscopy and
the work of Schleiden, Schwann and Virchow turned the metaphor into
\omterm{def:g10:scientific-biology:theory}{theory}. Viruses remain outside the free-living clause of the \omterm{def:g10:scientific-biology:theory}{theory}
(\cref{ch:g10:scientific-biology}).
\end{remark}

\section{Prokaryotic and eukaryotic cells}

\begin{definition}[Prokaryote and eukaryote]\label{def:g10:the-cell:prok-euk}
A \emph{prokaryotic cell}\index{prokaryote}\index{cell!prokaryotic}
lacks a membrane-bounded nucleus; its DNA lies in a nucleoid region.
Bacteria and archaea are prokaryotes. A \emph{eukaryotic
cell}\index{eukaryote}\index{cell!eukaryotic} has a true
\emph{nucleus}\index{nucleus (cell)} enclosed by a \omterm{def:g10:the-cell:nucleus}{nuclear envelope},
and typically many membrane-bounded \emph{organelles}\index{organelle}.
Animals, plants, fungi and protists are eukaryotes.
\end{definition}

\begin{proposition}[Shared features of all cells]\label{prop:g10:the-cell:shared}
All \omterm{def:g10:the-cell:cell}{cells} have: a plasma membrane; cytoplasm (cytosol plus structures);
\omterm{def:g10:the-cell:support}{ribosomes} for protein synthesis; and DNA as genetic material. Size and
internal complexity differ enormously.
\end{proposition}
\admitted

\begin{example}[Scale]\label{ex:g10:the-cell:scale}
A typical bacterium is about \qty{1}{\micro m}--\qty{5}{\micro m} long;
many animal \omterm{def:g10:the-cell:cell}{cells} are \qty{10}{\micro m}--\qty{30}{\micro m} across; a
human egg \omterm{def:g10:the-cell:cell}{cell} is about \qty{100}{\micro m}. Light microscopes resolve
down to roughly \qty{0.2}{\micro m}; smaller detail needs electron
microscopy.
\end{example}

\section{Organelles of eukaryotic cells}

\begin{definition}[Nucleus]\label{def:g10:the-cell:nucleus}
The \emph{nucleus}\index{nucleus (cell)} houses most of the \omterm{def:g10:the-cell:cell}{cell}'s DNA,
organised with proteins as \emph{chromatin}\index{chromatin}. The
\emph{nuclear envelope}\index{nuclear envelope} is a double membrane
with pores that control traffic of RNA and proteins. The
\emph{nucleolus}\index{nucleolus} assembles ribosomal subunits.
\end{definition}

\begin{definition}[Endomembrane system]\label{def:g10:the-cell:endomembrane}
The \emph{endoplasmic reticulum}\index{endoplasmic reticulum} (ER) is a
membrane network: rough ER (studded with \omterm{def:g10:the-cell:support}{ribosomes}) helps make and fold
proteins for secretion or membranes; smooth ER handles lipids and
detoxification. The \emph{Golgi
apparatus}\index{Golgi apparatus} modifies, sorts and ships proteins and
lipids in vesicles. \emph{Lysosomes}\index{lysosome} (especially in
animal \omterm{def:g10:the-cell:cell}{cells}) digest macromolecules with hydrolytic enzymes.
\emph{Vacuoles}\index{vacuole} store materials; plant \omterm{def:g10:the-cell:cell}{cells} often have a
large central vacuole that maintains turgor.
\end{definition}

\begin{definition}[Mitochondria and chloroplasts]\label{def:g10:the-cell:energy-organelles}
\emph{Mitochondria}\index{mitochondrion} carry out most of aerobic
cellular respiration, producing ATP
(\cref{ch:g10:cellular-respiration}). \emph{Chloroplasts}\index{chloroplast}
in plants and algae perform photosynthesis
(\cref{ch:g10:photosynthesis}). Both have double membranes and their own
DNA --- a clue to their ancient bacterial origins (endosymbiosis).
\end{definition}

\begin{definition}[Cytoskeleton, wall and junctions]\label{def:g10:the-cell:support}
The \emph{cytoskeleton}\index{cytoskeleton} of protein filaments shapes
the \omterm{def:g10:the-cell:cell}{cell}, moves \omterm{def:g10:the-cell:prok-euk}{organelles} and drives some motility.
\emph{Ribosomes}\index{ribosome} (not membrane-bounded) synthesise
proteins. Plant \omterm{def:g10:the-cell:cell}{cells} have a rigid \emph{cell wall}\index{cell wall} of
cellulose outside the plasma membrane; fungal walls use chitin; many
bacteria have peptidoglycan walls. Animal \omterm{def:g10:the-cell:cell}{cells} lack walls but may form
junctions with neighbours.
\end{definition}

\begin{example}[Division of labour]\label{ex:g10:the-cell:labour}
A pancreatic \omterm{def:g10:the-cell:cell}{cell} secreting digestive enzymes is rich in rough ER and
Golgi. A muscle \omterm{def:g10:the-cell:cell}{cell} is packed with mitochondria. A leaf mesophyll \omterm{def:g10:the-cell:cell}{cell}
is packed with \omterm{def:g10:the-cell:energy-organelles}{chloroplasts}. Structure matches job.
\end{example}

\section{Surface area, volume and size}

\begin{definition}[Surface-to-volume ratio]\label{def:g10:the-cell:sa-v}
As a \omterm{def:g10:the-cell:cell}{cell} grows, its \emph{volume}\index{volume (cell)} (demand for
nutrients and waste removal) increases faster than its
\emph{surface area}\index{surface area (cell)} (membrane available for
exchange). The \emph{surface-to-volume ratio}\index{surface-to-volume
ratio} therefore falls with size.
\end{definition}

\begin{proposition}[Why cells stay small]\label{prop:g10:the-cell:size}
For a cube of side $L$, \omterm{def:g10:the-cell:sa-v}{surface area} $S = 6L^2$ and \omterm{def:g10:the-cell:sa-v}{volume} $V = L^3$, so
$S/V = 6/L$. Larger $L$ means lower $S/V$: exchange across the membrane
cannot keep up with the interior's needs. \omterm{def:g10:the-cell:cell}{Cells} stay small, or become
flat, folded or multinucleate, to keep exchange efficient.
\end{proposition}

\begin{proof}
Direct calculation for the cube: $S/V = 6L^2/L^3 = 6/L$. The same
scaling --- area $\propto$ length$^2$, \omterm{def:g10:the-cell:sa-v}{volume} $\propto$ length$^3$ ---
holds for spheres and similar shapes.
\end{proof}

\section{Seeing cells: the light microscope}

\begin{definition}[Magnification and resolution]\label{def:g10:the-cell:microscope}
\emph{Magnification}\index{magnification} is how many times larger the
image is than the object. \emph{Resolution}\index{resolution
(microscope)} is the smallest distance at which two points are still
seen as separate. High magnification is useless without sufficient
resolution.
\end{definition}

\begin{method}[Using a compound light microscope]\label{met:g10:the-cell:scope}
\begin{enumerate}
  \item Start on the lowest power objective; centre the specimen.
  \item Focus with the coarse knob only on low power; then use fine
        focus.
  \item Increase power stepwise; only fine focus at high power.
  \item Adjust light (diaphragm) for contrast; never use the coarse
        knob on high power (risk of slide damage).
\end{enumerate}
\end{method}

\section{Problem: Building a cell map}

\begin{problem}[{Weekend problem --- annotate, compare and argue:
from a blank outline of plant and animal cells to a reasoned account of
specialisation}]
\label{pb:g10:the-cell:1}
You are given outline diagrams of a typical animal \omterm{def:g10:the-cell:cell}{cell} and a typical
plant mesophyll \omterm{def:g10:the-cell:cell}{cell} (or you draw them).

\medskip
\noindent\textbf{Part I --- Labelling.}
\begin{enumerate}
  \item Label at least eight structures on each diagram where present.
  \item Mark with an asterisk (*) every structure involved directly in
        protein secretion (ER, Golgi, vesicles, membrane).
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Comparison.}
\begin{enumerate}
  \item Construct a Venn diagram of plant-only, animal-only and shared
        features at the \omterm{def:g10:the-cell:prok-euk}{organelle} level.
  \item For each unique plant structure, state its functional payoff.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Specialisation.}
\begin{enumerate}
  \item Predict relative abundance of mitochondria in: skin fibroblast,
        cardiac muscle \omterm{def:g10:the-cell:cell}{cell}, mature red blood \omterm{def:g10:the-cell:cell}{cell} (mammal).
  \item Predict relative abundance of rough ER in: plasma B \omterm{def:g10:the-cell:cell}{cell}
        secreting antibodies versus a fat-storage \omterm{def:g10:the-cell:cell}{cell}.
  \item Write a short paragraph linking structure to function for one
        of these examples.
\end{enumerate}
\end{problem}
