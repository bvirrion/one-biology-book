\chapter{Born, Growing, Dying}\label{ch:g2:born-grow-die}

In the spring, the cherry tree by the gate is covered in blossom;
under it, a cat nurses her newborn kittens; and Grandpa shows a
photograph of himself when he was your age --- small, gap-toothed,
and already fond of cherries. Every \omterm{def:g1:living-or-not:living}{living thing} travels the same
road: it begins, it grows, and one day it ends. That road is called
the \omterm{def:g2:born-grow-die:cycle}{life cycle}.

\section{The stages of a life}

\begin{definition}[Life cycle]\label{def:g2:born-grow-die:cycle}
The \emph{life cycle}\index{life cycle} of a \omterm{def:g1:living-or-not:living}{living thing} is the
row of \emph{stages}\index{stage} its life goes through:
\begin{enumerate}
  \item it is \emph{born}\index{birth} --- the beginning;
  \item it is \emph{young}\index{young} --- it grows and changes
        fast;
  \item it is an \emph{adult}\index{adult} --- fully grown; adults
        can have young of their own;
  \item it grows \emph{old}\index{old age}, and at the end it
        \emph{dies}\index{death}.
\end{enumerate}
\end{definition}

\begin{example}[Three lives, same road]\label{ex:g2:born-grow-die:three}
A kitten is born, plays and grows into a cat, who may have kittens
herself, and grows old. A cherry seed sprouts, the young tree
grows taller for years, the \omterm{def:g2:born-grow-die:cycle}{adult} tree makes blossom and cherries,
and after many tens of years it dies. And you: baby, child, then
one day a grown-up. Kitten, cherry tree, child --- the same four
\omterm{def:g2:born-grow-die:cycle}{stages}.
\end{example}

\begin{remark}[Why it is called a cycle]\label{rem:g2:born-grow-die:wheel}
Each single life runs from \omterm{def:g2:born-grow-die:cycle}{birth} to death, like a road with two
ends. But before the end, \omterm{def:g2:born-grow-die:cycle}{adults} have young, and those young start
the road again. Seen over many generations, life turns like a
wheel --- that is why we say life \emph{cycle}.
\end{remark}

\section{Growing and changing}

\begin{proposition}[Growing up is more than getting bigger]\label{prop:g2:born-grow-die:change}
While it grows, a young \omterm{def:g1:living-or-not:living}{living thing} also \emph{changes}: a baby
gets teeth, loses them, and gets stronger ones; a chick's yellow
fluff becomes real \omterm{def:g1:animals-around-us:covering}{feathers}; a young tree's soft green \omterm{def:g1:plants-around-us:parts}{stem} turns
to hard wood. Growing is getting bigger \emph{and} becoming
different.
\end{proposition}
\admitted

\begin{example}[The photograph test]\label{ex:g2:born-grow-die:photo}
On Grandpa's old photographs you can follow one person through
every \omterm{def:g2:born-grow-die:cycle}{stage}: baby, schoolchild, young man, father, grandfather.
Same person, changing body. Photographs are little time machines
for watching a \omterm{def:g2:born-grow-die:cycle}{life cycle}.
\end{example}

\begin{method}[Putting life stages in order]\label{met:g2:born-grow-die:order}
Given pictures of a \omterm{def:g1:living-or-not:living}{living thing}'s life, mixed up:
\begin{enumerate}
  \item find the \omterm{def:g2:born-grow-die:cycle}{birth} picture --- egg, seed or newborn baby;
  \item find the \omterm{def:g2:born-grow-die:cycle}{adult} --- fully grown, able to have young;
  \item place the young in between, from smaller to bigger;
  \item check your row: does each picture lead to the next?
\end{enumerate}
\end{method}

\section{Every life ends --- and life goes on}

\begin{proposition}[Lifespans]\label{prop:g2:born-grow-die:lifespan}
Different \omterm{def:g1:living-or-not:living}{living things} live different lengths of time --- a
\emph{lifespan}\index{lifespan}. A butterfly may live a few weeks;
a cat, and a dog, ten to twenty years; a person often more than
eighty years; some trees, hundreds of years. Long or short, every
life has an end.
\end{proposition}
\admitted

\begin{example}[The old apple tree]\label{ex:g2:born-grow-die:oldtree}
The bent old apple tree at the end of the garden gives fewer
apples each year --- it is in its \omterm{def:g2:born-grow-die:cycle}{old age}. But its apples have
pips, and one pip, planted, is already a young apple tree
beginning the road. When the old tree dies, apple trees will not
be gone: its young carry on.
\end{example}

\begin{remark}[Talking about death]\label{rem:g2:born-grow-die:death}
Dying belongs to every \omterm{def:g2:born-grow-die:cycle}{life cycle}, and it is all right to feel sad
about it --- people do, especially about pets and about people we
love. \omterm{def:g1:living-or-not:living}{Biology} adds one comfort: because \omterm{def:g2:born-grow-die:cycle}{adults} have young before
they die, life itself continues, cycle after cycle.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:1}
Name the four \omterm{def:g2:born-grow-die:cycle}{stages} of a \omterm{def:g2:born-grow-die:cycle}{life cycle}, in order.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:2}
At which \omterm{def:g2:born-grow-die:cycle}{stage} can a \omterm{def:g1:living-or-not:living}{living thing} have young of its own?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:3}
Put in order, oldest to youngest: a schoolchild, a baby, a
grandmother, a mother.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:4}
Give the \omterm{def:g2:born-grow-die:cycle}{life cycle} of the cherry tree of
\cref{ex:g2:born-grow-die:three}, \omterm{def:g2:born-grow-die:cycle}{stage} by \omterm{def:g2:born-grow-die:cycle}{stage}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:5}
Give two examples from \cref{prop:g2:born-grow-die:change} of a
young \omterm{def:g1:living-or-not:living}{living thing} \emph{changing} while it grows, not just
getting bigger.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:6}
Who usually lives longer: a butterfly, a cat, or an oak tree? Put
the three in order of \omterm{prop:g2:born-grow-die:lifespan}{lifespan}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:7}
Why is the word \emph{cycle} used, when each single life runs
from \omterm{def:g2:born-grow-die:cycle}{birth} to death like a road? Use the figure's purple arrow.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:born-grow-die:8}
Try it: ask your family for photographs of one grown-up at
different ages, and put them in life-cycle order with
\cref{met:g2:born-grow-die:order}. Which \omterm{def:g2:born-grow-die:cycle}{stages} do the photographs
show? Which \omterm{def:g2:born-grow-die:cycle}{stage} is still to come?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:born-grow-die:9}
The old apple tree of \cref{ex:g2:born-grow-die:oldtree} will die
one day, yet the garden need not lose its apple trees. Explain
why, with the word \emph{generation} or the purple arrow of the
figure.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:born-grow-die:10}
A pebble never dies. Is that because it is very strong? Use
\cref{ch:g1:living-or-not}'s idea of living and non-living to
explain.
\end{exercise}
