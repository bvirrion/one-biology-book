\chapter{Pertukaran Gas dan Pengangkutan Getah Tumbuhan}\label{ch:b1:plant-transport}

Di puncak sebatang sekuoia seratus meter di atas tanah, air sedang
ditarik keluar dari tanahnya oleh penguapan dari \omterm{def:b1:flowering-plant-organization:organs}{daunnya}, menaiki
sebuah kolom \omterm{def:b1:cell-unit-of-life:cell}{sel} mati yang tak lebih lebar daripada sehelai rambut, di
bawah sebuah tegangan yang akan memutuskan sebuah kawat baja
berpenampang sama. Tak ada pompa; airnya diseret oleh \omterm{def:b1:water-small-molecules:water}{ikatan hidrogen}
yang sama yang membuatnya melekat pada dirinya sendiri. Sementara itu,
dalam seperangkat tabung kedua di sebelah yang pertama, gula yang
dibuat \omterm{def:b1:flowering-plant-organization:organs}{daunnya} mengalir ke arah yang lain, didorong tekanan yang
dibangkitkan \omterm{def:b1:flowering-plant-organization:organs}{daunnya} sendiri. Kedua sistem perpipaan sebatang
tumbuhan, \omterm{def:b1:flowering-plant-organization:tissues}{xilem} dan \omterm{def:b1:plant-transport:phloem}{floemnya}, dipacu oleh fisika yang disiapkan
tumbuhannya lalu dibiarkannya berjalan. Bab ini memerikan \omterm{def:b1:flowering-plant-organization:tissues}{stomata} yang
lewatnya sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} menukar air dengan karbon dioksida, mekanisme
kohesi--tegangan yang mengangkat getahnya, pembuluh yang membawanya,
\omterm{thm:b1:plant-transport:pressureflow}{mekanisme alir tekanan} yang menggerakkan gulanya, serta kompromi
antara minum dan makan yang menguasai keseluruhannya.

\section{Stomata: gerbangnya}

\begin{definition}[Stoma, sel penjaga]\label{def:b1:plant-transport:stoma}
\index{stoma}\index{sel penjaga}
Yang disebut \emph{stoma} adalah sebuah liang dalam \omterm{def:b1:flowering-plant-organization:tissues}{epidermis} sehelai
\omterm{def:b1:flowering-plant-organization:organs}{daun} (atau sebatang \omterm{def:b1:flowering-plant-organization:organs}{batang} hijau) yang dibatasi dua \emph{\omterm{def:b1:cell-unit-of-life:cell}{sel}
penjaga}, yaitu \omterm{def:b1:cell-unit-of-life:cell}{sel} berbentuk ginjal yang dindingnya menebal pada sisi
yang menghadap liangnya dan yang mikrofibril \omterm{def:b1:carbohydrates:polysaccharide}{selulosanya} berjalan
mengelilinginya seperti simpai. Ketika \omterm{def:b1:cell-unit-of-life:cell}{sel} penjaganya menyerap air
lalu menggembung, simpainya mencegahnya melebar lalu memaksanya
melengkung berjauhan: liangnya membuka. Ketika keduanya kehilangan
air, liangnya menutup. Sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} membawa beberapa ratus \omterm{def:b1:flowering-plant-organization:tissues}{stomata}
tiap milimeter persegi, sebagian besarnya pada permukaan bawahnya;
ketika terbuka, liangnya adalah satu atau dua persen luas \omterm{def:b1:flowering-plant-organization:organs}{daunnya} dan
hampir seluruh pertukaran gasnya lewat di situ.
\end{definition}

\begin{proposition}[Bagaimana sebuah stoma membuka, dan kapan]\label{prop:b1:plant-transport:opening}
\index{asam absisat}
\omterm{def:b1:plant-transport:stoma}{Sel penjaganya} membuka liangnya dengan memompa proton keluar
(\cref{ch:b1:plant-water-minerals}), yang memacu kalium dan klorida
masuk lalu membuat malat di dalamnya; zat terlarutnya menurunkan
\omterm{def:b1:membranes-transport:osmosis}{potensial airnya}, airnya mengikut dari \omterm{def:b1:cell-unit-of-life:cell}{sel} tetangganya, dan \omterm{def:b1:eukaryotic-cell:plantcell}{turgornya}
naik semegapascal. Membalikkan fluks ionnya menutup liangnya dalam
beberapa menit. Isyaratnya: \emph{cahaya} (cahaya biru langsung pada
\omterm{def:b1:plant-transport:stoma}{sel penjaganya}, dan turunnya $\mathrm{CO_2}$ dalamnya seiring
fotosintesisnya mulai) membuka; \emph{$\mathrm{CO_2}$ dalam yang
tinggi} menutup; \emph{cekaman air} menutup, lewat hormon \emph{\omterm{prop:b1:plant-transport:opening}{asam
absisat}}, yang dibuat dalam \omterm{def:b1:flowering-plant-organization:organs}{akar} dan \omterm{def:b1:flowering-plant-organization:organs}{daun} yang layu, yang memicu
kehilangan ion dari \omterm{def:b1:plant-transport:stoma}{sel penjaganya} dalam beberapa menit; udara kering
menutup. \omterm{def:b1:flowering-plant-organization:organs}{Daunnya} membuka gerbangnya ketika karbonnya sepadan dengan
airnya lalu menutupnya ketika airnya langka --- dan sebagian besar
tumbuhan berbuat demikian dalam sebuah irama harian, membuka pada
fajar dan menutup pada senja, yang bertahan berhari-hari dalam cahaya
yang tetap.
\end{proposition}

\section{Mengangkat air: kohesi dan tegangan}

\begin{theorem}[Mekanisme kohesi--tegangan]\label{thm:b1:plant-transport:cohesiontension}
\index{teori kohesi--tegangan}\index{transpirasi}
Air naik dalam \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} sebab ia \emph{ditarik} dari atas, bukan
didorong dari bawah. Penguapan dari dinding basah \omterm{def:b1:cell-unit-of-life:cell}{sel} mesofil \omterm{def:b1:flowering-plant-organization:organs}{daunnya}
ke dalam ruang udaranya (\emph{transpirasi}) menarik air dalam pori
halus dindingnya menjadi meniskus melengkung, yang tegangan
permukaannya menaruh air di belakangnya di bawah tegangan; tegangannya
diteruskan, lewat kolom air yang sinambung dalam \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} yang
tertahan bersama oleh kohesi (\omterm{def:b1:water-small-molecules:water}{ikatan hidrogen} pada
\cref{ch:b1:water-small-molecules}), menuruni \omterm{def:b1:flowering-plant-organization:organs}{batangnya} dan masuk ke
\omterm{def:b1:flowering-plant-organization:organs}{akarnya}, lalu mengangkat air dari tanahnya. Untuk menahan sebuah kolom
setinggi $h$ lawan gravitasinya dituntut sebuah tegangan $\rho g h$ ---
\qty{1}{MPa} tiap seratus meter --- ditambah apa yang dibutuhkan untuk
mengatasi gesekan alirannya, kira-kira sebanyak itu lagi; kolom
sebatang pohon tinggi berada pada \qtyrange{-2}{-3}{MPa}, masih di
dalam kekuatan tarik air dalam sebuah tabung halus.
\end{theorem}
\begin{proof}[Bukti]
Dixon dan Joly (1894) memperlihatkan bahwa air dalam sebuah tabung
kaca tersegel menahan tegangan beberapa megapascal tanpa putus, dan
bahwa sebuah cabang berdaun yang bertranspirasi, yang disegel pada
sebuah tabung air yang tercelup dalam raksa, mengangkat raksanya lebih
tinggi daripada yang sanggup dilakukan tekanan atmosfernya.
\emph{Bilik tekanan} Scholander (1965) mengukur tegangannya secara
langsung: sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} yang dipotong disegel ke dalam sebuah bilik
dengan tangkainya menjulur, dan tekanan gas yang tepat memaksa
getahnya kembali ke permukaan potongnya setara dengan tegangan yang
tadinya menahan getahnya --- \qtyrange{0.5}{1}{MPa} pada sebuah
tanaman yang tersiram baik pada tengah hari, \qtyrange{2}{4}{MPa} pada
sebuah semak gurun atau tajuk sebatang pohon tinggi. Sebuah
dendrometer memperlihatkan \omterm{def:b1:flowering-plant-organization:organs}{batang} sebatang pohon menyusut pada siang
hari, seiring tegangannya memeras pembuluhnya, lalu menggembung pada
malam hari; sebuah termometer dalam kayunya memperlihatkan getahnya
mengalir paling cepat ketika transpirasinya paling tinggi, lalu
berhenti pada malam hari. Tak ada \omterm{def:b1:cell-unit-of-life:cell}{sel} hidup yang dituntut: sebuah
\omterm{def:b1:flowering-plant-organization:organs}{batang} yang dimatikan dengan racun atau panas tetap menghantarkan.
\end{proof}

\begin{definition}[Saluran xilem, kavitasi]\label{def:b1:plant-transport:xylem}
\index{pembuluh xilem}\index{kavitasi}
Air menempuh perjalanan dalam \emph{\omterm{def:b1:flowering-plant-organization:tissues}{trakeid}} dan \emph{pembuluh}
\omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} (\cref{ch:b1:flowering-plant-organization}): \omterm{def:b1:cell-unit-of-life:cell}{sel} mati yang
isinya telah dikosongkan, dindingnya yang berlignin menebal dalam
cincin dan pilin yang menjaganya dari runtuh di bawah tegangan di
dalamnya, tersambung ujung dengan ujung lewat perforasi (pembuluh)
atau lewat noktah pada dinding sisinya (\omterm{def:b1:flowering-plant-organization:tissues}{trakeid}). Sebuah pembuluh
dapat selebar \qtyrange{20}{300}{\micro m} dan sepanjang beberapa
sentimeter sampai beberapa meter: tabung lebar membawa jauh lebih
banyak (alir lewat sebuah tabung naik sebagai pangkat empat jejarinya)
tetapi lebih rentan terhadap \emph{kavitasi} --- terbentuknya sebuah
gelembung gas secara mendadak dalam air yang tegang, yang mengosongkan
salurannya lalu menaruhnya di luar layanan. Noktah antarsaluran
menghentikan gelembungnya menyebar; pembekuan dan kekeringan
sama-sama menyebabkan kavitasi, dan pembuluh sebatang pohon sebagian
terisi kembali oleh \omterm{prop:b1:plant-water-minerals:rootpressure}{tekanan akar} pada musim semi atau diganti kayu
baru tiap tahun.
\end{definition}

\begin{example}[Puncak pohonnya]\label{ex:b1:plant-transport:top}
Pada \qty{100}{m} bobot kolomnya saja menuntut \qty{1}{MPa} tegangan;
gesekan dalam pembuluhnya menambahkan sebanyak itu lagi pada siang
hari, sehingga \omterm{def:b1:flowering-plant-organization:tissues}{xilem} pada tajuknya dekat \qty{-2}{MPa} dan \omterm{def:b1:cell-unit-of-life:cell}{sel}
\omterm{def:b1:flowering-plant-organization:organs}{daunnya}, untuk menarik air darinya, lebih rendah daripada itu:
\omterm{def:b1:flowering-plant-organization:tissues}{stomatanya} harus menutup lebih awal pada harinya dan fotosintesisnya
lebih lambat daripada fotosintesis sebuah cabang rendah, dan \omterm{def:b1:flowering-plant-organization:organs}{daun} di
puncaknya kecil dan tebal. Tinggi pohon yang tertinggi, sekitar
\qty{120}{m}, agaknya ditetapkan oleh hal ini: di atasnya, \omterm{def:b1:flowering-plant-organization:organs}{daunnya} tak
sanggup menahan tegangannya tanpa berkavitasi pada sebuah petang yang
kering.
\end{example}

\section{Menggerakkan gula: alir tekanan}

\begin{definition}[Floem, tabung tapis, sumber dan penampung]\label{def:b1:plant-transport:phloem}
\index{floem}\index{tabung tapis}\index{sumber}\index{penampung}
Gula menempuh perjalanan dalam \emph{floem}, dalam \emph{tabung
tapis}: yaitu deretan \omterm{def:b1:cell-unit-of-life:cell}{sel} hidup yang telah kehilangan \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya} dan
sebagian besar organelnya tetapi tetap memegang membrannya, tersambung
ujung dengan ujung lewat \emph{lempeng tapis} yang berperforasi, tiap
\omterm{def:b1:cell-unit-of-life:cell}{selnya} diasuh sebuah \emph{\omterm{def:b1:cell-unit-of-life:cell}{sel} pengiring} yang memasok \omterm{def:b1:proteins:peptide}{protein} dan
\omterm{def:b1:water-small-molecules:atp}{ATP-nya}. Getahnya \qtyrange{10}{30}{\%} sukrosa, dengan \omterm{def:b1:proteins:aminoacid}{asam amino},
ion dan molekul pengisyaratan, dan ia mengalir dari \emph{sumber} ---
\omterm{def:b1:flowering-plant-organization:organs}{daun} yang berfotosintesis, atau \omterm{def:b1:mammal-organization:organ}{organ} simpanan yang sedang dikosongkan
--- ke \emph{penampung} --- pucuk yang tumbuh, \omterm{def:b1:flowering-plant-organization:organs}{akar}, buah, biji, \omterm{def:b1:mammal-organization:organ}{organ}
simpanan yang sedang diisi --- pada \qtyrange{0.5}{1}{m/h}, pada arah
mana pun, dan pada arah yang berbeda dalam tabung yang berbeda.
\end{definition}

\begin{theorem}[Alir tekanan]\label{thm:b1:plant-transport:pressureflow}
\index{mekanisme alir tekanan}
Getah \omterm{def:b1:plant-transport:phloem}{floemnya} bergerak lewat alir ruah yang dipacu sebuah beda
tekanan hidrostatik antara sumber dan penampungnya (Münch, 1930). Pada
sumbernya, \omterm{def:b1:cell-unit-of-life:cell}{sel} pengiringnya \emph{memuatkan} sukrosa ke dalam \omterm{def:b1:plant-transport:phloem}{tabung
tapisnya} lawan landaiannya (lewat simpor proton, dari \omterm{def:b1:plant-water-minerals:pathways}{apoplasnya}, atau
lewat \omterm{def:b1:eukaryotic-cell:plantcell}{plasmodesma}); \omterm{def:b1:membranes-transport:osmosis}{potensial air} tabungnya turun, airnya masuk dari
\omterm{def:b1:flowering-plant-organization:tissues}{xilem} tetangganya, dan \omterm{def:b1:eukaryotic-cell:plantcell}{turgornya} naik menjadi \qty{1}{MPa} atau lebih.
Pada penampungnya, sukrosanya \emph{dibongkar} lalu dipakai atau
disimpan, \omterm{def:b1:membranes-transport:osmosis}{potensial airnya} naik, airnya pergi, dan \omterm{def:b1:eukaryotic-cell:plantcell}{turgornya} rendah.
Di antara keduanya getahnya mengalir menuruni landaian tekanannya
lewat lempeng tapisnya, sambil membawa apa pun yang terlarut di
dalamnya. Tumbuhannya membelanjakan energi hanya pada kedua ujungnya;
alirannya sendiri bersifat pasif.
\end{theorem}
\begin{proof}[Bukti]
Seekor kutu \omterm{def:b1:flowering-plant-organization:organs}{daun} yang mengisap sebuah \omterm{def:b1:flowering-plant-organization:organs}{batang} menusukkan stiletnya ke
dalam sebuah \omterm{def:b1:plant-transport:phloem}{tabung tapis} tunggal; potonglah kutunya lalu stiletnya,
yang tertinggal di tempatnya, mengeluarkan getah selama berjam-jam di
bawah tekanan tabungnya sendiri --- getah \omterm{def:b1:plant-transport:phloem}{floem} murni, yang kepekatan
sukrosa dan laju alirnya dapat diukur, dan yang tekanannya, yang
diukur dengan sebuah manometer pada stiletnya, sekitar \qty{1}{MPa}
dekat sebuah sumber dan lebih rendah menuju penampungnya.
$\mathrm{^{14}CO_2}$ bertanda yang diberikan kepada sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun}
muncul dalam \omterm{def:b1:plant-transport:phloem}{floem} \omterm{def:b1:flowering-plant-organization:organs}{batangnya} dalam beberapa menit lalu bergerak pada
laju yang diramalkan alir stiletnya; mendinginkan sepotong \omterm{def:b1:flowering-plant-organization:organs}{batangnya},
yang memperlambat \omterm{def:b1:cell-unit-of-life:cell}{sel} hidupnya tetapi nyaris tak memengaruhi sebuah
alir yang dipacu tekanan, hanya sedikit memperlambat pengangkutannya;
meracuni pemuatan \omterm{def:b1:flowering-plant-organization:organs}{daun} sumbernya menghentikannya.
\end{proof}

\begin{method}[Membaca sebuah percobaan translokasi]\label{met:b1:plant-transport:translocation}
\begin{enumerate}
  \item Berilah sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} $\mathrm{CO_2}$ bertanda; ambillah cuplikan
        \omterm{def:b1:flowering-plant-organization:organs}{batangnya} di atas dan di bawahnya pada selang waktu tertentu:
        letak tandanya terhadap waktu memberikan arah dan laju alirnya.
  \item Cincinlah \omterm{def:b1:flowering-plant-organization:organs}{batangnya} (singkirkan sebuah cincin kulit kayu, yang
        membawa \omterm{def:b1:plant-transport:phloem}{floemnya}, dengan meninggalkan \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}): gulanya
        bertimbun di atas cincinnya dan \omterm{def:b1:flowering-plant-organization:organs}{akarnya} kelaparan di bawahnya
        --- \omterm{def:b1:plant-transport:phloem}{floemnya}, bukan \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}, yang membawanya; pohonnya mati
        dalam satu musim.
  \item Kumpulkan getah dari stilet kutu \omterm{def:b1:flowering-plant-organization:organs}{daun} yang terpotong pada dua
        ketinggian: turunnya sukrosa dan tekanan di antara keduanya
        adalah landaian yang memacu alirannya.
  \item Ubahlah penampungnya: singkirkan buahnya, dan tandanya pergi ke
        \omterm{def:b1:flowering-plant-organization:organs}{akarnya}; naungi \omterm{def:b1:flowering-plant-organization:organs}{daunnya}, dan ia menjadi sebuah penampung
        sendiri. Pola alirannya adalah pola permintaannya.
\end{enumerate}
\end{method}

\begin{example}[Sebuah kentang dalam dua musim]\label{ex:b1:plant-transport:potato}
Pada musim panas \omterm{def:b1:flowering-plant-organization:organs}{daunnya} menjadi sumber dan umbi di bawah tanahnya
menjadi penampung: sukrosanya mengalir turun, dibongkar, diubah
menjadi \omterm{def:b1:carbohydrates:polysaccharide}{pati}, dan umbinya menggembung. Pada musim semi umbinya
bertunas: \omterm{def:b1:carbohydrates:polysaccharide}{patinya} diubah kembali menjadi sukrosa, dimuatkan ke dalam
\omterm{def:b1:plant-transport:phloem}{floemnya}, lalu mengalir naik ke tunas yang tumbuh --- tabung yang
sama, arah yang berlawanan, sebab sumber dan penampungnya telah
bertukar tempat.
\end{example}

\section{Kompromi itu}

\begin{proposition}[Efisiensi penggunaan air]\label{prop:b1:plant-transport:wue}
\index{efisiensi penggunaan air}
Tiap \omterm{def:b1:plant-transport:stoma}{stoma} yang memasukkan karbon dioksida kehilangan air, dan
pertukarannya tidak setara: bagian dalam \omterm{def:b1:flowering-plant-organization:organs}{daunnya} \omterm{def:b1:lipids:fattyacid}{jenuh} oleh uap dan
udara di luarnya kering, sedangkan $\mathrm{CO_2}$ berada pada
\qty{0.04}{\%} di luar dan lebih rendah di dalam, sehingga landaian
airnya yang mengarah keluar ratusan kali landaian karbon dioksidanya
yang mengarah masuk. Sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} C3 kehilangan
\qtyrange{200}{500}{} molekul air tiap $\mathrm{CO_2}$ yang
dipancangkan; sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} C4, yang memekatkan $\mathrm{CO_2}$ lalu
menjaga \omterm{def:b1:flowering-plant-organization:tissues}{stomatanya} lebih sempit, \qtyrange{100}{200}{}; sebatang
\omterm{def:b1:photosynthesis:c4cam}{tumbuhan CAM}, yang membuka hanya pada malam hari,
\qtyrange{20}{50}{} (\cref{ch:b1:photosynthesis}). Yang disebut
\emph{\omterm{prop:b1:plant-transport:wue}{efisiensi penggunaan air}} adalah karbon yang diperoleh tiap air
yang hilang, dan perbaikannya --- lewat menutup pada tengah hari,
lewat \omterm{def:b1:flowering-plant-organization:tissues}{stomata} yang terbenam dan \omterm{def:b1:flowering-plant-organization:tissues}{kutikula} yang tebal, lewat lintasan C4
dan CAM --- adalah bagian tumbuhan dalam tawarannya dengan sebuah
atmosfer yang kering.
\end{proposition}

\begin{example}[Tawarannya dalam angka]\label{ex:b1:plant-transport:bargain}
Sebatang jagung yang mentranspirasikan seliter setengah sehari
memancangkan sekitar \qty{10}{g} karbon dioksida: \qty{80}{mol} air
bagi \qty{0.23}{mol} $\mathrm{CO_2}$, tiga ratus lima puluh berbanding
satu. Sebuah ladang gandum dalam satu musim mentranspirasikan lima
ratus ton air tiap hektar untuk membuat delapan ton biji; airnya
adalah harga karbonnya, dan di sebagian besar ladang dunia itulah
harga yang membatasi panennya.
\end{example}

\section{Soal: Sekuoia}

\begin{problem}[{Soal akhir pekan --- sebatang pohon seratus meter pada
sebuah hari musim panas: tegangan pada tajuknya, laju getahnya, air
yang digerakkannya, gula yang dikirimnya turun dan karbon yang
dibelinya, berakhir pada tegangan xilem di tajuknya}]
\label{pb:b1:plant-transport:1}
Sebuah sekuoia setinggi \qty{100}{m} dengan sebuah tajuk seluas
\qty{2000}{m^2} \omterm{def:b1:flowering-plant-organization:organs}{daun}. Pada sebuah hari musim panas ia
mentranspirasikan \qty{600}{L} air dalam \qty{12}{h} lewat kayu gubal
berpenampang \qty{0.5}{m^2}, yang \qty{20}{\%}-nya berupa lumen
penghantar. Air: $\rho = \qty{1000}{kg/m^3}$, $g =
\qty{9.8}{m/s^2}$, \qty{18}{g/mol}. Gesekan sepanjang \omterm{def:b1:flowering-plant-organization:organs}{batangnya} memakan
tegangan sebanyak gravitasinya. \omterm{def:b1:membranes-transport:osmosis}{Potensial air} tanahnya \qty{-0.1}{MPa}.
Tajuknya memancangkan \qty{4}{\micro mol} $\mathrm{CO_2}$ tiap meter
persegi \omterm{def:b1:flowering-plant-organization:organs}{daun} tiap sekon selama \qty{12}{h}; sukrosa \qty{342}{g/mol}
dan membawa dua belas karbon; getah \omterm{def:b1:plant-transport:phloem}{floemnya} \qty{15}{\%} sukrosa
menurut massa (kerapatan \qty{1.06}{g/mL}) dan mengalir pada
\qty{0.6}{m/h}.

\medskip
\noindent\textbf{Bagian I --- Tegangan.}
\begin{enumerate}
  \item Hitunglah tegangan yang dibutuhkan untuk menopang kolomnya
        lawan gravitasinya.
  \item Tambahkan gesekannya: berapa tegangan \omterm{def:b1:flowering-plant-organization:tissues}{xilem} pada tajuknya?
  \item Hitunglah \omterm{def:b1:membranes-transport:osmosis}{potensial air} \omterm{def:b1:flowering-plant-organization:tissues}{xilem} pada tajuknya, dengan mengambil
        $\Psi_s = 0$ bagi getahnya.
  \item Berapa \omterm{def:b1:membranes-transport:osmosis}{potensial air} \omterm{def:b1:cell-unit-of-life:cell}{sel} \omterm{def:b1:flowering-plant-organization:organs}{daunnya} di tajuknya agar air masuk ke
        dalamnya? Bila $\Psi_s = \qty{-2.5}{MPa}$ pada \omterm{def:b1:cell-unit-of-life:cell}{sel} itu, berapa
        \omterm{def:b1:eukaryotic-cell:plantcell}{turgornya}?
  \item Bandingkan dengan sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} pada \qty{10}{m} di pohon yang sama ($\Psi_s =
        \qty{-1.5}{MPa}$).
  \item Air dalam tabung halus menahan sekitar \qty{-30}{MPa}; mengapa
        pohonnya tidak sekadar tumbuh sampai \qty{1000}{m}?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Alir.}
\begin{enumerate}[resume]
  \item Hitunglah laju transpirasinya dalam liter tiap jam dan dalam
        meter kubik tiap sekon.
  \item Hitunglah penampang penghantar kayu gubalnya.
  \item Hitunglah laju purata getahnya.
  \item Hitunglah massa air yang tertahan dalam lumen penghantar
        \omterm{def:b1:flowering-plant-organization:organs}{batangnya} (\qty{100}{m} darinya).
  \item Berapa lama sebuah molekul air menempuh perjalanan dari \omterm{def:b1:flowering-plant-organization:organs}{akar} ke
        tajuk pada laju itu?
  \item Hitunglah transpirasinya tiap meter persegi \omterm{def:b1:flowering-plant-organization:organs}{daun} tiap jam,
        dalam gram, dan tiap sekon dalam milimol.
  \item Getahnya mengalir menuruni sebuah landaian potensial dari
        \qty{-0.1}{} pada tanahnya ke nilai tajuknya pada pertanyaan 3
        sepanjang \qty{100}{m}. Berapa landaiannya dalam megapascal
        tiap meter? Bagaimana perbandingannya dengan bagian
        gravitasinya saja?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Gula.}
\begin{enumerate}[resume]
  \item Hitunglah $\mathrm{CO_2}$ yang dipancangkan tajuknya tiap hari,
        dalam mol.
  \item Hitunglah sukrosa yang setara dengannya, dalam mol dan dalam
        kilogram.
  \item Separuhnya direspirasikan dalam tajuknya; sisanya dikirim
        menuruni \omterm{def:b1:plant-transport:phloem}{floemnya}. Hitunglah massa sukrosa yang ditranslokasikan
        tiap hari dan massa getah yang membawanya.
  \item Hitunglah volume getahnya dan, pada \qty{0.6}{m/h}, penampang
        \omterm{def:b1:plant-transport:phloem}{tabung tapis} yang dibutuhkan untuk membawanya dalam \qty{24}{h}.
  \item Hitunglah waktu bagi sukrosanya menempuh perjalanan dari tajuk
        ke \omterm{def:b1:flowering-plant-organization:organs}{akar}.
  \item Hitunglah nisbah penggunaan airnya: mol air yang
        ditranspirasikan tiap mol $\mathrm{CO_2}$ yang dipancangkan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Kompromi.}
\begin{enumerate}[resume]
  \item \omterm{def:b1:membranes-transport:osmosis}{Potensial air} getah \omterm{def:b1:plant-transport:phloem}{floem} pada tajuknya, dengan
        $\Psi_s = \qty{-1.2}{MPa}$ dan \omterm{def:b1:eukaryotic-cell:plantcell}{turgor} \qty{1.0}{MPa}: hitunglah
        lalu bandingkanlah dengan potensial \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} di situ. Ke arah
        mana air bergerak antara keduanya, dan sesuaikah itu dengan
        pemuatannya?
  \item Pada \omterm{def:b1:flowering-plant-organization:organs}{akarnya} getah \omterm{def:b1:plant-transport:phloem}{floemnya} \qty{5}{\%} sukrosa ($\Psi_s =
        \qty{-0.4}{MPa}$) dan \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya} pada \qty{-0.3}{MPa}. \omterm{def:b1:eukaryotic-cell:plantcell}{Turgor}
        berapa yang membuat $\Psi$ \omterm{def:b1:plant-transport:phloem}{floemnya} setara dengan \omterm{def:b1:flowering-plant-organization:tissues}{xilemnya}? Beda
        tekanan berapa antara tajuk dan \omterm{def:b1:flowering-plant-organization:organs}{akarnya} yang memacu alirannya?
  \item \omterm{def:b1:flowering-plant-organization:tissues}{Stomatanya} menutup pada tengah hari dan transpirasinya
        terparuh. Hitunglah kembali tegangan gesekannya dan potensial
        \omterm{def:b1:flowering-plant-organization:tissues}{xilem} tajuknya. Apa yang diperoleh dan hilang dari pohonnya?
  \item Pada sebuah tahun kering tanahnya turun ke \qty{-1.0}{MPa}.
        Hitunglah kembali potensial tajuknya dengan transpirasi penuh.
        Sanggupkah \omterm{def:b1:flowering-plant-organization:organs}{daun} tajuknya ($\Psi_s = \qty{-2.5}{MPa}$) menahan
        \omterm{def:b1:eukaryotic-cell:plantcell}{turgor} apa pun? Apa yang harus diperbuat pohonnya?
  \item Jelaskan mengapa \omterm{def:b1:flowering-plant-organization:organs}{daun} yang paling tinggi adalah yang paling
        kecil dan paling tebal.
  \item Nyatakan hasilnya: tegangan \omterm{def:b1:flowering-plant-organization:tissues}{xilem} pada tajuk sebatang pohon
        \qty{100}{m} pada sebuah hari musim panas, dan laju getah yang
        menyertainya.
\end{enumerate}
\end{problem}
