\chapter{Pengungkapan Gen: Transkripsi dan Translasi}\label{ch:b1:gene-expression}

Sebuah \omterm{def:b1:genomes:gene}{gen} berisi tiga puluh ribu \omterm{thm:b1:nucleic-acids:helix}{pasangan basa} dibaca dalam seperempat
jam menjadi sebuah salinan kerja, dipangkas menjadi dua ribu huruf,
dikirim keluar dari \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}, lalu ditranslasikan dua puluh lima \omterm{def:b1:gene-expression:ribosome}{ribosom}
sekaligus menjadi sebuah \omterm{def:b1:proteins:peptide}{protein} berisi lima ratus \omterm{def:b1:proteins:aminoacid}{asam amino} --- satu
tiap empat sekon, selama salinannya bertahan. \omterm{def:b1:cell-unit-of-life:cell}{Sel} membelanjakan energi
untuk hal ini lebih banyak daripada untuk apa pun yang lain. Bab ini
mengikuti informasinya dari \omterm{def:b1:nucleic-acids:chain}{DNA} ke \omterm{def:b1:proteins:peptide}{protein}: penyalinan sebuah \omterm{def:b1:genomes:gene}{gen}
menjadi \omterm{def:b1:nucleic-acids:chain}{RNA}, penyuntingan \omterm{def:b1:nucleic-acids:chain}{RNA} itu pada eukariot, kode yang memetakan
triplet basa pada \omterm{def:b1:proteins:aminoacid}{asam amino}, \omterm{def:b1:gene-expression:ribosome}{ribosom} yang membacanya, dan apa yang
terjadi pada sebuah \omterm{def:b1:proteins:peptide}{protein} begitu ia dibuat.

\section{Dari gen ke protein}

\begin{proposition}[Aliran informasinya]\label{prop:b1:gene-expression:dogma}
\index{pengungkapan gen}\index{transkripsi}\index{translasi}
Informasi genetik mengalir dari \omterm{def:b1:nucleic-acids:chain}{DNA} ke \omterm{def:b1:nucleic-acids:chain}{RNA} ke \omterm{def:b1:proteins:peptide}{protein}.
\emph{Transkripsi} menyalin satu untai sebuah \omterm{def:b1:genomes:gene}{gen} menjadi sebuah \omterm{def:b1:nucleic-acids:chain}{RNA}
yang urutannya sama dengan untai yang lain (dengan U menggantikan T);
\emph{translasi} membaca \omterm{def:b1:nucleic-acids:rnas}{RNA pembawa pesan} tiga basa pada satu waktu
lalu merakit \omterm{def:b1:proteins:aminoacid}{asam amino} yang bersesuaian menjadi sebuah \omterm{def:b1:proteins:peptide}{polipeptida}.
Tiap langkahnya memakai pemasangan basa: \omterm{def:b1:nucleic-acids:chain}{DNA} dengan \omterm{def:b1:nucleic-acids:chain}{RNA} dalam
transkripsi, pembawa pesan dengan \omterm{def:b1:gene-expression:trna}{RNA pemindah} dalam translasi. Karena
itu urutan sebuah \omterm{def:b1:proteins:peptide}{protein} adalah transkrip urutan \omterm{def:b1:genomes:gene}{gennya}, dan perubahan
satu basa dapat mengubah satu \omterm{def:b1:proteins:aminoacid}{asam amino} --- tautan antara mutasi dan
fenotipenya. Informasi tidak mengalir balik dari \omterm{def:b1:proteins:peptide}{protein} ke asam
nukleat; satu langkah balik yang ada, \omterm{def:b1:nucleic-acids:chain}{RNA} yang disalin menjadi \omterm{def:b1:nucleic-acids:chain}{DNA} oleh
retrovirus, tidak menyentuh \omterm{def:b1:proteins:peptide}{proteinnya}.
\end{proposition}

\section{Transkripsi}

\begin{definition}[RNA polimerase, promotor, transkripsi]\label{def:b1:gene-expression:transcription}
\index{RNA polimerase}\index{promotor}
\emph{\omterm{def:b1:nucleic-acids:chain}{RNA} polimerase} mensintesis \omterm{def:b1:nucleic-acids:chain}{RNA} pada sebuah cetakan \omterm{def:b1:nucleic-acids:chain}{DNA},
$5' \to 3'$, dari keempat ribonukleosida trifosfatnya, tanpa sebuah
primer. Ia mulai pada sebuah \emph{promotor}, yaitu urutan tepat di
hulu \omterm{def:b1:genomes:gene}{gennya} yang dikenali dan diikatnya; pada bakteri sebuah subunit
yang disebut $\sigma$ (\emph{sigma}) yang mengenalinya, pada dua
bentangan lestari sepuluh dan tiga puluh lima pasangan sebelum
awalnya. \omterm{def:b1:enzymes:enzyme}{Enzimnya} membuka sekitar lima belas pasangan heliksnya menjadi
sebuah \emph{gelembung transkripsi}, menyalin \emph{untai cetakannya}
(yang dibaca $3' \to 5'$) menjadi sebuah \omterm{def:b1:nucleic-acids:chain}{RNA} yang urutannya serupa
dengan untai yang lain, yaitu untai \emph{penyandi}, lalu bergerak
sepanjangnya pada sekitar lima puluh \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} tiap sekon, sambil
menutup kembali heliksnya di belakangnya; \omterm{def:b1:nucleic-acids:chain}{RNA-nya} mengelupas seiring
dibuatnya. Ia berhenti pada sebuah \emph{pengakhir}: pada bakteri
sebuah urutan yang melengkapi dirinya sendiri yang \omterm{def:b1:nucleic-acids:chain}{RNA-nya} melipat
menjadi sebuah jepit rambut yang menarik transkripnya lepas. Banyak
polimerase dapat mengikuti satu sama lain sepanjang sebuah \omterm{def:b1:genomes:gene}{gen},
sehingga sebuah transkrip dapat dimulai tiap sekon.
\end{definition}

\begin{proposition}[Transkripsi eukariot dan pemrosesan RNA]\label{prop:b1:gene-expression:processing}
\index{penyambungan}\index{RNA pembawa pesan}\index{tudung}\index{ekor poli-A}
Eukariot memiliki tiga polimerase: I bagi \omterm{def:b1:nucleic-acids:rnas}{RNA ribosom} yang besar, II
bagi \omterm{def:b1:nucleic-acids:rnas}{RNA pembawa pesan}, III bagi \omterm{def:b1:gene-expression:trna}{RNA pemindah} dan \omterm{def:b1:nucleic-acids:chain}{RNA} kecil. Polimerase
II tidak mengenali \omterm{def:b1:gene-expression:transcription}{promotornya} sendirian: sehimpunan \emph{faktor
transkripsi umum} merakit diri padanya (pada sebuah urutan TATA sekitar
tiga puluh pasangan di hulu pada banyak \omterm{def:b1:genomes:gene}{gen}) lalu merekrut \omterm{def:b1:enzymes:enzyme}{enzimnya},
dan \omterm{def:b1:proteins:peptide}{protein} pengatur yang terikat dekat atau jauh menyetel lajunya
(\cref{ch:b1:expression-control}). Transkrip primernya
\emph{diproses} dalam \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}: sebuah \emph{tudung} guanina yang diubah
ditambahkan pada ujung $5'$-nya, sebuah ekor berisi sekitar dua ratus
adenina (\emph{poli-A}) pada ujung $3'$-nya, dan \omterm{def:b1:genomes:gene}{intronnya} disingkirkan
lewat \emph{penyambungan} --- sebuah kompleks \omterm{def:b1:nucleic-acids:chain}{RNA} kecil dan \omterm{def:b1:proteins:peptide}{protein},
yaitu \emph{spliceosom}, mengenali GU pada awal tiap \omterm{def:b1:genomes:gene}{intronnya} dan AG
pada ujungnya, memotong, lalu menyambung \omterm{def:b1:genomes:gene}{eksonnya}. Baru sesudah itu
pembawa pesan yang matang diekspor ke \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya}. Banyak \omterm{def:b1:genomes:gene}{gen} disambung
dengan lebih dari satu cara (\emph{penyambungan berselang-selang}),
sehingga satu \omterm{def:b1:genomes:gene}{gen} menghasilkan beberapa \omterm{def:b1:proteins:peptide}{protein}; kedua puluh ribu \omterm{def:b1:genomes:gene}{gen}
manusia membuat barangkali seratus ribu \omterm{def:b1:proteins:peptide}{protein}.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Sharp dan Roberts (1977) menghibridisasikan sebuah \omterm{def:b1:nucleic-acids:rnas}{RNA pembawa pesan}
\omterm{def:b1:genomes:virus}{virus} pada \omterm{def:b1:nucleic-acids:chain}{DNA} \omterm{def:b1:genomes:gene}{gennya} lalu melihat hibridnya di bawah \omterm{prop:b1:cell-unit-of-life:microscopes}{mikroskop
elektron}: \omterm{def:b1:nucleic-acids:chain}{RNA-nya} berpasangan dengan \omterm{def:b1:nucleic-acids:chain}{DNA-nya} pada beberapa bentangan,
dan di antaranya \omterm{def:b1:nucleic-acids:chain}{DNA-nya} menggelung keluar tanpa pasangan --- yaitu
\omterm{def:b1:genomes:gene}{intronnya}, yang hadir dalam \omterm{def:b1:genomes:gene}{gennya} dan tidak hadir dalam pesannya. \omterm{def:b1:genomes:gene}{Gen}
bakteri, yang diperlakukan sama, tidak memberikan gelung. Ukuran
transkrip primer sebuah \omterm{def:b1:genomes:gene}{gen}, yang diukur dalam \omterm{def:b1:cell-unit-of-life:prokeuk}{intinya}, sepadan dengan
\omterm{def:b1:genomes:gene}{gennya}; ukuran pembawa pesan \omterm{def:b1:cell-unit-of-life:cell}{sitoplasmanya} sepadan dengan \omterm{def:b1:genomes:gene}{eksonnya} saja.
\end{proof}

\section{Kode genetik}

\begin{definition}[Kode genetik]\label{def:b1:gene-expression:code}
\index{kode genetik}\index{kodon}\index{kerangka baca}
Yang disebut \emph{kode genetik} memetakan triplet basa pembawa pesan,
yaitu \emph{kodon}, pada \omterm{def:b1:proteins:aminoacid}{asam amino}. Dari 64 kodonnya, 61 menentukan
kedua puluh \omterm{def:b1:proteins:aminoacid}{asam aminonya} dan tiga (UAA, UAG, UGA) menjadi isyarat
\emph{berhenti}; AUG menentukan metionina dan sekaligus menjadi kodon
\emph{mulai}, sehingga tiap \omterm{def:b1:proteins:peptide}{polipeptida} baru diawali metionina. Kodenya
bersifat \emph{merosot} --- sebagian besar \omterm{def:b1:proteins:aminoacid}{asam amino} memiliki
beberapa kodon, yang sebagian besarnya berbeda pada kedudukan
ketiganya ---
\emph{tak bertindih}, dibaca tanpa tanda baca dalam sebuah \emph{kerangka
baca} yang ditetapkan kodon mulainya, dan hampir \emph{semesta}, sama
pada bakteri, tumbuhan dan hewan dengan ragam kecil dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria}:
sebuah \omterm{def:b1:genomes:gene}{gen} manusia yang ditaruh dalam sebuah bakteri dibaca dengan
benar.
\end{definition}

\begin{proposition}[Bagaimana kodenya dibaca]\label{prop:b1:gene-expression:nirenberg}
Kodenya adalah kode triplet yang tak bertindih, dan makna tiap \omterm{def:b1:gene-expression:code}{kodonnya}
dapat ditentukan lewat kimia.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Crick dan Brenner (1961) membuat mutasi pada sebuah \omterm{def:b1:genomes:gene}{gen} fag yang
menambahkan atau menyingkirkan satu basa: satu perubahan semacam itu
merusak fungsi \omterm{def:b1:genomes:gene}{gennya}, begitu pula dua, tetapi tiga penyisipan (atau
tiga penghapusan) yang berdekatan memulihkannya --- pesannya dibaca
bertiga, dari sebuah titik awal yang tetap, dan sebuah penyisipan
menggeser kerangka segala yang ada di hilirnya. Nirenberg dan Matthaei
(1961) menambahkan sebuah \omterm{def:b1:nucleic-acids:chain}{RNA} buatan berisi urasil saja pada sebuah
ekstrak \emph{E. coli} tanpa \omterm{def:b1:cell-unit-of-life:cell}{sel} bersama kedua puluh \omterm{def:b1:proteins:aminoacid}{asam aminonya}:
ekstrak itu membuat sebuah rantai fenilalanina saja, sehingga UUU
berarti Phe; pembawa pesan buatan yang lain, lalu pengikatan
trinukleotida tunggal pada \omterm{def:b1:gene-expression:ribosome}{ribosom} bersama \omterm{def:b1:gene-expression:trna}{RNA pemindahnya},
menetapkan keenam puluh empatnya pada 1966.
\end{proof}

\begin{example}[Membaca sebuah pesan]\label{ex:b1:gene-expression:reading}
\omterm{def:b1:membranes-transport:transporters}{Pembawa} pesan $5'$-\ldots{}GC\underline{AUG}GCUUUCGGAUAA\ldots{}-$3'$
dibaca dari AUG-nya: AUG GCU UUC GGA UAA --- Met-Ala-Phe-Gly-berhenti:
sebuah peptida berisi empat residu. Hapuslah G pertama sesudah AUG dan
kerangkanya bergeser: AUG CUU UCG GAU AA\ldots{} ---
Met-Leu-Ser-Asp\ldots, sebuah \omterm{def:b1:proteins:peptide}{protein} yang berbeda dengan panjang yang
berbeda. Ubahlah UUC menjadi UUU dan tak ada yang berubah: keduanya
Phe, sebuah mutasi \emph{bisu}.
\end{example}

\section{Translasi}

\begin{definition}[RNA pemindah dan sintetasenya]\label{def:b1:gene-expression:trna}
\index{RNA pemindah}\index{antikodon}\index{aminoasil-tRNA sintetase}
Sebuah \emph{\omterm{def:b1:nucleic-acids:chain}{RNA} pemindah} (\cref{ch:b1:nucleic-acids}) adalah
penghubung antara \omterm{def:b1:gene-expression:code}{kodon} dan \omterm{def:b1:proteins:aminoacid}{asam amino}: \emph{antikodon}nya, yaitu tiga
basa dalam sebuah gelung, berpasangan antiparalel dengan sebuah \omterm{def:b1:gene-expression:code}{kodon}
pesannya, dan ujung $3'$-nya membawa \omterm{def:b1:proteins:aminoacid}{asam amino} yang bersesuaian.
Pemasangan pada kedudukan \omterm{def:b1:gene-expression:code}{kodon} ketiganya longgar (\emph{goyah}),
sehingga sekitar empat puluh tRNA sudah cukup bagi enam puluh satu
\omterm{def:b1:gene-expression:code}{kodonnya}. Tiap tRNA dimuati \emph{aminoasil-tRNA sintetase}-nya
sendiri, yaitu sebuah \omterm{def:b1:enzymes:enzyme}{enzim} yang mengenali \omterm{def:b1:proteins:aminoacid}{asam aminonya} maupun
tRNA-nya lalu menyambungkan keduanya dalam dua langkah dengan biaya
satu \omterm{def:b1:water-small-molecules:atp}{ATP} (yang dibelah menjadi AMP: dua ikatan berenergi tinggi),
sambil mengoreksi \omterm{def:b1:proteins:aminoacid}{asam aminonya}. Kedua puluh \omterm{def:b1:enzymes:enzyme}{enzim} itulah tempat
kodenya sungguh-sungguh dilaksanakan: \omterm{def:b1:gene-expression:ribosome}{ribosomnya} tidak memeriksa \omterm{def:b1:proteins:aminoacid}{asam
amino} mana yang dibawa sebuah tRNA, hanya memeriksa bahwa antikodonnya
cocok.
\end{definition}

\begin{definition}[Ribosom]\label{def:b1:gene-expression:ribosome}
\index{ribosom}
Yang disebut \emph{ribosom} adalah sebuah partikel berisi dua subunit,
masing-masing dari \omterm{def:b1:nucleic-acids:rnas}{RNA ribosom} dan \omterm{def:b1:proteins:peptide}{protein} (pada bakteri: sebuah
subunit kecil 30S dan sebuah subunit besar 50S, bersama-sama 70S,
\qty{2.5}{MDa}; ribosom eukariot lebih besar, 80S). Subunit kecilnya
mengikat pembawa pesannya lalu memecahkan kodenya; subunit besarnya
memegang tRNA-nya lalu membentuk \omterm{def:b1:proteins:peptide}{ikatan peptidanya} --- katalisnya
adalah \omterm{def:b1:nucleic-acids:rnas}{RNA ribosomnya} sendiri. Tiga sisi tRNA menjangkau kedua
subunitnya: A (aminoasil, tempat tRNA bermuatan berikutnya masuk), P
(peptidil, yang memegang rantai yang sedang tumbuh), E (keluar).
\end{definition}

\begin{proposition}[Daur pemanjangannya]\label{prop:b1:gene-expression:elongation}
Translasi mulai ketika subunit kecilnya menemukan \omterm{def:b1:gene-expression:code}{kodon} mulainya ---
pada bakteri dengan memasangkan sebuah urutan \omterm{def:b1:nucleic-acids:rnas}{RNA ribosom} dengan sebuah
sisi tepat di hulu AUG-nya, pada eukariot dengan mengikat tudungnya lalu
memindai sampai AUG pertamanya --- dan tRNA pemulanya (metionina)
menetap pada sisi P-nya; subunit besarnya lalu bergabung. Tiap putaran
\emph{pemanjangan} menambahkan satu residu: sebuah tRNA bermuatan yang
\omterm{def:b1:gene-expression:trna}{antikodonnya} cocok dengan \omterm{def:b1:gene-expression:code}{kodon} sisi A-nya diantarkan sebuah faktor
pemanjangan lalu diperiksa (satu GTP); \omterm{def:b1:nucleic-acids:rnas}{RNA ribosom} subunit besarnya
memindahkan rantai yang tumbuh dari tRNA sisi P-nya ke \omterm{def:b1:proteins:aminoacid}{asam amino} tRNA
sisi A-nya, sehingga membentuk \omterm{def:b1:proteins:peptide}{ikatan peptidanya}; \omterm{def:b1:gene-expression:ribosome}{ribosomnya} bergerak
satu \omterm{def:b1:gene-expression:code}{kodon} (sebuah GTP kedua), sehingga menggeser tRNA-nya ke P dan E,
dan yang kosong pergi. Pada sebuah \omterm{def:b1:gene-expression:code}{kodon} berhenti sebuah \emph{faktor
pelepas} masuk ke sisi A-nya dan rantainya dihidrolisis lepas. \omterm{def:b1:gene-expression:ribosome}{Ribosom}
bakteri menambahkan lima belas sampai dua puluh residu tiap sekon,
\omterm{def:b1:gene-expression:ribosome}{ribosom} eukariot dua sampai lima; beberapa \omterm{def:b1:gene-expression:ribosome}{ribosom} membaca satu pesan
sekaligus, sehingga membentuk sebuah \emph{polisom}.
\end{proposition}

\begin{method}[Memperhitungkan sintesis sebuah protein]\label{met:b1:gene-expression:reckon}
\begin{enumerate}
  \item Panjangnya: sebuah \omterm{def:b1:proteins:peptide}{protein} berisi $n$ residu memerlukan sebuah
        urutan penyandi berisi $3n + 3$ \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} (\omterm{def:b1:gene-expression:code}{kodon} berhentinya
        termasuk), di dalam sebuah pembawa pesan yang lebih panjang
        sebesar ujung tak tertranslasinya dan, dalam \omterm{def:b1:genomes:gene}{gennya}, sebesar
        \omterm{def:b1:genomes:gene}{intronnya}.
  \item Waktunya: bagilah $n$ dengan laju pemanjangannya
        (\qtyrange{15}{20}{} tiap sekon pada bakteri, \qtyrange{2}{5}{}
        pada eukariot); pesannya dibaca satu \omterm{def:b1:gene-expression:ribosome}{ribosom} tiap
        \qtyrange{80}{100}{} \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}, sehingga keluaran tiap pesannya
        adalah satu rantai tiap (jarak / laju) sekon.
  \item Energinya: empat ikatan fosfat berenergi tinggi tiap residu
        (dua untuk memuati tRNA-nya, dua GTP pada \omterm{def:b1:gene-expression:ribosome}{ribosomnya}), ditambah
        transkripsi pesannya sebanyak dua tiap \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}, yang dibagi
        di antara semua rantai yang dihasilkan pesan itu.
  \item Kesetiaannya: sekitar satu \omterm{def:b1:proteins:aminoacid}{asam amino} yang keliru tiap $10^4$
        \omterm{def:b1:gene-expression:code}{kodon}; sebuah \omterm{def:b1:proteins:peptide}{protein} berisi \qty{500}{} residu keliru di suatu
        tempat pada satu salinan dari dua puluh --- yang tertanggungkan
        karena \omterm{def:b1:proteins:peptide}{protein} dapat diganti dan kekeliruannya tidak diwariskan.
\end{enumerate}
\end{method}

\section{Sesudah translasi}

\begin{proposition}[Apa yang terjadi pada sebuah polipeptida baru]\label{prop:b1:gene-expression:after}
\index{peptida isyarat}\index{perubahan pascatranslasi}
Rantainya melipat seiring munculnya, dibantu \omterm{prop:b1:proteins:denaturation}{pendamping protein}
(\cref{ch:b1:proteins}). Tujuannya tertulis dalam urutannya: sebuah
\emph{\omterm{prop:b1:gene-expression:after}{peptida isyarat}} pada ujung N-nya mengikat sebuah \emph{partikel
pengenal isyarat} yang menghentikan translasinya, menambatkan
\omterm{def:b1:gene-expression:ribosome}{ribosomnya} pada \omterm{def:b1:eukaryotic-cell:endomembrane}{retikulum endoplasma}, lalu merangkai rantainya ke dalam
lumennya (\cref{ch:b1:eukaryotic-cell}); urutan yang lain mengarahkan
\omterm{def:b1:proteins:peptide}{protein} ke dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria}, kloroplas, \omterm{def:b1:cell-unit-of-life:prokeuk}{inti} atau \omterm{def:b1:eukaryotic-cell:peroxisome}{peroksisomnya}
sesudah sintesisnya; sebuah \omterm{def:b1:proteins:peptide}{protein} tanpa isyarat tinggal dalam
\omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya}. Banyak \omterm{def:b1:proteins:peptide}{protein} \emph{diubah}: metionina pemulanya
disingkirkan, gula dipautkan dalam RE dan Golginya, fosfat ditambahkan
dan disingkirkan kinase serta fosfatase, \omterm{def:b1:lipids:fattyacid}{lipid} dipautkan, keping
dipotong keluar (insulin dibuat sebagai satu rantai lalu dipotong
menjadi dua; zimogen dipotong untuk mengaktifkannya). Dan tiap \omterm{def:b1:proteins:peptide}{protein}
akhirnya \emph{diuraikan}: ditandai dengan \omterm{def:b1:proteins:peptide}{protein} kecil
\emph{ubikuitin} lalu dibuka lipatannya dan dicerna \emph{proteasom},
sesudah hidup beberapa menit (\omterm{def:b1:proteins:peptide}{protein} pengatur) sampai beberapa bulan
(hemoglobin) --- sehingga \omterm{def:b1:proteins:peptide}{protein} yang hadir adalah \omterm{def:b1:proteins:peptide}{protein} yang sedang
dibuat \omterm{def:b1:cell-unit-of-life:cell}{selnya}.
\end{proposition}

\begin{example}[Angka sebuah sel]\label{ex:b1:gene-expression:numbers}
Sebuah \emph{E. coli} yang sedang tumbuh memegang sekitar \num{20000}
\omterm{def:b1:gene-expression:ribosome}{ribosom}, yang masing-masing menambahkan dua puluh residu tiap sekon:
$\num{4e5}$ residu tiap sekon, sejuta \omterm{def:b1:proteins:peptide}{protein} berisi \qty{300}{} residu
dalam sejam --- yakni isinya sendiri, dan itulah yang dituntut
pelipatduaan tiap jam. Separuh energinya pergi ke sintesis \omterm{def:b1:proteins:peptide}{protein}.
Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} hati memegang sepuluh juta \omterm{def:b1:gene-expression:ribosome}{ribosom} dan membuat \omterm{def:b1:proteins:peptide}{protein}
untuk dipakai berminggu-minggu alih-alih untuk membelah; sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}
plasma, yang menyekresikan antibodi, mengerahkan hampir seluruhnya bagi
satu \omterm{def:b1:proteins:peptide}{protein} lalu mencurahkan dua ribu molekul tiap sekon.
\end{example}

\section{Soal: Dari Sebuah Gen ke Sebuah Protein}

\begin{problem}[{Soal akhir pekan --- sebuah gen tiga puluh kilobasa
yang diikuti sampai menjadi protein berisi lima ratus residu:
transkripsinya yang diukur waktunya, penyambungannya yang ditimbang,
ribosomnya yang dihitung, energi dan kekeliruannya yang diperhitungkan,
berakhir pada biaya satu protein dalam ATP}]
\label{pb:b1:gene-expression:1}
Sebuah \omterm{def:b1:genomes:gene}{gen} manusia menjangkau \qty{30}{kb} dengan delapan \omterm{def:b1:genomes:gene}{ekson} yang
seluruhnya \qty{2000}{} \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} pembawa pesan matang, yang
\qty{1503}{} di antaranya menyandi (500 residu ditambah \omterm{def:b1:gene-expression:code}{kodon}
berhentinya). Polimerase II mentranskripsi pada \qty{30}{} \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}
tiap sekon; \omterm{def:b1:gene-expression:ribosome}{ribosom} memanjang pada \qty{4}{} residu tiap sekon dan
berjarak satu tiap \qty{80}{} \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}; waktu paruh pembawa pesannya
\qty{2}{h}. Biayanya: dua ikatan berenergi tinggi tiap \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} yang
ditranskripsikan, empat tiap residu yang ditranslasikan; ambil satu
ikatan berenergi tinggi sebagai satu \omterm{def:b1:water-small-molecules:atp}{ATP}. Laju kekeliruannya:
$10^{-5}$ tiap \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} yang ditranskripsikan, $10^{-4}$ tiap \omterm{def:b1:gene-expression:code}{kodon}
yang ditranslasikan.

\medskip
\noindent\textbf{Bagian I --- Transkripsi dan penyambungannya.}
\begin{enumerate}
  \item Berapa lama satu polimerase mentranskripsi \omterm{def:b1:genomes:gene}{gennya}?
  \item Berapa fraksi transkrip primernya yang disingkirkan
        penyambungannya?
  \item Berapa banyak \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} yang ditranskripsikan bagi tiap
        \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} pembawa pesan matangnya?
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibelanjakan untuk mentranskripsi satu
        transkrip primer.
  \item Bila sebuah polimerase mulai tiap \qty{10}{s}, berapa banyak
        polimerase yang ada pada \omterm{def:b1:genomes:gene}{gennya} sekaligus, dan berapa banyak
        pembawa pesan yang dihasilkan \omterm{def:b1:genomes:gene}{gennya} tiap jam?
  \item Hitunglah banyaknya \omterm{def:b1:genomes:gene}{intron} dan panjang rata-ratanya.
  \item Spliceosomnya menyingkirkan tiap \omterm{def:b1:genomes:gene}{intron} dalam sekitar semenit,
        serentak. Penyambungannya atau transkripsinya yang menetapkan
        waktu dari \omterm{def:b1:genomes:gene}{gen} ke pembawa pesan?
  \item Hitunglah panjang fisik \omterm{def:b1:genomes:gene}{gennya} dan pembawa pesan matangnya
        (\qty{0.34}{nm} tiap \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida}).
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Translasi.}
\begin{enumerate}[resume]
  \item Berapa lama satu \omterm{def:b1:gene-expression:ribosome}{ribosom} mentranslasikan \omterm{def:b1:proteins:peptide}{proteinnya}?
  \item Berapa banyak \omterm{def:b1:gene-expression:ribosome}{ribosom} yang membaca satu pembawa pesan
        sekaligus?
  \item Berapa banyak molekul \omterm{def:b1:proteins:peptide}{protein} yang dihasilkan satu pembawa
        pesan tiap jam?
  \item Sepanjang waktu paruhnya \qty{2}{h}, berapa banyak yang
        dihasilkannya seluruhnya (sebuah pesan berwaktu paruh
        $t_{1/2}$ menghasilkan, rata-rata, $t_{1/2}/\ln 2$ jam produksi
        penuh)?
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibelanjakan untuk mentranslasikan satu
        \omterm{def:b1:proteins:peptide}{protein}.
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibelanjakan bagi transkripsinya tiap
        \omterm{def:b1:proteins:peptide}{protein}, dengan membagi biaya transkripnya di antara \omterm{def:b1:proteins:peptide}{protein}
        pertanyaan 12.
  \item Hitunglah total biaya satu \omterm{def:b1:proteins:peptide}{protein} dan fraksinya yang berasal
        dari translasinya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Kekeliruannya.}
\begin{enumerate}[resume]
  \item Hitunglah peluang bahwa sebuah pembawa pesan tertentu membawa
        sekurang-kurangnya satu kekeliruan transkripsi dalam
        \qty{1503}{} \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} penyandinya.
  \item Hitunglah peluang bahwa sebuah molekul \omterm{def:b1:proteins:peptide}{protein} tertentu
        membawa sekurang-kurangnya satu kekeliruan translasi.
  \item Sekitar seperempat perubahan \omterm{def:b1:nucleic-acids:nucleotide}{nukleotidanya} bisu dan sepertiga
        penggantian \omterm{def:b1:proteins:aminoacid}{asam aminonya} tak merugikan. Berapa fraksi molekul
        \omterm{def:b1:proteins:peptide}{protein} yang dibuat itu yang cacat?
  \item Sebuah pembawa pesan yang cacat menghasilkan \omterm{def:b1:proteins:peptide}{protein} yang cacat
        selama dua jam; sebuah \omterm{def:b1:proteins:peptide}{protein} yang cacat hanyalah satu molekul.
        Jelaskan mengapa \omterm{def:b1:cell-unit-of-life:cell}{sel} sanggup menanggung laju kekeliruan
        translasi sepuluh kali laju transkripsinya, dan keduanya jauh
        di atas laju replikasinya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Anggaran \omterm{def:b1:cell-unit-of-life:cell}{selnya}.}
\omterm{def:b1:cell-unit-of-life:cell}{Sel} itu membuat \num{3000} salinan \omterm{def:b1:proteins:peptide}{protein} ini tiap jam, dan
seluruhnya \num{3e9} residu \omterm{def:b1:proteins:peptide}{protein} tiap hari.
\begin{enumerate}[resume]
  \item Berapa banyak pembawa pesan \omterm{def:b1:genomes:gene}{gen} ini yang harus hadir sekaligus
        (masing-masing menghasilkan sebanyak pertanyaan 11)?
  \item Berapa banyak \omterm{def:b1:gene-expression:ribosome}{ribosom} yang ditempati \omterm{def:b1:proteins:peptide}{protein} ini?
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} harian yang dibelanjakan \omterm{def:b1:cell-unit-of-life:cell}{sel} itu bagi seluruh
        sintesis \omterm{def:b1:proteins:peptide}{proteinnya} (empat tiap residu) dan, pada
        \qty{50}{kJ/mol}, dayanya dalam watt bagi sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}
        \qty{2000}{\micro m^3}.
  \item Bandingkan dengan daya total \omterm{def:b1:cell-unit-of-life:cell}{selnya} bila ia memakai
        \qty{1e9}{} \omterm{def:b1:water-small-molecules:atp}{ATP} tiap sekon.
  \item Sebuah obat menghambat spliceosomnya. Ramalkan pengaruhnya
        pada \omterm{def:b1:proteins:peptide}{protein} ini dan pada sebuah \omterm{def:b1:proteins:peptide}{protein} bakteri.
  \item Nyatakan hasilnya: biaya \omterm{def:b1:water-small-molecules:atp}{ATP} satu molekul \omterm{def:b1:proteins:peptide}{proteinnya}, yang
        dibagi antara translasi dan transkripsinya, serta waktu dari
        awal transkripsinya sampai \omterm{def:b1:proteins:peptide}{protein} jadi yang pertama.
\end{enumerate}
\end{problem}
