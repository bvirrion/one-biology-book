\chapter{Parts of a Plant}\label{ch:g2:plant-parts}

A \omterm{def:g1:plants-animals:plant}{plant} is more than a green blob. It has parts, and each part has
a job. Learn the names, and you can read any garden like a map.

\section{The main parts}

\begin{definition}[Main plant parts]\label{def:g2:plant-parts:main}
Most familiar \omterm{def:g1:plants-animals:plant}{plants} have:
\begin{itemize}
  \item \emph{roots}\index{root} under the ground;
  \item a \emph{stem}\index{stem} that holds the \omterm{def:g1:plants-animals:plant}{plant} up;
  \item \emph{leaves}\index{leaf} that are often green;
  \item sometimes \emph{flowers}\index{flower}, then
        \emph{fruits}\index{fruit} and \emph{seeds}\index{seed}.
\end{itemize}
\end{definition}

\section{Jobs of the parts}

\begin{definition}[Jobs]\label{def:g2:plant-parts:jobs}
\begin{itemize}
  \item \omterm{def:g2:plant-parts:main}{Roots} anchor the \omterm{def:g1:plants-animals:plant}{plant} and take up \omterm{def:g1:caring:needs}{water} and minerals from
        the soil.
  \item The \omterm{def:g2:plant-parts:main}{stem} carries \omterm{def:g1:caring:needs}{water} up and holds \omterm{def:g2:plant-parts:main}{leaves} to the light.
  \item \omterm{def:g2:plant-parts:main}{Leaves} capture light to help the \omterm{def:g1:plants-animals:plant}{plant} make \omterm{def:g1:caring:needs}{food}.
  \item \omterm{def:g2:plant-parts:main}{Flowers} help the \omterm{def:g1:plants-animals:plant}{plant} make \omterm{def:g2:plant-parts:main}{seeds}. \omterm{def:g2:plant-parts:main}{Fruits} protect \omterm{def:g2:plant-parts:main}{seeds} and
        help them travel.
\end{itemize}
\end{definition}

\begin{example}[Pull a weed carefully]\label{ex:g2:plant-parts:weed}
If you pull a dandelion, you feel the long \omterm{def:g2:plant-parts:main}{root}. Leave the \omterm{def:g2:plant-parts:main}{root} in
the soil and the \omterm{def:g1:plants-animals:plant}{plant} may \omterm{def:g2:what-alive:signs}{grow} back. The \omterm{def:g2:plant-parts:main}{root} is the \omterm{def:g1:plants-animals:plant}{plant}'s
anchor and \omterm{def:g1:caring:needs}{water} straw.
\end{example}

\begin{method}[Label a plant]\label{met:g2:plant-parts:label}
\begin{enumerate}
  \item Find the soil line: below $\to$ \omterm{def:g2:plant-parts:main}{roots}; above $\to$ shoot.
  \item Follow the \omterm{def:g2:plant-parts:main}{stem} upward.
  \item Spot \omterm{def:g2:plant-parts:main}{leaves} (flat and green) and any \omterm{def:g2:plant-parts:main}{flower} or \omterm{def:g2:plant-parts:main}{fruit}.
  \item Write the name next to each part.
\end{enumerate}
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g2:plant-parts:1}
Name the four main parts of a typical flowering \omterm{def:g1:plants-animals:plant}{plant}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:plant-parts:2}
Which part is usually underground?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:plant-parts:3}
What job do \omterm{def:g2:plant-parts:main}{leaves} do?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:plant-parts:4}
True or false? ``The \omterm{def:g2:plant-parts:main}{stem} only looks pretty; it has no job.''
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:plant-parts:5}
Match: \omterm{def:g2:plant-parts:main}{roots}, \omterm{def:g2:plant-parts:main}{stem}, \omterm{def:g2:plant-parts:main}{flower} --- to: makes \omterm{def:g2:plant-parts:main}{seeds}; takes \omterm{def:g1:caring:needs}{water} from
soil; holds the \omterm{def:g1:plants-animals:plant}{plant} up.
\end{exercise}
