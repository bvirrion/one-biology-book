\chapter{Reproduction in Organisms}\label{ch:g6:reproduction}

\omterm{def:g5:reproduction:def}{Reproduction} patterns differ across \omterm{def:g1:plants-animals:plant}{plants}, \omterm{def:g1:plants-animals:animal}{animals} and \omterm{def:g4:microbes:def}{microbes}, yet
all pass life to a new generation. This chapter compares sexual and
asexual strategies and walks through \omterm{def:g1:plants-animals:plant}{flowering-plant} and \omterm{def:g1:plants-animals:animal}{animal} \omterm{def:g5:reproduction:lifecycle}{life
cycles} in more detail than Grade~5.

\section{Strategies compared}

\begin{definition}[Asexual reproduction]\label{def:g6:reproduction:asexual}
\emph{Asexual reproduction}\index{asexual reproduction} produces
\omterm{def:g2:growing:offspring}{offspring} from one parent without fusion of \omterm{def:g6:reproduction:sexual}{gametes}. \omterm{def:g2:growing:offspring}{Offspring} are
usually genetically very similar to the parent.
\end{definition}

\begin{definition}[Sexual reproduction]\label{def:g6:reproduction:sexual}
\emph{Sexual reproduction}\index{sexual reproduction} involves
\emph{gametes}\index{gamete} and \emph{fertilisation}\index{fertilisation},
mixing genetic information. \omterm{def:g2:growing:offspring}{Offspring} vary.
\end{definition}

\begin{example}[Who uses what?]\label{ex:g6:reproduction:who}
\omterm{def:g4:microbes:def}{Bacteria} divide \omterm{prop:g5:reproduction:two}{asexually}. Many \omterm{def:g1:plants-animals:plant}{plants} use both \omterm{def:g2:plant-parts:main}{flowers} (sexual) and
runners or bulbs (asexual). \omterm{def:g2:animal-groups:five}{Mammals} \omterm{def:g2:what-alive:signs}{reproduce} \omterm{prop:g5:reproduction:two}{sexually}.
\end{example}

\section{Flowering plants}

\begin{definition}[Pollination and seed]\label{def:g6:reproduction:plant}
\emph{Pollination}\index{pollination} transfers pollen to a stigma.
After \omterm{def:g6:reproduction:sexual}{fertilisation}, ovules become \emph{seeds}\index{seed} and ovaries
often become \emph{fruits}\index{fruit}. Seeds disperse and may
germinate.
\end{definition}

\omimg{seed-sprouting.jpg}{0.52\linewidth}{\omterm{def:g3:seeds:germ}{Germination} starts the next
\omterm{def:g1:plants-animals:plant}{plant} generation when \omterm{def:g4:plants-need:needs}{water}, temperature and oxygen allow it.}

\begin{method}[From flower to seedling]\label{met:g6:reproduction:flower}
\begin{enumerate}
  \item \omterm{def:g2:plant-parts:main}{Flower} produces pollen and ovules.
  \item \omterm{def:g6:reproduction:plant}{Pollination} (wind, \omterm{def:g3:insects:insect}{insects} or other \omterm{def:g1:plants-animals:animal}{animals}).
  \item \omterm{def:g6:reproduction:sexual}{Fertilisation} inside the \omterm{def:g2:plant-parts:main}{flower}.
  \item \omterm{def:g6:reproduction:plant}{Seed} and \omterm{def:g6:reproduction:plant}{fruit} develop.
  \item Dispersal; \omterm{def:g3:seeds:germ}{germination} if conditions suit.
\end{enumerate}
\end{method}

\begin{example}[Insect pollination clues]\label{ex:g6:reproduction:insect}
Bright petals, scent and nectar often attract \omterm{def:g3:insects:insect}{insects} that carry pollen
between \omterm{def:g2:plant-parts:main}{flowers} --- a mutual benefit.
\end{example}

\section{Animals}

\begin{definition}[External and internal fertilisation]\label{def:g6:reproduction:fert}
\emph{External fertilisation}\index{fertilisation!external} joins \omterm{def:g6:reproduction:sexual}{gametes}
outside the \omterm{def:g1:my-body:body}{body} (many \omterm{def:g5:vertebrates:groups}{fish} and \omterm{def:g2:animal-groups:five}{amphibians} in \omterm{def:g4:plants-need:needs}{water}).
\emph{Internal fertilisation}\index{fertilisation!internal} joins them
inside the \omterm{def:g1:my-body:body}{body} (many \omterm{def:g2:animal-groups:five}{reptiles}, \omterm{def:g2:animal-groups:five}{birds}, \omterm{def:g2:animal-groups:five}{mammals}).
\end{definition}



\begin{definition}[Development patterns]\label{def:g6:reproduction:dev}
Some young resemble miniature \omterm{def:g3:life-cycles:butterfly}{adults} (many \omterm{def:g2:animal-groups:five}{mammals}). Others transform
through larval stages (\omterm{def:g3:insects:insect}{insects}, \omterm{def:g2:animal-groups:five}{amphibians}). \omterm{def:g3:life-cycles:butterfly}{Eggs} may develop outside
the parent or, in many \omterm{def:g2:animal-groups:five}{mammals}, inside the uterus with placental
support --- human details belong in specialised \omterm{def:g2:healthy:health}{health} education.
\end{definition}

\section{Choosing a strategy}

\begin{proposition}[Trade-offs]\label{prop:g6:reproduction:tradeoff}
\omterm{def:g6:reproduction:asexual}{Asexual reproduction} can be fast and \omterm{def:g1:caring:needs}{needs} one parent. \omterm{def:g6:reproduction:sexual}{Sexual
reproduction} produces \omterm{def:g5:evolution:variation}{variation} that may help \omterm{def:g6:ecology:levels}{populations} face change
or disease. Many \omterm{def:g6:classification:species}{species} combine or switch strategies.
\end{proposition}
\admitted
