\section*{Chapter \ref{ch:g7:nervous} --- The Nervous System}

\begin{solution}{exo:g7:nervous:1}
CNS: \omterm{def:g5:nerves:system}{brain} and \omterm{def:g5:nerves:system}{spinal cord}. Peripheral: \omterm{def:g5:nerves:system}{nerves} outside the CNS connecting
to receptors and effectors.
\end{solution}

\begin{solution}{exo:g7:nervous:2}
\omterm{def:g7:nervous:stim}{Stimulus}: detectable change; \omterm{def:g7:nervous:stim}{response}: resulting action. Example: bright
\omterm{def:g4:plants-need:needs}{light} $\to$ pupils constrict (or doorbell $\to$ stand up).
\end{solution}

\begin{solution}{exo:g7:nervous:3}
\omterm{def:g7:nervous:neuron}{Dendrites} receive; \omterm{def:g7:nervous:neuron}{cell body} integrates; \omterm{def:g7:nervous:neuron}{axon} carries signal away toward
\omterm{def:g7:nervous:neuron}{synapse}/next \omterm{def:g7:microscope:cell}{cell}.
\end{solution}

\begin{solution}{exo:g7:nervous:4}
Sensory: receptor $\to$ CNS. Motor: CNS $\to$ effector (\omterm{def:g3:bones:muscle}{muscle}/gland).
\end{solution}

\begin{solution}{exo:g7:nervous:5}
Fast automatic \omterm{def:g7:nervous:stim}{response}; protects the \omterm{def:g1:my-body:body}{body} and frees attention for other
tasks.
\end{solution}
