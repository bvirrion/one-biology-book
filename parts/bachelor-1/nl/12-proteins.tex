\chapter{Aminozuren en eiwitten}\label{ch:b1:proteins}

Kook een ei en het heldere wit wordt ondoorzichtig en stevig; er is niets
toegevoegd of weggenomen, maar elk eiwitmolecuul erin is ontvouwen en met zijn
buren verstrengeld geraakt. Koel het af en het blijft gekookt. Toch vouwt een
klein enzym dat in een reageerbuis voorzichtig is ontvouwen en daarna in een kale
\omterm{def:b1:water-small-molecules:buffer}{buffer} wordt gelaten, zich binnen minuten terug in zijn precieze werkvorm, zonder
hulp van wat dan ook: de vorm stond in de keten geschreven. \omterm{def:b1:proteins:peptide}{Eiwitten} zijn de
werkmoleculen van de \omterm{def:b1:cell-unit-of-life:cell}{cel} --- haar enzymen, pompen, motoren, geraamtes,
antistoffen, hormonen --- en elk ervan is een keten van \omterm{def:b1:proteins:aminoacid}{aminozuren}, gevouwen tot
een vorm die haar sequentie voorschrijft. Dit hoofdstuk beschrijft de twintig
\omterm{def:b1:proteins:aminoacid}{aminozuren}, de binding die hen verbindt, de vier niveaus van eiwitstructuur, de
vouwing die van een keten een machine maakt, het coöperatieve gedrag van \omterm{def:b1:proteins:peptide}{eiwitten}
met verscheidene subeenheden, en de methoden waarmee \omterm{def:b1:proteins:peptide}{eiwitten} worden gescheiden en
zichtbaar gemaakt.

\section{Aminozuren}

\begin{definition}[Aminozuur]\label{def:b1:proteins:aminoacid}
Een \emph{aminozuur}\index{aminozuur} heeft een centrale
\emph{$\alpha$-koolstof} met een aminogroep ($-\mathrm{NH_2}$), een carboxylgroep
($-\mathrm{COOH}$), een waterstof en een \emph{zijketen}\index{zijketen} $R$ die
de twintig standaardaminozuren van \omterm{def:b1:proteins:peptide}{eiwitten} onderscheidt. Bij de pH van de \omterm{def:b1:cell-unit-of-life:cell}{cel} is
de aminogroep geprotoneerd en de carboxylgroep geïoniseerd: het molecuul is een
\emph{zwitterion}\index{zwitterion}, $\mathrm{^+H_3N{-}CHR{-}COO^-}$, zonder netto
lading. De $\alpha$-koolstof is bij alle aminozuren behalve glycine chiraal, en
levende wezens gebruiken alleen de L-enantiomeer. De zijketens groeperen zich naar
hun chemie:
\begin{center}
\footnotesize
\begin{tabular}{lll}
\hline
klasse & aminozuren (drie- en éénlettercodes) & zijketen \\
\hline
apolair, alifatisch & Gly G, Ala A, Val V, Leu L, Ile I, Met M, Pro P & koolwaterstof; Pro een ring \\
aromatisch & Phe F, Tyr Y, Trp W & ringen; Tyr en Trp nemen UV op \\
\omterm{def:b1:water-small-molecules:water}{polair}, ongeladen & Ser S, Thr T, Cys C, Asn N, Gln Q & $-$OH, $-$SH, amide \\
positief geladen & Lys K, Arg R, His H & basen; His p$K_a$ 6.0 \\
negatief geladen & Asp D, Glu E & carboxylaten, p$K_a$ 4 \\
\hline
\end{tabular}
\end{center}
Negen ervan (His, Ile, Leu, Lys, Met, Phe, Thr, Trp, Val) zijn voor de mens
\emph{essentieel}: wij kunnen ze niet maken en moeten ze eten.
\end{definition}

\begin{proposition}[Ionisatie]\label{prop:b1:proteins:ionisation}
Elke ioniseerbare groep van een \omterm{def:b1:proteins:aminoacid}{aminozuur} heeft haar eigen p$K_a$: ongeveer 2 voor
de $\alpha$-carboxylgroep, 9.5 voor de $\alpha$-aminogroep, en voor de \omterm{def:b1:proteins:aminoacid}{zijketens}
3.9 (Asp), 4.1 (Glu), 6.0 (His), 8.3 (Cys), 10.1 (Tyr), 10.5 (Lys), 12.5 (Arg).
Het \emph{iso-elektrisch punt}\index{iso-elektrisch punt} p$I$ is de pH waarbij de
netto lading nul is --- het gemiddelde van de twee p$K_a$ die de neutrale vorm
omsluiten; de p$I$ van een \omterm{def:b1:proteins:peptide}{eiwit} wordt bepaald door zijn gehalte aan geladen
\omterm{def:b1:proteins:aminoacid}{zijketens}, en bij die pH is het het minst oplosbaar en beweegt het niet in een
elektrisch veld. Histidine, met een p$K_a$ dicht bij 7, is het residu dat binnen
het fysiologische bereik van lading verandert, en het zit in de actieve plaats van
veel enzymen.
\end{proposition}

\section{De peptidebinding}

\begin{definition}[Peptidebinding, polypeptide]\label{def:b1:proteins:peptide}
De \emph{peptidebinding}\index{peptidebinding} verbindt door condensatie de
carboxylgroep van het ene \omterm{def:b1:proteins:aminoacid}{aminozuur} met de aminogroep van het volgende (een
amidebinding, $\mathrm{-CO{-}NH-}$). Een zo verbonden keten \omterm{def:b1:proteins:aminoacid}{aminozuren} is een
\emph{polypeptide}\index{polypeptide}: een \emph{ruggengraat} van zich herhalende
$\mathrm{N{-}C_\alpha{-}C}$-eenheden waaruit de \omterm{def:b1:proteins:aminoacid}{zijketens} steken, met een
\emph{N-uiteinde} (vrije aminogroep) en een \emph{C-uiteinde} (vrije
carboxylgroep). Sequenties worden van N naar C geschreven, de volgorde van de
synthese (\cref{ch:b1:gene-expression}). Een \emph{eiwit}\index{eiwit} is een of
meer polypeptiden gevouwen tot een bepaalde vorm; gebruikelijke ketens tellen
\qtyrange{100}{1000}{} residuen met een gemiddelde massa van \qty{110}{Da}, zodat
een eiwit van \qty{300}{} residuen \qty{33}{kDa} weegt.
\end{definition}

\begin{proposition}[Meetkunde van de peptidebinding]\label{prop:b1:proteins:geometry}
De C--N-binding van een peptide heeft gedeeltelijk het karakter van een dubbele
binding (de amideresonantie): zij is kort, kan niet draaien, en houdt de zes
atomen $\mathrm{C_\alpha{-}C(=O){-}N(H){-}C_\alpha}$ in één vlak, vrijwel altijd
met de twee $\alpha$-koolstoffen \emph{trans}. De ruggengraat is daardoor een
keten van stijve vlakken die aan de $\alpha$-koolstoffen scharnieren, waarbij elk
scharnier twee draaiingen heeft, $\phi$ (om N--$\mathrm{C_\alpha}$) en $\psi$ (om
$\mathrm{C_\alpha}$--C), en botsingen tussen \omterm{def:b1:proteins:aminoacid}{zijketens} en ruggengraat de meeste
combinaties van $\phi$ en $\psi$ verbieden. De conformatie van een \omterm{def:b1:proteins:peptide}{eiwit} is in
wezen de lijst van zijn $\phi$- en $\psi$-hoeken.
\end{proposition}

\section{Secundaire structuur}

\begin{definition}[Secundaire structuur]\label{def:b1:proteins:secondary}
De \emph{secundaire structuur}\index{secundaire structuur} van een \omterm{def:b1:proteins:peptide}{eiwit} is de
plaatselijke, regelmatige vouwing van zijn ruggengraat, vastgehouden door
\omterm{def:b1:water-small-molecules:water}{waterstofbruggen} tussen de C=O- en N--H-groepen van de ruggengraat. Twee patronen
overheersen. De \emph{$\alpha$-helix}\index{alpha helix@$\alpha$-helix}: een
rechtsdraaiende spiraal van 3.6 residuen per winding, die \qty{0.15}{nm} per
residu stijgt (\qty{0.54}{nm} per winding), waarbij elke C=O is gebonden aan de
N--H vier residuen verderop en de \omterm{def:b1:proteins:aminoacid}{zijketens} naar buiten wijzen. De
\emph{$\beta$-plaat}\index{beta sheet@$\beta$-plaat}: ketens die vrijwel volledig
zijn gestrekt (\qty{0.35}{nm} per residu) en zij aan zij liggen, evenwijdig of
antiparallel, met bruggen tussen naburige strengen en \omterm{def:b1:proteins:aminoacid}{zijketens} die om en om boven
en onder de plaat uitsteken. Daartussen keren \emph{bochten} en lussen de richting
van de keten om. Proline, waarvan de ring $\phi$ vastlegt, breekt helices;
glycine, zonder \omterm{def:b1:proteins:aminoacid}{zijketen}, laat bochten toe die geen ander residu kan maken.
\end{definition}

\begin{example}[Helix en plaat in getallen]\label{ex:b1:proteins:numbers}
Een $\alpha$-helix die een membraan doorkruist moet \qty{3}{nm} hydrofobe kern
overbruggen: $3/0.15 = 20$ residuen, en inderdaad zijn de transmembraansegmenten
van \cref{ch:b1:membranes-transport} reeksen van ongeveer twintig hydrofobe
residuen. Een $\beta$-streng van dezelfde lengte overspant \qty{7}{nm}:
zijdefibroïne bestaat uit gestapelde antiparallelle platen van glycine en alanine,
en een zijdedraad is sterker dan staal van hetzelfde gewicht omdat de covalente
ruggengraten in de lengte van de vezel liggen.
\end{example}

\section{Tertiaire en quaternaire structuur}

\begin{definition}[Tertiaire en quaternaire structuur]\label{def:b1:proteins:tertiary}
De \emph{tertiaire structuur}\index{tertiaire structuur} is de volledige
driedimensionale schikking van één \omterm{def:b1:proteins:peptide}{polypeptide}: zijn helices, platen en lussen
samengepakt, vaak in verscheidene compacte \emph{domeinen} die zich onafhankelijk
vouwen. Zij wordt vastgehouden door niet-covalente krachten --- het
\emph{\omterm{prop:b1:water-small-molecules:properties}{hydrofobe effect}} dat apolaire \omterm{def:b1:proteins:aminoacid}{zijketens} in een kern weg van het water
begraaft, \omterm{def:b1:water-small-molecules:water}{waterstofbruggen}, zoutbruggen tussen geladen \omterm{def:b1:proteins:aminoacid}{zijketens},
vanderwaalspakking --- en soms door
\emph{disulfidebruggen}\index{disulfidebrug}, covalente S--S-bindingen tussen twee
cysteïnen, in \omterm{def:b1:proteins:peptide}{eiwitten} die naar de oxiderende buitenwereld worden afgescheiden. De
\emph{quaternaire structuur}\index{quaternaire structuur} is de samenvoeging van
verscheidene \omterm{def:b1:proteins:peptide}{polypeptiden} (subeenheden) tot één \omterm{def:b1:proteins:peptide}{eiwit}: hemoglobine bestaat uit
twee $\alpha$- en twee $\beta$-ketens. \emph{Globulaire} \omterm{def:b1:proteins:peptide}{eiwitten} zijn compact en
wateroplosbaar (enzymen, \omterm{def:b1:membranes-transport:transporters}{carriers}); \emph{fibrillaire} \omterm{def:b1:proteins:peptide}{eiwitten} zijn langgerekt en
onoplosbaar (de drievoudige helix van \omterm{def:b1:body-plans-tissues:connective}{collageen}, de gewonden spiralen van keratine
en myosine, zijde).
\end{definition}

\begin{theorem}[De sequentie bepaalt de vouwing]\label{thm:b1:proteins:anfinsen}
\index{eiwitvouwing}
De driedimensionale structuur van een \omterm{def:b1:proteins:peptide}{eiwit} wordt bepaald door zijn
aminozuursequentie: de natieve vouwing is de conformatie met de laagste \omterm{prop:b1:water-small-molecules:gibbs}{vrije
energie} die de keten in haar cellulaire omgeving kan bereiken, en de informatie
die daarvoor nodig is zit in de keten zelf.
\end{theorem}
\begin{proof}[Bewijsmateriaal]
Anfinsen (1961) ontvouwde ribonuclease A, een enzym van \qty{124}{} residuen met
vier \omterm{def:b1:proteins:tertiary}{disulfidebruggen}, in geconcentreerd ureum met een reducerend middel dat de
bruggen verbrak: de keten verloor alle activiteit. Door het ureum te verwijderen en
de cysteïnen in lucht langzaam opnieuw te laten oxideren, kreeg hij het enzym
volledig actief terug, met zijn vier bruggen opnieuw gevormd tussen dezelfde acht
cysteïnen uit de \qty{105}{} mogelijke paringen --- de keten had haar eigen
vouwing gevonden. Opnieuw geoxideerd terwijl zij nog in ureum zat, vormde zij
willekeurige bruggen en bleef zij inactief; een spoor van het reducerende middel
liet haar die vervolgens herschikken tot de natieve, meest stabiele reeks was
bereikt. Sindsdien zijn duizenden sequenties in de reageerbuis vanuit het niets
gevouwen, en kan de vouwing van een nieuwe sequentie eruit worden berekend.
\end{proof}

\begin{proposition}[Denaturatie]\label{prop:b1:proteins:denaturation}
\index{denaturatie (eiwit)}
Warmte, extreme pH, ureum, detergentia of organische oplosmiddelen ontvouwen
\omterm{def:b1:proteins:peptide}{eiwitten} door de zwakke bindingen te overweldigen die de vouwing bijeenhouden: de
keten verliest haar vorm en haar functie, legt haar hydrofobe kern bloot en klontert
met haar buren. Kleine \omterm{def:b1:proteins:peptide}{eiwitten} vouwen zich terug wanneer het denaturerende middel
wordt verwijderd; grote, en \omterm{def:b1:proteins:peptide}{eiwitten} in een volgepakte \omterm{def:b1:cell-unit-of-life:cell}{cel}, kunnen dat vaak niet,
en de \omterm{def:b1:cell-unit-of-life:cell}{cel} houdt er \emph{chaperonnes}\index{chaperonne} op na --- \omterm{def:b1:proteins:peptide}{eiwitten} die
ontvouwen ketens binden, hun hydrofobe oppervlakken afschermen en hun herhaalde
kansen geven om correct te vouwen --- en breekt af wat verkeerd gevouwen blijft.
Koorts van enkele graden, en de \qty{42}{\celsius} waarbij hersenei\-witten beginnen
te ontvouwen, tekenen de smalle marge van stabiliteit: een gebruikelijke vouwing is
maar \qtyrange{20}{60}{kJ/mol} stabieler dan de ontvouwen keten, de waarde van
enkele \omterm{def:b1:water-small-molecules:water}{waterstofbruggen}.
\end{proposition}

\begin{example}[Collageen]\label{ex:b1:proteins:collagen}
Een kwart van het \omterm{def:b1:proteins:peptide}{eiwit} van een zoogdier is \omterm{def:b1:body-plans-tissues:connective}{collageen}: drie ketens, elk een
linksdraaiende helix van duizend residuen met glycine op elke derde plaats
(Gly--X--Y, met Y vaak hydroxyproline), die samen tot een rechtsdraaiende
drievoudige helix van \qty{300}{nm} lang zijn gewonden en daarna zij aan zij tot
fibrillen worden gepakt die covalent zijn dwarsverbonden. Dat glycine geen \omterm{def:b1:proteins:aminoacid}{zijketen}
heeft, is wat de drie ketens bij de as laat samenkomen; één vervanging van glycine
door welk ander residu ook knikt de helix en geeft brosse botten. Vitamine C is
nodig om de prolinen te hydroxyleren; zonder haar is de helix onstabiel en falen de
fibrillen: scheurbuik.
\end{example}

\section{Allosterie: hemoglobine}

\begin{definition}[Allosterie, coöperativiteit]\label{def:b1:proteins:allostery}
Een \omterm{def:b1:proteins:peptide}{eiwit} is \emph{allosterisch}\index{allosterie} wanneer het binden van een
ligand op de ene plaats de conformatie van het \omterm{def:b1:proteins:peptide}{eiwit} verandert en daarmee zijn
affiniteit op een andere plaats. Wanneer de plaatsen gelijk zijn en de verandering
de affiniteit van de andere verhoogt, is het binden
\emph{coöperatief}\index{coöperativiteit}: de verzadigingscurve is sigmoïde in
plaats van hyperbolisch, en het \omterm{def:b1:proteins:peptide}{eiwit} schakelt over een smal bereik van
ligandconcentratie van vrijwel leeg naar vrijwel vol. De
\emph{hillvergelijking}\index{hillvergelijking} beschrijft dat:
\[
Y = \frac{P^n}{P_{50}^n + P^n},
\]
waarin $Y$ het deel van de bezette plaatsen is, $P$ de ligandconcentratie (of
partiële druk), $P_{50}$ de waarde bij halve verzadiging, en $n$ de
hillcoëfficiënt --- 1 voor onafhankelijke plaatsen, tot het aantal plaatsen bij
volmaakte coöperativiteit.
\end{definition}

\begin{proposition}[Hemoglobine en myoglobine]\label{prop:b1:proteins:hemoglobin}
Myoglobine, één keten met één heem, bindt $\mathrm{O_2}$ langs een hyperbool
($n = 1$, $P_{50} = \qty{0.37}{kPa}$): het is verzadigd bij elke druk die de
\omterm{def:b1:body-plans-tissues:tissue}{weefsels} bieden en dient als voorraad in de spier. Hemoglobine, vier
subeenheden, bindt \omterm{def:b1:proteins:allostery}{coöperatief} ($n \approx 2.8$, $P_{50} = \qty{3.5}{kPa}$): de
eerste gebonden $\mathrm{O_2}$ verschuift het viertal van een \emph{T-toestand}
met lage affiniteit naar een \emph{R-toestand} met hoge affiniteit, zodat het bij
de \qty{13}{kPa} van de long vrijwel volledig laadt en bij de \qty{5}{kPa} van de
\omterm{def:b1:body-plans-tissues:tissue}{weefsels} een groot deel lost. De affiniteit wordt verder verlaagd, en het lossen
bevorderd, door \omterm{def:b1:water-small-molecules:ph}{zuur} en $\mathrm{CO_2}$ (het \emph{bohreffect}) en door
2,3-bisfosfoglyceraat (BPG), dat de T-toestand bindt: in een werkende spier, of op
hoogte, komt bij dezelfde druk meer zuurstof vrij. De foetus maakt een hemoglobine
dat BPG zwak bindt, zodat zijn bloed zuurstof over de placenta van de moeder
aantrekt.
\end{proposition}

\begin{method}[De hillvergelijking gebruiken]\label{met:b1:proteins:hill}
\begin{enumerate}
  \item Lees $P_{50}$ af uit de curve bij $Y = 0.5$; schat $n$ uit de
        helling van $\log[Y/(1-Y)]$ tegen $\log P$ (de hillplot), die een
        rechte lijn met helling $n$ is in de buurt van $P_{50}$.
  \item Bereken $Y$ bij de laaddruk (longen) en bij de losdruk (\omterm{def:b1:body-plans-tissues:tissue}{weefsels});
        het verschil is het deel van de draagcapaciteit dat wordt
        afgeleverd.
  \item Vermenigvuldig met de capaciteit: bloed bevat \qty{150}{g/L}
        hemoglobine, waarbij elke gram \qty{1.34}{mL} $\mathrm{O_2}$ bindt,
        oftewel \qty{200}{mL} $\mathrm{O_2}$ per liter bij verzadiging.
  \item Om het bohreffect of BPG mee te nemen, gebruik de verschoven
        $P_{50}$ alleen bij de \omterm{def:b1:body-plans-tissues:tissue}{weefsels}, waar het \omterm{def:b1:water-small-molecules:ph}{zuur} zit.
\end{enumerate}
\end{method}

\section{Eiwitten zien en scheiden}

\begin{method}[Eiwitten scheiden]\label{met:b1:proteins:methods}
\begin{enumerate}
  \item \emph{Elektroforese in SDS}: het detergens bekleedt elke keten met
        negatieve lading naar rato van haar lengte, zodat de ketens alleen
        op grootte door een polyacrylamidegel migreren; een ladder van
        bekende massa's ijkt de gel, en de plaats van een gekleurde band
        geeft de massa op enkele procenten nauwkeurig.
  \item \emph{Chromatografie}: door een kolom met korrels die \omterm{def:b1:proteins:peptide}{eiwitten}
        ophouden op grootte (gelfiltratie: de grote komen het eerst naar
        buiten), op lading (ionenwisseling), of op specifieke binding
        (affiniteit: een vastgezet ligand houdt één \omterm{def:b1:proteins:peptide}{eiwit} vast en laat de
        rest door). Een zuivering wordt gevolgd aan de stijging van de
        specifieke activiteit (\cref{ch:b1:cell-unit-of-life}).
  \item \emph{Sequencing}: de sequentie van het gen, vertaald; of het \omterm{def:b1:proteins:peptide}{eiwit}
        zelf, in peptiden geknipt en met massaspectrometrie gelezen.
  \item \emph{Structuur}: röntgendiffractie van een kristal, of
        elektronenmicroscopie van bevroren losse deeltjes, geeft de plaats
        van elk atoom; de lintdiagrammen van dit hoofdstuk zijn daarvan de
        samenvattingen.
\end{enumerate}
\end{method}

\begin{example}[Een gel lezen]\label{ex:b1:proteins:gel}
Een ruw extract vertoont tientallen banden; na affiniteitschromatografie blijft er
één band over op de plaats van de merker van \qty{33}{kDa}. Zonder SDS en zonder
reducerend middel gelopen, migreert hetzelfde \omterm{def:b1:proteins:peptide}{eiwit} als een deeltje van
\qty{66}{kDa}: het is een dimeer van twee identieke ketens, bijeengehouden door een
\omterm{def:b1:proteins:tertiary}{disulfidebrug} die het reducerende middel verbreekt en het detergens scheidt. Twee
gels, één redenering, en de \omterm{def:b1:proteins:tertiary}{quaternaire structuur} is bekend.
\end{example}

\section{Vraagstuk: hemoglobine aan het werk}

\begin{problem}[{Weekendvraagstuk --- zuurstof van long naar spier gedragen, in
rust, in een sprint, op een berg en vóór de geboorte, eindigend op de zuurstof die
per liter bloed wordt afgeleverd}]
\label{pb:b1:proteins:1}
Bloed bevat \qty{150}{g/L} hemoglobine; \qty{1}{g} bindt \qty{1.34}{mL}
$\mathrm{O_2}$. Hemoglobine gehoorzaamt bij pH 7.4 aan de \omterm{def:b1:proteins:allostery}{hillvergelijking} met
$n = 2.8$ en $P_{50} = \qty{3.5}{kPa}$; myoglobine heeft $n = 1$ en
$P_{50} = \qty{0.37}{kPa}$. Partiële drukken van $\mathrm{O_2}$: \qty{13.3}{kPa}
in de longen op zeeniveau, \qty{5.3}{kPa} in rustende \omterm{def:b1:body-plans-tissues:tissue}{weefsels}. Neem
$2.8\ln 3.5 = 3.51$, $2.8\ln 5.3 = 4.67$, $2.8\ln 13.3 = 7.25$,
$2.8\ln 2.7 = 2.78$, $2.8\ln 4.5 = 4.21$, $2.8\ln 7 = 5.45$,
$2.8\ln 4 = 3.88$, $2.8\ln 2.6 = 2.68$.

\medskip
\noindent\textbf{Deel I --- In rust.}
\begin{enumerate}
  \item Bereken de zuurstofcapaciteit van een liter bloed.
  \item Bereken $P^{2.8}$ voor $P = 3.5$, $5.3$ en $13.3$ kPa (als
        $e^{2.8\ln P}$).
  \item Bereken de verzadiging $Y$ van hemoglobine in de longen en in de
        \omterm{def:b1:body-plans-tissues:tissue}{weefsels}.
  \item Bereken het afgeleverde deel van de lading en de zuurstof die per
        liter bloed wordt afgeleverd, in milliliters.
  \item Een mens in rust verbruikt \qty{250}{mL} $\mathrm{O_2}$ per minuut.
        Welk hartminuutvolume vraagt die aflevering?
  \item Bereken de verzadiging van myoglobine bij dezelfde twee drukken en
        de zuurstof die het per liter zou afleveren als het hemoglobine
        verving. Trek een conclusie.
  \item Leg in één zin uit wat \omterm{def:b1:proteins:allostery}{coöperativiteit} oplevert.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Een sprint.}
In een sprintende spier daalt de pH tot 7.2 en de plaatselijke $\mathrm{O_2}$-druk
tot \qty{2.7}{kPa}; het bohreffect verhoogt $P_{50}$ daar tot \qty{4.5}{kPa}.
\begin{enumerate}[resume]
  \item Bereken $Y$ bij \qty{2.7}{kPa} met $P_{50} = \qty{3.5}{kPa}$
        (zonder bohreffect).
  \item Bereken $Y$ bij \qty{2.7}{kPa} met $P_{50} = \qty{4.5}{kPa}$.
  \item Bereken de zuurstof die in beide gevallen per liter wordt
        afgeleverd (het laden in de longen blijft gelijk) en de winst van
        het bohreffect.
  \item Het myoglobine van de spier zit op $P = \qty{2.7}{kPa}$. Hoe
        verzadigd is het, en wat gebeurt ermee als de druk tijdens een
        samentrekking tot \qty{0.5}{kPa} daalt?
  \item Leg uit waarom het bohreffect een vorm van \omterm{def:b1:proteins:allostery}{allosterie} is en waar de
        protonen aangrijpen.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Een berg.}
Op \qty{4000}{m} is de $\mathrm{O_2}$-druk in de longen \qty{7}{kPa}. Binnen dagen
verhogen de rode \omterm{def:b1:cell-unit-of-life:cell}{cellen} hun BPG, wat $P_{50}$ naar \qty{4.0}{kPa} verschuift;
binnen weken brengt het beenmerg het hemoglobine op \qty{180}{g/L}.
\begin{enumerate}[resume]
  \item Bereken de verzadiging in de longen bij \qty{7}{kPa} met
        $P_{50} = \qty{3.5}{kPa}$, en de aflevering per liter (\omterm{def:b1:body-plans-tissues:tissue}{weefsels} op
        \qty{5.3}{kPa}). Vergelijk dat met zeeniveau.
  \item Bereken de verzadiging in longen en \omterm{def:b1:body-plans-tissues:tissue}{weefsels} met
        $P_{50} = \qty{4.0}{kPa}$, en de aflevering.
  \item Leg uit waarom het verlagen van de affiniteit helpt, hoewel het het
        laden in de longen verlaagt.
  \item Bereken de aflevering per liter met $P_{50} = \qty{4.0}{kPa}$ en
        \qty{180}{g/L} hemoglobine.
  \item Zou een nog lagere affiniteit ($P_{50} = \qty{6}{kPa}$) bij
        \qty{7}{kPa} helpen? Bereken de twee verzadigingen en beslis.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Vóór de geboorte.}
Foetaal hemoglobine heeft $P_{50} = \qty{2.6}{kPa}$; in de placenta is de
$\mathrm{O_2}$-druk aan beide zijden \qty{4}{kPa}.
\begin{enumerate}[resume]
  \item Bereken de verzadiging van het hemoglobine van de moeder bij
        \qty{4}{kPa}.
  \item Bereken de verzadiging van foetaal hemoglobine bij \qty{4}{kPa}.
  \item Leg uit hoe zuurstof van moeder naar foetus gaat hoewel de druk aan
        beide zijden gelijk is.
  \item Foetaal hemoglobine heeft $\gamma$-ketens in plaats van $\beta$,
        die BPG zwak binden. Leg uit hoe dat de lagere $P_{50}$ oplevert.
  \item Bij de geboorte moet het foetale bloed lossen in \omterm{def:b1:body-plans-tissues:tissue}{weefsels} op
        \qty{5.3}{kPa}. Bereken de foetale aflevering per liter vanuit
        longen op \qty{13.3}{kPa}, en zeg waarom de overgang naar
        volwassen hemoglobine in de eerste maanden nuttig is.
  \item Een mutatie brengt de volwassen $P_{50}$ op \qty{5}{kPa}. Voorspel
        de slagaderlijke verzadiging van die persoon op zeeniveau, en het
        gevolg voor de aflevering.
  \item Leg uit waarom een drager met $P_{50} = \qty{3.5}{kPa}$ en zonder
        \omterm{def:b1:proteins:allostery}{coöperativiteit} ($n = 1$) een slechte zuurstofdrager zou zijn:
        bereken zijn aflevering.
  \item Geef het resultaat: de zuurstof die per liter bloed wordt
        afgeleverd in rust en in een sprintende spier, en de twee
        eigenschappen van hemoglobine die het verschil maken.
\end{enumerate}
\end{problem}
