\chapter{Las cadenas alimentarias}\label{ch:g3:food-chains}

En el prado crece la hierba. Un saltamontes se come la hierba. Una
lagartija atrapa al saltamontes. Un ratonero, que da vueltas allá
arriba, se lleva a la lagartija. Cuatro comidas --- y de repente la
tarde del ratonero depende de lo verde que creciera la hierba. Las
comidas de la naturaleza se enlazan en cadenas, y aprender a leerlas
es aprender cómo se sostiene un prado entero.

\section{Leer y escribir una cadena alimentaria}

\begin{definition}[Cadena alimentaria]\label{def:g3:food-chains:chain}
Una \emph{cadena alimentaria}\index{cadena alimentaria} es una fila de
\omterm{def:g1:living-or-not:living}{seres vivos} en la que a cada uno se lo come el siguiente. Se escribe
con flechas, y cada flecha significa \emph{``es comido
por''}\index{es comido por}:
\[
\text{hierba} \longrightarrow \text{saltamontes}
\longrightarrow \text{lagartija} \longrightarrow \text{ratonero}.
\]
Cada \omterm{def:g1:living-or-not:living}{ser vivo} es un \emph{eslabón}\index{eslabón} de la cadena.
\end{definition}

\begin{remark}[Cuidado con el sentido de la flecha]\label{rem:g3:food-chains:arrow}
La flecha va del comido al que come --- sigue al alimento. ``Hierba
$\rightarrow$ saltamontes'' se lee ``la hierba es comida por el
saltamontes''. Escribir las flechas al revés --- apuntando a la
comida --- es el error más frecuente en las \omterm{def:g3:food-chains:chain}{cadenas alimentarias};
acuérdate de que la flecha enseña adónde \emph{va} el alimento.
\end{remark}

\begin{proposition}[Toda cadena empieza por una planta]\label{prop:g3:food-chains:start}
El primer eslabón de una \omterm{def:g3:food-chains:chain}{cadena alimentaria} es siempre una \omterm{def:g1:plants-around-us:plant}{planta} (o
algún otro \omterm{def:g1:living-or-not:living}{ser vivo} que fabrica su propio alimento). Solo las \omterm{def:g1:plants-around-us:plant}{plantas}
se alimentan sin comerse a nadie --- de la luz, el agua y el aire,
como decía la \cref{def:g1:plants-around-us:plant}. Todos los eslabones
\omterm{def:g1:animals-around-us:animal}{animales} viven, directamente o a través de otros, de ese primer
eslabón verde.
\end{proposition}
\admitted

\begin{example}[Cadenas de tres hábitats]\label{ex:g3:food-chains:three}
En la charca: \omterm{def:g1:plants-around-us:plant}{plantas} acuáticas $\rightarrow$ renacuajo
$\rightarrow$ larva de libélula $\rightarrow$ espinoso $\rightarrow$
garza. En el bosque: \omterm{def:g1:plants-around-us:parts}{hojas} de roble $\rightarrow$ \omterm{ex:g2:animals-grow-up:butterfly}{oruga} $\rightarrow$
herrerillo $\rightarrow$ gavilán. En el jardín: lechuga $\rightarrow$
caracol $\rightarrow$ erizo. Primer eslabón: siempre verde.
\end{example}

\begin{method}[Construir una cadena alimentaria]\label{met:g3:food-chains:build}
\begin{enumerate}
  \item elige un \omterm{def:g1:animals-around-us:animal}{animal} y pregunta: ¿qué come? Escribe el alimento a
        su izquierda, con la flecha apuntando al que come;
  \item hazle la misma pregunta al alimento y sigue hacia la izquierda
        hasta llegar a una \omterm{def:g1:plants-around-us:plant}{planta};
  \item pregunta: ¿quién se come a mi \omterm{def:g1:animals-around-us:animal}{animal} de partida? Alarga hacia
        la derecha mientras haya alguien que coma;
  \item comprueba que todas las flechas se leen \omterm{def:g3:food-chains:chain}{``es comido por''} ---
        con el alimento fluyendo hacia la derecha.
\end{enumerate}
\end{method}

\section{Cuando un eslabón cambia}

\begin{proposition}[Los eslabones dependen unos de otros]\label{prop:g3:food-chains:depend}
En una \omterm{def:g3:food-chains:chain}{cadena alimentaria}, cada eslabón depende de los que están
antes que él y alimenta a los que están después. Si un eslabón
escasea, los eslabones siguientes pasan hambre --- y el eslabón
\emph{anterior}, al que ya no se comen tanto, puede multiplicarse. Un
cambio en un eslabón recorre la cadena entera.
\end{proposition}
\admitted

\begin{example}[El verano seco]\label{ex:g3:food-chains:dry}
Una sequía agosta la hierba del prado. Los saltamontes, cortos de
comida, son pocos; las lagartijas que los cazaban pasan hambre y a su
vez alimentan mal al ratonero. Un primer eslabón amarillo y toda la
cadena de encima adelgaza --- el ratonero no comió hierba nunca y, aun
así, nota la sequía.
\end{example}

\begin{example}[Quitar al cazador]\label{ex:g3:food-chains:hunter}
Supón que los ratoneros desaparecen del valle. Las lagartijas, menos
cazadas, se multiplican; a los saltamontes se los come entonces con
más ganas y menguan; y la hierba, menos mordisqueada, hasta puede
espesarse. Quitar el \emph{último} eslabón también recorre la cadena
--- en la otra dirección.
\end{example}

\begin{remark}[Los prados de verdad son redes]\label{rem:g3:food-chains:web}
Nuestra cadena es un solo hilo, pero el ratonero también coge ratones,
la lagartija también come moscas y la hierba también alimenta a los
conejos. Muchas cadenas cruzadas hacen una red --- la \emph{red
alimentaria} que estudiará un curso posterior (el
\cref{ch:g2:what-animals-eat} ya la insinuó con el zorro, el conejo y
la hierba). Primero las cadenas: son los hilos con los que se teje la
red.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:1}
¿Qué significa la flecha de una \omterm{def:g3:food-chains:chain}{cadena alimentaria}? ¿Hacia dónde apunta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:2}
¿Qué clase de \omterm{def:g1:living-or-not:living}{ser vivo} es siempre el primer eslabón? ¿Por qué?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:3}
Escribe la cadena del prado de este capítulo con sus cuatro eslabones
y sus flechas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:4}
Corrige esta cadena: zorro $\rightarrow$ conejo $\rightarrow$ hierba.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:5}
Construye una cadena con estos eslabones: herrerillo, \omterm{def:g1:plants-around-us:parts}{hojas} de roble,
gavilán, \omterm{ex:g2:animals-grow-up:butterfly}{oruga}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:6}
En la cadena del jardín lechuga $\rightarrow$ caracol $\rightarrow$
erizo, ¿a quién se come el caracol? ¿Quién se come al caracol?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:7}
Usa el \cref{met:g3:food-chains:build} para hacer una cadena de al
menos tres eslabones que acabe en la garza de la charca.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:food-chains:8}
En el verano seco del \cref{ex:g3:food-chains:dry}, ¿por qué lo pasa
mal el ratonero si nunca come hierba?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:food-chains:9}
Los hortelanos que envenenan a todos los caracoles a veces se quedan
sin su erizo --- y después se encuentran \emph{más} babosas y
caracoles que antes. Explica las dos mitades con la
\cref{prop:g3:food-chains:depend}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:food-chains:10}
``La lagartija depende de la hierba''. ¿Verdadero o falso? Defiende tu
respuesta con la cadena del prado --- y di en qué dirección viaja un
fallo de la hierba.
\end{exercise}
