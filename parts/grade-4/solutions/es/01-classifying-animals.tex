\section*{Capítulo \ref{ch:g4:classifying-animals} --- Clasificar los animales}

\begin{solution}{exo:g4:classifying-animals:1}
Un \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} tiene un \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto} interno construido alrededor de una
\omterm{def:g4:classifying-animals:vertebrate}{columna vertebral} (la trucha, el gato); un \omterm{def:g4:classifying-animals:vertebrate}{invertebrado} no tiene
\omterm{def:g4:classifying-animals:vertebrate}{columna vertebral} (el caracol, la abeja, la lombriz, la medusa).
\end{solution}

\begin{solution}{exo:g4:classifying-animals:2}
\omterm{prop:g3:sorting-living-things:groups}{Peces}: escamas, aletas, respiran en el agua. \omterm{prop:g3:sorting-living-things:groups}{Anfibios}: \omterm{def:g1:the-five-senses:sense}{piel} desnuda y
húmeda, \omterm{def:g2:animals-grow-up:born}{huevos} en el agua, infancia de renacuajo. \omterm{prop:g4:classifying-animals:classes}{Reptiles}: \omterm{def:g1:the-five-senses:sense}{piel} seca
y escamosa, y casi todos ponen \omterm{def:g2:animals-grow-up:born}{huevos} con cáscara en tierra. \omterm{prop:g3:sorting-living-things:groups}{Aves}:
\omterm{def:g1:animals-around-us:covering}{plumas}, pico, \omterm{def:g2:animals-grow-up:born}{huevos} con cáscara. \omterm{prop:g3:sorting-living-things:groups}{Mamíferos}: \omterm{def:g1:animals-around-us:covering}{pelo}, y crías alimentadas
con \omterm{def:g2:animals-grow-up:born}{leche}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:3}
Sardina: \omterm{prop:g3:sorting-living-things:groups}{pez}. Cocodrilo: reptil. Tritón: anfibio. Avestruz: ave.
Ballena: mamífero. Lombriz: \omterm{def:g4:classifying-animals:vertebrate}{invertebrado} --- sin \omterm{def:g4:classifying-animals:vertebrate}{columna vertebral},
así que ninguna clase de \omterm{def:g4:classifying-animals:vertebrate}{vertebrados}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:4}
El gato, dentro de los \omterm{prop:g3:sorting-living-things:groups}{mamíferos}, dentro de los \omterm{def:g4:classifying-animals:vertebrate}{vertebrados}, dentro de
los \omterm{def:g1:animals-around-us:animal}{animales}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:5}
La \omterm{def:g1:the-five-senses:sense}{piel} --- seca y escamosa en la lagartija, húmeda y desnuda en el
tritón --- y los \omterm{def:g2:animals-grow-up:born}{huevos}: con cáscara y puestos en tierra en la
lagartija, desnudos y puestos en el agua en el tritón.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:6}
Los \omterm{def:g4:classifying-animals:vertebrate}{invertebrados}. Por ejemplo: \omterm{prop:g3:sorting-living-things:groups}{insectos}, arañas, caracoles, gusanos
(y medusas).
\end{solution}

\begin{solution}{exo:g4:classifying-animals:7}
Pingüino: ¿\omterm{def:g1:animals-around-us:covering}{plumas}? Sí --- un ave; se para en la primera pregunta.
Delfín: ¿\omterm{def:g1:animals-around-us:covering}{plumas}? No. ¿\omterm{def:g1:animals-around-us:covering}{Pelo} o \omterm{def:g2:animals-grow-up:born}{leche}? Alimenta a su cría con \omterm{def:g2:animals-grow-up:born}{leche} ---
un mamífero; se para ahí.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:8}
Por ejemplo: \omterm{def:g1:animals-around-us:covering}{plumas}, pico, alas, \omterm{def:g2:animals-grow-up:born}{huevos} con cáscara --- y una \omterm{def:g4:classifying-animals:vertebrate}{columna
vertebral}, porque las \omterm{prop:g3:sorting-living-things:groups}{aves} son \omterm{def:g4:classifying-animals:vertebrate}{vertebrados}.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:9}
Las preguntas se paran en ``\omterm{def:g2:animals-grow-up:born}{leche} para las crías'': la ballena
amamanta a su cría, así que es un mamífero, tenga o no tenga \omterm{def:g1:animals-around-us:covering}{pelo}.
``Vive en el mar'' es un \omterm{def:g2:where-animals-live:habitat}{hábitat}, no un rasgo del cuerpo --- y la
\omterm{def:g4:classifying-animals:classification}{clasificación} se construyó sobre lo que un \omterm{def:g1:animals-around-us:animal}{animal} \emph{tiene}
justamente porque las costumbres y las casas las pueden compartir unos
desconocidos: el tiburón y el delfín comparten dirección, no clase.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:10}
La \omterm{def:g4:classifying-animals:vertebrate}{columna vertebral}. La serpiente tiene una columna de cientos de
huesecillos --- un \omterm{def:g4:classifying-animals:vertebrate}{vertebrado}, con la \omterm{def:g1:the-five-senses:sense}{piel} seca y escamosa de un
reptil; la lombriz no tiene ninguna \omterm{def:g4:classifying-animals:vertebrate}{columna vertebral} --- un
\omterm{def:g4:classifying-animals:vertebrate}{invertebrado}. Que falten las patas es solo un rasgo que falta; la
pregunta de la \omterm{def:g4:classifying-animals:vertebrate}{columna vertebral} se hace antes.
\end{solution}

\begin{solution}{exo:g4:classifying-animals:11}
Ganaron los \omterm{prop:g3:sorting-living-things:groups}{mamíferos}: el \omterm{def:g1:animals-around-us:covering}{pelo} y la \omterm{def:g2:animals-grow-up:born}{leche} para las crías son el
conjunto de los \omterm{prop:g3:sorting-living-things:groups}{mamíferos}, y el ornitorrinco tiene los dos --- el
\omterm{def:g2:animals-grow-up:born}{huevo} es un rasgo a contracorriente. La \omterm{def:g4:classifying-animals:classification}{clasificación} pesa el conjunto
entero en vez de dejar que decida un rasgo suelto; un \omterm{def:g1:animals-around-us:animal}{animal} raro en
un aspecto se queda donde apuntan la mayoría de sus rasgos. (El
ornitorrinco es lo bastante raro como para que los \omterm{prop:g3:sorting-living-things:groups}{mamíferos} con
\omterm{def:g2:animals-grow-up:born}{huevos} formen su propio subgrupo pequeño.)
\end{solution}
