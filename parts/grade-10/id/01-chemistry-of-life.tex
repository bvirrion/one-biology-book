\chapter{Susunan Kimia Makhluk Hidup}\label{ch:g10:chemistry-of-life}

Biarkan sepotong roti terlalu lama di atas pemanggang dan roti itu
menghitam: yang tersisa sebagian besar karbon. Timbang sepotong apel
segar, keringkan sehari dalam oven hangat, lalu timbang lagi: lima
perenam massanya telah pergi sebagai uap air. Roti, apel, pohon ek,
ubur-ubur, kamu --- setiap makhluk hidup tersusun, pada dasarnya, dari
daftar bahan kimia pendek yang sama, dan daftar itu bukan daftar
penyusun batuan. Bab ini membongkar materi hidup, bahan demi bahan, dan
bertanya apa yang membuatnya khas.

\section{Unsur: atom yang sama dengan batuan, dalam perbandingan yang berbeda}

\begin{definition}[Komposisi kimia materi hidup]\label{def:g10:chemistry-of-life:composition}
Yang disebut \emph{komposisi kimia}\index{komposisi kimia} sebuah
cuplikan materi hidup adalah daftar unsur kimia yang dikandungnya,
beserta fraksi massa cuplikan yang diwakili masing-masing unsur.
Komposisi itu diukur dengan membongkar cuplikan secara kimia ---
membakarnya, menimbang apa yang menguap dan apa yang tersisa,
menganalisis gas yang dilepaskan --- bukan dengan mengamatinya.
\end{definition}

\begin{proposition}[Empat unsur menyusun 96\% tubuh makhluk hidup]\label{prop:g10:chemistry-of-life:four}
Menurut massa, materi hidup hampir seluruhnya tersusun dari empat
unsur: oksigen (O), karbon (C), hidrogen (H) dan nitrogen (N). Dalam
tubuh manusia keempatnya meliputi sekitar 65\%, 18.5\%, 9.5\% dan
3.2\% massa --- 96\% bersama-sama. Kalsium, fosfor, kalium, belerang,
natrium, klorin dan magnesium berbagi sebagian besar sisa 4\% itu, dan
selusin \emph{unsur renik} (besi, seng, iodin, tembaga\dots)
hadir jauh di bawah 0.1\% namun tak tergantikan. Keempat unsur yang sama
mendominasi dalam bakteri, tanaman gandum atau paus, dengan
perbandingan yang serupa.
\end{proposition}
\admitted

\begin{example}[Materi hidup dibandingkan batuan]\label{ex:g10:chemistry-of-life:crust}
Kerak Bumi juga 46\% oksigen, tetapi unsur berikutnya adalah silikon
(28\%), aluminium (8\%) dan besi (5.6\%). Karbon menyusun 0.02\% kerak
dan nitrogen 0.002\%, berhadapan dengan 18.5\% dan 3.2\% tubuh. Jadi
makhluk hidup bukanlah cuplikan lingkungannya: ia
\emph{memekatkan} beberapa unsur dengan faktor seribu atau lebih.
Penyaringan itulah tanda kimia pertama kehidupan.
\end{example}

\begin{remark}[Dari mana atom-atom itu datang]\label{rem:g10:chemistry-of-life:origin}
Setiap atom itu biasa saja: karbon dalam ototmu dahulu adalah karbon
dioksida udara sebelum sebuah tumbuhan mengikatnya, nitrogen dalam
proteinmu dahulu adalah nitrogen udara sebelum bakteri tanah
menangkapnya. Yang istimewa pada kehidupan bukanlah atomnya, melainkan
cara atom-atom itu dirakit dan perbandingan saat semuanya dikumpulkan.
\end{remark}

\section{Air dan garam mineral}

\begin{definition}[Materi mineral dan materi organik]\label{def:g10:chemistry-of-life:mineral-organic}
Molekul makhluk hidup terbagi menjadi dua kelompok. Kelompok pertama,
\emph{materi mineral}\index{materi mineral}, tersusun dari molekul yang
juga ada di luar makhluk hidup: air dan
\emph{garam mineral}\index{garam mineral} terlarut (ion seperti
$\mathrm{Na^+}$, $\mathrm{K^+}$, $\mathrm{Ca^{2+}}$, $\mathrm{Cl^-}$,
fosfat dan bikarbonat). Kelompok kedua,
\emph{materi organik}\index{materi organik}, tersusun dari molekul yang
dibangun di atas kerangka atom karbon yang terikat pada hidrogen, dan
biasanya pada oksigen serta nitrogen --- molekul yang, di Bumi, hanya
dibuat oleh makhluk hidup atau berasal dari sisa-sisanya.
\end{definition}

\begin{proposition}[Air adalah penyusun utama]\label{prop:g10:chemistry-of-life:water}
Air adalah molekul paling berlimpah pada setiap makhluk hidup: sekitar
60\% sampai 65\% massa manusia dewasa, 75\% pada bayi baru lahir, 85\%
pada apel, 95\% pada ubur-ubur, tetapi hanya 13\% pada butir gandum
yang dorman. Kadar air sebuah cuplikan diukur dengan menimbangnya,
mengeringkannya pada sekitar \qty{105}{\celsius} sampai massanya
berhenti turun, lalu menimbangnya lagi: massa yang hilang adalah airnya.
\end{proposition}
\admitted

\begin{example}[Mengeringkan sepotong apel]\label{ex:g10:chemistry-of-life:apple}
Sepotong apel segar bermassa \qty{24.0}{g} dikeringkan sampai massanya
mantap pada \qty{3.6}{g}. Air yang hilang: $24.0 - 3.6 = \qty{20.4}{g}$, yaitu
$20.4/24.0 = 0.85$, atau 85\% massa segarnya. Sisa \qty{3.6}{g} itu
adalah \emph{bahan kering}: molekul organik ditambah \omterm{def:g10:chemistry-of-life:mineral-organic}{garam mineral}.
Bakar bahan kering itu dan sekitar \qty{0.1}{g} abu pucat --- \omterm{def:g10:chemistry-of-life:mineral-organic}{garam
mineralnya} --- bertahan; sisanya, bagian organiknya, telah pergi
sebagai karbon dioksida dan uap air.
\end{example}

\begin{proposition}[Apa yang dikerjakan air]\label{prop:g10:chemistry-of-life:water-roles}
Air adalah tempat kimia sebuah sel berlangsung: air melarutkan garam,
gula dan sebagian besar molekul organik kecil, sehingga semuanya dapat
bertemu dan bereaksi; air mengangkutnya dari satu tempat ke tempat lain
(plasma darah 90\% air, getah jauh lebih banyak lagi); dan karena
memanaskannya menuntut banyak energi, air meredam suhu organisme
terhadap perubahan mendadak. Kehilangan 10\% air tubuh adalah keadaan
darurat medis; biji yang dorman, pada 13\%, sekadar mematikan kimianya
sampai air kembali.
\end{proposition}
\admitted

\begin{example}[Garam mineral yang dapat disebutkan namanya]\label{ex:g10:chemistry-of-life:salts}
Kalsium fosfat mengeraskan tulang dan gigi (kalsium 1.5\% massa tubuh,
hampir seluruhnya di rangka). Ion natrium dan klorida menetapkan kadar
garam darah, mendekati \qty{9}{g/L}; ion kalium terpekat di dalam sel;
besi duduk di jantung hemoglobin; iodin diperlukan untuk membuat hormon
kelenjar tiroid. \omterm{def:g10:chemistry-of-life:mineral-organic}{Garam mineral} tidak pernah menjadi sumber energi,
tetapi tubuh yang kekurangan salah satunya gagal pada satu tugas yang
tertentu.
\end{example}

\section{Molekul organik: empat keluarga}

\begin{definition}[Karbohidrat, lipid, protein, asam nukleat]\label{def:g10:chemistry-of-life:families}
Molekul organik tergolong ke dalam empat keluarga:
\begin{itemize}
  \item \emph{karbohidrat}\index{karbohidrat} (gula) hanya mengandung
        karbon, hidrogen dan oksigen. Gula sederhana berupa satuan
        tunggal, seperti glukosa $\mathrm{C_6H_{12}O_6}$; karbohidrat
        majemuk berupa rantai ratusan atau ribuan satuan glukosa ---
        pati dan selulosa pada tumbuhan, glikogen pada hewan.
  \item \emph{lipid}\index{lipid} (lemak dan minyak) juga hanya
        mengandung C, H dan O, tetapi dengan oksigen yang jauh lebih
        sedikit, dan tidak larut dalam air. Yang paling lazim adalah
        trigliserida, masing-masing terbuat dari satu molekul gliserol
        yang bergabung dengan tiga rantai asam lemak yang panjang.
  \item \emph{protein}\index{protein} berupa rantai
        \emph{asam amino}\index{asam amino}: dua puluh jenis asam
        amino, semuanya mengandung nitrogen, terangkai dalam urutan
        yang tepat menjadi rantai lima puluh sampai beberapa ribu
        satuan yang melipat menjadi bentuk yang tertentu.
  \item \emph{asam nukleat}\index{asam nukleat} (DNA dan RNA) berupa
        rantai nukleotida, yang mengandung fosfor di samping C, H,
        O dan N; keduanya menyimpan dan mewariskan keterangan genetik,
        dan menjadi pokok \cref{ch:g10:universal-dna}.
\end{itemize}
\end{definition}

\end{example}

\begin{proposition}[Untuk apa masing-masing keluarga]\label{prop:g10:chemistry-of-life:roles}
Keempat keluarga itu membagi pekerjaan sebuah sel, dengan tumpang tindih:
\begin{itemize}
  \item \omterm{def:g10:chemistry-of-life:families}{karbohidrat} adalah bahan bakar sehari-hari (glukosa) dan
        simpanan jangka pendeknya (pati, glikogen), dan selulosa adalah
        bahan bangunan dinding sel tumbuhan;
  \item \omterm{def:g10:chemistry-of-life:families}{lipid} adalah simpanan energi jangka panjang yang padat (\qty{38}{kJ} per
        gram, berhadapan dengan \qty{17}{kJ} per gram untuk \omterm{def:g10:chemistry-of-life:families}{karbohidrat} dan
        \omterm{def:g10:chemistry-of-life:families}{protein}), dan \omterm{def:g10:chemistry-of-life:families}{lipid} membran membentuk selubung setiap sel;
  \item \omterm{def:g10:chemistry-of-life:families}{protein} adalah mesinnya: enzim yang menjalankan setiap reaksi,
        serat yang mengerutkan otot, pengangkut, antibodi,
        reseptor;
  \item \omterm{def:g10:chemistry-of-life:families}{asam nukleat} menyimpan petunjuk untuk membangun \omterm{def:g10:chemistry-of-life:families}{protein}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Molekul besar dan wujud kehidupan]\label{rem:g10:chemistry-of-life:macromolecules}
Molekul batuan kecil dan berulang (kristal kuarsa adalah
$\mathrm{SiO_2}$ yang berulang tanpa akhir). Molekul khas
kehidupan sangat besar --- satu \omterm{def:g10:chemistry-of-life:families}{protein} dapat memuat sepuluh ribu atom, satu
molekul DNA bermiliar-miliar --- dan \emph{bermakna}: urutan satuan
pembangunnya membawa pesan. Tidak ada di dunia mineral yang menyerupai
molekul yang urutannya berarti sesuatu.
\end{remark}

\section{Mengenali penyusunnya}

\begin{method}[Uji baku untuk keempat keluarga]\label{met:g10:chemistry-of-life:tests}
Setiap keluarga punya satu uji sederhana yang memberi hasil kasatmata.
\begin{enumerate}
  \item \emph{Air}: sehelai kertas kobalt klorida biru berubah merah
        muda; atau keringkan cuplikannya dan ukur massa yang hilang.
  \item \emph{Pati}: setetes larutan iodin berubah dari cokelat
        kekuningan menjadi biru kehitaman.
  \item \emph{Gula sederhana} (glukosa, fruktosa\dots): panaskan
        dengan pereaksi Fehling atau Benedict; larutan birunya memberi
        endapan merah bata.
  \item \emph{\omterm{def:g10:chemistry-of-life:families}{Protein}}: uji biuret --- beberapa tetes tembaga
        sulfat dalam larutan basa berubah ungu.
  \item \emph{\omterm{def:g10:chemistry-of-life:families}{Lipid}}: cuplikannya meninggalkan noda tembus cahaya yang
        tidak mengering pada kertas; zat warna merah Sudan III mewarnainya.
  \item \emph{\omterm{def:g10:chemistry-of-life:mineral-organic}{Garam mineral}}: bakar bahan keringnya sampai habis; abu
        yang bertahan bersifat mineral. Ion tertentu dikenali lewat
        pengendapan (perak nitrat untuk klorida, misalnya).
\end{enumerate}
Selalu jalankan satu kontrol dengan air murni, dan satu lagi dengan
cuplikan yang sudah pasti positif, agar yakin pereaksinya bekerja dan
perubahan warnanya berarti seperti yang kamu kira.
\end{method}

\begin{example}[Menganalisis sebuah kentang]\label{ex:g10:chemistry-of-life:potato}
Sepotong kentang mentah: kertas kobalt klorida berubah merah muda
(air); iodin langsung berubah biru kehitaman (pati, berlimpah); Fehling
hanya memberi rona jingga yang samar (sedikit gula bebas); biuret
memberi ungu pucat (ada \omterm{def:g10:chemistry-of-life:families}{protein}, sekitar 2\% massa segarnya); tidak ada
noda tembus cahaya (hampir tidak ada \omterm{def:g10:chemistry-of-life:families}{lipid}). Pengeringan menyisakan 22\% massanya;
pembakaran sisa itu meninggalkan 1\% abu. Kentang adalah simpanan pati yang
terbungkus air --- tepat seperti yang diperlukan tumbuhan untuk
bertahan melewati musim dingin di bawah tanah.
\end{example}

\begin{proposition}[Kesatuan komposisi]\label{prop:g10:chemistry-of-life:unity}
Apa pun organismenya --- bakteri, jamur, tumbuhan, hewan --- uji yang
sama menemukan keempat keluarga yang sama, terbuat dari satuan
pembangun yang sama: dua puluh \omterm{def:g10:chemistry-of-life:families}{asam amino} yang sama, glukosa yang sama,
empat nukleotida yang sama. Makhluk hidup menjadi khas bukan karena memiliki
atom istimewa atau satuan pembangun istimewa, melainkan karena
perbandingan yang dipertahankannya, molekul yang sangat besar yang
dirakitnya, dan keterangan yang dibawa molekul-molekul itu. Kesatuan
kimia ini adalah salah satu alasan terkuat bahwa seluruh makhluk hidup
berbagi satu asal usul, alasan yang akan diangkat \cref{ch:g10:common-ancestry}.
\end{proposition}
\admitted

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:1}
Sebutkan empat unsur yang menyusun 96\% massa tubuh makhluk hidup,
beserta persentasenya kira-kira pada manusia.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:2}
Pilah menjadi \omterm{def:g10:chemistry-of-life:mineral-organic}{materi mineral} dan \omterm{def:g10:chemistry-of-life:mineral-organic}{materi organik}: air, glukosa, natrium
klorida, trigliserida, kalsium fosfat, \omterm{def:g10:chemistry-of-life:families}{asam amino}, karbon
dioksida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:3}
Sehelai daun selada segar bermassa \qty{12.0}{g} dikeringkan sampai
massanya tetap: \qty{0.6}{g}. Berapa kadar airnya, dalam persen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:4}
Uji mana yang kamu pakai untuk menunjukkan bahwa sebutir biji kacang
mengandung (a) pati, (b) \omterm{def:g10:chemistry-of-life:families}{protein}, (c) \omterm{def:g10:chemistry-of-life:families}{lipid}? Sebutkan hasil yang diharapkan pada tiap hal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:5}
Sebutkan satuan pembangun tiap keluarga: \omterm{def:g10:chemistry-of-life:families}{karbohidrat}, \omterm{def:g10:chemistry-of-life:families}{lipid},
\omterm{def:g10:chemistry-of-life:families}{protein}, \omterm{def:g10:chemistry-of-life:families}{asam nukleat}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:chemistry-of-life:6}
Dengan gambar pertama bab ini, berapa kali lipat karbon lebih pekat
dalam tubuh manusia daripada dalam kerak Bumi? Dan nitrogen?
Beri tanggapan.
\end{exercise}

\section{Soal: Karbon dalam Diri Seorang Manusia}

\begin{problem}[{Soal akhir pekan --- dari ladang gandum ke sebuah tubuh:
audit materi, dari air dalam sebutir gandum sampai jumlah kilogram
karbon yang dibawa seseorang}]
\label{pb:g10:chemistry-of-life:1}
Sebuah kelas mengikuti materi dari ladang gandum sampai ke tubuh salah
seorang muridnya. Data: sebutir gandum mengandung 13\% air, dan bahan
keringnya kira-kira 70\% pati, 12\% \omterm{def:g10:chemistry-of-life:families}{protein}, 2\% \omterm{def:g10:chemistry-of-life:families}{lipid}, 2\% mineral, sisanya
selulosa. Tepung roti kurang lebih mempertahankan komposisi bagian
dalam butirnya. Murid itu, Lea, bermassa \qty{60}{kg}; pakai
komposisi orang dewasa pada gambar kedua (air 60\%, \omterm{def:g10:chemistry-of-life:families}{protein} 17\%,
\omterm{def:g10:chemistry-of-life:families}{lipid} 16\%, mineral 6\%, \omterm{def:g10:chemistry-of-life:families}{karbohidrat} 1\%). Fraksi massa unsur
di dalam tiap keluarga: ambil \omterm{def:g10:chemistry-of-life:families}{karbohidrat} 40\% karbon, \omterm{def:g10:chemistry-of-life:families}{lipid} 77\%
karbon, \omterm{def:g10:chemistry-of-life:families}{protein} 53\% karbon dan 16\% nitrogen.

\medskip
\noindent\textbf{Bagian I --- Butir gandumnya.}
\begin{enumerate}
  \item Sebutir gandum bermassa \qty{45}{mg} dikeringkan. Berapa massa
        air yang hilang, dan berapa massa keringnya?
  \item Hitung massa pati, \omterm{def:g10:chemistry-of-life:families}{protein} dan \omterm{def:g10:chemistry-of-life:families}{lipid} dalam bahan
        kering satu butir.
  \item Uji mana pada \cref{met:g10:chemistry-of-life:tests} yang akan
        memastikan masing-masing ketiga penyusun itu pada butir yang ditumbuk?
  \item Butir itu ditanam dan disiram; tiga hari kemudian kecambahnya
        mengandung 88\% air. Jelaskan mengapa perkecambahan tidak dapat mulai pada 13\%.
  \item Satu hektar menghasilkan \qty{8}{t} gandum. Berapa massa air
        yang ikut terpanen, dan berapa massa bahan keringnya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Rotinya.}
\begin{enumerate}[resume]
  \item Sebuah roti dipanggang dari \qty{500}{g} tepung, \qty{320}{g}
        air dan \qty{10}{g} garam; roti itu kehilangan \qty{130}{g} air
        dalam oven. Berapa massa roti matangnya, dan berapa kadar
        airnya dalam persen?
  \item Dengan komposisi kering butir gandum sebagai komposisi tepung,
        perkirakan massa pati dan \omterm{def:g10:chemistry-of-life:families}{protein} dalam roti itu.
  \item Dengan \qty{17}{kJ/g} untuk \omterm{def:g10:chemistry-of-life:families}{karbohidrat} dan \omterm{def:g10:chemistry-of-life:families}{protein} serta
        \qty{38}{kJ/g} untuk \omterm{def:g10:chemistry-of-life:families}{lipid}, perkirakan kandungan energi roti
        itu.
  \item Lea memakan \qty{120}{g} roti ini. Berapa bagian dari kebutuhan
        harian sekitar \qty{10000}{kJ} yang tertutupi olehnya?
  \item Mengapa garam dalam roti itu bukan sumber energi, dan untuk apa
        garam itu di dalam tubuh?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Tubuhnya: air dan keluarga molekul.}
\begin{enumerate}[resume]
  \item Hitung massa air dalam tubuh Lea.
  \item Hitung massa \omterm{def:g10:chemistry-of-life:families}{protein}, \omterm{def:g10:chemistry-of-life:families}{lipid}, mineral dan \omterm{def:g10:chemistry-of-life:families}{karbohidrat}.
  \item Lea kehilangan \qty{1.2}{kg} lewat keringat selama lari
        panjang. Berapa bagian air tubuhnya itu? Mengapa harus segera
        diganti?
  \item Darahnya, sekitar \qty{4.5}{L}, mengandung 55\% plasma, dan
        plasma mengandung 90\% air. Perkirakan massa air dalam darahnya
        (ambil \qty{1}{kg} per liter). Berapa bagian air tubuhnya itu?
        Di mana sisanya?
  \item Bayi baru lahir mengandung 75\% air. Jelaskan dalam satu
        kalimat mengapa bayi lebih rentan terhadap kekurangan cairan
        daripada orang dewasa.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Tubuhnya: unsur-unsurnya.}
\begin{enumerate}[resume]
  \item Dengan fraksi karbon tiap keluarga, hitung massa karbon dalam
        \omterm{def:g10:chemistry-of-life:families}{karbohidrat}, \omterm{def:g10:chemistry-of-life:families}{lipid} dan \omterm{def:g10:chemistry-of-life:families}{protein} Lea, lalu massa karbon
        seluruhnya.
  \item Berapa bagian massa tubuhnya berupa karbon? Bandingkan dengan
        18.5\% pada \cref{prop:g10:chemistry-of-life:four} dan beri
        tanggapan atas kesesuaiannya.
  \item Hitung massa nitrogen dalam \omterm{def:g10:chemistry-of-life:families}{proteinnya}. Dari mana nitrogen itu
        akhirnya berasal?
  \item Air mengandung 89\% oksigen menurut massa. Hitung massa oksigen
        yang dibawa air tubuhnya saja, dan bandingkan dengan 65\% pada
        \cref{prop:g10:chemistry-of-life:four}. Molekul mana yang
        menjelaskan sebagian besar oksigen tubuh?
  \item Nyatakan hasilnya dalam satu kalimat: berapa kilogram karbon
        yang dibawa seorang bermassa \qty{60}{kg}, dan lewat dua proses
        biologi mana karbon itu meninggalkan udara dan masuk ke dalam
        tubuh?
\end{enumerate}
\end{problem}
