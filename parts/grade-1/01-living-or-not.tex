\chapter{Living or Not Living?}\label{ch:g1:living-or-not}

A snail slides slowly across the wet path. Next to it lies a gray
pebble, just as small, just as round. The snail and the pebble sit in
the same rain --- yet one of them is alive and the other is not. How
do we know? That is the very first question of \omterm{def:g1:living-or-not:living}{biology}, the science
of \omterm{def:g1:living-or-not:living}{living things}.

\section{What living things do}

\begin{definition}[Living thing]\label{def:g1:living-or-not:living}
A \emph{living thing}\index{living thing} is something that is
\emph{alive}\index{alive}: it is born, it feeds, it grows, it can
have babies, and one day it dies. Animals and plants are living
things. Biology\index{biology} is the science that studies them.
\end{definition}

\begin{example}[The snail and the pebble]\label{ex:g1:living-or-not:snail}
The snail was born from a tiny egg. It eats salad leaves. It grows
--- its shell grows with it. There can be baby snails. So the snail
is a \omterm{def:g1:living-or-not:living}{living thing}. The pebble eats nothing, grows not at all, and
never has baby pebbles: the pebble is not living.
\end{example}

\begin{proposition}[The signs of life]\label{prop:g1:living-or-not:signs}
A \omterm{def:g1:living-or-not:living}{living thing} shows these signs:
\begin{enumerate}
  \item it is \emph{born}\index{born};
  \item it \emph{feeds}\index{feeds} --- it takes in food or makes
        its own, like plants do with light;
  \item it \emph{grows}\index{grows};
  \item it can have babies --- new \omterm{def:g1:living-or-not:living}{living things} like itself;
  \item it \emph{dies}\index{dies} one day.
\end{enumerate}
Something that shows none of these signs is not living.
\end{proposition}
\admitted

\begin{remark}[Every sign counts]\label{rem:g1:living-or-not:allsigns}
One sign alone is not enough. A car ``drinks'' fuel, and an icicle
grows longer in the cold --- but the car was built, not born, and the
icicle has no baby icicles. To be alive, a thing must show the signs
of life \emph{together}: born, feeding, growing, making new \omterm{def:g1:living-or-not:living}{living
things}, dying.
\end{remark}

\section{Living, not living, never living}

\begin{definition}[Non-living thing]\label{def:g1:living-or-not:nonliving}
A \emph{non-living thing}\index{non-living thing} is something that
is not alive. There are two kinds:
\begin{itemize}
  \item things that were \emph{never alive}\index{never alive}: a
        pebble, water, the wind, a bicycle;
  \item things that are \emph{no longer alive}\index{no longer
        alive} or that \emph{come from} \omterm{def:g1:living-or-not:living}{living things}: a dead leaf on
        the ground, a wooden table, a woolly pullover.
\end{itemize}
\end{definition}

\begin{example}[The wooden table]\label{ex:g1:living-or-not:table}
A wooden table does not feed and does not grow: it is not living.
But its wood once was: it comes from a tree that was born from a
seed, grew for many years, and was cut down. The table is non-living
--- and it comes from a \omterm{def:g1:living-or-not:living}{living thing}.
\end{example}

\begin{example}[What about a robot?]\label{ex:g1:living-or-not:robot}
A toy robot moves, makes sounds, and its light blinks like a little
eye. Is it alive? Check the signs: it was built in a factory, not
born; its battery is not food that makes it grow; it will never have
baby robots. The robot moves, but it is not living.
\end{example}

\begin{method}[How to decide: living or not?]\label{met:g1:living-or-not:decide}
Ask the questions of life, one by one:
\begin{enumerate}
  \item Was it born? Does it feed? Does it grow?
  \item Can there be babies of it? Will it die one day?
  \item If the answers are yes: it is living.
  \item If the answers are no, ask one more question: does it
        \emph{come from} something living, like wood from a tree?
\end{enumerate}
\end{method}

\begin{remark}[Seeds look asleep]\label{rem:g1:living-or-not:seed}
A dry bean seed in a packet seems as still as a pebble. But put it
on wet cotton, and in a few days a little root appears: the seed was
alive all along, waiting. Slow, quiet \omterm{def:g1:living-or-not:living}{living things} are easy to
mistake for non-living ones --- the questions of life help us look
closer.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:1}
Say the five signs of life. Which of them can you see a kitten doing
today?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:2}
Living or not living? (a) a butterfly; (b) a bicycle; (c) a daisy;
(d) a cloud.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:3}
A dead leaf lies on the path. Is it living now? Was it ever living?
Where did it come from?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:4}
Tom says: ``The river is alive --- look, it moves and it makes
noise!'' Use the questions of \cref{met:g1:living-or-not:decide} to
answer him kindly.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:5}
Sort into the three boxes of the figure --- living, was living,
never living: a spider, a woolly sock, a raindrop, an apple on the
tree, a fallen pine cone.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:6}
A candle flame ``eats'' the wax, grows taller, and dies when you
blow it out. Is the flame a \omterm{def:g1:living-or-not:living}{living thing}? Which question of life
says no?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:7}
Name two things in your bedroom that come from \omterm{def:g1:living-or-not:living}{living things}, like
the wooden table of \cref{ex:g1:living-or-not:table}. Say which
\omterm{def:g1:living-or-not:living}{living thing} each one comes from.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:8}
Try it: put a dry bean on wet cotton near a window, and a pebble on
another piece of wet cotton next to it. Look every day for a week.
Which one changes? What does that show?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:9}
A toy robot and a snail both move across the floor. Give two
questions of life where their answers are different.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:10}
Mushrooms do not run like animals and are not green like most
plants. With the signs of \cref{prop:g1:living-or-not:signs},
explain why a mushroom is still a \omterm{def:g1:living-or-not:living}{living thing}.
\end{exercise}
