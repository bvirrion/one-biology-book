\chapter{المخططات التنظيمية والأنسجة عند الحيوان}\label{ch:b1:body-plans-tissues}

أربعة حيوانات على منضدة المختبر: هيدرا طولها سنتيمتر، ودودة أرض،
وجراد بحر، وضفدع. وتحت اختلافات القياس والسيرة تُبنى كلها على حفنة
من القرارات المشتركة --- أللجسم مقدمة ومؤخرة، ومن كم طبقة من الخلايا
نما، أثمة جوف حول المعي، أمبني الجسم من قطع متكررة، وأين الهيكل ---
وكل واحد منها مبني من أنواع \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة} الأربعة نفسها. يعرض هذا الفصل
المخططات التنظيمية عند الحيوان \omterm{def:b1:body-plans-tissues:tissue}{والأنسجة} التي تبنيها، وينتهي حيث
انتهت الفصول السابقة: عند جدار معي ثديي، منظورًا إليه بوصفه \omterm{def:b1:body-plans-tissues:tissue}{أنسجة}
مركبة في \omterm{def:b1:mammal-organization:organ}{عضو}.

\section{المخططات التنظيمية}

\begin{definition}[التناظر والتمركز الرأسي]\label{def:b1:body-plans-tissues:symmetry}
يكون للحيوان \emph{تناظر شعاعي}\index{تناظر شعاعي} حين ينتظم جسمه
حول محور مركزي بلا يمين ولا يسار، مثل الهيدرا أو قنديل البحر: فهو
يلاقي محيطه من كل الجهات على السواء، ويكون عادة ثابتًا أو منجرفًا.
ويكون له \emph{تناظر جانبي}\index{تناظر جانبي} حين يقسمه مستو واحد
إلى نصفين متماثلين، فيحدد مقدمة (أمامية) ومؤخرة (خلفية)، وظهرًا
وبطنًا: وهو مخطط حيوان يتحرك ورأسه إلى الأمام، \omterm{def:b1:mammal-organization:organ}{وأعضاء} حسه ومراكزه
العصبية مركزة في المقدمة (\omterm{def:b1:mammal-organization:bodyplan}{التمركز الرأسي}).
\end{definition}

\begin{definition}[الأوراق الجنينية وأجواف الجسم]\label{def:b1:body-plans-tissues:layers}
يكوّن جنين الحيوان طبقتين أو ثلاثًا من الخلايا، هي \emph{الأوراق
الجنينية}\index{ورقة جنينية}، تشتق منها كل \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}: \emph{الأديم
الظاهر}\index{أديم ظاهر} الخارجي (الجلد، \omterm{def:b1:mammal-organization:organ}{والجهاز} العصبي)،
و\emph{الأديم الباطن}\index{أديم باطن} الداخلي (بطانة المعي وما
ينمو عنه)، و\emph{الأديم المتوسط}\index{أديم متوسط} الأوسط في كل
الحيوانات عدا الإسفنجيات واللاسعات (العضل، والهيكل، \omterm{def:b1:body-plans-tissues:connective}{والنسيج الضام}،
والدم، والكلى، والمناسل). فاللاسعات \emph{ثنائية الورقتين}، وسائرها
\emph{ثلاثية \omterm{def:b1:flowering-plant-organization:organs}{الأوراق}}. وفي الحيوانات ثلاثية \omterm{def:b1:flowering-plant-organization:organs}{الأوراق} قد يملأ الأديم
المتوسط الحيز بين المعي وجدار الجسم (\emph{عديمة الجوف}: الديدان
المفلطحة)، أو يترك بينهما جوفًا: وهو \emph{جوف عام}\index{جوف عام}
حين يبطنه الأديم المتوسط كله (الحلقيات، والرخويات، والمفصليات،
وشوكيات الجلد، والفقاريات). \omterm{def:b1:mammal-organization:bodyplan}{والجوف العام} يعلّق المعي في حجيرته
الخاصة، ويتيح له أن يتحرك مستقلًا عن جدار الجسم، ويمنح رخوات الجسم
هيكلًا مائيًا.
\end{definition}

\begin{definition}[أوليات الفم وثانويات الفم]\label{def:b1:body-plans-tissues:proto}
في الحيوانات ذوات \omterm{def:b1:mammal-organization:bodyplan}{الجوف العام}، يقسم مصير أول فتحة للمعي الجنيني
نسبين كبيرين: ففي \emph{أوليات الفم}\index{أولي الفم} (الحلقيات،
والرخويات، والمفصليات) تصير فمًا، ويتكون \omterm{def:b1:mammal-organization:bodyplan}{الجوف العام} بانشقاق \omterm{def:b1:body-plans-tissues:layers}{الأديم
المتوسط}، وتكون الانقسامات الأولى للبيضة حلزونية ومصيرها محدد باكرًا؛
وفي \emph{ثانويات الفم}\index{ثانوي الفم} (شوكيات الجلد،
والفقاريات) تصير شرجًا، ويتكون \omterm{def:b1:mammal-organization:bodyplan}{الجوف العام} بانجياب المعي إلى الخارج،
وتبقى الخلايا الأولى مرنة المصير.
\end{definition}

\begin{definition}[التقطع والهياكل]\label{def:b1:body-plans-tissues:segments}
يكون الجسم \emph{متقطعًا}\index{تقطع} حين يُبنى من وحدات متكررة على
امتداد محوره (حلقات دودة الأرض، وقطع جراد البحر، وفقرات السمكة وكتلها
العضلية)، لكل منها مجموعتها من العضلات والأعصاب وكثيرًا من الزوائد.
وقد تتشابه القطع أو تتخصص في مناطق (رأس، وصدر، وبطن).
و\emph{الهيكل}\index{هيكل}، الذي تشد عليه العضلات، إما
\emph{مائي}\index{هيكل مائي} (سائل تحت ضغط في \omterm{def:b1:body-plans-tissues:layers}{جوف عام}، كما في دودة
الأرض)، أو \emph{هيكل خارجي}\index{هيكل خارجي} (\omterm{def:b1:flowering-plant-organization:tissues}{أديمة} صلبة خارج
الجسم، مفصلية، تُنسلخ ليكبر الحيوان، كما في المفصليات)، أو
\emph{هيكل داخلي}\index{هيكل داخلي} (غضروف أو عظم داخل الجسم، ينمو
معه، كما في الفقاريات).
\end{definition}

\begin{example}[قراءة حيوان من مخططه]\label{ex:b1:body-plans-tissues:reading}
حيوان له نهاية رأسية، ومعي نافذ، وجسم من حلقات متشابهة، ولا أجزاء
صلبة له: فهو أولي فم ذو \omterm{def:b1:body-plans-tissues:layers}{جوف عام} يتحرك \omterm{def:b1:body-plans-tissues:segments}{بهيكل مائي}، إذ يُضغط سائل جوف
كل قطعة بعضلات دائرية ثم طولانية بالتناوب --- أي دودة أرض تحفر.
وحيوان له \omterm{def:b1:flowering-plant-organization:tissues}{أديمة} مفصلية وقطع متخصصة: فهو مفصلي، لا بد لنموه أن يمر
بانسلاخات. وكتل عضلية جانبية متقطعة حول محور ظهري من الفقرات: فهو
فقاري، هيكله الداخلي ينمو باستمرار.
\end{example}

\section{أنواع الأنسجة الأربعة}

\begin{definition}[النسيج]\label{def:b1:body-plans-tissues:tissue}
\emph{النسيج}\index{نسيج} مجموعة خلايا متشابهة البنية والأصل، مع
\omterm{def:b1:body-plans-tissues:connective}{المادة خارج الخلوية} التي تنتجها، منظمة لأداء وظيفة مشتركة. وأنسجة
الحيوان أربعة أنواع: ظهارية، وضامة، وعضلية، وعصبية. وكل \omterm{def:b1:mammal-organization:organ}{عضو} تركيب من
اثنين منها على الأقل.
\end{definition}

\begin{definition}[النسيج الظهاري]\label{def:b1:body-plans-tissues:epithelium}
\emph{الظهارة}\index{ظهارة} صفيحة من خلايا متراصة تغطي سطحًا أو تبطن
تجويفًا، مستقرة على طبقة رقيقة من \omterm{def:b1:body-plans-tissues:connective}{المادة خارج الخلوية} هي
\emph{الصفيحة القاعدية}\index{الصفيحة القاعدية}. وخلاياها
\emph{مستقطبة}\index{القطبية الخلوية}: فالوجه القمي يطل على الخارج
أو على اللمعة، والوجه القاعدي على الصفيحة، ويحمل كل منهما بروتينات
مختلفة. وتربطها \emph{وصلات خلوية}\index{وصلة خلوية}:
\emph{وصلات محكمة} قرب القمة، تسد الحيز بين الخلايا حتى لا يعبر
الصفيحةَ شيء إلا عبر الخلايا؛ و\emph{وصلات التصاق}
و\emph{جسيمات رابطة} تثبت الخلايا تثبيتًا ميكانيكيًا؛
و\emph{وصلات فجوية}، وهي قنوات تمرر الشوارد والجزيئات الصغيرة من
خلية إلى خلية. وتُصنف الظهارات بعدد الطبقات (بسيطة، أو مطبقة) وبشكل
الخلايا (حرشفية، أو مكعبة، أو عمادية)، وتشمل الغدد، وهي خلايا ظهارية
متخصصة في الإفراز. وهي تغطي وتحمي وتمتص وتفرز.
\end{definition}

\begin{definition}[النسيج الضام]\label{def:b1:body-plans-tissues:connective}
\emph{النسيج الضام}\index{نسيج ضام} مكون من خلايا متفرقة في
\emph{مادة خارج خلوية}\index{المادة خارج الخلوية} وفيرة تفرزها هي:
ألياف بروتينية --- \emph{الكولاجين}\index{كولاجين} (مقاومة الشد؛ وهو
ربع كل بروتين الثديي)، و\emph{الإيلاستين}\index{إيلاستين} (الارتداد)
--- مغروسة في \emph{مادة أساسية} مماهة من السكريات المتعددة.
والنسب تحدد أنواعه: \omterm{def:b1:body-plans-tissues:tissue}{نسيج} ضام رخو تحت الظهارات وحول \omterm{def:b1:mammal-organization:organ}{الأعضاء} (أرومات
ليفية، وألياف قليلة، ومادة أساسية كثيرة)؛ \omterm{def:b1:body-plans-tissues:tissue}{ونسيج} ضام كثيف في الأوتار
والأربطة (كولاجين متواز)؛ \omterm{def:b1:body-plans-tissues:tissue}{ونسيج} شحمي (خلايا مدخرة للشحم)؛ وغضروف
(هلام متين من الكولاجين والسكريات المتعددة، بلا أوعية)؛ وعظم
(كولاجين متمعدن بفوسفات الكالسيوم، وخلايا حية في تجاويف، وأوعية)؛
ودم مادته البلازما. وهو يربط ويدعم ويدخر وينقل ويدافع.
\end{definition}

\begin{definition}[النسيج العضلي]\label{def:b1:body-plans-tissues:muscle}
\emph{النسيج العضلي}\index{نسيج عضلي} مكون من خلايا مستطيلة مكتظة
بخيوط بروتينية متقلصة (الأكتين والميوزين). وهو ثلاثة أنواع:
\emph{العضلة الهيكلية}، من ألياف طويلة عديدة النوى تبدو عليها تخططات
عرضية، مرتبطة بالهيكل وخاضعة للإرادة؛ و\emph{العضلة القلبية}، من
خلايا مخططة متفرعة موصولة طرفًا بطرف بوصلات تمرر النبضة من خلية إلى
خلية، وهي تتقلص على إيقاعها الخاص؛ و\emph{العضلة الملساء}، من خلايا
مغزلية غير مخططة في جدر \omterm{def:b1:mammal-organization:organ}{الأعضاء} الجوفاء (المعي، والأوعية، والرحم)،
تتقلص ببطء ودون إرادة. أما آلية التقلص فيعالجها كتاب السنة الثانية.
\end{definition}

\begin{definition}[النسيج العصبي]\label{def:b1:body-plans-tissues:nervous}
\emph{النسيج العصبي}\index{نسيج عصبي} مكون من
\emph{عصبونات}\index{عصبون}، وهي خلايا لها جسم وامتدادات طويلة
(تغصنات تستقبل الإشارات، ومحور يحملها بعيدًا) تنقل الإشارات
الكهربائية وتمررها إلى خلايا أخرى عند المشابك، ومن \emph{خلايا
دبقية}\index{خلية دبقية} تفوقها عددًا بمرات، وهي تدعم العصبونات
وتعزلها وتغذيها. وهو يحس ويدمج ويأمر.
\end{definition}

\begin{example}[من أين تأتي الأنسجة]\label{ex:b1:body-plans-tissues:origin}
تشتق الظهارات من \omterm{def:b1:body-plans-tissues:layers}{الأوراق الجنينية} الثلاث جميعًا (الجلد من \omterm{def:b1:body-plans-tissues:layers}{الأديم
الظاهر}، وبطانة المعي من \omterm{def:b1:body-plans-tissues:layers}{الأديم الباطن}، وبطانة الأوعية \omterm{def:b1:mammal-organization:bodyplan}{والجوف العام}
من \omterm{def:b1:body-plans-tissues:layers}{الأديم المتوسط})؛ ويشتق \omterm{def:b1:body-plans-tissues:connective}{النسيج الضام} والعضل والدم من \omterm{def:b1:body-plans-tissues:layers}{الأديم
المتوسط}؛ \omterm{def:b1:body-plans-tissues:nervous}{والنسيج العصبي} من \omterm{def:b1:body-plans-tissues:layers}{الأديم الظاهر}. فجلد الضفدع \omterm{def:b1:body-plans-tissues:epithelium}{ظهارة} أديمية
ظاهرة على \omterm{def:b1:body-plans-tissues:connective}{نسيج ضام} أديمي متوسط؛ ومعيه \omterm{def:b1:body-plans-tissues:epithelium}{ظهارة} أديمية باطنة ملفوفة في
عضل أديمي متوسط؛ ودماغه \omterm{def:b1:body-plans-tissues:layers}{أديم ظاهر} انطوى إلى الداخل.
\end{example}

\section{من الأنسجة إلى عضو: جدار المعي}

\begin{proposition}[طبقات جدار المعي الأربع]\label{prop:b1:body-plans-tissues:gutwall}
\index{جدار المعي}
من اللمعة إلى الخارج، يُبنى جدار الأنبوب الهضمي عند الفقاريات من
أربع طبقات متحدة المركز، كل منها تركيب من \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}:
\emph{المخاطية} (\omterm{def:b1:body-plans-tissues:epithelium}{ظهارة} --- ماصة ومفرزة --- على \omterm{def:b1:body-plans-tissues:connective}{نسيج ضام} رخو يحمل
أوعية دموية ولمفية، وتحتها صفيحة رقيقة من العضل الأملس)؛
و\emph{تحت المخاطية} (\omterm{def:b1:body-plans-tissues:connective}{نسيج ضام} كثيف بأوعية أكبر وشبكة عصبية)؛
و\emph{العضلية} (طبقتان من العضل الأملس، داخلية دائرية وخارجية
طولانية، بينهما شبكة عصبية، تنتجان حركات المزج والدفع)؛
و\emph{المصلية} (طبقة ضامة رقيقة تغطيها \omterm{def:b1:body-plans-tissues:epithelium}{ظهارة} \omterm{def:b1:mammal-organization:bodyplan}{الجوف العام}). والطبقات
الأربع نفسها تمتد من المريء إلى المستقيم؛ وما يتغير هو المخاطية ---
مطوية إلى زغابات في المعي الدقيق، غدية في المعدة.
\end{proposition}

\begin{method}[قراءة مقطع نسيجي]\label{met:b1:body-plans-tissues:histology}
\begin{enumerate}
  \item ابحث عن السطح الحر أو اللمعة: \omterm{def:b1:body-plans-tissues:epithelium}{فالظهارة} التي تبطنه هي الطبقة
        الأولى. وعُدّ صفوف نواها (بسيطة أو مطبقة) ولاحظ شكل الخلايا.
  \item وتحت \omterm{def:b1:body-plans-tissues:epithelium}{الظهارة} ابحث عن المنطقة الشاحبة الليفية ذات النوى
        المتفرقة والأوعية: فهي \omterm{def:b1:body-plans-tissues:connective}{النسيج الضام}.
  \item ويظهر العضل حزمًا كثيفة من خلايا مستطيلة: مخططة ونواها
        محيطية (هيكلية)، أو مخططة متفرعة (قلبية)، أو غير مخططة
        ونواتها مركزية (ملساء). ولاحظ اتجاه الألياف بالنسبة إلى
        المقطع.
  \item ويظهر \omterm{def:b1:body-plans-tissues:nervous}{النسيج العصبي} في جدار \omterm{def:b1:mammal-organization:organ}{عضو} تجمعات من أجسام عصبونية
        كبيرة شاحبة بين طبقتي العضل.
  \item وسمِّ \omterm{def:b1:mammal-organization:organ}{العضو} من تتابع الطبقات ومن تخصصات المخاطية (زغابات،
        وغدد، وكيراتين).
\end{enumerate}
\end{method}

\begin{example}[الزغابة سطح تبادل]\label{ex:b1:body-plans-tissues:villus}
زغابة المعي الدقيق عند الإنسان إصبع ارتفاعه \qty{0.6}{mm} وعرضه
\qty{0.15}{mm}، تغطيه طبقة واحدة من خلايا عمادية يحمل الوجه القمي
لكل منها نحو \qty{1500}{} زغيبة. والزغابات تضاعف سطح الأنبوب نحو سبع
مرات، والزغيبات تضاعفه عشرين مرة أخرى: فتصير \qty{0.6}{m^2} من أنبوب
أملس عشرات الأمتار المربعة من \omterm{def:b1:body-plans-tissues:epithelium}{ظهارة} ماصة سماكتها خلية واحدة، كل خلية
منها على بعد بضعة ميكرومترات من شعرية دموية ووعاء لمفي في اللب الضام
لزغابتها (\cref{ch:b1:digestion-absorption}). \omterm{def:b1:body-plans-tissues:epithelium}{والظهارة} تتجدد من
الغدد عند قاعدتها كل أربعة أيام: فالصفيحة ووصلاتها وقطبيتها يُعاد
بناؤها باستمرار.
\end{example}

\begin{remark}[أنسجة الكائنين النموذجين]\label{rem:b1:body-plans-tissues:plants}
جمل النبات النسيجية الثلاث في
\cref{ch:b1:flowering-plant-organization} لا نظائر دقيقة لها هنا.
فبشرته \omterm{def:b1:body-plans-tissues:epithelium}{ظهارة} من حيث الوظيفة لكن خلاياها تمسكها جدر لا وصلات؛ ونسيجه
الأساسي ادخار ودعامة وتركيب ضوئي في آن؛ ولا عضل له ولا عصب. \omterm{def:b1:body-plans-tissues:tissue}{فأنسجة}
الحيوان الأربعة عدة جسم يتحرك ويحس؛ \omterm{def:b1:body-plans-tissues:tissue}{وأنسجة} النبات الثلاثة عدة جسم
ينمو.
\end{remark}

\section{مسألة: جدار المعي الدقيق}

\begin{problem}[{مسألة نهاية الأسبوع --- سلسلة نسيجية من المعي الدقيق
عند الإنسان تُقرأ من الأنبوب نزولًا إلى الزغيبة: الطبقات، والزغابات،
والخلايا، والوصلات، والتجدد، وصولًا إلى السطح الماص لزغابة واحدة}]
\label{pb:b1:body-plans-tissues:1}
المعي الدقيق عند الإنسان أنبوب طوله \qty{6}{m} وقطره الداخلي
\qty{3}{cm}. وطيات جداره الدائرية تضاعف سطح الأنبوب الأملس ثلاث
مرات. ومخاطيته تحمل \qty{25}{} زغابة في المليمتر المربع من السطح
المطوي، كل منها أسطوانة ارتفاعها \qty{0.6}{mm} وقطرها
\qty{0.15}{mm}. والزغابات تغطيها خلايا عمادية وجهها القمي مربع ضلعه
\qty{5}{\micro m}؛ ويحمل كل وجه قمي \qty{1500}{} زغيبة، وهي أسطوانات
طولها \qty{1}{\micro m} وقطرها \qty{0.1}{\micro m}. \omterm{def:b1:body-plans-tissues:epithelium}{والظهارة} تتجدد كل
\qty{4}{} أيام.

\medskip
\noindent\textbf{الجزء الأول --- الطبقات.}
يُفحص مقطع عرضي تحت المجهر.
\begin{enumerate}
  \item اذكر طبقات الجدار الأربع من اللمعة إلى الخارج وأنواع \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}
        في كل منها.
  \item يبين المقطع زغابات. فأي منطقة من الأنبوب الهضمي هي؟ وماذا
        كان مقطع من المعدة ليبين بدلًا منها؟
  \item بين طبقتي العضل في العضلية تقع تجمعات من خلايا كبيرة شاحبة.
        فما هي، وما الذي تتحكم فيه؟
  \item فسّر لماذا تسير طبقتا العضل في اتجاهين متعامدين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- السطوح.}
\begin{enumerate}[resume]
  \item احسب السطح الداخلي للأنبوب الأملس.
  \item احسب السطح بعد الطيات الدائرية.
  \item احسب السطح الجانبي لزغابة واحدة (أسطوانة، مع إهمال القمة).
  \item احسب سطح الزغابات لكل مليمتر مربع من المخاطية المطوية
        والعامل الذي تضاعف به الزغابات السطح.
  \item احسب السطح بعد الزغابات.
  \item احسب السطح الجانبي لزغيبة واحدة ثم لعدد \qty{1500}{} زغيبة على
        خلية واحدة.
  \item احسب العامل الذي تضاعف به الزغيبات السطح القمي لخلية.
  \item احسب مجموع السطح الماص للمعي الدقيق.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الخلايا.}
\begin{enumerate}[resume]
  \item احسب عدد الخلايا الظهارية التي تغطي زغابة واحدة.
  \item احسب عدد الزغابات في المعي الدقيق وعدد الخلايا الظهارية التي
        تغطيها.
  \item تتجدد \omterm{def:b1:body-plans-tissues:epithelium}{الظهارة} كل \qty{4}{} أيام. فكم خلية تُطرح في اللمعة في
        اليوم، وكم في الثانية؟
  \item أين تُنتج الخلايا البديلة، وماذا يقول هذا عن موضع الخلايا
        المنقسمة بالنسبة إلى الخلايا الماصة؟
  \item ارتفاع الخلية الظهارية \qty{25}{\micro m}. احسب حجم \omterm{def:b1:body-plans-tissues:epithelium}{الظهارة}
        في المعي الدقيق وكتلتها (بكثافة \qty{1.05}{g/cm^3}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الوصلات والمادة والزغابة.}
\begin{enumerate}[resume]
  \item كل خلية مسدودة إلى جاراتها بوصلة محكمة تدور حول قمتها. احسب
        طول الوصلة المحكمة لكل خلية وطولها الكلي في المعي الدقيق.
  \item فسّر لماذا لا تستطيع مادة في اللمعة أن تبلغ الدم إلا بعبور
        غشاءين خلويين.
  \item لب الزغابة \omterm{def:b1:body-plans-tissues:connective}{نسيج ضام} رخو فيه شبكة شعرية ووعاء لمفي مركزي.
        سمِّ نوعي الألياف خارج الخلوية اللذين يحويهما والخلية التي
        تفرزهما.
  \item سمِّ \omterm{def:b1:body-plans-tissues:layers}{الورقة الجنينية} التي تنشأ عنها \omterm{def:b1:body-plans-tissues:epithelium}{الظهارة} والتي ينشأ عنها
        اللب.
  \item تحوي الزغابة أيضًا خصلة من العضل الأملس تجعلها تقصر إيقاعيًا.
        اقترح وظيفتها.
  \item مرض يسطح الزغابات (فيعود السطح إلى سطح الأنبوب المطوي). بأي
        عامل يهبط السطح الماص، وما النتائج؟
  \item احسب السطح الماص لزغابة واحدة مع زغيباتها، بالمليمتر المربع.
  \item اذكر النتيجة: السطح الماص لزغابة واحدة، وعدد الزغابات في
        المعي الدقيق.
\end{enumerate}
\end{problem}
