\chapter{Inspanning en de energiebehoefte van het lichaam}\label{ch:g10:exercise-and-energy}

Een wielrenster trapt op een laboratoriumfiets tegen een rem die haar
vermogen meet: \qty{100}{W}, dan \qty{150}{W}, dan \qty{200}{W}. Een
masker over haar gezicht stuurt elke ademteug naar een analysator. De
getallen op het scherm klimmen mee met de rem: hoe meer mechanisch werk
zij levert, des te meer zuurstof zij verbruikt en des te meer
koolstofdioxide zij uitademt. Haar spieren draaien de ademhaling van
\cref{ch:g10:cell-metabolism} op volle kracht, en de analysator leest de
balans daarvan af. Dit hoofdstuk volgt de energie van inspanning, van
het voedsel dat haar levert tot de zuurstof die haar vrijmaakt.

\section{Energie erin, werk eruit}

\begin{definition}[Energieverbruik]\label{def:g10:exercise-and-energy:expenditure}
Het \emph{energieverbruik}\index{energieverbruik} van het lichaam, in
joule, is de totale energie die zijn \omterm{def:g10:cells-common-unit:cell}{cellen} in een gegeven tijd uit
organische moleculen vrijmaken. Gedeeld door de tijd is het een
\emph{metabool vermogen}, in watt. In rust verbruikt een volwassene
ongeveer \qty{80}{W} --- zo'n \qty{7000}{kJ} per dag --- om het hart te
laten kloppen, de hersenen te laten werken, de temperatuur op
\qty{37}{\celsius} te houden en de \omterm{def:g10:cells-common-unit:cell}{cellen} in leven te houden; dat is de
\emph{basale \omterm{def:g10:cell-metabolism:metabolism}{stofwisseling}}. Elke bezigheid komt daar bovenop.
\end{definition}

\begin{proposition}[Waar de energie blijft]\label{prop:g10:exercise-and-energy:efficiency}
Een werkende spier zet slechts ongeveer een kwart van de energie die zij
vrijmaakt om in mechanisch werk; de overige drie kwart gaat weg als
warmte. Een mechanisch vermogen $P$ leveren kost het lichaam dus een
metabool vermogen van ongeveer $4P$ boven de rust: een wielrenster die
\qty{100}{W} op de pedalen zet, verbruikt zo'n \qty{400}{W} in haar
spieren, en haar lichaam moet \qty{300}{W} warmte kwijtraken --- vooral
door te zweten.
\end{proposition}
\admitted

\begin{example}[Dagbudgetten]\label{ex:g10:exercise-and-energy:budgets}
Een leerling die in de klas zit, naar school loopt en slaapt, verbruikt
ongeveer \qty{9000}{kJ} per dag; een uur voetbal voegt zo'n
\qty{2500}{kJ} toe; een dag in de bergen met een rugzak, in totaal
\qty{15000}{kJ}. Een wielrenner in een bergrit van een wedstrijd
verbruikt \qty{25000}{kJ} en kan gedurende de dag niet genoeg eten om
dat te dekken. De energie komt uit voedsel
(\cref{ch:g10:chemistry-of-life}): \qty{17}{kJ} per gram \omterm{def:g10:chemistry-of-life:families}{koolhydraat} of
\omterm{def:g10:chemistry-of-life:families}{eiwit}, \qty{38}{kJ} per gram \omterm{def:g10:chemistry-of-life:families}{lipide}.
\end{example}

\section{Zuurstofverbruik meet het energieverbruik}

\begin{proposition}[Het zuurstofequivalent]\label{prop:g10:exercise-and-energy:oxygen}
Omdat de energie van inspanning door de \omterm{prop:g10:cell-metabolism:respiration}{celademhaling} wordt vrijgemaakt,
die zuurstof verbruikt, meet het zuurstofverbruik van het lichaam zijn
\omterm{def:g10:exercise-and-energy:expenditure}{energieverbruik}: elke liter verbruikte zuurstof komt overeen met
ongeveer \qty{20}{kJ} die vrijkomt, wat de mengverhouding van verbrande
glucose en \omterm{def:g10:chemistry-of-life:families}{lipiden} ook is. In rust verbruikt iemand ongeveer
\qty{0.25}{L} zuurstof per minuut; tijdens inspanning stijgt het verbruik
evenredig met het geleverde vermogen, tot aan een persoonlijk plafond.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
De ademhalingsvergelijking uit \cref{ch:g10:cell-metabolism} geeft
\qty{2870}{kJ} voor \qty{6}{mol} zuurstof, dus \qty{478}{kJ} per mol, en
een mol gas neemt ongeveer \qty{24}{L} in: \qty{20}{kJ} per liter. Voor
\omterm{def:g10:chemistry-of-life:families}{lipiden} is het getal \qty{19.6}{kJ/L}, voor \omterm{def:g10:chemistry-of-life:families}{koolhydraten}
\qty{21.1}{kJ/L}, dus het mengsel doet er nauwelijks toe. Metingen in
afgesloten kamers, waarin de warmte die een proefpersoon afgeeft
rechtstreeks wordt opgevangen, komen tot op enkele procenten met het
zuurstofgetal overeen.
\end{proof}

\begin{definition}[Maximale zuurstofopname]\label{def:g10:exercise-and-energy:vo2max}
De \emph{maximale zuurstofopname}\index{maximale zuurstofopname},
geschreven als $\dot V\!\mathrm{O_2}$max, is het hoogste tempo waarmee
het lichaam van iemand zuurstof kan opnemen en gebruiken, in liter per
minuut --- of, om mensen van verschillende grootte te vergelijken, in
milliliter per minuut per kilogram lichaamsmassa. Zij bepaalt het
hoogste vermogen dat door de ademhaling alleen kan worden volgehouden.
Gebruikelijke waarden: \qty{35}{mL/min} per kilogram bij een zittende
volwassene, \num{50} bij een getrainde leerling, \num{80} of meer bij een
topsporter in het duurvermogen.
\end{definition}

\begin{example}[De kromme lezen]\label{ex:g10:exercise-and-energy:reading}
In de figuur kost \qty{100}{W} trappen \qty{1.5}{L/min} zuurstof:
$1.5 \times 20 = \qty{30}{kJ}$ per minuut, dus \qty{500}{W} metabool
vermogen --- de \qty{100}{W} werk plus \qty{400}{W} warmte en
rustbehoefte, in lijn met
\cref{prop:g10:exercise-and-energy:efficiency}. Het plafond van
\qty{3.3}{L/min} bij een proefpersoon van \qty{66}{kg} is
\qty{50}{mL/min} per kilogram.
\end{example}

\section{De brandstoffen van de spier}

\begin{proposition}[Drie bronnen van ATP]\label{prop:g10:exercise-and-energy:fuels}
Spiersamentrekking wordt in ATP betaald, en de \omterm{def:g10:cells-common-unit:cell}{cel} houdt daarvan maar
enkele seconden voorraad. Het wordt langs drie wegen vernieuwd, elk met
een eigen snelheid en een eigen capaciteit:
\begin{itemize}
  \item een kleine voorraad van een fosfaatverbinding in de spier, die
        ATP onmiddellijk opnieuw vormt maar ongeveer tien seconden meegaat
        --- de sprint;
  \item de melkzuurgisting van glucose in het \omterm{def:g10:cells-common-unit:cell}{cytoplasma}, snel maar
        verkwistend, die een maximale inspanning één tot twee minuten
        draagt en melkzuur oplevert;
  \item de \omterm{prop:g10:cell-metabolism:respiration}{celademhaling} in de \omterm{def:g10:cells-common-unit:organelle}{mitochondriën}, die trager op volle
        snelheid komt maar vrijwel onbeperkt kan doorgaan en glucose uit
        het bloed, \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} uit de spier en vetzuren uit het lichaamsvet
        verbrandt.
\end{itemize}
Het aandeel van elk hangt af van de intensiteit en de duur van de
inspanning.
\end{proposition}
\admitted

\begin{definition}[Glycogeen]\label{def:g10:exercise-and-energy:glycogen}
\emph{Glycogeen}\index{glycogeen} is de dierlijke vorm van opgeslagen
\omterm{def:g10:chemistry-of-life:families}{koolhydraat}: een vertakte keten van glucose-eenheden, bewaard als
korrels in de spiercellen (ongeveer \qty{400}{g} bij een volwassene) en
in de lever (ongeveer \qty{100}{g}). Spierglycogeen voedt de spier die
het opslaat; leverglycogeen wordt afgebroken tot glucose die aan het
bloed wordt afgegeven, en houdt de bloedglucose bij \qty{1}{g/L} voor de
hersenen en de andere weefsels. Vet, opgeslagen in vetcellen, is de veel
grotere reserve (\cref{ch:g10:chemistry-of-life}), maar het kan alleen
langzaam en alleen met zuurstof worden verbrand.
\end{definition}

\begin{example}[De muur van de marathon]\label{ex:g10:exercise-and-energy:wall}
Hardlopen kost ongeveer \qty{4}{kJ} per kilogram lichaamsmassa per
kilometer; een loper van \qty{70}{kg} verbruikt \qty{280}{kJ/km}, en een
marathon kost \qty{11800}{kJ}. De glycogeenvoorraden bevatten ongeveer
\qty{8500}{kJ}. Verbrandt de loper zijn \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} te snel, dan raken de
voorraden rond de dertigste kilometer op; de spieren moeten dan op vet
terugvallen, dat energie met nauwelijks de halve snelheid levert, en het
tempo stort in --- "tegen de muur lopen". Getrainde lopers verbranden
van meet af aan een groter aandeel vet, en eten \omterm{def:g10:chemistry-of-life:families}{koolhydraat} tijdens de
wedstrijd.
\end{example}

\section{Meten en redeneren}

\begin{method}[Een energieberekening voor inspanning]\label{met:g10:exercise-and-energy:calc}
\begin{enumerate}
  \item Uit de zuurstof: energie (\unit{kJ}) $=$ verbruikte zuurstof
        (\unit{L}) $\times 20$. Trek het rustverbruik af als de vraag
        alleen naar de kosten van de inspanning vraagt.
  \item Uit het mechanische vermogen: metabool vermogen $\approx 4 P$;
        energie $=$ vermogen $\times$ tijd (\qty{1}{W} gedurende
        \qty{1}{s} is \qty{1}{J}; \qty{1}{kWh} $= \qty{3600}{kJ}$).
  \item Uit de brandstof: verbrande massa $=$ energie $/$ energiewaarde
        (\qty{17}{kJ/g} \omterm{def:g10:chemistry-of-life:families}{koolhydraat}, \qty{38}{kJ/g} \omterm{def:g10:chemistry-of-life:families}{lipide}); \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen}
        wordt met drie keer zijn massa aan water opgeslagen.
  \item Toets de orde van grootte: een hele dag is ongeveer
        \qty{10000}{kJ}; een zwaar uur \qtyrange{2000}{3000}{kJ}; een
        marathon \qty{12000}{kJ}.
\end{enumerate}
\end{method}

\begin{example}[Een uur fietsen]\label{ex:g10:exercise-and-energy:hour}
Eén uur op \qty{150}{W} op de pedalen: metabool vermogen ongeveer
\qty{600}{W}, energie $600 \times 3600 = \qty{2.16e6}{J} \approx
\qty{2200}{kJ}$; zuurstof $2200/20 = \qty{110}{L}$, dus
\qty{1.8}{L/min} --- in overeenstemming met de kromme. Komt de helft uit
\omterm{def:g10:chemistry-of-life:families}{koolhydraat}, dan is dat $1100/17 \approx \qty{65}{g}$ glucose of
\omterm{def:g10:exercise-and-energy:glycogen}{glycogeen}, en $1100/38 \approx \qty{29}{g}$ vet.
\end{example}

\begin{remark}[Waarom de zuurstof moet aankomen]\label{rem:g10:exercise-and-energy:delivery}
Alles hierboven veronderstelt dat de zuurstof de \omterm{def:g10:cells-common-unit:organelle}{mitochondriën} zo snel
bereikt als zij haar kunnen gebruiken. Dat is de taak van de longen, het
hart en het bloed, waarvan de reactie op inspanning --- sneller ademen,
sneller en krachtiger kloppen, bloed dat naar de spieren wordt geleid ---
het onderwerp is van \cref{ch:g10:heart-lungs-effort}.
$\dot V\!\mathrm{O_2}$max is vooral een grens van de aanvoer, niet van de
honger van de spieren.
\end{remark}

\section{Probleem: Waar de muur staat}

\begin{problem}[{Weekendprobleem --- de energiebalans van een marathon:
de zuurstof die de loopster inademt, het glycogeen dat zij draagt, de
kilometer waarop het opraakt, en wat training verandert}]
\label{pb:g10:exercise-and-energy:1}
Nadia, \qty{60}{kg}, loopt een marathon (\qty{42.2}{km}) in
\qty{3}{h}\,\qty{30}{min}. Hardlopen kost \qty{4.0}{kJ} per kilogram per
kilometer. Haar spieren bevatten \qty{300}{g} \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} en haar lever
\qty{80}{g}; \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} en glucose geven \qty{17}{kJ/g}, vet
\qty{38}{kJ/g}; één liter zuurstof maakt \qty{20}{kJ} vrij. Haar
$\dot V\!\mathrm{O_2}$max is \qty{52}{mL/min} per kilogram.

\medskip
\noindent\textbf{Deel I --- De kosten van de wedstrijd.}
\begin{enumerate}
  \item Bereken de energiekosten van de marathon voor Nadia.
  \item Bereken haar gemiddelde metabole vermogen tijdens de wedstrijd,
        in watt.
  \item Bereken het volume zuurstof dat zij over de wedstrijd verbruikt,
        en vervolgens haar gemiddelde zuurstofverbruik in liter per
        minuut.
  \item Druk dat verbruik uit in milliliter per minuut per kilogram, en
        als percentage van haar $\dot V\!\mathrm{O_2}$max.
  \item Haar rustverbruik is \qty{0.2}{L/min}. Welk deel van haar
        verbruik op wedstrijddag komt door het lopen zelf?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De voorraden.}
\begin{enumerate}[resume]
  \item Bereken de energie in haar glycogeenvoorraden.
  \item Als zij alleen \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} verbrandde, bij welke kilometer zouden
        de voorraden dan opraken?
  \item In werkelijkheid verbrandt zij een mengsel: bij haar
        wedstrijdtempo 80\% \omterm{def:g10:chemistry-of-life:families}{koolhydraat} en 20\% vet. Bij welke kilometer
        raakt het \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} nu op?
  \item Hoeveel vet verbrandt zij in de wedstrijd? Vergelijk met de
        \qty{9}{kg} vet die een vrouw van \qty{60}{kg} doorgaans draagt.
  \item Leg met \cref{prop:g10:exercise-and-energy:fuels} uit waarom zij
        niet eenvoudigweg alleen vet kan verbranden zodra het \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen}
        op is en toch hetzelfde tempo kan aanhouden.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De muur verslaan.}
\begin{enumerate}[resume]
  \item Tijdens de wedstrijd drinkt zij een suikeroplossing die
        \qty{60}{g} \omterm{def:g10:chemistry-of-life:families}{koolhydraat} per uur levert. Hoeveel energie voegt dat
        over de wedstrijd toe, en hoeveel kilometer koolhydraatverbruik
        dekt dat?
  \item Doe vraag 8 opnieuw, met de drank erbij. Komt zij nu aan de
        finish voordat het \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} opraakt?
  \item Drie dagen voor de wedstrijd "stapelt" zij \omterm{def:g10:chemistry-of-life:families}{koolhydraat}, waardoor
        haar spierglycogeen tot \qty{500}{g} stijgt. Elke gram wordt met
        \qty{3}{g} water opgeslagen. Met hoeveel stijgt haar massa, en
        wat kost dat aan energie per kilometer?
  \item Training verschuift haar wedstrijdmengsel naar 55\% \omterm{def:g10:chemistry-of-life:families}{koolhydraat}
        en 45\% vet. Bij welke kilometer zou het \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} met stapelen
        en met de drank opraken? Becommentarieer dat.
  \item Waarom laten de hersenen, die alleen glucose gebruiken, het
        opraken van \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen} als uitputting voelen, ook als de spieren
        nog vet te verbranden hebben?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De getallen van een kampioen.}
Een topsporter van \qty{55}{kg} loopt de marathon in \qty{2}{h}\,
\qty{05}{min}, met een loopkosten van \qty{3.6}{kJ} per kilogram per
kilometer en een $\dot V\!\mathrm{O_2}$max van \qty{80}{mL/min} per
kilogram.
\begin{enumerate}[resume]
  \item Bereken de energiekosten van de wedstrijd voor de kampioen en
        zijn gemiddelde zuurstofverbruik in liter per minuut.
  \item Welk percentage van zijn $\dot V\!\mathrm{O_2}$max houdt hij vol?
        Vergelijk met het cijfer van Nadia.
  \item Twee dingen onderscheiden de kampioen: een hoger plafond en
        lagere kosten per kilometer. Welke van de twee is de
        "loopeconomie"? Bereken hoezeer alleen al het verschil in kosten
        de energie van de wedstrijd voor een loper van \qty{55}{kg}
        verandert.
  \item Zijn spieren bevatten na het stapelen \qty{500}{g} \omterm{def:g10:exercise-and-energy:glycogen}{glycogeen}.
        Haalt hij bij 70\% koolhydraatverbruik de finish daarop, zonder
        te drinken?
  \item Geef het resultaat: voor Nadia de kilometer waar de muur zonder
        voorbereiding staat, en de drie maatregelen die hem voorbij de
        finish verschuiven.
\end{enumerate}
\end{problem}
