\chapter{من البويضة إلى الولادة}\label{ch:g8:from-egg-to-birth}

روى \Cref{ch:g5:human-reproduction} الأشهر التسعة حكايةً؛
وأنت تملك اليوم كل آلة تحتاج الحكاية إلى عزفها كما ينبغي:
\omterm{def:g5:first-look-at-cells:cell}{الخلايا} وانقساماتها، والمشيجان وملتقاهما، \omterm{def:g8:puberty:hormones}{والهرمونات}،
وتوصيلات \omterm{prop:g4:journey-of-food:absorb}{الدم}. ويعيد هذا الفصل عرض النمو البشري والآلة
ظاهرة --- من \omterm{def:g5:first-look-at-cells:cell}{خلية} واحدة إلى ثلاثة آلاف مليار، في أربعين
أسبوعًا.

\section{الأسبوع الأول: من اللاقحة إلى الساكن}

\begin{proposition}[الرحلة نزولًا في قناة البيض]\label{prop:g8:from-egg-to-birth:firstweek}
\omterm{def:g5:flower-to-fruit:steps}{الإخصاب} في \omterm{def:g8:reproductive-systems:female}{قناة البيض}
(\cref{def:g8:reproductive-systems:female}) يشغّل ساعتين
في وقت واحد:
\begin{enumerate}
  \item \emph{الانقسام}: تنقسم \omterm{prop:g8:sexual-reproduction:core}{اللاقحة} في غضون يوم، ثم
        تنقسم مرة بعد مرة
        (\cref{ex:g5:first-look-at-cells:one}) --- خليتان،
        فأربع، فثمان، فكرة كالتوتة من عشرات \omterm{def:g5:first-look-at-cells:cell}{الخلايا} في
        اليوم الرابع، وكل ذلك وهي تنساب في \omterm{def:g8:reproductive-systems:female}{قناة البيض}
        نحو \omterm{prop:g5:human-reproduction:plan}{الرحم}؛
  \item \emph{الانغراس}: نحو اليوم السابع تبلغ الكرة
        الصغيرة \omterm{prop:g5:human-reproduction:plan}{الرحم} وتنغرس في البطانة المهيأة (الحدث 4
        في \cref{prop:g8:reproductive-systems:cycle}) ---
        فيصل الساكن، وتتوقف الدورة، وتعلن \omterm{def:g8:puberty:hormones}{الهرمونات}
        للجهاز كله: حمل.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[على أي شيء تعيش الانقسامات الأولى]\label{ex:g8:from-egg-to-birth:provisions}
خلال الأسبوع الأول لا يأكل المسافر وجبة ولا يتلقى
توصيلة: إنه يجري على زاد \omterm{def:g4:animal-reproduction:sexual}{البويضة} نفسها --- وهو تمامًا ما
كان تزويد \cref{ex:g8:reproductive-systems:egg} من أجله.
جاءت \omterm{def:g8:reproductive-systems:male}{النطفة} \omterm{def:g6:cell-unit-of-life:parts}{بنواة} الأب وقليل غير ذلك؛ وجاءت \omterm{def:g4:animal-reproduction:sexual}{البويضة}
\omterm{def:g6:cell-unit-of-life:parts}{بالنواة} الأخرى \emph{ومعها} المخزن. والانغراس هو، من بين
أشياء أخرى، المخزن يوشك أن ينفد والساكن يشترك في خدمة
الغرف.
\end{example}

\section{المضغة والمشيمة والجنين}

\begin{definition}[المشيمة]\label{def:g8:from-egg-to-birth:placenta}
من الكرة المنغرسة ينمو بناءان في وقت واحد: \emph{\omterm{def:g5:human-reproduction:pregnancy}{المضغة}}
نفسها، وواجهة إعالتها --- \emph{المشيمة}\index{مشيمة}،
قرص ينمو في جدار \omterm{prop:g5:human-reproduction:plan}{الرحم} حيث يقترب دم الأم ودم \omterm{def:g5:human-reproduction:pregnancy}{المضغة}
اقترابًا شديدًا \emph{من غير أن يختلطا}. وعبر سطوحها
الشاسعة الرقيقة --- مواصفة
\cref{prop:g7:how-animals-breathe:spec}، مبنية مرة أخرى
--- تعبر \omterm{def:g7:digestion-and-nutrients:families}{المغذيات} \omterm{def:g7:what-is-respiration:respiration}{والأكسجين} إلى \omterm{def:g5:human-reproduction:pregnancy}{المضغة}، وتعبر الفضلات
\omterm{def:g7:what-is-respiration:respiration}{وثاني أكسيد الكربون} راجعة (والحبل في
\cref{prop:g5:human-reproduction:cord} يمتد بين المشيمة
\omterm{def:g5:human-reproduction:pregnancy}{والمضغة}). المشيمة \omterm{def:g7:how-animals-breathe:four}{رئة} ومعي \omterm{def:g7:removing-wastes:kidneys}{وكلية} للراكب --- الحدود
الثلاثة في عضو واحد.
\end{definition}

\begin{proposition}[الجدول الزمني، معادًا بالآلات]\label{prop:g8:from-egg-to-birth:timetable}
معالم \Cref{ex:g5:human-reproduction:timetable}، وقد
صارت مفسرة:
\begin{itemize}
  \item الأسابيع 1--8، \emph{\omterm{def:g5:human-reproduction:pregnancy}{المضغة}}: يوضع مخطط الجسم ---
        تنقسم \omterm{def:g5:first-look-at-cells:cell}{الخلايا} وترتحل و\emph{تتخصص}
        (\cref{rem:g6:cell-unit-of-life:twoways}) فتصير
        قلبًا وعصبًا وعظمًا وسائر ما يلزم؛ ويدق \omterm{def:g4:heart-and-blood:heart}{القلب} منذ
        الأسبوع الرابع، لأن اقتصاد المخزن \omterm{def:g8:from-egg-to-birth:placenta}{والمشيمة} يحتاج
        إلى مضخته باكرًا؛
  \item الأسابيع 9--40، \emph{\omterm{def:g5:human-reproduction:pregnancy}{الجنين}}: المخطط قد بني،
        فالنمو والنضج --- \omterm{def:g5:brain-in-command:system}{الدماغ} يوصّل أسلاكه، \omterm{def:g7:how-animals-breathe:four}{والرئتان}
        تتمرنان فارغتين
        (\cref{exo:g5:human-reproduction:11} فسّر ظهورهما
        الأول مع الصرخة)، وركلات الشهر الرابع تعلن \omterm{def:g3:bones-and-muscles:muscle}{عضلات}
        عاملة؛
  \item وطوال ذلك: تبقي \omterm{def:g8:puberty:hormones}{الهرمونات} \omterm{prop:g5:human-reproduction:plan}{الرحم} هادئًا والدورة
        موقوفة --- كيمياء
        \cref{def:g8:puberty:hormones}، تدير شؤون \omterm{def:g5:human-reproduction:pregnancy}{الحمل}.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[لماذا تطلب الأسابيع الأولى أشد العناية]\label{ex:g8:from-egg-to-birth:earlycare}
حذّر \Cref{prop:g5:human-reproduction:care} من أن التبغ
والكحول يعبران إلى الطفل؛ ويقول الجدول الزمني متى يكون
ذلك أشد ما يكون خطرًا: في أسابيع \omterm{def:g5:human-reproduction:pregnancy}{المضغة} التي يوضع فيها
المخطط، حين يكون كل عضو حشدًا صغيرًا من \omterm{def:g5:first-look-at-cells:cell}{خلايا} تتخصص،
فتعليمة واحدة مضطربة قد يتردد صداها في البناء كله ---
وهي بعينها الأسابيع التي قد يمر فيها \omterm{def:g5:human-reproduction:pregnancy}{الحمل} من غير أن
يُلحظ. ومن هنا حساب الأطباء الصريح: العناية تبدأ لا عند
أول فحص، بل عند أول احتمال.
\end{example}

\section{الولادة، وتبديل الحدود}

\begin{proposition}[الولادة تبديلًا تأمر به الهرمونات]\label{prop:g8:from-egg-to-birth:birth}
نحو الأسبوع الأربعين، تنهي إشارات هرمونية --- إشارات
الطفل والأم في حوار --- هدوء \omterm{prop:g5:human-reproduction:plan}{الرحم} الطويل:
\begin{enumerate}
  \item \emph{المخاض}: ينقبض \omterm{prop:g5:human-reproduction:plan}{الرحم}، أقوى \omterm{def:g3:bones-and-muscles:muscle}{عضلات} الجسم
        (\cref{rem:g8:reproductive-systems:living} عرض
        عمله الألطف)، في موجات تشتد فتفتح الطريق وتدفع
        الطفل نزولًا في قناة \omterm{def:g2:born-grow-die:cycle}{الولادة}؛
  \item \emph{\omterm{def:g2:born-grow-die:cycle}{الولادة}}: بالرأس أولًا في العادة
        (\cref{ex:g5:human-reproduction:birth})؛
  \item \emph{تبديل الحدود}: تملأ الصرخة الأولى \omterm{def:g7:how-animals-breathe:four}{رئتين} لم
        تستعملا قط؛ وينتهي مرور \omterm{prop:g5:human-reproduction:cord}{الحبل السري} فيقطع؛ وفي
        دقائق يجري المولود على رئتيه وأمعائه المقبلة
        وكليتيه --- كل حد كانت \omterm{def:g8:from-egg-to-birth:placenta}{المشيمة} تغطيه
        (\cref{exo:g5:human-reproduction:11}، والخريطة
        الآن كاملة)؛
  \item \emph{الخلاص}: تنفصل \omterm{def:g8:from-egg-to-birth:placenta}{المشيمة}، وقد تمت مهمتها،
        فتتبع --- وهي العضو الوحيد الذي يبنيه الجسم
        لاستعمال واحد ثم يطرحه.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[أول تدقيق للمولود]\label{ex:g8:from-egg-to-birth:audit}
تعطي القابلات المولود درجات بعد دقائق من \omterm{def:g2:born-grow-die:cycle}{الولادة} ---
\omterm{def:g7:what-is-respiration:respiration}{التنفس}، \omterm{def:g4:heart-and-blood:heart}{ودقة القلب}، واللون، والحركة، والصرخة. اقرأ
القائمة بعيني هذا الكتاب: إنها حلقة التزويد والإزالة في
\cref{prop:g7:removing-wastes:complete}، مدققة في أول
تشغيل منفرد لها. فأول امتحان يجتازه كل إنسان امتحان في
علم وظائف الأعضاء.
\end{example}

\begin{method}[شرح أي مرحلة من الحمل]\label{met:g8:from-egg-to-birth:explain}
إذا سئلت عن أي لحظة من الأشهر التسعة، فأجب بأربع
آلات:
\begin{enumerate}
  \item \emph{\omterm{def:g5:first-look-at-cells:cell}{الخلايا}}: ماذا تفعل الانقسامات والتخصصات؟
  \item \emph{التزويد}: مخزن، أم \omterm{def:g8:from-egg-to-birth:placenta}{مشيمة} --- ما الذي يعبر،
        وما الذي يجب ألا يعبر؟
  \item \emph{\omterm{def:g8:puberty:hormones}{الهرمونات}}: ما الذي يُبقى، أو يؤمر به؟
  \item \emph{العناية}: ماذا تطلب المرحلة من عادات الأم
        ومن تقويم الأطباء؟
\end{enumerate}
\end{method}

\section{مسألة: خط زمن القابلة}

\begin{problem}[{مسألة نهاية الأسبوع --- حمل متابع
زيارةً زيارة}]\label{pb:g8:from-egg-to-birth:1}
ملف قابلة عن حمل واحد (متخيَّل): زيارة التسجيل في
الأسبوع 8، وفحصان بالصدى في الأسبوعين 12 و20، وفحصان في
28 و36، \omterm{def:g2:born-grow-die:cycle}{والولادة} في الأسبوع 40. اقرأ الملف بآلات
الفصل.

\medskip\noindent\textbf{الجزء الأول --- صفحات الملف
الأولى.}
\begin{enumerate}
  \item تأكد \omterm{def:g5:human-reproduction:pregnancy}{الحمل} باختبار هرموني قبل الزيارة الأولى
        بأسابيع. أي إشارة يقرأ مثل هذا الاختبار، ومن أي
        حدث في \cref{prop:g8:from-egg-to-birth:firstweek}
        فصاعدًا توجد؟
  \item في الأسبوع 8 تسجل القابلة ``\omterm{def:g5:human-reproduction:pregnancy}{المضغة} تامة المخطط،
        في حجم حبة الفول تقريبًا''. ترجم ذلك إلى لغة
        الجدول الزمني --- أي مرحلة تنتهي هنا؟
  \item تبدأ نصيحة زيارة التسجيل بلا كحول ولا دخان ---
        وبحمض الفوليك، وهو فيتامين البنّاء، ``من قبل
        البداية إن أمكن''. برر الأمور الثلاثة جميعًا وفق
        \cref{ex:g8:from-egg-to-birth:earlycare}.
  \item يظهر فحص الأسبوع 12 \omterm{def:g4:heart-and-blood:heart}{دقة القلب} التي بدأت في
        الأسبوع 4. لماذا هذا الباكر، بلغة التزويد؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- الأشهر الوسطى.}
\begin{enumerate}[resume]
  \item يقيس فحص الأسبوع 20 كل عضو وموضع \omterm{def:g8:from-egg-to-birth:placenta}{المشيمة}. اذكر
        شغل مرحلة \omterm{def:g5:human-reproduction:pregnancy}{الجنين}، ومهام \omterm{def:g8:from-egg-to-birth:placenta}{المشيمة} الثلاث التي يجري
        التحقق منها.
  \item تبلغ الأم عن أول الركلات قرب الأسبوع 18. أي آلة
        تعلن عن نفسها (\cref{prop:g5:brain-in-command:loop}
        يسمي السلسلة)؟
  \item في الأسبوع 28 تفحص القابلة سكر دم الأم وضغطه.
        لماذا تحمل دورة الأم حملًا مضاعفًا --- حدود من
        تدير؟
  \item الأسابيع 30--36: ينمو \omterm{def:g5:human-reproduction:pregnancy}{الجنين} في الأغلب وتتمرن
        \omterm{def:g7:how-animals-breathe:four}{الرئتان}. تتمرنان على ماذا --- ولأي لحظة واحدة؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- الصفحة الأخيرة.}
\begin{enumerate}[resume]
  \item الأسبوع 40: المخاض. سمّ \omterm{def:g3:bones-and-muscles:muscle}{العضلة}، ومهمتيها، ومن أمر
        بالعمل (\cref{prop:g8:from-egg-to-birth:birth}).
  \item قيد \omterm{def:g2:born-grow-die:cycle}{الولادة} في الملف: ``صرخة قوية في الحال؛ الحبل
        مشبوك ومقطوع؛ \omterm{def:g8:from-egg-to-birth:placenta}{المشيمة} تامة.'' دقّق العبارات
        الثلاث بتبديل الحدود.
  \item يحصل المولود على درجات عالية في قائمة الدقائق.
        اذكر ما الذي تشهد به القائمة، بلغة
        \cref{ex:g8:from-egg-to-birth:audit}.
  \item اختم الملف بجملة واحدة للوالدين: ماذا بنت
        الأسابيع الأربعون، ومن أي \omterm{def:g5:first-look-at-cells:cell}{خلية} واحدة، وبأي آلة.
\end{enumerate}
\end{problem}
