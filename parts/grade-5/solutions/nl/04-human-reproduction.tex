\section*{Hoofdstuk \ref{ch:g5:human-reproduction} --- Voortplanting bij de mens}

\begin{solution}{exo:g5:human-reproduction:1}
De \omterm{def:g4:animal-reproduction:sexual}{eicel} van de moeder, de \omterm{def:g4:animal-reproduction:sexual}{zaadcel} van de vader; hun samengaan is de
\omterm{def:g5:flower-to-fruit:steps}{bevruchting}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:2}
In de \omterm{prop:g5:human-reproduction:plan}{baarmoeder} van de moeder, ongeveer negen maanden --- de \omterm{def:g5:human-reproduction:pregnancy}{zwangerschap}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:3}
Eerst \omterm{def:g5:human-reproduction:pregnancy}{embryo}, dan \omterm{def:g5:human-reproduction:pregnancy}{foetus} --- vanaf ongeveer twee maanden, zodra het
bouwplan is uitgezet.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:4}
Via de \omterm{prop:g5:human-reproduction:cord}{navelstreng}: \omterm{def:g4:journey-of-food:digestion}{voedingsstoffen} en zuurstof steken over van het
\omterm{prop:g4:journey-of-food:absorb}{bloed} van de moeder naar dat van de \omterm{def:g3:stages-of-life:stages}{baby}, en de afvalstoffen van de \omterm{def:g3:stages-of-life:stages}{baby}
steken terug zodat het lichaam van de moeder ze afhandelt.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:5}
De plek waar jouw eigen \omterm{prop:g5:human-reproduction:cord}{navelstreng} vastzat, voordat die bij de \omterm{def:g2:born-grow-die:cycle}{geboorte}
werd doorgeknipt.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:6}
De eerste kreet vult de \omterm{def:g4:breathing:organs}{longen} voor het eerst met lucht: tot dat moment
hadden ze nooit geademd --- de zuurstof kwam altijd via de streng binnen.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:7}
\omterm{def:g5:flower-to-fruit:steps}{Bevruchting} in het lichaam van de moeder; een levendbarende \omterm{def:g2:born-grow-die:cycle}{geboorte} (de
\omterm{def:g3:stages-of-life:stages}{baby} groeit in de \omterm{prop:g5:human-reproduction:plan}{baarmoeder} en wordt \omterm{def:g2:animals-grow-up:born}{levend geboren}); daarna voeden met
\omterm{def:g2:animals-grow-up:born}{melk} --- precies de kenmerken van de klasse van de \omterm{prop:g3:sorting-living-things:groups}{zoogdieren}.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:8}
Omdat alles wat in het \omterm{prop:g4:journey-of-food:absorb}{bloed} van de moeder zit de \omterm{def:g3:stages-of-life:stages}{baby} via de streng
bereikt --- de schadelijke stoffen van rook en alcohol inbegrepen --- en
een \omterm{def:g5:human-reproduction:pregnancy}{foetus} veel kwetsbaarder is dan een volwassene.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:9}
Eeneiige tweeling: één bevruchte \omterm{def:g4:animal-reproduction:sexual}{eicel} die zich in twee \omterm{def:g5:human-reproduction:pregnancy}{embryo}'s
splitste. Gewone (twee-eiige) tweeling: twee \omterm{def:g4:animal-reproduction:sexual}{eicellen} bevrucht door twee
\omterm{def:g4:animal-reproduction:sexual}{zaadcellen}, die alleen de \omterm{prop:g5:human-reproduction:plan}{baarmoeder} deelden. Het eeneiige paar lijkt op
elkaar als kopieën omdat een lichaam uit zijn eerste \omterm{def:g5:first-look-at-cells:cell}{cel} gebouwd wordt
--- en zij zijn uit dezelfde begonnen.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:10}
Aan de kant van de olifant, en daar nog voorbij: meestal één \omterm{def:g3:stages-of-life:stages}{baby}, jaren
van hulpeloosheid, en zorg die niet alleen voeden en beschermen omvat
maar ook onderwijzen --- taal, vaardigheden, school. De miljoenen zonder
zorg van de kabeljauw zijn de tegenpool.
\end{solution}

\begin{solution}{exo:g5:human-reproduction:11}
De taak van zuurstof laden en koolstofdioxide lossen: tot de \omterm{def:g2:born-grow-die:cycle}{geboorte}
deed het \omterm{prop:g4:journey-of-food:absorb}{bloed} van de moeder dat via de streng; vanaf de eerste teug
nemen de luchtzakjes van de eigen \omterm{def:g4:breathing:organs}{longen} van de \omterm{def:g3:stages-of-life:stages}{baby} het over. De kreet
is goed getimed omdat hij op precies de minuut komt waarop de oude
bevoorradingslijn wordt afgesneden --- de \omterm{def:g4:breathing:organs}{longen} moeten beginnen op het
moment dat de streng ophoudt.
\end{solution}
