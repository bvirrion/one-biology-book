\section{Los rasgos y de dónde vienen}

\begin{definition}[Los rasgos]\label{def:g9:heredity-and-traits:traits}
Un \emph{rasgo}\index{rasgo} es cualquier característica describible de
un individuo: el color de los \omterm{def:g1:the-five-senses:sense}{ojos}, el grupo sanguíneo, la forma del
lóbulo de la \omterm{def:g1:the-five-senses:sense}{oreja}, la estatura, una voz para cantar, una cicatriz. La
colección entera de rasgos de un individuo la construyen dos obras: lo
que se \emph{heredó} y lo que añadieron el \emph{ambiente y la vida}
(el mundo de la \cref{def:g4:adapted-to-their-world:adaptation},
actuando sobre un cuerpo).
\end{definition}

\begin{definition}[La herencia]\label{def:g9:heredity-and-traits:heredity}
La \emph{herencia}\index{herencia} es la transmisión de \omterm{def:g9:heredity-and-traits:traits}{rasgos} de
padres a hijos por la reproducción --- a través, como enseñó el
\cref{ch:g8:sexual-reproduction}, de dos \omterm{def:g5:first-look-at-cells:cell}{células} sueltas. Los \omterm{def:g9:heredity-and-traits:traits}{rasgos}
que pasan por ahí son \emph{hereditarios}\index{rasgo hereditario}: el
color de los \omterm{def:g1:the-five-senses:sense}{ojos} y el grupo sanguíneo lo son; las cicatrices y las
\omterm{def:g1:the-five-senses:sense}{orejas} agujereadas no, por muchos años que se lleven. La prueba que
separa son los \omterm{def:g8:sexual-reproduction:gametes}{gametos}: lo que dos \omterm{def:g5:first-look-at-cells:cell}{células} no pueden llevar, las
familias no lo pueden transmitir.
\end{definition}

\begin{example}[Clasificar un álbum de familia]\label{ex:g9:heredity-and-traits:album}
Las fotografías del abuelo: su nariz inconfundible --- hereditaria, ahí
vuelve a estar en la tía y en el primo; su \omterm{def:g1:the-five-senses:sense}{piel} curtida de campesino
--- adquirida, el nieto de oficina no la tiene; su estatura ---
en parte las dos cosas, porque el nieto le saca la cabeza con la misma
constitución, alimentado por años más ricos
(\cref{prop:g3:balanced-diet:jobs}); su destreza con el violín --- no
hereditaria como destreza (\cref{ex:g8:nervous-communication:juggler}:
las \omterm{def:g8:nervous-communication:synapse}{sinapsis} se entrenan, no se transmiten), aunque la familia de
músicos nos dice que algo más sutil actúa en lo que se pasa: aptitud,
quizá; nunca el cableado ya entrenado.
\end{example}

\begin{proposition}[Heredado y adquirido]\label{prop:g9:heredity-and-traits:sorting}
Las dos obras se separan con tres pruebas:
\begin{enumerate}
  \item el \emph{patrón familiar}: los \omterm{def:g9:heredity-and-traits:heredity}{rasgos hereditarios} reaparecen
        por el árbol siguiendo las líneas de la reproducción; los
        adquiridos siguen a las vidas (todos los marineros curtidos,
        sean quienes sean sus padres);
  \item la \emph{presencia desde el principio}: el grupo sanguíneo
        queda fijado en la \omterm{def:g5:flower-to-fruit:steps}{fecundación}; la \omterm{def:g1:the-five-senses:sense}{piel} curtida y la destreza
        se construyen después;
  \item la \emph{transmisión}: los hijos de los amputados tienen sus
        miembros; los hijos de los culturistas empiezan sin entrenar.
        Los \omterm{def:g9:heredity-and-traits:traits}{rasgos} adquiridos mueren con quien los adquirió --- los
        \omterm{def:g8:sexual-reproduction:gametes}{gametos} no los llevan.
\end{enumerate}
La mayoría de los \omterm{def:g9:heredity-and-traits:traits}{rasgos} visibles --- la estatura, la constitución,
incluso el carácter --- están tejidos con las dos obras: un margen
heredado y un relleno vivido.
\end{proposition}
\admitted

\section{Leer los patrones familiares}

\begin{example}[Rasgos que se saltan una generación]\label{ex:g9:heredity-and-traits:skip}
Algunos patrones desconciertan a primera vista: dos padres de \omterm{def:g1:the-five-senses:sense}{ojos}
marrones con un hijo de \omterm{def:g1:the-five-senses:sense}{ojos} azules; un \omterm{def:g9:heredity-and-traits:traits}{rasgo} de la abuela ausente en
todos sus hijos y de vuelta en un nieto. Saltarse una generación es
corriente, y trae una conclusión pesada: un progenitor puede
\emph{transmitir lo que no muestra}. Las instrucciones del \omterm{def:g9:heredity-and-traits:traits}{rasgo}
viajaron escondidas por una generación --- así que lo que se hereda no
es el \omterm{def:g9:heredity-and-traits:traits}{rasgo} mismo, sino algo que lo \emph{determina}, que se puede
llevar sin verse.
\end{example}

\begin{proposition}[Lo que nos obligan a concluir los patrones]\label{prop:g9:heredity-and-traits:conclusions}
Tres conclusiones, cada una obligada por una observación:
\begin{enumerate}
  \item como los \omterm{def:g9:heredity-and-traits:traits}{rasgos} adquiridos no pasan, lo que pasa tienen que ser
        \emph{determinantes} --- instrucciones para construir \omterm{def:g9:heredity-and-traits:traits}{rasgos},
        no los \omterm{def:g9:heredity-and-traits:traits}{rasgos} mismos;
  \item como los \omterm{def:g9:heredity-and-traits:traits}{rasgos} se saltan generaciones, los determinantes se
        pueden \emph{llevar sin mostrarlos}: cada individuo lleva más
        de lo que enseña;
  \item como cada hijo recibe de los dos progenitores por medio de dos
        \omterm{def:g8:sexual-reproduction:gametes}{gametos} (\cref{prop:g8:sexual-reproduction:core}), cada
        progenitor aporta una \emph{selección} de sus determinantes ---
        y las selecciones difieren de un hijo a otro, o los hermanos
        saldrían iguales (el barajado de la
        \cref{prop:g8:sexual-reproduction:shuffle}, ahora situada: es
        un barajado \emph{de determinantes}).
\end{enumerate}
\end{proposition}
\admitted

\begin{remark}[La pregunta que esto plantea]\label{rem:g9:heredity-and-traits:question}
El álbum de familia nos ha acorralado en una pregunta maravillosa. En
algún sitio del casi nada de un \omterm{def:g8:reproductive-systems:male}{espermatozoide}
(\cref{ex:g8:reproductive-systems:sperm} --- un \omterm{def:g6:cell-unit-of-life:parts}{núcleo} apretado y una
cola) viajan las instrucciones de una nariz, de un grupo sanguíneo, de
un color de \omterm{def:g1:the-five-senses:sense}{ojos} --- llevables sin mostrarse, seleccionables por
mitades, legibles por el cuerpo que crece. \emph{¿Dónde están y de qué
están hechas?} La anatomía del \omterm{def:g8:reproductive-systems:male}{espermatozoide} ya apunta la respuesta:
todo lo que manda el padre cabe en un \omterm{def:g6:cell-unit-of-life:parts}{núcleo} (la
\cref{rem:g5:first-look-at-cells:nucleus} prometía que ese sitio
importaba). Los dos capítulos siguientes lo abren.
\end{remark}

\begin{method}[Auditar un rasgo]\label{met:g9:heredity-and-traits:audit}
Para cualquier \omterm{def:g9:heredity-and-traits:traits}{rasgo}, pasa las tres pruebas:
\begin{enumerate}
  \item sitúalo en el árbol familiar: ¿sigue las líneas de la
        reproducción, o las vidas y las costumbres?
  \item féchalo: ¿fijado desde el principio, o construido por el
        camino?
  \item aplica la prueba de los \omterm{def:g8:sexual-reproduction:gametes}{gametos}: ¿podrían llevarlo dos \omterm{def:g5:first-look-at-cells:cell}{células}?
        (una cicatriz no; un grupo sanguíneo sí);
  \item veredicto: heredado, adquirido o --- lo más corriente --- un
        margen heredado, vivido hasta el fondo.
\end{enumerate}
\end{method}

\section{Problema: El genealogista del pueblo}

\begin{problem}[{Problema de fin de semana --- tres generaciones,
un libro de rasgos}]\label{pb:g9:heredity-and-traits:1}
Un genealogista de pueblo anota los árboles de familia de la parroquia
con sus \omterm{def:g9:heredity-and-traits:traits}{rasgos}: la ``barbilla de la capilla'' (una barbilla
inconfundible que reaparece desde hace un siglo), ser zurdo, la tos de
harina de los panaderos, los grupos sanguíneos del registro de
donantes y las caras curtidas de los pastores.

\medskip\noindent\textbf{Parte I --- Clasificar el
libro.}
\begin{enumerate}
  \item Clasifica los cinco \omterm{def:g9:heredity-and-traits:traits}{rasgos} con el
        \cref{met:g9:heredity-and-traits:audit}, dando la prueba que
        decide en cada uno.
  \item La tos sigue a la panadería, no a la familia: casarse con
        alguien de la casa la trae, mudarse lejos la quita. ¿Qué prueba
        es esta, en acción?
  \item La barbilla de la capilla se salta la generación de en medio en
        dos \omterm{def:g1:plants-around-us:tree}{ramas}. Saca la conclusión que eso obliga.
  \item Los grupos sanguíneos sorprenden a veces al registro: un hijo
        del grupo O de unos padres del grupo A y del grupo B. ¿Qué
        tenía que llevar cada progenitor?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- La lectura en
determinantes.}
\begin{enumerate}[resume]
  \item Reescribe el siglo de la barbilla en lenguaje de
        determinantes: ¿qué viajó de verdad cuatro generaciones abajo,
        y en qué vehículos
        (\cref{def:g9:heredity-and-traits:heredity})?
  \item ¿Por qué difieren en todo lo demás los primos que llevan la
        barbilla? Sitúa el barajado.
  \item El genealogista no encuentra ni un caso --- en un siglo --- de
        una cicatriz, un oficio o una lengua que pase a un recién
        nacido. Enuncia la generalización y su razón a nivel de
        mecanismo.
  \item Los gemelos de una familia coinciden hasta en la barbilla y más
        allá. ¿Qué concluye el libro sobre su comienzo
        (\cref{rem:g5:human-reproduction:twins})?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- La conferencia.}
\begin{enumerate}[resume]
  \item El genealogista da una conferencia: ``la parroquia transmite
        dos \omterm{def:g9:heredity-and-traits:heredity}{herencias} --- una por los \omterm{def:g8:sexual-reproduction:gametes}{gametos} y otra por vivir
        juntos''. Asigna a los dos canales los cinco \omterm{def:g9:heredity-and-traits:traits}{rasgos} del libro.
  \item ``Cada uno de nosotros enseña menos de lo que lleva''.
        Defiéndelo a partir del libro, dos veces.
  \item Un oyente pregunta qué son físicamente las instrucciones del
        canal de los \omterm{def:g8:sexual-reproduction:gametes}{gametos}. Da la respuesta honrada de la
        \cref{rem:g9:heredity-and-traits:question} --- sitio conocido,
        contenido en el capítulo siguiente.
  \item Cierra la conferencia con el veredicto de una línea de la
        auditoría sobre el caso más corriente: la mayoría de los \omterm{def:g9:heredity-and-traits:traits}{rasgos}
        son \emph{qué}, vividos \emph{cómo}?
\end{enumerate}
\end{problem}
