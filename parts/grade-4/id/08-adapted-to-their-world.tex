\chapter{Menyesuaikan Diri dengan Dunianya}\label{ch:g4:adapted-to-their-world}

Taruhlah seekor beruang kutub di Sahara dan seekor unta di atas es
apung, dan tidak satu pun akan bertahan satu musim. Namun
masing-masing, ketika di rumahnya, adalah juara: beruangnya berenang
di laut berbongkah es dengan nyaman, untanya melintasi gurun yang akan
menghabisi seekor kuda dalam beberapa hari. \omterm{def:g1:living-or-not:living}{Makhluk hidup} cocok dengan
dunianya dengan ketepatan yang layak dipelajari dari dekat --- ciri
demi ciri.

\section{Apa arti adaptasi}

\begin{definition}[Adaptasi]\label{def:g4:adapted-to-their-world:adaptation}
Suatu \emph{adaptasi}\index{adaptasi} adalah ciri tubuh atau perilaku
sebuah \omterm{def:g1:living-or-not:living}{makhluk hidup} yang mencocokkannya dengan keadaan \omterm{def:g2:caring-for-nature:environment}{lingkungannya}
--- dengan dinginnya atau keringnya, dengan makanannya, dengan
pemburunya. \Cref{prop:g2:where-animals-live:fit} memperlihatkan \omterm{def:g1:animals-around-us:animal}{hewan}
yang cocok dengan \omterm{def:g2:where-animals-live:habitat}{habitatnya}; adaptasi adalah kecocokan itu, yang
diperiksa kepingan demi kepingan.
\end{definition}

\begin{example}[Unta, kepingan demi kepingan]\label{ex:g4:adapted-to-their-world:camel}
Masalah gurunnya: panas, tanpa air, pasir yang lembut. Jawaban untanya:
punuk berisi lemak --- \omterm{def:g3:animals-through-seasons:active}{cadangan makanan} bagi lintasan yang kurus;
kemampuan meminum seratus liter dalam hitungan menit lalu bertahan
berhari-hari tanpa minum; kaki lebar berjari dua yang tidak amblas di
pasir; lubang hidung yang dapat menutup terhadap debu yang beterbangan
dan \omterm{def:g1:animals-around-us:covering}{bulu} \omterm{def:g1:the-five-senses:sense}{mata} rangkap yang panjang untuk menemaninya. Setiap masalah
gurun, dijawab oleh sebuah ciri.
\end{example}

\begin{example}[Beruang kutub, kepingan demi kepingan]\label{ex:g4:adapted-to-their-world:bear}
Masalah es apungnya: dingin, air yang membekukan, mangsa yang sukar
didekati di atas es yang putih. Jawaban beruangnya: \omterm{def:g1:animals-around-us:covering}{bulu} yang tebal di
atas lapisan lemak yang dalam --- penyekat rangkap; telapak kaki yang
lebar dan sedikit berselaput --- sepatu salju dan dayung sekaligus;
mantel putih --- tak terlihat di atas salju tempatnya berburu.
Ketepatan yang sama, disetel untuk dunia yang berlawanan.
\end{example}

\begin{proposition}[Adaptasi menjawab masalah]\label{prop:g4:adapted-to-their-world:answers}
Di mana pun \omterm{def:g2:caring-for-nature:environment}{lingkungan} mengajukan masalah yang sulit, \omterm{def:g1:living-or-not:living}{makhluk hidup}
yang berada di rumahnya di situ membawa ciri yang menjawabnya:
\begin{itemize}
  \item dingin: \omterm{def:g1:animals-around-us:covering}{bulu} yang tebal, lapisan lemak, \omterm{def:g1:the-five-senses:sense}{telinga} kecil yang
        hanya sedikit kehilangan kehangatan;
  \item kekeringan: penyimpanan air, \omterm{def:g1:the-five-senses:sense}{kulit} berlilin, \omterm{def:g1:plants-around-us:parts}{akar} yang dalam;
  \item hidup di air: tubuh yang ramping, sirip dan kaki berselaput,
        mantel yang menolak air;
  \item pemburu di sekeliling: warna penyamaran, baju pelindung, duri,
        kecepatan, dan \omterm{def:g1:the-five-senses:sense}{mata} yang berjauhan letaknya untuk mengawasi ke
        segala arah;
  \item berburu untuk hidup: indra yang tajam, \omterm{def:g1:the-five-senses:sense}{mata} yang menghadap ke
        depan, cakar dan kecepatan
        (\cref{prop:g2:what-animals-eat:match}).
\end{itemize}
\end{proposition}
\admitted

\section{Tumbuhan pun beradaptasi}

\begin{example}[Kaktus]\label{ex:g4:adapted-to-their-world:cactus}
Gurun menguji \omterm{def:g1:plants-around-us:plant}{tumbuhan} paling keras --- \omterm{def:g1:plants-around-us:plant}{tumbuhan} tidak dapat berjalan
menuju air. Kaktus menjawabnya: \omterm{def:g1:plants-around-us:parts}{batang} tebal berbentuk tong yang
menyimpan air dan mengerjakan tugas penangkapan cahaya milik \omterm{def:g1:plants-around-us:parts}{daun};
\omterm{def:g1:plants-around-us:parts}{daun} yang menyusut menjadi duri, yang nyaris tidak kehilangan air ---
dan sekaligus membela tongnya terhadap \omterm{def:g1:animals-around-us:animal}{hewan} yang kehausan; \omterm{def:g1:plants-around-us:parts}{akar} yang
menyebar lebar dan dangkal untuk menangkap seluruh hujan yang jarang
itu sekaligus.
\end{example}

\begin{example}[Bantalan pegunungan]\label{ex:g4:adapted-to-their-world:alpine}
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} pegunungan tinggi menghadapi angin yang membekukan. Banyak di
antaranya tumbuh sebagai bantalan yang rapat dan merapat ke tanah: di
dalam bantalannya, anginnya terpatahkan dan udaranya tetap lebih
lunak --- sebuah bentuk yang sekaligus tempat berlindung. Sepupu dataran
rendah \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang sama tumbuh tinggi dan renggang; bentuk
pegunungannya memeluk batunya.
\end{example}

\begin{method}[Membaca dunia sebuah organisme dari tubuhnya]\label{met:g4:adapted-to-their-world:read}
Diberikan seekor \omterm{def:g1:animals-around-us:animal}{hewan} atau sebuah \omterm{def:g1:plants-around-us:plant}{tumbuhan} yang belum dikenal:
\begin{enumerate}
  \item daftarkan ciri-cirinya yang mencolok --- mantel, kaki, \omterm{def:g1:the-five-senses:sense}{mata},
        bentuk tubuh, \omterm{def:g1:plants-around-us:parts}{daun}, \omterm{def:g1:plants-around-us:parts}{akar};
  \item untuk masing-masing, tanyakan: \emph{masalah apa yang akan
        dipecahkan ciri ini?}
  \item rangkailah jawabannya menjadi gambaran rumahnya --- dingin
        atau panas, basah atau kering, pemburu atau yang diburu;
  \item periksa terkaanmu terhadap tempat organisme itu sungguh-sungguh
        hidup.
\end{enumerate}
Inilah \cref{met:g2:where-animals-live:matching} yang sudah dewasa:
dari memasangkan sebuah \omterm{def:g2:where-animals-live:habitat}{habitat} menjadi menyusun ulang sebuah dunia.
\end{method}

\begin{example}[Metodenya atas sebuah kulit misteri]\label{ex:g4:adapted-to-their-world:mystery}
Sebuah laci museum menyimpan \omterm{def:g1:the-five-senses:sense}{kulit} tanpa label: \omterm{def:g1:animals-around-us:covering}{bulu} lembut yang
lebat, \omterm{def:g1:the-five-senses:sense}{mata} yang mungil, cakar depan raksasa berbentuk sekop. \omterm{def:g1:animals-around-us:covering}{Bulunya}
berkata \omterm{prop:g3:sorting-living-things:groups}{mamalia}; sekopnya berkata menggali; \omterm{def:g1:the-five-senses:sense}{matanya} yang tak berguna
berkata hidup dalam gelap. Putusannya: seorang penggali bawah tanah
--- dan memang benar ia seekor tikus tanah
(\cref{ex:g2:where-animals-live:mole}), yang dibaca dengan benar hanya
dari tiga ciri.
\end{example}

\begin{remark}[Bagaimana kecocokannya terjadi?]\label{rem:g4:adapted-to-their-world:how}
Seekor unta tidak pernah \emph{memutuskan} untuk menumbuhkan kaki yang
lebar, dan tidak ada beruang kutub yang memilih mantel putihnya. Lalu
bagaimana \omterm{def:g1:living-or-not:living}{makhluk hidup} sampai begitu cocok dengan dunianya? Itulah
salah satu pertanyaan terdalam \omterm{def:g1:living-or-not:living}{biologi}, dan jawaban lengkapnya ---
bagaimana kecocokannya terbangun, generasi demi generasi --- adalah
kisah yang disimpan buku ini untuk tahun berikutnya. Untuk sekarang,
belajarlah \emph{melihat} kecocokannya: penglihatan itulah yang membuat
pertanyaannya dapat diajukan.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:1}
Apa itu \omterm{def:g4:adapted-to-their-world:adaptation}{adaptasi}? Berikan satu contoh dari \omterm{def:g1:animals-around-us:animal}{hewan} dan satu contoh dari
\omterm{def:g1:plants-around-us:plant}{tumbuhan}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:2}
Sebutkan tiga masalah gurun dan jawaban unta atas masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:3}
Sebutkan tiga masalah es apung dan jawaban beruang kutub atas
masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:4}
Tugas apa yang dikerjakan duri kaktus? Apa yang disimpan \omterm{def:g1:plants-around-us:parts}{batang}
tongnya, dan pekerjaan apa yang sudah diambil alih \omterm{def:g1:plants-around-us:parts}{batangnya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:5}
Mengapa banyak \omterm{def:g1:plants-around-us:plant}{tumbuhan} pegunungan tinggi tumbuh sebagai bantalan yang
rapat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:6}
Berikan dua \omterm{def:g4:adapted-to-their-world:adaptation}{adaptasi} \omterm{def:g1:animals-around-us:animal}{hewan} yang diburu dan dua \omterm{def:g4:adapted-to-their-world:adaptation}{adaptasi} pemburu, dari
\cref{prop:g4:adapted-to-their-world:answers}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:7}
Ceritakan kembali \cref{ex:g4:adapted-to-their-world:mystery}: tiga
ciri mana yang membongkar tikus tanahnya, dan apa yang dikatakan
masing-masing?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:8}
Cobalah: pilih \omterm{def:g1:animals-around-us:animal}{hewan} apa pun yang dapat kamu amati --- seekor bebek,
seekor kucing, seekor laba-laba --- lalu jalankan
\cref{met:g4:adapted-to-their-world:read} atasnya: tiga ciri, tiga
masalah yang terpecahkan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:9}
Rubah gurun bertelinga sangat besar; rubah kutub bertelinga mungil.
Dengan memakai baris tentang dingin pada
\cref{prop:g4:adapted-to-their-world:answers}, jelaskan keduanya ---
\omterm{def:g1:the-five-senses:sense}{telinga} yang besar pasti berbuat apa terhadap kehangatan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:10}
Seekor \omterm{def:g1:animals-around-us:animal}{hewan} misteri: tubuh yang ramping, kaki belakang berselaput,
\omterm{def:g1:animals-around-us:covering}{bulu} lebat yang menolak air, kumis, dan ia pandai menahan napas.
Susunlah ulang dunianya dengan
\cref{met:g4:adapted-to-their-world:read}. (Ia membangun bendungan.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:adapted-to-their-world:11}
Paus (\cref{ex:g3:sorting-living-things:surprises}) adalah \omterm{prop:g3:sorting-living-things:groups}{mamalia}
yang berbentuk seperti \omterm{prop:g3:sorting-living-things:groups}{ikan}: ramping, bersirip, licin. \omterm{prop:g3:sorting-living-things:groups}{Ikan} trout
adalah \omterm{prop:g3:sorting-living-things:groups}{ikan} yang berbentuk sama. Apa yang ditunjukkan kemiripan antara
dua perenang yang \emph{tidak berkerabat} ini tentang \omterm{def:g4:adapted-to-their-world:adaptation}{adaptasi} ---
siapa yang menetapkan bentuknya, keluarganya atau airnya?
\end{exercise}
