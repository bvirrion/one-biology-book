\chapter{Imunitas Adaptif dan Vaksinasi}\label{ch:g12:adaptive-immunity}

Pada 1796 seorang dokter desa menggoreskan bahan dari lepuh cacar sapi
milik seorang pemerah susu ke lengan seorang anak lelaki, menunggu enam
pekan, lalu menginokulasinya dengan cacar. Anak itu tidak jatuh sakit.
Tak ada apa pun dalam dunia sang dokter yang menjelaskan mengapa;
tubuhnya, rupanya, telah diajari. Dua abad kemudian cacar telah punah,
dan pengajaran itu dipahami sampai ke molekulnya: seperangkat \omterm{def:g10:cells-common-unit:cell}{sel} yang
bersama-sama membawa reseptor bagi hampir setiap molekul yang mungkin
pernah ada, yang berlipat ketika molekulnya muncul, dan yang
mengingat. Bab ini menguraikan sistem itu, kedua lengannya, ingatannya,
\omterm{def:g12:adaptive-immunity:vaccine}{vaksin} yang dibangun di atasnya --- dan virus yang menghancurkannya.

\section{Antigen dan limfosit}

\begin{definition}[Antigen, limfosit]\label{def:g12:adaptive-immunity:antigen}
Sebuah \emph{antigen}\index{antigen} adalah molekul apa pun yang dapat
dikenali \omterm{def:g12:innate-immunity:immunity}{sistem imun} adaptif: biasanya \omterm{def:g11:gene-expression:protein}{protein} atau gula sebuah
mikroba, sebuah racun, permukaan \omterm{def:g10:cells-common-unit:cell}{sel} asing --- atau, karena keliru,
salah satu molekul tubuh sendiri. \omterm{def:g10:cells-common-unit:cell}{Sel} yang mengenalinya adalah
\emph{limfosit}\index{limfosit}, yaitu \omterm{def:g10:cells-common-unit:cell}{sel} darah putih yang dibuat di
sumsum tulang dan berkumpul di kelenjar getah bening, limpa serta
jaringan limfoid usus. Setiap limfosit membawa ribuan salinan satu
\emph{reseptor}, dan setiap reseptor hanya mengikat satu antigen.
\emph{Limfosit B} mematang di dalam sumsum dan akan membuat \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi};
\emph{limfosit T} mematang di dalam timus dan bekerja lewat sentuhan
serta lewat isyarat.
\end{definition}

\begin{proposition}[Sebuah khazanah yang dibuat sejak awal]\label{prop:g12:adaptive-immunity:repertoire}
Reseptor tiap \omterm{def:g12:adaptive-immunity:antigen}{limfosit} dirakit, selagi \omterm{def:g10:cells-common-unit:cell}{selnya} mematang, oleh penyusunan
ulang ruas \omterm{def:g10:universal-dna:gene}{gen} secara acak, yang berbeda pada setiap \omterm{def:g10:cells-common-unit:cell}{sel}. Jadi tubuh
memuat sekitar $10^{8}$ reseptor yang berbeda bahkan sebelum pernah
bertemu \omterm{def:g12:adaptive-immunity:antigen}{antigen} --- cukup banyak sehingga hampir setiap molekul
menemukan beberapa \omterm{def:g10:cells-common-unit:cell}{sel} yang mengikatnya. \omterm{def:g12:adaptive-immunity:antigen}{Limfosit} yang reseptornya
mengikat molekul tubuh sendiri disingkirkan selama pematangan; sisanya
menunggu, beberapa \omterm{def:g10:cells-common-unit:cell}{sel} bagi tiap \omterm{def:g12:adaptive-immunity:antigen}{antigen} yang mungkin, sampai
\omterm{def:g12:adaptive-immunity:antigen}{antigennya} tiba.
\end{proposition}
\admitted

\section{Seleksi klonal}

\begin{proposition}[Seleksi dan pengembangan]\label{prop:g12:adaptive-immunity:selection}
Ketika sebuah \omterm{def:g12:adaptive-immunity:antigen}{antigen} masuk, hanya \omterm{def:g12:adaptive-immunity:antigen}{limfosit} yang reseptornya cocok
dengannya yang menanggapi: \omterm{def:g12:adaptive-immunity:antigen}{limfosit} itu \emph{terpilih}, lalu berlipat
lewat \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} menjadi sebuah klon berisi ribuan \omterm{def:g10:cells-common-unit:cell}{sel} identik sepanjang
beberapa hari --- \emph{seleksi klonal}\index{seleksi klonal}. Klon itu
lalu berdiferensiasi: sebagian besar \omterm{def:g10:cells-common-unit:cell}{selnya} menjadi \emph{\omterm{def:g10:cells-common-unit:cell}{sel} efektor}
yang memerangi \omterm{def:g12:adaptive-immunity:antigen}{antigennya} selama satu-dua pekan lalu mati; sebagian
kecil menjadi \emph{sel ingatan}\index{sel ingatan} yang berumur
panjang, yang bertahan bertahun-tahun dan, bila \omterm{def:g12:adaptive-immunity:antigen}{antigennya} kembali,
menanggapi lebih cepat dan dalam jumlah yang jauh lebih besar.
Tanggapannya khas karena hanya klon yang cocok yang berkembang;
tanggapannya lambat pada kali pertama karena ia mulai dari segelintir
\omterm{def:g10:cells-common-unit:cell}{sel}; dan cepat pada kali kedua karena \omterm{prop:g12:adaptive-immunity:selection}{sel ingatannya} sudah banyak.
\end{proposition}
\begin{proof}[Bukti percobaan]
\omterm{prop:g12:adaptive-immunity:antibodies}{Antibodi} terhadap suatu \omterm{def:g12:adaptive-immunity:antigen}{antigen} muncul di dalam darah sekitar sepekan
setelah suntikan pertama, naik selama dua pekan lalu menurun; suntikan
kedua beberapa bulan kemudian menghasilkan, dalam dua atau tiga hari,
\omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} sepuluh sampai seratus kali puncak yang pertama, yang bertahan
bertahun-tahun. Menyuntikkan \omterm{def:g12:adaptive-immunity:antigen}{antigen} kedua yang tak berkaitan
bersamaan dengan dosis kedua itu memberi, bagi \omterm{def:g12:adaptive-immunity:antigen}{antigen} tersebut,
tanggapan lambat bertipe pertama: ingatannya bersifat khas. \omterm{def:g12:adaptive-immunity:antigen}{Limfosit}
yang diambil dari hewan yang diimunisasi lalu dipindahkan ke hewan yang
belum pernah terpapar akan memindahkan ingatannya; \omterm{def:g12:adaptive-immunity:antigen}{limfosit} dari hewan
yang tak diimunisasi tidak.
\end{proof}

\section{Dua lengan: antibodi dan sel pembunuh}

\begin{proposition}[Limfosit B dan antibodi]\label{prop:g12:adaptive-immunity:antibodies}
Klon B yang terpilih berdiferensiasi menjadi \emph{\omterm{def:g10:cells-common-unit:cell}{sel} plasma}, yaitu
pabrik yang menyekresikan ribuan \emph{antibodi}\index{antibodi} tiap
detik: salinan larut dari reseptor klon itu, yaitu \omterm{def:g11:gene-expression:protein}{protein} berbentuk Y
yang kedua lengannya masing-masing mengikat \omterm{def:g12:adaptive-immunity:antigen}{antigennya}. \omterm{prop:g12:adaptive-immunity:antibodies}{Antibodi} tidak
membunuh; ia \emph{menandai}. Bila terikat pada virus atau racun,
\omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} menghalanginya masuk ke \omterm{def:g10:cells-common-unit:cell}{sel} atau bekerja; bila terikat pada
bakteri, \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} menyelubunginya, dan selubung itulah yang paling
mudah dicengkeram \omterm{def:g12:innate-immunity:phagocytosis}{fagosit} \cref{ch:g12:innate-immunity}. \omterm{def:g12:adaptive-immunity:antigen}{Antigen} yang
disilangikat banyak \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} menggumpal menjadi \emph{kompleks imun},
yang dibersihkan lewat \omterm{def:g12:innate-immunity:phagocytosis}{fagositosis}. Inilah lengan yang melawan penyusup
di dalam darah dan cairan tubuh.
\end{proposition}
\admitted

\begin{proposition}[Limfosit T]\label{prop:g12:adaptive-immunity:tcells}
\omterm{def:g12:adaptive-immunity:antigen}{Limfosit} T mengenali \omterm{def:g12:adaptive-immunity:antigen}{antigen} hanya sebagai serpihan yang dipajang di
permukaan \omterm{def:g10:cells-common-unit:cell}{sel} lain (\cref{ch:g12:innate-immunity}). Ada dua macam:
\begin{itemize}
  \item \emph{Sel T penolong}\index{sel T penolong} diaktifkan oleh \omterm{prop:g12:innate-immunity:handover}{sel
        dendritik} yang membawa \omterm{def:g12:adaptive-immunity:antigen}{antigennya} ke kelenjar getah bening;
        setelah terpilih dan berlipat, \omterm{def:g10:cells-common-unit:cell}{sel} itu menyekresikan sitokina
        yang mengizinkan dan memperkuat tanggapan \omterm{def:g10:cells-common-unit:cell}{sel} B serta \omterm{def:g10:cells-common-unit:cell}{sel} T
        lainnya. \omterm{def:g10:cells-common-unit:cell}{Sel} itulah pengoordinasinya; tanpanya, nyaris tak ada
        tanggapan adaptif yang terjadi.
  \item \emph{\omterm{def:g10:cells-common-unit:cell}{Sel} T sitotoksik} mengenali serpihan sebuah virus (atau
        \omterm{def:g11:gene-expression:protein}{protein} sebuah \omterm{def:g11:cancer:cancer}{tumor}) yang dipajang \omterm{def:g10:cells-common-unit:cell}{sel} yang terinfeksi, lalu
        membunuh \omterm{def:g10:cells-common-unit:cell}{sel} itu lewat sentuhan, sebelum virus di dalamnya
        sempat berlipat. Inilah lengan yang melawan penyusup yang
        bersembunyi di dalam \omterm{def:g10:cells-common-unit:cell}{sel}, tempat \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} tak dapat mencapainya.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Sebuah virus, ditemui dua kali]\label{ex:g12:adaptive-immunity:virus}
Influenza pertama: virusnya berlipat selama tiga hari sebelum tanggapan
adaptifnya siap; pada hari ke-7 \omterm{def:g10:cells-common-unit:cell}{sel} T sitotoksik membunuh \omterm{def:g10:cells-common-unit:cell}{sel} yang
terinfeksi di saluran napas dan \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} menetralkan virus yang bebas;
pada hari ke-10 orangnya pulih, dengan \omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} dari kedua macam.
Galur yang sama setahun kemudian: \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} yang sudah ada di dalam
darah menghadang sebagian besar virusnya di pintu masuk, \omterm{prop:g12:adaptive-immunity:selection}{sel ingatannya}
berkembang dalam dua hari, dan jangkitannya berakhir sebelum
disadari. Galur berbeda yang \omterm{def:g11:gene-expression:protein}{protein} permukaannya bermutasi lolos dari
\omterm{prop:g12:adaptive-immunity:antibodies}{antibodinya} --- itulah sebabnya \omterm{def:g12:adaptive-immunity:vaccine}{vaksinnya} dibuat ulang setiap tahun.
\end{example}

\section{Vaksinasi}

\begin{definition}[Vaksin]\label{def:g12:adaptive-immunity:vaccine}
Sebuah \emph{vaksin}\index{vaksin} adalah sediaan yang memancing
tanggapan adaptif primer, dan karenanya ingatan, terhadap sebuah patogen
tanpa menimbulkan penyakitnya: patogennya dilemahkan (campak,
tuberkulosis), dimatikan (polio suntikan), diringkas menjadi salah satu
\omterm{def:g11:gene-expression:protein}{proteinnya} atau racun yang dinonaktifkan (tetanus, hepatitis B), atau,
paling belakangan ini, menjadi petunjuk bagi \omterm{def:g10:cells-common-unit:cell}{sel} untuk membuat \omterm{def:g11:gene-expression:protein}{protein}
itu. Sebuah \emph{adjuvan} yang ditambahkan pada sediaannya memancing
peradangan kecil yang tanpanya \omterm{prop:g12:innate-immunity:handover}{sel dendritik} tak akan membawa
\omterm{def:g12:adaptive-immunity:antigen}{antigennya} ke kelenjar getah bening. Dosis penguat mengubah tanggapan
primer menjadi tanggapan sekunder dan memperpanjang ingatannya sampai
berpuluh tahun.
\end{definition}

\begin{proposition}[Melindungi kelompoknya]\label{prop:g12:adaptive-immunity:herd}
Orang yang divaksinasi terlindungi; \omterm{def:g12:selection-drift-speciation:population}{populasi} yang divaksinasi juga
melindungi anggotanya yang belum divaksinasi. Orang yang terjangkit
menularkan penyakitnya, rata-rata, kepada $R_0$ orang lain di dalam
\omterm{def:g12:selection-drift-speciation:population}{populasi} tanpa imunitas (sekitar 15 untuk campak, 2 sampai 3 untuk
influenza); bila sebagian sebesar $v$ sudah kebal, hanya
$R_0(1 - v)$ kontaknya yang dapat terjangkit, dan penyakitnya surut
ketika angka itu jatuh di bawah 1 --- yaitu ketika
\[
v > 1 - \frac{1}{R_0}.
\]
Untuk campak, 93\% \omterm{def:g12:selection-drift-speciation:population}{populasinya} harus kebal agar bayi yang terlalu muda
untuk divaksinasi dan orang yang tak dapat menerimanya ikut
terlindungi; untuk influenza, sekitar 60\%. \emph{Imunitas
kelompok}\index{imunitas kelompok} inilah sebabnya vaksinasi merupakan
tindakan bersama.
\end{proposition}
\admitted

\section{Ketika pengoordinasinya dihancurkan}

\begin{proposition}[HIV dan AIDS]\label{prop:g12:adaptive-immunity:hiv}
Virus imunodefisiensi manusia, HIV, menjangkiti \omterm{prop:g12:adaptive-immunity:tcells}{sel T penolong}, dan
masuk justru lewat reseptor yang menandai \omterm{def:g10:cells-common-unit:cell}{sel} itu, lalu berlipat di
dalamnya. Bertahun-tahun \omterm{def:g12:innate-immunity:immunity}{sistem imunnya} membunuh \omterm{def:g10:cells-common-unit:cell}{sel} yang terinfeksi
dan membuat \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} --- orangnya \emph{seropositif}, dan menular ---
sementara virusnya, yang terus bermutasi, lolos dari tiap tanggapan
dan perlahan-lahan menguras \omterm{def:g10:cells-common-unit:cell}{sel} penolongnya. Ketika cacahnya jatuh di
bawah kira-kira seperlima taraf normal, tanggapan adaptifnya tak lagi
dapat dikoordinasikan: \emph{AIDS}\index{AIDS}, saat jangkitan yang
dibersihkan tubuh sehat tanpa disadari --- sejenis jamur mulut, seekor
parasit paru yang ringan --- menjadi mematikan, dan \omterm{def:g11:cancer:cancer}{kanker} yang langka
pun muncul. Obat antivirus menghambat \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} virusnya, memulihkan \omterm{def:g10:cells-common-unit:cell}{sel}
penolongnya dan, bila diminum seumur hidup, menahan penyakitnya serta
menghentikan penularannya; belum ada \omterm{def:g12:adaptive-immunity:vaccine}{vaksinnya}, dan belum ada obat
yang menyembuhkannya.
\end{proposition}
\admitted

\begin{method}[Membaca sebuah tanggapan imun]\label{met:g12:adaptive-immunity:read}
\begin{enumerate}
  \item Kenali \omterm{def:g12:adaptive-immunity:antigen}{antigennya} dan di mana ia berada: bebas di dalam cairan
        (\omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} yang bekerja) atau di dalam \omterm{def:g10:cells-common-unit:cell}{sel} (\omterm{def:g10:cells-common-unit:cell}{sel} T sitotoksik
        yang bekerja).
  \item Bacalah waktunya: sepekan menuju puncak berarti tanggapan
        primer, beberapa hari berarti ingatan.
  \item Bacalah kekhasannya: tanggapan terhadap satu \omterm{def:g12:adaptive-immunity:antigen}{antigen}
        membiarkan tanggapan terhadap \omterm{def:g12:adaptive-immunity:antigen}{antigen} lain tak tersentuh.
  \item Carilah \omterm{prop:g12:adaptive-immunity:tcells}{sel T penolongnya}: tanpanya tak ada apa pun yang
        terjadi, dan itulah yang diperagakan \omterm{prop:g12:adaptive-immunity:hiv}{AIDS}.
  \item Untuk sebuah \omterm{def:g12:adaptive-immunity:vaccine}{vaksin}, sebutkan bentuk \omterm{def:g12:adaptive-immunity:antigen}{antigennya}, adjuvannya,
        jumlah dosisnya, dan bagian \omterm{def:g12:selection-drift-speciation:population}{populasi} yang diperlukan.
\end{enumerate}
\end{method}

\begin{remark}[Seleksi, sekali lagi]\label{rem:g12:adaptive-immunity:selection}
\omterm{def:g12:innate-immunity:immunity}{Sistem imun} adaptif bersifat Darwinian: sebuah \omterm{def:g12:selection-drift-speciation:population}{populasi} \omterm{def:g10:cells-common-unit:cell}{sel} dengan
reseptor acak, yang darinya lingkungan --- \omterm{def:g12:adaptive-immunity:antigen}{antigennya} --- memilih
beberapa untuk berlipat, dan yang keturunannya mewarisi reseptor
pemenangnya. Sistem itu menjalankan proses
\cref{ch:g12:selection-drift-speciation} di dalam satu tubuh dalam
sepekan, dan, lewat ingatannya, menyimpan hasilnya seumur hidup. Itu
pula sebabnya \omterm{def:g12:innate-immunity:immunity}{sistem imun} dapat diajari, yaitu apa yang dikerjakan
sebuah \omterm{def:g12:adaptive-immunity:vaccine}{vaksin}, dan sebabnya virus yang bermutasi lebih cepat daripada
kesanggupan sistemnya memilih --- HIV, influenza --- begitu sukar
dikalahkan.
\end{remark}

\section{Soal: Sembilan Puluh Tiga Persen}

\begin{problem}[{Soal akhir pekan --- sebuah kampanye vaksinasi
dihitung: tanggapan terhadap dua dosis, hitungan imunitas kelompok,
sebuah wabah campak dimodelkan, dan virus yang membongkar sistemnya}]
\label{pb:g12:adaptive-immunity:1}
Campak punya $R_0 = 15$. Sebuah kota berpenduduk 50\,000 orang punya
90\% penduduk yang kebal karena vaksinasi atau jangkitan masa lalu;
1500 di antaranya bayi di bawah satu tahun, terlalu muda untuk \omterm{def:g12:adaptive-immunity:vaccine}{vaksin}.

\medskip
\noindent\textbf{Bagian I --- Tanggapan seorang anak.}
Seorang anak menerima dosis pertama pada usia 12 bulan dan dosis kedua
pada usia 18 bulan.
\begin{enumerate}
  \item Uraikan kurva \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} setelah dosis pertama: tundaannya,
        puncaknya, penurunannya.
  \item Uraikan kurva itu setelah dosis kedua lalu jelaskan
        perbedaannya dengan \omterm{prop:g12:adaptive-immunity:selection}{seleksi klonal}.
  \item Virus \omterm{def:g12:adaptive-immunity:vaccine}{vaksinnya} dilemahkan tetapi hidup. Mengapa virus hidup
        yang dipakai, dan lengan tanggapan mana yang dilatihnya yang
        tak akan dilatih virus yang mati?
  \item Sepuluh tahun kemudian anak itu bertemu campak. Apa yang
        terjadi pada tiga hari pertama, dan mengapa ia tidak jatuh
        sakit?
  \item Sepupunya yang tak divaksinasi bertemu virus yang sama. Uraikan
        sepuluh hari pertamanya, dan apa yang disimpannya sesudah itu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Ambangnya.}
\begin{enumerate}[resume]
  \item Hitunglah bagian \omterm{def:g12:selection-drift-speciation:population}{populasi} yang harus kebal agar campaknya
        berhenti menyebar.
  \item Dengan 90\% yang kebal, berapa orang yang dapat dijangkiti satu
        orang yang terjangkit, rata-rata? Apakah kotanya terlindungi?
  \item Berapa orang di kota itu yang tidak kebal, dan berapa di
        antaranya yang bayi?
  \item Jika cakupannya naik menjadi 95\%, berapa orang rentan yang
        akan tersisa, dan berapa yang akan dihasilkan satu kasus
        rata-rata?
  \item Jelaskan dalam satu kalimat mengapa keselamatan bayinya
        bergantung pada keputusan orang dewasanya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Sebuah wabah.}
Seorang musafir membawa campak ke kota itu pada imunitas 90\%.
\begin{enumerate}[resume]
  \item Ambillah tiap kasus menjangkiti $15 \times 0.10 = 1.5$ orang
        lain, dalam generasi berturut-turut selama 12 hari. Berapa
        kasus pada generasi ketiga? Pada generasi keenam?
  \item Berapa kasus seluruhnya setelah enam generasi?
  \item Dari 5000 orang yang rentan, 30\% adalah bayi. Taksirlah cacah
        bayi di antara kasusnya setelah enam generasi, bila kasusnya
        jatuh merata pada yang rentan.
  \item Tiap generasi kasus meninggalkan penyintas yang kebal:
        jelaskan mengapa wabahnya melambat dengan sendirinya, dan
        kira-kira kapan.
  \item Ulangi pertanyaan 11 untuk kota dengan imunitas 95\%.
        Bandingkan kedua kota itu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Sistem yang dibongkar.}
Seorang pasien yang terjangkit HIV, tanpa pengobatan, kehilangan \omterm{prop:g12:adaptive-immunity:tcells}{sel T
penolong} dari 1000 menjadi 200 per mikroliter dalam 8 tahun.
\begin{enumerate}[resume]
  \item Hitunglah kehilangan rata-ratanya tiap tahun, dan tiap bulan.
  \item Pada 200 per mikroliter, \omterm{prop:g12:adaptive-immunity:hiv}{AIDS} mulai. Jelaskan, dengan gambar
        susunannya, mengapa \omterm{prop:g12:adaptive-immunity:antibodies}{antibodi} dan \omterm{def:g10:cells-common-unit:cell}{sel} pembunuhnya sama-sama
        terkena walaupun hanya \omterm{def:g10:cells-common-unit:cell}{sel} penolong yang terjangkit.
  \item \omterm{prop:g12:adaptive-immunity:selection}{Sel ingatan} campak pasien itu masih utuh. Apakah paparan
        campak sekarang akan berbahaya? Jelaskan.
  \item Pengobatan antivirus yang dimulai pada tahun ke-8 menaikkan
        cacahnya kembali ke 600 dalam dua tahun. Apa yang ditunjukkan
        hal ini tentang \omterm{prop:g12:adaptive-immunity:selection}{sel ingatan} dan khazanahnya, dan mengapa
        pengobatannya harus berlanjut?
  \item Nyatakan hasilnya: cakupan yang melindungi bayi kota itu,
        cacah kasus yang ditebarkan satu musafir dalam enam generasi
        pada 90\%, dan satu \omterm{def:g10:cells-common-unit:cell}{sel} yang kehilangannya meruntuhkan seluruh
        sistem adaptif.
\end{enumerate}
\end{problem}
