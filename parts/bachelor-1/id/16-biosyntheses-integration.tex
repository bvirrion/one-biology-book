\chapter{Biosintesis dan Sel yang Terpadu}\label{ch:b1:biosyntheses-integration}

Pada pukul delapan pagi sebuah hati sedang mengubah gula sarapan
menjadi \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} dan \omterm{def:b1:lipids:triglyceride}{lemak}; pada pukul tiga dini hari berikutnya hati
yang sama sedang membuat gula dari \omterm{def:b1:proteins:aminoacid}{asam amino} otot lalu mencurahkannya
ke dalam darah bagi sebuah otak yang tidak makan apa pun selama sembilan
jam. Tak ada yang berubah pada \omterm{def:b1:enzymes:enzyme}{enzimnya}; yang berubah adalah \omterm{def:b1:enzymes:enzyme}{enzim} mana
yang dinyalakan. Bab-bab sebelumnya membongkar molekul \omterm{def:b1:cell-unit-of-life:cell}{sel}; bab ini
menyusunnya kembali --- sintesis gula, \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}, \omterm{def:b1:lipids:triglyceride}{lemak} dan \omterm{def:b1:proteins:aminoacid}{asam amino}
--- lalu bertanya bagaimana sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel}, dan sebuah \omterm{def:b1:organism-environment:organism}{organisme}, memutuskan
pada tiap saat lintasan mana yang berjalan. Jawabannya adalah sehimpunan
kecil mata uang bersama, beberapa \omterm{def:b1:enzymes:enzyme}{enzim} di persimpangannya, dan hormon
yang memberi tahu tiap \omterm{def:b1:cell-unit-of-life:cell}{sel} keadaan keseluruhannya.

\section{Apa yang dibutuhkan sebuah sintesis}

\begin{proposition}[Ketiga tuntutan anabolisme]\label{prop:b1:biosyntheses-integration:needs}
\index{anabolisme}
Tiap biosintesis memerlukan \emph{kerangka karbon}, yang ditarik dari
segelintir zat antara \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis} dan \omterm{def:b1:respiration-fermentation:krebs}{daur Krebs}; \emph{daya reduksi},
yang dipasok sebagai \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}; dan \emph{energi}, yang dipasok sebagai
\omterm{def:b1:water-small-molecules:atp}{ATP}. Karena itu katabolisme dan anabolisme bertemu di persimpangan yang
sama --- glukosa-6-fosfat, triosa fosfat, piruvat, \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA},
$\alpha$-ketoglutarat, oksaloasetat --- tetapi berjalan pada \omterm{def:b1:enzymes:enzyme}{enzim} yang
terpisah pada langkah tak berbaliknya, sehingga tiap arahnya dapat
disakelar sendiri-sendiri, dan keduanya memakai \omterm{def:b1:enzymes:enzyme}{koenzim} yang terpisah
bagi elektronnya: NADH, yang dijaga teroksidasi, untuk membawa elektron
pergi dalam katabolisme; \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}, yang dijaga tereduksi, untuk
menyerahkannya dalam sintesis.
\end{proposition}

\begin{definition}[Lintasan pentosa fosfat]\label{def:b1:biosyntheses-integration:ppp}
\index{lintasan pentosa fosfat}\index{NADPH}
\emph{Lintasan pentosa fosfat} dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosol} mengoksidasi
glukosa-6-fosfat menjadi ribulosa-5-fosfat dan $\mathrm{CO_2}$, sambil
mereduksi dua $\mathrm{NADP^+}$ menjadi NADPH; cabangnya yang tak
mengoksidasi lalu saling mengubah gula fosfat berkarbon lima, empat,
enam dan tujuh, sehingga \omterm{def:b1:cell-unit-of-life:cell}{sel} dapat membuat ribosa-5-fosfat bagi
\omterm{def:b1:nucleic-acids:nucleotide}{nukleotidanya} ketika ia memerlukan pentosa, NADPH ketika ia memerlukan
daya reduksi (sintesis \omterm{def:b1:lipids:triglyceride}{lemak}, pertahanan terhadap pengoksidasi), atau
keduanya, lalu mengembalikan sisanya ke \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis}. Ia sumber utama
NADPH dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} hewan; pada tumbuhan \omterm{prop:b1:photosynthesis:lightreactions}{reaksi terang} kloroplasnya memasok
NADPH pada siang hari.
\end{definition}

\section{Membuat glukosa: glukoneogenesis}

\begin{definition}[Glukoneogenesis]\label{def:b1:biosyntheses-integration:gluconeogenesis}
\index{glukoneogenesis}
\emph{Glukoneogenesis} membuat glukosa dari prazat bukan \omterm{def:b1:carbohydrates:monosaccharide}{karbohidrat}
--- laktat, piruvat, gliserol dan kerangka karbon sebagian besar \omterm{def:b1:proteins:aminoacid}{asam
amino} --- di dalam hati (dan sedikit di ginjal). Ia menjalankan
\omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis} mundur lewat ketujuh langkahnya yang berbalikan dan
memintasi ketiga langkah tak berbaliknya dengan \omterm{def:b1:enzymes:enzyme}{enzim} yang berbeda:
piruvat dikarboksilasi menjadi oksaloasetat (dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria}, satu
\omterm{def:b1:water-small-molecules:atp}{ATP}) lalu didekarboksilasi menjadi fosfoenolpiruvat (satu GTP);
fruktosa-1,6-bisfosfat dihidrolisis menjadi fruktosa-6-fosfat;
glukosa-6-fosfat dihidrolisis menjadi glukosa bebas, yang meninggalkan
\omterm{def:b1:cell-unit-of-life:cell}{selnya}. Tiap glukosa: 6 setara \omterm{def:b1:water-small-molecules:atp}{ATP} dan 2 NADH, berhadapan dengan 2 \omterm{def:b1:water-small-molecules:atp}{ATP}
yang dihasilkan \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis} --- harga menjalankan lintasan menurun secara
mendaki.
\end{definition}

\begin{proposition}[Apa yang dapat dan tak dapat menjadi glukosa]\label{prop:b1:biosyntheses-integration:cannot}
Laktat, gliserol, dan \omterm{def:b1:proteins:aminoacid}{asam amino} yang menghasilkan piruvat atau zat
antara \omterm{def:b1:respiration-fermentation:krebs}{daur Krebs} (semuanya kecuali leusina dan lisina) dapat diubah
menjadi glukosa. \omterm{def:b1:lipids:fattyacid}{Asam lemak} tidak, pada hewan: oksidasi $\beta$-nya
memberikan \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA}, yang kedua karbonnya masuk ke daurnya lalu pergi
sebagai dua $\mathrm{CO_2}$ sebelum ada oksaloasetat yang diperoleh,
sehingga tak ada rute bersih dari \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA} ke gula. Tumbuhan, jamur
dan bakteri memiliki \emph{daur glioksilat}, sebuah jalan pintas yang
melewatkan kedua dekarboksilasinya dan membuat sebutir biji berminyak
yang berkecambah dapat mengubah \omterm{def:b1:lipids:triglyceride}{lemaknya} menjadi gula yang dibutuhkan
kecambahnya. Sebaliknya seekor mamalia yang kelaparan harus membuat
glukosanya dari \omterm{def:b1:proteins:peptide}{protein}.
\end{proposition}

\begin{example}[Daur Cori]\label{ex:b1:biosyntheses-integration:cori}
Sebuah otot yang berlari cepat memfermentasi \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya} menjadi
laktat, yang masuk ke darah; hati menyerap laktat itu, mengoksidasinya
menjadi piruvat, membuat glukosa darinya dengan 6 \omterm{def:b1:water-small-molecules:atp}{ATP} tiap glukosa,
lalu mengembalikan glukosanya ke darah agar ototnya memakainya lagi.
Ototnya memperoleh 2 \omterm{def:b1:water-small-molecules:atp}{ATP} tiap glukosa tanpa oksigen dan menyerahkan
tagihan 6 kepada hatinya, yang membayarnya dengan oksigen tanpa
tergesa-gesa: ``utang oksigen'' sebuah lari cepat sebagian besar dilunasi
di dalam hati.
\end{example}

\section{Menimbun: glikogen dan lemak}

\begin{definition}[Sintesis glikogen]\label{def:b1:biosyntheses-integration:glycogensynthesis}
Glukosa ditimbun sebagai \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} dengan mengaktifkan
glukosa-6-fosfat menjadi \emph{UDP-glukosa} (satu UTP), yang
ditambahkan \emph{\omterm{def:b1:carbohydrates:polysaccharide}{glikogen} sintase} pada ujung tak mereduksi pohon yang
sudah ada; sebuah \emph{\omterm{def:b1:enzymes:enzyme}{enzim} pencabang} memotong lalu memasangkan
kembali rantai pendek untuk membuat cabang $\alpha(1{\to}6)$-nya
(\cref{ch:b1:carbohydrates}). Sintase dan fosforilasenya diatur ke arah
yang berlawanan oleh isyarat yang sama: fosforilasi yang menyalakan
fosforilase memadamkan sintasenya, sehingga pohonnya tak pernah
dibangun dan dibongkar sekaligus.
\end{definition}

\begin{definition}[Sintesis asam lemak]\label{def:b1:biosyntheses-integration:fasynthesis}
\index{sintesis asam lemak}
\omterm{def:b1:lipids:fattyacid}{Asam lemak} dibuat dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosol} dari \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA} yang diekspor dari
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria}. \emph{\omterm{def:b1:respiration-fermentation:pdh}{Asetil-KoA} karboksilase}, langkah yang diatur,
mengarboksilasinya (satu \omterm{def:b1:water-small-molecules:atp}{ATP}) menjadi malonil-KoA; \emph{\omterm{def:b1:lipids:fattyacid}{asam lemak}
sintase} lalu menambahkan dua karbon pada satu waktu dari malonil-KoA
pada rantai yang sedang tumbuh, sambil mereduksi tiap penambahannya
dengan dua \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}, sampai palmitat (C16) dilepaskan: 8 \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA}, 7 \omterm{def:b1:water-small-molecules:atp}{ATP}
dan 14 \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} tiap palmitat. Pemanjangan dan penakjenuhan dalam \omterm{def:b1:eukaryotic-cell:endomembrane}{retikulum
endoplasma} membuat asam yang lain; pengesteran dengan
gliserol-3-fosfat (dari triosa fosfat) membuat \omterm{def:b1:lipids:triglyceride}{trigliserida}.
Lintasannya bukanlah oksidasi $\beta$ yang dibalik: \omterm{def:b1:enzymes:enzyme}{enzimnya} berbeda,
ruangnya berbeda, malonil-KoA alih-alih \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA}, \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} alih-alih
$\mathrm{FADH_2}$ dan NADH --- dan malonil-KoA menghambat pengangkutan
\omterm{def:b1:lipids:fattyacid}{asam lemak} ke dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria}, sehingga sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} tidak membuat
\omterm{def:b1:lipids:triglyceride}{lemak} sekaligus membakarnya.
\end{definition}

\begin{example}[Lemak dari gula]\label{ex:b1:biosyntheses-integration:fatfromsugar}
Sebuah hati yang diberi glukosa lebih banyak daripada yang dapat
ditimbunnya sebagai \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} membuat \omterm{def:b1:lipids:triglyceride}{lemak}: glukosa menjadi piruvat
menjadi \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA} (sambil kehilangan sepertiga karbonnya sebagai
$\mathrm{CO_2}$ dan memperoleh \omterm{def:b1:water-small-molecules:atp}{ATP}), \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA} menjadi palmitat dengan
biaya \omterm{def:b1:water-small-molecules:atp}{ATP} itu dan \omterm{def:b1:biosyntheses-integration:ppp}{NADPH} dari \omterm{def:b1:biosyntheses-integration:ppp}{lintasan pentosa fosfatnya}, palmitat
menjadi \omterm{def:b1:lipids:triglyceride}{trigliserida} yang dikirim ke \omterm{def:b1:body-plans-tissues:tissue}{jaringan} \omterm{def:b1:lipids:triglyceride}{lemak}. Sekitar
\qty{15}{\%} energi gulanya hilang dalam pengubahannya; sisanya ditimbun
sembilan kali lebih padat daripada \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} (\cref{ch:b1:lipids}).
Kebalikannya --- \omterm{def:b1:lipids:triglyceride}{lemak} menjadi gula --- mustahil.
\end{example}

\section{Nitrogen: asam amino dan nukleotida}

\begin{definition}[Asimilasi nitrogen, transaminasi]\label{def:b1:biosyntheses-integration:nitrogen}
\index{transaminasi}\index{glutamat}
Amonium masuk ke materi organik hampir seluruhnya lewat
\emph{glutamat}: glutamat dehidrogenase menambahkannya pada
$\alpha$-ketoglutarat, dan glutamina sintetase menambahkan yang kedua
pada \omterm{def:b1:proteins:aminoacid}{rantai samping} glutamat (satu \omterm{def:b1:water-small-molecules:atp}{ATP}), sehingga membuat glutamina.
Dari kedua penyumbang itu, \emph{transaminase} (aminotransferase,
dengan \omterm{def:b1:enzymes:enzyme}{koenzim} vitamin B$_6$) memindahkan gugus aminonya ke asam
$\alpha$-keto mana pun, sehingga membangun \omterm{def:b1:proteins:aminoacid}{asam amino} dari kerangka
karbon lintasan pusatnya: keluarga glutamat dari
$\alpha$-ketoglutarat, keluarga aspartat dari oksaloasetat, alanina dan
serina dari piruvat serta 3-fosfogliserat. Tumbuhan dan bakteri membuat
kedua puluhnya; hewan telah kehilangan lintasan panjang menuju sembilan
di antaranya, yaitu \omterm{def:b1:proteins:aminoacid}{asam amino} \emph{esensial}, dan mengambilnya dari
makanan. Transaminase yang sama, yang dijalankan mundur, melucuti
nitrogen dari \omterm{def:b1:proteins:aminoacid}{asam amino} yang berlebih untuk diekskresikan sebagai
amonia (ikan), urea (mamalia) atau asam urat (burung, serangga).
\end{definition}

\begin{example}[Nukleotida]\label{ex:b1:biosyntheses-integration:nucleotides}
Sebuah cincin \omterm{def:b1:nucleic-acids:nucleotide}{purin} dirakit pada ribosa-5-fosfat dari glisina,
aspartat, dua glutamina, dua satuan berkarbon satu dan $\mathrm{CO_2}$,
dengan biaya enam \omterm{def:b1:water-small-molecules:atp}{ATP}; sebuah \omterm{def:b1:nucleic-acids:nucleotide}{pirimidin} dari aspartat dan karbamoil
fosfat. Deoksinukleotida dibuat dari ribonukleotida dengan mereduksi
gulanya, memakai \omterm{def:b1:biosyntheses-integration:ppp}{NADPH}. Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} yang hendak membelah membuat sekitar
$10^{10}$ \omterm{def:b1:nucleic-acids:nucleotide}{nukleotida} dalam sejam; obat yang menghambat sintesis ini ---
metotreksat, 5-fluorourasil --- menghentikan \omterm{def:b1:cell-unit-of-life:cell}{sel} yang membelah lebih
dahulu, dan itulah sebabnya obat itu mengobati kanker.
\end{example}

\section{Mamalia yang terpadu}

\begin{proposition}[Kenyang dan berpuasa]\label{prop:b1:biosyntheses-integration:fedfasting}
\index{insulin}\index{glukagon}
\omterm{def:b1:mammal-organization:organ}{Organnya} berbagi pekerjaan. Sesudah sebuah santapan, \emph{insulin}
dari pankreas memberi tahu hati agar menimbun glukosa sebagai \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}
lalu mengubah kelebihannya menjadi \omterm{def:b1:lipids:triglyceride}{lemak}, memberi tahu otot agar
menyerap glukosa dan menimbun \omterm{def:b1:carbohydrates:polysaccharide}{glikogen}, memberi tahu \omterm{def:b1:body-plans-tissues:tissue}{jaringan} \omterm{def:b1:lipids:triglyceride}{lemak}
agar menyerap \omterm{def:b1:lipids:triglyceride}{lemak}; glukosa darahnya, yang telah naik ke
\qty{8}{mmol/L}, kembali ke \qty{5}{} dalam dua jam. Di antara waktu
makan, \emph{glukagon} memberi tahu hati agar menguraikan \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya}
dan, seiring habisnya \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} itu, membuat glukosa dari laktat,
gliserol serta \omterm{def:b1:proteins:aminoacid}{asam amino}, dan memberi tahu \omterm{def:b1:body-plans-tissues:tissue}{jaringan} \omterm{def:b1:lipids:triglyceride}{lemak} agar
melepas \omterm{def:b1:lipids:fattyacid}{asam lemak}, yang dibakar otot dan hati menggantikan glukosa;

\end{proposition}

\begin{method}[Membaca sebuah keadaan metabolik]\label{met:b1:biosyntheses-integration:state}
\begin{enumerate}
  \item Tanyakan bahan bakar mana yang berlimpah: glukosa yang tinggi
        berarti penimbunan (\omterm{def:b1:carbohydrates:polysaccharide}{glikogen}, lalu \omterm{def:b1:lipids:triglyceride}{lemak}); glukosa yang rendah
        berarti penggerakan.
  \item Ikuti isyaratnya: insulin mengaktifkan sintase dan
        menonaktifkan fosforilase serta lipase; glukagon melakukan
        kebalikannya --- sebagian besar lewat fosforilasi \omterm{def:b1:enzymes:enzyme}{enzim} yang sama
        (\cref{ch:b1:enzymes}).
  \item Periksa persimpangannya: \omterm{def:b1:water-small-molecules:atp}{ATP} dan sitrat menghambat PFK lalu
        membuka \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis}; AMP melakukan sebaliknya; malonil-KoA
        menghambat pembakaran \omterm{def:b1:lipids:triglyceride}{lemak} selama \omterm{def:b1:lipids:triglyceride}{lemak} sedang dibuat;
        \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA} mengaktifkan karboksilase yang memulai
        \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis}.
  \item Bagikan \omterm{def:b1:mammal-organization:organ}{organnya}: hati (menimbun, membuat dan mengekspor
        glukosa; membuat keton dan urea), otot (menimbun \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} bagi
        dirinya sendiri, mengekspor laktat dan alanina), \omterm{def:b1:body-plans-tissues:tissue}{jaringan} \omterm{def:b1:lipids:triglyceride}{lemak}
        (menimbun dan melepas \omterm{def:b1:lipids:triglyceride}{lemak}), otak (membakar glukosa, lalu
        keton).
\end{enumerate}
\end{method}

\begin{proposition}[Sumber dan lubuk]\label{prop:b1:biosyntheses-integration:sourcesink}
Pada siang hari kloroplas sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} mengekspor triosa fosfat ke
\omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya}, tempat ia dibuat menjadi \emph{sukrosa} untuk diekspor
dalam floem (\cref{ch:b1:plant-transport}) ke \omterm{def:b1:flowering-plant-organization:organs}{akar}, buah dan ujung yang
tumbuh; apa yang tak sanggup dibawa floemnya disimpan dalam kloroplas
sebagai \emph{\omterm{def:b1:carbohydrates:polysaccharide}{pati}}, sebuah simpanan sementara, lalu digerakkan pada
malam hari agar sukrosanya terus mengalir. Bila hati mamalia memutuskan
antara \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} dan ekspor lewat hormon, sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} memutuskan antara
\omterm{def:b1:carbohydrates:polysaccharide}{pati} dan sukrosa lewat aras fosfat dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya}, yang
mengendalikan pengekspor selubung kloroplasnya. Sebuah tumbuhan
mengatur lewat kepekatan metabolitnya sendiri, \omterm{def:b1:mammal-organization:organ}{organ} demi \omterm{def:b1:mammal-organization:organ}{organ};
mamalia menambahkan sebuah perintah saraf dan hormon atas
keseluruhannya.
\end{proposition}

\begin{example}[Pati semalam]\label{ex:b1:biosyntheses-integration:nightstarch}
Sehelai \omterm{def:b1:flowering-plant-organization:organs}{daun} menyisihkan sebagai \omterm{def:b1:carbohydrates:polysaccharide}{pati} sekitar separuh dari yang

sebuah jam, dalam sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} yang tak berotak.
\end{example}

\section{Soal: Dua Puluh Empat Jam Seorang Manusia}

\begin{problem}[{Soal akhir pekan --- sehari tanpa sarapan: simpanannya
yang ditimbang, otaknya yang diberi makan, proteinnya yang dihitung dan
lemaknya yang digerakkan, berakhir pada berapa jam puasa yang dapat
ditutup glikogen hatinya}]
\label{pb:b1:biosyntheses-integration:1}
Seorang dewasa \qty{70}{kg} mengeluarkan \qty{8.4}{MJ} sehari
(\qty{100}{W} rata-rata) dan memegang: \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} hati \qty{100}{g},
\omterm{def:b1:carbohydrates:polysaccharide}{glikogen} otot \qty{400}{g}, \omterm{def:b1:lipids:triglyceride}{lemak} \qty{12}{kg}, \omterm{def:b1:proteins:peptide}{protein} \qty{10}{kg}
(yang \qty{6}{kg} di antaranya dalam otot). Otaknya memakai
\qty{120}{g} glukosa sehari (\qty{5}{g/h}); \omterm{def:b1:body-plans-tissues:tissue}{jaringan} lain yang
bergantung glukosa (\omterm{def:b1:cell-unit-of-life:cell}{sel} darah merah, medula ginjal) \qty{40}{g}.
Energi: \omterm{def:b1:carbohydrates:monosaccharide}{karbohidrat} \qty{17}{kJ/g}, \omterm{def:b1:lipids:triglyceride}{lemak} \qty{38}{kJ/g}, \omterm{def:b1:proteins:peptide}{protein}
\qty{17}{kJ/g}; \qty{1}{g} \omterm{def:b1:proteins:peptide}{protein} menghasilkan \qty{0.55}{g} glukosa
lewat \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis}; \qty{1}{g} \omterm{def:b1:lipids:triglyceride}{lemak} menghasilkan \qty{0.1}{g}
glukosa (dari gliserolnya).

\medskip
\noindent\textbf{Bagian I --- Simpanannya.}
\begin{enumerate}
  \item Hitunglah energi yang dipegang tiap keempat simpanannya, dalam
        megajoule, dan berapa hari pengeluaran yang diwakili
        masing-masing.
  \item Simpanan mana yang dapat memasok glukosa ke darah secara
        langsung? Mengapa \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} otot tidak?
  \item Hitunglah total glukosa yang dibutuhkan tubuhnya tiap hari bagi
        \omterm{def:b1:body-plans-tissues:tissue}{jaringan} yang bergantung glukosa.
  \item Hitunglah glukosa dalam darahnya sendiri (\qty{5}{mmol/L},
        \qty{5}{L}, \qty{180}{g/mol}) dan berapa menit ia akan memberi
        makan otaknya saja.
  \item Berapa fraksi pengeluaran hariannya yang berupa glukosa otak?
  \item Sesudah sebuah santapan, \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} hatinya penuh dan glukosa
        darahnya masih naik. Sebutkan nasib berikutnya glukosa itu dan
        hormon yang mengarahkannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Malamnya.}
Santapan terakhirnya berakhir pukul 20:00; sejak itu hatinya sendirian
memasok glukosa darahnya.
\begin{enumerate}[resume]
  \item Pada \qty{160}{g} glukosa tiap hari, berapa permintaan tiap jam
        \omterm{def:b1:body-plans-tissues:tissue}{jaringan} yang bergantung glukosa?
  \item Berapa lama \qty{100}{g} \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} hati bertahan pada laju itu,
        bila tak ada yang lain memasok glukosa?
  \item Sesungguhnya \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis} memasok sekitar sepertiga
        glukosanya sejak jam-jam pertamanya. Hitunglah ulang lama
        \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya}.
  \item Pada pukul berapa pagi harinya \omterm{def:b1:carbohydrates:polysaccharide}{glikogen} hatinya habis menurut
        tiap taksirannya?
  \item Sementara itu ototnya membakar \omterm{def:b1:lipids:fattyacid}{asam lemak}. Hitunglah \omterm{def:b1:lipids:triglyceride}{lemak} yang
        dioksidasi semalam (\qty{12}{h}) bila otot itu menanggung
        \qty{60}{\%} pengeluaran istirahatnya dan hanya membakar \omterm{def:b1:lipids:triglyceride}{lemak}.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Hari berikutnya, tanpa makan.}
\begin{enumerate}[resume]
  \item Dengan \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya} habis, \qty{160}{g} glukosa itu harus datang
        dari \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis}. Hitunglah \omterm{def:b1:proteins:peptide}{protein} yang dibutuhkan tiap
        hari, dan massa otot yang diwakilinya.
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibelanjakan hati untuk membuat \qty{160}{g}
        glukosa dari piruvat (6 \omterm{def:b1:water-small-molecules:atp}{ATP} tiap glukosa) dan energinya pada
        \qty{50}{kJ/mol}.
  \item Hitunglah sumbangan gliserol \omterm{def:b1:lipids:triglyceride}{lemaknya}: berapa banyak \omterm{def:b1:lipids:triglyceride}{lemak} yang
        dioksidasi tiap hari bila ia menutup seluruh pengeluaran bukan
        glukosanya, dan berapa banyak glukosa yang dihasilkan
        gliserolnya?
  \item Hitunglah ulang \omterm{def:b1:proteins:peptide}{protein} yang dibutuhkan dengan sumbangan
        gliserolnya dikurangkan.
  \item Pada laju itu, berapa hari sampai separuh ototnya hilang?
  \item Sejak hari ketiga hatinya membuat badan keton dan otaknya
        mengambil dua pertiga energinya dari keton itu; kebutuhan
        glukosanya turun ke \qty{40}{g} bagi otaknya ditambah
        \qty{40}{g} bagi selebihnya. Hitunglah ulang kehilangan
        \omterm{def:b1:proteins:peptide}{proteinnya} tiap hari dan berapa hari sampai separuh ototnya
        hilang.
  \item Hitunglah berapa hari simpanan \omterm{def:b1:lipids:triglyceride}{lemaknya} bertahan pada
        \qty{8.4}{MJ} sehari (nilai itu turun ke \qty{6.5}{MJ} dalam
        puasa yang berkepanjangan: hitunglah ulang).
  \item Jelaskan mengapa seorang yang berpuasa mati karena kehilangan
        \omterm{def:b1:proteins:peptide}{protein} sebelum \omterm{def:b1:lipids:triglyceride}{lemaknya} habis, dan apa yang ditunda badan
        keton.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Katupnya.}
\begin{enumerate}[resume]
  \item Sebutkan \omterm{def:b1:enzymes:enzyme}{enzim} hati yang harus aktif pada pukul 03:00 dan tidak
        aktif pada pukul 09:00, dan \omterm{def:b1:enzymes:enzyme}{enzim} yang harus sebaliknya, bagi
        \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya}.
  \item Sebutkan isyarat (hormon) yang menyetel keduanya pada tiap
        arahnya, dan perubahan kimia yang menjadi cara kerjanya.
  \item Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} hati pada pukul 03:00 memiliki \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA} yang
        tinggi dari oksidasi \omterm{def:b1:lipids:triglyceride}{lemaknya}. Sebutkan \omterm{def:b1:enzymes:enzyme}{enzim} \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis}
        yang diaktifkannya dan \omterm{def:b1:enzymes:enzyme}{enzim} \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis} yang dihambat \omterm{def:b1:water-small-molecules:atp}{ATP} pada
        saat yang sama.
  \item Jelaskan mengapa \omterm{def:b1:biosyntheses-integration:gluconeogenesis}{glukoneogenesis} dan \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis}, yang berjalan

  \item Seorang penderita diabetes tanpa insulin memiliki glukosa darah
