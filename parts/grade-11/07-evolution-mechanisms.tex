\chapter{Mechanisms of Evolution}\label{ch:g11:evolution-mechanisms}

Life's diversity is not a static catalogue: \omterm{def:g11:evolution-mechanisms:pool}{populations} change genetically over
generations. \emph{\omterm{def:g11:evolution-mechanisms:evo}{Evolution}} in the population-genetic sense is change in \omterm{def:g11:evolution-mechanisms:evo}{allele
frequencies}. This chapter surveys evidence that \omterm{def:g11:evolution-mechanisms:evo}{evolution} has occurred, then the
main mechanisms that drive it --- \omterm{def:g11:evolution-mechanisms:mutation}{mutation}, selection, drift, and \omterm{def:g11:evolution-mechanisms:geneflow}{gene flow} ---
and a light introduction to \omterm{def:g11:evolution-mechanisms:hw}{Hardy--Weinberg equilibrium} as a null model.

\section{What evolution means}

\begin{definition}[Biological evolution]\label{def:g11:evolution-mechanisms:evo}
\emph{Evolution}\index{evolution} is change in the inherited characteristics of
\omterm{def:g11:evolution-mechanisms:pool}{populations} over generations. At the genetic level it is change in
\emph{allele frequencies}\index{allele frequency} in a \omterm{def:g11:evolution-mechanisms:pool}{population}'s \omterm{def:g11:evolution-mechanisms:pool}{gene pool}.
\end{definition}

\begin{definition}[Population and gene pool]\label{def:g11:evolution-mechanisms:pool}
A \emph{population}\index{population} is a group of individuals of one species
that interbreed in a place and time. The \emph{gene pool}\index{gene pool} is
the complete set of \omterm{def:g11:mendelian-genetics:allele}{alleles} at all loci in that population.
\end{definition}

\begin{remark}
Individual \omterm{def:g10:scientific-biology:organism}{organisms} develop and die; they do not evolve in the \omterm{def:g11:evolution-mechanisms:pool}{population} sense.
\omterm{def:g11:evolution-mechanisms:selection}{Natural selection} acts on individual \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes}, but the evolutionary outcome is
recorded as changed frequencies in the next generation's \omterm{def:g11:evolution-mechanisms:pool}{gene pool}.
\end{remark}

\section{Evidence for evolution}

\begin{proposition}[Lines of evidence]\label{prop:g11:evolution-mechanisms:evidence}
Multiple independent lines support common descent with modification:
\begin{itemize}
  \item \emph{Fossils}\index{fossil}: extinct forms in temporal \omterm{def:g10:biodiversity-classification:hierarchy}{order}; transitional
        fossils linking major groups.
  \item \emph{Biogeography}\index{biogeography}: related species on nearby
        islands; convergent forms in similar climates far apart.
  \item \emph{Comparative anatomy}: \emph{homologous structures}\index{homologous
        structure} (shared ancestry) versus \emph{analogous
        structures}\index{analogous structure} (shared function, different origin).
  \item \emph{Embryology}: similar early developmental stages among vertebrates.
  \item \emph{Molecular homology}: shared \omterm{def:g11:gene-expression:codon}{genetic code}; \omterm{def:g10:molecules-of-life:proteins}{sequence} similarity
        nested in the same pattern as other evidence.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Homology versus analogy]\label{ex:g11:evolution-mechanisms:homology}
The forelimb bones of a human, bat and whale are homologous: same skeletal
elements, different functions. The wings of a bat and an insect are analogous as
wings: flight evolved separately; the structures are not built on the same
ancestral wing.
\end{example}

\section{Mutation: the source of new alleles}

\begin{definition}[Mutation]\label{def:g11:evolution-mechanisms:mutation}
A \emph{mutation}\index{mutation} is a heritable change in \omterm{def:g11:dna-replication:helix}{DNA} \omterm{def:g10:molecules-of-life:proteins}{sequence}. Point
mutations, insertions, deletions and larger chromosomal changes can create new
\omterm{def:g11:mendelian-genetics:allele}{alleles}. Mutation rates per base are low, but \omterm{prop:g11:dna-replication:genome}{genomes} are large and \omterm{def:g11:evolution-mechanisms:pool}{populations}
can be huge, so new variation continually appears.
\end{definition}

\begin{proposition}[Mutation is random with respect to need]\label{prop:g11:evolution-mechanisms:random}
\omterm{def:g11:evolution-mechanisms:mutation}{Mutations} do not arise because they would be useful; they arise from \omterm{def:g11:dna-replication:semi}{replication}
errors and damage. Selection (below) filters them after the fact.
\end{proposition}
\admitted

\section{Natural selection}

\begin{definition}[Natural selection]\label{def:g11:evolution-mechanisms:selection}
\emph{Natural selection}\index{natural selection} is the differential survival
and \omterm{def:g10:scientific-biology:properties}{reproduction} of individuals due to differences in \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}. If those
differences are heritable, \omterm{def:g11:evolution-mechanisms:evo}{allele frequencies} change across generations.
\end{definition}

\begin{proposition}[Requirements for selection]\label{prop:g11:evolution-mechanisms:req}
Selection produces adaptive change when:
\begin{enumerate}
  \item individuals vary in a trait;
  \item more offspring are produced than can survive (or reproductive success
        varies);
  \item trait variants are heritable and affect survival or \omterm{def:g10:scientific-biology:properties}{reproduction}.
\end{enumerate}
\end{proposition}
\admitted

\begin{definition}[Fitness and adaptation]\label{def:g11:evolution-mechanisms:fitness}
In evolutionary genetics, \emph{fitness}\index{fitness} measures relative
reproductive success. An \emph{adaptation}\index{adaptation} is a trait shaped by
selection that improves fitness in a given environment.
\end{definition}

\begin{example}[Antibiotic resistance]\label{ex:g11:evolution-mechanisms:abx}
\omterm{def:g10:biodiversity-classification:domains}{Bacteria} vary genetically; some \omterm{def:g11:mendelian-genetics:allele}{alleles} already confer resistance. Antibiotics
kill sensitive \omterm{def:g10:the-cell:cell}{cells}, so resistant \omterm{def:g11:mendelian-genetics:allele}{alleles} increase in frequency. The drug did
not ``create'' the \omterm{def:g11:evolution-mechanisms:mutation}{mutation} on demand --- it selected among existing (and newly
arising) variants.
\end{example}

\section{Genetic drift and gene flow}

\begin{definition}[Genetic drift]\label{def:g11:evolution-mechanisms:drift}
\emph{Genetic drift}\index{genetic drift} is random change in \omterm{def:g11:evolution-mechanisms:evo}{allele frequencies}
due to sampling error in finite \omterm{def:g11:evolution-mechanisms:pool}{populations}. It is stronger in small \omterm{def:g11:evolution-mechanisms:pool}{populations}
and is not directed toward \omterm{def:g11:evolution-mechanisms:fitness}{adaptation}.
\end{definition}

\begin{definition}[Bottleneck and founder effect]\label{def:g11:evolution-mechanisms:bottle}
A \emph{population bottleneck}\index{bottleneck} is a sharp reduction in size;
the survivors' \omterm{def:g11:mendelian-genetics:allele}{alleles} may misrepresent the original pool. A
\emph{founder effect}\index{founder effect} occurs when a few individuals start a
new \omterm{def:g11:evolution-mechanisms:pool}{population}: their \omterm{def:g11:evolution-mechanisms:evo}{allele frequencies} can differ by chance from the source.
\end{definition}

\begin{definition}[Gene flow]\label{def:g11:evolution-mechanisms:geneflow}
\emph{Gene flow}\index{gene flow} (migration) is the transfer of \omterm{def:g11:mendelian-genetics:allele}{alleles} between
\omterm{def:g11:evolution-mechanisms:pool}{populations} when individuals (or gametes) move and reproduce. \omterm{def:g11:gene-expression:gene}{Gene} flow tends to
homogenize \omterm{def:g11:evolution-mechanisms:evo}{allele frequencies} among \omterm{def:g11:evolution-mechanisms:pool}{populations}.
\end{definition}

\begin{proposition}[Mechanisms compared]\label{prop:g11:evolution-mechanisms:compare}
\begin{center}\small
\begin{tabular}{lll}
\textbf{Mechanism} & \textbf{Direction} & \textbf{Adaptive?} \\
\omterm{def:g11:evolution-mechanisms:mutation}{mutation} & random new \omterm{def:g11:mendelian-genetics:allele}{alleles} & not by itself \\
selection & non-random, by \omterm{def:g11:evolution-mechanisms:fitness}{fitness} & can adapt \\
drift & random frequency change & no \\
\omterm{def:g11:evolution-mechanisms:geneflow}{gene flow} & mixes \omterm{def:g11:evolution-mechanisms:pool}{populations} & sometimes \\
\end{tabular}
\end{center}
\end{proposition}
\admitted

\section{Hardy--Weinberg: a null model}

\begin{definition}[Hardy--Weinberg principle]\label{def:g11:evolution-mechanisms:hw}
For an autosomal \omterm{def:g11:gene-expression:gene}{gene} with two \omterm{def:g11:mendelian-genetics:allele}{alleles} $A$ and $a$ at frequencies $p$ and $q$
($p + q = 1$), random mating in a large \omterm{def:g11:evolution-mechanisms:pool}{population} with no selection, no
\omterm{def:g11:evolution-mechanisms:mutation}{mutation}, no migration, and fair \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis} gives \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} frequencies
\[
p^{2}\ (AA),\qquad 2pq\ (Aa),\qquad q^{2}\ (aa)
\]
in the next generation. Those frequencies stay constant thereafter under the
same assumptions: the \omterm{def:g11:evolution-mechanisms:pool}{population} is in
\emph{Hardy--Weinberg equilibrium}\index{Hardy--Weinberg equilibrium}.
\end{definition}

\begin{method}[Using Hardy--Weinberg]\label{met:g11:evolution-mechanisms:hw}
\begin{enumerate}
  \item If a \omterm{def:g11:mendelian-genetics:dominance}{recessive} trait has \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype} frequency $q^{2}$, estimate
        $q = \sqrt{q^{2}}$, then $p = 1 - q$.
  \item Carrier frequency among \omterm{def:g11:mendelian-genetics:geno-pheno}{genotypes} is about $2pq$ (for rare $q$, roughly
        $2q$).
  \item Departures from $p^{2}:2pq:q^{2}$ suggest that one or more Hardy--Weinberg
        assumptions fail (selection, non-random mating, structure, \ldots).
\end{enumerate}
\end{method}

\begin{example}[Carrier frequency]\label{ex:g11:evolution-mechanisms:carrier}
If $q^{2} = 1/10000$ for a \omterm{def:g11:mendelian-genetics:dominance}{recessive} disease, $q = 1/100$, $p \approx 1$, and
$2pq \approx 1/50$: about one person in fifty is a \omterm{def:g11:mendelian-genetics:zygous}{heterozygous} carrier even
though only one in ten thousand is affected.
\end{example}

\begin{remark}
Hardy--Weinberg is a baseline, not a claim that real \omterm{def:g11:evolution-mechanisms:pool}{populations} never evolve.
Real \omterm{def:g11:evolution-mechanisms:pool}{populations} always violate at least some assumptions slightly; the model
shows what \omterm{def:g11:mendelian-genetics:allele}{allele} and \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} frequencies would do \emph{without} evolutionary
forces.
\end{remark}

\section{Speciation (sketch)}

\begin{definition}[Species sketch]\label{def:g11:evolution-mechanisms:species}
Under the \emph{biological species concept}\index{biological species concept},
species are groups of actually or potentially interbreeding natural \omterm{def:g11:evolution-mechanisms:pool}{populations}
that are reproductively isolated from other such groups. \emph{Speciation}
\index{speciation} is the origin of new species, often beginning with geographic
separation (\emph{allopatric speciation}\index{allopatric speciation}) and
genetic divergence.
\end{definition}

\begin{remark}
Macroevolution and phylogeny continue in later \omterm{def:g10:the-cell:sa-v}{volumes}; the mechanisms of this
chapter are the microevolutionary engine that, given time and isolation, yields
the tree of life.
\end{remark}

\begin{problem}[Finches, frequencies and a null model]\label{pb:g11:evolution-mechanisms:1}
A \omterm{def:g11:evolution-mechanisms:pool}{population} of seed-eating birds has a \omterm{def:g11:gene-expression:gene}{gene} with \omterm{def:g11:mendelian-genetics:allele}{alleles} $B$ (large beak,
frequency $p$) and $b$ (smaller beak, frequency $q$). Large seeds become rare
after a drought; birds with larger beaks survive better.
\begin{enumerate}
  \item Name the evolutionary mechanism that will increase $p$ if beak size is
        heritable, and state which Hardy--Weinberg assumption is violated.
  \item Before the drought, the \omterm{def:g11:evolution-mechanisms:pool}{population} was approximately in \omterm{def:g11:evolution-mechanisms:hw}{Hardy--Weinberg
        equilibrium} with $p = 0.4$. Compute expected \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} frequencies.
  \item After selection, suppose among breeders the frequency of $b$ has fallen
        to $q' = 0.5$. If survivors mate randomly, what \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} frequencies do
        you expect in the next generation (new Hardy--Weinberg with $p' = 0.5$)?
  \item A small island is colonized by five birds from the mainland. Why might
        \omterm{def:g11:evolution-mechanisms:evo}{allele frequencies} on the island differ from the mainland even if the
        environment is identical?
  \item Later, occasional migrants arrive from the mainland. Name this mechanism
        and predict its effect on divergence between island and mainland.
  \item \omterm{prop:g11:evolution-mechanisms:evidence}{Fossils} show a \omterm{def:g10:molecules-of-life:proteins}{sequence} of beak shapes over millennia. How does that
        evidence complement the short-term frequency change in living birds?
  \item Why is the phrase ``birds evolved large beaks so that they could eat
        large seeds'' teleological, and how would you rephrase it?
\end{enumerate}
\end{problem}
