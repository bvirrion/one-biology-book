\chapter{Pola Tubuh dan Jaringan Hewan}\label{ch:b1:body-plans-tissues}

Empat ekor hewan di atas meja laboratorium: seekor hidra sepanjang satu
sentimeter, seekor cacing tanah, seekor udang karang, seekor katak. Di
balik perbedaan ukuran dan perilakunya mereka dibangun atas segelintir
keputusan bersama --- apakah tubuhnya berujung depan dan belakang,
berapa lapis sel asal pertumbuhannya, apakah ada rongga di sekeliling
ususnya, apakah tubuhnya tersusun dari segmen yang berulang, di mana
rangkanya --- dan setiap satunya dibangun dari empat macam \omterm{def:b1:body-plans-tissues:tissue}{jaringan} yang
sama. Bab ini memaparkan \omterm{def:b1:mammal-organization:bodyplan}{pola tubuh} hewan dan \omterm{def:b1:body-plans-tissues:tissue}{jaringan} yang
membangunnya, lalu berakhir di tempat bab sebelumnya berhenti: pada
\omterm{prop:b1:body-plans-tissues:gutwall}{dinding usus} seekor mamalia, yang dibaca sebagai \omterm{def:b1:body-plans-tissues:tissue}{jaringan} yang dirakit
menjadi sebuah \omterm{def:b1:mammal-organization:organ}{organ}.

\section{Pola tubuh}

\begin{definition}[Simetri dan sefalisasi]\label{def:b1:body-plans-tissues:symmetry}
Seekor hewan memiliki \emph{simetri radial}\index{simetri radial}
apabila tubuhnya tersusun mengelilingi satu sumbu pusat tanpa kiri dan
kanan, seperti hidra atau ubur-ubur: ia menemui \omterm{def:b1:organism-environment:organism}{lingkungannya} sama rata
dari segala sisi, dan biasanya menetap atau hanyut. Ia memiliki
\emph{simetri bilateral}\index{simetri bilateral} apabila satu bidang
membaginya menjadi dua belahan cermin, yang menetapkan sisi depan
(anterior) dan belakang (posterior), punggung (dorsal) dan perut
(ventral): \omterm{def:b1:mammal-organization:bodyplan}{pola tubuh} hewan yang bergerak dengan kepala di depan, dengan
indra dan pusat sarafnya terpusat di sisi depan (\omterm{def:b1:mammal-organization:bodyplan}{sefalisasi}).
\end{definition}

\begin{definition}[Lapisan lembaga, rongga tubuh]\label{def:b1:body-plans-tissues:layers}
Embrio seekor hewan membentuk dua atau tiga lapis sel, yaitu
\emph{lapisan lembaga}\index{lapisan lembaga}, yang menurunkan semua
\omterm{def:b1:body-plans-tissues:tissue}{jaringannya}: \emph{ektoderm}\index{ektoderm} di luar (kulit, sistem
saraf), \emph{endoderm}\index{endoderm} di dalam (pelapis usus dan
tonjolannya) dan, pada semua kecuali spons dan knidaria, sebuah
\emph{mesoderm}\index{mesoderm} di tengah (otot, rangka, \omterm{def:b1:body-plans-tissues:connective}{jaringan ikat},
darah, ginjal, gonad). Knidaria bersifat \emph{diploblastik}, selebihnya
\emph{triploblastik}. Di antara hewan triploblastik, mesodermnya dapat
memenuhi ruang antara usus dan dinding tubuh (\emph{aselomata}: cacing
pipih), atau menyisakan rongga di antara keduanya: sebuah
\emph{selom}\index{selom} apabila rongga itu seluruhnya dilapisi
mesoderm (anelida, moluska, artropoda, ekinodermata, vertebrata). Selom
menggantung usus di dalam ruangnya sendiri, membuatnya dapat bergerak
lepas dari dinding tubuh, dan memberi hewan bertubuh lunak sebuah \omterm{def:b1:body-plans-tissues:segments}{rangka
hidrostatik}.
\end{definition}

\begin{definition}[Protostomia dan deuterostomia]\label{def:b1:body-plans-tissues:proto}
Di antara hewan selomata, nasib bukaan pertama usus embrio memisahkan
dua garis keturunan besar: pada \emph{protostomia}\index{protostomia}
(anelida, moluska, artropoda) bukaan itu menjadi mulut, \omterm{def:b1:body-plans-tissues:layers}{selomnya}
terbentuk lewat pembelahan \omterm{def:b1:body-plans-tissues:layers}{mesoderm}, dan pembelahan awal telurnya
berpilin dengan nasib yang tertetapkan sejak dini; pada
\emph{deuterostomia}\index{deuterostomia} (ekinodermata, vertebrata)
bukaan itu menjadi anus, \omterm{def:b1:body-plans-tissues:layers}{selomnya} terbentuk lewat pengantongan usus, dan
sel awalnya tetap lentur nasibnya.
\end{definition}

\begin{definition}[Segmentasi, rangka]\label{def:b1:body-plans-tissues:segments}
Sebuah tubuh disebut \emph{bersegmen}\index{segmentasi} apabila ia
dibangun dari satuan berulang sepanjang sumbunya (gelang-gelang cacing
tanah, segmen udang karang, \omterm{def:b1:flowering-plant-organization:organs}{ruas} tulang belakang dan blok otot seekor
ikan), masing-masing dengan perangkat otot, saraf dan sering pula
anggota tubuhnya sendiri. Segmen dapat seragam atau terkhususkan menjadi
kawasan (kepala, dada, perut). \emph{Rangka}\index{rangka}, tempat otot
menarik, dapat bersifat \emph{hidrostatik}\index{rangka hidrostatik}
(cairan bertekanan di dalam sebuah \omterm{def:b1:body-plans-tissues:layers}{selom}, seperti pada cacing tanah),
berupa \emph{rangka luar}\index{rangka luar} (\omterm{def:b1:flowering-plant-organization:tissues}{kutikula} kaku di luar
tubuh, berbuku, ditanggalkan agar dapat tumbuh, seperti pada artropoda),
atau berupa \emph{rangka dalam}\index{rangka dalam} (tulang rawan atau
tulang di dalam tubuh, yang tumbuh bersamanya, seperti pada vertebrata).
\end{definition}

\begin{proposition}[Empat pola tubuh dibandingkan]\label{prop:b1:body-plans-tissues:four}
\begin{center}
\small
\begin{tabular}{lllll}
\hline
 & hidra & cacing tanah & udang karang & katak \\
\hline
simetri & radial & bilateral & bilateral & bilateral \\
\omterm{def:b1:body-plans-tissues:layers}{lapisan lembaga} & 2 & 3 & 3 & 3 \\
\omterm{def:b1:body-plans-tissues:layers}{selom} & tidak ada & ada & menyusut & ada \\
garis keturunan & --- & \omterm{def:b1:body-plans-tissues:proto}{protostomia} & \omterm{def:b1:body-plans-tissues:proto}{protostomia} & \omterm{def:b1:body-plans-tissues:proto}{deuterostomia} \\
segmen & tidak & seragam & terkhususkan & \omterm{def:b1:flowering-plant-organization:organs}{ruas} tulang, blok otot \\
rangka & hidrostatik & hidrostatik & \omterm{def:b1:body-plans-tissues:segments}{rangka luar} & \omterm{def:b1:body-plans-tissues:segments}{rangka dalam} \\
usus & satu bukaan & mulut ke anus & mulut ke anus & mulut ke anus \\
peredaran & tidak ada & tertutup & terbuka & tertutup, berjantung \\
sistem saraf & jala & tali perut & tali perut & tali punggung \\
\hline
\end{tabular}
\end{center}
Pola-pola ini bukan anak tangga sebuah tangga: masing-masing adalah cara
yang lengkap dan berhasil untuk menjadi seekor hewan, dan udang karang
serta katak adalah sezaman, bukan leluhur satu sama lain.
\end{proposition}

\begin{example}[Membaca seekor hewan dari pola tubuhnya]\label{ex:b1:body-plans-tissues:reading}
Seekor hewan berujung kepala, berusus tembus, bertubuh gelang-gelang
serupa dan tanpa bagian keras: selomata \omterm{def:b1:body-plans-tissues:proto}{protostomia} yang bergerak dengan
\omterm{def:b1:body-plans-tissues:segments}{rangka hidrostatik}, dengan cairan \omterm{def:b1:body-plans-tissues:layers}{selom} tiap segmen ditekan bergiliran
oleh otot melingkar dan otot memanjang --- seekor cacing tanah yang
menggali. Seekor hewan berkutikula berbuku dengan segmen yang
terkhususkan: seekor artropoda, yang pertumbuhannya harus melewati
pergantian kulit. Bersimetri bilateral, dengan blok otot \omterm{def:b1:body-plans-tissues:segments}{bersegmen}
mengelilingi sumbu punggung berupa \omterm{def:b1:flowering-plant-organization:organs}{ruas} tulang belakang: seekor
vertebrata, yang \omterm{def:b1:body-plans-tissues:segments}{rangka dalamnya} tumbuh sinambung.
\end{example}

\section{Keempat jenis jaringan}

\begin{definition}[Jaringan]\label{def:b1:body-plans-tissues:tissue}
Yang disebut \emph{jaringan}\index{jaringan} adalah sekelompok sel yang
serupa struktur dan asal usulnya, beserta bahan ekstrasel yang
dihasilkannya, yang tertata untuk menjalankan satu fungsi bersama.
Jaringan hewan terbagi menjadi empat jenis: \omterm{def:b1:body-plans-tissues:epithelium}{epitel}, ikat, otot, saraf.
Setiap \omterm{def:b1:mammal-organization:organ}{organ} adalah gabungan sekurang-kurangnya dua di antaranya.
\end{definition}

\begin{definition}[Jaringan epitel]\label{def:b1:body-plans-tissues:epithelium}
Yang disebut \emph{epitel}\index{epitel} adalah selembar sel yang
tersusun rapat menyelimuti sebuah permukaan atau melapisi sebuah rongga,
yang bertumpu pada selapis tipis \omterm{def:b1:body-plans-tissues:connective}{matriks ekstrasel}, yaitu \emph{lamina
basal}\index{lamina basal}. Selnya \emph{terkutub}\index{kepolaran sel}:
sisi apikalnya menghadap dunia luar atau lumen, sisi basalnya menghadap
lamina, dan keduanya membawa protein yang berbeda. Sel itu disatukan
\emph{sambungan sel}\index{sambungan sel}: \emph{sambungan rapat} dekat
puncaknya, yang menutup ruang antarsel sehingga tak ada yang melintasi
lembar itu kecuali lewat selnya; \emph{sambungan lekat} dan
\emph{desmosom}, yang mengikat sel secara mekanis; \emph{sambungan
celah}, yaitu saluran yang meloloskan ion dan molekul kecil dari sel ke
sel. Epitel digolongkan menurut banyaknya lapisan (selapis, berlapis)
dan bentuk selnya (pipih, kubus, silindris), dan mencakup kelenjar,
yaitu sel epitel yang terkhususkan dalam sekresi. Epitel menyelimuti,
melindungi, menyerap dan menyekresi.
\end{definition}

\begin{definition}[Jaringan ikat]\label{def:b1:body-plans-tissues:connective}
Yang disebut \emph{jaringan ikat}\index{jaringan ikat} tersusun dari sel
yang tersebar di dalam \emph{matriks ekstrasel}\index{matriks ekstrasel}
melimpah yang disekresikannya: serabut protein ---
\emph{kolagen}\index{kolagen} (kekuatan tarik; seperempat seluruh
protein seekor mamalia), \emph{elastin}\index{elastin} (kelentingan) ---
yang terbenam dalam \emph{bahan dasar} berair dari polisakarida.
Perbandingannya menetapkan macamnya: \omterm{def:b1:body-plans-tissues:tissue}{jaringan} ikat longgar di bawah
\omterm{def:b1:body-plans-tissues:epithelium}{epitel} dan di sekeliling \omterm{def:b1:mammal-organization:organ}{organ} (fibroblas, sedikit serabut, banyak bahan
dasar); \omterm{def:b1:body-plans-tissues:tissue}{jaringan} ikat padat pada tendon dan ligamen (kolagen sejajar);
\omterm{def:b1:body-plans-tissues:tissue}{jaringan} lemak (sel penimbun lemak); tulang rawan (gel padat dari
kolagen dan polisakarida, tanpa pembuluh); tulang (kolagen yang
dimineralkan dengan kalsium fosfat, sel hidup di dalam rongga,
pembuluh); dan darah, yang matriksnya adalah plasma. \omterm{def:b1:body-plans-tissues:tissue}{Jaringan} ikat
mengikat, menopang, menimbun, mengangkut dan mempertahankan.
\end{definition}

\begin{definition}[Jaringan otot]\label{def:b1:body-plans-tissues:muscle}
Yang disebut \emph{jaringan otot}\index{jaringan otot} tersusun dari sel
memanjang yang padat berisi filamen protein yang dapat berkerut (aktin
dan miosin). Ada tiga macam: \emph{otot rangka}, berupa serabut panjang
berinti banyak yang memperlihatkan lurik melintang, melekat pada rangka
dan berada di bawah kendali sadar; \emph{otot jantung}, berupa sel
berlurik yang bercabang dan tersambung ujung ke ujung oleh sambungan
yang meneruskan denyut dari sel ke sel, berkerut menurut iramanya
sendiri; \emph{otot polos}, berupa sel berbentuk gelendong tanpa lurik
pada dinding \omterm{def:b1:mammal-organization:organ}{organ} berongga (usus, pembuluh, rahim), yang berkerut
lambat dan tanpa kehendak. Mekanisme pengerutannya dibahas dalam jilid
Tahun ke-2.
\end{definition}

\begin{definition}[Jaringan saraf]\label{def:b1:body-plans-tissues:nervous}
Yang disebut \emph{jaringan saraf}\index{jaringan saraf} tersusun dari
\emph{neuron}\index{neuron}, yaitu sel bertubuh dengan juluran panjang
(dendrit yang menerima isyarat, sebuah akson yang membawanya pergi) yang
menghantarkan isyarat listrik dan meneruskannya ke sel lain pada
sinapsis, serta dari \emph{sel glia}\index{sel glia}, yang beberapa kali
lebih banyak jumlahnya dan menopang, menyekat serta memberi makan
neuronnya. \omterm{def:b1:body-plans-tissues:tissue}{Jaringan} saraf mengindera, memadukan dan memerintah.
\end{definition}

\begin{example}[Asal usul jaringan]\label{ex:b1:body-plans-tissues:origin}
\omterm{def:b1:body-plans-tissues:epithelium}{Epitel} berasal dari ketiga \omterm{def:b1:body-plans-tissues:layers}{lapisan lembaga} (kulit dari \omterm{def:b1:body-plans-tissues:layers}{ektoderm},
pelapis usus dari \omterm{def:b1:body-plans-tissues:layers}{endoderm}, pelapis pembuluh dan \omterm{def:b1:body-plans-tissues:layers}{selom} dari \omterm{def:b1:body-plans-tissues:layers}{mesoderm});
\omterm{def:b1:body-plans-tissues:connective}{jaringan ikat}, otot dan darah dari \omterm{def:b1:body-plans-tissues:layers}{mesoderm}; \omterm{def:b1:body-plans-tissues:nervous}{jaringan saraf} dari
\omterm{def:b1:body-plans-tissues:layers}{ektoderm}. Kulit katak adalah \omterm{def:b1:body-plans-tissues:epithelium}{epitel} ektodermal di atas \omterm{def:b1:body-plans-tissues:connective}{jaringan ikat}
mesodermal; ususnya, \omterm{def:b1:body-plans-tissues:epithelium}{epitel} endodermal yang terbungkus otot mesodermal;
otaknya, \omterm{def:b1:body-plans-tissues:layers}{ektoderm} yang terlipat ke dalam.
\end{example}

\section{Dari jaringan menjadi organ: dinding usus}

\begin{proposition}[Keempat lapis dinding usus]\label{prop:b1:body-plans-tissues:gutwall}
\index{dinding usus}
Dari lumen ke luar, dinding saluran pencernaan vertebrata dibangun dari
empat lapis sepusat, masing-masing berupa gabungan \omterm{def:b1:body-plans-tissues:tissue}{jaringan}:
\emph{mukosa} (sebuah \omterm{def:b1:body-plans-tissues:epithelium}{epitel} --- penyerap dan penyekresi --- di atas
\omterm{def:b1:body-plans-tissues:connective}{jaringan ikat} longgar yang membawa pembuluh darah dan pembuluh limfa,
dengan selembar tipis otot polos di bawahnya); \emph{submukosa}
(\omterm{def:b1:body-plans-tissues:connective}{jaringan ikat} padat dengan pembuluh yang lebih besar dan sebuah jala
saraf); \emph{muskularis} (dua lapis otot polos, melingkar di dalam dan
memanjang di luar, dengan sebuah jala saraf di antaranya, yang
menghasilkan gerak pengadukan dan pendorongan); dan \emph{serosa}
(selapis tipis \omterm{def:b1:body-plans-tissues:connective}{jaringan ikat} yang diselimuti \omterm{def:b1:body-plans-tissues:epithelium}{epitel} \omterm{def:b1:body-plans-tissues:layers}{selom}). Keempat
lapis yang sama membentang dari kerongkongan sampai rektum; yang berubah
adalah mukosanya --- terlipat menjadi vili di usus halus, berkelenjar di
lambung.
\end{proposition}

\begin{method}[Membaca sebuah sayatan histologi]\label{met:b1:body-plans-tissues:histology}
\begin{enumerate}
  \item Carilah permukaan bebas atau lumennya: \omterm{def:b1:body-plans-tissues:epithelium}{epitel} yang melapisinya
        adalah lapis pertama. Hitunglah barisan intinya (selapis atau
        berlapis) dan perhatikan bentuk selnya.
  \item Di bawah \omterm{def:b1:body-plans-tissues:epithelium}{epitel} carilah zona pucat berserabut dengan inti yang
        tersebar dan pembuluh: \omterm{def:b1:body-plans-tissues:connective}{jaringan ikat}.
  \item Otot tampak sebagai berkas padat sel memanjang: berlurik dengan
        inti di tepi (rangka), berlurik dan bercabang (jantung), tanpa
        lurik dengan satu inti di tengah (polos). Perhatikan arah
        serabutnya terhadap sayatan.
  \item \omterm{def:b1:body-plans-tissues:nervous}{Jaringan saraf} pada dinding sebuah \omterm{def:b1:mammal-organization:organ}{organ} tampak sebagai
        gerombolan tubuh \omterm{def:b1:body-plans-tissues:nervous}{neuron} besar yang pucat di antara lapisan
        ototnya.
  \item Namailah \omterm{def:b1:mammal-organization:organ}{organnya} dari urutan lapisannya dan pengkhususan
        mukosanya (vili, kelenjar, keratin).
\end{enumerate}
\end{method}

\begin{example}[Vilus sebagai permukaan pertukaran]\label{ex:b1:body-plans-tissues:villus}
Sebuah vilus usus halus manusia adalah jari setinggi \qty{0.6}{mm} dan
selebar \qty{0.15}{mm}, yang diselimuti selapis sel silindris yang tiap
sisi apikalnya membawa sekitar \qty{1500}{} mikrovili. Vili
melipatgandakan permukaan tabung itu sekitar tujuh kali dan mikrovili
dua puluh kali lagi: \qty{0.6}{m^2} sebuah tabung polos menjadi puluhan
meter persegi \omterm{def:b1:body-plans-tissues:epithelium}{epitel} penyerap setebal satu sel, yang tiap selnya berada
dalam beberapa mikrometer dari sebuah kapiler dan sebuah pembuluh limfa
di dalam teras ikat vilusnya (\cref{ch:b1:digestion-absorption}). \omterm{def:b1:body-plans-tissues:epithelium}{Epitel}
itu diperbarui dari kelenjar di pangkalnya setiap empat hari: lembarnya,
sambungannya dan kekutubannya dibangun ulang tanpa henti.
\end{example}

\begin{remark}[Jaringan kedua organisme model]\label{rem:b1:body-plans-tissues:plants}
Ketiga sistem \omterm{def:b1:body-plans-tissues:tissue}{jaringan} tumbuhan pada
\cref{ch:b1:flowering-plant-organization} tidak memiliki padanan yang
persis di sini. \omterm{def:b1:flowering-plant-organization:tissues}{Epidermisnya} adalah \omterm{def:b1:body-plans-tissues:epithelium}{epitel} dalam fungsinya tetapi selnya
dipegang dinding, bukan sambungan; \omterm{def:b1:flowering-plant-organization:tissues}{jaringan dasarnya} sekaligus penimbun,
penopang dan pelaku fotosintesis; ia tidak memiliki otot maupun saraf.
Keempat \omterm{def:b1:body-plans-tissues:tissue}{jaringan} hewan adalah perkakas sebuah tubuh yang bergerak dan
mengindera; ketiga \omterm{def:b1:body-plans-tissues:tissue}{jaringan} tumbuhan, perkakas sebuah tubuh yang tumbuh.
\end{remark}

\section{Soal: Dinding Usus Halus}

\begin{problem}[{Soal akhir pekan --- sebuah seri histologi usus halus
manusia yang dibaca dari tabungnya sampai ke mikrovilusnya: lapisan,
vili, sel, sambungan dan pembaruan, berakhir pada permukaan penyerap
sebuah vilus tunggal}]
\label{pb:b1:body-plans-tissues:1}
Usus halus manusia adalah sebuah tabung sepanjang \qty{6}{m} dan
bergaris tengah dalam \qty{3}{cm}. Lipatan melingkar dindingnya
melipattigakan permukaan tabung polos itu. Mukosanya membawa
\qty{25}{} vilus tiap milimeter persegi permukaan yang terlipat,
masing-masing berupa tabung setinggi \qty{0.6}{mm} dan bergaris tengah
\qty{0.15}{mm}. Vili itu diselimuti sel silindris yang sisi apikalnya
berupa persegi bersisi \qty{5}{\micro m}; tiap sisi apikal membawa
\qty{1500}{} mikrovilus, yaitu tabung sepanjang \qty{1}{\micro m} dan
bergaris tengah \qty{0.1}{\micro m}. \omterm{def:b1:body-plans-tissues:epithelium}{Epitelnya} diperbarui setiap
\qty{4}{} hari.

\medskip
\noindent\textbf{Bagian I --- Lapisannya.}
Sebuah sayatan melintang diperiksa di bawah mikroskop.
\begin{enumerate}
  \item Sebutkan keempat lapis dindingnya dari lumen ke luar dan jenis
        \omterm{def:b1:body-plans-tissues:tissue}{jaringan} pada masing-masing.
  \item Sayatan itu memperlihatkan vili. Kawasan saluran pencernaan
        yang mana itu? Apa yang akan diperlihatkan sayatan lambung
        sebagai gantinya?
  \item Di antara kedua lapis otot muskularisnya terdapat gerombolan
        sel besar yang pucat. Apakah itu, dan apa yang dikendalikannya?
  \item Jelaskan mengapa kedua lapis otot itu membentang pada arah yang
        saling tegak lurus.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Permukaan.}
\begin{enumerate}[resume]
  \item Hitunglah permukaan dalam tabung polosnya.
  \item Hitunglah permukaannya setelah lipatan melingkarnya.
  \item Hitunglah permukaan samping satu vilus (sebagai tabung, dengan
        mengabaikan ujungnya).
  \item Hitunglah permukaan vilus tiap milimeter persegi mukosa yang
        terlipat dan faktor pelipatgandaan permukaan oleh vili itu.
  \item Hitunglah permukaannya setelah vilinya.
  \item Hitunglah permukaan samping satu mikrovilus dan \qty{1500}{}
        mikrovilus pada satu sel.
  \item Hitunglah faktor pelipatgandaan permukaan apikal sebuah sel
        oleh mikrovilinya.
  \item Hitunglah total permukaan penyerap usus halusnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Sel.}
\begin{enumerate}[resume]
  \item Hitunglah banyaknya sel \omterm{def:b1:body-plans-tissues:epithelium}{epitel} yang menyelimuti satu vilus.
  \item Hitunglah banyaknya vilus dalam usus halus dan banyaknya sel
        \omterm{def:b1:body-plans-tissues:epithelium}{epitel} yang menyelimutinya.
  \item \omterm{def:b1:body-plans-tissues:epithelium}{Epitelnya} diperbarui setiap \qty{4}{} hari. Berapa banyak sel
        yang dilepas ke dalam lumen tiap hari, dan tiap sekon?
  \item Di mana sel penggantinya dihasilkan, dan apa yang dikatakan hal
        ini tentang letak sel yang membelah terhadap sel yang menyerap?
  \item Sebuah sel \omterm{def:b1:body-plans-tissues:epithelium}{epitel} setinggi \qty{25}{\micro m}. Hitunglah volume
        \omterm{def:b1:body-plans-tissues:epithelium}{epitel} dalam usus halus dan massanya (massa jenis
        \qty{1.05}{g/cm^3}).
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Sambungan, matriks dan vilus.}
\begin{enumerate}[resume]
  \item Tiap sel ditutup terhadap tetangganya oleh sebuah sambungan
        rapat yang mengelilingi puncaknya. Hitunglah panjang sambungan
        rapat tiap sel dan panjang totalnya dalam usus halus.
  \item Jelaskan mengapa sebuah zat di dalam lumen hanya dapat mencapai
        darah dengan melintasi dua membran sel.
  \item Teras vilusnya adalah \omterm{def:b1:body-plans-tissues:connective}{jaringan ikat} longgar dengan jala kapiler
        dan sebuah pembuluh limfa di pusatnya. Sebutkan kedua macam
        serabut ekstrasel yang dikandungnya dan sel yang
        menyekresikannya.
  \item Sebutkan \omterm{def:b1:body-plans-tissues:layers}{lapisan lembaga} asal \omterm{def:b1:body-plans-tissues:epithelium}{epitelnya} dan asal terasnya.
  \item Vilus itu juga mengandung seuntai otot polos yang membuatnya
        memendek berirama. Ajukan fungsinya.
  \item Sebuah penyakit meratakan vilinya (permukaannya kembali menjadi
        permukaan tabung yang terlipat). Berapa kali lipat permukaan
        penyerapnya turun, dan apa akibatnya?
  \item Hitunglah permukaan penyerap satu vilus beserta mikrovilinya,
        dalam milimeter persegi.
  \item Nyatakan hasilnya: permukaan penyerap sebuah vilus tunggal, dan
        banyaknya vilus dalam usus halus.
\end{enumerate}
\end{problem}
