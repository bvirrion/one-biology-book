\section*{Chapter \ref{ch:g11:cell-cycle-mitosis} --- The Cell Cycle and Mitosis}

\begin{solution}{exo:g11:cell-cycle-mitosis:1}
$\mathrm{G_1}$: growth; S: \omterm{def:g11:dna-replication:replication}{DNA replication}; $\mathrm{G_2}$:
preparation for division; M: \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} (division of the \omterm{def:g10:cells-common-unit:organelle}{nucleus}) and
cytokinesis (division of the \omterm{def:g10:cells-common-unit:cell}{cytoplasm}).
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:2}
A \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid} is one of the two identical copies of a replicated
\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}; the \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromere} is the region where the two are held
together. A \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} has two \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} from the end of S until
anaphase.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:3}
Prophase: \omterm{def:g10:universal-dna:information}{chromosomes} condense, nuclear envelope breaks down.
Metaphase: \omterm{def:g10:universal-dna:information}{chromosomes} align on the equator. Anaphase: \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}
separate and move to the poles. Telophase: nuclear envelopes re-form,
then cytokinesis.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:4}
In S phase, half-way through replication.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:5}
46 \omterm{def:g10:universal-dna:information}{chromosomes}, 92 \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:6}
$\mathrm{G_1}$ \qty{11}{h}, S \qty{8}{h}, $\mathrm{G_2}$ \qty{4}{h},
M \qty{1}{h}. A \omterm{def:g10:cells-common-unit:cell}{cell} with $2q$ is in $\mathrm{G_2}$ or in \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}
before the end of anaphase.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:7}
Metaphase: 40 \omterm{def:g10:universal-dna:information}{chromosomes}, 80 \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}. Anaphase: 80 \omterm{def:g10:universal-dna:information}{chromosomes}
(single \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid}), 40 to each pole. Daughter: 40 \omterm{def:g10:universal-dna:information}{chromosomes}, 40
\omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:8}
Mitotic index $80/800 = 10\%$; \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} \qty{2.4}{h}. Prophase
$48/800 \times 24 = \qty{1.44}{h}$; metaphase \qty{0.48}{h} (29 min);
anaphase \qty{0.18}{h} (11 min); telophase \qty{0.3}{h} (18 min).
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:9}
Without a spindle the \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} cannot be pulled apart: the \omterm{def:g10:cells-common-unit:cell}{cell} is
stuck in metaphase with its \omterm{def:g10:universal-dna:information}{chromosomes} condensed and fully visible,
each with two \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}. That is exactly the state in which
\omterm{def:g10:universal-dna:information}{chromosomes} are photographed for a \omterm{def:g11:cell-cycle-mitosis:chromosome}{karyotype}.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:10}
Two metres of tangled thread cannot be sorted; condensed into rods a
few micrometres long, the 46 \omterm{def:g10:universal-dna:information}{chromosomes} can be lined up and pulled
apart without breaking or knotting. They uncoil afterwards because
\omterm{def:g10:universal-dna:gene}{genes} can only be read from the loose form.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:11}
Ten divisions: $2^{10} = 1024$ \omterm{def:g10:cells-common-unit:cell}{cells}. No: the early embryo does not
grow between divisions, and its cycles are almost all S and M, with
hardly any $\mathrm{G_1}$.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:12}
One daughter has 47 \omterm{def:g10:universal-dna:information}{chromosomes} (one extra), the other 45 (one
missing); their \omterm{def:g10:universal-dna:information}{DNA} is $q$ plus or minus one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}'s worth. In a
body \omterm{def:g10:cells-common-unit:cell}{cell} the error affects one \omterm{def:g10:cells-common-unit:cell}{cell} among trillions, which usually
dies or is eliminated; but if the \omterm{def:g10:cells-common-unit:cell}{cell} survives and divides, and the
lost or gained \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} deregulates its growth, a tumour can start.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:13}
About $\num{2e11}$ divisions per day, i.e. $\num{2e11}$ \omterm{def:g10:cells-common-unit:cell}{cells} cycling
at any time (each yields one new \omterm{def:g10:cells-common-unit:cell}{cell} per day). Blocking \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} stops
the supply; red \omterm{def:g10:cells-common-unit:cell}{cells} live 120 days but some white \omterm{def:g10:cells-common-unit:cell}{cells} only days, so
within a week the \omterm{def:g10:cells-common-unit:cell}{white-cell} count collapses and infections follow ---
the classic side effect of anti-cancer drugs.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:14}
In interphase the \omterm{def:g10:universal-dna:information}{chromosomes} are uncoiled and invisible; in anaphase
they are moving and mixed. In metaphase they are fully condensed,
separate, still made of two \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} and lying in one plane: each
can be identified by its size and shape.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:15}
Liver \omterm{def:g10:cells-common-unit:cell}{cells} resting in $\mathrm{G_1}$ re-entered the cycle, passed
through S, $\mathrm{G_2}$ and \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} repeatedly until the mass was
restored, then returned to rest. On day 0 almost all \omterm{def:g10:cells-common-unit:cell}{cells} held $q$;
on day 3 many held between $q$ and $2q$ (in S) or $2q$ (in
$\mathrm{G_2}$, \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}).
\end{solution}

\begin{solution}{pb:g11:cell-cycle-mitosis:1}
\textbf{1.} $100/1000 = 10\%$.

\textbf{2.} $0.10 \times 20 = \qty{2}{h}$.

\textbf{3.} Prophase $60/1000 \times 20 = \qty{1.2}{h}$ (72 min);
metaphase $0.36$ h (22 min); anaphase $0.16$ h (about 10 min);
telophase $0.28$ h (17 min).

\textbf{4.} Anaphase. The \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} are already condensed and
attached; the movement is a pull of a few micrometres by fibres that
shorten in minutes.

\textbf{5.} Rare stages are seen in few \omterm{def:g10:cells-common-unit:cell}{cells}, and a small count gives
a large error; and only the growing zone cycles --- elsewhere the
index would be zero and dilute the figures.

\textbf{6.} 16 \omterm{def:g10:universal-dna:information}{chromosomes}, 32 \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids}.

\textbf{7.} 32 \omterm{def:g10:universal-dna:information}{chromosomes} moving, 16 to each pole.

\textbf{8.} Prophase $2q$; anaphase $2q$ in the whole \omterm{def:g10:cells-common-unit:cell}{cell}; telophase
$q$ per \omterm{def:g10:cells-common-unit:organelle}{nucleus}.

\textbf{9.} S is $7/19$ of interphase: about $900 \times 7/19 \approx
330$ \omterm{def:g10:cells-common-unit:cell}{cells}.

\textbf{10.} Anaphase: the \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromeres} have just split and the 16
\omterm{def:g10:universal-dna:information}{chromosomes}' 32 \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatids} are now 32 \omterm{def:g10:universal-dna:information}{chromosomes} on their way to the
poles.

\textbf{11.} $\num{20000}/20 = 1000$ new \omterm{def:g10:cells-common-unit:cell}{cells} per hour.

\textbf{12.} 1000 \omterm{def:g10:cells-common-unit:cell}{cells} per hour across the whole zone; in one file the
zone is perhaps a hundred \omterm{def:g10:cells-common-unit:cell}{cells} wide in each direction, so roughly
$\num{24000}/(100 \times 100) \approx 2$--3 \omterm{def:g10:cells-common-unit:cell}{cells} per file per day,
each \qty{100}{\micro m}: a few tenths of a millimetre. The observed
\qty{1}{mm} means the zone is narrower than assumed or the \omterm{def:g10:cells-common-unit:cell}{cells}
elongate more; the order of magnitude is right.

\textbf{13.} One origin copies $2 \times 50 \times 7 \times 3600 =
\num{2.5e6}$ base pairs; $\num{1.6e10}/\num{2.5e6} \approx 6400$
origins at least.

\textbf{14.} Every stage lasts longer, so fewer \omterm{def:g10:cells-common-unit:cell}{cells} complete the
cycle per day; if all phases slow equally the proportions, and hence
the mitotic index, do not change.

\textbf{15.} \omterm{def:g10:cells-common-unit:cell}{Cells} already past S would complete $\mathrm{G_2}$ and
\omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} in the first hours; after a day no \omterm{def:g10:cells-common-unit:cell}{cell} can reach \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}, so
prophase to telophase have disappeared and only interphase \omterm{def:g10:cells-common-unit:cell}{cells}
remain, stuck in $\mathrm{G_1}$ or at the start of S.

\textbf{16.} $4/1000 \times 20 = \qty{0.08}{h}$, about 5 minutes,
against 10. Not surprising: with 4 or 8 \omterm{def:g10:cells-common-unit:cell}{cells} the count is dominated
by chance.

\textbf{17.} $12/2000 \times 20 = \qty{0.12}{h}$, about 7 minutes.

\textbf{18.} Human: $0.04 \times 24 \approx \qty{1}{h}$; garlic
\qty{2}{h}. Similar order, plant \omterm{def:g10:cells-common-unit:cell}{cells} somewhat slower.

\textbf{19.} Tumour \omterm{def:g10:cells-common-unit:cell}{cells} cycle continuously while most normal \omterm{def:g10:cells-common-unit:cell}{cells}
rest in $\mathrm{G_1}$, so a larger fraction is in \omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} at any
time. A spindle-blocking drug therefore hits tumour \omterm{def:g10:cells-common-unit:cell}{cells} more often
than normal ones --- but also the normal tissues that divide fast.

\textbf{20.} Cycle 20 hours: $\mathrm{G_1}$ 8, S 7, $\mathrm{G_2}$ 4,
\omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis} about 2 (prophase 72 min, metaphase 22, anaphase 7--10,
telophase 17). Anaphase, the shortest, is when the \omterm{def:g11:cell-cycle-mitosis:chromosome}{centromeres} split
and each pole receives one \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromatid} of each of the sixteen
\omterm{def:g10:universal-dna:information}{chromosomes}.
\end{solution}
