\chapter{La inmunidad adaptativa y la vacunación}\label{ch:g12:adaptive-immunity}

En 1796, un médico rural rascó materia de la pústula de viruela de las
vacas de una ordeñadora, la inoculó en el brazo de un niño, esperó seis
semanas y después le inoculó la viruela. El niño no enfermó. Nada en el
mundo de aquel médico explicaba por qué; al organismo, según parecía,
se le había enseñado. Dos siglos más tarde la viruela está extinguida,
y la enseñanza se entiende hasta las moléculas: un conjunto de \omterm{def:g10:cells-common-unit:cell}{células}
que llevan, entre todas, un receptor para casi cualquier molécula que
pueda existir, que se multiplican cuando aparece su molécula y que
recuerdan. Este capítulo describe ese sistema, sus dos brazos, su
memoria, las \omterm{def:g12:adaptive-immunity:vaccine}{vacunas} construidas sobre él --- y el virus que lo
destruye.

\section{Antígenos y linfocitos}

\begin{definition}[Antígeno, linfocito]\label{def:g12:adaptive-immunity:antigen}
Un \emph{antígeno}\index{antígeno} es cualquier molécula que el \omterm{def:g12:innate-immunity:immunity}{sistema
inmunitario} adaptativo puede reconocer: normalmente una \omterm{def:g11:gene-expression:protein}{proteína} o un
azúcar de un microbio, una toxina, la superficie de una \omterm{def:g10:cells-common-unit:cell}{célula} extraña
--- o, por error, una molécula propia del organismo. Las \omterm{def:g10:cells-common-unit:cell}{células} que
reconocen son los \emph{linfocitos}\index{linfocito}, glóbulos blancos
fabricados en la médula ósea y reunidos en los ganglios linfáticos, el
bazo y el tejido linfoide del intestino. Cada linfocito lleva miles de
copias de un \emph{receptor}, y cada receptor se une a un solo
antígeno. Los \emph{linfocitos B} maduran en la médula y fabricarán
\omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos}; los \emph{linfocitos T} maduran en el timo y actúan por
contacto y mediante señales.
\end{definition}

\begin{proposition}[Un repertorio hecho de antemano]\label{prop:g12:adaptive-immunity:repertoire}
El receptor de cada \omterm{def:g12:adaptive-immunity:antigen}{linfocito} se ensambla, mientras la \omterm{def:g10:cells-common-unit:cell}{célula} madura,
mediante una redistribución al azar de segmentos de \omterm{def:g10:universal-dna:gene}{genes}, distinta en
cada \omterm{def:g10:cells-common-unit:cell}{célula}. El organismo tiene así unos $10^{8}$ receptores distintos
antes de haber encontrado ningún \omterm{def:g12:adaptive-immunity:antigen}{antígeno} --- bastantes para que casi
cualquier molécula encuentre unas pocas \omterm{def:g10:cells-common-unit:cell}{células} que se le unan. Los
\omterm{def:g12:adaptive-immunity:antigen}{linfocitos} cuyo receptor se une a las moléculas propias del organismo
se eliminan durante la maduración; los demás esperan, unas pocas
\omterm{def:g10:cells-common-unit:cell}{células} por cada \omterm{def:g12:adaptive-immunity:antigen}{antígeno} posible, a que llegue su \omterm{def:g12:adaptive-immunity:antigen}{antígeno}.
\end{proposition}
\admitted

\section{La selección clonal}

\begin{proposition}[Selección y expansión]\label{prop:g12:adaptive-immunity:selection}
Cuando entra un \omterm{def:g12:adaptive-immunity:antigen}{antígeno}, solo responden los \omterm{def:g12:adaptive-immunity:antigen}{linfocitos} cuyo receptor
encaja con él: quedan \emph{seleccionados}, y se multiplican por
\omterm{def:g11:cell-cycle-mitosis:cycle}{mitosis}, a lo largo de varios días, hasta formar un clon de miles de
\omterm{def:g10:cells-common-unit:cell}{células} idénticas --- la \emph{selección clonal}\index{selección
clonal}. El clon se diferencia después: la mayoría de las \omterm{def:g10:cells-common-unit:cell}{células} se
convierten en \emph{\omterm{def:g10:cells-common-unit:cell}{células} efectoras}, que combaten al \omterm{def:g12:adaptive-immunity:antigen}{antígeno}
durante una o dos semanas y mueren; una minoría se convierte en
\emph{células de memoria}\index{célula de memoria} de vida larga, que
persisten años y que, si el \omterm{def:g12:adaptive-immunity:antigen}{antígeno} vuelve, responden más deprisa y en
un número muchísimo mayor. La respuesta es específica porque solo se
expanden los clones que encajan; es lenta la primera vez porque parten
de un puñado de \omterm{def:g10:cells-common-unit:cell}{células}; es rápida la segunda porque las \omterm{prop:g12:adaptive-immunity:selection}{células de
memoria} ya son muchas.
\end{proposition}
\begin{proof}[Pruebas experimentales]
Los \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} contra un \omterm{def:g12:adaptive-immunity:antigen}{antígeno} dado aparecen en la sangre alrededor
de una semana después de una primera inyección, suben durante dos
semanas y bajan; una segunda inyección unos meses después produce, en
dos o tres días, \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} a un nivel diez o cien veces el del primer
máximo, que persisten años. Inyectar un segundo \omterm{def:g12:adaptive-immunity:antigen}{antígeno}, sin relación
con el primero, a la vez que la segunda dosis da, para ese \omterm{def:g12:adaptive-immunity:antigen}{antígeno},
una respuesta lenta de primer tipo: la memoria es específica. Unos
\omterm{def:g12:adaptive-immunity:antigen}{linfocitos} tomados de un animal inmunizado y transferidos a otro que no
lo está le transfieren la memoria; los de un animal no inmunizado, no.
\end{proof}

\section{Dos brazos: anticuerpos y células asesinas}

\begin{proposition}[Los linfocitos B y los anticuerpos]\label{prop:g12:adaptive-immunity:antibodies}
Un clon B seleccionado se diferencia en \emph{\omterm{def:g10:cells-common-unit:cell}{células} plasmáticas},
fábricas que secretan miles de \emph{anticuerpos}\index{anticuerpo} por
segundo: copias solubles del receptor del clon, \omterm{def:g10:chemistry-of-life:families}{proteínas} en forma de Y
cuyos dos brazos se unen cada uno al \omterm{def:g12:adaptive-immunity:antigen}{antígeno}. Los \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} no
matan; \emph{marcan}. Unidos a un virus o a una toxina, le impiden
entrar en las \omterm{def:g10:cells-common-unit:cell}{células} o actuar; unidos a una bacteria, la recubren, y
ese recubrimiento es lo que mejor agarran los \omterm{def:g12:innate-immunity:phagocytosis}{fagocitos} del
\cref{ch:g12:innate-immunity}. Los \omterm{def:g12:adaptive-immunity:antigen}{antígenos} entrelazados por muchos
\omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} se agrupan en \emph{complejos inmunitarios}, que se
eliminan por \omterm{def:g12:innate-immunity:phagocytosis}{fagocitosis}. Este es el brazo contra los intrusos que están
en la sangre y en los líquidos.
\end{proposition}
\admitted

\begin{proposition}[Los linfocitos T]\label{prop:g12:adaptive-immunity:tcells}
Los \omterm{def:g12:adaptive-immunity:antigen}{linfocitos} T solo reconocen el \omterm{def:g12:adaptive-immunity:antigen}{antígeno} en forma de fragmentos
exhibidos en la superficie de otra \omterm{def:g10:cells-common-unit:cell}{célula}
(\cref{ch:g12:innate-immunity}). Hay dos clases:
\begin{itemize}
  \item Los \emph{linfocitos T colaboradores}\index{linfocito T
        colaborador} los activan las \omterm{prop:g12:innate-immunity:handover}{células dendríticas} que llevan el
        \omterm{def:g12:adaptive-immunity:antigen}{antígeno} al ganglio linfático; una vez seleccionados y
        multiplicados, secretan citocinas que autorizan y amplifican la
        respuesta de los \omterm{def:g12:adaptive-immunity:antigen}{linfocitos} B y de los demás \omterm{def:g12:adaptive-immunity:antigen}{linfocitos} T. Son
        los coordinadores; sin ellos casi no hay respuesta adaptativa.
  \item Los \emph{\omterm{def:g12:adaptive-immunity:antigen}{linfocitos} T citotóxicos} reconocen fragmentos de un
        virus (o de una \omterm{def:g11:gene-expression:protein}{proteína} tumoral) exhibidos por una \omterm{def:g10:cells-common-unit:cell}{célula}
        infectada, y matan a esa \omterm{def:g10:cells-common-unit:cell}{célula} por contacto, antes de que el
        virus que hay dentro pueda multiplicarse. Este es el brazo
        contra los intrusos escondidos dentro de las \omterm{def:g10:cells-common-unit:cell}{células}, adonde
        los \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} no llegan.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Un virus, encontrado dos veces]\label{ex:g12:adaptive-immunity:virus}
Primera gripe: el virus se multiplica durante tres días antes de que la
respuesta adaptativa esté lista; hacia el día 7, los \omterm{def:g12:adaptive-immunity:antigen}{linfocitos} T
citotóxicos van matando \omterm{def:g10:cells-common-unit:cell}{células} infectadas en las vías respiratorias y
los \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} van neutralizando al virus libre; hacia el día 10 la
persona se está recuperando, con \omterm{prop:g12:adaptive-immunity:selection}{células de memoria} de las dos clases.
La misma cepa un año después: los \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} ya presentes en la sangre
bloquean a la mayor parte del virus en la entrada, las \omterm{prop:g12:adaptive-immunity:selection}{células de
memoria} se expanden en dos días y la infección termina antes de que se
note. Una cepa distinta, con las \omterm{def:g10:chemistry-of-life:families}{proteínas} de superficie mutadas, se
escapa de los \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} --- que es por lo que la \omterm{def:g12:adaptive-immunity:vaccine}{vacuna} se rehace
todos los años.
\end{example}

\section{La vacunación}

\begin{definition}[Vacuna]\label{def:g12:adaptive-immunity:vaccine}
Una \emph{vacuna}\index{vacuna} es un preparado que provoca una
respuesta adaptativa primaria, y por tanto memoria, contra un patógeno
sin causar su enfermedad: el patógeno atenuado (sarampión,
tuberculosis), muerto (poliomielitis por inyección), reducido a una de
sus \omterm{def:g10:chemistry-of-life:families}{proteínas} o a una toxina inactivada (tétanos, hepatitis B) o, más
recientemente, a las instrucciones para que la \omterm{def:g10:cells-common-unit:cell}{célula} fabrique esa
\omterm{def:g11:gene-expression:protein}{proteína}. Un \emph{adyuvante} añadido al preparado provoca la pequeña
inflamación sin la cual las \omterm{prop:g12:innate-immunity:handover}{células dendríticas} no llevarían el
\omterm{def:g12:adaptive-immunity:antigen}{antígeno} al ganglio linfático. Las dosis de recuerdo convierten la
respuesta primaria en una secundaria y prolongan la memoria durante
décadas.
\end{definition}

\begin{proposition}[Proteger al grupo]\label{prop:g12:adaptive-immunity:herd}
Una persona vacunada está protegida; una \omterm{def:g12:selection-drift-speciation:population}{población} vacunada protege
además a sus miembros sin vacunar. Una persona infectada contagia una
enfermedad, en promedio, a $R_0$ personas más en una \omterm{def:g12:selection-drift-speciation:population}{población} sin
inmunidad (unas 15 en el sarampión, de 2 a 3 en la gripe); si una
fracción $v$ es inmune, solo pueden infectarse $R_0(1 - v)$ de esos
contactos, y la enfermedad retrocede cuando ese número baja de 1 ---
esto es, cuando
\[
v > 1 - \frac{1}{R_0}.
\]
En el sarampión, el 93\% de la \omterm{def:g12:selection-drift-speciation:population}{población} tiene que ser inmune para
proteger también a los lactantes demasiado pequeños para la \omterm{def:g12:adaptive-immunity:vaccine}{vacuna} y a
las personas que no pueden recibirla; en la gripe, alrededor del 60\%.
Esta \emph{inmunidad de grupo}\index{inmunidad de grupo} es por lo que
la vacunación es un acto colectivo.
\end{proposition}
\admitted

\section{Cuando se destruye al coordinador}

\begin{proposition}[El VIH y el sida]\label{prop:g12:adaptive-immunity:hiv}
El virus de la inmunodeficiencia humana, el VIH, infecta a los
\omterm{prop:g12:adaptive-immunity:tcells}{linfocitos T colaboradores}, y entra precisamente por el receptor que
los distingue, y se multiplica en ellos. Durante años, el \omterm{def:g12:innate-immunity:immunity}{sistema
inmunitario} mata \omterm{def:g10:cells-common-unit:cell}{células} infectadas y fabrica \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} --- la
persona es \emph{seropositiva}, y contagiosa --- mientras el virus,
que muta sin parar, se escapa de cada respuesta y va agotando poco a
poco a las \omterm{def:g10:cells-common-unit:cell}{células} colaboradoras. Cuando su recuento baja por debajo de
alrededor de una quinta parte de lo normal, la respuesta adaptativa ya
no puede coordinarse: es el \emph{sida}\index{sida}, en el que unas
infecciones que un organismo sano elimina sin enterarse --- un hongo de
la boca, un parásito leve de los pulmones --- se vuelven mortales, y
aparecen \omterm{def:g11:cancer:cancer}{cánceres} raros. Los antivirales bloquean las \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimas} del
virus, recuperan las \omterm{def:g10:cells-common-unit:cell}{células} colaboradoras y, tomados de por vida,
mantienen a raya la enfermedad y detienen la transmisión; todavía no
hay \omterm{def:g12:adaptive-immunity:vaccine}{vacuna}, ni curación.
\end{proposition}
\admitted

\begin{method}[Leer una respuesta inmunitaria]\label{met:g12:adaptive-immunity:read}
\begin{enumerate}
  \item Identifica el \omterm{def:g12:adaptive-immunity:antigen}{antígeno} y dónde está: libre en los líquidos
        (actúan los \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos}) o dentro de las \omterm{def:g10:cells-common-unit:cell}{células} (actúan los T
        citotóxicos).
  \item Lee los plazos: una semana hasta el máximo significa una
        respuesta primaria; unos pocos días, memoria.
  \item Lee la especificidad: una respuesta a un \omterm{def:g12:adaptive-immunity:antigen}{antígeno} deja intacta
        la respuesta a otro.
  \item Localiza a los \omterm{prop:g12:adaptive-immunity:tcells}{linfocitos T colaboradores}: sin ellos no pasa
        nada, que es lo que demuestra el \omterm{prop:g12:adaptive-immunity:hiv}{sida}.
  \item Para una \omterm{def:g12:adaptive-immunity:vaccine}{vacuna}, nombra la forma del \omterm{def:g12:adaptive-immunity:antigen}{antígeno}, el adyuvante, el
        número de dosis y la fracción de \omterm{def:g12:selection-drift-speciation:population}{población} necesaria.
\end{enumerate}
\end{method}

\begin{remark}[La selección, una vez más]\label{rem:g12:adaptive-immunity:selection}
El \omterm{def:g12:innate-immunity:immunity}{sistema inmunitario} adaptativo es darwiniano: una \omterm{def:g12:selection-drift-speciation:population}{población} de
\omterm{def:g10:cells-common-unit:cell}{células} con receptores al azar, de las que el ambiente --- el \omterm{def:g12:adaptive-immunity:antigen}{antígeno}
--- selecciona unas pocas para que se multipliquen, y cuyos
descendientes heredan el receptor ganador. Ejecuta el proceso del
\cref{ch:g12:selection-drift-speciation} dentro de un solo cuerpo en
una semana y, gracias a la memoria, guarda sus resultados de por vida.
Es también la razón por la que al \omterm{def:g12:innate-immunity:immunity}{sistema inmunitario} se le puede
enseñar, que es lo que hace una \omterm{def:g12:adaptive-immunity:vaccine}{vacuna}, y por la que un virus que muta
más deprisa de lo que el sistema puede seleccionar --- el VIH, la
gripe --- es tan difícil de vencer.
\end{remark}

\section{Problema: El noventa y tres por ciento}

\begin{problem}[{Problema de fin de semana --- una campaña de
vacunación calculada: las respuestas a dos dosis, la aritmética de la
inmunidad de grupo, un brote de sarampión modelizado y el virus que
deshace el sistema}]
\label{pb:g12:adaptive-immunity:1}
El sarampión tiene $R_0 = 15$. Un pueblo de 50\,000 habitantes tiene al
90\% de su \omterm{def:g12:selection-drift-speciation:population}{población} inmune por vacunación o por haber pasado la
enfermedad; 1500 son lactantes de menos de un año, demasiado pequeños
para la \omterm{def:g12:adaptive-immunity:vaccine}{vacuna}.

\medskip
\noindent\textbf{Parte I --- La respuesta de una persona.}
Un niño recibe la primera dosis a los 12 meses y la segunda a los 18.
\begin{enumerate}
  \item Describe la curva de \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos} tras la primera dosis:
        retraso, máximo, descenso.
  \item Descríbela tras la segunda dosis y explica la diferencia con la
        \omterm{prop:g12:adaptive-immunity:selection}{selección clonal}.
  \item El virus de la \omterm{def:g12:adaptive-immunity:vaccine}{vacuna} está atenuado, pero vivo. ¿Por qué se usa
        un virus vivo, y qué brazo de la respuesta entrena que no
        entrenaría uno muerto?
  \item Diez años después, la niña se encuentra con el sarampión. ¿Qué
        pasa en los tres primeros días, y por qué no enferma?
  \item Su primo sin vacunar se encuentra con el mismo virus. Describe
        sus diez primeros días, y qué le queda después.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El umbral.}
\begin{enumerate}[resume]
  \item Calcula la fracción de la \omterm{def:g12:selection-drift-speciation:population}{población} que tiene que ser inmune
        para que el sarampión deje de propagarse.
  \item Con un 90\% inmune, ¿a cuántas personas puede contagiar en
        promedio una persona infectada? ¿Está protegido el pueblo?
  \item ¿Cuántas personas del pueblo no son inmunes, y cuántas de ellas
        son los lactantes?
  \item Si la cobertura subiera al 95\%, ¿cuántas personas susceptibles
        quedarían, y qué produciría en promedio un caso?
  \item Explica en una frase por qué la seguridad de los lactantes
        depende de las decisiones de los adultos.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Un brote.}
Un viajero mete el sarampión en el pueblo con un 90\% de inmunidad.
\begin{enumerate}[resume]
  \item Toma que cada caso contagia a $15 \times 0.10 = 1.5$ personas,
        en generaciones sucesivas de 12 días. ¿Cuántos casos hay en la
        tercera generación? ¿Y en la sexta?
  \item ¿Cuántos casos hay en total al cabo de seis generaciones?
  \item De las 5000 personas susceptibles, el 30\% son lactantes.
        Estima el número de lactantes entre los casos al cabo de seis
        generaciones, si los casos caen por igual entre los
        susceptibles.
  \item Cada generación de casos deja supervivientes inmunes: explica
        por qué el brote se frena solo, y más o menos cuándo.
  \item Repite la pregunta 11 para un pueblo con un 95\% de inmunidad.
        Compara los dos pueblos.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El sistema deshecho.}
Un paciente infectado por el VIH, sin tratar, pierde \omterm{prop:g12:adaptive-immunity:tcells}{linfocitos T
colaboradores} de 1000 a 200 por microlitro en 8 años.
\begin{enumerate}[resume]
  \item Calcula la pérdida media por año y por mes.
  \item A 200 por microlitro empieza el \omterm{prop:g12:adaptive-immunity:hiv}{sida}. Explica, con la figura de
        la organización, por qué se ven afectados tanto los \omterm{prop:g12:adaptive-immunity:antibodies}{anticuerpos}
        como las \omterm{def:g10:cells-common-unit:cell}{células} asesinas, aunque solo estén infectadas las
        \omterm{def:g10:cells-common-unit:cell}{células} colaboradoras.
  \item Las \omterm{prop:g12:adaptive-immunity:selection}{células de memoria} del sarampión del paciente están
        intactas. ¿Sería peligrosa ahora una exposición al sarampión?
        Explícalo.
  \item Un tratamiento antiviral empezado en el año 8 devuelve el
        recuento a 600 en dos años. ¿Qué muestra esto sobre las \omterm{prop:g12:adaptive-immunity:selection}{células
        de memoria} y sobre el repertorio, y por qué el tratamiento
        tiene que continuar?
  \item Enuncia el resultado: la cobertura que protege a los lactantes
        del pueblo, el número de casos que un solo viajero siembra en
        seis generaciones con un 90\% y la única \omterm{def:g10:cells-common-unit:cell}{célula} cuya pérdida
        derriba el sistema adaptativo entero.
\end{enumerate}
\end{problem}
