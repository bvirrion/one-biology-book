\chapter{الهضم والمغذيات}\label{ch:g7:digestion-and-nutrients}

تتبّع \cref{ch:g4:journey-of-food} لقمة الخبز نازلةً في الأنبوب
ووعد بأن ``\omterm{def:g4:journey-of-food:digestion}{العصارات الهاضمة} تفكّكها''. وهذه السنة نفتح أبواب
المشغل: ماذا تفعل العصارات بالضبط، وما \omterm{def:g7:digestion-and-nutrients:families}{المغذيات} التي تنتجها، وكيف
يدبّر جدار \omterm{def:g4:journey-of-food:tract}{المعي الدقيق} أعظم عبور حدودي في الجسم؟ لقد عادت لقمة
الخبز --- تحت أدوات أفضل.

\section{العصارات في العمل}

\begin{proposition}[الهضم تفكيك كيميائي]\label{prop:g7:digestion-and-nutrients:chemical}
\omterm{def:g5:human-reproduction:pregnancy}{المضغ} والخضّ لا يفتّتان الغذاء إلا \emph{أصغر}؛ أما التفكيك
الحقيقي فكيميائي. وتحوي \emph{العصارات
الهاضمة}\index{عصارة هاضمة} --- \omterm{def:g4:journey-of-food:tract}{اللعاب}، وعصارة \omterm{def:g4:journey-of-food:tract}{المعدة}، والعصارات
المصبوبة في \omterm{def:g4:journey-of-food:tract}{المعي الدقيق} --- موادّ عاملة تقطع مكوّنات الغذاء
الكبيرة إلى صغيرة: النشا إلى سكريات، ومادة بناء اللحم \omterm{def:g2:animals-grow-up:born}{والبيض} إلى
وحداتها الصغيرة، والدهون إلى وحداتها. ولكل عصارة اختصاصاتها،
وكل واحدة تعمل أحسن ما تعمل في \omterm{def:g6:exploring-our-environment:conditions}{شروط} محطتها هي.
\end{proposition}
\admitted

\begin{example}[نسخة المنضدة]\label{ex:g7:digestion-and-nutrients:bench}
التجربة الكلاسيكية تجري \omterm{def:g4:journey-of-food:digestion}{الهضم} في زجاج. أنبوبان من عجينة نشا مطبوخ
عند دفء الجسم: أحدهما ينال لعابًا طازجًا، والآخر لعابًا مغليًّا
(فاسدًا). وبعد ساعة يجد اختبار النشا الأنبوب الثاني على حاله ---
ما زال نشا --- بينما تحوّلت عجينة الأنبوب الأول إلى سكريات، وهي
\omterm{def:g1:the-five-senses:sense}{عين} حلاوة \cref{ex:g4:journey-of-food:sweet}، منتَجةً بلا أي فم.
فالعصارة وحدها قامت بالتفكيك --- والشاهد المغلي يبين أن مادتها
العاملة تُتلفها الحرارة، كما تُتلف آلةً رقيقة.
\end{example}

\begin{remark}[ولماذا التفكيك أصلًا؟]\label{rem:g7:digestion-and-nutrients:why}
مكوّنات الغذاء الكبيرة لا تستطيع أن تعبر جدار المعي أكثر مما تستطيع
خزانة أن تمرّ في فتحة بريد. والجسم لا يريدها كاملة: بل يريد
\emph{أجزاءً} --- وحدات صغيرة يستطيع أن يعيد بناءها مادةً خاصة به
(\cref{prop:g6:matter-from-food:transform}). \omterm{def:g4:journey-of-food:digestion}{فالهضم} تفكيك للنقل
\emph{ولإعادة التركيب}: من أثاث إلى قطع مسطحة، في كل وجبة.
\end{remark}

\section{المغذيات}

\begin{definition}[أسر المغذيات]\label{def:g7:digestion-and-nutrients:families}
يعطي التفكيك \emph{المغذيات}\index{مغذيات} --- وهي المكوّنات
الصغيرة الجاهزة للدم:
\begin{itemize}
  \item \emph{السكريات} (\omterm{prop:g7:organs-at-work:bill}{والغلوكوز} قبل كل شيء، وهو وقود
        \cref{prop:g7:organs-at-work:bill})، من الأطعمة النشوية
        والحلوة؛
  \item ووحدات \emph{مادة البناء} الصغيرة، من اللحم والسمك
        \omterm{def:g2:animals-grow-up:born}{والبيض} والألبان والبقول --- وهي قرميد البناء في
        \cref{prop:g3:balanced-diet:jobs}؛
  \item ووحدات \emph{الدهون} --- وهي وقود كثيف ومادة بناء معًا؛
  \item مع ركّاب لا يحتاجون إلى تفكيك:
        \emph{الفيتامينات}، و\emph{\omterm{prop:g4:what-plants-need:needs}{الأملاح المعدنية}}، والماء
        نفسه.
\end{itemize}
وأسر الغذاء في \cref{def:g3:balanced-diet:families} هي في جوهرها
مورّدة لأسر المغذيات هذه --- قائمة المشتريات وراء قائمة المشتريات.
\end{definition}

\begin{example}[غداء واحد مفصَّلًا]\label{ex:g7:digestion-and-nutrients:lunch}
الغداء المتوازن في \cref{ex:g3:balanced-diet:lunch}، مقروءًا من
جديد مغذياتٍ: المعكرونة --- نشا، يُفكَّك غلوكوزًا باللتر؛ والسمك ---
مادة بناء، تُقطَّع إلى وحداتها؛ والزيت على الفاصولياء --- وحدات
دهون؛ والفاصولياء والتفاحة --- فيتامينات ومعادن وماء، مع ألياف.
ولاحظ أن الألياف هي الاستثناء الذي يبرهن على النظام: فلا عصارة
عندنا تستطيع قطعها، فلا تعبر شيئًا وتسافر أبعد --- وهي كتلة
البقايا (\cref{prop:g4:journey-of-food:absorb})، تقوم بكنسها
النافع الخاص.
\end{example}

\section{العبور الكبير}

\begin{proposition}[الامتصاص عند الزغابات]\label{prop:g7:digestion-and-nutrients:absorption}
العبور الموعود في
\cref{prop:g4:journey-of-food:absorb} يحدث عبر جدار \omterm{def:g4:journey-of-food:tract}{المعي الدقيق}
الداخلي، والجدار مبني له: فهو مطوي، ومفروش بملايين
\emph{الزغابات}\index{زغابة} --- نتوءات على شكل الأصابع بارتفاع
مليمتر، كل واحدة تلفّ أوعيتها الدموية الدقيقة. \omterm{prop:g7:digestion-and-nutrients:absorption}{والزغابات} هي سبب
بلوغ سطح العبور حجم فرش غرفة (\cref{rem:g4:journey-of-food:surface}):
طيّات على طيّات على أصابع. وتعبر \omterm{def:g7:digestion-and-nutrients:families}{المغذيات} جدرانها الرقيقة إلى \omterm{prop:g4:journey-of-food:absorb}{الدم}،
ويحملها \omterm{prop:g4:journey-of-food:absorb}{الدم} أولًا إلى عضو كبير للفرز والادّخار --- هو
\emph{الكبد}\index{كبد} --- قبل التوصيل العام.
\end{proposition}
\admitted

\begin{example}[المواصفة من جديد]\label{ex:g7:digestion-and-nutrients:spec}
كبيرة، رقيقة، رطبة، مبطنة \omterm{prop:g4:journey-of-food:absorb}{بالدم}: \omterm{prop:g7:digestion-and-nutrients:absorption}{فالزغابات} تلبّي بالضبط مواصفة
\cref{prop:g7:how-animals-breathe:spec} --- \omterm{def:g7:digestion-and-nutrients:families}{للمغذيات} بدل \omterm{def:g7:what-is-respiration:respiration}{الأكسجين}.
جيوب \omterm{def:g7:how-animals-breathe:four}{الرئتين}، \omterm{def:g1:animals-around-us:covering}{وريش} \omterm{def:g7:how-animals-breathe:four}{الغلاصم}، وزغابات المعي: فحيثما وجب على الجسم أن
ينقل كثيرًا عبر حدّ، بنى الآلة نفسها. تعرّف المواصفة مرة واحدة،
تعرفها في كل مكان.
\end{example}

\begin{method}[تتبّع مغذٍّ من طرف إلى طرف]\label{met:g7:digestion-and-nutrients:trace}
لأي لقمة، تتبّع مغذيًّا واحدًا:
\begin{enumerate}
  \item سمِّ أسرة الغذاء ومكوّناته الرئيسة؛
  \item ولاحظ، محطةً محطة، أين يبدأ تفكيكه وأين ينتهي، وأي
        العصارات عملت فيه؛
  \item واعبر معه عند \omterm{prop:g7:digestion-and-nutrients:absorption}{الزغابات}، إلى \omterm{prop:g4:journey-of-food:absorb}{الدم}، عبر \omterm{prop:g7:digestion-and-nutrients:absorption}{الكبد}؛
  \item وسلّمه: إلى \omterm{def:g5:first-look-at-cells:cell}{خلية} عاملة وقودًا
        (\cref{prop:g7:organs-at-work:bill})، أو إلى ورشة بناء
        مادةً (\cref{def:g6:matter-from-food:production}).
\end{enumerate}
\end{method}

\begin{example}[التتبّع مُجرًى على الخبز]\label{ex:g7:digestion-and-nutrients:breadtrace}
الخبز: من الأسرة النشوية، ونشا في معظمه. يبدأ التفكيك في الفم
(\omterm{def:g4:journey-of-food:tract}{اللعاب} --- وحلاوة المضغ الطويل)، ويتوقف في حمض \omterm{def:g4:journey-of-food:tract}{المعدة}، وينتهي في
عصارات \omterm{def:g4:journey-of-food:tract}{المعي الدقيق}: غلوكوز. والعبور: زغابات، فدم، فكبد ---
\omterm{prop:g7:digestion-and-nutrients:absorption}{والكبد} يودع جزءًا من \omterm{prop:g7:organs-at-work:bill}{الغلوكوز} لوقت لاحق، وهي إحدى خدماته الكثيرة.
والتسليم: \omterm{def:g3:bones-and-muscles:muscle}{عضلة} فخذ في وسط عدو تحرقه في غضون ساعة. وهكذا حُوسبت
لقمة الصف الرابع حسابًا كاملًا.
\end{example}

\section{مسألة: المعي الزجاجي}

\begin{problem}[{مسألة نهاية الأسبوع --- بناء الهضم على المنضدة،
أنبوبًا أنبوبًا}]\label{pb:g7:digestion-and-nutrients:1}
يبني الصف ``معًى زجاجيًّا'': رفًّا من الأنابيب عند دفء الجسم، كل
واحد يحاكي محطة من \omterm{def:g4:journey-of-food:tract}{الأنبوب الهضمي}. الأنبوب M (الفم): عجينة نشا مع
\omterm{def:g4:journey-of-food:tract}{لعاب} طازج. والأنبوب M$'$: عجينة نشا مع \omterm{def:g4:journey-of-food:tract}{لعاب} مغلي. والأنبوب S
(\omterm{def:g4:journey-of-food:tract}{المعدة}): مكعبات بياض \omterm{def:g2:animals-grow-up:born}{بيض} مع عصارة \omterm{def:g4:journey-of-food:tract}{معدة} من الصيدلي. والأنبوب
S$'$: المكعبات نفسها في ماء عادي. وتجري كلها ساعتين؛ ثم
الاختبارات --- للنشا، وللسكريات، ونظرة إلى المكعبات.

\medskip\noindent\textbf{القسم الأول --- التنبؤات
والنتائج.}
\begin{enumerate}
  \item تنبأ بنتائج الأنابيب الأربعة بعد ساعتين.
  \item يصير M سكريًّا، ولا يصير M$'$ كذلك. اذكر الخلاصة، ودور
        M$'$ بالضبط.
  \item مكعبات S تنكمش وتلين وتصفو؛ ومكعبات S$'$ تبقى على
        حالها. فماذا فعلت عصارة \omterm{def:g4:journey-of-food:tract}{المعدة} --- وأي أسرة من أسر
        \omterm{def:g7:digestion-and-nutrients:families}{المغذيات} تُنتَج؟
  \item لم يُمضغ أي أنبوب ولم يُخضّ. فعلامَ يبرهن الرفّ في شأن
        موضع عمل \omterm{def:g4:journey-of-food:digestion}{الهضم} الحقيقي؟
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- تحسين النموذج.}
\begin{enumerate}[resume]
  \item يقترح تلميذ إجراء الأنبوب M عند حرارة الثلاجة ``حفاظًا
        على الترتيب''. اعترض على ذلك، بنمط
        \cref{prop:g6:decomposers-and-soil:speed} مطبَّقًا على
        العصارات.
  \item ويقترح آخر أنبوبًا واحدًا كبيرًا بالعصارتين ممزوجتين.
        فأي واقعة من وقائع المحطة تلو المحطة في
        \cref{prop:g7:digestion-and-nutrients:chemical} يطمسها
        المزج؟
  \item المعي الزجاجي بلا جدار: فلا شيء يُمتصّ. فأي بنية سيحتاج
        إليها نموذج كامل بعد ذلك، وأي خاصيتين يجب أن تكونا في
        مادتها؟
  \item أضف أليافًا --- رقائق نخالة --- إلى أنبوب خامس مع كل
        عصارة متاحة. تنبأ بحاله بعد ساعتين، وبمعناه للنموذج.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- من الزجاج إلى المعي.}
\begin{enumerate}[resume]
  \item في الجسم، ستلقى سكريات M \omterm{prop:g7:digestion-and-nutrients:absorption}{الزغابات} بعد ذلك. تابع رحلتها
        في أربع محطات.
  \item عصارة \omterm{def:g4:journey-of-food:tract}{المعدة} تعمل في حمض قوي؛ وعصارات المعي تحتاج إلى
        تعديل ذلك الحمض. فعلامَ يدل هذا في شأن ``كل عصارة في
        \omterm{def:g6:exploring-our-environment:conditions}{شروط} محطتها'' --- وفي شأن ابتلاع المحطات بالترتيب؟
  \item تبيع شركة أدوية كبسولات عصارة لمن تقصّر محطاتهم عن
        عملها. اشرح، بالرفّ، لماذا كان مثل هذا العلاج ممكنًا
        أصلًا.
  \item لخّص التطبيق العملي في جملة واحدة تحوي \emph{العصارات}
        و\emph{\omterm{def:g7:digestion-and-nutrients:families}{المغذيات}} و\emph{أنابيب الشاهد}.
\end{enumerate}
\end{problem}
