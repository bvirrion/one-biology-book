\chapter{Plant Biology}\label{ch:g12:plant-biology}

A tree does not chase prey or flee a predator; it stands and still
manages water, sugar, \omterm{def:g10:scientific-biology:properties}{growth} and sex. \omterm{def:g12:plant-biology:tissues}{Vascular} plants solve transport
with two specialised pipelines, open and close stomata to balance
\omterm{def:g10:photosynthesis:def}{photosynthesis} against water loss, and steer organs with \omterm{def:g12:nerves-hormones:hormone}{hormones} such
as \omterm{def:g12:plant-biology:auxin}{auxin}. This chapter surveys tissues, \omterm{def:g12:plant-biology:xylem}{xylem} and \omterm{def:g12:plant-biology:phloem}{phloem}, stomatal
control, hormone-guided \omterm{def:g10:scientific-biology:properties}{growth}, and the outline of plant \omterm{def:g10:scientific-biology:properties}{reproduction}.

\section{Plant tissues and organs}

\begin{definition}[Plant organs]\label{def:g12:plant-biology:organs}
The three vegetative organs of a typical \omterm{def:g12:plant-biology:tissues}{vascular} plant are the
\emph{root}\index{root} (anchorage, water and mineral uptake), the
\emph{stem}\index{stem} (support, transport, sometimes \omterm{def:g10:photosynthesis:def}{photosynthesis})
and the \emph{leaf}\index{leaf} (primary photosynthetic organ).
Reproductive organs include \omterm{def:g12:plant-biology:flower}{flowers}, cones or equivalent structures.
\end{definition}

\begin{definition}[Tissue systems]\label{def:g12:plant-biology:tissues}
Three tissue systems span the plant body:
\emph{dermal}\index{dermal tissue} (epidermis, \omterm{rem:g12:plant-biology:cuticle}{cuticle}, \omterm{def:g12:plant-biology:stoma}{guard cells}),
\emph{ground}\index{ground tissue} (parenchyma, collenchyma, sclerenchyma
--- \omterm{def:g10:photosynthesis:def}{photosynthesis}, storage, support) and \emph{vascular}\index{vascular
tissue} (\omterm{def:g12:plant-biology:xylem}{xylem} and \omterm{def:g12:plant-biology:phloem}{phloem}).
\end{definition}

\begin{definition}[Meristems]\label{def:g12:plant-biology:meristem}
A \emph{meristem}\index{meristem} is a region of undifferentiated,
dividing \omterm{def:g10:the-cell:cell}{cells}. Apical meristems lengthen shoots and \omterm{def:g12:plant-biology:organs}{roots}; lateral
meristems (\omterm{def:g12:plant-biology:tissues}{vascular} and cork cambia) thicken woody \omterm{def:g12:plant-biology:organs}{stems} and \omterm{def:g12:plant-biology:organs}{roots}.
\end{definition}

\section{Xylem and phloem}

\begin{definition}[Xylem]\label{def:g12:plant-biology:xylem}
\emph{Xylem}\index{xylem} transports water and dissolved minerals from
\omterm{def:g12:plant-biology:organs}{roots} toward shoots. Conducting \omterm{def:g10:the-cell:cell}{cells} (tracheids and vessel elements) are
dead and hollow at maturity, with lignified walls that also provide
mechanical support.
\end{definition}

\begin{definition}[Phloem]\label{def:g12:plant-biology:phloem}
\emph{Phloem}\index{phloem} transports organic solutes --- mainly
sugars --- from sources (e.g.\ photosynthesising leaves) to sinks
(growing tips, \omterm{def:g12:plant-biology:organs}{roots}, fruits, storage organs). Sieve-tube elements are
living but depend on companion \omterm{def:g10:the-cell:cell}{cells}.
\end{definition}

\begin{proposition}[Cohesion--tension in xylem]
\label{prop:g12:plant-biology:cohesion}
Transpiration from leaves generates tension that pulls a continuous
water column upward; cohesion between water molecules and adhesion to
cellulose walls maintain the column under negative pressure
(\emph{cohesion--tension theory}\index{cohesion--tension theory}).
\end{proposition}
\admitted

\begin{proposition}[Pressure-flow in phloem]
\label{prop:g12:plant-biology:pressure-flow}
At sources, sugar loading raises osmotic concentration; water enters and
creates hydrostatic pressure that drives \omterm{def:g12:animal-exchange:bulk}{bulk flow} toward sinks, where
sugar is unloaded (\emph{pressure-flow model}\index{pressure-flow model}).
\end{proposition}
\admitted

\begin{example}[Wilting]\label{ex:g12:plant-biology:wilt}
If soil water cannot replace transpirational loss, \omterm{def:g12:plant-biology:xylem}{xylem} tension and
\omterm{def:g10:the-cell:cell}{cell} turgor fall; soft tissues wilt. Closing stomata slows loss but also
limits CO$_2$ entry for \omterm{def:g10:photosynthesis:def}{photosynthesis}.
\end{example}

\section{Stomata and gas exchange}

\begin{definition}[Stoma]\label{def:g12:plant-biology:stoma}
A \emph{stoma}\index{stoma} (plural stomata) is a pore in the epidermis,
flanked by two \emph{guard cells}\index{guard cell}, that regulates \omterm{def:g12:animal-exchange:gas}{gas
exchange} (CO$_2$ in, O$_2$ and water vapour out).
\end{definition}

\begin{proposition}[Guard-cell mechanics]\label{prop:g12:plant-biology:guard}
When \omterm{def:g12:plant-biology:stoma}{guard cells} take up K$^{+}$ and water, they swell and bow apart,
opening the pore; loss of turgor closes it. Light, CO$_2$, humidity and
the \omterm{def:g12:nerves-hormones:hormone}{hormone} abscisic acid (ABA) under drought modulate opening.
\end{proposition}
\admitted

\begin{remark}[Cuticle]\label{rem:g12:plant-biology:cuticle}
The waxy \emph{cuticle}\index{cuticle} waterproofs the epidermis; most
\omterm{def:g12:animal-exchange:gas}{gas exchange} is forced through stomata, making their regulation central
to plant water relations.
\end{remark}

\section{Plant hormones and auxin}

\begin{definition}[Plant hormone]\label{def:g12:plant-biology:hormone}
A \emph{plant hormone}\index{plant hormone} (phytohormone) is an organic
signal produced in one region that regulates \omterm{def:g10:scientific-biology:properties}{growth} or \omterm{def:g10:scientific-biology:properties}{development}
elsewhere at low concentration. Major classes include \omterm{def:g12:plant-biology:auxin}{auxins}, cytokinins,
gibberellins, abscisic acid and ethylene.
\end{definition}

\begin{definition}[Auxin]\label{def:g12:plant-biology:auxin}
\emph{Auxin}\index{auxin} (notably IAA) promotes \omterm{def:g10:the-cell:cell}{cell} elongation in
shoots, is involved in apical dominance, and mediates tropisms. It moves
directionally (\omterm{def:g10:molecules-of-life:water}{polar} transport) through tissues.
\end{definition}

\begin{proposition}[Phototropism]\label{prop:g12:plant-biology:photo}
Uneven light redistributes \omterm{def:g12:plant-biology:auxin}{auxin} toward the shaded side of a shoot tip;
shaded \omterm{def:g10:the-cell:cell}{cells} elongate more, bending the shoot toward the light
(\emph{phototropism}\index{phototropism}).
\end{proposition}
\admitted

\begin{example}[Other hormones at a glance]
\label{ex:g12:plant-biology:others}
Cytokinins promote \omterm{def:g10:the-cell:cell}{cell} division; gibberellins promote \omterm{def:g12:plant-biology:organs}{stem} elongation
and \omterm{def:g12:plant-biology:flower}{seed} germination; ABA enforces dormancy and stomatal closure under
stress; ethylene promotes fruit ripening and senescence.
\end{example}

\begin{method}[Explaining a tropism]\label{met:g12:plant-biology:tropism}
\begin{enumerate}
  \item Identify the directional stimulus (light, gravity, touch).
  \item Localise \omterm{def:g12:nerves-hormones:hormone}{hormone} redistribution or differential sensitivity.
  \item Link unequal elongation (or \omterm{def:g10:scientific-biology:properties}{growth} inhibition) to curvature.
\end{enumerate}
\end{method}

\section{Reproduction overview}

\begin{definition}[Alternation of generations]
\label{def:g12:plant-biology:alternation}
Land plants alternate a multicellular \omterm{def:g11:meiosis-life-cycles:ploidy}{diploid} \emph{sporophyte}
\index{sporophyte} (produces spores by \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}) with a multicellular
\omterm{def:g11:meiosis-life-cycles:ploidy}{haploid} \emph{gametophyte}\index{gametophyte} (produces gametes by
\omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}). In flowering plants the sporophyte dominates; gametophytes are
highly reduced (pollen, embryo sac).
\end{definition}

\begin{definition}[Flower, pollination, seed]
\label{def:g12:plant-biology:flower}
A \emph{flower}\index{flower} bears reproductive organs. \emph{Pollination}
\index{pollination} transfers pollen to the stigma; fertilisation yields
a zygote and (in angiosperms) endosperm. The \emph{seed}\index{seed}
packages embryo and reserves inside a protective coat; the fruit develops
from the ovary and aids dispersal.
\end{definition}

\begin{proposition}[Double fertilisation]\label{prop:g12:plant-biology:double}
In angiosperms, one sperm fuses with the egg (zygote) and another with
two \omterm{def:g10:molecules-of-life:water}{polar} nuclei (triploid endosperm) --- \emph{double fertilisation}
\index{double fertilisation}.
\end{proposition}
\admitted

\begin{remark}[Link to photosynthesis]\label{rem:g12:plant-biology:photo}
Leaves couple atmospheric CO$_2$ and light to sugar production; \omterm{def:g12:plant-biology:phloem}{phloem}
distributes that \omterm{def:g12:nerves-hormones:synapse}{chemical} energy. Transport and stomatal chapters are
therefore inseparable from carbon nutrition.
\end{remark}

\section{Problem: From root to fruit}

\begin{problem}[{Weekend problem --- integrate transport, stomata,
hormones and reproduction in a crop plant}]
\label{pb:g12:plant-biology:1}
Consider a tomato plant on a hot, dry afternoon, then later in the
season when fruits fill with sugar.

\medskip
\noindent\textbf{Part I --- Structure.}
\begin{enumerate}
  \item Assign \omterm{def:g12:plant-biology:tissues}{dermal}, \omterm{def:g12:plant-biology:tissues}{ground} and \omterm{def:g12:plant-biology:tissues}{vascular tissue} systems to \omterm{def:g12:plant-biology:organs}{leaf}
        functions: waterproofing, \omterm{def:g10:photosynthesis:def}{photosynthesis}, sugar export, water
        import.
  \item Why must \omterm{def:g12:plant-biology:xylem}{xylem} and \omterm{def:g12:plant-biology:phloem}{phloem} run continuously from \omterm{def:g12:plant-biology:organs}{root} to \omterm{def:g12:plant-biology:organs}{leaf}?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Water and sugar.}
\begin{enumerate}[resume]
  \item Describe the path of a water molecule from soil to air via
        \omterm{def:g12:plant-biology:xylem}{xylem} and stomata.
  \item On a hot dry day, predict stomatal behaviour and the effect on
        both water loss and CO$_2$ entry.
  \item When fruits act as sinks, outline how \omterm{def:g12:plant-biology:phloem}{phloem} delivers sucrose
        from mature leaves.
  \item If air is \omterm{rem:g10:molecules-of-life:sat}{saturated} with humidity, how does reduced
        transpiration affect \omterm{def:g12:plant-biology:xylem}{xylem} flow (qualitatively)?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- \omterm{def:g12:nerves-hormones:hormone}{Hormones} and \omterm{def:g10:scientific-biology:properties}{development}.}
\begin{enumerate}[resume]
  \item Explain bending of a shoot toward a window using \omterm{def:g12:plant-biology:auxin}{auxin}.
  \item Name a \omterm{def:g12:nerves-hormones:hormone}{hormone} that promotes fruit ripening and one that closes
        stomata under water stress.
  \item Why might apical dominance (\omterm{def:g12:plant-biology:auxin}{auxin} from the tip) matter for
        pruning practice?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- \omterm{def:g10:scientific-biology:properties}{Reproduction}.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:molecules-of-life:proteins}{Sequence}: \omterm{def:g12:plant-biology:flower}{flower} $\to$ \omterm{def:g12:plant-biology:flower}{pollination} $\to$ fertilisation $\to$
        \omterm{def:g12:plant-biology:flower}{seed} $\to$ fruit (one line each).
  \item What is \omterm{prop:g12:plant-biology:double}{double fertilisation}, and why is endosperm useful?
  \item In one paragraph, argue that plant success on land depends on
        the joint \omterm{def:g11:evolution-mechanisms:evo}{evolution} of \omterm{def:g12:plant-biology:tissues}{vascular tissue}, \omterm{rem:g12:plant-biology:cuticle}{cuticle}/stomata and
        protected embryos (\omterm{def:g12:plant-biology:flower}{seeds}).
\end{enumerate}
\end{problem}
