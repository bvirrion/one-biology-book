\section*{Chapitre \ref{ch:g2:from-seed-to-plant} --- De la graine à la plante}

\begin{solution}{exo:g2:from-seed-to-plant:1}
Une toute petite \omterm{def:g1:plants-around-us:plant}{plante} déjà commencée (le germe de la \omterm{def:g1:plants-around-us:plant}{plante}, avec
le début d'une \omterm{def:g1:plants-around-us:parts}{racine} et d'une \omterm{def:g1:plants-around-us:parts}{tige}) et une réserve de nourriture
pour la nourrir.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:2}
\omterm{def:g2:from-seed-to-plant:germination}{Germer}, c'est se réveiller et se mettre à pousser~: la \omterm{def:g2:from-seed-to-plant:seed}{graine}
gonfle, son enveloppe se fend, et elle commence à devenir une \omterm{def:g1:plants-around-us:plant}{plante}.
La \omterm{def:g1:plants-around-us:parts}{racine} sort la première, et plonge vers le bas.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:3}
De l'eau et de la chaleur. La lumière n'est pas dans la liste --- la
\omterm{def:g2:from-seed-to-plant:seed}{graine} germe très bien dans le noir, sous la terre.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:4}
Elle vit sur la réserve de nourriture rangée dans la \omterm{def:g2:from-seed-to-plant:seed}{graine} --- les
deux moitiés épaisses du haricot. Cette réserve nourrit la \omterm{prop:g2:from-seed-to-plant:cycle}{jeune
pousse} jusqu'à ce que ses \omterm{def:g1:plants-around-us:parts}{feuilles} vertes atteignent la lumière et
qu'elle puisse se nourrir seule.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:5}
Par exemple~: haricots, pois, lentilles, riz, \omterm{def:g2:from-seed-to-plant:seed}{graines} de tournesol,
pépins de pomme, noyau de pêche.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:6}
Parce qu'une \omterm{def:g2:from-seed-to-plant:seed}{graine} a besoin de chaleur autant que d'eau~: dans la
terre froide de l'hiver, elle attendrait, ou pourrirait, au lieu de
\omterm{def:g2:from-seed-to-plant:germination}{germer}. Le printemps apporte le sol tiède que la \omterm{def:g2:from-seed-to-plant:seed}{graine} attend.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:7}
\omterm{def:g2:from-seed-to-plant:seed}{Graine}, \omterm{prop:g2:from-seed-to-plant:cycle}{jeune pousse}, \omterm{def:g1:plants-around-us:plant}{plante} feuillue, \omterm{def:g1:plants-around-us:plant}{plante} en \omterm{def:g1:plants-around-us:parts}{fleur}, cosse pleine
de nouveaux haricots.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:8}
Les deux \omterm{def:g1:plants-around-us:parts}{racines} poussent vers le bas et les deux \omterm{def:g1:plants-around-us:parts}{tiges} vers le
haut, quel que soit le sens dans lequel les haricots étaient
couchés. La \omterm{prop:g2:from-seed-to-plant:cycle}{jeune pousse} trouve toujours le haut et le bas ---
exactement ce que la remarque prévoit.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:9}
Dans le tiroir, les \omterm{def:g2:from-seed-to-plant:seed}{graines} n'avaient pas d'eau, donc elles ne
pouvaient pas \omterm{def:g2:from-seed-to-plant:germination}{germer}~; mais une bonne \omterm{def:g2:from-seed-to-plant:seed}{graine} sans eau ne meurt pas
--- elle attend. Un an plus tard, avec de l'eau et de la chaleur,
les \omterm{def:g2:from-seed-to-plant:seed}{graines} en attente se sont réveillées normalement.
\end{solution}

\begin{solution}{exo:g2:from-seed-to-plant:10}
La réserve de nourriture de la \omterm{def:g2:from-seed-to-plant:seed}{graine} --- les deux moitiés épaisses
du haricot --- joue le rôle du jaune. L'\omterm{def:g2:animals-grow-up:born}{œuf} et la \omterm{def:g2:from-seed-to-plant:seed}{graine} sont tous
deux le commencement d'un nouvel \omterm{def:g1:living-or-not:living}{être vivant}, garnis de la
nourriture dont le jeune a besoin avant de pouvoir se nourrir seul.
\end{solution}
