\chapter{Les bactéries et la résistance aux antibiotiques}\label{ch:g11:antibiotic-resistance}

En 1941, un policier mourant d'une septicémie née d'une égratignure
reçut les premières doses de pénicilline jamais administrées à un
patient. Il se rétablit en quelques jours --- puis rechuta et mourut
quand la réserve, péniblement extraite d'une moisissure, fut épuisée.
En une décennie, la pénicilline était fabriquée à la tonne et des
infections qui tuaient depuis des millénaires devenaient curables en
une semaine. En deux décennies, des bactéries que la pénicilline
n'atteignait plus étaient courantes dans tous les hôpitaux. Ce
chapitre dit pourquoi~: comment les bactéries varient, comment un
\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} les trie, et comment l'arithmétique d'une population de
milliards d'individus change une \omterm{def:g11:mutations:mutation}{mutation} rare en un problème de
service entier.

\section{Comment les bactéries varient}

\begin{proposition}[Reproduction et variation chez les bactéries]\label{prop:g11:antibiotic-resistance:variation}
Une bactérie se reproduit en se divisant en deux après avoir copié son
unique \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}~: les filles sont, aux erreurs de copie près,
génétiquement identiques à la mère --- un \emph{clone}. En milieu
riche, une division toutes les 20 minutes change une \omterm{def:g10:cells-common-unit:cell}{cellule} en un
milliard en dix heures. La variation génétique du clone a deux
sources~:
\begin{itemize}
  \item les \emph{\omterm{def:g11:mutations:mutation}{mutations}} (\cref{ch:g11:mutations}), rares par
        division mais fréquentes dans une population de milliards~;
  \item le \emph{transfert horizontal de gènes}\index{transfert horizontal de gènes}~:
        de l'\omterm{def:g10:universal-dna:information}{ADN} reçu d'une autre bactérie, de la même \omterm{def:g10:biodiversity-scales:species}{espèce} ou d'une
        autre, le plus souvent sous forme de \emph{plasmide} --- un
        petit cercle d'\omterm{def:g10:universal-dna:information}{ADN}, distinct du \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}, qui porte quelques
        \omterm{def:g10:universal-dna:gene}{gènes} et peut passer d'une \omterm{def:g10:cells-common-unit:cell}{cellule} à l'autre par un pont de
        membrane.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[L'arithmétique d'une population]\label{ex:g11:antibiotic-resistance:arithmetic}
Une \omterm{def:g11:mutations:mutation}{mutation} de résistance apparaît environ une fois sur $10^8$
divisions. Un flacon de $10^9$ bactéries a donc connu, dans les
divisions qui l'ont produit, une dizaine de telles \omterm{def:g11:mutations:mutation}{mutations}~; une
plaie infectée ou un intestin, avec $10^{12}$ bactéries, une dizaine de
milliers. Toute population nombreuse de bactéries contient, avant tout
usage d'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique}, des \omterm{def:g10:cells-common-unit:cell}{cellules} qui y résistent --- les répliques
sur velours du \cref{ch:g11:mutations} l'ont prouvé.
\end{example}

\section{Les antibiotiques}

\begin{definition}[Antibiotique]\label{def:g11:antibiotic-resistance:antibiotic}
Un \emph{antibiotique}\index{antibiotique} est une molécule qui tue les
bactéries ou arrête leur croissance, à des concentrations inoffensives
pour les \omterm{def:g10:cells-common-unit:cell}{cellules} du patient, en agissant sur une structure ou un
processus que les bactéries possèdent et que les \omterm{def:g10:cells-common-unit:cell}{cellules} humaines
n'ont pas~: la pénicilline et ses proches bloquent la construction de
la \omterm{def:g10:cells-common-unit:organelle}{paroi bactérienne}~; la streptomycine et la tétracycline bloquent le
\omterm{def:g10:cells-common-unit:organelle}{ribosome} bactérien, qui diffère du nôtre~; d'autres bloquent les
\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymes} bactériennes de la \omterm{def:g11:dna-replication:replication}{réplication de l'ADN}. La plupart ont d'abord
été trouvés chez des moisissures et des bactéries du sol, qui les
fabriquent contre leurs concurrents. Les antibiotiques n'ont aucun
effet sur les virus, qui n'ont aucune de ces cibles.
\end{definition}

\begin{proposition}[Les mécanismes de la résistance]\label{prop:g11:antibiotic-resistance:mechanisms}
Une bactérie est \emph{résistante}\index{résistance aux antibiotiques} à
un \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} lorsqu'elle pousse à des concentrations qui arrêtent
ses parentes sensibles. La résistance a une base génétique de l'une de
quelques sortes~: un \omterm{def:g10:universal-dna:gene}{gène} codant une \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzyme} qui détruit l'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique}
(les pénicillinases coupent la pénicilline en deux)~; une \omterm{def:g11:mutations:mutation}{mutation} qui
modifie la cible de l'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} de sorte que le médicament ne s'y
fixe plus (une \omterm{def:g11:gene-expression:protein}{protéine} ribosomique changée)~; une pompe qui expulse le
médicament~; une paroi qui ne le laisse pas entrer. Chacune est un
\omterm{def:g10:universal-dna:gene}{allèle} ou un \omterm{def:g10:universal-dna:gene}{gène}, et chacune est héritée --- verticalement par le
clone et, quand elle est sur un plasmide, horizontalement par les
voisines.
\end{proposition}
\admitted

\begin{method}[Lire un antibiogramme]\label{met:g11:antibiotic-resistance:antibiogram}
Un \emph{antibiogramme} teste les bactéries d'un patient contre
plusieurs \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotiques} à la fois.
\begin{enumerate}
  \item Étaler les bactéries uniformément sur une boîte~; déposer des
        disques de papier imprégnés chacun d'un \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique}~; incuber
        une nuit.
  \item L'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} diffuse hors du disque, sa concentration
        décroissant avec la distance~; les bactéries poussent en tapis
        sauf là où la concentration les arrête~: une
        \emph{zone d'inhibition} claire entoure chaque disque.
  \item Mesurer le diamètre de chaque zone et le comparer au seuil
        établi pour cet \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique}~: un diamètre supérieur au seuil
        signifie sensible, inférieur résistant.
  \item Les plus grandes zones parmi les résultats sensibles orientent
        la prescription~; un disque sans aucune zone désigne un
        \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} que la souche ignore.
\end{enumerate}
\end{method}

\section{Comment la résistance se répand}

\begin{proposition}[La sélection par l'antibiotique]\label{prop:g11:antibiotic-resistance:selection}
Un \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} ne crée pas de bactéries résistantes~; il les
\emph{sélectionne}. Dans une population où quelques \omterm{def:g10:cells-common-unit:cell}{cellules} portent un
\omterm{def:g10:universal-dna:gene}{allèle} de résistance, le médicament tue ou arrête la majorité sensible
et laisse les quelques résistantes se multiplier sans concurrence~: en
quelques jours, la population est résistante. Plus un \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} est
utilisé --- chez les patients, chez les animaux, dans l'environnement
--- plus souvent ce tri se produit, et plus les bactéries d'un hôpital,
d'une ferme ou d'un pays deviennent résistantes. Une cure incomplète,
qui tue les \omterm{def:g10:cells-common-unit:cell}{cellules} les plus sensibles et épargne les moins sensibles,
sélectionne plus efficacement que tout le reste.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Les répliques sur velours de Lederberg ont montré que les \omterm{def:g10:cells-common-unit:cell}{cellules}
résistantes étaient présentes avant l'exposition. Des cultures menées
une semaine dans des concentrations croissantes d'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique}
finissent résistantes à cent fois la dose initiale, et leur résistance
est héréditaire et se rapporte à des \omterm{def:g11:mutations:mutation}{mutations} précises. Les pays qui
consomment le plus d'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotiques} par habitant ont les plus fortes
proportions de souches résistantes~; les services hospitaliers qui
réduisent leur consommation voient la résistance baisser en quelques
mois. Une souche de la bactérie cutanée \emph{Staphylococcus aureus}
résistante à la pénicilline est apparue quatre ans après
l'introduction du médicament et représente aujourd'hui la majorité des
souches hospitalières~; la souche résistante au médicament de
remplacement a suivi ce remplacement de deux ans.
\end{proof}

\begin{example}[D'une cellule à un service]\label{ex:g11:antibiotic-resistance:ward}
Un patient traité à la pénicilline porte, parmi $10^{10}$ bactéries,
une centaine de résistantes. En trois jours, les sensibles ont disparu
et le clone résistant a pris la place~: $10^{10}$ \omterm{def:g10:cells-common-unit:cell}{cellules}
résistantes. Sur les mains d'une infirmière, sur une barrière de lit,
chez le patient voisin, le clone continue --- et si son \omterm{def:g10:universal-dna:gene}{gène} de
résistance est sur un plasmide, il le passe en chemin à d'autres
\omterm{def:g10:biodiversity-scales:species}{espèces}. Rien dans cette suite n'exigeait que l'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} agisse sur
l'\omterm{def:g10:universal-dna:information}{ADN}~; il n'a fait qu'écarter la concurrence.
\end{example}

\section{Faire durer les antibiotiques}

\begin{proposition}[Conséquences et réponses]\label{prop:g11:antibiotic-resistance:consequences}
Les infections résistantes sont plus longues, plus coûteuses et plus
souvent mortelles~; certaines souches résistent aujourd'hui à tous les
médicaments disponibles. Comme la résistance est entraînée par l'usage,
la réponse consiste à utiliser les \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotiques} moins et mieux~:
seulement contre les infections bactériennes (jamais contre les rhumes
et les grippes virales), choisis après un antibiogramme quand c'est
possible, à pleine dose et pour toute la durée de la cure, avec le
médicament au spectre le plus étroit qui fonctionne~; à limiter leur
usage en élevage~; à prévenir les infections (hygiène, vaccination)
pour que moins de cures soient nécessaires~; et à isoler les \omterm{def:g11:genes-and-disease:genetic}{porteurs}
de souches résistantes à l'hôpital. Les nouveaux \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotiques}
achètent du temps~; l'arithmétique de la sélection garantit que la
résistance à chacun apparaîtra.
\end{proposition}
\admitted

\begin{remark}[L'évolution en une semaine]\label{rem:g11:antibiotic-resistance:evolution}
La \omterm{def:g11:mutations:mutation}{mutation} qui fournit au hasard la variation, l'environnement qui la
trie, les descendants des survivants qui héritent la différence~: les
bactéries de ce chapitre parcourent en une semaine le processus que la
dernière année décrira sur des millions d'années pour les baleines et
les chênes. La \omterm{prop:g11:antibiotic-resistance:mechanisms}{résistance aux antibiotiques}, c'est l'évolution par
sélection naturelle observée en temps réel, avec l'humain pour agent
de sélection --- et la démonstration la plus claire que le processus
n'est pas une théorie sur le passé mais un fait valable pour toute
population qui varie et se reproduit.
\end{remark}

\end{exercise}

\section{Problème~: le service}

\begin{problem}[{Problème du week-end --- une épidémie dans un service
hospitalier suivie depuis une seule cellule mutante~: l'arithmétique
d'une colonie, un antibiogramme lu, une cure interrompue, et la
politique qui arrête la propagation}]
\label{pb:g11:antibiotic-resistance:1}
La plaie d'un patient est infectée par $10^{10}$ bactéries d'une même
\omterm{def:g10:biodiversity-scales:species}{espèce}. Les bactéries se divisent toutes les 30 minutes~; une \omterm{def:g11:mutations:mutation}{mutation}
donnant la résistance à l'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} amoxicilline survient une fois
pour $10^8$ divisions. L'\omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} tue 90~\% des \omterm{def:g10:cells-common-unit:cell}{cellules} sensibles
toutes les 6 heures et n'a aucun effet sur les \omterm{def:g10:cells-common-unit:cell}{cellules} résistantes,
qui continuent de se diviser.

\medskip
\noindent\textbf{Partie I --- Avant le traitement.}
\begin{enumerate}
  \item Combien de divisions ont produit les $10^{10}$ bactéries à
        partir de la première \omterm{def:g10:cells-common-unit:cell}{cellule}~? (Compter le nombre de
        \omterm{def:g10:cells-common-unit:cell}{cellules}.)
  \item Estimer le nombre de \omterm{def:g10:cells-common-unit:cell}{cellules} résistantes présentes avant tout
        \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique}.
  \item Expliquer pourquoi l'infection du patient est néanmoins
        qualifiée de «~sensible~» sur l'antibiogramme.
  \item L'antibiogramme montre des zones de \qty{24}{mm} pour
        l'amoxicilline (seuil 17), \qty{12}{mm} pour l'érythromycine
        (seuil 20) et aucune pour la pénicilline. Interpréter les trois
        résultats.
  \item Le médecin prescrit de l'amoxicilline pour 7 jours. Pourquoi
        pas de la pénicilline, et pourquoi pas le médicament à la plus
        grande zone si l'une avait été plus grande~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Pendant le traitement.}
\begin{enumerate}[resume]
  \item Combien de \omterm{def:g10:cells-common-unit:cell}{cellules} sensibles reste-t-il après 1 jour~? Après 3
        jours~?
  \item En partant de votre réponse à la question 2, combien y a-t-il
        de \omterm{def:g10:cells-common-unit:cell}{cellules} résistantes après 1 jour si elles doublent toutes
        les 30 minutes sans limite~? (Utiliser
        $2^{48} \approx \num{2.8e14}$.) Pourquoi ce chiffre est-il
        irréaliste, et qu'est-ce qui le limite~?
  \item Supposons que le clone résistant soit limité à la place que
        libèrent les \omterm{def:g10:cells-common-unit:cell}{cellules} sensibles, de sorte qu'il ne puisse pas
        dépasser $10^{10}$. Vers quand remplit-il cette place~?
  \item Esquisser, ou décrire, les courbes des \omterm{def:g10:cells-common-unit:cell}{cellules} sensibles,
        résistantes et totales sur les 7 jours.
  \item En réalité, le système immunitaire du patient élimine en
        quelques jours les derniers millions de \omterm{def:g10:cells-common-unit:cell}{cellules}, quelles
        qu'elles soient. Expliquer pourquoi l'issue dépend de ce que le
        clone résistant croisse plus vite ou moins vite que le système
        immunitaire ne l'élimine.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La cure interrompue.}
\begin{enumerate}[resume]
  \item Le patient s'arrête au bout de 2 jours, se sentant bien.
        Combien de \omterm{def:g10:cells-common-unit:cell}{cellules} sensibles reste-t-il, et quelle fraction de
        la population survivante est résistante~?
  \item Les bactéries repoussent jusqu'à $10^{10}$ les jours suivants.
        Quelle proportion est désormais résistante, si les \omterm{def:g10:cells-common-unit:cell}{cellules}
        résistantes et sensibles croissent au même rythme~?
  \item Le patient rechute et reçoit de nouveau de l'amoxicilline.
        Prévoir le résultat.
  \item Un antibiogramme de la rechute ne montre aucune zone pour
        l'amoxicilline. Expliquer le changement par rapport au premier
        antibiogramme.
  \item Le \omterm{def:g10:universal-dna:gene}{gène} de résistance est sur un plasmide. Quel risque
        supplémentaire cela ajoute-t-il pour le service~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La politique du service.}
Le service traite 400 patients par an à l'amoxicilline~; 30~\% des
infections isolées dans le service sont désormais résistantes, contre
5~\% en ville.
\begin{enumerate}[resume]
  \item Expliquer la différence entre le service et la ville en termes
        de pression de sélection.
  \item Le service décide~: antibiogramme avant toute prescription,
        cures menées à leur terme, hygiène des mains et isolement des
        \omterm{def:g11:genes-and-disease:genetic}{porteurs} de souches résistantes. Dire sur quel mécanisme du
        chapitre chaque mesure agit.
  \item Au bout de deux ans, la résistance dans le service est de
        12~\%. Expliquer pourquoi elle a baissé, et pourquoi elle
        n'est pas revenue à 5~\%.
  \item Pourquoi un nouvel \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotique} introduit dans le service ne
        résoudrait-il pas le problème plus de quelques années~?
  \item Énoncer le résultat~: le nombre de \omterm{def:g10:cells-common-unit:cell}{cellules} résistantes que la
        plaie contenait avant tout médicament, le jour où elles en ont
        hérité, et l'unique habitude du patient qui a changé une
        infection curable en une épidémie.
\end{enumerate}
\end{problem}
