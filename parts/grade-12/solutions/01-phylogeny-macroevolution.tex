\section*{Chapter \ref{ch:g12:phylogeny-macroevolution} --- Phylogeny and Macroevolution}

\begin{solution}{exo:g12:phylogeny-macroevolution:1}
\omterm{def:g12:phylogeny-macroevolution:homology}{Homology}: similarity due to common ancestry (e.g.\ vertebrate
forelimbs). \omterm{def:g12:phylogeny-macroevolution:analogy}{Analogy}: similarity due to independent \omterm{def:g11:evolution-mechanisms:evo}{evolution}, often
convergence (e.g.\ bat and insect wings as flight surfaces).
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:2}
A and B are more closely related (more recent common ancestor). Tip
\omterm{def:g10:biodiversity-classification:hierarchy}{order} alone does not decide relatedness: \omterm{def:g12:phylogeny-macroevolution:node}{branches} can be rotated.
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:3}
(a) \omterm{def:g12:phylogeny-macroevolution:monophyly}{Monophyletic} (mammals form a \omterm{def:g12:phylogeny-macroevolution:node}{clade}). (b) \omterm{def:g12:phylogeny-macroevolution:monophyly}{Paraphyletic} (tetrapods
evolved from within ``fish''). (c) \omterm{def:g12:phylogeny-macroevolution:monophyly}{Polyphyletic} if defined only by
endothermy (birds and mammals acquired it separately; their exclusive
common ancestor was not endothermic in that sense).
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:4}
Relatedness is encoded by shared \omterm{def:g12:phylogeny-macroevolution:node}{nodes} (topology), not by left--right
layout. Rotating around a \omterm{def:g12:phylogeny-macroevolution:node}{node} preserves which lineages share which
ancestors.
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:5}
Allopatric: geographic barrier splits \omterm{def:g11:evolution-mechanisms:pool}{gene pools} (e.g.\ a river divides
a rodent range). Sympatric: \omterm{def:g12:phylogeny-macroevolution:speciation}{speciation} without complete geographic
isolation (e.g.\ polyploid plant; host-shift in insects within the same
region).
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:6}
Prezygotic: behavioural (different courtship songs); temporal
(different breeding seasons); mechanical (incompatible genitalia);
gametic (sperm fails to fertilise egg) --- any three. Postzygotic:
hybrid inviability (embryo dies); hybrid sterility (mule); hybrid
breakdown (F2 \omterm{def:g11:evolution-mechanisms:fitness}{fitness} collapse).
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:7}
Birds are nested within archosaurs with crocodiles; excluding birds
omits descendants of the reptile ancestor $\Rightarrow$ paraphyly.
\omterm{def:g12:phylogeny-macroevolution:monophyly}{Monophyletic} alternative: Sauropsida (or Reptilia \emph{sensu lato}
including birds).
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:8}
No. If the outgroup has it, the state is ancestral (or the character is
not polarised as derived for the whole ingroup). A \omterm{def:g10:biodiversity-classification:characters}{shared derived
character} of the whole ingroup should be present in the ingroup
ancestor and absent (as the derived state) from the outgroup.
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:9}
\omterm{def:g11:dna-replication:chromosome}{Chromosome} doubling yields a polyploid that is often reproductively
isolated from \omterm{def:g11:meiosis-life-cycles:ploidy}{diploids} at once (meiotic mismatch). Animals usually have
sex \omterm{def:g11:dna-replication:chromosome}{chromosomes} and developmental constraints that make viable
polyploids rarer; self-fertilisation and clonal options in plants help
polyploids establish.
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:10}
Root between the two pairs: $((W,X),(Y,Z))$. \omterm{def:g12:phylogeny-macroevolution:node}{Clade} of three that
becomes \omterm{def:g12:phylogeny-macroevolution:monophyly}{paraphyletic} if one is excluded: e.g.\ $\{W,X,Y\}$ is not a
\omterm{def:g12:phylogeny-macroevolution:node}{clade}; the \omterm{def:g12:phylogeny-macroevolution:monophyly}{monophyletic} set of three might be none if only balanced
pairs --- alternatively mark $\{W,X,Y,Z\}$ and note that $\{W,X,Y\}$
(excluding Z) is \omterm{def:g12:phylogeny-macroevolution:monophyly}{paraphyletic}. \omterm{def:g12:phylogeny-macroevolution:monophyly}{Polyphyletic} pair: $\{W,Y\}$ or
$\{X,Z\}$.
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:11}
Colonisers face empty \omterm{def:g10:ecosystems:habitat-niche}{niches} (seeds, insects, different islands).
Isolation among islands reduces \omterm{def:g11:evolution-mechanisms:geneflow}{gene flow}. Divergent selection on beak
and behaviour specialises subpopulations; \omterm{def:g12:phylogeny-macroevolution:isolation}{reproductive isolation}
accumulates and many \omterm{def:g12:phylogeny-macroevolution:species}{species} replace one generalist ancestor.
\end{solution}

\begin{solution}{exo:g12:phylogeny-macroevolution:12}
\omterm{def:g12:phylogeny-macroevolution:analogy}{Homoplasy} (convergence, reverse \omterm{def:g11:evolution-mechanisms:mutation}{mutation}), incomplete lineage sorting,
hybridisation, short internal \omterm{def:g12:phylogeny-macroevolution:node}{branches} with few supporting characters.
More loci, morphological fossils, biogeography or molecular clock
dating can tip support toward one topology.
\end{solution}

\begin{solution}{pb:g12:phylogeny-macroevolution:1}
\textbf{Part I.} (1)~Character 1 (state 1 in all A--D). (2)~Chars 2 (and
possibly others unique to A/B): 2 supports $\{A,B\}$; 3 and 6 support
$\{C,D\}$. (3)~Autapomorphies: they diagnose single tips, not groupings.
(4)~\omterm{def:g12:phylogeny-macroevolution:analogy}{Homoplasy} weakens character 2 as unique evidence for $\{A,B\}$;
other characters (if any) must carry the grouping.

\textbf{Part II.} (5)--(6)~Tree: O outside; split into $(A,B)$ and
$(C,D)$; sister of $\{A,B\}$ is $\{C,D\}$. (7)~$\{A,B,C\}$ is not
\omterm{def:g12:phylogeny-macroevolution:monophyly}{monophyletic} (excludes D, descendant of the $C$--$D$ ancestor shared with
the larger \omterm{def:g12:phylogeny-macroevolution:node}{clade}). (8)~\omterm{def:g12:phylogeny-macroevolution:monophyly}{Monophyletic}: $\{A,B\}$; \omterm{def:g12:phylogeny-macroevolution:monophyly}{paraphyletic}:
$\{A,B,C\}$; \omterm{def:g12:phylogeny-macroevolution:monophyly}{polyphyletic}: $\{A,C\}$.

\textbf{Part III.} (9)~E.g.\ continental separation or mountain uplift
splits ancestral range. (10)~E.g.\ ecological specialisation or
polyploidy. (11)~Survivors diversify into empty \omterm{def:g10:ecosystems:habitat-niche}{niches}; many short
\omterm{def:g12:phylogeny-macroevolution:node}{branches} near the base of surviving \omterm{def:g12:phylogeny-macroevolution:node}{clades}. (12)~\omterm{def:g12:phylogeny-macroevolution:monophyly}{Monophyletic} taxa share
a unique history and thus tend to share derived traits and responses;
similarity alone confounds \omterm{def:g12:phylogeny-macroevolution:homology}{homology} with \omterm{def:g12:phylogeny-macroevolution:analogy}{analogy}.
\end{solution}
