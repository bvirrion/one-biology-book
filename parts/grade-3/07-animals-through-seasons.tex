\chapter{Animals Through the Seasons}\label{ch:g3:animals-through-seasons}

In October the swallows line up on the wires and, one morning,
they are gone. The hedgehog under the hedge rolls itself into
its \omterm{def:g1:plants-around-us:parts}{leaf} nest and will not be seen before spring. The fox,
meanwhile, is out in every snow. Winter is a hard season ---
cold, and above all short of food --- and \omterm{def:g1:animals-around-us:animal}{animals} have three
great ways through it.

\section{Winter, the season of scarcity}

\begin{proposition}[Winter's double problem]\label{prop:g3:animals-through-seasons:problem}
Winter brings \omterm{def:g1:animals-around-us:animal}{animals} two problems at once:
\begin{enumerate}
  \item \emph{cold}, which drains body warmth faster --- and
        keeping warm costs \omterm{prop:g3:balanced-diet:jobs}{energy}
        (\cref{prop:g3:balanced-diet:jobs});
  \item \emph{scarce food}: \omterm{def:g1:plants-around-us:plant}{plants} stop growing, \omterm{prop:g3:sorting-living-things:groups}{insects}
        disappear, the first links of the \omterm{def:g3:food-chains:chain}{food chains} of
        \cref{ch:g3:food-chains} thin out --- so the scarcity
        travels up to everyone.
\end{enumerate}
More warmth needed, less food to pay for it: that is the winter
problem every \omterm{def:g1:animals-around-us:animal}{animal} must solve.
\end{proposition}
\admitted

\begin{example}[Where did the insect-eaters' food go?]\label{ex:g3:animals-through-seasons:insects}
The swallow feeds on flying \omterm{prop:g3:sorting-living-things:groups}{insects}, caught on the wing. In
November there are simply none: too cold. No change of menu can
fix that for a bird built for aerial hunting --- the swallow's
food supply vanishes completely.
\end{example}

\section{Three answers to winter}

\begin{definition}[Migration]\label{def:g3:animals-through-seasons:migration}
To \emph{migrate}\index{migrate} is to travel, each year, from
one region to another as the seasons change --- leaving before
winter for warmer lands where food remains, and returning in
spring. The swallow, the stork and the crane are
\emph{migrators}\index{migration}; some travel thousands of
kilometres, and find their way back to the very same nest.
\end{definition}

\begin{definition}[Hibernation]\label{def:g3:animals-through-seasons:hibernation}
To \emph{hibernate}\index{hibernate} is to spend the winter in
a deep, cold \omterm{prop:g1:healthy-habits:needs}{sleep}, hidden in a shelter: the body cools, heart
and breathing slow to almost nothing, and the \omterm{def:g1:animals-around-us:animal}{animal} lives
slowly off fat stored up in autumn. The hedgehog, the dormouse,
the marmot and the bat hibernate.
\end{definition}

\begin{definition}[Staying active]\label{def:g3:animals-through-seasons:active}
The rest \emph{stay active}\index{staying active (in winter)}
and change their equipment or their habits: a thicker
\emph{winter coat}\index{winter coat}, food
\emph{stores}\index{food store} gathered in autumn (the
squirrel's buried nuts), a switch of menu
(\cref{ex:g2:what-animals-eat:winter}'s blackbird), or simply
harder searching --- the fox, the deer, the robin.
\end{definition}

\begin{example}[Three neighbors, three strategies]\label{ex:g3:animals-through-seasons:three}
One hedgerow in January: the swallows that nested above it are
in Africa, catching \omterm{prop:g3:sorting-living-things:groups}{insects} under a warm sun; the hedgehog
below it \omterm{prop:g1:healthy-habits:needs}{sleeps} cold and slow in its \omterm{def:g1:plants-around-us:parts}{leaf} nest, burning
August's fat; the robin on top of it, in its puffed-up winter
plumage, patrols for the last berries. Come April, all three
will be busy in the same hedge again.
\end{example}

\begin{remark}[Sleep is not laziness, travel is not tourism]\label{rem:g3:animals-through-seasons:cost}
Each strategy has its price. The migrator risks an exhausting
journey; the hibernator must gather every gram of its winter
fat in autumn --- a hedgehog too thin in November will not wake
in spring; the active \omterm{def:g1:animals-around-us:animal}{animal} gambles on finding food through
the hungriest months. Winter is paid for in advance, or en
route, or day by day.
\end{remark}

\begin{method}[Guessing an animal's winter strategy]\label{met:g3:animals-through-seasons:guess}
\begin{enumerate}
  \item Ask what it eats, and whether that food exists here in
        winter;
  \item food completely gone (flying \omterm{prop:g3:sorting-living-things:groups}{insects}, nectar): expect
        \omterm{def:g3:animals-through-seasons:migration}{migration} or hibernation;
  \item food findable (\omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}, berries, prey): expect an active
        winter, perhaps with a thicker coat or stores;
  \item check in autumn: fattening up and digging a nest points
        to hibernation; gathering in flocks points to departure.
\end{enumerate}
\end{method}

\begin{example}[Spring undoes it all]\label{ex:g3:animals-through-seasons:spring}
In March and April the order reverses: \omterm{prop:g3:sorting-living-things:groups}{insects} return, the
\omterm{def:g3:animals-through-seasons:migration}{migrators} reappear at their old nests, the hibernators wake
thin and ravenous, \omterm{def:g3:animals-through-seasons:active}{winter coats} are shed. The seasons turn, and
the \omterm{def:g1:animals-around-us:animal}{animals}' calendar turns with them --- next autumn, the
swallows will line up on the wires again.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:1}
What double problem does winter set every \omterm{def:g1:animals-around-us:animal}{animal}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:2}
Name the three winter strategies, with two \omterm{def:g1:animals-around-us:animal}{animals} for each.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:3}
What happens to a hibernating hedgehog's body during its long
\omterm{prop:g1:healthy-habits:needs}{sleep}? What does it live on?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:4}
Why exactly must the swallow leave --- what is its food, and
what becomes of it here in winter?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:5}
Give three ways an active \omterm{def:g1:animals-around-us:animal}{animal} gets through winter, each with
its \omterm{def:g1:animals-around-us:animal}{animal}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:6}
What is each strategy's price, by
\cref{rem:g3:animals-through-seasons:cost}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:7}
In autumn, an \omterm{def:g1:animals-around-us:animal}{animal} is seen eating hugely and lining a hole
with dry \omterm{def:g1:plants-around-us:parts}{leaves}. Which strategy is it preparing? Which step of
\cref{met:g3:animals-through-seasons:guess} told you?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:8}
Try it: keep a winter list of the \omterm{prop:g3:sorting-living-things:groups}{birds} you actually see
outside during one cold week. Are the swallows on it? Which of
your \omterm{prop:g3:sorting-living-things:groups}{birds} chose the active strategy?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:animals-through-seasons:9}
Use \cref{met:g3:animals-through-seasons:guess} on: (a) the
squirrel (eats nuts and \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}); (b) the crane (eats frogs and
\omterm{prop:g3:sorting-living-things:groups}{insects} of wet meadows). What does each likely do in winter?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:animals-through-seasons:10}
A mild autumn keeps a young hedgehog awake and hunting late,
but it stays thin. Why is this dangerous for it, by
\cref{rem:g3:animals-through-seasons:cost}? What do \omterm{def:g1:animals-around-us:animal}{animal}
rescuers do with too-thin autumn hedgehogs?
\end{exercise}
