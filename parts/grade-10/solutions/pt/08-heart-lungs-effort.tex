\begin{solution}{exo:g10:heart-lungs-effort:5}

\begin{solution}{exo:g10:heart-lungs-effort:6}

\begin{solution}{exo:g10:heart-lungs-effort:7}
Repouso: \qty{2.5}{L/min}; exercício: $0.04 \times 24 \approx
\qty{1.0}{L/min}$. A parcela cai doze vezes, o fluxo absoluto apenas
por um fator de 2.6, porque o próprio débito cresceu.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:8}
$18 \times 0.14 = \qty{2.5}{L/min}$: cerca de três quartos dos
\qty{3.45}{L/min} de um estudante treinado --- perto disso, para uma
pessoa menor.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:9}
O cérebro não tolera nem segundos sem oxigênio e sem glicose, de modo
que o fornecimento dele é protegido. O intestino e os rins conseguem
suspender a digestão e a filtração por uma hora sem dano, e a parcela
deles é emprestada aos músculos.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:10}
$V_s = 5000/42 \approx \qty{120}{mL}$, contra cerca de \qty{70}{mL}: o
coração aumentado dela entrega o débito de repouso em bem menos
batimentos.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:11}
A ordem de movimento do encéfalo, e a antecipação do esforço, são
enviadas aos centros cardíacos e às glândulas suprarrenais antes de os
músculos agirem. Um corredor que dependesse só da detecção do gás
carbônico começaria com o débito de repouso, correria o primeiro minuto
com a \omterm{def:g10:cell-metabolism:fermentation}{fermentação} e seria obrigado a desacelerar pelo ácido láctico
antes de a entrega alcançá-lo.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:12}
$195 \times 0.110 \times 0.150 \approx \qty{3.2}{L/min}$ e
$195 \times 0.160 \times 0.150 \approx \qty{4.7}{L/min}$. O volume
sistólico é o que o treinamento aumenta: o segundo sujeito é o treinado
para provas de resistência.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:13}
Oxigênio inspirado: $0.21 \times 100 = \qty{21}{L/min}$; usado:
\qty{3.5}{L/min}, ou seja 17\%. O resto é expirado de novo --- o ar
expirado ainda contém cerca de 16\% de oxigênio, e é por isso que a
respiração boca a boca funciona.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:14}
Débito de no máximo cerca de $110 \times 0.12 \approx \qty{13}{L/min}$
em vez de 24: o $\dot V\!\mathrm{O_2}$max cai quase pela metade, para
cerca de \qty{2}{L/min}. O paciente consegue caminhar e pedalar
suavemente, mas não sustentar nenhum esforço puxado.
\end{solution}

\begin{solution}{exo:g10:heart-lungs-effort:15}
Entrega por litro reduzida em um quarto, extração inalterada em
proporção: o $\dot V\!\mathrm{O_2}$max cai 25\%. Nas semanas seguintes
o corpo fabrica mais hemácias, elevando o oxigênio carregado por litro
de volta na direção de \qty{200}{mL}.
\end{solution}

\begin{solution}{pb:g10:heart-lungs-effort:1}
\textbf{1.} $62 \times 0.072 \approx 4.5$; $105 \times 0.100 = 10.5$;
$140 \times 0.116 \approx 16.2$; $175 \times 0.120 = 21.0$;
$198 \times 0.120 \approx \qty{23.8}{L/min}$.

\textbf{2.} $23.8/4.5 \approx 5.3$: a frequência contribui com $198/62
\approx 3.2$, o volume sistólico com $120/72 \approx 1.7$, e $3.2 \times
1.7 \approx 5.3$.

\textbf{3.} Acima de cerca de \qty{225}{W} o volume sistólico fica em
\qty{120}{mL}; só a frequência cardíaca eleva mais o débito.

\textbf{4.} $220 - 17 = 203$: os 198 dele estão dentro da precisão da regra.

\textbf{5.} $5/4.5 \approx \qty{1.1}{min}$, cerca de \qty{67}{s}; a
\qty{300}{W}, $5/23.8 \approx \qty{0.21}{min}$, cerca de \qty{13}{s}.

\textbf{6.} $0.5 \times 13 = 6.5$; $1.2 \times 18 = 21.6$; $1.8 \times
24 = 43.2$; $2.3 \times 32 = 73.6$; $2.6 \times 44 \approx
\qty{114}{L/min}$.

\textbf{7.} $114/6.5 \approx 18$: o volume corrente subiu 5.2 vezes, a
frequência 3.4 vezes; o volume contribui mais.

\textbf{8.} $2.6/5.0 = 52\%$ da capacidade vital por incursão.

\textbf{9.} Não: a \omterm{def:g10:heart-lungs-effort:ventilation}{ventilação} tem uma grande reserva e o sangue que sai
dos pulmões permanece saturado. O limite é o volume sistólico do
coração, que fixa quanto oxigênio cada batimento entrega.

\textbf{10.} $0.7 \times 6.5 \approx \qty{4.6}{L/min}$ e $0.7 \times
114 \approx \qty{80}{L/min}$.

\textbf{11.} $4.5 \times 0.052 \approx 0.23$; $10.5 \times 0.090
\approx 0.95$; $16.2 \times 0.120 \approx 1.94$; $21.0 \times 0.145
\approx 3.05$; $23.8 \times 0.155 \approx \qty{3.7}{L/min}$.

\textbf{12.} $3700/68 \approx \qty{54}{mL/min}$ por quilograma. A
frequência chegou ao máximo dela e o aumento de 225 para \qty{300}{W} é
pequeno: este é o $\dot V\!\mathrm{O_2}$max dele, o de um estudante em
boa forma e treinado.

\textbf{13.} $3.7/0.23 \approx 16$: o fator 5.3 do coração multiplicado
pelo $155/52 \approx 3.0$ da extração.

\textbf{14.} $3.7 \times 20 = \qty{74}{kJ}$ por minuto, cerca de
\qty{1230}{W}; em repouso $0.23 \times 20/60 \approx \qty{77}{W}$;
acima do repouso, cerca de \qty{1150}{W}, de modo que o rendimento é
$300/1150 \approx
26\%$.

\textbf{15.} Aproximadamente uma reta de \num{0.23} a
\qty{3.05}{L/min} entre 0 e \qty{225}{W}, e depois encurvando: os
últimos \qty{75}{W} acrescentam apenas \qty{0.65}{L/min}. O teto é de
cerca de \qty{3.7}{L/min}.

\textbf{16.} A antecipação: a expectativa do esforço no encéfalo é
retransmitida aos centros cardíacos e às glândulas suprarrenais,
elevando a frequência antes de qualquer movimento.

\textbf{17.} Os \omterm{def:g10:heart-lungs-effort:ventilation}{alvéolos} oxigenam o sangue por completo mesmo com
\omterm{def:g10:heart-lungs-effort:ventilation}{ventilação} e débito máximos; os pulmões não são o limite. O limite é a
quantidade de sangue que o coração consegue enviar por minuto.

\textbf{18.} $20/23.8 \approx 84\%$. O intestino e os rins, que ficavam
com metade do débito de repouso, recebem agora cerca de \qty{1}{L/min};
as artérias deles se estreitaram e o fluxo foi redirecionado aos
músculos.

\textbf{19.} $198 \times 0.140 \times 0.155 \approx \qty{4.3}{L/min}$,
cerca de \qty{63}{mL/min} por quilograma.

\textbf{20.} O \omterm{def:g10:heart-lungs-effort:output}{débito cardíaco} multiplicado por 5.3, a extração de
oxigênio por 3, a \omterm{def:g10:heart-lungs-effort:ventilation}{ventilação} por 18; o treinamento mudou o volume
sistólico do coração e, com ele, o teto.
\end{solution}
