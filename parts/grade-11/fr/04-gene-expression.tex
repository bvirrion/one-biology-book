\chapter{Du gène à la protéine}\label{ch:g11:gene-expression}

En 1961, un biochimiste nourrit un extrait acellulaire de bactéries
avec un \omterm{def:g11:gene-expression:transcription}{ARN} artificiel fait uniquement de la lettre U, répétée.
L'extrait fabriqua une \omterm{def:g11:gene-expression:protein}{protéine} faite uniquement d'un \omterm{def:g10:chemistry-of-life:families}{acide aminé}, la
phénylalanine, répété. Cette seule expérience livrait le premier mot du
\omterm{def:g11:gene-expression:code}{code génétique}~: UUU signifie phénylalanine. En cinq ans, les
soixante-quatre mots étaient connus, et il s'avéra que c'étaient les
mêmes mots chez une bactérie, un blé et un humain. Ce chapitre suit le
chemin qui va de la séquence de \omterm{def:g10:universal-dna:nucleotide}{nucléotides} d'un \omterm{def:g10:universal-dna:gene}{gène} à la séquence
d'\omterm{def:g10:chemistry-of-life:families}{acides aminés} d'une \omterm{def:g11:gene-expression:protein}{protéine} --- le chemin que suit chaque caractère
du \cref{ch:g11:mutations}.

\section{Les gènes fabriquent des protéines}

\begin{proposition}[Un gène, une protéine]\label{prop:g11:gene-expression:onegene}
Un \omterm{def:g10:universal-dna:gene}{gène} est exprimé quand la \omterm{def:g10:cells-common-unit:cell}{cellule} utilise sa séquence pour
fabriquer la \omterm{def:g11:gene-expression:protein}{protéine} qu'il code. La plupart des caractères dépendent
de \omterm{def:g10:chemistry-of-life:families}{protéines} --- \omterm{def:g10:cell-metabolism:metabolism}{enzymes}, fibres de structure, transporteurs,
récepteurs --- et une \omterm{def:g11:mutations:mutation}{mutation} change un caractère en changeant la
\omterm{def:g11:gene-expression:protein}{protéine} que son \omterm{def:g10:universal-dna:gene}{gène} produit.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Beadle et Tatum (1941) ont irradié des spores d'une moisissure du pain
et recueilli des mutants qui ne pouvaient plus pousser sur milieu
minimum à moins d'y ajouter une substance précise, comme l'\omterm{def:g10:chemistry-of-life:families}{acide aminé}
arginine. Chaque mutant était privé d'une \omterm{def:g10:cell-metabolism:metabolism}{enzyme} de la chaîne de
réactions qui fabrique cette substance~; chaque défaut se transmettait
comme un \omterm{def:g10:universal-dna:gene}{gène} unique~; et des mutants bloqués à des étapes différentes
de la même chaîne portaient des \omterm{def:g11:mutations:mutation}{mutations} dans des \omterm{def:g10:universal-dna:gene}{gènes} différents. Un
\omterm{def:g10:universal-dna:gene}{gène}, une \omterm{def:g10:cell-metabolism:metabolism}{enzyme} --- et, comme les travaux ultérieurs l'ont généralisé,
un \omterm{def:g10:universal-dna:gene}{gène}, une chaîne polypeptidique.
\end{proof}

\begin{definition}[Protéine, séquence d'acides aminés]\label{def:g11:gene-expression:protein}
Une \emph{protéine}\index{protéine} est une chaîne d'\omterm{def:g10:chemistry-of-life:families}{acides aminés},
longue de quelques dizaines à plusieurs milliers d'unités, prises dans
un jeu de vingt sortes. La \emph{séquence} --- l'ordre des \omterm{def:g10:chemistry-of-life:families}{acides
aminés} --- est ce que le \omterm{def:g10:universal-dna:gene}{gène} spécifie~; une fois faite, la chaîne se
replie en une forme tridimensionnelle définie que cette séquence
détermine, et la forme détermine la fonction. Changez un \omterm{def:g10:chemistry-of-life:families}{acide aminé} et
le repliement, donc la fonction, peuvent changer.
\end{definition}

\section{La transcription~: le gène copié en ARN}

\begin{definition}[ARN messager et transcription]\label{def:g11:gene-expression:transcription}
L'\emph{ARN}\index{ARN} (acide ribonucléique) est une chaîne de
\omterm{def:g10:universal-dna:nucleotide}{nucléotides} à un seul brin, semblables à ceux de l'\omterm{def:g10:universal-dna:information}{ADN}, sauf que son
sucre est le ribose et que la base uracile (U) y remplace la thymine
(T). La \emph{transcription}\index{transcription} est la copie de la
séquence d'un \omterm{def:g10:universal-dna:gene}{gène} en une molécule d'\emph{ARN messager}\index{ARN messager}
(ARNm)~: l'\omterm{def:g10:cell-metabolism:metabolism}{enzyme} \emph{ARN polymérase} ouvre la \omterm{prop:g10:universal-dna:helix}{double hélice} le long
du \omterm{def:g10:universal-dna:gene}{gène}, lit un brin (le \emph{brin transcrit}) et assemble l'ARN
complémentaire, U en face de A, A en face de T, G en face de C et C en
face de G. L'ARNm a la séquence de l'autre brin du \omterm{def:g10:universal-dna:gene}{gène}, avec U pour
T~; il quitte le \omterm{def:g10:cells-common-unit:organelle}{noyau} pour le \omterm{def:g10:cells-common-unit:cell}{cytoplasme}.
\end{definition}

\begin{example}[Un transcrit]\label{ex:g11:gene-expression:transcript}
Le brin transcrit \texttt{3'-TAC GGA CTT ATC-5'} donne l'ARNm
\texttt{5'-AUG CCU GAA UAG-3'}. L'autre brin du \omterm{def:g10:universal-dna:gene}{gène} se lit
\texttt{5'-ATG CCT GAA TAG-3'}~: l'ARNm est la séquence de ce brin
avec U pour T, et c'est pourquoi on écrit conventionnellement les
séquences de \omterm{def:g10:universal-dna:gene}{gènes} sous la forme de ce brin, le \emph{brin codant}.
\end{example}

\section{Le code génétique}

\begin{definition}[Codon et code génétique]\label{def:g11:gene-expression:code}
L'ARNm est lu par groupes consécutifs de trois \omterm{def:g10:universal-dna:nucleotide}{nucléotides}, les
\emph{codons}\index{codon}, à partir d'un point de départ défini. Le
\emph{code génétique}\index{code génétique} est la correspondance entre
les 64 codons possibles et les 20 \omterm{def:g10:chemistry-of-life:families}{acides aminés}~: 61 codons spécifient
un \omterm{def:g10:chemistry-of-life:families}{acide aminé}, trois (UAA, UAG, UGA) sont des signaux d'\emph{arrêt}
qui terminent la chaîne, et AUG, qui spécifie la méthionine, sert aussi
de codon \emph{initiateur}. Le code est \emph{redondant} --- la plupart
des \omterm{def:g10:chemistry-of-life:families}{acides aminés} ont plusieurs codons --- et \emph{universel}~: la même
table sert à tous les organismes connus, à de rares variantes mineures
près.
\end{definition}

\begin{proposition}[Comment le code a été lu]\label{prop:g11:gene-expression:readcode}
Le sens de chaque \omterm{def:g11:gene-expression:code}{codon} a été établi expérimentalement.
\end{proposition}
\begin{proof}[Faits expérimentaux]
Nirenberg et Matthaei (1961) ont ajouté un \omterm{def:g11:gene-expression:transcription}{ARN} synthétique fait d'un
seul \omterm{def:g10:universal-dna:nucleotide}{nucléotide} répété à un extrait acellulaire contenant des
\omterm{def:g10:cells-common-unit:organelle}{ribosomes}, des \omterm{def:g10:chemistry-of-life:families}{acides aminés} et les \omterm{def:g10:cell-metabolism:metabolism}{enzymes} de la synthèse des
\omterm{def:g10:chemistry-of-life:families}{protéines}~: le poly-U a donné une chaîne de phénylalanines, le poly-C
une chaîne de prolines, le poly-A une chaîne de lysines. Des \omterm{def:g11:gene-expression:transcription}{ARN}
synthétiques à motifs de deux et de trois lettres, puis la fixation de
triplets isolés sur les \omterm{def:g10:cells-common-unit:organelle}{ribosomes}, ont attribué les \omterm{def:g11:gene-expression:code}{codons} restants
d'ici 1966. La longueur de trois lettres avait été déduite de
l'arithmétique (deux lettres ne donnent que 16 combinaisons, trop peu
pour 20 \omterm{def:g10:chemistry-of-life:families}{acides aminés}~; trois en donnent 64) et confirmée par des
\omterm{def:g11:mutations:mutation}{mutations}~: insérer un ou deux \omterm{def:g10:universal-dna:nucleotide}{nucléotides} dans un \omterm{def:g10:universal-dna:gene}{gène} en détruit la
\omterm{def:g11:gene-expression:protein}{protéine}, en insérer trois en restaure une presque normale.
\end{proof}

\begin{method}[Traduire une séquence]\label{met:g11:gene-expression:translate}
\begin{enumerate}
  \item Écrire le brin codant (ou transcrire le brin transcrit)~:
        remplacer T par U pour obtenir l'ARNm.
  \item Trouver le premier AUG~: c'est l'initiation, et le cadre de
        lecture est fixé à partir de lui.
  \item Découper l'ARNm en triplets consécutifs à partir de l'AUG et
        lire chacun dans la table.
  \item S'arrêter au premier \omterm{def:g11:gene-expression:code}{codon} d'arrêt~; les \omterm{def:g10:chemistry-of-life:families}{acides aminés}
        énumérés, dans l'ordre, forment la séquence de la \omterm{def:g11:gene-expression:protein}{protéine}.
  \item Pour un mutant, recommencer et comparer~: même \omterm{def:g11:gene-expression:protein}{protéine}
        (silencieuse), un \omterm{def:g10:chemistry-of-life:families}{acide aminé} changé (faux-sens), arrêt
        prématuré (non-sens), tout changé après la \omterm{def:g11:mutations:mutation}{mutation} (décalage
        du cadre de lecture).
\end{enumerate}
\end{method}

\begin{example}[Quatre mutations, quatre issues]\label{ex:g11:gene-expression:outcomes}
Brin codant \texttt{ATG CCT GAA TTC TAG}~: ARNm
\texttt{AUG CCU GAA UUC UAG}, \omterm{def:g11:gene-expression:protein}{protéine} Met--Pro--Glu--Phe.
\begin{itemize}
  \item \texttt{ATG CC\textbf{C} GAA TTC TAG}~: CCC est toujours Pro
        --- \omterm{def:g11:mutations:mutation}{mutation} \emph{silencieuse}.
  \item \texttt{ATG CCT G\textbf{C}A TTC TAG}~: GCA est Ala ---
        Met--Pro--Ala--Phe, \emph{faux-sens}.
  \item \texttt{ATG CCT \textbf{T}AA TTC TAG}~: UAA est un arrêt ---
        Met--Pro, \emph{non-sens}, une \omterm{def:g11:gene-expression:protein}{protéine} tronquée.
  \item \texttt{ATG CC\textbf{A}T GAA TTC TAG}~: un A inséré décale le
        cadre~: \texttt{AUG CCA UGA}\dots{} Met--Pro--arrêt ---
        \emph{décalage du cadre de lecture}.
\end{itemize}
\end{example}

\section{La traduction~: le message lu}

\begin{proposition}[La traduction]\label{prop:g11:gene-expression:translation}
La \emph{traduction}\index{traduction} se déroule sur les
\emph{\omterm{def:g10:cells-common-unit:organelle}{ribosomes}}, dans le \omterm{def:g10:cells-common-unit:cell}{cytoplasme}. Un \omterm{def:g10:cells-common-unit:organelle}{ribosome} se fixe sur l'ARNm au
niveau de son \omterm{def:g11:gene-expression:code}{codon} initiateur et le parcourt \omterm{def:g11:gene-expression:code}{codon} par \omterm{def:g11:gene-expression:code}{codon}. Chaque
\omterm{def:g11:gene-expression:code}{codon} est reconnu par un \emph{\omterm{def:g11:gene-expression:transcription}{ARN} de transfert} (ARNt), petit \omterm{def:g11:gene-expression:transcription}{ARN} qui
porte, à une extrémité, l'\emph{anticodon} de trois \omterm{def:g10:universal-dna:nucleotide}{nucléotides}
complémentaire du \omterm{def:g11:gene-expression:code}{codon} et, à l'autre, l'\omterm{def:g10:chemistry-of-life:families}{acide aminé} correspondant. Le
\omterm{def:g10:cells-common-unit:organelle}{ribosome} joint chaque \omterm{def:g10:chemistry-of-life:families}{acide aminé} apporté à la chaîne en croissance et
libère l'ARNt vide~; à un \omterm{def:g11:gene-expression:code}{codon} d'arrêt, il libère la chaîne achevée,
qui se replie. Plusieurs \omterm{def:g10:cells-common-unit:organelle}{ribosomes} lisent un même ARNm à la suite, et
un ARNm donne des centaines de copies de la \omterm{def:g11:gene-expression:protein}{protéine} avant d'être
dégradé.
\end{proposition}
\admitted

\begin{example}[Vitesse et rendement]\label{ex:g11:gene-expression:speed}
Un \omterm{def:g10:cells-common-unit:organelle}{ribosome} ajoute 5 à 20 \omterm{def:g10:chemistry-of-life:families}{acides aminés} par seconde~: une \omterm{def:g11:gene-expression:protein}{protéine} de
300 \omterm{def:g10:chemistry-of-life:families}{acides aminés} prend moins d'une minute. Les \omterm{def:g10:cells-common-unit:organelle}{ribosomes} se suivent
sur un ARNm à environ 80 \omterm{def:g10:universal-dna:nucleotide}{nucléotides} d'intervalle, si bien qu'un
message de 900 \omterm{def:g10:universal-dna:nucleotide}{nucléotides} en porte une douzaine à la fois~; sur sa vie
de quelques heures, un seul ARNm produit un millier de molécules de
\omterm{def:g11:gene-expression:protein}{protéine}. Une \omterm{def:g10:cells-common-unit:cell}{cellule} d'\emph{E.~coli} fabrique quelque
$\num{20000}$ molécules de \omterm{def:g11:gene-expression:protein}{protéine} par seconde~; le précurseur d'un
globule rouge, quelque $\num{5e8}$ molécules d'hémoglobine au cours de
son existence.
\end{example}

\section{Un gène, plusieurs messages}

\begin{proposition}[La maturation du message chez les eucaryotes]\label{prop:g11:gene-expression:splicing}
Dans les \omterm{def:g10:cells-common-unit:prokaryote}{cellules eucaryotes}, la séquence d'un \omterm{def:g10:universal-dna:gene}{gène} est interrompue par
des \emph{introns}\index{intron}, segments qui sont transcrits puis
découpés hors de l'\omterm{def:g11:gene-expression:transcription}{ARN} avant qu'il ne quitte le \omterm{def:g10:cells-common-unit:organelle}{noyau}~; les segments
conservés et raboutés, les \emph{exons}, forment l'ARNm mature. De
nombreux \omterm{def:g10:universal-dna:gene}{gènes} peuvent être découpés de plus d'une façon, en gardant
des jeux d'exons différents (l'\emph{épissage alternatif}), si bien
qu'un seul \omterm{def:g10:universal-dna:gene}{gène} donne plusieurs ARNm distincts et plusieurs \omterm{def:g10:chemistry-of-life:families}{protéines}
apparentées~: les $\num{20000}$ \omterm{def:g10:universal-dna:gene}{gènes} humains codent bien plus de
$\num{100000}$ \omterm{def:g10:chemistry-of-life:families}{protéines}.
\end{proposition}
\admitted

\begin{remark}[Le sens du flux d'information]\label{rem:g11:gene-expression:flow}
L'\omterm{def:g10:universal-dna:information}{ADN} est transcrit en \omterm{def:g11:gene-expression:transcription}{ARN}, l'\omterm{def:g11:gene-expression:transcription}{ARN} est traduit en \omterm{def:g11:gene-expression:protein}{protéine}, et la
\omterm{def:g11:gene-expression:protein}{protéine} fait le caractère~: l'information circule dans un seul sens.
Une \omterm{def:g11:gene-expression:protein}{protéine} changée ne réécrit jamais le \omterm{def:g10:universal-dna:gene}{gène}, et c'est pourquoi une
vie d'\omterm{def:g10:sport-and-health:training}{entraînement} ou de coups de soleil ne s'hérite pas, et pourquoi
seules les \omterm{def:g11:mutations:mutation}{mutations} de l'\omterm{def:g10:universal-dna:information}{ADN} (\cref{ch:g11:mutations}) changent ce que
la génération suivante reçoit. Les trois mêmes étapes se déroulent dans
chaque \omterm{def:g10:cells-common-unit:cell}{cellule} de chaque organisme, avec le même code~: l'universalité
du \cref{ch:g10:universal-dna}, désormais jusqu'au dernier mot.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g11:gene-expression:1}
Donner trois différences entre l'\omterm{def:g10:universal-dna:information}{ADN} et l'\omterm{def:g11:gene-expression:transcription}{ARN}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:gene-expression:2}
Transcrire en ARNm le brin transcrit \texttt{3'-TAC CGA AAA ATT-5'}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:gene-expression:3}
Traduire l'ARNm \texttt{AUG GGC AAA UGU UAA} à l'aide de la table du code.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:gene-expression:4}
Quels sont les rôles du \omterm{def:g10:cells-common-unit:organelle}{ribosome} et des \omterm{def:g11:gene-expression:transcription}{ARN} de transfert dans la
\omterm{prop:g11:gene-expression:translation}{traduction}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:gene-expression:5}
Pourquoi le code doit-il utiliser au moins trois \omterm{def:g10:universal-dna:nucleotide}{nucléotides} par \omterm{def:g10:chemistry-of-life:families}{acide
aminé}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:gene-expression:6}
Une \omterm{def:g11:gene-expression:protein}{protéine} compte 412 \omterm{def:g10:chemistry-of-life:families}{acides aminés}. Quelle est la longueur minimale
de la partie codante de son ARNm, \omterm{def:g11:gene-expression:code}{codon} d'arrêt compris, et celle du
\omterm{def:g10:universal-dna:gene}{gène} correspondant en paires de \omterm{def:g10:universal-dna:nucleotide}{bases}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:gene-expression:7}
Le brin codant \texttt{ATG AAA GGC TGG TAA} est muté en
\texttt{ATG AAA GGC TGA TAA}. Donner les deux \omterm{def:g10:chemistry-of-life:families}{protéines} et classer la
\omterm{def:g11:mutations:mutation}{mutation}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:gene-expression:8}
Même séquence de départ, mutée en \texttt{ATG AAG GGC TGG TAA} puis en
\texttt{ATG AAA GGG CTG GTA A}. Classer chacune et donner les \omterm{def:g10:chemistry-of-life:families}{protéines}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:gene-expression:9}
Expliquer, à l'aide de la table, pourquoi une substitution en troisième
position d'un \omterm{def:g11:gene-expression:code}{codon} est souvent silencieuse.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:gene-expression:10}
Un \omterm{def:g11:gene-expression:transcription}{ARN} poly-UC (\texttt{UCUCUCUC}\dots) donne une \omterm{def:g11:gene-expression:protein}{protéine} qui alterne
sérine et leucine. Montrer que c'est compatible avec un code à trois
lettres et s'en servir pour attribuer deux \omterm{def:g11:gene-expression:code}{codons}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:gene-expression:11}
Un \omterm{def:g10:universal-dna:gene}{gène} humain de \num{30000} paires de \omterm{def:g10:universal-dna:nucleotide}{bases} produit une \omterm{def:g11:gene-expression:protein}{protéine} de
500 \omterm{def:g10:chemistry-of-life:families}{acides aminés}. Expliquer l'écart.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:gene-expression:12}
Un médicament bloque les \omterm{def:g10:cells-common-unit:organelle}{ribosomes} bactériens mais non les \omterm{def:g10:cells-common-unit:organelle}{ribosomes}
humains. Expliquer pourquoi il peut être un antibiotique, et ce que son
existence implique au sujet des \omterm{def:g10:cells-common-unit:organelle}{ribosomes} des deux groupes.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:gene-expression:13}
Expliquer, dans le vocabulaire de ce chapitre, comment le \omterm{def:g10:universal-dna:gene}{gène} de
méduse du \cref{ch:g10:universal-dna} a pu être lu par une souris, et
ce qui se passerait si la souris utilisait un autre code.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:gene-expression:14}
Une \omterm{def:g11:mutations:mutation}{mutation} modifie l'ARNt dont l'anticodon lit UAG, de sorte qu'il
apporte un \omterm{def:g10:chemistry-of-life:families}{acide aminé} au lieu d'arrêter la \omterm{prop:g11:gene-expression:translation}{traduction}. Prévoir son
effet sur les \omterm{def:g10:chemistry-of-life:families}{protéines} de la \omterm{def:g10:cells-common-unit:cell}{cellule} en général, et sur un \omterm{def:g10:universal-dna:gene}{gène} mutant

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:gene-expression:15}
Discuter pourquoi un décalage du cadre de lecture près du début d'un
\omterm{def:g10:universal-dna:gene}{gène} est presque toujours plus grave qu'un décalage près de sa fin, et
pourquoi une \omterm{def:g11:mutations:mutation}{mutation} faux-sens peut aller de l'anodin au fatal.
\end{exercise}

\section{Problème~: un gène lu de quatre façons}

\begin{problem}[{Problème du week-end --- un gène court transcrit,
traduit, muté trois fois et chronométré~: de l'ADN aux acides aminés, et
les soixante secondes que prend une protéine}]
\label{pb:g11:gene-expression:1}
Le brin codant d'un \omterm{def:g10:universal-dna:gene}{gène} court se lit~:
\begin{center}
\texttt{ATG GCT TGG AAA CCC GAA TAC GGT CAT TTC TGA}
\end{center}

\medskip
\noindent\textbf{Partie I --- Lire le \omterm{def:g10:universal-dna:gene}{gène}.}
\begin{enumerate}
  \item Écrire le brin transcrit.
  \item Écrire l'ARNm transcrit à partir du \omterm{def:g10:universal-dna:gene}{gène}.
  \item Le traduire, \omterm{def:g11:gene-expression:code}{codon} par \omterm{def:g11:gene-expression:code}{codon}, et donner la séquence d'\omterm{def:g10:chemistry-of-life:families}{acides
        aminés} de la \omterm{def:g11:gene-expression:protein}{protéine}.
  \item Combien d'\omterm{def:g10:chemistry-of-life:families}{acides aminés} la \omterm{def:g11:gene-expression:protein}{protéine} contient-elle, et combien
        de \omterm{def:g10:universal-dna:nucleotide}{nucléotides} de l'ARNm servent à les spécifier, arrêt
        compris~?
  \item Quel \omterm{def:g11:gene-expression:code}{codon} a lancé la lecture, et qu'est-ce qui fixe le cadre
        de lecture de tous les autres~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Trois mutants.}
\begin{enumerate}[resume]
  \item Mutant 1~: \texttt{ATG GCT TGG AAA CCG GAA TAC GGT CAT TTC TGA}.
        Donner la \omterm{def:g11:gene-expression:protein}{protéine} et classer la \omterm{def:g11:mutations:mutation}{mutation}.
  \item Mutant 2~: \texttt{ATG GCT TGG AAA CCC GAA TAC GGT CAT TTC TGA}
        avec le neuvième \omterm{def:g11:gene-expression:code}{codon} \texttt{CAT} remplacé par \texttt{CGT}.
        Donner la \omterm{def:g11:gene-expression:protein}{protéine} et classer.
  \item Mutant 3~: \texttt{ATG GCT TGA AAA CCC GAA TAC GGT CAT TTC TGA}.
        Donner la \omterm{def:g11:gene-expression:protein}{protéine} et classer. Quelle substitution unique l'a
        produit~?
  \item Mutant 4~: un T est inséré après le sixième \omterm{def:g10:universal-dna:nucleotide}{nucléotide}~:
        \texttt{ATG GCT TTG GAA ACC CGA ATA CGG TCA TTT CTG A}. Donner
        la \omterm{def:g11:gene-expression:protein}{protéine} et classer.
  \item Classer les quatre mutants du moins au plus perturbant pour la
        \omterm{def:g11:gene-expression:protein}{protéine} et justifier.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Pourquoi le code est ce qu'il est.}
\begin{enumerate}[resume]
  \item Combien de \omterm{def:g11:gene-expression:code}{codons} spécifient la leucine~? La sérine~? Le
        tryptophane~? Quel \omterm{def:g10:chemistry-of-life:families}{acide aminé} a le plus de chances d'être
        changé par une substitution au hasard dans son \omterm{def:g11:gene-expression:code}{codon}, et lequel
        le moins~?
  \item Calculer la fraction des substitutions en troisième position
        d'un \omterm{def:g11:gene-expression:code}{codon} de la leucine \texttt{CUx} qui sont silencieuses.
  \item Expliquer pourquoi un code à deux lettres ne pourrait pas
        fonctionner, et pourquoi un code à quatre n'est pas nécessaire.
  \item Le poly-U donne de la poly-phénylalanine et le poly-A de la
        poly-lysine. Quelles entrées de la table ces deux expériences
        établissent-elles~?
  \item Le code est le même chez la bactérie et chez l'humain. Énoncer
        la conséquence pour la \omterm{prop:g10:universal-dna:universal}{transgenèse} et la conséquence pour la
        parenté.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Chronométrer l'usine.}
Un \omterm{def:g10:cells-common-unit:organelle}{ribosome} ajoute 10 \omterm{def:g10:chemistry-of-life:families}{acides aminés} par seconde et les \omterm{def:g10:cells-common-unit:organelle}{ribosomes} se
suivent à 80 \omterm{def:g10:universal-dna:nucleotide}{nucléotides} d'intervalle le long d'un ARNm~; chaque ARNm
vit 2 heures.
\begin{enumerate}[resume]
  \item Combien de temps un \omterm{def:g10:cells-common-unit:organelle}{ribosome} met-il à traduire la \omterm{def:g11:gene-expression:protein}{protéine} de
        ce \omterm{def:g10:universal-dna:gene}{gène}~? Et une \omterm{def:g11:gene-expression:protein}{protéine} de 600 \omterm{def:g10:chemistry-of-life:families}{acides aminés}~?
  \item Combien de \omterm{def:g10:cells-common-unit:organelle}{ribosomes} peuvent lire à la fois un ARNm de 1800
        \omterm{def:g10:universal-dna:nucleotide}{nucléotides}~?
  \item Si un nouveau \omterm{def:g10:cells-common-unit:organelle}{ribosome} démarre toutes les 8 secondes, combien
        de copies de la \omterm{def:g11:gene-expression:protein}{protéine} un ARNm donne-t-il au cours de sa
        vie~?
  \item Une bactérie a besoin de \num{2000} exemplaires d'une \omterm{def:g10:cell-metabolism:metabolism}{enzyme} en
        10 minutes. Combien d'ARNm du \omterm{def:g10:universal-dna:gene}{gène}, transcrits en même temps,
        faut-il~?
  \item Énoncer le résultat~: la \omterm{def:g11:gene-expression:protein}{protéine} codée par le \omterm{def:g10:universal-dna:gene}{gène}, celle des
        quatre \omterm{def:g11:mutations:mutation}{mutations} qui l'a abolie, et le temps qu'un \omterm{def:g10:cells-common-unit:organelle}{ribosome} met
        à la construire.
\end{enumerate}
\end{problem}
