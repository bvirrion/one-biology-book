\chapter{Protéger le système nerveux}\label{ch:g8:protecting-the-nervous-system}

Deux chapitres ont montré ce qu'est le \omterm{def:g5:brain-in-command:system}{système nerveux}~: un
câblage dont les jonctions apprennent, décident --- et
peuvent être trompées
(\cref{rem:g8:nervous-communication:vulnerable}). La
clôture de l'année transforme cette anatomie en protection.
Que font vraiment le manque de \omterm{prop:g1:healthy-habits:needs}{sommeil}, le bruit, les écrans
et les substances psychoactives, jonction par jonction, à la
machine qui est toi --- et à quoi ressemble, en pratique, sa
protection, à quatorze ans~?

\section{L'entretien~: le sommeil, et les limites des sens}

\begin{proposition}[Dormir, c'est entretenir ses synapses]\label{prop:g8:protecting-the-nervous-system:sleep}
Le classement et les réparations de
\cref{prop:g5:brain-in-command:protect} se lisent au niveau
des jonctions~: pendant le \omterm{prop:g1:healthy-habits:needs}{sommeil}, les \omterm{def:g8:nervous-communication:synapse}{synapses} renforcées
dans la journée sont consolidées --- «~l'entraînement rend
définitif~» (\cref{ex:g8:nervous-communication:juggler}) se
joue en grande partie la nuit --- et le ménage du \omterm{def:g5:brain-in-command:system}{cerveau}
évacue l'usure chimique du jour. Les nuits courtes laissent
les apprentissages de la veille à moitié classés et la
pesée du lendemain paresseuse~: le \omterm{def:g5:brain-in-command:system}{cerveau} \omterm{def:g3:stages-of-life:stages}{adolescent}, en
plein recâblage (\cref{rem:g8:puberty:moods}) et dont
l'horloge dérive vers le tard, a le plus besoin de ses
heures précisément quand le collège commence le plus tôt.
\end{proposition}
\admitted

\begin{example}[Le paradoxe des révisions]\label{ex:g8:protecting-the-nervous-system:revision}
Deux élèves avant un contrôle~: l'un révise jusqu'à minuit
puis jusqu'à deux heures~; l'autre révise jusqu'à dix
heures et dort. Le second gagne d'habitude --- la matière a
traversé ses jonctions \emph{et a été consolidée}~; les
heures supplémentaires du premier ont coûté le classement
même qui fait tenir une révision. «~La nuit porte
conseil~», c'est de l'ingénierie des \omterm{def:g8:nervous-communication:synapse}{synapses}.
\end{example}

\begin{proposition}[Le fort est une dose]\label{prop:g8:protecting-the-nervous-system:noise}
Les \omterm{def:g5:first-look-at-cells:cell}{cellules} sensorielles de l'\omterm{def:g1:the-five-senses:sense}{oreille} --- les récepteurs
fins comme des cheveux qui changent le son en \omterm{def:g8:nervous-communication:neuron}{messages
nerveux} --- ne repoussent pas
(\cref{prop:g1:the-five-senses:fragile} le disait avec
douceur~; voici le mécanisme). Le son fort les tue par la
dose~: le volume multiplié par la durée. Les concerts et
les écouteurs à fond dépensent, heure après heure, un
capital de récepteurs qu'aucun \omterm{prop:g1:healthy-habits:needs}{sommeil} ne rend --- le
sifflement après une soirée bruyante
(\cref{ex:g1:the-five-senses:loud}) est l'avertissement du
surdosage. La règle est un budget~: moins fort, moins
longtemps, avec des pauses --- et des bouchons là où les
amplificateurs font la loi.
\end{proposition}
\admitted

\section{Les substances qui trompent les jonctions}

\begin{proposition}[Les substances psychoactives agissent aux synapses]\label{prop:g8:protecting-the-nervous-system:substances}
Toute substance qui change l'humeur, la perception ou la
vigilance agit aux espaces de
\cref{def:g8:nervous-communication:synapse} --- en imitant
un messager, en en bloquant un, ou en forçant sa
libération~:
\begin{itemize}
  \item l'\emph{alcool} amortit les jonctions du système
        tout entier~: la boucle ralentit d'un bout à
        l'autre --- le jugement, le \omterm{def:g5:brain-in-command:reaction}{temps de réaction} (les
        mètres de \cref{ex:g5:brain-in-command:driver}
        s'allongent), l'équilibre, la mémoire~; les
        jonctions encore en construction du \omterm{def:g5:brain-in-command:system}{cerveau}
        \omterm{def:g3:stages-of-life:stages}{adolescent} reçoivent l'amortissement le plus
        profond~;
  \item la \emph{nicotine du tabac} imite un messager
        naturel et, répétée, recâble les jonctions de la
        récompense vers la \emph{\omterm{prop:g5:healthy-choices:substances}{dépendance}} (le piège de
        \cref{prop:g5:healthy-choices:substances}, muni
        maintenant de son mécanisme~: les \omterm{def:g8:nervous-communication:synapse}{synapses} se
        règlent pour attendre la dose --- et protestent
        sans elle)~;
  \item le \emph{cannabis} imite une autre famille de
        messagers et embrouille précisément les jonctions
        de la mémoire, de la motivation et du sens du
        temps --- le plus cher, là encore, dans un \omterm{def:g5:brain-in-command:system}{cerveau}
        en chantier~;
  \item plus la substance est dure, plus le forçage des
        jonctions est grossier --- et plus vite s'installe
        le recâblage de la récompense, moteur de la
        \omterm{prop:g5:healthy-choices:substances}{dépendance}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[La dépendance, mécaniquement]\label{rem:g8:protecting-the-nervous-system:addiction}
Le piège de \cref{rem:g5:healthy-choices:never} a
maintenant son schéma de câblage~: les jonctions de la
récompense, dosées à répétition, \emph{s'adaptent} ---
récepteurs raréfiés, seuils relevés --- jusqu'à ce que la
substance soit nécessaire pour se sentir normal, et que son
absence soit une protestation du système. Cette \omterm{def:g4:adapted-to-their-world:adaptation}{adaptation},
c'est l'asymétrie entrée facile, sortie difficile faite de
\omterm{def:g8:nervous-communication:synapse}{synapses}~: commencer est une décision~; arrêter, c'est aller
contre des jonctions recâblées. Le \omterm{def:g5:brain-in-command:system}{cerveau} \omterm{def:g3:stages-of-life:stages}{adolescent},
encore réglable, se recâble le plus vite --- et c'est
pourquoi toutes les statistiques trouvent les premiers
partis les plus profondément piégés, et pourquoi
«~ne jamais commencer~» reste le coup le moins cher.
\end{remark}

\begin{example}[La boucle des écrans]\label{ex:g8:protecting-the-nervous-system:screens}
Aucune molécule, les mêmes jonctions~: les fils d'actualité
et les jeux sont réglés --- exprès --- pour chatouiller le
câblage de la récompense avec de petites récompenses
imprévisibles, le rythme qui entraîne le plus fort les
jonctions. Le résultat est un cousin doux du piège~: on
vérifie et on n'arrive pas à s'arrêter, les soirées
s'évaporent, le \omterm{prop:g1:healthy-habits:needs}{sommeil} est repoussé (les heures de
\cref{prop:g5:healthy-choices:screens}, plus les coûts de
\cref{prop:g8:protecting-the-nervous-system:sleep}). Même
défense que toujours~: décider les heures à l'avance, et
garder la dernière sans écran.
\end{example}

\section{En pratique~: assurer soi-même sa protection}

\begin{method}[Le manuel du propriétaire d'un système nerveux]\label{met:g8:protecting-the-nervous-system:manual}
La liste d'entretien de \cref{met:g5:healthy-choices:run},
rééditée avec ses mécanismes~:
\begin{enumerate}
  \item \emph{dors tes heures} --- la consolidation et le
        ménage tournent la nuit~; protège la dernière
        heure des écrans~;
  \item \emph{fais le budget du volume} --- le capital de
        récepteurs ne repousse pas~: volume baissé, pauses
        prises, bouchons là où ça rugit~;
  \item \emph{le casque} là où les chutes sont rapides
        (\cref{ex:g5:brain-in-command:helmet}) --- un
        câblage ne se ressoude pas comme un \omterm{def:g3:bones-and-muscles:skeleton}{os}~;
  \item \emph{pas de trompeurs de jonctions} pour un
        \omterm{def:g5:brain-in-command:system}{cerveau} en chantier --- et le «~non~» décidé
        calmement à l'avance (la phrase de poche de
        \cref{ex:g5:healthy-choices:answers}, munie
        maintenant de toute sa justification)~;
  \item \emph{sers-toi de la machine} --- apprendre, faire
        du sport, de la musique~: les jonctions se
        renforcent avec le trafic, et une boucle entraînée
        est le capital que toutes les autres habitudes
        protègent.
\end{enumerate}
\end{method}

\begin{example}[L'aide existe, et elle marche]\label{ex:g8:protecting-the-nervous-system:help}
Pour toute personne déjà prise --- sa propre habitude ou
celle d'un ami, substances ou écrans~: les adresses de
\cref{ex:g8:contraception:help} servent ici aussi. Les
infirmières scolaires et les médecins traitent la
\omterm{prop:g5:healthy-choices:substances}{dépendance} pour ce que ce chapitre montre qu'elle est ---
un problème de câblage, pas un verdict sur le caractère ---
et des jonctions recâblées peuvent se régler à nouveau~:
c'est plus dur que de ne jamais commencer, c'est loin
d'être désespéré. Le pire plan est celui de la cour de
récréation~: le silence.
\end{example}

\section{Problème~: la soirée prévention}

\begin{problem}[{Problème du week-end --- la soirée
parents-élèves du collège, écrite avec ses
mécanismes}]\label{pb:g8:protecting-the-nervous-system:1}
La classe prépare la soirée prévention du collège~: quatre
stands --- \omterm{prop:g1:healthy-habits:needs}{sommeil}, son, substances, écrans --- chacun avec
une démonstration et une fiche mécanisme.

\medskip\noindent\textbf{Partie I --- Les fiches des
stands.}
\begin{enumerate}
  \item Écris la fiche \omterm{prop:g1:healthy-habits:needs}{sommeil}~: la démonstration est le
        score de deux volontaires au test de la règle
        (\cref{met:g5:brain-in-command:ruler}), reposé
        contre nuit courte. Prédis les scores et donne le
        mécanisme au niveau des jonctions.
  \item Écris la fiche son~: un sonomètre, une paire de
        bouchons, et la phrase «~on ne les fait pas
        repousser~». Complète le mécanisme.
  \item Écris la fiche substances~: un schéma de \omterm{def:g8:nervous-communication:synapse}{synapse},
        trois flèches étiquetées imiter, bloquer, forcer.
        Attribue l'alcool, la nicotine et un venin de
        \cref{exo:g8:nervous-communication:11} à leurs
        flèches.
  \item Écris la fiche écrans~: aucune molécule --- alors
        qu'est-ce que la démonstration du téléphone à
        notifications partage avec le stand des
        substances~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les questions de la
salle.}
\begin{enumerate}[resume]
  \item Un parent~: «~nous, on buvait à leur âge et nous
        voilà très bien.~» Réponds par l'asymétrie du
        chantier --- qu'est-ce qui diffère à quatorze ans~?
  \item Un élève~: «~une cigarette ne peut rien recâbler
        du tout.~» Réponds avec
        \cref{rem:g8:protecting-the-nervous-system:addiction}
        et \cref{rem:g5:healthy-choices:never} --- quel est
        le vrai prix de la première dose~?
  \item Un parent~: «~les écrans, ce n'est pas la
        télévision nouvelle version~?~» Distingue les
        rythmes~: qu'est-ce qui fait des petites récompenses
        imprévisibles le dresseur le plus dur~?
  \item Un élève, à voix basse~: «~et si quelqu'un
        n'arrive déjà plus à arrêter~?~» Donne la réponse
        de \cref{ex:g8:protecting-the-nervous-system:help},
        adresses comprises.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le mot de la
fin.}
\begin{enumerate}[resume]
  \item Le mot de la fin s'ouvre sur la jongleuse et se
        ferme sur la \omterm{prop:g5:healthy-choices:substances}{dépendance} --- «~la même capacité de
        réglage, lue deux fois~». Rédige ce plaidoyer en
        quatre phrases.
  \item Il remet ensuite à chaque élève la phrase de poche
        de \cref{ex:g5:healthy-choices:answers}. Explique,
        mécanisme compris, pourquoi cette phrase doit être
        composée \emph{maintenant}.
  \item L'affiche de la soirée montre les cinq points du
        manuel du propriétaire. Range-les pour la semaine
        d'un jeune de quatorze ans, du plus protecteur au
        moins, et défends ton premier choix.
  \item Termine la soirée en une phrase~: qu'est-ce qui est
        protégé, et pourquoi quatorze ans est-il le milieu
        de la décennie décisive~?
\end{enumerate}
\end{problem}
