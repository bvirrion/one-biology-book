\chapter{कोशिकीय श्वसन और किण्वन}\label{ch:b1:respiration-fermentation}

हवा से अलग रखे शर्करा-विलयन में खमीर का कोई फ़्लास्क कार्बन डाइऑक्साइड के
बुलबुले छोड़ता है और शर्करा को ऐल्कोहॉल में बदल देता है; हवा भीतर आने
दीजिए और बुलबुले धीमे पड़ जाते हैं, ऐल्कोहॉल बनना रुक जाता है, और उसी
शर्करा पर खमीर दस गुना तेज़ बढ़ता है। पाश्चर ने 1861 में यह देखा और
समझा नहीं सके; व्याख्या में एक सदी लगी और वह पूरे इस अध्याय में चलती है।
ऑक्सीजन किसी \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को ग्लूकोज़ से उससे पंद्रह गुना ऊर्जा निकालने देती है
जितनी वह उसके बिना पा सकती है, और वह ऐसा शर्करा के इलेक्ट्रॉनों को
वाहकों की एक शृंखला से नीचे ऑक्सीजन तक भेजकर करती है, गिरते हुए उनसे किसी
झिल्ली के आरपार प्रोटॉन पंप करवाकर, और प्रोटॉनों को ऐसे टरबाइन से वापस
आने देकर जो \omterm{def:b1:water-small-molecules:atp}{ATP} बनाता है। यह अध्याय \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस}, \omterm{def:b1:respiration-fermentation:krebs}{क्रेब्स चक्र}, \omterm{def:b1:respiration-fermentation:chain}{श्वसन
शृंखला} और उसके रसोपरासरणी युग्मन, उन किण्वनों का जो बिना ऑक्सीजन के काम
चला लेते हैं, और उसी मशीनरी से \omterm{def:b1:lipids:triglyceride}{वसाओं} तथा \omterm{def:b1:proteins:peptide}{प्रोटीनों} के जलने का वर्णन करता
है।

\section{ग्लूकोज़ का ऑक्सीकरण}

\begin{proposition}[समग्र अभिक्रिया और उसके चरण]\label{prop:b1:respiration-fermentation:overview}
\index{कोशिकीय श्वसन}
ग्लूकोज़ का पूर्ण ऑक्सीकरण,
\[
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \to 6\,\mathrm{CO_2} + 6\,\mathrm{H_2O},
\qquad \Delta G^{\circ\prime} = \qty{-2870}{kJ/mol},
\]
तीन चरणों में होता है, और उनमें इलेक्ट्रॉन कभी सीधे ऑक्सीजन से नहीं
मिलते। \emph{\omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस}}, जो \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} में है, ग्लूकोज़ को
दो पाइरुवेट में तोड़ता है और 2 \omterm{def:b1:water-small-molecules:atp}{ATP} तथा 2 NADH देता है। \emph{पाइरुवेट
ऑक्सीकरण और \omterm{def:b1:respiration-fermentation:krebs}{क्रेब्स चक्र}}, जो \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका आधात्री} में हैं, पाइरुवेट का
$\mathrm{CO_2}$ तक ऑक्सीकरण करते हैं, और इलेक्ट्रॉन 8 NADH तथा
2 $\mathrm{FADH_2}$ पर लादकर 2 GTP बनाते हैं। \emph{ऑक्सीकारक
फ़ॉस्फ़ोरिलीकरण}, जो \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका} की भीतरी झिल्ली में है, उन
इलेक्ट्रॉनों को वाहकों की उस शृंखला से होकर ऑक्सीजन तक पहुँचाता है जो
प्रोटॉन पंप करती है, और प्रोटॉन प्रवणता लगभग 26 \omterm{def:b1:water-small-molecules:atp}{ATP} का संश्लेषण चलाती है।
कुल मिलाकर प्रति ग्लूकोज़ कोई 30 \omterm{def:b1:water-small-molecules:atp}{ATP}: यानी दहन की \omterm{prop:b1:water-small-molecules:gibbs}{मुक्त ऊर्जा} का लगभग
आधा पकड़ा गया, और बाक़ी ऊष्मा के रूप में छूटा।
\end{proposition}

\begin{definition}[इलेक्ट्रॉन वाहक]\label{def:b1:respiration-fermentation:carriers}
\index{NAD}\index{FAD}
\emph{$\mathrm{NAD^+}$} (निकोटिनामाइड ऐडेनीन डाइन्यूक्लियोटाइड) किसी
\omterm{def:b1:enzymes:enzyme}{क्रियाधार} से दो इलेक्ट्रॉन और एक प्रोटॉन लेकर NADH बन जाता है; \emph{FAD}
दो इलेक्ट्रॉन और दो प्रोटॉन लेकर $\mathrm{FADH_2}$।
दोनों डिहाइड्रोजनेज़ों के \omterm{def:b1:enzymes:enzyme}{सहएंजाइम} हैं, और दोनों पुनःचक्रित होते हैं:
अपचय से अपचयित, और \omterm{def:b1:respiration-fermentation:chain}{श्वसन शृंखला} से फिर ऑक्सीकृत। \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} इनके केवल
माइक्रोमोल रखती है, और वे घंटे भर में हज़ारों बार पलटते हैं। उनके
अपचयोपचय विभव, यानी $E'_0 = \qty{-0.32}{V}$, जो
$\mathrm{NAD^+}/\mathrm{NADH}$ का है, उन्हें ऑक्सीजन ($\qty{+0.82}{V}$) से बहुत नीचे रखते हैं: NADH जो भी
इलेक्ट्रॉन-युग्म ऑक्सीजन को देता है वह $\Delta G^{\circ\prime} = -2F\Delta E =
\qty{-220}{kJ/mol}$ छोड़ता है, जो चार से अधिक \omterm{def:b1:water-small-molecules:atp}{ATP} के लिए काफ़ी है।
\end{definition}

\section{ग्लाइकोलिसिस}

\begin{definition}[ग्लाइकोलिसिस]\label{def:b1:respiration-fermentation:glycolysis}
\index{ग्लाइकोलिसिस}
\emph{ग्लाइकोलिसिस} \omterm{def:b1:eukaryotic-cell:organelle}{कोशिकाद्रव} में दस \omterm{def:b1:enzymes:enzyme}{एंजाइम} अभिक्रियाओं का वह क्रम है
जो एक ग्लूकोज़ (C6) को दो \emph{पाइरुवेट} (C3) में बदल देता है।
\emph{निवेश प्रावस्था} में शर्करा का फ़ॉस्फ़ोरिलीकरण करने पर दो \omterm{def:b1:water-small-molecules:atp}{ATP}
ख़र्च होते हैं (हेक्सोकाइनेज़, फिर \emph{फ़ॉस्फ़ोफ्रक्टोकाइनेज़}, PFK)
और छह-कार्बन डाइफ़ॉस्फ़ेट दो तीन-कार्बन फ़ॉस्फ़ेटों में टूट जाता है।
\emph{लाभ प्रावस्था} में हर ट्रायोस का $\mathrm{NAD^+}$ से ऑक्सीकरण होता
है (जो ग्लाइकोलिसिस का इकलौता ऑक्सीकरण है) और उसके फ़ॉस्फ़ेट
\emph{क्रियाधार-स्तरीय फ़ॉस्फ़ोरिलीकरण}\index{क्रियाधार-स्तरीय फ़ॉस्फ़ोरिलीकरण} से ADP पर पहुँचा दिए जाते हैं --- यानी किसी उच्च-ऊर्जा
मध्यवर्ती से सीधे ADP को दिया गया कोई फ़ॉस्फ़ेट, बिना किसी झिल्ली या
ऑक्सीजन के। बहीखाता:
\[
\text{ग्लूकोज़} + 2\,\mathrm{NAD^+} + 2\,\mathrm{ADP} + 2\,\mathrm{P_i}
\to 2\,\text{पाइरुवेट} + 2\,\mathrm{NADH} + 2\,\mathrm{H^+} + 2\,\mathrm{ATP} + 2\,\mathrm{H_2O} .
\]
उसे ऑक्सीजन नहीं चाहिए; और वह उपापचय का सबसे पुराना तथा सबसे सार्वत्रिक
पथ है।
\end{definition}

\begin{proposition}[फ़ॉस्फ़ोफ्रक्टोकाइनेज़ पर नियंत्रण]\label{prop:b1:respiration-fermentation:pfk}
PFK, जो \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} के प्रति प्रतिबद्ध पहला अनुत्क्रमणीय चरण है, एक
\omterm{def:b1:enzymes:allosteric}{अन्यस्थानिक एंजाइम} है (\cref{ch:b1:enzymes}) जिसे \omterm{def:b1:water-small-molecules:atp}{ATP} और साइट्रेट
संदमित करते हैं --- यानी वे संकेत कि ऊर्जा तथा क्रेब्स-चक्र का ईंधन
प्रचुर है --- और जिसे AMP तथा ADP सक्रिय करते हैं, यानी वे संकेत कि वे
प्रचुर नहीं हैं। \omterm{def:b1:water-small-molecules:atp}{ATP} ऊँचा होने पर \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} का प्रवाह सेकंडों में गिर
जाता है; और \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} के \omterm{def:b1:water-small-molecules:atp}{ATP} ख़र्च करते ही AMP तेज़ी से चढ़ता है (ऐडेनिलेट
काइनेज़ से, $2\,\mathrm{ADP} \rightleftharpoons \mathrm{ATP} + \mathrm{AMP}$), और
PFK खुल जाता है। \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} का \emph{ऊर्जा आवेश} उसी पथ को नियंत्रित करता है
जो उसे भरता है।
\end{proposition}

\section{पाइरुवेट ऑक्सीकरण और क्रेब्स चक्र}

\begin{definition}[पाइरुवेट डिहाइड्रोजनेज़, ऐसीटिल-CoA]\label{def:b1:respiration-fermentation:pdh}
पाइरुवेट \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका} में घुसता है और \emph{पाइरुवेट
डिहाइड्रोजनेज़} संकुल द्वारा \emph{ऐसीटिल-CoA}\index{ऐसीटिल-CoA} में
ऑक्सीकृत होता है --- यानी \omterm{def:b1:enzymes:enzyme}{सहएंजाइम} A (\cref{ch:b1:nucleic-acids}) द्वारा
किसी उच्च-ऊर्जा थायोएस्टर बंध पर ढोया गया दो-कार्बन ऐसीटिल समूह --- और
इसमें एक $\mathrm{CO_2}$ तथा एक NADH छूटते हैं। ऐसीटिल-CoA चक्र में सारे
ईंधनों का साझा प्रवेश है: \omterm{def:b1:carbohydrates:monosaccharide}{कार्बोहाइड्रेट} पाइरुवेट से, \omterm{def:b1:lipids:triglyceride}{वसाएँ}
$\beta$-ऑक्सीकरण से, और अनेक \omterm{def:b1:proteins:aminoacid}{ऐमीनो अम्ल} सीधे।
\end{definition}

\begin{definition}[क्रेब्स चक्र]\label{def:b1:respiration-fermentation:krebs}
\index{क्रेब्स चक्र}\index{सिट्रिक अम्ल चक्र}
\omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका आधात्री} का \emph{क्रेब्स चक्र} (सिट्रिक अम्ल चक्र)
\omterm{def:b1:respiration-fermentation:pdh}{ऐसीटिल-CoA} (C2) को ऑक्सैलोऐसीटेट (C4) से संघनित करके साइट्रेट (C6) बनाता
है और, आठ चरणों में, उन दो कार्बनों का $\mathrm{CO_2}$ तक ऑक्सीकरण करता
है तथा ऑक्सैलोऐसीटेट का पुनर्जनन कर देता है। प्रति ऐसीटिल समूह: 3 NADH,
1 $\mathrm{FADH_2}$, 1 GTP (\omterm{def:b1:respiration-fermentation:glycolysis}{क्रियाधार-स्तरीय फ़ॉस्फ़ोरिलीकरण} से) और
2 $\mathrm{CO_2}$। प्रति ग्लूकोज़ (दो ऐसीटिल): 6 NADH, 2 $\mathrm{FADH_2}$, 2 GTP, 4 $\mathrm{CO_2}$
--- जो पाइरुवेट डिहाइड्रोजनेज़ के 2 $\mathrm{CO_2}$ के साथ मिलकर समग्र

\begin{definition}[श्वसन शृंखला]\label{def:b1:respiration-fermentation:chain}
\index{श्वसन शृंखला}\index{इलेक्ट्रॉन अभिगमन शृंखला}
\emph{श्वसन शृंखला} \omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिका} की भीतरी झिल्ली के चार \omterm{def:b1:proteins:peptide}{प्रोटीन} संकुल और
दो गतिशील वाहक हैं। NADH अपने इलेक्ट्रॉन \emph{संकुल I} को देता है,
$\mathrm{FADH_2}$ (सक्सिनेट डिहाइड्रोजनेज़, यानी \emph{संकुल II} के

\emph{संकुल III} को देता है, जो उन्हें \emph{साइटोक्रोम $c$} को सौंप
देता है, यानी बाहरी फलक के किसी छोटे \omterm{def:b1:proteins:peptide}{प्रोटीन} को; और
\emph{संकुल IV} (साइटोक्रोम ऑक्सीडेज़) उन्हें, एक बार में चार,
$\mathrm{O_2}$ तक पहुँचाकर जल बना देता है। संकुल I, III और IV में से हर
एक पर इलेक्ट्रॉन विभव में गिरते हैं और वह ऊर्जा आधात्री से अंतःझिल्ली

\begin{theorem}[रसोपरासरणी युग्मन]\label{thm:b1:respiration-fermentation:chemiosmosis}
\index{रसोपरासरण सिद्धांत}\index{प्रोटॉन-प्रेरक बल}
शृंखला की बनाई प्रोटॉन प्रवणता --- यानी लगभग \qty{0.16}{V} का
\omterm{def:b1:membranes-transport:potential}{झिल्ली विभव} (आधात्री ऋणात्मक) और एक pH मात्रक, जो मिलकर लगभग
\qty{0.22}{V} का \emph{\omterm{thm:b1:respiration-fermentation:chemiosmosis}{प्रोटॉन-प्रेरक बल}} या प्रति मोल प्रोटॉन
\qty{21}{kJ} है --- इलेक्ट्रॉन अभिगमन और \omterm{def:b1:water-small-molecules:atp}{ATP} संश्लेषण के बीच का इकलौता
जोड़ है। भीतरी झिल्ली का घूर्णी प्रेरक \omterm{thm:b1:photosynthesis:chemiosmosis}{ATP सिंथेज़} प्रति बने \omterm{def:b1:water-small-molecules:atp}{ATP} लगभग
\qty{3}{H^+} निकलने देता है, और एक और फ़ॉस्फ़ेट भीतर लाने तथा \omterm{def:b1:water-small-molecules:atp}{ATP} बाहर
भेजने में ख़र्च होता है: यानी प्रति \omterm{def:b1:water-small-molecules:atp}{ATP} 4। इसलिए
\emph{P/O अनुपात}\index{P/O अनुपात}: प्रति NADH $10/4 = 2.5$ \omterm{def:b1:water-small-molecules:atp}{ATP} और
$6/4 = 1.5$ प्रति $\mathrm{FADH_2}$।
\end{theorem}
\begin{proof}[साक्ष्य]
\omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाएँ} \omterm{def:b1:water-small-molecules:atp}{ATP} केवल तभी बनाती हैं जब उनकी भीतरी झिल्ली साबुत और बंद
हो; जो टुकड़े इलेक्ट्रॉन तो ढोते हैं पर प्रवणता नहीं थाम सकते वे कुछ नहीं
बनाते। डाइनाइट्रोफ़ीनॉल जैसे अयुग्मक, जो झिल्ली के आरपार प्रोटॉन ढोते
हैं, \omterm{def:b1:water-small-molecules:atp}{ATP} संश्लेषण मिटा देते हैं जबकि ऑक्सीजन-खपत चलती रहती है और तेज़ भी
हो जाती है, और ऊर्जा ऊष्मा के रूप में निकल जाती है --- भूरी \omterm{def:b1:lipids:triglyceride}{वसा} यही
जान-बूझकर अपने अयुग्मक \omterm{def:b1:proteins:peptide}{प्रोटीन} से करती है
(\cref{ch:b1:mammal-organization}), और डाइनाइट्रोफ़ीनॉल कभी दुबले करने
की दवा के रूप में बिकता था जो अतिताप से मार डालती थी। शृंखला के \omterm{def:b1:enzymes:inhibitors}{संदमक}
(संकुल IV पर सायनाइड) दोनों रोक देते हैं; सिंथेज़ के \omterm{def:b1:enzymes:inhibitors}{संदमक}
(ऑलिगोमाइसिन) \omterm{def:b1:water-small-molecules:atp}{ATP} संश्लेषण रोक देते हैं और, प्रवणता के निकल न पाने से,
श्वसन भी रोक देते हैं, जिसे फिर कोई अयुग्मक लौटा देता है। अँधेरे में
\omterm{def:b1:eukaryotic-cell:mitochondrion}{सूत्रकणिकाओं} पर थोपी गई कोई कृत्रिम pH प्रवणता \omterm{def:b1:water-small-molecules:atp}{ATP} संश्लेषण चला देती है,
जैसे \omterm{def:b1:eukaryotic-cell:plastid}{हरितलवकों} में (\cref{ch:b1:photosynthesis})।
\end{proof}

  \item ऑक्सीकारक फ़ॉस्फ़ोरिलीकरण: प्रति सूत्रकणिकीय NADH $2.5$ \omterm{def:b1:water-small-molecules:atp}{ATP}
        (8: 20 \omterm{def:b1:water-small-molecules:atp}{ATP}), और $1.5$ प्रति $\mathrm{FADH_2}$ (2: 3 \omterm{def:b1:water-small-molecules:atp}{ATP})।
  \item कोशिकाद्रवी NADH भीतरी झिल्ली पार नहीं कर सकता; उसके इलेक्ट्रॉन
        किसी शटल से घुसते हैं जो उन्हें $\mathrm{FAD}$ तक पहुँचाता है
        (प्रति एक 1.5, पेशी तथा मस्तिष्क में) या $\mathrm{NAD^+}$ तक
        (प्रति एक 2.5, यकृत तथा हृदय में): यानी 3 या 5 \omterm{def:b1:water-small-molecules:atp}{ATP}।
  \item कुल: $2 + 2 + 20 + 3 + 3 = 30$, या बेहतर शटल के साथ $32$।
        दक्षता: कोशिकीय परिस्थितियों में
        $30\times 50/2870 \approx \qty{52}{\%}$।
\end{enumerate}
\end{method}

\section{किण्वन}

\begin{definition}[किण्वन]\label{def:b1:respiration-fermentation:fermentation}
\index{किण्वन}
ऑक्सीजन के बिना \omterm{def:b1:respiration-fermentation:chain}{श्वसन शृंखला} रुक जाती है, NADH फिर ऑक्सीकृत नहीं हो
सकता, और $\mathrm{NAD^+}$ की कमी से \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} सेकंडों में ठप हो जाता।
\emph{किण्वन} स्वयं पाइरुवेट को इलेक्ट्रॉन-ग्राही बनाकर उसका पुनर्जनन कर
देता है: \emph{लैक्टिक किण्वन} में (पेशी, लाल \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ},
लैक्टिक जीवाणु) पाइरुवेट लैक्टेट में अपचयित होता है; और \emph{ऐल्कोहॉली
किण्वन} में (खमीर, कुछ पौधे) वह विकार्बोक्सिलित होकर ऐसीटैल्डिहाइड बनता
है, जो एथेनॉल में अपचयित होता है और $\mathrm{CO_2}$ छोड़ता है। उपज
\omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} की अपनी है: प्रति ग्लूकोज़ 2 \omterm{def:b1:water-small-molecules:atp}{ATP}, यानी श्वसन का पंद्रहवाँ
भाग; कार्बन मुश्किल से ऑक्सीकृत होता है और उत्पाद ईंधन की लगभग सारी
ऊर्जा अब भी रखते हैं। \emph{अनॉक्सी श्वसन} --- यानी ऐसी
इलेक्ट्रॉन-अभिगमन शृंखला जो ऑक्सीजन के बजाय नाइट्रेट, सल्फ़ेट या
कार्बोनेट पर समाप्त होती है, जैसी अनेक जीवाणुओं में है --- अलग चीज़ है,
और वर्ष 2 के खंड में दी गई है।
\end{definition}

\begin{proposition}[पाश्चर प्रभाव]\label{prop:b1:respiration-fermentation:pasteur}
\index{पाश्चर प्रभाव}
जो \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} श्वसन कर सकती है वह ऑक्सीजन से वंचित होने पर ऑक्सीजन मिलने की
तुलना में कहीं तेज़ी से ग्लूकोज़ खपाती है, और उस पर कहीं कम बढ़ती है।
\end{proposition}
\begin{proof}[साक्ष्य]
पाश्चर (1861) ने पाया कि हवादार किसी बर्तन में खमीर शर्करा धीरे खपाता है
और बढ़ता जाता है, जबकि बंद बर्तन में वह शर्करा दस गुना तेज़ खपाता है,
कम बढ़ता है और ऐल्कोहॉल बनाता है। व्याख्या \omterm{def:b1:water-small-molecules:atp}{ATP} की उपज है: 30 के बजाय
प्रति ग्लूकोज़ 2 से उतना ही \omterm{def:b1:water-small-molecules:atp}{ATP} पाने के लिए \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} को
\omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस} पंद्रह गुना तेज़ चलाना पड़ता है, और PFK, जो \omterm{def:b1:water-small-molecules:atp}{ATP} तथा
साइट्रेट के उस संदमन से मुक्त हो जाता है जिसे श्वसन बनाए रखता है, उसे यह
करने देता है। दौड़ती कोई पेशी सेकंडों में वही प्रभाव दिखाती है: उसका
\omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} विश्राम से बीस गुना तेज़ गिरता है, और लैक्टेट चढ़ता है।
\end{proof}

\begin{definition}[श्वसन गुणांक]\label{def:b1:respiration-fermentation:rq}
\index{श्वसन गुणांक}
\emph{श्वसन गुणांक} RQ बनी $\mathrm{CO_2}$ का खपी $\mathrm{O_2}$ से
अनुपात है, मोल में। \omterm{def:b1:carbohydrates:monosaccharide}{कार्बोहाइड्रेट} 1.0 देता है
(प्रति ग्लूकोज़ हर एक के छह); \omterm{def:b1:lipids:triglyceride}{वसा} लगभग 0.7 (पामिटेट:
$\mathrm{C_{16}H_{32}O_2} + 23\,\mathrm{O_2} \to 16\,\mathrm{CO_2} +
16\,\mathrm{H_2O}$, $16/23 = 0.70$), क्योंकि उसके कार्बन अधिक अपचयित हैं
और उन्हें प्रति $\mathrm{CO_2}$ अधिक ऑक्सीजन चाहिए; और \omterm{def:b1:proteins:peptide}{प्रोटीन} लगभग 0.8।
\omterm{def:b1:respiration-fermentation:fermentation}{किण्वन} करता कोई संवर्धन बिना ऑक्सीजन लिए $\mathrm{CO_2}$ छोड़ता है: उसका
RQ अनंत है, और किसी मिश्रित संवर्धन का 1 से ऊपर का RQ \omterm{def:b1:respiration-fermentation:fermentation}{किण्वन} का हिस्सा
माप देता है। किसी पूरे जंतु पर गैस-विश्लेषण से मापा गया RQ बता देता है कि
वह कौन-सा ईंधन जला रहा है।
\end{definition}

\begin{example}[कोई RQ पढ़ना]\label{ex:b1:respiration-fermentation:readrq}
\omterm{def:b1:carbohydrates:monosaccharide}{कार्बोहाइड्रेट} के भोजन के बाद विश्राम कर रहा कोई मनुष्य: RQ 0.95, यानी
ग्लूकोज़ जल रहा है। वही व्यक्ति रात भर के उपवास के बाद: 0.75, यानी \omterm{def:b1:lipids:triglyceride}{वसा}।
तीसवें किलोमीटर पर कोई मैराथन धावक, जिसका \omterm{def:b1:carbohydrates:polysaccharide}{ग्लाइकोजन} चुक गया है: 0.72।
थोड़ी हवा के साथ ग्लूकोज़ पर खमीर का कोई फ़्लास्क: RQ 4 --- यानी उसके
ग्लूकोज़ का तीन-चौथाई किण्वित हो रहा है।
\end{example}

\section{समस्या: हवा के साथ और बिना हवा के खमीर}

\begin{problem}[{सप्ताहांत समस्या --- पाश्चर के बर्तन फिर से देखे गए:
ATP गिना गया, ग्लूकोज़ खपा, जैवभार बढ़ा और गैसें मापी गईं, और अंत में
ATP उपजों का अनुपात तथा P/O अनुपात}]
\label{pb:b1:respiration-fermentation:1}
खमीर को दो एक जैसे फ़्लास्कों में ग्लूकोज़ पर उगाया जाता है, एक हवादार
और एक बंद। 2.5 (NADH) तथा 1.5 ($\mathrm{FADH_2}$) के \omterm{thm:b1:respiration-fermentation:chemiosmosis}{P/O अनुपात},
कोशिकाद्रवी NADH के लिए $\mathrm{FADH_2}$ शटल, प्रति मोल \omterm{def:b1:water-small-molecules:atp}{ATP}
\qty{10.5}{g} शुष्क जैवभार की वृद्धि-उपज, और ग्लूकोज़
\qty{180}{g/mol} लीजिए।

\medskip
\noindent\textbf{भाग I --- \omterm{def:b1:water-small-molecules:atp}{ATP} की गिनती।}
\begin{enumerate}
  \item \omterm{def:b1:respiration-fermentation:glycolysis}{ग्लाइकोलिसिस}, पाइरुवेट डिहाइड्रोजनेज़ और \omterm{def:b1:respiration-fermentation:krebs}{क्रेब्स चक्र} द्वारा एक
        ग्लूकोज़ से बने \omterm{def:b1:water-small-molecules:atp}{ATP}, GTP, NADH तथा $\mathrm{FADH_2}$ गिनाइए।
  \item प्रति ग्लूकोज़ शृंखला द्वारा पंप किए गए प्रोटॉन निकालिए (प्रति
        सूत्रकणिकीय NADH 10, प्रति $\mathrm{FADH_2}$ 6; कोशिकाद्रवी NADH
        $\mathrm{FADH_2}$ के रूप में घुसता है)।
  \item प्रति \omterm{def:b1:water-small-molecules:atp}{ATP} \qty{4}{H^+} पर सिंथेज़ द्वारा बना \omterm{def:b1:water-small-molecules:atp}{ATP} निकालिए,
        और हवा के साथ प्रति ग्लूकोज़ कुल \omterm{def:b1:water-small-molecules:atp}{ATP}।
  \item बिना हवा के प्रति ग्लूकोज़ \omterm{def:b1:water-small-molecules:atp}{ATP} निकालिए।
  \item दोनों उपजों का अनुपात निकालिए।
  \item हवा के साथ पकड़ी गई \omterm{prop:b1:water-small-molecules:gibbs}{मुक्त ऊर्जा} ($\Delta G_{\text{ATP}}
        = \qty{-50}{kJ/mol}$) \qty{2870}{kJ} के भिन्न के रूप में
        निकालिए, और बिना हवा के उसी \qty{2870}{kJ} के भिन्न के रूप में
        तथा \omterm{def:b1:respiration-fermentation:fermentation}{किण्वन} से सचमुच मुक्त हुई ऊर्जा के भिन्न के रूप में भी
        (ग्लूकोज़ से दो एथेनॉल और दो $\mathrm{CO_2}$: \qty{-235}{kJ/mol})।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- खपा ग्लूकोज़, बढ़ा जैवभार।}
अपनी \omterm{def:b1:cell-unit-of-life:cell}{कोशिकाएँ} बनाए रखने के लिए बंद फ़्लास्क को उतनी ही दर से \omterm{def:b1:water-small-molecules:atp}{ATP} बनाना
पड़ता है जितनी हवादार फ़्लास्क को: प्रति घंटा \qty{30}{mmol} \omterm{def:b1:water-small-molecules:atp}{ATP}।
\begin{enumerate}[resume]
  \item हर फ़्लास्क में प्रति घंटा खपा ग्लूकोज़ निकालिए।
  \item बंद फ़्लास्क में प्रति घंटा बने एथेनॉल तथा $\mathrm{CO_2}$
        निकालिए, और हवादार फ़्लास्क में विनिमय हुई $\mathrm{CO_2}$ तथा
        $\mathrm{O_2}$।
  \item बंद फ़्लास्क प्रति मोल ग्लूकोज़ और प्रति ग्राम ग्लूकोज़ कितना
        जैवभार बना सकता है, यह निकालिए।
  \item हवादार फ़्लास्क प्रति ग्राम ग्लूकोज़ \qty{0.5}{g} जैवभार बनाता
        है। \qty{10.5}{g/mol} पर यह कितना \omterm{def:b1:water-small-molecules:atp}{ATP} दर्शाता है, यह निकालिए,
        और उपलब्ध 30 \omterm{def:b1:water-small-molecules:atp}{ATP} से तुलना कीजिए। ऑक्सी उपज को क्या सीमित करता
        है?
  \item प्रश्न 7 और 9 से \omterm{prop:b1:respiration-fermentation:pasteur}{पाश्चर प्रभाव} समझाइए, और उस \omterm{def:b1:enzymes:enzyme}{एंजाइम} का नाम
        बताइए जिसका नियमन उसे लागू करता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- मापी गई गैसें।}
कोई तीसरा फ़्लास्क, जो कम हवादार है, प्रति घंटा \qty{44}{mg}
$\mathrm{CO_2}$ छोड़ता है और \qty{20}{mg} $\mathrm{O_2}$ खपाता है।
\begin{enumerate}[resume]
  \item दोनों को मिलीमोल में बदलिए और RQ निकालिए।
  \item मान लीजिए प्रति घंटा श्वसित ग्लूकोज़ $x$ है और किण्वित
        ग्लूकोज़ $y$। $\mathrm{O_2}$ तथा $\mathrm{CO_2}$ के बहीखाते
        लिखिए और $x$ तथा $y$ के लिए हल कीजिए।
  \item ग्लूकोज़ का कितना भिन्न किण्वित होता है? \omterm{def:b1:water-small-molecules:atp}{ATP} का कितना भिन्न
        \omterm{def:b1:respiration-fermentation:fermentation}{किण्वन} से आता है?
  \item वही फ़्लास्क ग्लूकोज़ के बजाय पामिटेट पर (ऑक्सी):
        RQ निकालिए।
  \item कोई विद्यार्थी विश्राम कर रहे किसी जंतु में रात भर के उपवास के
        बाद 0.7 और भोजन के बाद 1.0 का RQ मापता है। व्याख्या कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- टरबाइन।}
\begin{enumerate}[resume]
  \item \qty{0.22}{V} का कोई \omterm{thm:b1:respiration-fermentation:chemiosmosis}{प्रोटॉन-प्रेरक बल} पार करते प्रोटॉनों के एक
        मोल की \omterm{prop:b1:water-small-molecules:gibbs}{मुक्त ऊर्जा} निकालिए।
  \item \qty{4}{H^+} की ऊर्जा और प्रति \omterm{def:b1:water-small-molecules:atp}{ATP} \qty{50}{kJ} पर सिंथेज़ की
        दक्षता निकालिए।
  \item NADH से $\mathrm{O_2}$ तक (\qty{1.14}{V}) गिरते दो
        इलेक्ट्रॉनों से \omterm{prop:b1:water-small-molecules:gibbs}{मुक्त ऊर्जा}, पंप किए गए दस प्रोटॉनों में संचित
        ऊर्जा, और शृंखला की दक्षता निकालिए।
  \item दोनों दक्षताएँ मिलाइए और \qty{220}{kJ} पर 2.5 \omterm{def:b1:water-small-molecules:atp}{ATP} $\times$
        \qty{50}{kJ} से तुलना कीजिए।
  \item हवादार फ़्लास्क में कोई अयुग्मक मिलाया जाता है।
        ऑक्सीजन-खपत, \omterm{def:b1:water-small-molecules:atp}{ATP} उपज, ग्लूकोज़-खपत, ऊष्मा-निर्गम और वृद्धि में
        परिवर्तनों की भविष्यवाणी कीजिए।
  \item इसके बजाय ऑलिगोमाइसिन मिलाया जाता है, जो सिंथेज़ रोक देता है।
        परिवर्तनों की भविष्यवाणी कीजिए, और यह भी कि उसके ऊपर कोई
        अयुग्मक मिला दिया जाए तो क्या होता है।
  \item किसी उत्परिवर्ती खमीर में संकुल I नहीं है; उसका NADH किसी
        वैकल्पिक \omterm{def:b1:enzymes:enzyme}{एंजाइम} से शृंखला में घुसता है जो कोई प्रोटॉन पंप नहीं
        करता। NADH के लिए उसका \omterm{thm:b1:respiration-fermentation:chemiosmosis}{P/O अनुपात} और प्रति ग्लूकोज़ उसका \omterm{def:b1:water-small-molecules:atp}{ATP}
        निकालिए।
  \item समझाइए कि ऑक्सीजन के अभाव में \omterm{def:b1:cell-unit-of-life:cell}{कोशिका} सूत्रकणिकीय NADH भंडार को
        लगभग पूरा अपचयित क्यों रखती है, और इससे \omterm{def:b1:respiration-fermentation:krebs}{क्रेब्स चक्र} का क्या
        होता है।
  \item परिणाम बताइए: हवा के साथ और बिना हवा के प्रति ग्लूकोज़ \omterm{def:b1:water-small-molecules:atp}{ATP},
        उनका अनुपात, और NADH तथा $\mathrm{FADH_2}$ के \omterm{thm:b1:respiration-fermentation:chemiosmosis}{P/O अनुपात}।
\end{enumerate}
\end{problem}
