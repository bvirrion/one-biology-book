\chapter{البكتيريا ومقاومة المضادات الحيوية}\label{ch:g11:antibiotic-resistance}

سنة 1941 أعطي شرطي يحتضر من تسمم دم بدأ بخدش أول جرعات البنسلين التي
استعملت على مريض قط. فتعافى في أيام --- ثم انتكس ومات حين نفد المخزون،
وقد استخلص من عفن بجهد جهيد. وفي غضون عقد صار البنسلين يصنع بالأطنان
وصارت عدوى ظلت تقتل آلاف السنين تشفى في أسبوع. وفي غضون عقدين صارت
البكتيريا التي لم يعد البنسلين يمسها شائعة في كل مستشفى. وهذا الفصل عن
السبب: كيف تتنوع البكتيريا، وكيف يفرزها \omterm{def:g11:antibiotic-resistance:antibiotic}{المضاد الحيوي}، وكيف يحوّل حساب
جمهرة من المليارات طفرةً نادرة إلى مشكلة تعم جناحًا بأكمله.

\section{كيف تتنوع البكتيريا}

\begin{proposition}[تكاثر البكتيريا وتنوعها]\label{prop:g11:antibiotic-resistance:variation}
تتكاثر البكتيريا بالانقسام إلى اثنتين بعد نسخ صبغيها الوحيد: فالبنتان،
عدا أخطاء النسخ، مطابقتان للأم وراثيًا --- أي \emph{نسيلة}. وفي وسط
غني، انقسام كل 20 دقيقة يحوّل \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة إلى مليار في عشر ساعات.
\omterm{def:g10:biodiversity-scales:biodiversity}{والتنوع الوراثي} في النسيلة يأتي من مصدرين:
\begin{itemize}
  \item \emph{\omterm{def:g11:mutations:mutation}{الطفرات}} (\cref{ch:g11:mutations})، وهي نادرة في
        الانقسام الواحد لكنها متواترة في جمهرة من المليارات؛
  \item \emph{النقل الأفقي للمورّثات}\index{نقل أفقي للمورّثات}: أي
        \omterm{def:g10:universal-dna:information}{DNA} يتلقى من بكتيريا أخرى، من \omterm{def:g10:biodiversity-scales:species}{النوع} نفسه أو من \omterm{def:g10:biodiversity-scales:species}{نوع} آخر،
        وغالبًا في صورة \emph{بلازميد} --- وهو حلقة صغيرة من \omterm{def:g10:universal-dna:information}{DNA}،
        منفصلة عن \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي}، تحمل بضع \omterm{def:g10:universal-dna:gene}{مورّثات} وتستطيع الانتقال من \omterm{def:g10:cells-common-unit:cell}{خلية}
        إلى \omterm{def:g10:cells-common-unit:cell}{خلية} عبر جسر من الغشاء.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[حساب جمهرة]\label{ex:g11:antibiotic-resistance:arithmetic}
تنشأ \omterm{def:g11:mutations:mutation}{طفرة} المقاومة نحو مرة في كل $10^8$ انقسام. فدورق فيه $10^9$
بكتيريا قد شهد إذن، في الانقسامات التي أنتجته، نحو عشر \omterm{def:g11:mutations:mutation}{طفرات} من هذا
القبيل؛ وجرح ملتهب أو معي، وفيه $10^{12}$ بكتيريا، نحو عشرة آلاف. فكل
جمهرة كبيرة من البكتيريا تحوي، قبل استعمال أي \omterm{def:g11:antibiotic-resistance:antibiotic}{مضاد حيوي}، \omterm{def:g10:cells-common-unit:cell}{خلايا} مقاومة
له --- وقد أثبتت ذلك أطباق النسخ في \cref{ch:g11:mutations}.
\end{example}

\section{المضادات الحيوية}

\begin{definition}[المضاد الحيوي]\label{def:g11:antibiotic-resistance:antibiotic}
\emph{المضاد الحيوي}\index{مضاد حيوي} جزيء يقتل البكتيريا أو يوقف
نموها، بتراكيز لا تؤذي \omterm{def:g10:cells-common-unit:cell}{خلايا} المريض، بتأثيره في بنية أو عملية توجد
عند البكتيريا وتفتقر إليها \omterm{def:g10:cells-common-unit:cell}{الخلايا} البشرية: فالبنسلين وأقرباؤه يحصرون
بناء الجدار الجرثومي؛ والستربتوميسين والتتراسيكلين يعطلان \omterm{def:g10:cells-common-unit:organelle}{الريبوزوم}
الجرثومي، وهو يختلف عن ريبوزومنا؛ وغيرها يحصر \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيمات} الجرثومية
\omterm{def:g11:dna-replication:replication}{لتضاعف DNA}. ومعظمها وجد أول ما وجد في العفن وبكتيريا التربة، وهي
تصنعها ضد منافساتها. والمضادات الحيوية لا أثر لها في الفيروسات، ولا
هدف من هذه الأهداف عندها.
\end{definition}

\begin{proposition}[آليات المقاومة]\label{prop:g11:antibiotic-resistance:mechanisms}
البكتيريا \emph{مقاومة}\index{مقاومة المضادات الحيوية} \omterm{def:g11:antibiotic-resistance:antibiotic}{لمضاد حيوي} حين
تنمو عند تراكيز توقف قريباتها الحساسة. وللمقاومة أساس وراثي من أصناف
قليلة: \omterm{def:g10:universal-dna:gene}{مورّثة} \omterm{def:g11:enzymes-and-phenotype:enzyme}{لإنزيم} يهدم \omterm{def:g11:antibiotic-resistance:antibiotic}{المضاد الحيوي} (فالبنسليناز يقطع البنسلين
شطرين)؛ أو \omterm{def:g11:mutations:mutation}{طفرة} تبدّل هدف \omterm{def:g11:antibiotic-resistance:antibiotic}{المضاد الحيوي} فلا يعود الدواء يرتبط (\omterm{def:g11:gene-expression:protein}{بروتين}
ريبوزومي متغير)؛ أو مضخة تطرد الدواء؛ أو جدار لا يدخله. وكل من ذلك
\omterm{def:g10:universal-dna:gene}{أليل} أو \omterm{def:g10:universal-dna:gene}{مورّثة}، وكل منه يورَّث --- شاقوليًا في النسيلة، وأفقيًا إلى
الجارات حين يكون على بلازميد.
\end{proposition}
\admitted

\begin{method}[قراءة مخطط المضادات الحيوية]\label{met:g11:antibiotic-resistance:antibiogram}
\emph{مخطط \omterm{def:g11:antibiotic-resistance:antibiotic}{المضادات الحيوية}} يختبر بكتيريا المريض في مقابل عدة \omterm{def:g11:antibiotic-resistance:antibiotic}{مضادات
حيوية} في آن واحد.
\begin{enumerate}
  \item انثر البكتيريا نثرًا متساويًا على طبق؛ وضع أقراص ورق مشبعًا
        كل منها \omterm{def:g11:antibiotic-resistance:antibiotic}{بمضاد حيوي} واحد؛ واحضنها ليلة.
  \item ينتشر \omterm{def:g11:antibiotic-resistance:antibiotic}{المضاد الحيوي} خارج القرص، ويهبط تركيزه مع المسافة؛
        وتنمو البكتيريا مرجًا إلا حيث يوقفها التركيز: فتنشأ
        \emph{منطقة تثبيط} صافية حول كل قرص.
  \item قس قطر كل منطقة وقارنه بالعتبة المقررة لذلك \omterm{def:g11:antibiotic-resistance:antibiotic}{المضاد الحيوي}:
        فقطر فوق العتبة يعني حساسة، ودونها مقاومة.
  \item وأكبر المناطق بين النتائج الحساسة توجّه الوصفة؛ والقرص الذي لا
        منطقة له البتة \omterm{def:g11:antibiotic-resistance:antibiotic}{مضاد حيوي} تتجاهله السلالة.
\end{enumerate}
\end{method}

\section{كيف تنتشر المقاومة}

\begin{proposition}[الانتقاء بالمضاد الحيوي]\label{prop:g11:antibiotic-resistance:selection}
\omterm{def:g11:antibiotic-resistance:antibiotic}{المضاد الحيوي} لا يخلق بكتيريا مقاومة؛ بل \emph{ينتقيها}. ففي جمهرة
تحمل فيها بضع \omterm{def:g10:cells-common-unit:cell}{خلايا} \omterm{def:g10:universal-dna:gene}{أليل} مقاومة، يقتل الدواء الأغلبية الحساسة أو
يوقفها ويترك القلة المقاومة تتكاثر بلا منافسة: فتصير الجمهرة مقاومة في
غضون أيام. وكلما زاد استعمال \omterm{def:g11:antibiotic-resistance:antibiotic}{المضاد الحيوي} --- عند المرضى، وعند
الحيوانات، وفي البيئة --- تكرر هذا الفرز، وزادت مقاومة بكتيريا
المستشفى أو المزرعة أو البلد. والجرعة الناقصة، وهي تقتل أشد \omterm{def:g10:cells-common-unit:cell}{الخلايا}
حساسية وتعفي أقلها، أشدها انتقاءً على الإطلاق.
\end{proposition}
\begin{proof}[شواهد]
بينت أطباق ليدربرغ أن \omterm{def:g10:cells-common-unit:cell}{الخلايا} المقاومة موجودة قبل التعرض. والمزارع
التي تنمّى أسبوعًا في تراكيز متصاعدة من \omterm{def:g11:antibiotic-resistance:antibiotic}{مضاد حيوي} تنتهي مقاومةً لمئة
ضعف الجرعة الابتدائية، ومقاومتها تورَّث وترسم على \omterm{def:g11:mutations:mutation}{طفرات} بعينها. والبلدان
التي تستعمل أكثر \omterm{def:g11:antibiotic-resistance:antibiotic}{المضادات الحيوية} للفرد فيها أعلى نسب السلالات
المقاومة؛ وأجنحة المستشفيات التي تخفض استعمالها ترى المقاومة تهبط على
مدى أشهر. وقد ظهرت سلالة من جرثومة الجلد \emph{Staphylococcus aureus}
مقاومة للبنسلين في غضون أربع سنوات من إدخال الدواء وصارت الآن معظم
سلالات المستشفيات؛ وتلت السلالةُ المقاومة للدواء البديل ذلك البديلَ
بسنتين.
\end{proof}

\begin{example}[من خلية واحدة إلى جناح]\label{ex:g11:antibiotic-resistance:ward}
مريض يعالج بالبنسلين يحمل، بين $10^{10}$ بكتيريا، مئةً مقاومة. وفي
ثلاثة أيام تزول الحساسة وتحل النسيلة المقاومة محلها: أي $10^{10}$
\omterm{def:g10:cells-common-unit:cell}{خلية} مقاومة. وعلى يدي ممرضة، وعلى حافة سرير، وفي المريض التالي، تستمر
النسيلة --- وإن كانت \omterm{def:g10:universal-dna:gene}{مورّثة} مقاومتها على بلازميد، سلّمتها إلى \omterm{def:g10:biodiversity-scales:species}{أنواع}
أخرى في الطريق. وما من شيء في هذا التتابع اقتضى من \omterm{def:g11:antibiotic-resistance:antibiotic}{المضاد الحيوي} أن
يؤثر في \omterm{def:g10:universal-dna:information}{DNA}؛ إنما أزال المنافسة وحسب.
\end{example}

\section{إبقاء المضادات الحيوية فعالة}

\begin{proposition}[العواقب والاستجابات]\label{prop:g11:antibiotic-resistance:consequences}
العدوى المقاومة أطول وأغلى وأكثر فتكًا؛ وبعض السلالات صارت تقاوم كل
دواء متاح. ولما كانت المقاومة يقودها الاستعمال، كانت الاستجابة أن تستعمل
\omterm{def:g11:antibiotic-resistance:antibiotic}{المضادات الحيوية} أقل وأفضل: فلا تستعمل إلا للعدوى الجرثومية (ولا تستعمل
أبدًا للزكام والنزلة الوافدة الفيروسيين)، ومختارة بعد مخطط \omterm{def:g11:antibiotic-resistance:antibiotic}{مضادات
حيوية} حيث أمكن، بالجرعة الكاملة وللمدة الكاملة، بأضيق دواء يفي
بالغرض؛ وأن يحد استعمالها في الزراعة؛ وأن تمنع العدوى (بالنظافة
والتلقيح) حتى تقل الجرعات اللازمة؛ وأن يعزل حاملو السلالات المقاومة في
المستشفيات. \omterm{def:g11:antibiotic-resistance:antibiotic}{والمضادات الحيوية} الجديدة تكسب وقتًا؛ أما حساب الانتقاء
فيضمن ظهور المقاومة لكل منها.
\end{proposition}
\admitted

\begin{remark}[تطور في أسبوع]\label{rem:g11:antibiotic-resistance:evolution}
\omterm{def:g11:mutations:mutation}{طفرة} توفّر التنوع عشوائيًا، ومحيط يفرزه، وذرية الناجيات ترث الفرق:
بكتيريا هذا الفصل تجري، في أسبوع، العملية التي ستصفها السنة الأخيرة
على مدى ملايين السنين للحيتان والبلوط. \omterm{prop:g11:antibiotic-resistance:mechanisms}{فمقاومة المضادات الحيوية} تطور
بالانتقاء الطبيعي مشاهدًا في الزمن الحقيقي، والإنسان هو عامل الانتقاء
--- وهي أوضح برهان على أن العملية ليست نظرية عن الماضي بل حقيقة عن أي
جمهرة تتنوع وتتكاثر.
\end{remark}

\section{مسألة: الجناح}

\begin{problem}[{مسألة نهاية الأسبوع --- تفشٍّ في جناح مستشفى يتابع من
خلية طافرة واحدة: حساب مستعمرة، ومخطط مضادات حيوية يقرأ، وجرعة تقطع،
والسياسة التي توقف الانتشار}]
\label{pb:g11:antibiotic-resistance:1}
جرح مريض ملتهب فيه $10^{10}$ بكتيريا من \omterm{def:g10:biodiversity-scales:species}{نوع} واحد. وتنقسم البكتيريا كل 30
دقيقة؛ وتقع \omterm{def:g11:mutations:mutation}{طفرة} تعطي مقاومة \omterm{def:g11:antibiotic-resistance:antibiotic}{للمضاد الحيوي} أموكسيسيلين مرة لكل $10^8$
انقسام. \omterm{def:g11:antibiotic-resistance:antibiotic}{والمضاد الحيوي} يقتل 90\% من \omterm{def:g10:cells-common-unit:cell}{الخلايا} الحساسة كل 6 ساعات ولا
يؤثر في \omterm{def:g10:cells-common-unit:cell}{الخلايا} المقاومة، وهي تواصل الانقسام.

\medskip
\noindent\textbf{الجزء الأول --- قبل العلاج.}
\begin{enumerate}
  \item كم انقسامًا أنتج البكتيريا $10^{10}$ من \omterm{def:g10:cells-common-unit:cell}{الخلية} الأولى؟ (عُدّ
        عدد \omterm{def:g10:cells-common-unit:cell}{الخلايا}.)
  \item قدّر عدد \omterm{def:g10:cells-common-unit:cell}{الخلايا} المقاومة الموجودة قبل إعطاء أي \omterm{def:g11:antibiotic-resistance:antibiotic}{مضاد حيوي}.
  \item فسّر لماذا توصف عدوى المريض مع ذلك بأنها «حساسة» على مخطط
        \omterm{def:g11:antibiotic-resistance:antibiotic}{المضادات الحيوية}.
  \item يبين مخطط المضادات مناطق قدرها \qty{24}{mm} للأموكسيسيلين
        (العتبة 17)، و\qty{12}{mm} للإريثروميسين (العتبة 20)، ولا
        منطقة للبنسلين. فسّر النتائج الثلاث.
  \item يصف الطبيب أموكسيسيلين 7 أيام. فلماذا لا بنسلين، ولماذا لا
        الدواء ذو أكبر منطقة لو كانت إحداها أكبر؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- أثناء العلاج.}
\begin{enumerate}[resume]
  \item كم \omterm{def:g10:cells-common-unit:cell}{خلية} حساسة تبقى بعد يوم واحد؟ وبعد 3 أيام؟
  \item انطلاقًا من جوابك على السؤال 2، كم \omterm{def:g10:cells-common-unit:cell}{خلية} مقاومة توجد بعد يوم
        واحد إذا تضاعفت كل 30 دقيقة بلا حد؟ (استعمل
        $2^{48} \approx \num{2.8e14}$.) ولماذا كان هذا الرقم غير
        واقعي، وما الذي يحده؟
  \item افترض أن النسيلة المقاومة محدودة بالحيز الذي تخليه \omterm{def:g10:cells-common-unit:cell}{الخلايا}
        الحساسة، فلا تستطيع تجاوز $10^{10}$. فمتى تقريبًا تملأ ذلك
        الحيز؟
  \item ارسم، أو صف، منحنيات \omterm{def:g10:cells-common-unit:cell}{الخلايا} الحساسة والمقاومة والمجموع على
        مدى الأيام السبعة.
  \item والحق أن جهاز المريض المناعي يصفّي آخر بضعة ملايين \omterm{def:g10:cells-common-unit:cell}{خلية} من أي
        صنف في غضون أيام. فسّر لماذا تتوقف النتيجة على ما إذا كانت
        النسيلة المقاومة تنمو أسرع مما يصفّيها الجهاز المناعي.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الجرعة المقطوعة.}
\begin{enumerate}[resume]
  \item يتوقف المريض بعد يومين وهو يشعر بأنه بخير. فكم \omterm{def:g10:cells-common-unit:cell}{خلية} حساسة
        تبقى، وأي نسبة من الجمهرة الناجية مقاومة؟
  \item تعود البكتيريا إلى النمو حتى $10^{10}$ في الأيام التالية. فأي
        نسبة منها مقاومة الآن، إذا نمت \omterm{def:g10:cells-common-unit:cell}{الخلايا} المقاومة والحساسة
        بالنسبة نفسها؟
  \item ينتكس المريض فيعطى أموكسيسيلين من جديد. تنبأ بالنتيجة.
  \item يبين مخطط \omterm{def:g11:antibiotic-resistance:antibiotic}{مضادات حيوية} للانتكاسة أنه لا منطقة للأموكسيسيلين.
        فسّر التغير عن المخطط الأول.
  \item \omterm{def:g10:universal-dna:gene}{مورّثة} المقاومة على بلازميد. فأي خطر إضافي يضيف ذلك على
        الجناح؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- سياسة الجناح.}
يعالج الجناح 400 مريض في السنة بالأموكسيسيلين؛ و30\% من حالات العدوى
المعزولة في الجناح مقاومة الآن، مقابل 5\% في المدينة.
\begin{enumerate}[resume]
  \item فسّر الفرق بين الجناح والمدينة بدلالة ضغط الانتقاء.
  \item يقرر الجناح: مخططات \omterm{def:g11:antibiotic-resistance:antibiotic}{مضادات حيوية} قبل كل وصفة، وجرعات كاملة،
        ونظافة أيدٍ، وعزل حاملي السلالات المقاومة. قل على أي آلية من
        هذا الفصل يعمل كل تدبير.
  \item بعد سنتين تصير المقاومة في الجناح 12\%. فسّر لماذا هبطت،
        ولماذا لم تعد إلى 5\%.
  \item لماذا لا يحل \omterm{def:g11:antibiotic-resistance:antibiotic}{مضاد حيوي} جديد، يدخل إلى الجناح، المشكلةَ أكثر
        من بضع سنوات؟
  \item اذكر النتيجة: عدد \omterm{def:g10:cells-common-unit:cell}{الخلايا} المقاومة التي حملها الجرح قبل أي
        دواء، واليوم الذي ورثت فيه ذلك، والعادة الواحدة عند المريض
        التي حوّلت عدوى قابلة للشفاء إلى تفشٍّ.
\end{enumerate}
\end{problem}
