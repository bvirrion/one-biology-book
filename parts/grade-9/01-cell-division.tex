\chapter{Cell Division}\label{ch:g9:cell-division}

\omterm{def:g4:ecosystems:levels}{Organisms} \omterm{def:g2:what-alive:signs}{grow}, repair \omterm{def:g7:tissues:tissue}{tissues} and \omterm{def:g2:what-alive:signs}{reproduce} using \omterm{def:g7:microscope:cell}{cell} division. \omterm{def:g9:cell-division:mitosis}{Mitosis}
makes \omterm{def:g1:my-body:body}{body} \omterm{def:g7:microscope:cell}{cells}; \omterm{def:g9:cell-division:meiosis}{meiosis} makes \omterm{def:g6:reproduction:sexual}{gametes}. This chapter separates the two
stories and links them to \omterm{def:g8:genetics:chromosome}{chromosome} numbers.

\section{Why cells divide}

\begin{definition}[Cell cycle]\label{def:g9:cell-division:cycle}
The \emph{cell cycle}\index{cell cycle} is the life of a \omterm{def:g7:microscope:cell}{cell} from
formation to division: \omterm{def:g2:growing:growth}{growth} and DNA copying, then \omterm{def:g9:cell-division:mitosis}{mitosis} and
\omterm{def:g9:cell-division:mitosis}{cytokinesis} (for \omterm{def:g1:my-body:body}{body} \omterm{def:g7:microscope:cell}{cells}).
\end{definition}

\begin{proposition}[Roles of division]\label{prop:g9:cell-division:roles}
Mitotic division enables \omterm{def:g2:growing:growth}{growth}, repair and \omterm{def:g6:reproduction:asexual}{asexual reproduction}. \omterm{def:g9:cell-division:meiosis}{Meiosis}
halves \omterm{def:g8:genetics:chromosome}{chromosome} number for \omterm{prop:g5:reproduction:two}{sexual reproduction} so \omterm{def:g6:reproduction:sexual}{fertilisation} can
restore the \omterm{def:g8:genetics:ploid}{diploid} number.
\end{proposition}
\admitted

\section{Mitosis}

\begin{definition}[Mitosis]\label{def:g9:cell-division:mitosis}
\emph{Mitosis}\index{mitosis} is nuclear division that produces two nuclei
with the same \omterm{def:g8:genetics:chromosome}{chromosome} number and (ideally) the same genetic information
as the parent \omterm{def:g8:organelles:nucleus}{nucleus}. \emph{Cytokinesis}\index{cytokinesis} splits the
\omterm{def:g7:microscope:parts}{cytoplasm} into two \omterm{def:g7:microscope:cell}{cells}.
\end{definition}

\begin{definition}[Chromosome copying]\label{def:g9:cell-division:copy}
Before \omterm{def:g9:cell-division:mitosis}{mitosis}, DNA is replicated. Each \omterm{def:g8:genetics:chromosome}{chromosome} then consists of two
sister \emph{chromatids}\index{chromatid} joined until they separate.
\end{definition}

\begin{proposition}[Mitosis outcome]\label{prop:g9:cell-division:mit-out}
One \omterm{def:g8:genetics:ploid}{diploid} parent \omterm{def:g7:microscope:cell}{cell} $\to$ two \omterm{def:g8:genetics:ploid}{diploid} daughter \omterm{def:g7:microscope:cell}{cells} with matching
\omterm{def:g8:genetics:chromosome}{chromosome} sets (barring mutation/error).
\end{proposition}
\admitted

\section{Meiosis}

\begin{definition}[Meiosis]\label{def:g9:cell-division:meiosis}
\emph{Meiosis}\index{meiosis} is a two-division process that produces four
\omterm{def:g8:genetics:ploid}{haploid} \omterm{def:g7:microscope:cell}{cells} (\omterm{def:g6:reproduction:sexual}{gametes} in \omterm{def:g1:plants-animals:animal}{animals}) from one \omterm{def:g8:genetics:ploid}{diploid} \omterm{def:g7:microscope:cell}{cell}. \omterm{def:g8:genetics:homolog}{Homologous
chromosomes} separate in meiosis I; sister \omterm{def:g9:cell-division:copy}{chromatids} separate in meiosis II.
\end{definition}

\begin{proposition}[Variation from meiosis]\label{prop:g9:cell-division:var}
Independent assortment of homologues and crossing over (exchange between
homologues) shuffle \omterm{def:g7:heredity:allele}{alleles}, so \omterm{def:g6:reproduction:sexual}{gametes} differ --- fuel for genetic
\omterm{def:g5:evolution:variation}{variation} after \omterm{def:g6:reproduction:sexual}{fertilisation}.
\end{proposition}
\admitted

\begin{method}[Mitosis or meiosis?]\label{met:g9:cell-division:which}
\begin{enumerate}
  \item Purpose \omterm{def:g2:growing:growth}{growth}/repair/asexual clone? $\to$ \omterm{def:g9:cell-division:mitosis}{mitosis}.
  \item Purpose make \omterm{def:g6:reproduction:sexual}{gametes} / halve \omterm{def:g7:heredity:gene}{chromosomes}? $\to$ \omterm{def:g9:cell-division:meiosis}{meiosis}.
  \item Check products: 2 identical \omterm{def:g8:genetics:ploid}{diploid} vs 4 diverse \omterm{def:g8:genetics:ploid}{haploid}
        (typical textbook \omterm{def:g1:plants-animals:animal}{animal} case).
\end{enumerate}
\end{method}

\begin{example}[Human numbers]\label{ex:g9:cell-division:human}
Human \omterm{def:g1:my-body:body}{body} \omterm{def:g7:microscope:cell}{cells}: 46 \omterm{def:g7:heredity:gene}{chromosomes}. Sperm/\omterm{def:g3:life-cycles:butterfly}{egg}: 23. Zygote: 46.
\end{example}
