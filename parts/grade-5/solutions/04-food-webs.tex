\section*{Chapter \ref{ch:g5:food-webs} --- Food Webs}

\begin{solution}{exo:g5:food-webs:1}
A \omterm{def:g5:food-webs:chain-web}{food chain} is one who-eats-whom path; a \omterm{def:g5:food-webs:chain-web}{food web} is many linked chains
in a \omterm{def:g4:ecosystems:levels}{community}.
\end{solution}

\begin{solution}{exo:g5:food-webs:2}
Toward the eater, showing the direction of \omterm{def:g2:food-energy:energy}{energy} flow.
\end{solution}

\begin{solution}{exo:g5:food-webs:3}
\omterm{def:g5:food-webs:roles}{Producers} make \omterm{def:g2:food-energy:food}{food} (usually via \omterm{def:g4:plants-need:needs}{light}). \omterm{def:g4:ecosystems:roles}{Consumers} eat other \omterm{def:g4:ecosystems:levels}{organisms}
or their remains.
\end{solution}

\begin{solution}{exo:g5:food-webs:4}
They break down dead material and recycle \omterm{def:g4:plants-need:needs}{nutrients}.
\end{solution}

\begin{solution}{exo:g5:food-webs:5}
Example: grass $\to$ \omterm{def:g3:insects:insect}{insect} $\to$ mouse $\to$ hawk. Grass (or other
\omterm{def:g1:plants-animals:plant}{plant}) is the \omterm{def:g5:food-webs:roles}{producer}.
\end{solution}
