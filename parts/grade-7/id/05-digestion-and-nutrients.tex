\chapter{Pencernaan dan Zat Gizi}\label{ch:g7:digestion-and-nutrients}

\Cref{ch:g4:journey-of-food} mengikuti sepotong gigitan roti menuruni
pipanya lalu berjanji bahwa ``\omterm{def:g4:journey-of-food:digestion}{getah pencernaan} membongkarnya''. Tahun
ini kita membuka pintu bengkelnya: apa tepatnya yang dikerjakan
getahnya, apa \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} yang dihasilkannya, dan bagaimana dinding usus
halusnya mengelola penyeberangan perbatasan terbesar di dalam tubuh?
Gigitan rotinya kembali --- di bawah alat yang lebih baik.

\section{Getahnya yang bekerja}

\begin{proposition}[Pencernaan adalah pembongkaran kimia]\label{prop:g7:digestion-and-nutrients:chemical}
Mengunyah dan mengaduk hanya memecah makanan menjadi lebih
\emph{kecil}; pembongkaran yang sesungguhnya bersifat kimia.
\emph{Getah pencernaan}\index{getah pencernaan} --- \omterm{def:g4:journey-of-food:tract}{air liur}, getah
\omterm{def:g4:journey-of-food:tract}{lambungnya}, getah yang dicurahkan ke dalam \omterm{def:g4:journey-of-food:tract}{usus halus} --- memuat zat
kerja yang memotong komponen besar makanan menjadi komponen kecil:
pati menjadi gula, materi pembangun pada daging dan \omterm{def:g2:animals-grow-up:born}{telur} menjadi
satuan kecilnya, lemak menjadi satuannya. Tiap getah punya
keahliannya, dan masing-masing bekerja paling baik pada \omterm{def:g6:exploring-our-environment:conditions}{kondisi}
stasiunnya sendiri.
\end{proposition}
\admitted

\begin{example}[Versi mejanya]\label{ex:g7:digestion-and-nutrients:bench}
Percobaan yang klasik menjalankan \omterm{def:g4:journey-of-food:digestion}{pencernaan} di dalam kaca. Dua tabung
berisi pasta pati yang sudah dimasak pada kehangatan tubuh: satu
mendapat \omterm{def:g4:journey-of-food:tract}{air liur} segar, satu mendapat \omterm{def:g4:journey-of-food:tract}{air liur} yang direbus (dirusak).
Sejam kemudian, uji patinya mendapati tabung 2 tidak berubah --- masih
pati --- sementara pasta tabung 1 sudah berubah menjadi gula, persis
rasa manis pada \cref{ex:g4:journey-of-food:sweet}, yang dihasilkan
tanpa mulut mana pun. Getahnya sendirilah yang mengerjakan
pembongkarannya --- dan kontrol yang direbus itu memperlihatkan bahwa
zat kerjanya dihancurkan panas, sebagaimana permesinan yang halus.
\end{example}

\begin{remark}[Mengapa harus dibongkar sama sekali?]\label{rem:g7:digestion-and-nutrients:why}
Komponen besar makanan tidak lebih mungkin menyeberangi dinding
ususnya daripada sebuah lemari melewati lubang surat. Dan tubuhnya
memang tidak menginginkannya secara utuh: ia menginginkan
\emph{bagian} --- satuan kecil yang dapat dibangunnya ulang menjadi
zatnya sendiri
(\cref{prop:g6:matter-from-food:transform}). \omterm{def:g4:journey-of-food:digestion}{Pencernaan} adalah
pembongkaran demi pengangkutan \emph{dan} demi perakitan ulang: perabot
menjadi rakit-sendiri, pada setiap santapan.
\end{remark}

\section{Zat gizinya}

\begin{definition}[Keluarga zat gizi]\label{def:g7:digestion-and-nutrients:families}
Pembongkarannya menghasilkan \emph{zat gizi}\index{zat gizi} ---
komponen kecil yang siap masuk \omterm{prop:g4:journey-of-food:absorb}{darah}:
\begin{itemize}
  \item \emph{gula} (terutama \omterm{prop:g7:organs-at-work:bill}{glukosa}, bahan bakar pada
        \cref{prop:g7:organs-at-work:bill}), dari makanan bertepung
        dan manis;
  \item satuan kecil \emph{materi pembangun}, dari daging, \omterm{prop:g3:sorting-living-things:groups}{ikan},
        \omterm{def:g2:animals-grow-up:born}{telur}, produk \omterm{def:g2:animals-grow-up:born}{susu} dan kacang-kacangan --- bata pembangunan
        pada \cref{prop:g3:balanced-diet:jobs};
  \item satuan \emph{lemak} --- bahan bakar yang pekat sekaligus bahan
        bangunan;
  \item ditambah para penumpang yang tidak membutuhkan pembongkaran:
        \emph{vitamin}, \emph{\omterm{prop:g4:what-plants-need:needs}{garam mineral}}, dan air itu sendiri.
\end{itemize}
\omterm{def:g3:balanced-diet:families}{Keluarga makanan} pada \cref{def:g3:balanced-diet:families} pada
dasarnya adalah pemasok keluarga zat gizi ini --- daftar belanja di
balik daftar belanja.
\end{definition}

\begin{example}[Satu makan siang, yang dirinci]\label{ex:g7:digestion-and-nutrients:lunch}
Makan siang seimbang pada \cref{ex:g3:balanced-diet:lunch}, yang
dibaca ulang sebagai \omterm{def:g7:digestion-and-nutrients:families}{zat gizi}: pastanya --- pati, yang dibongkar
menjadi \omterm{prop:g7:organs-at-work:bill}{glukosa} berliter-liter; \omterm{prop:g3:sorting-living-things:groups}{ikannya} --- materi pembangun, yang
dipotong menjadi satuannya; minyak pada buncisnya --- satuan lemak;
buncis dan apelnya --- vitamin, mineral, air, ditambah serat. Seratnya,
perhatikan, adalah perkecualian yang menegaskan sistemnya: tidak ada
getah kita yang dapat memotongnya, sehingga ia tidak menyeberangi apa
pun lalu meneruskan perjalanan --- bagian besar sisanya
(\cref{prop:g4:journey-of-food:absorb}), yang mengerjakan penyapuannya
sendiri yang berguna.
\end{example}

\section{Penyeberangan yang besar}

\begin{proposition}[Penyerapan pada jonjot usus]\label{prop:g7:digestion-and-nutrients:absorption}
Penyeberangan yang dijanjikan pada
\cref{prop:g4:journey-of-food:absorb} terjadi menyeberangi dinding
dalam usus halusnya, dan dindingnya dibangun untuk itu: terlipat, dan
terhampari jutaan \emph{jonjot usus}\index{jonjot usus} --- benjolan
berbentuk jari setinggi satu milimeter, masing-masing membungkus
\omterm{def:g4:heart-and-blood:vessels}{pembuluh darah} halusnya sendiri. Jonjot ususnyalah sebabnya permukaan
penyeberangannya mencapai ukuran sehampar karpet kamar
(\cref{rem:g4:journey-of-food:surface}): lipatan di atas lipatan di
atas jari. \omterm{def:g7:digestion-and-nutrients:families}{Zat gizi} menyeberangi dindingnya yang tipis ke dalam
\omterm{prop:g4:journey-of-food:absorb}{darahnya}, dan \omterm{prop:g4:journey-of-food:absorb}{darahnya} membawanya lebih dahulu ke sebuah organ besar
penyortir-dan-penyimpan --- yaitu \emph{hati}\index{hati} --- sebelum
pengantaran umumnya.
\end{proposition}
\admitted

\begin{example}[Spesifikasinya lagi]\label{ex:g7:digestion-and-nutrients:spec}
Luas, tipis, lembap, berlapis \omterm{prop:g4:journey-of-food:absorb}{darah}: jonjot ususnya persis memenuhi
spesifikasi pada \cref{prop:g7:how-animals-breathe:spec} --- untuk \omterm{def:g7:digestion-and-nutrients:families}{zat
gizi} alih-alih \omterm{def:g7:what-is-respiration:respiration}{oksigen}. Kantong \omterm{def:g7:how-animals-breathe:four}{paru-paru}, umbai \omterm{def:g7:how-animals-breathe:four}{insang}, \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot usus}:
di mana pun tubuh harus memindahkan banyak hal menyeberangi sebuah
perbatasan, ia membangun mesin yang sama. Kenali spesifikasinya sekali
saja, dan kamu mengenalinya di mana-mana.
\end{example}

\begin{method}[Melacak sebuah zat gizi dari ujung ke ujung]\label{met:g7:digestion-and-nutrients:trace}
Untuk suapan apa pun, lacaklah satu \omterm{def:g7:digestion-and-nutrients:families}{zat gizi}:
\begin{enumerate}
  \item sebutkan \omterm{def:g3:balanced-diet:families}{keluarga makanannya} dan komponen utamanya;
  \item stasiun demi stasiun, catatlah di mana pembongkarannya bermula
        dan berakhir, dan getah mana yang bekerja padanya;
  \item seberangilah jonjot ususnya bersamanya, ke dalam \omterm{prop:g4:journey-of-food:absorb}{darahnya},
        lewat \omterm{prop:g7:digestion-and-nutrients:absorption}{hatinya};
  \item antarkanlah ia: ke sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} yang bekerja sebagai bahan bakar
        (\cref{prop:g7:organs-at-work:bill}), atau ke sebuah lokasi
        pembangunan sebagai bahan
        (\cref{def:g6:matter-from-food:production}).
\end{enumerate}
\end{method}

\begin{example}[Pelacakannya, yang dijalankan atas roti]\label{ex:g7:digestion-and-nutrients:breadtrace}
Roti: keluarga bertepung, terutama pati. Pembongkarannya bermula di
mulut (\omterm{def:g4:journey-of-food:tract}{air liur} --- rasa manis kunyahan yang panjang), tertahan di
dalam asam \omterm{def:g4:journey-of-food:tract}{lambungnya}, lalu selesai di dalam getah usus halusnya:
\omterm{prop:g7:organs-at-work:bill}{glukosa}. Penyeberangannya: \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot usus}, \omterm{prop:g4:journey-of-food:absorb}{darah}, \omterm{prop:g7:digestion-and-nutrients:absorption}{hati} --- yang menabung
sebagian \omterm{prop:g7:organs-at-work:bill}{glukosanya} untuk nanti, salah satu di antara banyak
layanannya. Pengantarannya: sebuah \omterm{def:g3:bones-and-muscles:muscle}{otot} paha di tengah lari cepat
membakarnya dalam waktu sejam. Gigitan pada kelas empat, yang kini
terhitung sepenuhnya.
\end{example}

\section{Soal: Usus Kaca}

\begin{problem}[{Soal akhir pekan --- membangun pencernaan di atas
meja, tabung demi tabung}]\label{pb:g7:digestion-and-nutrients:1}
Kelasnya membangun sebuah ``usus kaca'': sebuah rak berisi tabung pada
kehangatan tubuh, masing-masing menirukan satu stasiun salurannya.
Tabung M (mulut): pasta pati ditambah \omterm{def:g4:journey-of-food:tract}{air liur} segar. Tabung M$'$:
pasta pati ditambah \omterm{def:g4:journey-of-food:tract}{air liur} yang direbus. Tabung S (\omterm{def:g4:journey-of-food:tract}{lambung}): kubus
putih \omterm{def:g2:animals-grow-up:born}{telur} ditambah getah \omterm{def:g4:journey-of-food:tract}{lambung} dari apotekernya. Tabung S$'$:
kubus yang sama di dalam air biasa. Semuanya dijalankan selama dua
jam; lalu ujinya --- untuk pati, untuk gula, dan sebuah pengamatan
atas kubusnya.

\medskip\noindent\textbf{Bagian I --- Ramalan dan
hasilnya.}
\begin{enumerate}
  \item Ramalkan hasil dua jam keempat tabungnya.
  \item M berubah bergula, M$'$ tidak. Nyatakan kesimpulannya, dan
        peran M$'$ yang tepat.
  \item Kubus S menyusut, melunak dan menjernih; kubus S$'$ tetap
        tidak berubah. Apa yang sudah dikerjakan getah \omterm{def:g4:journey-of-food:tract}{lambungnya} ---
        dan keluarga \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} jenis apa yang sedang dihasilkan?
  \item Tidak ada tabung yang dikunyah atau diaduk. Apa yang
        dibuktikan raknya tentang di mana letak pekerjaan sesungguhnya
        \omterm{def:g4:journey-of-food:digestion}{pencernaan}?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Memperbaiki modelnya.}
\begin{enumerate}[resume]
  \item Seorang murid mengusulkan menjalankan tabung M pada \omterm{def:g6:exploring-our-environment:conditions}{suhu}
        lemari es ``demi kerapian''. Bantahlah, dengan pola pada
        \cref{prop:g6:decomposers-and-soil:speed} yang diterapkan pada
        getahnya.
  \item Seorang lagi mengusulkan satu tabung besar dengan kedua
        getahnya dicampur. Fakta stasiun-demi-stasiun apa pada
        \cref{prop:g7:digestion-and-nutrients:chemical} yang akan
        dikaburkan campurannya?
  \item Usus kacanya tidak punya dinding: tidak ada yang diserap.
        Struktur apa yang berikutnya dibutuhkan sebuah model yang
        lengkap, dan dua sifat apa yang harus dimiliki bahannya?
  \item Tambahkan serat --- serpih dedak --- ke sebuah tabung kelima
        dengan setiap getah yang tersedia. Ramalkan keadaan dua
        jamnya, dan maknanya bagi modelnya.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Dari kaca ke usus.}
\begin{enumerate}[resume]
  \item Di dalam tubuh, gula M berikutnya akan berjumpa dengan jonjot
        ususnya. Lanjutkan perjalanannya dalam empat perhentian.
  \item Getah \omterm{def:g4:journey-of-food:tract}{lambungnya} bekerja di dalam asam yang kuat; getah
        ususnya membutuhkan asam itu dinetralkan. Apa yang dikatakannya
        tentang ``tiap getah pada \omterm{def:g6:exploring-our-environment:conditions}{kondisi} stasiunnya sendiri'' --- dan
        tentang menelan stasiunnya secara berurutan?
  \item Sebuah perusahaan farmasi menjual kapsul getah bagi orang yang
        stasiunnya sendiri kurang bekerja. Jelaskan, dengan raknya,
        mengapa terapi semacam itu bahkan mungkin ada.
  \item Rangkumlah praktikumnya dalam satu kalimat yang memuat
        \emph{getah}, \emph{\omterm{def:g7:digestion-and-nutrients:families}{zat gizi}} dan \emph{tabung kontrol}.
\end{enumerate}
\end{problem}
