\section*{Bab \ref{ch:g3:flowers-fruits-seeds} --- Bunga, Buah dan Biji}

\begin{solution}{exo:g3:flowers-fruits-seeds:1}
\omterm{def:g3:flowers-fruits-seeds:parts}{Mahkota bunga} (berwarna dan berbau harum, memikat \omterm{prop:g3:sorting-living-things:groups}{serangga}); \omterm{def:g3:flowers-fruits-seeds:parts}{benang
sari} (tangkai tipis yang membawa \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari}); \omterm{def:g3:flowers-fruits-seeds:parts}{putik} di tengahnya
(labu kecil yang memuat calon \omterm{def:g2:from-seed-to-plant:seed}{biji}).
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:2}
\omterm{def:g3:flowers-fruits-seeds:parts}{Serbuk sari} adalah bubuk kuning yang halus yang dibawa \omterm{def:g3:flowers-fruits-seeds:parts}{benang sari}.
Penyerbukan adalah sampainya \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} ke \omterm{def:g3:flowers-fruits-seeds:parts}{putik} --- kebanyakan
diantar \omterm{prop:g3:sorting-living-things:groups}{serangga} yang berkunjung, kadang oleh angin.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:3}
(a) Keduanya layu lalu gugur. (b) Ia membengkak lalu menjadi \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya}.
(c) Semuanya menjadi \omterm{def:g2:from-seed-to-plant:seed}{biji} di dalam \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya}.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:4}
April: \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} putih, lebah berkunjung --- penyerbukan. Mei: \omterm{def:g3:flowers-fruits-seeds:parts}{mahkota
bunganya} gugur; \omterm{def:g3:flowers-fruits-seeds:parts}{putik} tiap \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} yang terserbuki membengkak menjadi
bola hijau yang mungil. Juni: bolanya sudah tumbuh menjadi \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} ceri
yang merah dan manis dengan \omterm{def:g2:from-seed-to-plant:seed}{biji} besarnya --- \omterm{def:g2:from-seed-to-plant:seed}{bijinya} --- di dalam.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:5}
Karena tanpa kunjungan \omterm{prop:g3:sorting-living-things:groups}{serangga} kebanyakan \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} tidak pernah
terserbuki, dan \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} yang tidak terserbuki gugur tanpa menjadi \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}.
Pengantaran \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} oleh lebahlah yang mengubah \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} menjadi \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}
ceri.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:6}
Belah ia lalu carilah \omterm{def:g2:from-seed-to-plant:seed}{biji}: ada \omterm{def:g2:from-seed-to-plant:seed}{biji} di dalamnya berarti ia tumbuh
dari sebuah \omterm{def:g3:flowers-fruits-seeds:parts}{putik} --- sebuah \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}; tidak ada berarti bagian \omterm{def:g1:plants-around-us:plant}{tumbuhan}
yang lain. Tomatnya penuh \omterm{def:g2:from-seed-to-plant:seed}{biji}: sebuah \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}. Wortelnya tidak punya
\omterm{def:g2:from-seed-to-plant:seed}{biji}: bukan \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} melainkan \omterm{def:g1:plants-around-us:parts}{akar}.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:7}
Mentimun: berbiji --- \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}. Selada: tanpa \omterm{def:g2:from-seed-to-plant:seed}{biji} --- \omterm{def:g1:plants-around-us:parts}{daun}, bukan \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}.
Polong buncis hijau: sederet \omterm{def:g2:from-seed-to-plant:seed}{biji} --- \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}. Kentang: tanpa \omterm{def:g2:from-seed-to-plant:seed}{biji} ---
\omterm{def:g1:plants-around-us:parts}{batang} bawah tanah yang membengkak, bukan \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:8}
Jawaban terbuka. Kejutan yang biasa: tomat, mentimun dan polong buncis
yang ternyata \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}; stroberi yang membawa \omterm{def:g2:from-seed-to-plant:seed}{bijinya} yang mungil di bagian
luarnya.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:9}
Hampir tidak ada \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} prem. \omterm{prop:g3:flowers-fruits-seeds:transform}{Buah} hanya dapat datang dari \omterm{def:g3:flowers-fruits-seeds:parts}{putik} \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}
yang terserbuki; dengan \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya} mati pada bulan April, tidak ada lagi
yang tersisa untuk berubah, seindah apa pun musim panasnya.
\end{solution}

\begin{solution}{exo:g3:flowers-fruits-seeds:10}
Bagi pohonnya, apel adalah cara membuat dan melindungi \omterm{def:g2:from-seed-to-plant:seed}{biji}. Di intinya
terbaring \omterm{def:g2:from-seed-to-plant:seed}{biji-bijinya} --- dan tiap \omterm{def:g2:from-seed-to-plant:seed}{biji}, kalau ditanam, dapat
\omterm{def:g2:from-seed-to-plant:germination}{berkecambah} menjadi pohon apel baru. Bahwa kita menemukan
pembungkusnya lezat adalah keberuntungan kita (dan, bagi pohonnya,
sebuah cara agar \omterm{def:g2:from-seed-to-plant:seed}{bijinya} terbawa jauh dari rumah).
\end{solution}
