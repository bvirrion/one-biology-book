\chapter{L'hérédité et les caractères}\label{ch:g9:heredity-and-traits}

Les photographies de famille sont des données de \omterm{def:g1:living-or-not:living}{biologie}.
Le menton de la grand-mère sur le petit dernier, les
boucles de la mère sur aucun des deux enfants, un frère
grand et l'autre non --- la ressemblance court dans les
familles, et la différence aussi. Ce livre s'appuie sur ce
fait depuis des années (le poulain de
\cref{rem:g4:animal-reproduction:mix}, les frères et sœurs
de \cref{prop:g8:sexual-reproduction:shuffle})~; le
chapitre qui ouvre cette année s'y attaque enfin~: qu'est-ce
qui est hérité au juste, qu'est-ce qui ne l'est pas, et que
nous laissent conclure les motifs de la ressemblance
familiale.

\section{Les caractères, et d'où ils viennent}

\begin{definition}[Les caractères]\label{def:g9:heredity-and-traits:traits}
Un \emph{caractère}\index{caractère} est n'importe quelle
particularité descriptible d'un individu~: la couleur des
\omterm{def:g1:the-five-senses:sense}{yeux}, le groupe sanguin, la forme du lobe de l'\omterm{def:g1:the-five-senses:sense}{oreille}, la
taille, une voix de chanteur, une cicatrice. La collection
entière des caractères d'un individu est bâtie par deux
ouvrages~: ce qui a été \emph{hérité}, et ce que
\emph{le milieu et la vie} ont ajouté (le monde de
\cref{def:g4:adapted-to-their-world:adaptation}, agissant
sur un seul corps).
\end{definition}

\begin{definition}[L'hérédité]\label{def:g9:heredity-and-traits:heredity}
L'\emph{hérédité}\index{hérédité} est la transmission des
\omterm{def:g9:heredity-and-traits:traits}{caractères} des parents aux descendants par la reproduction
--- par deux \omterm{def:g5:first-look-at-cells:cell}{cellules} uniques, comme l'a montré
\cref{ch:g8:sexual-reproduction}. Les \omterm{def:g9:heredity-and-traits:traits}{caractères} qui
passent ainsi sont \emph{héréditaires}\index{caractère
héréditaire}~: la couleur des \omterm{def:g1:the-five-senses:sense}{yeux} et le groupe sanguin le
sont~; les cicatrices et les \omterm{def:g1:the-five-senses:sense}{oreilles} percées, non, si
longtemps portées soient-elles. Le test de tri, ce sont les
\omterm{def:g8:sexual-reproduction:gametes}{gamètes}~: ce que deux \omterm{def:g5:first-look-at-cells:cell}{cellules} ne peuvent pas porter, les
familles ne peuvent pas le transmettre.
\end{definition}

\begin{example}[Trier un album de famille]\label{ex:g9:heredity-and-traits:album}
Les photographies du grand-père~: son nez si reconnaissable
--- héréditaire, le revoici sur la tante et sur le cousin~;
son teint hâlé de paysan --- acquis, son petit-fils
employé de bureau ne l'a pas~; sa taille --- un peu des
deux, car le petit-fils le dépasse sur la même charpente,
nourri par des années plus riches
(\cref{prop:g3:balanced-diet:jobs})~; son talent de
violoniste --- non héréditaire en tant que talent
(\cref{ex:g8:nervous-communication:juggler}~: les \omterm{def:g8:nervous-communication:synapse}{synapses}
s'entraînent, elles ne se transmettent pas), même si cette
famille de musiciens nous dit que quelque chose de plus
subtil travaille dans ce qui se transmet~: une aptitude
peut-être, jamais le câblage entraîné lui-même.
\end{example}

\begin{proposition}[Hérité et acquis]\label{prop:g9:heredity-and-traits:sorting}
Les deux ouvrages se séparent sur trois tests~:
\begin{enumerate}
  \item le \emph{motif familial}~: les \omterm{def:g9:heredity-and-traits:heredity}{caractères
        héréditaires} reviennent dans l'arbre le long des
        lignes de la reproduction~; les \omterm{def:g9:heredity-and-traits:traits}{caractères} acquis
        suivent les vies (tous les marins sont hâlés, quels
        que soient leurs parents)~;
  \item la \emph{présence dès le départ}~: le groupe
        sanguin est fixé à la \omterm{def:g5:flower-to-fruit:steps}{fécondation}~; le hâle et le
        talent se construisent après~;
  \item la \emph{transmission}~: les enfants des amputés
        ont leurs \omterm{def:g1:my-body:parts}{membres}~; les enfants des culturistes
        partent sans entraînement. Les \omterm{def:g9:heredity-and-traits:traits}{caractères} acquis
        meurent avec celui qui les a acquis --- les \omterm{def:g8:sexual-reproduction:gametes}{gamètes}
        ne les portent pas.
\end{enumerate}
La plupart des \omterm{def:g9:heredity-and-traits:traits}{caractères} visibles --- la taille, la
carrure, et même le tempérament --- sont tissés des deux
ouvrages~: une marge héréditaire, et un remplissage vécu.
\end{proposition}
\admitted

\section{Lire les motifs familiaux}

\begin{example}[Les caractères qui sautent une génération]\label{ex:g9:heredity-and-traits:skip}
Certains motifs déroutent au premier regard~: deux parents
aux \omterm{def:g1:the-five-senses:sense}{yeux} bruns avec un enfant aux \omterm{def:g1:the-five-senses:sense}{yeux} bleus~; le \omterm{def:g9:heredity-and-traits:traits}{caractère}
de la grand-mère absent chez tous ses enfants et pourtant
de retour chez un petit-enfant. Le saut est fréquent, et il
porte une lourde conclusion~: un parent peut
\emph{transmettre ce qu'il ne montre pas}. Les consignes du
\omterm{def:g9:heredity-and-traits:traits}{caractère} ont voyagé cachées à travers une génération ---
ce qui est hérité n'est donc pas le \omterm{def:g9:heredity-and-traits:traits}{caractère} lui-même,
mais quelque chose qui le \emph{détermine}, transportable
sans être vu.
\end{example}

\begin{proposition}[Ce que les motifs nous forcent à conclure]\label{prop:g9:heredity-and-traits:conclusions}
Trois conclusions, chacune forcée par une observation~:
\begin{enumerate}
  \item puisque les \omterm{def:g9:heredity-and-traits:traits}{caractères} acquis ne passent pas, ce
        qui passe doit être des \emph{déterminants} --- des
        consignes pour construire les \omterm{def:g9:heredity-and-traits:traits}{caractères}, et non
        les \omterm{def:g9:heredity-and-traits:traits}{caractères} eux-mêmes~;
  \item puisque les \omterm{def:g9:heredity-and-traits:traits}{caractères} sautent des générations, les
        déterminants peuvent être \emph{portés sans être
        montrés}~: chaque individu en porte plus qu'il
        n'en affiche~;
  \item puisque chaque enfant reçoit de ses deux parents
        par deux \omterm{def:g8:sexual-reproduction:gametes}{gamètes}
        (\cref{prop:g8:sexual-reproduction:core}), chaque
        parent apporte une \emph{sélection} de ses
        déterminants --- et les sélections diffèrent d'un
        enfant à l'autre, sans quoi les frères et sœurs
        seraient identiques (le brassage de
        \cref{prop:g8:sexual-reproduction:shuffle}, enfin
        situé~: c'est un brassage \emph{de déterminants}).
\end{enumerate}
\end{proposition}
\admitted

\begin{remark}[La question que cela pose]\label{rem:g9:heredity-and-traits:question}
L'album de famille nous a acculés à une question
merveilleuse. Quelque part dans le presque-rien d'une
\omterm{def:g8:reproductive-systems:male}{cellule mâle} (\cref{ex:g8:reproductive-systems:sperm} --- un
\omterm{def:g6:cell-unit-of-life:parts}{noyau} tassé et une queue) voyagent des consignes pour un
nez, un groupe sanguin, une couleur des \omterm{def:g1:the-five-senses:sense}{yeux} --- portables
sans être montrées, sélectionnables par moitiés, lisibles
par le corps qui grandit. \emph{Où sont-elles, et de quoi
sont-elles faites~?} L'anatomie de la \omterm{def:g8:reproductive-systems:male}{cellule mâle} indique
déjà la réponse~: tout ce que le père envoie tient dans un
\omterm{def:g6:cell-unit-of-life:parts}{noyau} (\cref{rem:g5:first-look-at-cells:nucleus} promettait
que cet emplacement comptait). Les deux chapitres suivants
l'ouvrent.
\end{remark}

\begin{method}[Faire l'inventaire d'un caractère]\label{met:g9:heredity-and-traits:audit}
Pour tout \omterm{def:g9:heredity-and-traits:traits}{caractère}, applique les trois tests~:
\begin{enumerate}
  \item place-le sur l'arbre généalogique~: suit-il les
        lignes de la reproduction, ou les vies et les
        habitudes~?
  \item date-le~: fixé dès le départ, ou construit en
        chemin~?
  \item applique le test des \omterm{def:g8:sexual-reproduction:gametes}{gamètes}~: deux \omterm{def:g5:first-look-at-cells:cell}{cellules}
        pourraient-elles le porter~? (une cicatrice, non~;
        un groupe sanguin, oui)~;
  \item verdict~: hérité, acquis, ou --- le cas le plus
        fréquent --- une marge héréditaire, remplie par la vie.
\end{enumerate}
\end{method}

\section{Problème~: la généalogiste du village}

\begin{problem}[{Problème du week-end --- trois générations,
un seul registre de caractères}]\label{pb:g9:heredity-and-traits:1}
Une généalogiste de village annote les arbres
généalogiques de la paroisse avec des \omterm{def:g9:heredity-and-traits:traits}{caractères}~: le
«~menton de la chapelle~» (un menton reconnaissable qui
revient depuis un siècle), la main gauche, la toux des
\omterm{def:g7:how-animals-breathe:four}{poumons} enfarinés des boulangers, les groupes sanguins
relevés sur le registre des dons, et les visages burinés
des bergers.

\medskip\noindent\textbf{Partie I --- Le tri du
registre.}
\begin{enumerate}
  \item Trie les cinq \omterm{def:g9:heredity-and-traits:traits}{caractères} avec
        \cref{met:g9:heredity-and-traits:audit}, en
        donnant le test décisif pour chacun.
  \item La toux suit la boulangerie, pas la famille~:
        entrer dans la famille par mariage l'apporte,
        partir la fait perdre. Quel test est ici à
        l'œuvre~?
  \item Le menton de la chapelle saute la génération du
        milieu dans deux \omterm{def:g1:plants-around-us:tree}{branches}. Tire la conclusion que
        cela force.
  \item Les groupes sanguins surprennent parfois le
        registre~: un enfant du groupe O né de parents du
        groupe A et du groupe B. Qu'a dû porter chacun des
        parents~?
\end{enumerate}

\medskip\noindent\textbf{Partie II --- La lecture en
déterminants.}
\begin{enumerate}[resume]
  \item Réécris le siècle du menton dans la langue des
        déterminants~: qu'est-ce qui a voyagé réellement
        sur quatre générations, et dans quels véhicules
        (\cref{def:g9:heredity-and-traits:heredity})~?
  \item Pourquoi les cousins porteurs du menton
        diffèrent-ils sur tout le reste~? Situe le
        brassage.
  \item La généalogiste ne trouve aucun cas --- en un
        siècle --- de cicatrice, de métier ou de langue
        passant à un nouveau-né. Énonce la généralisation,
        et sa raison au niveau du mécanisme.
  \item Dans une famille, les jumeaux se ressemblent
        jusqu'au menton et au-delà. Que conclut le
        registre sur leur origine
        (\cref{rem:g5:human-reproduction:twins})~?
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La conférence.}
\begin{enumerate}[resume]
  \item La généalogiste fait une conférence~: «~la
        paroisse transmet deux héritages --- l'un par les
        \omterm{def:g8:sexual-reproduction:gametes}{gamètes}, l'autre par la vie commune.~» Attribue
        les cinq \omterm{def:g9:heredity-and-traits:traits}{caractères} du registre aux deux canaux.
  \item «~Chacun de nous montre moins qu'il ne porte.~»
        Défends la phrase à partir du registre, deux fois.
  \item Un auditeur demande ce que sont physiquement les
        consignes du canal des \omterm{def:g8:sexual-reproduction:gametes}{gamètes}. Donne la réponse
        honnête de \cref{rem:g9:heredity-and-traits:question}
        --- emplacement connu, contenu au prochain chapitre.
  \item Termine la conférence par le verdict en une ligne
        de l'inventaire sur le cas le plus fréquent~: la
        plupart des \omterm{def:g9:heredity-and-traits:traits}{caractères} sont \emph{quoi}, vécus
        \emph{comment}~?
\end{enumerate}
\end{problem}
