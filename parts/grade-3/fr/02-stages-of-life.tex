\chapter{Les étapes de la vie}\label{ch:g3:stages-of-life}

Regarde une photo de famille~: le bébé sur les genoux de mamie,
l'écolier à qui manquent deux dents de devant, le cousin \omterm{def:g3:stages-of-life:stages}{adolescent}
soudain plus grand que sa mère, les parents, mamie elle-même. Une
seule famille, un seul après-midi --- et tout le cycle de la vie
humaine assis sur un canapé. Cette année, nous regardons ces étapes
de près, et la façon dont on peut mesurer la croissance elle-même.

\section{Les étapes d'une vie humaine}

\begin{definition}[Les étapes de la vie humaine]\label{def:g3:stages-of-life:stages}
Une vie humaine traverse des étapes~:
\begin{enumerate}
  \item le \emph{bébé}\index{bébé} (à peu près les deux premières
        années)~: la croissance la plus rapide de toutes, les
        premières dents, les premiers pas, les premiers mots~;
  \item l'\emph{enfant}\index{enfant}~: une croissance régulière,
        les \omterm{prop:g2:teeth-and-eating-well:sets}{dents de lait} échangées contre les \omterm{prop:g2:teeth-and-eating-well:sets}{dents d'adulte}, des
        apprentissages de toutes sortes~;
  \item l'\emph{adolescent}\index{adolescent}~: une nouvelle
        poussée de croissance rapide, et le corps qui passe de
        celui d'un enfant à celui d'un adulte~;
  \item l'\emph{adulte}\index{adulte}~: la croissance en taille est
        finie~; les adultes peuvent devenir parents~;
  \item la \emph{vieillesse}\index{vieillesse}~: les cheveux
        blanchissent, la \omterm{def:g1:the-five-senses:sense}{peau} se ride, les forces baissent
        lentement --- la dernière étape naturelle du cycle.
\end{enumerate}
\end{definition}

\begin{example}[Lire le canapé]\label{ex:g3:stages-of-life:sofa}
Sur le canapé familial~: le bébé (étape 1), l'écolier édenté (étape
2 --- la relève de \cref{ch:g2:teeth-and-eating-well} en plein
travail), le cousin de seize ans (étape 3, en pleine poussée), les
parents (étape 4), mamie (étape 5). Cinq étapes, une famille.
\end{example}

\begin{remark}[On ne grandit pas régulièrement]\label{rem:g3:stages-of-life:bursts}
La croissance a un rythme~: très rapide chez le bébé --- la taille
augmente à peu près de moitié la première année --- plus lente et
régulière pendant l'enfance, rapide à nouveau chez l'\omterm{def:g3:stages-of-life:stages}{adolescent},
puis terminée. Deux poussées avec un long calme entre elles~: voilà
la forme normale d'une croissance.
\end{remark}

\section{Mesurer la croissance}

\begin{definition}[Courbe de croissance]\label{def:g3:stages-of-life:chart}
Une \emph{courbe de croissance}\index{courbe de croissance} est un
relevé de la taille (ou du poids) mesurée encore et encore au fil du
temps --- les traits du montant de porte de \cref{ch:g1:my-body},
rendus réguliers et datés. Les médecins en tiennent une pour chaque
enfant.
\end{definition}

\begin{example}[Lire la courbe]\label{ex:g3:stages-of-life:reading}
Sur la courbe, la ligne monte d'environ $\qty{25}{cm}$ la toute
première année --- et de $\qty{6}{cm}$ par an seulement pendant
l'enfance. Les deux parties raides sont les deux poussées de
\cref{rem:g3:stages-of-life:bursts}~; le bout plat dit que l'\omterm{def:g3:stages-of-life:stages}{adulte}
a cessé de grandir.
\end{example}

\begin{method}[Tenir sa propre courbe de croissance]\label{met:g3:stages-of-life:keep}
\begin{enumerate}
  \item Mesure la taille avec soin (la méthode du mur et du livre
        des années précédentes), toujours sans chaussures~;
  \item note la date et la taille dans un carnet, ou marque le
        montant de la porte et date le trait~;
  \item recommence tous les quelques mois --- la régularité compte
        plus que la fréquence~;
  \item pour voir l'histoire, place les points~: l'âge en bas, la
        taille sur le côté, comme dans la figure.
\end{enumerate}
\end{method}

\section{Les animaux et les plantes ont aussi des étapes}

\begin{proposition}[Des étapes dans tout le monde vivant]\label{prop:g3:stages-of-life:animals}
Tout \omterm{def:g1:living-or-not:living}{être vivant} traverse les étapes de sa propre version du cycle
--- jeune, \omterm{def:g3:stages-of-life:stages}{adulte}, \omterm{def:g3:stages-of-life:stages}{vieillesse}. L'étape jeune peut ressembler à
l'\omterm{def:g3:stages-of-life:stages}{adulte} (le chaton, le poulain) ou ne lui ressembler en rien (le
têtard, la \omterm{ex:g2:animals-grow-up:butterfly}{chenille} de \cref{ch:g2:animals-grow-up})~; les étapes
d'une \omterm{def:g1:plants-around-us:plant}{plante} vont de la \omterm{def:g2:from-seed-to-plant:seed}{graine} à la \omterm{prop:g2:from-seed-to-plant:cycle}{jeune pousse}, puis à la jeune
\omterm{def:g1:plants-around-us:plant}{plante} et à l'\omterm{def:g3:stages-of-life:stages}{adulte} en \omterm{def:g1:plants-around-us:parts}{fleur}. Ce qu'aucun d'eux ne peut faire,
c'est sauter une étape ou les parcourir à l'envers.
\end{proposition}
\admitted

\begin{example}[À quelle vitesse]\label{ex:g3:stages-of-life:speeds}
Les vitesses diffèrent énormément. Une mouche est \omterm{def:g3:stages-of-life:stages}{adulte} en quelques
semaines~; un chat en un an environ~; un humain en près de vingt
ans~; un chêne met plusieurs dizaines d'années avant ses premiers
glands. Lent ou rapide, l'ordre des étapes ne change jamais.
\end{example}

\begin{remark}[La vieillesse est une étape, pas un défaut]\label{rem:g3:stages-of-life:oldage}
Les cheveux blancs et les rides ne sont pas une maladie~: c'est la
dernière étape du corps, aussi naturelle que les premières dents du
bébé. Dans beaucoup d'espèces, et dans les familles humaines, les
anciens portent ce dont les jeunes ont le plus besoin ---
l'expérience.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:1}
Nomme les cinq étapes d'une vie humaine, dans l'ordre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:2}
À quelle étape~: (a) les \omterm{prop:g2:teeth-and-eating-well:sets}{dents de lait} tombent-elles~? (b) peut-on
devenir parent~? (c) la croissance est-elle la plus rapide de toute
la vie~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:3}
Quelles sont les deux poussées de croissance, et qu'y a-t-il entre
elles~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:4}
Sur la \omterm{def:g3:stages-of-life:chart}{courbe de croissance}, comment vois-tu, rien qu'à la ligne,
qu'un \omterm{def:g3:stages-of-life:stages}{adulte} a cessé de grandir~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:5}
À l'aide de la courbe, de combien de centimètres la ligne monte-t-elle
entre $6$ et $8$ ans~? Et pendant la seule première année~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:6}
Pourquoi faut-il toujours mesurer la taille sans chaussures, et de
la même façon chaque fois~? Que pourrait-il arriver sinon~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:7}
Donne les étapes de la vie d'un chêne, du gland au vieil arbre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:stages-of-life:8}
À toi d'essayer~: commence pour toi-même la \omterm{def:g3:stages-of-life:chart}{courbe de croissance} de
\cref{met:g3:stages-of-life:keep}, avec le premier point
d'aujourd'hui. À quelle étape de
\cref{def:g3:stages-of-life:stages} es-tu --- et quelle poussée est
encore devant toi~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:9}
Une mouche, un chat, un humain et un chêne parcourent les mêmes
étapes à des vitesses très différentes. Range-les de la plus rapide
à la plus lente à atteindre l'âge \omterm{def:g3:stages-of-life:stages}{adulte} --- et dis ce que leurs
quatre suites d'étapes ont en commun.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:stages-of-life:10}
Tom, $9$ ans, se mesure deux fois dans le même mois et ne trouve
«~aucune croissance~». Doit-il s'inquiéter~? Réponds avec
\cref{rem:g3:stages-of-life:bursts} et la forme de la courbe à son
âge.
\end{exercise}
