\section*{Chapitre \ref{ch:g11:genes-and-disease} --- Variation génétique et maladies}

\begin{solution}{exo:g11:genes-and-disease:1}
\omterm{def:g11:genes-and-disease:genetic}{Récessif}~: la maladie n'apparaît qu'avec deux copies de l'\omterm{def:g10:universal-dna:gene}{allèle}~;
\omterm{def:g11:genes-and-disease:genetic}{dominant}~: une copie suffit. Un \omterm{def:g11:genes-and-disease:genetic}{porteur} détient un \omterm{def:g10:universal-dna:gene}{allèle} muté \omterm{def:g11:genes-and-disease:genetic}{récessif}
et reste en bonne santé.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:2}
Atteint $1/4$, \omterm{def:g11:genes-and-disease:genetic}{porteur} $1/2$, ni l'un ni l'autre $1/4$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:3}
I-1 $Aa$, I-2 $Aa$ (parents d'un enfant atteint), II-2 $aa$, III-2
$aa$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:4}
Un homme n'a qu'un \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome} X~: un \omterm{def:g10:universal-dna:gene}{allèle} \omterm{def:g11:genes-and-disease:genetic}{récessif} qu'il porte est
exprimé, faute d'une seconde copie pour le masquer. Une femme doit
porter l'\omterm{def:g10:universal-dna:gene}{allèle} sur ses deux \omterm{def:g10:universal-dna:information}{chromosomes} X.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:5}
Une maladie dont le risque dépend de nombreux \omterm{def:g10:universal-dna:gene}{allèles} de faible effet
et de l'environnement~: le diabète de type 2, les maladies du cœur
(mais aussi l'asthme, la plupart des cancers).
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:6}
$aa \times AA$~: tous les enfants $Aa$, \omterm{def:g11:genes-and-disease:genetic}{porteurs} sains. Un enfant
\omterm{def:g11:genes-and-disease:genetic}{porteur} avec un \omterm{def:g11:genes-and-disease:genetic}{porteur}~: $1/4$ atteint, $1/2$ \omterm{def:g11:genes-and-disease:genetic}{porteurs}, $1/4$ non
\omterm{def:g11:genes-and-disease:genetic}{porteurs}.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:7}
II-1 est une enfant saine de deux \omterm{def:g11:genes-and-disease:genetic}{porteurs}~: porteuse avec une
probabilité de $2/3$. Risque avec un homme de la population~: $2/3 \times 1/25 \times 1/4
= 1/150$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:8}
$1/2$ pour chaque enfant. Aucun des quatre~: $(1/2)^4 = 1/16$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:9}
$X^hY \times X^HX^H$~: toutes les filles $X^HX^h$, porteuses saines~;
tous les fils $X^HY$, sains. Une fille porteuse avec un homme sain~: la
moitié de ses fils hémophiles, la moitié de ses filles porteuses.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:10}
Un \omterm{def:g10:universal-dna:gene}{allèle} donne 400 et 200~; l'autre, avec la délétion
supplémentaire, donne 300 et 200. Bandes à 400, 300 et 200 --- et
aucune à 600.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:11}
La fréquence d'un \omterm{def:g10:universal-dna:gene}{allèle} dans une population dépend de son histoire, et
non du seul taux de \omterm{def:g11:mutations:mutation}{mutation}~: l'\omterm{def:g10:universal-dna:gene}{allèle} européen est apparu il y a
longtemps et s'est répandu, peut-être parce que les \omterm{def:g11:genes-and-disease:genetic}{porteurs} avaient un
avantage (on soupçonne une résistance à une maladie diarrhéique)~; en
Asie de l'Est, ce ne fut pas le cas.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:12}
Chaque enfant est un tirage indépendant~: $1/4$ pour le deuxième. Au
moins l'un des deux suivants~: $1 - (3/4)^2 = 7/16$. Les \omterm{def:g10:universal-dna:gene}{allèles} sont
redistribués à chaque conception~; un enfant atteint auparavant
n'abaisse pas le risque du suivant.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:13}
Avant le test, porteuse avec une probabilité de $1/25$~; le test
manque 10~\% des \omterm{def:g11:genes-and-disease:genetic}{porteurs}, si bien que sur 1000 femmes, 40 sont
porteuses et 4 ont un test négatif, tandis que 960 non-porteuses ont un
test négatif~: probabilité d'être porteuse
$4/964 \approx 1/240$. Risque avec un \omterm{def:g11:genes-and-disease:genetic}{porteur} avéré~: $1/240 \times 1/4
\approx 1/960$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:14}
De beaucoup à peu d'\omterm{def:g10:universal-dna:gene}{allèles}, en restant sédentaire~: de 34~\% à 9~\%~;
de sédentaire à actif, avec beaucoup d'\omterm{def:g10:universal-dna:gene}{allèles}~: de 34~\% à 12~\%. Le
changement que l'on peut faire apporte presque la même réduction que
celui qu'on ne peut pas faire. Un score de risque sert à repérer qui
gagne le plus à changer de mode de vie, non à rendre un verdict.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:15}
Le test apporte une certitude --- la sortie d'une ombre à cinquante
pour cent s'il est négatif, la possibilité d'organiser sa vie et sa
famille s'il est positif --- et coûte, s'il est positif, la
connaissance d'une maladie incurable des décennies à l'avance, avec des
effets sur l'humeur, l'assurance et les proches qui partagent le \omterm{def:g10:universal-dna:gene}{gène}.
Aucun traitement ne découle du résultat, si bien que la seule raison de
savoir est celle de la personne elle-même~; c'est pourquoi le choix, et
son moment, lui appartiennent.
\end{solution}

\begin{solution}{pb:g11:genes-and-disease:1}
\textbf{1.} Récessive~: Zoé est atteinte et ses deux parents sont
sains, chacun porte donc un \omterm{def:g10:universal-dna:gene}{allèle} muté caché.

\textbf{2.} Léa $Aa$, Tom $Aa$, Zoé $aa$.

\textbf{3.} $AA$ avec une probabilité de $1/3$, $Aa$ avec $2/3$ (il est
sain, donc $aa$ est exclu).

\textbf{4.} Tom est \omterm{def:g11:genes-and-disease:genetic}{porteur}, donc l'un au moins de ses parents l'est~;
comme personne n'est atteint dans cette génération, Ana a reçu l'\omterm{def:g10:universal-dna:gene}{allèle}
de ce parent avec une probabilité de $1/2$~: $1/2$ (en prenant l'autre
parent pour un \omterm{def:g11:genes-and-disease:genetic}{non-porteur}).

\textbf{5.} $1/25$, le chiffre de la population.

\textbf{6.} $1/2 \times 1/25 \times 1/4 = 1/200$.

\textbf{7.} Atteint $1/4$~; \omterm{def:g11:genes-and-disease:genetic}{porteur} $1/2$.

\textbf{8.} $2/3 \times 1/25 \times 1/4 = 1/150$.

\textbf{9.} Zoé est $aa$~; son conjoint est $Aa$ avec une probabilité
de $1/25$. Atteint~: $1/25 \times 1/2 = 1/50$. \omterm{def:g11:genes-and-disease:genetic}{Porteur}~: chaque enfant
reçoit $a$ de Zoé, donc \omterm{def:g11:genes-and-disease:genetic}{porteur} avec une probabilité de $24/25 \times 1 + 1/25 \times
1/2 \approx 0.98$.

\textbf{10.} Lou est \omterm{def:g11:genes-and-disease:genetic}{porteur} avec une probabilité de $2/3$~; le cousin
a reçu $a$ d'Ana (porteuse avec $1/2$) avec une probabilité de $1/2$,
il est donc \omterm{def:g11:genes-and-disease:genetic}{porteur} avec une probabilité de $1/4$ (en négligeant
Karim)~: $2/3 \times 1/4
\times 1/4 = 1/24$, quatre fois plus qu'avec Max, parce que le cousin
vient de la même famille et est bien plus susceptible qu'un inconnu de
porter le même \omterm{def:g10:universal-dna:gene}{allèle}.

\textbf{11.} Zoé~: 600 et 300 seulement. Tom~: 900, 600 et 300.
\omterm{def:g11:genes-and-disease:genetic}{Non-porteur}~: 900 seulement.

\textbf{12.} $AA$~: il n'est pas \omterm{def:g11:genes-and-disease:genetic}{porteur}~; le risque pour son enfant
tombe de $1/150$ à zéro (sauf \omterm{def:g11:mutations:mutation}{mutation} nouvelle).

\textbf{13.} Ana est $Aa$~: risque $1 \times 1/25 \times 1/4 = 1/100$.

\textbf{14.} Sur 1000 personnes comme Karim, 40 sont porteuses dont 4
ont un test négatif, et 960 non-porteuses ont un test négatif~: $4/964 \approx
1/240$. Risque~: $1 \times 1/240 \times 1/4 \approx 1/960$.

\textbf{15.} $Aa$~: un \omterm{def:g11:genes-and-disease:genetic}{porteur} sain, comme sa mère.

\textbf{16.} $(1/25)^2 = 1/625$ des couples sont formés de deux
\omterm{def:g11:genes-and-disease:genetic}{porteurs}~; un quart de leurs enfants sont atteints~: $1/2500$.

\textbf{17.} Exactement la fréquence observée~: la fréquence des
\omterm{def:g11:genes-and-disease:genetic}{porteurs} et la règle du \omterm{def:g11:genes-and-disease:genetic}{récessif} en rendent compte.

\textbf{18.} Avec une fréquence de \omterm{def:g11:genes-and-disease:genetic}{porteurs} de $1/25$ et une fréquence
d'atteints de $1/2500$, les \omterm{def:g10:universal-dna:gene}{allèles} mutés portés par les \omterm{def:g11:genes-and-disease:genetic}{porteurs} sains
l'emportent sur ceux des personnes atteintes dans le rapport de
$2 \times 1/25$ à $2 \times
1/2500$, soit cent contre un. La mort des personnes atteintes ne retire
que 1~\% des copies par génération~; le reste est transmis
invisiblement par les \omterm{def:g11:genes-and-disease:genetic}{porteurs}.

\textbf{19.} La fréquence de l'\omterm{def:g10:universal-dna:gene}{allèle} va monter, mais très lentement~:
le 1~\% des copies qui se perdait à chaque génération est désormais
conservé, et le changement est donc de l'ordre de 1~\% de la fréquence
par génération.

\textbf{20.} Deux \omterm{def:g11:genes-and-disease:genetic}{porteurs} risquent un enfant atteint sur quatre~; le
frère ou la sœur sain d'un enfant atteint est \omterm{def:g11:genes-and-disease:genetic}{porteur} deux fois sur
trois~; le gel a changé des probabilités en \omterm{def:g11:enzymes-and-phenotype:phenotype}{génotypes} --- innocentant
Max, confirmant Ana, rassurant sans l'innocenter Karim --- et a
identifié le fœtus comme un \omterm{def:g11:genes-and-disease:genetic}{porteur} sain.
\end{solution}
