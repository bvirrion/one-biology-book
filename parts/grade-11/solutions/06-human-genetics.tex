\section*{Chapter \ref{ch:g11:human-genetics} --- Human Genetics}

\begin{solution}{exo:g11:human-genetics:1}
A filled square represents an affected male. An open circle represents an
unaffected female. Squares are males, circles are females; shading marks the
trait of interest in the \omterm{def:g11:human-genetics:pedigree}{pedigree}.
\end{solution}

\begin{solution}{exo:g11:human-genetics:2}
\omterm{def:g11:mendelian-genetics:zygous}{Heterozygous} carriers of a \omterm{def:g11:mendelian-genetics:dominance}{recessive allele} are unaffected, so intermediate
generations can look free of the trait. The \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype} reappears when two
carriers have a \omterm{def:g11:mendelian-genetics:zygous}{homozygous} \omterm{def:g11:mendelian-genetics:dominance}{recessive} child. That ``skipping'' is classic for
autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive} inheritance in \omterm{def:g11:human-genetics:pedigree}{pedigrees}.
\end{solution}

\begin{solution}{exo:g11:human-genetics:3}
Males have only one \omterm{def:g11:human-genetics:sexchrom}{X chromosome}, so they carry a single \omterm{def:g11:mendelian-genetics:allele}{allele} of each
\omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:gene-expression:gene}{gene}. That \omterm{def:g11:mendelian-genetics:allele}{allele} is expressed whether it would behave as \omterm{def:g11:mendelian-genetics:dominance}{recessive}
or \omterm{def:g11:mendelian-genetics:dominance}{dominant} in a female who has two \omterm{def:g11:human-genetics:sexchrom}{X chromosomes} --- the \omterm{def:g11:human-genetics:xlinked}{hemizygous} state.
\end{solution}

\begin{solution}{exo:g11:human-genetics:4}
The woman received her father's only X, which carries the colour-deficiency
\omterm{def:g11:mendelian-genetics:allele}{allele}, so she is at least a carrier $X^{A}X^{a}$. With a normal husband
$X^{A}Y$, sons are colour-deficient with probability $\tfrac12$; daughters are
not colour-deficient under complete recessivity, but half are carriers on
average.
\end{solution}

\begin{solution}{exo:g11:human-genetics:5}
A son receives his \omterm{def:g11:human-genetics:sexchrom}{X chromosome} from his mother and his Y from his father.
The father's X is transmitted only to daughters. Therefore an \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:allele}{allele}
cannot pass from father to son.
\end{solution}

\begin{solution}{exo:g11:human-genetics:6}
\omterm{def:g11:meiosis-life-cycles:nondisjunction}{Nondisjunction} is the failure of \omterm{def:g11:dna-replication:chromosome}{chromosomes} or chromatids to separate
properly during \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis} (or \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}). Resulting gametes may be $n+1$ or $n-1$,
producing \omterm{def:g11:meiosis-life-cycles:nondisjunction}{aneuploid} zygotes such as \omterm{prop:g11:human-genetics:trisomy}{trisomy 21}, \omterm{prop:g11:human-genetics:trisomy}{XXY} (Klinefelter) or X0
(Turner).
\end{solution}

\begin{solution}{exo:g11:human-genetics:7}
Both parents must be \omterm{def:g11:mendelian-genetics:zygous}{heterozygous} carriers $Aa$. Each future child, of either
sex, has probability $\tfrac14$ of being affected $aa$. Successive pregnancies
are independent: having one affected child does not use up the risk.
\end{solution}

\begin{solution}{exo:g11:human-genetics:8}
\omterm{def:g11:gene-expression:gene}{Single-gene} Mendelian traits often fall into discrete phenotypic classes (e.g.\
a $3:1$ ratio). \omterm{def:g11:human-genetics:polygenic}{Polygenic traits} typically show continuous variation with a
unimodal, often roughly bell-shaped, distribution in the \omterm{def:g10:ecosystems:levels}{population} because
many loci and environment create many intermediate classes.
\end{solution}

\begin{solution}{exo:g11:human-genetics:9}
An affected male passes his X to all daughters (who become carriers or
affected depending on the \omterm{def:g11:mendelian-genetics:allele}{allele}) and his Y to all sons. Sons therefore do not
inherit his \omterm{def:g11:human-genetics:xlinked}{X-linked} disease \omterm{def:g11:mendelian-genetics:allele}{allele} --- matching the \omterm{def:g11:human-genetics:pedigree}{pedigree} pattern of no
father-to-son transmission and obligatory carrier status of daughters for
\omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive} disease \omterm{def:g11:mendelian-genetics:allele}{alleles}.
\end{solution}

\begin{solution}{exo:g11:human-genetics:10}
An extra large autosome imbalances dosage for thousands of \omterm{def:g11:gene-expression:gene}{genes}, disrupting
\omterm{def:g10:scientific-biology:properties}{development} so severely that embryos usually die. \omterm{def:g11:dna-replication:chromosome}{Chromosome} 21 is relatively
gene-poor; three copies still cause Down syndrome but can be compatible with
live birth. \omterm{def:g11:gene-expression:gene}{Gene} dosage, not merely ``an extra \omterm{def:g11:dna-replication:chromosome}{chromosome}'' as an abstract
object, drives severity.
\end{solution}

\begin{solution}{exo:g11:human-genetics:11}
Children of the affected mother may inherit her \omterm{rem:g11:human-genetics:mtDNA}{mtDNA} variant (\omterm{rem:g11:human-genetics:mtDNA}{maternal
inheritance}), with severity possibly varying by \omterm{rem:g11:human-genetics:mtDNA}{heteroplasmy}. An affected
father usually does not transmit \omterm{rem:g11:human-genetics:mtDNA}{mtDNA} disease because sperm contribute
negligible cytoplasm and mitochondria to the zygote; \omterm{rem:g11:human-genetics:mtDNA}{mtDNA} comes almost
entirely from the egg.
\end{solution}

\begin{solution}{exo:g11:human-genetics:12}
Draw or confirm the \omterm{def:g11:human-genetics:pedigree}{pedigree}; both parents are $Aa$ carriers. For each
pregnancy: $\tfrac14$ affected ($aa$), $\tfrac12$ carrier ($Aa$), $\tfrac14$
neither ($AA$) --- or among unaffected children, $\tfrac23$ are carriers.
Explain test options and limits without directing the couple's decision.
Previous outcomes do not change the next pregnancy's Mendelian probabilities:
each fertilisation is an independent draw from the same gamete ratios.
\end{solution}

\begin{solution}{pb:g11:human-genetics:1}
\textbf{1.} Autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive} is readily compatible (daughter $aa$ from
unaffected carrier parents). \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive} affected daughters require an
affected father (or rare patterns); with both parents unaffected, an affected
daughter is unlikely for a rare \omterm{def:g11:human-genetics:xlinked}{X-linked} \omterm{def:g11:mendelian-genetics:dominance}{recessive} disease.

\textbf{2.} Parents $Aa \times Aa$; a future son is affected with probability
$\tfrac14$, same as a daughter for an autosomal \omterm{def:g11:gene-expression:gene}{gene}.

\textbf{3.} Autosomal \omterm{def:g11:mendelian-genetics:dominance}{recessive} affected man $aa \times AA$: all children
carriers, none affected. \omterm{def:g11:human-genetics:xlinked}{X-linked}: affected man $X^{a}Y \times X^{A}X^{A}$:
all daughters carriers $X^{A}X^{a}$, all sons $X^{A}Y$ unaffected --- no
father-to-son transmission of the disease \omterm{def:g11:mendelian-genetics:allele}{allele}.

\textbf{4.} \omterm{def:g11:human-genetics:pedigree}{Pedigree}: open circle (carrier mother), open square (father);
filled square (affected son); open circle daughter (often half-shaded if the
convention marks known carriers).

\textbf{5.} A karyotype shows three \omterm{def:g11:dna-replication:chromosome}{chromosomes} 21 (or extra $21$ material);
a \omterm{def:g11:gene-expression:gene}{single-gene} disorder typically has a normal \omterm{def:g11:dna-replication:chromosome}{chromosome} count and needs
DNA-level testing.

\textbf{6.} Children inherit a reshuffled mix of many height \omterm{def:g11:mendelian-genetics:allele}{alleles}; they can
receive a higher load of ``tall'' \omterm{def:g11:mendelian-genetics:allele}{alleles} than either parent average, and
environment (nutrition, health) can raise realized height.
\end{solution}
