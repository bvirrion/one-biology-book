\section*{Chapitre \ref{ch:g6:cell-unit-of-life} --- La cellule, unité du vivant}

\begin{solution}{exo:g6:cell-unit-of-life:1}
La \omterm{def:g6:cell-unit-of-life:parts}{membrane} --- la fine enveloppe qui tient la \omterm{def:g5:first-look-at-cells:cell}{cellule} d'un seul
tenant et contrôle ce qui entre et ce qui sort~; le \omterm{def:g6:cell-unit-of-life:parts}{cytoplasme} ---
la gelée de remplissage où se fait le travail~; le \omterm{def:g6:cell-unit-of-life:parts}{noyau} --- le
centre de commande qui dirige la vie de la \omterm{def:g5:first-look-at-cells:cell}{cellule} et sa division.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:2}
La paroi rigide --- la tenue des \omterm{def:g1:plants-around-us:plant}{plantes} sans \omterm{def:g3:bones-and-muscles:skeleton}{os}~; les grains verts
--- les ateliers du métier des \omterm{def:g5:food-webs:roles}{producteurs}, qui captent la
\omterm{def:g6:exploring-our-environment:conditions}{lumière}~; la grande poche d'eau --- sa pression garde la \omterm{def:g5:first-look-at-cells:cell}{cellule}
bien gonflée.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:3}
Oignon~: des parois droites qui se rejoignent en angles, le \omterm{def:g6:cell-unit-of-life:parts}{noyau}
repoussé sur le côté. Joue~: un contour souple et arrondi avec la
\omterm{def:g6:cell-unit-of-life:parts}{membrane} seule, le \omterm{def:g6:cell-unit-of-life:parts}{noyau} vers le milieu.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:4}
Un organisme complet fait d'exactement une \omterm{def:g5:first-look-at-cells:cell}{cellule}. La \omterm{ex:g6:producing-our-food:bread}{levure}~; la
rameuse en forme de pantoufle de la goutte de mare (ou ses nageurs
verts, la masse qui coule, les \omterm{def:g6:producing-our-food:fermentation}{microbes} du yaourt).
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:5}
Tout \omterm{def:g6:exploring-our-environment:components}{être vivant} est fait d'une ou de plusieurs \omterm{def:g5:first-look-at-cells:cell}{cellules}, les plus
petites \omterm{prop:g5:first-look-at-cells:principle}{unités du vivant}~; et toute \omterm{def:g5:first-look-at-cells:cell}{cellule} provient de la division
d'une \omterm{def:g5:first-look-at-cells:cell}{cellule} existante.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:6}
En se divisant en deux~: une \omterm{def:g5:first-look-at-cells:cell}{cellule} devient deux organismes
complets.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:7}
Les parois rigides et les poches d'eau bien gonflées~: des millions
de \omterm{def:g5:first-look-at-cells:cell}{cellules} à parois et sous pression s'épaulent comme un briquetage
gonflé. Se faner, c'est le relâchement des poches --- la pression
perdue faute d'eau, et toute la structure qui s'affaisse.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:8}
Les grains verts. Les \omterm{def:g1:animals-around-us:animal}{animaux} sont des \omterm{def:g5:food-webs:roles}{consommateurs} ---
transformateurs de \omterm{def:g6:plants-are-producers:matter}{matière organique} mangée, jamais fabricants à
partir du minéral --- si bien que leurs \omterm{def:g5:first-look-at-cells:cell}{cellules} ne portent aucun
grain de fabrication~: l'équipement va avec le métier, \omterm{def:g5:first-look-at-cells:cell}{cellule} par
\omterm{def:g5:first-look-at-cells:cell}{cellule}.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:9}
Par exemple~: les \omterm{def:g5:first-look-at-cells:cell}{cellules} de revêtement (celles de la joue), qui
tapissent les surfaces~; les \omterm{def:g5:first-look-at-cells:cell}{cellules} musculaires, qui se
contractent pour tirer les \omterm{def:g3:bones-and-muscles:skeleton}{os}~; les \omterm{def:g5:first-look-at-cells:cell}{cellules} nerveuses, qui portent
les rapports et les ordres de la boucle du \omterm{def:g5:brain-in-command:system}{cerveau}.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:10}
Éraflure qui guérit~: des \omterm{def:g5:first-look-at-cells:cell}{cellules} de \omterm{def:g1:the-five-senses:sense}{peau} se divisent pour
reconstruire la zone. Pâte qui double~: des \omterm{def:g5:first-look-at-cells:cell}{cellules} de \omterm{ex:g6:producing-our-food:bread}{levure} se
divisent encore et encore dans la pâte tiède et sucrée. Têtard qui
grandit~: ses \omterm{def:g5:first-look-at-cells:cell}{cellules} se divisent et se spécialisent à mesure que
le corps se bâtit.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:11}
Le métier des \omterm{def:g5:food-webs:roles}{producteurs}~: une fabrication \omterm{def:g6:cell-unit-of-life:unicellular}{unicellulaire} de \omterm{def:g6:plants-are-producers:matter}{matière
organique} à partir de fournitures minérales et de \omterm{def:g6:exploring-our-environment:conditions}{lumière}. Dans le
réseau de la mare, ce sont des premiers maillons --- l'étage vert
que broutent les \omterm{ex:g2:animals-grow-up:frog}{têtards} et les puces d'eau.
\end{solution}

\begin{solution}{exo:g6:cell-unit-of-life:12}
Chacune de tes \omterm{def:g5:first-look-at-cells:cell}{cellules} vient d'une division, dont la \omterm{def:g5:first-look-at-cells:cell}{cellule} mère
venait d'une division, sans une seule rupture --- en remontant par
la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée, par les \omterm{def:g5:first-look-at-cells:cell}{cellules} de tes parents, et
au-delà, dans le passé profond du vivant. Conséquence en une
phrase~: toute \omterm{def:g5:first-look-at-cells:cell}{cellule} vivante, y compris les tiennes, est le
dernier maillon d'une chaîne ininterrompue de divisions aussi
ancienne que la vie elle-même.
\end{solution}

\begin{solution}{pb:g6:cell-unit-of-life:1}
\textbf{1.} La paroi~: la paroi de la \omterm{def:g5:first-look-at-cells:cell}{cellule} végétale. La tache~:
le \omterm{def:g6:cell-unit-of-life:parts}{noyau}. Le coupable du tassement~: la grande poche d'eau, qui
presse le \omterm{def:g6:cell-unit-of-life:parts}{noyau} sur le côté.

\textbf{2.} A possède la paroi, la poche d'eau (et, dans les parties
vertes, des grains verts) qui manquent à B. Les deux partagent le
trousseau~: \omterm{def:g6:cell-unit-of-life:parts}{membrane}, \omterm{def:g6:cell-unit-of-life:parts}{cytoplasme}, \omterm{def:g6:cell-unit-of-life:parts}{noyau}.

\textbf{3.} Les grains sont les ateliers qui captent la \omterm{def:g6:exploring-our-environment:conditions}{lumière} ---
et un \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbe} souterrain vit dans le noir, où les ateliers resteraient
inactifs~: la \omterm{def:g1:plants-around-us:plant}{plante} les bâtit dans ses \omterm{def:g1:plants-around-us:parts}{feuilles} éclairées, pas dans
son cellier enterré.

\textbf{4.} Les \omterm{def:g5:first-look-at-cells:cell}{cellules} des élèves eux-mêmes. D'après la phrase 2,
chacune vient de la division d'une \omterm{def:g5:first-look-at-cells:cell}{cellule} existante --- une chaîne
de divisions qui remonte à la \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée de chaque élève.

\textbf{5.} La rameuse~: elle se déplace, se dirige, se nourrit ---
un organisme \omterm{def:g6:cell-unit-of-life:unicellular}{unicellulaire} complet (de type \omterm{def:g1:animals-around-us:animal}{animal}). Les nageurs
verts~: ils se déplacent et captent la \omterm{def:g6:exploring-our-environment:conditions}{lumière} --- des \omterm{def:g5:food-webs:roles}{producteurs}
\omterm{def:g6:cell-unit-of-life:unicellular}{unicellulaires}. La masse~: elle coule et engloutit sa nourriture ---
un chasseur \omterm{def:g6:cell-unit-of-life:unicellular}{unicellulaire}. Chacun montre les activités de toute une
vie dans une seule \omterm{def:g5:first-look-at-cells:cell}{cellule}.

\textbf{6.} Les grains verts --- l'équipement qui capte la \omterm{def:g6:exploring-our-environment:conditions}{lumière}
et fait tourner la fabrication des \omterm{def:g5:food-webs:roles}{producteurs}.

\textbf{7.} Elles se sont nourries du sucre du \omterm{def:g2:animals-grow-up:born}{lait} et ont libéré
l'acide qui l'a fait prendre en yaourt~: ce sont des \omterm{def:g6:exploring-our-environment:components}{êtres vivants}
\omterm{def:g6:cell-unit-of-life:unicellular}{unicellulaires} --- la main-d'œuvre de \omterm{def:g6:producing-our-food:fermentation}{microbes}, enfin vue en
personne.

\textbf{8.} Regarder un grain se diviser~: un qui devient deux, deux
qui deviennent quatre --- la division est la signature que la phrase
2 exige du vivant. (Sur une lame tiédie, les \omterm{def:g6:producing-our-food:fermentation}{microbes} du yaourt
obligent en quelques heures.)

\textbf{9.} A~: beaucoup de \omterm{def:g5:first-look-at-cells:cell}{cellules}, végétal. B~: beaucoup de
\omterm{def:g5:first-look-at-cells:cell}{cellules} (un échantillon d'un \omterm{def:g1:animals-around-us:animal}{animal} pluricellulaire), \omterm{def:g1:animals-around-us:animal}{animal}. C~:
des \omterm{def:g5:first-look-at-cells:cell}{cellules} uniques --- de type \omterm{def:g1:animals-around-us:animal}{animal} (la rameuse, la masse) et de
type végétal (les nageurs verts) \omterm{def:g3:bones-and-muscles:skeleton}{côte} à \omterm{def:g3:bones-and-muscles:skeleton}{côte}. D~: des \omterm{def:g5:first-look-at-cells:cell}{cellules}
uniques, ni végétales ni \omterm{def:g1:animals-around-us:animal}{animales} --- des \omterm{def:g6:producing-our-food:fermentation}{microbes}.

\textbf{10.} \omterm{def:g6:cell-unit-of-life:parts}{Membrane}, \omterm{def:g6:cell-unit-of-life:parts}{cytoplasme}, \omterm{def:g6:cell-unit-of-life:parts}{noyau}. Son universalité, de
l'oignon à la joue, de la mare au yaourt, est la preuve de l'\omterm{prop:g5:first-look-at-cells:principle}{unité
du vivant}~: un seul plan de construction sous tout cela.

\textbf{11.} L'oignon a poussé à partir d'une \omterm{def:g5:first-look-at-cells:cell}{cellule} fabriquée par
sa \omterm{def:g1:plants-around-us:plant}{plante} mère, par divisions~; l'élève a commencé par une seule
\omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée~; chaque chaîne de yaourt a commencé par un
\omterm{def:g1:healthy-habits:germs}{microbe} qui se divisait encore et encore. Tout ce qui a été dessiné
remonte à une \omterm{def:g5:first-look-at-cells:cell}{cellule} de départ unique.

\textbf{12.} Phrase 1 --- tout \omterm{def:g6:exploring-our-environment:components}{être vivant} est fait de \omterm{def:g5:first-look-at-cells:cell}{cellules}~:
mieux montrée par A et B (et par la visite dans son ensemble).
Phrase 2 --- toute \omterm{def:g5:first-look-at-cells:cell}{cellule} vient d'une \omterm{def:g5:first-look-at-cells:cell}{cellule} qui se divise~: mieux
montrée par C et D, où la division est un mode de vie visible.
\end{solution}
