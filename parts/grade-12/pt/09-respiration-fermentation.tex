\chapter{A respiração celular e a fermentação}\label{ch:g12:respiration-fermentation}

Os músculos de um maratonista gastam, ao longo de três horas e meia,
cerca de meio mol de \omterm{def:g12:respiration-fermentation:atp}{ATP} por minuto --- um quarto de quilograma da
molécula, produzida, usada e refeita milhares de vezes a partir de um
estoque que duraria cinco segundos. Cada uma dessas moléculas de \omterm{def:g12:respiration-fermentation:atp}{ATP} é
paga por uma glicose ou um ácido graxo desmontado, átomo por átomo, e
por uma molécula de oxigênio inspirada no começo e expirada como água
no fim. O \cref{ch:g10:cell-metabolism} deu o balanço; este capítulo
abre a \omterm{def:g10:cells-common-unit:organelle}{mitocôndria} e acompanha a glicose pelas três etapas que
transformam a energia dela em \omterm{def:g12:respiration-fermentation:atp}{ATP} --- e mostra o que uma \omterm{def:g10:cells-common-unit:cell}{célula} faz
quando o oxigênio acaba.

\section{O ATP, a moeda}

\begin{definition}[ATP]\label{def:g12:respiration-fermentation:atp}
O \emph{ATP}\index{ATP} (trifosfato de adenosina) é um \omterm{def:g10:universal-dna:nucleotide}{nucleotídeo} que
carrega três grupos fosfato em fila. Retirar o último libera cerca de
\qty{30}{kJ} por mol nas condições da \omterm{def:g10:cells-common-unit:cell}{célula}, e todo processo da \omterm{def:g10:cells-common-unit:cell}{célula}
que exige energia --- a contração, o transporte, a síntese, a luz de um
vaga-lume --- é impulsionado por essa retirada. O produto, o ADP, é
recarregado em ATP pelas reações deste capítulo. Uma \omterm{def:g10:cells-common-unit:cell}{célula} guarda ATP
para alguns segundos e o renova sem parar; um ser humano em repouso
recicla cerca da própria massa corporal em ATP num dia.
\end{definition}

\begin{proposition}[Onde a energia está]\label{prop:g12:respiration-fermentation:oxidation}
A glicose guarda energia nas ligações carbono--hidrogênio dela;
liberá-la significa transferir o hidrogênio, com os elétrons dele, para
o oxigênio --- oxidando a glicose a gás carbônico e reduzindo o
oxigênio a água. Feito num passo só, como numa chama, a energia sairia
como calor. A \omterm{def:g10:cells-common-unit:cell}{célula} o faz em dezenas de passos pequenos, cada um
conduzido por uma \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzima}, e captura parte da energia em vários deles: o
hidrogênio é primeiro carregado em moléculas transportadoras --- o
\emph{NAD reduzido} --- e só no fim entregue ao oxigênio.
\end{proposition}
\admitted

\section{Três etapas}

\begin{proposition}[A glicólise]\label{prop:g12:respiration-fermentation:glycolysis}
No \omterm{def:g10:cells-common-unit:cell}{citoplasma}, sem oxigênio, uma glicose (seis carbonos) é dividida em
duas moléculas de \emph{piruvato} (três carbonos cada) por dez passos
enzimáticos: a \emph{glicólise}\index{glicólise}. O balanço: dois \omterm{def:g12:respiration-fermentation:atp}{ATP}
consumidos para começar, quatro produzidos, portanto um ganho líquido
de 2 \omterm{def:g12:respiration-fermentation:atp}{ATP}, e dois NAD reduzidos carregados de hidrogênio. O piruvato
ainda guarda a maior parte da energia da glicose.
\end{proposition}
\admitted

\begin{proposition}[A mitocôndria completa a oxidação]\label{prop:g12:respiration-fermentation:mitochondrion}
Quando há oxigênio, o piruvato entra na
\emph{mitocôndria}\index{mitocôndria}, uma \omterm{def:g10:cells-common-unit:organelle}{organela} delimitada por duas
membranas, com a interna dobrada em \emph{cristas} que envolvem o
líquido da \emph{matriz}.
\begin{itemize}
  \item Na matriz, cada piruvato é desmontado num ciclo de reações, o
        \emph{ciclo de Krebs}\index{ciclo de Krebs}: os três carbonos
        dele saem como três $\mathrm{CO_2}$, o hidrogênio dele é
        carregado em transportadores (o NAD reduzido e um segundo
        transportador), e um pouco de \omterm{def:g12:respiration-fermentation:atp}{ATP} é feito diretamente --- um
        por piruvato.
  \item Na membrana interna, os transportadores carregados entregam os
        elétrons deles a uma cadeia de \omterm{def:g11:gene-expression:protein}{proteínas}, a \emph{cadeia
        respiratória}, que os passa adiante até o oxigênio, o qual se
        combina com prótons para formar água. A energia liberada ao
        longo da cadeia bombeia prótons através da membrana, e o
        retorno deles por uma \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzima} rotatória impulsiona a síntese de
        \omterm{def:g12:respiration-fermentation:atp}{ATP} --- cerca de 26 por glicose.
\end{itemize}
Total para uma glicose: cerca de 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} (2 da \omterm{prop:g12:respiration-fermentation:glycolysis}{glicólise}, 2 do ciclo,
uns 26 da cadeia), seis $\mathrm{CO_2}$ liberados, seis
$\mathrm{O_2}$ consumidos, e o resto dos \qty{2870}{kJ} como calor.
\end{proposition}
\begin{proof}[Evidence]
\omterm{def:g10:cells-common-unit:organelle}{Mitocôndrias} isoladas numa câmara fechada com uma sonda de oxigênio não
consomem oxigênio nenhum quando recebem glicose, mas o consomem
depressa quando recebem piruvato --- a \omterm{def:g10:cells-common-unit:organelle}{mitocôndria} não consegue partir
da glicose; o \omterm{def:g10:cells-common-unit:cell}{citoplasma} precisa fazê-lo. Acrescentar um veneno da
cadeia respiratória (o cianeto) interrompe o consumo na hora, seja qual
for o \omterm{def:g11:enzymes-and-phenotype:enzyme}{substrato} presente. Acrescentar ADP acelera o consumo e o
esgotamento dele o desacelera: o uso de oxigênio está acoplado à
síntese de \omterm{def:g12:respiration-fermentation:atp}{ATP}. E as \omterm{def:g10:cells-common-unit:organelle}{mitocôndrias} do músculo de voo de um beija-flor ou
da coxa de um maratonista têm cristas várias vezes mais densas que as
de um tecido em repouso.
\end{proof}

\begin{example}[Ler um traçado de oxigênio]\label{ex:g12:respiration-fermentation:trace}
\omterm{def:g10:cells-common-unit:organelle}{Mitocôndrias} isoladas na câmara: o oxigênio fica em \qty{8}{mg/L};
glicose é acrescentada, nada; piruvato é acrescentado, o oxigênio cai a
\qty{0.5}{mg/L} por minuto; ADP é acrescentado, a queda se acentua para
\qty{1.5}{mg/L} por minuto por um tempo, depois volta a \num{0.5};
cianeto é acrescentado, a queda para. Quatro fatos num traçado só: a
\omterm{def:g10:cells-common-unit:organelle}{mitocôndria} precisa de piruvato, não de glicose; o uso de oxigênio está
acoplado à síntese de \omterm{def:g12:respiration-fermentation:atp}{ATP}; o acoplamento se dá pela cadeia que o
cianeto bloqueia; e uma \omterm{def:g10:cells-common-unit:organelle}{mitocôndria} sem trabalho a fazer fica em marcha lenta.
\end{example}

\section{Sem oxigênio: a fermentação}

\begin{proposition}[A fermentação regenera o transportador]\label{prop:g12:respiration-fermentation:fermentation}
A \omterm{prop:g12:respiration-fermentation:glycolysis}{glicólise} carrega dois transportadores NAD por glicose; com oxigênio,
a \omterm{def:g10:cells-common-unit:organelle}{mitocôndria} os descarrega. Sem oxigênio, os transportadores estariam
todos carregados em segundos e a \omterm{prop:g12:respiration-fermentation:glycolysis}{glicólise} pararia. A
\emph{fermentação}\index{fermentação} é o jeito que o \omterm{def:g10:cells-common-unit:cell}{citoplasma} tem de
descarregá-los: o próprio piruvato aceita o hidrogênio e vira
\emph{lactato} (\omterm{def:g10:cells-common-unit:cell}{células} musculares, bactérias do leite) ou, depois de
perder um $\mathrm{CO_2}$, \emph{etanol} (levedura). Nenhum \omterm{def:g12:respiration-fermentation:atp}{ATP} a mais
é produzido; o rendimento são os 2 \omterm{def:g12:respiration-fermentation:atp}{ATP} da \omterm{prop:g12:respiration-fermentation:glycolysis}{glicólise}, umas quinze vezes
menos que o da respiração, e o produto --- lactato ou etanol --- ainda
guarda a maior parte da energia da glicose.
\end{proposition}
\admitted

\begin{example}[A escolha do músculo]\label{ex:g12:respiration-fermentation:muscle}
A fibra muscular de um velocista precisa de \omterm{def:g12:respiration-fermentation:atp}{ATP} mais rápido do que as
\omterm{def:g10:cells-common-unit:organelle}{mitocôndrias} dela e o suprimento de oxigênio conseguem produzir; a
\omterm{prop:g12:respiration-fermentation:glycolysis}{glicólise} com \omterm{def:g10:cell-metabolism:fermentation}{fermentação} láctica entrega \omterm{def:g12:respiration-fermentation:atp}{ATP} três vezes mais rápido,
ao custo de quinze vezes mais glicose, e o lactato se acumula até a dor
interromper o esforço. As fibras de uma maratonista rodam com
respiração, devagar e quase indefinidamente, queimando gordura além de
glicose; as \omterm{def:g10:cells-common-unit:organelle}{mitocôndrias} dela são maiores e mais numerosas, os
capilares dela mais densos, e o lactato dela fica baixo. Os tipos de
fibra do \cref{ch:g10:muscles-and-joints} são essas duas estratégias
embutidas em \omterm{def:g10:cells-common-unit:cell}{células}.
\end{example}

\begin{method}[Fechar um balanço energético]\label{met:g12:respiration-fermentation:budget}
\begin{enumerate}
  \item Converta a potência necessária em \omterm{def:g12:respiration-fermentation:atp}{ATP}: a cerca de \qty{30}{kJ}
        utilizáveis por mol de \omterm{def:g12:respiration-fermentation:atp}{ATP}, \qty{1}{kW} de potência metabólica
        são \qty{2}{mol} de \omterm{def:g12:respiration-fermentation:atp}{ATP} por minuto.
  \item Converta o \omterm{def:g12:respiration-fermentation:atp}{ATP} em combustível: 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} por glicose na
        respiração, 2 na \omterm{def:g10:cell-metabolism:fermentation}{fermentação}.
  \item Converta o combustível em oxigênio: 6 $\mathrm{O_2}$ por
        glicose respirada, \qty{24}{L} por mol de gás.
  \item Compare o oxigênio necessário com o que o sangue entrega
        (\cref{ch:g10:heart-lungs-effort}): a diferença é coberta pela
        \omterm{def:g10:cell-metabolism:fermentation}{fermentação}, com o lactato dela.
\end{enumerate}
\end{method}

\begin{remark}[O espelho da fotossíntese]\label{rem:g12:respiration-fermentation:mirror}
A \omterm{def:g10:cell-metabolism:autotrophy}{fotossíntese} carrega elétrons da água em transportadores com luz e os
usa para reduzir gás carbônico a açúcar, liberando oxigênio; a
respiração descarrega elétrons do açúcar em transportadores e os passa
ao oxigênio, produzindo água e liberando gás carbônico, e usa a energia
para fazer \omterm{def:g12:respiration-fermentation:atp}{ATP}. As duas são conduzidas por \omterm{def:g10:cells-common-unit:organelle}{organelas} que um dia foram
as duas bactérias livres (\cref{ch:g12:diversification-of-life}), com o
\omterm{def:g12:photosynthesis:chloroplast}{cloroplasto} fornecendo o que a \omterm{def:g10:cells-common-unit:organelle}{mitocôndria} consome; juntas, elas
convertem a luz do sol na moeda de toda \omterm{def:g10:cells-common-unit:cell}{célula}, e o oxigênio da
atmosfera é o saldo das contas delas.
\end{remark}

\section{Problema: meio mol de ATP por minuto}

\begin{problem}[{Problema de fim de semana --- a energia de uma
corredora acompanhada do watt à molécula: ATP contado, glicose e
oxigênio contabilizados, a parcela das mitocôndrias medida, e o sprint
que a cadeia não consegue pagar}]
\label{pb:g12:respiration-fermentation:1}
Uma corredora sustenta uma potência metabólica de \qty{1000}{W}, da
qual cerca de um terço é capturado como \omterm{def:g12:respiration-fermentation:atp}{ATP}, e o resto sai como calor.
Tome \qty{30}{kJ} de energia utilizável por mol de \omterm{def:g12:respiration-fermentation:atp}{ATP}, 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} por
glicose na respiração e 2 na \omterm{def:g10:cell-metabolism:fermentation}{fermentação}, \qty{2870}{kJ} por mol de
glicose, \qty{24}{L} por mol de gás, e glicose \qty{180}{g/mol}.

\medskip
\noindent\textbf{Parte I --- O \omterm{def:g12:respiration-fermentation:atp}{ATP}.}
\begin{enumerate}
  \item Quantos mols de \omterm{def:g12:respiration-fermentation:atp}{ATP} a corredora usa por minuto? (Só o terço
        capturado como \omterm{def:g12:respiration-fermentation:atp}{ATP} conta.)
  \item Que massa de \omterm{def:g12:respiration-fermentation:atp}{ATP} é essa, a \qty{507}{g/mol}? Compare com os
        poucos gramas que os músculos guardam a cada momento.
  \item Quantas vezes por minuto cada molécula de \omterm{def:g12:respiration-fermentation:atp}{ATP} é recarregada, se
        o estoque do corpo é de \qty{50}{g}?
  \item Se esse \omterm{def:g12:respiration-fermentation:atp}{ATP} fosse produzido só pela respiração, quantos mols de
        glicose por minuto seriam consumidos?
  \item Quantos litros de oxigênio por minuto isso exige? Compare com
        um $\dot V\!\mathrm{O_2}$max de \qty{4}{L/min}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A parcela das \omterm{def:g10:cells-common-unit:organelle}{mitocôndrias}.}
\begin{enumerate}[resume]
  \item Dos 30 \omterm{def:g12:respiration-fermentation:atp}{ATP} por glicose, quantos são produzidos no \omterm{def:g10:cells-common-unit:cell}{citoplasma} e
        quantos na \omterm{def:g10:cells-common-unit:organelle}{mitocôndria}? Que fração do \omterm{def:g12:respiration-fermentation:atp}{ATP} da corredora é
        mitocondrial?
  \item Quantos mols de $\mathrm{CO_2}$ ela expira por minuto, e de
        qual etapa da respiração eles vêm?
  \item Quanto dos \qty{2870}{kJ} de uma glicose termina em \omterm{def:g12:respiration-fermentation:atp}{ATP}? Para
        onde vai o resto, e o que a corredora faz a respeito?
  \item As \omterm{def:g10:cells-common-unit:organelle}{mitocôndrias} musculares dela têm uma área total de membrana
        interna de uns \qty{1000}{m^2}. Por que a cadeia precisa de
        tanta superfície?
  \item O \omterm{def:g10:sport-and-health:training}{treinamento} dobra as \omterm{def:g10:cells-common-unit:organelle}{mitocôndrias} das fibras dela. Quais das
        grandezas acima isso muda, e quais não muda?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O sprint.}
No sprint final, os músculos dela precisam de \qty{800}{W} de potência
como \omterm{def:g12:respiration-fermentation:atp}{ATP} por 30 segundos, enquanto a respiração, limitada pelo oxigênio
que o sangue entrega, consegue fornecer no máximo \qty{400}{W} como \omterm{def:g12:respiration-fermentation:atp}{ATP}.
\begin{enumerate}[resume]
  \item Quantos mols de \omterm{def:g12:respiration-fermentation:atp}{ATP} o sprint exige no total?
  \item Quanto disso a respiração consegue fornecer em 30 segundos, e
        quanto precisa vir da \omterm{def:g10:cell-metabolism:fermentation}{fermentação}?
  \item Quantos mols de glicose a parte fermentada consome, e quantos
        mols de lactato ela deixa?
  \item Compare a glicose usada por \omterm{def:g12:respiration-fermentation:atp}{ATP} nas duas rotas durante o
        sprint. Por que o músculo aceita o desperdício?
  \item Depois da corrida, o lactato é oxidado ou convertido de volta
        em glicose. Explique por que o consumo de oxigênio dela fica
        alto por vários minutos depois que ela para.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O outro combustível.}
A gordura fornece cerca de \qty{38}{kJ} por grama e, por grama, precisa
de uns 20\% mais oxigênio que a glicose para a mesma energia.
\begin{enumerate}[resume]
  \item Se metade dos \qty{1000}{W} dela viesse da gordura, que massa
        de gordura ela queimaria por hora?
  \item Por que a gordura, embora mais rica em energia por grama, dá
        menos potência por litro de oxigênio?
  \item Os ácidos graxos entram na respiração no \omterm{prop:g12:respiration-fermentation:mitochondrion}{ciclo de Krebs},
        pulando a \omterm{prop:g12:respiration-fermentation:glycolysis}{glicólise}. Qual das três etapas, então, a gordura não
        consegue usar, e o que decorre disso para um sprint?
  \item Por que o cérebro, que roda só com glicose, continua
        funcionando durante a corrida, embora os músculos estejam
        levando a maior parte da glicose do sangue?
  \item Enuncie o resultado: os mols de \omterm{def:g12:respiration-fermentation:atp}{ATP} que a corredora usa por
        minuto, os litros de oxigênio que os pagam, e a etapa da
        respiração que produz a maior parte deles.
\end{enumerate}
\end{problem}
