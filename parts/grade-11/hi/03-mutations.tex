\chapter{उत्परिवर्तन और आनुवंशिक विविधता}\label{ch:g11:mutations}

किसी ऐसे परिवार के दो साधारण माता-पिता के यहाँ, जिसमें इस स्थिति का कोई
इतिहास नहीं, ऐसा बच्चा जन्म लेता है जिसमें वर्णक बिलकुल नहीं है --- सफ़ेद
बाल, फीकी त्वचा, और ऐसी आँखें जो तेज़ रोशनी सह नहीं पातीं। जिस जीवाणु
कॉलोनी ने कभी कोई प्रतिजैविक देखा ही नहीं, उसकी दस करोड़ \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में
कुछ ऐसी होती हैं जो उसका प्रतिरोध पहले से ही करती हैं। किसी माली को गुलाब
की झाड़ी की एक ही शाखा पर अलग रंग के फूल मिलते हैं। हर हाल में किसी \omterm{def:g10:universal-dna:information}{DNA}
अनुक्रम में, एक बार, किसी एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} में, बदलाव आया है; और हर हाल में वह
बदलाव उसके बाद ईमानदारी से नक़ल होता रहता है। यह अध्याय उन्हीं बदलावों के
बारे में है: वे हैं क्या, आते कहाँ से हैं, \omterm{def:g10:cells-common-unit:cell}{कोशिका} उन्हें सीमित कैसे रखती
है, और जीवन इस पर क्यों टिका है कि वे कभी पूरी तरह रुकें नहीं।

\section{उत्परिवर्तन है क्या}

\begin{definition}[उत्परिवर्तन]\label{def:g11:mutations:mutation}
\emph{उत्परिवर्तन}\index{उत्परिवर्तन} किसी \omterm{def:g10:universal-dna:information}{DNA} अणु के \omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइडों} के
अनुक्रम में आया वह बदलाव है जो उस \omterm{def:g10:cells-common-unit:cell}{कोशिका} के वंशजों तक पहुँचता है जिसमें वह
हुआ हो। सबसे आम \emph{बिंदु उत्परिवर्तन} हैं, जो एक या कुछ ही
\omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइडों} पर असर डालते हैं:
\begin{itemize}
  \item \emph{प्रतिस्थापन}: एक \omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइड} की जगह दूसरा;
  \item \emph{निवेशन}: एक या अधिक \omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइड} जोड़े गए;
  \item \emph{विलोपन}: एक या अधिक \omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइड} हटाए गए।
\end{itemize}
बड़े बदलाव --- किसी खंड का दोहराया जाना, उलट जाना, हट जाना, या किसी पूरे
\omterm{def:g11:cell-cycle-mitosis:chromosome}{गुणसूत्र} का मिल जाना या खो जाना --- भी उत्परिवर्तन ही हैं।
\end{definition}

\begin{proposition}[उत्परिवर्तन होते कहाँ हैं, और किसके]\label{prop:g11:mutations:where}
किसी शरीर \omterm{def:g10:cells-common-unit:cell}{कोशिका} में हुआ \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} (\emph{कायिक \omterm{def:g11:mutations:mutation}{उत्परिवर्तन}}) केवल उसी
\omterm{def:g10:cells-common-unit:cell}{कोशिका} के वंशजों तक पहुँचता है: यानी ज़्यादा से ज़्यादा किसी एक व्यक्ति के
ऊतक का एक टुकड़ा। किसी जनन \omterm{def:g10:cells-common-unit:cell}{कोशिका} में हुआ \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} --- यानी किसी युग्मक
या उसके किसी पूर्ववर्ती में --- उससे बनने वाली संतान तक और उसकी हर \omterm{def:g10:cells-common-unit:cell}{कोशिका}
तक पहुँचता है: और \emph{जनन-वंश \omterm{def:g11:mutations:mutation}{उत्परिवर्तन}} ही अकेली ऐसी क़िस्म है जो
\omterm{def:g10:biodiversity-scales:species}{जाति} की आनुवंशिकता में घुसती है। गुलाब की झाड़ी की वह अजीब शाखा कायिक है;
और वर्णकहीन बच्चे की स्थिति किसी पूर्वज में हुए जनन-वंश \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} से
आई, जो पीढ़ियों तक चुपचाप ढोया गया।
\end{proposition}
\admitted

\begin{example}[उत्परिवर्तन की दर]\label{ex:g11:mutations:frequency}
प्रतिकृतिकरण $10^9$ \omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइडों} में लगभग एक ग़लती छोड़ता है
(\cref{ch:g11:dna-replication})। इसलिए $\num{3000}$ क्षारक-जोड़ों का कोई
\omterm{def:g10:universal-dna:gene}{जीन} हर $\num{300000}$ प्रतिकृतियों में लगभग एक बार ग़लत उतरता है;
और किसी मनुष्य में, जिसका जीनोम जीवन भर में क़रीब $10^{16}$ बार नक़ल होता
है, हर \omterm{def:g10:universal-dna:gene}{जीन} में बहुत-सी \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} जमा हो जाते हैं। हर बच्चा
क़रीब 60 ऐसे नए \omterm{def:g11:mutations:mutation}{उत्परिवर्तनों} के साथ जन्म लेता है जो किसी भी जनक के पास
नहीं थे --- और लगभग सारे जीनोम के उन 98\% हिस्से में जो \omterm{def:g10:universal-dna:gene}{जीन} है ही नहीं।
\end{example}

\section{कारण: स्वतःस्फूर्त और प्रेरित}

\begin{proposition}[स्वतःस्फूर्त उत्परिवर्तन]\label{prop:g11:mutations:spontaneous}
\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} हर \omterm{def:g10:cells-common-unit:cell}{कोशिका} में, बिना किसी बाहरी कारक के, उठते रहते हैं:
पॉलिमरेज़ की जाँच से बच निकलने वाली विरली बेमेल जोड़ियों से, और ख़ुद \omterm{def:g10:universal-dna:information}{DNA}
की रासायनिक अस्थिरता से, जिसके \omterm{def:g10:universal-dna:nucleotide}{क्षारक} साधारण \omterm{def:g10:cell-metabolism:metabolism}{उपापचय} के जल तथा
क्रियाशील अणुओं से हर \omterm{def:g10:cells-common-unit:cell}{कोशिका} में हर दिन हज़ारों बार क्षतिग्रस्त होते हैं।
ऐसे \emph{स्वतःस्फूर्त \omterm{def:g11:mutations:mutation}{उत्परिवर्तन}} \emph{यादृच्छिक} होते हैं: वे किसी
भी स्थान पर, ऐसी दर से होते हैं जो इस बात पर निर्भर नहीं करती कि वह
बदलाव \omterm{def:g10:cells-common-unit:cell}{कोशिका} के काम का होगा या नहीं।
\end{proposition}
\begin{proof}[साक्ष्य]
लेडरबर्ग (1952) ने ऐसे जीवाणु, जो कभी किसी प्रतिजैविक के सामने आए ही नहीं
थे, एक प्लेट पर फैलाए, उन्हें कॉलोनियों में बढ़ने दिया, फिर प्लेट पर
मख़मल का एक गद्दा दबाकर उसकी छाप --- वही कॉलोनियाँ, उन्हीं जगहों पर ---
प्रतिजैविक वाली कई प्लेटों पर छाप दी। प्रतिरोधी कॉलोनियाँ, जो थोड़ी ही
थीं, हर प्रतिजैविक प्लेट पर \emph{उन्हीं जगहों} पर उभरीं। चूँकि वे मूल
प्लेट की उन्हीं कॉलोनियों से आई थीं, जिन्होंने वह प्रतिजैविक कभी देखा ही
नहीं था, इसलिए प्रतिरोध वाला \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} उस संपर्क से पहले ही मौजूद था:
प्रतिजैविक ने उसे उजागर किया, पैदा नहीं किया। वे \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} कॉलोनियों की
वृद्धि के दौरान यादृच्छिक रूप से हुए थे, बिना इस बात की परवाह किए कि आगे
चलकर वे किस काम आएँगे।
\end{proof}

\begin{definition}[उत्परिवर्तजन]\label{def:g11:mutations:mutagen}
\emph{उत्परिवर्तजन}\index{उत्परिवर्तजन} वह भौतिक या रासायनिक कारक है जो
\omterm{def:g10:universal-dna:information}{DNA} को क्षति पहुँचाकर \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} की दर बढ़ा देता है। पराबैंगनी प्रकाश
किसी लड़ी पर पड़ोसी थाइमीनों को आपस में जोड़ देता है; एक्स-किरणें और
रेडियोसक्रियता लड़ियाँ तोड़ देती हैं; और तंबाकू के धुएँ का बेंज़ोपाइरीन या
फफूँदी लगे अनाज का एफ़्लाटॉक्सिन जैसे रसायन \omterm{def:g10:universal-dna:nucleotide}{क्षारकों} से चिपककर कुंडली को
मरोड़ देते हैं। जो \omterm{def:g10:cells-common-unit:cell}{कोशिका} किसी उत्परिवर्तजन के सामने आती है वह अपने
स्वतःस्फूर्त \omterm{def:g11:mutations:mutation}{उत्परिवर्तनों} के ऊपर \emph{प्रेरित \omterm{def:g11:mutations:mutation}{उत्परिवर्तन}} भी ढोती है
--- और वे भी यादृच्छिक स्थानों पर ही, बस संख्या में अधिक।
\end{definition}

\begin{example}[खमीर पर पराबैंगनी]\label{ex:g11:mutations:uv}
प्लेटों पर फैलाई गई खमीर \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} बढ़ते समय तक पराबैंगनी प्रकाश के सामने
रखी जाती हैं: \qty{10}{s} बाद आधी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} कॉलोनी बना ही नहीं पातीं,
\qty{20}{s} बाद चौथाई, \qty{30}{s} बाद आठवाँ हिस्सा --- यानी हर ख़ुराक
बचे हुओं को आधा कर देती है। और बचे हुओं में उन कॉलोनियों का अनुपात जिनका
कोई लक्षण दिखने लायक़ बदल गया हो (रंग, आकार, किसी ख़ास शर्करा पर उगने में
असमर्थता) दस लाख में एक से बढ़कर हज़ार में एक हो जाता है। वह लैंप \omterm{def:g10:universal-dna:information}{DNA} को
मरम्मत से परे क्षति पहुँचाकर मारता है, और उससे ज़रा कम क्षति पहुँचाकर
उत्परिवर्तित करता है।
\end{example}

\section{कोशिका पलटकर लड़ती है: मरम्मत}

\begin{proposition}[DNA की मरम्मत]\label{prop:g11:mutations:repair}
कोई \omterm{def:g10:cells-common-unit:cell}{कोशिका} दिन भर में \omterm{def:g10:universal-dna:information}{DNA} की दसियों हज़ार क्षतियाँ झेलती है और उनमें से
लगभग कोई भी नहीं रखती: समर्पित \omterm{def:g10:cell-metabolism:metabolism}{एंजाइम} तंत्र दोहरी कुंडली पर गश्त लगाते
हैं, किसी क्षतिग्रस्त या बेमेल \omterm{def:g10:universal-dna:nucleotide}{क्षारक} को पहचानते हैं, एक लड़ी का ख़राब
खंड काटकर निकाल देते हैं और दूसरी लड़ी को साँचा बनाकर उसे दोबारा बना देते
हैं --- यानी \cref{ch:g10:universal-dna} वाली \omterm{prop:g10:universal-dna:helix}{पूरकता} एक बार फिर। कोई
क्षति \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} तभी बनती है जब वह अगले प्रतिकृतिकरण के उसे नक़ल कर लेने
से पहले मरम्मत से बच निकले। मरम्मत ही वह वजह है कि \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} की दर
$10^{-9}$ है, $10^{-5}$ नहीं।
\end{proposition}
\begin{proof}[साक्ष्य]
जिन लोगों को ज़ीरोडर्मा पिगमेंटोसम नाम की वंशागत स्थिति होती है, उनमें
पराबैंगनी क्षति काटकर निकालने वाले एंजाइमों में से एक नहीं होता। उनकी
त्वचा, साधारण दिन के उजाले के सामने आकर, सामान्य त्वचा से हज़ार गुना अधिक
बार और बचपन से ही कैंसर पैदा करती है; और धूप से बचाकर रखी जाए तो वह
सामान्य रहती है। वही पराबैंगनी, वही क्षतियाँ --- पर मरम्मत के बिना
क्षतियाँ बनी रह जाती हैं, और \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} उन \omterm{def:g10:universal-dna:gene}{जीनों} में जमा होते जाते हैं जो
\omterm{def:g10:cells-common-unit:cell}{कोशिका} विभाजन को नियंत्रित करते हैं (\cref{ch:g11:cancer})। मरम्मत
एंजाइमों से रहित खमीर और जीवाणु प्रभेद प्रयोगशाला में वैसी ही
अति-संवेदनशीलता दिखाते हैं।
\end{proof}

\section{किसी उत्परिवर्तन के नतीजे}

\begin{proposition}[अनुक्रम से लक्षणप्ररूप तक]\label{prop:g11:mutations:consequences}
कोई \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} किसी \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} को नए \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} में बदल देता है। उसका असर इस
पर निर्भर है कि वह कहाँ पड़ता है और उस \omterm{def:g10:universal-dna:gene}{जीन} के कूट वाले \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} के साथ क्या
करता है (\cref{ch:g11:gene-expression} इसके नियम देता है):
\begin{itemize}
  \item किसी \omterm{def:g10:universal-dna:gene}{जीन} के बाहर, या ऐसे स्थान पर जो \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} को बदलता नहीं:
        \emph{कोई असर नहीं} --- और यही बहुत बड़ा बहुमत है;
  \item किसी \omterm{def:g10:universal-dna:gene}{जीन} के भीतर, एक \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्ल} बदलते हुए: ऐसा \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} जो ज़रा
        अलग ढंग से काम करता है, या बिलकुल नहीं करता, या कभी-कभार बेहतर;
  \item किसी \omterm{def:g10:universal-dna:gene}{जीन} के भीतर, संदेश को खिसकाते या काटते हुए: आमतौर पर ऐसा
        \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} जो काम ही नहीं करता।
\end{itemize}
बदला हुआ \omterm{def:g10:chemistry-of-life:families}{प्रोटीन} हानिकारक है, तटस्थ है या उपयोगी, यह पर्यावरण पर निर्भर
करता है: हीमोग्लोबिन के \omterm{def:g10:universal-dna:gene}{जीन} का वही \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} बिना मलेरिया वाले इलाक़े में
बोझ है और जहाँ मलेरिया हो वहाँ बचाव।
\end{proposition}
\admitted

\begin{example}[वर्णक खोया, वर्णक बचा]\label{ex:g11:mutations:pigment}
मेलानिन वर्णक कोई \omterm{def:g10:cell-metabolism:metabolism}{एंजाइम} बनाता है। उसके \omterm{def:g10:universal-dna:gene}{जीन} के दर्जनों अलग-अलग
\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} ज्ञात हैं, और उनमें से हर एक ऐसा \omterm{def:g10:cell-metabolism:metabolism}{एंजाइम} देता है जो काम नहीं
करता तथा, ऐसे दो \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोते व्यक्ति में, अध्याय के आरंभ वाली वर्णकहीन
स्थिति। उनमें से हर एक यादृच्छिक रूप से, एक बार, किसी जनन \omterm{def:g10:cells-common-unit:cell}{कोशिका} में उठा
था; और वह दिखने वाली स्थिति तभी उभरती है जब दो वाहक अपने चुप \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} एक ही
बच्चे तक पहुँचा दें। वही \omterm{def:g10:universal-dna:gene}{जीन}, किसी और \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} में, ज़रा कम सक्रिय \omterm{def:g10:cell-metabolism:metabolism}{एंजाइम}
बनाता है --- और उसके साथ हल्की त्वचा, जो ऊँचे अक्षांशों पर हज़ारों साल
रहने वाली आबादियों में आम है, जहाँ ज़रा कम वर्णक त्वचा को कमज़ोर धूप से
अधिक विटामिन D बनाने देता है।
\end{example}

\begin{proposition}[उत्परिवर्तन, विविधता का स्रोत]\label{prop:g11:mutations:source}
हर \omterm{def:g10:universal-dna:gene}{जीन} का हर \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} किसी पहले वाले \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} के \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} से ही शुरू हुआ
था। \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} अकेली ऐसी प्रक्रिया है जो नई \omterm{def:g10:universal-dna:information}{आनुवंशिक जानकारी} बनाती है;
और आख़िरी वर्ष वाला लैंगिक फेंटना केवल उसी को दोबारा मिलाता है जो
\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} ने बनाया है। विरला, यादृच्छिक और व्यक्ति के लिए ज़्यादातर
तटस्थ या हानिकारक होते हुए भी \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} ही पीढ़ियों के साथ वह विविधता
जुटाता है जिस पर \cref{ch:g10:biodiversity-scales} वाला जैव विकास टिका
है।
\end{proposition}
\admitted

\begin{method}[किसी उत्परिवर्तन का विश्लेषण]\label{met:g11:mutations:analyse}
\begin{enumerate}
  \item उत्परिवर्ती अनुक्रम की मूल से अक्षर-दर-अक्षर तुलना कीजिए और
        बदलाव का नाम बताइए (प्रतिस्थापन, निवेशन, विलोपन; और कितने
        \omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइड})।
  \item उसकी जगह पहचानिए: किसी \omterm{def:g10:universal-dna:gene}{जीन} में या उसके बाहर; और यदि \omterm{def:g10:universal-dna:gene}{जीन} में हो,
        तो यह भी कि संदेश का पढ़ना खिसका है या नहीं।
  \item जिस \omterm{def:g10:cells-common-unit:cell}{कोशिका} में वह हुआ उसे पहचानिए: कायिक (एक व्यक्ति, एक ऊतक) या
        जनन-वंश (आगे पहुँचने वाला)।
  \item यदि दर असामान्य हो तो कारण ढूँढ़िए (किसी \omterm{def:g11:mutations:mutagen}{उत्परिवर्तजन} से संपर्क,
        ख़राब मरम्मत); वरना उसे संयोग मानिए।
  \item नतीजे को \omterm{def:g10:chemistry-of-life:families}{प्रोटीन}, \omterm{def:g10:cells-common-unit:cell}{कोशिका}, जीव और आबादी के स्तर पर आँकिए --- इसी
        क्रम में, और किसी बताए हुए पर्यावरण में।
\end{enumerate}
\end{method}

\begin{remark}[यादृच्छिकता, ठीक-ठीक]\label{rem:g11:mutations:random}
``यादृच्छिक'' का मतलब यह नहीं कि सारे स्थान बराबर उत्परिवर्तित होते हैं
--- कुछ अनुक्रम अधिक नाज़ुक होते हैं --- और न यह कि \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} बेवजह होते
हैं। उसका मतलब है कि किसी \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} का होना इस बात से स्वतंत्र है कि वह
\omterm{def:g10:cells-common-unit:cell}{कोशिका} के काम आएगा या नहीं: प्रतिकृति प्लेटों वाले प्रतिजैविक-प्रतिरोधी
जीवाणुओं को प्रतिजैविक ने प्रतिरोधी बनाया ही नहीं था, और कोई जीव किसी काम
की दिशा में उत्परिवर्तित होने का ``फ़ैसला'' कर ही नहीं सकता। दिशा वरण
देता है, बाद में; \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} केवल जुटाता है।
\end{remark}

\num{3000} क्षारक-जोड़ों वाला कोई \omterm{def:g10:universal-dna:gene}{जीन} $10^9$ \omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइडों} में एक
ग़लती के साथ नक़ल होता है। $10^{12}$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के किसी ऊतक में, जो 40
विभाजनों से एक ही \omterm{def:g10:cells-common-unit:cell}{कोशिका} से उतरा है, मोटे तौर पर कितनी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} उस \omterm{def:g10:universal-dna:gene}{जीन} में
कोई नया \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} ढोती हैं? (केवल आख़िरी विभाजन में हुई ग़लतियाँ
गिनिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mutations:10}
गुलाब की किसी झाड़ी पर एक शाखा अलग रंग के फूल देती है। उस शाखा की क़लमों
से नए रंग वाली झाड़ियाँ बनती हैं; पर उसके फूलों के बीजों से साधारण गुलाब।
दोनों नतीजे समझाइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:mutations:11}
ज़ीरोडर्मा पिगमेंटोसम वाले व्यक्ति में त्वचा के कैंसर क्यों होते हैं, पर
मसलन आँत के कैंसर क्यों नहीं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mutations:12}
धूप-रोधी क्रीम पराबैंगनी सोख लेती है। बचाव वाले चित्र की मदद से समझाइए कि
पराबैंगनी का दसवाँ हिस्सा गुज़रने देने वाली क्रीम धूप के एक घंटे में त्वचा
की प्रति \omterm{def:g10:cells-common-unit:cell}{कोशिका} क्षतियों की संख्या का क्या करती है, और ``धूप से जलन नहीं''
का मतलब ``कोई \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} नहीं'' क्यों नहीं है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mutations:13}
हीमोग्लोबिन \omterm{def:g10:universal-dna:gene}{जीन} का कोई \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} एक प्रति ढोने वालों को मलेरिया से बचाता है
पर दो प्रतियाँ ढोने वालों में गंभीर रोग पैदा करता है। चर्चा कीजिए कि यह
\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} ``हानिकारक'' है या ``उपयोगी'', और यह तय क्या करता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mutations:14}
बिना प्रतिजैविक वाले शोरबे के जीवाणुओं में 30 पीढ़ियों बाद दस लाख में एक
प्रतिरोधी \omterm{def:g10:cells-common-unit:cell}{कोशिका} मिलती है। यदि फिर प्रतिजैविक डाला जाए, तो दिन भर में उस
संवर्धन का क्या होता है? समझाइए कि डॉक्टर प्रतिजैविक का पूरा कोर्स लेने पर
क्यों ज़ोर देते हैं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:mutations:15}
``यदि \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} लगभग हमेशा हानिकारक हैं, तो जैव विकास उन पर खड़ा नहीं
हो सकता।'' यादृच्छिकता, तटस्थ \omterm{def:g11:mutations:mutation}{उत्परिवर्तनों} के अनुपात, पर्यावरण और
समय-पैमाने की धारणाओं से एक अनुच्छेद में चर्चा कीजिए।
\end{exercise}

\section{समस्या: मख़मल की छाप}

\begin{problem}[{सप्ताहांत समस्या --- लेडरबर्ग का प्रतिकृति-रोपण दोबारा
खड़ा किया गया: कॉलोनियाँ गिनी गईं, प्रतिरोध का नक़्शा बना, उत्परिवर्तनों
का उद्गम तय हुआ, और उनकी दर नापी गई}]
\label{pb:g11:mutations:1}
एक विद्यार्थी ऐसे संवर्धन से, जिसने प्रतिजैविक स्ट्रेप्टोमाइसिन कभी देखा
ही नहीं, लगभग 200 जीवाणु बिना प्रतिजैविक वाली एक प्लेट पर फैलाता है। रात
भर में 200 कॉलोनियाँ उग आती हैं, हर एक किसी एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} से और हर एक में
लगभग $10^7$ \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}। प्लेट पर मख़मल का गद्दा दबाकर उसे
स्ट्रेप्टोमाइसिन वाली तीन प्लेटों पर छापा जाता है। अगले दिन प्लेट A दो
प्रतिरोधी कॉलोनियाँ दिखाती है, प्लेट B दो, प्लेट C दो --- और तीनों पर
उन्हीं दो जगहों पर।

\medskip
\noindent\textbf{भाग I --- प्रयोग।}
\begin{enumerate}
  \item मूल प्लेट की कॉलोनियों का इकहरी \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} से आना क्यों ज़रूरी है,
        और फैलाए गए जीवाणुओं की संख्या छोटी क्यों रखी जाती है?
  \item समझाइए कि मख़मल क्या पहुँचाता है, और किसी कॉलोनी की जगह क्यों
        बनी रहती है।
  \item 200 में से दो जगहें हर प्रतिकृति पर प्रतिरोधी कॉलोनियाँ देती हैं।
        तीनों प्रतिकृतियों का आपस में मेल खाना क्या दिखाता है?
  \item इसके बजाय मान लीजिए कि प्रतिजैविक ने संपर्क में आते ही कुछ थोड़ी
        \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} में प्रतिरोध पैदा किया हो। तब तीनों प्रतिकृतियाँ किस
        ढर्रे में दिखतीं, इसका वर्णन कीजिए।
  \item इन दोनों परिकल्पनाओं में से --- संपर्क से पहले \omterm{def:g11:mutations:mutation}{उत्परिवर्तन}, या
        संपर्क से प्रेरण --- देखा गया ढर्रा किसका समर्थन करता है? तर्क एक
        वाक्य में बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- मूल प्लेट पर वापस।}
\begin{enumerate}[resume]
  \item विद्यार्थी \emph{मूल} प्लेट की (जो कभी प्रतिजैविक से छुई ही नहीं)
        उन दो प्रतिरोधी जगहों से और दो दूसरी जगहों से \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} लेता है,
        और हर नमूने को स्ट्रेप्टोमाइसिन पर परखता है। नतीजों का अनुमान
        लगाइए।
  \item प्रतिरोधी जगहों वाली दोनों कॉलोनियों में लगभग $10^7$
        \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} हैं। क्या उनकी सारी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} प्रतिरोधी हैं, या कुछ ही?
        तय करने के लिए आपको क्या जानना होगा?
  \item कोई कॉलोनी एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} से क़रीब 23 विभाजनों में उगती है। यदि
        \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} 20वें विभाजन पर हुआ हो, तो कॉलोनी की \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} का
        कितना अंश उसे ढोता है? और यदि तीसरे विभाजन पर?
  \item समझाइए कि पूरी कॉलोनी को परखने से ऐसा \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} भी क्यों उजागर
        हो जाता है जो उसकी \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के केवल एक अंश में हुआ हो।
  \item यह प्रयोग तभी क्यों चलता है जब प्रतिकृति प्लेटों की कॉलोनियाँ
        \emph{ठीक} वहीं उगें जहाँ मूल कॉलोनियाँ थीं?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- दर नापना।}
\begin{enumerate}[resume]
  \item विद्यार्थी यही प्रयोग 200-200 कॉलोनियों वाली 50 मूल प्लेटों के
        साथ दोहराता है: यानी 10\,000 कॉलोनियाँ, जिनमें से 30 प्रतिरोधी
        प्रतिकृतियाँ देती हैं। कॉलोनियों का कितना अंश कम से कम एक
        प्रतिरोधी \omterm{def:g10:cells-common-unit:cell}{कोशिका} ढोता है?
  \item हर कॉलोनी एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} से लगभग $10^7$ \omterm{def:g10:cells-common-unit:cell}{कोशिका} विभाजनों में उठी।
        आँकिए कि किसी दिए हुए विभाजन से प्रतिरोध का \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} होने की
        प्रायिकता कितनी है।
  \item स्ट्रेप्टोमाइसिन का प्रतिरोध किसी एक \omterm{def:g10:universal-dna:gene}{जीन} के एक ख़ास स्थान पर
        बदलाव माँगता है। अपने अनुमान की तुलना प्रति \omterm{def:g10:universal-dna:nucleotide}{न्यूक्लियोटाइड}
        $10^{-9}$ की प्रतिकृतिकरण ग़लती-दर से कीजिए, और टिप्पणी कीजिए।
  \item बिना प्रतिजैविक वाले शोरबे में अकेली एक \omterm{def:g10:cells-common-unit:cell}{कोशिका} से $10^9$
        \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} का संवर्धन उगाया जाता है। अपनी दर की मदद से आँकिए कि
        प्रतिजैविक अंत में डाले जाने पर उसमें कितनी प्रतिरोधी \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ}
        होने की संभावना है।
  \item यदि प्रतिरोध का \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} उस संवर्धन की वृद्धि में जल्दी हुआ
        हो, तो प्रतिरोधी \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} की संख्या कैसे बदलती है? बार-बार किए गए
        संवर्धन बहुत अलग-अलग गिनतियाँ क्यों देते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- इसका मतलब क्या है।}
\begin{enumerate}[resume]
  \item ``\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} यादृच्छिक हैं'' और ``\omterm{def:g11:mutations:mutation}{उत्परिवर्तन} विरले हैं'' में
        अंतर समझाइए, और यह भी कि प्रयोग इनमें से किसे स्थापित करता है।
  \item प्रतिजैविक ने वह प्रतिरोध बनाया नहीं। तो उसने उस आबादी के साथ
        क्या किया?
  \item इस प्रयोग को \cref{prop:g11:mutations:source} की भाषा में
        उतारिए: विविधता किसने जुटाई, और छाँटा किसने?
  \item कोई विद्यार्थी आपत्ति करता है कि मूल प्लेट की वे दोनों प्रतिरोधी
        कॉलोनियाँ ``बिलकुल बाक़ियों जैसी ही दिख रही थीं''। समझाइए कि कोई
        \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} पर्यावरण बदलने तक अदृश्य क्यों रह सकता है।
  \item परिणाम लिखिए: मख़मल की तीन प्रतिकृतियों पर एक जैसी जगहों ने
        \omterm{def:g11:mutations:mutation}{उत्परिवर्तनों} के उद्गम के बारे में क्या सिद्ध किया, और आपने प्रति
        विभाजन कौन-सी दर नापी।
\end{enumerate}
\end{problem}
