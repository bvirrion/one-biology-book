\section*{Bab \ref{ch:g10:cell-metabolism} --- Metabolisme Sel}

\begin{solution}{exo:g10:cell-metabolism:1}
\omterm{def:g10:cell-metabolism:metabolism}{Metabolisme} adalah himpunan seluruh reaksi kimia dalam sebuah \omterm{def:g10:cells-common-unit:cell}{sel}, yang
membongkar molekul demi energi dan membangun molekul \omterm{def:g10:cells-common-unit:cell}{sel} itu sendiri.
\omterm{def:g10:cell-metabolism:metabolism}{Enzim} adalah \omterm{def:g10:chemistry-of-life:families}{protein} yang mempercepat satu reaksi tertentu, mengikat
substratnya dan melepaskan hasilnya, tanpa terpakai habis.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:2}
Respirasi: $\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \to
6\,\mathrm{CO_2} + 6\,\mathrm{H_2O}$ + energi. \omterm{def:g10:cell-metabolism:autotrophy}{Fotosintesis}:
$6\,\mathrm{CO_2} + 6\,\mathrm{H_2O}$ + cahaya $\to \mathrm{C_6H_{12}O_6}
+ 6\,\mathrm{O_2}$. Neraca materinya secara keseluruhan terbalik;
tetapi reaksinya, \omterm{def:g10:cell-metabolism:metabolism}{enzimnya}, tempatnya (\omterm{def:g10:cells-common-unit:organelle}{mitokondria} dibandingkan
\omterm{def:g10:cells-common-unit:organelle}{kloroplas}) dan sumber energinya (ikatan kimia dibandingkan cahaya)
sepenuhnya berbeda.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:3}
\omterm{def:g10:cell-metabolism:autotrophy}{Heterotrof}: jamur, \omterm{def:g10:cells-common-unit:cell}{sel} akar wortel, bakteri yoghurt, \omterm{def:g10:cells-common-unit:cell}{sel} otot manusia.
\omterm{def:g10:cell-metabolism:autotrophy}{Autotrof}: lumut, chlorella.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:4}
Karbon dioksida mengembangkan balonnya; etanol menumpuk dalam
cairannya. Labunya kekurangan oksigen: jika udara dialirkan lewat
gelembung, raginya akan berespirasi, tidak membuat etanol dan memakai
gula jauh lebih sedikit.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:5}
Katalase adalah \omterm{def:g10:chemistry-of-life:families}{protein}; perebusan merusak bentuknya dan dengan
demikian kegiatannya, sedangkan hidrogen peroksida tidak terurai cepat
dengan sendirinya.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:6}
$(7.9 - 0.9)/7 = \qty{1.0}{mg/L}$ tiap menit. Lengkungnya mendatar
karena oksigennya habis; raginya lalu beralih ke \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}, yang tidak
memakai oksigen.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:7}
$0.8 + 0.2 = \qty{1.0}{mg/L}$ tiap menit: kemiringan terangnya adalah
penghasilan dikurangi respirasi yang berlanjut dalam cahaya.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:8}
\qty{18}{g} sama dengan \qty{0.1}{mol} glukosa; ia memerlukan
\qty{0.6}{mol} oksigen, $0.6 \times 32 = \qty{19.2}{g}$, dan
melepaskan \qty{0.6}{mol} karbon dioksida, $0.6 \times 44 = \qty{26.4}{g}$.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:9}
Kacang polong yang berkecambah merespirasikan cadangannya dan
melepaskan panas; kacang yang direbus sudah mati, \omterm{def:g10:cell-metabolism:metabolism}{enzimnya} rusak,
sehingga tidak ada reaksi dan tidak ada panas. Jika tertutup rapat
udara, oksigennya akan habis dan respirasinya berhenti.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:10}
Separuh yang terbuka terwarnai biru kehitaman (pati yang dibuat lewat
\omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} dalam cahaya); separuh yang tertutup tidak. Kegelapan
selama 48 jam itu membiarkan daunnya menghabiskan pati sebelumnya,
sehingga pati yang ditemukan berasal dari cahaya hari itu.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:11}
\omterm{def:g10:cell-metabolism:fermentation}{Fermentasi} menghasilkan ATP kira-kira lima belas kali lebih sedikit
tiap glukosa daripada respirasi; untuk memperoleh ATP yang
diperlukannya, \omterm{def:g10:cells-common-unit:cell}{sel} yang berfermentasi harus membongkar gula lima belas
kali lebih banyak tiap menit.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:12}
Selama lombanya, kebutuhan ATP ototnya melampaui oksigen yang dapat
diantar darahnya; \omterm{def:g10:cells-common-unit:cell}{selnya} beralih ke \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} laktat, yang hasilnya,
yaitu asam laktat, lalu menumpuk. Dalam lari pelan pasokan oksigennya
mengimbangi, respirasi sendirian menyediakan ATP-nya, dan tidak ada
asam yang dibuat.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:13}
Lingkungannya berubah (tanpa cahaya, dengan makanan organik); \omterm{def:g10:universal-dna:gene}{gennya}
tidak. Keturunannya membangun kembali \omterm{def:g10:cells-common-unit:organelle}{kloroplasnya}, sehingga
petunjuknya tidak pernah hilang: yang hilang adalah perlengkapan yang
terpakai, bukan kemungkinan yang dibolehkan \omterm{def:g10:universal-dna:gene}{gennya}.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:14}
Tumbuhannya berfotosintesis dalam cahaya, memasok oksigen dan \omterm{def:g10:chemistry-of-life:mineral-organic}{materi
organik} (daun yang dimakan siputnya); siputnya berespirasi, memasok
karbon dioksida kepada tumbuhannya. Dalam gelap tumbuhannya hanya
berespirasi: keduanya memakai oksigen, tidak ada yang menghasilkannya,
dan keduanya mati dalam beberapa hari.
\end{solution}

\begin{solution}{exo:g10:cell-metabolism:15}
$30/2 = 15$ kali lebih banyak glukosa tiap menit. Pembuat bir
menginginkan etanol, yang hanya dibuat \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}; ragi yang
berespirasi akan mengubah gulanya menjadi karbon dioksida, air dan ragi
yang lebih banyak, tanpa alkohol.
\end{solution}

\begin{solution}{pb:g10:cell-metabolism:1}
\textbf{1.} \omterm{def:g10:cell-metabolism:fermentation}{Fermentasi} alkohol, karena tong yang tertutup itu tidak
memuat oksigen: $\mathrm{C_6H_{12}O_6} \to 2\,\mathrm{C_2H_5OH} +
2\,\mathrm{CO_2}$.

\textbf{2.} $200/180 \approx \qty{1.11}{mol}$.

\textbf{3.} Etanol $2 \times 1.11 = \qty{2.22}{mol}$, yaitu $2.22
\times 46 \approx \qty{102}{g}$; volumenya $102/0.79 \approx
\qty{129}{mL}$; sekitar 13\% menurut volume.

\textbf{4.} $2.22 \times 44 \approx \qty{98}{g}$ karbon dioksida;
$2.22 \times 24 \approx \qty{53}{L}$ gas.

\textbf{5.} Sekitar \qty{53000}{L} $= \qty{53}{m^3}$. Ruang bawah
tanahnya bervolume $4 \times 6 \times 2.5 = \qty{60}{m^3}$: gas yang
dilepaskan akan hampir mengisinya.

\textbf{6.} $\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \to
6\,\mathrm{CO_2} + 6\,\mathrm{H_2O}$.

\textbf{7.} Oksigen $6 \times 1.11 = \qty{6.67}{mol}$, $6.67 \times 32
\approx \qty{213}{g}$; karbon dioksida $6.67 \times 44 \approx
\qty{293}{g}$.

\textbf{8.} Tiga kali lebih banyak: respirasi mengubah keenam karbon
glukosanya menjadi karbon dioksida, \omterm{def:g10:cell-metabolism:fermentation}{fermentasi} hanya dua, sedangkan
empat lainnya tinggal dalam etanolnya.

\textbf{9.} Dengan respirasi, lima belas kali lebih lama: tiap glukosa
memberi ATP lima belas kali lebih banyak.

\textbf{10.} Etanol hanya dibuat lewat \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}; tong yang dialiri
udara akan memberi karbon dioksida, air dan ragi, tanpa anggur.

\textbf{11.} $1.11 \times 70 \approx \qty{78}{kJ}$ tiap liter;
\qty{78000}{kJ}, yaitu \qty{78}{MJ}, untuk seluruh tongnya.

\textbf{12.} $78/4.2 \approx \qty{18.5}{\celsius}$.

\textbf{13.} $20 + 18.5 \approx \qty{38.5}{\celsius}$: tepat di tepi
batas raginya, dan ruang bawah tanah yang hangat akan mendorongnya
melampaui batas. Tong yang besar melepaskan panas dengan lambat karena
permukaannya kecil dibandingkan volumenya, sehingga ia harus
didinginkan dengan sengaja.

\textbf{14.} \omterm{def:g10:cell-metabolism:metabolism}{Enzim} adalah \omterm{def:g10:chemistry-of-life:families}{protein}; panas membuka lipatannya dan
\omterm{def:g10:cell-metabolism:metabolism}{enzimnya} berhenti bekerja. Tanpa \omterm{def:g10:cell-metabolism:metabolism}{enzimnya}, raginya tidak dapat mengubah
gulanya, sebanyak apa pun yang tersisa.

\textbf{15.} Dalam ikatan kimia etanolnya: sekitar
\qty{2800}{kJ} tiap mol glukosa, dan itulah sebabnya etanol terbakar.

\textbf{16.} Di lantai, dalam lapisan yang menebal seiring berjalannya
\omterm{def:g10:cell-metabolism:fermentation}{fermentasi}. Lilin yang dibawa rendah padam di tempat karbon dioksida
telah menggeser oksigennya --- sebuah peringatan sebelum orangnya
terkena.

\textbf{17.} \omterm{def:g10:cell-metabolism:fermentation}{Fermentasi} laktat. Jalur itu menghasilkan ATP lima belas
kali lebih sedikit; otak dan jantung tidak dapat memenuhi kebutuhannya
dengan cara itu, dan asam laktatnya menumpuk. Hilangnya kesadaran
menyusul dalam satu atau dua menit.

\textbf{18.} Etanol pada kepekatan yang tinggi merusak membran dan
\omterm{def:g10:cell-metabolism:metabolism}{enzim} raginya: hasil \omterm{def:g10:cells-common-unit:cell}{sel} itu sendiri meracuni \omterm{def:g10:cell-metabolism:metabolism}{metabolismenya}.

\textbf{19.} Gelembungnya berisi karbon dioksida dari \omterm{def:g10:cell-metabolism:fermentation}{fermentasi}.
Etanolnya menguap dalam oven (etanol mendidih pada \qty{78}{\celsius}).

\textbf{20.} Sekitar 13\% alkohol menurut volume dan \qty{53}{L} gas
tiap liter sarinya; keduanya mengikuti dari raginya yang
memfermentasi, tanpa oksigen, tiap glukosa menjadi dua etanol dan dua
karbon dioksida.
\end{solution}
