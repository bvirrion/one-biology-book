\chapter{Hewan di Sekitar Kita}\label{ch:g1:animals-around-us}

Seekor burung hitam melompat-lompat di halaman rumput, seekor kucing
mengawasinya dari atas tembok, seekor semut memanggul remah roti yang
lebih besar daripada dirinya di ambang pintu. \omterm{def:g1:animals-around-us:animal}{Hewan} ada di mana-mana
di sekitar kita --- di kebun, di taman, di dalam rumah. Mari belajar
memandang mereka seperti seorang ahli \omterm{def:g1:living-or-not:living}{biologi}: pada tubuhnya, dan pada
caranya bergerak.

\section{Apa itu hewan?}

\begin{definition}[Hewan]\label{def:g1:animals-around-us:animal}
Suatu \emph{hewan}\index{hewan} adalah \omterm{def:g1:living-or-not:living}{makhluk hidup} yang makan dengan
memakan \omterm{def:g1:living-or-not:living}{makhluk hidup} lain --- tumbuhan, atau hewan lain --- dan yang
biasanya dapat berpindah tempat sendiri. Kucing, burung hitam, ikan
dan semut semuanya hewan. Kita pun hewan.
\end{definition}

\begin{example}[Hewan di satu kebun]\label{ex:g1:animals-around-us:garden}
Satu kebun kecil dapat menjadi rumah bagi banyak \omterm{def:g1:animals-around-us:animal}{hewan}: seekor burung
hitam yang menarik cacing dari tanah, cacingnya sendiri, seekor siput
di daun, seekor laba-laba di jaringnya, seekor lebah di bunga, dan
kucing tetangga. Besar atau mungil, semuanya makan dan bergerak.
\end{example}

\begin{remark}[Kecil bukan berarti tak penting]\label{rem:g1:animals-around-us:small}
Seekor semut sama-sama \omterm{def:g1:animals-around-us:animal}{hewan} seperti seekor gajah. Ukuran tubuh tidak
menentukan siapa yang \omterm{def:g1:animals-around-us:animal}{hewan}: makan dan bergeraklah yang menentukan.
Beberapa \omterm{def:g1:animals-around-us:animal}{hewan} paling menakjubkan di sekitar kita justru yang paling
mungil.
\end{remark}

\section{Bagian-bagian tubuh hewan}

\begin{definition}[Bagian-bagian tubuh]\label{def:g1:animals-around-us:parts}
Kebanyakan \omterm{def:g1:animals-around-us:animal}{hewan} mempunyai \emph{kepala}\index{kepala} (dengan mulut
dan sebagian besar alat indra), tubuh, dan \emph{anggota
gerak}\index{anggota gerak} --- kaki, sayap atau sirip --- yang
dipakainya untuk bergerak. Banyak pula yang mempunyai
\emph{ekor}\index{ekor}.
\end{definition}

\begin{example}[Hitung kakinya]\label{ex:g1:animals-around-us:legs}
Jumlah kaki memang layak dihitung. Burung hitam berjalan dengan $2$
kaki, kucing dengan $4$, kepik dengan $6$, laba-laba dengan $8$ ---
sedangkan cacing tanah dan siput sama sekali tidak berkaki, namun
tetap sampai ke tujuannya.
\end{example}

\begin{definition}[Penutup tubuh]\label{def:g1:animals-around-us:covering}
Tubuh \omterm{def:g1:animals-around-us:animal}{hewan} dilindungi oleh sebuah penutup:
\emph{bulu}\index{bulu} (rambut) pada kucing, \emph{bulu
burung}\index{bulu burung} pada burung hitam, \emph{sisik}\index{sisik}
pada ikan, \emph{cangkang}\index{cangkang} pada siput, dan
\emph{kulit}\index{kulit} yang licin dan basah pada katak.
\end{definition}

\begin{example}[Siapa memakai apa]\label{ex:g1:animals-around-us:covering}
Belai seekor kucing: \omterm{def:g1:animals-around-us:covering}{bulu}. Pungut sehelai \omterm{def:g1:animals-around-us:covering}{bulu burung} hitam yang
jatuh: ringan dan lembut. Amati ikan di pasar dari dekat: deretan
\omterm{def:g1:animals-around-us:covering}{sisik} mengilap, seperti genting mungil. Tiga \omterm{def:g1:animals-around-us:animal}{hewan}, tiga mantel yang
berbeda.
\end{example}

\section{Bagaimana hewan bergerak}

\begin{proposition}[Cara-cara bergerak]\label{prop:g1:animals-around-us:moving}
\omterm{def:g1:animals-around-us:animal}{Hewan} bergerak dengan cara yang sepadan dengan \omterm{def:g1:animals-around-us:parts}{anggota geraknya}:
\begin{itemize}
  \item \omterm{def:g1:animals-around-us:animal}{hewan} berkaki \emph{berjalan}, berlari, melompat atau merangkak
        --- kucing, burung hitam di halaman rumput, semut;
  \item \omterm{def:g1:animals-around-us:animal}{hewan} bersayap \emph{terbang}\index{terbang} --- burung hitam,
        lebah, kupu-kupu;
  \item \omterm{def:g1:animals-around-us:animal}{hewan} bersirip \emph{berenang}\index{berenang} --- ikan;
  \item \omterm{def:g1:animals-around-us:animal}{hewan} tanpa \omterm{def:g1:animals-around-us:parts}{anggota gerak} merayap atau menggeliat --- siput
        dengan satu kakinya yang lunak, cacing tanah yang memanjang
        lalu memendek.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Satu hewan, beberapa cara]\label{rem:g1:animals-around-us:several}
Banyak \omterm{def:g1:animals-around-us:animal}{hewan} mempunyai lebih dari satu cara bergerak. Burung hitam
berjalan \emph{dan} terbang; katak berenang \emph{dan} melompat; bebek
berjalan, berenang \emph{dan} terbang. Amati seekor \omterm{def:g1:animals-around-us:animal}{hewan} agak lama
sebelum memutuskan bagaimana ia bergerak.
\end{remark}

\begin{method}[Mengamati seekor hewan]\label{met:g1:animals-around-us:observe}
Untuk melukiskan seekor \omterm{def:g1:animals-around-us:animal}{hewan} seperti seorang ahli \omterm{def:g1:living-or-not:living}{biologi}:
\begin{enumerate}
  \item amati bagian-bagiannya: kepala, tubuh, \omterm{def:g1:animals-around-us:parts}{anggota gerak}, ekor;
  \item hitung kakinya, jangan menebak;
  \item sebutkan penutupnya: \omterm{def:g1:animals-around-us:covering}{bulu}, \omterm{def:g1:animals-around-us:covering}{bulu burung}, \omterm{def:g1:animals-around-us:covering}{sisik}, \omterm{def:g1:animals-around-us:covering}{cangkang}, \omterm{def:g1:animals-around-us:covering}{kulit};
  \item perhatikan caranya bergerak: berjalan, melompat, terbang,
        berenang, merayap;
  \item sebutkan di mana kamu menemukannya: rumput, pohon, kolam, di
        bawah batu.
\end{enumerate}
Jangan pernah menyakiti \omterm{def:g1:animals-around-us:animal}{hewannya}: amati, lalu biarkan ia meneruskan
perjalanannya.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:1}
Sebutkan empat bagian utama tubuh kucing pada gambarnya. Bagian mana
yang membawa mata dan mulut?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:2}
Berapa jumlah kakinya: (a) seekor ayam? (b) seekor anjing? (c) seekor
kepik? (d) seekor laba-laba? (e) seekor cacing tanah?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:3}
Sebutkan penutup tubuh masing-masing \omterm{def:g1:animals-around-us:animal}{hewan}: seekor domba, seekor
bebek, seekor ikan trout, seekor siput.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:4}
\omterm{def:g1:animals-around-us:animal}{Hewan} mana saja dalam bab ini yang dapat terbang? Apa yang dimiliki
mereka semua sebagai pengganti kaki depan atau lengan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:5}
Siput dan cacing tanah tidak berkaki. Bagaimana cara masing-masing
bergerak?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:6}
Seekor bebek dapat bergerak dengan tiga cara. Sebutkan ketiganya, dan
sebutkan bagian tubuh mana yang dipakainya untuk tiap cara.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:7}
Mengapa semut termasuk \omterm{def:g1:animals-around-us:animal}{hewan} menurut \cref{def:g1:animals-around-us:animal}?
Jawablah dengan kedua gagasan dalam definisinya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:8}
Cobalah: amati satu \omterm{def:g1:animals-around-us:animal}{hewan} di luar rumah agak lama --- seekor burung,
seekor serangga, seekor siput. Lukiskan ia dengan lima langkah
\cref{met:g1:animals-around-us:observe}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:9}
Seorang teman berkata: ``Ikan tidak mungkin \omterm{def:g1:animals-around-us:animal}{hewan} --- ikan tidak
berjalan!'' Apa jawabanmu? Pakai \cref{prop:g1:animals-around-us:moving}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:10}
\omterm{def:g1:animals-around-us:animal}{Hewan} misteri: ia mempunyai $2$ kaki, $2$ sayap, \omterm{def:g1:animals-around-us:covering}{bulu burung}, dan ia
tidak dapat terbang, tetapi ia berlari sangat cepat. Apakah ia burung?
Mungkinkah ia pelari tercepat di lingkunganmu? (Pikirkan burung unta.)
\end{exercise}
