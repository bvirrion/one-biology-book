\section*{Capítulo \ref{ch:g11:genes-and-disease} --- Variação genética e doença}

\begin{solution}{exo:g11:genes-and-disease:1}
\omterm{def:g11:genes-and-disease:genetic}{Recessivo}: a doença só aparece com duas cópias do \omterm{def:g10:universal-dna:gene}{alelo}; \omterm{def:g11:genes-and-disease:genetic}{dominante}: uma
cópia basta. Um \omterm{def:g11:genes-and-disease:genetic}{portador} tem um \omterm{def:g10:universal-dna:gene}{alelo} mutante \omterm{def:g11:genes-and-disease:genetic}{recessivo} e é saudável.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:2}
Afetado $1/4$, \omterm{def:g11:genes-and-disease:genetic}{portador} $1/2$, nenhum dos dois $1/4$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:3}
I-1 $Aa$, I-2 $Aa$ (pais de uma criança afetada), II-2 $aa$, III-2 $aa$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:4}
Um homem tem um \omterm{def:g11:cell-cycle-mitosis:chromosome}{cromossomo} X: um \omterm{def:g10:universal-dna:gene}{alelo} \omterm{def:g11:genes-and-disease:genetic}{recessivo} nele é expresso, sem
uma segunda cópia para mascará-lo. Uma mulher precisa do \omterm{def:g10:universal-dna:gene}{alelo} nos dois
\omterm{def:g10:universal-dna:information}{cromossomos} X dela.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:5}
Uma doença cujo risco depende de muitos \omterm{def:g10:universal-dna:gene}{alelos} de pequeno efeito e do
ambiente: o diabetes tipo 2, a doença do coração (e também a asma, a
maior parte dos cânceres).
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:6}
$aa \times AA$: todos os filhos $Aa$, \omterm{def:g11:genes-and-disease:genetic}{portadores} saudáveis. Um filho
\omterm{def:g11:genes-and-disease:genetic}{portador} com uma pessoa portadora: $1/4$ afetados, $1/2$ \omterm{def:g11:genes-and-disease:genetic}{portadores},
$1/4$ não \omterm{def:g11:genes-and-disease:genetic}{portadores}.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:7}
II-1 é uma filha saudável de dois \omterm{def:g11:genes-and-disease:genetic}{portadores}: portadora com
probabilidade $2/3$. Risco com um homem da população: $2/3 \times 1/25 \times 1/4
= 1/150$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:8}
$1/2$ para cada filho. Nenhum dos quatro: $(1/2)^4 = 1/16$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:9}
$X^hY \times X^HX^H$: todas as filhas $X^HX^h$, portadoras saudáveis;
todos os filhos homens $X^HY$, saudáveis. Uma filha portadora com um
homem saudável: metade dos filhos homens hemofílicos, metade das filhas
portadoras.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:10}
Um \omterm{def:g10:universal-dna:gene}{alelo} dá 400 e 200; o outro, com a deleção adicional, dá 300 e 200.
Bandas em 400, 300 e 200 --- e nenhuma em 600.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:11}
A frequência de um \omterm{def:g10:universal-dna:gene}{alelo} numa população depende da história dele, e não
só da taxa de \omterm{def:g11:mutations:mutation}{mutação}: o \omterm{def:g10:universal-dna:gene}{alelo} europeu surgiu há muito tempo e se
espalhou, talvez porque os \omterm{def:g11:genes-and-disease:genetic}{portadores} tivessem alguma vantagem
(suspeita-se de resistência a uma doença diarreica); no leste da Ásia
isso não aconteceu.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:12}
Cada filho é um sorteio independente: $1/4$ para o segundo. Pelo menos
um dos dois seguintes: $1 - (3/4)^2 = 7/16$. Os \omterm{def:g10:universal-dna:gene}{alelos} são distribuídos
de novo a cada concepção; um filho afetado anterior não reduz o risco do
seguinte.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:13}
Antes do teste, portadora com probabilidade $1/25$; o teste deixa
escapar 10\% das portadoras, de modo que entre 1000 mulheres 40 são
portadoras e 4 dão negativo, enquanto 960 não portadoras dão negativo:
probabilidade de ser portadora $4/964 \approx 1/240$. Risco com um
\omterm{def:g11:genes-and-disease:genetic}{portador} conhecido: $1/240 \times 1/4
\approx 1/960$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:14}
De muitos a poucos \omterm{def:g10:universal-dna:gene}{alelos}, sedentário: de 34\% a 9\%; de sedentário a
ativo, com muitos \omterm{def:g10:universal-dna:gene}{alelos}: de 34\% a 12\%. A mudança que se pode fazer
dá quase a mesma redução da mudança que não se pode fazer. Um escore de
risco é útil para identificar quem mais ganha ao mudar o modo de vida,
e não como veredito.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:15}
O teste oferece certeza --- livrar-se de uma sombra de cinquenta por
cento se for negativo, a possibilidade de planejar uma vida e uma
família se for positivo --- e custa, se for positivo, o conhecimento de
uma doença incurável décadas antes, com efeitos sobre o humor, os
seguros e os parentes que compartilham o \omterm{def:g10:universal-dna:gene}{gene}. Nenhum tratamento
decorre do resultado, de modo que a única razão para saber é a da
própria pessoa; é por isso que a escolha, e o momento dela, são dela.
\end{solution}

\begin{solution}{pb:g11:genes-and-disease:1}
\textbf{1.} Recessiva: Zoé é afetada e os dois pais dela são saudáveis,
de modo que cada um carrega um \omterm{def:g10:universal-dna:gene}{alelo} mutante escondido.

\textbf{2.} Léa $Aa$, Tom $Aa$, Zoé $aa$.

\textbf{3.} $AA$ com probabilidade $1/3$, $Aa$ com $2/3$ (ele é
saudável, de modo que $aa$ está excluído).

\textbf{4.} Tom é \omterm{def:g11:genes-and-disease:genetic}{portador}, logo pelo menos um dos pais dele também é;
sem nenhuma pessoa afetada naquela geração, Ana recebeu o \omterm{def:g10:universal-dna:gene}{alelo} desse
genitor com probabilidade $1/2$: $1/2$ (tomando o outro genitor como
não \omterm{def:g11:genes-and-disease:genetic}{portador}).

\textbf{5.} $1/25$, o número da população.

\textbf{6.} $1/2 \times 1/25 \times 1/4 = 1/200$.

\textbf{7.} Afetado $1/4$; \omterm{def:g11:genes-and-disease:genetic}{portador} $1/2$.

\textbf{8.} $2/3 \times 1/25 \times 1/4 = 1/150$.

\textbf{9.} Zoé é $aa$; o parceiro dela é $Aa$ com probabilidade
$1/25$. Afetado: $1/25 \times 1/2 = 1/50$. \omterm{def:g11:genes-and-disease:genetic}{Portador}: todo filho recebe
$a$ de Zoé, logo é \omterm{def:g11:genes-and-disease:genetic}{portador} com probabilidade $24/25 \times 1 + 1/25 \times
1/2 \approx 0.98$.

\textbf{10.} Lou é \omterm{def:g11:genes-and-disease:genetic}{portador} com probabilidade $2/3$; o primo recebeu
$a$ de Ana ($1/2$ portadora) com probabilidade $1/2$, ou seja é
\omterm{def:g11:genes-and-disease:genetic}{portador} com probabilidade $1/4$ (ignorando Karim): $2/3 \times 1/4
\times 1/4 = 1/24$, quatro vezes mais que para Max, porque o primo vem
da mesma família e tem chance bem maior que um estranho de carregar o
mesmo \omterm{def:g10:universal-dna:gene}{alelo}.

\textbf{11.} Zoé: só 600 e 300. Tom: 900, 600 e 300. Não \omterm{def:g11:genes-and-disease:genetic}{portador}: só 900.

\textbf{12.} $AA$: ele não é \omterm{def:g11:genes-and-disease:genetic}{portador}; o risco para o filho dele cai de
$1/150$ a zero (salvo uma \omterm{def:g11:mutations:mutation}{mutação} nova).

\textbf{13.} Ana é $Aa$: risco $1 \times 1/25 \times 1/4 = 1/100$.

\textbf{14.} Entre 1000 pessoas como Karim, 40 são portadoras, das
quais 4 dão negativo, e 960 não portadoras dão negativo: $4/964 \approx
1/240$. Risco: $1 \times 1/240 \times 1/4 \approx 1/960$.

\textbf{15.} $Aa$: um \omterm{def:g11:genes-and-disease:genetic}{portador} saudável, como a mãe.

\textbf{16.} $(1/25)^2 = 1/625$ dos casais são dois \omterm{def:g11:genes-and-disease:genetic}{portadores}; um
quarto dos filhos deles é afetado: $1/2500$.

\textbf{17.} Exatamente a frequência observada: a frequência de
\omterm{def:g11:genes-and-disease:genetic}{portadores} e a regra recessiva dão conta dela.

\textbf{18.} Com uma frequência de \omterm{def:g11:genes-and-disease:genetic}{portadores} de $1/25$ e uma
frequência de afetados de $1/2500$, os \omterm{def:g10:universal-dna:gene}{alelos} mutantes carregados por
\omterm{def:g11:genes-and-disease:genetic}{portadores} saudáveis superam os das pessoas afetadas na proporção de
$2 \times 1/25$ para $2 \times
1/2500$, cem para um. A morte dos afetados retira apenas 1\% das cópias
por geração; o resto é transmitido de forma invisível pelos \omterm{def:g11:genes-and-disease:genetic}{portadores}.

\textbf{19.} A frequência do \omterm{def:g10:universal-dna:gene}{alelo} vai subir, mas muito devagar: o 1\%
das cópias que se perdia a cada geração agora é conservado, de modo que
a mudança é da ordem de 1\% da frequência por geração.

\textbf{20.} Dois \omterm{def:g11:genes-and-disease:genetic}{portadores} correm o risco de um filho afetado em
quatro; um irmão saudável de uma criança afetada é \omterm{def:g11:genes-and-disease:genetic}{portador} duas vezes
em três; o gel transformou probabilidades em \omterm{def:g11:enzymes-and-phenotype:phenotype}{genótipos} --- inocentando
Max, confirmando Ana, tranquilizando mas não inocentando Karim --- e
identificou o feto como \omterm{def:g11:genes-and-disease:genetic}{portador} saudável.
\end{solution}
