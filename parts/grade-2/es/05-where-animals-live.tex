\chapter{Dónde viven los animales}\label{ch:g2:where-animals-live}

No te encontrarás una trucha en el patio del colegio, ni una ardilla
en el fondo de la charca. Cada \omterm{def:g1:animals-around-us:animal}{animal} vive en el sitio que le conviene
--- allí donde encuentra su comida, su refugio y sus vecinos. Los
biólogos tienen un nombre para el terreno propio de un \omterm{def:g1:animals-around-us:animal}{animal}.

\section{Los hábitats}

\begin{definition}[Hábitat]\label{def:g2:where-animals-live:habitat}
El \emph{hábitat}\index{hábitat} de un \omterm{def:g1:animals-around-us:animal}{animal} es el tipo de sitio
donde vive de forma natural y donde encuentra todo lo que necesita:
comida, agua, refugio y un rincón seguro para sus crías. La charca, el
bosque, el prado, el seto y la orilla del mar son hábitats.
\end{definition}

\begin{example}[Cuatro hábitats, cuatro barrios]\label{ex:g2:where-animals-live:four}
En la \emph{charca}: ranas, libélulas, espinosos, caracoles de agua.
En el \emph{bosque}: ardillas, ciervos, pájaros carpinteros, hormigas
rojas. En el \emph{prado}: conejos, saltamontes, mariposas, ratones de
campo. Bajo las piedras de un \emph{muro} viejo: cochinillas, arañas y
a veces una lagartija tomando el sol encima.
\end{example}

\begin{remark}[Un hábitat, muchas casas]\label{rem:g2:where-animals-live:layers}
Un \omterm{def:g2:where-animals-live:habitat}{hábitat} tiene muchos rincones. En un mismo bosque, el pájaro
carpintero vive arriba en los troncos, la ardilla en las \omterm{def:g1:plants-around-us:tree}{ramas}, el
ciervo entre los matorrales y las hormigas rojas en el suelo, en su
nido amontonado. El mismo bosque, pisos distintos --- por eso pueden
compartir un \omterm{def:g2:where-animals-live:habitat}{hábitat} muchos \omterm{def:g1:animals-around-us:animal}{animales} sin estorbarse.
\end{remark}

\section{Hechos a la medida del hábitat}

\begin{proposition}[Los animales encajan en su hábitat]\label{prop:g2:where-animals-live:fit}
El cuerpo y las costumbres de un \omterm{def:g1:animals-around-us:animal}{animal} encajan con el sitio donde vive:
\begin{itemize}
  \item los pies palmeados de la rana empujan el agua --- hechos para
        la charca;
  \item las garras que se agarran y la gran cola de equilibrio de la
        ardilla están hechas para los árboles;
  \item las patas excavadoras y las largas \omterm{def:g1:the-five-senses:sense}{orejas} de alarma del conejo
        están hechas para el prado abierto con madrigueras debajo;
  \item las \omterm{def:g1:animals-around-us:covering}{plumas} del pato repelen el agua y sus pies palmeados reman.
\end{itemize}
Llevado al \omterm{def:g2:where-animals-live:habitat}{hábitat} equivocado, un \omterm{def:g1:animals-around-us:animal}{animal} pierde estas ventajas.
\end{proposition}
\admitted

\begin{example}[El equipo del topo]\label{ex:g2:where-animals-live:mole}
El topo vive bajo tierra, en túneles que él mismo cava bajo el prado.
Sus manos son palas anchas de cavar, su \omterm{def:g1:animals-around-us:covering}{pelo} se aplasta en los dos
sentidos para que la tierra no lo despeine nunca, y sus \omterm{def:g1:the-five-senses:sense}{ojos} débiles
apenas importan en la oscuridad. Fuera es torpe; en su propio \omterm{def:g2:where-animals-live:habitat}{hábitat}
es un campeón.
\end{example}

\begin{method}[Emparejar un animal con su hábitat]\label{met:g2:where-animals-live:matching}
Ante un \omterm{def:g1:animals-around-us:animal}{animal} desconocido:
\begin{enumerate}
  \item mira su equipo: ¿pies palmeados? ¿garras? ¿alas? ¿patas
        excavadoras?
  \item pregunta qué come y dónde se encuentra esa comida;
  \item pregunta dónde puede refugiarse y criar;
  \item el \omterm{def:g2:where-animals-live:habitat}{hábitat} que responde a las tres cosas es su casa probable.
\end{enumerate}
\end{method}

\section{Los hábitats se pueden perder}

\begin{proposition}[Sin hábitat no hay habitantes]\label{prop:g2:where-animals-live:loss}
Un \omterm{def:g1:animals-around-us:animal}{animal} no puede vivir sin su \omterm{def:g2:where-animals-live:habitat}{hábitat}. Si se rellena una charca,
desaparecen con ella sus ranas y sus libélulas; si se arranca un seto,
los pájaros que anidaban allí se quedan sin refugio. Proteger a los
\omterm{def:g1:animals-around-us:animal}{animales} empieza por proteger los sitios donde viven.
\end{proposition}
\admitted

\begin{example}[La charca del colegio]\label{ex:g2:where-animals-live:schoolpond}
Un colegio cava una charca pequeña en un rincón del patio. En dos
años, las libélulas la patrullan, los zapateros andan sobre ella y las
ranas se han mudado allí --- nadie las llevó. Haz el \omterm{def:g2:where-animals-live:habitat}{hábitat} y los
habitantes lo encuentran.
\end{example}

\begin{remark}[Mirar sin molestar]\label{rem:g2:where-animals-live:respect}
Cuando visitamos un \omterm{def:g2:where-animals-live:habitat}{hábitat}, somos invitados. Observa en silencio,
vuelve a dejar las piedras como estaban, no toques \omterm{def:g2:animals-grow-up:born}{huevos} ni nidos y
llévate tu basura a casa --- el \omterm{def:g2:where-animals-live:habitat}{hábitat} debería olvidar que estuviste
allí. Los capítulos siguientes dan más maneras de cuidar la naturaleza.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:1}
¿Qué es un \omterm{def:g2:where-animals-live:habitat}{hábitat}? Nombra las cuatro cosas que un \omterm{def:g1:animals-around-us:animal}{animal} encuentra en él.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:2}
Di dos \omterm{def:g1:animals-around-us:animal}{animales} de la charca, dos del bosque y dos del prado.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:3}
¿Qué \omterm{def:g2:where-animals-live:habitat}{hábitat} le va a cada \omterm{def:g1:animals-around-us:animal}{animal}: al espinoso, al pájaro carpintero,
al saltamontes, a la cochinilla?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:4}
¿Cuál es el equipo de nadar de la rana? ¿Y el equipo de trepar a los
árboles de la ardilla?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:5}
Enumera las tres piezas del equipo de subsuelo del topo, según el
\cref{ex:g2:where-animals-live:mole}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:6}
En la \cref{rem:g2:where-animals-live:layers}, ¿qué ``piso'' del
bosque usa cada \omterm{def:g1:animals-around-us:animal}{animal}: el pájaro carpintero, la ardilla, el ciervo,
la hormiga roja?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:7}
¿Qué les pasa a las ranas y a las libélulas cuando se rellena una
charca? ¿Qué enseña eso sobre proteger a los \omterm{def:g1:animals-around-us:animal}{animales}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:where-animals-live:8}
Pruébalo: elige un \omterm{def:g2:where-animals-live:habitat}{hábitat} pequeño cerca de ti --- bajo una piedra, un
arbusto, un arriate. Observa en silencio y haz la lista de todos los
\omterm{def:g1:animals-around-us:animal}{animales} que veas en un tiempo fijo. Vuelve a poner cada piedra en su
sitio. ¿Cuántos habitantes tenía tu pequeño \omterm{def:g2:where-animals-live:habitat}{hábitat}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:where-animals-live:9}
Un \omterm{def:g1:animals-around-us:animal}{animal} misterioso tiene los pies palmeados, \omterm{def:g1:animals-around-us:covering}{plumas} espesas que
repelen el agua y un pico plano para cribar. Usa el
\cref{met:g2:where-animals-live:matching} para nombrar su \omterm{def:g2:where-animals-live:habitat}{hábitat} ---
y quizá el \omterm{def:g1:animals-around-us:animal}{animal}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:where-animals-live:10}
Tomás quiere ``ayudar'' a una rana llevándosela a casa a un terrario
seco y con comida de sobra. Con la
\cref{prop:g2:where-animals-live:fit} y la
\cref{prop:g2:where-animals-live:loss}, explica por qué eso no
ayudaría nada a la rana.
\end{exercise}
