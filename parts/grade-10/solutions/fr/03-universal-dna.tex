\section*{Chapitre \ref{ch:g10:universal-dna} --- L'ADN, molécule génétique universelle}

\begin{solution}{exo:g10:universal-dna:1}
Un groupement phosphate, un sucre (le désoxyribose) et une base.
\omterm{def:g10:universal-dna:nucleotide}{Bases}~: adénine, thymine, guanine, cytosine.
\end{solution}

\begin{solution}{exo:g10:universal-dna:2}
\texttt{ATGCCTAAGT}, écrit base sous base.
\end{solution}

\begin{solution}{exo:g10:universal-dna:3}
C $= 21\%$~; A $=$ T $= (100 - 42)/2 = 29\%$ chacune.
\end{solution}

\begin{solution}{exo:g10:universal-dna:4}
Un \omterm{def:g10:universal-dna:gene}{gène} est un segment d'\omterm{def:g10:universal-dna:information}{ADN} dont la séquence porte les instructions
d'une \omterm{def:g10:chemistry-of-life:families}{protéine}, à une position fixe sur un \omterm{def:g10:universal-dna:information}{chromosome}. Un \omterm{def:g10:universal-dna:gene}{allèle} est
l'une des versions d'un \omterm{def:g10:universal-dna:gene}{gène}, qui diffère des autres par quelques \omterm{def:g10:universal-dna:nucleotide}{bases}
de sa séquence.
\end{solution}

\begin{solution}{exo:g10:universal-dna:5}
Un organisme dans l'\omterm{def:g10:universal-dna:information}{ADN} duquel un \omterm{def:g10:universal-dna:gene}{gène} d'une autre espèce a été
inséré, et qui le lit et le transmet. Exemples~: des souris portant le
\omterm{def:g10:universal-dna:gene}{gène} GFP de méduse~; \emph{E.~coli} portant le \omterm{def:g10:universal-dna:gene}{gène} humain de
l'insuline.
\end{solution}

\begin{solution}{exo:g10:universal-dna:6}
$1500 \times 0.34 = \qty{510}{nm}$~; $1500/10 = 150$ tours.
\end{solution}

\begin{solution}{exo:g10:universal-dna:7}
$2 \times \num{3.2e9} \times \qty{0.34}{nm} \approx \qty{2.2}{m}$, soit
environ $\num{3.6e5}$ fois le diamètre du \omterm{def:g10:cells-common-unit:organelle}{noyau}.
\end{solution}

\begin{solution}{exo:g10:universal-dna:8}
Une fraction «~purifiée~» peut encore contenir des traces d'autres
molécules, si bien que son activité seule ne prouve pas que l'\omterm{def:g10:universal-dna:information}{ADN} en
est responsable. Détruire spécifiquement l'\omterm{def:g10:universal-dna:information}{ADN}, dans l'extrait complet,
et voir l'activité disparaître alors que les enzymes destructrices de
\omterm{def:g10:chemistry-of-life:families}{protéines} et d'ARN la laissent intacte, impute l'effet à l'\omterm{def:g10:universal-dna:information}{ADN} et à
rien d'autre.
\end{solution}

\begin{solution}{exo:g10:universal-dna:9}
Une répétition fixe donnerait la même composition (25~\% de chaque
base) chez toutes les espèces. Chargaff a trouvé une part d'A+T très
variable d'une espèce à l'autre~: l'ordre des \omterm{def:g10:universal-dna:nucleotide}{bases} n'est pas un motif
fixe, donc la séquence est libre de varier --- et c'est ce qui la rend
capable de porter une information.
\end{solution}

\begin{solution}{exo:g10:universal-dna:10}
La séquence est lue dans l'ordre pour construire la \omterm{def:g10:chemistry-of-life:families}{protéine}~; une
base changée à une position qui spécifie un \omterm{def:g10:chemistry-of-life:families}{acide aminé} peut
substituer un \omterm{def:g10:chemistry-of-life:families}{acide aminé} à un autre et modifier la forme ou
l'activité de la \omterm{def:g10:chemistry-of-life:families}{protéine}. Si la base changée se trouve dans une partie
non lue du \omterm{def:g10:universal-dna:gene}{gène}, ou si elle spécifie le même \omterm{def:g10:chemistry-of-life:families}{acide aminé}, la \omterm{def:g10:chemistry-of-life:families}{protéine}
est inchangée.
\end{solution}

\begin{solution}{exo:g10:universal-dna:11}
Toutes les \omterm{def:g10:cells-common-unit:cell}{cellules} de la souris descendent par divisions de l'œuf
fécondé~; à chaque division, l'\omterm{def:g10:universal-dna:information}{ADN}, transgène compris, est copié
(\omterm{prop:g10:universal-dna:helix}{complémentarité}) et partagé. Les \omterm{def:g10:cells-common-unit:cell}{cellules} de peau, de foie et les
spermatozoïdes portent donc tous le \omterm{def:g10:universal-dna:gene}{gène} --- et le spermatozoïde le
transmet à la génération suivante.
\end{solution}

\begin{solution}{exo:g10:universal-dna:12}
$\num{4.6e6} \times \qty{0.34}{nm} \approx \qty{1.6}{mm}$, soit quelque
800 fois la longueur de la \omterm{def:g10:cells-common-unit:cell}{cellule}. Mais la molécule ne fait que
\qty{2}{nm} de large~: son volume, environ $\pi \times 1^2 \times \num{1.6e6}\,
\unit{nm^3} \approx \num{5e-3}\,\unit{\micro m^3}$, fait environ 1~\%
de celui de la \omterm{def:g10:cells-common-unit:cell}{cellule}~; repliée et enroulée, elle y tient largement.
\end{solution}

\begin{solution}{exo:g10:universal-dna:13}
L'\omterm{def:g10:universal-dna:information}{ADN} humain est 700 fois plus long mais ne porte qu'environ 5 fois
plus de \omterm{def:g10:universal-dna:gene}{gènes}~: l'essentiel n'est pas constitué de \omterm{def:g10:universal-dna:gene}{gènes}. Une partie de
l'\omterm{def:g10:universal-dna:information}{ADN} supplémentaire règle le moment où les \omterm{def:g10:universal-dna:gene}{gènes} sont utilisés, une
autre est de la séquence répétée sans fonction connue. Le nombre de
\omterm{def:g10:universal-dna:nucleotide}{bases} mesure la taille du texte, non la complexité de l'organisme~; le
nombre de \omterm{def:g10:universal-dna:gene}{gènes} et, surtout, la façon dont ils sont combinés et
régulés comptent davantage.
\end{solution}

\begin{solution}{exo:g10:universal-dna:14}
Cela contredirait l'universalité de l'\omterm{def:g10:universal-dna:information}{ADN}
(\cref{prop:g10:universal-dna:universal}). Puisqu'une molécule et un
code partagés sont hérités d'un ancêtre commun, une telle espèce
devrait descendre d'une origine de la vie distincte~: ses \omterm{def:g10:universal-dna:gene}{gènes} ne
pourraient pas être lus par nos \omterm{def:g10:cells-common-unit:cell}{cellules} et la \omterm{prop:g10:universal-dna:universal}{transgenèse} entre les
deux lignées serait impossible.
\end{solution}

\begin{solution}{exo:g10:universal-dna:15}
Le produit bactérien est exactement la \omterm{def:g10:chemistry-of-life:families}{protéine} humaine (l'insuline de
porc diffère par un \omterm{def:g10:chemistry-of-life:families}{acide aminé} et peut provoquer des réactions
immunitaires), et il est fabriqué en quantité illimitée, exempt d'agents
pathogènes du porc. Une agence sanitaire demanderait encore si la
\omterm{def:g10:chemistry-of-life:families}{protéine} purifiée est exempte de contaminants bactériens et si elle est
correctement repliée et active.
\end{solution}

\begin{solution}{pb:g10:universal-dna:1}
\textbf{1.} $720 \times 0.34 \approx \qty{245}{nm}$~; $720/10 = 72$
tours.

\textbf{2.} \texttt{TACTCATTTCCTCTTCTTG}.

\textbf{3.} T $= 27\%$~; G $=$ C $= (100 - 54)/2 = 23\%$.

\textbf{4.} $720/238 \approx 3.0$~: trois \omterm{def:g10:universal-dna:nucleotide}{bases} par \omterm{def:g10:chemistry-of-life:families}{acide aminé} (les
trois dernières signalent la fin de la chaîne).

\textbf{5.} $4^{720}$ séquences, soit environ $10^{433}$. La chimie
autorise n'importe quel ordre, si bien que l'ordre particulier choisi
parmi toutes ces possibilités est un message --- une information.

\textbf{6.} L'universalité~: tout être vivant utilise les mêmes quatre
\omterm{def:g10:universal-dna:nucleotide}{nucléotides} et lit une séquence selon les mêmes règles.

\textbf{7.} Toutes les \omterm{def:g10:cells-common-unit:cell}{cellules} de la souris descendent de l'œuf par
divisions, si bien qu'un \omterm{def:g10:universal-dna:gene}{gène} inséré dans l'œuf est copié dans chaque
\omterm{def:g10:cells-common-unit:cell}{cellule}. Inséré dans la peau d'un adulte, il ne serait présent que dans
les \omterm{def:g10:cells-common-unit:cell}{cellules} de peau traitées et leurs descendantes.

\textbf{8.} Chaque type cellulaire contient tout l'\omterm{def:g10:universal-dna:information}{ADN}, transgène
compris~: la spécialisation ne jette aucun \omterm{def:g10:universal-dna:gene}{gène}.

\textbf{9.} Oui~: les \omterm{def:g10:cells-common-unit:cell}{cellules} germinales portent le \omterm{def:g10:universal-dna:gene}{gène}. Si la souris
en a un exemplaire sur un \omterm{def:g10:universal-dna:information}{chromosome} d'une paire, environ la moitié de
sa descendance en hérite et luit.

\textbf{10.} La \omterm{def:g10:chemistry-of-life:families}{protéine} n'est pas une information~: elle est consommée
et jamais renouvelée, si bien que la lueur s'éteint. Le \omterm{def:g10:universal-dna:gene}{gène} est copié
à chaque division et lu en permanence, si bien que la lueur est
permanente et héréditaire.

\textbf{11.} La fraction \omterm{def:g10:universal-dna:information}{ADN}~; le traitement par une enzyme qui détruit
l'\omterm{def:g10:universal-dna:information}{ADN}.

\textbf{12.} Des bactéries ont capté de l'\omterm{def:g10:universal-dna:information}{ADN} d'une autre souche, l'ont
intégré, ont exprimé et transmis le caractère~: un \omterm{def:g10:universal-dna:gene}{gène} transféré d'un
organisme à un autre, ce que la \omterm{prop:g10:universal-dna:universal}{transgenèse} fait délibérément.

\textbf{13.} A$\,=\,$T et G$\,=\,$C soutiennent l'appariement~; la part
variable d'A+T montre que l'ordre des paires est libre.

\textbf{14.} A et G sont de grandes \omterm{def:g10:universal-dna:nucleotide}{bases} à deux cycles, T et C de
petites \omterm{def:g10:universal-dna:nucleotide}{bases} à un cycle~: une grande appariée à une petite couvre
toujours la même distance, si bien que les deux squelettes restent à
\qty{2}{nm} l'un de l'autre sur toute la molécule. Deux grandes \omterm{def:g10:universal-dna:nucleotide}{bases}
feraient une bosse, deux petites laisseraient un vide.

\textbf{15.} Parce que chaque brin est le complémentaire de l'autre~:
il suffit de les séparer, de bâtir un partenaire sur chacun selon la
règle d'appariement, et deux molécules identiques en résultent.

\textbf{16.} $2 \times \num{2.7e9} \times \qty{0.34}{nm} \approx
\qty{1.8}{m}$.

\textbf{17.} $\qty{1.8}{m} / \qty{6}{\micro m} = \num{3e5}$~: une
compaction d'un facteur trois cent mille.

\textbf{18.} $720 / \num{5.4e9} \approx \num{1.3e-7}$~: environ un
dix-millionième de l'\omterm{def:g10:universal-dna:information}{ADN} de la \omterm{def:g10:cells-common-unit:cell}{cellule}.

\textbf{19.} $\num{2e4} \times \num{3e4} = \num{6e8}$ paires de \omterm{def:g10:universal-dna:nucleotide}{bases},
soit environ 22~\% du génome~; le reste est entre les \omterm{def:g10:universal-dna:gene}{gènes}.

\textbf{20.} Une \omterm{def:g10:cells-common-unit:cell}{cellule} de souris porte environ \qty{1.8}{m} d'\omterm{def:g10:universal-dna:information}{ADN}~;
comme cet \omterm{def:g10:universal-dna:information}{ADN} est bâti et lu exactement comme celui de la méduse,
l'unique \omterm{def:g10:universal-dna:gene}{gène} étranger parmi ses vingt mille est lu comme les autres.
\end{solution}
