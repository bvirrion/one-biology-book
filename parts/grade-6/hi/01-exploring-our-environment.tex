\chapter{अपने पर्यावरण की खोज}\label{ch:g6:exploring-our-environment}

जुलाई की दोपहर स्कूल का अहाता पार करो और दीवार का दक्षिणी मुँह हाथ के
नीचे तपता मिलेगा, डामर के ऊपर हवा थरथराती दिखेगी, और नज़र भर में हरा
सिर्फ़ किसी दरार में उगा एक खरपतवार होगा। अब उत्तरी दीवार के पीछे जाओ:
ठंडी हवा, नम मिट्टी, \omterm{prop:g4:plants-without-seeds:damp}{मॉस}, और किसी पत्थर के नीचे कठ-जूएँ। दो दुनियाएँ,
तीन मीटर के फ़ासले पर। यह साल पर्यावरणों की खोज से शुरू होता है, वैसे
ही जैसे वैज्ञानिक करते हैं --- उन्हें नापकर, और यह पूछकर कि हर \omterm{def:g6:exploring-our-environment:components}{सजीव} ठीक
वहीं क्यों रहता है जहाँ वह रहता है।

\section{पर्यावरण किन चीज़ों से बना है}

\begin{definition}[पर्यावरण के घटक]\label{def:g6:exploring-our-environment:components}
किसी \emph{पर्यावरण}\index{पर्यावरण} में --- चाहे वह अहाता हो, जंगल हो
या तालाब का किनारा --- तीन तरह के घटक होते हैं:
\begin{enumerate}
  \item \emph{सजीव}\index{सजीव}: वहाँ मौजूद \omterm{def:g1:animals-around-us:animal}{जंतु}, \omterm{def:g1:plants-around-us:plant}{पौधे}, कुकुरमुत्ते
        और सूक्ष्म जीवन;
  \item \emph{खनिज पदार्थ}\index{खनिज पदार्थ}, जो \omterm{def:g1:living-or-not:nonliving}{कभी जीवित नहीं} था:
        चट्टान, पत्थर, मिट्टी का खनिज हिस्सा, पानी, हवा;
  \item सजीवों के \emph{निशान और अवशेष}: सूखे पत्ते, गिरी हुई लकड़ी,
        मल, \omterm{def:g1:animals-around-us:covering}{पंख}, \omterm{def:g1:animals-around-us:covering}{खोल} --- यानी
        \cref{def:g1:living-or-not:nonliving} वाला ``पहले जीवित था''
        डिब्बा, जो अब खोजी के लिए सुराग़ों की पेटी है।
\end{enumerate}
\end{definition}

\begin{example}[उत्तरी दीवार के पीछे की सूची]\label{ex:g6:exploring-our-environment:inventory}
ठंडे कोने में दस मिनट की सूची: \omterm{def:g6:exploring-our-environment:components}{सजीव} --- \omterm{prop:g4:plants-without-seeds:damp}{मॉस}, \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न}, कठ-जूएँ, एक घोंघा,
एक कनखजूरा, आइवी; खनिज --- दीवार का पत्थर, नम मिट्टी, एक पोखर; निशान
--- सूखे पत्ते, घोंघे का एक \omterm{def:g1:animals-around-us:covering}{खोल}, मुँडेर पर \omterm{prop:g3:sorting-living-things:groups}{पक्षियों} का मल। तीन सूचियाँ,
और वह कोना किसी वैज्ञानिक के स्थल की तरह दर्ज हो गया।
\end{example}

\begin{omfigure}
\includegraphics[width=0.62\linewidth]{images/book1/ai/g6-env-wallcorner.png}

{\small ठंडा कोना, सूची के बीच: \omterm{prop:g4:plants-without-seeds:damp}{मॉस}, \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न}, पत्थर पर एक घोंघा --- और
उठाई हुई सिल के नीचे कठ-जुओं का \omterm{def:g6:exploring-our-environment:microhabitat}{सूक्ष्म आवास} (उसे वापस रख दो)।}
\end{omfigure}

\begin{remark}[निशान भी आँकड़े हैं]\label{rem:g6:exploring-our-environment:traces}
ज़्यादातर \omterm{def:g1:animals-around-us:animal}{जंतु} छिप जाते हैं या भाग जाते हैं, इसलिए खोजी को अकसर
\emph{निशान} ही पढ़ने पड़ते हैं: कुतरा हुआ हेज़लनट जंगली चूहे का नाम
बताता है, मकड़ी का जाला अपने मालिक का, और कीचड़ में बने पदचिह्न बगुले
का। जो सूची सिर्फ़ उन्हें गिनती है जो देखे जाने भर देर स्थिर रहे, वह आधे
बाशिंदे छोड़ देती है।
\end{remark}

\section{परिस्थितियाँ नापना}

\begin{definition}[पर्यावरणीय परिस्थितियाँ]\label{def:g6:exploring-our-environment:conditions}
किसी \omterm{def:g6:exploring-our-environment:components}{पर्यावरण} की \emph{परिस्थितियाँ}\index{परिस्थितियाँ (पर्यावरणीय)}
उसकी वे भौतिक ख़ासियतें हैं जिन्हें नापा जा सकता है:
\begin{itemize}
  \item \emph{ताप}\index{ताप}, जो थर्मामीटर से डिग्री सेल्सियस में
        नापा जाता है;
  \item \emph{आर्द्रता}\index{आर्द्रता} --- यानी हवा या मिट्टी कितनी
        नम है;
  \item \emph{प्रकाश}\index{प्रकाश (परिस्थिति)} --- पूरी धूप से लेकर
        गहरी छाया तक।
\end{itemize}
परिस्थितियाँ जगह-जगह बदलती हैं --- अकसर कुछ ही मीटरों में तीखे ढंग से
--- और दिन भर तथा साल भर भी।
\end{definition}

\begin{example}[अहाता, नापा हुआ]\label{ex:g6:exploring-our-environment:measured}
जुलाई की दोपहर का सर्वेक्षण, उपकरणों के साथ: दक्षिणी दीवार की \omterm{def:g1:plants-around-us:parts}{जड़} ---
\qty{38}{\degreeCelsius}, सूखी, पूरी धूप; उत्तरी कोना ---
\qty{22}{\degreeCelsius}, नम मिट्टी, गहरी छाया। तीन मीटर का फ़ासला,
सोलह डिग्री का फ़र्क़। शुरू के पैराग्राफ़ की दोनों दुनियाएँ अब
\emph{तालिका की दो पंक्तियाँ} हैं --- और यही प्रगति है, क्योंकि
तालिकाओं की तुलना की जा सकती है।
\end{example}

\begin{proposition}[परिस्थितियाँ बाशिंदे तय करती हैं]\label{prop:g6:exploring-our-environment:decide}
किसी जगह के \omterm{def:g6:exploring-our-environment:components}{सजीव} उसकी \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} से मेल खाते हैं: हर \omterm{def:g5:biodiversity:species}{प्रजाति} सिर्फ़
\omterm{def:g6:exploring-our-environment:conditions}{ताप}, \omterm{def:g6:exploring-our-environment:conditions}{आर्द्रता} और \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} की अपनी परास के भीतर ही फलती-फूलती है। \omterm{prop:g4:plants-without-seeds:damp}{मॉस} और
कठ-जुओं को नमी चाहिए (कठ-जूँ का आवरण, \omterm{prop:g3:sorting-living-things:groups}{कीट} के आवरण के उलट, पानी को निकल
जाने देता है --- वह घंटे भर की धूप में सूखकर मर जाता है); और दरार वाला
खरपतवार सूखा तथा चौंध, दोनों झेल लेता है। जहाँ \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} अलग होती
हैं, वहाँ बाशिंदे भी अलग होते हैं --- \cref{ch:g2:where-animals-live}
के हर \omterm{def:g2:where-animals-live:habitat}{आवास} के पीछे यही नियम है।
\end{proposition}
\admitted

\begin{example}[कठ-जूँ चुनाव-प्रयोग]\label{ex:g6:exploring-our-environment:chooser}
वह मशहूर प्रयोग: एक बंद डिब्बा, जिसका आधा हिस्सा नम काग़ज़ से और आधा
सूखे से ढका हो, और बीच की रेखा पर छोड़े गए दर्जन भर कठ-जूएँ। कुछ ही
मिनटों में लगभग सारे नम तरफ़ आ जाते हैं। अब इसे छायादार आधे बनाम उजाले
वाले आधे के साथ दोहराओ: वे छाया में इकट्ठा होते हैं। कोई उन्हें हाँक
नहीं रहा --- हर \omterm{def:g1:animals-around-us:animal}{जंतु} घूमता और मुड़ता हुआ वहीं जा पहुँचता है जहाँ उसके
शरीर को आराम है। कठ-जुओं का नक़्शा असल में नमी का ही नक़्शा \emph{है}।
\end{example}

\begin{method}[किसी स्थल का सर्वेक्षण]\label{met:g6:exploring-our-environment:survey}
किसी छोटे स्थल का सर्वेक्षण क्षेत्र-वैज्ञानिक की तरह करने के लिए:
\begin{enumerate}
  \item एक छोटा इलाक़ा चुनकर चिह्नित करो --- ठीक से छाना हुआ एक वर्ग
        मीटर सरसरी नज़र डाले गए एक हेक्टेयर से ज़्यादा बताता है;
  \item \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} नापो: \omterm{def:g6:exploring-our-environment:conditions}{ताप}, नमी, \omterm{def:g6:exploring-our-environment:conditions}{प्रकाश} --- हर स्थल पर वही उपकरण,
        वही तरीक़ा
        (\cref{met:g4:what-plants-need:fairtest} वाला न्याय का
        नियम);
  \item तीनों तरह के घटकों की सूची बनाओ --- \omterm{def:g6:exploring-our-environment:components}{सजीव}, \omterm{def:g6:exploring-our-environment:components}{खनिज पदार्थ}, निशान
        --- और ठीक से खोजो: पत्थरों के नीचे (उन्हें वापस रखो:
        \cref{met:g2:caring-for-nature:rules}), दरारों में, \omterm{def:g1:plants-around-us:parts}{पत्तियों}
        की निचली सतह पर;
  \item सब कुछ मौक़े पर ही दर्ज करो --- सूचियाँ, रेखाचित्र, गिनतियाँ;
        याददाश्त कोई दर्ज करने वाला उपकरण नहीं है;
  \item स्थलों की तुलना करो: बाशिंदों के हर फ़र्क़ को \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} के
        किसी फ़र्क़ के साथ जोड़ो।
\end{enumerate}
\end{method}

\begin{remark}[अभिलेख याददाश्त से बेहतर क्यों हैं]\label{rem:g6:exploring-our-environment:records}
भीतर लौटकर याददाश्त पर भरोसा करने वाले दो सर्वेक्षक इसी पर झगड़ेंगे कि
घोंघे तीन थे या पाँच। कॉपी फ़ैसला कर देती है --- या ईमानदारी से मान लेती
है कि घोंघे गिने ही नहीं गए थे। मौक़े पर लिखे गए अभिलेख ही किसी सैर को
आँकड़ों में बदलते हैं; इस पुस्तक का हर विज्ञान उन्हीं पर चलता है।
\end{remark}

\section{एक पर्यावरण, कई सूक्ष्म दुनियाएँ}

\begin{definition}[सूक्ष्म आवास]\label{def:g6:exploring-our-environment:microhabitat}
एक ही \omterm{def:g6:exploring-our-environment:components}{पर्यावरण} के भीतर हर वह छोटी जगह जिसकी अपनी \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} और अपने
बाशिंदे हों --- किसी पत्थर की निचली सतह, दीवार की कोई दरार, बाड़ की
छायादार \omterm{def:g1:plants-around-us:parts}{जड़} --- \emph{सूक्ष्म आवास}\index{सूक्ष्म आवास} कहलाती है।
\Cref{rem:g2:where-animals-live:layers} की जंगल की मंज़िलें सूक्ष्म \omterm{def:g2:where-animals-live:habitat}{आवास}
ही थीं; और सर्वेक्षक के एक वर्ग मीटर में आम तौर पर कई सूक्ष्म \omterm{def:g2:where-animals-live:habitat}{आवास}
होते हैं।
\end{definition}

\begin{example}[पत्थर, ऊपर और नीचे]\label{ex:g6:exploring-our-environment:stone}
ठंडे कोने का एक पत्थर: ऊपर लाइकेन और धूप सेंकती एक मक्खी --- उजाला,
सूखा, गरम। नीचे कठ-जूएँ, एक कनखजूरा, पीली-सी \omterm{def:g1:plants-around-us:parts}{जड़ें} --- अँधेरा, नम, ठंडा।
पाँच सेंटीमीटर की चट्टान से अलग हुए दो पूरे \omterm{def:g6:exploring-our-environment:microhabitat}{सूक्ष्म आवास}, और हर एक में
ठीक वही \omterm{def:g6:exploring-our-environment:components}{सजीव} जिनकी परास उसकी \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} से मेल खाती है।
\end{example}

\begin{exercise}[$\star\star$]\label{exo:g6:exploring-our-environment:10}

\cref{met:g6:exploring-our-environment:survey} की मदद से सर्वेक्षक के
निर्देश फिर से लिखो --- हर चरण पर क्या किया जाना चाहिए था?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:exploring-our-environment:11}
आइवी उत्तरी दीवार पर चढ़ती है, दक्षिणी पर नहीं। इसकी व्याख्या
\cref{prop:g6:exploring-our-environment:decide} के रूप में सुझाओ, और
एक ऐसा प्रेक्षण या माप भी जो उसकी जाँच कर सके।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:exploring-our-environment:12}
कठ-जूँ वाले डिब्बे पर एक विद्यार्थी आपत्ति करता है: ``हो सकता है वे बस
दीवार के पास इकट्ठा हुए हों, और संयोग से नम तरफ़ ज़्यादा दीवार रही
हो।'' इस आपत्ति का उत्तर देने वाला प्रयोग का बेहतर रूप बनाओ। (न्यायी
जाँच के बारे में सोचो।)
\end{exercise}

\section{समस्या: दो नालियाँ}

\begin{problem}[{सप्ताहांत समस्या --- प्रकृतिविद क्लब सड़क किनारे की
दो नालियों का सर्वेक्षण करता है और दो अलग-अलग दुनियाएँ
पाता है}]\label{pb:g6:exploring-our-environment:1}
प्रकृतिविद क्लब एक ही गली के किनारे बनी दो नालियों का सर्वेक्षण करता
है, जो $50$ मीटर की दूरी पर हैं। उत्तरी नाली एक ऊँची बाड़ के उत्तरी किनारे
पर है; दक्षिणी नाली खुले में है और दक्षिण की ओर मुँह किए है। क्लब पूरी तरह
क़ायदे से काम करता है: वही उपकरण, हर स्थल पर वही एक वर्ग मीटर, वही
खोज-समय।

\medskip\noindent\textbf{भाग I --- माप।}
\begin{enumerate}
  \item थर्मामीटर एक ही घंटे में उत्तरी नाली में \qty{19}{\degreeCelsius}
        और दक्षिणी नाली में \qty{31}{\degreeCelsius} दिखाते हैं। इन स्थलों
        की कौन-सी अकेली ख़ासियत इस फ़र्क़ को सबसे अच्छी तरह समझाती
        है?
  \item उत्तरी नाली की मिट्टी उँगलियों से चिपकती है; दक्षिणी नाली की भुरभुराकर
        धूल हो जाती है। दोनों प्रेक्षणों को
        \cref{def:g6:exploring-our-environment:conditions} की
        शब्दावली में बदलो।
  \item क्लब दोपहर में नापता है। एक कारण बताओ कि उत्तरी दक्षिणी नालीोपहर में और
        दक्षिणी नाली शाम छह बजे नापी जाती तो तुलना कम न्यायी क्यों होती।
  \item क्लब दोनों स्थलों पर \emph{वही} एक वर्ग मीटर का चौखटा और वही
        खोज-समय क्यों इस्तेमाल करता है?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- सूचियाँ।}
\begin{enumerate}[resume]
  \item उत्तरी नाली की सूची: \omterm{prop:g4:plants-without-seeds:damp}{मॉस}, \omterm{prop:g4:plants-without-seeds:damp}{फ़र्न}, कठ-जूएँ, घोंघे, एक भेक। दक्षिणी नाली की
        सूची: सूखी घासें, मोम जैसी \omterm{def:g1:plants-around-us:parts}{पत्तियों} वाला एक बड़ा खरपतवार,
        चींटियाँ, धूप सेंकती एक छिपकली, टिड्डे। हर सूची के सदस्यों को
        तीनों तरह के घटकों में छाँटो --- और यह भी लिखो कि सूचियों में
        क्या छूट रहा है
        (\cref{def:g6:exploring-our-environment:components} जाँचो)।
  \item कठ-जुओं, भेक और छिपकली में से हर एक के लिए बताओ कि किस नाली की
        \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} उसकी ज़रूरतों से मेल खाती हैं, और क्यों
        (\cref{prop:g6:exploring-our-environment:decide} और हर \omterm{def:g1:animals-around-us:animal}{जंतु} के
        बारे में अपनी जानकारी काम में लो)।
  \item दक्षिणी नाली में क्लब को घोंघे का एक \omterm{def:g1:animals-around-us:covering}{खोल} मिलता है, पर घोंघा नहीं, और
        वह पूछता है कि क्या वहाँ घोंघे रहते हैं। ऐसा निशान कौन-सी दो
        ईमानदार संभावनाएँ खुली छोड़ देता है?
  \item उत्तरी नाली में एक चपटे पत्थर के नीचे: आठ कठ-जूएँ, एक कनखजूरा,
        सफ़ेद \omterm{def:g1:plants-around-us:parts}{जड़ें}। इस \omterm{def:g6:exploring-our-environment:microhabitat}{सूक्ष्म आवास} की \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} बताओ, और वह नियम
        भी जिसने इन बाशिंदों को वहाँ पहुँचाया।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- व्याख्या।}
\begin{enumerate}[resume]
  \item बाशिंदों के हर फ़र्क़ को \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} के किसी फ़र्क़ से जोड़ो:
        ``उत्तरी में \omterm{prop:g4:plants-without-seeds:damp}{मॉस} और भेक हैं क्योंकि \dots; दक्षिणी में छिपकलियाँ और मोम
        जैसी \omterm{def:g1:plants-around-us:parts}{पत्तियों} वाले खरपतवार हैं क्योंकि \dots'' पूरा करो।
  \item बाड़ को उखाड़ दिए जाने की योजना है। दो गर्मियों बाद उत्तरी नाली की
        \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} की भविष्यवाणी करो --- और सवाल 9 वाली जोड़ी की
        मदद से उसके बाशिंदों की भी।
  \item एक सदस्य भविष्यवाणी की ठीक से जाँच का प्रस्ताव रखता है: बाड़
        हट जाने के बाद दोनों नालियों का फिर से सर्वेक्षण। इस दोबारा
        सर्वेक्षण के गिनती में आने के लिए \emph{वही} स्थल, उपकरण और
        मौसम क्यों ज़रूरी हैं?
  \item क्लब की रिपोर्ट इस वाक्य पर ख़त्म होती है: ``बाशिंदे
        \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} का नक़्शा बना देते हैं।'' इस वाक्य को कठ-जूँ वाले
        चुनाव-प्रयोग (\cref{ex:g6:exploring-our-environment:chooser})
        से समझाओ, और बताओ कि ऐसा नक़्शा अपने आप बनने के लिए किसी हाँकने
        की ज़रूरत क्यों नहीं।
\end{enumerate}
\end{problem}
