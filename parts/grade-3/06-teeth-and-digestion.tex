\chapter{Teeth and Digestion}\label{ch:g3:teeth}

\omterm{def:g2:food-energy:food}{Food} must be broken into tiny pieces your \omterm{def:g1:my-body:body}{body} can use. The journey
starts in the \omterm{def:g3:teeth:path}{mouth} with teeth, then continues through a long tube
called the digestive system.

\section{Teeth}

\begin{definition}[Tooth]\label{def:g3:teeth:tooth}
A \emph{tooth}\index{tooth} is a hard structure in the \omterm{def:g3:teeth:path}{mouth} used
to bite and chew. Different teeth have different shapes for
different jobs.
\end{definition}

\begin{definition}[Tooth types]\label{def:g3:teeth:types}
\begin{itemize}
  \item \emph{incisors}\index{incisor}: front teeth --- cut \omterm{def:g2:food-energy:food}{food};
  \item \emph{canines}\index{canine (tooth)}: pointed --- tear \omterm{def:g2:food-energy:food}{food};
  \item \emph{molars}\index{molar}: flat and wide --- grind \omterm{def:g2:food-energy:food}{food}.
\end{itemize}
\end{definition}

\begin{method}[Care for teeth]\label{met:g3:teeth:care}
\begin{enumerate}
  \item Brush morning and night with a soft brush.
  \item Clean all surfaces gently.
  \item Limit frequent sugary snacks and drinks.
  \item Visit a dentist when \omterm{def:g3:life-cycles:butterfly}{adults} arrange it.
\end{enumerate}
\end{method}

\section{Digestion}

\begin{definition}[Digestion]\label{def:g3:teeth:digestion}
\emph{Digestion}\index{digestion} is the breaking down of \omterm{def:g2:food-energy:food}{food} into
small pieces the \omterm{def:g1:my-body:body}{body} can absorb and use for \omterm{def:g2:food-energy:energy}{energy} and \omterm{def:g2:growing:growth}{growth}.
\end{definition}

\omimg{digestive-system.jpg}{0.52\linewidth}{\omterm{def:g3:teeth:path}{Mouth}, \omterm{def:g3:teeth:path}{gullet}, \omterm{def:g3:teeth:path}{stomach}
and \omterm{def:g3:teeth:path}{intestines} --- the \omterm{def:g2:food-energy:food}{food} pathway (simplified).}

\begin{definition}[Main stops]\label{def:g3:teeth:path}
\begin{itemize}
  \item \emph{mouth}\index{mouth}: teeth chew; saliva moistens;
  \item \emph{gullet}\index{gullet} (oesophagus): tube to the
        stomach;
  \item \emph{stomach}\index{stomach}: churns \omterm{def:g2:food-energy:food}{food} with juices;
  \item \emph{intestines}\index{intestine}: absorb useful parts;
        waste \omterm{def:g2:plant-parts:main}{leaves} the \omterm{def:g1:my-body:body}{body}.
\end{itemize}
\end{definition}

\begin{example}[A sandwich]\label{ex:g3:teeth:sandwich}
You bite with \omterm{def:g3:teeth:types}{incisors}, chew with \omterm{def:g3:teeth:types}{molars}, swallow, and the \omterm{def:g3:teeth:path}{stomach}
and \omterm{def:g3:teeth:path}{intestines} finish the work. Hours later, unused waste \omterm{def:g2:plant-parts:main}{leaves}
the \omterm{def:g1:my-body:body}{body}.
\end{example}
