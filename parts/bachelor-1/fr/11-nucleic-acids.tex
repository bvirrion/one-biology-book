\chapter{Nucléotides et acides nucléiques}\label{ch:b1:nucleic-acids}

Versez de l'éthanol froid sur une solution de \omterm{def:b1:cell-unit-of-life:cell}{cellules} broyées et un
nuage blanc de fibres apparaît, que l'on peut enrouler sur une baguette
de verre~: l'\omterm{def:b1:nucleic-acids:chain}{ADN}, la molécule qui porte chaque instruction de
l'\omterm{def:b1:organism-environment:organism}{organisme}, visible à l'œil nu. Deux mètres en sont repliés dans chaque
noyau humain~; le tout est écrit dans un alphabet de quatre lettres, lu
dans un seul sens, et copié en séparant deux brins qui s'ajustent l'un à
l'autre comme une main et son gant. Ce chapitre décrit les \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides}
--- lettres, \omterm{def:b1:water-small-molecules:atp}{monnaie énergétique} et coenzymes à la fois --- les chaînes
qu'ils forment, la \omterm{thm:b1:nucleic-acids:helix}{double hélice} et les faits qui l'établissent, ce qui
se passe quand on la fond et qu'on la réassocie, et les \omterm{def:b1:nucleic-acids:chain}{ARN} qui portent
son message dans la \omterm{def:b1:cell-unit-of-life:cell}{cellule}.

\section{Les nucléotides}

\begin{definition}[Nucléotide]\label{def:b1:nucleic-acids:nucleotide}
Un \emph{nucléotide}\index{nucleotide@nucléotide} est fait de trois parties~: une
\emph{base azotée}\index{base azotee@base azotée} --- une
\emph{purine}\index{purine} (adénine A, guanine G~: deux cycles accolés)
ou une \emph{pyrimidine}\index{pyrimidine} (cytosine C, thymine T,
uracile U~: un cycle)~; un sucre à cinq carbones, le \emph{ribose} dans
l'\omterm{def:b1:nucleic-acids:chain}{ARN} ou le \emph{2$'$-désoxyribose} dans l'\omterm{def:b1:nucleic-acids:chain}{ADN} (privé de l'hydroxyle
porté par le carbone $2'$)~; et un à trois groupes \emph{phosphate} sur
le carbone $5'$ du sucre. La base est fixée au carbone $1'$~; les
carbones du sucre sont numérotés avec des primes pour les distinguer de
ceux de la base. Base plus sucre font un \emph{nucléoside} (adénosine,
guanosine, cytidine, thymidine, uridine)~; avec des phosphates, c'est un
nucléotide (AMP, ADP, \omterm{def:b1:water-small-molecules:atp}{ATP} et leurs semblables).
\end{definition}

\begin{proposition}[Les nucléotides remplissent trois offices]\label{prop:b1:nucleic-acids:threejobs}
Outre qu'ils sont les monomères de l'\omterm{def:b1:nucleic-acids:chain}{ADN} et de l'\omterm{def:b1:nucleic-acids:chain}{ARN}, les \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides}
sont la \emph{\omterm{def:b1:water-small-molecules:atp}{monnaie énergétique}} de la \omterm{def:b1:cell-unit-of-life:cell}{cellule} --- l'\omterm{def:b1:water-small-molecules:atp}{ATP} et le GTP,
dont les \omterm{def:b1:water-small-molecules:atp}{liaisons phosphoanhydride}
(\cref{ch:b1:water-small-molecules}) paient les synthèses, les
transports et les mouvements~; ses \emph{coenzymes} --- $\mathrm{NAD^+}$,
$\mathrm{NADP^+}$ et FAD, qui transportent les électrons dans la
respiration et la photosynthèse
(\cref{ch:b1:respiration-fermentation}), et le coenzyme A, qui
transporte des groupes acyles~; et ses \emph{messagers} --- l'AMP
cyclique, qui relaie les signaux hormonaux à l'intérieur des \omterm{def:b1:cell-unit-of-life:cell}{cellules}.
Tous sont des \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} à adénine ou bâtis sur eux~: la même molécule
qui épelle le message alimente son exécution.
\end{proposition}

\section{La chaîne polynucléotidique}

\begin{definition}[Liaison phosphodiester, polarité]\label{def:b1:nucleic-acids:chain}
Les \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} sont unis en un \emph{polynucléotide}\index{polynucleotide@polynucléotide}
par des \emph{liaisons phosphodiester}\index{liaison phosphodiester}~: le
phosphate porté par le carbone $5'$ d'un \omterm{def:b1:nucleic-acids:nucleotide}{nucléotide} est lié à
l'hydroxyle $3'$ du suivant. La chaîne possède donc un
\emph{squelette} sucre--phosphate, uniforme et chargé négativement (une
charge par phosphate), d'où les bases font saillie en une séquence~; et
elle possède un sens~: une \emph{extrémité $5'$}\index{extremite 5'@extrémité $5'$}
portant un phosphate libre et une \emph{extrémité
$3'$}\index{extremite 3'@extrémité $3'$} portant un hydroxyle libre. Les
séquences s'écrivent et se lisent de $5' \to 3'$, le sens dans lequel
les chaînes sont synthétisées. L'\emph{ADN}\index{ADN} (acide
désoxyribonucléique) emploie le désoxyribose et les bases A, G, C, T~;
l'\emph{ARN}\index{ARN} (acide ribonucléique) emploie le ribose et A, G,
C, U.
\end{definition}

\begin{example}[L'ARN est fragile, l'ADN est durable]\label{ex:b1:nucleic-acids:fragile}
L'hydroxyle $2'$ du ribose peut attaquer la \omterm{def:b1:nucleic-acids:chain}{liaison phosphodiester}
voisine~: en milieu légèrement alcalin, l'\omterm{def:b1:nucleic-acids:chain}{ARN} est hydrolysé en
\omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} en quelques heures, tandis que l'\omterm{def:b1:nucleic-acids:chain}{ADN}, dépourvu de cet
hydroxyle, survit à une ébullition en milieu alcalin et a été retrouvé
intact dans des os vieux de dizaines de milliers d'années. L'uracile de
l'\omterm{def:b1:nucleic-acids:chain}{ARN} est une thymine sans son groupe méthyle~; l'\omterm{def:b1:nucleic-acids:chain}{ADN} emploie la thymine
pour que la cytosine, qui perd lentement son groupe aminé et devient de
l'uracile, puisse être reconnue comme lésée et réparée --- un uracile
dans l'\omterm{def:b1:nucleic-acids:chain}{ADN} est toujours une erreur.
\end{example}

\section{La double hélice}

\begin{proposition}[Les règles de Chargaff]\label{prop:b1:nucleic-acids:chargaff}
\index{regles de Chargaff@règles de Chargaff}
Dans l'\omterm{def:b1:nucleic-acids:chain}{ADN} de n'importe quel \omterm{def:b1:organism-environment:organism}{organisme}, la quantité d'adénine égale
celle de thymine et la quantité de guanine celle de cytosine ($A = T$,
$G = C$, donc \omterm{def:b1:nucleic-acids:nucleotide}{purines} $=$ \omterm{def:b1:nucleic-acids:nucleotide}{pyrimidines}), tandis que le rapport $(G + C)/(A + T)$
varie d'une espèce à l'autre (de \qty{25}{\%} à \qty{75}{\%} de
$G + C$).
\end{proposition}
\begin{proof}[Faits expérimentaux]
\omterm{prop:b1:nucleic-acids:chargaff}{Chargaff} (1950) hydrolysa l'\omterm{def:b1:nucleic-acids:chain}{ADN} de nombreux \omterm{def:b1:organism-environment:organism}{organismes} et sépara les
bases par chromatographie, en dosant chacune par son absorption dans
l'ultraviolet. Les égalités furent vérifiées aux erreurs d'expérience
près pour tous les \omterm{def:b1:nucleic-acids:chain}{ADN}~; la teneur en $G + C$, elle, ne l'était pas, et
elle était la même dans tous les \omterm{def:b1:body-plans-tissues:tissue}{tissus} d'un même \omterm{def:b1:organism-environment:organism}{organisme}.
\end{proof}

\begin{theorem}[La double hélice de Watson et Crick]\label{thm:b1:nucleic-acids:helix}
\index{double helice@double hélice}\index{paire de bases}\index{antiparallele@antiparallèle}
L'\omterm{def:b1:nucleic-acids:chain}{ADN} est fait de deux chaînes polynucléotidiques enroulées autour d'un
axe commun en une \emph{\omterm{thm:b1:nucleic-acids:helix}{double hélice}} droite~: les squelettes
sucre--phosphate à l'extérieur, les bases à l'intérieur, planes et
empilées perpendiculairement à l'axe. Les chaînes sont
\emph{antiparallèles} (l'une va de $5' \to 3'$ vers le haut, l'autre
vers le bas) et \emph{complémentaires}~: face à chaque A se trouve un T,
tenu par deux \omterm{def:b1:water-small-molecules:water}{liaisons hydrogène}, et face à chaque G un C, tenu par
trois~; chaque \emph{\omterm{thm:b1:nucleic-acids:helix}{paire de bases}} a la même largeur, de sorte que
l'hélice est uniforme quelle que soit la séquence. Dans la forme B,
la plus courante, l'hélice fait \qty{2}{nm} de diamètre, s'élève de
\qty{0.34}{nm} par \omterm{thm:b1:nucleic-acids:helix}{paire de bases} et de \qty{3.4}{nm} par tour de dix
paires, et présente un large \emph{grand sillon} et un étroit
\emph{petit sillon} où les bords des bases sont exposés aux protéines.
\end{theorem}
\begin{proof}[Faits expérimentaux]
Les clichés de diffraction des rayons X de fibres d'\omterm{def:b1:nucleic-acids:chain}{ADN} obtenus par
Franklin (1952) montraient la croix caractéristique d'une hélice, avec
une répétition de \qty{3.4}{nm} le long de l'axe, une réflexion intense
à \qty{0.34}{nm} (unités planes empilées) et un diamètre de
\qty{2}{nm}~; les phosphates, atomes les plus lourds, devaient être à
l'extérieur. Watson et Crick (1953) trouvèrent qu'une \omterm{def:b1:nucleic-acids:nucleotide}{purine} appariée à
une \omterm{def:b1:nucleic-acids:nucleotide}{pyrimidine} --- et seulement A avec T et G avec C --- donne des
paires de largeur égale s'ajustant au diamètre de \qty{2}{nm}, ce qui
expliquait les \omterm{prop:b1:nucleic-acids:chargaff}{règles de Chargaff}, et que deux chaînes antiparallèles
ainsi appariées bâtissent une hélice régulière aux dimensions de la
fibre. Le modèle prédisait que chaque chaîne est une matrice pour
l'autre~; la démonstration de la réplication semi-conservative par
Meselson et Stahl (rappelée du volume du lycée, et reprise au
\cref{ch:b1:replication-mitosis}) l'a confirmé.
\end{proof}

\begin{example}[Lire un brin]\label{ex:b1:nucleic-acids:complement}
Si un brin se lit $5'$-ATGGCATTC-$3'$, son partenaire, écrit
$5' \to 3'$, est $5'$-GAATGCCAT-$3'$~: complémenter chaque base, puis
inverser l'ordre. Neuf paires, \qty{3.1}{nm} d'hélice, vingt-deux
\omterm{def:b1:water-small-molecules:water}{liaisons hydrogène} (cinq G--C, quatre A--T). Un chromosome humain de
\num{2.5e8} paires mesure \qty{8.5}{cm} sous cette forme et doit être
replié cent mille fois pour tenir dans un noyau (\cref{ch:b1:genomes}).
\end{example}

\begin{proposition}[Ce qui tient l'hélice]\label{prop:b1:nucleic-acids:stability}
Les \omterm{def:b1:water-small-molecules:water}{liaisons hydrogène} entre bases appariées donnent la spécificité~;
l'\emph{empilement}\index{empilement des bases} des bases planes les
unes sur les autres --- contacts de \omterm{def:b1:water-small-molecules:bonds}{van der Waals} et \omterm{prop:b1:water-small-molecules:properties}{effet hydrophobe}
qui garde les bases hors de l'eau --- donne l'essentiel de la
stabilité~; les squelettes chargés négativement se repoussent, et
l'hélice n'est stable qu'en présence de cations (les $\mathrm{Mg^{2+}}$
et $\mathrm{Na^+}$ physiologiques, ou les protéines basiques de la
chromatine) qui écrantent la charge. Une paire G--C, avec trois \omterm{def:b1:water-small-molecules:water}{liaisons
hydrogène} et un \omterm{prop:b1:nucleic-acids:stability}{empilement} plus fort, tient mieux qu'une paire A--T.
\end{proposition}

\section{Fusion et hybridation}

\begin{definition}[Dénaturation, température de fusion]\label{def:b1:nucleic-acids:melting}
Le chauffage, ou un pH extrême, sépare les deux brins de l'\omterm{def:b1:nucleic-acids:chain}{ADN}~: c'est
la \emph{dénaturation}\index{denaturation (ADN)@dénaturation (ADN)} (ou fusion), une
transition coopérative sur quelques degrés, dont le milieu est la
\emph{température de fusion}\index{temperature de fusion@température de fusion} $T_m$. Elle
s'accompagne de l'\emph{effet hyperchrome}~: les brins simples absorbent
environ \qty{40}{\%} de lumière ultraviolette de plus à \qty{260}{nm}
que les mêmes bases empilées dans une hélice. $T_m$ croît avec la teneur
en $G + C$ (\qty{0.4}{\celsius} par pour cent), avec la concentration en
sel et avec la longueur~; dans un sel à \qty{0.15}{mol/L}, un long \omterm{def:b1:nucleic-acids:chain}{ADN} à
\qty{50}{\%} de $G + C$ fond vers \qty{90}{\celsius}. Refroidis
lentement, les brins retrouvent leurs complémentaires et reforment
l'hélice~: c'est la \emph{renaturation}, ou l'\emph{hybridation}\index{hybridation}
lorsque les brins viennent de sources différentes.
\end{definition}

\begin{method}[Doser les acides nucléiques par absorbance]\label{met:b1:nucleic-acids:absorbance}
\begin{enumerate}
  \item Les bases absorbent l'ultraviolet avec un maximum à
        \qty{260}{nm}. Dans une cuve de \qty{1}{cm}, une absorbance de
        $1.0$ correspond à \qty{50}{\micro g/mL} d'\omterm{def:b1:nucleic-acids:chain}{ADN} double brin
        (\qty{40}{\micro g/mL} d'\omterm{def:b1:nucleic-acids:chain}{ARN}, \qty{33}{\micro g/mL} d'\omterm{def:b1:nucleic-acids:chain}{ADN}
        simple brin).
  \item Le rapport $A_{260}/A_{280}$ vaut $1.8$ pour de l'\omterm{def:b1:nucleic-acids:chain}{ADN} pur et
        $2.0$ pour de l'\omterm{def:b1:nucleic-acids:chain}{ARN} pur~; les protéines absorbent à
        \qty{280}{nm} et l'abaissent.
  \item Pour suivre la fusion, enregistrer $A_{260}$ pendant un
        chauffage lent~; le milieu de la courbe est $T_m$ et sa raideur
        mesure l'homogénéité de l'échantillon.
  \item Pour détecter une séquence, dénaturer l'échantillon et ajouter
        un brin simple marqué qui lui soit complémentaire (une
        \emph{sonde})~; il ne s'hybride que là où se trouve son
        complémentaire, et le marquage en signale l'endroit --- sur un
        gel, sur un chromosome, dans un \omterm{def:b1:body-plans-tissues:tissue}{tissu}.
\end{enumerate}
\end{method}

{\small \omterm{def:b1:nucleic-acids:chain}{ADN} précipité par de l'éthanol froid à partir d'un lysat

\begin{example}[Une sonde trouve un gène]\label{ex:b1:nucleic-acids:probe}
Une sonde de vingt \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} complémentaire d'un gène, à
\qty{5}{\celsius} sous son propre $T_m$, se lie à son complémentaire
exact et à rien d'autre dans trois milliards de \omterm{thm:b1:nucleic-acids:helix}{paires de bases}~: un
seul mésappariement sur vingt abaisse $T_m$ de plusieurs degrés et
l'hybride mésapparié fond. La spécificité de l'appariement des bases,
multipliée sur vingt positions, est ce sur quoi repose tout test d'\omterm{def:b1:nucleic-acids:chain}{ADN},
de la recherche de paternité à la détection d'un pathogène.
\end{example}

\section{L'ARN}

\begin{definition}[Les ARN]\label{def:b1:nucleic-acids:rnas}
L'\omterm{def:b1:nucleic-acids:chain}{ARN} est le plus souvent simple brin, et se replie sur lui-même partout
où de courtes portions complémentaires le permettent, en épingles, en
boucles et en renflements --- une \emph{structure secondaire} --- qui se
tasse ensuite en une forme. Trois classes portent le flux de
l'information génétique (\cref{ch:b1:gene-expression})~: l'\emph{ARN
messager}\index{ARN messager} (ARNm), copie d'un gène que le ribosome
lit~; l'\emph{ARN de transfert}\index{ARN de transfert} (ARNt), quelque
\qty{75}{} \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} repliés en trèfle, portant un acide aminé à son
extrémité $3'$ et un \emph{anticodon} qui s'apparie au message~;
l'\emph{ARN ribosomique}\index{ARN ribosomique} (ARNr), cœur structural
et catalytique du ribosome, les quatre cinquièmes de l'\omterm{def:b1:nucleic-acids:chain}{ARN} d'une
\omterm{def:b1:cell-unit-of-life:cell}{cellule}. Beaucoup d'autres \omterm{def:b1:nucleic-acids:chain}{ARN} régulent, épissent et guident (volume de
troisième année).
\end{definition}

\begin{remark}[Un ARN peut être un catalyseur]\label{rem:b1:nucleic-acids:ribozyme}
Certains \omterm{def:b1:nucleic-acids:chain}{ARN} sont des enzymes (les \emph{ribozymes})~: la liaison
peptidique de toute protéine est formée par l'\omterm{def:b1:nucleic-acids:rnas}{ARN ribosomique} et non
par une protéine, et certains \omterm{def:b1:nucleic-acids:chain}{ARN} se coupent et s'épissent eux-mêmes.
Une molécule qui porte à la fois l'information et catalyse des réactions
est ce dont un premier système vivant a pu être fait~; le monde
ADN--protéines serait alors une division du travail plus tardive, avec
l'\omterm{def:b1:nucleic-acids:chain}{ARN} maintenu en son centre.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:1}
Nommer les trois parties d'un \omterm{def:b1:nucleic-acids:nucleotide}{nucléotide} et les deux différences entre
un ribonucléotide et un désoxyribonucléotide.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:2}
Écrire le brin complémentaire de $5'$-GGATCCTA-$3'$ dans le sens
$5' \to 3'$, et compter ses \omterm{def:b1:water-small-molecules:water}{liaisons hydrogène}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:3}
Un \omterm{def:b1:nucleic-acids:chain}{ADN} contient \qty{22}{\%} d'adénine. Donner les pourcentages de T, de
G et de C, ainsi que la teneur en $G + C$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:nucleic-acids:4}
D'après la figure de la fusion, quel est le $T_m$ d'un \omterm{def:b1:nucleic-acids:chain}{ADN} à
\qty{50}{\%} de $G + C$ dans les mêmes conditions, et de combien
l'absorbance croît-elle à la fusion~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:5}
Calculer la longueur du chromosome d'\emph{E. coli} ($\num{4.6e6}$
\omterm{thm:b1:nucleic-acids:helix}{paires de bases}) en millimètres, son nombre de tours, et le facteur de
compaction nécessaire pour tenir dans une \omterm{def:b1:cell-unit-of-life:cell}{cellule} de \qty{2}{\micro m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:6}
Expliquer pourquoi les deux brins de l'\omterm{def:b1:nucleic-acids:chain}{ADN} doivent être antiparallèles,
à l'aide de la géométrie du squelette sucre--phosphate et de l'exigence
que toutes les \omterm{thm:b1:nucleic-acids:helix}{paires de bases} aient la même largeur.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:7}
Une solution d'\omterm{def:b1:nucleic-acids:chain}{ADN} a $A_{260} = 0.35$ dans une cuve de \qty{1}{cm} et
$A_{280} = 0.20$. Calculer la concentration et commenter la pureté.
Combien y a-t-il de \omterm{thm:b1:nucleic-acids:helix}{paires de bases} dans \qty{1}{mL}, si la masse
moyenne d'une paire vaut \qty{650}{Da}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:8}
Expliquer l'effet hyperchrome et pourquoi la courbe de fusion d'un \omterm{def:b1:nucleic-acids:chain}{ADN}
pur est raide alors que celle d'un mélange d'\omterm{def:b1:nucleic-acids:chain}{ADN} de nombreuses espèces
est étalée.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:nucleic-acids:9}
Une sonde de vingt \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} a un $T_m = \qty{62}{\celsius}$ avec son
complémentaire exact~; un mésappariement abaisse $T_m$ d'environ
\qty{5}{\celsius}. À quelle température faut-il conduire l'\omterm{def:b1:nucleic-acids:melting}{hybridation}
pour détecter la séquence exacte et rejeter les mésappariements
uniques~? Que se passe-t-il à \qty{45}{\celsius}~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:10}
Les \omterm{prop:b1:nucleic-acids:chargaff}{règles de Chargaff} valent pour l'\omterm{def:b1:nucleic-acids:chain}{ADN} double brin mais non pour l'\omterm{def:b1:nucleic-acids:chain}{ARN}
ni pour l'\omterm{def:b1:nucleic-acids:chain}{ADN} de certains petits virus. Expliquer ce que chaque
exception apprend sur la structure de ces acides nucléiques.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:11}
Deux \omterm{def:b1:water-small-molecules:water}{liaisons hydrogène} tiennent une paire A--T, et pourtant une hélice
de mille paires A--T est stable à \qty{60}{\celsius} alors qu'une paire
isolée ne se forme pas du tout. Expliquer à l'aide des idées du
\cref{ch:b1:water-small-molecules} (liaisons faibles, coopérativité,
\omterm{prop:b1:nucleic-acids:stability}{empilement}), et dire ce que cela implique pour la fidélité d'une sonde.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:nucleic-acids:12}
«~La structure de l'\omterm{def:b1:nucleic-acids:chain}{ADN} est l'explication de l'hérédité.~» Discuter en
un paragraphe~: ce que la \omterm{thm:b1:nucleic-acids:helix}{double hélice} explique immédiatement (la
copie, la mutation comme changement de séquence), ce qu'elle n'explique
pas à elle seule (comment la séquence devient une protéine), et pourquoi
le modèle de Watson et Crick a été accepté avant tout test biochimique.
\end{exercise}

\section{Problème~: le génome d'un virus}

\begin{problem}[{Problème du week-end --- l'ADN du phage $\lambda$
mesuré, pesé, compté, fondu et empaqueté dans sa capside, jusqu'à sa
température de fusion et sa longueur développée}]
\label{pb:b1:nucleic-acids:1}
Le génome du bactériophage $\lambda$ est un unique \omterm{def:b1:nucleic-acids:chain}{ADN} linéaire double
brin de \num{48502} \omterm{thm:b1:nucleic-acids:helix}{paires de bases}, dont la teneur en $G + C$ vaut
\qty{49.9}{\%}. Il est empaqueté dans une capside icosaédrique de
\qty{55}{nm} de diamètre intérieur. Prendre \qty{0.34}{nm} par \omterm{thm:b1:nucleic-acids:helix}{paire de
bases}, \qty{2.0}{nm} pour le diamètre de l'hélice, \qty{650}{Da} par
\omterm{thm:b1:nucleic-acids:helix}{paire de bases}, et $T_m = 81.5 + 16.6\log_{10}[\mathrm{Na^+}]
+ 0.41\,(\%\,G{+}C) - 500/L$ en degrés Celsius, avec $[\mathrm{Na^+}]$
en \unit{mol/L} et $L$ la longueur en \omterm{thm:b1:nucleic-acids:helix}{paires de bases}.

\medskip
\noindent\textbf{Partie I --- Mesuré et pesé.}
\begin{enumerate}
  \item Calculer la longueur développée de la molécule, en micromètres.
  \item Calculer le nombre de tours de l'hélice.
  \item Calculer sa masse molaire, en daltons.
  \item Calculer la masse d'une molécule, en grammes.
  \item Calculer le nombre de groupes phosphate et, de là, la charge
        nette de la molécule.
  \item Calculer le nombre de \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} de chaque base (A, T, G, C).
  \item Calculer le nombre total de \omterm{def:b1:water-small-molecules:water}{liaisons hydrogène} qui tiennent les
        deux brins.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Fondu.}
\begin{enumerate}[resume]
  \item Calculer $T_m$ dans un sodium à \qty{0.1}{mol/L}.
  \item Calculer $T_m$ dans un sodium à \qty{0.01}{mol/L}, et expliquer
        pourquoi le sel stabilise l'hélice.
  \item Le génome de $\lambda$ contient une région de \qty{1000}{}
        paires à \qty{35}{\%} de $G + C$ et une autre à \qty{62}{\%}.
        Calculer le $T_m$ de chaque région dans un sodium à
        \qty{0.1}{mol/L} et décrire l'allure de la courbe de fusion de
        la molécule entière.
  \item Une solution d'\omterm{def:b1:nucleic-acids:chain}{ADN} de $\lambda$ a $A_{260} = 0.50$ avant la
        fusion. Quelle est sa concentration, et que vaudra $A_{260}$
        après fusion complète~?
  \item Combien de molécules de $\lambda$ y a-t-il dans \qty{1}{mL} de
        cette solution~?
  \item Les deux extrémités du génome de $\lambda$ sont des débords
        simple brin de 12 \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides}, complémentaires l'un de l'autre,
        par lesquels la molécule se circularise chez l'hôte. Calculer
        leur $T_m$ (prendre $L = 12$) et dire pourquoi le cercle tient
        malgré tout à \qty{37}{\celsius} dans la \omterm{def:b1:cell-unit-of-life:cell}{cellule}.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Empaqueté.}
\begin{enumerate}[resume]
  \item Calculer le volume de l'\omterm{def:b1:nucleic-acids:chain}{ADN} assimilé à un cylindre.
  \item Calculer le volume intérieur de la capside.
  \item Quelle fraction de la capside l'\omterm{def:b1:nucleic-acids:chain}{ADN} occupe-t-il~? Comparer avec
        l'\omterm{prop:b1:nucleic-acids:stability}{empilement} le plus compact de cylindres parallèles
        (\qty{91}{\%}).
  \item L'\omterm{def:b1:nucleic-acids:chain}{ADN} porte \num{97000} charges négatives dans une sphère de
        \qty{55}{nm}. Expliquer ce qui doit accompagner l'\omterm{def:b1:nucleic-acids:chain}{ADN} dans la
        capside, et pourquoi l'\omterm{def:b1:nucleic-acids:chain}{ADN} empaqueté exerce sur la coque une
        pression (de quelques dizaines d'atmosphères) qui aide à
        l'injecter dans une bactérie.
  \item La molécule doit être courbée en boucles de rayon comparable à
        celui de la capside. L'\omterm{def:b1:nucleic-acids:chain}{ADN} résiste à la courbure en deçà d'un
        rayon d'environ \qty{50}{nm}. Commenter.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Lu.}
\begin{enumerate}[resume]
  \item Un brin du génome commence par $5'$-GGGCGGCGACCT-$3'$. Écrire le
        brin complémentaire de $5' \to 3'$.
  \item Si un gène de \num{1200} \omterm{thm:b1:nucleic-acids:helix}{paires de bases} est transcrit en un
        ARNm, quelle est la longueur de cet ARNm en \omterm{def:b1:nucleic-acids:nucleotide}{nucléotides} et, à
        \qty{330}{Da} par \omterm{def:b1:nucleic-acids:nucleotide}{nucléotide}, sa masse~?
  \item Le génome compte \qty{75}{} gènes pour \num{48502} paires.
        Quelle fraction du génome est codante si le gène moyen fait
        \num{600} paires~? Comparer avec un génome humain
        (\qty{1.5}{\%}).
  \item Le génome est répliqué en \qty{20}{min} à \qty{37}{\celsius} par
        les enzymes de l'hôte. Calculer la vitesse en \omterm{thm:b1:nucleic-acids:helix}{paires de bases}
        par seconde si la réplication démarre à une origine unique et
        progresse dans les deux sens.
  \item Combien de \omterm{def:b1:water-small-molecules:water}{liaisons hydrogène} sont rompues par seconde aux deux
        fourches au cours de la réplication (utiliser la moyenne par
        paire de la question 7)~?
  \item Une mutation change une paire G--C en A--T. De combien le $T_m$
        de la molécule entière varie-t-il~? La fusion pourrait-elle le
        détecter~?
  \item Énoncer le résultat~: la \omterm{def:b1:nucleic-acids:melting}{température de fusion} de l'\omterm{def:b1:nucleic-acids:chain}{ADN} de
        $\lambda$ dans un sodium à \qty{0.1}{mol/L} et sa longueur
        développée, ainsi que la compaction qui le fait tenir dans la
        capside.
\end{enumerate}
\end{problem}
