\chapter{تغذية النبات بالماء والمعادن}\label{ch:b1:plant-water-minerals}

نبتة ذرة في تموز تسحب لترًا من الماء في اليوم من تربة تبدو جافة
لليد، وتسحب معه ثلاثين ميليغرامًا من النتروجين، نتراتٍ، تحتاج إليها
لبناء نمو يوم واحد. وليست لها \omterm{def:b1:membranes-transport:transporters}{مضخة}. فالماء يدخل لأن \omterm{def:b1:flowering-plant-organization:organs}{الجذر} أجف من
التربة، بالمعنى الدقيق في \cref{ch:b1:membranes-transport}؛ والنترات
تدخل لأن \omterm{def:b1:cell-unit-of-life:cell}{خلايا} \omterm{def:b1:flowering-plant-organization:organs}{الجذر} تنفق \omterm{def:b1:water-small-molecules:atp}{ATP} لإدخالها؛ وكلاهما يجب أن يمر بطبقة
واحدة من \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} جدرانها مختومة \omterm{def:b1:lipids:wax}{بالشمع} قبل أن يبلغ \omterm{def:b1:flowering-plant-organization:tissues}{الخشب}. ويصف هذا
الفصل \omterm{def:b1:flowering-plant-organization:organs}{الجذر} بوصفه عضوًا ماصًا، والكمونات المائية التي تسوق الماء من
التربة إلى \omterm{def:b1:flowering-plant-organization:tissues}{الخشب}، والعناصر المعدنية التي يحتاج إليها النبات وكيف
يلتقطها، والحالة الخاصة للنتروجين، والفطريات والبكتيريا التي
تستخدمها أكثر \omterm{def:b1:flowering-plant-organization:organs}{الجذور} لتقوم بجزء من العمل.

\section{الجذر بوصفه عضوًا ماصًا}

\begin{definition}[الأبوبلاست والسمبلاست والبشرة الداخلية]\label{def:b1:plant-water-minerals:pathways}
\index{أبوبلاست}\index{سمبلاست}\index{شريط كاسباري}
يعبر الماء والمنحلات قشرة \omterm{def:b1:flowering-plant-organization:organs}{الجذر} بطريقين: \emph{الأبوبلاست}، وهو
جملة جدر \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} والمسافات بينها المتصلة، ينتقلان فيها دون عبور أي
غشاء؛ و\emph{السمبلاست}، وهو \omterm{def:b1:cell-unit-of-life:cell}{سيتوبلازم} \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} المتصل الموصولة
\omterm{def:b1:eukaryotic-cell:plantcell}{بالروابط السيتوبلازمية} (\cref{ch:b1:eukaryotic-cell})، يُدخل إليه بعبور
غشاء بلازمي واحد. وعند الحد الداخلي للقشرة تسدّ \emph{\omterm{prop:b1:flowering-plant-organization:stemroot}{البشرة
الداخلية}} (\cref{ch:b1:flowering-plant-organization}) الأبوبلاست:
إذ يشرّب جدرَها الشعاعية شريط من السوبرين واللغنين، هو \emph{شريط
كاسباري}، فيكون على كل ما يدخل الأسطوانة الوعائية أن يمر بغشاء \omterm{def:b1:cell-unit-of-life:cell}{خلية}
من \omterm{prop:b1:flowering-plant-organization:stemroot}{البشرة الداخلية}. فنواقل الغشاء إذن هي التي تقرر ما يبلغ \omterm{def:b1:flowering-plant-organization:tissues}{الخشب}،
وتمنع ما حُمِّل في \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} من التسرب راجعًا.
\end{definition}

\begin{proposition}[الضغط الجذري]\label{prop:b1:plant-water-minerals:rootpressure}
\index{ضغط جذري}\index{إدماع}
تضخ \omterm{prop:b1:flowering-plant-organization:stemroot}{البشرة الداخلية} \omterm{def:b1:cell-unit-of-life:cell}{والخلايا} التي داخلها الأيونات إلى \omterm{def:b1:flowering-plant-organization:tissues}{الخشب}، فيهبط
\omterm{def:b1:membranes-transport:osmosis}{الكمون المائي} \omterm{def:b1:flowering-plant-organization:tissues}{للخشب} ويجتذب الماء تناضحيًا من القشرة؛ وحين يتنتح
المجموع الخضري قليلًا (ليلًا، أو في هواء رطب) يتراكم الماء ويُدفع
عصير \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} صاعدًا تحت \emph{\omterm{prop:b1:plant-water-minerals:rootpressure}{ضغط جذري}} موجب مقداره أعشار من
الميغاباسكال --- يكفي لإخراج قطيرات من أطراف \omterm{def:b1:flowering-plant-organization:organs}{أوراق} النباتات الصغيرة
(\emph{الإدماع}) ولإعادة ملء الأوعية المسدودة بالهواء، ولا يكفي
لرفع العصير إلى قمة شجرة. وفي النهار يتولى الأمر سحبُ النتح
(\cref{ch:b1:plant-transport}) ويزول \omterm{prop:b1:plant-water-minerals:rootpressure}{الضغط الجذري}.
\end{proposition}
\begin{proof}[شواهد]
ساق تُقطع قرب الأرض ينضح جذمورها عصيرًا ساعات، ويسجل مانومتر يوصل
به ضغطًا مقداره \qtyrange{0.1}{0.3}{MPa}. والنضاحة أغنى بالأيونات
من محلول التربة، ويتوقف جريانها إذا بُرّدت \omterm{def:b1:flowering-plant-organization:organs}{الجذور} أو سُمّمت بمثبط
للتنفس: فالجريان يسوقه تحميل الأيونات الفاعل، لا أي ضخ للماء.
\end{proof}

\section{الماء: من التربة إلى الخشب}

\begin{proposition}[سلّم الكمون المائي]\label{prop:b1:plant-water-minerals:ladder}
\index{كمون مائي}
ينتقل الماء من التربة إلى \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} لأن \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي}
(\cref{ch:b1:membranes-transport}) يهبط عند كل خطوة:
\begin{center}
\small
\begin{tabular}{lr}
\hline
الحيّز & $\Psi$ (\unit{MPa}) \\
\hline
تربة رطبة & $-0.03$ إلى $-0.3$ \\
تربة عند نقطة الذبول & $-1.5$ \\
\omterm{def:b1:cell-unit-of-life:cell}{خلية} من قشرة \omterm{def:b1:flowering-plant-organization:organs}{الجذر} & $-0.3$ إلى $-0.6$ \\
\omterm{def:b1:flowering-plant-organization:tissues}{خشب} \omterm{def:b1:flowering-plant-organization:organs}{الجذر} & $-0.5$ إلى $-0.8$ \\
\omterm{def:b1:cell-unit-of-life:cell}{خلايا} \omterm{def:b1:flowering-plant-organization:organs}{الورقة} & $-1.0$ إلى $-2.0$ \\
هواء عند رطوبة نسبية \qty{50}{\%} & $-95$ \\
\hline
\end{tabular}
\end{center}
كمون التربة يحدده في أكثره التوتر الذي تمسك به مسامّها الماء
(الخاصية الشعرية)؛ وكمون \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} تحدده منحلاتها وامتلاؤها؛ وكمون
\omterm{def:b1:flowering-plant-organization:tissues}{الخشب} يحدده توتر العمود المتنتح؛ وكمون الهواء تحدده رطوبته،
$\Psi = (RT/V_w)\ln(\text{الرطوبة النسبية})$، وهو هائل حتى في الهواء
الرطب. وأشد الخطوات انحدارًا بفارق كبير هي الأخيرة، من \omterm{def:b1:flowering-plant-organization:organs}{الورقة} إلى
الهواء: فهناك تُنفق قوة السَّوق كلها تقريبًا، ولذلك يتحكم النبات
فيها بالثغور.
\end{proposition}

\begin{method}[قراءة مقطع كمون مائي]\label{met:b1:plant-water-minerals:profile}
\begin{enumerate}
  \item اسرد الحيزات بالترتيب وكمون $\Psi$ لكل منها؛ فالماء لا
        يجري إلا من الأعلى إلى الأخفض، فوجب أن تتناقص المتتالية على
        طول المسار وإلا انعكس الجريان.
  \item اقسم $\Psi$ لكل \omterm{def:b1:cell-unit-of-life:cell}{خلية} إلى $\Psi_s$ (المنحلات، سالب) و
        $\Psi_p$ (\omterm{def:b1:eukaryotic-cell:plantcell}{الامتلاء}، موجب)؛ \omterm{def:b1:cell-unit-of-life:cell}{فخلية} \omterm{def:b1:flowering-plant-organization:organs}{جذر} عند $\Psi = -0.4$ مع
        $\Psi_s = -0.9$ لها $\Psi_p = +0.5$. وماء \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} له
        $\Psi_s \approx 0$ و$\Psi_p$ سالب: فهو تحت توتر.
  \item والجريان عبر خطوة هو فرق $\Psi$ مضروبًا في المواصلة
        المائية للحاجز؛ \omterm{prop:b1:flowering-plant-organization:stemroot}{والبشرة الداخلية} والثغور هما خطوتا المواصلة
        المنخفضة حيث ينظّم النبات.
  \item جفّف التربة يهبط كمونها $\Psi$؛ ومتى بلغ كمون \omterm{def:b1:flowering-plant-organization:organs}{الورقة} توقف
        الجريان وذبل النبات؛ ومتى بلغ $\Psi_p$ في \omterm{def:b1:cell-unit-of-life:cell}{خلايا} \omterm{def:b1:flowering-plant-organization:organs}{الورقة} الصفر
        تدلّت \omterm{def:b1:flowering-plant-organization:organs}{الورقة}. والتعافي يقتضي أن تتغير التربة لا النبات.
\end{enumerate}
\end{method}

\begin{example}[لتر في اليوم]\label{ex:b1:plant-water-minerals:litre}
\omterm{def:b1:flowering-plant-organization:organs}{جذور} نبتة الذرة الدقيقة تعرض نحو \qty{0.2}{m^2} من السطح بمواصلة
مائية مقدارها نحو \qty{2e-7}{m.s^{-1}.MPa^{-1}}؛ وبين تربة عند
\qty{-0.2}{MPa} \omterm{def:b1:flowering-plant-organization:tissues}{وخشب} \omterm{def:b1:flowering-plant-organization:organs}{جذر} عند \qty{-0.6}{MPa} يكون الدفق
$2\times 10^{-7}\times 0.4\times 0.2 = \qty{1.6e-8}{m^3/s}$ --- أي
لتر ونصف في اليوم، تُفقد كلها من \omterm{def:b1:flowering-plant-organization:organs}{الأوراق}. ولا يقوم \omterm{def:b1:flowering-plant-organization:organs}{الجذر} بأي عمل
لتحريكه: فنتح \omterm{def:b1:flowering-plant-organization:organs}{الورقة} هو الذي يضبط التوتر، والتربة تمد بالماء ما دام
كمونها فوق كمون \omterm{def:b1:flowering-plant-organization:organs}{الجذر}.
\end{example}

\section{التغذية المعدنية}

\begin{definition}[العناصر الأساسية]\label{def:b1:plant-water-minerals:elements}
\index{عنصر أساسي}\index{عنصر كبير المقدار}\index{عنصر ضئيل المقدار}
العنصر \emph{أساسي} للنبات إذا لم يستطع النبات إتمام دورة حياته
بدونه ولم يستطع عنصر آخر أن يحل محله. وهي سبعة عشر: الكربون
والهيدروجين والأكسجين من الهواء والماء؛ و\emph{العناصر الكبيرة
المقدار} المأخوذة من التربة بالغرامات لكل كيلوغرام من الكتلة الجافة
--- النتروجين (نتراتٍ أو أمونيوم)، والفوسفور (فوسفاتٍ)،
والبوتاسيوم، والكالسيوم، والمغنيسيوم، والكبريت (كبريتاتٍ)؛
و\emph{العناصر الضئيلة المقدار} اللازمة بالميليغرامات أو أقل ---
الحديد والمنغنيز والزنك والنحاس والبورون والموليبدنوم والكلور
والنيكل. فالنتروجين يبني \omterm{def:b1:proteins:peptide}{البروتينات} والأحماض النووية؛ والفوسفور
\omterm{def:b1:nucleic-acids:nucleotide}{النوكليوتيدات} \omterm{def:b1:membranes-transport:membrane}{والأغشية}؛ والبوتاسيوم كاتيون \omterm{def:b1:cell-unit-of-life:cell}{الخلية} الرئيس وعامل
\omterm{def:b1:membranes-transport:osmosis}{تناضح} الثغور؛ والمغنيسيوم يجلس في \omterm{def:b1:photosynthesis:pigments}{اليخضور}؛ والحديد في السيتوكرومات
والفيريدوكسين؛ والموليبدنوم في \omterm{prop:b1:plant-water-minerals:nitrate}{مختزلة النترات} وفي النتروجيناز.
\end{definition}

\begin{proposition}[كيف وُضعت القائمة]\label{prop:b1:plant-water-minerals:hydroponics}
عُرفت العناصر الأساسية بتنمية النباتات في ماء لا يحوي إلا أملاحًا
معلومة.
\end{proposition}
\begin{proof}[شواهد]
نمّى زاكس وكنوب (في ستينيات القرن التاسع عشر) نباتات حتى النضج في
محاليل من بضعة أملاح معدنية، بلا تربة البتة، فبرهنا أن التربة لا
تمد بشيء غير المعادن والدعامة؛ وحذف ملح واحد في كل مرة أعطى مرضًا
مميزًا لكل عنصر، وإعادته شفت منه. أما العناصر الضئيلة المقدار
فوُجدت لاحقًا، مع تعلّم الكيميائيين تنقية الأملاح: فنبات محروم من
الموليبدنوم، وهو لازم بجزء من عشرة ملايين، يعجز عن اختزال النترات.
والتقنية، وهي الزراعة المائية، تغذّي اليوم البيوت الزجاجية.
\end{proof}

\begin{method}[قراءة عوز]\label{met:b1:plant-water-minerals:deficiency}
\begin{enumerate}
  \item اسأل أين يظهر العرض. فالعناصر التي يستطيع النبات نقلها في
        اللحاء (N وP وK وMg) تُسحب من \omterm{def:b1:flowering-plant-organization:organs}{الأوراق} القديمة لتغذية الفتية:
        فتظهر القديمة العوز أولًا. أما التي لا يستطيع نقلها (Ca
        وFe وB) فتبقى حيث وُضعت: فتظهره \omterm{def:b1:flowering-plant-organization:organs}{الأوراق} الفتية والقمم
        النامية.
  \item اسأل ماذا يفعل العنصر: فبلا نتروجين يقل \omterm{def:b1:photosynthesis:pigments}{اليخضور} \omterm{def:b1:proteins:peptide}{والبروتين}
        --- شحوب وضمور؛ وبلا مغنيسيوم أو حديد يخفق \omterm{def:b1:photosynthesis:pigments}{اليخضور} --- \omterm{def:b1:flowering-plant-organization:organs}{أوراق}
        صفراء بعروق خضراء؛ وبلا بوتاسيوم تحترق حواف \omterm{def:b1:flowering-plant-organization:organs}{الورقة}؛ وبلا
        كالسيوم تموت القمم النامية.
  \item تأكد بإمداد العنصر المشتبه به وحده ومراقبة النمو الجديد.
\end{enumerate}
\end{method}

\begin{proposition}[الالتقاط الفاعل للأيونات]\label{prop:b1:plant-water-minerals:uptake}
\index{مضخة بروتونات}
يحمل محلول التربة الأيونات بالميكرومولات إلى بضعة مليمولات في
اللتر؛ \omterm{def:b1:cell-unit-of-life:cell}{وخلايا} \omterm{def:b1:flowering-plant-organization:organs}{الجذر} تحمل البوتاسيوم عند \qty{100}{mmol/L}،
والفوسفات عند \qty{10}{}، والنترات عند عدة مليمولات. فالالتقاط إذن
صعودي في أكثره ويُدفع ثمنه من \omterm{def:b1:water-small-molecules:atp}{ATP}، لكن بطريق غير مباشر: إذ تصدّر
\emph{\omterm{prop:b1:plant-water-minerals:uptake}{مضخة بروتونات}} في \omterm{def:b1:membranes-transport:membrane}{الغشاء البلازمي}
(\cref{ch:b1:membranes-transport}) أيونات $\mathrm{H^+}$، فتجعل
الخارج حمضيًا والداخل سالبًا بمقدار
\qtyrange{-120}{-200}{mV}؛ فتدخل الكاتيونات مثل $\mathrm{K^+}$
عندئذ عبر قنوات مع التدرج الكهربائي، وتدخل الأنيونات
($\mathrm{NO_3^-}$ و$\mathrm{H_2PO_4^-}$ و$\mathrm{SO_4^{2-}}$)
بنواقل مشتركة تمتطي البروتونات راجعة. \omterm{def:b1:membranes-transport:transporters}{والنواقل} انتقائية: فالبوتاسيوم
يُلتقط أسهل من الصوديوم بمئة مرة مع تساوي الشحنة وتقارب الحجم،
ويُبقي النبات الصوديوم خارجًا حتى في تربة مالحة. وتنفق \omterm{def:b1:flowering-plant-organization:organs}{الجذور} ثلث
تنفس النبات على هذا النقل.
\end{proposition}

\section{النتروجين}

\begin{proposition}[من النترات إلى الحمض الأميني]\label{prop:b1:plant-water-minerals:nitrate}
\index{مختزلة النترات}\index{تمثيل النتروجين}
تأخذ أكثر النباتات النتروجين نتراتٍ. ففي \omterm{def:b1:eukaryotic-cell:organelle}{العصارة الخلوية} تختزلها
\emph{\omterm{prop:b1:plant-water-minerals:nitrate}{مختزلة النترات}}، وهي \omterm{def:b1:enzymes:enzyme}{إنزيم} موليبدني، إلى نتريت بإلكترونات من
NADH؛ وفي \omterm{def:b1:eukaryotic-cell:plastid}{البلاستيدة} تختزل \emph{مختزلة النتريت} النتريتَ إلى
أمونيوم بإلكترونات من الفيريدوكسين؛ ويُثبَّت الأمونيوم فورًا في
الغلوتامين \omterm{def:b1:biosyntheses-integration:nitrogen}{والغلوتامات} (\cref{ch:b1:biosyntheses-integration})
--- فالأمونيوم نفسه سام ولا يتراكم أبدًا. والاختزال كله يكلف ثمانية
إلكترونات لكل نتروجين، أي عُشر دفق إلكترونات التركيب الضوئي في
\omterm{def:b1:flowering-plant-organization:organs}{ورقة} على تربة غنية بالنترات، \omterm{def:b1:flowering-plant-organization:organs}{والورقة} تقوم بأكثره في الضوء، من
فيريدوكسين \omterm{def:b1:photosynthesis:chloroplast}{البلاستيدة الخضراء} مباشرة. أما الأمونيوم، حيث تعرضه
التربة، فيُلتقط مباشرة ويتخطى هذه الكلفة.
\end{proposition}

\begin{definition}[تثبيت النتروجين والعقيدات]\label{def:b1:plant-water-minerals:nodules}
\index{تثبيت النتروجين}\index{عقيدة جذرية}\index{Rhizobium}
لا يستطيع أي حقيقي نواة استعمال $\mathrm{N_2}$ الهواء. وبعض
البكتيريا يستطيعه: إذ يكسر \emph{النتروجيناز} الرابطة الثلاثية في
$\mathrm{N_2}$ ويختزله إلى نشادرين، بكلفة ستة عشر \omterm{def:b1:water-small-molecules:atp}{ATP} وثمانية
إلكترونات لكل جزيء، ولا يفعل ذلك إلا حيث يُستبعد الأكسجين.
والبقوليات (البازلاء والفول والبرسيم) تؤوي مثل هذه البكتيريا
(\emph{Rhizobium}) في \emph{عقيدات جذرية}: فيبني النبات حجرة،
ويطعم البكتيريا سكرًا، ويبقي الأكسجين منخفضًا \omterm{def:b1:proteins:peptide}{ببروتين} أحمر، هو
الليغهيموغلوبين، يوصله إلى تنفس البكتيريا بتركيز أخفض من أن يؤذي
\omterm{def:b1:enzymes:enzyme}{الإنزيم}؛ وتردّ البكتيريا أمونيومًا يحوّله النبات أحماضًا أمينية.
وحقل برسيم يثبّت مئة كيلوغرام من النتروجين في الهكتار في السنة،
يدفع النبات مقابلها عُشر تركيبه الضوئي. أما دورة النتروجين كلها
فهي من كتاب السنة الثانية.
\end{definition}

\begin{example}[ثمن النتروجين]\label{ex:b1:plant-water-minerals:priceN}
تحتاج نبتة الذرة إلى نحو \qty{30}{mg} من النتروجين في اليوم؛ ومن
نترات عند \qty{2}{mmol/L} تحصل على ذلك في لتر الماء الذي تنتحه،
وتنفق بضع مئات من المليمولات من الإلكترونات على اختزاله. أما نبتة
برسيم تثبّت المقدار نفسه فتدفع \qty{16}{} \omterm{def:b1:water-small-molecules:atp}{ATP} لكل
$\mathrm{N_2}$ زائد المرجع --- أي نحو \qty{6}{g} من السكر لكل
غرام من النتروجين، أي نحو \qty{0.2}{g} في اليوم، وهو عُشر إنتاجها
--- وتستطيع أن تنمو على تربة لا نترات فيها البتة. وقد ناوب
المزارعون البقوليات مع الحبوب ألفي سنة لنقل ذلك النتروجين من حقل
إلى الذي يليه.
\end{example}

\section{شركاء في التربة}

\begin{definition}[الميكوريزا]\label{def:b1:plant-water-minerals:mycorrhiza}
\index{ميكوريزا}
\emph{الميكوريزا} تشارك بين \omterm{def:b1:flowering-plant-organization:organs}{جذر} وفطر، توجد في تسعة نباتات من كل
عشرة. وفي أشيع أشكالها تنمو خيوط الفطر داخل \omterm{def:b1:cell-unit-of-life:cell}{خلايا} القشرة وتتفرع
فيها \emph{شجيرات}، وهي \omterm{prop:b1:organism-environment:surfaces}{سطح التبادل}، بينما تستكشف خارج \omterm{def:b1:flowering-plant-organization:organs}{الجذر} شبكة
من الخيوط قطرها بضعة ميكرومترات التربةَ لأمتار. فيوصل الفطر
الفوسفات وأيونات أخرى ضعيفة الحركة، ويوصل الماء، مما لم تكن أشعار
\omterm{def:b1:flowering-plant-organization:organs}{الجذر} نفسها لتبلغه؛ ويوصل النبات السكر، عُشر تركيبه الضوئي أو
أكثر. ولأشجار الغابات المعتدلة شكل ثانٍ يغلّف فيه الفطر أطراف
\omterm{def:b1:flowering-plant-organization:organs}{الجذور} ويتخلل بين \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}؛ وأجسامه الثمرية هي فطور أرض الغابة.
\end{definition}
\begin{proof}[شواهد]
البادرات المنمّاة في تربة معقّمة تنمو نموًا ضعيفًا وتلتقط فوسفاتًا
قليلًا؛ فإذا لُقّحت بأبواغ الفطر نمت أكبر بعدة أمثال. والفوسفات
المشع الموضوع في تربة بعيدة عن متناول \omterm{def:b1:flowering-plant-organization:organs}{الجذور} لا يظهر في النبات إلا
إذا وصلت الخيوطُ بين الاثنين، والكربون الموسوم المطعَم \omterm{def:b1:flowering-plant-organization:organs}{للأوراق} يظهر
في الخيوط في غضون يوم. وقطع الخيوط بشبكة دقيقة تدع المنحلات تمر ولا
تدع الفطر يلغي الكسب.
\end{proof}

\includegraphics[width=0.52\linewidth]{images/book3/ai/b3-23-mycorrhiza.png}

{\small \omterm{def:b1:flowering-plant-organization:organs}{جذر} مصبوغ لإظهار فطره الميكوريزي: خيوط تجري بين \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}
وتتفرع شجيرات داخلها --- أي سطح ماص ثانٍ، يمدّه النبات بالسكر.}
\end{omfigure}

\begin{example}[لماذا يحتاج الفوسفات إلى فطر]\label{ex:b1:plant-water-minerals:phosphate}
يرتبط الفوسفات بحبيبات التربة وينتشر ميليمترًا في اليوم؛ وشعرة
\omterm{def:b1:flowering-plant-organization:organs}{الجذر} تلتقط كل الفوسفات في متناولها في ساعات ثم تنتظر. والخيوط
الفطرية، وهي أدق من شعرة \omterm{def:b1:flowering-plant-organization:organs}{الجذر} بعشر مرات، تبلغ أبعد منها بعشر مرات
لكل غرام من \omterm{def:b1:body-plans-tissues:tissue}{النسيج}، ومتر من الخيط يكلف النبات جزءًا من ألف مما
يكلفه متر من \omterm{def:b1:flowering-plant-organization:organs}{الجذر}. أما النترات، وهي تتحرك بحرية مع ماء التربة،
فتكفيها أشعار \omterm{def:b1:flowering-plant-organization:organs}{الجذر} نفسها، والنباتات على الترب الغنية بالنترات
تقلل فطرها؛ أما الفوسفات فالفطر امتداد \omterm{def:b1:flowering-plant-organization:organs}{الجذر} إليه.
\end{example}

\section{مسألة: سلّم الكمون المائي في جذر الذرة}

\begin{problem}[{مسألة نهاية الأسبوع --- الماء والنترات من تربة
تموزية إلى نبتة ذرة: الكمونات عند كل خطوة، والدفق عبر الجذور،
والنتروجين الذي يحمله وكلفة اختزاله، وختامًا الدفق المائي اليومي إلى
الجذر}]
\label{pb:b1:plant-water-minerals:1}
نبتة ذرة في تموز: سطح \omterm{def:b1:flowering-plant-organization:organs}{الجذور} الدقيقة \qty{0.2}{m^2}، والمواصلة
المائية \omterm{def:b1:flowering-plant-organization:organs}{للجذر} \qty{2e-7}{m.s^{-1}.MPa^{-1}}؛ ومساحة \omterm{def:b1:flowering-plant-organization:organs}{الأوراق}
\qty{0.5}{m^2}. \omterm{def:b1:membranes-transport:osmosis}{والكمون المائي} للتربة \qty{-0.2}{MPa}؛ \omterm{def:b1:cell-unit-of-life:cell}{وخلايا} قشرة
\omterm{def:b1:flowering-plant-organization:organs}{الجذر} لها $\Psi_s = \qty{-0.9}{MPa}$ و$\Psi_p = \qty{0.5}{MPa}$؛
\omterm{def:b1:flowering-plant-organization:tissues}{وخشب} \omterm{def:b1:flowering-plant-organization:organs}{الجذر} \qty{-0.6}{MPa}؛ \omterm{def:b1:cell-unit-of-life:cell}{وخلايا} \omterm{def:b1:flowering-plant-organization:organs}{الورقة} \qty{-1.3}{MPa}؛ والهواء
عند رطوبة نسبية \qty{50}{\%} ($RT/V_w = \qty{137}{MPa}$). ونترات
التربة \qty{2}{mmol/L}؛ ويحتاج النبات إلى \qty{30}{mg} من
النتروجين في اليوم. واختزال النترات إلى أمونيوم يستهلك 8 إلكترونات
لكل نتروجين؛ والتركيب الضوئي يحرك 4 إلكترونات لكل $\mathrm{CO_2}$
وتثبّت \omterm{def:b1:flowering-plant-organization:organs}{الورقة} \qty{4}{\micro mol} من $\mathrm{CO_2}$ في المتر
المربع في الثانية مدة \qty{12}{h}.

\medskip
\noindent\textbf{الجزء الأول --- السلّم.}
\begin{enumerate}
  \item احسب \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي} \omterm{def:b1:cell-unit-of-life:cell}{لخلايا} قشرة \omterm{def:b1:flowering-plant-organization:organs}{الجذر}.
  \item اكتب المتتالية: تربة، فقشرة، \omterm{def:b1:flowering-plant-organization:tissues}{فخشب}، \omterm{def:b1:flowering-plant-organization:organs}{فورقة}، وتحقق من أن الماء
        يجري إلى الداخل وإلى الأعلى عند كل خطوة.
  \item احسب \omterm{def:b1:membranes-transport:osmosis}{الكمون المائي} للهواء.
  \item احسب الهبوط من التربة إلى \omterm{def:b1:flowering-plant-organization:organs}{الورقة} والهبوط من \omterm{def:b1:flowering-plant-organization:organs}{الورقة} إلى
        الهواء؛ وأي كسر من المجموع هي الخطوة الأخيرة؟
  \item تجف التربة إلى \qty{-0.9}{MPa}. فماذا يحدث للجريان إلى
        \omterm{def:b1:cell-unit-of-life:cell}{خلايا} القشرة، ولامتلائها؟
  \item تجف التربة إلى \qty{-1.5}{MPa}. فماذا يحدث \omterm{def:b1:flowering-plant-organization:organs}{للأوراق}، ولماذا
        يعيدها السقي؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الدفق.}
\begin{enumerate}[resume]
  \item احسب الدفق المائي إلى \omterm{def:b1:flowering-plant-organization:organs}{الجذور} (المواصلة $\times$
        السطح $\times$ الفرق بين التربة \omterm{def:b1:flowering-plant-organization:tissues}{وخشب} \omterm{def:b1:flowering-plant-organization:organs}{الجذر}).
  \item حوّله إلى لترات في اليوم.
  \item كله يُفقد بالنتح. احسب النتح لكل متر مربع من الورق في
        الساعة، بالغرامات.
  \item عبّر عن النتح لكل متر مربع من الورق في الثانية بمليمولات
        الماء (\qty{18}{g/mol}).
  \item تنغلق الثغور عند الظهيرة فيرتفع $\Psi$ في \omterm{def:b1:flowering-plant-organization:organs}{الورقة} إلى
        \qty{-0.9}{MPa} ويرتفع في \omterm{def:b1:flowering-plant-organization:tissues}{الخشب} إلى \qty{-0.4}{MPa}.
        أعد حساب الدفق. فماذا فعل النبات، ولماذا؟
  \item لماذا يلتقط \omterm{def:b1:flowering-plant-organization:organs}{الجذر} الماء دون إنفاق طاقة، بينما عليه أن ينفق
        طاقة لالتقاط النترات؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- النتروجين.}
\begin{enumerate}[resume]
  \item احسب النترات المحمولة إلى \omterm{def:b1:flowering-plant-organization:organs}{الجذر} بالدفق المائي في السؤال 8،
        بالمليمولات وبالميليغرامات من النتروجين.
  \item هل تغطي حاجة النبات؟ وماذا يحدث للفائض؟
  \item يراكم \omterm{def:b1:flowering-plant-organization:organs}{الجذر} النترات إلى \qty{5}{mmol/L} في خلاياه. احسب
        \omterm{prop:b1:water-small-molecules:gibbs}{الطاقة الحرة} لنقل مول واحد من \qty{2}{mmol/L} خارجًا إلى
        \qty{5}{mmol/L} داخلًا ضد كمون غشائي \qty{-150}{mV}
        ($RT = \qty{2.48}{kJ/mol}$؛ وهو أنيون يدخل \omterm{def:b1:cell-unit-of-life:cell}{خلية} سالبة).
  \item الالتقاط نقل مشترك مع بروتونين. احسب \omterm{prop:b1:water-small-molecules:gibbs}{الطاقة الحرة} التي
        يحررها بروتونان يدخلان \omterm{def:b1:cell-unit-of-life:cell}{خلية} عند \qty{-150}{mV} من pH 5.5
        إلى pH 7.5، وتحقق من أنها تكفي.
  \item احسب الإلكترونات اللازمة في اليوم لاختزال \qty{30}{mg} من
        النتروجين إلى أمونيوم.
  \item احسب الإلكترونات التي تحركها \omterm{def:b1:flowering-plant-organization:organs}{الورقة} عبر جملتيها الضوئيتين
        في اليوم، والكسر المحوَّل إلى النترات.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- البدائل.}
\begin{enumerate}[resume]
  \item لو عرضت التربة أمونيومًا بدل النترات، فماذا كان النبات
        يوفّر، وأي مشكلة كان التقاط الأمونيوم يخلقها لتوازن pH
        عنده؟
  \item برسيم يثبّت \qty{30}{mg} من النتروجين في اليوم نفسه.
        وبإنفاق \qty{16}{} \omterm{def:b1:water-small-molecules:atp}{ATP} و\qty{8}{} إلكترونات لكل $\mathrm{N_2}$،
        احسب \omterm{def:b1:water-small-molecules:atp}{ATP} الذي ينفقه النتروجيناز في اليوم.
  \item وعند \qty{6}{g} من السكر لكل غرام من النتروجين المثبَّت في
        المجمل، احسب كلفة البرسيم اليومية سكرًا وقارنها بإنتاج
        \qty{3}{g} من السكر في اليوم. أتستحق الصفقة في تربة نتراتها
        \qty{2}{mmol/L}؟ وفي تربة بلا نترات؟
  \item الفوسفات عند \qty{2}{\micro mol/L} في محلول التربة. احسب
        الفوسفات الذي يجلبه الدفق المائي في السؤال 8، وقارنه بحاجة
        قدرها \qty{4}{mg} من الفوسفور في اليوم
        (\qty{31}{g/mol}). فعلى ماذا يجب أن يعتمد النبات؟
  \item فسّر في جملتين كيف يتيح \omterm{def:b1:plant-water-minerals:pathways}{شريط كاسباري} \omterm{def:b1:flowering-plant-organization:organs}{للجذر} أن يبقي النترات
        عند \qty{5}{mmol/L} في خشبه والتربة عند \qty{2}{}.
  \item طافر يفتقر إلى \omterm{def:b1:plant-water-minerals:pathways}{شريط كاسباري}. تنبأ بتركيب عصير خشبه وبسلوكه
        في تربة مالحة.
  \item اذكر النتيجة: الدفق المائي إلى \omterm{def:b1:flowering-plant-organization:organs}{جذر} الذرة في اليوم،
        والنتروجين الذي يوصله، وكسر إلكترونات \omterm{def:b1:flowering-plant-organization:organs}{الورقة} المنفق على
        اختزاله.
\end{enumerate}
\end{problem}
