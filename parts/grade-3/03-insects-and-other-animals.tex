\chapter{Insects and Other Animals}\label{ch:g3:insects}

\omterm{def:g3:insects:insect}{Insects} are everywhere: bees on \omterm{def:g2:plant-parts:main}{flowers}, ants on paths, beetles under
bark. They are \omterm{def:g1:plants-animals:animal}{animals} without backbones, with a special \omterm{def:g1:my-body:body}{body} plan.
This chapter teaches you to spot an \omterm{def:g3:insects:insect}{insect} and compare it with other
small \omterm{def:g1:plants-animals:animal}{animals}.

\section{What is an insect?}

\begin{definition}[Insect]\label{def:g3:insects:insect}
An \emph{insect}\index{insect} is an \omterm{def:g2:animal-groups:invert}{invertebrate} with three \omterm{def:g1:my-body:body}{body}
parts (\omterm{def:g1:my-body:parts}{head}, thorax, abdomen), six \omterm{def:g1:my-body:parts}{legs}, and usually two antennae.
Many have wings.
\end{definition}

\begin{example}[Insects]\label{ex:g3:insects:examples}
Ant, bee, butterfly, fly, beetle, dragonfly, grasshopper. A \omterm{def:g3:insects:compare}{spider}
is \emph{not} an \omterm{def:g3:insects:insect}{insect} --- it has eight \omterm{def:g1:my-body:parts}{legs} and two main \omterm{def:g1:my-body:body}{body}
parts.
\end{example}

\section{Insects and others}

\begin{definition}[Compare]\label{def:g3:insects:compare}
\begin{itemize}
  \item \emph{spiders}\index{spider}: eight \omterm{def:g1:my-body:parts}{legs}; two \omterm{def:g1:my-body:body}{body} parts;
  \item \emph{worms}\index{worm}: long soft \omterm{def:g1:my-body:body}{body}; no \omterm{def:g1:my-body:parts}{legs};
  \item \emph{snails}\index{snail}: soft \omterm{def:g1:my-body:body}{body}, often a shell; one
        large \omterm{def:g1:my-body:parts}{foot}.
\end{itemize}
All are \omterm{def:g2:animal-groups:invert}{invertebrates}, but only \omterm{def:g3:insects:insect}{insects} have the \omterm{def:g1:my-body:parts}{six-leg} plan.
\end{definition}

\begin{method}[Is it an insect?]\label{met:g3:insects:test}
\begin{enumerate}
  \item Count the \omterm{def:g1:my-body:parts}{legs}: six?
  \item Can you see three \omterm{def:g1:my-body:body}{body} parts?
  \item Are there antennae?
  \item If yes to these, it is likely an \omterm{def:g3:insects:insect}{insect}.
\end{enumerate}
\end{method}

\begin{example}[Helpful insects]\label{ex:g3:insects:help}
Bees carry pollen so \omterm{def:g1:plants-animals:plant}{plants} can make \omterm{def:g2:plant-parts:main}{seeds} and \omterm{def:g2:plant-parts:main}{fruit}. Ladybirds eat
pests. Earthworms (not \omterm{def:g3:insects:insect}{insects}) mix soil. Many small \omterm{def:g1:plants-animals:animal}{animals} help
\omterm{def:g2:habitats:habitat}{habitats} work.
\end{example}

\begin{proposition}[Insects matter]\label{prop:g3:insects:matter}
\omterm{def:g3:insects:insect}{Insects} pollinate \omterm{def:g1:plants-animals:plant}{plants}, recycle material and feed \omterm{def:g2:animal-groups:five}{birds} and other
\omterm{def:g1:plants-animals:animal}{animals}. They are a huge part of \omterm{def:g1:plants-animals:animal}{animal} life on land.
\end{proposition}
\admitted

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g3:insects:1}

\begin{exercise}[$\star$]\label{exo:g3:insects:4}

\end{exercise}
