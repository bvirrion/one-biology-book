\section*{Chapter \ref{ch:g4:microbes} --- Microbes Around Us}

\begin{solution}{exo:g4:microbes:1}
A \omterm{def:g4:microbes:def}{microbe} is a \omterm{def:g1:living-nonliving:living}{living thing} too small to see clearly without a
\omterm{prop:g4:cells:building}{microscope} (\omterm{def:g4:microbes:def}{bacteria}, many tiny \omterm{def:g4:microbes:def}{fungi}, and similar \omterm{def:g4:ecosystems:levels}{organisms}).
\end{solution}

\begin{solution}{exo:g4:microbes:2}
Examples: yeast in bread; \omterm{def:g4:microbes:def}{bacteria} in yoghurt/cheese; soil \omterm{def:g4:ecosystems:roles}{decomposers};
helpful gut \omterm{def:g4:microbes:def}{microbes}. Any two valid uses.
\end{solution}

\begin{solution}{exo:g4:microbes:3}
A \omterm{def:g4:microbes:pathogen}{pathogen} is a \omterm{def:g4:microbes:def}{microbe} or \omterm{def:g4:microbes:def}{virus} that can cause disease.
\end{solution}

\begin{solution}{exo:g4:microbes:4}
\omterm{def:g1:my-body:parts}{Hands} can carry \omterm{def:g4:microbes:pathogen}{pathogens} from surfaces or toilets onto \omterm{def:g2:food-energy:food}{food} and into
the \omterm{def:g3:teeth:path}{mouth}; washing reduces that transfer.
\end{solution}
