\chapter{من البذرة إلى النبات}\label{ch:g2:from-seed-to-plant}

هزّ قرن فول يابسًا فتتساقط منه حبات صغيرة صلبة --- جافة، باهتة،
تبدو بلا حياة كالحصى. ومع ذلك ازرع واحدة منها في التراب واسقها،
فيحدث تحت الأرض في أيام قليلة أمر عجيب. لقد كان \omterm{def:g1:plants-around-us:plant}{نبات} كامل مطويًّا
في الداخل، ينتظر إشارته.

\section{ما البذرة}

\begin{definition}[البذرة]\label{def:g2:from-seed-to-plant:seed}
\emph{البذرة}\index{بذرة} بداية \omterm{def:g1:plants-around-us:plant}{نبات} جديد، يصنعها \omterm{def:g1:plants-around-us:plant}{النبات} الأم.
وتحمل البذرة داخل غلافها الواقي:
\begin{itemize}
  \item \omterm{def:g1:plants-around-us:plant}{نبتة} صغيرة بدأت فعلًا --- وهي \emph{رشيم}\index{رشيم}
        \omterm{def:g1:plants-around-us:plant}{النبات}، وفيه بداية \omterm{def:g1:plants-around-us:parts}{جذر} وبداية برعم؛
  \item ومخزن غذاء يغذّيها حتى تستطيع أن تغذّي نفسها.
\end{itemize}
\end{definition}

\begin{example}[فتح فولة منقوعة]\label{ex:g2:from-seed-to-plant:open}
انقع فولة بيضاء كبيرة ليلة، فينزلق غلافها بسهولة. وتنفتح الفولة
على نصفين سميكين --- وهما مخزن الغذاء --- وهناك، مستقرة عند إحدى
الحافتين، \omterm{def:g1:plants-around-us:plant}{نبتة} شاحبة صغيرة، فيها طرف \omterm{def:g1:plants-around-us:parts}{جذر} \omterm{def:g1:plants-around-us:parts}{وأوراق} أُولى مطوية، في
صغر حبة أرز. كان \omterm{def:g1:plants-around-us:plant}{النبات} الجديد هناك طوال الوقت.
\end{example}

\begin{remark}[البذور تعبر مطابخنا]\label{rem:g2:from-seed-to-plant:kitchen}
الفول، والبازلاء، والعدس، والأرز، \omterm{def:g2:from-seed-to-plant:seed}{وبذور} عبّاد الشمس، \omterm{def:g2:from-seed-to-plant:seed}{وبذور}
التفاحة، ونواة الخوخ: المطبخ ملآن \omterm{def:g2:from-seed-to-plant:seed}{بالبذور}. وبعضها صغير جدًّا \omterm{def:g2:from-seed-to-plant:seed}{كبذور}
الخشخاش؛ وجوز الهند \omterm{def:g2:from-seed-to-plant:seed}{بذرة} عملاقة. وكل واحدة منها، لو أُتيحت لها
الفرصة، لحاولت أن تصير نباتًا.
\end{remark}

\section{الإنبات}

\begin{definition}[الإنبات]\label{def:g2:from-seed-to-plant:germination}
\emph{الإنبات}\index{إنبات} هو اللحظة التي تستيقظ فيها \omterm{def:g2:from-seed-to-plant:seed}{البذرة}
وتبدأ النمو: فتنتفخ بالماء، وينشقّ غلافها، ويخرج الجذير أولًا
فيغوص إلى أسفل، ثم يرتفع البرعم حاملًا \omterm{def:g1:plants-around-us:parts}{الأوراق} الخضراء الأولى نحو
الضوء. ونقول إن \omterm{def:g2:from-seed-to-plant:seed}{البذرة} \emph{تنبت}\index{نبت}.
\end{definition}

\begin{proposition}[ما تحتاج إليه البذرة لتنبت]\label{prop:g2:from-seed-to-plant:needs}
تحتاج \omterm{def:g2:from-seed-to-plant:seed}{البذرة} \omterm{def:g2:from-seed-to-plant:germination}{لتنبت} إلى \emph{الماء} و\emph{الدفء}. ولا تحتاج بعد
إلى الضوء: فمخزن الغذاء يغذّي \omterm{prop:g2:from-seed-to-plant:cycle}{البادرة} في البداية، ولو تحت التربة
في الظلام. ومن غير ماء، أو في البرد، تكتفي \omterm{def:g2:from-seed-to-plant:seed}{البذرة} الجيدة
بالانتظار.
\end{proposition}
\admitted

\begin{example}[لماذا تنتظر البذار الربيع]\label{ex:g2:from-seed-to-plant:spring}
يبذر البستانيون الفول في الربيع لا في الشتاء: \omterm{def:g2:from-seed-to-plant:seed}{فالبذرة} في تربة
باردة تنتظر --- أو تتعفن --- بدل أن \omterm{def:g2:from-seed-to-plant:germination}{تنبت}. والأيام الدافئة والتربة
المسقية هما بالضبط الإشارتان في
\cref{prop:g2:from-seed-to-plant:needs}.
\end{example}

\begin{method}[بذر بذرة ومراقبة إنباتها]\label{met:g2:from-seed-to-plant:sow}
\begin{enumerate}
  \item بطّن مرطبانًا زجاجيًّا بقطن أو ورق مبلل؛
  \item وأدخل فولة بين الزجاج والقطن، لتستطيع رؤيتها؛
  \item وأبقِ المرطبان في غرفة دافئة والقطن مبللًا؛
  \item وانظر كل يوم وسجّل ما ترى: الانتفاخ، وانشقاق الغلاف،
        \omterm{def:g1:plants-around-us:parts}{والجذر} إلى أسفل، والبرعم إلى أعلى.
\end{enumerate}
والزجاج يجعل مرئيًّا ما يحدث عادة تحت التربة.
\end{method}

\begin{remark}[الجذر أسفل والبرعم أعلى --- دائمًا]\label{rem:g2:from-seed-to-plant:updown}
مهما تكن وضعية الفولة في المرطبان --- على جنبها، أو مقلوبة ---
\omterm{def:g1:plants-around-us:parts}{فالجذر} يتجه دائمًا إلى أسفل والبرعم يتجه دائمًا إلى أعلى. \omterm{prop:g2:from-seed-to-plant:cycle}{والبادرة}
لا تبصر، ومع ذلك تعرف أين الأعلى. \omterm{def:g1:plants-around-us:plant}{النباتات} تحسّ أكثر مما تُظهر.
\end{remark}

\section{من البادرة إلى نبات ذي بذور}

\begin{proposition}[دورة حياة النبات]\label{prop:g2:from-seed-to-plant:cycle}
تنمو \emph{البادرة}\index{بادرة} فتصير نباتًا له \omterm{def:g1:plants-around-us:parts}{جذور} \omterm{def:g1:plants-around-us:parts}{وساق}
\omterm{def:g1:plants-around-us:parts}{وأوراق}؛ ثم يزهر \omterm{def:g1:plants-around-us:plant}{النبات}؛ وتصنع الأزهار \emph{بذورًا} جديدة؛ وتستطيع
الدورة أن تبدأ من جديد. \omterm{def:g1:plants-around-us:plant}{ونبتة} فول واحدة تستطيع أن تصنع عشرات
الحبات الجديدة في صيف واحد.
\end{proposition}
\admitted

\begin{example}[فولة واحدة وصيف واحد]\label{ex:g2:from-seed-to-plant:summer}
تُبذر في أيار \omterm{def:g2:from-seed-to-plant:germination}{فتنبت} في أسبوع؛ وفي حزيران تصير \omterm{def:g1:plants-around-us:plant}{نبتة} مورقة تتسلق
دعامتها؛ وفي تموز تأتي أزهار بيضاء، ثم قرون خضراء؛ وفي آب تتدلى
القرون ثقيلة، في كل واحد صف من حبات الفول الجديدة --- \omterm{def:g2:from-seed-to-plant:seed}{بذور} السنة
القادمة، صنعتها \omterm{def:g1:plants-around-us:plant}{نبتة} هذه السنة.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:1}
ما الشيئان اللذان يقعان داخل غلاف \omterm{def:g2:from-seed-to-plant:seed}{البذرة}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:2}
ما معنى أن \emph{\omterm{def:g2:from-seed-to-plant:germination}{تنبت}} \omterm{def:g2:from-seed-to-plant:seed}{البذرة}؟ وما الذي يخرج من \omterm{def:g2:from-seed-to-plant:seed}{البذرة} أولًا:
\omterm{def:g1:plants-around-us:parts}{الجذر} أم البرعم؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:3}
سمِّ الشيئين اللذين تحتاج إليهما \omterm{def:g2:from-seed-to-plant:seed}{البذرة} \omterm{def:g2:from-seed-to-plant:germination}{لتنبت}. وأي حاجة مشهورة من
حاجات \omterm{def:g1:plants-around-us:plant}{النباتات} الكبيرة \emph{ليست} في القائمة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:4}
كيف تنمو بادرة مدفونة في الظلام قبل أن تبلغ الضوء؟ ومن أين يأتي
غذاؤها في البداية؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:5}
سمِّ أربع \omterm{def:g2:from-seed-to-plant:seed}{بذور} يمكن أن تجدها في مطبخ.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:6}
لماذا يبذر البستانيون في الربيع لا في الشتاء؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:7}
رتّب: قرن فيه حبات جديدة --- \omterm{def:g2:from-seed-to-plant:seed}{بذرة} --- \omterm{def:g1:plants-around-us:plant}{نبات} مزهر --- بادرة ---
\omterm{def:g1:plants-around-us:plant}{نبات} مورق.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:from-seed-to-plant:8}
جرّب: هيّئ مرطبان \cref{met:g2:from-seed-to-plant:sow} الزجاجي
بفولتين --- واحدة على جنبها وأخرى مقلوبة. وبعد \omterm{def:g2:from-seed-to-plant:germination}{الإنبات} قارن
جذريهما وبرعميهما. بم تتنبأ \cref{rem:g2:from-seed-to-plant:updown}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:from-seed-to-plant:9}
تحفظ نينا كيس \omterm{def:g2:from-seed-to-plant:seed}{بذور} فول في درج جاف سنة كاملة، ومع ذلك \omterm{def:g2:from-seed-to-plant:germination}{تنبت} حين
تبذرها أخيرًا. فلماذا لم تنمُ ولم تمت في الدرج؟ استعمل
\cref{prop:g2:from-seed-to-plant:needs}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:from-seed-to-plant:10}
كتكوت \cref{ch:g2:animals-grow-up} ينمو داخل بيضته، ويغذّيه المحّ.
فما الذي يقوم بدور المحّ داخل الفولة؟ وبم تشترك \omterm{def:g2:animals-grow-up:born}{البيضة} \omterm{def:g2:from-seed-to-plant:seed}{والبذرة}؟
\end{exercise}
