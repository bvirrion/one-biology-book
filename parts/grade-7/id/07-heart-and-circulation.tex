\chapter{Jantung dan Peredaran Darah}\label{ch:g7:heart-and-circulation}

\Cref{ch:g4:heart-and-blood} memperkenalkan pompanya dan ronde
pengantarannya; sejak itu \omterm{prop:g4:journey-of-food:absorb}{darahnya} sudah memperoleh tanggung jawab ---
\omterm{def:g7:digestion-and-nutrients:families}{zat gizi} pada \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot ususnya}
(\cref{prop:g7:digestion-and-nutrients:absorption}), \omterm{def:g7:what-is-respiration:respiration}{oksigen} pada
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya}
(\cref{prop:g7:lungs-and-gas-exchange:numbers}), tagihan pada setiap
\omterm{def:g3:bones-and-muscles:muscle}{otot} yang bekerja (\cref{prop:g7:organs-at-work:bill}). Saatnya
memetakan jaringan jalannya dengan benar: keempat ruang \omterm{def:g4:heart-and-blood:heart}{jantungnya},
kedua peredarannya, dan ketiga jenis \omterm{def:g4:heart-and-blood:vessels}{pembuluhnya}.

\section{Keempat ruang jantung}

\begin{definition}[Serambi dan bilik]\label{def:g7:heart-and-circulation:rooms}
Kedua sisi \omterm{def:g4:heart-and-blood:heart}{jantung} (\cref{def:g4:heart-and-blood:heart}) masing-masing
memuat dua ruang: sebuah \emph{serambi}\index{serambi} di lantai atas
yang menerima \omterm{prop:g4:journey-of-food:absorb}{darah} yang datang, dan sebuah \emph{bilik}\index{bilik}
di lantai bawah --- ruang pemompa yang berdinding tebal --- yang
mendorongnya keluar. Di antara dan di seberangnya terdapat
\emph{katup}\index{katup jantung} --- pintu satu arah --- yang menutup
dengan bunyi di belakang tiap semburannya; sepasang bunyinya persis
merupakan ``lub-dub'' yang didengar sebuah stetoskop (atau sebuah
\omterm{def:g1:the-five-senses:sense}{telinga} di atas dada).
\end{definition}

\begin{proposition}[Satu denyut, secara berurutan]\label{prop:g7:heart-and-circulation:beat}
Tiap denyutnya menjalankan urutan yang sama pada kedua sisinya
sekaligus: \omterm{def:g7:heart-and-circulation:rooms}{serambinya} \omterm{def:g3:bones-and-muscles:muscle}{berkontraksi}, mengisi penuh \omterm{def:g7:heart-and-circulation:rooms}{biliknya}; \omterm{def:g7:heart-and-circulation:rooms}{biliknya}
\omterm{def:g3:bones-and-muscles:muscle}{berkontraksi}, mendorong \omterm{prop:g4:journey-of-food:absorb}{darahnya} keluar --- yang kanan menuju
\omterm{def:g7:how-animals-breathe:four}{paru-paru} dan yang kiri menuju tubuhnya --- sementara \omterm{def:g7:heart-and-circulation:rooms}{katup} masuknya
membanting menutup (lub); \omterm{def:g7:heart-and-circulation:rooms}{biliknya} mengendur, \omterm{def:g7:heart-and-circulation:rooms}{katup} keluarnya
membanting menutup (dub), dan ruangnya terisi kembali. Lalu berulang,
$100000$ kali sehari
(\cref{rem:g4:heart-and-blood:rest}).
\end{proposition}
\admitted

\section{Dua peredaran, tiga pembuluh}

\begin{definition}[Peredaran ganda]\label{def:g7:heart-and-circulation:double}
Jaringan jalan \omterm{prop:g4:journey-of-food:absorb}{darahnya} adalah dua gelang yang melewati satu pompa
(kedua sisi pada \cref{def:g4:heart-and-blood:heart}, yang sekarang
menjadi peredaran):
\begin{itemize}
  \item \emph{peredaran paru}: \omterm{def:g7:heart-and-circulation:rooms}{bilik} kanan $\rightarrow$ \omterm{def:g7:how-animals-breathe:four}{paru-paru}
        (memuat \omterm{def:g7:what-is-respiration:respiration}{oksigen}, membongkar \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}) $\rightarrow$
        \omterm{def:g7:heart-and-circulation:rooms}{serambi} kiri;
  \item \emph{peredaran tubuh}: \omterm{def:g7:heart-and-circulation:rooms}{bilik} kiri $\rightarrow$ seluruh tubuh
        (mengantarkan, memungut) $\rightarrow$ \omterm{def:g7:heart-and-circulation:rooms}{serambi} kanan.
\end{itemize}
Setiap tetesnya bergantian peredaran: \omterm{def:g7:how-animals-breathe:four}{paru-paru}, tubuh, \omterm{def:g7:how-animals-breathe:four}{paru-paru},
tubuh --- dan itulah sebabnya \omterm{def:g7:heart-and-circulation:rooms}{bilik} kiri, yang mendorong mengelilingi
gelang tubuh yang jauh lebih besar, merupakan ruang \omterm{def:g4:heart-and-blood:heart}{jantung} yang
paling tebal dindingnya.
\end{definition}

\begin{definition}[Arteri, kapiler, vena]\label{def:g7:heart-and-circulation:vessels}
Tiga jenis \omterm{def:g4:heart-and-blood:vessels}{pembuluh} menyusun tiap gelangnya
(\cref{def:g4:heart-and-blood:vessels}, yang sekarang dinamai):
\begin{itemize}
  \item \emph{arteri}\index{arteri}: berdinding tebal, lentur,
        membawa \omterm{prop:g4:journey-of-food:absorb}{darah} \emph{menjauh} dari \omterm{def:g4:heart-and-blood:heart}{jantung} di bawah tekanan ---
        \omterm{prop:g4:heart-and-blood:pulse}{denyut nadinya}
        (\cref{prop:g4:heart-and-blood:pulse}) adalah regangannya pada
        tiap semburan;
  \item \emph{kapiler}\index{kapiler}: \omterm{def:g4:heart-and-blood:vessels}{pembuluh} sehalus rambut yang
        menyelusup di setiap organ --- dindingnya setebal satu \omterm{def:g5:first-look-at-cells:cell}{sel},
        tempat \emph{semua} pertukarannya terjadi: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolus}, \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot
        usus}, \omterm{def:g3:bones-and-muscles:muscle}{otot}, \omterm{def:g1:the-five-senses:sense}{kulit};
  \item \emph{vena}\index{vena}: lebar, berdinding lebih tipis,
        mengembalikan \omterm{prop:g4:journey-of-food:absorb}{darah} \emph{ke} \omterm{def:g4:heart-and-blood:heart}{jantung} pada tekanan yang
        rendah, dengan \omterm{def:g7:heart-and-circulation:rooms}{katup} satu arahnya sendiri --- garis kebiruan
        pada \cref{rem:g4:heart-and-blood:see}.
\end{itemize}
\end{definition}

\begin{example}[Satu tetes, satu keliling penuh]\label{ex:g7:heart-and-circulation:tour}
Ronde pengantaran \cref{ex:g4:heart-and-blood:round}, dengan nama pada
petanya: dari \omterm{def:g7:heart-and-circulation:rooms}{bilik} kanan ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler} paru (\omterm{def:g7:what-is-respiration:respiration}{oksigen} naik di
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya}, warnanya merah cerah), lalu ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} kiri dan \omterm{def:g7:heart-and-circulation:rooms}{bilik}
kiri, lalu ke \omterm{def:g7:heart-and-circulation:vessels}{arteri} dan ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler} sebuah \omterm{def:g3:bones-and-muscles:muscle}{otot} paha (\omterm{def:g7:what-is-respiration:respiration}{oksigen} dan
\omterm{prop:g7:organs-at-work:bill}{glukosa} turun, \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} naik, warnanya lebih gelap), lalu ke
\omterm{def:g7:heart-and-circulation:vessels}{vena}, dengan \omterm{def:g7:heart-and-circulation:rooms}{katup} yang meringankan perjalanan mendakinya, dan
akhirnya ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} kanan. Dua peredaran, kurang dari semenit,
tanpa belokan salah yang tersedia: katupnyalah yang mengurus itu.
\end{example}

\begin{remark}[Mengapa kapilernya yang menjadi intinya]\label{rem:g7:heart-and-circulation:capillaries}
\omterm{def:g7:heart-and-circulation:vessels}{Arteri} dan \omterm{def:g7:heart-and-circulation:vessels}{vena} hanyalah jalan raya; setiap pengantaran yang
sesungguhnya --- setiap pertukaran yang dipelajari tahun ini ---
terjadi menyeberangi dinding \omterm{def:g7:heart-and-circulation:vessels}{kapiler}. Spesifikasi pada
\cref{prop:g7:how-animals-breathe:spec} hidup di sini juga: dinding
setebal satu \omterm{def:g5:first-look-at-cells:cell}{sel} (tipis), yang menyelusupi setiap jaringan sepanjang
puluhan ribu kilometer
(\cref{def:g4:heart-and-blood:vessels} --- luas), yang terbasuh oleh
\omterm{def:g6:exploring-our-environment:conditions}{kelembapan} tubuhnya. \omterm{def:g4:heart-and-blood:heart}{Jantung} ada untuk menjaga \omterm{prop:g4:journey-of-food:absorb}{darah} tetap bergerak
\emph{melewati \omterm{def:g7:heart-and-circulation:vessels}{kapiler}}; sisanya adalah perpipaan menuju dan dari
situ.
\end{remark}

\section{Membaca dan menjaga jaringannya}

\begin{example}[Pengukuran di tepi jalan]\label{ex:g7:heart-and-circulation:measures}
Tiga bacaan sehari-hari atas jaringannya: \emph{\omterm{prop:g4:heart-and-blood:pulse}{denyut nadi}} ---
regangan \omterm{def:g7:heart-and-circulation:vessels}{arterinya}, yang menghitung semburannya
(\cref{met:g4:heart-and-blood:pulse}); \emph{tekanan \omterm{prop:g4:journey-of-food:absorb}{darah}}, yaitu
kedua bilangan pada manset --- tekanan \omterm{def:g7:heart-and-circulation:vessels}{arterinya} ketika menyembur dan
di antara semburannya; dan \emph{lub-dub} pada stetoskopnya --- pintu
\omterm{def:g7:heart-and-circulation:rooms}{katupnya} yang terdengar menutup. Tiga jendela, tanpa satu sayatan pun.
\end{example}

\begin{proposition}[Menjaga jaringannya]\label{prop:g7:heart-and-circulation:keep}
Musuh dan kawan jangka panjang jaringannya sudah diketahui:
\begin{itemize}
  \item \emph{olahraga} memperkuat pompanya
        (\cref{prop:g7:organs-at-work:training}) dan menjaga jalan
        rayanya tetap lentur;
  \item \emph{makan yang seimbang}
        (\cref{met:g3:balanced-diet:check}) melindungi dinding
        \omterm{def:g7:heart-and-circulation:vessels}{arterinya}, yang perlahan dikeraki lalu dikakukan oleh \omterm{prop:g4:journey-of-food:absorb}{darah}
        yang terlalu kaya;
  \item \emph{asap} menyerang \omterm{def:g4:heart-and-blood:vessels}{pembuluhnya} selain \omterm{def:g7:how-animals-breathe:four}{paru-parunya}
        (\cref{prop:g7:lungs-and-gas-exchange:smoke}) ---
        menyempitkan, mengakukan, merebut kursi;
  \item pasokan \omterm{def:g4:heart-and-blood:heart}{jantungnya} sendiri --- ia diberi makan oleh \omterm{def:g7:heart-and-circulation:vessels}{arteri}
        kecilnya sendiri, bukan dari \omterm{prop:g4:journey-of-food:absorb}{darah} di dalam ruangnya ---
        bergantung pada ketiganya.
\end{itemize}
Kebiasaan pada \cref{prop:g5:healthy-choices:contract} secara konkret
adalah pemeliharaan \omterm{def:g4:heart-and-blood:vessels}{pembuluh}.
\end{proposition}
\admitted

\begin{method}[Memetakan organ apa pun ke dalam jaringannya]\label{met:g7:heart-and-circulation:map}
Untuk organ apa pun yang dijumpai buku ini:
\begin{enumerate}
  \item tempatkan ia pada peredaran tubuhnya: \omterm{def:g7:heart-and-circulation:vessels}{arteri} masuk, \omterm{def:g7:heart-and-circulation:vessels}{vena}
        keluar, \omterm{def:g7:heart-and-circulation:vessels}{kapiler} di dalamnya;
  \item daftarkan apa yang menyeberangi dinding \omterm{def:g7:heart-and-circulation:vessels}{kapilernya}, masuk dan
        keluar (\omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot usus}: \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} masuk; \omterm{def:g3:bones-and-muscles:muscle}{otot}: bahan bakar dan
        \omterm{def:g7:what-is-respiration:respiration}{oksigen} masuk, \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} keluar; \omterm{def:g1:the-five-senses:sense}{kulit}: panas keluar);
  \item catatlah sambungan khususnya, kalau ada (\omterm{prop:g4:journey-of-food:absorb}{darah} ususnya singgah
        di \omterm{prop:g7:digestion-and-nutrients:absorption}{hatinya} lebih dahulu; \omterm{def:g7:how-animals-breathe:four}{paru-parunya} duduk pada peredarannya
        sendiri);
  \item periksalah alirannya terhadap tugas organnya --- bagan
        jaringannya seharusnya menceritakan ulang bab organ itu.
\end{enumerate}
\end{method}

\section{Soal: Pagi Sang Dokter Jantung}

\begin{problem}[{Soal akhir pekan --- empat konsultasi, satu peta
jaringannya}]\label{pb:g7:heart-and-circulation:1}
Bayangilah seorang dokter \omterm{def:g4:heart-and-blood:heart}{jantung} selama satu pagi berisi empat
konsultasi (rekaan), dengan peta di tangan.

\medskip\noindent\textbf{Bagian I --- Garis dasar yang
sehat.}
\begin{enumerate}
  \item Pasien 1, seorang \omterm{def:g3:stages-of-life:stages}{remaja} yang bugar, mendapat ketiga bacaan
        tepi jalan pada
        \cref{ex:g7:heart-and-circulation:measures}. Sebutkan
        masing-masingnya, dan bagian permesinan mana yang dibaca
        masing-masingnya.
  \item \omterm{prop:g4:heart-and-blood:pulse}{Denyut nadi} istirahatnya $58$. Dokter \omterm{def:g4:heart-and-blood:heart}{jantungnya} tersenyum.
        Mengapa
        (\cref{exo:g4:heart-and-blood:11})?
  \item Lub-dub-nya tegas dan ganda. Apa tepatnya yang membanting
        menutup, dua kali, pada tiap denyutnya?
  \item Sketsakan jalan \omterm{prop:g4:journey-of-food:absorb}{darahnya} dari \omterm{def:g7:heart-and-circulation:rooms}{serambi} kanan mengelilingi kedua
        peredarannya lalu kembali, dalam delapan perhentian bernama.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Tiga masalah.}
\begin{enumerate}[resume]
  \item Pasien 2 punya sebuah \omterm{def:g7:heart-and-circulation:rooms}{katup} yang bocor antara \omterm{def:g7:heart-and-circulation:rooms}{serambi} kiri dan
        \omterm{def:g7:heart-and-circulation:rooms}{biliknya}: sebagian \omterm{prop:g4:journey-of-food:absorb}{darahnya} menyembur mundur pada tiap
        denyutnya, dengan bising \omterm{def:g4:heart-and-blood:heart}{jantung} yang berdesing. Ramalkan
        akibatnya bagi pengantaran peredaran tubuhnya dan bagi daya
        tahan pasiennya.
  \item Pasien 3, seorang perokok berat dengan pola makan yang kaya,
        punya \omterm{def:g7:heart-and-circulation:vessels}{arteri} yang mengerak dan mengaku. Bacaan mana pada
        konsultasi 1 yang akan memburuk, dan mengapa \omterm{def:g7:heart-and-circulation:rooms}{bilik} kirinya
        bekerja lembur?
  \item Kerak pasien 3 mencakup \omterm{def:g7:heart-and-circulation:vessels}{arteri} pemberi makan \omterm{def:g4:heart-and-blood:heart}{jantungnya}
        sendiri. Jelaskan kekhawatiran khusus dokter \omterm{def:g4:heart-and-blood:heart}{jantungnya} dalam
        satu kalimat.
  \item Pasien 4 punya \omterm{def:g1:my-body:joint}{pergelangan kaki} yang bengkak sesudah hari-hari
        panjang berdiri: kembalian mendaki \omterm{def:g7:heart-and-circulation:vessels}{venanya} kepayahan. Struktur
        mana yang gagal, dan mengapa berjalan --- dengan \omterm{def:g3:bones-and-muscles:muscle}{otot} betis
        yang memeras \omterm{def:g7:heart-and-circulation:vessels}{venanya} --- meringankannya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Bagan dindingnya.}
\begin{enumerate}[resume]
  \item Bagan dinding kliniknya memperlihatkan gelang gandanya.
        Jelaskan kepada seorang pasien yang menunggu mengapa \omterm{prop:g4:journey-of-food:absorb}{darahnya}
        harus mengunjungi \omterm{def:g7:how-animals-breathe:four}{paru-parunya} di antara setiap dua keliling
        tubuh.
  \item Tambahkan keterangan \omterm{def:g4:heart-and-blood:vessels}{pembuluhnya}: \omterm{def:g7:heart-and-circulation:vessels}{arteri}, \omterm{def:g7:heart-and-circulation:vessels}{kapiler}, \omterm{def:g7:heart-and-circulation:vessels}{vena} ---
        satu baris masing-masing tentang bangunan, tekanan dan
        tugasnya.
  \item Kaki bagannya berbunyi: ``semua pengobatan melayani
        \omterm{def:g7:heart-and-circulation:vessels}{kapilernya}.'' Belalah ia dengan
        \cref{rem:g7:heart-and-circulation:capillaries}.
  \item Tutuplah paginya dengan resep pencegahan yang diserahkan
        dokter \omterm{def:g4:heart-and-blood:heart}{jantungnya} kepada setiap pasien --- tiga kebiasaan,
        masing-masing terikat pada mekanismenya di dalam
        \cref{prop:g7:heart-and-circulation:keep}.
\end{enumerate}
\end{problem}
