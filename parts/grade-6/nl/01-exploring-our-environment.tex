\chapter{Onze omgeving verkennen}\label{ch:g6:exploring-our-environment}

Steek op een julimiddag het schoolplein over en de zuidkant van de muur
is heet onder je hand, de lucht boven het asfalt trilt, en het enige
groen in zicht is een onkruidje in een spleet. Stap achter de noordmuur:
koele lucht, vochtige grond, mos, pissebedden onder een steen. Twee
werelden, drie meter uit elkaar. Dit jaar begint met \omterm{def:g6:exploring-our-environment:components}{omgevingen} verkennen
zoals wetenschappers dat doen --- ze meten, en vragen waarom elk \omterm{def:g6:exploring-our-environment:components}{levend
wezen} precies daar leeft waar het leeft.

\section{Waaruit een omgeving bestaat}

\begin{definition}[De bestanddelen van een omgeving]\label{def:g6:exploring-our-environment:components}
Een \emph{omgeving}\index{omgeving} --- een plein, een bos, een
vijveroever --- bevat drie \omterm{def:g5:biodiversity:species}{soorten} bestanddelen:
\begin{enumerate}
  \item \emph{levende wezens}\index{levend wezen}: de \omterm{def:g1:animals-around-us:animal}{dieren}, \omterm{def:g1:plants-around-us:plant}{planten},
        paddenstoelen en het microscopische leven dat er is;
  \item \emph{minerale materie}\index{minerale materie}, die \omterm{def:g1:living-or-not:nonliving}{nooit
        geleefd} heeft: rots, steen, het minerale deel van de bodem,
        water, lucht;
  \item \emph{sporen en resten} van levende wezens: dode \omterm{def:g1:plants-around-us:parts}{bladeren},
        gevallen hout, uitwerpselen, \omterm{def:g1:animals-around-us:covering}{veren}, \omterm{def:g1:animals-around-us:covering}{schelpen} --- de bak ``heeft
        geleefd'' uit \cref{def:g1:living-or-not:nonliving}, nu een
        aanwijzingenset voor de verkenner.
\end{enumerate}
\end{definition}

\begin{example}[Inventarisatie achter de noordmuur]\label{ex:g6:exploring-our-environment:inventory}
Tien minuten inventariseren in de koele hoek: levend --- mos, \omterm{prop:g4:plants-without-seeds:damp}{varens},
pissebedden, een slak, een duizendpoot, klimop; mineraal --- de steen van
de muur, vochtige grond, een plas; sporen --- dode \omterm{def:g1:plants-around-us:parts}{bladeren}, een
slakkenhuis, vogelpoep op de muurafdekking. Drie lijsten, en de hoek is
beschreven als de locatie van een wetenschapper.
\end{example}

\begin{remark}[Sporen tellen als gegevens]\label{rem:g6:exploring-our-environment:traces}
De meeste \omterm{def:g1:animals-around-us:animal}{dieren} verstoppen zich of vluchten, dus leest de verkenner vaak
\emph{sporen} in plaats van \omterm{def:g1:animals-around-us:animal}{dieren}: een aangeknaagde hazelnoot noemt de
bosmuis, een spinnenweb noemt zijn eigenaar, sporen in de modder noemen
de reiger. Een inventarisatie die alleen opsomt wat stil genoeg zat om
gezien te worden, mist de helft van de bewoners.
\end{remark}

\section{De omstandigheden meten}

\begin{definition}[Omgevingsomstandigheden]\label{def:g6:exploring-our-environment:conditions}
De \emph{omstandigheden}\index{omstandigheden (in de omgeving)} van een
\omterm{def:g6:exploring-our-environment:components}{omgeving} zijn haar meetbare natuurkundige eigenschappen:
\begin{itemize}
  \item de \emph{temperatuur}\index{temperatuur}, gemeten met een
        thermometer, in graden Celsius;
  \item de \emph{vochtigheid}\index{vochtigheid} --- hoe vochtig de
        lucht of de bodem is;
  \item het \emph{licht}\index{licht (omstandigheid)} --- van volle zon
        tot diepe schaduw.
\end{itemize}
Omstandigheden verschillen van plek tot plek --- vaak scherp binnen een
paar meter --- en door de dag en het jaar heen.
\end{definition}

\begin{example}[Het plein, gemeten]\label{ex:g6:exploring-our-environment:measured}
De julimiddagopname, met instrumenten: voet van de zuidmuur ---
\qty{38}{\degreeCelsius}, droog, volle zon; noordhoek ---
\qty{22}{\degreeCelsius}, vochtige grond, diepe schaduw. Drie meter
afstand, zestien graden verschil. De twee werelden uit de openingsalinea
zijn nu twee \emph{rijen in een tabel} --- en dat is vooruitgang, want
tabellen kun je vergelijken.
\end{example}

\begin{proposition}[De omstandigheden bepalen de bewoners]\label{prop:g6:exploring-our-environment:decide}
De \omterm{def:g6:exploring-our-environment:components}{levende wezens} van een plek passen bij haar \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}: elke \omterm{def:g5:biodiversity:species}{soort}
gedijt alleen binnen haar eigen bereik van \omterm{def:g6:exploring-our-environment:conditions}{temperatuur}, \omterm{def:g6:exploring-our-environment:conditions}{vochtigheid} en
\omterm{def:g6:exploring-our-environment:conditions}{licht}. Mos en pissebedden hebben vocht nodig (de bedekking van de
pissebed laat, anders dan die van een insect, water ontsnappen --- hij
droogt in een uur zon dodelijk uit); het onkruidje in de spleet verdraagt
droogte en felle zon. Waar de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} verschillen, verschillen de
bewoners --- de regel achter elk \omterm{def:g2:where-animals-live:habitat}{leefgebied} uit
\cref{ch:g2:where-animals-live}.
\end{proposition}
\admitted

\begin{example}[De pissebeddenkiezer]\label{ex:g6:exploring-our-environment:chooser}
De klassieke proef: een gesloten doos, half bekleed met vochtig papier en
half met droog, en een stuk of twaalf pissebedden losgelaten langs de
middenlijn. Binnen een paar minuten zit bijna alles op de vochtige kant.
Doe het opnieuw met een beschaduwde helft tegenover een verlichte helft:
ze verzamelen zich in de schaduw. Niemand drijft ze op --- elk \omterm{def:g1:animals-around-us:animal}{dier} dat
rondloopt en keert, komt uit waar zijn lichaam zich prettig voelt. De
kaart van de pissebedden \emph{is} een kaart van de \omterm{def:g6:exploring-our-environment:conditions}{vochtigheid}.
\end{example}

\begin{method}[Een locatie in kaart brengen]\label{met:g6:exploring-our-environment:survey}
Om een kleine locatie te onderzoeken zoals een veldbioloog:
\begin{enumerate}
  \item kies een klein stuk en markeer het --- één vierkante meter, goed
        doorzocht, vertelt meer dan een hectare waar je even naar kijkt;
  \item meet de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}: \omterm{def:g6:exploring-our-environment:conditions}{temperatuur}, \omterm{def:g6:exploring-our-environment:conditions}{vochtigheid}, \omterm{def:g6:exploring-our-environment:conditions}{licht} ---
        dezelfde instrumenten, dezelfde manier, op elke locatie (de
        eerlijkheidsregel uit
        \cref{met:g4:what-plants-need:fairtest});
  \item inventariseer de drie \omterm{def:g5:biodiversity:species}{soorten} bestanddelen --- \omterm{def:g6:exploring-our-environment:components}{levende wezens},
        minerale materie, sporen --- en zoek goed: onder stenen (leg ze
        terug: \cref{met:g2:caring-for-nature:rules}), in spleten, aan
        de onderkant van \omterm{def:g1:plants-around-us:parts}{bladeren};
  \item noteer alles ter plekke --- lijsten, schetsen, aantallen; het
        geheugen is geen meetinstrument;
  \item vergelijk de locaties: koppel elk verschil in bewoners aan een
        verschil in \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}.
\end{enumerate}
\end{method}

\begin{remark}[Waarom aantekeningen het winnen van herinneringen]\label{rem:g6:exploring-our-environment:records}
Terug binnen krijgen twee onderzoekers die op hun geheugen vertrouwden
ruzie over de vraag of het er nu drie of vijf slakken waren. Het
notitieboekje beslist --- of geeft eerlijk toe dat de slakken niet
geteld zijn. Geschreven aantekeningen, ter plekke gemaakt, maken van een
wandeling gegevens; elke wetenschap in dit boek draait erop.
\end{remark}

\section{Eén omgeving, veel microwerelden}

\begin{definition}[Microhabitat]\label{def:g6:exploring-our-environment:microhabitat}
Binnen één \omterm{def:g6:exploring-our-environment:components}{omgeving} is elk klein plekje met zijn eigen \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} en
zijn eigen bewoners --- de onderkant van een steen, een spleet in de
muur, de beschaduwde voet van de heg --- een
\emph{microhabitat}\index{microhabitat}. De bosverdiepingen uit
\cref{rem:g2:where-animals-live:layers} waren microhabitats; de vierkante
meter van de onderzoeker bevat er meestal meerdere.
\end{definition}

\begin{example}[De steen, boven en onder]\label{ex:g6:exploring-our-environment:stone}
Eén steen in de koele hoek: bovenop korstmos en een zonnende vlieg ---
verlicht, droog, warm. Eronder pissebedden, een duizendpoot, bleke
worteltjes --- donker, vochtig, koel. Twee complete \omterm{def:g6:exploring-our-environment:microhabitat}{microhabitats},
gescheiden door vijf centimeter steen, elk bewoond door precies die
\omterm{def:g6:exploring-our-environment:components}{levende wezens} bij wier bereik de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} passen.
\end{example}

Noem de drie \omterm{def:g5:biodiversity:species}{soorten} bestanddelen van een \omterm{def:g6:exploring-our-environment:components}{omgeving}, met van elk twee

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:2}
Welke \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} meet een veldbioloog op een locatie, en waarmee

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:3}
Deel in bij de drie \omterm{def:g5:biodiversity:species}{soorten} bestanddelen: een varen, een slakkenhuis,
een plas, vogelpoep, een duizendpoot, de steen van de muur.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:4}
Uit \cref{ex:g6:exploring-our-environment:measured}: hoe groot zijn de
verschillen in \omterm{def:g6:exploring-our-environment:conditions}{temperatuur} en \omterm{def:g6:exploring-our-environment:conditions}{licht} tussen de twee plekken? Welke
bewoners passen bij elk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:5}

\begin{exercise}[$\star$]\label{exo:g6:exploring-our-environment:6}

\begin{exercise}[$\star\star$]\label{exo:g6:exploring-our-environment:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:exploring-our-environment:8}

\begin{exercise}[$\star\star$]\label{exo:g6:exploring-our-environment:9}

\begin{exercise}[$\star\star$]\label{exo:g6:exploring-our-environment:10}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:exploring-our-environment:11}

\begin{exercise}[$\star\star\star$]\label{exo:g6:exploring-our-environment:12}

\end{exercise}

\section{Probleem: De twee sloten}

\begin{problem}[{Weekendprobleem --- de natuurclub onderzoekt twee
sloten langs de weg en vindt twee verschillende
werelden}]\label{pb:g6:exploring-our-environment:1}
De natuurclub onderzoekt twee sloten langs hetzelfde weggetje, $50$ meter
uit elkaar. Sloot N loopt langs de noordkant van een hoge heg; sloot Z
ligt in het open veld, op het zuiden. De club werkt volgens het boekje:
dezelfde instrumenten, dezelfde vierkante meter per locatie, dezelfde
zoektijd.

\medskip\noindent\textbf{Deel I --- De metingen.}
\begin{enumerate}
  \item De thermometers wijzen op hetzelfde uur \qty{19}{\degreeCelsius}
        aan in sloot N en \qty{31}{\degreeCelsius} in sloot Z. Welk
        enkel kenmerk van de locaties verklaart het verschil het best?
  \item De grond van sloot N plakt aan je vingers; die van sloot Z
        verkruimelt tot stof. Vertaal allebei de waarnemingen in de
        woordenschat van
        \cref{def:g6:exploring-our-environment:conditions}.
  \item De club meet 's middags. Geef één reden waarom de vergelijking
        minder eerlijk zou zijn als sloot N 's middags en sloot Z om
        zes uur 's avonds gemeten werd.
  \item Waarom gebruikt de club op beide locaties \emph{hetzelfde}
        vierkantemeterframe en dezelfde zoektijd?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De inventarisaties.}
\begin{enumerate}[resume]
  \item De lijst van sloot N: mos, \omterm{prop:g4:plants-without-seeds:damp}{varens}, pissebedden, slakken, een pad.
        De lijst van sloot Z: droge grassen, een grote \omterm{def:g1:plants-around-us:plant}{plant} met
        wasachtige \omterm{def:g1:plants-around-us:parts}{bladeren}, mieren, een zonnende hagedis, sprinkhanen.
        Deel de leden van elke lijst in bij de drie \omterm{def:g5:biodiversity:species}{soorten} bestanddelen
        --- en merk op wat er in de lijsten ontbreekt (kijk naar
        \cref{def:g6:exploring-our-environment:components}).
  \item Zeg voor de pissebedden, de pad en de hagedis elk bij welke sloot
        de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} bij hun behoeften passen, en waarom (gebruik
        \cref{prop:g6:exploring-our-environment:decide} en wat je over
        elk \omterm{def:g1:animals-around-us:animal}{dier} weet).
  \item In sloot Z vindt de club een slakkenhuis, maar geen slak, en
        vraagt zich af of er slakken leven. Welke twee eerlijke
        mogelijkheden laat zo'n spoor open?
  \item Onder een platte steen in sloot N: acht pissebedden, een
        duizendpoot, witte worteltjes. Noem de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} van dit
        \omterm{def:g6:exploring-our-environment:microhabitat}{microhabitat}, en geef de regel die die bewoners daar bracht.
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De verklaring.}
\begin{enumerate}[resume]
  \item Koppel elk verschil in bewoners aan een verschil in
        \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}: vul aan ``N heeft mos en padden omdat \dots; Z
        heeft hagedissen en \omterm{def:g1:plants-around-us:plant}{planten} met wasachtige \omterm{def:g1:plants-around-us:parts}{bladeren} omdat
        \dots''.
  \item De heg staat op de nominatie om gerooid te worden. Voorspel de
        \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} van sloot N twee zomers later --- en haar
        bewoners, met de koppeling uit vraag 9.
  \item Een lid stelt voor de voorspelling netjes te controleren: beide
        sloten opnieuw onderzoeken nadat de heg weg is. Waarom moeten
        \emph{dezelfde} locaties, instrumenten en het seizoen gebruikt
        worden wil dat tweede onderzoek meetellen?
  \item Het verslag van de club eindigt met: ``de bewoners brengen de
        \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} in kaart''. Leg die zin uit met de
        pissebeddenkiezer
        (\cref{ex:g6:exploring-our-environment:chooser}), en zeg waarom
        er geen opdrijven nodig is om zo'n kaart zichzelf te laten
        tekenen.
\end{enumerate}
\end{problem}
