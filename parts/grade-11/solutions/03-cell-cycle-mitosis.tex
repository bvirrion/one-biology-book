\section*{Chapter \ref{ch:g11:cell-cycle-mitosis} --- The Cell Cycle and Mitosis}

\begin{solution}{exo:g11:cell-cycle-mitosis:1}
\omterm{def:g11:cell-cycle-mitosis:cycle}{Interphase} comprises G$_1$, \omterm{def:g11:cell-cycle-mitosis:interphase}{S} and G$_2$. In \omterm{def:g11:cell-cycle-mitosis:interphase}{S phase} the \omterm{def:g11:dna-replication:helix}{DNA} is replicated so
that each \omterm{def:g11:dna-replication:chromosome}{chromosome} consists of two \omterm{def:g11:cell-cycle-mitosis:chromatid}{sister chromatids} joined at the
\omterm{def:g11:cell-cycle-mitosis:chromatid}{centromere}, preparing the \omterm{prop:g11:dna-replication:genome}{genome} for equal segregation at \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:2}
\omterm{def:g11:cell-cycle-mitosis:chromatid}{Sister chromatids} are identical copies of one \omterm{def:g11:dna-replication:chromosome}{chromosome} produced by \omterm{def:g11:dna-replication:semi}{DNA
replication} and joined at the \omterm{def:g11:cell-cycle-mitosis:chromatid}{centromere} until anaphase. Homologous \omterm{def:g11:dna-replication:chromosome}{chromosomes}
are the maternal and paternal partners of a pair: same \omterm{def:g11:gene-expression:gene}{genes} in the same \omterm{def:g10:biodiversity-classification:hierarchy}{order},
but not necessarily the same alleles, and not products of the latest \omterm{def:g11:cell-cycle-mitosis:interphase}{S phase}.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:3}
\omterm{def:g10:biodiversity-classification:hierarchy}{Order}: prophase, prometaphase, metaphase, anaphase, telophase. \omterm{prop:g11:cell-cycle-mitosis:stages}{Metaphase}:
\omterm{def:g11:dna-replication:chromosome}{chromosomes} are aligned at the spindle equator with bipolar attachment.
\omterm{prop:g11:cell-cycle-mitosis:stages}{Anaphase}: cohesin is cleaved and \omterm{def:g11:cell-cycle-mitosis:chromatid}{sister chromatids} separate to opposite poles.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:4}
In G$_2$ the \omterm{def:g10:the-cell:cell}{cell} still has $46$ \omterm{def:g11:dna-replication:chromosome}{chromosomes} but $92$ chromatids (each
\omterm{def:g11:dna-replication:chromosome}{chromosome} is duplicated). After \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis} and \omterm{def:g11:cell-cycle-mitosis:cytokinesis}{cytokinesis} each daughter in
G$_1$ again has $46$ \omterm{def:g11:dna-replication:chromosome}{chromosomes} and $46$ chromatids --- each \omterm{def:g11:dna-replication:chromosome}{chromosome} is a
single \omterm{def:g11:dna-replication:helix}{DNA} molecule.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:5}
\omterm{def:g11:cell-cycle-mitosis:mitosis}{Mitosis} divides the nucleus; \omterm{def:g11:cell-cycle-mitosis:cytokinesis}{cytokinesis} divides the cytoplasm. They usually
coordinate but can be uncoupled: \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis} without \omterm{def:g11:cell-cycle-mitosis:cytokinesis}{cytokinesis} yields multiple
nuclei in one \omterm{def:g10:the-cell:cell}{cell}; \omterm{def:g11:cell-cycle-mitosis:cytokinesis}{cytokinesis} without a prior proper \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis} does not
distribute \omterm{prop:g11:dna-replication:genome}{genomes} equally and is not the standard healthy pathway.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:6}
Alignment at the metaphase plate with bipolar \omterm{def:g11:cell-cycle-mitosis:chromatid}{kinetochore} attachment positions
each pair of sisters so that, when cohesin is cleaved, one chromatid is pulled
to each pole. That geometry is the physical basis of equal partition of the
replicated \omterm{prop:g11:dna-replication:genome}{genome}.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:7}
G$_1$/\omterm{def:g11:cell-cycle-mitosis:interphase}{S} monitors \omterm{def:g10:scientific-biology:properties}{growth} signals, \omterm{def:g10:the-cell:cell}{cell} size and \omterm{def:g11:dna-replication:helix}{DNA} damage before committing to
\omterm{def:g11:dna-replication:semi}{replication}. G$_2$/M checks that \omterm{def:g11:dna-replication:semi}{DNA replication} is complete and damage is
addressed. The spindle (metaphase--anaphase) \omterm{def:g11:cell-cycle-mitosis:checkpoint}{checkpoint} monitors attachment of
\omterm{def:g11:cell-cycle-mitosis:chromatid}{kinetochores} to microtubules before allowing anaphase.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:8}
Plant \omterm{def:g10:the-cell:cell}{cells} build a \omterm{prop:g11:cell-cycle-mitosis:cyto-compare}{cell plate} from vesicles that fuse at the midplane; the
plate matures into a new \omterm{def:g10:the-cell:support}{cell wall} (and middle lamella) separating the
daughters. They do not pinch with an animal-style contractile ring because the
existing wall is rigid.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:9}
Without microtubules the spindle cannot form and \omterm{def:g11:cell-cycle-mitosis:chromatid}{kinetochores} cannot attach,
so \omterm{def:g11:dna-replication:chromosome}{chromosomes} fail to congress properly. The spindle \omterm{def:g11:cell-cycle-mitosis:checkpoint}{checkpoint} detects
unattached \omterm{def:g11:cell-cycle-mitosis:chromatid}{kinetochores} and can arrest the \omterm{def:g10:the-cell:cell}{cell} before anaphase, preventing
catastrophic mis-segregation --- or, if overridden, aneuploidy results.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:10}
For $2n = 6$: G$_1$ has $6$ \omterm{def:g11:dna-replication:chromosome}{chromosomes} and $6$ chromatids; G$_2$ has $6$
\omterm{def:g11:dna-replication:chromosome}{chromosomes} and $12$ chromatids; at anaphase there are $12$ \omterm{def:g11:dna-replication:chromosome}{chromosomes}
(separated sisters moving apart); each daughter G$_1$ again has $6$
\omterm{def:g11:dna-replication:chromosome}{chromosomes} and $6$ chromatids.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:11}
Animal \omterm{def:g10:the-cell:cell}{cells} use a contractile actin--myosin ring to form a \omterm{prop:g11:cell-cycle-mitosis:cyto-compare}{cleavage furrow}
and pinch the cytoplasm. Plant \omterm{def:g10:the-cell:cell}{cells} assemble a \omterm{prop:g11:cell-cycle-mitosis:cyto-compare}{cell plate} from vesicles at
the midplane that becomes a new wall. A \omterm{prop:g11:cell-cycle-mitosis:cyto-compare}{cleavage furrow} alone cannot divide a
walled plant \omterm{def:g10:the-cell:cell}{cell}: the rigid wall resists pinching, so new wall must be built
between the nuclei.
\end{solution}

\begin{solution}{exo:g11:cell-cycle-mitosis:12}
A failed G$_1$/\omterm{def:g11:cell-cycle-mitosis:interphase}{S} damage \omterm{def:g11:cell-cycle-mitosis:checkpoint}{checkpoint} lets \omterm{def:g10:the-cell:cell}{cells} replicate damaged \omterm{def:g11:dna-replication:helix}{DNA} and pass
mutations to daughters. A failed spindle \omterm{def:g11:cell-cycle-mitosis:checkpoint}{checkpoint} allows anaphase with
mis-attached \omterm{def:g11:dna-replication:chromosome}{chromosomes}, producing aneuploidy and genomic instability. \omterm{rem:g11:cell-cycle-mitosis:cancer}{Cancer}
is more than ``faster \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}'': it also involves inappropriate \omterm{def:g10:scientific-biology:properties}{growth} signals,
evasion of death, and often invasion --- \omterm{def:g11:cell-cycle-mitosis:checkpoint}{checkpoint} failure is one central
piece of loss of cycle control.
\end{solution}

\begin{solution}{pb:g11:cell-cycle-mitosis:1}
\textbf{1--2.} G$_1$: four unreplicated \omterm{def:g11:dna-replication:chromosome}{chromosomes} (two homologous pairs).
G$_2$: each \omterm{def:g11:dna-replication:chromosome}{chromosome} shown as two chromatids joined at a \omterm{def:g11:cell-cycle-mitosis:chromatid}{centromere}.

\textbf{3.} One \omterm{def:g11:dna-replication:chromosome}{chromosome} condenses in prophase, attaches to opposite poles by
prometaphase, aligns at metaphase, and its sisters separate at anaphase into
two daughter \omterm{def:g11:dna-replication:chromosome}{chromosomes} that decondense in telophase.

\textbf{4.} Each daughter receives $4$ \omterm{def:g11:dna-replication:chromosome}{chromosomes}. Yes: \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis} aims at two
genetically equivalent \omterm{def:g11:cell-cycle-mitosis:diploid}{diploid} nuclei (mutation aside).

\textbf{5.} One nucleus ends with $2n+1 = 5$ \omterm{def:g11:dna-replication:chromosome}{chromosomes} and the other with
$2n-1 = 3$ for mis-segregation of one \omterm{def:g11:dna-replication:chromosome}{chromosome}'s sisters (other \omterm{def:g11:dna-replication:chromosome}{chromosomes}
normal).

\textbf{6.} \omterm{prop:g11:cell-cycle-mitosis:stages}{Anaphase} separates sisters that exist only because \omterm{def:g11:cell-cycle-mitosis:interphase}{S phase} copied
the \omterm{def:g11:dna-replication:helix}{DNA}; incomplete \omterm{def:g11:dna-replication:semi}{replication} would send broken or partial \omterm{prop:g11:dna-replication:genome}{genomes} to
daughters.

\textbf{7.} Frequent \omterm{def:g11:cell-cycle-mitosis:mitosis}{mitosis}: e.g.\ intestinal epithelium, skin epidermis,
bone marrow. Mostly G$_0$: e.g.\ mature neurons, mature skeletal muscle fibres.
\end{solution}
