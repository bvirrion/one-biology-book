\section*{Bab \ref{ch:b1:lipids} --- Lipid}

\begin{solution}{exo:b1:lipids:1}
Sebuah \omterm{def:b1:lipids:fattyacid}{lipid} adalah molekul biologis yang tidak larut dalam air dan
larut dalam pelarut nonpolar. \omterm{def:b1:lipids:triglyceride}{Lemak}, \omterm{def:b1:lipids:amphiphilic}{fosfolipid}, \omterm{def:b1:lipids:amphiphilic}{sterol} dan karotenoid
tidak berbagi kerangka bersama, hanya perilaku ini terhadap air, dan
itulah yang memberi mereka peran bersamanya (simpanan, selaput, membran).
\end{solution}

\begin{solution}{exo:b1:lipids:2}
C18:2 $\Delta^{9,12}$. Ikatan rangkap terakhirnya mulai pada karbon 12
dari karboksilnya, yakni pada karbon 6 dari ujung metilnya: omega-6.
\end{solution}

\begin{solution}{exo:b1:lipids:3}
Sebuah \omterm{def:b1:lipids:triglyceride}{trigliserida} tidak berkepala polar: karena seluruhnya hidrofob,
ia dikucilkan dari air sebagai fase ruah, yaitu sebuah tetesan. Sebuah
\omterm{def:b1:lipids:amphiphilic}{fosfolipid} bersifat \omterm{def:b1:lipids:amphiphilic}{amfifilik}: kepala polarnya harus tinggal dalam air
sementara ekornya harus meninggalkannya, dan hanya selembar setebal dua
molekul yang sanggup memenuhi keduanya.
\end{solution}

\begin{solution}{exo:b1:lipids:4}
Sekitar \qty{41}{\celsius}, \qty{-5}{\celsius} dan \qty{20}{\celsius}
(kurva \omterm{def:b1:lipids:amphiphilic}{kolesterolnya} tidak memiliki transisi yang tajam; separuh
kenaikannya dekat \qty{20}{\celsius}). Sebuah bakteri pada
\qty{10}{\celsius} memerlukan susunan \omterm{def:b1:lipids:fattyacid}{tak jenuhnya}, yang cair jauh di
bawah suhu tumbuhnya.
\end{solution}

\begin{solution}{exo:b1:lipids:5}
$15\,000\times 38 = \qty{570}{MJ}$; $570/8 = 71$ hari. Glikogen: kering
$570\,000/17 = \qty{33.5}{kg}$, terhidrasi \qty{134}{kg}.
\end{solution}

\begin{solution}{exo:b1:lipids:6}
C22:0 $>$ C18:0 $>$ C14:0 $>$ C18:1 $>$ C18:3. Di antara asam \omterm{def:b1:lipids:fattyacid}{jenuh},
rantai yang lebih panjang membuat lebih banyak sentuhan \omterm{def:b1:water-small-molecules:bonds}{van der Waals};
satu ikatan rangkap \emph{cis} menghapus lebih banyak kemasan daripada
yang ditambahkan empat karbon (C18:1 meleleh di bawah C14:0); tiap
ikatan berikutnya menurunkannya lagi.
\end{solution}

\begin{solution}{exo:b1:lipids:7}
$2\times\qty{140}{\micro m^2}/\qty{0.6}{nm^2} = 2\times 1.4\times
10^{-10}/6\times 10^{-19} = \num{4.7e8}$ molekul; massanya
$\num{4.7e8}\times 750/\num{6e23} = \qty{5.8e-13}{g} = \qty{0.58}{pg}$.
\end{solution}

\begin{solution}{exo:b1:lipids:8}
Volumenya \qty{1.11}{mL} $= \qty{1.11e-6}{m^3}$; luas permukaan bolanya
$= 6V/d = 6\times 1.11\times 10^{-6}/10^{-6} = \qty{6.7}{m^2}$. Satu
tetes bervolume itu: $d = \qty{1.28}{cm}$, permukaannya
\qty{5.2}{cm^2}: emulsinya memiliki tiga belas ribu kali lebih banyak
antarmuka. Lipase larut dalam air dan hanya bekerja pada antarmukanya;
lajunya sebanding dengan antarmuka itu.
\end{solution}

\begin{solution}{exo:b1:lipids:9}
Di atas $T_m$ cincin kakunya menghalangi gerak rantai tetangganya dan
mengurangi \omterm{def:b1:lipids:fluidity}{kecairannya}; di bawah $T_m$ kebesarannya mencegah rantainya
berkemas menjadi gel yang teratur dan mencegah pembekuannya. Karena tak
ada pengemasan bekerja sama yang mungkin, tak ada suhu ketika seluruh
dwilapisnya berubah keadaan sekaligus: transisinya terbentang.
\end{solution}

\begin{solution}{exo:b1:lipids:10}
Rantainya \omterm{def:b1:lipids:fattyacid}{jenuh} semua: pada \qty{25}{\celsius} membrannya sudah dekat
pada transisinya dan kaku; pada \qty{5}{\celsius} membrannya menggel.
Membran yang menggel membuat proteinnya berhenti bekerja dan, ketika
dihangatkan kembali atau ditekan secara mekanis, retak dan bocor: \omterm{def:b1:cell-unit-of-life:cell}{selnya}
kehilangan isinya dan \omterm{def:b1:body-plans-tissues:tissue}{jaringannya} mati (cedera dingin). Tipe liarnya
mengurangi kejenuhan \omterm{def:b1:lipids:fattyacid}{lipidnya} pada musim gugur dan tetap cair pada
\qty{5}{\celsius}.
\end{solution}

\begin{solution}{exo:b1:lipids:11}
\qty{1.07}{kg} air tiap kilogram \omterm{def:b1:lipids:triglyceride}{lemak}. Oksigennya: \qty{2000}{L}; pada
penyarian \qty{5}{\%} dari oksigen \qty{21}{\%}, udara yang dibutuhkan
$2000/(0.21\times 0.05) = \qty{190}{m^3}$; air yang hilang pada
\qty{30}{g/m^3}: \qty{5.7}{kg}. Unta itu kehilangan air lima kali lebih
banyak lewat napasnya daripada yang dihasilkan \omterm{def:b1:lipids:triglyceride}{lemaknya}: punuknya adalah
simpanan energi, bukan simpanan air.
\end{solution}

\begin{solution}{exo:b1:lipids:12}
Bila sebuah dwilapis sudah ada, \omterm{prop:b1:water-small-molecules:properties}{efek hidrofob} membuatnya menutup
kembali, mengatup menjadi vesikel dan menerima \omterm{def:b1:lipids:fattyacid}{lipid} baru tanpa masukan
energi: perakitannya spontan. Tetapi \omterm{def:b1:cell-unit-of-life:cell}{sel} membuat \omterm{def:b1:lipids:fattyacid}{lipidnya} dengan enzim
yang duduk dalam sebuah membran yang sudah ada (RE), yang menyisipkannya
di tempat; sebuah membran baru tumbuh lewat pemuaian membran lama dan
dibagi pada pembelahan, sehingga tiap membran tiap \omterm{def:b1:cell-unit-of-life:cell}{sel} menurun dari
membran \omterm{def:b1:cell-unit-of-life:cell}{sel} sebelumnya. \omterm{prop:b1:lipids:selfassembly}{Perakitan diri} menerangkan fisika lembarnya,
bukan asal usulnya di dalam \omterm{def:b1:cell-unit-of-life:cell}{sel}.
\end{solution}

\begin{solution}{pb:b1:lipids:1}
\textbf{1.} C16:0 (0), C18:1 $\Delta^9$ (1), C22:6 $\Delta^{4,7,10,13,16,19}$
(6).
\textbf{2.} Hangat: $0.40\times 0 + 0.45\times 1 + 0.15\times 6 = 1.35$;
dingin: $0 + 0.40 + 2.10 = 2.50$ ikatan rangkap tiap rantai.
\textbf{3.} Tiap ikatan \emph{cis} menekuk rantainya dan mencegah
kemasan rapat dengan tetangganya; sentuhan \omterm{def:b1:water-small-molecules:bonds}{van der Waalsnya} lebih
sedikit, kalor untuk mengacaukan rantainya lebih sedikit, $T_m$-nya
lebih rendah.
\textbf{4.} $1.15$ ikatan lebih banyak tiap rantai: sekitar
\qty{45}{\celsius} lebih rendah.
\textbf{5.} Dalam sebuah gel \omterm{def:b1:lipids:fattyacid}{lipidnya} tidak sanggup bergerak: pembawanya
tidak dapat berubah bentuk, pompanya berhenti, salurannya macet;
gradien ionnya meluruh, penghantaran sarafnya gagal, dan ikannya lumpuh
lalu mati oleh dingin yang ditoleransi ikan yang teraklimatisasi dingin.
\textbf{6.} Desaturase, yang memasukkan ikatan rangkap ke dalam rantai
asil \omterm{def:b1:lipids:triglyceride}{lemak}; enzim itu adalah \omterm{def:b1:membranes-transport:proteins}{protein integral} \omterm{def:b1:eukaryotic-cell:endomembrane}{retikulum endoplasma},
tempat \omterm{def:b1:lipids:fattyacid}{lipidnya} dibuat.
\textbf{7.} Pada keduanya ia menyangga \omterm{def:b1:lipids:fluidity}{kecairannya}: mengakukan membran
dingin yang sangat \omterm{def:b1:lipids:fattyacid}{tak jenuh} di atas $T_m$-nya yang rendah, dan menahan
membran hangat agar tidak menggel pada hari yang sejuk.
\textbf{8.} $30\,000\times 38 = \qty{1140}{MJ}$.
\textbf{9.} $1140/60 = 19$ hari.
\textbf{10.} Kering: $\num{1.14e6}/17 = \qty{67}{kg}$; terhidrasi:
\qty{268}{kg}.
\textbf{11.} $268 - 30 = \qty{238}{kg}$, hampir separuh massa untanya.
\textbf{12.} Selapis \omterm{def:b1:lipids:triglyceride}{lemak} di bawah seluruh kulitnya akan menyekat
tubuhnya dan mencegahnya melepas kalor di gurun; memusatkan \omterm{def:b1:lipids:triglyceride}{lemaknya}
dalam satu punuk membiarkan sisa kulitnya bebas melepas kalor.
\textbf{13.} $1.07\times 30 = \qty{32}{kg}$ air.
\textbf{14.} $2.0\times 30\,000 = \qty{60000}{L} = \qty{60}{m^3}$
oksigen.
\textbf{15.} $60/(0.21\times 0.05) = \qty{5700}{m^3}$ udara.
\textbf{16.} $5700\times 30 = \qty{171}{kg}$ air.
\textbf{17.} Yang hilang lima kali yang diperoleh: mengoksidasi punuknya
memakan biaya air. Punuknya adalah simpanan energi yang membuat unta itu
sanggup bertahan tanpa makanan; penghematan airnya datang dari
ginjalnya, ketahanannya terhadap dehidrasi dan kesanggupannya
membiarkan suhunya naik.
\textbf{18.} $500\times 3.5\times 6 = \qty{10.5}{MJ}$ ditimbun sebagai kalor lalu dilepas pada malam hari; menguapkannya justru akan memakan biaya $10\,500/2.4 = \qty{4.4}{kg}$ air sehari.
\textbf{19.} Sebuah membran hanya sanggup meregang beberapa persen,
sehingga melipatduakan volume sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} merah menuntut kelebihan
membran yang besar yang terlipat dalam bentuk bikonkaf istirahatnya, dan
sebuah \omterm{def:b1:eukaryotic-cell:cytoskeleton}{rangka sel} yang membiarkannya terbuka tanpa robek: \omterm{def:b1:cell-unit-of-life:cell}{sel} merah unta
berbentuk lonjong, dengan membran lebih banyak tiap volumenya daripada
\omterm{def:b1:cell-unit-of-life:cell}{sel} merah manusia.
\textbf{20.} $2\times 80\times 10^{-12}/0.65\times 10^{-18} = \num{2.5e8}$.
\textbf{21.} $\num{2.5e8}\times\num{8e12}\times 40 = \num{7.9e22}$.
\textbf{22.} $\num{7.9e22}\times 750/\num{6e23} = \qty{99}{g}$.
\textbf{23.} Difusi menyamping menjaga kepalanya tetap dalam air dan
ekornya tetap dalam \omterm{def:b1:lipids:triglyceride}{minyak} pada tiap langkahnya, tanpa memakan biaya;
berbalik menuntut kepala polarnya melintasi teras hidrofobnya, sebuah
penghalang energi besar yang hanya diatasi guncangan termal sekali
seminggu.
\textbf{24.} Perbedaan antara kedua lembarnya meluruh hanya secepat
\omterm{def:b1:lipids:fattyacid}{lipidnya} berbalik; pada sekali seminggu tiap molekul, ketaksetangkupan
yang dibangun enzim yang memindahkan \omterm{def:b1:lipids:fattyacid}{lipid} tertentu ke seberangnya
(flipase) bertahan tanpa batas melawan pembalikan spontannya.
\textbf{25.} Sekitar \qty{240}{kg} dihemat --- hampir separuh massa
tubuhnya --- dengan membawa \qty{30}{kg} \omterm{def:b1:lipids:triglyceride}{lemak} alih-alih \qty{270}{kg}
glikogen terhidrasi; dan punuknya bukan simpanan air: mengoksidasinya
kehilangan air lima kali lebih banyak lewat napasnya daripada yang
dihasilkannya.
\end{solution}
