\section*{Chapitre \ref{ch:g4:heart-and-blood} --- Le cœur et le sang}

\begin{solution}{exo:g4:heart-and-blood:1}
Les \omterm{def:g4:journey-of-food:digestion}{nutriments}, recueillis à l'\omterm{def:g4:journey-of-food:tract}{intestin grêle}~; et le dioxygène,
recueilli aux \omterm{def:g4:breathing:organs}{poumons}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:2}
Le dioxyde de carbone --- déposé aux \omterm{def:g4:breathing:organs}{poumons}, où il est expiré.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:3}
C'est un \omterm{def:g3:bones-and-muscles:muscle}{muscle} creux, à peu près de la taille du poing de son
propriétaire, au milieu de la poitrine derrière la cage thoracique.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:4}
Le côté droit envoie le \omterm{prop:g4:journey-of-food:absorb}{sang} aux \omterm{def:g4:breathing:organs}{poumons} charger du dioxygène~; le
côté gauche envoie le \omterm{prop:g4:journey-of-food:absorb}{sang} riche en dioxygène dans tout le corps.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:5}
La petite vague que chaque \omterm{def:g4:heart-and-blood:heart}{battement} pousse le long des vaisseaux,
sentie au \omterm{def:g1:my-body:joint}{poignet} ou au cou. La compter, c'est compter les
\omterm{def:g4:heart-and-blood:heart}{battements} du \omterm{def:g4:heart-and-blood:heart}{cœur} par minute.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:6}
Le pouce a lui-même un petit \omterm{prop:g4:heart-and-blood:pulse}{pouls} bien perceptible~: tu risquerais
de compter ton pouce au lieu du \omterm{def:g1:my-body:joint}{poignet} que tu examines.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:7}
Des \omterm{def:g1:my-body:parts}{jambes} qui sprintent brûlent vite \omterm{def:g4:journey-of-food:digestion}{nutriments} et dioxygène et
doivent être réapprovisionnées vite~; le \omterm{def:g4:heart-and-blood:heart}{cœur} accélère et renforce
ses \omterm{def:g4:heart-and-blood:heart}{battements} pour que le \omterm{prop:g4:journey-of-food:absorb}{sang} fasse ses tournées plus souvent.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:8}
Réponse libre. En général, environ $80$ à $100$ \omterm{def:g4:heart-and-blood:heart}{battements} par
minute au repos, et $120$ à $150$ après les sauts étoilés --- toute
la chaîne d'approvisionnement au travail.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:9}
En quittant les \omterm{def:g4:breathing:organs}{poumons}, elle venait de charger du dioxygène ---
rouge vif. Dans le \omterm{def:g3:bones-and-muscles:muscle}{muscle}, elle a remis ce dioxygène et s'est
chargée de dioxyde de carbone --- plus foncée. La couleur suit la
cargaison.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:10}
L'\omterm{def:g4:journey-of-food:tract}{intestin grêle} met les \omterm{def:g4:journey-of-food:digestion}{nutriments} dans le \omterm{prop:g4:journey-of-food:absorb}{sang}~; les \omterm{def:g4:breathing:organs}{poumons} y
mettent le dioxygène et en retirent le dioxyde de carbone~; le \omterm{def:g4:heart-and-blood:heart}{cœur}
est la pompe qui maintient le \omterm{prop:g4:journey-of-food:absorb}{sang} en mouvement entre eux et jusqu'à
chaque consommateur --- \omterm{def:g3:bones-and-muscles:muscle}{muscles}, cerveau, \omterm{def:g3:bones-and-muscles:skeleton}{os}, \omterm{def:g1:the-five-senses:sense}{peau}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:11}
Le \omterm{def:g4:heart-and-blood:heart}{cœur} entraîné est un \omterm{def:g3:bones-and-muscles:muscle}{muscle} plus fort~: chaque \omterm{def:g4:heart-and-blood:heart}{battement} déplace
plus de \omterm{prop:g4:journey-of-food:absorb}{sang}. Livrer le même approvisionnement demande alors moins
de \omterm{def:g4:heart-and-blood:heart}{battements} par minute --- si bien que les $50$ tranquilles du
cycliste font le travail des $75$ du non-entraîné, avec de la
puissance en réserve pour l'effort.
\end{solution}
