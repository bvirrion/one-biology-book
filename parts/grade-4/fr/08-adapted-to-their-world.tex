\chapter{Adaptés à leur monde}\label{ch:g4:adapted-to-their-world}

Mets un ours blanc au Sahara et un chameau sur la banquise, et
aucun des deux ne tiendrait une saison. Et pourtant, chez lui,
chacun est un champion~: l'ours nage tranquillement dans des mers
glacées, le chameau traverse des déserts qui viendraient à bout
d'un cheval en quelques jours. Les \omterm{def:g1:living-or-not:living}{êtres vivants} s'ajustent à leur
monde avec une exactitude qui mérite d'être étudiée de près ---
caractère par caractère.

\section{Ce que veut dire adaptation}

\begin{definition}[Adaptation]\label{def:g4:adapted-to-their-world:adaptation}
Une \emph{adaptation}\index{adaptation} est un caractère du corps ou
du comportement d'un \omterm{def:g1:living-or-not:living}{être vivant} qui l'ajuste aux conditions de son
milieu --- au froid ou à la sécheresse, à sa nourriture, à ses
chasseurs. \Cref{prop:g2:where-animals-live:fit} montrait les
\omterm{def:g1:animals-around-us:animal}{animaux} ajustés à leur milieu~; l'adaptation est cet ajustement,
examiné pièce par pièce.
\end{definition}

\begin{example}[Le chameau, pièce par pièce]\label{ex:g4:adapted-to-their-world:camel}
Les problèmes du désert~: la chaleur, l'absence d'eau, le sable
mou. Les réponses du chameau~: des bosses de graisse --- une \omterm{def:g3:animals-through-seasons:active}{réserve
de nourriture} pour les traversées maigres~; la capacité de boire
cent litres en quelques minutes puis de tenir des jours sans~; de
larges pieds à deux doigts qui ne s'enfoncent pas dans le sable~; des
narines qui se ferment contre la poussière soufflée et de longs cils
doubles assortis. Chaque problème du désert, réglé par un caractère.
\end{example}

\begin{example}[L'ours blanc, pièce par pièce]\label{ex:g4:adapted-to-their-world:bear}
Les problèmes de la banquise~: le froid, l'eau glacée, des proies
difficiles à approcher sur la glace blanche. Les réponses de
l'ours~: une épaisse fourrure sur une profonde couche de graisse ---
une double isolation~; des pattes larges et légèrement palmées ---
raquettes et pagaies à la fois~; un \omterm{def:g1:animals-around-us:covering}{pelage} blanc --- invisible sur
la neige où il chasse. La même exactitude, réglée sur le monde
inverse.
\end{example}

\begin{proposition}[Les adaptations répondent à des problèmes]\label{prop:g4:adapted-to-their-world:answers}
Partout où le milieu pose un problème difficile, les \omterm{def:g1:living-or-not:living}{êtres vivants}
qui y sont chez eux portent des caractères qui y répondent~:
\begin{itemize}
  \item le froid~: fourrure épaisse, couches de graisse, petites
        \omterm{def:g1:the-five-senses:sense}{oreilles} qui perdent peu de chaleur~;
  \item la sécheresse~: réserves d'eau, peaux cireuses, \omterm{def:g1:plants-around-us:parts}{racines}
        profondes~;
  \item la vie dans l'eau~: corps fuselés, nageoires et pieds
        palmés, \omterm{def:g1:animals-around-us:covering}{pelages} qui laissent glisser l'eau~;
  \item des chasseurs tout autour~: couleurs de camouflage,
        armures, piquants, vitesse, et \omterm{def:g1:the-five-senses:sense}{yeux} écartés pour surveiller
        de tous côtés~;
  \item chasser pour vivre~: sens aiguisés, \omterm{def:g1:the-five-senses:sense}{yeux} vers l'avant,
        griffes et vitesse
        (\cref{prop:g2:what-animals-eat:match}).
\end{itemize}
\end{proposition}
\admitted

\section{Les plantes aussi sont adaptées}

\begin{example}[Le cactus]\label{ex:g4:adapted-to-their-world:cactus}
C'est aux \omterm{def:g1:plants-around-us:plant}{plantes} que le désert impose l'épreuve la plus dure ---
elles ne peuvent pas marcher jusqu'à l'eau. Le cactus répond~: une
\omterm{def:g1:plants-around-us:parts}{tige} épaisse en forme de tonneau qui stocke l'eau et fait le travail
de capture de la lumière à la place des \omterm{def:g1:plants-around-us:parts}{feuilles}~; des \omterm{def:g1:plants-around-us:parts}{feuilles}
réduites à des épines, qui ne perdent presque pas d'eau --- et qui
défendent le tonneau contre les \omterm{def:g1:animals-around-us:animal}{animaux} assoiffés~; des \omterm{def:g1:plants-around-us:parts}{racines}
étalées largement et près de la surface pour capter d'un coup toute
une averse rare.
\end{example}

\begin{example}[Les coussins d'altitude]\label{ex:g4:adapted-to-their-world:alpine}
Les \omterm{def:g1:plants-around-us:plant}{plantes} de haute montagne affrontent un vent glacial. Beaucoup
poussent en coussins serrés, collés au sol~: à l'intérieur du
coussin, le vent est brisé et l'air reste plus doux — une forme qui
est un abri. Les cousines de plaine des mêmes \omterm{def:g1:plants-around-us:plant}{plantes} poussent
hautes et lâches~; les formes de montagne épousent le rocher.
\end{example}

\begin{method}[Lire le monde d'un organisme sur son corps]\label{met:g4:adapted-to-their-world:read}
Devant un \omterm{def:g1:animals-around-us:animal}{animal} ou une \omterm{def:g1:plants-around-us:plant}{plante} que tu ne connais pas~:
\begin{enumerate}
  \item énumère ses caractères frappants --- \omterm{def:g1:animals-around-us:covering}{pelage}, pieds, \omterm{def:g1:the-five-senses:sense}{yeux},
        forme du corps, \omterm{def:g1:plants-around-us:parts}{feuilles}, \omterm{def:g1:plants-around-us:parts}{racines}~;
  \item pour chacun, demande-toi~: \emph{quel problème cela
        résoudrait-il~?}
  \item assemble les réponses en une image de son foyer --- froid
        ou chaud, humide ou sec, chasseur ou chassé~;
  \item vérifie ta supposition en regardant où l'organisme vit
        vraiment.
\end{enumerate}
C'est \cref{met:g2:where-animals-live:matching} devenue grande~: de
l'appariement à un milieu à la reconstruction d'un monde.
\end{method}

\begin{example}[La méthode sur une peau mystère]\label{ex:g4:adapted-to-their-world:mystery}
Un tiroir de muséum contient une \omterm{def:g1:the-five-senses:sense}{peau} sans étiquette~: fourrure
dense et douce, tout petits \omterm{def:g1:the-five-senses:sense}{yeux}, énormes griffes avant en forme de
pelle. La fourrure dit mammifère~; les pelles disent creuser~; les
\omterm{def:g1:the-five-senses:sense}{yeux} inutiles disent une vie dans le noir. Verdict~: un fouisseur
souterrain --- et c'est bien une taupe
(\cref{ex:g2:where-animals-live:mole}), lue correctement à partir de
trois caractères seulement.
\end{example}

\begin{remark}[D'où vient cet ajustement~?]\label{rem:g4:adapted-to-their-world:how}
Un chameau ne \emph{décide} jamais de se faire pousser de larges
pieds, et aucun ours blanc ne choisit son \omterm{def:g1:animals-around-us:covering}{pelage} blanc. Comment,
alors, les \omterm{def:g1:living-or-not:living}{êtres vivants} en viennent-ils à s'ajuster si bien à leur
monde~? C'est l'une des questions les plus profondes de la \omterm{def:g1:living-or-not:living}{biologie},
et sa réponse complète --- comment l'ajustement se construit,
génération après génération --- est une histoire que ce livre garde
pour une année ultérieure. Pour l'instant, apprends à \emph{voir}
l'ajustement~: c'est ce regard qui a rendu la question possible.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:1}
Qu'est-ce qu'une \omterm{def:g4:adapted-to-their-world:adaptation}{adaptation}~? Donne un exemple \omterm{def:g1:animals-around-us:animal}{animal} et un exemple
végétal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:2}
Nomme trois problèmes du désert et la réponse du chameau à chacun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:3}
Nomme trois problèmes de la banquise et la réponse de l'ours blanc à
chacun.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:4}
Quels travaux font les épines du cactus~? Que stocke sa \omterm{def:g1:plants-around-us:parts}{tige} en
tonneau, et quel travail cette \omterm{def:g1:plants-around-us:parts}{tige} a-t-elle repris~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:5}
Pourquoi beaucoup de \omterm{def:g1:plants-around-us:plant}{plantes} de haute montagne poussent-elles en
coussins serrés~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:6}
Donne deux \omterm{def:g4:adapted-to-their-world:adaptation}{adaptations} d'\omterm{def:g1:animals-around-us:animal}{animaux} chassés et deux de chasseurs,
d'après \cref{prop:g4:adapted-to-their-world:answers}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:7}
Raconte \cref{ex:g4:adapted-to-their-world:mystery}~: quels trois
caractères ont trahi la taupe, et que disait chacun d'eux~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:adapted-to-their-world:8}
À toi d'essayer~: choisis un \omterm{def:g1:animals-around-us:animal}{animal} que tu peux observer --- un
canard, un chat, une araignée --- et applique-lui
\cref{met:g4:adapted-to-their-world:read}~: trois caractères, trois
problèmes résolus.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:9}
Le fennec du désert a d'énormes \omterm{def:g1:the-five-senses:sense}{oreilles}~; le renard polaire, de
toutes petites. À l'aide de la ligne sur le froid de
\cref{prop:g4:adapted-to-their-world:answers}, explique les deux ---
que doivent faire de la chaleur de grandes \omterm{def:g1:the-five-senses:sense}{oreilles}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:adapted-to-their-world:10}
Un \omterm{def:g1:animals-around-us:animal}{animal} mystère~: corps fuselé, pieds arrière palmés, fourrure
dense qui laisse glisser l'eau, moustaches, et il retient bien son
souffle. Reconstruis son monde avec
\cref{met:g4:adapted-to-their-world:read}. (Il construit des
barrages.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:adapted-to-their-world:11}
La baleine (\cref{ex:g3:sorting-living-things:surprises}) est un
mammifère en forme de poisson~: fuselée, munie de nageoires, lisse.
La truite est un poisson de la même forme. Que montre cette
ressemblance entre deux nageurs \emph{sans parenté} au sujet des
\omterm{def:g4:adapted-to-their-world:adaptation}{adaptations} --- qui a fixé la forme, la famille ou l'eau~?
\end{exercise}
