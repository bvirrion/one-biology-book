\chapter{Leven door de seizoenen}\label{ch:g6:life-through-seasons}

Fotografeer dezelfde kastanjeboom op de eerste van elke maand en prik de
twaalf foto's op een rij: kale takken, \omterm{def:g6:life-through-seasons:buds}{knoppen}, \omterm{def:g1:plants-around-us:parts}{bladeren}, kaarsen van
\omterm{def:g3:flowers-fruits-seeds:parts}{bloemen}, kastanjes, goud, weer kale takken. Er bewoog niets --- de boom
stond het hele jaar stil --- en toch veranderde alles.
\Cref{ch:g3:animals-through-seasons} volgde de \omterm{def:g1:animals-around-us:animal}{dieren} door het jaar; nu
volgen we, met de ogen van een onderzoeker, de hele \omterm{def:g6:exploring-our-environment:components}{omgeving}, om te
beginnen de \omterm{def:g1:plants-around-us:plant}{planten}.

\section{De omgeving verandert met de seizoenen}

\begin{proposition}[Seizoensverandering van de omstandigheden]\label{prop:g6:life-through-seasons:conditions}
Door het jaar heen schommelen de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} van een \omterm{def:g6:exploring-our-environment:components}{omgeving}
(\cref{def:g6:exploring-our-environment:conditions}) sterk: de dagen
korten en lengen, de temperaturen dalen en stijgen, de bodemvochtigheid
volgt regen en droogte. Een locatie die in januari en in juni onderzocht
wordt, geeft twee verschillende rijen in de tabel --- dus verwacht je,
volgens \cref{prop:g6:exploring-our-environment:decide}, twee
verschillende bezettingen van bewoners.
\end{proposition}
\admitted

\begin{example}[Eén heg, twee onderzoeken]\label{ex:g6:life-through-seasons:hedge}
De heg langs het weggetje, twee keer onderzocht. Juni: dicht \omterm{def:g1:plants-around-us:parts}{blad}, tien
plantensoorten bloeiend aan zijn voet, luide bijen, nestelende merels,
slakken na de regen. Januari: kale twijgen, groen teruggebracht tot
klimop en mos, geen vliegend insect, merels die in de \omterm{def:g1:plants-around-us:parts}{bladeren} op de
grond scharrelen, slakken diep in de berm in hun huisje afgesloten.
Dezelfde plek, dezelfde lijst bestanddelen --- levend, mineraal, sporen
--- maar de kolom van het levende is van gedaante veranderd.
\end{example}

\section{Hoe planten het slechte seizoen doorkomen}

\begin{definition}[Bladverliezend en groenblijvend]\label{def:g6:life-through-seasons:trees}
Bomen vallen uiteen naar hun winterantwoord:
\begin{itemize}
  \item \emph{bladverliezende}\index{bladverliezend} bomen --- eik,
        kastanje, beuk --- laten in de herfst al hun \omterm{def:g1:plants-around-us:parts}{bladeren} vallen en
        staan kaal;
  \item \emph{groenblijvende}\index{groenblijvend} bomen --- den, hulst,
        klimop tegen de muur --- houden de hele winter werkende \omterm{def:g1:plants-around-us:parts}{bladeren},
        meestal taaie, wasachtige of naaldvormige
        (\cref{prop:g3:sorting-living-things:plants} kwam de naalden al
        tegen).
\end{itemize}
\end{definition}

\begin{definition}[Knoppen]\label{def:g6:life-through-seasons:buds}
Een \omterm{def:g6:life-through-seasons:trees}{bladverliezende} boom is in de winter niet dood: de volgende lente zit
al op zijn twijgen ingepakt als \emph{knoppen}\index{knop} ---
piepkleine complete scheuten, met de \omterm{def:g1:plants-around-us:parts}{bladeren} klein opgevouwen, verpakt
in weerbestendige \omterm{def:g1:animals-around-us:covering}{schubben}. De ``nieuwe'' \omterm{def:g1:plants-around-us:parts}{bladeren} van de lente zijn de
zomer ervoor gebouwd; de knop is er de winteropslag van.
\end{definition}

\begin{example}[Een paardenkastanjeknop openen]\label{ex:g6:life-through-seasons:bud}
De grote kleverige \omterm{def:g6:life-through-seasons:buds}{knoppen} van de paardenkastanje gaan goed open onder
een mesje (werk voor een volwassene): onder de gelakte \omterm{def:g1:animals-around-us:covering}{schubben}, in
katoenachtig dons verpakt, liggen miniatuurbladeren opgevouwen ---
volmaakt, bleek, wachtend. Een twijg die je in februari afknipt en
binnen in water zet, ``loopt weken te vroeg uit'': de scheuten waren af;
alleen de warmte ontbrak nog.
\end{example}

\begin{proposition}[De winterantwoorden van de kruidlaag]\label{prop:g6:life-through-seasons:herbs}
Kleinere \omterm{def:g1:plants-around-us:plant}{planten} hebben hun eigen strategieën:
\begin{itemize}
  \item \emph{eenjarigen}\index{eenjarige plant} --- de klaproos, de
        boon --- gaan in de herfst helemaal dood en brengen de winter
        door als \emph{\omterm{def:g2:from-seed-to-plant:seed}{zaad}} (\cref{prop:g2:from-seed-to-plant:needs}:
        een \omterm{def:g2:from-seed-to-plant:seed}{zaad} wacht de kou veilig af);
  \item \emph{overblijvende planten}\index{overblijvende plant} --- de
        narcis, de paardenbloem --- laten hun bovengrondse delen
        afsterven en wachten ondergronds als \omterm{ex:g4:plants-without-seeds:bulbtuber}{bol}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{knol} of taaie
        wortelstok (de voorraadkamers uit
        \cref{ex:g4:plants-without-seeds:bulbtuber}, in winterdienst);
  \item groenblijvers van de bodem --- mos, klimop --- houden het gewoon
        vol.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Waar zijn de narcissen gebleven?]\label{ex:g6:life-through-seasons:daffodil}
Het narcissenveldje bloeit in maart, wordt in mei geel, en is in juli
spoorloos verdwenen --- en toch komt het volgende maart terug, dichter
dan eerst. De hele zomer, herfst en winter bestond het als bollen onder
het gazon: de \omterm{def:g1:plants-around-us:plant}{plant} is nooit weggegaan; ze trok zich terug in de opslag.
\end{example}

\section{De kalender van de dieren, opnieuw bekeken}

\begin{proposition}[Dierantwoorden, in onderzoekstermen]\label{prop:g6:life-through-seasons:animals}
De drie strategieën uit \cref{ch:g3:animals-through-seasons} --- trekken,
\omterm{def:g3:animals-through-seasons:hibernation}{winterslaap}, actief blijven --- lezen nu als antwoorden op de gemeten
veranderingen: elke \omterm{def:g5:biodiversity:species}{soort} volgt de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} en het voedsel die zij
nodig heeft.
\begin{itemize}
  \item de zwaluw volgt haar \omterm{def:g3:animals-through-seasons:active}{voedselvoorraad} naar waar de
        \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} \omterm{prop:g3:sorting-living-things:groups}{insecten} aan het vliegen houden;
  \item de egel en de afgesloten slak wachten de maanden af waarin hun
        lichaam niet kan doorwerken;
  \item \omterm{prop:g3:sorting-living-things:groups}{insecten} voegen een vierde antwoord toe: de meeste brengen de
        winter door als \emph{\omterm{def:g2:animals-grow-up:born}{ei}, larve of \omterm{ex:g2:animals-grow-up:butterfly}{pop}}
        (\cref{ex:g2:animals-grow-up:butterfly}), weggestopt in de
        bodem, de bast of de bladlaag --- het volwassen \omterm{def:g1:animals-around-us:animal}{dier} gaat dood,
        de \omterm{def:g5:biodiversity:species}{soort} overwintert als haar jonge stadia.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Waar de insecten van juni in januari zijn]\label{ex:g6:life-through-seasons:insects}
De heg van januari laat geen enkel insect zien --- en toch zijn die van
juni er allemaal: de vlinders als poppen onder de afdekstenen, de
sprinkhanen als eitjes in de bodem, de lieveheersbeestjes als
opeengedrukte slapende volwassenen diep in de bladlaag, de bijen als een
strakke warme tros in de kast. Het onderzoek van juni loopt rond; dat van
januari moet graven, optillen en turen --- de bewoners zijn van
\emph{vorm} veranderd, niet van adres.
\end{example}

\begin{method}[Het jaar rond onderzoeken]\label{met:g6:life-through-seasons:yearsurvey}
Om het seizoensleven van een locatie te bestuderen:
\begin{enumerate}
  \item onderzoek dezelfde gemarkeerde locatie
        (\cref{met:g6:exploring-our-environment:survey}) één keer per
        seizoen --- hetzelfde vierkant, dezelfde instrumenten, dezelfde
        zoektijd;
  \item noteer elke keer de \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} \emph{en} de bewoners;
        fotografeer de locatie vanaf hetzelfde punt;
  \item zoek bewust naar de verborgen vormen van de winter: \omterm{def:g6:life-through-seasons:buds}{knoppen},
        bollen, eitjes, poppen, afgesloten slakken --- en leg elke
        afdekking terug;
  \item leg de vier onderzoeken naast elkaar: koppel elke verandering in
        bewoners aan een verandering in \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}, en noem van elke
        afwezige de wintervorm of de bestemming.
\end{enumerate}
\end{method}

\begin{remark}[Niets verdwijnt zomaar]\label{rem:g6:life-through-seasons:vanish}
De tabel van het hele jaar leert één diepe regel: geen enkele \omterm{def:g5:biodiversity:species}{soort} uit
het juni-onderzoek is zomaar opgehouden te bestaan. Elk ervan is elders
(getrokken), in \omterm{prop:g1:healthy-habits:needs}{slaap} (\omterm{def:g3:animals-through-seasons:hibernation}{winterslaap}), in opslag (\omterm{ex:g4:plants-without-seeds:bulbtuber}{bol}, \omterm{def:g2:from-seed-to-plant:seed}{zaad}) of in een
ander levensstadium (\omterm{def:g2:animals-grow-up:born}{ei}, \omterm{ex:g2:animals-grow-up:butterfly}{pop}). ``Weg'' is in de \omterm{def:g1:living-or-not:living}{biologie} nooit een
volledig antwoord --- de vervolgvraag luidt altijd \emph{weg waarheen,
als wat?}
\end{remark}

\section{Probleem: De twaalf foto's}

\begin{problem}[{Weekendprobleem --- een jaar van de kastanjeboom, beeld
voor beeld gelezen}]\label{pb:g6:life-through-seasons:1}
Lena fotografeert een jaar lang op de eerste van elke maand de
kastanjeboom bij het hek, en prikt de twaalf afdrukken op een rij. Onder
elke afdruk noteert ze de daglengte en de \omterm{def:g6:exploring-our-environment:conditions}{temperatuur} van die ochtend.

\medskip\noindent\textbf{Deel I --- De rij lezen.}
\begin{enumerate}
  \item Januari en december laten kale takken zien. Wat op de twijgen
        bewijst dat de boom in beide beelden leeft en klaar is voor de
        lente?
  \item Tussen welke twee beelden springen de \omterm{def:g6:life-through-seasons:buds}{knoppen} open? Lena's
        aantekeningen luiden daar: dagen van $11$--$13$ uur, ochtenden
        van $8$--\qty{12}{\degreeCelsius}. Op welke veranderende
        \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden} antwoordt de boom?
  \item Het meibeeld laat rechtopstaande ``kaarsen'' van witte \omterm{def:g3:flowers-fruits-seeds:parts}{bloemen}
        zien. Wat moet er met \cref{ch:g3:flowers-fruits-seeds} (of
        \cref{ch:g5:flower-to-fruit}) tussen het mei- en het
        septemberbeeld langskomen wil er in oktober kastanjes bestaan
        --- en wat is een kastanje precies?
  \item Het oktoberbeeld laat gouden \omterm{def:g1:plants-around-us:parts}{bladeren} zien; dat van november
        kale takken en een bruin tapijt eronder. Noem het type boom uit
        \cref{def:g6:life-through-seasons:trees}, en waar zijn volgende
        lente in de beelden erna is opgeslagen.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De buren van de boom.}
\begin{enumerate}[resume]
  \item Onder de boom vangt het maartbeeld een plek met narcissen die op
        het julibeeld ontbreekt. Waar is die plek in juli, en in welke
        vorm?
  \item Klaprozen zomen in juni het pad; op het januaribeeld is hun
        plek kale aarde. In welke vorm is de klaproos in januari
        aanwezig, en volgens welke strategie uit
        \cref{prop:g6:life-through-seasons:herbs}?
  \item Het junibeeld vangt een zwaluw boven de boom en bijen op de
        \omterm{def:g3:flowers-fruits-seeds:parts}{bloemen} eronder; geen van beide verschijnt van november tot
        februari. Zeg van elk waar het dan is, en als wat.
  \item Op elk beeld staat een ekster. Welke dierstrategie is dat, en
        wat moet de ekster de hele winter zien te klaren waar de zwaluw
        met vertrekken aan ontsnapt?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De regel van de rij.}
\begin{enumerate}[resume]
  \item Lena vat samen: ``het jaar van de boom volgt het jaar van de
        \omterm{def:g6:exploring-our-environment:conditions}{omstandigheden}''. Ondersteun haar met twee koppelingen van een
        verandering van beeld tot beeld in de boom aan een verandering
        in haar aantekeningen over daglengte en \omterm{def:g6:exploring-our-environment:conditions}{temperatuur}.
  \item Haar broertje zegt dat het narcissenveldje in juli ``dood'' is.
        Verbeter hem met
        \cref{rem:g6:life-through-seasons:vanish} --- en geef de
        vervolgvraag die zijn woord ``weg'' altijd verdient.
  \item Voorspel beeld 13 --- de foto van volgend jaar januari --- in
        detail: takken, twijgen, de grond eronder. Welke delen van je
        voorspelling zijn zeker genoeg om op te wedden, en waarom?
  \item Een kastanjetwijg die in februari geknipt wordt en binnen in
        warm water wordt gezet, loopt in drie weken uit. Wat laat dat
        zien over \emph{welke} omstandigheid de \omterm{def:g6:life-through-seasons:buds}{knoppen} stond af te
        wachten --- en ontwerp de eerlijke tweelingtwijgproef
        (\cref{met:g4:what-plants-need:fairtest}) die het zou vastpinnen.
\end{enumerate}
\end{problem}
