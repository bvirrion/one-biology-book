\chapter{A reprodução e o ambiente}\label{ch:g8:reproduction-and-environment}

Por que o coro dos sapos começa em maio e não em novembro? Por que um
bom ano de faia enche a mata de \omterm{prop:g2:from-seed-to-plant:cycle}{plântulas}, e um junho frio e chuvoso
esvazia de sapinhos a lagoa da primavera seguinte? O capítulo passado
construiu o mecanismo da geração seguinte; este o liga ao mundo: a
reprodução é regulada, provida --- e limitada --- pelo \omterm{def:g6:exploring-our-environment:components}{ambiente}.

\section{Regulada pelas estações}

\begin{proposition}[A reprodução segue o calendário da fartura]\label{prop:g8:reproduction-and-environment:timing}
As \omterm{def:g5:biodiversity:species}{espécies} regulam a reprodução delas para que os \emph{filhotes}
cheguem quando as \omterm{def:g6:exploring-our-environment:conditions}{condições} e o alimento os servirem
(\cref{prop:g6:life-through-seasons:conditions}):
\begin{itemize}
  \item o chapim põe os \omterm{def:g2:animals-grow-up:born}{ovos} de modo que os filhotes nasçam no pico de
        \omterm{ex:g2:animals-grow-up:butterfly}{lagartas} da primavera (a cadeia do carvalho do
        \cref{ex:g3:food-chains:three}, ajustada ao dia);
  \item o sapo desova na lagoa que esquenta, dando aos \omterm{ex:g2:animals-grow-up:frog}{girinos} um
        verão de crescimento antes do inverno;
  \item as \omterm{def:g1:plants-around-us:plant}{plantas} florescem na estação dos polinizadores delas e
        amadurecem as \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} no calendário da \omterm{def:g6:how-plants-spread:dispersal}{dispersão}
        (\cref{pb:g6:life-through-seasons:1});
  \item o berro do cervo no outono marca os \omterm{def:g2:born-grow-die:cycle}{nascimentos} para a
        \emph{primavera} seguinte --- com a gestação já contada.
\end{itemize}
As deixas que acionam a maquinaria são os próprios sinais do
\omterm{def:g6:exploring-our-environment:components}{ambiente}: dias que se alongam acima de tudo, calor, chuva.
\end{proposition}
\admitted

\begin{example}[Os cordeiros de galpão]\label{ex:g8:reproduction-and-environment:lambs}
Quem cria ovelhas explora a deixa diretamente: as ovelhas entram em
condição reprodutiva conforme os dias \emph{encurtam}; mantido sob \omterm{def:g6:exploring-our-environment:conditions}{luz}
artificial primeiro longa e depois encurtada, um rebanho pode ser
levado a parir a tempo do mercado lucrativo do início da temporada. A
maquinaria lê a \omterm{def:g6:exploring-our-environment:conditions}{luz}, e não o calendário --- mude a \omterm{def:g6:exploring-our-environment:conditions}{luz} e o calendário
segue. Uma deixa conhecida é uma alavanca.
\end{example}

\section{Provida pelo hábitat}

\begin{proposition}[A reprodução é paga em matéria]\label{prop:g8:reproduction-and-environment:cost}
\omterm{def:g2:animals-grow-up:born}{Ovos}, \omterm{def:g2:animals-grow-up:born}{leite}, néctar, \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} e chifres são todos \emph{produção} no
sentido da \cref{def:g6:matter-from-food:production} --- \omterm{def:g6:plants-are-producers:matter}{matéria
orgânica} que os pais precisam obter antes. Então a riqueza do \omterm{def:g2:where-animals-live:habitat}{hábitat}
define o orçamento:
\begin{itemize}
  \item uma galinha bem alimentada põe todo dia; uma faminta para ---
        o balanço do \cref{pb:g6:matter-from-food:1} fechando na linha
        dos \omterm{def:g2:animals-grow-up:born}{ovos};
  \item a faia frutifica em massa só nos ``anos de safra'', depois de
        boas estações de crescimento --- as safras de \omterm{def:g2:from-seed-to-plant:seed}{semente} são
        verões guardados
        (\cref{ex:g6:plants-are-producers:pantry});
  \item os anos duros reduzem a ninhada: muitas \omterm{prop:g3:sorting-living-things:groups}{aves} põem menos \omterm{def:g2:animals-grow-up:born}{ovos},
        ou nenhum, quando a presa está escassa.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Contando a ninhada dos chapins]\label{ex:g8:reproduction-and-environment:clutch}
Os estudos com caixas de ninho mostram as ninhadas dos chapins
acompanhando a oferta de \omterm{ex:g2:animals-grow-up:butterfly}{lagartas}: em carvalhais ricos, dez \omterm{def:g2:animals-grow-up:born}{ovos} ou
mais; em quintais pobres, seis ou sete. A ave não decide isso com um
balanço na mão --- a condição do corpo dela, construída com o alimento
do \omterm{def:g2:where-animals-live:habitat}{hábitat}, define o que ela consegue prover. O tamanho da ninhada é a
qualidade do \omterm{def:g2:where-animals-live:habitat}{hábitat}, escrita em \omterm{def:g2:animals-grow-up:born}{ovos}.
\end{example}

\section{Limitada pelo mundo: as populações}

\begin{definition}[População]\label{def:g8:reproduction-and-environment:population}
Uma \emph{população}\index{população} são todos os indivíduos de uma
\omterm{def:g5:biodiversity:species}{espécie} que vivem num mesmo lugar --- os sapos da lagoa, as faias da
mata. A reprodução acrescenta; as mortes e as partidas subtraem; o
\omterm{def:g6:exploring-our-environment:components}{ambiente} define o teto.
\end{definition}

\begin{proposition}[Por que as populações não explodem]\label{prop:g8:reproduction-and-environment:limits}
Toda \omterm{def:g5:biodiversity:species}{espécie} produz em excesso --- milhares de \omterm{def:g2:animals-grow-up:born}{ovos} por casal de
sapos, milhares de \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} por faia
(\cref{prop:g4:animal-reproduction:strategies}) --- e, mesmo assim, as
lagoas não ficam sólidas de sapo. O excedente é gasto contra os freios
do \omterm{def:g6:exploring-our-environment:components}{ambiente}:
\begin{itemize}
  \item a \emph{predação} --- as garças e as larvas de libélula do
        \cref{ex:g4:animal-reproduction:pondcount};
  \item o \emph{alimento e o espaço} --- só existe tanta \omterm{ex:g2:animals-grow-up:butterfly}{lagarta},
        tanto buraco de ninho, tanta clareira de \omterm{def:g6:exploring-our-environment:conditions}{luz} no chão da mata;
  \item as \emph{\omterm{def:g6:exploring-our-environment:conditions}{condições}} --- geada tardia nas \omterm{def:g3:flowers-fruits-seeds:parts}{flores}
        (\cref{exo:g3:flowers-fruits-seeds:9}), seca na lagoa que
        encolhe com os \omterm{ex:g2:animals-grow-up:frog}{girinos} dentro.
\end{itemize}
Em média, num \omterm{def:g2:where-animals-live:habitat}{hábitat} estável, cada casal substitui a si mesmo --- o
equilíbrio da \cref{prop:g5:food-webs:balance}, visto do lado dos
pais.
\end{proposition}
\admitted

\begin{example}[Duas primaveras na lagoa]\label{ex:g8:reproduction-and-environment:twosprings}
Primavera 1, amena e chuvosa: água alta, \omterm{prop:g3:sorting-living-things:groups}{insetos} cedo --- milhares de
\omterm{def:g2:animals-grow-up:born}{ovos}, centenas de sapinhos, a \omterm{def:g8:reproduction-and-environment:population}{população} sobe. Primavera 2, uma seca: a
lagoa encolhe até virar uma poça, os \omterm{ex:g2:animals-grow-up:frog}{girinos} se amontoam, o \omterm{def:g7:what-is-respiration:respiration}{oxigênio}
cai (\cref{prop:g7:oxygen-in-the-water:drains}), as garças se
banqueteiam no cardume concentrado --- um punhado de sapinhos. Os
mesmos sapos, o mesmo mecanismo, dois vereditos: o \omterm{def:g6:exploring-our-environment:components}{ambiente} assina
embaixo de cada geração.
\end{example}

\begin{remark}[Os seres humanos mexem nos freios]\label{rem:g8:reproduction-and-environment:humans}
Quase todos os efeitos humanos sobre a vida selvagem passam por esses
freios, e não pelo mecanismo central: lagoas aterradas e cercas vivas
arrancadas removem o \omterm{def:g2:where-animals-live:habitat}{hábitat} de reprodução
(\cref{prop:g2:where-animals-live:loss}); os pesticidas esvaziam o
pico de \omterm{ex:g2:animals-grow-up:butterfly}{lagartas} a que os chapins se ajustaram; as caixas de ninho de
quintal e os campos alagados restaurados do
\cref{ex:g5:biodiversity:storks} devolvem teto. Proteger uma \omterm{def:g5:biodiversity:species}{espécie}
na prática quer dizer proteger o \omterm{def:g6:exploring-our-environment:components}{ambiente} em que a reprodução dela
está ligada --- deixa, orçamento e freios, tudo junto.
\end{remark}

\begin{method}[Auditar uma mudança de população]\label{met:g8:reproduction-and-environment:audit}
Quando uma \omterm{def:g8:reproduction-and-environment:population}{população} sobe ou cai, audite nesta ordem:
\begin{enumerate}
  \item o \emph{tempo}: a deixa e o pico do recurso ainda coincidiam?
        (uma primavera cedo demais pode deixar os filhotes na mão
        depois das \omterm{ex:g2:animals-grow-up:butterfly}{lagartas});
  \item o \emph{orçamento}: o que o \omterm{def:g2:where-animals-live:habitat}{hábitat} da estação permitiu que
        os pais provessem?
  \item os \emph{freios}: qual deles mudou --- predadores, alimento,
        espaço, \omterm{def:g6:exploring-our-environment:conditions}{condições}?
  \item o \emph{\omterm{def:g2:where-animals-live:habitat}{hábitat}}: o próprio terreno de reprodução foi ganho
        ou perdido?
\end{enumerate}
\end{method}

\section{Problema: O balanço das caixas de ninho}

\begin{problem}[{Problema de fim de semana --- dez anos de chapins,
auditados caixa por caixa}]\label{pb:g8:reproduction-and-environment:1}
Uma escola monitora vinte caixas de ninho durante dez anos: datas de
postura, tamanhos de ninhada, filhotes que saem do ninho por caixa ---
ao lado do registro do tempo e do diário do zelador.

\medskip\noindent\textbf{Parte I --- Lendo os anos
normais.}
\begin{enumerate}
  \item Anos 1 a 4: postura no fim de abril, ninhadas de oito a dez
        \omterm{def:g2:animals-grow-up:born}{ovos}, quase todos os filhotes saindo do ninho. Diga a deixa
        que marca o fim de abril e o pico a que esse tempo mira.
  \item As ninhadas do ano 3 saíram um ou dois \omterm{def:g2:animals-grow-up:born}{ovos} maiores, depois
        de um maio quente e cheio de \omterm{ex:g2:animals-grow-up:butterfly}{lagartas} no ano 2 --- os pais
        passaram bem o inverno. Que proposição essa correlação
        ilustra?
  \item Mesmo nos bons anos, cerca de um quinto dos \omterm{def:g2:animals-grow-up:born}{ovos} nunca chega
        a sair do ninho. Liste os freios provavelmente responsáveis,
        a partir da
        \cref{prop:g8:reproduction-and-environment:limits}.
  \item Por que a escola registra o tempo e os serviços do terreno ao
        lado dos \omterm{def:g2:animals-grow-up:born}{ovos}? (Responda com as quatro perguntas da
        auditoria.)
\end{enumerate}

\medskip\noindent\textbf{Parte II --- As anomalias.}
\begin{enumerate}[resume]
  \item Ano 5: um março estranhamente quente trouxe as \omterm{ex:g2:animals-grow-up:butterfly}{lagartas} duas
        semanas antes; as datas de postura quase não mudaram. Preveja
        o sucesso de ninho do ano 5 e diga a linha de falha da
        auditoria.
  \item Ano 6: o diário do zelador registra a retirada dos \omterm{def:g1:plants-around-us:tree}{galhos}
        mortos dos carvalhos velhos e a faixa de campo ceifada todo
        mês. O sucesso de ninho cai, embora o tempo tenha sido
        bondoso. Que linhas da auditoria o diário implica?
  \item Ano 7: um casal de gaviões-miúdos se instala. As ninhadas não
        mudam, mas a contagem de filhotes que saem do ninho cai. Que
        freio se moveu, e por que a linha da \emph{ninhada} não se
        moveu junto?
  \item Ano 8: as caixas dobram para quarenta; o total de filhotes
        sobe, mas o sucesso \emph{por caixa} continua estável. Que
        teto as caixas novas levantaram, e qual elas deixaram
        intocado?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O relatório.}
\begin{enumerate}[resume]
  \item Escreva a frase de resumo dos dez anos ligando o tamanho da
        ninhada à qualidade do \omterm{def:g2:where-animals-live:habitat}{hábitat}, citando o
        \cref{ex:g8:reproduction-and-environment:clutch}.
  \item A prefeitura propõe derrubar a fileira de carvalhos para um
        estacionamento, ``compensando'' com mais cinquenta caixas de
        ninho. Audite a oferta: qual dos itens deixa, orçamento,
        freios e \omterm{def:g2:where-animals-live:habitat}{hábitat} as caixas substituem, e qual só os carvalhos
        fornecem?
  \item Recomende duas medidas de terreno para o ano 11, cada uma
        ligada à linha de auditoria dela.
  \item Encerre com a moral do capítulo numa frase: além de dois
        pais, de que todo filhote precisa?
\end{enumerate}
\end{problem}
