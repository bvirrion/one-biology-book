\chapter{العضلات والمفاصل}\label{ch:g10:muscles-and-joints}

يثبّت لاعب كرة قدم قدمه، ويستدير، ثم يسقط ممسكًا ركبته. والتشخيص، بعد
ساعة: \omterm{def:g10:muscles-and-joints:joint}{رباط} ممزق. ما من عظم كسر، وما من عضلة أخفقت؛ وإنما شريط من نسيج
طوله بضعة سنتيمترات، مهمته أن يبقي عظمين على استقامة واحدة، قد استسلم
تحت التواء لم يبن له قط. ولنفهم ما الذي تمزق، ولماذا ستحتاج الركبة
إلى أشهر لتتعافى، علينا أن نعرف مم يتألف \omterm{def:g10:muscles-and-joints:joint}{المفصل}، وكيف تشد العضلات
عليه، وأي قوى تجري فيه حين يهبط جسم كتلته سبعون كيلوغرامًا على رجل
واحدة.

\section{المفصل}

\begin{definition}[المفصل]\label{def:g10:muscles-and-joints:joint}
\emph{المفصل}\index{مفصل} هو الموضع الذي يلتقي فيه عظمان. وفي المفصل
المتحرك كالركبة والورك والكتف والمرفق، تغطى نهايتا العظمين بطبقة من
\emph{الغضروف}\index{غضروف} الأملس، سمكها بضعة مليمترات، وتحبسان في
كيس محكم هو \emph{المحفظة}، تفرز بطانته \emph{السائل الزليلي} الذي
يزلق السطحين. وتمتد أشرطة من نسيج متين، هي
\emph{الأربطة}\index{رباط}، من عظم إلى عظم عبر المفصل فتحد حركاته
بالحركات المقصودة. وفي الركبة يقع هلالان من الغضروف، هما
\emph{الهلالتان}، بين العظمين فيوزعان الحمل.
\end{definition}

\begin{proposition}[وظيفة كل جزء]\label{prop:g10:muscles-and-joints:parts}
\omterm{def:g10:muscles-and-joints:joint}{الغضروف} والسائل الزليلي يجعلان السطحين ينزلقان باحتكاك أقل من احتكاك
الجليد على الجليد؛ والهلالتان توزعان حملًا كان ليتركز لولاهما على بضعة
مليمترات مربعة؛ \omterm{def:g10:muscles-and-joints:joint}{والأربطة} تتيح للركبة أن تنثني وتنبسط لكنها تقاوم
الالتواء والانثناء الجانبي؛ والمحفظة تحبس السائل. ولكل جزء إخفاقه
الخاص: \omterm{def:g10:muscles-and-joints:joint}{فالغضروف} يبلى (الفصال)، والهلالة تتمزق تحت التواء والركبة
منثنية، \omterm{def:g10:muscles-and-joints:joint}{والرباط} يلتوي أو ينقطع حين يدفع \omterm{def:g10:muscles-and-joints:joint}{المفصل} وراء مداه.
\end{proposition}
\admitted

\begin{example}[التواء وتمزق وخلع]\label{ex:g10:muscles-and-joints:injuries}
التوِ كاحلك على حافة رصيف: فتتمدد \omterm{def:g10:muscles-and-joints:joint}{الأربطة} الخارجية وراء حدها --- وهو
\emph{الالتواء}؛ وينتفخ \omterm{def:g10:muscles-and-joints:joint}{المفصل} إذ تنزف المحفظة ويتراكم السائل. واشدد
شدًا قويًا على أوتار المأبض وهي باردة: فتنقطع بعض أليافها --- وهو
\emph{تمزق} عضلي. واسقط على ذراع ممدودة: فقد يغادر رأس العضد جوفه ---
وهو \emph{خلع}، تتمدد فيه \omterm{def:g10:muscles-and-joints:joint}{الأربطة} والمحفظة أو تتمزق. وما من واحدة
منها كسر عظم؛ ومع ذلك قد يطول شفاؤها أكثر من كسر.
\end{example}

\section{العضلات تشد، أزواجًا}

\begin{definition}[العضلة الهيكلية والوتر]\label{def:g10:muscles-and-joints:muscle}
\emph{العضلة الهيكلية}\index{عضلة هيكلية} عضو مؤلف من آلاف \omterm{def:g10:cells-common-unit:cell}{الخلايا}
الطويلة، هي \emph{الألياف العضلية}، محزومة معًا؛ ويقصر كل ليف حين
ينبهه عصب. وعند الطرفين تضيق العضلة إلى \emph{وتر}\index{وتر}، وهو
حبل من نسيج غير قابض متصل بالعظم. والعضلة لا تستطيع إلا أن
\emph{تشد}، بالتقصر، ولا إلا على طولها الخاص: فهي تحرك \omterm{def:g10:muscles-and-joints:joint}{المفصل} بالشد
على العظم الذي وراءه.
\end{definition}

\begin{proposition}[الأزواج المتضادة]\label{prop:g10:muscles-and-joints:antagonists}
لما كانت العضلة لا تستطيع الدفع، وجب لكل حركة \omterm{def:g10:muscles-and-joints:joint}{مفصل} عضلتان أو
مجموعتان تعملان متضادتين: \emph{قابضة} تثني \omterm{def:g10:muscles-and-joints:joint}{المفصل} و\emph{باسطة}
تبسطه. فحين تنقبض إحداهما، ترتخي الأخرى وتتمدد؛ ولحفظ وضع ما تنقبضان
معًا. والعضلة ذات الرأسين تثني المرفق وذات الرؤوس الثلاثة تبسطه؛
ورباعية الرؤوس في مقدمة الفخذ تبسط الركبة، وأوتار المأبض خلفها تثنيها.
\end{proposition}
\admitted

\begin{proposition}[الروافع والقوى]\label{prop:g10:muscles-and-joints:lever}
العظم الذي يدور عند \omterm{def:g10:muscles-and-joints:joint}{مفصل} رافعة. \omterm{def:g10:muscles-and-joints:muscle}{ووتر} العضلة يتصل على مسافة قصيرة $d$
من المحور، بينما يؤثر الحمل على مسافة أكبر $D$؛ وللتوازن يجب أن تشد
العضلة بقوة
\[
F_{\text{عضلة}} = F_{\text{حمل}} \times \frac{D}{d},
\]
أي عدة أمثال الحمل. والترتيب يقايض القوة بالسرعة والمدى: فتقصر صغير في
العضلة ينتج حركة كبيرة سريعة لليد أو القدم، بثمن قوى كبيرة في \omterm{def:g10:muscles-and-joints:muscle}{الوتر}
وفي \omterm{def:g10:muscles-and-joints:joint}{المفصل}.
\end{proposition}
\admitted

\begin{example}[حمل كيس]\label{ex:g10:muscles-and-joints:bag}
كيس كتلته \qty{5}{kg} (ووزنه \qty{49}{N}) يتدلى من اليد، على بعد
\qty{35}{cm} من المرفق؛ \omterm{def:g10:muscles-and-joints:muscle}{ووتر} ذات الرأسين يتصل على بعد \qty{5}{cm}
منه. فتشد ذات الرأسين بقوة $49 \times 35/5 = \qty{343}{N}$ --- أي سبعة
أمثال الحمل، وهو وزن \qty{35}{kg} --- ويضغط \omterm{def:g10:muscles-and-joints:joint}{مفصل} المرفق بعضه على بعض
بنحو \qty{300}{N}. والعضلات والأوتار والغضاريف مبنية لهذه القوى؛ وإنما
يتلفها قوة مفاجئة في الاتجاه الخطأ، أو حمل يتكرر وراء ما يستطيع النسيج
إصلاحه.
\end{example}

\section{داخل العضلة}

\begin{proposition}[بنية الليف العضلي]\label{prop:g10:muscles-and-joints:fibre}
الليف العضلي \omterm{def:g10:cells-common-unit:cell}{خلية} عملاقة واحدة، عرضها \num{10}--\qty{100}{\micro m}
وطولها يبلغ عدة سنتيمترات، وفيها أنوية كثيرة. وهي مكتظة
\emph{بلييفات عضلية} متوازية، كل منها سلسلة من وحدات متكررة تتشابك
فيها خيوط بروتينية من صنفين، غليظة ورقيقة. والتقلص هو انزلاق الخيوط
الرقيقة بين الغليظة، يقوده ATP: فتقصر كل وحدة بكسر من طولها، ويقصر
اللييف بمجموعها، ويقصر الليف معه. \omterm{def:g10:cells-common-unit:organelle}{والمتقدرات} بين اللييفات تقدم ATP؛
ونهاية عصبية على كل ليف تعطي الأمر.
\end{proposition}
\admitted

\begin{definition}[أصناف الألياف]\label{def:g10:muscles-and-joints:types}
\omterm{def:g10:muscles-and-joints:muscle}{الألياف العضلية} على صنفين رئيسيين. \emph{الألياف البطيئة} تنقبض ببطء
وضعف لكن بلا كلل: فهي غنية \omterm{def:g10:cells-common-unit:organelle}{بالمتقدرات} وبالشعيرات الدموية، حمراء
\omterm{def:g10:chemistry-of-life:families}{ببروتين} يخزن الأكسجين، تجري على التنفس وتسود في عضلات القوام وعند
رياضيي التحمل. أما \emph{الألياف السريعة} فتنقبض بسرعة وقوة لكنها
تتعب في غضون دقيقة: فهي أفقر \omterm{def:g10:cells-common-unit:organelle}{بالمتقدرات}، وأشحب لونًا، وتعتمد على
مخزون الفوسفات \omterm{def:g10:cell-metabolism:fermentation}{والتخمر} وتسود عند العدّائين السريعين. ونسبة كل صنف في
عضلة ما موروثة في معظمها؛ والتدريب يغير حجم الألياف وعدتها أكثر مما
يغير عددها.
\end{definition}

\begin{example}[قراءة خزعة عضلية]\label{ex:g10:muscles-and-joints:biopsy}
تلوين لإنزيمات \omterm{def:g10:cells-common-unit:organelle}{المتقدرات} على شريحة رقيقة من عضلة الفخذ يبين فسيفساء:
ألياف داكنة (بطيئة) وألياف شاحبة (سريعة). فعينة عدّاء ماراثون 80\%
داكنة؛ وعينة عدّاء سريع 30\%؛ وعينة شخص غير مدرَّب نحو النصف والنصف.
والتلوين نفسه على العدّاء نفسه بعد سنة من تدريب السرعة يبين الألياف
الشاحبة وقد تضخمت لكنها لم تزدد عددًا.
\end{example}

\section{التعب والأذى والإصلاح}

\begin{proposition}[حدود العضلة]\label{prop:g10:muscles-and-joints:limits}
تخفق العضلة على ثلاثة أنحاء متمايزة. \emph{التعب}: فتهبط القوة حين لا
يجاري إمداد ATP الطلب --- في غضون دقيقة في الألياف السريعة، وعبر ساعات
في البطيئة --- ويتعافى بالراحة. و\emph{التشنج}: انقباض لاإرادي مفاجئ
مستمر، سببه غالبًا التجفاف أو فقد الملح. و\emph{الإصابة}: ألياف تمزقها
قوة تفوق متانتها، وذلك عادةً حين تكون العضلة باردة أو متعبة أو ممدودة
وهي تنقبض. والألياف تصلح عبر أسابيع من \omterm{def:g10:cells-common-unit:cell}{خلايا} احتياطية في العضلة؛ أما
\omterm{def:g10:muscles-and-joints:muscle}{الوتر} أو \omterm{def:g10:muscles-and-joints:joint}{الرباط}، وإمدادهما الدموي ضعيف، فيصلحان عبر أشهر.
\end{proposition}
\admitted

\begin{method}[تحليل إصابة عضلية هيكلية]\label{met:g10:muscles-and-joints:analyse}
\begin{enumerate}
  \item عيّن الحركة التي سببتها واتجاهها: هل حرّك \omterm{def:g10:muscles-and-joints:joint}{المفصل} داخل مداه
        (فهي مشكلة عضلية) أم وراءه (فهي مشكلة \omterm{def:g10:muscles-and-joints:joint}{رباط} أو محفظة)؟
  \item عيّن النسيج: \omterm{def:g10:muscles-and-joints:muscle}{فالألياف العضلية} تتمزق (ألم في بطن العضلة،
        وضعف)؛ \omterm{def:g10:muscles-and-joints:joint}{والأربطة} تلتوي (ألم وانتفاخ عند \omterm{def:g10:muscles-and-joints:joint}{المفصل}، وعدم ثبات)؛
        والأوتار تلتهب تحت الحمل المتكرر؛ \omterm{def:g10:muscles-and-joints:joint}{والغضروف} يبلى صامتًا عبر
        السنين.
  \item قدّر القوة: استعمل قاعدة الرافعة؛ وتذكر أن الهبوط أو الكبح
        يضاعف وزن الجسم عدة مرات.
  \item تنبأ بالتعافي من الإمداد الدموي للنسيج: العضلة في أسابيع،
        \omterm{def:g10:muscles-and-joints:joint}{والرباط} \omterm{def:g10:muscles-and-joints:muscle}{والوتر} في أشهر، \omterm{def:g10:muscles-and-joints:joint}{والغضروف} لا يكاد يتعافى.
\end{enumerate}
\end{method}

\begin{remark}[لماذا ينفع الإحماء]\label{rem:g10:muscles-and-joints:warmup}
عشر دقائق من الحركة الهينة ترفع حرارة العضلات درجة أو درجتين، فتجعل
بروتيناتها أقبل للتمدد وإنزيماتها أسرع؛ وتفتح الشعيرات الدموية التي
ستوصل أكسجين \cref{ch:g10:heart-lungs-effort}، وتغلّظ غشاء السائل
الزليلي على \omterm{def:g10:muscles-and-joints:joint}{الغضروف}. والعضلة الباردة التي يطلب منها انقباض أقصى هي
أشيع سبب للتمزق؛ \omterm{def:g10:muscles-and-joints:joint}{والمفصل} الذي يحمَّل قبل أن ينتشر سائله يبلى أسرع.
\end{remark}

\begin{exercise}[$\star$]\label{exo:g10:muscles-and-joints:3}

\begin{exercise}[$\star$]\label{exo:g10:muscles-and-joints:4}
ميّز بين الالتواء والتمزق العضلي والخلع بحسب النسيج المصاب.

\begin{exercise}[$\star$]\label{exo:g10:muscles-and-joints:5}
أعطِ فرقين بين \omterm{def:g10:muscles-and-joints:muscle}{الألياف العضلية} البطيئة والسريعة.

\begin{exercise}[$\star\star$]\label{exo:g10:muscles-and-joints:6}

وذات الرأسين تتصل على بعد \qty{4}{cm} منه. احسب القوة في ذات الرأسين

\begin{exercise}[$\star\star$]\label{exo:g10:muscles-and-joints:7}
فسّر لماذا يجب أن تنقبض ذات الرأسين وذات الرؤوس الثلاثة معًا لإبقاء
الذراع ساكنة في مواجهة دفع غير متوقع.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:muscles-and-joints:8}
ليف عضلي طوله \qty{3}{cm} يقصر 20\% حين ينقبض. فبكم تتحرك اليد إذا
اتصل \omterm{def:g10:muscles-and-joints:muscle}{وتر} الليف على بعد \qty{5}{cm} من المرفق وكانت اليد على بعد

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:muscles-and-joints:9}
لماذا كان الكتف، وهو أكثر مفاصل الجسم حركة، أكثرها خلعًا كذلك؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:muscles-and-joints:10}
تتشنج عضلة ساق عدّاء في نهاية سباق حار. اقترح سببين من الفصل وعلاجًا
واحدًا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:muscles-and-joints:11}
باستعمال \cref{ex:g10:muscles-and-joints:biopsy}، فسّر لماذا يتحسن
عدّاء سريع يتدرب للماراثون أقل مما قد يتحسن شخص غير مدرَّب في عمره.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:muscles-and-joints:12}
عند الهبوط من قفزة تنثني الركبة ويحمل \omterm{def:g10:muscles-and-joints:muscle}{وتر} رباعية الرؤوس نحو أربعة
أمثال وزن الجسم. فلرياضي كتلته \qty{70}{kg}، احسب تلك القوة. ومقطع
\omterm{def:g10:muscles-and-joints:muscle}{الوتر} نحو \qty{1}{cm^2} وهو ينقطع عند نحو \qty{100}{N/mm^2}؛ فكم يبعد
عن الانقطاع؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:muscles-and-joints:13}
\omterm{def:g10:muscles-and-joints:joint}{الرباط} الممزق يشفى عبر أشهر، والعضلة الممزقة عبر أسابيع، \omterm{def:g10:muscles-and-joints:joint}{والغضروف}
المتضرر لا يكاد يشفى. اربط الثلاثة بالإمداد الدموي للنسج وبالخلايا
المتاحة للإصلاح.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:muscles-and-joints:14}
كمال الأجسام يضخم العضلات؛ والتمطيط يطيل مدى \omterm{def:g10:muscles-and-joints:joint}{المفصل}. فأي نسيج يغيّر
كل منهما، ولماذا لا يغير أي منهما عدد الألياف؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:muscles-and-joints:15}
\omterm{def:g10:muscles-and-joints:joint}{غضروف} ركبة مسن قد بلي إلى نصف سمكه. فسّر، انطلاقًا من الفصل، لماذا
يؤلم \omterm{def:g10:muscles-and-joints:joint}{المفصل} على السلالم، ولماذا صارت الهلالتان أهم من ذي قبل، ولماذا
كانت العضلات حول الركبة جزءًا من العلاج.
\end{exercise}

\section{مسألة: القوى في ركبة}

\begin{problem}[{مسألة نهاية الأسبوع --- ركبة لاعب كرة قدم تفكك:
رافعة الرجل، وشد عضلة الفخذ، والرباط الذي تمزق، والأرقام التي يحملها
وتر في كل خطوة}]
\label{pb:g10:muscles-and-joints:1}
كريم، \qty{70}{kg}، أصيبت ركبته وهو يستدير بحدة على قدم مثبَّتة. خذ
$g = \qty{9.8}{N/kg}$. ويتصل \omterm{def:g10:muscles-and-joints:muscle}{وتر} رباعية الرؤوس بالظنبوب على بعد
\qty{5}{cm} من محور الركبة؛ ويؤثر دفع الأرض على القدم على بعد نحو
\qty{20}{cm} منه حين تكون الركبة منثنية كثيرًا، كما في الهبوط.

\medskip
\noindent\textbf{الجزء الأول --- الوقوف والخطو.}
\begin{enumerate}
  \item احسب وزن كريم.
  \item عند الوقوف على رجل واحدة والركبة منثنية قليلًا، تدفع الأرض
        القدم إلى الأعلى بوزنه كله، مؤثرة على بعد \qty{10}{cm} من محور
        الركبة. احسب القوة في \omterm{def:g10:muscles-and-joints:muscle}{وتر} رباعية الرؤوس.
  \item عند صعود درجة، تنثني الركبة أكثر وتؤثر قوة الأرض على بعد
        \qty{20}{cm} من المحور. احسب قوة \omterm{def:g10:muscles-and-joints:muscle}{الوتر} من جديد.
  \item عبّر عن النتيجتين بمضاعفات وزن الجسم، وفسّر لماذا تتعب السلالم
        الفخذين.
  \item أي مجموعة عضلية هي مضادة رباعية الرؤوس، وماذا تفعل أثناء
        الخطوة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الهبوط.}
\begin{enumerate}[resume]
  \item عند الهبوط من ضربة رأس، تدفع الأرض بثلاثة أمثال وزن كريم، على
        بعد \qty{20}{cm} من المحور. احسب قوة \omterm{def:g10:muscles-and-joints:muscle}{الوتر}.
  \item مقطع \omterm{def:g10:muscles-and-joints:muscle}{الوتر} \qty{1.2}{cm^2}. احسب الإجهاد فيه، بالنيوتن لكل
        مليمتر مربع، وقارنه بإجهاد الانقطاع البالغ نحو
        \qty{100}{N/mm^2}.
  \item يضغط سطحا \omterm{def:g10:muscles-and-joints:joint}{المفصل} أحدهما على الآخر بنحو قوة \omterm{def:g10:muscles-and-joints:muscle}{الوتر} زائد قوة
        الأرض. احسب ذلك الحمل، ثم الضغط على \omterm{def:g10:muscles-and-joints:joint}{الغضروف} إذا كانت مساحة
        التماس \qty{6}{cm^2} مع الهلالتين و\qty{1}{cm^2} من دونهما.
  \item استنتج لماذا تبلي ركبة فقدت هلالة غضروفها أسرع.
  \item فسّر ما الذي يسهم به السائل الزليلي أثناء الهبوط.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الإصابة.}
\begin{enumerate}[resume]
  \item كانت قدم كريم مثبَّتة وجسمه يستدير: فأي اتجاه حركة فرض على
        الركبة، وأي نسيج مبني لمقاومته؟
  \item لماذا لم تحمِ رباعية الرؤوس \omterm{def:g10:muscles-and-joints:joint}{الرباط}، وهي قوية بما يكفي لتشد
        بعدة كيلونيوتن؟
  \item انتفخت الركبة في غضون ساعة. فما الذي ملأها، وأي جزء من \omterm{def:g10:muscles-and-joints:joint}{المفصل}
        قد خرق؟
  \item يبين التصوير أن \omterm{def:g10:muscles-and-joints:joint}{الرباط} انقطع وأن هلالة تمزقت. باستعمال
        \cref{met:g10:muscles-and-joints:analyse}، تنبأ بزمن تعافي كل
        منهما وعلّل من إمدادهما الدموي.
  \item يسأل كريم لماذا يريد الجراح تقوية أوتار مأبضه قبل العملية.
        أجب بما جاء في \cref{prop:g10:muscles-and-joints:antagonists}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- كل خطوة تحسب.}
\begin{enumerate}[resume]
  \item يخطو كريم نحو 8000 خطوة في اليوم؛ نصفها على الرجل المصابة.
        باستعمال قوة السؤال 2، كم مرة في اليوم يحمل \omterm{def:g10:muscles-and-joints:muscle}{الوتر} ذلك الحمل
        على الأقل؟
  \item وعلى مدى سنة، ما العدد الكلي للتحميلات؟ فسّر لماذا يحتاج \omterm{def:g10:muscles-and-joints:muscle}{وتر}
        التهب من فرط الاستعمال إلى الراحة لا إلى القوة.
  \item يهبط في مباراة من قفزات نحو 40 مرة. قارن الحمل الكلي المحمول
        في الهبوطات (السؤال 6) بالحمل المحمول في خطوات اليوم.
  \item الجري يضاعف قوة الأرض قياسًا إلى المشي. فلماذا ينصح الأطباء
        بركوب الدراجة والسباحة لا بالجري لركبة تتعافى؟
  \item اذكر النتيجة: مضاعف وزن الجسم الذي يحمله \omterm{def:g10:muscles-and-joints:muscle}{وتر} رباعية الرؤوس على
        درجة وعند هبوط، ونسيج الركبة الوحيد الذي لم تختبر تلك الأرقام
        متانته قط --- حتى جاءت الاستدارة.
\end{enumerate}
\end{problem}
