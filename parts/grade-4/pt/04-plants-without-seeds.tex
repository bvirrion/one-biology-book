\chapter{Plantas sem sementes}\label{ch:g4:plants-without-seeds}

O lado sombreado do muro velho está estofado de musgo; debaixo das
faias, as \omterm{prop:g4:plants-without-seeds:damp}{samambaias} desenrolam as grandes \omterm{def:g1:plants-around-us:parts}{folhas} verdes delas.
Procure nelas o verão inteiro: você não vai achar \omterm{def:g3:flowers-fruits-seeds:parts}{flor}, nem \omterm{prop:g3:flowers-fruits-seeds:transform}{fruto},
nem \omterm{def:g2:from-seed-to-plant:seed}{semente} --- nunca. E mesmo assim os \omterm{prop:g4:plants-without-seeds:damp}{musgos} e as \omterm{prop:g4:plants-without-seeds:damp}{samambaias} se
espalham muito bem. As \omterm{def:g1:plants-around-us:plant}{plantas}, ao que parece, têm mais de um jeito
de fazer a geração seguinte.

\section{Samambaias e musgos: reprodução por esporos}

\begin{definition}[Esporo]\label{def:g4:plants-without-seeds:spore}
Um \emph{esporo}\index{esporo} é um grão microscópico, feito em
quantidades enormes, do qual uma \omterm{def:g1:plants-around-us:plant}{planta} nova pode crescer. Ao
contrário de uma \omterm{def:g2:from-seed-to-plant:seed}{semente}
(\cref{def:g2:from-seed-to-plant:seed}), um esporo é uma partícula
sozinha, \emph{sem \omterm{def:g3:animals-through-seasons:active}{reserva de alimento} e sem plantinha pronta lá
dentro} --- é um começo no que ele tem de mais nu.
\end{definition}

\begin{example}[Embaixo das folhas da samambaia]\label{ex:g4:plants-without-seeds:fern}
Vire uma \omterm{def:g1:plants-around-us:parts}{folha} de samambaia no fim do verão: fileiras caprichadas de
pontinhos marrons alinhadas na parte de baixo. Cada pontinho é um
pacote com milhares de \omterm{def:g4:plants-without-seeds:spore}{esporos}; quando amadurecem, os pacotes se
abrem e os \omterm{def:g4:plants-without-seeds:spore}{esporos} partem pelo ar como a poeira mais fina. Um \omterm{def:g4:plants-without-seeds:spore}{esporo}
que cai em chão sombreado e úmido pode virar --- por etapas --- uma
samambaia nova.
\end{example}

\begin{proposition}[As plantas de esporo precisam de umidade]\label{prop:g4:plants-without-seeds:damp}
As \omterm{def:g1:plants-around-us:plant}{plantas} de \omterm{def:g4:plants-without-seeds:spore}{esporo} --- \emph{musgos}\index{musgo} e
\emph{samambaias}\index{samambaia} --- só conseguem se reproduzir
onde há \emph{umidade}: os \omterm{def:g4:plants-without-seeds:spore}{esporos} delas, e as etapas delicadas que
crescem a partir deles, ressecam e morrem no ar seco. É por isso que
o musgo forra o lado sombreado do muro e nunca o lado ensolarado,
e por isso que as \omterm{prop:g4:plants-without-seeds:damp}{samambaias} se aglomeram em matas
úmidas, margens de riacho e poços.
\end{proposition}
\admitted

\begin{remark}[Um esporo não é uma semente]\label{rem:g4:plants-without-seeds:sporeseed}
Os dois são começos que viajam, mas a \omterm{def:g2:from-seed-to-plant:seed}{semente} é um kit de
sobrevivência --- casca, \omterm{def:g3:animals-through-seasons:active}{reserva de alimento}, plantinha pronta ---
enquanto o \omterm{def:g4:plants-without-seeds:spore}{esporo} é nu. Daí os números: os milhões de \omterm{def:g4:plants-without-seeds:spore}{esporos} da
samambaia contra a dúzia de \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} da vagem de feijão. Começos nus
são baratos de fazer e fáceis de perder; os dois planos ecoam o
bacalhau e a elefanta da
\cref{prop:g4:animal-reproduction:strategies}.
\end{remark}

\section{Plantas novas sem plantar nada}

\begin{definition}[Reprodução vegetativa]\label{def:g4:plants-without-seeds:vegetative}
Muitas \omterm{def:g1:plants-around-us:plant}{plantas} também conseguem fazer \omterm{def:g1:plants-around-us:plant}{plantas} novas
\emph{diretamente a partir de um pedaço delas mesmas} --- sem
\omterm{def:g4:plants-without-seeds:spore}{esporos}, sem \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}, com um único pai. Isso é a \emph{reprodução
vegetativa}\index{reprodução vegetativa}. A \omterm{def:g1:plants-around-us:plant}{planta} nova é uma
continuação viva da antiga, e se parece com ela exatamente.
\end{definition}

\begin{example}[Os estolhos do morangueiro]\label{ex:g4:plants-without-seeds:runner}
Um morangueiro lança \omterm{def:g1:plants-around-us:parts}{caules} horizontais compridos e finos ---
\emph{estolhos}\index{estolho} --- por cima da terra. Onde um
estolho encosta no chão, ele cria \omterm{def:g1:plants-around-us:parts}{raízes}, brota \omterm{def:g1:plants-around-us:parts}{folhas} e vira um
morangueiro novo e completo; o estolho depois seca, e a \omterm{def:g1:plants-around-us:plant}{planta} filha
fica livre. Uma \omterm{def:g1:plants-around-us:plant}{planta} vira um canteiro sem uma única \omterm{def:g2:from-seed-to-plant:seed}{semente}
plantada.
\end{example}

\begin{example}[Bulbos e tubérculos]\label{ex:g4:plants-without-seeds:bulbtuber}
O \emph{bulbo}\index{bulbo} da tulipa e o
\emph{tubérculo}\index{tubérculo} da batata são despensas
subterrâneas --- e kits de partida. Deixado na terra, um \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbo} se
divide em \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulbos} filhos, cada um uma tulipa futura; cada ``\omterm{def:g1:the-five-senses:sense}{olho}'' de
uma batata brotando pode virar uma \omterm{def:g1:plants-around-us:plant}{planta} nova inteira. Quem cuida
de horta \omterm{def:g1:plants-around-us:plant}{planta} batatas, e não \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} de batata.
\end{example}

\begin{method}[Fazer uma estaca]\label{met:g4:plants-without-seeds:cutting}
Quem cuida de horta copia a \omterm{def:g4:plants-without-seeds:vegetative}{reprodução vegetativa} com
\emph{estacas}\index{estaca}:
\begin{enumerate}
  \item corte um broto jovem e saudável, do tamanho de uma mão, da
        planta-mãe (hortelã, salgueiro e gerânio funcionam bem);
  \item tire as \omterm{def:g1:plants-around-us:parts}{folhas} de baixo e ponha o broto em pé na água ou em
        terra úmida;
  \item espere: em dias ou semanas, nascem \omterm{def:g1:plants-around-us:parts}{raízes} do \omterm{def:g1:plants-around-us:parts}{caule} cortado;
  \item plante a estaca --- é uma \omterm{def:g1:plants-around-us:plant}{planta} nova e completa, idêntica
        ao seu único pai.
\end{enumerate}
\end{method}

\begin{remark}[Um pai, cópias exatas]\label{rem:g4:plants-without-seeds:copy}
As \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes} misturam dois pais (a
\cref{rem:g4:animal-reproduction:mix} também vale para as \omterm{def:g1:plants-around-us:plant}{plantas},
por meio do \omterm{def:g3:flowers-fruits-seeds:parts}{pólen}); a \omterm{def:g4:plants-without-seeds:vegetative}{reprodução vegetativa} copia um só. É por isso
que toda \omterm{def:g1:plants-around-us:plant}{planta} de uma variedade de maçã com nome é obtida de
\omterm{met:g4:plants-without-seeds:cutting}{estacas} e enxertos, e nunca de caroços: o pomar quer cópias exatas
de uma única árvore excelente. O preço de copiar é a mesmice --- um
pomar de árvores idênticas divide fraquezas idênticas, como um ano
posterior vai mostrar.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:1}
Diga duas \omterm{def:g1:plants-around-us:plant}{plantas} que nunca fazem \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}. Como elas se reproduzem?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:2}
Dê duas diferenças entre um \omterm{def:g4:plants-without-seeds:spore}{esporo} e uma \omterm{def:g2:from-seed-to-plant:seed}{semente}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:3}
Onde são feitos os \omterm{def:g4:plants-without-seeds:spore}{esporos} da samambaia? Descreva o que você
encontra embaixo de uma \omterm{def:g1:plants-around-us:parts}{folha} no fim do verão.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:4}
Por que o musgo cresce no lado sombreado do muro e não no lado
ensolarado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:5}
O que é a \omterm{def:g4:plants-without-seeds:vegetative}{reprodução vegetativa}? Quantos pais ela exige?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:6}
Conte a história de um estolho de morangueiro, do primeiro \omterm{def:g1:plants-around-us:parts}{caule} até
a \omterm{def:g1:plants-around-us:plant}{planta} filha independente.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:7}
Por que quem cuida de horta \omterm{def:g1:plants-around-us:plant}{planta} batatas em vez de semear \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}
de batata? O que cada ``\omterm{def:g1:the-five-senses:sense}{olho}'' consegue fazer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:plants-without-seeds:8}
Faça a experiência: faça uma estaca de hortelã seguindo o
\cref{met:g4:plants-without-seeds:cutting} e ponha ela em pé num copo
de água no peitoril da janela. O que aparece, e depois de quanto
tempo mais ou menos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:9}
A samambaia faz milhões de \omterm{def:g4:plants-without-seeds:spore}{esporos}; a vagem de feijão, uma dúzia de
\omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}. Explique a diferença de números com a
\cref{rem:g4:plants-without-seeds:sporeseed} --- e diga o par de
\omterm{def:g1:animals-around-us:animal}{animais} cujas estratégias elas ecoam.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:plants-without-seeds:10}
Quem cuida de horta encontra os morangueiros novos idênticos ao
pai, enquanto os pés de feijão dela variam um pouco em relação aos
pais deles. Explique as duas observações com a
\cref{rem:g4:plants-without-seeds:copy}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:plants-without-seeds:11}
Um \omterm{def:g1:plants-around-us:tree}{galho} de salgueiro arrancado por uma enchente de inverno, levado
rio abaixo e encalhado num banco de lama, muitas vezes vira um
salgueiro novo. Que tipo de reprodução é esse --- e por que os rios
tantas vezes são margeados por salgueiros? O que a enchente fez que
quem cuida de horta faz no
\cref{met:g4:plants-without-seeds:cutting} com a tesoura?
\end{exercise}
