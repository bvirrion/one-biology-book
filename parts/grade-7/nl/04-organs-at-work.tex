\chapter{Organen aan het werk}\label{ch:g7:organs-at-work}

Sprint drie trappen op en maak de balans op: brandende \omterm{def:g1:my-body:parts}{benen}, hamerend
\omterm{def:g4:heart-and-blood:heart}{hart}, jagende adem --- en, gek genoeg, een blozende, warme \omterm{def:g1:the-five-senses:sense}{huid}. Niemand
gaf opdracht tot die veranderingen; ze gaven zichzelf opdracht, en in
volmaakte verhouding. Dit hoofdstuk meet wat een werkend orgaan werkelijk
verbruikt, en hoe het lichaam zijn leveringen herschikt zodra het werk
begint.

\section{Wat een werkende spier verbruikt}

\begin{proposition}[De rekening van de spier]\label{prop:g7:organs-at-work:bill}
Een werkende \omterm{def:g3:bones-and-muscles:muscle}{spier} geeft uit, en die uitgaven zijn te meten. Vergeleken
met dezelfde \omterm{def:g3:bones-and-muscles:muscle}{spier} in rust neemt een hardwerkende \omterm{def:g3:bones-and-muscles:muscle}{spier}:
\begin{itemize}
  \item veel meer \emph{\omterm{def:g7:what-is-respiration:respiration}{zuurstof}} op uit het \omterm{prop:g4:journey-of-food:absorb}{bloed} dat haar passeert ---
        een veelvoud van haar rustverbruik;
  \item veel meer \emph{glucose}\index{glucose} --- de kant-en-klare
        suikervoedingsstof van het \omterm{prop:g4:journey-of-food:absorb}{bloed}, haar dagelijkse brandstof;
  \item en geeft navenant meer \emph{\omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide}} af --- en
        \emph{warmte}.
\end{itemize}
Dit is het verbranden in de \omterm{def:g5:first-look-at-cells:cell}{cel} uit
\cref{prop:g7:what-is-respiration:power}, afgelezen aan de meters van een
orgaan: brandstof en \omterm{def:g7:what-is-respiration:respiration}{zuurstof} erin, werk, afvalgas en warmte eruit.
\end{proposition}
\admitted

\begin{example}[Meten over een spier heen]\label{ex:g7:organs-at-work:measure}
Fysiologen vergelijken het \omterm{prop:g4:journey-of-food:absorb}{bloed} dat een \omterm{def:g3:bones-and-muscles:muscle}{spier} \emph{binnengaat} met het
\omterm{prop:g4:journey-of-food:absorb}{bloed} dat haar \emph{verlaat}. In rust heeft het vertrekkende \omterm{prop:g4:journey-of-food:absorb}{bloed} een
bescheiden deel van zijn \omterm{def:g7:what-is-respiration:respiration}{zuurstof} en \omterm{prop:g7:organs-at-work:bill}{glucose} afgestaan. Bij zwaar werk
laat diezelfde vergelijking het vertrekkende \omterm{prop:g4:journey-of-food:absorb}{bloed} uitgekleed zien --- het
meeste van zijn afleverbare \omterm{def:g7:what-is-respiration:respiration}{zuurstof} weggenomen --- terwijl de
\omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} omhoog is gesprongen. Hetzelfde orgaan, dezelfde meters:
alleen het werk veranderde, en de rekening volgde het.
\end{example}

\begin{remark}[De warmte is het bewijs van betaling]\label{rem:g7:organs-at-work:warmth}
De warmte van een werkend lichaam is geen toeval: brandstof verbranden
levert altijd warmte op, en \omterm{def:g3:bones-and-muscles:muscle}{spier} is de stookkelder van het lichaam.
Rillen is diezelfde truc met opzet gedraaid --- \omterm{def:g3:bones-and-muscles:muscle}{spieren} die aan het werk
gezet worden voor niets anders dan warmte (het vaste verbruik uit
\cref{ex:g3:balanced-diet:rest}, even verdubbeld). De blozende \omterm{def:g1:the-five-senses:sense}{huid} van
de sprint is het koelsysteem dat de kachelwarmte afvoert --- het onderwerp
van de volgende paragraaf.
\end{remark}

\section{Leveringen op afroep}

\begin{proposition}[Het lichaam herschikt zijn bevoorrading]\label{prop:g7:organs-at-work:supply}
Op het moment dat het werk begint, herschikken de bevoorradingslijnen zich
om de rekening te betalen (\cref{ex:g4:heart-and-blood:effort} kondigde het
aan; hier is de volledige uitgifte):
\begin{enumerate}
  \item de \emph{\omterm{def:g7:what-is-respiration:respiration}{ademhaling}} wordt dieper en sneller --- ze laadt het
        \omterm{prop:g4:journey-of-food:absorb}{bloed} sneller bij in de \omterm{def:g7:how-animals-breathe:four}{longen}
        (\cref{ex:g4:breathing:running});
  \item het \emph{\omterm{def:g4:heart-and-blood:heart}{hart}} klopt sneller en harder --- meer rondes per
        minuut (\cref{prop:g4:heart-and-blood:pulse});
  \item het \omterm{prop:g4:journey-of-food:absorb}{bloed} wordt \emph{omgeleid}: de vaten die de werkende \omterm{def:g3:bones-and-muscles:muscle}{spieren}
        (en de koelende \omterm{def:g1:the-five-senses:sense}{huid}) bedienen worden wijder, terwijl die van het
        rustende \omterm{def:g4:journey-of-food:tract}{spijsverteringskanaal} nauwer worden --- hetzelfde \omterm{prop:g4:journey-of-food:absorb}{bloed},
        omgeleid naar de verdieping die uitgeeft.
\end{enumerate}
Alle drie volgen ze het werk vanzelf, gestuurd door de ongevraagde
diensten van de \omterm{def:g5:brain-in-command:system}{hersenen}
(\cref{rem:g5:brain-in-command:more}).
\end{proposition}
\admitted

\begin{example}[De uitgifte, van binnenuit gevoeld]\label{ex:g7:organs-at-work:felt}
De balans na de trapsprint, regel voor regel verklaard: jagende adem ---
bijladen; hamerend \omterm{def:g4:heart-and-blood:heart}{hart} --- snellere rondes; brandende \omterm{def:g1:my-body:parts}{benen} --- de
rekening van de \omterm{def:g3:bones-and-muscles:muscle}{spieren} die tot het uiterste betaald wordt; blozende
warme \omterm{def:g1:the-five-senses:sense}{huid} --- kachelwarmte die naar het koelende oppervlak gebracht
wordt. Zelfs de klassieke steek in de zij en de loomheid na een grote
maaltijd passen erin: \omterm{def:g4:journey-of-food:digestion}{vertering} en sprint strijden om het omgeleide
\omterm{prop:g4:journey-of-food:absorb}{bloed}, en daarom raadde
\cref{met:g4:journey-of-food:habits} af om met een volle \omterm{def:g4:journey-of-food:tract}{maag} te rennen.
\end{example}

\begin{definition}[Herstel]\label{def:g7:organs-at-work:recovery}
Als de inspanning stopt, stoppen \omterm{prop:g4:heart-and-blood:pulse}{polsslag} en \omterm{def:g7:what-is-respiration:respiration}{ademhaling} niet: ze zakken
maar geleidelijk terug --- de \emph{herstelfase}\index{herstel}. Het
lichaam vereffent schulden: het laadt zijn zuurstofvoorraden bij, ruimt
het afval van de inspanning op en raakt de opgehoopte warmte kwijt. Hoe
fitter het lichaam, hoe sneller het herstel --- en daarom is de hersteltijd
zelf een fitheidsmeting.
\end{definition}

\section{Trainen, en de grenzen van de machine}

\begin{proposition}[Trainen verhoogt het plafond]\label{prop:g7:organs-at-work:training}
Bij regelmatige oefening wordt de hele bevoorradingsketen sterker
(\cref{rem:g4:heart-and-blood:rest},
\cref{met:g4:breathing:care}): het \omterm{def:g4:heart-and-blood:heart}{hart} verplaatst meer \omterm{prop:g4:journey-of-food:absorb}{bloed} per slag, de
ademhalingsspieren gaan dieper, de \omterm{def:g3:bones-and-muscles:muscle}{spieren} zelf groeien en slaan meer
brandstof op, en hun fijne vaatjes vermeerderen zich. Gevolg: dezelfde
inspanning kost een getraind lichaam een lagere \omterm{prop:g4:heart-and-blood:pulse}{polsslag}, en zijn plafond
--- de zwaarste inspanning die het kan betalen --- gaat omhoog. De machine
wordt door haar eigen gebruik verbouwd.
\end{proposition}
\admitted

\begin{example}[Twee lopers, één heuvel]\label{ex:g7:organs-at-work:runners}
Een ongetrainde en een getrainde loper nemen dezelfde heuvel.
Ongetraind: halverwege een \omterm{prop:g4:heart-and-blood:pulse}{polsslag} van $180$, hijgend, gedwongen te
lopen --- de bevoorradingslijnen kunnen de rekening niet betalen, en de
schuld roept halt. Getraind: bovenaan $150$, praten lukt nog, \omterm{def:g7:organs-at-work:recovery}{herstel} in
drie minuten. Dezelfde heuvel, dezelfde rekening --- verschillende
plafonds, gebouwd door maanden van precies zulke heuvels rustig genomen.
\end{example}

\begin{method}[Je eigen machine lezen]\label{met:g7:organs-at-work:read}
Een zelfproef met een horloge
(\cref{met:g4:heart-and-blood:pulse} levert de polstechniek):
\begin{enumerate}
  \item noteer de \omterm{prop:g4:heart-and-blood:pulse}{polsslag} in rust, zittend en kalm;
  \item doe drie minuten gestage inspanning --- opstapjes, touwtjespringen;
  \item noteer meteen de \omterm{prop:g4:heart-and-blood:pulse}{polsslag}, en daarna elke minuut tot hij weer op
        rust staat: de hersteltijd;
  \item herhaal het wekelijks gedurende een periode met regelmatige
        beweging: zie de rustpolsslag, de inspanningspolsslag en de
        hersteltijd allemaal wegzakken --- de verbouwing van
        \cref{prop:g7:organs-at-work:training}, in je eigen gegevens.
\end{enumerate}
\end{method}

\section{Probleem: Het fitheidsonderzoek van de school}

\begin{problem}[{Weekendprobleem --- een periode polsgegevens, gelezen als
een fysioloog}]\label{pb:g7:organs-at-work:1}
De klas doet elke maandag een periode lang
\cref{met:g7:organs-at-work:read}: rustpolsslag, \omterm{prop:g4:heart-and-blood:pulse}{polsslag} na drie minuten
opstapjes, en hersteltijd. Twee geanonimiseerde dossiers: leerling A ---
week 1: rust $84$, inspanning $168$, \omterm{def:g7:organs-at-work:recovery}{herstel} $9$ minuten; week 12: rust
$72$, inspanning $150$, \omterm{def:g7:organs-at-work:recovery}{herstel} $5$ minuten. Leerling B --- week 1: rust
$70$, inspanning $150$, \omterm{def:g7:organs-at-work:recovery}{herstel} $4$ minuten; week 12: rust $69$,
inspanning $148$, \omterm{def:g7:organs-at-work:recovery}{herstel} $4$ minuten.

\medskip\noindent\textbf{Deel I --- Binnen één sessie.}
\begin{enumerate}
  \item Noem tijdens de opstapjes wat er gebeurt met de opname van
        \omterm{def:g7:what-is-respiration:respiration}{zuurstof} en \omterm{prop:g7:organs-at-work:bill}{glucose} door een dijspier, en met haar afgifte van
        \omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} en warmte.
  \item Leg uit waarom \omterm{def:g7:what-is-respiration:respiration}{ademhaling} en \omterm{prop:g4:heart-and-blood:pulse}{polsslag} tijdens de minuten
        inspanning \emph{samen} omhoog gaan.
  \item Een leerling voelt zich bij minuut drie warm en ziet er blozend
        uit. Geef het verslag van kachel en koeling.
  \item Waarom zakt de \omterm{prop:g4:heart-and-blood:pulse}{polsslag} niet meteen naar rust op het moment dat
        de opstapjes stoppen?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De periode lezen.}
\begin{enumerate}[resume]
  \item Beschrijf de veranderingen bij A gedurende de periode, en noem de
        verbouwingen uit
        \cref{prop:g7:organs-at-work:training} achter elk getal.
  \item De getallen van B bewogen nauwelijks. Geef de twee eerlijke
        lezingen --- en welk extra feit over het leven van B buiten de les
        tussen die twee zou beslissen.
  \item In week 5 meet een leerling meteen na een haastige lunch en krijgt
        vreemde getallen. Welke regel uit
        \cref{prop:g7:organs-at-work:supply} stoorde, en wat zou het
        protocol over meettijden moeten zeggen?
  \item Waarom moet elke leerling altijd op dezelfde manier meten ---
        dezelfde inspanning, dezelfde timing? (Een oude bekende onder de
        methoden; noem haar.)
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De brief van de
fysioloog.}
\begin{enumerate}[resume]
  \item Schrijf de samenvatting van A's periode in twee zinnen voor het
        prikbord van de klas, met de woorden \emph{plafond} en
        \emph{\omterm{def:g7:organs-at-work:recovery}{herstel}}.
  \item Een ouder vraagt of een lage rustpolsslag zorgwekkend is.
        Antwoord met de wielrenner uit
        \cref{exo:g4:heart-and-blood:11}.
  \item De klas wil volgende periode één getal volgen. Pleit voor de
        hersteltijd boven de inspanningspolsslag, met
        \cref{def:g7:organs-at-work:recovery}.
  \item Sluit af met de moraal van het onderzoek in één zin: wat heeft de
        machine van A verbouwd?
\end{enumerate}
\end{problem}
