\section*{Bab \ref{ch:g7:removing-wastes} --- Menyingkirkan Buangan}

\begin{solution}{exo:g7:removing-wastes:1}
Sebuah buangan pembawa nitrogen yang ditinggalkan oleh pemakaian \omterm{def:g7:digestion-and-nutrients:families}{zat
gizi} materi pembangun di dalam \omterm{def:g5:first-look-at-cells:cell}{selnya}. Tidak berbahaya ketika berada
dalam lalu lintas yang lewat, ia meracuni kalau menumpuk --- sehingga
ia harus pergi terus-menerus.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:2}
\omterm{def:g7:how-animals-breathe:four}{Paru-paru}: \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida}. \omterm{def:g7:removing-wastes:kidneys}{Ginjal}: \omterm{prop:g7:removing-wastes:rubbish}{urea} dan kelebihan air serta
garamnya. \omterm{def:g1:the-five-senses:sense}{Kulit}: panas, dan air serta garam keringatnya (dengan jejak
\omterm{prop:g7:removing-wastes:rubbish}{urea}).
\end{solution}

\begin{solution}{exo:g7:removing-wastes:3}
Dua \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}, sebesar kepalan tangan dan berbentuk kacang, terletak rendah
di punggung, masing-masing diberi makan oleh sebuah \omterm{def:g7:heart-and-circulation:vessels}{arteri} yang murah
\omterm{prop:g7:digestion-and-nutrients:absorption}{hati}. \omterm{def:g7:removing-wastes:kidneys}{Urine} menetes dari masing-masingnya menuruni pipanya ke \omterm{def:g7:removing-wastes:kidneys}{kandung
kemih} --- tangki penampung yang dikosongkan sesuka \omterm{prop:g7:digestion-and-nutrients:absorption}{hati}.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:4}
Mula-mula menyaring dengan murah \omterm{prop:g7:digestion-and-nutrients:absorption}{hati}: berpuluh-puluh liter bagian
berair \omterm{prop:g4:journey-of-food:absorb}{darahnya} didorong melewati ayakannya setiap hari, yang baik dan
yang buruk sekaligus. Lalu mengambil kembali dengan hemat di sepanjang
pipa yang berkelok. Bingkainya: berpuluh-puluh liter yang diayak,
kira-kira satu setengah liter yang pergi sebagai \omterm{def:g7:removing-wastes:kidneys}{urine}.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:5}
Sepenuhnya: \omterm{prop:g7:organs-at-work:bill}{glukosanya} (dan nyaris seluruh airnya). Secukupnya: air
dan garam, yang disetel menurut kebutuhan hari itu. Sama sekali tidak:
\omterm{prop:g7:removing-wastes:rubbish}{urea} --- yang menjadi inti seluruh latihannya.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:6}
Pengambilan kembalinya disetel: sesudah banyak minum, lebih sedikit
air yang diambil kembali --- \omterm{def:g7:removing-wastes:kidneys}{urine} yang berlimpah dan pucat; sesudah
berkeringat atau sedikit minum, nyaris seluruhnya diambil kembali ---
\omterm{def:g7:removing-wastes:kidneys}{urine} yang sedikit dan gelap. \omterm{def:g7:removing-wastes:kidneys}{Ginjalnya} menjaga rekening air tubuhnya.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:7}
\omterm{def:g7:what-is-respiration:respiration}{Karbon dioksida} adalah gas, dan gas menyeberang pada sebuah perbatasan
udara --- \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya}; \omterm{prop:g7:removing-wastes:rubbish}{urea} bukan gas dan tidak dapat pergi bersama
napasnya, tetapi ia menumpang dalam keadaan \omterm{def:g7:oxygen-in-the-water:dissolved}{terlarut} lalu tersaring
dengan indah pada sebuah perbatasan cair --- ayakannya. Tiap buangan
keluar lewat perbatasan yang cocok dengan wujudnya: pembagian kerjanya
adalah kimia yang dipasangkan dengan arsitektur.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:8}
Ayakannya melewatkan \omterm{prop:g7:organs-at-work:bill}{glukosa} dengan bebas --- itu memang wajar; tahap
pengambilan kembalinya lalu seharusnya memulihkan setiap satuannya.
\omterm{prop:g7:organs-at-work:bill}{Glukosa} yang tiba di dalam \omterm{def:g7:removing-wastes:kidneys}{urine} berarti pengambilan kembalinya
kewalahan atau rusak: entah gula \omterm{prop:g4:journey-of-food:absorb}{darahnya} begitu tinggi sampai
\omterm{def:g7:organs-at-work:recovery}{pemulihannya} tidak dapat mengimbanginya, entah \omterm{def:g7:organs-at-work:recovery}{pemulihannya} sendiri
yang gagal --- keduanya layak mendapat perhatian dokter.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:9}
Dari \omterm{def:g7:heart-and-circulation:vessels}{kapiler} \omterm{def:g3:bones-and-muscles:muscle}{otot} ke \omterm{def:g7:heart-and-circulation:vessels}{vena}, lalu ke \omterm{def:g7:heart-and-circulation:rooms}{serambi} dan \omterm{def:g7:heart-and-circulation:rooms}{bilik} kanan, lalu ke
peredaran paru (\omterm{prop:g7:removing-wastes:rubbish}{ureanya} menumpang terus; \omterm{def:g7:how-animals-breathe:four}{paru-parunya} tidak dapat
menyentuhnya), lalu ke \omterm{def:g4:heart-and-blood:heart}{jantung} kirinya, lalu ke \omterm{def:g7:heart-and-circulation:vessels}{arteri} tubuh menuju
\omterm{def:g7:removing-wastes:kidneys}{ginjalnya}, lalu diayak pada sebuah satuan penyaring, \emph{tidak}
diambil kembali, lalu menuruni pipa pengumpulnya sampai ke \omterm{def:g7:removing-wastes:kidneys}{kandung
kemih}.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:10}
Masuknya: tanpa pembongkaran, menyeberang di \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot ususnya} ke dalam
\omterm{prop:g4:journey-of-food:absorb}{darahnya}. Pengangkutannya: \omterm{prop:g7:digestion-and-nutrients:absorption}{hatinya} lebih dahulu, lalu kedua
peredarannya. Pemakaiannya: menjadi medium bagi kimia basah setiap
\omterm{def:g5:first-look-at-cells:cell}{selnya}, dan menjadi pasokan bagi sistem pendinginannya. Keluarnya:
terbagi di antara \omterm{def:g7:removing-wastes:kidneys}{urine} (yang disetel pengambilan kembalinya),
keringat (bagian olahraganya), \omterm{def:g6:exploring-our-environment:conditions}{kelembapan} napasnya, dan sedikit di
dalam sisanya.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:11}
Ayakannya --- yang menyaring buangan dari \omterm{prop:g4:journey-of-food:absorb}{darah} menyeberangi sebuah
\omterm{def:g6:cell-unit-of-life:parts}{membran} halus; mesinnya adalah satuan penyaring di luar tubuh. Sesinya
berulang karena \omterm{def:g5:first-look-at-cells:cell}{selnya} tidak pernah berhenti membuat \omterm{prop:g7:removing-wastes:rubbish}{urea}
(\cref{prop:g7:removing-wastes:rubbish}): sebuah penyaring yang
berjalan dua kali seminggu harus mengejar hari-hari penumpukannya,
sedangkan ayakan \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} berjalan setiap menit.
\end{solution}

\begin{solution}{exo:g7:removing-wastes:12}
Misalnya: satu \omterm{prop:g4:journey-of-food:absorb}{darah}, yang digerakkan mengelilingi peredaran paru dan
peredaran tubuh oleh \omterm{def:g4:heart-and-blood:heart}{jantung} berempat ruangnya, memuat \omterm{def:g7:what-is-respiration:respiration}{oksigen} di
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya} dan \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} di \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot ususnya}, lalu mengantarkan keduanya
ke \omterm{def:g7:heart-and-circulation:vessels}{kapiler} setiap \omterm{def:g5:first-look-at-cells:cell}{selnya}, tempat api lambat \omterm{def:g7:what-is-respiration:respiration}{respirasinya} membakar
bahan bakar dengan \omterm{def:g7:what-is-respiration:respiration}{oksigen} untuk menenagai segala pekerjaannya. \omterm{prop:g4:journey-of-food:absorb}{Darah}
yang sama memungut abu apinya --- \omterm{def:g7:what-is-respiration:respiration}{karbon dioksida} ke \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya}, \omterm{prop:g7:removing-wastes:rubbish}{urea}
dan kelebihannya ke penyaring ayak-dan-ambil-kembali \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} ---
sehingga keempat perbatasannya melayani satu ekonomi. Setiap organ
tahun ini adalah sebuah stasiun pada gelang ini, dan gelangnya ada
supaya api di dalam \omterm{def:g5:first-look-at-cells:cell}{selnya} tidak pernah kekurangan bahan bakar,
\omterm{def:g7:what-is-respiration:respiration}{oksigen}, atau pemungutan sampah.
\end{solution}

\begin{solution}{pb:g7:removing-wastes:1}
\textbf{1.} Menyeberang di \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot usus} halusnya, ke dalam \omterm{def:g7:heart-and-circulation:vessels}{kapilernya}
--- jaringan \omterm{prop:g4:journey-of-food:absorb}{darahnya} yang menerimanya.

\textbf{2.} \omterm{prop:g7:digestion-and-nutrients:absorption}{Hatinya}. Penjalurannya melayani \omterm{def:g7:digestion-and-nutrients:families}{zat gizi} yang berbagi
\omterm{prop:g4:journey-of-food:absorb}{darah} yang sama --- penyortiran dan penabungannya --- dan airnya
sekadar menumpang jalur bakunya melewatinya.

\textbf{3.} Peredaran paru dan peredaran tubuh, dalam pergantian yang
ketat melewati \omterm{def:g4:heart-and-blood:heart}{jantung} berempat ruangnya.

\textbf{4.} Ia adalah medium bagi kimia \omterm{def:g5:first-look-at-cells:cell}{selnya} --- getah dan
pembakarannya berjalan dalam keadaan \omterm{def:g7:oxygen-in-the-water:dissolved}{terlarut} --- dan pembawa bagi
segala yang diangkut \omterm{prop:g4:journey-of-food:absorb}{darahnya}; ia juga mengisi cadangan pendinginan
yang akan dibelanjakan sore harinya sebagai keringat.

\textbf{5.} \omterm{def:g1:the-five-senses:sense}{Kulitnya}: keringat --- air dan garam --- yang tugas
sesungguhnya pada hari itu adalah mendinginkan \omterm{def:g3:bones-and-muscles:muscle}{otot} yang bekerja
sepanas tungku (\cref{rem:g7:organs-at-work:warmth}).

\textbf{6.} Berkeringat membelanjakan rekening airnya; tahap
pengambilan kembali \omterm{def:g7:removing-wastes:kidneys}{ginjalnya} menjawabnya dengan mengambil kembali
nyaris seluruh air yang diayak --- \omterm{def:g7:removing-wastes:kidneys}{urine} yang sedikit, pekat dan
gelap --- sementara rasa hausnya menuntut agar rekening itu diisi
kembali.

\textbf{7.} \omterm{def:g7:what-is-respiration:respiration}{Karbon dioksida}: \omterm{def:g7:heart-and-circulation:vessels}{kapiler} \omterm{def:g3:bones-and-muscles:muscle}{otot}, \omterm{def:g7:heart-and-circulation:vessels}{vena}, \omterm{def:g4:heart-and-blood:heart}{jantung} kanan,
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolus} --- lalu diembuskan keluar. \omterm{prop:g7:removing-wastes:rubbish}{Urea}: kembalian yang sama, lalu
melewati \omterm{def:g7:how-animals-breathe:four}{paru-parunya}, \omterm{def:g4:heart-and-blood:heart}{jantung} kirinya, \omterm{def:g7:heart-and-circulation:vessels}{arteri} \omterm{def:g7:removing-wastes:kidneys}{ginjalnya}, lalu diayak
dan dikirim menuju \omterm{def:g7:removing-wastes:kidneys}{kandung kemih}.

\textbf{8.} \omterm{prop:g7:removing-wastes:rubbish}{Urea} dan garam dari \omterm{prop:g4:journey-of-food:absorb}{darah} yang berubah oleh keringatnya:
produksi buangannya tidak pernah berhenti, sehingga ayakannya pun
tidak boleh berhenti. \emph{Volume} \omterm{def:g7:removing-wastes:kidneys}{urinenya} turun --- pengambilan
kembalinya menahan airnya --- tetapi pembersihannya berjalan terus.

\textbf{9.} Permukaan \omterm{def:g7:lungs-and-gas-exchange:alveoli}{alveolusnya} lembap, dan tiap tarikan napas
menyapu keluar udara yang sudah dihangatkan dan dilembapkan --- air
pada cermin yang berembun. Sebagian dari gelas mana pun pergi sebagai
\omterm{def:g6:exploring-our-environment:conditions}{kelembapan} napasnya.

\textbf{10.} Masuknya: \omterm{prop:g7:digestion-and-nutrients:absorption}{jonjot usus}. Simpanan dan pemakaiannya: volume
\omterm{prop:g4:journey-of-food:absorb}{darah}, kimia \omterm{def:g5:first-look-at-cells:cell}{sel}, cadangan pendinginan. Keluarnya: \omterm{def:g7:removing-wastes:kidneys}{urine} (yang disetel
pengambilan kembalinya), keringat (bagian pertandingannya), \omterm{def:g6:exploring-our-environment:conditions}{kelembapan}
napasnya, dan sebagian kecil yang tidak pernah terserap, yang keluar
bersama sisanya.

\textbf{11.} Karena setiap organ menimba pasokan dari, dan membongkar
buangan ke dalam, satu \omterm{prop:g4:journey-of-food:absorb}{darah} yang dibagi bersama: sebuah \omterm{def:g7:how-animals-breathe:four}{paru-paru}
yang gagal membuat setiap \omterm{def:g3:bones-and-muscles:muscle}{otot} kelaparan, sebuah \omterm{def:g7:removing-wastes:kidneys}{ginjal} yang gagal
meracuni setiap \omterm{def:g5:first-look-at-cells:cell}{sel}, sebuah \omterm{def:g7:heart-and-circulation:vessels}{arteri} yang mengerak mencekik stasiun mana
pun yang terletak di seberangnya. Satu jaringan berarti satu nasib,
stasiun demi stasiun.

\textbf{12.} \omterm{def:g7:what-is-respiration:respiration}{Respirasi} --- api lambat di dalam setiap \omterm{def:g5:first-look-at-cells:cell}{sel}: masuknya,
pengangkutannya, pembakarannya, pembangunannya dan setiap perbatasan
buangannya ada untuk memberinya makan, memasoknya, dan mengangkut
pergi abunya.
\end{solution}
