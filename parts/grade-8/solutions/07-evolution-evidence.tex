\section*{Chapter \ref{ch:g8:evolution} --- Evidence for Evolution}

\begin{solution}{exo:g8:evolution:1}
Change in \omterm{def:g4:traits:inherited}{inherited traits} of \omterm{def:g6:ecology:levels}{populations} over generations; long-term
diversity from common ancestors.
\end{solution}

\begin{solution}{exo:g8:evolution:2}
Preserved remains/traces of past life; show extinct forms, sequences and
change through time.
\end{solution}

\begin{solution}{exo:g8:evolution:3}
Parts with shared ancestral design despite different functions; e.g.\
\omterm{def:g5:vertebrates:split}{vertebrate} forelimbs.
\end{solution}

\begin{solution}{exo:g8:evolution:4}
More similar DNA/\omterm{def:g8:genetics:gene}{genes} usually imply more recent common ancestors.
\end{solution}

\begin{solution}{exo:g8:evolution:5}
Geographic distribution of \omterm{def:g6:classification:species}{species}.
\end{solution}
