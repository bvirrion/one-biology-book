\chapter{Penganekaragaman Makhluk Hidup}\label{ch:g12:diversification-of-life}

Gandum roti punya 42 \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}, dalam enam perangkat berisi tujuh: ia
adalah jumlah tiga rumput liar yang bersilangan, dua kali, di ladang
para petani pertama. Seekor ular sanca punya tiga ratus ruas tulang
belakang dan seekor mencit tiga puluh, namun keduanya membangun tulang
belakangnya dengan \omterm{def:g10:universal-dna:gene}{gen} yang sama. Sepetak lumut kerak pada sebuah batu
adalah dua organisme, jamur dan alga, yang telah hidup sebagai satu
begitu lama sehingga tidak satu pun dapat ditemukan sendirian. Dan
seekor simpanse muda belajar memecah kacang dengan sebuah batu dengan
menonton induknya, dalam hutan yang simpansenya seratus kilometer jauh
belum pernah mempelajarinya. \omterm{def:g11:mutations:mutation}{Mutasi} dan pengocokan bab terakhir bukan
satu-satunya cara makhluk hidup menjadi berbeda; bab ini meninjau
cara-cara lainnya.

\section{Lebih daripada mutasi}

\begin{proposition}[Sumber penganekaragamannya]\label{prop:g12:diversification-of-life:sources}
\omterm{def:g11:mutations:mutation}{Mutasi} titik dan pengocokan seksualnya menciptakan \omterm{def:g10:universal-dna:gene}{alel} baru dan
gabungan baru darinya. Di luar keduanya, keragaman makhluk hidup telah
dihasilkan proses yang mengubah genomnya pada skala yang lebih besar
--- menggandakan \omterm{def:g10:universal-dna:gene}{gen}, mengimpor \omterm{def:g10:universal-dna:gene}{gen} dari \omterm{def:g10:biodiversity-scales:species}{spesies} lain, menambahkan
perangkat \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} yang utuh --- oleh perubahan cara \omterm{def:g10:universal-dna:gene}{gen} yang sudah ada
dipakai selama perkembangannya, oleh pergaulan antarspesies, dan, pada
hewan, oleh perilaku yang diwariskan sama sekali tanpa \omterm{def:g10:universal-dna:gene}{gen}.
\end{proposition}
\admitted

\section{Gen baru dari gen lama}

\begin{proposition}[Penggandaan gen dan pemencarannya]\label{prop:g12:diversification-of-life:duplication}
\omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{Pindah silang} yang tak seimbang
(\cref{ch:g12:meiosis-genetic-shuffling}) atau kesalahan penyalinan
dapat meninggalkan sebuah \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} dengan dua salinan sebuah \omterm{def:g10:universal-dna:gene}{gen}. Satu
salinannya meneruskan tugas aslinya; salinan yang lain, yang bebas dari
kekangan itu, menumpuk \omterm{def:g11:mutations:mutation}{mutasi} dan mungkin memperoleh fungsi baru ---
atau hilang. Jika berulang sepanjang waktu evolusi, hal ini
menghasilkan \emph{keluarga gen}\index{keluarga gen}: yaitu perangkat
\omterm{def:g10:universal-dna:gene}{gen} berkerabat yang berasal dari satu moyang, yang derajat
keserupaannya merekam urutan penggandaannya.
\end{proposition}
\begin{proof}[Bukti percobaan]
Ketiga \omterm{def:g11:the-eye:photoreceptors}{opsin} dan rodopsin pada \cref{ch:g11:the-eye} adalah salah satu
keluarga semacam itu; globin adalah keluarga yang lain: beberapa \omterm{def:g10:universal-dna:gene}{gen}
bagi rantai hemoglobin --- satu dipakai pada embrio, satu pada janin,
dua pada orang dewasa --- dan \omterm{def:g10:universal-dna:gene}{gen} mioglobin, yaitu simpanan oksigen
otot, terletak berkelompok pada dua \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}, 40\% sampai 80\% serupa
urutannya, dan pohonnya cocok dengan urutan munculnya kelompok
\omterm{def:g10:common-ancestry:bodyplan}{vertebrata}. Genom membawa ratusan keluarga semacam itu, dan bagian yang
cukup besar dari tiap genom terdiri atas ruas yang tergandakan.
\end{proof}

\begin{proposition}[Pemindahan gen mendatar]\label{prop:g12:diversification-of-life:horizontal}
\omterm{def:g10:universal-dna:gene}{Gen} juga dapat masuk ke dalam sebuah genom dari \omterm{def:g10:biodiversity-scales:species}{spesies} lain. Bakteri
bertukar plasmid dengan bebas
(\cref{ch:g11:antibiotic-resistance}); virus membawa serpihan \omterm{def:g10:universal-dna:information}{DNA}
inangnya dari satu \omterm{def:g10:cells-common-unit:cell}{sel} ke \omterm{def:g10:cells-common-unit:cell}{sel} lain dan meninggalkan salinan \omterm{def:g10:universal-dna:gene}{gennya}
sendiri di sana; dan perbandingan genom menunjukkan bahwa bahkan hewan
dan tumbuhan pun telah memperoleh \omterm{def:g10:universal-dna:gene}{gen} dengan cara ini. Sekitar 8\%
genom manusia berasal dari virus, dan sedikitnya satu dari \omterm{def:g10:universal-dna:gene}{gen} itu,
yang dahulu menjadi alat sebuah virus untuk memadukan \omterm{def:g10:cells-common-unit:cell}{sel}, kini
membangun plasenta.
\end{proposition}
\admitted

\begin{proposition}[Hibridisasi dan poliploidi]\label{prop:g12:diversification-of-life:polyploidy}
Dua \omterm{def:g10:biodiversity-scales:species}{spesies} yang berkerabat kadang-kadang dapat bersilangan; hibridnya
membawa satu perangkat \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} dari masing-masing dan lazimnya mandul,
karena \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya} tidak punya pasangan untuk berpasangan pada
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosisnya}. Jika, karena sebuah kesalahan pembelahan, hibridnya
melipatduakan \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya}, tiap \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya} memperoleh pasangan,
\omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosisnya} bekerja, dan sebuah \omterm{def:g10:biodiversity-scales:species}{spesies} baru yang subur --- sebuah
\emph{poliploid}\index{poliploidi} --- seketika ada, terpencil dari
kedua induknya. Separuh seluruh \omterm{def:g10:biodiversity-scales:species}{spesies} tumbuhan berbunga punya
poliploidi dalam sejarahnya; gandum roti, kapas, tembakau dan kentang
termasuk di antaranya.
\end{proposition}
\begin{proof}[Bukti percobaan]
Gandum: einkorn liar punya $2n = 14$; emmer, $2n = 28$, memuat
perangkat einkornnya dan perangkat rumput liar lain, \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} demi
\omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}; gandum roti, $2n = 42$, menambahkan perangkat rumput yang
ketiga. Ketiga perangkatnya dapat dikenali dalam \omterm{def:g11:cell-cycle-mitosis:chromosome}{kariotipe} gandum
rotinya dan dicocokkan dengan \omterm{def:g10:biodiversity-scales:species}{spesies} liar yang masih hidup, dan
persilangannya dapat diulang di laboratorium. Sebatang rumput pantai
yang muncul di pesisir Eropa pada abad kesembilan belas, yang subur dan
kuat, punya perangkat \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} yang berlipat dua dari hibrid mandul dua
\omterm{def:g10:biodiversity-scales:species}{spesies} yang tidak pernah bertemu sebelum kapal membawa salah satunya
menyeberangi Atlantik.
\end{proof}

\section{Gen yang sama, pemakaian yang berbeda}

\begin{proposition}[Gen perkembangan dan saat pengungkapannya]\label{prop:g12:diversification-of-life:development}
\omterm{def:g10:universal-dna:gene}{Gen} yang menata \omterm{def:g10:common-ancestry:bodyplan}{pola tubuh} seekor hewan selama perkembangannya ---
ujung mana yang menjadi kepala, di mana tungkainya berada, berapa ruas
yang dimiliki badannya --- dibagi bersama, dengan sedikit perubahan
urutan, lintas hewan: keluarga \emph{gen perkembangan}\index{gen perkembangan}
yang sama menata seekor lalat, seekor mencit dan seorang
manusia. Yang berbeda antarspesies bukanlah \omterm{def:g10:universal-dna:gene}{gen} itu melainkan
\emph{di mana, kapan dan sekuat apa} \omterm{def:g10:universal-dna:gene}{gen} itu dinyalakan. Perubahan
pengaturan sebuah \omterm{prop:g12:diversification-of-life:development}{gen perkembangan} dapat mengubah bentuk sebuah bagian
tubuh atau jumlah bagian yang berulang tanpa mengubah satu \omterm{def:g11:gene-expression:protein}{protein} pun.
\end{proposition}
\begin{proof}[Bukti percobaan]
\omterm{def:g10:universal-dna:gene}{Gen} yang menetapkan daerah badannya terungkapkan dalam urutan yang sama
dari kepala ke ekor pada seekor mencit dan seekor ular sanca; pada ular
sancanya daerah yang membuat ruas tulang belakang bertulang rusuk
membentang di sepanjang hampir seluruh tubuhnya, dan daerah yang akan
membuat tungkainya tidak pernah menerima isyarat untuk berbuat demikian:
tiga ratus ruas tulang belakang dan tanpa kaki, dari \omterm{def:g10:universal-dna:gene}{gen} yang sama yang
terungkapkan sepanjang bentang yang berbeda. Di antara burung finch
Darwin, dalamnya paruhnya mengikuti aras pengungkapan satu isyarat
pertumbuhan dalam paruh embrionya: makin banyak isyaratnya, makin dalam
paruhnya. Menaikkan isyarat itu secara buatan pada embrio ayam
menghasilkan paruh yang dalam dan menyerupai paruh finch.
\end{proof}

\begin{example}[Paruh menurut dosisnya]\label{ex:g12:diversification-of-life:beaks}
Tiga belas \omterm{def:g10:biodiversity-scales:species}{spesies} burung finch di satu gugus pulau punya paruh dari
setipis jarum sampai sedalam pemecah kacang, dan makan sesuai
paruhnya. \omterm{def:g10:universal-dna:gene}{Gen} paruhnya sama; embrio \omterm{def:g10:biodiversity-scales:species}{spesies} berparuh dalam
mengungkapkan satu isyarat pertumbuhan lebih awal dan lebih kuat dalam
jaringan yang menjadi paruhnya. Satu perubahan pada saat atau jumlah
satu isyarat, bukan sebuah \omterm{def:g10:universal-dna:gene}{gen} baru, memisahkan pemakan serangga dari
pemecah biji --- dan itulah sebabnya beda semacam itu dapat muncul
dalam beberapa ribu keturunan.
\end{example}

\section{Hidup bersama: simbiosis}

\begin{definition}[Simbiosis]\label{def:g12:diversification-of-life:symbiosis}
Sebuah \emph{simbiosis}\index{simbiosis} adalah pergaulan dua \omterm{def:g10:biodiversity-scales:species}{spesies}
yang erat dan berlangsung lama yang keduanya beroleh untung. Lumut
kerak adalah jamur yang merumahkan \omterm{def:g10:cells-common-unit:cell}{sel} alga yang memberinya makan lewat
\omterm{def:g10:cell-metabolism:autotrophy}{fotosintesis} sementara jamurnya menaungi dan mengairinya; karang adalah
hewan yang merumahkan alga dalam jaringannya; akar sebagian besar
tumbuhan terbungkus jamur yang memasok mineral sebagai penukar gula;
usus hewan memuat bakteri yang mencerna apa yang tidak dapat dicerna
hewannya. Pergaulannya menghasilkan organisme, struktur dan cara hidup
yang tidak dapat dicapai satu pun mitranya sendirian.
\end{definition}

\begin{proposition}[Simbiosis di dalam tiap sel]\label{prop:g12:diversification-of-life:endosymbiosis}
\omterm{def:g10:cells-common-unit:organelle}{Mitokondria} tiap \omterm{def:g10:cells-common-unit:prokaryote}{sel eukariotik}, dan \omterm{def:g10:cells-common-unit:organelle}{kloroplas} \omterm{def:g10:cells-common-unit:cell}{sel} tumbuhan, berasal
dari bakteri yang hidup bebas yang ditelan sebuah \omterm{def:g10:cells-common-unit:cell}{sel} moyang sekitar
dua miliar tahun lalu dan lalu tinggal. \omterm{def:g10:cells-common-unit:prokaryote}{Sel eukariotiknya} sendiri
adalah hasil sebuah \omterm{def:g12:diversification-of-life:symbiosis}{simbiosis}.
\end{proposition}
\begin{proof}[Bukti percobaan]
\omterm{def:g10:cells-common-unit:organelle}{Mitokondria} dan \omterm{def:g10:cells-common-unit:organelle}{kloroplas} punya \omterm{def:g10:universal-dna:information}{DNA-nya} sendiri, yaitu lingkaran kecil
seperti \omterm{def:g10:universal-dna:information}{DNA} bakteri, dan \omterm{def:g10:cells-common-unit:organelle}{ribosomnya} sendiri, dari jenis bakteri, yang
peka terhadap \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotik} yang menghalangi \omterm{def:g10:cells-common-unit:organelle}{ribosom} bakteri tetapi tidak
menghalangi \omterm{def:g10:cells-common-unit:organelle}{ribosom} eukariot; keduanya berlipat ganda dengan membelah
dua, dan tidak dapat dibuat baru oleh \omterm{def:g10:cells-common-unit:cell}{selnya}; keduanya terbungkus dua
membran, dengan membran dalamnya menyerupai membran bakteri; dan
\omterm{def:g10:universal-dna:gene}{gennya} paling dekat, dalam urutannya, dengan \omterm{def:g10:universal-dna:gene}{gen} kelompok bakteri hidup
tertentu --- bakteri ungu bagi \omterm{def:g10:cells-common-unit:organelle}{mitokondria}, sianobakteri bagi \omterm{def:g10:cells-common-unit:organelle}{kloroplas}.
\end{proof}

\section{Keragaman tanpa gen: perilaku}

\begin{proposition}[Perilaku yang diwariskan]\label{prop:g12:diversification-of-life:behaviour}
Pada banyak hewan, sebagian dari apa yang dikerjakan seorang individu
dipelajari dari yang lain --- lewat peniruan, pengajaran atau penyalinan
--- dan diteruskan pada gilirannya: yaitu sebuah
\emph{budaya}\index{budaya (hewan)}, yang diwariskan tanpa \omterm{def:g10:universal-dna:gene}{gen}. Karena
itu populasi \omterm{def:g10:biodiversity-scales:species}{spesies} yang sama dapat berbeda perilakunya sebagaimana
berbeda \omterm{def:g10:universal-dna:gene}{alelnya}, dan sebuah perilaku dapat menyebar dalam sebuah
populasi dalam satu keturunan alih-alih sepanjang banyak keturunan yang
akan dituntut sebuah \omterm{def:g11:mutations:mutation}{mutasi}.
\end{proposition}
\begin{proof}[Bukti percobaan]
Masyarakat simpanse di hutan yang berbeda memakai alat yang berbeda ---
batu untuk memecah kacang di sini, ranting untuk memancing rayap di
sana, daun sebagai spons di tempat lain --- dan bedanya tidak mengikuti
kekerabatan genetik atau tersedianya bahan; simpanse muda memperoleh
perangkat masyarakatnya dengan menonton yang lebih tua. Burung penyanyi
yang dibesarkan terpencil menyanyikan lagu yang kasar; jika dibesarkan
sambil mendengar yang dewasa, ia menyanyikan dialek setempat, yang
berubah lintas wilayahnya seperti logat manusia. Sekelompok monyet
makaka di sebuah pulau belajar mencuci ubi jalar di laut dari satu
betina muda dalam sepuluh tahun, dan keturunannya masih melakukannya.
\end{proof}

\begin{omfigure}
\includegraphics[width=0.6\linewidth]{images/book2/ai/g12-diversif-chimp-tool.png}

{\small Seekor simpanse memancing rayap dengan ranting yang telah
disiapkan. Tekniknya dipelajari dengan menonton, dan berbeda dari satu
masyarakat ke masyarakat lain: yaitu sebuah tradisi, yang diwariskan tanpa \omterm{def:g10:universal-dna:gene}{gen}.}
\end{omfigure}

\begin{method}[Menyebutkan sumber sebuah beda]\label{met:g12:diversification-of-life:name}
Dihadapkan pada sebuah beda antara dua organisme atau dua populasi:
\begin{enumerate}
  \item Apakah bedanya pada urutan sebuah \omterm{def:g10:universal-dna:gene}{gen} (\omterm{def:g11:mutations:mutation}{mutasi}, \omterm{def:g10:universal-dna:gene}{alel})? Pada
        jumlah salinan sebuah \omterm{def:g10:universal-dna:gene}{gen} (penggandaan)? Pada kehadiran sebuah
        \omterm{def:g10:universal-dna:gene}{gen} dari tempat lain (pemindahan mendatar)? Pada jumlah
        perangkat \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya} (poliploidi)?
  \item Apakah bedanya pada kapan, di mana atau seberapa banyak sebuah
        \omterm{def:g10:universal-dna:gene}{gen} bersama diungkapkan selama perkembangannya (pengaturan)?
  \item Apakah bedanya merupakan hasil pergaulan dengan \omterm{def:g10:biodiversity-scales:species}{spesies} lain
        (\omterm{def:g12:diversification-of-life:symbiosis}{simbiosis})?
  \item Apakah bedanya dipelajari (perilaku yang diwariskan lewat
        budaya) --- diwariskan, tetapi tidak secara genetik?
\end{enumerate}
Lebih dari satu jawaban sering benar; pertanyaannya adalah daftar isi
bab ini.
\end{method}

\begin{remark}[Penganekaragaman dan seleksi]\label{rem:g12:diversification-of-life:selection}
Tiap proses di sini menghasilkan keragaman; tidak satu pun memutuskan
apa yang selamat. Gandum yang \omterm{prop:g12:diversification-of-life:polyploidy}{poliploid}, pola pengungkapan seekor ular
sanca, sepetak lumut kerak, sebuah tradisi memecah kacang, semuanya
bertahan karena berhasil di tempat kemunculannya. Pemilahan keragamannya
--- seleksi, dan kebetulan --- menjadi pokok bab berikutnya; bab ini
telah membentangkan berapa banyak yang harus dipilah, dan berapa banyak
cara membuatnya.
\end{remark}

\section{Soal: Tiga Rumput dalam Seketul Roti}

\begin{problem}[{Soal akhir pekan --- sejarah gandum roti yang disusun
ulang dari kromosomnya, gen seekor ular sanca yang dibaca di samping
gen seekor mencit, dan bakteri di dalam tiap sel yang diadili}]
\label{pb:g12:diversification-of-life:1}
Einkorn (AA), rumput kambing 1 (BB) dan rumput kambing 2 (DD)
masing-masing punya $2n = 14$. Emmer adalah AABB dan gandum roti AABBDD.

\medskip
\noindent\textbf{Bagian I --- \omterm{def:g11:cell-cycle-mitosis:chromosome}{Kromosom} gandumnya.}
\begin{enumerate}
  \item Sebutkan jumlah \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} emmer dan gandum rotinya.
  \item Hibrid einkorn dan rumput kambing 1 punya 14 \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}, 7 dari
        tiap macamnya. Berapa pasang homolog yang terbentuk pada
        \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosisnya}? Seperti apa gametnya?
  \item Jelaskan mengapa pelipatduaan \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya} memulihkan
        kesuburannya: hitunglah pasangannya.
  \item Hibrid emmer dan rumput kambing 2 punya 21 \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}. Yang mana
        di antaranya yang dapat berpasangan, dan mengapa ia mandul?
  \item Setelah pelipatduaannya, gandum roti punya 42. Berapa pasangan
        yang terbentuk pada \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{meiosisnya}, dan berapa \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} yang dibawa
        sebutir serbuk sarinya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Membaca sejarahnya.}
\begin{enumerate}[resume]
  \item Gandum roti membawa tiga salinan sebagian besar \omterm{def:g10:universal-dna:gene}{gennya}, satu
        pada tiap perangkatnya. Jelaskan mengapa \omterm{def:g11:mutations:mutation}{mutasi} yang memusnahkan
        satu salinannya lazimnya tanpa pengaruh yang kasatmata, dan apa
        yang dibolehkannya seiring waktu.
  \item Emmer dibudidayakan 10\,000 tahun lalu dan gandum roti muncul
        pada situs arkeologi 8\,000 tahun lalu. Di mana dan kapan
        hibridisasi keduanya barangkali terjadi? Apa yang harus tumbuh
        berdampingan?
  \item Persilangan laboratorium antara emmer dan rumput kambing 2,
        yang disusul pelipatduaan kimia, memberi tumbuhan subur yang
        menyerupai gandum roti. Kesimpulan mana dari sejarah ini yang
        dibenarkannya?
  \item Gandum roti tidak dapat bersilangan dengan einkorn untuk
        memberi keturunan yang subur. Apakah ia \omterm{def:g10:biodiversity-scales:species}{spesies} yang terpisah?
        Berikan alasannya dengan rumusan \cref{ch:g10:biodiversity-scales}.
  \item Seorang pemulia tanaman ingin memasukkan \omterm{def:g10:universal-dna:gene}{gen} ketahanan penyakit
        dari rumput kambing 2 ke dalam gandum roti. Jelaskan mengapa
        hal itu lebih mudah daripada dari tumbuhan yang tak berkerabat.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- \omterm{def:g10:universal-dna:gene}{Gen} ular sancanya.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:universal-dna:gene}{Gen} daerah badannya 98\% serupa antara mencit dan ular sanca.
        Apa yang dinyatakannya tentang sumber beda pada tubuhnya?
  \item Pada embrio mencitnya, \omterm{def:g10:universal-dna:gene}{gen} ``badan bertulang rusuk''-nya aktif
        pada 13 ruas; pada ular sancanya pada sekitar 300. Sebutkan
        macam perubahan yang terlibat.
  \item Embrio ular sancanya membentuk tunas mungil di tempat tungkai
        belakangnya akan berada, yang lalu berhenti tumbuh. Apa yang
        disarankannya tentang \omterm{def:g10:universal-dna:gene}{gen} tungkainya dan isyarat yang akan
        mengaktifkannya?
  \item Mencit yang direkayasa agar mengungkapkan \omterm{def:g10:universal-dna:gene}{gen} ``badan bertulang
        rusuk''-nya pada daerah pinggangnya menumbuhkan tulang rusuk
        tambahan di sana. Apa yang ditambahkan percobaan itu pada
        perbandingannya?
  \item Bandingkan kedua macam perubahannya: perubahan gandum roti dan
        perubahan ular sancanya. Mana yang mengubah besarnya genom, mana
        yang mengubah pemakaiannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Bakteri di dalamnya.}
\begin{enumerate}[resume]
  \item Sebutkan empat ciri \omterm{def:g10:cells-common-unit:organelle}{mitokondria} yang cocok dengan asal bakteri,
        dan untuk tiap cirinya sebutkan apa yang akan kamu harapkan
        seandainya \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya} dibangun \omterm{def:g10:cells-common-unit:cell}{selnya} dari nol.
  \item \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya} hanya membawa 37 \omterm{def:g10:universal-dna:gene}{gen}, sedangkan
        \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya} memerlukan sekitar 1500 \omterm{def:g11:gene-expression:protein}{protein}. Di mana \omterm{def:g10:universal-dna:gene}{gen}
        lainnya berada, dan apa yang terjadi padanya sepanjang dua
        miliar tahun?
  \item Tiap \omterm{def:g10:cells-common-unit:organelle}{mitokondria} seseorang berasal dari \omterm{def:g10:cells-common-unit:cell}{sel} telurnya, tidak
        satu pun dari spermanya. Apa akibatnya bagi pewarisan \omterm{def:g10:universal-dna:gene}{gen}
        \omterm{def:g10:cells-common-unit:organelle}{mitokondria}, dan bagi pelacakan keturunan dari pihak ibu?
  \item Lumut kerak dan \omterm{def:g10:cells-common-unit:organelle}{mitokondria} sama-sama \omterm{def:g12:diversification-of-life:symbiosis}{simbiosis}. Dalam hal apa
        \omterm{def:g12:diversification-of-life:symbiosis}{simbiosis} \omterm{def:g10:cells-common-unit:organelle}{mitokondrianya} melangkah lebih jauh?
  \item Nyatakan hasilnya: ketiga \omterm{def:g10:biodiversity-scales:species}{spesies} dalam seketul roti dan jumlah
        \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} yang disumbangkan masing-masing; satu kata yang
        menyebutkan apa yang berbeda antara pemakaian \omterm{def:g10:universal-dna:gene}{gen} bersama oleh
        ular sanca dan oleh mencitnya; dan \omterm{def:g10:cells-common-unit:organelle}{organel} yang dahulu adalah
        sebuah bakteri.
\end{enumerate}
\end{problem}
