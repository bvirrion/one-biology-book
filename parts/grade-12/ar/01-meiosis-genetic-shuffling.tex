\chapter{الانقسام المنصّف وخلط المورّثات}\label{ch:g12:meiosis-genetic-shuffling}

والدان، لكل منهما 46 صبغيًا، ينتجان طفلًا له 46 --- لا 92. فبين جسدي
الوالدين والطفل، في مكان ما، ينتصف العدد، وينتصف على نحو يوزع على كل
طفل يدًا لم يمسك بمثلها أخ له قط: الوالدان نفسهما، وأربعون سنة من
الأطفال، ولا اثنان متشابهان إلا التوأمين المتطابقين. والانتصاف انقسام
خاص يسمى \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{الانقسام المنصّف}؛ والتوزيع هو ما يفعله \omterm{def:g10:universal-dna:information}{بالصبغيات} في الطريق.
ويتابع هذا الفصل \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة عبر \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{الانقسام المنصّف}، ويحصي التوافيق التي
يستطيع إنتاجها، ويقرأ نتائج تهجينات تكشف، من الخارج، ما جرى في الداخل.

\section{انتصاف الطقم}

\begin{definition}[ثنائي الصيغة وأحادي الصيغة والانقسام المنصّف]\label{def:g12:meiosis-genetic-shuffling:meiosis}
\omterm{def:g10:cells-common-unit:cell}{الخلية} الحاملة نسختين من كل \omterm{def:g10:universal-dna:information}{صبغي} --- واحدة من كل والد، ويكوّن الاثنان
زوجًا من \emph{الصبغيات المتماثلة} --- هي \omterm{def:g10:cells-common-unit:cell}{خلية} \emph{ثنائية الصيغة
الصبغية}\index{ثنائي الصيغة الصبغية}، ويكتب طقمها $2n$ ($2n = 46$ عند
الإنسان). \omterm{def:g10:cells-common-unit:cell}{والخلية} الحاملة نسخة واحدة من كل \omterm{def:g10:universal-dna:information}{صبغي} \emph{أحادية الصيغة
الصبغية}\index{أحادي الصيغة الصبغية}، ويكتب طقمها $n$. و\emph{الانقسام
المنصّف}\index{انقسام منصّف} تتابع من انقسامين تنتج به \omterm{def:g10:cells-common-unit:cell}{خلية} واحدة ثنائية
الصيغة، بعد تضاعف واحد لجزيء \omterm{def:g10:universal-dna:information}{DNA}، أربع \omterm{def:g10:cells-common-unit:cell}{خلايا} أحادية الصيغة --- هي الأمشاج
أو سلائفها. و\emph{الإخصاب} يجمع مشيجين أحاديي الصيغة فيعيد العدد
الثنائي: وتناوب الانقسام المنصّف والإخصاب هو دورة كل \omterm{def:g10:biodiversity-scales:species}{نوع} يتكاثر جنسيًا.
\end{definition}

\begin{proposition}[الانقسامان]\label{prop:g12:meiosis-genetic-shuffling:divisions}
قبل \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{الانقسام المنصّف} يتضاعف \omterm{def:g10:universal-dna:information}{DNA}: فيصير لكل \omterm{def:g10:universal-dna:information}{صبغي} \omterm{def:g11:cell-cycle-mitosis:chromosome}{صبغيدان}. ثم:
\begin{itemize}
  \item \emph{الانقسام الأول} (المنصّف الأول، \emph{الاختزالي}):
        تتزاوج الصبغيات المتماثلة، اثنين اثنين، على خط الاستواء، ويشد
        عضوا كل زوج إلى قطبين متقابلين. فتتلقى كل \omterm{def:g10:cells-common-unit:cell}{خلية} بنت \emph{صبغيًا
        واحدًا من كل زوج}، ما زال مؤلفًا من \omterm{def:g11:cell-cycle-mitosis:chromosome}{صبغيدين}: $n$ صبغيًا، وكمية
        \omterm{def:g10:universal-dna:information}{DNA} تساوي $q$ (كمية \omterm{def:g10:cells-common-unit:cell}{الخلية} ثنائية الصيغة قبل التضاعف).
  \item \emph{الانقسام الثاني} (المنصّف الثاني، \emph{المعادل}): وهو
        \omterm{def:g11:cell-cycle-mitosis:cycle}{كالانقسام المتساوي}، ينفصل فيه صبغيدا كل \omterm{def:g10:universal-dna:information}{صبغي}. وتتلقى كل \omterm{def:g10:cells-common-unit:cell}{خلية} من
        \omterm{def:g10:cells-common-unit:cell}{الخلايا} الأربع $n$ صبغيًا وحيد \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغيد}، وكمية \omterm{def:g10:universal-dna:information}{DNA} تساوي $q/2$.
\end{itemize}
فالانقسام الأول ينصّف عدد \omterm{def:g10:universal-dna:information}{الصبغيات}؛ والثاني ينصّف كمية \omterm{def:g10:universal-dna:information}{DNA} فيعيدها إلى
\omterm{def:g11:cell-cycle-mitosis:chromosome}{صبغيد} واحد لكل \omterm{def:g10:universal-dna:information}{صبغي}.
\end{proposition}
\admitted

\begin{example}[أرقام بشرية]\label{ex:g12:meiosis-genetic-shuffling:numbers}
\omterm{def:g10:cells-common-unit:cell}{خلية} جنسية في الخصية، $2n = 46$، تتضاعف فتصير 46 صبغيًا ثنائي \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغيد}
($2q$)؛ وبعد المنصّف الأول، خليتان من 23 صبغيًا ثنائي \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغيد} ($q$)؛
وبعد المنصّف الثاني، أربع \omterm{def:g10:cells-common-unit:cell}{خلايا} منوية من 23 صبغيًا وحيد \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغيد}
($q/2$). وعند الإخصاب، $23 + 23 =
46$ و$q/2 + q/2 = q$: فتعود اللاقحة \omterm{def:g10:cells-common-unit:cell}{خلية} ثنائية الصيغة في راحتها. وفي
المبيض، تنتج الانقسامات نفسها بييضة واحدة كبيرة وثلاث \omterm{def:g10:cells-common-unit:cell}{خلايا} صغيرة
تطرح؛ والحساب هو الحساب.
\end{example}

\section{الخلط}

\begin{proposition}[التوزع المستقل]\label{prop:g12:meiosis-genetic-shuffling:assortment}
في الطور الاستوائي الأول، يصطف كل زوج من المتماثلين على خط الاستواء
مستقلًا عن سائر الأزواج: فأي عضو من الزوج يذهب إلى أي قطب أمر تقرره
المصادفة، زوجًا زوجًا. ومن ثم تستطيع \omterm{def:g10:cells-common-unit:cell}{خلية} لها $n$ زوجًا أن تنتج $2^n$
توفيقًا مختلفًا من \omterm{def:g10:universal-dna:information}{الصبغيات} الأبوية والأمومية في أمشاجها --- أي
$2^{23} \approx \num{8.4e6}$ عند الإنسان. والإخصاب، إذ يجمع مشيجين
كهذين على غير انتظام، يعطي $2^{23}
\times 2^{23} \approx \num{7e13}$ توفيقًا من \omterm{def:g10:universal-dna:information}{الصبغيات} لأطفال زوجين
اثنين، قبل إحصاء أي مصدر آخر للتنوع.
\end{proposition}
\admitted

\begin{proposition}[التصالب]\label{prop:g12:meiosis-genetic-shuffling:crossingover}
بينما المتماثلان متزاوجان، في أول المنصّف الأول، ينكسر \omterm{def:g11:cell-cycle-mitosis:chromosome}{صبغيدان} غير
شقيقين عند النقطة نفسها ويلتحمان متقاطعين: وهو \emph{التصالب}\index{تصالب}.
فيصير كل \omterm{def:g11:cell-cycle-mitosis:chromosome}{صبغيد}، فيما وراء نقطة التبادل، حاملًا \omterm{def:g10:universal-dna:gene}{أليلات} المتماثل الآخر.
\omterm{def:g10:universal-dna:information}{والصبغيات} التي تبلغ الأمشاج ليست إذن صبغيات الوالدين بل صبغيات
\emph{مؤتلفة}، تمزج على طولها \omterm{def:g10:universal-dna:gene}{الأليلات} المتلقاة من الجدين. ويقع تبادل
إلى ثلاثة تبادلات على كل زوج في كل \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{انقسام منصّف}، في مواضع متغيرة: فعدد
الأمشاج الممكنة، عمليًا، غير محدود.
\end{proposition}
\begin{proof}[شواهد]
تحت المجهر، تبدي المتماثلات المتزاوجة في المنصّف الأول نقاطًا متقاطعة،
هي \emph{التصالبات}، يبدل عندها \omterm{def:g11:cell-cycle-mitosis:chromosome}{صبغيدان} شريكيهما بادي العيان.
والتهجينات التي يقع فيها مورّثتان على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} نفسه تعطي، بين النسل،
أقلية تحمل توافيق جديدة من أليلي المورّثتين --- توافيق لم يحملها أي من
الوالدين --- بنسبة تتوقف على المسافة بين المورّثتين؛ والمورّثتان
المتباعدتان على \omterm{def:g10:universal-dna:information}{صبغي} واحد تأتلفان بتواتر المورّثتين الواقعتين على
صبغيين مختلفين. والواسمات الجزيئية، متابعة عبر العائلات، تبين قطع كل
\omterm{def:g10:universal-dna:information}{صبغي} وهي تتبدل بين الأصلين الجدّيين عند نقطة أو نقطتين لكل \omterm{def:g10:universal-dna:information}{صبغي} في كل
جيل.
\end{proof}

\begin{method}[التهجين الاختباري]\label{met:g12:meiosis-genetic-shuffling:testcross}
لمعرفة ما ينتجه فرد من أمشاج، هجّنه بشريك لا يحمل إلا \omterm{def:g10:universal-dna:gene}{أليلات} متنحية
(\emph{تهجين اختباري}): فيبدي كل نسل عندئذ \omterm{def:g10:universal-dna:gene}{الأليلات} التي حملها مشيج
الوالد المختبر بالضبط. ولمورّثتين، والوالد المختبر $AaBb$ والشريك
$aabb$:
\begin{enumerate}
  \item أحص أصناف النسل الأربعة: $AB$ و$Ab$ و$aB$ و$ab$ (وكل منها يحمل
        أيضًا $ab$ من الشريك).
  \item إذا تساوى تواتر الأربعة ($1:1:1:1$)، فالمورّثتان تقعان على
        صبغيين مختلفين وتتوزعان توزعًا مستقلًا.
  \item وإذا كان صنفان (التوفيقان الأبويان) أكثر بكثير من الصنفين
        الآخرين (المؤتلفين)، فالمورّثتان تقعان على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} نفسه؛
        والمؤتلفان يأتيان من تصالب، ونسبتهما تقيس المسافة بين
        المورّثتين.
  \item والمؤتلفان دائمًا أقلية، ودائمًا في صنفين متساويين؛ والأبويان
        كذلك.
\end{enumerate}
\end{method}

\begin{example}[تهجينان في ذبابة الفاكهة]\label{ex:g12:meiosis-genetic-shuffling:fly}
الذبابة A، متغايرة \omterm{def:g10:universal-dna:gene}{الأليلين} في لون الجسم وشكل الجناح، هجّنت تهجينًا

سوداء عادية --- أربعة أصناف متساوية، فالمورّثتان على صبغيين مختلفين.
والذبابة B، متغايرة \omterm{def:g10:universal-dna:gene}{الأليلين} في لون الجسم ولون العين: 412 رمادية
حمراء، و405 سوداء أرجوانية، و44 رمادية أرجوانية، و39 سوداء حمراء ---
صنفان أبويان نحو 45\% لكل منهما، وصنفان مؤتلفان نحو 5\%: فالمورّثتان
على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} نفسه، متقاربتان، يفصل بينهما تصالب في 9\% من الانقسامات
المنصّفة.
\end{example}

\section{حين يخطئ الخلط}

\begin{proposition}[أخطاء الانقسام المنصّف]\label{prop:g12:meiosis-genetic-shuffling:errors}
يحدث أحيانًا أن يخفق زوج من المتماثلين في الانفصال في المنصّف الأول، أو
\omterm{def:g11:cell-cycle-mitosis:chromosome}{صبغيدان} في المنصّف الثاني: فيتلقى مشيج نسختين من \omterm{def:g10:universal-dna:information}{صبغي} ولا يتلقى آخر
شيئًا. وإذا أخصبا أعطيا لاقحة بثلاث نسخ (\emph{تثلث \omterm{def:g10:universal-dna:information}{صبغي}}) أو بنسخة
واحدة (\emph{انفراد \omterm{def:g10:universal-dna:information}{صبغي}}). ومعظم هذه اللاقحات لا تنمو؛ وما ينمو منها
يحمل طقمًا من السمات المميزة. فالتثلث \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} 21 (نحو ولادة واحدة من كل
700) يسبب إعاقة ذهنية وعيوبًا قلبية، وهو أشيعها؛ ويرتفع تواتره ارتفاعًا
حادًا مع عمر الأم. \omterm{prop:g12:meiosis-genetic-shuffling:crossingover}{والتصالب} غير المتساوي --- إذ يتبادل \omterm{def:g11:cell-cycle-mitosis:chromosome}{صبغيدان} قطعتين
غير متساويتين --- ينتج صبغيًا \omterm{def:g10:universal-dna:gene}{بمورّثة} مضاعفة وآخر بحذف: وهي الآلية التي
تضاعفت بها \omterm{def:g10:universal-dna:gene}{مورّثات} \omterm{def:g11:the-eye:photoreceptors}{الأوبسين} في \cref{ch:g11:the-eye}، والتي تنشأ بها
\omterm{def:g10:universal-dna:gene}{مورّثات} جديدة (\cref{ch:g12:diversification-of-life}).
\end{proposition}
\admitted

\begin{example}[صبغيات جنسية مغلوطة العدد]\label{ex:g12:meiosis-genetic-shuffling:sexchrom}
مشيج بلا \omterm{def:g10:universal-dna:information}{صبغي} جنسي، إذا أخصبه مشيج حامل X، أعطى لاقحة XO: فتاة قصيرة

خصيتاه صغيرتين ولا تصنعان منويات. وفي الحالتين تقرر \omterm{prop:g11:becoming-male-female:sry}{SRY} طبيعة القند كما
في \cref{ch:g11:becoming-male-female}؛ ويقرر عدد صبغيات X كثيرًا من

\end{example}

\begin{remark}[لماذا الجنس]\label{rem:g12:meiosis-genetic-shuffling:whysex}
الجرثومة تنسخ نفسها؛ والوردة تنبت من عقلة. والتكاثر الجنسي أكلف ---
والدان، \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{وانقسام منصّف}، وبحث عن قرين --- وحاصله نسل كل فرد منه مختلف عن
والديه وعن إخوته. وذلك الاختلاف هو المقصود: ففي بيئة متغيرة، وفي وجه
طفيليات تتطور، يرجح أن تضم عشيرة من أفراد متنوعين أفرادًا ينجون أكثر مما
تضم عشيرة من النسخ. وخلط هذا الفصل هو المادة الخام التي سيفرزها انتقاء
الفصلين التاليين.
\end{remark}

\section{مسألة: ذبابات عالم الوراثة}

\begin{problem}[{مسألة نهاية الأسبوع --- مورّثتان تتابعان عبر انقسام
منصّف وتهجين اختباري: الأمشاج محصاة، والارتباط مقرَّر، والائتلاف مقيس،
وعدد الأيدي التي يستطيع زوجان توزيعها}]
\label{pb:g12:meiosis-genetic-shuffling:1}
في ذبابة الفاكهة، \omterm{def:g10:universal-dna:gene}{الأليل} $b$ (الجسم الأسود) \omterm{def:g11:genes-and-disease:genetic}{متنح} أمام $b^+$ (الرمادي)،
و$vg$ (الأجنحة الضامرة) \omterm{def:g11:genes-and-disease:genetic}{متنح} أمام $vg^+$ (العادية). هجّنت أنثى رمادية
عادية الجناحين، والداها من سلالة نقية رمادية عادية وسلالة نقية سوداء
ضامرة، بذكر أسود ضامر الجناحين. والنسل: 965 رمادية عادية، و944 سوداء
ضامرة، و206 رمادية ضامرة، و185 سوداء عادية.

\medskip
\noindent\textbf{القسم الأول --- التهجين.}
\begin{enumerate}
  \item أعط \omterm{def:g11:enzymes-and-phenotype:phenotype}{النمط الوراثي} للأنثى وللذكر، مورّثةً مورّثةً.
  \item ما الأمشاج التي ينتجها الذكر؟ ولم يجعل ذلك التهجين تهجينًا
        اختباريًا؟
  \item اذكر الأمشاج الأربعة التي تستطيع الأنثى إنتاجها، والنسل الذي
        يعطيه كل منها.
  \item احسب النسبة المئوية لكل صنف بين النسل البالغ 2300.
  \item هل المورّثتان على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} نفسه أم على صبغيين مختلفين؟ برهن من
        النسب.
\end{enumerate}

\medskip
\noindent\textbf{القسم الثاني --- داخل \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{الانقسام المنصّف} عند الأنثى.}
\begin{enumerate}[resume]
  \item أي الصنفين هما التوفيقان الأبويان، ومن أين يأتيان في صبغيات
        الأنثى؟
  \item وأيهما مؤتلف؟ وبأي حدث من أحداث المنصّف الأول نشأ؟
  \item احسب تواتر الائتلاف (المؤتلفون على المجموع). وأي كسر من
        الانقسامات المنصّفة عند الأنثى وقع فيه تصالب بين المورّثتين؟
  \item فسّر لماذا كان الصنفان المؤتلفان متساويين تقريبًا، ولماذا كان
        الصنفان الأبويان كذلك.
  \item المورّثتان نفسهما، في أنثى والداها من سلالة نقية رمادية ضامرة
        وسلالة نقية سوداء عادية: تنبأ بالأصناف الأربعة وتواتراتها.
\end{enumerate}

\medskip
\noindent\textbf{القسم الثالث --- \omterm{def:g10:universal-dna:gene}{مورّثة} ثالثة.}
\omterm{def:g10:universal-dna:gene}{مورّثة} ثالثة، هي $cn$ (العينان الزنجفريتان)، هجّنت تهجينًا اختباريًا مع
$b$ في أنثى أخرى: المؤتلفون 9\%. ومع $vg$: المؤتلفون 8\%.
\begin{enumerate}[resume]
  \item هل $cn$ على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} نفسه الذي عليه $b$ و$vg$؟
  \item مستعينًا بتواترات الائتلاف الثلاثة (17\% للزوج $b$--$vg$،
        و9\% للزوج $b$--$cn$، و8\% للزوج $cn$--$vg$)، رتّب \omterm{def:g10:universal-dna:gene}{المورّثات}
        الثلاث على \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي}.
  \item فسّر لماذا تتجامع التواترات (تقريبًا)، وماذا يقول ذلك عن العلاقة
        بين الائتلاف والمسافة.
  \item مورّثتان على طرفي \omterm{def:g10:universal-dna:information}{صبغي} طويل تعطيان 50\% مؤتلفين. فسّر لماذا لا
        يستطيع التواتر أن يتجاوز 50\%.
  \item \omterm{def:g10:universal-dna:gene}{مورّثة} رابعة تعطي 50\% مؤتلفين مع الثلاث جميعًا. ماذا تستنتج،
        وماذا لا تستطيع أن تستنتج؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الرابع --- حجم مجموعة الورق.}
لذبابة الفاكهة $2n = 8$.
\begin{enumerate}[resume]
  \item كم توفيقًا صبغيًا تستطيع أمشاج ذبابة أن تبديه بالتوزع المستقل
        وحده؟ وكم يستطيع نسل زوجين من الذباب؟
  \item بوقوع تصالب واحد في موضع متغير على كل زوج، فسّر لماذا يصير عدد
        الأمشاج المتمايزة غير محدود عمليًا.
  \item لزوجين من البشر، احسب عدد التوافيق الصبغية لأطفالهما بالتوزع
        وحده، وقارنه بعدد البشر الذين عاشوا قط (نحو $10^{11}$).
  \item فسّر لماذا كان أخوان أشبه ببعضهما مع ذلك من غريبين، من حيث
        \omterm{def:g10:universal-dna:gene}{الأليلات} التي أمكن أن يتلقياها.
  \item اذكر النتيجة: تواتر الائتلاف بين $b$ و$vg$، وما الذي يقيسه،
        وآليتا \omterm{def:g12:meiosis-genetic-shuffling:meiosis}{الانقسام المنصّف} اللتان تجعلان كل مشيج من أمشاج الأنثى
        مختلفًا.
\end{enumerate}
\end{problem}
