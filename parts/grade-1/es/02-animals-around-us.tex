\chapter{Los animales que nos rodean}\label{ch:g1:animals-around-us}

Un mirlo da saltitos por el césped, un gato lo vigila desde el muro,
una hormiga arrastra por el umbral una miga más grande que ella.
Los \omterm{def:g1:animals-around-us:animal}{animales} están por todas partes a nuestro alrededor --- en el
jardín, en el parque, en casa. Aprendamos a mirarlos como los mira un
biólogo: su cuerpo y su manera de moverse.

\section{¿Qué es un animal?}

\begin{definition}[Animal]\label{def:g1:animals-around-us:animal}
Un \emph{animal}\index{animal} es un \omterm{def:g1:living-or-not:living}{ser vivo} que se alimenta
comiendo otros \omterm{def:g1:living-or-not:living}{seres vivos} --- plantas u otros animales --- y que
normalmente puede desplazarse por sí mismo. Un gato, un mirlo, un pez
y una hormiga son animales. Nosotros también somos animales.
\end{definition}

\begin{example}[Los animales de un jardín]\label{ex:g1:animals-around-us:garden}
Un jardín pequeño puede alojar muchos \omterm{def:g1:animals-around-us:animal}{animales}: un mirlo que saca una
lombriz de la tierra, la propia lombriz, un caracol sobre una hoja,
una araña en su tela, una abeja en una flor y el gato del vecino.
Grandes o diminutos, todos se alimentan y todos se mueven.
\end{example}

\begin{remark}[Pequeño no quiere decir poco importante]\label{rem:g1:animals-around-us:small}
Una hormiga es tan \omterm{def:g1:animals-around-us:animal}{animal} como un elefante. El tamaño del cuerpo no
decide quién es \omterm{def:g1:animals-around-us:animal}{animal}: lo deciden comer y moverse. Algunos de los
\omterm{def:g1:animals-around-us:animal}{animales} más asombrosos que nos rodean son los más diminutos.
\end{remark}

\section{Las partes del cuerpo de un animal}

\begin{definition}[Partes del cuerpo]\label{def:g1:animals-around-us:parts}
Casi todos los \omterm{def:g1:animals-around-us:animal}{animales} tienen una \emph{cabeza}\index{cabeza} (con
la boca y casi todos los órganos de los sentidos), un cuerpo y
\emph{extremidades}\index{extremidad} --- patas, alas o aletas ---
para moverse. Muchos tienen además una \emph{cola}\index{cola}.
\end{definition}

\begin{example}[Cuenta las patas]\label{ex:g1:animals-around-us:legs}
Las patas vienen en números que vale la pena contar. El mirlo anda
sobre $2$ patas, el gato sobre $4$, la mariquita sobre $6$, la araña
sobre $8$ --- y la lombriz y el caracol no tienen ninguna, y aun así
llegan adonde van.
\end{example}

\begin{definition}[Cubierta del cuerpo]\label{def:g1:animals-around-us:covering}
El cuerpo de un \omterm{def:g1:animals-around-us:animal}{animal} está protegido por una cubierta:
\emph{pelo}\index{pelo} en el gato, \emph{plumas}\index{pluma}
en el mirlo, \emph{escamas}\index{escama} en el pez, una
\emph{concha}\index{concha} en el caracol y una \emph{piel}\index{piel}
lisa y húmeda en la rana.
\end{definition}

\begin{example}[Quién lleva qué]\label{ex:g1:animals-around-us:covering}
Acaricia un gato: \omterm{def:g1:animals-around-us:covering}{pelo}. Recoge una \omterm{def:g1:animals-around-us:covering}{pluma} caída de un mirlo: ligera y
suave. Mira de cerca un pez en el mercado: filas de escamas brillantes,
como tejas diminutas. Tres \omterm{def:g1:animals-around-us:animal}{animales}, tres abrigos distintos.
\end{example}

\section{Cómo se mueven los animales}

\begin{proposition}[Maneras de moverse]\label{prop:g1:animals-around-us:moving}
Los \omterm{def:g1:animals-around-us:animal}{animales} se mueven de una manera que va con sus \omterm{def:g1:animals-around-us:parts}{extremidades}:
\begin{itemize}
  \item los \omterm{def:g1:animals-around-us:animal}{animales} con patas \emph{andan}, corren, saltan o reptan
        --- el gato, el mirlo en el césped, la hormiga;
  \item los \omterm{def:g1:animals-around-us:animal}{animales} con alas \emph{vuelan}\index{volar} --- el mirlo,
        la abeja, la mariposa;
  \item los \omterm{def:g1:animals-around-us:animal}{animales} con aletas \emph{nadan}\index{nadar} --- el pez;
  \item los \omterm{def:g1:animals-around-us:animal}{animales} sin \omterm{def:g1:animals-around-us:parts}{extremidades} se deslizan o se retuercen ---
        el caracol sobre su único pie blando, la lombriz que se estira
        y se encoge.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Un animal, varias maneras]\label{rem:g1:animals-around-us:several}
Muchos \omterm{def:g1:animals-around-us:animal}{animales} tienen más de una manera de moverse. El mirlo anda
\emph{y} vuela; la rana nada \emph{y} salta; el pato anda, nada
\emph{y} vuela. Observa un \omterm{def:g1:animals-around-us:animal}{animal} un buen rato antes de decidir cómo
se mueve.
\end{remark}

\begin{method}[Observar un animal]\label{met:g1:animals-around-us:observe}
Para describir un \omterm{def:g1:animals-around-us:animal}{animal} como un biólogo:
\begin{enumerate}
  \item mira las partes: cabeza, cuerpo, \omterm{def:g1:animals-around-us:parts}{extremidades}, cola;
  \item cuenta las patas, sin adivinar;
  \item nombra la cubierta: \omterm{def:g1:animals-around-us:covering}{pelo}, \omterm{def:g1:animals-around-us:covering}{plumas}, escamas, \omterm{def:g1:animals-around-us:covering}{concha}, \omterm{def:g1:animals-around-us:covering}{piel};
  \item observa cómo se mueve: andar, saltar, volar, nadar, deslizarse;
  \item di dónde lo has encontrado: césped, árbol, charca, bajo una piedra.
\end{enumerate}
No hagas nunca daño al \omterm{def:g1:animals-around-us:animal}{animal}: obsérvalo y déjalo seguir su camino.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:1}
Nombra las cuatro partes principales del cuerpo del gato de la
figura. ¿Qué parte lleva los ojos y la boca?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:2}
¿Cuántas patas: (a) una gallina? (b) un perro? (c) una mariquita?
(d) una araña? (e) una lombriz?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:3}
Di la cubierta de cada \omterm{def:g1:animals-around-us:animal}{animal}: una oveja, un pato, una trucha, un
caracol.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:4}
¿Qué \omterm{def:g1:animals-around-us:animal}{animales} de este capítulo pueden volar? ¿Qué tienen todos ellos
en lugar de patas delanteras o brazos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:5}
El caracol y la lombriz no tienen patas. ¿Cómo se mueve cada uno de
ellos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:6}
Un pato se puede mover de tres maneras. Nómbralas y di qué parte de
su cuerpo usa para cada una.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:7}
¿Por qué es una hormiga un \omterm{def:g1:animals-around-us:animal}{animal}, según la
\cref{def:g1:animals-around-us:animal}? Responde con las dos ideas de
la definición.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:animals-around-us:8}
Pruébalo: observa un rato a un \omterm{def:g1:animals-around-us:animal}{animal} en la calle --- un pájaro, un
insecto, un caracol. Descríbelo con los cinco pasos del
\cref{met:g1:animals-around-us:observe}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:9}
Un amigo dice: ``¡Los peces no pueden ser \omterm{def:g1:animals-around-us:animal}{animales}, si no andan!''
¿Qué le responderías? Usa la \cref{prop:g1:animals-around-us:moving}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:animals-around-us:10}
\omterm{def:g1:animals-around-us:animal}{Animal} misterioso: tiene $2$ patas, $2$ alas y \omterm{def:g1:animals-around-us:covering}{plumas}, y no puede
volar, pero corre muy deprisa. ¿Es un pájaro? ¿Podría ser el mejor
corredor de tu barrio? (Piensa en un avestruz.)
\end{exercise}
