\chapter{Tulang dan Otot}\label{ch:g3:bones-and-muscles}

Melompatlah, lalu mendaratlah: sesuatu di dalam dirimu baru saja
bekerja seperti mesin yang sempurna. Tiang-tiang kaku menahan
bentukmu, engsel-engsel membuat tungkaimu dapat melipat lalu
melenting, dan ratusan motor kecil menarik tepat ketika dibutuhkan.
\omterm{def:g1:plants-around-us:parts}{Batangnya} adalah tulangmu, engselnya adalah sendimu, motornya adalah
ototmu --- perlengkapan tubuh bagi setiap gerakan yang akan pernah
dibuatnya.

\section{Rangka}

\begin{definition}[Rangka dan tulang]\label{def:g3:bones-and-muscles:skeleton}
Yang disebut \emph{rangka}\index{rangka} adalah kerangka kira-kira
$200$ \emph{tulang}\index{tulang} di dalam tubuh. Bagian-bagian
utamanya:
\begin{itemize}
  \item \emph{tengkorak}\index{tengkorak}, sebuah helm tulang yang
        kokoh mengelilingi otak;
  \item \emph{tulang belakang}\index{tulang belakang}, sebuah tiang
        yang dapat melengkung, tersusun dari tulang-tulang kecil yang
        ditumpuk dari leher sampai pinggul, menahan batang tubuh tetap
        tegak;
  \item \emph{tulang rusuk}\index{tulang rusuk}, tulang melengkung
        yang membentuk sangkar mengelilingi jantung dan paru-paru;
  \item \emph{tulang \omterm{def:g1:my-body:parts}{anggota gerak}}: tulang panjang pada \omterm{def:g1:my-body:parts}{lengan} dan
        \omterm{def:g1:my-body:parts}{tungkai} --- tulang paha adalah yang terpanjang di tubuh.
\end{itemize}
\end{definition}

\begin{example}[Rasakan kerangkamu]\label{ex:g3:bones-and-muscles:feel}
Tidak perlu sinar-X: tekan pelan-pelan dan \omterm{def:g3:bones-and-muscles:skeleton}{rangkanya} melapor. Buku
jari; punggungan \omterm{def:g3:bones-and-muscles:skeleton}{tulang} kering; rantai benjolan kecil di tengah
punggungmu --- \omterm{def:g3:bones-and-muscles:skeleton}{tulang belakang}, satu \omterm{def:g3:bones-and-muscles:skeleton}{tulang} kecil sesudah yang lain;
\omterm{def:g3:bones-and-muscles:skeleton}{tulang rusuk} yang dapat kamu hitung di bawah jarimu. Kerangkanya tepat
di bawah \omterm{def:g1:the-five-senses:sense}{kulit}.
\end{example}

\begin{remark}[Keras di luar, hidup di dalam]\label{rem:g3:bones-and-muscles:alive}
Sebuah \omterm{def:g3:bones-and-muscles:skeleton}{tulang} bukan bahan mati seperti sebatang kapur: ia hidup,
diberi makan oleh darah, tumbuh sementara kamu tumbuh --- dan mampu
memperbaiki dirinya sendiri. \omterm{def:g1:my-body:parts}{Lengan} patah yang digips menyatu kembali
karena \omterm{def:g3:bones-and-muscles:skeleton}{tulangnya} sendiri perlahan membangun ulang patahannya. Batang
kapur tidak pernah melakukan itu.
\end{remark}

\section{Sendi, tempat kerangkanya menekuk}

\begin{proposition}[Tiang kaku, tempat menekuk]\label{prop:g3:bones-and-muscles:joints}
\omterm{def:g3:bones-and-muscles:skeleton}{Tulang} itu sendiri tidak pernah menekuk --- \omterm{def:g3:bones-and-muscles:skeleton}{tulang} yang tertekuk
adalah \omterm{def:g3:bones-and-muscles:skeleton}{tulang} yang patah. Semua tekukan tubuh terjadi pada
\emph{sendi}\index{sendi}, tempat dua \omterm{def:g3:bones-and-muscles:skeleton}{tulang} bertemu: \omterm{def:g1:my-body:joint}{bahu}, \omterm{def:g1:my-body:joint}{siku},
\omterm{def:g1:my-body:joint}{pergelangan tangan}, pinggul, \omterm{def:g1:my-body:joint}{lutut}, \omterm{def:g1:my-body:joint}{pergelangan kaki}, dan banyak \omterm{def:g1:my-body:joint}{sendi}
kecil \omterm{def:g3:bones-and-muscles:skeleton}{tulang belakang}, yang tekukan kecilnya menjumlah menjadi
punggung yang lentur.
\end{proposition}
\admitted

\begin{example}[Kiat tulang belakang]\label{ex:g3:bones-and-muscles:spinetrick}
Tidak ada satu pun \omterm{def:g3:bones-and-muscles:skeleton}{tulang} di punggungmu yang dapat melengkung ---
namun kamu dapat menggulung badan ke depan hampir menjadi bola. \omterm{def:g3:bones-and-muscles:skeleton}{Tulang
belakang} adalah tiang yang terdiri atas lebih dari $20$ \omterm{def:g3:bones-and-muscles:skeleton}{tulang} kecil,
masing-masing berputar beberapa derajat terhadap tetangganya: banyak
tekukan kecil yang membuat satu tekukan besar. Itu pula sebabnya kamu
harus menekuk \omterm{def:g1:my-body:joint}{lutut}, bukan punggung, untuk mengangkat sesuatu yang
berat --- \omterm{def:g1:my-body:joint}{lutut} dibangun untuk lipatan besar, \omterm{def:g3:bones-and-muscles:skeleton}{tulang belakang} untuk
lipatan yang lembut.
\end{example}

\section{Otot, si penggerak}

\begin{definition}[Otot]\label{def:g3:bones-and-muscles:muscle}
Suatu \emph{otot}\index{otot} adalah motor berdaging yang melekat pada
\omterm{def:g3:bones-and-muscles:skeleton}{tulang}. Sebuah otot dapat \emph{berkontraksi}\index{berkontraksi} ---
memendek dan menebal --- dan ketika berkontraksi ia \emph{menarik}
\omterm{def:g3:bones-and-muscles:skeleton}{tulang} tempatnya melekat. Otot hanya menarik; tidak ada otot yang
dapat mendorong.
\end{definition}

\begin{proposition}[Bekerja berpasangan]\label{prop:g3:bones-and-muscles:pairs}
Karena \omterm{def:g3:bones-and-muscles:muscle}{otot} hanya menarik, menggerakkan sebuah \omterm{def:g1:my-body:joint}{sendi} ke dua arah
membutuhkan sebuah \emph{pasangan}: satu \omterm{def:g3:bones-and-muscles:muscle}{otot} di depan \omterm{def:g1:my-body:parts}{lengan}
menekukkan \omterm{def:g1:my-body:joint}{siku}, pasangannya di belakang meluruskannya kembali;
masing-masing mengendur ketika yang lain menarik. Hampir setiap \omterm{def:g1:my-body:joint}{sendi}
tubuh digerakkan oleh pasangan semacam itu.
\end{proposition}
\admitted

\begin{example}[Rasakan pasangannya bekerja]\label{ex:g3:bones-and-muscles:pair}
Genggam \omterm{def:g1:my-body:parts}{lengan} atasmu dengan tangan yang lain lalu tekuk sikumu
pelan-pelan: \omterm{def:g3:bones-and-muscles:muscle}{otot} di depan membengkak dan mengeras --- ia sedang
\omterm{def:g3:bones-and-muscles:muscle}{berkontraksi}. Sekarang luruskan lenganmu: bagian depannya melunak, dan
di belakang \omterm{def:g1:my-body:parts}{lengan} pasangannya justru mengencang. Kamu baru saja
merasakan \cref{prop:g3:bones-and-muscles:pairs} terjadi.
\end{example}

\begin{method}[Merawat tulang dan otot]\label{met:g3:bones-and-muscles:care}
\begin{enumerate}
  \item Bergeraklah setiap hari --- \omterm{def:g3:bones-and-muscles:skeleton}{tulang} menjadi lebih kuat, begitu
        pula \omterm{def:g3:bones-and-muscles:muscle}{otot}, dengan dipakai;
  \item makanlah makanan para pembangun: produk \omterm{def:g2:animals-grow-up:born}{susu} untuk \omterm{def:g3:bones-and-muscles:skeleton}{tulang}, dan
        santapan yang beragam (\cref{ch:g2:teeth-and-eating-well});
  \item lakukan pemanasan sebelum olahraga yang berat, supaya \omterm{def:g3:bones-and-muscles:muscle}{otot}
        meregang dan tidak robek;
  \item angkat benda berat dengan \omterm{def:g1:my-body:joint}{lutut} tertekuk dan punggung lurus;
  \item pakai helm dan pelindung di tempat yang mudah membuat jatuh:
        \omterm{def:g3:bones-and-muscles:skeleton}{tulang} memang memperbaiki diri, tetapi pelan.
\end{enumerate}
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:1}
Sebutkan keempat bagian \omterm{def:g3:bones-and-muscles:skeleton}{rangka} pada
\cref{def:g3:bones-and-muscles:skeleton}, dan apa yang dilindungi atau
ditopang masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:2}
Kira-kira ada berapa banyak \omterm{def:g3:bones-and-muscles:skeleton}{tulang} pada \omterm{def:g3:bones-and-muscles:skeleton}{rangka}? Yang mana yang
terpanjang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:3}
Dapatkah sebuah \omterm{def:g3:bones-and-muscles:skeleton}{tulang} menekuk? Di mana semua tekukan tubuh terjadi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:4}
\omterm{def:g3:bones-and-muscles:skeleton}{Tulang belakang} dapat melengkung walaupun tidak satu pun \omterm{def:g3:bones-and-muscles:skeleton}{tulangnya}
dapat. Jelaskan kiatnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:5}
Apa yang dilakukan sebuah \omterm{def:g3:bones-and-muscles:muscle}{otot} ketika \omterm{def:g3:bones-and-muscles:muscle}{berkontraksi}? Apa yang tidak
pernah dapat dilakukan \omterm{def:g3:bones-and-muscles:muscle}{otot} mana pun?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:6}
Mengapa menekuk dan meluruskan \omterm{def:g1:my-body:joint}{siku} membutuhkan dua \omterm{def:g3:bones-and-muscles:muscle}{otot}? Ceritakan
apa yang dilakukan masing-masing pada tiap saatnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:7}
Bagaimana \omterm{def:g1:my-body:parts}{lengan} patah yang digips menyatu kembali? Apa yang
ditunjukkannya tentang \omterm{def:g3:bones-and-muscles:skeleton}{tulang}, menurut
\cref{rem:g3:bones-and-muscles:alive}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:8}
Cobalah: lakukan uji genggam \cref{ex:g3:bones-and-muscles:pair} pada
lenganmu sendiri. Lukiskan apa yang dilakukan bagian depan \omterm{def:g1:my-body:parts}{lengan}
ketika kamu menekuk, dan ketika kamu meluruskan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:9}
Mengapa orang dewasa berkata ``angkatlah dengan lututmu, bukan dengan
punggungmu''? Jawablah dengan
\cref{ex:g3:bones-and-muscles:spinetrick}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:10}
Sebuah boneka bertali dapat dibuat melambai: tiap tali hanya dapat
\emph{menarik} \omterm{def:g1:my-body:parts}{lengannya} ke satu arah. Bagaimana ini persis merupakan
keadaan \omterm{def:g3:bones-and-muscles:muscle}{otot}, dan apa jawaban pembuat bonekanya atas hal itu?
Bandingkan dengan \cref{prop:g3:bones-and-muscles:pairs}.
\end{exercise}
