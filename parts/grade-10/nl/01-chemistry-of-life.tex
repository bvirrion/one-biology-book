\chapter{De chemische samenstelling van levende wezens}\label{ch:g10:chemistry-of-life}

Laat een boterham te lang op de grill liggen en hij wordt zwart: wat
overblijft is vooral koolstof. Weeg een vers schijfje appel, droog het
een dag in een warme oven en weeg opnieuw: vijf zesde van de massa is
als waterdamp verdwenen. Brood, appel, eik, kwal, jij --- elk levend
wezen is in de grond opgebouwd uit dezelfde korte lijst chemische
ingrediënten, en die lijst is niet de lijst waaruit het gesteente
bestaat. Dit hoofdstuk haalt levende materie uit elkaar, ingrediënt
voor ingrediënt, en vraagt wat haar zo bijzonder maakt.

\section{Elementen: dezelfde atomen als het gesteente, in andere verhoudingen}

\begin{definition}[Chemische samenstelling van levende materie]\label{def:g10:chemistry-of-life:composition}
De \emph{chemische samenstelling}\index{chemische samenstelling} van een
monster levende materie is de lijst van de chemische elementen die het
bevat, met voor elk element het deel van de massa van het monster dat
het vertegenwoordigt. Zij wordt gemeten door het monster chemisch uit
elkaar te halen --- verbranden, wegen wat verdampt en wat achterblijft,
de vrijgekomen gassen analyseren --- niet door ernaar te kijken.
\end{definition}

\begin{proposition}[Vier elementen vormen 96\% van een levend lichaam]\label{prop:g10:chemistry-of-life:four}
Naar massa bestaat levende materie bijna geheel uit vier elementen:
zuurstof (O), koolstof (C), waterstof (H) en stikstof (N). In het
menselijk lichaam zijn zij goed voor ongeveer 65\%, 18.5\%, 9.5\% en
3.2\% van de massa --- samen 96\%. Calcium, fosfor, kalium, zwavel,
natrium, chloor en magnesium delen het grootste deel van de resterende
4\%, en een tiental \emph{sporenelementen} (ijzer, zink, jood,
koper\dots) zijn ver beneden 0.1\% aanwezig en toch onmisbaar. Dezelfde
vier elementen overheersen in een bacterie, een tarweplant of een
walvis, in vergelijkbare verhoudingen.
\end{proposition}
\admitted

\begin{example}[Levende materie tegenover gesteente]\label{ex:g10:chemistry-of-life:crust}
De aardkorst bestaat eveneens voor 46\% uit zuurstof, maar de volgende
elementen zijn silicium (28\%), aluminium (8\%) en ijzer (5.6\%).
Koolstof maakt 0.02\% van de korst uit en stikstof 0.002\%, tegenover
18.5\% en 3.2\% van een lichaam. Een levend wezen is dus geen monster
van zijn omgeving: het \emph{concentreert} enkele elementen met een
factor duizend of meer. Die selectie is het eerste chemische kenmerk
van leven.
\end{example}

\begin{remark}[Waar de atomen vandaan komen]\label{rem:g10:chemistry-of-life:origin}
Elk van die atomen is doodgewoon: de koolstof van je spieren was
koolstofdioxide uit de lucht voordat een plant haar vastlegde, de
stikstof van je \omterm{def:g10:chemistry-of-life:families}{eiwitten} was stikstof uit de lucht voordat
bodembacteriën haar invingen. Het bijzondere aan leven zijn niet de
atomen, maar de manier waarop ze worden samengevoegd en de verhoudingen
waarin ze worden verzameld.
\end{remark}

\section{Water en minerale zouten}

\begin{definition}[Minerale stof en organische stof]\label{def:g10:chemistry-of-life:mineral-organic}
De moleculen van een levend wezen vallen in twee groepen. De
\emph{minerale stof}\index{minerale stof} bestaat uit moleculen die ook
buiten levende wezens voorkomen: water en de opgeloste
\emph{minerale zouten}\index{minerale zouten} (ionen zoals
$\mathrm{Na^+}$, $\mathrm{K^+}$, $\mathrm{Ca^{2+}}$, $\mathrm{Cl^-}$,
fosfaat en bicarbonaat). De \emph{organische stof}\index{organische stof}
bestaat uit moleculen die gebouwd zijn op een skelet van koolstofatomen
gebonden aan waterstof, en meestal ook aan zuurstof en stikstof ---
moleculen die op aarde alleen door levende wezens of uit hun resten
worden gemaakt.
\end{definition}

\begin{proposition}[Water is de belangrijkste bouwstof]\label{prop:g10:chemistry-of-life:water}
Water is het meest voorkomende molecuul van elk levend wezen: ongeveer
60\% tot 65\% van de massa van een volwassen mens, 75\% van een
pasgeborene, 85\% van een appel, 95\% van een kwal, maar slechts 13\%
van een rustende tarwekorrel. Het watergehalte van een monster wordt
gemeten door het te wegen, het bij ongeveer \qty{105}{\celsius} te
drogen tot de massa niet verder daalt, en het opnieuw te wegen: het
massaverlies is het water.
\end{proposition}
\admitted

\begin{example}[Een schijfje appel drogen]\label{ex:g10:chemistry-of-life:apple}
Een schijfje verse appel met massa \qty{24.0}{g} wordt gedroogd tot de
massa zich op \qty{3.6}{g} instelt. Verloren water:
$24.0 - 3.6 = \qty{20.4}{g}$, dus $20.4/24.0 = 0.85$, oftewel 85\% van
de verse massa. De \qty{3.6}{g} die overblijven zijn de \emph{droge
stof}: organische moleculen plus \omterm{def:g10:chemistry-of-life:mineral-organic}{minerale zouten}. Verbrand die droge
stof en er blijft ongeveer \qty{0.1}{g} bleke as --- de \omterm{def:g10:chemistry-of-life:mineral-organic}{minerale
zouten} --- over; de rest, het organische deel, is als koolstofdioxide
en waterdamp verdwenen.
\end{example}

\begin{proposition}[Wat water doet]\label{prop:g10:chemistry-of-life:water-roles}
Water is de plaats waar de chemie van een cel zich afspeelt: het lost
zouten, suikers en de meeste kleine organische moleculen op, zodat die
elkaar kunnen ontmoeten en reageren; het vervoert ze van de ene plaats
naar de andere (bloedplasma bestaat voor 90\% uit water, sap voor nog
veel meer); en omdat er veel energie nodig is om het te verwarmen,
dempt het de temperatuur van het organisme tegen plotselinge
schommelingen. Een verlies van 10\% van het lichaamswater is een
medisch noodgeval; een rustend zaad, op 13\%, heeft zijn chemie
eenvoudigweg uitgeschakeld tot het water terugkeert.
\end{proposition}
\admitted

\begin{example}[Minerale zouten die je kunt noemen]\label{ex:g10:chemistry-of-life:salts}
Calciumfosfaat verstevigt bot en tanden (calcium is 1.5\% van de
lichaamsmassa, vrijwel alles in het skelet). Natrium- en chloride-ionen
bepalen het zoutgehalte van het bloed, dicht bij \qty{9}{g/L};
kaliumionen zijn geconcentreerd in de cellen; ijzer zit in het hart van
hemoglobine; jood is nodig om het hormoon van de schildklier te maken.
Een mineraal zout is nooit een energiebron, maar een lichaam dat er één
mist, faalt bij een nauwkeurig omschreven taak.
\end{example}

\section{De organische moleculen: vier families}

\begin{definition}[Koolhydraten, lipiden, eiwitten, nucleïnezuren]\label{def:g10:chemistry-of-life:families}
Organische moleculen behoren tot vier families.
\begin{itemize}
  \item De \emph{koolhydraten}\index{koolhydraat} (suikers) bevatten
        alleen koolstof, waterstof en zuurstof. De enkelvoudige suikers
        zijn losse eenheden, zoals glucose $\mathrm{C_6H_{12}O_6}$; de
        samengestelde koolhydraten zijn ketens van honderden of
        duizenden glucose-eenheden --- zetmeel en cellulose bij planten,
        glycogeen bij dieren.
  \item De \emph{lipiden}\index{lipide} (vetten en oliën) bevatten
        eveneens alleen C, H en O, maar met veel minder zuurstof, en ze
        lossen niet op in water. De gewoonste zijn de triglyceriden,
        elk gemaakt van één glycerolmolecuul verbonden met drie lange
        vetzuurketens.
  \item De \emph{eiwitten}\index{eiwit} zijn ketens van
        \emph{aminozuren}\index{aminozuur}: twintig typen aminozuur,
        alle stikstofhoudend, in een nauwkeurige volgorde geregen tot
        ketens van vijftig tot enkele duizenden eenheden die zich tot
        een bepaalde vorm vouwen.
  \item De \emph{nucleïnezuren}\index{nucleïnezuur} (DNA en RNA) zijn
        ketens van nucleotiden, die naast C, H, O en N ook fosfor
        bevatten; zij bewaren en geven de genetische informatie door en
        zijn het onderwerp van \cref{ch:g10:universal-dna}.
\end{itemize}
\end{definition}

\begin{example}[Een voedingsetiket lezen]\label{ex:g10:chemistry-of-life:label}
Op een etiket van volkorenbrood staat, per \qty{100}{g}: \omterm{def:g10:chemistry-of-life:families}{koolhydraten}

mineraal zout. Een etiket is een chemische analyse, afgerond.
\end{example}

\begin{proposition}[Waar elke familie voor dient]\label{prop:g10:chemistry-of-life:roles}
De families verdelen het werk van een cel, met overlappingen:
\begin{itemize}
  \item \omterm{def:g10:chemistry-of-life:families}{koolhydraten} zijn de dagelijkse brandstof (glucose) en de
        voorraad daarvan op korte termijn (zetmeel, glycogeen), en
        cellulose is het bouwmateriaal van plantaardige celwanden;
  \item \omterm{def:g10:chemistry-of-life:families}{lipiden} vormen de compacte energievoorraad op lange termijn
        (\qty{38}{kJ} per gram, tegen \qty{17}{kJ} per gram voor
        \omterm{def:g10:chemistry-of-life:families}{koolhydraten} en \omterm{def:g10:chemistry-of-life:families}{eiwitten}), en membraanlipiden vormen de
        omhulling van elke cel;
  \item \omterm{def:g10:chemistry-of-life:families}{eiwitten} zijn de machines: de enzymen die elke reactie
        uitvoeren, de vezels die spieren doen samentrekken, de
        transporteurs, de antistoffen, de receptoren;
  \item \omterm{def:g10:chemistry-of-life:families}{nucleïnezuren} bewaren de instructies om de \omterm{def:g10:chemistry-of-life:families}{eiwitten} te bouwen.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Grote moleculen en de vorm van het leven]\label{rem:g10:chemistry-of-life:macromolecules}
De moleculen van gesteente zijn klein en eentonig (een kwartskristal is
$\mathrm{SiO_2}$ dat zich eindeloos herhaalt). De kenmerkende moleculen
van het leven zijn reusachtig --- één \omterm{def:g10:chemistry-of-life:families}{eiwit} kan tienduizend atomen
bevatten, een DNA-molecuul miljarden --- en \emph{informatief}: de
volgorde van hun bouwstenen draagt een boodschap. Niets in de minerale
wereld lijkt op een molecuul waarvan de volgorde iets betekent.
\end{remark}

\section{De bestanddelen aantonen}

\begin{method}[Standaardproeven voor de families]\label{met:g10:chemistry-of-life:tests}
Elke familie heeft een eenvoudige proef die een zichtbaar resultaat
geeft.
\begin{enumerate}
  \item \emph{Water}: een strookje blauw kobaltchloridepapier wordt
        roze; of droog het monster en meet het massaverlies.
  \item \emph{Zetmeel}: een druppel joodoplossing verkleurt van
        geelbruin naar blauwzwart.
  \item \emph{Enkelvoudige suikers} (glucose, fructose\dots): verwarm
        met de reagens van Fehling of van Benedict; de blauwe oplossing
        geeft een baksteenrode neerslag.
  \item \emph{\omterm{def:g10:chemistry-of-life:families}{Eiwitten}}: de biureetproef --- enkele druppels
        kopersulfaat in alkalische oplossing worden violet.
  \item \emph{\omterm{def:g10:chemistry-of-life:families}{Lipiden}}: het monster laat een doorschijnende vlek op
        papier achter die niet opdroogt; de rode kleurstof Soedan III
        kleurt haar.
  \item \emph{\omterm{def:g10:chemistry-of-life:mineral-organic}{Minerale zouten}}: verbrand de droge stof volledig; de as
        die overblijft is mineraal. Afzonderlijke ionen worden
        aangetoond door neerslag (zilvernitraat voor chloride,
        bijvoorbeeld).
\end{enumerate}
Voer altijd een controle uit met zuiver water, en één met een monster
waarvan de uitslag positief hoort te zijn, om zeker te weten dat het
reagens werkt en dat een kleurverandering betekent wat je denkt.
\end{method}

\begin{example}[Een aardappel onderzoeken]\label{ex:g10:chemistry-of-life:potato}
Een schijfje rauwe aardappel: kobaltchloridepapier wordt roze (water);
jood kleurt onmiddellijk blauwzwart (zetmeel, in overvloed); Fehling
geeft slechts een vage oranje zweem (weinig vrije suiker); biureet geeft
een bleek violet (wat \omterm{def:g10:chemistry-of-life:families}{eiwit}, ongeveer 2\% van de verse massa); geen
doorschijnende vlek (vrijwel geen \omterm{def:g10:chemistry-of-life:families}{lipide}). Drogen laat 22\% van de massa
over; die verbranden laat 1\% as over. De aardappel is een
zetmeelvoorraad verpakt in water --- precies wat een plant nodig heeft
om de winter ondergronds te overleven.
\end{example}

\begin{proposition}[Eenheid van samenstelling]\label{prop:g10:chemistry-of-life:unity}
Wat het organisme ook is --- bacterie, paddenstoel, plant, dier --- de
proeven vinden telkens dezelfde vier families, gemaakt van dezelfde
bouwstenen: dezelfde twintig \omterm{def:g10:chemistry-of-life:families}{aminozuren}, dezelfde glucose, dezelfde vier
nucleotiden. Een levend wezen onderscheidt zich niet doordat het
bijzondere atomen of bijzondere bouwstenen bezit, maar door de
verhoudingen die het in stand houdt, de zeer grote moleculen die het
samenstelt en de informatie die die moleculen dragen. Deze chemische
eenheid is een van de sterkste argumenten dat alle levende wezens een
gemeenschappelijke oorsprong delen, een argument dat
\cref{ch:g10:common-ancestry} zal opnemen.
\end{proposition}
\admitted

\end{exercise}

\end{exercise}

(b) \omterm{def:g10:chemistry-of-life:families}{eiwit}, (c) \omterm{def:g10:chemistry-of-life:families}{lipiden} bevat? Geef telkens het verwachte resultaat.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:chemistry-of-life:5}

\begin{exercise}[$\star\star$]\label{exo:g10:chemistry-of-life:6}

\begin{exercise}[$\star\star$]\label{exo:g10:chemistry-of-life:7}

\section{Probleem: De koolstof in een mens}

\begin{problem}[{Weekendprobleem --- van een tarweveld tot een lichaam:
een boekhouding van materie, van het water in een korrel tot het aantal
kilogram koolstof dat een mens draagt}]
\label{pb:g10:chemistry-of-life:1}
Een klas volgt materie van een tarweveld tot het lichaam van een van de
leerlingen. Gegevens: een tarwekorrel bestaat voor 13\% uit water, en
zijn droge stof voor ongeveer 70\% uit zetmeel, 12\% \omterm{def:g10:chemistry-of-life:families}{eiwit}, 2\% \omterm{def:g10:chemistry-of-life:families}{lipide},
2\% mineraal, de rest cellulose. Broodmeel behoudt ongeveer de
samenstelling van het binnenste van de korrel. De leerlinge, Lea, heeft
een massa van \qty{60}{kg}; gebruik de samenstelling van een volwassene
uit de tweede figuur (water 60\%, \omterm{def:g10:chemistry-of-life:families}{eiwitten} 17\%, \omterm{def:g10:chemistry-of-life:families}{lipiden} 16\%,
mineralen 6\%, \omterm{def:g10:chemistry-of-life:families}{koolhydraten} 1\%). Massafracties van de elementen binnen
elke familie: neem \omterm{def:g10:chemistry-of-life:families}{koolhydraten} als 40\% koolstof, \omterm{def:g10:chemistry-of-life:families}{lipiden} als 77\%
koolstof, \omterm{def:g10:chemistry-of-life:families}{eiwitten} als 53\% koolstof en 16\% stikstof.

\medskip
\noindent\textbf{Deel I --- De korrel.}
\begin{enumerate}
  \item Een korrel met massa \qty{45}{mg} wordt gedroogd. Hoeveel water
        verliest hij, en wat is zijn droge massa?
  \item Bereken de massa's zetmeel, \omterm{def:g10:chemistry-of-life:families}{eiwit} en \omterm{def:g10:chemistry-of-life:families}{lipide} in de droge stof van
        één korrel.
  \item Welke proeven uit \cref{met:g10:chemistry-of-life:tests} zouden
        elk van deze drie bestanddelen bevestigen op een geplette
        korrel?
  \item De korrel wordt gezaaid en begoten; drie dagen later bestaat de
        kiemplant voor 88\% uit water. Verklaar waarom de kieming bij
        13\% niet kan beginnen.
  \item Eén hectare levert \qty{8}{t} korrel op. Hoeveel water wordt
        daarmee geoogst, en hoeveel droge stof?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het brood.}
\begin{enumerate}[resume]
  \item Een brood wordt gebakken van \qty{500}{g} meel, \qty{320}{g}
        water en \qty{10}{g} zout; in de oven verliest het \qty{130}{g}
        water. Wat is de massa van het gebakken brood, en het
        watergehalte in procenten?
  \item Schat, met de droge samenstelling van de korrel voor het meel,
        de massa's zetmeel en \omterm{def:g10:chemistry-of-life:families}{eiwit} in het brood.
  \item Schat, met \qty{17}{kJ/g} voor \omterm{def:g10:chemistry-of-life:families}{koolhydraten} en \omterm{def:g10:chemistry-of-life:families}{eiwitten} en
        \qty{38}{kJ/g} voor \omterm{def:g10:chemistry-of-life:families}{lipiden}, de energie-inhoud van het brood.
  \item Lea eet \qty{120}{g} van dit brood. Welk deel van een dagelijkse
        behoefte van ongeveer \qty{10000}{kJ} dekt dat?
  \item Waarom is het zout in het brood geen energiebron, en waarvoor
        dient het in het lichaam?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het lichaam: water en families.}
\begin{enumerate}[resume]
  \item Bereken de massa water in het lichaam van Lea.
  \item Bereken de massa's \omterm{def:g10:chemistry-of-life:families}{eiwit}, \omterm{def:g10:chemistry-of-life:families}{lipide}, mineraal en \omterm{def:g10:chemistry-of-life:families}{koolhydraat}.
  \item Lea verliest tijdens een lange loop \qty{1.2}{kg} door zweten.
        Welk deel van haar lichaamswater is dat? Waarom moet het snel
        worden aangevuld?
  \item Haar bloed, ongeveer \qty{4.5}{L}, bestaat voor 55\% uit plasma,
        en plasma voor 90\% uit water. Schat de massa water in haar
        bloed (neem \qty{1}{kg} per liter). Welk deel van haar
        lichaamswater is dat? Waar zit de rest?
  \item Een pasgeborene bestaat voor 75\% uit water. Verklaar in één zin
        waarom een zuigeling kwetsbaarder is voor uitdroging dan een
        volwassene.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het lichaam: elementen.}
\begin{enumerate}[resume]
  \item Bereken met de koolstoffracties van elke familie de massa
        koolstof in de \omterm{def:g10:chemistry-of-life:families}{koolhydraten}, \omterm{def:g10:chemistry-of-life:families}{lipiden} en \omterm{def:g10:chemistry-of-life:families}{eiwitten} van Lea, en
        vervolgens de totale massa koolstof.
  \item Welk deel van haar lichaamsmassa is koolstof? Vergelijk met de
        18.5\% van \cref{prop:g10:chemistry-of-life:four} en
        becommentarieer de overeenkomst.
  \item Bereken de massa stikstof in haar \omterm{def:g10:chemistry-of-life:families}{eiwitten}. Waar kwam die
        stikstof uiteindelijk vandaan?
  \item Water bestaat voor 89\% van de massa uit zuurstof. Bereken de
        massa zuurstof die alleen al haar lichaamswater draagt, en
        vergelijk met de 65\% van
        \cref{prop:g10:chemistry-of-life:four}. Welk molecuul verklaart
        het grootste deel van de zuurstof in het lichaam?
  \item Geef het resultaat in één zin: hoeveel kilogram koolstof draagt
        een mens van \qty{60}{kg}, en via welke twee biologische
        processen heeft die koolstof de lucht verlaten en het lichaam
        bereikt?
\end{enumerate}
\end{problem}
