\section*{Chapter \ref{ch:g11:dna-replication} --- DNA Structure and Replication}

\begin{solution}{exo:g11:dna-replication:1}
A \omterm{def:g11:dna-replication:helix}{DNA} \omterm{def:g11:dna-replication:nucleotide}{nucleotide} has a \omterm{def:g11:dna-replication:nucleotide}{deoxyribose} sugar, a phosphate group, and a nitrogenous
base. The four bases are adenine (A), thymine (T), cytosine (C) and guanine (G).
\end{solution}

\begin{solution}{exo:g11:dna-replication:2}
Complement of \texttt{5'-GATC-3'} is \texttt{5'-GATC-3'} (the \omterm{def:g10:molecules-of-life:proteins}{sequence} is its
own reverse complement). Pairing: G$\leftrightarrow$C, A$\leftrightarrow$T,
T$\leftrightarrow$A, C$\leftrightarrow$G, then reverse to write $5'\to 3'$.
\end{solution}

\begin{solution}{exo:g11:dna-replication:3}
Each daughter duplex retains one parental strand and receives one newly
synthesized strand.
\end{solution}

\begin{solution}{exo:g11:dna-replication:4}
One strand runs $5'\to 3'$ in a given direction along the duplex while its
partner runs $5'\to 3'$ the opposite way: the free $5'$ end of one strand lies
beside the free $3'$ end of the other. That opposite orientation is
\omterm{def:g11:dna-replication:antiparallel}{antiparallel}.
\end{solution}

\begin{solution}{exo:g11:dna-replication:5}
\omterm{def:g11:dna-replication:enzymes}{Helicase}: unwinds and separates parental strands. \omterm{def:g11:dna-replication:enzymes}{Primase}: synthesizes short
\omterm{def:g10:molecules-of-life:nucleic}{RNA} \omterm{def:g11:dna-replication:enzymes}{primers}. \omterm{def:g11:dna-replication:enzymes}{DNA polymerase}: extends \omterm{def:g11:dna-replication:enzymes}{primers} with \omterm{def:g11:dna-replication:helix}{DNA}, $5'\to 3'$, using the
template. \omterm{def:g11:dna-replication:enzymes}{DNA ligase}: seals nicks between \omterm{prop:g11:dna-replication:leading-lagging}{Okazaki fragments} (phosphodiester
bond).
\end{solution}

\begin{solution}{exo:g11:dna-replication:6}
Polymerase can only add \omterm{def:g10:molecules-of-life:nucleic}{nucleotides} $5'\to 3'$. At a fork moving one way, only
one new strand can grow continuously toward the fork (leading). The other
template is oriented so that synthesis must proceed away from the fork in
short stretches (\omterm{prop:g11:dna-replication:leading-lagging}{Okazaki fragments}) as more template is exposed.
\end{solution}

\begin{solution}{exo:g11:dna-replication:7}
New strand: \texttt{5'-TTGCAA-3'}. Base pairs: T--A, T--A, G--C, C--G, A--T,
A--T $\to$ \omterm{def:g10:molecules-of-life:water}{hydrogen bonds} $2+2+3+3+2+2 = 14$.
\end{solution}

\begin{solution}{exo:g11:dna-replication:8}
A single origin would force one (or two) forks to travel the entire \omterm{def:g11:dna-replication:chromosome}{chromosome}
length; eukaryotic \omterm{def:g11:dna-replication:helix}{DNA} is so long that that would take far longer than the
available S phase. Many origins let many forks work in parallel.
\end{solution}

\begin{solution}{exo:g11:dna-replication:9}
Fully conservative \omterm{def:g11:dna-replication:semi}{replication} after one generation would give two bands: fully
heavy parental duplexes and fully light new duplexes. Intermediate density only
means every duplex is hybrid (one heavy strand + one light strand), which is
\omterm{def:g11:dna-replication:semi}{semi-conservative}, not conservative.
\end{solution}

\begin{solution}{exo:g11:dna-replication:10}
\omterm{def:g10:molecules-of-life:nucleic}{RNA} \omterm{def:g11:dna-replication:enzymes}{primers} are removed (nucleases / polymerase replacing \omterm{def:g10:molecules-of-life:nucleic}{RNA} with \omterm{def:g11:dna-replication:helix}{DNA}); \omterm{def:g11:dna-replication:enzymes}{DNA
polymerase} fills the gaps with \omterm{def:g11:dna-replication:helix}{DNA}; \omterm{def:g11:dna-replication:enzymes}{DNA ligase} seals the remaining nicks so the
\omterm{prop:g11:dna-replication:leading-lagging}{lagging strand} becomes one continuous \omterm{def:g11:dna-replication:helix}{DNA} chain.
\end{solution}

\begin{solution}{pb:g11:dna-replication:1}
\textbf{1.} Circle with origin O; two forks move opposite ways along the circle
and meet at the terminus roughly opposite O.

\textbf{2.} Two forks copy different halves at once, roughly halving the time
compared with one fork travelling the whole circumference.

\textbf{3.} At each fork, the new strand growing continuously toward the fork
is leading; the discontinuous strand is lagging (depends on which parental
strand is drawn $3'\to 5'$ toward the fork).

\textbf{4.} Time $\approx \num{4.6e6}/1000 = \num{4600}\,\mathrm{s}
\approx 77$ minutes (\omterm{def:g10:biodiversity-classification:hierarchy}{order} of one hour).

\textbf{5.} With two forks sharing the work, roughly half that time, \omterm{def:g10:biodiversity-classification:hierarchy}{order} of
$40$ minutes (actual \emph{E.\ coli} doubling can be faster still with
overlapping rounds).

\textbf{6.} \omterm{def:g11:dna-replication:enzymes}{Helicase} (unwind), \omterm{def:g11:dna-replication:enzymes}{primase} (\omterm{def:g10:molecules-of-life:nucleic}{RNA} \omterm{def:g11:dna-replication:enzymes}{primers}), \omterm{def:g11:dna-replication:enzymes}{DNA polymerase} (elongation
and much proofreading); ligase also needed to join \omterm{prop:g11:dna-replication:leading-lagging}{Okazaki fragments}.

\textbf{7.} Both daughter circles: each is \omterm{def:g11:dna-replication:semi}{semi-conservative}, so each contains
exactly one parental strand.
\end{solution}
