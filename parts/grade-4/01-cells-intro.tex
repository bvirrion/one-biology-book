\chapter{A First Look at Cells}\label{ch:g4:cells}

Your \omterm{def:g1:my-body:body}{body}, a \omterm{def:g2:plant-parts:main}{leaf}, and a beetle look very different. Yet each is built
from tiny living units called \emph{\omterm{def:g4:cells:cell}{cells}}. In this chapter you will
learn what a \omterm{def:g4:cells:cell}{cell} is, why we need \omterm{prop:g4:cells:building}{microscopes} to see most of them, and
how \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g4:cells:cell}{cells} and \omterm{def:g1:plants-animals:animal}{animal} \omterm{def:g4:cells:cell}{cells} are alike and different.

\section{What is a cell?}

\omimg{plant-cell.jpg}{0.48\linewidth}{A \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g4:cells:cell}{cell} has a wall and green \omterm{def:g4:cells:plant}{chloroplasts} --- look for the green parts.}

\begin{definition}[Cell]\label{def:g4:cells:cell}
A \emph{cell}\index{cell} is the smallest unit of a \omterm{def:g1:living-nonliving:living}{living thing} that
can carry out the jobs of life: taking in materials, using \omterm{def:g2:food-energy:energy}{energy}, and
(usually) making more cells. \omterm{def:g1:living-nonliving:living}{Living things} may be one cell or many
cells working together.
\end{definition}

\begin{example}[Everywhere you look]\label{ex:g4:cells:examples}
\omterm{def:g1:senses:five}{Skin} is a sheet of many \omterm{def:g4:cells:cell}{cells}. A tree \omterm{def:g1:my-body:parts}{trunk} is packed with \omterm{def:g4:cells:cell}{cells}. Some
\omterm{def:g1:living-nonliving:living}{living things}, such as certain \omterm{def:g2:habitats:kinds}{pond} microbes, are only a single \omterm{def:g4:cells:cell}{cell}
and still do everything they need to stay alive.
\end{example}

\begin{proposition}[Building blocks of life]\label{prop:g4:cells:building}
All \omterm{def:g1:plants-animals:plant}{plants} and \omterm{def:g1:plants-animals:animal}{animals} are made of \omterm{def:g4:cells:cell}{cells}. Larger \omterm{def:g1:living-nonliving:living}{living things} have
more \omterm{def:g4:cells:cell}{cells}, not ``bigger life material'' with no structure.
\end{proposition}
\admitted

\section{Seeing the invisible}

Most \omterm{def:g4:cells:cell}{cells} are far too small to see with the naked \omterm{def:g1:senses:five}{eye}. Scientists use
a \emph{microscope}\index{microscope} to magnify them.

\begin{definition}[Microscope]\label{def:g4:cells:microscope}
A \emph{microscope}\index{microscope} is a tool that makes tiny objects
look larger so we can study their details. A light \omterm{prop:g4:cells:building}{microscope} uses
lenses and a lamp (or daylight) to enlarge a thin sample on a glass
slide.
\end{definition}

\omimg{microscope.jpg}{0.52\linewidth}{A compound light \omterm{prop:g4:cells:building}{microscope}
enlarges tiny specimens so \omterm{def:g4:cells:cell}{cells} become visible.}

\begin{method}[Looking at a prepared slide]\label{met:g4:cells:slide}
\begin{enumerate}
  \item \omterm{def:g1:caring:needs}{Place} the slide carefully on the stage and clip it in \omterm{def:g1:caring:needs}{place}.
  \item Start with the lowest power lens and look through the eyepiece.
  \item Use the focus knobs slowly until the image is sharp.
  \item Move to a higher power only after the object is centred and in
        focus.
\end{enumerate}
\end{method}

\section{Parts shared by many cells}

\begin{definition}[Cell membrane and nucleus]\label{def:g4:cells:parts}
The \emph{cell membrane}\index{cell membrane} is a thin outer \omterm{def:g1:senses:five}{skin} that
controls what enters and \omterm{def:g2:plant-parts:main}{leaves} the \omterm{def:g4:cells:cell}{cell}. The
\emph{nucleus}\index{nucleus (cell)} is a control centre that holds
instructions for the \omterm{def:g4:cells:cell}{cell}'s work. The watery material around the nucleus
is the \emph{cytoplasm}\index{cytoplasm}, where many of the \omterm{def:g4:cells:cell}{cell}'s
activities happen.
\end{definition}

\section{Plant cells and animal cells}

\begin{definition}[Cell wall and chloroplast]\label{def:g4:cells:plant}
A \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g4:cells:cell}{cell} has a stiff \emph{cell wall}\index{cell wall} outside the
membrane that supports the \omterm{def:g1:plants-animals:plant}{plant}. Many \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g4:cells:cell}{cells} also have
\emph{chloroplasts}\index{chloroplast} --- green parts that capture light
\omterm{def:g2:food-energy:energy}{energy} to make \omterm{def:g2:food-energy:food}{food}. \omterm{def:g1:plants-animals:animal}{Animal} \omterm{def:g4:cells:cell}{cells} do not have \omterm{def:g4:cells:cell}{cell} walls or
chloroplasts.
\end{definition}

\begin{example}[Onion and cheek]\label{ex:g4:cells:onion}
Onion \omterm{def:g1:senses:five}{skin} \omterm{def:g4:cells:cell}{cells} show clear walls under a \omterm{prop:g4:cells:building}{microscope}. A gentle swab from
the inside of a cheek (only with proper \omterm{def:g3:life-cycles:butterfly}{adult} supervision and sterile
tools in a school lab) shows \omterm{def:g1:plants-animals:animal}{animal} \omterm{def:g4:cells:cell}{cells} with membranes but no walls.
\end{example}

\begin{proposition}[Same idea, different details]\label{prop:g4:cells:compare}
\omterm{def:g1:plants-animals:plant}{Plant} and \omterm{def:g1:plants-animals:animal}{animal} \omterm{def:g4:cells:cell}{cells} share the basic plan of membrane, \omterm{def:g4:cells:parts}{cytoplasm} and
\omterm{def:g4:cells:parts}{nucleus}. Extra \omterm{def:g1:plants-animals:plant}{plant} structures match the \omterm{def:g1:plants-animals:plant}{plant}'s \omterm{def:g1:caring:needs}{needs}: support and
making \omterm{def:g2:food-energy:food}{food} from light.
\end{proposition}
\admitted

\section{Many cells, many jobs}

\begin{definition}[Specialised cell]\label{def:g4:cells:special}
In multi-celled \omterm{def:g1:living-nonliving:living}{living things}, a \emph{specialised cell}\index{cell!specialised}
is a \omterm{def:g4:cells:cell}{cell} with a shape and features suited to a particular job --- for
example, a long nerve \omterm{def:g4:cells:cell}{cell} that carries messages, or a red blood \omterm{def:g4:cells:cell}{cell}
that carries oxygen.
\end{definition}

\begin{example}[Teamwork]\label{ex:g4:cells:team}
\omterm{def:g3:bones:muscle}{Muscle} \omterm{def:g4:cells:cell}{cells} contract to move the \omterm{def:g1:my-body:body}{body}. \omterm{def:g2:plant-parts:main}{Leaf} \omterm{def:g4:cells:cell}{cells} with many
\omterm{def:g4:cells:plant}{chloroplasts} make sugar. \omterm{def:g2:plant-parts:main}{Root} \omterm{def:g4:cells:cell}{cells} often take up \omterm{def:g1:caring:needs}{water}. Groups of
similar \omterm{def:g4:cells:cell}{cells} form tissues, which build organs such as the heart or a
\omterm{def:g2:plant-parts:main}{leaf}.
\end{example}
