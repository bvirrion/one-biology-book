\chapter{Natural Selection}\label{ch:g9:selection}

\omterm{def:g8:evolution:def}{Evolution} \omterm{def:g1:caring:needs}{needs} a mechanism. \omterm{def:g9:selection:ns}{Natural selection} explains how \omterm{def:g8:populations:pop}{populations}
adapt: heritable \omterm{def:g5:evolution:variation}{variation}, differential survival and \omterm{def:g5:reproduction:def}{reproduction}, and
change across generations.

\section{Darwin's core argument}

\begin{definition}[Natural selection]\label{def:g9:selection:ns}
\emph{Natural selection}\index{natural selection} is the process by which
individuals with heritable \omterm{def:g7:heredity:trait}{traits} that improve survival or \omterm{def:g5:reproduction:def}{reproduction} in
a given \omterm{def:g4:traits:env}{environment} leave more \omterm{def:g2:growing:offspring}{offspring}, so those \omterm{def:g7:heredity:trait}{traits} become more
common in the \omterm{def:g8:populations:pop}{population} over generations.
\end{definition}

\begin{proposition}[Requirements]\label{prop:g9:selection:req}
\omterm{def:g9:selection:ns}{Natural selection} \omterm{def:g1:caring:needs}{needs}:
\begin{enumerate}
  \item \omterm{def:g5:evolution:variation}{variation} among individuals;
  \item heritability of (some of) that \omterm{def:g5:evolution:variation}{variation};
  \item differential reproductive success linked to the \omterm{def:g7:heredity:trait}{trait}.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Peppered moths story]\label{ex:g9:selection:moth}
In polluted woodlands, darker moths were better camouflaged on sooty
trees; their relative frequency rose. When \omterm{def:g4:plants-need:needs}{air} cleaned, patterns shifted
again --- selection tracks \omterm{def:g4:traits:env}{environment}.
\end{example}

\section{Fitness and adaptation}

\begin{definition}[Fitness]\label{def:g9:selection:fitness}
In evolutionary \omterm{def:g6:method:science}{biology}, \emph{fitness}\index{fitness (evolution)} means
reproductive success relative to others --- not gym strength.
\end{definition}

\begin{definition}[Adaptation]\label{def:g9:selection:adapt}
An \emph{adaptation}\index{adaptation} is a heritable \omterm{def:g7:heredity:trait}{trait} that improves
\omterm{def:g9:selection:fitness}{fitness} in a particular \omterm{def:g4:traits:env}{environment} (camouflage, drought tolerance, beak
shape matching \omterm{def:g2:food-energy:food}{food}).
\end{definition}

\begin{proposition}[Selection is not random]\label{prop:g9:selection:not-random}
\omterm{ex:g9:dna:mut}{Mutations} may arise randomly with respect to need, but which variants
spread is often non-random: \omterm{def:g4:traits:env}{environments} filter \omterm{def:g7:heredity:trait}{traits}. \omterm{def:g8:evolution:def}{Evolution} by
selection is undirected regarding future goals, yet historically
patterned.
\end{proposition}
\admitted

\begin{method}[Applying selection reasoning]\label{met:g9:selection:apply}
\begin{enumerate}
  \item Name the \omterm{def:g7:heredity:trait}{trait} \omterm{def:g5:evolution:variation}{variation}.
  \item State the environmental challenge.
  \item Explain who \omterm{def:g7:tissues:plant-organs}{leaves} more \omterm{def:g2:growing:offspring}{offspring} and why.
  \item Predict how the \omterm{def:g8:populations:pop}{population} changes over generations.
\end{enumerate}
\end{method}

\begin{example}[Antibiotic resistance]\label{ex:g9:selection:ab}
\omterm{def:g4:microbes:def}{Bacteria} with resistance \omterm{def:g7:heredity:allele}{alleles} survive treatment and \omterm{def:g2:what-alive:signs}{reproduce}; resistant
types increase. Misuse of \omterm{prop:g8:microbiology:compare}{antibiotics} strengthens this selection.
\end{example}
