\chapter{التبادل الغازي التنفسي}\label{ch:b1:gas-exchange}

لتر الهواء يحمل من الأكسجين ثلاثين ضعف ما يحمله لتر الماء، ويزن
أخف منه ثماني مئة مرة، ويدع الأكسجين ينتشر فيه أسرع عشرة آلاف مرة.
وعلى سمكة السلمون المرقط أن تمرر مئة وخمسين غرامًا من الماء على
غلاصمها مقابل كل ميليغرام من الأكسجين تأخذه، وهي تستخلص أربعة
أخماس ما يحمله الماء؛ أما الإنسان فيحرك بضعة غرامات من الهواء لكل
ميليغرام ويهدر ثلاثة أرباعها. والعضوان جوابان عن المعادلة نفسها ---
أي دفق غاز عبر سطح --- في وسطين. ويعرض هذا الفصل الفيزياء، ويشتق
ما ينبغي أن يكون عليه \omterm{prop:b1:organism-environment:surfaces}{سطح التبادل}، ويصف \omterm{def:b1:gas-exchange:gill}{الغلاصم} والرئات \omterm{prop:b1:gas-exchange:designs}{والقصبات
الهوائية}، ويتتبع الأكسجين وثاني أكسيد الكربون في الدم، وينظر في
كيفية التحكم في التهوية وفيما يتغير في المرتفعات وتحت الماء.

\section{فيزياء غاز في وسطين}

\begin{definition}[الضغط الجزئي والانحلالية]\label{def:b1:gas-exchange:partial}
\index{ضغط جزئي}
في خليط من الغازات يمارس كل غاز \emph{ضغطًا جزئيًا} يتناسب مع
كسره: ففي هواء عند \qty{101}{kPa} يكون للأكسجين
(\qty{20.9}{\%}) $P_{\mathrm{O_2}} = \qty{21.2}{kPa}$، ولثاني
أكسيد الكربون (\qty{0.04}{\%}) \qty{0.04}{kPa}. وينحل الغاز في
سائل بنسبة ضغطه الجزئي (قانون هنري): فالماء المتوازن مع الهواء عند
\qty{15}{\celsius} يحمل \qty{0.32}{mmol/L} من الأكسجين
(\qty{10}{mg/L}، أي \qty{7}{mL/L} من الغاز)، في مقابل
\qty{8.7}{mmol/L} (\qty{210}{mL/L}) في الهواء نفسه. والضغط الجزئي
--- وهو نفسه في الماء والهواء عند التوازن --- هو ما يقود الانتشار
بين الوسطين؛ والمحتوى هو ما يستطيع المتنفس استخلاصه. وثاني أكسيد
الكربون أذوب من الأكسجين بخمس وعشرين مرة، فلا يكون قط الغاز الأصعب
تبادلًا.
\end{definition}

\begin{theorem}[قانون فيك لسطح تبادل]\label{thm:b1:gas-exchange:fick}
\index{قانون فيك}
سرعة عبور غاز غشاءً مساحته $A$ وسماكته $x$ يفصل وسطين عند ضغطين
جزئيين $P_1$ و$P_2$ هي
\[
\dot V = K\,\frac{A}{x}\,(P_1 - P_2),
\]
حيث $K$، وهو ثابت الانتشار (ثابت كروغ) للغشاء، يجمع انتشارية الغاز
وانحلاليته في \omterm{def:b1:body-plans-tissues:tissue}{النسيج}. فالدفق ترفعه مساحة أكبر، وحاجز أرق، وفرق
\omterm{def:b1:gas-exchange:partial}{ضغط جزئي} أكبر؛ ولا شيء غير ذلك.
\end{theorem}
\begin{proof}
يعطي \omterm{thm:b1:membranes-transport:fick}{قانون فيك} الأول للغاز المنحل في الغشاء
(\cref{ch:b1:membranes-transport}) دفقًا في وحدة المساحة يتناسب مع
تدرج التركيز؛ وبقانون هنري يتناسب التركيز عند كل وجه مع \omterm{def:b1:gas-exchange:partial}{الضغط
الجزئي} في الوسط هناك، فيكون التدرج $(P_1 - P_2)/x$ مضروبًا في
الانحلالية، ويستوعب الثابت الانحلالية والانتشارية.
\end{proof}

\begin{proposition}[ما ينبغي أن يكون عليه سطح التبادل]\label{prop:b1:gas-exchange:requirements}
\index{سطح تبادل}
كائن أكبر من ميليمتر لا يمكن إمداده بالانتشار عبر سطحه الخارجي
(\cref{ch:b1:organism-environment}). فهو يحتاج إلى \emph{سطح
تنفسي} واسع (لرفع $A$)، ورقيق (لخفض $x$)، ورطب (فالغازات لا تعبر
\omterm{def:b1:membranes-transport:membrane}{الأغشية} إلا منحلة)، و\emph{مهوًّى} --- أي يُجدَّد الوسط الخارجي حتى
يبقى $P_1$ عاليًا --- و\emph{مروًّى} --- أي يُجدَّد الدم خلفه حتى يبقى
$P_2$ منخفضًا؛ وإلى دوران يحمل الغاز من السطح إلى \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}.
\omterm{def:b1:gas-exchange:gill}{والغلاصم} والرئات \omterm{prop:b1:gas-exchange:designs}{والقصبات الهوائية} ثلاث طرائق لطي مثل هذا السطح في
جسم وتحريك المائعين عبره.
\end{proposition}

\section{الغلاصم والتيار المعاكس}

\begin{definition}[الغلصمة]\label{def:b1:gas-exchange:gill}
\index{غلصمة}
\emph{الغلصمة} انبعاج خارجي لسطح الجسم في الماء، مطوي في خيوط
و\emph{صفائح} سماكتها بضعة ميكرومترات يجري فيها الدم في شعيرات.
وغلاصم السمكة تتدلى من أربعة أقواس عظمية في كل جانب؛ والماء
المسحوب عند الفم والمدفوع خارجًا من تحت أغطية الغلاصم بمضختين يجري
بين الصفائح في اتجاه، والدم في الصفائح يجري في الاتجاه الآخر: أي
\emph{تيار معاكس}\index{تبادل بتيار معاكس}. ولسمكة سلمون مرقط
كتلتها كيلوغرام نحو \qty{2000}{cm^2} من سطح الصفائح، سماكته
\qty{2}{\micro m}.
\end{definition}

\begin{theorem}[التبادل بتيار معاكس]\label{thm:b1:gas-exchange:countercurrent}
حين يجري الماء والدم في اتجاهين متعاكسين على طول \omterm{prop:b1:gas-exchange:requirements}{سطح تبادل}، يستطيع
الدم المغادر \omterm{def:b1:gas-exchange:gill}{للغلصمة} أن يقارب \omterm{def:b1:gas-exchange:partial}{الضغط الجزئي} للماء \emph{الداخل}
إليها، ويمكن تجريد الماء من معظم أكسجينه: أي استخلاص \qty{80}{\%}
أو أكثر. وحين يجريان في الاتجاه نفسه (تيار موافق) يلتقي المائعان
عند قيمة وسطى مشتركة ولا يستطيع الاستخلاص أن يتجاوز \qty{50}{\%}.
\end{theorem}
\begin{proof}
على طول مبادل بتيار موافق يهبط الفرق $P_{\text{الماء}} -
P_{\text{الدم}}$ من قيمته عند المدخل إلى الصفر مع تقارب الاثنين
إلى الضغط نفسه؛ ومتى تساويا لم يعد غاز ينتقل، ونقطة اللقاء عند
دفقين متساويين هي المنتصف. أما في الجريان \omterm{def:b1:gas-exchange:gill}{بتيار معاكس} فالدم، وهو
يتجه نحو مخرجه، يلقى ماءً أحدث دائمًا، فيُحفظ فرق موجب على الطول
كله؛ فالدم عند مخرجه يواجه الماء الداخل عند \qty{21}{kPa}،
والماء عند مخرجه يواجه الدم الوريدي الداخل عند بضعة كيلوباسكالات.
والحد يضبطه السطح، لا التوازن.
\end{proof}

\begin{example}[ثمن الماء]\label{ex:b1:gas-exchange:priceofwater}
سمكة سلمون مرقط مستريحة كتلتها \qty{1}{kg} تأخذ نحو \qty{50}{mL}
من الأكسجين في الساعة. ومن ماء فيه \qty{7}{mL/L}، باستخلاص
\qty{80}{\%}، عليها أن تضخ \qty{9}{L} من الماء في الساعة عبر
غلاصمها --- أي تسعة كيلوغرامات من وسط أشد لزوجة من الهواء بخمسين
مرة، عبر قنوات عرضها جزء من الميليمتر. فعُشر استقلابها يذهب إلى
التنفس؛ والإنسان المستريح ينفق جزءًا من خمسين. وأدفئ البركة إلى
\qty{30}{\celsius} فالسمكة نفسها، وهي تحتاج إلى أكسجين أكثر من ماء
يحمل أقل، عليها أن تضخ ثلاثة أمثال ذلك.
\end{example}

\section{تنفس الهواء: الرئات والقصبات الهوائية}

\begin{definition}[رئة الثدييات]\label{def:b1:gas-exchange:lung}
\index{رئة}\index{سنخ}
\emph{الرئة} انبعاج داخلي لسطح الجسم في حجرة داخلية رطبة، تبقي
السطح مبتلًا في الهواء الجاف. ورئة الثدييات شجرة من الممرات
الهوائية تنتهي بنحو ثلاث مئة مليون \emph{سنخ}، وهي أكياس قطرها
\qty{0.2}{mm} جدرانها --- \omterm{def:b1:cell-unit-of-life:cell}{خلية} ظهارية واحدة وبطانة شعيرة واحدة،
\qty{0.5}{\micro m} في المجمل --- تبلغ \qty{100}{m^2} في الإنسان،
ملفوفة بشبكة شعيرات يمر بها كل الناتج القلبي. وتُهوَّى الرئة
\emph{مدًّا وجزرًا}: \omterm{def:b1:mammal-organization:bodyplan}{فالحجاب الحاجز} وعضلات الأضلاع توسّع الصدر
فيدخل الهواء؛ ثم ترتخي فيخرج الهواء من الطريق نفسه. والنفس الواحد
يحرك \qty{500}{mL}، منها \qty{150}{mL} لا تملأ إلا الممرات
(\emph{الحيّز الميت}) ولا تبلغ سنخًا أبدًا؛ والغاز السنخي، وهو
يتجدد بسبعه في كل نفس، يبقى عند $P_{\mathrm{O_2}} =
\qty{13.3}{kPa}$ و$P_{\mathrm{CO_2}} = \qty{5.3}{kPa}$ --- وهما
الضغطان اللذان يتوازن معهما الدم.
\end{definition}

\begin{proposition}[ثلاثة تصاميم للهواء]\label{prop:b1:gas-exchange:designs}
\index{قصبات هوائية}
\emph{الثدييات} تهوّي مدًّا وجزرًا: وهو بسيط، لكن الهواء العذب
يختلط بالبائت، ويكون الأكسجين السنخي ثلثي أكسجين الجو، والاستخلاص
ربعًا. و\emph{الطيور} تمرر الهواء في اتجاه واحد عبر أنابيب دقيقة
(قصيبات موازية) بين مجموعتين من الأكياس الهوائية تعملان منفاخًا،
فيجري الهواء العذب عبر \omterm{prop:b1:organism-environment:surfaces}{سطح التبادل} في الشهيق والزفير معًا، في تيار
متقاطع مع الدم: فاستخلاصها أعلى، وإوزة رأسية الأشرطة تطير فوق
الهيمالايا حيث يُغشى على ثديي. و\emph{الحشرات} لا دم تنفسيًا لها
البتة: فالهواء يدخل عبر صمامات (\emph{ثغور تنفسية}) إلى
\emph{\omterm{prop:b1:gas-exchange:designs}{قصبات هوائية}}، وهي أنابيب تشدّها تسمكات حلزونية تتفرع حتى
الميكرومتر وتوصل الأكسجين بالانتشار مباشرة إلى كل \omterm{def:b1:cell-unit-of-life:cell}{خلية}، يعينها ضخ
البطن في الحشرات الكبيرة أو النشيطة. والانتشار على طول الأنابيب
يحصر التصميم في الأجسام الصغيرة --- وهذا أحد أسباب ألا تكون حشرة
أكبر من فأر.
\end{proposition}

\begin{method}[قراءة ميزانية تهوية]\label{met:b1:gas-exchange:budget}
\begin{enumerate}
  \item التهوية الدقيقية $=$ الحجم المدّي $\times$ عدد الأنفاس في
        الدقيقة؛ والتهوية السنخية $=$ (الحجم المدّي $-$ الحيّز الميت)
        $\times$ عدد الأنفاس في الدقيقة. ولا يتبادل الغاز إلا
        الثانية.
  \item أخذ الأكسجين $=$ التهوية السنخية $\times$ (الكسر الملهَم
        $-$ الكسر السنخي)؛ وفي الراحة،
        $\qty{4.2}{L/min}\times(0.21 - 0.14) \approx \qty{0.3}{L/min}$.
  \item الاستخلاص $=$ الأخذ $/$ (التهوية $\times$ الكسر الملهَم):
        أي ربع في \omterm{def:b1:gas-exchange:lung}{رئة} مدّية، وأربعة أخماس في \omterm{def:b1:gas-exchange:gill}{غلصمة} \omterm{def:b1:gas-exchange:gill}{بتيار معاكس}.
  \item وفي جانب الدم: الأخذ $=$ الناتج القلبي $\times$
        (المحتوى الشرياني $-$ الوريدي)؛ أي \qty{5}{L/min}$\times$
        \qty{50}{mL/L} في الراحة، وهي \qty{250}{mL/min} نفسها.
        وعلى الميزانيتين أن تتطابقا.
\end{enumerate}
\end{method}

\section{الغازات في الدم}

\begin{proposition}[نقل الأكسجين وثاني أكسيد الكربون]\label{prop:b1:gas-exchange:transport}
\index{أنهيدراز الكربونيك}\index{أثر بور}\index{أثر هالدين}
لا تحل البلازما إلا \qty{3}{mL} من الأكسجين في اللتر عند الضغط
الشرياني؛ أما الهيموغلوبين (\cref{ch:b1:proteins}) فيحمل
\qty{200}{mL}، يُحمَّل في \omterm{def:b1:gas-exchange:lung}{الرئة} عند \qty{13.3}{kPa} ويفرّغ خمسه إلى
نصفه في \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}. ويسافر ثاني أكسيد الكربون بثلاث طرائق:
\qty{7}{\%} منحلًا، و\qty{23}{\%} مرتبطًا بالزمر الأمينية في
الهيموغلوبين، و\qty{70}{\%} بيكربونات: ففي الكرية الحمراء يحوّل
\emph{\omterm{prop:b1:gas-exchange:transport}{أنهيدراز الكربونيك}} $\mathrm{CO_2}$ والماء إلى \omterm{def:b1:water-small-molecules:ph}{حمض} الكربونيك
في مللي ثوانٍ، فينفصل \omterm{def:b1:water-small-molecules:ph}{الحمض}، وتغادر البيكربونات \omterm{def:b1:cell-unit-of-life:cell}{الخلية} في مبادلة
مع الكلوريد (\emph{انزياح الكلوريد})، ويلتقط الهيموغلوبين
البروتون. والغازان يعين أحدهما الآخر: \omterm{def:b1:water-small-molecules:ph}{فالحمض} و$\mathrm{CO_2}$
يخفضان ألفة الهيموغلوبين للأكسجين (\emph{\omterm{prop:b1:gas-exchange:transport}{أثر بور}})، فيتحرر
الأكسجين حيث يُصنع $\mathrm{CO_2}$؛ والهيموغلوبين منزوع الأكسجين
يرتبط بالبروتونات و$\mathrm{CO_2}$ ارتباطًا أفضل (\emph{\omterm{prop:b1:gas-exchange:transport}{أثر
هالدين}})، فيُلتقط $\mathrm{CO_2}$ حيث يتحرر الأكسجين، ويُطلق في
\omterm{def:b1:gas-exchange:lung}{الرئة} مع تحميل الأكسجين.
\end{proposition}

\section{التحكم والارتفاع والغوص}

\begin{proposition}[التهوية يتحكم فيها ثاني أكسيد الكربون]\label{prop:b1:gas-exchange:control}
\index{مستقبل كيميائي}
يتولد نظم التنفس في جذع الدماغ وتضبطه \emph{مستقبلات كيميائية}:
مركزية تغمرها السوائل حول الدماغ فتستجيب لتغير pH الذي يحدثه
$\mathrm{CO_2}$؛ ومحيطية في الشريانين السباتيين تستجيب لهبوط
الأكسجين الشرياني دون نحو \qty{8}{kPa}. وفي الحياة العادية يحكم
$\mathrm{CO_2}$: فارتفاع \qty{1}{kPa} في
$P_{\mathrm{CO_2}}$ الشرياني يضاعف التهوية تقريبًا، بينما على
الأكسجين أن يهبط ثلثًا قبل أن تستجيب التهوية. وحبس النفس ينتهي حين
يبلغ $\mathrm{CO_2}$، لا الأكسجين، الحد؛ وفرط التهوية قبله يخفض
$\mathrm{CO_2}$ ويدع الغائص يبقى تحت الماء حتى ينفد الأكسجين ---
فيفقد وعيه قبل أن تعود إليه الرغبة في التنفس.
\end{proposition}

\begin{example}[الارتفاع والعمق]\label{ex:b1:gas-exchange:altitudediving}
عند \qty{4000}{m} يحمل الهواء \qty{12}{kPa} من الأكسجين \omterm{def:b1:gas-exchange:lung}{والأسناخ}
\qty{7}{}: فتدفع الأجسام السباتية التهوية إلى الأعلى، وهو ما يطرد
$\mathrm{CO_2}$ ويجعل الدم قلويًا --- وهو كابح تحرره الكليتان على
مدى أيام بطرح البيكربونات؛ وترفع الكريات الحمراء BPG فيها في أيام
ويرفع النقي الهيموغلوبين في أسابيع
(\cref{ch:b1:proteins}). أما الفقمة الغائصة عشرين دقيقة فتفعل
العكس: تزفر قبل الغوص، وتبطئ قلبها إلى العشر، وتقطع الدم عن كل شيء
إلا الدماغ والقلب، وتشغّل عضلاتها على الأكسجين المرتبط بميوغلوبينها
هي، وهو أغلظ تركيزًا من ميوغلوبيننا بعشر مرات، ثم على \omterm{def:b1:respiration-fermentation:fermentation}{التخمر}،
محتملةً لاكتاتًا يصفّيه الجسم عند السطح. وتنطبق رئتاها دون
\qty{50}{m}، وهو ما يبقي النتروجين خارج دمها.
\end{example}

بأي أشكال ثلاثة يُحمل ثاني أكسيد الكربون في الدم، وبأي نسب؟

\begin{exercise}[$\star\star$]\label{exo:b1:gas-exchange:6}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:gas-exchange:7}
\omterm{def:b1:gas-exchange:lung}{رئة} مساحتها \qty{100}{m^2} وسماكتها \qty{0.5}{\micro m} تنقل

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:gas-exchange:8}
فسّر انزياح الكلوريد: لماذا تغادر البيكربونات الكرية الحمراء،
ولماذا يدخل الكلوريد، وماذا كان يحدث لحجم \omterm{def:b1:cell-unit-of-life:cell}{الخلية} بلا هذه المبادلة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:gas-exchange:9}
لماذا يخاطر غائص فرط تهويته قبل غوصة حبس نفس بالغرق، بينما يصعد
الذي لم يفعل سالمًا، منزعجًا لكن واعيًا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:gas-exchange:10}
قصبة هوائية في حشرة طولها \qty{1}{mm} وقطرها \qty{20}{\micro m}
تمد عضلة تستهلك \qty{1e-9}{mol} من الأكسجين في الثانية. فبمعامل
$D = \qty{2e-5}{m^2/s}$ في الهواء و\qty{8.7}{mol/m^3} من الأكسجين
عند الثغر، احسب التركيز عند طرف العضلة (\omterm{thm:b1:membranes-transport:fick}{بقانون فيك}:
$J = DA\,\Delta c/L$). وأعد الحساب لقصبة طولها \qty{10}{mm}. فما
الذي يحد حجم الحشرات؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:gas-exchange:11}
الطيور والأسماك كلاهما يستعمل جريانًا غير مدّي. قارن مبادليهما
(اتجاه الوسط، وعلاقته بجريان الدم، والاستخلاص) وفسّر لماذا لا
تستطيع \omterm{def:b1:gas-exchange:lung}{رئة} مدّية بلوغ استخلاصهما مهما يكن سطحها.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:gas-exchange:12}
«\omterm{def:b1:gas-exchange:lung}{الرئة} ينظّمها الغاز الذي تطرحه، لا الغاز الذي تأخذه.» ناقش في
فقرة: لماذا يكون $\mathrm{CO_2}$ الإشارة الأفضل عند سطح البحر،
ومتى يخفق الترتيب، وكيف يكشفه الارتفاع.
\end{exercise}

\section{مسألة: سمكة سلمون مرقط وإنسان}

\begin{problem}[{مسألة نهاية الأسبوع --- غلصمة ورئة جنبًا إلى
جنب: الأكسجين في لتر الوسط، والتهوية والاستخلاص، والتيار المعاكس
والجريان المدّي، والدم خلف كل منهما، وختامًا كفاءة استخلاص الغلصمة
ذات التيار المعاكس}]
\label{pb:b1:gas-exchange:1}
إنسان مستريح يأخذ \qty{250}{mL} من الأكسجين في الدقيقة، متنفسًا
\qty{12}{} مرة في الدقيقة بحجم مدّي \qty{500}{mL} وحيّز ميت
\qty{150}{mL}؛ والهواء الملهَم \qty{21}{\%} أكسجينًا، والغاز
السنخي \qty{14}{\%}؛ والناتج القلبي \qty{5}{L/min}، والمحتوى
الشرياني من الأكسجين \qty{200}{mL/L}. وسمكة سلمون مرقط مستريحة
كتلتها \qty{1}{kg} تأخذ \qty{50}{mL} من الأكسجين في الساعة في ماء
عند \qty{15}{\celsius} يحمل \qty{7}{mL/L}؛ وهيموغلوبينها يحمل
\qty{100}{mL/L} عند الإشباع؛ ودمها يغادر \omterm{def:b1:gas-exchange:gill}{الغلاصم} مشبعًا
\qty{95}{\%} ويعود مشبعًا \qty{15}{\%}. والماء أكثف من الهواء
\qty{800}{} مرة.

\medskip
\noindent\textbf{الجزء الأول --- وسطان.}
\begin{enumerate}
  \item عبّر عن محتوى الهواء ومحتوى ماء السمكة من الأكسجين
        بالمليمولات في اللتر (\qty{22.4}{L/mol}).
  \item احسب نسبة المحتويين.
  \item احسب كتلة الوسط (الهواء \qty{1.2}{g/L}؛ والماء
        \qty{1000}{g/L}) التي تحمل \qty{1}{mmol} من الأكسجين في كل
        حالة.
  \item احسب حجم ماء السمكة الذي يحمل من الأكسجين قدر ما يحمل لتر
        من الهواء.
  \item فسّر في جملة واحدة لماذا يجب أن تُهوَّى \omterm{def:b1:gas-exchange:gill}{غلاصم} الأسماك في
        اتجاه واحد بينما يجوز للرئات أن تكون مدّية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- \omterm{def:b1:gas-exchange:lung}{الرئة}.}
\begin{enumerate}[resume]
  \item احسب التهوية الدقيقية والتهوية السنخية.
  \item أي كسر من كل نفس هو حيّز ميت، وأي كسر من التهوية الدقيقية
        يُهدر تبعًا لذلك؟
  \item احسب الأكسجين الموصل إلى \omterm{def:b1:gas-exchange:lung}{الأسناخ} في الدقيقة والأكسجين
        المأخوذ (التهوية السنخية $\times$ فرق الكسرين). وتحقق
        بمقارنته بالقيمة \qty{250}{mL/min}.
  \item احسب كفاءة الاستخلاص: الأخذ مقسومًا على الأكسجين في
        التهوية الدقيقية الملهَمة.
  \item احسب $P_{\mathrm{O_2}}$ السنخي من الكسر السنخي (الضغط
        الكلي \qty{101}{kPa}، وبخار الماء \qty{6.3}{kPa} يُطرح
        أولًا).
  \item وفي جانب الدم، احسب الفرق الشرياني الوريدي اللازم
        لتحقيق \qty{250}{mL/min} عند \qty{5}{L/min}، والمحتوى الوريدي
        وإشباعه.
  \item في أثناء التمرين يرتفع الأخذ إلى \qty{3}{L/min} بناتج
        قلبي \qty{20}{L/min}. احسب الفرق الشرياني الوريدي والإشباع
        الوريدي.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{def:b1:gas-exchange:gill}{الغلصمة}.}
\begin{enumerate}[resume]
  \item احسب أخذ السمكة من الأكسجين بالميليلترات في الدقيقة.
  \item باستخلاص \qty{80}{\%} (\omterm{def:b1:gas-exchange:gill}{بتيار معاكس})، احسب الماء المهوَّى في
        الدقيقة وفي الساعة.
  \item وبالاستخلاص الذي يعطيه مبادل بتيار موافق بسطح مساوٍ
        (\qty{43}{\%})، احسب الماء اللازم. فماذا يوفر \omterm{def:b1:gas-exchange:gill}{التيار
        المعاكس}؟
  \item احسب كتلة الماء المضخوخ في الساعة، وقارنها بكتلة الهواء
        الذي يحركه الإنسان في الساعة.
  \item احسب الفرق الشرياني الوريدي في محتوى الأكسجين عند السمكة
        والناتج القلبي الذي يقتضيه أخذها.
  \item احسب نسبة دفق الماء إلى دفق الدم عند \omterm{def:b1:gas-exchange:gill}{الغلصمة}، ونسبة دفق
        الهواء إلى دفق الدم عند \omterm{def:b1:gas-exchange:lung}{رئة} الإنسان (التهوية السنخية على
        الناتج القلبي). وعلّق.
  \item يكلف ضخ الماء السمكة نحو \qty{10}{\%} من أخذها للأكسجين.
        فإذا دفئ الماء إلى \qty{25}{\celsius}
        (\qty{5.5}{mL/L}) وارتفع أخذها بالنصف، فبأي معامل يجب أن
        ترتفع التهوية، وماذا يحدث للكسر المنفق على التنفس؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- المبادل نفسه.}
\begin{enumerate}[resume]
  \item في \omterm{def:b1:gas-exchange:gill}{الغلصمة} ذات \omterm{def:b1:gas-exchange:gill}{التيار المعاكس} يغادر الدم عند \qty{18}{kPa}
        مواجهًا ماءً يدخل عند \qty{21}{kPa}، ويغادر الماء عند
        \qty{4}{kPa} مواجهًا دمًا يدخل عند \qty{3}{kPa}. احسب
        فرق الضغط عند كل طرف.
  \item وفي مبادل بتيار موافق يغادر المائعان كلاهما عند
        \qty{12}{kPa}. احسب الفرق عند كل طرف وفسّر لماذا يتوقف
        التبادل قبل أن يُستنفد الماء.
  \item \omterm{def:b1:gas-exchange:lung}{رئة} الإنسان: السنخي \qty{13.3}{kPa}، والدم الوريدي يصل عند
        \qty{5.3}{kPa}، والشرياني يغادر عند \qty{13.3}{kPa}.
        فأي ترتيب تشبه \omterm{def:b1:gas-exchange:lung}{الرئة} المدّية، ولماذا لا يستطيع دمها
        الشرياني أن يتجاوز القيمة السنخية؟
  \item صفائح السمكة مجموعها \qty{2000}{cm^2} بسماكة
        \qty{2}{\micro m}؛ \omterm{def:b1:gas-exchange:lung}{وأسناخ} الإنسان \qty{100}{m^2} بسماكة
        \qty{0.5}{\micro m}. احسب نسبة $A/x$ بينهما، ونسبة أخذيهما
        للأكسجين. فماذا توحي المقارنة عن فرق الضغط الوسطي عبر صفيحة
        قياسًا إلى جدار سنخي؟
  \item فسّر لماذا لا يكون ثاني أكسيد الكربون حادًّا للسمكة أبدًا
        مع أن الماء يحمل من الأكسجين قدرًا ضئيلًا.
  \item اذكر النتيجة: كفاءة استخلاص \omterm{def:b1:gas-exchange:gill}{غلصمة} السمكة \omterm{def:b1:gas-exchange:lung}{ورئة} الإنسان،
        والسمتان في \omterm{def:b1:gas-exchange:gill}{الغلصمة} اللتان تصنعان الفرق.
\end{enumerate}
\end{problem}
