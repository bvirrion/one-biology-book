\chapter{القلب والدم}\label{ch:g4:heart-and-blood}

اضغط بإصبعين --- لا بالإبهام --- على باطن رسغك، تحت أصل الإبهام،
وانتظر. وإذا بها: طرقة صغيرة ثابتة، مرة بعد مرة، نحو مرة في
الثانية. تلك الطرقة قلبك وهو يعمل، محسوسًا من بعيد. وهو ينبض منذ ما
قبل ولادتك، وهو محرّك خدمة التوصيل في الجسم.

\section{الدم، خدمة التوصيل في الجسم}

\begin{proposition}[ماذا يحمل الدم]\label{prop:g4:heart-and-blood:carries}
\emph{الدم}\index{دم} سائل يبلغ كل ركن من الجسم، وهو ناقل الجسم:
\begin{itemize}
  \item يجمع من \omterm{def:g4:journey-of-food:tract}{المعي الدقيق} \emph{\omterm{def:g4:journey-of-food:digestion}{المغذيات}}
        (\cref{prop:g4:journey-of-food:absorb})؛
  \item ويجمع من الرئتين \emph{الأكسجين}
        (\cref{prop:g4:breathing:exchange})؛
  \item ويوصّل الاثنين إلى كل جزء من الجسم --- \omterm{def:g3:bones-and-muscles:muscle}{العضلات}، والدماغ،
        \omterm{def:g3:bones-and-muscles:skeleton}{والعظام}، \omterm{def:g1:the-five-senses:sense}{والجلد}؛
  \item ويحمل الفضلات في طريق العودة، فيوصّل ثاني أكسيد الكربون
        إلى الرئتين ليُزفر.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[جولة توصيل واحدة]\label{ex:g4:heart-and-blood:round}
تتبّع قطرة دم: فهي تصير في الرئتين حمراء زاهية محمّلة بالأكسجين؛
ويرسلها \omterm{def:g4:heart-and-blood:heart}{القلب} إلى الخارج؛ وفي \omterm{def:g3:bones-and-muscles:muscle}{عضلة} \omterm{def:g1:my-body:parts}{رجل} تجري تسلّم الأكسجين
\omterm{def:g4:journey-of-food:digestion}{والمغذيات}، وتأخذ ثاني أكسيد الكربون، فتغمق؛ وحين تعود إلى \omterm{def:g4:heart-and-blood:heart}{القلب}
تُرسل إلى الرئتين لتفرّغ وتُعبَّأ من جديد --- ثم تخرج ثانية. والجولة
الكاملة تستغرق أقل من دقيقة بكثير.
\end{example}

\begin{definition}[الأوعية الدموية]\label{def:g4:heart-and-blood:vessels}
يسافر \omterm{prop:g4:journey-of-food:absorb}{الدم} في أنابيب تسمى \emph{الأوعية}\index{وعاء دموي}
الدموية: واسعة تغادر \omterm{def:g4:heart-and-blood:heart}{القلب} وتدخله، ثم تتفرع أدقّ فأدقّ --- حتى
أوعية في رقّة الشعرة تتخلل أجزاء الجسم، وعندها تُسلَّم التوصيلات.
ولو صُفّت أوعية جسم واحد طرفًا إلى طرف لامتدت عشرات آلاف
الكيلومترات.
\end{definition}

\begin{remark}[رؤية الأوعية]\label{rem:g4:heart-and-blood:see}
انظر إلى باطن رسغك: خطوط مزرقّة تحت \omterm{def:g1:the-five-senses:sense}{الجلد} --- وهي \omterm{def:g4:heart-and-blood:vessels}{أوعية دموية} في
طريقها راجعةً إلى \omterm{def:g4:heart-and-blood:heart}{القلب}. \omterm{def:g1:my-body:joint}{والركبة} المسحوجة تُظهر الطرف الآخر من
الشبكة: فحتى أرقّ جرح يجد دمًا، لأن أدقّ \omterm{def:g4:heart-and-blood:vessels}{الأوعية} في كل مكان.
\end{remark}

\section{القلب، وهو المضخة}

\begin{definition}[القلب]\label{def:g4:heart-and-blood:heart}
\emph{القلب}\index{قلب} \omterm{def:g3:bones-and-muscles:muscle}{عضلة} جوفاء، في حجم قبضة صاحبها تقريبًا، في
وسط الصدر خلف القفص الصدري. وهو بالعصر --- أي \emph{بالدقّة}\index{دقة
القلب} --- يدفع \omterm{prop:g4:journey-of-food:absorb}{الدم} في \omterm{def:g4:heart-and-blood:vessels}{الأوعية}. وله جانبان: الجانب الأيمن يضخّ
\omterm{prop:g4:journey-of-food:absorb}{الدم} إلى \emph{الرئتين} ليحمّل الأكسجين؛ والجانب الأيسر يضخّ \omterm{prop:g4:journey-of-food:absorb}{الدم}
الغني بالأكسجين إلى \emph{الجسم كله}.
\end{definition}

\begin{proposition}[النبض]\label{prop:g4:heart-and-blood:pulse}
كل دقّة تدفع موجة صغيرة من \omterm{prop:g4:journey-of-food:absorb}{الدم} في \omterm{def:g4:heart-and-blood:vessels}{الأوعية}؛ وتلك الموجة، محسوسةً
عند \omterm{def:g1:my-body:joint}{الرسغ} أو العنق، هي \emph{النبض}\index{نبض}. وعدّ \omterm{prop:g4:heart-and-blood:pulse}{النبض} عدّ
لدقات \omterm{def:g4:heart-and-blood:heart}{القلب}: نحو $80$ إلى $100$ في الدقيقة عند الطفل المستريح،
و$60$ إلى $80$ عند البالغ --- وأكثر من ذلك بكثير في أثناء الجهد.
\end{proposition}
\admitted

\begin{example}[القلب يجاري الرجلين]\label{ex:g4:heart-and-blood:effort}
حين تجري تحرق \omterm{def:g3:bones-and-muscles:muscle}{عضلات} رجليك \omterm{def:g4:journey-of-food:digestion}{المغذيات} والأكسجين بسرعة؛ فوجب أن تسرع
التوصيلات. فيدقّ \omterm{def:g4:heart-and-blood:heart}{القلب} أسرع \emph{وأقوى} --- وقد يتجاوز \omterm{prop:g4:heart-and-blood:pulse}{نبض} الطفل
$150$ في العدو --- بينما يسرع \omterm{def:g4:breathing:movements}{التنفس} ليعيد تحميل \omterm{prop:g4:journey-of-food:absorb}{الدم}
(\cref{ex:g4:breathing:running}). \omterm{def:g4:heart-and-blood:heart}{فالقلب} والرئتان نصفا سلسلة تموين
واحدة.
\end{example}

\begin{method}[أخذ النبض]\label{met:g4:heart-and-blood:pulse}
\begin{enumerate}
  \item اقلب يدًا راحتها إلى أعلى؛ وضع إصبعين من اليد الأخرى على
        \omterm{def:g1:my-body:joint}{الرسغ}، تحت أصل الإبهام مباشرة --- ولا تضع الإبهام أبدًا،
        فله \omterm{prop:g4:heart-and-blood:pulse}{نبض} صغير خاص به؛
  \item واضغط برفق حتى تحسّ بالطرق؛
  \item وعُدّ الطرقات دقيقة واحدة (وساعة بثوانٍ تعين)؛
  \item واكتب العدد، ومعه ما كان الجسم يفعله: راحة، أو مشيًا، أو
        بعد عدو.
\end{enumerate}
\end{method}

\begin{remark}[عضلة لا تستريح --- تقريبًا]\label{rem:g4:heart-and-blood:rest}
بين دقّتين يرتخي \omterm{def:g4:heart-and-blood:heart}{القلب} جزءًا من الثانية --- وتلك الشظية كل ما
يأخذه من راحة. ليلًا ونهارًا، نحو $100000$ دقّة في اليوم، طول
العمر. وهو كأي \omterm{def:g3:bones-and-muscles:muscle}{عضلة} (\cref{met:g3:bones-and-muscles:care}) يقوى
بالتمرين: \omterm{def:g4:heart-and-blood:heart}{فالقلب} المدرَّب يحرّك دمًا أكثر في الدقّة الواحدة، فيستطيع
أن يدقّ أهدأ.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:1}
سمِّ شيئين يوصّلهما \omterm{prop:g4:journey-of-food:absorb}{الدم}، وأين يجمع كل واحد منهما.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:2}
أي غاز فضلة يحمله \omterm{prop:g4:journey-of-food:absorb}{الدم} بعيدًا، وأين يُنزله؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:3}
مم يتكوّن \omterm{def:g4:heart-and-blood:heart}{القلب}، وما حجمه، وأين يقع؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:4}
ماذا يفعل كل جانب من جانبي \omterm{def:g4:heart-and-blood:heart}{القلب}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:5}
ما \omterm{prop:g4:heart-and-blood:pulse}{النبض}، وعلام يدلك عدّه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:6}
لماذا يجب أن يؤخذ \omterm{prop:g4:heart-and-blood:pulse}{النبض} بالأصابع لا بالإبهام؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:7}
لماذا يتسارع \omterm{prop:g4:heart-and-blood:pulse}{النبض} في أثناء العدو؟ احكِ الحكاية من وجهة نظر \omterm{def:g3:bones-and-muscles:muscle}{عضلات}
\omterm{def:g1:my-body:parts}{الرجلين}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:heart-and-blood:8}
جرّب: خذ نبضك مستريحًا وفق \cref{met:g4:heart-and-blood:pulse}، ثم
اقفز قفزات نجمية ثلاثين ثانية، ثم خذه من جديد. اذكر العددين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:heart-and-blood:9}
في \cref{ex:g4:heart-and-blood:round}، تكون قطرة \omterm{prop:g4:journey-of-food:absorb}{الدم} حمراء زاهية
وهي تغادر الرئتين وأغمق وهي تغادر \omterm{def:g3:bones-and-muscles:muscle}{عضلة} \omterm{def:g1:my-body:parts}{الرجل}. فما الذي تغير فيها
في كل مرة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:heart-and-blood:10}
\omterm{prop:g4:journey-of-food:absorb}{الدم} يصل ثلاثة أعضاء من صفّك: \omterm{def:g4:journey-of-food:tract}{المعي الدقيق}، والرئتين، \omterm{def:g4:heart-and-blood:heart}{والقلب}.
ارسم الشبكة أو صفها: بم يسهم كل واحد في خدمة التوصيل؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:heart-and-blood:11}
\omterm{prop:g4:heart-and-blood:pulse}{نبض} دراج مدرَّب مستريح قد يكون $50$؛ \omterm{prop:g4:heart-and-blood:pulse}{ونبض} بالغ غير مدرَّب $75$.
استعمل \cref{rem:g4:heart-and-blood:rest} لتشرح كيف يكون
\emph{الأبطأ} \emph{أقوى} --- وما الذي يجب أن تفعله كل دقّة من دقات
الدرّاج؟
\end{exercise}
