\section*{Bab \ref{ch:g12:innate-immunity} --- Imunitas Bawaan}

\begin{solution}{exo:g12:innate-immunity:1}
Bawaan: ada sejak lahir, bekerja dalam hitungan jam, mengenali golongan
penyusup lewat pola bersama, dan menanggapi dengan cara yang sama
setiap kali. Adaptif: perlu beberapa hari, mengenali tiap penyusup satu
per satu, mengingatnya (dan hanya ada pada \omterm{def:g10:common-ancestry:bodyplan}{vertebrata}).
\end{solution}

\begin{solution}{exo:g12:innate-immunity:2}
Kemerahan dan panas: pembuluh yang melebar sehingga membawa lebih
banyak darah. Bengkak: dinding pembuluh yang bocor sehingga plasmanya
masuk ke jaringan. Nyeri: ujung saraf yang dipekakan oleh \omterm{prop:g12:innate-immunity:inflammation}{mediatornya}.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:3}
Pola molekul yang dibagi oleh golongan mikroba (serpihan dinding, \omterm{def:g10:chemistry-of-life:families}{asam
nukleat} virus) dan molekul yang dilepaskan \omterm{def:g10:cells-common-unit:cell}{sel} yang rusak. \omterm{def:g10:cells-common-unit:cell}{Sel} itu
melepaskan \omterm{prop:g12:innate-immunity:inflammation}{mediator}: histamin, prostaglandin, sitokina.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:4}
Kontak dan pengikatan oleh reseptor; penelanan oleh membrannya;
pencernaan di dalam vesikel dengan \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} dan zat kimia yang reaktif;
pemajangan serpihan \omterm{def:g11:gene-expression:protein}{protein} di permukaannya.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:5}
Keduanya menghambat \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} pembuat prostaglandin, sehingga mengurangi
bengkak, nyeri dan demam. Keduanya tak menyingkirkan sebabnya dan tak
membunuh satu mikroba pun.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:6}
Bawaan pada sekitar setengah hari; adaptif pada sekitar hari ke-9;
kurvanya berpotongan sekitar hari ke-5 sampai ke-6.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:7}
Pemicunya adalah sebuah pola molekul yang dikenali \omterm{def:g10:cells-common-unit:cell}{sel} sentinel, bukan
penyusup yang hidup. Larutan garam yang steril tak membawa pola dan tak
membawa kerusakan: paling banter ada tanggapan kecil terhadap jarumnya.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:8}
Neutrofil yang mati, bakteri yang mati dan yang hidup, puing yang
tercerna dan plasma. Neutrofilnya berdatangan selama berjam-jam lalu
mati sepanjang sehari; nanah adalah yang menumpuk setelah mereka
bertempur.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:9}
Demam jerami adalah histamin yang dilepaskan \omterm{def:g10:cells-common-unit:cell}{sel} mast terhadap serbuk
sari: menghambat histamin menyingkirkan tanggapannya. Pada sebuah
jangkitan, histamin hanya menggerakkan menit-menit pertama; bengkak,
nyeri dan perekrutan yang belakangan berjalan dengan prostaglandin dan
sitokina, yang tak disentuh antihistamin.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:10}
Pada jam ke-12 100\% berbanding 55\%; pada jam ke-36 50\% berbanding
30\%. Obatnya menurunkan bengkaknya pada setiap saat tetapi kurvanya
tetap jatuh ke nol pada sekitar jam ke-72: obat itu tak memperpendek
tanggapannya.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:11}
Serat \omterm{def:g10:muscles-and-joints:joint}{ligamen} yang robek dan \omterm{def:g10:cells-common-unit:cell}{sel} yang remuk melepaskan molekul khas \omterm{def:g10:cells-common-unit:cell}{sel}
yang rusak, yang dikenali reseptor \omterm{def:g10:cells-common-unit:cell}{sel} sentinel sebagaimana ia
mengenali mikroba: kerusakan saja sudah menjadi pemicu.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:12}
Tanpa neutrofil, hari-hari pertama itu hilang dan bakteri biasa pun
menguasai tubuhnya: tanggapan bawaanlah yang menahan garisnya seketika.
Tanpa \omterm{def:g12:innate-immunity:immunity}{imunitas adaptif}, tanggapan bawaan membersihkan jangkitan yang
pertama, tetapi jangkitan yang berlarut atau berulang, yang memerlukan
antibodi khas dan ingatan, akhirnya menang: tanggapan adaptif
merampungkan apa yang ditahan tanggapan bawaan.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:13}
Abses adalah peradangan yang sedang melawan bakteri hidup: meredamnya
memperlambat pembersihannya dan berisiko menyebar. Artritis adalah
peradangan tanpa mikroba apa pun, yang diarahkan kepada \omterm{def:g10:muscles-and-joints:joint}{sendinya}
sendiri: di sana tanggapannya justru penyakitnya, dan menguranginya
adalah pengobatannya.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:14}
Reseptor bawaan ditetapkan oleh \omterm{def:g10:universal-dna:gene}{gen} dan mengenali pola yang dibagi
seluruh golongan; tak ada apa pun pada pertemuan kedua yang mengubahnya.
Sistem adaptif membangun reseptor yang khas bagi tiap penyusup dan
menyimpan \omterm{def:g10:cells-common-unit:cell}{sel} yang membuatnya: ia mengenali individunya, dan menyimpan
pengenalannya.
\end{solution}

\begin{solution}{exo:g12:innate-immunity:15}
Peradangan menjadi obatnya ketika diarahkan kepada penyusup atau cedera
yang nyata, sepadan, dan mereda dalam beberapa hari: ia membersihkan
lokasinya dan memulai perbaikan. Ia menjadi penyakitnya ketika
diarahkan kepada sesuatu yang tak berbahaya (serbuk sari, \omterm{def:g10:muscles-and-joints:joint}{sendi} tubuh
sendiri), berlebihan, atau tak berkesudahan: \omterm{prop:g12:innate-immunity:inflammation}{mediator} dan \omterm{def:g12:innate-immunity:phagocytosis}{fagositnya}
lalu merusak jaringan yang seharusnya dilindunginya. Yang menentukan
adalah sasaran dan lamanya, bukan mekanismenya, yang sama pada
keduanya.
\end{solution}

\begin{solution}{pb:g12:innate-immunity:1}
\textbf{1.} Tiga kali melipat dua dalam 2 jam: $\num{10000} \times 8 =
\num{80000}$; enam kali dalam 4 jam: $\num{640000}$.

\textbf{2.} $200 \times 5 = 1000$ per jam --- jauh di bawah
$\num{10000}$ bakteri baru yang ditambahkan pelipatan jam pertama:
pertumbuhannya berlanjut.

\textbf{3.} $\num{50000} \times 20 = 10^6$ bakteri.

\textbf{4.} Pada jam ke-2 bakterinya kurang dari $\num{80000}$;
neutrofil jam pertama saja sanggup menyingkirkan sejuta. \omterm{def:g12:innate-immunity:phagocytosis}{Fagositnya}
menang sejak jam ke-2.

\textbf{5.} Dalam beberapa jam bakterinya lenyap; yang tersisa adalah
neutrofil yang mati, puing dan cairan yang akan menjadi nanahnya,
serta makrofag yang membersihkannya.

\textbf{6.} Kemerahan dan panas: histamin dan prostaglandin melebarkan
pembuluhnya. Bengkak: histamin membuat dindingnya bocor dan plasmanya
merembes keluar. Nyeri: prostaglandin mempekakan ujung sarafnya.

\textbf{7.} Pembuluh yang bocor itulah jalan neutrofil dan \omterm{def:g11:gene-expression:protein}{protein}
pembela plasma mencapai jaringannya; cairan yang menekan saraf yang
dipekakan itulah nyerinya.

\textbf{8.} Tiap detak jantung mendorong lebih banyak darah ke dalam
pembuluh yang melebar; jaringan yang membengkak tak dapat memuai,
sehingga tiap denyutnya menaikkan tekanan pada saraf yang dipekakan:
sebuah denyutan seirama jantung.

\textbf{9.} Neutrofil yang mati, bakteri yang mati dan yang hidup,
puing yang tercerna, dan plasma.

\textbf{10.} Polanya lenyap bersama bakterinya dan puingnya sudah
dibersihkan: tanpa pola, tak ada pengenalan oleh sentinelnya, tak ada
\omterm{prop:g12:innate-immunity:inflammation}{mediator}; pembuluhnya pun kembali normal.

\textbf{11.} Antihistamin mengurangi kemerahan, panas dan kebocoran
dini yang disebabkan histamin; ia tak mengurangi bengkak dan nyeri
belakangan yang disebabkan prostaglandin, dan tak pula perekrutan
neutrofilnya.

\textbf{12.} Ibuprofen menghambat \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzim} pembuat prostaglandin: nyeri,
sebagian bengkaknya, dan demamnya pun turun.

\textbf{13.} Krim itu mematikan sebagian besar \omterm{prop:g12:innate-immunity:inflammation}{mediatornya}, sehingga
neutrofil yang dipanggil lebih sedikit; dengan \omterm{def:g12:innate-immunity:phagocytosis}{fagosit} yang lebih
sedikit bakterinya mungkin tak terbersihkan dan jangkitannya dapat
menyebar.

\textbf{14.} Ketiganya bekerja pada tandanya dan tak satu pun membunuh
bakteri; kortikosteroidnya, dengan memangkas perekrutan \omterm{def:g12:innate-immunity:phagocytosis}{fagosit}, dapat
membuat jangkitannya bertahan lebih lama.

\textbf{15.} Sitokina yang mencapai otak, lalu menaikkan titik acuan
suhunya; panas tambahannya mempercepat \omterm{def:g12:innate-immunity:phagocytosis}{fagosit} dan memperlambat
bakterinya.

\textbf{16.} Sitokina yang diserapnya di lokasi, yang memberi tahu
limfositnya penyusup macam apa yang datang; \omterm{def:g10:cells-common-unit:cell}{sel} itu melapor kepada
limfosit di kelenjar getah bening.

\textbf{17.} Sekitar sepekan sampai sepuluh hari. Untuk serpihan kayu
yang dibersihkan tanggapan bawaan dalam beberapa jam, antibodi itu tak
akan diperlukan --- tetapi antibodi itu tetap dibuat, dan diingat.

\textbf{18.} Tanggapan adaptifnya lebih cepat dan lebih kuat pada kali
kedua berkat ingatannya; tanggapan bawaannya persis sama.

\textbf{19.} \omterm{def:g11:antibiotic-resistance:antibiotic}{Antibiotik} membunuh bakteri, sehingga menggantikan \omterm{def:g12:innate-immunity:phagocytosis}{fagosit}
yang tak ada dalam melawan jangkitan bakteri; \omterm{def:g11:antibiotic-resistance:antibiotic}{antibiotik} tak dapat
menggantikan pembersihan puing oleh \omterm{def:g12:innate-immunity:phagocytosis}{fagosit}, kerjanya terhadap jamur,
atau laporannya ke kelenjar getah bening.

\textbf{20.} Sekitar jam ke-2 \omterm{def:g12:innate-immunity:phagocytosis}{fagositnya} menyalip bakterinya; histamin
dan prostaglandin di balik kemerahan, panas dan bengkaknya, serta
prostaglandin di balik nyerinya; dan \omterm{prop:g12:innate-immunity:handover}{sel dendritik} yang membawa
laporannya.
\end{solution}
