\section*{Chapitre \ref{ch:g9:chromosomes} --- Les chromosomes}

\begin{solution}{exo:g9:chromosomes:1}
Un corps en forme de fil, dans le \omterm{def:g6:cell-unit-of-life:parts}{noyau} --- son contenu
enroulé pour le transport. Visible, bien coloré, quand la
\omterm{def:g5:first-look-at-cells:cell}{cellule} se divise~; déroulé et invisible entre deux
divisions.
\end{solution}

\begin{solution}{exo:g9:chromosomes:2}
46 dans chaque \omterm{def:g5:first-look-at-cells:cell}{cellule} du corps~; rangés en 23 paires~; 23
--- un partenaire par paire --- dans les \omterm{def:g8:sexual-reproduction:gametes}{gamètes}.
\end{solution}

\begin{solution}{exo:g9:chromosomes:3}
$23 + 23 = 46$~: la \omterm{def:g5:flower-to-fruit:steps}{fécondation} rétablit les paires, un
partenaire de chacune venant de la mère, l'autre du père ---
les deux parents dans chaque \omterm{def:g5:first-look-at-cells:cell}{cellule} du corps de l'enfant.
\end{solution}

\begin{solution}{exo:g9:chromosomes:4}
Des \omterm{def:g8:sexual-reproduction:gametes}{gamètes} non réduits doubleraient le jeu à chaque
génération~: des \omterm{def:g8:sexual-reproduction:gametes}{gamètes} à 46 font des enfants à 92, puis
des petits-enfants à 184. La réduction de moitié maintient
l'\omterm{def:g5:biodiversity:species}{espèce} à 46.
\end{solution}

\begin{solution}{exo:g9:chromosomes:5}
XX (le \omterm{def:g9:chromosomes:chromosomes}{caryotype} d'une fille) et XY (celui d'un garçon).
Celui du père~: les \omterm{def:g4:animal-reproduction:sexual}{cellules œufs} portent toutes un X, les
\omterm{def:g8:reproductive-systems:male}{cellules mâles} portent X ou Y moitié-moitié --- c'est celle
qui arrive qui décide.
\end{solution}

\begin{solution}{exo:g9:chromosomes:6}
L'ensemble reconnaissable de \omterm{def:g9:heredity-and-traits:traits}{caractères} appelé trisomie 21.
Cela prouve que les fils portent les consignes des
\omterm{def:g9:heredity-and-traits:traits}{caractères}~: le compte d'un seul fil a changé, beaucoup de
\omterm{def:g9:heredity-and-traits:traits}{caractères} se déplacent.
\end{solution}

\begin{solution}{exo:g9:chromosomes:7}
Dans la division réductrice~: quel partenaire de chaque
paire reçoit un \omterm{def:g8:sexual-reproduction:gametes}{gamète} se règle indépendamment, paire par
paire --- $2^{23}$ demi-jeux possibles par parent, avant
que la \omterm{def:g5:flower-to-fruit:steps}{fécondation} ne multiplie entre eux deux tels espaces.
\end{solution}

\begin{solution}{exo:g9:chromosomes:8}
Des déterminants, pas des \omterm{def:g9:heredity-and-traits:traits}{caractères}~: les fils portent des
consignes --- compactes, transportées par la \omterm{def:g5:first-look-at-cells:cell}{cellule}.
Portés sans être montrés~: les deux partenaires voyagent
dans chaque \omterm{def:g5:first-look-at-cells:cell}{cellule}, quoi qui soit affiché. Des sélections
qui diffèrent d'un enfant à l'autre~: la réduction
distribue un partenaire par paire, indépendamment à chaque
fois.
\end{solution}

\begin{solution}{exo:g9:chromosomes:9}
Chaque \omterm{def:g5:flower-to-fruit:steps}{fécondation} est un nouveau pile ou face~: la
\omterm{def:g8:reproduction-and-environment:population}{population} des \omterm{def:g8:reproductive-systems:male}{cellules mâles} est moitié X, moitié Y à
chaque tentative, et aucune \omterm{def:g2:born-grow-die:cycle}{naissance} passée n'atteint la
course. Trois filles, ce sont trois tirages honnêtes ---
le quatrième l'est tout autant.
\end{solution}

\begin{solution}{exo:g9:chromosomes:10}
(a) Une \omterm{def:g5:first-look-at-cells:cell}{cellule} du corps d'une fille ou d'une femme~: jeu
complet, XX. (b) Une \omterm{def:g8:reproductive-systems:male}{cellule mâle} porteuse du Y~: un
\omterm{def:g8:sexual-reproduction:gametes}{gamète}, un demi-jeu --- sa \omterm{def:g5:flower-to-fruit:steps}{fécondation} ferait un garçon.
(c) Une \omterm{def:g5:first-look-at-cells:cell}{cellule} du corps avec l'exemplaire en trop à la
paire 21~: 47 fils --- le \omterm{def:g9:chromosomes:chromosomes}{caryotype} de la trisomie 21, XX~:
celui d'une fille.
\end{solution}

\begin{solution}{exo:g9:chromosomes:11}
Chaque \omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fournie par les reines portait un X ---
leur apport à la paire 23 n'admet aucune autre possibilité.
La variété X ou Y voyage dans les seuls \omterm{def:g8:sexual-reproduction:gametes}{gamètes} des rois~:
si le registre doit attribuer la pièce du pile ou face,
elle se tenait dans les \omterm{def:g8:reproductive-systems:male}{cellules mâles} royales --- et de
toute façon ce tirage est un hasard, imputable à personne.
\end{solution}

\begin{solution}{exo:g9:chromosomes:12}
L'énigme~: un seul jeu, beaucoup de sortes de \omterm{def:g5:first-look-at-cells:cell}{cellules} ---
si chaque \omterm{def:g5:first-look-at-cells:cell}{cellule} contient les mêmes 46 fils de consignes,
pourquoi un \omterm{def:g8:nervous-communication:neuron}{neurone} n'est-il pas une \omterm{def:g5:first-look-at-cells:cell}{cellule} de \omterm{def:g1:the-five-senses:sense}{peau}~?
Direction de la résolution~: les \omterm{def:g5:first-look-at-cells:cell}{cellules} doivent différer
par les consignes qu'elles \emph{lisent}, non par celles
qu'elles détiennent --- le jeu est une bibliothèque, et la
spécialisation une politique d'emprunt. Ce que sont les
unités lisibles sur un \omterm{def:g9:chromosomes:chromosomes}{chromosome} est la question du
prochain chapitre.
\end{solution}

\begin{solution}{pb:g9:chromosomes:1}
\textbf{1.} A~: une \omterm{def:g5:first-look-at-cells:cell}{cellule} du corps d'un garçon ou d'un
homme --- 46, XY. B~: une \omterm{def:g5:first-look-at-cells:cell}{cellule} du corps d'une fille ou
d'une femme --- 46, XX. C~: un \omterm{def:g8:sexual-reproduction:gametes}{gamète} --- 23, un partenaire
par paire~; son X en fait soit une \omterm{def:g4:animal-reproduction:sexual}{cellule œuf}, soit une
\omterm{def:g8:reproductive-systems:male}{cellule mâle} à X.

\textbf{2.} Un \omterm{def:g8:sexual-reproduction:gametes}{gamète}. Son 23 enregistre la division
réductrice particulière~: à partir d'une \omterm{def:g5:first-look-at-cells:cell}{cellule} à 46, un
partenaire de chaque paire pris, aucune paire complète.

\textbf{3.} 47 fils, l'exemplaire en trop à la paire 21,
23\textsuperscript{e} paire XX~: le \omterm{def:g9:chromosomes:chromosomes}{caryotype} d'une fille porteuse de
la trisomie 21 --- un fil mal compté, un développement
déplacé.

\textbf{4.} Celui du père --- le X ou Y de sa \omterm{def:g8:reproductive-systems:male}{cellule mâle}.
Les chances à la rencontre~: une sur deux, dans chaque sens.

\textbf{5.} Parce que le corps a grandi à partir d'une seule
\omterm{def:g4:animal-reproduction:sexual}{cellule œuf} fécondée par des divisions de recopiage~:
chaque division double le jeu et en distribue un exemplaire
complet à chaque \omterm{def:g5:first-look-at-cells:cell}{cellule} fille --- 46 copies fidèles, de la
\omterm{def:g1:the-five-senses:sense}{peau} au \omterm{def:g8:nervous-communication:neuron}{neurone}.

\textbf{6.} La division réductrice des \omterm{def:g8:sexual-reproduction:gametes}{gamètes}, avec sa
règle~: un partenaire de chacune des 23 paires --- un
demi-jeu complet, aucune paire doublée, aucune oubliée.

\textbf{7.} Que l'enfant en reçoit un X~: quel que soit le
partenaire qui arrive de l'autre côté, l'apport de C à la
paire 23 est fixé --- un \omterm{def:g8:sexual-reproduction:gametes}{gamète} à X écrit X dans la
23\textsuperscript{e} paire de l'enfant sans condition.

\textbf{8.} Chacune des 23 paires fournit l'un ou l'autre
partenaire, indépendamment~: $2 \times 2 \times \dots$, 23
fois --- $2^{23}$ demi-jeux, plusieurs millions, pour un
seul parent. La \omterm{def:g5:flower-to-fruit:steps}{fécondation} apparie une telle distribution
de chaque parent~: les deux espaces se multiplient, et
l'arithmétique du brassage explique pourquoi les frères et
sœurs ne se répètent jamais.

\textbf{9.} Rien n'a été fait de mal~: le 21 en trop naît
d'un accident de la réduction de moitié --- les partenaires
d'une paire qui ne se séparent pas --- une malchance du
mécanisme, et non des actes ni de la valeur des parents. Le
47\textsuperscript{e} fil déplace le développement de l'enfant~; il ne
change rien au fait que cet enfant est entièrement leur
enfant --- la vérité de la consultation et la vérité de la
cytogénétique s'accordent.

\textbf{10.} Une sur deux, exactement comme à chaque
\omterm{def:g2:born-grow-die:cycle}{naissance}~: la \omterm{def:g8:reproduction-and-environment:population}{population} des \omterm{def:g8:reproductive-systems:male}{cellules mâles} est moitié X,
moitié Y à chaque \omterm{def:g5:flower-to-fruit:steps}{fécondation}, et les tirages passés n'y
laissent aucune trace. La série semble porter un sens~; le
mécanisme, lui, ne garde aucune mémoire.

\textbf{11.} La décision est prise à la \omterm{def:g5:flower-to-fruit:steps}{fécondation}, par la
\omterm{def:g8:reproductive-systems:male}{cellule mâle} qui arrive --- à l'intérieur de l'apport du
père, dans les derniers millimètres de la course. Un régime
n'atteint ni la fabrication moitié-moitié de la \omterm{def:g8:reproduction-and-environment:population}{population}
des \omterm{def:g8:reproductive-systems:male}{cellules mâles}, ni l'issue de la course~: l'affirmation
n'a aucun maillon à bloquer ni à infléchir --- la question 1
de \cref{met:g8:contraception:evaluate}, ratée une fois de
plus.

\textbf{12.} L'\omterm{def:g9:heredity-and-traits:heredity}{hérédité} voyage sur des fils dénombrables
dans le \omterm{def:g6:cell-unit-of-life:parts}{noyau}~: réduits de moitié dans les \omterm{def:g8:sexual-reproduction:gametes}{gamètes},
rétablis à la \omterm{def:g5:flower-to-fruit:steps}{fécondation}, déplaçant des \omterm{def:g9:heredity-and-traits:traits}{caractères} quand
ils sont mal comptés, décidant du sexe à la paire 23 ---
quatre photographies, un seul véhicule.
\end{solution}
