\chapter{Pubertas}\label{ch:g8:puberty}

Antara kira-kira sembilan dan enam belas tahun, setiap tubuh
manusia membangun ulang dirinya: ledakan pertumbuhan kedua
pada \cref{rem:g3:stages-of-life:bursts}, dan jauh lebih
banyak daripada tinggi badan. \Cref{rem:g5:human-reproduction:adults}
menjanjikan bab ini bertahun-tahun yang lalu. Inilah dia ---
faktanya, ketidakadilan jadwalnya yang termasyhur, dan
permesinannya, yang akan menyerahkan kepada kita penemuan
yang lebih besar daripada \omterm{def:g8:puberty:puberty}{pubertas} itu sendiri: tubuh punya
sebuah sistem komunikasi kedua.

\section{Apa yang berubah}

\begin{definition}[Pubertas]\label{def:g8:puberty:puberty}
Yang disebut \emph{pubertas}\index{pubertas} adalah tahap
ketika tubuh seorang anak menjadi tubuh seorang dewasa ---
terutama, dewasa dalam pengertian
\cref{ch:g8:sexual-reproduction}: organ reproduksinya matang
lalu mulai menghasilkan \emph{\omterm{def:g8:sexual-reproduction:gametes}{gamet}}, dan tubuhnya mengambil
bentuk dewasanya. Inilah inti biologis masa \omterm{def:g3:stages-of-life:stages}{remaja}
(\cref{def:g3:stages-of-life:stages}).
\end{definition}

\begin{proposition}[Perubahan yang terlihat]\label{prop:g8:puberty:changes}
Di samping ledakan pertumbuhannya, selama beberapa tahun:
\begin{itemize}
  \item pada anak perempuan, umumnya dari kira-kira sembilan
        sampai tiga belas tahun: payudaranya berkembang,
        pinggulnya melebar, rambut tubuh muncul --- dan
        \emph{haid pertamanya} menandai mulai berjalannya
        daur pada \cref{ch:g8:reproductive-systems};
  \item pada anak laki-laki, umumnya setahun atau dua tahun
        kemudian: suaranya memberat (``pecah''), \omterm{def:g1:my-body:joint}{bahunya}
        melebar, rambut tubuh dan wajah muncul, dan buah
        zakarnya mulai menghasilkan \omterm{def:g4:animal-reproduction:sexual}{sel sperma};
  \item pada keduanya: \omterm{def:g1:the-five-senses:sense}{kulitnya} menjadi lebih berminyak
        (usia jerawat), keringatnya berubah, suasana hatinya
        ikut terombang-ambing --- \omterm{def:g5:brain-in-command:system}{otaknya} pun sedang
        dirangkai ulang menuju bentuk dewasanya.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[Jadwalnya bersifat pribadi]\label{rem:g8:puberty:timetable}
\Cref{rem:g1:my-body:different} mengatakannya tentang tinggi
badan; \omterm{def:g8:puberty:puberty}{pubertas} mengulanginya lebih keras. Teman sekelas
yang seusia dapat berdiri berjarak bertahun-tahun di jalan
ini --- awal maupun terlambat sama-sama wajar, urutan
perubahannya berbeda-beda, dan hampir semua orang mendarat
di tempat yang sama pada akhirnya. Rentang di atas itu
lebar, dan yang paling lebar padanya adalah betapa sedikit
yang dikatakannya tentang satu orang mana pun.
\end{remark}

\section{Permesinannya: sistem pengantar pesan kedua}

\begin{definition}[Hormon, perjumpaan pertama]\label{def:g8:puberty:hormones}
\omterm{def:g8:puberty:puberty}{Pubertas} tidak diputuskan di dalam organ yang diubahnya. Ia
\emph{diperintahkan} --- oleh pengantar pesan kimiawi yang
disebut \emph{hormon}\index{hormon}: zat yang dilepaskan ke
dalam \emph{\omterm{prop:g4:journey-of-food:absorb}{darah}} oleh organ khusus, diangkut ke
mana-mana, lalu dipatuhi oleh organ yang dibangun untuk
mendengarnya. Pada masa \omterm{def:g8:puberty:puberty}{pubertas}, sebuah pusat kendali di
dasar \omterm{def:g5:brain-in-command:system}{otak} terbangun, hormonnya memerintah organ
reproduksinya, dan hormon \emph{organ itu} --- diangkut ke
seluruh tubuh oleh jasa pengantaran pada
\cref{prop:g4:heart-and-blood:carries} --- memerintahkan
seluruh pembangunan ulangnya: pertumbuhan, rambut, suara,
daur.
\end{definition}

\begin{example}[Satu isyarat, banyak lokasi pembangunan]\label{ex:g8:puberty:sites}
\omterm{prop:g4:journey-of-food:absorb}{Darah} mengangkut sebuah \omterm{def:g8:puberty:hormones}{hormon} ke mana-mana sekaligus ---
itulah sebabnya \omterm{def:g8:puberty:puberty}{pubertas} adalah pembangunan ulang yang
\emph{terpadu}: suara, \omterm{def:g1:my-body:joint}{bahu} dan rambut mengikuti pengantar
pesan yang sama, pada satu jadwal, walaupun lokasinya
terletak berjauhan setengah tubuh. Tidak ada kabel \omterm{def:g5:brain-in-command:system}{saraf}
yang menghubungkan kotak suara dengan janggut; siaran aliran
darahnyalah yang menghubungkannya
(\cref{ch:g8:hormonal-communication} menjadikan sistem ini
sebuah bab tersendiri).
\end{example}

\begin{remark}[Mengapa cuaca suasana hatinya?]\label{rem:g8:puberty:moods}
Pengantar pesan yang sama sampai ke \omterm{def:g5:brain-in-command:system}{otak}, yang dirinya
sendiri sedang di tengah perombakan: suasana hati yang
berayun, ketajaman rasa yang baru, \omterm{prop:g1:healthy-habits:needs}{tidur} yang hanyut makin
malam --- cuaca, bukan kesalahan. Mengetahui permesinannya
menolong di kedua sisinya: badainya adalah kimia ditambah
pembangunan, ia wajar, dan ia berlalu.
\end{remark}

\section{Menjalani pembangunan ulangnya}

\begin{proposition}[Perawatan selama pembangunan]\label{prop:g8:puberty:care}
Tubuh yang sedang dibangun ulang punya kebutuhan seorang
pembangun, semuanya sudah dipenuhi sebelumnya di buku ini:
\begin{itemize}
  \item bahan dan bahan bakar: ledakannya dibayar dengan
        materi --- makan yang seimbang
        (\cref{met:g3:balanced-diet:check}), termasuk materi
        pembangunnya
        (\cref{exo:g7:digestion-and-nutrients:11});
  \item \omterm{prop:g1:healthy-habits:needs}{tidur}: perombakan \omterm{def:g5:brain-in-command:system}{otaknya} dan pertumbuhan tubuhnya
        sama-sama berjalan pada malam hari
        (\cref{prop:g5:brain-in-command:protect}) --- dan
        jam yang hanyut makin malam itu tetap butuh jamnya;
  \item gerak: \omterm{def:g3:bones-and-muscles:skeleton}{tulang} dan \omterm{def:g4:heart-and-blood:heart}{jantungnya} sedang berada pada
        tahun pembangunannya yang terakhir dan terbaik
        (\cref{met:g3:bones-and-muscles:care});
  \item dan tanpa sabotase: zat pada
        \cref{prop:g5:healthy-choices:substances} memukul
        paling keras tubuh dan \omterm{def:g5:brain-in-command:system}{otak} yang masih dalam
        pembangunan.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Pertanyaan punya alamat]\label{ex:g8:puberty:questions}
\omterm{def:g8:puberty:puberty}{Pubertas} menimbulkan pertanyaan --- tentang waktunya,
tentang tubuh, tentang perasaan --- dan pertanyaan itu punya
alamat yang tepat: perawat sekolah dan dokter menjawabnya
setiap hari, dengan rahasia terjaga dan tanpa terkejut;
orang dewasa yang dipercaya semuanya pernah melewatinya; dan
bab buku ini dibangun untuk dibaca ulang. Yang tidak layak
dipercaya: dongeng halaman sekolah dan sudut terburuk
internet, yang berdagang justru atas ketidakpastian usia ini.
\end{example}

\section{Soal: Konferensi Grafik Pertumbuhan}

\begin{problem}[{Soal akhir pekan --- empat grafik
pertumbuhan (rekaan), dibaca bersama seluruh bab}]\label{pb:g8:puberty:1}
Seorang dokter sekolah meninjau empat grafik, tinggi badan
terhadap usia, masing-masing dengan tanggal pengukurannya:
L, yang kurvanya melonjak curam dari sebelas ke tiga belas
tahun lalu kini mendatar pada usia lima belas; M, mendaki
datar pada usia tiga belas, ledakannya belum dimulai; N, di
tengah ledakannya pada usia empat belas; O, yang ruas
curamnya berjalan dari sembilan ke dua belas lalu mendatar.

\medskip\noindent\textbf{Bagian I --- Membaca kurvanya.}
\begin{enumerate}
  \item Tempatkan tiap grafiknya pada peta
        \cref{rem:g3:stages-of-life:bursts}: ledakan mana
        yang sedang tampak, dan di sebelah mana tiap murid
        berdiri padanya?
  \item M mencemaskan dirinya ``tertinggal''. Susunlah
        jawaban dokternya dari
        \cref{rem:g8:puberty:timetable}, beserta ramalan
        yang didukung oleh pendakian datar grafiknya.
  \item O bertanya apakah dataran itu berarti ``selesai
        tumbuh pada usia dua belas''. Betulkan bacaannya:
        apa yang diakhiri dataran itu, dan apa yang lanjut?
  \item Mengapa dokternya membaca \emph{kemiringannya},
        bukan tingginya, untuk menilai posisi tiap murid?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Di balik kurvanya.}
\begin{enumerate}[resume]
  \item Sebutkan permesinan yang memulai tiap ledakannya,
        dan rute yang ditempuh perintahnya.
  \item Dokternya juga melacak \omterm{prop:g1:healthy-habits:needs}{tidur} dan makanannya pada
        kunjungan ini. Kaitkan masing-masing dengan barisnya
        pada \cref{prop:g8:puberty:care}.
  \item N melaporkan badai suasana hati dan waktu \omterm{prop:g1:healthy-habits:needs}{tidur} yang
        hanyut makin malam. Berikan penjelasan dua bagian
        pada \cref{rem:g8:puberty:moods} --- dan akibat
        praktisnya bagi \omterm{prop:g1:healthy-habits:needs}{tidur} N pada pekan sekolahnya.
  \item L bertanya kepada dokternya secara pribadi tentang
        sebuah perubahan yang tak dapat diperlihatkan
        grafiknya. Nyatakan, tanpa mereka-reka pertanyaannya,
        mengapa ruang periksanya adalah alamat yang tepat
        (\cref{ex:g8:puberty:questions}).
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Catatan
konferensinya.}
\begin{enumerate}[resume]
  \item Tuliskan catatan satu barisnya bagi tiap grafik:
        tahapnya, kewajarannya, dan pokok perawatan apa pun.
  \item Keempat grafiknya berbeda bertahun-tahun, namun
        dokternya mengarsipkan keempatnya sebagai ``wajar''.
        Berikan alasannya dalam satu kalimat.
  \item Seorang orang tua meminta ``sesuatu untuk
        mempercepat M''. Jawablah dengan catatan jadwalnya
        dan permesinannya: apa yang ada untuk diperbaiki?
  \item Tutuplah dengan ringkasan satu kalimat babnya untuk
        papan pengumuman kelas: apa itu \omterm{def:g8:puberty:puberty}{pubertas}, siapa yang
        menjalankannya, dan apa yang dibutuhkannya.
\end{enumerate}
\end{problem}
