\chapter{Het afval afvoeren}\label{ch:g7:removing-wastes}

Leveringen hebben dit jaar overheerst: \omterm{def:g7:what-is-respiration:respiration}{zuurstof} erin, \omterm{def:g7:digestion-and-nutrients:families}{voedingsstoffen}
erin, brandstof naar elke werkende \omterm{def:g5:first-look-at-cells:cell}{cel}. Maar elke werkplaats maakt afval,
en een bezorgdienst die het vuilnis nooit ophaalt vergiftigt al snel haar
eigen stad. De rekening van de \omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} is al vereffend --- die
wordt bij de \omterm{def:g7:lungs-and-gas-exchange:alveoli}{longblaasjes} uitgeademd. Dit slothoofdstuk volgt het
\emph{andere} afval, naar de discreetste organen van het lichaam: de
\omterm{def:g7:removing-wastes:kidneys}{nieren}.

\section{Het afval voorbij de koolstofdioxide}

\begin{proposition}[Het overige vuilnis van de cellen]\label{prop:g7:removing-wastes:rubbish}
Het verbranden en bouwen van de cellen
(\cref{prop:g7:what-is-respiration:power},
\cref{prop:g6:matter-from-food:transform}) leveren meer op dan
\omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide}. Vooral het gebruik van \omterm{def:g7:digestion-and-nutrients:families}{voedingsstoffen} uit de bouwstof
laat een stikstofhoudend afval achter --- \emph{ureum}\index{ureum} ---
dat het \omterm{prop:g4:journey-of-food:absorb}{bloed} uit elk weefsel ophaalt, precies zoals het \omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide}
ophaalt. \omterm{prop:g7:removing-wastes:rubbish}{Ureum} is onschadelijk in het passerende verkeer en giftig als het
zich opstapelt: het moet er onophoudelijk uit.
\end{proposition}
\admitted

\begin{example}[Twee ophaalrondes, één bloed]\label{ex:g7:removing-wastes:rounds}
Dezelfde druppel uit \cref{ex:g7:heart-and-circulation:tour} draait beide
ophaaldiensten tegelijk: bij de \omterm{def:g7:heart-and-circulation:vessels}{haarvaten} van de dij neemt hij
\omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} \emph{en} \omterm{prop:g7:removing-wastes:rubbish}{ureum} aan boord. Het eerste zet hij bij de
\omterm{def:g7:how-animals-breathe:four}{longen} af, het volgende circuit al. Het tweede rijdt door --- de \omterm{def:g7:how-animals-breathe:four}{longen}
voorbij, die het niet kunnen aanraken --- tot de lichaamscirculatie het
door de organen voert die precies voor deze vracht gebouwd zijn.
\end{example}

\section{De nieren, de filters van het bloed}

\begin{definition}[Nieren en het urinestelsel]\label{def:g7:removing-wastes:kidneys}
De twee \emph{nieren}\index{nier} --- zo groot als een vuist, boonvormig,
laag in de rug --- zijn filters die het \omterm{prop:g4:journey-of-food:absorb}{bloed} schoonmaken. Elke nier
krijgt een royale \omterm{def:g7:heart-and-circulation:vessels}{slagader} rechtstreeks van de hoofdlijn van de
lichaamscirculatie. Binnenin wordt het \omterm{prop:g4:journey-of-food:absorb}{bloed} door ongeveer een miljoen
microscopische filtereenheden gefilterd; het schoongemaakte \omterm{prop:g4:journey-of-food:absorb}{bloed} keert
terug via de nierader, en de verwijderde vracht --- \omterm{prop:g7:removing-wastes:rubbish}{ureum}, overtollig
water, overtollige zouten --- vertrekt als \emph{urine}\index{urine}, die
door twee buisjes omlaag naar de \emph{blaas}\index{blaas} druppelt, de
opslagtank die naar believen geleegd wordt.
\end{definition}

\begin{proposition}[Filteren, en dan terugwinnen]\label{prop:g7:removing-wastes:filter}
De truc van de \omterm{def:g7:removing-wastes:kidneys}{nier} is vrijgevigheid gevolgd door zuinigheid. De
filtereenheden duwen eerst een enorme hoeveelheid van het waterige deel
van het \omterm{prop:g4:journey-of-food:absorb}{bloed} --- vele tientallen liters per dag --- door hun fijne zeven,
met \omterm{prop:g7:removing-wastes:rubbish}{ureum}, suikers, zouten en al. Daarna, langs het kronkelende buisje van
elke eenheid, \emph{wint het lichaam terug} wat het wil houden:
vrijwel al het water, alle \omterm{prop:g7:organs-at-work:bill}{glucose}, en zouten op maat --- en het laat het
\omterm{prop:g7:removing-wastes:rubbish}{ureum} en de overschotten achter. Van de tientallen liters die gezeefd
worden, vertrekt ongeveer anderhalve liter als \omterm{def:g7:removing-wastes:kidneys}{urine}: filter alles, neem
het goede terug, en het afval heeft zich nergens meer te verstoppen.
\end{proposition}
\admitted

\begin{example}[De zuinigheid aflezen]\label{ex:g7:removing-wastes:thrift}
De trap van het terugwinnen verklaart de alledaagse waarnemingen. Drink
veel, en de \omterm{def:g7:removing-wastes:kidneys}{nieren} winnen minder water terug: veel bleke \omterm{def:g7:removing-wastes:kidneys}{urine}. \omterm{rem:g7:removing-wastes:sweat}{Zweet} een
hete wedstrijd door, en ze winnen bijna alles terug: weinig donkere \omterm{def:g7:removing-wastes:kidneys}{urine}
--- het lichaam bewaakt zijn water
(\cref{prop:g4:what-plants-need:needs} leerde de waarde van water voor
\omterm{def:g1:plants-around-us:plant}{planten}; de \omterm{def:g7:removing-wastes:kidneys}{nier} is de dierlijke kant van dezelfde huishouding). En
\omterm{prop:g7:organs-at-work:bill}{glucose} in de \omterm{def:g7:removing-wastes:kidneys}{urine}, die de zeef doorlaat maar die het terugwinnen
volledig zou moeten ophalen, is een klassiek medisch signaal dat er iets
--- het terugwinnen of de bloedsuiker --- niet in orde is.
\end{example}

\begin{remark}[De huid steekt een handje toe]\label{rem:g7:removing-wastes:sweat}
\emph{Zweet}\index{zweet} is een nevenactiviteit van dezelfde
onderneming: water en zouten (met sporen \omterm{prop:g7:removing-wastes:rubbish}{ureum}) die aan de \omterm{def:g1:the-five-senses:sense}{huid} afgegeven
worden --- al is de eerste taak ervan het koelen uit
\cref{ex:g7:organs-at-work:felt}. De \omterm{def:g7:how-animals-breathe:four}{longen}, de \omterm{def:g7:removing-wastes:kidneys}{nieren} en de \omterm{def:g1:the-five-senses:sense}{huid} verdelen
de volledige afvalverwijdering van het lichaam: \omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} naar de
eerste, \omterm{prop:g7:removing-wastes:rubbish}{ureum} en de overschotten naar de tweede, warmte en een zilt
straaltje naar de derde. Er blijft niets giftigs liggen om zich op te
stapelen --- in een gezond lichaam wint het vuilnis nooit.
\end{remark}

\section{De machine van het jaar, compleet}

\begin{proposition}[De volledige lus van aanvoer en afvoer]\label{prop:g7:removing-wastes:complete}
Met de \omterm{def:g7:removing-wastes:kidneys}{nieren} erbij sluit de rondgang van dit jaar zich tot één machine.
Het \omterm{prop:g4:journey-of-food:absorb}{bloed}, dat door het dubbele circuit gepompt wordt
(\cref{def:g7:heart-and-circulation:double}), laadt \omterm{def:g7:what-is-respiration:respiration}{zuurstof} bij de
\omterm{def:g7:lungs-and-gas-exchange:alveoli}{longblaasjes} en \omterm{def:g7:digestion-and-nutrients:families}{voedingsstoffen} bij de \omterm{prop:g7:digestion-and-nutrients:absorption}{darmvlokken}; levert allebei aan
elke werkende \omterm{def:g5:first-look-at-cells:cell}{cel} om te verbranden en te bouwen; en haalt het afval op ---
\omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} naar de \omterm{def:g7:how-animals-breathe:four}{longen}, \omterm{prop:g7:removing-wastes:rubbish}{ureum} en de overschotten naar de \omterm{def:g7:removing-wastes:kidneys}{nieren}.
Elk orgaan uit leerjaar vier en zeven is een station van die ene lus; elke
uitwisseling steekt een haarvatwand over; en het geheel draait op het
trage vuur uit \cref{prop:g7:what-is-respiration:power}, \omterm{def:g5:first-look-at-cells:cell}{cel} voor \omterm{def:g5:first-look-at-cells:cell}{cel}.
\end{proposition}
\admitted

\begin{method}[De doorlichting van het hele lichaam]\label{met:g7:removing-wastes:audit}
Licht voor elke stof haar doorgang door:
\begin{enumerate}
  \item binnenkomst: \omterm{def:g4:journey-of-food:tract}{mond} en \omterm{prop:g7:digestion-and-nutrients:absorption}{darmvlokken}, of \omterm{def:g1:the-five-senses:sense}{neus} en \omterm{def:g7:lungs-and-gas-exchange:alveoli}{longblaasjes};
  \item vervoer: welke stukken van de circuits, welke bijzondere haltes
        (de \omterm{prop:g7:digestion-and-nutrients:absorption}{lever});
  \item gebruik: als brandstof verbrand, als materiaal ingebouwd, of geen
        van beide;
  \item vertrek: \omterm{def:g7:lungs-and-gas-exchange:alveoli}{longblaasjes} (gas), \omterm{def:g7:removing-wastes:kidneys}{nieren} (\omterm{prop:g7:removing-wastes:rubbish}{ureum}, overschotten), \omterm{def:g1:the-five-senses:sense}{huid}
        (het aandeel van het zweet) --- of, als het nooit opgenomen is, de
        route van de restanten
        (\cref{prop:g4:journey-of-food:absorb}).
\end{enumerate}
Laat haar lopen over water, over \omterm{prop:g7:organs-at-work:bill}{glucose}, over een zoute chip: de machine
heeft voor elke stap een station, en dat is de bevinding van het jaar in
één oefening.
\end{method}

\section{Probleem: De dag van een watermolecuul}

\begin{problem}[{Weekendprobleem --- één glas water, door de hele machine
doorgelicht}]\label{pb:g7:removing-wastes:1}
Volg een glas water dat op een sportdag bij het ontbijt gedronken wordt,
als de grote doorlichting van het jaar.

\medskip\noindent\textbf{Deel I --- Binnenkomst en vervoer.}
\begin{enumerate}
  \item Water hoeft niet uit elkaar gehaald te worden
        (\cref{def:g7:digestion-and-nutrients:families}). Noem zijn
        oversteek bij de binnenkomst en het vaatnetwerk dat het opvangt.
  \item Zijn \omterm{prop:g4:journey-of-food:absorb}{bloed} doet de gebruikelijke eerste halte aan. Welk orgaan, en
        waarom is die route verstandig voor \omterm{def:g7:digestion-and-nutrients:families}{voedingsstoffen}, ook al hoeft
        het water zelf niet gesorteerd te worden?
  \item Halverwege de ochtend is het water overal waar het \omterm{prop:g4:journey-of-food:absorb}{bloed} komt.
        Noem de twee circuits waarin het nu afwisselend meerijdt.
  \item Een deel ervan zit al snel in werkende cellen. Noem twee taken die
        water daar doet (denk eraan dat de scheikunde van de cellen ---
        sappen, verbranden --- nat verloopt).
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De wedstrijd van de
middag.}
\begin{enumerate}[resume]
  \item Tijdens de wedstrijd verlaat een deel van het water het lichaam
        via een grens die in de eerste plaats geen uitscheidingsorgaan is.
        Noem haar, haar vracht naast water, en haar eigenlijke taak op die
        dag.
  \item De wedstrijd is voorbij, de speelster heeft dorst en haar \omterm{def:g7:removing-wastes:kidneys}{urine} is
        die avond schaars en donker. Verklaar allebei met de trap van het
        terugwinnen.
  \item Ondertussen maakten de hardwerkende \omterm{def:g3:bones-and-muscles:muscle}{spieren} van de wedstrijd extra
        \omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} en \omterm{prop:g7:removing-wastes:rubbish}{ureum}. Wijs elk zijn uitgangsgrens toe, en de
        stukken van het circuit die het daarheen brachten.
  \item Waarom werkten de \omterm{def:g7:removing-wastes:kidneys}{nieren} de hele wedstrijd door, hoewel de
        urineproductie daalde? Wat verwijderden ze nog steeds?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De doorlichting afgesloten.}
\begin{enumerate}[resume]
  \item De volgende ochtend is een deel van het glas met de adem
        vertrokken. Leg uit hoe water bij de \omterm{def:g7:lungs-and-gas-exchange:alveoli}{longblaasjes} naar buiten gaat
        (\cref{ex:g4:breathing:mirror} liet het jaren geleden al zien).
  \item Stel de volledige boekhouding van het glas op: binnenkomst,
        voorraden en gebruik, en zijn vier uitgangen --- \omterm{def:g7:removing-wastes:kidneys}{urine}, zweet,
        adem, en het kleine aandeel in de restanten.
  \item De doorlichting laat zien dat één \omterm{prop:g4:journey-of-food:absorb}{bloed} achtereenvolgens \omterm{def:g7:how-animals-breathe:four}{longen},
        \omterm{prop:g7:digestion-and-nutrients:absorption}{darmvlokken}, \omterm{def:g3:bones-and-muscles:muscle}{spieren}, \omterm{def:g7:removing-wastes:kidneys}{nieren} en \omterm{def:g1:the-five-senses:sense}{huid} bedient. Zeg in twee zinnen
        waarom dit ontwerp met één netwerk de gezondheid van elk orgaan de
        zaak van elk ander orgaan maakt.
  \item Sluit het jaar af: welk enkele idee --- noem het --- bleek elk
        station van deze doorlichting te dienen?
        (\cref{prop:g7:what-is-respiration:power} heeft het antwoord.)
\end{enumerate}
\end{problem}
