\section*{Chapitre \ref{ch:g7:what-is-respiration} --- Qu'est-ce que la respiration~?}

\begin{solution}{exo:g7:what-is-respiration:1}
La \omterm{def:g7:what-is-respiration:respiration}{respiration} est l'échange gazeux par lequel un \omterm{def:g6:exploring-our-environment:components}{être vivant} prend
du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} dans son milieu et rejette du \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone}. La
\omterm{def:g7:what-is-respiration:respiration}{ventilation} n'est que le pompage visible de la poitrine --- une
façon de servir cet échange~; l'échange lui-même a lieu dans chaque
\omterm{def:g5:first-look-at-cells:cell}{cellule} vivante.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:2}
Une sonde à \omterm{def:g7:what-is-respiration:respiration}{dioxygène} dans un récipient fermé --- elle montre le
niveau de \omterm{def:g7:what-is-respiration:respiration}{dioxygène} qui baisse~; et l'\omterm{prop:g7:what-is-respiration:shown}{eau de chaux} --- un liquide
clair qui devient laiteux quand du \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} y passe.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:3}
Pour prouver que le changement vient du sujet vivant et non du
récipient, de la chaleur du jour ou de l'\omterm{prop:g7:what-is-respiration:shown}{eau de chaux} elle-même~: le
jumeau de l'expérience honnête, la vie étant la seule différence.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:4}
L'\omterm{prop:g3:balanced-diet:jobs}{énergie}~: dans chaque \omterm{def:g5:first-look-at-cells:cell}{cellule}, les \omterm{def:g4:journey-of-food:digestion}{nutriments} sont lentement
brûlés avec le \omterm{def:g7:what-is-respiration:respiration}{dioxygène}, ce qui libère leur \omterm{prop:g3:balanced-diet:jobs}{énergie} pour le travail
de la \omterm{def:g5:first-look-at-cells:cell}{cellule}. Le \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} est le reste consumé, évacué.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:5}
Une souris, un cloporte, des pois en \omterm{def:g2:from-seed-to-plant:germination}{germination}, un \omterm{def:g6:decomposers-and-soil:decomposers}{champignon}, des
limnées --- et une cuillerée de terre vivante.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:6}
Il faut la tester dans le noir~: la fabrication mue par la \omterm{def:g6:exploring-our-environment:conditions}{lumière}
étant arrêtée, le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} du bocal baisse et l'\omterm{prop:g7:what-is-respiration:shown}{eau de chaux} se
trouble --- la \omterm{def:g7:what-is-respiration:respiration}{respiration} démasquée.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:7}
Ressemblances~: toutes deux consomment du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} et rejettent du
\omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone}~; toutes deux libèrent l'\omterm{prop:g3:balanced-diet:jobs}{énergie} du combustible.
Différences~: la combustion de la \omterm{def:g5:first-look-at-cells:cell}{cellule} est lente et sans flamme,
en des milliers de petites étapes~; et son \omterm{prop:g3:balanced-diet:jobs}{énergie} est captée pour
le travail au lieu d'être perdue d'un coup en \omterm{def:g6:exploring-our-environment:conditions}{lumière} et en chaleur.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:8}
Enfant qui sprinte (travail musculaire intense) $>$ enfant qui lit
(l'abonnement tranquille du corps) $>$ pois en \omterm{def:g2:from-seed-to-plant:germination}{germination} (qui
bâtit vite, mais qui est petit) $>$ pois sec (dormant --- la
\omterm{def:g7:what-is-respiration:respiration}{respiration} presque au point mort). Le rythme suit le travail.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:9}
Toute la nuit, la \omterm{def:g7:what-is-respiration:respiration}{respiration} des pois qui germent a consommé le
\omterm{def:g7:what-is-respiration:respiration}{dioxygène} du bocal et rejeté du \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone}~; au matin, il
reste trop peu de \omterm{def:g7:what-is-respiration:respiration}{dioxygène} pour entretenir une flamme. L'\omterm{prop:g7:what-is-respiration:shown}{eau de
chaux} se serait troublée régulièrement pendant la nuit.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:10}
Deux récipients avec sondes et pièges à \omterm{prop:g7:what-is-respiration:shown}{eau de chaux}~: l'un avec de
la terre vivante fraîche, l'autre avec la même terre cuite et rendue
sans vie --- la seule différence. Si le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} du bocal vivant
baisse et que son \omterm{prop:g7:what-is-respiration:shown}{eau de chaux} se trouble pendant que le jumeau cuit
ne change pas, c'est la vie de la terre qui l'a fait.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:11}
Le grain humide se réveille~: la \omterm{def:g2:from-seed-to-plant:germination}{germination} commence, et la
\omterm{def:g7:what-is-respiration:respiration}{respiration} passe du murmure au rythme du pois qui germe --- des
milliards de grains brûlant leurs réserves en même temps, ce qui
chauffe le tas comme les équipes au travail du compost. La chaleur
accélère encore le travail~: un tas humide peut se chauffer
lui-même jusqu'à s'avarier. Le grain sec dort au froid~; le grain
humide est un four lent.
\end{solution}

\begin{solution}{exo:g7:what-is-respiration:12}
Comportement~: la \omterm{def:g7:what-is-respiration:respiration}{ventilation} s'observe --- les mouvements de la
poitrine, quinze fois par minute
(\cref{ex:g4:breathing:watch}). Chimie~: la \omterm{def:g7:what-is-respiration:respiration}{respiration} a lieu chez
des êtres qui n'ont pas de poitrine du tout --- pois, \omterm{def:g6:decomposers-and-soil:decomposers}{champignons},
terre --- et n'est détectée que par des tests de gaz. Service~: la
\omterm{def:g7:what-is-respiration:respiration}{ventilation} existe pour renouveler l'air des \omterm{def:g4:breathing:organs}{poumons} afin que
l'échange gazeux puisse continuer --- elle s'emballe exactement
quand la combustion des \omterm{def:g5:first-look-at-cells:cell}{cellules} s'emballe
(\cref{ex:g4:breathing:running}).
\end{solution}

\begin{solution}{pb:g7:what-is-respiration:1}
\textbf{1.} Aucun changement~: \omterm{def:g7:what-is-respiration:respiration}{dioxygène} stable, \omterm{prop:g7:what-is-respiration:shown}{eau de chaux}
limpide. A est le témoin sans vie --- l'immobilité à laquelle on
compare tous les autres bocaux.

\textbf{2.} B~: un changement à peine mesurable --- les pois secs
sont dormants, la \omterm{def:g7:what-is-respiration:respiration}{respiration} au ralenti. C~: une baisse nette du
\omterm{def:g7:what-is-respiration:respiration}{dioxygène} et un trouble --- la \omterm{def:g2:from-seed-to-plant:germination}{germination} est une construction à
plein régime, et le travail se paie en \omterm{def:g7:what-is-respiration:respiration}{respiration}.

\textbf{3.} D~: la baisse et le trouble les plus rapides de tous ---
un mammifère chaud et remuant brûle fort. La durée est limitée parce
que la souris consomme le \omterm{def:g7:what-is-respiration:respiration}{dioxygène} de son bocal~; l'expérience doit
s'arrêter bien avant que l'air ne vienne à manquer.

\textbf{4.} E~: \omterm{def:g7:what-is-respiration:respiration}{dioxygène} stable ou en hausse --- la fabrication de
la \omterm{def:g1:plants-around-us:plant}{plante} éclairée absorbe du \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} et rejette plus de
\omterm{def:g7:what-is-respiration:respiration}{dioxygène} que sa \omterm{def:g7:what-is-respiration:respiration}{respiration} n'en consomme. F~: \omterm{def:g7:what-is-respiration:respiration}{dioxygène} en baisse,
\omterm{prop:g7:what-is-respiration:shown}{eau de chaux} qui se trouble --- dans le noir, la fabrication
s'arrête et la \omterm{def:g7:what-is-respiration:respiration}{respiration}, jamais arrêtée, se montre à découvert.

\textbf{5.} F montre que la \omterm{def:g1:plants-around-us:plant}{plante} respire toujours --- \omterm{def:g7:what-is-respiration:respiration}{dioxygène}
absorbé, \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} rejeté, comme la souris. E cache cette
même \omterm{def:g7:what-is-respiration:respiration}{respiration} sous le plus gros échange du jour~: les entrées et
les sorties de la fabrication l'engloutissent. Ensemble, les jumeaux
séparent les deux échanges.

\textbf{6.} Le pois en \omterm{def:g2:from-seed-to-plant:germination}{germination} travaille trois fois plus fort
par gramme --- il bâtit \omterm{def:g1:plants-around-us:parts}{racines} et \omterm{def:g1:plants-around-us:parts}{tige} à plein régime --- et la
\omterm{def:g7:what-is-respiration:respiration}{respiration} est le bruit de moteur de ce travail.

\textbf{7.} Les \omterm{def:g3:bones-and-muscles:muscle}{muscles} et le corps chaud de la souris brûlent des
\omterm{def:g4:journey-of-food:digestion}{nutriments} avec du \omterm{def:g7:what-is-respiration:respiration}{dioxygène}~; cette combustion rejette du \omterm{def:g7:what-is-respiration:respiration}{dioxyde
de carbone} dans l'air du bocal~; l'air prélevé traverse l'\omterm{prop:g7:what-is-respiration:shown}{eau de
chaux}~; le \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone} la rend laiteuse.

\textbf{8.} «~Tout \omterm{def:g6:exploring-our-environment:components}{être vivant} respire --- \omterm{def:g1:animals-around-us:animal}{animal}, \omterm{def:g1:plants-around-us:plant}{plante}, \omterm{def:g2:from-seed-to-plant:seed}{graine}
éveillée ou à peine endormie --- en prenant du \omterm{def:g7:what-is-respiration:respiration}{dioxygène} et en
rejetant du \omterm{def:g7:what-is-respiration:respiration}{dioxyde de carbone}.~»

\textbf{9.} \omterm{def:g7:what-is-respiration:respiration}{Dioxygène} en baisse, \omterm{prop:g7:what-is-respiration:shown}{eau de chaux} qui se trouble --- le
\omterm{def:g6:decomposers-and-soil:decomposers}{champignon} respire. Cela ajoute que la règle vaut aussi pour les
\omterm{def:g6:decomposers-and-soil:decomposers}{champignons}~: ni \omterm{def:g1:plants-around-us:plant}{plante} ni \omterm{def:g1:animals-around-us:animal}{animal}, et aucune exception.

\textbf{10.} Parce qu'un silo est le bocal C à l'échelle d'un
entrepôt~: du grain gardé humide se met à \omterm{def:g2:from-seed-to-plant:germination}{germer} et à respirer fort,
ce qui chauffe et avarie le stock --- les règles de sécheresse du
responsable sont un contrôle de la \omterm{def:g7:what-is-respiration:respiration}{respiration}.

\textbf{11.} Tout ce qu'il contient~: les limnées et les petits
\omterm{def:g1:animals-around-us:animal}{animaux}, les \omterm{def:g6:decomposers-and-soil:decomposers}{décomposeurs} de la vase, les \omterm{def:g1:plants-around-us:plant}{plantes} et les \omterm{def:g5:first-look-at-cells:cell}{cellules}
vertes uniques pendant leurs heures sombres. Cela compte parce que
tous les habitants de la mare puisent dans un seul petit stock de
\omterm{def:g7:what-is-respiration:respiration}{dioxygène} dissous --- l'économie qu'ouvrent les chapitres suivants.

\textbf{12.} Seul D --- la souris --- a montré la \omterm{def:g7:what-is-respiration:respiration}{ventilation}~: des
mouvements visibles de la poitrine au service de son échange gazeux.
Tous les bocaux contenant de la vie ont montré la \omterm{def:g7:what-is-respiration:respiration}{respiration}~: B
faiblement, C, D et F nettement, E masquée par la fabrication
diurne mais respirant tout de même.
\end{solution}
