\chapter{الخلية: وحدة الحياة}\label{ch:b1:cell-unit-of-life}

في عام 1665 وضع روبرت هوك شريحة من الفلين تحت عدسة فرآها مقسومة إلى
علب صغيرة سماها \omterm{def:b1:cell-unit-of-life:cell}{خلايا}، على اسم غرف الدير. وبعد عشر سنين رأى تاجر
أقمشة هولندي، أنطوني فان ليفنهوك، وقد صقل عدساته بقوة لم يبلغها
سواه، \omterm{def:b1:organism-environment:organism}{كائنات} حية تسبح في قطرة من ماء بركة، وفي كشاطة أسنانه، وفي
منيّه. ثم لزم قرن ونصف آخر ليُقال ما رآه الرجلان: أن كل \omterm{def:b1:organism-environment:organism}{كائن حي}
مكون من \omterm{def:b1:cell-unit-of-life:cell}{خلايا}، وأن كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} تأتي من \omterm{def:b1:cell-unit-of-life:cell}{خلية}. يعرض هذا الفصل \omterm{prop:b1:cell-unit-of-life:theory}{النظرية
الخلوية} وشواهدها، ويقابل بين نوعي \omterm{def:b1:cell-unit-of-life:cell}{الخلية}، ويصف الأدوات --- المجاهر،
والنابذة، ودورق الزرع --- التي تتيح لنا أن نرى \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}، وأن نفككها،
وأن نبقيها حية خارج \omterm{def:b1:organism-environment:organism}{الكائن الحي}.

\section{النظرية الخلوية}

\begin{definition}[الخلية]\label{def:b1:cell-unit-of-life:cell}
\emph{الخلية}\index{خلية} أصغر وحدة من المادة الحية: حجم من
\emph{السيتوبلازم}\index{سيتوبلازم} المائي يحده \emph{غشاء
بلازمي}\index{غشاء بلازمي}، يحوي المادة الوراثية (DNA) والآلة التي
تعبر عنها، قادر على أخذ المغذيات، وتحويل الطاقة، وصون تنظيمه الخاص،
والانقسام إلى خليتين حين تسمح الظروف.
\end{definition}

\begin{proposition}[النظرية الخلوية]\label{prop:b1:cell-unit-of-life:theory}
\index{النظرية الخلوية}
\begin{enumerate}
  \item كل \omterm{def:b1:organism-environment:organism}{كائن حي} مكون من \omterm{def:b1:cell-unit-of-life:cell}{خلية} واحدة أو أكثر، \omterm{def:b1:cell-unit-of-life:cell}{والخلية} هي وحدة بنية
        الأحياء ووحدة عملها (شلايدن وشفان، 1838--1839).
  \item كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} تنشأ من \omterm{def:b1:cell-unit-of-life:cell}{خلية} سابقة لها بالانقسام (فيرشو، 1855:
        \emph{كل \omterm{def:b1:cell-unit-of-life:cell}{خلية} من \omterm{def:b1:cell-unit-of-life:cell}{خلية}})؛ ولا توالد ذاتيًا.
  \item المادة الوراثية تُمرَّر من \omterm{def:b1:cell-unit-of-life:cell}{خلية} إلى \omterm{def:b1:cell-unit-of-life:cell}{خلية} عند الانقسام، حتى
        إن استمرار الحياة هو استمرار \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}.
\end{enumerate}
\end{proposition}
\begin{proof}[شواهد]
فلين هوك (1665) والمقاطع النباتية عند غرو ومالبيغي أظهرت \omterm{def:b1:body-plans-tissues:tissue}{الأنسجة}
النباتية صفوفًا من الحجيرات؛ وعدسات ليفنهوك (سبعينيات القرن السابع
عشر) أظهرت \omterm{def:b1:organism-environment:organism}{كائنات} وحيدة \omterm{def:b1:cell-unit-of-life:cell}{الخلية} \omterm{def:b1:cell-unit-of-life:cell}{وخلايا} دم ونطافًا. ثم وجد قرنان من
تحسين العدسات \omterm{def:b1:cell-unit-of-life:cell}{خلايا} في كل \omterm{def:b1:body-plans-tissues:tissue}{نسيج} فُحص، حيوانيًا كان أم نباتيًا، بعد أن
تعرف شفان على \omterm{def:b1:cell-unit-of-life:cell}{خلايا} منواة في الغضروف والأجنة. ورُئي الانقسام مباشرة
في الطحالب وفي الأجنة، ولم يجد فيرشو نسيجًا، سليمًا ولا مريضًا، لا
تأتي خلاياه من \omterm{def:b1:cell-unit-of-life:cell}{خلايا} أخرى. ثم أغلقت دوارق باستور المعقوفة العنق
(1861) --- مرقٌ بقي عقيمًا شهورًا حين لم يبلغه الغبار، وفار في أيام
حين بلغه --- مسألة التوالد الذاتي. أما الجملة الثالثة فعمل أواخر
القرن التاسع عشر، الذي تتبع الصبغيات عبر الانقسام
(\cref{ch:b1:replication-mitosis}).
\end{proof}

\begin{example}[خلية واحدة، وخلايا كثيرة]\label{ex:b1:cell-unit-of-life:onemany}
البكتيريا والخميرة والأميبة \omterm{def:b1:organism-environment:organism}{كائنات} كاملة من \omterm{def:b1:cell-unit-of-life:cell}{خلية} واحدة، تفعل كل شيء
--- التغذي، والحركة، والانقسام --- في بضعة ميكرومترات. أما الإنسان
فنحو $\num{3e13}$ \omterm{def:b1:cell-unit-of-life:cell}{خلية} من مئتي نوع، لا تستطيع واحدة منها أن تعيش
وحدها في بركة، وكل منها منحدرة بانقسامات غير منقطعة من بيضة ملقحة
واحدة. وأرنب \cref{ch:b1:mammal-organization} وبلوطة
\cref{ch:b1:flowering-plant-organization} هما النظرية نفسها مطبقة
مرتين.
\end{example}

\section{نوعا الخلية}

\begin{definition}[الخلايا بدائية النواة وحقيقية النواة]\label{def:b1:cell-unit-of-life:prokeuk}
\emph{الخلية بدائية النواة}\index{خلية بدائية النواة} (البكتيريا،
والعتائق) لا نواة لها: فحمضها النووي، وهو عادة صبغي حلقي واحد، يقع
في منطقة من \omterm{def:b1:cell-unit-of-life:cell}{السيتوبلازم} تسمى \emph{النويد}\index{نويد}، ولا حجيرات
محدودة بغشاء فيها. وهي صغيرة (\qtyrange{0.5}{5}{\micro m}) يحدها
\omterm{def:b1:cell-unit-of-life:cell}{غشاء بلازمي} ومن خارجه \emph{جدار خلوي}\index{جدار خلوي}. أما
\emph{الخلية حقيقية النواة}\index{خلية حقيقية النواة} (الطلائعيات،
والفطريات، والنباتات، والحيوانات) فتحفظ حمضها النووي، في هيئة عدة
صبغيات خطية، داخل \emph{نواة}\index{نواة} يحدها غشاء مضاعف، وتقسم
سيتوبلازمها إلى \emph{عضيات}\index{عضية} محدودة بأغشية
(\cref{ch:b1:eukaryotic-cell}). وهي أكبر
(\qtyrange{10}{100}{\micro m}) ولها \omterm{def:b1:body-plans-tissues:segments}{هيكل داخلي} من خيوط بروتينية.
\end{definition}

\begin{proposition}[الخلية البكتيرية]\label{prop:b1:cell-unit-of-life:bacterium}
\emph{E. coli}، وهي البكتيريا النموذج، عصية طولها \qty{2}{\micro m}
وعرضها \qty{0.8}{\micro m}: أي حجم يناهز \qty{1}{\micro m^3} يحوي
صبغيًا واحدًا من $\num{4.6e6}$ زوج قواعد (\qty{1.5}{mm} من DNA مطوي
عشرة آلاف طية)، ونحو \num{20000} ريبوزوم، وبضعة ملايين من جزيئات
البروتين من \num{2000} نوع، وكثيرًا ما يحوي
\emph{بلازميدًا}\index{بلازميد} أو أكثر --- وهي جزيئات DNA حلقية
صغيرة تحمل مورّثات إضافية. وغلافها، من الداخل إلى الخارج، \omterm{def:b1:cell-unit-of-life:cell}{غشاء
بلازمي}، وجدار رقيق من \emph{الببتيدوغليكان}\index{ببتيدوغليكان}
(شبكة من سلاسل سكرية تربطها ببتيدات قصيرة، وهي تقاوم الضغط الداخلي
\omterm{def:b1:cell-unit-of-life:cell}{للخلية})، وغشاء خارجي. والبكتيريا ذات جدار الببتيدوغليكان السميك بلا
غشاء خارجي تحتفظ بصباغ البنفسج البلوري في \emph{صبغة
غرام}\index{صبغة غرام} (إيجابية الغرام)؛ وذات الجدار الرقيق تحت غشاء
خارجي تفقده (سلبية الغرام، مثل \emph{E. coli}). والأسواط تدور دوران
المراوح؛ والشعيرات تثبت \omterm{def:b1:cell-unit-of-life:cell}{الخلية} على السطوح وعلى \omterm{def:b1:cell-unit-of-life:cell}{خلايا} أخرى.
\end{proposition}

\begin{method}[صبغة غرام]\label{met:b1:cell-unit-of-life:gram}
\begin{enumerate}
  \item ثبّت لطاخة رقيقة من البكتيريا على شريحة بالحرارة.
  \item اصبغ بالبنفسج البلوري (فتصير كل \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} بنفسجية)، ثم باليود،
        الذي يكوّن معقدًا كبيرًا مع الصباغ داخل \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}.
  \item اشطف بالكحول: فالببتيدوغليكان السميك عند إيجابيات الغرام
        ينكمش ويحبس المعقد؛ أما الجدار الرقيق عند سلبيات الغرام،
        بعد أن يذيب الكحول الغشاء الخارجي، فيدع المعقد ينغسل.
  \item اصبغ صبغة مقابلة بصباغ وردي (السفرانين): فتبدو إيجابيات
        الغرام بنفسجية وسلبياته وردية. والنتيجة تتنبأ ببنية الجدار،
        ومعها الحساسية لكثير من المضادات الحيوية.
\end{enumerate}
\end{method}

\begin{example}[لماذا كانت الخلية حقيقية النواة محجّرة]\label{ex:b1:cell-unit-of-life:whycompart}
\omterm{def:b1:cell-unit-of-life:cell}{خلية} قياسها \qty{20}{\micro m} حجمها ألف ضعف حجم بكتيريا لكن سطحها
الغشائي مئة ضعف سطحها فقط
(\cref{prop:b1:organism-environment:sv}). والأغشية هي حيث تحمل
\omterm{def:b1:cell-unit-of-life:cell}{الخلية} سلاسل تحويل طاقتها وجمل نقلها؛ فتستعيد \omterm{def:b1:cell-unit-of-life:prokeuk}{حقيقية النواة} السطح
بطي الأغشية داخل نفسها --- الغشاء الداخلي للمتقدرة، وصفائح الشبكة
الإندوبلازمية --- وتكسب فوق ذلك حجرات منفصلة تستطيع أن تجري فيها
كيمياءات متنافرة (الهضم في الجسيم الحال، والتركيب في العصارة
الخلوية) جنبًا إلى جنب.
\end{example}

\section{رؤية الخلايا: المجهرية}

\begin{definition}[التكبير والقدرة الفاصلة]\label{def:b1:cell-unit-of-life:resolution}
المجهر \emph{يكبّر}\index{تكبير} (أي يعظّم الصورة)
و\emph{يفصل}\index{فصل} (أي يميز النقاط المتجاورة). ولا يحد ما يمكن
رؤيته إلا الثاني: \emph{فالقدرة الفاصلة}\index{قدرة فاصلة} هي أصغر
مسافة $d$ بين نقطتين تبقيان متمايزتين،
\[
d = \frac{0.61\,\lambda}{\mathrm{NA}},
\]
حيث $\lambda$ طول موجة الإضاءة و$\mathrm{NA}$ الفتحة العددية للشيئية
(جيب نصف زاويتها في جمع الضوء مضروبًا في قرينة انكسار الوسط، وهي
$1.4$ على الأكثر بالزيت). والتكبير فوق القدرة الفاصلة يعظّم الضبابة
لا التفصيل.
\end{definition}

\begin{proposition}[المجهران الضوئي والإلكتروني]\label{prop:b1:cell-unit-of-life:microscopes}
بالضوء المرئي ($\lambda \approx \qty{550}{nm}$) و$\mathrm{NA} = 1.4$
يفصل \emph{المجهر الضوئي}\index{مجهر ضوئي} \qty{0.24}{\micro m}:
أي \omterm{def:b1:cell-unit-of-life:cell}{الخلايا}، والنوى، والبلاستيدات الخضراء، والمتقدرات نقاطًا،
والبكتيريا عصيات --- وعينات حية متحركة. ويُحصَّل التباين في \omterm{def:b1:cell-unit-of-life:cell}{خلية}
شفافة بالصبغ (وهو يقتل عادة)، أو \emph{بتباين الطور} (بتحويل فروق
قرينة الانكسار إلى فروق في السطوع)، أو \emph{بالتألق} (بوسم جزيء
واحد بصباغ أو ببروتين متألق وتصوير ضوئه وحده). أما \emph{المجهر
الإلكتروني}\index{مجهر إلكتروني} فيستعمل إلكترونات طول موجتها بضعة
بيكومترات؛ وقدرته الفاصلة العملية، التي تحدها عدساته، نحو
\qty{0.2}{nm} في النفوذ (TEM: تمر الإلكترونات عبر مقطع سماكته
\qty{50}{nm} مصبوغ بمعادن ثقيلة، فيعطي البنية الداخلية \omterm{def:b1:cell-unit-of-life:prokeuk}{للعضيات})

\begin{method}[اختيار الأداة]\label{met:b1:cell-unit-of-life:choose}
\begin{enumerate}
  \item لمتابعة عملية حية (انقسام \omterm{def:b1:cell-unit-of-life:cell}{خلية}، أو حركة، أو بروتين متألق
        ينتقل بين حجيرتين): المجهرية الضوئية، بتباين الطور أو
        بالتألق، على \omterm{def:b1:cell-unit-of-life:cell}{خلايا} غير مثبتة.
  \item لعد \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} في \omterm{def:b1:body-plans-tissues:tissue}{نسيج} وتعيينها: تحضيرة مثبتة مقطعة مصبوغة تحت
        \omterm{prop:b1:cell-unit-of-life:microscopes}{المجهر الضوئي}.
  \item لرؤية البنية الداخلية \omterm{def:b1:cell-unit-of-life:prokeuk}{لعضية} أو غشاء أو فيروس: المجهرية
        الإلكترونية النافذة لمقاطع رقيقة، أو لجسيمات مصبوغة صبغًا
        سالبًا.
  \item لرؤية تضاريس سطح \omterm{def:b1:cell-unit-of-life:cell}{خلية} أو \omterm{def:b1:body-plans-tissues:tissue}{نسيج}: المجهرية الإلكترونية الماسحة.
  \item لتحديد موضع بروتين واحد: التألق المناعي (ضوئي) أو الوسم
        بالذهب المناعي (إلكتروني)، باستعمال ضد مناعي له.
\end{enumerate}
\end{method}

\begin{proposition}[الترسيب]\label{prop:b1:cell-unit-of-life:stokes}
جسيم كروي نصف قطره $r$ وكثافته $\rho_p$ في مائع كثافته $\rho_f$
ولزوجته $\eta$، في حقل نابذ $g_c$ (مضاعف للتسارع $g$)، يترسب بالسرعة
المستقرة
\[
v = \frac{2}{9}\,\frac{r^2\,(\rho_p - \rho_f)\,g_c}{\eta}
\]
(قانون ستوكس). والسرعة تنمو مثل مربع نصف القطر: \omterm{def:b1:cell-unit-of-life:prokeuk}{فنواة} قياسها
\qty{5}{\micro m} تسقط أسرع مئة مرة من متقدرة قياسها
\qty{0.5}{\micro m}، وهذه تسقط أسرع ألف مرة من ريبوزوم قياسه
\qty{15}{nm}. ولهذا يفصلها الأنبوب نفسه عند سرعات متزايدة.
\end{proposition}
\begin{proof}
عند السرعة المستقرة يوازن جر ستوكس، $6\pi\eta r v$، الوزن الصافي
للجسيم في المائع، $\frac{4}{3}\pi r^3(\rho_p - \rho_f)g_c$؛ وحل
المعادلة من أجل $v$ يعطي الصيغة.
\end{proof}

\begin{definition}[زرع الخلايا]\label{def:b1:cell-unit-of-life:culture}
\emph{زرع الخلايا}\index{زرع الخلايا} يبقي \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} حية منقسمة خارج
\omterm{def:b1:organism-environment:organism}{الكائن الحي}، في وسط عقيم يمدها بالمغذيات والأملاح وعوامل النمو،
وبسطح تلتصق به إن كانت \omterm{def:b1:cell-unit-of-life:cell}{خلايا} حيوانية، عند حرارة \omterm{def:b1:organism-environment:organism}{الكائن الحي}.
والبكتيريا والخمائر تنمو في أوساط بسيطة وتتضاعف في دقائق إلى ساعات؛
\omterm{def:b1:cell-unit-of-life:cell}{والخلايا} الحيوانية المأخوذة من \omterm{def:b1:body-plans-tissues:tissue}{نسيج} (\emph{المزارع الأولية}) تنقسم
عددًا محدودًا من المرات، بينما تنقسم \emph{السلالات الخلوية}
المشتقة من أورام أو من \omterm{def:b1:cell-unit-of-life:cell}{خلايا} محوَّلة انقسامًا \omterm{prop:b1:flowering-plant-organization:modular}{غير محدود}. والزرع يجعل
\omterm{def:b1:cell-unit-of-life:cell}{الخلية} موضوعًا تجريبيًا: نوع واحد، ووسط واحد، ومتغير واحد في كل مرة.
\end{definition}

\begin{example}[منحني نمو]\label{ex:b1:cell-unit-of-life:growthcurve}
دورق من \emph{E. coli} في وسط غني عند \qty{37}{\celsius}: تأخر نصف
ساعة، ثم نمو أسي --- إذ يتضاعف العدد كل \qty{20}{min}، فيكون
$N = N_0\,2^{t/20}$ --- إلى أن ينفد مغذٍ أو تتراكم الفضلات فتبلغ
المزرعة هضبة قرب $10^9$ \omterm{def:b1:cell-unit-of-life:cell}{خلية} في المليلتر. فمن \omterm{def:b1:cell-unit-of-life:cell}{خلية} واحدة يُبلغ
$2^{30} \approx 10^9$ \omterm{def:b1:cell-unit-of-life:cell}{خلية} في عشر ساعات. والمنحني نفسه، بزمن تضاعف
قدره يوم، يصف مزرعة \omterm{def:b1:cell-unit-of-life:cell}{خلايا} بشرية.
\end{example}

1)، ثم $\num{10000}\,g$ لمدة \qty{20}{min} (الراسب 2)، ثم

الأخير. ويُقاس محتوى البروتين ونشاطا إنزيمين في \omterm{def:b1:cell-unit-of-life:fractionation}{المجانس} وفي كل جزء:

\medskip
\noindent\textbf{الجزء الأول --- ماذا يقع أين.}
\begin{enumerate}
  \item سمِّ \omterm{def:b1:cell-unit-of-life:prokeuk}{العضية} الرئيسة في كل راسب.
  \item لماذا يجب أن يكون وسط المجانسة باردًا متساوي التوتر؟
  \item نازعة هيدروجين السكسينات واسم لأي حجيرة؟ ونازعة هيدروجين
        اللاكتات لأي حجيرة؟
  \item احسب استرجاع البروتين: أي مجموعه في الأجزاء بالنسبة المئوية
        من \omterm{def:b1:cell-unit-of-life:fractionation}{المجانس}. وعلّق.
  \item احسب استرجاع نشاط كل إنزيم بالطريقة نفسها.
  \item احسب محتوى البروتين لكل غرام من الكبد والكسر من الكتلة
        الطرية للكبد الذي هو بروتين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- قانون ستوكس.}
المتقدرات: نصف القطر \qty{0.5}{\micro m}، وفائض الكثافة
\qty{100}{kg/m^3}. والنوى: نصف القطر \qty{4}{\micro m}، والفائض
نفسه. والريبوزومات: نصف القطر \qty{12}{nm}، وفائض الكثافة
\qty{600}{kg/m^3}.
\begin{enumerate}[resume]
  \item احسب سرعة ترسيب متقدرة عند $\num{10000}\,g$ والزمن اللازم
        لقطع \qty{5}{cm}. فهل تكفي \qty{20}{min}؟
  \item احسب المسافة التي تقطعها متقدرة في أثناء الدوران الأول
        ($1000\,g$، \qty{10}{min}). فأي كسر من المتقدرات، الموزعة
        توزيعًا منتظمًا في البدء، سيترسب في المرحلة الأولى؟
  \item قارن مع \qty{15}{\%} من نازعة هيدروجين السكسينات الموجودة في
        الراسب 1. وما الذي يستطيع أيضًا أن يجر المتقدرات إلى الراسب 1؟
  \item احسب الزمن اللازم \omterm{def:b1:cell-unit-of-life:prokeuk}{لنواة} لتقطع \qty{5}{cm} عند $1000\,g$.
  \item احسب الزمن اللازم لريبوزوم ليقطع \qty{5}{cm} عند
        $\num{100000}\,g$. فهل يترسب ريبوزوم عند $\num{10000}\,g$ في
        \qty{20}{min}؟
  \item فسّر لماذا كان تتابع الدورانات بقوة متزايدة ولماذا وجب أن
        تكون كل مرحلة طويلة بما يكفي لا أطول مما ينبغي.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- النشاط النوعي والتنقية.}
النشاط النوعي لإنزيم في جزء هو نشاطه لكل مليغرام من البروتين. وعامل
تنقية جزء من أجل إنزيم هو نشاطه النوعي مقسومًا على النشاط النوعي في
\omterm{def:b1:cell-unit-of-life:fractionation}{المجانس}؛ والمردود هو نشاط الجزء مقسومًا على نشاط \omterm{def:b1:cell-unit-of-life:fractionation}{المجانس}.
\begin{enumerate}[resume]
  \item احسب النشاط النوعي لنازعة هيدروجين السكسينات في \omterm{def:b1:cell-unit-of-life:fractionation}{المجانس} وفي
        كل جزء.
  \item احسب عامل التنقية والمردود لنازعة هيدروجين السكسينات في
        الراسب 2.
  \item احسب عامل تنقية نازعة هيدروجين السكسينات في الراسب 3 وفسّر
        قيمته.
  \item احسب النشاط النوعي لنازعة هيدروجين اللاكتات في \omterm{def:b1:cell-unit-of-life:fractionation}{المجانس} وفي
        الطافي الأخير، وعامل التنقية والمردود الموافقين.
  \item لماذا كان عامل تنقية الإنزيم العصاري أدنى من عامل تنقية
        الإنزيم المتقدري مع أن مردوده أعلى؟
  \item يحوي الراسب 2 \qty{12}{U} من نازعة هيدروجين اللاكتات. فهل
        الإنزيم متقدري جزئيًا؟ اقترح تفسيرًا أفضل وطريقة لاختباره.
  \item أي كسر من بروتين الكبد متقدري، إذا كانت كل نازعة هيدروجين
        السكسينات متقدرية والراسب 2 نقي بنسبة \qty{60}{\%}؟ (إرشاد:
        استعمل النشاط النوعي للمتقدرات النقية، مستنتجًا من الراسب 2،
        والنشاط الكلي.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- التحقق من الأجزاء.}
\begin{enumerate}[resume]
  \item يُفحص الراسب 2 بالمجهرية الإلكترونية: فأي أداة وأي تحضير،
        وماذا ينبغي أن يُرى؟
  \item كيف تتحقق من الراسب 1 بحثًا عن \omterm{def:b1:cell-unit-of-life:cell}{خلايا} غير مكسورة، وماذا تفعل
        إن وجدت كثيرًا منها؟
  \item يُعاد تعليق الراسب 2 ويُوضع طبقةً على تدرج كثافة سكروزي، ثم
        يُدار حتى تقف كل جسيمة حيث تساوي كثافتها كثافة الوسط.
        فالمتقدرات تشكل نطاقًا عند \qty{1.18}{g/mL}، والجسيمات الحالة
        عند \qty{1.22}{g/mL}، والبيروكسيسومات عند \qty{1.24}{g/mL}.
        فسّر لماذا ينجح هذا الفصل حيث لم يستطع \omterm{def:b1:cell-unit-of-life:fractionation}{الطرد المركزي
        التفاضلي} أن يحققه.
  \item يرتفع نشاط الفوسفاتاز الحمضي في الراسب 2 ارتفاعًا حادًا حين
        يُضاف مُنظِّف إلى المقايسة. فسّر، وقل ماذا يخبرنا هذا عن حال
        الجسيمات الحالة في الراسب.
  \item يُجانِس طالب في ماء نقي بدل السكروز المتساوي التوتر. تنبأ
        بالأثر في الراسب 2 وفي توزع نازعة هيدروجين السكسينات
        والفوسفاتاز الحمضي.
  \item لخّص النتيجة: عامل التنقية ومردود الجزء المتقدري، وما الذي
        يحد كلًا منهما.
\end{enumerate}
\end{problem}
