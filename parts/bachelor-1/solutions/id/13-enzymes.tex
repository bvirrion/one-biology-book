\section*{Bab \ref{ch:b1:enzymes} --- Enzim dan Katalisis Biokimia}

\begin{solution}{exo:b1:enzymes:1}
Ia menurunkan \omterm{prop:b1:enzymes:whatitdoes}{energi pengaktifannya} (tinggi \omterm{prop:b1:enzymes:whatitdoes}{keadaan transisinya}), dan
dengan itu lajunya, pada kedua arahnya; ia membiarkan $\Delta G$,
tetapan kesetimbangan dan letak kesetimbangannya tak berubah. Pada
profilnya puncaknya diturunkan, kedua ujungnya tidak.
\end{solution}

\begin{solution}{exo:b1:enzymes:2}
$K_m$: kepekatan \omterm{def:b1:enzymes:enzyme}{substrat} pada setengah $V_{\max}$ (\unit{mol/L}).
$V_{\max}$: laju pada \omterm{def:b1:enzymes:enzyme}{substrat} yang menjenuhkan (\unit{mol/s}, atau
\unit{\micro mol/min}). $k_{\text{cat}}$: molekul \omterm{def:b1:enzymes:enzyme}{substrat} yang diubah
tiap sekon tiap \omterm{def:b1:enzymes:enzyme}{enzim} pada kejenuhannya (\unit{s^{-1}}).
$k_{\text{cat}}/K_m$: efisiensinya, yaitu tetapan laju orde dua pada
\omterm{def:b1:enzymes:enzyme}{substrat} yang rendah
(\unit{L.mol^{-1}.s^{-1}}).
\end{solution}

\begin{solution}{exo:b1:enzymes:3}
Tanpa \omterm{def:b1:enzymes:inhibitors}{penghambat}: perpotongannya 1, sehingga $V_{\max} = 1$;
perpotongan $x$-nya $-0.5$, sehingga $K_m = 2$. Bersaing:
$V_{\max} = 1$, perpotongan $x$ $-0.25$, $K_m = 4$. Tak bersaing:
perpotongannya 2, $V_{\max} = 0.5$; perpotongan $x$ $-0.5$,
$K_m = 2$.
\end{solution}

\begin{solution}{exo:b1:enzymes:4}
Pengaturan \omterm{def:b1:proteins:allostery}{alosterik} (umpan balik), milisekon; \omterm{prop:b1:enzymes:control}{perubahan kovalen}
(fosforilasi), sekon sampai menit; pengaktifan proteolitik sebuah
zimogen, sekali, sesuai kebutuhan; pengendalian jumlahnya lewat
pengungkapan gen dan penguraian, menit sampai jam.
\end{solution}

\begin{solution}{exo:b1:enzymes:5}
$v_0 = 50[S]/(0.1 + [S])$: 8.3, 25, 45.5 dan \qty{49.5}{\micro mol/min}.
$0.9 = [S]/(0.1 + [S])$ memberi $[S] = 0.9\,K_m/0.1 = \qty{0.9}{mmol/L}$.
\end{solution}

\begin{solution}{exo:b1:enzymes:6}
$e^{20\,000/2580} = e^{7.75} = 2300$. Bagi $10^{10}$: $\Delta E_a =
RT\ln 10^{10} = 2.58\times 23.0 = \qty{59}{kJ/mol}$.
\end{solution}

\begin{solution}{exo:b1:enzymes:7}
$V_{\max} = \qty{0.12}{mol/L}$ tiap menit dalam \qty{1}{mL}:
\qty{1.2e-4}{mol/min} $= \qty{2.0e-6}{mol/s}$ hasil; \omterm{def:b1:enzymes:enzyme}{enzimnya}
\qty{2e-12}{mol}: $k_{\text{cat}} = 2\times 10^{-6}/2\times 10^{-12} =
\qty{1e6}{s^{-1}}$. Efisiensinya $10^6/0.012 = \qty{8e7}{L.mol^{-1}.s^{-1}}$,
dalam faktor sepuluh dari batas difusinya: nyaris tiap perjumpaan
dengan sebuah molekul \omterm{def:b1:enzymes:enzyme}{substrat} menghasilkan.
\end{solution}

\begin{solution}{exo:b1:enzymes:8}
$1 + 2/K_i = 2$: $K_i = \qty{2}{mmol/L}$. Bagi \qty{90}{\%}:
$[S] = 9K_m^{\text{app}}$: \qty{9}{mmol/L} tanpa, \qty{18}{mmol/L}
dengan \omterm{def:b1:enzymes:inhibitors}{penghambatnya}.
\end{solution}

\begin{solution}{exo:b1:enzymes:9}
Menghambat langkah pertama yang menentukan menghentikan seluruh
lintasannya sekaligus dan tidak memboroskan zat antara, yang jika tidak
akan menumpuk; hasil akhirnya adalah isyarat yang penting ---
kelimpahannya justru adalah informasi bahwa lintasannya tak dibutuhkan
lagi --- sedangkan aras sebuah zat antara tidak mengatakan apa pun
tentang permintaannya.
\end{solution}

\begin{solution}{exo:b1:enzymes:10}
Sebuah protease yang aktif di dalam \omterm{def:b1:cell-unit-of-life:cell}{sel} yang membuatnya akan mencerna
\omterm{def:b1:cell-unit-of-life:cell}{sel} itu; karena dibuat sebagai zimogen yang tak aktif, ia tak
berbahaya sampai enteropeptidase, sebuah \omterm{def:b1:enzymes:enzyme}{enzim} pelapis usus, memotong
tripsinogen menjadi tripsin di duodenum, tempat pencernaannya
diinginkan; tripsin lalu mengaktifkan zimogen yang lain (sebuah
untaian). Karena sedikit tripsin dapat memulai untaian itu terlalu
dini, pankreas mengemaskan sebuah \omterm{def:b1:enzymes:inhibitors}{penghambat} tripsin yang khas ke dalam
butirannya sebagai jaminan; gagalnya pengaturan itu adalah pankreatitis.
\end{solution}

\begin{solution}{exo:b1:enzymes:11}
Tanpa: $3^3/(3^3 + 3^3) = 0.50$. Dengan: $27/(166 + 27) = 0.14$:
lajunya turun ke \qty{28}{\%} nilainya. Sebuah \omterm{def:b1:enzymes:enzyme}{enzim} Michaelis--Menten
yang $K_m$-nya naik dari 3 ke 5.5: $3/6 = 0.50$ ke $3/8.5 = 0.35$,
yakni \qty{70}{\%}. \omterm{def:b1:proteins:allostery}{Kerja sama} membuat geseran kurva yang sama lebih
dari dua kali lebih mangkus dekat titik kerjanya: sebuah sakelar
alih-alih sebuah peredup.
\end{solution}

\begin{solution}{exo:b1:enzymes:12}
Sebuah katalis telanjang (permukaan platina, sebuah asam) mempercepat
banyak reaksi tanpa pilih kasih; sebuah \omterm{def:b1:enzymes:enzyme}{enzim} mempercepat satu reaksi
pada satu \omterm{def:b1:enzymes:enzyme}{substrat} --- kekhasan \omterm{def:b1:enzymes:enzyme}{sisi aktifnya} sudah merupakan informasi
tentang apa yang diinginkan \omterm{def:b1:cell-unit-of-life:cell}{selnya}. Pengaturannya menambahkan lapisan
kedua: sisi \omterm{def:b1:proteins:allostery}{alosterik}, sisi fosforilasi dan peptida zimogennya memberi
tahu \omterm{def:b1:enzymes:enzyme}{enzimnya} kapan dan di mana harus bertindak, sebagai tanggapan atas
keadaan \omterm{def:b1:cell-unit-of-life:cell}{selnya}. Biayanya adalah sebuah \omterm{def:b1:proteins:peptide}{protein} besar (ratusan residu
bagi sebuah sisi yang berisi selusin), putaran yang lebih lambat
daripada katalis anorganik terbaik, dan mesin untuk membuat, mengubah
serta memusnahkannya; keuntungannya adalah kimia yang berjalan hanya di
tempat dan saat ia berguna, dan itulah sebuah metabolisme.
\end{solution}

\begin{solution}{pb:b1:enzymes:1}
\textbf{1.} $1/[S]$: 2, 1, 0.5, 0.2, 0.1, 0.05. $1/v_0$: 5.0, 3.0, 2.0,
1.4, 1.2, 1.1.
\textbf{2.} Tiap langkah $1/[S]$ mengubah $1/v_0$ sebanding dengannya:
kemiringannya $(5.0 - 1.1)/(2 - 0.05) = 2.0$; perpotongannya $1.0$.
\textbf{3.} $V_{\max} = 1/1.0 = \qty{1.0}{\micro mol/min}$; $K_m =
\text{kemiringan}\times V_{\max} = \qty{2.0}{mmol/L}$.
\textbf{4.} $1.0\times 5/(2 + 5) = 0.714$: seperti yang terukur.
\textbf{5.} $\qty{10e-9}{mol/L}\times\qty{1e-3}{L} = \qty{1e-11}{mol}$;
$V_{\max} = \qty{1e-6}{mol/min} = \qty{1.67e-8}{mol/s}$;
$k_{\text{cat}} = 1.67\times 10^{-8}/10^{-11} = \qty{1670}{s^{-1}}$.
\textbf{6.} $1670/(2\times 10^{-3}) = \qty{8.3e5}{L.mol^{-1}.s^{-1}}$:
seribu kali di bawah batas difusinya; hanya satu tumbukan dari seribu
yang menghasilkan.
\textbf{7.} $0.95 = [S]/(2 + [S])$: $[S] = 19\,K_m = \qty{38}{mmol/L}$.
\textbf{8.} $1/v_0$: 9.0, 5.0, 3.0, 1.8, 1.4, 1.2. Kemiringan $(9.0 -
1.2)/1.95 = 4.0$; perpotongan $1.0$.
\textbf{9.} $V_{\max}^{\text{app}} = \qty{1.0}{\micro mol/min}$ (tak
berubah); $K_m^{\text{app}} = 4.0\times 1.0 = \qty{4.0}{mmol/L}$.
\textbf{10.} $K_m$ berlipat dua, $V_{\max}$ tak berubah: bersaing.
\textbf{11.} $4 = 2(1 + 1/K_i)$: $K_i = \qty{1.0}{mmol/L}$.
\textbf{12.} Dengan \omterm{def:b1:enzymes:inhibitors}{penghambatnya}, $v = V_{\max}[S]/(K_m(1 + [I]/K_i) +
[S])$; membagi dua lajunya menuntut $K_m(1 + [I]/K_i) + [S] = 2(K_m +
[S])$, yakni $[I] = K_i(K_m + [S])/K_m$: pada \qty{2}{mmol/L}, $[I] =
1\times 4/2 = \qty{2}{mmol/L}$; pada \qty{20}{mmol/L}, $[I] = 22/2 =
\qty{11}{mmol/L}$. Makin banyak \omterm{def:b1:enzymes:enzyme}{substratnya}, makin banyak \omterm{def:b1:enzymes:inhibitors}{penghambat}
yang dibutuhkan.
\textbf{13.} Sebuah molekul yang mirip \omterm{def:b1:enzymes:enzyme}{substratnya} (atau \omterm{prop:b1:enzymes:whatitdoes}{keadaan
transisinya}) --- sebuah analog ester yang tak dapat dihidrolisis ---
yang mengikat pada \omterm{def:b1:enzymes:enzyme}{sisi aktifnya}.
\textbf{14.} $1/v_0$: 10.0, 6.0, 4.0, 2.8, 2.4, 2.2; kemiringan $4.0$,
perpotongan $2.0$: $V_{\max}^{\text{app}} = \qty{0.5}{\micro mol/min}$,
$K_m^{\text{app}} = 4.0\times 0.5 = \qty{2.0}{mmol/L}$.
\textbf{15.} $V_{\max}$ terbagi dua, $K_m$ tak berubah: tak bersaing;
$2 = 1 + 1/K_i$: $K_i = \qty{1.0}{mmol/L}$.
\textbf{16.} J mengikat pada sisi selain \omterm{def:b1:enzymes:enzyme}{sisi aktifnya}, pada \omterm{def:b1:enzymes:enzyme}{enzim}
bebas maupun $ES$, dengan afinitas yang sama: \omterm{def:b1:enzymes:enzyme}{substratnya} tidak
bersaing dengannya, dan pada $[S]$ berapa pun separuh molekul \omterm{def:b1:enzymes:enzyme}{enzimnya}
dinonaktifkan.
\textbf{17.} Dengan J lajunya berlipat dua (separuh dari \omterm{def:b1:enzymes:enzyme}{enzim} yang dua
kali lebih banyak tetap aktif); dengan I lajunya juga berlipat dua
(lajunya sebanding dengan \omterm{def:b1:enzymes:enzyme}{enzimnya} pada tiap $[S]$): melipatduakan
\omterm{def:b1:enzymes:enzyme}{enzimnya} tak pernah membedakan kedua \omterm{def:b1:enzymes:inhibitors}{penghambatnya}.
\textbf{18.} Lajunya pada \qty{25}{\celsius}: $V_{\max}[S]/(K_m + [S])$
dengan $V_{\max} = k_{\text{cat}}[E]$: tiap liter, $1670\times 10^{-8} =
\qty{1.67e-5}{mol/L/s}$; $\times 0.5/2.5 = \qty{3.3e-6}{mol/L/s}$; pada
\qty{37}{\celsius}, $\times 2^{1.2} = 2.3$: \qty{7.7e-6}{mol/L/s}; dalam
\qty{1e-12}{L}: \qty{7.7e-18}{mol/s} $= \num{4.6e6}$ molekul tiap sekon.
\textbf{19.} Tak perlu naik: lajunya (\num{4.6e6}) melebihi
permintaannya (\num{3e6}); seandainya harus naik, \omterm{def:b1:cell-unit-of-life:cell}{sel} dapat menaikkan
\omterm{def:b1:enzymes:enzyme}{substratnya}, atau memfosforilasi \omterm{def:b1:enzymes:enzyme}{enzimnya} agar $K_m$-nya turun, atau
membuat lebih banyak \omterm{def:b1:enzymes:enzyme}{enzim}.
\textbf{20.} $K_m^{\text{app}} = 2(1 + 0.4/0.2) = \qty{6}{mmol/L}$:
lajunya $\times 0.5/6.5$ alih-alih $0.5/2.5$: \qty{38}{\%} dari
sebelumnya, \num{1.8e6} tiap sekon --- hasilnya mencekik sintesisnya
sendiri sampai di bawah permintaannya, sehingga ia terpakai dan arasnya
turun, yang lalu melepaskan \omterm{def:b1:enzymes:enzyme}{enzimnya}: sebuah pasokan yang menyetel
dirinya sendiri.
\textbf{21.} Sebelumnya: $0.5/2.5 = 0.20$ dari $V_{\max}$; sesudahnya:
$0.5/0.9 =
0.56$: hampir tiga kali lipat.
\textbf{22.} Nyaris nol: pada pH 5 histidina \omterm{def:b1:enzymes:enzyme}{sisi aktifnya} terprotonasi
dan tak sanggup berlaku sebagai basa, dan residu asam sisinya
dinetralkan; keadaan pengionan \omterm{def:b1:enzymes:enzyme}{enzimnya}, dan mungkin lipatannya,
keliru. Ia dibangun bagi \omterm{def:b1:eukaryotic-cell:organelle}{sitosol}, bukan bagi \omterm{def:b1:eukaryotic-cell:endomembrane}{lisosom}.
\textbf{23.} $K_m$-nya sedikit berubah (pengikatannya sebagian besar
oleh selebihnya sisi itu); $k_{\text{cat}}$-nya runtuh beberapa orde
besar, sebab histidinanya adalah basa katalitik yang mengaktifkan air
atau serinanya.
\textbf{24.} Dekat $K_m$ lajunya menanggapi perubahan \omterm{def:b1:enzymes:enzyme}{substratnya}
(kemiringannya dekat $V_{\max}/2K_m$); jauh di atasnya \omterm{def:b1:enzymes:enzyme}{enzimnya} \omterm{def:b1:lipids:fattyacid}{jenuh}
dan tak peka, dan kelebihan \omterm{def:b1:enzymes:enzyme}{substratnya} akan menjadi beban osmotik dan
kimia. Bekerja dekat $K_m$ membuat lintasannya dapat dikendalikan
pasokannya.
\textbf{25.} $K_m = \qty{2.0}{mmol/L}$, $V_{\max} = \qty{1.0}{\micro mol/min}$
(\qty{10}{nmol/L} \omterm{def:b1:enzymes:enzyme}{enzim}), $k_{\text{cat}} = \qty{1670}{s^{-1}}$; I
bersaing, $K_i = \qty{1.0}{mmol/L}$; J tak bersaing, $K_i =
\qty{1.0}{mmol/L}$.
\end{solution}
