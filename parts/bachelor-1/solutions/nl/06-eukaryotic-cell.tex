\section*{Hoofdstuk \ref{ch:b1:eukaryotic-cell} --- Functionele organisatie van de eukaryote cel}

\begin{solution}{exo:b1:eukaryotic-cell:1}
\omterm{def:b1:eukaryotic-cell:nucleus}{Celkern} (chromosomen, transcriptie); ruw ER (secretie- en membraaneiwitten);
glad ER (lipiden, calcium, ontgifting); \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} (sorteren, wijzigen,
verpakken); \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomen} (vertering); \omterm{def:b1:eukaryotic-cell:peroxisome}{peroxisomen} (oxidaties, katalase);
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondriën} (ademhaling, ATP).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:2}
Gemaakt door ribosomen die aan het \omterm{def:b1:eukaryotic-cell:endomembrane}{ruwe} ER gebonden zijn en tijdens de synthese
het lumen ingeschoven (\omterm{prop:b1:eukaryotic-cell:secretory}{signaalsequentie}); blaasjes knoppen van het ER af en
versmelten met het cis-golgi; het eiwit doorkruist de stapel, wordt gewijzigd en
gesorteerd; het verlaat de trans-zijde in een condenserende \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole} die tot
secretiekorrel rijpt; de korrel versmelt op een signaal met de apicale membraan
en leegt zich in de afvoerbuis.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:3}
$7 + 32 = \qty{39}{\%}$. Het binnenmembraan is tot \omterm{def:b1:eukaryotic-cell:mitochondrion}{cristae} gevouwen omdat het de
ademhalingsketen en het ATP-synthase draagt, waarvan de doorzet evenredig is met
zijn oppervlak; het buitenmembraan is slechts een omhulsel.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:4}
Plant: cellulosewand, grote \omterm{def:b1:eukaryotic-cell:plantcell}{vacuole}, \omterm{def:b1:eukaryotic-cell:plastid}{plastiden} (en plasmodesmata). Dier:
\omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomen}, centriolen (en het vermogen om te kruipen en te fagocyteren).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:5}
ER na \qty{3}{min} (dan al dalend), \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} na ongeveer \qty{15}{min},
korrels na ongeveer \qty{60}{min}. Transport van ER naar korrel ongeveer
\qty{45}{min}; de helft afgescheiden na ongeveer \qty{110}{min}.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:6}
Het \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat}, dat door dyneïne op \omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtubuli} bij het \omterm{def:b1:eukaryotic-cell:cytoskeleton}{centrosoom} wordt
gehouden, valt uiteen en raakt verspreid; het blaasjesverkeer vertraagt tot
diffusie en het transport naar de rand stopt; de spoelfiguur kan niet worden
gevormd en \omterm{def:b1:cell-unit-of-life:cell}{cellen} blijven in de mitose steken; trilharen, waarvan de kern uit
\omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtubuli} bestaat, staan stil (bestaande trilharen zijn stabiel, maar worden
niet vernieuwd).
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:7}
Twee membranen waarvan het binnenste van het bacteriële type is; ringvormig DNA
zonder histonen; ribosomen van bacteriële afmeting, gevoelig voor antibiotica;
deling door splitsing; gensequenties die zich bij de alfaproteobacteriën
groeperen. Het kan niet alleen leven omdat de meeste van zijn genen naar de kern
zijn verhuisd: het voert de meeste van zijn eiwitten uit het \omterm{def:b1:eukaryotic-cell:organelle}{cytosol} in.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:8}
De suikers worden in het ER en het \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} aan de lumenzijde aangebracht.
Elke versmelting van een blaasje behoudt de zijdigheid: de lumenzijde van een
blaasje wordt de lumenzijde van het volgende compartiment, en wanneer het blaasje
met de plasmamembraan versmelt, opent zijn lumen zich naar buiten, zodat de
suikers naar de buitenkant wijzen. Wat luminaal is, ligt topologisch buiten.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:9}
Elke korrel $\pi d^2 = \qty{3.14}{\micro m^2}$; per dag
\qty{6280}{\micro m^2}, oftewel \qty{157}{} keer de apicale zijde. De \omterm{def:b1:cell-unit-of-life:cell}{cel} moet
membraan met dezelfde snelheid door \omterm{def:b1:eukaryotic-cell:endocytosis}{endocytose} terughalen, zodat de apicale
membraan elke negen minuten wordt vervangen en het teruggehaalde membraan voor
hergebruik naar het \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} gaat.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:10}
De keten van \qty{320}{} residuen is de voorloper met zijn \omterm{prop:b1:eukaryotic-cell:secretory}{signaalsequentie} van
\qty{20}{} residuen, die normaal bij binnenkomst in het ER wordt afgeknipt. Met
blaasjes erbij komt de keten tijdens de synthese het lumen binnen, wordt het
signaal afgeknipt, en beschermt de membraan het product tegen het protease. De
mutant zonder signaal wordt door de ER-machinerie niet herkend: hij wordt op
vrije ribosomen gemaakt en blijft op volle lengte in het \omterm{def:b1:eukaryotic-cell:organelle}{cytosol}, en wordt nooit
afgescheiden.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:11}
Membranen die door \omterm{def:b1:eukaryotic-cell:endocytosis}{endocytose} naar binnen zijn gehaald en versleten \omterm{def:b1:cell-unit-of-life:prokeuk}{organellen}
worden ter vertering aan de \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomen} afgeleverd; zonder het enzym kan de lipide
die zij bevatten niet worden afgebroken en vult zij de \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomen}, die opzwellen
tot de \omterm{def:b1:cell-unit-of-life:cell}{cel} het begeeft. \omterm{def:b1:cell-unit-of-life:cell}{Cellen} met de snelste membraanomzet (\omterm{def:b1:body-plans-tissues:nervous}{neuronen}, die
synaptische blaasjes onafgebroken hergebruiken; macrofagen) stapelen haar het
snelst op. Een ingespoten enzym dat het suikersignaal draagt dat lysosomale
enzymen kenmerkt, kan aan receptoren op het celoppervlak binden en door
receptorgemedieerde \omterm{def:b1:eukaryotic-cell:endocytosis}{endocytose} in de \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomen} worden opgenomen --- het beginsel
van de enzymvervangingstherapie.
\end{solution}

\begin{solution}{exo:b1:eukaryotic-cell:12}
Endosymbiose verklaart \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitochondriën} en \omterm{def:b1:eukaryotic-cell:plastid}{plastiden} goed (twee membranen, een eigen
genoom en eigen ribosomen, splitsing, bacteriële verwanten), en daarmee de
energiestofwisseling van de eukaryoot. Zij verklaart niet de \omterm{def:b1:eukaryotic-cell:nucleus}{celkern}, het ER en
het \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat}, of het \omterm{def:b1:eukaryotic-cell:cytoskeleton}{cytoskelet}, die geen bacteriële tegenhangers hebben en
uitvindingen van de gastheerlijn lijken --- instulpingen van haar membraan en
nieuwe eiwitten. ``Federatie'' past bij de \omterm{def:b1:cell-unit-of-life:prokeuk}{organellen}; ``bijeengehouden door een
membraanstelsel'' wijst op het grotere, onverklaarde deel: een gastheercel die al
eukaryoot van bouw was voordat zij de bacteriën opnam.
\end{solution}

\begin{solution}{pb:b1:eukaryotic-cell:1}
\textbf{1.} Een korte puls merkt een smalle lichting moleculen, zodat hun plaats
één tijdstip markeert; de overmaat ongemerkt aminozuur stopt de verdere inbouw,
zodat de lichting niet door latere synthese wordt uitgesmeerd.
\textbf{2.} ER: \qty{3}{min} (het vroegste punt); \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} en
condenserende \omterm{def:b1:eukaryotic-cell:plantcell}{vacuolen}: ongeveer \qty{15}{min}; korrels: ongeveer \qty{60}{min}.
\textbf{3.} ER naar \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} ongeveer \qty{10}{min}; \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} naar
korrel ongeveer \qty{45}{min}.
\textbf{4.} Grotendeels in korrels (\qty{70}{\%}), een tiende in het
\omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat}, een tiende al afgescheiden: de constitutieve route en de eerst
gevormde korrels geven al iets af, terwijl het meeste op een prikkel wacht.
\textbf{5.} Het afknoppen en vervoeren van blaasjes vraagt energie en vloeibare
membranen; in de kou stopt de stap van ER naar \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} en hoopt het eiwit
zich vóór de blokkade op.
\textbf{6.} $10\,\mathrm{g}/\num{8e10} = \qty{1.25e-10}{g} =
\qty{125}{pg}$ per \omterm{def:b1:cell-unit-of-life:cell}{cel} per dag.
\textbf{7.} $V = \frac{\pi}{6}(1)^3 = \qty{0.52}{\micro m^3} =
\qty{5.2e-13}{mL}$; eiwit $\qty{1.05e-13}{g} = \qty{0.1}{pg}$.
\textbf{8.} $125/0.1 = 1250$ korrels per dag, $0.87$ per minuut.
\textbf{9.} $V = 12^3 = \qty{1728}{\micro m^3} = \qty{1.73e-9}{cm^3}$;
massa \qty{1.8e-9}{g} = \qty{1800}{pg}. Zij scheidt $125/1800 = 7\%$ van haar
massa per dag af (de hele \omterm{def:b1:cell-unit-of-life:cell}{cel} is maar voor \qty{20}{\%} eiwit, dus een derde van
haar eiwitgehalte).
\textbf{10.} Molaire massa $250\times 110 = \qty{27500}{g/mol}$;
$\qty{1.05e-13}{g}/27500\times\num{6e23} = \num{2.3e6}$ moleculen.
\textbf{11.} $250/5 = \qty{50}{s}$.
\textbf{12.} Moleculen per dag $1250\times\num{2.3e6} = \num{2.9e9}$;
één ribosoom maakt er $86\,400/50 = 1728$ per dag; nodig:
$\num{1.7e6}$ ribosomen, ongeveer \qty{40}{\%} van die van de \omterm{def:b1:cell-unit-of-life:cell}{cel}.
\textbf{13.} $\pi d^2 = \qty{3.14}{\micro m^2}$.
\textbf{14.} $1250\times 3.14 = \qty{3900}{\micro m^2}$.
\textbf{15.} $3900/144 = 27$ keer: de \omterm{def:b1:cell-unit-of-life:cell}{cel} haalt hetzelfde oppervlak door
\omterm{def:b1:eukaryotic-cell:endocytosis}{endocytose} terug en geeft het aan het \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat} terug.
\textbf{16.} $20\,000/3900 = 5$ dagen: het ER zou maar vijf dagen
korrelmembraan kunnen leveren als er niets terugkwam; het membraan moet
onafgebroken worden hergebruikt.
\textbf{17.} Elke microvillus $\pi\times 0.1\times 1 =
\qty{0.31}{\micro m^2}$; \qty{630}{\micro m^2} voor 2000; factor $5.4$.
\textbf{18.} Aan het begin van de keten: de \omterm{prop:b1:eukaryotic-cell:secretory}{signaalsequentie}, twintig
overwegend hydrofobe residuen.
\textbf{19.} Het signaal stuurde de groeiende keten de blaasjes in, waar een
signaalpeptidase het verwijderde; de membraan schermt het eiwit van het protease
af.
\textbf{20.} Alleen eiwitten met een signaal komen het ER binnen; een cytosolisch
eiwit mist het en blijft buiten.
\textbf{21.} Het enzym wordt op vrije ribosomen gemaakt en blijft in het \omterm{def:b1:eukaryotic-cell:organelle}{cytosol};
als verteringsenzym (als inactieve voorloper richt het weinig schade aan, maar
eenmaal geactiveerd) zou het de \omterm{def:b1:cell-unit-of-life:cell}{cel} van binnenuit verteren.
\textbf{22.} In het trans-golgi: het lysosomale enzym draagt een onderscheidend
merk (een gefosforyleerde suiker) dat door een receptor wordt herkend die het in
blaasjes voor de \omterm{def:b1:eukaryotic-cell:endomembrane}{lysosomen} verpakt in plaats van in secretiekorrels.
\textbf{23.} In het ER (en het \omterm{def:b1:eukaryotic-cell:endomembrane}{golgi-apparaat}): het vormen van blaasjes, het
motortransport langs \omterm{def:b1:eukaryotic-cell:cytoskeleton}{microtubuli} en de membraanversmelting verbruiken alle ATP;
en de eiwitsynthese zelf stopt, zodat de puls niet eens wordt voltooid.
\textbf{24.} \qty{45}{min} transport bij $0.87$ korrels per minuut: ongeveer
\qty{40}{} korrels aan eiwit onderweg.
\textbf{25.} Transport van synthese tot korrel ongeveer \qty{45}{min}
(\qty{55}{min} het ER meegerekend), afscheiding na ongeveer twee uur in de proef
(vier uur of meer in de levende klier, omdat korrels op een maaltijd wachten);
\qty{0.9}{} korrel per minuut vrijgegeven; \qty{7}{\%} van de massa van de \omterm{def:b1:cell-unit-of-life:cell}{cel}
per dag.
\end{solution}
