\chapter{الهضم والامتصاص}\label{ch:b1:digestion-absorption}

بقرة تأكل عشرة كيلوغرامات من العشب في اليوم وتعيش على \omterm{def:b1:water-small-molecules:ph}{حمض} الخل؛
وحصان يأكل العشب نفسه ويعيش، في جزء منه، على الغلوكوز؛ وأرنب يأكل
برازه الليلي ويمرض لولاه. ولا يستطيع أي منها \omterm{def:b1:digestion-absorption:nutrition}{هضم} السللوز، وكلها
تهضمه، بأن تؤوي بكتيريا تستطيعه. والإنسان، وهو لا يستطيعه، يهضم
وجبة من خبز ولحم وزبد إلى \omterm{def:b1:carbohydrates:monosaccharide}{سكاكر} \omterm{def:b1:proteins:aminoacid}{وأحماض أمينية} \omterm{def:b1:lipids:fattyacid}{وحموض دسمة} في بضع
ساعات، ويمتص تسعة لترات من الماء في اليوم عبر جدار بسماكة \omterm{def:b1:cell-unit-of-life:cell}{خلية}
واحدة. ويصف هذا الفصل \omterm{prop:b1:digestion-absorption:tract}{السبيل الهضمي} عند الثدييات، \omterm{def:b1:enzymes:enzyme}{والإنزيمات}
والإفرازات التي تفكك الغذاء، \omterm{def:b1:cell-unit-of-life:cell}{والخلايا} التي تدخل القطع، والإشارات
التي توقّت ذلك كله، والشراكات مع الجراثيم التي تعيش بها آكلات
العشب على ما لا يمسه \omterm{def:b1:enzymes:enzyme}{إنزيم} ثديي.

\section{التغذي غير الذاتي}

\begin{definition}[التغذية والهضم والامتصاص]\label{def:b1:digestion-absorption:nutrition}
\index{هضم}\index{امتصاص}
على \omterm{def:b1:organism-environment:trophy}{غير ذاتي التغذي} أن يأخذ من غذائه الطاقة والكربون، \omterm{def:b1:proteins:aminoacid}{والأحماض
الأمينية} \omterm{def:b1:lipids:fattyacid}{والحموض الدسمة} التي لا يستطيع صنعها (\cref{ch:b1:biosyntheses-integration})،
والفيتامينات والمعادن والماء. ويأتي الغذاء جزيئات كبيرة --- نشاء
وبروتينًا ودهنًا --- لا تستطيع عبور غشاء. و\emph{الهضم} هو
حلمأتها، \omterm{def:b1:enzymes:enzyme}{بالإنزيمات}، إلى مونوميرات صغيرة بما يكفي لالتقاطها:
غلوكوز، \omterm{def:b1:proteins:aminoacid}{وأحماض أمينية}، \omterm{def:b1:lipids:fattyacid}{وحموض دسمة} وغليسرول، \omterm{def:b1:nucleic-acids:nucleotide}{ونوكليوتيدات}.
و\emph{الامتصاص} هو نقل تلك المونوميرات عبر \omterm{def:b1:body-plans-tissues:epithelium}{ظهارة} الأمعاء إلى الدم
أو \omterm{def:b1:mammal-organization:compartments}{اللمف}. والهضم في الحيوانات \emph{خارج خلوي}، في لمعة أنبوب
مفتوح من طرفيه، وهو ما يتيح معالجة الغذاء على مراحل وهو يتحرك،
ويتيح لمناطق الأنبوب أن تتخصص.
\end{definition}

\begin{proposition}[السبيل الهضمي عند الثدييات]\label{prop:b1:digestion-absorption:tract}
\index{سبيل هضمي}
من الفم إلى الشرج ينقسم أنبوب الثدييات مناطق، لكل منها ظهارتها
وإفرازاتها وعضلها: \emph{الفم}، حيث تطحن الأسنان ويرطّب اللعاب
ويبدأ \omterm{def:b1:digestion-absorption:nutrition}{هضم} النشاء؛ و\emph{المريء}، وهو \omterm{def:b1:membranes-transport:transporters}{قناة}؛ و\emph{المعدة}، وهي
كيس حمضي تُنشر فيه \omterm{def:b1:proteins:peptide}{البروتينات} وتُقص، ويُحبس الغذاء ويُطلق عجينة
(الكيموس)؛ و\emph{المعي الدقيق} --- الاثنا عشري والصائم واللفائفي،
ستة أمتار في الإنسان --- حيث تكمل إفرازات \emph{الكبد} (الصفراء)
و\emph{البنكرياس} (\omterm{def:b1:enzymes:enzyme}{إنزيمات} وبيكربونات) الهضمَ وحيث يجري \omterm{def:b1:digestion-absorption:nutrition}{الامتصاص}
كله تقريبًا؛ و\emph{المعي الغليظ}، حيث يُستعاد الماء ويخمّر مجتمع
جرثومي كثيف ما بقي؛ ثم المستقيم. والعضل في الجدار
(\cref{ch:b1:body-plans-tissues}) يخلط المحتويات ويدفعها
بحركة \emph{التمعّج}؛ والعاصرات تحبسها عند كل مرحلة.
\end{proposition}

\section{تفكيك الغذاء}

كل \omterm{def:b1:enzymes:enzyme}{إنزيم} منها حلمأة (\cref{ch:b1:enzymes}) نوعية لصنف واحد من
الروابط؛ وكل منها يعمل عند pH منطقته؛ والبروتيازات تُصنع وتُخزن
زيموجينات خاملة، لا تُنشَّط إلا في اللمعة. والصفراء، التي يصنعها
الكبد وتُخزن في المرارة، لا \omterm{def:b1:enzymes:enzyme}{إنزيم} فيها: بل \emph{أملاح الصفراء}

سطحها، وتحمل النواتج في مذيلات إلى \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} الماصة
(\cref{ch:b1:lipids}).
\end{proposition}

\begin{example}[كيمياء وجبة]\label{ex:b1:digestion-absorption:meal}
شريحة خبز بزبد وجبن: يهاجم أميلازُ اللعاب النشاءَ في الفم، ويوقفه
حمضُ المعدة، ويستأنفه أميلاز البنكرياس في الاثني عشر، ويكمله
مالتاز الحافة الفرشاتية على \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} المعوية نفسها؛ \omterm{def:b1:proteins:peptide}{وبروتين} الجبن
ينشره \omterm{def:b1:water-small-molecules:ph}{الحمض}، ويقصه الببسين، ويقصه أكثر التربسين والكيموتربسين،
ويُكمَل إلى \omterm{def:b1:proteins:aminoacid}{أحماض أمينية} وثنائيات ببتيد عند الحافة الفرشاتية؛
والزبد يذوب، وتستحلبه الصفراء، ويشطره الليباز إلى \omterm{def:b1:lipids:fattyacid}{حموض دسمة}
وأحاديات غليسريد، ويُعبَّر به في مذيلات إلى الغشاء. ثلاث ساعات من
الفم إلى الدم.
\end{example}

\begin{proposition}[التحكم في الإفراز]\label{prop:b1:digestion-absorption:control}
\index{غاسترين}\index{سكرتين}\index{كوليسيستوكينين}
الإفراز موقّت على الغذاء. فرؤية الوجبة وشمّها، عبر العصب المبهم،
تبدآن سيلان اللعاب والعصارة المعدية قبل اللقمة الأولى. والغذاء في
المعدة يمطّ جدارها وتحرر ببتيداته هرمون \emph{الغاسترين}، الذي
يحفّز إفراز \omterm{def:b1:water-small-molecules:ph}{الحمض}؛ \omterm{def:b1:water-small-molecules:ph}{والحمض} الداخل إلى الاثني عشر يحرر
\emph{السكرتين}، الذي يجعل البنكرياس يصبّ البيكربونات لتعديله؛
\omterm{def:b1:lipids:fattyacid}{والدهن} والببتيدات في الاثني عشر تحرر \emph{الكوليسيستوكينين}، الذي
يجعل المرارة تتقلص والبنكرياس يطلق إنزيماته، ويبطئ إفراغ المعدة
حتى لا يُغرق المعي. فكل إفراز يستدعيه الركيزة التي يعمل عليها.
\end{proposition}
\begin{proof}[شواهد]
راقب بومون (1833) معدة رجل بقي فيها جرح مفتوح، فرأى العصارة
المعدية لا تظهر إلا حين يلمس الغذاء الجدار، وتهضم اللحم في كأس كما
تهضمه في المعدة. وركّب بافلوف (في تسعينيات القرن التاسع عشر) على
الكلاب جيوبًا من المعدة ونواسير من قناة البنكرياس: فكانت رؤية
الغذاء تجعل المعدة تفرز (وإذا قُطع المبهم لم تفرز)؛ وكان \omterm{def:b1:water-small-molecules:ph}{الحمض}
الموضوع في الاثني عشر يجعل البنكرياس يفرز حتى بعد قطع كل عصب، وبيّن
بايليس وستارلينغ (1902) أن مستخلصًا من بطانة الاثني عشر يُحقن في
الدم يفعل الشيء نفسه --- وهو أول هرمون، السكرتين.
\end{proof}

\section{إدخال القطع}

\begin{definition}[السطح الماص]\label{def:b1:digestion-absorption:surface}
\index{زغابة}\index{خلية معوية}
يطوي المعي الدقيق بطانته طيات دائرية، ثم \emph{زغابات} ارتفاعها
ميليمتر، تغطي كل واحدة منها طبقة واحدة من \emph{\omterm{def:b1:cell-unit-of-life:cell}{الخلايا} المعوية}
تحمل أغشيتها القمّية زُغيبات --- وهي \emph{الحافة الفرشاتية} ---
حتى إن أنبوبًا مساحته \qty{0.6}{m^2} يعرض نحو
\qty{200}{m^2} من الغشاء (\cref{ch:b1:body-plans-tissues}). وفي
كل زغابة شبكة شعيرات تحمل \omterm{def:b1:carbohydrates:monosaccharide}{السكاكر} \omterm{def:b1:proteins:aminoacid}{والأحماض الأمينية} إلى الكبد عبر
الوريد البابي، و\emph{وعاء لبني} مركزي، وهو وعاء لمفي يحمل \omterm{def:b1:lipids:fattyacid}{الدهن}.
وتعيش \omterm{def:b1:cell-unit-of-life:cell}{الخلايا} المعوية أربعة أيام وتُستبدل من الخبايا بين الزغابات.
\end{definition}

\begin{proposition}[طرق الامتصاص]\label{prop:b1:digestion-absorption:routes}
\index{كيلوميكرون}
\emph{\omterm{def:b1:carbohydrates:monosaccharide}{السكاكر}}: يدخل الغلوكوز والغالاكتوز \omterm{def:b1:digestion-absorption:surface}{الخلية المعوية} \omterm{def:b1:membranes-transport:transporters}{بالناقل}
المشترك صوديوم-غلوكوز (\cref{ch:b1:membranes-transport})، ويدخل
الفركتوز بحامل، ويخرج الجميع بحامل على الجانب القاعدي إلى الدم.
و\emph{\omterm{def:b1:proteins:aminoacid}{الأحماض الأمينية}} تدخل بعدة نواقل مشتركة مقترنة بالصوديوم،
والببتيدات الصغيرة \omterm{def:b1:membranes-transport:transporters}{بناقل} مقترن بالبروتون، لتُقص داخل \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.
و\emph{\omterm{def:b1:lipids:fattyacid}{الدهون}}: تغادر \omterm{def:b1:lipids:fattyacid}{الحموض الدسمة} وأحاديات الغليسريد مذيلاتها
وتنتشر عبر الغشاء القمّي؛ وداخل \omterm{def:b1:cell-unit-of-life:cell}{الخلية} تُعاد أسترتها إلى \omterm{def:b1:lipids:triglyceride}{ثلاثيات
غليسريد}، وتُغلَّف \omterm{def:b1:proteins:peptide}{ببروتين} في \emph{كيلوميكرونات} قطرها ميكرومتر،
وتُطرح \omterm{prop:b1:eukaryotic-cell:secretory}{بالإخراج الخلوي} إلى الوعاء اللبني، الذي لا يبلغ لمفه الدم
إلا عند العنق --- \omterm{def:b1:lipids:fattyacid}{فالدهن} يتجاوز الكبد. و\emph{الماء} يتبع
المنحلات تناضحيًا: فتسعة لترات (\qty{9}{L}) تدخل الأمعاء في اليوم
(اثنان مشروبان وسبعة مفرزة لعابًا وعصارة معدية وصفراء وعصارة
بنكرياسية ومعوية)، ويُعاد امتصاصها كلها إلا عُشر لتر، أكثرها في
المعي الدقيق والباقي في القولون. و\emph{الفيتامينات}
و\emph{الحديد} و\emph{الكالسيوم} لكل منها حوامله؛ والفيتامين
B$_{12}$ يحتاج إلى \omterm{def:b1:proteins:peptide}{بروتين} من المعدة ليُمتص أصلًا.
\end{proposition}

\begin{method}[قراءة تجربة امتصاص]\label{met:b1:digestion-absorption:everted}
\begin{enumerate}
  \item اقلب قطعة من المعي رأسًا على عقب (\emph{كيسًا مقلوبًا})،
        واملأها بمحلول ملحي، واغمرها في محلول المادة: فما يظهر في
        الداخل قد عبر \omterm{def:b1:body-plans-tissues:epithelium}{الظهارة} من اللمعة السابقة.
  \item قِس النقل في مقابل التركيز: فإن كان يشبع ويوقفه البرد أو
        سمّ لتخليق \omterm{def:b1:water-small-molecules:atp}{ATP} فهو نقل فاعل أو بحامل؛ وإن كان خطيًا غير
        حساس فهو انتشار.
  \item انزع الصوديوم من الحمام: فإن انهار نقل الغلوكوز أو \omterm{def:b1:proteins:aminoacid}{الحمض
        الأميني} فقد كان مقترنًا بالصوديوم. وأضف سكرًا منافسًا: فإن
        هبط النقل فالاثنان يتقاسمان حاملًا.
  \item قارن بين القطع: فأكثر \omterm{def:b1:carbohydrates:monosaccharide}{السكاكر} \omterm{def:b1:proteins:aminoacid}{والأحماض الأمينية} تعبر في
        الصائم، وأملاح الصفراء والفيتامين B$_{12}$ في اللفائفي
        وحده، والماء والملح في كل مكان.
\end{enumerate}
\end{method}

\begin{example}[ميزانية الماء]\label{ex:b1:digestion-absorption:water}
من اللترات التسعة التي تدخل أمعاء الإنسان كل يوم يمتص المعي الدقيق
ثمانية، والقولون واحدًا آخر، ويغادر \qty{0.1}{L} في البراز. والتقاط
المعي الدقيق للغلوكوز المقترن بالصوديوم يجرّ الماء معه --- ولهذا
يعيد محلول من الغلوكوز والملح، يُؤخذ بالفم، إماهة طفل مصاب
بالكوليرا حين يكون الذيفان قد حوّل الخبايا إلى مفرزات: \omterm{def:b1:membranes-transport:transporters}{فالناقل}
المشترك سليم ويمتص الماء أسرع مما يفقده الذيفان
(\cref{ch:b1:membranes-transport}).
\end{example}

\section{العيش على العشب}

\begin{proposition}[هضم السللوز بالتعايش]\label{prop:b1:digestion-absorption:cellulose}
\index{كرش}\index{مجترّ}\index{تخمر المعي الخلفي}
لا يصنع أي ثديي سللولازًا. وآكلات العشب تهضم السللوز بإيواء
بكتيريا وطلائعيات وفطريات تستطيعه، في حجرة تخمير تكسر فيها
الجراثيم الروابط $\beta(1{\to}4)$ (\cref{ch:b1:carbohydrates})
وتخمّر \omterm{def:b1:carbohydrates:monosaccharide}{السكاكر}، بلا أكسجين، إلى \emph{\omterm{def:b1:lipids:fattyacid}{حموض دسمة} طيارة}
--- أسيتات وبروبيونات وبوتيرات --- زائد ثاني أكسيد الكربون
والميثان. ويمتص المضيف الأحماض ويعيش عليها. و\emph{المجترّات}
(الأبقار والأغنام والأيائل) تضع الحجرة \emph{قبل} المعدة الحقيقية:
وهي \emph{الكرش}، حوض سعته مئة لتر في البقرة، يُخمَّر فيه الغذاء
يومًا أو يومين، ويُجترّ ويُمضغ ثانية، ومنه تمر الجراثيم نفسها إلى
المنفحة والمعي لتُهضم بوصفها مصدر البقرة الرئيس \omterm{def:b1:proteins:peptide}{للبروتين}.
أما \emph{مخمّرات المعي الخلفي} (الخيول والأرانب والفيلة) فتضع
الحجرة \emph{بعد} المعي الدقيق، في أعور وقولون متضخمين: فتهضم
النشاء \omterm{def:b1:proteins:peptide}{والبروتين} بنفسها أولًا، وتخمّر الألياف بعدئذ، وتخسر
\omterm{def:b1:proteins:peptide}{البروتين} الجرثومي في برازها --- إلا إن أكلت، كالأرنب، برازها
الليلي اللين وأمرّته مرة ثانية (\emph{أكل البراز الأعوري}).
\end{proposition}

\begin{example}[طريقان لأكل مرج]\label{ex:b1:digestion-absorption:cowhorse}
تخمّر البقرة ثلثي الألياف وتحصل على ستين في المئة من طاقتها أسيتات
وبروبيونات --- ومنها يجب أن يصنع كبدها كل غرام من الغلوكوز تحتاج
إليه، إذ لا يكاد يبلغ معيَها منه شيء؛ وكرشها كذلك يحوّل نتروجين
البولة، المعاد من دمها إلى لعابها، إلى \omterm{def:b1:proteins:peptide}{بروتين} جرثومي، فتستطيع أن
تعيش على تبن أفقر \omterm{def:b1:proteins:peptide}{بالبروتين} مما يستطيعه أي ثديي آخر. أما الحصان
فيخمّر أقل من نصف الألياف لكنه يهضم نشاء العشب وبروتينه بنفسه،
ويمتص الغلوكوز، ويمرّر الغذاء في يوم ونصف بدل ثلاثة: فيحصل على أقل
من كل كيلوغرام ويأكل كيلوغرامات أكثر. فالبقرة تربح في المرعى
الفقير، والحصان حين يجب أن يجري.
\end{example}

\begin{remark}[طول الأمعاء]\label{rem:b1:digestion-absorption:length}
معي آكل اللحم أربعة أمثال طول جسمه، ومعي آكل العشب عشرة إلى عشرين
مثلًا: فكلما فقر الغذاء طال الأنبوب وكبرت الحجرات. وللشرغوف الذي
يأكل الطحالب معي طويل ملتف يقصر فجأة عند التحول حين يقبل الضفدع
على الحشرات؛ وهي \omterm{def:b1:nucleic-acids:nucleotide}{القاعدة} نفسها، مكتوبة في عمر حيوان واحد.
\end{remark}

\section{مسألة: البقرة والحصان}

\begin{problem}[{مسألة نهاية الأسبوع --- عشرة كيلوغرامات من العشب
عبر كرش وعبر أعور: الألياف المخمَّرة، والأحماض الممتصة، والبروتين
الجرثومي المكسوب أو المفقود، والميثان المنفوث، وختامًا كسر طاقة
البقرة الآتي من الحموض الدسمة الطيارة}]
\label{pb:b1:digestion-absorption:1}
تأكل بقرة وحصان كل منهما \qty{10}{kg} من المادة الجافة للعشب في
اليوم، فيها بالكتلة \qty{50}{\%} سللوز وهيميسللوز (ألياف)،
و\qty{20}{\%} نشاء \omterm{def:b1:carbohydrates:monosaccharide}{وسكاكر}، و\qty{15}{\%} \omterm{def:b1:proteins:peptide}{بروتين}، و\qty{15}{\%}
لغنين ورماد غير قابلين \omterm{def:b1:digestion-absorption:nutrition}{للهضم}. والطاقة: \omterm{def:b1:carbohydrates:monosaccharide}{الكربوهيدرات} \omterm{def:b1:proteins:peptide}{والبروتين}
\qty{17.5}{kJ/g}. وتخمير \qty{1}{g} من \omterm{def:b1:carbohydrates:monosaccharide}{الكربوهيدرات} يعطي حموضًا
دسمة طيارة تحمل \qty{75}{\%} من طاقته، وميثانًا يحمل
\qty{8}{\%}، والحرارةَ ونموَّ الجراثيم الباقي؛ ويعطي كذلك
\qty{0.15}{g} من \omterm{def:b1:proteins:peptide}{البروتين} الجرثومي. ويخمّر الكرش \qty{70}{\%} من
الألياف وكل النشاء \omterm{def:b1:carbohydrates:monosaccharide}{والسكاكر}؛ ويخمّر معي الحصان الخلفي \qty{45}{\%}
من الألياف، بينما يهضم معيه الدقيق \qty{90}{\%} من النشاء
و\qty{70}{\%} من \omterm{def:b1:proteins:peptide}{البروتين} قبل أن تبلغ الألياف المعي الخلفي.
ويحتاج الحيوانان كلاهما إلى \qty{110}{MJ} من الطاقة الأيضية في
اليوم.

\medskip
\noindent\textbf{الجزء الأول --- كرش البقرة.}
\begin{enumerate}
  \item احسب كتلة الألياف، وكتلة النشاء \omterm{def:b1:carbohydrates:monosaccharide}{والسكاكر}، وكتلة \omterm{def:b1:proteins:peptide}{البروتين}
        المأكولة في اليوم.
  \item احسب كتلة \omterm{def:b1:carbohydrates:monosaccharide}{الكربوهيدرات} المخمَّرة في الكرش.
  \item احسب طاقة تلك \omterm{def:b1:carbohydrates:monosaccharide}{الكربوهيدرات} والطاقة المستعادة حموضًا دسمة
        طيارة.
  \item احسب الطاقة المفقودة ميثانًا، وكتلة الميثان
        (\qty{55}{kJ/g}) وحجمه (\qty{1.4}{L/g}).
  \item احسب \omterm{def:b1:proteins:peptide}{البروتين} الجرثومي المنتَج.
  \item فسّر لماذا يكون مورد البقرة من \omterm{def:b1:proteins:peptide}{البروتين} هو الجراثيم في
        معظمه، وأين تُهضم.
  \item احسب كسر الألياف الذي يغادر الكرش غير مخمَّر، ومصيره.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- ميزانية البقرة.}
وتهضم البقرة كذلك \qty{60}{\%} من \omterm{def:b1:proteins:peptide}{بروتين} الغذاء الذي يفلت من
الكرش (ويفلت \qty{40}{\%} من \omterm{def:b1:proteins:peptide}{البروتين} المأكول) وكل \omterm{def:b1:proteins:peptide}{البروتين}
الجرثومي، عند \qty{17.5}{kJ/g}.
\begin{enumerate}[resume]
  \item احسب الطاقة التي تحصل عليها البقرة من \omterm{def:b1:proteins:peptide}{البروتين} (الفالت من
        الغذاء زائد الجرثومي).
  \item احسب طاقة البقرة الأيضية الكلية: \omterm{def:b1:lipids:fattyacid}{الحموض الدسمة} الطيارة
        زائد \omterm{def:b1:proteins:peptide}{البروتين}.
  \item احسب كسر تلك الطاقة الموفَّر حموضًا دسمة طيارة.
  \item هل تبلغ البقرة حاجتها البالغة \qty{110}{MJ}؟ وماذا تفعل إن
        لم تبلغها؟
  \item لا يكاد يُمتص غلوكوز عند البقرة. سمّ \omterm{def:b1:water-small-molecules:ph}{الحمض} الذي يصنع كبدها
        منه الغلوكوز، وسمّ العملية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- معي الحصان الخلفي.}
\begin{enumerate}[resume]
  \item احسب الطاقة التي يحصل عليها الحصان من النشاء المهضوم في
        معيه الدقيق (الممتص غلوكوزًا، \qty{17.5}{kJ/g}).
  \item احسب الطاقة من \omterm{def:b1:proteins:peptide}{بروتين} الغذاء المهضوم في المعي الدقيق.
  \item احسب الألياف المخمَّرة في المعي الخلفي والطاقة المستعادة
        حموضًا دسمة طيارة.
  \item احسب \omterm{def:b1:proteins:peptide}{البروتين} الجرثومي المنتَج في المعي الخلفي. وماذا يصير
        إليه؟
  \item احسب طاقة الحصان الأيضية الكلية والكسر الموفَّر أحماضًا
        دسمة طيارة.
  \item هل يبلغ الحصان \qty{110}{MJ}؟ وكم عشبًا يحتاج أن يأكل، وكيف
        يدبّر ذلك؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- مقارنة التصميمين.}
\begin{enumerate}[resume]
  \item احسب الطاقة التي يستخلصها كل حيوان من كيلوغرام المادة
        الجافة للعشب.
  \item ميثان البقرة \qty{8}{\%} من الطاقة المخمَّرة. احسب حصته من
        مدخول البقرة من الطاقة الإجمالية، وكتلة الميثان الذي ينفثه
        قطيع من مئة بقرة في السنة.
  \item يبقى غذاء البقرة \qty{70}{h} في السبيل، وغذاء الحصان
        \qty{36}{h}. احسب كتلة المادة الجافة التي يحملها كل منهما
        في أمعائه في أي لحظة، عند مدخول ثابت.
  \item الأرنب يخمّر الألياف في أعوره ثم يأكل برازه الأعوري. فأي
        كسر من \omterm{def:b1:proteins:peptide}{البروتين} الجرثومي في السؤال 16 كان يستطيع الحصان
        استعادته لو فعل مثله؟ ولماذا لا يفعله حصان؟
  \item فسّر لماذا تستطيع البقرة أن تعيش على نظام من القش فيه
        \qty{4}{\%} \omterm{def:b1:proteins:peptide}{بروتين} زائد بولة، ولا يستطيعه الحصان.
  \item حصان عدّاء يحرق غلوكوزًا من غليكوجينه هو؛ والبقرة لا
        تستطيع العدو. اربط هذا بالتصميمين.
  \item اذكر النتيجة: كسر طاقة البقرة الأيضية الآتي من الأحماض
        الدسمة الطيارة، وكسر الحصان، والطاقة التي يستخلصها كل
        منهما من كيلوغرام العشب.
\end{enumerate}
\end{problem}
