\chapter{The Five Senses}\label{ch:g1:the-five-senses}

Close your \omterm{def:g1:the-five-senses:sense}{eyes} and the world does not disappear: you still hear
the classroom, smell lunch cooking, feel the chair under you. The
body has five doors to the world --- the senses --- and each door
is an organ of the body, working for you all day long.

\section{Five senses, five organs}

\begin{definition}[Sense and sense organ]\label{def:g1:the-five-senses:sense}
A \emph{sense}\index{sense} is a way the body finds out about the
world. Each sense works through a \emph{sense
organ}\index{sense organ}, a part of the body built for that job:
\begin{enumerate}
  \item \emph{sight}\index{sight} --- the \emph{eyes}\index{eye}
        see light, colors and shapes;
  \item \emph{hearing}\index{hearing} --- the \emph{ears}\index{ear}
        hear sounds;
  \item \emph{smell}\index{smell} --- the \emph{nose}\index{nose}
        smells odors;
  \item \emph{taste}\index{taste} --- the \emph{tongue}\index{tongue}
        tastes food;
  \item \emph{touch}\index{touch} --- the \emph{skin}\index{skin},
        all over the body, feels rough and smooth, warm and cold.
\end{enumerate}
\end{definition}

\begin{example}[Breakfast, sense by sense]\label{ex:g1:the-five-senses:breakfast}
One bowl of hot chocolate uses all five: the \omterm{def:g1:the-five-senses:sense}{eyes} see the steam,
the nose smells the cocoa, the \omterm{def:g1:the-five-senses:sense}{skin} of the hands feels the warm
bowl, the \omterm{def:g1:the-five-senses:sense}{ears} hear the spoon clink, and the tongue --- the best
part --- tastes it.
\end{example}

\begin{remark}[Touch is everywhere]\label{rem:g1:the-five-senses:skin}
Four \omterm{def:g1:the-five-senses:sense}{sense organs} live on the head, but the organ of touch is the
\omterm{def:g1:the-five-senses:sense}{skin}, and the \omterm{def:g1:the-five-senses:sense}{skin} covers you from head to toe. That is why you can
feel a breeze on your neck, a pebble in your shoe and a ladybug
landing on your \omterm{def:g1:my-body:parts}{arm} --- all without looking.
\end{remark}

\section{The senses report to the brain}

\begin{proposition}[Messages to the brain]\label{prop:g1:the-five-senses:brain}
A \omterm{def:g1:the-five-senses:sense}{sense organ} does not understand what it senses --- it \emph{sends
a message}. All the messages travel inside the body to the
\emph{brain}\index{brain}, in the head, which puts them together:
``that is Grandma's voice'', ``that smell is toast''. Without the
brain, the \omterm{def:g1:the-five-senses:sense}{eyes} and \omterm{def:g1:the-five-senses:sense}{ears} would report to no one.
\end{proposition}
\admitted

\begin{example}[Naming things in the dark]\label{ex:g1:the-five-senses:dark}
A friend puts a familiar object in your hands while your \omterm{def:g1:the-five-senses:sense}{eyes} are
closed: a key, a sponge, a spoon. You name it almost at once. The
\omterm{def:g1:the-five-senses:sense}{skin} sent its message --- cold, hard, jagged --- and the brain
answered: ``a key''. Senses collect; the brain understands.
\end{example}

\begin{remark}[Helping each other]\label{rem:g1:the-five-senses:together}
The senses work as a team. Food ``tastes'' dull when a cold blocks
your nose, because smell was doing half the work of flavor. And
when one sense cannot do its job at all, the others learn to do
more: people who cannot see read with their fingertips, using
touch, and cross the street with their \omterm{def:g1:the-five-senses:sense}{ears}.
\end{remark}

\section{Taking care of the sense organs}

\begin{proposition}[Fragile doors]\label{prop:g1:the-five-senses:fragile}
\omterm{def:g1:the-five-senses:sense}{Sense organs} are precious and easy to hurt. Too-loud sound hurts
the \omterm{def:g1:the-five-senses:sense}{ears}; looking at the Sun hurts the \omterm{def:g1:the-five-senses:sense}{eyes}; a too-hot spoonful
burns the tongue. A hurt \omterm{def:g1:the-five-senses:sense}{sense organ} may stay hurt --- so we
protect them.
\end{proposition}
\admitted

\begin{method}[Protecting my senses]\label{met:g1:the-five-senses:protect}
\begin{enumerate}
  \item Never look straight at the Sun, not even for a moment;
  \item keep sounds gentle: no shouting into \omterm{def:g1:the-five-senses:sense}{ears}, headphones at
        low volume;
  \item never push anything into your \omterm{def:g1:the-five-senses:sense}{ear} or your nose;
  \item blow on hot food before tasting;
  \item read and draw with good light.
\end{enumerate}
\end{method}

\begin{example}[The concert next door]\label{ex:g1:the-five-senses:loud}
After very loud music, \omterm{def:g1:the-five-senses:sense}{ears} sometimes whistle for a while. That
whistling is the \omterm{def:g1:the-five-senses:sense}{ear} saying ``too much!''. People who work with
loud machines wear \omterm{def:g1:the-five-senses:sense}{ear} covers for exactly this reason --- their
\omterm{def:g1:the-five-senses:sense}{ears} must last them a lifetime.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:1}
Name the five senses and, for each, its \omterm{def:g1:the-five-senses:sense}{sense organ}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:2}
Which \omterm{def:g1:the-five-senses:sense}{sense organ} reports: (a) the color of a balloon? (b) the
softness of a blanket? (c) the smell of rain? (d) the sound of a
bell?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:3}
Which \omterm{def:g1:the-five-senses:sense}{sense organ} is not on the head? How far does it reach?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:4}
In \cref{ex:g1:the-five-senses:dark}, the \omterm{def:g1:the-five-senses:sense}{eyes} were closed. Which
organ sent the message, and which part of the body understood it?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:5}
With a cold and a blocked nose, food seems to lose its flavor.
Which two senses usually team up when you eat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:6}
Give three rules of \cref{met:g1:the-five-senses:protect}, and say
which \omterm{def:g1:the-five-senses:sense}{sense organ} each one protects.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:7}
Why do people who work with loud machines wear \omterm{def:g1:the-five-senses:sense}{ear} covers? Use
\cref{prop:g1:the-five-senses:fragile}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:the-five-senses:8}
Try it: with \omterm{def:g1:the-five-senses:sense}{eyes} closed, let a helper hold three foods under your
nose --- perhaps banana, cheese, orange. How many did your nose
name correctly, without the \omterm{def:g1:the-five-senses:sense}{eyes}' help?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:the-five-senses:9}
A friend claims: ``The \omterm{def:g1:the-five-senses:sense}{eye} understands what it sees.'' Correct the
sentence using \cref{prop:g1:the-five-senses:brain}: what does the
\omterm{def:g1:the-five-senses:sense}{eye} really do, and who does the understanding?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:the-five-senses:10}
How can a person who cannot see ``read'' a book, and cross a
street safely? Say which senses take over which jobs.
\end{exercise}
