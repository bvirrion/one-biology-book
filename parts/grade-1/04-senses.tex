\chapter{The Senses}\label{ch:g1:senses}

How do you know the world around you? You see a colour, hear a
sound, \omterm{def:g1:senses:five}{smell} bread, \omterm{def:g1:senses:five}{taste} an apple, feel a soft toy. Your \omterm{def:g1:my-body:body}{body} uses
five main \emph{\omterm{def:g1:senses:sense}{senses}}.

\section{Five senses}

\begin{definition}[Sense]\label{def:g1:senses:sense}
A \emph{sense}\index{sense} is a way your \omterm{def:g1:my-body:body}{body} notices the world.
The five main senses are \omterm{def:g1:senses:five}{sight}, \omterm{def:g1:senses:five}{hearing}, \omterm{def:g1:senses:five}{smell}, \omterm{def:g1:senses:five}{taste} and \omterm{def:g1:senses:five}{touch}.
\end{definition}

\omimg{five-senses.jpg}{0.52\linewidth}{\omterm{def:g1:senses:five}{Eye}, \omterm{def:g1:senses:five}{ear}, \omterm{def:g1:senses:five}{nose}, \omterm{def:g1:senses:five}{tongue} and
\omterm{def:g1:my-body:parts}{hand} --- five tools for exploring the world.}

\begin{definition}[The five senses]\label{def:g1:senses:five}
\begin{itemize}
  \item \emph{sight}\index{sight}\index{eye}: eyes see light,
        colours and shapes;
  \item \emph{hearing}\index{hearing}\index{ear}: ears hear sounds;
  \item \emph{smell}\index{smell}\index{nose}: the nose smells
        odours;
  \item \emph{taste}\index{taste}\index{tongue}: the tongue tastes
        food;
  \item \emph{touch}\index{touch}\index{skin}: skin feels heat,
        cold, pressure and texture.
\end{itemize}
\end{definition}

\section{Using the senses safely}

\begin{method}[Explore with care]\label{met:g1:senses:care}
\begin{enumerate}
  \item Look with your \omterm{def:g1:senses:five}{eyes} --- never look straight at the Sun.
  \item Listen, but protect \omterm{def:g1:senses:five}{ears} from very loud noise.
  \item \omterm{def:g1:senses:five}{Smell} gently; do not sniff unknown chemicals.
  \item \omterm{def:g1:senses:five}{Taste} only food that a trusted adult says is safe.
  \item \omterm{def:g1:senses:five}{Touch} carefully: some things are hot, sharp or fragile.
\end{enumerate}
\end{method}

\begin{example}[A picnic]\label{ex:g1:senses:picnic}
You \emph{see} green grass. You \emph{hear} birds. You \emph{\omterm{def:g1:senses:five}{smell}}
fresh bread. You \emph{\omterm{def:g1:senses:five}{taste}} a sweet fruit. You \emph{feel} a
cool breeze on your \omterm{def:g1:senses:five}{skin}. One moment, all five \omterm{def:g1:senses:sense}{senses} at work!
\end{example}

\begin{example}[Guess the sense]\label{ex:g1:senses:guess}
Ringing bell $\to$ \omterm{def:g1:senses:five}{hearing}. Red apple $\to$ \omterm{def:g1:senses:five}{sight} (and later \omterm{def:g1:senses:five}{taste}).
Soft blanket $\to$ \omterm{def:g1:senses:five}{touch}. Baking cake $\to$ \omterm{def:g1:senses:five}{smell}.
\end{example}
