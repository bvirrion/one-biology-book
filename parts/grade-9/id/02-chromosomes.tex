\chapter{Kromosom}\label{ch:g9:chromosomes}

\Cref{rem:g9:heredity-and-traits:question} menyudutkan
penentunya ke dalam \omterm{def:g6:cell-unit-of-life:parts}{inti selnya}. Tangkaplah sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} pada
saat yang tepat dan \omterm{def:g6:cell-unit-of-life:parts}{inti selnya} memperlihatkan kartunya:
isinya menggulung menjadi benda yang dapat dihitung dan
difoto --- \omterm{def:g9:chromosomes:chromosomes}{kromosomnya}. Menghitungnya, di dalam \omterm{def:g5:first-look-at-cells:cell}{sel} tubuh
dan di dalam \omterm{def:g8:sexual-reproduction:gametes}{gametnya}, ternyata menjelaskan hampir segala
yang dituntut album keluarganya: separuhnya, pasangannya,
pengocokannya, dan lemparan koin yang memutuskan apakah
seorang bayi laki-laki atau perempuan.

\section{Inti selnya, tertangkap basah}

\begin{definition}[Kromosom]\label{def:g9:chromosomes:chromosomes}
Sebuah \emph{kromosom}\index{kromosom} adalah benda mirip
benang di dalam \omterm{def:g6:cell-unit-of-life:parts}{inti selnya}, terlihat (bila diwarnai baik,
pada saat pembelahannya) selagi isi \omterm{def:g6:cell-unit-of-life:parts}{inti selnya} menggulung
untuk diangkut. Di antara pembelahannya benang yang sama
terurai dan tak terlihat --- keadaan bekerjanya; pada
pembelahannya ia memadat --- keadaan berpindahnya. Difoto
lalu dipilah, kromosom sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} membentuk
\emph{kariotipe}\index{kariotipe} \omterm{def:g5:first-look-at-cells:cell}{selnya}: setumpuk kartunya,
digelar.
\end{definition}

\begin{proposition}[Bilangan manusianya]\label{prop:g9:chromosomes:counts}
\omterm{def:g9:chromosomes:chromosomes}{Kariotipenya} menyerahkan tiga bilangan yang membawa seluruh
babnya:
\begin{enumerate}
  \item setiap \omterm{def:g5:first-look-at-cells:cell}{sel} tubuh manusia memuat \emph{46} \omterm{def:g9:chromosomes:chromosomes}{kromosom}
        --- \omterm{def:g1:the-five-senses:sense}{kulit}, \omterm{prop:g7:digestion-and-nutrients:absorption}{hati}, \omterm{def:g8:nervous-communication:neuron}{neuron} sama saja (pembelahan pada
        \cref{prop:g5:first-look-at-cells:growth} menyalin
        tumpukannya dengan setia ke setiap \omterm{def:g5:first-look-at-cells:cell}{sel});
  \item bila digelar, ke-46 itu terpilah menjadi \emph{23
        pasang}, dan kedua mitra sepasangnya serupa dalam
        ukuran serta pola pitanya;
  \item \emph{\omterm{def:g8:sexual-reproduction:gametes}{gametnya}} melanggar aturannya: \omterm{def:g4:animal-reproduction:sexual}{sel telur} dan
        \omterm{def:g8:reproductive-systems:male}{sel sperma} mengangkut \emph{23} --- satu \omterm{def:g9:chromosomes:chromosomes}{kromosom}
        dari tiap pasangnya, tidak ada pasang yang lengkap.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Hitungan pembuahannya]\label{ex:g9:chromosomes:arithmetic}
Bilangannya berderap bersama seperti sebuah bukti. \omterm{def:g8:sexual-reproduction:gametes}{Gametnya}
masing-masing 23; \omterm{def:g5:flower-to-fruit:steps}{pembuahannya}
(\cref{prop:g8:sexual-reproduction:core}) menjumlahkannya:
\omterm{prop:g8:sexual-reproduction:core}{zigotnya} memuat $23 + 23 = 46$ --- pasangannya pulih, satu
mitra tiap pasangnya dari ibunya, satu dari ayahnya. Setiap
\omterm{def:g5:first-look-at-cells:cell}{sel} tubuh anaknya lalu mengangkut sumbangan kedua orang
tuanya --- dan itu persis yang diamati
\cref{rem:g4:animal-reproduction:mix} dari luar, anak kuda
dan belangnya, satu angkatan bab yang lalu.
\end{example}

\begin{remark}[Mengapa gametnya harus separuh]\label{rem:g9:chromosomes:whyhalve}
Andaikan tidak: \omterm{def:g8:sexual-reproduction:gametes}{gamet} berkromosom 46 akan membuat anak
berkromosom 92, cucu berkromosom 184 --- tumpukannya
berlipat dua tiap angkatan. Pembelahan pembagi dua yang
khusus pada \omterm{def:g8:sexual-reproduction:gametes}{gametnya} --- masing-masing menerima \emph{satu
mitra dari tiap pasangnya} --- itulah yang menjaga bilangan
spesiesnya tetap 46, angkatan demi angkatan. Dan perhatikan
bonusnya: \emph{mitra yang mana} dari tiap pasang yang
diterima sebuah \omterm{def:g8:sexual-reproduction:gametes}{gamet} diputuskan secara bebas, pasang demi
pasang --- $2^{23}$ tumpukan yang mungkin dari satu orang
tua, pengocokan pada
\cref{prop:g8:sexual-reproduction:shuffle} akhirnya terhitung.
\end{remark}

\section{Kromosomnya mengangkut penentunya}

\begin{proposition}[Bukti bahwa tumpukannya penting]\label{prop:g9:chromosomes:evidence}
Bahwa \omterm{def:g9:chromosomes:chromosomes}{kromosomnya} \emph{adalah} kendaraan penentunya
bukanlah anggapan melainkan bukti:
\begin{itemize}
  \item ia berperilaku persis seperti yang dituntut
        \cref{prop:g9:heredity-and-traits:conclusions}:
        hadir di setiap \omterm{def:g5:first-look-at-cells:cell}{sel}, dibagi dua ke dalam \omterm{def:g8:sexual-reproduction:gametes}{gametnya},
        dipulihkan pada \omterm{def:g5:flower-to-fruit:steps}{pembuahannya}, dapat dipilih mitra
        demi mitra;
  \item kecelakaan \omterm{def:g9:chromosomes:chromosomes}{kariotipenya} membuktikan muatannya:
        sebuah tumpukan dengan satu \omterm{def:g9:chromosomes:chromosomes}{kromosom} berlebih atau
        hilang mengubah perkembangan seluruh tubuhnya ---
        sebuah salinan tambahan pada pasang 21 yang kecil,
        misalnya, menghasilkan perangkat \omterm{def:g9:heredity-and-traits:traits}{sifat} yang dikenali
        sebagai sindrom Down. Satu benang salah hitung,
        banyak \omterm{def:g9:heredity-and-traits:traits}{sifat} bergeser: benangnya mengangkut petunjuk
        \omterm{def:g9:heredity-and-traits:traits}{sifatnya};
  \item dan satu pasang mengumumkan muatannya pada setiap
        \omterm{def:g9:chromosomes:chromosomes}{kariotipe}: \emph{\omterm{def:g9:chromosomes:chromosomes}{kromosom} kelaminnya}.
\end{itemize}
\end{proposition}
\admitted

\begin{definition}[X, Y, dan lemparan koinnya]\label{def:g9:chromosomes:xy}
Pasang 23 hadir dalam dua versi: dua \omterm{def:g9:chromosomes:chromosomes}{kromosom} \emph{X} yang
serupa --- \omterm{def:g9:chromosomes:chromosomes}{kariotipe} seorang anak perempuan; atau satu X dan
satu \emph{Y} yang kecil --- seorang anak laki-laki.
Pembagian duanya lalu menjalankan lemparan koinnya: setiap
\omterm{def:g4:animal-reproduction:sexual}{sel telur} mengangkut sebuah X; sebuah \omterm{def:g8:reproductive-systems:male}{sel sperma} mengangkut
X atau Y, separuh-separuh. \omterm{def:g5:flower-to-fruit:steps}{Pembuahan} oleh sperma-X
menghasilkan XX, oleh sperma-Y menghasilkan XY --- satu
peluang dari dua (bahasa peluang pada
\cref{ch:g9:heredity-and-traits}, akhirnya dengan
mekanismenya), diputuskan oleh perenang mana yang tiba,
pada saat \omterm{def:g5:flower-to-fruit:steps}{pembuahannya}.
\end{definition}

\begin{example}[Pertanyaan lama, jawaban pendek]\label{ex:g9:chromosomes:answers}
Tumpukannya melunasi perkara bersejarah. Apakah seorang anak
laki-laki atau perempuan sama sekali tidak berutang kepada
keinginan, makanan atau musim orang tuanya --- itu lemparan
koinnya, yang dijalankan pada mekanisme
\cref{def:g9:chromosomes:xy} (dan, untuk catatan
ketidakadilan zaman dahulu: perbedaan yang memutuskan itu
menumpang di dalam \emph{spermanya}). Separuh yang sama besar
juga menjawab album keluarganya: ibu dan ayahnya menyumbang
masing-masing satu setengah-tumpukan penuh --- 23 dan 23 ---
dan hitungan \omterm{def:g9:heredity-and-traits:heredity}{pewarisan sifatnya} menghormati kedua garisnya
sama rata.
\end{example}

\begin{method}[Membaca sebuah kariotipe]\label{met:g9:chromosomes:read}
Bila diserahi sebuah foto \omterm{def:g9:chromosomes:chromosomes}{kariotipe}:
\begin{enumerate}
  \item hitunglah: 46 diharapkan pada sebuah \omterm{def:g5:first-look-at-cells:cell}{sel} tubuh, 23
        pada sebuah \omterm{def:g8:sexual-reproduction:gametes}{gamet};
  \item pasangkan: mitranya dicocokkan menurut ukuran dan
        pola pitanya, 22 pasang biasa ditambah yang ke-23;
  \item bacalah yang ke-23: XX atau XY;
  \item periksa bilangannya: pasang mana pun yang beranggota
        satu atau tiga menjelaskan, menurut
        \cref{prop:g9:chromosomes:evidence}, sebuah
        perkembangan yang bergeser.
\end{enumerate}
\end{method}

\section{Soal: Meja Sitogenetika}

\begin{problem}[{Soal akhir pekan --- empat kariotipe pada
satu meja cahaya}]\label{pb:g9:chromosomes:1}
Meja ajar sebuah laboratorium sitogenetika rumah sakit:
\omterm{def:g9:chromosomes:chromosomes}{kariotipe} A --- 46 \omterm{def:g9:chromosomes:chromosomes}{kromosom}, pasang ke-23 XY; B --- 46,
pasang ke-23 XX; C --- 23 \omterm{def:g9:chromosomes:chromosomes}{kromosom}, satu X di antaranya;
D --- 47, dengan tiga salinan pada pasang 21, ke-23 XX.

\medskip\noindent\textbf{Bagian I --- Bacaannya.}
\begin{enumerate}
  \item Bacalah A, B dan C dengan
        \cref{met:g9:chromosomes:read}: \omterm{def:g5:first-look-at-cells:cell}{sel} siapa saja
        masing-masing mungkin?
  \item Bilangan C bukanlah sebuah kekeliruan. \omterm{def:g5:first-look-at-cells:cell}{Sel} jenis apa
        ia, dan riwayat pembelahan apa yang dicatat 23-nya?
  \item Bacalah D: bilangannya, pasangnya, dan perkembangan
        yang diumumkannya.
  \item \omterm{def:g8:sexual-reproduction:gametes}{Gamet} orang tua yang mana yang memutuskan pasang
        ke-23 pada A, dan berapa peluangnya waktu itu?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pertanyaan
hitungannya.}
\begin{enumerate}[resume]
  \item Seorang peserta pelatihan bertanya mengapa \omterm{def:g5:first-look-at-cells:cell}{sel} tubuh
        satu orang semuanya sesuai dengan \omterm{def:g9:chromosomes:chromosomes}{kariotipe} A.
        Jawablah dengan pembelahan penyalin pada
        \cref{prop:g5:first-look-at-cells:growth}.
  \item Yang lain bertanya bagaimana dua orang tua bersel 46
        membuat 23 pada C. Sebutkan pembelahan khusus dan aturannya.
  \item Gabungkan C dengan sebuah \omterm{def:g4:animal-reproduction:sexual}{sel telur} pembawa X, dan
        dengan tetangga pembawa Y di laboratoriumnya sendiri
        --- tunggu: C \emph{memang} bebas Y. Nyatakan apa
        yang dijamin X pada C tentang anak yang dibantunya.
  \item Hitunglah tumpukan yang mungkin dibagikan satu orang
        tua: nyatakan kebebasan pasang demi pasangnya dan
        kesimpulan $2^{23}$-nya --- lalu tambahkan pemasangan
        dua pembagian semacam itu oleh \omterm{def:g5:flower-to-fruit:steps}{pembuahannya}.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Ruang
konselingnya.}
\begin{enumerate}[resume]
  \item Orang tua seorang anak berkariotipe D bertanya ``apa
        yang kami lakukan salah?'' Berikan jawaban jujur
        mekanismenya tentang kecelakaan pembagian duanya ---
        dan apa yang diubah serta tidak diubah oleh ke-47
        benang anaknya tentang kasih sayang di ruangan itu.
        (Jawablah sebagai konselornya, dengan \omterm{def:g1:living-or-not:living}{biologi} lurus.)
  \item Sepasang suami istri dengan tiga anak perempuan
        bertanya berapa peluang seorang anak laki-laki
        berikutnya. Jawablah dengan lemparan koinnya, lalu
        betulkan naluri deretan-keberuntungannya.
  \item Seorang pengunjung mengklaim sebuah pola makan dapat
        memilih jenis kelamin bayinya. Auditlah klaim itu
        terhadap \cref{def:g9:chromosomes:xy}: di mana
        keputusannya dibuat, dan apa yang dapat dijangkaunya?
  \item Tutuplah meja ujinya dengan satu kalimat: apa yang
        dibuktikan keempat fotonya bersama-sama tentang di
        mana \omterm{def:g9:heredity-and-traits:heredity}{pewarisan sifatnya} menumpang.
\end{enumerate}
\end{problem}
