\chapter{Immunity}\label{ch:g12:immunity}

\omterm{def:g12:immunity:pathogen}{Pathogens} --- viruses, bacteria, fungi, parasites --- continually test
every animal. Survival depends on layered defences: walls that keep
intruders out, rapid general responses that limit early damage, and a
slower, precise adaptive system that learns and remembers. This chapter
builds those layers, explains \omterm{def:g12:immunity:bcell}{antibodies} and lymphocytes, and shows why
\omterm{def:g12:immunity:vaccine}{vaccination} works.

\section{Barriers and innate immunity}

\begin{definition}[Pathogen and antigen]\label{def:g12:immunity:pathogen}
A \emph{pathogen}\index{pathogen} is a disease-causing \omterm{def:g10:scientific-biology:organism}{organism} or
virus. An \emph{antigen}\index{antigen} is any molecule (often from a
pathogen) that can be specifically recognised by the adaptive immune
system.
\end{definition}

\begin{definition}[Innate immunity]\label{def:g12:immunity:innate}
\emph{Innate immunity}\index{innate immunity} comprises rapid, non-specific
(or broadly specific) defences present from birth: barriers, phagocytes,
complement, \omterm{def:g12:immunity:inflammation}{inflammation} and antiviral interferons. It does not build
long-term antigen-specific memory in the adaptive sense.
\end{definition}

\begin{proposition}[Physical and chemical barriers]
\label{prop:g12:immunity:barriers}
Intact skin, mucous membranes, mucus, cilia, acid (stomach, skin),
lysozyme in tears and saliva, and the normal microbiota all reduce
colonisation. Breach of barriers is often the first step in infection.
\end{proposition}
\admitted

\begin{definition}[Phagocytosis]\label{def:g12:immunity:phagocytosis}
\emph{Phagocytosis}\index{phagocytosis} is engulfment of particles by
\omterm{def:g10:the-cell:cell}{cells} such as \emph{macrophages}\index{macrophage} and
\emph{neutrophils}\index{neutrophil}. \omterm{def:g12:immunity:pathogen}{Pathogen} fragments may then be
displayed to lymphocytes (linking innate and adaptive arms).
\end{definition}

\begin{definition}[Inflammation]\label{def:g12:immunity:inflammation}
\emph{Inflammation}\index{inflammation} is a local response to injury or
infection: increased blood flow, vessel leakiness, recruitment of
immune \omterm{def:g10:the-cell:cell}{cells}, heat, redness, swelling and pain. It isolates damage and
delivers defences, but chronic inflammation can harm tissues.
\end{definition}

\section{Adaptive immunity}

\begin{definition}[Adaptive immunity]\label{def:g12:immunity:adaptive}
\emph{Adaptive immunity}\index{adaptive immunity} is antigen-specific,
slower on first encounter, and capable of \emph{immunological memory}
\index{immunological memory}. It is mediated chiefly by \emph{B
lymphocytes}\index{B lymphocyte} and \emph{T lymphocytes}\index{T
lymphocyte}.
\end{definition}

\begin{definition}[B cells and antibodies]\label{def:g12:immunity:bcell}
\emph{B cells}\index{B cell} recognise native \omterm{def:g12:immunity:pathogen}{antigen} via surface
immunoglobulin. When activated (often with \omterm{def:g10:the-cell:cell}{T-cell} help), they
differentiate into \emph{plasma cells}\index{plasma cell} that secrete
\emph{antibodies}\index{antibody} (immunoglobulins) and into memory B
\omterm{def:g10:the-cell:cell}{cells}.
\end{definition}

\begin{definition}[Antibody structure and functions]
\label{def:g12:immunity:antibody}
An \omterm{def:g12:immunity:bcell}{antibody} is a Y-shaped \omterm{def:g10:molecules-of-life:proteins}{protein} with variable regions that bind a
specific epitope. Functions include: neutralisation (blocking \omterm{def:g12:immunity:pathogen}{pathogen}
or toxin), opsonisation (tagging for \omterm{def:g12:immunity:phagocytosis}{phagocytosis}), agglutination, and
activation of complement.
\end{definition}

\begin{definition}[T cells]\label{def:g12:immunity:tcell}
\emph{T cells}\index{T cell} recognise \omterm{def:g12:immunity:pathogen}{antigen} fragments displayed on
other \omterm{def:g10:the-cell:cell}{cells} by MHC molecules. \emph{Helper T cells}\index{helper T cell}
($T_H$) coordinate responses (activate \omterm{def:g12:immunity:bcell}{B cells} and \omterm{def:g12:immunity:phagocytosis}{macrophages});
\emph{cytotoxic T cells}\index{cytotoxic T cell} ($T_C$) kill virus-infected
or abnormal \omterm{def:g10:the-cell:cell}{cells}; regulatory T \omterm{def:g10:the-cell:cell}{cells} restrain over-reaction.
\end{definition}

\begin{proposition}[Clonal selection]\label{prop:g12:immunity:clonal}
Each lymphocyte clone bears receptors of one \omterm{def:g10:membranes-enzymes:ES}{specificity}. \omterm{def:g12:immunity:pathogen}{Antigen}
selectively activates matching clones to proliferate and differentiate
--- \emph{clonal selection}\index{clonal selection}. This explains
\omterm{def:g10:membranes-enzymes:ES}{specificity} and the expansion of useful \omterm{def:g10:the-cell:cell}{cells} during infection.
\end{proposition}
\admitted

\begin{example}[Primary vs secondary response]
\label{ex:g12:immunity:primary}
On first exposure, \omterm{def:g12:immunity:bcell}{antibody} levels rise over one to two weeks. On
re-exposure, memory \omterm{def:g10:the-cell:cell}{cells} mount a faster, larger \omterm{prop:g12:immunity:kinetics}{secondary response} ---
often clearing the \omterm{def:g12:immunity:pathogen}{pathogen} before disease.
\end{example}

\begin{proposition}[Primary vs secondary kinetics]
\label{prop:g12:immunity:kinetics}
The \emph{primary response}\index{primary immune response} after first
encounter with an \omterm{def:g12:immunity:pathogen}{antigen} is delayed (lag while rare clones expand) and
modest in peak \omterm{def:g12:immunity:bcell}{antibody} level. The \emph{secondary response}\index{secondary
immune response} after re-exposure is faster, reaches a higher titre, and is
often dominated by higher-affinity \omterm{def:g12:immunity:bcell}{antibody} --- the operational definition of
\omterm{def:g12:immunity:adaptive}{immunological memory} in blood measurements.
\end{proposition}
\admitted

\section{Vaccination and memory}

\begin{definition}[Vaccination]\label{def:g12:immunity:vaccine}
\emph{Vaccination}\index{vaccination} deliberately exposes the immune
system to a safe form of \omterm{def:g12:immunity:pathogen}{antigen} (killed or attenuated \omterm{def:g12:immunity:pathogen}{pathogen},
subunit, toxoid, mRNA/\omterm{def:g10:molecules-of-life:proteins}{protein} \omterm{def:g12:immunity:pathogen}{antigen}, \ldots) so that memory \omterm{def:g10:the-cell:cell}{cells} form
without causing the full disease.
\end{definition}

\begin{proposition}[How vaccines protect]\label{prop:g12:immunity:vaccine}
Vaccines prime adaptive memory. On later encounter with the real
\omterm{def:g12:immunity:pathogen}{pathogen}, \omterm{prop:g12:immunity:kinetics}{secondary responses} neutralise or clear it quickly. \omterm{def:g11:evolution-mechanisms:pool}{Population}
uptake also reduces transmission (\emph{herd immunity}\index{herd
immunity} when coverage is high enough for a given \omterm{def:g12:immunity:pathogen}{pathogen}).
\end{proposition}
\admitted

\begin{method}[Reasoning about an immune response]
\label{met:g12:immunity:reason}
\begin{enumerate}
  \item Which barrier failed, if any?
  \item What innate mechanisms act in the first hours?
  \item Is the \omterm{def:g12:immunity:pathogen}{pathogen} intracellular (favour $T_C$) or extracellular
        (favour \omterm{def:g12:immunity:bcell}{antibodies})?
  \item Will memory from prior infection or vaccine alter the kinetics?
\end{enumerate}
\end{method}

\begin{remark}[Self-tolerance]\label{rem:g12:immunity:tolerance}
Lymphocytes that strongly recognise \omterm{def:g12:immunity:pathogen}{self-antigens} are deleted or
suppressed. Failure of tolerance contributes to autoimmune disease;
over-reaction to harmless \omterm{def:g12:immunity:pathogen}{antigens} underlies \omterm{def:g12:immunity:allergy-auto}{allergy}.
\end{remark}

\begin{example}[Virus-infected cell]\label{ex:g12:immunity:virus}
A virus-infected \omterm{def:g10:the-cell:cell}{cell} displays viral peptides on MHC \omterm{def:g10:biodiversity-classification:hierarchy}{class} I; \omterm{def:g12:immunity:tcell}{cytotoxic
T cells} induce apoptosis. \omterm{def:g12:immunity:bcell}{Antibodies} can still block free virions
between \omterm{def:g10:the-cell:cell}{cells}; interferons slow viral spread early on.
\end{example}

\section{Integration and disorders (overview)}

\begin{definition}[Active and passive immunity]
\label{def:g12:immunity:active-passive}
\emph{Active immunity}\index{active immunity} is generated by the
host's own response (infection or vaccine) and yields memory.
\emph{Passive immunity}\index{passive immunity} is acquired ready-made
\omterm{def:g12:immunity:bcell}{antibodies} (maternal transfer, antiserum) --- immediate but temporary,
without host memory from that transfer alone.
\end{definition}

\begin{definition}[Allergy and autoimmunity]
\label{def:g12:immunity:allergy-auto}
\emph{Allergy}\index{allergy} (immediate hypersensitivity) is an
inappropriate adaptive response to a normally harmless environmental
\omterm{def:g12:immunity:pathogen}{antigen} (\emph{allergen}\index{allergen}), often involving IgE, mast
\omterm{def:g10:the-cell:cell}{cells} and histamine --- symptoms range from hay fever to anaphylaxis.
\emph{Autoimmunity}\index{autoimmunity} is adaptive attack on the body's
own \omterm{def:g12:immunity:pathogen}{antigens} after failure of self-tolerance (e.g.\ certain forms of
diabetes mellitus type 1, rheumatoid arthritis). Both show that
\omterm{def:g10:membranes-enzymes:ES}{specificity} without proper regulation damages the host.
\end{definition}

\begin{proposition}[Immunodeficiency and hyper-reaction]
\label{prop:g12:immunity:disorders}
Immunodeficiency (genetic, infection such as HIV targeting \omterm{def:g12:immunity:tcell}{helper T
cells}, or drugs) raises infection risk. Hypersensitivity and
\omterm{def:g12:immunity:allergy-auto}{autoimmunity} show that regulation is as important as firepower: the same
clonal machinery that clears \omterm{def:g12:immunity:pathogen}{pathogens} can mis-target pollen or self
tissue.
\end{proposition}
\admitted

\begin{example}[Too little vs too much]
\label{ex:g12:immunity:balance}
A child with severe combined immunodeficiency lacks functional T and \omterm{def:g12:immunity:bcell}{B
cells} and suffers recurrent infections --- too little adaptive firepower.
Another child with peanut \omterm{def:g12:immunity:allergy-auto}{allergy} has a rapid, potentially life-threatening
response to a harmless food \omterm{def:g10:molecules-of-life:proteins}{protein} --- misdirected firepower. Therapy aims
differ: restore defences versus dampen or desensitise them.
\end{example}

\section{Problem: An outbreak and a vaccine}

\begin{problem}[{Weekend problem --- follow a pathogen from barrier
breach to memory, then design a public-health vaccine response}]
\label{pb:g12:immunity:1}
A novel influenza-like virus spreads in a city. Some people were
vaccinated months earlier with a closely related strain; others were
not.

\medskip
\noindent\textbf{Part I --- First line and innate response.}
\begin{enumerate}
  \item List barriers the virus must overcome to infect the respiratory
        epithelium.
  \item Name two innate mechanisms that act within the first day.
  \item Why is fever both a possible aid and a possible danger?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Adaptive response.}
\begin{enumerate}[resume]
  \item Which lymphocyte types limit free virus versus kill infected
        \omterm{def:g10:the-cell:cell}{cells}?
  \item Outline \omterm{prop:g12:immunity:clonal}{clonal selection} for a \omterm{def:g10:the-cell:cell}{B-cell} clone specific to a viral
        surface \omterm{def:g10:molecules-of-life:proteins}{protein}.
  \item Sketch primary vs \omterm{def:g10:ecosystems:trophic-roles}{secondary} \omterm{def:g12:immunity:bcell}{antibody} kinetics and mark the
        vaccinated group's likely curve on re-exposure.
  \item Explain why antigenic drift (small surface changes) can reduce
        vaccine match.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- \omterm{def:g12:immunity:vaccine}{Vaccination} and \omterm{def:g11:evolution-mechanisms:pool}{population}.}
\begin{enumerate}[resume]
  \item Define \omterm{def:g12:immunity:active-passive}{active immunity} in the vaccinee.
  \item What is \omterm{prop:g12:immunity:vaccine}{herd immunity}, and why does required coverage depend on
        how contagious the virus is?
  \item Compare killed, live-attenuated and subunit/mRNA vaccine ideas
        for safety vs immunogenicity (qualitative).
  \item Give one reason passive monoclonal \omterm{def:g12:immunity:bcell}{antibodies} might be given to
        a high-risk patient at exposure, and one limitation.
  \item In one paragraph, integrate barriers, innate speed, adaptive
        memory and \omterm{def:g12:immunity:vaccine}{vaccination} into a single coherent defence story.
\end{enumerate}
\end{problem}
