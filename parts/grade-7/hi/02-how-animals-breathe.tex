\chapter{जंतु साँस कैसे लेते हैं}\label{ch:g7:how-animals-breathe}

पुल के नीचे की ट्राउट, लॉन के नीचे का केंचुआ, उसके ऊपर उड़ता व्याध-पतंगा
और उस पर खड़े तुम --- इस मिनट सब एक ही काम कर रहे हो: अपने आस-पास से
\omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} ले रहे हो
(\cref{def:g7:what-is-respiration:respiration})। पर तुम्हारे आस-पास
अलग-अलग हैं --- हवा, \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी}, पानी --- और इसलिए साज़-सामान भी। यह अध्याय
जंतु-जगत की साँस लेने वाली मशीनों का चक्कर है।

\section{एक ही समस्या, चार मशीनें}

\begin{proposition}[साझी विशेष-सूची]\label{prop:g7:how-animals-breathe:spec}
हर साँस वाला अंग, चाहे किसी का भी हो, एक ही विशेष-सूची पर बना है, जो
\cref{rem:g4:breathing:sponge} में पहले भी मिल चुकी है: एक
\emph{बड़ी}, \emph{पतली}, \emph{नम} सतह, जहाँ \omterm{def:g1:animals-around-us:animal}{जंतु} का \omterm{prop:g4:journey-of-food:absorb}{रक्त} (या शरीर का
तरल) अपने आस-पास के इतना पास आ जाए कि \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} भीतर और \omterm{def:g7:what-is-respiration:respiration}{कार्बन
डाइऑक्साइड} बाहर पार कर सके। बड़ी, ताकि पर्याप्त पार हो सके; पतली, ताकि
पार करना आसान हो; और नम, क्योंकि गैसें घुली हुई ही पार करती हैं। मशीनें
सिर्फ़ इसमें अलग हैं कि वे उस सतह को आस-पास के सामने कैसे थामती हैं।
\end{proposition}
\admitted

\begin{definition}[चार मशीनें]\label{def:g7:how-animals-breathe:four}
\omterm{def:g1:animals-around-us:animal}{जंतु} अपनी विनिमय-सतहें मुख्य रूप से चार ढंग से थामते हैं:
\begin{enumerate}
  \item \emph{फेफड़े}\index{फेफड़ा}: हवा से भरी वे कोठरियाँ जो शरीर के
        \emph{भीतर} तह की हुई होती हैं और साँस की गतियों से हवादार
        रहती हैं --- \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी}, \omterm{prop:g3:sorting-living-things:groups}{पक्षी}, \omterm{prop:g4:classifying-animals:classes}{सरीसृप}, वयस्क \omterm{prop:g3:sorting-living-things:groups}{उभयचर};
  \item \emph{क्लोम}\index{क्लोम}: पंखदार सतहें, जो \emph{शरीर के बाहर
        या उसके किनारे} थामी जाती हैं और पानी की धार से धुलती रहती हैं
        --- \omterm{prop:g3:sorting-living-things:groups}{मछलियाँ} (\cref{exo:g4:breathing:11}), \omterm{ex:g2:animals-grow-up:frog}{टैडपोल}, पानी के
        बहुत-से डिंभ, सीपियाँ;
  \item \emph{श्वासनलिकाएँ}\index{श्वासनलिका (कीट की)}: महीन हवा-नलियों
        का वह जाल जो शरीर पर बने छिद्रों से सीधे हर अंग तक पाइप किया
        होता है --- \omterm{prop:g3:sorting-living-things:groups}{कीट}; हवा घर-घर पहुँचाई जाती है, और वह भी बड़ी हद तक
        \omterm{prop:g4:journey-of-food:absorb}{रक्त} की मदद के बिना;
  \item \emph{त्वचीय श्वसन}\index{त्वचीय श्वसन}: पूरी नम \omterm{def:g1:the-five-senses:sense}{त्वचा} ही
        विनिमय-सतह --- पूरी तरह केंचुए में; और \omterm{prop:g3:sorting-living-things:groups}{उभयचरों} में अपने सादे
        फेफड़ों के एक बड़े पूरक के रूप में।
\end{enumerate}
\end{definition}

\begin{example}[मशीनों को चलते देखना]\label{ex:g7:how-animals-breathe:watch}
हर मशीन अपनी लय दिखाती है। ट्राउट का मुँह और क्लोम-ढक्कन बारी-बारी से,
लगातार पंप करते हैं --- पानी मुँह से भीतर, \omterm{def:g7:how-animals-breathe:four}{क्लोमों} के ऊपर से होकर,
ढक्कनों के नीचे से बाहर। आराम करते टिड्डे का पेट धड़कता है --- अपनी
हवा-नलियाँ पंप करता हुआ। मेंढक का गला फड़फड़ाता है --- हवा को अपने
फेफड़ों में धकेलता हुआ (यह करने के लिए उसके पास \omterm{def:g3:bones-and-muscles:skeleton}{पसलियों} का पिंजरा नहीं
है)। सिर्फ़ केंचुआ कुछ नहीं दिखाता: \omterm{def:g7:how-animals-breathe:four}{त्वचीय श्वसन} में कोई गति होती ही
नहीं, और ठीक इसीलिए
\cref{def:g7:what-is-respiration:respiration} ने अदला-बदली को पंप करने
से अलग किया था।
\end{example}

\section{मशीन माध्यम से मेल खाती है --- और जीवन से भी}

\begin{proposition}[कौन-सी मशीन कहाँ, और क्यों]\label{prop:g7:how-animals-breathe:fits}
मशीनें माध्यमों और जीवनों पर बैठ जाती हैं
(\cref{def:g4:adapted-to-their-world:adaptation}, साँस पर काम करता
हुआ):
\begin{itemize}
  \item \omterm{def:g7:how-animals-breathe:four}{क्लोम} पानी के लिए ठीक हैं: उनकी पंखदार सतहें पानी में अलग-अलग
        तैरती रहती हैं और लगातार धुलती हैं --- और हवा में वे बेकार होकर
        बैठ जाती हैं (\cref{exo:g4:breathing:11} वाली दम तोड़ती
        ट्राउट);
  \item \omterm{def:g7:how-animals-breathe:four}{फेफड़े} हवा के लिए ठीक हैं: भीतर तह किए हुए वे वहाँ भी नम बने
        रहते हैं जहाँ खुला पानी उड़ जाता --- पर वे पानी की विरल घुली
        \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} नहीं बटोर सकते;
  \item \omterm{def:g7:how-animals-breathe:four}{श्वासनलिकाएँ} छोटे शरीरों के लिए ठीक हैं: मिलीमीटर भर की दूरियों
        पर बेहतरीन, पर लंबी दूरियों पर बहुत धीमी --- \omterm{prop:g3:sorting-living-things:groups}{कीटों} के छोटे बने
        रहने का एक कारण यही है;
  \item और \omterm{def:g7:how-animals-breathe:four}{त्वचीय श्वसन} नम जगहों के छोटे, नम-चमड़ी वाले जीवनों के लिए
        ठीक है: केंचुए की पूरी सतह उसके पतले शरीर की मुश्किल से सेवा
        करती है, और सूखी हवा उसे मार डालती है --- बारिश के बाद पगडंडी
        पर फँसे केंचुए याद करो।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[एक ही जंतु में दो जीवन, दो मशीनें]\label{ex:g7:how-animals-breathe:frog}
मेंढक पूरा अध्याय अपने एक ही जीवन में चला देता है: \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} के रूप में
\omterm{def:g7:how-animals-breathe:four}{क्लोम} (पानी में रहने वाले की मशीन,
\cref{ex:g2:animals-grow-up:frog}), और वयस्क के रूप में \omterm{def:g7:how-animals-breathe:four}{फेफड़े} तथा नम
\omterm{def:g1:the-five-senses:sense}{त्वचा} --- \omterm{def:g7:how-animals-breathe:four}{फेफड़े} सादे, और \omterm{def:g1:the-five-senses:sense}{त्वचा} बहुत सारा काम सँभाले हुए, ख़ास तौर पर
तालाब की तह वाली ठंडी सर्दी भर
(\cref{exo:g6:life-through-seasons:11}: वह सुप्त मेंढक अकेली \omterm{def:g1:the-five-senses:sense}{त्वचा} से
साँस लेता है)। \omterm{def:g2:animals-grow-up:metamorphosis}{कायांतरण} साँस की मशीनरी तक फिर से गढ़ देता है।
\end{example}

\begin{example}[पानी में रहते हुए हवा में साँस लेना]\label{ex:g7:how-animals-breathe:airinwater}
फेफड़ों वाले पानी के बाशिंदों को आना-जाना पड़ता है। डॉल्फ़िन और ह्वेल
फुहार छोड़ने और साँस भरने के लिए सतह पर आते हैं
(\cref{ex:g6:classification-kinship:whale}: \omterm{prop:g3:sorting-living-things:groups}{स्तनधारी} \omterm{def:g7:how-animals-breathe:four}{फेफड़े}, थलचर
पूर्वजों से विरासत में मिले); तालाब का घोंघा फिर से भरने के लिए सतह पर
चढ़ता है; और गोताख़ोर भृंग अपने डैनों के ढक्कनों के नीचे चाँदी जैसा हवा
का एक बुलबुला ढोता है --- यानी ग़ोताख़ोर की टंकी, जो सतह पर फिर भरी जाती
है। \omterm{def:g2:where-animals-live:habitat}{आवास} पर सौदा हो सकता है; मशीन के माध्यम पर नहीं।
\end{example}

\begin{method}[किसी जंतु की साँस पढ़ना]\label{met:g7:how-animals-breathe:read}
कोई अनजान \omterm{def:g1:animals-around-us:animal}{जंतु} सामने हो, तो:
\begin{enumerate}
  \item वह रहता कहाँ है --- और क्या वह वहाँ से निकल सकता है? (पानी,
        हवा, \omterm{def:g6:decomposers-and-soil:soil}{मिट्टी}; आना-जाना करने वाला या वहीं का बाशिंदा);
  \item मशीन के निशान ढूँढ़ो: क्लोम-ढक्कन या पंखदार गुच्छे; नथुने और
        हिलती छाती; किसी \omterm{prop:g3:sorting-living-things:groups}{कीट} के शरीर पर नन्हे छिद्रों की क़तारें; या
        कुछ भी दिखता नहीं, बस एक नम \omterm{def:g1:the-five-senses:sense}{त्वचा};
  \item लय देखो: पंप करते ढक्कन, धड़कता पेट, उठती छाती, सतह तक के
        चक्कर;
  \item मशीन का नाम बताओ --- और उसे
        \cref{prop:g7:how-animals-breathe:fits} की मदद से माध्यम से
        मिलाकर जाँचो।
\end{enumerate}
\end{method}

\begin{remark}[विशेष-सूची कभी नहीं बदलती]\label{rem:g7:how-animals-breathe:same}
इस सारी विविधता के नीचे इन चारों में से किसी भी मशीन को पास से देखो और
\cref{prop:g7:how-animals-breathe:spec} फिर सामने आ जाती है: \omterm{def:g7:how-animals-breathe:four}{क्लोम} के
\omterm{def:g1:animals-around-us:covering}{पंख}, \omterm{def:g7:how-animals-breathe:four}{फेफड़े} की कोठरियाँ, \omterm{def:g7:how-animals-breathe:four}{श्वासनलिकाओं} के सबसे महीन सिरे और केंचुए की
\omterm{def:g1:the-five-senses:sense}{त्वचा}, सब आस-पास तथा शरीर की बड़ी-पतली-नम मिलन-जगहें हैं। चार मशीनें,
एक विशेष-सूची --- यानी \omterm{prop:g5:first-look-at-cells:principle}{जीवन की एकता}
(\cref{prop:g5:first-look-at-cells:principle}), साँस के साज़-सामान में
लिखी हुई।
\end{remark}

\section{समस्या: तालाब में साँस लेने वालों की गिनती}

\begin{problem}[{सप्ताहांत समस्या --- एक तालाब, और काम पर हर
मशीन}]\label{pb:g7:how-animals-breathe:1}
जाल, देखने वाले डिब्बे और कॉपी के साथ तालाब पर एक दोपहर। गिनती में:
ट्राउट के बच्चे और काँटा-मछलियाँ; \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} (कुछ की पिछली \omterm{def:g1:my-body:parts}{टाँगें} आ चुकी
हैं); वयस्क मेंढक; पानी की सतह की परत पर घूमते तालाब के घोंघे; चाँदी
चमकाता एक गोताख़ोर भृंग; पत्थरों के नीचे पंखदार गुच्छों वाले मेफ़्लाई के
डिंभ; और किनारे के नम लॉन में केंचुए।

\medskip\noindent\textbf{भाग I --- मशीनों के नाम।}
\begin{enumerate}
  \item गिनती के हर सदस्य को उसकी साँस वाली मशीन दो।
  \item मेफ़्लाई के डिंभ के पंखदार गुच्छे ठहरे हुए पानी में भी लगातार
        हिलते रहते हैं। वे हैं क्या, और यह हिलाना किसलिए है?
  \item गिनती के किन सदस्यों को सतह तक जाना ही पड़ता है, और ठीक वही
        क्यों?
  \item बिना \omterm{def:g1:my-body:parts}{टाँगों} वाले \omterm{ex:g2:animals-grow-up:frog}{टैडपोल} और \omterm{def:g1:my-body:parts}{टाँगों} वाले भीतर से उतने ही अलग हो
        सकते हैं जितने बाहर से।
        \cref{ex:g7:how-animals-breathe:frog} के अनुसार भीतर कौन-सा
        पुनर्निर्माण चल रहा है?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- दबाव में मशीनें।}
\begin{enumerate}[resume]
  \item गरम, बिना हवा वाला एक हफ़्ता तालाब की घुली हुई \omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन} घटा देता
        है (\cref{ch:g7:oxygen-in-the-water} समझाएगा कि क्यों)। गिनती
        को क्रम में लगाओ: पहले तकलीफ़ किसे होगी --- \omterm{def:g7:how-animals-breathe:four}{क्लोम} वालों को, या
        सतह पर आने-जाने वालों को? क्यों?
  \item उसी हफ़्ते काँटा-मछलियाँ सतह के ठीक नीचे लटकी रहती हैं और उनके
        मुँह सतह की परत पर चलते रहते हैं। इस व्यवहार की व्याख्या
        विशेष-सूची से करो: पानी के सबसे ऊपरी मिलीमीटर में ख़ास क्या है?
  \item एक लापरवाह पानी में उतरने वाला महीन कीचड़ के बादल उड़ा देता है।
        कौन-सी मशीन जाम होती है, और तुम्हें उन \omterm{def:g1:animals-around-us:animal}{जंतुओं} से क्या करने की
        उम्मीद होगी?
  \item रात भर की बारिश के बाद लॉन के केंचुए पगडंडी पर पड़े मिलते हैं।
        इस जाने-पहचाने नज़ारे को समझाओ: कौन-सी मशीन, और कौन-सी
        परिस्थिति ने उन्हें ऊपर धकेला?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- बड़ा नक़्शा।}
\begin{enumerate}[resume]
  \item गिनती को तालिका के रूप में बनाओ: \omterm{def:g1:animals-around-us:animal}{जंतु}, मशीन, माध्यम, और
        आने-जाने वाला या बाशिंदा।
        \cref{rem:g7:how-animals-breathe:same} के अनुसार मशीन वाले हर
        स्तंभ के \omterm{def:g1:my-body:parts}{सिर} पर कौन-सी एक विशेष-सूची बैठती है?
  \item जनवरी में तालाब ऊपर से जम जाता है और सतह बंद हो जाती है। सर्दी
        की नियतियों की भविष्यवाणी करो: मेंढक
        (\cref{exo:g6:life-through-seasons:11}), घोंघे, भृंग,
        \omterm{prop:g3:sorting-living-things:groups}{मछलियाँ} --- कौन मुश्किल में है, और किसका इंतज़ाम है?
  \item एक नया मेहमान आता है: वह जल-मकड़ी जो रेशम की घंटी बुनती है और
        उसमें सतह की हवा भर लेती है। उसकी इस तरकीब को भृंग की तरकीब के
        बग़ल में रखो, और उस मशीन का नाम बताओ जिसकी सेवा वह करती रहती
        है।
  \item निष्कर्ष दो: एक छोटे-से तालाब को पूरी तरह बसने के लिए साँस की
        चार अलग-अलग मशीनें क्यों चाहिए? एक वाक्य में, और उसमें
        \emph{माध्यम} तथा \emph{\omterm{def:g7:what-is-respiration:respiration}{ऑक्सीजन}} शब्द इस्तेमाल करो।
\end{enumerate}
\end{problem}
