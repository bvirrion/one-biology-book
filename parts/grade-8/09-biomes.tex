\chapter{Biomes of the Earth}\label{ch:g8:biomes}

Climate paints the planet in broad living zones. \omterm{def:g8:biomes:biome}{Biomes} are large-scale
\omterm{def:g7:eco-energy:ecosystem}{ecosystem} types shaped mainly by temperature and precipitation --- from
rainforests to \omterm{def:g8:biomes:terra}{deserts} and \omterm{def:g8:biomes:terra}{tundra}.

\section{What is a biome?}

\begin{definition}[Biome]\label{def:g8:biomes:biome}
A \emph{biome}\index{biome} is a major regional \omterm{def:g8:populations:community}{community} of \omterm{def:g4:ecosystems:levels}{organisms}
with a \omterm{def:g3:classify:char}{characteristic} climate and \omterm{def:g8:genetics:dom}{dominant} \omterm{def:g1:plants-animals:plant}{plant} forms (for example
\omterm{def:g8:biomes:terra}{tropical rainforest}, \omterm{def:g2:habitats:kinds}{grassland}, \omterm{def:g8:biomes:terra}{tundra}). Local \omterm{def:g7:eco-energy:ecosystem}{ecosystems} sit within
biomes.
\end{definition}

\begin{proposition}[Climate drives biomes]\label{prop:g8:biomes:climate}
Average temperature and rainfall (and their \omterm{def:g1:seasons:season}{seasons}) largely determine
which \omterm{def:g1:plants-animals:plant}{plants} can dominate. \omterm{def:g1:plants-animals:animal}{Animals} and soil communities follow the \omterm{def:g1:plants-animals:plant}{plant}
base and climate stresses.
\end{proposition}
\admitted

\section{A tour of major biomes}

\begin{definition}[Selected terrestrial biomes]\label{def:g8:biomes:terra}
\begin{itemize}
  \item \emph{Tropical rainforest}\index{tropical rainforest}: warm, very
        wet; high \omterm{def:g6:biodiversity:def}{biodiversity}; layered trees.
  \item \emph{Savanna}\index{savanna}: warm; wet/dry \omterm{def:g1:seasons:season}{seasons}; grasses with
        scattered trees.
  \item \emph{Desert}\index{desert}: very low rainfall; \omterm{def:g1:plants-animals:plant}{plants} adapted to
        save \omterm{def:g4:plants-need:needs}{water}.
  \item \emph{Temperate grassland}\index{grassland!temperate}: fertile
        soils; grasses dominate; few trees.
  \item \emph{Temperate forest}\index{forest!temperate}: moderate climate;
        deciduous or mixed trees.
  \item \emph{Taiga (boreal forest)}\index{taiga}: cold; conifers.
  \item \emph{Tundra}\index{tundra}: very cold; low \omterm{def:g1:plants-animals:plant}{plants}; permafrost in
        many areas.
\end{itemize}
\end{definition}

\begin{example}[Desert adaptations]\label{ex:g8:biomes:desert}
Cacti store \omterm{def:g4:plants-need:needs}{water} and have reduced \omterm{def:g7:tissues:plant-organs}{leaves} (spines); many \omterm{def:g1:plants-animals:animal}{animals} are
nocturnal to avoid heat --- climate shapes \omterm{def:g7:heredity:trait}{traits}.
\end{example}

\section{Aquatic systems (brief)}

\begin{definition}[Aquatic biomes]\label{def:g8:biomes:aqua}
\omterm{def:g2:habitats:kinds}{Oceans}, coral reefs, estuaries, lakes and rivers are shaped by salinity,
depth, \omterm{def:g4:plants-need:needs}{light} and \omterm{def:g4:plants-need:needs}{nutrients} as much as by ``rainfall''. Coral reefs rival
rainforests in diversity but need warm, clear, sunlit \omterm{def:g4:plants-need:needs}{water}.
\end{definition}

\begin{proposition}[Human pressure]\label{prop:g8:biomes:human}
\omterm{def:g8:biomes:biome}{Biomes} face deforestation, desertification risk, overfishing, \omterm{def:g7:impact:pollution}{pollution} and
climate shifts that move temperature/rainfall envelopes. \omterm{def:g7:impact:conserve}{Conservation}
often means protecting large connected \omterm{def:g7:impact:habitat}{habitats} within \omterm{def:g8:biomes:biome}{biomes}.
\end{proposition}
\admitted

\begin{method}[Identifying a biome from a description]\label{met:g8:biomes:id}
\begin{enumerate}
  \item Note temperature range and precipitation pattern.
  \item Identify \omterm{def:g8:genetics:dom}{dominant} vegetation.
  \item Match to the nearest \omterm{def:g8:biomes:biome}{biome} type; allow for local exceptions
        (mountains create vertical zonation).
\end{enumerate}
\end{method}
