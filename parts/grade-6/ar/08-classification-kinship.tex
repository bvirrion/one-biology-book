\chapter{التصنيف والقرابة}\label{ch:g6:classification-kinship}

لماذا ينجح \omterm{def:g4:classifying-animals:classification}{تصنيف} \cref{ch:g4:classifying-animals} هذا النجاح؟
افرز \omterm{def:g1:animals-around-us:animal}{الحيوانات} \omterm{def:g1:animals-around-us:covering}{بالريش} \omterm{def:g1:animals-around-us:covering}{والفرو}، فتصطفّ ألف \omterm{def:g6:classification-kinship:attribute}{صفة} أخرى طائعةً --- فذوات
\omterm{def:g2:animals-grow-up:born}{الحليب} لها \omterm{def:g4:classifying-animals:vertebrate}{عمود فقري}، ولابسات \omterm{def:g1:animals-around-us:covering}{الريش} تضع بيضًا مقشورًا، ولا يحتاج
أحد إلى التحقق. وقد تركت \cref{rem:g4:classifying-animals:tree}
السؤال معلَّقًا. وهذا الفصل يجيب عنه بكلمة واحدة --- \omterm{prop:g6:classification-kinship:kinship}{القرابة} ---
ويعيد بناء الصناديق شجرةَ نسب.

\section{الصفات، مشتركةً ومتداخلة}

\begin{definition}[الصفة]\label{def:g6:classification-kinship:attribute}
\emph{الصفة}\index{صفة} خاصية \emph{يملكها} الكائن، مصوغة صياغة
قابلة للتحقق: له هيكل ذو \omterm{def:g4:classifying-animals:vertebrate}{عمود فقري}؛ له أربعة \omterm{def:g1:animals-around-us:parts}{أطراف}؛ له \omterm{def:g1:animals-around-us:covering}{فرو} ويُرضع
حليبًا؛ له \omterm{def:g1:animals-around-us:covering}{ريش}. \omterm{def:g4:classifying-animals:classification}{والتصنيف} مبني على الصفات --- أي محكّات
\cref{def:g3:sorting-living-things:criterion} الجيدة، مصوغةً صوغًا
دقيقًا.
\end{definition}

\begin{proposition}[الصفات تتداخل]\label{prop:g6:classification-kinship:nest}
إذا فُحصت الصفات عبر \omterm{def:g5:biodiversity:species}{أنواع} كثيرة وقعت في مجموعات
\emph{متداخلة} --- فكل \omterm{def:g1:animals-around-us:animal}{حيوان} ذي \omterm{def:g1:animals-around-us:covering}{فرو} له أربعة \omterm{def:g1:animals-around-us:parts}{أطراف} أيضًا؛ وكل
\omterm{def:g1:animals-around-us:animal}{حيوان} رباعي \omterm{def:g1:my-body:parts}{الأطراف} له \omterm{def:g4:classifying-animals:vertebrate}{عمود فقري}؛ ولا يقع النمط المعكوس أبدًا (\omterm{def:g1:animals-around-us:covering}{فرو}
بلا \omterm{def:g4:classifying-animals:vertebrate}{عمود فقري}) ولا مجموعات متقاطعة (\omterm{def:g1:animals-around-us:covering}{فرو} \omterm{def:g1:animals-around-us:covering}{وريش} معًا). وذلك التداخل هو
سبب ملاءمة المجموعات داخل المجموعات
(\cref{def:g4:classifying-animals:classification}) للعالم الحي
أصلًا.
\end{proposition}
\admitted

\begin{example}[وضع ثلاثة حيوانات]\label{ex:g6:classification-kinship:placing}
راجع الصناديق للحوت والسلمون المرقط والشحرور. الحوت: \omterm{def:g4:classifying-animals:vertebrate}{عمود فقري}
نعم، وأربعة \omterm{def:g1:animals-around-us:parts}{أطراف} نعم (زعنفتان أماميتان، وأثران خفيّان خلفهما)،
\omterm{def:g1:animals-around-us:covering}{وفرو} \omterm{def:g2:animals-grow-up:born}{وحليب} نعم --- فالصندوق الأعمق، مع القطة. والسلمون المرقط:
\omterm{def:g4:classifying-animals:vertebrate}{عمود فقري}، ثم قف --- فالصندوق الخارجي وحده. والشحرور: \omterm{def:g4:classifying-animals:vertebrate}{عمود فقري}،
وأربعة \omterm{def:g1:animals-around-us:parts}{أطراف} (\omterm{def:g1:my-body:parts}{رجلان} وجناحان)، \omterm{def:g1:animals-around-us:covering}{وريش} --- فصندوق رباعيات \omterm{def:g1:my-body:parts}{الأطراف}، في
حجيرة \omterm{def:g1:animals-around-us:covering}{الريش} الخاصة. ولا \omterm{def:g1:animals-around-us:animal}{حيوان} يطلب صندوقين متقاطعين معًا أبدًا.
\end{example}

\section{القرابة: سبب التداخل}

\begin{proposition}[الصفات المشتركة تعني سلفًا مشتركًا]\label{prop:g6:classification-kinship:kinship}
تفسير التداخل هو الوراثة
(\cref{rem:g4:animal-reproduction:mix}: فالصغار ينالون من
والديهم). \omterm{def:g5:biodiversity:species}{فالأنواع} المشتركة في \omterm{def:g6:classification-kinship:attribute}{صفة} تشترك فيها لأنها
\emph{ورثتها من سلف مشترك} --- \omterm{def:g5:biodiversity:species}{نوع} جدّ بعيد. وكلما زادت الصفات
المشتركة بين نوعين قربت \emph{قرابتهما}\index{قرابة}: فالقطة
والحوت، وكلاهما ذو \omterm{def:g1:animals-around-us:covering}{فرو} يُرضع حليبًا، أقرب قرابةً أحدهما للآخر من
\omterm{prop:g6:classification-kinship:kinship}{قرابة} أي منهما للسلمون المرقط. ومجموعة \omterm{def:g4:classifying-animals:classification}{التصنيف} في جوهرها
\emph{أسرة}: سلف وكل نسله.
\end{proposition}
\admitted

\begin{example}[قراءة الحوت على صوابها أخيرًا]\label{ex:g6:classification-kinship:whale}
ولهذا وجب أن يُحلّ لغز الحوت
(\cref{exo:g4:classifying-animals:9}) كما حُلّ: \omterm{def:g1:animals-around-us:covering}{فالفرو} \omterm{def:g2:animals-grow-up:born}{والحليب}
يعلّمان خط أسرة الحوت --- إذ كانت أسلافه ثدييات يابسة تُرضع
حليبًا --- أما شكله الشبيه بالسمكة فمن صنع الماء
(\cref{exo:g4:adapted-to-their-world:11})، مطبوعًا على أسرة لا
\omterm{prop:g6:classification-kinship:kinship}{قرابة} لها بالسمك بحكم طريقة العيش نفسها. فصفات النسب تفرز الأسر؛
وتشابهات \omterm{def:g2:where-animals-live:habitat}{الموئل} يستعيرها الغرباء. \omterm{def:g4:classifying-animals:classification}{والتصنيف} يثق بالأولى ولا يثق
بالثانية.
\end{example}

\begin{definition}[شجرة القرابة]\label{def:g6:classification-kinship:tree}
تُرسم \omterm{prop:g6:classification-kinship:kinship}{القرابة} \emph{شجرةً}\index{شجرة القرابة}: فنقاط التفرع تمثّل
الأسلاف المشتركين، وكل \omterm{def:g6:classification-kinship:attribute}{صفة} تظهر مرة واحدة عند الفرع الذي نشأت
عنده --- وكل \omterm{def:g5:biodiversity:species}{نوع} بعد ذلك المفرق يرثها. والصناديق المتداخلة والشجرة
تحملان المعلومة نفسها: فالصندوق فرع بكل أغصانه.
\end{definition}

\begin{method}[بناء شجرة قرابة صغيرة]\label{met:g6:classification-kinship:build}
إذا أُعطيت حفنة من \omterm{def:g5:biodiversity:species}{الأنواع}:
\begin{enumerate}
  \item جدولها بصفاتها --- \omterm{def:g5:biodiversity:species}{الأنواع} في الأسطر، والصفات القابلة
        للتحقق في الأعمدة، وعلامات حيث تُملك؛
  \item وجمّعها بحسب العلامات المشتركة، وأوسع \omterm{def:g6:classification-kinship:attribute}{صفة} أولًا: فيظهر
        التداخل في الجدول؛
  \item وارسم الشجرة: مفرقًا واحدًا لكل \omterm{def:g6:classification-kinship:attribute}{صفة}، والأوسع أقرب إلى
        الجذع؛ وضع كل \omterm{def:g5:biodiversity:species}{نوع} بعد المفارق التي يملك صفاتها بالضبط؛
  \item واقرأ \omterm{prop:g6:classification-kinship:kinship}{القرابة}: فقرب نوعين هو آخر مفرق يتشاركانه.
\end{enumerate}
\end{method}

\begin{remark}[ما تلمّح إليه الشجرة في هدوء]\label{rem:g6:classification-kinship:implies}
خذ الشجرة مأخذ الجدّ فتقل شيئًا هائلًا: تتبّع أي فرعين إلى الوراء
\emph{فيلتقيان}. فالقطة والسلمون المرقط يتشاركان نوعًا جدًّا؛ وكذلك،
أبعد إلى الوراء، القطة والحلزون، والأقحوانة والطفل. فتكون
\omterm{def:g6:exploring-our-environment:components}{الكائنات الحية} كلها أسرة واحدة، ووحدتها
(\cref{prop:g5:first-look-at-cells:principle}) شبهًا عائليًّا ---
ويكون على \omterm{def:g5:biodiversity:species}{الأنواع} أن تستطيع أن \emph{تتغير} عبر الأجيال، وإلا لما
حدث التفرع قط. أما أصحيح ذلك وكيف يمكن أن يعمل، فهو السؤال الأكبر
في سنة \cref{ch:g9:evidence-of-evolution} --- وهذه الشجرة هي الرسم
الذي يفرضه.
\end{remark}

\begin{exercise}[$\star\star\star$]\label{exo:g6:classification-kinship:12}

\end{exercise}

\section{مسألة: درج المتحف}

\begin{problem}[{مسألة نهاية الأسبوع --- ست عينات، وجدول صفات
واحد، وشجرة واحدة}]\label{pb:g6:classification-kinship:1}
يعير متحف الصفَّ درجًا فيه ست عينات: سلمون مرقط، وضفدع، وسحلية،
وحمامة، وفأر، \omterm{def:g3:bones-and-muscles:skeleton}{وجمجمة} دلفين مع قوالب زعانفه. والمهمة: أن تصنّفها
بالصفات وترسم شجرة قرابتها.

\medskip\noindent\textbf{القسم الأول --- الجدول.}
\begin{enumerate}
  \item ابنِ الجدول للستّ، بالصفات: \omterm{def:g4:classifying-animals:vertebrate}{عمود فقري}؛ وأربعة \omterm{def:g1:animals-around-us:parts}{أطراف}؛
        \omterm{def:g1:the-five-senses:sense}{وجلد} جاف حرشفي؛ \omterm{def:g1:animals-around-us:covering}{وريش}؛ \omterm{def:g1:animals-around-us:covering}{وفرو} \omterm{def:g2:animals-grow-up:born}{وحليب}. (وللدلفين، وهو شبه عارٍ
        من الشعر، تشهد \omterm{prop:g3:balanced-diet:jobs}{بطاقة} المتحف: صغاره تُرضع حليبًا ---
        و\cref{ex:g3:sorting-living-things:surprises} تشرح لماذا
        تحسم تلك الشهادة الأمر.)
  \item وأي \omterm{def:g6:classification-kinship:attribute}{صفة} تشترك فيها الستّ جميعًا؟ وما معنى أوسع اشتراك
        بلغة \omterm{prop:g6:classification-kinship:kinship}{القرابة}؟
  \item وأي العينات تُعلَّم بعبارة ``أربعة \omterm{def:g1:animals-around-us:parts}{أطراف}''؟ اشرح علامتَي
        الحمامة والدلفين --- ما الأجنحة والزعانف بمنطق الشجرة؟
  \item ولماذا لا تظهر ``تعيش في الماء'' في الجدول البتة، مع أن
        السلمون المرقط والدلفين كليهما يعيش فيه؟ استشهد بقاعدة
        الفصل.
\end{enumerate}

\medskip\noindent\textbf{القسم الثاني --- الشجرة.}
\begin{enumerate}[resume]
  \item ارسم الشجرة: رتّب المفارق --- \omterm{def:g3:bones-and-muscles:skeleton}{العمود الفقري}، \omterm{def:g1:my-body:parts}{فالأطراف}
        الأربعة، ثم الأكسية (الحراشف، \omterm{def:g1:animals-around-us:covering}{والريش}، \omterm{def:g1:animals-around-us:covering}{والفرو}) على فروعها
        الخاصة --- وضع الستّ جميعًا.
  \item اقرأ من شجرتك: أقرب نسيب للفأر في الدرج، وأقرب نسيب
        للسلمون المرقط.
  \item يقف الضفدع بعد ``\omterm{def:g1:my-body:parts}{الأطراف} الأربعة'' وقبل كل مفرق كساء.
        فعلامَ يعلّمه جلده العاري الرطب
        (\cref{rem:g4:classifying-animals:confusion})، وأين
        يغادر فرعه؟
  \item تسقط \omterm{prop:g3:balanced-diet:jobs}{بطاقة} عن عينة سابعة: \omterm{def:g1:the-five-senses:sense}{جلد} جاف حرشفي، وأربعة \omterm{def:g1:animals-around-us:parts}{أطراف}،
        \omterm{def:g4:classifying-animals:vertebrate}{وعمود فقري}. ضعها على الشجرة وسمِّ طائفتها --- من غير أن
        ترى البطاقة.
\end{enumerate}

\medskip\noindent\textbf{القسم الثالث --- المعنى.}
\begin{enumerate}[resume]
  \item يتشارك الدلفين والسلمون المرقط الانسيابية؛ ويتشارك
        الدلفين والفأر نسب \omterm{def:g1:animals-around-us:covering}{الفرو} \omterm{def:g2:animals-grow-up:born}{والحليب}. فأي زوج يقع أقرب على
        الشجرة، ولماذا يجب على الشجرة أن تتجاهل الانسيابية؟
  \item اذكر ماذا يزعم مفرق ``\omterm{def:g1:my-body:parts}{الأطراف} الأربعة'' تاريخيًّا: ما
        الشيء الواحد الذي ينحدر منه، منذ زمن بعيد، الضفدعُ
        والسحلية والحمامة والفأر والدلفين جميعًا؟
  \item شجرة الصف تجعل كل فرع يلتقي في النهاية عند الجذع. قل في
        جملة واحدة ماذا يزعم ذلك الرسم في هدوء عن العينات الستّ
        جميعًا --- وعن أفراد الصف الراسمين له.
  \item يقول متشكك: ``شجرتكم مجرد نظام حفظ ملفات.'' أعطِ الردّ
        من شطرين: علام يكون تداخل الصفات \emph{واقعةً}، وما الذي
        \emph{تشرحه} الشجرة مما يتركه مجرد حفظ الملفات بلا شرح.
\end{enumerate}
\end{problem}
