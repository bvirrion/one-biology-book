\chapter{التنوع الحيوي}\label{ch:g5:biodiversity}

خذ مترًا مربعًا واحدًا من مرج مزهر وابدأ العدّ: عشرات الأصناف من
\omterm{def:g1:plants-around-us:plant}{النباتات}، وخنافس وبقّ لا تُعدّ، وعناكب وحلزونات وديدان تحتها
وفراشات فوقها. ثم جرّب المتر المربع نفسه من ساحة مدرسة مبلّطة.
وللفرق بين ذينك العددين اسم --- \omterm{def:g5:biodiversity:biodiversity}{التنوع الحيوي} --- وقد صار من أهم
الكلمات في \omterm{def:g1:living-or-not:living}{علم الأحياء}.

\section{تنوع الحياة}

\begin{definition}[النوع]\label{def:g5:biodiversity:species}
\emph{النوع}\index{نوع} صنف بعينه من الكائنات الحية: أي الأفراد
كلهم المبنيون على النموذج نفسه، والذين يتكاثرون فيما بينهم. فعصفور
الدوري نوع؛ وكذلك الأقحوان الشائع، والثعلب الأحمر، وحلزون الحديقة.
\omterm{prop:g4:classifying-animals:classes}{وطوائف} \cref{ch:g4:classifying-animals} صناديق تضم أنواعًا كثيرة:
فصندوق ``\omterm{prop:g3:sorting-living-things:groups}{الثدييات}'' يضم آلافًا منها.
\end{definition}

\begin{definition}[التنوع الحيوي]\label{def:g5:biodiversity:biodiversity}
\emph{التنوع الحيوي}\index{تنوع حيوي} في مكان ما هو تنوع الحياة
التي يحتويها --- وقبل كل شيء كم \emph{نوعًا} مختلفًا يعيش هناك،
وكيف حال كل واحد منها. فالمرج المزهر الغني \omterm{def:g5:biodiversity:species}{بالأنواع} تنوعه الحيوي
عالٍ؛ أما الساحة المبلّطة فلا تنوع فيها تقريبًا.
\end{definition}

\begin{omfigure}
\includegraphics[width=0.62\linewidth]{images/book1/ai/g5-biodiversity-meadow.png}

{\small متر مربع واحد من المرج في وسط العدّ: أعشاب، وأزهار،
وفراشات، وخنافس، وحلزون --- والعدّ لم يكد يبدأ.}
\end{omfigure}

\begin{example}[عدّ الكوكب]\label{ex:g5:biodiversity:planet}
سمّى العلماء نحو مليوني \omterm{def:g5:biodiversity:species}{نوع} حتى الآن --- وهم على يقين من أن ذلك
جزء من المجموع فحسب، إذ تُكتشف \omterm{def:g5:biodiversity:species}{أنواع} جديدة في كل بعثة جادّة:
والخنافس قبل كل شيء (فلا مجموعة أكثر منها أنواعًا مسمّاة)، وكذلك
\omterm{prop:g3:sorting-living-things:groups}{الأسماك} والضفادع \omterm{def:g1:plants-around-us:plant}{والنباتات}، بل ثدييات أحيانًا. وعدد \omterm{def:g5:biodiversity:species}{أنواع} الأرض
الكامل ما زال مجهولًا --- وأكثر التقديرات تبلغ ملايين فوق ما
سُمّي.
\end{example}

\begin{remark}[تنوع على كل مقياس]\label{rem:g5:biodiversity:scales}
يظهر \omterm{def:g5:biodiversity:biodiversity}{التنوع الحيوي} على مقاييس عدة في وقت واحد: \omterm{def:g5:biodiversity:species}{أنواع} كثيرة في \omterm{def:g2:where-animals-live:habitat}{موئل}
واحد؛ \omterm{def:g2:where-animals-live:habitat}{وموائل} مختلفة كثيرة جنبًا إلى جنب (بركة، وسياج نباتي، ومرج
--- ولكل واحد أنواعه، كما بينت
\cref{ex:g2:where-animals-live:four})؛ وداخل \omterm{def:g5:biodiversity:species}{النوع} الواحد لا
يتشابه فردان تمام التشابه --- فلا حلزونا حديقة يلبسان \omterm{def:g1:animals-around-us:covering}{الصدفة} نفسها
تمامًا. تنوع داخل تنوع داخل تنوع.
\end{remark}

\section{لماذا يهمّ التنوع الحيوي}

\begin{proposition}[الشبكة تحتاج إلى خيوطها]\label{prop:g5:biodiversity:web}
\omterm{def:g5:biodiversity:species}{أنواع} \omterm{def:g2:where-animals-live:habitat}{الموئل} يغذّي بعضها بعضًا، ويؤويه، ويؤبّره، وينظّف بعده ---
فهي خيوط شبكة واحدة (\cref{rem:g3:food-chains:web}). والشبكة
الغنية بالخيوط الكثيرة تستطيع أن تمتصّ صدمة: فإن أخفق \omterm{def:g5:biodiversity:species}{نوع} فريسة،
انتقل صائدوه إلى غيره. أما الشبكة الفقيرة فتتمزق بسهولة: إذ يترك
كل \omterm{def:g5:biodiversity:species}{نوع} مفقود للباقين خيارات أقل. \omterm{def:g5:biodiversity:biodiversity}{فالتنوع الحيوي} قوة الشبكة.
\end{proposition}
\admitted

\begin{example}[اختبار البستان]\label{ex:g5:biodiversity:orchard}
بستان فيه أسيجة نباتية وأشرطة أزهار برية وأشجار عتيقة يؤوي عشرات
\omterm{def:g5:biodiversity:species}{الأنواع} المعينة: فالنحل والذباب المحلّق يؤبّران الزهر
(\cref{rem:g3:flowers-fruits-seeds:bees})، والدعسوقات تأكل مَنّ
\omterm{def:g1:plants-around-us:plant}{النبات}، \omterm{prop:g3:sorting-living-things:groups}{والطيور} تجوب باحثة عن \omterm{ex:g2:animals-grow-up:butterfly}{اليرقات}، والديدان تُبقي التربة
مفتوحة. فاقتلع الأسيجة واجزز الأزهار، فيرحل المعينون --- وعندئذ
على مزارع الفاكهة أن يؤدي أعمالهم وحده، وهي أعمال كانت الشبكة
تؤديها مجانًا.
\end{example}

\begin{example}[الناس يعيشون من الشبكة]\label{ex:g5:biodiversity:people}
كل ما \omterm{def:g3:food-chains:chain}{يأكله} البشر يعود إلى \omterm{def:g5:biodiversity:species}{أنواع} حية
(\cref{rem:g4:what-plants-need:everyone})؛ وأكثر أدويتنا وُجدت أول
مرة في \omterm{def:g1:plants-around-us:plant}{نباتات} وعفن وكائنات حية أخرى؛ والقطن والصوف والخشب \omterm{def:g1:the-five-senses:sense}{والجلد}
منتجات \omterm{def:g5:biodiversity:species}{أنواع}. فخيوط الشبكة تمرّ مباشرة في كل مطبخ وصيدلية وخزانة
ثياب.
\end{example}

\section{التنوع الحيوي يمكن أن يُفقد --- ويمكن أن يُعان}

\begin{proposition}[كيف يُفقد التنوع]\label{prop:g5:biodiversity:loss}
يتضاءل \omterm{def:g5:biodiversity:species}{النوع} حين يُنتزع ما يعتمد عليه:
\begin{itemize}
  \item \emph{موئله} يُدمَّر --- كالبركة المجفَّفة في
        \cref{prop:g2:where-animals-live:loss}، والسياج
        المقتلَع، والمرج المبلّط؛
  \item وبيئته تُسمَّم --- بالنفايات والمواد الكيميائية
        (\cref{prop:g2:caring-for-nature:harm})؛
  \item ويُؤخذ منه أكثر من اللازم --- بإفراط الصيد أو القطف أو
        القنص.
\end{itemize}
\omterm{def:g5:biodiversity:species}{والنوع} الذي ذهب كليًّا --- في كل مكان وإلى الأبد ---
\emph{منقرض}\index{انقراض}. والانقراض ليس كلعبة خاسرة: فلا جولة
تالية.
\end{proposition}
\admitted

\begin{method}[إعانة التنوع الحيوي بدءًا من الصغير]\label{met:g5:biodiversity:help}
وصفات \cref{met:g2:caring-for-nature:rules} تكبر معنا:
\begin{enumerate}
  \item احفظ التنوع وازرعه --- فسياج من خمس شجيرات خير من سور،
        وشريط أزهار خير من حصى عارٍ؛
  \item واترك أركانًا برية: فالخشب الميت وأكوام \omterm{def:g1:plants-around-us:parts}{الأوراق} والعشب غير
        المجزوز \omterm{def:g2:where-animals-live:habitat}{موائل}
        (\cref{ex:g2:caring-for-nature:home})؛
  \item وأعِن المعينين: بصناديق أعشاش، ومآوي حشرات، وبركة مهما
        تصغر
        (\cref{ex:g2:where-animals-live:schoolpond})؛
  \item وعُدّ ما يعيش حولك سنة بعد سنة --- فما يُعدّ يُلاحَظ، وما
        يُلاحَظ يُحمى.
\end{enumerate}
\end{method}

\begin{example}[عودة اللقالق]\label{ex:g5:biodiversity:storks}
حيث أُعيدت المروج الرطبة ورُفعت منصّات الأعشاش، عادت اللقالق ---
وقد كادت تختفي من أقاليم بأكملها --- لتتوالد، وعادت معها الضفادع
والرعّاشات وأزهار المروج. \omterm{def:g5:biodiversity:species}{فالأنواع} تعود حين تُربط خيوطها من جديد:
والحماية تنجح، ويمكن مشاهدة نتائجها من ساحة القرية.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:biodiversity:1}
ما \omterm{def:g5:biodiversity:species}{النوع}؟ أعطِ ثلاثة أمثلة من هذا الفصل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:biodiversity:2}
ما \omterm{def:g5:biodiversity:biodiversity}{التنوع الحيوي}؟ قارن المرج بالساحة المبلّطة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:biodiversity:3}
كم نوعًا سُمّي حتى الآن تقريبًا؟ ولماذا كان العدد الحقيقي أعلى
يقينًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:biodiversity:4}
سمِّ مقاييس التنوع الثلاثة في \cref{rem:g5:biodiversity:scales}،
مع مثال كل واحد.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:biodiversity:5}
لماذا تنجو الشبكة الغنية بالأنواع من الصدمات أحسن من الفقيرة؟
استعمل حجة انتقال الصائدين إلى فريسة أخرى.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:biodiversity:6}
عدّد أربعة \omterm{def:g5:biodiversity:species}{أنواع} معينة في البستان والعمل المجاني الذي يؤديه كل
واحد.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:biodiversity:7}
اذكر الطرق الثلاث التي يتضاءل بها \omterm{def:g5:biodiversity:species}{النوع}، مع مثال على كل واحدة. وما
معنى \emph{\omterm{prop:g5:biodiversity:loss}{منقرض}}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:biodiversity:8}
جرّب: عُدّ \omterm{def:g5:biodiversity:species}{الأنواع} --- \omterm{def:g1:plants-around-us:plant}{نباتات} \omterm{def:g1:animals-around-us:animal}{وحيوانات}، مسمّاة أو موصوفة --- في
مربع صغير تختاره في الخارج. ثم عُدّ مربعًا مبلّطًا للمقارنة.
وعدداك قياس \omterm{def:g5:biodiversity:biodiversity}{للتنوع الحيوي}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:biodiversity:9}
``فقد \omterm{def:g5:biodiversity:species}{نوع} خنفساء واحد من بين آلاف لا يمكن أن يهمّ.'' أجب وفق
\cref{prop:g5:biodiversity:web} --- وبأن لا أحد يعرف كل الخيوط
التي يمسكها \omterm{def:g5:biodiversity:species}{نوع} ما.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:biodiversity:10}
تسمّى حكاية اللقالق نجاحًا لحماية \emph{\omterm{def:g2:where-animals-live:habitat}{الموئل}} لا لحماية اللقالق
واحدًا واحدًا. اشرح الفرق مستعملًا
\cref{prop:g2:where-animals-live:loss} وما عاد إلى جانب اللقالق.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:biodiversity:11}
يستبدل صاحب بستان بالأسيجة وأشرطة الأزهار أسوارًا وحصًى، ثم يضطر
إلى استئجار \omterm{def:g5:first-look-at-cells:cell}{خلايا} نحل للتأبير والرشّ ضد مَنّ \omterm{def:g1:plants-around-us:plant}{النبات} --- فينفق مالًا
على أعمال كانت تُؤدى مجانًا. استعمل \cref{ex:g5:biodiversity:orchard}
لتشرح ما الذي أُزيل حقًّا --- ولماذا لا تظهر حسابات عمل الطبيعة
المجاني في العادة إلا بعد أن يتوقف.
\end{exercise}
