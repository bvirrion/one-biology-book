\chapter{Comment les espèces changent}\label{ch:g9:how-species-change}

La salle des preuves a établi le fait~; ce chapitre
construit le moteur. Toutes ses pièces sont déjà entre tes
mains~: la \omterm{prop:g9:genes-and-dna:explains}{mutation} qui écrit des orthographes neuves
(\cref{prop:g9:genes-and-dna:explains}), le brassage qui
distribue des jeux uniques
(\cref{prop:g9:unique-individuals:theorem}), les freins du
milieu qui prélèvent la plus grande part du surplus de
chaque génération
(\cref{prop:g8:reproduction-and-environment:limits}).
Assemblés, ils tournent tout seuls --- et ce fonctionnement
autonome est la \omterm{prop:g9:how-species-change:selection}{sélection naturelle}, l'idée simple la plus
lourde de conséquences de toute la \omterm{def:g1:living-or-not:living}{biologie}.

\section{Le moteur, assemblé}

\begin{proposition}[La sélection naturelle]\label{prop:g9:how-species-change:selection}
Trois faits, tous indéniables, et leur conséquence~:
\begin{enumerate}
  \item la \emph{variété}~: les individus d'une \omterm{def:g8:reproduction-and-environment:population}{population}
        diffèrent par des \omterm{def:g9:heredity-and-traits:heredity}{caractères héréditaires}
        (\cref{prop:g9:unique-individuals:reserve} l'a mise
        en banque)~;
  \item la \emph{surproduction et les freins}~: il naît
        beaucoup plus de jeunes que le milieu n'en laisse
        survivre --- les freins prélèvent le surplus
        (\cref{ex:g8:reproduction-and-environment:twosprings})~;
  \item le \emph{prélèvement inégal}~: les freins ne
        prélèvent pas au hasard --- les individus dont les
        \omterm{def:g9:heredity-and-traits:heredity}{caractères héréditaires} conviennent aux \omterm{def:g6:exploring-our-environment:conditions}{conditions}
        du moment leur échappent un peu plus souvent, et
        laissent donc plus de descendants,
        \emph{qui héritent de ces \omterm{def:g9:heredity-and-traits:traits}{caractères}}.
\end{enumerate}
Conséquence~: génération après génération, la part des
\omterm{def:g9:genes-and-dna:gene}{allèles} qui conviennent grandit dans la \omterm{def:g8:reproduction-and-environment:population}{population}. Cette
dérive de la constitution héréditaire d'une \omterm{def:g8:reproduction-and-environment:population}{population},
poussée par rien d'autre que la variété rencontrant les
freins, est la \emph{sélection
naturelle}\index{sélection naturelle} --- et avec assez de
temps, elle fait l'\emph{évolution}\index{évolution}.
\end{proposition}
\admitted

\begin{example}[Les phalènes, mécanisées]\label{ex:g9:how-species-change:moths}
Les phalènes du bouleau de
\cref{ex:g9:evidence-of-evolution:live}, pièce par pièce.
Variété~: des formes claire et sombre \omterm{def:g9:heredity-and-traits:heredity}{héréditaires}. Frein~:
les \omterm{prop:g3:sorting-living-things:groups}{oiseaux} chassent les phalènes à vue sur l'écorce.
Prélèvement inégal~: sur des troncs noircis de suie, c'est
la forme claire qui se voit --- elle est mangée davantage,
si bien que la part de l'\omterm{def:g9:genes-and-dna:gene}{allèle} sombre grimpe vers les bois
sombres observés~; l'air s'éclaircit, l'écorce pâlit, et le
même moteur tourne à l'envers. Aucune phalène n'a changé~;
c'est la constitution de la \emph{\omterm{def:g8:reproduction-and-environment:population}{population}} qui a changé
--- la sélection corrige les fréquences du jeu, non les
individus.
\end{example}

\begin{remark}[Ce que la sélection n'est pas]\label{rem:g9:how-species-change:not}
Trois confusions tenaces, mises à la retraite~:
\begin{itemize}
  \item \emph{aucune intention}~: les \omterm{def:g6:exploring-our-environment:conditions}{conditions}
        sélectionnent comme l'\omterm{def:g6:exploring-our-environment:conditions}{humidité} «~choisissait~» les
        cloportes
        (\cref{ex:g6:exploring-our-environment:chooser})
        --- par conséquences, non par choix~; rien ne
        planifie~;
  \item \emph{aucune perfection}~: la sélection ne compare
        que les variantes \emph{présentes} --- elle trouve
        la meilleure disponible, non la meilleure
        imaginable (le rechargement maladroit du plan des
        \omterm{def:g1:my-body:parts}{membres} partagé,
        \cref{exo:g9:evidence-of-evolution:7}, en est la
        signature)~;
  \item \emph{aucun héritage de l'effort}~: le \omterm{def:g1:my-body:parts}{bras} du
        forgeron meurt avec le forgeron
        (\cref{prop:g9:heredity-and-traits:sorting})~; la
        sélection ne travaille que sur ce que portent les
        \omterm{def:g8:sexual-reproduction:gametes}{gamètes}. Les girafes n'ont pas allongé leur cou à
        force de tirer --- des variantes à cou plus long
        se sont mieux nourries et ont laissé plus de
        descendants que les variantes à cou plus court.
\end{itemize}
\end{remark}

\section{Le moteur observé et exploité}

\begin{example}[La sélection à l'hôpital]\label{ex:g9:how-species-change:resistance}
Le cas médical, mécanisé pour l'usage de
\cref{ch:g9:microbes-and-infection}~: une \omterm{def:g8:reproduction-and-environment:population}{population} de
\omterm{def:g6:producing-our-food:fermentation}{microbes} contient de rares mutants résistants (le filet de
la \omterm{prop:g9:genes-and-dna:explains}{mutation})~; un antibiotique est le frein~; il prélève les
sensibles et épargne les résistants, qui héritent du
service. Des jours, non des millénaires --- les générations
tournent en quelques heures, et la vitesse de la sélection
suit la leur. Toutes les règles sur les cures
d'antibiotiques à finir et les antibiotiques à ne pas
répandre dans l'\omterm{def:g6:producing-our-food:farming}{élevage} sont de la conduite de moteur~:
applique le frein à moitié, et tu élèves ce que tu voulais
tuer.
\end{example}

\begin{example}[La sélection avec une main humaine sur le frein]\label{ex:g9:how-species-change:artificial}
La sélectionneuse de \cref{pb:g8:sexual-reproduction:1}
faisait tourner le même moteur en se mettant elle-même à la
place du frein --- gardant les pommes croquantes et
résistantes, éliminant le reste. Chaque race de chien,
chaque variété de \omterm{def:g6:producing-our-food:farming}{culture}, chaque lignée laitière est de la
\emph{sélection artificielle} accumulée~: la preuve de la
puissance du moteur à l'échelle d'une vie humaine, du loup
au chien de salon en quelques milliers de générations.
Darwin --- le naturaliste qui a assemblé l'argument de ce
chapitre --- ouvrait son grand livre exactement sur cette
comparaison.
\end{example}

\begin{proposition}[Du changement aux espèces nouvelles]\label{prop:g9:how-species-change:speciation}
La sélection explique qu'une \omterm{def:g8:reproduction-and-environment:population}{population} change~; un
ingrédient de plus explique que des \omterm{def:g8:reproduction-and-environment:population}{populations} se
\emph{séparent}. Sépare deux \omterm{def:g8:reproduction-and-environment:population}{populations} d'une même \omterm{def:g5:biodiversity:species}{espèce}
--- une mer monte, une chaîne de montagnes coupe, une île
reçoit des naufragés (\cref{exo:g6:how-plants-spread:11})
--- et chacune fait tourner le moteur dans ses propres
\omterm{def:g6:exploring-our-environment:conditions}{conditions}, en accumulant ses propres \omterm{def:g9:genes-and-dna:gene}{allèles}. Avec assez
de divergence, les deux ne se reproduisent plus entre elles
là où elles se rencontrent~: par la définition même de
\cref{def:g5:biodiversity:species}, deux \omterm{def:g5:biodiversity:species}{espèces} se
tiennent là où il y en avait une. Les fourches de l'arbre
(\cref{def:g6:classification-kinship:tree}) sont exactement
de telles séparations, à hauteur de temps profond.
\end{proposition}
\admitted

\begin{method}[Faire tourner le moteur sur n'importe quel cas]\label{met:g9:how-species-change:run}
Pour tout exemple d'\omterm{prop:g9:how-species-change:selection}{évolution} qu'on te propose~:
\begin{enumerate}
  \item nomme la \emph{variété} et vérifie qu'elle est
        héréditaire
        (\cref{met:g9:heredity-and-traits:audit})~;
  \item nomme le \emph{frein}, et dis comment il prélève
        inégalement dans cette variété~;
  \item suis la \emph{comptabilité}~: de qui les \omterm{def:g9:genes-and-dna:gene}{allèles}
        gagnent-ils de la part à la génération suivante~?
  \item pour une séparation, ajoute la \emph{séparation}
        elle-même et les \omterm{def:g6:exploring-our-environment:conditions}{conditions} qui divergent~;
  \item vérification de l'échelle de temps~: ce sont les
        générations, non les années, qui sont les
        battements du moteur.
\end{enumerate}
\end{method}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:1}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:2}
Dans le cas des phalènes, nomme la variété, le frein et le

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:3}
Qu'est-ce qui change sous l'effet de la sélection --- les
individus, ou la \omterm{def:g8:reproduction-and-environment:population}{population}~? Explique.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:4}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:6}
Qu'avaient en commun la sélectionneuse de pommes et les

\begin{exercise}[$\star\star$]\label{exo:g9:how-species-change:7}

\begin{exercise}[$\star\star$]\label{exo:g9:how-species-change:9}

\section{Problème~: les carnets de l'archipel}

\begin{problem}[{Problème du week-end --- deux siècles sur
un archipel volcanique}]\label{pb:g9:how-species-change:1}
Un archipel (composite, inspiré de l'histoire)~: des îles
volcaniques, jamais reliées au continent, qui abritent des
\omterm{prop:g3:sorting-living-things:groups}{oiseaux} du genre des pinsons dont la forme du bec diffère
d'une île à l'autre --- et diffère de leur unique parent
continental évident. Les carnets des naturalistes couvrent
deux siècles.

\medskip\noindent\textbf{Partie I --- Le motif.}
\begin{enumerate}
  \item Les îles n'abritent que des \omterm{def:g5:biodiversity:species}{espèces} capables de
        traverser la mer --- des \omterm{prop:g3:sorting-living-things:groups}{oiseaux}, des \omterm{prop:g3:flowers-fruits-seeds:transform}{graines}
        flottantes, une chauve-souris, aucun mammifère
        terrestre ni grenouille indigènes. Quel chapitre
        prédisait cette liste de passagers
        (\cref{exo:g6:how-plants-spread:11})~?
  \item Tous les pinsons de l'archipel ressemblent à
        l'unique \omterm{def:g5:biodiversity:species}{espèce} continentale plus qu'à n'importe
        quel autre oiseau de la Terre. Que conclut sur leur
        origine le raisonnement de \omterm{prop:g6:classification-kinship:kinship}{parenté}
        (\cref{prop:g6:classification-kinship:kinship})~?
  \item Les becs diffèrent selon le régime de l'île~: des
        casseurs de \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} sur les îles à \omterm{prop:g3:flowers-fruits-seeds:transform}{graines}, des
        sondeurs sur les îles à cactus. Lis cet
        appariement avec
        \cref{prop:g4:adapted-to-their-world:answers} ---
        puis dis ce que ce chapitre ajoute que le chapitre
        sur l'\omterm{def:g4:adapted-to-their-world:adaptation}{adaptation} ne pouvait pas donner.
  \item Pourquoi des îles volcaniques jamais reliées
        sont-elles la vitrine du moteur~? Nomme
        l'ingrédient de la formation d'\omterm{def:g5:biodiversity:species}{espèces} qu'elles
        fournissent en gros.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Les années du
moteur dans les carnets.}
\begin{enumerate}[resume]
  \item Une année de sécheresse~: les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} tendres
        manquent, les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} dures restent~; le pied à
        coulisse des bagueurs montre que les jeunes de
        l'année suivante ont en moyenne des becs plus
        épais. Fais tourner la comptabilité du moteur à
        travers cette phrase.
  \item Des années humides suivent, les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} tendres
        reviennent, et la moyenne dérive en sens inverse.
        Que prouve ce retour en arrière au sujet de la
        «~direction~» de la sélection~?
  \item Les pinsons d'une île, transportés par une tempête
        sur une autre, se reproduisent mal avec les
        résidents. Quelle proposition est en train d'être
        observée en cours de route~?
  \item Un carnet de l'époque créationniste objecte~:
        «~personne n'a jamais vu une \omterm{def:g5:biodiversity:species}{espèce} en devenir une
        autre.~» Réponds par l'échelle de temps et par la
        méthode des témoins convergents
        (\cref{met:g9:evidence-of-evolution:weigh}).
\end{enumerate}

\medskip\noindent\textbf{Partie III --- La conférence de
synthèse.}
\begin{enumerate}[resume]
  \item Raconte l'histoire de l'archipel selon les cinq
        étapes du moteur
        (\cref{met:g9:how-species-change:run}), de
        l'arrivée des naufragés aux \omterm{def:g5:biodiversity:species}{espèces} insulaires.
  \item Explique pourquoi le pinson de chaque île est
        «~le meilleur disponible, non le meilleur
        imaginable~» --- qu'est-ce que le jeu de chaque île
        pouvait offrir à la sélection~?
  \item Relie l'archipel au service hospitalier de
        \cref{ex:g9:how-species-change:resistance}~: même
        moteur, cadence différente --- dis ce qui fixe
        cette cadence.
  \item Termine la conférence par la comparaison de Darwin
        rendue explicite~: ce qu'ont en commun la main de
        l'éleveur de pigeons et les \omterm{prop:g3:flowers-fruits-seeds:transform}{graines} dures de la
        sécheresse.
\end{enumerate}
\end{problem}
