\section*{Capítulo \ref{ch:g4:heart-and-blood} --- O coração e o sangue}

\begin{solution}{exo:g4:heart-and-blood:1}
\omterm{def:g4:journey-of-food:digestion}{Nutrientes}, recolhidos no \omterm{def:g4:journey-of-food:tract}{intestino delgado}; e oxigênio, recolhido
nos pulmões.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:2}
Gás carbônico --- deixado nos pulmões, onde é expirado.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:3}
Ele é um \omterm{def:g3:bones-and-muscles:muscle}{músculo} oco, mais ou menos do tamanho do \omterm{def:g1:my-body:joint}{punho} fechado do
dono, no meio do peito, atrás da caixa torácica.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:4}
O lado direito bombeia o \omterm{prop:g4:journey-of-food:absorb}{sangue} até os pulmões para carregar
oxigênio; o lado esquerdo bombeia o \omterm{prop:g4:journey-of-food:absorb}{sangue} rico em oxigênio para o
corpo inteiro.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:5}
A pequena onda que cada \omterm{def:g4:heart-and-blood:heart}{batida do coração} empurra pelos \omterm{def:g4:heart-and-blood:vessels}{vasos},
sentida no \omterm{prop:g4:heart-and-blood:pulse}{pulso} ou no pescoço. Contar essa onda é contar as \omterm{def:g4:heart-and-blood:heart}{batidas
do coração} por minuto.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:6}
O polegar tem um \omterm{prop:g4:heart-and-blood:pulse}{pulso} próprio bem perceptível, então você poderia
acabar contando o seu polegar em vez do \omterm{prop:g4:heart-and-blood:pulse}{pulso} que está medindo.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:7}
As \omterm{def:g1:my-body:parts}{pernas} em corrida queimam \omterm{def:g4:journey-of-food:digestion}{nutrientes} e oxigênio depressa e
precisam recebê-los de volta depressa; o \omterm{def:g4:heart-and-blood:heart}{coração} acelera e reforça
as \omterm{def:g4:heart-and-blood:heart}{batidas} para o \omterm{prop:g4:journey-of-food:absorb}{sangue} fazer as rodadas de entrega com mais
frequência.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:8}
Resposta aberta. Em geral cerca de $80$--$100$ em repouso e
$120$--$150$ depois dos polichinelos --- a cadeia de abastecimento em
ação.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:9}
Ao sair dos pulmões ela tinha acabado de carregar oxigênio ---
vermelha viva. No \omterm{def:g3:bones-and-muscles:muscle}{músculo} ela entregou o oxigênio e recolheu gás
carbônico --- mais escura. A cor acompanha a carga.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:10}
O \omterm{def:g4:journey-of-food:tract}{intestino delgado} põe \omterm{def:g4:journey-of-food:digestion}{nutrientes} no \omterm{prop:g4:journey-of-food:absorb}{sangue}; os pulmões põem
oxigênio e tiram gás carbônico; o \omterm{def:g4:heart-and-blood:heart}{coração} é a bomba que mantém o
\omterm{prop:g4:journey-of-food:absorb}{sangue} circulando entre eles e chegando a todos os consumidores ---
\omterm{def:g3:bones-and-muscles:muscle}{músculos}, cérebro, \omterm{def:g3:bones-and-muscles:skeleton}{ossos}, \omterm{def:g1:the-five-senses:sense}{pele}.
\end{solution}

\begin{solution}{exo:g4:heart-and-blood:11}
O \omterm{def:g4:heart-and-blood:heart}{coração} treinado é um \omterm{def:g3:bones-and-muscles:muscle}{músculo} mais forte: cada \omterm{def:g4:heart-and-blood:heart}{batida} move mais
\omterm{prop:g4:journey-of-food:absorb}{sangue}. Entregar o mesmo abastecimento passa então a exigir menos
\omterm{def:g4:heart-and-blood:heart}{batidas} por minuto --- então os calmos $50$ do ciclista fazem o
trabalho dos $75$ de quem não treina, com potência sobrando de
reserva para o esforço.
\end{solution}
