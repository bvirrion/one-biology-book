\chapter{A comunicação nervosa}\label{ch:g8:nervous-communication}

O \cref{ch:g5:brain-in-command} desenhou o circuito --- informar,
decidir, ordenar --- e mediu o quinto de segundo dele. Neste ano
abrimos os cabos. O que de fato viaja por um \omterm{def:g8:nervous-communication:neuron}{nervo}? De que são feitos
os \omterm{def:g5:brain-in-command:system}{nervos}? E o que acontece nas fendas em que um fio termina e o
seguinte começa --- as fendas em que, como se verá, tanto a memória
quanto muitos venenos fazem o trabalho deles?

\section{A fiação e as células dela}

\begin{definition}[Neurônios e mensagens nervosas]\label{def:g8:nervous-communication:neuron}
As \omterm{def:g5:first-look-at-cells:cell}{células} de trabalho do \omterm{def:g5:brain-in-command:system}{sistema nervoso} são os
\emph{neurônios}\index{neurônio} (a
\cref{rem:g6:cell-unit-of-life:twoways} já os listou): cada um tem um
corpo celular e uma fibra comprida e fina --- algumas fibras vão da
\omterm{def:g3:bones-and-muscles:skeleton}{coluna} até o dedo do pé, as \omterm{def:g5:first-look-at-cells:cell}{células} mais compridas do corpo. Um
\emph{nervo}\index{nervo} é um cabo que reúne milhares dessas fibras.
Por cada fibra viajam as \emph{mensagens nervosas}\index{mensagem
nervosa}: sinais elétricos breves, correndo a até \qty{100}{m/s} (o
número da \cref{def:g5:brain-in-command:system}, agora com o portador
dele).
\end{definition}

\begin{proposition}[Codificação por frequência]\label{prop:g8:nervous-communication:code}
Os sinais em si são todos iguais --- o que varia é o \emph{ritmo}
deles. Um toque leve manda um tamborilar lento pela fibra sensitiva;
um toque forte, um rufar rápido. A língua do \omterm{def:g5:brain-in-command:system}{sistema nervoso} tem uma
palavra só, repetida em velocidades diferentes: a intensidade é
codificada como frequência, em toda linha, de entrada e de saída.
\end{proposition}
\admitted

\begin{example}[Três linhas, uma língua]\label{ex:g8:nervous-communication:lines}
O elenco do circuito (\cref{prop:g5:brain-in-command:loop}), refeito
na fiação: os \emph{\omterm{def:g8:nervous-communication:neuron}{neurônios} sensitivos} levam os informes --- da \omterm{def:g1:the-five-senses:sense}{pele}
à medula, do \omterm{def:g1:the-five-senses:sense}{olho} ao \omterm{def:g5:brain-in-command:system}{cérebro}; os \emph{\omterm{def:g8:nervous-communication:neuron}{neurônios} motores} levam as
ordens --- da medula ao \omterm{def:g3:bones-and-muscles:muscle}{músculo}, terminando onde a fibra encontra o
\omterm{def:g3:bones-and-muscles:muscle}{músculo} e onde o rufar dela define a força da contração
(\cref{def:g3:bones-and-muscles:muscle}); e, entre uns e outros, no
\omterm{def:g5:brain-in-command:system}{cérebro} e na medula, incontáveis \emph{\omterm{def:g8:nervous-communication:neuron}{neurônios} de conexão} fazem a
decisão. Reflexo ou sinfonia, o tráfego são os mesmos sinais em ritmos
diferentes em linhas diferentes.
\end{example}

\section{As fendas: as sinapses}

\begin{definition}[A sinapse]\label{def:g8:nervous-communication:synapse}
Os \omterm{def:g8:nervous-communication:neuron}{neurônios} não se tocam. Onde a ponta de uma fibra encontra a \omterm{def:g5:first-look-at-cells:cell}{célula}
seguinte existe uma fenda microscópica --- a
\emph{sinapse}\index{sinapse}. O sinal elétrico não consegue saltá-la;
em vez disso, a fibra que chega libera um \emph{mensageiro químico} na
fenda, que a atravessa, se acopla do outro lado e --- se chegar o
bastante --- inicia o sinal de novo no \omterm{def:g8:nervous-communication:neuron}{neurônio} seguinte. Elétrico ao
longo do fio, químico em cada junção: o sistema inteiro alterna os
dois.
\end{definition}

\begin{proposition}[Por que fendas, e não solda]\label{prop:g8:nervous-communication:whygaps}
As fendas parecem um defeito e são a genialidade do sistema:
\begin{itemize}
  \item elas tornam as junções \emph{ajustáveis} --- uma \omterm{def:g8:nervous-communication:synapse}{sinapse}
        atravessada com frequência fica mais forte, uma \omterm{def:g8:nervous-communication:synapse}{sinapse} sem
        uso desaparece: é o aprendizado (os ``caminhos melhores'' da
        \cref{rem:g5:brain-in-command:training}, encontrados --- são
        \omterm{def:g8:nervous-communication:synapse}{sinapses} fortalecidas);
  \item elas tornam os sinais de \emph{mão única} --- os mensageiros
        são liberados só de um lado, então o tráfego não pode correr
        para trás;
  \item elas deixam um \omterm{def:g8:nervous-communication:neuron}{neurônio} \emph{pesar} milhares de entradas ---
        vozes que estimulam e vozes que freiam, somadas em cada \omterm{def:g5:first-look-at-cells:cell}{célula}
        antes de ela disparar: é a decisão do circuito, feita nas
        junções.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[As sinapses da malabarista]\label{ex:g8:nervous-communication:juggler}
A malabarista da \cref{rem:g5:brain-in-command:training}, enfim
explicada: um mês de prática conduziu os mesmos sinais pelas mesmas
junções até essas \omterm{def:g8:nervous-communication:synapse}{sinapses} ficarem mais fortes --- o roteamento que
antes era pesado com esforço agora corre já pré-pesado. A habilidade é
química consolidada nas fendas; ``a prática torna permanente'' é uma
afirmação sobre \omterm{def:g8:nervous-communication:synapse}{sinapses}.
\end{example}

\begin{remark}[As fendas são os pontos vulneráveis]\label{rem:g8:nervous-communication:vulnerable}
Tudo o que age sobre o \omterm{def:g5:brain-in-command:system}{sistema nervoso} a partir de fora age nas
\omterm{def:g8:nervous-communication:synapse}{sinapses} --- o único lugar em que o sinal vira química, passível de
acoplamento e de engano. Os anestésicos as silenciam; os venenos as
travam (alguns paralisam bloqueando o mensageiro da junção com o
\omterm{def:g3:bones-and-muscles:muscle}{músculo}, outros inundando essa junção); e o álcool da
\cref{prop:g5:healthy-choices:substances}, e toda droga que muda o
humor ou a percepção, funciona imitando, bloqueando ou esgotando os
mensageiros das fendas. O
\cref{ch:g8:protecting-the-nervous-system} se apoia exatamente nesse
ponto.
\end{remark}

\section{Fios rápidos, correio lento: dois sistemas comparados}

\begin{proposition}[Nervos contra hormônios]\label{prop:g8:nervous-communication:compare}
O corpo mostra agora dois sistemas de comunicação, construídos para
trabalhos opostos:
\begin{itemize}
  \item o \emph{nervoso}: sinais em milésimos de segundo, por linhas
        dedicadas, para endereços precisos --- um \omterm{def:g3:bones-and-muscles:muscle}{músculo}, uma glândula
        --- e que somem quando o sinal para: é o sistema das pegadas, da
        fala e dos reflexos;
  \item o \emph{hormonal}
        (\cref{def:g8:puberty:hormones}): mensageiros em minutos ou
        meses, transmitidos pelo \omterm{prop:g4:journey-of-food:absorb}{sangue} a todo órgão construído para
        escutar, comandando programas lentos e de corpo inteiro ---
        crescimento, ciclos, a reconstrução da \omterm{def:g8:puberty:puberty}{puberdade}.
\end{itemize}
Telégrafo e serviço postal: o corpo roda os dois, e o
\cref{ch:g8:hormonal-communication} dá ao serviço postal um capítulo
próprio.
\end{proposition}
\admitted

\begin{method}[Diagnosticar uma comunicação, sistema por sistema]\label{met:g8:nervous-communication:diagnose}
Diante de qualquer coordenação do corpo, atribua o sistema dela:
\begin{enumerate}
  \item cronometre: milésimos de segundo --- nervoso; minutos a anos
        --- hormonal (ou os dois, em revezamento);
  \item mapeie o alcance: um endereço preciso --- nervoso; corpo
        inteiro --- hormonal;
  \item ache a rota: um caminho de fio que dá para nomear ---
        nervoso; transmissão pela corrente sanguínea --- hormonal;
  \item confira a persistência: some com o sinal --- nervoso;
        continua enquanto o mensageiro circula --- hormonal.
\end{enumerate}
\end{method}

\begin{example}[O método em três casos]\label{ex:g8:nervous-communication:cases}
O piscar diante de um jato de ar: milésimos de segundo, um anel
muscular, caminho de fio, some na hora --- nervoso. O salto de
crescimento: anos, corpo inteiro, corrente sanguínea, funcionando por
estações --- hormonal. O \omterm{def:g4:heart-and-blood:heart}{coração} disparado da corrida
(\cref{prop:g7:organs-at-work:supply}): os dois em revezamento --- os
sinais nervosos aceleram o ritmo em uma batida, e um mensageiro da
corrente sanguínea (aquela famosa onda quente que se sente num susto)
mantém o alerta do corpo inteiro por minutos. Primeiro o telégrafo,
depois o correio.
\end{example}

\section{Problema: A bancada do reflexo}

\begin{problem}[{Problema de fim de semana --- o reflexo patelar,
instrumentado}]\label{pb:g8:nervous-communication:1}
A demonstração clássica de bancada: a pessoa voluntária sentada, o
martelinho logo abaixo da rótula, o chute involuntário do pé --- agora
instrumentado com cronômetros e com este capítulo.

\medskip\noindent\textbf{Parte I --- O circuito.}
\begin{enumerate}
  \item O martelo estica um tendão; o pé chuta mais ou menos um
        cinquentavo de segundo depois. Monte o circuito: que tipos de
        \omterm{def:g8:nervous-communication:neuron}{neurônio}, passando por onde --- e por que o \omterm{def:g5:brain-in-command:system}{cérebro} não está
        neste circuito.
  \item Um cinquentavo de segundo é mais rápido que o quinto de
        segundo da pegada de bola. Explique a diferença com o
        comprimento do circuito e o número de junções.
  \item A pessoa voluntária não consegue segurar o chute, mesmo
        avisada. O que isso diz sobre onde o reflexo é decidido --- e
        sobre o tráfego de mão única nas \omterm{def:g8:nervous-communication:synapse}{sinapses}?
  \item Logo depois do chute, a pessoa \emph{sente} a batidinha.
        Explique a segunda viagem, mais lenta, desse atraso.
\end{enumerate}

\medskip\noindent\textbf{Parte II --- Variações de bancada.}
\begin{enumerate}[resume]
  \item Bata mais forte: o chute fica mais forte. Que regra de
        codificação da \cref{prop:g8:nervous-communication:code} se
        mostrou?
  \item A médica da turma explica que esse reflexo é testado em toda
        consulta de rotina. O que um reflexo patelar presente e
        simétrico atesta sobre a medula, os \omterm{def:g5:brain-in-command:system}{nervos} e as junções?
  \item Uma pessoa voluntária que cerra a mandíbula chuta com mais
        força --- uma amplificação conhecida. O que isso revela sobre
        as vozes dos \omterm{def:g8:nervous-communication:neuron}{neurônios} de conexão chegando até circuitos
        ``sem \omterm{def:g5:brain-in-command:system}{cérebro}''?
  \item Um anestésico local na \omterm{def:g1:my-body:parts}{perna} elimina o chute e a sensação.
        Diga o ponto de ataque, segundo a
        \cref{rem:g8:nervous-communication:vulnerable}.
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O relatório.}
\begin{enumerate}[resume]
  \item Desenhe o reflexo como o circuito de três passos da
        \cref{prop:g5:brain-in-command:loop}, marcando onde este aqui
        toma um atalho.
  \item Compare o reflexo e a pegada de bola numa tabela de duas
        \omterm{def:g3:bones-and-muscles:skeleton}{colunas}: caminho, junções, tempo, possibilidade de segurar,
        papel da prática.
  \item A última seção do relatório pergunta: ``por que os circuitos
        rápidos de proteção contornam o \omterm{def:g5:brain-in-command:system}{cérebro}?''. Responda com a
        lógica da panela quente
        (\cref{exo:g5:brain-in-command:5}).
  \item Uma frase para encerrar: o que a bancada mostrou, ao mesmo
        tempo, sobre a língua, a fiação e as junções do \omterm{def:g5:brain-in-command:system}{sistema
        nervoso}.
\end{enumerate}
\end{problem}
