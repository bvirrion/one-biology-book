\chapter{फूल से फल तक}\label{ch:g5:flower-to-fruit}

\Cref{ch:g3:flowers-fruits-seeds} ने चेरी के \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} को चेरी बनते देखा था
और एक सवाल खुला छोड़ दिया था: जब \omterm{def:g3:flowers-fruits-seeds:parts}{पराग} का एक कण \omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसर} पर उतरता है
तब \emph{सचमुच होता क्या है}? इस साल, \cref{ch:g4:animal-reproduction}
के \omterm{def:g5:flower-to-fruit:steps}{निषेचन} और \cref{ch:g5:first-look-at-cells} की \omterm{def:g5:first-look-at-cells:cell}{कोशिकाओं} से लैस होकर,
हम \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} की मशीनरी ठीक से खोल सकते हैं।

\section{फूलों में नर और मादा, दोनों भाग होते हैं}

\begin{proposition}[एक ही फूल में दो भूमिकाएँ]\label{prop:g5:flower-to-fruit:roles}
\omterm{def:g3:flowers-fruits-seeds:parts}{फूल} वाले \omterm{def:g1:plants-around-us:plant}{पौधों} का जनन ठीक
\cref{def:g4:animal-reproduction:sexual} के अर्थ में \emph{लैंगिक}
जनन है --- और ज़्यादातर \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} दोनों भूमिकाएँ एक साथ निभाते हैं:
\begin{itemize}
  \item \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{पुंकेसर}} \omterm{def:g4:animal-reproduction:sexual}{नर} भाग हैं: \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{पराग}} का हर कण \omterm{def:g1:plants-around-us:plant}{पौधे} की \omterm{def:g4:animal-reproduction:sexual}{नर}
        जननिक \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} लिए रहता है;
  \item \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसर}} \omterm{def:g4:animal-reproduction:sexual}{मादा} भाग है: उसके भीतर गहराई में
        \emph{बीजांड}\index{बीजांड} पड़े रहते हैं, और हर \omterm{prop:g5:flower-to-fruit:roles}{बीजांड} में
        एक \omterm{def:g4:animal-reproduction:sexual}{अंडाणु} होता है।
\end{itemize}
\cref{def:g3:flowers-fruits-seeds:parts} के ``भावी \omterm{def:g2:from-seed-to-plant:seed}{बीजों}'' को अब उनका
असली नाम मिल गया: \omterm{prop:g5:flower-to-fruit:roles}{बीजांड}, जो निषेचित होने के इंतज़ार में हैं।
\end{proposition}
\admitted

\begin{remark}[जंतुओं जैसी ही बनावट]\label{rem:g5:flower-to-fruit:sameplan}
\omterm{def:g4:animal-reproduction:sexual}{नर} \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ}, \omterm{def:g4:animal-reproduction:sexual}{अंडाणु}, \omterm{def:g5:flower-to-fruit:steps}{निषेचन}: \cref{ch:g4:animal-reproduction} की
शब्दावली शब्द दर शब्द यहाँ उतर आती है। यह फिर \omterm{prop:g5:first-look-at-cells:principle}{जीवन की एकता} है
(\cref{prop:g5:first-look-at-cells:principle}) --- गुलबहार और
डॉल्फ़िन एक ही अंतर्निहित बनावट पर जनन करते हैं, चाहे उनकी मशीनरी
कितनी ही अलग दिखे।
\end{remark}

\section{परागण, फिर निषेचन}

\begin{definition}[दोनों क़दम]\label{def:g5:flower-to-fruit:steps}
\omterm{def:g2:from-seed-to-plant:seed}{बीज} बनने में दो अलग-अलग क़दम लगते हैं:
\begin{enumerate}
  \item \emph{परागण}\index{परागण}: \omterm{def:g3:flowers-fruits-seeds:parts}{पराग} का एक कण किसी \omterm{def:g3:flowers-fruits-seeds:parts}{पुंकेसर} से
        किसी \omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसर} की चोटी तक सफ़र करता है --- किसी \omterm{prop:g3:sorting-living-things:groups}{कीट} के साथ
        (\cref{prop:g3:flowers-fruits-seeds:transform}) या हवा के
        साथ;
  \item \emph{निषेचन}\index{निषेचन (पौधों में)}: उतरा हुआ कण
        \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसर} के भीतर नीचे तक} एक महीन नली उगाता है; उसी
        से होकर \omterm{def:g4:animal-reproduction:sexual}{नर} \omterm{def:g5:first-look-at-cells:cell}{कोशिका} किसी \omterm{prop:g5:flower-to-fruit:roles}{बीजांड} तक पहुँचती है और उसके \omterm{def:g4:animal-reproduction:sexual}{अंडाणु} से
        जुड़ जाती है।
\end{enumerate}
परागण यात्रा है; निषेचन मुलाक़ात।
\end{definition}

ख़ाली शाखा काफ़ी दिखती है।
\end{example}

\section{मुलाक़ात के बाद: बीज और फल}

\begin{proposition}[हर भाग क्या बनता है]\label{prop:g5:flower-to-fruit:becomes}
\omterm{def:g5:flower-to-fruit:steps}{निषेचन} के बाद \cref{prop:g3:flowers-fruits-seeds:transform} वाला बदलाव
नई सटीकता के साथ चलता है:
\begin{itemize}
  \item हर \emph{निषेचित \omterm{prop:g5:flower-to-fruit:roles}{बीजांड}} एक \emph{\omterm{def:g2:from-seed-to-plant:seed}{बीज}} बन जाता है --- उसके
        भीतर का नन्हा \omterm{def:g1:plants-around-us:plant}{पौधा}
        (\cref{def:g2:from-seed-to-plant:seed}) निषेचित \omterm{def:g4:animal-reproduction:sexual}{अंडाणु} से
        उगता है, ठीक वैसे ही जैसे कोई \omterm{def:g1:animals-around-us:animal}{जंतु} अपने \omterm{def:g4:animal-reproduction:sexual}{अंडाणु} से उगता है
        (\cref{ex:g5:first-look-at-cells:one});
  \item उनके चारों ओर का \emph{\omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसर}} फूलकर \emph{\omterm{prop:g3:flowers-fruits-seeds:transform}{फल}} बन जाता
        है;
  \item \omterm{def:g3:flowers-fruits-seeds:parts}{पंखुड़ियाँ} और \omterm{def:g3:flowers-fruits-seeds:parts}{पुंकेसर}, अपना काम पूरा करके, मुरझाकर गिर जाते
        हैं।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[दाने गिनना]\label{ex:g5:flower-to-fruit:pips}
सेब को आड़ा काटो: पाँच छोटे कक्ष, और हर एक में कुछ दाने। हर दाना एक
ऐसा \omterm{prop:g5:flower-to-fruit:roles}{बीजांड} है जिस तक कोई एक पराग-नली पहुँची थी। जिस सेब का एक पहलू
दबा हो और उस तरफ़ के कक्ष ख़ाली हों, वह ख़राब \omterm{def:g5:flower-to-fruit:steps}{परागण} वाले \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} की कहानी
कहता है --- सारे \omterm{prop:g5:flower-to-fruit:roles}{बीजांडों} को निषेचित करने के लिए बहुत कम कण उतरे थे,
और \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} असमान फूला।
\end{example}

\begin{example}[हवा से उड़ता पराग]\label{ex:g5:flower-to-fruit:wind}
घासों, गेहूँ और हेज़ल के \omterm{def:g1:plants-around-us:tree}{पेड़ों} की न \omterm{def:g3:flowers-fruits-seeds:parts}{पंखुड़ियाँ} होती हैं, न महक, न रस
--- आने वालों को देने को कुछ भी नहीं। उनका \omterm{def:g3:flowers-fruits-seeds:parts}{पराग} \emph{हवा} पर सवार
होता है: बहुत बड़ी मात्रा में बना, धूल जितना महीन, और हवा में उछाल
दिया गया, कि जहाँ क़िस्मत ले जाए वहाँ उतर जाए। यह रणनीति सटीकता के
बदले मात्रा लेती है ---
\cref{rem:g4:plants-without-seeds:sporeseed} वाला बीजाणु-बादल का तर्क,
\omterm{def:g3:flowers-fruits-seeds:parts}{पराग} के साथ खेला हुआ। (चलते-चलते: जून में हवा में उड़ता यही \omterm{def:g3:flowers-fruits-seeds:parts}{पराग} नाकों
को छींक-बुख़ार में डालता है।)
\end{example}

\begin{method}[बिना फल वाले पौधे की जाँच]\label{met:g5:flower-to-fruit:diagnose}
\omterm{def:g1:plants-around-us:plant}{पौधे} में \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} तो ख़ूब आए पर \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} नहीं लगा। क्रम से पूछो:
\begin{enumerate}
  \item क्या \omterm{def:g5:flower-to-fruit:steps}{परागण} करने वाले उस तक पहुँचे? (घर के भीतर, जाली में,
        आस-पास कोई \omterm{prop:g3:sorting-living-things:groups}{कीट} नहीं, या फूलने के हफ़्ते ठंडे और बरसाती?)
  \item अगर नहीं, तो हाथ से \omterm{def:g5:flower-to-fruit:steps}{परागण} करो: मुलायम कूची को एक \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} के
        \omterm{def:g3:flowers-fruits-seeds:parts}{पुंकेसरों} पर लगाओ, फिर दूसरे \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} के \omterm{def:g3:flowers-fruits-seeds:parts}{स्त्रीकेसर} पर;
  \item क्या \omterm{prop:g5:flower-to-fruit:roles}{बीजांडों} के निषेचित होने से पहले ही पाले या किसी नुक़सान
        ने \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} मार दिए थे
        (\cref{exo:g3:flowers-fruits-seeds:9})?
  \item अगर अब \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} लग जाता है, तो \omterm{def:g1:plants-around-us:plant}{पौधे} की मशीनरी दुरुस्त थी --- सिर्फ़
        पहले क़दम की यात्रा नाकाम हुई थी।
\end{enumerate}
\end{method}

\begin{exercise}[$\star\star$]\label{exo:g5:flower-to-fruit:10}
चेरी उगाने वाले का वसंत ठंडा और बरसाती रहा: दो हफ़्ते मधुमक्खियाँ
मुश्किल से उड़ीं, हालाँकि पाला नहीं पड़ा और \omterm{def:g1:plants-around-us:tree}{पेड़ों} पर शानदार \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} आए।
जुलाई की फ़सल की भविष्यवाणी करो और उसे
\cref{def:g5:flower-to-fruit:steps} से समझाओ।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:flower-to-fruit:11}
हरितगृह में टमाटर उगाने वाला भीतर ही भँवरों का छत्ता लगवा देता है,
हालाँकि सीलबंद हरितगृह से बाहर के किसी \omterm{def:g3:flowers-fruits-seeds:parts}{फूल} तक पहुँचा नहीं जा सकता। इस
अध्याय से ठीक-ठीक समझाओ कि भँवरे कौन-सी सेवा बेचते हैं --- और टमाटरों
के लिए (जिनका \omterm{def:g3:flowers-fruits-seeds:parts}{पराग} हिलाने भर से झड़ जाता है) \omterm{def:g1:plants-around-us:plant}{पौधों} पर हवा फेंकते पंखे
उनकी जगह कुछ हद तक क्यों ले सकते हैं, जबकि चेरी के \omterm{def:g1:plants-around-us:tree}{पेड़ों} के लिए नहीं
ले सकते।
\end{exercise}
