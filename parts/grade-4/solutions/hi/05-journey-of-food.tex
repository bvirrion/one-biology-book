\section*{अध्याय \ref{ch:g4:journey-of-food} --- भोजन की यात्रा}

\begin{solution}{exo:g4:journey-of-food:1}
मुँह, \omterm{def:g4:journey-of-food:tract}{ग्रासनली}, \omterm{def:g4:journey-of-food:tract}{आमाशय}, \omterm{def:g4:journey-of-food:tract}{छोटी आँत}, \omterm{def:g4:journey-of-food:tract}{बड़ी आँत} --- और फिर बचा-खुचा शरीर से
बाहर चला जाता है।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:2}
वह \omterm{def:g4:journey-of-food:tract}{आमाशय} की पेशीदार दीवारों से मथा जाता है और पाचक रसों में मिलाया
जाता है --- कुछ घंटों तक।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:3}
\omterm{def:g4:journey-of-food:digestion}{पाचन} भोजन का, नली के साथ-साथ, इतने सरल और छोटे घटकों में बदल जाना है कि
वे शरीर में उतर सकें। वही घटक \omterm{def:g4:journey-of-food:digestion}{पोषक तत्व} हैं।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:4}
\omterm{def:g4:journey-of-food:tract}{छोटी आँत} में, उसकी पतली तह वाली दीवारों से होकर, \omterm{prop:g4:journey-of-food:absorb}{रक्त} में। जो नहीं उतर
सकता वह आगे \omterm{def:g4:journey-of-food:tract}{बड़ी आँत} तक जाता है, वहाँ अपना पानी गँवाता है, और मल के
रूप में शरीर से बाहर निकल जाता है।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:5}
\omterm{def:g4:journey-of-food:tract}{लार} पहले से ही रोटी को सरल, मीठे घटकों में खोल रही होती है --- \omterm{def:g4:journey-of-food:digestion}{पाचन}
मुँह में ही शुरू हो जाता है, और वह मिठास उसी का स्वाद है।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:6}
क्योंकि \omterm{prop:g4:journey-of-food:absorb}{रक्त} में उतरने के लिए दीवार का क्षेत्रफल चाहिए: कई मीटर की
लंबाई और भीतर की गहरी तहें एक विशाल उतरने वाली सतह --- सपाट फैलाओ तो
मोटे तौर पर एक छोटे कमरे का क़ालीन --- पेट में समेट देती हैं।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:7}
जैसे: ठीक से चबाओ (\omterm{def:g4:journey-of-food:tract}{आमाशय} के दाँत नहीं होते); शांति से, तय समय पर खाओ
(नली बिना जल्दबाज़ी के सबसे अच्छा काम करती है); पानी पियो (\omterm{def:g4:journey-of-food:digestion}{पाचन} गीले
में चलता है); रेशा खाओ (वह बचे-खुचे को चलता रखता है); भारी खाने के बाद
आराम करो (\omterm{def:g4:journey-of-food:tract}{आमाशय} और \omterm{def:g1:my-body:parts}{टाँगें} शरीर की मेहनत के लिए होड़ करते हैं)।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:8}
उत्तर खुला है। आम तौर पर मुँह में आधे घंटे से भी कम --- और बाक़ी यात्रा
में एक से दो दिन। मुँह का हिस्सा समय में बहुत छोटा और असर में निर्णायक
है: दाँतों वाला पड़ाव वही अकेला है।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:9}
टूटीं: ठीक से चबाना (खाना आधा चबाकर निगला गया) और खाने के बाद आराम
करना (ठीक बाद ही दौड़)। कल: धीरे-धीरे खाओ, और दौड़ने से पहले खाने को
बैठ जाने दो।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:10}
निगला हुआ भोजन अब भी नली के भीतर ही है --- डोनट के छेद की तरह, अभी
सचमुच शरीर में नहीं। पहले उसे पचकर \omterm{def:g4:journey-of-food:digestion}{पोषक तत्व} बनना होगा; और वे \omterm{def:g4:journey-of-food:digestion}{पोषक
तत्व} सिर्फ़ \omterm{def:g4:journey-of-food:tract}{छोटी आँत} में ही दीवार पार करके \omterm{prop:g4:journey-of-food:absorb}{रक्त} में उतरते हैं। मुँह में
पहुँचना शरीर में पहुँचना नहीं है।
\end{solution}

\begin{solution}{exo:g4:journey-of-food:11}
\omterm{def:g4:journey-of-food:digestion}{पाचन} को भोजन खोलकर \omterm{def:g4:journey-of-food:digestion}{पोषक तत्व} बनाने होते हैं, और घास रोटी से कहीं ज़्यादा
मुश्किल से खुलती है --- सख़्त रेशे, और हर कौर में थोड़ा-सा ही सार
(\cref{prop:g2:what-animals-eat:daily})। इसलिए घास खाने वाली भारी
मशीनरी लाती है: दो बार इस्तेमाल होते पीसने वाले \omterm{def:g2:teeth-and-eating-well:kinds}{चर्वणक} (जुगाली), लंबी
प्रक्रिया के लिए \omterm{def:g4:journey-of-food:tract}{आमाशय} के चार कक्ष, और आँत में ऐसे मददगार जो वह \omterm{def:g1:animals-around-us:covering}{खोल}
सकें जो उसके अपने रस नहीं \omterm{def:g1:animals-around-us:covering}{खोल} पाते। सख़्त कच्चा माल, बड़ा कारख़ाना।
\end{solution}
