\section*{Chapter \ref{ch:g4:body-systems} --- Human Body Systems}

\begin{solution}{exo:g4:body-systems:1}
An \omterm{def:g4:body-systems:organs}{organ system} is a group of \omterm{def:g4:body-systems:organs}{organs} that work together to carry out a
major job in the \omterm{def:g1:my-body:body}{body}.
\end{solution}

\begin{solution}{exo:g4:body-systems:2}
Any three: \omterm{def:g3:teeth:path}{mouth}, oesophagus, \omterm{def:g3:teeth:path}{stomach}, small \omterm{def:g3:teeth:path}{intestine}, large
\omterm{def:g3:teeth:path}{intestine}, liver, etc.
\end{solution}

\begin{solution}{exo:g4:body-systems:3}
The \omterm{def:g4:body-systems:circ}{heart} pumps blood around the \omterm{def:g1:my-body:body}{body}.
\end{solution}

\begin{solution}{exo:g4:body-systems:4}
The \omterm{def:g4:body-systems:resp}{respiratory system} (lungs and airways).
\end{solution}

\begin{solution}{exo:g4:body-systems:5}
\omterm{def:g3:teeth:digestion}{Digestion} breaks \omterm{def:g2:food-energy:food}{food} down and absorbs \omterm{def:g4:plants-need:needs}{nutrients}; blood in the
\omterm{def:g4:body-systems:circ}{circulatory system} carries those \omterm{def:g4:plants-need:needs}{nutrients} to \omterm{def:g1:my-body:body}{body} \omterm{def:g4:cells:cell}{cells}.
\end{solution}
