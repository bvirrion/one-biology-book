ونزف كل ثمانية وعشرين يومًا أو نحوها. ومختبر يقيس أربعة \omterm{def:g11:hormones-and-reproduction:hormone}{هرمونات} في

مختلفة؛ أما نظام الضبط --- مركز في الدماغ، وغدة تحته، والقند، ورسائل
يحملها الدم في الاتجاهين --- فهو هو. ويقرأ هذا الفصل الجداول، ويبين
كيف أتاح فهمها منع حمل، أو الحصول عليه.

\section{الهرمونات وحلقاتها}

\begin{definition}[الهرمون والتغذية الراجعة]\label{def:g11:hormones-and-reproduction:hormone}
\emph{الهرمون}\index{هرمون} جزيء تفرزه غدة في الدم ويؤثر، عن بعد، في
\omterm{def:g10:cells-common-unit:cell}{الخلايا} التي تحمل \emph{مستقبلًا} له. والنظام الهرموني تنظمه
\emph{التغذية الراجعة}\index{تغذية راجعة}: إذ يؤثر الهرمون المنتَج في
آخر السلسلة رجوعًا في الغدد التي في أولها. و\emph{التغذية الراجعة
السالبة} --- إذ يخفض الناتجُ الأمرَ --- تبقي مستوى ثابتًا؛
و\emph{التغذية الراجعة الموجبة} --- إذ يزيد الناتج الأمر --- تنتج
اندفاعة.
\end{definition}

\begin{proposition}[السلسلة المشتركة]\label{prop:g11:hormones-and-reproduction:chain}
تضبط القندين في الجنسين السلسلة الثلاثية الطبقات نفسها. فمركز في
الدماغ، هو \emph{الوطاء}، يفرز على نبضات هرمونًا محرِّرًا، هو GnRH، في
الأوعية الصغيرة المؤدية إلى \emph{الغدة النخامية} تحته مباشرة؛ فتستجيب
النخامية بإفراز هرمونين في الدم العام، هما \emph{LH} و\emph{FSH}؛
وهذان يؤثران في القندين، وهما ينتجان الأمشاج \omterm{def:g11:hormones-and-reproduction:hormone}{والهرمونات} الجنسية معًا
--- التستوستيرون في الخصية، والإستروجينات والبروجسترون في المبيض.
\omterm{def:g11:hormones-and-reproduction:hormone}{والهرمونات} الجنسية تؤثر في الجسم، وتؤثر رجوعًا في الوطاء والنخامية.
\end{proposition}
\admitted

\section{الذكر: مستوى ثابت}

\begin{proposition}[ضبط الخصية]\label{prop:g11:hormones-and-reproduction:male}
للخصية حجرتان. \emph{فالأنيبيبات المنوية}، ينبهها FSH والتستوستيرون،
تنتج النطاف باستمرار من \omterm{prop:g11:becoming-male-female:puberty}{البلوغ}، نحو مئة مليون في اليوم. أما \omterm{def:g10:cells-common-unit:cell}{الخلايا}
الواقعة بين الأنيبيبات، وينبهها LH، فتفرز \emph{التستوستيرون}، وهو
يحفظ السبيل التناسلي والصفات الثانوية وإنتاج النطاف. والتستوستيرون
يثبّط الوطاء والنخامية: فحين يرتفع، يهبط GnRH وLH وFSH وتفرز الخصية
أقل؛ وحين يهبط، ترتفع. وهذه \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة} السالبة تبقي المستوى شبه
ثابت، ضمن تقلبات يومية صغيرة، من \omterm{prop:g11:becoming-male-female:puberty}{البلوغ} إلى الشيخوخة.
\end{proposition}
\begin{proof}[شواهد]
خصاء حيوان يرفع LH عنده عدة أضعاف في غضون أيام؛ وحقنة من التستوستيرون
تعيد LH إلى الهبوط. وآفة في الوطاء تلغي LH وFSH والتستوستيرون معًا؛
وحقن GnRH على نبضات يعيد الثلاثة، أما التسريب المستمر \omterm{def:g11:hormones-and-reproduction:hormone}{لهرمون} GnRH فلا
يعيدها --- فالنبض مهم. والرجال الذين يعالجون بالتستوستيرون لأسباب أخرى
يرون إنتاجهم من النطاف يهبط: \omterm{def:g11:hormones-and-reproduction:hormone}{فالهرمون} الخارجي يطفئ نخاميتهم.
\end{proof}

\section{الأنثى: دورات}

\begin{proposition}[الدورة المبيضية]\label{prop:g11:hormones-and-reproduction:ovary}
من \omterm{prop:g11:becoming-male-female:puberty}{البلوغ} إلى سن اليأس يجري المبيض دورة نحو 28 يومًا.
\begin{itemize}
  \item \emph{الطور الجريبي} (الأيام 1--13): تحت FSH، ينمو جريب ---
        وهو بييضة تحيط بها خلاياها المرضعة --- ويفرز مقادير متزايدة من
        \emph{الإستروجينات}.
  \item \emph{الإباضة} (اليوم 14): اندفاعة من LH تمزق الجريب الناضج،
        فيطلق بييضته في قناة البيض.
  \item \emph{الطور الأصفري} (الأيام 15--28): يصير الجريب الفارغ
        \emph{الجسم الأصفر}، وهو يفرز \emph{البروجسترون}
        والإستروجينات؛ فإذا لم يبدأ حمل تنكس بعد نحو اثني عشر يومًا،
        وهبط الهرمونان، وبدأت دورة جديدة.
\end{itemize}
\end{proposition}
\admitted

\begin{proposition}[الدورة الرحمية والتغذيات الراجعة]\label{prop:g11:hormones-and-reproduction:uterus}
تتبع بطانة الرحم \omterm{def:g11:hormones-and-reproduction:hormone}{هرمونات} المبيض: فالإستروجينات تجعلها تغلظ أثناء الطور
الجريبي؛ والبروجسترون يجعلها مفرزة ومهيأة لاستقبال جنين أثناء الطور
الأصفري؛ وهبوطهما معًا في نهاية الدورة يجعلها تتفكك وتنزف --- وهو
\emph{الطمث}، من اليوم 1 إلى 5 من الدورة التالية. والتغذيات الراجعة
تغير إشارتها أثناء الدورة: فعند المستويات المنخفضة والمعتدلة، تثبّط
الإستروجينات والبروجسترون الوطاء والنخامية (\omterm{def:g11:hormones-and-reproduction:hormone}{تغذية راجعة} سالبة)؛ أما
الإستروجين المرتفع للجريب الناضج، إذا دام يومين، \emph{فينبّههما}
(\omterm{def:g11:hormones-and-reproduction:hormone}{تغذية راجعة} موجبة)، وهذا ينتج اندفاعة LH التي تطلق الإباضة. وبعد
الإباضة، يعيد بروجسترون الجسم الأصفر \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة} السالبة، فلا تقع
اندفاعة ثانية.
\end{proposition}
\begin{proof}[شواهد]
تعطي القياسات الدموية اليومية منحنيات الشكل التالي: فاندفاعة LH تتلو
ذروة الإستروجين بيوم وتسبق الإباضة بيوم؛ والبروجسترون لا يظهر إلا بعد
الإباضة. واستئصال المبيضين يرفع LH وFSH؛ وجرعات صغيرة من الإستروجينات
تخفضهما؛ وجرعة كبيرة مستمرة تعطى في منتصف الدورة تثير اندفاعة LH ---
وهي علامة انقلاب إشارة \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة} مع الجرعة. والمرأة التي لا
تستطيع نخاميتها الاستجابة لارتفاع الإستروجين لا تحدث عندها اندفاعة ولا
تبيض، وإن نمت جريباتها.
\end{proof}

\begin{method}[قراءة جدول دورة]\label{met:g11:hormones-and-reproduction:read}
\begin{enumerate}
  \item ابحث عن ذروة LH: فالإباضة في اليوم التالي؛ والطور الجريبي كل
        ما قبلها، والطور الأصفري كل ما بعدها.
  \item تحقق من الترتيب: ذروة إستروجين، ثم اندفاعة LH، ثم إباضة، ثم
        بروجسترون. والمنحني الخارج عن الترتيب علامة على اضطراب.
  \item عُدّ: الطور الأصفري قريب من 14 يومًا عند كل امرأة؛ أما طول
        الدورة فيتفاوت في الطور الجريبي. والإباضة إذن قبل الحيض التالي
        بأربعة عشر يومًا، لا بعد السابق بأربعة عشر.
  \item فسّر دواءً أو آفةً بالسؤال أي \omterm{def:g11:hormones-and-reproduction:hormone}{هرمون} يضيفه أو يزيله، وما الذي
        تفعله \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة} عندئذ بالبقية.
\end{enumerate}
\end{method}

\section{الاختيار: منع الحمل}

\begin{proposition}[منع الحمل الهرموني]\label{prop:g11:hormones-and-reproduction:contraception}
تقدم الحبة المركبة، كل يوم، جرعات صغيرة من إستروجين ومن جزيء شبيه
بالبروجسترون. وعند تلك المستويات تكون \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة} سالبة طوال
الوقت: فيبقى FSH منخفضًا، ولا ينضج جريب، ولا تقع ذروة إستروجين، ولا
اندفاعة LH، ولا إباضة. وبطانة الرحم، وقد أبقيت رقيقة، ومخاط عنق الرحم،
وقد أبقي غليظًا، يضيفان حاجزين. وإيقاف الحبوب أسبوعًا يدع \omterm{def:g11:hormones-and-reproduction:hormone}{الهرمونات}
تهبط والبطانة تنزف. وطرق أخرى تعمل في مواضع أخرى: فجهاز نحاسي في الرحم
يمنع الإخصاب والانغراس؛ وحبة الطوارئ، إذا أخذت في غضون أيام من الجماع،
تؤخر الإباضة؛ والعوازل تمنع الأمشاج من اللقاء --- وهي الطريقة الوحيدة
التي تمنع كذلك العدوى المنقولة جنسيًا.
\end{proposition}
\admitted

\begin{example}[قراءة أثر الحبة]\label{ex:g11:hormones-and-reproduction:pill}
على جدول امرأة تأخذ الحبة، تكون المنحنيات الأربعة مستوية:
الإستروجينات والبروجسترون عند المستوى المنخفض الثابت الذي تقدمه
الحبوب، وLH وFSH دون حدهما الأدنى في الدورة، ولا ذروة في أي مكان.
وأخطئ الحبوب بضعة أيام في أول العلبة فيفلت FSH: فقد يبدأ جريب، وقد
يتبع الباقي. فالحبة تنجح لأن \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة} تعمل، والحبة المنسية ثقب
في \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة}.
\end{example}

\section{الحصول: الإنجاب المساعَد}

\begin{proposition}[حين لا يأتي الحمل]\label{prop:g11:hormones-and-reproduction:art}
نحو زوجين من كل سبعة يستشيران بعد سنة من المحاولة. والأسباب مشتركة:
إباضة غائبة أو غير منتظمة، وقناتا بيض مسدودتان، ونطاف قليلة جدًا أو
غير متحركة، وفي ثلث الحالات لا سبب معيَّن. والعلاجات تتبع البيولوجيا:
\begin{itemize}
  \item \emph{استحثاث الإباضة}: حقن FSH، أو دواء يحصر \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة}
        السالبة للإستروجين، يجعل الجريبات تنضج؛ ثم تطلق حقنة شبيهة
        \omterm{def:g11:hormones-and-reproduction:hormone}{بهرمون} LH الإباضةَ في ساعة مختارة؛
  \item \emph{التلقيح}: نطاف محضَّرة توضع في الرحم عند الإباضة؛
  \item \emph{الإخصاب في المختبر}: بعد التنبيه، تجمع البييضات من
        المبيض بإبرة، وتخصَّب في طبق --- بحقن نطفة واحدة في كل بييضة
        إن لزم --- ويوضع جنين أو جنينان في الرحم بعد ثلاثة إلى خمسة
        أيام. ونحو محاولة من كل ثلاث أو أربع تفضي إلى ولادة؛ والباقي
        يمكن إعادته بأجنة مجمدة.
\end{itemize}
وكل خطوة من السلسلة الطبيعية تخفق يحل محلها التدخل الموافق.
\end{proposition}
\admitted

\begin{example}[أرقام محاولة واحدة]\label{ex:g11:hormones-and-reproduction:ivf}
يعطي التنبيه 10 بييضات؛ منها 8 ناضجة؛ و5 تخصَّب؛ و3 تبلغ طور اليوم
الخامس؛ فينقل 1 ويجمَّد 2. وينجح النقل في نحو ثلث الحالات؛ ومع
الجنينين المجمدين ترتفع فرصة الولادة من الجمع الواحد إلى نحو 60\%. وقد
أنتجت التقنية عدة ملايين من الأطفال منذ 1978؛ وأثارت كذلك أسئلة ---
عن الأجنة غير المستعملة، وعن عمر الوالدين، وعما يجوز اختياره --- لا
تجيب عنها البيولوجيا.
\end{example}

\begin{remark}[نظامان يلتقيان]\label{rem:g11:hormones-and-reproduction:brain}
\omterm{def:g11:hormones-and-reproduction:hormone}{هرمونات} هذا الفصل تضبط القندين والجسم؛ وهي لا تضبط، عند الإنسان،
السلوك الجنسي كما تفعل عند كثير من الحيوانات، التي يرتبط تزاوجها
بالدورة المبيضية. فالنشاط الجنسي البشري مستقل في معظمه عن الدورة
ويتوقف على أنظمة الرغبة واللذة والتعلق في الدماغ --- وهي دارات مكافأة
مشتركة مع دوافع أخرى --- وعلى تاريخ الشخص وثقافته وخياراته. فالتكاثر
والجنسانية يتداخلان؛ وهما ليسا الشيء نفسه، \omterm{def:g11:hormones-and-reproduction:hormone}{والهرمونات} لا تحكم إلا
الأول.
\end{remark}

\section{مسألة: الجدول}

\begin{problem}[{مسألة نهاية الأسبوع --- دورة امرأة واحدة تُجدوَل
هرمونًا هرمونًا، وتقرأ بتغذياتها الراجعة، وتضطرب بحبة، وتساعَد حين
تخفق}]
\label{pb:g11:hormones-and-reproduction:1}
تؤخذ عينات دم يوميًا عبر دورة واحدة. والقيم مقرَّبة، بالنسب المئوية من
أعظمي كل \omterm{def:g11:hormones-and-reproduction:hormone}{هرمون}:

\begin{center}
\begin{tabular}{lccccccccc}
اليوم & 2 & 6 & 10 & 12 & 13 & 14 & 18 & 22 & 27 \\ \hline
إستروجينات & 10 & 20 & 55 & 100 & 80 & 45 & 45 & 60 & 15 \\
LH & 15 & 15 & 20 & 35 & 100 & 60 & 15 & 15 & 12 \\
FSH & 45 & 35 & 30 & 35 & 70 & 45 & 22 & 25 & 25 \\
بروجسترون & 3 & 3 & 3 & 3 & 4 & 5 & 55 & 100 & 15 \\
\end{tabular}
\end{center}

\medskip
\noindent\textbf{الجزء الأول --- قراءة الجدول.}
\begin{enumerate}
  \item في أي يوم تقع الإباضة؟ علّل من هرمونين.
  \item أعطِ الطورين الجريبي والأصفري، بطوليهما.
  \item أي بنية من المبيض تفرز إستروجينات اليوم 10؟ وإستروجينات اليوم
        22؟ وأيها تفرز بروجسترون اليوم 22؟
  \item صف حال بطانة الرحم في الأيام 6 و22 و28.
  \item لماذا يكون FSH أعلى ما يكون في أول الدورة وأدنى ما يكون في
        الطور الأصفري؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- التغذيات الراجعة.}
\begin{enumerate}[resume]
  \item بين اليومين 2 و10، ترتفع الإستروجينات ويهبط FSH. فأي \omterm{def:g11:hormones-and-reproduction:hormone}{تغذية
        راجعة} تعمل؟ فسّر.
  \item بين اليومين 12 و13، تكون الإستروجينات في ذروتها ويرتفع LH سبعة
        أضعاف. فأي \omterm{def:g11:hormones-and-reproduction:hormone}{تغذية راجعة}؟ وأي شرط على مستوى الإستروجين يجعل ذلك
        ممكنًا؟
  \item بين اليومين 18 و27، تكون الإستروجينات معتدلة والبروجسترون
        مرتفعًا وLH منخفضًا. فأي \omterm{def:g11:hormones-and-reproduction:hormone}{تغذية راجعة}، ولماذا لا تقع اندفاعة
        ثانية؟
  \item ينكس الجسم الأصفر في اليوم 26. تنبأ \omterm{def:g11:hormones-and-reproduction:hormone}{بهرمونات} اليوم 28 وبما
        يتلوها.
  \item فسّر لماذا كانت الدورة ذاتية الاستئناف: أي ارتفاع في اليوم 1
        يطلق الجريب التالي؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الحبة.}
تأخذ المرأة نفسها الحبة المركبة شهرًا، وهي تقدم إستروجينات عند 25\%
وهرمونًا شبيهًا بالبروجسترون عند 40\% من أعظمييهما، كل يوم.
\begin{enumerate}[resume]
  \item تنبأ بمستويي FSH وLH طوال الشهر، وبمصير الجريبات.
  \item هل تقع ذروة إستروجين؟ واندفاعة LH؟ وإباضة؟
  \item خلال الأيام السبعة الخالية من الحبوب، ماذا يحدث لبطانة الرحم،
        ولماذا لا يكون ذلك طمثًا حقيقيًا؟
  \item تنسى الحبوب في الأيام 5 إلى 8. فسّر الخطر، خطوة خطوة.
  \item فسّر لماذا تستطيع حبة بروجسترون وحده، عند مستوى يحفظ \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية
        الراجعة} السالبة، أن تمنع الإباضة كذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- حين تخفق السلسلة.}
\begin{enumerate}[resume]
  \item جدول امرأة أخرى يبين FSH وLH دائمًا دون 10\% والإستروجينات دون
        10\%. فأين الخلل؟ وأي علاج يعيد الإباضة، وبأي طريق؟
  \item جدول امرأة ثالثة طبيعي لكن قناتي بيضها مسدودتان. فأي خطوات
        \cref{prop:g11:hormones-and-reproduction:art} تحل محل الخطوة
        المسدودة؟
  \item في الإخصاب المختبري تعطى حقنة واحدة من \omterm{def:g11:hormones-and-reproduction:hormone}{هرمون} شبيه \omterm{def:g11:hormones-and-reproduction:hormone}{بهرمون} LH قبل 36
        ساعة من جمع البييضات. فسّر ما الذي تحاكيه ولماذا كان التوقيت
        دقيقًا.
  \item بعد التنبيه ينضج 10 جريبات بدل 1. فسّر، بدلالة FSH، لماذا لا تنضج
        الدورة الطبيعية إلا واحدًا عادةً.
  \item اذكر النتيجة: يوم الإباضة على الجدول، وعلامتا \omterm{def:g11:hormones-and-reproduction:hormone}{التغذية الراجعة}
        اللتان أنتجتاه، والآلية التي كبحته بها الحبة.
\end{enumerate}
\end{problem}
