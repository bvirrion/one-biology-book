\chapter{Photosynthesis and Respiration}\label{ch:g7:photo-resp}

\omterm{def:g1:plants-animals:plant}{Plants} make \omterm{def:g2:food-energy:food}{food} from \omterm{def:g4:plants-need:needs}{light}; almost all \omterm{def:g1:living-nonliving:living}{living things} release \omterm{def:g6:nutrition:def}{energy} from
\omterm{def:g2:food-energy:food}{food} by respiration. The two processes are not opposites in every detail,
but they link in a global cycle of carbon and oxygen.

\section{Photosynthesis}

\begin{definition}[Photosynthesis]\label{def:g7:photo-resp:photo}
\emph{Photosynthesis}\index{photosynthesis} is the process by which green
\omterm{def:g1:plants-animals:plant}{plants} (and some other \omterm{def:g4:ecosystems:levels}{organisms}) use \omterm{def:g4:plants-need:needs}{light} \omterm{def:g6:nutrition:def}{energy} to build sugars from
carbon dioxide and \omterm{def:g4:plants-need:needs}{water}, releasing oxygen. \omterm{def:g7:microscope:parts}{Chloroplasts} contain the
green pigment \emph{chlorophyll}\index{chlorophyll}.
\end{definition}

\begin{proposition}[Word equation]\label{prop:g7:photo-resp:photo-eq}
In outline:
\[
\text{carbon dioxide} + \text{water}
\xrightarrow{\text{light, chlorophyll}}
\text{glucose} + \text{oxygen}.
\]
\omterm{def:g4:plants-need:needs}{Light} \omterm{def:g6:nutrition:def}{energy} is stored in chemical form in the sugar.
\end{proposition}
\admitted

\begin{definition}[Requirements]\label{def:g7:photo-resp:needs}
\omterm{def:g7:photo-resp:photo}{Photosynthesis} \omterm{def:g1:caring:needs}{needs} \omterm{def:g4:plants-need:needs}{light}, \omterm{def:g7:photo-resp:photo}{chlorophyll}, carbon dioxide and \omterm{def:g4:plants-need:needs}{water}. Rate
can be limited by whichever factor is in shortest supply (a limiting
factor).
\end{definition}

\begin{example}[Starch test idea]\label{ex:g7:photo-resp:starch}
\omterm{def:g4:plants-need:parts}{Leaves} that have photosynthesised may store \omterm{def:g6:nutrition:groups}{carbohydrate} as starch. A
classic school test uses iodine solution (after treatment) --- blue-black
colour suggests starch. Destarched \omterm{def:g7:tissues:plant-organs}{leaves} kept in darkness test negative
until \omterm{def:g4:plants-need:needs}{light} returns.
\end{example}

\section{Respiration}

\begin{definition}[Respiration]\label{def:g7:photo-resp:resp}
\emph{Cellular respiration}\index{respiration!cellular} is the release of
\omterm{def:g6:nutrition:def}{energy} from \omterm{def:g2:food-energy:food}{food} (often glucose) in \omterm{def:g7:microscope:cell}{cells}, making that \omterm{def:g6:nutrition:def}{energy} available
for life processes. In aerobic respiration, oxygen is used and carbon
dioxide and \omterm{def:g4:plants-need:needs}{water} are produced.
\end{definition}

\begin{proposition}[Aerobic word equation]\label{prop:g7:photo-resp:resp-eq}
\[
\text{glucose} + \text{oxygen} \to
\text{carbon dioxide} + \text{water} + \text{energy (usable form)}.
\]
\end{proposition}
\admitted

\section{How the processes link}

\begin{proposition}[Complementary roles]\label{prop:g7:photo-resp:link}
\omterm{def:g7:photo-resp:photo}{Photosynthesis} stores \omterm{def:g4:plants-need:needs}{light} \omterm{def:g6:nutrition:def}{energy} in sugars and releases $\mathrm{O_2}$.
Respiration releases \omterm{def:g6:nutrition:def}{energy} from sugars and produces $\mathrm{CO_2}$ and
\omterm{def:g4:plants-need:needs}{water}. \omterm{def:g1:plants-animals:plant}{Plants} photosynthesise in \omterm{def:g4:plants-need:needs}{light} and respire all the time; \omterm{def:g1:plants-animals:animal}{animals}
only respire (they do not photosynthesise).
\end{proposition}
\admitted

\begin{example}[Pondweed bubbles]\label{ex:g7:photo-resp:bubbles}
In bright \omterm{def:g4:plants-need:needs}{light}, aquatic \omterm{def:g1:plants-animals:plant}{plants} may release oxygen bubbles --- evidence
that \omterm{def:g7:photo-resp:photo}{photosynthesis} is producing gas faster than the \omterm{def:g1:plants-animals:plant}{plant} uses it.
\end{example}
