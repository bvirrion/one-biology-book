\section*{Chapter \ref{ch:g11:genes-and-disease} --- Genetic Variation and Disease}

\begin{solution}{exo:g11:genes-and-disease:1}
\omterm{def:g11:genes-and-disease:genetic}{Recessive}: the disease appears only with two copies of the \omterm{def:g10:universal-dna:gene}{allele};
\omterm{def:g11:genes-and-disease:genetic}{dominant}: one copy suffices. A carrier holds one \omterm{def:g11:genes-and-disease:genetic}{recessive} mutant
\omterm{def:g10:universal-dna:gene}{allele} and is healthy.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:2}
Affected $1/4$, carrier $1/2$, neither $1/4$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:3}
I-1 $Aa$, I-2 $Aa$ (parents of an affected child), II-2 $aa$, III-2
$aa$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:4}
A male has one X \omterm{def:g11:cell-cycle-mitosis:chromosome}{chromosome}: a \omterm{def:g11:genes-and-disease:genetic}{recessive} \omterm{def:g10:universal-dna:gene}{allele} on it is expressed,
having no second copy to mask it. A female needs the \omterm{def:g10:universal-dna:gene}{allele} on both
of her X \omterm{def:g10:universal-dna:information}{chromosomes}.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:5}
A disease whose risk depends on many \omterm{def:g10:universal-dna:gene}{alleles} of small effect and on
the environment: type 2 diabetes, heart disease (also asthma, most
cancers).
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:6}
$aa \times AA$: all children $Aa$, healthy carriers. A carrier child
with a carrier: $1/4$ affected, $1/2$ carriers, $1/4$ non-carriers.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:7}
II-1 is a healthy child of two carriers: carrier with probability
$2/3$. Risk with a man of the population: $2/3 \times 1/25 \times 1/4
= 1/150$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:8}
$1/2$ for each child. None of four: $(1/2)^4 = 1/16$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:9}
$X^hY \times X^HX^H$: all daughters $X^HX^h$, healthy carriers; all
sons $X^HY$, healthy. A carrier daughter with a healthy man: half her
sons haemophiliac, half her daughters carriers.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:10}
One \omterm{def:g10:universal-dna:gene}{allele} gives 400 and 200; the other, with the further deletion,
gives 300 and 200. Bands at 400, 300 and 200 --- and no 600.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:11}
The frequency of an \omterm{def:g10:universal-dna:gene}{allele} in a population depends on its history,
not only on the \omterm{def:g11:mutations:mutation}{mutation} rate: the European \omterm{def:g10:universal-dna:gene}{allele} arose long ago and
spread, perhaps because carriers had some advantage (resistance to a
diarrhoeal disease is suspected); in eastern Asia it did not.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:12}
Each child is an independent draw: $1/4$ for the second. At least one
of the next two: $1 - (3/4)^2 = 7/16$. The \omterm{def:g10:universal-dna:gene}{alleles} are dealt afresh
at each conception; a previous affected child does not lower the
next one's risk.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:13}
Before the test, carrier with probability $1/25$; the test misses
10\% of carriers, so among 1000 women 40 are carriers and 4 test
negative, while 960 non-carriers test negative: carrier probability
$4/964 \approx 1/240$. Risk with a known carrier: $1/240 \times 1/4
\approx 1/960$.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:14}
Many to few \omterm{def:g10:universal-dna:gene}{alleles}, sedentary: 34\% to 9\%; sedentary to active, many
\omterm{def:g10:universal-dna:gene}{alleles}: 34\% to 12\%. The change one can make gives nearly the same
reduction as the change one cannot. A risk score is useful to
identify who gains most from changing their way of life, not as a
verdict.
\end{solution}

\begin{solution}{exo:g11:genes-and-disease:15}
The test offers certainty --- freedom from a fifty-per-cent shadow if
negative, the ability to plan a life and a family if positive --- and
costs, if positive, the knowledge of an incurable disease decades in
advance, with effects on mood, insurance and relatives who share the
\omterm{def:g10:universal-dna:gene}{gene}. No treatment follows from the result, so the only reason to
know is the person's own; that is why the choice, and its timing, are
theirs.
\end{solution}

\begin{solution}{pb:g11:genes-and-disease:1}
\textbf{1.} \omterm{def:g11:genes-and-disease:genetic}{Recessive}: Zoé is affected and both her parents are
healthy, so each carries one hidden mutant \omterm{def:g10:universal-dna:gene}{allele}.

\textbf{2.} Léa $Aa$, Tom $Aa$, Zoé $aa$.

\textbf{3.} $AA$ with probability $1/3$, $Aa$ with $2/3$ (he is
healthy, so $aa$ is excluded).

\textbf{4.} Tom is a carrier, so one of his parents at least is; with
no affected person in that generation, Ana received the \omterm{def:g10:universal-dna:gene}{allele} from
that parent with probability $1/2$: $1/2$ (taking the other parent as
a non-carrier).

\textbf{5.} $1/25$, the population figure.

\textbf{6.} $1/2 \times 1/25 \times 1/4 = 1/200$.

\textbf{7.} Affected $1/4$; carrier $1/2$.

\textbf{8.} $2/3 \times 1/25 \times 1/4 = 1/150$.

\textbf{9.} Zoé is $aa$; her partner is $Aa$ with probability $1/25$.
Affected: $1/25 \times 1/2 = 1/50$. Carrier: every child receives $a$
from Zoé, so a carrier with probability $24/25 \times 1 + 1/25 \times
1/2 \approx 0.98$.

\textbf{10.} Lou is a carrier with probability $2/3$; the cousin
received $a$ from Ana ($1/2$ carrier) with probability $1/2$, i.e. is a
carrier with probability $1/4$ (ignoring Karim): $2/3 \times 1/4
\times 1/4 = 1/24$, four times higher than for Max, because the cousin
comes from the same family and is far likelier than a stranger to
carry the same \omterm{def:g10:universal-dna:gene}{allele}.

\textbf{11.} Zoé: 600 and 300 only. Tom: 900, 600 and 300. Non-carrier:
900 only.

\textbf{12.} $AA$: he is not a carrier; the risk for his child falls
from $1/150$ to zero (barring a new \omterm{def:g11:mutations:mutation}{mutation}).

\textbf{13.} Ana is $Aa$: risk $1 \times 1/25 \times 1/4 = 1/100$.

\textbf{14.} Among 1000 people like Karim, 40 are carriers of whom 4
test negative, and 960 non-carriers test negative: $4/964 \approx
1/240$. Risk: $1 \times 1/240 \times 1/4 \approx 1/960$.

\textbf{15.} $Aa$: a healthy carrier, like its mother.

\textbf{16.} $(1/25)^2 = 1/625$ of couples are two carriers; a quarter
of their children are affected: $1/2500$.

\textbf{17.} Exactly the observed frequency: the carrier frequency
and the \omterm{def:g11:genes-and-disease:genetic}{recessive} rule account for it.

\textbf{18.} With a carrier frequency of $1/25$ and an affected
frequency of $1/2500$, the mutant \omterm{def:g10:universal-dna:gene}{alleles} carried by healthy carriers
outnumber those in affected people by $2 \times 1/25$ to $2 \times
1/2500$, a hundred to one. Death of the affected removes only 1\% of
the copies per generation; the rest are transmitted invisibly by
carriers.

\textbf{19.} The \omterm{def:g10:universal-dna:gene}{allele}'s frequency will rise, but very slowly: the
1\% of copies that used to be lost each generation are now kept, so
the change is of the order of 1\% of the frequency per generation.

\textbf{20.} Two carriers risk one affected child in four; a healthy
sibling of an affected child is a carrier two times in three; the gel
turned probabilities into \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypes} --- clearing Max, confirming Ana,
reassuring but not clearing Karim --- and identified the foetus as a
healthy carrier.
\end{solution}
