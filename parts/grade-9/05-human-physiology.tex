\chapter{Human Physiology Overview}\label{ch:g9:physiology}

\omterm{def:g9:physiology:def}{Physiology} is how the \omterm{def:g1:my-body:body}{body} works. This chapter surveys major systems ---
exchange, transport, \omterm{def:g6:method:control}{control} and support --- as cooperating parts of one
\omterm{def:g4:ecosystems:levels}{organism}.

\section{Systems at a glance}

\begin{definition}[Physiology]\label{def:g9:physiology:def}
\emph{Physiology}\index{physiology} is the study of the functions of
\omterm{def:g7:tissues:organ}{organs} and systems in living \omterm{def:g4:ecosystems:levels}{organisms}.
\end{definition}

\begin{proposition}[Integration]\label{prop:g9:physiology:integrate}
No system works alone: \omterm{def:g7:digestion:def}{digestion} supplies fuel; circulation delivers it;
respiration supplies oxygen; excretion removes wastes; \omterm{def:g5:nerves:system}{nerves} and hormones
coordinate; \omterm{def:g3:bones:muscle}{muscles} and \omterm{def:g3:bones:bone}{skeleton} move the \omterm{def:g1:my-body:body}{body}.
\end{proposition}
\admitted

\section{Exchange with the environment}

\begin{definition}[Respiratory system]\label{def:g9:physiology:resp}
The \emph{respiratory system}\index{respiratory system} moves \omterm{def:g4:plants-need:needs}{air} to the
\omterm{def:g5:circulation:breathing}{lungs} so oxygen can enter \omterm{def:g7:digestion:blood}{blood} and carbon dioxide can leave. \omterm{def:g5:circulation:breathing}{Gas exchange}
occurs across moist, thin surfaces with large area (alveoli).
\end{definition}

\begin{definition}[Digestive and excretory]\label{def:g9:physiology:gut}
The \omterm{def:g4:body-systems:digestion}{digestive system} breaks down \omterm{def:g2:food-energy:food}{food} and absorbs \omterm{def:g4:plants-need:needs}{nutrients}. The
\emph{excretory (urinary) system}\index{excretory system} removes nitrogen
wastes and helps balance \omterm{def:g4:plants-need:needs}{water} and salts (kidneys).
\end{definition}

and pass \omterm{def:g4:plants-need:needs}{nutrients} to the \omterm{def:g7:digestion:blood}{blood}.}

\section{Internal transport and defence}

\begin{definition}[Circulatory system]\label{def:g9:physiology:circ}
The \emph{circulatory system}\index{circulatory system} (\omterm{def:g7:digestion:heart}{heart}, \omterm{def:g7:digestion:blood}{blood},
vessels) transports gases, \omterm{def:g4:plants-need:needs}{nutrients}, hormones, wastes and immune \omterm{def:g7:microscope:cell}{cells}.
\end{definition}

\begin{definition}[Immune system]\label{def:g9:physiology:immune}
The \emph{immune system}\index{immune system} defends against \omterm{def:g8:microbiology:pathogen}{pathogens}
using \omterm{def:g8:immune:barriers}{barriers}, innate \omterm{def:g7:nervous:stim}{responses} and adaptive memory (\omterm{def:g8:immune:adaptive}{antibodies},
\omterm{def:g8:immune:adaptive}{lymphocytes}).
\end{definition}

\section{Control and movement}

\begin{definition}[Nervous and endocrine]\label{def:g9:physiology:control}
The \emph{nervous system}\index{nervous system} sends fast electrical
signals. The \emph{endocrine system}\index{endocrine system} uses hormones
in \omterm{def:g7:digestion:blood}{blood} for slower, longer-lasting \omterm{def:g6:method:control}{control} (\omterm{def:g2:growing:growth}{growth}, metabolism, stress).
\end{definition}

\begin{definition}[Musculoskeletal]\label{def:g9:physiology:msk}
\omterm{def:g3:bones:bone}{Bones} support and protect; \omterm{def:g3:bones:joint}{joints} allow movement; skeletal \omterm{def:g3:bones:muscle}{muscles} pull on
\omterm{def:g3:bones:bone}{bones} under nervous \omterm{def:g6:method:control}{control}.
\end{definition}

\begin{example}[Running]\label{ex:g9:physiology:run}
\omterm{def:g3:bones:muscle}{Muscles} demand oxygen and glucose; \omterm{def:g5:circulation:breathing}{breathing} and \omterm{def:g7:digestion:heart}{heart} rate rise;
sweating helps cool; \omterm{def:g5:nerves:system}{nerves} coordinate stride --- multi-system teamwork.
\end{example}

\begin{method}[Tracing a molecule]\label{met:g9:physiology:trace}
Pick a molecule (oxygen, glucose, urea) and list: entry/production \omterm{def:g7:tissues:organ}{organ}
$\to$ transport $\to$ use or exit. This reveals system links.
\end{method}
