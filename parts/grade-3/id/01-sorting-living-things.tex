\chapter{Mengelompokkan Makhluk Hidup}\label{ch:g3:sorting-living-things}

Tuangkan sekotak kancing bercampur ke atas meja, dan tanganmu mulai
mengelompokkan dengan sendirinya: yang besar dengan yang besar, yang
merah dengan yang merah, yang berlubang empat dengan yang berlubang
empat. Para naturalis merasakan dorongan yang sama di hadapan
campur-baur kumbang, \omterm{prop:g3:sorting-living-things:groups}{burung}, \omterm{def:g1:plants-around-us:plant}{tumbuhan} paku dan \omterm{prop:g3:sorting-living-things:groups}{ikan} di dunia yang
hidup --- dan mengelompokkan \omterm{def:g1:living-or-not:living}{makhluk hidup} dengan cermat ternyata
menjadi salah satu gagasan \omterm{def:g1:living-or-not:living}{biologi} yang paling ampuh.

\section{Pengelompokan membutuhkan aturan}

\begin{definition}[Pengelompokan dan kriteria]\label{def:g3:sorting-living-things:criterion}
\emph{Mengelompokkan}\index{pengelompokan} sebuah kumpulan berarti
menyatukan hal-hal yang memang sepadan. Suatu
\emph{kriteria}\index{kriteria} adalah aturan yang dipakai untuk
pengelompokan itu --- ``berkaki enam'', ``berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}''. Kriteria
yang baik bagi \omterm{def:g1:living-or-not:living}{makhluk hidup} adalah ciri tubuh yang dapat kamu
\emph{amati}: sesuatu yang \emph{dimiliki} \omterm{def:g1:animals-around-us:animal}{hewan} atau \omterm{def:g1:plants-around-us:plant}{tumbuhannya},
bukan tempat ia hidup atau apa yang kebetulan sedang dilakukannya.
\end{definition}

\begin{example}[Dua pengelompokan atas hewan yang sama]\label{ex:g3:sorting-living-things:two}
Ambil kucing, bebek, \omterm{prop:g3:sorting-living-things:groups}{ikan} trout dan kepik. Dikelompokkan menurut
``hidup dekat air'': bebek dan \omterm{prop:g3:sorting-living-things:groups}{ikan} trout bersama. Dikelompokkan
menurut ``berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}'': bebeknya berdiri sendiri. Keduanya
\omterm{def:g3:sorting-living-things:criterion}{pengelompokan} --- tetapi hanya yang kedua memakai tubuh \omterm{def:g1:animals-around-us:animal}{hewannya}
sendiri, dan jenis itulah yang dipercaya ahli \omterm{def:g1:living-or-not:living}{biologi}: seekor \omterm{def:g1:animals-around-us:animal}{hewan}
dapat pindah rumah, tetapi ia tidak dapat menukar \omterm{def:g1:animals-around-us:covering}{bulu burungnya}
dengan \omterm{def:g1:animals-around-us:covering}{sisik}.
\end{example}

\begin{remark}[Mengapa tidak mengelompokkan menurut rumah atau makanan?]\label{rem:g3:sorting-living-things:why}
\omterm{def:g2:where-animals-live:habitat}{Habitat} dan daftar makanan berubah bersama musim dan bersama nasib ---
\omterm{prop:g3:sorting-living-things:groups}{burung} hitam pada \cref{ch:g2:what-animals-eat} beralih dari cacing ke
buah beri tanpa menjadi \omterm{def:g1:animals-around-us:animal}{hewan} yang lain. Ciri tubuh tetap. Itulah
sebabnya ``apa yang dimilikinya'' mengalahkan ``apa yang
dilakukannya'' sebagai aturan \omterm{def:g3:sorting-living-things:criterion}{pengelompokan}.
\end{remark}

\section{Kelompok-kelompok besar hewan}

\begin{proposition}[Beberapa kelompok besar]\label{prop:g3:sorting-living-things:groups}
\omterm{def:g3:sorting-living-things:criterion}{Pengelompokan} menurut ciri tubuh mengumpulkan \omterm{def:g1:animals-around-us:animal}{hewan} ke dalam
kelompok-kelompok besar, di antaranya:
\begin{itemize}
  \item \emph{serangga}\index{serangga}: berkaki enam, bertubuh tiga
        bagian --- semut, kepik, lebah, kupu-kupu;
  \item \emph{ikan}\index{ikan}: bersirip dan bersisik, hidupnya
        dijalani di dalam air --- \omterm{prop:g3:sorting-living-things:groups}{ikan} trout, \omterm{prop:g3:sorting-living-things:groups}{ikan} mas koki;
  \item \emph{burung}\index{burung}: berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}, bersayap dan
        berparuh --- bebek, \omterm{prop:g3:sorting-living-things:groups}{burung} hitam, \omterm{prop:g3:sorting-living-things:groups}{burung} unta;
  \item \emph{mamalia}\index{mamalia}: berbulu atau berambut, dan
        induknya memberi anaknya \omterm{def:g2:animals-grow-up:born}{susu} --- kucing, sapi, kelelawar,
        paus --- serta manusia;
  \item \emph{amfibi}\index{amfibi}: berkulit lembap, \omterm{def:g2:animals-grow-up:born}{telurnya}
        diletakkan di air, masa kecilnya sebagai \omterm{ex:g2:animals-grow-up:frog}{berudu} --- katak,
        salamander.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Kejutan yang ditangani dengan benar]\label{ex:g3:sorting-living-things:surprises}
\omterm{def:g3:sorting-living-things:criterion}{Kriteria} menyelamatkan kita dari terkaan yang malas. Kelelawar terbang
seperti \omterm{prop:g3:sorting-living-things:groups}{burung} --- tetapi ia berbulu, tidak berbulu \omterm{prop:g3:sorting-living-things:groups}{burung}, dan
memberi anaknya \omterm{def:g2:animals-grow-up:born}{susu}: seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia}. Paus berenang seperti \omterm{prop:g3:sorting-living-things:groups}{ikan} ---
\omterm{def:g1:animals-around-us:covering}{bulunya} menyusut hampir habis, tetapi ada \omterm{def:g2:animals-grow-up:born}{susu} bagi anaknya dan tidak
ada \omterm{def:g1:animals-around-us:covering}{sisik}: seekor \omterm{prop:g3:sorting-living-things:groups}{mamalia} juga. \omterm{prop:g3:sorting-living-things:groups}{Burung} unta sama sekali tidak dapat
terbang, namun \omterm{def:g1:animals-around-us:covering}{bulu burung} dan paruhnya berkata: \omterm{prop:g3:sorting-living-things:groups}{burung}. Kriterialah,
bukan kesan, yang memutuskan.
\end{example}

\begin{method}[Mengelompokkan hewan yang belum dikenal]\label{met:g3:sorting-living-things:sort}
\begin{enumerate}
  \item Amati tubuhnya: \omterm{def:g1:the-five-senses:sense}{kulit}, \omterm{def:g1:animals-around-us:covering}{bulu}, \omterm{def:g1:animals-around-us:covering}{bulu burung} atau \omterm{def:g1:animals-around-us:covering}{sisik}? berapa
        kaki? bersayap? berparuh?
  \item Tanyakan tentang anaknya kalau kamu bisa: \omterm{def:g2:animals-grow-up:born}{susu}? \omterm{def:g2:animals-grow-up:born}{telur}? tahap
        \omterm{ex:g2:animals-grow-up:frog}{berudu}?
  \item Bandingkan dengan kelompok-kelompok pada
        \cref{prop:g3:sorting-living-things:groups} lalu tempatkan
        \omterm{def:g1:animals-around-us:animal}{hewannya} di tempat yang cocok dengan \emph{ciri}-cirinya;
  \item kalau sebuah kesan (``ia terbang!'') bertentangan dengan
        sebuah ciri (``ia berbulu''), percayai cirinya.
\end{enumerate}
\end{method}

\section{Tumbuhan pun dikelompokkan}

\begin{proposition}[Mengelompokkan tumbuhan]\label{prop:g3:sorting-living-things:plants}
\omterm{def:g1:plants-around-us:plant}{Tumbuhan} dikelompokkan menurut cirinya persis seperti \omterm{def:g1:animals-around-us:animal}{hewan}. \omterm{def:g3:sorting-living-things:criterion}{Kriteria}
yang berguna: apakah ia membuat \emph{\omterm{def:g1:plants-around-us:parts}{bunga}} dan \omterm{def:g2:from-seed-to-plant:seed}{biji} (\omterm{def:g1:plants-around-us:parts}{bunga} aster,
pohon berangan) atau tidak pernah sama sekali (\omterm{def:g1:plants-around-us:plant}{tumbuhan} paku, lumut)?
Apakah \omterm{def:g1:plants-around-us:parts}{batangnya} lunak dan hijau, atau berkayu seperti \omterm{def:g1:plants-around-us:tree}{batang pokok}
sebuah \emph{pohon}? Apakah \omterm{def:g1:plants-around-us:parts}{daunnya} lebar, atau berupa jarum seperti
\omterm{def:g1:plants-around-us:parts}{daun} pinus?
\end{proposition}
\admitted

\begin{example}[Tiga sudut taman]\label{ex:g3:sorting-living-things:park}
Di satu taman: semak mawar --- berbunga, berbatang kayu; lumut di
tembok sisi utara --- tidak pernah berbunga, sama sekali tidak
berbatang sejati; pohon pinus --- sebuah pohon berdaun jarum dan
berbuah runjung. Tiga \omterm{def:g1:plants-around-us:plant}{tumbuhan}, tiga jawaban berbeda atas \omterm{def:g3:sorting-living-things:criterion}{kriteria}
yang sama.
\end{example}

\begin{remark}[Pengelompokan adalah langkah pertama sebuah ilmu]\label{rem:g3:sorting-living-things:science}
Kelompok yang tersusun baik memungkinkan orang mengajukan pertanyaan
yang lebih baik: apa pun yang benar tentang \omterm{prop:g3:sorting-living-things:groups}{serangga} pada umumnya ---
berkaki enam, bertubuh tiga bagian, sering bersayap --- seketika
diketahui bagi ratusan ribu spesies sekaligus. Bab-bab berikutnya akan
mengubah \omterm{def:g3:sorting-living-things:criterion}{pengelompokan} menjadi pohon keluarga semua \omterm{def:g1:living-or-not:living}{makhluk hidup} ---
untuk sekarang, belajarlah mengelompokkan menurut apa yang dapat kamu
lihat.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:1}
Apa itu \omterm{def:g3:sorting-living-things:criterion}{kriteria}? Berikan satu \omterm{def:g3:sorting-living-things:criterion}{kriteria} yang baik untuk mengelompokkan
\omterm{def:g1:animals-around-us:animal}{hewan}, dan satu yang dihindari ahli \omterm{def:g1:living-or-not:living}{biologi}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:2}
Sebutkan ciri (atau ciri-ciri) yang menentukan tiap kelompok:
\omterm{prop:g3:sorting-living-things:groups}{serangga}, \omterm{prop:g3:sorting-living-things:groups}{ikan}, \omterm{prop:g3:sorting-living-things:groups}{burung}, \omterm{prop:g3:sorting-living-things:groups}{mamalia}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:3}
Tempatkan tiap \omterm{def:g1:animals-around-us:animal}{hewan} dalam kelompoknya: lebah, \omterm{prop:g3:sorting-living-things:groups}{ikan} mas koki, \omterm{prop:g3:sorting-living-things:groups}{burung}
unta, sapi, katak.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:4}
Kelelawar terbang. Mengapa ia \omterm{prop:g3:sorting-living-things:groups}{mamalia} dan bukan \omterm{prop:g3:sorting-living-things:groups}{burung}? Ciri mana yang
memutuskan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:5}
Paus hidup di laut. Mengapa ia bukan \omterm{prop:g3:sorting-living-things:groups}{ikan}? Ciri mana yang memutuskan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:6}
Seekor laba-laba berkaki delapan. Dapatkah ia menjadi \omterm{prop:g3:sorting-living-things:groups}{serangga}? Apa
jawaban \omterm{def:g3:sorting-living-things:criterion}{kriterianya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:7}
Kelompokkan \omterm{def:g1:plants-around-us:plant}{tumbuhan} berikut dengan \omterm{def:g3:sorting-living-things:criterion}{kriteria} pada
\cref{prop:g3:sorting-living-things:plants}: \omterm{def:g1:plants-around-us:parts}{bunga} aster, lumut, pohon
pinus, semak mawar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:sorting-living-things:8}
Cobalah: kelompokkan \omterm{def:g1:animals-around-us:animal}{hewan} di jalanmu atau kebunmu --- setiap \omterm{def:g1:animals-around-us:animal}{hewan}
yang dapat kamu temukan dalam sekali jalan --- ke dalam
kelompok-kelompok bab ini. Kelompok mana yang menang jumlahnya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sorting-living-things:9}
Lea mengelompokkan \omterm{def:g1:animals-around-us:animal}{hewan} menjadi ``\omterm{def:g1:animals-around-us:animal}{hewan} peliharaan'' dan ``\omterm{def:g1:animals-around-us:animal}{hewan}
liar''. Temannya mengelompokkan \omterm{def:g1:animals-around-us:animal}{hewan} yang sama menjadi ``\omterm{prop:g3:sorting-living-things:groups}{mamalia}''
dan ``\omterm{prop:g3:sorting-living-things:groups}{burung}''. \omterm{def:g3:sorting-living-things:criterion}{Pengelompokan} siapa yang termasuk jenis ahli \omterm{def:g1:living-or-not:living}{biologi},
dan mengapa? Pakai \cref{rem:g3:sorting-living-things:why}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:sorting-living-things:10}
Seekor penguin berenang dengan sangat baik, tidak dapat terbang, dan
tertutup oleh sesuatu yang tampak seperti genting mengilap. Seorang
naturalis mengamati lebih dekat: ``gentingnya'' adalah \omterm{def:g1:animals-around-us:covering}{bulu burung}
yang tersusun rapat, dan ada sebuah paruh. Jalankan
\cref{met:g3:sorting-living-things:sort} atas penguinnya, langkah demi
langkah.
\end{exercise}
