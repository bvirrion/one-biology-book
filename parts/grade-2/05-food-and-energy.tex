\chapter{Food and Energy}\label{ch:g2:food-energy}

Why do you eat breakfast? Why does a \omterm{def:g1:plants-animals:plant}{plant} need light? \omterm{def:g1:living-nonliving:living}{Living
things} need \emph{\omterm{def:g2:food-energy:energy}{energy}} to \omterm{def:g2:what-alive:signs}{grow}, move and stay warm. \omterm{def:g2:food-energy:food}{Food} and
light are how that \omterm{def:g2:food-energy:energy}{energy} arrives.

\section{Energy for life}

\begin{definition}[Energy]\label{def:g2:food-energy:energy}
\emph{Energy}\index{energy} is what \omterm{def:g1:living-nonliving:living}{living things} use to do work:
\omterm{def:g2:what-alive:signs}{grow}, move, keep warm and repair the \omterm{def:g1:my-body:body}{body}. Without energy, life
stops.
\end{definition}
\omimg{food-chain.jpg}{0.50\linewidth}{Food chains show who eats what.}


\begin{definition}[Food]\label{def:g2:food-energy:food}
\emph{Food}\index{food} is material that \omterm{def:g1:plants-animals:animal}{animals} eat to get \omterm{def:g2:food-energy:energy}{energy}
and building blocks for the \omterm{def:g1:my-body:body}{body}. \omterm{def:g1:plants-animals:plant}{Plants} make their own food using
light.
\end{definition}

\section{Plants and animals}

\begin{proposition}[Two ways to get energy]\label{prop:g2:food-energy:two}
\begin{itemize}
  \item \omterm{def:g1:plants-animals:plant}{Plants} use light, \omterm{def:g1:caring:needs}{air} and \omterm{def:g1:caring:needs}{water} to make \omterm{def:g2:food-energy:food}{food} in their
        \omterm{def:g2:plant-parts:main}{leaves}.
  \item \omterm{def:g1:plants-animals:animal}{Animals} must eat \omterm{def:g1:plants-animals:plant}{plants} or other \omterm{def:g1:plants-animals:animal}{animals}.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Lunch box]\label{ex:g2:food-energy:lunch}
Bread comes from wheat (a \omterm{def:g1:plants-animals:plant}{plant}). Cheese comes from milk (from a
\omterm{def:g2:animal-groups:five}{mammal} that ate \omterm{def:g1:plants-animals:plant}{plants}). An apple is from a tree. Your lunch stores
\omterm{def:g2:food-energy:energy}{energy} that began with sunlight on \omterm{def:g1:plants-animals:plant}{plants}.
\end{example}

\begin{method}[Trace the energy]\label{met:g2:food-energy:trace}
\begin{enumerate}
  \item Start with the Sun.
  \item Find a \omterm{def:g1:plants-animals:plant}{plant} that used the light.
  \item Find who ate the \omterm{def:g1:plants-animals:plant}{plant} (or who ate the plant-eater).
  \item End with you or another \omterm{def:g1:plants-animals:animal}{animal}.
\end{enumerate}
\end{method}

\begin{definition}[Healthy fuel]\label{def:g2:food-energy:healthy}
Different \omterm{def:g2:food-energy:food}{foods} help in different ways. \omterm{def:g2:plant-parts:main}{Fruit}, vegetables, grains,
proteins and \omterm{def:g1:caring:needs}{water} work together. Sweets give a quick boost but are
not enough alone. Drink \omterm{def:g1:caring:needs}{water} often.
\end{definition}
