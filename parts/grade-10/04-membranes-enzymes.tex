\chapter{Membranes and Enzymes}\label{ch:g10:membranes-enzymes}

Life is chemistry in compartments. Membranes set the boundaries and gate
the traffic; \omterm{def:g10:membranes-enzymes:enzyme}{enzymes} accelerate the reactions that would otherwise be too
slow. This chapter links the fluid-mosaic membrane to \omterm{def:g10:membranes-enzymes:diffusion}{diffusion}, \omterm{def:g10:membranes-enzymes:osmosis}{osmosis}
and \omterm{def:g10:membranes-enzymes:active}{active transport}, then treats \omterm{def:g10:membranes-enzymes:enzyme}{enzyme} catalysis and the factors that
speed or stall it.

\section{The plasma membrane}

\begin{definition}[Fluid mosaic model]\label{def:g10:membranes-enzymes:fluid-mosaic}
The \emph{plasma membrane}\index{plasma membrane} is a
\emph{phospholipid bilayer}\index{bilayer} studded with \omterm{def:g10:molecules-of-life:proteins}{proteins},
described by the \emph{fluid mosaic model}\index{fluid mosaic model}:
\omterm{def:g10:molecules-of-life:lipids}{lipids} and many \omterm{def:g10:molecules-of-life:proteins}{proteins} diffuse laterally in the plane of the membrane
(``fluid''), forming a patchwork of components (``mosaic'').
\omterm{def:g10:molecules-of-life:carbs}{Carbohydrate} chains on \omterm{def:g10:molecules-of-life:lipids}{lipids} and \omterm{def:g10:molecules-of-life:proteins}{proteins} form the glycocalyx on the
outer face.
\end{definition}

\begin{definition}[Selective permeability]\label{def:g10:membranes-enzymes:perm}
A membrane is \emph{selectively permeable}\index{selective permeability}:
small nonpolar molecules (e.g.\ $\mathrm{O_2}$, $\mathrm{CO_2}$) cross
readily; ions and large \omterm{def:g10:molecules-of-life:water}{polar molecules} need \omterm{def:g10:molecules-of-life:proteins}{proteins}; water crosses
slowly on its own and faster through aquaporins.
\end{definition}

\section{Passive and active transport}

\begin{definition}[Diffusion]\label{def:g10:membranes-enzymes:diffusion}
\emph{Diffusion}\index{diffusion} is the net movement of particles from
higher to lower concentration (down the concentration gradient) due to
random thermal motion. It requires no cellular ATP.
\end{definition}

\begin{definition}[Facilitated diffusion]\label{def:g10:membranes-enzymes:facilitated}
\emph{Facilitated diffusion}\index{facilitated diffusion} is passive
transport of a solute through a channel or carrier \omterm{def:g10:molecules-of-life:proteins}{protein}, still down
its gradient --- faster than unaided crossing, still without ATP.
\end{definition}

\begin{definition}[Osmosis]\label{def:g10:membranes-enzymes:osmosis}
\emph{Osmosis}\index{osmosis} is the net \omterm{def:g10:membranes-enzymes:diffusion}{diffusion} of water across a
\omterm{def:g10:membranes-enzymes:perm}{selectively permeable} membrane from a region of higher water potential
(typically lower solute concentration) to lower water potential (higher
solute concentration).
\end{definition}

\begin{definition}[Tonicity]\label{def:g10:membranes-enzymes:tonicity}
Relative to a \omterm{def:g10:the-cell:cell}{cell}:
\begin{enumerate}
  \item \emph{Isotonic}\index{isotonic} --- same effective solute;
        no net water movement;
  \item \emph{Hypotonic}\index{hypotonic} --- lower external solute;
        water enters the \omterm{def:g10:the-cell:cell}{cell};
  \item \emph{Hypertonic}\index{hypertonic} --- higher external solute;
        water leaves the \omterm{def:g10:the-cell:cell}{cell}.
\end{enumerate}
Animal \omterm{def:g10:the-cell:cell}{cells} may burst in strongly hypotonic media (lysis) or shrink in
hypertonic media (crenation). Plant \omterm{def:g10:the-cell:cell}{cells} become turgid in hypotonic
media (wall prevents burst) and plasmolysed in hypertonic media.
\end{definition}

\begin{definition}[Active transport]\label{def:g10:membranes-enzymes:active}
\emph{Active transport}\index{active transport} moves a solute
\emph{against} its concentration gradient using energy (usually ATP) and
a membrane \omterm{def:g10:molecules-of-life:proteins}{protein} pump (e.g.\ sodium--potassium pump).
\emph{Endocytosis}\index{endocytosis} and
\emph{exocytosis}\index{exocytosis} move bulk material in vesicles.
\end{definition}

\begin{method}[Classifying membrane traffic]\label{met:g10:membranes-enzymes:classify}
Ask three questions:
\begin{enumerate}
  \item Is the net movement with or against the gradient?
  \item Is a membrane \omterm{def:g10:molecules-of-life:proteins}{protein} required?
  \item Is metabolic energy (ATP) required?
\end{enumerate}
With gradient, no ATP $\to$ passive (simple or facilitated). Against
gradient, ATP $\to$ active.
\end{method}

\section{Enzymes as biological catalysts}

\begin{definition}[Enzyme]\label{def:g10:membranes-enzymes:enzyme}
An \emph{enzyme}\index{enzyme} is a biological catalyst, almost always a
\omterm{def:g10:molecules-of-life:proteins}{protein}, that speeds a specific reaction without being consumed. The
reactant is the \emph{substrate}\index{substrate}; it binds at the
\emph{active site}\index{active site}.
\end{definition}

\begin{proposition}[Lowering activation energy]\label{prop:g10:membranes-enzymes:Ea}
\omterm{def:g10:membranes-enzymes:enzyme}{Enzymes} lower the \emph{activation energy}\index{activation energy}
barrier of a reaction, increasing the rate at which equilibrium is
approached. They do not change the overall free-energy difference between
reactants and products.
\end{proposition}
\admitted

\begin{definition}[Enzyme--substrate complex]\label{def:g10:membranes-enzymes:ES}
\omterm{def:g10:membranes-enzymes:enzyme}{Substrate} binding forms an \emph{enzyme--substrate
complex}\index{enzyme--substrate complex}. The \omterm{def:g10:membranes-enzymes:enzyme}{active site} is shaped for
specific \omterm{def:g10:membranes-enzymes:enzyme}{substrates} (\emph{specificity}\index{enzyme specificity});
slight induced fit improves catalysis. Products leave and the \omterm{def:g10:membranes-enzymes:enzyme}{enzyme} is
reused.
\end{definition}

\begin{example}[Catalase]\label{ex:g10:membranes-enzymes:catalase}
Catalase decomposes hydrogen peroxide to water and oxygen --- essential
detoxification. One \omterm{def:g10:membranes-enzymes:enzyme}{enzyme} molecule can process millions of peroxide
molecules per second under good conditions.
\end{example}

\section{Factors affecting enzyme activity}

\begin{proposition}[Temperature and pH]\label{prop:g10:membranes-enzymes:temp-pH}
Raising temperature increases collision rate and activity up to an
\emph{optimum}\index{optimum (enzyme)}; beyond that, denaturation
collapses the \omterm{def:g10:membranes-enzymes:enzyme}{active site} and activity falls. Each \omterm{def:g10:membranes-enzymes:enzyme}{enzyme} also has a pH
\omterm{prop:g10:membranes-enzymes:temp-pH}{optimum} (e.g.\ pepsin acidic, many cytoplasmic \omterm{def:g10:membranes-enzymes:enzyme}{enzymes} near neutral);
extreme pH disrupts ionic bonds that hold shape.
\end{proposition}
\admitted

\begin{proposition}[Substrate and enzyme concentration]\label{prop:g10:membranes-enzymes:conc}
At fixed \omterm{def:g10:membranes-enzymes:enzyme}{enzyme} level, rate rises with \omterm{def:g10:membranes-enzymes:enzyme}{substrate} concentration until
\omterm{def:g10:membranes-enzymes:enzyme}{active sites} are \omterm{rem:g10:molecules-of-life:sat}{saturated} ($V_{\max}$ region). Adding more \omterm{def:g10:membranes-enzymes:enzyme}{enzyme}
raises the ceiling, all else equal.
\end{proposition}
\admitted

\begin{definition}[Inhibitors]\label{def:g10:membranes-enzymes:inhibitors}
A \emph{competitive inhibitor}\index{competitive inhibitor} resembles the
\omterm{def:g10:membranes-enzymes:enzyme}{substrate} and blocks the \omterm{def:g10:membranes-enzymes:enzyme}{active site}. A
\emph{non-competitive inhibitor}\index{non-competitive inhibitor} binds
elsewhere and changes active-site shape. Many toxins and some drugs are
\omterm{def:g10:membranes-enzymes:enzyme}{enzyme} inhibitors; \omterm{def:g10:the-cell:cell}{cells} also use inhibitors for regulation.
\end{definition}

\begin{method}[Reading an enzyme graph]\label{met:g10:membranes-enzymes:graph}
\begin{enumerate}
  \item Identify the independent factor (temperature, pH, [S], \dots).
  \item Locate the \omterm{prop:g10:membranes-enzymes:temp-pH}{optimum} if present.
  \item Explain the rising limb (more collisions / more occupied sites)
        and the falling limb (denaturation or charge disruption).
  \item State what was held constant.
\end{enumerate}
\end{method}

\section{Problem: Membrane clinic and enzyme lab}

\begin{problem}[{Weekend problem --- diagnose osmotic stress and design
an enzyme assay: from wilted celery to a rate curve}]
\label{pb:g10:membranes-enzymes:1}
A greengrocer stores celery in very salty water ``to keep it crisp''.
The sticks become limp. Separately, a student investigates lipase
activity by measuring pH drop as fatty acids are released.

\medskip
\noindent\textbf{Part I --- Celery forensics.}
\begin{enumerate}
  \item Explain the limp celery using \omterm{def:g10:membranes-enzymes:osmosis}{osmosis} and plasmolysis.
  \item Prescribe a recovery treatment and explain why it should work.
  \item Contrast what would happen to an animal \omterm{def:g10:the-cell:cell}{cell} in the same salty
        water.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Lipase assay.}
\begin{enumerate}
  \item Why does pH fall as lipase works on a \omterm{def:g10:molecules-of-life:lipids}{lipid} emulsion?
  \item Sketch expected initial rate versus temperature, labelling the
        \omterm{prop:g10:membranes-enzymes:temp-pH}{optimum} region.
  \item How would a \omterm{def:g10:membranes-enzymes:inhibitors}{competitive inhibitor} of lipase change the curve of
        rate versus \omterm{def:g10:membranes-enzymes:enzyme}{substrate} concentration?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Integration.}
\begin{enumerate}
  \item Membrane \omterm{def:g10:molecules-of-life:proteins}{proteins} include pumps and channels. Sort one example
        of each into active versus passive transport.
  \item Why must \omterm{def:g10:the-cell:cell}{cells} spend ATP to stay away from equilibrium with
        their surroundings?
\end{enumerate}
\end{problem}
