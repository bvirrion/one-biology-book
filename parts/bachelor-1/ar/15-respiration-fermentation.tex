\chapter{التنفس الخلوي والتخمر}\label{ch:b1:respiration-fermentation}

دورق من الخميرة في محلول سكري محجوب عنه الهواء يفور بثاني أكسيد
الكربون ويحول السكر إلى كحول؛ وأدخل إليه الهواء فيبطؤ الفوران، ويقف
الكحول، وتنمو الخميرة أسرع بعشر مرات على السكر نفسه. رأى باستور هذا
سنة 1861 ولم يستطع تفسيره؛ وقد استغرق التفسير قرنًا ويمتد عبر هذا
الفصل كله. فالأكسجين يتيح \omterm{def:b1:cell-unit-of-life:cell}{للخلية} أن تستخرج من الغلوكوز خمسة عشر ضعف
الطاقة التي تحصلها بدونه، وهو يفعل ذلك بتمرير إلكترونات السكر هابطةً
على سلسلة من الحوامل إلى الأكسجين، وضخ البروتونات عبر غشاء وهي تهبط،
ثم إرجاع البروتونات عبر عنفة تصنع \omterm{def:b1:water-small-molecules:atp}{ATP}. يصف هذا الفصل \omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري}،
\omterm{def:b1:respiration-fermentation:krebs}{ودورة كريبس}، \omterm{def:b1:respiration-fermentation:chain}{وسلسلة التنفس} واقترانها التناضحي الكيميائي، والتخمرات
التي تستغني عن الأكسجين، وحرق الشحوم \omterm{def:b1:proteins:peptide}{والبروتينات} بالآلة نفسها.

\section{أكسدة الغلوكوز}

\begin{proposition}[التفاعل الإجمالي ومراحله]\label{prop:b1:respiration-fermentation:overview}
\index{التنفس الخلوي}
الأكسدة التامة للغلوكوز،
\[
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \to 6\,\mathrm{CO_2} + 6\,\mathrm{H_2O},
\qquad \Delta G^{\circ\prime} = \qty{-2870}{kJ/mol},
\]
تجري على ثلاث مراحل لا تدع الإلكترونات تلاقي الأكسجين مباشرة أبدًا.
\emph{\omterm{def:b1:respiration-fermentation:glycolysis}{فالتحلل السكري}}، في \omterm{def:b1:eukaryotic-cell:organelle}{العصارة الخلوية}، يشطر الغلوكوز إلى بيروفاتين
ويعطي 2 \omterm{def:b1:water-small-molecules:atp}{ATP} و2 NADH. \emph{وأكسدة البيروفات \omterm{def:b1:respiration-fermentation:krebs}{ودورة كريبس}}، في \omterm{def:b1:eukaryotic-cell:mitochondrion}{حشوة
المتقدرة}، تؤكسدان البيروفات إلى $\mathrm{CO_2}$، فتحملان الإلكترونات
على 8 NADH و2 $\mathrm{FADH_2}$ وتصنعان 2 GTP. \emph{والفسفرة
المؤكسدة}، في الغشاء المتقدري الداخلي، تمرر تلك الإلكترونات إلى
الأكسجين عبر سلسلة حوامل تضخ البروتونات، ويقود تدرج البروتونات تركيب
نحو 26 \omterm{def:b1:water-small-molecules:atp}{ATP}. وفي المجموع نحو 30 \omterm{def:b1:water-small-molecules:atp}{ATP} لكل غلوكوز: أي نحو نصف \omterm{prop:b1:water-small-molecules:gibbs}{الطاقة
الحرة} للاحتراق ملتقطًا، والباقي مطلقًا حرارةً.
\end{proposition}

\begin{definition}[حوامل الإلكترونات]\label{def:b1:respiration-fermentation:carriers}
\index{NAD}\index{FAD}
\emph{$\mathrm{NAD^+}$} (ثنائي \omterm{def:b1:nucleic-acids:nucleotide}{نوكليوتيد} النيكوتيناميد والأدنين) يقبل
إلكترونين وبروتونًا واحدًا من ركيزة فيصير NADH؛ و\emph{FAD} يقبل
إلكترونين وبروتونين فيصير $\mathrm{FADH_2}$. وكلاهما \omterm{def:b1:enzymes:enzyme}{مرافق إنزيمي}
لنازعات الهيدروجين، وكلاهما يُعاد تدويره: يُرجعه الهدم، وتعيد \omterm{def:b1:respiration-fermentation:chain}{سلسلة
التنفس} أكسدته. \omterm{def:b1:cell-unit-of-life:cell}{والخلية} لا تحمل منهما إلا ميكرومولات، تدور آلاف المرات
في الساعة. وكمونات أكسدتهما وإرجاعهما، $E'_0 = \qty{-0.32}{V}$ عند
$\mathrm{NAD^+}/\mathrm{NADH}$، تضعهما دون الأكسجين ($\qty{+0.82}{V}$)
بكثير: فكل زوج إلكترونات يسلمه NADH إلى الأكسجين يحرر
$\Delta G^{\circ\prime} = -2F\Delta E =
\qty{-220}{kJ/mol}$، وهو يكفي
لأكثر من أربعة \omterm{def:b1:water-small-molecules:atp}{ATP}.
\end{definition}

\section{التحلل السكري}

\begin{definition}[التحلل السكري]\label{def:b1:respiration-fermentation:glycolysis}
\index{التحلل السكري}
\emph{التحلل السكري} تتابع من عشرة تفاعلات إنزيمية في \omterm{def:b1:eukaryotic-cell:organelle}{العصارة الخلوية}
يحول غلوكوزًا واحدًا (C6) إلى \emph{بيروفاتين} (C3). ففي \emph{طور
الاستثمار} يُنفق \omterm{def:b1:water-small-molecules:atp}{ATP} اثنان لفسفرة السكر (الهكسوكيناز، ثم
\emph{الفسفوفركتوكيناز}، PFK) ويُشطر ثنائي الفوسفات السداسي إلى
فوسفاتين ثلاثيي الكربون. وفي \emph{طور المردود} يُؤكسد كل سكر ثلاثي
بجزيء $\mathrm{NAD^+}$ (وهي الأكسدة الوحيدة في التحلل السكري) وتُنقل
فوسفاته إلى ADP \emph{بفسفرة على مستوى الركيزة}\index{الفسفرة على مستوى الركيزة}
--- أي فوسفات يُسلَّم مباشرة من وسيط غني بالطاقة إلى ADP، بلا غشاء
ولا أكسجين. والميزان:
\[
\text{غلوكوز} + 2\,\mathrm{NAD^+} + 2\,\mathrm{ADP} + 2\,\mathrm{P_i}
\to 2\,\text{بيروفات} + 2\,\mathrm{NADH} + 2\,\mathrm{H^+} + 2\,\mathrm{ATP} + 2\,\mathrm{H_2O} .
\]
وهو لا يحتاج إلى أكسجين؛ وهو أقدم سبل الاستقلاب وأعمّها.
\end{definition}

\begin{proposition}[التحكم عند الفسفوفركتوكيناز]\label{prop:b1:respiration-fermentation:pfk}
PFK، وهو أول خطوة لاعكوسة ملتزمة \omterm{def:b1:respiration-fermentation:glycolysis}{بالتحلل السكري}، \omterm{def:b1:enzymes:allosteric}{إنزيم ألوستيري}
(\cref{ch:b1:enzymes}) يثبطه \omterm{def:b1:water-small-molecules:atp}{ATP} والسترات --- وهما إشارتا وفرة الطاقة
ووقود \omterm{def:b1:respiration-fermentation:krebs}{دورة كريبس} --- وينشطه AMP وADP، وهما إشارتا نقصهما. فحين يرتفع
\omterm{def:b1:water-small-molecules:atp}{ATP} يهبط التدفق عبر \omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري} في ثوان؛ وحين تنفق \omterm{def:b1:cell-unit-of-life:cell}{الخلية} \omterm{def:b1:water-small-molecules:atp}{ATP} يرتفع
AMP ارتفاعًا حادًا (عبر كيناز الأدنيلات،
$2\,\mathrm{ADP} \rightleftharpoons \mathrm{ATP} + \mathrm{AMP}$)،
فينفتح PFK. وهكذا فإن \emph{الشحنة الطاقية} في \omterm{def:b1:cell-unit-of-life:cell}{الخلية} تنظم السبيل الذي
يملؤها.
\end{proposition}

\section{أكسدة البيروفات ودورة كريبس}

\begin{definition}[نازعة هيدروجين البيروفات وأسيتيل-CoA]\label{def:b1:respiration-fermentation:pdh}
يدخل البيروفات \omterm{def:b1:eukaryotic-cell:mitochondrion}{المتقدرة} ويؤكسده معقد \emph{نازعة هيدروجين البيروفات}
إلى \emph{أسيتيل-CoA}\index{أسيتيل-CoA} --- وهي زمرة أسيتيل ثنائية
الكربون يحملها \omterm{def:b1:water-small-molecules:ph}{المرافق} A (\cref{ch:b1:nucleic-acids}) على رابطة
ثيوإستيرية غنية بالطاقة --- فيتحرر $\mathrm{CO_2}$ واحد وNADH واحد.
وأسيتيل-CoA هو المدخل المشترك لكل الوقود إلى الدورة: \omterm{def:b1:carbohydrates:monosaccharide}{الكربوهيدرات} عبر
البيروفات، والشحوم عبر أكسدة $\beta$، وكثير من الحموض الأمينية
مباشرة.
\end{definition}

\begin{definition}[دورة كريبس]\label{def:b1:respiration-fermentation:krebs}
\index{دورة كريبس}\index{دورة حمض الليمون}
\emph{دورة كريبس} (دورة \omterm{def:b1:water-small-molecules:ph}{حمض} الليمون) في \omterm{def:b1:eukaryotic-cell:mitochondrion}{حشوة المتقدرة} تكثّف
\omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA} (C2) مع الأوكسالوأسيتات (C4) في سترات (C6) ثم تؤكسد، في
ثماني خطوات، الكربونين إلى $\mathrm{CO_2}$ وهي تجدد الأوكسالوأسيتات.
ولكل زمرة أسيتيل: 3 NADH، و1 $\mathrm{FADH_2}$، و1 GTP (\omterm{def:b1:respiration-fermentation:glycolysis}{بفسفرة على
مستوى الركيزة})، و2 $\mathrm{CO_2}$. ولكل غلوكوز (أسيتيلان): 6 NADH،
و2 $\mathrm{FADH_2}$، و2 GTP، و4 $\mathrm{CO_2}$ --- وهذه مع
$\mathrm{CO_2}$ الاثنين من نازعة هيدروجين البيروفات تكوّن الستة في
المعادلة الإجمالية. ولا يُستهلك أكسجين في الدورة نفسها؛ وهي لا تجري
إلا ما دامت \omterm{def:b1:respiration-fermentation:chain}{سلسلة التنفس} تعيد أكسدة NADH لها. وهي أيضًا
\emph{مزدوجة الاتجاه}: إذ تُسحب وسائطها للتركيب الحيوي
(\cref{ch:b1:biosyntheses-integration}) وتُعوَّض من الحموض الأمينية
والبيروفات.
\end{definition}

\begin{example}[تتبع الكربونات]\label{ex:b1:respiration-fermentation:carbons}
أطعم \omterm{def:b1:cell-unit-of-life:cell}{خلية} غلوكوزًا موسومًا بذرة $\mathrm{^{14}C}$ على الكربون 1: فيظهر
الوسم في كربون الميثيل في البيروفات، ثم في \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA}، ثم يدخل
السترات، ولا يغادر في هيئة $\mathrm{CO_2}$ إلا في الدورة الثانية أو
الثالثة --- فالكربونان المحرران في دورة ما ينتميان إلى أوكسالوأسيتات
الدورة السابقة. وقد استنتج كريبس (1937) الدورة من ملاحظة أن مقادير
حفزية من أي من حموضها تنشّط أكسدة بيروفات أكثر بكثير مما تستطيع أن
تفسره وحدها: فقد كانت تُجدَّد.
\end{example}

\section{الفسفرة المؤكسدة}

\begin{definition}[سلسلة التنفس]\label{def:b1:respiration-fermentation:chain}
\index{سلسلة التنفس}\index{سلسلة نقل الإلكترونات}
\emph{سلسلة التنفس} أربعة معقدات بروتينية في الغشاء المتقدري الداخلي
وحاملان متنقلان. فجزيء NADH يعطي إلكتروناته \emph{للمعقد I}، ويعطي
$\mathrm{FADH_2}$ إلكتروناته (عبر نازعة هيدروجين السكسينات، وهي
\emph{المعقد II})؛ وكلاهما يرجع \emph{اليوبيكينون} (Q)، وهو كينون
منحل في \omterm{def:b1:lipids:fattyacid}{الدهن} ينتشر في الغشاء؛ وQ يمررها إلى \emph{المعقد III}، الذي
يسلمها إلى \emph{السيتوكروم $c$}، وهو \omterm{def:b1:proteins:peptide}{بروتين} صغير على الوجه الخارجي؛
و\emph{المعقد IV} (أكسيداز السيتوكروم) يوصلها، أربعةً أربعة، إلى
$\mathrm{O_2}$، فيصنع الماء. وعند كل من المعقدات I وIII وIV تهبط
الإلكترونات في الكمون وتضخ الطاقة البروتونات من \omterm{def:b1:eukaryotic-cell:mitochondrion}{الحشوة} إلى الحيز بين
الغشاءين: نحو 4 و4 و2 لكل زوج إلكترونات، أي عشرة لكل NADH وستة لكل
$\mathrm{FADH_2}$ (الذي يدخل تحت المعقد I).
\end{definition}

\begin{theorem}[الاقتران التناضحي الكيميائي]\label{thm:b1:respiration-fermentation:chemiosmosis}
\index{النظرية التناضحية الكيميائية}\index{القوة المحركة للبروتون}
تدرج البروتونات الذي تبنيه السلسلة --- وهو نحو \qty{0.16}{V} من \omterm{def:b1:membranes-transport:potential}{كمون
الغشاء} (\omterm{def:b1:eukaryotic-cell:mitochondrion}{والحشوة} سالبة) زائد وحدة pH واحدة، أي \emph{قوة محركة
للبروتون} تناهز \qty{0.22}{V}، أو \qty{21}{kJ} لكل مول بروتونات ---
هو الصلة الوحيدة بين نقل الإلكترونات وتركيب \omterm{def:b1:water-small-molecules:atp}{ATP}. \omterm{thm:b1:photosynthesis:chemiosmosis}{وسنتاز ATP}، وهو محرك
دوّار في الغشاء الداخلي، يمرر نحو \qty{3}{H^+} لكل \omterm{def:b1:water-small-molecules:atp}{ATP} يُصنع، ويُنفق
واحد آخر في استيراد الفوسفات وتصدير \omterm{def:b1:water-small-molecules:atp}{ATP}: أي 4 لكل \omterm{def:b1:water-small-molecules:atp}{ATP}. ومن هنا
\emph{نسبة P/O}\index{نسبة P/O}: $10/4 = 2.5$ \omterm{def:b1:water-small-molecules:atp}{ATP} لكل NADH
و$6/4 = 1.5$ لكل $\mathrm{FADH_2}$.
\end{theorem}
\begin{proof}[شواهد]
لا تصنع \omterm{def:b1:eukaryotic-cell:mitochondrion}{المتقدرات} \omterm{def:b1:water-small-molecules:atp}{ATP} إلا حين يكون غشاؤها الداخلي سليمًا مغلقًا؛
والشظايا التي تنقل الإلكترونات ولا تستطيع إمساك تدرج لا تصنع شيئًا.
وفواصل الاقتران، مثل ثنائي نتروفينول، التي تنقل البروتونات عبر
الغشاء، تلغي تركيب \omterm{def:b1:water-small-molecules:atp}{ATP} بينما يستمر استهلاك الأكسجين بل ويتسارع،
فتغادر الطاقة حرارةً --- \omterm{def:b1:lipids:triglyceride}{والشحم} البني يفعل هذا عمدًا \omterm{def:b1:proteins:peptide}{ببروتين} فصل
اقتران خاص به (\cref{ch:b1:mammal-organization})، وقد بيع ثنائي
نتروفينول يومًا دواءً للتنحيف فقتل بفرط الحرارة. ومثبطات السلسلة
(السيانيد على المعقد IV) توقف الأمرين؛ ومثبطات السنتاز (الأوليغوميسين)
توقف تركيب \omterm{def:b1:water-small-molecules:atp}{ATP}، ثم توقف التنفس أيضًا لعجز التدرج عن التفريغ، ويعيده
فاصل اقتران عندئذ. وتدرج pH اصطناعي مفروض على \omterm{def:b1:eukaryotic-cell:mitochondrion}{متقدرات} في الظلام يقود
تركيب \omterm{def:b1:water-small-molecules:atp}{ATP}، كما في \omterm{def:b1:eukaryotic-cell:plastid}{البلاستيدات الخضراء} (\cref{ch:b1:photosynthesis}).
\end{proof}

\begin{method}[ميزانية ATP لغلوكوز واحد]\label{met:b1:respiration-fermentation:balance}
\begin{enumerate}
  \item \omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري}: $+2$ \omterm{def:b1:water-small-molecules:atp}{ATP} (صافيًا)، و$+2$ NADH في \omterm{def:b1:eukaryotic-cell:organelle}{العصارة
        الخلوية}.
  \item نازعة هيدروجين البيروفات: $+2$ NADH. \omterm{def:b1:respiration-fermentation:krebs}{ودورة كريبس}: $+6$ NADH،
        و$+2$ $\mathrm{FADH_2}$، و$+2$ GTP ($=$ \omterm{def:b1:water-small-molecules:atp}{ATP}).
  \item الفسفرة المؤكسدة: $2.5$ \omterm{def:b1:water-small-molecules:atp}{ATP} لكل NADH متقدري (أي 8: 20 \omterm{def:b1:water-small-molecules:atp}{ATP})،
        و$1.5$ لكل $\mathrm{FADH_2}$ (أي 2: 3 \omterm{def:b1:water-small-molecules:atp}{ATP}).
  \item NADH العصاري لا يستطيع عبور الغشاء الداخلي؛ وإلكتروناته تدخل
        بمكوك يسلمها إلى $\mathrm{FAD}$ (أي 1.5 لكل واحد، في العضلة
        والدماغ) أو إلى $\mathrm{NAD^+}$ (أي 2.5 لكل واحد، في الكبد
        والقلب): أي 3 أو 5 \omterm{def:b1:water-small-molecules:atp}{ATP}.
  \item المجموع: $2 + 2 + 20 + 3 + 3 = 30$، أو $32$ بالمكوك الأفضل.
        والمردود: $30\times 50/2870 \approx \qty{52}{\%}$ في شروط
        \omterm{def:b1:cell-unit-of-life:cell}{الخلية}.
\end{enumerate}
\end{method}

\section{التخمر}

\begin{definition}[التخمر]\label{def:b1:respiration-fermentation:fermentation}
\index{تخمر}
بلا أكسجين تتوقف \omterm{def:b1:respiration-fermentation:chain}{سلسلة التنفس}، ولا يمكن إعادة أكسدة NADH، وكان
\omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري} ليقف في ثوان لنقص $\mathrm{NAD^+}$. \emph{والتخمر}
يجدده باستعمال البيروفات نفسه متقبلًا للإلكترونات: ففي \emph{التخمر
اللبني} (العضلة، والكريات الحمراء، وبكتيريا اللبن) يُرجَع البيروفات
إلى لاكتات؛ وفي \emph{التخمر الكحولي} (الخميرة، وبعض النباتات) يُنزع
كربوكسيله إلى أسيتألدهيد يُرجَع إلى إيثانول، فيتحرر $\mathrm{CO_2}$.
والمردود مردود \omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري} نفسه: 2 \omterm{def:b1:water-small-molecules:atp}{ATP} لكل غلوكوز، أي جزء من خمسة
عشر من مردود التنفس؛ والكربون لا يكاد يُؤكسد والنواتج ما زالت تحمل
جل طاقة الوقود. أما \emph{التنفس اللاهوائي} --- وهو سلسلة نقل
إلكترونات تنتهي على النترات أو الكبريتات أو الكربونات بدل الأكسجين،
عند كثير من البكتيريا --- فشيء آخر، يعالجه كتاب السنة الثانية.
\end{definition}

\begin{proposition}[أثر باستور]\label{prop:b1:respiration-fermentation:pasteur}
\index{أثر باستور}
\omterm{def:b1:cell-unit-of-life:cell}{الخلية} القادرة على التنفس تستهلك الغلوكوز أسرع بكثير حين تُحرم
الأكسجين منها حين يُوفَّر لها، وتنمو عليه أقل بكثير.
\end{proposition}
\begin{proof}[شواهد]
وجد باستور (1861) خميرة في خزان مهوى تستهلك السكر ببطء وتتكاثر،
بينما استهلكته في خزان محكم أسرع بعشر مرات، وتكاثرت قليلًا، وصنعت
كحولًا. والتفسير هو مردود \omterm{def:b1:water-small-molecules:atp}{ATP}: فللحصول على \omterm{def:b1:water-small-molecules:atp}{ATP} نفسه من 2 لكل غلوكوز
كما من 30، لا بد \omterm{def:b1:cell-unit-of-life:cell}{للخلية} أن تجري \omterm{def:b1:respiration-fermentation:glycolysis}{التحلل السكري} أسرع بخمس عشرة مرة،
وPFK، متحررًا من تثبيط \omterm{def:b1:water-small-molecules:atp}{ATP} والسترات الذي يبقيه التنفس، يتيح لها ذلك.
وعضلة تعدو تظهر الأثر نفسه في ثوان: فغليكوجينها يهبط أسرع بعشرين مرة
مما يهبط في الراحة، وترتفع اللاكتات.
\end{proof}

\begin{definition}[الحاصل التنفسي]\label{def:b1:respiration-fermentation:rq}
\index{الحاصل التنفسي}
\emph{الحاصل التنفسي} RQ هو نسبة $\mathrm{CO_2}$ المنتج إلى
$\mathrm{O_2}$ المستهلك، بالمول. \omterm{def:b1:carbohydrates:monosaccharide}{والكربوهيدرات} تعطي 1.0 (ستة من كل
منهما لكل غلوكوز)؛ \omterm{def:b1:lipids:triglyceride}{والشحم} نحو 0.7 (البالميتات:
$\mathrm{C_{16}H_{32}O_2} + 23\,\mathrm{O_2} \to 16\,\mathrm{CO_2} +
16\,\mathrm{H_2O}$، أي $16/23 = 0.70$)، لأن كربوناته أشد اختزالًا
وتحتاج إلى أكسجين أكثر لكل $\mathrm{CO_2}$؛ \omterm{def:b1:proteins:peptide}{والبروتين} نحو 0.8.
والمزرعة المخمِّرة تطلق $\mathrm{CO_2}$ بلا أخذ أكسجين: فحاصلها
لانهائي، وحاصل مزرعة مختلطة فوق 1 يقيس حصة \omterm{def:b1:respiration-fermentation:fermentation}{التخمر}. ومقيسًا على حيوان
كامل بتحليل الغازات، يخبر الحاصل التنفسي أي وقود يحرقه.
\end{definition}

\begin{example}[قراءة حاصل تنفسي]\label{ex:b1:respiration-fermentation:readrq}
إنسان مستريح بعد وجبة كربوهيدراتية: RQ يساوي 0.95، فهو يحرق الغلوكوز.
والشخص نفسه بعد صيام ليلة: 0.75، فهو \omterm{def:b1:lipids:triglyceride}{الشحم}. وعدّاء ماراثون عند
الكيلومتر الثلاثين، وقد نفد غليكوجينه: 0.72. ودورق خميرة على غلوكوز
بقليل من الهواء: RQ يساوي 4 --- فثلاثة أرباع غلوكوزه يُخمَّر.
\end{example}

\begin{proposition}[الشحوم والبروتينات تدخل الآلة نفسها]\label{prop:b1:respiration-fermentation:otherfuels}
\index{beta@أكسدة $\beta$}
تُنشَّط \omterm{def:b1:lipids:fattyacid}{الحموض الدسمة} إلى أسيل-CoA، وتُقصَّر في \omterm{def:b1:eukaryotic-cell:mitochondrion}{حشوة المتقدرة} كربونين
كربونين \emph{بأكسدة $\beta$}، وكل دورة تعطي \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA} واحدًا،
وNADH واحدًا، و$\mathrm{FADH_2}$ واحدًا: فالبالميتات (C16) يعطي 8
\omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA}، و7 NADH، و7 $\mathrm{FADH_2}$ --- أي نحو 106 \omterm{def:b1:water-small-molecules:atp}{ATP}، أو
\qty{6.6}{} \omterm{def:b1:water-small-molecules:atp}{ATP} لكل كربون مقابل 5 عند الغلوكوز. أما الحموض الأمينية
فتفقد نتروجينها نشادرًا (يُحوَّل إلى بولة في الكبد) وتدخل هياكلها
الكربونية بيروفاتًا أو \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA} أو وسائط \omterm{def:b1:respiration-fermentation:krebs}{لدورة كريبس}. فكل وقود
يلتقي عند \omterm{def:b1:respiration-fermentation:pdh}{أسيتيل-CoA} والسلسلة؛ \omterm{def:b1:cell-unit-of-life:cell}{والخلية} تختار بينها بالهرمونات وبحال
\omterm{def:b1:water-small-molecules:atp}{ATP} لديها، كما يصف \cref{ch:b1:biosyntheses-integration}.
\end{proposition}

\begin{example}[غرام من كل]\label{ex:b1:respiration-fermentation:gramofeach}
غرام واحد من الغلوكوز (\qty{5.6}{mmol}) يعطي نحو \qty{170}{mmol} من
\omterm{def:b1:water-small-molecules:atp}{ATP}؛ وغرام واحد من البالميتات (\qty{3.9}{mmol}) نحو \qty{410}{mmol}
--- أي \qty{38}{kJ/g} مقابل \qty{17}{kJ/g} في \cref{ch:b1:lipids}،
محوَّلة إلى عملة. \omterm{def:b1:lipids:triglyceride}{والشحم} يكلف أكسجينًا أكثر لكل \omterm{def:b1:water-small-molecules:atp}{ATP}، ولهذا تحرق عضلة
عدّاء، وهي محدودة الأكسجين، \omterm{def:b1:carbohydrates:polysaccharide}{الغليكوجين}، ويحرق طائر مهاجر، وهو غني
بالأكسجين محدود الكتلة، \omterm{def:b1:lipids:triglyceride}{الشحم}.
\end{example}

\section{مسألة: خميرة بالهواء وبدونه}

\begin{problem}[{مسألة نهاية الأسبوع --- خزانا باستور من جديد: ATP
معدود، وغلوكوز مستهلك، وكتلة حيوية نامية، وغازات مقيسة، وصولًا إلى
نسبة مردودي ATP ونسبة P/O}]
\label{pb:b1:respiration-fermentation:1}
تُنمّى خميرة على غلوكوز في دورقين متطابقين، أحدهما مهوى والآخر محكم.
استعمل نسبتي P/O 2.5 (عند NADH) و1.5 (عند $\mathrm{FADH_2}$)،
ومكوك $\mathrm{FADH_2}$ من أجل NADH العصاري، ومردود نمو قدره \qty{10.5}{g} من
الكتلة الحيوية الجافة لكل مول \omterm{def:b1:water-small-molecules:atp}{ATP}، وغلوكوزًا عند \qty{180}{g/mol}.

\medskip
\noindent\textbf{الجزء الأول --- \omterm{def:b1:water-small-molecules:atp}{ATP} معدودًا.}
\begin{enumerate}
  \item اذكر \omterm{def:b1:water-small-molecules:atp}{ATP} وGTP وNADH و$\mathrm{FADH_2}$ الناتجة من غلوكوز واحد
        \omterm{def:b1:respiration-fermentation:glycolysis}{بالتحلل السكري} ونازعة هيدروجين البيروفات \omterm{def:b1:respiration-fermentation:krebs}{ودورة كريبس}.
  \item احسب البروتونات التي تضخها السلسلة لكل غلوكوز (10 لكل NADH
        متقدري، و6 لكل $\mathrm{FADH_2}$؛ وNADH العصاري يدخل
        $\mathrm{FADH_2}$).
  \item احسب \omterm{def:b1:water-small-molecules:atp}{ATP} الذي يصنعه السنتاز عند \qty{4}{H^+} لكل \omterm{def:b1:water-small-molecules:atp}{ATP}، ومجموع
        \omterm{def:b1:water-small-molecules:atp}{ATP} لكل غلوكوز مع الهواء.
  \item احسب \omterm{def:b1:water-small-molecules:atp}{ATP} لكل غلوكوز بلا هواء.
  \item احسب نسبة المردودين.
  \item احسب \omterm{prop:b1:water-small-molecules:gibbs}{الطاقة الحرة} الملتقطة مع الهواء
        ($\Delta G_{\text{ATP}}
        = \qty{-50}{kJ/mol}$) بوصفها كسرًا من
        \qty{2870}{kJ}، وبلا هواء بوصفها كسرًا من \qty{2870}{kJ} ومن
        الطاقة التي يحررها \omterm{def:b1:respiration-fermentation:fermentation}{التخمر} فعلًا (غلوكوز إلى إيثانولين
        و$\mathrm{CO_2}$ اثنين: \qty{-235}{kJ/mol}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الغلوكوز المستهلك والكتلة النامية.}
على الدورق المحكم أن يصنع \omterm{def:b1:water-small-molecules:atp}{ATP} بالمعدل نفسه الذي يصنعه المهوى ليصون
خلاياه: أي \qty{30}{mmol} من \omterm{def:b1:water-small-molecules:atp}{ATP} في الساعة.
\begin{enumerate}[resume]
  \item احسب الغلوكوز المستهلك في الساعة في كل دورق.
  \item احسب الإيثانول و$\mathrm{CO_2}$ المنتجين في الساعة في الدورق
        المحكم، و$\mathrm{CO_2}$ و$\mathrm{O_2}$ المتبادلين في المهوى.
  \item احسب الكتلة الحيوية التي يستطيع الدورق المحكم بناءها لكل مول
        غلوكوز، ولكل غرام غلوكوز.
  \item الدورق المهوى يبني \qty{0.5}{g} من الكتلة الحيوية لكل غرام
        غلوكوز. احسب \omterm{def:b1:water-small-molecules:atp}{ATP} الذي يمثله هذا عند \qty{10.5}{g/mol}، وقارنه
        بالثلاثين \omterm{def:b1:water-small-molecules:atp}{ATP} المتاحة. فما الذي يحد المردود الهوائي؟
  \item فسّر \omterm{prop:b1:respiration-fermentation:pasteur}{أثر باستور} من السؤالين 7 و9، وسمِّ \omterm{def:b1:enzymes:enzyme}{الإنزيم} الذي ينفذه
        تنظيمه.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الغازات مقيسة.}
دورق ثالث، رديء التهوية، يطلق \qty{44}{mg} من $\mathrm{CO_2}$
ويستهلك \qty{20}{mg} من $\mathrm{O_2}$ في الساعة.
\begin{enumerate}[resume]
  \item حوّل الكميتين إلى مليمولات واحسب \omterm{def:b1:respiration-fermentation:rq}{الحاصل التنفسي}.
  \item ليكن $x$ الغلوكوز المتنفَّس و$y$ الغلوكوز المخمَّر في الساعة.
        اكتب ميزاني $\mathrm{O_2}$ و$\mathrm{CO_2}$ وحل من أجل $x$
        و$y$.
  \item أي كسر من الغلوكوز يُخمَّر؟ وأي كسر من \omterm{def:b1:water-small-molecules:atp}{ATP} يأتي من \omterm{def:b1:respiration-fermentation:fermentation}{التخمر}؟
  \item الدورق نفسه على البالميتات بدل الغلوكوز (هوائيًا): احسب
        \omterm{def:b1:respiration-fermentation:rq}{الحاصل التنفسي}.
  \item يقيس طالب حاصلًا تنفسيًا قدره 0.7 عند حيوان مستريح بعد صيام
        ليلة و1.0 بعد وجبة. فسّر.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- العنفة.}
\begin{enumerate}[resume]
  \item احسب \omterm{prop:b1:water-small-molecules:gibbs}{الطاقة الحرة} لمول واحد من البروتونات يعبر قوة محركة
        للبروتون قدرها \qty{0.22}{V}.
  \item احسب طاقة \qty{4}{H^+} ومردود السنتاز عند \qty{50}{kJ} لكل
        \omterm{def:b1:water-small-molecules:atp}{ATP}.
  \item احسب \omterm{prop:b1:water-small-molecules:gibbs}{الطاقة الحرة} المحررة بهبوط إلكترونين من NADH إلى
        $\mathrm{O_2}$ (\qty{1.14}{V})، والطاقة المخزونة في
        البروتونات العشرة المضخوخة، ومردود السلسلة.
  \item اجمع المردودين وقارن مع 2.5 \omterm{def:b1:water-small-molecules:atp}{ATP} $\times$ \qty{50}{kJ} على
        \qty{220}{kJ}.
  \item يُضاف فاصل اقتران إلى الدورق المهوى. تنبأ بالتغيرات في
        استهلاك الأكسجين، ومردود \omterm{def:b1:water-small-molecules:atp}{ATP}، واستهلاك الغلوكوز، وإنتاج
        الحرارة، والنمو.
  \item ويُضاف الأوليغوميسين، الذي يسد السنتاز، بدلًا منه. تنبأ
        بالتغيرات، وبما يحدث إن أُضيف فاصل اقتران فوقه.
  \item خميرة طافرة تفتقر إلى المعقد I؛ وNADH لديها يدخل السلسلة عبر
        \omterm{def:b1:enzymes:enzyme}{إنزيم} بديل لا يضخ بروتونات. احسب \omterm{thm:b1:respiration-fermentation:chemiosmosis}{نسبة P/O} عند NADH فيها \omterm{def:b1:water-small-molecules:atp}{وATP}
        لكل غلوكوز.
  \item فسّر لماذا تبقي \omterm{def:b1:cell-unit-of-life:cell}{الخلية} مخزون NADH المتقدري مرجعًا كله تقريبًا
        حين ينعدم الأكسجين، وماذا يفعل هذا \omterm{def:b1:respiration-fermentation:krebs}{بدورة كريبس}.
  \item اذكر النتيجة: \omterm{def:b1:water-small-molecules:atp}{ATP} لكل غلوكوز مع الهواء وبدونه، ونسبتهما،
        ونسبتا P/O عند NADH وعند $\mathrm{FADH_2}$.
\end{enumerate}
\end{problem}
