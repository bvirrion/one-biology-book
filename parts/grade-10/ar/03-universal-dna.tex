\chapter{DNA: جزيء وراثي واحد للحياة كلها}\label{ch:g10:universal-dna}

في مختبر معتم، تحت مصباح أزرق، يتوهج أحد فأرين أبيضين بالأخضر من
أنفه إلى ذنبه. إنه يحمل، في كل \omterm{def:g10:cells-common-unit:cell}{خلية} من خلاياه، \omterm{def:g10:universal-dna:gene}{مورّثة} واحدة أخذت من
قنديل بحر يضيء في المحيط الهادئ --- \omterm{def:g10:cells-common-unit:cell}{وخلايا} الفأر تقرأ \omterm{def:g10:universal-dna:gene}{مورّثة} قنديل
البحر تلك كما تقرؤها \omterm{def:g10:cells-common-unit:cell}{خلايا} القنديل نفسها بالضبط. وتنجح التجربة لأن كل
كائن حي يكتب مورّثاته في الجزيء نفسه، بالأحرف الأربعة نفسها، ويقرؤها
بالقواعد نفسها. وهذا الفصل عن ذلك الجزيء، \omterm{def:g10:universal-dna:information}{DNA}: كيف بني، وما الذي
تفسره بنيته، وما الذي يثبته الفأر المتوهج.

\section{المادة الوراثية}

\begin{definition}[المعلومة الوراثية]\label{def:g10:universal-dna:information}
\emph{المعلومة الوراثية}\index{معلومة وراثية} في \omterm{def:g10:cells-common-unit:cell}{خلية} ما هي مجموع
التعليمات، المنتقلة من \omterm{def:g10:cells-common-unit:cell}{الخلية} إلى خليتيها البنتين ومن الآباء إلى
الأبناء، التي تحدد صفات الكائن \omterm{def:g10:chemistry-of-life:families}{والبروتينات} التي تستطيع خلاياه صنعها.
وتحملها في كل كائن حي جزيئات \emph{DNA}\index{DNA} (\omterm{def:g10:chemistry-of-life:families}{الحمض النووي}
الريبي منقوص الأكسجين)، محزومة مع بروتينات في
\emph{الصبغيات}\index{صبغي} --- في \omterm{def:g10:cells-common-unit:organelle}{نواة} \omterm{def:g10:cells-common-unit:prokaryote}{الخلية الحقيقية النواة}، وحرة
في \omterm{def:g10:cells-common-unit:cell}{سيتوبلازم} البكتيريا.
\end{definition}

\begin{proposition}[DNA هو جزيء الوراثة]\label{prop:g10:universal-dna:material}
المادة الوراثية في \omterm{def:g10:cells-common-unit:cell}{الخلية} هي \omterm{def:g10:universal-dna:information}{DNA} فيها، لا بروتيناتها ولا أي مكون آخر:
فنقل \omterm{def:g10:universal-dna:information}{DNA} منقى من سلالة بكتيريا إلى أخرى ينقل صفات السلالة الأولى.
\end{proposition}
\begin{proof}[شواهد]
وجد غريفيث (1928) أن بكتيريا ذات الرئة غير المؤذية، إذا خلطت \omterm{def:g10:cells-common-unit:cell}{بخلايا}
مقتولة من سلالة قاتلة، صارت قاتلة وبقيت كذلك عبر الأجيال: فشيء من
\omterm{def:g10:cells-common-unit:cell}{الخلايا} الميتة قد «حوّلها». ثم فصل آفري وماكليود وماكارتي (1944)
مستخلص \omterm{def:g10:cells-common-unit:cell}{الخلايا} الميتة إلى عائلات جزيئاته واختبر كلًا منها: فلم يحوّل
إلا جزء \omterm{def:g10:universal-dna:information}{DNA}؛ ومعالجة المستخلص بإنزيم يهدم \omterm{def:g10:universal-dna:information}{DNA} ألغت الأثر، بينما لم
تلغه إنزيمات تهدم \omterm{def:g10:chemistry-of-life:families}{البروتينات} أو RNA. فالمادة المحوِّلة، ومن ثم حاملة
الصفة الموروثة، هي \omterm{def:g10:universal-dna:information}{DNA}.
\end{proof}

\begin{example}[أين يقع DNA]\label{ex:g10:universal-dna:where}
صباغ خاص بجزيء \omterm{def:g10:universal-dna:information}{DNA} يلون \omterm{def:g10:cells-common-unit:organelle}{نواة} \omterm{def:g10:cells-common-unit:cell}{خلية} الخد ولا يلون شيئًا آخر في \omterm{def:g10:cells-common-unit:cell}{السيتوبلازم}؛
وفي \omterm{def:g10:cells-common-unit:cell}{خلية} منقسمة يلون \omterm{def:g10:universal-dna:information}{الصبغيات} المكتنزة. وتأخذ البكتيريا الصباغ في
منطقة مركزية بلا غلاف. وفي الحالتين تتضاعف كمية \omterm{def:g10:universal-dna:information}{DNA} قبل كل انقسام
وتنصف بين الخليتين البنتين --- وهو السلوك المنتظر من معلومة يجب أن
تنسخ وتتقاسم.
\end{example}

\section{بنية DNA}

\begin{definition}[النوكليوتيد]\label{def:g10:universal-dna:nucleotide}
\omterm{def:g10:universal-dna:information}{DNA} سلسلة من \emph{النوكليوتيدات}\index{نوكليوتيد}. وكل نوكليوتيد
مؤلف من ثلاثة أجزاء: زمرة فوسفاتية، وسكر (الريبوز منقوص الأكسجين)،
وواحدة من أربع \emph{قواعد}\index{قاعدة (DNA)} آزوتية: الأدنين (A)،
والتيمين (T)، والغوانين (G)، والسيتوزين (C). ويتناوب السكر والفوسفات
فيكوّنان العمود الفقري للسلسلة؛ وتبرز القواعد منه، واحدة لكل
نوكليوتيد. ولا تختلف النوكليوتيدات الأربعة إلا بقاعدتها، فيوصف شريط
\omterm{def:g10:universal-dna:information}{DNA} بمتتالية قواعده، مكتوبة سلسلة من الأحرف:
\texttt{ATGGCTTAC\dots}
\end{definition}

\begin{proposition}[اللولب المزدوج]\label{prop:g10:universal-dna:helix}
جزيء \omterm{def:g10:universal-dna:information}{DNA} مؤلف من شريطين ملتفين أحدهما حول الآخر في \emph{لولب
مزدوج}\index{لولب مزدوج} يمنة عرضه نحو \qty{2}{nm}، بدورة كاملة كل
عشرة أزواج قواعد (\qty{3.4}{nm}). ويسير الشريطان في اتجاهين متعاكسين
ويتواجهان بقواعدهما، وهي تتزاوج وفق قاعدة صارمة من
\emph{التتام}\index{تتام}: فA لا تزاوج إلا T، وG لا تزاوج إلا C.
ومتتالية أحد الشريطين تثبّت إذن متتالية الآخر.
\end{proposition}
\begin{proof}[شواهد]
قاس شارغاف (1950) تركيب القواعد في \omterm{def:g10:universal-dna:information}{DNA} من أنواع كثيرة فوجد في كل
منها من A بقدر ما فيه من T، ومن G بقدر ما فيه من C، بينما تفاوتت نسبة
A+T إلى G+C من نوع إلى نوع --- أي قاعدة تزاوج، لا مصادفة كيميائية. ثم
بينت صور فرانكلين بالأشعة السينية لألياف \omterm{def:g10:universal-dna:information}{DNA} (1952) بصمة لولب قطره
\qty{2}{nm} وخطوته \qty{3.4}{nm}. وبنى واطسون وكريك (1953) النموذج
الوحيد الذي يوافق الأمرين: عمودان خارجيان، وقواعد مزدوجة في الداخل،
وزوجا A--T وG--C لهما العرض نفسه فيكون اللولب منتظمًا.
\end{proof}

\begin{example}[قاعدة شارغاف بالأرقام]\label{ex:g10:universal-dna:chargaff}
\omterm{def:g10:universal-dna:information}{DNA} البشري: A 30.9\%، وT 29.4\%، وG 19.9\%، وC 19.8\% ---
أي A$\,\approx\,$T وG$\,\approx\,$C في حدود دقة القياس، وA+T
$= 60\%$. و\emph{E.~coli}: A 24.7\%، وT 23.6\%، وG 26.0\%، وC 25.7\%،
وA+T $= 48\%$. والخميرة: A+T $= 64\%$. فإذا حوت عينة \omterm{def:g10:universal-dna:information}{DNA} 32\% من A،
تنبأ \omterm{prop:g10:universal-dna:helix}{التتام} بنسبة 32\% من T وبنسبة $(100 - 64)/2 = 18\%$ من كل من G وC.
\end{example}

\begin{method}[كتابة الشريط المتمم]\label{met:g10:universal-dna:complement}
إذا أعطي شريط، فاكتب تحت كل قاعدة شريكتها (A$\leftrightarrow$T،
G$\leftrightarrow$C)، ثم اقرأ الناتج في الاتجاه المعاكس إن اقتضى ذلك
اصطلاح التمرين. فمقابل \texttt{5'-ATGGCTTAC-3'} يكون الشريط الشريك،
مقروءًا في اتجاهه الخاص، \texttt{5'-GTAAGCCAT-3'}. أما العد: فالشريط
الذي فيه $n_\mathrm{A}$ من الأدنين يواجه شريطًا فيه $n_\mathrm{A}$ من
التيمين، ومنه في الجزيء كله $n_\mathrm{A} = n_\mathrm{T}$
و$n_\mathrm{G} = n_\mathrm{C}$.
\end{method}

\begin{proposition}[ما الذي تفسره البنية]\label{prop:g10:universal-dna:explains}
تتبع من البنية خاصتان للمادة الوراثية.
\begin{itemize}
  \item \emph{المعلومة}: ترتيب القواعد على شريط لا تقيده الكيمياء ---
        فكل متتالية ممكنة --- فتستطيع المتتالية أن تحمل رسالة، كما
        يحمل ترتيب الأحرف نصًا. \omterm{def:g10:cells-common-unit:cell}{والخلية} البشرية تحوي $\num{3.2e9}$
        زوج قواعد في كل طقم من \omterm{def:g10:universal-dna:information}{الصبغيات}.
  \item \emph{النسخ}: لما كان كل شريط يحدد الآخر، أمكن مضاعفة الجزيء
        بفصل الشريطين وبناء شريك جديد على كل منهما. وآلية ذلك موضوع
        \cref{ch:g11:dna-replication}.
\end{itemize}
\end{proposition}
\admitted

\section{المورّثات والأليلات والصبغيات}

\begin{definition}[المورّثة والأليل]\label{def:g10:universal-dna:gene}
\emph{المورّثة}\index{مورّثة} قطعة من جزيء \omterm{def:g10:universal-dna:information}{DNA} --- طولها عادة بضعة
آلاف من أزواج القواعد --- تحمل متتاليتها تعليمات صنع \omterm{def:g10:chemistry-of-life:families}{بروتين} واحد (أو،
أحيانًا، جزيء RNA عامل واحد). وهي تشغل موضعًا ثابتًا على \omterm{def:g10:universal-dna:information}{صبغي} معين.
أما نسخ المورّثة التي تختلف ببضع قواعد فهي \emph{أليلاتها}\index{أليل}:
الموضع نفسه، والوظيفة نفسها، ومتتالية مختلفة قليلًا، ومنه \omterm{def:g10:chemistry-of-life:families}{بروتين} مختلف
قليلًا في الغالب.
\end{definition}

\begin{example}[أرقام]\label{ex:g10:universal-dna:numbers}
يحمل المجين البشري --- أي طقم كامل من \omterm{def:g10:universal-dna:information}{الصبغيات} --- $\num{3.2e9}$ زوج
قواعد ونحو $\num{20000}$ \omterm{def:g10:universal-dna:gene}{مورّثة}؛ ولا تشغل \omterm{def:g10:universal-dna:gene}{المورّثات} إلا بضعة بالمئة
من المتتالية. وتحمل \omterm{def:g10:cells-common-unit:cell}{خلية} \emph{E.~coli} جزيء \omterm{def:g10:universal-dna:information}{DNA} حلقيًا واحدًا فيه
$\num{4.6e6}$ زوج قواعد ونحو 4300 \omterm{def:g10:universal-dna:gene}{مورّثة}. ويختلف إنسانان لا قرابة
بينهما في نحو قاعدة من كل ألف: أي في نحو ثلاثة ملايين موضع، معظمها
خارج \omterm{def:g10:universal-dna:gene}{المورّثات} ولا أثر له، وقليل منها \omterm{def:g10:universal-dna:gene}{أليلات} تغير صفة.
\end{example}

\begin{proposition}[الصبغيات]\label{prop:g10:universal-dna:chromosome}
\omterm{def:g10:universal-dna:information}{الصبغي} جزيء \omterm{def:g10:universal-dna:information}{DNA} واحد، ملفوف حول بروتينات حازمة ومكوَّر على نفسه.
ولو مُدّ \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:universal-dna:information}{صبغي} بشري واحد لبلغ طوله من \qty{1.5}{cm} إلى
\qty{8.5}{cm}؛ وصبغيات \omterm{def:g10:cells-common-unit:cell}{الخلية} الستة والأربعون تحوي معًا نحو
\qty{2}{m} من \omterm{def:g10:universal-dna:information}{DNA} في \omterm{def:g10:cells-common-unit:organelle}{نواة} عرضها \qty{6}{\micro m}. ولا تُرى \omterm{def:g10:universal-dna:information}{الصبغيات}
أجسامًا منفصلة إلا أثناء انقسام \omterm{def:g10:cells-common-unit:cell}{الخلية}، حين تكون أشد ما تكون تكوّرًا؛
أما بين الانقسامات فهي منشورة تملأ \omterm{def:g10:cells-common-unit:organelle}{النواة} كخيوط متشابكة.
\end{proposition}
\admitted

\section{جزيء واحد لكل الكائنات الحية}

\begin{proposition}[كونية DNA]\label{prop:g10:universal-dna:universal}
تحمل الكائنات الحية جميعًا --- بكتيريا وفطريات ونباتات وحيوانات ---
معلومتها الوراثية \omterm{def:g10:universal-dna:information}{DNA} مبنيًا من \omterm{def:g10:universal-dna:nucleotide}{النوكليوتيدات} الأربعة نفسها، وتقرأ
متتالية \omterm{def:g10:universal-dna:gene}{المورّثة} بالقواعد نفسها. ومن ثم \omterm{def:g10:universal-dna:gene}{فمورّثة} تؤخذ من نوع وتدخل في
\omterm{def:g10:universal-dna:information}{DNA} نوع آخر تقرؤها \omterm{def:g10:cells-common-unit:cell}{خلايا} المتلقي، فتصنع \omterm{def:g10:chemistry-of-life:families}{البروتين} الذي ترمز له:
\emph{فالنقل المورّثي}\index{نقل مورّثي} ممكن، والكائن المنقول إليه
يورّث \omterm{def:g10:universal-dna:gene}{المورّثة} الغريبة نسله كأي \omterm{def:g10:universal-dna:gene}{مورّثة} أخرى.
\end{proposition}
\begin{proof}[شواهد]
أدخلت \omterm{def:g10:universal-dna:gene}{مورّثة} \omterm{def:g10:chemistry-of-life:families}{البروتين} الفلوري الأخضر (GFP) لقنديل البحر
\emph{Aequorea victoria} في بكتيريا وخميرة ونباتات وأسماك وفئران
وأرانب: فصنع كل منها \omterm{def:g10:chemistry-of-life:families}{البروتين} وتوهج بالأخضر تحت الضوء الأزرق، وورثت
الصفة. وأدخلت \omterm{def:g10:universal-dna:gene}{مورّثة} الأنسولين البشرية في \emph{E.~coli} سنة 1978
فصارت البكتيريا تنتج أنسولينًا بشريًا، وهو مصدر الهرمون للسكريين منذ
ذلك الحين. وما من نوع معروف تكتب مورّثاته بكيمياء أخرى.
\end{proof}

\begin{example}[استعمالات النقل المورّثي]\label{ex:g10:universal-dna:uses}
أنسولين بشري وهرمون نمو من بكتيريا؛ \omterm{def:g10:universal-dna:gene}{ومورّثة} مقاومة لحشرة ضارة تدخل في
الذرة أو القطن؛ وأرز يحمل \omterm{def:g10:universal-dna:gene}{مورّثات} لصنع طليعة الفيتامين A؛ \omterm{def:g10:universal-dna:gene}{ومورّثة} GFP
تلحق \omterm{def:g10:universal-dna:gene}{بمورّثة} أخرى فتضيء \omterm{def:g10:cells-common-unit:cell}{الخلايا} التي تنشط فيها تلك \omterm{def:g10:universal-dna:gene}{المورّثة} تحت
المجهر --- وهذا الاستعمال الأخير، وهو أداة بحث، أشيعها جميعًا. ولكل
استعمال أسئلته في السلامة وفي العواقب على البيئة؛ وكل منها يقوم على
الحقيقة نفسها، وهي كونية الجزيء الوراثي.
\end{example}

\begin{remark}[الوحدة والتنوع على المستوى الجزيئي]\label{rem:g10:universal-dna:unity}
كونية \omterm{def:g10:universal-dna:information}{DNA} هي الوجه الجزيئي لوحدة الحياة التي لقيناها في
\cref{ch:g10:chemistry-of-life}: كيمياء واحدة، وجزيء معلومة واحد،
ونظام قراءة واحد لكل كائن على الأرض، وهو ما يتنبأ به أصل مشترك
(\cref{ch:g10:common-ancestry}). أما تنوع الحياة فمكتوب في الجزيء
نفسه أيضًا، اختلافات في المتتالية: بين \omterm{def:g10:universal-dna:gene}{أليلين} بضع قواعد، وبين نوعين
بضعة بالمئة من المجين، وبين بكتيريا وإنسان تنظيم النص كله.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g10:universal-dna:1}
سمِّ أجزاء \omterm{def:g10:universal-dna:nucleotide}{النوكليوتيد} الثلاثة وقواعد \omterm{def:g10:universal-dna:information}{DNA} الأربع.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-dna:2}
اكتب الشريط المتمم للمتتالية \texttt{TACGGATTCA}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-dna:3}
عينة \omterm{def:g10:universal-dna:information}{DNA} فيها 21\% غوانين. أعطِ نسب C وA وT.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-dna:4}
عرّف \omterm{def:g10:universal-dna:gene}{المورّثة} \omterm{def:g10:universal-dna:gene}{والأليل} في جملة واحدة لكل منهما، مستعملًا كلمة
«متتالية».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:universal-dna:5}
ما الكائن المنقول إليه مورّثيًا؟ أعطِ مثالين من الفصل.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-dna:6}
\omterm{def:g10:universal-dna:gene}{مورّثة} طولها 1500 زوج قواعد. ما طولها بالنانومترات، وكم دورة لولب
تصنع؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-dna:7}
احسب الطول الكلي لجزيئات \omterm{def:g10:universal-dna:information}{DNA} في \omterm{def:g10:cells-common-unit:cell}{خلية} بشرية واحدة، علمًا بأن في كل طقم
صبغيات $\num{3.2e9}$ زوج قواعد وأن في \omterm{def:g10:cells-common-unit:cell}{الخلية} طقمين. قارن بقطر \omterm{def:g10:cells-common-unit:organelle}{النواة}،
\qty{6}{\micro m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-dna:8}
في تجربة آفري، لماذا لزم اختبار المستخلص المعالج بإنزيم يهدم \omterm{def:g10:universal-dna:information}{DNA}، لا
جزء \omterm{def:g10:universal-dna:information}{DNA} المنقى وحده؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-dna:9}
فسّر لماذا استبعدت معطيات شارغاف نموذجًا تُرصّ فيه القواعد بترتيب
دوري منتظم مثل \texttt{ATGCATGC\dots}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-dna:10}
أليلان \omterm{def:g10:universal-dna:gene}{لمورّثة} يختلفان بقاعدة واحدة من بين 2000. فسّر كيف يستطيع
اختلاف بهذا الصغر أن يغير بروتينًا، ولماذا قد لا يغير شيئًا كذلك.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:universal-dna:11}
\omterm{def:g10:universal-dna:gene}{مورّثة} GFP المدخلة في بيضة فأر ملقحة توجد في \omterm{def:g10:cells-common-unit:cell}{خلايا} جلده \omterm{def:g10:cells-common-unit:cell}{وخلايا} كبده
\omterm{def:g10:cells-common-unit:cell}{وخلايا} نطافه. فسّر ذلك بالنظرية الخلوية وبما
جاء في \cref{prop:g10:universal-dna:explains}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-dna:12}
جزيء \omterm{def:g10:universal-dna:information}{DNA} في بكتيريا طوله $\num{4.6e6}$ زوج قواعد. احسب طوله؛ وقارنه
بالطول \qty{2}{\micro m} في \omterm{def:g10:cells-common-unit:cell}{الخلية}؛ ثم فسّر كيف يتسع له مع ذلك مكان في
البكتيريا.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-dna:13}
يزعم صديق أن \omterm{def:g10:universal-dna:information}{DNA} البشري لا بد أن يكون «أعقد» من \omterm{def:g10:universal-dna:information}{DNA} بكتيريا لأن فيه

بالعددين $\num{20000}$ و4300 \omterm{def:g10:universal-dna:gene}{مورّثة}، وناقش ما هو \omterm{def:g10:universal-dna:information}{DNA} الزائد وما ليس هو.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-dna:14}

قضايا الفصل يناقض ذلك، وماذا يعني بالنسبة إلى الأصل المشترك بين ذلك

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:universal-dna:15}

\end{exercise}

\section{مسألة: مورّثة قنديل البحر}

\begin{problem}[{مسألة نهاية الأسبوع --- مورّثة واحدة من قنديل بحر
تتبع إلى فأر: طولها، وأحرفها، ولماذا يقدر الفأر على قراءتها، والمترا
DNA اللذان يحملانها في كل خلية}]
\label{pb:g10:universal-dna:1}
\omterm{def:g10:chemistry-of-life:families}{البروتين} الفلوري الأخضر لقنديل البحر \emph{Aequorea victoria} سلسلة
من 238 حمضًا أمينيًا. ومورّثته، كما تستعمل في المختبر، قطعة \omterm{def:g10:universal-dna:information}{DNA} فيها
720 زوج قواعد. خذ \qty{0.34}{nm} لكل زوج قواعد، وعشرة أزواج قواعد لكل
دورة، ومجين فأر فيه $\num{2.7e9}$ زوج قواعد لكل طقم صبغيات، وطقمين في
\omterm{def:g10:cells-common-unit:cell}{الخلية}.

\medskip
\noindent\textbf{الجزء الأول --- الجزيء.}
\begin{enumerate}
  \item احسب طول \omterm{def:g10:universal-dna:gene}{مورّثة} GFP بالنانومترات، وعدد دورات اللولب التي
        تصنعها.
  \item يبدأ أحد شريطي \omterm{def:g10:universal-dna:gene}{المورّثة} هكذا:
        \texttt{ATGAGTAAAGGAGAAGAAC}. اكتب الشريط المتمم.
  \item في \omterm{def:g10:universal-dna:gene}{المورّثة} كلها، يمثل الأدنين 27\% من القواعد. أعطِ نسب
        القواعد الثلاث الأخرى.
  \item في \omterm{def:g10:universal-dna:gene}{المورّثة} 720 زوج قواعد وهي ترمز إلى 238 حمضًا أمينيًا. احسب
        النسبة، واقترح ما توحي به عن عدد القواعد التي تحدد حمضًا
        أمينيًا واحدًا (وهي فرضية سيؤكدها
        \cref{ch:g11:gene-expression}).
  \item كم متتالية مختلفة من 720 قاعدة ممكنة من حيث المبدأ؟ اكتب
        الجواب قوةً للعدد 4، ثم فسّر في جملة واحدة لماذا يمكن أن تسمى
        المتتالية معلومة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- لماذا يقدر الفأر على قراءتها.}
\begin{enumerate}[resume]
  \item اذكر خاصة \omterm{def:g10:universal-dna:information}{DNA} التي تجعل \omterm{def:g10:cells-common-unit:cell}{خلية} فأر قادرة على استعمال \omterm{def:g10:universal-dna:gene}{مورّثة}
        قنديل بحر.
  \item تدخل \omterm{def:g10:universal-dna:gene}{المورّثة} في بيضة ملقحة لا في جلد فأر بالغ. باستعمال
        النظرية الخلوية، فسّر ما الفرق الذي يحدثه ذلك في المكان الذي
        تنتهي إليه \omterm{def:g10:universal-dna:gene}{المورّثة}.
  \item يتوهج الفأر البالغ في كل نسيج من أنسجته. فما الذي يبينه ذلك
        عن محتوى أصناف خلاياه المختلفة من \omterm{def:g10:universal-dna:information}{DNA}؟
  \item يزاوج الفأر المتوهج بفأر عادي. تنبأ، مع التعليل، بما إذا كان
        بعض النسل سيتوهج.
  \item يحقن فأر شاهد \omterm{def:g10:chemistry-of-life:families}{ببروتين} GFP منقى بدل \omterm{def:g10:universal-dna:gene}{المورّثة}: فيتوهج توهجًا
        خافتًا بضعة أيام ثم يتوقف. فسّر الفرق بين التجربتين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الشواهد وراء القصة.}
\begin{enumerate}[resume]
  \item في تجربة آفري، سمِّ الجزء الذي حوّل البكتيريا والمعالجة التي
        ألغت التحويل.
  \item فسّر لماذا كانت تجربة التحويل صورة طبيعية من \omterm{prop:g10:universal-dna:universal}{النقل المورّثي}.
  \item وجد شارغاف A$\,=\,$T وG$\,=\,$C في كل نوع، لكن A+T يتفاوت من
        25\% إلى 75\%. أي المشاهدتين تدعم ازدواج القواعد، وأيهما يبين
        أن المتتالية حرة؟
  \item أعطت صور فرانكلين قطر اللولب \qty{2}{nm}. فسّر لماذا يجعل
        ازدواج قاعدة كبيرة مع صغيرة (A مع T، وG مع C) لا كبيرتين معًا
        اللولب منتظمًا.
  \item لماذا أوحى اكتشاف البنية في الحال بكيفية نسخ \omterm{def:g10:universal-dna:information}{DNA}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- متران في \omterm{def:g10:cells-common-unit:organelle}{نواة}.}
\begin{enumerate}[resume]
  \item احسب الطول الكلي لجزيئات \omterm{def:g10:universal-dna:information}{DNA} في \omterm{def:g10:cells-common-unit:cell}{خلية} فأر واحدة.
  \item عرض \omterm{def:g10:cells-common-unit:organelle}{النواة} \qty{6}{\micro m}. بأي عامل يجب تقصير \omterm{def:g10:universal-dna:information}{DNA} بالتكوّر
        ليتسع فيها؟
  \item ما نسبة \omterm{def:g10:universal-dna:gene}{مورّثة} GFP من \omterm{def:g10:universal-dna:information}{DNA} \omterm{def:g10:cells-common-unit:cell}{الخلية}؟
  \item في الفأر نحو $\num{2e4}$ \omterm{def:g10:universal-dna:gene}{مورّثة} متوسط طولها $\num{3e4}$ زوج
        قواعد بأجزائها غير المرمِّزة. فما نسبة المجين التي تشغلها
        \omterm{def:g10:universal-dna:gene}{المورّثات}؟
  \item اذكر النتيجة في جملة واحدة: كم من \omterm{def:g10:universal-dna:information}{DNA} تحمل \omterm{def:g10:cells-common-unit:cell}{خلية} فأر، وأي
        حقيقة واحدة عن ذلك \omterm{def:g10:universal-dna:information}{DNA} تتيح قراءة \omterm{def:g10:universal-dna:gene}{مورّثة} قنديل بحر واحدة بين
        آلاف مورّثاته؟
\end{enumerate}
\end{problem}
