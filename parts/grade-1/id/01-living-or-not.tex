\chapter{Hidup atau Tak Hidup?}\label{ch:g1:living-or-not}

Seekor siput merayap pelan di jalan setapak yang basah. Di sebelahnya
tergeletak sebutir kerikil kelabu, sama kecilnya, sama bulatnya. Siput
dan kerikil itu berada di bawah hujan yang sama --- namun yang satu
hidup dan yang lain tidak. Dari mana kita tahu? Itulah pertanyaan
paling awal dalam \omterm{def:g1:living-or-not:living}{biologi}, ilmu tentang \omterm{def:g1:living-or-not:living}{makhluk hidup}.

\section{Apa yang dilakukan makhluk hidup}

\begin{definition}[Makhluk hidup]\label{def:g1:living-or-not:living}
Suatu \emph{makhluk hidup}\index{makhluk hidup} adalah sesuatu yang
\emph{hidup}\index{hidup}: ia lahir, ia makan, ia tumbuh, ia dapat
mempunyai anak, dan pada suatu hari ia mati. Hewan dan tumbuhan adalah
makhluk hidup. Biologi\index{biologi} adalah ilmu yang mempelajarinya.
\end{definition}

\begin{example}[Siput dan kerikil]\label{ex:g1:living-or-not:snail}
Siputnya lahir dari sebutir telur kecil. Ia makan daun selada. Ia
tumbuh --- cangkangnya ikut tumbuh. Dapat ada anak-anak siput. Jadi
siputnya adalah \omterm{def:g1:living-or-not:living}{makhluk hidup}. Kerikilnya tidak makan apa pun, sama
sekali tidak tumbuh, dan tidak pernah punya anak kerikil: kerikilnya
tidak hidup.
\end{example}

\begin{proposition}[Tanda-tanda kehidupan]\label{prop:g1:living-or-not:signs}
Suatu \omterm{def:g1:living-or-not:living}{makhluk hidup} menunjukkan tanda-tanda berikut:
\begin{enumerate}
  \item ia \emph{lahir}\index{lahir};
  \item ia \emph{makan}\index{makan} --- ia mengambil makanan atau
        membuatnya sendiri, seperti tumbuhan dengan cahaya;
  \item ia \emph{tumbuh}\index{tumbuh};
  \item ia dapat mempunyai anak --- \omterm{def:g1:living-or-not:living}{makhluk hidup} baru seperti dirinya;
  \item ia \emph{mati}\index{mati} pada suatu hari.
\end{enumerate}
Sesuatu yang tidak menunjukkan satu pun tanda ini tidaklah hidup.
\end{proposition}
\admitted

\begin{remark}[Setiap tanda diperhitungkan]\label{rem:g1:living-or-not:allsigns}
Satu tanda saja tidak cukup. Sebuah mobil ``meminum'' bahan bakar, dan
sebatang es gantung memanjang dalam udara dingin --- tetapi mobilnya
dibuat, bukan dilahirkan, dan es itu tidak punya anak es. Untuk dapat
disebut hidup, sesuatu harus menunjukkan tanda-tanda kehidupan
\emph{bersama-sama}: lahir, makan, tumbuh, membuat \omterm{def:g1:living-or-not:living}{makhluk hidup} baru,
lalu mati.
\end{remark}

\section{Hidup, tak hidup, tak pernah hidup}

\begin{definition}[Benda tak hidup]\label{def:g1:living-or-not:nonliving}
Suatu \emph{benda tak hidup}\index{benda tak hidup} adalah sesuatu
yang tidak hidup. Ada dua macam:
\begin{itemize}
  \item benda yang \emph{tidak pernah hidup}\index{tidak pernah hidup}:
        kerikil, air, angin, sepeda;
  \item benda yang \emph{tidak hidup lagi}\index{tidak hidup
        lagi} atau yang \emph{berasal dari} \omterm{def:g1:living-or-not:living}{makhluk hidup}: daun kering
        di tanah, meja kayu, baju hangat dari wol.
\end{itemize}
\end{definition}

\begin{example}[Meja kayu]\label{ex:g1:living-or-not:table}
Sebuah meja kayu tidak makan dan tidak tumbuh: ia tidak hidup. Tetapi
kayunya dahulu hidup: kayu itu berasal dari sebatang pohon yang lahir
dari biji, tumbuh bertahun-tahun, lalu ditebang. Mejanya tak hidup ---
dan ia berasal dari \omterm{def:g1:living-or-not:living}{makhluk hidup}.
\end{example}

\begin{example}[Bagaimana dengan robot?]\label{ex:g1:living-or-not:robot}
Sebuah robot mainan bergerak, mengeluarkan bunyi, dan lampunya
berkedip seperti mata kecil. Apakah ia hidup? Periksa tanda-tandanya:
ia dibuat di pabrik, bukan dilahirkan; baterainya bukan makanan yang
membuatnya tumbuh; ia tidak akan pernah punya anak robot. Robotnya
bergerak, tetapi ia tidak hidup.
\end{example}

\begin{method}[Cara memutuskan: hidup atau tidak?]\label{met:g1:living-or-not:decide}
Ajukan pertanyaan-pertanyaan kehidupan, satu per satu:
\begin{enumerate}
  \item Apakah ia lahir? Apakah ia makan? Apakah ia tumbuh?
  \item Dapatkah ada anak-anaknya? Akankah ia mati suatu hari?
  \item Jika jawabannya ya: ia hidup.
  \item Jika jawabannya tidak, ajukan satu pertanyaan lagi: apakah ia
        \emph{berasal dari} sesuatu yang hidup, seperti kayu dari pohon?
\end{enumerate}
\end{method}

\begin{remark}[Biji tampak tertidur]\label{rem:g1:living-or-not:seed}
Sebutir biji kacang kering di dalam bungkus tampak sediam kerikil.
Tetapi letakkan ia di atas kapas basah, dan dalam beberapa hari muncul
akar kecil: biji itu hidup selama ini, sedang menunggu. \omterm{def:g1:living-or-not:living}{Makhluk hidup}
yang lambat dan diam mudah dikira \omterm{def:g1:living-or-not:nonliving}{benda tak hidup} --- pertanyaan
kehidupan menolong kita melihat lebih teliti.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:1}
Sebutkan lima tanda kehidupan. Tanda mana saja yang dapat kamu lihat
dilakukan seekor anak kucing hari ini?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:2}
Hidup atau tak hidup? (a) seekor kupu-kupu; (b) sebuah sepeda; (c)
setangkai bunga aster; (d) sebuah awan.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:3}
Sehelai daun kering tergeletak di jalan setapak. Apakah ia hidup
sekarang? Pernahkah ia hidup? Dari mana asalnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:4}
Tomi berkata: ``Sungai itu hidup --- lihatlah, ia bergerak dan ia
bersuara!'' Pakai pertanyaan-pertanyaan \cref{met:g1:living-or-not:decide}
untuk menjawabnya dengan ramah.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:5}
Kelompokkan ke dalam tiga kotak pada gambarnya --- hidup, dahulu
hidup, tak pernah hidup: seekor laba-laba, sebuah kaus kaki wol,
setetes air hujan, sebuah apel di pohon, sebuah buah pinus yang jatuh.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:6}
Nyala lilin ``memakan'' lilinnya, tumbuh makin tinggi, lalu mati
ketika kamu tiup. Apakah nyala itu \omterm{def:g1:living-or-not:living}{makhluk hidup}? Pertanyaan kehidupan
yang mana yang menjawab tidak?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:7}
Sebutkan dua benda di kamar tidurmu yang berasal dari \omterm{def:g1:living-or-not:living}{makhluk hidup},
seperti meja kayu pada \cref{ex:g1:living-or-not:table}. Sebutkan pula
\omterm{def:g1:living-or-not:living}{makhluk hidup} mana asal masing-masing.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:living-or-not:8}
Cobalah: letakkan sebutir kacang kering di atas kapas basah dekat
jendela, dan sebutir kerikil di atas kapas basah lain di sebelahnya.
Amati setiap hari selama seminggu. Mana yang berubah? Apa artinya itu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:9}
Sebuah robot mainan dan seekor siput sama-sama bergerak di lantai.
Berikan dua pertanyaan kehidupan yang jawabannya berbeda bagi keduanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:living-or-not:10}
Jamur tidak berlari seperti hewan dan tidak berwarna hijau seperti
kebanyakan tumbuhan. Dengan tanda-tanda pada
\cref{prop:g1:living-or-not:signs}, jelaskan mengapa jamur tetap
\omterm{def:g1:living-or-not:living}{makhluk hidup}.
\end{exercise}
