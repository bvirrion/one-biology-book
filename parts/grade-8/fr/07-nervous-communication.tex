\chapter{La communication nerveuse}\label{ch:g8:nervous-communication}

\Cref{ch:g5:brain-in-command} traçait la boucle --- signaler,
décider, ordonner --- et en mesurait le cinquième de seconde. Cette
année, nous ouvrons les câbles. Qu'est-ce qui circule vraiment le
long d'un \omterm{def:g8:nervous-communication:neuron}{nerf}~? De quoi les \omterm{def:g5:brain-in-command:system}{nerfs} sont-ils faits~? Et que se
passe-t-il aux espaces où un fil s'arrête et où le suivant commence
--- ces espaces où, on va le voir, la mémoire et bien des poisons
font leur travail~?

\section{Le câblage et ses cellules}

\begin{definition}[Neurones et messages nerveux]\label{def:g8:nervous-communication:neuron}
Les \omterm{def:g5:first-look-at-cells:cell}{cellules} de travail du \omterm{def:g5:brain-in-command:system}{système nerveux} sont les
\emph{neurones}\index{neurone}
(\cref{rem:g6:cell-unit-of-life:twoways} les énumérait)~: chacun a un
corps cellulaire et une longue fibre fine --- certaines fibres vont
de la \omterm{def:g4:classifying-animals:vertebrate}{colonne vertébrale} à l'orteil, les plus longues \omterm{def:g5:first-look-at-cells:cell}{cellules} du
corps. Un \emph{nerf}\index{nerf} est un câble qui regroupe des
milliers de telles fibres. Le long de chaque fibre circulent des
\emph{messages nerveux}\index{message nerveux}~: de brefs signaux
électriques, filant jusqu'à \qty{100}{m/s}
(le chiffre de \cref{def:g5:brain-in-command:system}, muni
maintenant de son porteur).
\end{definition}

\begin{proposition}[Un codage par la fréquence]\label{prop:g8:nervous-communication:code}
Les signaux eux-mêmes sont tous identiques --- ce qui varie, c'est
leur \emph{cadence}. Un contact léger envoie un crépitement lent le
long de la fibre sensitive~; un contact fort, un roulement rapide.
La langue du \omterm{def:g5:brain-in-command:system}{système nerveux} a un seul mot répété à des vitesses
différentes~: l'intensité est codée en fréquence, sur chaque ligne,
à l'aller comme au retour.
\end{proposition}
\admitted

\begin{example}[Trois lignes, une seule langue]\label{ex:g8:nervous-communication:lines}
Les acteurs de la boucle (\cref{prop:g5:brain-in-command:loop}),
recâblés~: les \emph{\omterm{def:g8:nervous-communication:neuron}{neurones} sensitifs} portent les rapports ---
de la \omterm{def:g1:the-five-senses:sense}{peau} à la \omterm{def:g5:brain-in-command:system}{moelle épinière}, de l'\omterm{def:g1:the-five-senses:sense}{œil} au \omterm{def:g5:brain-in-command:system}{cerveau}~; les
\emph{\omterm{def:g8:nervous-communication:neuron}{neurones} moteurs} portent les ordres --- de la moelle au
\omterm{def:g3:bones-and-muscles:muscle}{muscle}, en se terminant là où la fibre rencontre le \omterm{def:g3:bones-and-muscles:muscle}{muscle} et où son
roulement fixe la force de la contraction
(\cref{def:g3:bones-and-muscles:muscle})~; et entre les deux, dans
le \omterm{def:g5:brain-in-command:system}{cerveau} et la moelle, d'innombrables \emph{\omterm{def:g8:nervous-communication:neuron}{neurones}
d'association} font le travail de décision. Réflexe ou symphonie, le
trafic est fait des mêmes signaux à des cadences différentes sur des
lignes différentes.
\end{example}

\section{Les espaces~: les synapses}

\begin{definition}[La synapse]\label{def:g8:nervous-communication:synapse}
Les \omterm{def:g8:nervous-communication:neuron}{neurones} ne se touchent pas. Là où le bout d'une fibre rencontre
la \omterm{def:g5:first-look-at-cells:cell}{cellule} suivante s'ouvre un espace microscopique --- la
\emph{synapse}\index{synapse}. Le signal électrique ne peut pas le
franchir~; à la place, la fibre qui arrive libère un
\emph{messager chimique} dans l'espace~; celui-ci le traverse,
s'amarre de l'autre côté et --- s'il en arrive assez --- relance le
signal dans le \omterm{def:g8:nervous-communication:neuron}{neurone} suivant. Électrique le long du fil, chimique
à chaque jonction~: tout le système alterne les deux.
\end{definition}

\begin{proposition}[Pourquoi des espaces, et non des soudures]\label{prop:g8:nervous-communication:whygaps}
Ces espaces ont l'air d'un défaut et sont le génie du système~:
\begin{itemize}
  \item ils rendent les jonctions \emph{réglables} --- une \omterm{def:g8:nervous-communication:synapse}{synapse}
        souvent franchie se renforce, une \omterm{def:g8:nervous-communication:synapse}{synapse} inutilisée
        s'efface~: c'est l'apprentissage (les «~meilleurs chemins~»
        de \cref{rem:g5:brain-in-command:training}, retrouvés ---
        ce sont des \omterm{def:g8:nervous-communication:synapse}{synapses} renforcées)~;
  \item ils rendent les signaux \emph{à sens unique} --- les
        messagers ne sont libérés que d'un seul côté, si bien que le
        trafic ne peut pas remonter~;
  \item ils permettent à un \omterm{def:g8:nervous-communication:neuron}{neurone} de \emph{peser} des milliers
        d'entrées --- des voix qui excitent et des voix qui freinent,
        additionnées dans chaque \omterm{def:g5:first-look-at-cells:cell}{cellule} avant qu'elle ne déclenche~:
        la décision de la boucle, exécutée aux jonctions.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Les synapses de la jongleuse]\label{ex:g8:nervous-communication:juggler}
La jongleuse de \cref{rem:g5:brain-in-command:training}, enfin
expliquée~: un mois d'entraînement a conduit les mêmes signaux à
travers les mêmes jonctions jusqu'à ce que ces \omterm{def:g8:nervous-communication:synapse}{synapses} se
renforcent --- l'aiguillage laborieusement pesé tourne maintenant
prépesé. L'habileté est de la chimie consolidée aux espaces~;
«~l'entraînement rend définitif~» est une affirmation sur les
\omterm{def:g8:nervous-communication:synapse}{synapses}.
\end{example}

\begin{remark}[Les espaces sont les points vulnérables]\label{rem:g8:nervous-communication:vulnerable}
Tout ce qui agit de l'extérieur sur le \omterm{def:g5:brain-in-command:system}{système nerveux} agit aux
\omterm{def:g8:nervous-communication:synapse}{synapses} --- le seul endroit où le signal devient de la chimie,
amarrable et trompable. Les anesthésiques les font taire~; les
venins les bloquent (certains paralysent en bloquant le messager de
la jonction avec le \omterm{def:g3:bones-and-muscles:muscle}{muscle}, d'autres en l'inondant)~; et l'alcool de
\cref{prop:g5:healthy-choices:substances}, comme toute drogue qui
change l'humeur ou la perception, agit en imitant, en bloquant ou en
épuisant les messagers de ces espaces.
\Cref{ch:g8:protecting-the-nervous-system} bâtit exactement sur ce
point.
\end{remark}

\section{Fils rapides, courrier lent~: deux systèmes comparés}

\begin{proposition}[Les nerfs face aux hormones]\label{prop:g8:nervous-communication:compare}
Le corps montre désormais deux systèmes de communication, bâtis pour
des travaux opposés~:
\begin{itemize}
  \item le système \emph{nerveux}~: des signaux en millièmes de
        seconde, le long de lignes dédiées, vers des adresses
        précises --- un \omterm{def:g3:bones-and-muscles:muscle}{muscle}, une glande --- et qui cessent dès
        que le signal cesse~: le système des prises au vol, de la
        parole et des réflexes~;
  \item le système \emph{hormonal}
        (\cref{def:g8:puberty:hormones})~: des messagers en minutes
        ou en mois, diffusés par le \omterm{prop:g4:journey-of-food:absorb}{sang} à chaque organe bâti pour
        les entendre, qui commandent des programmes lents et
        étendus à tout le corps --- la croissance, les cycles, la
        reconstruction de la \omterm{def:g8:puberty:puberty}{puberté}.
\end{itemize}
Le télégraphe et la poste~: le corps fait tourner les deux, et
\cref{ch:g8:hormonal-communication} donne à la poste son propre
chapitre.
\end{proposition}
\admitted

\begin{method}[Diagnostiquer une communication, système par système]\label{met:g8:nervous-communication:diagnose}
Devant n'importe quelle coordination du corps, attribue son système~:
\begin{enumerate}
  \item chronomètre-la~: des millièmes de seconde --- nerveux~; des
        minutes ou des années --- hormonal (ou les deux, en
        relais)~;
  \item cartographie sa portée~: une seule adresse précise ---
        nerveux~; tout le corps --- hormonal~;
  \item trouve le trajet~: un chemin de fils nommable --- nerveux~;
        une diffusion par le \omterm{prop:g4:journey-of-food:absorb}{sang} --- hormonal~;
  \item vérifie la persistance~: cesse avec le signal --- nerveux~;
        se poursuit tant que le messager circule --- hormonal.
\end{enumerate}
\end{method}

\begin{example}[La méthode sur trois cas]\label{ex:g8:nervous-communication:cases}
Le clignement devant un souffle d'air~: des millièmes de seconde, un
seul anneau musculaire, un trajet câblé, fini aussitôt --- nerveux.
La poussée de croissance~: des années, tout le corps, le \omterm{prop:g4:journey-of-food:absorb}{sang}, une
action qui dure des saisons --- hormonal. Le \omterm{def:g4:heart-and-blood:heart}{cœur} emballé du sprint
(\cref{prop:g7:organs-at-work:supply})~: les deux en relais --- des
signaux nerveux montent la cadence en un battement, et un messager
sanguin (la fameuse bouffée que l'on ressent comme une vague chaude
de frayeur) maintient l'alerte de tout le corps pendant des minutes.
Le télégraphe d'abord, la poste ensuite.
\end{example}

\section{Problème~: le banc des réflexes}

\begin{problem}[{Problème du week-end --- le réflexe rotulien,
instrumenté}]\label{pb:g8:nervous-communication:1}
La démonstration classique~: le volontaire assis, le petit marteau
sous la rotule, le coup de pied involontaire --- instrumenté cette
fois de chronomètres et de ce chapitre.

\medskip\noindent\textbf{Partie I --- Le circuit.}
\begin{enumerate}
  \item Le marteau étire un tendon~; le pied donne son coup un
        cinquantième de seconde plus tard environ. Expose le
        circuit~: quelles sortes de \omterm{def:g8:nervous-communication:neuron}{neurones}, par où --- et pourquoi
        le \omterm{def:g5:brain-in-command:system}{cerveau} n'est pas sur cette boucle.
  \item Un cinquantième de seconde est plus rapide que le cinquième
        de la prise au vol. Explique l'écart par la longueur du
        circuit et le nombre de jonctions.
  \item Le volontaire ne peut pas retenir le coup de pied, même
        prévenu. Que dit cela sur l'endroit où le réflexe est décidé
        --- et sur le trafic à sens unique des \omterm{def:g8:nervous-communication:synapse}{synapses}~?
  \item Aussitôt après le coup de pied, le volontaire \emph{sent} la
        tape. Explique ce second voyage, plus lent, qui explique le
        décalage.
\end{enumerate}

\medskip\noindent\textbf{Partie II --- Variantes de paillasse.}
\begin{enumerate}[resume]
  \item Tape plus fort~: le coup de pied se renforce. Quelle règle
        de codage de \cref{prop:g8:nervous-communication:code} vient
        de se montrer~?
  \item Le médecin de la classe explique que ce réflexe est testé à
        chaque visite. Que certifie un réflexe rotulien présent et
        symétrique, au sujet de la moelle, des \omterm{def:g5:brain-in-command:system}{nerfs} et des
        jonctions~?
  \item Un volontaire qui serre la mâchoire donne un coup plus fort
        --- une amplification connue. Que révèle cela sur les voix
        des \omterm{def:g8:nervous-communication:neuron}{neurones} d'association, qui atteignent même les boucles
        «~sans \omterm{def:g5:brain-in-command:system}{cerveau}~»~?
  \item Un anesthésique local dans la \omterm{def:g1:my-body:parts}{jambe} supprime le coup de pied
        et la sensation. Nomme le point d'attaque, d'après
        \cref{rem:g8:nervous-communication:vulnerable}.
\end{enumerate}

\medskip\noindent\textbf{Partie III --- Le compte rendu.}
\begin{enumerate}[resume]
  \item Dessine le réflexe comme la boucle en trois étapes de
        \cref{prop:g5:brain-in-command:loop}, en marquant où
        celui-ci prend un raccourci.
  \item Compare le réflexe et la prise au vol dans un tableau à deux
        colonnes~: trajet, jonctions, durée, possibilité de le
        retenir, rôle de l'entraînement.
  \item La dernière partie du compte rendu demande~: «~pourquoi les
        boucles de protection rapides court-circuitent-elles le
        \omterm{def:g5:brain-in-command:system}{cerveau}~?~» Réponds avec la logique de la poêle brûlante
        (\cref{exo:g5:brain-in-command:5}).
  \item Une phrase pour conclure~: ce que le banc a montré à la fois
        de la langue du \omterm{def:g5:brain-in-command:system}{système nerveux}, de son câblage et de ses
        jonctions.
\end{enumerate}
\end{problem}
