\section*{Bab \ref{ch:g10:chemistry-of-life} --- Susunan Kimia Makhluk Hidup}

\begin{solution}{exo:g10:chemistry-of-life:1}
Oksigen (sekitar 65\%), karbon (18.5\%), hidrogen (9.5\%) dan nitrogen
(3.2\%): bersama-sama 96\% massa tubuh.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:2}
Mineral: air, natrium klorida, kalsium fosfat, karbon dioksida (sebuah
senyawa karbon, tetapi dibuat di luar kehidupan dan tanpa kerangka
karbon--hidrogen). Organik: glukosa, trigliseridanya, \omterm{def:g10:chemistry-of-life:families}{asam aminonya}.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:3}
Air yang hilang $12.0 - 0.6 = \qty{11.4}{g}$; $11.4/12.0 = 0.95$: daun
itu mengandung 95\% air.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:4}
(a) Larutan iodin pada biji yang ditumbuk: biru kehitaman. (b) Pereaksi
biuret: ungu. (c) Tumbuk bijinya di atas kertas: noda tembus cahaya
yang tidak mengering; Sudan III mewarnainya merah.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:5}
\omterm{def:g10:chemistry-of-life:families}{Karbohidrat}: gula sederhana seperti glukosa. \omterm{def:g10:chemistry-of-life:families}{Lipid}: gliserol dan
asam lemak. \omterm{def:g10:chemistry-of-life:families}{Protein}: \omterm{def:g10:chemistry-of-life:families}{asam amino} (dua puluh jenis). \omterm{def:g10:chemistry-of-life:families}{Asam nukleat}:
nukleotida (empat jenis).
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:6}
Karbon: $18.5/0.02 \approx 900$ kali lebih pekat. Nitrogen:
$3.2/0.002 = 1600$ kali. Materi hidup bukan cuplikan
lingkungannya: ia mengumpulkan beberapa unsur langka dengan faktor
seribuan, dan itulah tanda kimia kehidupan.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:7}
Air: $0.62 \times 70 = \qty{43.4}{kg}$. Bahan kering: $70 - 43.4 =
\qty{26.6}{kg}$. \omterm{def:g10:chemistry-of-life:families}{Protein}: $0.17 \times 70 = \qty{11.9}{kg}$.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:8}
\omterm{def:g10:chemistry-of-life:families}{Karbohidrat} $60 \times 17 = \qty{1020}{kJ}$; \omterm{def:g10:chemistry-of-life:families}{lipid} $12 \times 38 =
\qty{456}{kJ}$; \omterm{def:g10:chemistry-of-life:families}{protein} $8 \times 17 = \qty{136}{kJ}$; seluruhnya
\qty{1612}{kJ}, sekitar \qty{1600}{kJ}. \omterm{def:g10:chemistry-of-life:families}{Lipid} memasok $456/1612 \approx
28\%$ energinya dengan hanya 12\% massanya.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:9}
Roti: biru kehitaman --- tepung sebagian besar pati. Keju: tidak ada
perubahan warna selain rona iodinnya sendiri --- \omterm{def:g10:chemistry-of-life:families}{protein} dan \omterm{def:g10:chemistry-of-life:families}{lipid}, tanpa pati. Gula:
tidak berubah --- sukrosa adalah gula sederhana, bukan pati; iodin
hanya mengenali rantai pati yang panjang.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:10}
Bahan keringnya sebagian besar organik: molekul karbon--hidrogen yang
bergabung dengan oksigen, melepaskan energi, karbon dioksida dan air.
Abunya adalah \omterm{def:g10:chemistry-of-life:mineral-organic}{garam mineral}, yang sudah sepenuhnya bergabung dengan
oksigen; tidak ada lagi di dalamnya yang dapat terbakar.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:11}
Pereaksi Fehling bereaksi dengan gugus kimia bebas yang dibawa molekul
glukosa tunggal; dalam pati satuan glukosanya saling bergabung dan
gugus itu terkunci di dalam rantai, sehingga pereaksinya tidak
menemukan apa pun. Sebaliknya iodin menyusup ke dalam rantai pati
panjang yang melingkar, yang tidak dibentuk molekul glukosa tunggal.
Tiap uji mengenali sebuah struktur, bukan sebuah unsur.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:12}
Butir gandum itu telah menyerap air besar-besaran. Pada 13\%
cadangannya padat dan enzimnya tidak dapat bekerja: kimia butir itu
dimatikan dan ia bertahan dalam keadaan dorman. Air melarutkan
simpanannya, membiarkan enzimnya bekerja dan mengantarkan hasilnya ke
embrio yang tumbuh: kimianya hidup lagi hanya setelah air berlimpah.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:13}
Garam: $9 \times 0.5 = \qty{4.5}{g}$; glukosa: $1 \times 0.5 =
\qty{0.5}{g}$. Air murni akan mengencerkan plasmanya; air lalu akan
masuk ke dalam sel darah merah, yang akan menggembung dan pecah.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:14}
Glikogen: $500 \times 17 = \qty{8500}{kJ}$. \omterm{def:g10:chemistry-of-life:families}{Lipid}: $10000 \times 38 =
\qty{3.8e5}{kJ}$. Untuk menyimpan \qty{3.8e5}{kJ} sebagai glikogen: $380000/17
\approx \qty{22.4}{kg}$ glikogen, ditambah tiga kali massa itu berupa
air, seluruhnya sekitar \qty{90}{kg} --- lebih berat daripada orangnya.
Lemak menyimpan kira-kira sembilan kali lebih banyak energi tiap
kilogram yang dibawa, dan itulah sebabnya hewan yang harus bergerak menyimpan cadangannya sebagai \omterm{def:g10:chemistry-of-life:families}{lipid}.
\end{solution}

\begin{solution}{exo:g10:chemistry-of-life:15}
Atomnya memang biasa saja, tetapi (1) perbandingannya tidak: materi
hidup memekatkan karbon seribu kali lipat dan nitrogen jauh lebih
banyak lagi, berhadapan dengan kerak oksigen--silikon; (2) molekulnya
tidak: hanya makhluk hidup yang merakit karbon menjadi rantai raksasa
--- \omterm{def:g10:chemistry-of-life:families}{protein}, pati, DNA --- yang ukurannya tidak ada tandingannya di
dunia mineral; (3) molekul itu membawa keterangan dalam urutan
satuannya, dan tidak ada kristal yang berbuat begitu. Kehidupan
istimewa secara kimia dalam penataannya, bukan dalam atomnya.
\end{solution}

\begin{solution}{pb:g10:chemistry-of-life:1}
\textbf{1.} Air $0.13 \times 45 = \qty{5.9}{mg}$; massa kering
\qty{39.2}{mg}.

\textbf{2.} Pati $0.70 \times 39.2 \approx \qty{27.4}{mg}$; \omterm{def:g10:chemistry-of-life:families}{protein}
$0.12 \times 39.2 \approx \qty{4.7}{mg}$; \omterm{def:g10:chemistry-of-life:families}{lipid} $0.02 \times 39.2
\approx \qty{0.8}{mg}$.

\textbf{3.} Iodin (biru kehitaman) untuk pati; biuret (ungu) untuk
\omterm{def:g10:chemistry-of-life:families}{protein}; noda tembus cahaya pada kertas atau Sudan III untuk \omterm{def:g10:chemistry-of-life:families}{lipid}.

\textbf{4.} Pada 13\% simpanannya padat dan enzimnya tidak aktif; air
harus melarutkan cadangannya dan membiarkan reaksinya berjalan sebelum
embrionya dapat tumbuh. Perkecambahan mulai dengan menyerap air, dan
88\% pada kecambahnya adalah tanda bahwa kimianya menyala.

\textbf{5.} Air $0.13 \times 8 = \qty{1.04}{t}$; bahan kering
\qty{6.96}{t}, sekitar \qty{7}{t}.

\textbf{6.} Massanya $500 + 320 + 10 - 130 = \qty{700}{g}$. Air dalam
rotinya: dari tepung $0.13 \times 500 = \qty{65}{g}$, ditambah \qty{320}{g},
dikurangi \qty{130}{g}: \qty{255}{g}, yaitu $255/700 \approx 36\%$.

\textbf{7.} Bahan kering tepung $0.87 \times 500 = \qty{435}{g}$; pati
$0.70 \times 435 \approx \qty{305}{g}$; \omterm{def:g10:chemistry-of-life:families}{protein} $0.12 \times 435 \approx
\qty{52}{g}$ (\omterm{def:g10:chemistry-of-life:families}{lipid} sekitar \qty{9}{g}).

\textbf{8.} $305 \times 17 + 52 \times 17 + 9 \times 38 \approx 5185 +
884 + 342 \approx \qty{6400}{kJ}$.

\textbf{9.} $120/700 \approx 0.17$ bagian rotinya: sekitar \qty{1100}{kJ},
yaitu 11\% dari \qty{10000}{kJ}.

\textbf{10.} Garam bersifat mineral: ia sudah teroksidasi sepenuhnya
dan tidak punya ikatan karbon--hidrogen untuk dibakar. Di dalam tubuh
ionnya menetapkan kadar garam darah dan diperlukan bagi isyarat saraf
dan otot.

\textbf{11.} $0.60 \times 60 = \qty{36}{kg}$ air.

\textbf{12.} \omterm{def:g10:chemistry-of-life:families}{Protein} $0.17 \times 60 = \qty{10.2}{kg}$; \omterm{def:g10:chemistry-of-life:families}{lipid} $0.16
\times 60 = \qty{9.6}{kg}$; mineral \qty{3.6}{kg}; \omterm{def:g10:chemistry-of-life:families}{karbohidrat}
\qty{0.6}{kg}.

\textbf{13.} $1.2/36 \approx 3.3\%$ air tubuhnya. Volume darah dan
pengaturan suhu sama-sama bergantung padanya; kehilangan 10\%
berbahaya, sehingga minum harus memulihkannya dalam hitungan jam.

\textbf{14.} Plasma $0.55 \times 4.5 \approx \qty{2.5}{L}$; air $0.9
\times 2.5 \approx \qty{2.2}{kg}$, sekitar 6\% air tubuhnya. Sisanya
ada di dalam sel (kira-kira dua pertiga seluruh air tubuh) dan dalam
cairan antarsel.

\textbf{15.} Seorang bayi menukar bagian airnya yang jauh lebih besar
setiap hari (minum, urine, kulit) daripada orang dewasa, sehingga
kehilangan beberapa jam tanpa asupan menyingkirkan bagian yang
berbahaya dari seluruh airnya.

\textbf{16.} \omterm{def:g10:chemistry-of-life:families}{Karbohidrat} $0.40 \times 0.6 = \qty{0.24}{kg}$; \omterm{def:g10:chemistry-of-life:families}{lipid}
$0.77 \times 9.6 \approx \qty{7.4}{kg}$; \omterm{def:g10:chemistry-of-life:families}{protein} $0.53 \times 10.2
\approx \qty{5.4}{kg}$. Seluruhnya sekitar \qty{13}{kg} karbon.

\textbf{17.} $13/60 \approx 22\%$, berhadapan dengan 18.5\%.
Kesesuaiannya baik mengingat fraksi keluarga yang dibulatkan dan
perubahan kadar \omterm{def:g10:chemistry-of-life:families}{lipid} antarindividu; tubuh yang lebih ramping memberi
angka yang lebih rendah.

\textbf{18.} $0.16 \times 10.2 \approx \qty{1.6}{kg}$ nitrogen.
Nitrogen itu berasal dari nitrogen udara, ditangkap bakteri tanah,
diserap tumbuhan dan dimakan sepanjang rantai makanan.

\textbf{19.} $0.89 \times 36 \approx \qty{32}{kg}$ oksigen, yaitu
$32/60 \approx 53\%$ massa tubuh. Air sendirian menjelaskan sebagian
besar dari 65\% itu; sisanya adalah oksigen di dalam molekul organik.

\textbf{20.} Seorang bermassa \qty{60}{kg} membawa sekitar \qty{13}{kg}
karbon. Karbon itu meninggalkan udara sebagai karbon dioksida lewat
fotosintesis pada tumbuhan, dan masuk ke dalam tubuh lewat makan dan
pencernaan sepanjang rantai makanan.
\end{solution}
