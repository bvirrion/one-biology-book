\chapter{Human Nutrition}\label{ch:g6:nutrition}

\omterm{def:g2:food-energy:food}{Food} supplies \omterm{def:g6:nutrition:def}{energy} and building materials. Knowing nutrient groups and
how \omterm{def:g6:nutrition:digestion}{digestion} releases them helps you make \omterm{def:g1:senses:sense}{sense} of healthy eating ---
without turning meals into a guilt contest.

\section{Why we need food}

\begin{definition}[Nutrition]\label{def:g6:nutrition:def}
\emph{Nutrition}\index{nutrition} is how \omterm{def:g4:ecosystems:levels}{organisms} obtain and use \omterm{def:g2:food-energy:food}{food}.
In humans, \omterm{def:g2:food-energy:food}{food} provides \emph{energy}\index{energy (food)}, materials
for \omterm{def:g2:growing:growth}{growth} and repair, and substances that keep processes working.
\end{definition}

\begin{proposition}[Balanced intake]\label{prop:g6:nutrition:balance}
No single \omterm{def:g2:food-energy:food}{food} supplies everything in ideal amounts for life. A varied
diet across the week is the practical aim.
\end{proposition}
\admitted

\section{Nutrient groups}

\begin{definition}[Main nutrients]\label{def:g6:nutrition:groups}
\begin{itemize}
  \item \emph{Carbohydrates}\index{carbohydrate} --- main quick \omterm{def:g6:nutrition:def}{energy}
        (sugars and starches).
  \item \emph{Proteins}\index{protein} --- \omterm{def:g2:growing:growth}{growth}, repair, enzymes.
  \item \emph{Fats (lipids)}\index{fat}\index{lipid} --- \omterm{def:g6:nutrition:def}{energy} store,
        \omterm{def:g5:cell-structure:animal}{cell membranes}, some vitamins.
  \item \emph{Vitamins}\index{vitamin} and \emph{minerals}\index{mineral}
        --- needed in smaller amounts for \omterm{def:g2:healthy:health}{health} (for example calcium
        for \omterm{def:g3:bones:bone}{bones}, iron for \omterm{def:g5:circulation:blood}{blood}).
  \item \emph{Fibre}\index{fibre (diet)} --- \omterm{def:g1:plants-animals:plant}{plant} material that aids
        digestive \omterm{def:g2:healthy:health}{health}.
  \item \emph{Water}\index{water (body)} --- essential solvent and
        transport medium.
\end{itemize}
\end{definition}

\section{Digestion: releasing nutrients}

\begin{definition}[Digestion]\label{def:g6:nutrition:digestion}
\emph{Digestion}\index{digestion} breaks large \omterm{def:g2:food-energy:food}{food} molecules into
smaller ones that can be absorbed into the \omterm{def:g5:circulation:blood}{blood}. It includes chewing,
churning and chemical action by enzymes.
\end{definition}

\omimg{digestive-system.jpg}{0.52\linewidth}{The \omterm{def:g4:body-systems:digestion}{digestive system} is a
tube with specialised regions from \omterm{def:g3:teeth:path}{mouth} to \omterm{def:g3:teeth:path}{intestine}.}

\begin{method}[Follow a mouthful]\label{met:g6:nutrition:path}
\begin{enumerate}
  \item \omterm{def:g3:teeth:path}{Mouth}: teeth and saliva begin breakdown.
  \item \omterm{def:g3:teeth:path}{Stomach}: acid and enzymes continue \omterm{def:g6:nutrition:groups}{protein} \omterm{def:g6:nutrition:digestion}{digestion}; \omterm{def:g2:food-energy:food}{food} is
        mixed.
  \item Small \omterm{def:g3:teeth:path}{intestine}: most \omterm{def:g6:nutrition:digestion}{digestion} finished; \omterm{def:g4:plants-need:needs}{nutrients} absorbed.
  \item Large \omterm{def:g3:teeth:path}{intestine}: \omterm{def:g4:plants-need:needs}{water} reclaimed; fibre and wastes form faeces.
\end{enumerate}
\end{method}

\section{Energy and activity}

\begin{definition}[Energy balance (simple)]\label{def:g6:nutrition:energy}
\omterm{def:g2:food-energy:food}{Food} \omterm{def:g6:nutrition:def}{energy} is measured in kilojoules or calories. Active bodies
generally need more \omterm{def:g6:nutrition:def}{energy}. \omterm{def:g2:growing:growth}{Growth}, pregnancy and illness change \omterm{def:g1:caring:needs}{needs}.
Extreme restriction without medical advice is harmful.
\end{definition}
