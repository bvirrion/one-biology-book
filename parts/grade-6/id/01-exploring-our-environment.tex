\chapter{Menjelajahi Lingkungan Kita}\label{ch:g6:exploring-our-environment}

Seberangilah halaman sekolah pada tengah hari bulan Juli dan sisi
selatan temboknya terasa panas di bawah tanganmu, udara di atas
aspalnya beriak, dan satu-satunya benda hijau yang terlihat adalah
rumput liar di sebuah celah. Melangkahlah ke balik tembok utara: udara
sejuk, tanah lembap, \omterm{prop:g4:plants-without-seeds:damp}{lumut}, kutu kayu di bawah sebuah batu. Dua dunia,
berjarak tiga meter. Tahun ini bermula dengan menjelajahi \omterm{def:g6:exploring-our-environment:components}{lingkungan}
dengan cara para ilmuwan --- mengukurnya, dan menanyakan mengapa tiap
\omterm{def:g6:exploring-our-environment:components}{makhluk hidup} tinggal persis di tempat ia tinggal.

\section{Terbuat dari apa sebuah lingkungan}

\begin{definition}[Komponen sebuah lingkungan]\label{def:g6:exploring-our-environment:components}
Suatu \emph{lingkungan}\index{lingkungan} --- sebuah halaman, sebuah
hutan, tepian sebuah kolam --- memuat tiga macam komponen:
\begin{enumerate}
  \item \emph{makhluk hidup}\index{makhluk hidup}: \omterm{def:g1:animals-around-us:animal}{hewan}, \omterm{def:g1:plants-around-us:plant}{tumbuhan},
        jamur dan kehidupan renik yang ada di situ;
  \item \emph{materi mineral}\index{materi mineral}, yang \omterm{def:g1:living-or-not:nonliving}{tidak pernah
        hidup}: batuan, batu, bagian mineral tanah, air, udara;
  \item \emph{jejak dan sisa} makhluk hidup: \omterm{def:g1:plants-around-us:parts}{daun} kering, kayu yang
        jatuh, tinja, \omterm{def:g1:animals-around-us:covering}{bulu burung}, \omterm{def:g1:animals-around-us:covering}{cangkang} --- kotak ``dahulu
        hidup'' pada \cref{def:g1:living-or-not:nonliving}, yang
        sekarang menjadi perlengkapan petunjuk bagi sang penjelajah.
\end{enumerate}
\end{definition}

\begin{example}[Pendataan di balik tembok utara]\label{ex:g6:exploring-our-environment:inventory}
Pendataan sepuluh menit di sudut yang sejuk: yang hidup --- \omterm{prop:g4:plants-without-seeds:damp}{lumut},
\omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku}, kutu kayu, seekor siput, seekor kelabang, tanaman ivi;
yang mineral --- batu temboknya, tanah yang lembap, sebuah genangan;
jejaknya --- \omterm{def:g1:plants-around-us:parts}{daun} kering, sebuah \omterm{def:g1:animals-around-us:covering}{cangkang} siput, tinja \omterm{prop:g3:sorting-living-things:groups}{burung} di
puncak temboknya. Tiga daftar, dan sudut itu terlukiskan seperti
sebuah lokasi milik ilmuwan.
\end{example}

seekor siput di batunya --- dan di bawah lempeng yang diangkat,

\begin{remark}[Jejak terhitung sebagai data]\label{rem:g6:exploring-our-environment:traces}
Kebanyakan \omterm{def:g1:animals-around-us:animal}{hewan} bersembunyi atau melarikan diri, sehingga sang
penjelajah sering membaca \emph{jejak} sebagai gantinya: sebutir \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}
hazel yang tergigit menyebutkan tikus hutannya, sebuah jaring
laba-laba menyebutkan pemiliknya, tapak di lumpur menyebutkan
kuntulnya. Pendataan yang hanya mendaftarkan apa yang cukup diam untuk
dilihat melewatkan separuh penghuninya.
\end{remark}

\section{Mengukur kondisinya}

\begin{definition}[Kondisi lingkungan]\label{def:g6:exploring-our-environment:conditions}
Yang disebut \emph{kondisi}\index{kondisi lingkungan} sebuah
\omterm{def:g6:exploring-our-environment:components}{lingkungan} adalah ciri fisiknya yang dapat diukur:
\begin{itemize}
  \item \emph{suhu}\index{suhu}, yang diukur dengan termometer, dalam
        derajat Celsius;
  \item \emph{kelembapan}\index{kelembapan} --- seberapa lembap udara
        atau tanahnya;
  \item \emph{cahaya}\index{cahaya sebagai kondisi} --- dari matahari
        penuh sampai naungan yang pekat.
\end{itemize}
Kondisi berubah dari satu tempat ke tempat lain --- sering dengan
tajam dalam hitungan meter --- dan sepanjang hari serta sepanjang
tahun.
\end{definition}

\begin{example}[Halamannya, yang sudah diukur]\label{ex:g6:exploring-our-environment:measured}
Survei tengah hari bulan Juli itu, dengan alat ukur: kaki tembok
selatan --- \qty{38}{\degreeCelsius}, kering, matahari penuh; sudut
utara --- \qty{22}{\degreeCelsius}, tanah lembap, naungan pekat. Tiga
meter jarak, enam belas derajat perbedaan. Kedua dunia pada alinea
pembuka sekarang menjadi dua \emph{baris dalam sebuah tabel} --- dan
itu sebuah kemajuan, karena tabel dapat dibandingkan.
\end{example}

\begin{proposition}[Kondisi menentukan penghuninya]\label{prop:g6:exploring-our-environment:decide}
\omterm{def:g6:exploring-our-environment:components}{Makhluk hidup} sebuah tempat cocok dengan \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}: tiap \omterm{def:g5:biodiversity:species}{spesies}
tumbuh subur hanya dalam rentang \omterm{def:g6:exploring-our-environment:conditions}{suhu}, \omterm{def:g6:exploring-our-environment:conditions}{kelembapan} dan \omterm{def:g6:exploring-our-environment:conditions}{cahayanya}
sendiri. \omterm{prop:g4:plants-without-seeds:damp}{Lumut} dan kutu kayu membutuhkan \omterm{def:g6:exploring-our-environment:conditions}{kelembapan} (penutup kutu
kayu, berbeda dengan penutup \omterm{prop:g3:sorting-living-things:groups}{serangga}, membiarkan air lolos --- ia
mengering sampai mati dalam satu jam di bawah matahari); rumput liar
di celah itu tahan kekeringan dan silau. Di tempat \omterm{def:g6:exploring-our-environment:conditions}{kondisinya} berbeda,
penghuninya berbeda --- aturan di balik setiap \omterm{def:g2:where-animals-live:habitat}{habitat} pada
\cref{ch:g2:where-animals-live}.
\end{proposition}
\admitted

\begin{example}[Pemilih kutu kayu]\label{ex:g6:exploring-our-environment:chooser}
Percobaan yang klasik: sebuah kotak tertutup, separuhnya dilapisi
kertas lembap, separuhnya kertas kering, dan selusin kutu kayu
dilepaskan di sepanjang garis tengahnya. Dalam beberapa menit, nyaris
semuanya berada di sisi yang lembap. Jalankan lagi dengan separuh yang
ternaungi melawan separuh yang tersinari: mereka berkumpul di
naungannya. Tidak seorang pun menggiring mereka --- tiap \omterm{def:g1:animals-around-us:animal}{hewan}, yang
mengembara dan berbelok, berakhir di tempat tubuhnya merasa nyaman.
Peta kutu kayu \emph{adalah} peta \omterm{def:g6:exploring-our-environment:conditions}{kelembapan}.
\end{example}

\begin{method}[Mensurvei sebuah lokasi]\label{met:g6:exploring-our-environment:survey}
Untuk mensurvei sebuah lokasi kecil seperti seorang ilmuwan lapangan:
\begin{enumerate}
  \item pilih dan tandailah sebuah daerah kecil --- satu meter persegi
        yang dicari dengan baik menceritakan lebih banyak daripada
        satu hektare yang dilirik sekilas;
  \item ukurlah \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}: \omterm{def:g6:exploring-our-environment:conditions}{suhu}, \omterm{def:g6:exploring-our-environment:conditions}{kelembapan}, \omterm{def:g6:exploring-our-environment:conditions}{cahaya} --- alat yang
        sama, cara yang sama, di tiap lokasinya
        (aturan keadilan pada
        \cref{met:g4:what-plants-need:fairtest});
  \item datalah ketiga macam komponennya --- \omterm{def:g6:exploring-our-environment:components}{makhluk hidup}, \omterm{def:g6:exploring-our-environment:components}{materi
        mineral}, jejak --- dengan pencarian yang benar: di bawah batu
        (kembalikan batunya:
        \cref{met:g2:caring-for-nature:rules}), di celah-celah, di
        sisi bawah \omterm{def:g1:plants-around-us:parts}{daun};
  \item catatlah segalanya di tempat --- daftar, sketsa, hitungan;
        ingatan bukan alat perekam;
  \item bandingkan lokasinya: pasangkan tiap perbedaan penghuninya
        dengan sebuah perbedaan \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}.
\end{enumerate}
\end{method}

\begin{remark}[Mengapa catatan mengalahkan ingatan]\label{rem:g6:exploring-our-environment:records}
Sesudah kembali ke dalam ruangan, dua orang penyurvei yang
mengandalkan ingatan akan berbantah tentang apakah siputnya berjumlah
tiga atau lima. Buku catatannya menyelesaikannya --- atau mengakui
dengan jujur bahwa siputnya memang tidak dihitung. Catatan tertulis,
yang dibuat di lokasi, adalah yang mengubah sebuah jalan-jalan menjadi
data; setiap ilmu dalam buku ini berjalan di atasnya.
\end{remark}

\section{Satu lingkungan, banyak dunia mikro}

\begin{definition}[Mikrohabitat]\label{def:g6:exploring-our-environment:microhabitat}
Di dalam satu \omterm{def:g6:exploring-our-environment:components}{lingkungan}, tiap tempat kecil yang punya \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}
sendiri dan penghuninya sendiri --- sisi bawah sebuah batu, sebuah
celah di tembok, kaki pagar tanaman yang ternaungi --- disebut sebuah
\emph{mikrohabitat}\index{mikrohabitat}. Lantai hutan pada
\cref{rem:g2:where-animals-live:layers} adalah mikrohabitat; dan satu
meter persegi milik sang penyurvei biasanya memuat beberapa di
antaranya.
\end{definition}

\begin{example}[Batunya, di atas dan di bawah]\label{ex:g6:exploring-our-environment:stone}
Satu batu di sudut yang sejuk: di atasnya, \omterm{prop:g4:plants-without-seeds:damp}{lumut} kerak dan seekor
lalat yang berjemur --- tersinari, kering, hangat. Di bawahnya, kutu
kayu, seekor kelabang, \omterm{def:g1:plants-around-us:parts}{akar-akar} pucat --- gelap, lembap, sejuk. Dua
\omterm{def:g6:exploring-our-environment:microhabitat}{mikrohabitat} yang lengkap, yang dipisahkan lima sentimeter batu,
masing-masing dihuni persis oleh \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang rentangnya cocok
dengan \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}.
\end{example}

\section{Soal: Dua Parit}

\begin{problem}[{Soal akhir pekan --- klub naturalis mensurvei dua
parit tepi jalan dan menemukan dua dunia yang
berbeda}]\label{pb:g6:exploring-our-environment:1}
Klub naturalisnya mensurvei dua parit di sepanjang jalan kecil yang
sama, berjarak $50$ meter. Parit N berjalan di sisi utara sebuah pagar
tanaman yang tinggi; parit S berjalan di tempat terbuka, menghadap
selatan. Klubnya bekerja sesuai buku: alat yang sama, satu meter
persegi yang sama di tiap lokasi, waktu pencarian yang sama.

\medskip\noindent\textbf{Bagian I --- Pengukurannya.}
\begin{enumerate}
  \item Termometernya menunjukkan \qty{19}{\degreeCelsius} di parit N
        dan \qty{31}{\degreeCelsius} di parit S pada jam yang sama.
        Ciri tunggal lokasi yang mana yang paling baik menjelaskan
        perbedaannya?
  \item Tanah parit N melekat di jari; tanah parit S remuk menjadi
        debu. Terjemahkan kedua pengamatan itu ke dalam perbendaharaan
        kata \cref{def:g6:exploring-our-environment:conditions}.
  \item Klubnya mengukur pada tengah hari. Berikan satu alasan mengapa
        perbandingannya akan kurang adil kalau parit N diukur pada
        tengah hari dan parit S pada pukul enam sore.
  \item Mengapa klubnya memakai bingkai satu meter persegi yang
        \emph{sama} dan waktu pencarian yang sama di kedua lokasinya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pendataannya.}
\begin{enumerate}[resume]
  \item Daftar parit N: \omterm{prop:g4:plants-without-seeds:damp}{lumut}, \omterm{prop:g4:plants-without-seeds:damp}{tumbuhan paku}, kutu kayu, siput, seekor
        kodok. Daftar parit S: rerumputan kering, sebuah rumput liar
        besar berdaun lilin, semut, seekor kadal yang berjemur,
        belalang. Kelompokkan anggota tiap daftarnya ke dalam ketiga
        macam komponennya --- lalu catat apa yang hilang dari
        daftarnya (periksa
        \cref{def:g6:exploring-our-environment:components}).
  \item Untuk kutu kayu, kodok dan kadal, sebutkan \omterm{def:g6:exploring-our-environment:conditions}{kondisi} parit mana
        yang cocok dengan kebutuhannya, dan mengapa (pakai
        \cref{prop:g6:exploring-our-environment:decide} dan apa yang
        kamu ketahui tentang tiap \omterm{def:g1:animals-around-us:animal}{hewannya}).
  \item Di parit S klubnya menemukan sebuah \omterm{def:g1:animals-around-us:covering}{cangkang} siput, tetapi
        tidak ada siputnya, lalu bertanya apakah siput tinggal di
        situ. Dua kemungkinan jujur apa yang dibiarkan terbuka oleh
        jejak semacam ini?
  \item Di bawah sebuah batu pipih di parit N: delapan kutu kayu,
        seekor kelabang, \omterm{def:g1:plants-around-us:parts}{akar-akar} putih. Sebutkan \omterm{def:g6:exploring-our-environment:conditions}{kondisi}
        \omterm{def:g6:exploring-our-environment:microhabitat}{mikrohabitatnya}, lalu nyatakan aturan yang menempatkan
        penghuni itu di situ.
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Penjelasannya.}
\begin{enumerate}[resume]
  \item Pasangkan tiap perbedaan penghuninya dengan sebuah perbedaan
        \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}: lengkapilah ``N punya \omterm{prop:g4:plants-without-seeds:damp}{lumut} dan kodok karena
        \dots; S punya kadal dan rumput liar berdaun lilin karena
        \dots''.
  \item Pagar tanamannya dijadwalkan untuk dibongkar. Ramalkan \omterm{def:g6:exploring-our-environment:conditions}{kondisi}
        parit N dua musim panas kemudian --- beserta penghuninya,
        dengan memakai pemasangan pada pertanyaan 9.
  \item Seorang anggota mengusulkan memeriksa ramalannya dengan benar:
        mensurvei kedua paritnya lagi sesudah pagar tanamannya lenyap.
        Mengapa lokasi, alat dan musim yang \emph{sama} harus dipakai
        agar survei ulangnya berarti?
  \item Laporan klubnya berakhir: ``penghuninya memetakan \omterm{def:g6:exploring-our-environment:conditions}{kondisinya}.''
        Jelaskan kalimat itu dengan pemilih kutu kayu
        (\cref{ex:g6:exploring-our-environment:chooser}), lalu
        sebutkan mengapa tidak dibutuhkan penggiringan agar peta
        semacam itu menggambar dirinya sendiri.
\end{enumerate}
\end{problem}
