\chapter{Respirasi Sel dan Fermentasi}\label{ch:b1:respiration-fermentation}

Sebuah labu berisi khamir dalam larutan gula dengan udara yang
disingkirkan menggelembungkan karbon dioksida dan mengubah gulanya
menjadi alkohol; masukkan udara dan gelembungnya melambat, alkoholnya
berhenti, dan khamirnya tumbuh sepuluh kali lebih cepat pada gula yang
sama. Pasteur melihat hal ini pada 1861 dan tak sanggup
menerangkannya; penjelasannya memakan waktu seabad dan mengalir
sepanjang seluruh bab ini. Oksigen membuat sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} dapat menyarikan
dari glukosa lima belas kali energi yang dapat diperolehnya tanpa
oksigen, dan ia melakukannya dengan meneruskan elektron gulanya
menuruni sebuah rantai pembawa menuju oksigen, memompa proton
melintasi sebuah membran sementara elektron itu jatuh, lalu
membiarkan protonnya kembali lewat sebuah turbin yang membuat \omterm{def:b1:water-small-molecules:atp}{ATP}. Bab
ini memerikan \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis}, \omterm{def:b1:respiration-fermentation:krebs}{daur Krebs}, \omterm{def:b1:respiration-fermentation:chain}{rantai respirasi} beserta
gandengan kemiosmosisnya, \omterm{def:b1:respiration-fermentation:fermentation}{fermentasi} yang bekerja tanpa oksigen, dan
pembakaran \omterm{def:b1:lipids:triglyceride}{lemak} serta \omterm{def:b1:proteins:peptide}{protein} oleh mesin yang sama.

\section{Oksidasi glukosa}

\begin{proposition}[Reaksi keseluruhannya dan tahapnya]\label{prop:b1:respiration-fermentation:overview}
\index{respirasi sel}
Oksidasi glukosa yang lengkap,
\[
\mathrm{C_6H_{12}O_6} + 6\,\mathrm{O_2} \to 6\,\mathrm{CO_2} + 6\,\mathrm{H_2O},
\qquad \Delta G^{\circ\prime} = \qty{-2870}{kJ/mol},
\]
dijalankan dalam tiga tahap yang tak pernah membiarkan elektronnya
bertemu oksigennya secara langsung. \emph{\omterm{def:b1:respiration-fermentation:glycolysis}{Glikolisis}}, dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosol},
membelah glukosa menjadi dua piruvat dan menghasilkan 2 \omterm{def:b1:water-small-molecules:atp}{ATP} serta 2
NADH. \emph{Oksidasi piruvat dan \omterm{def:b1:respiration-fermentation:krebs}{daur Krebs}}, dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{matriks
mitokondria}, mengoksidasi piruvatnya menjadi $\mathrm{CO_2}$, sambil
memuatkan elektronnya pada 8 NADH dan 2 $\mathrm{FADH_2}$ serta membuat
2 GTP. \emph{Fosforilasi oksidatif}, dalam membran dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria},
meneruskan elektron itu ke oksigen lewat sebuah rantai pembawa yang
memompa proton, dan gradien protonnya menggerakkan sintesis sekitar 26
\omterm{def:b1:water-small-molecules:atp}{ATP}. Seluruhnya, sekitar 30 \omterm{def:b1:water-small-molecules:atp}{ATP} tiap glukosa: sekitar separuh \omterm{prop:b1:water-small-molecules:gibbs}{energi
bebas} pembakarannya tertangkap, sisanya dilepas sebagai kalor.
\end{proposition}

\begin{definition}[Pembawa elektron]\label{def:b1:respiration-fermentation:carriers}
\index{NAD}\index{FAD}
\emph{$\mathrm{NAD^+}$} (nikotinamida adenina dinukleotida) menerima
dua elektron dan satu proton dari sebuah \omterm{def:b1:enzymes:enzyme}{substrat} lalu menjadi NADH;
\emph{FAD} menerima dua elektron dan dua proton lalu menjadi
$\mathrm{FADH_2}$. Keduanya \omterm{def:b1:enzymes:enzyme}{koenzim} dehidrogenase, dan keduanya didaur
ulang: direduksi katabolisme, dioksidasi kembali \omterm{def:b1:respiration-fermentation:chain}{rantai respirasinya}.
Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} hanya memegang beberapa mikromol keduanya, yang berputar
ribuan kali tiap jam. Potensial redoksnya,
$E'_0 = \qty{-0.32}{V}$ bagi $\mathrm{NAD^+}/\mathrm{NADH}$,
menempatkannya jauh di bawah oksigen ($\qty{+0.82}{V}$): tiap pasang
elektron yang diserahkan NADH kepada oksigen melepas
$\Delta G^{\circ\prime} = -2F\Delta E =
\qty{-220}{kJ/mol}$, cukup bagi lebih dari empat \omterm{def:b1:water-small-molecules:atp}{ATP}.
\end{definition}

\section{Glikolisis}

\begin{definition}[Glikolisis]\label{def:b1:respiration-fermentation:glycolysis}
\index{glikolisis}
Yang disebut \emph{glikolisis} adalah urutan sepuluh reaksi \omterm{def:b1:enzymes:enzyme}{enzim} dalam
\omterm{def:b1:eukaryotic-cell:organelle}{sitosol} yang mengubah satu glukosa (C6) menjadi dua \emph{piruvat} (C3).
Dalam \emph{tahap penanaman modal} dua \omterm{def:b1:water-small-molecules:atp}{ATP} dibelanjakan untuk
memfosforilasi gulanya (heksokinase, lalu \emph{fosfofruktokinase},
PFK) dan difosfat berkarbon enamnya dibelah menjadi dua fosfat
berkarbon tiga. Dalam \emph{tahap panen} tiap triosanya dioksidasi
$\mathrm{NAD^+}$ (satu-satunya oksidasi glikolisis) dan fosfatnya
dipindahkan ke ADP lewat \emph{fosforilasi aras
substrat}\index{fosforilasi aras substrat} --- sebuah fosfat yang
diserahkan langsung dari sebuah zat antara berenergi tinggi kepada ADP,
tanpa membran maupun oksigen. Neracanya:
\[
\text{glukosa} + 2\,\mathrm{NAD^+} + 2\,\mathrm{ADP} + 2\,\mathrm{P_i}
\to 2\,\text{piruvat} + 2\,\mathrm{NADH} + 2\,\mathrm{H^+} + 2\,\mathrm{ATP} + 2\,\mathrm{H_2O} .
\]
Ia tidak memerlukan oksigen; ia lintasan metabolisme yang tertua dan
paling semesta.
\end{definition}

\begin{proposition}[Kendali pada fosfofruktokinase]\label{prop:b1:respiration-fermentation:pfk}
PFK, langkah tak berbalik pertama yang menentukan \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis}, adalah
sebuah \omterm{def:b1:enzymes:allosteric}{enzim alosterik} (\cref{ch:b1:enzymes}) yang dihambat \omterm{def:b1:water-small-molecules:atp}{ATP} dan
sitrat --- isyarat bahwa energi dan bahan bakar \omterm{def:b1:respiration-fermentation:krebs}{daur Krebsnya} berlimpah
--- serta diaktifkan AMP dan ADP, isyarat bahwa keduanya tidak
berlimpah. Ketika \omterm{def:b1:water-small-molecules:atp}{ATP-nya} tinggi fluks lewat \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisisnya} turun dalam
beberapa sekon; ketika \omterm{def:b1:cell-unit-of-life:cell}{sel} membelanjakan \omterm{def:b1:water-small-molecules:atp}{ATP}, AMP naik tajam (lewat
adenilat kinase,
$2\,\mathrm{ADP} \rightleftharpoons \mathrm{ATP} + \mathrm{AMP}$), dan PFK
membuka. \emph{Muatan energi} \omterm{def:b1:cell-unit-of-life:cell}{sel} mengatur
lintasan yang mengisinya.
\end{proposition}

\section{Oksidasi piruvat dan daur Krebs}

\begin{definition}[Piruvat dehidrogenase, asetil-KoA]\label{def:b1:respiration-fermentation:pdh}
Piruvat masuk ke \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria} lalu dioksidasi kompleks \emph{piruvat
dehidrogenase} menjadi \emph{asetil-KoA}\index{asetil-KoA} --- sebuah
gugus asetil berkarbon dua yang dibawa \omterm{def:b1:enzymes:enzyme}{koenzim} A
(\cref{ch:b1:nucleic-acids}) pada sebuah ikatan tioester berenergi
tinggi --- sambil melepas satu $\mathrm{CO_2}$ dan satu NADH.
Asetil-KoA adalah pintu masuk bersama semua bahan bakar ke dalam
daurnya: \omterm{def:b1:carbohydrates:monosaccharide}{karbohidrat} lewat piruvat, \omterm{def:b1:lipids:triglyceride}{lemak} lewat oksidasi $\beta$,
banyak \omterm{def:b1:proteins:aminoacid}{asam amino} secara langsung.
\end{definition}

\begin{definition}[Daur Krebs]\label{def:b1:respiration-fermentation:krebs}
\index{daur Krebs}\index{daur asam sitrat}
\emph{Daur Krebs} (daur asam sitrat) dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{matriks mitokondria}
mengondensasikan \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA} (C2) dengan oksaloasetat (C4) menjadi
sitrat (C6) lalu, dalam delapan langkah, mengoksidasi kedua karbonnya
menjadi $\mathrm{CO_2}$ sambil membentuk kembali oksaloasetatnya. Tiap
gugus asetil: 3 NADH, 1 $\mathrm{FADH_2}$, 1 GTP (lewat \omterm{def:b1:respiration-fermentation:glycolysis}{fosforilasi aras
substrat}) dan 2 $\mathrm{CO_2}$. Tiap glukosa (dua asetil): 6 NADH, 2
$\mathrm{FADH_2}$, 2 GTP, 4 $\mathrm{CO_2}$ --- yang bersama 2
$\mathrm{CO_2}$ piruvat dehidrogenasenya membuat enam pada persamaan
keseluruhannya. Tak ada oksigen yang dipakai dalam daurnya sendiri; ia
berjalan hanya selama \omterm{def:b1:respiration-fermentation:chain}{rantai respirasinya} mengoksidasi kembali NADH-nya.
Ia juga bersifat \emph{amfibolik}: zat antaranya ditarik keluar untuk
biosintesis (\cref{ch:b1:biosyntheses-integration}) dan diisi kembali
dari \omterm{def:b1:proteins:aminoacid}{asam amino} serta piruvat.
\end{definition}

\begin{example}[Mengikuti karbonnya]\label{ex:b1:respiration-fermentation:carbons}
Berilah sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} glukosa yang ditandai $\mathrm{^{14}C}$ pada karbon
1: tandanya muncul pada karbon metil piruvatnya, lalu \omterm{def:b1:respiration-fermentation:pdh}{asetil-KoA-nya},
masuk ke sitrat, dan pergi sebagai $\mathrm{CO_2}$ hanya pada putaran
kedua atau ketiga daurnya --- kedua karbon yang dilepas pada satu
putaran adalah milik oksaloasetat putaran sebelumnya. Krebs (1937)
menyimpulkan daurnya dari pengamatan bahwa sejumlah katalitik asam mana
pun dalam daurnya merangsang oksidasi piruvat jauh lebih banyak
daripada yang dapat dipertanggungjawabkannya sendiri: asam itu sedang
dibentuk kembali.
\end{example}

\section{Fosforilasi oksidatif}

\begin{definition}[Rantai respirasi]\label{def:b1:respiration-fermentation:chain}
\index{rantai respirasi}\index{rantai pengangkut elektron}
Yang disebut \emph{rantai respirasi} adalah empat kompleks \omterm{def:b1:membranes-transport:membrane}{protein
membran} dalam \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria} beserta dua pembawa yang bergerak. NADH
memberikan elektronnya kepada \emph{kompleks I}, $\mathrm{FADH_2}$
(lewat suksinat dehidrogenase, \emph{kompleks II}) memberikan
elektronnya sendiri; keduanya mereduksi \emph{ubikuinon} (Q), sebuah
kuinon yang larut \omterm{def:b1:lipids:fattyacid}{lipid} dan berdifusi dalam membrannya; Q meneruskannya
ke \emph{kompleks III}, yang menyerahkannya ke \emph{sitokrom $c$},
sebuah \omterm{def:b1:proteins:peptide}{protein} kecil pada sisi luarnya; dan \emph{kompleks IV}
(sitokrom oksidase) mengantarkannya, empat sekaligus, ke
$\mathrm{O_2}$, sehingga membuat air. Pada tiap kompleks I, III dan IV
elektronnya turun potensialnya dan energinya memompa proton dari
\omterm{def:b1:eukaryotic-cell:mitochondrion}{matriksnya} ke dalam ruang antarmembran: sekitar 4, 4 dan 2 tiap pasang
elektron, sepuluh tiap NADH, enam tiap $\mathrm{FADH_2}$ (yang masuk di
bawah kompleks I).
\end{definition}

\begin{theorem}[Gandengan kemiosmosis]\label{thm:b1:respiration-fermentation:chemiosmosis}
\index{teori kemiosmosis}\index{gaya gerak proton}
Gradien proton yang dibangun rantainya --- sekitar \qty{0.16}{V}
\omterm{def:b1:membranes-transport:potential}{potensial membran} (\omterm{def:b1:eukaryotic-cell:mitochondrion}{matriksnya} negatif) ditambah satu satuan pH, sebuah
\emph{\omterm{thm:b1:respiration-fermentation:chemiosmosis}{gaya gerak proton}} sekitar \qty{0.22}{V}, atau \qty{21}{kJ} tiap
mol proton --- adalah satu-satunya tautan antara pengangkutan elektron
dan sintesis \omterm{def:b1:water-small-molecules:atp}{ATP}. \omterm{thm:b1:photosynthesis:chemiosmosis}{ATP sintase}, sebuah motor putar membran dalamnya,
meloloskan sekitar \qty{3}{H^+} tiap \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibuat, dan satu lagi
dibelanjakan untuk mengimpor fosfatnya serta mengekspor \omterm{def:b1:water-small-molecules:atp}{ATP-nya}: 4 tiap
\omterm{def:b1:water-small-molecules:atp}{ATP}. Karena itulah \emph{rasio P/O}\index{rasio P/O}: $10/4 = 2.5$ \omterm{def:b1:water-small-molecules:atp}{ATP}
tiap NADH dan $6/4 = 1.5$ tiap $\mathrm{FADH_2}$.
\end{theorem}
\begin{proof}[Bukti eksperimental]
\omterm{def:b1:eukaryotic-cell:mitochondrion}{Mitokondria} hanya membuat \omterm{def:b1:water-small-molecules:atp}{ATP} ketika membran dalamnya utuh dan
tertutup; serpihan yang mengangkut elektron tetapi tak sanggup memegang
gradien tidak membuat apa-apa. Zat pemutus gandengan seperti
dinitrofenol, yang mengantarkan proton melintasi membrannya,
meniadakan sintesis \omterm{def:b1:water-small-molecules:atp}{ATP-nya} sementara konsumsi oksigennya berlanjut dan
bahkan makin cepat, dengan energinya pergi sebagai kalor --- \omterm{def:b1:lipids:triglyceride}{lemak}
cokelat melakukan hal ini dengan sengaja lewat \omterm{def:b1:proteins:peptide}{protein} pemutus
gandengannya sendiri (\cref{ch:b1:mammal-organization}), dan
dinitrofenol dahulu dijual sebagai obat pelangsing yang membunuh lewat
suhu tubuh yang terlampau tinggi. \omterm{def:b1:enzymes:inhibitors}{Penghambat} rantainya (sianida pada
kompleks IV) menghentikan keduanya; \omterm{def:b1:enzymes:inhibitors}{penghambat} sintasenya (oligomisin)
menghentikan sintesis \omterm{def:b1:water-small-molecules:atp}{ATP} dan, karena gradiennya tak sanggup melepas
muatan, menghentikan respirasinya pula, yang lalu dipulihkan sebuah zat
pemutus gandengan. Sebuah gradien pH buatan yang dikenakan pada
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria} dalam gelap menggerakkan sintesis \omterm{def:b1:water-small-molecules:atp}{ATP}, seperti pada
kloroplas (\cref{ch:b1:photosynthesis}).
\end{proof}

\begin{method}[Neraca ATP sebuah glukosa]\label{met:b1:respiration-fermentation:balance}
\begin{enumerate}
  \item \omterm{def:b1:respiration-fermentation:glycolysis}{Glikolisis}: $+2$ \omterm{def:b1:water-small-molecules:atp}{ATP} (bersih), $+2$ NADH dalam \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya}.
  \item Piruvat dehidrogenase: $+2$ NADH. \omterm{def:b1:respiration-fermentation:krebs}{Daur Krebs}: $+6$ NADH, $+2$
        $\mathrm{FADH_2}$, $+2$ GTP ($=$ \omterm{def:b1:water-small-molecules:atp}{ATP}).
  \item Fosforilasi oksidatif: $2.5$ \omterm{def:b1:water-small-molecules:atp}{ATP} tiap NADH \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria}
        (8: 20 \omterm{def:b1:water-small-molecules:atp}{ATP}), $1.5$ tiap $\mathrm{FADH_2}$ (2: 3 \omterm{def:b1:water-small-molecules:atp}{ATP}).
  \item NADH \omterm{def:b1:eukaryotic-cell:organelle}{sitosol} tak sanggup melintasi membran dalamnya;
        elektronnya masuk lewat sebuah ulang alik yang mengantarkannya
        ke $\mathrm{FAD}$ (1.5 masing-masing, dalam otot dan otak) atau
        ke $\mathrm{NAD^+}$ (2.5 masing-masing, dalam hati dan
        jantung): 3 atau 5 \omterm{def:b1:water-small-molecules:atp}{ATP}.
  \item Seluruhnya: $2 + 2 + 20 + 3 + 3 = 30$, atau $32$ dengan ulang
        alik yang lebih baik. Efisiensinya:
        $30\times 50/2870 \approx \qty{52}{\%}$ pada keadaan seluler.
\end{enumerate}
\end{method}

\section{Fermentasi}

\begin{definition}[Fermentasi]\label{def:b1:respiration-fermentation:fermentation}
\index{fermentasi}
Tanpa oksigen \omterm{def:b1:respiration-fermentation:chain}{rantai respirasinya} berhenti, NADH tak sanggup
dioksidasi kembali, dan \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisisnya} akan terhenti dalam beberapa
sekon karena kekurangan $\mathrm{NAD^+}$. \emph{Fermentasi} membentuk
kembali zat itu dengan memakai piruvatnya sendiri sebagai
penerima elektron: dalam \emph{fermentasi laktat} (otot, \omterm{def:b1:cell-unit-of-life:cell}{sel} merah,
bakteri asam laktat) piruvatnya direduksi menjadi laktat; dalam
\emph{fermentasi alkohol} (khamir, sebagian tumbuhan) ia
didekarboksilasi menjadi asetaldehida, yang direduksi menjadi etanol,
sambil melepas $\mathrm{CO_2}$. Hasilnya adalah hasil \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis}
sendiri: 2 \omterm{def:b1:water-small-molecules:atp}{ATP} tiap glukosa, seperlima belas hasil respirasinya;
karbonnya nyaris tak teroksidasi dan hasilnya masih memegang hampir
seluruh energi bahan bakarnya. \emph{Respirasi anaerob} --- sebuah
\omterm{def:b1:respiration-fermentation:chain}{rantai pengangkut elektron} yang berakhir pada nitrat, sulfat atau
karbonat alih-alih oksigen, pada banyak bakteri --- adalah hal yang
lain, yang dibahas dalam jilid Tahun ke-2.
\end{definition}

\begin{proposition}[Efek Pasteur]\label{prop:b1:respiration-fermentation:pasteur}
\index{efek Pasteur}
Sebuah \omterm{def:b1:cell-unit-of-life:cell}{sel} yang sanggup berespirasi memakai glukosa jauh lebih cepat
ketika kekurangan oksigen daripada ketika dipasok oksigen, dan tumbuh
jauh lebih sedikit darinya.
\end{proposition}
\begin{proof}[Bukti eksperimental]
Pasteur (1861) mendapati khamir dalam sebuah tong yang dialiri udara
memakai gulanya perlahan sambil berbiak, sedangkan dalam tong tersegel
ia memakai gulanya sepuluh kali lebih cepat, sedikit berbiak dan
membuat alkohol. Penjelasannya adalah hasil \omterm{def:b1:water-small-molecules:atp}{ATP-nya}: untuk memperoleh
\omterm{def:b1:water-small-molecules:atp}{ATP} yang sama dari 2 tiap glukosa seperti dari 30, \omterm{def:b1:cell-unit-of-life:cell}{sel} itu harus
menjalankan \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisisnya} lima belas kali lebih cepat, dan PFK, yang
terlepas dari penghambatan oleh \omterm{def:b1:water-small-molecules:atp}{ATP} serta sitrat yang dipertahankan
respirasinya, mengizinkannya. Sebuah otot yang berlari cepat
memperlihatkan efek yang sama dalam beberapa sekon: \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya} turun
dua puluh kali lebih cepat daripada saat beristirahat, dan laktatnya
naik.
\end{proof}

\begin{definition}[Kuosien respirasi]\label{def:b1:respiration-fermentation:rq}
\index{kuosien respirasi}
Yang disebut \emph{kuosien respirasi} RQ adalah rasio $\mathrm{CO_2}$
yang dihasilkan terhadap $\mathrm{O_2}$ yang dipakai, dalam mol.
\omterm{def:b1:carbohydrates:monosaccharide}{Karbohidrat} memberikan 1.0 (enam masing-masing tiap glukosa); \omterm{def:b1:lipids:triglyceride}{lemak}
sekitar 0.7 (palmitat:
$\mathrm{C_{16}H_{32}O_2} + 23\,\mathrm{O_2} \to 16\,\mathrm{CO_2} +
16\,\mathrm{H_2O}$, $16/23 = 0.70$), karena karbonnya lebih tereduksi
dan memerlukan lebih banyak oksigen tiap $\mathrm{CO_2}$; \omterm{def:b1:proteins:peptide}{protein}
sekitar 0.8. Sebuah biakan yang berfermentasi melepas $\mathrm{CO_2}$
tanpa mengambil oksigen: RQ-nya tak hingga, dan RQ sebuah biakan
campuran di atas 1 mengukur bagian \omterm{def:b1:respiration-fermentation:fermentation}{fermentasinya}. Diukur pada seekor
hewan utuh lewat analisis gas, RQ-nya memberi tahu bahan bakar mana
yang sedang dibakarnya.
\end{definition}

\begin{example}[Membaca sebuah RQ]\label{ex:b1:respiration-fermentation:readrq}
Seorang manusia yang beristirahat sesudah santapan berkarbohidrat: RQ
0.95, glukosanya yang terbakar. Orang yang sama sesudah berpuasa
semalam: 0.75, \omterm{def:b1:lipids:triglyceride}{lemaknya}. Seorang pelari maraton pada kilometer ketiga
puluh, \omterm{def:b1:carbohydrates:polysaccharide}{glikogennya} habis: 0.72. Sebuah labu khamir pada glukosa dengan
sedikit udara: RQ 4 --- tiga perempat glukosanya sedang difermentasi.
\end{example}

\section{Soal: Khamir dengan dan tanpa Udara}

\begin{problem}[{Soal akhir pekan --- tong Pasteur yang ditinjau
kembali: ATP yang dihitung, glukosa yang dipakai, biomassa yang tumbuh
dan gas yang diukur, berakhir pada rasio hasil ATP-nya dan rasio P/O}]
\label{pb:b1:respiration-fermentation:1}
Khamir ditumbuhkan pada glukosa dalam dua labu yang serupa, satu
dialiri udara dan satu tersegel. Pakailah \omterm{thm:b1:respiration-fermentation:chemiosmosis}{rasio P/O} 2.5 (NADH) dan 1.5
($\mathrm{FADH_2}$), ulang alik $\mathrm{FADH_2}$ bagi NADH \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya},
hasil pertumbuhan \qty{10.5}{g} biomassa kering tiap mol \omterm{def:b1:water-small-molecules:atp}{ATP}, dan
glukosa pada
\qty{180}{g/mol}.

\medskip
\noindent\textbf{Bagian I --- \omterm{def:b1:water-small-molecules:atp}{ATP} yang dihitung.}
\begin{enumerate}
  \item Sebutkan \omterm{def:b1:water-small-molecules:atp}{ATP}, GTP, NADH dan $\mathrm{FADH_2}$ yang dihasilkan
        dari satu glukosa oleh \omterm{def:b1:respiration-fermentation:glycolysis}{glikolisis}, piruvat dehidrogenase dan
        \omterm{def:b1:respiration-fermentation:krebs}{daur Krebs}.
  \item Hitunglah proton yang dipompa rantainya tiap glukosa (10 tiap
        NADH \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondria}, 6 tiap $\mathrm{FADH_2}$; NADH \omterm{def:b1:eukaryotic-cell:organelle}{sitosolnya}
        masuk sebagai $\mathrm{FADH_2}$).
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang dibuat sintasenya pada \qty{4}{H^+} tiap
        \omterm{def:b1:water-small-molecules:atp}{ATP}, dan total \omterm{def:b1:water-small-molecules:atp}{ATP} tiap glukosa dengan udara.
  \item Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} tiap glukosa tanpa udara.
  \item Hitunglah rasio kedua hasilnya.
  \item Hitunglah \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} yang tertangkap dengan udara
        ($\Delta G_{\text{ATP}}
        = \qty{-50}{kJ/mol}$) sebagai fraksi \qty{2870}{kJ}, dan tanpa
        udara sebagai fraksi \qty{2870}{kJ} itu dan sebagai fraksi
        energi yang sesungguhnya dilepas \omterm{def:b1:respiration-fermentation:fermentation}{fermentasi} (glukosa menjadi
        dua etanol dan dua $\mathrm{CO_2}$: \qty{-235}{kJ/mol}).
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Glukosa yang dipakai, biomassa yang tumbuh.}
Labu yang tersegel harus membuat \omterm{def:b1:water-small-molecules:atp}{ATP} pada laju yang sama dengan labu
yang dialiri udara untuk memelihara \omterm{def:b1:cell-unit-of-life:cell}{selnya}: \qty{30}{mmol} \omterm{def:b1:water-small-molecules:atp}{ATP} tiap jam.
\begin{enumerate}[resume]
  \item Hitunglah glukosa yang dipakai tiap jam pada tiap labunya.
  \item Hitunglah etanol dan $\mathrm{CO_2}$ yang dihasilkan tiap jam
        dalam labu tersegelnya, serta $\mathrm{CO_2}$ dan
        $\mathrm{O_2}$ yang dipertukarkan dalam labu yang dialiri udara.
  \item Hitunglah biomassa yang dapat dibangun labu tersegelnya tiap
        mol glukosa, dan tiap gram glukosa.
  \item Labu yang dialiri udara membangun \qty{0.5}{g} biomassa tiap
        gram glukosa. Hitunglah \omterm{def:b1:water-small-molecules:atp}{ATP} yang diwakilinya pada
        \qty{10.5}{g/mol}, lalu bandingkan dengan 30 \omterm{def:b1:water-small-molecules:atp}{ATP} yang tersedia.
        Apa yang membatasi hasil aerobnya?
  \item Jelaskan \omterm{prop:b1:respiration-fermentation:pasteur}{efek Pasteur} dari pertanyaan 7 dan 9, dan sebutkan
        \omterm{def:b1:enzymes:enzyme}{enzim} yang pengaturannya melaksanakannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Gas yang diukur.}
Sebuah labu ketiga, yang kurang dialiri udara, melepas \qty{44}{mg}
$\mathrm{CO_2}$ dan memakai \qty{20}{mg} $\mathrm{O_2}$ tiap jam.
\begin{enumerate}[resume]
  \item Ubahlah keduanya menjadi milimol lalu hitunglah RQ-nya.
  \item Misalkan $x$ adalah glukosa yang direspirasi dan $y$ glukosa
        yang difermentasi tiap jam. Tuliskan neraca $\mathrm{O_2}$ dan
        $\mathrm{CO_2}$-nya lalu selesaikan bagi $x$ dan $y$.
  \item Berapa fraksi glukosanya yang difermentasi? Berapa fraksi
        \omterm{def:b1:water-small-molecules:atp}{ATP-nya} yang datang dari \omterm{def:b1:respiration-fermentation:fermentation}{fermentasi}?
  \item Labu yang sama pada palmitat alih-alih glukosa (secara aerob):
        hitunglah RQ-nya.
  \item Seorang mahasiswa mengukur RQ 0.7 pada seekor hewan yang
        beristirahat sesudah berpuasa semalam dan 1.0 sesudah sebuah
        santapan. Tafsirkan.
\end{enumerate}

\noindent\textbf{Bagian IV --- Turbinnya.}
\begin{enumerate}[resume]
  \item Hitunglah \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} satu mol proton yang melintasi \omterm{thm:b1:respiration-fermentation:chemiosmosis}{gaya
        gerak proton} \qty{0.22}{V}.
  \item Hitunglah energi \qty{4}{H^+} dan efisiensi sintasenya pada
        \qty{50}{kJ} tiap \omterm{def:b1:water-small-molecules:atp}{ATP}.
  \item Hitunglah \omterm{prop:b1:water-small-molecules:gibbs}{energi bebas} yang dilepas dua elektron yang jatuh
        dari NADH ke $\mathrm{O_2}$ (\qty{1.14}{V}), energi yang
        tersimpan dalam kesepuluh proton yang dipompa, dan efisiensi
        rantainya.
  \item Gabungkan kedua efisiensinya lalu bandingkan dengan 2.5 \omterm{def:b1:water-small-molecules:atp}{ATP}
        $\times$ \qty{50}{kJ} atas \qty{220}{kJ}.
  \item Sebuah zat pemutus gandengan ditambahkan pada labu yang dialiri
        udara. Ramalkan perubahan konsumsi oksigennya, hasil \omterm{def:b1:water-small-molecules:atp}{ATP-nya},
        konsumsi glukosanya, keluaran kalornya dan pertumbuhannya.
  \item Oligomisin, yang menghambat sintasenya, ditambahkan sebagai
        gantinya. Ramalkan perubahannya, dan apa yang terjadi bila
        sebuah zat pemutus gandengan lalu ditambahkan di atasnya.
  \item Sebuah khamir mutan tidak memiliki kompleks I; NADH-nya masuk
        ke rantainya lewat sebuah \omterm{def:b1:enzymes:enzyme}{enzim} alternatif yang tidak memompa
        proton. Hitunglah \omterm{thm:b1:respiration-fermentation:chemiosmosis}{rasio P/O-nya} bagi NADH dan \omterm{def:b1:water-small-molecules:atp}{ATP-nya} tiap
        glukosa.
  \item Jelaskan mengapa \omterm{def:b1:cell-unit-of-life:cell}{sel} menjaga lubuk NADH \omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondrianya} nyaris
        tereduksi penuh ketika oksigennya tak ada, dan apa yang
        diperbuatnya pada \omterm{def:b1:respiration-fermentation:krebs}{daur Krebs}.
  \item Nyatakan hasilnya: \omterm{def:b1:water-small-molecules:atp}{ATP} tiap glukosa dengan dan tanpa udara,
        rasionya, serta \omterm{thm:b1:respiration-fermentation:chemiosmosis}{rasio P/O} NADH dan $\mathrm{FADH_2}$.
\end{enumerate}
\end{problem}
