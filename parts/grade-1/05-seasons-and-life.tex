\chapter{Seasons and Life}\label{ch:g1:seasons}

The year turns. Weather changes. Trees may lose leaves. Birds may
fly away and return. \omterm{def:g1:living-nonliving:living}{Living things} feel the \omterm{def:g1:seasons:season}{seasons} and change with
them.

\section{Four seasons}

\begin{definition}[Season]\label{def:g1:seasons:season}
A \emph{season}\index{season} is a time of year with its own usual
weather. In many places there are four seasons: \omterm{def:g1:seasons:four}{spring}, \omterm{def:g1:seasons:four}{summer},
\omterm{def:g1:seasons:four}{autumn} and \omterm{def:g1:seasons:four}{winter}.
\end{definition}

\omimg{four-seasons.jpg}{0.52\linewidth}{\omterm{def:g1:seasons:four}{Spring} buds, \omterm{def:g1:seasons:four}{summer} green,
\omterm{def:g1:seasons:four}{autumn} colours, \omterm{def:g1:seasons:four}{winter} rest --- one tree through the year.}

\begin{definition}[The four seasons]\label{def:g1:seasons:four}
\begin{itemize}
  \item \emph{spring}\index{spring}: warmer days; \omterm{def:g1:plants-animals:plant}{plants} grow new
        leaves and flowers;
  \item \emph{summer}\index{summer}: warm or hot; many \omterm{def:g1:plants-animals:plant}{plants} are
        full and green;
  \item \emph{autumn}\index{autumn} (fall): cooler; some leaves
        change colour and fall;
  \item \emph{winter}\index{winter}: cold in many places; some
        \omterm{def:g1:plants-animals:plant}{plants} rest; some \omterm{def:g1:plants-animals:animal}{animals} sleep or hide.
\end{itemize}
\end{definition}

\section{How life changes}

\begin{example}[Plants through the year]\label{ex:g1:seasons:plants}
In \omterm{def:g1:seasons:four}{spring}, buds open. In \omterm{def:g1:seasons:four}{summer}, trees give shade. In \omterm{def:g1:seasons:four}{autumn}, some
leaves fall. In \omterm{def:g1:seasons:four}{winter}, many trees look bare --- but the tree is
still alive, resting for \omterm{def:g1:seasons:four}{spring}.
\end{example}

\begin{example}[Animals through the year]\label{ex:g1:seasons:animals}
Some birds fly to warmer places for \omterm{def:g1:seasons:four}{winter} and return in \omterm{def:g1:seasons:four}{spring}.
Some \omterm{def:g1:plants-animals:animal}{animals} grow thicker fur. Squirrels store food. Frogs may sleep
in the mud until warmer days.
\end{example}

\begin{method}[Watch a season]\label{met:g1:seasons:watch}
\begin{enumerate}
  \item Choose one tree or bush near you.
  \item Once a month, look at its leaves, flowers or fruit.
  \item Draw or write what you see: green? bare? flowers?
  \item Link the change to the \omterm{def:g1:seasons:season}{season} name.
\end{enumerate}
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g1:seasons:1}
Name the four \omterm{def:g1:seasons:season}{seasons} in order, starting with \omterm{def:g1:seasons:four}{spring}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:seasons:2}
In which \omterm{def:g1:seasons:season}{season} do many trees grow new green leaves?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:seasons:3}
True or false? ``In \omterm{def:g1:seasons:four}{winter} a bare tree is dead.''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:seasons:4}
What might a bird do when \omterm{def:g1:seasons:four}{winter} is very cold where it lives?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:seasons:5}
Match: \omterm{def:g1:seasons:four}{spring}, \omterm{def:g1:seasons:four}{summer}, \omterm{def:g1:seasons:four}{autumn}, \omterm{def:g1:seasons:four}{winter} --- to: falling leaves; new
flowers; hottest time; coldest time (in many places).
\end{exercise}
