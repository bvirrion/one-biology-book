\chapter{العظام والعضلات}\label{ch:g3:bones-and-muscles}

اقفز ثم اهبط: لقد عمل شيء في داخلك كآلة كاملة. قضبان صلبة حفظت
شكلك، \omterm{def:g1:my-body:joint}{ومفاصل} أتاحت لرجليك أن تنطويا وتنتفضا، ومئات المحركات
الصغيرة سحبت في اللحظة المطلوبة تمامًا. فالقضبان عظامك، \omterm{prop:g3:bones-and-muscles:joints}{والمفاصل}
مفاصلك، والمحركات عضلاتك --- وهي عتاد الجسم لكل حركة سيقوم بها
يومًا.

\section{الهيكل العظمي}

\begin{definition}[الهيكل العظمي والعظام]\label{def:g3:bones-and-muscles:skeleton}
\emph{الهيكل العظمي}\index{هيكل عظمي} إطار من نحو $200$
\emph{عظمة}\index{عظم} داخل الجسم. ومناطقه الرئيسة:
\begin{itemize}
  \item \emph{الجمجمة}\index{جمجمة}، وهي خوذة صلبة من عظم حول
        الدماغ؛
  \item و\emph{العمود الفقري}\index{عمود فقري}، وهو عمود قابل
        للانثناء من عظام صغيرة مرصوفة من العنق إلى الوركين، يحفظ
        \omterm{def:g1:my-body:parts}{البدن} منتصبًا؛
  \item و\emph{الأضلاع}\index{ضلع}، وهي عظام منحنية تكوّن قفصًا حول
        القلب والرئتين؛
  \item و\emph{عظام \omterm{def:g1:my-body:parts}{الأطراف}}: عظام طويلة في \omterm{def:g1:my-body:parts}{الذراعين} \omterm{def:g1:my-body:parts}{والرجلين} ---
        وعظم الفخذ أطول عظام الجسم.
\end{itemize}
\end{definition}

\begin{example}[تحسّس إطارك]\label{ex:g3:bones-and-muscles:feel}
لا حاجة إلى أشعة: اضغط برفق فيبلّغك \omterm{def:g3:bones-and-muscles:skeleton}{الهيكل العظمي}. \omterm{def:g1:my-body:joint}{مفاصل} الأصابع؛
وحرف قصبة \omterm{def:g1:plants-around-us:parts}{الساق}؛ وسلسلة النتوءات الصغيرة في وسط ظهرك --- وهي
\omterm{def:g3:bones-and-muscles:skeleton}{العمود الفقري}، \omterm{def:g3:bones-and-muscles:skeleton}{عظمة} صغيرة بعد أخرى؛ \omterm{def:g3:bones-and-muscles:skeleton}{والأضلاع} التي تستطيع عدّها
تحت أصابعك. فالإطار تحت \omterm{def:g1:the-five-senses:sense}{الجلد} مباشرة.
\end{example}

\begin{remark}[صلب من خارجه، حي في داخله]\label{rem:g3:bones-and-muscles:alive}
\omterm{def:g3:bones-and-muscles:skeleton}{العظم} ليس مادة ميتة كعود الطباشير: بل هو حي، يغذّيه الدم، وينمو
معك وأنت تنمو --- وهو قادر على إصلاح نفسه. \omterm{def:g1:my-body:parts}{والذراع} المكسورة في
جبيرة تلتئم لأن \omterm{def:g3:bones-and-muscles:skeleton}{العظم} نفسه يعيد بناء الكسر ببطء. أما عيدان
الطباشير فلا تفعل ذلك أبدًا.
\end{remark}

\section{المفاصل، حيث ينثني الإطار}

\begin{proposition}[قضبان صلبة ومواضع انثناء]\label{prop:g3:bones-and-muscles:joints}
\omterm{def:g3:bones-and-muscles:skeleton}{العظام} نفسها لا تنثني أبدًا --- \omterm{def:g3:bones-and-muscles:skeleton}{والعظم} المنثني \omterm{def:g3:bones-and-muscles:skeleton}{عظم} مكسور. وكل
انثناء في الجسم يحدث عند \emph{المفاصل}\index{مفصل}، حيث تلتقي
عظمتان: \omterm{def:g1:my-body:joint}{الكتف}، \omterm{def:g1:my-body:joint}{والمرفق}، \omterm{def:g1:my-body:joint}{والرسغ}، والورك، \omterm{def:g1:my-body:joint}{والركبة}، \omterm{def:g1:my-body:joint}{والكاحل}، \omterm{def:g1:my-body:joint}{ومفاصل}
\omterm{def:g3:bones-and-muscles:skeleton}{العمود الفقري} الصغيرة الكثيرة التي تتجمع انثناءاتها الصغيرة فتصير
ظهرًا لدنًا.
\end{proposition}
\admitted

\begin{example}[حيلة العمود الفقري]\label{ex:g3:bones-and-muscles:spinetrick}
لا \omterm{def:g3:bones-and-muscles:skeleton}{عظمة} واحدة في ظهرك تستطيع أن تلتف --- ومع ذلك تستطيع أن تنكفئ
إلى الأمام حتى تكاد تصير كرة. \omterm{def:g3:bones-and-muscles:skeleton}{فالعمود الفقري} عمود من أكثر من $20$
\omterm{def:g3:bones-and-muscles:skeleton}{عظمة} صغيرة، تدور كل واحدة درجات قليلة على جارتيها: انثناءات صغيرة
كثيرة تصنع انثناءً كبيرًا واحدًا. ولهذا أيضًا ينبغي أن تثني ركبتيك
لا ظهرك لترفع ثقيلًا --- فالركبتان مبنيتان للطيّات الكبيرة، \omterm{def:g3:bones-and-muscles:skeleton}{والعمود
الفقري} للرقيقة منها.
\end{example}

\section{العضلات، وهي المحرّكات}

\begin{definition}[العضلة]\label{def:g3:bones-and-muscles:muscle}
\emph{العضلة}\index{عضلة} محرّك لحمي متصل \omterm{def:g3:bones-and-muscles:skeleton}{بالعظام}. وتستطيع العضلة
أن \emph{تنقبض}\index{انقباض} --- أي أن تقصر وتغلظ --- وهي إذ
تنقبض \emph{تسحب} \omterm{def:g3:bones-and-muscles:skeleton}{العظمة} المتصلة بها. والعضلات تسحب فحسب؛ ولا
عضلة تستطيع أن تدفع.
\end{definition}

\begin{proposition}[العمل أزواجًا]\label{prop:g3:bones-and-muscles:pairs}
لما كانت \omterm{def:g3:bones-and-muscles:muscle}{العضلات} تسحب فحسب، احتاج تحريك \omterm{def:g1:my-body:joint}{المفصل} في الاتجاهين إلى
\emph{زوج}: \omterm{def:g3:bones-and-muscles:muscle}{فعضلة} أمام \omterm{def:g1:my-body:parts}{الذراع} تثني \omterm{def:g1:my-body:joint}{المرفق}، وشريكتها خلفه تمدّه؛
وترتخي كل واحدة حين تسحب الأخرى. ويكاد كل \omterm{def:g1:my-body:joint}{مفصل} في الجسم يتحرك
بأزواج كهذه.
\end{proposition}
\admitted

\begin{example}[تحسّس الزوج وهو يعمل]\label{ex:g3:bones-and-muscles:pair}
أمسك عضدك بيدك الأخرى واثنِ \omterm{def:g1:my-body:joint}{المرفق} ببطء: \omterm{def:g3:bones-and-muscles:muscle}{فالعضلة} الأمامية تنتفخ
وتقسو --- إنها \omterm{def:g3:bones-and-muscles:muscle}{تنقبض}. والآن مُدّ \omterm{def:g1:my-body:parts}{الذراع}: فتلين الأمامية، وتشتدّ
شريكتها خلف \omterm{def:g1:my-body:parts}{الذراع} بدلها. وبذلك تكون قد تحسّست
\cref{prop:g3:bones-and-muscles:pairs} تحدث.
\end{example}

\begin{method}[العناية بالعظام والعضلات]\label{met:g3:bones-and-muscles:care}
\begin{enumerate}
  \item تحرّك كل يوم --- \omterm{def:g3:bones-and-muscles:skeleton}{فالعظام} تقوى بالاستعمال، \omterm{def:g3:bones-and-muscles:muscle}{والعضلات} كذلك؛
  \item وكُلْ أغذية البنّائين: الألبان \omterm{def:g3:bones-and-muscles:skeleton}{للعظام}، ووجبات متنوعة
        (\cref{ch:g2:teeth-and-eating-well})؛
  \item وسخّن قبل التمرين الشاق، لتمتدّ \omterm{def:g3:bones-and-muscles:muscle}{العضلات} بدل أن تتمزق؛
  \item وارفع الأثقال بركبتين منثنيتين وظهر مستقيم؛
  \item والبس الخوذة والواقيات حيث يُرجَّح السقوط: \omterm{def:g3:bones-and-muscles:skeleton}{فالعظام} تُصلح
        نفسها، لكن ببطء.
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:1}
سمِّ مناطق \omterm{def:g3:bones-and-muscles:skeleton}{الهيكل العظمي} الأربع في
\cref{def:g3:bones-and-muscles:skeleton}، وما تحميه أو تحمله كل
واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:2}
كم \omterm{def:g3:bones-and-muscles:skeleton}{عظمة} \omterm{def:g3:bones-and-muscles:skeleton}{للهيكل العظمي} تقريبًا؟ وأيها أطولها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:3}
أتستطيع \omterm{def:g3:bones-and-muscles:skeleton}{العظمة} أن تنثني؟ وأين يحدث كل انثناء في الجسم؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:4}
يستطيع \omterm{def:g3:bones-and-muscles:skeleton}{العمود الفقري} أن يلتف مع أن ولا واحدة من عظامه تستطيع ذلك.
اشرح حيلته.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:5}
ماذا تفعل \omterm{def:g3:bones-and-muscles:muscle}{العضلة} حين \omterm{def:g3:bones-and-muscles:muscle}{تنقبض}؟ وما الذي لا تستطيعه أي \omterm{def:g3:bones-and-muscles:muscle}{عضلة} أبدًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:6}
لماذا يحتاج ثني \omterm{def:g1:my-body:joint}{المرفق} ومدّه إلى عضلتين؟ اذكر ماذا تفعل كل واحدة
في كل لحظة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:7}
كيف تلتئم \omterm{def:g1:my-body:parts}{ذراع} مكسورة في جبيرة؟ وعلام يدل هذا في شأن \omterm{def:g3:bones-and-muscles:skeleton}{العظم} وفق
\cref{rem:g3:bones-and-muscles:alive}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:bones-and-muscles:8}
جرّب: طبّق اختبار المسك في \cref{ex:g3:bones-and-muscles:pair}
على ذراعك. صف ماذا يفعل مقدَّم \omterm{def:g1:my-body:parts}{الذراع} حين تثني، وحين تمدّ.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:9}
لماذا يقول الكبار ``ارفع بركبتيك لا بظهرك''؟ أجب وفق
\cref{ex:g3:bones-and-muscles:spinetrick}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:bones-and-muscles:10}
دمية على خيوط يمكن أن تُجعل تلوّح: وكل خيط لا يستطيع إلا أن
\emph{يسحب} ذراعها في اتجاه واحد. فكيف يطابق هذا حال \omterm{def:g3:bones-and-muscles:muscle}{العضلات}
تمامًا، وما جواب صانع الدمى عنه؟ قارن ذلك
مع \cref{prop:g3:bones-and-muscles:pairs}.
\end{exercise}
