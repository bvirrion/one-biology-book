\chapter{Bagaimana Spesies Berubah}\label{ch:g9:how-species-change}

Aula buktinya menegakkan kenyataannya; bab ini membangun
mesinnya. Bagiannya semua sudah ada di tanganmu: \omterm{prop:g9:genes-and-dna:explains}{mutasi} yang
menuliskan ejaan baru
(\cref{prop:g9:genes-and-dna:explains}), pengocokan yang
membagikan tumpukan yang unik
(\cref{prop:g9:unique-individuals:theorem}), penahan
\omterm{def:g6:exploring-our-environment:components}{lingkungan} yang memungut sebagian besar kelebihan tiap
angkatannya
(\cref{prop:g8:reproduction-and-environment:limits}).
Dirakit, semuanya berjalan sendiri --- berjalan sendiri
itulah \omterm{prop:g9:how-species-change:selection}{seleksi alam}, dan itulah gagasan sederhana yang
paling berakibat dalam \omterm{def:g1:living-or-not:living}{biologi}.

\section{Mesinnya, dirakit}

\begin{proposition}[Seleksi alam]\label{prop:g9:how-species-change:selection}
Tiga kenyataan, tak satu pun dapat disangkal, dan akibatnya:
\begin{enumerate}
  \item \emph{keragaman}: individu sebuah \omterm{def:g8:reproduction-and-environment:population}{populasi} berbeda
        dalam \omterm{def:g9:heredity-and-traits:heredity}{sifat turunannya}
        (\cref{prop:g9:unique-individuals:reserve}
        menyimpannya);
  \item \emph{kelebihan produksi dan penahannya}: anak yang
        dihasilkan jauh lebih banyak daripada yang diizinkan
        \omterm{def:g6:exploring-our-environment:components}{lingkungannya} bertahan hidup --- penahannya memungut
        kelebihannya (\cref{ex:g8:reproduction-and-environment:twosprings});
  \item \emph{pemungutan yang tidak merata}: penahannya
        tidak memungut secara acak --- individu yang \omterm{def:g9:heredity-and-traits:heredity}{sifat
        turunannya} cocok dengan keadaan sekarang lolos
        darinya sedikit lebih sering, sehingga meninggalkan
        lebih banyak keturunan, \emph{yang mewarisi \omterm{def:g9:heredity-and-traits:traits}{sifat} itu}.
\end{enumerate}
Akibatnya: angkatan demi angkatan, bagian \omterm{def:g9:genes-and-dna:gene}{alel} yang cocok di
dalam \omterm{def:g8:reproduction-and-environment:population}{populasinya} bertumbuh. Hanyutnya susunan turunan
sebuah \omterm{def:g8:reproduction-and-environment:population}{populasi}, yang digerakkan oleh tidak lain daripada
keragaman yang bertemu penahannya, itulah \emph{seleksi
alam}\index{seleksi alam} --- dan waktu yang cukup
menjadikannya \emph{evolusi}\index{evolusi}.
\end{proposition}
\admitted

\begin{example}[Ngengatnya, dimesinkan]\label{ex:g9:how-species-change:moths}
Ngengat berbintik pada \cref{ex:g9:evidence-of-evolution:live},
bagian demi bagian. Keragamannya: bentuk turunan yang pucat
dan yang gelap. Penahannya: \omterm{prop:g3:sorting-living-things:groups}{burung} memburu ngengat dengan
penglihatan pada kulit kayunya. Pemungutan yang tidak merata:
pada \omterm{def:g1:plants-around-us:parts}{batang} yang menghitam jelaga, bentuk pucatnyalah yang
terlihat --- lebih banyak dimakan, sehingga bagian \omterm{def:g9:genes-and-dna:gene}{alel}
gelapnya mendaki menuju hutan gelap yang teramati; udaranya
bersih, kulit kayunya memucat, dan mesin yang sama berjalan
terbalik. Tidak ada ngengat yang berubah; susunan
\emph{\omterm{def:g8:reproduction-and-environment:population}{populasinya}} yang berubah --- seleksinya menyunting
frekuensi tumpukannya, bukan individunya.
\end{example}

\begin{remark}[Apa yang bukan seleksi]\label{rem:g9:how-species-change:not}
Tiga kekeliruan yang bertahan, dipensiunkan:
\begin{itemize}
  \item \emph{tanpa niat}: keadaannya menyeleksi dengan cara
        \omterm{def:g6:exploring-our-environment:conditions}{kelembapan} ``menyeleksi'' kutu kayunya
        (\cref{ex:g6:exploring-our-environment:chooser}) ---
        lewat akibat, bukan pilihan; tidak ada yang merencanakan;
  \item \emph{tanpa kesempurnaan}: seleksinya hanya
        membandingkan ragam yang \emph{ada} --- ia menemukan
        yang lebih baik di antara yang tersedia, bukan yang
        terbaik yang dapat dibayangkan (alih tugas yang
        janggal pada rancangan \omterm{def:g1:my-body:parts}{anggota gerak} bersamanya,
        \cref{exo:g9:evidence-of-evolution:7}, adalah tanda
        tangannya);
  \item \emph{tanpa pewarisan jerih payah}: \omterm{def:g1:my-body:parts}{lengan} si pandai
        besi mati bersama si pandai besi
        (\cref{prop:g9:heredity-and-traits:sorting});
        seleksinya hanya bekerja pada apa yang diangkut
        \omterm{def:g8:sexual-reproduction:gametes}{gametnya}. Jerapah tidak memanjangkan lehernya dengan
        meregang --- ragam yang berleher lebih panjanglah
        yang makan dan berbiak lebih baik daripada yang pendek.
\end{itemize}
\end{remark}

\section{Mesinnya diamati dan dimanfaatkan}

\begin{example}[Seleksi di rumah sakit]\label{ex:g9:how-species-change:resistance}
Kasus kedokterannya, dimesinkan untuk dipakai
\cref{ch:g9:microbes-and-infection}: sebuah \omterm{def:g8:reproduction-and-environment:population}{populasi} \omterm{def:g6:producing-our-food:fermentation}{mikroba}
memuat mutan tahan yang jarang (tetesan \omterm{prop:g9:genes-and-dna:explains}{mutasinya}); sebuah
antibiotik adalah penahannya; ia memungut yang rentan lalu
melepaskan yang tahan, dan yang tahan itulah yang mewarisi
bangsalnya. Hitungan hari, bukan milenium --- angkatannya
berputar dalam hitungan jam, dan kecepatan seleksinya
berskala bersamanya. Setiap aturan tentang menuntaskan
kursus antibiotik dan tidak menaburkan antibiotik ke dalam
pertanian adalah pengelolaan mesinnya: lemahkan penahannya
secara compang-camping, dan kamu membiakkan apa yang hendak
kamu bunuh.
\end{example}

\begin{example}[Seleksi dengan tangan manusia pada penahannya]\label{ex:g9:how-species-change:artificial}
Pemulia pada \cref{pb:g8:sexual-reproduction:1} menjalankan
mesin yang sama dengan dirinya sendiri sebagai penahannya
--- mempertahankan yang renyah dan tahan, menyingkirkan
sisanya. Setiap ras anjing, setiap varietas \omterm{def:g6:producing-our-food:farming}{tanaman pangan},
setiap galur sapi perah adalah \emph{seleksi buatan} yang
menumpuk: bukti kekuatan mesinnya pada skala waktu manusia,
dari serigala menjadi anjing pangkuan dalam beberapa ribu
angkatan. Darwin --- naturalis yang merakit hujah bab ini
--- membuka buku besarnya justru dengan perbandingan itu.
\end{example}

\begin{proposition}[Dari perubahan menjadi spesies baru]\label{prop:g9:how-species-change:speciation}
Seleksinya menjelaskan sebuah \omterm{def:g8:reproduction-and-environment:population}{populasi} yang berubah; satu
bahan lagi menjelaskan \omterm{def:g8:reproduction-and-environment:population}{populasi} yang \emph{terbelah}.
Pisahkan dua \omterm{def:g8:reproduction-and-environment:population}{populasi} satu \omterm{def:g5:biodiversity:species}{spesies} --- lautnya naik, sebuah
pegunungan membelah, sebuah pulau menerima orang terdampar
(\cref{exo:g6:how-plants-spread:11}) --- dan masing-masing
menjalankan mesinnya di bawah keadaannya sendiri, menumpuk
\omterm{def:g9:genes-and-dna:gene}{alelnya} sendiri. Diberi pemencaran yang cukup, keduanya
tidak lagi saling berbiak di tempat mereka bertemu: menurut
rumusan \cref{def:g5:biodiversity:species} sendiri, dua
\omterm{def:g5:biodiversity:species}{spesies} berdiri di tempat satu \omterm{def:g5:biodiversity:species}{spesies} dahulu berdiri.
Percabangan pohonnya (\cref{def:g6:classification-kinship:tree})
justru adalah pembelahan semacam itu, sebanyak yang
dikandung waktu yang dalam.
\end{proposition}
\admitted

\begin{method}[Menjalankan mesinnya pada kasus apa pun]\label{met:g9:how-species-change:run}
Untuk setiap contoh \omterm{prop:g9:how-species-change:selection}{evolusi} yang diklaim:
\begin{enumerate}
  \item sebutkan \emph{keragamannya} lalu periksa bahwa ia
        turunan (\cref{met:g9:heredity-and-traits:audit});
  \item sebutkan \emph{penahannya}, dan bagaimana ia
        memungut tidak merata menembus keragaman itu;
  \item ikuti \emph{pembukuannya}: \omterm{def:g9:genes-and-dna:gene}{alel} siapa yang menambah
        bagiannya pada angkatan berikutnya?
  \item untuk sebuah pembelahan, tambahkan
        \emph{pemisahannya} dan keadaan yang memencar;
  \item periksa skala waktunya: angkatan, bukan tahun,
        adalah detak mesinnya.
\end{enumerate}
\end{method}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:1}
Nyatakan ketiga kenyataan \omterm{prop:g9:how-species-change:selection}{seleksi alamnya} dan akibatnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:2}
Pada kasus ngengatnya, sebutkan keragamannya, penahannya,
dan pemungutan yang tidak meratanya --- ke kedua arahnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:3}
Apa yang berubah di bawah seleksinya --- individunya, atau
\omterm{def:g8:reproduction-and-environment:population}{populasinya}? Jelaskan.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:4}
Betulkan, dengan mekanismenya di tangan: ``jerapah
meregangkan lehernya lalu meneruskan regangannya''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:5}
Mengapa ketahanan antibiotik berevolusi dalam hitungan hari
sementara ngengatnya butuh berdasawarsa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:how-species-change:6}
Apa yang sama antara pemulia apelnya dan \omterm{prop:g3:sorting-living-things:groups}{burung} pemakan

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:how-species-change:7}
Mengapa seleksinya tak dapat menghasilkan makhluk ``terbaik

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:how-species-change:8}

\omterm{def:g2:from-seed-to-plant:seed}{bijinya} mengeras, paruh besar memecahkan apa yang tak dapat

\begin{exercise}[$\star\star$]\label{exo:g9:how-species-change:9}
Rakitlah pembentukan \omterm{def:g5:biodiversity:species}{spesies} bagi orang terdampar pulau pada

\begin{exercise}[$\star\star$]\label{exo:g9:how-species-change:10}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:how-species-change:12}

berbalik?
\end{exercise}

\section{Soal: Buku Catatan Kepulauannya}

\begin{problem}[{Soal akhir pekan --- dua abad di sebuah
kepulauan gunung api}]\label{pb:g9:how-species-change:1}
Sebuah kepulauan (gabungan, terilhami sejarah): pulau gunung
api, tidak pernah menyatu dengan daratan utamanya,
menampung \omterm{prop:g3:sorting-living-things:groups}{burung} mirip finch yang bentuk paruhnya berbeda
dari pulau ke pulau --- dan berbeda dari satu-satunya
kerabat daratan utamanya yang jelas. Buku catatan para
naturalis membentang dua abad.

\medskip\noindent\textbf{Bagian I --- Polanya.}
\begin{enumerate}
  \item Pulaunya hanya memuat \omterm{def:g5:biodiversity:species}{spesies} yang dapat
        menyeberangi lautnya --- \omterm{prop:g3:sorting-living-things:groups}{burung}, \omterm{def:g2:from-seed-to-plant:seed}{biji} hanyut, satu
        kelelawar, tanpa \omterm{prop:g3:sorting-living-things:groups}{mamalia} darat atau kodok asli. Bab
        yang mana yang meramalkan daftar penumpangnya
        (\cref{exo:g6:how-plants-spread:11})?
  \item Semua \omterm{prop:g3:sorting-living-things:groups}{burung} finch kepulauannya menyerupai satu
        \omterm{def:g5:biodiversity:species}{spesies} daratan utamanya lebih daripada \omterm{prop:g3:sorting-living-things:groups}{burung} lain
        mana pun di Bumi. Apa yang disimpulkan penalaran
        \omterm{prop:g6:classification-kinship:kinship}{kekerabatannya}
        (\cref{prop:g6:classification-kinship:kinship})
        tentang asalnya?
  \item Paruhnya berbeda menurut makanan pulaunya: pemecah
        \omterm{def:g2:from-seed-to-plant:seed}{biji} di pulau berbiji, penusuk di pulau berkaktus.
        Bacalah kecocokannya dengan
        \cref{prop:g4:adapted-to-their-world:answers} ---
        lalu katakan apa yang ditambahkan bab ini yang tak
        dapat ditambahkan bab \omterm{def:g4:adapted-to-their-world:adaptation}{adaptasinya}.
  \item Mengapa pulau gunung api yang tak pernah menyatu
        adalah panggung utama mesinnya? Sebutkan bahan
        pembentukan \omterm{def:g5:biodiversity:species}{spesies} yang dipasoknya secara borongan.
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Tahun buku
catatan mesinnya.}
\begin{enumerate}[resume]
  \item Sebuah tahun kemarau: \omterm{def:g2:from-seed-to-plant:seed}{biji} yang lunak gagal, \omterm{def:g2:from-seed-to-plant:seed}{biji}
        yang keras bertahan; jangka sorong para pencincin
        memperlihatkan anak yang menetas tahun berikutnya
        rata-rata berparuh lebih dalam. Jalankan pembukuan
        mesinnya menembus kalimat itu.
  \item Tahun basah menyusul, \omterm{def:g2:from-seed-to-plant:seed}{biji} yang lunak kembali, dan
        rata-ratanya hanyut kembali. Apa yang dibuktikan
        pembalikan itu tentang ``arah'' seleksinya?
  \item \omterm{prop:g3:sorting-living-things:groups}{Burung} finch satu pulau, yang dipindahkan badai ke
        pulau lain, berbiak dengan buruk bersama \omterm{def:g8:reproduction-and-environment:population}{populasi}
        setempatnya. Proposisi mana yang sedang disaksikan?
  \item Sebuah buku catatan dari zaman kreasionis membantah:
        ``tidak seorang pun pernah melihat satu \omterm{def:g5:biodiversity:species}{spesies}
        menjadi \omterm{def:g5:biodiversity:species}{spesies} lain.'' Jawablah dengan skala
        waktunya dan cara saksi yang bertemu
        (\cref{met:g9:evidence-of-evolution:weigh}).
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Ceramah
sintesisnya.}
\begin{enumerate}[resume]
  \item Sampaikan kisah kepulauannya dalam lima langkah
        mesinnya (\cref{met:g9:how-species-change:run}),
        dari tibanya orang terdampar sampai \omterm{def:g5:biodiversity:species}{spesies} pulaunya.
  \item Jelaskan mengapa \omterm{prop:g3:sorting-living-things:groups}{burung} finch tiap pulaunya adalah
        ``yang lebih baik di antara yang tersedia, bukan yang
        terbaik yang dapat dibayangkan'' --- apa yang dapat
        ditawarkan tumpukan tiap pulaunya kepada seleksinya?
  \item Hubungkan kepulauannya dengan bangsal rumah sakit
        pada \cref{ex:g9:how-species-change:resistance}:
        mesin yang sama, laju detak yang berbeda ---
        nyatakan apa yang menetapkan lajunya.
  \item Tutuplah ceramahnya dengan perbandingan Darwin yang
        dibuat tersurat: apa yang sama antara tangan sang
        penggemar merpati dan \omterm{def:g2:from-seed-to-plant:seed}{biji} keras kemaraunya.
\end{enumerate}
\end{problem}
