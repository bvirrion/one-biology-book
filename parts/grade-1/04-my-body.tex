\chapter{My Body}\label{ch:g1:my-body}

Stand in front of a mirror and wave: a whole living body waves
back. It has been growing since before you were born, it bends in
some places and not in others, and it is yours for life. Time to
look at it the way we looked at the cat and the daisy --- part by
part.

\section{The parts of my body}

\begin{definition}[Parts of the human body]\label{def:g1:my-body:parts}
The human body has three main parts:
\begin{enumerate}
  \item the \emph{head}\index{head}, which carries the face, the
        eyes, the ears, the nose and the mouth;
  \item the \emph{trunk}\index{trunk (body)}, the middle of the
        body --- chest, belly and back;
  \item the \emph{limbs}\index{limb}: two \emph{arms}\index{arm}
        ending in hands, two \emph{legs}\index{leg} ending in feet.
\end{enumerate}
The head sits on the \emph{neck}\index{neck}, which joins it to the
trunk.
\end{definition}

\begin{example}[Same plan as the cat?]\label{ex:g1:my-body:cat}
Compare yourself with the cat: head, body, four \omterm{def:g1:my-body:parts}{limbs} --- the plan
looks alike. But the cat walks on all four; you stand on two \omterm{def:g1:my-body:parts}{legs}
and keep two \omterm{def:g1:my-body:parts}{arms} free for carrying, drawing and waving. And where
the cat has a tail, you have none.
\end{example}

\section{Bending: the joints}

\begin{definition}[Joint]\label{def:g1:my-body:joint}
A \emph{joint}\index{joint} is a place where the body can bend or
turn. The \emph{elbow}\index{elbow} bends the \omterm{def:g1:my-body:parts}{arm}, the
\emph{knee}\index{knee} bends the \omterm{def:g1:my-body:parts}{leg}, the
\emph{shoulder}\index{shoulder} turns the whole \omterm{def:g1:my-body:parts}{arm}, the
\emph{wrist}\index{wrist} and the \emph{ankle}\index{ankle} bend
the hand and the foot. Between two joints, the body stays straight.
\end{definition}

\begin{example}[Try it: the stiff-knee walk]\label{ex:g1:my-body:stiff}
Try walking without bending your \omterm{def:g1:my-body:joint}{knees}. It works --- just barely
--- with stiff little duck steps. Now try picking a pencil off the
floor without bending \omterm{def:g1:my-body:joint}{knees}, hips or back. \omterm{def:g1:my-body:joint}{Joints} are what make the
body's movements smooth, quick and easy.
\end{example}

\begin{remark}[Right and left]\label{rem:g1:my-body:sides}
The body has two sides, right and left, almost like a mirror: one
\omterm{def:g1:my-body:parts}{arm} and one \omterm{def:g1:my-body:parts}{leg} on each side, one eye and one ear on each side of
the head. Knowing your right hand from your left is body knowledge
too --- it tells people which way you mean.
\end{remark}

\section{My body grows}

\begin{proposition}[Growing up]\label{prop:g1:my-body:growth}
A child's body \emph{grows}\index{growing}: it gets taller and
heavier, year after year, until grown-up size. Growing is slow and
steady --- too slow to feel, easy to measure.
\end{proposition}
\admitted

\begin{example}[Marks on the door frame]\label{ex:g1:my-body:doorframe}
Stand against the door frame; a grown-up marks your height with a
pencil. Do it again after some months: the new mark is higher! You
never felt yourself growing, but the two marks, one above the
other, show it plainly.
\end{example}

\begin{omfigure}
\includegraphics[width=0.46\linewidth]{images/book1/ai/g1-body-doorframe.jpg}

{\small Measuring day at the door frame: the flat book brings the
top of the head straight to the wood.}
\end{omfigure}

\begin{method}[Measuring my height]\label{met:g1:my-body:measure}
\begin{enumerate}
  \item Take off your shoes;
  \item stand straight, back and heels against the wall, looking
        ahead;
  \item a helper lays a flat book on your head, level, touching
        the wall;
  \item mark the spot under the book, and write the date next to
        it.
\end{enumerate}
Measured the same way each time, the marks can be trusted --- and
compared.
\end{method}

\begin{remark}[Every body is different]\label{rem:g1:my-body:different}
In one class, some children are taller, some shorter; some grow
sooner, some later. All of that is normal. Bodies are like faces:
built on the same plan, and each one different. The marks on your
door frame are yours --- they are not for racing the neighbor.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:g1:my-body:1}
Name the three main parts of the human body. What joins the head
to the trunk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:2}
Where does each \omterm{def:g1:my-body:parts}{limb} end --- what is at the end of an \omterm{def:g1:my-body:parts}{arm}, and at
the end of a \omterm{def:g1:my-body:parts}{leg}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:3}
Which \omterm{def:g1:my-body:joint}{joint} bends: (a) the \omterm{def:g1:my-body:parts}{arm} in the middle? (b) the \omterm{def:g1:my-body:parts}{leg} in the
middle? (c) the hand at the end of the \omterm{def:g1:my-body:parts}{arm}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:4}
Touch your right ear with your left hand. Which \omterm{def:g1:my-body:joint}{joints} did your
\omterm{def:g1:my-body:parts}{arm} just use? Name at least two.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:5}
In \cref{ex:g1:my-body:cat}, find one way your body is like the
cat's and two ways it is different.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:6}
Why does the helper in \cref{met:g1:my-body:measure} lay a flat
book on your head instead of just looking?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:7}
Two marks on the door frame: last year's, and today's a little
higher. What do the marks show? Did you feel it happening?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:my-body:8}
Try it: do the \omterm{def:g1:my-body:joint}{stiff-knee} walk of \cref{ex:g1:my-body:stiff}, then
try clapping without bending your \omterm{def:g1:my-body:joint}{elbows}. Which everyday movements
turn out to need \omterm{def:g1:my-body:joint}{joints}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:9}
Mia is the smallest in her class and worries about it. What does
\cref{rem:g1:my-body:different} answer her?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:my-body:10}
A puppet made of one wooden stick cannot sit on a chair. Where
would you cut the stick and add turning points so it can sit? Name
the body \omterm{def:g1:my-body:joint}{joints} your cuts copy.
\end{exercise}
