\chapter{Os animais ao longo das estações}\label{ch:g3:animals-through-seasons}

Em outubro as andorinhas se enfileiram nos fios e, numa manhã, já
foram embora. O ouriço debaixo da cerca viva se enrola no ninho de
\omterm{def:g1:plants-around-us:parts}{folhas} dele e não vai aparecer antes da primavera. A raposa, por sua
vez, sai por aí em toda nevada. O inverno é uma estação dura ---
fria e, acima de tudo, pobre em alimento --- e os \omterm{def:g1:animals-around-us:animal}{animais} têm três
grandes jeitos de atravessar essa estação.

\section{O inverno, a estação da escassez}

\begin{proposition}[O problema duplo do inverno]\label{prop:g3:animals-through-seasons:problem}
O inverno traz aos \omterm{def:g1:animals-around-us:animal}{animais} dois problemas de uma vez:
\begin{enumerate}
  \item o \emph{frio}, que rouba o calor do corpo mais depressa ---
        e manter-se quente custa \omterm{prop:g3:balanced-diet:jobs}{energia}
        (\cref{prop:g3:balanced-diet:jobs});
  \item o \emph{alimento escasso}: as \omterm{def:g1:plants-around-us:plant}{plantas} param de crescer, os
        \omterm{prop:g3:sorting-living-things:groups}{insetos} somem, os primeiros elos das \omterm{def:g3:food-chains:chain}{cadeias alimentares} do
        \cref{ch:g3:food-chains} ficam ralos --- e assim a escassez
        sobe até todo mundo.
\end{enumerate}
Mais calor necessário, menos alimento para pagar por ele: esse é o
problema do inverno que todo \omterm{def:g1:animals-around-us:animal}{animal} precisa resolver.
\end{proposition}
\admitted

\begin{example}[Para onde foi o alimento de quem come inseto?]\label{ex:g3:animals-through-seasons:insects}
A andorinha se alimenta de \omterm{prop:g3:sorting-living-things:groups}{insetos} voadores, apanhados em pleno voo.
Em novembro simplesmente não há nenhum: frio demais. Nenhuma troca
de cardápio resolve isso para uma ave feita para caçar no ar --- o
estoque de alimento da andorinha some por completo.
\end{example}

\section{Três respostas ao inverno}

\begin{definition}[Migração]\label{def:g3:animals-through-seasons:migration}
\emph{Migrar}\index{migrar} é viajar, todo ano, de uma região para
outra conforme as estações mudam --- partindo antes do inverno para
terras mais quentes onde ainda há alimento, e voltando na primavera.
A andorinha, a cegonha e o grou são \emph{migradores}\index{migração};
alguns viajam milhares de quilômetros e acham o caminho de volta ao
mesmíssimo ninho.
\end{definition}

\begin{definition}[Hibernação]\label{def:g3:animals-through-seasons:hibernation}
\emph{Hibernar}\index{hibernar} é passar o inverno num \omterm{prop:g1:healthy-habits:needs}{sono} profundo
e frio, escondido num abrigo: o corpo esfria, o coração e a
respiração ficam quase parados, e o \omterm{def:g1:animals-around-us:animal}{animal} vive devagarinho da
gordura acumulada no outono. O ouriço, o arganaz, a marmota e o
morcego hibernam.
\end{definition}

\begin{definition}[Continuar ativo]\label{def:g3:animals-through-seasons:active}
Os demais \emph{continuam ativos}\index{continuar ativo (no
inverno)} e mudam o equipamento ou os hábitos: um \emph{pelo de
inverno}\index{pelo de inverno} mais grosso,
\emph{reservas}\index{reserva de alimento} juntadas no outono (as
nozes enterradas pelo esquilo), uma troca de cardápio (o sabiá do
\cref{ex:g2:what-animals-eat:winter}) ou simplesmente uma procura
mais dura --- a raposa, o veado, o pintarroxo.
\end{definition}

\begin{example}[Três vizinhos, três estratégias]\label{ex:g3:animals-through-seasons:three}
Uma cerca viva em janeiro: as andorinhas que fizeram ninho em cima
dela estão na África, caçando \omterm{prop:g3:sorting-living-things:groups}{insetos} sob um sol quente; o ouriço lá
embaixo dorme frio e devagar no ninho de \omterm{def:g1:plants-around-us:parts}{folhas}, queimando a gordura
de agosto; o pintarroxo no alto, com a plumagem de inverno toda
inflada, patrulha à procura das últimas frutinhas. Em abril, os três
vão estar de novo ocupados na mesma cerca viva.
\end{example}

\begin{remark}[Dormir não é preguiça, viajar não é turismo]\label{rem:g3:animals-through-seasons:cost}
Cada estratégia tem o seu preço. O migrador arrisca uma viagem
exaustiva; o hibernador precisa juntar cada grama da gordura de
inverno no outono --- um ouriço magro demais em novembro não vai
acordar na primavera; o \omterm{def:g1:animals-around-us:animal}{animal} ativo aposta em achar alimento
durante os meses de mais fome. O inverno é pago adiantado, ou no
meio do caminho, ou dia após dia.
\end{remark}

\begin{method}[Adivinhar a estratégia de inverno de um animal]\label{met:g3:animals-through-seasons:guess}
\begin{enumerate}
  \item Pergunte o que ele come e se esse alimento existe aqui no
        inverno;
  \item alimento totalmente sumido (\omterm{prop:g3:sorting-living-things:groups}{insetos} voadores, néctar):
        espere \omterm{def:g3:animals-through-seasons:migration}{migração} ou hibernação;
  \item alimento que dá para achar (\omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}, frutinhas, presas):
        espere um inverno ativo, talvez com um pelo mais grosso ou
        com reservas;
  \item confira no outono: engordar e cavar um ninho aponta para a
        hibernação; juntar-se em bandos aponta para a partida.
\end{enumerate}
\end{method}

\begin{example}[A primavera desfaz tudo]\label{ex:g3:animals-through-seasons:spring}
Em março e abril a ordem se inverte: os \omterm{prop:g3:sorting-living-things:groups}{insetos} voltam, os
\omterm{def:g3:animals-through-seasons:migration}{migradores} reaparecem nos ninhos antigos, os hibernadores acordam
magros e com muita fome, os \omterm{def:g3:animals-through-seasons:active}{pelos de inverno} caem. As estações
giram, e o calendário dos \omterm{def:g1:animals-around-us:animal}{animais} gira com elas --- no outono
seguinte, as andorinhas vão se enfileirar nos fios de novo.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:1}
Que problema duplo o inverno impõe a todo \omterm{def:g1:animals-around-us:animal}{animal}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:2}
Diga as três estratégias de inverno, com dois \omterm{def:g1:animals-around-us:animal}{animais} para cada uma.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:3}
O que acontece com o corpo de um ouriço que hiberna durante o longo
\omterm{prop:g1:healthy-habits:needs}{sono} dele? Do que ele vive?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:4}
Por que exatamente a andorinha precisa ir embora --- qual é o
alimento dela e o que acontece com esse alimento aqui no inverno?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:5}
Dê três jeitos que um \omterm{def:g1:animals-around-us:animal}{animal} ativo tem de atravessar o inverno, cada
um com o seu \omterm{def:g1:animals-around-us:animal}{animal}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:6}
Qual é o preço de cada estratégia, segundo a
\cref{rem:g3:animals-through-seasons:cost}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:7}
No outono, um \omterm{def:g1:animals-around-us:animal}{animal} é visto comendo muitíssimo e forrando um buraco
com \omterm{def:g1:plants-around-us:parts}{folhas} secas. Que estratégia ele está preparando? Que passo do
\cref{met:g3:animals-through-seasons:guess} disse isso a você?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:animals-through-seasons:8}
Faça a experiência: mantenha uma lista de inverno com as \omterm{prop:g3:sorting-living-things:groups}{aves} que
você realmente vir lá fora durante uma semana fria. As andorinhas
estão na lista? Quais das suas \omterm{prop:g3:sorting-living-things:groups}{aves} escolheram a estratégia ativa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:animals-through-seasons:9}
Use o \cref{met:g3:animals-through-seasons:guess} em: (a) o esquilo
(come nozes e \omterm{prop:g3:flowers-fruits-seeds:transform}{sementes}); (b) o grou (come sapos e \omterm{prop:g3:sorting-living-things:groups}{insetos} de campos
alagados). O que cada um provavelmente faz no inverno?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:animals-through-seasons:10}
Um outono ameno deixa um ouriço jovem acordado e caçando até tarde,
mas ele continua magro. Por que isso é perigoso para ele, segundo a
\cref{rem:g3:animals-through-seasons:cost}? O que quem resgata
\omterm{def:g1:animals-around-us:animal}{animais} faz com ouriços magros demais no outono?
\end{exercise}
