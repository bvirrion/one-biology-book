\section*{Chapter \ref{ch:g10:biodiversity-classification} --- Biodiversity and Classification}

\begin{solution}{exo:g10:biodiversity-classification:1}
\omterm{def:g10:biodiversity-classification:biodiversity}{Genetic diversity}: many alleles in a rice landrace collection. \omterm{def:g10:biodiversity-classification:biodiversity}{Species
diversity}: number (and evenness) of bird species in a wetland. \omterm{def:g10:biodiversity-classification:biodiversity}{Ecosystem
diversity}: coral reef versus mangrove versus seagrass beds in one coastal
region. \omterm{def:g10:biodiversity-classification:biodiversity}{Biodiversity} is variety at all three levels, not species counts alone.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:2}
Under the \omterm{def:g10:biodiversity-classification:species}{biological species concept}, a species is a group of natural
populations that interbreed to produce viable fertile offspring and are
reproductively isolated from other such groups. It is hard to apply to
asexual \omterm{def:g10:scientific-biology:organism}{organisms} (including most bacteria), fossils, and many plants that
hybridise freely.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:3}
Correct forms: \emph{Homo sapiens} (genus \emph{Homo}); \emph{Felis catus}
(genus \emph{Felis}); \emph{Escherichia coli} (genus \emph{Escherichia}).
Genus is capitalised; specific epithet lower case; both italicised.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:4}
From broad to narrow: \omterm{def:g10:biodiversity-classification:hierarchy}{domain}, \omterm{def:g10:biodiversity-classification:hierarchy}{kingdom}, \omterm{def:g10:biodiversity-classification:hierarchy}{phylum}, \omterm{def:g10:biodiversity-classification:hierarchy}{class}, \omterm{def:g10:biodiversity-classification:hierarchy}{order}, \omterm{def:g10:biodiversity-classification:hierarchy}{family}, genus,
species. Each rank is a \omterm{def:g10:biodiversity-classification:hierarchy}{taxon} nested inside the rank above.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:5}
A node represents a common ancestor (a branching point where lineages
diverged). Tips represent the taxa being compared --- species or larger groups
at the ends of branches.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:6}
A and B are more closely related because they share a more recent common
ancestor with each other than either does with C. A consistent \omterm{def:g10:biodiversity-classification:clade}{cladogram} has
A and B as sister tips with C branching earlier from their shared stem.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:7}
Molecular and biochemical evidence show a deep evolutionary split: \omterm{def:g10:biodiversity-classification:domains}{Archaea} are
more closely related to \omterm{def:g10:the-cell:prok-euk}{eukaryotes} than to \omterm{def:g10:biodiversity-classification:domains}{Bacteria}. ``\omterm{def:g10:the-cell:prok-euk}{Prokaryote}'' describes
\omterm{def:g10:the-cell:cell}{cell} organisation (no nucleus), not a single \omterm{def:g10:biodiversity-classification:clade}{clade}, so the two prokaryotic
\omterm{def:g10:biodiversity-classification:domains}{domains} are separate.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:8}
A streamlined shape is convergent adaptation for swimming, not proof of close
ancestry. Whales share derived mammalian traits (mammary glands, lungs,
middle-ear bones, hair in \omterm{def:g10:scientific-biology:properties}{development}, etc.) with other mammals. Phylogeny
groups by ancestry and \omterm{def:g10:biodiversity-classification:characters}{shared derived characters}, not by habitat look-alikes.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:9}
Morphological (and sometimes molecular if \omterm{def:g10:molecules-of-life:nucleic}{DNA} remains) species concepts are
usable on fossils. The \omterm{def:g10:biodiversity-classification:species}{biological species concept} is not directly testable
because breeding behaviour and fertility cannot be observed in extinct
populations known only from bones.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:10}
Tree \omterm{def:g10:biodiversity-classification:hierarchy}{order}: W branches first (backbone only); then X joins the four-limbed
lineage; Y and Z are sisters with fur; Z alone gains opposable thumbs. Mark
character gains on the internal branches where each trait first appears in the
\omterm{def:g10:scientific-biology:hypothesis}{hypothesis}.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:11}
Forest loss eliminates populations and can extinguish species (\omterm{def:g10:biodiversity-classification:biodiversity}{species
diversity} down). Survivors in fragments may be small and inbred (\omterm{def:g10:biodiversity-classification:biodiversity}{genetic
diversity} down). Entire habitat types vanish from the landscape (\omterm{def:g10:biodiversity-classification:biodiversity}{ecosystem
diversity} down). One pressure therefore hits all three levels.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:12}
Correct forms: \emph{Canis lupus}, \emph{Homo sapiens}, \emph{Quillaja
saponaria}. Genera: \emph{Canis}, \emph{Homo}, \emph{Quillaja}. A shared genus
means taxonomists place the species as close relatives in the same small
group --- more recent common ancestry than species in different genera of the
same \omterm{def:g10:biodiversity-classification:hierarchy}{family}, under evolutionary classification.
\end{solution}

\begin{solution}{exo:g10:biodiversity-classification:13}
Ranks are nested: membership in a \omterm{def:g10:biodiversity-classification:hierarchy}{family} implies membership in all higher
ranks that contain that \omterm{def:g10:biodiversity-classification:hierarchy}{family} (\omterm{def:g10:biodiversity-classification:hierarchy}{order}, \omterm{def:g10:biodiversity-classification:hierarchy}{class}, \omterm{def:g10:biodiversity-classification:hierarchy}{phylum}, \omterm{def:g10:biodiversity-classification:hierarchy}{kingdom}, \omterm{def:g10:biodiversity-classification:hierarchy}{domain}). Genus
is below \omterm{def:g10:biodiversity-classification:hierarchy}{family}, so two species can share a \omterm{def:g10:biodiversity-classification:hierarchy}{family} yet sit in different genera.
Humans (\emph{Homo sapiens}) and chimpanzees (\emph{Pan troglodytes}) share
\omterm{def:g10:biodiversity-classification:hierarchy}{family} Hominidae (and all ranks above) but not the genus \emph{Homo}.
\end{solution}

\begin{solution}{pb:g10:biodiversity-classification:1}
\textbf{Part I.} Biological concept: no fertile interbreeding with echidnas
or other species in nature; distinct mating signals or isolation. If breeding
data are unavailable: diagnosable morphology, distinct \omterm{def:g10:molecules-of-life:nucleic}{DNA} clusters, and
ecological niche as operational criteria.

\textbf{Part II.} \omterm{def:g10:biodiversity-classification:hierarchy}{Domain} \omterm{def:g10:biodiversity-classification:domains}{Eukarya}; \omterm{def:g10:biodiversity-classification:hierarchy}{kingdom} Animalia; \omterm{def:g10:biodiversity-classification:hierarchy}{phylum} Chordata; \omterm{def:g10:biodiversity-classification:hierarchy}{class}
Mammalia (fur; egg-laying places it among monotreme-like mammals). Binomial
e.g.\ \emph{Tachyglossus insula} (invented; correct form matters more than the
words).

\textbf{Part III.} Fish outside the tetrapod \omterm{def:g10:biodiversity-classification:clade}{clade}; lizard with tetrapods but
outside mammals; echidna + new \omterm{def:g10:scientific-biology:organism}{organism} as sisters; human nested in mammals.
Mammal \omterm{def:g10:biodiversity-classification:clade}{clade} mark: fur/hair or mammary glands as \omterm{def:g10:biodiversity-classification:characters}{shared derived characters}
relative to lizards and fish.
\end{solution}
