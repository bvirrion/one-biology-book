\chapter{Clasificar los animales}\label{ch:g4:classifying-animals}

Un museo de historia natural guarda millones de \omterm{def:g1:animals-around-us:animal}{animales} --- cajones
de escarabajos, salas de \omterm{def:g3:bones-and-muscles:skeleton}{esqueletos}, tarros de \omterm{prop:g3:sorting-living-things:groups}{peces}. Y sin embargo un
conservador llega a cualquiera de ellos en unos minutos. El secreto no
es la memoria, sino el \emph{orden}: cada \omterm{def:g1:animals-around-us:animal}{animal} tiene su sitio en un
gran sistema de grupos dentro de grupos, construido con las reglas de
\omterm{def:g4:classifying-animals:classification}{clasificación} del \cref{ch:g3:sorting-living-things} --- y este curso
aprendemos cómo funciona el sistema.

\section{Grupos dentro de grupos}

\begin{definition}[Clasificación]\label{def:g4:classifying-animals:classification}
\emph{Clasificar}\index{clasificar} \omterm{def:g1:living-or-not:living}{seres vivos} es agruparlos por los
rasgos del cuerpo que comparten y agrupar después esos grupos en
grupos mayores --- \emph{grupos encajados dentro de
grupos}\index{grupos encajados}, como cajas dentro de cajas. El
conjunto es una \emph{clasificación}\index{clasificación}.
\end{definition}

\begin{example}[Las direcciones del gato]\label{ex:g4:classifying-animals:cat}
Un gato, varias direcciones encajadas: el gato es un \emph{mamífero}
(\omterm{def:g1:animals-around-us:covering}{pelo}, \omterm{def:g2:animals-grow-up:born}{leche} para las crías); todo mamífero es un \emph{\omterm{def:g4:classifying-animals:vertebrate}{vertebrado}}
(un \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto} con \omterm{def:g4:classifying-animals:vertebrate}{columna vertebral} por dentro); todo \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} es un
\emph{\omterm{def:g1:animals-around-us:animal}{animal}}. El gato dentro de los \omterm{prop:g3:sorting-living-things:groups}{mamíferos}, los \omterm{prop:g3:sorting-living-things:groups}{mamíferos} dentro
de los \omterm{def:g4:classifying-animals:vertebrate}{vertebrados}, los \omterm{def:g4:classifying-animals:vertebrate}{vertebrados} dentro de los \omterm{def:g1:animals-around-us:animal}{animales}: cada caja
está dentro de otra mayor.
\end{example}

\begin{definition}[Vertebrados e invertebrados]\label{def:g4:classifying-animals:vertebrate}
Un \emph{vertebrado}\index{vertebrado} es un \omterm{def:g1:animals-around-us:animal}{animal} con un \omterm{def:g3:bones-and-muscles:skeleton}{esqueleto}
interno construido alrededor de una \emph{columna
vertebral}\index{columna vertebral} --- como la que notaste en el
\cref{ex:g3:bones-and-muscles:feel}. Un \omterm{def:g1:animals-around-us:animal}{animal} sin columna vertebral
es un \emph{invertebrado}\index{invertebrado}: \omterm{prop:g3:sorting-living-things:groups}{insectos}, arañas,
caracoles, gusanos, medusas --- la inmensa mayoría de los \omterm{def:g1:animals-around-us:animal}{animales}.
\end{definition}

\section{Las cinco clases de vertebrados}

\begin{proposition}[Las cinco clases de vertebrados]\label{prop:g4:classifying-animals:classes}
Los \omterm{def:g4:classifying-animals:vertebrate}{vertebrados} se clasifican en cinco grandes grupos, llamados
\emph{clases}\index{clase (de vertebrados)}:
\begin{enumerate}
  \item los \emph{\omterm{prop:g3:sorting-living-things:groups}{peces}}: escamas, aletas, respiran en el agua ---
        trucha, tiburón, sardina;
  \item los \emph{anfibios}\index{anfibio}: \omterm{def:g1:the-five-senses:sense}{piel} desnuda y húmeda,
        \omterm{def:g2:animals-grow-up:born}{huevos} en el agua, infancia de renacuajo --- rana, sapo,
        tritón;
  \item los \emph{reptiles}\index{reptil}: \omterm{def:g1:the-five-senses:sense}{piel} seca y escamosa, casi
        todos ponen \omterm{def:g2:animals-grow-up:born}{huevos} con cáscara en tierra --- lagartija,
        serpiente, tortuga, cocodrilo;
  \item las \emph{\omterm{prop:g3:sorting-living-things:groups}{aves}}: \omterm{def:g1:animals-around-us:covering}{plumas}, pico, \omterm{def:g2:animals-grow-up:born}{huevos} con cáscara --- mirlo,
        avestruz, pingüino;
  \item los \emph{\omterm{prop:g3:sorting-living-things:groups}{mamíferos}}: \omterm{def:g1:animals-around-us:covering}{pelo}, crías alimentadas con \omterm{def:g2:animals-grow-up:born}{leche} ---
        gato, ballena, murciélago, persona.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Clasificar un envío al zoo]\label{ex:g4:classifying-animals:zoo}
Llegan seis cajas: una pitón (\omterm{def:g1:the-five-senses:sense}{piel} seca y escamosa, \omterm{def:g2:animals-grow-up:born}{huevos} con cáscara
--- reptil), un sapo (\omterm{def:g1:the-five-senses:sense}{piel} desnuda y húmeda --- anfibio), un pingüino
(\omterm{def:g1:animals-around-us:covering}{plumas} --- ave), un tiburón (aletas y branquias --- \omterm{prop:g3:sorting-living-things:groups}{pez}), un delfín
(\omterm{def:g2:animals-grow-up:born}{leche} para su cría --- mamífero) y una caja de caracoles gigantes
(sin \omterm{def:g4:classifying-animals:vertebrate}{columna vertebral} --- \omterm{def:g4:classifying-animals:vertebrate}{invertebrados}, en ninguna clase de
\omterm{def:g4:classifying-animals:vertebrate}{vertebrados}).
\end{example}

\begin{remark}[¿Reptil o anfibio?]\label{rem:g4:classifying-animals:confusion}
Las lagartijas y los tritones se parecen a primera vista --- cuatro
patas, cola larga. Lo decide la \omterm{def:g1:the-five-senses:sense}{piel}: la de la lagartija es seca y
escamosa, la del tritón húmeda y desnuda; los \omterm{def:g2:animals-grow-up:born}{huevos} de la lagartija
tienen cáscara y se ponen en tierra, los del tritón se ponen desnudos
en el agua. La misma silueta, clases distintas.
\end{remark}

\section{Usar una clasificación}

\begin{method}[Colocar un vertebrado en su clase]\label{met:g4:classifying-animals:place}
Hazte las preguntas en orden y párate en el primer sí:
\begin{enumerate}
  \item ¿\omterm{def:g1:animals-around-us:covering}{plumas}? --- un ave.
  \item ¿\omterm{def:g1:animals-around-us:covering}{pelo} (o \omterm{def:g2:animals-grow-up:born}{leche} para las crías)? --- un mamífero.
  \item ¿\omterm{def:g1:the-five-senses:sense}{piel} seca y escamosa? --- un reptil.
  \item ¿\omterm{def:g1:the-five-senses:sense}{piel} desnuda y húmeda e infancia de renacuajo? --- un
        anfibio.
  \item ¿aletas y branquias, y vida en el agua? --- un \omterm{prop:g3:sorting-living-things:groups}{pez}.
\end{enumerate}
¿No tiene \omterm{def:g4:classifying-animals:vertebrate}{columna vertebral}? Entonces era un \omterm{def:g4:classifying-animals:vertebrate}{invertebrado}, y las
preguntas no valen.
\end{method}

\begin{proposition}[Lo que una clase te da gratis]\label{prop:g4:classifying-animals:free}
Saber la clase de un \omterm{def:g1:animals-around-us:animal}{animal} es saber un montón de datos antes de
mirarlo: si solo te dicen ``es un mamífero'', ya esperas \omterm{def:g1:animals-around-us:covering}{pelo}, \omterm{def:g2:animals-grow-up:born}{leche}
para las crías y --- como en casi todos los \omterm{prop:g3:sorting-living-things:groups}{mamíferos} --- crías
nacidas vivas. Ese ahorro es el sentido entero de la \omterm{def:g4:classifying-animals:classification}{clasificación}:
aprende el grupo una vez y todos sus miembros llegarán medio conocidos.
\end{proposition}
\admitted

\begin{example}[Llega una especie nueva]\label{ex:g4:classifying-animals:new}
Los naturalistas siguen descubriendo \omterm{def:g1:animals-around-us:animal}{animales} nunca vistos. Cuando se
encuentra un \omterm{def:g1:animals-around-us:animal}{animal} parecido a un murciélago con \omterm{def:g1:animals-around-us:covering}{pelo} y \omterm{def:g2:animals-grow-up:born}{leche}, va
derecho a los \omterm{prop:g3:sorting-living-things:groups}{mamíferos} --- y en el acto los biólogos esperan una
\omterm{def:g4:classifying-animals:vertebrate}{columna vertebral}, un corazón de cuatro cavidades y sangre caliente,
antes de comprobar nada de eso. La \omterm{def:g4:classifying-animals:classification}{clasificación} paga en predicciones.
\end{example}

\begin{remark}[Hacia el árbol de familia]\label{rem:g4:classifying-animals:tree}
¿Por qué viajan siempre juntos el \omterm{def:g1:animals-around-us:covering}{pelo}, la \omterm{def:g2:animals-grow-up:born}{leche} y la \omterm{def:g4:classifying-animals:vertebrate}{columna
vertebral}? La respuesta completa --- que los miembros de un grupo
heredan de antepasados comunes los rasgos que comparten --- pertenece
a la historia del parentesco y de la evolución, que se retoma en
cursos posteriores. Por ahora, las cajas encajadas sencillamente
funcionan, y funcionan de maravilla.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:1}
¿Qué es un \omterm{def:g4:classifying-animals:vertebrate}{vertebrado}? ¿Qué es un \omterm{def:g4:classifying-animals:vertebrate}{invertebrado}? Da dos ejemplos de
cada uno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:2}
Nombra las cinco clases de \omterm{def:g4:classifying-animals:vertebrate}{vertebrados} con los rasgos que las definen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:3}
Coloca cada \omterm{def:g1:animals-around-us:animal}{animal}: la sardina, el cocodrilo, el tritón, el avestruz,
la ballena, la lombriz.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:4}
Escribe las direcciones encajadas del gato, de la caja más pequeña a
la mayor, como en el \cref{ex:g4:classifying-animals:cat}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:5}
Una lagartija y un tritón se parecen. ¿Qué dos rasgos separan sus
clases?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:6}
¿Qué grupo reúne a la inmensa mayoría de los \omterm{def:g1:animals-around-us:animal}{animales}? Nombra cuatro
de sus miembros.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:7}
Pásale en voz alta el \cref{met:g4:classifying-animals:place} al
pingüino y al delfín, pregunta a pregunta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:classifying-animals:8}
Solo te dicen: ``este \omterm{def:g1:animals-around-us:animal}{animal} es un ave''. Enumera tres cosas que ya
esperas sin mirar, según la \cref{prop:g4:classifying-animals:free}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:9}
La ballena casi no tiene \omterm{def:g1:animals-around-us:covering}{pelo} y vive como un \omterm{prop:g3:sorting-living-things:groups}{pez}. Con el
\cref{met:g4:classifying-animals:place} y el
\cref{ex:g3:sorting-living-things:surprises}, explica por qué es de
todas formas un mamífero --- y por qué ``vive en el mar'' no fue nunca
un rasgo de \omterm{def:g4:classifying-animals:classification}{clasificación}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:classifying-animals:10}
La serpiente no tiene patas, y la lombriz tampoco. ¿Por qué una es un
reptil \omterm{def:g4:classifying-animals:vertebrate}{vertebrado} y la otra un \omterm{def:g4:classifying-animals:vertebrate}{invertebrado}? ¿Qué rasgo único manda
más que las patas que faltan?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:classifying-animals:11}
El ornitorrinco desconcertó a los naturalistas: \omterm{def:g1:animals-around-us:covering}{pelo} y \omterm{def:g2:animals-grow-up:born}{leche} y, aun
así, pone \omterm{def:g2:animals-grow-up:born}{huevos} con cáscara. ¿Qué clase ganó y por qué? ¿Qué enseña
el rompecabezas sobre \omterm{def:g4:classifying-animals:classification}{clasificar} por \emph{conjuntos} de rasgos en vez
de por uno solo?
\end{exercise}
