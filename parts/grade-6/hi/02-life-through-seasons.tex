\chapter{ऋतुओं में जीवन}\label{ch:g6:life-through-seasons}

हर महीने की पहली तारीख़ को उसी चेस्टनट के \omterm{def:g1:plants-around-us:tree}{पेड़} की तस्वीर लो और बारहों
तस्वीरें एक क़तार में लगा दो: नंगी शाखाएँ, \omterm{def:g6:life-through-seasons:buds}{कलियाँ}, \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}, फूलों की
मोमबत्तियाँ, \omterm{prop:g3:flowers-fruits-seeds:transform}{फल}, सुनहरा रंग, और फिर नंगी शाखाएँ। कुछ भी हिला नहीं ---
\omterm{def:g1:plants-around-us:tree}{पेड़} पूरे साल अपनी जगह खड़ा रहा --- फिर भी सब कुछ बदल गया।
\Cref{ch:g3:animals-through-seasons} ने साल भर \omterm{def:g1:animals-around-us:animal}{जंतुओं} का पीछा किया था;
अब हम सर्वेक्षक की \omterm{def:g1:the-five-senses:sense}{आँखों} से पूरे \omterm{def:g6:exploring-our-environment:components}{पर्यावरण} का पीछा करेंगे, और पहले
\omterm{def:g1:plants-around-us:plant}{पौधों} का।

\section{ऋतुओं के साथ पर्यावरण बदलता है}

\begin{proposition}[परिस्थितियों का ऋतु-बदलाव]\label{prop:g6:life-through-seasons:conditions}
साल भर में किसी \omterm{def:g6:exploring-our-environment:components}{पर्यावरण} की \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ}
(\cref{def:g6:exploring-our-environment:conditions}) ख़ूब झूलती हैं:
दिन छोटे और बड़े होते हैं, \omterm{def:g6:exploring-our-environment:conditions}{ताप} गिरता और चढ़ता है, और मिट्टी की नमी
बारिश तथा सूखे के पीछे चलती है। जनवरी और जून में सर्वेक्षण किया गया कोई
स्थल तालिका में दो अलग पंक्तियाँ देता है --- इसलिए
\cref{prop:g6:exploring-our-environment:decide} के अनुसार वहाँ बाशिंदों
के दो अलग समूहों की उम्मीद करनी चाहिए।
\end{proposition}
\admitted

\begin{example}[एक बाड़, दो सर्वेक्षण]\label{ex:g6:life-through-seasons:hedge}
गली की बाड़, दो बार सर्वेक्षण की गई। जून: घनी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}, \omterm{def:g1:plants-around-us:parts}{जड़} पर फूलती
\omterm{def:g1:plants-around-us:plant}{पौधों} की दस \omterm{def:g5:biodiversity:species}{प्रजातियाँ}, ज़ोर से भिनभिनाती मधुमक्खियाँ, घोंसले बनाती
कलचिड़ियाँ, बारिश के बाद निकले घोंघे। जनवरी: नंगी टहनियाँ, हरे रंग में
सिर्फ़ आइवी और \omterm{prop:g4:plants-without-seeds:damp}{मॉस}, कोई उड़ता \omterm{prop:g3:sorting-living-things:groups}{कीट} नहीं, पत्तों के कूड़े में चारा ढूँढ़ती
कलचिड़ियाँ, और किनारे में गहरे धँसे अपने \omterm{def:g1:animals-around-us:covering}{खोलों} में बंद घोंघे। वही जगह,
वही घटकों की सूची --- \omterm{def:g6:exploring-our-environment:components}{सजीव}, खनिज, निशान --- पर \omterm{def:g6:exploring-our-environment:components}{सजीव} वाला ख़ाना पूरी तरह
बदल चुका है।
\end{example}

\section{पौधे बुरा मौसम कैसे काटते हैं}

\begin{definition}[पर्णपाती और सदाबहार]\label{def:g6:life-through-seasons:trees}
\omterm{def:g1:plants-around-us:tree}{पेड़} अपने सर्दी वाले उत्तर से बँट जाते हैं:
\begin{itemize}
  \item \emph{पर्णपाती}\index{पर्णपाती} \omterm{def:g1:plants-around-us:tree}{पेड़} --- बलूत, चेस्टनट, बीच
        --- पतझड़ में अपनी सारी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} गिरा देते हैं और नंगे खड़े
        रहते हैं;
  \item \emph{सदाबहार}\index{सदाबहार} \omterm{def:g1:plants-around-us:tree}{पेड़} --- चीड़, होली, दीवार पर
        आइवी --- पूरी सर्दी काम करती \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} बनाए रखते हैं, जो आम तौर
        पर सख़्त, मोम जैसी या सुई के आकार की होती हैं
        (\cref{prop:g3:sorting-living-things:plants} में सुइयों से
        मुलाक़ात हो ही चुकी है)।
\end{itemize}
\end{definition}

\begin{definition}[कलियाँ]\label{def:g6:life-through-seasons:buds}
सर्दी में \omterm{def:g6:life-through-seasons:trees}{पर्णपाती} \omterm{def:g1:plants-around-us:tree}{पेड़} मरा हुआ नहीं होता: अगला वसंत उसकी टहनियों पर
पहले से \emph{कलियों}\index{कली} के रूप में बँधा रखा है --- यानी नन्हे
पूरे \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर}, जिनकी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} छोटी-सी तह की हुई हैं और जो मौसम-रोधी \omterm{def:g1:animals-around-us:covering}{शल्कों}
में लिपटी हैं। वसंत की ``नई'' \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} पिछली गर्मियों में ही बना दी गई
थीं; कली उनके लिए सर्दी का भंडार-घर है।
\end{definition}

\begin{example}[चेस्टनट की कली खोलना]\label{ex:g6:life-through-seasons:bud}
हॉर्स चेस्टनट की बड़ी चिपचिपी \omterm{def:g6:life-through-seasons:buds}{कलियाँ} किसी धार से अच्छी तरह खुलती हैं
(यह काम किसी बड़े का): रोग़नी \omterm{def:g1:animals-around-us:covering}{शल्कों} के नीचे, रुई जैसी लोई में लिपटी,
तह की हुई नन्ही \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} पड़ी होती हैं --- पूरी, पीली-सी, इंतज़ार करती
हुईं। फ़रवरी में काटी गई और घर के भीतर पानी में खड़ी की गई टहनी हफ़्तों
पहले ही ``पत्ते निकाल'' देती है: \omterm{prop:g2:from-seed-to-plant:cycle}{अंकुर} तो तैयार थे; बस गरमाहट गुम थी।
\end{example}

भीतर अगले वसंत की \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} तह की हुई और तैयार पड़ी हैं।}

\begin{proposition}[शाक-परत के सर्दी वाले उत्तर]\label{prop:g6:life-through-seasons:herbs}
छोटे \omterm{def:g1:plants-around-us:plant}{पौधों} की अपनी रणनीतियाँ हैं:
\begin{itemize}
  \item \emph{एकवर्षी}\index{एकवर्षी पौधा} --- पोस्ता, सेम --- पतझड़
        में पूरी तरह मर जाते हैं और सर्दी \emph{\omterm{def:g2:from-seed-to-plant:seed}{बीजों}} के रूप में
        काटते हैं (\cref{prop:g2:from-seed-to-plant:needs}: \omterm{def:g2:from-seed-to-plant:seed}{बीज} ठंड
        सुरक्षित काट लेता है);
  \item \emph{बहुवर्षी}\index{बहुवर्षी पौधा} --- डैफ़ोडिल, कुकरौंधा
        --- अपने ऊपरी हिस्से सूखने देते हैं और ज़मीन के नीचे \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंदों},
        \omterm{ex:g4:plants-without-seeds:bulbtuber}{कंदों} या सख़्त जड़-ठूँठों के रूप में इंतज़ार करते हैं
        (\cref{ex:g4:plants-without-seeds:bulbtuber} के भंडार-घर,
        सर्दी की ड्यूटी पर);
  \item ज़मीन के \omterm{def:g6:life-through-seasons:trees}{सदाबहार} --- \omterm{prop:g4:plants-without-seeds:damp}{मॉस}, आइवी --- बस झेल जाते हैं।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[डैफ़ोडिल कहाँ चले गए?]\label{ex:g6:life-through-seasons:daffodil}
डैफ़ोडिल की क्यारी मार्च में फूलती है, मई में पीली पड़ती है, और जुलाई तक
बिना निशान ग़ायब हो जाती है --- फिर भी अगले मार्च वह और घनी होकर लौट
आती है। पूरी गर्मी, पतझड़ और सर्दी वह लॉन के नीचे \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंदों} के रूप में
मौजूद थी: \omterm{def:g1:plants-around-us:plant}{पौधा} कभी गया ही नहीं; वह भंडार में चला गया था।
\end{example}

\section{जंतुओं का पंचांग, फिर से}

\begin{proposition}[जंतुओं के उत्तर, सर्वेक्षण की भाषा में]\label{prop:g6:life-through-seasons:animals}
\Cref{ch:g3:animals-through-seasons} की तीनों रणनीतियाँ --- \omterm{def:g3:animals-through-seasons:migration}{प्रवास},
\omterm{def:g3:animals-through-seasons:hibernation}{शीतनिद्रा}, सक्रिय रहना --- अब सर्वेक्षण में दर्ज बदलावों के उत्तर की
तरह पढ़ी जाती हैं: हर \omterm{def:g5:biodiversity:species}{प्रजाति} अपनी ज़रूरी \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} और अपने भोजन के
पीछे चलती है।
\begin{itemize}
  \item अबाबील अपने भोजन के पीछे वहाँ जाती है जहाँ की \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ}
        \omterm{prop:g3:sorting-living-things:groups}{कीटों} को उड़ता रखती हैं;
  \item साही और \omterm{def:g1:animals-around-us:covering}{खोल} में बंद घोंघा उन महीनों का इंतज़ार करते हैं जिनमें
        उनके शरीर काम नहीं कर सकते;
  \item और \omterm{prop:g3:sorting-living-things:groups}{कीट} एक चौथा उत्तर जोड़ते हैं: ज़्यादातर सर्दी
        \emph{\omterm{def:g2:animals-grow-up:born}{अंडों}, डिंभों या \omterm{ex:g2:animals-grow-up:butterfly}{प्यूपाओं}} के रूप में
        (\cref{ex:g2:animals-grow-up:butterfly}) मिट्टी, छाल या
        पत्तों के कूड़े में दबे बिताते हैं --- वयस्क मर जाता है, और
        \omterm{def:g5:biodiversity:species}{प्रजाति} अपनी नन्ही अवस्थाओं के रूप में सर्दी काटती है।
\end{itemize}
\end{proposition}
\admitted

\begin{example}[जून के कीट जनवरी में कहाँ हैं]\label{ex:g6:life-through-seasons:insects}
जनवरी की बाड़ में एक भी \omterm{prop:g3:sorting-living-things:groups}{कीट} नहीं दिखता --- फिर भी जून के सारे \omterm{prop:g3:sorting-living-things:groups}{कीट} वहीं
मौजूद हैं: तितलियाँ मुँडेर के पत्थरों के नीचे \omterm{ex:g2:animals-grow-up:butterfly}{प्यूपा} के रूप में, टिड्डे
मिट्टी में \omterm{def:g2:animals-grow-up:born}{अंडों} के रूप में, गुबरैले पत्तों के कूड़े में गहरे धँसे सोते
हुए वयस्कों के रूप में, और मधुमक्खियाँ छत्ते में कसकर बने गरम गुच्छे के
रूप में। जून का सर्वेक्षण टहलकर हो जाता है; जनवरी के सर्वेक्षण को खोदना,
उठाना और झाँकना पड़ता है --- बाशिंदों ने अपना \emph{रूप} बदला है, पता
नहीं।
\end{example}

\begin{method}[साल भर का सर्वेक्षण]\label{met:g6:life-through-seasons:yearsurvey}
किसी स्थल के ऋतु-जीवन का अध्ययन करने के लिए:
\begin{enumerate}
  \item उसी चिह्नित स्थल का
        (\cref{met:g6:exploring-our-environment:survey})
        हर ऋतु में एक बार सर्वेक्षण करो --- वही वर्ग, वही उपकरण,
        वही खोज-समय;
  \item हर बार \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियाँ} \emph{और} बाशिंदे, दोनों दर्ज करो; और
        स्थल की तस्वीर हर बार उसी जगह से खींचो;
  \item सर्दी के छिपे रूप जान-बूझकर खोजो: \omterm{def:g6:life-through-seasons:buds}{कलियाँ}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद}, \omterm{def:g2:animals-grow-up:born}{अंडे},
        \omterm{ex:g2:animals-grow-up:butterfly}{प्यूपा}, बंद घोंघे --- और हर ढक्कन वापस रख दो;
  \item चारों सर्वेक्षण अगल-बगल रखो: बाशिंदों के हर बदलाव को
        \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} के किसी बदलाव से जोड़ो, और हर ग़ैरहाज़िर का सर्दी
        वाला रूप या ठिकाना बताओ।
\end{enumerate}
\end{method}

\begin{remark}[कुछ भी यूँ ही ग़ायब नहीं होता]\label{rem:g6:life-through-seasons:vanish}
साल भर की तालिका एक गहरा नियम सिखाती है: जून के सर्वेक्षण की कोई भी
\omterm{def:g5:biodiversity:species}{प्रजाति} यूँ ही ख़त्म नहीं हो गई। हर एक या तो कहीं और है (\omterm{def:g3:animals-through-seasons:migration}{प्रवास} पर),
या सो रही है (\omterm{def:g3:animals-through-seasons:hibernation}{शीतनिद्रा} में), या भंडार में है (\omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद}, \omterm{def:g2:from-seed-to-plant:seed}{बीज}), या किसी
और जीवन-अवस्था में है (\omterm{def:g2:animals-grow-up:born}{अंडा}, \omterm{ex:g2:animals-grow-up:butterfly}{प्यूपा})। \omterm{def:g1:living-or-not:living}{जीवविज्ञान} में ``चला गया'' कभी
पूरा उत्तर नहीं होता --- अगला सवाल हमेशा यह है कि \emph{कहाँ गया, और
किस रूप में?}
\end{remark}

\omterm{def:g1:animals-around-us:animal}{जंतुओं} के सर्दी वाले चारों उत्तर बताओ, और हर एक के साथ एक \omterm{def:g5:biodiversity:species}{प्रजाति} भी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:life-through-seasons:7}
जून की बाड़ के हर बाशिंदे का जनवरी वाला रूप और पता बताओ: तितली, टिड्डा,
गुबरैला, घोंघा, अबाबील।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:life-through-seasons:8}
चारों ऋतु-सर्वेक्षणों में वही स्थल, वही उपकरण और वही खोज-समय क्यों
ज़रूरी हैं? \cref{ch:g6:exploring-our-environment} का कौन-सा नियम यहाँ
काम कर रहा है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:life-through-seasons:9}
जनवरी की बाड़ के सर्वेक्षण में जून के मुक़ाबले कहीं कम \omterm{def:g6:exploring-our-environment:components}{सजीव} दर्ज होते
हैं। \cref{ex:g6:life-through-seasons:insects} की मदद से दो कारण बताओ
कि यह फ़र्क़ असली नुक़सान को बढ़ा-चढ़ाकर क्यों दिखाता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:life-through-seasons:10}
एक माली अगस्त में डैफ़ोडिल की ``मरी हुई'' ज़मीन खोदता है और उसे ठोस
\omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद} मिलते हैं। इस खोज को
\cref{prop:g6:life-through-seasons:herbs} की भाषा में बदलो: \omterm{def:g1:plants-around-us:plant}{पौधा} किस
हालत में है, और मार्च में \omterm{ex:g4:plants-without-seeds:bulbtuber}{शल्ककंद} क्या करेंगे?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:life-through-seasons:11}
\cref{rem:g6:life-through-seasons:vanish} लागू करो: जनवरी में किसी
तालाब के मेंढक कहीं नहीं मिलते। पूछे जाने वाले दोनों सवाल बताओ, और
मेंढक-भेक वाला उनका उत्तर भी।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:life-through-seasons:12}
\omterm{def:g6:life-through-seasons:trees}{सदाबहार} \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} सख़्त, मोम जैसी या सुई के आकार की होती हैं; \omterm{def:g6:life-through-seasons:trees}{पर्णपाती}
\omterm{def:g1:plants-around-us:parts}{पत्तियाँ} चौड़ी और पतली। सुझाओ कि \omterm{def:g6:life-through-seasons:trees}{सदाबहार} \omterm{def:g1:plants-around-us:parts}{पत्ती} की बनावट सर्दी की किस

\omterm{def:g6:life-through-seasons:trees}{पर्णपाती} \omterm{def:g1:plants-around-us:tree}{पेड़} अपनी पतली \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} गिरा देना ही क्यों पसंद करता है।
\end{exercise}

\section{समस्या: बारह तस्वीरें}

\begin{problem}[{सप्ताहांत समस्या --- चेस्टनट के पेड़ का एक साल, तस्वीर
दर तस्वीर पढ़ा हुआ}]\label{pb:g6:life-through-seasons:1}
लीना साल भर हर महीने की पहली तारीख़ को फाटक के पास वाले चेस्टनट के \omterm{def:g1:plants-around-us:tree}{पेड़}
की तस्वीर खींचती है, और बारहों तस्वीरें एक क़तार में लगा देती है। हर
तस्वीर के नीचे वह दिन की लंबाई और उस सुबह का \omterm{def:g6:exploring-our-environment:conditions}{ताप} लिख देती है।

\medskip\noindent\textbf{भाग I --- क़तार को पढ़ना।}
\begin{enumerate}
  \item जनवरी और दिसंबर में नंगी शाखाएँ दिखती हैं। टहनियों पर ऐसा क्या
        है जो दोनों तस्वीरों में साबित करता है कि \omterm{def:g1:plants-around-us:tree}{पेड़} जीवित है और
        वसंत के लिए तैयार भी?
  \item \omterm{def:g6:life-through-seasons:buds}{कलियाँ} किन दो तस्वीरों के बीच फूटती हैं? लीना के वहाँ के नोट
        कहते हैं: दिन $11$--$13$ घंटे, सुबहें
        $8$--\qty{12}{\degreeCelsius}। \omterm{def:g1:plants-around-us:tree}{पेड़} किन बदलती
        \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} का उत्तर दे रहा है?
  \item मई की तस्वीर में सफ़ेद फूलों की सीधी खड़ी ``मोमबत्तियाँ''
        दिखती हैं। \cref{ch:g3:flowers-fruits-seeds} (या
        \cref{ch:g5:flower-to-fruit}) की मदद से बताओ कि अक्तूबर के \omterm{prop:g3:flowers-fruits-seeds:transform}{फल}
        बनने के लिए मई और सितंबर की तस्वीरों के बीच किसका आना ज़रूरी है
        --- और यह भी कि वह \omterm{prop:g3:flowers-fruits-seeds:transform}{फल} ठीक-ठीक क्या है।
  \item अक्तूबर की तस्वीर में सुनहरी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} दिखती हैं; नवंबर की में
        नंगी शाखाएँ और नीचे भूरा क़ालीन।
        \cref{def:g6:life-through-seasons:trees} से \omterm{def:g1:plants-around-us:tree}{पेड़} का प्रकार
        बताओ, और यह भी कि आगे की तस्वीरों में उसका अगला वसंत कहाँ
        रखा है।
\end{enumerate}

\medskip\noindent\textbf{भाग II --- \omterm{def:g1:plants-around-us:tree}{पेड़} के पड़ोसी।}
\begin{enumerate}[resume]
  \item \omterm{def:g1:plants-around-us:tree}{पेड़} के नीचे मार्च की तस्वीर डैफ़ोडिलों की एक क्यारी पकड़ती है,
        जो जुलाई की तस्वीर में नहीं है। जुलाई में वह क्यारी कहाँ है,
        और किस रूप में?
  \item जून में पगडंडी के किनारे पोस्ते खिले हैं; जनवरी की तस्वीर में
        उनकी जगह नंगी मिट्टी है। जनवरी में पोस्ता किस रूप में मौजूद
        है, और \cref{prop:g6:life-through-seasons:herbs} की किस
        रणनीति से?
  \item जून की तस्वीर \omterm{def:g1:plants-around-us:tree}{पेड़} के ऊपर एक अबाबील और नीचे फूलों पर
        मधुमक्खियाँ पकड़ती है; नवंबर से फ़रवरी तक इनमें से कोई नहीं
        दिखता। हर एक के लिए बताओ कि वह तब कहाँ है, और किस रूप में।
  \item एक मगपाई हर एक तस्वीर में दिखती है। यह कौन-सी जंतु-रणनीति है,
        और मगपाई को पूरी सर्दी वह क्या सँभालना पड़ता है जिससे अबाबील
        चले जाकर बच जाती है?
\end{enumerate}

\medskip\noindent\textbf{भाग III --- क़तार का नियम।}
\begin{enumerate}[resume]
  \item लीना सार बताती है: ``\omterm{def:g1:plants-around-us:tree}{पेड़} का साल \omterm{def:g6:exploring-our-environment:conditions}{परिस्थितियों} के साल के पीछे
        चलता है।'' \omterm{def:g1:plants-around-us:tree}{पेड़} में तस्वीर-दर-तस्वीर हुए दो बदलावों को उसके
        दिन-लंबाई तथा \omterm{def:g6:exploring-our-environment:conditions}{ताप} के नोटों के दो बदलावों से जोड़कर उसका समर्थन
        करो।
  \item उसका छोटा भाई कहता है कि जुलाई वाली डैफ़ोडिल की क्यारी ``मर
        गई''। \cref{rem:g6:life-through-seasons:vanish} से उसे सुधारो
        --- और वह अगला सवाल भी बताओ जिसका हक़ उसके ``चला गया'' शब्द को
        हमेशा है।
  \item तेरहवीं तस्वीर की --- यानी अगली जनवरी की तस्वीर की ---
        ब्यौरेवार भविष्यवाणी करो: शाखाएँ, टहनियाँ, नीचे की ज़मीन।
        तुम्हारी भविष्यवाणी के कौन-से हिस्से दाँव लगाने लायक़ पक्के
        हैं, और क्यों?
  \item फ़रवरी में काटी गई और घर के भीतर गरम पानी में खड़ी की गई
        चेस्टनट की टहनी तीन हफ़्तों में पत्ते निकाल देती है। इससे पता
        चलता है कि \omterm{def:g6:life-through-seasons:buds}{कलियाँ} \emph{किस} परिस्थिति का इंतज़ार कर रही थीं
        --- और वह दो-टहनी वाली न्यायी जाँच
        (\cref{met:g4:what-plants-need:fairtest}) बनाओ जो इसे पक्का
        कर दे।
\end{enumerate}
\end{problem}
