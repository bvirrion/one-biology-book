\chapter{Hoe dieren opgroeien}\label{ch:g2:animals-grow-up}

Een kitten lijkt vanaf zijn allereerste dag op een kleine kat. Maar de
kikker in de vijver begon als een visvormige donderpad zonder poten, en
de vlinder op de vlinderstruik bracht weken door als kruipende \omterm{ex:g2:animals-grow-up:butterfly}{rups}.
Jonge \omterm{def:g1:animals-around-us:animal}{dieren} lijken niet allemaal op hun ouders --- sommige moeten op
weg naar de volwassenheid volledig van gedaante veranderen.

\section{Uit het ei of levend geboren}

\begin{definition}[Twee manieren om geboren te worden]\label{def:g2:animals-grow-up:born}
Jonge \omterm{def:g1:animals-around-us:animal}{dieren} komen op twee manieren ter wereld:
\begin{itemize}
  \item de meeste komen uit een \emph{ei}\index{ei} dat de moeder legt:
        vogels, vissen, kikkers, slakken, insecten;
  \item andere worden \emph{levend geboren}\index{levend geboren}: ze
        komen al gevormd uit het lichaam van de moeder en drinken haar
        \emph{melk}\index{melk}: katten, honden, koeien, konijnen ---
        en mensen.
\end{itemize}
\end{definition}

\begin{example}[In het kippenhok en in de mand]\label{ex:g2:animals-grow-up:henhouse}
De kip legt eieren en gaat erop zitten om ze warm te houden; na drie
weken breken de kuikens de schaal en komen ze uit. In de mand bij de
kachel zijn de kittens van de poes diezelfde nacht \omterm{def:g2:animals-grow-up:born}{levend geboren}, en
binnen het uur hadden ze haar \omterm{def:g2:animals-grow-up:born}{melk} gevonden.
\end{example}

\begin{remark}[Een ei is geen steen]\label{rem:g2:animals-grow-up:egg}
In een bebroed \omterm{def:g2:animals-grow-up:born}{ei} wordt in alle stilte een kuiken gebouwd, gevoed door
de gele dooier. Het \omterm{def:g2:animals-grow-up:born}{ei} wacht niet --- het werkt. Daarom houdt de kip
haar eieren warm, en daarom mag een \omterm{def:g2:animals-grow-up:born}{ei} uit de koelkast, dat toch nooit
zou zijn uitgekomen, de kou niet worden verweten.
\end{remark}

\section{Groeien zonder van gedaante te veranderen}

\begin{definition}[Groeien als de ouders]\label{def:g2:animals-grow-up:direct}
Veel jonge \omterm{def:g1:animals-around-us:animal}{dieren} zijn \emph{miniaturen van hun
ouders}\index{miniatuur van de ouders}: het kitten is een kleine kat,
het veulen een klein paard, het kuiken een kleine kip. Opgroeien
betekent voor hen groter en sterker worden --- de vorm van het lichaam
blijft dezelfde.
\end{definition}

\begin{example}[Van kitten tot kat]\label{ex:g2:animals-grow-up:kitten}
Het pasgeboren kitten drinkt \omterm{def:g2:animals-grow-up:born}{melk} met gesloten \omterm{def:g1:the-five-senses:sense}{ogen}. Na een maand
speelt het en springt het op alles af; na zes maanden jaagt het als een
volwassen kat; na een jaar is het volwassen, misschien met eigen
kittens. Kleine kat, grotere kat, kat: onderweg steeds dezelfde vorm.
\end{example}

\section{Groeien door van gedaante te veranderen}

\begin{definition}[Gedaanteverwisseling]\label{def:g2:animals-grow-up:metamorphosis}
Sommige jonge \omterm{def:g1:animals-around-us:animal}{dieren} lijken in niets op hun ouders en moeten hun hele
lichaamsvorm veranderen om volwassen te worden. Die grote verandering
heet \emph{gedaanteverwisseling}\index{gedaanteverwisseling}. De
donderpad wordt een kikker; de \omterm{ex:g2:animals-grow-up:butterfly}{rups} wordt een vlinder.
\end{definition}

\begin{example}[De weg van de kikker]\label{ex:g2:animals-grow-up:frog}
Uit de eitjes van de kikker in de vijver komen
\emph{donderpadden}\index{donderpad}: met een ronde kop, een lange
\omterm{def:g1:animals-around-us:parts}{staart}, zonder poten, ademend in het water zoals vissen. Dan, week na
week: eerst komen de achterpoten, dan de voorpoten; de \omterm{def:g1:animals-around-us:parts}{staart} krimpt
weg; en op een dag klimt een kikkertje de oever op. Hetzelfde \omterm{def:g1:animals-around-us:animal}{dier} ---
opnieuw gebouwd.
\end{example}

\begin{example}[De weg van de vlinder]\label{ex:g2:animals-grow-up:butterfly}
Uit het eitje van de vlinder komt een \emph{rups}\index{rups}, die
weinig anders doet dan \omterm{def:g1:plants-around-us:parts}{bladeren} eten en groeien. Op een dag wikkelt ze
zich in een omhulsel, de \emph{pop}\index{pop}, en houdt ze op met
bewegen. Binnenin wordt het lichaam opnieuw gebouwd; na een paar weken
splijt het omhulsel, en er klimt een vlinder uit die natte nieuwe
vleugels ontvouwt. \omterm{def:g2:animals-grow-up:born}{Ei}, \omterm{ex:g2:animals-grow-up:butterfly}{rups}, \omterm{ex:g2:animals-grow-up:butterfly}{pop}, vlinder: vier stadia van één en
hetzelfde \omterm{def:g1:animals-around-us:animal}{dier}.
\end{example}

\begin{method}[De levenscyclus van een dier lezen]\label{met:g2:animals-grow-up:read}
Voor elk \omterm{def:g1:animals-around-us:animal}{dier} dat je tegenkomt:
\begin{enumerate}
  \item vraag hoe het \omterm{def:g2:born-grow-die:cycle}{geboren} wordt: uit een \omterm{def:g2:animals-grow-up:born}{ei}, of levend?
  \item vraag hoe het jong eruitziet: een miniatuur van zijn ouders, of
        een andere vorm?
  \item als het een andere vorm heeft, zal het van gedaante veranderen:
        zoek de stadia van zijn \omterm{def:g2:animals-grow-up:metamorphosis}{gedaanteverwisseling};
  \item teken de cyclus als een wiel, met een pijl van de volwassene
        terug naar de eitjes.
\end{enumerate}
\end{method}

\begin{remark}[Melk, of meteen aan het werk]\label{rem:g2:animals-grow-up:care}
Kittens worden wekenlang gevoed en gewassen door hun moeder; een
pasgeboren donderpad krijgt helemaal geen hulp en moet zich vanaf de
eerste dag zelf voeden. Sommige jonge \omterm{def:g1:animals-around-us:animal}{dieren} worden verzorgd, andere
beginnen hun leven alleen --- beide wegen leiden naar de volwassenheid.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:1}
Geef de twee manieren waarop jonge \omterm{def:g1:animals-around-us:animal}{dieren} ter wereld komen, met van elk
een voorbeeld.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:2}
Uit een \omterm{def:g2:animals-grow-up:born}{ei}, of \omterm{def:g2:animals-grow-up:born}{levend geboren}: (a) een kuiken; (b) een kitten; (c) een
kikker; (d) een konijn; (e) een slak?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:3}
Wat drinkt een kitten als eerste? Welke \omterm{def:g1:animals-around-us:animal}{dieren} uit dit hoofdstuk
drinken bij hun \omterm{def:g2:born-grow-die:cycle}{geboorte} hetzelfde?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:4}
Wat is \omterm{def:g2:animals-grow-up:metamorphosis}{gedaanteverwisseling}? Noem de twee beroemde voorbeelden uit dit
hoofdstuk.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:5}
Zet de stadia van de kikker op volgorde: kikker --- eitjes ---
donderpad met poten --- donderpad.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:6}
Zet de stadia van de vlinder op volgorde: vlinder --- \omterm{ex:g2:animals-grow-up:butterfly}{rups} --- \omterm{def:g2:animals-grow-up:born}{ei} ---
\omterm{ex:g2:animals-grow-up:butterfly}{pop}. In welk stadium eet het \omterm{def:g1:animals-around-us:animal}{dier} \omterm{def:g1:plants-around-us:parts}{bladeren} en groeit het het meest?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:7}
Welk jong \omterm{def:g1:animals-around-us:animal}{dier} is een miniatuur van zijn ouders: de donderpad of het
veulen? Wat betekent opgroeien voor dat \omterm{def:g1:animals-around-us:animal}{dier}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:animals-grow-up:8}
Waarom gaat de kip op haar eieren zitten? Wat gebeurt er binnenin,
volgens \cref{rem:g2:animals-grow-up:egg}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:animals-grow-up:9}
Zoë vindt in de vijver een klein \omterm{def:g1:animals-around-us:animal}{dier} met een ronde kop, een lange
\omterm{def:g1:animals-around-us:parts}{staart}, twee achterpoten en geen voorpoten. Zeg met behulp van de
figuur van de kikkercyclus precies waar het op zijn weg is, wat het
daarvoor was, en wat er nu komt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:animals-grow-up:10}
``De \omterm{ex:g2:animals-grow-up:butterfly}{rups} is in haar omhulsel gestorven, en er is een vlinder in
getrokken.'' Wat gebeurde er werkelijk in de \omterm{ex:g2:animals-grow-up:butterfly}{pop}? Antwoord met
\cref{def:g2:animals-grow-up:metamorphosis} en
\cref{ex:g2:animals-grow-up:butterfly}.
\end{exercise}
