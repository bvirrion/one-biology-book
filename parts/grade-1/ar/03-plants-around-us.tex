\chapter{النباتات من حولنا}\label{ch:g1:plants-around-us}

\omterm{def:g1:plants-around-us:parts}{زهرة} أقحوان في العشب، وطحلب على الجدار، \omterm{def:g1:plants-around-us:tree}{وشجرة} الكستناء الكبيرة في
ساحة المدرسة: لن يهرب أي منها يومًا، ومع ذلك فكلها حية. \omterm{def:g1:plants-around-us:plant}{النباتات}
هي الكائنات الحية الهادئة --- تولد وتتغذى وتنمو، مئات السنين
أحيانًا، من غير أن تبرح مكانها.

\section{النباتات كائنات حية}

\begin{definition}[النبات]\label{def:g1:plants-around-us:plant}
\emph{النبات}\index{نبات} \omterm{def:g1:living-or-not:living}{كائن حي} يبقى ثابتًا في الأرض ويصنع غذاءه
بنفسه مستعملًا الضوء. فالعشب والأقحوان وشجيرات الطماطم والأشجار
كلها نباتات. وأكثر النباتات خضراء.
\end{definition}

\begin{example}[الأقحوانة حية]\label{ex:g1:plants-around-us:daisy}
راجع علامات الحياة على \omterm{def:g1:plants-around-us:parts}{زهرة} أقحوان: وُلدت من بذرة؛ وهي تتغذى ---
بالماء من التربة والضوء من السماء؛ وتنمو؛ وتصنع بذورًا تصير
أقحوانًا جديدًا؛ وتموت في الصقيع. فكل العلامات حاضرة. الأقحوانة
حية كالقطة تمامًا --- غير أنها أهدأ منها بكثير.
\end{example}

\begin{remark}[حي من غير أن يجري]\label{rem:g1:plants-around-us:still}
\omterm{def:g1:plants-around-us:plant}{النباتات} لا تمشي، لكنها ليست ساكنة تمامًا. فعبّاد الشمس يستدير
ليتبع الضوء في أثناء النهار، \omterm{def:g1:plants-around-us:plant}{والنبتة} على حافة النافذة تميل ببطء
نحو النافذة. \omterm{def:g1:plants-around-us:plant}{النباتات} تتحرك \emph{بالنمو} --- ببطء لا تدركه
عيوننا، وبسرعة تكفي مراقبة أسبوع.
\end{remark}

\section{أجزاء النبات}

\begin{definition}[أجزاء النبات]\label{def:g1:plants-around-us:parts}
\omterm{def:g1:plants-around-us:plant}{للنبات} المزهر أربعة أجزاء رئيسة:
\begin{enumerate}
  \item \emph{الجذور}\index{جذر}، وهي مختفية في التربة، تثبّت
        \omterm{def:g1:plants-around-us:plant}{النبات} في مكانه وتشرب الماء؛
  \item \emph{الساق}\index{ساق}، وهي تقيم \omterm{def:g1:plants-around-us:plant}{النبات} وتحمل الماء إلى
        الأوراق؛
  \item \emph{الأوراق}\index{ورقة}، وهي مسطحة خضراء تلتقط الضوء؛
  \item \emph{الزهرة}\index{زهرة}، وهي التي ستصنع البذور.
\end{enumerate}
\end{definition}

\end{omfigure}

\begin{example}[اقتلاع عشبة]\label{ex:g1:plants-around-us:weed}
اقتلع عشبة برفق من تراب رخو مبلل، فترفع معها الجزء المختفي: خصلة
من \omterm{def:g1:plants-around-us:parts}{جذور} شاحبة، أطول أحيانًا من \omterm{def:g1:plants-around-us:plant}{النبتة} الخضراء كلها فوقها. الآن
يمكن قراءة \omterm{def:g1:plants-around-us:plant}{النبتة} من أسفلها إلى أعلاها: \omterm{def:g1:plants-around-us:parts}{جذور}، \omterm{def:g1:plants-around-us:parts}{وساق}، \omterm{def:g1:plants-around-us:parts}{وأوراق} ---
\omterm{def:g1:plants-around-us:parts}{وزهرة} إن حالفك الحظ.
\end{example}

\begin{definition}[الشجرة]\label{def:g1:plants-around-us:tree}
\emph{الشجرة}\index{شجرة} \omterm{def:g1:plants-around-us:plant}{نبات} كبير ساقه من خشب. وتسمى \omterm{def:g1:plants-around-us:parts}{ساق} الشجرة
الخشبية \emph{الجذع}\index{جذع}؛ وهو ينقسم إلى
\emph{أغصان}\index{غصن} تحمل \omterm{def:g1:plants-around-us:parts}{الأوراق}.
\end{definition}

\begin{example}[شجرة الكستناء]\label{ex:g1:plants-around-us:tree}
\omterm{def:g1:plants-around-us:tree}{لشجرة} الكستناء في الساحة الخطة نفسها التي للأقحوانة، لكنها أكبر
وأصلب: \omterm{def:g1:plants-around-us:parts}{جذور} تحت الأرض، وجذع مكان \omterm{def:g1:plants-around-us:parts}{الساق} الطرية، \omterm{def:g1:plants-around-us:tree}{وأغصان} عليها آلاف
\omterm{def:g1:plants-around-us:parts}{الأوراق}، وفي الربيع أزهار. الأقحوانة تعيش سنة واحدة؛ أما الكستناء
فقد تعيش أكثر من مئة سنة.
\end{example}

\section{ما يحتاج إليه النبات}

\begin{proposition}[حاجات النبات]\label{prop:g1:plants-around-us:needs}
يحتاج \omterm{def:g1:plants-around-us:plant}{النبات} لكي يعيش وينمو إلى \emph{الماء} و\emph{الضوء} ومكان
لجذوره، وهو \emph{التربة}\index{تربة} في الغالب. \omterm{def:g1:plants-around-us:plant}{والنبات} الذي
يُترك في الظلام، أو لا يُسقى أبدًا، يذبل ويموت.
\end{proposition}
\admitted

\begin{example}[النبتة المنسية]\label{ex:g1:plants-around-us:wilt}
انسَ سقي \omterm{def:g1:plants-around-us:plant}{نبتة} الصف قبل العطلة: فتتدلى أوراقها طرية حزينة --- إنها
تذبل. فاسقها، وبعد ساعات قليلة تنتصب \omterm{def:g1:plants-around-us:parts}{الأوراق} من جديد. كان الماء هو
الحاجة الناقصة.
\end{example}

\begin{method}[العناية بنبتة]\label{met:g1:plants-around-us:care}
\begin{enumerate}
  \item ضع الأصيص حيث يبلغه ضوء النهار؛
  \item اسقِ التربة حين تجدها جافة --- لا كل ساعة، \omterm{def:g1:plants-around-us:parts}{فالجذور} تحتاج
        إلى الهواء أيضًا؛
  \item لا تنزع \omterm{def:g1:plants-around-us:parts}{الأوراق}: فهي ملاقط الضوء عند \omterm{def:g1:plants-around-us:plant}{النبتة}؛
  \item وتحلَّ بالصبر: \omterm{def:g1:plants-around-us:plant}{النبتة} تجيب في أيام، لا في دقائق.
\end{enumerate}
\end{method}

\begin{remark}[النباتات تُطعم الجميع]\label{rem:g1:plants-around-us:food}
\omterm{def:g1:plants-around-us:plant}{النباتات} مهمة لكل \omterm{def:g1:animals-around-us:animal}{حيوان}. فالحلزون يأكل الخس، والشحرور يأكل التوت،
ونحن نأكل خبزًا مصنوعًا من القمح --- وهو \omterm{def:g1:plants-around-us:plant}{نبات}. وحتى القطة التي لا
تأكل الخس، تأكل \omterm{def:g1:animals-around-us:animal}{حيوانات} أكلت \omterm{def:g1:plants-around-us:plant}{نباتات}. ولولا \omterm{def:g1:plants-around-us:plant}{النباتات} لما وجد أحد
غيرها ما يأكله؛ وسيتتبع فصل لاحق هذه السلاسل الغذائية.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:1}
سمِّ أجزاء \omterm{def:g1:plants-around-us:plant}{النبات} المزهر الأربعة، من تحت التربة إلى الأعلى.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:2}
أي جزء من \omterm{def:g1:plants-around-us:plant}{النبات}: (أ) يشرب الماء من التربة؟ (ب) يلتقط الضوء؟
(ج) سيصنع البذور؟ (د) يقيم \omterm{def:g1:plants-around-us:plant}{النبات} منتصبًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:3}
الأقحوانة لا تمشي أبدًا. راجع علامات الحياة واذكر لماذا هي \omterm{def:g1:living-or-not:living}{كائن
حي} مع ذلك.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:4}
ما الاسمان الخاصان \omterm{def:g1:plants-around-us:parts}{لساق} \omterm{def:g1:plants-around-us:tree}{الشجرة} وللأجزاء التي تحمل أوراقها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:5}
اذكر الأشياء الثلاثة التي يحتاج إليها \omterm{def:g1:plants-around-us:plant}{النبات} ليعيش، من
\cref{prop:g1:plants-around-us:needs}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:6}
\omterm{def:g1:plants-around-us:parts}{أوراق} \omterm{def:g1:plants-around-us:plant}{نبتة} الصف طرية متدلية. فما الذي نُسي على الأرجح؟ وماذا يحدث
بعد أن يُعطى لها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:7}
بم تتشابه الأقحوانة \omterm{def:g1:plants-around-us:tree}{وشجرة} الكستناء؟ وبأي شيئين تختلفان؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:8}
جرّب: ضع \omterm{def:g1:plants-around-us:plant}{نبتة} فول (أو أصيص رشاد) قرب النافذة وأخرى في خزانة
مظلمة، واسقِ الاثنتين. وانظر بعد أسبوع. أي فرق تتوقع؟ وأي حاجة
تفسره؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-around-us:9}
يقول ليو: ``\omterm{def:g1:plants-around-us:plant}{النباتات} ليست حية حقًّا --- فهي لا تفعل شيئًا أبدًا.''
أعطِه أمرين تفعلهما \omterm{def:g1:plants-around-us:plant}{النبتة} ببطء وثبات، من
\cref{rem:g1:plants-around-us:still} و\cref{ex:g1:plants-around-us:daisy}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-around-us:10}
الجزرة برتقالية وتنمو تحت الأرض. وبالنظر إلى شكل أجزاء \omterm{def:g1:plants-around-us:plant}{النبات}، أي
جزء من \omterm{def:g1:plants-around-us:plant}{نبتة} الجزر تظن أننا نأكله؟ وأي دليل تعطيه عبارة ``تحت
الأرض''؟
\end{exercise}
