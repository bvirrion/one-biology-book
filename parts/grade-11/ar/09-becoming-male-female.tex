\chapter{كيف يصير المرء ذكرًا أو أنثى}\label{ch:g11:becoming-male-female}

في الأسبوع السابع بعد الإخصاب، يكون عند الجنين البشري زوج من القندين
ليسا خصيتين ولا مبيضين، وطقمان كاملان من القنوات جنبًا إلى جنب ---
أحدهما قد يصير السبيل الذكري والآخر قد يصير الأنثوي. وما من شيء مرئي
يميز جنينًا سيصير صبيًا من جنين سيصير بنتًا. وعلى مدى الأسابيع التالية
ينمو أحد طقمي القنوات ويضمر الآخر، ويأخذ القندان إحدى الصورتين، وبحلول
الشهر الرابع يكون التشريح قد تقرر. والقرار تتخذه \omterm{def:g10:universal-dna:gene}{مورّثة} واحدة، وينفذه
هرمونان. ويتابع هذا الفصل ذلك التتابع، من \omterm{def:g10:universal-dna:information}{الصبغيات} المتلقاة عند الإخصاب
إلى تغيرات \omterm{prop:g11:becoming-male-female:puberty}{البلوغ}.

\section{الصبغيات الجنسية}

\begin{definition}[الصبغيات الجنسية]\label{def:g11:becoming-male-female:chromosomes}
من أزواج \omterm{def:g10:universal-dna:information}{الصبغيات} البشرية الثلاثة والعشرين، 22 زوجًا متشابهة في
الجنسين؛ أما الزوج الثالث والعشرون فمؤلف من \emph{الصبغيات
الجنسية}\index{صبغيات جنسية}: صبغيا X عند الأنثى (XX)، \omterm{def:g10:universal-dna:information}{وصبغي} X \omterm{def:g10:universal-dna:information}{وصبغي} Y
عند الذكر (XY). \omterm{def:g11:cell-cycle-mitosis:chromosome}{والصبغي} Y صغير، يحمل \omterm{def:g10:universal-dna:gene}{مورّثات} قليلة، ولا ينتقل إلا من
الأب إلى الابن؛ وكل بييضة تحمل X، والنطفة تحمل إما X وإما Y، فنطفة
الأب هي التي تقرر الجنس \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} للطفل، بحظوظ متساوية.
\end{definition}

\begin{proposition}[الطور غير المتمايز]\label{prop:g11:becoming-male-female:undifferentiated}
حتى الأسبوع السابع، تكون أجنة الجنسين الصبغيين متطابقة في تشريحها
التناسلي: زوج من القندين غير المتمايزين، إلى جانب كل منهما قناتان ---
\emph{قناة فولف}، وهي سليفة السبيل الذكري (البربخ، والأسهر)،
و\emph{قناة مولر}، وهي سليفة السبيل الأنثوي (قناة البيض، والرحم).
فالنمو الجنسي هو تحوّل هذا المخطط المشترك إلى إحدى صورتين؛ وتمايز
القند هو الذي يبدأه.
\end{proposition}
\admitted

\section{المورّثة التي تقرر}

\begin{proposition}[SRY والقند]\label{prop:g11:becoming-male-female:sry}
يحمل \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} Y \omterm{def:g10:universal-dna:gene}{مورّثة}، هي \emph{SRY}\index{مورّثة SRY}، تعبَّر في القند
غير المتمايز في الأسبوع السابع. وبروتينها يشغّل \omterm{def:g10:universal-dna:gene}{المورّثات} التي تحوّل
القند إلى \emph{خصية}. وفي غيابها --- عند جنين XX --- يصير القند
\emph{مبيضًا} بعد بضعة أسابيع. فوجود \omterm{prop:g11:becoming-male-female:sry}{SRY}، لا عدد صبغيات X، هو الذي
يقرر القند.
\end{proposition}
\begin{proof}[شواهد]
أفراد XY نادرون تحمل \omterm{prop:g11:becoming-male-female:sry}{مورّثة SRY} عندهم طفرةً ينمو عندهم مبيضان وجسم
أنثوي؛ وأفراد XX نادرون يحملون قطعة من Y فيها \omterm{prop:g11:becoming-male-female:sry}{SRY}، منقولة إلى \omterm{def:g10:universal-dna:information}{صبغي} X،
تنمو عندهم خصيتان وجسم ذكري. وفي الفئران: جنين XX أدخلت فيه \omterm{def:g10:universal-dna:gene}{مورّثة}
الفأر \emph{Sry} ينمو ذكرًا. \omterm{def:g10:universal-dna:gene}{فمورّثة} واحدة من \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي} Y كافية، ولازمة،
لصنع خصية.
\end{proof}

\begin{proposition}[هرمونا الخصية الجنينية]\label{prop:g11:becoming-male-female:hormones}
حالما تتكون الخصية تفرز هرمونين. \emph{فالتستوستيرون}\index{تستوستيرون}
يحفظ قناتي فولف ويجعلهما تنموان إلى السبيل الذكري الداخلي، ويشكّل
الأعضاء التناسلية الخارجية في الصورة الذكرية. أما \emph{الهرمون المضاد
لمولر} (AMH) فيجعل قناتي مولر تضمران. والمبيض لا يفرز أيًّا منهما في
هذا الطور: فمن دون التستوستيرون تضمر قناتا فولف، ومن دون AMH تصير
قناتا مولر قناتي البيض والرحم وأعلى المهبل. فالسبيل الأنثوي هو إذن
الصورة التي تأخذها القنوات في غياب هرموني الخصية.
\end{proposition}
\begin{proof}[شواهد]
استأصل جوست (1947) قندي أجنة أرانب في الرحم قبل التمايز: فنما عند كل
جنين، XX كان أو XY، سبيل أنثوي. وزرع خصية إلى جانب قندي جنين مخصي
أنتج سبيلًا ذكريًا؛ أما غرس بلورة من التستوستيرون وحدها فحفظ قناتي فولف
لكنه لم يزل قناتي مولر، فبقيتا إلى جانبهما. فالخصية تصنع إذن ناتجين ---
واحدًا يبني القنوات الذكرية (التستوستيرون) وواحدًا، عرف لاحقًا بأنه
AMH، يزيل الأنثوية --- والمبيض غير لازم لتكوّن السبيل الأنثوي.
\end{proof}

\begin{example}[التتابع بالترتيب]\label{ex:g11:becoming-male-female:cascade}
إخصاب بنطفة حاملة \omterm{def:g10:universal-dna:information}{للصبغي} Y؛ ثم الأسبوع السابع، تعبَّر \omterm{prop:g11:becoming-male-female:sry}{SRY} فيصير القند خصية؛
ثم الأسابيع 8 إلى 12، التستوستيرون وAMH: فتحفظ قناتا فولف، وتزول قناتا
مولر، وتشكَّل الأعضاء التناسلية الخارجية ذكريةً؛ ثم سكون حتى \omterm{prop:g11:becoming-male-female:puberty}{البلوغ}.
وإخصاب بنطفة حاملة \omterm{def:g10:universal-dna:information}{للصبغي} X: فلا \omterm{prop:g11:becoming-male-female:sry}{SRY}؛ ثم الأسابيع 10 إلى 12، يصير القند
مبيضًا؛ ولا هرموني خصية: فتنمو قناتا مولر، وتضمر قناتا فولف، وتأخذ
الأعضاء التناسلية الخارجية الصورة الأنثوية. وكل خطوة تتوقف على التي
قبلها، وإخفاق أي خطوة يغير كل ما بعدها.
\end{example}

\section{البلوغ}

\begin{proposition}[البلوغ]\label{prop:g11:becoming-male-female:puberty}
الأعضاء التناسلية، وقد تكوّنت قبل الولادة، تبقى خاملة طوال الطفولة.
وعند \emph{البلوغ}\index{بلوغ} --- حول الحادية عشرة إلى الثالثة عشرة،
وأبكر عند البنات --- يبدأ مركز في الدماغ بإطلاق هرمون، على نبضات، يجعل
الغدة النخامية تفرز هرمونين في الدم، هما LH وFSH. وهذان يؤثران في
القندين: فتبدأ الخصية بصنع النطاف وبإفراز التستوستيرون بمقادير بالغة؛
ويبدأ المبيض دوراته ويفرز الإستروجينات. ثم تنتج الهرمونات الجنسية
\emph{الصفات الجنسية الثانوية}: نمو الأعضاء التناسلية، وشعر البدن،
والصوت، والنهدين، وتوزع العضل والشحم، \omterm{def:g11:mutations:mutation}{وطفرة} النمو ثم انغلاق صفائح
النمو في العظام. ويتم \omterm{prop:g11:becoming-male-female:puberty}{البلوغ} حين يصير الفرد قادرًا على التكاثر.
\end{proposition}
\admitted

\section{الجنس بمعانيه العدة}

\begin{proposition}[التمييز بين المستويات]\label{prop:g11:becoming-male-female:levels}
تنطبق كلمتا «ذكر» و«أنثى» على مستويات عدة تتوافق عادةً لكنها متمايزة:
\begin{itemize}
  \item الجنس \emph{\omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي}}: XX أو XY، وهو ثابت عند الإخصاب؛
  \item الجنس \emph{القندي}: مبيض أو خصية، تقرره \omterm{prop:g11:becoming-male-female:sry}{SRY}؛
  \item الجنس \emph{التشريحي}: السبل والأعضاء التناسلية الخارجية،
        تشكّلها هرمونات القند الجنيني، والصفات الثانوية، وتتشكل عند
        \omterm{prop:g11:becoming-male-female:puberty}{البلوغ}؛
  \item \emph{الهوية الجندرية}، وهي إحساس الشخص بأنه رجل أو امرأة أو
        لا هذا ولا ذاك، و\emph{الميل الجنسي}، أي من ينجذب إليه الشخص
        --- وكلاهما واقعتان عن الشخص لا تستنتجان من المستويات
        السابقة ولا تختاران.
\end{itemize}
وبيولوجيا هذا الفصل تصف الثلاثة الأولى؛ وهي لا تحدد الأخير ولا تحكم
عليه.
\end{proposition}
\admitted

\begin{example}[حين تختلف المستويات]\label{ex:g11:becoming-male-female:disagree}
نحو شخص من كل بضعة آلاف يولد بتركيب لا ينتجه التتابع عادةً: فشخص XY
تفتقر خلاياه إلى مستقبل التستوستيرون ينمو عنده جسم أنثوي، بخصيتين لم
تنزلا قط وبلا رحم؛ وشخص XX عنده انتقال \omterm{def:g10:universal-dna:information}{صبغي} \omterm{prop:g11:becoming-male-female:sry}{للمورّثة SRY} مكوِّن للخصية ينمو
عنده جسم ذكري؛ وشخص XY عنده \omterm{prop:g11:becoming-male-female:sry}{SRY} غير عاملة ينمو أنثى. وهذه المتغايرات
تبين، أفضل مما تبين الحال العادية، أن كل مستوى حصيلة خطوة بعينها ---
\omterm{def:g10:universal-dna:gene}{مورّثة}، أو هرمون، أو مستقبل --- وأن المستويات وقائع منفصلة.
\end{example}

\begin{method}[الاستدلال في حالة]\label{met:g11:becoming-male-female:case}
\begin{enumerate}
  \item \omterm{def:g10:universal-dna:information}{الصبغيات}: هل \omterm{prop:g11:becoming-male-female:sry}{SRY} موجودة وعاملة؟
  \item القند: خصية إن عملت \omterm{prop:g11:becoming-male-female:sry}{SRY}، ومبيض في غير ذلك.
  \item هرمونا القند الجنيني: تستوستيرون وAMH، أو لا هذا ولا ذاك؛ وهل
        تستطيع النسج الاستجابة (المستقبلات)؟
  \item القنوات: قناة فولف تحفظ وقناة مولر تزال إن عمل الهرمونان؛
        وقناة مولر تنمو إن لم يعمل أي منهما؛ والقناتان معًا إن عمل
        التستوستيرون دون AMH.
  \item \omterm{prop:g11:becoming-male-female:puberty}{البلوغ}: أي هرمون جنسي يفرزه القند يقرر الصفات الثانوية ---
        وذلك عبر المستقبلات كذلك.
\end{enumerate}
\end{method}

\begin{remark}[مورّثة واحدة، وكل ما بعدها]\label{rem:g11:becoming-male-female:onegene}
يرجع الفرق كله بين التشريحين إلى ما إذا كانت \omterm{def:g10:universal-dna:gene}{مورّثة} واحدة على \omterm{def:g10:universal-dna:information}{صبغي} واحد
موجودة في الأسبوع السابع. وما يتبع ذلك تتابع من بيولوجيا \omterm{def:g10:cells-common-unit:cell}{الخلية}
العادية --- تعبير مورّثي، وهرمونات، ومستقبلات --- من الصنف الذي وصفه كل
فصل من هذه السنة. والمخطط الذي يبدأ منه التتابع مشترك بين الجنسين،
والصورة الأنثوية هي التي يبنيها حين لا تتدخل الخصية.
\end{remark}

\section{مسألة: أرانب جوست}

\begin{problem}[{مسألة نهاية الأسبوع --- التجارب التي وجدت هرموني
الخصية، يعاد بناؤها خطوة خطوة، ثم تطبق على ثلاث حالات بشرية وعلى
توقيت البلوغ}]
\label{pb:g11:becoming-male-female:1}
عند الأرنب، يتمايز القندان حول اليوم 19 من الحمل وتتمايز القنوات حول
اليوم 22. وقد أجرى جوست جراحته على الأجنة في الرحم في اليوم 19 وفحصها
عند الولادة (اليوم 30).

\medskip
\noindent\textbf{الجزء الأول --- التجارب.}
\begin{enumerate}
  \item أجنة من الجنسين الصبغيين، استؤصل قنداها في اليوم 19: كلها
        تولد بقناتي بيض ورحم وبلا قنوات ذكرية. فما الذي يبينه ذلك عن
        دور المبيض، وعن دور الخصية؟
  \item لماذا كان لا بد من الجراحة قبل اليوم 22؟
  \item أجنة خصيت في اليوم 19، وزرعت خصية في البطن: فنمت قنوات ذكرية
        ولا قنوات أنثوية. فما الذي تقدمه الطعم، وكيف يبلغ القنوات؟
  \item أجنة خصيت في اليوم 19، وغرست بلورة من التستوستيرون: فوجدت
        قنوات ذكرية ووجدت قنوات أنثوية كذلك. فما الذي يفعله
        التستوستيرون، وما الذي لا يفعله؟
  \item استنتج أن على الخصية أن تصنع مادة ثانية، واذكر أثرها. وكيف
        أمكن تأكيد وجودها لاحقًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- قند واحد يستأصل.}
\begin{enumerate}[resume]
  \item تستأصل خصية واحدة من جنين XY في اليوم 19. تنبأ بالقنوات على كل
        جانب من الجسم، علمًا بأن الهرمونين يؤثران في الجانب الأقرب
        غالبًا.
  \item تزرع خصية إلى جانب مبيض واحد من جنين XX. تنبأ بكل جانب.
  \item تغرس بلورة من التستوستيرون إلى جانب مبيض واحد من جنين XX.
        تنبأ بكل جانب.
  \item لماذا تقوّي هذه التجارب أحادية الجانب استنتاج الجزء الأول؟
  \item فسّر لماذا يكون جنين XX المزروعة فيه خصية عقيمًا مع ذلك حين
        يبلغ.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- ثلاث حالات بشرية.}
طبّق \cref{met:g11:becoming-male-female:case} على كل منها.
\begin{enumerate}[resume]
  \item شخص XX يحمل، على أحد صبغيي X، قطعة من Y فيها \omterm{prop:g11:becoming-male-female:sry}{SRY}.
  \item شخص XY يحمل \omterm{prop:g11:becoming-male-female:sry}{SRY} غير عاملة.
  \item شخص XY خصيتاه طبيعيتان لكن خلاياه لا تستطيع الاستجابة لهرمون AMH.
  \item في أي الحالات الثلاث يرجح أن يكون الشخص خصبًا؟ علّل.
  \item ما الذي تبينه الحالات الثلاث عن العلاقة بين الجنس \omterm{def:g11:cell-cycle-mitosis:chromosome}{الصبغي}
        والجنس التشريحي؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- \omterm{prop:g11:becoming-male-female:puberty}{البلوغ} بالأرقام.}
من شكل الهرمونات، خذ \omterm{def:g11:mutations:mutation}{طفرة} النمو تبدأ بعد سنة من بلوغ الهرمون 20\% من
مستواه البالغ، وخذ العظام تكف عن النمو حين يبلغ 100\%.
\begin{enumerate}[resume]
  \item قدّر العمر الذي تبدأ عنده \omterm{def:g11:mutations:mutation}{طفرة} النمو عند البنات وعند الصبيان،
        والعمر الذي يتوقف عنده النمو.
  \item كم سنة من النمو لكل جنس بين بداية \omterm{def:g11:mutations:mutation}{الطفرة} ونهايتها؟
  \item صبي ينمو \qty{8}{cm} في السنة أثناء \omterm{def:g11:mutations:mutation}{الطفرة} وبنت \qty{7}{cm}.
        قدّر الطول المكتسب أثناء \omterm{def:g11:mutations:mutation}{الطفرة} عند كل منهما، وعلّق على الفرق
        الوسطي في طول البالغ (نحو \qty{13}{cm}).
  \item بنت يبدأ ارتفاع الإستروجين عندها في السابعة تنمو سريعًا باكرًا
        وتتوقف باكرًا. فسّر لماذا يؤخر الأطباء أحيانًا بلوغًا بالغ
        الباكورة بالأدوية.
  \item اذكر النتيجة: الهرمونان اللذان كشفتهما أرانب جوست، وأثر كل
        منهما، \omterm{def:g10:universal-dna:gene}{والمورّثة} الواحدة التي تشغّل العضو الذي يصنعهما.
\end{enumerate}
\end{problem}
