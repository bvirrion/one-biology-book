\chapter{A reprodução humana}\label{ch:g5:human-reproduction}

Toda pessoa que você conhece --- amigos, professores, a pessoa mais
velha da sua família --- chegou do mesmo jeito: como um bebê, nascido
depois de meses crescendo dentro da mãe. Os seres humanos são
\omterm{prop:g3:sorting-living-things:groups}{mamíferos} (\cref{prop:g4:classifying-animals:classes}), e a
reprodução humana funciona pelo plano dos \omterm{prop:g3:sorting-living-things:groups}{mamíferos} do
\cref{ch:g4:animal-reproduction} --- com os números, as datas e os
detalhes da nossa própria \omterm{def:g5:biodiversity:species}{espécie}, que este capítulo apresenta com
clareza.

\section{A versão humana do plano}

\begin{proposition}[A reprodução humana é reprodução de mamífero]\label{prop:g5:human-reproduction:plan}
Como todos os \omterm{prop:g3:sorting-living-things:groups}{mamíferos}, os seres humanos se reproduzem sexuadamente
(\cref{def:g4:animal-reproduction:sexual}):
\begin{itemize}
  \item o corpo da mãe produz \emph{\omterm{def:g4:animal-reproduction:sexual}{células reprodutoras femininas}};
        o do pai produz \emph{\omterm{def:g4:animal-reproduction:sexual}{espermatozoides}};
  \item a \omterm{def:g5:flower-to-fruit:steps}{fecundação} --- um \omterm{def:g4:animal-reproduction:sexual}{espermatozoide} se juntando a uma \omterm{def:g4:animal-reproduction:sexual}{célula
        reprodutora feminina} --- acontece \emph{dentro do corpo da
        mãe} (\cref{prop:g4:animal-reproduction:where});
  \item a \omterm{def:g4:animal-reproduction:sexual}{célula reprodutora feminina} fecundada --- uma única \omterm{def:g5:first-look-at-cells:cell}{célula}
        (\cref{ex:g5:first-look-at-cells:one}) --- se instala no
        \emph{útero}\index{útero} da mãe e cresce ali;
  \item os seres humanos são \omterm{def:g4:animal-reproduction:ovivivi}{vivíparos}
        (\cref{def:g4:animal-reproduction:ovivivi}): o bebê nasce de
        barriga e no começo é alimentado com \omterm{def:g2:animals-grow-up:born}{leite}.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[A reprodução exige corpos adultos]\label{rem:g5:human-reproduction:adults}
O corpo de uma criança não pode fazer bebês: os órgãos reprodutores
só começam a funcionar na \emph{puberdade}, a mudança da etapa
\omterm{def:g3:stages-of-life:stages}{adolescente} da \cref{def:g3:stages-of-life:stages}, quando o corpo se
reconstrói de corpo de criança em corpo de adulto. O que a puberdade
muda, e como, é o assunto de um capítulo inteiro nos anos finais do
ensino fundamental deste livro.
\end{remark}

\section{Nove meses no útero}

\begin{definition}[Gravidez]\label{def:g5:human-reproduction:pregnancy}
A \emph{gravidez}\index{gravidez} é o tempo que o futuro bebê passa
crescendo no \omterm{prop:g5:human-reproduction:plan}{útero} da mãe: cerca de \emph{nove meses} nos seres
humanos. O bebê em crescimento é chamado primeiro de
\emph{embrião}\index{embrião} e depois --- assim que o plano do corpo
dele está traçado, com uns dois meses --- de \emph{feto}\index{feto}.
\end{definition}

\begin{proposition}[Alimentado pelo cordão]\label{prop:g5:human-reproduction:cord}
Dentro do \omterm{prop:g5:human-reproduction:plan}{útero}, o bebê boia numa bolsa de líquido e não consegue
comer nem respirar. Tudo chega pronto do \omterm{prop:g4:journey-of-food:absorb}{sangue} da mãe: \omterm{def:g4:journey-of-food:digestion}{nutrientes} e
oxigênio (\cref{prop:g4:heart-and-blood:carries}) atravessam para o
\omterm{prop:g4:journey-of-food:absorb}{sangue} do bebê pelo \emph{cordão umbilical}\index{cordão umbilical},
na barriga dele --- e os resíduos dele atravessam de volta pelo mesmo
caminho, para o corpo da mãe cuidar deles. O seu umbigo é onde o seu
próprio cordão estava preso.
\end{proposition}
\admitted

\begin{example}[O calendário dos nove meses]\label{ex:g5:human-reproduction:timetable}
Mês 1: um aglomerado de \omterm{def:g5:first-look-at-cells:cell}{células} se dividindo, menor que um grão de
arroz, mas já crescendo depressa. Mês 2: um \omterm{def:g5:human-reproduction:pregnancy}{embrião} com o esboço de
uma cabeça, um \omterm{def:g4:heart-and-blood:heart}{coração} batendo, brotos de \omterm{def:g1:my-body:parts}{braços} e \omterm{def:g1:my-body:parts}{pernas} --- ainda
menor que um polegar. Meses 3 e 4: um \omterm{def:g5:human-reproduction:pregnancy}{feto}, completo no plano, que
começa a se mexer; a mãe logo sente os chutes. Meses 5 a 8:
crescendo e amadurecendo --- chega a audição e uma primeira camada
de pelinhos. Mês 9: o bebê, em geral de cabeça para baixo e com uns
\qty{3}{kg}, está pronto.
\end{example}

\begin{example}[O nascimento]\label{ex:g5:human-reproduction:birth}
No \omterm{def:g2:born-grow-die:cycle}{nascimento}, \omterm{def:g3:bones-and-muscles:muscle}{músculos} poderosos do \omterm{prop:g5:human-reproduction:plan}{útero} empurram o bebê para o
mundo, em geral com a cabeça primeiro. O primeiro choro enche os
pulmões de ar --- eles nunca tinham respirado antes
(\cref{prop:g5:human-reproduction:cord}) ---, o cordão, já sem
utilidade, é cortado sem dor, e o plano dos \omterm{prop:g3:sorting-living-things:groups}{mamíferos} continua do
lado de fora: \omterm{def:g2:animals-grow-up:born}{leite}, calor e cuidado (o
\cref{ex:g4:animal-reproduction:whalecalf} aprovaria).
\end{example}

\begin{remark}[Gêmeos]\label{rem:g5:human-reproduction:twins}
Às vezes dois bebês crescem juntos no \omterm{prop:g5:human-reproduction:plan}{útero}. Se uma \omterm{def:g4:animal-reproduction:sexual}{célula
reprodutora feminina} fecundada se partiu em dois \omterm{def:g5:human-reproduction:pregnancy}{embriões}, os gêmeos
são \emph{idênticos} --- construídos a partir da mesma \omterm{def:g5:first-look-at-cells:cell}{célula}
inicial, iguais como duas cópias. Se duas \omterm{def:g4:animal-reproduction:sexual}{células reprodutoras
femininas} foram fecundadas por dois \omterm{def:g4:animal-reproduction:sexual}{espermatozoides}, os gêmeos não
são mais parecidos que irmãos comuns --- eles só dividiram o \omterm{prop:g5:human-reproduction:plan}{útero}.
Dois tipos de gêmeos, distinguidos pelo começo deles.
\end{remark}

\section{Cuidado antes e depois}

\begin{proposition}[O bebê divide os suprimentos da mãe]\label{prop:g5:human-reproduction:care}
Tudo o que chega ao bebê viaja pelo \omterm{prop:g4:journey-of-food:absorb}{sangue} da mãe --- e é por isso
que uma gestante se alimenta bem (\cref{ch:g3:balanced-diet}),
descansa, é acompanhada por médicos ou parteiras e evita o que
poderia fazer mal ao bebê: a fumaça do cigarro e o álcool também
passam pelo cordão, e um \omterm{def:g5:human-reproduction:pregnancy}{feto} é muito mais frágil que um adulto.
Cuidar da mãe \emph{é} cuidar do bebê.
\end{proposition}
\admitted

\begin{example}[Uma espécie muito cuidada]\label{ex:g5:human-reproduction:longcare}
Mesmo entre os \omterm{prop:g3:sorting-living-things:groups}{mamíferos}, os seres humanos levam ao extremo a
estratégia de poucos e bem cuidados da
\cref{prop:g4:animal-reproduction:strategies}: em geral um bebê de
cada vez, indefeso por muito mais tempo que um potrinho ou um
gatinho, e cuidado --- alimentado, ensinado, amado --- durante muitos
\emph{anos}. A infância humana é uma das mais longas do mundo
\omterm{def:g1:animals-around-us:animal}{animal}; este livro, um curso de uma década, é ele mesmo um pedaço
desse cuidado longo.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:1}
Que \omterm{def:g5:first-look-at-cells:cell}{célula} vem da mãe, qual vem do pai, e como se chama a junção
delas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:2}
Onde o bebê humano cresce, e por quanto tempo mais ou menos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:3}
Quais são os dois nomes do bebê em crescimento, em ordem, e mais ou
menos quando o primeiro vira o segundo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:4}
O bebê no \omterm{prop:g5:human-reproduction:plan}{útero} não come nem respira. Como ele recebe \omterm{def:g4:journey-of-food:digestion}{nutrientes} e
oxigênio, e para onde vão os resíduos dele?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:5}
De que o seu umbigo é a marca?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:6}
O que acontece no primeiro choro, e por que essa é uma mudança tão
grande para o corpo do bebê?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:7}
Que características fazem da reprodução humana uma reprodução
tipicamente de mamífero? Diga três, citando a
\cref{prop:g4:classifying-animals:classes} ou o
\cref{ch:g4:animal-reproduction}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:human-reproduction:8}
Por que os médicos pedem que as gestantes evitem a fumaça do cigarro
e o álcool? Use a \cref{prop:g5:human-reproduction:care}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:9}
Explique os dois tipos de gêmeos pelo começo deles. Que tipo é igual
``como duas cópias'', e por que isso decorre do
\cref{ex:g5:first-look-at-cells:one}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:human-reproduction:10}
Ponha o ser humano na linha das estratégias da
\cref{prop:g4:animal-reproduction:strategies}, comparando com o
bacalhau e a elefanta: número de filhotes, duração do cuidado e o
que esse cuidado longo contém no caso humano.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g5:human-reproduction:11}
No \omterm{def:g2:born-grow-die:cycle}{nascimento}, o corte do cordão encerra a linha de abastecimento do
bebê exatamente no momento em que os pulmões tomam o primeiro
fôlego. Usando a \cref{prop:g5:human-reproduction:cord} e a
\cref{prop:g4:breathing:exchange}, explique que trabalho passa,
nesse minuto, do corpo da mãe para os pulmões do próprio bebê --- e
por que o primeiro choro é bem cronometrado.
\end{exercise}
