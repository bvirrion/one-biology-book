\chapter{एंजाइम और लक्षणप्ररूप}\label{ch:g11:enzymes-and-phenotype}

स्याम की कोई बिल्ली अपने शरीर के अधिकांश हिस्से पर मलाई जैसी और अपने
कानों, थूथन, पंजों तथा पूँछ पर --- यानी ठंडे हिस्सों पर --- गहरी भूरी होती
है। उसकी पीठ पर एक टुकड़ा मूँड़ दीजिए, रोएँ दोबारा उगते समय उस टुकड़े को
बर्फ़ की थैली के नीचे ठंडा रखिए, और नए रोएँ गहरे रंग के आते हैं। बिल्ली के
\omterm{def:g10:universal-dna:gene}{जीन} नहीं बदले; बस उसका वर्णक बनाने वाला \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} \qty{35}{\celsius} से नीचे
काम करता है और उससे ऊपर नहीं। किसी \omterm{def:g10:universal-dna:gene}{जीन} और किसी दिखने वाले लक्षण के बीच
लगभग हमेशा कोई \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} खड़ा होता है; और उस \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} तथा उस लक्षण के बीच वह
पर्यावरण जिसमें वह काम करता है। यह अध्याय इन्हीं दो बिचौलियों के बारे में
है।

\section{तीन पैमानों पर लक्षणप्ररूप}

\begin{definition}[जीनप्ररूप और लक्षणप्ररूप]\label{def:g11:enzymes-and-phenotype:phenotype}
किसी व्यक्ति का \emph{जीनप्ररूप}\index{जीनप्ररूप} उसके ढोए हुए विकल्पियों
का समुच्चय है; और उसका \emph{लक्षणप्ररूप}\index{लक्षणप्ररूप} उसके देखे जा
सकने वाले अभिलक्षणों का समुच्चय। लक्षणप्ररूप का वर्णन तीन पैमानों पर किया
जा सकता है:
\begin{itemize}
  \item \emph{आणविक} पैमाना: कौन-से \omterm{def:g11:gene-expression:protein}{प्रोटीन} मौजूद हैं, किस रूप में, और
        कितनी सक्रियता के साथ;
  \item \emph{कोशिकीय} पैमाना: \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} का आकार, अंतर्वस्तु और व्यवहार;
  \item \emph{जीव} का पैमाना: दिखने वाले लक्षण, आँखों के रंग से लेकर किसी
        रोग के लक्षणों तक।
\end{itemize}
हर पैमाना अपने नीचे वाले पैमाने का नतीजा है।
\end{definition}

\begin{example}[तीन पैमानों पर हँसिया-कोशिका रोग]\label{ex:g11:enzymes-and-phenotype:sickle}
आणविक: हीमोग्लोबिन शृंखला का एक \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्ल} (छठा, यानी ग्लूटामेट) वैलीन से
बदल जाता है; और ऑक्सीजन कम हो तो वे बदले हुए अणु आपस में चिपककर लंबे तंतु
बना लेते हैं। कोशिकीय: वे तंतु लाल रक्त \omterm{def:g10:cells-common-unit:cell}{कोशिका} को कड़े हँसिये जैसा बना देते
हैं, जो छोटी वाहिकाएँ रोक देता है और आसानी से टूट जाता है। जीव का:
रक्ताल्पता, वाहिकाएँ रुकने पर दर्द के दौरे, तथा प्लीहा और गुर्दों को
नुक़सान। एक \omterm{def:g10:universal-dna:gene}{जीन} के एक \omterm{def:g11:gene-expression:code}{कोडॉन} में एक प्रतिस्थापन, और उसके नतीजे के तीन वर्णन।
\end{example}

\begin{proposition}[लक्षणप्ररूप प्रोटीन बनाते हैं]\label{prop:g11:enzymes-and-phenotype:proteins}
\omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} का हर स्तर प्रोटीनों से बनता है: कोई \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} किसी \omterm{def:g11:gene-expression:protein}{प्रोटीन} को तय
करता है (\cref{ch:g11:gene-expression}); उस \omterm{def:g11:gene-expression:protein}{प्रोटीन} की सक्रियता \omterm{def:g10:cells-common-unit:cell}{कोशिका} का
रसायन तथा बनावट तय करती है; और \omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} जीव के लक्षण तय करती हैं। उनमें से
अधिकांश \omterm{def:g11:gene-expression:protein}{प्रोटीन} \emph{\omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम}} हैं, इसलिए अधिकांश लक्षण एंजाइमों से चलने
वाली अभिक्रियाओं की शृंखलाओं का नतीजा हैं।
\end{proposition}
\admitted

\section{एंजाइम}

\begin{definition}[एंजाइम, क्रियाधार, सक्रिय स्थल]\label{def:g11:enzymes-and-phenotype:enzyme}
\emph{एंजाइम}\index{एंजाइम} वह \omterm{def:g11:gene-expression:protein}{प्रोटीन} है जो किसी एक रासायनिक अभिक्रिया को
उत्प्रेरित करता है: वह उस अभिक्रिया को लाखों गुना तेज़ चला देता है और ख़ुद
ख़र्च नहीं होता। जिस अणु को वह बदलता है वह उसका
\emph{क्रियाधार}\index{क्रियाधार} है; और वह अभिक्रिया एंजाइम की एक जेब में
होती है, यानी \emph{सक्रिय स्थल}\index{सक्रिय स्थल} में, जिसका आकार और
रसायन उसी क्रियाधार से मेल खाता है, किसी और से नहीं। यह
\emph{विशिष्टता} एंजाइम के त्रि-आयामी मोड़ से, यानी उसके \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्ल} अनुक्रम
से, यानी उसके \omterm{def:g10:universal-dna:gene}{जीन} से निकलती है।
\end{definition}

\begin{example}[लैक्टेज़]\label{ex:g11:enzymes-and-phenotype:lactase}
दूध की शर्करा, यानी लैक्टोस, को छोटी आँत की परत में लैक्टेज़ \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} दो सरल
शर्कराओं में तोड़ देता है; और बिना टूटे वह बिना पचे बृहदांत्र तक पहुँच जाती
है, जहाँ जीवाणु उसका \omterm{def:g10:cell-metabolism:fermentation}{किण्वन} करते हैं, और नतीजे में गैस, मरोड़ तथा दस्त होते
हैं। सारे शिशु लैक्टेज़ बनाते हैं; अधिकांश मनुष्यों में दूध छुड़ाने के बाद
उसका \omterm{def:g10:universal-dna:gene}{जीन} बंद कर दिया जाता है, और वयस्क लैक्टोस असहिष्णु हो जाते हैं।
दुग्ध-पालन के लंबे इतिहास वाली आबादियों में वह \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} आम है जो उस \omterm{def:g10:universal-dna:gene}{जीन} को
जीवन भर सक्रिय रखता है --- उत्तरी यूरोप में 90\%, पूर्वी एशिया में 10\%।
असहिष्णु लोगों को बेची जाने वाली लैक्टेज़ की बूँदों में वही \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} होता है,
जो कवकों से बनाया जाता है।
\end{example}

\begin{proposition}[सक्रियता की शर्तें]\label{prop:g11:enzymes-and-phenotype:conditions}
किसी \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} की सक्रियता उसके परिवेश पर निर्भर है। वह ताप के साथ तब तक बढ़ती
है जब तक \omterm{def:g11:gene-expression:protein}{प्रोटीन} खुलने न लगे, और फिर ढह जाती है --- मानव \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम}
\qty{37}{\celsius} के पास सबसे अच्छा काम करते हैं और लगभग \num{45} से ऊपर
नष्ट हो जाते हैं; हर \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} अपनी जगह के अनुकूल अम्लता की एक परास में काम
करता है (आमाशय के अम्ल में पेप्सिन, उदासीन आँत में ट्रिप्सिन); और उसकी दर
अपने \omterm{def:g11:enzymes-and-phenotype:enzyme}{क्रियाधार} की सांद्रता के साथ तब तक बढ़ती है जब तक कोई अधिकतम न आ जाए,
यानी जब तक हर \omterm{def:g11:enzymes-and-phenotype:enzyme}{सक्रिय स्थल} व्यस्त न हो जाए।
\end{proposition}
\admitted

\section{अभिक्रियाओं की शृंखलाएँ}

\begin{proposition}[उपापचयी पथ]\label{prop:g11:enzymes-and-phenotype:pathway}
कोई \omterm{def:g10:cells-common-unit:cell}{कोशिका} किसी उत्पाद को शायद ही एक ही क़दम में बनाती है: कोई शुरुआती अणु
एंजाइमों की एक क़तार से बदला जाता है, जिसमें हर \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} पिछले के उत्पाद पर
काम करता है, और यही \emph{उपापचयी पथ}\index{उपापचयी पथ} है। हर \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} का
अपना \omterm{def:g10:universal-dna:gene}{जीन} उसका कूट है। उनमें से किसी एक का भी काम न करने वाला \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} उस
क़दम पर पथ रोक देता है: उत्पाद बनता नहीं, और रुके हुए \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} का \omterm{def:g11:enzymes-and-phenotype:enzyme}{क्रियाधार}
जमा हो सकता है। इस तरह दो अलग-अलग \omterm{def:g10:universal-dna:gene}{जीन} एक ही दिखने वाली ख़राबी दे सकते हैं,
और एक ही \omterm{def:g10:universal-dna:gene}{जीन} की ख़राबी कई ख़राबियाँ।
\end{proposition}
\admitted

\section{जीनप्ररूप, पर्यावरण, लक्षणप्ररूप}

\begin{proposition}[पर्यावरण लक्षणप्ररूप को गढ़ता है]\label{prop:g11:enzymes-and-phenotype:environment}
\omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} \emph{और} पर्यावरण, दोनों का नतीजा है। पर्यावरण
आणविक पैमाने पर असर डालता है (ताप और अम्लता \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} की सक्रियता तय करते
हैं, किसी \omterm{def:g11:enzymes-and-phenotype:enzyme}{क्रियाधार} की मौजूदगी तय करती है कि कोई पथ चलेगा या नहीं),
कोशिकीय पैमाने पर (\omterm{def:g10:cells-common-unit:cell}{कोशिकाएँ} संकेतों, पोषक तत्वों, प्रकाश का जवाब देती हैं)
और जीव के पैमाने पर (आहार, व्यायाम, रोग)। एक जैसे \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} वाले व्यक्ति
--- समरूप जुड़वाँ, किसी एक पौधे की क़लमें --- अपने-अपने इतिहास के हिसाब से
\omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} में अलग होते हैं; और अलग-अलग \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} वाले व्यक्ति उपयुक्त
हालात में एक ही \omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} पर आ मिल सकते हैं।
\end{proposition}
\begin{proof}[साक्ष्य]
स्याम की बिल्ली का ताप-संवेदी टायरोसिनेज़: वर्णक केवल ठंडे सिरों पर बनता
है, और ठंडा रखा गया मूँड़ा हुआ टुकड़ा गहरे रंग में दोबारा उगता है। आहार से
क़ाबू में रखा गया फेनिलकीटोनूरिया। अलग-अलग पाले गए समरूप जुड़वाँ, जिनकी
लंबाई में कई सेंटीमीटर और शरीर-द्रव्यमान में उससे भी अधिक का फ़र्क़ रहता
है। एक ही क्लोन का पौधा, जो समुद्र-तल पर और ऊँचाई पर उगाए जाने पर ऊँचाई
में दस गुना अलग होता है और प्रतिरोपण करने पर लौट आता है। हर हाल में
\omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} जमा हुआ है और \omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} हालात के साथ हिलता है।
\end{proof}

\begin{method}[जीन से लक्षण तक तर्क करना]\label{met:g11:enzymes-and-phenotype:reason}
\begin{enumerate}
  \item \omterm{def:g11:gene-expression:protein}{प्रोटीन} पहचानिए: कौन-सा \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम}, किस पथ में, और वह करता क्या है।
  \item \omterm{def:g11:gene-expression:protein}{प्रोटीन} पर उस \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} का असर पहचानिए: सामान्य, घटा हुआ,
        अनुपस्थित, या सक्रियता की शर्तों में बदला हुआ।
  \item नतीजे का पीछा पैमानों पर ऊपर तक कीजिए: \omterm{def:g10:cells-common-unit:cell}{कोशिका} में किसकी कमी है या
        क्या जमा हो रहा है, और फिर जीव क्या दिखाता है।
  \item पूछिए कि कितने काम करते \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} चाहिए: अधिकांश एंजाइमों के लिए एक
        ही काफ़ी है, इसलिए ख़राबी केवल दो बेकार विकल्पियों के साथ उभरती है
        (यानी कोई अप्रभावी लक्षण)।
  \item पूछिए कि पर्यावरण क्या जोड़ता है: \omterm{def:g11:enzymes-and-phenotype:enzyme}{क्रियाधार} मौजूद है या नहीं, ताप,
        कोई इलाज।
\end{enumerate}
\end{method}

\begin{remark}[एक काम करता युग्मविकल्पी आमतौर पर काफ़ी क्यों है]\label{rem:g11:enzymes-and-phenotype:recessive}
\omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} उत्प्रेरक हैं: थोड़ी-सी मात्रा भी बहुत सारा \omterm{def:g11:enzymes-and-phenotype:enzyme}{क्रियाधार} पलट देती है।
सामान्य से आधा टायरोसिनेज़ भी काफ़ी मेलानिन बना देता है, और सामान्य से आधा
लैक्टेज़ भी दूध का एक गिलास पचा देता है। इसलिए एक बेकार \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोता कोई
वाहक आमतौर पर बिना वाहक वाले से अलग पहचाना ही नहीं जा सकता --- यानी
प्राथमिक खंड का वही ``अप्रभावी'', अब समझाया हुआ --- और वह लक्षण तभी उभरता
है जब दोनों \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} चूक जाएँ। जो \omterm{def:g11:gene-expression:protein}{प्रोटीन} \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} नहीं हैं और जिन्हें मात्रा
में या ठीक-ठीक आकार में चाहिए, वे अक्सर अपवाद हैं: आधा-सामान्य हीमोग्लोबिन
फ़र्क़ डाल देता है।
\end{remark}

नहीं है; \qty{43}{\celsius} है। ताप वाले वक्र से समझाइए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:enzymes-and-phenotype:14}
समरूप जुड़वाँ का \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} एक ही होता है। उनमें से एक मैराथन धावक बन जाता
है, दूसरा गतिहीन। उनके बीच कोशिकीय या आणविक पैमाने पर अपेक्षित तीन
लक्षणप्ररूपी फ़र्क़ गिनाइए, और बताइए कि उनमें से हर एक ``आनुवंशिक'' लक्षणों
के बारे में क्या कहता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:enzymes-and-phenotype:15}
``चूँकि \omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} पर्यावरण पर निर्भर है, इसलिए \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} बहुत मायने नहीं
रखता।'' फेनिलकीटोनूरिया, वर्णकहीनता और स्याम की बिल्ली की मदद से चर्चा
कीजिए: हर हाल में \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} क्या तय कर देता है, और पर्यावरण क्या बदल सकता
है तथा क्या नहीं।
\end{exercise}

\section{समस्या: ठंडे कानों वाली बिल्ली}

\begin{problem}[{सप्ताहांत समस्या --- स्याम की बिल्ली के रोएँ एंजाइम-दर-एंजाइम
समझाए गए: वह ताप-संवेदी टायरोसिनेज़, शरीर के तापों का नक़्शा, वह मूँड़ा हुआ
टुकड़ा, और वह आहार जो किसी लक्षणप्ररूप को दोबारा लिख देता है}]
\label{pb:g11:enzymes-and-phenotype:1}
टायरोसिनेज़ \omterm{def:g10:universal-dna:gene}{जीन} का स्याम \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ऐसे \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} का कूट है जिसकी सक्रियता,
प्रयोगशाला में नापी गई, यह है: \qty{30}{\celsius} पर \qty{100}{\percent},
\qty{33}{\celsius} पर \qty{80}{\percent}, \qty{35}{\celsius} पर
\qty{40}{\percent}, \qty{37}{\celsius} पर \qty{5}{\percent}, और
\qty{39}{\celsius} पर \qty{0}{\percent}। साधारण \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} का \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम}
\qty{40}{\celsius} तक \qty{100}{\percent} बनाए रखता है। किसी बिल्ली की
त्वचा के ताप: धड़ \qty{38}{\celsius}, टाँगें \qty{36}{\celsius}, पंजे
\qty{33}{\celsius}, कान और थूथन \qty{31}{\celsius}। मेलानिन रोओं में तब
दिखता है जब \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} लगभग \qty{30}{\percent} से अधिक पर काम करे।

\medskip
\noindent\textbf{भाग I --- नक़्शा।}
\begin{enumerate}
  \item शरीर के हर क्षेत्र के लिए स्याम \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} की सक्रियता पढ़िए और
        बताइए कि वहाँ के रोएँ वर्णकयुक्त हैं या नहीं।
  \item बनने वाले रोओं का चित्र खींचिए, या उनका वर्णन कीजिए। क्या वह
        स्याम की बिल्ली से मेल खाता है?
  \item साधारण \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} वाली बिल्ली के लिए भी यही कीजिए।
  \item स्याम का कोई बिल्ली का बच्चा एक-सा मलाई रंग लिए जन्म लेता है। यह
        जानते हुए समझाइए कि गर्भाशय \qty{38}{\celsius} पर है।
  \item गहरे और फीके रोओं के बीच की सरहद त्वचा के किस ताप पर पड़ती है?
        \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} के वक्र का कौन-सा हिस्सा उसे तय करता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- प्रयोग।}
\begin{enumerate}[resume]
  \item धड़ पर एक टुकड़ा मूँड़कर रोएँ दोबारा उगते समय ठंडी थैली के नीचे
        \qty{30}{\celsius} पर रखा जाता है। उसके रंग का अनुमान लगाइए।
  \item पंजे पर एक टुकड़ा मूँड़कर पट्टी से \qty{38}{\celsius} पर गरम रखा
        जाता है। अनुमान लगाइए।
  \item थैलियाँ हटा देने के बाद दोबारा उगे रोएँ महीनों तक अपना रंग बनाए
        रखते हैं, फिर धीरे-धीरे बदल जाते हैं। दोनों तथ्य समझाइए।
  \item समझाइए कि ये प्रयोग यह क्यों दिखाते हैं कि पर्यावरण आणविक पैमाने
        पर असर डालता है, \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} पर नहीं।
  \item कोई विद्यार्थी सुझाता है कि गहरे सिरे उन \omterm{def:g10:cells-common-unit:cell}{कोशिकाओं} के किसी अलग
        \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} की वजह से हैं। इसके ख़िलाफ़ दो तर्क दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- तीनों पैमाने।}
\begin{enumerate}[resume]
  \item स्याम के \omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} का वर्णन आणविक, कोशिकीय और जीव के पैमाने पर
        कीजिए।
  \item स्याम \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} साधारण \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} से एक \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्ल} बदलने वाले अकेले एक
        प्रतिस्थापन से अलग है। समझाइए कि एक \omterm{def:g10:chemistry-of-life:families}{ऐमीनो अम्ल} किसी \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} को
        ताप-संवेदी कैसे बना सकता है।
  \item एक स्याम और एक साधारण \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} ढोती बिल्ली पूरी तरह वर्णकयुक्त
        होती है। \cref{rem:g11:enzymes-and-phenotype:recessive} की मदद से
        समझाइए।
  \item किसी वर्णकहीन बिल्ली में उसी \omterm{def:g10:universal-dna:gene}{जीन} के दो ऐसे \omterm{def:g10:universal-dna:gene}{युग्मविकल्पी} हैं जो किसी भी
        ताप पर काम करते \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} का कूट नहीं हैं। तीनों पैमानों पर उसके
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} की तुलना स्याम वाले से कीजिए।
  \item हिमालयी ख़रगोश और कुछ चूहे भी वही ढर्रा दिखाते हैं। इससे उनके
        टायरोसिनेज़ \omterm{def:g10:universal-dna:gene}{जीन} के बारे में क्या पता चलता है, और आप उस
        \omterm{def:g11:mutations:mutation}{उत्परिवर्तन} से क्या उम्मीद करेंगे?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- एक आहार जो \omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} दोबारा लिख देता है।}
किसी नवजात में फेनिलकीटोनूरिया पाया जाता है: रक्त का फेनिलऐलानिन \num{60}
की जगह \qty{1200}{\micro mol/L}। मस्तिष्क के विकास के लिए सुरक्षित स्तर
\qty{360}{\micro mol/L} से नीचे है।
\begin{enumerate}[resume]
  \item फेनिलऐलानिन कितने गुना चढ़ा हुआ है, और पथ वाले चित्र का कौन-सा
        \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} ग़ायब है?
  \item समझाइए कि ख़तरे में पड़ा अंग बच्चे का मस्तिष्क क्यों है, मसलन उसकी
        त्वचा क्यों नहीं।
  \item आहार फेनिलऐलानिन का सेवन सामान्य के पाँचवें हिस्से तक घटा देता है;
        और रक्त का स्तर भी उसी अनुपात में गिरता है। क्या बच्चा सुरक्षित
        है? \omterm{def:g11:enzymes-and-phenotype:enzyme}{एंजाइम} के अभाव में \omterm{def:g11:enzymes-and-phenotype:enzyme}{क्रियाधार} की आपूर्ति और \omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} के
        रिश्ते के बारे में इससे क्या पता चलता है?
  \item फिर भी थोड़ा फेनिलऐलानिन अपरिहार्य है, क्योंकि शरीर उसे बना नहीं
        सकता। समझाइए कि इसीलिए आहार को नियंत्रित करना पड़ता है, न कि
        फेनिलऐलानिन को बस हटा देना।
  \item परिणाम लिखिए: स्याम की बिल्ली के लिए और उस बच्चे के लिए,
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप} ने क्या तय किया, पर्यावरण ने क्या तय किया, और वह अकेला
        वाक्य जो \omterm{def:g11:enzymes-and-phenotype:phenotype}{जीनप्ररूप}, पर्यावरण तथा \omterm{def:g11:enzymes-and-phenotype:phenotype}{लक्षणप्ररूप} को जोड़ता है।
\end{enumerate}
\end{problem}
