\chapter{Hoe dieren ademhalen}\label{ch:g7:how-animals-breathe}

De forel onder de brug, de regenworm onder het gazon, de libel erboven en
jij erop doen op dit moment allemaal hetzelfde: \omterm{def:g7:what-is-respiration:respiration}{zuurstof} uit je \omterm{def:g6:exploring-our-environment:components}{omgeving}
opnemen (\cref{def:g7:what-is-respiration:respiration}). Maar jullie
\omterm{def:g6:exploring-our-environment:components}{omgevingen} verschillen --- lucht, bodem, water --- en de uitrusting dus
ook. Dit hoofdstuk is een rondgang langs de ademhalingsmachines van het
dierenrijk.

\section{Eén probleem, vier machines}

\begin{proposition}[Het gemeenschappelijke bestek]\label{prop:g7:how-animals-breathe:spec}
Elk ademhalingsorgaan is, wie de eigenaar ook is, volgens één bestek
gebouwd, dat we al in \cref{rem:g4:breathing:sponge} tegenkwamen: een
\emph{groot}, \emph{dun}, \emph{vochtig} oppervlak waar het \omterm{prop:g4:journey-of-food:absorb}{bloed} (of de
lichaamsvloeistof) van het \omterm{def:g1:animals-around-us:animal}{dier} dicht genoeg bij zijn \omterm{def:g6:exploring-our-environment:components}{omgeving} komt om
\omterm{def:g7:what-is-respiration:respiration}{zuurstof} naar binnen en \omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} naar buiten te laten oversteken.
Groot voor genoeg oversteek; dun voor een gemakkelijke oversteek; vochtig
omdat de gassen opgelost oversteken. De machines verschillen in hoe ze
dat oppervlak tegen de \omterm{def:g6:exploring-our-environment:components}{omgeving} aan houden.
\end{proposition}
\admitted

\begin{definition}[De vier machines]\label{def:g7:how-animals-breathe:four}
\omterm{def:g1:animals-around-us:animal}{Dieren} houden hun uitwisselingsoppervlak op vier hoofdmanieren vast:
\begin{enumerate}
  \item \emph{longen}\index{long}: met lucht gevulde zakjes die
        \emph{binnen} in het lichaam gevouwen zijn en door
        ademhalingsbewegingen geventileerd worden --- \omterm{prop:g3:sorting-living-things:groups}{zoogdieren}, \omterm{prop:g3:sorting-living-things:groups}{vogels},
        \omterm{prop:g4:classifying-animals:classes}{reptielen}, volwassen \omterm{prop:g3:sorting-living-things:groups}{amfibieën};
  \item \emph{kieuwen}\index{kieuw}: geveerde oppervlakken die
        \emph{buiten of aan de rand van het lichaam} gehouden worden en
        door een waterstroom gespoeld worden --- \omterm{prop:g3:sorting-living-things:groups}{vissen}
        (\cref{exo:g4:breathing:11}), \omterm{ex:g2:animals-grow-up:frog}{donderpadden}, veel waterlarven,
        mosselen;
  \item \emph{tracheeën}\index{trachee (insect)}: een netwerk van fijne
        luchtbuisjes dat vanuit openingen langs het lichaam rechtstreeks
        naar elk orgaan loopt --- \omterm{prop:g3:sorting-living-things:groups}{insecten}; de lucht wordt aan huis
        bezorgd, grotendeels zonder hulp van het \omterm{prop:g4:journey-of-food:absorb}{bloed};
  \item \emph{huidademhaling}\index{huidademhaling}: de hele vochtige
        \omterm{def:g1:the-five-senses:sense}{huid} als uitwisselingsoppervlak --- bij de regenworm helemaal;
        bij \omterm{prop:g3:sorting-living-things:groups}{amfibieën} als een grote aanvulling op hun eenvoudige longen.
\end{enumerate}
\end{definition}

\begin{example}[De machines aan het werk zien]\label{ex:g7:how-animals-breathe:watch}
Elke machine laat haar eigen ritme zien. De bek en de kieuwdeksels van de
forel pompen in gestage afwisseling --- water naar binnen bij de bek, over
de \omterm{def:g7:how-animals-breathe:four}{kieuwen}, naar buiten onder de deksels. Het achterlijf van de rustende
sprinkhaan klopt --- die pompt zijn luchtbuisjes. De keel van de kikker
trilt --- die duwt lucht zijn \omterm{def:g7:how-animals-breathe:four}{longen} in (hij heeft geen ribbenkast die dat
doet). Alleen de regenworm laat niets zien: \omterm{def:g7:how-animals-breathe:four}{huidademhaling} kent helemaal
geen bewegingen, en juist daarom onderscheidde
\cref{def:g7:what-is-respiration:respiration} de wisseling van het pompen.
\end{example}

\section{De machine past bij het medium --- en bij het leven}

\begin{proposition}[Waarom elke machine waar]\label{prop:g7:how-animals-breathe:fits}
De machines vallen samen met media en levenswijzen
(\cref{def:g4:adapted-to-their-world:adaptation} aan het werk op de
\omterm{def:g7:what-is-respiration:respiration}{ademhaling}):
\begin{itemize}
  \item \omterm{def:g7:how-animals-breathe:four}{kieuwen} passen bij water: hun geveerde oppervlakken drijven daar
        uit elkaar en worden onophoudelijk gespoeld --- en ze klappen in
        de lucht nutteloos in elkaar (de stikkende forel uit
        \cref{exo:g4:breathing:11});
  \item \omterm{def:g7:how-animals-breathe:four}{longen} passen bij lucht: naar binnen gevouwen blijven ze vochtig
        waar open water verloren zou gaan --- maar ze kunnen de schaarse
        opgeloste \omterm{def:g7:what-is-respiration:respiration}{zuurstof} van water niet oogsten;
  \item \omterm{def:g7:how-animals-breathe:four}{tracheeën} passen bij kleine lichamen: prachtig over millimeters,
        leveren ze over lange afstanden te traag --- een van de redenen
        dat \omterm{prop:g3:sorting-living-things:groups}{insecten} klein blijven;
  \item \omterm{def:g7:how-animals-breathe:four}{huidademhaling} past bij kleine levens met een vochtige \omterm{def:g1:the-five-senses:sense}{huid} op
        vochtige plekken: het hele oppervlak van de worm is amper genoeg
        voor zijn dunne lichaam, en droge lucht doodt hem --- denk aan de
        regenwormen die na de regen gestrand liggen.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[Twee levens, twee machines in één dier]\label{ex:g7:how-animals-breathe:frog}
De kikker doorloopt het hele hoofdstuk in één leven: \omterm{def:g7:how-animals-breathe:four}{kieuwen} als
donderpad (de machine van een waterbewoner,
\cref{ex:g2:animals-grow-up:frog}), \omterm{def:g7:how-animals-breathe:four}{longen} plus een vochtige \omterm{def:g1:the-five-senses:sense}{huid} als
volwassene --- met eenvoudige \omterm{def:g7:how-animals-breathe:four}{longen} en een \omterm{def:g1:the-five-senses:sense}{huid} die veel van het werk
doet, vooral tijdens de koude winter op de vijverbodem
(\cref{exo:g6:life-through-seasons:11}: die winterrustende kikker ademt
alleen door zijn \omterm{def:g1:the-five-senses:sense}{huid}). De \omterm{def:g2:animals-grow-up:metamorphosis}{gedaanteverwisseling} bouwt zelfs de
ademhalingsmachinerie om.
\end{example}

\begin{example}[Lucht ademen terwijl je in water leeft]\label{ex:g7:how-animals-breathe:airinwater}
Waterbewoners met \omterm{def:g7:how-animals-breathe:four}{longen} moeten pendelen. De dolfijn en de walvis komen
bovendrijven om te blazen en in te ademen
(\cref{ex:g6:classification-kinship:whale}: zoogdierlongen, geërfd van
landvoorouders); de poelslak klimt naar het oppervlak om bij te vullen;
de duikerkever draagt een zilveren luchtbel onder zijn dekschilden mee ---
een duikfles, aan het oppervlak bijgevuld. Over het \omterm{def:g2:where-animals-live:habitat}{leefgebied} valt te
onderhandelen; over het medium van de machine niet.
\end{example}

\begin{method}[De ademhaling van een dier aflezen]\label{met:g7:how-animals-breathe:read}
Bij een onbekend \omterm{def:g1:animals-around-us:animal}{dier}:
\begin{enumerate}
  \item waar leeft het --- en kan het weg? (water, lucht, bodem;
        pendelaar of vaste bewoner);
  \item zoek de tekenen van de machine: kieuwdeksels of geveerde
        pluimen; neusgaten en een borst die beweegt; rijen piepkleine
        openingen langs een insectenlichaam; of niets zichtbaars behalve
        een vochtige \omterm{def:g1:the-five-senses:sense}{huid};
  \item kijk naar het ritme: pompende deksels, kloppend achterlijf,
        rijzende borst, bezoekjes aan het oppervlak;
  \item noem de machine --- en toets haar aan het medium met
        \cref{prop:g7:how-animals-breathe:fits}.
\end{enumerate}
\end{method}

\begin{remark}[Het bestek verandert nooit]\label{rem:g7:how-animals-breathe:same}
Onder alle verscheidenheid: bekijk een van de vier machines van dichtbij
en \cref{prop:g7:how-animals-breathe:spec} duikt weer op. De \omterm{def:g1:animals-around-us:covering}{veren} van de
\omterm{def:g7:how-animals-breathe:four}{kieuw}, de zakjes van de \omterm{def:g7:how-animals-breathe:four}{long}, de fijnste uiteinden van de \omterm{def:g7:how-animals-breathe:four}{tracheeën} en de
\omterm{def:g1:the-five-senses:sense}{huid} van de worm zijn allemaal grote, dunne, vochtige ontmoetingsplaatsen
van \omterm{def:g6:exploring-our-environment:components}{omgeving} en lichaam. Vier machines, één bestek --- de \omterm{prop:g5:first-look-at-cells:principle}{eenheid van het
leven} (\cref{prop:g5:first-look-at-cells:principle}), geschreven in
ademhalingsuitrusting.
\end{remark}

\section{Probleem: De ademhalerstelling van de vijver}

\begin{problem}[{Weekendprobleem --- één vijver, elke machine aan het
werk}]\label{pb:g7:how-animals-breathe:1}
Een middag bij de vijver met schepnet, kijkdoos en notitieboekje. De
telling: jonge forellen en stekelbaarzen; \omterm{ex:g2:animals-grow-up:frog}{donderpadden} (sommige met
achterpoten); volwassen kikkers; poelslakken die langs de oppervlaktefilm
varen; een duikerkever die zilver flitst; haftlarven met geveerde pluimen
onder stenen; en, in het vochtige gras van de oever, regenwormen.

\medskip\noindent\textbf{Deel I --- De machines benoemen.}
\begin{enumerate}
  \item Wijs elk lid van de telling zijn ademhalingsmachine toe.
  \item De geveerde pluimen van de haftlarve wuiven onophoudelijk, ook in
        stilstaand water. Wat zijn ze, en waar dient dat wuiven voor?
  \item Welke leden van de telling moeten het oppervlak opzoeken, en
        waarom precies die?
  \item De \omterm{ex:g2:animals-grow-up:frog}{donderpadden} zonder poten en die met poten verschillen
        vanbinnen misschien net zo veel als vanbuiten. Welke verbouwing
        is er aan de gang, volgens
        \cref{ex:g7:how-animals-breathe:frog}?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De machines onder
druk.}
\begin{enumerate}[resume]
  \item Een hete, windstille week verlaagt de opgeloste \omterm{def:g7:what-is-respiration:respiration}{zuurstof} van de
        vijver (\cref{ch:g7:oxygen-in-the-water} zal uitleggen waarom).
        Zet de telling op volgorde: wie lijdt het eerst --- de
        kieuwgebruikers of de oppervlaktependelaars? Waarom?
  \item Die week hangen de stekelbaarzen vlak onder het oppervlak, met
        hun bek werkend aan de film. Leg dat gedrag uit met het bestek:
        wat is er bijzonder aan de bovenste millimeter van het water?
  \item Een onvoorzichtige waadvogel wervelt wolken fijn slib op. Welke
        machine verstopt, en wat zou je verwachten dat de getroffen
        \omterm{def:g1:animals-around-us:animal}{dieren} doen?
  \item Na een nacht regen liggen de regenwormen van het gazon op het pad.
        Verklaar dat klassieke tafereel: welke machine, en welke
        omstandigheid, dreef ze naar boven?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- De grotere kaart.}
\begin{enumerate}[resume]
  \item Zet de telling in een tabel: \omterm{def:g1:animals-around-us:animal}{dier}, machine, medium, pendelaar of
        vaste bewoner. Welk enkele bestek staat er volgens
        \cref{rem:g7:how-animals-breathe:same} boven elke machinekolom?
  \item De vijver vriest in januari dicht en het oppervlak wordt
        afgesloten. Voorspel de winterlotgevallen: de kikkers
        (\cref{exo:g6:life-through-seasons:11}), de slakken, de kever, de
        \omterm{prop:g3:sorting-living-things:groups}{vissen} --- wie zit in de problemen, en voor wie is gezorgd?
  \item Er komt een nieuwkomer bij: een waterspin die een zijden klok
        weeft en die met lucht van het oppervlak vult. Zet haar truc naast
        die van de kever, en noem de machine die ze blijft bedienen.
  \item Concludeer: waarom heeft één kleine vijver vier verschillende
        ademhalingsmachines nodig om volledig bewoond te zijn? Eén zin,
        met \emph{medium} en \emph{\omterm{def:g7:what-is-respiration:respiration}{zuurstof}} erin.
\end{enumerate}
\end{problem}
