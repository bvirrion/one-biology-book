\chapter{سكر الدم وداء السكري}\label{ch:g12:glucose-and-diabetes}

خمسة لترات من الدم، وغرام من الغلوكوز في كل لتر: أي خمسة غرامات جملةً،
نحو مكعب سكر. والدماغ وحده يحرق ذلك المقدار كل ساعة، ليلًا ونهارًا، ولا
يستطيع أن يستعمل غيره. ومع ذلك فمن لا يأكل شيئًا يومًا كاملًا، أو يأكل
كعكة كاملة دفعة واحدة، يبقي التركيز في حدود أعشار قليلة من الغرام في
اللتر من القيمة نفسها --- ومن أخفق بنكرياسه لا يفعل، والعواقب من العطش
إلى الغيبوبة. وهذا الفصل في أشد الأرقام مراقبةً في الجسم، وفي الأعضاء
التي تبقيه ثابتًا، وفيما يحدث، في صورتي \omterm{def:g12:glucose-and-diabetes:diabetes}{داء السكري}، حين تعجز عن ذلك.

\section{ثابت منظَّم}

\begin{definition}[سكر الدم]\label{def:g12:glucose-and-diabetes:glycaemia}
\emph{سكر الدم}\index{سكر الدم} هو تركيز الغلوكوز في بلازما الدم: نحو
\qty{1}{g/L} (\qty{5.5}{mmol/L}) قبل الوجبة عند شخص سليم، ويرتفع إلى
\qty{1.4}{g/L} في غضون ساعة من الأكل ويعود في غضون ساعتين، ويهبط إلى
\qty{0.7}{g/L} بعد يوم من الصيام. ودون \qty{0.5}{g/L} يخفق الدماغ
(تخليط، ثم غيبوبة)؛ وفوق \qty{1.8}{g/L} ينسكب الغلوكوز في البول،
وسنوات فوق \qty{1.3}{g/L} تتلف الأوعية والأعصاب. وسكر الدم مقدار
\emph{منظَّم}: أي قيمة تبقى قرب نقطة ضبط بنظام يقيسها ويصححها.
\end{definition}

\begin{proposition}[مخزن الغلوكوز وجريانه]\label{prop:g12:glucose-and-diabetes:pool}
\qty{5}{g} من الغلوكوز في الدم مخزن تمر عبره جريانات كبيرة.
\emph{الداخل}: المعي بعد الوجبة (حتى \qty{100}{g} في ساعة)، وبين
الوجبات \emph{الكبد}، الذي يحرر الغلوكوز من \emph{غليكوجينه} المخزون
(\qty{100}{g}، أي إمداد نصف يوم) ويصنع غلوكوزًا جديدًا من \omterm{def:g10:chemistry-of-life:families}{الحموض
الأمينية} واللاكتات. و\emph{الخارج}: الدماغ (\qty{5}{g/h}، ثابتًا)،
والعضلات (من غرامات قليلة إلى \qty{100}{g/h} أثناء العمل)، وسائر
الأنسجة، والادخار غليكوجينًا في الكبد والعضلات أو دسمًا. ومخزن قدره
\qty{5}{g} يغذّى ويستنزف بعشرات الغرامات في الساعة لا يستقر إلا لأن
الجريانات تعدَّل دقيقةً دقيقةً.
\end{proposition}
\admitted

\begin{example}[يوم من سكر الدم]\label{ex:g12:glucose-and-diabetes:day}
فطور في الثامنة: يرتفع \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} من \num{0.9} إلى \qty{1.3}{g/L} عند
التاسعة ويعود إلى \num{1.0} عند العاشرة، وقد ذهب الفائض إلى \omterm{def:g10:exercise-and-energy:glycogen}{غليكوجين}
الكبد والعضلات. ومن العاشرة إلى الواحدة يحرر الكبد \omterm{def:g10:exercise-and-energy:glycogen}{الغليكوجين} ويبقى
المستوى قرب \num{0.9}. وجري في الخامسة مساء يسحب \qty{60}{g} إلى
العضلات في ساعة؛ فيفرغ الكبد نصف مخزونه ليواكب، ويهبط المستوى إلى
\num{0.8}. وفي الليل يصنع الكبد غلوكوزًا جديدًا من \omterm{def:g10:chemistry-of-life:families}{الحموض الأمينية}،
وعند السابعة يكون المستوى \qty{0.9}{g/L}. فثلاثمئة غرام مرت عبر مخزن
من خمسة.
\end{example}

\section{البنكرياس يمسك الميزان}

\begin{proposition}[هرمونان من الجزر]\label{prop:g12:glucose-and-diabetes:hormones}
تتناثر في البنكرياس، بين \omterm{def:g10:cells-common-unit:cell}{الخلايا} التي تصنع \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيمات} الهاضمة، مليون
عنقود صغير من \omterm{def:g10:cells-common-unit:cell}{الخلايا}، هي \emph{الجزر}\index{جزر البنكرياس}. وخلاياها
تقيس \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} المار بها وتفرز فيه هرمونين:
\begin{itemize}
  \item \emph{الأنسولين}\index{أنسولين}، من \omterm{def:g10:cells-common-unit:cell}{الخلايا} $\beta$، حين
        يرتفع \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم}: فيجعل \omterm{def:g10:cells-common-unit:cell}{خلايا} العضلة والدهن تأخذ الغلوكوز، ويجعل
        الكبد والعضلات تدخره غليكوجينًا؛ فيهبط \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم}؛
  \item \emph{الغلوكاغون}\index{غلوكاغون}، من \omterm{def:g10:cells-common-unit:cell}{الخلايا} $\alpha$، حين
        يهبط: فيجعل الكبد يفكك \omterm{def:g10:exercise-and-energy:glycogen}{الغليكوجين} ويحرر الغلوكوز؛ فيرتفع \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر
        الدم}.
\end{itemize}
والهرمونان يعملان على الأعضاء نفسها في اتجاهين متعاكسين؛ وميزانهما،
المعاد ضبطه كل بضع دقائق \omterm{def:g12:glucose-and-diabetes:glycaemia}{بسكر الدم} المقيس، هو التنظيم. \omterm{prop:g12:glucose-and-diabetes:hormones}{والأنسولين} هو
\omterm{def:g11:hormones-and-reproduction:hormone}{الهرمون} الوحيد الذي يخفض \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم}؛ أما رافعوه فعدة.
\end{proposition}
\begin{proof}[شواهد]
نزع بنكرياس كلب (سنة 1889) يجعله سكريًا في غضون يوم: فيتضاعف سكر دمه
ثلاثًا ويظهر الغلوكوز في بوله؛ أما ربط القناة التي تحمل \omterm{def:g11:enzymes-and-phenotype:enzyme}{الإنزيمات}
الهاضمة فلا يفعل، فالأثر ليس هضميًا. وحقن مستخلص من \omterm{prop:g12:glucose-and-diabetes:hormones}{الجزر} (سنة 1921)
يخفض سكر دم كلب سكري، وسكر دم طفل سكري، في غضون ساعات؛ وقد نقيت المادة
الفعالة، وهي \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين}، في السنة نفسها. \omterm{prop:g12:glucose-and-diabetes:hormones}{والجزر} المعزولة في طبق تفرز
\omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} حين يرفع غلوكوز الوسط وتفرز \omterm{prop:g12:glucose-and-diabetes:hormones}{الغلوكاغون} حين يخفض. \omterm{prop:g12:glucose-and-diabetes:hormones}{والغلوكاغون}
المحقون يرفع سكر دم صائم في غضون دقائق، ولا يفعل إلا إذا كان الكبد
يحمل غليكوجينًا.
\end{proof}

\begin{example}[الحلقة في العمل]\label{ex:g12:glucose-and-diabetes:loop}
بعد الوجبة يتسلق \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم}؛ وفي غضون دقائق تحرر \omterm{def:g10:cells-common-unit:cell}{الخلايا} $\beta$
\omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين}، وتفتح \omterm{def:g10:cells-common-unit:cell}{خلايا} العضلة والدهن ناقلات الغلوكوز عندها، ويتحول
الكبد إلى صنع \omterm{def:g10:exercise-and-energy:glycogen}{الغليكوجين}، ويعود المستوى فيهبط --- وعندئذ يهبط إفراز
\omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} أيضًا. وبين الوجبات يقابل الانحدار نحو الأسفل \omterm{prop:g12:glucose-and-diabetes:hormones}{بالغلوكاغون}،
فيفكك \omterm{def:g10:exercise-and-energy:glycogen}{الغليكوجين}، ويثبت المستوى. وما من \omterm{def:g11:hormones-and-reproduction:hormone}{هرمون} منهما يغيب قط؛ وإنما
تتحرك نسبتهما، ويتبعها \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} في نطاق ضيق.
\end{example}

\section{حين تخفق الحلقة: داء السكري}

\begin{definition}[داء السكري]\label{def:g12:glucose-and-diabetes:diabetes}
\emph{داء السكري}\index{داء السكري} فائض مزمن في \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم}: فوق
\qty{1.26}{g/L} صائمًا، أو فوق \qty{2}{g/L} بعد ساعتين من حمل غلوكوز
قياسي. وعلاماته الباكرة تلي الفائض: غلوكوز في البول، يجر معه الماء
(بول غزير، وعطش)، ونقص وزن حين تعجز \omterm{def:g10:cells-common-unit:cell}{الخلايا} عن استعمال الغلوكوز
حولها. أما تلفه المتأخر --- في أوعية العين والكلية الصغيرة، وفي
الأعصاب، وفي الشرايين --- فآت من سنوات يتفاعل فيها الغلوكوز مع
\omterm{def:g10:chemistry-of-life:families}{البروتينات}. ويشترك في الاسم مرضان.
\end{definition}

\begin{proposition}[النمط 1 والنمط 2]\label{prop:g12:glucose-and-diabetes:types}
\begin{itemize}
  \item \emph{\omterm{def:g12:glucose-and-diabetes:diabetes}{داء السكري} من النمط 1} (نحو 10\% من الحالات) هو تخريب
        \omterm{def:g10:cells-common-unit:cell}{الخلايا} $\beta$ بجهاز مناعة المريض نفسه، في الطفولة أو الشباب
        عادة: \omterm{prop:g12:glucose-and-diabetes:hormones}{فالأنسولين} غائب. فيرتفع \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} بلا كابح، ويحرق الدسم
        لانعدام غلوكوز صالح للاستعمال وتتراكم نواتجه الحمضية؛ ومن غير
        أنسولين محقون يكون المرض مميتًا في غضون أشهر. والعلاج أنسولين،
        مرات عدة في اليوم، مجرَّعًا على الوجبات وعلى \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} المقيس،
        مدى العمر.
  \item \emph{\omterm{def:g12:glucose-and-diabetes:diabetes}{داء السكري} من النمط 2} (نحو 90\%) ينشأ عند البالغين،
        وأكثرهم زائدو الوزن قليلو الحركة: فتستجيب الأنسجة للأنسولين
        استجابة تقل شيئًا فشيئًا (\emph{مقاومة \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين}})، وتعوّض
        \omterm{def:g10:cells-common-unit:cell}{الخلايا} $\beta$ بإفراز المزيد، ثم تنهك بعد سنين. \omterm{prop:g12:glucose-and-diabetes:hormones}{فالأنسولين}
        موجود لكنه غير كاف. والاستعداد وراثي جزئيًا
        (\cref{ch:g11:genes-and-disease})؛ والزناد نمط العيش، وأول
        العلاج تغييره --- إنقاص الوزن والرياضة --- قبل الأدوية،
        وقبل \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} في وقت متأخر.
\end{itemize}
\end{proposition}
\begin{proof}[شواهد]
مرضى النمط 1 عندهم أجسام مضادة ضد \omterm{def:g10:cells-common-unit:cell}{خلايا} جزرهم نفسها قبل الأعراض بسنين، وعند
التشخيص تكاد جزرهم تخلو من \omterm{def:g10:cells-common-unit:cell}{الخلايا} $\beta$ وتكون مرتشحة \omterm{def:g10:cells-common-unit:cell}{بخلايا} بيض؛
وأنسولين دمهم غير قابل للكشف، وهم يستجيبون للأنسولين المحقون في الحال.
أما مرضى النمط 2 فأنسولينهم عند التشخيص عادي أو مرتفع؛ \omterm{def:g10:cells-common-unit:cell}{وخلايا} عضلاتهم
ودهنهم تأخذ غلوكوزًا أقل عند جرعة \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} نفسها مما تأخذ \omterm{def:g10:cells-common-unit:cell}{خلايا} شخص
سليم؛ والتوأمان المتطابقان يشتركان في المرض في 70\% من الحالات، وإنقاص
الوزن أو برنامج مشي وحده يعيد \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} إلى الطبيعي في كثير من الحالات
الباكرة.
\end{proof}

مراهقة عندها النمط 1 تقيس سكر دمها قبل كل وجبة بقطرة دم، وتحصي
السكريات في صحنها، وتحقن جرعة من \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} السريع تتناسب معها --- نحو
وحدة لكل \qty{10}{g} --- مع أنسولين بطيء مرة في اليوم للخلفية. فإن زاد
\omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين}، أو فاتت وجبة، هبط المستوى دون
\qty{0.6}{g/L}: تعرق، ورجفان، وتخليط، يصححه السكر في دقائق. وإن نقص،
بقي المستوى فوق \qty{2}{g/L} مع عطش وتعب، وتحمّلت الأوعية التلف في
صمت. فهي تفعل، بيدها، ما كانت خلاياها $\beta$ تفعله \omterm{def:g11:hormones-and-reproduction:hormone}{بالتغذية الراجعة}.
\end{example}

\begin{method}[قراءة سجل سكر دم]\label{met:g12:glucose-and-diabetes:read}
\begin{enumerate}
  \item عيّن نقطة الضبط والنطاق: صائمًا \qtyrange{0.7}{1.1}{g/L}، وبعد
        الوجبات دون \num{1.4}؛ وبعد ساعتين من حمل غلوكوز دون \num{1.4}
        (سليم)، و\qtyrange{1.4}{2}{} (مضطرب)، وفوق \qty{2}{g/L} (\omterm{def:g12:glucose-and-diabetes:diabetes}{داء
        السكري}).
  \item وعند كل ارتفاع، اسأل ما الذي أدخل الغلوكوز (وجبة، أو كبد) وما
        الذي ينبغي أن يخرجه (أهداف \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين})؛ وعند كل هبوط، ما الذي
        أخرجه (دماغ، أو عضلة) وما الذي ينبغي أن يرده (\omterm{prop:g12:glucose-and-diabetes:hormones}{الغلوكاغون}،
        والكبد).
  \item واقرأ الهرمونين مع الغلوكوز: فينبغي أن يرتفع \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} مع \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر
        الدم} وأن يهبط \omterm{prop:g12:glucose-and-diabetes:hormones}{الغلوكاغون}؛ وغياب \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} يعني النمط 1،
        وارتفاع \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} مع ارتفاع \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} يعني مقاومة.
  \item وحوّل الوحدات عند الحاجة: فالتركيز \qty{1}{g/L} من الغلوكوز
        هو \qty{5.55}{mmol/L}.
\end{enumerate}
\end{method}

\begin{remark}[نموذج في التنظيم]\label{rem:g12:glucose-and-diabetes:model}
مستقبل، ونقطة ضبط، ومنفذان متعاكسان، \omterm{def:g11:hormones-and-reproduction:hormone}{وتغذية راجعة} سلبية: فضبط \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم}
هو تصميم ضبط التستوستيرون نفسه في
\cref{ch:g11:hormones-and-reproduction}، وهو تصميم معظم ثوابت الجسم
--- الحرارة، والماء، وضغط الدم، والحموضة. \omterm{def:g12:glucose-and-diabetes:diabetes}{وداء السكري} هو ما يبدو عليه
ثابت منظَّم حين تنكسر حلقته: فالقيمة ما زالت تستجيب للداخل، لكن لا شيء
يعيدها. وقراءة منحنيات هذا الفصل تمرين على قراءة أي منها.
\end{remark}

\section{مسألة: خمسة غرامات من السكر}

\begin{problem}[{مسألة نهاية الأسبوع --- غلوكوز الدم يحاسب عليه على
مدى يوم: المخزن وجريانه، واختبار تحمل يقرأ، وجرعة أنسولين تحسب، وحلقة
لا بد أن تعوَّض باليد}]
\label{pb:g12:glucose-and-diabetes:1}
خذ حجم دم قدره \qty{5}{L}، واستهلاك دماغ قدره \qty{5}{g/h} من
الغلوكوز، ومخزون \omterm{def:g10:exercise-and-energy:glycogen}{غليكوجين} كبدي قدره \qty{100}{g}، وغليكوجينًا عضليًا
قدره \qty{400}{g}. والوحدة الواحدة من \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} السريع تتصرف بنحو
\qty{10}{g} من السكريات.

\medskip
\noindent\textbf{القسم الأول --- المخزن.}
\begin{enumerate}
  \item احسب كتلة الغلوكوز في الدم عند \qty{1}{g/L}، والتركيز نفسه
        بوحدة \unit{mmol/L} (الغلوكوز \qty{180}{g/mol}).
  \item كم كان غلوكوز الدم سيكفي الدماغ وحده، لو لم يعوَّض بشيء؟
  \item وبين الوجبات يعوّضه الكبد. فكم يكفي \omterm{def:g10:exercise-and-energy:glycogen}{غليكوجين} الكبد الدماغَ؟
        وماذا يحدث بعد ذلك؟
  \item \omterm{def:g10:exercise-and-energy:glycogen}{غليكوجين} العضلات أربعة أمثال \omterm{def:g10:exercise-and-energy:glycogen}{غليكوجين} الكبد، لكن العضلات لا
        تستطيع أن تحرر غلوكوزًا في الدم. فسّر لماذا لا يعين \omterm{def:g10:exercise-and-energy:glycogen}{غليكوجين}
        العضلات الدماغ، ولماذا هو موجود.
  \item وجبة تسلّم \qty{60}{g} من الغلوكوز على مدى ساعة. لو لم ينزع
        شيء منه، فماذا كان \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} سيبلغ؟ وما الذي ينزعه فعلًا، وإلى
        أين يذهب؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثاني --- الاختبار.}
مريض يشرب \qty{75}{g} من الغلوكوز. \omterm{def:g12:glucose-and-diabetes:glycaemia}{وسكر الدم} \omterm{prop:g12:glucose-and-diabetes:hormones}{والأنسولين} المقيسان:

\begin{center}
\begin{tabular}{lcccccc}
الدقائق & 0 & 30 & 60 & 90 & 120 & 180 \\ \hline
\omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} (g/L) & 1.15 & 1.8 & 2.05 & 2.0 & 1.85 & 1.6 \\
\omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} (\% من ذروة سليمة) & 90 & 140 & 180 & 200 & 190 & 160 \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item صنّف المريض بعتبات \cref{met:g12:glucose-and-diabetes:read}.
  \item أالأنسولين ناقص؟ فما الذي يخفق إذن؟
  \item سمّ المرض، وأعط سمتين من سيرة المريض تتوقعهما.
  \item كيف كان الجدول نفسه سيبدو عند مريض من النمط 1 بلا علاج؟
  \item ما أول علاج يوصف، وبأي آلية يعمل؟
\end{enumerate}

\medskip
\noindent\textbf{القسم الثالث --- إدارة الحلقة باليد.}
سكر دم مريضة من النمط 1 \qty{1.0}{g/L} قبل العشاء؛ والوجبة فيها
\qty{90}{g} من السكريات.
\begin{enumerate}[resume]
  \item احسب جرعة \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} السريع للوجبة.
  \item وهي تحقنها لكنها لا تأكل إلا نصف الوجبة. قدّر إلى أين قد يهبط
        \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم}، وسمّ الأعراض والعلاج.
  \item وهي لا تحقن شيئًا وتأكل الوجبة كلها. قدّر \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} لو دخلت
        \qty{90}{g} كلها مخزنًا قدره \qty{5}{L}، وقل ماذا تفعل
        الكليتان فوق \qty{1.8}{g/L}.
  \item ولماذا وجب أن تحقن أنسولينًا بطيئًا أيضًا في الليل، وهي لا
        تأكل شيئًا؟
  \item فسّر بأي معنى يعوّض المقياس والقلم \omterm{def:g10:cells-common-unit:cell}{الخلية} $\beta$، وما الذي لا
        يستطيعان تعويضه.
\end{enumerate}

\medskip
\noindent\textbf{القسم الرابع --- الحلقة، مستدلًا عليها.}
\begin{enumerate}[resume]
  \item دواء ينبه \omterm{def:g10:cells-common-unit:cell}{الخلايا} $\beta$ إلى إفراز مزيد من \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين}. فلأي
        نمط من \omterm{def:g12:glucose-and-diabetes:diabetes}{داء السكري} يستطيع أن ينفع، ولماذا لا ينفع للآخر؟
  \item شخص عنده النمط 2 يفقد \qty{10}{kg} ويمشي ساعة في اليوم؛ وبعد
        ستة أشهر يكون اختبار التحمل طبيعيًا. فسّر أي جزء من الحلقة
        أصلح.
  \item فسّر لماذا لا يكون \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين} \omterm{prop:g12:glucose-and-diabetes:hormones}{والغلوكاغون} مرتفعين معًا قط عند
        شخص سليم، وماذا تفعل جزيرة في طبق إذا ضبط غلوكوز الوسط عند
        \qty{1}{g/L}.
  \item \omterm{prop:g12:glucose-and-diabetes:hormones}{الغلوكاغون} \omterm{prop:g10:heart-lungs-effort:control}{والأدرينالين} والكورتيزول \omterm{def:g11:hormones-and-reproduction:hormone}{وهرمون} النمو ترفع \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم}
        كلها؛ ولا يخفضه إلا \omterm{prop:g12:glucose-and-diabetes:hormones}{الأنسولين}. اقترح لماذا كان للجسم عدة
        ضمانات ضد انخفاض \omterm{def:g12:glucose-and-diabetes:glycaemia}{سكر الدم} وضمانة واحدة ضد ارتفاعه.
  \item اذكر النتيجة: الغلوكوز الذي في الدم، والساعات التي يستطيع
        الكبد تغطيتها، وتشخيص القسم الثاني، وجرعة القسم الثالث.
\end{enumerate}
\end{problem}
