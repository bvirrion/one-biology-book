\chapter{Plant Structure}\label{ch:g7:plant-structure}

\omterm{def:g1:plants-animals:plant}{Plants} are \omterm{def:g6:uni-multi:multi}{multicellular organisms} adapted to capture \omterm{def:g4:plants-need:needs}{light}, take up \omterm{def:g4:plants-need:needs}{water}
and \omterm{def:g6:nutrition:groups}{minerals}, and move materials between \omterm{def:g7:tissues:plant-organs}{roots} and \omterm{def:g7:tissues:plant-organs}{leaves}. This chapter
surveys \omterm{def:g7:tissues:plant-organs}{roots}, \omterm{def:g7:tissues:plant-organs}{stems}, \omterm{def:g7:tissues:plant-organs}{leaves} and transport \omterm{def:g7:tissues:tissue}{tissues}.

\section{Organs of a flowering plant}

\begin{definition}[Root, stem, leaf]\label{def:g7:plant-structure:organs}
The \emph{root}\index{root} anchors the \omterm{def:g1:plants-animals:plant}{plant} and absorbs \omterm{def:g4:plants-need:needs}{water} and
dissolved \omterm{def:g6:nutrition:groups}{minerals}. The \emph{stem}\index{stem} supports \omterm{def:g7:tissues:plant-organs}{leaves} and
\omterm{def:g2:plant-parts:main}{flowers} and contains transport pathways. The \emph{leaf}\index{leaf} is
the main photosynthetic \omterm{def:g7:tissues:organ}{organ} in most \omterm{def:g1:plants-animals:plant}{plants}.
\end{definition}

\omimg{plant-parts.jpg}{0.52\linewidth}{External \omterm{def:g1:plants-animals:plant}{plant} parts: \omterm{def:g7:tissues:plant-organs}{roots},
\omterm{def:g7:plant-structure:organs}{stem}, \omterm{def:g7:tissues:plant-organs}{leaves} and reproductive structures.}

\begin{example}[Root hairs]\label{ex:g7:plant-structure:hairs}
Near \omterm{def:g7:plant-structure:organs}{root} tips, epidermal \omterm{def:g7:microscope:cell}{cells} may form long \emph{root
hairs}\index{root hair} that greatly increase surface area for absorption.
\end{example}

\section{Inside the leaf}

\begin{definition}[Leaf tissues]\label{def:g7:plant-structure:leaf}
A typical broad \omterm{def:g7:plant-structure:organs}{leaf} has:
\begin{itemize}
  \item upper and lower \emph{epidermis}\index{epidermis (plant)} (outer
        \omterm{def:g5:nerves:senses}{skin}), often with a waxy \emph{cuticle}\index{cuticle};
  \item \emph{palisade mesophyll}\index{mesophyll!palisade} --- packed
        with \omterm{def:g7:microscope:parts}{chloroplasts} for \omterm{def:g5:photosynthesis:def}{photosynthesis};
  \item \emph{spongy mesophyll}\index{mesophyll!spongy} --- \omterm{def:g4:plants-need:needs}{air} spaces for
        gas diffusion;
  \item \emph{stomata}\index{stoma} (singular stoma) --- pores, usually on
        the underside, each bordered by guard \omterm{def:g7:microscope:cell}{cells} that open and close.
\end{itemize}
\omterm{def:g5:circulation:heart}{Veins} contain \omterm{def:g7:plant-structure:vascular}{xylem} and \omterm{def:g7:plant-structure:vascular}{phloem}.
\end{definition}

\section{Transport: xylem and phloem}

\begin{definition}[Xylem and phloem]\label{def:g7:plant-structure:vascular}
\emph{Xylem}\index{xylem} \omterm{def:g7:tissues:tissue}{tissue} transports \omterm{def:g4:plants-need:needs}{water} and dissolved \omterm{def:g6:nutrition:groups}{minerals}
mainly upward from \omterm{def:g7:tissues:plant-organs}{roots} to \omterm{def:g7:tissues:plant-organs}{leaves}. \emph{Phloem}\index{phloem} \omterm{def:g7:tissues:tissue}{tissue}
transports sugars (and other organic compounds) from \omterm{def:g7:tissues:plant-organs}{leaves} to growing
and storage parts --- often called \emph{translocation}\index{translocation}.
\end{definition}

\begin{proposition}[Why transport matters]\label{prop:g7:plant-structure:why}
\omterm{def:g4:plants-need:parts}{Leaves} need \omterm{def:g4:plants-need:needs}{water} for \omterm{def:g5:photosynthesis:def}{photosynthesis} and to keep \omterm{def:g7:microscope:cell}{cells} firm (turgor).
\omterm{def:g4:plants-need:parts}{Roots} and growing tips need sugars made in \omterm{def:g7:tissues:plant-organs}{leaves}. Vascular \omterm{def:g7:tissues:tissue}{tissues} link
distant \omterm{def:g7:tissues:organ}{organs} into one working \omterm{def:g1:plants-animals:plant}{plant}.
\end{proposition}
\admitted

\begin{method}[Reading a stem cross-section]\label{met:g7:plant-structure:stem}
\begin{enumerate}
  \item Find the outer \omterm{def:g7:plant-structure:leaf}{epidermis} (and bark in woody \omterm{def:g7:tissues:plant-organs}{stems}).
  \item Locate vascular bundles: \omterm{def:g7:plant-structure:vascular}{xylem} usually toward the centre, \omterm{def:g7:plant-structure:vascular}{phloem}
        toward the outside (pattern depends on \omterm{def:g6:classification:species}{species}).
  \item Note ground \omterm{def:g7:tissues:tissue}{tissue} filling the rest (cortex, pith).
\end{enumerate}
\end{method}
