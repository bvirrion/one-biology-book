\chapter{Pola Makan yang Seimbang}\label{ch:g3:balanced-diet}

Sebuah lokasi pembangunan membutuhkan bata, semen, papan --- dan bahan
bakar bagi mesinnya. Tubuhmu adalah lokasi pembangunan yang tidak
pernah tutup: ia tumbuh, memperbaiki dirinya dan berlari-lari
sepanjang hari. Kirimannya datang tiga atau empat kali sehari, di atas
piring. Kirimannya harus memuat apa? Itulah pertanyaan tentang pola
makan.

\section{Mengapa tubuh membutuhkan makanan}

\begin{proposition}[Dua tugas makanan]\label{prop:g3:balanced-diet:jobs}
Makanan mengerjakan dua tugas yang berbeda:
\begin{enumerate}
  \item ia menyediakan \emph{energi}\index{energi dari makanan} ---
        tenaga untuk bergerak, untuk tetap hangat, untuk menjaga
        jantung dan otak tetap berjalan, bahkan selama \omterm{prop:g1:healthy-habits:needs}{tidur};
  \item ia menyediakan \emph{bahan bangunan} --- yang dipakai tubuh
        untuk bertambah tinggi, membangun \omterm{def:g3:bones-and-muscles:muscle}{otot} dan \omterm{def:g3:bones-and-muscles:skeleton}{tulang}, memperbaiki
        luka lecet dan mengganti bagian yang aus.
\end{enumerate}
Sebuah pola makan harus mengirimkan keduanya, setiap hari.
\end{proposition}
\admitted

\begin{example}[Bahan bakar bahkan saat beristirahat]\label{ex:g3:balanced-diet:rest}
Bahkan ketika berbaring diam sempurna, kamu membelanjakan \omterm{prop:g3:balanced-diet:jobs}{energi}:
jantung berdetak, dada naik lalu turun, tubuh tetap hangat pada
kira-kira $\qty{37}{\degreeCelsius}$ apa pun cuacanya. Beban tetap
itulah sebabnya melewatkan santapan membuatmu lemah dan kedinginan,
bukan sekadar lapar.
\end{example}

\section{Keluarga makanan}

\begin{definition}[Keluarga makanan]\label{def:g3:balanced-diet:families}
Makanan dikelompokkan ke dalam \emph{keluarga
makanan}\index{keluarga makanan} menurut apa yang dibawanya:
\begin{itemize}
  \item \emph{\omterm{prop:g3:flowers-fruits-seeds:transform}{buah} dan sayur} --- vitamin, serat dan air; keluarga
        kesegaran;
  \item \emph{serealia dan makanan bertepung} (roti, pasta, nasi,
        kentang) --- keluarga \omterm{prop:g3:balanced-diet:jobs}{energi} yang bertahan;
  \item \emph{produk \omterm{def:g2:animals-grow-up:born}{susu}} (\omterm{def:g2:animals-grow-up:born}{susu}, yogurt, keju) --- keluarga pembangun
        \omterm{def:g3:bones-and-muscles:skeleton}{tulang};
  \item \emph{daging, \omterm{prop:g3:sorting-living-things:groups}{ikan} dan \omterm{def:g2:animals-grow-up:born}{telur}} --- keluarga pembangun tubuh;
  \item \emph{lemak} (mentega, minyak) --- \omterm{prop:g3:balanced-diet:jobs}{energi} yang pekat,
        dibutuhkan dalam jumlah kecil;
  \item \emph{makanan manis} --- kesenangan yang cepat, sama sekali
        tidak dibutuhkan: keluarga kudapan;
  \item dan \emph{air}\index{air minum} --- satu-satunya minuman yang
        diperlukan tubuh, sepanjang hari.
\end{itemize}
\end{definition}

\begin{example}[Menyeimbangkan makan siang]\label{ex:g3:balanced-diet:lunch}
Pasta dengan saus tomat, sepotong \omterm{prop:g3:sorting-living-things:groups}{ikan}, buncis hijau, sebuah yogurt,
sebuah apel, air: setiap keluarga hadir, kudapan manis tidak ada ---
makan siang yang seimbang. Keripik, minuman bersoda dan sebatang
cokelat memberi makan tepat dua keluarga (lemak dan gula) dan
melewatkan keluarga yang ditunggu lokasi pembangunannya.
\end{example}

\begin{remark}[Tidak ada makanan terlarang, hanya yang terlupa]\label{rem:g3:balanced-diet:balance}
Keseimbangan diukur sepanjang beberapa hari, bukan pada satu gigitan.
Sepotong kue ulang tahun tidak melanggar aturan mana pun --- masalahnya
tidak pernah satu kudapan, melainkan kudapan \emph{sebagai ganti}
keluarganya, hari demi hari. Pertanyaan yang berguna pada tiap santapan
bukan ``apakah makanan ini buruk?'' melainkan ``keluarga mana yang
sudah kulupakan hari ini?''
\end{remark}

\section{Mencocokkan makanan dengan harinya}

\begin{proposition}[Kebutuhan berubah menurut hari dan orangnya]\label{prop:g3:balanced-diet:needs}
Berapa banyak \omterm{prop:g3:balanced-diet:jobs}{energi} yang dibutuhkan sebuah tubuh bergantung pada
siapa dia dan apa yang dikerjakannya: seorang anak yang sedang tumbuh
membutuhkan bahan bangunan yang secara nisbi lebih banyak daripada
orang dewasa; sehari berolahraga atau dinginnya musim dingin membakar
lebih banyak \omterm{prop:g3:balanced-diet:jobs}{energi} daripada sesore membaca. Rasa lapar tumbuh seiring
pengeluaran tubuh.
\end{proposition}
\admitted

\begin{example}[Mengapa sarapan berarti]\label{ex:g3:balanced-diet:breakfast}
Menjelang pagi, tubuh sudah berjalan semalaman --- jantung, napas,
kehangatan --- sama sekali tanpa kiriman. Sarapan mengisi ulang
persediaannya untuk pelajaran pagi dan berlari-lari. Lewatkan sarapan,
dan menjelang tengah pagi lokasinya kehabisan: perut berbunyi,
perhatian mengembara, tidak bertenaga ketika bermain.
\end{example}

\begin{method}[Memeriksa keseimbangan sehari]\label{met:g3:balanced-diet:check}
Pada akhir hari, telusuri daftarnya:
\begin{enumerate}
  \item apakah ada \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} atau sayur pada setiap santapan? (sasaran:
        kira-kira lima porsi dalam sehari)
  \item apakah ada makanan bertepung pada tiap santapan untuk \omterm{prop:g3:balanced-diet:jobs}{energi}
        yang bertahan?
  \item apakah pembangunnya hadir --- produk \omterm{def:g2:animals-grow-up:born}{susu} beberapa kali, dan
        daging, \omterm{prop:g3:sorting-living-things:groups}{ikan} atau \omterm{def:g2:animals-grow-up:born}{telur} sekali atau dua kali?
  \item apakah air menjadi minuman hari itu?
  \item kudapan manis: sedikit saja, bukan bagian utama?
\end{enumerate}
Keluarga mana pun yang hilang, undanglah ia besok.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:1}
Apa kedua tugas makanan? Sebutkan satu \omterm{def:g3:balanced-diet:families}{keluarga makanan} yang terutama
melayani tiap tugasnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:2}
Sebutkan keluarga-keluarga makanan, dan satu minuman yang diperlukan
tubuh.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:3}
Kelompokkan belanjaan berikut ke dalam keluarganya: nasi, yogurt,
wortel, \omterm{def:g2:animals-grow-up:born}{telur}, mentega, apel, roti.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:4}
Mengapa tubuh membelanjakan \omterm{prop:g3:balanced-diet:jobs}{energi} bahkan selama \omterm{prop:g1:healthy-habits:needs}{tidur}? Sebutkan dua
pengeluaran pada malam hari.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:5}
Periksa makan siang ``pasta, \omterm{prop:g3:sorting-living-things:groups}{ikan}, buncis hijau, yogurt, apel, air''
terhadap piring seimbang pada gambarnya: tiap benda itu termasuk
bagian piring yang mana?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:6}
Mengapa sarapan sangat penting? Apa yang terjadi menjelang tengah pagi
tanpa sarapan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:7}
Siapa yang secara nisbi membutuhkan lebih banyak bahan bangunan:
seorang anak atau orang dewasa? Mengapa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:balanced-diet:8}
Cobalah: jalankan \cref{met:g3:balanced-diet:check} atas harimu
sendiri, dengan jujur. Keluarga mana yang kamu lupakan --- dan mana
yang muncul melebihi bagiannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:balanced-diet:9}
``Permen itu terlarang!'' Apakah itu yang dikatakan bab ini? Berikan
aturan yang lebih tepat pada \cref{rem:g3:balanced-diet:balance}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:balanced-diet:10}
Ketika berlibur bermain ski, semua orang lebih lapar daripada di
rumah. Jelaskan dengan \cref{prop:g3:balanced-diet:needs} --- sebutkan
dua pengeluaran tambahan yang ditambahkan sehari yang dingin dan penuh
olahraga.
\end{exercise}
