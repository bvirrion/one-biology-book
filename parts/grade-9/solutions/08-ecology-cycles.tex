\section*{Chapter \ref{ch:g9:cycles} --- Matter Cycles in Ecosystems}

\begin{solution}{exo:g9:cycles:1}
\omterm{def:g2:food-energy:energy}{Energy} is lost as heat at each transfer and must be resupplied by the Sun;
elements are reused via decomposition and physical processes.
\end{solution}

\begin{solution}{exo:g9:cycles:2}
Evaporation, \omterm{def:g9:plant-phys:trans}{transpiration}, condensation, precipitation, runoff,
infiltration (any four).
\end{solution}

\begin{solution}{exo:g9:cycles:3}
Removes $\mathrm{CO_2}$ from \omterm{def:g4:plants-need:needs}{air} into organic molecules.
\end{solution}

\begin{solution}{exo:g9:cycles:4}
Respiration, combustion, decomposition (any two).
\end{solution}

\begin{solution}{exo:g9:cycles:5}

to build \omterm{def:g6:nutrition:groups}{proteins}/nucleic acids.
\end{solution}
