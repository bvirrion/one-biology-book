\chapter{المناعة التكيّفية والتلقيح}\label{ch:g12:adaptive-immunity}

في سنة 1796 خدش طبيب ريفي مادةً من بثرة جدري البقر عند حالبة إلى ذراع
صبي، وانتظر ستة أسابيع، ثم لقّحه بالجدري. فلم يمرض الصبي. ولم يكن في
عالم الطبيب ما يفسر ذلك؛ فقد بدا أن الجسم قد علّم. وبعد قرنين انقرض
الجدري، وفهم ذلك التعليم حتى الجزيئات: طقم من \omterm{def:g10:cells-common-unit:cell}{الخلايا} يحمل، فيما
بينها، مستقبلًا لأي جزيء يمكن أن يوجد تقريبًا، وتتكاثر حين يظهر
جزيئها، وتتذكر. ويصف هذا الفصل ذلك الجهاز، وذراعيه، وذاكرته، واللقاحات
المبنية عليه --- والفيروس الذي يخربه.

\section{المستضدات واللمفاويات}

\begin{definition}[المستضد واللمفاوية]\label{def:g12:adaptive-immunity:antigen}
\emph{المستضد}\index{مستضد} أي جزيء يستطيع \omterm{def:g12:innate-immunity:immunity}{الجهاز المناعي} التكيّفي أن
يتعرفه: وهو عادة \omterm{def:g11:gene-expression:protein}{بروتين} أو سكر عند جرثومة، أو ذيفان، أو سطح \omterm{def:g10:cells-common-unit:cell}{خلية}
غريبة --- أو، خطأً، جزيء من جزيئات الجسم نفسه. \omterm{def:g10:cells-common-unit:cell}{والخلايا} المتعرِّفة هي
\emph{اللمفاويات}\index{لمفاوية}، وهي \omterm{def:g10:cells-common-unit:cell}{خلايا} بيض تصنع في نقي العظم
وتتجمع في العقد اللمفية والطحال والنسيج اللمفاوي في المعي. وكل لمفاوية
تحمل آلاف النسخ من \emph{مستقبل} واحد، وكل مستقبل لا يرتبط إلا بمستضد
واحد. و\emph{اللمفاويات البائية} تنضج في النقي وستصنع \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة}؛
أما \emph{اللمفاويات التائية} فتنضج في التوتة وتعمل بالتماس
وبالإشارات.
\end{definition}

\begin{proposition}[ذخيرة تصنع سلفًا]\label{prop:g12:adaptive-immunity:repertoire}
مستقبل كل \omterm{def:g12:adaptive-immunity:antigen}{لمفاوية} يركَّب، أثناء نضج \omterm{def:g10:cells-common-unit:cell}{الخلية}، بإعادة ترتيب عشوائية لقطع
مورّثية، مختلفة في كل \omterm{def:g10:cells-common-unit:cell}{خلية}. فيحمل الجسم بذلك نحو $10^{8}$ مستقبل مختلف
قبل أن يلقى مستضدًا قط --- بما يكفي لأن يجد أي جزيء تقريبًا \omterm{def:g10:cells-common-unit:cell}{خلايا} قليلة
ترتبط به. \omterm{def:g12:adaptive-immunity:antigen}{واللمفاويات} التي يرتبط مستقبلها بجزيئات الجسم نفسه تزال
أثناء النضج؛ والباقي ينتظر، \omterm{def:g10:cells-common-unit:cell}{خلايا} قليلة لكل \omterm{def:g12:adaptive-immunity:antigen}{مستضد} ممكن، حتى يصل
مستضدها.
\end{proposition}
\admitted

\section{الانتقاء النسيلي}

\begin{proposition}[الانتقاء والتكاثر]\label{prop:g12:adaptive-immunity:selection}
حين يدخل \omterm{def:g12:adaptive-immunity:antigen}{مستضد}، لا تستجيب إلا \omterm{def:g12:adaptive-immunity:antigen}{اللمفاويات} التي يوافقه مستقبلها: فهي
\emph{تنتقى}، وتتكاثر \omterm{def:g11:cell-cycle-mitosis:cycle}{بالانقسام المتساوي} إلى نسيلة من آلاف \omterm{def:g10:cells-common-unit:cell}{الخلايا}
المتطابقة على مدى أيام عدة --- وهو \emph{الانتقاء
النسيلي}\index{انتقاء نسيلي}. ثم تتمايز النسيلة: فتصير معظم \omterm{def:g10:cells-common-unit:cell}{الخلايا}
\emph{\omterm{def:g10:cells-common-unit:cell}{خلايا} منفِّذة} تقاتل \omterm{def:g12:adaptive-immunity:antigen}{المستضد} أسبوعًا أو أسبوعين ثم تموت؛ وتصير
أقلية \emph{خلايا ذاكرة}\index{خلية ذاكرة} طويلة العمر، تبقى سنين،
وتستجيب، إن عاد \omterm{def:g12:adaptive-immunity:antigen}{المستضد}، أسرع وبأعداد أكبر بكثير. والاستجابة نوعية
لأن النسائل الموافقة وحدها هي التي تتكاثر؛ وهي بطيئة في المرة الأولى
لأنها تبدأ من حفنة \omterm{def:g10:cells-common-unit:cell}{خلايا}؛ وسريعة في الثانية لأن \omterm{prop:g12:adaptive-immunity:selection}{خلايا الذاكرة} كثيرة
أصلًا.
\end{proposition}
\begin{proof}[شواهد]
\omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} \omterm{def:g12:adaptive-immunity:antigen}{لمستضد} بعينه تظهر في الدم بعد نحو أسبوع من الحقنة
الأولى، وترتفع أسبوعين ثم تنحدر؛ وحقنة ثانية بعد أشهر تنتج، في غضون
يومين أو ثلاثة، أجسامًا مضادة بعشرة أمثال الذروة الأولى إلى مئة مثل،
وتبقى سنين. وحقن \omterm{def:g12:adaptive-immunity:antigen}{مستضد} ثان لا صلة له في الوقت نفسه مع الجرعة الثانية
يعطي، لذلك \omterm{def:g12:adaptive-immunity:antigen}{المستضد}، استجابة بطيئة من النمط الأول: فالذاكرة نوعية.
\omterm{def:g12:adaptive-immunity:antigen}{واللمفاويات} المأخوذة من حيوان ممنَّع والمنقولة إلى حيوان ساذج تنقل
الذاكرة؛ أما المأخوذة من حيوان غير ممنَّع فلا تنقلها.
\end{proof}

\section{ذراعان: الأجسام المضادة والخلايا القاتلة}

\begin{proposition}[اللمفاويات البائية والأجسام المضادة]\label{prop:g12:adaptive-immunity:antibodies}
النسيلة البائية المنتقاة تتمايز إلى \emph{\omterm{def:g10:cells-common-unit:cell}{خلايا} بلازمية}، وهي مصانع
تفرز آلاف \emph{الأجسام المضادة}\index{جسم مضاد} في الثانية: وهي نسخ
منحلة من مستقبل النسيلة، بروتينات على شكل حرف Y يرتبط كل من ذراعيها
\omterm{def:g12:adaptive-immunity:antigen}{بالمستضد}. \omterm{prop:g12:adaptive-immunity:antibodies}{والأجسام المضادة} لا تقتل؛ بل \emph{تعلّم}. فهي إذا ارتبطت
بفيروس أو بذيفان منعته من دخول \omterm{def:g10:cells-common-unit:cell}{الخلايا} أو من العمل؛ وإذا ارتبطت
بجرثومة كستها، والكسوة هي ما تمسكه بالعات \cref{ch:g12:innate-immunity}
خير مسك. والمستضدات التي تشابكها \omterm{prop:g12:adaptive-immunity:antibodies}{أجسام مضادة} كثيرة تتكتل في
\emph{معقدات مناعية}، تصفَّى \omterm{def:g12:innate-immunity:phagocytosis}{بالبلعمة}. وهذه هي الذراع في وجه الدخلاء في
الدم والسوائل.
\end{proposition}
\admitted

\begin{proposition}[اللمفاويات التائية]\label{prop:g12:adaptive-immunity:tcells}
\omterm{def:g12:adaptive-immunity:antigen}{اللمفاويات} التائية لا تتعرف \omterm{def:g12:adaptive-immunity:antigen}{المستضد} إلا قطعًا معروضة على سطح \omterm{def:g10:cells-common-unit:cell}{خلية} أخرى
(\cref{ch:g12:innate-immunity}). وهي صنفان:
\begin{itemize}
  \item \emph{الخلايا التائية المساعدة}\index{خلية تائية مساعدة} تنشّطها
        \omterm{def:g10:cells-common-unit:cell}{الخلايا} التغصنية التي تجلب \omterm{def:g12:adaptive-immunity:antigen}{المستضد} إلى العقدة اللمفية؛ وهي،
        بعد أن تنتقى وتتكاثر، تفرز سيتوكينات تأذن باستجابة \omterm{def:g12:adaptive-immunity:antigen}{اللمفاويات}
        البائية وسائر \omterm{def:g12:adaptive-immunity:antigen}{اللمفاويات} التائية وتضخّمها. فهي المنسّقة؛ ومن
        غيرها لا تكاد تقع استجابة تكيّفية.
  \item و\emph{\omterm{def:g10:cells-common-unit:cell}{الخلايا} التائية السامة \omterm{def:g10:cells-common-unit:cell}{للخلايا}} تتعرف قطعًا من فيروس (أو
        من \omterm{def:g11:gene-expression:protein}{بروتين} ورمي) تعرضها \omterm{def:g10:cells-common-unit:cell}{خلية} مصابة، وتقتل تلك \omterm{def:g10:cells-common-unit:cell}{الخلية} بالتماس،
        قبل أن يستطيع الفيروس الذي فيها أن يتكاثر. وهذه هي الذراع في
        وجه الدخلاء المختبئين داخل \omterm{def:g10:cells-common-unit:cell}{الخلايا}، حيث لا تبلغ \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام
        المضادة}.
\end{itemize}
\end{proposition}
\admitted

\begin{example}[فيروس يلقى مرتين]\label{ex:g12:adaptive-immunity:virus}
الأنفلونزا الأولى: يتكاثر الفيروس ثلاثة أيام قبل أن تجهز الاستجابة
التكيّفية؛ وبحلول اليوم السابع تكون \omterm{def:g10:cells-common-unit:cell}{الخلايا} التائية السامة تقتل \omterm{def:g10:cells-common-unit:cell}{الخلايا}
المصابة في المسالك الهوائية وتكون \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} تعادل الفيروس الحر؛
وبحلول اليوم العاشر يتعافى الشخص، ومعه \omterm{prop:g12:adaptive-immunity:selection}{خلايا ذاكرة} من الصنفين. والسلالة
نفسها بعد سنة: تسد \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} الموجودة أصلًا في الدم معظم الفيروس
عند الدخول، وتتكاثر \omterm{prop:g12:adaptive-immunity:selection}{خلايا الذاكرة} في غضون يومين، وتنتهي العدوى قبل أن
يلاحظها أحد. أما سلالة مختلفة طفرت بروتينات سطحها فتفلت من \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام
المضادة} --- ولهذا يعاد صنع \omterm{def:g12:adaptive-immunity:vaccine}{اللقاح} كل سنة.
\end{example}

\section{التلقيح}

\begin{definition}[اللقاح]\label{def:g12:adaptive-immunity:vaccine}
\emph{اللقاح}\index{لقاح} مستحضر يثير استجابة تكيّفية أولية، ومن ثم
ذاكرة، ضد ممرض من غير أن يسبب مرضه: أي الممرض موهَّنًا (الحصبة،
والسل)، أو مقتولًا (شلل الأطفال بالحقن)، أو مختصرًا إلى واحد من
بروتيناته أو إلى ذيفان معطَّل (الكزاز، والتهاب الكبد B)، أو، في الأحدث،
إلى تعليمات تصنع بها \omterm{def:g10:cells-common-unit:cell}{الخلية} ذلك \omterm{def:g11:gene-expression:protein}{البروتين}. و\emph{المساعد} المضاف إلى
المستحضر يثير الالتهاب الصغير الذي لولاه ما حملت \omterm{def:g10:cells-common-unit:cell}{الخلايا} التغصنية
\omterm{def:g12:adaptive-immunity:antigen}{المستضد} إلى العقدة اللمفية. والجرعات المعزِّزة تحوّل الاستجابة الأولية
إلى ثانوية وتمد الذاكرة عقودًا.
\end{definition}

\begin{proposition}[حماية الجماعة]\label{prop:g12:adaptive-immunity:herd}
الشخص الملقَّح محمي؛ \omterm{def:g12:selection-drift-speciation:population}{والجمهرة} الملقَّحة تحمي أفرادها غير الملقَّحين أيضًا.
فالشخص المصاب ينقل المرض، وسطيًا، إلى $R_0$ من غيره في \omterm{def:g12:selection-drift-speciation:population}{جمهرة} لا مناعة
لها (نحو 15 في الحصبة، و2 إلى 3 في الأنفلونزا)؛ فإذا كان كسر $v$
ممنَّعًا، لم يمكن أن يصاب إلا $R_0(1 - v)$ من تلك المخالطات، وانحسر
المرض حين يهبط ذلك العدد دون 1 --- أي حين يكون
\[
v > 1 - \frac{1}{R_0}.
\]
ففي الحصبة لا بد أن يكون 93\% من \omterm{def:g12:selection-drift-speciation:population}{الجمهرة} ممنَّعين لحماية الرضّع الأصغر
من أن يلقَّحوا ومن لا يستطيعون تلقيه أيضًا؛ وفي الأنفلونزا نحو 60\%.
وهذه \emph{مناعة القطيع}\index{مناعة القطيع} هي السبب في أن التلقيح
عمل جماعي.
\end{proposition}
\admitted

\section{حين يخرَّب المنسّق}

\begin{proposition}[HIV والإيدز]\label{prop:g12:adaptive-immunity:hiv}
فيروس العوز المناعي البشري، HIV، يصيب \omterm{prop:g12:adaptive-immunity:tcells}{الخلايا التائية المساعدة}،
داخلًا من المستقبل عينه الذي يميزها، ويتكاثر فيها. ويظل \omterm{def:g12:innate-immunity:immunity}{الجهاز
المناعي} سنين يقتل \omterm{def:g10:cells-common-unit:cell}{الخلايا} المصابة ويصنع أجسامًا مضادة --- والشخص
\emph{إيجابي المصل}، ومعدٍ --- بينما يفلت الفيروس، وهو يطفر باستمرار،
من كل استجابة ويستنزف \omterm{def:g10:cells-common-unit:cell}{الخلايا} المساعدة ببطء. فإذا هبط عددها دون نحو
خمس الطبيعي، لم يعد في الوسع تنسيق الاستجابة التكيّفية: وهو
\emph{الإيدز}\index{إيدز}، وفيه تصير عداوى يصفّيها الجسم السليم من غير
أن يلاحظها --- فطر في الفم، وطفيلي خفيف في الرئتين --- مميتة، وتظهر
سرطانات نادرة. والأدوية المضادة للفيروسات تسد \omterm{def:g11:enzymes-and-phenotype:enzyme}{إنزيمات} الفيروس، وتعيد
\omterm{def:g10:cells-common-unit:cell}{الخلايا} المساعدة، وتحجز المرض وتوقف العدوى إن أخذت مدى العمر؛ ولا \omterm{def:g12:adaptive-immunity:vaccine}{لقاح}
بعد، ولا شفاء.
\end{proposition}
\admitted

\begin{method}[قراءة استجابة مناعية]\label{met:g12:adaptive-immunity:read}
\begin{enumerate}
  \item عيّن \omterm{def:g12:adaptive-immunity:antigen}{المستضد} وأين هو: حرًا في السوائل (فتعمل \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة})
        أم داخل \omterm{def:g10:cells-common-unit:cell}{الخلايا} (فتعمل \omterm{def:g10:cells-common-unit:cell}{الخلايا} التائية السامة).
  \item واقرأ التوقيت: أسبوع إلى الذروة يعني استجابة أولية، وأيام
        قليلة تعني ذاكرة.
  \item واقرأ النوعية: فالاستجابة \omterm{def:g12:adaptive-immunity:antigen}{لمستضد} تترك الاستجابة لآخر على حالها.
  \item وحدد موضع \omterm{prop:g12:adaptive-immunity:tcells}{الخلايا التائية المساعدة}: فمن غيرها لا يحدث شيء، وهو
        ما يبرهنه \omterm{prop:g12:adaptive-immunity:hiv}{الإيدز}.
  \item وفي \omterm{def:g12:adaptive-immunity:vaccine}{اللقاح}، سمّ صورة \omterm{def:g12:adaptive-immunity:antigen}{المستضد}، والمساعد، وعدد الجرعات، وكسر
        \omterm{def:g12:selection-drift-speciation:population}{الجمهرة} اللازم.
\end{enumerate}
\end{method}

\begin{remark}[الانتقاء، مرة أخرى]\label{rem:g12:adaptive-immunity:selection}
\omterm{def:g12:innate-immunity:immunity}{الجهاز المناعي} التكيّفي داروينيّ: \omterm{def:g12:selection-drift-speciation:population}{جمهرة} من \omterm{def:g10:cells-common-unit:cell}{الخلايا} ذوات مستقبلات
عشوائية، تنتقي منها البيئة --- أي \omterm{def:g12:adaptive-immunity:antigen}{المستضد} --- قليلًا لتتكاثر، ويرث
نسلها المستقبل الرابح. فهو يدير مسار
\cref{ch:g12:selection-drift-speciation} داخل جسد واحد في أسبوع، ويحفظ
نتائجه، بالذاكرة، مدى العمر. وهو أيضًا سبب إمكان تعليم \omterm{def:g12:innate-immunity:immunity}{الجهاز المناعي}،
وهو ما يفعله \omterm{def:g12:adaptive-immunity:vaccine}{اللقاح}، وسبب صعوبة التغلب على فيروس يطفر أسرع مما يستطيع
الجهاز أن ينتقي --- مثل HIV والأنفلونزا.
\end{remark}

\section{مسألة: ثلاثة وتسعون في المئة}

\begin{problem}[{مسألة نهاية الأسبوع --- حملة تلقيح تحسب: الاستجابتان
لجرعتين، وحساب مناعة القطيع، وفاشية حصبة تنمذج، والفيروس الذي ينقض
الجهاز}]
\label{pb:g12:adaptive-immunity:1}
للحصبة $R_0 = 15$. وفي بلدة من 50\,000 نسمة 90\% من \omterm{def:g12:selection-drift-speciation:population}{الجمهرة} ممنَّعون
بالتلقيح أو بعدوى سابقة؛ و1500 منهم رضّع دون السنة، أصغر من أن
يلقَّحوا.

\medskip
\noindent\textbf{القسم الأول --- استجابة شخص واحد.}
يتلقى طفل الجرعة الأولى في الشهر الثاني عشر والثانية في الشهر الثامن عشر.
\begin{enumerate}
  \item صف منحني \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} بعد الجرعة الأولى: التأخر، والذروة،
        والانحدار.
  \item وصفه بعد الجرعة الثانية وفسّر الفرق \omterm{prop:g12:adaptive-immunity:selection}{بالانتقاء النسيلي}.
  \item فيروس \omterm{def:g12:adaptive-immunity:vaccine}{اللقاح} موهَّن لكنه حي. فلماذا يستعمل فيروس حي، وأي ذراع
        من الاستجابة يدربها ما كان الميت ليدربه؟
  \item وبعد عشر سنين تلقى الطفلة الحصبة. فماذا يحدث في الأيام الثلاثة
        الأولى، ولماذا لا تمرض؟
  \item وابن عمها غير الملقَّح يلقى الفيروس نفسه. صف أيامه العشرة
        الأولى، وما الذي يبقى عنده بعدها.
\end{enumerate}

\medskip
\noindent\textbf{القسم الثاني --- العتبة.}
\begin{enumerate}[resume]
  \item احسب كسر \omterm{def:g12:selection-drift-speciation:population}{الجمهرة} الذي لا بد أن يكون ممنَّعًا لتكف الحصبة عن
        الانتشار.
  \item ومع 90\% ممنَّعين، كم شخصًا يستطيع مصاب واحد أن يعدي وسطيًا؟
        أالبلدة محمية؟
  \item وكم شخصًا في البلدة غير ممنَّع، وكم منهم من الرضّع؟
  \item ولو ارتفعت التغطية إلى 95\%، فكم قابلًا للإصابة يبقى، وماذا
        كانت حالة واحدة ستنتج وسطيًا؟
  \item فسّر في جملة واحدة لماذا كانت سلامة الرضّع تتوقف على قرارات
        البالغين.
\end{enumerate}

\medskip
\noindent\textbf{القسم الثالث --- فاشية.}
مسافر يدخل الحصبة إلى البلدة عند مناعة 90\%.
\begin{enumerate}[resume]
  \item خذ كل حالة تعدي $15 \times 0.10 = 1.5$ من غيرها، في أجيال
        متعاقبة من 12 يومًا. فكم حالة في الجيل الثالث؟ وكم في السادس؟
  \item وكم حالة جملةً بعد ستة أجيال؟
  \item ومن الخمسة آلاف قابل للإصابة، 30\% رضّع. قدّر عدد الرضّع بين
        الحالات بعد ستة أجيال، إذا وقعت الحالات بالتساوي على القابلين
        للإصابة.
  \item كل جيل من الحالات يترك ناجين ممنَّعين: فسّر لماذا تبطئ الفاشية
        من تلقاء نفسها، ومتى تقريبًا.
  \item أعد السؤال 11 لبلدة عند مناعة 95\%. وقارن البلدتين.
\end{enumerate}

\medskip
\noindent\textbf{القسم الرابع --- الجهاز منقوضًا.}
مريض مصاب بفيروس HIV، بلا علاج، يفقد \omterm{def:g10:cells-common-unit:cell}{خلايا} تائية مساعدة من 1000 إلى 200 في
الميكرولتر في ثماني سنين.
\begin{enumerate}[resume]
  \item احسب متوسط الفقد في السنة، وفي الشهر.
  \item وعند 200 في الميكرولتر يبدأ \omterm{prop:g12:adaptive-immunity:hiv}{الإيدز}. فسّر، بشكل التنظيم، لماذا
        تتأثر \omterm{prop:g12:adaptive-immunity:antibodies}{الأجسام المضادة} \omterm{def:g10:cells-common-unit:cell}{والخلايا} القاتلة معًا مع أن \omterm{def:g10:cells-common-unit:cell}{الخلايا}
        المساعدة وحدها هي المصابة.
  \item \omterm{prop:g12:adaptive-immunity:selection}{وخلايا ذاكرة} الحصبة عند المريض سليمة. أفتكون مخالطة الحصبة
        الآن خطرة؟ فسّر.
  \item والعلاج المضاد للفيروسات إذا بدئ في السنة الثامنة يرفع العدد
        إلى 600 في غضون سنتين. فماذا يبين ذلك عن \omterm{prop:g12:adaptive-immunity:selection}{خلايا الذاكرة} وعن
        الذخيرة، ولماذا وجب أن يستمر العلاج؟
  \item اذكر النتيجة: التغطية التي تحمي رضّع البلدة، وعدد الحالات التي
        يبذرها مسافر واحد في ستة أجيال عند 90\%، \omterm{def:g10:cells-common-unit:cell}{والخلية} الواحدة التي
        يهدم فقدها الجهاز التكيّفي كله.
\end{enumerate}
\end{problem}
