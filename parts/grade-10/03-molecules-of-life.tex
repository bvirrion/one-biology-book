\chapter{Molecules of Life}\label{ch:g10:molecules-of-life}

\omterm{def:g10:the-cell:cell}{Cells} are chemical machines. A handful of element types --- mainly
carbon, hydrogen, oxygen, nitrogen, phosphorus and sulfur --- build a
vast library of molecules. This chapter surveys water and the four great
classes of biological macromolecules: \omterm{def:g10:molecules-of-life:carbs}{carbohydrates}, \omterm{def:g10:molecules-of-life:lipids}{lipids}, \omterm{def:g10:molecules-of-life:proteins}{proteins}
and \omterm{def:g10:molecules-of-life:nucleic}{nucleic acids} --- structure first, then role.

\section{Water: the solvent of life}

\begin{definition}[Polarity of water]\label{def:g10:molecules-of-life:water}
A water molecule ($\mathrm{H_2O}$) is \emph{polar}\index{polar molecule}: the
oxygen atom pulls shared electrons more strongly than hydrogen, so oxygen
carries a partial negative charge and each hydrogen a partial positive
charge. Neighbouring water molecules form \emph{hydrogen
bonds}\index{hydrogen bond} --- weak attractions that constantly break
and reform.
\end{definition}

\begin{proposition}[Properties that matter for life]\label{prop:g10:molecules-of-life:water-props}
Hydrogen bonding helps explain why water: is an excellent solvent for
ions and \omterm{def:g10:molecules-of-life:water}{polar molecules}; resists temperature change (high specific heat);
cools surfaces when it evaporates; and forms a less dense solid (ice
floats), protecting aquatic life in winter.
\end{proposition}
\admitted

\begin{definition}[Hydrophilic and hydrophobic]\label{def:g10:molecules-of-life:phil-phob}
\emph{Hydrophilic}\index{hydrophilic} substances interact readily with
water (ions, sugars, many \omterm{def:g10:molecules-of-life:proteins}{proteins}). \emph{Hydrophobic}\index{hydrophobic}
substances do not (oils, \omterm{def:g10:molecules-of-life:lipids}{fats}); they cluster away from water --- a key
driver of membrane structure (\cref{ch:g10:membranes-enzymes}).
\end{definition}

\section{Building macromolecules}

\begin{definition}[Monomer and polymer]\label{def:g10:molecules-of-life:polymer}
A \emph{monomer}\index{monomer} is a small molecular building block. A
\emph{polymer}\index{polymer} is a long chain of monomers.
\emph{Dehydration synthesis}\index{dehydration synthesis} (condensation)
joins monomers by removing water; \emph{hydrolysis}\index{hydrolysis}
splits polymers by adding water.
\end{definition}

\section{Carbohydrates}

\begin{definition}[Carbohydrates]\label{def:g10:molecules-of-life:carbs}
\emph{Carbohydrates}\index{carbohydrate} are sugars and their \omterm{def:g10:molecules-of-life:polymer}{polymers},
built from C, H and O (roughly $\mathrm{(CH_2O)_n}$).
\emph{Monosaccharides}\index{monosaccharide} (e.g.\ glucose, fructose,
ribose) are single sugar units. \emph{Disaccharides}\index{disaccharide}
(e.g.\ sucrose, lactose) join two. \emph{Polysaccharides}\index{polysaccharide}
are long chains: starch and glycogen store glucose; cellulose builds
plant \omterm{def:g10:the-cell:support}{cell walls}; chitin builds fungal walls and arthropod exoskeletons.
\end{definition}

\begin{example}[Glucose as fuel and brick]\label{ex:g10:molecules-of-life:glucose}
Glucose ($\mathrm{C_6H_{12}O_6}$) is a central energy source in cellular respiration
and the \omterm{def:g10:molecules-of-life:polymer}{monomer} of starch, glycogen and cellulose. Same \omterm{def:g10:molecules-of-life:polymer}{monomer},
different linkages and branching --- different properties (digestible
store versus tough fibre).
\end{example}

\begin{proposition}[Storage versus structure]\label{prop:g10:molecules-of-life:poly-roles}
Starch (plants) and glycogen (animals) are compact, enzyme-accessible
glucose stores. Cellulose's geometry makes straight, hydrogen-bonded
fibres that few animals can digest without microbial help --- structure
over snack.
\end{proposition}
\admitted

\section{Lipids}

\begin{definition}[Lipids]\label{def:g10:molecules-of-life:lipids}
\emph{Lipids}\index{lipid} are a diverse group of \omterm{def:g10:molecules-of-life:phil-phob}{hydrophobic} molecules,
not true \omterm{def:g10:molecules-of-life:polymer}{polymers}. Major types:
\begin{enumerate}
  \item \emph{Fats and oils}\index{fat}\index{triglyceride}
        (triglycerides): glycerol plus three fatty acids --- energy
        storage, insulation;
  \item \emph{Phospholipids}\index{phospholipid}: glycerol, two fatty
        acids and a \omterm{def:g10:molecules-of-life:water}{polar} phosphate head --- bilayer membranes;
  \item \emph{Steroids}\index{steroid}: four fused carbon rings
        (e.g.\ cholesterol, some hormones).
\end{enumerate}
\end{definition}

\begin{remark}[Saturated and unsaturated]\label{rem:g10:molecules-of-life:sat}
Fatty acids with only single C--C bonds are \emph{saturated}\index{saturated
fatty acid}; kinks from double bonds make \emph{unsaturated}\index{unsaturated
fatty acid} chains. Kinks lower packing density and melting point ---
oils versus solid \omterm{def:g10:molecules-of-life:lipids}{fats}.
\end{remark}

\section{Proteins}

\begin{definition}[Proteins and amino acids]\label{def:g10:molecules-of-life:proteins}
\emph{Proteins}\index{protein} are \omterm{def:g10:molecules-of-life:polymer}{polymers} of \emph{amino
acids}\index{amino acid} linked by \emph{peptide bonds}\index{peptide
bond}. Twenty common amino acids differ in their side chains (R groups).
A protein's \emph{sequence}\index{primary structure} folds into a unique
three-dimensional shape that determines function.
\end{definition}

\begin{definition}[Levels of protein structure]\label{def:g10:molecules-of-life:protein-levels}
\emph{Primary}: amino-acid order. \emph{Secondary}: local folds
($\alpha$-helix, $\beta$-sheet) stabilised by \omterm{def:g10:molecules-of-life:water}{hydrogen bonds}.
\emph{Tertiary}: overall 3-D shape of one chain. \emph{Quaternary}:
assembly of multiple chains (e.g.\ haemoglobin).
\end{definition}

\begin{proposition}[Shape is function]\label{prop:g10:molecules-of-life:shape}
Enzymes, antibodies, transporters, hormones and structural fibres are
\omterm{def:g10:molecules-of-life:proteins}{proteins} whose jobs require a precise shape. \emph{Denaturation}\index{denaturation}
(by heat, pH extremes or some chemicals) unfolds the \omterm{def:g10:molecules-of-life:proteins}{protein} and destroys
activity --- often irreversibly.
\end{proposition}
\admitted

\begin{example}[Roles]\label{ex:g10:molecules-of-life:protein-roles}
Catalysis (enzymes), transport (haemoglobin), movement (actin/myosin),
defence (antibodies), signalling (insulin), structure (collagen,
keratin). One chemistry, many careers.
\end{example}

\section{Nucleic acids}

\begin{definition}[Nucleic acids]\label{def:g10:molecules-of-life:nucleic}
\emph{Nucleic acids}\index{nucleic acid} store and transmit hereditary
information. \emph{DNA}\index{DNA} (deoxyribonucleic acid) is a double
helix of two complementary strands; \emph{RNA}\index{RNA} (ribonucleic
acid) is usually single-stranded and helps express genes. \omterm{def:g10:molecules-of-life:polymer}{Monomers} are
\emph{nucleotides}\index{nucleotide}: sugar + phosphate + nitrogenous
base.
\end{definition}

\begin{definition}[Base pairing]\label{def:g10:molecules-of-life:base-pair}
In \omterm{def:g10:molecules-of-life:nucleic}{DNA}, adenine (A) pairs with thymine (T), and cytosine (C) with
guanine (G), via \omterm{def:g10:molecules-of-life:water}{hydrogen bonds}. \omterm{def:g10:molecules-of-life:nucleic}{RNA} uses uracil (U) in place of thymine.
Complementarity allows accurate copying and information transfer.
\end{definition}

\begin{proposition}[Information flow (overview)]\label{prop:g10:molecules-of-life:info}
\omterm{def:g10:molecules-of-life:nucleic}{DNA} stores instructions; segments called genes are transcribed to \omterm{def:g10:molecules-of-life:nucleic}{RNA};
\omterm{def:g10:molecules-of-life:nucleic}{RNA} guides assembly of \omterm{def:g10:molecules-of-life:proteins}{proteins}. \omterm{def:g10:molecules-of-life:proteins}{Proteins} then build and run the \omterm{def:g10:the-cell:cell}{cell}.
Details occupy later genetics chapters.
\end{proposition}
\admitted

\section{Problem: From molecule to meal}

\begin{problem}[{Weekend problem --- decode a lunch tray at the molecular
level: identify macromolecules, predict digestion products and link each
class to a cell function}]
\label{pb:g10:molecules-of-life:1}
A lunch tray holds: wholegrain bread, olive oil on salad, grilled fish,
fresh fruit and a glass of milk.

\medskip
\noindent\textbf{Part I --- Inventory.}
\begin{enumerate}
  \item For each food, list the dominant macromolecule classes present.
  \item Identify one example of a storage \omterm{def:g10:molecules-of-life:carbs}{polysaccharide} and one of a
        structural \omterm{def:g10:molecules-of-life:carbs}{polysaccharide} you might meet in this meal or in
        related plant material (e.g.\ fruit peel cellulose).
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Digestion chemistry.}
\begin{enumerate}
  \item Name the \omterm{def:g10:molecules-of-life:polymer}{monomer} end-products of digesting starch, \omterm{def:g10:molecules-of-life:proteins}{protein} and
        \omterm{def:g10:molecules-of-life:lipids}{triglyceride} \omterm{def:g10:molecules-of-life:lipids}{fat}.
  \item State whether each process is mainly \omterm{def:g10:molecules-of-life:polymer}{hydrolysis} or dehydration.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Cellular destiny.}
\begin{enumerate}
  \item Match \omterm{def:g10:molecules-of-life:polymer}{monomers}/products to uses: build new \omterm{def:g10:molecules-of-life:proteins}{protein}; fuel
        respiration; store as \omterm{def:g10:molecules-of-life:lipids}{fat}; contribute to membranes.
  \item Explain why the fish's \omterm{def:g10:molecules-of-life:nucleic}{DNA}, though present, is not used as a
        human gene after you eat it.
\end{enumerate}
\end{problem}
