\chapter{Enzymen en het fenotype}\label{ch:g11:enzymes-and-phenotype}

Een siamese kat is over het grootste deel van haar lichaam roomkleurig en
donkerbruin op oren, snuit, poten en staart --- de koude delen. Scheer een
plek op haar rug kaal, houd die plek onder een koelelement koel terwijl de
vacht teruggroeit, en de nieuwe haren komen donker op. De \omterm{def:g10:universal-dna:gene}{genen} van de kat
zijn niet veranderd; haar pigmentmakende \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} werkt eenvoudigweg beneden
\qty{35}{\celsius} en niet erboven. Tussen een \omterm{def:g10:universal-dna:gene}{gen} en een zichtbaar
kenmerk staat vrijwel altijd een \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym}; en tussen het \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} en het kenmerk
staat de omgeving waarin het werkt. Dit hoofdstuk gaat over die twee
tussenschakels.

\section{Het fenotype op drie schalen}

\begin{definition}[Genotype en fenotype]\label{def:g11:enzymes-and-phenotype:phenotype}
Het \emph{genotype}\index{genotype} van een individu is de verzameling
\omterm{def:g10:universal-dna:gene}{allelen} die het draagt; zijn \emph{fenotype}\index{fenotype} is de
verzameling van zijn waarneembare eigenschappen. Het fenotype kan op drie
schalen worden beschreven:
\begin{itemize}
  \item de \emph{moleculaire} schaal: welke \omterm{def:g10:chemistry-of-life:families}{eiwitten} aanwezig zijn, in
        welke vorm, met welke werkzaamheid;
  \item de \emph{cellulaire} schaal: de vorm, de inhoud en het gedrag van
        de \omterm{def:g10:cells-common-unit:cell}{cellen};
  \item de schaal van het \emph{organisme}: de zichtbare kenmerken, van
        oogkleur tot de verschijnselen van een ziekte.
\end{itemize}
Elke schaal is het gevolg van de schaal eronder.
\end{definition}

\begin{example}[Sikkelcelziekte op drie schalen]\label{ex:g11:enzymes-and-phenotype:sickle}
Moleculair: één \omterm{def:g10:chemistry-of-life:families}{aminozuur} van de keten van hemoglobine (glutamaat, het zesde)
is door valine vervangen; de veranderde moleculen plakken bij weinig
zuurstof aan elkaar tot lange vezels. Cellulair: die vezels vervormen de
rode bloedcel tot een stijve sikkel die kleine vaten verstopt en
gemakkelijk breekt. Organisme: bloedarmoede, pijnaanvallen wanneer vaten
verstopt raken, schade aan milt en nieren. Eén substitutie in één \omterm{def:g11:gene-expression:code}{codon}
van één \omterm{def:g10:universal-dna:gene}{gen}, en drie beschrijvingen van het gevolg ervan.
\end{example}

\begin{proposition}[Eiwitten maken het fenotype]\label{prop:g11:enzymes-and-phenotype:proteins}
Elk niveau van het \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} wordt door \omterm{def:g10:chemistry-of-life:families}{eiwitten} voortgebracht: een \omterm{def:g10:universal-dna:gene}{allel}
bepaalt een \omterm{def:g11:gene-expression:protein}{eiwit} (\cref{ch:g11:gene-expression}); de werkzaamheid van
het \omterm{def:g11:gene-expression:protein}{eiwit} bepaalt de chemie en de bouw van de \omterm{def:g10:cells-common-unit:cell}{cel}; de \omterm{def:g10:cells-common-unit:cell}{cellen} bepalen de
kenmerken van het organisme. De meeste van die \omterm{def:g10:chemistry-of-life:families}{eiwitten} zijn
\emph{\omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymen}}, en de meeste kenmerken zijn dus de uitkomst van ketens
van reacties die door \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymen} worden uitgevoerd.
\end{proposition}
\admitted

\section{Enzymen}

\begin{definition}[Enzym, substraat, actieve plaats]\label{def:g11:enzymes-and-phenotype:enzyme}
Een \emph{enzym}\index{enzym} is een \omterm{def:g11:gene-expression:protein}{eiwit} dat één chemische reactie
katalyseert: het laat de reactie miljoenen keren sneller verlopen zonder
zelf te worden verbruikt. Het molecuul dat het omzet is zijn
\emph{substraat}\index{substraat}; de reactie vindt plaats in een holte
van het enzym, de \emph{actieve plaats}\index{actieve plaats}, waarvan de
vorm en de chemie bij dat ene substraat passen en bij geen ander. Die
\emph{specificiteit} volgt uit de driedimensionale vouwing van het enzym,
dus uit zijn aminozuurvolgorde, dus uit zijn \omterm{def:g10:universal-dna:gene}{gen}.
\end{definition}

\begin{example}[Lactase]\label{ex:g11:enzymes-and-phenotype:lactase}
Melksuiker, lactose, wordt door het \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} lactase in de bekleding van de
dunne darm in twee enkelvoudige suikers gesplitst; zonder die splitsing
gaat zij onverteerd door naar de dikke darm, waar bacteriën haar
vergisten, met gas, krampen en diarree als gevolg. Alle zuigelingen maken
lactase; bij het grootste deel van de mensheid wordt het \omterm{def:g10:universal-dna:gene}{gen} ervan na het
spenen uitgeschakeld, en zijn volwassenen lactose-intolerant. Bij
bevolkingen met een lange geschiedenis van veehouderij komt een \omterm{def:g10:universal-dna:gene}{allel}
veel voor dat het \omterm{def:g10:universal-dna:gene}{gen} levenslang actief houdt --- 90\% in Noord-Europa,
10\% in Oost-Azië. De lactasedruppels die aan intoleranten worden
verkocht, bevatten hetzelfde \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym}, gemaakt door schimmels.
\end{example}

\begin{proposition}[Voorwaarden voor werkzaamheid]\label{prop:g11:enzymes-and-phenotype:conditions}
De werkzaamheid van een \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} hangt van zijn omgeving af. Zij stijgt met
de temperatuur tot het \omterm{def:g11:gene-expression:protein}{eiwit} zich begint te ontvouwen, en stort dan in
--- menselijke \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymen} werken het best rond \qty{37}{\celsius} en worden
boven ongeveer \num{45} vernield; elk \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} werkt binnen een zuurgraad
die bij zijn plaats past (pepsine in het zuur van de maag, trypsine in de
neutrale darm); en zijn snelheid stijgt met de concentratie van zijn
\omterm{def:g11:enzymes-and-phenotype:enzyme}{substraat} tot een maximum, wanneer elke \omterm{def:g11:enzymes-and-phenotype:enzyme}{actieve plaats} bezet is.
\end{proposition}
\admitted

\section{Reactieketens}

\begin{proposition}[Metabole routes]\label{prop:g11:enzymes-and-phenotype:pathway}
Een \omterm{def:g10:cells-common-unit:cell}{cel} maakt een product zelden in één stap: een beginmolecuul wordt
omgezet door een reeks \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymen}, die elk op het product van het vorige
inwerken, langs een \emph{metabole route}\index{metabole route}. Elk \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym}
wordt door een eigen \omterm{def:g10:universal-dna:gene}{gen} gecodeerd. Een niet werkend \omterm{def:g10:universal-dna:gene}{allel} van welk \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym}
ook blokkeert de route bij die stap: het product wordt niet gemaakt, en
het \omterm{def:g11:enzymes-and-phenotype:enzyme}{substraat} van het geblokkeerde \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} kan zich ophopen. Twee
verschillende \omterm{def:g10:universal-dna:gene}{genen} kunnen zo hetzelfde zichtbare gebrek opleveren, en het
gebrek van één \omterm{def:g10:universal-dna:gene}{gen} kan er verscheidene opleveren.
\end{proposition}
\admitted

\begin{example}[Fenylketonurie, en hoe een dieet een fenotype verandert]\label{ex:g11:enzymes-and-phenotype:pku}
Ongeveer één pasgeborene op tienduizend draagt twee niet werkende \omterm{def:g10:universal-dna:gene}{allelen}
van het \omterm{def:g10:universal-dna:gene}{gen} voor \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} 1. Fenylalanine uit voedsel wordt niet omgezet; de
concentratie ervan in het bloed stijgt twintigvoudig en zij vergiftigt de
zich ontwikkelende hersenen. Onbehandeld loopt het kind een ernstige
verstandelijke beperking op. Bij de geboorte opgespoord met een druppel
bloed op een kaartje --- elke pasgeborene wordt getest --- en gevoed met
een voeding die vrijwel geen fenylalanine bevat, ontwikkelt datzelfde kind
zich normaal. Het \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} is onveranderd; het \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} is, op elke schaal
boven de moleculaire, dat van een gezond mens. De omgeving, hier de
voeding, gaf de doorslag.
\end{example}

\section{Genotype, omgeving, fenotype}

\begin{proposition}[De omgeving vormt het fenotype]\label{prop:g11:enzymes-and-phenotype:environment}
Het \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} is het resultaat van het \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} \emph{én} de omgeving. De
omgeving grijpt in op de moleculaire schaal (temperatuur en zuurgraad
bepalen de werkzaamheid van \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymen}, de aanwezigheid van een \omterm{def:g11:enzymes-and-phenotype:enzyme}{substraat}
bepaalt of een route loopt), op de cellulaire schaal (\omterm{def:g10:cells-common-unit:cell}{cellen} die op
signalen, voedingsstoffen en licht reageren) en op de schaal van het
organisme (voeding, beweging, ziekte). Individuen met een gelijk \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype}
--- eeneiige tweelingen, stekken van één plant --- verschillen in
\omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} naar hun voorgeschiedenis; en individuen met verschillende
\omterm{def:g11:enzymes-and-phenotype:phenotype}{genotypen} kunnen onder geschikte omstandigheden bij hetzelfde \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype}
uitkomen.
\end{proposition}
\begin{proof}[Bewijsmateriaal]
De temperatuurgevoelige tyrosinase van de siamese kat: pigment vormt zich
alleen in de koele uiteinden, en een gekoelde kaalgeschoren plek groeit
donker terug. Fenylketonurie die met voeding onder controle wordt
gehouden. Eeneiige tweelingen die apart zijn grootgebracht en enkele
centimeters in lengte en nog meer in lichaamsmassa verschillen. Een plant
van dezelfde kloon op zeeniveau en op hoogte gekweekt, die tienvoudig in
hoogte verschilt en terugkeert wanneer zij wordt verplant. In elk geval
staat het \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} vast en beweegt het \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} mee met de
omstandigheden.
\end{proof}

\begin{method}[Redeneren van gen naar kenmerk]\label{met:g11:enzymes-and-phenotype:reason}
\begin{enumerate}
  \item Stel het \omterm{def:g11:gene-expression:protein}{eiwit} vast: welk \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym}, in welke route, en wat het doet.
  \item Stel het gevolg van het \omterm{def:g10:universal-dna:gene}{allel} voor het \omterm{def:g11:gene-expression:protein}{eiwit} vast: normaal,
        verminderd, afwezig, of veranderd in zijn voorwaarden voor
        werkzaamheid.
  \item Volg het gevolg langs de schalen omhoog: wat de \omterm{def:g10:cells-common-unit:cell}{cel} mist of
        ophoopt, en daarna wat het organisme vertoont.
  \item Vraag je af hoeveel werkende \omterm{def:g10:universal-dna:gene}{allelen} nodig zijn: voor de meeste
        \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzymen} volstaat er één, zodat het gebrek pas met twee niet
        werkende \omterm{def:g10:universal-dna:gene}{allelen} verschijnt (een recessief kenmerk).
  \item Vraag je af wat de omgeving toevoegt: \omterm{def:g11:enzymes-and-phenotype:enzyme}{substraat} wel of niet
        aanwezig, temperatuur, een behandeling.
\end{enumerate}
\end{method}

\begin{remark}[Waarom één werkend allel meestal genoeg is]\label{rem:g11:enzymes-and-phenotype:recessive}
\omterm{def:g11:enzymes-and-phenotype:enzyme}{Enzymen} zijn katalysatoren: een kleine hoeveelheid zet een grote
hoeveelheid \omterm{def:g11:enzymes-and-phenotype:enzyme}{substraat} om. De halve normale hoeveelheid tyrosinase maakt
nog altijd genoeg melanine, de halve normale hoeveelheid lactase verteert
nog altijd een glas melk. Een drager van één niet werkend \omterm{def:g10:universal-dna:gene}{allel} is dus
meestal niet van een niet-drager te onderscheiden --- het "recessieve"
van het deel over de basisschool, nu verklaard --- en het kenmerk
verschijnt pas wanneer beide \omterm{def:g10:universal-dna:gene}{allelen} falen. \omterm{def:g10:chemistry-of-life:families}{Eiwitten} die geen \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} zijn
en in hoeveelheid of in precieze vorm nodig zijn, vormen vaak de
uitzondering: half zoveel hemoglobine maakt wel verschil.
\end{remark}

\section{Probleem: De kat met koude oren}

\begin{problem}[{Weekendprobleem --- de vacht van een siamese kat enzym
voor enzym verklaard: de temperatuurgevoelige tyrosinase, de kaart van
lichaamstemperaturen, de kaalgeschoren plek, en een dieet dat een fenotype
herschrijft}]
\label{pb:g11:enzymes-and-phenotype:1}
Het siamese \omterm{def:g10:universal-dna:gene}{allel} van het tyrosinasegen codeert voor een \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} waarvan de
werkzaamheid, in het laboratorium gemeten, als volgt is:
\qty{100}{\percent} bij \qty{30}{\celsius}, \qty{80}{\percent} bij
\qty{33}{\celsius}, \qty{40}{\percent} bij \qty{35}{\celsius},
\qty{5}{\percent} bij \qty{37}{\celsius}, \qty{0}{\percent} bij
\qty{39}{\celsius}. Het \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} van het gewone \omterm{def:g10:universal-dna:gene}{allel} houdt
\qty{100}{\percent} tot \qty{40}{\celsius}. Huidtemperaturen van een kat:
romp \qty{38}{\celsius}, poten \qty{36}{\celsius}, voeten
\qty{33}{\celsius}, oren en snuit \qty{31}{\celsius}. Melanine is in de
vacht zichtbaar wanneer het \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} voor meer dan ongeveer
\qty{30}{\percent} werkt.

\medskip
\noindent\textbf{Deel I --- De kaart.}
\begin{enumerate}
  \item Lees voor elk lichaamsdeel de werkzaamheid van het siamese \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym}
        af en zeg of de vacht gepigmenteerd is.
  \item Teken, of beschrijf, de vacht die daaruit volgt. Klopt zij met een
        siamees?
  \item Doe hetzelfde voor een kat met het gewone \omterm{def:g10:universal-dna:gene}{allel}.
  \item Een siamees katje wordt overal roomkleurig geboren. Verklaar dat,
        wetend dat de baarmoeder \qty{38}{\celsius} is.
  \item Bij welke huidtemperatuur ligt de grens tussen donkere en bleke
        vacht? Welk deel van de kromme van het \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} legt haar vast?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De proef.}
\begin{enumerate}[resume]
  \item Een plek op de romp wordt kaalgeschoren en onder een koelelement
        op \qty{30}{\celsius} gehouden terwijl de vacht teruggroeit.
        Voorspel de kleur ervan.
  \item Een plek op een voet wordt kaalgeschoren en met een verband warm
        gehouden op \qty{38}{\celsius}. Voorspel.
  \item Nadat de elementen zijn weggehaald, behoudt de teruggegroeide
        vacht haar kleur maandenlang en wordt zij dan geleidelijk
        vervangen. Verklaar beide feiten.
  \item Leg uit waarom deze proeven laten zien dat de omgeving op de
        moleculaire schaal ingrijpt en niet op het \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype}.
  \item Een leerling stelt voor dat de donkere uiteinden aan een ander
        \omterm{def:g10:universal-dna:gene}{allel} in die \omterm{def:g10:cells-common-unit:cell}{cellen} te danken zijn. Geef twee tegenargumenten.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De drie schalen.}
\begin{enumerate}[resume]
  \item Beschrijf het siamese \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} op de moleculaire en de cellulaire
        schaal en op die van het organisme.
  \item Het siamese \omterm{def:g10:universal-dna:gene}{allel} verschilt van het gewone door één substitutie
        die één \omterm{def:g10:chemistry-of-life:families}{aminozuur} verandert. Leg uit hoe één \omterm{def:g10:chemistry-of-life:families}{aminozuur} een \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym}
        temperatuurgevoelig kan maken.
  \item Een kat met één siamees en één gewoon \omterm{def:g10:universal-dna:gene}{allel} is volledig
        gepigmenteerd. Leg dat uit met
        \cref{rem:g11:enzymes-and-phenotype:recessive}.
  \item Een albinokat heeft twee \omterm{def:g10:universal-dna:gene}{allelen} van hetzelfde \omterm{def:g10:universal-dna:gene}{gen} die bij geen
        enkele temperatuur een werkend \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} opleveren. Vergelijk haar
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} op de drie schalen met dat van de siamees.
  \item Himalayakonijnen en bepaalde muizen vertonen hetzelfde patroon.
        Wat suggereert dat over hun tyrosinasegen, en wat zou je van de
        \omterm{def:g11:mutations:mutation}{mutatie} verwachten?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Een dieet dat een \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} herschrijft.}
Bij een pasgeborene wordt fenylketonurie vastgesteld: fenylalanine in het
bloed \qty{1200}{\micro mol/L} in plaats van \num{60}. Het veilige niveau
voor de hersenontwikkeling ligt onder \qty{360}{\micro mol/L}.
\begin{enumerate}[resume]
  \item Met welke factor is de fenylalanine verhoogd, en welk \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} uit de
        routefiguur ontbreekt?
  \item Leg uit waarom juist de hersenen van het kind, en niet
        bijvoorbeeld zijn huid, het orgaan zijn dat gevaar loopt.
  \item Het dieet brengt de inname van fenylalanine terug tot een vijfde
        van het gewone; het bloedgehalte daalt evenredig. Is het kind
        veilig? Wat zegt dat over het verband tussen de aanvoer van
        \omterm{def:g11:enzymes-and-phenotype:enzyme}{substraat} en het \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} wanneer het \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzym} ontbreekt?
  \item Toch is enige fenylalanine onmisbaar, omdat het lichaam haar niet
        kan maken. Leg uit waarom de voeding moet worden geregeld en
        fenylalanine niet eenvoudigweg kan worden weggelaten.
  \item Geef het resultaat: wat het \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype} bij de siamese kat en bij het
        kind vastlegde, wat de omgeving bepaalde, en de ene zin die
        \omterm{def:g11:enzymes-and-phenotype:phenotype}{genotype}, omgeving en \omterm{def:g11:enzymes-and-phenotype:phenotype}{fenotype} met elkaar verbindt.
\end{enumerate}
\end{problem}
