\chapter{التنوع الحيوي على كل سلّم}\label{ch:g10:biodiversity-scales}

ضع مربعًا من الخيط ضلعه متر على مرج وعُدّ ما ينبت داخله: العشب طبعًا،
لكن أيضًا النفل والأقحوان والهندباء ولسان الحمل والطحلب، وإن نظرت
بعدسة فعشرات الحشرات وحلزون. ثم انقل المربع إلى إسفلت الموقف: فلا شيء،
أو عشبة واحدة في شق. ثم انقله إلى مرج قديم لم يحرث قط: فأربعون نوعًا
من النبات، وحشرات لا تحصى. والكلمة التي تسمي ما يقيسه المربع هي
\emph{\omterm{def:g10:biodiversity-scales:biodiversity}{التنوع الحيوي}}، وهذا الفصل عن كيفية قياسه، وعلى أي سلالم يوجد،
ولماذا هو غير ثابت وغير مضمون.

\section{ثلاثة سلالم للتنوع}

\begin{definition}[التنوع الحيوي]\label{def:g10:biodiversity-scales:biodiversity}
\emph{التنوع الحيوي}\index{تنوع حيوي} هو تنوع الكائنات الحية، منظورًا
إليه على ثلاثة سلالم:
\begin{itemize}
  \item تنوع \emph{الأنظمة البيئية}\index{نظام بيئي} --- أي المجتمعات
        الحية والأوساط التي تشغلها (بركة، غابة زان، شعبة مرجانية،
        مرج)، ولكل منها مجموعة أنواعها الخاصة؛
  \item تنوع \omterm{def:g10:biodiversity-scales:species}{الأنواع} داخل نظام بيئي أو على الأرض --- أي عدد \omterm{def:g10:biodiversity-scales:species}{الأنواع}
        ووفرتها النسبية؛
  \item \emph{التنوع الوراثي}\index{تنوع وراثي} داخل \omterm{def:g10:biodiversity-scales:species}{النوع} --- أي عدد
        \omterm{def:g10:universal-dna:gene}{أليلات} مورّثاته وطريقة توزعها على أفراده وجمهراته.
\end{itemize}
\end{definition}

\begin{definition}[النوع]\label{def:g10:biodiversity-scales:species}
\emph{النوع}\index{نوع} جماعة من الكائنات الحية يشبه بعضها بعضًا،
وتستطيع أن تتناسل فيما بينها وتنتج نسلًا خصبًا؛ وجمهرتان لا تستطيعان
التناسل، أو هجائنهما عقيمة، نوعان متمايزان. أما عمليًا --- في
الأحافير، وفي البكتيريا، وفي كائنات لم تر قط وهي تتناسل --- فتعرف
الأنواع بخصائصها المشتركة، وسيبين
\cref{ch:g12:selection-drift-speciation} أن الحد ليس حادًا دائمًا.
\end{definition}

\begin{example}[الحصان والحمار والبغل]\label{ex:g10:biodiversity-scales:mule}
الحصان والحمار يشبه أحدهما الآخر ويستطيعان التناسل: ونسلهما البغل.
لكن البغل عقيم، فالحصان والحمار نوعان. أما البودل والكلب السلوقي فلا
يشبه أحدهما الآخر في شيء، ويتناسلان بيسر، ويعطيان جراءً خصبة: فهما
\omterm{def:g10:biodiversity-scales:species}{نوع} واحد، هو الكلب --- وهو قادر كذلك على التناسل مع الذئب، سلفه.
\end{example}

\section{تنوع الأنواع}

\begin{proposition}[عدّ الأنواع]\label{prop:g10:biodiversity-scales:count}
وصف نحو مليوني \omterm{def:g10:biodiversity-scales:species}{نوع} وسمّيت. ومعظمها حشرات (أكثر من مليون)، تليها
النباتات (نحو 300\,000)، ثم الفطريات والعنكبيات والرخويات؛ أما
الفقاريات، وهي المجموعة التي نعرفها أفضل من غيرها، فنحو 70\,000 \omterm{def:g10:biodiversity-scales:species}{نوع}،
منها 6\,400 من الثدييات. أما عدد \omterm{def:g10:biodiversity-scales:species}{الأنواع} التي لم توصف بعد فيقدر،
انطلاقًا من نسبة اكتشاف \omterm{def:g10:biodiversity-scales:species}{الأنواع} الجديدة، بعدة ملايين أخرى --- ولعلها
ثمانية إلى عشرة ملايين \omterm{def:g10:biodiversity-scales:species}{نوع} من حقيقيات \omterm{def:g10:cells-common-unit:organelle}{النواة} إجمالًا، إلى جانب تنوع
هائل من البكتيريا لم يعد في معظمه.
\end{proposition}
\admitted

\begin{method}[قياس تنوع موقع]\label{met:g10:biodiversity-scales:sampling}
لا يعد الموقع كله قط؛ بل \emph{يعاين}.
\begin{enumerate}
  \item اختر وحدة معاينة موافقة للكائنات: مربعًا من \qty{1}{m^2}
        لنباتات المرج، وكنسة شبكة للحشرات، وغربال تربة لديدان الأرض،
        وطول مقطع للطيور.
  \item ضع الوحدات عشوائيًا أو على فترات منتظمة، بعدد يكفي حتى تصير
        إضافة وحدة أخرى قليلة الأثر في المجموع.
  \item عيّن \omterm{def:g10:biodiversity-scales:species}{الأنواع} وعُدّها في كل وحدة. ويصف العينةَ عددان:
        \emph{غنى \omterm{def:g10:biodiversity-scales:species}{الأنواع}}، وهو عدد \omterm{def:g10:biodiversity-scales:species}{الأنواع} الموجودة، و\emph{الوفرة
        النسبية} لكل منها --- فموقع يمثل فيه \omterm{def:g10:biodiversity-scales:species}{نوع} واحد 95\% من الأفراد
        أقل تنوعًا من موقع تتقاسمها فيه عشرة \omterm{def:g10:biodiversity-scales:species}{أنواع} بالتساوي، ولو
        تساوى الغنى.
  \item لا تقارن المواقع إلا بالطريقة نفسها والجهد نفسه، وفي الموسم
        نفسه.
\end{enumerate}
\end{method}

\begin{example}[موقعان عشبيان]\label{ex:g10:biodiversity-scales:lawns}
عشرة مربعات على مرج مدرسة مجزوز: 6 \omterm{def:g10:biodiversity-scales:species}{أنواع} نباتية، يغطي العشب منها
90\%. وعشرة مربعات على حافة غير مجزوزة بجانبه: 23 نوعًا، لا يتجاوز
أحدها 30\%. فالحافة أغنى وأكثر توازنًا؛ والفرق ليس التربة ولا المناخ،
وهما نفسهما، بل الجز، الذي يزيل كل نبات لا يستطيع الإزهار قرب الأرض.
\end{example}

\section{التنوع الوراثي داخل النوع}

\begin{proposition}[أفراد النوع ليسوا متطابقين]\label{prop:g10:biodiversity-scales:genetic}
داخل \omterm{def:g10:biodiversity-scales:species}{النوع}، توجد معظم \omterm{def:g10:universal-dna:gene}{المورّثات} في عدة \omterm{def:g10:universal-dna:gene}{أليلات}، ويحمل الأفراد تراكيب
مختلفة منها. و\emph{تواتر} \omterm{def:g10:universal-dna:gene}{الأليل} في جمهرة --- أي نسبة نسخ \omterm{def:g10:universal-dna:gene}{المورّثة}
التي هي هذا \omterm{def:g10:universal-dna:gene}{الأليل} --- يتفاوت من جمهرة إلى أخرى. \omterm{def:g10:biodiversity-scales:biodiversity}{والتنوع الوراثي} هو
ما يتيح \omterm{def:g10:biodiversity-scales:species}{للنوع} أن يستجيب لمحيط متغير: فالجمهرة التي يتشابه فيها كل
الأفراد لا شيء فيها ينتقى.
\end{proposition}
\admitted

\begin{example}[أليلات زمر الدم حول العالم]\label{ex:g10:biodiversity-scales:abo}
\omterm{def:g10:universal-dna:gene}{لمورّثة} زمر الدم ABO ثلاثة \omterm{def:g10:universal-dna:gene}{أليلات} شائعة، هي A وB وO. وتواتراتها في
غرب أوروبا نحو 0.27 و0.07 و0.66؛ وفي شرق آسيا نحو 0.20 و0.20 و0.60؛
وفي بعض جمهرات أمريكا الجنوبية الأصلية يبلغ \omterm{def:g10:universal-dna:gene}{الأليل} O وحده 0.98.
\omterm{def:g10:biodiversity-scales:species}{فالنوع} نفسه، \omterm{def:g10:universal-dna:gene}{والمورّثة} نفسها، وتوزيعات مختلفة لأليلاتها: \omterm{def:g10:biodiversity-scales:biodiversity}{التنوع
الوراثي} للإنسان حقيقي، وهو منتشر عبر الجمهرات لا فاصل بينها --- فمعظم
تنوع \omterm{def:g10:biodiversity-scales:species}{النوع} يوجد داخل أي جمهرة واحدة.
\end{example}

\begin{example}[نوع لا يكاد يملك منه شيئًا]\label{ex:g10:biodiversity-scales:cheetah}
الفهود الصيادة متشابهة وراثيًا إلى حد أن طعوم الجلد بين حيوانات لا
قرابة بينها تقبل كما لو كانت بين توأمين: فقد مر \omterm{def:g10:biodiversity-scales:species}{النوع} بعنق زجاجة من
أفراد قليلين جدًا قبل نحو عشرة آلاف سنة ولم يستعد أليلاته قط. فمهما
جاء من مرض جديد أو مناخ جديد، يلقاه كل فهد بالعدة نفسها؛ \omterm{def:g10:biodiversity-scales:species}{والنوع} باق،
لكن بلا احتياط.
\end{example}

\section{التنوع الحيوي يتغير: في الماضي وفي الحاضر}

\begin{proposition}[التنوع الحيوي حال لا ثابت]\label{prop:g10:biodiversity-scales:changes}
مجموعة \omterm{def:g10:biodiversity-scales:species}{الأنواع} الحية في زمن ما مرحلة من تاريخ. ويبين السجل الأحفوري
أنواعًا تظهر، وتدوم بضعة ملايين من السنين وسطيًا، وتختفي؛ وقد ازداد
تنوع الحياة الكلي على مدى 540 مليون سنة الأخيرة، قاطعته خمسة
\emph{انقراضات جماعية}\index{انقراض جماعي} زال في كل منها أكثر من
ثلاثة أرباع \omterm{def:g10:biodiversity-scales:species}{الأنواع} في زمن قصير جيولوجيًا. وأحدثها، قبل 66 مليون سنة،
أنهى الديناصورات غير الطيرية وفتح الطريق أمام تنوع الثدييات.
\end{proposition}
\begin{proof}[شواهد]
عدّ العائلات الأحفورية في طبقات صخرية مؤرخة، مجموعًا للسجل البحري
كله، يعطي منحني الشكل التالي: ارتفاع من بضع مئات إلى قرابة ألف عائلة،
مع هبوطات حادة عند 445 و370 و252 و201 و66 مليون سنة مضت. وحادثة
252 مليون سنة، وهي الكبرى، أزالت نحو 95\% من \omterm{def:g10:biodiversity-scales:species}{الأنواع} البحرية؛
والطبقات التي فوقها مباشرة تكاد تخلو من الأحافير، ولا تملؤها مجموعات
جديدة إلا على مدى ملايين السنين التالية.
\end{proof}

\begin{proposition}[التغير الحاضر]\label{prop:g10:biodiversity-scales:present}
تختفي \omterm{def:g10:biodiversity-scales:species}{الأنواع} في زماننا بنسبة تقدر بمئة إلى ألف ضعف وسطي السجل
الأحفوري، بسبب تدمير \omterm{def:g10:biodiversity-scales:biodiversity}{الأنظمة البيئية} (قطع الغابات، وتجفيف الأراضي
الرطبة، والتوسع العمراني)، وبسبب الاستغلال المفرط، \omterm{def:g10:biodiversity-scales:species}{والأنواع} الدخيلة،
والتلوث، وتغير المناخ --- وكلها من صنع الإنسان. فقد زال نصف الغابات
المدارية التي كانت سنة 1950؛ وثلث \omterm{def:g10:biodiversity-scales:species}{أنواع} البرمائيات مهدد؛ وضاق \omterm{def:g10:biodiversity-scales:biodiversity}{التنوع
الوراثي} للمحاصيل والماشية إلى بضعة أصناف. \omterm{def:g10:biodiversity-scales:biodiversity}{والتنوع الحيوي} مورد ---
غذاءً ودواءً وتأبيرًا وتربةً واستقرارًا للأنظمة البيئية --- وفقده
غير قابل للرد على سلّم زمن الإنسان.
\end{proposition}
\admitted

\begin{example}[رد خسارة]\label{ex:g10:biodiversity-scales:reversing}
حيث أعيد غرس السياج النباتي الذي أزيل في الستينيات، عاد غنى الطيور
والحشرات في غضون عقد؛ وحيث أعيدت الذئاب إلى حديقة وطنية سنة 1995،
كفّت الأيائل التي تصيدها عن تعرية صفصاف ضفاف النهر، فنما الصفصاف،
وعادت القنادس، ومعها الطيور والأسماك التي تعتمد على برك القنادس.
فحماية \omterm{def:g10:biodiversity-scales:biodiversity}{نظام بيئي} أو ترميمه يرد التنوع على السلالم الثلاثة دفعة واحدة؛
أما \omterm{def:g10:biodiversity-scales:species}{نوع} يبقى حيًا في حديقة حيوان فلا يحفظ منها إلا واحدًا.
\end{example}

\section{مسألة: عائد السياج النباتي}

\begin{problem}[{مسألة نهاية الأسبوع --- مسح ميداني لمزرعتين، إحداهما
بسياج نباتي والأخرى بلا سياج: أنواع تعد، ووفرات تقارن، وأليلات حلزون
تتبع، والعدد الذي يساويه سياج}]
\label{pb:g10:biodiversity-scales:1}
مزرعتان متجاورتان تزرعان القمح نفسه في التربة نفسها. المزرعة A حفظت
سياجها النباتي؛ والمزرعة B أزالته سنة 1970. ويعاين صف المزرعتين في
حزيران بالطريقة نفسها: 20 مربعًا من \qty{1}{m^2} على حواف الحقول
للنباتات، و20 كنسة شبكة للحشرات، و10 دقائق من الإنصات عند 5 نقاط
للطيور.

\medskip
\noindent\textbf{الجزء الأول --- النباتات.}
النتائج، المزرعة A: 31 نوعًا نباتيًا في المربعات العشرين؛ والمزرعة B:
9. وفي المزرعة B يمثل عشب واحد 84\% من النباتات المعدودة؛ وفي المزرعة
A يمثل أشيع \omterm{def:g10:biodiversity-scales:species}{الأنواع} 22\%.
\begin{enumerate}
  \item أعطِ غنى \omterm{def:g10:biodiversity-scales:species}{الأنواع} على حافتي كل مزرعة والنسبة بينهما.
  \item أي المزرعتين مجتمعها النباتي أكثر توازنًا؟ فسّر ما الذي
        يضيفه «التوازن» إلى «الغنى».
  \item في المزرعة A كان عدد \omterm{def:g10:biodiversity-scales:species}{الأنواع} الموجودة بعد 5 و10 و15 و20
        مربعًا: 18 و25 و29 و31. فهل المعاينة كافية؟ علّل.
  \item وكانت السلسلة نفسها في المزرعة B: 6 و8 و9 و9. قارن السلسلتين
        وقل أي شكل يأخذه منحني معاينة «كاملة».
  \item لماذا يجب معاينة المزرعتين في الشهر نفسه؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الحشرات والطيور.}
الحشرات: المزرعة A 46 نوعًا، والمزرعة B 14. والطيور: المزرعة A 17
نوعًا، والمزرعة B 5؛ وفي المزرعة B كان 60\% من الطيور المسموعة من \omterm{def:g10:biodiversity-scales:species}{نوع}
واحد.
\begin{enumerate}[resume]
  \item احسب نسبة غنى الحشرات ونسبة غنى الطيور بين المزرعتين. فهل
        تروي المجموعات الثلاث (النباتات والحشرات والطيور) القصة
        نفسها؟
  \item اقترح سلسلة أسباب تربط إزالة السياج بهبوط تنوع الحشرات، ثم
        بتنوع الطيور.
  \item محصول القمح في الهكتار واحد في المزرعتين. فهل يعني ذلك أن لا
        أثر للسياج في المحصول؟ اقترح خدمة واحدة قد يؤديها السياج لا
        يبينها رقم المحصول.
  \item سمِّ نوعًا واحدًا يتوقع الصف أن يجده في المزرعة A وحدها وفسّر
        السبب.
  \item أي سلالم \omterm{def:g10:biodiversity-scales:biodiversity}{التنوع الحيوي} اختزل في المزرعة B اختزالًا أظهر:
        \omterm{def:g10:biodiversity-scales:biodiversity}{الأنظمة البيئية} أم \omterm{def:g10:biodiversity-scales:species}{الأنواع} أم \omterm{def:g10:universal-dna:gene}{المورّثات}؟ علّل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{def:g10:universal-dna:gene}{أليلات} حلزون.}
حلزون البستان يحمل \omterm{def:g10:universal-dna:gene}{مورّثة} لها أليلان شائعان للون الصدفة، أصفر (y)
وبني (b). وفي سياج المزرعة A عوين 400 حلزون: 100 منها لها أليلا y،
و200 لها y وb، و100 لها أليلا b. وفي الضفة الوحيدة الباقية في المزرعة
B، 100 حلزون: 4 بأليلي y، و32 \omterm{def:g10:universal-dna:gene}{بأليل} من كل، و64 بأليلي b.
\begin{enumerate}[resume]
  \item عُدّ نسخ y ونسخ b في كل عينة (فكل حلزون يحمل نسختين). واستنتج
        تواتر y في كل مزرعة.
  \item أي الجمهرتين أكثر تنوعًا وراثيًا عند هذه \omterm{def:g10:universal-dna:gene}{المورّثة}؟ قارن نسبة
        الحلزونات الحاملة \omterm{def:g10:universal-dna:gene}{للأليلين} معًا.
  \item السمنة تصيد الحلزون بالنظر؛ وعلى ضفة المزرعة B العارية تكون
        الصدفات البنية أخفى. اقترح كيف نشأت تواترات المزرعة B.
  \item لو كان سياج رطب مظلل يفضل الصدفات الصفراء، فبم تتنبأ لحلزونات
        المزرعة B لو أعيد غرس السياج ولم يتغير شيء آخر؟
  \item حلزونات المزرعة B بضع مئات؛ وحلزونات المزرعة A عشرات الألوف.
        فأي الجمهرتين أشد عرضة لفقد \omterm{def:g10:universal-dna:gene}{أليل} بأكمله بالمصادفة؟ ولماذا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- العائد.}
\begin{enumerate}[resume]
  \item عبر المجموعات الثلاث، احسب غنى \omterm{def:g10:biodiversity-scales:species}{الأنواع} الكلي في كل مزرعة، ثم
        مجموع المزرعة A مقسومًا على مجموع المزرعة B.
  \item يشغل سياج المزرعة A 3\% من مساحتها. اذكر في جملة واحدة أي
        نسبة من الأرض تحوي أي نسبة من \omterm{def:g10:biodiversity-scales:species}{الأنواع}.
  \item السياج الذي أعيد غرسه في المزرعة B سنة 1995 أعيد مسحه سنة
        2010: 24 نوعًا نباتيًا، و35 من الحشرات، و12 من الطيور. فأي
        نسبة من غنى المزرعة A استعادتها كل مجموعة؟
  \item أي سلالم \omterm{def:g10:biodiversity-scales:biodiversity}{التنوع الحيوي} الثلاثة أبطأ تعافيًا بعد إعادة الغرس،
        ولماذا؟
  \item اذكر النتيجة: العامل الذي ضاعف به السياج النباتي غنى \omterm{def:g10:biodiversity-scales:species}{أنواع}
        مزرعة، وسلّم \omterm{def:g10:biodiversity-scales:biodiversity}{التنوع الحيوي} الوحيد الذي لا يستطيع سياج أعيد
        غرسه أن يرده وحده.
\end{enumerate}
\end{problem}
