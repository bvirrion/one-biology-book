\chapter{Local Ecosystems}\label{ch:g4:ecosystems}

A school garden, a \omterm{def:g2:habitats:kinds}{pond}, a park and a stretch of coastline are more than
pretty scenery. Each is a \omterm{def:g1:caring:needs}{place} where \omterm{def:g1:living-nonliving:living}{living things} interact with one
another and with non-living conditions. That whole package is an
\omterm{def:g4:ecosystems:levels}{ecosystem} --- and local examples are perfect for careful observation.

\section{From organism to ecosystem}

\begin{definition}[Organism to ecosystem]\label{def:g4:ecosystems:levels}
An \emph{organism}\index{organism} is one living individual. A
\emph{population}\index{population} is all the members of one kind of
organism in an area. A \emph{community}\index{community} is all the
populations living together there. An
\emph{ecosystem}\index{ecosystem} is the community plus the non-living
surroundings (\omterm{def:g4:plants-need:needs}{light}, \omterm{def:g4:plants-need:needs}{water}, soil, \omterm{def:g4:plants-need:needs}{air}, temperature).
\end{definition}

\begin{example}[Park bench view]\label{ex:g4:ecosystems:park}
One oak is an \omterm{def:g4:ecosystems:levels}{organism}. All the oaks in the park form a \omterm{def:g4:ecosystems:levels}{population}.
Oaks, squirrels, \omterm{def:g2:animal-groups:five}{birds}, \omterm{def:g3:insects:insect}{insects} and fungi form a \omterm{def:g4:ecosystems:levels}{community}. Add
sunlight, rain and soil and you have a park \omterm{def:g4:ecosystems:levels}{ecosystem}.
\end{example}

\section{Living and non-living parts}

\begin{definition}[Biotic and abiotic]\label{def:g4:ecosystems:bio-abio}
\emph{Biotic}\index{biotic} factors are living (or once-living) parts of
an \omterm{def:g4:ecosystems:levels}{ecosystem}: \omterm{def:g1:plants-animals:plant}{plants}, \omterm{def:g1:plants-animals:animal}{animals}, microbes, dead \omterm{def:g2:plant-parts:main}{leaves}. \emph{Abiotic}
\index{abiotic} factors are non-living: sunlight, temperature, \omterm{def:g4:plants-need:needs}{water},
rocks, wind.
\end{definition}

\section{Who eats whom?}

\begin{definition}[Producers and consumers]\label{def:g4:ecosystems:roles}
\emph{Producers}\index{producer} (mainly green \omterm{def:g1:plants-animals:plant}{plants} and algae) make
\omterm{def:g2:food-energy:food}{food} using \omterm{def:g4:plants-need:needs}{light}. \emph{Consumers}\index{consumer} eat other \omterm{def:g1:living-nonliving:living}{living
things} or their remains. \emph{Decomposers}\index{decomposer} (many
fungi and bacteria) break down dead material and return \omterm{def:g4:plants-need:needs}{nutrients} to
the soil or \omterm{def:g4:plants-need:needs}{water}.
\end{definition}

\omimg{food-chain.jpg}{0.52\linewidth}{A \omterm{def:g3:food-chains:chain}{food chain} shows one path of
who-eats-whom. Arrows point in the direction \omterm{def:g2:food-energy:energy}{energy} flows --- toward the
eater.}

\begin{example}[Garden chain]\label{ex:g4:ecosystems:chain}
Lettuce $\to$ \omterm{def:g3:insects:compare}{snail} $\to$ thrush. \omterm{def:g2:food-energy:energy}{Energy} from sunlight, stored in the
lettuce, moves along the chain when \omterm{def:g1:plants-animals:animal}{animals} feed.
\end{example}

\begin{proposition}[Sun as the main start]\label{prop:g4:ecosystems:sun}
In most local \omterm{def:g4:ecosystems:levels}{ecosystems} on land and in sunlit \omterm{def:g4:plants-need:needs}{water}, \omterm{def:g2:food-energy:energy}{energy} enters
through \omterm{def:g4:ecosystems:roles}{producers} that capture sunlight.
\end{proposition}
\admitted

\section{Habitats nearby}

\begin{definition}[Habitat]\label{def:g4:ecosystems:habitat}
A \emph{habitat}\index{habitat} is the \omterm{def:g1:caring:needs}{place} where an \omterm{def:g4:ecosystems:levels}{organism} lives ---
its address in the \omterm{def:g4:ecosystems:levels}{ecosystem}.
\end{definition}



\begin{method}[Surveying a small plot]\label{met:g4:ecosystems:survey}
\begin{enumerate}
  \item Choose a safe small area (for example one square metre of
        grass, with permission).
  \item Record \omterm{def:g4:ecosystems:bio-abio}{abiotic} notes: sunny or shady, damp or dry, soil or
        pavement.
  \item List \omterm{def:g1:plants-animals:plant}{plants} and any \omterm{def:g1:plants-animals:animal}{animals} or signs of \omterm{def:g1:plants-animals:animal}{animals} (tracks,
        webs, droppings).
  \item Sketch a simple \omterm{def:g3:food-chains:chain}{food chain} using what you found.
  \item Leave the \omterm{def:g1:caring:needs}{place} as you found it.
\end{enumerate}
\end{method}

\section{When ecosystems change}

\begin{example}[Human influence]\label{ex:g4:ecosystems:human}
Litter, drained \omterm{def:g2:habitats:kinds}{ponds}, planted gardens, roads and pollution all change
local \omterm{def:g4:ecosystems:levels}{ecosystems}. Some changes help wildlife (native \omterm{def:g2:plant-parts:main}{flowers} for
pollinators); others harm it.
\end{example}
