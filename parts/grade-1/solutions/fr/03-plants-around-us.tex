\section*{Chapitre \ref{ch:g1:plants-around-us} --- Les plantes autour de nous}

\begin{solution}{exo:g1:plants-around-us:1}
Les \omterm{def:g1:plants-around-us:parts}{racines} (sous la terre), la \omterm{def:g1:plants-around-us:parts}{tige}, les \omterm{def:g1:plants-around-us:parts}{feuilles}, la \omterm{def:g1:plants-around-us:parts}{fleur}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:2}
(a) Les \omterm{def:g1:plants-around-us:parts}{racines}. (b) Les \omterm{def:g1:plants-around-us:parts}{feuilles}. (c) La \omterm{def:g1:plants-around-us:parts}{fleur}. (d) La \omterm{def:g1:plants-around-us:parts}{tige}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:3}
Elle est née d'une graine, elle se nourrit (l'eau du sol, la lumière
du ciel), elle grandit, elle fait des graines qui deviennent de
nouvelles pâquerettes, et elle meurt un jour. Tous les signes de la
vie sont là~; marcher n'en fait pas partie.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:4}
La \omterm{def:g1:plants-around-us:parts}{tige} de l'\omterm{def:g1:plants-around-us:tree}{arbre} est le tronc~; les \omterm{def:g1:plants-around-us:parts}{feuilles} sont portées par les
\omterm{def:g1:plants-around-us:tree}{branches}.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:5}
De l'eau, de la lumière et une place pour ses \omterm{def:g1:plants-around-us:parts}{racines} --- le plus
souvent du sol.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:6}
L'arrosage. La \omterm{def:g1:plants-around-us:plant}{plante} s'est fanée faute d'eau~; quelques heures
après avoir été arrosée, ses \omterm{def:g1:plants-around-us:parts}{feuilles} se redresseront.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:7}
Pareil~: toutes deux sont des \omterm{def:g1:plants-around-us:plant}{plantes}, avec des \omterm{def:g1:plants-around-us:parts}{racines}, une \omterm{def:g1:plants-around-us:parts}{tige},
des \omterm{def:g1:plants-around-us:parts}{feuilles} et des \omterm{def:g1:plants-around-us:parts}{fleurs}, et toutes deux ont besoin d'eau et de
lumière. Différent~: la \omterm{def:g1:plants-around-us:parts}{tige} de l'\omterm{def:g1:plants-around-us:tree}{arbre} est un tronc de bois alors
que celle de la pâquerette est tendre et verte, et l'\omterm{def:g1:plants-around-us:tree}{arbre} est bien
plus grand et vit bien plus longtemps.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:8}
La \omterm{def:g1:plants-around-us:plant}{plante} de la fenêtre devrait être verte et droite~; celle du
placard pâle, étirée et faible, même arrosée. Le besoin qui manque
est la lumière --- l'eau seule ne suffit pas.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:9}
Par exemple~: la \omterm{def:g1:plants-around-us:plant}{plante} grandit --- plus haute de semaine en
semaine --- et elle se tourne ou penche vers la lumière, comme le
tournesol qui suit le jour. Lent ne veut pas dire immobile.
\end{solution}

\begin{solution}{exo:g1:plants-around-us:10}
La \omterm{def:g1:plants-around-us:parts}{racine}. «~Sous la terre~» est la place des \omterm{def:g1:plants-around-us:parts}{racines} dans la figure
--- la carotte que nous mangeons est une grosse \omterm{def:g1:plants-around-us:parts}{racine} orange que la
\omterm{def:g1:plants-around-us:plant}{plante} a remplie de nourriture.
\end{solution}
