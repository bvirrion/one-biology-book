\chapter{Growing Up}\label{ch:g2:growing}

You were once a baby. Now you are bigger and can do more. All
\omterm{def:g2:what-alive:organism}{organisms} \omterm{def:g2:what-alive:signs}{grow} and change. This chapter looks at growing up in
people, \omterm{def:g1:plants-animals:animal}{animals} and \omterm{def:g1:plants-animals:plant}{plants}.

\section{Growth}

\begin{definition}[Growth]\label{def:g2:growing:growth}
\emph{Growth}\index{growth} means becoming larger and developing
new skills or parts over time. Growth \omterm{def:g1:caring:needs}{needs} \omterm{def:g2:food-energy:food}{food} (or \omterm{def:g1:plants-animals:plant}{plant} \omterm{def:g2:food-energy:food}{food} from
light), \omterm{def:g1:caring:needs}{water} and time.
\end{definition}

\begin{example}[You]\label{ex:g2:growing:you}
As a baby you crawled. Later you walked and talked. Your \omterm{def:g1:my-body:parts}{hands} and
\omterm{def:g1:my-body:parts}{feet} grew. Your brain learned. Growing up is \omterm{def:g1:my-body:body}{body} and mind together.
\end{example}

\section{Young animals and plants}

\begin{definition}[Offspring]\label{def:g2:growing:offspring}
\emph{Offspring}\index{offspring} are the young of a \omterm{def:g1:living-nonliving:living}{living thing}:
babies, chicks, puppies, seedlings. Parents and young are the same
kind of \omterm{def:g2:what-alive:organism}{organism}.
\end{definition}

\begin{example}[Animal young]\label{ex:g2:growing:young}
A kitten \omterm{def:g2:what-alive:signs}{grows} into a cat. A tadpole \omterm{def:g2:what-alive:signs}{grows} into a frog (it changes
shape a lot). A chick \omterm{def:g2:what-alive:signs}{grows} feathers and learns to fly. Some young
look like small adults; some look different at first.
\end{example}

\begin{example}[Plant young]\label{ex:g2:growing:seedling}
A \omterm{def:g2:plant-parts:main}{seed} sprouts into a seedling. The seedling \omterm{def:g2:what-alive:signs}{grows} a stronger \omterm{def:g2:plant-parts:main}{stem}
and more \omterm{def:g2:plant-parts:main}{leaves}. One day it may \omterm{def:g2:plant-parts:main}{flower} and make its own \omterm{def:g2:plant-parts:main}{seeds} ---
the cycle continues (see Grade~3 life cycles).
\end{example}

\begin{method}[Watch growth]\label{met:g2:growing:watch}
\begin{enumerate}
  \item Measure height (yours, a \omterm{def:g1:plants-animals:plant}{plant}, or a pet) once a week.
  \item Write the number and the date.
  \item After a month, compare: what changed?
\end{enumerate}
Numbers make \omterm{def:g2:growing:growth}{growth} visible.
\end{method}
