\chapter{A célula: a unidade comum da vida}\label{ch:g10:cells-common-unit}

Raspe o interior da sua bochecha com um cotonete, espalhe o material
numa lâmina, acrescente uma gota de corante azul e olhe ao microscópio
com quatrocentas vezes de \omterm{def:g10:cells-common-unit:magnification}{aumento}: um punhado de ladrilhos achatados e
irregulares, cada um com uma mancha mais escura perto do centro.
Descole a película fina do interior de uma escama de cebola e olhe de
novo: tijolos retangulares bem arrumados, cada um com a mancha dele.
A água de uma lagoa mostra criaturas ovais remando com uma franja de
pelos. As três amostras não têm nada em comum a olho nu. Sob a lente,
elas são construídas com o mesmo objeto --- a \omterm{def:g10:cells-common-unit:cell}{célula} --- e este capítulo
trata do que esse objeto é, de como ele é visto e de por que todo ser
vivo é feito dele.

\section{A teoria celular}

\begin{definition}[Célula]\label{def:g10:cells-common-unit:cell}
Uma \emph{célula}\index{célula} é a menor unidade de um ser vivo que
está ela própria viva: um volume de conteúdo aquoso, o
\emph{citoplasma}\index{citoplasma}, delimitado por uma fina
\emph{membrana plasmática}\index{membrana plasmática} que a separa do
que a cerca e ao mesmo tempo deixa passar substâncias selecionadas. Uma
célula capta matéria e energia, as transforma, cresce e se divide em
duas células.
\end{definition}

\begin{proposition}[A teoria celular]\label{prop:g10:cells-common-unit:theory}
Três enunciados, estabelecidos pela microscopia no século XIX, formam a
teoria celular:
\begin{enumerate}
  \item todo ser vivo é feito de uma ou de várias \omterm{def:g10:cells-common-unit:cell}{células};
  \item a \omterm{def:g10:cells-common-unit:cell}{célula} é a unidade básica de estrutura e de funcionamento dos
        seres vivos --- nada menor está vivo por conta própria;
  \item toda \omterm{def:g10:cells-common-unit:cell}{célula} vem de uma \omterm{def:g10:cells-common-unit:cell}{célula} preexistente, por divisão.
\end{enumerate}
\end{proposition}
\begin{proof}[Evidências]
O enunciado 1 se apoia no exame sistemático de vegetais (Schleiden,
1838) e de animais (Schwann, 1839) com microscópios melhorados: todo
tecido examinado, por mais diferente que parecesse, se resolvia em
\omterm{def:g10:cells-common-unit:cell}{células}. O enunciado 3 foi estabelecido por Virchow (1855) e pela
observação de \omterm{def:g10:cells-common-unit:cell}{células} em divisão em tecidos em crescimento; a
alternativa --- \omterm{def:g10:cells-common-unit:cell}{células} se formando espontaneamente a partir de um
líquido --- nunca foi observada, e os experimentos de Pasteur sobre os
micróbios (o \cref{ch:g11:antibiotic-resistance} volta a eles) fecharam
a questão. O enunciado 2 decorre do primeiro e do fato de que \omterm{def:g10:cells-common-unit:cell}{células}
isoladas, mantidas num líquido adequado, continuam vivendo, enquanto
nenhum fragmento de \omterm{def:g10:cells-common-unit:cell}{célula} continua.
\end{proof}

\begin{example}[Uma célula ou muitas]\label{ex:g10:cells-common-unit:onemany}
Uma levedura, um paramécio ou uma bactéria é uma única \omterm{def:g10:cells-common-unit:cell}{célula} que faz
tudo --- se alimenta, se move, se divide. Um carvalho ou um ser humano é
uma cooperativa de \omterm{def:g10:cells-common-unit:cell}{células} de muitos tipos, cada uma cumprindo uma
parte do trabalho: um corpo humano contém cerca de $\num{3e13}$
\omterm{def:g10:cells-common-unit:cell}{células}, de aproximadamente duzentos tipos. Entre os dois extremos
estão as colônias e os organismos pluricelulares simples, mas não
existe ser vivo construído com outra coisa que não \omterm{def:g10:cells-common-unit:cell}{células}.
\end{example}

\section{Ver as células: escalas e microscópios}

\begin{definition}[Aumento e resolução]\label{def:g10:cells-common-unit:magnification}
O \emph{aumento}\index{aumento} de um instrumento é a razão entre o
tamanho da imagem e o tamanho do objeto. A
\emph{resolução}\index{resolução} dele é a menor distância entre dois
pontos que ele ainda mostra separados. Ampliar uma imagem além do que a
resolução justifica aumenta o borrão, não o detalhe.
\end{definition}

\begin{proposition}[Dois instrumentos, duas escalas]\label{prop:g10:cells-common-unit:microscopes}
O microscópio ótico aumenta até cerca de $\times 1500$ com uma
\omterm{def:g10:cells-common-unit:magnification}{resolução} de cerca de \qty{0.2}{\micro m}: ele mostra \omterm{def:g10:cells-common-unit:cell}{células}, \omterm{def:g10:cells-common-unit:organelle}{núcleos},
\omterm{def:g10:cells-common-unit:organelle}{cloroplastos} e bactérias, em cores e vivos. O microscópio eletrônico
usa um feixe de elétrons em vez de luz: \omterm{def:g10:cells-common-unit:magnification}{resolução} abaixo de
\qty{1}{nm}, \omterm{def:g10:cells-common-unit:magnification}{aumento} até $\times 10^6$; ele revela a estrutura interna
da \omterm{def:g10:cells-common-unit:cell}{célula} --- membranas, \omterm{def:g10:cells-common-unit:organelle}{mitocôndrias}, \omterm{def:g10:cells-common-unit:organelle}{ribossomos}, até moléculas
grandes ---, mas numa amostra que está morta, seca e cortada em fatias
mais finas que um micrômetro.
\end{proposition}
\admitted

\begin{method}[Medir um objeto numa micrografia]\label{met:g10:cells-common-unit:measure}
Uma micrografia traz ou um \omterm{def:g10:cells-common-unit:magnification}{aumento} ou uma barra de escala.
\begin{enumerate}
  \item Com uma barra de escala: meça a barra na página (digamos
        \qty{12}{mm} para uma barra rotulada \qty{10}{\micro m}) e o
        objeto (\qty{30}{mm}); o tamanho real é $30 \times 10/12 =
        \qty{25}{\micro m}$.
  \item Com um \omterm{def:g10:cells-common-unit:magnification}{aumento}: tamanho real $=$ tamanho medido dividido pelo
        \omterm{def:g10:cells-common-unit:magnification}{aumento}; uma \omterm{def:g10:cells-common-unit:cell}{célula} desenhada com \qty{20}{mm} de comprimento a
        $\times 400$ tem $20/400 = \qty{0.05}{mm} = \qty{50}{\micro m}$.
  \item Enuncie sempre a unidade, e converta: $\qty{1}{mm} =
        \qty{1000}{\micro m}$, $\qty{1}{\micro m} = \qty{1000}{nm}$.
\end{enumerate}
\end{method}

\begin{example}[Células da bochecha]\label{ex:g10:cells-common-unit:cheek}
Uma \omterm{def:g10:cells-common-unit:cell}{célula} da bochecha desenhada com \qty{24}{mm} de largura a partir
de uma imagem a $\times 400$ tem $24/400 = \qty{0.06}{mm} = \qty{60}{\micro m}$
de largura, e o \omterm{def:g10:cells-common-unit:organelle}{núcleo} dela, desenhado com \qty{3}{mm}, tem cerca de
\qty{8}{\micro m}. Uma hemácia tem \qty{7}{\micro m}; as bactérias da
boca, de \qty{1}{\micro m} de comprimento, aparecem como pontos mal
resolvidos no mesmo \omterm{def:g10:cells-common-unit:magnification}{aumento}.
\end{example}

\section{Dentro da célula}

\begin{definition}[Organela]\label{def:g10:cells-common-unit:organelle}
Uma \emph{organela}\index{organela} é uma estrutura de dentro de uma
\omterm{def:g10:cells-common-unit:cell}{célula}, delimitada por uma membrana própria, que cumpre uma tarefa
específica. As principais são:
\begin{itemize}
  \item o \emph{núcleo}\index{núcleo}, que contém os cromossomos --- o
        DNA do \cref{ch:g10:universal-dna} --- e comanda a atividade da
        \omterm{def:g10:cells-common-unit:cell}{célula};
  \item as \emph{mitocôndrias}\index{mitocôndria}, de cerca de
        \qty{1}{\micro m} de comprimento, onde moléculas orgânicas são
        degradadas com oxigênio para liberar energia utilizável (a
        respiração celular, \cref{ch:g10:cell-metabolism});
  \item nas \omterm{def:g10:cells-common-unit:cell}{células} vegetais, os \emph{cloroplastos}\index{cloroplasto},
        verdes de clorofila, onde acontece a fotossíntese, e um grande
        \emph{vacúolo}\index{vacúolo} cheio de água que mantém a \omterm{def:g10:cells-common-unit:cell}{célula}
        túrgida.
\end{itemize}
Do lado de fora da membrana, uma \omterm{def:g10:cells-common-unit:cell}{célula} vegetal fica encaixotada numa
rígida \emph{parede celular}\index{parede celular} de celulose; as
\omterm{def:g10:cells-common-unit:cell}{células} animais não têm nenhuma. O \omterm{def:g10:cells-common-unit:cell}{citoplasma} de toda \omterm{def:g10:cells-common-unit:cell}{célula} também
contém milhões de \emph{ribossomos}\index{ribossomo}, partículas de
cerca de \qty{25}{nm} que montam as \omterm{def:g10:chemistry-of-life:families}{proteínas}, visíveis só ao
microscópio eletrônico.
\end{definition}

\begin{definition}[Procariontes e eucariontes]\label{def:g10:cells-common-unit:prokaryote}
Uma \omterm{def:g10:cells-common-unit:cell}{célula} cujo DNA está encerrado num \omterm{def:g10:cells-common-unit:organelle}{núcleo} é uma
\emph{célula eucariótica}\index{célula eucariótica}: animais, vegetais,
fungos e muitos organismos unicelulares são eucariontes. Uma
\emph{célula procariótica}\index{célula procariótica} não tem \omterm{def:g10:cells-common-unit:organelle}{núcleo}
nem \omterm{def:g10:cells-common-unit:organelle}{organelas} delimitadas por membrana: o DNA dela fica livre no
\omterm{def:g10:cells-common-unit:cell}{citoplasma}, o tamanho dela é de cerca de \qty{1}{\micro m}, e ela é
cercada por uma parede de tipo próprio. As bactérias são procariontes.
\end{definition}

\begin{example}[Lendo uma bactéria]\label{ex:g10:cells-common-unit:bacterium}
Uma micrografia eletrônica da bactéria intestinal \emph{Escherichia
coli} mostra um bastonete de \qty{2}{\micro m} de comprimento e
\qty{0.5}{\micro m} de largura: uma parede, uma membrana logo por
dentro dela, um \omterm{def:g10:cells-common-unit:cell}{citoplasma} granuloso abarrotado de \omterm{def:g10:cells-common-unit:organelle}{ribossomos} e uma
região central mais pálida onde fica a única molécula circular de DNA.
Nenhum \omterm{def:g10:cells-common-unit:organelle}{núcleo}, nenhuma \omterm{def:g10:cells-common-unit:organelle}{mitocôndria}. E no entanto a \omterm{def:g10:cells-common-unit:cell}{célula} se alimenta,
cresce e se divide a cada vinte minutos num caldo rico: tudo o que a
teoria celular exige, num milésimo do volume de uma \omterm{def:g10:cells-common-unit:cell}{célula} de bochecha.
\end{example}

\section{Muitas células, muitas formas: a especialização}

\begin{proposition}[Células especializadas]\label{prop:g10:cells-common-unit:specialised}
Num organismo pluricelular as \omterm{def:g10:cells-common-unit:cell}{células} compartilham o mesmo plano básico
e o mesmo DNA, mas cada tipo tem uma forma e um equipamento ajustados à
tarefa dele: um neurônio estende uma fibra de até um metro para
conduzir sinais; uma hemácia perdeu o \omterm{def:g10:cells-common-unit:organelle}{núcleo} e é um disco achatado
abarrotado de hemoglobina; uma fibra muscular é uma \omterm{def:g10:cells-common-unit:cell}{célula} gigante
cheia de \omterm{def:g10:chemistry-of-life:families}{proteínas} contráteis; um pelo absorvente é um tubo fino que
maximiza a superfície para absorver água; uma célula-guarda de uma
folha muda de forma para abrir ou fechar um poro. A especialização é
uma questão de quais partes do plano comum são desenvolvidas.
\end{proposition}
\admitted

\begin{example}[Superfície e volume]\label{ex:g10:cells-common-unit:surface}
Por que as \omterm{def:g10:cells-common-unit:cell}{células} são tão pequenas? Tudo o que uma \omterm{def:g10:cells-common-unit:cell}{célula} capta ou
elimina atravessa a membrana dela, cuja área cresce com o quadrado do
tamanho, enquanto as necessidades crescem com o volume, o cubo. Uma
\omterm{def:g10:cells-common-unit:cell}{célula} cúbica de aresta \qty{10}{\micro m} tem uma superfície de
\qty{600}{\micro m^2} para um volume de \qty{1000}{\micro m^3}:
\qty{0.6}{\micro m^2} de membrana por micrômetro cúbico. Com aresta de
\qty{100}{\micro m} a razão cai para \qty{0.06}{\micro m^2}: dez vezes
menos membrana por unidade de conteúdo. Uma \omterm{def:g10:cells-common-unit:cell}{célula} grande passaria fome
no centro dela; as \omterm{def:g10:cells-common-unit:cell}{células} que são grandes (um óvulo de rã, uma fibra
muscular) ou são sobretudo reserva inerte, ou são alongadas para que
nenhum ponto fique longe da superfície.
\end{example}

\includegraphics[width=0.62\linewidth]{images/book2/ai/g10-cells-paramecium.png}

{\small Uma gota de água de lagoa ao microscópio ótico: um paramécio,
uma única \omterm{def:g10:cells-common-unit:cell}{célula} de um quarto de milímetro de comprimento, rema entre
algas verdes com a batida de milhares de pelos minúsculos. Uma \omterm{def:g10:cells-common-unit:cell}{célula},

\end{omfigure}

\begin{remark}[A célula como sistema aberto]\label{rem:g10:cells-common-unit:open}
Uma \omterm{def:g10:cells-common-unit:cell}{célula} não é um saco lacrado. Água, oxigênio, glicose e íons
atravessam a membrana dela continuamente, e os resíduos dela saem pelo
mesmo caminho; uma \omterm{def:g10:cells-common-unit:cell}{célula} de bochecha colocada em água pura incha e
estoura em poucos minutos, e uma colocada em água muito salgada
murcha. O controle que a membrana exerce sobre o que entra e o que sai
é assunto do volume do 1\textsuperscript{o} ano da graduação; por ora,
lembre que uma \omterm{def:g10:cells-common-unit:cell}{célula} vive trocando com o que a cerca, e morre quando
as trocas param.
\end{remark}

\section{Problema: contar as células de um corpo}

\begin{problem}[{Problema de fim de semana --- uma sessão de microscopia,
da película de cebola na lâmina a uma estimativa do número de células de
um corpo humano}]
\label{pb:g10:cells-common-unit:1}
Uma turma passa uma tarde ao microscópio. As objetivas disponíveis dão
\omterm{def:g10:cells-common-unit:magnification}{aumentos} totais de $\times 40$, $\times 100$ e $\times 400$; o campo de
visão a $\times 100$ tem \qty{1.8}{mm} de largura.

\medskip
\noindent\textbf{Parte I --- A cebola.}
\begin{enumerate}
  \item A $\times 100$, cerca de 9 \omterm{def:g10:cells-common-unit:cell}{células} de cebola cabem lado a lado
        na largura do campo de visão. Estime a largura de uma \omterm{def:g10:cells-common-unit:cell}{célula},
        em micrômetros.
  \item A $\times 400$ o campo de visão é quatro vezes mais estreito.
        Qual é a largura dele, e quantas dessas \omterm{def:g10:cells-common-unit:cell}{células} cabem nela?
  \item Um aluno desenha uma \omterm{def:g10:cells-common-unit:cell}{célula} com \qty{28}{mm} de largura a
        partir da imagem a $\times 400$. Qual é o \omterm{def:g10:cells-common-unit:magnification}{aumento} do desenho em
        relação à \omterm{def:g10:cells-common-unit:cell}{célula} real?
  \item O iodo tinge o \omterm{def:g10:cells-common-unit:organelle}{núcleo} de amarelo-castanho; o aluno o mede com
        \qty{4}{mm} no desenho. Diâmetro real?
  \item Cite as estruturas que o aluno deveria rotular no desenho, e
        uma estrutura presente que não pode ser vista nesse \omterm{def:g10:cells-common-unit:magnification}{aumento}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A bochecha e a lagoa.}
\begin{enumerate}[resume]
  \item As \omterm{def:g10:cells-common-unit:cell}{células} da bochecha coradas com azul de metileno aparecem
        como polígonos achatados de cerca de \qty{60}{\micro m} de
        largura, mas com apenas \qty{5}{\micro m} de espessura. Por que
        elas são achatadas, dado o lugar de onde vêm?
  \item Uma gota de água de lagoa mostra um paramécio de
        \qty{240}{\micro m} de comprimento. Ele pode ser visto a olho
        nu? Em que \omterm{def:g10:cells-common-unit:magnification}{aumento} ele preenche um quarto do campo?
  \item Pontinhos móveis de cerca de \qty{1}{\micro m} de comprimento
        são visíveis perto do paramécio a $\times 400$. O que são eles,
        muito provavelmente, e o que um microscópio eletrônico
        acrescentaria?
  \item O paramécio é uma única \omterm{def:g10:cells-common-unit:cell}{célula}; a \omterm{def:g10:cells-common-unit:cell}{célula} da bochecha é uma
        entre trilhões. Numa frase para cada, diga o que
        ``especializada'' quer dizer para a \omterm{def:g10:cells-common-unit:cell}{célula} da bochecha e por
        que isso não pode se aplicar ao paramécio.
  \item Um aluno acrescenta uma gota de água salgada às \omterm{def:g10:cells-common-unit:cell}{células} da
        bochecha e as vê murchar. Explique com a
        \cref{rem:g10:cells-common-unit:open}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O tamanho de uma \omterm{def:g10:cells-common-unit:cell}{célula}.}
\begin{enumerate}[resume]
  \item Tome uma \omterm{def:g10:cells-common-unit:cell}{célula} humana típica como um cubo de aresta
        \qty{20}{\micro m}. Calcule o volume dela em micrômetros
        cúbicos, depois em milímetros cúbicos
        ($\qty{1}{mm^3} = 10^9\,\unit{\micro m^3}$).
  \item Calcule a área da superfície dela e a razão superfície/volume.
  \item Repita para uma bactéria tomada como um cubo de aresta
        \qty{1}{\micro m}. Por que fator a razão dela é maior?
  \item As \omterm{def:g10:cells-common-unit:cell}{células} têm uma massa aproximadamente igual à de um volume
        igual de água ($\qty{1}{g}$ por \unit{cm^3}). Estime a massa da
        \omterm{def:g10:cells-common-unit:cell}{célula} de \qty{20}{\micro m} em nanogramas
        ($\qty{1}{g} = 10^9\,\unit{ng}$).
  \item Uma hemácia é um disco de \qty{7}{\micro m} de diâmetro e
        \qty{2}{\micro m} de espessura. Estime o volume dela (trate-a
        como um cilindro, $V = \pi r^2 h$) e compare com o cubo da
        questão 11.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O número de \omterm{def:g10:cells-common-unit:cell}{células} de um corpo.}
\begin{enumerate}[resume]
  \item Um aluno de massa \qty{60}{kg} tem um volume de cerca de
        \qty{60}{L}. Se o corpo fosse feito só de \omterm{def:g10:cells-common-unit:cell}{células} cúbicas de
        \qty{20}{\micro m}, quantas ele conteria?
  \item De fato, cerca de um terço do volume do corpo é líquido entre
        as \omterm{def:g10:cells-common-unit:cell}{células} e outro material não celular. Corrija a estimativa.
  \item Só as hemácias somam cerca de $\num{2.5e13}$. Usando o volume
        da questão 15, que volume elas ocupam? Isso é compatível com um
        volume de sangue de cerca de \qty{4.5}{L} do qual 45\% são
        \omterm{def:g10:cells-common-unit:cell}{células}?
  \item Combine as questões 17 e 18: dê uma ordem de grandeza para o
        número total de \omterm{def:g10:cells-common-unit:cell}{células} do corpo, e diga por que uma contagem
        precisa é impossível.
  \item Enuncie o resultado: quantas \omterm{def:g10:cells-common-unit:cell}{células}, grosso modo, formam um
        ser humano, e por qual processo --- a partir de que número
        inicial --- todas elas foram produzidas?
\end{enumerate}
\end{problem}
