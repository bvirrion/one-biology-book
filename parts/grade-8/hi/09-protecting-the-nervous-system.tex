\chapter{तंत्रिका तंत्र की हिफ़ाज़त}\label{ch:g8:protecting-the-nervous-system}

दो अध्याय दिखा चुके हैं कि \omterm{def:g5:brain-in-command:system}{तंत्रिका तंत्र} है क्या: ऐसे तार, जिनके जोड़
सीखते हैं, फ़ैसला करते हैं --- और ठगे भी जा सकते हैं
(\cref{rem:g8:nervous-communication:vulnerable})। इस साल का समापन उसी
शरीर-रचना को हिफ़ाज़त में बदल देता है। \omterm{prop:g1:healthy-habits:needs}{नींद} की कमी, शोर, परदे और मन पर
असर करने वाले पदार्थ, उस मशीन के साथ --- जो तुम ख़ुद हो --- जोड़-दर-जोड़
असल में करते क्या हैं; और चौदह की उम्र में, अमल में, उसकी हिफ़ाज़त दिखती
कैसी है?

\section{रखरखाव: नींद, और इंद्रियों की हदें}

\begin{proposition}[नींद जोड़ों का रखरखाव है]\label{prop:g8:protecting-the-nervous-system:sleep}
\cref{prop:g5:brain-in-command:protect} वाली फ़ाइलिंग-और-मरम्मत का एक
जोड़ के दर्जे वाला पाठ भी है: \omterm{prop:g1:healthy-habits:needs}{नींद} में दिन भर मज़बूत हुई
\omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियाँ} पक्की की जाती हैं --- यानी ``प्रैक्टिस पक्का कर देती
है'' (\cref{ex:g8:nervous-communication:juggler}) बड़ी हद तक रात को ही
होता है --- और \omterm{def:g5:brain-in-command:system}{मस्तिष्क} की सफ़ाई दिन भर की रासायनिक घिसाई हटा देती है।
छोटी रातें कल का सीखा आधा दर्ज छोड़ देती हैं और कल का तौलना सुस्त कर
देती हैं: और \omterm{def:g3:stages-of-life:stages}{किशोर} \omterm{def:g5:brain-in-command:system}{मस्तिष्क}, जो बीच जुड़ाई में है
(\cref{rem:g8:puberty:moods}) तथा जिसकी घड़ी देर की ओर खिसकी हुई है,
उसे अपने घंटे ठीक तभी सबसे ज़्यादा चाहिए जब स्कूल सबसे जल्दी शुरू होता
है।
\end{proposition}
\admitted

\begin{example}[दोहराई की उलझन]\label{ex:g8:protecting-the-nervous-system:revision}
इम्तिहान से पहले दो विद्यार्थी: एक आधी रात तक, और फिर दो बजे तक दोहराता
है; दूसरा दस बजे तक दोहराकर सो जाता है। आम तौर पर दूसरा जीतता है ---
क्योंकि सामग्री उसके जोड़ों को पार भी कर गई \emph{और} पक्की भी हो गई;
जबकि पहले के फ़ालतू घंटों ने ठीक वही फ़ाइलिंग गँवा दी जो दोहराई को
चिपकाती है। ``सोकर सोचो'' दरअसल \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियों} की इंजीनियरी है।
\end{example}

\begin{proposition}[ज़ोर की आवाज़ एक ख़ुराक है]\label{prop:g8:protecting-the-nervous-system:noise}
\omterm{def:g1:the-five-senses:sense}{कान} की ख़बर वाली \omterm{def:g5:first-look-at-cells:cell}{कोशिकाएँ} --- यानी वे बाल जैसे महीन पाने वाले, जो आवाज़
को \omterm{def:g8:nervous-communication:neuron}{तंत्रिका-संदेशों} में बदलते हैं --- दोबारा नहीं उगतीं
(\cref{prop:g1:the-five-senses:fragile} ने यह नरमी से कहा था; यहाँ उसकी
कार्य-प्रणाली है)। तेज़ आवाज़ उन्हें ख़ुराक के हिसाब से मारती है: यानी
आवाज़ का ज़ोर गुणा उसका समय। पूरे ज़ोर पर बजते समारोह और \omterm{def:g1:the-five-senses:sense}{कान} के छोटे
स्पीकर, घंटे-दर-घंटे, पाने वालों की वह पूँजी ख़र्च करते हैं जिसे कोई
\omterm{prop:g1:healthy-habits:needs}{नींद} वापस नहीं लाती --- और किसी शोर भरी रात के बाद की वह सीटी
(\cref{ex:g1:the-five-senses:loud}) ज़्यादा ख़ुराक की चेतावनी है। नियम
एक बजट है: कम, कम देर, और बीच में ठहराव --- और जहाँ ऐंप्लीफ़ायरों का
राज हो वहाँ \omterm{def:g1:the-five-senses:sense}{कान} के ठेपे।
\end{proposition}
\admitted

\section{जोड़ों को ठगने वाले पदार्थ}

\begin{proposition}[मन पर असर करने वाले पदार्थ तंत्रिका-संधियों पर काम करते हैं]\label{prop:g8:protecting-the-nervous-system:substances}
जो भी पदार्थ मिज़ाज, बोध या चौकसी बदलता है वह
\cref{def:g8:nervous-communication:synapse} वाली ख़ाली जगहों पर ही काम
करता है --- किसी संदेशवाहक की नक़ल करके, किसी को रोककर, या किसी का छूटना
ज़बरदस्ती करवाकर:
\begin{itemize}
  \item \emph{शराब} पूरे तंत्र के जोड़ों को दबा देती है: फंदा सिरे से
        सिरे तक धीमा पड़ जाता है --- समझ-बूझ, \omterm{def:g5:brain-in-command:reaction}{प्रतिक्रिया काल}
        (\cref{ex:g5:brain-in-command:driver} वाले मीटर लंबे हो जाते
        हैं), संतुलन, याददाश्त; और \omterm{def:g3:stages-of-life:stages}{किशोर} \omterm{def:g5:brain-in-command:system}{मस्तिष्क} के अभी बन रहे जोड़ यह
        दबाव सबसे गहरा झेलते हैं;
  \item \emph{तंबाकू का निकोटीन} एक क़ुदरती संदेशवाहक की नक़ल करता है,
        और बार-बार लेने पर इनाम वाले जोड़ों को \emph{\omterm{prop:g5:healthy-choices:substances}{लत}} की ओर नए सिरे
        से जोड़ देता है (\cref{prop:g5:healthy-choices:substances} वाला
        जाल, अब अपनी कार्य-प्रणाली के साथ: \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियाँ} उस ख़ुराक
        की उम्मीद करने लगती हैं --- और उसके बिना बग़ावत करती हैं);
  \item \emph{गाँजा} संदेशवाहकों के एक और कुल की नक़ल करता है, और ठीक
        याददाश्त, प्रेरणा तथा समय के जोड़ों को गँदला कर देता है ---
        और यह भी, फिर से, बीच उसारी वाले \omterm{def:g5:brain-in-command:system}{मस्तिष्क} में सबसे महँगा पड़ता
        है;
  \item और पदार्थ जितना कड़ा हो, जोड़ों पर उसकी ज़बरदस्ती उतनी ही
        भोंड़ी --- तथा \omterm{prop:g5:healthy-choices:substances}{लत} का इंजन बनी वह इनाम वाली जुड़ाई उतनी ही
        तेज़।
\end{itemize}
\end{proposition}
\admitted

\begin{remark}[लत, कार्य-प्रणाली के हिसाब से]\label{rem:g8:protecting-the-nervous-system:addiction}
\Cref{rem:g5:healthy-choices:never} वाले जाल का अब एक तार-नक़्शा भी है:
इनाम वाले जोड़, बार-बार ख़ुराक पाकर, \emph{ढल} जाते हैं --- ग्राही छँट
जाते हैं, दहलीज़ें ऊँची हो जाती हैं --- यहाँ तक कि सामान्य महसूस करने के
लिए ही वह पदार्थ चाहिए होता है, और उसका न होना पूरे तंत्र की बग़ावत बन
जाता है। यही ढलना वह ``घुसना आसान, निकलना मुश्किल'' वाला बेमेल है, जो
\omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधियों} से बना है: शुरू करना एक फ़ैसला है; रुकना नए सिरे से जुड़े
जोड़ों के ख़िलाफ़ लड़ाई। और अभी सधने लायक़ \omterm{def:g3:stages-of-life:stages}{किशोर} \omterm{def:g5:brain-in-command:system}{मस्तिष्क} सबसे तेज़ी से
जुड़ता है --- इसीलिए हर आँकड़ा जल्दी शुरू करने वालों को सबसे गहरा फँसा
पाता है, और इसीलिए ``शुरू ही मत करो'' सबसे सस्ता क़दम बना रहता है।
\end{remark}

\begin{example}[परदे का फंदा]\label{ex:g8:protecting-the-nervous-system:screens}
कोई अणु नहीं, पर जोड़ वही: फ़ीड और खेल जान-बूझकर ऐसे सधे होते हैं कि
इनाम वाली जुड़ाई को अनजाने छोटे-छोटे इनामों से गुदगुदाते रहें --- यानी
वही समय-सारणी, जो जोड़ों को सबसे कड़ाई से सिखाती है। नतीजा उसी जाल का एक
हलका रिश्तेदार है: बार-बार देखना, जो रुकने का विरोध करता है; शामें, जो
ग़ायब हो जाती हैं; और \omterm{prop:g1:healthy-habits:needs}{नींद}, जो देर की ओर धकेल दी जाती है
(\cref{prop:g5:healthy-choices:screens} वाले घंटे, और साथ में
\cref{prop:g8:protecting-the-nervous-system:sleep} वाली क़ीमत)। बचाव
वही पुराना: घंटे पहले से तय करो, और आख़िरी घंटा परदों से ख़ाली रखो।
\end{example}

\section{अमल: अपनी हिफ़ाज़त ख़ुद चलाना}

\begin{method}[तंत्रिका तंत्र के मालिक की पुस्तिका]\label{met:g8:protecting-the-nervous-system:manual}
\cref{met:g5:healthy-choices:run} वाली देखभाल की सूची, अब
कार्य-प्रणालियों के साथ दोबारा जारी:
\begin{enumerate}
  \item \emph{पूरे घंटे सोओ} --- पक्का करना और सफ़ाई, दोनों रात को
        चलते हैं; और आख़िरी घंटे को परदों से बचाओ;
  \item \emph{ज़ोर का बजट रखो} --- पाने वालों की पूँजी दोबारा नहीं
        उगती: आवाज़ नीची, बीच में ठहराव, और जहाँ गरजता हो वहाँ \omterm{def:g1:the-five-senses:sense}{कान} के
        ठेपे;
  \item जहाँ गिरना तेज़ हो वहाँ \emph{हेलमेट}
        (\cref{ex:g5:brain-in-command:helmet}) --- तार \omterm{def:g3:bones-and-muscles:skeleton}{हड्डी} की तरह
        जुड़ते नहीं;
  \item बीच उसारी वाले \omterm{def:g5:brain-in-command:system}{मस्तिष्क} के लिए \emph{कोई जोड़-ठग नहीं} ---
        और वह ``ना'' पहले से, ठंडे दिमाग़ से तय की हुई
        (\cref{ex:g5:healthy-choices:answers} वाला जेब में रखा वाक्य,
        अब अपने पूरे औचित्य के साथ);
  \item \emph{मशीन इस्तेमाल करो} --- पढ़ाई, खेल, संगीत: जोड़ आवाजाही से
        मज़बूत होते हैं, और सधा हुआ फंदा वह संपत्ति है जिसकी हिफ़ाज़त
        बाक़ी हर आदत करती है।
\end{enumerate}
\end{method}

\begin{example}[मदद मौजूद है, और काम करती है]\label{ex:g8:protecting-the-nervous-system:help}
जो कोई पहले से उलझा हो --- अपनी आदत में या किसी दोस्त की, पदार्थों में
या परदों में: \cref{ex:g8:contraception:help} वाले पते यहाँ भी काम आते
हैं। स्कूल की नर्सें और डॉक्टर \omterm{prop:g5:healthy-choices:substances}{लत} को ठीक वही मानकर चलते हैं जो यह अध्याय
दिखाता है --- यानी तारों की एक हालत, न कि चरित्र का कोई फ़ैसला --- और नए
सिरे से जुड़े जोड़ दोबारा ढल भी सकते हैं: शुरू न करने से मुश्किल, पर
नाउम्मीदी से बहुत दूर। और सबसे बुरी योजना वही मैदान वाली है: चुप्पी।
\end{example}

\section{समस्या: बचाव वाली शाम}

\begin{problem}[{सप्ताहांत समस्या --- स्कूल की अभिभावक-और-विद्यार्थी
शाम, कार्य-प्रणालियों के साथ लिखी
हुई}]\label{pb:g8:protecting-the-nervous-system:1}
कक्षा स्कूल की बचाव वाली शाम की तैयारी करती है: चार ठिकाने --- \omterm{prop:g1:healthy-habits:needs}{नींद},
आवाज़, पदार्थ, परदे --- और हर एक के साथ एक प्रदर्शन तथा एक
कार्य-प्रणाली वाला कार्ड।

\medskip\noindent\textbf{भाग I --- ठिकानों के कार्ड।}
\begin{enumerate}
  \item \omterm{prop:g1:healthy-habits:needs}{नींद} वाला कार्ड लिखो: प्रदर्शन में दो स्वयंसेवकों के
        पैमाना-परीक्षण के अंक हैं
        (\cref{met:g5:brain-in-command:ruler}), यानी आराम किया हुआ
        बनाम छोटी रात वाला। अंकों की भविष्यवाणी करो और जोड़ वाली
        कार्य-प्रणाली दो।
  \item आवाज़ वाला कार्ड लिखो: एक डेसिबल मापक, \omterm{def:g1:the-five-senses:sense}{कान} के ठेपों की एक
        जोड़ी, और वह वाक्य ``इन्हें दोबारा नहीं उगाया जा सकता''।
        कार्य-प्रणाली भरो।
  \item पदार्थ वाला कार्ड लिखो: किसी \omterm{def:g8:nervous-communication:synapse}{तंत्रिका-संधि} का एक चित्र, और
        उस पर नक़ल, रोक तथा ज़बरदस्ती के तीन तीर। शराब, निकोटीन और
        \cref{exo:g8:nervous-communication:11} वाले विष को उनके-उनके
        तीर सौंपो।
  \item परदे वाला कार्ड लिखो: कोई अणु नहीं --- तो फिर इस ठिकाने के
        सूचना-वाले-फ़ोन के प्रदर्शन में पदार्थ वाले ठिकाने से साझा
        क्या है?
\end{enumerate}

\medskip\noindent\textbf{भाग II --- सभा से आए सवाल।}
\begin{enumerate}[resume]
  \item एक अभिभावक: ``हमने भी इनकी उम्र में पी थी और हम तो ठीक ही
        निकले।'' जवाब बीच-उसारी वाले बेमेल से दो --- चौदह की उम्र में
        अलग क्या है?
  \item एक विद्यार्थी: ``एक सिगरेट भला क्या जोड़ बदल देगी।'' जवाब
        \cref{rem:g8:protecting-the-nervous-system:addiction} और
        \cref{rem:g5:healthy-choices:never} से दो --- पहली ख़ुराक की
        असली क़ीमत क्या है?
  \item एक अभिभावक: ``परदे तो बस नया टेलीविज़न ही हैं न?''
        समय-सारणियों का फ़र्क़ बताओ: अनजाने छोटे इनाम ज़्यादा कड़ा
        उस्ताद किस बात से बनते हैं?
  \item एक विद्यार्थी, धीरे से: ``और अगर किसी से रुका ही न जा रहा
        हो?'' \cref{ex:g8:protecting-the-nervous-system:help} वाला
        जवाब दो, पतों समेत।
\end{enumerate}

\medskip\noindent\textbf{भाग III --- समापन भाषण।}
\begin{enumerate}[resume]
  \item भाषण बाज़ीगर से खुलता है और \omterm{prop:g5:healthy-choices:substances}{लत} पर बंद होता है --- ``वही सधने
        की गुंजाइश, दो बार पढ़ी हुई''। वह तर्क चार वाक्यों में लिखो।
  \item फिर वह हर विद्यार्थी को \cref{ex:g5:healthy-choices:answers}
        वाला जेब में रखने का वाक्य थमाता है। कार्य-प्रणाली समेत
        समझाओ कि वह वाक्य \emph{अभी} क्यों गढ़ लेना चाहिए।
  \item शाम के पोस्टर पर मालिक की पुस्तिका की पाँचों मदें हैं।
        उन्हें चौदह साल के किसी बच्चे के हफ़्ते के लिए क्रम में लगाओ,
        सबसे ज़्यादा हिफ़ाज़त देने वाली पहले, और अपनी पहली पसंद का
        बचाव करो।
  \item शाम का समापन एक वाक्य में करो: हिफ़ाज़त किसकी हो रही है, और
        चौदह उस फ़ैसलाकुन दशक का बीच क्यों है?
\end{enumerate}
\end{problem}
