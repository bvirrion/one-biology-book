\chapter{Menggolongkan Dunia Hidup}\label{ch:b1:classifying-biodiversity}

Sekitar \num{2} juta \omterm{def:b1:classifying-biodiversity:species}{spesies} telah dinamai, dan tiap tahun sekitar
\num{18000} ditambahkan; cacah yang ada ditaksir mendekati \num{9} juta
bagi eukariotnya saja, dan tak ada yang tahu berapa banyak bakterinya.
Sebuah nama hanyalah permulaan: seorang ahli biologi ingin tahu
\omterm{def:b1:classifying-biodiversity:species}{spesies} mana yang paling dekat dengan mana, dan jawabannya adalah
sebuah pohon, sebab \omterm{def:b1:classifying-biodiversity:species}{spesiesnya} berkerabat lewat keturunan. Bab ini
memberikan aturan penamaan, cara hubungannya disimpulkan dari
cirinya --- serta kriterianya, yaitu \omterm{def:b1:classifying-biodiversity:parsimony}{parsimoni}, yang dengannya pohon
yang bersaing dinilai --- bentuk pohon kehidupan sebagaimana ia
dipahami sekarang, dan keadaan inventaris apa yang hidup.

\section{Spesies dan nama}

\begin{definition}[Spesies]\label{def:b1:classifying-biodiversity:species}
\index{spesies}\index{konsep spesies biologi}
Yang disebut \emph{spesies}, menurut \emph{konsep spesies biologi},
adalah sekelompok \omterm{def:b1:populations:population}{populasi} yang anggotanya dapat berbiak silang lalu
menghasilkan anak yang subur dan terasing secara pembiakan dari
kelompok semacam itu yang lain. Konsepnya tak dapat diterapkan pada
\omterm{def:b1:organism-environment:organism}{organisme} yang tidak berbiak secara seksual (sebagian besar bakteri,
banyak protista dan sebagian tumbuhan serta hewan), pada fosil, atau
pada \omterm{def:b1:populations:population}{populasi} yang tak pernah bertemu; di situ sebuah spesies dikenali
dari morfologinya (konsep \emph{morfologi}), dari garis keturunannya
yang khas pada sebuah pohon (konsep \emph{\omterm{def:b1:classifying-biodiversity:systematics}{filogeni}}), atau dari sebuah
ambang penyimpangan genetik. Seluruh konsepnya mencoba menamai hal
yang sama: sebuah garis keturunan yang berevolusi terpisah dari yang
lain.
\end{definition}

\begin{definition}[Tata nama dan peringkat]\label{def:b1:classifying-biodiversity:nomenclature}
\index{tata nama binomial}\index{takson}\index{genus}\index{spesimen tipe}
Tiap \omterm{def:b1:classifying-biodiversity:species}{spesies} memiliki sebuah nama \emph{binomial}: nama
\emph{genus}nya (berhuruf besar) diikuti sebuah julukan khas, keduanya
dalam huruf miring --- \emph{Homo sapiens}, \emph{Quercus robur},
\emph{Escherichia coli} --- sering diikuti penulis dan tahun
pemeriannya. Sistemnya, yang ditetapkan Linnaeus pada 1753 (tumbuhan)
dan 1758 (hewan), menyarangkan \omterm{def:b1:classifying-biodiversity:species}{spesies} dalam sebuah hierarki
\emph{peringkat}: genus, famili, ordo, kelas, filum (atau divisi),
kerajaan, domain. Sebuah kelompok pada peringkat mana pun adalah sebuah
\emph{takson}. Tiap namanya terpaut pada sebuah \emph{spesimen tipe},
yang disimpan dalam sebuah museum atau herbarium, yang menetapkan apa
yang diacu namanya; nama sah yang tertua memiliki keutamaan; dan
aturannya dibakukan sehingga satu \omterm{def:b1:organism-environment:organism}{organisme} memiliki satu nama di
seluruh dunia. Peringkatnya adalah kelaziman --- tak ada sifat yang
dimiliki tiap famili dan tak dimiliki sebuah genus; hanya \omterm{def:b1:classifying-biodiversity:species}{spesiesnya}
yang merupakan satuan alami.
\end{definition}

\section{Menyusun ulang hubungannya}

\begin{definition}[Sistematika, filogeni, kladistika]\label{def:b1:classifying-biodiversity:systematics}
\index{sistematika}\index{filogeni}\index{kladistika}\index{klade}
Yang disebut \emph{sistematika} adalah kajian keanekaragaman \omterm{def:b1:organism-environment:organism}{organisme}
dan hubungan di antaranya; sebuah \emph{filogeni} adalah pohon
hubungan itu, yaitu sebuah hipotesis tentang urutan garis keturunannya
terbelah. \emph{Fenetika} mengelompokkan \omterm{def:b1:organism-environment:organism}{organisme} menurut kemiripan
keseluruhannya, dengan menghitung seluruh cirinya sama rata;
\emph{kladistika} mengelompokkannya hanya menurut ciri turunan yang
dikongsi, sehingga tiap kelompoknya adalah sebuah \emph{klade} ---
seorang moyang dan seluruh keturunannya. Penggolongan modern
bertujuan membuat tiap \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} yang dinamai sebuah klade.
\end{definition}

\begin{definition}[Ciri: homologi, homoplasi, polaritas]\label{def:b1:classifying-biodiversity:characters}
\index{homologi}\index{homoplasi}\index{sinapomorfi}\index{kelompok luar}
Yang disebut \emph{ciri} adalah tiap sifat terwaris dengan dua
\emph{keadaan} atau lebih (tungkai hadir atau tidak; sebuah basa
tertentu dalam sebuah \omterm{def:b1:genomes:gene}{gen} A, C, G atau T). Ciri dua \omterm{def:b1:organism-environment:organism}{organisme} bersifat
\emph{homolog} ketika terwaris dari seorang moyang bersama,
\emph{homoplastik} ketika mirip lewat pemusatan atau pembalikan (sayap
burung dan kelelawar; tubuh ramping lumba-lumba dan hiu). Sebuah
keadaan ciri bersifat \emph{moyang} (plesiomorfik) bila hadir pada
moyang kelompoknya, \emph{turunan} (apomorfik) bila ia muncul di dalam
kelompoknya; keadaan turunan yang dikongsi sebuah \omterm{def:b1:classifying-biodiversity:systematics}{klade}, yaitu
\emph{sinapomorfi}nya, adalah satu-satunya bukti baginya. Polaritasnya
ditentukan lewat perbandingan dengan sebuah \emph{kelompok luar},
yaitu sebuah \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} yang diketahui terletak di luar kelompok yang
dikaji: keadaan yang diperlihatkannya diambil sebagai keadaan
moyangnya.
\end{definition}

\begin{definition}[Monofili, parafili, polifili]\label{def:b1:classifying-biodiversity:monophyly}
\index{monofili}\index{parafili}\index{polifili}
Sebuah kelompok bersifat \emph{monofiletik} (sebuah \omterm{def:b1:classifying-biodiversity:systematics}{klade}) bila ia
memuat seorang moyang dan seluruh keturunannya; \emph{parafiletik}
bila ia memuat seorang moyang dan hanya sebagian keturunannya
(``reptilia'', yang mengecualikan burung yang menurun darinya;
``ikan'', yang mengecualikan tetrapodanya; ``invertebrata'');
\emph{polifiletik} bila ia mengelompokkan keturunan moyang yang
berbeda lewat sebuah ciri yang memusat (``hewan berdarah panas'',
burung dan mamalia; ``alga''). Kelompok parafiletik berguna sebagai
perian jenjang penataan tetapi bukanlah \omterm{def:b1:classifying-biodiversity:systematics}{klade} dan tak lagi diberi nama
resmi.
\end{definition}

\begin{definition}[Parsimoni]\label{def:b1:classifying-biodiversity:parsimony}
\index{parsimoni}\index{panjang pohon}
Di antara seluruh pohon yang dapat menghubungkan seperangkat \omterm{def:b1:classifying-biodiversity:nomenclature}{takson},
yang \emph{paling parsimonius} adalah yang menuntut paling sedikit
perubahan evolusi keadaan cirinya --- \emph{panjang pohon} yang
terkecil, yaitu jumlah pada seluruh cirinya dari cacah minimum
perubahan yang diperlukan masing-masing pada pohon itu. Sebuah ciri
yang paskan dengan pohonnya lewat satu perubahan sesuai dengannya;
yang memerlukan dua atau lebih adalah sebuah \omterm{def:b1:classifying-biodiversity:characters}{homoplasi} pada pohon itu.
Parsimoni bukanlah klaim bahwa evolusi itu hemat; ia adalah aturan
bahwa sebuah hipotesis tidak boleh melipatgandakan pemusatan melampaui
apa yang dipaksakan datanya.
\end{definition}

\begin{method}[Membangun sebuah kladogram dari sebuah matriks ciri]\label{met:b1:classifying-biodiversity:cladogram}
\begin{enumerate}
  \item Pilihlah \omterm{def:b1:classifying-biodiversity:nomenclature}{taksonnya} dan sebuah \omterm{def:b1:classifying-biodiversity:characters}{kelompok luar}; daftarkan cirinya
        beserta keadaannya, dengan menyandikan keadaan \omterm{def:b1:classifying-biodiversity:characters}{kelompok
        luarnya} 0 (moyang) dan keadaan turunannya 1.
  \item Kelompokkan \omterm{def:b1:classifying-biodiversity:nomenclature}{taksonnya} menurut keadaan turunan yang dikongsi,
        mulai dari ciri yang dikongsi paling banyak \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} (\omterm{def:b1:classifying-biodiversity:systematics}{klade}
        terdalamnya) lalu menyarang ke dalam; tiap 1 yang dikongsi
        seperangkat \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} adalah sebuah calon \omterm{def:b1:classifying-biodiversity:characters}{sinapomorfi} bagi
        perangkat itu.
  \item Tempat cirinya berselisih (dua pengelompokan tak dapat
        keduanya menjadi \omterm{def:b1:classifying-biodiversity:systematics}{klade}), gambarlah tiap pohon alternatifnya
        lalu hitunglah panjangnya: bagi tiap cirinya, perubahan
        tersedikit yang menempatkan keadaannya pada pohonnya.
  \item Peganglah pohon terpendeknya; ciri yang memerlukan perubahan
        tambahan padanya adalah \omterm{def:b1:classifying-biodiversity:characters}{homoplasinya}. Bila dua pohon seri,
        datanya tidak memutuskan dan lebih banyak ciri dibutuhkan.
  \item Bacalah pohonnya: namailah \omterm{def:b1:classifying-biodiversity:systematics}{kladenya}, catatlah kelompok
        tradisional mana yang parafiletik, lalu periksalah bahwa ia
        sesuai dengan data bebas (urutan molekul, fosil).
\end{enumerate}
\end{method}

\begin{example}[Dua pohon, yang dihitung]\label{ex:b1:classifying-biodiversity:count}
Empat \omterm{def:b1:classifying-biodiversity:nomenclature}{takson}: kadal, burung, mamalia, katak (\omterm{def:b1:classifying-biodiversity:characters}{kelompok luar}). Cirinya:
telur amniotik (kadal, burung, mamalia: 1), tengkorak diapsid (kadal,
burung), endotermi (burung, mamalia). Pohon A, (kadal, burung) lalu
mamalia: telurnya 1 perubahan, tengkoraknya 1, endoterminya 2 (muncul
dua kali) --- panjangnya 4. Pohon B, (burung, mamalia) lalu kadal:
telurnya 1, endoterminya 1, tengkoraknya 2 --- panjangnya 4. Sebuah
seri; tambahkan bulu pada sisik kakinya dan sisik \omterm{def:b1:flowering-plant-organization:tissues}{epidermisnya} (kadal,
burung): pohon A panjangnya 5, pohon B panjangnya 6. Pohon A lebih
disukai, dan endoterminya adalah sebuah \omterm{def:b1:classifying-biodiversity:characters}{homoplasi}: ia berevolusi dua
kali.
\end{example}

\begin{remark}[Ciri molekul]\label{rem:b1:classifying-biodiversity:molecular}
Sebuah urutan \omterm{def:b1:genomes:gene}{gen} memasok ratusan ciri sekaligus, tiap tapaknya
sebuah ciri dengan empat keadaan, yang dapat dibandingkan pada
\omterm{def:b1:organism-environment:organism}{organisme} yang tidak berbagi sifat kasatmata mana pun --- sebuah
bakteri dan sebatang pohon ek. Urutannya disejajarkan, tapak
homolognya dibandingkan, dan pohonnya dibangun lewat \omterm{def:b1:classifying-biodiversity:parsimony}{parsimoni} atau
lewat cara statistik yang memodelkan laju penggantiannya; jam
molekulnya, yaitu laju perbedaan netralnya bertimbun, memberi tarikh
belahannya. Caranya, dan jebakan cabang panjang serta laju yang tak
setara, dibahas dalam jilid Tahun ke-2.
\end{remark}

\section{Pohon kehidupan}

\begin{proposition}[Tiga domain]\label{prop:b1:classifying-biodiversity:domains}
\index{domain}\index{arkea}\index{bakteri}\index{eukariot}
Kehidupan terbagi menjadi tiga domain, yaitu \emph{Bacteria},
\emph{Archaea} dan \emph{Eukarya}. Keduanya, yaitu Bacteria dan
Archaea, sama-sama prokariot dalam penataan \omterm{def:b1:cell-unit-of-life:cell}{selnya} (\cref{ch:b1:cell-unit-of-life}) tetapi berbeda dalam
\omterm{def:b1:lipids:fattyacid}{lipid} membran, \omterm{prop:b1:carbohydrates:wall}{dinding sel}, \omterm{def:b1:gene-expression:transcription}{RNA polimerase} dan \omterm{def:b1:gene-expression:ribosome}{ribosomnya}; mesin
transkripsi dan translasi Archaea lebih dekat pada mesin eukariotnya
daripada pada mesin bakterinya, dan \omterm{def:b1:cell-unit-of-life:prokeuk}{sel eukariotnya} muncul dari sebuah
garis keturunan arkea yang memperoleh sebuah bakteri sebagai
\omterm{def:b1:eukaryotic-cell:mitochondrion}{mitokondrianya} (\cref{ch:b1:eukaryotic-cell}). Di dalam Eukarya
kerajaan lamanya --- hewan, tumbuhan, jamur dan sebuah sisa
``protista'' --- diganti segelintir garis keturunan utama, yang
hewan, jamur dan tumbuhan daratnya adalah tiga ranting kecil, dan
``protista''nya sebuah kumpulan parafiletik segala yang lain.
\end{proposition}
\begin{proof}[Bukti]
Woese dan Fox (1977) membandingkan urutan \omterm{def:b1:nucleic-acids:rnas}{RNA ribosom} subunit
kecilnya --- sebuah molekul yang dimiliki tiap \omterm{def:b1:cell-unit-of-life:cell}{sel}, yang berfungsi
sama dan dapat disejajarkan pada seluruh kehidupan --- di antara
bakteri, metanogen dan eukariotnya, lalu menemukan bahwa metanogennya
berbeda dari bakteri yang lain sebanyak keduanya berbeda dari
eukariotnya. Prokariotnya adalah dua kelompok, bukan satu; pohon yang
digambar dari satu \omterm{def:b1:genomes:gene}{gen} ini telah disahkan oleh \omterm{def:b1:genomes:genome}{genom} utuh dan oleh
biokimia Archaea yang khas (\omterm{def:b1:lipids:fattyacid}{lipid} berpaut eter, tanpa \omterm{def:b1:carbohydrates:polysaccharide}{peptidoglikan}).
\end{proof}

\begin{proposition}[Garis keturunan eukariot yang utama]\label{prop:b1:classifying-biodiversity:eukaryotes}
Eukariotnya jatuh ke dalam beberapa kelompok besar:
\emph{opistokonta} (hewan, jamur dan kerabat bersel tunggalnya, yang
\omterm{def:b1:cell-unit-of-life:cell}{selnya} berenang dengan satu flagelum belakang); \emph{Archaeplastida}
(tumbuhan darat, alga hijau dan merah, dengan kloroplas primernya);
\emph{Amoebozoa}; \emph{Excavata} (flagelata seperti
\emph{Trypanosoma} dan \emph{Euglena}); kelompok \emph{SAR}
(stramenopil seperti diatom dan alga cokelat; alveolata seperti
siliata, dinoflagelata dan \omterm{def:b1:species-interactions:symbiosis}{parasit} malarianya; rhizaria seperti
foraminifera dan radiolaria). Karena itu hewan dan jamur lebih dekat
satu sama lain daripada masing-masing dengan tumbuhan; ``alga''
tersebar pada empat kelompok; dan mayoritas mikroskopis keanekaragaman
eukariotnya terletak di luar ketiga kerajaan yang bernama umum.
\end{proposition}

\section{Inventaris kehidupan}

\begin{definition}[Keanekaragaman hayati]\label{def:b1:classifying-biodiversity:biodiversity}
\index{keanekaragaman hayati}\index{kepunahan}
Yang disebut \emph{keanekaragaman hayati} adalah keragaman kehidupan
pada tiap arasnya: keanekaragaman genetik di dalam \omterm{def:b1:classifying-biodiversity:species}{spesies}, cacah dan
kelimpahan \omterm{def:b1:classifying-biodiversity:species}{spesiesnya}, serta keragaman \omterm{def:b1:ecosystem-organization:ecosystem}{komunitas} dan \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistemnya}.
Inventarisnya belum selesai: sekitar \num{2} juta \omterm{def:b1:classifying-biodiversity:species}{spesies} telah
diperikan, yang sejuta di antaranya serangga, \num{380000} tumbuhan,
\num{150000} jamur, \num{70000} vertebrata dan beberapa puluh ribu
masing-masing moluska, krustasea, protista dan prokariotnya; cacah
\omterm{def:b1:classifying-biodiversity:species}{spesies} eukariotnya ditaksir, dari laju \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} tinggi yang baru masih
ditemukan, mendekati \num{9} juta, sehingga sebagian besar \omterm{def:b1:classifying-biodiversity:species}{spesiesnya}
tak bernama, dan sebagian besar di antaranya kecil, tropis, atau
keduanya. \omterm{def:b1:classifying-biodiversity:species}{Spesies} selalu \emph{punah} --- laju latarnya, dari fosil,
sekitar satu \omterm{def:b1:classifying-biodiversity:species}{spesies} tiap sejuta tiap tahun --- dan laju sekarangnya,
yang dipacu kehilangan \omterm{def:b1:ecosystem-organization:niche}{habitat}, eksploitasi, \omterm{def:b1:classifying-biodiversity:species}{spesies} invasif dan
iklimnya, ditaksir seratus sampai seribu kali itu.
\end{definition}

\begin{method}[Menaksir berapa banyak spesies yang ada]\label{met:b1:classifying-biodiversity:estimate}
\begin{enumerate}
  \item Hitunglah \omterm{def:b1:classifying-biodiversity:species}{spesies} yang diperikan dalam sebuah kelompok dan
        laju yang baru masih ditambahkan; bila lajunya tidak melambat,
        inventarisnya jauh dari lengkap.
  \item Ekstrapolasikan dari sebuah kelompok yang dikenal baik ke
        sebuah kelompok yang dikenal buruk lewat sebuah nisbah: bila
        fauna iklim sedangnya memuat dua \omterm{def:b1:classifying-biodiversity:species}{spesies} serangga tiap \omterm{def:b1:classifying-biodiversity:species}{spesies}
        tumbuhan, dan tropisnya memiliki \num{250000} tumbuhan,
        tropisnya memuat sekurangnya setengah juta serangga.
  \item Cupliklah secara intensif sebuah luas kecil (mengasapi sebuah
        tajuk pohon tropis dengan insektisida, mengurutkan \omterm{def:b1:nucleic-acids:chain}{DNA} seliter
        air laut) lalu naikkan skalanya dari fraksi \omterm{def:b1:classifying-biodiversity:species}{spesies} yang baru.
  \item Pakailah pola pada seluruh peringkatnya: cacah filum, kelas,
        ordo dan familinya nyaris lengkap lalu naik teratur menuju aras
        \omterm{def:b1:classifying-biodiversity:species}{spesiesnya}; mengekstrapolasikan keteraturannya memberi totalnya
        (sekitar \num{8.7} juta eukariot).
  \item Laporkan taksirannya beserta jangkaunya; jawaban yang jujur
        bagi prokariotnya adalah sebuah jangkau beberapa orde besaran.
\end{enumerate}
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:classifying-biodiversity:1}
Tuliskan nama kucing rumah dengan benar, kenalilah \omterm{def:b1:classifying-biodiversity:nomenclature}{genus} dan
julukannya, lalu tempatkanlah ia dalam famili, ordo, kelas dan
filumnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:classifying-biodiversity:2}
Definisikan \omterm{def:b1:classifying-biodiversity:characters}{homologi} dan \omterm{def:b1:classifying-biodiversity:characters}{homoplasi} dengan sebuah contoh
masing-masing, lalu katakan yang mana yang menjadi bukti bagi sebuah
\omterm{def:b1:classifying-biodiversity:systematics}{klade}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:classifying-biodiversity:3}
Nyatakan apakah ``ikan'', ``burung'' dan ``hewan berdarah panas''
bersifat monofiletik, parafiletik atau polifiletik, beserta alasannya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:classifying-biodiversity:4}
Sebutkan ketiga domainnya lalu berikan dua ciri yang memisahkan
Archaea dari Bacteria.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:classifying-biodiversity:5}
Mengapa \omterm{def:b1:classifying-biodiversity:species}{konsep spesies biologi} tak dapat diterapkan pada bakteri atau
pada fosil, dan apa yang dipakai sebagai gantinya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:classifying-biodiversity:6}
Tiga \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} X, Y, Z dan sebuah \omterm{def:b1:classifying-biodiversity:characters}{kelompok luar}. Ciri 1 bersifat turunan
pada X dan Y; ciri 2 pada Y dan Z; ciri 3 pada X dan Y; ciri 4 pada X
dan Y. Gambarlah kedua pohon yang mungkin, hitunglah panjangnya, lalu
berikan yang paling parsimonius beserta \omterm{def:b1:classifying-biodiversity:characters}{homoplasinya}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:classifying-biodiversity:7}
Mengapa sebuah \omterm{def:b1:classifying-biodiversity:characters}{kelompok luar} dibutuhkan untuk membangun sebuah
kladogram, dan apa yang menjadi salah bila \omterm{def:b1:classifying-biodiversity:characters}{kelompok luar} yang dipilih
sebenarnya termasuk di dalam kelompoknya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:classifying-biodiversity:8}
Jelaskan mengapa hewan dan jamur dianggap kerabat yang lebih dekat
daripada hewan dan tumbuhan, dan ciri apa yang dikongsi opistokontanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:classifying-biodiversity:9}
Woese menemukan bahwa metanogen berbeda dari bakteri lain sebanyak
berbeda dari eukariotnya. Mengapa hal ini membenarkan sebuah domain
baru alih-alih sebuah kerajaan baru di dalam bakterinya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:classifying-biodiversity:10}
Sebuah tajuk pohon tropis yang diasapi insektisida menghasilkan 1200
\omterm{def:b1:classifying-biodiversity:species}{spesies} kumbang, yang \qty{20}{\%} di antaranya khas bagi \omterm{def:b1:classifying-biodiversity:species}{spesies}
pohon itu; kumbangnya adalah \qty{40}{\%} serangga tajuknya, dan
tajuknya memuat dua pertiga serangga pohonnya; ada \num{50000} \omterm{def:b1:classifying-biodiversity:species}{spesies}
pohon tropis. Taksirlah cacah \omterm{def:b1:classifying-biodiversity:species}{spesies} serangga tropisnya lalu
sebutkan anggapannya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:classifying-biodiversity:11}
Sebuah \omterm{def:b1:classifying-biodiversity:species}{spesies} diperikan dari sebuah spesimen museum tunggal;
kemudian, \omterm{def:b1:populations:population}{populasi} hidup ditemukan yang sedikit berbeda. Jelaskan
peran \omterm{def:b1:classifying-biodiversity:nomenclature}{spesimen tipenya} dalam memutuskan pada apa namanya berlaku, dan
mengapa keutamaannya berarti.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:classifying-biodiversity:12}
``\omterm{def:b1:classifying-biodiversity:parsimony}{Parsimoni} menganggap bahwa evolusi menempuh jalan terpendek.''
Bahaslah: apa yang benar-benar dianggap \omterm{def:b1:classifying-biodiversity:parsimony}{parsimoni}, kapan ia gagal
(pemusatan, garis keturunan yang berevolusi cepat), dan bagaimana data
molekul serta fosil dipakai untuk memeriksa sebuah pohon.
\end{exercise}

\noindent\textbf{Bagian I --- Membaca \omterm{def:b1:eukaryotic-cell:mitochondrion}{matriksnya}.}
\begin{enumerate}
  \item Ciri mana yang dikongsi seluruh \omterm{def:b1:classifying-biodiversity:nomenclature}{taksonnya} kecuali \omterm{def:b1:classifying-biodiversity:characters}{kelompok

        masing-masing.
  \item Ciri mana yang hadir pada sebuah \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} tunggal? Adakah
        keduanya berguna dalam menempatkan \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} itu?
  \item Pasangan ciri mana yang saling berselisih, dan di antara ketiga
        \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} yang mana?
  \item Nyatakan polaritas ciri 3 (\omterm{def:b1:gas-exchange:lung}{paru-paru}) lalu jelaskan bagaimana
        \omterm{def:b1:classifying-biodiversity:characters}{kelompok luarnya} menetapkannya.
  \item Berapa banyak pohon berakar yang berbeda dapat menghubungkan
        ketiga amniotanya satu sama lain, begitu sisa pohonnya
        ditetapkan? Ciri mana yang berperan dalam pilihannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Membangun pohonnya.}
\begin{enumerate}[resume]
  \item Gambarlah kladogram yang disiratkan ciri 1 sampai 5: \omterm{def:b1:classifying-biodiversity:systematics}{klade}
        bersarangnya dari rahang sampai telur amniotiknya.
  \item Di dalam amniotanya, gambarlah pohon A, yang di dalamnya kadal
        dan merpatinya menjadi \omterm{def:b1:classifying-biodiversity:nomenclature}{takson} saudara, dan pohon B, yang di
        dalamnya tikus dan merpatinya menjadi saudara.
  \item Hitunglah \omterm{def:b1:classifying-biodiversity:parsimony}{panjang pohon} A: bagi tiap dari kesepuluh cirinya,
        cacah minimum perubahan yang dituntutnya.
  \item Hitunglah \omterm{def:b1:classifying-biodiversity:parsimony}{panjang pohon} B dengan cara yang sama.
  \item Pohon mana yang paling parsimonius, dan seberapa jauh?
  \item Sebutkan ciri yang homoplastik pada pohon yang disukainya lalu
        perikan tafsiran evolusinya.
  \item Bila ciri 8 dan 10 disingkirkan, berapa kedua panjangnya akan
        menjadi, dan apa yang akan Anda simpulkan?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Kelompok pada pohonnya.}
\begin{enumerate}[resume]
  \item Pada pohon yang disukainya, adakah kelompok (hiu, trout, ikan
        paru) --- yaitu ``ikan'' --- monofiletik? \omterm{def:b1:classifying-biodiversity:nomenclature}{Takson} mana yang harus
        ditambahkan?
  \item Adakah (kadal, tikus, merpati), yaitu amniotanya, monofiletik?
        Sebutkan \omterm{def:b1:classifying-biodiversity:characters}{sinapomorfinya}.
  \item Adakah (kadal) sendirian ditambah buaya dan kura-kuranya ---
        yaitu ``reptilia'' tradisionalnya --- monofiletik begitu
        merpatinya ditempatkan? Kelompok macam apakah ia?
  \item Adakah (tikus, merpati), yang dikelompokkan lewat endoterminya,
        sebuah \omterm{def:b1:classifying-biodiversity:systematics}{klade}? Kelompok macam apakah ia?
  \item Ikan parunya sering disebut seekor ikan. Pada pohonnya, adakah
        ia lebih dekat dengan troutnya atau dengan kataknya? Ciri mana
        yang memutuskan?
  \item Berikan peringkat yang akan ditempati ketiga amniotanya dalam
        sebuah penggolongan Linnaeus, lalu jelaskan mengapa
        peringkatnya tidak menyampaikan keterangan tentang dalamnya
        belahannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Inventarisnya.}
\begin{enumerate}[resume]
  \item Sekitar \num{70000} \omterm{def:b1:classifying-biodiversity:species}{spesies} vertebrata telah diperikan dan
        laju perian barunya adalah \num{300} setahun bagi ikannya dan
        \num{20} bagi mamalianya. Inventaris mana yang lebih dekat
        pada selesai?
  \item Sebuah survei \omterm{def:b1:nucleic-acids:chain}{DNA} dalam \qty{1}{L} air laut menemukan
        \num{2000} urutan \omterm{def:b1:gene-expression:ribosome}{ribosom} yang berbeda, yang \qty{90}{\%} di
        antaranya tak cocok dengan \omterm{def:b1:organism-environment:organism}{organisme} terperikan mana pun. Apa
        yang disiratkannya bagi inventaris prokariotnya?
  \item Bila laju \omterm{def:b1:classifying-biodiversity:biodiversity}{kepunahan} latarnya adalah satu \omterm{def:b1:classifying-biodiversity:species}{spesies} tiap sejuta
        tiap tahun dan ada \num{9} juta \omterm{def:b1:classifying-biodiversity:species}{spesies} eukariot, berapa banyak
        yang seharusnya punah setahun? Laju yang teramati bagi kelompok
        yang dikaji baik sekitar \num{1000} kali lebih tinggi: berapa
        banyak tiap tahun yang diberikannya bagi seluruh eukariotnya?
  \item Mengapa sebuah \omterm{def:b1:classifying-biodiversity:species}{spesies} yang punah sebelum diperikan menjadi
        sebuah kehilangan bagi \omterm{def:b1:classifying-biodiversity:systematics}{sistematika} maupun bagi \omterm{def:b1:ecosystem-organization:ecosystem}{ekosistemnya}?
  \item Jelaskan mengapa cacah famili yang diperikan menjadi penuntun
        yang lebih baik bagi totalnya daripada cacah \omterm{def:b1:classifying-biodiversity:species}{spesies} yang
        diperikan.
  \item Nyatakan hasilnya: pohon paling parsimonius kedelapan
        vertebratanya, yang ditulis dalam kurung bersarang, beserta
        panjangnya.
\end{enumerate}
\end{problem}
