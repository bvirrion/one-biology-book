\chapter{Os micróbios e a infecção}\label{ch:g9:microbes-and-infection}

Eles seguem este livro como uma sombra desde os germes
invisíveis do primeiro ano
(\cref{def:g1:healthy-habits:germs}): subiram a mão de obra
na padaria (\cref{ch:g6:producing-our-food}), a equipes de
\omterm{def:g6:decomposers-and-soil:decomposers}{decompositores} no \omterm{def:g6:decomposers-and-soil:soil}{solo}, a adversário em \omterm{prop:g9:how-species-change:selection}{evolução} na
enfermaria do hospital
(\cref{ex:g9:how-species-change:resistance}). Este capítulo
enfim presta contas completas dos \omterm{def:g6:producing-our-food:fermentation}{micróbios}: o que eles são,
como a minoria prejudicial ataca e como as defesas externas
do corpo --- e as nossas --- seguram as portas.

\section{O mundo microbiano}

\begin{definition}[Os micróbios, classificados]\label{def:g9:microbes-and-infection:sorted}
Os \emph{microrganismos}\index{microrganismo} --- os
\omterm{def:g6:producing-our-food:fermentation}{micróbios} --- são \omterm{def:g6:exploring-our-environment:components}{seres vivos} pequenos demais para se ver a
\omterm{def:g1:the-five-senses:sense}{olho} nu. Os tipos principais:
\begin{itemize}
  \item as \emph{bactérias}\index{bactéria}: \omterm{def:g5:first-look-at-cells:cell}{células} isoladas
        sem \omterm{def:g6:cell-unit-of-life:parts}{núcleo} verdadeiro, muito menores que as suas
        \omterm{def:g5:first-look-at-cells:cell}{células}, que se dividem depressa (o mundo da
        \cref{def:g6:cell-unit-of-life:unicellular}) --- em
        toda parte, em números fora de qualquer conta;
  \item os \emph{\omterm{def:g6:decomposers-and-soil:decomposers}{fungos} \omterm{def:g6:cell-unit-of-life:unicellular}{unicelulares} e afins}: as leveduras
        (\cref{ex:g6:producing-our-food:bread}), os \omterm{def:g4:plants-without-seeds:spore}{esporos}
        dos bolores, os nadadores da gota de água de lagoa;
  \item os \emph{vírus}\index{vírus}: menores ainda, e
        diferentes por natureza --- um pedacinho embalado de
        texto genético (as letras da
        \cref{def:g9:genes-and-dna:dna}) dentro de um casaco,
        \emph{sem \omterm{def:g5:first-look-at-cells:cell}{célula} e sem vida próprias}: um vírus não
        faz nada até entrar numa \omterm{def:g5:first-look-at-cells:cell}{célula} viva e virar a
        maquinaria de leitura dela para copiar o texto do
        invasor. Nos sinais da
        \cref{prop:g1:living-or-not:signs}, os vírus ficam
        bem na beirada da vida.
\end{itemize}
\end{definition}

\begin{proposition}[Quase sempre inofensivos, muitas vezes essenciais]\label{prop:g9:microbes-and-infection:mostly}
A fama de prejudiciais é um relatório de minoria. A maioria
dos \omterm{def:g6:producing-our-food:fermentation}{micróbios} nos ignora; muitos nos servem: as equipes de
\omterm{def:g6:decomposers-and-soil:decomposers}{decompositores}
(\cref{def:g6:decomposers-and-soil:decomposers}), as mãos de
obra dos alimentos, os químicos do \omterm{def:g6:decomposers-and-soil:soil}{solo} --- e a sua própria
\emph{flora residente}\index{flora residente}: os trilhões de
\omterm{def:g9:microbes-and-infection:sorted}{bactérias} inofensivas que atapetam a \omterm{def:g1:the-five-senses:sense}{pele} e o intestino,
ajudando a \omterm{def:g4:journey-of-food:digestion}{digestão} e, por pura ocupação, expulsando os
invasores do terreno. Só uma minoria pequena --- os
\emph{patógenos}\index{patógeno} --- causa doença.
\end{proposition}
\admitted

\begin{example}[Um corpo, três relações]\label{ex:g9:microbes-and-infection:relations}
Na sua \omterm{def:g1:the-five-senses:sense}{pele} neste minuto: residentes aos bilhões, segurando a
superfície (relação um: aliança). No seu iogurte do café da
manhã: trabalhadores, azedando sob encomenda (relação dois:
emprego, \cref{ex:g6:producing-our-food:yogurt}). No prego
enferrujado do galpão: \omterm{def:g4:plants-without-seeds:spore}{esporos} da \omterm{def:g9:microbes-and-infection:sorted}{bactéria} do tétano,
esperando um corte fundo (relação três: a minoria patogênica,
e a razão do \cref{ch:g9:immune-defenses} e do reforço da
vacina contra o tétano).
\end{example}

\section{A infecção: o ataque, em etapas}

\begin{definition}[As etapas da infecção]\label{def:g9:microbes-and-infection:stages}
A \emph{infecção}\index{infecção} acontece em etapas, cada
uma com o nome dela:
\begin{enumerate}
  \item a \emph{transmissão}: o patógeno viaja ---
        superfícies tocadas e mãos, gotículas de tosse,
        comida e água contaminadas, \omterm{prop:g4:journey-of-food:absorb}{sangue} e a via sexual que
        a \cref{def:g8:contraception:condom} assinalou;
  \item a \emph{entrada}: por uma brecha --- \omterm{def:g1:the-five-senses:sense}{pele} cortada, ou
        as portas abertas do corpo: as vias aéreas, o \omterm{def:g4:journey-of-food:tract}{tubo
        digestivo} e as vias acima;
  \item a \emph{multiplicação}: no calor, na \omterm{def:g6:exploring-our-environment:conditions}{umidade} e com
        comida, os que chegaram se dividem --- as \omterm{def:g9:microbes-and-infection:sorted}{bactérias}
        pela regra das \omterm{def:g6:exploring-our-environment:conditions}{condições} da
        \cref{prop:g6:decomposers-and-soil:speed}, dobrando
        de número em horas; os \omterm{def:g9:microbes-and-infection:sorted}{vírus} transformando \omterm{def:g5:first-look-at-cells:cell}{células} em
        copiadoras;
  \item a \emph{doença}: os sintomas --- vindos do próprio
        estrago, dos venenos das \omterm{def:g9:microbes-and-infection:sorted}{bactérias} e, em parte
        (próximo capítulo), da batalha da própria defesa.
\end{enumerate}
\end{definition}

\begin{example}[Dois invasores, em contraste]\label{ex:g9:microbes-and-infection:contrast}
A \omterm{def:g9:microbes-and-infection:sorted}{bactéria} do corte contra o \omterm{def:g9:microbes-and-infection:sorted}{vírus} do resfriado do inverno. A
\omterm{def:g9:microbes-and-infection:sorted}{bactéria}: entra pela brecha, multiplica-se \emph{entre} as
\omterm{def:g5:first-look-at-cells:cell}{células} no tecido quente e úmido, e os resíduos dela
envenenam a vizinhança --- a latejada e o pus de um dedo
infeccionado. O \omterm{def:g9:microbes-and-infection:sorted}{vírus}: pega carona numa gotícula até o
revestimento das vias aéreas, entra nas próprias \omterm{def:g5:first-look-at-cells:cell}{células}
desse revestimento, e cada \omterm{def:g5:first-look-at-cells:cell}{célula} sequestrada solta uma
multidão de cópias antes de falhar --- a garganta em carne
viva e o nariz escorrendo do rastro dele. Mecânicas
diferentes; o antibiótico que mata de fome a primeira não
encosta no segundo, porque o segundo toma emprestada a
\emph{sua} maquinaria, e os remédios contra ele são difíceis
de mirar.
\end{example}

\section{Segurando as portas}

\begin{proposition}[As primeiras defesas]\label{prop:g9:microbes-and-infection:doors}
Antes de qualquer batalha, o corpo segura as fronteiras dele:
\begin{itemize}
  \item a \emph{\omterm{def:g1:the-five-senses:sense}{pele}}: seca, em camadas, que se renova
        sozinha --- a muralha (o órgão da
        \cref{def:g1:the-five-senses:sense}, de guarda);
  \item os \emph{revestimentos e as vassouras deles}: as
        superfícies de limpeza das vias aéreas (a
        \cref{prop:g7:lungs-and-gas-exchange:smoke} as
        mostrou sobrecarregadas), a química de enxágue das
        lágrimas e da \omterm{def:g4:journey-of-food:tract}{saliva}, o banho ácido do \omterm{def:g4:journey-of-food:tract}{estômago} (o
        \cref{pb:g7:digestion-and-nutrients:1} esteriliza
        enquanto digere);
  \item os \emph{residentes}: o terreno ocupado pela flora,
        com pouco espaço para pousos;
\end{itemize}
e a \omterm{def:g1:healthy-habits:hygiene}{higiene} os estende: o sabonete do
\cref{met:g1:healthy-habits:hands} (agora com o mecanismo
completo dele --- a transmissão interrompida), a água segura,
a comida cozida e mantida no frio (a recusa das \omterm{def:g6:exploring-our-environment:conditions}{condições} da
\cref{rem:g6:producing-our-food:goodbad}), a agulha
esterilizada, a barreira da \omterm{def:g8:contraception:condom}{camisinha}. Toda regra de \omterm{def:g1:healthy-habits:hygiene}{higiene}
deste livro é uma etapa de transmissão ou de entrada,
bloqueada.
\end{proposition}
\admitted

\begin{example}[Os dois médicos que mudaram os números]\label{ex:g9:microbes-and-infection:history}
Dois nomes ancoram a história. Semmelweis, na Viena dos anos
1840: as mortes por febre puerperal despencaram na
maternidade dele quando os médicos passaram a lavar as mãos
entre a sala de autópsia e o parto --- a transmissão
interrompida antes que alguém tivesse visto o viajante.
Pasteur, uma geração depois, mostrou os próprios viajantes:
\omterm{def:g6:producing-our-food:fermentation}{micróbios}, e não o ``ar ruim'', azedando o vinho, matando os
bichos-da-seda, infeccionando as feridas --- e o calor (a
\emph{pasteurização} dele) ou a esterilização detendo-os. A
teoria dos germes transformou os hospitais de perigos em
defesas no espaço de uma vida.
\end{example}

\begin{method}[Auditando um risco de infecção]\label{met:g9:microbes-and-infection:audit}
Para qualquer situação --- cozinha, ferida, enfermaria,
multidão:
\begin{enumerate}
  \item nomeie o patógeno plausível e o tipo dele (\omterm{def:g9:microbes-and-infection:sorted}{bactéria},
        \omterm{def:g9:microbes-and-infection:sorted}{vírus}, \omterm{def:g6:decomposers-and-soil:decomposers}{fungo});
  \item trace a via de transmissão dele até você;
  \item encontre a porta de entrada dele;
  \item pergunte o que o multiplica --- calor, \omterm{def:g6:exploring-our-environment:conditions}{umidade},
        comida, tempo;
  \item bloqueie a etapa mais barata: lavar, cozinhar,
        cobrir, gelar, esterilizar, vacinar (o
        \cref{ch:g9:immune-defenses} explica a última).
\end{enumerate}
\end{method}

\section{Problema: O caderno do surto}

\begin{problem}[{Problema de fim de semana --- um surto na
escola, rastreado etapa por etapa}]\label{pb:g9:microbes-and-infection:1}
Um surto (fictício) numa escola: em uma semana, metade da
turma 9B relata febre e tosse; depois os casos pipocam nas
outras turmas. A enfermeira da escola mantém um caderno do
surto e pede à turma de \omterm{def:g1:living-or-not:living}{biologia} que ajude a lê-lo.

\medskip\noindent\textbf{Parte I --- Lendo a propagação.}
\begin{enumerate}
  \item O primeiro caso --- o caso-índice --- sentava na 9B;
        os oito seguintes dividem a fileira dela e as duplas
        de tênis de mesa dela. Nomeie a via de transmissão
        provável e a porta de entrada.
  \item Os casos dobram a cada dois dias no começo. A
        aritmética de que etapa está aparecendo, e de quem é
        a maquinaria que faz a cópia, se o agente for um
        \omterm{def:g9:microbes-and-infection:sorted}{vírus}?
  \item A lista de conferência da enfermeira: início da
        febre, tosse, garganta em carne viva. Que etapa os
        sintomas marcam, e de que duas fontes eles surgem?
  \item Por que o surto pulou de turma exatamente nos dois
        atos de toda a escola daquela semana?
\end{enumerate}

\medskip\noindent\textbf{Parte II --- O contra-ataque.}
\begin{enumerate}[resume]
  \item As medidas da escola: alunos doentes em casa; janelas
        abertas (\cref{exo:g4:breathing:10}); lavagem de mãos
        treinada; equipamentos compartilhados limpos.
        Atribua a cada medida a etapa que ela bloqueia.
  \item Um pai exige antibióticos para todos. Redija a
        resposta da médica para um surto viral --- e o aviso
        sobre a seleção
        (\cref{exo:g9:how-species-change:10}) caso os
        antibióticos fossem espalhados assim mesmo.
  \item Ao mesmo tempo, a cantina relata dois casos de
        problema de \omterm{def:g4:journey-of-food:tract}{estômago} rastreados até uma sobremesa
        fora da geladeira. Audite esse surto paralelo pela
        \cref{met:g9:microbes-and-infection:audit} --- outro
        tipo de agente, outras etapas.
  \item Por que a enfermeira mapeia \emph{quem sentava onde}
        em vez de tratar os casos como aleatórios? O que é um
        mapa de surto, em termos de transmissão?
\end{enumerate}

\medskip\noindent\textbf{Parte III --- O balanço.}
\begin{enumerate}[resume]
  \item Escreva a biografia do surto nas quatro etapas, do
        caso-índice ao apagamento.
  \item O surto se apagou antes de chegar a todos os alunos.
        Ofereça dois mecanismos --- os bloqueios das medidas
        e a dica do próximo capítulo: quem já estava imune
        pela cepa prima do ano passado.
  \item Acrescente o parágrafo de história: de que duas
        lições do século XIX
        (\cref{ex:g9:microbes-and-infection:history})
        descendeu a resposta inteira da escola?
  \item Encerre o caderno com o método de auditoria virado
        lema: uma frase sobre onde os surtos são mais baratos
        de parar.
\end{enumerate}
\end{problem}
