\chapter{Protecting Nature}\label{ch:g4:protecting}

Local \omterm{def:g4:ecosystems:levels}{ecosystems} give us clean \omterm{def:g4:plants-need:needs}{air}, \omterm{def:g2:food-energy:food}{food}, beauty and \omterm{def:g1:caring:needs}{places} to play.
They are also under pressure from litter, \omterm{def:g4:protecting:habitat-loss}{habitat loss} and pollution.
This chapter explains why nature \omterm{def:g1:caring:needs}{needs} care and what practical actions
help.

\section{Why nature matters}

\begin{definition}[Biodiversity]\label{def:g4:protecting:biodiversity}
\emph{Biodiversity}\index{biodiversity} means the variety of \omterm{def:g1:living-nonliving:living}{living
things} in a \omterm{def:g1:caring:needs}{place} --- many kinds of \omterm{def:g1:plants-animals:plant}{plants}, \omterm{def:g1:plants-animals:animal}{animals}, \omterm{def:g4:microbes:def}{fungi} and
\omterm{def:g4:microbes:def}{microbes}. Rich biodiversity often makes \omterm{def:g4:ecosystems:levels}{ecosystems} more stable and
interesting.
\end{definition}

\begin{proposition}[Services from nature]\label{prop:g4:protecting:services}
Healthy \omterm{def:g4:ecosystems:levels}{ecosystems} help provide oxygen, clean \omterm{def:g4:plants-need:needs}{water}, soil fertility,
\omterm{def:g2:food-energy:food}{food}, materials and enjoyment. Protecting nature protects these
benefits for people too.
\end{proposition}
\admitted

\section{Problems living things face}

\begin{definition}[Habitat loss]\label{def:g4:protecting:habitat-loss}
\emph{Habitat loss}\index{habitat loss} happens when the \omterm{def:g1:caring:needs}{place} a
species \omterm{def:g1:caring:needs}{needs} is destroyed or broken up --- by building, farming
expansion, draining wetlands or cutting \omterm{def:g2:habitats:kinds}{forests} without care.
\end{definition}

\begin{example}[Everyday pressures]\label{ex:g4:protecting:pressures}
Plastic litter can harm \omterm{def:g1:plants-animals:animal}{animals} that eat or become tangled in it.
Polluted \omterm{def:g4:plants-need:needs}{water} kills \omterm{def:g2:animal-groups:five}{fish} and \omterm{def:g3:insects:insect}{insects}. \omterm{def:g4:plants-need:needs}{Lights} and noise disturb some
wildlife. Invasive \omterm{def:g1:plants-animals:plant}{plants} can crowd out native species.
\end{example}

\section{Conservation ideas}

\begin{definition}[Conservation]\label{def:g4:protecting:conservation}
\emph{Conservation}\index{conservation} means protecting and carefully
managing nature so species and \omterm{def:g4:ecosystems:levels}{ecosystems} can continue into the future.
\end{definition}

\begin{method}[Actions that help]\label{met:g4:protecting:actions}
\begin{enumerate}
  \item Reduce, reuse, recycle --- less waste and less demand for new
        materials.
  \item Keep litter out of soil and \omterm{def:g4:plants-need:needs}{water}; join clean-ups when safe.
  \item \omterm{def:g1:plants-animals:plant}{Plant} native \omterm{def:g2:plant-parts:main}{flowers} for pollinators where allowed.
  \item Save \omterm{def:g2:food-energy:energy}{energy} and \omterm{def:g4:plants-need:needs}{water} at home and school.
  \item Learn about local wildlife and respect nature reserves' rules.
  \item Speak up for green spaces in your \omterm{def:g4:ecosystems:levels}{community}.
\end{enumerate}
\end{method}

\begin{example}[School grounds]\label{ex:g4:protecting:school}
A corner of long grass, a log pile and a small \omterm{def:g2:habitats:kinds}{pond} (safely managed)
can \omterm{def:g1:caring:needs}{shelter} \omterm{def:g3:insects:insect}{insects}, \omterm{def:g2:animal-groups:five}{birds} and \omterm{def:g2:animal-groups:five}{amphibians} better than bare tarmac
alone.
\end{example}

\section{You as a nature scientist}

\begin{method}[Notice, record, care]\label{met:g4:protecting:notice}
\begin{enumerate}
  \item Observe a local green space across \omterm{def:g1:seasons:season}{seasons}.
  \item Record species you can identify (apps or guides help).
  \item Note threats (litter hotspots, trampled \omterm{def:g1:plants-animals:plant}{plants}).
  \item Choose one improvement you can actually do.
  \item Check later whether the \omterm{def:g1:caring:needs}{place} improved.
\end{enumerate}
\end{method}
