\chapter{Menjadi Lelaki atau Perempuan}\label{ch:g11:becoming-male-female}

Pada pekan ketujuh setelah pembuahan, embrio manusia punya sepasang
gonad yang bukan testis dan bukan pula ovarium, serta dua perangkat
saluran yang lengkap dan berdampingan --- satu yang dapat menjadi
saluran lelaki, satu yang dapat menjadi saluran perempuan. Tidak ada
yang kasatmata yang membedakan embrio yang akan menjadi anak lelaki
dari embrio yang akan menjadi anak perempuan. Sepanjang pekan-pekan
berikutnya satu perangkat salurannya tumbuh dan yang lain mengerut,
gonadnya mengambil salah satu bentuknya, dan pada bulan keempat
anatominya sudah diputuskan. Keputusannya diambil satu \omterm{def:g10:universal-dna:gene}{gen}, dan
dijalankan dua hormon. Bab ini mengikuti runtunan itu, dari \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}
yang diterima pada pembuahan sampai perubahan pada \omterm{prop:g11:becoming-male-female:puberty}{pubertasnya}.

\section{Kromosom kelamin}

\begin{definition}[Kromosom kelamin]\label{def:g11:becoming-male-female:chromosomes}
Dari 23 pasang \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} manusia, 22 pasang serupa pada kedua kelaminnya;
pasangan ke-23 terbuat dari \emph{kromosom kelamin}\index{kromosom kelamin}:
dua \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X pada perempuan (XX), satu X dan satu Y pada lelaki (XY).
Y-nya kecil, membawa sedikit \omterm{def:g10:universal-dna:gene}{gen}, dan hanya diwariskan dari ayah kepada
anak lelakinya; tiap \omterm{def:g10:cells-common-unit:cell}{sel} telur membawa satu X, dan sebuah sperma
membawa X atau Y, sehingga sperma ayahnyalah yang memutuskan kelamin
\omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} anaknya, dengan peluang yang sama.
\end{definition}

\begin{proposition}[Tahap yang belum terbedakan]\label{prop:g11:becoming-male-female:undifferentiated}
Sampai pekan ketujuh, embrio dari kedua kelamin \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya} serupa
anatomi perkembangbiakannya: sepasang gonad yang belum terbedakan, yang
masing-masing bersisian dengan dua saluran --- yaitu \emph{saluran
Wolff}, bakal saluran lelaki (epididimis, vas deferens), dan
\emph{saluran Müller}, bakal saluran perempuan (saluran telur, rahim).
Perkembangan kelaminnya adalah perubahan rencana bersama ini menjadi
salah satu dari dua bentuknya; pembedaan gonadnyalah yang memulainya.
\end{proposition}
\admitted

\section{Gen yang memutuskan}

\begin{proposition}[SRY dan gonadnya]\label{prop:g11:becoming-male-female:sry}
\omterm{def:g11:cell-cycle-mitosis:chromosome}{Kromosom} Y membawa sebuah \omterm{def:g10:universal-dna:gene}{gen}, yaitu \emph{SRY}\index{gen SRY}, yang
terungkapkan dalam gonad yang belum terbedakan pada pekan ketujuh.
\omterm{def:g11:gene-expression:protein}{Proteinnya} menyalakan \omterm{def:g10:universal-dna:gene}{gen} yang mengubah gonadnya menjadi sebuah
\emph{testis}. Tanpa \omterm{def:g10:universal-dna:gene}{gen} itu --- pada embrio XX --- gonadnya menjadi
sebuah \emph{ovarium} beberapa pekan kemudian. Kehadiran \omterm{prop:g11:becoming-male-female:sry}{SRY}, bukan
jumlah \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X-nya, yang memutuskan gonadnya.
\end{proposition}
\begin{proof}[Bukti percobaan]
Individu XY yang langka yang \omterm{prop:g11:becoming-male-female:sry}{gen SRY-nya} membawa sebuah \omterm{def:g11:mutations:mutation}{mutasi}
mengembangkan ovarium dan tubuh perempuan; individu XX yang langka yang
membawa serpihan Y berisi \omterm{prop:g11:becoming-male-female:sry}{SRY}, yang berpindah ke sebuah X,
mengembangkan testis dan tubuh lelaki. Mencit: embrio XX yang ke
dalamnya \omterm{def:g10:universal-dna:gene}{gen} \emph{Sry} mencit disisipkan berkembang sebagai jantan.
Satu \omterm{def:g10:universal-dna:gene}{gen} dari Y-nya sudah cukup, dan diperlukan, untuk membuat sebuah testis.
\end{proof}

\begin{proposition}[Hormon testis janin]\label{prop:g11:becoming-male-female:hormones}
Begitu terbentuk, testisnya mengeluarkan dua hormon.
\emph{Testosteron}\index{testosteron} menjaga saluran Wolffnya dan
membuatnya berkembang menjadi saluran dalam lelaki, serta membentuk
alat kelamin luarnya dalam bentuk lelaki. \emph{Hormon anti-Müller}
(AMH) membuat saluran Müllernya mengerut. Sebuah ovarium tidak
mengeluarkan satu pun dari keduanya pada tahap ini: tanpa testosteron
saluran Wolffnya mengerut, dan tanpa AMH saluran Müllernya menjadi
saluran telur, rahim dan bagian atas vagina. Jadi saluran perempuan
adalah bentuk yang diambil salurannya tanpa hormon testis.
\end{proposition}
\begin{proof}[Bukti percobaan]
Jost (1947) menyingkirkan gonad janin kelinci dalam rahimnya sebelum
pembedaannya: tiap janinnya, XX maupun XY, mengembangkan saluran
betina. Mencangkokkan sebuah testis di samping gonad janin yang
dikebiri menghasilkan saluran jantan; menanamkan sebuah kristal
testosteron saja menjaga saluran Wolffnya tetapi tidak menyingkirkan
saluran Müllernya, yang bertahan di sampingnya. Jadi testisnya membuat
dua hasil --- satu yang membangun saluran jantannya (testosteron) dan
satu, yang kemudian dikenali sebagai AMH, yang menyingkirkan saluran
betinanya --- dan ovariumnya tidak diperlukan agar saluran betinanya
terbentuk.
\end{proof}

\begin{example}[Riam peristiwanya secara berurutan]\label{ex:g11:becoming-male-female:cascade}
Pembuahan oleh sperma \omterm{def:g11:genes-and-disease:genetic}{pembawa} Y; pekan ke-7, \omterm{prop:g11:becoming-male-female:sry}{SRY} terungkapkan,
gonadnya menjadi testis; pekan ke-8 sampai ke-12, testosteron dan AMH:
saluran Wolff terjaga, saluran Müller lenyap, dan alat kelamin
luarnya terbentuk lelaki; lalu tenang sampai \omterm{prop:g11:becoming-male-female:puberty}{pubertasnya}. Pembuahan oleh sperma
\omterm{def:g11:genes-and-disease:genetic}{pembawa} X: tanpa \omterm{prop:g11:becoming-male-female:sry}{SRY}; pekan ke-10 sampai ke-12, gonadnya menjadi
ovarium; tanpa hormon testis: saluran Müllernya berkembang, saluran
Wolffnya mengerut, alat kelamin luarnya mengambil bentuk perempuan.
Tiap langkahnya bergantung pada langkah sebelumnya, dan kegagalan pada
langkah mana pun mengubah segalanya di hilirnya.
\end{example}

\section{Pubertas}

\begin{proposition}[Pubertas]\label{prop:g11:becoming-male-female:puberty}
Organ perkembangbiakannya, yang terbentuk sebelum lahir, tetap tidak
aktif sepanjang masa kanak-kanak. Pada \emph{pubertas}\index{pubertas}
--- sekitar umur sebelas sampai tiga belas tahun, lebih awal pada anak
perempuan --- sebuah pusat di otaknya mulai melepaskan, dalam denyut,
sebuah hormon yang membuat kelenjar hipofisisnya mengeluarkan dua
hormon, yaitu LH dan FSH, ke dalam darahnya. Keduanya bekerja pada
gonadnya: testisnya mulai membuat sperma dan mengeluarkan testosteron
dalam jumlah dewasa; ovariumnya mulai daurnya dan mengeluarkan
estrogen. Hormon kelaminnya lalu menghasilkan \emph{ciri kelamin
sekunder}: pertumbuhan organ perkembangbiakannya, rambut tubuh, suara,
payudara, pembagian otot dan lemaknya, pacu tumbuhnya lalu menutupnya
lempeng pertumbuhan tulangnya. \omterm{prop:g11:becoming-male-female:puberty}{Pubertasnya} lengkap ketika individunya
dapat berkembang biak.
\end{proposition}
\admitted

\begin{example}[Membaca lengkungnya]\label{ex:g11:becoming-male-female:curves}
Estrogen naik sejak sekitar umur 9 pada anak perempuan, testosteron
sejak sekitar umur 11 pada anak lelaki; masing-masing mencapai separuh
aras dewasanya dalam dua tahun sejak mulainya dan datarannya dua atau
tiga tahun kemudian. Pacu tumbuhnya menyusul kenaikannya sekitar satu
tahun, dan tulangnya berhenti tumbuh ketika hormonnya mencapai aras
dewasa --- dan itulah sebabnya anak perempuan, yang mulai lebih awal,
juga berhenti tumbuh lebih awal dan berakhir lebih pendek rata-rata.
\end{example}

\section{Kelamin, dalam beberapa artinya}

\begin{proposition}[Membedakan arasnya]\label{prop:g11:becoming-male-female:levels}
``Lelaki'' dan ``perempuan'' berlaku pada beberapa aras yang lazimnya
sesuai tetapi berbeda:
\begin{itemize}
  \item kelamin \emph{\omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom}}: XX atau XY, yang tetap sejak
        pembuahan;
  \item kelamin \emph{gonad}: ovarium atau testis, yang diputuskan \omterm{prop:g11:becoming-male-female:sry}{SRY};
  \item kelamin \emph{anatomi}: saluran dan alat kelamin luarnya, yang
        dibentuk hormon gonad janinnya, serta ciri sekundernya, yang
        dibentuk pada \omterm{prop:g11:becoming-male-female:puberty}{pubertasnya};
  \item \emph{identitas gender}, yaitu rasa yang dimiliki seseorang
        sebagai lelaki, perempuan atau bukan keduanya, dan
        \emph{orientasi seksual}, yaitu kepada siapa seseorang tertarik
        --- keduanya kenyataan tentang orangnya yang tidak diturunkan
        dari aras sebelumnya dan tidak dipilih.
\end{itemize}
Biologi bab ini melukiskan ketiga aras yang pertama; ia tidak
menetapkan dan tidak pula menghakimi aras yang terakhir.
\end{proposition}
\admitted

\begin{example}[Ketika arasnya tidak sesuai]\label{ex:g11:becoming-male-female:disagree}
Sekitar satu orang tiap beberapa ribu lahir dengan gabungan yang
lazimnya tidak dihasilkan riam peristiwanya: seorang XY yang \omterm{def:g10:cells-common-unit:cell}{selnya}
tidak punya reseptor bagi testosteron mengembangkan tubuh perempuan,
dengan testis yang tidak pernah turun dan tanpa rahim; seorang XX
dengan translokasi \omterm{prop:g11:becoming-male-female:sry}{SRY} yang membentuk testis mengembangkan tubuh
lelaki; seorang XY dengan \omterm{prop:g11:becoming-male-female:sry}{SRY} yang tidak bekerja berkembang sebagai
perempuan. Ragam ini menunjukkan, lebih baik daripada hal yang biasa,
bahwa tiap arasnya adalah hasil sebuah langkah tertentu --- sebuah \omterm{def:g10:universal-dna:gene}{gen},
sebuah hormon, sebuah reseptor --- dan bahwa arasnya adalah kenyataan
yang terpisah.
\end{example}

\begin{method}[Menalar tentang sebuah kasus]\label{met:g11:becoming-male-female:case}
\begin{enumerate}
  \item \omterm{def:g11:cell-cycle-mitosis:chromosome}{Kromosomnya}: apakah \omterm{prop:g11:becoming-male-female:sry}{SRY} hadir dan bekerja?
  \item Gonadnya: testis jika \omterm{prop:g11:becoming-male-female:sry}{SRY} bekerja, ovarium jika tidak.
  \item Hormon gonad janinnya: testosteron dan AMH, atau tidak
        keduanya; dan dapatkah jaringannya menanggapi (reseptor)?
  \item Salurannya: Wolff terjaga dan Müller tersingkir jika kedua
        hormonnya bekerja; Müller berkembang jika tidak satu pun
        bekerja; keduanya jika testosteron bekerja tanpa AMH.
  \item \omterm{prop:g11:becoming-male-female:puberty}{Pubertasnya}: hormon kelamin mana yang dikeluarkan gonadnya
        memutuskan ciri sekundernya --- lagi-lagi lewat reseptornya.
\end{enumerate}
\end{method}

\begin{remark}[Satu gen, dan segalanya sesudahnya]\label{rem:g11:becoming-male-female:onegene}
Seluruh beda antara kedua anatominya terlacak kembali kepada apakah
satu \omterm{def:g10:universal-dna:gene}{gen} pada satu \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} hadir pada pekan ketujuh. Yang menyusulnya
adalah riam biologi \omterm{def:g10:cells-common-unit:cell}{sel} yang biasa --- pengungkapan \omterm{def:g10:universal-dna:gene}{gen}, hormon,
reseptor --- dari jenis yang dilukiskan tiap bab tahun ini. Rencana
yang menjadi titik awal riamnya sama pada kedua kelaminnya, dan bentuk
perempuannya adalah bentuk yang dibangunnya ketika testisnya tidak
turun tangan.
\end{remark}

\section{Latihan}

\section{Soal: Kelinci Jost}

\begin{problem}[{Soal akhir pekan --- percobaan yang menemukan kedua
hormon testisnya, yang disusun ulang langkah demi langkah, lalu
diterapkan pada tiga kasus manusia dan pada saat pubertasnya}]
\label{pb:g11:becoming-male-female:1}
Pada kelinci, gonadnya terbedakan sekitar hari ke-19 kebuntingannya dan
salurannya sekitar hari ke-22. Jost mengoperasi janin dalam rahimnya
pada hari ke-19 dan memeriksanya saat lahir (hari ke-30).

\medskip
\noindent\textbf{Bagian I --- Percobaannya.}
\begin{enumerate}
  \item Janin dari kedua kelamin \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosomnya}, yang gonadnya
        disingkirkan pada hari ke-19: semuanya lahir dengan saluran
        telur dan rahim serta tanpa saluran jantan. Apa yang
        ditunjukkannya tentang peran ovariumnya, dan tentang peran testisnya?
  \item Mengapa penting sekali mengoperasinya sebelum hari ke-22?
  \item Janin yang dikebiri pada hari ke-19, dengan sebuah testis yang
        dicangkokkan ke dalam perutnya: saluran jantan, tanpa saluran
        betina. Apa yang dipasok cangkoknya, dan bagaimana ia mencapai
        salurannya?
  \item Janin yang dikebiri pada hari ke-19, dengan sebuah kristal
        testosteron yang ditanamkan: saluran jantannya hadir, saluran
        betinanya juga hadir. Apa kerja testosteron, dan apa yang tidak
        dikerjakannya?
  \item Turunkan bahwa testisnya pasti membuat zat kedua, dan nyatakan
        pengaruhnya. Bagaimana keberadaannya kemudian dapat dibenarkan?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Satu gonad yang disingkirkan.}
\begin{enumerate}[resume]
  \item Satu testis disingkirkan dari janin XY pada hari ke-19.
        Ramalkan salurannya pada tiap sisi tubuhnya, jika diketahui
        bahwa hormonnya sebagian besar bekerja pada sisi yang lebih dekat.
  \item Sebuah testis dicangkokkan di samping satu ovarium janin XX.
        Ramalkan tiap sisinya.
  \item Sebuah kristal testosteron ditanamkan di samping satu ovarium
        janin XX. Ramalkan tiap sisinya.
  \item Mengapa percobaan sesisi ini memperkuat kesimpulan Bagian I?
  \item Jelaskan mengapa janin XX dengan cangkok testis tetap akan
        mandul setelah dewasa.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Tiga kasus manusia.}
Terapkan \cref{met:g11:becoming-male-female:case} pada masing-masingnya.
\begin{enumerate}[resume]
  \item Seorang XX membawa, pada satu \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} X-nya, serpihan Y yang
        memuat \omterm{prop:g11:becoming-male-female:sry}{SRY}.
  \item Seorang XY membawa \omterm{prop:g11:becoming-male-female:sry}{SRY} yang tidak bekerja.
  \item Seorang XY dengan testis yang normal punya \omterm{def:g10:cells-common-unit:cell}{sel} yang tidak dapat
        menanggapi AMH.
  \item Pada hal mana di antara ketiganya orangnya mungkin subur?
        Berikan alasannya.
  \item Apa yang ditunjukkan ketiga kasus itu tentang hubungan antara
        kelamin \omterm{def:g11:cell-cycle-mitosis:chromosome}{kromosom} dan kelamin anatominya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- \omterm{prop:g11:becoming-male-female:puberty}{Pubertas} dalam angka.}
Dari gambar hormonnya, ambil pacu tumbuhnya mulai satu tahun setelah
hormonnya mencapai 20\% aras dewasanya dan tulangnya berhenti tumbuh
ketika hormonnya mencapai 100\%.
\begin{enumerate}[resume]
  \item Perkirakan umur ketika pacu tumbuhnya mulai pada anak perempuan
        dan pada anak lelaki, serta umur ketika pertumbuhannya berhenti.
  \item Berapa tahun pertumbuhan yang dimiliki tiap kelaminnya antara
        awal pacu tumbuhnya dan akhirnya?
  \item Seorang anak lelaki tumbuh \qty{8}{cm} tiap tahun selama pacu
        tumbuhnya dan seorang anak perempuan \qty{7}{cm}. Perkirakan
        tinggi yang diperoleh masing-masing selama pacu tumbuhnya, dan
        beri tanggapan atas beda tinggi dewasa rata-ratanya (sekitar
        \qty{13}{cm}).
  \item Seorang anak perempuan yang kenaikan estrogennya mulai pada
        umur 7 akan tumbuh cepat lebih awal dan berhenti lebih awal.
        Jelaskan mengapa dokter kadang-kadang menunda \omterm{prop:g11:becoming-male-female:puberty}{pubertas} yang
        sangat dini dengan obat.
  \item Nyatakan hasilnya: kedua hormon yang disingkapkan kelinci Jost,
        pengaruh masing-masing, dan satu \omterm{def:g10:universal-dna:gene}{gen} yang menyalakan organ yang
        membuatnya.
\end{enumerate}
\end{problem}
