\section*{Bab \ref{ch:g11:dna-replication} --- Replikasi DNA}

\begin{solution}{exo:g11:dna-replication:1}
Tiap untai bertugas sebagai cetakan yang di atasnya untai komplementer
yang baru dibangun. Ia berhasil hanya karena A berpasangan dengan T dan
G dengan C: urutan satu untai menetapkan urutan untai yang lain,
sehingga satu untai sendirian membawa seluruh keterangannya.
\end{solution}

\begin{solution}{exo:g11:dna-replication:2}
Semikonservatif: tiap molekul anaknya menyimpan satu untai induk dan
memperoleh satu untai baru. Konservatif: molekul induknya tetap utuh
dan satu untai ganda yang sepenuhnya baru dibangun di sampingnya.
\end{solution}

\begin{solution}{exo:g11:dna-replication:3}
Fase S dalam daur \omterm{def:g10:cells-common-unit:cell}{selnya}; jumlah \omterm{def:g10:universal-dna:information}{DNA} tiap \omterm{def:g10:cells-common-unit:organelle}{inti sel} berlipat dua.
\end{solution}

\begin{solution}{exo:g11:dna-replication:4}
\omterm{prop:g11:dna-replication:forks}{DNA polimerase} menambahkan pada untai yang sedang tumbuh \omterm{def:g10:universal-dna:nucleotide}{nukleotida}
yang komplementer dengan cetakannya, satu demi satu. \omterm{prop:g11:dna-replication:forks}{Garpu replikasi}
adalah titik bergerak tempat untai induknya dipisahkan dan kedua untai
barunya sedang dibangun.
\end{solution}

\begin{solution}{exo:g11:dna-replication:5}
\texttt{CTAATGTCCG}, \omterm{def:g10:universal-dna:nucleotide}{basa} di bawah \omterm{def:g10:universal-dna:nucleotide}{basa}.
\end{solution}

\begin{solution}{exo:g11:dna-replication:6}
Konservatif, generasi 1: satu pita berat dan satu pita ringan, tanpa
hibrid --- padahal hanya pita hibrid yang terlihat. Dispersif, generasi
2: satu pita tunggal seperempat jalan dari ringan ke berat --- padahal
dua pita yang berbeda terlihat. Tiap model meramalkan pita yang tidak teramati.
\end{solution}

\begin{solution}{exo:g11:dna-replication:7}
$2^5 = 32$ molekul, 2 hibrid dan 30 ringan: pita ringannya lima belas
kali pita hibridnya, tanpa pita berat.
\end{solution}

\begin{solution}{exo:g11:dna-replication:8}
Generasi 0: satu pita ringan. Generasi 1: satu pita hibrid. Generasi 2:
pita hibrid dan pita berat yang sama pekatnya --- yaitu bayangan cermin
percobaan aslinya.
\end{solution}

\begin{solution}{exo:g11:dna-replication:9}
$\num{2.5e8}/50 = \num{5e6}\,\unit{s}$, sekitar 58 hari --- jauh lebih
lama daripada 8 jam. \omterm{def:g10:universal-dna:information}{Kromosomnya} harus disalin dari banyak titik awal
sekaligus, yang masing-masing membuka dua garpu.
\end{solution}

\begin{solution}{exo:g11:dna-replication:10}
Kedua untai dari $\num{3.2e9}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa}, dua kali: $\num{6.4e9}$
pasangan \omterm{def:g10:universal-dna:nucleotide}{basa}, yaitu $\num{6.4e9}$ \omterm{def:g10:universal-dna:nucleotide}{nukleotida} baru yang disambungkan
(satu tiap pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} yang disalin). Pada $10^{-9}$ tiap \omterm{def:g10:universal-dna:nucleotide}{nukleotida},
sekitar 6 kesalahan.
\end{solution}

\begin{solution}{exo:g11:dna-replication:11}
Membuka sebuah gelembung menyingkapkan cetakan pada kedua sisi titik
awalnya; tiap garpu menyalin menjauhi titik awalnya, sehingga keduanya
berjalan dalam arah berlawanan dan bersama-sama meliput seluruh
daerahnya. Ketika dua gelembung bertemu, garpunya berpadu dan untai
anaknya bersambung ujung ke ujung menjadi molekul yang menerus.
\end{solution}

\begin{solution}{exo:g11:dna-replication:12}
Bakterinya berhenti membelah (tanpa replikasi, tanpa pembelahan).
Lukanya berhenti sembuh, karena perbaikannya menuntut pembelahan \omterm{def:g10:cells-common-unit:cell}{sel}.
\omterm{def:g10:cells-common-unit:cell}{Sel} sarafnya tidak terpengaruh: ia tidak lagi mereplikasi \omterm{def:g10:universal-dna:information}{DNA-nya}. Obat
yang menghentikan \omterm{def:g10:cells-common-unit:cell}{sel} yang cepat membelah sambil menyayangi \omterm{def:g10:cells-common-unit:cell}{sel} yang
tidak membelah menyarankan pengobatan melawan bakteri atau melawan \omterm{def:g10:cells-common-unit:cell}{sel}
kanker yang membelah.
\end{solution}

\begin{solution}{exo:g11:dna-replication:13}
Kedua untai aslinya tidak pernah dimusnahkan: pada tiap generasi
masing-masing kembali berpasangan dengan untai ringan yang baru,
sehingga tepat dua molekul hibrid selalu ada. Setelah 10 generasi: $2/2^{10} = 2/1024
\approx 0.2\%$ --- ada tetapi terlalu samar untuk terlihat.
\end{solution}

\begin{solution}{exo:g11:dna-replication:14}
$\num{6.4e9} \times 10^{-5} = \num{64000}$ kesalahan tiap pembelahan.
Dengan dua puluh ribu \omterm{def:g10:universal-dna:gene}{gen}, ratusan \omterm{def:g10:universal-dna:gene}{gen} akan rusak pada tiap
pembelahannya; setelah beberapa pembelahan \omterm{def:g10:cells-common-unit:cell}{selnya} akan membawa cacat
yang mematikan dan galurnya akan punah.
\end{solution}

\begin{solution}{exo:g11:dna-replication:15}
Putaran replikasi yang baru mulai pada titik awalnya sebelum putaran
sebelumnya selesai: \omterm{def:g10:universal-dna:information}{kromosomnya} membawa beberapa pasang garpu yang
bersarang sekaligus. Tiap pembelahan lalu menerima \omterm{def:g10:universal-dna:information}{kromosom} yang
sebagiannya sudah disalin untuk pembelahan berikutnya, dan
pembelahannya dapat saling menyusul lebih cepat daripada satu penyalinan.
\end{solution}

\begin{solution}{pb:g11:dna-replication:1}
\textbf{1.} Nitrogen menyusun 16\% massanya dan 7\% lebih berat:
$0.16 \times 0.07 \approx 1.1\%$. Cara gradiennya memisahkan massa
jenis yang berbeda sepersekian persen, sehingga 1\% memberi pita yang
terpisah baik.

\textbf{2.} Agar hampir seluruh \omterm{def:g10:universal-dna:information}{DNA-nya}, bukan hanya separuh atau tiga
perempatnya, menjadi berat: setelah 14 generasi kurang dari satu bagian
dalam $10^4$ untai ringan aslinya yang tersisa.

\textbf{3.} Tepat di tengah antara kedua pitanya: massa jenisnya adalah
rata-ratanya.

\textbf{4.} Satu bakteri memuat beberapa femtogram \omterm{def:g10:universal-dna:information}{DNA}; potretnya
menuntut mikrogram, jadi miliaran \omterm{def:g10:cells-common-unit:cell}{sel}.

\textbf{5.} Dengan menyimpan satu biakan dalam medium berat sebagai
kontrol dan memeriksa bahwa pitanya tetap berat, dan dengan memeriksa
pita ringan bakteri yang ditumbuhkan dalam medium ringan sepanjang waktu.

\textbf{6.} Semikonservatif: generasi 1, satu pita hibrid; generasi 2,
pita hibrid dan pita ringan, sama pekatnya.

\textbf{7.} Konservatif: generasi 1, pita berat dan pita ringan, sama
pekatnya; generasi 2, satu berat berbanding tiga ringan.

\textbf{8.} Dispersif: generasi 1, satu pita pada kedudukan hibridnya;
generasi 2, satu pita seperempat jalan dari ringan ke berat.

\textbf{9.} Model konservatif, yang meramalkan tidak ada pita hibrid,
tersingkir. Semikonservatif dan dispersif sama-sama meramalkan pita
hibrid tunggal itu dan keduanya bertahan.

\textbf{10.} Semikonservatif bertahan: hanya ia yang meramalkan dua
pita yang terpisah. Model dispersif meramalkan satu pita tunggal yang
menghanyut makin ringan, tidak pernah terbelah.

\textbf{11.} $2^n$ molekul, yang tepat 2 di antaranya membawa untai
yang asli.

\textbf{12.} Ringan : hibrid $= (2^n - 2) : 2$: generasi 3, $3:1$;
generasi 4, $7:1$; generasi 6, $31:1$.

\textbf{13.} Bagian hibridnya $2/2^n < 0.05$ ketika $2^n > 40$: mulai
generasi 6 ($2/64 \approx 3\%$).

\textbf{14.} Dua pita, satu berat dan satu ringan, yang sama
jumlahnya: molekul hibrid adalah satu untai berat ditambah satu untai
ringan. Hal ini menguji model dispersif, yang meramalkan tiap untai
tunggalnya bermassa jenis antara, dan menyingkirkannya secara mandiri.

\textbf{15.} Jika berangkat dari ringan, satu generasi dalam medium
berat memberi satu pita hibrid dan dua generasi memberi hibrid ditambah
berat: pola yang sama tetapi tercermin, yang menunjukkan bahwa hasilnya
tidak bergantung pada isotop mana yang ``lama''.

\textbf{16.} $\num{4.6e6}/(2 \times 1000) = \qty{2300}{s}$, sekitar
38 menit.

\textbf{17.} 77 menit. Pada sebuah lingkaran, kedua untainya
tersingkap pada kedua sisi bukaan mana pun; dua garpu yang
meninggalkan satu titik awal dalam arah berlawanan adalah cara
paling sederhana untuk meliput seluruh lingkarannya.

\textbf{18.} Satu titik awal menyalin $2 \times 50 \times 8 \times 3600 =
\num{2.88e6}$ pasangan \omterm{def:g10:universal-dna:nucleotide}{basa} dalam 8 jam; $\num{6.4e9}/\num{2.88e6}
\approx 2200$ titik awal sebagai jumlah paling sedikit (kenyataannya
puluhan ribu, yang menyala pada saat yang berbeda-beda).

\textbf{19.} Bakterinya: $\num{9.2e6} \times 10^{-9} \approx 0.01$
kesalahan tiap replikasi; \omterm{def:g10:cells-common-unit:cell}{sel} manusianya: sekitar 6. Sebagian besar
genom manusia bukan \omterm{def:g10:universal-dna:gene}{gen}, sehingga sebagian besar kesalahannya jatuh di
tempat yang tidak mengubah apa pun, dan \omterm{def:g10:cells-common-unit:cell}{selnya} punya dua salinan tiap \omterm{def:g10:universal-dna:gene}{gen}.

\textbf{20.} Kedua pita pada generasi kedua --- hibrid dan ringan dalam
jumlah yang sama --- membuktikan bahwa replikasinya semikonservatif;
kedua garpunya menyalin \omterm{def:g10:universal-dna:information}{kromosom} bakterinya dalam sekitar 38 menit,
kira-kira sama dengan waktu generasi bakterinya, sehingga penyalinannya
menetapkan irama pembelahannya.
\end{solution}
