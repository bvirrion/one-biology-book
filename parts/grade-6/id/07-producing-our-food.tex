\chapter{Menghasilkan Makanan Kita}\label{ch:g6:producing-our-food}

Tidak ada satu pun di atas meja sarapanmu yang ditemukan di alam
liar. Gandum pada rotinya dibiakkan lalu ditanam; \omterm{def:g2:animals-grow-up:born}{susunya} datang dari
kawanan yang dipelihara untuk itu; yogurtnya \emph{dibuat} dari \omterm{def:g2:animals-grow-up:born}{susu}
itu oleh miliaran pekerja hidup yang terlalu kecil untuk dilihat.
Manusia memberi makan dirinya dengan mengelola \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} ---
menumbuhkannya, memeliharanya, dan mempekerjakan sebagian yang paling
kecil. Bab ini berkeliling di bengkel makanannya.

\section{Menumbuhkan dan memelihara}

\begin{definition}[Tanaman pangan dan ternak]\label{def:g6:producing-our-food:farming}
Produksi makanan manusia mengelola \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} pada dua lantai daur
materinya:
\begin{itemize}
  \item \emph{penanaman pangan}\index{tanaman pangan}: mengelola
        \omterm{def:g5:food-webs:roles}{produsen} --- gandum, kentang, sayur, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} --- demi \omterm{def:g6:plants-are-producers:matter}{materi
        organiknya}
        (\cref{prop:g6:plants-are-producers:produce});
  \item \emph{pemeliharaan ternak}\index{ternak}: mengelola \omterm{def:g5:food-webs:roles}{konsumen}
        --- sapi, domba, babi, ayam --- demi \omterm{def:g6:matter-from-food:production}{produksi materinya}
        (\cref{def:g6:matter-from-food:production}): \omterm{def:g2:animals-grow-up:born}{susu}, daging,
        \omterm{def:g2:animals-grow-up:born}{telur}, wol.
\end{itemize}
Petani mengelola \omterm{def:g6:exploring-our-environment:conditions}{kondisi} yang terus dijumpai buku ini: mineral \omterm{def:g6:decomposers-and-soil:soil}{tanah}
bagi tanamannya (\cref{rem:g4:what-plants-need:soil}), pakan bagi
\omterm{def:g1:animals-around-us:animal}{hewannya} --- seluruh pembukuan pada
\cref{pb:g6:matter-from-food:1}, yang dijalankan dengan sengaja.
\end{definition}

\begin{example}[Sebuah ladang gandum itu apa]\label{ex:g6:producing-our-food:wheat}
Sebuah ladang gandum adalah \omterm{def:g6:exploring-our-environment:components}{lingkungan} yang dikelola
(\cref{prop:g5:people-and-nature:managed}) dan disetel bagi satu
\omterm{def:g5:food-webs:roles}{produsen}: \omterm{def:g6:decomposers-and-soil:soil}{tanahnya} diisi ulang lalu digemburkan bagi \omterm{def:g1:plants-around-us:parts}{akarnya},
penyemaiannya diatur waktunya menurut kehangatan pada
\cref{prop:g2:from-seed-to-plant:needs}, rumput liar --- penjajah
saingan pada \cref{ch:g6:how-plants-spread} --- dijauhkan. Panennya
adalah simpanan \omterm{def:g2:from-seed-to-plant:seed}{biji} \omterm{def:g1:plants-around-us:plant}{tumbuhannya}: butir gandum, bekal sejuta tanaman
gandum yang tidak akan pernah disemai, yang dialihkan ke toko roti.
\end{example}

\begin{remark}[Setiap makanan adalah sebuah tahap daur hidup]\label{rem:g6:producing-our-food:stage}
Bacalah makanan apa pun secara biologis dan ia adalah organ atau tahap
sebuah organisme: butir gandum dan kacang adalah simpanan \omterm{def:g2:from-seed-to-plant:seed}{biji};
kentang, \omterm{ex:g4:plants-without-seeds:bulbtuber}{umbi batang}; apel, \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} yang dibangun untuk dimakan
(\cref{exo:g6:how-plants-spread:12}); \omterm{def:g2:animals-grow-up:born}{susu} adalah perawatan induk
\omterm{prop:g3:sorting-living-things:groups}{mamalia} (\cref{def:g2:animals-grow-up:born}); sebutir \omterm{def:g2:animals-grow-up:born}{telur} adalah
sebuah ruang bayi yang terkemas. Produksi makanan adalah seni
mengelola \omterm{def:g2:born-grow-die:cycle}{daur hidup} sehingga bekalnya tiba di atas meja kita.
\end{remark}

\section{Angkatan kerja yang tak terlihat: makanan buatan mikroba}

\begin{definition}[Fermentasi]\label{def:g6:producing-our-food:fermentation}
Sebagian makanan dibuat dengan mempekerjakan \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang
terlalu kecil untuk dilihat --- \emph{mikroba}\index{mikroba pangan}
--- pada sebuah bahan baku. Kegiatan makan mereka mengubahnya:
perubahan itu disebut
\emph{fermentasi}\index{fermentasi}, dan manusia sudah menjalankannya
selama ribuan tahun, jauh sebelum siapa pun tahu bahwa pekerjanya ada.
\end{definition}

\begin{example}[Roti mengembang]\label{ex:g6:producing-our-food:bread}
Adonan mentah roti terbuat dari tepung dan air --- ditambah
\emph{ragi}\index{ragi}, sejenis \omterm{def:g6:decomposers-and-soil:decomposers}{jamur} bersel tunggal
(\cref{rem:g5:first-look-at-cells:onecelled} sudah berjumpa dengan
kehidupan semacam itu). \omterm{ex:g6:producing-our-food:bread}{Raginya} makan dari gula tepungnya lalu
melepaskan sebuah gas; gelembung gasnya, yang terperangkap di dalam
adonan yang lentur, mengembungkannya --- tukang rotinya menyebutnya
mengembang. Panas ovennya lalu mengakhiri giliran kerja para
pekerjanya dan memaku hasil kerjanya: lubang pada remah rotinya adalah
tanda tangan \omterm{ex:g6:producing-our-food:bread}{raginya}.
\end{example}

\begin{omfigure}
\includegraphics[width=0.58\linewidth]{images/book1/ai/g6-food-bread.jpg}

{\small Tanda tangan angkatan kerjanya, yang terpanggang di tempat:
tiap lubang pada remah rotinya adalah gelembung yang ditiup \omterm{ex:g6:producing-our-food:bread}{raginya}.}
\end{omfigure}

\begin{example}[Susu menjadi yogurt dan keju]\label{ex:g6:producing-our-food:yogurt}
\omterm{def:g2:animals-grow-up:born}{Susu} hangat yang ditaburi \omterm{def:g6:producing-our-food:fermentation}{mikroba} yang tepat mengental dalam semalam
menjadi yogurt: pekerjanya makan dari gula \omterm{def:g2:animals-grow-up:born}{susunya} lalu melepaskan
sebuah asam, dan asamnya memadatkan \omterm{def:g2:animals-grow-up:born}{susunya} --- rasa asam segarnya
\emph{adalah} asam itu. Keju bermula serupa, lalu berjalan lebih jauh:
dadihnya dipres, digarami dan dimatangkan berbulan-bulan, sementara
\omterm{def:g6:producing-our-food:fermentation}{mikrobanya} terus bekerja sepanjang pematangannya --- lubang, kerak dan
kekuatan bau sebuah keju adalah catatan giliran kerjanya yang panjang.
\end{example}

\begin{remark}[Mikroba yang menolong dan yang merugikan]\label{rem:g6:producing-our-food:goodbad}
\omterm{def:g1:healthy-habits:germs}{Kuman} pada \cref{def:g1:healthy-habits:germs} mengajarkan kewaspadaan
tentang \omterm{def:g6:producing-our-food:fermentation}{mikroba}; bab ini menambahkan paruh yang lain: \omterm{def:g6:producing-our-food:fermentation}{mikroba} yang
\emph{dipilih}, yang bekerja dalam \omterm{def:g6:exploring-our-environment:conditions}{kondisi} yang \emph{dikendalikan},
termasuk pegawai kita yang tertua --- di dalam roti, yogurt, keju, dan
tong cuka. Kerajinannya selalu sama: sambutlah pekerja yang dipilih,
dan pertahankanlah \omterm{def:g6:exploring-our-environment:conditions}{kondisi} yang cocok bagi mereka dan tidak bagi para
perusak. Penggaraman, pendinginan, pengeringan dan pemasakan semuanya
adalah cara menolak \omterm{def:g6:exploring-our-environment:conditions}{kondisi} bagi \omterm{def:g6:producing-our-food:fermentation}{mikroba} yang tidak diinginkan
(\cref{prop:g6:decomposers-and-soil:speed} berlaku pada sepotong ham
persis seperti pada setumpuk \omterm{def:g1:plants-around-us:parts}{daun}).
\end{remark}

\section{Dari ladang ke meja, dengan jujur}

\begin{proposition}[Biaya yang ada di atas meja]\label{prop:g6:producing-our-food:costs}
Setiap piring adalah keluaran dari \omterm{def:g6:decomposers-and-soil:soil}{tanah} dan \omterm{def:g6:exploring-our-environment:components}{makhluk hidup} yang
dikelola, dan ia membawa hitung-hitungannya:
\begin{itemize}
  \item makanan \omterm{def:g1:plants-around-us:plant}{tumbuhan} tiba satu lantai dari \omterm{def:g6:exploring-our-environment:conditions}{cahayanya}; makanan
        \omterm{def:g1:animals-around-us:animal}{hewan} menempuh jalur lewat seorang \omterm{def:g5:food-webs:roles}{konsumen} lebih dahulu lalu
        membayar bea pada
        \cref{prop:g6:matter-from-food:losses}
        (\cref{exo:g6:matter-from-food:12});
  \item produksi yang bertahan harus memegang ketiga aturan pada
        \cref{prop:g5:people-and-nature:lasting} --- berilah makan
        \omterm{def:g6:decomposers-and-soil:soil}{tanahnya}, pertahankan para penolongnya, jangan mengambil lebih
        cepat daripada pembaruannya;
  \item dan pemborosan membatalkan semuanya: makanan yang dibuang
        membelanjakan \omterm{def:g6:decomposers-and-soil:soil}{tanah}, air dan kerjanya tanpa berguna bagi
        siapa pun.
\end{itemize}
\end{proposition}
\admitted

\begin{method}[Membaca kisah sebuah makanan]\label{met:g6:producing-our-food:read}
Untuk makanan apa pun di dapur:
\begin{enumerate}
  \item sebutkan organismenya dan organ atau tahap yang menjadi
        wujudnya (\cref{rem:g6:producing-our-food:stage});
  \item tempatkan ia pada daurnya: materi \omterm{def:g5:food-webs:roles}{produsen}, hasil \omterm{def:g5:food-webs:roles}{konsumen},
        atau buatan \omterm{def:g6:producing-our-food:fermentation}{mikroba}?
  \item kalau buatan \omterm{def:g6:producing-our-food:fermentation}{mikroba}, sebutkan bahan bakunya dan apa yang
        diubah para pekerjanya;
  \item lacaklah ia sampai ke \omterm{def:g1:plants-around-us:parts}{daunnya}
        (\cref{met:g6:plants-are-producers:trace}) --- jejaknya tetap
        selalu berakhir di dalam \omterm{def:g6:exploring-our-environment:conditions}{cahaya}.
\end{enumerate}
\end{method}

\begin{example}[Metodenya ketika sarapan]\label{ex:g6:producing-our-food:breakfast}
Yogurt, menurut metodenya: (1) \omterm{def:g2:animals-grow-up:born}{susu} --- hasil perawatan induk seekor
sapi --- yang diubah; (2) hasil \omterm{def:g5:food-webs:roles}{konsumen}, yang diselesaikan \omterm{def:g6:producing-our-food:fermentation}{mikroba};
(3) bahan bakunya \omterm{def:g2:animals-grow-up:born}{susu}, pekerjanya mengasamkan lalu memadatkannya; (4)
\omterm{def:g2:animals-grow-up:born}{susu} dari sapinya, sapi dari rumput, rumput dari \omterm{def:g6:exploring-our-environment:conditions}{cahaya}. Tiga organisme
dan sebuah angkatan kerja bermiliar-miliar, di dalam satu pot kecil.
\end{example}

\section{Soal: Toko Roti Kelas}

\begin{problem}[{Soal akhir pekan --- satu adonan roti, yang diikuti
dari biji sampai irisan}]\label{pb:g6:producing-our-food:1}
Untuk pasar sekolahnya, kelasnya memanggang roti dari awal --- dan
membiarkan buku catatan \omterm{def:g1:living-or-not:living}{biologinya} terbuka sejak kunjungan ke ladang
gandum pada bulan Juli sampai roti bulatnya pada bulan November.

\medskip\noindent\textbf{Bagian I --- Ladangnya (Juli).}
\begin{enumerate}
  \item Petaninya memperlihatkan gandum yang masak: ``panennya adalah
        bekal \omterm{def:g1:plants-around-us:plant}{tumbuhannya}.'' Uraikan frasa itu dengan
        \cref{def:g2:from-seed-to-plant:seed} --- tiap butirnya itu
        apa, dan dikemas untuk siapa?
  \item Di dalam \omterm{def:g1:plants-around-us:parts}{daun} siapa, dan dengan masukan apa, \omterm{def:g6:plants-are-producers:matter}{materi organik}
        butirnya dibuat?
  \item Petaninya hanya menyemai sebagian panennya pada musim gugur
        berikutnya lalu menjual sisanya. Sebutkan kedua nasib sebutir
        gandum, dalam istilah \omterm{def:g2:born-grow-die:cycle}{daur hidup}.
  \item \omterm{def:g6:decomposers-and-soil:soil}{Tanah} ladangnya mendapat pupuk kandang sesudah panen. Aturan
        pengembalian yang mana ini, dan kebutuhan mana di antara
        kelima kebutuhan \omterm{def:g1:plants-around-us:plant}{tumbuhannya} yang dilayaninya?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Toko rotinya (November).}
\begin{enumerate}[resume]
  \item Kelasnya mencampur tepung, air, garam dan \omterm{ex:g6:producing-our-food:bread}{ragi}, lalu meletakkan
        adonannya di sudut yang hangat. Mengapa hangat? Jawablah bagi
        \omterm{ex:g6:producing-our-food:bread}{raginya} sebagaimana bagi pekerja hidup mana pun
        (\cref{prop:g6:decomposers-and-soil:speed} memegang polanya).
  \item Sejam kemudian adonannya sudah membesar dua kali lipat.
        Jelaskan pengembangannya: \omterm{ex:g6:producing-our-food:bread}{raginya} makan dari apa, apa yang
        dilepaskannya, dan apa yang memerangkapnya?
  \item Adonan salah satu regu, yang terlupakan di dekat jendela dingin
        yang berangin, nyaris tidak mengembang. Diagnosislah --- lalu
        sebutkan perbandingan uji adil yang tanpa sengaja dijalankan
        kedua regunya.
  \item Di dalam ovennya, roti bulatnya berhenti mengembang lalu
        memadat. Apa yang sudah dilakukan panasnya terhadap angkatan
        kerjanya, dan apa yang tersisa sebagai tanda tangannya pada
        remah rotinya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Pembukuannya.}
\begin{enumerate}[resume]
  \item Lacaklah irisan roti yang sudah jadi sampai ke \omterm{def:g6:exploring-our-environment:conditions}{cahayanya}, mata
        rantai demi mata rantai.
  \item Pasar sekolahnya juga menjual roti isi mentega dan roti isi
        ham. Urutkan irisan rotinya, menteganya dan hamnya menurut
        lantai yang dilalui materinya, lalu nyatakan mana yang membayar
        bea terbesar pada
        \cref{prop:g6:matter-from-food:losses}.
  \item Adonan yang tidak terjual dikompos, bukan dibuang ke tempat
        sampah. Ikutilah \omterm{def:g6:plants-are-producers:matter}{materi organiknya} satu putaran lebih jauh
        mengelilingi daurnya --- siapa yang mengambilnya, dan menuju
        apa?
  \item Tutuplah buku catatannya dengan satu kalimat yang memuat:
        \omterm{def:g5:food-webs:roles}{produsen}, \omterm{def:g5:food-webs:roles}{konsumen} (kalau ada), angkatan kerja \omterm{def:g6:producing-our-food:fermentation}{mikroba}, dan
        \omterm{def:g6:exploring-our-environment:conditions}{cahaya}.
\end{enumerate}
\end{problem}
