\section*{Chapter \ref{ch:g4:plants-without-seeds} --- Plants Without Seeds}

\begin{solution}{exo:g4:plants-without-seeds:1}
\omterm{prop:g4:plants-without-seeds:damp}{Mosses} and \omterm{prop:g4:plants-without-seeds:damp}{ferns}. They reproduce by \omterm{def:g4:plants-without-seeds:spore}{spores} --- microscopic
grains released in enormous numbers.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:2}
A \omterm{def:g4:plants-without-seeds:spore}{spore} is microscopic and bare --- no \omterm{def:g3:animals-through-seasons:active}{food store}, no
ready-made tiny \omterm{def:g1:plants-around-us:plant}{plant}; a \omterm{def:g2:from-seed-to-plant:seed}{seed} carries both, inside a protective
coat. And \omterm{def:g4:plants-without-seeds:spore}{spores} are made by the million, \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} by the dozen.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:3}
On the underside of the fronds: neat rows of brown dots, each a
packet of thousands of \omterm{def:g4:plants-without-seeds:spore}{spores} that split open when ripe and let
them drift off like dust.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:4}
Moss reproduces by \omterm{def:g4:plants-without-seeds:spore}{spores}, and \omterm{def:g4:plants-without-seeds:spore}{spores} and their delicate young
stages need dampness --- they dry out and die in dry air. The
shaded north wall stays damp; the sunny south wall bakes.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:5}
Making a new \omterm{def:g1:plants-around-us:plant}{plant} directly from a piece of the parent \omterm{def:g1:plants-around-us:plant}{plant}
--- runner, \omterm{ex:g4:plants-without-seeds:bulbtuber}{bulb}, \omterm{ex:g4:plants-without-seeds:bulbtuber}{tuber}, cutting. One parent only.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:6}
The \omterm{def:g1:plants-around-us:plant}{plant} sends a long thin \omterm{def:g1:plants-around-us:parts}{stem} --- the runner --- across the
soil; where it touches down it \omterm{def:g1:plants-around-us:parts}{roots} and sprouts \omterm{def:g1:plants-around-us:parts}{leaves},
becoming a complete little \omterm{def:g1:plants-around-us:plant}{plant}; the runner then withers and
the child stands free, identical to its parent.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:7}
A planted potato (a \omterm{ex:g4:plants-without-seeds:bulbtuber}{tuber}) sprouts reliably from its
``\omterm{def:g1:the-five-senses:sense}{eyes}'', each of which can grow into a whole \omterm{def:g1:plants-around-us:plant}{plant} identical
to the good variety planted --- faster and surer than \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds}.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:8}
Open answer. White \omterm{def:g1:plants-around-us:parts}{roots} appear from the submerged \omterm{def:g1:plants-around-us:parts}{stem},
typically after one to two weeks; the cutting is then ready to
\omterm{def:g1:plants-around-us:plant}{plant} --- a complete new mint, from one parent.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:9}
The \omterm{def:g4:plants-without-seeds:spore}{spore} is bare and cheap: without \omterm{def:g3:animals-through-seasons:active}{food store} or plantlet,
almost every one is lost, so millions must be made. The \omterm{def:g2:from-seed-to-plant:seed}{seed}
is an expensive survival kit, so a dozen well-equipped ones
suffice. They echo the codfish (millions, uncared for) and
the elephant (few, well provided).
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:10}
The strawberries came by runner --- \omterm{def:g4:plants-without-seeds:vegetative}{vegetative reproduction},
one parent, exact copies. The beans came from \omterm{prop:g3:flowers-fruits-seeds:transform}{seeds} made by
\omterm{def:g3:flowers-fruits-seeds:parts}{flowers}, mixing two parents' contributions (\omterm{def:g3:flowers-fruits-seeds:parts}{pollen} from one
\omterm{def:g3:flowers-fruits-seeds:parts}{flower}, \omterm{def:g4:animal-reproduction:sexual}{egg cells} in another), so each bean \omterm{def:g1:plants-around-us:plant}{plant} varies a
little.
\end{solution}

\begin{solution}{exo:g4:plants-without-seeds:11}
\omterm{def:g4:plants-without-seeds:vegetative}{Vegetative reproduction} --- the \omterm{def:g1:plants-around-us:tree}{branch} is a natural cutting.
The flood did the gardener's scissor-work: it cut the shoot,
carried it, and planted it in wet mud, willow's ideal rooting
bed. Rivers line themselves with willows by breaking and
planting them in the same stroke.
\end{solution}
