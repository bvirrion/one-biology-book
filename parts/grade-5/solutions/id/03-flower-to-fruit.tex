\section*{Bab \ref{ch:g5:flower-to-fruit} --- Dari Bunga Menjadi Buah}

\begin{solution}{exo:g5:flower-to-fruit:1}
\omterm{def:g3:flowers-fruits-seeds:parts}{Benang sarinya} \omterm{def:g4:animal-reproduction:sexual}{jantan}: butir \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sarinya} membawa \omterm{def:g5:first-look-at-cells:cell}{sel} kelamin
\omterm{def:g4:animal-reproduction:sexual}{jantan}. \omterm{def:g3:flowers-fruits-seeds:parts}{Putiknya} \omterm{def:g4:animal-reproduction:sexual}{betina}: di dalamnya terbaring \omterm{prop:g5:flower-to-fruit:roles}{bakal biji},
masing-masing memuat sebuah \omterm{def:g4:animal-reproduction:sexual}{sel telur}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:2}
\omterm{def:g5:flower-to-fruit:steps}{Penyerbukan}: sebutir \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} mengembara dari \omterm{def:g3:flowers-fruits-seeds:parts}{benang sari} ke puncak
sebuah \omterm{def:g3:flowers-fruits-seeds:parts}{putik}. \omterm{def:g5:flower-to-fruit:steps}{Pembuahan}: buluh butir itu membawa \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g4:animal-reproduction:sexual}{jantannya} turun
ke sebuah \omterm{prop:g5:flower-to-fruit:roles}{bakal biji}, dan di situ ia bergabung dengan \omterm{def:g4:animal-reproduction:sexual}{sel telurnya}.
\omterm{def:g5:flower-to-fruit:steps}{Penyerbukan} datang lebih dahulu --- perjalanan sebelum pertemuan.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:3}
Dibuat di sebuah \omterm{def:g3:flowers-fruits-seeds:parts}{benang sari}; dibawa (oleh \omterm{prop:g3:sorting-living-things:groups}{serangga} atau angin) ke
puncak sebuah \omterm{def:g3:flowers-fruits-seeds:parts}{putik} --- \omterm{def:g5:flower-to-fruit:steps}{penyerbukan}; menumbuhkan buluh halus turun
menembus \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya}; \omterm{def:g5:first-look-at-cells:cell}{sel} \omterm{def:g4:animal-reproduction:sexual}{jantannya} mencapai sebuah \omterm{prop:g5:flower-to-fruit:roles}{bakal biji} lalu
bergabung dengan \omterm{def:g4:animal-reproduction:sexual}{sel telurnya} --- \omterm{def:g5:flower-to-fruit:steps}{pembuahan}.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:4}
Tiap \omterm{prop:g5:flower-to-fruit:roles}{bakal biji} yang terbuahi menjadi sebutir \omterm{def:g2:from-seed-to-plant:seed}{biji}; \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya}
membengkak menjadi \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya}; \omterm{def:g3:flowers-fruits-seeds:parts}{mahkota bunga} (dan \omterm{def:g3:flowers-fruits-seeds:parts}{benang sarinya}) layu
lalu gugur.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:5}
Tiap \omterm{def:g2:from-seed-to-plant:seed}{bijinya} adalah sebuah \omterm{prop:g5:flower-to-fruit:roles}{bakal biji} terbuahi yang telah menjadi
\omterm{def:g2:from-seed-to-plant:seed}{biji}. Bilik yang kosong berarti \omterm{prop:g5:flower-to-fruit:roles}{bakal biji} yang tidak terbuahi ---
terlalu sedikit \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} yang mendarat, dan \omterm{prop:g3:flowers-fruits-seeds:transform}{buahnya} membengkak
tidak merata di sekitar kekurangannya.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:6}
\omterm{def:g3:flowers-fruits-seeds:parts}{Serbuk sarinya} menumpang angin: sehalus debu, dibuat dalam jumlah amat
besar, dilemparkan ke udara. Harganya adalah pemborosan --- hampir
setiap butirnya mendarat di tempat yang tidak berguna, sehingga jumlah
harus menggantikan ketepatan.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:7}
Tanpa kunjungan, tidak ada \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} di \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya} --- \omterm{def:g5:flower-to-fruit:steps}{penyerbukan},
langkah yang pertama, tidak pernah terjadi, sehingga tidak ada buluh
yang tumbuh dan tidak ada \omterm{prop:g5:flower-to-fruit:roles}{bakal biji} yang terbuahi; \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya} layu
seluruhnya.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:8}
Jawaban terbuka. Yang diharapkan: \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} yang diserbuki dengan tangan
dan ditandai benang jelas lebih sering menghasilkan \omterm{prop:g3:flowers-fruits-seeds:transform}{buah} --- kuasnya
mengerjakan tugas sang lebah.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:9}
\omterm{def:g5:first-look-at-cells:cell}{Sel} \omterm{def:g4:animal-reproduction:sexual}{jantan}: di dalam butir \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} pada \omterm{def:g3:flowers-fruits-seeds:parts}{benang sarinya}. \omterm{def:g4:animal-reproduction:sexual}{Sel telur}:
di dalam \omterm{prop:g5:flower-to-fruit:roles}{bakal biji} di dalam \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya}. \omterm{def:g5:flower-to-fruit:steps}{Pembuahan}: di dasar buluh
\omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari}, di dalam \omterm{prop:g5:flower-to-fruit:roles}{bakal bijinya}. \omterm{def:g5:first-look-at-cells:cell}{Sel} pertama individu barunya: \omterm{def:g4:animal-reproduction:sexual}{sel
telur} terbuahi di dalam \omterm{prop:g5:flower-to-fruit:roles}{bakal biji} --- yang tumbuh menjadi \omterm{def:g1:plants-around-us:plant}{tumbuhan}
mungil di dalam \omterm{def:g2:from-seed-to-plant:seed}{bijinya}, seperti seekor \omterm{def:g1:animals-around-us:animal}{hewan} tumbuh dari \omterm{def:g5:first-look-at-cells:cell}{sel}
pertamanya sendiri.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:10}
Panen yang buruk --- kebanyakan \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya} gugur tanpa \omterm{prop:g3:flowers-fruits-seeds:transform}{buah}. Permesinannya
sehat dan \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya} tidak terluka; yang gagal adalah langkah 1,
\omterm{def:g5:flower-to-fruit:steps}{penyerbukan}: selama dua minggu yang menentukan, para kurirnya tinggal
di rumah, sehingga kebanyakan \omterm{def:g3:flowers-fruits-seeds:parts}{putiknya} tidak pernah menerima \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk
sari} dan kebanyakan \omterm{prop:g5:flower-to-fruit:roles}{bakal bijinya} tidak pernah terbuahi.
\end{solution}

\begin{solution}{exo:g5:flower-to-fruit:11}
Lebah bombusnya menjual \omterm{def:g5:flower-to-fruit:steps}{penyerbukan} --- langkah pengangkutannya:
membawa \omterm{def:g3:flowers-fruits-seeds:parts}{serbuk sari} dari \omterm{def:g3:flowers-fruits-seeds:parts}{benang sari} ke \omterm{def:g3:flowers-fruits-seeds:parts}{putik} di dalam sebuah bangunan
yang tidak dapat dimasuki penyerbuk luar mana pun. \omterm{def:g3:flowers-fruits-seeds:parts}{Serbuk sari} tomat
mudah terguncang lepas, sehingga getaran kipas dan udara yang bergerak
dapat mengerjakan pengangkutan kasar di dalam tiap \omterm{def:g3:flowers-fruits-seeds:parts}{bunganya} --- seorang
kurir mesin. \omterm{def:g3:flowers-fruits-seeds:parts}{Serbuk sari} ceri harus mengembara dari pohon ke pohon,
dari \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} ke \omterm{def:g3:flowers-fruits-seeds:parts}{bunga}, dengan ketepatan yang hanya disediakan tubuh yang
mengunjungi \omterm{def:g3:flowers-fruits-seeds:parts}{bunga} satu per satu; angin di dalam sebuah aula tidak dapat
membidik, sehingga bagi ceri kurir yang hidup itu tidak punya pengganti
yang murah.
\end{solution}
