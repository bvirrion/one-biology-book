\section*{Bab \ref{ch:g9:evidence-of-evolution} --- Bukti Evolusi}

\begin{solution}{exo:g9:evidence-of-evolution:1}
Jejak kehidupan masa lalu yang terawetkan, tersegel di dalam
batuan endapan. Lapisannya bertumpuk menurut urutan
pengendapannya, yang tertua terbawah --- sehingga naik
menembus tumpukannya berarti maju menembus waktunya.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:2}
Faunanya berubah lapisan demi lapisan, dan sebagian besar
\omterm{def:g5:biodiversity:species}{spesies} yang dalam sudah \omterm{prop:g5:biodiversity:loss}{punah}; kelompok besarnya muncul
dalam urutan bersarang pohonnya --- \omterm{prop:g3:sorting-living-things:groups}{ikan} di dalam, kerabat
berkaki empat di atasnya, \omterm{prop:g3:sorting-living-things:groups}{mamalia} dan \omterm{prop:g3:sorting-living-things:groups}{burung} tinggi; dan
peralihannya berdiri pada titik percabangannya.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:3}
\omterm{def:g1:animals-around-us:covering}{Bulu burung} dan sayap --- perangkat \omterm{prop:g3:sorting-living-things:groups}{burungnya} --- pada sebuah
kerangka bergigi dan berekor panjang bertulang --- perangkat
\omterm{prop:g4:classifying-animals:classes}{reptilnya}: kedua tumpukan pada satu \omterm{def:g1:animals-around-us:animal}{hewan}, tepat pada
percabangan yang digambar pohonnya.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:4}
\omterm{def:g3:bones-and-muscles:skeleton}{Tulang} pinggul-dan-tungkai pausnya yang tersembunyi:
keturunan yang berjalan. Sayap tersegel kumbang yang tak
dapat terbang: nenek moyang yang terbang. \omterm{def:g3:bones-and-muscles:skeleton}{Tulang} ekor
manusianya (atau umbai cacingnya yang menyusut): sebuah garis
yang berekor dan bermakanan berbeda. Masing-masing adalah
bagian yang dipertahankan, yang bersaksi tentang sebuah
pembangunan yang dahulu.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:5}
Hitunglah perbedaan huruf sebuah \omterm{def:g9:genes-and-dna:gene}{gen} bersama antarspesies:
makin sedikit, makin dekat kerabatnya. Jaraknya, bila
disusun, merakit sebuah pohon --- dan pohon itu cocok dengan
pohon yang digambar \omterm{def:g3:bones-and-muscles:skeleton}{tulang} dan \omterm{def:g9:evidence-of-evolution:fossils}{fosilnya}.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:6}
Bentuk gelap ngengat berbintiknya yang mengikuti jelaga dan
udara yang bersih; hama yang kebal insektisida dan antibiotik
yang muncul dalam hitungan tahun; ukuran paruh \omterm{prop:g3:sorting-living-things:groups}{burung}
finchnya yang bergeser mengikuti kemarau dan kelimpahannya.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:7}
Rekayasa yang segar dapat memberi tiap \omterm{def:g1:my-body:parts}{anggota gerak}
mekanisme terbaiknya --- penopang bagi sayap, sirip hidrofoil
bagi sirip dayung --- dan tidak akan mengulang jumlah \omterm{def:g3:bones-and-muscles:skeleton}{tulang}
yang sewenang-wenang. Satu susunan itu, yang direntangkan
dengan janggal ke setiap pekerjaan, adalah hasil pembangunan
di atas warisannya, dan bukan hasil rancangan yang bebas.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:8}
Kelasnya: \omterm{def:g2:animals-grow-up:born}{susu}, bernapas dengan udara, \omterm{def:g1:animals-around-us:covering}{bulu} \omterm{def:g5:human-reproduction:pregnancy}{janin} --- pusaka
\omterm{prop:g3:sorting-living-things:groups}{mamalianya} (\cref{ex:g6:classification-kinship:whale}). Sisa
rudimenternya: \omterm{def:g3:bones-and-muscles:skeleton}{tulang} pinggul dan \omterm{def:g1:my-body:parts}{tungkai} yang tersembunyi.
\omterm{def:g9:evidence-of-evolution:fossils}{Fosilnya}: deret berurutan waktu dari \omterm{prop:g3:sorting-living-things:groups}{mamalia} pantai berkaki
empat sampai perenang tanpa \omterm{def:g1:my-body:parts}{tungkai}. Molekulnya: jarak
\omterm{def:g9:genes-and-dna:gene}{gennya} menempatkannya di antara \omterm{prop:g3:sorting-living-things:groups}{mamalia}, dengan kerabat
berkuku sebagai yang terdekat. Empat saksi, satu berkas.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:9}
Karena saksinya tak mungkin bersekongkol: bebatuannya
diendapkan, tubuhnya dibangun dan molekulnya dieja oleh
proses yang tidak punya jalan masuk ke kesaksian satu sama
lain. Saksi yang saling bebas dan sepakat melipatgandakan
kelayakan percayanya --- satu pohon dari tiga sumber adalah
pertemuan yang dihargai caranya.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:10}
Karena klaimnya melarangnya: \omterm{prop:g3:sorting-living-things:groups}{mamalia} tidak boleh muncul di
bawah lapisan nenek moyangnya. Setiap penggalian adalah
kesempatan baru untuk memalsukannya --- dan ketiadaannya,
yang bertahan menembus setiap penggalian yang pernah dibuat,
adalah klaimnya yang lulus ujinya setiap hari.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:11}
\omterm{def:g5:human-reproduction:pregnancy}{Embrio} \omterm{def:g4:classifying-animals:vertebrate}{vertebratanya} sepakat secara mencolok sepanjang pekan
awal penggelaran rancangannya lalu memencar belakangan.
Permesinan bab perkembangannya mengatakan mengapa: petunjuk
awalnya menggelar rancangan tubuh bersamanya --- bab teksnya
yang terdalam dan tertua --- sementara pembacaan yang
kemudian menuliskan kekhususan \omterm{def:g5:biodiversity:species}{spesiesnya}. Teks awal yang
sama, keturunan dalam yang sama.
\end{solution}

\begin{solution}{exo:g9:evidence-of-evolution:12}
Klaimnya: \omterm{def:g5:biodiversity:species}{spesies} berubah dan berbagi keturunan. Ramalannya:
fauna berurutan waktu dalam sarang pohonnya; peralihan pada
percabangannya; sisa rudimenter pembangunan yang dahulu;
jarak molekul yang berjenjang mengikuti \omterm{prop:g6:classification-kinship:kinship}{kekerabatannya}.
Saksinya: bebatuannya menyerahkan urutan dan peralihannya;
tubuhnya menyerahkan rancangan dan sisa rudimenternya;
molekulnya menyerahkan pohon yang cocok; saksi langsungnya
menyerahkan perubahan yang sedang bergerak. Vonisnya:
keturunan yang berubah berdiri di atas kesaksian yang saling
bebas dan bertemu --- lalu menyerahkan pertanyaannya kepada
bab berikutnya: \emph{apa yang menggerakkan perubahannya?}
\end{solution}

\begin{solution}{pb:g9:evidence-of-evolution:1}
\textbf{1.} Urutannya: trilobitnya terbawah, lalu \omterm{prop:g3:sorting-living-things:groups}{ikan}
pertamanya, tapak kerabat \omterm{prop:g3:sorting-living-things:groups}{amfibinya}, \omterm{prop:g4:classifying-animals:classes}{reptilnya},
\emph{Archaeopteryx}, rahang \omterm{prop:g3:sorting-living-things:groups}{mamalia} awalnya. Keterangannya:
``Dibaca ke atas, bebatuannya berganti pemain persis dalam
urutan yang diramalkan pohon keluarganya --- dengan yang
di-antaranya berdiri pada percabangannya.''

\textbf{2.} Keterangannya: ``Satu rancangan \omterm{def:g3:bones-and-muscles:skeleton}{tulang} ---
direntangkan untuk terbang, mendayung, berlari kencang dan
menulis: warisan yang dialihtugaskan, tidak pernah dirancang
ulang.'' Lemarinya: pinggul pausnya, sayap tersegel
kumbangnya, \omterm{def:g3:bones-and-muscles:skeleton}{tulang} ekor manusianya.

\textbf{3.} Pohon yang disiratkan hitungan hurufnya ---
simpanse terdekat, \omterm{ex:g6:producing-our-food:bread}{ragi} terjauh --- harus cocok dengan pohon
para ahli anatomi dan paleontologi di ruangan sebelahnya:
percabangan yang sama, diturunkan secara bebas.

\textbf{4.} Ngengatnya: warna sebuah \omterm{def:g8:reproduction-and-environment:population}{populasi} yang mengikuti
\omterm{def:g6:exploring-our-environment:components}{lingkungannya} dalam hitungan dasawarsa. \omterm{prop:g3:sorting-living-things:groups}{Burung} finchnya:
sebaran paruh yang bergerak mengikuti pasokan \omterm{def:g2:from-seed-to-plant:seed}{bijinya},
terukur dengan jangka sorong --- perubahan yang disuapi
pewarisan, disaksikan langsung.

\textbf{5.} Karena keturunannya bercabang: kita dan monyet
hari ini adalah sepupu pada cabang yang terpisah dari sebuah
percabangan bersama --- tidak ada yang masih hidup menjadi
nenek moyang siapa pun, dan keberadaan seorang sepupu tidak
lebih menyangkal \omterm{prop:g6:classification-kinship:kinship}{kekerabatan} daripada sepupumu menyangkal
kekerabatanmu.

\textbf{6.} \emph{Archaeopteryx}, yang berdiri antara
kerangka \omterm{prop:g4:classifying-animals:classes}{reptil} dan perangkat \omterm{prop:g3:sorting-living-things:groups}{burungnya}; \omterm{def:g9:evidence-of-evolution:fossils}{fosil} paus
bertungkai, yang berdiri antara \omterm{prop:g3:sorting-living-things:groups}{mamalia} pantai dan
perenangnya. Keduanya adalah ramalan sebelum menjadi temuan.

\textbf{7.} Dalam ilmu, ``teori'' menamai sebuah penjelasan
yang diuji sampai ia menata sebuah bidang --- bukan sebuah
terkaan. Yang ditegakkan ruangannya adalah kenyataan
sejarah tentang perubahan dan keturunannya, lewat saksi yang
saling bebas dan bertemu; teori yang sebenarnya adalah
mekanisme yang dijelaskan aula berikutnya.

\textbf{8.} Sebuah kesalahan zaman yang sungguhan --- \omterm{def:g3:bones-and-muscles:skeleton}{tulang}
\omterm{prop:g3:sorting-living-things:groups}{mamalia} di bawah lapisan trilobitnya, terbukti --- akan
mematahkan urutan yang dituntut klaimnya. Bahwa tidak ada
penggalian dalam dua abad yang menghasilkannya adalah
klaimnya yang bertahan pada ujinya yang terberat, setiap
hari, di mana pun penggaliannya berlanjut.

\textbf{9.} ``Kotak-di-dalam-kotak yang kamu jumpai di
sebelah tidak pernah sekadar kemudahan pengarsipan:
\omterm{def:g6:classification-kinship:attribute}{atributnya} bersarang karena keturunannya bercabang ---
\omterm{def:g4:classifying-animals:classification}{klasifikasinya} memang pohon keluarganya, digambar sebelum
seorang pun tahu apa yang digambarnya.''

\textbf{10.} ``Setiap pameran di sayap ini, dan setiap
pengunjung di hadapannya, menuliskan pembangunannya dengan
empat huruf yang sama yang dibaca oleh sandi yang sama:
cabangnya berbeda, bahasa \omterm{def:g1:plants-around-us:tree}{batang pokoknya} tidak pernah.''

\textbf{11.} ``Sayap ini memperlihatkan bahwa \omterm{def:g5:biodiversity:species}{spesies}
berubah, bahwa mereka berbagi keturunan, dan bahwa rekamannya
menyimpan yang di-antaranya. Yang belum dikatakannya adalah
apa yang \emph{menggerakkan} perubahannya --- aula
mekanismenya berikutnya, dan jawabannya berjalan di atas
keragaman, pewarisan dan penahan \omterm{def:g6:exploring-our-environment:components}{lingkungannya}.''

\textbf{12.} ``Seluruh kehidupan adalah satu album keluarga,
dan bebatuannya menyimpan halaman awalnya --- diberi waktu,
sepupu tumbuh sejauh beda paus dan kelelawar, dan tetap
keluarga.''
\end{solution}
