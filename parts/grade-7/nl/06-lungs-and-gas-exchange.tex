\chapter{De longen en de gaswisseling}\label{ch:g7:lungs-and-gas-exchange}

Elke minuut van je leven gaat er ongeveer acht liter lucht door twee
organen die je nooit gezien hebt. \Cref{ch:g4:breathing} schetste de weg
en de wisseling; \cref{ch:g7:how-animals-breathe} plaatste de menselijke
machine tussen de vier van het dierenrijk. Dit hoofdstuk maakt het beeld
af: de ventilatiemachinerie, de luchtzakjes bij hun eigenlijke naam, de
getallen van de wisseling --- en wat rook met het hele werk doet.

\section{De ventilatiemachinerie}

\begin{proposition}[Hoe de lucht bewogen wordt]\label{prop:g7:lungs-and-gas-exchange:ventilation}
De \omterm{def:g7:how-animals-breathe:four}{longen} hebben geen eigen \omterm{def:g3:bones-and-muscles:muscle}{spieren}; de borst pompt ze als een blaasbalg:
\begin{itemize}
  \item \omterm{def:g4:breathing:movements}{inademen} is \emph{werk}: de ribspieren tillen de kast op en maken
        haar breder, en het \emph{middenrif}\index{middenrif} --- de brede
        spiervloer onder de \omterm{def:g7:how-animals-breathe:four}{longen} --- vlakt naar beneden af; de
        verbrede borst rekt de \omterm{def:g7:how-animals-breathe:four}{longen} op, en er stroomt lucht naar
        binnen om ze te vullen;
  \item \omterm{def:g4:breathing:movements}{uitademen} is in rust vooral \emph{loslaten}: de \omterm{def:g3:bones-and-muscles:muscle}{spieren}
        ontspannen, de opgerekte borst veert terug, de lucht stroomt naar
        buiten. (Kaarsen uitblazen voegt spierkracht aan de \omterm{def:g1:animals-around-us:covering}{veer} toe.)
\end{itemize}
Ongeveer $15$ keer per minuut in rust
(\cref{def:g4:breathing:movements}), ongevraagd
(\cref{rem:g5:brain-in-command:more}) --- en op afroep dieper en sneller
(\cref{prop:g7:organs-at-work:supply}).
\end{proposition}
\admitted

\begin{example}[De blaasbalg voelen]\label{ex:g7:lungs-and-gas-exchange:feel}
Handen op je \omterm{def:g3:bones-and-muscles:skeleton}{ribben}, adem diep in: de kast komt onder je handpalmen
omhoog en wordt breder (\cref{ex:g4:breathing:watch}). Hijg nu als een
hond: die snelle oppervlakkige pufjes zijn bijna helemaal \omterm{prop:g7:lungs-and-gas-exchange:ventilation}{middenrif}. De
hik is, voor de volledigheid, een reeks schokkerige valse starts van het
\omterm{prop:g7:lungs-and-gas-exchange:ventilation}{middenrif} --- de blaasbalg die op eigen houtje misvuurt.
\end{example}

\section{De longblaasjes en de wisseling}

\begin{definition}[Longblaasjes]\label{def:g7:lungs-and-gas-exchange:alveoli}
De piepkleine luchtzakjes uit \cref{def:g4:breathing:organs} zijn de
\emph{longblaasjes}\index{longblaasje}: honderden miljoenen dunwandige
bolletjes in trosjes aan de fijnste uiteinden van de luchtwegen, elk
omwikkeld met fijne \omterm{def:g4:heart-and-blood:vessels}{bloedvaten}. Hun opgetelde oppervlak is de halve
tennisbaan uit \cref{rem:g4:breathing:sponge} --- de menselijke uitvoering
van het groot-dun-vochtige bestek uit
\cref{prop:g7:how-animals-breathe:spec}.
\end{definition}

\begin{proposition}[De wisseling, in getallen]\label{prop:g7:lungs-and-gas-exchange:numbers}
Vergelijk de lucht die naar binnen gaat met de lucht die naar buiten
komt: ingeademde lucht bestaat voor ongeveer $21$ procent uit \omterm{def:g7:what-is-respiration:respiration}{zuurstof} en
is vrijwel vrij van \omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide}; uitgeademde lucht is gezakt tot
ongeveer $16$ procent \omterm{def:g7:what-is-respiration:respiration}{zuurstof} en opgelopen tot ongeveer $4$ procent
\omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} (en is bovendien warm en vochtig ---
\cref{ex:g4:breathing:mirror}). Die verschillen zijn de wisseling: door de
wanden van de \omterm{def:g7:lungs-and-gas-exchange:alveoli}{longblaasjes} steekt \omterm{def:g7:what-is-respiration:respiration}{zuurstof} over naar het \omterm{prop:g4:journey-of-food:absorb}{bloed} en
\omterm{def:g7:what-is-respiration:respiration}{koolstofdioxide} terug
(\cref{prop:g4:breathing:exchange}), elke minuut van elke dag.
\end{proposition}
\admitted

\begin{example}[De boekhouding klopt]\label{ex:g7:lungs-and-gas-exchange:accounting}
Acht liter per minuut, met een twintigste van de \omterm{def:g7:what-is-respiration:respiration}{zuurstof} van elke
ademteug achtergehouden: die rekensom levert ruwweg de zuurstofrekening
van het rustende lichaam --- en bij inspanning, met een verdrievoudigde en
verdiepte \omterm{def:g7:what-is-respiration:respiration}{ademhaling} en versnelde rondes van het \omterm{def:g4:heart-and-blood:heart}{hart}
(\cref{prop:g7:organs-at-work:supply}), vermenigvuldigt de levering zich
tot ze bij de rekening van de \omterm{def:g3:bones-and-muscles:muscle}{spieren} past. De getallen van de machine en
de boekhouding van het lichaam kloppen; het zijn dezelfde boeken, aan twee
loketten bijgehouden.
\end{example}

\section{De machinerie misbruikt: rook}

\begin{proposition}[Wat rook met het werk doet]\label{prop:g7:lungs-and-gas-exchange:smoke}
Tabaksrook valt elke verdieping van dit hoofdstuk aan
(\cref{prop:g5:healthy-choices:substances} gaf de waarschuwing; hier is de
techniek):
\begin{itemize}
  \item zijn teer beslaat en prikkelt de luchtwegen: de schoonmakende
        oppervlakken haperen, en de ``rokershoest'' is de overbelaste
        bezem van de luchtwegen;
  \item de wanden van de \omterm{def:g7:lungs-and-gas-exchange:alveoli}{longblaasjes}, chronisch geprikkeld, breken af:
        blaasjes smelten samen tot minder, grotere, slappere ruimtes ---
        er gaat \emph{oppervlak verloren}, en daarmee wisseling (het
        bestek, gesloopt);
  \item een van de gassen van de rook bezet de zuurstofstoelen van het
        \omterm{prop:g4:journey-of-food:absorb}{bloed} en verdringt de \omterm{def:g7:what-is-respiration:respiration}{zuurstof} van haar eigen vervoer;
  \item en de luchtwegen worden nauwer --- elke ademteug door de borst
        van een roker is een teug door een half verstopte blaasbalg.
\end{itemize}
De schade is traag, cumulatief en maar deels omkeerbaar: de \omterm{def:g7:how-animals-breathe:four}{long} die
nooit begonnen is, houdt haar halve tennisbaan heel
(\cref{rem:g5:healthy-choices:never}).
\end{proposition}
\admitted

\begin{example}[Twee traptesten]\label{ex:g7:lungs-and-gas-exchange:stairs}
De drie trappen uit \cref{ch:g7:organs-at-work}, genomen door twee
volwassenen van dezelfde leeftijd en met dezelfde trainingsgewoonten,
waarvan er één al lang rookt: de roker komt hijgend boven bij een
\omterm{prop:g4:heart-and-blood:pulse}{polsslag} die de ander alleen sprintend haalt. Minder oppervlak, bezette
stoelen, nauwere luchtwegen --- de bevoorradingsketen betaalt de rekening
van \cref{prop:g7:organs-at-work:bill} door een gesmoorde inlaat.
Fitheidstests lezen de geschiedenis van de \omterm{def:g7:how-animals-breathe:four}{longen} net zo betrouwbaar als
welke scan ook.
\end{example}

\begin{method}[De machinerie onderhouden]\label{met:g7:lungs-and-gas-exchange:keep}
\Cref{met:g4:breathing:care}, bijgewerkt met de redenen uit dit
hoofdstuk:
\begin{enumerate}
  \item geen rook --- niet die van jou en niet die van anderen: verloren
        oppervlak komt nauwelijks terug;
  \item lucht de kamers (\cref{exo:g4:breathing:10}) en kies voor sport
        liever schone buitenlucht;
  \item oefen de blaasbalg: inspanning waarbij je diep ademt bouwt de
        ademhalingsspieren op en houdt het volledige oppervlak
        geventileerd;
  \item adem door je \omterm{def:g1:the-five-senses:sense}{neus} in kou en stof
        (\cref{exo:g4:breathing:7}) --- het filter beschermt de bezem.
\end{enumerate}
\end{method}

\section{Probleem: Het dossier van de ademhalingsmachine}

\begin{problem}[{Weekendprobleem --- een technische inspectie van de
menselijke blaasbalg}]\label{pb:g7:lungs-and-gas-exchange:1}
Een technische klas ``inspecteert'' de menselijke ademhalingsmachine op
papier: inlaat, pomp, uitwisselingsoppervlak, transportaansluiting en
onderhoudsdossier.

\medskip\noindent\textbf{Deel I --- De pomp.}
\begin{enumerate}
  \item Stel de cyclus van de pomp op: de twee fasen, de bewegers van
        elke fase, en welke fase in rust \omterm{prop:g3:balanced-diet:jobs}{energie} kost.
  \item De pomp heeft geen contact met de eigen \omterm{def:g3:bones-and-muscles:muscle}{spier} van het gepompte
        orgaan --- de \omterm{def:g7:how-animals-breathe:four}{longen} hebben er geen. Hoe beweegt het verbreden van
        de borst dan überhaupt lucht?
  \item Geef de rustfrequentie van de pomp en haar twee \omterm{def:g4:adapted-to-their-world:adaptation}{aanpassingen} bij
        inspanning
        (\cref{prop:g7:organs-at-work:supply}).
  \item Hijgen, zuchten, kaarsen uitblazen: wijs elk zijn plaats in de
        mechanica van de cyclus toe.
\end{enumerate}

\medskip\noindent\textbf{Deel II --- Het
uitwisselingsoppervlak.}
\begin{enumerate}[resume]
  \item Specificeer het oppervlak: naam, aantal, wand, omwikkeling,
        opgetelde omvang.
  \item Toets het bestek punt voor punt aan
        \cref{prop:g7:how-animals-breathe:spec}.
  \item De inlaatlucht en de uitlaatlucht verschillen met de percentages
        uit het hoofdstuk. Bereken het zuurstofverschil als een deel van
        de $21$ procent van de inlaat --- ruwweg welk aandeel van de
        ingeademde \omterm{def:g7:what-is-respiration:respiration}{zuurstof} wordt achtergehouden?
  \item Waarom moet de uitlaat warm en vochtig blijven --- van welke
        eigenschap van het oppervlak is dat de prijs?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Het
onderhoudsdossier.}
\begin{enumerate}[resume]
  \item Archiveer de vier aanvallen van rook onder: inlaat, pomp,
        oppervlak, transport. Eén regel per stuk.
  \item Het dossier noteert: ``verlies van oppervlak is verlies van
        kapitaal''. Leg dat uit, tegenover de hoest en de verkoudheden die
        lopende reparaties zijn.
  \item Schrijf het onderhoudsschema ---
        \cref{met:g7:lungs-and-gas-exchange:keep} --- als vier
        richtlijnen van een ingenieur, met redenen.
  \item Sluit de inspectie af: één zin over waarom juist deze machine, als
        enige van de pompen en grenzen van het lichaam, half onder de
        vrijwillige controle van de eigenaar staat --- en waar die
        controle voor dient (spraak, zang, de trage uitademing van de
        duiker).
\end{enumerate}
\end{problem}
