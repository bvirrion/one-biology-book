\chapter{Mendelian Genetics}\label{ch:g11:mendelian-genetics}

Gregor Mendel tracked single visible traits through generations of pea plants and
inferred particles of heredity --- what we now call \omterm{def:g11:gene-expression:gene}{genes} and \omterm{def:g11:mendelian-genetics:allele}{alleles}. This
chapter builds the vocabulary of \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} and \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}, dominance, segregation,
independent assortment, and the Punnett square as a bookkeeping tool.

\section{Genes, alleles, genotype, phenotype}

\begin{definition}[Allele]\label{def:g11:mendelian-genetics:allele}
An \emph{allele}\index{allele} is one of the alternative versions of a \omterm{def:g11:gene-expression:gene}{gene}. In
a \omterm{def:g11:meiosis-life-cycles:ploidy}{diploid} individual, the two alleles of a \omterm{def:g11:gene-expression:gene}{gene} (one on each homologue) may be
identical or different.
\end{definition}

\begin{definition}[Genotype and phenotype]\label{def:g11:mendelian-genetics:geno-pheno}
The \emph{genotype}\index{genotype} is the allelic composition of an individual
(for one \omterm{def:g11:gene-expression:gene}{gene} or many). The \emph{phenotype}\index{phenotype} is the observable
trait --- morphological, physiological, or molecular --- that results from the
genotype interacting with the environment.
\end{definition}

\begin{definition}[Homozygous and heterozygous]\label{def:g11:mendelian-genetics:zygous}
An individual is \emph{homozygous}\index{homozygous} for a \omterm{def:g11:gene-expression:gene}{gene} if both \omterm{def:g11:mendelian-genetics:allele}{alleles}
are the same, and \emph{heterozygous}\index{heterozygous} if they differ. We
often write $AA$ or $aa$ for homozygotes and $Aa$ for a heterozygote.
\end{definition}

\begin{definition}[Dominance]\label{def:g11:mendelian-genetics:dominance}
When the heterozygote $Aa$ shows the same \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype} as $AA$, \omterm{def:g11:mendelian-genetics:allele}{allele} $A$ is
\emph{dominant}\index{dominant allele} and $a$ is \emph{recessive}\index{recessive
allele}. Dominance is a relationship between \omterm{def:g11:mendelian-genetics:allele}{alleles} for a given \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}; it
is not ``strength'' in any absolute sense.
\end{definition}

\begin{example}[Pea seed colour]\label{ex:g11:mendelian-genetics:peas}
In Mendel's peas, yellow seeds ($Y$) are \omterm{def:g11:mendelian-genetics:dominance}{dominant} to green ($y$). \omterm{def:g11:mendelian-genetics:geno-pheno}{Genotypes}
$YY$ and $Yy$ are yellow; only $yy$ is green. A yellow seed can be \omterm{def:g11:mendelian-genetics:zygous}{homozygous} or
\omterm{def:g11:mendelian-genetics:zygous}{heterozygous} --- the \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype} alone does not always reveal the \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype}.
\end{example}

\begin{remark}
Not all traits follow simple complete dominance. \omterm{def:g11:mendelian-genetics:incomplete}{Incomplete dominance},
\omterm{def:g11:mendelian-genetics:codominance}{codominance} and \omterm{def:g11:mendelian-genetics:ABO}{multiple alleles} are developed later in this chapter;
polygenic inheritance appears in \cref{ch:g11:human-genetics}. Mendel's
\omterm{def:g10:scientific-biology:theory}{laws} of segregation and assortment still organize the \omterm{def:g11:gene-expression:gene}{single-gene} cases.
\end{remark}

\section{Segregation}

\begin{proposition}[Law of segregation]\label{prop:g11:mendelian-genetics:segregation}
During gamete formation, the two \omterm{def:g11:mendelian-genetics:allele}{alleles} of a \omterm{def:g11:gene-expression:gene}{gene} separate so that each gamete
receives one \omterm{def:g11:mendelian-genetics:allele}{allele}. Fertilization restores a pair of \omterm{def:g11:mendelian-genetics:allele}{alleles} at random from the
two parents.
\end{proposition}
\admitted

\begin{remark}
Segregation is the genetic consequence of \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}: homologous \omterm{def:g11:dna-replication:chromosome}{chromosomes} ---
and thus the \omterm{def:g11:mendelian-genetics:allele}{alleles} they carry --- go to different gametes
(\cref{ch:g11:meiosis-life-cycles}).
\end{remark}

\begin{method}[Monohybrid Punnett square]\label{met:g11:mendelian-genetics:mono}
\begin{enumerate}
  \item Write parental \omterm{def:g11:mendelian-genetics:geno-pheno}{genotypes} and the gametes each can produce (equal
        frequency if fair \omterm{def:g11:meiosis-life-cycles:meiosis}{meiosis}).
  \item Draw a grid with one parent's gametes on top and the other's on the side.
  \item Fill each \omterm{def:g10:the-cell:cell}{cell} with the zygote \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype}; read \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes} from dominance
        rules.
  \item Convert counts to ratios (\omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} and \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}).
\end{enumerate}
\end{method}

\begin{example}[Monohybrid cross]\label{ex:g11:mendelian-genetics:mono}
Cross $Yy \times Yy$ (both yellow). Gametes $Y$ and $y$ from each parent.
Offspring: $\tfrac14\,YY$, $\tfrac12\,Yy$, $\tfrac14\,yy$ --- phenotypic ratio
$3$ yellow $:$ $1$ green among large samples.
\end{example}

\section{Test crosses}

\begin{definition}[Test cross]\label{def:g11:mendelian-genetics:testcross}
A \emph{test cross}\index{test cross} mates an individual of \omterm{def:g11:mendelian-genetics:dominance}{dominant} \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}
with a \omterm{def:g11:mendelian-genetics:zygous}{homozygous} \omterm{def:g11:mendelian-genetics:dominance}{recessive} ($aa$). The offspring \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes} reveal whether the
tested parent was $AA$ (all \omterm{def:g11:mendelian-genetics:dominance}{dominant} \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}) or $Aa$ (about half \omterm{def:g11:mendelian-genetics:dominance}{recessive}
\omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}).
\end{definition}

\begin{example}[Yellow seed mystery]\label{ex:g11:mendelian-genetics:test}
Yellow $Y\_ \times yy$: if any green offspring appear, the yellow parent was
$Yy$. If many offspring are all yellow, the parent was likely $YY$.
\end{example}

\section{Independent assortment}

\begin{proposition}[Law of independent assortment]\label{prop:g11:mendelian-genetics:assortment}
\omterm{def:g11:mendelian-genetics:allele}{Alleles} of \omterm{def:g11:gene-expression:gene}{genes} on different \omterm{def:g11:dna-replication:chromosome}{chromosomes} (more precisely, unlinked \omterm{def:g11:gene-expression:gene}{genes})
segregate independently: the \omterm{def:g11:mendelian-genetics:allele}{allele} a gamete gets at one locus does not bias the
\omterm{def:g11:mendelian-genetics:allele}{allele} it gets at another.
\end{proposition}
\admitted

\begin{remark}
Independent assortment reflects the independent orientation of different
\omterm{prop:g11:meiosis-life-cycles:prophaseI}{bivalents} at metaphase I. \omterm{def:g11:gene-expression:gene}{Genes} close together on the same \omterm{def:g11:dna-replication:chromosome}{chromosome} are
\emph{linked}\index{genetic linkage} and do not assort independently --- they
recombine only via \omterm{prop:g11:meiosis-life-cycles:prophaseI}{crossing over}.
\end{remark}

\begin{method}[Dihybrid cross]\label{met:g11:mendelian-genetics:dihybrid}
For two unlinked \omterm{def:g11:gene-expression:gene}{genes}, a double heterozygote $AaBb$ produces four gamete types
$AB$, $Ab$, $aB$, $ab$ in equal frequency. Crossing $AaBb \times AaBb$ yields
the classic $9:3:3:1$ phenotypic ratio when both \omterm{def:g11:gene-expression:gene}{genes} show complete dominance
and independent assortment.
\end{method}

\begin{example}[Mendel's dihybrid ratio]\label{ex:g11:mendelian-genetics:dihybrid}
Seed colour $Y/y$ and shape $R/r$ (round \omterm{def:g11:mendelian-genetics:dominance}{dominant} to wrinkled). Selfing
$YyRr$ gives approximately $9$ yellow round $:$ $3$ yellow wrinkled $:$
$3$ green round $:$ $1$ green wrinkled among large F$_2$ samples.
\end{example}

\begin{proposition}[Product rule for unlinked genes]\label{prop:g11:mendelian-genetics:product}
If traits assort independently, the probability of a combined \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} (or
\omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype}) is the product of the single-locus probabilities. Example: probability
of $aa$ and $bb$ from $AaBb \times AaBb$ is
$\tfrac14 \times \tfrac14 = \tfrac1{16}$.
\end{proposition}
\admitted

\section{Beyond complete dominance}

\begin{definition}[Incomplete dominance]\label{def:g11:mendelian-genetics:incomplete}
Under \emph{incomplete dominance}\index{incomplete dominance}, the heterozygote
shows a \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype} intermediate between the two homozygotes. \omterm{def:g11:mendelian-genetics:allele}{Alleles} still
segregate $1:2:1$; only the mapping from \omterm{def:g11:mendelian-genetics:geno-pheno}{genotype} to \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype} differs from
complete dominance.
\end{definition}

\begin{example}[Snapdragon flower colour]\label{ex:g11:mendelian-genetics:snapdragon}
In some snapdragons, $C^{R}C^{R}$ is red, $C^{W}C^{W}$ is white, and
$C^{R}C^{W}$ is pink. Crossing two pink plants yields approximately
$1$ red $:$ $2$ pink $:$ $1$ white --- the same genotypic ratios as
$Aa \times Aa$, now visible as three \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotypes}.
\end{example}

\begin{definition}[Codominance]\label{def:g11:mendelian-genetics:codominance}
Under \emph{codominance}\index{codominance}, both \omterm{def:g11:mendelian-genetics:allele}{alleles} are fully expressed
in the heterozygote: the \omterm{def:g11:mendelian-genetics:geno-pheno}{phenotype} shows both products rather than a blended
intermediate.
\end{definition}

\begin{example}[Roan cattle]\label{ex:g11:mendelian-genetics:roan}
In some cattle, red and white coat \omterm{def:g11:mendelian-genetics:allele}{alleles} are codominant: the heterozygote is
\emph{roan}\index{roan}, with both red and white hairs visible, not a uniform
pink blend.
\end{example}

\begin{definition}[Multiple alleles and ABO]\label{def:g11:mendelian-genetics:ABO}
A \omterm{def:g11:gene-expression:gene}{gene} may have more than two \omterm{def:g11:mendelian-genetics:allele}{alleles} in the \omterm{def:g10:ecosystems:levels}{population}
(\emph{multiple alleles}\index{multiple alleles}). Human ABO blood groups are
determined by three common \omterm{def:g11:mendelian-genetics:allele}{alleles} of one \omterm{def:g11:gene-expression:gene}{gene}: $I^{A}$, $I^{B}$ and $i$.
$I^{A}$ and $I^{B}$ are codominant with each other; both are \omterm{def:g11:mendelian-genetics:dominance}{dominant} to $i$.
\end{definition}

\begin{proposition}[ABO phenotypes]\label{prop:g11:mendelian-genetics:ABO}
\begin{center}\small
\begin{tabular}{ll}
\textbf{\omterm{def:g11:mendelian-genetics:geno-pheno}{Genotype}(s)} & \textbf{Blood type} \\
$I^{A}I^{A}$, $I^{A}i$ & A \\
$I^{B}I^{B}$, $I^{B}i$ & B \\
$I^{A}I^{B}$ & AB \\
$ii$ & O \\
\end{tabular}
\end{center}
Type AB expresses both A and B antigens (\omterm{def:g11:mendelian-genetics:codominance}{codominance}); type O expresses neither.
\end{proposition}
\admitted

\begin{example}[ABO cross]\label{ex:g11:mendelian-genetics:ABO-cross}
Parents $I^{A}i$ (type A) and $I^{B}i$ (type B) can produce children of types
A, B, AB or O --- each with probability $\tfrac14$ if gametes are equally
likely. A child of type O proves both parents carried $i$.
\end{example}

\section{Testing ratios: the chi-square idea}

\begin{method}[Chi-square goodness of fit (idea)]
\label{met:g11:mendelian-genetics:chi-square}
Mendelian models predict ratios (e.g.\ $3:1$ or $9:3:3:1$). Real counts
fluctuate by chance. A simple comparison:
\begin{enumerate}
  \item From the total number of offspring $N$ and the predicted fractions,
        compute \emph{expected} counts $E$ in each \omterm{def:g10:biodiversity-classification:hierarchy}{class}.
  \item Compare with \emph{observed} counts $O$.
  \item Form $\chi^{2} = \sum \dfrac{(O-E)^{2}}{E}$ over classes.
  \item Large $\chi^{2}$ means the data sit far from the model (perhaps wrong
        model, linkage, or scoring error); small $\chi^{2}$ is consistent with
        chance deviation. Full critical values and degrees of freedom belong
        to statistics courses --- the \omterm{def:g10:scientific-biology:science}{biology} idea is that ratios are
        \emph{tested}, not assumed from a handful of offspring.
\end{enumerate}
\end{method}

\begin{example}[Checking a $3:1$]\label{ex:g11:mendelian-genetics:chi}
From $Yy \times Yy$, $80$ seeds: $62$ yellow, $18$ green. Expected under
$3:1$: $60$ yellow, $20$ green. Deviations are small relative to sample size,
so the data are compatible with the Mendelian model; with only $8$ seeds the
same percentages would be uninformative.
\end{example}

\begin{problem}[Breeding a true-breeding line]\label{pb:g11:mendelian-genetics:1}
A plant species has flower colour $R$ red \omterm{def:g11:mendelian-genetics:dominance}{dominant} to $r$ white, and leaf edge
$S$ smooth \omterm{def:g11:mendelian-genetics:dominance}{dominant} to $s$ serrated. The \omterm{def:g11:gene-expression:gene}{genes} are unlinked. You have a red,
smooth plant that might be \omterm{def:g11:mendelian-genetics:zygous}{heterozygous} at either locus, and true-breeding white
serrated plants ($rrss$).
