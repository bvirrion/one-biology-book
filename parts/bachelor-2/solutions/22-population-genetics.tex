\section*{Chapter \ref{ch:b2:population-genetics} --- Population Genetics}

\begin{solution}{exo:b2:population-genetics:1}
$p(M) = (1280 + 320)/2000 = 0.8$, $q(N) = 0.2$. Expected $MM$:
$0.64\times 1000 = 640$; $MN$: $2\times 0.8\times 0.2\times 1000 =
320$; $NN$: 40 --- exactly the observed numbers. The population is
in equilibrium at this locus.
\end{solution}

\begin{solution}{exo:b2:population-genetics:2}
$q = \sqrt{1/20000} = 0.0071$; carriers $2pq \approx 0.014$, one
person in 70. Carriers outnumber affected people by $2p/q \approx
280$ to one.
\end{solution}

\begin{solution}{exo:b2:population-genetics:3}
\omterm{def:b2:population-genetics:fitness}{Fitness}: the expected number of offspring of a \omterm{def:b2:meiosis-heredity:vocabulary}{genotype} relative to
the best. \omterm{def:b2:population-genetics:fitness}{Selection coefficient} $s$: the \omterm{def:b2:population-genetics:fitness}{fitness} deficit, $w = 1 -
s$. Heritability $h^{2}$: the fraction of the phenotypic variance
that is additive genetic, equivalently the slope of offspring on
parents. Forces: \omterm{def:b2:genome-diversification:mutation}{mutation}, selection, drift, migration, non-random
mating (the last changes \omterm{def:b2:meiosis-heredity:vocabulary}{genotype}, not \omterm{def:b2:meiosis-heredity:vocabulary}{allele}, frequencies).
\end{solution}

\begin{solution}{exo:b2:population-genetics:4}
Directional: melanism in the peppered moth, agent the birds hunting
by sight on soot-darkened trunks. Stabilising: human birth weight,
agent the mortality of very small and very large babies. Balancing:
sickle cell, agents malaria (against $AA$) and anaemia (against
$SS$) together.
\end{solution}

\begin{solution}{exo:b2:population-genetics:5}
$q_t = q_0/(1 + tq_0)$: halving takes $1/q_0 = 50$ generations;
reaching $0.001$ takes $1/0.001 - 1/0.02 = 950$ generations. Almost
all copies of the \omterm{def:b2:meiosis-heredity:vocabulary}{allele} sit in heterozygotes, invisible to
selection; preventing the affected from reproducing (which nature
already does, since they are dead) changes nothing measurable in a
human lifetime.
\end{solution}

\begin{solution}{exo:b2:population-genetics:6}
$p = 0.3$: $\bar w = 0.09 + 0.42 + 0.8\times 0.49 = 0.902$; $p' =
(0.09 + 0.21)/0.902 = 0.333$; $\Delta p = +0.033$. $p = 0.9$: $\bar w
= 0.998$, $p' = 0.9018$, $\Delta p = +0.0018$. At $p = 0.9$ only
$q^{2} = 0.01$ of the population is $aa$: the \omterm{def:b2:meiosis-heredity:vocabulary}{allele} selection acts
against is hidden in heterozygotes, and $\Delta p = spq^{2}/\bar w$
carries the factor $q^{2}$.
\end{solution}

\begin{solution}{exo:b2:population-genetics:7}
With $s = 0.12$ against $AA$ and $t = 0.8$ against $SS$, $\hat q =
s/(s + t) = 0.12/0.92 = 0.13$. Affected births $\hat q^{2} = 0.017$,
about one child in 60 --- the price of the protection the
heterozygotes enjoy.
\end{solution}

\begin{solution}{exo:b2:population-genetics:8}
\omterm{def:b2:meiosis-heredity:vocabulary}{Recessive} lethal: $\hat q = \sqrt{\mu} = \sqrt{2\times 10^{-5}} =
0.0045$; affected births $\hat q^{2} = \mu = 2\times 10^{-5}$, one in
\num{50000}. \omterm{def:b2:meiosis-heredity:vocabulary}{Dominant} lethal: every copy is removed in the
generation it appears, so affected births are simply the new
\omterm{def:b2:genome-diversification:mutation}{mutations}, $2\mu = 4\times 10^{-5}$ (two gametes per birth), one in
\num{25000} --- the \omterm{def:b2:meiosis-heredity:vocabulary}{recessive} hides its \omterm{def:b2:meiosis-heredity:vocabulary}{alleles} two hundred to one,
the \omterm{def:b2:meiosis-heredity:vocabulary}{dominant} hides nothing.
\end{solution}

\begin{solution}{exo:b2:population-genetics:9}
$H_t = 0.5\,(1 - 1/100)^{t}$: after 10 generations $0.45$; after 50,
$0.30$; after 200, $0.067$. For $(1 - 1/2N)^{100} = 0.9$:
$100/(2N) \approx -\ln 0.9 = 0.105$, so $N \approx 475$, about 500
breeding individuals.
\end{solution}

\begin{solution}{exo:b2:population-genetics:10}
Twenty gene copies: $P(\text{absent}) = 0.9^{20} = 0.12$. A single
copy among 20 has frequency $0.05$, and a neutral \omterm{def:b2:meiosis-heredity:vocabulary}{allele} fixes with
probability equal to its frequency: $0.05$. On a mainland of, say,
\num{100000} birds a single copy has probability $5\times 10^{-6}$
of fixing. The island is a place where rare \omterm{def:b2:meiosis-heredity:vocabulary}{alleles} are lost and
where those that survive can take over: drift both impoverishes and
differentiates.
\end{solution}

\begin{solution}{exo:b2:population-genetics:11}
$R = h^{2}S = 0.3\times 1.4\times \qty{1000}{L} = \qty{420}{L}$ per
generation. With bulls from the top \qty{1}{\%}, the average
differential is $(1.4 + 2.7)/2 = 2.05$ standard deviations and $R =
0.3\times 2050 = \qty{615}{L}$. The response slows because selection
uses up the additive variance --- favourable \omterm{def:b2:meiosis-heredity:vocabulary}{alleles} go to fixation
and $h^{2}$ falls --- and because the correlated changes in other
traits (fertility, health) impose costs that selection on yield
alone does not see.
\end{solution}

\begin{solution}{exo:b2:population-genetics:12}
Selection sees \omterm{def:b2:meiosis-heredity:vocabulary}{phenotypes}. A \omterm{def:b2:meiosis-heredity:vocabulary}{recessive} \omterm{def:b2:meiosis-heredity:vocabulary}{allele} in a heterozygote
makes no \omterm{def:b2:meiosis-heredity:vocabulary}{phenotype}, so selection cannot remove it, and its
frequency falls as $1/t$, slower and slower. A neutral \omterm{def:b2:genome-diversification:mutation}{mutation}
makes no \omterm{def:b2:population-genetics:fitness}{fitness} difference; its fate is drift's alone, and it
fixes with probability $1/2N$. A character with zero heritability
may be strongly selected --- the survivors may differ greatly from
the population --- yet the next generation is unchanged, because
the differences were not inherited. Selection can act only on
heritable differences that make a difference to \omterm{def:b2:population-genetics:fitness}{fitness}; everything
else evolves by chance or not at all.
\end{solution}

\begin{solution}{pb:b2:population-genetics:1}
\textbf{1.} Melanic fraction $1 - q^{2} = 1 - 0.999^{2} = 0.002$,
one moth in 500; the fraction of $M$ \omterm{def:b2:meiosis-heredity:vocabulary}{alleles} in heterozygotes is
$2pq/(2pq + 2p^{2}) = q = 0.999$: essentially all.
\textbf{2.} With $\bar w = 1 - sq^{2}$, the pale \omterm{def:b2:meiosis-heredity:vocabulary}{allele}'s copies
after selection are $pq\cdot 1 + q^{2}(1 - s)$ out of $\bar w$, so $q'
= (pq + q^{2} - sq^{2})/(1 - sq^{2}) = q(1 - sq)/(1 - sq^{2})$.
\textbf{3.} $q_1 = 0.999\times(1 - 0.2997)/(1 - 0.2994) = 0.99857$:
a change of $-0.0014$.
\textbf{4.} $q = \sqrt{0.1} = 0.32$.
\textbf{5.} The recursion, iterated, takes 28 generations for $s =
0.3$; the approximation gives the same order (the change is
$-0.0003$ at first, then accelerates as $p$ grows). The observed 50
generations correspond to a smaller $s$, about $0.2$ (the recursion
gives 45 generations for $s = 0.2$): selection of \qtyrange{20}{30}{\%}
per generation, enormous by the standards of most \omterm{def:b2:meiosis-heredity:vocabulary}{alleles}.
\textbf{6.} With $w_{MM} = w_{Mm} = 1 - s$, $w_{mm} = 1$: $\bar w = 1
- s(1 - q^{2})$ and $p' = p(1 - s)/\bar w$, so $\Delta p = p[(1 - s) -
\bar w]/\bar w = -spq^{2}/\bar w$. The pale form is favoured only as
the homozygote $mm$, a fraction $q^{2} = 0.1$ at the start: the
return begins slowly. As $q \to 1$, $\Delta p \approx -sp$ and $M$
declines exponentially, by a factor $1 - s$ each generation: a
\omterm{def:b2:meiosis-heredity:vocabulary}{dominant} \omterm{def:b2:meiosis-heredity:vocabulary}{allele} has nowhere to hide. The rise was the mirror image:
$\Delta p \approx +sp$ while $M$ was rare (exponential start), then
$\Delta q \approx -sq^{2}p \to$ a crawl as the pale \omterm{def:b2:meiosis-heredity:vocabulary}{allele} hid in
heterozygotes. The recursion gives 27 generations for the return
from $p = 0.68$ to $0.001$, 16 of them to reach \qty{5}{\%} melanic.
\textbf{7.} $24$ gene copies: $0.95^{24} = 0.29$.
\textbf{8.} $H_1 = 0.4\,(1 - 1/24) = 0.383$.
\textbf{9.} $H_{100} = 0.383\,(1 - 1/80)^{100} = 0.383\times 0.284 =
0.109$: \qty{27}{\%} of the mainland's $0.4$.
\textbf{10.} $P(\text{fixation}) = 1/(2N) = 1/80 = 0.0125$; expected
time to fixation $4N = 160$ generations.
\textbf{11.} Without inbreeding, $q^{2} = 0.0025$; with $F = 0.3$,
$q^{2} + Fpq = 0.0025 + 0.3\times 0.95\times 0.05 = 0.017$: seven
times more affected birds.
\textbf{12.} One migrant per generation ($m = 1/40$) is enough to
prevent the island from diverging at neutral loci and to
replenish the \omterm{def:b2:meiosis-heredity:vocabulary}{alleles} drift has lost; it also imports the
mainland's deleterious \omterm{def:b2:meiosis-heredity:vocabulary}{alleles} at their mainland frequency, which,
under the island's inbreeding, are exposed as homozygotes --- the
load is set by the balance of immigration and exposure.
\textbf{13.} $\hat q = \sqrt{10^{-5}} = 0.0032$; affected births $\hat
q^{2} = 10^{-5}$.
\textbf{14.} $\hat q = \sqrt{10^{-5}/0.1} = 0.01$; affected births
$10^{-4}$: a tenfold weaker selection lets the \omterm{def:b2:meiosis-heredity:vocabulary}{allele} become three
times commoner and the disease ten times.
\textbf{15.} $\hat p = \mu/(hs) = 10^{-5}/0.05 = 2\times 10^{-4}$;
carriers $2\hat p = 4\times 10^{-4}$.
\textbf{16.} From $\hat q = 0.0032$ to $0.01$: the \omterm{def:b2:meiosis-heredity:vocabulary}{allele} rises at a
rate set by \omterm{def:b2:genome-diversification:mutation}{mutation}, $\mu = 10^{-5}$ per generation, so the
approach takes of order $\Delta q/\mu \approx 700$ generations ---
\num{20000} years: any decision made now is felt by a population
that will be nothing like ours.
\textbf{17.} Each locus contributes $2\hat q = 0.0063$ lethal copies
per person; over \num{20000} loci, about $130$ --- but that assumes
every gene can mutate to a \omterm{def:b2:meiosis-heredity:vocabulary}{recessive} lethal; with the conventional
estimate of a few thousand essential genes, a person carries
several to a few dozen lethal equivalents in the \omterm{def:b2:meiosis-heredity:vocabulary}{heterozygous} state.
\textbf{18.} Each is at a different locus and each is rare, so the
chance that a random couple both carry the same one is small;
cousins share an eighth of their genes by descent, so for each of a
cousin's several lethals the risk that the child is \omterm{def:b2:meiosis-heredity:vocabulary}{homozygous} is
$1/16$ per shared lethal --- the excess child mortality of cousin
marriages is measured at a few per cent.
\textbf{19.} $S = 1.16\times \qty{800}{L} = \qty{930}{L}$; $R = 0.4\times
930 = \qty{370}{L}$.
\textbf{20.} Five generations: about \qty{1860}{L}; smaller in
practice because the additive variance is depleted, because $h^{2}$
was estimated in the original herd, and because the environment
(feed, housing) that made the top cows may not be transmitted.
\textbf{21.} Average differential $(1.16 + 2.4)/2 = 1.78$ standard
deviations, $S = \qty{1424}{L}$, $R = 0.4\times 1424 = \qty{570}{L}$.
\textbf{22.} \omterm{def:b2:asexual-reproduction:clone}{Clones} have no genetic variance, so $h^{2} = 0$ and $R =
0$ whatever $S$ is: heritability is a property of a population, the
share of its variance that is genetic, not a property of the trait.
\textbf{23.} $R = 0.7\times 0.6 = \qty{0.42}{mm}$ deeper.
\textbf{24.} The same response $\qty{370}{L}$ needs $S = 370/0.1 =
\qty{3700}{L}$, that is $4.6$ standard deviations --- keeping fewer
than one cow in ten thousand: impractical, which is why
low-heritability traits are improved slowly or by other means
(progeny testing, larger families).
\textbf{25.} Moth: $s \approx 0.2$ to $0.3$, 28 generations at $0.3$
and 45 at $0.2$, against 50 observed; island: $H_{100} \approx 0.11$,
\qty{27}{\%} of the mainland; herd: \qty{370}{L} per generation from
cows alone, \qty{570}{L} with selected bulls.
\end{solution}
