\section*{Chapter \ref{ch:b2:blood-circulation} --- Blood and the Circulatory System}

\begin{solution}{exo:b2:blood-circulation:1}
\omterm{def:b2:blood-circulation:blood}{Plasma} \qty{55}{\%} (water, \qty{70}{g/L} of protein, salts,
nutrients, gases, hormones); cells \qty{45}{\%}. Red cells $5\times
10^{12}$/L, \qty{7}{\micro m} discs, oxygen transport; white cells
$7\times 10^{9}$/L, \qtyrange{10}{15}{\micro m}, immunity; \omterm{def:b2:blood-circulation:blood}{platelets}
$3\times 10^{11}$/L, \qty{2}{\micro m} fragments, haemostasis.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:2}
Fish: heart $\to$ gills $\to$ body $\to$ heart, one circuit, the body
receiving \omterm{def:b2:blood-circulation:blood}{blood} at the low pressure left after the gills. Mammal:
right heart $\to$ lungs (\qty{25}{mmHg}) $\to$ left heart $\to$ body
(\qty{120}{mmHg}) $\to$ right heart. The double circuit gives the
body high pressure without exposing the lungs to it, and lets the
two circuits be regulated separately.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:3}
\omterm{def:b2:blood-circulation:vessels}{Artery}: thick wall with elastic layers and muscle --- conducts \omterm{def:b2:blood-circulation:blood}{blood}
at high pressure and smooths the pulse. \omterm{def:b2:blood-circulation:vessels}{Arteriole}: wall mostly
smooth muscle --- sets resistance and distributes flow. \omterm{def:b2:blood-circulation:vessels}{Capillary}:
one endothelial cell thick --- exchange. \omterm{def:b2:blood-circulation:vessels}{Vein}: thin, distensible
wall with valves --- returns \omterm{def:b2:blood-circulation:blood}{blood} at low pressure and stores most
of it.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:4}
$70\times 72 = \qty{5}{L/min}$, \qty{7200}{L} a day, against
\qty{5}{L} of \omterm{def:b2:blood-circulation:blood}{blood}: the volume passes through the heart \num{1440}
times a day. No organ could make or consume seven tonnes of \omterm{def:b2:blood-circulation:blood}{blood} a
day; the same \omterm{def:b2:blood-circulation:blood}{blood} must return --- it circulates.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:5}
$\Delta P = 8\eta LQ/\pi r^{4} = 8\times 3\times 10^{-3}\times
0.4\times 8.3\times 10^{-5}/(\pi\times 2.44\times 10^{-8}) =
\qty{10}{Pa}$, \qty{0.08}{mmHg}: the aorta costs nothing; the
pressure is spent in the \omterm{def:b2:blood-circulation:vessels}{arterioles}.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:6}
$(15/12)^{4} = 2.4$: resistance up 2.4-fold, flow down to
\qty{41}{\%}. $(15/20)^{4} = 0.32$: resistance down to a third, flow
up 3.2-fold.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:7}
Aorta $83/3 = \qty{28}{cm/s}$; capillaries $83/3000 =
\qty{0.028}{cm/s}$; transit $0.1/0.028 = \qty{3.6}{s}$. In exercise
the output rises fivefold and the open \omterm{def:b2:blood-circulation:vessels}{capillary} area perhaps
threefold, so the transit time falls to about \qty{2}{s} --- still
enough for the oxygen to leave.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:8}
Arterial end $35 - 25 = \qty{+10}{mmHg}$ (filtration); venous end $15
- 25 = \qty{-10}{mmHg}$ (reabsorption). With $P_c = 28$ at the venous
end the net is $+3$: fluid filters along the whole length and none
is reabsorbed --- oedema, the swollen ankles of heart failure.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:9}
$40/66000 = \qty{0.61}{mol/m^3}$; $\pi = 0.61\times 8.314\times 310 =
\qty{1570}{Pa} = \qty{11.8}{mmHg}$; half the albumin, \qty{5.9}{mmHg}.
Sodium crosses the \omterm{def:b2:blood-circulation:vessels}{capillary} wall freely, so its concentration is the
same on both sides and it exerts no osmotic pressure across it.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:10}
$RC = \qty{2}{s}$: $120e^{-0.4} = \qty{80}{mmHg}$. $C = 1$: $RC = 1$,
$120e^{-0.8} = \qty{54}{mmHg}$ --- and the same stroke volume raises
the systolic pressure more in a stiff aorta. A wide \omterm{thm:b2:blood-circulation:windkessel}{pulse pressure}
means a higher peak against which the heart must eject, more work
and more oxygen for the same output.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:11}
Rest: $90/83 = \qty{1.08}{mmHg.s/mL}$; exercise: $105/417 =
\qty{0.25}{mmHg.s/mL}$, a fourfold fall. Since $R \propto r^{-4}$,
$r$ has risen by $4^{1/4} = 1.41$: the \omterm{def:b2:blood-circulation:vessels}{arterioles} have widened by
\qty{41}{\%} on average.
\end{solution}

\begin{solution}{exo:b2:blood-circulation:12}
Continuity fixes the velocity at each level by the total
cross-section, and the tree is built so that the section is a
thousand times the aorta's in the capillaries and small again in
the \omterm{def:b2:blood-circulation:vessels}{veins}; \omterm{thm:b2:blood-circulation:poiseuille}{Poiseuille's law} puts the resistance in the \omterm{def:b2:blood-circulation:vessels}{arterioles},
where muscle can change it, and leaves the wide vessels nearly free
of loss. An \omterm{def:b2:blood-circulation:circuits}{open circulation} has no capillaries to slow the \omterm{def:b2:blood-circulation:blood}{blood} at
a defined place, no vessels in which to set a resistance, and no way
to direct flow to one organ --- it can only stir.
\end{solution}

\begin{solution}{pb:b2:blood-circulation:1}
\textbf{1.} $70\times 72 = \qty{5.0}{L/min}$; \qty{7300}{L} a day.
\textbf{2.} $7300/5 \approx 1450$ times.
\textbf{3.} A quarter of \qty{57}{g} at 72 beats a minute: \qty{14}{g}
$\times 4320 = \qty{62}{kg}$ an hour --- about the weight of a man.
\textbf{4.} No food could supply, and no tissue consume, a man's
weight of \omterm{def:b2:blood-circulation:blood}{blood} every hour; the only escape is that the same \omterm{def:b2:blood-circulation:blood}{blood}
returns to the heart.
\textbf{5.} A cord tied round the arm swells the \omterm{def:b2:blood-circulation:vessels}{veins} below it, not
above, so \omterm{def:b2:blood-circulation:blood}{blood} in the \omterm{def:b2:blood-circulation:vessels}{veins} moves toward the heart; pressing the
\omterm{def:b2:blood-circulation:blood}{blood} in a \omterm{def:b2:blood-circulation:vessels}{vein} toward the hand, it stops at the valves and cannot be
pushed back --- the valves permit flow toward the heart only.
\textbf{6.} The capillaries joining arteries to \omterm{def:b2:blood-circulation:vessels}{veins}; Malpighi saw
them in the frog's lung in 1661.
\textbf{7.} Area $\pi\times 1.25^{2} = \qty{4.9}{cm^2}$; $83/4.9 =
\qty{17}{cm/s}$.
\textbf{8.} $8\times 3\times 10^{-3}\times 0.4\times 8.3\times
10^{-5}/(\pi\times 2.44\times 10^{-8}) = \qty{10}{Pa}$, \qty{0.08}{mmHg}.
\textbf{9.} $R_{1} = 8\times 3\times 10^{-3}\times 10^{-3}/(\pi\times
5.06\times 10^{-20}) = \qty{1.5e14}{Pa.s/m^3}$; in parallel,
$1.5\times 10^{14}/3\times 10^{6} = \qty{5.0e7}{Pa.s/m^3}$.
\textbf{10.} $8.3\times 10^{-5}\times 5.0\times 10^{7} = \qty{4200}{Pa}
= \qty{31}{mmHg}$.
\textbf{11.} Open capillaries $10^{10}$, each $\pi(3\times 10^{-6})^{2}
= \qty{2.8e-11}{m^2}$: \qty{0.28}{m^2} $= \qty{2800}{cm^2}$; $v =
83/2800 = \qty{0.03}{cm/s}$; transit $0.1/0.03 = \qty{3.3}{s}$.
\textbf{12.} Resistance $\times(1/0.9)^{4} = 1.52$: \qty{47}{mmHg}; the
heart must raise the arterial pressure by \qty{16}{mmHg} or the
output falls by a third.
\textbf{13.} $+10$, $-10$ and $0$ mmHg.
\textbf{14.} Filtration $10\times 2 = \qty{20}{L}$ a day; reabsorption
$8.5\times 2 = \qty{17}{L}$.
\textbf{15.} \qty{3}{L} a day of lymph.
\textbf{16.} $\pi_c = \qty{12.5}{mmHg}$: net $+22.5$ at the arterial
end and $+2.5$ at the venous end --- filtration everywhere, no
reabsorption: oedema.
\textbf{17.} Net $+10$ and $+3$, average $+6.5$: filtration along the
whole \omterm{def:b2:blood-circulation:vessels}{capillary}, some \qty{26}{L} a day; the lymphatics cannot carry
it and the fluid accumulates in the tissues, feet first.
\textbf{18.} $4\times 10^{10}\times 2\pi\times 3\times 10^{-6}\times
10^{-3} = \qty{750}{m^2}$; $t = x^{2}/2D = (2\times 10^{-5})^{2}/
(2\times 10^{-9}) = \qty{0.2}{s}$.
\textbf{19.} $(93 - 3)/83 = \qty{1.08}{mmHg.s/mL}$.
\textbf{20.} $RC = \qty{2.2}{s}$; $120e^{-0.8/2.2} = 120\times 0.69 =
\qty{83}{mmHg}$.
\textbf{21.} $RC = \qty{1.1}{s}$; $120e^{-0.73} = \qty{58}{mmHg}$: the
\omterm{thm:b2:blood-circulation:windkessel}{pulse pressure} widens from 37 to \qty{62}{mmHg}.
\textbf{22.} $120e^{-0.3/2.2} = \qty{105}{mmHg}$: at a fast rate the
pressure barely falls between beats.
\textbf{23.} $C\Delta P = 2\times 20 = \qty{40}{mL}$ stored; the other
\qty{30}{mL} run off through the \omterm{def:b2:blood-circulation:vessels}{arterioles} during systole itself.
\textbf{24.} The contracting ventricle squeezes its own vessels shut
in systole, so the heart muscle is perfused in diastole, by the
pressure that the recoiling aorta maintains; a stiff aorta that lets
the diastolic pressure collapse starves the heart between beats.
\textbf{25.} Output \qty{5}{L/min}; arteriolar drop \qty{31}{mmHg};
net filtrate \qty{3}{L} a day to the lymph; diastolic \qty{83}{mmHg}
for $C = 2$ and \qty{58}{mmHg} for $C = 1$.
\end{solution}
