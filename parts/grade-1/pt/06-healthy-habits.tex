\chapter{Hábitos saudáveis}\label{ch:g1:healthy-habits}

Por que todo mundo vive dizendo ``lave as mãos'', ``escove os
dentes'', ``hora de dormir''? Não é para estragar a diversão. O seu
corpo é o \omterm{def:g1:living-or-not:living}{ser vivo} que você vai ter por mais tempo --- a vida
inteira --- e alguns hábitos pequenos, feitos todo dia, mantêm ele
forte. Aqui estão esses hábitos, e por que funcionam.

\section{Manter-se limpo}

\begin{definition}[Micróbios]\label{def:g1:healthy-habits:germs}
Os \emph{micróbios}\index{micróbio} são \omterm{def:g1:living-or-not:living}{seres vivos} pequenos demais
para enxergar, que estão em toda parte --- nas mãos, nas maçanetas,
na terra. Quase todos são inofensivos, mas alguns podem deixar a
gente doente quando entram no corpo. Como são invisíveis, mãos que
parecem limpas ainda podem estar carregando micróbios.
\end{definition}

\begin{definition}[Higiene]\label{def:g1:healthy-habits:hygiene}
\emph{Higiene}\index{higiene} são os hábitos que mantêm o corpo
limpo e os \omterm{def:g1:healthy-habits:germs}{micróbios} do lado de fora: lavar as mãos, lavar o corpo,
escovar os dentes, manter a comida limpa.
\end{definition}

\begin{method}[Lavar bem as mãos]\label{met:g1:healthy-habits:hands}
\begin{enumerate}
  \item Molhe as mãos e pegue sabão;
  \item esfregue tudo --- palmas, costas das mãos, entre os dedos,
        em volta dos polegares --- enquanto conta devagar até $20$
        ou canta uma musiquinha;
  \item enxágue em água corrente;
  \item seque com uma toalha limpa.
\end{enumerate}
Lave antes de comer, depois do banheiro e depois de brincar na rua
ou de mexer com \omterm{def:g1:animals-around-us:animal}{animais}.
\end{method}

\begin{example}[A experiência do glitter]\label{ex:g1:healthy-habits:glitter}
Espalhe uma pitada de glitter nas palmas das mãos, depois cumprimente
um amigo, abra uma porta, pegue um lápis: glitter por toda parte. Os
\omterm{def:g1:healthy-habits:germs}{micróbios} viajam exatamente assim, só que sem ninguém ver. Agora lave
com sabão, contando até $20$ --- o glitter vai embora, e os \omterm{def:g1:healthy-habits:germs}{micróbios}
também.
\end{example}

\section{Dentes para guardar}

\begin{proposition}[Por que os dentes precisam ser escovados]\label{prop:g1:healthy-habits:teeth}
Depois de comer, restinhos de comida ficam nos dentes. Os \omterm{def:g1:healthy-habits:germs}{micróbios}
da boca se alimentam desses restos --- principalmente dos doces ---
e devagarinho fazem buracos nos dentes, que doem. A escovação varre
os restos antes que os \omterm{def:g1:healthy-habits:germs}{micróbios} possam se banquetear.
\end{proposition}
\admitted

\begin{method}[Escovar os meus dentes]\label{met:g1:healthy-habits:brush}
\begin{enumerate}
  \item Escove depois do café da manhã e antes de dormir --- todo dia;
  \item escove todos os lados de todos os dentes: a frente, o fundo
        e o alto, onde você mastiga;
  \item vá com calma --- mais ou menos o tempo de uma música;
  \item troque de escova quando as cerdas ficarem abertas.
\end{enumerate}
\end{method}

\begin{remark}[Os dentes de leite também contam]\label{rem:g1:healthy-habits:babyteeth}
``Esses dentes vão cair mesmo --- para que escovar?'' Porque eles
ainda vão mastigar por você durante anos, porque buracos neles doem
do mesmo jeito e porque os dentes de adulto já estão esperando
embaixo e merecem chegar a uma boca limpa e saudável.
\end{remark}

\section{Comer, se mexer, dormir}

\begin{proposition}[As três necessidades do dia]\label{prop:g1:healthy-habits:needs}
Um corpo que está crescendo precisa, todo dia, de:
\begin{enumerate}
  \item \emph{comida variada}\index{comida variada} --- um pouco de
        tudo: legumes e frutas, pão e cereais, leite e queijo, carne
        ou peixe ou ovos, e água para beber; doces só de vez em
        quando;
  \item \emph{movimento}\index{movimento} --- correr, subir, andar
        de bicicleta, brincar na rua: mexer o corpo deixa os
        músculos, os ossos e o coração mais fortes;
  \item \emph{sono}\index{sono} --- de noite o corpo descansa, se
        conserta e cresce; uma criança precisa de uma noite longa,
        bem mais longa que a de um adulto.
\end{enumerate}
\end{proposition}
\admitted

\begin{example}[Um bom dia para o corpo]\label{ex:g1:healthy-habits:day}
Café da manhã com fruta; mãos lavadas antes do almoço; uma tarde
correndo no parque; dentes escovados depois do jantar; luz apagada
cedo, com uma história. Nada de extraordinário --- e é justamente
esse o ponto: a saúde é feita de dias comuns como esse.
\end{example}

\begin{remark}[Cansaço é um recado]\label{rem:g1:healthy-habits:tired}
Bocejo, \omterm{def:g1:the-five-senses:sense}{olhos} ardendo, mau humor sem motivo: é o corpo dizendo
``preciso dormir''. Escutar esse recado também é um hábito saudável.
Um cérebro que dormiu bem aprende melhor de manhã, brinca melhor e
ri com mais facilidade.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:1}
O que são \omterm{def:g1:healthy-habits:germs}{micróbios}? Por que mãos que parecem limpas ainda podem
precisar de lavagem?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:2}
Dê três momentos do dia em que as mãos precisam ser lavadas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:3}
No \cref{met:g1:healthy-habits:hands}, quanto tempo deve durar a
esfregação com sabão? Diga dois pontos fáceis de esquecer.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:4}
Por que os dentes ficam com buracos quando não são escovados? Conte
a história com os \omterm{def:g1:healthy-habits:germs}{micróbios} e os restos de comida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:5}
Quando os dentes devem ser escovados, e quanto tempo isso deve durar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:6}
Diga as três necessidades diárias da \cref{prop:g1:healthy-habits:needs}
e um exemplo de cada uma no seu próprio dia.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:7}
O que o corpo faz enquanto dorme? Quem precisa da noite mais longa,
uma criança ou um adulto?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:healthy-habits:8}
Dê dois sinais de que um corpo está pedindo \omterm{prop:g1:healthy-habits:needs}{sono}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:healthy-habits:9}
O Samuel diz: ``Os meus dentes de leite vão cair mesmo, então
escovar é inútil.'' Responda a ele com a
\cref{rem:g1:healthy-habits:babyteeth} --- pelo menos dois motivos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:healthy-habits:10}
Na experiência do glitter do \cref{ex:g1:healthy-habits:glitter}, o
que o glitter representa? O que a experiência mostra sobre
cumprimentar com a mão --- e sobre o sabão?
\end{exercise}
