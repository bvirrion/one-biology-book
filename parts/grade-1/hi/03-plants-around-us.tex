\chapter{हमारे आस-पास के पौधे}\label{ch:g1:plants-around-us}

लॉन में एक गुलबहार, दीवार पर मॉस, स्कूल के अहाते में बड़ा-सा
चेस्टनट का \omterm{def:g1:plants-around-us:tree}{पेड़}: इनमें से कोई कभी भागकर नहीं जाएगा, फिर भी ये सब
जीवित हैं। \omterm{def:g1:plants-around-us:plant}{पौधे} चुपचाप जीने वाले \omterm{def:g1:living-or-not:living}{सजीव} हैं --- वे जन्म लेते हैं, पोषण
लेते हैं, बढ़ते हैं, कभी-कभी सैकड़ों साल तक, और यह सब अपनी जगह से हिले
बिना।

\section{पौधे सजीव हैं}

\begin{definition}[पौधा]\label{def:g1:plants-around-us:plant}
\emph{पौधा}\index{पौधा} वह \omterm{def:g1:living-or-not:living}{सजीव} है जो ज़मीन में जमा रहता है और प्रकाश
की मदद से अपना भोजन ख़ुद बनाता है। घास, गुलबहार, टमाटर के पौधे और
\omterm{def:g1:plants-around-us:tree}{पेड़}, सब पौधे हैं। ज़्यादातर पौधे हरे होते हैं।
\end{definition}

\begin{example}[गुलबहार जीवित है]\label{ex:g1:plants-around-us:daisy}
गुलबहार पर जीवन के लक्षण जाँचो: वह एक बीज से जन्मी; वह पोषण लेती है
--- मिट्टी से पानी और आसमान से प्रकाश; वह बढ़ती है; वह ऐसे बीज बनाती
है जिनसे नई गुलबहार बनती हैं; और पाले में वह मर जाती है। सारे लक्षण
मौजूद हैं। गुलबहार उतनी ही जीवित है जितनी बिल्ली --- बस वह इस बारे
में कहीं ज़्यादा चुप रहती है।
\end{example}

\begin{omfigure}
\includegraphics[width=0.58\linewidth]{images/book1/ai/g1-plants-daisy.jpg}

{\small लॉन में गुलबहार: बीजों से जन्मीं, प्रकाश और पानी से पोषण
लेतीं, चुपचाप जीवित।}
\end{omfigure}

\begin{remark}[बिना दौड़े भी जीवित]\label{rem:g1:plants-around-us:still}
\omterm{def:g1:plants-around-us:plant}{पौधे} चलते नहीं, पर वे पूरी तरह निश्चल भी नहीं हैं। सूरजमुखी दिन भर
प्रकाश की ओर मुड़ता रहता है, और खिड़की पर रखा \omterm{def:g1:plants-around-us:plant}{पौधा} धीरे-धीरे खिड़की
की ओर झुक जाता है। \omterm{def:g1:plants-around-us:plant}{पौधे} \emph{बढ़कर} चलते हैं --- हमारी आँखों के लिए
बहुत धीरे, पर हफ़्ते भर देखने के लिए काफ़ी तेज़।
\end{remark}

\section{पौधे के भाग}

\begin{definition}[पौधे के भाग]\label{def:g1:plants-around-us:parts}
फूल वाले \omterm{def:g1:plants-around-us:plant}{पौधे} के चार मुख्य भाग होते हैं:
\begin{enumerate}
  \item \emph{जड़ें}\index{जड़}, जो मिट्टी में छिपी रहती हैं, \omterm{def:g1:plants-around-us:plant}{पौधे} को
        अपनी जगह थामे रखती हैं और पानी पीती हैं;
  \item \emph{डंठल}\index{डंठल}, जो \omterm{def:g1:plants-around-us:plant}{पौधे} को खड़ा रखता है और पत्तियों
        तक पानी पहुँचाता है;
  \item \emph{पत्तियाँ}\index{पत्ती}, चपटी और हरी, जो प्रकाश पकड़ती
        हैं;
  \item \emph{फूल}\index{फूल}, जो आगे चलकर बीज बनाएगा।
\end{enumerate}
\end{definition}

भीतर \omterm{def:g1:plants-around-us:parts}{जड़ें} अपना काम नज़रों से ओझल, मिट्टी के नीचे करती हैं।}

\begin{example}[खरपतवार उखाड़ना]\label{ex:g1:plants-around-us:weed}
ढीली गीली मिट्टी से कोई खरपतवार धीरे से उखाड़ो और छिपा हुआ हिस्सा
बाहर आ जाएगा: पीली-सी \omterm{def:g1:plants-around-us:parts}{जड़ों} का गुच्छा, कभी-कभी ऊपर के पूरे हरे \omterm{def:g1:plants-around-us:plant}{पौधे}
से भी लंबा। अब \omterm{def:g1:plants-around-us:plant}{पौधे} को नीचे से ऊपर तक पढ़ा जा सकता है: \omterm{def:g1:plants-around-us:parts}{जड़ें}, \omterm{def:g1:plants-around-us:parts}{डंठल},
\omterm{def:g1:plants-around-us:parts}{पत्तियाँ} --- और क़िस्मत अच्छी हो तो एक \omterm{def:g1:plants-around-us:parts}{फूल} भी।
\end{example}

\begin{definition}[पेड़]\label{def:g1:plants-around-us:tree}
\emph{पेड़}\index{पेड़} वह बड़ा \omterm{def:g1:plants-around-us:plant}{पौधा} है जिसका \omterm{def:g1:plants-around-us:parts}{डंठल} लकड़ी का बना होता
है। पेड़ के लकड़ी वाले \omterm{def:g1:plants-around-us:parts}{डंठल} को \emph{तना}\index{तना} कहते हैं; वह
\emph{शाखाओं}\index{शाखा} में बँट जाता है, जो \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} लिए रहती
हैं।
\end{definition}

\begin{example}[चेस्टनट का पेड़]\label{ex:g1:plants-around-us:tree}
अहाते के चेस्टनट के \omterm{def:g1:plants-around-us:tree}{पेड़} की बनावट गुलबहार जैसी ही है, बस बड़ी और
सख़्त: ज़मीन के नीचे \omterm{def:g1:plants-around-us:parts}{जड़ें}, मुलायम \omterm{def:g1:plants-around-us:parts}{डंठल} की जगह एक तना, हज़ारों
\omterm{def:g1:plants-around-us:parts}{पत्तियों} वाली शाखाएँ, और वसंत में \omterm{def:g1:plants-around-us:parts}{फूल}। गुलबहार एक साल जीती है;
चेस्टनट सौ साल से भी ज़्यादा जी सकता है।
\end{example}

\section{पौधे को किन चीज़ों की ज़रूरत है}

\begin{proposition}[पौधे की ज़रूरतें]\label{prop:g1:plants-around-us:needs}
जीने और बढ़ने के लिए \omterm{def:g1:plants-around-us:plant}{पौधे} को \emph{पानी}, \emph{प्रकाश} और अपनी \omterm{def:g1:plants-around-us:parts}{जड़ों}
के लिए एक जगह चाहिए, जो आम तौर पर \emph{मिट्टी}\index{मिट्टी} होती
है। अँधेरे में रखा या कभी न सींचा गया \omterm{def:g1:plants-around-us:plant}{पौधा} मुरझाकर मर जाता है।
\end{proposition}
\admitted

\begin{example}[भुला दिया गया पौधा]\label{ex:g1:plants-around-us:wilt}
छुट्टियों से पहले कक्षा के \omterm{def:g1:plants-around-us:plant}{पौधे} को सींचना भूल जाओ: उसकी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ}
नीचे लटक जाती हैं, नरम और उदास --- वह मुरझा जाता है। उसे पानी दो, और
कुछ ही घंटों बाद \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} फिर तन जाती हैं। पानी ही वह गुम ज़रूरत थी।
\end{example}

\begin{method}[पौधे की देखभाल]\label{met:g1:plants-around-us:care}
\begin{enumerate}
  \item गमले को वहाँ रखो जहाँ दिन का उजाला पहुँचे;
  \item जब मिट्टी सूखी लगे तब पानी दो --- हर घंटे नहीं, \omterm{def:g1:plants-around-us:parts}{जड़ों} को हवा
        भी चाहिए;
  \item \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} मत तोड़ो: वे ही \omterm{def:g1:plants-around-us:plant}{पौधे} की प्रकाश पकड़ने वाली चीज़ें
        हैं;
  \item धीरज रखो: \omterm{def:g1:plants-around-us:plant}{पौधा} दिनों में उत्तर देता है, मिनटों में नहीं।
\end{enumerate}
\end{method}

\begin{remark}[पौधे सबका पेट भरते हैं]\label{rem:g1:plants-around-us:food}
\omterm{def:g1:plants-around-us:plant}{पौधे} हर \omterm{def:g1:animals-around-us:animal}{जंतु} के लिए मायने रखते हैं। घोंघा सलाद खाता है, कलचिड़ी
जामुन खाती है, हम गेहूँ की बनी रोटी खाते हैं --- और गेहूँ एक \omterm{def:g1:plants-around-us:plant}{पौधा}
है। यहाँ तक कि बिल्ली भी, जो सलाद नहीं खाती, ऐसे \omterm{def:g1:animals-around-us:animal}{जंतुओं} को खाती है
जिन्होंने \omterm{def:g1:plants-around-us:plant}{पौधे} खाए थे। \omterm{def:g1:plants-around-us:plant}{पौधों} के बिना किसी और के पास खाने को कुछ न
होता; आगे का एक अध्याय इन्हीं आहार शृंखलाओं का पीछा करता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:1}
मिट्टी के नीचे से लेकर सबसे ऊपर तक, \omterm{def:g1:plants-around-us:parts}{फूल} वाले \omterm{def:g1:plants-around-us:plant}{पौधे} के चार भागों के नाम
बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:2}
\omterm{def:g1:plants-around-us:plant}{पौधे} का कौन-सा भाग: (क) मिट्टी से पानी पीता है? (ख) प्रकाश पकड़ता
है? (ग) आगे चलकर बीज बनाएगा? (घ) \omterm{def:g1:plants-around-us:plant}{पौधे} को सीधा खड़ा रखता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:3}
गुलबहार कभी चलती नहीं। जीवन के लक्षणों से गुज़रो और बताओ कि फिर भी वह
\omterm{def:g1:living-or-not:living}{सजीव} क्यों है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:4}
\omterm{def:g1:plants-around-us:tree}{पेड़} के \omterm{def:g1:plants-around-us:parts}{डंठल} का और उन भागों का ख़ास नाम क्या है जो उसकी \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} लिए
रहते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:5}
\cref{prop:g1:plants-around-us:needs} से \omterm{def:g1:plants-around-us:plant}{पौधे} को जीने के लिए ज़रूरी
तीन चीज़ें बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:6}
कक्षा के \omterm{def:g1:plants-around-us:plant}{पौधे} की \omterm{def:g1:plants-around-us:parts}{पत्तियाँ} नरम होकर लटक गई हैं। शायद क्या भुला दिया
गया है? देने पर क्या होगा?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:7}
गुलबहार और चेस्टनट का \omterm{def:g1:plants-around-us:tree}{पेड़} किन बातों में एक जैसे हैं? किन दो बातों
में अलग हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:plants-around-us:8}
करके देखो: सेम का एक \omterm{def:g1:plants-around-us:plant}{पौधा} (या हलीम का गमला) खिड़की के पास रखो और
दूसरा अँधेरी अलमारी में, और दोनों को पानी दो। हफ़्ते भर बाद देखो।
तुम्हें क्या फ़र्क़ मिलने की उम्मीद है, और कौन-सी ज़रूरत उसे समझाती
है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-around-us:9}
लियो कहता है: ``\omterm{def:g1:plants-around-us:plant}{पौधे} सचमुच जीवित नहीं हैं --- वे कभी कुछ करते ही
नहीं।'' \cref{rem:g1:plants-around-us:still} और
\cref{ex:g1:plants-around-us:daisy} से उसे दो ऐसी चीज़ें बताओ जो \omterm{def:g1:plants-around-us:plant}{पौधा}
धीरे-धीरे, पर ज़रूर करता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:plants-around-us:10}
गाजर नारंगी होती है और ज़मीन के नीचे उगती है। \omterm{def:g1:plants-around-us:plant}{पौधे} के भागों वाले चित्र
को देखकर बताओ कि गाजर के \omterm{def:g1:plants-around-us:plant}{पौधे} का कौन-सा भाग हम खाते हैं। ``ज़मीन के
नीचे'' से क्या सुराग़ मिलता है?
\end{exercise}
