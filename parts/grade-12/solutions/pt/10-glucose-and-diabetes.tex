\section*{Capítulo \ref{ch:g12:glucose-and-diabetes} --- A glicose do sangue e o diabetes}

\begin{solution}{exo:g12:glucose-and-diabetes:1}
Cerca de \qty{1}{g/L}, ou seja, \qty{5.5}{mmol/L}. Abaixo de
\qty{0.5}{g/L} o cérebro falha; acima de \qty{1.8}{g/L} a glicose passa
para a urina (e anos acima de \qty{1.3}{g/L} danificam os vasos).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:2}
A \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina}, das \omterm{def:g10:cells-common-unit:cell}{células} $\beta$, baixa a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} (captação e
armazenamento pelo músculo, pela gordura e pelo fígado); o \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon},
das \omterm{def:g10:cells-common-unit:cell}{células} $\alpha$, a eleva (liberação de glicose pelo fígado).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:3}
O fígado, como \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} (e ele também consegue produzir glicose nova
a partir de \omterm{def:g10:chemistry-of-life:families}{aminoácidos} e de lactato).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:4}
Tipo 1: as \omterm{def:g10:cells-common-unit:cell}{células} $\beta$ são destruídas e a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} está ausente.
Tipo 2: a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} está presente, mas os tecidos respondem mal a ela, e
as \omterm{def:g10:cells-common-unit:cell}{células} $\beta$ acabam se cansando.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:5}
Saudável, cerca de \qty{0.95}{g/L}; tolerância diminuída, cerca de
\qty{1.7}{g/L} (entre 1.4 e 2); tipo 2, cerca de \qty{2.7}{g/L} (acima de 2).
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:6}
O fígado libera glicose do \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} dele na mesma taxa em que o
cérebro a retira, sob o controle do \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon}, e assim o reservatório
fica perto de \qty{5}{g} embora o conteúdo dele seja renovado a cada hora.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:7}
O ducto leva \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzimas} digestivas ao intestino; bloqueá-lo deixa a
\omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} normal, e assim o efeito do pâncreas sobre a glicose não se dá
pela digestão. Retirar o órgão inteiro retira as \omterm{prop:g12:glucose-and-diabetes:hormones}{ilhotas}: a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} é
produzida no pâncreas e chega aos alvos dela pelo sangue, e não pelo ducto.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:8}
O \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} age fazendo o fígado quebrar o \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} dele; depois de
uma noite de jejum a reserva está lá e a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} sobe; depois de três
dias o \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} está esgotado e o \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} não tem nada para liberar.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:9}
Acima de cerca de \qty{1.8}{g/L} os rins não conseguem reabsorver toda
a glicose que filtram; ela passa para a urina e, por osmose, arrasta
água junto. A água perdida precisa ser reposta: sede.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:10}
A \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} empurra a glicose para dentro dos músculos e das \omterm{def:g10:cells-common-unit:cell}{células}
adiposas e interrompe a liberação do fígado, mas glicose nenhuma entra:
a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} cai abaixo de \qty{0.6}{g/L} em uma ou duas horas, com suor,
tremor e confusão; açúcar tomado na hora corrige.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:11}
Um músculo em trabalho capta glicose do sangue por uma via que não
precisa de \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina}, e o músculo treinado recupera a sensibilidade à
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} entre as sessões; a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} cai durante e depois do
exercício, e a resistência dos tecidos, a causa da doença, diminui.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:12}
Sem \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina}: tipo 1, as \omterm{def:g10:cells-common-unit:cell}{células} $\beta$ das \omterm{prop:g12:glucose-and-diabetes:hormones}{ilhotas} estão destruídas e
não secretam nada. Três vezes a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} normal: tipo 2, as \omterm{prop:g12:glucose-and-diabetes:hormones}{ilhotas}
estão secretando a plena carga contra tecidos que não respondem mais.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:13}
Saudável: a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} sobe a cerca de \qty{1.3}{g/L} aos 30--60 minutos
e volta em 2 horas; a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} sobe com ela e cai com ela. Tipo 1 sem
tratamento: a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} sobe acima de \qty{2}{g/L} e fica; a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina}
plana em zero --- nada retira a glicose. Tipo 2: a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} sobe alto e
cai devagar; a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} sobe mais que o normal e fica alta --- muito
\omterm{def:g11:hormones-and-reproduction:hormone}{hormônio}, pouco efeito.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:14}
$5 + 80 = \qty{85}{g}$ em \qty{5}{L}: \qty{17}{g/L}. Isso nunca
acontece porque a absorção leva uma ou duas horas e, à medida que a
glicose chega, a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} a manda para o \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} do fígado, para o
\omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} do músculo e para a gordura, enquanto o cérebro e os tecidos
queimam um pouco: o reservatório nunca guarda mais que alguns gramas
acima do normal.
\end{solution}

\begin{solution}{exo:g12:glucose-and-diabetes:15}
Os dois: um sensor (as \omterm{def:g10:cells-common-unit:cell}{células} das \omterm{prop:g12:glucose-and-diabetes:hormones}{ilhotas}; o hipotálamo e a hipófise),
um ponto de ajuste, efetores e uma \omterm{def:g11:hormones-and-reproduction:hormone}{retroalimentação} negativa que remove
o sinal. A \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} tem dois \omterm{def:g11:hormones-and-reproduction:hormone}{hormônios} opostos agindo sobre os mesmos
órgãos, e assim a variável consegue ser empurrada ativamente para cima
e para baixo; a alça da \omterm{prop:g11:becoming-male-female:hormones}{testosterona} tem um \omterm{def:g11:hormones-and-reproduction:hormone}{hormônio} e corrige
sobretudo por mais ou menos dele. Dois efetores opostos dão um controle
mais rápido e mais apertado de uma variável que muda depressa nos dois sentidos.
\end{solution}

\begin{solution}{pb:g12:glucose-and-diabetes:1}
\textbf{1.} \qty{5}{g}; $\qty{1}{g/L} \div \qty{180}{g/mol} = \qty{5.55}{mmol/L}$.

\textbf{2.} Uma hora.

\textbf{3.} $100/5 = 20$ horas. Depois disso, o fígado produz glicose
nova a partir de \omterm{def:g10:chemistry-of-life:families}{aminoácidos} (e de lactato), ao custo das \omterm{def:g11:gene-expression:protein}{proteínas} do corpo.

\textbf{4.} As \omterm{def:g10:cells-common-unit:cell}{células} musculares não têm a \omterm{def:g11:enzymes-and-phenotype:enzyme}{enzima} que libera glicose
livre do \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} para o sangue; o \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} delas é queimado no
próprio lugar, para a contração delas mesmas.

\textbf{5.} $5 + 60 = \qty{65}{g}$ em \qty{5}{L}: \qty{13}{g/L}. A
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} manda a glicose para o \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} do fígado e do músculo e
para a gordura, e os tecidos queimam um pouco; a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} tem pico em
torno de \qty{1.4}{g/L}.

\textbf{6.} Jejum \qty{1.15}{g/L} (abaixo de 1.26) mas \qty{1.85}{g/L}
aos 120 minutos (entre 1.4 e 2): tolerância diminuída, no limiar do
\omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes}.

\textbf{7.} Não: a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} está no dobro de um pico saudável. O que
está falhando é a resposta dos tecidos a ela --- resistência à \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina}.

\textbf{8.} Tipo 2 (ou o estágio de alerta dele). Esperado: adulto, com
sobrepeso, sedentário, muitas vezes com parentes diabéticos.

\textbf{9.} \omterm{def:g12:glucose-and-diabetes:glycaemia}{Glicemia} começando acima de \qty{1.5}{g/L}, subindo acima
de \num{2.5} e sem cair; \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} perto de zero o tempo todo.

\textbf{10.} Perda de peso e exercício regular: eles restauram a
sensibilidade dos tecidos à \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} e deixam o músculo em trabalho
captar glicose sem ela.

\textbf{11.} $90/10 = 9$ unidades.

\textbf{12.} A dose dá conta de \qty{90}{g}, mas só \qty{45}{g} chegam:
a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} excedente puxa a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} para baixo pelo equivalente a
\qty{45}{g} no reservatório e nas reservas --- bem abaixo de
\qty{0.6}{g/L}: suor, tremor, confusão; açúcar na hora.

\textbf{13.} $(5 + 90)/5 = \qty{19}{g/L}$ se nada fosse retirado; na
realidade, alguns \unit{g/L}. Acima de \qty{1.8}{g/L} os rins deixam a
glicose ir para a urina, com água: urina abundante e sede.

\textbf{14.} Entre as refeições, o fígado liberaria glicose sem freio
(o \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon} age, nada se opõe a ele) e os tecidos queimariam gordura
com produtos ácidos: um fundo de \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} é necessário para o
equilíbrio básico, e não só para as refeições.

\textbf{15.} O medidor mede a \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} como a \omterm{def:g10:cells-common-unit:cell}{célula} $\beta$ media; a
caneta entrega a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} que ela teria secretado. Eles não conseguem
substituir o ajuste contínuo, minuto a minuto: a paciente mede algumas
vezes por dia e estima, onde a \omterm{def:g10:cells-common-unit:cell}{célula} media o tempo todo e respondia
com exatidão.

\textbf{16.} Tipo 2, cujas \omterm{def:g10:cells-common-unit:cell}{células} $\beta$ existem e conseguem ser
estimuladas. No tipo 1 não há \omterm{def:g10:cells-common-unit:cell}{células} $\beta$ para estimular.

\textbf{17.} A resposta dos tecidos à \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} --- o lado efetor da
alça: com menos gordura e músculo ativo, um sinal normal de \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina}
volta a produzir uma captação normal, e as \omterm{def:g10:cells-common-unit:cell}{células} $\beta$ são aliviadas.

\textbf{18.} Cada um é secretado em resposta ao desvio oposto da mesma
variável: uma \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} alta chama a \omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} e silencia o \omterm{prop:g12:glucose-and-diabetes:hormones}{glucagon}, uma
baixa faz o contrário. A \qty{1}{g/L} a ilhota secreta os dois a uma
taxa baixa e equilibrada --- o estado de repouso dela.

\textbf{19.} Uma \omterm{def:g12:glucose-and-diabetes:glycaemia}{glicemia} baixa mata o cérebro em minutos, uma alta o
danifica devagar ao longo de anos: o perigo agudo tem várias guardas
independentes, o lento tem uma só --- e é por isso que a falha da
\omterm{prop:g12:glucose-and-diabetes:hormones}{insulina} sozinha basta para causar uma doença, e por que o \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes} é
comum.

\textbf{20.} Cerca de \qty{5}{g} de glicose no sangue; o \omterm{def:g10:exercise-and-energy:glycogen}{glicogênio} do
fígado cobre cerca de 20 horas; o paciente da Parte II tem tolerância
diminuída no limite do \omterm{def:g12:glucose-and-diabetes:diabetes}{diabetes} tipo 2; a dose da Parte III é de 9
unidades.
\end{solution}
