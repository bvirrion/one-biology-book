\chapter{Staying Healthy}\label{ch:g2:healthy}

A healthy \omterm{def:g1:my-body:body}{body} helps you learn, play and \omterm{def:g2:what-alive:signs}{grow}. \omterm{def:g2:healthy:health}{Health} is not one
magic trick --- it is a set of daily habits: \omterm{def:g2:food-energy:food}{food}, \omterm{def:g1:caring:needs}{water}, sleep,
movement, cleanliness and kindness to yourself.

\section{Healthy habits}

\begin{definition}[Health]\label{def:g2:healthy:health}
\emph{Health}\index{health} means the \omterm{def:g1:my-body:body}{body} and mind are working
well. You have \omterm{def:g2:food-energy:energy}{energy}, you recover from small illnesses, and you
can enjoy work and play.
\end{definition}

\begin{definition}[Healthy habits]\label{def:g2:healthy:habits}
Useful daily habits include:
\begin{itemize}
  \item eating a mix of \omterm{def:g2:food-energy:food}{foods} and drinking \omterm{def:g1:caring:needs}{water};
  \item moving and playing every day;
  \item sleeping enough at night;
  \item washing \omterm{def:g1:my-body:parts}{hands} and brushing teeth;
  \item telling a trusted adult when you feel unwell.
\end{itemize}
\end{definition}

\section{Food, sleep and movement}

\begin{method}[Build a healthy day]\label{met:g2:healthy:day}
\begin{enumerate}
  \item Drink \omterm{def:g1:caring:needs}{water}; eat \omterm{def:g2:plant-parts:main}{fruit} or vegetables with meals.
  \item Play outside or move indoors --- run, dance, stretch.
  \item Wash \omterm{def:g1:my-body:parts}{hands} before eating and after the toilet.
  \item Brush teeth morning and night.
  \item Go to bed on time so your \omterm{def:g1:my-body:body}{body} can repair and \omterm{def:g2:what-alive:signs}{grow}.
\end{enumerate}
\end{method}

\begin{example}[Hand washing]\label{ex:g2:healthy:hands}
Soap and \omterm{def:g1:caring:needs}{water} remove dirt and many tiny germs. Wash for as long as
it takes to sing a short song twice. Dry on a clean towel.
\end{example}

\begin{example}[Sleep]\label{ex:g2:healthy:sleep}
Children need plenty of sleep. Sleep helps memory and mood. A calm
evening (less screen glare, quiet story) makes sleep easier.
\end{example}

\begin{proposition}[Balance]\label{prop:g2:healthy:balance}
Healthy living balances \omterm{def:g2:food-energy:food}{food}, movement, sleep and hygiene. Treats
can fit sometimes; everyday choices matter most.
\end{proposition}
\admitted
