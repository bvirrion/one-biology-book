\section*{Chapter \ref{ch:g6:classification} --- Classification of Life}

\begin{solution}{exo:g6:classification:1}
To organise \omterm{def:g4:protecting:biodiversity}{biodiversity}, show relationships, and use shared names for
clear communication.
\end{solution}

\begin{solution}{exo:g6:classification:2}
A group that can typically interbreed to produce fertile \omterm{def:g2:growing:offspring}{offspring}
(working definition; exceptions exist).
\end{solution}

\begin{solution}{exo:g6:classification:3}
Two: genus and \omterm{def:g6:classification:species}{species}.
\end{solution}

\begin{solution}{exo:g6:classification:4}
Genus is more specific (narrower) than family.
\end{solution}

\begin{solution}{exo:g6:classification:5}
\emph{Aurora minor} (genus capitalised, \omterm{def:g6:classification:species}{species} lower case; italics if
possible).
\end{solution}
