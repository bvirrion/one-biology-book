\chapter{El ejercicio y las necesidades energéticas del cuerpo}\label{ch:g10:exercise-and-energy}

Una ciclista pedalea en una bicicleta de laboratorio contra un freno
que mide su potencia: \qty{100}{W}, luego \qty{150}{W}, luego
\qty{200}{W}. Una mascarilla sobre su cara envía cada respiración a un
analizador. Las cifras de la pantalla suben al compás del freno: cuanto
más trabajo mecánico entrega, más oxígeno consume y más dióxido de
carbono espira. Sus músculos están llevando al máximo la respiración
del \cref{ch:g10:cell-metabolism}, y el analizador lee ese balance.
Este capítulo sigue la energía del ejercicio desde el alimento que la
aporta hasta el oxígeno que la libera.

\section{Energía que entra, trabajo que sale}

\begin{definition}[Gasto energético]\label{def:g10:exercise-and-energy:expenditure}
El \emph{gasto energético}\index{gasto energético} del cuerpo, en
julios, es la energía total que sus \omterm{def:g10:cells-common-unit:cell}{células} liberan a partir de
moléculas orgánicas en un tiempo dado. Dividido por el tiempo, es una
\emph{potencia metabólica}, en vatios. En reposo, un adulto gasta unos
\qty{80}{W} --- unos \qty{7000}{kJ} al día --- para mantener el corazón
latiendo, el cerebro funcionando, la temperatura en \qty{37}{\celsius}
y las \omterm{def:g10:cells-common-unit:cell}{células} vivas; es el \emph{\omterm{def:g10:cell-metabolism:metabolism}{metabolismo} basal}. Toda actividad se
suma a él.
\end{definition}

\begin{proposition}[Adónde va la energía]\label{prop:g10:exercise-and-energy:efficiency}
Un músculo que trabaja convierte en trabajo mecánico solo alrededor de
una cuarta parte de la energía que libera; las otras tres cuartas
partes salen como calor. Entregar una potencia mecánica $P$ le cuesta
por tanto al cuerpo una potencia metabólica de unas $4P$ por encima del
reposo: una ciclista que da \qty{100}{W} en los pedales gasta unos
\qty{400}{W} en sus músculos, y su cuerpo debe deshacerse de
\qty{300}{W} de calor --- sudando, sobre todo.
\end{proposition}
\admitted

\begin{example}[Presupuestos diarios]\label{ex:g10:exercise-and-energy:budgets}
Un estudiante que está sentado en clase, camina al instituto y duerme
gasta unos \qty{9000}{kJ} al día; una hora de fútbol añade unos
\qty{2500}{kJ}; un día de montaña con mochila, \qty{15000}{kJ} en
total. Un ciclista en una etapa de montaña de una carrera gasta
\qty{25000}{kJ} y no puede comer bastante durante el día para
cubrirlos. La energía viene del alimento
(\cref{ch:g10:chemistry-of-life}): \qty{17}{kJ} por gramo de \omterm{def:g10:chemistry-of-life:families}{glúcido} o
de \omterm{def:g10:chemistry-of-life:families}{proteína}, \qty{38}{kJ} por gramo de \omterm{def:g10:chemistry-of-life:families}{lípido}.
\end{example}

\section{El consumo de oxígeno mide el gasto energético}

\begin{proposition}[El equivalente energético del oxígeno]\label{prop:g10:exercise-and-energy:oxygen}
Como la energía del ejercicio la libera la \omterm{prop:g10:cell-metabolism:respiration}{respiración celular}, que
consume oxígeno, el consumo de oxígeno del cuerpo mide su \omterm{def:g10:exercise-and-energy:expenditure}{gasto
energético}: cada litro de oxígeno consumido corresponde a unos
\qty{20}{kJ} liberados, sea cual sea la mezcla de glucosa y \omterm{def:g10:chemistry-of-life:families}{lípidos}
quemada. En reposo, una persona consume unos \qty{0.25}{L} de oxígeno
por minuto; durante el ejercicio, el consumo sube en proporción a la
potencia entregada, hasta un techo personal.
\end{proposition}
\begin{proof}[Pruebas experimentales]
La ecuación de la respiración del \cref{ch:g10:cell-metabolism} da
\qty{2870}{kJ} por \qty{6}{mol} de oxígeno, es decir \qty{478}{kJ} por
mol, y un mol de gas ocupa unos \qty{24}{L}: \qty{20}{kJ} por litro.
Para los \omterm{def:g10:chemistry-of-life:families}{lípidos} la cifra es \qty{19.6}{kJ/L} y para los \omterm{def:g10:chemistry-of-life:families}{glúcidos}
\qty{21.1}{kJ/L}, así que la mezcla apenas importa. Las medidas hechas
en cámaras cerradas, donde se recoge directamente el calor que libera
un sujeto, coinciden con la cifra del oxígeno dentro de un pequeño
porcentaje.
\end{proof}

\begin{definition}[Consumo máximo de oxígeno]\label{def:g10:exercise-and-energy:vo2max}
El \emph{consumo máximo de oxígeno}\index{consumo máximo de oxígeno},
escrito $\dot V\!\mathrm{O_2}$máx, es la mayor velocidad a la que el
cuerpo de una persona puede captar y usar oxígeno, en litros por minuto
--- o, para comparar personas de tamaños distintos, en mililitros por
minuto y por kilogramo de masa corporal. Fija la mayor potencia que
puede sostenerse solo con la respiración. Valores típicos:
\qty{35}{mL/min} por kilogramo en un adulto sedentario, \num{50} en un
estudiante entrenado, \num{80} o más en un campeón de resistencia.
\end{definition}

\begin{example}[Leer la curva]\label{ex:g10:exercise-and-energy:reading}
En la figura, pedalear a \qty{100}{W} cuesta \qty{1.5}{L/min} de
oxígeno: $1.5 \times 20 = \qty{30}{kJ}$ por minuto, es decir
\qty{500}{W} de potencia metabólica --- los \qty{100}{W} de trabajo más
\qty{400}{W} de calor y de necesidades de reposo, en línea con la
\cref{prop:g10:exercise-and-energy:efficiency}. El techo de
\qty{3.3}{L/min} para un sujeto de \qty{66}{kg} son \qty{50}{mL/min}
por kilogramo.
\end{example}

\section{Los combustibles del músculo}

\begin{proposition}[Tres fuentes de ATP]\label{prop:g10:exercise-and-energy:fuels}
La contracción muscular se paga en ATP, y la \omterm{def:g10:cells-common-unit:cell}{célula} solo guarda unos
segundos de reserva. Se renueva por tres vías, cada una con su
velocidad y su capacidad:
\begin{itemize}
  \item una pequeña reserva de un compuesto fosforado del músculo, que
        rehace ATP al instante pero dura unos diez segundos --- el
        esprint;
  \item la \omterm{def:g10:cell-metabolism:fermentation}{fermentación} láctica de la glucosa en el \omterm{def:g10:cells-common-unit:cell}{citoplasma}, rápida
        pero derrochadora, que sostiene un esfuerzo máximo durante uno
        o dos minutos y produce ácido láctico;
  \item la \omterm{prop:g10:cell-metabolism:respiration}{respiración celular} en las \omterm{def:g10:cells-common-unit:organelle}{mitocondrias}, más lenta en
        alcanzar su ritmo pleno pero casi ilimitada en duración, que
        quema glucosa de la sangre, \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} del músculo y ácidos
        grasos de la grasa corporal.
\end{itemize}
La proporción de cada una depende de la intensidad y de la duración del
ejercicio.
\end{proposition}
\admitted

\begin{definition}[Glucógeno]\label{def:g10:exercise-and-energy:glycogen}
El \emph{glucógeno}\index{glucógeno} es la forma animal de \omterm{def:g10:chemistry-of-life:families}{glúcido} de
reserva: una cadena ramificada de unidades de glucosa, guardada en
gránulos en las \omterm{def:g10:cells-common-unit:cell}{células} musculares (unos \qty{400}{g} en un adulto) y
en el hígado (unos \qty{100}{g}). El glucógeno del músculo alimenta al
músculo que lo almacena; el del hígado se degrada en glucosa que pasa a
la sangre y mantiene la glucemia cerca de \qty{1}{g/L} para el cerebro
y los demás tejidos. La grasa, guardada en las \omterm{def:g10:cells-common-unit:cell}{células} adiposas, es la
reserva mucho mayor (\cref{ch:g10:chemistry-of-life}), pero solo puede
quemarse despacio y solo con oxígeno.
\end{definition}

\begin{example}[El muro del maratón]\label{ex:g10:exercise-and-energy:wall}
Correr consume unos \qty{4}{kJ} por kilogramo de masa corporal y por
kilómetro; una corredora de \qty{70}{kg} gasta \qty{280}{kJ/km}, y un
maratón cuesta \qty{11800}{kJ}. Las reservas de \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} guardan unos
\qty{8500}{kJ}. Si la corredora quema \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} demasiado deprisa, las
reservas se agotan hacia el kilómetro treinta; los músculos han de
recurrir entonces a la grasa, que aporta energía a apenas la mitad de
velocidad, y el ritmo se hunde --- es ``chocar contra el muro''. Los
corredores entrenados queman desde el principio una proporción mayor de
grasa y comen \omterm{def:g10:chemistry-of-life:families}{glúcidos} durante la carrera.
\end{example}

\section{Medir y razonar}

\begin{method}[Un cálculo energético del ejercicio]\label{met:g10:exercise-and-energy:calc}
\begin{enumerate}
  \item A partir del oxígeno: energía (\unit{kJ}) $=$ oxígeno
        consumido (\unit{L}) $\times 20$. Resta el consumo de reposo si
        la pregunta pide el coste del ejercicio solo.
  \item A partir de la potencia mecánica: potencia metabólica
        $\approx 4 P$; energía $=$ potencia $\times$ tiempo
        (\qty{1}{W} durante \qty{1}{s} es \qty{1}{J}; \qty{1}{kWh} $= \qty{3600}{kJ}$).
  \item A partir del combustible: masa quemada $=$ energía $/$ valor
        energético (\qty{17}{kJ/g} los \omterm{def:g10:chemistry-of-life:families}{glúcidos}, \qty{38}{kJ/g} los
        \omterm{def:g10:chemistry-of-life:families}{lípidos}); el \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} se almacena con tres veces su masa de
        agua.
  \item Comprueba los órdenes de magnitud: un día entero son unos
        \qty{10000}{kJ}; una hora dura, \qtyrange{2000}{3000}{kJ}; un
        maratón, \qty{12000}{kJ}.
\end{enumerate}
\end{method}

\begin{example}[Una hora de bicicleta]\label{ex:g10:exercise-and-energy:hour}
Una hora a \qty{150}{W} en los pedales: potencia metabólica de unos
\qty{600}{W}, energía $600 \times 3600 = \qty{2.16e6}{J} \approx
\qty{2200}{kJ}$; oxígeno $2200/20 = \qty{110}{L}$, es decir
\qty{1.8}{L/min} --- coherente con la curva. Si la mitad viene de los
\omterm{def:g10:chemistry-of-life:families}{glúcidos}: $1100/17 \approx \qty{65}{g}$ de glucosa o de \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno}, y
$1100/38 \approx \qty{29}{g}$ de grasa.
\end{example}

\begin{remark}[Por qué el oxígeno tiene que llegar]\label{rem:g10:exercise-and-energy:delivery}
Todo lo anterior supone que el oxígeno llega a las \omterm{def:g10:cells-common-unit:organelle}{mitocondrias} tan
deprisa como estas pueden usarlo. De eso se encargan los pulmones, el
corazón y la sangre, cuya respuesta al ejercicio --- respirar más
deprisa, latidos más rápidos y más fuertes, sangre desviada a los
músculos --- es el tema del \cref{ch:g10:heart-lungs-effort}. El
$\dot V\!\mathrm{O_2}$máx es sobre todo un límite de entrega, no del
apetito de los músculos.
\end{remark}

\section{Problema: Dónde está el muro}

\begin{problem}[{Problema de fin de semana --- el presupuesto
energético de un maratón: el oxígeno que respira la corredora, el
glucógeno que lleva, el kilómetro en que se agota y qué cambia el
entrenamiento}]
\label{pb:g10:exercise-and-energy:1}
Nadia, de \qty{60}{kg}, corre un maratón (\qty{42.2}{km}) en
\qty{3}{h}\,\qty{30}{min}. Correr cuesta \qty{4.0}{kJ} por kilogramo y
por kilómetro. Sus músculos guardan \qty{300}{g} de \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} y su
hígado \qty{80}{g}; el \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} y la glucosa dan \qty{17}{kJ/g}, y la
grasa \qty{38}{kJ/g}; un litro de oxígeno libera \qty{20}{kJ}. Su
$\dot V\!\mathrm{O_2}$máx es de \qty{52}{mL/min} por kilogramo.

\medskip
\noindent\textbf{Parte I --- El coste de la carrera.}
\begin{enumerate}
  \item Calcula el coste energético del maratón para Nadia.
  \item Calcula su potencia metabólica media durante la carrera, en
        vatios.
  \item Calcula el volumen de oxígeno que consume en la carrera, y
        después su consumo medio de oxígeno en litros por minuto.
  \item Expresa ese consumo en mililitros por minuto y por kilogramo, y
        como porcentaje de su $\dot V\!\mathrm{O_2}$máx.
  \item Su consumo en reposo es de \qty{0.2}{L/min}. ¿Qué fracción de
        su consumo del día de carrera corresponde a correr?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Las reservas.}
\begin{enumerate}[resume]
  \item Calcula la energía que guardan sus reservas de \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno}.
  \item Si quemara solo \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno}, ¿en qué kilómetro se agotarían las
        reservas?
  \item En realidad quema una mezcla: a su ritmo de carrera, un 80\% de
        \omterm{def:g10:chemistry-of-life:families}{glúcidos} y un 20\% de grasa. ¿En qué kilómetro se agota ahora
        el \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno}?
  \item ¿Qué masa de grasa quema en la carrera? Compárala con los
        \qty{9}{kg} de grasa que suele llevar una mujer de \qty{60}{kg}.
  \item Explica, con la \cref{prop:g10:exercise-and-energy:fuels}, por
        qué no puede simplemente quemar solo grasa una vez agotado el
        \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} y mantener el mismo ritmo.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Vencer al muro.}
\begin{enumerate}[resume]
  \item Durante la carrera bebe una disolución azucarada que aporta
        \qty{60}{g} de \omterm{def:g10:chemistry-of-life:families}{glúcidos} por hora. ¿Cuánta energía añade eso a
        lo largo de la carrera, y cuántos kilómetros de consumo de
        \omterm{def:g10:chemistry-of-life:families}{glúcidos} cubre?
  \item Rehaz la pregunta 8 con la bebida. ¿Termina ahora antes de que
        se agote el \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno}?
  \item Tres días antes de la carrera hace una ``carga'' de \omterm{def:g10:chemistry-of-life:families}{glúcidos} y
        sube el \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} de sus músculos a \qty{500}{g}. Cada gramo se
        almacena con \qty{3}{g} de agua. ¿Cuánto sube su masa y cuánto
        cuesta eso en energía por kilómetro?
  \item El entrenamiento desplaza su mezcla a ritmo de carrera a un
        55\% de \omterm{def:g10:chemistry-of-life:families}{glúcidos} y un 45\% de grasa. Con la carga y la bebida,
        ¿en qué kilómetro se agotaría el \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno}? Coméntalo.
  \item ¿Por qué el cerebro, que solo usa glucosa, hace que el
        agotamiento del \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} se sienta como extenuación aunque a
        los músculos les quede grasa que quemar?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Las cifras de un campeón.}
Un corredor de élite de \qty{55}{kg} corre el maratón en
\qty{2}{h}\, \qty{05}{min}, con un coste de carrera de \qty{3.6}{kJ}
por kilogramo y por kilómetro y un $\dot V\!\mathrm{O_2}$máx de
\qty{80}{mL/min} por kilogramo.
\begin{enumerate}[resume]
  \item Calcula el coste energético de la carrera para el campeón y su
        consumo medio de oxígeno en litros por minuto.
  \item ¿Qué porcentaje de su $\dot V\!\mathrm{O_2}$máx sostiene?
        Compáralo con la cifra de Nadia.
  \item Dos cosas distinguen al campeón: un techo más alto y un coste
        por kilómetro más bajo. ¿Cuál de las dos es la
        ``economía''? Calcula cuánto cambia la energía de la carrera,
        para un corredor de \qty{55}{kg}, solo la diferencia de coste.
  \item Sus músculos guardan \qty{500}{g} de \omterm{def:g10:exercise-and-energy:glycogen}{glucógeno} tras la carga.
        Con un consumo del 70\% de \omterm{def:g10:chemistry-of-life:families}{glúcidos}, ¿llega a meta con él sin
        beber?
  \item Enuncia el resultado: para Nadia, el kilómetro en el que está
        el muro sin preparación, y las tres medidas que lo llevan más
        allá de la línea de meta.
\end{enumerate}
\end{problem}
