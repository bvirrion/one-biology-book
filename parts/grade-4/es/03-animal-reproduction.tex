\chapter{Cómo se reproducen los animales}\label{ch:g4:animal-reproduction}

En mayo la charca es un escándalo de cantos de rana; en junio está
negra de \omterm{ex:g2:animals-grow-up:frog}{renacuajos}. El canto, resulta, era el principio de la
historia: la reproducción --- cómo hacen los \omterm{def:g1:animals-around-us:animal}{animales} la generación
siguiente --- sigue reglas que valen desde las ranas hasta los
elefantes, con variaciones maravillosas en los detalles.

\section{Hacen falta dos}

\begin{definition}[Reproducción sexual]\label{def:g4:animal-reproduction:sexual}
En los \omterm{def:g1:animals-around-us:animal}{animales}, hacer crías exige normalmente dos progenitores: una
\emph{hembra}\index{hembra} y un \emph{macho}\index{macho}. La hembra
produce \emph{óvulos}\index{óvulo}; el macho produce
\emph{espermatozoides}\index{espermatozoide}. Un \omterm{def:g1:animals-around-us:animal}{animal} nuevo empieza
cuando un espermatozoide se une a un óvulo --- esa unión es la
\emph{fecundación}\index{fecundación}. A la reproducción con dos
progenitores se la llama \emph{reproducción
sexual}\index{reproducción sexual}.
\end{definition}

\begin{example}[Por qué cantan las ranas]\label{ex:g4:animal-reproduction:song}
El coro de mayo son los \omterm{def:g4:animal-reproduction:sexual}{machos} llamando a las \omterm{def:g4:animal-reproduction:sexual}{hembras} a la charca.
Después, cada pareja suelta sus células en el agua --- los \omterm{def:g4:animal-reproduction:sexual}{óvulos} de
la \omterm{def:g4:animal-reproduction:sexual}{hembra}, los \omterm{def:g4:animal-reproduction:sexual}{espermatozoides} del \omterm{def:g4:animal-reproduction:sexual}{macho} ---, donde ocurre la
\omterm{def:g4:animal-reproduction:sexual}{fecundación}. Los puntos negros de la gelatina, unos días más tarde,
son las crías que ya están creciendo: el canto era el paso uno del
ciclo de la rana del \cref{ch:g2:animals-grow-up}.
\end{example}

\includegraphics[width=0.56\linewidth]{images/book1/ai/g4-repro-frogspawn.jpg}

{\small Unos días después del coro: puesta de rana entre los juncos,
cada esfera de gelatina un \omterm{def:g2:animals-grow-up:born}{huevo} fecundado con una ranita empezando
dentro.}
\end{omfigure}

\begin{remark}[Dos progenitores, suerte mezclada]\label{rem:g4:animal-reproduction:mix}
Una cría recibe algo de cada progenitor --- por eso un potro puede
tener el color de su madre y la mancha blanca de su padre. Qué entrega
exactamente cada progenitor, y cómo, es una de las grandes preguntas
que este libro tiene por delante; por ahora quédate con que todo \omterm{def:g1:animals-around-us:animal}{animal}
nacido de \omterm{def:g4:animal-reproduction:sexual}{reproducción sexual} tiene dos progenitores y se parece a los
dos.
\end{remark}

\section{Huevos puestos o crías nacidas vivas}

\begin{definition}[Ovíparo y vivíparo]\label{def:g4:animal-reproduction:ovivivi}
Las dos maneras de nacer de la \cref{def:g2:animals-grow-up:born}
reciben ahora sus nombres científicos:
\begin{itemize}
  \item un \omterm{def:g1:animals-around-us:animal}{animal} es \emph{ovíparo}\index{ovíparo} si la \omterm{def:g4:animal-reproduction:sexual}{hembra} pone
        \emph{\omterm{def:g2:animals-grow-up:born}{huevos}} y las crías terminan de crecer dentro del \omterm{def:g2:animals-grow-up:born}{huevo},
        fuera de su cuerpo: \omterm{prop:g3:sorting-living-things:groups}{aves}, casi todos los \omterm{prop:g3:sorting-living-things:groups}{peces}, casi todos los
        \omterm{prop:g4:classifying-animals:classes}{reptiles}, \omterm{prop:g3:sorting-living-things:groups}{anfibios}, \omterm{prop:g3:sorting-living-things:groups}{insectos}, caracoles;
  \item un \omterm{def:g1:animals-around-us:animal}{animal} es \emph{vivíparo}\index{vivíparo} si la cría crece
        \emph{dentro del cuerpo de la madre}, alimentada por ella, y
        \emph{nace viva}: casi todos los \omterm{prop:g3:sorting-living-things:groups}{mamíferos}.
\end{itemize}
\end{definition}

\begin{example}[Dos guarderías]\label{ex:g4:animal-reproduction:nurseries}
El pollito de la gallina pasa tres semanas calentito dentro de un
\omterm{def:g2:animals-grow-up:born}{huevo} con cáscara, viviendo de la yema
(\cref{rem:g2:animals-grow-up:egg}); el potro pasa once meses dentro
de la yegua, alimentado sin parar por su cuerpo, y nace listo para
ponerse de pie en menos de una hora. El mismo objetivo, dos
guarderías: una fiambrera dentro de una cáscara y un servicio de
habitaciones por dentro.
\end{example}

\begin{proposition}[Dónde ocurre la fecundación]\label{prop:g4:animal-reproduction:where}
En los que viven en el agua, como los \omterm{prop:g3:sorting-living-things:groups}{peces} y las ranas, la
\omterm{def:g4:animal-reproduction:sexual}{fecundación} ocurre normalmente \emph{en el agua}, fuera del cuerpo de
la \omterm{def:g4:animal-reproduction:sexual}{hembra} --- los \omterm{def:g4:animal-reproduction:sexual}{óvulos} y los \omterm{def:g4:animal-reproduction:sexual}{espermatozoides} se sueltan juntos y se
encuentran a miles. En los \omterm{def:g1:animals-around-us:animal}{animales} terrestres --- \omterm{prop:g3:sorting-living-things:groups}{aves}, \omterm{prop:g4:classifying-animals:classes}{reptiles},
\omterm{prop:g3:sorting-living-things:groups}{mamíferos}, \omterm{prop:g3:sorting-living-things:groups}{insectos} --- las células se secarían al aire libre, así que
el \omterm{def:g4:animal-reproduction:sexual}{macho} coloca sus \omterm{def:g4:animal-reproduction:sexual}{espermatozoides} \emph{dentro del cuerpo de la
\omterm{def:g4:animal-reproduction:sexual}{hembra}} durante el \emph{apareamiento}\index{apareamiento}, y la
\omterm{def:g4:animal-reproduction:sexual}{fecundación} ocurre allí.
\end{proposition}
\admitted

\section{Pocas crías y muchos cuidados, o muchas y ninguno}

\begin{proposition}[Dos estrategias para la generación siguiente]\label{prop:g4:animal-reproduction:strategies}
Compara las cifras: un bacalao suelta \emph{millones} de \omterm{def:g2:animals-grow-up:born}{huevos} al año
y no cuida ninguno; una elefanta tiene \emph{una} cría cada varios
años y la vigila durante una década. Entre esos extremos vale una
regla en todo el mundo \omterm{def:g1:animals-around-us:animal}{animal}: cuantas más crías produce una especie,
menos cuidados recibe cada una --- y cuantas menos produce, más
protegida y alimentada está cada una. Las dos estrategias funcionan:
por término medio, sobreviven bastantes crías para sustituir a los
progenitores.
\end{proposition}
\admitted

\begin{example}[Contar la charca]\label{ex:g4:animal-reproduction:pondcount}
La puesta de una pareja de ranas tiene miles de \omterm{def:g2:animals-grow-up:born}{huevos}. Las garzas,
los espinosos y las larvas de libélula (véase el
\cref{ex:g3:food-chains:three}) se comen \omterm{ex:g2:animals-grow-up:frog}{renacuajos} toda la primavera
--- y de los miles, un puñado de ranitas trepa a la orilla en julio.
La puesta enorme no es un desperdicio: es la respuesta de la rana a los
peligros de la charca, sin ninguna guardia paterna.
\end{example}

\begin{example}[El otro extremo]\label{ex:g4:animal-reproduction:whalecalf}
La cría de la ballena nace sola, nada enseguida junto a su madre, bebe
su \omterm{def:g2:animals-grow-up:born}{leche} riquísima durante muchos meses y está defendida contra todos
los peligros. Una sola cría cada vez --- pero esa lo recibe todo. Los
\omterm{prop:g3:sorting-living-things:groups}{mamíferos}, con su \omterm{def:g2:animals-grow-up:born}{leche} (\cref{def:g2:animals-grow-up:born}), son los
grandes especialistas de la estrategia de pocas crías muy cuidadas.
\end{example}

\begin{method}[Leer la estrategia de una especie]\label{met:g4:animal-reproduction:read}
Dada una especie, pregunta:
\begin{enumerate}
  \item ¿cuántas crías cada vez --- una, una docena, miles?
  \item ¿quién las vigila y las alimenta --- los dos progenitores, la
        madre, nadie?
  \item ¿dónde empiezan --- en un \omterm{def:g2:animals-grow-up:born}{huevo} fuera o creciendo dentro?
  \item comprueba el equilibrio de la
        \cref{prop:g4:animal-reproduction:strategies}: muchas crías
        con pocos cuidados, o pocas con muchos.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:1}
¿Qué es la \omterm{def:g4:animal-reproduction:sexual}{fecundación}? ¿Qué célula viene de cada progenitor?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:2}
Define \omterm{def:g4:animal-reproduction:ovivivi}{ovíparo} y \omterm{def:g4:animal-reproduction:ovivivi}{vivíparo}, con dos \omterm{def:g1:animals-around-us:animal}{animales} de cada clase.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:3}
¿Por qué cantaban las ranas en mayo? Vuelve a contar la historia de la
charca en tres pasos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:4}
¿Dónde ocurre la \omterm{def:g4:animal-reproduction:sexual}{fecundación} en la trucha y dónde en la gallina? ¿Por
qué la diferencia, según la \cref{prop:g4:animal-reproduction:where}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:5}
Compara las guarderías del pollito y del potro: ¿dónde crece cada uno
y qué lo alimenta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:6}
Enuncia la regla que relaciona el número de crías con los cuidados que
reciben. Da los dos ejemplos extremos del capítulo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:7}
Los millones de \omterm{def:g2:animals-grow-up:born}{huevos} del bacalao casi nunca llegan a ser bacalaos
adultos. ¿Por qué funciona la estrategia de todas formas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:animal-reproduction:8}
Pásale el \cref{met:g4:animal-reproduction:read} a la gata: camada
típica de unas cinco crías, amamantadas y vigiladas por la madre
durante semanas, crecidas dentro de ella. ¿A qué lado de las
estrategias está más cerca?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:animal-reproduction:9}
Un potro se parece a menudo a sus dos progenitores a la vez. ¿Qué
demuestra eso sobre lo que recibe la cría, según la
\cref{rem:g4:animal-reproduction:mix}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:animal-reproduction:10}
El \omterm{def:g4:animal-reproduction:sexual}{macho} del espinoso construye un nido, abanica los \omterm{def:g2:animals-grow-up:born}{huevos} y espanta
a los intrusos --- una devoción rara en un \omterm{prop:g3:sorting-living-things:groups}{pez}. Con la
\cref{prop:g4:animal-reproduction:strategies}, ¿qué predecirías sobre
el tamaño de su puesta, comparada con los millones del bacalao?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g4:animal-reproduction:11}
Las tortugas marinas salen a la arena a enterrar sus \omterm{def:g2:animals-grow-up:born}{huevos} en la
arena caliente y vuelven al mar. ¿Ovíparas o vivíparas? ¿Con cuidados
o sin ellos? ¿Y por qué tiene que poner en \emph{tierra} este reptil,
que se pasa nadando toda la vida? (Usa los rasgos de los \omterm{prop:g4:classifying-animals:classes}{reptiles} de
la \cref{prop:g4:classifying-animals:classes}.)
\end{exercise}
